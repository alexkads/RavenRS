Expand description

SwiftUI-style modifiers available on every view.

use ravenrs_style::Semantic;
use ravenrs_widgets::{Text, ViewExt};

let chip = Text::new("New").padding((4.0, 10.0)).background(Semantic::Control).corner_radius(999.0);
