Struct Sidebar

pub struct Sidebar<C, P = ()> { /* private fields */ }
Expand description

A navigation column: Sidebar::new((SidebarSection::new("Workspace"), SidebarItem::new(...), ...)).

Its children scroll between an optional header and footer. With Self::collapsible it closes to icons or off the screen, by Cmd/Ctrl+B or a SidebarTrigger. With Self::narrow_below small windows show it as a sheet over the page.

use ravenrs_widgets::*;

let state = SidebarState::new(true);
let sidebar = Sidebar::new((
    SidebarGroup::new("Workspace", (
        SidebarItem::new(IconName::House, "Home").selected(true),
        SidebarItem::new(IconName::Users, "People").badge(Badge::new("3")),
    )),
))
.collapsible(SidebarCollapse::Icons)
.narrow_below(768.0)
.state(state.clone());
let trigger = SidebarTrigger::new(state);
