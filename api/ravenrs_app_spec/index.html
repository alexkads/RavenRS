Source

What #[derive(BusinessObject)] and #[derive(BusinessEnum)] read from a struct or an enum, as plain data: the derive turns it into code, and ravenrs-app (feature source) turns it into entities while a program runs, so both accept and refuse the same things, with the same words.

let file = ravenrs_app_spec::parse_file(
    r#"
    #[derive(BusinessObject)]
    #[object(caption = "Banks")]
    struct Bank {
        #[field(typed_key, caption = "Code")]
        code: String,
    }
    "#,
)
.map_err(|errors| errors[0].message.clone())?;
assert_eq!(file.objects[0].name.to_string(), "Bank");
assert!(file.objects[0].fields[0].settings.typed_key);
