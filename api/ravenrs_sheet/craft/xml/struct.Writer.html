ravenrs_sheet::craft::xml

Struct Writer 

Source 
pub struct Writer { /* private fields */ }
Expand description

Builds XML text without a tree. Element names are closed for the caller, attribute values and text are escaped, and with Writer::pretty each element goes on its own line.

Names are written as given: they are the caller’s own constants, not checked.

let mut w = ravenrs_sheet::craft::xml::Writer::document();
w.tag("w:p").attr("w:id", 7).open();
w.leaf("w:t", &[("xml:space", "preserve")], " a < b ");
w.empty("w:br", &[]);
assert_eq!(
    w.finish(),
    "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<w:p w:id=\"7\"><w:t xml:space=\"preserve\"> a &lt; b </w:t><w:br/></w:p>"
);
