Struct App

pub struct App { /* private fields */ }
Expand description

An application from its records: the module with the actions every application needs, the screens, and records in memory unless told otherwise.

use ravenrs_app::{BusinessObject, ObjectSpaceExt};
use ravenrs_app_ui::{App, Language};

#[derive(BusinessObject)]
#[object(caption = "Contas", navigation = "Cadastros")]
struct Account {
    #[field(key)]
    id: i64,
    #[field(caption = "Nome")]
    name: String,
}

let window = App::new("Raven Contas")
    .language(Language::BrazilianPortuguese)
    .entity::<Account>()
    .seed(|space| {
        space.add(&Account { id: 0, name: "Carteira".into() }).map_err(|e| e.to_string())
    })
    .build()?;
