1//! A BPMN 2.0 editor made only of RavenRS components: `FlowLibrary`, `FlowToolbox`,
2//! `FlowCanvas` and `FlowInspector` take BPMN from `ravenrs_flow_bpmn::notation()`, `Chat`
3//! talks to a model running on this machine through `ravenrs-ai`, and `ModelManager` gets one.
4//! Nothing on the screen is drawn here: the app only joins the parts.
5//!
6//! - The palette has the elements of BPMN; connections take the right kind as they are made.
7//! - Check, in the canvas's controls, says what is wrong and outlines where.
8//! - Processes are kept by name, and open and save as BPMN 2.0 XML with their drawing, which
9//!   other BPMN tools read.
10//! - The assistant drafts a process from a description, or reviews the one open.
11

21/// What asks for a review instead of a draft.
22pub const REVIEW: &str = "Revisar este processo";
23
24/// The editor.
25pub struct Editor {
26    storage: Rc<dyn Storage>,
27    /// The local models, on the desktop; the browser runs none.
28    #[cfg(not(target_arch = "wasm32"))]
29    models: Option<ModelManager>,
30}
31
32impl std::fmt::Debug for Editor {
33    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
34        f.debug_struct("Editor").finish_non_exhaustive()
35    }
36}

38/// The editor with its processes kept only while it runs, and no model.
39pub fn app() -> Localized<Editor> {
40    app_kept_in(Rc::new(MemoryStorage::default()), None)
41}
42
43/// The editor keeping its processes in `storage`, and its local models in `models` (on the
44/// desktop; in the browser there is no local model).
45pub fn app_kept_in(storage: Rc<dyn Storage>, models: Option<PathBuf>) -> Localized<Editor> {
46    #[cfg(not(target_arch = "wasm32"))]
47    let editor = Editor {
48        storage,
49        models: models.map(|folder| {
50            ModelManager::new(CATALOG, ModelFolder::new(folder)).labels(ModelManagerLabels::pt_br())
51        }),
52    };
53    #[cfg(target_arch = "wasm32")]
54    let editor = {
55        let _ = models;
56        Editor { storage }
57    };
58    Localized::new(Labels::pt_br(), editor)
59}

61/// What the assistant does with what the person writes, and the part that picks the model.
62type Assistant = (Rc<dyn Fn(String)>, AnyView);
63
64impl Editor {
65    /// In the browser: no local model runs there, and the chat says so.
66    #[cfg(target_arch = "wasm32")]
67    fn assistant(
68        &self,
69        cx: &mut BuildCx<'_>,
70        said: State<Vec<ChatMessage>>,
71        _graph: State<FlowGraph>,
72        _history: State<FlowHistory>,
73        _view: State<FlowViewport>,
74    ) -> Assistant {
75        let _ = cx;
76        let reply: Rc<dyn Fn(String)> = Rc::new(move |_| {
77            said.update(|m| {
78                m.push(ChatMessage::assistant(
79                    "A IA local roda no aplicativo para desktop; no navegador ela não está disponível.",
80                ))
81            });
82        });
83        (reply, Box::new(Alert::message("No navegador, a IA local fica desligada.")))
84    }
85
86    /// On the desktop: a description becomes a process (undone with Cmd/Ctrl+Z), and the review
87    /// reads the one open. Without a model, it says how to get one.
88    #[cfg(not(target_arch = "wasm32"))]
89    fn assistant(
90        &self,
91        cx: &mut BuildCx<'_>,
92        messages: State<Vec<ChatMessage>>,
93        graph: State<FlowGraph>,
94        history: State<FlowHistory>,
95        view: State<FlowViewport>,
96    ) -> Assistant {
97        let assistant = self.models.as_ref().and_then(ModelManager::assistant);
98        let (said, g, h, v, waker) =
99            (messages.clone(), graph.clone(), history.clone(), view.clone(), cx.ui_waker());
100        let on_send: Rc<dyn Fn(String)> = Rc::new(move |text: String| {
101            let Some(model) = &assistant else {
102                said.update(|m| m.push(ChatMessage::assistant("Escolha um modelo em Modelos de IA local. Ele roda neste computador e nada sai daqui.")));
103                return;
104            };
105            let reply = if text == REVIEW {
106                model.reply(plan::REVIEW_RULES.to_owned(), plan::summary(&g.get()), waker.clone(), |_| {})
107            } else {
108                let (told, g, h, v) = (said.clone(), g.clone(), h.clone(), v.clone());
109                model.reply(plan::DRAFT_RULES.to_owned(), text, waker.clone(), move |answer| {
110                    let words = match plan::plan(&answer) {
111                        Ok(drafted) => {
112                            let before = g.get();
113                            h.update(|history| history.record(before));
114                            let steps = drafted.graph.nodes.len();
115                            g.set(drafted.graph);
116                            v.set(FlowViewport::fit());
117                            let mut words = format!(
118                                "Processo montado com {steps} elementos. Cmd/Ctrl+Z volta ao anterior."
119                            );
120                            for note in drafted.mended {
121                                words.push_str(&format!("\n• {note}"));
122                            }
123                            words
124                        }
125                        Err(why) => format!("Não consegui montar um processo com essa resposta: {why}"),
126                    };
127                    told.update(|m| m.push(ChatMessage::assistant(words)));
128                })
129            };
130            said.update(|m| m.push(reply));
131        });
132        let models: AnyView = match &self.models {
133            Some(models) => Box::new(Disclosure::new("Modelos de IA local", models.clone())),
134            None => Box::new(Alert::message("Sem pasta de modelos: a IA local fica desligada.")),
135        };
136
137        (on_send, models)
138    }
139}

141impl View for Editor {
142    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
143        let bpmn = ravenrs_flow_bpmn::notation();
144        let graph = cx.use_state(ravenrs_flow_bpmn::example_process);
145        let selection = cx.use_state(FlowSelection::default);
146        let history = cx.use_state(FlowHistory::new);
147        let view = cx.use_state(FlowViewport::fit);
148        let arrivals = cx.use_state(|| None::<FlowNode>);
149        let messages = cx.use_state(Vec::<ChatMessage>::new);
150
151        let (on_send, models) =
152            self.assistant(cx, messages.clone(), graph.clone(), history.clone(), view.clone());
153        let chat = Chat::new(messages.binding())
154            .intro("Descreva um processo, com quem faz o quê e quando decide, e eu monto o diagrama. A IA roda neste computador.")
155            .placeholder("Descreva o processo")
156            .examples([REVIEW])
157            .on_send(move |text| on_send(text));
158        cx.child(
159            &HStack::new((
160                VStack::new((
161                    FlowLibrary::new(graph.binding())
162                        .title("Processos")
163                        .key("processes")
164                        .notation(&bpmn)
165                        .starter("Compra aprovada", ravenrs_flow_bpmn::example_process())
166                        .starter("Pedido entre cliente e loja", ravenrs_flow_bpmn::example_collaboration())
167                        .history(history.binding())
168                        .selection(selection.binding())
169                        .view(view.binding())
170                        .storage(self.storage.clone()),
171                    FlowToolbox::new().title("Elementos BPMN").notation(&bpmn).arrivals(arrivals.binding()),
172                ))
173                .spacing(0.0)
174                .align(Align::Stretch),
175                FlowCanvas::new(graph.binding())
176                    .label("Processo")
177                    .notation(&bpmn)
178                    .view(view.binding())
179                    .selection(selection.binding())
180                    .history(history.binding())
181                    .arrivals(arrivals.binding())
182                    .on_export(ravenrs_flow_export::saver(cx.file_picker(), "Processo")),
183                VStack::new((
184                    FlowInspector::new(graph.binding(), selection.binding())
185                        .notation(&bpmn)
186                        .history(history.binding()),
187                    VStack::new((chat,)).grow(1.0).align(Align::Stretch),
188                    models,
189                ))
190                .width(320.0)
191                .spacing(8.0)
192                .padding(8.0)
193                .align(Align::Stretch),
194            ))
195            .align(Align::Stretch)
196            .grow(1.0),
197        )
198    }
199}

201#[cfg(target_arch = "wasm32")]
202mod web_entry {
203    use std::cell::RefCell;
204
205    use wasm_bindgen::prelude::*;
206
207    thread_local! {
208        static APP: RefCell<Option<ravenrs::web::WebApp>> = const { RefCell::new(None) };
209    }
210
211    /// Runs when the module is instantiated: the diagrams are kept in the browser's storage.
212    #[wasm_bindgen(start)]
213    pub fn start() {
214        console_error_panic_hook::set_once();
215        wasm_bindgen_futures::spawn_local(async {
216            let storage = match ravenrs::web::IndexedDbStorage::open("ravenrs-bpmn-editor").await {
217                Ok(storage) => storage,
218                Err(e) => {
219                    web_sys::console::error_1(&e.to_string().into());
220                    return;
221                }
222            };
223            let app = super::app_kept_in(std::rc::Rc::new(storage), None);
224            match ravenrs::web::start(app, ravenrs::web::WebOptions::new("raven-canvas")).await {
225                Ok(app) => APP.with(|a| *a.borrow_mut() = Some(app)),
226                Err(e) => web_sys::console::error_1(&e.to_string().into()),
227            }
228        });
229    }
