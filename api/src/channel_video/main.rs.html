1//! The videos of the project's channel, recorded from the real apps.
2//!
3//! ```text
4//! cargo run --release -p channel-video -- record OUT
5//! FF_AI_MODELS=~/.raven-financas/modelos cargo run --release -p channel-video -- narrate OUT --voz pm_santa
6//! cargo run --release -p channel-video -- export OUT
7//! ```
8//!
9//! `record` writes the films, `narrate` records what their boxes say with the local
10//! Kokoro voice (`--voz` chooses `pf_dora`, `pm_alex` or `pm_santa`, the default), and `export` writes the wide and the vertical video, the subtitles,
11//! the chapters for the description and a cover. `frames OUT 0:6000` draws
12//! single moments (film:ms) as pictures, to look at before exporting.
13#![allow(clippy::expect_used, clippy::print_stdout, clippy::print_stderr, reason = "a tool run by hand")]

133/// The opening card, fades and the halo around the pointer.
134fn edit(film: Movie, title: &str, subtitle: &str) -> Movie {
135    let mut project = film.project().clone();
136    let total = project.duration_ms();
137    project.add_overlay(OverlayKind::FadeOut, total.saturating_sub(800), 800);
138    project.add_overlay(OverlayKind::PointerHighlight, 0, total);
139    if !title.is_empty() {
140        project.insert_title(0, title, subtitle, 2500);
141    }
142    project.add_overlay(OverlayKind::FadeIn, 0, 600);
143    film.with_project(project)
144}

146fn record(out: &Path) {
147    let mut take = Recorder::new(component_gallery::Gallery::new(), Theme::raven_light(), SIZE);
148    gallery(&mut take).unwrap_or_else(|e| panic!("the gallery script no longer fits: {e}"));
149    let first = edit(take.finish().expect("film"), "RavenRS", "Interfaces em Rust, desenhadas na GPU");
150    let today = Date::new(2026, 11, 15).expect("valid date");
151    let db = Database::in_memory().expect("database");
152    seed::load_demo(&db, today).expect("seed");
153    let mut take = Recorder::new(App::new(db, today, None, update::idle()), themes::light(), SIZE);
154    finance(&mut take).unwrap_or_else(|e| panic!("the finance script no longer fits: {e}"));
155    let second = edit(take.finish().expect("film"), "", "");
156    for (name, film) in [("1-componentes", &first), ("2-financas", &second)] {
157        std::fs::write(out.join(format!("{name}.ravenmovie")), film.to_bytes()).expect("film");
158        println!("{name}: {:.1}s, {} boxes", film.duration_ms() as f32 / 1000.0, film.hotspots().len());
159    }
160}

175/// The Kokoro voice chosen with `--voz NAME`, among [`ravenrs_ai::KOKORO_SPEAKERS`].
176fn chosen_speaker(args: &[String]) -> Result<String, String> {
177    let speaker = match args.iter().position(|arg| arg == "--voz") {
178        Some(at) => args.get(at + 1).ok_or("--voz pede o nome da voz")?.clone(),
179        None => DEFAULT_SPEAKER.to_owned(),
180    };
181    if ravenrs_ai::KOKORO_SPEAKERS.contains(&speaker.as_str()) {
182        Ok(speaker)
183    } else {
184        Err(format!("voz {speaker} desconhecida; use uma de {}", ravenrs_ai::KOKORO_SPEAKERS.join(", ")))
185    }
186}

188/// What a box says, as the voice says it: short sentences together, a quarter second after each.
189fn spoken_sound(
190    engine: &mut ravenrs_ai::SpeechEngine,
191    settings: &ravenrs_ai::SpeechSettings,
192    text: &str,
193) -> Vec<f32> {
194    let silence = vec![0.0_f32; (ravenrs_ai::SPEECH_RATE / 4) as usize];
195    let mut sound = Vec::new();
196    // Short sentences go together: Kokoro says a few words alone with a flat tune.
197    for piece in engine.pieces(&family_finance::ui::spoken(text)) {
198        sound.extend(engine.speak(&piece, settings, &mut |_| true).expect("sound"));
199        sound.extend_from_slice(&silence);
200    }
201    sound
202}

204fn narrate(out: &Path, speaker: String) {
205    use ravenrs_ai::{ModelFolder, SPEECH_RATE, SpeechEngine, SpeechSettings, model_by_id};
206    let mut said: Vec<String> = films(out)
207        .iter()
208        .flat_map(|film| {
209            film.hotspots().iter().map(|s| format!("{}. {}", s.title, s.text)).collect::<Vec<_>>()
210        })
211        .collect();
212    said.dedup();
213    let root = std::env::var("FF_AI_MODELS").expect("FF_AI_MODELS: the folder of the models");
214    let model = model_by_id("kokoro-82m").expect("known");
215    let mut engine = SpeechEngine::load(&ModelFolder::new(root), &model).expect("the Kokoro voice");
216    println!("voz: {speaker}");
217    let settings = SpeechSettings { speaker: Some(speaker), ..SpeechSettings::default() };
218    let wavs = out.join("falas");
219    std::fs::create_dir_all(&wavs).expect("folder");
220    let mut recordings = Vec::new();
221    for (index, text) in said.iter().enumerate() {
222        let sound = spoken_sound(&mut engine, &settings, text);
223        let allowed = 0.8 + text.chars().count() as f32 * 0.075;
224        println!("{:.1}s of {allowed:.1}s | {text}", sound.len() as f32 / SPEECH_RATE as f32);
225        std::fs::write(wavs.join(format!("{index:02}.wav")), wav_of(&sound, SPEECH_RATE)).expect("wav");
226        std::fs::write(wavs.join(format!("{index:02}.txt")), text).expect("text");
227        recordings.push((text.clone(), sound));
228    }
229    std::fs::write(out.join("narracao.voz"), Narration::pack(&recordings, SPEECH_RATE)).expect("narration");
230}

232fn wav_of(sound: &[f32], rate: u32) -> Vec<u8> {
233    let mut wav = Vec::with_capacity(44 + sound.len() * 2);
234    wav.extend_from_slice(b"RIFF");
235    wav.extend_from_slice(&(36 + sound.len() as u32 * 2).to_le_bytes());
236    wav.extend_from_slice(b"WAVEfmt ");
237    wav.extend_from_slice(&16_u32.to_le_bytes());
238    for value in [1_u16, 1] {
239        wav.extend_from_slice(&value.to_le_bytes());
240    }
241    wav.extend_from_slice(&rate.to_le_bytes());
242    wav.extend_from_slice(&(rate * 2).to_le_bytes());
243    for value in [2_u16, 16] {
244        wav.extend_from_slice(&value.to_le_bytes());
245    }
246    wav.extend_from_slice(b"data");
247    wav.extend_from_slice(&(sound.len() as u32 * 2).to_le_bytes());
248    for sample in sound {
249        wav.extend_from_slice(&((sample.clamp(-1.0, 1.0) * 32767.0) as i16).to_le_bytes());
250    }
251    wav
252}

254fn export(out: &Path, only: Option<&str>) {
255    let films = films(out);
256    let narration = std::fs::read(out.join("narracao.voz"))
257        .ok()
258        .map(|bytes| Narration::new(Box::leak(bytes.into_boxed_slice())));
259    if narration.is_none() {
260        eprintln!("no narracao.voz: the video goes without the voice");
261    }
262    let make = |format: VideoFormat, films: &[Movie]| {
263        let mut video = Video::new(films[0].clone(), format).labels(MovieLabels::pt_br()).brand(BRAND);
264        for film in &films[1..] {
265            video = video.then(film.clone());
266        }
267        match &narration {
268            Some(narration) => video.narration(narration.clone()),
269            None => video,
270        }
271    };
272    let started = std::time::Instant::now();
273    let write = |name: &str, video: Video| {
274        let report = video
275            .write(out.join(name), |done, all| {
276                if done % 300 == 0 || done == all {
277                    eprintln!("{name}: {done}/{all}");
278                }
279            })
280            .expect("the video");
281        println!(
282            "{name}: {:.1}s, {:.1} MB, {:.0}s to make{}",
283            report.duration_ms as f32 / 1000.0,
284            report.bytes as f32 / 1e6,
285            started.elapsed().as_secs_f32(),
286            if report.too_long { ", too long for the format" } else { "" }
287        );
288        for text in report.unrecorded {
289            println!("  not said: {text}");
290        }
291    };
292    if only != Some("vertical") {
293        let wide = make(VideoFormat::WIDE, &films);
294        std::fs::write(out.join("capa.png"), wide.thumbnail(1_200).expect("cover")).expect("cover");
295        write("ravenrs-youtube.mov", wide);
296        std::fs::write(out.join("ravenrs-youtube.srt"), subtitles(&films)).expect("srt");
297        let description = format!(
298            "RavenRS é um framework de interface em Rust: componentes desenhados pela GPU, o mesmo código no Mac, no Linux, no Windows e no navegador.\n\n\
299             Neste vídeo: a galeria de componentes (painel, grade de dados com cem mil linhas, gráficos, faixa de comandos, animações e temas) e o Raven Finanças, um aplicativo completo feito só com o RavenRS.\n\n\
300             O vídeo foi gravado pelo próprio RavenRS, a partir de um roteiro, com o crate ravenrs-movie-video.\n\n\
301             Código: https://github.com/alexkads/RavenRS\n\n{}",
302            chapters(&films)
303        );
304        std::fs::write(out.join("descricao.txt"), description).expect("description");
305    }
306    if only != Some("wide") {
307        let followed: Vec<Movie> = films.to_vec();
308        write("ravenrs-vertical.mov", make(VideoFormat::VERTICAL, &followed));
309    }
310}

312/// Pictures of moments of the first film, wide and vertical, to look at.
313fn frames(out: &Path, moments: &[String]) {
314    let films = films(out);
315    for (film, at) in moments.iter().filter_map(|m| m.split_once(':')) {
316        let movie = &films[film.parse::<usize>().expect("film")];
317        let at: u64 = at.parse().expect("ms");
318        for (name, format, movie) in
319            [("wide", VideoFormat::WIDE, movie.clone()), ("vertical", VideoFormat::VERTICAL, movie.clone())]
320        {
321            let video = Video::new(movie, format).labels(MovieLabels::pt_br()).brand(BRAND);
322            std::fs::write(
323                out.join(format!("q-{film}-{at}-{name}.png")),
324                video.thumbnail(at).expect("picture"),
325            )
326            .expect("png");
327        }
328    }
329}

331fn main() {
332    let args: Vec<String> = std::env::args().collect();
333    let out = PathBuf::from(
334        args.get(2).expect("usage: channel-video record|narrate|export OUT [--voz pf_dora|pm_alex|pm_santa]"),
335    );
336    std::fs::create_dir_all(&out).expect("folder");
337    match args[1].as_str() {
338        "record" => record(&out),
339        "narrate" => match chosen_speaker(&args[3..]) {
340            Ok(speaker) => narrate(&out, speaker),
341            Err(error) => {
342                eprintln!("{error}");
343                std::process::exit(2);
344            }
345        },
346        "export" => export(&out, args.get(3).map(String::as_str)),
347        "frames" => frames(&out, &args[3..]),
348        other => panic!("unknown step {other}"),
349    }
350}

360    /// Lowercase words without accents or punctuation, to compare what was said with what was heard.
361    fn words(text: &str) -> Vec<String> {
362        let plain: String = text
363            .to_lowercase()
364            .chars()
365            .map(|c| match c {
366                'á' | 'à' | 'â' | 'ã' => 'a',
367                'é' | 'ê' => 'e',
368                'í' => 'i',
369                'ó' | 'ô' | 'õ' => 'o',
370                'ú' => 'u',
371                'ç' => 'c',
372                c if c.is_alphanumeric() => c,
373                _ => ' ',
374            })
375            .collect();
376        plain.split_whitespace().map(str::to_owned).collect()
377    }

379    /// #97: each of the three voices says a box of the films, and Whisper hears it back. Needs
380    /// `FF_AI_MODELS` with `kokoro-82m` and `whisper-base`; run it in release.
381    #[test]
382    #[ignore = "needs FF_AI_MODELS with kokoro-82m and whisper-base"]
383    fn whisper_hears_what_each_voice_says() {
384        use ravenrs_ai::{
385            HEARING_RATE, ModelFolder, SPEECH_RATE, SpeechEngine, SpeechSettings, Transcriber, model_by_id,
386        };
387        let root = std::env::var("FF_AI_MODELS").expect("FF_AI_MODELS");
388        let folder = ModelFolder::new(root);
389        let mut engine =
390            SpeechEngine::load(&folder, &model_by_id("kokoro-82m").expect("known")).expect("Kokoro");
391        let mut ears =
392            Transcriber::load(&folder, &model_by_id("whisper-base").expect("known")).expect("Whisper");
393        let text = "Lançamentos. Cada gasto entra com o valor, a categoria e quem pagou, e o saldo da conta acompanha.";
394        let expected = words(text);
395        for voice in ["pf_dora", "pm_alex", "pm_santa"] {
396            let settings = SpeechSettings { speaker: Some(voice.to_owned()), ..SpeechSettings::default() };
397            let sound = spoken_sound(&mut engine, &settings, text);
398            // Whisper hears at its own rate; the voice speaks at 24 kHz.
399            let step = SPEECH_RATE as f32 / HEARING_RATE as f32;
400            let heard_rate: Vec<f32> = (0..(sound.len() as f32 / step) as usize)
401                .map(|i| {
402                    let at = i as f32 * step;
403                    let (a, f) = (at as usize, at.fract());
404                    sound[a] * (1.0 - f) + sound.get(a + 1).copied().unwrap_or(0.0) * f
405                })
406                .collect();
407            let heard = ears.transcribe(&heard_rate, "pt").expect("heard").text;
408            let got = words(&heard);
409            let found = expected.iter().filter(|w| got.contains(w)).count();
410            println!("{voice}: {found}/{} | {heard}", expected.len());
411            assert!(found * 10 >= expected.len() * 8, "{voice} said it so that Whisper heard: {heard}");
412        }
413    }
