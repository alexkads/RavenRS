1//! Turns a report's values and .NET-style format specifiers into what a
2//! spreadsheet stores: a number and an Excel number format.
3
4use ravenrs_report::{Culture, DateTime, Value};
5
6/// Days between Excel's day zero (in its 1900 system) and 1970-01-01.
7const EXCEL_EPOCH: i64 = 25_569;
8
9/// The number to store for `value` and its Excel format, or `None` when the
10/// cell should keep the printed text.
11pub(crate) fn typed(value: &Value, spec: Option<&str>, culture: &Culture) -> Option<(f64, Option<String>)> {
12    let spec = spec.unwrap_or_default().trim();
13    match value {
14        Value::Number(n) if n.is_finite() => Some((*n, Some(number_format(spec, culture)?))),
15        Value::Date(d) => date(*d, spec, culture),
16        // Text takes number and date formats too, as the engine prints it that way.
17        Value::Text(s) if !spec.is_empty() => match DateTime::parse(s) {
18            Some(d) => date(d, spec, culture),
19            None => {
20                let n = s.trim().parse::<f64>().ok().filter(|n| n.is_finite())?;
21                Some((n, Some(number_format(spec, culture)?)))
22            }
23        },
24        _ => None,
25    }
26}
27
28fn date(d: DateTime, spec: &str, culture: &Culture) -> Option<(f64, Option<String>)> {
29    // Before March 1900, Excel's calendar disagrees with the real one.
30    let day = d.days + EXCEL_EPOCH;
31    if day < 61 {
32        return None;
33    }
34    let serial = day as f64 + f64::from(d.seconds) / 86_400.0;
35    Some((serial, Some(date_format(spec, culture)?)))
36}

38/// The Excel format for a number specifier; `None` when Excel has no equivalent.
39fn number_format(spec: &str, culture: &Culture) -> Option<String> {
40    if spec.is_empty() || spec.eq_ignore_ascii_case("g") {
41        return Some("General".to_owned());
42    }
43    let mut chars = spec.chars();
44    let kind = chars.next()?;
45    let rest = chars.as_str();
46    if rest.is_empty() || rest.parse::<usize>().is_ok() {
47        let digits = rest.parse::<usize>().ok();
48        let decimals = |default: usize| {
49            let n = digits.unwrap_or(default);
50            if n == 0 { String::new() } else { format!(".{}", "0".repeat(n)) }
51        };
52        match kind.to_ascii_uppercase() {
53            'C' => {
54                let space = if culture.currency_space { "\\ " } else { "" };
55                return Some(format!("\"{}\"{space}#,##0{}", culture.currency, decimals(2)));
56            }
57            'N' => return Some(format!("#,##0{}", decimals(2))),
58            'F' => return Some(format!("0{}", decimals(2))),
59            'P' => return Some(format!("0{}%", decimals(2))),
60            'D' => return Some("0".repeat(digits.unwrap_or(1).max(1))),
61            'X' => return None,
62            _ => {}
63        }
64    }
65    let sections: Vec<String> = spec.split(';').map(custom_number_section).collect::<Option<_>>()?;
66    Some(sections.join(";"))
67}
68
69/// A custom section such as `R$ #,##0.00`, with the literal text around the digits quoted.
70fn custom_number_section(section: &str) -> Option<String> {
71    let is_digit = |c: char| matches!(c, '0' | '#' | ',' | '.');
72    let start = section.find(is_digit)?;
73    let end = section.rfind(is_digit)? + 1;
74    let literal = |s: &str| {
75        let mut out = String::new();
76        let plain = s.replace(['\'', '"'], "");
77        let without_percent = plain.replace('%', "");
78        if !without_percent.is_empty() {
79            out = format!("\"{without_percent}\"");
80        }
81        if plain.contains('%') {
82            out.push('%');
83        }
84        out
85    };
86    Some(format!("{}{}{}", literal(&section[..start]), &section[start..end], literal(&section[end..])))
87}

89/// The Excel format for a date specifier; `None` when Excel has no equivalent.
90fn date_format(spec: &str, culture: &Culture) -> Option<String> {
91    let pattern = match spec {
92        "" | "d" => culture.short_date.to_owned(),
93        "D" => culture.long_date.to_owned(),
94        "t" => culture.short_time.to_owned(),
95        "T" => format!("{}:ss", culture.short_time),
96        "g" => format!("{} {}", culture.short_date, culture.short_time),
97        "G" => format!("{} {}:ss", culture.short_date, culture.short_time),
98        "f" => format!("{} {}", culture.long_date, culture.short_time),
99        "F" => format!("{} {}:ss", culture.long_date, culture.short_time),
100        "M" | "m" | "Y" | "y" => return None,
101        custom => custom.to_owned(),
102    };
103    // Month and day names follow the report's language, whatever the machine's is.
104    let locale = if culture.name == "pt-BR" { "[$-416]" } else { "" };
105    Some(format!("{locale}{}", translate_date(&pattern)))
106}

108/// Translates a .NET date pattern to Excel's letters.
109fn translate_date(pattern: &str) -> String {
110    let mut out = String::new();
111    let mut chars = pattern.chars().peekable();
112    while let Some(c) = chars.next() {
113        match c {
114            '\'' | '"' => {
115                let literal: String = chars.by_ref().take_while(|x| *x != c).collect();
116                out.push('"');
117                out.push_str(&literal);
118                out.push('"');
119            }
120            'M' => out.push('m'),
121            'H' => out.push('h'),
122            'y' | 'd' | 'h' | 'm' | 's' => out.push(c),
123            't' => {
124                while chars.peek() == Some(&'t') {
125                    chars.next();
126                }
127                out.push_str("AM/PM");
128            }
129            '/' | ':' | '-' | ' ' | ',' | '.' => out.push(c),
130            other => {
131                out.push('"');
132                out.push(other);
133                out.push('"');
134            }
135        }
136    }
137    out
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143
144    fn number(spec: &str) -> Option<String> {
145        number_format(spec, &Culture::pt_br())
146    }
147
148    #[test]
149    fn standard_number_formats() {
150        assert_eq!(number("C2").as_deref(), Some("\"R$\"\\ #,##0.00"));
151        assert_eq!(number("N0").as_deref(), Some("#,##0"));
152        assert_eq!(number("N").as_deref(), Some("#,##0.00"));
153        assert_eq!(number("F1").as_deref(), Some("0.0"));
154        assert_eq!(number("P1").as_deref(), Some("0.0%"));
155        assert_eq!(number("D4").as_deref(), Some("0000"));
156        assert_eq!(number("").as_deref(), Some("General"));
157        assert_eq!(number("X2"), None);
158        assert_eq!(number_format("C2", &Culture::en_us()).as_deref(), Some("\"$\"#,##0.00"));
159    }
160
161    #[test]
162    fn custom_number_formats_quote_their_literals() {
163        assert_eq!(number("#,##0.00").as_deref(), Some("#,##0.00"));
164        assert_eq!(number("R$ #,##0.00").as_deref(), Some("\"R$ \"#,##0.00"));
165        assert_eq!(number("0.0%").as_deref(), Some("0.0%"));
166        assert_eq!(number("#,##0.00;(#,##0.00)").as_deref(), Some("#,##0.00;\"(\"#,##0.00\")\""));
167        assert_eq!(number("abc"), None);
168    }
169
170    #[test]
171    fn date_formats_translate_letters_and_literals() {
172        let br = Culture::pt_br();
173        assert_eq!(date_format("dd/MM/yyyy", &br).as_deref(), Some("[$-416]dd/mm/yyyy"));
174        assert_eq!(date_format("d", &br).as_deref(), Some("[$-416]dd/mm/yyyy"));
175        assert_eq!(date_format("g", &br).as_deref(), Some("[$-416]dd/mm/yyyy hh:mm"));
176        assert_eq!(date_format("D", &br).as_deref(), Some("[$-416]dddd, d \"de\" mmmm \"de\" yyyy"));
177        assert_eq!(date_format("h:mm tt", &Culture::en_us()).as_deref(), Some("h:mm AM/PM"));
178        assert_eq!(date_format("M", &br), None);
179    }
180
181    #[test]
182    fn dates_become_excel_serial_numbers() {
183        let br = Culture::pt_br();
184        let date = DateTime::from_ymd(2026, 11, 1).map(Value::Date).unwrap_or_default();
185        // 2026-11-01 is serial 46327 in Excel.
186        assert_eq!(typed(&date, Some("dd/MM/yyyy"), &br).map(|t| t.0), Some(46_327.0));
187        let early = DateTime::from_ymd(1899, 1, 1).map(Value::Date).unwrap_or_default();
188        assert_eq!(typed(&early, None, &br), None, "before 1900 stays text");
189    }
190
191    #[test]
192    fn text_is_typed_only_with_a_format() {
193        let br = Culture::pt_br();
194        assert_eq!(typed(&Value::Text("12.5".into()), Some("N2"), &br).map(|t| t.0), Some(12.5));
195        assert_eq!(typed(&Value::Text("12.5".into()), None, &br), None);
196        assert_eq!(typed(&Value::Text("abc".into()), Some("N2"), &br), None);
197        assert_eq!(typed(&Value::Bool(true), None, &br), None);
198    }
