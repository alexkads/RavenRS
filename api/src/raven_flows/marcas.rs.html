1//! A run of the runtime as the canvas is told it: each node's part becomes a `FlowNodeRun`,
2//! with its status and how many items left by each port, and the canvas draws it with
3//! `FlowCanvas::execution`, on a copy, never touching the flow.
4
5use std::collections::BTreeMap;
6
7use ravenrs::prelude::{FlowNodeRun, FlowNodeStatus, FlowRunView};
8use ravenrs_flows::{NodeRun, NodeStatus};
9
10/// The run whose nodes did `parts`, as the canvas draws it.
11pub fn run_view(parts: &BTreeMap<String, NodeRun>) -> FlowRunView {
12    parts.iter().fold(FlowRunView::new(), |view, (id, part)| {
13        let run = part
14            .outputs
15            .iter()
16            .fold(FlowNodeRun::new(status(part)), |run, (port, items)| run.items(port.as_str(), items.len()));
17        view.node(id.as_str(), run)
18    })
19}
20
21/// How a node's part shows on its card.
22fn status(part: &NodeRun) -> FlowNodeStatus {
23    let why = || part.error.clone().unwrap_or_else(|| crate::casa::node_words(part.status).to_owned());
24    match part.status {
25        NodeStatus::Done | NodeStatus::Reused => FlowNodeStatus::Success,
26        NodeStatus::Pinned => FlowNodeStatus::Pinned,
27        NodeStatus::Failed | NodeStatus::TimedOut | NodeStatus::Uncertain => FlowNodeStatus::Error(why()),
28        NodeStatus::Waiting => FlowNodeStatus::Waiting("Aguardando".to_owned()),
29        NodeStatus::Skipped => FlowNodeStatus::Skipped,
30        NodeStatus::Cancelled => FlowNodeStatus::Cancelled,
31        NodeStatus::Disabled => FlowNodeStatus::Disabled,
32        _ => FlowNodeStatus::Normal,
33    }
34}
35
36/// "1 item", "4 itens".
37pub fn items(n: usize) -> String {
38    if n == 1 { "1 item".to_owned() } else { format!("{n} itens") }
39}
40
41/// How many items a node gave, on all its outputs.
42pub fn given(part: &NodeRun) -> usize {
43    part.outputs.values().map(Vec::len).sum()
44}
