1//! The runs: the list, and the one chosen drawn on its diagram, with Repetir and Aprovar e
2//! continuar.
3
4use std::collections::BTreeSet;
5
6use ravenrs::prelude::*;
7
8use super::{Cell, RavenFlows};
9use crate::casa;
10use crate::marcas::run_view;
11
12impl RavenFlows {
13    /// The runs, and the one chosen with its diagram.
14    pub(super) fn runs(&self) -> AnyView {
15        let s = self.s.clone();
16        let (casa, catalog) = (self.casa.clone(), self.catalog.clone());
17        let now = casa::now_ms();
18        let names: std::collections::BTreeMap<String, String> =
19            casa.flows().into_iter().map(|f| (f.id, f.name)).collect();
20        let runs = casa.runs();
21        let rows: Vec<Vec<Cell>> = runs
22            .iter()
23            .map(|r| {
24                vec![
25                    r.id.clone().into(),
26                    names.get(&r.flow_id).cloned().unwrap_or_else(|| r.flow_id.clone()).into(),
27                    casa::run_words(r.status).into(),
28                    casa::ago(r.created_ms, now).into(),
29                    f64::from(r.revision).into(),
30                    r.actor.clone().unwrap_or_default().into(),
31                ]
32            })
33            .collect();
34        let grid = DataGrid::new(
35            GridData::new(rows),
36            [
37                GridColumn::text("Id").hidden(),
38                GridColumn::text("Fluxo").fill(),
39                GridColumn::text("Estado").width(180.0),
40                GridColumn::text("Quando").width(140.0),
41                GridColumn::number("Revisão").width(110.0),
42                GridColumn::text("Pedida por").width(160.0),
43            ],
44        )
45        .key_column(0)
46        .selected_keys(s.run_keys.binding())
47        .empty_text("Nenhuma execução ainda. Abra um fluxo e execute.")
48        .height(280.0);
49
50        let chosen = s.run_keys.get().into_iter().next().and_then(|id| casa.run_record(&id));
51        let detail: AnyView = match chosen {
52            None => Box::new(
53                Text::new("Escolha uma execução para ver o que cada nó fez.").foreground(Semantic::InkMuted),
54            ),
55            Some(run) => {
56                // The flow of the run chosen, drawn again only when another run is chosen.
57                if s.run_shown.get().as_deref() != Some(run.id.as_str()) {
58                    let graph =
59                        casa.flow(&run.flow_id).map(|f| crate::nos::drawn(&f, &catalog)).unwrap_or_default();
60                    s.run_graph.set(graph);
61                    s.run_shown.set(Some(run.id.clone()));
62                }
63                let repeat = {
64                    let (casa, s, id) = (casa.clone(), s.clone(), run.id.clone());
65                    move || match casa.repeat(&id) {
66                        Ok(new) => {
67                            s.run_keys.set(BTreeSet::from([new]));
68                            s.message.set("Execução repetida.".to_owned());
69                        }
70                        Err(why) => s.message.set(why),
71                    }
72                };
73                let approve = {
74                    let (casa, s, id) = (casa.clone(), s.clone(), run.id.clone());
75                    move || match casa.approve(&id) {
76                        Ok(()) => s.message.set("Aprovado. A execução continua.".to_owned()),
77                        Err(why) => s.message.set(why),
78                    }
79                };
80                let waiting = casa.waiting_for(&run.id).is_some();
81                let mut actions = HStack::new((
82                    Text::new(format!(
83                        "Execução de {} · {}",
84                        names.get(&run.flow_id).cloned().unwrap_or_default(),
85                        casa::run_words(run.status)
86                    ))
87                    .style(TextStyle::Headline),
88                    Spacer::new(),
89                    Button::new("Repetir")
90                        .icon(IconName::RotateCcw)
91                        .style(ButtonStyle::Secondary)
92                        .on_click(repeat),
93                ))
94                .spacing(8.0)
95                .align(Align::Center);
96                let approve_button: AnyView = if waiting {
97                    Box::new(Button::new("Aprovar e continuar").icon(IconName::Check).on_click(approve))
98                } else {
99                    Box::new(Spacer::new().width(0.0))
100                };
101                actions = actions.spacing(8.0);
102                Box::new(
103                    VStack::new((
104                        HStack::new((actions.grow(1.0), approve_button)).spacing(8.0).align(Align::Center),
105                        crate::nos::kinds(
106                            FlowCanvas::new(s.run_graph.binding())
107                                .execution(run_view(&casa.parts(&run.id)))
108                                .label("Caminho executado")
109                                .read_only(true)
110                                .grow(1.0),
111                            &catalog,
112                        ),
113                    ))
114                    .spacing(8.0)
115                    .align(Align::Stretch)
116                    .grow(1.0),
117                )
118            }
119        };
120        Box::new(
121            VStack::new((
122                PageHeader::new("Execuções").description("Guardadas em ~/.raven-flows/execucoes.db."),
123                grid,
124                detail,
125            ))
126            .spacing(16.0)
127            .padding(24.0)
128            .align(Align::Stretch)
129            .grow(1.0),
130        )
131    }
132}
