22/// The pages of the sidebar.
23const FLOWS: usize = 0;

27/// What the screens keep between builds.
28struct Screens {
29    page: State<usize>,
30    /// The flow open in the editor.
31    open: State<Option<String>>,
32    graph: State<FlowGraph>,
33    history: State<FlowHistory>,
34    selection: State<FlowSelection>,
35    view: State<FlowViewport>,
36    arrivals: State<Option<FlowNode>>,
37    /// The parameters of the selected node, as typed.
38    parameters: State<String>,
39    /// The node whose parameters are in `parameters`.
40    parameters_of: State<Option<String>>,
41    /// The name of the next flow.
42    name: State<String>,
43    flow_keys: State<BTreeSet<String>>,
44    run_keys: State<BTreeSet<String>>,
45    /// The diagram of the run open in Execuções, and which run it is.
46    run_graph: State<FlowGraph>,
47    run_shown: State<Option<String>>,
48    days: State<Option<f64>>,
49    news: State<usize>,
50    message: State<String>,
51    confirm: ConfirmDialog,
52}

96impl View for RavenFlows {
97    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
98        self.casa.start(cx.ui_waker());
99        // The updater's threads wake the screen when they have something to say.
100        let waker = cx.ui_waker();
101        self.updater.connect(move || waker.wake());
102        self.updater.poll();
103        let s = &self.s;
104        let page = s.page.get();
105        let item = |icon, label: &str, at: usize| {
106            let p = s.page.clone();
107            let open = s.open.clone();
108            SidebarItem::new(icon, label).selected(page == at).on_click(move || {
109                p.set(at);
110                if at != FLOWS {
111                    open.set(None);
112                }
113            })
114        };
115        let waiting =
116            self.casa.runs().iter().filter(|r| r.status == ravenrs_flows_runtime::RunStatus::Waiting).count();
117        let mut runs_item = item(IconName::History, "Execuções", RUNS);
118        if waiting > 0 {
119            runs_item = runs_item.count_spoken(waiting.to_string(), format!("{waiting} aguardando"));
120        }
121        let sidebar = Sidebar::new((SidebarGroup::new(
122            "Automação",
123            (
124                item(IconName::GitBranch, "Fluxos", FLOWS),
125                runs_item,
126                item(IconName::Settings, "Configurações", SETTINGS),
127            ),
128        ),))
129        .label("Navegação")
130        .header(SidebarBrand::new(Icon::new(IconName::GitBranch).color(Semantic::Accent), "Raven Flows"));
131
132        let content: AnyView = match (page, s.open.get()) {
133            (FLOWS, Some(id)) => self.editor(cx, &id),
134            (FLOWS, None) => self.flows(),
135            (RUNS, _) => self.runs(),
136            _ => self.settings(),
137        };
138        let status = self.status_bar();
139        cx.child(
140            &VStack::new((
141                HStack::new((sidebar, VStack::new((content,)).align(Align::Stretch).grow(1.0)))
142                    .align(Align::Stretch)
143                    .grow(1.0),
144                status,
145                s.confirm.clone(),
146            ))
147            .spacing(0.0)
148            .align(Align::Stretch)
149            .grow(1.0),
150        )
151    }
152}

154impl RavenFlows {
155    /// The list of flows.
156    fn flows(&self) -> AnyView {
157        let s = self.s.clone();
158        let now = casa::now_ms();
159        let rows: Vec<Vec<Cell>> = self
160            .casa
161            .flows()
162            .iter()
163            .map(|f| {
164                let last = match self.casa.last_run(&f.id) {
165                    Some(r) => format!("{} {}", casa::run_words(r.status), casa::ago(r.created_ms, now)),
166                    None => "Nunca executado".to_owned(),
167                };
168                vec![
169                    f.id.clone().into(),
170                    f.name.clone().into(),
171                    (f.nodes.len() as f64).into(),
172                    (f64::from(f.revision)).into(),
173                    last.into(),
174                ]
175            })
176            .collect();
177        let (casa, catalog) = (self.casa.clone(), self.catalog.clone());
178        let opener = {
179            let (casa, catalog, s) = (casa.clone(), catalog.clone(), s.clone());
180            move |row: &Vec<Cell>| open_flow(&casa, &catalog, &s, &row[0].key_text())
181        };
182        let create = {
183            let (casa, catalog, s) = (casa.clone(), catalog.clone(), s.clone());
184            move || {
185                let name = s.name.get();
186                let name =
187                    if name.trim().is_empty() { "Novo fluxo".to_owned() } else { name.trim().to_owned() };
188                match casa.create(&name) {
189                    Ok(flow) => {
190                        s.name.set(String::new());
191                        s.message.set(format!("Fluxo {} criado.", flow.name));
192                        open_flow(&casa, &catalog, &s, &flow.id);
193                    }
194                    Err(why) => s.message.set(why),
195                }
196            }
197        };
198        let open_chosen = {
199            let (casa, catalog, s) = (casa.clone(), catalog.clone(), s.clone());
200            move || match s.flow_keys.get().into_iter().next() {
201                Some(id) => open_flow(&casa, &catalog, &s, &id),
202                None => s.message.set("Escolha na lista o fluxo a abrir.".to_owned()),
203            }
204        };
205        let delete = {
206            let (casa, s) = (casa.clone(), s.clone());
207            move || {
208                let keys = s.flow_keys.get();
209                if keys.is_empty() {
210                    s.message.set("Escolha na lista o fluxo a excluir.".to_owned());
211                    return;
212                }
213                let (casa, s2) = (casa.clone(), s.clone());
214                let names: Vec<String> = keys.iter().filter_map(|k| casa.flow(k).map(|f| f.name)).collect();
215                s.confirm.ask(
216                    "Excluir fluxo?",
217                    format!("{} sai da lista. As execuções ficam até a limpeza.", names.join(", ")),
218                    "Excluir",
219                    move || {
220                        for key in s2.flow_keys.get() {
221                            if let Err(why) = casa.delete(&key) {
222                                s2.message.set(why);
223                                return;
224                            }
225                        }
226                        s2.flow_keys.set(BTreeSet::new());
227                        s2.message.set("Fluxo excluído.".to_owned());
228                    },
229                );
230            }
231        };
232        Box::new(
233            VStack::new((
234                PageHeader::new("Fluxos").description("Cada fluxo é um arquivo em ~/.raven-flows/fluxos."),
235                HStack::new((
236                    TextField::new(s.name.binding()).placeholder("Nome do novo fluxo"),
237                    Button::new("Criar fluxo").icon(IconName::Plus).on_click(create),
238                    Spacer::new(),
239                    Button::new("Abrir")
240                        .icon(IconName::FileText)
241                        .style(ButtonStyle::Secondary)
242                        .on_click(open_chosen),
243                    Button::new("Excluir").style(ButtonStyle::Secondary).on_click(delete),
244                ))
245                .spacing(8.0)
246                .align(Align::Center),
247                DataGrid::new(
248                    GridData::new(rows),
249                    [
250                        GridColumn::text("Id").hidden(),
251                        GridColumn::text("Nome").fill(),
252                        GridColumn::number("Nós").width(90.0),
253                        GridColumn::number("Revisão").width(110.0),
254                        GridColumn::text("Última execução").width(260.0),
255                    ],
256                )
257                .key_column(0)
258                .selected_keys(s.flow_keys.binding())
259                .empty_text("Nenhum fluxo ainda. Dê um nome e crie o primeiro.")
260                .on_row_open(opener)
261                .grow(1.0),
262            ))
263            .spacing(16.0)
264            .padding(24.0)
265            .align(Align::Stretch)
266            .grow(1.0),
267        )
268    }

270    /// The settings: theme, how long runs are kept, the version and its news.
271    fn settings(&self) -> AnyView {
272        let s = self.s.clone();
273        let casa = self.casa.clone();
274        let prune = {
275            let s = s.clone();
276            move || {
277                let days = s.days.get().unwrap_or(30.0).max(0.0) as u64;
278                match casa.prune(days) {
279                    Ok(n) => s.message.set(format!("{n} execuções antigas removidas.")),
280                    Err(why) => s.message.set(why),
281                }
282            }
283        };
284        Box::new(
285            VStack::new((
286                PageHeader::new("Configurações").description("Valem para este computador."),
287                HStack::new((Text::new("Tema"), Spacer::new(), ThemeSwitcher::new())).align(Align::Center),
288                HStack::new((
289                    Text::new("Guardar execuções por (dias)"),
290                    Spacer::new(),
291                    NumberField::new(s.days.binding()).range(0.0, 3650.0).label("Dias"),
292                    Button::new("Limpar agora").style(ButtonStyle::Secondary).on_click(prune),
293                ))
294                .spacing(8.0)
295                .align(Align::Center),
296                HStack::new((
297                    Text::new(format!("Versão {}", crate::VERSION)).style(TextStyle::Headline),
298                    Spacer::new(),
299                    Button::new("Procurar atualização")
300                        .icon(IconName::RotateCcw)
301                        .style(ButtonStyle::Secondary)
302                        .on_click({
303                            let updater = self.updater.clone();
304                            move || updater.ask()
305                        }),
306                ))
307                .spacing(8.0)
308                .align(Align::Center),
309                self.failure(),
310                ReleaseNotes::new(release_notes(self.updater.timeline()), s.news.binding()).height(360.0),
311            ))
312            .spacing(16.0)
313            .padding(24.0)
314            .align(Align::Stretch)
315            .grow(1.0),
316        )
317    }

320/// The news of every version, newest first, in the format Raven Finanças keeps them:
321/// what changed, who made it and the issues it closed.
322pub fn release_notes(timeline: Vec<News>) -> Vec<ReleaseNote> {
323    timeline
324        .into_iter()
325        .map(|news| {
326            let issues: Vec<String> = news.issues.iter().map(|n| format!("#{n}")).collect();
327            ReleaseNote::new(news.versao, news.titulo)
328                .date(news.data.unwrap_or_default())
329                .changes(news.novidades)
330                .section("Quem contribuiu", news.contribuidores.join(", "))
331                .section("Issues resolvidas", issues.join(", "))
332        })
333        .collect()
334}

336impl RavenFlows {
337    /// The status bar: the version, what the updater says with its buttons, and the last
338    /// message of the app.
339    fn status_bar(&self) -> StatusBar {
340        let page = self.s.page.clone();
341        let bar = StatusBar::new()
342            .left((StatusBar::action(format!("Raven Flows {}", crate::VERSION))
343                .on_click(move || page.set(SETTINGS)),))
344            .right((StatusBar::text(self.s.message.get()).truncate(),));
345        let Some(notice) = self.updater.notice() else { return bar };
346        let buttons: Vec<Button> = notice.actions.iter().map(|&action| self.notice_button(action)).collect();
347        let bar = bar.center((
348            StatusBar::text(notice.text).foreground(Semantic::Ink).truncate().grow(1.0),
349            HStack::new(buttons).spacing(4.0),
350        ));
351        if notice.important { bar.center_tone(Tone::Warning) } else { bar }
352    }
353
354    /// A button of the updater's notice, doing what Raven Finanças does with it.
355    fn notice_button(&self, action: NoticeAction) -> Button {
356        let (updater, page) = (self.updater.clone(), self.s.page.clone());
357        let button = StatusBar::action(action.label());
358        let button = if action.is_primary() { button.style(ButtonStyle::Primary) } else { button };
359        button.on_click(move || match action {
360            NoticeAction::News | NoticeAction::HowToUpdate | NoticeAction::Details => page.set(SETTINGS),
361            NoticeAction::Update | NoticeAction::Retry => {
362                if let Some(news) = updater.status().news().cloned() {
363                    updater.install(news);
364                }
365            }
366            NoticeAction::Reopen => updater.reopen(),
367            NoticeAction::CheckAgain => updater.ask(),
368            NoticeAction::Later | NoticeAction::Close => updater.dismiss(),
369        })
370    }
371
372    /// Why the last update failed, where it stopped, the end of the installer's log and how
373    /// to install by hand; nothing when it did not fail.
374    fn failure(&self) -> AnyView {
375        let UpdateStatus::Failed(_, reason) = self.updater.status() else {
376            return Box::new(Spacer::new().height(0.0));
377        };
378        let stage = self.updater.stage().map(|s| format!(" Parou em: {s}.")).unwrap_or_default();
379        let tail = crate::versao::APP.installer_log_tail(12).unwrap_or_default();
380        Box::new(
381            VStack::new((
382                Alert::new(
383                    "A atualização não deu certo",
384                    format!("{reason}.{stage} A versão em uso continua funcionando."),
385                )
386                .tone(Tone::Danger),
387                CodeEditor::new(State::new(tail).binding())
388                    .language(CodeLanguage::Plain)
389                    .read_only(true)
390                    .rows(8),
391                Text::new(format!("Para instalar à mão: {}", crate::versao::APP.install_command()))
392                    .style(TextStyle::Callout)
393                    .foreground(Semantic::InkMuted),
394            ))
395            .spacing(8.0)
396            .align(Align::Stretch),
397        )
398    }
