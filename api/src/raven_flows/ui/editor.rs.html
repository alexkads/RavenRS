14/// Opens the flow `id` in the editor.
15pub(super) fn open_flow(casa: &Casa, catalog: &Catalog, s: &Screens, id: &str) {
16    let Some(flow) = casa.flow(id) else {
17        s.message.set("Este fluxo não existe mais.".to_owned());
18        return;
19    };
20    s.graph.set(nos::drawn(&flow, catalog));
21    s.history.set(FlowHistory::new());
22    s.selection.set(FlowSelection::default());
23    s.view.set(FlowViewport::fit());
24    s.parameters_of.set(None);
25    s.open.set(Some(id.to_owned()));
26    s.page.set(FLOWS);
27}

29/// The flow as drawn, with the id, name and revision the drawing keeps.
30fn drawn(graph: &FlowGraph, catalog: &Catalog) -> FlowDefinition {
31    FlowDefinition::from_graph(graph, catalog)
32}
33
34/// Saves the drawing; the saved flow, or why not.
35fn save(casa: &Casa, catalog: &Catalog, s: &Screens) -> Result<FlowDefinition, String> {
36    let mut flow = drawn(&s.graph.get(), catalog);
37    casa.save(&mut flow)?;
38    let revision = flow.revision;
39    s.graph.update(|g| {
40        if let Some(head) = g.data.get_mut(GRAPH_KEY) {
41            head["revision"] = revision.into();
42        }
43    });
44    Ok(flow)
45}

47/// The items that reached `node` in the last run of the flow `flow`; none before any run.
48fn input_of(casa: &Casa, flow: &str, node: &str) -> Vec<Json> {
49    let Some(run) = casa.last_run(flow) else { return Vec::new() };
50    casa.parts(&run.id)
51        .get(node)
52        .map(|part| part.inputs.values().flatten().map(|item| item.json.clone()).collect())
53        .unwrap_or_default()
54}

56/// The parameters of the card `id` of `graph`.
57fn parameters_of(graph: &FlowGraph, id: &str) -> Map<String, Json> {
58    graph
59        .find_node(id)
60        .and_then(|card| card.data.get(GRAPH_KEY))
61        .and_then(|setup| setup.get("parameters"))
62        .and_then(Json::as_object)
63        .cloned()
64        .unwrap_or_default()
65}
66
67/// The parameters of the card `id` of `graph`, as JSON text.
68fn parameters_text(graph: &FlowGraph, id: &str) -> String {
69    serde_json::to_string_pretty(&Json::Object(parameters_of(graph, id))).unwrap_or_default()
70}

72/// Puts `text` as the parameters of the card `id`: refused, with why, when it is not a JSON
73/// object.
74pub fn apply_parameters(graph: &mut FlowGraph, id: &str, text: &str) -> Result<(), String> {
75    let parameters: Map<String, Json> = match serde_json::from_str::<Json>(text) {
76        Ok(Json::Object(map)) => map,
77        Ok(_) => return Err("Os parâmetros são um objeto JSON, entre { e }.".to_owned()),
78        Err(e) => return Err(format!("JSON inválido: {e}")),
79    };
80    let card = graph.nodes.iter_mut().find(|n| n.id == id).ok_or("O nó não está mais no fluxo.")?;
81    let mut setup: NodeDef = card
82        .data
83        .get(GRAPH_KEY)
84        .and_then(|v| serde_json::from_value(v.clone()).ok())
85        .unwrap_or_else(|| NodeDef::new(&card.id, &card.kind));
86    setup.parameters = parameters;
87    let mut data = if card.data.is_object() { card.data.clone() } else { Json::Object(Map::new()) };
88    data[GRAPH_KEY] = serde_json::to_value(&setup).map_err(|e| e.to_string())?;
89    card.data = data;
90    Ok(())
91}

93impl RavenFlows {
94    /// The editor of the flow `id`.
95    pub(super) fn editor(&self, cx: &mut BuildCx<'_>, id: &str) -> AnyView {
96        let s = self.s.clone();
97        let (casa, catalog) = (self.casa.clone(), self.catalog.clone());
98        let _ = cx;
99        // The parameters of the node selected now.
100        let selected = s.selection.get().nodes.first().cloned();
101        if s.parameters_of.get() != selected {
102            s.parameters
103                .set(selected.as_deref().map(|n| parameters_text(&s.graph.get(), n)).unwrap_or_default());
104            s.parameters_of.set(selected.clone());
105        }
106        let name = s
107            .graph
108            .get()
109            .data
110            .get(GRAPH_KEY)
111            .and_then(|h| h.get("name"))
112            .and_then(Json::as_str)
113            .unwrap_or("Fluxo")
114            .to_owned();
115
116        let back = {
117            let s = s.clone();
118            move || s.open.set(None)
119        };
120        let save_click = {
121            let (casa, catalog, s) = (casa.clone(), catalog.clone(), s.clone());
122            move || match save(&casa, &catalog, &s) {
123                Ok(flow) => s.message.set(format!("Salvo, revisão {}.", flow.revision)),
124                Err(why) => s.message.set(why),
125            }
126        };
127        let run_click = {
128            let (casa, catalog, s) = (casa.clone(), catalog.clone(), s.clone());
129            move || {
130                let flow = drawn(&s.graph.get(), &catalog);
131                if let Err(errors) = Plan::compile(&flow, &catalog) {
132                    let words: Vec<String> = errors.iter().map(ToString::to_string).collect();
133                    s.message.set(format!("O fluxo não pode rodar: {}", words.join("; ")));
134                    return;
135                }
136                match save(&casa, &catalog, &s).and_then(|flow| casa.run(&flow)) {
137                    Ok(_) => s.message.set("Execução pedida.".to_owned()),
138                    Err(why) => s.message.set(why),
139                }
140            }
141        };
142        let apply = {
143            let s = s.clone();
144            move || {
145                let Some(node) = s.parameters_of.get() else { return };
146                let mut graph = s.graph.get();
147                match apply_parameters(&mut graph, &node, &s.parameters.get()) {
148                    Ok(()) => {
149                        let before = s.graph.get();
150                        s.history.update(|h| h.record(before));
151                        s.graph.set(graph);
152                        s.message.set("Parâmetros aplicados. Salve para guardar.".to_owned());
153                    }
154                    Err(why) => s.message.set(why),
155                }
156            }
157        };
158
159        let mut canvas = FlowCanvas::new(s.graph.binding())
160            .label("Área do fluxo")
161            .view(s.view.binding())
162            .selection(s.selection.binding())
163            .history(s.history.binding())
164            .arrivals(s.arrivals.binding())
165            .editable_titles(true)
166            .minimap(true)
167            // A connection let go in the open searches the kinds of node, created connected.
168            .open_end(nos::models(&catalog))
169            .grow(1.0);
170        // The last run, drawn on the canvas as it moves on; the flow itself is not touched.
171        if let Some(run) = casa.last_run(id) {
172            canvas = canvas.execution(marcas::run_view(&casa.parts(&run.id)));
173        }
174        let canvas = nos::kinds(canvas, &catalog);
175        let side: AnyView = match &selected {
176            Some(node) => {
177                let (kind, title) = s
178                    .graph
179                    .get()
180                    .find_node(node)
181                    .map(|n| (n.kind.clone(), n.title.clone()))
182                    .unwrap_or_default();
183                let editor: AnyView = match nos::schema(&kind) {
184                    // The form of the kind's schema edits the card's parameters, and Aplicar is one
185                    // change of the history.
186                    Some(fields) => {
187                        let (read, write, id, id2) = (s.graph.clone(), s.clone(), node.clone(), node.clone());
188                        let values = Binding::new(
189                            move || ParamValues::from(parameters_of(&read.get(), &id)),
190                            move |values: ParamValues| {
191                                let mut graph = write.graph.get();
192                                let text = Json::Object(values.into_map()).to_string();
193                                match apply_parameters(&mut graph, &id2, &text) {
194                                    Ok(()) => {
195                                        let before = write.graph.get();
196                                        write.history.update(|h| h.record(before));
197                                        write.graph.set(graph);
198                                        write
199                                            .message
200                                            .set("Parâmetros aplicados. Salve para guardar.".to_owned());
201                                    }
202                                    Err(why) => write.message.set(why),
203                                }
204                            },
205                        );
206                        Box::new(ParameterForm::new(fields, values))
207                    }
208                    None => Box::new(
209                        VStack::new((
210                            Text::new("Em JSON: este tipo de nó guarda listas ou objetos.")
211                                .style(TextStyle::Callout)
212                                .foreground(Semantic::InkMuted),
213                            CodeEditor::new(s.parameters.binding()).language(CodeLanguage::Plain).fill(),
214                            Button::new("Aplicar parâmetros")
215                                .icon(IconName::Check)
216                                .style(ButtonStyle::Secondary)
217                                .on_click(apply),
218                        ))
219                        .spacing(8.0)
220                        .align(Align::Stretch)
221                        .grow(1.0),
222                    ),
223                };
224                Box::new(
225                    VStack::new((
226                        Text::new(format!("Parâmetros de {title}")).style(TextStyle::Headline),
227                        editor,
228                        Text::new("Entrada na última execução").style(TextStyle::Headline),
229                        // What reached the node, in the inspector's four ways.
230                        DataInspector::of(input_of(&casa, id, node)).limit(100).label("Entrada"),
231                    ))
232                    .spacing(8.0)
233                    .padding(12.0)
234                    .align(Align::Stretch)
235                    .width(340.0),
236                )
237            }
238            None => {
239                Box::new(VStack::new((nos::toolbox(&catalog).arrivals(s.arrivals.binding()),)).width(240.0))
240            }
241        };
242
243        let mut table = Table::new([
244            Column::new("Nó"),
245            Column::new("Estado"),
246            Column::new("Itens"),
247            Column::new("Tentativas"),
248            Column::new("Erro"),
249        ]);
250        let mut summary = "Ainda não executado.".to_owned();
251        if let Some(run) = casa.last_run(id) {
252            summary = format!("Última execução: {}", casa::run_words(run.status));
253            let parts = casa.parts(&run.id);
254            // In the flow's order, the nodes that took part.
255            for card in s.graph.get().nodes.iter().filter(|n| parts.contains_key(&n.id)) {
256                let part = &parts[&card.id];
257                let given = match part.status {
258                    NodeStatus::Waiting | NodeStatus::Skipped => "—".to_owned(),
259                    _ => marcas::items(marcas::given(part)),
260                };
261                table = table.row([
262                    card.title.clone(),
263                    casa::node_words(part.status).to_owned(),
264                    given,
265                    part.attempts.to_string(),
266                    part.error.clone().unwrap_or_default(),
267                ]);
268            }
269        }
270
271        Box::new(
272            VStack::new((
273                HStack::new((
274                    Button::new("Voltar aos fluxos")
275                        .icon(IconName::ArrowLeft)
276                        .style(ButtonStyle::Ghost)
277                        .on_click(back),
278                    Text::new(name).style(TextStyle::Title),
279                    Spacer::new(),
280                    Button::new("Salvar")
281                        .icon(IconName::Save)
282                        .style(ButtonStyle::Secondary)
283                        .on_click(save_click),
284                    Button::new("Executar fluxo").icon(IconName::Play).on_click(run_click),
285                ))
286                .spacing(8.0)
287                .align(Align::Center)
288                .padding(12.0),
289                HStack::new((side, canvas)).align(Align::Stretch).grow(1.0),
290                VStack::new((Text::new(summary).style(TextStyle::Headline), table))
291                    .spacing(8.0)
292                    .padding(12.0),
293            ))
294            .spacing(0.0)
295            .align(Align::Stretch)
296            .grow(1.0),
297        )
298    }
