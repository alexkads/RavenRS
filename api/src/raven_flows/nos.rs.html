8/// How a kind shows: its name, its icon and its group in the toolbox.
9pub fn look(kind: &str) -> (&'static str, IconName, &'static str) {
10    match kind {
11        "manual" => ("Início manual", IconName::Play, "Gatilhos"),
12        "if" => ("Se", IconName::Split, "Lógica"),
13        "switch" => ("Escolher caminho", IconName::GitBranch, "Lógica"),
14        "merge" => ("Juntar", IconName::Merge, "Lógica"),
15        "batches" => ("Em lotes", IconName::Layers, "Lógica"),
16        "wait" => ("Esperar", IconName::Timer, "Tempo"),
17        "delay" => ("Esperar, mesmo fechado", IconName::Clock, "Tempo"),
18        "wait_event" => ("Esperar aprovação", IconName::Bell, "Tempo"),
19        "set" => ("Definir campos", IconName::FileText, "Dados"),
20        "filter" => ("Filtrar", IconName::Filter, "Dados"),
21        "sort" => ("Ordenar", IconName::ListOrdered, "Dados"),
22        "group" => ("Agrupar", IconName::Sigma, "Dados"),
23        "split" => ("Separar lista", IconName::Split, "Dados"),
24        "join" => ("Juntar em lista", IconName::Layers, "Dados"),
25        "codex" => ("Codex CLI", IconName::Terminal, "Agentes"),
26        "claude" => ("Claude CLI", IconName::Sparkles, "Agentes"),
27        "code_review" => ("Revisão de código", IconName::Eye, "Agentes"),
28        _ => ("Nó", IconName::Square, "Outros"),
29    }
30}

32/// The parameters of a kind as a schema for the `ParameterForm`, for the kinds whose parameters
33/// are plain values; the others are edited as JSON.
34pub fn schema(kind: &str) -> Option<Vec<ParamField>> {
35    let condition = || {
36        ParamField::new("condition", "Condição", ParamKind::Text)
37            .required()
38            .hint("Na linguagem de critérios, como [valor] > 100")
39    };
40    let ms = || ParamField::new("ms", "Espera (ms)", ParamKind::Number).required().default(1000);
41    Some(match kind {
42        "if" | "filter" => vec![condition()],
43        "wait" | "delay" => vec![ms()],
44        "wait_event" => vec![
45            ParamField::new("event", "Evento", ParamKind::Text)
46                .required()
47                .hint("O nome que o app usa para aprovar"),
48            ParamField::new("keep", "Deixar passar os itens que esperavam", ParamKind::Boolean).default(true),
49            ParamField::new("as", "Campo do evento em cada item", ParamKind::Text)
50                .default("evento")
51                .shown_when("keep", true),
52        ],
53        "codex" | "claude" => vec![
54            ParamField::new("prompt", "Instruções", ParamKind::LongText)
55                .required()
56                .hint("Texto com expressões, como Liste os arquivos de {{ pasta }}"),
57            ParamField::new("dir", "Pasta de trabalho", ParamKind::Text),
58            ParamField::new("read_only", "Só ler os arquivos", ParamKind::Boolean)
59                .default(true)
60                .hint("Desligado, o agente pode mudar arquivos da pasta"),
61            ParamField::new("timeout_ms", "Tempo limite (ms)", ParamKind::Number).default(600_000),
62        ],
63        "code_review" => vec![
64            ParamField::new("prompt", "O que revisar", ParamKind::LongText)
65                .required()
66                .hint("O diff ou os arquivos; a revisão só lê"),
67            ParamField::new("dir", "Pasta de trabalho", ParamKind::Text),
68        ],
69        _ => return None,
70    })
71}

73/// The kinds of node of the app: the built-in ones and the coding agents installed on the
74/// computer, whose login stays with each CLI.
75pub fn catalog() -> Catalog {
76    use ravenrs_agent_cli::{Agent, Cli};
77    ravenrs_agent_cli::with_agents(
78        Catalog::builtin(),
79        Cli::new(Agent::Codex),
80        Cli::new(Agent::Claude),
81        Cli::new(Agent::Codex),
82    )
83}

85/// The groups of the toolbox, in order.
86const GROUPS: [&str; 6] = ["Gatilhos", "Lógica", "Dados", "Tempo", "Agentes", "Outros"];
87
88/// The toolbox, a group per kind of work, with every kind of `catalog`.
89pub fn toolbox(catalog: &Catalog) -> FlowToolbox {
90    let specs = catalog.specs();
91    let mut toolbox = FlowToolbox::new().title("Nós").searchable(true);
92    for group in GROUPS {
93        let mut any = false;
94        for spec in specs.iter().filter(|s| look(&s.kind).2 == group) {
95            if !any {
96                toolbox = toolbox.group(group);
97                any = true;
98            }
99            toolbox =
100                toolbox.model(look(&spec.kind).0, ravenrs_flows::card(spec, &spec.kind, look(&spec.kind).0));
101        }
102    }
103    toolbox
104}

106/// `flow` drawn for the canvas, each node the catalog names in English called by its name in
107/// Portuguese; a title the person gave stays.
108pub fn drawn(flow: &ravenrs_flows::FlowDefinition, catalog: &Catalog) -> FlowGraph {
109    let mut graph = flow.to_graph(catalog);
110    let specs = catalog.specs();
111    for card in &mut graph.nodes {
112        if specs.iter().any(|s| s.kind == card.kind && s.title == card.title) {
113            card.title = look(&card.kind).0.to_owned();
114        }
115    }
116    graph
117}

119/// Every kind of `catalog` as a card, with its name in Portuguese: what the toolbox offers and
120/// what a connection let go in the open can lead to.
121pub fn models(catalog: &Catalog) -> Vec<(String, FlowNode)> {
122    catalog
123        .specs()
124        .iter()
125        .map(|spec| {
126            let (name, _, _) = look(&spec.kind);
127            (name.to_owned(), ravenrs_flows::card(spec, &spec.kind, name))
128        })
129        .collect()
130}

132/// The canvas with every kind of `catalog` drawn with its icon and name.
133pub fn kinds(mut canvas: FlowCanvas, catalog: &Catalog) -> FlowCanvas {
134    for spec in catalog.specs() {
135        let (name, icon, group) = look(&spec.kind);
136        let tone = if group == "Gatilhos" { Semantic::Warning } else { Semantic::Accent };
137        canvas = canvas.kind(spec.kind.as_str(), FlowKind::new(icon, name).tone(tone));
138    }
139    canvas
140}
