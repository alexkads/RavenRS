1//! The printers among the drivers that come with the program. This is only a view of that catalog,
2//! for choosing by name: what is known about each one lives in `ravenrs-perifericos`.
3
4use std::sync::OnceLock;
5
6use ravenrs_perifericos::{Ajustes, Manifesto, TipoPeriferico, manifestos_embutidos};
7
8use crate::language::Language;
9
10/// A printer model and the languages it takes, the first one preferred. The choice stays editable.
11#[derive(Clone, Debug, PartialEq, Eq)]
12pub struct PrinterPreset {
13    /// The exact model.
14    pub name: String,
15    /// The driver this comes from.
16    pub driver_id: String,
17    /// Its languages. PPLB is [`Language::Epl`].
18    pub languages: Vec<Language>,
19    /// What differs on this model, for [`LabelJob::ajustes`](crate::LabelJob::ajustes).
20    pub ajustes: Ajustes,
21    /// What to set on the printer before using it, to show to whoever installs it.
22    pub setup: Option<String>,
23}
24
25fn preset_de(driver: &Manifesto) -> Option<PrinterPreset> {
26    if !matches!(driver.tipo, TipoPeriferico::ImpressoraTermica | TipoPeriferico::Etiquetadora) {
27        return None;
28    }
29    let protocolos = std::iter::once(&driver.protocolo).chain(&driver.protocolos_alternativos);
30    Some(PrinterPreset {
31        name: driver.nome_do_modelo().to_owned(),
32        driver_id: driver.id.clone(),
33        languages: protocolos.filter_map(|protocolo| Language::do_protocolo(protocolo.nome)).collect(),
34        ajustes: driver.ajustes,
35        setup: driver.instalacao.clone(),
36    })
37}
38
39/// The printers known by name.
40pub fn presets() -> &'static [PrinterPreset] {
41    static PRESETS: OnceLock<Vec<PrinterPreset>> = OnceLock::new();
42    PRESETS.get_or_init(|| manifestos_embutidos().iter().filter_map(preset_de).collect())
43}

45#[cfg(test)]
46mod tests {
47    use super::*;
48
49    #[test]
50    fn every_preset_has_a_language_and_a_distinct_name() {
51        assert_eq!(presets().len(), 9);
52        for (index, preset) in presets().iter().enumerate() {
53            assert!(!preset.languages.is_empty(), "{}", preset.name);
54            assert!(presets().iter().skip(index + 1).all(|other| other.name != preset.name));
55        }
56    }
57
58    #[test]
59    fn no_preset_has_only_a_brand_for_a_name() {
60        for vague in ["TSC", "Argox", "Elgin", "Epson", "Zebra"] {
61            assert!(presets().iter().all(|preset| preset.name != vague), "{vague}");
62        }
63    }
64
65    #[test]
66    fn printers_keep_their_languages_and_adjustments() {
67        let find = |name: &str| presets().iter().find(|preset| preset.name == name);
68        for name in ["Epson TM-T20", "Elgin i9", "Bematech MP-4200 TH"] {
69            assert_eq!(
70                find(name).map(|preset| preset.languages.clone()),
71                Some(vec![Language::EscPos]),
72                "{name}"
73            );
74        }
75        assert_eq!(
76            find("Elgin L42 Pro").map(|preset| preset.languages.clone()),
77            Some(vec![Language::Zpl, Language::Epl, Language::Ppla])
78        );
79        let bematech = find("Bematech MP-4200 TH");
80        assert!(bematech.is_some_and(|preset| preset.setup.is_some() && preset.ajustes.bipe.is_some()));
81    }
82}
