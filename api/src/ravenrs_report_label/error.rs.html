1//! What can go wrong between a document and a printed label. The messages are for the person at the
2//! counter, so they are in Portuguese and say what to change.
3
4use std::fmt;
5
6/// Why a label could not be made or sent.
7#[derive(Clone, Debug, PartialEq)]
8#[non_exhaustive]
9pub enum LabelError {
10    /// A barcode cannot keep its minimum module in its box.
11    BarcodeDoesNotFit {
12        /// The format's name.
13        symbology: &'static str,
14        /// What it encodes.
15        content: String,
16        /// Where its box starts on the label, in millimeters.
17        at_mm: (f32, f32),
18        /// The size available across, in millimeters.
19        available_mm: f32,
20        /// The size the code needs across, quiet zone included, in millimeters.
21        needed_mm: f32,
22        /// The module that fits, in whole dots. It can be zero.
23        module_dots: u32,
24        /// The smallest module allowed at this resolution, in whole dots.
25        minimum_dots: u32,
26        /// The smallest module allowed for this format, in millimeters.
27        minimum_mm: f32,
28        /// The printer's resolution.
29        dpi: u32,
30    },
31    /// The content cannot be encoded in the format.
32    InvalidBarcode {
33        /// The format's name.
34        symbology: &'static str,
35        /// What it should encode.
36        content: String,
37        /// What the encoder said.
38        reason: String,
39    },
40    /// The document has no pages to print.
41    EmptyDocument,
42    /// The page cannot become a picture at this resolution.
43    InvalidPage(String),
44    /// A setting of the job is out of range for the language.
45    InvalidJob(String),
46    /// The printer's name or address is not acceptable.
47    InvalidDestination(String),
48    /// The printer or the file did not take the bytes.
49    Output(String),
50    /// What went wrong on one label of the document.
51    OnLabel {
52        /// The label, from 1.
53        label: usize,
54        /// What went wrong.
55        source: Box<LabelError>,
56    },
57}
58
59fn mm(value: f32, decimals: usize) -> String {
60    format!("{value:.decimals$}").replace('.', ",")
61}
62
63impl fmt::Display for LabelError {
64    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
65        match self {
66            Self::BarcodeDoesNotFit {
67                symbology,
68                content,
69                at_mm,
70                available_mm,
71                needed_mm,
72                module_dots,
73                minimum_dots,
74                minimum_mm,
75                dpi,
76            } => write!(
77                f,
78                "o código {symbology} \"{content}\", no controle em {} x {} mm, não cabe: tem {} mm, \
79                 precisa de {} mm com a zona quieta. A {dpi} dpi o módulo daria {module_dots} ponto(s), \
80                 e o mínimo é {minimum_dots} ({} mm). Aumente o controle, use uma etiqueta maior ou \
81                 uma impressora de resolução maior.",
82                mm(at_mm.0, 1),
83                mm(at_mm.1, 1),
84                mm(*available_mm, 1),
85                mm(*needed_mm, 1),
86                mm(*minimum_mm, 3),
87            ),
88            Self::InvalidBarcode { symbology, content, reason } => {
89                write!(f, "o código {symbology} não aceita \"{content}\": {reason}")
90            }
91            Self::EmptyDocument => write!(f, "o documento não tem nenhuma página para imprimir"),
92            Self::InvalidPage(why) => write!(f, "a página não vira etiqueta: {why}"),
93            Self::InvalidJob(why) => write!(f, "impressão inválida: {why}"),
94            Self::InvalidDestination(why) => write!(f, "destino inválido: {why}"),
95            Self::Output(why) => write!(f, "falha ao enviar para a impressora: {why}"),
96            Self::OnLabel { label, source } => write!(f, "etiqueta {label}: {source}"),
97        }
98    }
99}
100
101impl std::error::Error for LabelError {
102    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
103        match self {
104            Self::OnLabel { source, .. } => Some(source.as_ref()),
105            _ => None,
106        }
107    }
108}
