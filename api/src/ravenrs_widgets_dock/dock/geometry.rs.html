11/// How far inside a group, as a part of it, a drop still means its edge.
12const EDGE_PART: f32 = 0.25;
13/// How close to the border of the area a drop means a dock that is not showing.
14const BORDER_ZONE: f32 = 28.0;
15/// How close to the top a drop opens the top dock. Much thinner than the other borders: the tabs of
16/// the groups at the top lie under it, and dropping among them has to keep working.
17const TOP_ZONE: f32 = 4.0;
18/// How thick the rail is where the panels hidden at the edge of a dock have their tabs.
19#[doc(hidden)]
20pub const RAIL: f32 = 28.0;
21/// How thick the line that marks a gap between tabs is.
22const GAP_LINE: f32 = 2.0;

24/// The measures an arrangement is laid out with.
25#[derive(Clone, Copy, Debug, PartialEq)]
26#[doc(hidden)]
27pub struct Metrics {
28    /// How thick a bar between parts is.
29    pub bar: f32,
30    /// How thick the strip that collapses a dock is, or zero without one.
31    pub toggle: f32,
32    /// The smallest and the largest the left dock gets.
33    pub left: (f32, f32),
34    /// The smallest and the largest the right dock gets.
35    pub right: (f32, f32),
36    /// The smallest and the largest the bottom dock gets.
37    pub bottom: (f32, f32),
38    /// The smallest and the largest the top dock gets.
39    pub top: (f32, f32),
40    /// The room the center keeps, which docks do not take from it.
41    pub center_min: (f32, f32),
42    /// The smallest a group gets along a split.
43    pub pane_min: f32,
44}

46#[cfg_attr(feature = "live", ravenrs_live::expose)]
47impl Metrics {
48    fn limits(&self, side: DockSide) -> (f32, f32) {
49        match side {
50            DockSide::Left => self.left,
51            DockSide::Right => self.right,
52            DockSide::Bottom => self.bottom,
53            DockSide::Top => self.top,
54            _ => (0.0, f32::MAX),
55        }
56    }
57}

108/// The sizes the parts of a split get in `room`: their shares, with each part at least at its
109/// smallest and the difference taken from those that have some to spare.
110#[doc(hidden)]
111pub fn distribute(room: f32, shares: &[f32], mins: &[f32]) -> Vec<f32> {
112    let mut sizes: Vec<f32> = shares.iter().zip(mins).map(|(share, min)| (share * room).max(*min)).collect();
113    let mut over = sizes.iter().sum::<f32>() - room;
114    if over > 0.0 {
115        let mut spare: f32 = sizes.iter().zip(mins).map(|(size, min)| size - min).sum();
116        if spare > 0.0 {
117            let part = (over / spare).min(1.0);
118            for (size, min) in sizes.iter_mut().zip(mins) {
119                *size -= (*size - min) * part;
120            }
121            spare = over.min(spare);
122            over -= spare;
123        }
124        // Too little room for every smallest: the last parts give way, down to nothing.
125        for size in sizes.iter_mut().rev() {
126            if over <= 0.0 {
127                break;
128            }
129            let taken = size.min(over);
130            *size -= taken;
131            over -= taken;
132        }
133    }
134    sizes
135}

137/// The least a node can be along `axis`.
138fn least(node: &DockNode, axis: Axis, metrics: &Metrics) -> f32 {
139    match node {
140        DockNode::Tabs { .. } => metrics.pane_min,
141        DockNode::Split { axis: own, children } => {
142            let parts = children.iter().map(|c| least(&c.node, axis, metrics));
143            if *own == axis {
144                parts.sum::<f32>() + children.len().saturating_sub(1) as f32 * metrics.bar
145            } else {
146                parts.fold(0.0, f32::max)
147            }
148        }
149    }
150}

160fn place_node(
161    node: &DockNode,
162    rect: Rect,
163    side: DockSide,
164    path: &mut Vec<usize>,
165    metrics: &Metrics,
166    out: &mut Arranged,
167) {
168    match node {
169        DockNode::Tabs { panels, active } => {
170            out.groups.push(GroupPlace { panels: panels.clone(), active: *active, rect });
171        }
172        DockNode::Split { axis, children } => place_split(*axis, children, rect, side, path, metrics, out),
173    }
174}

176fn place_split(
177    axis: Axis,
178    children: &[DockChild],
179    rect: Rect,
180    side: DockSide,
181    path: &mut Vec<usize>,
182    metrics: &Metrics,
183    out: &mut Arranged,
184) {
185    let vertical = axis == Axis::Vertical;
186    let length = if vertical { rect.height() } else { rect.width() };
187    let room = (length - children.len().saturating_sub(1) as f32 * metrics.bar).max(0.0);
188    let shares: Vec<f32> = children.iter().map(|c| c.share).collect();
189    let mins: Vec<f32> = children.iter().map(|c| least(&c.node, axis, metrics)).collect();
190    let sizes = distribute(room, &shares, &mins);
191    let mut at = if vertical { rect.y() } else { rect.x() };
192    for (index, child) in children.iter().enumerate() {
193        let size = sizes.get(index).copied().unwrap_or(0.0);
194        let part = if vertical {
195            Rect::new(rect.x(), at, rect.width(), size)
196        } else {
197            Rect::new(at, rect.y(), size, rect.height())
198        };
199        path.push(index);
200        place_node(&child.node, part, side, path, metrics, out);
201        path.pop();
202        at += size;
203        if let (Some(next), Some(next_size), Some(next_min)) =
204            (children.get(index + 1), sizes.get(index + 1), mins.get(index + 1))
205        {
206            let bar = if vertical {
207                Rect::new(rect.x(), at, rect.width(), metrics.bar)
208            } else {
209                Rect::new(at, rect.y(), metrics.bar, rect.height())
210            };
211            let both = size + next_size;
212            let min = mins[index].min(both);
213            out.bars.push(BarPlace {
214                kind: BarKind::Split {
215                    side,
216                    path: path.clone(),
217                    index,
218                    both,
219                    names: (shown_name(&child.node), shown_name(&next.node)),
220                },
221                axis,
222                rect: bar,
223                value: size,
224                min,
225                max: (both - next_min).max(min),
226            });
227            at += metrics.bar;
228        }
229    }
230}

232/// How large a dock gets out of `room`, and how large it could get while the center and `taken`
233/// keep theirs.
234fn dock_size(
235    layout: &DockLayout,
236    side: DockSide,
237    room: f32,
238    taken: f32,
239    keep: f32,
240    metrics: &Metrics,
241) -> (f32, f32) {
242    let (min, max) = metrics.limits(side);
243    let most = (room - taken - keep - metrics.bar).clamp(min, max.max(min));
244    (layout.size(side).clamp(min, most.max(min)).min((room - taken).max(0.0)), most)
245}

247/// Lays an arrangement out in an area of `size`.
248#[doc(hidden)]
249pub fn arrange(layout: &DockLayout, size: Size, metrics: &Metrics) -> Arranged {
250    let mut out = Arranged { size, ..Arranged::default() };
251    let has = |side| layout.node(side).is_some();
252    let open = |side| has(side) && layout.is_open(side);
253    let mut x0 = 0.0;
254    let mut x1 = size.width;
255    let mut y1 = size.height;
256    let toggle = metrics.toggle;
257    let toggles = |side| has(side) && toggle > 0.0;
258    let rails = |side| !layout.auto_hidden_panels(side).is_empty();
259    // The rails go outermost, then the strips, then the docks.
260    if rails(DockSide::Left) {
261        out.rails.push((DockSide::Left, Rect::new(0.0, 0.0, RAIL, size.height)));
262        x0 += RAIL;
263    }
264    if rails(DockSide::Right) {
265        x1 -= RAIL;
266        out.rails.push((DockSide::Right, Rect::new(x1, 0.0, RAIL, size.height)));
267    }
268    if toggles(DockSide::Left) {
269        out.toggles.push((DockSide::Left, Rect::new(x0, 0.0, toggle, size.height)));
270        x0 += toggle;
271    }
272    if toggles(DockSide::Right) {
273        x1 -= toggle;
274        out.toggles.push((DockSide::Right, Rect::new(x1, 0.0, toggle, size.height)));
275    }
276    let width = (x1 - x0).max(0.0);
277    let (center_w, center_h) = metrics.center_min;
278    // The right gives way first when a narrow window has too little for both.
279    let left_size = open(DockSide::Left).then(|| {
280        let right_least = if open(DockSide::Right) { metrics.right.0 + metrics.bar } else { 0.0 };
281        dock_size(layout, DockSide::Left, width, right_least, center_w, metrics)
282    });
283    let right_size = open(DockSide::Right).then(|| {
284        let taken = left_size.map_or(0.0, |(left, _)| left + metrics.bar);
285        dock_size(layout, DockSide::Right, width, taken, center_w, metrics)
286    });
287    let mut path = Vec::new();
288    if let (Some((left, most)), Some(node)) = (left_size, layout.node(DockSide::Left)) {
289        place_node(node, Rect::new(x0, 0.0, left, size.height), DockSide::Left, &mut path, metrics, &mut out);
290        x0 += left;
291        let (min, _) = metrics.left;
292        out.bars.push(BarPlace {
293            kind: BarKind::Dock(DockSide::Left),
294            axis: Axis::Horizontal,
295            rect: Rect::new(x0, 0.0, metrics.bar, size.height),
296            value: left,
297            min,
298            max: most,
299        });
300        x0 += metrics.bar;
301    } else if !open(DockSide::Left) {
302        let (min, max) = metrics.left;
303        let ghost = layout.size(DockSide::Left).clamp(min, max).min(width / 2.0);
304        out.ghosts.push((DockSide::Left, Rect::new(x0, 0.0, ghost, size.height)));
305    }
306    if let (Some((right, most)), Some(node)) = (right_size, layout.node(DockSide::Right)) {
307        x1 -= right;
308        place_node(
309            node,
310            Rect::new(x1, 0.0, right, size.height),
311            DockSide::Right,
312            &mut path,
313            metrics,
314            &mut out,
315        );
316        x1 -= metrics.bar;
317        let (min, _) = metrics.right;
318        out.bars.push(BarPlace {
319            kind: BarKind::Dock(DockSide::Right),
320            axis: Axis::Horizontal,
321            rect: Rect::new(x1, 0.0, metrics.bar, size.height),
322            value: right,
323            min,
324            max: most,
325        });
326    } else if !open(DockSide::Right) {
327        let (min, max) = metrics.right;
328        let ghost = layout.size(DockSide::Right).clamp(min, max).min(width / 2.0);
329        out.ghosts.push((DockSide::Right, Rect::new(x1 - ghost, 0.0, ghost, size.height)));
330    }
331    let middle = (x1 - x0).max(0.0);
332    let mut y0 = 0.0;
333    if rails(DockSide::Top) {
334        out.rails.push((DockSide::Top, Rect::new(x0, 0.0, middle, RAIL)));
335        y0 += RAIL;
336    }
337    if rails(DockSide::Bottom) {
338        y1 -= RAIL;
339        out.rails.push((DockSide::Bottom, Rect::new(x0, y1, middle, RAIL)));
340    }
341    if toggles(DockSide::Top) {
342        out.toggles.push((DockSide::Top, Rect::new(x0, y0, middle, toggle)));
343        y0 += toggle;
344    }
345    if toggles(DockSide::Bottom) {
346        y1 -= toggle;
347        out.toggles.push((DockSide::Bottom, Rect::new(x0, y1, middle, toggle)));
348    }
349    let height = (y1 - y0).max(0.0);
350    // The top gives way first when a short window has too little for both.
351    let bottom_size = open(DockSide::Bottom).then(|| {
352        let top_least = if open(DockSide::Top) { metrics.top.0 + metrics.bar } else { 0.0 };
353        dock_size(layout, DockSide::Bottom, height, top_least, center_h, metrics)
354    });
355    let top_size = open(DockSide::Top).then(|| {
356        let taken = bottom_size.map_or(0.0, |(bottom, _)| bottom + metrics.bar);
357        dock_size(layout, DockSide::Top, height, taken, center_h, metrics)
358    });
359    if let (Some((bottom, most)), Some(node)) = (bottom_size, layout.node(DockSide::Bottom)) {
360        y1 -= bottom;
361        place_node(node, Rect::new(x0, y1, middle, bottom), DockSide::Bottom, &mut path, metrics, &mut out);
362        y1 -= metrics.bar;
363        let (min, _) = metrics.bottom;
364        out.bars.push(BarPlace {
365            kind: BarKind::Dock(DockSide::Bottom),
366            axis: Axis::Vertical,
367            rect: Rect::new(x0, y1, middle, metrics.bar),
368            value: bottom,
369            min,
370            max: most,
371        });
372    } else if !open(DockSide::Bottom) {
373        let (min, max) = metrics.bottom;
374        let ghost = layout.size(DockSide::Bottom).clamp(min, max).min(height / 2.0);
375        out.ghosts.push((DockSide::Bottom, Rect::new(x0, y1 - ghost, middle, ghost)));
376    }
377    if let (Some((top, most)), Some(node)) = (top_size, layout.node(DockSide::Top)) {
378        place_node(node, Rect::new(x0, y0, middle, top), DockSide::Top, &mut path, metrics, &mut out);
379        y0 += top;
380        let (min, _) = metrics.top;
381        out.bars.push(BarPlace {
382            kind: BarKind::Dock(DockSide::Top),
383            axis: Axis::Vertical,
384            rect: Rect::new(x0, y0, middle, metrics.bar),
385            value: top,
386            min,
387            max: most,
388        });
389        y0 += metrics.bar;
390    } else if !open(DockSide::Top) {
391        let (min, max) = metrics.top;
392        let ghost = layout.size(DockSide::Top).clamp(min, max).min(height / 2.0);
393        out.ghosts.push((DockSide::Top, Rect::new(x0, y0, middle, ghost)));
394    }
395    out.center = Rect::new(x0, y0, middle, (y1 - y0).max(0.0));
396    if let Some(node) = layout.node(DockSide::Center) {
397        place_node(node, out.center, DockSide::Center, &mut path, metrics, &mut out);
398    }
399    out
400}

411/// Which edge of `rect` the point counts for, if it is close enough to one.
412fn edge_of(rect: Rect, at: Point) -> Option<DockEdge> {
413    let part = |near: f32, length: f32| if length > 0.0 { near / length } else { 1.0 };
414    let candidates = [
415        (part(at.x - rect.x(), rect.width()), DockEdge::Left),
416        (part(rect.max_x() - at.x, rect.width()), DockEdge::Right),
417        (part(at.y - rect.y(), rect.height()), DockEdge::Top),
418        (part(rect.max_y() - at.y, rect.height()), DockEdge::Bottom),
419    ];
420    candidates
421        .into_iter()
422        .filter(|(near, _)| *near < EDGE_PART)
423        .min_by(|a, b| a.0.total_cmp(&b.0))
424        .map(|(_, edge)| edge)
425}

427/// The half of `rect` a split on `edge` would take.
428fn half(rect: Rect, edge: DockEdge) -> Rect {
429    let (w, h) = (rect.width() / 2.0, rect.height() / 2.0);
430    match edge {
431        DockEdge::Left => Rect::new(rect.x(), rect.y(), w, rect.height()),
432        DockEdge::Right => Rect::new(rect.x() + w, rect.y(), w, rect.height()),
433        DockEdge::Top => Rect::new(rect.x(), rect.y(), rect.width(), h),
434        DockEdge::Bottom => Rect::new(rect.x(), rect.y() + h, rect.width(), h),
435    }
436}

438/// What lies under the pointer at `at` for the panel `dragged`. `strip` says how tall the tabs of
439/// a group are, and `tab` where a tab is on screen now, relative to the area. It finds a place
440/// whether or not dropping there changes anything; the caller tries that.
441#[doc(hidden)]
442pub fn aim(
443    arranged: &Arranged,
444    at: Point,
445    strip: &dyn Fn(&GroupPlace) -> f32,
446    tab: &dyn Fn(&str) -> Option<Rect>,
447) -> Option<DropAim> {
448    let size = arranged.size;
449    if at.x < 0.0 || at.y < 0.0 || at.x > size.width || at.y > size.height {
450        return None;
451    }
452    // A dock that does not show opens from the border it belongs to.
453    for (side, rect) in &arranged.ghosts {
454        let near = match side {
455            DockSide::Left => at.x < BORDER_ZONE,
456            DockSide::Right => at.x > size.width - BORDER_ZONE,
457            DockSide::Bottom => at.y > size.height - BORDER_ZONE,
458            DockSide::Top => at.y < TOP_ZONE && at.x >= rect.x() && at.x <= rect.max_x(),
459            _ => false,
460        };
461        if near {
462            return Some(DropAim { target: DockDrop::Side(*side), mark: *rect });
463        }
464    }
465    let group = arranged.groups.iter().find(|group| group.rect.contains(at));
466    let Some(group) = group else {
467        return arranged
468            .center
469            .contains(at)
470            .then_some(DropAim { target: DockDrop::Side(DockSide::Center), mark: arranged.center });
471    };
472    let first = group.panels.first()?;
473    let height = strip(group);
474    if at.y < group.rect.y() + height {
475        return Some(aim_tab(group, first, at, height, tab));
476    }
477    let body = Rect::new(
478        group.rect.x(),
479        group.rect.y() + height,
480        group.rect.width(),
481        (group.rect.height() - height).max(0.0),
482    );
483    Some(match edge_of(body, at) {
484        Some(edge) => {
485            DropAim { target: DockDrop::Split { beside: first.clone(), edge }, mark: half(body, edge) }
486        }
487        None => {
488            DropAim { target: DockDrop::Tab { beside: first.clone(), at: group.panels.len() }, mark: body }
489        }
490    })
491}

493/// What a floating panel dragged by its header would do if let go here. Only the tabs of a group
494/// and the borders of the docks that do not show take it, so that it passes over the groups
495/// without landing in them; with `splits`, everything that takes a tab takes it.
496#[doc(hidden)]
497pub fn float_aim(
498    arranged: &Arranged,
499    at: Point,
500    strip: &dyn Fn(&GroupPlace) -> f32,
501    tab: &dyn Fn(&str) -> Option<Rect>,
502    splits: bool,
503) -> Option<DropAim> {
504    let found = aim(arranged, at, strip, tab)?;
505    if splits {
506        return Some(found);
507    }
508    let on_tabs =
509        arranged.groups.iter().any(|group| group.rect.contains(at) && at.y < group.rect.y() + strip(group));
510    (on_tabs || matches!(found.target, DockDrop::Side(_))).then_some(found)
511}

513/// Where a floating panel shows in an area of `size`: moved to fit, and made smaller only when it
514/// cannot fit at all, never below `least` while the area has room for that.
515#[doc(hidden)]
516pub fn clamp_floating(rect: Rect, size: Size, least: Size) -> Rect {
517    let room = |want: f32, least: f32, room: f32| want.max(least.min(room)).min(room.max(0.0));
518    let width = room(rect.width(), least.width, size.width);
519    let height = room(rect.height(), least.height, size.height);
520    let x = rect.x().clamp(0.0, (size.width - width).max(0.0));
521    let y = rect.y().clamp(0.0, (size.height - height).max(0.0));
522    Rect::new(x, y, width, height)
523}

525/// A floating panel's box drawn to the edges near it: of the area, and of the other floating
526/// panels, when one is within `reach`, with the lines where it touches them, to mark.
527#[doc(hidden)]
528pub fn snap_floating(rect: Rect, size: Size, others: &[Rect], reach: f32) -> (Rect, Vec<Rect>) {
529    // Each line it could go to: where its edge would be, and the span to mark there.
530    let (width, height) = (rect.width(), rect.height());
531    let mut xs: Vec<(f32, f32, f32, f32)> =
532        vec![(0.0, 0.0, 0.0, size.height), (size.width - width, size.width, 0.0, size.height)];
533    let mut ys: Vec<(f32, f32, f32, f32)> =
534        vec![(0.0, 0.0, 0.0, size.width), (size.height - height, size.height, 0.0, size.width)];
535    for other in others {
536        let (top, bottom) = (other.y(), other.max_y());
537        let (left, right) = (other.x(), other.max_x());
538        xs.extend([
539            (other.max_x(), other.max_x(), top, bottom),
540            (other.x() - width, other.x(), top, bottom),
541            (other.x(), other.x(), top, bottom),
542            (other.max_x() - width, other.max_x(), top, bottom),
543        ]);
544        ys.extend([
545            (other.max_y(), other.max_y(), left, right),
546            (other.y() - height, other.y(), left, right),
547            (other.y(), other.y(), left, right),
548            (other.max_y() - height, other.max_y(), left, right),
549        ]);
550    }
551    let nearest = |at: f32, lines: &[(f32, f32, f32, f32)]| {
552        lines
553            .iter()
554            .copied()
555            .filter(|line| (line.0 - at).abs() <= reach)
556            .min_by(|a, b| (a.0 - at).abs().total_cmp(&(b.0 - at).abs()))
557    };
558    let x = nearest(rect.x(), &xs);
559    let y = nearest(rect.y(), &ys);
560    let snapped = Rect::new(x.map_or(rect.x(), |l| l.0), y.map_or(rect.y(), |l| l.0), width, height);
561    let mut lines = Vec::new();
562    if let Some((_, at, from, to)) = x {
563        let (from, to) = (from.min(snapped.y()), to.max(snapped.max_y()));
564        lines.push(Rect::new(at - 1.0, from, 2.0, to - from));
565    }
566    if let Some((_, at, from, to)) = y {
567        let (from, to) = (from.min(snapped.x()), to.max(snapped.max_x()));
568        lines.push(Rect::new(from, at - 1.0, to - from, 2.0));
569    }
570    (snapped, lines)
571}

573fn aim_tab(
574    group: &GroupPlace,
575    first: &str,
576    at: Point,
577    height: f32,
578    tab: &dyn Fn(&str) -> Option<Rect>,
579) -> DropAim {
580    let frames: Vec<Option<Rect>> = group.panels.iter().map(|panel| tab(panel)).collect();
581    let start = frames.first().copied().flatten().map_or(group.rect.x(), |frame| frame.x());
582    let extents: Vec<f32> = frames.iter().map(|frame| frame.map_or(0.0, |f| f.width())).collect();
583    let spacing = match (frames.first().copied().flatten(), frames.get(1).copied().flatten()) {
584        (Some(a), Some(b)) => (b.x() - a.max_x()).max(0.0),
585        _ => 0.0,
586    };
587    let gap = reorder::gap_at(&extents, spacing, at.x - start);
588    let line = frames
589        .get(gap)
590        .copied()
591        .flatten()
592        .map(|frame| frame.x())
593        .or_else(|| frames.last().copied().flatten().map(|frame| frame.max_x()))
594        .unwrap_or(group.rect.x());
595    DropAim {
596        target: DockDrop::Tab { beside: first.to_owned(), at: gap },
597        mark: Rect::new(line - GAP_LINE / 2.0, group.rect.y(), GAP_LINE, height),
598    }
599}
