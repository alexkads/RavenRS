1//! A spatial index over spill areas: which areas cover a cell, and which meet a range, without
2//! looking at every area. Areas are registered in 16 x 16 cell blocks, so a lookup touches one
3//! block (a cell) or the blocks a range covers (a range) rather than every spill on the sheet.
4//! An area that would take more than `BIG_AREA_BLOCKS` blocks (a full-height column spill takes
5//! 65,536) is not registered in blocks but kept in a short list that every lookup tests directly,
6//! so registering costs what the area's size warrants only for small areas.
7
8use crate::base::{CellRef, RangeRef};
9use crate::calc::recalc::{FxHasher, Key};
10use std::hash::BuildHasherDefault;
11
12type HashMap<K, V> = std::collections::HashMap<K, V, BuildHasherDefault<FxHasher>>;
13
14/// Block edge in cells (rows and columns).
15const BLOCK: u32 = 16;
16
17/// Areas spanning more blocks than this are tested directly instead of being registered.
18const BIG_AREA_BLOCKS: u64 = 1024;
19
20/// One indexed area: its rectangle, and whether it covers no cells (an array with no rows or no
21/// columns still meets the ranges its anchor is in, as a 1 x 1 area does for taint tracking).
22#[derive(Clone, Copy)]
23struct Entry {
24    rect: RangeRef,
25    empty: bool,
26}
27
28#[derive(Default)]
29struct SheetAreas {
30    entries: HashMap<CellRef, Entry>,
31    blocks: HashMap<(u32, u32), Vec<CellRef>>,
32    /// Anchors of the areas too big to register in blocks.
33    big: Vec<CellRef>,
34}
35
36/// What covers a cell.
37pub(crate) enum Cover {
38    None,
39    One(CellRef),
40    /// Several areas overlap here: the caller decides which one counts.
41    Many,
42}
43
44#[derive(Default)]
45pub(crate) struct SpillIndex {
46    sheets: HashMap<usize, SheetAreas>,
47}
48
49fn block_span(rect: &RangeRef) -> (u32, u32, u32, u32) {
50    (rect.start.row / BLOCK, rect.end.row / BLOCK, rect.start.col / BLOCK, rect.end.col / BLOCK)
51}
52
53fn block_count(rect: &RangeRef) -> u64 {
54    let (r0, r1, c0, c1) = block_span(rect);
55    u64::from(r1 - r0 + 1) * u64::from(c1 - c0 + 1)
56}
57
58/// Whether registering `areas` in blocks costs less than checking each of them against each of
59/// `lookups` cells would: a few block entries cost about one containment check each. A tall or
60/// wide area is not registered at all (see `BIG_AREA_BLOCKS`) and so costs nothing here.
61pub(crate) fn indexing_pays<'a>(areas: impl ExactSizeIterator<Item = &'a RangeRef>, lookups: usize) -> bool {
62    const BLOCK_COST: u64 = 16;
63    let budget = (lookups as u64).saturating_mul(areas.len() as u64) / BLOCK_COST;
64    let mut blocks = 0u64;
65    for r in areas {
66        let n = block_count(r);
67        if n > BIG_AREA_BLOCKS {
68            continue;
69        }
70        blocks = blocks.saturating_add(n);
71        if blocks > budget {
72            return false;
73        }
74    }
75    true
76}

78impl SpillIndex {
79    /// Whether any area is indexed on `sheet`.
80    #[cfg(test)]
81    pub(crate) fn has_sheet(&self, sheet: usize) -> bool {
82        self.sheets.get(&sheet).is_some_and(|s| !s.entries.is_empty())
83    }
84
85    /// Indexes the area an array of `rows` x `cols` anchored at `anchor` covers (replacing the
86    /// one already indexed for that anchor).
87    pub(crate) fn insert(&mut self, (sheet, anchor): Key, rows: usize, cols: usize) {
88        let end =
89            CellRef::new(anchor.row.saturating_add((rows as u32).saturating_sub(1)), anchor.col.saturating_add((cols as u32).saturating_sub(1)));
90        self.insert_rect((sheet, anchor), RangeRef::new(anchor, end), rows == 0 || cols == 0);
91    }
92
93    /// Indexes an area given as a range.
94    pub(crate) fn insert_rect(&mut self, (sheet, anchor): Key, rect: RangeRef, empty: bool) {
95        self.remove((sheet, anchor));
96        let areas = self.sheets.entry(sheet).or_default();
97        if block_count(&rect) > BIG_AREA_BLOCKS {
98            areas.big.push(anchor);
99        } else {
100            let (r0, r1, c0, c1) = block_span(&rect);
101            for br in r0..=r1 {
102                for bc in c0..=c1 {
103                    areas.blocks.entry((br, bc)).or_default().push(anchor);
104                }
105            }
106        }
107        areas.entries.insert(anchor, Entry { rect, empty });
108    }
109
110    pub(crate) fn remove(&mut self, (sheet, anchor): Key) {
111        let Some(areas) = self.sheets.get_mut(&sheet) else { return };
112        let Some(old) = areas.entries.remove(&anchor) else { return };
113        if block_count(&old.rect) > BIG_AREA_BLOCKS {
114            areas.big.retain(|a| *a != anchor);
115            return;
116        }
117        let (r0, r1, c0, c1) = block_span(&old.rect);
118        for br in r0..=r1 {
119            for bc in c0..=c1 {
120                if let Some(list) = areas.blocks.get_mut(&(br, bc)) {
121                    list.retain(|a| *a != anchor);
122                    if list.is_empty() {
123                        areas.blocks.remove(&(br, bc));
124                    }
125                }
126            }
127        }
128    }
129
130    /// The areas that cover `c`.
131    pub(crate) fn covering(&self, sheet: usize, c: CellRef) -> Cover {
132        let Some(areas) = self.sheets.get(&sheet) else { return Cover::None };
133        let in_block = areas.blocks.get(&(c.row / BLOCK, c.col / BLOCK)).map_or(&[][..], |l| l.as_slice());
134        let mut found = Cover::None;
135        for a in in_block.iter().chain(&areas.big) {
136            if let Some(e) = areas.entries.get(a)
137                && !e.empty
138                && e.rect.contains(c)
139            {
140                found = match found {
141                    Cover::None => Cover::One(*a),
142                    _ => return Cover::Many,
143                };
144            }
145        }
146        found
147    }
148
149    /// The anchors of the areas whose rectangle meets `range` (an empty area counts through its
150    /// anchor cell), sorted by anchor.
151    pub(crate) fn meeting(&self, sheet: usize, range: RangeRef, out: &mut Vec<CellRef>) {
152        out.clear();
153        let Some(areas) = self.sheets.get(&sheet) else { return };
154        let (r0, r1, c0, c1) = block_span(&range);
155        let touched = u64::from(r1 - r0 + 1) * u64::from(c1 - c0 + 1);
156        if touched > 4 * areas.entries.len() as u64 + 16 {
157            // A range wider than the areas are many: look at each area once.
158            out.extend(areas.entries.iter().filter(|(_, e)| e.rect.intersects(&range)).map(|(a, _)| *a));
159        } else {
160            for br in r0..=r1 {
161                for bc in c0..=c1 {
162                    if let Some(list) = areas.blocks.get(&(br, bc)) {
163                        out.extend(list.iter().copied().filter(|a| areas.entries.get(a).is_some_and(|e| e.rect.intersects(&range))));
164                    }
165                }
166            }
167            out.extend(areas.big.iter().copied().filter(|a| areas.entries.get(a).is_some_and(|e| e.rect.intersects(&range))));
168        }
169        out.sort_unstable();
170        out.dedup();
171    }
172
173    /// The rectangle indexed for `anchor` and whether it covers no cells.
174    pub(crate) fn area(&self, sheet: usize, anchor: CellRef) -> Option<(RangeRef, bool)> {
175        self.sheets.get(&sheet)?.entries.get(&anchor).map(|e| (e.rect, e.empty))
176    }
177}

187    #[test]
188    fn finds_the_areas_covering_a_cell_and_meeting_a_range() {
189        let mut ix = SpillIndex::default();
190        ix.insert((0, at(0, 0)), 40, 3);
191        ix.insert((0, at(30, 1)), 1, 40);
192        ix.insert((1, at(0, 0)), 5, 5);
193        assert!(matches!(ix.covering(0, at(39, 2)), Cover::One(a) if a == at(0, 0)));
194        assert!(matches!(ix.covering(0, at(30, 2)), Cover::Many));
195        assert!(matches!(ix.covering(0, at(40, 0)), Cover::None));
196        assert!(matches!(ix.covering(2, at(0, 0)), Cover::None));
197        let mut hits = Vec::new();
198        ix.meeting(0, RangeRef::new(at(30, 30), at(50, 60)), &mut hits);
199        assert_eq!(hits, vec![at(30, 1)]);
200        ix.meeting(0, RangeRef::new(at(0, 0), at(1_000_000, 16_000)), &mut hits);
201        assert_eq!(hits, vec![at(0, 0), at(30, 1)]);
202        ix.remove((0, at(30, 1)));
203        assert!(matches!(ix.covering(0, at(30, 2)), Cover::One(_)));
204        assert!(ix.has_sheet(1) && !ix.has_sheet(2));
205        ix.insert((0, at(0, 0)), 1, 1);
206        assert!(matches!(ix.covering(0, at(39, 2)), Cover::None));
207    }

209    #[test]
210    fn tall_or_wide_areas_are_not_registered_in_blocks() {
211        let column = RangeRef::new(at(0, 0), at(1_048_575, 0));
212        assert!(block_count(&column) > BIG_AREA_BLOCKS);
213        let mut ix = SpillIndex::default();
214        ix.insert_rect((0, at(0, 0)), column, false);
215        assert!(ix.sheets[&0].blocks.is_empty());
216        assert!(matches!(ix.covering(0, at(900_000, 0)), Cover::One(a) if a == at(0, 0)));
217        assert!(matches!(ix.covering(0, at(900_000, 1)), Cover::None));
218        let mut hits = Vec::new();
219        ix.meeting(0, RangeRef::new(at(5, 0), at(6, 3)), &mut hits);
220        assert_eq!(hits, vec![at(0, 0)]);
221        ix.remove((0, at(0, 0)));
222        assert!(matches!(ix.covering(0, at(900_000, 0)), Cover::None) && ix.sheets[&0].big.is_empty());
223        // Registering a big area costs nothing (lookups test it directly), so a small paste still indexes next to many small areas.
224        assert!(indexing_pays([column].iter(), 17));
225        // Many small areas pay back for a paste of a hundred cells, but not for a handful.
226        let small: Vec<RangeRef> = (0..1_000).map(|i| RangeRef::new(at(i * 3, 0), at(i * 3 + 2, 0))).collect();
227        assert!(indexing_pays(small.iter(), 100));
228        assert!(!indexing_pays(small.iter(), 17));
229    }

231    /// The index against the old way: test every area for every lookup. Mixed shapes: small,
232    /// thin and tall, wide, big (kept out of the blocks), empty, on two sheets, replaced and removed.
233    #[test]
234    fn matches_a_scan_of_every_area_on_mixed_shapes() {
235        let mut seed = 0x2545_F491_4F6C_DD1Du64;
236        let mut next = |n: u32| {
237            seed ^= seed << 13;
238            seed ^= seed >> 7;
239            seed ^= seed << 17;
240            (seed >> 11) as u32 % n
241        };
242        let mut ix = SpillIndex::default();
243        let mut model: std::collections::BTreeMap<(usize, CellRef), (RangeRef, bool)> = Default::default();
244        for step in 0..3_000 {
245            let sheet = next(2) as usize;
246            let anchor = at(next(400) * 5, next(40) * 3);
247            if next(4) == 0 {
248                ix.remove((sheet, anchor));
249                model.remove(&(sheet, anchor));
250            } else {
251                let (rows, cols) = match next(8) {
252                    0 => (0, 1 + next(5) as usize),
253                    1 => (1 + next(5) as usize, 0),
254                    2 => (5_000 + next(100_000) as usize, 1),
255                    3 => (1, 1 + next(40_000) as usize),
256                    4 => (200 + next(300) as usize, 40 + next(40) as usize),
257                    _ => (1 + next(60) as usize, 1 + next(40) as usize),
258                };
259                ix.insert((sheet, anchor), rows, cols);
260                let end = at(anchor.row + (rows as u32).saturating_sub(1), anchor.col + (cols as u32).saturating_sub(1));
261                model.insert((sheet, anchor), (RangeRef::new(anchor, end), rows == 0 || cols == 0));
262            }
263            if step % 7 != 0 {
264                continue;
265            }
266            let sheet = next(3) as usize;
267            for _ in 0..20 {
268                let cell = at(next(2_100), next(130));
269                let want: Vec<CellRef> =
270                    model.iter().filter(|((s, _), (r, e))| *s == sheet && !e && r.contains(cell)).map(|((_, a), _)| *a).collect();
271                match ix.covering(sheet, cell) {
272                    Cover::None => assert!(want.is_empty(), "{cell:?}"),
273                    Cover::One(a) => assert_eq!(want, vec![a], "{cell:?}"),
274                    Cover::Many => assert!(want.len() > 1, "{cell:?}"),
275                }
276                let (r0, c0) = (next(2_100), next(130));
277                let (tall, wide) = (if next(5) == 0 { 3_000 } else { 40 }, if next(5) == 0 { 130 } else { 20 });
278                let range = RangeRef::new(at(r0, c0), at(r0 + next(tall), c0 + next(wide)));
279                let want: Vec<CellRef> = model.iter().filter(|((s, _), (r, _))| *s == sheet && r.intersects(&range)).map(|((_, a), _)| *a).collect();
280                let mut got = Vec::new();
281                ix.meeting(sheet, range, &mut got);
282                assert_eq!(got, want, "{range:?}");
283            }
284        }
285    }
