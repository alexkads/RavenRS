6/// Formulas a formula may calculate on demand one inside the other (each needing one another's
7/// value before it is done). Each level takes stack, so a chain deeper than this is suspended and
8/// calculated from the top instead (see [`PassHost::calculate`]): long dependency chains calculate
9/// whatever order they come in, with bounded stack.
10const MAX_NESTED: usize = 64;
11
12/// Formula evaluations iterative calculation may spend on one recalculation (passes × formulas
13/// evaluated again), however high Maximum Iterations is.
14const MAX_ITERATION_WORK: usize = 10_000_000;
15
16pub(super) struct PassHost<'a> {
17    wb: &'a Workbook,
18    exprs: &'a HashMap<Key, std::sync::Arc<Expr>>,
19    pub(super) results: HashMap<Key, Value>,
20    pending: HashSet<Key>,
21    in_progress: HashSet<Key>,
22    /// New spills found during the pass: anchor → array.
23    pub(super) spills: HashMap<Key, std::sync::Arc<Array>>,
24    /// Used range per sheet: the workbook doesn't change during a pass, and working it out walks
25    /// every row, so once per sheet rather than once per range read.
26    used: HashMap<usize, Option<RangeRef>>,
27    rng: &'a mut u64,
28    now: f64,
29    cycle: bool,
30    pub(super) cycles: Vec<Key>,
31    depth: usize,
32    /// A formula needed at [`MAX_NESTED`] levels deep: the evaluations in progress are abandoned
33    /// and it is calculated first.
34    suspend: Option<Key>,
35    /// Formulas whose evaluation was abandoned for another one, to be calculated again after it.
36    /// They all need it, so reading one of them while it is calculated is a circular reference,
37    /// as reading a formula in progress is.

39    /// The formulas abandoned since the last suspension.
40    abandoned: Vec<Key>,
41    /// Other formulas read at [`MAX_NESTED`] levels deep by the evaluations being abandoned. They

43    /// read at the limit would restart the abandoned evaluations once per cell in it).
44    also: Vec<Key>,
45    /// Suspensions so far (tests check that a wide read at the limit needs one, not one per cell).
46    #[cfg(test)]
47    pub(super) suspensions: usize,
48    /// Spilled values looked at by range reads (tests check that a read outside a large spill
49    /// looks at none of it).
50    #[cfg(test)]
51    pub(super) spill_visits: usize,
52    /// The formulas in `cycles`.
53    cycle_seen: HashSet<Key>,
54    /// Iterative calculation: a circular reference reads the cell's previous value instead of

65    /// Whether the formula being evaluated has read a previous value.

126/// Whether an array result at `k` can't spill: it is in a table, or its area runs off the sheet,
127/// overlaps merged cells, holds other cells, or another formula's spill (unless that formula is
128/// recalculated in this pass, and so may spill elsewhere now).
129fn spill_blocked(host: &PassHost<'_>, (sheet, anchor): Key, a: &Array) -> bool {
130    let Some(sh) = host.wb.sheet(sheet) else { return false };
131    let end =
132        CellRef::new(anchor.row.saturating_add((a.rows as u32).saturating_sub(1)), anchor.col.saturating_add((a.cols as u32).saturating_sub(1)));
133    let range = RangeRef::new(anchor, end);
134    let own = sh.spill_ranges.get(&anchor);
135    let spilled_over = |c: CellRef| {
136        !own.is_some_and(|o| o.contains(c))
137            && sh.spill_ranges.iter().any(|(o, r)| {
138                // A formula or Python cell recalculated in this pass lays its array out again.
139                *o != anchor && r.contains(c) && !host.exprs.contains_key(&(sheet, *o)) && !host.python.contains_key(&(sheet, *o))
140            })
141    };
142    end.row >= crate::base::MAX_ROWS
143        || end.col >= crate::base::MAX_COLS
144        || sh.merges.iter().any(|m| m.intersects(&range))
145        || sh.tables.iter().any(|t| t.range.contains(anchor))
146        || sh.cells.iter_range(range).any(|(c, cell)| c != anchor && (!cell.value.is_empty() || cell.formula.is_some()))
147        || sh.spill.range(range.start..=range.end).any(|(c, _)| range.contains(*c) && spilled_over(*c))
148}

150/// Whether `arr`, anchored at `anchor`, would spill exactly what the sheet already holds over
151/// `range`: the same area, nothing new in the way, and the same values.
152pub(super) fn spills_as_before(sheet: &crate::model::Sheet, anchor: CellRef, range: RangeRef, arr: &Array) -> bool {
153    let end =
154        CellRef::new(anchor.row.saturating_add((arr.rows as u32).saturating_sub(1)), anchor.col.saturating_add((arr.cols as u32).saturating_sub(1)));
155    RangeRef::new(anchor, end) == range
156        && !sheet.merges.iter().any(|m| m.intersects(&range))
157        && !sheet.tables.iter().any(|t| t.range.contains(anchor))
158        && range
159            .iter()
160            .enumerate()
161            .all(|(i, c)| c == anchor || (!sheet.cells.has(c) && sheet.spill.get(&c) == Some(arr.data.get(i).unwrap_or(&Value::Empty))))
162}

171/// The spilled values `sheet` stores inside `range`, passed to `f` until it returns false. Values
172/// outside the range are not walked: with fewer rows in the range than spilled values, each row
173/// is looked up on its own; otherwise one ordered scan visits each spilled value at most once.
174/// Returns how many values were looked at.
175fn stored_spill_in<'s>(sheet: &'s crate::model::Sheet, range: RangeRef, f: &mut dyn FnMut(CellRef, &'s Value) -> bool) -> usize {
176    let mut seen = 0usize;
177    if sheet.spill.is_empty() {
178        return 0;
179    }
180    if (range.height() as usize) < sheet.spill.len() {
181        for row in range.start.row..=range.end.row {
182            for (c, v) in sheet.spill.range(CellRef::new(row, range.start.col)..=CellRef::new(row, range.end.col)) {
183                seen += 1;
184                if !f(*c, v) {
185                    return seen;
186                }
187            }
188        }
189    } else {
190        for (c, v) in sheet.spill.range(range.start..=range.end) {
191            seen += 1;
192            if range.contains(*c) && !f(*c, v) {
193                return seen;
194            }
195        }
196    }
197    seen
198}

200impl PassHost<'_> {
201    /// Calculates `k` (and the formulas it reads) with bounded stack: when [`Self::compute`] would
202    /// go deeper than [`MAX_NESTED`], it suspends what is in progress, the formula it needed is
203    /// calculated from here, then the suspended ones again (reading the results kept so far).
204    pub(super) fn calculate(&mut self, k: Key) {
205        // Formulas to calculate, innermost last, each with the ones abandoned to calculate the
206        // next.
207        let mut work: Vec<(Key, Vec<Key>)> = vec![(k, Vec::new())];
208        while let Some(&(top, _)) = work.last() {
209            self.compute(top);
210            match self.suspend.take() {
211                Some(next) => {
212                    #[cfg(test)]
213                    {
214                        self.suspensions += 1;
215                    }
216                    let held = std::mem::take(&mut self.abandoned);
217                    self.waiting.extend(held.iter().copied());
218                    if let Some(entry) = work.last_mut() {
219                        entry.1 = held;
220                    }
221                    // The other formulas read at the limit are calculated after `next`, before the
222                    // held evaluations restart. Each has nothing held of its own, so the held ones
223                    // are released only when the last of them is done.
224                    let mut seen: HashSet<Key> = HashSet::default();
225                    seen.insert(next);
226                    for s in std::mem::take(&mut self.also) {
227                        if seen.insert(s) {
228                            work.push((s, Vec::new()));
229                        }
230                    }
231                    work.push((next, Vec::new()));
232                }
233                None => {
234                    work.pop();
235                    if let Some(entry) = work.last_mut() {
236                        for a in entry.1.drain(..) {
237                            self.waiting.remove(&a);
238                        }
239                    }
240                }
241            }
242        }
243    }

245    fn compute(&mut self, k: Key) -> Value {
246        if let Some(v) = self.results.get(&k) {
247            let v = v.clone();
248            if self.iterate && self.tainted.contains(&k) {
249                self.taint = true;
250            }
251            return v;
252        }
253        if let Some(result) = self.python.get(&k).cloned() {
254            return self.python_result(k, result);
255        }
256        let Some(expr) = self.exprs.get(&k) else {
257            return self.wb.sheet(k.0).map(|s| s.value(k.1)).unwrap_or_default();
258        };
259        if self.suspend.is_some() {
260            // Unwinding to `calculate`: the value is thrown away. A formula read at the limit is
261            // calculated with the suspended one, so the evaluations are not restarted for each.
262            if self.depth >= MAX_NESTED && !self.in_progress.contains(&k) && !self.waiting.contains(&k) {
263                self.also.push(k);
264            }
265            return Value::Error(CellError::Calc);
266        }
267        if self.in_progress.contains(&k) || self.waiting.contains(&k) {
268            self.cycle = true;
269            if self.iterate {
270                return self.circular_read(k);
271            }
272            // Once each, though a suspended evaluation meets it again.
273            if self.cycle_seen.insert(k) {
274                self.cycles.push(k);
275            }
276            return Value::Error(CellError::Circ);
277        }
278        if self.depth >= MAX_NESTED {
279            self.suspend = Some(k);
280            return Value::Error(CellError::Calc);
281        }
282        if self.depth == 0 {
283            self.stack_top = stack_position();
284        }
285        self.in_progress.insert(k);
286        self.depth += 1;
287        let outer_taint = std::mem::replace(&mut self.taint, false);
288        if self.iterate {
289            self.stack.push(k);
290        }
291        let expr = std::sync::Arc::clone(expr);
292        let outer_out_of_stack = std::mem::replace(&mut self.out_of_stack, false);
293        let v = {
294            let mut ev = Evaluator::new(self, k.0, k.1);
295            ev.value(&expr)
296        };
297        let out_of_stack = std::mem::replace(&mut self.out_of_stack, outer_out_of_stack);
298        if out_of_stack && self.depth > 1 && self.suspend.is_none() {
299            // It ran out of the stack left below the formulas that read it: it is calculated
300            // from the top instead, as a formula needed at `MAX_NESTED` levels deep is.
301            self.suspend = Some(k);
302            self.depth -= 1;
303            self.in_progress.remove(&k);
304            if self.iterate {
305                self.stack.pop();
306            }
307            self.taint = outer_taint;
308            return Value::Error(CellError::Calc);
309        }
310        if self.suspend.is_some() {
311            // Abandoned: `calculate` evaluates this formula again once the one it needs is done.
312            self.depth -= 1;
313            self.in_progress.remove(&k);
314            self.abandoned.push(k);
315            if self.iterate {
316                self.stack.pop();
317            }
318            self.taint = outer_taint;
319            return Value::Error(CellError::Calc);
320        }
321        // A legacy array formula fills the range it was entered in.
322        let legacy = self.wb.sheet(k.0).and_then(|s| s.cell(k.1)).and_then(|c| c.formula.as_ref()).and_then(|f| f.array);
323        let v = match legacy {
324            Some(range) => fit_to_range(v, range),
325            None => v,
326        };
327        self.depth -= 1;
328        self.in_progress.remove(&k);
329        self.pending.remove(&k);
330        if self.iterate {
331            self.stack.pop();
332            if self.taint {
333                self.tainted.insert(k);
334            } else {
335                self.tainted.remove(&k);
336            }
337        }
338        self.taint |= outer_taint;
339        // A blank cell shows as 0 in a formula's result, inside an array as well (`=A1:A3`,
340        // FILTER or SORT of a range with blanks). GROUPBY and PIVOTBY lay out blank cells
341        // of their own, which stay blank.
342        let keep_blanks = matches!(&*expr, Expr::Call(n, _) if matches!(n.as_str(), "GROUPBY" | "PIVOTBY"));
343        let v = match v {
344            Value::Array(a) => {
345                let a = if keep_blanks || !a.data.iter().any(|x| matches!(x, Value::Empty)) {
346                    a
347                } else {
348                    let data = a.data.iter().map(|x| if matches!(x, Value::Empty) { Value::Number(0.0) } else { x.clone() }).collect();
349                    Array::new(a.rows, a.cols, data).map(std::sync::Arc::new).unwrap_or(a)
350                };
351                // Precision as displayed: each element takes the format of the cell it lands in.
352                let a = match self.precision.as_mut() {
353                    Some(p) if a.data.iter().any(|x| matches!(x, Value::Number(_))) => {
354                        let cols = a.cols.max(1);
355                        let data = a
356                            .data
357                            .iter()
358                            .enumerate()
359                            .map(|(i, x)| {
360                                let at = CellRef::new(k.1.row.saturating_add((i / cols) as u32), k.1.col.saturating_add((i % cols) as u32));
361                                p.round_value(self.wb, k.0, at, x.clone())
362                            })
363                            .collect();
364                        Array::new(a.rows, a.cols, data).map(std::sync::Arc::new).unwrap_or(a)
365                    }
366                    _ => a,
367                };
368                // A dynamic-array formula anchors a spill range even while its result is one
369                // value -- a FILTER matching one row, a SORT or UNIQUE over one item -- so `C1#`
370                // is the 1 x 1 range and `ROWS(C1#)` is 1, as in Excel (#1491). A formula that
371                // returns a scalar never reaches this arm, so it does not become an anchor.
372                self.spills.insert(k, a.clone());
373                a.data.first().cloned().unwrap_or_default()
374            }
375            Value::Empty => Value::Number(0.0),
376            v @ Value::Number(_) => match self.precision.as_mut() {
377                Some(p) => p.round_value(self.wb, k.0, k.1, v),
378                None => v,
379            },
380            v => v,
381        };
382        // An array that can't spill is #SPILL! for the formulas evaluated after it in this pass too.
383        let v = match self.spills.get(&k) {
384            Some(a) if spill_blocked(self, k, a) => {
385                self.spills.remove(&k);
386                Value::Error(CellError::Spill)
387            }
388            _ => v,
389        };
390        self.results.insert(k, v.clone());
391        v
392    }

394    /// A Python cell's kept result as its value: an array spills from the cell unless it is
395    /// blocked, malformed or larger than [`crate::model::python::MAX_RESULT_CELLS`] (only a
396    /// crafted file holds one), which shows `#SPILL!`.
397    fn python_result(&mut self, k: Key, result: Value) -> Value {
398        self.pending.remove(&k);
399        let v = match result {
400            Value::Array(a) => {
401                let sane = a.rows >= 1
402                    && a.cols >= 1
403                    && a.rows.checked_mul(a.cols).is_some_and(|n| n == a.data.len() && n <= crate::model::python::MAX_RESULT_CELLS);
404                if sane {
405                    self.spills.insert(k, a.clone());
406                    a.data.first().cloned().unwrap_or_default()
407                } else {
408                    Value::Error(CellError::Spill)
409                }
410            }
411            v => v,
412        };
413        let v = match self.spills.get(&k) {
414            Some(a) if spill_blocked(self, k, a) => {
415                self.spills.remove(&k);
416                Value::Error(CellError::Spill)
417            }
418            _ => v,
419        };
420        self.results.insert(k, v.clone());
421        v
422    }

424    /// A formula's previous value for iterative calculation: from the last iteration, else the
425    /// value it shows. Blank, and the `#CIRC!` of a loop calculated with iteration off, are 0.
426    fn previous(&self, k: Key) -> Value {
427        let v = match self.prev.get(&k) {
428            Some(v) => v.clone(),
429            None => self.wb.sheet(k.0).map(|s| s.value(k.1)).unwrap_or_default(),
430        };
431        if matches!(v, Value::Empty | Value::Error(CellError::Circ)) { Value::Number(0.0) } else { v }
432    }

434    /// Reading `k` while it is being evaluated, with iterative calculation on: its previous value.
435    /// The formulas around the loop are the ones whose changes decide when iteration stops.
436    fn circular_read(&mut self, k: Key) -> Value {
437        self.taint = true;
438        match self.stack.iter().rposition(|s| *s == k) {
439            Some(i) => self.members.extend(self.stack.get(i..).unwrap_or_default().iter().copied()),
440            // Too deep to follow: the formula itself is watched.
441            None => {
442                self.members.insert(k);
443            }
444        }
445        self.previous(k)
446    }

448    /// Iterative calculation, after a first pass over `order` that met circular references. The
449    /// formulas that read a previous value (the loops and what depends on them) are evaluated
450    /// again in the same order, each pass starting from the last one's values and reading what it
451    /// has already produced (Gauss-Seidel), until no formula on a loop changes by `max_change` or
452    /// more, or after `max_iterations` passes in all. Returns the number of passes.
453    pub(super) fn iterate_loops(&mut self, order: &[Key], max_iterations: u32, max_change: f64) -> u32 {
454        let again: Vec<Key> = order.iter().copied().filter(|k| self.tainted.contains(k)).collect();
455        let set: HashSet<Key> = again.iter().copied().collect();
456        // Bounds the work whatever Maximum Iterations says.
457        let max_passes = u32::try_from(MAX_ITERATION_WORK / again.len().max(1)).unwrap_or(u32::MAX).max(1);
458        let mut passes = 1u32;
459        while passes < max_iterations.min(max_passes) && !self.converged(max_change) {
460            self.prev = again.iter().filter_map(|k| self.results.remove(k).map(|v| (*k, v))).collect();
461            self.pending = set.clone();
462            self.spills.retain(|a, _| !set.contains(a));
463            for k in &again {
464                self.calculate(*k);
465            }
466            passes += 1;
467        }
468        passes
469    }
470
471    /// Whether every formula on a loop changed by less than `max_change` in the last pass (text,
472    /// logical and error values: not at all).
473    fn converged(&self, max_change: f64) -> bool {
474        self.members.iter().all(|k| {
475            let Some(new) = self.results.get(k) else { return true };
476            match (new, &self.previous(*k)) {
477                (Value::Number(a), Value::Number(b)) => {
478                    let d = (a - b).abs();
479                    d < max_change || d == 0.0
480                }
481                (new, old) => new == old,
482            }
483        })
484    }
485}

487/// A legacy (Ctrl+Shift+Enter) array formula's result fitted to the range it was entered in, as
488/// Excel does: a single row or column repeats across the range, a single value fills it, cells
489/// past the result show `#N/A`, and a result larger than the range is cut off.
490fn fit_to_range(v: Value, range: RangeRef) -> Value {
491    if range.count() > crate::calc::eval::MAX_CELLS {
492        return Value::Error(CellError::Num);
493    }
494    let (h, w) = (range.height() as usize, range.width() as usize);
495    let Value::Array(a) = v else {
496        return if h * w == 1 { v } else { Array::new(h, w, vec![v; h * w]).map(Value::from).unwrap_or(Value::Error(CellError::Value)) };
497    };
498    if h * w == 1 {
499        return a.data.first().cloned().unwrap_or_default();
500    }
501    let mut data = Vec::with_capacity(h * w);
502    for r in 0..h {
503        for c in 0..w {
504            let (r, c) = (if a.rows == 1 { 0 } else { r }, if a.cols == 1 { 0 } else { c });
505            data.push(a.get(r, c).cloned().unwrap_or(Value::Error(CellError::NA)));
506        }
507    }
508    Array::new(h, w, data).map(Value::from).unwrap_or(Value::Error(CellError::Value))
509}

511impl Host for PassHost<'_> {
512    fn workbook(&self) -> &Workbook {
513        self.wb
514    }
515    fn stack_top(&self) -> Option<usize> {
516        (self.depth > 1).then_some(self.stack_top)
517    }
518    fn stack_exhausted(&mut self) {
519        self.out_of_stack = true;
520    }
521    fn spill_range(&mut self, sheet: usize, anchor: CellRef) -> Option<RangeRef> {
522        let k = (sheet, anchor);
523        if !(self.pending.contains(&k) || self.results.contains_key(&k)) {
524            return self.wb.sheet(sheet)?.spill_ranges.get(&anchor).copied();
525        }
526        // Recalculated in this pass: its new array, unless something blocks it (#SPILL!).
527        self.compute(k);
528        let arr = self.spills.get(&k)?;
529        let end = CellRef::new(
530            anchor.row.saturating_add((arr.rows as u32).saturating_sub(1)),
531            anchor.col.saturating_add((arr.cols as u32).saturating_sub(1)),
532        );
533        let range = RangeRef::new(anchor, end);
534        let sh = self.wb.sheet(sheet)?;
535        let blocked = end.row >= crate::base::MAX_ROWS
536            || end.col >= crate::base::MAX_COLS
537            || sh.cells.iter_range(range).any(|(c, cell)| c != anchor && (cell.formula.is_some() || !cell.value.is_empty()))
538            || sh.merges.iter().any(|m| m.intersects(&range));
539        (!blocked).then_some(range)
540    }
541    fn cell_value(&mut self, sheet: usize, c: CellRef) -> Value {
542        let k = (sheet, c);
543        if self.pending.contains(&k) || self.results.contains_key(&k) {
544            return self.compute(k);
545        }
546        // A cell covered by a spill computed in this pass.
547        for (anchor, arr) in &self.spills {
548            if anchor.0 == sheet && c.row >= anchor.1.row && c.col >= anchor.1.col {
549                let (dr, dc) = ((c.row - anchor.1.row) as usize, (c.col - anchor.1.col) as usize);
550                if dr < arr.rows && dc < arr.cols {
551                    if self.iterate && self.tainted.contains(anchor) {
552                        self.taint = true;
553                    }
554                    return arr.get(dr, dc).cloned().unwrap_or_default();
555                }
556            }
557        }
558        self.wb.sheet(sheet).map(|s| s.value(c)).unwrap_or_default()
559    }
560    fn range_values(&mut self, sheet: usize, range: RangeRef) -> Vec<Value> {
561        let (h, w) = (range.height() as usize, range.width() as usize);
562        let mut out = vec![Value::Empty; h * w];
563        let Some(sh) = self.wb.sheet(sheet) else { return out };
564        // Sparse walk: only stored cells; formula and Python cells that are (or were) dirty go
565        // through `compute`, constants are copied straight from the store.
566        let mut formulas: Vec<(usize, CellRef)> = Vec::new();
567        for (c, cell) in sh.cells.iter_range(range) {
568            let i = (c.row - range.start.row) as usize * w + (c.col - range.start.col) as usize;
569            if cell.formula.is_some() || cell.py.is_some() {
570                formulas.push((i, c));
571            } else if let Some(slot) = out.get_mut(i) {
572                *slot = cell.value.clone();
573            }
574        }
575        let _seen = stored_spill_in(sh, range, &mut |c, v| {
576            let i = (c.row - range.start.row) as usize * w + (c.col - range.start.col) as usize;
577            if let Some(slot) = out.get_mut(i)
578                && slot.is_empty()
579            {
580                *slot = v.clone();
581            }
582            true
583        });
584        #[cfg(test)]
585        {
586            self.spill_visits += _seen;
587        }
588        for (i, c) in formulas {
589            let v = self.cell_value(sheet, c);
590            if let Some(slot) = out.get_mut(i) {
591                *slot = v;
592            }
593        }
594        if !self.spills.is_empty() {
595            // Only the part of each array inside the range is walked: a large array beside a small
596            // read costs nothing.
597            let mut new: Vec<(usize, Value)> = Vec::new();
598            for (a, arr) in self.spills.iter().filter(|(a, _)| a.0 == sheet) {
599                let area = array_area(a.1, arr);
600                if !area.intersects(&range) {
601                    continue;
602                }
603                let (r0, r1) = (area.start.row.max(range.start.row), area.end.row.min(range.end.row));
604                let (c0, c1) = (area.start.col.max(range.start.col), area.end.col.min(range.end.col));
605                for row in r0..=r1 {
606                    for col in c0..=c1 {
607                        let c = CellRef::new(row, col);
608                        if c == a.1 {
609                            continue;
610                        }
611                        let v = arr.get((row - a.1.row) as usize, (col - a.1.col) as usize).cloned().unwrap_or_default();
612                        new.push(((row - range.start.row) as usize * w + (col - range.start.col) as usize, v));
613                    }
614                }
615            }
616            #[cfg(test)]
617            {
618                self.spill_visits += new.len();
619            }
620            // Iterating: an array spilled by a formula that read a previous value.
621            if self.iterate
622                && self.spills.iter().any(|(a, arr)| {
623                    let end = CellRef::new(
624                        a.1.row.saturating_add((arr.rows as u32).saturating_sub(1)),
625                        a.1.col.saturating_add((arr.cols as u32).saturating_sub(1)),
626                    );
627                    a.0 == sheet && self.tainted.contains(a) && RangeRef::new(a.1, end).intersects(&range)
628                })
629            {
630                self.taint = true;
631            }
632            for (i, v) in new {
633                if let Some(slot) = out.get_mut(i) {
634                    *slot = v;
635                }
636            }
637        }
638        out
639    }
640    fn visit_range(&mut self, sheet: usize, range: RangeRef, f: &mut dyn FnMut(&Value) -> bool) -> bool {
641        let wb = self.wb;
642        let Some(sh) = wb.sheet(sheet) else { return false };
643        // Formula and Python cells first, in row-major order like `range_values`: reading one
644        // calculates it (a Python cell lays out its result) and the pass remembers the value, so
645        // the walk below gets the same value again without this method holding any.
646        for (c, cell) in sh.cells.iter_range(range) {
647            if cell.formula.is_some() || cell.py.is_some() {
648                let _ = self.cell_value(sheet, c);
649            }
650        }
651        // Spilled values sit between the stored cells (and the formulas just calculated may have
652        // produced some): leave such blocks to `range_values`.
653        let arrays = self.spills.iter().any(|(a, arr)| a.0 == sheet && array_area(a.1, arr).intersects(&range));
654        let mut stored = false;
655        let _seen = stored_spill_in(sh, range, &mut |_, _| {
656            stored = true;
657            false
658        });
659        #[cfg(test)]
660        {
661            self.spill_visits += _seen;
662        }
663        if arrays || stored {
664            return false;
665        }
666        for (c, cell) in sh.cells.iter_range(range) {
667            let go = if cell.formula.is_some() || cell.py.is_some() { f(&self.cell_value(sheet, c)) } else { f(&cell.value) };
668            if !go {
669                break;
670            }
671        }
672        true
673    }
674    fn used_range(&mut self, sheet: usize) -> Option<RangeRef> {
675        let wb = self.wb;
676        *self.used.entry(sheet).or_insert_with(|| wb.sheet(sheet).and_then(|s| s.used_range()))
677    }

680    }
681    fn random(&mut self) -> f64 {
682        // xorshift64*
683        let mut x = *self.rng;
684        x ^= x >> 12;
685        x ^= x << 25;
686        x ^= x >> 27;
687        *self.rng = x;
688        (x.wrapping_mul(0x2545_F491_4F6C_DD1D) >> 11) as f64 / (1u64 << 53) as f64
689    }
