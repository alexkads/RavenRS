8    pub(super) fn call(&mut self, name: &str, args: &[Expr]) -> Ev {
9        match name {
10            "IF" => self.if_call(args),
11            "IFS" => {
12                if args.len() < 2 || !args.len().is_multiple_of(2) {
13                    return err(CellError::Value);
14                }
15                for pair in args.chunks(2) {
16                    let [c, v] = pair else { break };
17                    let cv = self.value(c);
18                    match cv.scalar().to_bool() {
19                        Ok(true) => return self.eval(v),
20                        Ok(false) => {}
21                        Err(e) => return err(e),
22                    }
23                }
24                err(CellError::NA)
25            }
26            "IFERROR" | "IFNA" => self.if_error_call(name, args),
27            "CHOOSE" => self.choose_call(args),
28            "SWITCH" => self.switch_call(args),
29            "ROW" | "COLUMN" => self.row_column_call(name, args),
30            "ROWS" | "COLUMNS" => {
31                if args.len() != 1 {
32                    return err(CellError::Value);
33                }
34                let is_rows = name == "ROWS";
35                match self.eval(&args[0]) {
36                    Ev::R(a) => match a.as_slice() {
37                        [a] => Ev::V(Value::number(if is_rows { a.range.height() } else { a.range.width() } as f64)),
38                        _ => err(CellError::Ref),
39                    },
40                    Ev::V(Value::Array(arr)) => Ev::V(Value::number(if is_rows { arr.rows } else { arr.cols } as f64)),
41                    Ev::V(Value::Error(e)) => err(e),
42                    Ev::V(_) => Ev::V(Value::Number(1.0)),
43                    Ev::L(_) => err(CellError::Value),
44                }
45            }
46            "AREAS" => match args.first().map(|a| self.eval(a)) {
47                Some(Ev::R(a)) => Ev::V(Value::number(a.len() as f64)),
48                Some(Ev::V(Value::Error(e))) => err(e),
49                _ => err(CellError::Value),
50            },
51            "ISREF" => Ev::V(Value::Bool(matches!(args.first().map(|a| self.eval(a)), Some(Ev::R(_))))),
52            "ISFORMULA" | "FORMULATEXT" => {
53                let Some(Ev::R(a)) = args.first().map(|a| self.eval(a)) else { return err(CellError::NA) };
54                let Some(a) = a.first() else { return err(CellError::NA) };
55                let f = self.host.workbook().sheet(a.sheet).and_then(|s| s.cell(a.range.start)).and_then(|c| c.formula.clone());
56                if name == "ISFORMULA" {
57                    Ev::V(Value::Bool(f.is_some()))
58                } else {
59                    match f {
60                        Some(f) => Ev::V(Value::text(format!("={}", f.text))),
61                        None => err(CellError::NA),
62                    }
63                }
64            }
65            "SHEET" | "SHEETS" => self.sheet_call(name, args),
66            "OFFSET" => self.offset(args),
67            "INDIRECT" => self.indirect(args),
68            "INDEX" => self.index(args),
69            "XLOOKUP" => self.xlookup(args),
70            "CELL" => self.cell_info(args),
71            "TRIMRANGE" => self.trimrange(args),
72            "SUBTOTAL" | "AGGREGATE" => self.subtotal(name, args),
73            "TEXT" => {
74                if args.len() != 2 {
75                    return err(CellError::Value);
76                }
77                let v = self.arg_val(args, 0).unwrap_or_default();
78                let f = self.arg_val(args, 1).unwrap_or_default();
79                let sys = self.host.workbook().date_system;
80                let one = |v: &Value, f: &Value| -> Value {
81                    if let Value::Error(e) = v {
82                        return Value::Error(*e);
83                    }
84                    let code = match f.to_text() {
85                        Ok(c) => c,
86                        Err(e) => return Value::Error(e),
87                    };
88                    match crate::numfmt::text_function(v, &code, sys) {
89                        Ok(s) => Value::text(s),
90                        Err(e) => Value::Error(e),
91                    }
92                };
93                Ev::V(zip2(&v, &f, one))
94            }
95            "LET" => self.let_(args),
96            "LAMBDA" => {
97                if args.is_empty() {
98                    return err(CellError::Value);
99                }
100                let mut params = Vec::new();
101                for p in &args[..args.len() - 1] {
102                    match p {
103                        Expr::Name(n) => params.push(n.trim_start_matches('[').trim_end_matches(']').to_string()),
104                        _ => return err(CellError::Value),
105                    }
106                }
107                let Some(body) = args.last() else { return err(CellError::Value) };
108                Ev::L(Arc::new(Lambda { params, body: body.clone(), env: self.env.clone() }))
109            }
110            "ISOMITTED" => {
111                let r = matches!(args.first(), Some(Expr::Name(n)) if self.env.iter().rev().find(|(k, _)| crate::formula::same_name(k, n)).is_some_and(|(_, v)| matches!(v, Ev::V(Value::Empty))));
112                Ev::V(Value::Bool(r))
113            }
114            "MAP" | "REDUCE" | "SCAN" | "BYROW" | "BYCOL" | "MAKEARRAY" => self.higher_order(name, args),
115            "GROUPBY" | "PIVOTBY" => {
116                // The aggregation may be a function given by name (`SUM`): pass it as its name.
117                let i = if name == "GROUPBY" { 2 } else { 3 };
118                match args.get(i).and_then(|a| self.function_by_name(a)) {
119                    Some(f) => {
120                        let mut args = args.to_vec();
121                        if let Some(a) = args.get_mut(i) {
122                            *a = Expr::Text(f.into());
123                        }
124                        self.builtin(name, &args)
125                    }
126                    None => self.builtin(name, args),
127                }
128            }
129            "NOW" | "TODAY" | "RAND" | "RANDBETWEEN" | "RANDARRAY" => {
130                self.host.note_volatile();
131                self.builtin(name, args)
132            }
133            _ => self.builtin(name, args),
134        }
135    }

137    fn if_call(&mut self, args: &[Expr]) -> Ev {
138        if args.is_empty() || args.len() > 3 {
139            return err(CellError::Value);
140        }
141        let condition = self.arg_val(args, 0).unwrap_or_default();
142        if let Value::Array(array) = &condition {
143            let when_true = self.arg_val(args, 1).unwrap_or(Value::Bool(true));
144            let when_false = if args.len() > 2 { self.arg_val(args, 2).unwrap_or(Value::Bool(false)) } else { Value::Bool(false) };
145            return Ev::V(zip3(array, &when_true, &when_false, |condition, when_true, when_false| match condition.to_bool() {
146                Ok(true) => when_true.clone(),
147                Ok(false) => when_false.clone(),
148                Err(error) => Value::Error(error),
149            }));
150        }
151        match condition.to_bool() {
152            Ok(true) => match args.get(1) {
153                Some(Expr::Missing) => Ev::V(Value::Number(0.0)),
154                Some(expression) => self.eval(expression),
155                None => Ev::V(Value::Bool(true)),
156            },
157            Ok(false) => match args.get(2) {
158                Some(Expr::Missing) => Ev::V(Value::Number(0.0)),
159                Some(expression) => self.eval(expression),
160                None => Ev::V(Value::Bool(false)),
161            },
162            Err(error) => err(error),
163        }
164    }

166    fn if_error_call(&mut self, name: &str, args: &[Expr]) -> Ev {
167        if args.len() != 2 {
168            return err(CellError::Value);
169        }
170        let value = self.arg_ev(args, 0).unwrap_or(Ev::V(Value::Empty));
171        let na_only = name == "IFNA";
172        let catches = |error: CellError| !na_only || error == CellError::NA;
173        match &value {
174            Ev::V(Value::Error(error)) if catches(*error) => self.eval(&args[1]),
175            Ev::V(Value::Array(array)) if array.data.iter().any(|item| item.as_error().is_some_and(catches)) => {
176                let alternate = self.arg_val(args, 1).unwrap_or_default();
177                let data =
178                    array.data.iter().map(|item| if item.as_error().is_some_and(catches) { alternate.scalar() } else { item.clone() }).collect();
179                Ev::V(Array::new(array.rows, array.cols, data).map(Value::from).unwrap_or(Value::Error(CellError::Value)))
180            }
181            Ev::R(_) => self.if_error_reference(value.clone(), args, catches),
182            _ => value,
183        }
184    }

186    fn if_error_reference(&mut self, value: Ev, args: &[Expr], catches: impl Fn(CellError) -> bool + Copy) -> Ev {
187        match self.deref(value.clone()) {
188            Value::Error(error) if catches(error) => self.eval(&args[1]),
189            Value::Array(ref array) if array.data.iter().any(|item| item.as_error().is_some_and(catches)) => {
190                let alternate = self.arg_val(args, 1).unwrap_or_default();
191                let data =
192                    array.data.iter().map(|item| if item.as_error().is_some_and(catches) { alternate.scalar() } else { item.clone() }).collect();
193                Ev::V(Array::new(array.rows, array.cols, data).map(Value::from).unwrap_or(Value::Error(CellError::Value)))
194            }
195            _ => value,
196        }
197    }

199    fn choose_call(&mut self, args: &[Expr]) -> Ev {
200        if args.len() < 2 {
201            return err(CellError::Value);
202        }
203        let index = self.value(&args[0]);
204        if let Value::Array(_) = index {
205            let Some(spec) = crate::functions::lookup("CHOOSE") else { return err(CellError::Name) };
206            let mut evaluated = vec![Arg { value: index, from_ref: false }];
207            for argument in &args[1..] {
208                evaluated.push(Arg { value: self.value(argument), from_ref: false });
209            }
210            let mut context = FnCtx { ev: self, criteria: None, lambda: None };
211            return crate::functions::call(spec, &evaluated, &mut context).pipe(Ev::V);
212        }
213        let index = match index.to_number() {
214            Ok(number) => number.trunc() as i64,
215            Err(error) => return err(error),
216        };
217        if index < 1 || index as usize >= args.len() {
218            return err(CellError::Value);
219        }
220        self.eval(&args[index as usize])
221    }

223    fn switch_call(&mut self, args: &[Expr]) -> Ev {
224        if args.len() < 3 {
225            return err(CellError::Value);
226        }
227        let value = self.arg_val(args, 0).unwrap_or_default().scalar();
228        if let Value::Error(error) = value {
229            return err(error);
230        }
231        let choices = &args[1..];
232        for pair in choices.as_chunks::<2>().0 {
233            let (candidate, result) = (&pair[0], &pair[1]);
234            let candidate = self.value(candidate).scalar();
235            if !candidate.is_error() && compare(&value, &candidate) == std::cmp::Ordering::Equal && same_kind(&value, &candidate) {
236                return self.eval(result);
237            }
238        }
239        if let Some(default) = choices.get(choices.len().saturating_sub(1)).filter(|_| choices.len() % 2 == 1) {
240            self.eval(default)
241        } else {
242            err(CellError::NA)
243        }
244    }

246    fn row_column_call(&mut self, name: &str, args: &[Expr]) -> Ev {
247        let is_row = name == "ROW";
248        if args.is_empty() || matches!(args.first(), Some(Expr::Missing)) {
249            return Ev::V(Value::number(if is_row { self.at.row + 1 } else { self.at.col + 1 } as f64));
250        }
251        let areas = match self.eval(&args[0]) {
252            Ev::R(areas) => areas,
253            Ev::V(Value::Error(error)) => return err(error),
254            _ => return err(CellError::Value),
255        };
256        let Some(area) = areas.first() else { return err(CellError::Ref) };
257        let range = self.trim_lines(area);
258        if is_row {
259            let values: Vec<Value> = (range.start.row..=range.end.row).map(|row| Value::number(row as f64 + 1.0)).collect();
260            if values.len() == 1 { Ev::V(values.into_iter().next().unwrap_or_default()) } else { Ev::V(Array::column(values).into()) }
261        } else {
262            let values: Vec<Value> = (range.start.col..=range.end.col).map(|col| Value::number(col as f64 + 1.0)).collect();
263            if values.len() == 1 { Ev::V(values.into_iter().next().unwrap_or_default()) } else { Ev::V(Array::row(values).into()) }
264        }
265    }

267    fn sheet_call(&mut self, name: &str, args: &[Expr]) -> Ev {
268        let is_sheet = name == "SHEET";
269        if args.is_empty() {
270            let count = if is_sheet { self.sheet + 1 } else { self.host.workbook().sheets.len() };
271            return Ev::V(Value::number(count as f64));
272        }
273        match self.eval(&args[0]) {
274            Ev::R(areas) if is_sheet => areas.first().map_or(err(CellError::Ref), |area| Ev::V(Value::number(area.sheet as f64 + 1.0))),
275            Ev::R(areas) => {
276                let mut sheets: Vec<usize> = areas.iter().map(|area| area.sheet).collect();
277                sheets.dedup();
278                Ev::V(Value::number(sheets.len() as f64))
279            }
280            Ev::V(Value::Text(name)) if is_sheet => {
281                self.host.workbook().sheet_index(&name).map_or(err(CellError::NA), |index| Ev::V(Value::number(index as f64 + 1.0)))
282            }
283            _ => err(CellError::NA),
284        }
285    }

287    pub(super) fn builtin(&mut self, name: &str, args: &[Expr]) -> Ev {
288        let Some(spec) = crate::functions::lookup(name) else {
289            // A defined name holding a LAMBDA, or a LET-bound lambda.
290            return match self.name(name) {
291                Ev::L(l) => self.apply_lambda(&l, args),
292                Ev::V(Value::Error(x)) => err(x),
293                _ => err(CellError::Name),
294            };
295        };
296        if args.len() < spec.min_args || args.len() > spec.max_args {
297            return err(CellError::Value);
298        }
299        if let Some(kind) = Agg::of(spec.name) {
300            return self.aggregate(kind, args);
301        }
302        let mut evaluated = Vec::with_capacity(args.len());
303        // SUMIF/AVERAGEIF: the size of the criteria range (argument 1), which the sum range takes.
304        let mut criteria_shape = None;
305        // Database functions: the criteria range (argument 2), whose formulas are computed criteria.
306        let mut criteria = None;
307        // GROUPBY/PIVOTBY: the LAMBDA given as their function.
308        let mut lambda = None;
309        let lambda_at = match name {
310            "GROUPBY" => Some(2),
311            "PIVOTBY" => Some(3),
312            _ => None,
313        };
314        for (i, a) in args.iter().enumerate() {
315            let mut ev = self.eval(a);
316            if lambda_at == Some(i)
317                && let Ev::L(l) = &ev
318            {
319                lambda = Some(l.clone());
320                ev = Ev::V(Value::Empty);
321            }
322            if spec.category == crate::functions::Category::Database
323                && i == 2
324                && let Ev::R(areas) = &ev
325                && let [area] = areas.as_slice()
326            {
327                criteria = Some(*area);
328            }
329            if resizes_sum_range(name, args.len()) {
330                if i == 0 {
331                    criteria_shape = ev_shape(&ev);
332                } else if i == 2
333                    && let (Some((h, w)), Ev::R(areas)) = (criteria_shape, &ev)
334                    && let [area] = areas.as_slice()
335                {
336                    ev = Ev::R(vec![Area { sheet: area.sheet, range: resized_sum_range(area.range, h, w) }]);
337                }
338            }
339            let from_ref = matches!(ev, Ev::R(_));
340            let value = match ev {
341                Ev::R(areas) if areas.len() > 1 => {
342                    // Multi-area references (unions): flatten into one row for aggregates.
343                    let mut vals = Vec::new();
344                    for ar in &areas {
345                        match self.area_value(ar) {
346                            Value::Array(a) => vals.extend(a.data.iter().cloned()),
347                            v => vals.push(v),
348                        }
349                    }
350                    Value::from(Array::row(vals))
351                }
352                ev => self.deref(ev),
353            };
354            evaluated.push(Arg { value, from_ref });
355        }
356        let mut ctx = FnCtx { ev: self, criteria, lambda };
357        crate::functions::call(spec, &evaluated, &mut ctx).pipe(Ev::V)
358    }

360    /// Performs aggregation operations without copying its ranges: blocks are folded over the stored
361    /// cells, so no array is built and there is no `MAX_CELLS` limit. Arguments are evaluated and read one
362    /// after another, as the array path does, so volatile formulas (like `RAND`) see the same order.
363    /// Only `SUM`, `COUNT` and `AVG` are supported currently.
364    fn aggregate(&mut self, kind: Agg, args: &[Expr]) -> Ev {
365        let mut acc = Acc::default();
366        let mut first_error = None;
367        for a in args {
368            let ev = self.eval(a);
369            let steps = match ev {
370                Ev::R(areas) => areas.iter().map(|area| self.aggregate_area(kind, area, &mut acc)).collect(),
371                ev => {
372                    let value = self.deref(ev);
373                    vec![self.aggregate_value(kind, value, false, &mut acc)]
374                }
375            };
376            if first_error.is_none() {
377                first_error = steps.into_iter().find_map(Result::err);
378            }
379        }
380        first_error.unwrap_or_else(|| Ev::V(acc.finish(kind)))
381    }

383    /// Adds one area to `acc`; `Err` is the result the function ends with.
384    fn aggregate_area(&mut self, kind: Agg, a: &Area, acc: &mut Acc) -> Result<(), Ev> {
385        if !a.range.is_single() {
386            let mut error = None;
387            let walked = self.host.visit_range(a.sheet, a.range, &mut |v| match acc.add_cell(kind, v) {
388                Ok(()) => true,
389                Err(e) => {
390                    error = Some(e);
391                    false
392                }
393            });
394            if walked {
395                return error.map_or(Ok(()), |e| Err(err(e)));
396            }
397        }
398        let value = self.area_value(a);
399        self.aggregate_value(kind, value, true, acc)
400    }

402    /// Adds one argument's value by calling `SUM` and `COUNT` on it, so coercion and error rules
403    /// stay in one place (`AVERAGE` of an argument is its `SUM` over its `COUNT`).
404    fn aggregate_value(&mut self, kind: Agg, value: Value, from_ref: bool, acc: &mut Acc) -> Result<(), Ev> {
405        let (Some(sum), Some(count)) = (crate::functions::lookup("SUM"), crate::functions::lookup("COUNT")) else {
406            return Err(err(CellError::Name));
407        };
408        let args = [Arg { value, from_ref }];
409        if kind != Agg::Count {
410            match crate::functions::call(sum, &args, &mut FnCtx { ev: self, criteria: None, lambda: None }) {
411                Value::Number(n) => acc.sum += n,
412                other => return Err(Ev::V(other)),
413            }
414        }
415        if kind != Agg::Sum {
416            match crate::functions::call(count, &args, &mut FnCtx { ev: self, criteria: None, lambda: None }) {
417                Value::Number(n) => acc.count = acc.count.saturating_add(n as u64),
418                other => return Err(Ev::V(other)),
419            }
420        }
421        Ok(())
422    }

424    /// A worksheet function given by name where a function is expected (`BYROW(A1:C3,SUM)`,
425    /// `GROUPBY(A2:A9,B2:B9,SUM)`): its name, unless a LET/LAMBDA variable or a defined name has
426    /// that name.
427    pub(super) fn function_by_name(&self, e: &Expr) -> Option<String> {
428        let Expr::Name(n) = e else { return None };
429        if self.env.iter().any(|(k, _)| crate::formula::same_name(k, n)) || self.host.workbook().name(n, self.sheet).is_some() {
430            return None;
431        }
432        let u = n.to_ascii_uppercase();
433        crate::functions::lookup(&u).map(|_| u)
434    }
