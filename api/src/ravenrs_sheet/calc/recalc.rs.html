1//! Dependency graph and recalculation.
2//!
3//! The graph maps every formula cell to its static precedents (resolved areas). A reverse index
4//! answers "which formulas read this cell?": single-cell precedents in a hash map, ranges as row
5//! intervals per column (wide ranges per sheet), each an interval index so a lookup costs
6//! O(log n + matches) however many ranges share the column. After an edit, the dirty set is the
7//! transitive closure of dependents; it is evaluated in topological order, with on-demand
8//! evaluation as a safety net for dynamic references. Cycles produce `#CIRC!` unless iterative
9//! calculation is on: then a formula read while it is being evaluated gives its previous value,
10//! and the formulas that read one are evaluated again until the loop settles
11//! (`PassHost::iterate_loops`).
12

48/// Pops one finished component off Tarjan's stack, labelling its nodes with `comp_index`. The
49/// component is everything up to and including `root`.
50fn pop_component(stack: &mut Vec<usize>, on_stack: &mut [bool], comp: &mut [usize], comp_index: usize, root: usize) {
51    while let Some(w) = stack.pop() {
52        on_stack[w] = false;
53        comp[w] = comp_index;
54        if w == root {
55            break;
56        }
57    }
58}

60/// Strongly connected components of a graph given as adjacency lists (Tarjan, iterative): each
61/// node's component index, and the number of nodes in each component.
62fn components(edges: &[Vec<usize>]) -> (Vec<usize>, Vec<usize>) {
63    let n = edges.len();
64    let (mut low, mut num) = (vec![0usize; n], vec![usize::MAX; n]);
65    let (mut on_stack, mut stack, mut comp) = (vec![false; n], Vec::new(), vec![usize::MAX; n]);
66    let (mut next, mut comps) = (0usize, 0usize);
67    for root in 0..n {
68        if num[root] != usize::MAX {
69            continue;
70        }
71        let mut call: Vec<(usize, usize)> = vec![(root, 0)];
72        num[root] = next;
73        low[root] = next;
74        next += 1;
75        stack.push(root);
76        on_stack[root] = true;
77        while let Some(top) = call.last_mut() {
78            let v = top.0;
79            if let Some(&w) = edges.get(v).and_then(|e| e.get(top.1)) {
80                top.1 += 1;
81                if num[w] == usize::MAX {
82                    num[w] = next;
83                    low[w] = next;
84                    next += 1;
85                    stack.push(w);
86                    on_stack[w] = true;
87                    call.push((w, 0));
88                } else if on_stack[w] {
89                    low[v] = low[v].min(num[w]);
90                }
91                continue;
92            }
93            call.pop();
94            if let Some(&(u, _)) = call.last() {
95                low[u] = low[u].min(low[v]);
96            }
97            if low[v] == num[v] {
98                pop_component(&mut stack, &mut on_stack, &mut comp, comps, v);
99                comps += 1;
100            }
101        }
102    }
103    let mut size = vec![0usize; comps];
104    for &c in &comp {
105        if let Some(s) = size.get_mut(c) {
106            *s += 1;
107        }
108    }
109    (comp, size)
110}

124    /// Rebuilds the graph from scratch.
125    pub fn rebuild(&mut self, wb: &Workbook) {
126        self.graph = Graph::default();
127        let mut circ = Vec::new();
128        for (si, sheet) in wb.sheets.iter().enumerate() {
129            for (c, cell) in sheet.cells.iter() {
130                if let Some(f) = &cell.formula
131                    && let Some((areas, dynamic)) = f.with_expr(|e| precedents(wb, si, e))
132                {
133                    self.graph.insert((si, c), Node { areas, dynamic });
134                    if cell.value == Value::Error(CellError::Circ) {
135                        circ.push((si, c));
136                    }
137                } else if cell.py.is_some() {
138                    // A Python cell lays out its kept result, so it is calculated like a formula
139                    // that reads nothing: its spill is blocked and unblocked like a formula's.
140                    self.graph.insert((si, c), Node { areas: Vec::new(), dynamic: false });
141                }
142            }
143        }
144        self.graph.settle();
145        self.circular = if wb.calc.iterative { Vec::new() } else { self.loops_among(&circ) };
146    }

148    /// The circular references of a rebuilt workbook (after undo, redo or a rollback, which put
149    /// back cells without calculating them), from the formulas whose cached value is `#CIRC!`:
150    /// the listed cells still on a loop, and one cell of each loop none of them is on.
151    fn loops_among(&self, circ: &[Key]) -> Vec<Key> {
152        if circ.is_empty() {
153            return Vec::new();
154        }
155        let n = circ.len();
156        let index: HashMap<Key, usize> = circ.iter().enumerate().map(|(i, k)| (*k, i)).collect();
157        // Edges from a cell to the `#CIRC!` formulas that read it; loops are the same either way.
158        let mut deps = Vec::new();
159        let edges: Vec<Vec<usize>> = circ
160            .iter()
161            .map(|k| {
162                deps.clear();
163                self.graph.dependents(k.0, k.1, &mut deps);
164                let mut e: Vec<usize> = deps.iter().filter_map(|d| index.get(d).copied()).collect();
165                e.sort_unstable();
166                e.dedup();
167                e
168            })
169            .collect();
170        // Loop members: nodes in a component of more than one, or with an edge to themselves.
171        let (comp, size) = components(&edges);
172        let on_loop = |i: usize| comp.get(i).and_then(|c| size.get(*c)).is_some_and(|s| *s > 1) || edges.get(i).is_some_and(|e| e.contains(&i));
173        // A formula with dynamic references (INDIRECT, OFFSET) may close a loop the graph can't see.
174        let mut out: Vec<Key> =
175            self.circular.iter().copied().filter(|k| index.get(k).is_some_and(|&i| on_loop(i) || self.graph.dynamic.contains(k))).collect();
176        let mut covered: HashSet<usize> = out.iter().filter_map(|k| index.get(k)).filter_map(|&i| comp.get(i).copied()).collect();
177        let mut order: Vec<usize> = (0..n).collect();
178        order.sort_by_key(|&i| circ.get(i).map(|(s, c)| (*s, c.row, c.col)));
179        for i in order {
180            if let (Some(&c), Some(&k)) = (comp.get(i), circ.get(i))
181                && on_loop(i)
182                && covered.insert(c)
183            {
184                out.push(k);
185            }
186        }
187        out
188    }

190    /// Recalculates every formula (F9 / Ctrl+Alt+F9, and after loading).
191    pub fn recalc_all(&mut self, wb: &mut Workbook) {
192        self.rebuild(wb);
193        // Spills are laid out again from the formulas where they are now (a structural edit moves
194        // or deletes the formulas, not the values they spilled).
195        for i in 0..wb.sheets.len() {
196            if wb.sheet(i).is_some_and(|s| !s.spill_ranges.is_empty() || !s.spill.is_empty())
197                && let Some(sh) = wb.sheet_mut(i)
198            {
199                sh.spill.clear();
200                sh.spill_ranges.clear();
201            }
202        }
203        let all: Vec<Key> = self.graph.nodes.keys().copied().collect();
204        self.run(wb, all, true);
205    }

207    /// Updates the graph for cells whose content changed and recalculates what depends on them.
208    pub fn cells_changed(&mut self, wb: &mut Workbook, changed: &[Key]) {
209        let mut cleared: Vec<(usize, RangeRef)> = Vec::new();
210        for &k in changed {
211            let precedent =
212                wb.sheet(k.0).and_then(|s| s.cell(k.1)).and_then(|c| c.formula.as_ref()).and_then(|f| f.with_expr(|e| precedents(wb, k.0, e)));
213            let py = wb.sheet(k.0).and_then(|s| s.cell(k.1)).and_then(|c| c.py.as_ref());
214            let python = py.is_some();
215            // A Python cell whose result (if any) is not an array spills nothing.
216            let spills = py.and_then(|p| p.result_value()).is_some_and(|v| matches!(v, Value::Array(_)));
217            match precedent {
218                Some((areas, dynamic)) => {
219                    self.graph.insert(k, Node { areas, dynamic });
220                }
221                // See `rebuild`: its spill is laid out again from its result when it is calculated.
222                None if python => {
223                    self.graph.insert(k, Node { areas: Vec::new(), dynamic: false });
224                    // New code has no result yet: the spill the cell had (its old result's, or a
225                    // formula's) goes, and what read it is recalculated, as for a formula
226                    // replaced by a constant.
227                    if !spills
228                        && let Some(sh) = wb.sheet_mut(k.0)
229                        && let Some(r) = sh.spill_ranges.remove(&k.1)
230                    {
231                        sh.clear_spill(r);
232                        cleared.push((k.0, r));
233                    }
234                }
235                None => {
236                    self.graph.remove(k);
237                    // A formula replaced by a constant (or cleared) takes its spilled values with it.
238                    if let Some(sh) = wb.sheet_mut(k.0)
239                        && let Some(r) = sh.spill_ranges.remove(&k.1)
240                    {
241                        for c in r.iter() {
242                            sh.spill.remove(&c);
243                        }
244                        cleared.push((k.0, r));
245                    }
246                    // A formula that doesn't parse evaluates to #NAME?.
247                    if let Some(cell) = wb.sheet_mut(k.0).and_then(|s| s.cells.get_mut(k.1))
248                        && cell.formula.is_some()
249                    {
250                        cell.value = Value::Error(CellError::Name);
251                    }
252                }
253            }
254        }
255        self.graph.settle();
256        if wb.calc.mode == CalcMode::Manual {
257            // Only the edited formulas themselves are evaluated.
258            let own: Vec<Key> = changed.iter().copied().filter(|k| self.graph.nodes.contains_key(k)).collect();
259            self.run(wb, own, false);
260            return;
261        }
262        let mut seeds: Vec<Key> = changed.to_vec();
263        let mut emptied: Vec<Key> = Vec::new();
264        // Typing into (or clearing) a cell a dynamic array spills over (or would spill over)
265        // re-evaluates the anchor, which may now be blocked or unblocked.
266        for &k in changed {
267            if let Some(sh) = wb.sheet(k.0) {
268                for (anchor, r) in &sh.spill_ranges {
269                    if *anchor != k.1 && r.contains(k.1) {
270                        seeds.push((k.0, *anchor));
271                    }
272                }
273                for (c, cell) in sh.cells.iter_range(RangeRef::new(CellRef::new(k.1.row.saturating_sub(64), k.1.col.saturating_sub(64)), k.1)) {
274                    if c != k.1 && cell.value == Value::Error(CellError::Spill) {
275                        seeds.push((k.0, c));
276                    }
277                }
278                if !sh.cells.has(k.1) {
279                    match emptied.iter_mut().find(|x| x.0 == k.0) {
280                        Some(x) => x.1 = CellRef::new(x.1.row.max(k.1.row), x.1.col.max(k.1.col)),
281                        None => emptied.push(k),
282                    }
283                }
284            }
285        }
286        // An emptied cell can unblock an anchor any distance above or to the left of it.
287        for (si, end) in emptied {
288            for &(s, c) in self.graph.nodes.keys() {
289                if s == si && c.row <= end.row && c.col <= end.col && wb.sheet(s).is_some_and(|sh| sh.value(c) == Value::Error(CellError::Spill)) {
290                    seeds.push((s, c));
291                }
292            }
293        }
294        // Spill areas of changed anchors also change. They stay rectangles: one seed per spilled
295        // cell listed the formulas reading the area once per cell.
296        let mut areas: Vec<(usize, RangeRef)> = cleared;
297        for &k in changed {
298            if let Some(r) = wb.sheet(k.0).and_then(|s| s.spill_ranges.get(&k.1)) {
299                areas.push((k.0, *r));
300            }
301        }
302        let t0 = prof_now();
303        let dirty = self.dirty_closure(wb, &seeds, &areas);
304        let t1 = prof_now();
305        let n = dirty.len();
306        self.run(wb, dirty, false);
307        if std::env::var_os("GRIDCRAFT_PROFILE").is_some() {
308            eprintln!("recalc: {} seeds, closure {} cells {:.1} ms, run {:.1} ms", seeds.len(), n, t1 - t0, prof_now() - t1);
309        }
310    }

312    /// The formulas to recalculate after `seeds` and every cell of `areas` changed: the seeds
313    /// that are formulas, and everything that reads them, directly or through other formulas.
314    fn dirty_closure(&self, wb: &Workbook, seeds: &[Key], areas: &[(usize, RangeRef)]) -> Vec<Key> {
315        let mut seen: HashSet<Key> = HashSet::default();
316        let mut queue: VecDeque<Key> = VecDeque::new();
317        // Each range-index entry is listed once in the whole walk: many changed cells under one
318        // broad reference (or many formulas inside one) would otherwise list every formula reading
319        // it once per cell, which squared the work.
320        let mut taken = Taken::default();
321        let mut deps = Vec::new();
322        for k in seeds {
323            if self.graph.nodes.contains_key(k) && seen.insert(*k) {
324                queue.push_back(*k);
325            }
326            deps.clear();
327            self.graph.dependents_untaken(k.0, k.1, &mut taken, &mut deps);
328            for d in deps.drain(..) {
329                if seen.insert(d) {
330                    queue.push_back(d);
331                }
332            }
333        }
334        for &(si, r) in areas {
335            if let Some(sh) = wb.sheet(si) {
336                for (c, _) in sh.cells.iter_range(r) {
337                    if self.graph.nodes.contains_key(&(si, c)) && seen.insert((si, c)) {
338                        queue.push_back((si, c));
339                    }
340                }
341            }
342            deps.clear();
343            self.graph.dependents_of_range(si, r, &mut deps);
344            for d in deps.drain(..) {
345                if seen.insert(d) {
346                    queue.push_back(d);
347                }
348            }
349        }
350        for k in &self.graph.dynamic {
351            if seen.insert(*k) {
352                queue.push_back(*k);
353            }
354        }
355        while let Some(k) = queue.pop_front() {
356            deps.clear();
357            self.graph.dependents_untaken(k.0, k.1, &mut taken, &mut deps);
358            for d in deps.drain(..) {
359                if seen.insert(d) {
360                    queue.push_back(d);
361                }
362            }
363        }
364        seen.into_iter().collect()
365    }

367    /// Evaluates the dirty set and writes results (and spills) back.
368    fn run(&mut self, wb: &mut Workbook, mut dirty: Vec<Key>, full: bool) {
369        let mut dynamic_again = !self.graph.dynamic.is_empty();
370        self.last_recalc_cells = 0;
371        for _round in 0..8 {
372            if dirty.is_empty() {
373                break;
374            }
375            // Row-major order is a good topological guess; on-demand evaluation fixes the rest.
376            dirty.sort_by_key(|(s, c)| (*s, c.row, c.col));
377            let mut exprs: HashMap<Key, std::sync::Arc<Expr>> = HashMap::with_capacity_and_hasher(dirty.len(), Default::default());
378            for k in &dirty {
379                if let Some(e) = wb.sheet(k.0).and_then(|s| s.cell(k.1)).and_then(|c| c.formula.as_ref()).and_then(|f| f.expr_arc()) {
380                    exprs.insert(*k, e);
381                }
382            }
383            let python: HashMap<Key, Value> = dirty
384                .iter()
385                .filter_map(|k| wb.sheet(k.0).and_then(|s| s.cell(k.1)).and_then(|c| c.py.as_ref()).and_then(|p| p.result_value()).map(|v| (*k, v)))
386                .collect();
387            let now = self.now();
388            let mut rng = self.rng;
389            let tp = prof_now();
390            let (results, spills, cycles) = {
391                let mut host = PassHost::new(wb, &exprs, &mut rng, now).with_python(python);
392                host.iterate = wb.calc.iterative;
393                for k in &dirty {
394                    host.calculate(*k);
395                }
396                if host.iterate && !host.members.is_empty() {
397                    let passes = host.iterate_loops(&dirty, wb.calc.max_iterations, wb.calc.max_change);
398                    if std::env::var_os("GRIDCRAFT_PROFILE").is_some() {
399                        eprintln!("  iterative calculation: {} passes over {} formulas", passes, host.tainted.len());
400                    }
401                }
402                (host.results, host.spills, host.cycles)
403            };
404            self.rng = rng;
405            if std::env::var_os("GRIDCRAFT_PROFILE").is_some() {
406                eprintln!("  eval {} formulas {:.1} ms", results.len(), prof_now() - tp);
407            }
408            let tw = prof_now();
409            self.last_recalc_cells += results.len();
410            // The loops this round finds replace the ones listed for the formulas it evaluated.
411            if !self.circular.is_empty() {
412                let evaluated: HashSet<Key> = dirty.iter().copied().collect();
413                self.circular.retain(|k| !evaluated.contains(k));
414            }
415            self.circular.extend(cycles);
416            let mut spill_changes = self.write_back(wb, &dirty, results, &spills);
417            // Formulas with references only known while evaluating (INDIRECT, OFFSET) may have read
418            // a spill area before it was laid out: they are evaluated once more.
419            if dynamic_again && !spills.is_empty() {
420                dynamic_again = false;
421                spill_changes.extend(self.graph.dynamic.iter().copied());
422            }
423            if full {
424                // Everything was already evaluated once; spill dependents need another pass.
425            }
426            if std::env::var_os("GRIDCRAFT_PROFILE").is_some() {
427                eprintln!("  write-back {:.1} ms", prof_now() - tw);
428            }
429            spill_changes.sort_by_key(|(s, c)| (*s, c.row, c.col));
430            spill_changes.dedup();
431            if spill_changes.is_empty() {
432                break;
433            }
434            dirty = self.dirty_closure(wb, &spill_changes, &[]);
435        }
436        // Loops among formulas the run did not evaluate stay listed while they are still formulas.
437        // Iterative calculation reports none.
438        if wb.calc.iterative {
439            self.circular.clear();
440        } else {
441            self.circular.retain(|k| self.graph.nodes.contains_key(k));
442        }
443    }

445    fn write_back(
446        &mut self,
447        wb: &mut Workbook,
448        dirty: &[Key],
449        results: HashMap<Key, Value>,
450        spills: &HashMap<Key, std::sync::Arc<Array>>,
451    ) -> Vec<Key> {
452        let mut spill_changes = Vec::new();
453        let mut freed = Vec::new();
454        for (key, value) in results {
455            let Some(sheet) = wb.sheet_mut(key.0) else { continue };
456            if let Some(cell) = sheet.cells.get_mut(key.1)
457                && cell.value != value
458            {
459                cell.value = value;
460            }
461        }
462        for key in dirty {
463            let new = spills.get(key);
464            let Some(sheet) = wb.sheet_mut(key.0) else { continue };
465            let old = sheet.spill_ranges.get(&key.1).copied();
466            if old.is_none() && new.is_none() {
467                continue;
468            }
469            // An unchanged spill does not alter any values its dependents read.
470            if let (Some(range), Some(array)) = (old, new)
471                && spills_as_before(sheet, key.1, range, array)
472            {
473                continue;
474            }
475            if let Some(range) = old {
476                for cell in range.iter() {
477                    sheet.spill.remove(&cell);
478                }
479                sheet.spill_ranges.remove(&key.1);
480            }
481            let mut changed_range = old;
482            if let Some(array) = new {
483                let end = CellRef::new(key.1.row.saturating_add(array.rows as u32 - 1), key.1.col.saturating_add(array.cols as u32 - 1));
484                let range = RangeRef::new(key.1, end);
485                let blocked = spill_is_blocked(sheet, key.1, range, end);
486                if blocked {
487                    if let Some(cell) = sheet.cells.get_mut(key.1) {
488                        cell.value = Value::Error(CellError::Spill);
489                    }
490                    let mut blocked_dependents = Vec::new();
491                    self.graph.dependents_of_range(key.0, range, &mut blocked_dependents);
492                    spill_changes.extend(blocked_dependents.into_iter().filter(|dependent| dependent != key));
493                } else {
494                    fill_spill(sheet, key.1, range, array);
495                    changed_range = Some(changed_range.map_or(range, |previous| previous.union(&range)));
496                }
497            }
498            if let Some(range) = changed_range {
499                let mut dependents = Vec::new();
500                self.graph.dependents_of_range(key.0, range, &mut dependents);
501                let freed_old_range = match (old, sheet.spill_ranges.get(&key.1)) {
502                    (Some(previous), Some(current)) => !(current.contains(previous.start) && current.contains(previous.end)),
503                    (_, None) => true,
504                    (None, Some(_)) => false,
505                };
506                spill_changes.extend(dependents.into_iter().filter(|dependent| dependent != key || freed_old_range));
507            }
508            if let Some(previous) = old
509                && !sheet.spill_ranges.get(&key.1).is_some_and(|current| current.contains(previous.start) && current.contains(previous.end))
510            {
511                freed.push((key.0, previous));
512            }
513        }
514        for (sheet_index, range) in freed {
515            for &(sheet, cell) in self.graph.nodes.keys() {
516                let was_blocked_cell = cell.row <= range.end.row
517                    && cell.col <= range.end.col
518                    && wb.sheet(sheet).is_some_and(|sheet| sheet.value(cell) == Value::Error(CellError::Spill));
519                if sheet == sheet_index && was_blocked_cell {
520                    spill_changes.push((sheet, cell));
521                }
522            }
523        }
524        spill_changes
525    }
526}

528fn spill_is_blocked(sheet: &crate::model::Sheet, anchor: CellRef, range: RangeRef, end: CellRef) -> bool {
529    end.row >= crate::base::MAX_ROWS
530        || end.col >= crate::base::MAX_COLS
531        || range.iter().any(|cell| cell != anchor && (sheet.cells.has(cell) || sheet.spill.contains_key(&cell)))
532        || sheet.merges.iter().any(|merge| merge.intersects(&range))
533        || sheet.tables.iter().any(|table| table.range.contains(anchor))
534}

536fn fill_spill(sheet: &mut crate::model::Sheet, anchor: CellRef, range: RangeRef, array: &Array) {
537    for (index, cell) in range.iter().enumerate() {
538        if cell != anchor {
539            sheet.spill.insert(cell, array.data.get(index).cloned().unwrap_or_default());
540        }
541    }
542    sheet.spill_ranges.insert(anchor, range);
543}

1002        }
1003

1005        // of a real pass partway through; `visit_range` has to cope with the formulas it reads

1009

1012            seen.push(v.clone());
1013            true
1014        })
