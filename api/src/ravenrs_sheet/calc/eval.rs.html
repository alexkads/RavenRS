3//! Expressions evaluate to [`Ev`]: a value, a reference (one or more areas, kept unresolved so

26/// Largest array a reference may expand to.
27pub const MAX_CELLS: u64 = 16_000_000;
28/// References up to this many cells (one whole row) are always read at their full size.
29const FULL_SIZE_CELLS: u64 = 16_384;
30/// Bounded references up to this many cells (one column's worth) are read at their full size too, as
31/// Excel does, so their shape lines up with the arrays next to them (`D3:AZ400*B3:B400` is 398 by
32/// 49 even when the data ends at column F) and their blanks count. References that run to the edge
33/// of the sheet (`A:A`, `A2:A1048576`, several whole rows) and larger blocks stop at the sheet's used
34/// range, so that formulas over them stay fast.
35const FULL_SIZE_BOUNDED_CELLS: u64 = 1_048_576;
36const MAX_DEPTH: usize = 400;
37/// Lambda invocations one cell may make. A semantically bounded formula in Excel can invoke
38/// its lambda at most once per cell of an array (capped here by `MAX_CELLS`), so this covers
39/// every legitimate call; hostile work that grows past one dimension (a LAMBDA that recurses
40/// or nest `MAP` into `MAP`) then stops at the same bound instead of squaring.
41const LAMBDA_FUEL: u64 = MAX_CELLS;
42/// Work one cell's formula may do: one unit per expression evaluated, per cell read from a
43/// reference, per array value handed to a function, per pair of areas intersected and per
44/// character of a defined name's formula parsed where the name is used. Each limit
45/// above bounds one kind of step, but a compact formula could still repeat bounded steps
46/// exponentially often (defined names that each use the one before twice, a computed criterion
47/// whose database function reruns it, an aggregate over a large array inside every LAMBDA call).
48/// Past this the formula is `#CALC!`. It is far above what a formula that reads every cell of a
49/// full-size array a few times needs.
50const EVAL_WORK: u64 = 16 * MAX_CELLS;

82/// LAMBDA calls that one formula may have in progress at once, counted the same way on every build.
83/// The next call answers `#NUM!`. A countdown from 131, the deepest that `MAX_DEPTH` allows, has 132
84/// calls in progress, so it still runs.
85const MAX_LAMBDA_CALLS: usize = 140;
86
87/// Native stack, in bytes, that one formula's evaluation may use below its first `eval`. Past it,
88/// `eval` answers `#NUM!`. It guards the stack and leaves 128 KiB of a 1 MiB thread for the frames
89/// above the formula. A cell that another cell's formula computes on demand shares the budget of the
90/// outermost formula ([`Host::stack_top`]).
91const STACK_BUDGET: usize = 896 * 1024;
92
93/// The address of a local variable, which shows how far down the native stack this call is.
94#[inline(always)]
95pub(crate) fn stack_position() -> usize {
96    let marker = 0u8;
97    std::hint::black_box(&marker) as *const u8 as usize
98}

100/// What the evaluator needs from its surroundings.
101pub trait Host {
102    fn workbook(&self) -> &Workbook;
103    /// The current value of a cell (evaluating it first if it is dirty).
104    fn cell_value(&mut self, sheet: usize, c: CellRef) -> Value;
105    fn now_serial(&self) -> f64;
106    fn random(&mut self) -> f64;
107    /// Called when a volatile or dynamic-reference function is used.
108    fn note_volatile(&mut self) {}
109    /// The spill range of the dynamic array anchored at `anchor` (`None`: it doesn't spill).
110    fn spill_range(&mut self, sheet: usize, anchor: CellRef) -> Option<RangeRef> {
111        self.workbook().sheet(sheet)?.spill_ranges.get(&anchor).copied()
112    }
113    /// Values of a block, row-major. Hosts can override with a sparse fast path.
114    fn range_values(&mut self, sheet: usize, range: RangeRef) -> Vec<Value> {
115        range.iter().map(|c| self.cell_value(sheet, c)).collect()
116    }
117    /// The sheet's used range (`Sheet::used_range`, which walks every stored row). Hosts that
118    /// keep the workbook unchanged while evaluating can remember it.
119    fn used_range(&mut self, sheet: usize) -> Option<RangeRef> {
120        self.workbook().sheet(sheet).and_then(|s| s.used_range())
121    }
122    /// Calls `f` on each stored value of a block, row-major (formula cells evaluated first),
123    /// until `f` returns false. `false` when this host can't (callers use `range_values`).
124    fn visit_range(&mut self, _sheet: usize, _range: RangeRef, _f: &mut dyn FnMut(&Value) -> bool) -> bool {
125        false
126    }
127    /// `(sum, count)` of the numbers in a block that holds nothing but constants (no formula,
128    /// error or spilled value), added row-major from `0.0` exactly as `visit_range` would hand
129    /// them over. `None`: the block isn't like that, or this host can't tell.
130    fn range_sum(&mut self, _sheet: usize, _range: RangeRef) -> Option<(f64, u64)> {
131        None
132    }
133    /// While a formula is calculated on demand inside another one: `stack_position()` where the
134    /// outermost one started, which the stack budget is counted from.
135    fn stack_top(&self) -> Option<usize> {
136        None
137    }
138    /// Called when a formula's evaluation runs past the stack budget.
139    fn stack_exhausted(&mut self) {}
140}

142pub struct Evaluator<'h> {
143    pub host: &'h mut dyn Host,
144    pub sheet: usize,
145    pub at: CellRef,
146    env: Vec<(String, Ev)>,
147    depth: usize,
148    /// LAMBDA calls in progress.
149    calls: usize,
150    /// `stack_position()` at the outermost `eval`.
151    stack_top: usize,
152    /// Lambda invocations this cell still has left. Every call spends one charge, so a
153    /// compact hostile formula (a LAMBDA that calls itself, nested `MAP`s over big arrays)
154    /// runs out of work after the same number of calls that a legitimately wide one could
155    /// make, instead of an amount of work that squares with array nesting (#2382).
156    fuel: u64,
157    /// Work this cell still has left (see [`EVAL_WORK`]), shared by every branch of the formula
158    /// and by the formulas it computes on the way (names, computed criteria).
159    work: u64,
160    /// A spend found too little work left. A spend that leaves exactly nothing is fine, so this
161    /// can't be told from `work` alone; a database function checks it after its computed
162    /// criteria ran, whose failures it can't otherwise see.
163    exhausted: bool,
164    /// Computed criteria being evaluated, outermost first: one that is reached again from inside
165    /// itself counts as not met instead of repeating its work at every level.
166    active_criteria: Vec<(usize, CellRef)>,
167}

169struct FnCtx<'a, 'h> {
170    ev: &'a mut Evaluator<'h>,
171    /// The criteria argument of a database function, when it is a reference.
172    criteria: Option<Area>,
173    /// The LAMBDA given as the function argument of GROUPBY/PIVOTBY.
174    lambda: Option<Arc<Lambda>>,
175}

177impl Ctx for FnCtx<'_, '_> {
178    fn date_system(&self) -> crate::base::DateSystem {
179        self.ev.host.workbook().date_system
180    }
181    fn now_serial(&self) -> f64 {
182        self.ev.host.now_serial()
183    }
184    fn random(&mut self) -> f64 {
185        self.ev.host.random()
186    }
187    /// Excel's "formula as criteria": the formula is written for the first record, so its relative
188    /// references move down to `record`'s row, and the database's labels name its fields.
189    /// Known gap: the shifted references are not dependencies of the calling cell. One that lands
190    /// outside both the database and the criteria cell's own precedents (say `=G2<H2` with column H
191    /// outside the database) is read here but not tracked, so editing that cell doesn't recalculate
192    /// the D-function until something else does.
193    fn computed_criterion(&mut self, row: usize, col: usize, db: &Array, record: usize) -> Option<bool> {
194        let area = self.criteria?;
195        let at = area.range.start.offset(row as i64, col as i64).filter(|c| area.range.contains(*c))?;
196        if self.ev.active_criteria.contains(&(area.sheet, at)) {
197            return Some(false);
198        }
199        let expr = self.ev.host.workbook().sheet(area.sheet)?.cell(at)?.formula.as_ref()?.expr_arc()?;
200        let shifted = crate::formula::adjust::shift_relative((*expr).clone(), record as i64 - 1, 0);
201        let env = (0..db.cols)
202            .filter_map(|c| match db.get(0, c) {
203                Some(Value::Text(t)) if !t.trim().is_empty() => {
204                    Some((t.trim().to_string(), Ev::V(db.get(record, c).cloned().unwrap_or(Value::Empty))))
205                }
206                _ => None,
207            })
208            .collect();
209        let mut ev = Evaluator {
210            host: &mut *self.ev.host,
211            sheet: area.sheet,
212            at,
213            env,
214            depth: self.ev.depth,
215            calls: self.ev.calls,
216            stack_top: self.ev.stack_top,
217            fuel: self.ev.fuel,
218            work: self.ev.work,
219            exhausted: false,
220            active_criteria: std::mem::take(&mut self.ev.active_criteria),
221        };
222        ev.active_criteria.push((area.sheet, at));
223        let v = matches!(ev.value(&shifted).scalar().to_bool(), Ok(true));
224        // The child's spend counts against this cell's budget.
225        self.ev.fuel = ev.fuel;
226        self.ev.work = ev.work;
227        self.ev.exhausted |= ev.exhausted;
228        let mut active = std::mem::take(&mut ev.active_criteria);
229        active.pop();
230        self.ev.active_criteria = active;
231        Some(v)
232    }
233    fn exhausted(&self) -> bool {
234        self.ev.exhausted
235    }
236    fn has_lambda(&self) -> bool {
237        self.lambda.is_some()
238    }
239    /// A LAMBDA must give one value per group; an array result is #CALC!.
240    fn call_lambda(&mut self, arg: Value) -> Option<Value> {
241        let l = self.lambda.clone()?;
242        let [p] = l.params.as_slice() else { return Some(Value::Error(CellError::Value)) };
243        let r = self.ev.call_lambda_values(&l, vec![(p.clone(), Ev::V(arg))]);
244        Some(match self.ev.deref(r) {
245            Value::Array(a) if a.data.len() != 1 => Value::Error(CellError::Calc),
246            v => v.scalar(),
247        })
248    }
249}

268    /// Spends `units` of this cell's work; `false` (and nothing left) when there isn't enough.
269    pub(crate) fn spend(&mut self, units: u64) -> bool {
270        match self.work.checked_sub(units) {
271            Some(w) => {
272                self.work = w;
273                true
274            }
275            None => {
276                self.work = 0;
277                self.exhausted = true;
278                false
279            }
280        }
281    }

283    /// Charges the values of an array (one unit per value; one for anything else).
284    pub(crate) fn spend_on(&mut self, v: &Value) -> bool {
285        match v {
286            Value::Array(a) => self.spend(a.data.len() as u64),
287            _ => self.spend(1),
288        }

297    fn resolve_sheet(&self, s: &SheetSel) -> Result<Vec<usize>, CellError> {
298        let wb = self.host.workbook();
299        match s {
300            SheetSel::Current => Ok(vec![self.sheet]),
301            SheetSel::Named(n) => wb.sheet_index(n).map(|i| vec![i]).ok_or(CellError::Ref),
302            SheetSel::Span(a, b) => {
303                let (i, j) = (wb.sheet_index(a).ok_or(CellError::Ref)?, wb.sheet_index(b).ok_or(CellError::Ref)?);
304                Ok((i.min(j)..=i.max(j)).collect())
305            }
306        }
307    }

314    /// Dereferences: one cell → its value; a block → an array; several areas → `#VALUE!`.
315    ///
316    /// A block keeps its full shape (blanks included) when it holds at most `FULL_SIZE_CELLS`
317    /// cells, or when it holds at most `FULL_SIZE_BOUNDED_CELLS` cells (one full column's worth)
318    /// and does not run to the sheet's last row or column. Every other block (larger, or a
319    /// bigger one that runs to the sheet's edge, such as `A:A` or `1:1`) is trimmed to the
320    /// sheet's used range before its values are read. An array over `MAX_CELLS` is `#NUM!`.
321    pub fn deref(&mut self, ev: Ev) -> Value {
322        match ev {
323            Ev::V(v) => v,
324            Ev::L(_) => Value::Error(CellError::Calc),
325            Ev::R(areas) => match areas.as_slice() {
326                [a] => self.area_value(a),
327                _ => Value::Error(CellError::Value),
328            },
329        }
330    }

332    pub fn area_value(&mut self, a: &Area) -> Value {
333        if a.range.is_single() {
334            return self.host.cell_value(a.sheet, a.range.start);
335        }
336        // A reference keeps its size, so the array has its shape and its blanks count
337        // (`=A1:A10`, `SUMPRODUCT(--(A1:A10=""))`).
338        let range = self.read_range(a);
339        let (h, w) = (range.height() as usize, range.width() as usize);
340        if (h as u64) * (w as u64) > MAX_CELLS {
341            return Value::Error(CellError::Num);
342        }
343        if !self.spend((h as u64) * (w as u64)) {
344            return Value::Error(CellError::Calc);
345        }
346        let data = self.host.range_values(a.sheet, range);
347        Array::new(h, w, data).map(Value::from).unwrap_or(Value::Error(CellError::Value))
348    }

350    /// The block that reading `a` gives: `a` itself, or when it is large (or runs to the sheet's
351    /// edge) the part inside the sheet's used range.
352    pub(crate) fn read_range(&mut self, a: &Area) -> RangeRef {
353        let cells = u64::from(a.range.height()) * u64::from(a.range.width());
354        // `A:A`, `1:1` and `A2:A1048576` run to the edge of the sheet: they stand for "the rest of
355        // the column", not for a block of a chosen size.
356        let to_edge = a.range.end.row == MAX_ROWS - 1 || a.range.end.col == MAX_COLS - 1;
357        let full_size = cells <= FULL_SIZE_CELLS || (!to_edge && cells <= FULL_SIZE_BOUNDED_CELLS);
358        if full_size { a.range } else { self.trim(a) }
359    }
360
361    /// Trims the bottom/right of an area to the sheet's used range (keeps the top-left so
362    /// positions stay right).
363    pub fn trim(&mut self, a: &Area) -> RangeRef {
364        if self.host.workbook().sheet(a.sheet).is_none() {
365            return a.range;
366        }
367        let used = self.host.used_range(a.sheet);
368        let (mut r1, mut c1) = (a.range.end.row, a.range.end.col);
369        match used {
370            Some(u) => {
371                r1 = r1.min(u.end.row.max(a.range.start.row));
372                c1 = c1.min(u.end.col.max(a.range.start.col));
373            }
374            None => {
375                r1 = a.range.start.row;
376                c1 = a.range.start.col;
377            }
378        }
379        RangeRef::new(a.range.start, CellRef::new(r1, c1))
380    }

382    pub fn eval(&mut self, e: &Expr) -> Ev {
383        let here = stack_position();
384        if self.depth == 0 {
385            self.stack_top = self.host.stack_top().unwrap_or(here);
386        }
387        if self.stack_top.abs_diff(here) > STACK_BUDGET {
388            self.host.stack_exhausted();
389            return err(CellError::Num);
390        }
391        if !self.spend(1) {
392            return err(CellError::Calc);
393        }
394        self.depth += 1;
395        if self.depth > MAX_DEPTH {
396            self.depth -= 1;
397            return err(CellError::Num);
398        }
399        let r = self.eval_inner(e);
400        self.depth -= 1;
401        r
402    }

404    fn eval_inner(&mut self, e: &Expr) -> Ev {
405        match e {
406            Expr::Number(n) => Ev::V(Value::number(*n)),
407            Expr::Text(t) => Ev::V(Value::Text(t.clone())),
408            Expr::Bool(b) => Ev::V(Value::Bool(*b)),
409            Expr::Error(x) => err(*x),
410            Expr::Missing => Ev::V(Value::Empty),
411            Expr::Paren(x) => self.eval(x),
412            Expr::Array(rows) => {
413                let h = rows.len();
414                let w = rows.first().map_or(0, Vec::len);
415                // A ragged literal is an error either way; one wide row and many empty ones must
416                // not reserve their rows-times-width product first.
417                if rows.iter().any(|row| row.len() != w) {
418                    return err(CellError::Value);
419                }
420                let mut data = Vec::with_capacity(h * w);
421                for row in rows {
422                    for el in row {
423                        let v = self.value(el);
424                        data.push(v);
425                    }
426                }
427                Ev::V(Array::new(h, w, data).map(Value::from).unwrap_or(Value::Error(CellError::Value)))
428            }
429            Expr::Ref(r) => match self.reference_areas(r) {
430                Ok(a) => Ev::R(a),
431                Err(x) => err(x),
432            },
433            Expr::Name(n) => self.name(n),
434            Expr::Struct(s) => self.structured(s),
435            Expr::Unary(op, x) => self.unary(*op, x),
436            Expr::Binary(op, a, b) => self.binary(*op, a, b),
437            Expr::Call(name, args) => self.call(name, args),
438            Expr::Invoke(callee, args) => {
439                let f = self.eval(callee);
440                match f {
441                    Ev::L(l) => self.apply_lambda(&l, args),
442                    Ev::V(Value::Error(x)) => err(x),
443                    _ => err(CellError::Value),
444                }
445            }
446        }
447    }

449    fn name(&mut self, n: &str) -> Ev {
450        if let Some((_, v)) = self.env.iter().rev().find(|(k, _)| crate::formula::same_name(k, n)) {
451            return v.clone();
452        }
453        let wb = self.host.workbook();
454        // Sheet-qualified `Sheet1!Name`. The parser stores the sheet unquoted, so it is used as
455        // is (`' Sales '` keeps its spaces) and split at the last `!` (`'a!b'!Name`).
456        let (scope_sheet, bare) = match n.rsplit_once('!') {
457            Some((s, b)) => match wb.sheet_index(s) {
458                Some(i) => (i, b),
459                None => return err(CellError::Name),
460            },
461            None => (self.sheet, n),
462        };
463        if let Some(def) = wb.name(bare, scope_sheet) {
464            let text = def.formula.clone();
465            let scope = def.scope.unwrap_or(self.sheet);
466            // Each use parses and evaluates the definition again: names that each use the one
467            // before twice double the work per name, so it is charged.
468            if !self.spend(text.len() as u64) {
469                return err(CellError::Calc);
470            }
471            return match crate::formula::parse_name(&text) {
472                Ok(expr) => {
473                    // Relative references in a name are relative to A1: used here, they move by
474                    // this cell's offset from it.
475                    let expr = if crate::formula::adjust::has_relative_refs(&expr) {
476                        crate::formula::adjust::shift_relative_wrapping(expr, self.at.row as i64, self.at.col as i64)
477                    } else {
478                        expr
479                    };
480                    let saved = self.sheet;
481                    // Unqualified references in a name refer to the scope sheet.
482                    self.sheet = scope;
483                    let r = self.eval(&expr);
484                    self.sheet = saved;
485                    r
486                }
487                Err(_) => err(CellError::Name),
488            };
489        }
490        // A table name alone means its data body.
491        if let Some((si, ti)) = wb.table(bare)
492            && let Some(t) = wb.sheet(si).and_then(|s| s.tables.get(ti))
493            && let Some(d) = t.data_range()
494        {
495            return Ev::R(vec![Area { sheet: si, range: d }]);
496        }
497        err(CellError::Name)
498    }

500    fn structured(&mut self, s: &StructRef) -> Ev {
501        let wb = self.host.workbook();
502        let found = if s.table.is_empty() {
503            wb.sheet(self.sheet).and_then(|sh| sh.tables.iter().position(|t| t.range.contains(self.at)).map(|ti| (self.sheet, ti)))
504        } else {
505            wb.table(&s.table)
506        };
507        let Some((si, ti)) = found else { return err(CellError::Ref) };
508        let Some(t) = wb.sheet(si).and_then(|sh| sh.tables.get(ti)) else { return err(CellError::Ref) };
509        let mut r0 = t.range.start.row;
510        let mut r1 = t.range.end.row;
511        let data = t.data_range();
512        let items = &s.specifiers;
513        if items.is_empty() || items == &[StructItem::Data] {
514            match data {
515                Some(d) => {
516                    r0 = d.start.row;
517                    r1 = d.end.row;
518                }
519                None => return err(CellError::Ref),
520            }
521        } else if items.contains(&StructItem::ThisRow) {
522            if self.at.row < t.range.start.row || self.at.row > t.range.end.row {
523                return err(CellError::Value);
524            }
525            r0 = self.at.row;
526            r1 = self.at.row;
527        } else if !items.contains(&StructItem::All) {
528            let has = |i: StructItem| items.contains(&i);
529            let hdr = t.header_row.then_some(t.range.start.row);
530            let tot = t.totals_row.then_some(t.range.end.row);
531            let mut lo = u32::MAX;
532            let mut hi = 0;
533            let mut add = |a: u32, b: u32| {
534                lo = lo.min(a);
535                hi = hi.max(b);
536            };
537            if has(StructItem::Headers) {
538                match hdr {
539                    Some(h) => add(h, h),
540                    None => return err(CellError::Ref),
541                }
542            }
543            if has(StructItem::Data)
544                && let Some(d) = data
545            {
546                add(d.start.row, d.end.row);
547            }
548            if has(StructItem::Totals) {
549                match tot {
550                    Some(h) => add(h, h),
551                    None => return err(CellError::Ref),
552                }
553            }
554            if lo > hi {
555                return err(CellError::Ref);
556            }
557            r0 = lo;
558            r1 = hi;
559        }
560        let (mut c0, mut c1) = (t.range.start.col, t.range.end.col);
561        if let Some(a) = &s.col_start {
562            let Some(i) = t.column_index(a) else { return err(CellError::Ref) };
563            c0 = t.range.start.col + i as u32;
564            c1 = c0;
565            if let Some(b) = &s.col_end {
566                let Some(j) = t.column_index(b) else { return err(CellError::Ref) };
567                let cj = t.range.start.col + j as u32;
568                c0 = c0.min(cj);
569                c1 = c1.max(cj);
570            }
571        }
572        Ev::R(vec![Area { sheet: si, range: RangeRef::new(CellRef::new(r0, c0), CellRef::new(r1, c1)) }])
573    }
574
575    fn unary(&mut self, op: UnOp, x: &Expr) -> Ev {
576        if let UnOp::Trim(mode) = op {
577            return self.trim_ref(mode, x);
578        }
579        if op == UnOp::Spill {
580            return match self.eval(x) {
581                Ev::R(areas) => match areas.as_slice() {
582                    [a] if a.range.is_single() => match self.host.spill_range(a.sheet, a.range.start) {
583                        Some(range) => Ev::R(vec![Area { sheet: a.sheet, range }]),
584                        None => err(CellError::Ref),
585                    },
586                    _ => err(CellError::Ref),
587                },
588                Ev::V(Value::Error(e)) => err(e),
589                _ => err(CellError::Ref),
590            };
591        }
592        if op == UnOp::At {
593            let ev = self.eval(x);
594            return match ev {
595                Ev::R(areas) => match areas.as_slice() {
596                    [a] => match self.implicit_intersect(a) {
597                        Some(c) => Ev::V(self.host.cell_value(a.sheet, c)),
598                        None => err(CellError::Value),
599                    },
600                    _ => err(CellError::Value),
601                },
602                Ev::V(v) => Ev::V(v.scalar()),
603                l => l,
604            };
605        }
606        let v = self.value(x);
607        let sys = self.host.workbook().date_system;
608        Ev::V(map_unary(
609            &v,
610            &|n| match op {
611                UnOp::Neg => Value::number(-n),
612                UnOp::Plus => Value::number(n),
613                UnOp::Percent => Value::number(n / 100.0),
614                UnOp::At | UnOp::Spill | UnOp::Trim(_) => Value::number(n),
615            },
616            op == UnOp::Plus,
617            sys,
618        ))
619    }
620
621    fn implicit_intersect(&self, a: &Area) -> Option<CellRef> {
622        let r = a.range;
623        if r.is_single() {
624            return Some(r.start);
625        }
626        if r.width() == 1 && r.start.row <= self.at.row && self.at.row <= r.end.row {
627            return Some(CellRef::new(self.at.row, r.start.col));
628        }
629        if r.height() == 1 && r.start.col <= self.at.col && self.at.col <= r.end.col {
630            return Some(CellRef::new(r.start.row, self.at.col));
631        }
632        // A 2-D range intersects on both the formula's row and column (observed in Excel, on
633        // another sheet too); one that includes the formula's own cell is a circular reference.
634        if r.start.row <= self.at.row && self.at.row <= r.end.row && r.start.col <= self.at.col && self.at.col <= r.end.col {
635            return Some(self.at);
636        }
637        None
638    }
639
640    fn binary(&mut self, op: BinOp, a: &Expr, b: &Expr) -> Ev {
641        match op {
642            BinOp::Range => {
643                let (x, y) = (self.eval(a), self.eval(b));
644                let (Ev::R(xa), Ev::R(ya)) = (&x, &y) else {
645                    if let Ev::V(Value::Error(e)) = x {
646                        return err(e);
647                    }
648                    if let Ev::V(Value::Error(e)) = y {
649                        return err(e);
650                    }
651                    return err(CellError::Value);
652                };
653                let all: Vec<&Area> = xa.iter().chain(ya.iter()).collect();
654                let Some(first) = all.first() else { return err(CellError::Ref) };
655                if all.iter().any(|ar| ar.sheet != first.sheet) {
656                    return err(CellError::Value);
657                }
658                let mut r = first.range;
659                for ar in &all {
660                    r = r.union(&ar.range);
661                }
662                Ev::R(vec![Area { sheet: first.sheet, range: r }])
663            }
664            BinOp::Union => {
665                let (x, y) = (self.eval(a), self.eval(b));
666                match (x, y) {
667                    (Ev::R(mut xa), Ev::R(ya)) => {
668                        // A union reused through LET doubles at each step: its areas count.
669                        if !self.spend((xa.len() as u64).saturating_add(ya.len() as u64)) {
670                            return err(CellError::Calc);
671                        }
672                        xa.extend(ya);
673                        Ev::R(xa)
674                    }
675                    (Ev::V(Value::Error(e)), _) | (_, Ev::V(Value::Error(e))) => err(e),
676                    _ => err(CellError::Value),
677                }
678            }
679            BinOp::Intersect => {
680                let (x, y) = (self.eval(a), self.eval(b));
681                match (x, y) {
682                    (Ev::R(xa), Ev::R(ya)) => {
683                        // Every pair is looked at, and each may be kept: charged before the loop,
684                        // so unions squared through LET can't build an enormous list.
685                        if !self.spend((xa.len() as u64).saturating_mul(ya.len() as u64)) {
686                            return err(CellError::Calc);
687                        }
688                        let mut out = Vec::new();
689                        for p in &xa {
690                            for q in &ya {
691                                if p.sheet == q.sheet
692                                    && let Some(i) = p.range.intersection(&q.range)
693                                {
694                                    out.push(Area { sheet: p.sheet, range: i });
695                                }
696                            }
697                        }
698                        if out.is_empty() { err(CellError::Null) } else { Ev::R(out) }
699                    }
700                    (Ev::V(Value::Error(e)), _) | (_, Ev::V(Value::Error(e))) => err(e),
701                    _ => err(CellError::Value),
702                }
703            }
704            _ => {
705                // A long chain such as A1+A2+…+A500 nests to the left: fold it in a loop instead
706                // of one level of recursion per operator (past the depth limit it was an error).
707                let mut rest = vec![(op, b)];
708                let mut left = a;
709                while let Expr::Binary(o, l, r) = left
710                    && !matches!(o, BinOp::Range | BinOp::Union | BinOp::Intersect)
711                {
712                    rest.push((*o, r.as_ref()));
713                    left = l.as_ref();
714                }
715                let mut acc = self.value(left);
716                while let Some((o, r)) = rest.pop() {
717                    let y = self.value(r);
718                    acc = binary_values(o, &acc, &y, self.host.workbook().date_system);
719                }
720                Ev::V(acc)
721            }
722        }
723    }

728        args.get(i).map(|e| self.eval(e))

733    fn arg_num(&mut self, args: &[Expr], i: usize, default: f64) -> Result<f64, CellError> {

735            None | Some(Expr::Missing) => Ok(default),
736            Some(e) => {

739            }
740        }

888    #[test]
889    fn lambda_calls_spend_fuel_and_run_out_quietly() {
890        // Three applications, one charge each: 3 inside the budget. (A nested MAP wouldn't do
891        // here: an inner MAP result is an array, and MAP of array elements is #CALC! on shape,
892        // which happens well before the budget does.)
893        let r = eval_with(3, "=LAMBDA(x,LAMBDA(y,LAMBDA(z,z+1)(3))(2))(1)");
894        assert_eq!(r, Value::number(4.0), "{r:?}");
895        // One charge short: the nest bottoms out in #CALC! instead of running unbounded.
896        let r = eval_with(2, "=LAMBDA(x,LAMBDA(y,LAMBDA(z,z+1)(3))(2))(1)");
897        assert!(matches!(r, Value::Error(CellError::Calc)), "{r:?}");
898        // One charge short of a single application: #CALC! with a zero budget.
899        let r = eval_with(0, "=LAMBDA(x,x+1)(2)");
900        assert!(matches!(r, Value::Error(CellError::Calc)), "{r:?}");
901        let r = eval_with(1, "=LAMBDA(x,x+1)(2)");
902        assert_eq!(r, Value::number(3.0));
903        // A flat helper map keeps working (3 of 3 invokes inside the budget).
904        let r = eval_with(3, "=MAP({1,2,3},LAMBDA(x,x*2))");
905        assert!(matches!(r, Value::Array(_)) && !matches!(&r, Value::Array(a) if a.data.iter().any(|v| matches!(v, Value::Error(_)))), "{r:?}");
906    }
