9/// A cell's formula.
10#[derive(Clone, Debug, PartialEq, Eq)]
11pub struct Formula {
12    /// The formula in A1 notation, without the leading `=`. Function names are the English
13    /// ones; newer functions come without their `_xlfn.` prefix.
14    pub text: String,
15    /// The range of a Ctrl+Shift+Enter array formula, on its top left cell.
16    pub array: Option<Range>,
17}

86#[derive(Clone, Debug, Default, PartialEq)]
87pub struct Sheet {
88    pub name: String,
89    pub visibility: Visibility,
90    /// The stored cells. A blank cell with plain formatting is not stored.
91    pub cells: BTreeMap<CellRef, Cell>,
92    /// Merged ranges of more than one cell.
93    pub merges: Vec<Range>,
94    /// Columns that have a ColInfo record, by zero-based index (0 to 255).
95    pub columns: BTreeMap<u16, Column>,
96    /// Rows that have a Row record, by zero-based index.
97    pub rows: BTreeMap<u32, Row>,
98    /// The default row height in twips, when the sheet sets one.
99    pub default_row_height: Option<u16>,
100    /// `None` when the sheet has no usable Window2 record.
101    pub view: Option<View>,
102    pub calc: Calc,
103    /// The sheet's `Print_Area` name as formula text (`Sheet1!$A$1:$C$9`).
104    pub print_area: Option<String>,
105    /// The sheet's `Print_Titles` name as formula text (`Sheet1!$1:$2`).
106    pub print_titles: Option<String>,
107    /// Data validation rules (Dv records), in file order.
108    pub validations: Vec<DataValidation>,
109    /// Cell comments (Note records with their text), in file order.
110    pub notes: Vec<Note>,
111    /// Conditional formats (CF records), in file order: the first has the highest priority.
112    pub conditional_formats: Vec<ConditionalFormat>,
113    /// Pictures placed on the sheet, in drawing order.
114    pub pictures: Vec<Picture>,
115    /// The AutoFilter's range: the sheet's `_FilterDatabase` name, when the sheet has an
116    /// AutoFilterInfo record (the name can outlive the filter). The criteria are not read.
117    pub autofilter: Option<Range>,
118}

241/// A data validation rule (Dv, [MS-XLS] 2.4.95), in the terms SpreadsheetML uses for the same
242/// rule (`dataValidation`).
243#[derive(Clone, Debug, PartialEq, Eq)]
244pub struct DataValidation {
245    /// The ranges it covers. A range over every BIFF8 row (column) is a whole column (row) of
246    /// the 1,048,576 × 16,384 grid, as SpreadsheetML stores it.
247    pub ranges: Vec<Range>,
248    pub kind: ValidationType,
249    /// `Between` for the types without an operator.
250    pub operator: Operator,
251    /// The first formula as A1 text without `=`, relative references from the top left cell of
252    /// all the ranges. A literal list is quoted, its items separated by commas and each `"`
253    /// doubled (`"Red,Green"`), as SpreadsheetML's `formula1`. Empty for "any value".
254    pub formula1: String,
255    /// The upper bound of `Between` and `NotBetween`.
256    pub formula2: Option<String>,
257    pub allow_blank: bool,
258    /// A list shows its drop-down arrow in the cell.
259    pub in_cell_dropdown: bool,
260    pub prompt_title: String,
261    pub prompt: String,
262    pub show_prompt: bool,
263    pub error_title: String,
264    pub error: String,
265    pub error_style: ErrorStyle,
266    pub show_error: bool,
267}

375/// A workbook read from a BIFF8 file.
376#[derive(Clone, Debug, Default, PartialEq)]
377pub struct Workbook {
378    /// Dates count from 1904-01-01 instead of 1899-12-31.
379    pub date_1904: bool,
380    pub precision_as_displayed: bool,
381    /// The first worksheet's calculation settings, which Excel applies to the workbook.
382    pub calc: Calc,
383    /// The worksheets, in workbook order. Chart, macro and dialog sheets are left out.
384    pub sheets: Vec<Sheet>,
385    /// The sheet the workbook was saved on, as an index into `sheets`: `None` when the file has
386    /// no Window1 record or names a sheet that was not read. It may be a hidden sheet; Excel
387    /// then opens on the first visible one.
388    pub active_sheet: Option<usize>,
389    pub names: Vec<DefinedName>,
390    /// The font of the Normal style: what an unformatted cell shows. `None` when the file has
391    /// no font at all.
392    pub normal_font: Option<Font>,
393    /// One entry per XF record, in file order. A format is *plain* when it is the Normal font
394    /// with everything else at [`CellFormat::default`]; cells, rows and columns with a plain
395    /// format carry `None` instead of its index.
396    pub formats: Vec<CellFormat>,
397    /// User-defined cell styles (name, formatting). Built-in styles are left out.
398    pub cell_styles: Vec<(String, CellFormat)>,
399}
