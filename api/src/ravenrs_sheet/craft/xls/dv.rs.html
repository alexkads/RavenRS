1//! Data validation: a DVal record ([MS-XLS] 2.4.96) starts a sheet's rules and says how many
2//! follow, and each Dv record after it (2.4.95) is one rule: its type, operator, alert style,
3//! messages, up to two formulas (DVParsedFormula, 2.5.198.9) and the ranges it covers (SqRefU,
4//! 2.5.247).
5//!
6//! Excel reads exactly the DVal's count of Dv records, and none without a DVal. It opens an
7//! unknown rule type as "any value", but refuses the whole file for an unknown alert style or
8//! operator, or for a formula or range list that runs past its record; here such a rule is
9//! skipped instead and the rest are kept.
10
11use crate::craft::sheet::{CellRef, MAX_COLS, MAX_ROWS, Range};

18/// Rules kept per sheet; more are skipped with a warning. Each rule's messages are charged to
19/// the text allowance, but its ranges are not, so the count is capped on its own.
20pub(crate) const MAX_RULES: usize = 65_536;
21
22/// The last row and column of the BIFF8 grid (65,536 × 256).
23const LAST_ROW: u16 = 0xFFFF;
24const LAST_COL: u16 = 0x00FF;
25
26/// PtgStr: a string constant ([MS-XLS] 2.5.198.89).
27const PTG_STR: u8 = 0x17;
28
29fn invalid(msg: &str) -> Error {
30    Error::Invalid(msg.into())
31}

33/// The number of Dv records a DVal says follow it: flags (2 bytes), the input window's
34/// position (4 + 4), the drop-down arrow's object id (4), then `idvMac` (4).
35pub(crate) fn dval(c: &mut Cursor<'_, '_>) -> Result<u32, Error> {
36    c.skip(14)?;
37    c.u32()
38}

40/// One Dv record as stored, before its formulas are decoded.
41pub(crate) struct Dv {
42    flags: u32,
43    prompt_title: String,
44    error_title: String,
45    prompt: String,
46    error: String,
47    rgce1: Vec<u8>,
48    rgce2: Vec<u8>,
49    ranges: Vec<Range>,
50}

52impl Dv {
53    /// Bytes held in the rule's four messages.
54    pub(crate) fn text_len(&self) -> usize {
55        [&self.prompt_title, &self.error_title, &self.prompt, &self.error].iter().map(|s| s.len()).sum()
56    }
57}

59/// Reads a Dv record: flags, the four messages, the two formulas and the ranges.
60pub(crate) fn dv(c: &mut Cursor<'_, '_>) -> Result<Dv, Error> {
61    let flags = c.u32()?;
62    let prompt_title = message(c)?;
63    let error_title = message(c)?;
64    let prompt = message(c)?;
65    let error = message(c)?;
66    let rgce1 = formula(c)?;
67    let rgce2 = formula(c)?;
68    let cref = usize::from(c.u16()?);
69    // Each Ref8U is 8 bytes; a count the record can't hold makes Excel refuse the file.
70    if cref.saturating_mul(8) > c.remaining() {
71        return Err(invalid("data validation ranges run past the end of the record"));
72    }
73    let mut ranges = Vec::with_capacity(cref);
74    for _ in 0..cref {
75        let (r0, r1, c0, c1) = (c.u16()?, c.u16()?, c.u16()?, c.u16()?);
76        ranges.extend(range(r0, r1, c0, c1));
77    }
78    Ok(Dv { flags, prompt_title, error_title, prompt, error, rgce1, rgce2, ranges })
79}

81/// A title or message. Excel stores an empty one as a single NUL character.
82fn message(c: &mut Cursor<'_, '_>) -> Result<String, Error> {
83    let s = c.string()?;
84    Ok(if s == "\0" { String::new() } else { s })
85}

87/// A DVParsedFormula's tokens: their size, 2 unused bytes (which hold leftover data in Excel's
88/// files, so they are never read), then the tokens.
89fn formula(c: &mut Cursor<'_, '_>) -> Result<Vec<u8>, Error> {
90    let cce = usize::from(c.u16()?);
91    c.skip(2)?;
92    if cce > c.remaining() {
93        return Err(invalid("data validation formula runs past the end of the record"));
94    }
95    (0..cce).map(|_| c.u8()).collect()
96}

98/// A Ref8U as a range. Columns past the BIFF8 grid are clamped to its last column, as Excel
99/// does; a reversed range is dropped (Excel drops it too). A range over every row (column) of
100/// the BIFF8 grid is a whole column (row), as SpreadsheetML stores the same rule.
101fn range(r0: u16, r1: u16, c0: u16, c1: u16) -> Option<Range> {
102    let (c0, c1) = (c0.min(LAST_COL), c1.min(LAST_COL));
103    if r0 > r1 || c0 > c1 {
104        return None;
105    }
106    let (r0, r1) = if r0 == 0 && r1 == LAST_ROW { (0, MAX_ROWS - 1) } else { (u32::from(r0), u32::from(r1)) };
107    let (c0, c1) = if c0 == 0 && c1 == LAST_COL { (0, MAX_COLS - 1) } else { (u32::from(c0), u32::from(c1)) };
108    Some(Range::new(CellRef::new(r0, c0), CellRef::new(r1, c1)))
109}

111/// The rule of a Dv, or `None` when it has no ranges, when Excel would refuse it (an alert
112/// style or operator outside the spec's values), or when a formula can't be decoded.
113pub(crate) fn validation(dv: Dv, links: &Links) -> Option<DataValidation> {
114    // Relative references are offsets from the top left cell of all the ranges, whatever order
115    // they are listed in.
116    let base = CellRef::new(dv.ranges.iter().map(|r| r.first.row).min()?, dv.ranges.iter().map(|r| r.first.col).min()?);
117    let f = dv.flags;
118    let bit = |n: u32| f & (1 << n) != 0;
119    let kind = match f & 0xF {
120        1 => ValidationType::Whole,
121        2 => ValidationType::Decimal,
122        3 => ValidationType::List,
123        4 => ValidationType::Date,
124        5 => ValidationType::Time,
125        6 => ValidationType::TextLength,
126        7 => ValidationType::Custom,
127        // 0, and the undefined 8 to 15, which Excel opens as "any value".
128        _ => ValidationType::Any,
129    };
130    let error_style = match (f >> 4) & 0x7 {
131        0 => ErrorStyle::Stop,
132        1 => ErrorStyle::Warning,
133        2 => ErrorStyle::Information,
134        _ => return None,
135    };
136    // The operator is undefined for "any value", lists and custom formulas, whatever it holds.
137    let compares = !matches!(kind, ValidationType::Any | ValidationType::List | ValidationType::Custom);
138    let operator = if compares {
139        match (f >> 20) & 0xF {
140            0 => Operator::Between,
141            1 => Operator::NotBetween,
142            2 => Operator::Equal,
143            3 => Operator::NotEqual,
144            4 => Operator::Greater,
145            5 => Operator::Less,
146            6 => Operator::GreaterOrEqual,
147            7 => Operator::LessOrEqual,
148            _ => return None,
149        }
150    } else {
151        Operator::Between
152    };
153    let text = |rgce: &[u8]| decode(rgce, &[], base, Mode::Cell, links).ok().map(|e| expr::print(&e));
154    let formula1 = match kind {
155        ValidationType::Any => String::new(),
156        _ if dv.rgce1.is_empty() => String::new(),
157        ValidationType::List if bit(7) => match literal_list(&dv.rgce1) {
158            Some(list) => list,
159            None => text(&dv.rgce1)?,
160        },
161        _ => text(&dv.rgce1)?,
162    };
163    let formula2 =
164        if compares && matches!(operator, Operator::Between | Operator::NotBetween) && !dv.rgce2.is_empty() { Some(text(&dv.rgce2)?) } else { None };
165    Some(DataValidation {
166        ranges: dv.ranges,
167        kind,
168        operator,
169        formula1,
170        formula2,
171        allow_blank: bit(8),
172        // fSuppressCombo hides a list's in-cell arrow; it means nothing for other kinds.
173        in_cell_dropdown: !(kind == ValidationType::List && bit(9)),
174        prompt_title: dv.prompt_title,
175        prompt: dv.prompt,
176        show_prompt: bit(18),
177        error_title: dv.error_title,
178        error: dv.error,
179        error_style,
180        show_error: bit(19),
181    })
182}

184/// A literal list: one PtgStr holding the items separated by NUL characters, written as
185/// SpreadsheetML writes it: in quotes, the items separated by commas, each `"` doubled. `None`
186/// when the formula isn't a lone PtgStr.
187fn literal_list(rgce: &[u8]) -> Option<String> {
188    let (&ptg, rest) = rgce.split_first()?;
189    let (&cch, rest) = rest.split_first()?;
190    let (&flags, chars) = rest.split_first()?;
191    if ptg != PTG_STR {
192        return None;
193    }
194    let cch = usize::from(cch);
195    let s = if flags & 1 != 0 {
196        if chars.len() != cch * 2 {
197            return None;
198        }
199        let units: Vec<u16> = chars.as_chunks::<2>().0.iter().map(|u| u16::from_le_bytes(*u)).collect();
200        String::from_utf16_lossy(&units)
201    } else {
202        if chars.len() != cch {
203            return None;
204        }
205        chars.iter().map(|&b| char::from(b)).collect()
206    };
207    let items: Vec<String> = s.split('\0').map(|item| item.replace('"', "\"\"")).collect();
208    Some(format!("\"{}\"", items.join(",")))
209}
