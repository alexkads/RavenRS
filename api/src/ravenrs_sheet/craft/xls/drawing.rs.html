1//! Pictures, from the Office Drawing records ([MS-ODRAW]): the workbook's
2//! MsoDrawingGroup holds the pictures (a BStore of FBSE entries, each with its BLIP), and each
3//! sheet's MsoDrawing records hold its shapes; a shape whose properties name a BLIP (`pib`,
4//! 0x0104) and that has a client anchor (cells and fractions of them) is a picture.
5
6use std::sync::Arc;
7
8use crate::craft::xls::model::ClientAnchor;
9
10/// Pictures read from one workbook at most.
11const MAX_BLIPS: usize = 4096;
12/// Bytes inflated from all of a workbook's metafiles together.
13const MAX_METAFILES: usize = 256 << 20;
14/// Nesting of drawing containers followed at most.
15const MAX_DEPTH: usize = 16;
16
17/// One stored picture: its bytes (shared by every shape that shows it) and MIME type, or `None`
18/// for a format that is not read (PICT, TIFF) or a damaged one, so that `pib` indexes still
19/// line up.
20pub(crate) type Blip = Option<(Arc<Vec<u8>>, &'static str)>;
21
22/// An Office Drawing record header: version/instance, type, length.
23fn header(b: &[u8], at: usize) -> Option<(u16, u16, usize)> {
24    let h = b.get(at..at.checked_add(8)?)?;
25    let vi = u16::from_le_bytes([*h.first()?, *h.get(1)?]);
26    let ty = u16::from_le_bytes([*h.get(2)?, *h.get(3)?]);
27    let len = u32::from_le_bytes([*h.get(4)?, *h.get(5)?, *h.get(6)?, *h.get(7)?]);
28    Some((vi, ty, usize::try_from(len).ok()?))
29}
30
31/// The records directly inside `b` (a container's body): (version/instance, type, body).
32fn children(b: &[u8]) -> Vec<(u16, u16, &[u8])> {
33    let mut out = Vec::new();
34    let mut at = 0usize;
35    while let Some((vi, ty, len)) = header(b, at) {
36        let start = at + 8;
37        // A record claiming more than is there keeps what is there: the last one in a buffer
38        // is often cut where its sheet's drawing records end.
39        let end = start.saturating_add(len).min(b.len());
40        out.push((vi, ty, b.get(start..end).unwrap_or(&[])));
41        at = end;
42    }
43    out
44}

46/// The pictures of the workbook's MsoDrawingGroup, in BStore order (`pib` 1 is the first).
47pub(crate) fn blip_store(dgg: &[u8]) -> Vec<Blip> {
48    let mut out = Vec::new();
49    let mut inflate_left = MAX_METAFILES;
50    for (_, ty, body) in children(dgg) {
51        if ty != 0xF000 {
52            continue;
53        }
54        for (_, ty, store) in children(body) {
55            if ty != 0xF001 {
56                continue;
57            }
58            for (_, ty, fbse) in children(store) {
59                if ty == 0xF007 && out.len() < MAX_BLIPS {
60                    out.push(fbse_blip(fbse, &mut inflate_left).map(|(b, m)| (Arc::new(b), m)));
61                }
62            }
63        }
64    }
65    out
66}

68/// A picture's bytes and MIME type.
69type Bytes = Option<(Vec<u8>, &'static str)>;
70
71/// The picture of one FBSE: 36 bytes of description, its name, then the BLIP record.
72/// `inflate_left` is what the workbook's metafiles may still inflate to.
73fn fbse_blip(fbse: &[u8], inflate_left: &mut usize) -> Bytes {
74    let name_len = usize::from(*fbse.get(33)?);
75    let rest = fbse.get(36usize.checked_add(name_len)?..)?;
76    let (vi, ty, len) = header(rest, 0)?;
77    let data = rest.get(8..8usize.checked_add(len)?.min(rest.len()))?;
78    let instance = vi >> 4;
79    // EMF and WMF: one or two identifiers, a metafile header, then the (usually zlib) bytes.
80    match (ty, instance) {
81        (0xF01A, 0x3D4) => return metafile(data, 1, false, inflate_left),
82        (0xF01A, 0x3D5) => return metafile(data, 2, false, inflate_left),
83        (0xF01B, 0x216) => return metafile(data, 1, true, inflate_left),
84        (0xF01B, 0x217) => return metafile(data, 2, true, inflate_left),
85        // DIB: identifiers, a tag byte, then a BITMAPINFOHEADER and the pixels.
86        (0xF01F, 0x7A8) => return dib_as_bmp(data.get(17..)?),
87        (0xF01F, 0x7A9) => return dib_as_bmp(data.get(33..)?),
88        _ => {}
89    }
90    // PNG and JPEG keep one or two 16-byte identifiers and a tag byte before the bytes.
91    let (uids, mime) = match (ty, instance) {
92        (0xF01E, 0x6E0) => (1, "image/png"),
93        (0xF01E, 0x6E1) => (2, "image/png"),
94        (0xF01D | 0xF02A, 0x46A | 0x6E2) => (1, "image/jpeg"),
95        (0xF01D | 0xF02A, 0x46B | 0x6E3) => (2, "image/jpeg"),
96        _ => return None,
97    };
98    let bytes = data.get(16 * uids + 1..)?;
99    (!bytes.is_empty()).then(|| (bytes.to_vec(), mime))
100}

102/// Largest metafile inflated from one BLIP.
103const MAX_METAFILE: usize = 64 << 20;
104
105/// An EMF or WMF BLIP's file: after `uids` identifiers comes the 34-byte OfficeArtMetafileHeader
106/// (uncompressed size, bounds, size in EMUs, saved size, compression 0 = DEFLATE in a zlib wrapper
107/// or 0xFE = none, filter), then the saved bytes. A WMF is stored without its placeable header;
108/// one is put back from the bounds so the picture keeps its size.
109fn metafile(data: &[u8], uids: usize, wmf: bool, inflate_left: &mut usize) -> Bytes {
110    let h = data.get(16 * uids..16 * uids + 34)?;
111    let u32_at = |at: usize| Some(u32::from_le_bytes(h.get(at..at + 4)?.try_into().ok()?));
112    let i32_at = |at: usize| Some(i32::from_le_bytes(h.get(at..at + 4)?.try_into().ok()?));
113    let size = usize::try_from(u32_at(0)?).ok()?;
114    let saved = usize::try_from(u32_at(28)?).ok()?;
115    let body = data.get(16 * uids + 34..)?;
116    let body = body.get(..saved.min(body.len()))?;
117    let bytes = match *h.get(32)? {
118        0x00 => {
119            let out = miniz_oxide::inflate::decompress_to_vec_zlib_with_limit(body, size.min(MAX_METAFILE).min(*inflate_left)).ok()?;
120            *inflate_left = inflate_left.saturating_sub(out.len());
121            out
122        }
123        0xFE => body.to_vec(),
124        _ => return None,
125    };
126    if bytes.is_empty() {
127        return None;
128    }
129    if !wmf {
130        return Some((bytes, "image/x-emf"));
131    }
132    // Bounds in the metafile's units, size in EMUs (914400 an inch): its units per inch.
133    let (l, t, r, b) = (i32_at(4)?, i32_at(8)?, i32_at(12)?, i32_at(16)?);
134    let width_emu = f64::from(i32_at(20)?);
135    let units = f64::from(r.saturating_sub(l));
136    let inch = if width_emu > 0.0 && units > 0.0 { (units * 914_400.0 / width_emu).round().clamp(1.0, 65_535.0) as u16 } else { 1440 };
137    let clamp16 = |v: i32| v.clamp(i32::from(i16::MIN), i32::from(i16::MAX)) as i16;
138    let mut out = Vec::with_capacity(22 + bytes.len());
139    out.extend_from_slice(&0x9AC6_CDD7u32.to_le_bytes());
140    out.extend_from_slice(&0u16.to_le_bytes());
141    for v in [l, t, r, b] {
142        out.extend_from_slice(&clamp16(v).to_le_bytes());
143    }
144    out.extend_from_slice(&inch.to_le_bytes());
145    out.extend_from_slice(&0u32.to_le_bytes());
146    // Checksum: the XOR of the ten words before it.
147    let sum = out.chunks(2).fold(0u16, |x, w| x ^ u16::from_le_bytes([w[0], *w.get(1).unwrap_or(&0)]));
148    out.extend_from_slice(&sum.to_le_bytes());
149    out.extend_from_slice(&bytes);
150    Some((out, "image/x-wmf"))
151}

153/// A packed DIB (header, palette, pixels) as a BMP file: the 14-byte file header in front, its
154/// pixel offset counted from the info header's size and palette.
155fn dib_as_bmp(dib: &[u8]) -> Bytes {
156    let u32_at = |at: usize| Some(u32::from_le_bytes(dib.get(at..at + 4)?.try_into().ok()?));
157    let header = u32_at(0)?;
158    if !(12..=124).contains(&header) || dib.len() <= header as usize {
159        return None;
160    }
161    let palette = if header == 12 {
162        0
163    } else {
164        let bits = u16::from_le_bytes(dib.get(14..16)?.try_into().ok()?);
165        let used = u32_at(32)?;
166        let colors = if used > 0 {
167            used.min(256)
168        } else if bits <= 8 {
169            1 << bits
170        } else {
171            0
172        };
173        // BI_BITFIELDS with a 40-byte header keeps its three masks after it.
174        let masks = if header == 40 && u32_at(16)? == 3 { 12 } else { 0 };
175        colors * 4 + masks
176    };
177    let offset = 14 + header + palette;
178    let total = u32::try_from(dib.len()).ok()?.checked_add(14)?;
179    let mut out = Vec::with_capacity(total as usize);
180    out.extend_from_slice(b"BM");
181    out.extend_from_slice(&total.to_le_bytes());
182    out.extend_from_slice(&0u32.to_le_bytes());
183    out.extend_from_slice(&offset.to_le_bytes());
184    out.extend_from_slice(dib);
185    Some((out, "image/bmp"))
186}

188/// A picture shape: its BLIP (1-based) and client anchor.
189#[derive(Clone, Copy, Debug, PartialEq, Eq)]
190pub(crate) struct PictureShape {
191    pub pib: u32,
192    pub anchor: ClientAnchor,
193}
194
195/// The picture shapes in a sheet's drawing (its MsoDrawing records joined).
196pub(crate) fn picture_shapes(dg: &[u8]) -> Vec<PictureShape> {
197    let mut out = Vec::new();
198    walk(dg, 0, &mut out);
199    out
200}
201
202fn walk(b: &[u8], depth: usize, out: &mut Vec<PictureShape>) {
203    if depth > MAX_DEPTH {
204        return;
205    }
206    for (vi, ty, body) in children(b) {
207        if ty == 0xF004 {
208            if let Some(p) = shape(body) {
209                out.push(p);
210            }
211        } else if vi & 0x0F == 0x0F {
212            walk(body, depth + 1, out);
213        }
214    }
215}
216
217/// A shape container's BLIP and anchor, when it has both.
218fn shape(sp: &[u8]) -> Option<PictureShape> {
219    let (mut pib, mut anchor) = (None, None);
220    for (vi, ty, body) in children(sp) {
221        match ty {
222            // Properties: `instance` of them, six bytes each (id with flags, value).
223            0xF00B => {
224                for p in body.chunks(6).take(usize::from(vi >> 4)) {
225                    if let [a, b, c, d, e, f] = *p
226                        && u16::from_le_bytes([a, b]) & 0x3FFF == 0x0104
227                    {
228                        pib = Some(u32::from_le_bytes([c, d, e, f]));
229                    }
230                }
231            }
232            // Client anchor: flags, then eight 16-bit values.
233            0xF010 => {
234                let mut v = [0u16; 8];
235                for (i, slot) in v.iter_mut().enumerate() {
236                    let at = 2 + i * 2;
237                    *slot = u16::from_le_bytes([*body.get(at)?, *body.get(at + 1)?]);
238                }
239                let [col1, dx1, row1, dy1, col2, dx2, row2, dy2] = v;
240                anchor = Some(ClientAnchor { col1, dx1, row1, dy1, col2, dx2, row2, dy2 });
241            }
242            _ => {}
243        }
244    }
245    Some(PictureShape { pib: pib.filter(|p| *p > 0)?, anchor: anchor? })
246}

252    fn rec(vi: u16, ty: u16, body: &[u8]) -> Vec<u8> {
253        [&vi.to_le_bytes()[..], &ty.to_le_bytes(), &(body.len() as u32).to_le_bytes(), body].concat()
254    }

256    #[test]
257    fn pictures_and_their_anchors_are_found_and_hostile_drawings_are_harmless() {
258        let png = [0x89, b'P', b'N', b'G'];
259        let blip = rec(0x6E00, 0xF01E, &[&[0u8; 17][..], &png].concat());
260        let fbse = rec(0x0062, 0xF007, &[&[0u8; 36][..], &blip].concat());
261        let wmf = rec(0x0062, 0xF007, &[&[0u8; 36][..], &rec(0x2160, 0xF01B, &[0; 60])].concat());
262        let dgg = rec(0x000F, 0xF000, &rec(0x002F, 0xF001, &[fbse, wmf].concat()));
263        let store = blip_store(&dgg);
264        assert_eq!(store.len(), 2);
265        assert_eq!(store[0], Some((Arc::new(png.to_vec()), "image/png")));
266        assert_eq!(store[1], None, "a damaged metafile keeps its slot");
267        // A shape with pib 1 and an anchor from B2 (half way across) to D5.
268        let opt = rec(0x0013, 0xF00B, &[&0x4104u16.to_le_bytes()[..], &1u32.to_le_bytes()].concat());
269        let anchor_body: Vec<u8> = [0u16, 1, 512, 1, 0, 3, 0, 4, 0].iter().flat_map(|v| v.to_le_bytes()).collect();
270        let sp = rec(0x000F, 0xF004, &[opt, rec(0x0000, 0xF010, &anchor_body)].concat());
271        let dg = rec(0x000F, 0xF002, &rec(0x000F, 0xF003, &sp));
272        let anchor = ClientAnchor { col1: 1, dx1: 512, row1: 1, dy1: 0, col2: 3, dx2: 0, row2: 4, dy2: 0 };
273        assert_eq!(picture_shapes(&dg), vec![PictureShape { pib: 1, anchor }]);
274        // Truncated or nonsense drawings give nothing, never a panic.
275        for bad in [&[][..], &[0xFF; 8][..], &[0; 70][..]] {
276            assert_eq!(metafile(bad, 1, false, &mut usize::MAX.clone()), None);
277            let _ = metafile(bad, 2, true, &mut usize::MAX.clone());
278            let _ = dib_as_bmp(bad);
279        }
280        for bad in [&dg[..dg.len() - 5], &[0xFF; 40][..], &[][..], &dgg[..20]] {
281            let _ = picture_shapes(bad);
282            let _ = blip_store(bad);
283        }
284    }

286    #[test]
287    fn metafiles_share_one_inflate_allowance() {
288        // A zlib metafile of 1000 zero bytes, as an EMF BLIP with one identifier.
289        let raw = vec![0u8; 1000];
290        let z = miniz_oxide::deflate::compress_to_vec_zlib(&raw, 6);
291        let mut header = Vec::new();
292        for v in [1000u32, 0, 0, 10, 10, 0, 0, z.len() as u32] {
293            header.extend_from_slice(&v.to_le_bytes());
294        }
295        header.extend_from_slice(&[0x00, 0xFE]);
296        let data = [&[0u8; 16][..], &header, &z].concat();
297        let mut left = 1500;
298        assert_eq!(metafile(&data, 1, false, &mut left).map(|(b, _)| b.len()), Some(1000));
299        assert_eq!(left, 500);
300        // The second would inflate past what is left: refused, not cut.
301        assert_eq!(metafile(&data, 1, false, &mut left), None);
302    }

304    #[test]
305    fn a_wmf_gets_its_placeable_header_back_and_a_dib_becomes_a_bmp() {
306        // Bounds 0,0–2000×1000 units over 2 × 1 inches (EMUs): 1000 units an inch; stored as is.
307        let wmf = b"WMF records".to_vec();
308        let mut header = Vec::new();
309        for v in [wmf.len() as u32, 0, 0, 2000, 1000, 2 * 914_400, 914_400, wmf.len() as u32] {
310            header.extend_from_slice(&v.to_le_bytes());
311        }
312        header.extend_from_slice(&[0xFE, 0xFE]);
313        let data = [&[0u8; 16][..], &header, &wmf].concat();
314        let (bytes, mime) = metafile(&data, 1, true, &mut usize::MAX.clone()).expect("wmf");
315        assert_eq!(mime, "image/x-wmf");
316        assert_eq!(&bytes[..4], &0x9AC6_CDD7u32.to_le_bytes());
317        assert_eq!(u16::from_le_bytes([bytes[14], bytes[15]]), 1000, "units per inch");
318        let sum = bytes[..20].chunks(2).fold(0u16, |x, w| x ^ u16::from_le_bytes([w[0], w[1]]));
319        assert_eq!(u16::from_le_bytes([bytes[20], bytes[21]]), sum);
320        assert_eq!(&bytes[22..], &wmf[..]);
321        // A 1 × 1, 24-bit DIB: the pixels start right after the file and info headers.
322        let mut dib = Vec::new();
323        for v in [40u32, 1, 1] {
324            dib.extend_from_slice(&v.to_le_bytes());
325        }
326        dib.extend_from_slice(&1u16.to_le_bytes());
327        dib.extend_from_slice(&24u16.to_le_bytes());
328        dib.extend_from_slice(&[0; 24]);
329        dib.extend_from_slice(&[0, 0, 255, 0]);
330        let (bmp, mime) = dib_as_bmp(&dib).expect("bmp");
331        assert_eq!(mime, "image/bmp");
332        assert_eq!(&bmp[..2], b"BM");
333        assert_eq!(u32::from_le_bytes(bmp[10..14].try_into().unwrap()), 54);
334        assert_eq!(bmp.len(), 14 + dib.len());
335    }
