1//! Password-encrypted workbooks ([MS-XLS] 2.2.10, [MS-OFFCRYPTO] 2.3.5, 2.3.6). A
2//! FilePass record in the workbook globals holds a salt and a password verifier; the record data
3//! after it is encrypted with RC4, rekeyed every 1024 bytes of the stream. Record headers, and the
4//! data of a few records, are left in the clear.
5//!
6//! Two schemes are read: RC4 (version 1.1), keyed from the password through MD5, and RC4
7//! CryptoAPI (versions 2.2, 3.2 and 4.2, what current Excel writes for an `.xls` saved with a
8//! password), keyed through SHA-1 with a 40- to 128-bit key. XOR obfuscation, and CryptoAPI with
9//! another cipher or hash, are reported as unsupported. Clean-room from Microsoft's public
10//! specifications.
11
12use md5::{Digest, Md5};
13use sha1::Sha1;
14
15/// Why a Workbook stream could not be decrypted.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub(crate) enum DecryptError {
18    WrongPassword,
19    /// A scheme that is not read, named to complete "uses {0} encryption".
20    Unsupported(String),
21    Malformed(String),
22}

24/// The password Excel encrypts with when a workbook has none of its own (a workbook protected
25/// against changes, not against opening). Excel tries it before asking.
26pub(crate) const DEFAULT_PASSWORD: &str = "VelvetSweatshop";

28/// FilePass: the globals record that marks an encrypted workbook.
29pub(crate) const FILEPASS: u16 = 0x002F;
30const EOF: u16 = 0x000A;
31const BOUNDSHEET: u16 = 0x0085;
32/// Records whose data is never encrypted: BOF, FilePass, UsrExcl, FileLock, InterfaceHdr,
33/// RRDInfo and RRDHead.
34const CLEAR: [u16; 7] = [0x0809, FILEPASS, 0x0194, 0x0195, 0x00E1, 0x0196, 0x0138];
35/// The RC4 key changes every this many bytes of the stream.
36const REKEY: usize = 1024;
37/// CryptoAPI `AlgID` for RC4 ([MS-OFFCRYPTO] 2.3.2); 0 means the flags decide, which without
38/// `fAES` is RC4 too.
39const CALG_RC4: u32 = 0x6801;
40/// CryptoAPI `AlgIDHash` for SHA-1; 0 means SHA-1 too.
41const CALG_SHA1: u32 = 0x8004;
42/// EncryptionHeader flags: `fExternal` (an external provider does the encryption) and `fAES`.
43const F_EXTERNAL: u32 = 0x10;
44const F_AES: u32 = 0x20;
45/// A CryptoAPI salt is 16 bytes, the verifier too, and the verifier's SHA-1 hash 20.
46const SALT_LEN: usize = 16;
47const SHA1_LEN: usize = 20;
48
49fn malformed(m: &str) -> DecryptError {
50    DecryptError::Malformed(m.into())
51}

53/// The record header at `pos`: its type and data length.
54fn header(stream: &[u8], pos: usize) -> Option<(u16, usize)> {
55    let h = stream.get(pos..pos.checked_add(4)?)?;
56    let id = u16::from_le_bytes([*h.first()?, *h.get(1)?]);
57    Some((id, usize::from(u16::from_le_bytes([*h.get(2)?, *h.get(3)?]))))
58}

60/// The FilePass record's data, when the workbook globals hold one.
61pub(crate) fn filepass(stream: &[u8]) -> Option<&[u8]> {
62    let mut pos = 0usize;
63    while let Some((id, len)) = header(stream, pos) {
64        let start = pos + 4;
65        match id {
66            EOF => return None,
67            FILEPASS => return stream.get(start..start.saturating_add(len)),
68            _ => pos = start + len,
69        }
70    }
71    None
72}

74/// Decrypts a Workbook stream in place with `password`. The FilePass record stays where it is
75/// (removing it would move the substreams the BoundSheet records point at).
76pub(crate) fn decrypt(stream: &mut [u8], password: &str) -> Result<(), DecryptError> {
77    let key = password_key(filepass(stream).ok_or_else(|| malformed("no FilePass record"))?, password)?;
78    let mut keystream = Keystream::new(key);
79    let mut pos = 0usize;
80    while let Some((id, len)) = header(stream, pos) {
81        let start = pos + 4;
82        let end = start.saturating_add(len).min(stream.len());
83        // BoundSheet8's stream position (its first 4 bytes) stays in the clear.
84        let clear = if CLEAR.contains(&id) {
85            len
86        } else if id == BOUNDSHEET {
87            4
88        } else {
89            0
90        };
91        let from = start.saturating_add(clear).min(end);
92        if let Some(data) = stream.get_mut(from..end) {
93            keystream.apply(from, data);
94        }
95        pos = start + len;
96    }
97    Ok(())
98}

100/// What the block keys are made from.
101enum Key {
102    /// RC4 1.1: the 5 bytes every block key is the MD5 of, with the block number.
103    Rc4([u8; 5]),
104    /// RC4 CryptoAPI: H0, which every block key is the SHA-1 of, with the block number, cut to
105    /// `bits`.
106    CryptoApi { h0: [u8; SHA1_LEN], bits: u32 },
107}
108
109impl Key {
110    /// The cipher for 1024-byte block `block`, at its start.
111    fn block(&self, block: usize) -> Rc4 {
112        let block = u32::try_from(block).unwrap_or(u32::MAX).to_le_bytes();
113        match self {
114            // MD5(key ‖ LE32(block)) ([MS-OFFCRYPTO] 2.3.6.2).
115            Key::Rc4(key) => Rc4::new(&md5(&[key, &block])),
116            // SHA-1(H0 ‖ LE32(block)), its first bits / 8 bytes; a 40-bit key is padded with
117            // zeros to 128 bits ([MS-OFFCRYPTO] 2.3.5.2).
118            Key::CryptoApi { h0, bits } => {
119                let hash = sha1(&[h0, &block]);
120                let mut key = [0u8; 16];
121                let len = usize::try_from(*bits / 8).unwrap_or(0).min(key.len());
122                key.iter_mut().zip(hash.iter()).take(len).for_each(|(k, h)| *k = *h);
123                Rc4::new(if *bits == 40 { &key } else { key.get(..len).unwrap_or(&key) })
124            }
125        }
126    }
127}

129/// The key every block key is derived from, after checking `password` against the FilePass
130/// verifier.
131fn password_key(filepass: &[u8], password: &str) -> Result<Key, DecryptError> {
132    let short = || malformed("truncated FilePass record");
133    match u16_at(filepass, 0).ok_or_else(short)? {
134        0 => return Err(DecryptError::Unsupported("XOR obfuscation".into())),
135        1 => {}
136        other => return Err(DecryptError::Malformed(format!("unknown encryption type {other}"))),
137    }
138    match (u16_at(filepass, 2).ok_or_else(short)?, u16_at(filepass, 4).ok_or_else(short)?) {
139        (1, 1) => rc4_key(filepass, password),
140        (2..=4, 2) => cryptoapi_key(filepass, password),
141        (major, minor) => Err(DecryptError::Unsupported(format!("version {major}.{minor}"))),
142    }
143}

145fn u16_at(b: &[u8], at: usize) -> Option<u16> {
146    let v = b.get(at..at.checked_add(2)?)?;
147    Some(u16::from_le_bytes([*v.first()?, *v.get(1)?]))
148}

150fn u32_at(b: &[u8], at: usize) -> Option<u32> {
151    let v = b.get(at..at.checked_add(4)?)?;
152    Some(u32::from_le_bytes([*v.first()?, *v.get(1)?, *v.get(2)?, *v.get(3)?]))
153}

155/// RC4 1.1 ([MS-OFFCRYPTO] 2.3.6.1, 2.3.6.2): the key, after checking the verifier.
156fn rc4_key(filepass: &[u8], password: &str) -> Result<Key, DecryptError> {
157    let short = || malformed("truncated FilePass record");
158    let salt = filepass.get(6..22).ok_or_else(short)?;
159    let verifier = filepass.get(22..38).ok_or_else(short)?;
160    let verifier_hash = filepass.get(38..54).ok_or_else(short)?;
161
162    // H0 = MD5(UTF-16LE password); H1 = MD5(16 × (first 5 bytes of H0 ‖ salt)); the key is the
163    // first 5 bytes of H1.
164    let pw: Vec<u8> = password.encode_utf16().flat_map(u16::to_le_bytes).collect();
165    let h0 = md5(&[&pw]);
166    let truncated = h0.get(..5).unwrap_or_default();
167    let mut m = Md5::new();
168    for _ in 0..16 {
169        m.update(truncated);
170        m.update(salt);
171    }
172    let mut key = [0u8; 5];
173    key.iter_mut().zip(m.finalize().iter()).for_each(|(k, h)| *k = *h);
174    let key = Key::Rc4(key);
175
176    // The verifier and its hash are encrypted one after the other with the block-0 key.
177    let mut rc4 = key.block(0);
178    let verifier: Vec<u8> = verifier.iter().map(|b| b ^ rc4.next_byte()).collect();
179    let verifier_hash: Vec<u8> = verifier_hash.iter().map(|b| b ^ rc4.next_byte()).collect();
180    if md5(&[&verifier]).as_slice() != verifier_hash.as_slice() {
181        return Err(DecryptError::WrongPassword);
182    }
183    Ok(key)
184}

186/// RC4 CryptoAPI ([MS-OFFCRYPTO] 2.3.5.1, 2.3.5.2, 2.3.5.6): after the version, the flags, the
187/// EncryptionHeader's size and the header itself, then the EncryptionVerifier (salt size, salt,
188/// encrypted verifier, verifier hash size, encrypted verifier hash). Every size is the file's
189/// word, so each is checked before it is used.
190fn cryptoapi_key(filepass: &[u8], password: &str) -> Result<Key, DecryptError> {
191    let short = || malformed("truncated FilePass record");
192    let unsupported = |scheme: String| Err(DecryptError::Unsupported(scheme));
193    let flags = u32_at(filepass, 6).ok_or_else(short)?;
194    let header_size = usize::try_from(u32_at(filepass, 10).ok_or_else(short)?).unwrap_or(usize::MAX);
195    let header_end = 14usize.saturating_add(header_size);
196    let header = filepass.get(14..header_end).ok_or_else(short)?;
197    // Eight words (Flags, SizeExtra, AlgID, AlgIDHash, KeySize, ProviderType and two reserved),
198    // then the provider's name, which says nothing the reader needs.
199    let (true, Some(alg), Some(alg_hash), Some(bits)) = (header.len() >= 32, u32_at(header, 8), u32_at(header, 12), u32_at(header, 16)) else {
200        return Err(malformed("EncryptionHeader too short"));
201    };
202    if flags & F_EXTERNAL != 0 {
203        return unsupported("CryptoAPI with an external provider".into());
204    }
205    if flags & F_AES != 0 {
206        return unsupported("CryptoAPI AES".into());
207    }
208    if !matches!(alg, 0 | CALG_RC4) {
209        return unsupported(format!("CryptoAPI with cipher 0x{alg:04X}"));
210    }
211    if !matches!(alg_hash, 0 | CALG_SHA1) {
212        return unsupported(format!("RC4 CryptoAPI with hash 0x{alg_hash:04X}"));
213    }
214    // 0 means 40 bits; otherwise 40 to 128 bits in steps of 8.
215    let bits = if bits == 0 { 40 } else { bits };
216    if !(40..=128).contains(&bits) || bits % 8 != 0 {
217        return unsupported(format!("RC4 CryptoAPI with a {bits}-bit key"));
218    }
219
220    let verifier_at = header_end;
221    let field = |at: usize, len: usize| filepass.get(verifier_at.saturating_add(at)..verifier_at.saturating_add(at + len));
222    if u32_at(filepass, verifier_at) != Some(SALT_LEN as u32) {
223        return Err(field(0, 4).map_or_else(short, |_| malformed("unexpected salt size")));
224    }
225    let salt = field(4, SALT_LEN).ok_or_else(short)?;
226    let verifier = field(4 + SALT_LEN, SALT_LEN).ok_or_else(short)?;
227    if u32_at(filepass, verifier_at.saturating_add(4 + 2 * SALT_LEN)) != Some(SHA1_LEN as u32) {
228        return Err(field(4 + 2 * SALT_LEN, 4).map_or_else(short, |_| malformed("unexpected verifier hash size")));
229    }
230    let verifier_hash = field(8 + 2 * SALT_LEN, SHA1_LEN).ok_or_else(short)?;
231
232    // H0 = SHA-1(salt ‖ UTF-16LE password).
233    let pw: Vec<u8> = password.encode_utf16().flat_map(u16::to_le_bytes).collect();
234    let key = Key::CryptoApi { h0: sha1(&[salt, &pw]), bits };
235
236    // As with RC4 1.1, the verifier and its hash run on through one block-0 cipher.
237    let mut rc4 = key.block(0);
238    let verifier: Vec<u8> = verifier.iter().map(|b| b ^ rc4.next_byte()).collect();
239    let verifier_hash: Vec<u8> = verifier_hash.iter().map(|b| b ^ rc4.next_byte()).collect();
240    if sha1(&[&verifier]).as_slice() != verifier_hash.as_slice() {
241        return Err(DecryptError::WrongPassword);
242    }
243    Ok(key)
244}

246fn md5(parts: &[&[u8]]) -> [u8; 16] {
247    let mut m = Md5::new();
248    for p in parts {
249        m.update(p);
250    }
251    let mut out = [0u8; 16];
252    out.iter_mut().zip(m.finalize().iter()).for_each(|(o, h)| *o = *h);
253    out
254}
255
256fn sha1(parts: &[&[u8]]) -> [u8; SHA1_LEN] {
257    let mut m = Sha1::new();
258    for p in parts {
259        m.update(p);
260    }
261    let mut out = [0u8; SHA1_LEN];
262    out.iter_mut().zip(m.finalize().iter()).for_each(|(o, h)| *o = *h);
263    out
264}

266/// The keystream by stream offset: byte `n` comes from block `n / 1024`'s key, at position
267/// `n % 1024`, whether or not the bytes before it were encrypted.
268struct Keystream {
269    key: Key,
270    block: usize,
271    rc4: Rc4,
272    /// The stream offset of the next byte `rc4` gives.
273    at: usize,
274}
275
276impl Keystream {
277    fn new(key: Key) -> Keystream {
278        Keystream { rc4: key.block(0), key, block: 0, at: 0 }
279    }
280
281    /// XORs `data`, found at stream offset `offset`, with the keystream there. Offsets only
282    /// move forward between calls in the normal case; going back rekeys.
283    fn apply(&mut self, offset: usize, data: &mut [u8]) {
284        for (n, b) in data.iter_mut().enumerate() {
285            *b ^= self.byte_at(offset.saturating_add(n));
286        }
287    }
288
289    fn byte_at(&mut self, offset: usize) -> u8 {
290        let block = offset / REKEY;
291        if block != self.block || offset < self.at {
292            self.block = block;
293            self.rc4 = self.key.block(block);
294            self.at = block * REKEY;
295        }
296        while self.at < offset {
297            self.rc4.next_byte();
298            self.at += 1;
299        }
300        self.at += 1;
301        self.rc4.next_byte()
302    }
303}

305/// The RC4 stream cipher.
306struct Rc4 {
307    s: [u8; 256],
308    i: u8,
309    j: u8,
310}
311
312impl Rc4 {
313    /// RC4 keyed with `key` (5 to 16 bytes here; an empty key acts as one zero byte).
314    fn new(key: &[u8]) -> Rc4 {
315        let mut s = [0u8; 256];
316        for (n, x) in (0u8..=255).zip(s.iter_mut()) {
317            *x = n;
318        }
319        let mut j = 0u8;
320        for n in 0..256 {
321            let k = key.get(n % key.len().max(1)).copied().unwrap_or(0);
322            j = j.wrapping_add(s[n]).wrapping_add(k);
323            s.swap(n, usize::from(j));
324        }
325        Rc4 { s, i: 0, j: 0 }
326    }
327
328    fn next_byte(&mut self) -> u8 {
329        self.i = self.i.wrapping_add(1);
330        self.j = self.j.wrapping_add(self.s[usize::from(self.i)]);
331        self.s.swap(usize::from(self.i), usize::from(self.j));
332        let t = self.s[usize::from(self.i)].wrapping_add(self.s[usize::from(self.j)]);
333        self.s[usize::from(t)]
334    }
335}

337/// Encrypts a Workbook stream the way [`decrypt`] reads it, for tests: the FilePass record
338/// (RC4 1.1 with a fixed salt and verifier) goes right after the globals' BOF. `None` when the
339/// stream does not start with a record.
340pub(crate) fn encrypt(stream: &[u8], password: &str) -> Option<Vec<u8>> {
341    let salt: [u8; 16] = std::array::from_fn(|n| n as u8 + 0x40);
342    let verifier: [u8; 16] = std::array::from_fn(|n| n as u8 + 0x60);
343    let pw: Vec<u8> = password.encode_utf16().flat_map(u16::to_le_bytes).collect();
344    let h0 = md5(&[&pw]);
345    let mut m = Md5::new();
346    for _ in 0..16 {
347        m.update(h0.get(..5)?);
348        m.update(salt);
349    }
350    let mut key = [0u8; 5];
351    key.iter_mut().zip(m.finalize().iter()).for_each(|(k, h)| *k = *h);
352    let mut rc4 = Key::Rc4(key).block(0);
353    let mut filepass = vec![1, 0, 1, 0, 1, 0];
354    filepass.extend(salt);
355    filepass.extend(verifier.iter().map(|b| b ^ rc4.next_byte()));
356    filepass.extend(md5(&[&verifier]).iter().map(|b| b ^ rc4.next_byte()));
357    with_filepass(stream, &filepass, password)
358}

360/// The FilePass data [`encrypt_cryptoapi`] writes: RC4 CryptoAPI version 4.2, as current Excel
361/// writes it, with a `bits`-bit key (40 to 128) and a fixed salt and verifier.
362pub(crate) fn cryptoapi_filepass(password: &str, bits: u32) -> Vec<u8> {
363    let salt: [u8; SALT_LEN] = std::array::from_fn(|n| (n as u8).wrapping_mul(7).wrapping_add(0x11));
364    let verifier: [u8; SALT_LEN] = std::array::from_fn(|n| n as u8 + 0x80);
365    let pw: Vec<u8> = password.encode_utf16().flat_map(u16::to_le_bytes).collect();
366    let mut rc4 = Key::CryptoApi { h0: sha1(&[&salt, &pw]), bits }.block(0);
367    // fCryptoAPI and fDocProps, as Excel sets them.
368    let flags: u32 = 0x0C;
369    let provider = "Test provider\0".encode_utf16().flat_map(u16::to_le_bytes);
370    let header: Vec<u8> = [flags, 0, CALG_RC4, CALG_SHA1, bits, 1, 0, 0].iter().flat_map(|w| w.to_le_bytes()).chain(provider).collect();
371    let mut filepass = vec![1, 0, 4, 0, 2, 0];
372    filepass.extend(flags.to_le_bytes());
373    filepass.extend((header.len() as u32).to_le_bytes());
374    filepass.extend(header);
375    filepass.extend((SALT_LEN as u32).to_le_bytes());
376    filepass.extend(salt);
377    filepass.extend(verifier.iter().map(|b| b ^ rc4.next_byte()));
378    filepass.extend((SHA1_LEN as u32).to_le_bytes());
379    filepass.extend(sha1(&[&verifier]).iter().map(|b| b ^ rc4.next_byte()));
380    filepass
381}

383/// [`encrypt`] with RC4 CryptoAPI ([`cryptoapi_filepass`]).
384pub(crate) fn encrypt_cryptoapi(stream: &[u8], password: &str, bits: u32) -> Option<Vec<u8>> {
385    with_filepass(stream, &cryptoapi_filepass(password, bits), password)
386}

388/// `stream` with a FilePass record holding `filepass` right after its first BOF, encrypted with
389/// `password`.
390fn with_filepass(stream: &[u8], filepass: &[u8], password: &str) -> Option<Vec<u8>> {
391    let (_, bof) = header(stream, 0)?;
392    let bof_len = 4 + bof;
393    let mut out = stream.get(..bof_len)?.to_vec();
394    out.extend(FILEPASS.to_le_bytes());
395    out.extend(u16::try_from(filepass.len()).ok()?.to_le_bytes());
396    out.extend(filepass);
397    let shift = u32::try_from(out.len() - bof_len).ok()?;
398    let mut rest = stream.get(bof_len..)?.to_vec();
399    // BoundSheet8 positions move by the FilePass record's size.
400    let mut pos = 0;
401    while let Some((id, len)) = header(&rest, pos) {
402        if id == BOUNDSHEET {
403            let at = rest.get_mut(pos + 4..pos + 8)?;
404            let p = u32::from_le_bytes([*at.first()?, *at.get(1)?, *at.get(2)?, *at.get(3)?]).checked_add(shift)?;
405            at.copy_from_slice(&p.to_le_bytes());
406        }
407        pos += 4 + len;
408    }
409    out.extend(rest);
410    // The cipher is its own inverse.
411    decrypt(&mut out, password).ok()?;
412    Some(out)
