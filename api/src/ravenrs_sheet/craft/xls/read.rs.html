61/// The root-level stream called `name` (compared as compound files compare names).
62fn root_stream(cf: &crate::craft::cfb::CompoundFile<'_>, name: &str) -> Option<usize> {
63    cf.child(0, name).filter(|&id| cf.entry(id).is_some_and(|e| e.kind == crate::craft::cfb::EntryKind::Stream))
64}

66/// The Workbook stream of a compound file, or why there is none.
67fn workbook_stream(bytes: &[u8]) -> Result<Vec<u8>, Error> {
68    let cf = crate::craft::cfb::CompoundFile::open(bytes).map_err(|e| invalid(e.to_string()))?;
69    let Some(id) = root_stream(&cf, "Workbook") else {
70        return Err(if root_stream(&cf, "Book").is_some() {
71            Error::Biff5
72        } else {
73            invalid("the file has no Workbook stream, so it is not an Excel workbook")
74        });
75    };
76    cf.read_capped(id, MAX_STREAM).map_err(|e| match e {
77        crate::craft::cfb::Error::LimitExceeded(_) => Error::TooLarge("the Workbook stream is larger than 1 GiB".into()),
78        e => invalid(e.to_string()),
79    })
80}

82/// Whether the bytes are a compound file holding an Excel workbook stream (BIFF8 `Workbook`
83/// or BIFF5 `Book`). The stream itself is not read.
84pub fn is_xls(bytes: &[u8]) -> bool {
85    crate::craft::cfb::CompoundFile::open(bytes).is_ok_and(|cf| root_stream(&cf, "Workbook").or_else(|| root_stream(&cf, "Book")).is_some())

88/// Most shared formulas one sheet expands, and the longest template text one attach may
89/// produce: a template shared by its followers was costing every one of them a full decode and
90/// print however often that could never succeed, so each anchor is tried once and failed ones
91/// are skipped for the rest of the sheet. Beyond the cap the remaining references keep their
92/// cached values.
93const MAX_SHARED_ATTACHES: usize = 262_144;
94const MAX_SHARED_TEMPLATE: usize = 8192;

96struct Budget {
97    limits: Limits,
98    cells: usize,
99    text: usize,
100    records: usize,
101    /// A shared formula's text was refused: no later shared formula, in any sheet, is expanded.
102    shared_text_spent: bool,
103}
104
105impl Budget {
106    fn new(limits: Limits) -> Self {
107        Budget { limits, cells: 0, text: 0, records: 0, shared_text_spent: false }
108    }
109    fn record(&mut self) -> Result<(), Error> {
110        self.records = self.records.saturating_add(1);
111        if self.records > self.limits.max_records {
112            return Err(Error::TooLarge("too many records".into()));
113        }
114        Ok(())
115    }
116    fn cell(&mut self) -> Result<(), Error> {
117        self.cells = self.cells.saturating_add(1);
118        if self.cells > self.limits.max_cells {
119            return Err(Error::TooLarge(format!("more than {} cells", self.limits.max_cells)));
120        }
121        Ok(())
122    }
123    /// Charges `len` bytes of text if they fit the allowance, leaving it untouched otherwise:
124    /// for text the read can do without (expanded shared formulas keep their cached values).
125    fn try_text(&mut self, len: usize) -> bool {
126        let total = self.text.saturating_add(len);
127        let fits = total <= self.limits.max_text;
128        if fits {
129            self.text = total;
130        }
131        fits
132    }
133    /// [`try_text`](Self::try_text) for a shared formula's text. The first refusal ends shared
134    /// expansion for the whole workbook, so a shorter formula on a later sheet doesn't fit in the
135    /// room the refused one left, and refused formulas aren't decoded again on each sheet.
136    fn try_shared_text(&mut self, len: usize) -> bool {
137        self.shared_text_spent = self.shared_text_spent || !self.try_text(len);
138        !self.shared_text_spent
139    }
140    fn text(&mut self, len: usize) -> Result<(), Error> {
141        self.text = self.text.saturating_add(len);
142        if self.text > self.limits.max_text {
143            return Err(Error::TooLarge(format!("more than {} bytes of text", self.limits.max_text)));
144        }
145        Ok(())
146    }
147}

205/// Decrypts an encrypted stream: with `password`, or without one with Excel's default password,
206/// as Excel opens a workbook protected only against changes.
207fn decrypt(stream: &mut [u8], password: Option<&str>) -> Result<(), Error> {
208    use crate::craft::xls::crypt::DecryptError;
209    match crate::craft::xls::crypt::decrypt(stream, password.unwrap_or(crate::craft::xls::crypt::DEFAULT_PASSWORD)) {
210        Ok(()) => Ok(()),
211        Err(DecryptError::WrongPassword) if password.is_none() => Err(Error::Encrypted),
212        Err(DecryptError::WrongPassword) => Err(Error::WrongPassword),
213        Err(DecryptError::Unsupported(s)) => Err(Error::UnsupportedEncryption(s)),
214        Err(DecryptError::Malformed(s)) => Err(invalid(format!("the encryption record is damaged: {s}"))),
215    }
216}

218fn read_stream(mut stream: Vec<u8>, limits: Limits, password: Option<&str>) -> Result<Read, Error> {
219    if crate::craft::xls::crypt::filepass(&stream).is_some() {
220        decrypt(&mut stream, password)?;
221    }
222    let mut budget = Budget::new(limits);
223    let mut report = Report::default();
224    let g = globals(&stream, &mut budget, &mut report)?;
225
226    let mut wb = Workbook { date_1904: g.date1904, precision_as_displayed: g.precision_as_displayed, ..Workbook::default() };
227    // Per XF: whether its format is plain.
228    let plain = styles::build(&g.styles, &g.formats, &mut wb);
229    if g.styles.short {
230        report.warn(Warning::ShortFormatting);
231    }
232
233    // BoundSheet8 index → index in `wb.sheets`.
234    let mut model_index: Vec<Option<usize>> = Vec::with_capacity(g.sheets.len());
235    let mut parsed = Vec::new();
236    for bs in &g.sheets {
237        if bs.kind != 0 {
238            report.warn(match bs.kind {
239                2 => Warning::ChartSheet,
240                1 => Warning::MacroSheet,
241                _ => Warning::OtherSheet,
242            });
243            model_index.push(None);
244            continue;
245        }
246        let Some(st) = worksheet(&stream, bs, &g, &plain, &mut budget, &mut report)? else {
247            report.warn(Warning::DialogSheet);
248            model_index.push(None);
249            continue;
250        };
251        model_index.push(Some(parsed.len()));
252        parsed.push((bs.visibility, st));
253    }
254    // Every sheet's mandatory text is charged by now; shared formulas take what is left.
255    let mut filtered = Vec::with_capacity(parsed.len());
256    for (visibility, st) in parsed {
257        filtered.push(st.filtered);
258        let mut sheet = finish_sheet(st, &mut budget, &mut report);
259        sheet.visibility = visibility;
260        // Every worksheet repeats the workbook's calculation settings; Excel applies the first.
261        if wb.sheets.is_empty() {
262            wb.calc = sheet.calc;
263        }
264        wb.sheets.push(sheet);
265    }
266    if wb.sheets.is_empty() {
267        return Err(invalid("the workbook has no worksheets"));
268    }
269    wb.active_sheet = g.active_tab.and_then(|t| model_index.get(usize::from(t)).copied().flatten());
270    defined_names(&g, &model_index, &filtered, &mut wb, &mut report);
271    Ok(Read { workbook: wb, warnings: report.warnings })
272}

274fn bof(rec: &Record<'_>) -> Result<(u16, u16), Error> {
275    if rec.id != BOF {
276        return Err(invalid("a substream does not start with a BOF record"));
277    }
278    let mut c = Cursor::new(&rec.parts);
279    Ok((c.u16()?, c.u16()?))
280}

282#[allow(clippy::cognitive_complexity, reason = "one arm per globals record of [MS-XLS] 2.1.7.20.3")]
283fn globals(stream: &[u8], budget: &mut Budget, report: &mut Report) -> Result<Globals, Error> {
284    let mut recs = Records::new(stream, 0);
285    let first = recs.next_record()?.ok_or_else(|| invalid("empty Workbook stream"))?;
286    let (vers, dt) = bof(&first)?;
287    if vers != BIFF8 {
288        return Err(if vers == 0x0500 { Error::Biff5 } else { invalid("unsupported BIFF version (only Excel 97-2003 BIFF8 files are read)") });
289    }
290    if dt != 0x0005 {
291        return Err(invalid("the Workbook stream does not start with the workbook globals"));
292    }
293    let mut g = Globals::default();
294    let mut offsets = HashSet::new();
295    let mut dgg: Vec<u8> = Vec::new();
296    while let Some(rec) = recs.next_record()? {
297        budget.record()?;
298        let mut c = Cursor::new(&rec.parts);
299        match rec.id {
300            EOF => break,
301            // FilePass: only reached once the stream was decrypted.
302            0x002F => {}
303            0x0085 => {
304                if g.sheets.len() >= MAX_SHEETS {
305                    return Err(Error::TooLarge("too many sheets".into()));
306                }
307                let pos = c.u32()? as usize;
308                if !offsets.insert(pos) {
309                    return Err(invalid("two sheets point at the same substream"));
310                }
311                let state = c.u8()?;
312                let kind = c.u8()?;
313                let name = c.short_string()?;
314                let visibility = match state & 3 {
315                    0 => Visibility::Visible,
316                    1 => Visibility::Hidden,
317                    _ => Visibility::VeryHidden,
318                };
319                g.links.sheets.push(name.clone());
320                g.sheets.push(BoundSheet { pos, name, visibility, kind });
321            }
322            0x0022 => g.date1904 = c.u16()? == 1,
323            0x000E => g.precision_as_displayed = c.u16()? == 0,
324            0x041E => {
325                let ifmt = c.u16()?;
326                let cch = usize::from(c.u16()?);
327                let high = c.u8()? & 1 != 0;
328                // Some writers count one character more than the record holds; Excel opens
329                // those files, so keep the characters that are there.
330                let fit = c.remaining() / if high { 2 } else { 1 };
331                if cch > fit {
332                    report.warn(Warning::ShortNumberFormat);
333                }
334                let code = c.chars(cch.min(fit), high)?;
335                budget.text(code.len())?;
336                g.formats.insert(ifmt, code);
337            }
338            // Window1: position, size and flags (five u16), then the active sheet. Only the first
339            // window counts; a record too short for it is ignored.
340            0x003D if g.active_tab.is_none() && c.remaining() >= 12 => {
341                c.skip(10)?;
342                g.active_tab = Some(c.u16()?);
343            }
344            0x00E0 => g.styles.xf(&mut c, MAX_XFS),
345            0x0031 => g.styles.font(&mut c),
346            0x0092 => g.styles.palette(&mut c),
347            0x0293 => g.styles.style(&mut c),
348            // MsoDrawingGroup: one buffer of drawing records, cut at MAX_DRAWING.
349            0x00EB if dgg.len() < MAX_DRAWING => dgg.extend(c.rest()),
350            0x00FC => g.sst = shared_strings(&mut c, budget)?,
351            0x01AE => {
352                let _ctab = c.u16()?;
353                let cch = c.u16()?;
354                g.links.books.push(SupBook { internal: cch == 0x0401, addin: cch == 0x3A01, names: vec![] });
355            }
356            0x0023 => {
357                // ExternName: flags, then an AddinUdf or ExternDocName body; both start with
358                // four bytes before the name.
359                c.u16()?;
360                c.skip(4)?;
361                let name = c.short_string()?;
362                if let Some(book) = g.links.books.last_mut()
363                    && book.names.len() < MAX_NAMES
364                {
365                    book.names.push(name);
366                }
367            }
368            0x0017 => {
369                let n = usize::from(c.u16()?);
370                for _ in 0..n.min(c.remaining() / 6) {
371                    let book = c.u16()?;
372                    let first = c.u16()? as i16;
373                    let last = c.u16()? as i16;
374                    g.links.xti.push(Xti { book, first, last });
375                }
376            }
377            0x0018 => {
378                if g.names.len() >= MAX_NAMES {
379                    continue;
380                }
381                let lbl = label(&mut c)?;
382                g.links.names.push(lbl.name.clone());
383                g.names.push(lbl);
384            }
385            _ => {}
386        }
387    }
388    // The same test `defined_names` applies, before any cell formula is decoded.
389    let skipped = g.names.iter().map(|l| !l.rgce.is_empty() && decode(&l.rgce, &l.rgcb, CellRef::new(0, 0), Mode::Name, &g.links).is_err()).collect();
390    g.links.skipped = skipped;
391    g.blips = drawing::blip_store(&dgg);
392    Ok(g)
393}

413fn label(c: &mut Cursor<'_, '_>) -> Result<Lbl, Error> {
414    let flags = c.u16()?;
415    let _key = c.u8()?;
416    let cch = usize::from(c.u8()?);
417    let cce = usize::from(c.u16()?);
418    c.skip(2)?;
419    let itab = c.u16()?;
420    c.skip(4)?;
421    let high = c.u8()? & 1 != 0;
422    let raw = c.chars(cch, high)?;
423    let builtin = flags & 0x0020 != 0;
424    let (name, builtin) = if builtin {
425        let code = raw.chars().next().map_or(0, |ch| ch as u32);
426        let code = u8::try_from(code).unwrap_or(u8::MAX);
427        (BUILTIN_NAMES.get(usize::from(code)).map_or_else(|| raw.clone(), |n| (*n).to_string()), Some(code))
428    } else {
429        (raw, None)
430    };
431    let rest = c.rest();
432    let rgce = rest.get(..cce).unwrap_or(&rest).to_vec();
433    let rgcb = rest.get(cce..).unwrap_or(&[]).to_vec();
434    Ok(Lbl { name, builtin, hidden: flags & 1 != 0, macro_name: flags & 0x000A != 0, itab, rgce, rgcb })
435}

437fn shared_strings(c: &mut Cursor<'_, '_>, budget: &mut Budget) -> Result<Vec<Arc<str>>, Error> {
438    let _total = c.u32()?;
439    let unique = c.u32()? as usize;
440    if unique > MAX_STRINGS {
441        return Err(Error::TooLarge("too many shared strings".into()));
442    }
443    // Each string takes at least three bytes: the declared count can't exceed what's there.
444    let mut out = Vec::with_capacity(unique.min(c.remaining() / 3));
445    for _ in 0..unique {
446        if c.remaining() == 0 {
447            break;
448        }
449        let s = c.rich_string()?;
450        budget.text(s.len())?;
451        out.push(Arc::from(s));
452    }
453    Ok(out)
454}

468fn parsed(c: &mut Cursor<'_, '_>) -> Result<Parsed, Error> {
469    let cce = usize::from(c.u16()?);
470    let rest = c.rest();
471    if cce > rest.len() {
472        return Err(invalid("formula runs past the end of its record"));
473    }
474    Ok(Parsed { rgce: rest.get(..cce).unwrap_or(&[]).to_vec(), rgcb: rest.get(cce..).unwrap_or(&[]).to_vec() })
475}

509impl SheetState<'_> {
510    /// Records why a formula was kept as its value (each reason once).
511    fn unsupported(&mut self, why: &'static str) {
512        if !self.unsupported.contains(&why) {
513            self.unsupported.push(why);
514        }
515    }
516
517    /// The format index a cell, row or column keeps: none for a plain or missing XF.
518    fn format(&self, ixfe: u16) -> Option<u16> {
519        (self.plain.get(usize::from(ixfe)) == Some(&false)).then_some(ixfe)
520    }
521
522    fn pos(&mut self, row: u16, col: u16) -> Option<CellRef> {
523        let (r, c) = (u32::from(row), u32::from(col));
524        if r >= GRID_ROWS || c >= GRID_COLS {
525            self.dropped += 1;
526            return None;
527        }
528        Some(CellRef::new(r, c))
529    }
530
531    fn put(&mut self, budget: &mut Budget, row: u16, col: u16, ixfe: u16, value: Value) -> Result<Option<CellRef>, Error> {
532        let Some(pos) = self.pos(row, col) else { return Ok(None) };
533        let format = self.format(ixfe);
534        if value.is_empty() && format.is_none() {
535            return Ok(Some(pos));
536        }
537        budget.cell()?;
538        self.sheet.cells.insert(pos, Cell { value, format, formula: None });
539        Ok(Some(pos))
540    }
541
542    /// Decodes a formula and attaches it to the cell at `pos`, or records why it can't be.
543    fn attach(&mut self, pos: CellRef, p: &Parsed, mode: Mode, array: Option<Range>) {
544        match decode(&p.rgce, &p.rgcb, pos, mode, &self.g.links) {
545            // A one-cell Ctrl+Shift+Enter array keeps its range: it evaluates as an array.
546            Ok(e) => self.sheet.cells.entry(pos).or_default().formula = Some(Formula { text: crate::craft::xls::expr::print(&e), array }),
547            Err(Unsupported(why)) => self.unsupported(why),
548        }
549    }
550
551    fn formula_at(&self, pos: CellRef) -> Option<&Formula> {
552        self.sheet.cells.get(&pos).and_then(|c| c.formula.as_ref())
553    }
554}

556/// RK number ([MS-XLS] 2.5.217): a 30-bit integer or the top of an IEEE double, maybe ×100.
557fn rk(v: u32) -> Value {
558    let n = if v & 2 != 0 { f64::from((v as i32) >> 2) } else { f64::from_bits(u64::from(v & !3) << 32) };
559    Value::number(if v & 1 != 0 { n / 100.0 } else { n })
560}

562/// Attaches every pending formula to its anchor's template. An anchor whose template can't be
563/// read is refused once: its followers re-decode the same template for nothing, so they are
564/// skipped instead (they keep their cached values).
565fn resolve_pending(
566    st: &mut SheetState<'_>,
567    budget: &mut Budget,
568    arrays: &HashMap<CellRef, (Range, Parsed)>,
569    shared: &HashMap<CellRef, Parsed>,
570    pending: &[Pending],
571) {
572    const TOO_MUCH: &str = "more shared-formula text than the read allows";
573    let mut failed: HashSet<CellRef> = HashSet::new();
574    let mut shared_attaches = 0usize;
575    for p in pending {
576        if let Some((range, f)) = arrays.get(&p.anchor) {
577            if p.pos == p.anchor {
578                st.attach(p.pos, f, Mode::Cell, Some(*range));
579            }
580        } else if let Some(f) = shared.get(&p.anchor) {
581            if budget.shared_text_spent {
582                st.unsupported(TOO_MUCH);
583                continue;
584            }
585            if failed.contains(&p.anchor) || shared_attaches >= MAX_SHARED_ATTACHES {
586                continue;
587            }
588            shared_attaches += 1;
589            st.attach(p.pos, f, Mode::Shared, None);
590            // Every follower keeps its own text: charge it to the workbook's text allowance, and
591            // past it drop the formula again so the cell keeps its cached value.
592            if let Some(len) = st.formula_at(p.pos).map(|f| f.text.len())
593                && !budget.try_shared_text(len)
594            {
595                if let Some(cell) = st.sheet.cells.get_mut(&p.pos) {
596                    cell.formula = None;
597                }
598                st.unsupported(TOO_MUCH);
599                continue;
600            }
601            if !st.formula_at(p.pos).is_some_and(|f| f.text.chars().count() <= MAX_SHARED_TEMPLATE) {
602                failed.insert(p.anchor);
603            }
604        } else {
605            st.unsupported("a shared formula that is missing from the file");
606        }
607    }
608    if shared_attaches >= MAX_SHARED_ATTACHES {
609        st.unsupported("more shared formulas than the expansion cap; some formulas kept their values");
610    }
611}

613/// The other cells of a legacy array formula hold only cached results: blanked, so that the
614/// formula fills its range on recalculation. If the formula itself couldn't be read, they keep
615/// their values.
616fn blank_array_followers(st: &mut SheetState<'_>, pending: &[Pending]) {
617    for p in pending {
618        let filled = st.formula_at(p.anchor).and_then(|f| f.array).is_some_and(|r| r.contains(p.pos));
619        if p.pos == p.anchor || !filled {
620            continue;
621        }
622        match st.sheet.cells.get_mut(&p.pos) {
623            Some(cell) if cell.format.is_some() => *cell = Cell { value: Value::Empty, format: cell.format, formula: None },
624            _ => {
625                st.sheet.cells.remove(&p.pos);
626            }
627        }
628    }
629}

631fn worksheet<'g>(
632    stream: &[u8],
633    bs: &BoundSheet,
634    g: &'g Globals,
635    plain: &'g [bool],
636    budget: &mut Budget,
637    report: &mut Report,
638) -> Result<Option<SheetState<'g>>, Error> {
639    if bs.pos >= stream.len() {
640        return Err(invalid(format!("sheet '{}' points outside the Workbook stream", bs.name)));
641    }
642    let mut recs = Records::new(stream, bs.pos);
643    let first = recs.next_record()?.ok_or_else(|| invalid("missing sheet substream"))?;
644    let (vers, dt) = bof(&first)?;
645    if vers != BIFF8 || dt != 0x0010 {
646        return Ok(None);
647    }
648    let mut st = SheetState {
649        g,
650        plain,
651        sheet: Sheet { name: bs.name.clone(), ..Sheet::default() },
652        pending: vec![],
653        shared: HashMap::new(),
654        arrays: HashMap::new(),
655        last_formula: None,
656        pending_string: None,
657        dropped: 0,
658        unsupported: vec![],
659        dv_left: 0,
660        note_object: None,
661        note_texts: HashMap::new(),
662        notes: vec![],
663        cf: None,
664        palette: g.styles.color_table(),
665        drawing: vec![],
666        filtered: false,
667    };
668    let mut depth = 1usize;
669    while let Some(rec) = recs.next_record()? {
670        budget.record()?;
671        match rec.id {
672            BOF => {
673                // An embedded chart's substream.
674                depth += 1;
675                continue;
676            }
677            EOF => {
678                depth -= 1;
679                if depth == 0 {
680                    break;
681                }
682                continue;
683            }
684            _ if depth > 1 => continue,
685            WSBOOL if rec.parts.first().and_then(|p| p.first()).is_some_and(|b| b & WSBOOL_DIALOG != 0) => return Ok(None),
686            _ => {}
687        }
688        sheet_record(&rec, &mut st, budget, report)?;
689    }
690    Ok(Some(st))
691}

693/// Attaches a sheet's shared and array formulas (their definitions follow the first cell that
694/// uses them). Run for every sheet only after all sheets' mandatory text is charged, so the
695/// optional shared-formula text can't use up the allowance a later sheet's strings need.
696fn finish_sheet(mut st: SheetState<'_>, budget: &mut Budget, report: &mut Report) -> Sheet {
697    let (arrays, shared) = (std::mem::take(&mut st.arrays), std::mem::take(&mut st.shared));
698    let pending = std::mem::take(&mut st.pending);
699    resolve_pending(&mut st, budget, &arrays, &shared, &pending);
700    blank_array_followers(&mut st, &pending);
701    let mut skipped_pictures = false;
702    for shape in drawing::picture_shapes(&std::mem::take(&mut st.drawing)) {
703        let blip = usize::try_from(shape.pib).ok().and_then(|i| i.checked_sub(1)).and_then(|i| st.g.blips.get(i)).cloned().flatten();
704        match blip {
705            Some((data, mime)) if st.sheet.pictures.len() < MAX_PICTURES => st.sheet.pictures.push(Picture { anchor: shape.anchor, data, mime }),
706            _ => skipped_pictures = true,
707        }
708    }
709    if skipped_pictures {
710        report.warn(Warning::PicturesSkipped);
711    }
712    // A Note names its text's object, whose TxO came earlier in the substream.
713    for (cell, object, author, visible) in std::mem::take(&mut st.notes) {
714        let text = st.note_texts.get(&object).cloned().unwrap_or_default();
715        st.sheet.notes.push(Note { cell, author, text, visible });
716    }
717    if st.dropped > 0 {
718        report.warn(Warning::CellsOutsideGrid(st.dropped));
719    }
720    for why in &st.unsupported {
721        report.warn(Warning::FormulaKept(why));
722    }
723    st.sheet
724}

726/// The shortest WINDOW2 a file holds (the chart-sheet form); a shorter one is truncated.
727const WINDOW2_MIN_LEN: usize = 10;
728
729#[allow(clippy::cognitive_complexity, reason = "one arm per worksheet record of [MS-XLS] 2.1.7.20.5")]
730fn sheet_record(rec: &Record<'_>, st: &mut SheetState<'_>, budget: &mut Budget, report: &mut Report) -> Result<(), Error> {
731    let mut c = Cursor::new(&rec.parts);
732    // A formula's string result is the String record right after it (shared/array records may sit between).
733    if !matches!(rec.id, 0x0207 | 0x04BC | 0x0221 | 0x0236) {
734        st.pending_string = None;
735    }
736    match rec.id {
737        0x0203 => {
738            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
739            let v = Value::number(c.f64()?);
740            st.put(budget, r, col, x, v)?;
741        }
742        0x027E => {
743            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
744            let v = rk(c.u32()?);
745            st.put(budget, r, col, x, v)?;
746        }
747        0x00BD => {
748            let r = c.u16()?;
749            let first = c.u16()?;
750            let n = c.remaining().saturating_sub(2) / 6;
751            for i in 0..n {
752                let x = c.u16()?;
753                let v = rk(c.u32()?);
754                let col = u16::try_from(usize::from(first) + i).unwrap_or(u16::MAX);
755                st.put(budget, r, col, x, v)?;
756            }
757        }
758        0x00FD => {
759            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
760            let i = c.u32()? as usize;
761            match st.g.sst.get(i) {
762                Some(s) => {
763                    st.put(budget, r, col, x, Value::Text(s.clone()))?;
764                }
765                None => report.warn(Warning::MissingSharedString),
766            }
767        }
768        // Label, and RString (a label followed by formatting runs, which are skipped).
769        0x0204 | 0x00D6 => {
770            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
771            let s = c.string()?;
772            budget.text(s.len())?;
773            st.put(budget, r, col, x, Value::Text(Arc::from(s)))?;
774        }
775        0x0205 => {
776            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
777            let (v, is_err) = (c.u8()?, c.u8()?);
778            let v = if is_err != 0 { Value::Error(cell_error(v)) } else { Value::Bool(v != 0) };
779            st.put(budget, r, col, x, v)?;
780        }
781        0x0201 => {
782            let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
783            st.put(budget, r, col, x, Value::Empty)?;
784        }
785        0x00BE => {
786            let r = c.u16()?;
787            let first = c.u16()?;
788            let n = c.remaining().saturating_sub(2) / 2;
789            for i in 0..n {
790                let x = c.u16()?;
791                let col = u16::try_from(usize::from(first) + i).unwrap_or(u16::MAX);
792                st.put(budget, r, col, x, Value::Empty)?;
793            }
794        }
795        0x0006 => formula_cell(&mut c, st, budget)?,
796        0x0207 => {
797            if let Some((pos, format)) = st.pending_string.take() {
798                let s = c.string()?;
799                budget.text(s.len())?;
800                let cell = st.sheet.cells.entry(pos).or_insert_with(|| Cell { value: Value::Empty, format, formula: None });
801                cell.value = Value::Text(Arc::from(s));
802            }
803        }
804        0x04BC => {
805            // RefU (the cells that may use it), reserved, cUse.
806            c.skip(8)?;
807            let f = parsed(&mut c)?;
808            if let Some(at) = st.last_formula {
809                st.shared.insert(at, f);
810            }
811        }
812        0x0221 => {
813            let (r0, r1) = (c.u16()?, c.u16()?);
814            let (c0, c1) = (c.u8()?, c.u8()?);
815            c.skip(6)?;
816            let f = parsed(&mut c)?;
817            let range = Range::new(CellRef::new(u32::from(r0), u32::from(c0)), CellRef::new(u32::from(r1.max(r0)), u32::from(c1.max(c0))));
818            if let Some(at) = st.last_formula {
819                st.arrays.insert(at, (range, f));
820            }
821        }
822        0x000D => {
823            st.sheet.calc.mode = Some(match c.u16()? as i16 {
824                0 => CalcMode::Manual,
825                -1 => CalcMode::AutomaticExceptTables,
826                _ => CalcMode::Automatic,
827            });
828        }
829        0x000C => st.sheet.calc.max_iterations = Some(c.u16()?),
830        0x0011 => st.sheet.calc.iterative = Some(c.u16()? == 1),
831        0x0010 => st.sheet.calc.max_change = Some(c.f64()?),
832        0x0236 => report.warn(Warning::DataTable),
833        0x00E5 => merges(&mut c, st)?,
834        0x007D => {
835            let (first, last) = (c.u16()?, c.u16()?);
836            let width = c.u16()?;
837            let x = c.u16()?;
838            let flags = c.u16()?;
839            let column = Column {
840                width,
841                custom_width: flags & 0x0002 != 0,
842                hidden: flags & 0x0001 != 0,
843                outline: ((flags >> 8) & 7) as u8,
844                collapsed: flags & 0x1000 != 0,
845                format: st.format(x),
846            };
847            // An empty range when `first` is past `last`.
848            for col in first..=last.min(255) {
849                st.sheet.columns.insert(col, column);
850            }
851        }
852        0x0208 => {
853            let r = c.u16()?;
854            c.skip(4)?;
855            let height = c.u16()?;
856            c.skip(4)?;
857            let flags = c.u32()?;
858            let format = if flags & 0x80 != 0 { st.format(((flags >> 16) & 0x0FFF) as u16) } else { None };
859            let row = Row {
860                height: height & 0x7FFF,
861                custom_height: flags & 0x40 != 0,
862                hidden: flags & 0x20 != 0,
863                outline: (flags & 7) as u8,
864                collapsed: flags & 0x10 != 0,
865                format,
866            };
867            st.sheet.rows.insert(u32::from(r), row);
868        }
869        0x0225 => {
870            let flags = c.u16()?;
871            let height = c.u16()?;
872            if flags & 0x0002 == 0 && height > 0 {
873                st.sheet.default_row_height = Some(height);
874            }
875        }
876        0x023E => {
877            let flags = if c.remaining() >= WINDOW2_MIN_LEN { c.u16().ok() } else { None };
878            let Some(flags) = flags else { return Ok(()) };
879            let on = |bit: u16| flags & bit != 0;
880            st.sheet.view = Some(View {
881                show_formulas: on(0x0001),
882                show_gridlines: on(0x0002),
883                show_headings: on(0x0004),
884                show_zeros: on(0x0010),
885                right_to_left: on(0x0040),
886            });
887        }
888        0x005D | 0x01B6 | 0x001C | 0x00EC => match object_record(rec, &mut c, st, budget, report) {
889            Err(Error::Invalid(_)) => {
890                if rec.id == 0x005D {
891                    st.note_object = None;
892                }
893                report.warn(Warning::DamagedObjects);
894            }
895            r => r?,
896        },
897        0x009D => st.filtered = true,
898        0x01B0 => match cond::condfmt(&mut c) {
899            Ok(cf) => st.cf = Some(cf),
900            Err(_) => {
901                st.cf = None;
902                report.warn(Warning::ConditionalFormattingSkipped);
903            }
904        },
905        0x01B1 => {
906            if conditional_format(&mut c, st).is_none() {
907                report.warn(Warning::ConditionalFormattingSkipped);
908            }
909        }
910        0x0879 => report.warn(Warning::ConditionalFormatting12),
911        // A short DVal starts no rules.
912        0x01B2 => st.dv_left = crate::craft::xls::dv::dval(&mut c).unwrap_or(0),
913        0x01BE => {
914            if st.dv_left == 0 {
915                report.warn(Warning::DataValidationIgnored);
916                return Ok(());
917            }
918            st.dv_left -= 1;
919            let rule = match crate::craft::xls::dv::dv(&mut c) {
920                Ok(raw) => {
921                    budget.text(raw.text_len())?;
922                    crate::craft::xls::dv::validation(raw, &st.g.links)
923                }
924                Err(_) => None,
925            };
926            match rule {
927                Some(v) if st.sheet.validations.len() < crate::craft::xls::dv::MAX_RULES => st.sheet.validations.push(v),
928                _ => report.warn(Warning::DataValidationSkipped),
929            }
930        }
931        0x01B8 => report.warn(Warning::Hyperlinks),
932        0x00B0 => report.warn(Warning::PivotTables),
933        _ => {}
934    }
935    Ok(())
936}

938/// The drawing-layer records of a sheet: comment objects (Obj, TxO, Note) and the drawing
939/// stream (MsoDrawing) that places pictures.
940fn object_record(rec: &Record<'_>, c: &mut Cursor<'_, '_>, st: &mut SheetState<'_>, budget: &mut Budget, report: &mut Report) -> Result<(), Error> {
941    match rec.id {
942        // Obj: FtCmo first (ft 0x15, cb, object type, id).
943        0x005D => {
944            let ft = c.u16()?;
945            c.u16()?;
946            let ot = c.u16()?;
947            let id = c.u16()?;
948            st.note_object = (ft == 0x15 && ot == notes::OT_NOTE).then_some(id);
949            if ft == 0x15 && ot == 5 {
950                report.warn(Warning::Charts);
951            }
952        }
953        // TxO: the text of the comment object just described.
954        0x01B6 => {
955            if let Some(id) = st.note_object.take() {
956                c.skip(10)?;
957                let cch = usize::from(c.u16()?);
958                let text = notes::txo_text(&rec.parts, cch);
959                budget.text(text.len())?;
960                st.note_texts.insert(id, text);
961            }
962        }
963        // Note: the cell, flags (0x0002: shown), the object id and the author.
964        0x001C => {
965            let (row, col, flags, object) = (c.u16()?, c.u16()?, c.u16()?, c.u16()?);
966            let author = c.string().unwrap_or_default();
967            budget.text(author.len())?;
968            if let Some(cell) = st.pos(row, col)
969                && st.notes.len() < MAX_NOTES
970            {
971                st.notes.push((cell, object, author, flags & 0x0002 != 0));
972            }
973        }
974        0x00EC if st.drawing.len() < MAX_DRAWING => st.drawing.extend(c.rest()),
975        _ => {}
976    }
977    Ok(())
978}

980/// One CF record of the current CondFmt; `None` when it is not kept.
981fn conditional_format(c: &mut Cursor<'_, '_>, st: &mut SheetState<'_>) -> Option<()> {
982    let g = st.g;
983    let rule = cond::cf(c, &st.palette, |ifmt| g.formats.get(&ifmt).cloned()).ok()?;
984    let (base, ranges) = st.cf.as_ref()?;
985    if ranges.is_empty() || st.sheet.conditional_formats.len() >= cond::MAX_RULES {
986        return None;
987    }
988    // The formulas are offsets from the top left cell of the CondFmt's bounding range.
989    let formula = |rgce: &[u8]| {
990        (!rgce.is_empty()).then(|| decode(rgce, &[], *base, Mode::Cell, &g.links).ok().map(|e| crate::craft::xls::expr::print(&e))).flatten()
991    };
992    let rule_kind = match (rule.kind, rule.operator) {
993        (1, Some(operator)) => ConditionRule::CellIs { operator, formula1: formula(&rule.rgce1)?, formula2: formula(&rule.rgce2) },
994        (2, _) => ConditionRule::Expression { formula: formula(&rule.rgce1)? },
995        _ => return None,
996    };
997    st.sheet.conditional_formats.push(ConditionalFormat { ranges: ranges.clone(), rule: rule_kind, format: rule.format });
998    Some(())
999}

1001/// MergeCells ([MS-XLS] 2.4.168).
1002fn merges(c: &mut Cursor<'_, '_>, st: &mut SheetState<'_>) -> Result<(), Error> {
1003    let n = usize::from(c.u16()?);
1004    for _ in 0..n.min(c.remaining() / 8) {
1005        let (r0, r1, c0, c1) = (c.u16()?, c.u16()?, c.u16()?, c.u16()?);
1006        if r1 < r0 || c1 < c0 || u32::from(c1) >= GRID_COLS {
1007            continue;
1008        }
1009        if st.sheet.merges.len() >= MAX_MERGES {
1010            return Err(Error::TooLarge("too many merged ranges".into()));
1011        }
1012        let range = Range::new(CellRef::new(u32::from(r0), u32::from(c0)), CellRef::new(u32::from(r1), u32::from(c1)));
1013        if !range.is_single() {
1014            st.sheet.merges.push(range);
1015        }
1016    }
1017    Ok(())
1018}

1020fn formula_cell(c: &mut Cursor<'_, '_>, st: &mut SheetState<'_>, budget: &mut Budget) -> Result<(), Error> {
1021    let (r, col, x) = (c.u16()?, c.u16()?, c.u16()?);
1022    let mut val = [0u8; 8];
1023    for b in val.iter_mut() {
1024        *b = c.u8()?;
1025    }
1026    let _flags = c.u16()?;
1027    c.skip(4)?;
1028    let f = parsed(c)?;
1029    let [b0, _, b2, _, _, _, e0, e1] = val;
1030    let mut string_result = false;
1031    let value = if [e0, e1] == [0xFF, 0xFF] {
1032        match b0 {
1033            0x00 => {
1034                string_result = true;
1035                Value::Empty
1036            }
1037            0x01 => Value::Bool(b2 != 0),
1038            0x02 => Value::Error(cell_error(b2)),
1039            0x03 => Value::Text(Arc::from("")),
1040            _ => Value::Empty,
1041        }
1042    } else {
1043        Value::number(f64::from_le_bytes(val))
1044    };
1045    let Some(pos) = st.pos(r, col) else { return Ok(()) };
1046    budget.cell()?;
1047    let format = st.format(x);
1048    st.sheet.cells.insert(pos, Cell { value, format, formula: None });
1049    st.last_formula = Some(pos);
1050    if string_result {
1051        st.pending_string = Some((pos, format));
1052    }
1053    // PtgExp: part of a shared or array formula anchored at another cell.
1054    if f.rgce.len() == 5 && f.rgce.first() == Some(&0x01) {
1055        let get = |i: usize| f.rgce.get(i).copied().unwrap_or(0);
1056        let anchor = CellRef::new(u32::from(u16::from_le_bytes([get(1), get(2)])), u32::from(u16::from_le_bytes([get(3), get(4)])));
1057        st.pending.push(Pending { pos, anchor });
1058        return Ok(());
1059    }
1060    st.attach(pos, &f, Mode::Cell, None);
1061    Ok(())
1062}

1064fn defined_names(g: &Globals, model_index: &[Option<usize>], filtered: &[bool], wb: &mut Workbook, report: &mut Report) {
1065    let scope_of = |itab: u16| -> Option<Option<usize>> {
1066        if itab == 0 {
1067            return Some(None);
1068        }
1069        model_index.get(usize::from(itab) - 1).copied().flatten().map(Some)
1070    };
1071    for lbl in &g.names {
1072        let Some(scope) = scope_of(lbl.itab) else { continue };
1073        let text = match decode(&lbl.rgce, &lbl.rgcb, CellRef::new(0, 0), Mode::Name, &g.links) {
1074            Ok(e) => crate::craft::xls::expr::print(&e),
1075            Err(_) if lbl.rgce.is_empty() => continue,
1076            Err(Unsupported(why)) => {
1077                report.warn(Warning::NameSkipped(why));
1078                continue;
1079            }
1080        };
1081        match lbl.builtin {
1082            Some(6 | 7) => {
1083                let Some(sheet) = scope.and_then(|si| wb.sheets.get_mut(si)) else { continue };
1084                if lbl.builtin == Some(6) {
1085                    sheet.print_area = Some(text);
1086                } else {
1087                    sheet.print_titles = Some(text);
1088                }
1089            }
1090            // _FilterDatabase: the AutoFilter's range, on a sheet whose filter is on.
1091            Some(13) => {
1092                let Some(si) = scope else { continue };
1093                if !filtered.get(si).copied().unwrap_or(false) {
1094                    continue;
1095                }
1096                let area = text.rsplit('!').next().unwrap_or("").replace('$', "");
1097                if let (Some(range), Some(sheet)) = (Range::parse(&area), wb.sheets.get_mut(si))
1098                    && sheet.autofilter.is_none()
1099                {
1100                    sheet.autofilter = Some(range);
1101                }
1102            }
1103            Some(_) => {}
1104            None if lbl.macro_name => report.warn(Warning::MacroNames),
1105            None => wb.names.push(DefinedName { name: lbl.name.clone(), scope, formula: text, hidden: lbl.hidden }),
1106        }
1107    }
1108}
