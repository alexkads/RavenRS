1//! Conditional formatting: a CondFmt record names the ranges ([MS-XLS] 2.4.56), and each CF
2//! record after it is one rule (2.4.42): "cell value is" with an operator and one or two
3//! formulas, or a formula that is true. Its look is a DXFN (2.5.81), where every attribute it
4//! leaves unset ("ninch") is left as the cell has it.
5
6use crate::craft::sheet::{CellRef, Range, builtin_format};
7
8use crate::craft::xls::Error;
9use crate::craft::xls::model::{BorderLine, DifferentialFormat, DxfFill, Operator, Underline};
10use crate::craft::xls::records::Cursor;
11use crate::craft::xls::styles::{border_style, color, pattern};

13/// Ranges one CondFmt record lists at most.
14pub(crate) const MAX_CF_RANGES: usize = 4096;
15/// Conditional formats kept per sheet; more are skipped with a warning.
16pub(crate) const MAX_RULES: usize = 65_536;
17
18/// A CondFmt record: the top left cell of its bounding range (what the rules' relative
19/// references are offsets from) and its ranges.
20pub(crate) fn condfmt(c: &mut Cursor<'_, '_>) -> Result<(CellRef, Vec<Range>), Error> {
21    // Rule count, then flags and an id.
22    c.skip(4)?;
23    // refBound: rows (first, last), columns (first, last).
24    let (r0, _, c0, _) = (c.u16()?, c.u16()?, c.u16()?, c.u16()?);
25    let n = usize::from(c.u16()?).min(MAX_CF_RANGES).min(c.remaining() / 8);
26    let mut out = Vec::with_capacity(n);
27    for _ in 0..n {
28        let (r0, r1, c0, c1) = (c.u16()?, c.u16()?, c.u16()?, c.u16()?);
29        out.push(Range::new(CellRef::new(u32::from(r0), u32::from(c0)), CellRef::new(u32::from(r1), u32::from(c1))));
30    }
31    Ok((CellRef::new(u32::from(r0), u32::from(c0)), out))
32}

34/// One CF record, before its formulas are decoded.
35pub(crate) struct Cf {
36    /// 1: the cell value compared; 2: a formula.
37    pub kind: u8,
38    pub operator: Option<Operator>,
39    pub format: DifferentialFormat,
40    pub rgce1: Vec<u8>,
41    pub rgce2: Vec<u8>,
42}
43
44/// Reads a CF record: its kind, operator, look and the two formulas' tokens. `num_fmt` gives the
45/// code of a number format by id.
46pub(crate) fn cf(c: &mut Cursor<'_, '_>, palette: &[u32; 64], num_fmt: impl Fn(u16) -> Option<String>) -> Result<Cf, Error> {
47    let kind = c.u8()?;
48    let operator = match c.u8()? {
49        1 => Some(Operator::Between),
50        2 => Some(Operator::NotBetween),
51        3 => Some(Operator::Equal),
52        4 => Some(Operator::NotEqual),
53        5 => Some(Operator::Greater),
54        6 => Some(Operator::Less),
55        7 => Some(Operator::GreaterOrEqual),
56        8 => Some(Operator::LessOrEqual),
57        _ => None,
58    };
59    let (cce1, cce2) = (usize::from(c.u16()?), usize::from(c.u16()?));
60    let format = dxfn(c, palette, num_fmt)?;
61    let rest = c.rest();
62    let rgce1 = rest.get(..cce1).unwrap_or(&[]).to_vec();
63    let rgce2 = rest.get(cce1..cce1.saturating_add(cce2)).unwrap_or(&[]).to_vec();
64    Ok(Cf { kind, operator, format, rgce1, rgce2 })
65}

67fn bit(flags: u32, n: u32) -> bool {
68    flags & (1 << n) != 0
69}
70
71/// A DXFN: which blocks follow (number format, font, alignment, border, pattern, protection)
72/// and, per attribute, whether it is left unset.
73fn dxfn(c: &mut Cursor<'_, '_>, palette: &[u32; 64], num_fmt: impl Fn(u16) -> Option<String>) -> Result<DifferentialFormat, Error> {
74    let flags = c.u32()?;
75    let flags2 = c.u16()?;
76    let mut d = DifferentialFormat::default();
77    if bit(flags, 25) {
78        if flags2 & 1 != 0 {
79            // DXFNumUsr: its size, then the format code.
80            c.u16()?;
81            d.number_format = Some(c.string()?);
82        } else {
83            c.u8()?;
84            let id = u16::from(c.u8()?);
85            d.number_format = num_fmt(id).or_else(|| builtin_format(u32::from(id)).map(str::to_string));
86        }
87    }
88    if bit(flags, 26) {
89        // DXFFntD, 118 bytes: name (64), height, options (italic 0x02, strikeout 0x80), weight,
90        // script, underline, colour, then which of those are unset. As in a SpreadsheetML
91        // differential font, the face and size stay the cell's.
92        let mut b = [0u8; 118];
93        for slot in b.iter_mut() {
94            *slot = c.u8()?;
95        }
96        let u32_at = |at: usize| b.get(at..at + 4).and_then(|x| <[u8; 4]>::try_from(x).ok()).map_or(u32::MAX, u32::from_le_bytes);
97        let u16_at = |at: usize| b.get(at..at + 2).and_then(|x| <[u8; 2]>::try_from(x).ok()).map_or(0, u16::from_le_bytes);
98        let (options, weight, underline, icv) = (u32_at(68), u16_at(72), b.get(76).copied().unwrap_or(0), u32_at(80));
99        let (style_unset, underline_unset, weight_unset) = (u32_at(88), u32_at(96) != 0, u32_at(100) != 0);
100        if !weight_unset {
101            d.bold = Some(weight >= 600);
102        }
103        if style_unset & 0x02 == 0 {
104            d.italic = Some(options & 0x02 != 0);
105        }
106        if style_unset & 0x80 == 0 {
107            d.strike = Some(options & 0x80 != 0);
108        }
109        if !underline_unset {
110            d.underline = Some(match underline {
111                1 => Underline::Single,
112                2 => Underline::Double,
113                0x21 => Underline::SingleAccounting,
114                0x22 => Underline::DoubleAccounting,
115                _ => Underline::None,
116            });
117        }
118        // 0xFFFFFFFF: the colour is unset.
119        if let Ok(icv) = u16::try_from(icv) {
120            d.font_color = Some(color(icv, palette));
121        }
122    }
123    if bit(flags, 27) {
124        c.skip(8)?;
125    }
126    if bit(flags, 28) {
127        let (a, b) = (c.u32()?, c.u32()?);
128        let line = |unset: bool, dg: u32, icv: u32| {
129            (!unset).then(|| BorderLine { style: border_style(dg & 0x0F), color: color((icv & 0x7F) as u16, palette) })
130        };
131        d.left = line(bit(flags, 10), a, a >> 16);
132        d.right = line(bit(flags, 11), a >> 4, a >> 23);
133        d.top = line(bit(flags, 12), a >> 8, b);
134        d.bottom = line(bit(flags, 13), a >> 12, b >> 7);
135    }
136    if bit(flags, 29) {
137        let (p1, p2) = (c.u16()?, c.u16()?);
138        d.fill = Some(DxfFill {
139            pattern: (!bit(flags, 16)).then(|| pattern((p1 >> 10) & 0x3F)),
140            fg: (!bit(flags, 17)).then(|| color(p2 & 0x7F, palette)),
141            bg: (!bit(flags, 18)).then(|| color((p2 >> 7) & 0x7F, palette)),
142        });
143    }
144    if bit(flags, 30) {
145        c.skip(2)?;
146    }
147    Ok(d)
148}
