1//! Cell comments (notes): the Note record names the cell, its author and the
2//! drawing object that holds the text ([MS-XLS] 2.4.179 Note); the Obj record before it says
3//! the object is a comment ([MS-XLS] 2.4.181 Obj, 2.5.143 FtCmo, `ot` 0x19); the TxO record
4//! that follows holds the text in its Continue records ([MS-XLS] 2.4.329 TxO).
5
6/// Characters in one comment at most.
7pub(crate) const MAX_NOTE_CHARS: usize = 32_767;
8
9/// FtCmo's object type for a comment.
10pub(crate) const OT_NOTE: u16 = 0x19;
11
12/// The text of a TxO record: `cch` characters in the Continue records after its fixed part,
13/// each starting with its own compression flag (0: one byte a character, 1: UTF-16). The
14/// formatting runs after the text are ignored.
15pub(crate) fn txo_text(parts: &[&[u8]], cch: usize) -> String {
16    let cch = cch.min(MAX_NOTE_CHARS);
17    let mut units: Vec<u16> = Vec::with_capacity(cch);
18    for part in parts.iter().skip(1) {
19        if units.len() >= cch {
20            break;
21        }
22        let Some((&flag, rest)) = part.split_first() else { continue };
23        if flag & 1 != 0 {
24            for pair in rest.as_chunks::<2>().0 {
25                if units.len() >= cch {
26                    break;
27                }
28                units.push(u16::from_le_bytes(*pair));
29            }
30        } else {
31            for b in rest {
32                if units.len() >= cch {
33                    break;
34                }
35                units.push(u16::from(*b));
36            }
37        }
38    }
39    String::from_utf16_lossy(&units)
40}

42#[cfg(test)]
43mod tests {
44    use super::*;
45
46    #[test]
47    fn txo_text_reads_compressed_and_wide_parts_and_stops_at_its_length() {
48        let fixed: &[u8] = &[0; 18];
49        assert_eq!(txo_text(&[fixed, &[0, b'H', b'i']], 2), "Hi");
50        let wide: Vec<u8> = [1u8].into_iter().chain("Đỗ".encode_utf16().flat_map(u16::to_le_bytes)).collect();
51        assert_eq!(txo_text(&[fixed, &wide], 2), "Đỗ");
52        // A text over two Continue records, then the formatting runs (not text).
53        assert_eq!(txo_text(&[fixed, &[0, b'a', b'b'], &[0, b'c'], &[0, 0, 0, 0]], 3), "abc");
54        assert_eq!(txo_text(&[fixed], 5), "", "no text record");
55        assert_eq!(txo_text(&[fixed, &[1, 0x41]], 1), "", "half a UTF-16 unit is dropped");
56    }
57}
