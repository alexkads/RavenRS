2//! Clean-room reader for Excel 97-2003 workbooks (`.xls`, BIFF8), written from Microsoft's
3//! public specifications **[MS-XLS]: Excel Binary File Format (.xls) Structure** (the Workbook
4//! stream) and [MS-CFB] (the compound file around it, read by `craft-cfb`).
5//!
6//! [`read`] gives a [`Workbook`]: the worksheets with their cells (values, cached formula
7//! results, and formulas as A1 text, including shared and array formulas), the cell formats
8//! (number formats, fonts, fills, borders, alignment, protection, with palette colours
9//! resolved), user-defined cell styles, merged cells, column and row records, sheet display
10//! flags, calculation settings, defined names, print areas and print titles, data validation
11//! rules, cell comments, conditional formats ("cell value is" and "formula is"), pictures with
12//! their cell anchors, the AutoFilter range, and the 1904 date flag. Widths, heights and anchors
13//! stay in the file's units.
14//!
15//! Everything else (charts, other shapes, newer conditional formats, hyperlinks,
16//! PivotTables, macros) is skipped and named in [`Read::warnings`]. A workbook encrypted with
17//! RC4 or RC4 CryptoAPI is read with its password ([`read_with_password`]; without one, Excel's
18//! default password is tried and [`Error::Encrypted`] says one is needed). Other encryption and
19//! versions before BIFF8 are refused ([`Error::UnsupportedEncryption`], [`Error::Biff5`]). A formula that
20//! cannot be put back into text (links to other workbooks, macro commands, unknown tokens)
21//! leaves its cell with the saved result and a [`Warning::FormulaKept`].
22//!
23//! Input is hostile: [`Limits`] caps records, cells and text, the remaining counts have fixed
24//! caps, formula nesting is bounded, and nothing indexes by a number from the file.
25//!
26//! ```
27//! # fn main() -> Result<(), ravenrs_sheet::craft::xls::Error> {
28//! # let bytes = ravenrs_sheet::craft::xls::testing::workbook(&[("Sheet1", vec![vec!["a".into(), "2.5".into()]])]);
29//! let read = ravenrs_sheet::craft::xls::read(&bytes)?;
30//! let sheet = &read.workbook.sheets[0];
31//! assert_eq!(sheet.name, "Sheet1");
32//! assert_eq!(sheet.cells[&ravenrs_sheet::craft::xls::CellRef::new(0, 1)].value, ravenrs_sheet::craft::xls::Value::Number(2.5));
33//! # Ok(())
34//! # }
35//! ```

60/// Why a workbook could not be read.
61#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
62#[non_exhaustive]
63pub enum Error {
64    /// Not a compound file, no Workbook stream, or a damaged stream.
65    #[error("not a readable Excel 97-2003 workbook: {0}")]
66    Invalid(String),
67    /// More than a [`Limits`] value or a fixed cap allows.
68    #[error("the workbook is too large: {0}")]
69    TooLarge(String),
70    /// The workbook is encrypted with a password (it has a FilePass record): read it with
71    /// [`read_with_password`].
72    #[error("the workbook is password-protected (encrypted)")]
73    Encrypted,
74    /// The password does not open the workbook.
75    #[error("the password is not the workbook's")]
76    WrongPassword,
77    /// The workbook is encrypted with a scheme that is not read (XOR obfuscation, CryptoAPI with
78    /// AES or another cipher or hash, an external provider), named.
79    #[error("the workbook uses {0} encryption, which is not read")]
80    UnsupportedEncryption(String),
81    /// An Excel 5.0/95 workbook (BIFF5), which is not read.
82    #[error("Excel 5.0/95 workbooks (BIFF5) are not supported")]
83    Biff5,
84}

86/// What a read skipped or repaired. Each distinct warning is reported once.
87#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
88#[non_exhaustive]
89pub enum Warning {
90    #[error("some cell formatting records were shorter than expected; the formatting present was kept")]
91    ShortFormatting,
92    #[error("some number formats were shorter than their declared length; the characters present were kept")]
93    ShortNumberFormat,
94    #[error("chart sheets were skipped")]
95    ChartSheet,
96    #[error("Excel 4.0 macro sheets were skipped")]
97    MacroSheet,
98    #[error("VBA modules and other non-worksheet sheets were skipped")]
99    OtherSheet,
100    #[error("dialog sheets and unreadable sheets were skipped")]
101    DialogSheet,
102    #[error("some cells refer to missing shared strings and were left empty")]
103    MissingSharedString,
104    #[error("what-if data tables were not read; their values were kept")]
105    DataTable,
106    /// Some comment or drawing object records were shorter than they say and were skipped.
107    #[error("some comment or object records were damaged and skipped")]
108    DamagedObjects,
109    /// Charts embedded in a worksheet are not read.
110    #[error("charts were not read")]
111    Charts,
112    /// Pictures in a format that is not read (PICT, TIFF), damaged ones, ones past the limits,
113    /// and shapes that name a missing picture.
114    #[error("some pictures were damaged, in a format that is not read (PICT, TIFF), or past the limits, and were skipped")]
115    PicturesSkipped,
116    /// Some CF records could not be read: a rule type other than "cell value is" and "formula
117    /// is", a formula that could not be decoded, a damaged record, or more than the reader keeps.
118    #[error("some conditional formatting rules could not be read and were skipped")]
119    ConditionalFormattingSkipped,
120    /// CF12 records (Excel 2007 data bars, colour scales, icon sets) are not read.
121    #[error("newer conditional formats (data bars, colour scales, icon sets) were not read")]
122    ConditionalFormatting12,
123    /// Some Dv records could not be read: Excel would refuse them (an unknown alert style or
124    /// operator, a formula or range list past the record), no range was left, a formula could not
125    /// be decoded, or the sheet had more rules than the reader keeps.
126    #[error("some data validation rules could not be read and were skipped")]
127    DataValidationSkipped,
128    /// Dv records with no DVal record before them, or past the number of rules it gives: Excel
129    /// does not read them either.
130    #[error("some data validation records were ignored, as Excel ignores them: no DVal record before them, or past the number of rules it gives")]
131    DataValidationIgnored,
132    #[error("hyperlinks were not read")]
133    Hyperlinks,
134    #[error("PivotTables were not read; their values were kept")]
135    PivotTables,
136    /// Cells beyond the BIFF8 grid (65,536 rows, 256 columns) in one sheet.
137    #[error("{0} cells outside the grid were skipped")]
138    CellsOutsideGrid(usize),
139    /// Some formulas use the named feature; their saved results were kept as values.
140    #[error("some formulas use {0}; their saved results were kept as values")]
141    FormulaKept(&'static str),
142    /// Some defined names use the named feature and were skipped.
143    #[error("some defined names use {0} and were skipped")]
144    NameSkipped(&'static str),
145    #[error("macro names were skipped")]
146    MacroNames,
147}

156/// Caps on what one workbook may hold. The defaults suit a spreadsheet application; a caller
157/// that only wants a table of values can lower them.
158#[derive(Debug, Clone, Copy, PartialEq, Eq)]
159pub struct Limits {
160    /// Records read from the whole Workbook stream (a full 65,536 × 256 sheet has fewer than the
161    /// default). One allowance for every sheet: many sheets pointing at one huge substream do
162    /// not each get their own.
163    pub max_records: usize,
164    /// Stored cells in the workbook.
165    pub max_cells: usize,
166    /// Bytes of text (strings, number formats, expanded shared formulas) in the workbook.
167    pub max_text: usize,
168}
169
170impl Default for Limits {
171    fn default() -> Self {
172        Limits { max_records: 40_000_000, max_cells: 4_000_000, max_text: 256 * 1024 * 1024 }
173    }
