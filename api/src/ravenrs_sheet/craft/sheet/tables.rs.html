44pub fn builtin_format(id: u32) -> Option<&'static str> {
45    BUILTIN_FORMATS.iter().find(|(i, _)| *i == id).map(|(_, c)| *c)
46}
47
48/// Id to write for a format code when it is a locale-independent built-in. The currency formats
49/// 5 to 8 and the accounting formats with a currency, 42 and 44, are left out (`None`, to be
50/// written as custom formats) because their meaning depends on the reader's locale.
51pub fn builtin_format_id(code: &str) -> Option<u32> {
52    BUILTIN_FORMATS.iter().find(|(i, c)| *c == code && !matches!(i, 5..=8 | 42 | 44)).map(|(i, _)| *i)
53}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    #[test]
71    fn builtin_format_ids() {
72        assert_eq!((0..=60).filter(|&id| builtin_format(id).is_some()).count(), 36);
73        assert_eq!(builtin_format(0), Some("General"));
74        assert_eq!(builtin_format(14), Some("m/d/yyyy"));
75        assert_eq!(builtin_format(22), Some("m/d/yyyy h:mm"));
76        assert_eq!(builtin_format(49), Some("@"));
77        // 23 to 36 are locale formats that are not defined here.
78        assert_eq!((23..=36).find_map(builtin_format), None);
79        assert_eq!(builtin_format(50), None);
80        for id in 0..=60 {
81            if let Some(code) = builtin_format(id) {
82                let back = builtin_format_id(code);
83                assert_eq!(back, (!matches!(id, 5..=8 | 42 | 44)).then_some(id), "{id}");
84            }
85        }
86        assert_eq!(builtin_format_id("yyyy"), None);
87    }
88
89    #[test]
90    fn the_palette_repeats_its_eight_fixed_colours() {
91        assert_eq!(DEFAULT_PALETTE.get(..8), DEFAULT_PALETTE.get(8..16));
92        assert_eq!(DEFAULT_PALETTE.get(63), Some(&0x333333));
93    }
