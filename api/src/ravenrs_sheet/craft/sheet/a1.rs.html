1//! Cell positions and their A1 text.
2
3use std::fmt::Write;

5/// Rows in the grid of Excel 2007 and later.
6pub const MAX_ROWS: u32 = 1_048_576;
7/// Columns in the grid of Excel 2007 and later (`A` to `XFD`).
8pub const MAX_COLS: u32 = 16_384;
9
10/// A cell position, zero-based. Ordered by row, then column.
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
12pub struct CellRef {
13    pub row: u32,
14    pub col: u32,
15}
16
17impl CellRef {
18    pub const fn new(row: u32, col: u32) -> Self {
19        CellRef { row, col }
20    }
21
22    /// The A1 text (`B3`).
23    pub fn a1(self) -> String {
24        let mut s = String::new();
25        self.push_a1(&mut s);
26        s
27    }
28
29    /// Appends the A1 text.
30    pub fn push_a1(self, s: &mut String) {
31        push_col(s, self.col);
32        let _ = write!(s, "{}", u64::from(self.row) + 1);
33    }
34
35    /// Reads `B3`, `$B$3`, `b3`: a cell inside the grid, with nothing after it.
36    pub fn parse(s: &str) -> Option<CellRef> {
37        match CellRef::parse_prefix(s)? {
38            (cell, "") => Some(cell),
39            _ => None,
40        }
41    }
42
43    /// Reads an A1 reference at the start of `s` and returns it with the rest of `s`.
44    pub fn parse_prefix(s: &str) -> Option<(CellRef, &str)> {
45        let s = s.strip_prefix('$').unwrap_or(s);
46        let letters = s.bytes().take_while(u8::is_ascii_alphabetic).count();
47        let (cols, rest) = s.split_at_checked(letters)?;
48        let col = col_from_letters(cols)?;
49        let rest = rest.strip_prefix('$').unwrap_or(rest);
50        let digits = rest.bytes().take_while(u8::is_ascii_digit).count();
51        if digits == 0 || digits > 7 {
52            return None;
53        }
54        let (rows, rest) = rest.split_at_checked(digits)?;
55        let row: u32 = rows.parse().ok()?;
56        if row == 0 || row > MAX_ROWS {
57            return None;
58        }
59        Some((CellRef::new(row - 1, col), rest))
60    }
61}
62
63impl std::fmt::Display for CellRef {
64    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
65        f.write_str(&self.a1())
66    }
67}

69/// A rectangle of cells, both corners included; `first` is the top left one. The default is
70/// `A1`.
71#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
72pub struct Range {
73    pub first: CellRef,
74    pub last: CellRef,
75}
76
77impl Range {
78    /// The rectangle spanned by two corners, in any order.
79    pub fn new(a: CellRef, b: CellRef) -> Self {
80        Range { first: CellRef::new(a.row.min(b.row), a.col.min(b.col)), last: CellRef::new(a.row.max(b.row), a.col.max(b.col)) }
81    }
82
83    pub fn contains(&self, c: CellRef) -> bool {
84        (self.first.row..=self.last.row).contains(&c.row) && (self.first.col..=self.last.col).contains(&c.col)
85    }
86
87    pub fn is_single(&self) -> bool {
88        self.first == self.last
89    }
90
91    /// How many cells the rectangle holds.
92    pub fn count(&self) -> u64 {
93        (u64::from(self.last.row - self.first.row) + 1) * (u64::from(self.last.col - self.first.col) + 1)
94    }
95
96    /// The A1 text: `B3` for one cell, `A1:C2` otherwise.
97    pub fn a1(&self) -> String {
98        let mut s = self.first.a1();
99        if !self.is_single() {
100            s.push(':');
101            self.last.push_a1(&mut s);
102        }
103        s
104    }
105
106    /// Reads `A1:C2`, `$A$1:$C$2`, one cell (`B3`), whole columns (`A:C`) and whole rows
107    /// (`1:3`), with surrounding space ignored. Whole columns and rows span the grid.
108    pub fn parse(s: &str) -> Option<Range> {
109        let s = s.trim();
110        let Some((a, b)) = s.split_once(':') else {
111            return CellRef::parse(s).map(|c| Range { first: c, last: c });
112        };
113        if let (Some(x), Some(y)) = (CellRef::parse(a), CellRef::parse(b)) {
114            return Some(Range::new(x, y));
115        }
116        let (a, b) = (a.strip_prefix('$').unwrap_or(a), b.strip_prefix('$').unwrap_or(b));
117        if let (Some(x), Some(y)) = (col_from_letters(a), col_from_letters(b)) {
118            return Some(Range::new(CellRef::new(0, x), CellRef::new(MAX_ROWS - 1, y)));
119        }
120        let row = |t: &str| -> Option<u32> {
121            if t.is_empty() || t.len() > 7 || !t.bytes().all(|c| c.is_ascii_digit()) {
122                return None;
123            }
124            t.parse::<u32>().ok().filter(|r| (1..=MAX_ROWS).contains(r)).map(|r| r - 1)
125        };
126        Some(Range::new(CellRef::new(row(a)?, 0), CellRef::new(row(b)?, MAX_COLS - 1)))
127    }
128}
129
130impl std::fmt::Display for Range {
131    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
132        f.write_str(&self.a1())
133    }
134}

136/// Appends a column's letters: 0 is `A`, 25 is `Z`, 26 is `AA`.
137pub fn push_col(s: &mut String, col: u32) {
138    let mut letters = [0u8; 8];
139    let mut i = letters.len();
140    let mut n = u64::from(col) + 1;
141    while n > 0 && i > 0 {
142        i -= 1;
143        if let Some(l) = letters.get_mut(i) {
144            *l = b'A' + ((n - 1) % 26) as u8;
145        }
146        n = (n - 1) / 26;
147    }
148    for &l in letters.get(i..).unwrap_or_default() {
149        s.push(char::from(l));
150    }
151}

153/// A column's letters: 0 is `A`, 16383 is `XFD`.
154pub fn col_letters(col: u32) -> String {
155    let mut s = String::new();
156    push_col(&mut s, col);
157    s
158}
159
160/// `A` is 0, `xfd` is 16383. `None` for empty text, anything but one to three letters, or a
161/// column beyond `XFD`.
162pub fn col_from_letters(s: &str) -> Option<u32> {
163    if s.is_empty() || s.len() > 3 || !s.bytes().all(|b| b.is_ascii_alphabetic()) {
164        return None;
165    }
166    let n = s.bytes().fold(0u32, |n, b| n * 26 + u32::from(b.to_ascii_uppercase() - b'A') + 1);
167    (n <= MAX_COLS).then(|| n - 1)
168}

170/// Whether `name` reads as an R1C1 reference: `R`, `C`, `RC`, `R1`, `C1`, `R1C`, `RC1`, `R1C1`
171/// in any case.
172fn is_r1c1_token(name: &str) -> bool {
173    let mut chars = name.chars().peekable();
174    let mut letters = 0;
175    for letter in ['r', 'c'] {
176        if chars.peek().is_some_and(|c| c.to_ascii_lowercase() == letter) {
177            chars.next();
178            letters += 1;
179            while chars.peek().is_some_and(char::is_ascii_digit) {
180                chars.next();
181            }
182        }
183    }
184    letters > 0 && chars.next().is_none()
185}

187/// Whether Excel quotes a sheet name in a reference (`'My Sheet'!A1`): it starts with a digit or
188/// `.`, holds anything but letters (any script), digits, `_` and `.`, reads as an A1 cell or as
189/// an R1C1 reference, or is a boolean.
190pub fn sheet_needs_quotes(name: &str) -> bool {
191    let Some(first) = name.chars().next() else {
192        return true;
193    };
194    if first.is_ascii_digit() || first == '.' || name.chars().any(|c| !(c.is_alphanumeric() || c == '_' || c == '.')) {
195        return true;
196    }
197    // No `$` can be here (it is not alphanumeric), so this is exactly "an A1 cell in the grid".
198    CellRef::parse(name).is_some() || is_r1c1_token(name) || name.eq_ignore_ascii_case("TRUE") || name.eq_ignore_ascii_case("FALSE")
199}
200
201/// A sheet name as it is written in a reference: quoted, with `'` doubled, when Excel quotes it.
202pub fn quote_sheet(name: &str) -> String {
203    if sheet_needs_quotes(name) { format!("'{}'", name.replace('\'', "''")) } else { name.to_string() }
204}

210    #[test]
211    fn a1_text_both_ways() {
212        assert_eq!(CellRef::new(2, 27).a1(), "AB3");
213        assert_eq!(CellRef::parse("AB3"), Some(CellRef::new(2, 27)));
214        assert_eq!(CellRef::parse("$ab$3"), Some(CellRef::new(2, 27)));
215        assert_eq!(CellRef::parse("XFD1048576"), Some(CellRef::new(MAX_ROWS - 1, MAX_COLS - 1)));
216        assert_eq!(CellRef::new(MAX_ROWS - 1, MAX_COLS - 1).a1(), "XFD1048576");
217        for bad in ["", "A", "1", "A0", "XFE1", "A1048577", "A1x", "A+1", "A-1", "AAAA1", "A12345678", "É1"] {
218            assert_eq!(CellRef::parse(bad), None, "{bad}");
219        }
220        assert_eq!(CellRef::parse_prefix("B2:C3"), Some((CellRef::new(1, 1), ":C3")));
221        assert_eq!(col_letters(16383), "XFD");
222        assert_eq!((col_letters(26), col_letters(u32::MAX).len()), ("AA".to_string(), 7));
223        assert_eq!(col_from_letters("xfd"), Some(16383));
224        assert_eq!(col_from_letters("XFE"), None);
225    }

227    #[test]
228    fn ranges() {
229        let r = Range::parse(" $C$2:A1 ");
230        assert_eq!(r, Some(Range { first: CellRef::new(0, 0), last: CellRef::new(1, 2) }));
231        assert_eq!(r.map(|r| (r.a1(), r.count())), Some(("A1:C2".to_string(), 6)));
232        assert_eq!(Range::parse("B3").map(|r| (r.is_single(), r.a1())), Some((true, "B3".to_string())));
233        assert_eq!(Range::parse("B:$A"), Some(Range::new(CellRef::new(0, 0), CellRef::new(MAX_ROWS - 1, 1))));
234        assert_eq!(Range::parse("3:2"), Some(Range::new(CellRef::new(1, 0), CellRef::new(2, MAX_COLS - 1))));
235        assert_eq!(Range::parse("1:1048577"), None);
236        assert_eq!(Range::parse("0:1"), None);
237        assert_eq!(Range::parse("A1:"), None);
238        assert_eq!(Range::parse("A:1"), None);
239        assert!(Range::new(CellRef::new(0, 0), CellRef::new(MAX_ROWS - 1, MAX_COLS - 1)).count() == 17_179_869_184);
240    }
