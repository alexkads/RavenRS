15/// The archive, its declared entry count checked before it is indexed. `max_total` is the
16/// archive's own total; a reader that keeps its own counts passes `u64::MAX`.
17fn open_zip<'a>(bytes: &'a [u8], limits: &Limits, max_total: u64) -> Result<Zip<'a>, Error> {
18    let zip =
19        crate::craft::zip::Archive::open(bytes, crate::craft::zip::Limits { max_entries: limits.max_entries as u64, max_total_bytes: max_total })
20            .map_err(crate::craft::opc::zip_error)?;
21    // `zip` keeps one entry of a repeated name. Part names are compared by spelling rules this
22    // crate applies itself (see `Package::new`); an exact repeat is noted, as other spellings are.
23    if zip.has_duplicate_names() {
24        log::warn!("opc: the archive repeats an entry name; one of the entries is read");
25    }
26    Ok(zip)
27}

29/// A part name as stored: `\` as `/` (code that splits paths on `/` must not meet a `\`), no
30/// leading `/`.
31fn stored_name(name: &str) -> String {
32    name.replace('\\', "/").trim_start_matches('/').to_string()
33}

35/// The inflated parts of a package, within [`Limits`]. Shared, so a part the caller keeps as it
36/// is (a picture, an embedded object) is handed over rather than copied.
37#[derive(Clone, Debug, Default)]
38pub struct Package {
39    files: BTreeMap<String, Arc<Vec<u8>>>,
40    /// Lowercased part name → its stored spelling: part names are case-insensitive (ECMA-376
41    /// Part 2 §6.2.2.3), so every spelling is one part.
42    lower: HashMap<String, String>,
43}
44
45impl Package {
46    /// Inflates every part.
47    ///
48    /// An entry the archive cannot produce (a failed checksum, an unknown method) is left out
49    /// with a logged warning, and what it inflated is charged to [`Limits::max_work`]: an entry
50    /// inflates in full before its checksum is compared, so failed entries would otherwise
51    /// inflate without limit. A part over a limit refuses the package or is left out
52    /// ([`Limits::oversized`]).
53    pub fn open(bytes: &[u8], limits: &Limits) -> Result<Package, Error> {
54        // This reader keeps its own two counts (bytes kept, bytes inflated), so the archive's is
55        // not used.
56        let mut zip = open_zip(bytes, limits, u64::MAX)?;
57        let skip = limits.oversized == Oversized::Skip;
58        let mut files = BTreeMap::new();
59        let (mut total, mut work) = (0u64, 0u64);
60        let entries: Vec<String> = zip.names().map(|n| n.into_owned()).collect();
61        for (i, entry) in entries.iter().enumerate() {
62            if skip && work >= limits.max_work {
63                log::warn!("opc: decompression budget spent; skipping the remaining {} entries", entries.len().saturating_sub(i));
64                break;
65            }
66            if entry.ends_with('/') {
67                continue;
68            }
69            let name = stored_name(entry);
70            let size = zip.declared_size(entry).unwrap_or(0);
71            if size > limits.max_part {
72                if skip {
73                    log::warn!("opc: skipping oversized part {name} ({size} bytes)");
74                    continue;
75                }
76                return Err(Error::PartTooLarge { part: name, limit: limits.max_part });
77            }
78            let left = limits.max_total.saturating_sub(total).min(limits.max_work.saturating_sub(work));
79            let room = limits.max_part.min(left);
80            if size > room {
81                // It fits the part limit but not what is left: refused before inflating anything.
82                if skip {
83                    log::warn!("opc: part {name} exceeds size limits; skipped");
84                    continue;
85                }
86                return Err(Error::PackageTooLarge { limit: limits.max_total.min(limits.max_work) });
87            }
88            match zip.read(entry, room) {
89                Ok(Some(buf)) => {
90                    work = work.saturating_add(buf.len() as u64);
91                    total = total.saturating_add(buf.len() as u64);
92                    files.insert(name, Arc::new(buf));
93                }
94                Ok(None) => {}
95                // A damaged entry, or one that inflates past the size it declares: left out and
96                // charged what it may have inflated, its declared size (the read stops one byte
97                // past it).
98                Err(e) => {
99                    work = work.saturating_add(size);
100                    log::warn!("opc: zip entry {name} unreadable: {e}");
101                }
102            }
103        }
104        Ok(Package::new(files))
105    }

107    /// A package from parts already in memory (a Flat OPC file, a converted document), by part
108    /// name.
109    pub fn from_parts(files: BTreeMap<String, Vec<u8>>) -> Package {
110        Package::new(files.into_iter().map(|(name, bytes)| (stored_name(&name), Arc::new(bytes))).collect())
111    }
112
113    fn new(mut files: BTreeMap<String, Arc<Vec<u8>>>) -> Package {
114        let mut lower: HashMap<String, String> = HashMap::new();
115        let mut collides = false;
116        for name in files.keys() {
117            match lower.entry(name.to_ascii_lowercase()) {
118                Entry::Occupied(_) => collides = true,
119                Entry::Vacant(v) => {
120                    v.insert(name.clone());
121                }
122            }
123        }
124        if !collides {
125            return Package { files, lower };
126        }
127        // Entries whose names differ only in case are one part, so every spelling of the name
128        // reaches the same stored entry. The spelling kept is the one a relationship names, so
129        // the part the document points at is the one read; the first in byte order only when no
130        // relationship names any of them.
131        let mut named = HashSet::new();
132        for (name, bytes) in &files {
133            targets_of(name, bytes, |t| {
134                named.insert(t);
135            });
136        }
137        for name in files.keys().rev().filter(|n| named.contains(n.as_str())) {
138            lower.insert(name.to_ascii_lowercase(), name.clone());
139        }
140        let before = files.len();
141        files.retain(|name, _| lower.get(&name.to_ascii_lowercase()) == Some(name));
142        let dropped = before.saturating_sub(files.len());
143        log::warn!("opc: {dropped} zip entries are other spellings of parts already read; they are left out");
144        Package { files, lower }
145    }

147    /// The bytes of all its parts.
148    pub fn size(&self) -> u64 {
149        self.files.values().map(|f| f.len() as u64).sum()
150    }
151
152    /// Every part name as stored, in byte order.
153    pub fn names(&self) -> impl Iterator<Item = &str> {
154        self.files.keys().map(String::as_str)
155    }
156
157    /// The stored spelling of part `name` (leading `/` ignored; case-insensitive).
158    pub fn canonical(&self, name: &str) -> Option<&str> {
159        self.lower.get(&normalize(name)).map(String::as_str)
160    }
161
162    /// Part bytes by name (leading `/` ignored; case-insensitive).
163    pub fn get(&self, name: &str) -> Option<&[u8]> {
164        self.get_shared(name).map(|v| v.as_slice())
165    }
166
167    /// [`Package::get`], as the shared bytes themselves: a part the caller keeps whole then
168    /// costs no copy.
169    pub fn get_shared(&self, name: &str) -> Option<&Arc<Vec<u8>>> {
170        self.canonical(name).and_then(|n| self.files.get(n))
171    }
172
173    /// Part `name` as XML, read with `options` (which carry the caller's limits and its choice
174    /// of names and strictness). `Ok(None)` when there is no such part.
175    pub fn xml(&self, name: &str, options: &Options) -> Result<Option<El>, Error> {
176        match self.get(name) {
177            Some(b) => options.parse(b).map(Some).map_err(|error| Error::Xml { part: name.to_string(), error }),
178            None => Ok(None),
179        }
180    }
181
182    /// The relationships of `part` (from `dir/_rels/name.rels`); those of the package itself
183    /// for `""`. Internal targets name the part as stored, so a part reached under several
184    /// spellings is one part. A missing or unreadable relationships part gives none.
185    pub fn rels(&self, part: &str) -> Rels {
186        match self.get(&rels_path(part)) {
187            Some(bytes) => Rels::parse_in(bytes, part, |n| self.canonical(n).map(str::to_string)),
188            None => Rels::default(),
189        }
190    }
191
192    /// `[Content_Types].xml`.
193    pub fn content_types(&self) -> ContentTypes {
194        self.get("[Content_Types].xml").map(ContentTypes::parse).unwrap_or_default()
195    }
196}

198/// A package whose parts are inflated when they are asked for, against a running total: for a
199/// reader that needs a few parts of a large file, or one part at a time.
200pub struct LazyPackage<'a> {
201    zip: Zip<'a>,
202    /// Normalized part name → entry name (the first entry of a name).
203    names: HashMap<String, String>,
204    total: u64,
205    limits: Limits,
206}
207
208impl<'a> LazyPackage<'a> {
209    /// Indexes the archive. Nothing is inflated; [`Limits::max_work`] and
210    /// [`Limits::oversized`] are not used.
211    pub fn open(bytes: &'a [u8], limits: &Limits) -> Result<LazyPackage<'a>, Error> {
212        // This reader keeps its own count, which a failed read spends (see `read_limited`).
213        let zip = open_zip(bytes, limits, u64::MAX)?;
214        let mut names = HashMap::new();
215        for n in zip.names() {
216            names.entry(normalize(&n)).or_insert_with(|| n.into_owned());
217        }
218        Ok(LazyPackage { zip, names, total: 0, limits: *limits })
219    }
220
221    /// Every part's normalized name ([`normalize`]), sorted.
222    pub fn names(&self) -> Vec<&str> {
223        let mut v: Vec<&str> = self.names.keys().map(String::as_str).collect();
224        v.sort_unstable();
225        v
226    }
227
228    pub fn has(&self, part: &str) -> bool {
229        self.names.contains_key(&normalize(part))
230    }
231
232    /// The stored spelling of part `name` (leading `/` ignored; case-insensitive), as
233    /// [`Package::canonical`] gives it: for a writer that keeps the package's part names.
234    pub fn canonical(&self, name: &str) -> Option<String> {
235        self.names.get(&normalize(name)).map(|n| n.trim_start_matches('/').replace('\\', "/"))
236    }
237
238    /// Bytes charged to [`Limits::max_total`] so far. It is the whole of it once a read has
239    /// failed: see [`LazyPackage::read_limited`].
240    pub fn spent(&self) -> u64 {
241        self.total
242    }
243
244    /// A part's bytes, up to [`Limits::max_part`]; `Ok(None)` when it doesn't exist.
245    pub fn read(&mut self, part: &str) -> Result<Option<Vec<u8>>, Error> {
246        self.read_limited(part, self.limits.max_part)
247    }
248
249    /// A part the reader can do without (the caller warns and goes on when it fails). A read
250    /// that fails part-way is charged this part's whole allowance (`maximum`, and the byte that
251    /// detects an overrun) instead of ending the package's: the decompression work stays
252    /// bounded, and the parts read after it still open.
253    pub fn read_optional_limited(&mut self, part: &str, maximum: u64) -> Result<Option<Vec<u8>>, Error> {
254        let before = self.total;
255        let read = self.read_limited(part, maximum);
256        if let Err(Error::Zip(_)) = &read {
257            self.total = before.saturating_add(maximum.min(self.limits.max_part).saturating_add(1)).min(self.limits.max_total);
258        }
259        read
260    }
261
262    /// A part's bytes, up to `maximum` (and [`Limits::max_part`]): for a small metadata part
263    /// that must not expand to the general limit.
264    ///
265    /// A read that fails or overruns spends the whole of [`Limits::max_total`]: it has already
266    /// cost decompression work the total never saw, and a loop over many broken parts must not
267    /// pay it again and again. Every later read is then a cheap error.
268    pub fn read_limited(&mut self, part: &str, maximum: u64) -> Result<Option<Vec<u8>>, Error> {
269        let maximum = maximum.min(self.limits.max_part);
270        let Some(entry) = self.names.get(&normalize(part)) else {
271            return Ok(None);
272        };
273        let size = self.zip.declared_size(entry).unwrap_or(0);
274        if size > maximum {
275            return Err(Error::PartTooLarge { part: part.to_string(), limit: maximum });
276        }
277        let remaining = self.limits.max_total.saturating_sub(self.total);
278        if size > remaining {
279            self.total = self.limits.max_total;
280            return Err(Error::PackageTooLarge { limit: self.limits.max_total });
281        }
282        match self.zip.read(entry, maximum.min(remaining)) {
283            Ok(Some(out)) => {
284                self.total = self.total.saturating_add(out.len() as u64);
285                Ok(Some(out))
286            }
287            Ok(None) => Ok(None),
288            // Damaged, or inflating past the size it declares.
289            Err(e) => {
290                self.total = self.limits.max_total;
291                Err(Error::Zip(format!("{part}: {e}")))
292            }
293        }
294    }
295
296    /// Part `name` as XML, read with `options`. `Ok(None)` when there is no such part.
297    pub fn read_xml(&mut self, part: &str, options: &Options) -> Result<Option<El>, Error> {
298        match self.read(part)? {
299            Some(b) => options.parse(&b).map(Some).map_err(|error| Error::Xml { part: part.to_string(), error }),
300            None => Ok(None),
301        }
302    }
303
304    /// The relationships of `part`. A missing or unparseable relationships part gives none; one
305    /// that cannot be inflated is the error.
306    pub fn rels(&mut self, part: &str) -> Result<Rels, Error> {
307        let Some(bytes) = self.read(&rels_path(part))? else { return Ok(Rels::default()) };
308        Ok(Rels::parse_in(&bytes, part, |n| self.has(n).then(|| n.to_string())))
309    }
310}
311
312/// The first part whose declared size is over [`Limits::max_part`], with that size: for telling
313/// the user which part made a package too large.
314pub fn declared_oversized(bytes: &[u8], limits: &Limits) -> Option<(String, u64)> {
315    let mut zip = open_zip(bytes, limits, u64::MAX).ok()?;
316    let names: Vec<String> = zip.names().map(|n| n.into_owned()).collect();
317    names.into_iter().find_map(|n| zip.declared_size(&n).filter(|size| *size > limits.max_part).map(|size| (n, size)))
318}

320#[cfg(test)]
321mod tests {
322    use super::*;
323    use crate::craft::opc::zip64_entry_count_sample;
324
325    #[test]
326    fn a_package_declaring_too_many_entries_is_refused_before_zip_reads_them() {
327        let limits = Limits::new(20_000, 1 << 20, 1 << 20);
328        let bomb = zip64_entry_count_sample(20_001);
329        let refused = Some(Error::TooManyEntries { declared: 20_001, limit: 20_000 });
330        assert_eq!(Package::open(&bomb, &limits).err(), refused);
331        assert_eq!(LazyPackage::open(&bomb, &limits).err(), refused);
332        assert!(declared_oversized(&bomb, &limits).is_none());
333        // Within the limit, the same archive is the `zip` crate's to judge.
334        assert!(!matches!(Package::open(&zip64_entry_count_sample(5), &limits), Err(Error::TooManyEntries { .. })));
335    }
