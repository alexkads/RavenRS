2//! Clean-room Office document cryptography, written from Microsoft's public Open Specification
3//! **[MS-OFFCRYPTO]: Office Document Cryptography Structure** and ECMA-376.
4//!
5//! A password-encrypted `.docx`, `.xlsx` or `.pptx` is not a zip but a compound file
6//! ([MS-CFB], read and written with `craft-cfb`) holding `EncryptionInfo` (how the key comes
7//! from the password) and `EncryptedPackage` (the zip, AES encrypted). This crate goes between
8//! the two: file bytes in, package bytes out, and back. It knows nothing about what the package
9//! holds.
10//!
11//! - [`is_encrypted`] tells such a file from any other compound file; [`check`] says, without
12//!   a password, whether a password could open it.
13//! - [`decrypt`] reads *agile* encryption (Office 2010 and later: AES-CBC in 4096-byte
14//!   segments, SHA-1 to SHA-512, a password verifier and an HMAC over the encrypted stream,
15//!   which is required and checked) and *standard* encryption (Office 2007: AES-ECB, SHA-1).
16//!   RC4, extensible, certificate-only and rights-managed files are refused by name
17//!   ([`Error::Unsupported`]).
18//! - [`encrypt`] writes agile encryption as Office does: AES-256, SHA-512, 100,000 spins.
19//! - [`protection`] has the password hashes of document, sheet and workbook protection.
20//!
21//! Files are hostile: every stream read is capped, every length is checked before slicing, and
22//! the spin count is bounded by the specification's maximum (10,000,000). Passwords and keys
23//! are wiped from memory when dropped and never logged.
24//!
25//! ```
26//! # fn main() -> Result<(), ravenrs_sheet::craft::offcrypto::Error> {
27//! let package = b"PK\x03\x04 a zip package";
28//! let file = ravenrs_sheet::craft::offcrypto::encrypt_with_spin_count(package, "secret", 1000)?;
29//! assert!(ravenrs_sheet::craft::offcrypto::is_encrypted(&file));
30//! assert_eq!(ravenrs_sheet::craft::offcrypto::decrypt(&file, None), Err(ravenrs_sheet::craft::offcrypto::Error::PasswordRequired));
31//! assert_eq!(ravenrs_sheet::craft::offcrypto::decrypt(&file, Some("guess")), Err(ravenrs_sheet::craft::offcrypto::Error::WrongPassword));
32//! assert_eq!(ravenrs_sheet::craft::offcrypto::decrypt(&file, Some("secret"))?, package);
33//! # Ok(())
34//! # }
35//! ```

45/// Why a file could not be decrypted or encrypted.
46#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
47#[non_exhaustive]
48pub enum Error {
49    /// The file can be decrypted, and no password was given.
50    #[error("a password is required")]
51    PasswordRequired,
52    #[error("the password is incorrect")]
53    WrongPassword,
54    /// A scheme that is not read, named to complete "uses {0} encryption": `RC4`, `Extensible`,
55    /// `external`, `certificate`, `rights-management`, a cipher or hash name, `version 1.1`.
56    #[error("{0} encryption isn't supported")]
57    Unsupported(String),
58    /// The file is damaged, or fails its integrity check.
59    #[error("the encrypted file is damaged ({0})")]
60    Malformed(String),
61    /// A stream is larger than its cap.
62    #[error("the encrypted file is too large ({0})")]
63    TooLarge(String),
64    /// A password that can't be used to encrypt: empty, or longer than [`MAX_PASSWORD_CHARS`].
65    #[error("{0}")]
66    BadPassword(String),
67}
