60/// Is this an encrypted Office package (a compound file with an `EncryptedPackage` stream)
61/// rather than a zip? A rights-managed file counts: it has the stream but no `EncryptionInfo`,

63pub fn is_encrypted(bytes: &[u8]) -> bool {

67/// Whether the right password could decrypt this file, checked without one: an error names a
68/// scheme no password opens here ([`Error::Unsupported`]) or damage, so a caller can say so
69/// instead of asking for a password first.
70pub fn check(bytes: &[u8]) -> Result<(), Error> {
71    match decrypt(bytes, None) {
72        Ok(_) | Err(Error::PasswordRequired) => Ok(()),
73        Err(e) => Err(e),
74    }

77/// The zip package inside an encrypted Office file. Without a password (`None`) this says
78/// [`Error::PasswordRequired`] once it knows the file is one it can decrypt; a password that
79/// doesn't fit gives [`Error::WrongPassword`].
80pub fn decrypt(bytes: &[u8], password: Option<&str>) -> Result<Vec<u8>, Error> {
81    let cf = crate::craft::cfb::CompoundFile::open(bytes).map_err(|e| bad(format!("not a readable compound file ({e})")))?;
82    let package = root_stream(&cf, "EncryptedPackage").ok_or_else(|| bad("no EncryptedPackage stream"))?;
83    // Rights management (IRM) keeps the package but describes its encryption elsewhere.
84    let info = root_stream(&cf, "EncryptionInfo").ok_or_else(|| Error::Unsupported("rights-management".into()))?;
85    let info = stream(&cf, info, "EncryptionInfo", MAX_INFO)?;
86    let package = stream(&cf, package, "EncryptedPackage", bytes.len() as u64)?;
87    decrypt_streams(&info, &package, password)
88}

90/// [`decrypt`] from the contents of the two streams.
91pub fn decrypt_streams(info: &[u8], package: &[u8], password: Option<&str>) -> Result<Vec<u8>, Error> {
92    if matches!(info.get(..4), Some([2..=4, 0, 2, 0])) {
93        return decrypt_standard(info, package, password);
94    }
95    let info = parse_info(info)?;
96    let Some(password) = password else { return Err(Error::PasswordRequired) };
97    let pe = &info.password;
98    let hn = iterate(pe.hash, &pe.salt, password, pe.spin_count);
99    // The verifier (§2.3.4.13): a random value and its hash, both encrypted with keys derived
100    // from the password. They match only for the right password.
101    let key_input = derive_key(pe.hash, &hn, &BK_VERIFIER_INPUT, pe.key_bytes);
102    let key_value = derive_key(pe.hash, &hn, &BK_VERIFIER_VALUE, pe.key_bytes);
103    let iv = make_iv(pe.hash, &pe.salt, None, pe.block_size);
104    let input = Zeroizing::new(cbc_decrypt(&key_input, &iv, &pe.verifier_input)?);
105    let input = input.get(..pe.salt_size).ok_or_else(|| bad("the password verifier is too short"))?;
106    let value = Zeroizing::new(cbc_decrypt(&key_value, &iv, &pe.verifier_hash)?);
107    let value = value.get(..pe.hash.size()).ok_or_else(|| bad("the password verifier hash is too short"))?;
108    if !same(&pe.hash.hash(&[input]), value) {
109        return Err(Error::WrongPassword);
110    }
111    // The intermediate key that encrypts the package.
112    let key_key = derive_key(pe.hash, &hn, &BK_KEY_VALUE, pe.key_bytes);
113    let key = Zeroizing::new(cbc_decrypt(&key_key, &iv, &pe.key_value)?);
114    let key = Zeroizing::new(key.get(..info.key_bytes).ok_or_else(|| bad("the encrypted key is too short"))?.to_vec());
115    // Data integrity (§2.3.4.14): an HMAC of the whole EncryptedPackage stream. It is required:
116    // a file that could drop it would let altered ciphertext, or an altered StreamSize, through
117    // unauthenticated (`parse_info` refuses a file without it).
118    let (enc_key, enc_value) = &info.integrity;
119    let hmac_key = Zeroizing::new(cbc_decrypt(&key, &make_iv(info.hash, &info.salt, Some(&BK_HMAC_KEY), info.block_size), enc_key)?);
120    // The key is the salt, zero-padded to the block size; HMAC pads short keys with zeros
121    // anyway, so the padded key gives the same HMAC.
122    let hmac_key = hmac_key.get(..hmac_key.len().min(info.hash.size())).unwrap_or(&[]);
123    let expected = cbc_decrypt(&key, &make_iv(info.hash, &info.salt, Some(&BK_HMAC_VALUE), info.block_size), enc_value)?;
124    let expected = expected.get(..info.hash.size()).ok_or_else(|| bad("the integrity check value is too short"))?;
125    if !same(&info.hash.hmac(hmac_key, package)?, expected) {
126        return Err(bad("the document is damaged (its integrity check failed)"));
127    }
128    // The package itself (§2.3.4.15): StreamSize, then 4096-byte segments, each with its own IV.
129    let size = package
130        .get(..8)
131        .and_then(|b| <[u8; 8]>::try_from(b).ok())
132        .map(u64::from_le_bytes)
133        .ok_or_else(|| bad("the encrypted package is truncated"))?;
134    let data = package.get(8..).unwrap_or(&[]);
135    let size = usize::try_from(size).ok().filter(|s| *s <= data.len()).ok_or_else(|| bad("the encrypted package is truncated"))?;
136    let needed = size.div_ceil(BLOCK).saturating_mul(BLOCK);
137    let data = data.get(..needed).ok_or_else(|| bad("the encrypted package is truncated"))?;
138    let mut out = Vec::with_capacity(needed);
139    for (i, seg) in data.chunks(SEGMENT).enumerate() {
140        let i = u32::try_from(i).map_err(|_| bad("the encrypted package is too large"))?;
141        let iv = make_iv(info.hash, &info.salt, Some(&i.to_le_bytes()), info.block_size);
142        out.extend_from_slice(&cbc_decrypt(&key, &iv, seg)?);
143    }
144    out.truncate(size);
145    Ok(out)
146}

148/// [`decrypt`] for Standard encryption (§2.3.4.5–§2.3.4.9): one AES key from the password,
149/// checked with the verifier, and the whole package AES-ECB encrypted after its 8-byte size.
150fn decrypt_standard(info: &[u8], package: &[u8], password: Option<&str>) -> Result<Vec<u8>, Error> {
151    let s = parse_standard(info)?;
152    let Some(password) = password else { return Err(Error::PasswordRequired) };
153    let key = standard_key(s.salt, password, s.key_bytes);
154    // The verifier (§2.3.4.9): a random value and its SHA-1, both encrypted with the key.
155    let verifier = Zeroizing::new(ecb(&key, s.verifier, false)?);
156    let hash = Zeroizing::new(ecb(&key, s.verifier_hash, false)?);
157    let hash = hash.get(..SHA1_SIZE).ok_or_else(|| bad("the password verifier hash is too short"))?;
158    if !same(&Hash::Sha1.hash(&[&verifier]), hash) {
159        return Err(Error::WrongPassword);
160    }
161    // The package (§2.3.4.4): StreamSize, then the data; bytes past the last block are ignored.
162    let size = package
163        .get(..8)
164        .and_then(|b| <[u8; 8]>::try_from(b).ok())
165        .map(u64::from_le_bytes)
166        .ok_or_else(|| bad("the encrypted package is truncated"))?;
167    let data = package.get(8..).unwrap_or(&[]);
168    let size = usize::try_from(size).ok().filter(|s| *s <= data.len()).ok_or_else(|| bad("the encrypted package is truncated"))?;
169    let needed = size.div_ceil(BLOCK).saturating_mul(BLOCK);
170    let data = data.get(..needed).ok_or_else(|| bad("the encrypted package is truncated"))?;
171    let mut out = ecb(&key, data, false)?;
172    out.truncate(size);
173    Ok(out)
174}

176/// What a Standard `EncryptionInfo` stream says (§2.3.4.5): the key size, and the verifier's salt
177/// and encrypted values.
178struct Standard<'a> {
179    key_bytes: usize,
180    salt: &'a [u8],
181    verifier: &'a [u8],
182    verifier_hash: &'a [u8],
183}
184
185fn parse_standard(info: &[u8]) -> Result<Standard<'_>, Error> {
186    let short = || bad("EncryptionInfo is truncated");
187    let u32_at = |at: usize| -> Result<u32, Error> {
188        let b = at.checked_add(4).and_then(|end| info.get(at..end)).ok_or_else(short)?;
189        <[u8; 4]>::try_from(b).map(u32::from_le_bytes).map_err(|_| short())
190    };
191    let flags = u32_at(4)?;
192    if flags & F_EXTERNAL != 0 {
193        return Err(Error::Unsupported("external".into()));
194    }
195    if flags & STANDARD_FLAGS != STANDARD_FLAGS {
196        return Err(Error::Unsupported("RC4".into()));
197    }
198    // EncryptionHeader (§2.3.2): Flags, SizeExtra, AlgID, AlgIDHash, KeySize, ProviderType, two
199    // reserved words, then the CSP name, which isn't needed.
200    let header_size = usize::try_from(u32_at(8)?).map_err(|_| short())?;
201    if header_size < 32 {
202        return Err(bad("the encryption header is too short"));
203    }
204    let verifier_at = header_size.checked_add(12).filter(|&v| v <= info.len()).ok_or_else(short)?;
205    // AlgID 0 and AlgIDHash 0 mean "set by the flags": AES-128 and SHA-1.
206    let bits = match u32_at(20)? {
207        0 | ALG_AES_128 => 128,
208        ALG_AES_192 => 192,
209        ALG_AES_256 => 256,
210        id => return Err(Error::Unsupported(format!("cipher 0x{id:04X}"))),
211    };
212    let hash_id = u32_at(24)?;
213    if !matches!(hash_id, 0 | ALG_SHA1) {
214        return Err(Error::Unsupported(format!("hash 0x{hash_id:04X}")));
215    }
216    if u32_at(28)? != bits {
217        return Err(bad("the key size doesn't match the encryption algorithm"));
218    }
219    // EncryptionVerifier (§2.3.3): SaltSize (16), Salt, EncryptedVerifier, VerifierHashSize (20
220    // for SHA-1), EncryptedVerifierHash (32 bytes for AES).
221    if u32_at(verifier_at)? != SALT as u32 {
222        return Err(bad("the salt isn't 16 bytes"));
223    }
224    let field = |at: usize, len: usize| verifier_at.checked_add(at).and_then(|a| info.get(a..a.checked_add(len)?)).ok_or_else(short);
225    let salt = field(4, SALT)?;
226    let verifier = field(4 + SALT, BLOCK)?;
227    if u32_at(verifier_at.saturating_add(4 + SALT + BLOCK))? != SHA1_SIZE as u32 {
228        return Err(bad("the password verifier hash isn't SHA-1"));
229    }
230    let verifier_hash = field(8 + SALT + BLOCK, 2 * BLOCK)?;
231    Ok(Standard { key_bytes: bits as usize / 8, salt, verifier, verifier_hash })
232}

234/// Encrypt a zip package with `password` (agile encryption: AES-256-CBC, SHA-512, Word's spin
235/// count), as Word's "Encrypt with Password" does.

238}

240/// [`encrypt`] with a chosen spin count (tests use a small one; files should use the default).
241pub fn encrypt_with_spin_count(package: &[u8], password: &str, spin_count: u32) -> Result<Vec<u8>, Error> {
242    check_password(password)?;
243    let spin_count = spin_count.min(MAX_SPIN_COUNT);
244    let hash = Hash::Sha512;
245    let key_bytes = 32;
246    let key_data_salt = random(SALT)?;
247    let password_salt = random(SALT)?;
248    let key = random(key_bytes)?;
249    let verifier = random(SALT)?;
250    // The HMAC key is a salt as long as the hash (MS-OFFCRYPTO §2.3.4.14).
251    let hmac_salt = random(hash.size())?;
252
253    // The package, segment by segment; the last segment is zero-padded to the block size.
254    let mut stream = Vec::with_capacity(package.len().saturating_add(8 + BLOCK));
255    stream.extend_from_slice(&(package.len() as u64).to_le_bytes());
256    for (i, seg) in package.chunks(SEGMENT).enumerate() {
257        let i = u32::try_from(i).map_err(|_| bad("the document is too large to encrypt"))?;
258        let iv = make_iv(hash, &key_data_salt, Some(&i.to_le_bytes()), BLOCK);
259        stream.extend_from_slice(&cbc_encrypt(&key, &iv, &padded(seg))?);
260    }
261
262    // The password key encryptor (§2.3.4.13).
263    let hn = iterate(hash, &password_salt, password, spin_count);
264    let iv = make_iv(hash, &password_salt, None, BLOCK);
265    let enc_verifier_input = cbc_encrypt(&derive_key(hash, &hn, &BK_VERIFIER_INPUT, key_bytes), &iv, &padded(&verifier))?;
266    let enc_verifier_hash = cbc_encrypt(&derive_key(hash, &hn, &BK_VERIFIER_VALUE, key_bytes), &iv, &padded(&hash.hash(&[&verifier])))?;
267    let enc_key_value = cbc_encrypt(&derive_key(hash, &hn, &BK_KEY_VALUE, key_bytes), &iv, &padded(&key))?;
268
269    // Data integrity (§2.3.4.14).
270    let enc_hmac_key = cbc_encrypt(&key, &make_iv(hash, &key_data_salt, Some(&BK_HMAC_KEY), BLOCK), &padded(&hmac_salt))?;
271    let hmac = hash.hmac(&hmac_salt, &stream)?;
272    let enc_hmac_value = cbc_encrypt(&key, &make_iv(hash, &key_data_salt, Some(&BK_HMAC_VALUE), BLOCK), &padded(&hmac))?;
273
274    let b64 = |b: &[u8]| base64::engine::general_purpose::STANDARD.encode(b);
275    let params = format!(
276        r#"saltSize="{SALT}" blockSize="{BLOCK}" keyBits="{}" hashSize="{}" cipherAlgorithm="AES" cipherChaining="ChainingModeCBC" hashAlgorithm="{}""#,
277        key_bytes * 8,
278        hash.size(),
279        hash.name(),
280    );
281    let xml = format!(
282        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\r\n\
283         <encryption xmlns=\"{NS_ENCRYPTION}\" xmlns:p=\"{NS_PASSWORD}\">\
284         <keyData {params} saltValue=\"{}\"/>\
285         <dataIntegrity encryptedHmacKey=\"{}\" encryptedHmacValue=\"{}\"/>\
286         <keyEncryptors><keyEncryptor uri=\"{NS_PASSWORD}\">\
287         <p:encryptedKey spinCount=\"{spin_count}\" {params} saltValue=\"{}\" encryptedVerifierHashInput=\"{}\" encryptedVerifierHashValue=\"{}\" encryptedKeyValue=\"{}\"/>\
288         </keyEncryptor></keyEncryptors></encryption>",
289        b64(&key_data_salt),
290        b64(&enc_hmac_key),
291        b64(&enc_hmac_value),
292        b64(&password_salt),
293        b64(&enc_verifier_input),
294        b64(&enc_verifier_hash),
295        b64(&enc_key_value),
296    );
297    // EncryptionInfo (§2.3.4.10): version 4.4, reserved 0x40, the XML.
298    let mut info = vec![4, 0, 4, 0, 0x40, 0, 0, 0];
299    info.extend_from_slice(xml.as_bytes());
300
301    compound(&[
302        ("\u{6}DataSpaces/Version", data_space_version()),
303        ("\u{6}DataSpaces/DataSpaceMap", data_space_map()),
304        ("\u{6}DataSpaces/DataSpaceInfo/StrongEncryptionDataSpace", data_space_definition()),
305        ("\u{6}DataSpaces/TransformInfo/StrongEncryptionTransform/\u{6}Primary", primary_transform()),
306        ("EncryptionInfo", info),
307        ("EncryptedPackage", stream),
308    ])
309}

317/// A package encrypted with the *Standard* method (§2.3.4.5–§2.3.4.9: AES-ECB, SHA-1, 50,000
318/// spins), as Word 2007 wrote it, for tests and fixtures: files are saved with the agile method
319/// ([`encrypt`]). `key_bits` is 128, 192 or 256. The compound file holds just the two streams
320/// [`decrypt`] reads.
321#[doc(hidden)]
322pub fn encrypt_standard(package: &[u8], password: &str, key_bits: u32) -> Result<Vec<u8>, Error> {
323    check_password(password)?;
324    let alg_id = match key_bits {
325        128 => ALG_AES_128,
326        192 => ALG_AES_192,
327        256 => ALG_AES_256,
328        _ => return Err(bad(format!("unsupported AES key size {key_bits}"))),
329    };
330    let salt = random(SALT)?;
331    let verifier = random(SALT)?;
332    let key = standard_key(&salt, password, key_bits as usize / 8);
333    let mut header = Vec::new();
334    for v in [STANDARD_FLAGS, 0, alg_id, ALG_SHA1, key_bits, PROV_RSA_AES, 0, 0] {
335        header.extend_from_slice(&v.to_le_bytes());
336    }
337    // CSPName: an empty null-terminated UTF-16 string.
338    header.extend_from_slice(&[0, 0]);
339    let mut info = vec![3, 0, 2, 0];
340    info.extend_from_slice(&STANDARD_FLAGS.to_le_bytes());
341    info.extend_from_slice(&(header.len() as u32).to_le_bytes());
342    info.extend_from_slice(&header);
343    info.extend_from_slice(&(SALT as u32).to_le_bytes());
344    info.extend_from_slice(&salt);
345    info.extend_from_slice(&ecb(&key, &verifier, true)?);
346    info.extend_from_slice(&(SHA1_SIZE as u32).to_le_bytes());
347    info.extend_from_slice(&ecb(&key, &padded(&Hash::Sha1.hash(&[&verifier])), true)?);
348    let mut stream = (package.len() as u64).to_le_bytes().to_vec();
349    stream.extend_from_slice(&ecb(&key, &padded(package), true)?);
350    compound(&[("EncryptionInfo", info), ("EncryptedPackage", stream)])
351}

353/// Refuse a password Word wouldn't accept: empty, or longer than [`MAX_PASSWORD_CHARS`].
354pub fn check_password(password: &str) -> Result<(), Error> {
355    if password.is_empty() {
356        return Err(Error::BadPassword("the password is empty".into()));
357    }
358    if password.chars().count() > MAX_PASSWORD_CHARS {
359        return Err(Error::BadPassword(format!("passwords are at most {MAX_PASSWORD_CHARS} characters")));
360    }
361    Ok(())
362}

376/// The hash algorithms agile encryption names that are supported.
377#[derive(Clone, Copy, Debug, PartialEq, Eq)]
378pub(crate) enum Hash {
379    Sha1,
380    Sha256,
381    Sha384,
382    Sha512,
383}
384
385impl Hash {
386    pub(crate) fn parse(s: &str) -> Option<Hash> {
387        match s.to_ascii_uppercase().replace('-', "").as_str() {
388            "SHA1" => Some(Hash::Sha1),
389            "SHA256" => Some(Hash::Sha256),
390            "SHA384" => Some(Hash::Sha384),
391            "SHA512" => Some(Hash::Sha512),
392            _ => None,
393        }
394    }
395
396    fn name(self) -> &'static str {
397        match self {
398            Hash::Sha1 => "SHA-1",
399            Hash::Sha256 => "SHA256",
400            Hash::Sha384 => "SHA384",
401            Hash::Sha512 => "SHA512",
402        }
403    }
404
405    pub(crate) fn size(self) -> usize {
406        match self {
407            Hash::Sha1 => 20,
408            Hash::Sha256 => 32,
409            Hash::Sha384 => 48,
410            Hash::Sha512 => 64,
411        }
412    }
413
414    /// The hash of the concatenated `parts`.
415    pub(crate) fn hash(self, parts: &[&[u8]]) -> Vec<u8> {
416        fn go<D: Digest>(parts: &[&[u8]]) -> Vec<u8> {
417            let mut d = D::new();
418            for p in parts {
419                d.update(p);
420            }
421            d.finalize().to_vec()
422        }
423        match self {
424            Hash::Sha1 => go::<sha1::Sha1>(parts),
425            Hash::Sha256 => go::<sha2::Sha256>(parts),
426            Hash::Sha384 => go::<sha2::Sha384>(parts),
427            Hash::Sha512 => go::<sha2::Sha512>(parts),
428        }
429    }
430
431    fn hmac(self, key: &[u8], msg: &[u8]) -> Result<Vec<u8>, Error> {
432        fn go<M: Mac + KeyInit>(key: &[u8], msg: &[u8]) -> Result<Vec<u8>, Error> {
433            let mut m = <M as KeyInit>::new_from_slice(key).map_err(|_| bad("bad integrity key"))?;
434            m.update(msg);
435            Ok(m.finalize().into_bytes().to_vec())
436        }
437        match self {
438            Hash::Sha1 => go::<hmac::Hmac<sha1::Sha1>>(key, msg),
439            Hash::Sha256 => go::<hmac::Hmac<sha2::Sha256>>(key, msg),
440            Hash::Sha384 => go::<hmac::Hmac<sha2::Sha384>>(key, msg),
441            Hash::Sha512 => go::<hmac::Hmac<sha2::Sha512>>(key, msg),
442        }
443    }
444}

446/// The password hash after `spin_count` iterations (§2.3.4.11): H0 = H(salt + password as
447/// UTF-16LE), Hn = H(iterator + Hn-1).
448fn iterate(hash: Hash, salt: &[u8], password: &str, spin_count: u32) -> Zeroizing<Vec<u8>> {
449    let pw: Zeroizing<Vec<u8>> = Zeroizing::new(password.encode_utf16().flat_map(u16::to_le_bytes).collect());
450    let mut h = Zeroizing::new(hash.hash(&[salt, &pw]));
451    for i in 0..spin_count.min(MAX_SPIN_COUNT) {
452        *h = hash.hash(&[&i.to_le_bytes(), &h]);
453    }
454    h
455}
456
457/// A key from the iterated password hash and a block key (§2.3.4.11): H(Hn + blockKey),
458/// truncated to `len`, or padded with 0x36.
459fn derive_key(hash: Hash, hn: &[u8], block_key: &[u8], len: usize) -> Zeroizing<Vec<u8>> {
460    let mut k = Zeroizing::new(hash.hash(&[hn, block_key]));
461    k.resize(len, 0x36);
462    k
463}
464
465/// Standard encryption's key (§2.3.4.7): the SHA-1 password hash after 50,000 iterations, hashed
466/// with block number 0; then X1 = SHA-1 of that hash XOR-ed into 64 bytes of 0x36, X2 the same
467/// with 0x5C, and the key is the first `len` bytes of X1 + X2.
468fn standard_key(salt: &[u8], password: &str, len: usize) -> Zeroizing<Vec<u8>> {
469    let hn = iterate(Hash::Sha1, salt, password, STANDARD_SPIN_COUNT);
470    let hfinal = Zeroizing::new(Hash::Sha1.hash(&[&hn, &0u32.to_le_bytes()]));
471    let x = |fill: u8| {
472        let mut buf = Zeroizing::new([fill; 64]);
473        for (b, h) in buf.iter_mut().zip(hfinal.iter()) {
474            *b ^= h;
475        }
476        Zeroizing::new(Hash::Sha1.hash(&[&buf[..]]))
477    };
478    let mut key = Zeroizing::new(x(0x36).to_vec());
479    key.extend_from_slice(&x(0x5C));
480    key.truncate(len);
481    key
482}

484/// AES-ECB without padding (Standard encryption, §2.3.4.7–§2.3.4.9): `data` must be whole blocks.
485fn ecb(key: &[u8], data: &[u8], encrypt: bool) -> Result<Vec<u8>, Error> {
486    let aes = Aes::new(key)?;
487    let (blocks, rest) = data.as_chunks::<BLOCK>();
488    if !rest.is_empty() {
489        return Err(bad("encrypted data isn't a whole number of blocks"));
490    }
491    let mut out = Vec::with_capacity(data.len());
492    for b in blocks {
493        let mut x = aes::Block::from(*b);
494        if encrypt {
495            aes.encrypt(&mut x);
496        } else {
497            aes.decrypt(&mut x);
498        }
499        out.extend_from_slice(&x);
500    }
501    Ok(out)
502}

504/// An initialization vector (§2.3.4.12): H(salt + blockKey), or the salt itself, truncated to
505/// the block size or padded with 0x36.
506fn make_iv(hash: Hash, salt: &[u8], block_key: Option<&[u8]>, block_size: usize) -> Vec<u8> {
507    let mut iv = match block_key {
508        Some(b) => hash.hash(&[salt, b]),
509        None => salt.to_vec(),
510    };
511    iv.resize(block_size, 0x36);
512    iv
513}

515/// `data` zero-padded to a whole number of blocks.
516fn padded(data: &[u8]) -> Zeroizing<Vec<u8>> {
517    let mut v = Zeroizing::new(data.to_vec());
518    v.resize(data.len().div_ceil(BLOCK).saturating_mul(BLOCK), 0);
519    v
520}

522pub(crate) fn random(n: usize) -> Result<Zeroizing<Vec<u8>>, Error> {
523    let mut v = Zeroizing::new(vec![0u8; n]);
524    getrandom::fill(&mut v).map_err(|e| bad(format!("no secure random numbers available ({e})")))?;
525    Ok(v)

528/// Compare without stopping at the first difference.
529pub(crate) fn same(a: &[u8], b: &[u8]) -> bool {
530    a.len() == b.len() && a.iter().zip(b).fold(0u8, |acc, (x, y)| acc | (x ^ y)) == 0

533/// AES with a 128-, 192- or 256-bit key.
534enum Aes {
535    A128(aes::Aes128),
536    A192(aes::Aes192),
537    A256(aes::Aes256),
538}
539
540impl Aes {
541    fn new(key: &[u8]) -> Result<Aes, Error> {
542        let err = |_| bad("bad AES key");
543        Ok(match key.len() {
544            16 => Aes::A128(aes::Aes128::new_from_slice(key).map_err(err)?),
545            24 => Aes::A192(aes::Aes192::new_from_slice(key).map_err(err)?),
546            32 => Aes::A256(aes::Aes256::new_from_slice(key).map_err(err)?),
547            n => return Err(bad(format!("unsupported AES key length {} bits", n * 8))),
548        })
549    }
550
551    fn encrypt(&self, b: &mut aes::Block) {
552        match self {
553            Aes::A128(c) => c.encrypt_block(b),
554            Aes::A192(c) => c.encrypt_block(b),
555            Aes::A256(c) => c.encrypt_block(b),
556        }
557    }
558
559    fn decrypt(&self, b: &mut aes::Block) {
560        match self {
561            Aes::A128(c) => c.decrypt_block(b),
562            Aes::A192(c) => c.decrypt_block(b),
563            Aes::A256(c) => c.decrypt_block(b),
564        }
565    }
566}
567
568/// AES-CBC without padding: `data` must be whole blocks.
569fn cbc_encrypt(key: &[u8], iv: &[u8], data: &[u8]) -> Result<Vec<u8>, Error> {
570    let aes = Aes::new(key)?;
571    let (blocks, rest) = data.as_chunks::<BLOCK>();
572    let mut prev: [u8; BLOCK] = iv.try_into().map_err(|_| bad("bad initialization vector"))?;
573    if !rest.is_empty() {
574        return Err(bad("encrypted data isn't a whole number of blocks"));
575    }
576    let mut out = Vec::with_capacity(data.len());
577    for b in blocks {
578        let mut x = aes::Block::default();
579        for ((o, p), v) in x.iter_mut().zip(b).zip(&prev) {
580            *o = p ^ v;
581        }
582        aes.encrypt(&mut x);
583        prev.copy_from_slice(&x);
584        out.extend_from_slice(&x);
585    }
586    Ok(out)
587}
588
589/// AES-CBC decryption without padding.
590fn cbc_decrypt(key: &[u8], iv: &[u8], data: &[u8]) -> Result<Vec<u8>, Error> {
591    let aes = Aes::new(key)?;
592    let (blocks, rest) = data.as_chunks::<BLOCK>();
593    if !rest.is_empty() {
594        return Err(bad("encrypted data isn't a whole number of blocks"));
595    }
596    let mut prev: [u8; BLOCK] = iv.try_into().map_err(|_| bad("bad initialization vector"))?;
597    let mut out = Vec::with_capacity(data.len());
598    for b in blocks {
599        let mut x = aes::Block::default();
600        x.copy_from_slice(b);
601        aes.decrypt(&mut x);
602        for (o, v) in x.iter_mut().zip(&prev) {
603            *o ^= v;
604        }
605        prev = *b;
606        out.extend_from_slice(&x);
607    }
608    Ok(out)
609}

643fn parse_info(info: &[u8]) -> Result<Info, Error> {
644    let (major, minor) = match info.get(..4) {
645        Some([a, b, c, d]) => (u16::from_le_bytes([*a, *b]), u16::from_le_bytes([*c, *d])),
646        _ => return Err(bad("EncryptionInfo is truncated")),
647    };
648    match (major, minor) {
649        (4, 4) => {}
650        (3 | 4, 3) => return Err(Error::Unsupported("Extensible".into())),
651        _ => return Err(Error::Unsupported(format!("version {major}.{minor}"))),
652    }
653    let options = crate::craft::xml::Options {
654        limits: crate::craft::xml::Limits { max_input: MAX_INFO as usize, ..crate::craft::xml::Limits::default() },
655        ..crate::craft::xml::Options::default()
656    };
657    let root = options.parse(info.get(8..).unwrap_or(&[])).map_err(|e| bad(format!("EncryptionInfo isn't readable XML ({e})")))?;
658    if root.local() != "encryption" {
659        return Err(bad("EncryptionInfo has no <encryption> element"));
660    }
661    let key_data = root.elements().find(|e| e.local() == "keyData").ok_or_else(|| bad("EncryptionInfo has no keyData"))?;
662    let kd = params(key_data)?;
663    // The integrity check is mandatory (§2.3.4.14): a file is not trusted to opt out of it.
664    let di = root
665        .elements()
666        .find(|e| e.local() == "dataIntegrity")
667        .ok_or_else(|| bad("EncryptionInfo has no dataIntegrity, so the document cannot be checked for tampering"))?;
668    let integrity = (b64_attr(di, "encryptedHmacKey")?, b64_attr(di, "encryptedHmacValue")?);
669    // The password key encryptor: the one with the password URI, whose `encryptedKey` has a
670    // spin count (a certificate key encryptor has neither).
671    let encryptors = root.elements().find(|e| e.local() == "keyEncryptors").ok_or_else(|| bad("EncryptionInfo has no keyEncryptors"))?;
672    let ek = encryptors
673        .elements()
674        .filter(|e| e.local() == "keyEncryptor" && e.attr("uri").is_none_or(|uri| uri == NS_PASSWORD))
675        .flat_map(|e| e.elements())
676        .find(|e| e.local() == "encryptedKey" && e.attr("spinCount").is_some())
677        .ok_or_else(|| Error::Unsupported("certificate".into()))?;
678    let pe = params(ek)?;
679    let spin_count: u32 = ek.attr("spinCount").and_then(|s| s.trim().parse().ok()).ok_or_else(|| bad("bad spinCount"))?;
680    if spin_count > MAX_SPIN_COUNT {
681        return Err(bad(format!("spinCount {spin_count} is over the allowed maximum")));
682    }
683    Ok(Info {
684        hash: kd.hash,
685        salt: kd.salt,
686        block_size: kd.block_size,
687        key_bytes: kd.key_bytes,
688        integrity,
689        password: PasswordEncryptor {
690            hash: pe.hash,
691            salt: pe.salt,
692            salt_size: pe.salt_size,
693            block_size: pe.block_size,
694            key_bytes: pe.key_bytes,
695            spin_count,
696            verifier_input: b64_attr(ek, "encryptedVerifierHashInput")?,
697            verifier_hash: b64_attr(ek, "encryptedVerifierHashValue")?,
698            key_value: b64_attr(ek, "encryptedKeyValue")?,
699        },
700    })
701}
702
703fn params(e: &El) -> Result<Params, Error> {
704    let num = |k: &str| -> Result<usize, Error> { e.attr(k).and_then(|s| s.trim().parse::<usize>().ok()).ok_or_else(|| bad(format!("bad {k}"))) };
705    let cipher = e.attr("cipherAlgorithm").unwrap_or("");
706    if !cipher.eq_ignore_ascii_case("AES") {
707        return Err(Error::Unsupported(cipher.to_string()));
708    }
709    let chaining = e.attr("cipherChaining").unwrap_or("");
710    if chaining != "ChainingModeCBC" {
711        return Err(Error::Unsupported(format!("{chaining} chaining")));
712    }
713    let name = e.attr("hashAlgorithm").unwrap_or("");
714    let hash = Hash::parse(name).ok_or_else(|| Error::Unsupported(format!("{name} hash")))?;
715    if num("hashSize")? != hash.size() {
716        return Err(bad("hashSize doesn't match the hash algorithm"));
717    }
718    let block_size = num("blockSize")?;
719    if block_size != BLOCK {
720        return Err(bad(format!("AES needs a block size of {BLOCK}, not {block_size}")));
721    }
722    let key_bits = num("keyBits")?;
723    if !matches!(key_bits, 128 | 192 | 256) {
724        return Err(bad(format!("unsupported AES key size {key_bits}")));
725    }
726    let salt = b64_attr(e, "saltValue")?;
727    let salt_size = num("saltSize")?;
728    if salt.is_empty() || salt.len() != salt_size {
729        return Err(bad("the salt doesn't match saltSize"));
730    }
731    Ok(Params { hash, salt, salt_size, block_size, key_bytes: key_bits / 8 })
732}

734fn b64_attr(e: &El, k: &str) -> Result<Vec<u8>, Error> {
735    let s = e.attr(k).ok_or_else(|| bad(format!("{k} is missing")))?;
736    let s: String = s.chars().filter(|c| !c.is_ascii_whitespace()).collect();
737    base64::engine::general_purpose::STANDARD.decode(s).map_err(|_| bad(format!("{k} isn't valid base64")))
738}

740/// A UNICODE-LP-P4 string (§2.1.2): byte length, UTF-16LE, zero-padded to 4 bytes.
741fn lp_p4(out: &mut Vec<u8>, s: &str) {
742    let units: Vec<u8> = s.encode_utf16().flat_map(u16::to_le_bytes).collect();
743    out.extend_from_slice(&(units.len() as u32).to_le_bytes());
744    out.extend_from_slice(&units);
745    while !out.len().is_multiple_of(4) {
746        out.push(0);
747    }
748}
