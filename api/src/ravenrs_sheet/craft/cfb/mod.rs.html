2//! Clean-room Compound File Binary format reader and writer ("OLE structured storage": the
3//! container of legacy Office files, Outlook `.msg` messages, encrypted OOXML packages and AAF),
4//! written from the public Microsoft Open Specification **[MS-CFB]: Compound File Binary File
5//! Format** (protocol revision 10.0 and earlier editions; the on-disk format has not changed
6//! since 1.0).
7//!
8//! A compound file is a small FAT file system inside one file: fixed-size sectors (512 bytes in
9//! version 3, 4096 bytes in version 4) chained by a file allocation table, a directory of 128-byte
10//! entries arranged as one red-black tree of siblings per storage, and a "mini stream" with a mini
11//! FAT for streams smaller than 4096 bytes.
12//!
13//! - [`CompoundFile::open`] parses a file held in memory (no copies of large streams; every chain
14//!   is bounds- and cycle-checked, so damaged files give errors, never panics or endless loops).
15//!   [`Limits`] caps the stream size and the number of directory entries.
16//! - [`Writer`] builds a tree of storages and streams and lays it out in one pass
17//!   ([`Writer::finish`]): stream sectors, mini stream, mini FAT, directory, FAT and DIFAT.
18//!   [`write_streams`] is the one-call form for a list of `(path, bytes)`.
19//! - [`directory_names`] and [`has_entry_named`] answer "does this file hold an entry called X"
20//!   for files too damaged to open.
21//!
22//! No dependencies beyond `std`, no `unsafe`, nothing platform-specific.

56/// Errors from the reader and writer.
57#[derive(Debug, Clone, PartialEq, Eq)]
58#[non_exhaustive]
59pub enum Error {
60    /// Not a compound file (bad signature or byte order mark).
61    NotCfb,
62    /// Structurally invalid or truncated.
63    Invalid(String),
64    /// No entry with that name / path.
65    NotFound(String),
66    /// A name the format cannot store (too long, illegal characters, duplicate).
67    BadName(String),
68    /// The file is well-formed so far but asks for more than a [`Limits`] value or a per-call cap.
69    LimitExceeded(String),
70}
71
72impl fmt::Display for Error {
73    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
74        match self {
75            Error::NotCfb => f.write_str("not a compound file"),
76            Error::Invalid(s) => write!(f, "invalid compound file: {s}"),
77            Error::NotFound(s) => write!(f, "no entry {s:?}"),
78            Error::BadName(s) => write!(f, "invalid entry name {s:?}"),
79            Error::LimitExceeded(s) => write!(f, "compound file limit exceeded: {s}"),
80        }
81    }
82}

93fn upper(u: u16) -> u16 {
94    if (b'a' as u16..=b'z' as u16).contains(&u) {
95        return u - 32;
96    }
97    if u < 0x80 {
98        return u;
99    }
100    match char::from_u32(u as u32) {
101        Some(c) => {
102            let mut up = c.to_uppercase();
103            match (up.next(), up.next()) {
104                (Some(x), None) if (x as u32) <= 0xFFFF => x as u32 as u16,
105                _ => u,
106            }
107        }
108        None => u,
109    }
110}

112/// The sibling order of directory entries (§2.6.4): shorter names first, then a code-unit
113/// comparison of the upper-cased names.
114pub fn compare_names(a: &str, b: &str) -> Ordering {
115    // No allocation: the iterators are walked in place (`child` calls this once per sibling).
116    let (la, lb) = (a.encode_utf16().count(), b.encode_utf16().count());
117    la.cmp(&lb).then_with(|| a.encode_utf16().map(upper).cmp(b.encode_utf16().map(upper)))
118}

120/// Check a directory entry name (§2.6.1): 1–31 UTF-16 code units, none of `/ \ : !`, and no NUL
121/// (the stored name is NUL-terminated, so a NUL would cut the name short when read back).
122pub fn validate_name(name: &str) -> Result<()> {
123    let n = name.encode_utf16().count();
124    if n == 0 || n > MAX_NAME_LEN || name.contains(['/', '\\', ':', '!', '\0']) {
125        return Err(Error::BadName(name.to_string()));
126    }
127    Ok(())
128}
