1//! A lenient look at the directory, for telling what a file is when it may be too damaged to
2//! open: only the header, the DIFAT and the directory chain are followed. No FAT is built and
3//! nothing is allocated beyond the names themselves.
4
5use crate::craft::cfb::read::{Limits, entry_name, span, u16_at, u32_at};
6use crate::craft::cfb::{ENDOFCHAIN, MAXREGSECT, compare_names, sniff};
7
8/// The names of the directory entries of a compound file, in directory order (the root entry
9/// first; unallocated entries, which have no name, are left out), or `None` when `bytes` is not a
10/// compound file or its directory cannot be followed.
11///
12/// More forgiving than [`crate::craft::cfb::CompoundFile::open`]: the byte order mark, the FAT sector count,
13/// the mini FAT, the mini stream and the sibling trees are not looked at, so entries no storage
14/// links to are listed too. A directory chain that loops, or that holds more than
15/// [`Limits::max_dir_entries`] entries, gives the names read up to that point.
16pub fn directory_names(bytes: &[u8], limits: &Limits) -> Option<Vec<String>> {
17    if bytes.len() < 512 || !sniff(bytes) {
18        return None;
19    }
20    let shift = u16_at(bytes, 30);
21    if shift != 9 && shift != 12 {
22        return None;
23    }
24    let size = 1usize << shift;
25    let per = size / 4;
26    // No chain is longer than the file has sectors: the bound on both walks below.
27    let max_sectors = bytes.len() / size + 1;
28    let sector = |id: u32| -> Option<&[u8]> {
29        let (start, end) = span(usize::try_from(id).ok()?, 1, size)?;
30        bytes.get(start..end)
31    };
32    // The DIFAT: the FAT's sector ids, 109 in the header then chained DIFAT sectors.
33    let mut difat: Vec<u32> = (0..109).map(|i| u32_at(bytes, 76 + i * 4)).filter(|&s| s <= MAXREGSECT).collect();
34    let mut next = u32_at(bytes, 68);
35    for _ in 0..max_sectors {
36        if next > MAXREGSECT {
37            break;
38        }
39        let s = sector(next)?;
40        difat.extend((0..per - 1).map(|i| u32_at(s, i * 4)).filter(|&x| x <= MAXREGSECT));
41        next = u32_at(s, (per - 1) * 4);
42    }
43    let fat_next = |id: u32| -> Option<u32> {
44        let idx = usize::try_from(id).ok()?;
45        Some(u32_at(sector(*difat.get(idx / per)?)?, (idx % per) * 4))
46    };
47    let mut names = Vec::new();
48    let mut id = u32_at(bytes, 48);
49    for _ in 0..max_sectors {
50        if id == ENDOFCHAIN {
51            return Some(names);
52        }
53        if id > MAXREGSECT {
54            return None;
55        }
56        for e in sector(id)?.as_chunks::<128>().0 {
57            // an allocated entry's name length counts its terminating NUL: 2 to 64 bytes
58            if !(2..=64).contains(&u16_at(e, 64)) {
59                continue;
60            }
61            if names.len() >= limits.max_dir_entries {
62                return Some(names);
63            }
64            names.push(entry_name(e));
65        }
66        id = fat_next(id)?;
67    }
68    // A cyclic chain: report what was read.
69    Some(names)
70}

72/// Whether the compound file `bytes` holds a directory entry called `name` (compared as sibling
73/// names are, so case-insensitively), at any depth. `false` when `bytes` does not start with the
74/// compound file signature.
75///
76/// When the directory cannot be followed (a truncated or damaged file), falls back to searching
77/// the bytes for the name in UTF-16LE, so a damaged file still answers. That fallback is
78/// case-sensitive and can match stream contents: use it to choose an error message, not to make a
79/// security decision.
80pub fn has_entry_named(bytes: &[u8], name: &str, limits: &Limits) -> bool {
81    if !sniff(bytes) {
82        return false;
83    }
84    match directory_names(bytes, limits) {
85        Some(names) => names.iter().any(|n| compare_names(n, name).is_eq()),
86        None => {
87            let needle: Vec<u8> = name.encode_utf16().flat_map(u16::to_le_bytes).collect();
88            !needle.is_empty() && bytes.windows(needle.len()).any(|w| w == needle.as_slice())
89        }
90    }
91}
