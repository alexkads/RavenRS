2//! Bounded XML for hostile input.
3//!
4//! - [`parse`] and [`Options::parse`] read bytes into an owned tree of [`El`] and [`Node`], with
5//!   limits on input size, nesting, node count and attributes per element ([`Limits`]).
6//! - [`Options::parse_streaming`] hands repeated elements to a callback, so a large part never
7//!   becomes a whole tree. [`Options::events`] is the pull reader underneath.
8//! - [`Names`] chooses how names are read: as written, by local name, or rewritten to the
9//!   caller's canonical prefixes by namespace URI.
10//! - [`esc`], [`esc_attr`], [`El::to_xml`], [`El::to_xml_pretty`] and [`Writer`] write XML.
11//!
12//! No document type definition is processed, and no entity is expanded except the five
13//! predefined ones (`&amp;` `&lt;` `&gt;` `&quot;` `&apos;`) and numeric character references.
14//! The text the reader produces is therefore never more than three times its input.
15//!
16//! ```
17//! let root = ravenrs_sheet::craft::xml::parse(b"<a x='1'><b>one &amp; two</b><b/></a>")?;
18//! assert_eq!(root.attr_u32("x"), Some(1));
19//! assert_eq!(root.child("b").map(|b| b.text().into_owned()).as_deref(), Some("one & two"));
20//! assert_eq!(root.to_xml(), r#"<a x="1"><b>one &amp; two</b><b/></a>"#);
21//! # Ok::<(), ravenrs_sheet::craft::xml::Error>(())
22//! ```

40/// Why a document was not read.
41#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
42pub enum Error {
43    /// The input is longer than [`Limits::max_input`].
44    #[error("the XML is larger than {0} bytes")]
45    TooLarge(usize),
46    /// Elements nest deeper than [`Limits::max_depth`].
47    #[error("XML elements nest deeper than {0} levels")]
48    TooDeep(usize),
49    /// The tree would hold more than [`Limits::max_nodes`] nodes.
50    #[error("the XML holds more than {0} nodes")]
51    TooManyNodes(usize),
52    /// One start tag has more than [`Limits::max_attrs`] attributes.
53    #[error("an XML element has more than {0} attributes")]
54    TooManyAttrs(usize),
55    /// The tree would hold more than [`Limits::max_attrs_total`] attributes.
56    #[error("the XML holds more than {0} attributes")]
57    TooManyAttrsTotal(usize),
58    /// The document has a document type declaration that [`Options::doctype`] refuses.
59    #[error("the XML has a document type declaration, which is not accepted")]
60    Doctype,
61    /// A reference to an entity that is neither predefined nor numeric, with
62    /// [`UnknownEntity::Error`].
63    #[error("the entity `&{name};` at byte {at} is not defined")]
64    UnknownEntity { at: usize, name: String },
65    /// Text that is not UTF-8, in a strict parse.
66    #[error("the XML is not valid UTF-8 at byte {0}")]
67    Encoding(usize),
68    /// Markup that cannot be read. `at` is a byte offset into the UTF-8 text ([`to_utf8`]).
69    #[error("malformed XML at byte {at}: {msg}")]
70    Syntax { at: usize, msg: String },
71    /// There is no root element.
72    #[error("the XML has no root element")]
73    NoRoot,
74}
75
76impl Error {
77    /// Was a limit exceeded (as opposed to the markup being wrong)?
78    pub fn is_limit(&self) -> bool {
79        matches!(self, Error::TooLarge(_) | Error::TooDeep(_) | Error::TooManyNodes(_) | Error::TooManyAttrs(_) | Error::TooManyAttrsTotal(_))
80    }
81}
