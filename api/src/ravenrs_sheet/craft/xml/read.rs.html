13/// The namespace the `xml` prefix is always bound to.
14pub const XML_NS: &str = "http://www.w3.org/XML/1998/namespace";

16/// What the reader refuses to exceed.
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub struct Limits {
19    /// Most bytes of input. Default 256 MiB.
20    pub max_input: usize,
21    /// Deepest element nesting: the root is at depth 1. Default 256.
22    pub max_depth: usize,
23    /// Most nodes in a tree: elements, text stretches and whatever else is kept. Elements
24    /// handed to a streaming callback leave the count with everything inside them. Default
25    /// 2,000,000: a tree node costs about 72 bytes before its strings, and text lies between
26    /// elements at most, so this is about a million elements of an indented document.
27    pub max_nodes: usize,
28    /// Most attributes on one start tag, namespace declarations and malformed ones included.
29    /// With [`Names::Canonical`] it is also the most namespace declarations in scope at once.
30    /// Default 1024.
31    pub max_attrs: usize,
32    /// Most attributes kept in a tree at once, over all its elements. An attribute costs 48
33    /// bytes before its strings and five bytes of input (` a=""`), so without this a document
34    /// inside every other limit could take ten times its size in attribute headers. Elements
35    /// handed to a streaming callback take their attributes out of the count. Default
36    /// 4,000,000: two for every node at the node limit.
37    pub max_attrs_total: usize,
38}
39
40impl Limits {
41    pub const DEFAULT: Limits =
42        Limits { max_input: 256 * 1024 * 1024, max_depth: 256, max_nodes: 2_000_000, max_attrs: 1024, max_attrs_total: 4_000_000 };
43}
44
45impl Default for Limits {
46    fn default() -> Limits {
47        Limits::DEFAULT
48    }

51/// How element and attribute names are read.
52#[derive(Clone, Copy, Default)]
53pub enum Names<'a> {
54    /// The qualified name as written: `p:sp`, `r:id`. Match with a `&str` when the prefixes

56    #[default]

81/// Canonical prefixes by namespace URI.
82///
83/// - An element in a listed namespace is named `prefix:local`; in an unlisted one,
84///   `unknown:local`; in no namespace (or with a prefix nothing declares), `local`. A namespace
85///   listed with the empty prefix reads unprefixed too: a vocabulary's own namespace.
86/// - A prefixed attribute is named the same way. An unprefixed attribute is in no namespace
87///   and keeps its name.
88/// - The `xml` prefix needs no entry: `xml:space` stays `xml:space`.
89#[derive(Clone, Copy)]
90pub struct Canonical<'a> {
91    /// (prefix, namespace URI). The first entry with a URI wins, so several URIs may share a
92    /// prefix (the Strict and Transitional forms of a vocabulary).
93    pub namespaces: &'a [(&'a str, &'a str)],
94    /// The prefix for a namespace that is not listed. `"?"` by default: it cannot be part of an
95    /// XML name, so such names never collide with real ones and [`El::write_xml`] leaves them
96    /// out.
97    pub unknown: &'a str,
98    /// Overrides the prefix of an element, for vocabularies where one namespace is read as
99    /// several depending on the element and its parent. `None` keeps [`NsName::prefix`]. The
100    /// prefixes it returns are constants of the caller's.
101    pub element: Option<&'a ElementHook<'a>>,
102    /// Rewrites an attribute value while the namespace declarations in scope are still known:
103    /// for values that hold prefixes (`mc:Choice/@Requires`, `xsi:type`). Called with the
104    /// element's name, the attribute's name, the value and the scope; `None` keeps the value.
105    pub attr_value: Option<&'a AttrValueHook<'a>>,
106}
107
108impl<'a> Canonical<'a> {
109    pub const fn new(namespaces: &'a [(&'a str, &'a str)]) -> Canonical<'a> {
110        Canonical { namespaces, unknown: "?", element: None, attr_value: None }
111    }
112    /// The prefix for `uri`: the listed one, `xml` for [`XML_NS`], else [`Canonical::unknown`].
113    pub fn prefix(&self, uri: &str) -> &'a str {
114        match self.namespaces.iter().find(|(_, u)| *u == uri) {
115            Some((p, _)) => p,
116            None if uri == XML_NS => "xml",
117            None => self.unknown,
118        }
119    }
120}

128    /// The namespace URI `prefix` is bound to; `""` is the default namespace.
129    pub fn uri(&self, prefix: &str) -> Option<&'s str> {
130        if prefix == "xml" {
131            return Some(XML_NS);
132        }

153pub enum Doctype {
154    /// Skip it.
155    #[default]

163/// How to read a document. `Options::default()` keeps names as written and is lenient.
164#[derive(Clone, Copy)]
165pub struct Options<'a> {
166    pub limits: Limits,
167    pub names: Names<'a>,
168    /// Keep namespace declarations (`xmlns`, `xmlns:p`) as attributes, as written. On by
169    /// default; [`Options::local`] and [`Options::canonical`] turn it off.
170    pub ns_decls: bool,
171    /// Refuse what a lenient read repairs. Lenient: an end tag closes the innermost open
172    /// element whatever its name, an end tag with nothing open is skipped, elements still open
173    /// at the end of the input are closed, reading stops when the root element ends, a
174    /// malformed attribute is skipped, and bytes that are not UTF-8 become U+FFFD. Strict: each
175    /// of those is an error, as are a second root element and text outside the root.
176    pub strict: bool,
177    pub unknown_entity: UnknownEntity,
178    pub doctype: Doctype,
179    /// Keep CDATA sections as [`Node::CData`] instead of joining them to the text around them.
180    pub cdata: bool,
181    /// Keep processing instructions inside the root element as [`Node::Pi`].
182    pub pis: bool,
183    /// Keep comments inside the root element as [`Node::Comment`].
184    pub comments: bool,
185    /// Drop the text of an element that has child elements when all of it is whitespace: the
186    /// indentation of an indented document.
187    pub drop_blank_text: bool,
188    /// For [`Options::parse_streaming`]: how far below the root an element may be and still be
189    /// handed out. 2 by default: children and grandchildren of the root.
190    pub split_depth: usize,
191}
192
193impl Default for Options<'_> {
194    fn default() -> Self {
195        Options {
196            limits: Limits::DEFAULT,
197            names: Names::AsWritten,
198            ns_decls: true,
199            strict: false,
200            unknown_entity: UnknownEntity::Drop,
201            doctype: Doctype::Ignore,
202            cdata: false,
203            pis: false,
204            comments: false,
205            drop_blank_text: false,
206            split_depth: 2,
207        }
208    }
209}

235    /// Reads a document into its root element.
236    pub fn parse(&self, bytes: &[u8]) -> Result<El, Error> {
237        self.parse_streaming(bytes, &[], |_| Ok::<(), Error>(()))
238    }
239
240    /// Reads a document, handing each complete element whose name is in `split` to `each`
241    /// instead of keeping it in the tree: worksheet rows, shared strings. Only elements at
242    /// most [`Options::split_depth`] levels below the root are handed out; a deeper element of
243    /// the same name stays in the tree. `split` names are matched against names as this reader
244    /// produces them. Elements handed out do not count toward [`Limits::max_nodes`] once they
245    /// are gone, so a part of any length can be read; each one is itself held to the limit.
246    ///
247    /// An error from `each` stops the read and is returned.
248    pub fn parse_streaming<E: From<Error>>(&self, bytes: &[u8], split: &[&str], each: impl FnMut(El) -> Result<(), E>) -> Result<El, E> {
249        match self.parse_streaming_partial(bytes, split, each) {
250            (_, Some(e)) => Err(e),
251            (root, None) => root.ok_or_else(|| Error::NoRoot.into()),
252        }
253    }
254
255    /// [`Options::parse_streaming`] that also returns what it had read when it failed: the
256    /// root with the children that were complete (and not handed out) before the error, and
257    /// the elements still open at that point with what they held. A worksheet whose rows
258    /// cannot all be read still gives its properties and merged cells. `(None, Some(_))` when
259    /// nothing was read; `(Some(_), None)` is a whole document.
260    ///
261    /// The partial tree is not a document: it stops where the error was, and an open element
262    /// named in `split` is attached to it instead of being handed to `each`.
263    pub fn parse_streaming_partial<E: From<Error>>(
264        &self,
265        bytes: &[u8],
266        split: &[&str],
267        mut each: impl FnMut(El) -> Result<(), E>,
268    ) -> (Option<El>, Option<E>) {
269        if bytes.len() > self.limits.max_input {
270            return (None, Some(Error::TooLarge(self.limits.max_input).into()));
271        }
272        let text = to_utf8(bytes);
273        let mut events = Events::new(*self, &text, false);
274        let mut tree = Builder { opts: self, split, stack: Vec::new(), marks: Vec::new(), kept: 0, attrs: 0, root: None, at: 0 };
275        let mut err = tree.run(&mut events, &mut each).err();
276        // What was open when the read stopped goes into its parent as it is.
277        while let Some(el) = tree.stack.pop() {
278            match tree.stack.last_mut() {
279                Some(parent) => parent.children.push(Node::El(el)),
280                None => tree.root = Some(el),
281            }
282        }
283        if tree.root.is_none() && err.is_none() {
284            err = Some(Error::NoRoot.into());
285        }
286        (tree.root, err)
287    }

289    /// The first element that matches, from its start tag alone: its name and attributes, no
290    /// children. Nothing after the tag is read, so unbalanced markup around it neither hides
291    /// nor moves it. `None` when there is none or the markup before it cannot be read.
292    pub fn first_start(&self, bytes: &[u8], pat: impl Pat) -> Option<El> {
293        if bytes.len() > self.limits.max_input {
294            return None;
295        }
296        let text = to_utf8(bytes);
297        // No tree is built, and markup that never closes its tags would otherwise look deep.
298        let opts = Options { limits: Limits { max_depth: usize::MAX, ..self.limits }, ..*self };
299        let mut events = Events::new(opts, &text, false);
300        loop {
301            match events.next().ok()?? {
302                Event::Start { name, attrs } if pat.matches(&name) => return Some(El { name, attrs, children: Vec::new() }),
303                _ => {}
304            }
305        }
306    }

308    /// A pull reader over `utf8`, which is UTF-8 text: pass other input through [`to_utf8`]
309    /// first. No tree is built, so [`Limits::max_nodes`] and [`Limits::max_input`] do not
310    /// apply; depth and attribute limits do.
311    pub fn events<'i>(&self, utf8: &'i [u8]) -> Events<'a, 'i> {
312        Events::new(*self, utf8, true)
313    }

317#[derive(Clone, Debug, PartialEq)]
318pub enum Event {

320    Start {

348/// A pull reader. See [`Options::events`].
349pub struct Events<'o, 'i> {
350    reader: Reader<&'i [u8]>,
351    opts: Options<'o>,
352    /// Whether [`Event::End`] carries the name.
353    end_names: bool,
354    open: Vec<Open>,
355    scope: Vec<(String, String)>,
356    /// The end of an empty element, due after its start.
357    pending: bool,
358    /// Whether a root element has started.
359    rooted: bool,

363fn at(pos: u64) -> usize {
364    usize::try_from(pos).unwrap_or(usize::MAX)
365}

367fn line_ends(s: Cow<'_, str>) -> String {
368    // XML 1.0 §2.11: a literal CR LF or CR is read as LF.
369    if s.contains('\r') { s.replace("\r\n", "\n").replace('\r', "\n") } else { s.into_owned() }
370}

372fn is_ns_decl(key: &[u8]) -> bool {
373    key == b"xmlns" || key.starts_with(b"xmlns:")
374}

376impl<'o, 'i> Events<'o, 'i> {
377    fn new(opts: Options<'o>, utf8: &'i [u8], end_names: bool) -> Self {
378        let mut reader = Reader::from_reader(utf8);
379        let cfg = reader.config_mut();
380        // The checks a strict read wants are made here, where the answer can be lenient.
381        cfg.check_end_names = false;
382        cfg.allow_unmatched_ends = true;
383        cfg.allow_dangling_amp = true;
384        cfg.trim_text_start = false;
385        cfg.trim_text_end = false;
386        cfg.expand_empty_elements = false;
387        Events { reader, opts, end_names, open: Vec::new(), scope: Vec::new(), pending: false, rooted: false, span: 0..0 }
388    }
389
390    /// The bytes of the input the last event was read from. Both events of an empty element
391    /// have the whole element.
392    pub fn span(&self) -> Range<usize> {
393        self.span.clone()
394    }
395
396    fn syntax<T>(&self, msg: impl Into<String>) -> Result<T, Error> {
397        Err(Error::Syntax { at: self.span.start, msg: msg.into() })
398    }
399
400    fn decode<'b>(&self, bytes: &'b [u8]) -> Result<Cow<'b, str>, Error> {
401        match std::str::from_utf8(bytes) {
402            Ok(s) => Ok(Cow::Borrowed(s)),
403            Err(e) if self.opts.strict => Err(Error::Encoding(self.span.start.saturating_add(e.valid_up_to()))),
404            Err(_) => Ok(String::from_utf8_lossy(bytes)),
405        }
406    }
407
408    fn unknown(&self, name: &str) -> Result<Option<String>, Error> {
409        match self.opts.unknown_entity {
410            UnknownEntity::Drop => Ok(None),
411            UnknownEntity::Literal => Ok(Some(format!("&{name};"))),
412            UnknownEntity::Error => Err(Error::UnknownEntity { at: self.span.start, name: name.to_string() }),
413        }
414    }
415
416    /// The next event, or `None` at the end of the input.
417    #[allow(clippy::should_implement_trait)]
418    pub fn next(&mut self) -> Result<Option<Event>, Error> {
419        if self.pending {
420            self.pending = false;
421            return Ok(Some(self.end()));
422        }
423        loop {
424            let from = at(self.reader.buffer_position());
425            let ev = match self.reader.read_event() {
426                Ok(ev) => ev,
427                Err(e) => return Err(Error::Syntax { at: at(self.reader.error_position()), msg: e.to_string() }),
428            };
429            self.span = from..at(self.reader.buffer_position());
430            if let Some(out) = self.step(ev)? {
431                return Ok(out);
432            }
433        }
434    }
435
436    /// `None` to read on, `Some(None)` at the end of the input.
437    fn step(&mut self, ev: Q<'i>) -> Result<Option<Option<Event>>, Error> {
438        let event = match ev {
439            Q::Start(e) => self.start(&e)?,
440            Q::Empty(e) => {
441                self.pending = true;
442                self.start(&e)?
443            }
444            Q::End(e) => {
445                let Some(top) = self.open.last() else {
446                    return if self.opts.strict { self.syntax("an end tag has no start tag") } else { Ok(None) };
447                };
448                if self.opts.strict && top.raw != e.name().as_ref() {
449                    return self.syntax(format!("the end tag `{}` does not match its start tag", String::from_utf8_lossy(e.name().as_ref())));
450                }
451                self.end()
452            }
453            Q::Text(t) => {
454                if t.is_empty() {
455                    return Ok(None);
456                }
457                // Outside the root element only white space may stand (XML 1.0 [1], [27]).
458                if self.outside_root() && t.iter().any(|b| !matches!(b, b' ' | b'\t' | b'\r' | b'\n')) {
459                    return self.syntax("text outside the root element");
460                }
461                Event::Text(line_ends(self.decode(&t)?))
462            }
463            Q::CData(_) | Q::GeneralRef(_) if self.outside_root() => return self.syntax("text outside the root element"),
464            Q::CData(t) => Event::CData(line_ends(self.decode(&t)?)),
465            Q::GeneralRef(r) => {
466                let name = self.decode(&r)?;
467                match entity(&name) {
468                    Some(c) => Event::Text(c.to_string()),
469                    None => match self.unknown(&name)? {
470                        Some(literal) => Event::Text(literal),
471                        None => return Ok(None),
472                    },
473                }
474            }
475            Q::PI(p) => Event::Pi { target: self.decode(p.target())?.into_owned(), data: self.decode(p.content())?.trim_start().to_string() },
476            Q::Comment(c) => Event::Comment(self.decode(&c)?.into_owned()),
477            Q::DocType(d) => {
478                let refuse = match self.opts.doctype {
479                    Doctype::Ignore => false,
480                    // A `[` in a quoted identifier is refused too, which fails safe.
481                    Doctype::RejectInternalSubset => d.contains(&b'['),
482                    Doctype::Reject => true,
483                };
484                return if refuse { Err(Error::Doctype) } else { Ok(None) };
485            }
486            Q::Decl(_) => return Ok(None),
487            Q::Eof => {
488                return if self.opts.strict && !self.open.is_empty() { self.syntax("an element is not closed") } else { Ok(Some(None)) };
489            }
490        };
491        Ok(Some(Some(event)))
492    }
493
494    fn end(&mut self) -> Event {
495        let (scope, name) = self.open.pop().map_or((0, String::new()), |o| (o.scope, o.name));
496        self.scope.truncate(scope);
497        Event::End { name }
498    }

500    /// The URI `prefix` is bound to here; `""` is the default namespace.
501    fn uri(&self, prefix: &str) -> Option<&str> {
502        Scope { bindings: &self.scope }.uri(prefix)
503    }
504
505    fn element_name(&self, raw: &str) -> String {
506        match self.opts.names {
507            Names::AsWritten => raw.to_string(),
508            Names::Local => local_of(raw).to_string(),
509            Names::Canonical(canon) => {
510                let (prefix, local) = raw.split_once(':').unwrap_or(("", raw));
511                let Some(uri) = self.uri(prefix) else { return local.to_string() };
512                let listed = canon.prefix(uri);
513                let parent = self.open.last().map(|o| o.name.as_str());
514                let chosen = canon.element.and_then(|hook| hook(&NsName { prefix: listed, uri, local, parent })).unwrap_or(listed);
515                if chosen.is_empty() { local.to_string() } else { format!("{chosen}:{local}") }
516            }
517        }
518    }
519
520    fn attr_name(&self, raw: &str) -> String {
521        match self.opts.names {
522            Names::AsWritten => raw.to_string(),
523            Names::Local => local_of(raw).to_string(),
524            Names::Canonical(canon) => match raw.split_once(':') {
525                Some((prefix, local)) => match self.uri(prefix).map(|uri| canon.prefix(uri)) {
526                    Some(p) if !p.is_empty() => format!("{p}:{local}"),
527                    _ => local.to_string(),
528                },
529                None => raw.to_string(),
530            },
531        }
532    }

534    fn attr_value(&self, raw: &[u8]) -> Result<String, Error> {
535        match decode_refs(&self.decode(raw)?, true, self.opts.unknown_entity) {
536            Ok(v) => Ok(v),
537            Err(name) => Err(Error::UnknownEntity { at: self.span.start, name }),
538        }
539    }

541    /// Reads the tag's namespace declarations into the scope, so that the tag's own name and
542    /// attributes resolve against them.
543    fn declare(&mut self, e: &BytesStart<'_>) -> Result<(), Error> {
544        let mut attrs = e.attributes();
545        attrs.with_checks(false);
546        // The attribute pass that follows counts every attribute; this one only has to stop.
547        for a in attrs.take(self.opts.limits.max_attrs).flatten() {
548            let key = a.key.as_ref();
549            if is_ns_decl(key) {
550                let prefix = self.decode(key.get(6..).unwrap_or_default())?.into_owned();
551                let uri = self.attr_value(&a.value)?;
552                self.scope.push((prefix, uri));
553            }
554        }
555        // Every name is resolved against the scope, so the scope as a whole is bounded too:
556        // declarations piled up over many levels would make each lookup long.
557        if self.scope.len() > self.opts.limits.max_attrs {
558            return Err(Error::TooManyAttrs(self.opts.limits.max_attrs));
559        }
560        Ok(())
561    }
562
563    /// In a strict read: is the reader before or after the root element?
564    fn outside_root(&self) -> bool {
565        self.opts.strict && self.open.is_empty()
566    }
567
568    fn start(&mut self, e: &BytesStart<'_>) -> Result<Event, Error> {
569        let limits = self.opts.limits;
570        if self.outside_root() && self.rooted {
571            return self.syntax("more than one root element");
572        }
573        self.rooted = true;
574        if self.open.len() >= limits.max_depth {
575            return Err(Error::TooDeep(limits.max_depth));
576        }
577        let canonical = match self.opts.names {
578            Names::Canonical(c) => Some(c),
579            _ => None,
580        };
581        let scope = self.scope.len();
582        if canonical.is_some() {
583            self.declare(e)?;
584        }
585        let raw = e.name();
586        let name = self.element_name(&self.decode(raw.as_ref())?);
587        let mut out = Vec::new();
588        let mut attrs = e.attributes();
589        attrs.with_checks(false);
590        for (seen, a) in attrs.enumerate() {
591            if seen >= limits.max_attrs {
592                return Err(Error::TooManyAttrs(limits.max_attrs));
593            }
594            let a = match a {
595                Ok(a) => a,
596                Err(err) if self.opts.strict => return self.syntax(err.to_string()),
597                Err(_) => continue,
598            };
599            let key = a.key.as_ref();
600            let decl = is_ns_decl(key);
601            if decl && !self.opts.ns_decls {
602                continue;
603            }
604            let key = self.decode(key)?;
605            let key = if decl { key.into_owned() } else { self.attr_name(&key) };
606            let mut value = self.attr_value(&a.value)?;
607            if let Some(hook) = canonical.and_then(|c| c.attr_value)
608                && let Some(rewritten) = hook(&name, &key, &value, &Scope { bindings: &self.scope })
609            {
610                value = rewritten;
611            }
612            out.push((key, value));
613        }
614        let keep_name = self.end_names || canonical.is_some_and(|c| c.element.is_some());
615        self.open.push(Open {
616            scope,
617            raw: if self.opts.strict { raw.as_ref().to_vec() } else { Vec::new() },
618            name: if keep_name { name.clone() } else { String::new() },
619        });
620        Ok(Event::Start { name, attrs: out })
621    }
622}

624struct Builder<'b, 'o> {
625    opts: &'b Options<'o>,
626    split: &'b [&'b str],
627    stack: Vec<El>,
628    /// The node and attribute counts when each open element started.
629    marks: Vec<(usize, usize)>,
630    kept: usize,
631    /// Attributes in the elements built so far and not handed out.
632    attrs: usize,
633    root: Option<El>,
634    /// Where the event being handled starts, for errors.
635    at: usize,
636}
637
638impl Builder<'_, '_> {
639    fn run<E: From<Error>>(&mut self, events: &mut Events<'_, '_>, each: &mut impl FnMut(El) -> Result<(), E>) -> Result<(), E> {
640        while let Some(ev) = events.next()? {
641            self.at = events.span().start;
642            self.event(ev, each)?;
643            if !self.opts.strict && self.root.is_some() {
644                break;
645            }
646        }
647        // A lenient read keeps what a truncated document had; a strict one failed above.
648        while let Some(el) = self.stack.pop() {
649            self.finish(el, each)?;
650        }
651        Ok(())
652    }
653
654    fn count(&mut self) -> Result<(), Error> {
655        self.kept = self.kept.saturating_add(1);
656        if self.kept > self.opts.limits.max_nodes { Err(Error::TooManyNodes(self.opts.limits.max_nodes)) } else { Ok(()) }
657    }
658
659    fn push(&mut self, node: Node) -> Result<(), Error> {
660        if !self.stack.is_empty() {
661            self.count()?;
662        }
663        if let Some(top) = self.stack.last_mut() {
664            top.children.push(node);
665        }
666        Ok(())
667    }
668
669    fn text(&mut self, text: String) -> Result<(), Error> {
670        match self.stack.last_mut() {
671            Some(top) => {
672                if let Some(Node::Text(prev)) = top.children.last_mut() {
673                    prev.push_str(&text);
674                    return Ok(());
675                }
676                self.push(Node::Text(text))
677            }
678            None if self.opts.strict && !text.trim().is_empty() => Err(Error::Syntax { at: self.at, msg: "text outside the root element".into() }),
679            None => Ok(()),
680        }
681    }
682
683    fn event<E: From<Error>>(&mut self, ev: Event, each: &mut impl FnMut(El) -> Result<(), E>) -> Result<(), E> {
684        match ev {
685            Event::Start { name, attrs } => {
686                if self.root.is_some() {
687                    return Err(Error::Syntax { at: self.at, msg: "more than one root element".into() }.into());
688                }
689                self.marks.push((self.kept, self.attrs));
690                self.attrs = self.attrs.saturating_add(attrs.len());
691                if self.attrs > self.opts.limits.max_attrs_total {
692                    return Err(Error::TooManyAttrsTotal(self.opts.limits.max_attrs_total).into());
693                }
694                self.stack.push(El { name, attrs, children: Vec::new() });
695            }
696            Event::End { .. } => {
697                if let Some(el) = self.stack.pop() {
698                    self.finish(el, each)?;
699                }
700            }
701            Event::Text(t) => self.text(t)?,
702            Event::CData(t) if self.opts.cdata => self.push(Node::CData(t))?,
703            Event::CData(t) => self.text(t)?,
704            Event::Pi { target, data } if self.opts.pis => self.push(Node::Pi { target, data })?,
705            Event::Comment(c) if self.opts.comments => self.push(Node::Comment(c))?,
706            Event::Pi { .. } | Event::Comment(_) => {}
707        }
708        Ok(())
709    }
710
711    /// Attaches a complete element to its parent, hands it out, or makes it the root.
712    fn finish<E: From<Error>>(&mut self, mut el: El, each: &mut impl FnMut(El) -> Result<(), E>) -> Result<(), E> {
713        let (mark, attrs_mark) = self.marks.pop().unwrap_or((0, 0));
714        if self.opts.drop_blank_text && el.children.iter().any(|n| matches!(n, Node::El(_))) {
715            let is_text = |n: &Node| matches!(n, Node::Text(_));
716            if el.children.iter().all(|n| !is_text(n) || matches!(n, Node::Text(t) if t.trim().is_empty())) {
717                let before = el.children.len();
718                el.children.retain(|n| !is_text(n));
719                self.kept = self.kept.saturating_sub(before - el.children.len());
720            }
721        }
722        let depth = self.stack.len();
723        if depth == 0 {
724            self.count()?;
725            self.root = Some(el);
726        } else if depth <= self.opts.split_depth && self.split.contains(&el.name.as_str()) {
727            each(el)?;
728            // The element and everything inside it have left the tree.
729            self.kept = mark;
730            self.attrs = attrs_mark;
731        } else {
732            self.push(Node::El(el))?;
733        }
734        Ok(())
735    }
736}

738/// The input as UTF-8 text for the reader: a UTF-8 byte-order mark is removed, and UTF-16 (with
739/// a byte-order mark, or recognisable without one by the zero bytes of its first two
740/// characters) is transcoded. An unpaired surrogate becomes U+FFFD and a trailing odd byte is
741/// dropped. The result is at most 1.5 times the input. Any other input is returned as it is;
742/// the encoding an XML declaration names is not consulted.
743pub fn to_utf8(bytes: &[u8]) -> Cow<'_, [u8]> {
744    let (little_endian, body) = match bytes {
745        [0xEF, 0xBB, 0xBF, rest @ ..] => return Cow::Borrowed(rest),
746        [0xFF, 0xFE, rest @ ..] => (true, rest),
747        [0xFE, 0xFF, rest @ ..] => (false, rest),
748        [a, 0, b, 0, ..] if *a != 0 && *b != 0 => (true, bytes),
749        [0, a, 0, b, ..] if *a != 0 && *b != 0 => (false, bytes),
750        _ => return Cow::Borrowed(bytes),
751    };
752    let units = body.as_chunks::<2>().0.iter().map(|pair| if little_endian { u16::from_le_bytes(*pair) } else { u16::from_be_bytes(*pair) });
753    Cow::Owned(char::decode_utf16(units).map(|c| c.unwrap_or(char::REPLACEMENT_CHARACTER)).collect::<String>().into_bytes())
754}

756/// The character a reference names, without its `&` and `;`: one of the five predefined
757/// entities (`amp`, `lt`, `gt`, `quot`, `apos`) or a numeric character reference (`#65`,
758/// `#x41`, zero-padded or not). `None` for anything else, for a number that is not a character
759/// XML can carry, and for a number with a sign.
760pub fn entity(name: &str) -> Option<char> {
761    let number = match name {
762        "amp" => return Some('&'),
763        "lt" => return Some('<'),
764        "gt" => return Some('>'),
765        "quot" => return Some('"'),
766        "apos" => return Some('\''),
767        _ => name.strip_prefix('#')?,
768    };
769    let (digits, radix) = match number.strip_prefix(['x', 'X']) {
770        Some(hex) => (hex, 16),
771        None => (number, 10),
772    };
773    if digits.is_empty() {
774        return None;
775    }
776    let code = digits.chars().try_fold(0u32, |n, c| n.checked_mul(radix)?.checked_add(c.to_digit(radix)?))?;
777    char::from_u32(code).filter(|c| is_xml_char(*c))
778}

780/// Resolves the references in `s`, and with `attr` turns literal whitespace into spaces as
781/// attribute-value normalisation does. An `&` that starts no reference stays as it is. `Err` is
782/// the name of an unknown entity under [`UnknownEntity::Error`].
783fn decode_refs(s: &str, attr: bool, unknown: UnknownEntity) -> Result<String, String> {
784    let special = |c: char| c == '&' || (attr && matches!(c, '\t' | '\n' | '\r'));
785    let mut out = String::with_capacity(s.len());
786    let mut rest = s;
787    while let Some(i) = rest.find(special) {
788        out.push_str(rest.get(..i).unwrap_or_default());
789        let tail = rest.get(i..).unwrap_or_default();
790        let Some(after) = tail.strip_prefix('&') else {
791            out.push(' ');
792            rest = tail.strip_prefix("\r\n").or_else(|| tail.get(1..)).unwrap_or_default();
793            continue;
794        };
795        // Only the bytes a name or a number can have are scanned, so a run of `&` costs one
796        // look each however far away the next `;` is.
797        let len = after.bytes().take_while(|b| b.is_ascii_alphanumeric() || matches!(b, b'#' | b'_' | b'-' | b'.' | b':')).count();
798        let name = after.get(..len).unwrap_or_default();
799        if len == 0 || after.as_bytes().get(len) != Some(&b';') {
800            out.push('&');
801            rest = after;
802            continue;
803        }
804        match (entity(name), unknown) {
805            (Some(c), _) => out.push(c),
806            (None, UnknownEntity::Drop) => {}
807            (None, UnknownEntity::Literal) => {
808                out.push('&');
809                out.push_str(name);
810                out.push(';');
811            }
812            (None, UnknownEntity::Error) => return Err(name.to_string()),
813        }
814        rest = after.get(len + 1..).unwrap_or_default();
815    }
816    out.push_str(rest);
817    Ok(out)
818}

820/// `s` with its predefined and numeric references resolved ([`entity`]). Anything else that
821/// starts with `&` stays as it is.
822pub fn unescape(s: &str) -> String {
823    decode_refs(s, false, UnknownEntity::Literal).unwrap_or_else(|_| s.to_string())
