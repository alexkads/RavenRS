1//! ZIP structure checks that run before a ZIP reader indexes the archive, and bounded entry reads.
2//!
3//! A ZIP file ends with an end-of-central-directory record (EOCD) naming the offset and size of a
4//! central directory that lists every entry, optionally preceded by a ZIP64 pair when the archive
5//! is too large for the classic fields. [`preflight`] validates those records and walks the
6//! directory on the raw bytes, so a hostile entry count or directory size is refused before a
7//! reader can spend memory and time on it. [`read_entry`] then bounds one entry's inflation.
8//!
9//! Only single-disk archives are accepted: split archives and self-extracting prefixed archives
10//! are rejected, because their relative offsets would make this check disagree with the bytes a
11//! later reader indexes. The algorithm follows the ZIP specification (PKWARE APPNOTE, "ZIP File
12//! Format"); no third-party code is involved.
13

19const EOCD_SIGNATURE: &[u8; 4] = b"PK\x05\x06";
20const ZIP64_EOCD_SIGNATURE: &[u8; 4] = b"PK\x06\x06";
21const ZIP64_LOCATOR_SIGNATURE: &[u8; 4] = b"PK\x06\x07";
22const CENTRAL_FILE_SIGNATURE: &[u8; 4] = b"PK\x01\x02";
23const CENTRAL_DIGITAL_SIGNATURE: &[u8; 4] = b"PK\x05\x05";

25/// The classic end-of-central-directory record, fixed part.
26const EOCD_LEN: usize = 22;
27/// The ZIP64 end-of-central-directory locator.
28const ZIP64_LOCATOR_LEN: usize = 20;
29/// The fixed part of a ZIP64 end-of-central-directory record: 12 bytes of signature and record
30/// size, then 44 bytes of fixed fields.
31const ZIP64_EOCD_FIXED_LEN: usize = 44;
32const ZIP64_EOCD_HEADER_LEN: u64 = 12;
33/// The fixed part of a central-directory file header, before the name, extra field and comment.
34const CENTRAL_FILE_LEN: usize = 46;
35/// The largest archive comment the classic end record can describe.
36const MAX_COMMENT_LEN: usize = u16::MAX as usize;
37/// Central-directory file header field offsets.
38const NAME_LEN_AT: usize = 28;
39const EXTRA_LEN_AT: usize = 30;
40const COMMENT_LEN_AT: usize = 32;
41const ENTRY_DISK_AT: usize = 34;

43/// Application-owned limits for [`preflight`].
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub struct Limits {
46    /// The most central-directory entries the archive may declare.
47    pub max_entries: u64,
48    /// The most bytes the central directory may occupy.
49    pub max_central_directory_bytes: u64,
50    /// The most bytes a central-directory entry name may occupy.
51    pub max_name_bytes: u16,
52}

54/// What [`preflight`] found: the directory's declared totals.
55#[derive(Clone, Copy, Debug, PartialEq, Eq)]
56pub struct Directory {
57    /// The number of central-directory entries, counted by walking them.
58    pub entries: u64,
59    /// The declared size of the central directory in bytes.
60    pub central_directory_bytes: u64,
61}

63/// Why [`preflight`] refused an archive.
64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub enum Error {
66    /// No end-of-central-directory record spans the end of the file.
67    MissingEndRecord,
68    /// More than one end record could claim the end of the file, so which one a reader would use
69    /// is not determined; none is trusted.
70    AmbiguousEndRecords,
71    /// The structure is malformed or unsupported. The message names what was wrong.
72    Invalid(&'static str),
73    /// The archive declares more entries than the limit admits.
74    TooManyEntries {
75        /// The declared entry count.
76        actual: u64,
77        /// The caller's limit.
78        limit: u64,
79    },
80    /// The central directory is larger than the limit admits.
81    CentralDirectoryTooLarge {
82        /// The declared directory size in bytes.
83        actual: u64,
84        /// The caller's limit.
85        limit: u64,
86    },
87    /// A central-directory entry name is longer than the limit admits.
88    NameTooLong {
89        /// The declared name length in bytes.
90        actual: u64,
91        /// The caller's limit.
92        limit: u64,
93    },
94}
95
96impl fmt::Display for Error {
97    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
98        match self {
99            Self::MissingEndRecord => f.write_str("the ZIP end record is missing"),
100            Self::AmbiguousEndRecords => f.write_str("the ZIP has more than one plausible end record"),
101            Self::Invalid(message) => write!(f, "invalid ZIP structure: {message}"),
102            Self::TooManyEntries { actual, limit } => write!(f, "the ZIP declares {actual} entries; the limit is {limit}"),
103            Self::CentralDirectoryTooLarge { actual, limit } => write!(f, "the ZIP central directory is {actual} bytes; the limit is {limit}"),
104            Self::NameTooLong { actual, limit } => write!(f, "a ZIP entry name is {actual} bytes; the limit is {limit}"),
105        }
106    }
107}

111/// Why [`read_entry`] refused an entry.
112///
113/// Not `PartialEq`: an [`std::io::Error`] is not comparable, so tests compare the displayed
114/// message instead.
115#[derive(Debug)]
116pub enum ReadError {
117    /// The entry declares more bytes than the per-entry limit admits.
118    DeclaredTooLarge {
119        /// The declared size.
120        declared: u64,
121        /// The caller's per-entry limit.
122        limit: u64,
123    },
124    /// The entry produced more bytes than the smaller of its declared size and the per-entry
125    /// limit: the declared size lied, or the data is a decompression bomb.
126    Overrun {
127        /// The per-entry bound that was passed.
128        limit: u64,
129    },
130    /// The entry needed more bytes than the total budget had left.
131    Budget(Exceeded),
132    /// The reader failed.
133    Io(std::io::Error),
134}
135
136impl fmt::Display for ReadError {
137    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
138        match self {
139            Self::DeclaredTooLarge { declared, limit } => write!(f, "the entry declares {declared} bytes; the limit is {limit}"),
140            Self::Overrun { limit } => write!(f, "the entry produced more than the {limit} bytes it is allowed"),
141            Self::Budget(exceeded) => exceeded.fmt(f),
142            Self::Io(error) => error.fmt(f),
143        }
144    }
145}

169/// The most bytes [`read_entry`] reserves up front, however large the entry declares itself.
170///
171/// A declared size is attacker-controlled, so reserving it would let a ten-byte archive claim a
172/// gigabyte; the buffer grows as real bytes arrive instead.
173pub const MAX_PREALLOCATION: u64 = 64 * 1024;
174
175/// Validate the end records and walk the declared central directory without allocating.
176///
177/// Checks, in order:
178///
179/// * the end-of-central-directory record: exactly one candidate must span the end of the file, so
180///   an archive that hides a second plausible record is refused rather than guessed at;
181/// * the ZIP64 pair when the classic record uses a sentinel or a locator is present: a single disk,
182///   a locator immediately before the end record, the ZIP64 record ending at that locator, and no
183///   disagreement between the classic and ZIP64 fields;
184/// * the declared entry count, directory size and entry-name lengths against `limits`;
185/// * the directory arithmetic: its offset and size must not overflow and must reach the end
186///   records, so a lying offset or size cannot point the reader elsewhere;
187/// * every entry header in the directory: its signature, its disk number, its variable-length
188///   fields, and that the entries exactly fill the declared directory, allowing one optional
189///   central-directory digital signature at its end.
190///
191/// Returns the declared entry count and directory size. Malformed input is an error, never a panic.
192pub fn preflight(bytes: &[u8], limits: Limits) -> Result<Directory, Error> {
193    let eocd_at = find_end_record(bytes)?;
194    let directory = end_record_totals(bytes, eocd_at)?;
195
196    if directory.entries > limits.max_entries {
197        return Err(Error::TooManyEntries { actual: directory.entries, limit: limits.max_entries });
198    }
199    if directory.size > limits.max_central_directory_bytes {
200        return Err(Error::CentralDirectoryTooLarge { actual: directory.size, limit: limits.max_central_directory_bytes });
201    }
202
203    let directory_end = directory.offset.checked_add(directory.size).ok_or(Error::Invalid("central-directory offset overflow"))?;
204    let structure_at = u64::try_from(directory.structure_at).map_err(|_| Error::Invalid("end-record offset does not fit"))?;
205    if directory_end != structure_at {
206        return Err(Error::Invalid("central-directory offset and size do not reach the end records"));
207    }
208    let start = usize::try_from(directory.offset).map_err(|_| Error::Invalid("central-directory offset does not fit this platform"))?;
209    let end = usize::try_from(directory_end).map_err(|_| Error::Invalid("central-directory end does not fit this platform"))?;
210
211    walk_directory(bytes, start, end, directory.entries, limits.max_name_bytes)?;
212    Ok(Directory { entries: directory.entries, central_directory_bytes: directory.size })
213}

215/// Read one entry up to the smallest of `max`, the total budget's remainder and `declared`.
216///
217/// The declared size is refused outright when it exceeds `max`. At most [`MAX_PREALLOCATION`] bytes
218/// are reserved however large the entry claims to be. The bytes are read up to the bound and only
219/// the bytes actually read are charged to `total`, so a declaration that overstates the entry
220/// costs nothing extra.
221///
222/// More bytes arriving than the bound admits is the hostile case: [`ReadError::Overrun`] when the
223/// per-entry bound or the declaration was what stopped the read (a lying size or a decompression
224/// bomb), and [`ReadError::Budget`] when the total budget was already the tighter bound.
225///
226/// Works with any [`Read`], including a `zip` crate entry reader. An I/O failure is
227/// [`ReadError::Io`]; the bytes read before it are not charged.
228pub fn read_entry(r: impl Read, declared: u64, max: u64, total: &mut Budget) -> Result<Vec<u8>, ReadError> {
229    if declared > max {
230        return Err(ReadError::DeclaredTooLarge { declared, limit: max });
231    }
232    let per_entry = declared.min(max);
233    let budget_left = total.remaining();
234    let bound = per_entry.min(budget_left);
235    // One byte past the bound has to be readable to tell "exactly the bound" from "more".
236    let mut out = Vec::with_capacity(usize::try_from(bound.min(MAX_PREALLOCATION)).unwrap_or(0));
237    let mut reader = r.take(bound.saturating_add(1));
238    reader.read_to_end(&mut out)?;
239    let read = u64::try_from(out.len()).unwrap_or(u64::MAX);
240    if read > bound {
241        // The budget was the tighter bound, so the total limit is what refused the entry.
242        if per_entry > budget_left {
243            return Err(ReadError::Budget(Exceeded { what: total.what(), limit: total.limit(), requested: read }));
244        }
245        return Err(ReadError::Overrun { limit: bound });
246    }
247    total.charge(read)?;
248    Ok(out)
249}

251/// The declared entry count, directory size and offset, and where the end records begin.
252struct EndRecord {
253    entries: u64,
254    size: u64,
255    offset: u64,
256    /// The offset of the first end record: the ZIP64 record when there is one, else the EOCD.
257    structure_at: usize,
258}

260/// Find the single end-of-central-directory record that ends the file.
261///
262/// A record is a candidate when its signature is present, its declared comment length puts its end
263/// exactly at the end of the file, and it lies in the last 64 KiB plus 22 bytes — the only place
264/// the format allows one. The search runs backwards, so the record nearest the end is considered
265/// first; finding a second candidate means the archive is ambiguous and is refused.
266fn find_end_record(bytes: &[u8]) -> Result<usize, Error> {
267    let last = bytes.len().checked_sub(EOCD_LEN).ok_or(Error::MissingEndRecord)?;
268    let first = bytes.len().saturating_sub(EOCD_LEN.saturating_add(MAX_COMMENT_LEN));
269    let mut found = None;
270    for at in (first..=last).rev() {
271        if !signature_at(bytes, at, EOCD_SIGNATURE) {
272            continue;
273        }
274        let Some(comment_len) = u16_at(bytes, at + 20).map(usize::from) else { continue };
275        let Some(end) = at.checked_add(EOCD_LEN).and_then(|value| value.checked_add(comment_len)) else { continue };
276        if end != bytes.len() {
277            continue;
278        }
279        if found.replace(at).is_some() {
280            return Err(Error::AmbiguousEndRecords);
281        }
282    }
283    found.ok_or(Error::MissingEndRecord)
284}

286/// Read the totals the end records declare, using the ZIP64 pair when one is present.
287fn end_record_totals(bytes: &[u8], eocd_at: usize) -> Result<EndRecord, Error> {
288    let disk = u16_at(bytes, eocd_at + 4).ok_or(Error::Invalid("truncated end record"))?;
289    let directory_disk = u16_at(bytes, eocd_at + 6).ok_or(Error::Invalid("truncated end record"))?;
290    let disk_entries = u16_at(bytes, eocd_at + 8).ok_or(Error::Invalid("truncated end record"))?;
291    let entries = u16_at(bytes, eocd_at + 10).ok_or(Error::Invalid("truncated end record"))?;
292    let directory_size = u32_at(bytes, eocd_at + 12).ok_or(Error::Invalid("truncated end record"))?;
293    let directory_offset = u32_at(bytes, eocd_at + 16).ok_or(Error::Invalid("truncated end record"))?;
294
295    let sentinel = disk == u16::MAX
296        || directory_disk == u16::MAX
297        || disk_entries == u16::MAX
298        || entries == u16::MAX
299        || directory_size == u32::MAX
300        || directory_offset == u32::MAX;
301    let locator_at = eocd_at.checked_sub(ZIP64_LOCATOR_LEN).filter(|at| signature_at(bytes, *at, ZIP64_LOCATOR_SIGNATURE));
302
303    let Some(locator_at) = locator_at else {
304        if sentinel {
305            return Err(Error::Invalid("ZIP64 sentinel without a ZIP64 end record"));
306        }
307        if disk != 0 || directory_disk != 0 || disk_entries != entries {
308            return Err(Error::Invalid("multi-disk ZIP archive"));
309        }
310        return Ok(EndRecord {
311            entries: u64::from(entries),
312            size: u64::from(directory_size),
313            offset: u64::from(directory_offset),
314            structure_at: eocd_at,
315        });
316    };
317
318    let zip64_disk = u32_at(bytes, locator_at + 4).ok_or(Error::Invalid("truncated ZIP64 locator"))?;
319    let zip64_offset = u64_at(bytes, locator_at + 8).ok_or(Error::Invalid("truncated ZIP64 locator"))?;
320    let zip64_disks = u32_at(bytes, locator_at + 16).ok_or(Error::Invalid("truncated ZIP64 locator"))?;
321    if zip64_disk != 0 || zip64_disks != 1 {
322        return Err(Error::Invalid("multi-disk ZIP64 archive"));
323    }
324    let zip64_at = usize::try_from(zip64_offset).map_err(|_| Error::Invalid("ZIP64 end-record offset does not fit this platform"))?;
325    if !signature_at(bytes, zip64_at, ZIP64_EOCD_SIGNATURE) {
326        return Err(Error::Invalid("the ZIP64 end record is missing at its declared offset"));
327    }
328    let zip64_disk_number = u32_at(bytes, zip64_at + 16).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
329    let zip64_directory_disk = u32_at(bytes, zip64_at + 20).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
330    let zip64_disk_entries = u64_at(bytes, zip64_at + 24).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
331    let zip64_entries = u64_at(bytes, zip64_at + 32).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
332    let zip64_directory_size = u64_at(bytes, zip64_at + 40).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
333    let zip64_directory_offset = u64_at(bytes, zip64_at + 48).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
334    let zip64_record_size = u64_at(bytes, zip64_at + 4).ok_or(Error::Invalid("truncated ZIP64 end record"))?;
335    if zip64_record_size < u64::try_from(ZIP64_EOCD_FIXED_LEN).unwrap_or(44) {
336        return Err(Error::Invalid("the ZIP64 end record is too short"));
337    }
338    // The ZIP64 record ends exactly where its locator begins; anything else means the locator
339    // points somewhere other than this record.
340    let record_end = u64::try_from(zip64_at)
341        .ok()
342        .and_then(|at| at.checked_add(ZIP64_EOCD_HEADER_LEN))
343        .and_then(|at| at.checked_add(zip64_record_size))
344        .ok_or(Error::Invalid("ZIP64 end-record length overflow"))?;
345    if record_end != u64::try_from(locator_at).unwrap_or(u64::MAX) {
346        return Err(Error::Invalid("the ZIP64 end record does not end at its locator"));
347    }
348    if zip64_disk_number != 0 || zip64_directory_disk != 0 || zip64_disk_entries != zip64_entries {
349        return Err(Error::Invalid("multi-disk ZIP64 archive"));
350    }
351    if (disk != u16::MAX && disk != 0)
352        || (directory_disk != u16::MAX && directory_disk != 0)
353        || (disk_entries != u16::MAX && u64::from(disk_entries) != zip64_entries)
354        || (entries != u16::MAX && u64::from(entries) != zip64_entries)
355        || (directory_size != u32::MAX && u64::from(directory_size) != zip64_directory_size)
356        || (directory_offset != u32::MAX && u64::from(directory_offset) != zip64_directory_offset)
357    {
358        return Err(Error::Invalid("the classic and ZIP64 end records disagree"));
359    }
360    Ok(EndRecord { entries: zip64_entries, size: zip64_directory_size, offset: zip64_directory_offset, structure_at: zip64_at })
361}

363/// Walk the declared directory, checking each entry header and that the entries fill it exactly.
364fn walk_directory(bytes: &[u8], start: usize, end: usize, entries: u64, max_name_bytes: u16) -> Result<(), Error> {
365    if end > bytes.len() || start > end {
366        return Err(Error::Invalid("the central directory is outside the archive"));
367    }
368    let mut at = start;
369    for _ in 0..entries {
370        if !signature_at(bytes, at, CENTRAL_FILE_SIGNATURE) {
371            return Err(Error::Invalid("a central-directory entry signature is missing"));
372        }
373        let name = u16_at(bytes, at + NAME_LEN_AT).ok_or(Error::Invalid("truncated central-directory entry"))?;
374        let extra = usize::from(u16_at(bytes, at + EXTRA_LEN_AT).ok_or(Error::Invalid("truncated central-directory entry"))?);
375        let comment = usize::from(u16_at(bytes, at + COMMENT_LEN_AT).ok_or(Error::Invalid("truncated central-directory entry"))?);
376        let disk = u16_at(bytes, at + ENTRY_DISK_AT).ok_or(Error::Invalid("truncated central-directory entry"))?;
377        if disk != 0 {
378            return Err(Error::Invalid("a central-directory entry refers to another disk"));
379        }
380        if name == 0 || name > max_name_bytes {
381            return Err(Error::NameTooLong { actual: u64::from(name), limit: u64::from(max_name_bytes) });
382        }
383        at = at
384            .checked_add(CENTRAL_FILE_LEN)
385            .and_then(|value| value.checked_add(usize::from(name)))
386            .and_then(|value| value.checked_add(extra))
387            .and_then(|value| value.checked_add(comment))
388            .ok_or(Error::Invalid("central-directory entry length overflow"))?;
389        if at > end {
390            return Err(Error::Invalid("a central-directory entry exceeds the declared directory size"));
391        }
392    }
393    // The directory normally ends where its last entry does, optionally followed by one
394    // central-directory digital signature record.
395    if at < end && signature_at(bytes, at, CENTRAL_DIGITAL_SIGNATURE) {
396        let signature_len = usize::from(u16_at(bytes, at + 4).ok_or(Error::Invalid("truncated central-directory signature"))?);
397        at = at
398            .checked_add(6)
399            .and_then(|value| value.checked_add(signature_len))
400            .ok_or(Error::Invalid("central-directory signature length overflow"))?;
401    }
402    if at != end {
403        return Err(Error::Invalid("the central-directory entry count and size disagree"));
404    }
405    Ok(())
406}

408fn bytes_at(bytes: &[u8], at: usize, len: usize) -> Option<&[u8]> {
409    bytes.get(at..at.checked_add(len)?)
410}
411
412fn u16_at(bytes: &[u8], at: usize) -> Option<u16> {
413    Some(u16::from_le_bytes(bytes_at(bytes, at, 2)?.try_into().ok()?))
414}
415
416fn u32_at(bytes: &[u8], at: usize) -> Option<u32> {
417    Some(u32::from_le_bytes(bytes_at(bytes, at, 4)?.try_into().ok()?))
418}
419
420fn u64_at(bytes: &[u8], at: usize) -> Option<u64> {
421    Some(u64::from_le_bytes(bytes_at(bytes, at, 8)?.try_into().ok()?))
422}
423
424fn signature_at(bytes: &[u8], at: usize, signature: &[u8; 4]) -> bool {
425    bytes_at(bytes, at, signature.len()) == Some(signature.as_slice())
426}
