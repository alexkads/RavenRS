1//! A ZIP archive being written.
2
3use std::io::{Cursor, Write};
4
5use crate::craft::zip::Error;
6
7/// How an entry is stored.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum Compression {
10    /// Uncompressed: a `mimetype` entry that must be readable in place, or data that is
11    /// compressed already.
12    Stored,
13    /// Deflate at level 6.
14    Deflated,
15    /// Deflate at the given level (0 to 9; higher is clamped).
16    Level(u8),
17}
18
19/// A ZIP archive being written into memory. The output is deterministic: the same (1980)
20/// timestamp on every entry. Entries are written in the order they are added.
21pub struct Writer {
22    zip: zip::ZipWriter<Cursor<Vec<u8>>>,
23}
24
25impl Default for Writer {
26    fn default() -> Writer {
27        Writer { zip: zip::ZipWriter::new(Cursor::new(Vec::new())) }
28    }
29}
30
31impl Writer {
32    pub fn new() -> Writer {
33        Writer::default()
34    }
35
36    /// Adds an entry. A name the archive already has is an error.
37    pub fn add(&mut self, name: &str, data: &[u8], compression: Compression) -> Result<(), Error> {
38        let (method, level) = match compression {
39            Compression::Stored => (zip::CompressionMethod::Stored, None),
40            Compression::Deflated => (zip::CompressionMethod::Deflated, Some(6)),
41            Compression::Level(l) => (zip::CompressionMethod::Deflated, Some(i64::from(l.min(9)))),
42        };
43        let options = zip::write::SimpleFileOptions::default()
44            .compression_method(method)
45            .compression_level(level)
46            .last_modified_time(zip::DateTime::default())
47            // Deflate can grow incompressible data a little, so ZIP64 well before 4 GiB.
48            .large_file(data.len() as u64 >= u64::from(u32::MAX) / 2);
49        let fail = |e: &dyn std::fmt::Display| Error::Write(format!("{name}: {e}"));
50        self.zip.start_file(name, options).map_err(|e| fail(&e))?;
51        self.zip.write_all(data).map_err(|e| fail(&e))
52    }
53
54    /// The finished archive.
55    pub fn finish(self) -> Result<Vec<u8>, Error> {
56        self.zip.finish().map(Cursor::into_inner).map_err(|e| Error::Write(e.to_string()))
57    }
58}
