1//! An archive open for reading: entry-count guard, duplicate names, budgeted entry reads.
2
3use std::borrow::Cow;
4use std::io::{Cursor, Read};
5
6use crate::craft::zip::preflight::{ReadError, read_entry};
7use crate::craft::zip::{Budget, Error, check_entry_count};
8
9/// What an [`Archive`] refuses to exceed. Both numbers are the caller's.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub struct Limits {
12    /// Most entries the archive may declare.
13    pub max_entries: u64,
14    /// Most bytes all the entries read from the archive may expand to together.
15    pub max_total_bytes: u64,
16}
17
18/// A ZIP archive in memory. Nothing is decompressed until an entry is read.
19///
20/// A clone shares the parsed directory and the bytes, and carries a copy of the budget: a
21/// caller that reads through a clone keeps its own count.
22#[derive(Clone)]
23pub struct Archive<'a> {
24    zip: zip::ZipArchive<Cursor<&'a [u8]>>,
25    budget: Budget,
26    duplicates: bool,
27}
28
29const CENTRAL_HEADER: &[u8; 4] = b"PK\x01\x02";
30const CENTRAL_HEADER_LEN: usize = 46;
31
32/// The central directory headers that follow one another from `start`, counted up to `stop`.
33fn count_central_headers(bytes: &[u8], start: u64, stop: usize) -> usize {
34    let u16_at = |at: usize| bytes.get(at..at.checked_add(2)?).map(|b| usize::from(u16::from_le_bytes([b[0], b[1]])));
35    let Ok(mut at) = usize::try_from(start) else { return 0 };
36    let mut count = 0;
37    while count < stop && bytes.get(at..).is_some_and(|b| b.starts_with(CENTRAL_HEADER)) {
38        let (Some(name), Some(extra), Some(comment)) = (u16_at(at + 28), u16_at(at + 30), u16_at(at + 32)) else { break };
39        count += 1;
40        at = at.saturating_add(CENTRAL_HEADER_LEN + name + extra + comment);
41    }
42    count
43}
44
45impl<'a> Archive<'a> {
46    /// Indexes `bytes`. The declared entry count is checked against
47    /// [`Limits::max_entries`] before `zip` parses the central directory, and again after.
48    pub fn open(bytes: &'a [u8], limits: Limits) -> Result<Self, Error> {
49        check_entry_count(bytes, limits.max_entries)?;
50        let zip = zip::ZipArchive::new(Cursor::new(bytes)).map_err(|e| Error::NotZip(e.to_string()))?;
51        let found = zip.len() as u64;
52        if found > limits.max_entries {
53            return Err(Error::TooManyEntries { declared: found, limit: limits.max_entries });
54        }
55        // `zip` keeps one entry per name, so a directory with more headers than `zip` has entries
56        // repeats a name.
57        let duplicates = count_central_headers(bytes, zip.central_directory_start(), zip.len().saturating_add(1)) > zip.len();
58        Ok(Archive { zip, budget: Budget::new("archive", limits.max_total_bytes), duplicates })
59    }
60
61    /// The number of distinct entry names.
62    pub fn len(&self) -> usize {
63        self.zip.len()
64    }
65
66    pub fn is_empty(&self) -> bool {
67        self.zip.is_empty()
68    }
69
70    /// Do two entries share a name? `zip` then keeps one of them, and which one a writer meant
71    /// is not known: a format that must not guess refuses the archive.
72    pub fn has_duplicate_names(&self) -> bool {
73        self.duplicates
74    }
75
76    /// The entry names in the archive's order, folders included. A name that is not text is
77    /// left out.
78    pub fn names(&self) -> impl Iterator<Item = Cow<'_, str>> {
79        self.zip.file_names().filter_map(Result::ok)
80    }
81
82    /// Is there an entry `name` (exactly as stored)?
83    pub fn contains(&self, name: &str) -> bool {
84        self.zip.index_for_name(name).is_some()
85    }
86
87    /// The size entry `name` declares. It is a claim, not a measurement.
88    pub fn declared_size(&mut self, name: &str) -> Option<u64> {
89        let index = self.zip.index_for_name(name)?;
90        self.zip.by_index_raw(index).ok().map(|f| f.size())
91    }
92
93    /// The size entry `name` takes in the archive, compressed. With [`Archive::declared_size`] it
94    /// gives the compression ratio the entry claims.
95    pub fn compressed_size(&mut self, name: &str) -> Option<u64> {
96        let index = self.zip.index_for_name(name)?;
97        self.zip.by_index_raw(index).ok().map(|f| f.compressed_size())
98    }
99
100    /// The first `n` bytes of entry `name` (all of it when it is shorter), or `None` when there
101    /// is no such entry: for telling a file by its signature without inflating it. The bytes are
102    /// charged to the total; a total with less left than the bytes wanted refuses before reading.
103    ///
104    /// The entry's checksum is not checked: it covers the whole entry.
105    pub fn read_head(&mut self, name: &str, n: usize) -> Result<Option<Vec<u8>>, Error> {
106        let Some(index) = self.zip.index_for_name(name) else { return Ok(None) };
107        let entry = |msg: String| Error::Entry { name: name.to_string(), msg };
108        let f = self.zip.by_index(index).map_err(|e| entry(e.to_string()))?;
109        // Never more than the entry declares: a shorter entry costs only what it has.
110        let want = (n as u64).min(f.size());
111        if self.budget.remaining() < want {
112            return Err(Error::BudgetSpent { name: name.to_string(), limit: self.budget.limit() });
113        }
114        let mut out = Vec::with_capacity(n.min(64 * 1024));
115        f.take(want).read_to_end(&mut out).map_err(|e| entry(e.to_string()))?;
116        self.budget.charge(out.len() as u64).map_err(|e| Error::BudgetSpent { name: name.to_string(), limit: e.limit })?;
117        Ok(Some(out))
118    }
119
120    /// The bytes of entry `name`, or `None` when there is no such entry.
121    ///
122    /// The entry may declare, and expand to, at most `max` bytes, and at most what is left of
123    /// [`Limits::max_total_bytes`]. An entry that expands past the size it declares is refused.
124    /// Only the bytes of a read that succeeds are charged to the total.
125    pub fn read(&mut self, name: &str, max: u64) -> Result<Option<Vec<u8>>, Error> {
126        let Some(index) = self.zip.index_for_name(name) else { return Ok(None) };
127        let entry = |msg: String| Error::Entry { name: name.to_string(), msg };
128        let f = self.zip.by_index(index).map_err(|e| entry(e.to_string()))?;
129        let declared = f.size();
130        match read_entry(f, declared, max, &mut self.budget) {
131            Ok(data) => Ok(Some(data)),
132            Err(ReadError::DeclaredTooLarge { limit, .. }) | Err(ReadError::Overrun { limit }) => {
133                Err(Error::EntryTooLarge { name: name.to_string(), limit })
134            }
135            Err(ReadError::Budget(e)) => Err(Error::BudgetSpent { name: name.to_string(), limit: e.limit }),
136            Err(ReadError::Io(e)) => Err(entry(e.to_string())),
137        }
138    }
139
140    /// The total budget: what the entries read so far have used, and what is left.
141    pub fn budget(&self) -> &Budget {
142        &self.budget
143    }
144
145    /// The total budget, for a caller with its own charging rule (charging a failed read, say).
146    pub fn budget_mut(&mut self) -> &mut Budget {
147        &mut self.budget
148    }
149}
