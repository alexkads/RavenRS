3//! ZIP archives from untrusted files, read within limits and written.
4//!
5//! The `zip` crate does the decoding. This crate is what every reader of a ZIP-based format
6//! otherwise writes again around it:
7//!
8//! - [`Archive::open`] refuses an archive that declares more entries than the caller allows
9//!   before `zip` parses them ([`check_entry_count`]), and tells whether two entries share a name.
10//! - [`Archive::read`] reads an entry within a per-entry limit and the archive's total
11//!   [`Budget`], counting the bytes that really arrive, not the size the archive declares.
12//! - [`Writer`] writes stored and deflated entries, ZIP64 where an entry needs it, with the same
13//!   timestamp on every entry so equal input gives equal bytes.
14//! - [`preflight`] is a stricter check of the whole end-record and central-directory structure
15//!   on the raw bytes, for a caller that wants nothing ambiguous to reach a ZIP reader at all.
16//!
17//! Every limit is the caller's number. Entry-name rules (which names are safe, which spellings
18//! are one part) stay with the format that knows them.
19

35/// The payload of a ZIP's first entry when it is a stored file named `mimetype` of at most 256
36/// bytes: how OpenDocument, EPUB, IDML, OpenRaster and Krita mark their archives so the type is
37/// readable at a fixed offset without unzipping.
38pub fn stored_mimetype(bytes: &[u8]) -> Option<&[u8]> {
39    let u16_at = |o: usize| bytes.get(o..o + 2).and_then(|s| <[u8; 2]>::try_from(s).ok()).map(u16::from_le_bytes);
40    let u32_at = |o: usize| bytes.get(o..o + 4).and_then(|s| <[u8; 4]>::try_from(s).ok()).map(u32::from_le_bytes);
41    if u32_at(0)? != 0x0403_4b50 || u16_at(8)? != 0 {
42        return None;
43    }
44    let size = usize::try_from(u32_at(18)?).ok()?;
45    let name_len = usize::from(u16_at(26)?);
46    let extra_len = usize::from(u16_at(28)?);
47    if bytes.get(30..30 + name_len)? != b"mimetype" || size > 256 {
48        return None;
49    }
50    let start = 30 + name_len + extra_len;
51    bytes.get(start..start.checked_add(size)?)
52}

54/// Why an archive was not read or not written.
55#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
56pub enum Error {
57    /// The bytes are not a ZIP archive the `zip` crate can index.
58    #[error("not a ZIP archive: {0}")]
59    NotZip(String),
60    /// The archive declares more entries than [`Limits::max_entries`].
61    #[error("the archive declares {declared} entries, more than {limit}")]
62    TooManyEntries { declared: u64, limit: u64 },
63    /// More end of central directory records than an archive can have.
64    #[error("more than {0} end of central directory records")]
65    TooManyEndRecords(usize),
66    /// An entry declares, or expands to, more than the limit given for it.
67    #[error("{name} is larger than {limit} bytes")]
68    EntryTooLarge { name: String, limit: u64 },
69    /// The entries read so far have used up [`Limits::max_total_bytes`].
70    #[error("{name} does not fit in what is left of the {limit} bytes the archive may expand to")]
71    BudgetSpent { name: String, limit: u64 },
72    /// An entry could not be decoded (a failed checksum, an unknown method, encryption).
73    #[error("{name}: {msg}")]
74    Entry { name: String, msg: String },
75    /// The archive could not be written.
76    #[error("the ZIP archive could not be written: {0}")]
77    Write(String),
78}
79
80impl Error {
81    /// Was a limit exceeded (as opposed to the archive being broken)?
82    pub fn is_limit(&self) -> bool {
83        matches!(self, Error::TooManyEntries { .. } | Error::TooManyEndRecords(_) | Error::EntryTooLarge { .. } | Error::BudgetSpent { .. })
84    }
85}

89/// A named limit and the total charged against it so far.
90///
91/// The name is what a breach message should mention ("document", "part", "images"), and the limit
92/// is the application's own number. The crate never invents either, so one budget can cover a
93/// whole import and be reported in the application's terms.
94#[derive(Clone, Copy, Debug, PartialEq, Eq)]
95pub struct Budget {
96    what: &'static str,
97    limit: u64,
98    used: u64,
99}
100
101impl Budget {
102    /// A budget named `what` that admits up to `limit` units in total.
103    pub const fn new(what: &'static str, limit: u64) -> Self {
104        Self { what, limit, used: 0 }
105    }
106
107    /// Charge `n` units, or refuse without changing the total.
108    ///
109    /// A refused charge leaves the total untouched, so a caller may carry on with a smaller
110    /// charge, and the amount refused is reported in [`Exceeded::requested`].
111    pub fn charge(&mut self, n: u64) -> Result<(), Exceeded> {
112        if n > self.remaining() {
113            return Err(Exceeded { what: self.what, limit: self.limit, requested: n });
114        }
115        self.used = self.used.saturating_add(n);
116        Ok(())
117    }
118
119    /// The units still available; zero when the budget is exhausted.
120    pub const fn remaining(&self) -> u64 {
121        self.limit.saturating_sub(self.used)
122    }
123
124    /// The units charged so far.
125    pub const fn used(&self) -> u64 {
126        self.used
127    }
128
129    /// The limit this budget was created with.
130    pub const fn limit(&self) -> u64 {
131        self.limit
132    }
133
134    /// The name this budget was created with.
135    pub const fn what(&self) -> &'static str {
136        self.what
137    }
138}
139
140/// A charge that a [`Budget`] refused: what was limited, the limit, and the amount asked for.
141#[derive(Clone, Copy, Debug, PartialEq, Eq)]
142pub struct Exceeded {
143    /// The budget's name, so a message can say which limit was hit.
144    pub what: &'static str,
145    /// The limit that was already reached.
146    pub limit: u64,
147    /// The amount that did not fit.
148    pub requested: u64,
149}
150
151impl fmt::Display for Exceeded {
152    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
153        write!(f, "the {} limit is {}; another {} was requested", self.what, self.limit, self.requested)
154    }
155}

159#[cfg(test)]
160mod tests {
161    use super::*;
162
163    #[test]
164    fn budgets_charge_and_report_the_remainder() {
165        let mut budget = Budget::new("document", 10);
166        assert_eq!(budget.what(), "document");
167        assert_eq!(budget.limit(), 10);
168        assert_eq!(budget.remaining(), 10);
169        assert_eq!(budget.used(), 0);
170        assert_eq!(budget.charge(4), Ok(()));
171        assert_eq!(budget.used(), 4);
172        assert_eq!(budget.remaining(), 6);
173        assert_eq!(budget.charge(6), Ok(()));
174        assert_eq!(budget.remaining(), 0);
175    }
176
177    #[test]
178    fn a_refused_charge_leaves_the_total_untouched() {
179        let mut budget = Budget::new("part", 100);
180        assert_eq!(budget.charge(60), Ok(()));
181        let exceeded = budget.charge(41).err();
182        assert_eq!(exceeded, Some(Exceeded { what: "part", limit: 100, requested: 41 }));
183        assert_eq!(budget.used(), 60);
184        // The refused amount is not charged, so the budget stays usable.
185        assert_eq!(budget.charge(40), Ok(()));
186        assert_eq!(budget.used(), 100);
187        assert!(budget.charge(1).is_err());
188    }
189
190    #[test]
191    fn a_zero_budget_admits_nothing_and_the_message_names_the_limit() {
192        let mut budget = Budget::new("entries", 0);
193        assert_eq!(budget.remaining(), 0);
194        assert_eq!(budget.charge(0), Ok(()));
195        let message = budget.charge(1).err().map(|e| e.to_string());
196        assert_eq!(message.as_deref(), Some("the entries limit is 0; another 1 was requested"));
197    }
198
199    #[test]
200    fn a_limit_at_the_maximum_does_not_overflow() {
201        let mut budget = Budget::new("total", u64::MAX);
202        assert_eq!(budget.charge(u64::MAX), Ok(()));
203        assert_eq!(budget.remaining(), 0);
204        assert!(budget.charge(u64::MAX).is_err());
205    }
