3/// A part's relationships with full type URIs. `Ok(vec![])` when it has none; an error when the
4/// rels part is damaged or too large (its part then can't be kept faithfully).
5fn raw_rels(cx: &mut Ctx<'_>, part: &str) -> Result<Vec<PreservedRel>, String> {
6    let name = rels_part_name(part);
7    let bytes = match cx.pkg.read_limited(&name, MAX_META) {
8        Ok(Some(b)) => b,
9        Ok(None) => return Ok(vec![]),
10        Err(e) => return Err(e.to_string()),
11    };
12    let root = xml::parse(&bytes).map_err(|e| format!("{name}: {e}"))?;
13    let (dir, _) = package::split_dir(part);
14    let mut out = vec![];
15    for r in root.kids("Relationship") {
16        if out.len() >= MAX_RELS {
17            return Err(format!("{name}: too many relationships"));
18        }
19        let (Some(id), Some(target)) = (r.attr("Id"), r.attr("Target")) else { continue };
20        let external = r.attr("TargetMode").is_some_and(|m| m.eq_ignore_ascii_case("External"));
21        out.push(PreservedRel {
22            id: id.to_string(),
23            rel_type: r.attr("Type").unwrap_or("").to_string(),
24            target: if external { target.to_string() } else { package::resolve(dir, target) },
25            external,
26        });
27    }
28    Ok(out)
29}

33/// The `<sheet>` entries of the workbook that are chart sheets, each with the number of worksheets
34/// (model sheets) before it.
35fn chart_sheet_entries<'a>(wb_xml: &'a xml::El, wb_rels: &[PreservedRel], file_to_model: &[Option<usize>]) -> Vec<(usize, &'a xml::El)> {
36    let mut out = vec![];
37    let mut worksheets = 0usize;
38    for (i, se) in wb_xml.child("sheets").into_iter().flat_map(|s| s.kids("sheet")).enumerate() {
39        if file_to_model.get(i).copied().flatten().is_some() {
40            worksheets += 1;
41        } else if se.attr("id").and_then(|id| wb_rels.iter().find(|r| r.id == id)).is_some_and(|r| is_kind(r, &["chartsheet"])) {
42            out.push((worksheets, se));
43        }
44    }
45    out
46}

48/// Collects kept content while a package is read.
49pub struct KeptReader {
50    on: bool,
51    strict: bool,
52    defaults: HashMap<String, String>,
53    overrides: HashMap<String, String>,
54    budget: usize,
55    parts_left: usize,
56}

58impl KeptReader {
59    pub fn new(cx: &mut Ctx<'_>) -> KeptReader {
60        let mut k = KeptReader {
61            on: true,
62            strict: false,
63            defaults: HashMap::new(),
64            overrides: HashMap::new(),
65            budget: kept_total(),
66            parts_left: MAX_KEPT_PARTS,
67        };
68        if let Ok(root) = raw_rels(cx, "")
69            && root.iter().any(|r| r.rel_type.starts_with("http://purl.oclc.org/"))
70        {
71            k.on = false;
72            k.strict = true;
73        }
74        match cx.pkg.read_limited("[Content_Types].xml", MAX_META).ok().flatten().and_then(|b| xml::parse(&b).ok()) {
75            Some(types) => {
76                for d in types.kids("Default") {
77                    if let (Some(e), Some(t)) = (d.attr("Extension"), d.attr("ContentType")) {
78                        k.defaults.insert(e.to_ascii_lowercase(), t.to_string());
79                    }
80                }
81                for o in types.kids("Override") {
82                    if let (Some(p), Some(t)) = (o.attr("PartName"), o.attr("ContentType")) {
83                        k.overrides.insert(normalize(p), t.to_string());
84                    }
85                }
86            }
87            None => k.on = false,
88        }
89        k
90    }

92    fn content_type(&self, part: &str) -> Option<String> {
93        let key = normalize(part);
94        if let Some(t) = self.overrides.get(&key) {
95            return Some(t.clone());
96        }
97        let (_, file) = package::split_dir(&key);
98        let ext = file.rsplit_once('.').map(|(_, e)| e)?;
99        self.defaults.get(ext).cloned()
100    }

102    /// The parts reachable from `root` that aren't in `seen` yet. Fails (and keeps nothing) when the
103    /// graph reaches rewritten content, a missing or damaged part, or a size limit.
104    fn closure(&mut self, cx: &mut Ctx<'_>, root: &PreservedRel, seen: &HashSet<String>, through: &[&str]) -> Result<Vec<PreservedPart>, String> {
105        if root.external {
106            return Ok(vec![]);
107        }
108        let blocked = |r: &PreservedRel| rewritten(r) && !through.iter().any(|k| k.eq_ignore_ascii_case(kind(&r.rel_type)));
109        if blocked(root) {
110            return Err(format!("it refers to {}", kind(&root.rel_type)));
111        }
112        let mut parts = vec![];
113        let mut bytes = 0usize;
114        let mut local: HashSet<String> = HashSet::new();
115        let mut queue = VecDeque::from([(root.target.clone(), 0usize)]);
116        while let Some((name, depth)) = queue.pop_front() {
117            let key = normalize(&name);
118            if seen.contains(&key) || !local.insert(key) {
119                continue;
120            }
121            if depth > MAX_CHAIN {
122                return Err("its relationships are nested too deeply".into());
123            }
124            if parts.len() >= self.parts_left {
125                return Err("it has too many parts".into());
126            }
127            let content_type = self.content_type(&name).ok_or_else(|| format!("{name} has no content type"))?;
128            let data = match cx.pkg.read_limited(&name, MAX_KEPT_PART) {
129                Ok(Some(d)) => d,
130                Ok(None) => return Err(format!("{name} is missing")),
131                Err(e) => return Err(e.to_string()),
132            };
133            bytes = bytes.saturating_add(data.len());
134            if bytes > self.budget {
135                return Err("it is too large".into());
136            }
137            let rels = raw_rels(cx, &name)?;
138            for r in rels.iter().filter(|r| !r.external) {
139                if blocked(r) {
140                    return Err(format!("it refers to {}", kind(&r.rel_type)));
141                }
142                queue.push_back((r.target.clone(), depth + 1));
143            }
144            parts.push(PreservedPart { name, content_type, data: Arc::from(data), rels });
145        }
146        self.budget = self.budget.saturating_sub(bytes);
147        self.parts_left = self.parts_left.saturating_sub(parts.len());
148        Ok(parts)
149    }

151    /// Keeps a set of relationships (all or nothing) and the parts behind them.
152    fn keep(&mut self, cx: &mut Ctx<'_>, p: &mut Preserved, roots: &[PreservedRel], seen: &mut HashSet<String>) -> Result<(), String> {
153        self.keep_through(cx, p, roots, seen, &[])
154    }

156    /// [`Self::keep`] for roots whose graph may reach the rewritten kinds in `through`.
157    fn keep_through(
158        &mut self,
159        cx: &mut Ctx<'_>,
160        p: &mut Preserved,
161        roots: &[PreservedRel],
162        seen: &mut HashSet<String>,
163        through: &[&str],
164    ) -> Result<(), String> {
165        let mut new_parts: Vec<PreservedPart> = vec![];
166        let mut scratch = seen.clone();
167        for r in roots {
168            let parts = self.closure(cx, r, &scratch, through)?;
169            scratch.extend(parts.iter().map(|q| normalize(&q.name)));
170            new_parts.extend(parts);
171        }
172        *seen = scratch;
173        p.parts.extend(new_parts);
174        for r in roots {
175            if !p.rels.iter().any(|x| x.id == r.id) {
176                p.rels.push(r.clone());
177            }
178        }
179        Ok(())
180    }

182    /// Package-root and workbook-level content.
183    pub fn workbook(&mut self, cx: &mut Ctx<'_>, wb_part: &str, wb_xml: &xml::El, file_to_model: &[Option<usize>]) -> Option<Arc<Preserved>> {
184        if !self.on {
185            let wb_rels = raw_rels(cx, wb_part).unwrap_or_default();
186            for (_, se) in chart_sheet_entries(wb_xml, &wb_rels, file_to_model) {
187                cx.warn(format!("chart sheet \"{}\" was skipped and is not kept on save", se.attr("name").unwrap_or("")));
188            }
189            if self.strict {
190                cx.warn("unmodelled content of a Strict Open XML workbook (custom XML, links, controls…) is not kept on save");
191            }
192            return None;
193        }
194        let mut p = Preserved::default();
195        let mut seen = HashSet::new();
196        // Package root: custom document properties, ribbon customisation, task panes.
197        if let Ok(root) = raw_rels(cx, "") {
198            for r in root.iter().filter(|r| is_kind(r, PACKAGE_ROOTS)) {
199                let mut q = Preserved::default();
200                match self.keep(cx, &mut q, std::slice::from_ref(r), &mut seen) {
201                    Ok(()) => {
202                        p.parts.extend(q.parts);
203                        p.package_rels.extend(q.rels);
204                    }
205                    Err(why) => cx.warn(format!("{} was dropped: {why}", kind(&r.rel_type))),
206                }
207            }
208        }
209        let wb_rels = raw_rels(cx, wb_part).unwrap_or_default();
210        for r in wb_rels.iter().filter(|r| is_kind(r, WORKBOOK_ROOTS)) {
211            if let Err(why) = self.keep(cx, &mut p, std::slice::from_ref(r), &mut seen) {
212                cx.warn(format!("{} part {} was dropped: {why}", kind(&r.rel_type), r.target));
213            }
214        }
215        // Chart sheets: the sheet entry (name, state, position) plus the chart sheet part and the
216        // drawing, chart, style and colour parts behind it.
217        for (worksheets, se) in chart_sheet_entries(wb_xml, &wb_rels, file_to_model) {
218            let name = se.attr("name").unwrap_or("").to_string();
219            let Some(r) = se.attr("id").and_then(|id| wb_rels.iter().find(|r| r.id == id)) else { continue };
220            match self.keep_through(cx, &mut p, std::slice::from_ref(r), &mut seen, CHART_SHEET_GRAPH) {
221                Ok(()) => p.sheets.push(PreservedSheet {
222                    name,
223                    visibility: match se.attr("state") {
224                        Some("hidden") => Visibility::Hidden,
225                        Some("veryHidden") => Visibility::VeryHidden,
226                        _ => Visibility::Visible,
227                    },
228                    rel_id: r.id.clone(),
229                    before: worksheets,
230                    sheet_id: se.attr_u32("sheetId").unwrap_or(0),
231                }),
232                Err(why) => cx.warn(format!("chart sheet \"{name}\" was dropped: {why}")),
233            }
234        }
235        // Links to external workbooks: `[1]` in formulas is the position in <externalReferences>,
236        // so they are kept all together or not at all.
237        let ids: Vec<&str> =
238            wb_xml.child("externalReferences").map(|e| e.kids("externalReference").filter_map(|x| x.attr("id")).collect()).unwrap_or_default();
239        if !ids.is_empty() {
240            let roots: Option<Vec<PreservedRel>> =
241                ids.iter().map(|id| wb_rels.iter().find(|r| r.id == *id && is_kind(r, &["externalLink"])).cloned()).collect();
242            let kept = match roots {
243                Some(roots) => self.keep(cx, &mut p, &roots, &mut seen),
244                None => Err("a link has no part".into()),
245            };
246            match kept {
247                Ok(()) => {
248                    let mut x = format!("<externalReferences xmlns:r=\"{NS_REL}\">");
249                    for id in &ids {
250                        let _ = write!(x, "<externalReference r:id=\"{}\"/>", esc_attr(id));
251                    }
252                    x.push_str("</externalReferences>");
253                    p.elements.push(PreservedXml { slot: "externalReferences".into(), xml: x });
254                    cx.warn("links to external workbooks were kept; they are not followed or refreshed");
255                }
256                Err(why) => cx.warn(format!("links to external workbooks were dropped: {why}")),
257            }
258        }
259        // Excel trusts the saved cell values only when both are present; without them it
260        // recalculates everything on open, and links into closed workbooks come back as errors.
261        if let Some(v) = wb_xml.child("fileVersion") {
262            let mut x = String::from("<fileVersion");
263            for k in FILE_VERSION_ATTRS {
264                if let Some(a) = v.attr(k) {
265                    let _ = write!(x, " {k}=\"{}\"", esc_attr(a));
266                }
267            }
268            x.push_str("/>");
269            p.elements.push(PreservedXml { slot: "fileVersion".into(), xml: x });
270        }
271        let calc_features = wb_xml
272            .child("extLst")
273            .and_then(|l| l.kids("ext").find(|e| e.attr("uri").is_some_and(|u| u.eq_ignore_ascii_case(CALC_FEATURES_URI))))
274            .and_then(|e| e.child("calcFeatures"));
275        if let Some(c) = calc_features {
276            let mut x = format!("<ext uri=\"{CALC_FEATURES_URI}\" xmlns:xcalcf=\"{NS_CALC_FEATURES}\"><xcalcf:calcFeatures>");
277            for name in c.kids("feature").filter_map(|f| f.attr("name")) {
278                let _ = write!(x, "<xcalcf:feature name=\"{}\"/>", esc_attr(name));
279            }
280            x.push_str("</xcalcf:calcFeatures></ext>");
281            p.elements.push(PreservedXml { slot: "ext".into(), xml: x });
282        }
283        (!p.is_empty()).then(|| Arc::new(p))
284    }

286    /// Sheet-level content of the sheet part `part`. `controls` says which form controls the
287    /// sheet reader modelled (their VML shapes are left out of the kept content) and whether the
288    /// VML drawing holds others that only it lists.
289    pub fn sheet(&mut self, cx: &mut Ctx<'_>, part: &str, name: &str, controls: &crate::xlsx::controls::ReadControls) -> Option<Arc<Preserved>> {
290        if !self.on {
291            return None;
292        }
293        let rels = match raw_rels(cx, part) {
294            Ok(r) => r,
295            Err(e) => {
296                cx.warn(format!("sheet \"{name}\": relationships could not be read ({e}); unmodelled content was dropped"));
297                return None;
298            }
299        };
300        let mut p = Preserved::default();
301        let mut seen = HashSet::new();
302        for r in rels.iter().filter(|r| is_kind(r, SHEET_ROOTS)) {
303            if let Err(why) = self.keep(cx, &mut p, std::slice::from_ref(r), &mut seen) {
304                cx.warn(format!("sheet \"{name}\": {} was dropped: {why}", kind(&r.rel_type)));
305            }
306        }
307        // Notes have one VML drawing per sheet; another one holds the header and footer pictures.
308        let vml_drawings = rels.iter().filter(|r| is_kind(r, &["vmlDrawing"])).count();
309        let notes = rels.iter().filter(|r| is_kind(r, &["comments"])).count();
310        if rels.iter().any(|r| is_kind(r, SHEET_ELEMENT_KINDS)) || vml_drawings > notes || controls.vml_only_unmodelled {
311            self.sheet_elements(cx, part, name, &rels, &mut p, &mut seen, controls);
312        }
313        (!p.is_empty()).then(|| Arc::new(p))
314    }

316    #[allow(clippy::too_many_arguments)]
317    fn sheet_elements(
318        &mut self,
319        cx: &mut Ctx<'_>,
320        part: &str,
321        name: &str,
322        rels: &[PreservedRel],
323        p: &mut Preserved,
324        seen: &mut HashSet<String>,
325        controls: &crate::xlsx::controls::ReadControls,
326    ) {
327        let read_controls = &controls.modelled;
328        let scan = match cx.pkg.read(part) {
329            Ok(Some(bytes)) => scan_sheet(&bytes),
330            _ => None,
331        };
332        let Some(scan) = scan else {
333            cx.warn(format!("sheet \"{name}\": controls and embedded objects could not be read and were dropped"));
334            return;
335        };
336        // Printer settings (a binary part per sheet) hang off <pageSetup r:id>.
337        if let Some(id) = scan.page_setup.as_deref()
338            && let Some(r) = rels.iter().find(|r| r.id == id && is_kind(r, &["printerSettings"]))
339        {
340            match self.keep(cx, p, std::slice::from_ref(r), seen) {
341                Ok(()) => p.elements.push(PreservedXml { slot: "pageSetup".into(), xml: r.id.clone() }),
342                Err(why) => cx.warn(format!("sheet \"{name}\": printer settings were dropped: {why}")),
343            }
344        }
345        // Header and footer pictures (`&G`): a VML drawing of their own and the images it shows,
346        // kept byte for byte (GridCraft writes the header text, not the pictures).
347        if let Some(id) = scan.legacy_drawing_hf.as_deref()
348            && scan.legacy_drawing.as_deref() != Some(id)
349            && let Some(r) = rels.iter().find(|r| r.id == id && is_kind(r, &["vmlDrawing"]))
350        {
351            match self.keep_through(cx, p, std::slice::from_ref(r), seen, &["vmlDrawing"]) {
352                Ok(()) => {
353                    let xml = format!("<legacyDrawingHF xmlns:r=\"{NS_REL}\" r:id=\"{}\"/>", esc_attr(&r.id));
354                    p.elements.push(PreservedXml { slot: "legacyDrawingHF".into(), xml });
355                    cx.warn(format!("sheet \"{name}\": pictures in the header and footer are not shown; they are kept for saving"));
356                }
357                Err(why) => cx.warn(format!("sheet \"{name}\": header and footer pictures were dropped: {why}")),
358            }
359        }
360        // Form controls the sheet reader modelled are written from the model; the others stay.
361        let mut frags: Vec<(String, String)> = scan
362            .fragments
363            .into_iter()
364            .filter_map(|(slot, x)| if slot == "controls" { without_controls(&x, read_controls).map(|x| (slot, x)) } else { Some((slot, x)) })
365            .collect();
366        // Controls and OLE objects are drawn from shapes in the legacy VML drawing. Controls only
367        // the drawing lists need it too, whether or not a `<controls>` element is left.
368        if frags.iter().any(|(slot, _)| slot != "picture") || controls.vml_only_unmodelled {
369            match self.vml(cx, rels, scan.legacy_drawing.as_deref(), seen, read_controls) {
370                Ok((vml, parts)) => {
371                    p.parts.extend(parts);
372                    p.vml = Some(vml);
373                }
374                Err(why) => {
375                    cx.warn(format!("sheet \"{name}\": controls and embedded objects were dropped: {why}"));
376                    frags.retain(|(slot, _)| slot == "picture");
377                }
378            }
379        }
380        for (slot, x) in frags {
381            let mut ids = vec![];
382            let ok = map_rel_ids(&x, &mut |id| {
383                ids.push(id.to_string());
384                Some(id.to_string())
385            })
386            .is_some();
387            let roots: Option<Vec<PreservedRel>> = ids.iter().map(|id| rels.iter().find(|r| r.id == *id).cloned()).collect();
388            let kept = match (ok, roots) {
389                (true, Some(roots)) => self.keep(cx, p, &roots, seen),
390                (false, _) => Err("its XML is damaged".into()),
391                (_, None) => Err("it refers to a missing relationship".into()),
392            };
393            match kept {
394                Ok(()) => p.elements.push(PreservedXml { slot, xml: x }),
395                Err(why) => cx.warn(format!("sheet \"{name}\": {slot} were dropped: {why}")),
396            }
397        }
398        if p.elements.iter().any(|e| e.slot == "oleObjects" || e.slot == "controls") || (controls.vml_only_unmodelled && p.vml.is_some()) {
399            cx.warn(format!("sheet \"{name}\": embedded objects and form controls are not shown; they are kept for saving"));
400        } else {
401            p.vml = None;
402        }
403    }

405    /// The sheet's VML drawing without its notes and the form controls read into the model
406    /// (GridCraft writes those), and the parts its shapes refer to (preview images).
407    fn vml(
408        &mut self,
409        cx: &mut Ctx<'_>,
410        rels: &[PreservedRel],
411        rid: Option<&str>,
412        seen: &mut HashSet<String>,
413        read_controls: &HashSet<u64>,
414    ) -> Result<(PreservedPart, Vec<PreservedPart>), String> {
415        let rel = rid.and_then(|id| rels.iter().find(|r| r.id == id && !r.external)).ok_or("the sheet has no VML drawing")?;
416        // Charged to the kept total like any other part: many sheets can point at one highly
417        // compressed VML drawing, and each keeps its own copy.
418        if self.budget == 0 {
419            return Err("it is too large".into());
420        }
421        let limit = MAX_META.min(u64::try_from(self.budget).unwrap_or(u64::MAX));
422        let data = match cx.pkg.read_limited(&rel.target, limit) {
423            Ok(Some(d)) => d,
424            Ok(None) => return Err(format!("{} is missing", rel.target)),
425            Err(IoError::TooLarge(_)) => return Err("it is too large".into()),
426            Err(e) => return Err(e.to_string()),
427        };
428        self.budget = self.budget.saturating_sub(data.len());
429        let text = String::from_utf8(data).map_err(|_| "the VML drawing is not UTF-8".to_string())?;
430        let text = strip_shapes(&text, read_controls);
431        let vrels = raw_rels(cx, &rel.target)?;
432        let mut q = Preserved::default();
433        let roots: Vec<PreservedRel> = vrels.iter().filter(|r| !r.external).cloned().collect();
434        self.keep(cx, &mut q, &roots, seen)?;
435        let vml = PreservedPart { name: rel.target.clone(), content_type: CT_VML.into(), data: Arc::from(text.into_bytes()), rels: vrels };
436        Ok((vml, q.parts))
437    }

451/// Finds the sheet's `<picture>`, `<oleObjects>`, `<controls>` (also inside
452/// `<mc:AlternateContent>`) and the `<legacyDrawing>` and `<legacyDrawingHF>` ids. `None` when the XML is damaged.
453fn scan_sheet(bytes: &[u8]) -> Option<SheetScan> {
454    let data = xml::to_utf8(bytes);
455    let text = std::str::from_utf8(&data).ok()?;
456    let mut reader = Reader::from_str(text);
457    let mut decls: Vec<(String, String)> = vec![];
458    let mut out = SheetScan { fragments: vec![], legacy_drawing: None, legacy_drawing_hf: None, page_setup: None };
459    let mut in_root = false;
460    loop {
461        let start = usize::try_from(reader.buffer_position()).ok()?;
462        match reader.read_event().ok()? {
463            Event::Eof => break,
464            Event::Start(e) if !in_root => {
465                in_root = true;
466                for a in e.attributes().with_checks(false).flatten() {
467                    if let Some(p) = a.key.as_ref().strip_prefix(b"xmlns:")
468                        && let (Ok(p), Ok(v)) = (std::str::from_utf8(p), std::str::from_utf8(&a.value))
469                        && !v.contains('"')
470                    {
471                        decls.push((p.to_string(), v.to_string()));
472                    }
473                }
474            }
475            Event::Start(e) => {
476                let local = e.local_name().as_ref().to_vec();
477                match local.as_slice() {
478                    b"pageSetup" => out.page_setup = rel_id_attr(&e),
479                    b"legacyDrawing" => out.legacy_drawing = rel_id_attr(&e),
480                    b"legacyDrawingHF" => out.legacy_drawing_hf = rel_id_attr(&e),
481                    _ => {}
482                }
483                reader.read_to_end(e.name()).ok()?;
484                let end = usize::try_from(reader.buffer_position()).ok()?;
485                let wanted = matches!(local.as_slice(), b"picture" | b"oleObjects" | b"controls" | b"AlternateContent");
486                if wanted && end.saturating_sub(start) <= MAX_FRAGMENT {
487                    let frag = text.get(start..end)?;
488                    push_fragment(&mut out, frag, &local, &decls);
489                }
490            }
491            Event::Empty(e) if in_root => {
492                let local = e.local_name().as_ref().to_vec();
493                if local == b"legacyDrawing" || local == b"legacyDrawingHF" || local == b"pageSetup" {
494                    let id = rel_id_attr(&e);
495                    match local.as_slice() {
496                        b"pageSetup" => out.page_setup = id,
497                        b"legacyDrawing" => out.legacy_drawing = id,
498                        _ => out.legacy_drawing_hf = id,
499                    }
500                } else if local == b"picture"
501                    && let Some(frag) = text.get(start..usize::try_from(reader.buffer_position()).ok()?)
502                {
503                    push_fragment(&mut out, frag, &local, &decls);
504                }
505            }
506            Event::End(_) => break,
507            _ => {}
508        }
509    }
510    Some(out)
511}

513/// The `r:id` of an element (an `id` attribute with a namespace prefix).
514fn rel_id_attr(e: &BytesStart<'_>) -> Option<String> {
515    e.attributes()
516        .with_checks(false)
517        .flatten()
518        .find(|a| a.key.local_name().as_ref() == b"id" && a.key.prefix().is_some())
519        .and_then(|a| String::from_utf8(a.value.to_vec()).ok())
520}

522fn push_fragment(out: &mut SheetScan, frag: &str, local: &[u8], decls: &[(String, String)]) {
523    let frag = with_decls(frag, decls);
524    let slot = if local == b"AlternateContent" {
525        // The kept element is the one the first choice holds.
526        let Ok(el) = xml::parse(frag.as_bytes()) else { return };
527        match el.child("Choice").and_then(|c| c.children.first()) {
528            Some(c) => c.name.clone(),
529            None => return,
530        }
531    } else {
532        String::from_utf8_lossy(local).into_owned()
533    };
534    if SHEET_SLOTS.contains(&slot.as_str()) && !out.fragments.iter().any(|(s, _)| *s == slot) {
535        out.fragments.push((slot, frag));
536    }
537}

539/// Adds the root's namespace declarations a fragment doesn't declare itself.
540fn with_decls(frag: &str, decls: &[(String, String)]) -> String {
541    let Some(name_end) = frag.char_indices().skip(1).find(|(_, c)| c.is_whitespace() || *c == '/' || *c == '>').map(|(i, _)| i) else {
542        return frag.to_string();
543    };
544    let head = frag.get(..frag.find('>').unwrap_or(frag.len())).unwrap_or("");
545    let mut add = String::new();
546    for (p, v) in decls {
547        if !head.contains(&format!("xmlns:{p}=")) {
548            let _ = write!(add, " xmlns:{p}=\"{v}\"");
549        }
550    }
551    format!("{}{add}{}", frag.get(..name_end).unwrap_or(""), frag.get(name_end..).unwrap_or(""))
552}

554/// Byte ranges of the `<v:shape …>…</v:shape>` (or self-closed) elements in VML text.
555fn shape_spans(vml: &str) -> Vec<(usize, usize)> {
556    let mut spans = vec![];
557    let mut from = 0;
558    while let Some(i) = vml.get(from..).and_then(|s| s.find("<v:shape")).map(|i| i + from) {
559        let after = i + "<v:shape".len();
560        let is_shape = vml.get(after..).and_then(|s| s.chars().next()).is_some_and(|c| c.is_whitespace() || c == '>' || c == '/');
561        if !is_shape {
562            from = after;
563            continue;
564        }
565        let Some(tag_end) = vml.get(i..).and_then(|s| s.find('>')).map(|j| i + j + 1) else { break };
566        let end = if vml.get(..tag_end).is_some_and(|t| t.ends_with("/>")) {
567            tag_end
568        } else {
569            match vml.get(tag_end..).and_then(|s| s.find("</v:shape>")) {
570                Some(j) => tag_end + j + "</v:shape>".len(),
571                None => break,
572            }
573        };
574        spans.push((i, end));
575        from = end;
576    }
577    spans
578}

580/// VML without its note shapes and the shapes numbered in `drop`, each judged as the control
581/// reader judges it, from the shape's own tags and never from text elsewhere in it: a note by its
582/// client data's `ObjectType` ([`crate::xlsx::controls::is_note_data`]), its number from its `id`, else
583/// its `o:spid` ([`crate::xlsx::controls::number_of`]). Each tag is read on its own ([`xml::first_start`]),
584/// so a caption's unclosed `<br>` doesn't hide the client data after it. A shape with no number
585/// that isn't a note is kept.
586fn strip_shapes(vml: &str, drop: &HashSet<u64>) -> String {
587    let mut out = String::with_capacity(vml.len());
588    let mut last = 0;
589    for (a, b) in shape_spans(vml) {
590        let shape = vml.get(a..b).unwrap_or("").as_bytes();
591        let note = xml::first_start(shape, "ClientData").is_some_and(|d| crate::xlsx::controls::is_note_data(&d));
592        let numbered = xml::first_start(shape, "shape").and_then(|s| crate::xlsx::controls::number_of(&s)).is_some_and(|n| drop.contains(&n));
593        if note || numbered {
594            out.push_str(vml.get(last..a).unwrap_or(""));
595            last = b;
596        }
597    }
598    out.push_str(vml.get(last..).unwrap_or(""));
599    out
600}

602/// Shape numbers (`_x0000_s1025` → 1025) used in VML text.
603fn shape_numbers(vml: &str) -> Vec<u64> {
604    vml.match_indices("_x0000_s")
605        .filter_map(|(i, m)| {
606            let digits: String = vml.get(i + m.len()..)?.chars().take_while(char::is_ascii_digit).take(18).collect();
607            digits.parse().ok()
608        })
609        .collect()
610}

612/// The highest shape block VML text claims in `<o:idmap data>` or uses in a shape id.
613pub(super) fn vml_block_max(vml: &str) -> u64 {
614    let used = shape_numbers(vml).into_iter().max().unwrap_or(0) / 1024;
615    let claimed = vml
616        .find("<o:idmap")
617        .and_then(|i| vml.get(i..)?.find("data=\"").map(|d| i + d + "data=\"".len()))
618        .and_then(|d| Some((d, d + vml.get(d..)?.find('"')?)))
619        .and_then(|(d, e)| vml.get(d..e))
620        .map(|list| list.split(',').filter_map(|b| b.trim().parse::<u64>().ok()).max().unwrap_or(0))
621        .unwrap_or(0);
622    used.max(claimed)
623}

625/// The first shape block free in kept VML: GridCraft's own shapes (notes, then form controls) are
626/// numbered from `block * 1024 + 1` when they join it ([`merge_vml`]).
627pub(super) fn first_free_block(kept: &str) -> u64 {
628    shape_numbers(kept).into_iter().max().unwrap_or(0) / 1024 + 1
629}

631/// Kept VML (controls, OLE objects) plus GridCraft's note and form control shapes: ours get ids
632/// in fresh shape blocks so they can't collide with the kept shapes the sheet XML refers to.
633pub(super) fn merge_vml(kept: &str, ours: &str) -> String {
634    let Some(close) = kept.rfind("</xml>") else { return ours.to_string() };
635    let block = first_free_block(kept);
636    let mut add = String::new();
637    // Our shape types (notes, form controls) that the kept drawing doesn't define.
638    let mut from = 0;
639    while let Some(i) = ours.get(from..).and_then(|s| s.find("<v:shapetype")).map(|i| i + from) {
640        let Some(j) = ours.get(i..).and_then(|s| s.find("</v:shapetype>")).map(|j| i + j + "</v:shapetype>".len()) else { break };
641        let ty = ours.get(i..j).unwrap_or("");
642        let id = attr_value(ty, "id").unwrap_or("");
643        if !id.is_empty() && !kept.contains(&format!("id=\"{id}\"")) && !kept.contains(&format!("id='{id}'")) {
644            add.push_str(ty);
645        }
646        from = j;
647    }
648    let spans = shape_spans(ours);
649    let blocks = (spans.len() as u64 + 1).div_ceil(1024).max(1);
650    for (k, (a, b)) in spans.into_iter().enumerate() {
651        let shape = ours.get(a..b).unwrap_or("");
652        let id = block.saturating_mul(1024).saturating_add(1 + k as u64);
653        match (shape.find("id=\"_x0000_s"), shape.find("id=\"_x0000_s").and_then(|i| shape.get(i + 4..)?.find('"').map(|j| i + 4 + j))) {
654            (Some(i), Some(j)) => {
655                let _ = write!(add, "{}id=\"_x0000_s{id}\"{}", shape.get(..i).unwrap_or(""), shape.get(j + 1..).unwrap_or(""));
656            }
657            _ => add.push_str(shape),
658        }
659    }
660    let mut head = kept.get(..close).unwrap_or("").to_string();
661    // Claim the blocks in the drawing's id map.
662    if let Some(i) = head.find("<o:idmap")
663        && let Some(d) = head.get(i..).and_then(|s| s.find("data=\"")).map(|d| i + d + "data=\"".len())
664        && let Some(e) = head.get(d..).and_then(|s| s.find('"')).map(|e| d + e)
665    {
666        let claimed: String = (block..block.saturating_add(blocks)).map(|b| format!(",{b}")).collect();
667        head.insert_str(e, &claimed);
668    }
669    format!("{head}{add}{}", kept.get(close..).unwrap_or(""))
670}

672/// The value of the first attribute `name` in XML text `x`, in either quote style (`name="v"` or
673/// `name='v'`), when it has one. The name must start a token (`id` is not the end of `spid`).
674fn attr_value<'a>(x: &'a str, name: &str) -> Option<&'a str> {
675    let mut from = 0;
676    while let Some(i) = x.get(from..)?.find(name).map(|i| i + from) {
677        from = i + name.len();
678        let starts_token = x.get(..i).is_some_and(|before| before.ends_with(|c: char| c.is_whitespace() || c == '<'));
679        let Some(rest) = x.get(from..).map(str::trim_start).and_then(|r| r.strip_prefix('=')).map(str::trim_start) else { continue };
680        if !starts_token {
681            continue;
682        }
683        if let Some(q) = rest.chars().next().filter(|q| *q == '"' || *q == '\'') {
684            let body = rest.get(q.len_utf8()..)?;
685            return body.find(q).and_then(|e| body.get(..e));
686        }
687    }
688    None
689}

691/// A kept `<controls>` fragment without the controls whose `shapeId` is in `drop` (they are
692/// written from the model): a bare `<control>` child goes when it is one of them, and within an
693/// `mc:AlternateContent` child only those controls go, in each of its branches, the wrapper
694/// staying while a control remains. `None` when nothing is left. A fragment that doesn't parse
695/// is returned unchanged.
696fn without_controls(frag: &str, drop: &HashSet<u64>) -> Option<String> {
697    if drop.is_empty() {
698        return Some(frag.to_string());
699    }
700    let mut reader = Reader::from_str(frag);
701    let mut out = String::with_capacity(frag.len());
702    let (mut last, mut left, mut inside) = (0usize, 0usize, false);
703    loop {
704        let Ok(start) = usize::try_from(reader.buffer_position()) else { return Some(frag.to_string()) };
705        let ev = match reader.read_event() {
706            Ok(ev) => ev,
707            Err(_) => return Some(frag.to_string()),
708        };
709        let child = match ev {
710            Event::Eof => break,
711            Event::Start(e) if e.local_name().as_ref() == b"controls" => {
712                inside = true;
713                continue;
714            }
715            Event::End(e) if e.local_name().as_ref() == b"controls" => {
716                inside = false;
717                continue;
718            }
719            Event::Start(e) if inside => {
720                if reader.read_to_end(e.name()).is_err() {
721                    return Some(frag.to_string());
722                }
723                true
724            }
725            Event::Empty(_) if inside => true,
726            _ => false,
727        };
728        if !child {
729            continue;
730        }
731        let Ok(end) = usize::try_from(reader.buffer_position()) else { return Some(frag.to_string()) };
732        let text = frag.get(start..end).unwrap_or("");
733        let Some((kept, controls, dropped)) = controls_kept(text, drop) else { return Some(frag.to_string()) };
734        if dropped == 0 {
735            // Nothing of ours in it: the child stays as it is.
736            left += 1;
737        } else {
738            out.push_str(frag.get(last..start).unwrap_or(""));
739            if controls > 0 {
740                out.push_str(&kept);
741                left += 1;
742            }
743            last = end;
744        }
745    }
746    out.push_str(frag.get(last..).unwrap_or(""));
747    (left > 0).then_some(out)
748}

750/// One child of a `<controls>` element with the `<control>` elements numbered in `drop` taken
751/// out, wherever they are in it (a bare control, or the Choice and Fallback branches of an
752/// `mc:AlternateContent`), as (the text, the controls that remain, the controls taken out). The
753/// `shapeId` is read as the control reader reads it (an attribute, in either quote style). `None`
754/// when the child doesn't parse.
755fn controls_kept(child: &str, drop: &HashSet<u64>) -> Option<(String, usize, usize)> {
756    let mut reader = Reader::from_str(child);
757    let mut out = String::with_capacity(child.len());
758    let (mut last, mut kept, mut dropped) = (0usize, 0usize, 0usize);
759    loop {
760        let start = usize::try_from(reader.buffer_position()).ok()?;
761        let (e, has_end) = match reader.read_event().ok()? {
762            Event::Eof => break,
763            Event::Start(e) => (e, true),
764            Event::Empty(e) => (e, false),
765            _ => continue,
766        };
767        if e.local_name().as_ref() != b"control" {
768            // A wrapper or a branch: its controls are the events that follow.
769            continue;
770        }
771        let number = e
772            .try_get_attribute("shapeId")
773            .ok()
774            .flatten()
775            .and_then(|a| std::str::from_utf8(&a.value).ok().and_then(crate::xlsx::controls::shape_number));
776        if has_end {
777            reader.read_to_end(e.name()).ok()?;
778        }
779        let end = usize::try_from(reader.buffer_position()).ok()?;
780        if number.is_some_and(|n| drop.contains(&n)) {
781            out.push_str(child.get(last..start)?);
782            last = end;
783            dropped += 1;
784        } else {
785            kept += 1;
786        }
787    }
788    out.push_str(child.get(last..)?);
789    Some((out, kept, dropped))
790}

792/// Rewrites relationship-id attributes (`r:id`, `r:embed`…: any attribute in the relationships
793/// namespace) of an XML fragment through `map`. `None` when the fragment is damaged or `map` has
794/// no id for one of them.
795pub fn map_rel_ids(x: &str, map: &mut dyn FnMut(&str) -> Option<String>) -> Option<String> {
796    let mut reader = Reader::from_str(x);
797    let mut writer = Writer::new(Vec::with_capacity(x.len()));
798    let mut prefixes: HashSet<Vec<u8>> = HashSet::new();
799    let mut depth = 0usize;
800    loop {
801        match reader.read_event().ok()? {
802            Event::Eof => break,
803            Event::Start(e) => {
804                depth += 1;
805                if depth > MAX_FRAGMENT_DEPTH {
806                    return None;
807                }
808                let e = map_start(&e, &mut prefixes, map)?;
809                writer.write_event(Event::Start(e)).ok()?;
810            }
811            Event::Empty(e) => {
812                let e = map_start(&e, &mut prefixes, map)?;
813                writer.write_event(Event::Empty(e)).ok()?;
814            }
815            Event::End(e) => {
816                depth = depth.saturating_sub(1);
817                writer.write_event(Event::End(e)).ok()?;
818            }
819            ev => writer.write_event(ev).ok()?,
820        }
821    }
822    String::from_utf8(writer.into_inner()).ok()
823}

825fn map_start(e: &BytesStart<'_>, prefixes: &mut HashSet<Vec<u8>>, map: &mut dyn FnMut(&str) -> Option<String>) -> Option<BytesStart<'static>> {
826    for a in e.attributes().with_checks(false) {
827        let a = a.ok()?;
828        if let Some(p) = a.key.as_ref().strip_prefix(b"xmlns:")
829            && (a.value.as_ref() == NS_REL.as_bytes() || a.value.as_ref() == NS_REL_STRICT.as_bytes())
830        {
831            prefixes.insert(p.to_vec());
832        }
833    }
834    let name = String::from_utf8(e.name().as_ref().to_vec()).ok()?;
835    let mut out = BytesStart::new(name);
836    for a in e.attributes().with_checks(false) {
837        let a = a.ok()?;
838        let key = a.key.as_ref();
839        let is_rel = !key.starts_with(b"xmlns") && a.key.prefix().is_some_and(|p| prefixes.contains(p.as_ref()));
840        if is_rel {
841            let old = std::str::from_utf8(&a.value).ok()?;
842            let new = map(old)?;
843            out.push_attribute(Attribute { key: QName(key), value: Cow::Owned(esc_attr(&new).into_bytes()) });
844        } else {
845            out.push_attribute(Attribute { key: QName(key), value: a.value.clone() });
846        }
847    }
848    Some(out)
849}
