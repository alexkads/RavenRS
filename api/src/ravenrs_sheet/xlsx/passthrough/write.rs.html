6/// Names given to kept parts while a package is written.
7#[derive(Default)]
8pub struct KeptOut {
9    used: HashSet<String>,
10    n: u32,
11    /// The highest VML shape block a kept header and footer drawing uses: notes go after it.
12    vml_blocks: u32,
13    /// Package-root relationships (type, target, external), added when `_rels/.rels` is written.
14    root: Vec<(String, String, bool)>,
15}
16
17impl KeptOut {
18    /// Reserves a part name GridCraft writes itself, so kept parts don't take it. False when it is
19    /// taken.
20    pub fn reserve(&mut self, name: &str) -> bool {
21        self.used.insert(name.to_ascii_lowercase())
22    }
23
24    /// Reserves the VML shape blocks the kept header and footer drawings of `wb` use, so that the
25    /// notes drawings GridCraft writes don't reuse their shape ids.
26    pub fn reserve_vml_blocks(&mut self, wb: &Workbook) {
27        let vmls =
28            wb.sheets.iter().filter_map(|s| s.preserved.as_deref()).flat_map(|p| &p.parts).filter(|q| q.content_type.eq_ignore_ascii_case(CT_VML));
29        let max = vmls.map(|q| vml_block_max(&String::from_utf8_lossy(&q.data))).max().unwrap_or(0);
30        self.vml_blocks = u32::try_from(max).unwrap_or(u32::MAX);
31    }
32
33    /// The shape block of the `n`th notes drawing.
34    pub fn notes_block(&self, n: u32) -> u32 {
35        n.saturating_add(self.vml_blocks)
36    }
37
38    /// A part name for a kept part: its original name when that can't collide with anything
39    /// written, otherwise a fresh one in the same folder.
40    pub(super) fn claim(&mut self, original: &str) -> String {
41        let lower = original.to_ascii_lowercase();
42        let own = OWN_DIRS.iter().any(|d| lower.starts_with(d));
43        if own && self.used.insert(lower) {
44            return original.to_string();
45        }
46        let (dir, file) = package::split_dir(original);
47        let (stem, ext) = match file.rsplit_once('.') {
48            Some((s, e)) => (s, format!(".{e}")),
49            None => (file, String::new()),
50        };
51        let stem = if own { format!("{stem}_") } else { "kept".to_string() };
52        loop {
53            self.n = self.n.saturating_add(1);
54            let file = format!("{stem}{}{ext}", self.n);
55            let cand = if dir.is_empty() { file } else { format!("{dir}/{file}") };
56            if self.used.insert(cand.to_ascii_lowercase()) || self.n == u32::MAX {
57                return cand;
58            }
59        }
60    }
61}

66fn rel_target(r: &PreservedRel, names: &NameMap) -> Option<String> {
67    if r.external { Some(r.target.clone()) } else { names.get(&normalize(&r.target)).map(|n| format!("/{n}")) }

70/// A kept part's relationships, ids unchanged, targets renamed.
71pub(super) fn part_rels_xml(rels: &[PreservedRel], names: &NameMap) -> String {
72    let mut s = format!("{XML_DECL}<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">");
73    for r in rels {
74        let Some(target) = rel_target(r, names) else { continue };
75        let _ = write!(
76            s,
77            "<Relationship Id=\"{}\" Type=\"{}\" Target=\"{}\"{}/>",
78            esc_attr(&r.id),
79            esc_attr(&r.rel_type),
80            esc_attr(&target),
81            if r.external { " TargetMode=\"External\"" } else { "" }
82        );
83    }
84    s.push_str("</Relationships>");
85    s
86}

88pub(super) fn write_parts(out: &mut Out, parts: &[PreservedPart]) -> NameMap {
89    let mut names = NameMap::new();
90    for p in parts {
91        names.entry(normalize(&p.name)).or_insert_with(|| out.kept.claim(&p.name));
92    }
93    let mut written = HashSet::new();
94    for p in parts {
95        let Some(new) = names.get(&normalize(&p.name)).cloned() else { continue };
96        if !written.insert(new.clone()) {
97            continue;
98        }
99        out.part(&new, Some(&p.content_type), p.data.to_vec());
100        if !p.rels.is_empty() {
101            out.part(&rels_part_name(&new), None, part_rels_xml(&p.rels, &names).into_bytes());
102        }
103    }
104    names
105}

107/// Adds the owner's relationships; returns old id → new id.
108pub(super) fn attach(rels: &mut Rels, list: &[PreservedRel], names: &NameMap) -> HashMap<String, String> {
109    let mut ids = HashMap::new();
110    for r in list {
111        if let Some(target) = rel_target(r, names) {
112            let id = rels.add_raw(&r.rel_type, &target, r.external);
113            ids.insert(r.id.clone(), id);
114        }
115    }
116    ids
117}

119pub(super) fn remap(x: &str, ids: &HashMap<String, String>) -> Option<String> {
120    map_rel_ids(x, &mut |id| ids.get(id).cloned())

132/// The `sheetId`s of the written sheets: each keeps the one the file gave it, unless another
133/// sheet took it first (a copied sheet) or it has none (a new sheet), and those get the next id
134/// that no sheet asks for.
135pub struct SheetIds {
136    wanted: HashSet<u32>,
137    given: HashSet<u32>,
138    next: u32,
139}
140
141impl SheetIds {
142    pub fn new(wb: &Workbook, kept: &WorkbookKept) -> SheetIds {
143        let wanted: HashSet<u32> = wb.sheets.iter().map(|s| s.file_id).chain(kept.sheets.iter().map(|s| s.sheet_id)).filter(|&id| id != 0).collect();
144        SheetIds { wanted, given: HashSet::new(), next: 1 }
145    }
146
147    pub fn take(&mut self, wanted: u32) -> u32 {
148        if wanted != 0 && self.given.insert(wanted) {
149            return wanted;
150        }
151        while self.wanted.contains(&self.next) || self.given.contains(&self.next) {
152            self.next += 1;
153        }
154        self.given.insert(self.next);
155        self.next
156    }
157}

167impl WorkbookKept {
168    /// Where the model's sheet `i` is in the file's sheet list: chart sheets come between worksheets.
169    pub fn file_index(&self, i: usize) -> usize {
170        i + self.sheets.iter().filter(|s| s.before <= i).count()
171    }
172
173    /// The `<sheet>` entries of the chart sheets that go right before the model's sheet `i`
174    /// (`i` = the number of sheets for those that come last), with their ids from `ids`.
175    pub fn sheet_entries(&self, i: usize, last: bool, ids: &mut SheetIds) -> String {
176        let mut x = String::new();
177        for s in self.sheets.iter().filter(|s| if last { s.before >= i } else { s.before == i }) {
178            let state = match s.visibility {
179                Visibility::Visible => "",
180                Visibility::Hidden => " state=\"hidden\"",
181                Visibility::VeryHidden => " state=\"veryHidden\"",
182            };
183            let name: String = s.name.chars().take(31).collect();
184            let id = ids.take(s.sheet_id);
185            let _ = write!(x, "<sheet name=\"{}\" sheetId=\"{id}\"{state} r:id=\"{}\"/>", esc_attr(&name), esc_attr(&s.rid));
186        }
187        x
188    }
189}

191/// The part names reachable from `root` through kept parts' relationships (normalized).
192fn reachable(p: &Preserved, root: &PreservedRel) -> HashSet<String> {
193    let by_name: HashMap<String, &PreservedPart> = p.parts.iter().map(|q| (normalize(&q.name), q)).collect();
194    let mut seen = HashSet::new();
195    let mut queue = VecDeque::from([normalize(&root.target)]);
196    while let Some(n) = queue.pop_front() {
197        if !seen.insert(n.clone()) {
198            continue;
199        }
200        if let Some(q) = by_name.get(&n) {
201            queue.extend(q.rels.iter().filter(|r| !r.external).map(|r| normalize(&r.target)));
202        }
203    }
204    seen
205}

207/// Every formula (`<c:f>`) of a chart part, depth first.
208fn chart_formulas(el: &xml::El, out: &mut Vec<String>) {
209    if el.name == "f" && !el.text.trim().is_empty() {
210        out.push(el.text.trim().to_string());
211    }
212    for c in &el.children {
213        chart_formulas(c, out);
214    }
215}

217/// The sheet names a formula such as `'My Sheet'!$A$1:$A$9,Data!$B$1` refers to. References to
218/// other workbooks (`[1]Sheet!A1`) are not sheets of this one and are left out.
219pub(crate) fn sheet_names_in(formula: &str) -> Vec<String> {
220    let mut names = vec![];
221    let chars: Vec<char> = formula.chars().collect();
222    let at = |i: usize| chars.get(i).copied();
223    let mut i = 0;
224    while let Some(c) = at(i) {
225        let mut name = String::new();
226        let start = i;
227        if c == '\'' {
228            i += 1;
229            while let Some(c) = at(i) {
230                if c == '\'' {
231                    if at(i + 1) == Some('\'') {
232                        name.push('\'');
233                        i += 2;
234                        continue;
235                    }
236                    break;
237                }
238                name.push(c);
239                i += 1;
240            }
241            i += 1;
242        } else {
243            while let Some(c) = at(i).filter(|c| !matches!(c, '!' | ',' | '(' | ')' | '\'' | ' ')) {
244                name.push(c);
245                i += 1;
246            }
247        }
248        if at(i) == Some('!') {
249            if !name.is_empty() && !name.starts_with('[') {
250                names.push(name);
251            }
252            i += 1;
253        } else if i == start {
254            i += 1;
255        }
256    }
257    names
258}

260/// Whether a kept chart sheet can be written back: its chart's series must still refer to sheets
261/// that exist under the same names (the chart is kept as it was, so a series on a deleted or
262/// renamed sheet would be left pointing at nothing), and no worksheet may have taken its name.
263fn chart_sheet_usable(wb: &Workbook, p: &Preserved, s: &PreservedSheet) -> bool {
264    let same = |a: &str, b: &str| a.chars().take(31).collect::<String>().eq_ignore_ascii_case(&b.chars().take(31).collect::<String>());
265    if wb.sheets.iter().any(|w| same(&w.name, &s.name)) {
266        return false;
267    }
268    let Some(root) = p.rels.iter().find(|r| r.id == s.rel_id) else { return false };
269    let parts = reachable(p, root);
270    let mut formulas = vec![];
271    for q in p.parts.iter().filter(|q| parts.contains(&normalize(&q.name)) && q.content_type.contains("chart")) {
272        if let Ok(x) = xml::parse(&q.data) {
273            chart_formulas(&x, &mut formulas);
274        }
275    }
276    formulas.iter().flat_map(|f| sheet_names_in(f)).all(|n| wb.sheets.iter().any(|w| same(&w.name, &n)))
277}

279/// Writes the workbook's kept parts and relationships. Chart sheets that can't be written back
280/// (see [`chart_sheet_usable`]) are left out with the parts only they reach.
281pub fn write_workbook(wb: &Workbook, wb_rels: &mut Rels, out: &mut Out) -> WorkbookKept {
282    let Some(p) = wb.preserved.as_deref() else { return WorkbookKept::default() };
283    let (live, dropped): (Vec<&PreservedSheet>, Vec<&PreservedSheet>) = p.sheets.iter().partition(|s| chart_sheet_usable(wb, p, s));
284    let root_of = |s: &PreservedSheet| p.rels.iter().find(|r| r.id == s.rel_id);
285    let dropped_ids: HashSet<&str> = dropped.iter().map(|s| s.rel_id.as_str()).collect();
286    // Parts reached only by dropped chart sheets go with them.
287    let mut gone: HashSet<String> = dropped.iter().filter_map(|s| root_of(s)).flat_map(|r| reachable(p, r)).collect();
288    if !gone.is_empty() {
289        let others = p.rels.iter().filter(|r| !dropped_ids.contains(r.id.as_str())).flat_map(|r| reachable(p, r));
290        for n in others {
291            gone.remove(&n);
292        }
293    }
294    let parts: Vec<PreservedPart> = p.parts.iter().filter(|q| !gone.contains(&normalize(&q.name))).cloned().collect();
295    let rels: Vec<PreservedRel> = p.rels.iter().filter(|r| !dropped_ids.contains(r.id.as_str())).cloned().collect();
296    let names = write_parts(out, &parts);
297    let ids = attach(wb_rels, &rels, &names);
298    for r in &p.package_rels {
299        if let Some(target) = rel_target(r, &names) {
300            out.kept.root.push((r.rel_type.clone(), target, r.external));
301        }
302    }
303    WorkbookKept {
304        elements: p.elements.iter().filter(|e| e.slot == "externalReferences").filter_map(|e| remap(&e.xml, &ids)).collect(),
305        sheets: live
306            .iter()
307            .filter_map(|s| {
308                Some(KeptSheet {
309                    before: s.before,
310                    name: s.name.clone(),
311                    visibility: s.visibility,
312                    rid: ids.get(&s.rel_id)?.clone(),
313                    sheet_id: s.sheet_id,
314                })
315            })
316            .collect(),
317    }
318}
