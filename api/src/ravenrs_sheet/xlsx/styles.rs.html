17/// Longest font name kept, in characters (Excel's own limit is 31).
18pub const MAX_FONT_NAME_CHARS: usize = 255;

20/// Styles read from a file.
21#[derive(Debug, Default)]
22pub struct StylesIn {
23    /// Cell formats (`cellXfs`) in file order.
24    pub xfs: Vec<Style>,
25    /// Differential formats (`dxfs`) in file order.
26    pub dxfs: Vec<Style>,
27    pub palette: Vec<u32>,
28    /// Named cell styles (`cellStyles` with their `cellStyleXfs`).
29    pub named: Vec<(String, Style)>,
30    /// Custom number formats (`numFmts`) by id.
31    pub num_fmts: HashMap<u32, String>,
32}

34pub fn read_color(e: &El, palette: &[u32]) -> Color {
35    if e.flag("auto", false) {
36        return Color::Auto;
37    }
38    if let Some(rgb) = e.attr("rgb") {
39        return Color::from_hex(rgb).unwrap_or(Color::Auto);
40    }
41    if let Some(t) = e.attr_u32("theme") {
42        if t > 11 {
43            return Color::Auto;
44        }
45        let tint = e.attr_f64("tint").map(|v| (v * 1000.0).round().clamp(-1000.0, 1000.0) as i16).unwrap_or(0);
46        return Color::Theme(t as u8, tint);
47    }
48    if let Some(i) = e.attr_u32("indexed") {
49        return palette.get(i as usize).map(|&c| Color::Rgb(c)).unwrap_or(Color::Auto);
50    }
51    Color::Auto
52}
53
54fn opt_color(e: Option<&El>, palette: &[u32]) -> Color {
55    e.map(|e| read_color(e, palette)).unwrap_or(Color::Auto)
56}

58/// `<b/>`-style boolean property elements: present means true unless `val` says otherwise.

60    e.child(name).is_some_and(|c| c.flag("val", true))

63const DEFAULT_FONT_FAMILY: u8 = 2;
64const MAX_FONT_FAMILY: u8 = 14;

66pub fn read_font(e: &El, palette: &[u32]) -> Font {
67    let mut f = Font::default();
68    if let Some(n) = e.child("name").or_else(|| e.child("rFont")).and_then(|n| n.attr("val")) {
69        // Every cell format that uses the font copies its name: keep it short.
70        f.name = match n.char_indices().nth(MAX_FONT_NAME_CHARS) {
71            Some((at, _)) => n.get(..at).unwrap_or_default().to_string(),
72            None => n.to_string(),
73        };
74    }
75    if let Some(sz) = e.child("sz").and_then(|s| s.attr_f64("val")) {
76        f.size = sz.clamp(1.0, 409.0) as f32;
77    }
78    f.bold = bool_prop(e, "b");
79    f.italic = bool_prop(e, "i");
80    f.strike = bool_prop(e, "strike");
81    if let Some(u) = e.child("u") {
82        f.underline = match u.attr("val").unwrap_or("single") {
83            "double" => Underline::Double,
84            "singleAccounting" => Underline::SingleAccounting,
85            "doubleAccounting" => Underline::DoubleAccounting,
86            "none" => Underline::None,
87            _ => Underline::Single,
88        };
89    }
90    if let Some(v) = e.child_val("vertAlign") {
91        f.vert = match v {
92            "superscript" => VertAlign::Superscript,
93            "subscript" => VertAlign::Subscript,
94            _ => VertAlign::Baseline,
95        };
96    }
97    f.charset = e.child("charset").and_then(|c| c.attr_u32("val")).and_then(|v| u8::try_from(v).ok());
98    f.family = e
99        .child("family")
100        .and_then(|c| c.attr_u32("val"))
101        .and_then(|v| u8::try_from(v).ok())
102        .filter(|v| *v <= MAX_FONT_FAMILY && *v != DEFAULT_FONT_FAMILY);
103    f.minor = e.child_val("scheme") == Some("minor");
104    f.color = opt_color(e.child("color"), palette);
105    // Text 1 (dk1) is what "automatic" font colour means in practice.
106    if f.color == Color::Theme(1, 0) {
107        f.color = Color::Auto;
108    }
109    f
110}

160/// A `<fill>`. In differential formats a solid fill's colour is the *background* colour and the
161/// pattern type may be omitted.
162pub fn read_fill(e: &El, palette: &[u32], dxf: bool) -> Fill {
163    if let Some(p) = e.child("patternFill") {
164        let fg = opt_color(p.child("fgColor"), palette);
165        let bg = opt_color(p.child("bgColor"), palette);
166        let pt = p.attr("patternType");
167        if dxf && (pt.is_none() || pt == Some("solid")) {
168            let c = if bg != Color::Auto { bg } else { fg };
169            if c == Color::Auto && pt.is_none() {
170                return Fill::default();
171            }
172            return Fill::solid(c);
173        }
174        let pattern = pattern_from(pt.unwrap_or("none"));
175        if pattern == PatternType::None {
176            return Fill::default();
177        }
178        return Fill { pattern, fg, bg };
179    }
180    if let Some(g) = e.child("gradientFill") {
181        // Approximated by the first stop's colour.
182        if let Some(c) = g.kids("stop").next().and_then(|s| s.child("color")) {
183            return Fill::solid(read_color(c, palette));
184        }
185    }
186    Fill::default()
187}

227fn read_line(e: Option<&El>, palette: &[u32]) -> BorderLine {
228    let Some(e) = e else { return BorderLine::default() };
229    let style = border_style(e.attr("style").unwrap_or("none"));
230    if style == BorderStyle::None {
231        return BorderLine::default();
232    }
233    BorderLine { style, color: opt_color(e.child("color"), palette) }
234}

236pub fn read_border(e: &El, palette: &[u32]) -> Borders {
237    let diag = read_line(e.child("diagonal"), palette);
238    Borders {
239        left: read_line(e.child("left").or_else(|| e.child("start")), palette),
240        right: read_line(e.child("right").or_else(|| e.child("end")), palette),
241        top: read_line(e.child("top"), palette),
242        bottom: read_line(e.child("bottom"), palette),
243        diag_down: if e.flag("diagonalDown", false) { diag } else { BorderLine::default() },
244        diag_up: if e.flag("diagonalUp", false) { diag } else { BorderLine::default() },
245    }
246}

248fn read_alignment(a: &El) -> Alignment {
249    let mut al = Alignment {
250        h: match a.attr("horizontal").unwrap_or("general") {
251            "left" => HAlign::Left,
252            "center" => HAlign::Center,
253            "right" => HAlign::Right,
254            "fill" => HAlign::Fill,
255            "justify" => HAlign::Justify,
256            "centerContinuous" => HAlign::CenterAcross,
257            "distributed" => HAlign::Distributed,
258            _ => HAlign::General,
259        },
260        v: match a.attr("vertical").unwrap_or("bottom") {
261            "top" => VAlign::Top,
262            "center" => VAlign::Center,
263            "justify" => VAlign::Justify,
264            "distributed" => VAlign::Distributed,
265            _ => VAlign::Bottom,
266        },
267        wrap: a.flag("wrapText", false),
268        shrink: a.flag("shrinkToFit", false),
269        indent: a.attr_u32("indent").unwrap_or(0).min(250) as u8,
270        rotation: 0,
271    };
272    if let Some(r) = a.attr_i64("textRotation") {
273        al.rotation = match r {
274            0..=90 => r as i16,
275            91..=180 => -((r - 90) as i16),
276            255 => 255,
277            _ => 0,
278        };
279    }
280    al
281}

287/// A number format by id. Custom formats are built once and shared (`NumFmt` is an `Arc`), so
288/// many cell formats naming one long code don't each copy it.
289fn num_fmt_code(id: u32, custom: &HashMap<u32, NumFmt>) -> NumFmt {
290    if let Some(c) = custom.get(&id) {
291        return c.clone();
292    }
293    NumFmt::new(builtin_format(id).unwrap_or("General"))
294}

296fn read_xf(xf: &El, fonts: &[Font], fills: &[Fill], borders: &[Borders], fmts: &HashMap<u32, NumFmt>, checkboxes: &[bool]) -> Style {
297    let mut s = Style::default();
298    if let Some(f) = xf.attr_u32("fontId").and_then(|i| fonts.get(i as usize)) {
299        s.font = f.clone();
300    } else if let Some(f) = fonts.first() {
301        s.font = f.clone();
302    }
303    if let Some(f) = xf.attr_u32("fillId").and_then(|i| fills.get(i as usize)) {
304        s.fill = *f;
305    }
306    if let Some(b) = xf.attr_u32("borderId").and_then(|i| borders.get(i as usize)) {
307        s.borders = *b;
308    }
309    s.num_fmt = num_fmt_code(xf.attr_u32("numFmtId").unwrap_or(0), fmts);
310    // A checkbox cell control (Insert › Checkbox) is a feature-property-bag complement on the xf.
311    if crate::xlsx::feature_bags::xf_is_checkbox(xf, checkboxes) {
312        s.num_fmt = NumFmt::new("checkbox");
313    }
314    if let Some(a) = xf.child("alignment") {
315        s.align = read_alignment(a);
316    }
317    if let Some(p) = xf.child("protection") {
318        s.protection = read_protection(p);
319    }
320    s
321}

323/// `checkboxes`: for each `xfComplement` index, whether it is a checkbox (see
324/// [`crate::xlsx::feature_bags::checkbox_complements`]); empty when the file has no feature property bags.
325/// `minor_font`: the theme's body font, which fonts with `<scheme val="minor"/>` follow.
326pub fn read_styles(root: &El, checkboxes: &[bool], minor_font: &str) -> StylesIn {
327    let mut palette: Vec<u32> = INDEXED.to_vec();
328    if let Some(ix) = root.path(&["colors", "indexedColors"]) {
329        let custom: Vec<u32> = ix
330            .kids("rgbColor")
331            .map(|c| match read_color(c, &INDEXED) {
332                Color::Rgb(v) => v,
333                _ => 0,
334            })
335            .collect();
336        if !custom.is_empty() {
337            palette = custom;
338        }
339    }
340    let mut fmts = HashMap::new();
341    if let Some(n) = root.child("numFmts") {
342        for f in n.kids("numFmt") {
343            if let (Some(id), Some(code)) = (f.attr_u32("numFmtId"), f.attr("formatCode")) {
344                fmts.insert(id, code.to_string());
345            }
346        }
347    }
348    // One shared `NumFmt` per custom id a cell format or differential format refers to (`NumFmt`
349    // is an `Arc`); declared but unused codes get no second copy.
350    let mut used: HashSet<u32> = HashSet::new();
351    for list in ["cellXfs", "cellStyleXfs"] {
352        if let Some(x) = root.child(list) {
353            used.extend(x.kids("xf").map(|xf| xf.attr_u32("numFmtId").unwrap_or(0)));
354        }
355    }
356    if let Some(d) = root.child("dxfs") {
357        used.extend(
358            d.kids("dxf").filter_map(|x| x.child("numFmt")).filter(|n| n.attr("formatCode").is_none()).filter_map(|n| n.attr_u32("numFmtId")),
359        );
360    }
361    let shared_fmts: HashMap<u32, NumFmt> = fmts.iter().filter(|(id, _)| used.contains(id)).map(|(&id, code)| (id, NumFmt::new(code))).collect();
362    // The scheme link only means something on a font in the body face; elsewhere it stays as a new font has it.
363    let body_font = |e: &El| {
364        let mut f = read_font(e, &palette);
365        f.minor |= f.name != minor_font;
366        f
367    };
368    let fonts: Vec<Font> = root.child("fonts").map(|f| f.kids("font").map(body_font).collect()).unwrap_or_default();
369    let fills: Vec<Fill> = root.child("fills").map(|f| f.kids("fill").map(|e| read_fill(e, &palette, false)).collect()).unwrap_or_default();
370    let borders: Vec<Borders> = root.child("borders").map(|f| f.kids("border").map(|e| read_border(e, &palette)).collect()).unwrap_or_default();
371    let xfs = root
372        .child("cellXfs")
373        .map(|x| x.kids("xf").map(|xf| read_xf(xf, &fonts, &fills, &borders, &shared_fmts, checkboxes)).collect())
374        .unwrap_or_default();
375    let style_xfs: Vec<Style> = root
376        .child("cellStyleXfs")
377        .map(|x| x.kids("xf").map(|xf| read_xf(xf, &fonts, &fills, &borders, &shared_fmts, &[])).collect())
378        .unwrap_or_default();
379    let mut named = Vec::new();
380    if let Some(cs) = root.child("cellStyles") {
381        for c in cs.kids("cellStyle") {
382            if let (Some(name), Some(s)) = (c.attr("name"), c.attr_u32("xfId").and_then(|i| style_xfs.get(i as usize)))
383                && name != "Normal"
384            {
385                named.push((name.to_string(), s.clone()));
386            }
387        }
388    }
389    let dxfs = root.child("dxfs").map(|d| d.kids("dxf").map(|x| read_dxf(x, &palette, &shared_fmts)).collect()).unwrap_or_default();
390    StylesIn { xfs, dxfs, palette, named, num_fmts: fmts }
391}

393/// One differential format (`dxf`): the style it sets over a cell's own.
394fn read_dxf(x: &El, palette: &[u32], fmts: &HashMap<u32, NumFmt>) -> Style {
395    let mut s = Style::default();
396    if let Some(e) = x.child("font") {
397        let f = read_font(e, palette);
398        // Differential fonts rarely name a face; unless stated, keep the default face and size.
399        let has_size = e.child("sz").and_then(|n| n.attr_f64("val")).is_some();
400        let has_name = e.child("name").and_then(|n| n.attr("val")).is_some();
401        let name = if has_name { f.name.clone() } else { s.font.name.clone() };
402        let size = if has_size { f.size } else { s.font.size };
403        let def = Font::default();
404        let st = &mut s.dxf_stated;
405        st.bold = e.child("b").is_some() && !f.bold;
406        st.italic = e.child("i").is_some() && !f.italic;
407        st.strike = e.child("strike").is_some() && !f.strike;
408        st.underline = e.child("u").is_some() && f.underline == Underline::None;
409        st.size = has_size && size == def.size;
410        st.name = has_name && name == def.name;
411        st.scheme = e.child_val("scheme").map(str::to_string);
412        // A differential font's scheme is in `dxf_stated`.
413        s.font = Font { name, size, minor: s.font.minor, ..f };
414    }
415    if let Some(n) = x.child("numFmt") {
416        if let Some(code) = n.attr("formatCode") {
417            s.num_fmt = NumFmt::new(code);
418        } else if let Some(id) = n.attr_u32("numFmtId") {
419            s.num_fmt = num_fmt_code(id, fmts);
420        }
421    }
422    if let Some(f) = x.child("fill") {
423        s.fill = read_fill(f, palette, true);
424    }
425    if let Some(b) = x.child("border") {
426        s.borders = read_border(b, palette);
427    }
428    if let Some(a) = x.child("alignment") {
429        s.align = read_alignment(a);
430    }
431    if let Some(p) = x.child("protection") {
432        s.protection = read_protection(p);
433        s.dxf_stated.protection = s.protection == Protection::default();
434    }
435    s
436}

440pub fn color_attrs(c: &Color) -> String {
441    match *c {
442        Color::Auto => "auto=\"1\"".into(),
443        Color::Rgb(v) => format!("rgb=\"FF{:06X}\"", v & 0xFFFFFF),
444        Color::Theme(i, t) => {
445            if t == 0 {
446                format!("theme=\"{i}\"")
447            } else {
448                format!("theme=\"{i}\" tint=\"{}\"", num(t as f64 / 1000.0))
449            }
450        }
451    }
452}

454/// `minor`: the font follows the theme's body font (`<scheme val="minor"/>`). `dxf`: a
455/// differential format's font, which writes only what it states.
456fn font_xml(f: &Font, dxf: Option<&DxfStated>, minor: bool) -> String {
457    let stated = dxf.cloned().unwrap_or_default();
458    let mut s = String::from("<font>");
459    for (on, said, tag) in [(f.bold, stated.bold, "b"), (f.italic, stated.italic, "i"), (f.strike, stated.strike, "strike")] {
460        if on {
461            let _ = write!(s, "<{tag}/>");
462        } else if said {
463            let _ = write!(s, "<{tag} val=\"0\"/>");
464        }
465    }
466    match f.underline {
467        Underline::None if stated.underline => s.push_str("<u val=\"none\"/>"),
468        Underline::None => {}
469        Underline::Single => s.push_str("<u/>"),
470        Underline::Double => s.push_str("<u val=\"double\"/>"),
471        Underline::SingleAccounting => s.push_str("<u val=\"singleAccounting\"/>"),
472        Underline::DoubleAccounting => s.push_str("<u val=\"doubleAccounting\"/>"),
473    }
474    match f.vert {
475        VertAlign::Baseline => {}
476        VertAlign::Superscript => s.push_str("<vertAlign val=\"superscript\"/>"),
477        VertAlign::Subscript => s.push_str("<vertAlign val=\"subscript\"/>"),
478    }
479    let def = Font::default();
480    if dxf.is_none() || stated.size || f.size != def.size {
481        let _ = write!(s, "<sz val=\"{}\"/>", num(f.size as f64));
482    }
483    if f.color != Color::Auto {
484        let _ = write!(s, "<color {}/>", color_attrs(&f.color));
485    } else if dxf.is_none() {
486        s.push_str("<color theme=\"1\"/>");
487    }
488    if dxf.is_none() {
489        let family = f.family.filter(|v| *v <= MAX_FONT_FAMILY).unwrap_or(DEFAULT_FONT_FAMILY);
490        let _ = write!(s, "<name val=\"{}\"/><family val=\"{family}\"/>", esc_attr(&f.name));
491        if let Some(charset) = f.charset {
492            let _ = write!(s, "<charset val=\"{charset}\"/>");
493        }
494        if minor {
495            s.push_str("<scheme val=\"minor\"/>");
496        }
497    } else {
498        if stated.name || f.name != def.name {
499            let _ = write!(s, "<name val=\"{}\"/>", esc_attr(&f.name));
500        }
501        if let Some(sc) = &stated.scheme {
502            let _ = write!(s, "<scheme val=\"{}\"/>", esc_attr(sc));
503        }
504    }
505    s.push_str("</font>");
506    s
507}

509fn fill_xml(f: &Fill, dxf: bool) -> String {
510    if f.pattern == PatternType::None {
511        return "<fill><patternFill patternType=\"none\"/></fill>".into();
512    }
513    if dxf && f.pattern == PatternType::Solid {
514        let c = color_attrs(&f.fg);
515        return format!("<fill><patternFill patternType=\"solid\"><fgColor {c}/><bgColor {c}/></patternFill></fill>");
516    }
517    let mut s = format!("<fill><patternFill patternType=\"{}\">", pattern_name(f.pattern));
518    if f.fg != Color::Auto {
519        let _ = write!(s, "<fgColor {}/>", color_attrs(&f.fg));
520    } else if f.pattern != PatternType::Gray125 {
521        s.push_str("<fgColor indexed=\"64\"/>");
522    }
523    if f.bg != Color::Auto {
524        let _ = write!(s, "<bgColor {}/>", color_attrs(&f.bg));
525    } else {
526        s.push_str("<bgColor indexed=\"64\"/>");
527    }
528    s.push_str("</patternFill></fill>");
529    s
530}

532fn line_xml(tag: &str, l: &BorderLine) -> String {
533    if l.is_none() {
534        return format!("<{tag}/>");
535    }
536    let color = if l.color == Color::Auto { "<color indexed=\"64\"/>".to_string() } else { format!("<color {}/>", color_attrs(&l.color)) };
537    format!("<{tag} style=\"{}\">{color}</{tag}>", border_style_name(l.style))
538}

540fn border_xml(b: &Borders) -> String {
541    let mut s = String::from("<border");
542    if !b.diag_up.is_none() {
543        s.push_str(" diagonalUp=\"1\"");
544    }
545    if !b.diag_down.is_none() {
546        s.push_str(" diagonalDown=\"1\"");
547    }
548    s.push('>');
549    s.push_str(&line_xml("left", &b.left));
550    s.push_str(&line_xml("right", &b.right));
551    s.push_str(&line_xml("top", &b.top));
552    s.push_str(&line_xml("bottom", &b.bottom));
553    let diag = if !b.diag_down.is_none() { b.diag_down } else { b.diag_up };
554    s.push_str(&line_xml("diagonal", &diag));
555    s.push_str("</border>");
556    s
557}
558
559fn alignment_xml(a: &Alignment) -> Option<String> {
560    if *a == Alignment::default() {
561        return None;
562    }
563    let mut s = String::from("<alignment");
564    let h = match a.h {
565        HAlign::General => None,
566        HAlign::Left => Some("left"),
567        HAlign::Center => Some("center"),
568        HAlign::Right => Some("right"),
569        HAlign::Fill => Some("fill"),
570        HAlign::Justify => Some("justify"),
571        HAlign::CenterAcross => Some("centerContinuous"),
572        HAlign::Distributed => Some("distributed"),
573    };
574    if let Some(h) = h {
575        let _ = write!(s, " horizontal=\"{h}\"");
576    }
577    let v = match a.v {
578        VAlign::Bottom => None,
579        VAlign::Top => Some("top"),
580        VAlign::Center => Some("center"),
581        VAlign::Justify => Some("justify"),
582        VAlign::Distributed => Some("distributed"),
583    };
584    if let Some(v) = v {
585        let _ = write!(s, " vertical=\"{v}\"");
586    }
587    let rot = match a.rotation {
588        255 => 255,
589        r if (0..=90).contains(&r) => r,
590        r if (-90..0).contains(&r) => 90 - r,
591        _ => 0,
592    };
593    if rot != 0 {
594        let _ = write!(s, " textRotation=\"{rot}\"");
595    }
596    if a.wrap {
597        s.push_str(" wrapText=\"1\"");
598    }
599    if a.indent > 0 {
600        let _ = write!(s, " indent=\"{}\"", a.indent);
601    }
602    if a.shrink {
603        s.push_str(" shrinkToFit=\"1\"");
604    }
605    s.push_str("/>");
606    Some(s)
607}

609fn protection_xml(p: &Protection) -> Option<String> {
610    if *p == Protection::default() {
611        return None;
612    }
613    Some(format!("<protection locked=\"{}\" hidden=\"{}\"/>", p.locked as u8, p.hidden as u8))
614}

616/// Interns `items` into a deduplicated list, returning the index.
617fn intern<T: Clone + Eq + std::hash::Hash>(list: &mut Vec<T>, map: &mut HashMap<T, u32>, item: &T) -> u32 {
618    if let Some(&i) = map.get(item) {
619        return i;
620    }
621    let i = list.len() as u32;
622    list.push(item.clone());
623    map.insert(item.clone(), i);
624    i
625}

627/// Number format ids assigned while writing.
628struct NumFmts {
629    custom: Vec<(u32, String)>,
630    map: HashMap<String, u32>,
631}
632
633impl NumFmts {
634    fn id(&mut self, code: &str) -> u32 {
635        if let Some(id) = builtin_id(code) {
636            return id;
637        }
638        if let Some(&id) = self.map.get(code) {
639            return id;
640        }
641        let id = 164 + self.custom.len() as u32;
642        self.custom.push((id, code.to_string()));
643        self.map.insert(code.to_string(), id);
644        id
645    }
646}

648/// `styles.xml` for the workbook: one cell format per model style, in `StyleId` order (so a
649/// cell's `s` attribute is its style id), plus the given differential formats.
650/// `extra_fmts` are number formats used outside cell styles (PivotTable value fields); the
651/// returned map gives the id of every custom format written.
652pub fn write_styles(wb: &Workbook, dxfs: &[Style], extra_fmts: &[String]) -> (String, HashMap<String, u32>) {
653    let mut nf = NumFmts { custom: vec![], map: HashMap::new() };
654    let mut fonts: Vec<Font> = vec![];
655    let mut font_map = HashMap::new();
656    let mut fills: Vec<Fill> = vec![Fill::default(), Fill { pattern: PatternType::Gray125, fg: Color::Auto, bg: Color::Auto }];
657    let mut fill_map: HashMap<Fill, u32> = fills.iter().enumerate().map(|(i, f)| (*f, i as u32)).collect();
658    let mut borders: Vec<Borders> = vec![Borders::default()];
659    let mut border_map: HashMap<Borders, u32> = HashMap::from([(Borders::default(), 0)]);
660    // The default style's font is font 0.
661    intern(&mut fonts, &mut font_map, &wb.styles.get(crate::model::StyleId::DEFAULT).font);
662
663    let mut xfs = String::new();
664    let mut count = 0;
665    for (_, st) in wb.styles.iter() {
666        let font = intern(&mut fonts, &mut font_map, &st.font);
667        let fill = intern(&mut fills, &mut fill_map, &st.fill);
668        let border = intern(&mut borders, &mut border_map, &st.borders);
669        // A checkbox is General plus the feature-property-bag complement (see `feature_bags`).
670        let checkbox = st.num_fmt.as_str() == "checkbox";
671        let fmt = if checkbox { 0 } else { nf.id(st.num_fmt.as_str()) };
672        let _ = write!(xfs, "<xf numFmtId=\"{fmt}\" fontId=\"{font}\" fillId=\"{fill}\" borderId=\"{border}\" xfId=\"0\"");
673        if fmt != 0 {
674            xfs.push_str(" applyNumberFormat=\"1\"");
675        }
676        if font != 0 {
677            xfs.push_str(" applyFont=\"1\"");
678        }
679        if fill != 0 {
680            xfs.push_str(" applyFill=\"1\"");
681        }
682        if border != 0 {
683            xfs.push_str(" applyBorder=\"1\"");
684        }
685        let al = alignment_xml(&st.align);
686        let pr = protection_xml(&st.protection);
687        if al.is_some() {
688            xfs.push_str(" applyAlignment=\"1\"");
689        }
690        if pr.is_some() {
691            xfs.push_str(" applyProtection=\"1\"");
692        }
693        if al.is_none() && pr.is_none() && !checkbox {
694            xfs.push_str("/>");
695        } else {
696            xfs.push('>');
697            xfs.push_str(&al.unwrap_or_default());
698            xfs.push_str(&pr.unwrap_or_default());
699            if checkbox {
700                xfs.push_str(&crate::xlsx::feature_bags::checkbox_xf_ext());
701            }
702            xfs.push_str("</xf>");
703        }
704        count += 1;
705    }
706
707    // Named cell styles.
708    let mut style_xfs = String::from("<xf numFmtId=\"0\" fontId=\"0\" fillId=\"0\" borderId=\"0\"/>");
709    let mut cell_styles = String::from("<cellStyle name=\"Normal\" xfId=\"0\" builtinId=\"0\"/>");
710    let mut named = 1;
711    for (name, st) in &wb.cell_styles {
712        if name.is_empty() || name == "Normal" {
713            continue;
714        }
715        let font = intern(&mut fonts, &mut font_map, &st.font);
716        let fill = intern(&mut fills, &mut fill_map, &st.fill);
717        let border = intern(&mut borders, &mut border_map, &st.borders);
718        let fmt = nf.id(st.num_fmt.as_str());
719        let _ = write!(style_xfs, "<xf numFmtId=\"{fmt}\" fontId=\"{font}\" fillId=\"{fill}\" borderId=\"{border}\"");
720        let al = alignment_xml(&st.align);
721        let pr = protection_xml(&st.protection);
722        if al.is_none() && pr.is_none() {
723            style_xfs.push_str("/>");
724        } else {
725            let _ = write!(style_xfs, ">{}{}</xf>", al.unwrap_or_default(), pr.unwrap_or_default());
726        }
727        let _ = write!(cell_styles, "<cellStyle name=\"{}\" xfId=\"{named}\" customBuiltin=\"1\"/>", esc_attr(name));
728        named += 1;
729    }
730
731    let mut dx = String::new();
732    for d in dxfs {
733        dx.push_str("<dxf>");
734        let def = Style::default();
735        let st = &d.dxf_stated;
736        if d.font.bold
737            || d.font.italic
738            || d.font.strike
739            || d.font.underline != Underline::None
740            || d.font.color != Color::Auto
741            || d.font.vert != VertAlign::Baseline
742            || d.font.size != def.font.size
743            || d.font.name != def.font.name
744            || st.bold
745            || st.italic
746            || st.strike
747            || st.underline
748            || st.size
749            || st.name
750            || st.scheme.is_some()
751        {
752            dx.push_str(&font_xml(&d.font, Some(st), false));
753        }
754        if d.num_fmt != def.num_fmt {
755            let id = nf.id(d.num_fmt.as_str());
756            let _ = write!(dx, "<numFmt numFmtId=\"{id}\" formatCode=\"{}\"/>", esc_attr(d.num_fmt.as_str()));
757        }
758        if d.fill != def.fill {
759            dx.push_str(&fill_xml(&d.fill, true));
760        }
761        if let Some(a) = alignment_xml(&d.align) {
762            dx.push_str(&a);
763        }
764        if d.borders != def.borders {
765            dx.push_str(&border_xml(&d.borders));
766        }
767        if st.protection || d.protection != def.protection {
768            let _ = write!(dx, "<protection locked=\"{}\" hidden=\"{}\"/>", d.protection.locked as u8, d.protection.hidden as u8);
769        }
770        dx.push_str("</dxf>");
771    }
772    for f in extra_fmts {
773        nf.id(f);
774    }
775
776    let mut s = String::from(
777        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<styleSheet xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\">",
778    );
779    if !nf.custom.is_empty() {
780        let _ = write!(s, "<numFmts count=\"{}\">", nf.custom.len());
781        for (id, code) in &nf.custom {
782            let _ = write!(s, "<numFmt numFmtId=\"{id}\" formatCode=\"{}\"/>", esc_attr(code));
783        }
784        s.push_str("</numFmts>");
785    }
786    let _ = write!(s, "<fonts count=\"{}\">", fonts.len());
787    for (i, f) in fonts.iter().enumerate() {
788        // A Normal font in the theme's body face follows the theme, as Excel writes it (#805); so
789        // does every other body-face font that kept the scheme link (#2517).
790        s.push_str(&font_xml(f, None, (i == 0 || f.minor) && f.name == wb.theme.minor_font));
791    }
792    let _ = write!(s, "</fonts><fills count=\"{}\">", fills.len());
793    for f in &fills {
794        s.push_str(&fill_xml(f, false));
795    }
796    let _ = write!(s, "</fills><borders count=\"{}\">", borders.len());
797    for b in &borders {
798        s.push_str(&border_xml(b));
799    }
800    let _ = write!(s, "</borders><cellStyleXfs count=\"{named}\">{style_xfs}</cellStyleXfs>");
801    let _ = write!(s, "<cellXfs count=\"{count}\">{xfs}</cellXfs>");
802    let _ = write!(s, "<cellStyles count=\"{named}\">{cell_styles}</cellStyles>");
803    let _ = write!(s, "<dxfs count=\"{}\">{dx}</dxfs>", dxfs.len());
804    s.push_str("<tableStyles count=\"0\" defaultTableStyle=\"TableStyleMedium2\" defaultPivotStyle=\"PivotStyleLight16\"/></styleSheet>");
805    (s, nf.map)
806}

808/// Index of a differential format, adding it when new.
809pub fn dxf_id(dxfs: &mut Vec<Style>, s: &Style) -> u32 {
810    if let Some(i) = dxfs.iter().position(|d| d == s) {
811        return i as u32;
812    }
813    dxfs.push(s.clone());
814    (dxfs.len() - 1) as u32
