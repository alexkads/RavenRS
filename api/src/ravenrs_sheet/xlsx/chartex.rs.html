21/// Kinds written as chartex parts.
22pub fn is_chartex(k: ChartKind) -> bool {
23    matches!(k, ChartKind::Histogram | ChartKind::BoxWhisker | ChartKind::Waterfall | ChartKind::Funnel | ChartKind::Treemap | ChartKind::Sunburst)
24}

26/// The `mc:Choice` requirement for a kind's drawing frame: (prefix, namespace). Funnel charts
27/// came in a later revision of the format.
28pub fn requires(k: ChartKind) -> (&'static str, &'static str) {
29    match k {
30        ChartKind::Funnel => ("cx2", "http://schemas.microsoft.com/office/drawing/2015/10/21/chartex"),
31        _ => ("cx1", "http://schemas.microsoft.com/office/drawing/2015/9/8/chartex"),
32    }
33}

37fn kind_of(layout: &str) -> Option<ChartKind> {
38    Some(match layout {
39        "clusteredColumn" | "paretoLine" => ChartKind::Histogram,
40        "boxWhisker" => ChartKind::BoxWhisker,
41        "waterfall" => ChartKind::Waterfall,
42        "funnel" => ChartKind::Funnel,
43        "treemap" => ChartKind::Treemap,
44        "sunburst" => ChartKind::Sunburst,
45        _ => return None,
46    })
47}

49fn formula(dim: &El) -> Option<String> {
50    let f = dim.child("f")?;
51    Some(crate::xlsx::fmla::from_file(&crate::xlsx::chart::chart_ref_from_file(&f.text)))
52}
53
54fn text(tx: &El) -> Option<String> {
55    if let Some(d) = tx.child("txData") {
56        if let Some(f) = d.child("f") {
57            return Some(crate::xlsx::chart::chart_ref_from_file(&f.text));
58        }
59        return d.child("v").map(|v| v.text.clone());
60    }
61    let rich = tx.child("rich")?;
62    let paras: Vec<String> = rich.kids("p").map(|p| p.kids("r").filter_map(|r| r.child("t")).map(|t| t.text.as_str()).collect()).collect();
63    let s = paras.join("\n");
64    if s.is_empty() { None } else { Some(s) }
65}

67pub fn read_chartex(root: &El) -> Option<Chart> {
68    let chart = root.child("chart")?;
69    let plot = chart.child("plotArea")?;
70    let region = plot.child("plotAreaRegion")?;
71    // Data by id: (categories, values).
72    let data: Vec<(String, Option<String>, Option<String>)> = root
73        .child("chartData")
74        .map(|cd| {
75            cd.kids("data")
76                .map(|d| {
77                    let id = d.attr("id").unwrap_or("").to_string();
78                    let cat = d.kids("strDim").chain(d.kids("numDim")).find(|x| matches!(x.attr("type"), Some("cat" | "x"))).and_then(formula);
79                    let val = d.kids("numDim").find(|x| matches!(x.attr("type"), Some("val" | "size" | "y"))).and_then(formula);
80                    (id, cat, val)
81                })
82                .collect()
83        })
84        .unwrap_or_default();
85    let mut kind = None;
86    let mut series = vec![];
87    let mut data_labels = false;
88    for s in region.kids("series") {
89        let layout = s.attr("layoutId").unwrap_or("");
90        // A Pareto line belongs to its histogram.
91        if layout == "paretoLine" {
92            continue;
93        }
94        let k = kind_of(layout)?;
95        kind.get_or_insert(k);
96        let id = s.child_val("dataId").unwrap_or("");
97        let (cat, val) = data.iter().find(|(i, _, _)| i == id).map(|(_, c, v)| (c.clone(), v.clone())).unwrap_or_default();
98        data_labels |= s.path(&["dataLabels", "visibility"]).is_some_and(|v| v.flag("value", false));
99        series.push(Series {
100            quartiles: match s.path(&["layoutPr", "statistics"]).and_then(|el| el.attr("quartileMethod")) {
101                Some("inclusive") => QuartileMethod::Inclusive,
102                _ => QuartileMethod::Exclusive,
103            },
104            name: s.child("tx").and_then(text),
105            categories: cat,
106            values: val.unwrap_or_default(),
107            // The default colour (the series' theme accent, which the writer spells out) isn't kept.
108            color: s.path(&["spPr", "solidFill"]).and_then(dml_color).filter(|c| *c != crate::model::Color::Theme(4 + (series.len() % 6) as u8, 0)),
109            ..Default::default()
110        });
111    }
112    let legend = match chart.child("legend") {
113        None => LegendPos::None,
114        Some(l) => match l.attr("pos").unwrap_or("r") {
115            "b" => LegendPos::Bottom,
116            "t" => LegendPos::Top,
117            "l" => LegendPos::Left,
118            _ => LegendPos::Right,
119        },
120    };
121    let axis_title = |id: &str| plot.kids("axis").find(|a| a.attr("id") == Some(id)).and_then(|a| a.path(&["title", "tx"])).and_then(text);
122    Some(Chart {
123        id: 0,
124        kind: kind?,
125        anchor: Default::default(),
126        title: chart.path(&["title", "tx"]).and_then(text),
127        series,
128        legend,
129        data_labels,
130        data_table: None,
131        gridlines: plot.kids("axis").any(|a| a.child("majorGridlines").is_some()),
132        style: 2,
133        hole_size: 75,
134        first_slice_angle: 0,
135        x_title: axis_title("0"),
136        y_title: axis_title("1"),
137        source: None,
138        by_rows: false,
139        pivot: None,
140    })
141}

145/// Prefix of the hidden defined names chartex formulas go through: a chartex part refers to
146/// its data by names (`_xlchart.v1.0`), which the workbook defines, not by references.
147pub const NAME_PREFIX: &str = "_xlchart.v1.";
148
149/// `<cx:f>` for a reference: a new hidden workbook name for it (collected in `names`, written
150/// with the workbook's defined names). A reference along a row says so; otherwise each column
151/// is one level.
152fn f_xml(wb: &Workbook, si: usize, f: &str, names: &mut Vec<String>) -> String {
153    let row = ref_range(wb, si, f).is_some_and(|(_, r)| r.height() == 1 && r.width() > 1);
154    names.push(f.to_string());
155    format!("<cx:f{}>{NAME_PREFIX}{}</cx:f>", if row { " dir=\"row\"" } else { "" }, names.len() - 1)
156}
157
158/// A reference's sheet and range (for the row/column direction).
159fn ref_range(wb: &Workbook, si: usize, f: &str) -> Option<(usize, RangeRef)> {
160    let Ok(Expr::Ref(r)) = crate::formula::parse(f) else { return None };
161    let sheet = match &r.sheet {
162        SheetSel::Current => si,
163        SheetSel::Named(n) => wb.sheet_index(n)?,
164        SheetSel::Span(..) => return None,
165    };
166    Some((sheet, r.range()))
167}
168
169fn tx_xml(t: &str, wb: &Workbook, si: usize, names: &mut Vec<String>) -> String {
170    if ref_range(wb, si, t).is_some() {
171        let v = crate::xlsx::chart::ref_values(wb, si, t).and_then(|v| v.into_iter().next()).map(|v| v.display()).unwrap_or_default();
172        format!("<cx:tx><cx:txData>{}<cx:v>{}</cx:v></cx:txData></cx:tx>", f_xml(wb, si, t, names), esc(&v))
173    } else {
174        let lit = t.trim().strip_prefix('"').and_then(|x| x.strip_suffix('"')).unwrap_or(t);
175        format!("<cx:tx><cx:txData><cx:v>{}</cx:v></cx:txData></cx:tx>", esc(lit))
176    }
177}

179/// The chart style part a chartex part comes with: every element at its plain default.
180pub fn style_xml() -> String {
181    let mut s = String::from(
182        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<cs:chartStyle xmlns:cs=\"http://schemas.microsoft.com/office/drawing/2012/chartStyle\" xmlns:a=\"http://schemas.openxmlformats.org/drawingml/2006/main\" id=\"201\">",
183    );
184    for e in [
185        "axisTitle",
186        "categoryAxis",
187        "chartArea",
188        "dataLabel",
189        "dataLabelCallout",
190        "dataPoint",
191        "dataPoint3D",
192        "dataPointLine",
193        "dataPointMarker",
194        "dataPointMarkerLayout",
195        "dataPointWireframe",
196        "dataTable",
197        "downBar",
198        "dropLine",
199        "errorBar",
200        "floor",
201        "gridlineMajor",
202        "gridlineMinor",
203        "hiLoLine",
204        "leaderLine",
205        "legend",
206        "plotArea",
207        "plotArea3D",
208        "seriesAxis",
209        "seriesLine",
210        "title",
211        "trendline",
212        "trendlineLabel",
213        "upBar",
214        "valueAxis",
215        "wall",
216    ] {
217        if e == "dataPointMarkerLayout" {
218            s.push_str("<cs:dataPointMarkerLayout symbol=\"circle\" size=\"5\"/>");
219        } else {
220            let _ = write!(
221                s,
222                "<cs:{e}><cs:lnRef idx=\"0\"/><cs:fillRef idx=\"0\"/><cs:effectRef idx=\"0\"/><cs:fontRef idx=\"minor\"><a:schemeClr val=\"tx1\"/></cs:fontRef></cs:{e}>"
223            );
224        }
225    }
226    s.push_str("</cs:chartStyle>");
227    s
228}

230/// The colour style part a chartex part comes with: the theme accents in turn.
231pub fn colors_xml() -> String {
232    let mut s = String::from(
233        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<cs:colorStyle xmlns:cs=\"http://schemas.microsoft.com/office/drawing/2012/chartStyle\" xmlns:a=\"http://schemas.openxmlformats.org/drawingml/2006/main\" meth=\"cycle\" id=\"10\">",
234    );
235    for i in 1..=6 {
236        let _ = write!(s, "<a:schemeClr val=\"accent{i}\"/>");
237    }
238    s.push_str("<cs:variation/></cs:colorStyle>");
239    s

242/// A solid fill (and outline, for line-drawn parts like whiskers) as `cx:spPr`.
243fn fill_xml(color: &str, line: bool) -> String {
244    let ln = if line { format!("<a:ln w=\"9525\"><a:solidFill>{color}</a:solidFill></a:ln>") } else { String::new() };
245    format!("<cx:spPr><a:solidFill>{color}</a:solidFill>{ln}</cx:spPr>")
246}

248/// Light grey for gridlines and the chart border: the text colour at a quarter strength.
249const GREY: &str = "<a:schemeClr val=\"tx1\"><a:lumMod val=\"25000\"/><a:lumOff val=\"75000\"/></a:schemeClr>";
250
251fn accent(i: usize) -> String {
252    format!("<a:schemeClr val=\"accent{}\"/>", i % 6 + 1)
253}

255/// Chartex part XML for a model chart on sheet `si`. The references it uses are appended to
256/// `names` (see [`NAME_PREFIX`]).
257pub fn write_chartex(wb: &Workbook, si: usize, ch: &Chart, names: &mut Vec<String>) -> String {
258    let hierarchy = matches!(ch.kind, ChartKind::Treemap | ChartKind::Sunburst);
259    let mut s = String::from(
260        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<cx:chartSpace xmlns:a=\"http://schemas.openxmlformats.org/drawingml/2006/main\" xmlns:r=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships\" xmlns:cx=\"http://schemas.microsoft.com/office/drawing/2014/chartex\"><cx:chartData>",
261    );
262    for (i, ser) in ch.series.iter().enumerate() {
263        let _ = write!(s, "<cx:data id=\"{i}\">");
264        if let Some(c) = ser.categories.as_deref().filter(|_| ch.kind != ChartKind::Histogram) {
265            let _ = write!(s, "<cx:strDim type=\"cat\">{}</cx:strDim>", f_xml(wb, si, c, names));
266        }
267        let _ = write!(s, "<cx:numDim type=\"{}\">{}</cx:numDim>", if hierarchy { "size" } else { "val" }, f_xml(wb, si, &ser.values, names));
268        s.push_str("</cx:data>");
269    }
270    s.push_str("</cx:chartData><cx:chart>");
271    if let Some(t) = ch.title.as_deref().filter(|t| !t.is_empty()) {
272        let _ = write!(s, "<cx:title pos=\"t\" align=\"ctr\" overlay=\"0\"><cx:tx><cx:txData><cx:v>{}</cx:v></cx:txData></cx:tx></cx:title>", esc(t));
273    }
274    s.push_str("<cx:plotArea><cx:plotAreaRegion>");
275    let layout = match ch.kind {
276        ChartKind::BoxWhisker => "boxWhisker",
277        ChartKind::Waterfall => "waterfall",
278        ChartKind::Funnel => "funnel",
279        ChartKind::Treemap => "treemap",
280        ChartKind::Sunburst => "sunburst",
281        _ => "clusteredColumn",
282    };
283    for (i, ser) in ch.series.iter().enumerate() {
284        let _ = write!(s, "<cx:series layoutId=\"{layout}\" uniqueId=\"{{00000000-0000-0000-0000-{:08X}{i:04X}}}\">", ch.id);
285        if let Some(n) = &ser.name {
286            s.push_str(&tx_xml(n, wb, si, names));
287        }
288        // Fills are written out: they don't come from the style part. Waterfall decreases and
289        // treemap/sunburst boxes get their own colours.
290        let color = ser.color.as_ref().and_then(|c| srgb(c, &wb.theme)).unwrap_or_else(|| accent(i));
291        s.push_str(&fill_xml(&color, ch.kind == ChartKind::BoxWhisker));
292        let points = || crate::xlsx::chart::ref_values(wb, si, &ser.values).unwrap_or_default().into_iter().take(10_000).enumerate();
293        match ch.kind {
294            ChartKind::Waterfall => {
295                for (j, v) in points() {
296                    if v.as_f64().is_some_and(|n| n < 0.0) {
297                        let _ = write!(s, "<cx:dataPt idx=\"{j}\">{}</cx:dataPt>", fill_xml(&accent(i + 1), false));
298                    }
299                }
300            }
301            ChartKind::Treemap | ChartKind::Sunburst if ser.color.is_none() => {
302                for (j, _) in points() {
303                    let _ = write!(s, "<cx:dataPt idx=\"{j}\">{}</cx:dataPt>", fill_xml(&accent(j), false));
304                }
305            }
306            _ => {}
307        }
308        if ch.data_labels {
309            s.push_str("<cx:dataLabels><cx:visibility seriesName=\"0\" categoryName=\"0\" value=\"1\"/></cx:dataLabels>");
310        }
311        let _ = write!(s, "<cx:dataId val=\"{i}\"/>");
312        match ch.kind {
313            ChartKind::Histogram => s.push_str("<cx:layoutPr><cx:binning intervalClosed=\"r\"/></cx:layoutPr>"),
314            ChartKind::BoxWhisker => {
315                let method = match ser.quartiles {
316                    QuartileMethod::Exclusive => "exclusive",
317                    QuartileMethod::Inclusive => "inclusive",
318                };
319                let _ = write!(
320                    s,
321                    "<cx:layoutPr><cx:visibility meanLine=\"0\" meanMarker=\"1\" nonoutliers=\"0\" outliers=\"1\"/><cx:statistics quartileMethod=\"{method}\"/></cx:layoutPr>"
322                );
323            }
324            ChartKind::Treemap => s.push_str("<cx:layoutPr><cx:parentLabelLayout val=\"overlapping\"/></cx:layoutPr>"),
325            _ => {}
326        }
327        s.push_str("</cx:series>");
328    }
329    s.push_str("</cx:plotAreaRegion>");
330    let gap = match ch.kind {
331        ChartKind::Histogram => Some("0"),
332        ChartKind::BoxWhisker => Some("1"),
333        ChartKind::Waterfall => Some("0.5"),
334        ChartKind::Funnel => Some("0.06"),
335        _ => None,
336    };
337    let title = |t: Option<&str>| {
338        t.filter(|t| !t.is_empty())
339            .map(|t| format!("<cx:title><cx:tx><cx:txData><cx:v>{}</cx:v></cx:txData></cx:tx></cx:title>", esc(t)))
340            .unwrap_or_default()
341    };
342    if let Some(gap) = gap {
343        let _ =
344            write!(s, "<cx:axis id=\"0\"><cx:catScaling gapWidth=\"{}\"/>{}<cx:tickLabels/></cx:axis>", esc_attr(gap), title(ch.x_title.as_deref()));
345        if ch.kind != ChartKind::Funnel {
346            let _ = write!(
347                s,
348                "<cx:axis id=\"1\"><cx:valScaling/>{}{}<cx:tickLabels/></cx:axis>",
349                title(ch.y_title.as_deref()),
350                if ch.gridlines {
351                    format!("<cx:majorGridlines><cx:spPr><a:ln w=\"9525\"><a:solidFill>{GREY}</a:solidFill></a:ln></cx:spPr></cx:majorGridlines>")
352                } else {
353                    String::new()
354                }
355            );
356        }
357    }
358    s.push_str("</cx:plotArea>");
359    let pos = match ch.legend {
360        LegendPos::None => None,
361        LegendPos::Bottom => Some("b"),
362        LegendPos::Top => Some("t"),
363        LegendPos::Left => Some("l"),
364        LegendPos::Right => Some("r"),
365    };
366    if let Some(p) = pos {
367        let _ = write!(s, "<cx:legend pos=\"{p}\" align=\"ctr\" overlay=\"0\"/>");
368    }
369    // A white chart area with a light border, like other charts.
370    let _ = write!(
371        s,
372        "</cx:chart><cx:spPr><a:solidFill><a:schemeClr val=\"bg1\"/></a:solidFill><a:ln w=\"9525\"><a:solidFill>{GREY}</a:solidFill></a:ln></cx:spPr></cx:chartSpace>"
373    );
374    s
375}

377#[cfg(test)]
378mod quartile_tests {
379    use super::*;
380
381    #[test]
382    fn box_whisker_reads_quartile_method_and_defaults() {
383        for (statistics, expected) in [
384            ("", QuartileMethod::Exclusive),
385            ("<cx:statistics/>", QuartileMethod::Exclusive),
386            (r#"<cx:statistics quartileMethod="unknown"/>"#, QuartileMethod::Exclusive),
387            (r#"<cx:statistics quartileMethod="exclusive"/>"#, QuartileMethod::Exclusive),
388            (r#"<cx:statistics quartileMethod="inclusive"/>"#, QuartileMethod::Inclusive),
389        ] {
390            let xml = format!(
391                r#"<cx:chartSpace xmlns:cx="http://schemas.microsoft.com/office/drawing/2014/chartex"><cx:chart><cx:plotArea><cx:plotAreaRegion><cx:series layoutId="boxWhisker"><cx:layoutPr>{statistics}</cx:layoutPr></cx:series></cx:plotAreaRegion></cx:plotArea></cx:chart></cx:chartSpace>"#
392            );
393            let root = crate::xlsx::xml::parse(xml.as_bytes()).unwrap();
394            let chart = read_chartex(&root).unwrap();
395            assert_eq!(chart.series[0].quartiles, expected);
396        }
397    }
