16const NS_RICH: &str = "http://schemas.microsoft.com/office/spreadsheetml/2017/richdata";
17const NS_IMAGE_REL: &str = "http://schemas.microsoft.com/office/spreadsheetml/2022/richvaluerel";
18const RICH_EXT: &str = "{3E2802C4-A4D2-4D8B-9148-E3BE6C30E623}";

25/// Header dimensions bound decode allocations in consumers. Full decoding is left to the UI.
26fn image_type(data: &[u8]) -> Option<(&'static str, &'static str)> {
27    let dimensions = if data.len() >= 33 && data.starts_with(b"\x89PNG\r\n\x1a\n") && data.get(8..16)? == b"\0\0\0\rIHDR" {
28        let w = u32::from_be_bytes(data.get(16..20)?.try_into().ok()?);
29        let h = u32::from_be_bytes(data.get(20..24)?.try_into().ok()?);
30        Some((w, h, "image/png", "png"))
31    } else if data.starts_with(&[0xff, 0xd8]) {
32        jpeg_dimensions(data).map(|(w, h)| (w, h, "image/jpeg", "jpeg"))
33    } else {
34        None
35    };
36    let (w, h, mime, ext) = dimensions?;
37    (w > 0 && h > 0 && w <= 8192 && h <= 8192 && u64::from(w) * u64::from(h) <= MAX_PIXELS).then_some((mime, ext))
38}

40fn jpeg_dimensions(data: &[u8]) -> Option<(u32, u32)> {
41    let mut offset = 2usize;
42    while offset < data.len() {
43        if *data.get(offset)? != 0xff {
44            return None;
45        }
46        while data.get(offset) == Some(&0xff) {
47            offset += 1;
48        }
49        let marker = *data.get(offset)?;
50        offset += 1;
51        if matches!(marker, 0xd9 | 0xda) {
52            return None;
53        }
54        if marker == 0x01 || (0xd0..=0xd7).contains(&marker) {
55            continue;
56        }
57        let len = usize::from(u16::from_be_bytes(data.get(offset..offset.checked_add(2)?)?.try_into().ok()?));
58        let end = offset.checked_add(len)?;
59        if len < 2 || end > data.len() {
60            return None;
61        }
62        if matches!(marker, 0xc0..=0xc3 | 0xc5..=0xc7 | 0xc9..=0xcb | 0xcd..=0xcf) && len >= 8 {
63            let h = u16::from_be_bytes(data.get(offset + 3..offset + 5)?.try_into().ok()?);
64            let w = u16::from_be_bytes(data.get(offset + 5..offset + 7)?.try_into().ok()?);
65            return Some((u32::from(w), u32::from(h)));
66        }
67        offset = end;
68    }
69    None
70}

72fn part(cx: &mut Ctx<'_>, rels: &[Rel], kind: &str) -> Option<String> {
73    let mut target: Option<String> = None;
74    for rel in rels.iter().filter(|rel| rel.kind == kind) {
75        if rel.external {
76            cx.warn(format!("external {kind} relationship is unsupported; cell pictures may be missing"));
77            return None;
78        }
79        if target.as_ref().is_some_and(|old| normalize(old) != normalize(&rel.target)) {
80            cx.warn(format!("conflicting {kind} relationships; cell pictures may be missing"));
81            return None;
82        }
83        target = Some(rel.target.clone());
84    }
85    target
86}

88fn read_xml(cx: &mut Ctx<'_>, path: &str, root: &str) -> Result<Option<El>, IoError> {
89    let bytes = match cx.pkg.read_limited(path, MAX_XML) {
90        Ok(Some(bytes)) => bytes,
91        Err(error @ IoError::TooLarge(_)) => return Err(error),
92        other => {
93            cx.warn(format!("missing or damaged cell-picture part {path}: {}", other.err().map(|e| e.to_string()).unwrap_or_default()));
94            return Ok(None);
95        }
96    };
97    match xml::parse(&bytes) {
98        Ok(el) if el.name == root => Ok(Some(el)),
99        _ => {
100            cx.warn(format!("invalid cell-picture part {path}"));
101            Ok(None)
102        }
103    }
104}

106fn entries<'a>(cx: &mut Ctx<'_>, el: &'a El, child: &'a str) -> Result<Vec<&'a El>, IoError> {
107    let entries: Vec<_> = el.kids(child).take(MAX_RECORDS + 1).collect();
108    if entries.len() > MAX_RECORDS {
109        return Err(IoError::TooLarge("too many cell-picture metadata records".into()));
110    }
111    if el.attr("count").is_some() && el.attr_u32("count").map(|n| n as usize) != Some(entries.len()) {
112        cx.warn(format!("cell-picture {} count does not match its records", el.name));
113    }
114    Ok(entries)
115}

117/// Resolves every value-metadata block without assuming the indices at each layer coincide.
118pub fn read(cx: &mut Ctx<'_>, workbook_rels: &[Rel]) -> Result<Vec<Option<Arc<CellPicture>>>, IoError> {
119    let Some(metadata_path) = part(cx, workbook_rels, "sheetMetadata") else { return Ok(vec![]) };
120    let Some(metadata) = read_xml(cx, &metadata_path, "metadata")? else { return Ok(vec![]) };
121    let Some(types) = metadata.child("metadataTypes") else { return Ok(vec![]) };
122    let types = entries(cx, types, "metadataType")?;
123    let Some(type_id) = types.iter().position(|el| el.attr("name") == Some("XLRICHVALUE")).map(|n| n + 1) else { return Ok(vec![]) };
124    if types.iter().filter(|el| el.attr("name") == Some("XLRICHVALUE")).count() != 1 {
125        cx.warn("duplicate XLRICHVALUE metadata types; cell pictures were not loaded");
126        return Ok(vec![]);
127    }
128    // Support workbook-owned parts and the implicit owner chain described by MS-XLSX.
129    let mut rels = workbook_rels.to_vec();
130    rels.extend(cx.pkg.rels(&metadata_path)?);
131    let Some(data_path) = part(cx, &rels, "rdRichValue") else {
132        cx.warn("missing rich-value data relationship; cell pictures were not loaded");
133        return Ok(vec![]);
134    };
135    rels.extend(cx.pkg.rels(&data_path)?);
136    let (Some(structure_path), Some(image_rels_path)) = (part(cx, &rels, "rdRichValueStructure"), part(cx, &rels, "richValueRel")) else {
137        cx.warn("missing rich-value structure or image relationships; cell pictures were not loaded");
138        return Ok(vec![]);
139    };
140    let Some(data) = read_xml(cx, &data_path, "rvData")? else { return Ok(vec![]) };
141    let Some(structures) = read_xml(cx, &structure_path, "rvStructures")? else { return Ok(vec![]) };
142    let Some(image_rel_list) = read_xml(cx, &image_rels_path, "richValueRels")? else { return Ok(vec![]) };
143    let image_rels = cx.pkg.rels(&image_rels_path)?;
144    let structures = entries(cx, &structures, "s")?;
145    let image_rel_list = entries(cx, &image_rel_list, "rel")?;
146    let values = entries(cx, &data, "rv")?;
147    let mut pictures = Vec::with_capacity(values.len());
148    let mut cache = HashMap::new();
149    let mut total = 0usize;
150    for value in values {
151        let picture = read_value(cx, value, &structures, &image_rel_list, &image_rels, &mut cache, &mut total)?;
152        if picture.is_none() {
153            cx.warn("some rich values are unsupported or malformed; their scalar fallbacks were retained (cell pictures may be missing)");
154        }
155        pictures.push(picture);
156    }
157    let Some(future) = metadata.kids("futureMetadata").find(|el| el.attr("name") == Some("XLRICHVALUE")) else {
158        cx.warn("missing XLRICHVALUE future metadata; cell pictures were not loaded");
159        return Ok(vec![]);
160    };
161    let future: Vec<_> = entries(cx, future, "bk")?
162        .into_iter()
163        .map(|block| {
164            block
165                .child("extLst")
166                .and_then(|el| el.kids("ext").find(|ext| ext.attr("uri").is_some_and(|uri| uri.eq_ignore_ascii_case(RICH_EXT))))
167                .and_then(|ext| ext.child("rvb"))
168                .and_then(|rvb| rvb.attr_u32("i"))
169                .map(|n| n as usize)
170        })
171        .collect();
172    let Some(blocks) = metadata.child("valueMetadata") else { return Ok(vec![]) };
173    let blocks = entries(cx, blocks, "bk")?;
174    Ok(blocks
175        .into_iter()
176        .map(|block| {
177            block
178                .kids("rc")
179                .find(|rc| rc.attr_u32("t").map(|n| n as usize) == Some(type_id))
180                .and_then(|rc| rc.attr_u32("v"))
181                .and_then(|n| future.get(n as usize))
182                .copied()
183                .flatten()
184                .and_then(|n| pictures.get(n))
185                .cloned()
186                .flatten()
187        })
188        .collect())
189}

191fn read_value(
192    cx: &mut Ctx<'_>,
193    value: &El,
194    structures: &[&El],
195    image_rel_list: &[&El],
196    image_rels: &[Rel],
197    cache: &mut HashMap<(String, String), Arc<CellPicture>>,
198    total: &mut usize,
199) -> Result<Option<Arc<CellPicture>>, IoError> {
200    let Some(structure) = value.attr_u32("s").and_then(|n| structures.get(n as usize)) else { return Ok(None) };
201    if structure.attr("t") != Some("_localImage") {
202        return Ok(None);
203    }
204    let keys: Vec<_> = structure.kids("k").collect();
205    let values: Vec<_> = value.kids("v").collect();
206    if keys.len() != values.len() {
207        return Ok(None);
208    }
209    let lookup = |name: &str, ty: &str| {
210        let mut matches = keys.iter().zip(&values).filter(|(k, _)| k.attr("n") == Some(name));
211        let (key, value) = matches.next()?;
212        (matches.next().is_none() && key.attr("t") == Some(ty)).then_some(value.text.as_str())
213    };
214    let Some(rel_id) = lookup("_rvRel:LocalImageIdentifier", "i")
215        .and_then(|v| v.trim().parse::<usize>().ok())
216        .and_then(|n| image_rel_list.get(n))
217        .and_then(|rel| rel.attr("id"))
218    else {
219        return Ok(None);
220    };
221    let mut matches = image_rels.iter().filter(|rel| rel.id == rel_id);
222    let Some(rel) = matches.next() else { return Ok(None) };
223    if matches.next().is_some() || rel.kind != "image" || rel.external {
224        cx.warn("a cell picture has an external or invalid image relationship; kept its scalar fallback");
225        return Ok(None);
226    }
227    let alt = lookup("Text", "s").unwrap_or("").to_string();
228    let key = (normalize(&rel.target), alt.clone());
229    if let Some(picture) = cache.get(&key) {
230        return Ok(Some(picture.clone()));
231    }
232    let data = match cx.pkg.read_limited(&rel.target, MAX_IMAGE) {
233        Ok(Some(data)) => data,
234        other => {
235            cx.warn(format!(
236                "missing, damaged or oversized cell picture {}: {}; kept its scalar fallback",
237                rel.target,
238                other.err().map(|e| e.to_string()).unwrap_or_default()
239            ));
240            return Ok(None);
241        }
242    };
243    let Some((mime, _)) = image_type(&data) else {
244        cx.warn("a cell picture is not a supported PNG/JPEG or exceeds 8192 pixels per side / 16 million pixels; kept its scalar fallback");
245        return Ok(None);
246    };
247    if total.saturating_add(data.len()) > MAX_IMAGES_TOTAL {
248        cx.warn("cell pictures exceed the 128 MiB workbook limit; remaining pictures kept their scalar fallbacks");
249        return Ok(None);
250    }
251    *total += data.len();
252    let picture = Arc::new(CellPicture { data, mime: mime.to_string(), alt });
253    cache.insert(key, picture.clone());
254    Ok(Some(picture))
255}

257/// One rich value per shared model payload; copied cells can reuse a metadata block.
258#[derive(Default)]
259pub struct PicturesOut {
260    pictures: Vec<Arc<CellPicture>>,
261    indices: HashMap<*const CellPicture, u32>,
262}
263
264impl PicturesOut {
265    pub fn is_empty(&self) -> bool {
266        self.pictures.is_empty()
267    }
268    pub fn index(&mut self, picture: &Arc<CellPicture>) -> u32 {
269        let key = Arc::as_ptr(picture);
270        if let Some(index) = self.indices.get(&key) {
271            return *index;
272        }
273        let index = self.pictures.len().saturating_add(1) as u32;
274        self.pictures.push(picture.clone());
275        self.indices.insert(key, index);
276        index
277    }
278}

280pub fn write(out: &mut Out, workbook_rels: &mut Rels) -> Result<(), IoError> {
281    let pictures = std::mem::take(&mut out.cell_pictures).pictures;
282    if pictures.len() > MAX_RECORDS {
283        return Err(IoError::TooLarge("too many cell pictures".into()));
284    }
285    let count = pictures.len();
286    let mut data = format!("{XML_DECL}<rvData xmlns=\"{NS_RICH}\" count=\"{count}\">");
287    let mut rel_list = format!("{XML_DECL}<richValueRels xmlns=\"{NS_IMAGE_REL}\" xmlns:r=\"{NS_REL}\">");
288    let mut image_rels = Rels::default();
289    let mut total = 0usize;
290    for (i, picture) in pictures.iter().enumerate() {
291        total = total.saturating_add(picture.data.len());
292        if picture.data.len() as u64 > MAX_IMAGE || total > MAX_IMAGES_TOTAL {
293            return Err(IoError::TooLarge("cell pictures exceed the 16 MiB image or 128 MiB workbook limit".into()));
294        }
295        let (mime, ext) = image_type(&picture.data)
296            .ok_or_else(|| IoError::Format("cell pictures must be PNG/JPEG with at most 8192 pixels per side and 16 million pixels".into()))?;
297        let image_name = format!("cellImage{}.{}", i + 1, ext);
298        out.default_type(ext, mime);
299        out.part(&format!("xl/media/{image_name}"), None, picture.data.clone());
300        let rid = image_rels.add("image", &format!("../media/{image_name}"));
301        let _ = write!(rel_list, "<rel r:id=\"{rid}\"/>");
302        let _ = write!(data, "<rv s=\"0\"><v>{i}</v><v>5</v><v>{}</v></rv>", esc(&picture.alt));
303    }
304    data.push_str("</rvData>");
305    rel_list.push_str("</richValueRels>");
306    let structures = format!(
307        "{XML_DECL}<rvStructures xmlns=\"{NS_RICH}\" count=\"1\"><s t=\"_localImage\"><k n=\"_rvRel:LocalImageIdentifier\" t=\"i\"/><k n=\"CalcOrigin\" t=\"i\"/><k n=\"Text\" t=\"s\"/></s></rvStructures>"
308    );
309    for (name, content_type, relation, bytes) in [
310        (
311            "rdrichvalue.xml",
312            "application/vnd.ms-excel.rdRichValue+xml",
313            "http://schemas.microsoft.com/office/2017/06/relationships/rdRichValue",
314            data.into_bytes(),
315        ),
316        (
317            "rdrichvaluestructure.xml",
318            "application/vnd.ms-excel.rdRichValueStructure+xml",
319            "http://schemas.microsoft.com/office/2017/06/relationships/rdRichValueStructure",
320            structures.into_bytes(),
321        ),
322        (
323            "richValueRel.xml",
324            "application/vnd.ms-excel.richvaluerel+xml",
325            "http://schemas.microsoft.com/office/2022/10/relationships/richValueRel",
326            rel_list.into_bytes(),
327        ),
328    ] {
329        out.part(&format!("xl/richData/{name}"), Some(content_type), bytes);
330        workbook_rels.add_uri(relation, &format!("richData/{name}"));
331    }
332    out.part("xl/richData/_rels/richValueRel.xml.rels", None, image_rels.xml().into_bytes());
333    out.part(
334        "xl/metadata.xml",
335        Some("application/vnd.openxmlformats-officedocument.spreadsheetml.sheetMetadata+xml"),
336        metadata(count, out.dynamic_arrays).into_bytes(),
337    );
338    workbook_rels.add("sheetMetadata", "metadata.xml");
339    Ok(())
340}

342fn metadata(count: usize, dynamic: bool) -> String {
343    let type_id = if dynamic { 2 } else { 1 };
344    let mut xml = format!(
345        "{XML_DECL}<metadata xmlns=\"{NS_MAIN}\" xmlns:rv=\"{NS_RICH}\" xmlns:xda=\"http://schemas.microsoft.com/office/spreadsheetml/2017/dynamicarray\"><metadataTypes count=\"{type_id}\">"
346    );
347    if dynamic {
348        xml.push_str("<metadataType name=\"XLDAPR\" minSupportedVersion=\"120000\" copy=\"1\" pasteAll=\"1\" pasteValues=\"1\" merge=\"1\" splitFirst=\"1\" rowColShift=\"1\" clearFormats=\"1\" clearComments=\"1\" assign=\"1\" coerce=\"1\" cellMeta=\"1\"/>");
349    }
350    xml.push_str("<metadataType name=\"XLRICHVALUE\" minSupportedVersion=\"120000\" copy=\"1\" pasteAll=\"1\" pasteValues=\"1\" merge=\"1\" splitFirst=\"1\" rowColShift=\"1\" clearFormats=\"1\" clearComments=\"1\" assign=\"1\" cellMeta=\"0\"/></metadataTypes>");
351    if dynamic {
352        // Lower case, as the dynamic-array-only metadata in `write.rs` spells it: Excel refuses
353        // a workbook whose XLDAPR extension URI is in upper case (#2339).
354        xml.push_str("<futureMetadata name=\"XLDAPR\" count=\"1\"><bk><extLst><ext uri=\"{bdbb8cdc-fa1e-496e-a857-3c3f30c029c3}\"><xda:dynamicArrayProperties fDynamic=\"1\" fCollapsed=\"0\"/></ext></extLst></bk></futureMetadata>");
355    }
356    let _ = write!(xml, "<futureMetadata name=\"XLRICHVALUE\" count=\"{count}\">");
357    for i in 0..count {
358        let _ = write!(xml, "<bk><extLst><ext uri=\"{RICH_EXT}\"><rv:rvb i=\"{i}\"/></ext></extLst></bk>");
359    }
360    xml.push_str("</futureMetadata>");
361    if dynamic {
362        xml.push_str("<cellMetadata count=\"1\"><bk><rc t=\"1\" v=\"0\"/></bk></cellMetadata>");
363    }
364    let _ = write!(xml, "<valueMetadata count=\"{count}\">");
365    for i in 0..count {
366        let _ = write!(xml, "<bk><rc t=\"{type_id}\" v=\"{i}\"/></bk>");
367    }
368    xml.push_str("</valueMetadata></metadata>");
369    xml
