1//! The Python part ([MS-XLSX] 2.1.23, 2.3.10): where a workbook keeps the code of its Python cells.
2//!
3//! A Python cell's formula is `_xlfn._xlws.PY(index, return_type, …)` ([MS-XLSX] 2.3.10.2): the
4//! first argument is the zero-based index of a `pythonScript` element in this part, whose `code`
5//! is the cell's Python code, and the second is 0 (an Excel value) or 1 (a Python object). Further
6//! arguments are references the code names as `%Px%`, `x` being the argument's zero-based index
7//! ([MS-XLSX] 2.3.10.4). The part's root is `python` ([MS-XLSX] 2.4.102, CT_Python 2.6.249), with
8//! `pythonScripts` (2.6.253) holding the `pythonScript` elements (2.6.252).
9//!
10//! GridCraft writes one script per Python cell, in the order the cells are written, with the code
11//! as it is (its `xl()` references stay text in the code, so no further arguments). Reading
12//! resolves the index and puts each `%Px%` back as the text of argument `x`. A formula that names
13//! no script the part has keeps only its cached value. Hostile input is bounded: the part's size,
14//! the number of scripts, each script's length, and the code all cells take from the part
15//! together.
16

25/// Namespace of the part.
26pub const NS: &str = "http://schemas.microsoft.com/office/spreadsheetml/2023/python";
27/// Relationship type from the workbook part.
28pub const REL: &str = "http://schemas.microsoft.com/office/2023/09/relationships/Python";
29/// Last segment of [`REL`], as [`crate::xlsx::package::Rel::kind`] holds it.
30const REL_KIND: &str = "Python";
31/// Content type of the part.
32pub const CONTENT_TYPE: &str = "application/vnd.ms-excel.python+xml";
33/// Part name we write (the format names none; the relationship finds it).
34pub const PART: &str = "xl/python.xml";

36/// Largest Python part read.
37const MAX_PART_BYTES: u64 = 64 * 1024 * 1024;
38/// Most scripts read from the part.
39const MAX_SCRIPTS: usize = 1_000_000;
40/// Most characters kept of one script: one more than a cell's code may have, so a longer script
41/// still fails [`PyCode::fits`] and its cell keeps only its value, with the usual warning.
42const MAX_SCRIPT_CHARS: usize = MAX_FORMULA_CHARS + 1;
43/// Most bytes of code all the cells of a workbook take from the part together. Several formulas
44/// may name one script, so without it a small part could fill memory.
45pub const MAX_CODE_BYTES: usize = 64 * 1024 * 1024;
46
47/// The scripts read from a workbook's Python part, and what the cells took from them.
48#[derive(Default)]
49pub struct ScriptsIn {
50    scripts: Vec<String>,
51    /// Bytes of code the cells may still take ([`MAX_CODE_BYTES`] in all).
52    budget: usize,
53    /// Python cells that kept only their value: no such script, or over the budget.
54    missing: usize,
55}
56
57impl ScriptsIn {
58    /// The warning for the Python cells that kept only their value, if any.
59    pub fn warning(&self) -> Option<String> {
60        (self.missing > 0).then(|| {
61            format!(
62                "{} Python cell(s) name code the file's Python part doesn't hold (or holds too much of); they kept only their values",
63                self.missing
64            )
65        })
66    }
67}

69/// Reads the workbook's Python part, if it has one. A damaged part is a warning: its cells keep
70/// their values.
71pub fn read(cx: &mut Ctx<'_>, rels: &[crate::xlsx::package::Rel]) -> Result<ScriptsIn, IoError> {
72    let mut out = ScriptsIn { budget: MAX_CODE_BYTES, ..ScriptsIn::default() };
73    let Some(rel) = rels.iter().find(|r| !r.external && r.kind.eq_ignore_ascii_case(REL_KIND)) else { return Ok(out) };
74    let bytes = match cx.pkg.read_optional_limited(&rel.target, MAX_PART_BYTES) {
75        Ok(Some(b)) => b,
76        Ok(None) => {
77            cx.warn(format!("missing part {}", rel.target));
78            return Ok(out);
79        }
80        Err(IoError::TooLarge(m)) => {
81            cx.warn(format!("the Python part is too large ({m}); Python cells kept only their values"));
82            return Ok(out);
83        }
84        // A damaged entry (a bad checksum or compressed stream) is a damaged part: its cells keep
85        // their values, as when the part is missing.
86        Err(e) => {
87            cx.warn(format!("the Python part is damaged ({e}); Python cells kept only their values"));
88            return Ok(out);
89        }
90    };
91    let mut over = 0usize;
92    let parsed = xml::parse_streaming(&bytes, &["pythonScript"], &mut |el: El| {
93        if out.scripts.len() < MAX_SCRIPTS {
94            out.scripts.push(script_code(&el));
95        } else {
96            over += 1;
97        }
98        Ok(())
99    });
100    match parsed {
101        Ok(root) if root.name != "python" => {
102            cx.warn(format!("the Python part has the root <{}>; Python cells kept only their values", root.name));
103            out.scripts.clear();
104        }
105        Ok(_) => {}
106        Err(e) => cx.warn(format!("the Python part is damaged ({e}); kept {} scripts", out.scripts.len())),
107    }
108    if over > 0 {
109        cx.warn(format!("the Python part holds more than {MAX_SCRIPTS} scripts; {over} were not read"));
110    }
111    Ok(out)
112}

114/// Encoded characters of a script that are decoded: enough for its first [`MAX_SCRIPT_CHARS`]
115/// characters of code, whatever its escapes. One character takes at most 14 encoded characters
116/// (a UTF-16 pair of `_xHHHH_` escapes) and a `\r\n` line break becomes one character, so
117/// `2 × (MAX_SCRIPT_CHARS + 1)` decoded characters cover the code kept, and two more escapes'
118/// worth keep the last of them, and the line break after it, from being cut.
119const MAX_ENCODED_CHARS: usize = 14 * (2 * (MAX_SCRIPT_CHARS + 1) + 2);
120
121/// A `pythonScript` element's code (ST_Xstring), with `\n` line breaks, cut after
122/// [`MAX_SCRIPT_CHARS`] characters. The limit is measured on the decoded code, so escapes in the
123/// file (`_x000D_`, `_x0009_`) don't cut a script that fits; decoding is linear and reads at most
124/// [`MAX_ENCODED_CHARS`] characters.
125fn script_code(el: &El) -> String {
126    let text = el.child("code").map(|c| c.text.as_str()).unwrap_or_default();
127    let encoded = match text.char_indices().nth(MAX_ENCODED_CHARS) {
128        Some((at, _)) => text.get(..at).unwrap_or_default(),
129        None => text,
130    };
131    let code = crate::model::python::normalize_code(&xml::decode_xstring(encoded));
132    match code.char_indices().nth(MAX_SCRIPT_CHARS) {
133        Some((at, _)) => code.get(..at).unwrap_or_default().to_string(),
134        None => code,
135    }
136}

138/// `PY(index, 0|1, params…)`, as a file's formula holds it: the index, whether the return type is
139/// a Python object, and the parameters. `None` for anything else (the code in a string literal
140/// is [`PyCode::from_formula`]'s form).
141fn py_call(expr: &Expr) -> Option<(usize, bool, &[Expr])> {
142    let Expr::Call(name, args) = expr else { return None };
143    let base = name.trim_start_matches("_XLFN.").trim_start_matches("_XLWS.");
144    if !base.eq_ignore_ascii_case("PY") {
145        return None;
146    }
147    let index = match args.first()? {
148        Expr::Number(n) if *n >= 0.0 && n.fract() == 0.0 && *n < MAX_SCRIPTS as f64 => *n as usize,
149        _ => return None,
150    };
151    let object = match args.get(1)? {
152        Expr::Number(n) if *n == 0.0 => false,
153        Expr::Number(n) if *n == 1.0 => true,
154        _ => return None,
155    };
156    Some((index, object, args.get(2..).unwrap_or_default()))
157}

159/// What a file formula is to Python.
160#[derive(Debug, PartialEq)]
161pub enum PyFormula {
162    /// Not a `PY(…)` call: an ordinary formula.
163    Not,
164    /// A Python cell with this code.
165    Code(PyCode),
166    /// A `PY(index, …)` call naming a script the part doesn't hold (or over the budget): the
167    /// cell keeps only its cached value, since no calculation can evaluate the formula.
168    Missing,
169}
170
171/// What a file formula makes: a Python cell from the Python part when it names a script, or from
172/// string literals ([`PyCode::from_formula`]). `formula` is the parsed formula, `raw` the
173/// formula's own text, which keeps line breaks in literal code.
174pub fn cell_code(scripts: &mut ScriptsIn, formula: &crate::model::Formula, raw: &str) -> PyFormula {
175    // The cheap test first: most formulas are not `PY(` calls.
176    if !crate::model::python::is_py_formula(&formula.text) {
177        return PyFormula::Not;
178    }
179    let expr = formula.expr_arc();
180    let Some((index, object, params)) = expr.as_deref().and_then(py_call) else {
181        return PyCode::from_formula(raw).map_or(PyFormula::Not, PyFormula::Code);
182    };
183    let code = scripts.scripts.get(index).map(|code| with_params(code, params));
184    match code.filter(|c| c.len() <= scripts.budget) {
185        Some(code) => {
186            scripts.budget -= code.len();
187            PyFormula::Code(PyCode { object, ..PyCode::new(&code) })
188        }
189        None => {
190            scripts.missing += 1;
191            PyFormula::Missing
192        }
193    }
194}

196/// `code` with every `%Px%` (`x` at least 2, naming one of `params`) replaced by the parameter's
197/// text. Stops growing past [`MAX_SCRIPT_CHARS`] characters: such code can't be a cell's.
198fn with_params(code: &str, params: &[Expr]) -> String {
199    if params.is_empty() || !code.contains("%P") {
200        return code.to_string();
201    }
202    let texts: Vec<String> = params.iter().map(crate::formula::print).collect();
203    let mut out = String::with_capacity(code.len());
204    let mut chars = 0usize;
205    let mut push = |out: &mut String, piece: &str| {
206        out.push_str(piece);
207        chars = chars.saturating_add(piece.chars().count());
208        chars > MAX_SCRIPT_CHARS
209    };
210    let mut rest = code;
211    while let Some(at) = rest.find("%P") {
212        if push(&mut out, rest.get(..at).unwrap_or_default()) {
213            return out;
214        }
215        let after = rest.get(at + 2..).unwrap_or_default();
216        let digits = after.bytes().take_while(u8::is_ascii_digit).count();
217        let param = (1..=6)
218            .contains(&digits)
219            .then(|| after.get(..digits).and_then(|d| d.parse::<usize>().ok()))
220            .flatten()
221            .filter(|_| after.as_bytes().get(digits) == Some(&b'%'))
222            .and_then(|x| x.checked_sub(2))
223            .and_then(|x| texts.get(x));
224        let full = match param {
225            Some(text) => {
226                rest = after.get(digits + 1..).unwrap_or_default();
227                push(&mut out, text)
228            }
229            None => {
230                rest = after;
231                push(&mut out, "%P")
232            }
233        };
234        if full {
235            return out;
236        }
237    }
238    out.push_str(rest);
239    out
240}

242/// The formula a Python cell is saved as: `_xlfn._xlws.PY(index,0|1)`, its code being script
243/// `index` of the part ([`write`]).
244pub fn file_formula(index: usize, object: bool) -> String {
245    format!("{FILE_PREFIX}PY({index},{})", u8::from(object))
246}

248/// The Python part holding `scripts`, in order. Only written when there is at least one: the
249/// format requires one `pythonScript` in `pythonScripts`.
250pub fn write(scripts: &[String]) -> String {
251    let mut xml = format!("{}<python xmlns=\"{NS}\"><pythonScripts>", crate::xlsx::write::XML_DECL);
252    for code in scripts {
253        xml.push_str("<pythonScript><code>");
254        xml.push_str(&xml::esc(&xml::encode_xstring(code)));
255        xml.push_str("</code></pythonScript>");
256    }
257    xml.push_str("</pythonScripts></python>");
258    xml
259}

273    #[test]
274    fn reads_the_index_form_and_parameters() {
275        let mut s = scripts(&["1 + 1", "df = xl(\"%P2%\", headers=True)\nxl('%P3%') + %P9% + %P1%"]);
276        let code = |s: &mut ScriptsIn, text: &str| match cell_code(s, &formula(text), text) {
277            PyFormula::Code(py) => Some(py),
278            _ => None,
279        };
280        let a = code(&mut s, "_xlfn._xlws.PY(0,1)").unwrap();
281        assert_eq!((a.code.as_str(), a.object), ("1 + 1", true));
282        let b = code(&mut s, "_xlfn._xlws.PY(1,0,A1:B3,Sheet2!$C$4)").unwrap();
283        assert_eq!(b.code, "df = xl(\"A1:B3\", headers=True)\nxl('Sheet2!$C$4') + %P9% + %P1%");
284        assert!(!b.object);
285        // The literal form still reads (other applications, older files).
286        assert_eq!(code(&mut s, "_xlfn._xlws.PY(\"x = 1\",0)").unwrap().code, "x = 1");
287        // A script the part doesn't hold: the cell keeps its value, and is counted for the warning.
288        let gone = "_xlfn._xlws.PY(7,0)";
289        assert_eq!(cell_code(&mut s, &formula(gone), gone), PyFormula::Missing);
290        assert!(s.warning().is_some());
291        for not in ["_xlfn._xlws.PY(-1,0)", "_xlfn._xlws.PY(0.5,0)", "_xlfn._xlws.PY(0,2)", "PY(A1,0)", "SUM(0,1)", "PYTHON_TYPE(A1)"] {
292            assert_eq!(cell_code(&mut s, &formula(not), not), PyFormula::Not, "{not}");
293        }
294    }

296    #[test]
297    fn hostile_scripts_are_bounded() {
298        // Many formulas naming one script take from one budget.
299        let code = "x".repeat(1_000);
300        let mut s = ScriptsIn { scripts: vec![code], budget: 2_500, missing: 0 };
301        let f = formula("_xlfn._xlws.PY(0,0)");
302        let taken = (0..5).filter(|_| matches!(cell_code(&mut s, &f, ""), PyFormula::Code(_))).count();
303        assert_eq!(taken, 2);
304        assert_eq!(s.missing, 3);
305        // A parameter repeated many times stops growing the code just past a cell's limit.
306        let many = "%P2%".repeat(5_000);
307        let out = with_params(&many, &[Expr::Name("A_VERY_LONG_NAME_FOR_A_RANGE".into())]);
308        assert!((MAX_SCRIPT_CHARS..MAX_SCRIPT_CHARS + 40).contains(&out.chars().count()), "{}", out.len());
309        assert!(!PyCode::new(&out).fits());
310        // Odd tokens stay as they are.
311        assert_eq!(with_params("%P%P2%%P99999999%%P", &[Expr::Number(1.0)]), "%P1%P99999999%%P");
312        // A long script is cut just past the limit, so it fails the cell's check.
313        let el = El {
314            name: "pythonScript".into(),
315            children: vec![El { name: "code".into(), text: "y".repeat(20_000), ..El::default() }],
316            ..El::default()
317        };
318        assert_eq!(script_code(&el).chars().count(), MAX_SCRIPT_CHARS);
319    }
