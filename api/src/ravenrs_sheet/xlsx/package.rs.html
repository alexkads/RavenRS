1//! OPC package access through `craft-opc`: zip parts read against GridCraft's limits,
2//! relationships with GridCraft's `Rel`, part-name resolution, and a deterministic zip writer.
3//! Parts are parsed with this crate's XML reader.
4
5use crate::xlsx::IoError;
6use crate::xlsx::xml::{self, El};
7
8pub use crate::craft::opc::{normalize, resolve, split_dir};
9
10/// Largest decompressed part accepted.
11pub const MAX_PART: u64 = 512 * 1024 * 1024;
12/// Largest total decompressed size read from one package.
13pub const MAX_TOTAL: u64 = 1024 * 1024 * 1024;
14/// Most entries accepted in the zip directory.
15pub const MAX_ENTRIES: usize = 200_000;
16
17const LIMITS: crate::craft::opc::Limits = crate::craft::opc::Limits::new(MAX_ENTRIES, MAX_PART, MAX_TOTAL);
18
19pub struct Package<'a> {
20    opc: crate::craft::opc::LazyPackage<'a>,
21}
22
23/// A relationship.
24#[derive(Clone, Debug)]
25pub struct Rel {
26    pub id: String,
27    /// Last path segment of the relationship type (`worksheet`, `styles`, …): the same for the
28    /// transitional and strict namespaces.
29    pub kind: String,
30    /// Resolved part name (no leading `/`), or the raw target for external relationships.
31    pub target: String,
32    pub external: bool,
33}
34
35/// GridCraft's wording of a package error.
36fn error(e: crate::craft::opc::Error) -> IoError {
37    use crate::craft::opc::Error as E;
38    match e {
39        E::Zip(m) | E::Write(m) => IoError::Zip(m),
40        E::TooManyEntries { declared, .. } => IoError::TooLarge(format!("{declared} zip entries")),
41        E::PartTooLarge { part, limit } => IoError::TooLarge(format!("{part} expands to more than {limit} bytes")),
42        E::PackageTooLarge { .. } => IoError::TooLarge("package expands to more than 1 GB".into()),
43        E::Xml { part, error } => IoError::Xml(format!("{part}: {error}")),
44        e if e.is_limit() => IoError::TooLarge(e.to_string()),
45        e => IoError::Zip(e.to_string()),
46    }
47}
48
49impl<'a> Package<'a> {
50    pub fn open(bytes: &'a [u8]) -> Result<Package<'a>, IoError> {
51        Ok(Package { opc: crate::craft::opc::LazyPackage::open(bytes, &LIMITS).map_err(error)? })
52    }
53
54    /// Every part's (normalized) name, sorted.
55    #[cfg(test)]
56    pub fn names(&self) -> Vec<String> {
57        self.opc.names().into_iter().map(str::to_string).collect()
58    }
59
60    pub fn has(&self, part: &str) -> bool {
61        self.opc.has(part)
62    }
63
64    /// Reads a part's bytes; `Ok(None)` when it doesn't exist.
65    pub fn read(&mut self, part: &str) -> Result<Option<Vec<u8>>, IoError> {
66        self.opc.read(part).map_err(error)
67    }
68
69    /// Reads a part the reader can do without (the caller warns and goes on when it fails). A read
70    /// that fails partway is charged this part's whole allowance instead of ending the package's.
71    pub fn read_optional_limited(&mut self, part: &str, maximum: u64) -> Result<Option<Vec<u8>>, IoError> {
72        self.opc.read_optional_limited(part, maximum).map_err(error)
73    }
74
75    /// Reads a small metadata part without allowing it to expand to the general part limit.
76    pub fn read_limited(&mut self, part: &str, maximum: u64) -> Result<Option<Vec<u8>>, IoError> {
77        self.opc.read_limited(part, maximum).map_err(error)
78    }
79
80    pub fn read_xml(&mut self, part: &str) -> Result<Option<El>, IoError> {
81        match self.read(part)? {
82            Some(b) => xml::parse(&b).map(Some).map_err(|e| prefix(part, e)),
83            None => Ok(None),
84        }
85    }
86
87    /// Relationships of a part (`xl/workbook.xml` → `xl/_rels/workbook.xml.rels`). Missing or
88    /// broken rels parts give an empty list.
89    pub fn rels(&mut self, part: &str) -> Result<Vec<Rel>, IoError> {
90        let rels = self.opc.rels(part).map_err(error)?;
91        Ok(rels
92            .list
93            .into_iter()
94            .map(|r| {
95                let kind = r.kind().to_string();
96                Rel { id: r.id, kind, target: r.target, external: r.external }
97            })
98            .collect())
99    }
100}
101
102/// The first part whose declared unpacked size is over [`MAX_PART`], with that size.
103pub fn oversized_part(bytes: &[u8]) -> Option<(String, u64)> {
104    crate::craft::opc::declared_oversized(bytes, &LIMITS)
105}
106
107fn prefix(part: &str, e: IoError) -> IoError {
108    match e {
109        IoError::Xml(m) => IoError::Xml(format!("{part}: {m}")),
110        o => o,
111    }
112}

116/// A zip package being written: every entry deflated, with one fixed timestamp.
117pub struct ZipOut {
118    zip: crate::craft::opc::ZipOut,
119}
120
121impl ZipOut {
122    pub fn new() -> ZipOut {
123        ZipOut { zip: crate::craft::opc::ZipOut::new() }
124    }
125    pub fn add(&mut self, name: &str, data: &[u8]) -> Result<(), IoError> {
126        self.zip.add(name, data, crate::craft::opc::Compression::Deflate).map_err(error)
127    }
128    pub fn finish(self) -> Result<Vec<u8>, IoError> {
129        self.zip.finish().map_err(error)
130    }
131}

133#[cfg(test)]
134mod tests {

137    #[test]
138    fn resolving() {
139        assert_eq!(resolve("xl", "worksheets/sheet1.xml"), "xl/worksheets/sheet1.xml");
140        assert_eq!(resolve("xl/worksheets", "../drawings/d1.xml"), "xl/drawings/d1.xml");
141        assert_eq!(resolve("xl/worksheets", "/xl/tables/t.xml"), "xl/tables/t.xml");
142        assert_eq!(resolve("", "xl/workbook.xml"), "xl/workbook.xml");
143        assert_eq!(resolve("a", "../../../x"), "x");
144        assert_eq!(split_dir("xl/workbook.xml"), ("xl", "workbook.xml"));
145        assert_eq!(split_dir("x.xml"), ("", "x.xml"));
146    }
