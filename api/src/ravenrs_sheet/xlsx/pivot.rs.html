1//! PivotTables: pivot cache definitions and records (ECMA-376 Part 1 §18.10.1) and PivotTable
2//! definitions (§18.10.1.73), written from the model's [`PivotTable`]s and read back into them.
3//!
4//! Writing: one cache per PivotTable, with a records snapshot of the source list (omitted, with
5//! `saveData="0"`, for very large sources) and `refreshOnLoad="1"` so a reader recomputes the
6//! report from the source on open. Fields used on an axis get shared items (or, for date
7//! grouping, `fieldGroup` group items) so the table's `<items>` can refer to them. The report
8//! cells the engine wrote stay on the sheet as plain cells.
9//!
10//! Reading: the cache's source and field names plus the table's areas, functions, show-values-as,
11//! layout flags, grand totals and style become a model [`PivotTable`]. Anything we don't model
12//! (OLAP sources, calculated fields, numeric grouping…) becomes a warning.

56/// The pivot item label of a value (blank shows as "(blank)").
57fn label(v: &Value) -> String {
58    match v {
59        Value::Empty => "(blank)".into(),
60        Value::Number(n) => number_to_text(*n),
61        Value::Text(t) => t.to_string(),
62        Value::Bool(b) => if *b { "TRUE" } else { "FALSE" }.into(),
63        Value::Error(e) => e.as_str().into(),
64        Value::Array(_) => label(&v.scalar()),
65    }
66}

141/// `2024-03-05T00:00:00` for a date serial.
142fn iso(sys: DateSystem, serial: f64) -> Option<String> {
143    let d = datetime_from_serial(sys, serial)?;
144    Some(format!("{:04}-{:02}-{:02}T{:02}:{:02}:{:02}", d.year, d.month, d.day, d.hour, d.minute, d.second))
145}
146
147/// `3/5/2024` for a date serial (group boundary captions).
148fn mdy(sys: DateSystem, serial: f64) -> String {
149    match datetime_from_serial(sys, serial) {
150        Some(d) => format!("{}/{}/{}", d.month, d.day, d.year),
151        None => number_to_text(serial),
152    }
153}

157/// Identity of a shared item (text items match case-insensitively, as in Excel).
158#[derive(Clone, PartialEq, Eq, Hash)]
159enum Key {
160    Blank,
161    Num(u64),
162    Text(String),
163    Bool(bool),
164    Err(&'static str),
165}
166
167fn key(v: &Value) -> Key {
168    match v {
169        Value::Empty => Key::Blank,
170        Value::Number(n) => Key::Num(if *n == 0.0 { 0 } else { n.to_bits() }),
171        Value::Text(t) => Key::Text(t.to_lowercase()),
172        Value::Bool(b) => Key::Bool(*b),
173        Value::Error(e) => Key::Err(e.as_str()),
174        Value::Array(_) => key(&v.scalar()),
175    }
176}
