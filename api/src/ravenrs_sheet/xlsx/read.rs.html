16/// State shared while reading one package.
17pub struct Ctx<'a> {
18    pub pkg: Package<'a>,
19    pub report: ReadReport,
20    pub sst: Vec<Arc<str>>,
21    /// Phonetic (furigana) data of the shared strings, one entry per base text.
22    pub phonetics: Vec<PhoneticString>,
23    /// Cell format index → model style.
24    pub xf_map: Vec<StyleId>,
25    pub dxfs: Vec<Style>,
26    pub palette: Vec<u32>,
27    pub date_system: DateSystem,
28    pub styles: StyleTable,
29    /// Threaded-comment person id → display name.
30    pub persons: std::collections::HashMap<String, String>,
31    pub next_id: u32,
32    /// Custom number formats from the styles part, by id.
33    pub num_fmts: std::collections::HashMap<u32, String>,
34    /// Value metadata blocks, indexed by the worksheet's one-based `vm` attribute.
35    pub cell_pictures: Vec<Option<Arc<CellPicture>>>,
36    /// x14 conditional-format ids of the current sheet's data bars, with the index of the rule in `cond_formats`.
37    pub cf_ext_ids: Vec<(String, usize)>,
38    /// Bytes charged so far for shared formulas expanded into dependent cells (see
39    /// [`crate::xlsx::sheet_read::MAX_SHARED_FORMULA_BYTES`]).
40    pub shared_formula_bytes: u64,
41    /// Whether GridCraft calculates a function, when the caller says: cell formulas wrapped by
42    /// Google Sheets' export are then read as the formula inside (see [`crate::xlsx::read_xlsx_with`]).
43    pub known_function: Option<&'a dyn Fn(&str) -> bool>,
44    /// The code of the workbook's Python cells, from its Python part.
45    pub python: crate::xlsx::python_part::ScriptsIn,
46}

48impl Ctx<'_> {
49    pub fn warn(&mut self, m: impl Into<String>) {
50        self.report.warn(m);
51    }
52    /// Stores the phonetic data read from `xl/sharedStrings.xml`.
53    ///
54    /// One entry per base text: the writer rebuilds the shared string table from the values the
55    /// sheets use and holds each text once (the format's `uniqueCount`), so only one reading can be
56    /// written per text. A later `<si>` with a text that is already there is dropped — with a
57    /// warning when its data differs, so nothing is lost silently.
58    pub fn keep_phonetics(&mut self, list: Vec<PhoneticString>) {
59        let mut kept: Vec<PhoneticString> = Vec::new();
60        let mut index: std::collections::HashMap<String, usize> = std::collections::HashMap::new();
61        let mut conflicts = 0usize;
62        for p in list {
63            match index.get(p.text.as_str()) {
64                Some(&i) => {
65                    if kept.get(i).is_some_and(|k| *k != p) {
66                        conflicts += 1;
67                    }
68                }
69                None => {
70                    index.insert(p.text.clone(), kept.len());
71                    kept.push(p);
72                }
73            }
74        }
75        if conflicts > 0 {
76            self.warn(format!("shared strings: {conflicts} phonetic readings were dropped because another <si> with the same text already had one"));
77        }
78        self.phonetics = kept;
79    }
80    /// A fresh id for a chart, picture, shape or table.
81    pub fn next_object_id(&mut self) -> u32 {
82        let id = self.next_id;
83        self.next_id = self.next_id.saturating_add(1);
84        id
85    }
86    pub fn style(&self, xf: Option<u32>) -> StyleId {
87        xf.and_then(|i| self.xf_map.get(i as usize).copied()).unwrap_or_default()
88    }
89    /// Reads and parses an optional part: problems other than size limits become warnings.
90    pub fn optional_xml(&mut self, part: &str) -> Result<Option<El>, IoError> {
91        match self.pkg.read_xml(part) {
92            Ok(Some(e)) => Ok(Some(e)),
93            Ok(None) => {
94                self.warn(format!("missing part {part}"));
95                Ok(None)
96            }
97            Err(IoError::TooLarge(m)) => Err(IoError::TooLarge(m)),
98            Err(e) => {
99                self.warn(format!("skipped {part}: {e}"));
100                Ok(None)
101            }
102        }
103    }
104}

110fn read_shared_strings(cx: &mut Ctx<'_>, wb_rels: &[Rel]) -> Result<(), IoError> {
111    // Shared strings, streamed item by item. Phonetic (furigana) runs are kept next to the text
112    // they annotate, so an open + save does not delete them.
113    if let Some(r) = find_rel(wb_rels, "sharedStrings").cloned() {
114        match cx.pkg.read(&r.target)? {
115            Some(b) => {
116                let mut sst = Vec::new();
117                let mut phonetics = Vec::new();
118                let mut dropped = 0usize;
119                let res = xml::parse_streaming(&b, &["si"], &mut |si| {
120                    let text = xml::rich_text(&si);
121                    let raw = si.kids("rPh").count();
122                    let (runs, props) = xml::phonetic_data(&si, &text);
123                    dropped += raw.saturating_sub(runs.len());
124                    if !runs.is_empty() || props.is_some() {
125                        phonetics.push(PhoneticString { text: text.clone(), runs, props });
126                    }
127                    sst.push(Arc::from(text));
128                    Ok(())
129                });
130                if let Err(e) = res {
131                    cx.warn(format!("shared strings are damaged ({e}); kept {} strings", sst.len()));
132                }
133                if dropped > 0 {
134                    cx.warn(format!("shared strings: {dropped} phonetic runs were dropped (bad sb/eb or no reading)"));
135                }
136                cx.sst = sst;
137                cx.keep_phonetics(phonetics);
138            }
139            None => cx.warn(format!("missing part {}", r.target)),
140        }
141    }
142
143    Ok(())
144}

146fn read_workbook_styles(cx: &mut Ctx<'_>, wb: &mut Workbook, wb_rels: &[Rel]) -> Result<(), IoError> {
147    // Theme, then styles: a font's link to the theme's body font is read against it.
148    if let Some(r) = find_rel(wb_rels, "theme").cloned()
149        && let Some(x) = cx.optional_xml(&r.target)?
150    {
151        wb.theme = crate::xlsx::theme::read_theme(&x);
152    }
153    // Cell checkboxes live in the feature property bags, which styles refer to.
154    let mut checkboxes = Vec::new();
155    if let Some(r) = find_rel(wb_rels, "FeaturePropertyBag").cloned()
156        && let Some(x) = cx.optional_xml(&r.target)?
157    {
158        checkboxes = crate::xlsx::feature_bags::checkbox_complements(&x);
159    }
160    if let Some(r) = find_rel(wb_rels, "styles").cloned() {
161        if let Some(x) = cx.optional_xml(&r.target)? {
162            let StylesIn { xfs, dxfs, palette, named, num_fmts } = read_styles(&x, &checkboxes, &wb.theme.minor_font);
163            cx.num_fmts = num_fmts;
164            // Cell format 0 is the Normal cell format: everything on it (font, number format,
165            // fill, borders, alignment, protection) is what a cell without its own format shows
166            // (#1013).
167            if let Some(normal) = xfs.first() {
168                cx.styles = StyleTable::with_default(normal.clone());
169            }
170            cx.xf_map = xfs.into_iter().enumerate().map(|(i, s)| if i == 0 { StyleId::DEFAULT } else { cx.styles.intern(s) }).collect();
171            cx.dxfs = dxfs;
172            cx.palette = palette;
173            wb.cell_styles = named;
174        }
175    } else {
176        cx.warn("no styles part");
177    }
178    Ok(())
179}

181/// Read the workbook XML selected by the package's root relationships.
182fn workbook_xml(pkg: &mut Package<'_>, part: &str) -> Result<El, IoError> {
183    let xml = match pkg.read_xml(part)? {
184        Some(xml) => xml,
185        None => {
186            if pkg.has("xl/workbook.xml") && part != "xl/workbook.xml" {
187                pkg.read_xml("xl/workbook.xml")?.ok_or_else(|| IoError::Format("no workbook part".into()))?
188            } else if pkg.has("content.xml") {
189                return Err(IoError::Format("this is an OpenDocument file, not XLSX".into()));
190            } else {
191                return Err(IoError::Format("no workbook part".into()));
192            }
193        }
194    };
195    if xml.name != "workbook" {
196        return Err(IoError::Format(format!("unexpected root element <{}> in workbook part", xml.name)));
197    }
198    Ok(xml)
199}

201/// Read workbook-level settings before processing the individual sheets.
202fn read_workbook_properties(cx: &mut Ctx<'_>, wb: &mut Workbook, xml: &El) {
203    if let Some(p) = xml.child("workbookPr") {
204        if p.flag("date1904", false) {
205            cx.date_system = DateSystem::D1904;
206        }
207        wb.remove_personal_info = p.flag("filterPrivacy", false);
208        wb.show_objects = match p.attr("showObjects") {
209            Some("none") => ShowObjects::None,
210            Some("placeholders") => ShowObjects::Placeholders,
211            _ => ShowObjects::All,
212        };
213    }
214    wb.date_system = cx.date_system;
215    if let Some(c) = xml.child("calcPr") {
216        wb.calc.mode = match c.attr("calcMode") {
217            Some("manual") => CalcMode::Manual,
218            Some("autoNoTable") => CalcMode::AutomaticExceptTables,
219            _ => CalcMode::Automatic,
220        };
221        wb.calc.iterative = c.flag("iterate", false);
222        if let Some(n) = c.attr_u32("iterateCount") {
223            wb.calc.set_max_iterations(n);
224        }
225        if let Some(d) = c.attr_f64("iterateDelta") {
226            wb.calc.set_max_change(d);
227        }
228        wb.calc.precision_as_displayed = !c.flag("fullPrecision", true);
229        wb.calc.r1c1 = c.attr("refMode") == Some("R1C1");
230    }
231    if let Some(p) = xml.child("workbookProtection") {
232        wb.protected_structure = p.flag("lockStructure", false);
233        wb.structure_password = if wb.protected_structure { crate::xlsx::sheet_read::read_password(p, "workbook") } else { None };
234    }
235}

237/// The single window modelled by GridCraft: prefer the first visible window,
238/// including the format's default visibility, and otherwise retain the first one.
239fn selected_workbook_view(workbook: &El) -> Option<&El> {
240    let views = workbook.child("bookViews")?;
241    views.kids("workbookView").find(|view| view.attr("visibility").unwrap_or("visible") == "visible").or_else(|| views.child("workbookView"))
242}

244pub fn read_xlsx<'a>(bytes: &'a [u8], known_function: Option<&'a dyn Fn(&str) -> bool>) -> Result<(Workbook, ReadReport), IoError> {
245    let pkg = Package::open(bytes)?;
246    let mut cx = Ctx {
247        pkg,
248        report: ReadReport::default(),
249        sst: vec![],
250        phonetics: vec![],
251        xf_map: vec![],
252        dxfs: vec![],
253        palette: crate::xlsx::tables::INDEXED.to_vec(),
254        date_system: DateSystem::D1900,
255        styles: StyleTable::default(),
256        persons: Default::default(),
257        next_id: 1,
258        num_fmts: Default::default(),
259        cell_pictures: vec![],
260        cf_ext_ids: vec![],
261        shared_formula_bytes: 0,
262        known_function,
263        python: Default::default(),
264    };
265    let root_rels = cx.pkg.rels("")?;
266    let wb_part = find_rel(&root_rels, "officeDocument").map(|r| r.target.clone()).unwrap_or_else(|| "xl/workbook.xml".into());
267    let wb_xml = workbook_xml(&mut cx.pkg, &wb_part)?;
268    let wb_rels = cx.pkg.rels(&wb_part)?;
269    let mut kept = crate::xlsx::passthrough::KeptReader::new(&mut cx);
270    cx.cell_pictures = match crate::xlsx::richdata::read(&mut cx, &wb_rels) {
271        Ok(pictures) => pictures,
272        Err(error) => {
273            cx.warn(format!("cell-picture data could not be read ({error}); scalar fallbacks were retained"));
274            vec![]
275        }
276    };
277    // The code of Python cells, which their formulas name by index.
278    cx.python = crate::xlsx::python_part::read(&mut cx, &wb_rels)?;
279    let mut wb = Workbook::new();
280    wb.sheets.clear();
281
282    read_workbook_styles(&mut cx, &mut wb, &wb_rels)?;
283    read_shared_strings(&mut cx, &wb_rels)?;
284
285    if let Some(r) = find_rel(&wb_rels, "person").cloned()
286        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
287    {
288        for p in x.kids("person") {
289            if let (Some(id), Some(n)) = (p.attr("id"), p.attr("displayName")) {
290                cx.persons.insert(id.to_string(), n.to_string());
291            }
292        }
293    }
294
295    read_workbook_properties(&mut cx, &mut wb, &wb_xml);
296    let view_el = selected_workbook_view(&wb_xml);
297    let active_tab = view_el.and_then(|v| v.attr_u32("activeTab")).unwrap_or(0) as usize;
298    if let Some(v) = view_el {
299        // The window's display settings (Excel's "Display options for this workbook").
300        // Absent attributes take the format's defaults, which is also what a workbook
301        // without a <workbookView> gets.
302        wb.view = WorkbookView {
303            show_sheet_tabs: v.flag("showSheetTabs", true),
304            show_horizontal_scroll: v.flag("showHorizontalScroll", true),
305            show_vertical_scroll: v.flag("showVerticalScroll", true),
306            tab_ratio: v.attr_u32("tabRatio").unwrap_or(600).min(WorkbookView::MAX_TAB_RATIO),
307        };
308    }
309
310    // Sheets. `file_to_model[i]` maps the i-th <sheet> to a model sheet index.
311    let mut file_to_model: Vec<Option<usize>> = vec![];
312    let mut pivot_caches = crate::xlsx::pivot::CachesIn::new(&wb_xml, &wb_rels);
313    let sheet_els: Vec<El> = wb_xml.child("sheets").map(|s| s.kids("sheet").cloned().collect()).unwrap_or_default();
314    let mut names = SheetNames::default();
315    for (i, se) in sheet_els.iter().enumerate() {
316        let name = se.attr("name").unwrap_or("").to_string();
317        let rel = se.attr("id").and_then(|id| wb_rels.iter().find(|r| r.id == id)).cloned();
318        let Some(rel) = rel else {
319            cx.warn(format!("sheet \"{name}\" has no part; skipped"));
320            file_to_model.push(None);
321            continue;
322        };
323        if rel.kind == "chartsheet" {
324            // Kept as it is and written back in place (see `passthrough`); nothing to model.
325            file_to_model.push(None);
326            continue;
327        }
328        if rel.kind != "worksheet" {
329            cx.warn(format!("sheet \"{name}\" is a {} and was skipped", rel.kind));
330            file_to_model.push(None);
331            continue;
332        }
333        let name = names.unique(&name, i);
334        let (mut sheet, read_controls) = match crate::xlsx::sheet_read::read_sheet(&mut cx, &rel.target, &name) {
335            Ok(s) => s,
336            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
337            Err(e) => {
338                cx.warn(format!("sheet \"{name}\" could not be read: {e}"));
339                (Sheet::new(name.clone()), Default::default())
340            }
341        };
342        match crate::xlsx::pivot::read_pivots(&mut cx, &mut pivot_caches, &rel.target, &name) {
343            Ok(p) => sheet.pivots = p,
344            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
345            Err(e) => cx.warn(format!("PivotTables on sheet \"{name}\" could not be read: {e}")),
346        }
347        sheet.preserved = kept.sheet(&mut cx, &rel.target, &name, &read_controls);
348        sheet.file_id = se.attr_u32("sheetId").unwrap_or(0);
349        sheet.visibility = match se.attr("state") {
350            Some("hidden") => Visibility::Hidden,
351            Some("veryHidden") => Visibility::VeryHidden,
352            _ => Visibility::Visible,
353        };
354        file_to_model.push(Some(wb.sheets.len()));
355        wb.sheets.push(Arc::new(sheet));
356    }
357    if wb.sheets.is_empty() {
358        cx.warn("the workbook has no worksheets; added an empty sheet");
359        wb.sheets.push(Arc::new(Sheet::new("Sheet1")));
360    }
361    wb.set_active_sheet(file_to_model.get(active_tab).copied().flatten().unwrap_or(0));
362
363    // Defined names. Chartex parts refer to their data through hidden `_xlchart.*` names.
364    let mut chart_names: Vec<(String, String)> = vec![];
365    if let Some(dn) = wb_xml.child("definedNames") {
366        for d in dn.kids("definedName") {
367            let Some(name) = d.attr("name") else { continue };
368            let scope = match d.attr_u32("localSheetId") {
369                Some(i) => match file_to_model.get(i as usize).copied().flatten() {
370                    Some(m) => Some(m),
371                    None => continue,
372                },
373                None => None,
374            };
375            let text = d.text.trim();
376            let lname = name.to_ascii_lowercase();
377            if lname == "_xlnm._filterdatabase" {
378                continue;
379            }
380            if lname.starts_with("_xlchart.") {
381                chart_names.push((lname, crate::xlsx::fmla::from_file(&crate::xlsx::chart::chart_ref_from_file(text))));
382                continue;
383            }
384            if lname == "_xlnm.print_area" || lname == "_xlnm.print_titles" {
385                if let Some(si) = scope
386                    && let Some(sheet) = wb.sheet_mut(si)
387                {
388                    if lname == "_xlnm.print_area" {
389                        sheet.print.print_area = parse_print_area(text);
390                    } else {
391                        let (rows, cols) = parse_print_titles(text);
392                        sheet.print.title_rows = rows;
393                        sheet.print.title_cols = cols;
394                    }
395                }
396                continue;
397            }
398            wb.names.push(DefinedName {
399                name: name.to_string(),
400                scope,
401                formula: crate::xlsx::fmla::from_file(text),
402                comment: d.attr("comment").unwrap_or("").to_string(),
403                hidden: d.flag("hidden", false),
404            });
405        }
406    }
407
408    if !chart_names.is_empty() {
409        let resolve = |f: &mut String| {
410            if let Some((_, r)) = chart_names.iter().find(|(n, _)| f.eq_ignore_ascii_case(n)) {
411                *f = r.clone();
412            }
413        };
414        for sheet in wb.sheets.iter_mut().filter(|s| !s.charts.is_empty()) {
415            for ch in Arc::make_mut(sheet).charts.iter_mut() {
416                for s in ch.series.iter_mut() {
417                    resolve(&mut s.values);
418                    s.categories.iter_mut().for_each(&resolve);
419                    s.name.iter_mut().for_each(&resolve);
420                }
421            }
422        }
423    }
424
425    // Document properties.
426    if let Some(r) = find_rel(&root_rels, "core-properties").cloned()
427        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
428    {
429        let get = |n: &str| x.child(n).map(|e| e.text.clone()).unwrap_or_default();
430        wb.props.title = get("title");
431        wb.props.subject = get("subject");
432        wb.props.author = get("creator");
433        wb.props.keywords = get("keywords");
434        wb.props.description = get("description");
435    }
436    if let Some(r) = find_rel(&root_rels, "extended-properties").cloned()
437        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
438    {
439        wb.props.company = x.child("Company").map(|e| e.text.clone()).unwrap_or_default();
440    }
441
442    // A VBA project is carried through unchanged: GridCraft never runs macros, but preserving the
443    // binary keeps macro-enabled workbooks from being silently stripped on save.
444    if let Some(r) = find_rel(&wb_rels, "vbaProject").cloned() {
445        match cx.pkg.read(&r.target) {
446            Ok(Some(data)) => {
447                wb.vba_project = Some(data.into());
448                cx.warn("macros (VBA project) were kept but not executed");
449            }
450            Ok(None) => cx.warn("macros (VBA project) reference a missing part and were dropped"),
451            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
452            Err(e) => cx.warn(format!("macros (VBA project) could not be read and were dropped: {e}")),
453        }
454    }
455    wb.preserved = kept.workbook(&mut cx, &wb_part, &wb_xml, &file_to_model);
456    if let Some(w) = cx.python.warning() {
457        cx.warn(w);
458    }
459    // Calls of LAMBDA names are spelled like the names (the parser upper-cases them).
460    wb.apply_name_call_case();
461    wb.styles = cx.styles;
462    wb.phonetics = cx.phonetics;
463    Ok((wb, cx.report))
464}

466/// The sheet names an import has assigned so far, for repairing repeated ones.
467#[derive(Default)]
468pub(crate) struct SheetNames {
469    /// Assigned names, ASCII-lowercased.
470    taken: HashSet<String>,
471    /// For each lowercased stem, the next suffix to try: a file full of repeated names would
472    /// otherwise retry every occupied suffix for every sheet.
473    next: HashMap<String, u64>,
474}
475
476impl SheetNames {
477    pub(crate) fn of(wb: &Workbook) -> Self {
478        SheetNames { taken: wb.sheets.iter().map(|s| s.name.to_ascii_lowercase()).collect(), next: HashMap::new() }
479    }
480
481    /// A valid name for the i-th sheet, unique among the names assigned so far, which it joins.
482    pub(crate) fn unique(&mut self, name: &str, i: usize) -> String {
483        let mut base: String = name.chars().filter(|c| !matches!(c, ':' | '\\' | '/' | '?' | '*' | '[' | ']')).take(31).collect();
484        if base.trim().is_empty() {
485            base = format!("Sheet{}", i + 1);
486        }
487        let name = if self.taken.contains(&base.to_ascii_lowercase()) {
488            let stem: String = base.chars().take(25).collect();
489            let next = self.next.entry(stem.to_ascii_lowercase()).or_insert(2);
490            let found = (*next..).find(|n| !self.taken.contains(&format!("{stem} ({n})").to_ascii_lowercase()));
491            match found {
492                Some(n) => {
493                    *next = n + 1;
494                    format!("{stem} ({n})")
495                }
496                None => base,
497            }
498        } else {
499            base
500        };
501        self.taken.insert(name.to_ascii_lowercase());
502        name
503    }
504}

506/// Strips a sheet qualifier (`'My Sheet'!$A$1` → `$A$1`).
507fn strip_sheet(s: &str) -> &str {
508    match s.rfind('!') {
509        Some(i) => s.get(i + 1..).unwrap_or(""),
510        None => s,
511    }
512}

514pub fn parse_print_area(text: &str) -> Option<RangeRef> {
515    let first = text.split(',').next()?;
516    let r = strip_sheet(first.trim()).replace('$', "");
517    RangeRef::parse(&r)
518}

520/// A print-titles span: the first and last row (or column) repeated on each page.
521type TitleSpan = Option<(u32, u32)>;
522
523pub fn parse_print_titles(text: &str) -> (TitleSpan, TitleSpan) {
524    let mut rows = None;
525    let mut cols = None;
526    for part in text.split(',') {
527        let p = strip_sheet(part.trim()).replace('$', "");
528        let Some((a, b)) = p.split_once(':') else { continue };
529        if let (Ok(r0), Ok(r1)) = (a.parse::<u32>(), b.parse::<u32>()) {
530            if r0 >= 1 && r1 >= r0 && r1 <= crate::base::MAX_ROWS {
531                rows = Some((r0 - 1, r1 - 1));
532            }
533        } else if let (Some(c0), Some(c1)) = (crate::base::letters_to_col(a), crate::base::letters_to_col(b))
534            && c1 >= c0
535        {
536            cols = Some((c0, c1));
537        }
538    }
539    (rows, cols)
540}
