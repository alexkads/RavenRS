16/// State shared while reading one package.
17pub struct Ctx<'a> {
18    pub pkg: Package<'a>,
19    pub report: ReadReport,
20    pub sst: Vec<Arc<str>>,
21    /// Phonetic (furigana) data of the shared strings, one entry per base text.
22    pub phonetics: Vec<PhoneticString>,
23    /// Cell format index → model style.
24    pub xf_map: Vec<StyleId>,
25    pub dxfs: Vec<Style>,
26    pub palette: Vec<u32>,
27    pub date_system: DateSystem,
28    pub styles: StyleTable,
29    /// Threaded-comment person id → display name.
30    pub persons: std::collections::HashMap<String, String>,
31    pub next_id: u32,
32    /// Custom number formats from the styles part, by id.
33    pub num_fmts: std::collections::HashMap<u32, String>,
34    /// Value metadata blocks, indexed by the worksheet's one-based `vm` attribute.
35    pub cell_pictures: Vec<Option<Arc<CellPicture>>>,
36    /// x14 conditional-format ids of the current sheet's data bars, with the index of the rule in `cond_formats`.
37    pub cf_ext_ids: Vec<(String, usize)>,
38    /// Cells of the worksheet being read whose formulas lack dynamic-array metadata (legacy
39    /// formulas, evaluated with implicit intersection), collected per sheet.
40    pub legacy_formulas: Vec<crate::base::CellRef>,
41    /// Bytes charged so far for shared formulas expanded into dependent cells (see
42    /// [`crate::xlsx::sheet_read::MAX_SHARED_FORMULA_BYTES`]).
43    pub shared_formula_bytes: u64,
44    /// Whether GridCraft calculates a function, when the caller says: cell formulas wrapped by
45    /// Google Sheets' export are then read as the formula inside (see [`crate::xlsx::read_xlsx_with`]).
46    pub known_function: Option<&'a dyn Fn(&str) -> bool>,
47    /// The code of the workbook's Python cells, from its Python part.
48    pub python: crate::xlsx::python_part::ScriptsIn,
49}

51impl Ctx<'_> {
52    pub fn warn(&mut self, m: impl Into<String>) {
53        self.report.warn(m);
54    }
55    /// Stores the phonetic data read from `xl/sharedStrings.xml`.
56    ///
57    /// One entry per base text: the writer rebuilds the shared string table from the values the
58    /// sheets use and holds each text once (the format's `uniqueCount`), so only one reading can be
59    /// written per text. A later `<si>` with a text that is already there is dropped — with a
60    /// warning when its data differs, so nothing is lost silently.
61    pub fn keep_phonetics(&mut self, list: Vec<PhoneticString>) {
62        let mut kept: Vec<PhoneticString> = Vec::new();
63        let mut index: std::collections::HashMap<String, usize> = std::collections::HashMap::new();
64        let mut conflicts = 0usize;
65        for p in list {
66            match index.get(p.text.as_str()) {
67                Some(&i) => {
68                    if kept.get(i).is_some_and(|k| *k != p) {
69                        conflicts += 1;
70                    }
71                }
72                None => {
73                    index.insert(p.text.clone(), kept.len());
74                    kept.push(p);
75                }
76            }
77        }
78        if conflicts > 0 {
79            self.warn(format!("shared strings: {conflicts} phonetic readings were dropped because another <si> with the same text already had one"));
80        }
81        self.phonetics = kept;
82    }
83    /// A fresh id for a chart, picture, shape or table.
84    pub fn next_object_id(&mut self) -> u32 {
85        let id = self.next_id;
86        self.next_id = self.next_id.saturating_add(1);
87        id
88    }
89    pub fn style(&self, xf: Option<u32>) -> StyleId {
90        xf.and_then(|i| self.xf_map.get(i as usize).copied()).unwrap_or_default()
91    }
92    /// Reads and parses an optional part: problems other than size limits become warnings.
93    pub fn optional_xml(&mut self, part: &str) -> Result<Option<El>, IoError> {
94        match self.pkg.read_xml(part) {
95            Ok(Some(e)) => Ok(Some(e)),
96            Ok(None) => {
97                self.warn(format!("missing part {part}"));
98                Ok(None)
99            }
100            Err(IoError::TooLarge(m)) => Err(IoError::TooLarge(m)),
101            Err(e) => {
102                self.warn(format!("skipped {part}: {e}"));
103                Ok(None)
104            }
105        }
106    }
107}

113fn read_shared_strings(cx: &mut Ctx<'_>, wb_rels: &[Rel]) -> Result<(), IoError> {
114    // Shared strings, streamed item by item. Phonetic (furigana) runs are kept next to the text
115    // they annotate, so an open + save does not delete them.
116    if let Some(r) = find_rel(wb_rels, "sharedStrings").cloned() {
117        match cx.pkg.read(&r.target)? {
118            Some(b) => {
119                let mut sst = Vec::new();
120                let mut phonetics = Vec::new();
121                let mut dropped = 0usize;
122                let res = xml::parse_streaming(&b, &["si"], &mut |si| {
123                    let text = xml::rich_text(&si);
124                    let raw = si.kids("rPh").count();
125                    let (runs, props) = xml::phonetic_data(&si, &text);
126                    dropped += raw.saturating_sub(runs.len());
127                    if !runs.is_empty() || props.is_some() {
128                        phonetics.push(PhoneticString { text: text.clone(), runs, props });
129                    }
130                    sst.push(Arc::from(text));
131                    Ok(())
132                });
133                if let Err(e) = res {
134                    cx.warn(format!("shared strings are damaged ({e}); kept {} strings", sst.len()));
135                }
136                if dropped > 0 {
137                    cx.warn(format!("shared strings: {dropped} phonetic runs were dropped (bad sb/eb or no reading)"));
138                }
139                cx.sst = sst;
140                cx.keep_phonetics(phonetics);
141            }
142            None => cx.warn(format!("missing part {}", r.target)),
143        }
144    }
145
146    Ok(())
147}

149fn read_workbook_styles(cx: &mut Ctx<'_>, wb: &mut Workbook, wb_rels: &[Rel]) -> Result<(), IoError> {
150    // Theme, then styles: a font's link to the theme's body font is read against it.
151    if let Some(r) = find_rel(wb_rels, "theme").cloned()
152        && let Some(x) = cx.optional_xml(&r.target)?
153    {
154        wb.theme = crate::xlsx::theme::read_theme(&x);
155    }
156    // Cell checkboxes live in the feature property bags, which styles refer to.
157    let mut checkboxes = Vec::new();
158    if let Some(r) = find_rel(wb_rels, "FeaturePropertyBag").cloned()
159        && let Some(x) = cx.optional_xml(&r.target)?
160    {
161        checkboxes = crate::xlsx::feature_bags::checkbox_complements(&x);
162    }
163    if let Some(r) = find_rel(wb_rels, "styles").cloned() {
164        if let Some(x) = cx.optional_xml(&r.target)? {
165            let StylesIn { xfs, dxfs, palette, named, num_fmts } = read_styles(&x, &checkboxes, &wb.theme.minor_font);
166            cx.num_fmts = num_fmts;
167            // Cell format 0 is the Normal cell format: everything on it (font, number format,
168            // fill, borders, alignment, protection) is what a cell without its own format shows
169            // (#1013).
170            if let Some(normal) = xfs.first() {
171                cx.styles = StyleTable::with_default(normal.clone());
172            }
173            cx.xf_map = xfs.into_iter().enumerate().map(|(i, s)| if i == 0 { StyleId::DEFAULT } else { cx.styles.intern(s) }).collect();
174            cx.dxfs = dxfs;
175            cx.palette = palette;
176            wb.cell_styles = named;
177        }
178    } else {
179        cx.warn("no styles part");
180    }
181    Ok(())
182}

184/// Read the workbook XML selected by the package's root relationships.
185fn workbook_xml(pkg: &mut Package<'_>, part: &str) -> Result<El, IoError> {
186    let xml = match pkg.read_xml(part)? {
187        Some(xml) => xml,
188        None => {
189            if pkg.has("xl/workbook.xml") && part != "xl/workbook.xml" {
190                pkg.read_xml("xl/workbook.xml")?.ok_or_else(|| IoError::Format("no workbook part".into()))?
191            } else if pkg.has("content.xml") {
192                return Err(IoError::Format("this is an OpenDocument file, not XLSX".into()));
193            } else {
194                return Err(IoError::Format("no workbook part".into()));
195            }
196        }
197    };
198    if xml.name != "workbook" {
199        return Err(IoError::Format(format!("unexpected root element <{}> in workbook part", xml.name)));
200    }
201    Ok(xml)
202}

204/// Read workbook-level settings before processing the individual sheets.
205fn read_workbook_properties(cx: &mut Ctx<'_>, wb: &mut Workbook, xml: &El) {
206    if let Some(p) = xml.child("workbookPr") {
207        if p.flag("date1904", false) {
208            cx.date_system = DateSystem::D1904;
209        }
210        wb.remove_personal_info = p.flag("filterPrivacy", false);
211        wb.show_objects = match p.attr("showObjects") {
212            Some("none") => ShowObjects::None,
213            Some("placeholders") => ShowObjects::Placeholders,
214            _ => ShowObjects::All,
215        };
216    }
217    wb.date_system = cx.date_system;
218    if let Some(c) = xml.child("calcPr") {
219        wb.calc.mode = match c.attr("calcMode") {
220            Some("manual") => CalcMode::Manual,
221            Some("autoNoTable") => CalcMode::AutomaticExceptTables,
222            _ => CalcMode::Automatic,
223        };
224        wb.calc.iterative = c.flag("iterate", false);
225        if let Some(n) = c.attr_u32("iterateCount") {
226            wb.calc.set_max_iterations(n);
227        }
228        if let Some(d) = c.attr_f64("iterateDelta") {
229            wb.calc.set_max_change(d);
230        }
231        wb.calc.precision_as_displayed = !c.flag("fullPrecision", true);
232        wb.calc.r1c1 = c.attr("refMode") == Some("R1C1");
233    }
234    if let Some(p) = xml.child("workbookProtection") {
235        wb.protected_structure = p.flag("lockStructure", false);
236        wb.structure_password = if wb.protected_structure { crate::xlsx::sheet_read::read_password(p, "workbook") } else { None };
237    }
238}

240/// The single window modelled by GridCraft: prefer the first visible window,
241/// including the format's default visibility, and otherwise retain the first one.
242fn selected_workbook_view(workbook: &El) -> Option<&El> {
243    let views = workbook.child("bookViews")?;
244    views.kids("workbookView").find(|view| view.attr("visibility").unwrap_or("visible") == "visible").or_else(|| views.child("workbookView"))
245}

247pub fn read_xlsx<'a>(bytes: &'a [u8], opts: crate::xlsx::ReadOptions<'a>) -> Result<(Workbook, ReadReport), IoError> {
248    let functions = opts.functions;
249    let known_function = opts.known_function;
250    let pkg = Package::open(bytes)?;
251    let mut cx = Ctx {
252        pkg,
253        report: ReadReport::default(),
254        sst: vec![],
255        phonetics: vec![],
256        xf_map: vec![],
257        dxfs: vec![],
258        palette: crate::xlsx::tables::INDEXED.to_vec(),
259        date_system: DateSystem::D1900,
260        styles: StyleTable::default(),
261        persons: Default::default(),
262        next_id: 1,
263        num_fmts: Default::default(),
264        cell_pictures: vec![],
265        cf_ext_ids: vec![],
266        legacy_formulas: vec![],
267        shared_formula_bytes: 0,
268        known_function,
269        python: Default::default(),
270    };
271    let root_rels = cx.pkg.rels("")?;
272    let wb_part = find_rel(&root_rels, "officeDocument").map(|r| r.target.clone()).unwrap_or_else(|| "xl/workbook.xml".into());
273    let wb_xml = workbook_xml(&mut cx.pkg, &wb_part)?;
274    let wb_rels = cx.pkg.rels(&wb_part)?;
275    let mut kept = crate::xlsx::passthrough::KeptReader::new(&mut cx);
276    cx.cell_pictures = match crate::xlsx::richdata::read(&mut cx, &wb_rels) {
277        Ok(pictures) => pictures,
278        Err(error) => {
279            cx.warn(format!("cell-picture data could not be read ({error}); scalar fallbacks were retained"));
280            vec![]
281        }
282    };
283    // The code of Python cells, which their formulas name by index.
284    cx.python = crate::xlsx::python_part::read(&mut cx, &wb_rels)?;
285    let mut wb = Workbook::new();
286    wb.sheets.clear();
287
288    read_workbook_styles(&mut cx, &mut wb, &wb_rels)?;
289    read_shared_strings(&mut cx, &wb_rels)?;
290
291    if let Some(r) = find_rel(&wb_rels, "person").cloned()
292        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
293    {
294        for p in x.kids("person") {
295            if let (Some(id), Some(n)) = (p.attr("id"), p.attr("displayName")) {
296                cx.persons.insert(id.to_string(), n.to_string());
297            }
298        }
299    }
300
301    read_workbook_properties(&mut cx, &mut wb, &wb_xml);
302    let view_el = selected_workbook_view(&wb_xml);
303    let active_tab = view_el.and_then(|v| v.attr_u32("activeTab")).unwrap_or(0) as usize;
304    if let Some(v) = view_el {
305        // The window's display settings (Excel's "Display options for this workbook").
306        // Absent attributes take the format's defaults, which is also what a workbook
307        // without a <workbookView> gets.
308        wb.view = WorkbookView {
309            show_sheet_tabs: v.flag("showSheetTabs", true),
310            show_horizontal_scroll: v.flag("showHorizontalScroll", true),
311            show_vertical_scroll: v.flag("showVerticalScroll", true),
312            tab_ratio: v.attr_u32("tabRatio").unwrap_or(600).min(WorkbookView::MAX_TAB_RATIO),
313        };
314    }
315
316    // Sheets. `file_to_model[i]` maps the i-th <sheet> to a model sheet index.
317    let mut file_to_model: Vec<Option<usize>> = vec![];
318    // Legacy formula cells per model sheet, rewritten once names and tables are known.
319    let mut legacy_formulas: Vec<(usize, Vec<crate::base::CellRef>)> = vec![];
320    let mut pivot_caches = crate::xlsx::pivot::CachesIn::new(&wb_xml, &wb_rels);
321    let sheet_els: Vec<El> = wb_xml.child("sheets").map(|s| s.kids("sheet").cloned().collect()).unwrap_or_default();
322    let mut names = SheetNames::default();
323    for (i, se) in sheet_els.iter().enumerate() {
324        let name = se.attr("name").unwrap_or("").to_string();
325        let rel = se.attr("id").and_then(|id| wb_rels.iter().find(|r| r.id == id)).cloned();
326        let Some(rel) = rel else {
327            cx.warn(format!("sheet \"{name}\" has no part; skipped"));
328            file_to_model.push(None);
329            continue;
330        };
331        if rel.kind == "chartsheet" {
332            // Kept as it is and written back in place (see `passthrough`); nothing to model.
333            file_to_model.push(None);
334            continue;
335        }
336        if rel.kind != "worksheet" {
337            cx.warn(format!("sheet \"{name}\" is a {} and was skipped", rel.kind));
338            file_to_model.push(None);
339            continue;
340        }
341        let name = names.unique(&name, i);
342        cx.legacy_formulas.clear();
343        let (mut sheet, read_controls) = match crate::xlsx::sheet_read::read_sheet(&mut cx, &rel.target, &name) {
344            Ok(s) => s,
345            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
346            Err(e) => {
347                cx.warn(format!("sheet \"{name}\" could not be read: {e}"));
348                cx.legacy_formulas.clear();
349                (Sheet::new(name.clone()), Default::default())
350            }
351        };
352        legacy_formulas.push((wb.sheets.len(), std::mem::take(&mut cx.legacy_formulas)));
353        match crate::xlsx::pivot::read_pivots(&mut cx, &mut pivot_caches, &rel.target, &name) {
354            Ok(p) => sheet.pivots = p,
355            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
356            Err(e) => cx.warn(format!("PivotTables on sheet \"{name}\" could not be read: {e}")),
357        }
358        sheet.preserved = kept.sheet(&mut cx, &rel.target, &name, &read_controls);
359        sheet.file_id = se.attr_u32("sheetId").unwrap_or(0);
360        sheet.visibility = match se.attr("state") {
361            Some("hidden") => Visibility::Hidden,
362            Some("veryHidden") => Visibility::VeryHidden,
363            _ => Visibility::Visible,
364        };
365        file_to_model.push(Some(wb.sheets.len()));
366        wb.sheets.push(Arc::new(sheet));
367    }
368    if wb.sheets.is_empty() {
369        cx.warn("the workbook has no worksheets; added an empty sheet");
370        wb.sheets.push(Arc::new(Sheet::new("Sheet1")));
371    }
372    wb.set_active_sheet(file_to_model.get(active_tab).copied().flatten().unwrap_or(0));
373
374    // Defined names. Chartex parts refer to their data through hidden `_xlchart.*` names.
375    let mut chart_names: Vec<(String, String)> = vec![];
376    if let Some(dn) = wb_xml.child("definedNames") {
377        for d in dn.kids("definedName") {
378            let Some(name) = d.attr("name") else { continue };
379            let scope = match d.attr_u32("localSheetId") {
380                Some(i) => match file_to_model.get(i as usize).copied().flatten() {
381                    Some(m) => Some(m),
382                    None => continue,
383                },
384                None => None,
385            };
386            let text = d.text.trim();
387            let lname = name.to_ascii_lowercase();
388            if lname == "_xlnm._filterdatabase" {
389                continue;
390            }
391            if lname.starts_with("_xlchart.") {
392                chart_names.push((lname, crate::xlsx::fmla::from_file(&crate::xlsx::chart::chart_ref_from_file(text))));
393                continue;
394            }
395            if lname == "_xlnm.print_area" || lname == "_xlnm.print_titles" {
396                if let Some(si) = scope
397                    && let Some(sheet) = wb.sheet_mut(si)
398                {
399                    if lname == "_xlnm.print_area" {
400                        sheet.print.print_area = parse_print_area(text);
401                    } else {
402                        let (rows, cols) = parse_print_titles(text);
403                        sheet.print.title_rows = rows;
404                        sheet.print.title_cols = cols;
405                    }
406                }
407                continue;
408            }
409            wb.names.push(DefinedName {
410                name: name.to_string(),
411                scope,
412                formula: crate::xlsx::fmla::from_file(text),
413                comment: d.attr("comment").unwrap_or("").to_string(),
414                hidden: d.flag("hidden", false),
415            });
416        }
417    }
418
419    if !chart_names.is_empty() {
420        let resolve = |f: &mut String| {
421            if let Some((_, r)) = chart_names.iter().find(|(n, _)| f.eq_ignore_ascii_case(n)) {
422                *f = r.clone();
423            }
424        };
425        for sheet in wb.sheets.iter_mut().filter(|s| !s.charts.is_empty()) {
426            for ch in Arc::make_mut(sheet).charts.iter_mut() {
427                for s in ch.series.iter_mut() {
428                    resolve(&mut s.values);
429                    s.categories.iter_mut().for_each(&resolve);
430                    s.name.iter_mut().for_each(&resolve);
431                }
432            }
433        }
434    }
435
436    // Document properties.
437    if let Some(r) = find_rel(&root_rels, "core-properties").cloned()
438        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
439    {
440        let get = |n: &str| x.child(n).map(|e| e.text.clone()).unwrap_or_default();
441        wb.props.title = get("title");
442        wb.props.subject = get("subject");
443        wb.props.author = get("creator");
444        wb.props.keywords = get("keywords");
445        wb.props.description = get("description");
446    }
447    if let Some(r) = find_rel(&root_rels, "extended-properties").cloned()
448        && let Ok(Some(x)) = cx.pkg.read_xml(&r.target)
449    {
450        wb.props.company = x.child("Company").map(|e| e.text.clone()).unwrap_or_default();
451    }
452
453    // A VBA project is carried through unchanged: GridCraft never runs macros, but preserving the
454    // binary keeps macro-enabled workbooks from being silently stripped on save.
455    if let Some(r) = find_rel(&wb_rels, "vbaProject").cloned() {
456        match cx.pkg.read(&r.target) {
457            Ok(Some(data)) => {
458                wb.vba_project = Some(data.into());
459                cx.warn("macros (VBA project) were kept but not executed");
460            }
461            Ok(None) => cx.warn("macros (VBA project) reference a missing part and were dropped"),
462            Err(IoError::TooLarge(m)) => return Err(IoError::TooLarge(m)),
463            Err(e) => cx.warn(format!("macros (VBA project) could not be read and were dropped: {e}")),
464        }
465    }
466    wb.preserved = kept.workbook(&mut cx, &wb_part, &wb_xml, &file_to_model);
467    if let Some(w) = cx.python.warning() {
468        cx.warn(w);
469    }
470    // Calls of LAMBDA names are spelled like the names (the parser upper-cases them).
471    wb.apply_name_call_case();
472    // Legacy formulas read as dynamic-array formulas with Excel's implicit `@`.
473    crate::xlsx::array_formula::apply_legacy(&mut wb, legacy_formulas, functions);
474    wb.styles = cx.styles;
475    wb.phonetics = cx.phonetics;
476    Ok((wb, cx.report))
477}

479/// The sheet names an import has assigned so far, for repairing repeated ones.
480#[derive(Default)]
481pub(crate) struct SheetNames {
482    /// Assigned names, ASCII-lowercased.
483    taken: HashSet<String>,
484    /// For each lowercased stem, the next suffix to try: a file full of repeated names would
485    /// otherwise retry every occupied suffix for every sheet.
486    next: HashMap<String, u64>,
487}
488
489impl SheetNames {
490    pub(crate) fn of(wb: &Workbook) -> Self {
491        SheetNames { taken: wb.sheets.iter().map(|s| s.name.to_ascii_lowercase()).collect(), next: HashMap::new() }
492    }
493
494    /// A valid name for the i-th sheet, unique among the names assigned so far, which it joins.
495    pub(crate) fn unique(&mut self, name: &str, i: usize) -> String {
496        let mut base: String = name.chars().filter(|c| !matches!(c, ':' | '\\' | '/' | '?' | '*' | '[' | ']')).take(31).collect();
497        if base.trim().is_empty() {
498            base = format!("Sheet{}", i + 1);
499        }
500        let name = if self.taken.contains(&base.to_ascii_lowercase()) {
501            let stem: String = base.chars().take(25).collect();
502            let next = self.next.entry(stem.to_ascii_lowercase()).or_insert(2);
503            let found = (*next..).find(|n| !self.taken.contains(&format!("{stem} ({n})").to_ascii_lowercase()));
504            match found {
505                Some(n) => {
506                    *next = n + 1;
507                    format!("{stem} ({n})")
508                }
509                None => base,
510            }
511        } else {
512            base
513        };
514        self.taken.insert(name.to_ascii_lowercase());
515        name
516    }
517}

519/// Strips a sheet qualifier (`'My Sheet'!$A$1` → `$A$1`).
520fn strip_sheet(s: &str) -> &str {
521    match s.rfind('!') {
522        Some(i) => s.get(i + 1..).unwrap_or(""),
523        None => s,
524    }
525}
526
527pub fn parse_print_area(text: &str) -> Option<RangeRef> {
528    let first = text.split(',').next()?;
529    let r = strip_sheet(first.trim()).replace('$', "");
530    RangeRef::parse(&r)
531}

533/// A print-titles span: the first and last row (or column) repeated on each page.
534type TitleSpan = Option<(u32, u32)>;
535
536pub fn parse_print_titles(text: &str) -> (TitleSpan, TitleSpan) {
537    let mut rows = None;
538    let mut cols = None;
539    for part in text.split(',') {
540        let p = strip_sheet(part.trim()).replace('$', "");
541        let Some((a, b)) = p.split_once(':') else { continue };
542        if let (Ok(r0), Ok(r1)) = (a.parse::<u32>(), b.parse::<u32>()) {
543            if r0 >= 1 && r1 >= r0 && r1 <= crate::base::MAX_ROWS {
544                rows = Some((r0 - 1, r1 - 1));
545            }
546        } else if let (Some(c0), Some(c1)) = (crate::base::letters_to_col(a), crate::base::letters_to_col(b))
547            && c1 >= c0
548        {
549            cols = Some((c0, c1));
550        }
551    }
552    (rows, cols)
553}
