15fn group_kind(g: &El) -> Option<ChartKind> {
16    let grouping = g.child_val("grouping").unwrap_or("standard");
17    let stacked = grouping == "stacked";
18    let pct = grouping == "percentStacked";
19    Some(match g.name.as_str() {
20        "barChart" | "bar3DChart" => {
21            let bar = g.child_val("barDir") == Some("bar");
22            match (bar, stacked, pct) {
23                (false, true, _) => ChartKind::ColumnStacked,
24                (false, _, true) => ChartKind::ColumnStacked100,
25                (false, _, _) => ChartKind::ColumnClustered,
26                (true, true, _) => ChartKind::BarStacked,
27                (true, _, true) => ChartKind::BarStacked100,
28                (true, _, _) => ChartKind::BarClustered,
29            }
30        }
31        "lineChart" | "line3DChart" => {
32            if stacked || pct {
33                ChartKind::LineStacked
34            } else {
35                let group_markers = g.child_val("marker").is_none_or(|v| v == "1" || v == "true");
36                let all_none = g.kids("ser").all(|s| s.path(&["marker", "symbol"]).and_then(|m| m.attr("val")) == Some("none"));
37                if !group_markers || (all_none && g.kids("ser").next().is_some()) { ChartKind::Line } else { ChartKind::LineMarkers }
38            }
39        }
40        "pieChart" | "pie3DChart" | "ofPieChart" => ChartKind::Pie,
41        "doughnutChart" => ChartKind::Doughnut,
42        "areaChart" | "area3DChart" => {
43            if pct {
44                ChartKind::AreaStacked100
45            } else if stacked {
46                ChartKind::AreaStacked
47            } else {
48                ChartKind::Area
49            }
50        }
51        "scatterChart" => {
52            let no_lines = g.kids("ser").all(|s| s.path(&["spPr", "ln"]).is_some_and(|l| l.child("noFill").is_some()));
53            if g.child_val("scatterStyle") == Some("marker") || (no_lines && g.kids("ser").next().is_some()) {
54                ChartKind::Scatter
55            } else {
56                ChartKind::ScatterLines
57            }
58        }
59        "bubbleChart" => ChartKind::Bubble,
60        "radarChart" => ChartKind::Radar,
61        "stockChart" => ChartKind::Stock,
62        _ => return None,
63    })
64}

67const THIS_BOOK: &str = "[0]!";

69/// A chart reference (`<c:f>`) read from a file → model text: drops a leading `=` and the
70/// `[0]!` in front of workbook-scoped names (`[0]!Sales` → `Sales`), which the formula parser
71/// would take for a structured reference. External books (`[1]Sheet1!$A$1`), quoted sheet
72/// names and string literals are kept as they are.
73pub(crate) fn chart_ref_from_file(text: &str) -> String {
74    let t = text.trim();
75    let t = t.strip_prefix('=').unwrap_or(t);
76    let mut out = String::with_capacity(t.len());
77    let mut quote: Option<char> = None;
78    // The last character kept: `[0]!` only counts at the start or after an operator,
79    // never inside a structured reference (`T[[0]]`) or a longer name.
80    let mut prev: Option<char> = None;
81    let mut i = 0;
82    while let Some(c) = t.get(i..).and_then(|r| r.chars().next()) {
83        let rest = t.get(i..).unwrap_or("");
84        if quote.is_none()
85            && rest.starts_with(THIS_BOOK)
86            && prev.is_none_or(|p| !(p.is_alphanumeric() || "_.\\[]'\"!$#".contains(p)))
87            && rest.get(THIS_BOOK.len()..).and_then(|n| n.chars().next()).is_some_and(|n| n.is_alphabetic() || n == '_' || n == '\\')
88        {
89            i += THIS_BOOK.len();
90            continue;
91        }
92        match (c, quote) {
93            ('"' | '\'', None) => quote = Some(c),
94            (q, Some(open)) if q == open => quote = None,
95            _ => {}
96        }
97        out.push(c);
98        prev = Some(c);
99        i += c.len_utf8();
100    }
101    out
102}

104fn data_ref(e: Option<&El>) -> Option<String> {
105    let e = e?;
106    for k in ["numRef", "strRef", "multiLvlStrRef"] {
107        if let Some(f) = e.child(k).and_then(|r| r.child("f")) {
108            return Some(chart_ref_from_file(&f.text));
109        }
110    }
111    // Literal values become an array constant.
112    for k in ["numLit", "strLit"] {
113        if let Some(l) = e.child(k) {
114            let pts: Vec<String> = l
115                .kids("pt")
116                .filter_map(|p| p.child("v"))
117                .map(|v| if k == "numLit" { v.text.trim().to_string() } else { format!("\"{}\"", v.text.replace('"', "\"\"")) })
118                .collect();
119            return Some(format!("{{{}}}", pts.join(",")));
120        }
121    }
122    None
123}

125fn read_series(s: &El) -> Series {
126    let name = s
127        .child("tx")
128        .and_then(|tx| tx.path(&["strRef", "f"]).map(|f| chart_ref_from_file(&f.text)).or_else(|| tx.child("v").map(|v| v.text.clone())));
129    let sp = s.child("spPr");
130    let color = sp
131        .and_then(|p| p.child("solidFill"))
132        .and_then(dml_color)
133        .or_else(|| sp.and_then(|p| p.path(&["ln", "solidFill"])).and_then(dml_color))
134        .or_else(|| s.path(&["marker", "spPr", "solidFill"]).and_then(dml_color));
135    Series {
136        name,
137        categories: data_ref(s.child("cat")).or_else(|| data_ref(s.child("xVal"))),
138        values: data_ref(s.child("val")).or_else(|| data_ref(s.child("yVal"))).unwrap_or_default(),
139        bubble_sizes: data_ref(s.child("bubbleSize")),
140        color,
141        secondary: false,
142        kind: None,
143        quartiles: Default::default(),
144        smooth: bool_val(s, "smooth"),
145        trendlines: s.kids("trendline").filter_map(read_trendline).collect(),
146    }
147}

174/// `<c:trendline>` (ECMA-376 Part 1, 21.2.2.211). Unknown types are dropped.
175fn read_trendline(t: &El) -> Option<Trendline> {
176    let kind = match t.child_val("trendlineType").unwrap_or("linear") {
177        "linear" => TrendlineKind::Linear,
178        "exp" => TrendlineKind::Exponential,
179        "log" => TrendlineKind::Logarithmic,
180        "power" => TrendlineKind::Power,
181        "poly" => TrendlineKind::Polynomial(t.child_val("order").and_then(|v| v.trim().parse::<u8>().ok()).unwrap_or(2).clamp(2, 6)),
182        "movingAvg" => TrendlineKind::MovingAverage(t.child_val("period").and_then(|v| v.trim().parse::<u32>().ok()).unwrap_or(2).clamp(2, 255)),
183        _ => return None,
184    };
185    let ln = t.path(&["spPr", "ln"]);
186    Some(Trendline {
187        kind,
188        name: t.child("name").map(|n| n.text.clone()),
189        forward: Trendline::clamp_forecast(f64_val(t, "forward").unwrap_or(0.0)),
190        backward: Trendline::clamp_forecast(f64_val(t, "backward").unwrap_or(0.0)),
191        intercept: f64_val(t, "intercept"),
192        display_equation: bool_val(t, "dispEq"),
193        display_r_squared: bool_val(t, "dispRSqr"),
194        color: ln.and_then(|l| l.child("solidFill")).and_then(dml_color),
195        width: ln.and_then(|l| l.attr_f64("w")).map(|w| (w / EMU_PER_PT) as f32).filter(|w| w.is_finite() && *w >= 0.0),
196        dash: ln.and_then(|l| l.child("prstDash")).and_then(|d| d.attr("val")).and_then(|v| DASHES.iter().find(|(_, n)| *n == v).map(|(d, _)| *d)),
197    })
198}

200fn rich_title(t: &El) -> Option<String> {
201    let tx = t.child("tx")?;
202    if let Some(rich) = tx.child("rich") {
203        let paras: Vec<String> =
204            rich.kids("p").map(|p| p.kids("r").filter_map(|r| r.child("t")).map(|t| t.text.as_str()).collect::<String>()).collect();
205        let s = paras.join("\n");
206        return if s.is_empty() { None } else { Some(s) };
207    }
208    if let Some(sr) = tx.child("strRef") {
209        if let Some(v) = sr.path(&["strCache", "pt", "v"]) {
210            return Some(v.text.clone());
211        }
212        return sr.child("f").map(|f| f.text.trim().to_string());
213    }
214    None
215}

221pub fn read_chart(root: &El) -> Option<Chart> {
222    let chart = root.child("chart")?;
223    let plot = chart.child("plotArea")?;
224    let groups: Vec<(&El, ChartKind)> = plot.children.iter().filter_map(|g| group_kind(g).map(|k| (g, k))).collect();
225    let (first, first_kind) = *groups.first()?;
226    let first_axes = axis_ids(first);
227    let mut series = vec![];
228    for (gi, (g, k)) in groups.iter().enumerate() {
229        for s in g.kids("ser") {
230            let mut ser = read_series(s);
231            if groups.len() > 1 {
232                // Every series of a combo chart keeps its own group's kind, the first group's included.
233                ser.kind = Some(*k);
234            }
235            if gi > 0 {
236                ser.secondary = axis_ids(g) != first_axes;
237            }
238            series.push(ser);
239        }
240    }
241    let kind = if groups.len() > 1 { ChartKind::Combo } else { first_kind };
242    let show_val = |e: &El| e.child("dLbls").is_some_and(|d| bool_val(d, "showVal"));
243    let data_labels = groups.iter().any(|(g, _)| show_val(g) || g.kids("ser").any(show_val));
244    let val_axes: Vec<&El> = plot.kids("valAx").collect();
245    let cat_ax = plot.child("catAx").or_else(|| plot.child("dateAx"));
246    let scatter = matches!(first_kind, ChartKind::Scatter | ChartKind::ScatterLines | ChartKind::Bubble);
247    let (x_ax, y_ax) = if scatter {
248        let x = val_axes.iter().find(|a| matches!(a.child_val("axPos"), Some("b") | Some("t"))).copied();
249        let y = val_axes.iter().find(|a| matches!(a.child_val("axPos"), Some("l") | Some("r"))).copied();
250        (x, y)
251    } else {
252        (cat_ax, val_axes.first().copied())
253    };
254    let legend = match chart.child("legend") {
255        None => LegendPos::None,
256        Some(l) => match l.child_val("legendPos").unwrap_or("r") {
257            "b" => LegendPos::Bottom,
258            "t" => LegendPos::Top,
259            "l" => LegendPos::Left,
260            _ => LegendPos::Right,
261        },
262    };
263    let doughnut = plot.child("doughnutChart");
264    let hole_size = doughnut.and_then(|g| g.child_val("holeSize")).and_then(|v| v.trim().parse::<u8>().ok()).unwrap_or(75).clamp(10, 90);
265    let first_slice_angle =
266        plot.child("pieChart").or(doughnut).and_then(|g| g.child_val("firstSliceAng")).and_then(|v| v.trim().parse::<u16>().ok()).unwrap_or(0) % 360;
267    Some(Chart {
268        id: 0,
269        kind,
270        anchor: Default::default(),
271        title: chart.child("title").and_then(rich_title),
272        series,
273        legend,
274        data_labels,
275        data_table: plot.child("dTable").map(|t| crate::model::ChartDataTable {
276            show_keys: bool_val(t, "showKeys"),
277            horizontal_border: bool_val(t, "showHorzBorder"),
278            vertical_border: bool_val(t, "showVertBorder"),
279            outline: bool_val(t, "showOutline"),
280        }),
281        gridlines: val_axes.iter().any(|a| a.child("majorGridlines").is_some()),
282        style: root.child_val("style").and_then(|v| v.parse().ok()).unwrap_or(2),
283        hole_size,
284        first_slice_angle,
285        x_title: x_ax.and_then(|a| a.child("title")).and_then(rich_title),
286        y_title: y_ax.and_then(|a| a.child("title")).and_then(rich_title),
287        source: None,
288        by_rows: false,
289        pivot: root.path(&["pivotSource", "name"]).map(|n| strip_book(n.text.trim()).to_string()).filter(|n| !n.is_empty()),
290    })
291}

308fn group_for(k: ChartKind) -> G {
309    match k {
310        ChartKind::ColumnStacked => G::Bar { bar: false, grouping: "stacked" },
311        ChartKind::ColumnStacked100 => G::Bar { bar: false, grouping: "percentStacked" },
312        ChartKind::BarClustered => G::Bar { bar: true, grouping: "clustered" },
313        ChartKind::BarStacked => G::Bar { bar: true, grouping: "stacked" },
314        ChartKind::BarStacked100 => G::Bar { bar: true, grouping: "percentStacked" },
315        ChartKind::Line => G::Line { stacked: false, markers: false },
316        ChartKind::LineMarkers => G::Line { stacked: false, markers: true },
317        ChartKind::LineStacked => G::Line { stacked: true, markers: false },
318        ChartKind::Pie => G::Pie,
319        ChartKind::Doughnut => G::Doughnut,
320        ChartKind::Area => G::Area { grouping: "standard" },
321        ChartKind::AreaStacked => G::Area { grouping: "stacked" },
322        ChartKind::AreaStacked100 => G::Area { grouping: "percentStacked" },
323        ChartKind::Scatter => G::Scatter { lines: false },
324        ChartKind::ScatterLines => G::Scatter { lines: true },
325        ChartKind::Bubble => G::Bubble,
326        ChartKind::Radar => G::Radar,
327        ChartKind::Stock => G::Stock,
328        // Combo charts start from clustered columns; chartex kinds have their own part.
329        _ => G::Bar { bar: false, grouping: "clustered" },
330    }
331}

335/// The sheet and range a same-workbook reference (`Sheet1!$B$2:$B$9`) names.
336fn ref_target(wb: &Workbook, sheet_idx: usize, f: &str) -> Option<(usize, RangeRef)> {
337    let e = crate::formula::parse(f).ok()?;
338    let Expr::Ref(r) = e else { return None };
339    let si = match &r.sheet {
340        SheetSel::Current => sheet_idx,
341        SheetSel::Named(n) => wb.sheet_index(n)?,
342        SheetSel::Span(..) => return None,
343    };
344    Some((si, r.range()))
345}

347/// Values of a same-workbook reference (`Sheet1!$B$2:$B$9`), row-major.
348pub(crate) fn ref_values(wb: &Workbook, sheet_idx: usize, f: &str) -> Option<Vec<Value>> {
349    let (si, mut range) = ref_target(wb, sheet_idx, f)?;
350    let sheet = wb.sheet(si)?;
351    // Whole columns/rows: cut the far end at the used range. The start stays put, so point indices keep matching the reference.
352    if let Some(u) = sheet.used_range() {
353        if range.is_full_cols() {
354            range.end.row = range.end.row.min(u.end.row.max(range.start.row));
355        }
356        if range.is_full_rows() {
357            range.end.col = range.end.col.min(u.end.col.max(range.start.col));
358        }
359    }
360    if range.count() > MAX_CACHE {
361        return None;
362    }
363    Some(range.iter().map(|c| sheet.value(c)).collect())
364}

370/// A chart reference that is a name: its sheet when qualified (`Sheet1!Sales`), the bare
371/// name, and the definition formulas would find from sheet `si` (none when undefined). `None`
372/// when it isn't a name, or names a sheet the workbook doesn't have.
373fn chart_name<'a>(wb: &'a Workbook, si: usize, f: &str) -> Option<(Option<usize>, String, Option<&'a DefinedName>)> {
374    let Ok(Expr::Name(n)) = crate::formula::parse(f.trim()) else { return None };
375    // The parser has already unquoted the sheet (`'Bob''s'!Name` → `Bob's!Name`, #2012): split
376    // at the last `!` (`'a!b'!Name`) and keep the name as is (`' Sales '` keeps its spaces).
377    match n.rsplit_once('!') {
378        Some((s, bare)) => {
379            let i = wb.sheet_index(s)?;
380            Some((Some(i), bare.to_string(), wb.name(bare, i)))
381        }
382        None => {
383            let def = wb.name(&n, si);
384            Some((None, n, def))
385        }
386    }
387}

389/// A chart reference for `<c:f>`. Excel needs a name qualified there: with a bare `Name` it
390/// draws the whole chart blank, with no repair prompt (observed in Excel for Windows,
391/// Microsoft 365). A workbook-scoped name is written `[0]!Name`, as Excel writes it, a
392/// sheet-scoped one `Sheet!Name`; an undefined name gets `[0]!`, as the file it came from had.
393fn chart_ref_to_file(wb: &Workbook, si: usize, f: &str) -> String {
394    let on_sheet = |i: usize, n: &str| wb.sheet(i).map(|sh| format!("{}!{n}", crate::formula::quote_sheet(&sh.name)));
395    match chart_name(wb, si, f) {
396        Some((Some(i), n, _)) => on_sheet(i, &n).unwrap_or_else(|| crate::xlsx::fmla::text_to_file(f)),
397        Some((None, n, def)) => def.and_then(|d| d.scope).and_then(|i| on_sheet(i, &n)).unwrap_or_else(|| format!("{THIS_BOOK}{n}")),
398        None => crate::xlsx::fmla::text_to_file(f),
399    }
400}

407fn num_data(tag: &str, f: Option<&str>, wb: &Workbook, si: usize) -> String {
408    let f = f.unwrap_or("").trim();
409    if f.is_empty() {
410        return format!("<c:{tag}><c:numLit><c:ptCount val=\"0\"/></c:numLit></c:{tag}>");
411    }
412    if f.starts_with('{') {
413        let vals: Vec<&str> = f.trim_matches(|c| c == '{' || c == '}').split([',', ';']).collect();
414        let mut s = format!("<c:{tag}><c:numLit><c:formatCode>General</c:formatCode><c:ptCount val=\"{}\"/>", vals.len());
415        for (i, v) in vals.iter().enumerate() {
416            if let Ok(n) = v.trim().parse::<f64>() {
417                let _ = write!(s, "<c:pt idx=\"{i}\"><c:v>{}</c:v></c:pt>", num(n));
418            }
419        }
420        let _ = write!(s, "</c:numLit></c:{tag}>");
421        return s;
422    }
423    let mut s = format!("<c:{tag}><c:numRef><c:f>{}</c:f>", esc(&chart_ref_to_file(wb, si, f)));
424    if let Some(vals) = ref_values(wb, si, f) {
425        let _ = write!(s, "<c:numCache><c:formatCode>General</c:formatCode><c:ptCount val=\"{}\"/>", vals.len());
426        for (i, v) in vals.iter().enumerate() {
427            if let Value::Number(n) = v {
428                let _ = write!(s, "<c:pt idx=\"{i}\"><c:v>{}</c:v></c:pt>", num(*n));
429            }
430        }
431        s.push_str("</c:numCache>");
432    }
433    let _ = write!(s, "</c:numRef></c:{tag}>");
434    s
435}

437/// Splits the inside of an array constant on `,` and `;` outside double quotes; inside quotes `""` is a literal quote.
438/// Unquoted items are trimmed; quoted ones keep the text between the quotes and drop the spaces around them.
439fn split_array_items(inner: &str) -> Vec<String> {
440    let mut items = Vec::new();
441    let mut cur = String::new();
442    let mut quoted = false;
443    let mut in_quote = false;
444    let mut chars = inner.chars().peekable();
445    let finish = |cur: &mut String, quoted: &mut bool| {
446        let text = std::mem::take(cur);
447        let item = if *quoted { text } else { text.trim().to_string() };
448        *quoted = false;
449        item
450    };
451    while let Some(c) = chars.next() {
452        match c {
453            '"' if in_quote && chars.peek() == Some(&'"') => {
454                cur.push('"');
455                chars.next();
456            }
457            '"' => {
458                if !quoted && cur.trim().is_empty() {
459                    cur.clear();
460                }
461                in_quote = !in_quote;
462                quoted = true;
463            }
464            ',' | ';' if !in_quote => items.push(finish(&mut cur, &mut quoted)),
465            c if quoted && !in_quote && c.is_whitespace() => {}
466            _ => cur.push(c),
467        }
468    }
469    items.push(finish(&mut cur, &mut quoted));
470    items
471}

473fn str_data(tag: &str, f: &str, wb: &Workbook, si: usize) -> String {
474    let f = f.trim();
475    if f.starts_with('{') {
476        let vals = split_array_items(f.trim_matches(|c| c == '{' || c == '}'));
477        let mut s = format!("<c:{tag}><c:strLit><c:ptCount val=\"{}\"/>", vals.len());
478        for (i, v) in vals.iter().enumerate() {
479            let _ = write!(s, "<c:pt idx=\"{i}\"><c:v>{}</c:v></c:pt>", esc(v));
480        }
481        let _ = write!(s, "</c:strLit></c:{tag}>");
482        return s;
483    }
484    let mut s = format!("<c:{tag}><c:strRef><c:f>{}</c:f>", esc(&chart_ref_to_file(wb, si, f)));
485    if let Some(vals) = ref_values(wb, si, f) {
486        let _ = write!(s, "<c:strCache><c:ptCount val=\"{}\"/>", vals.len());
487        for (i, v) in vals.iter().enumerate() {
488            if !v.is_empty() {
489                let _ = write!(s, "<c:pt idx=\"{i}\"><c:v>{}</c:v></c:pt>", esc(&v.display()));
490            }
491        }
492        s.push_str("</c:strCache>");
493    }
494    let _ = write!(s, "</c:strRef></c:{tag}>");
495    s
496}

498/// `<c:cat>` of a series: a string reference, or a multi-level one (`multiLvlStrRef`) when the
499/// category range has more columns than a column of values (or more rows than a row of values:
500/// year beside month). Its cache lists the innermost level first, then each outer level with
501/// only the cells that start a group, as Excel writes it.
502fn cat_data(f: &str, values: &str, wb: &Workbook, si: usize) -> String {
503    multi_level_cat(f.trim(), values, wb, si).unwrap_or_else(|| str_data("cat", f, wb, si))
504}
505
506fn multi_level_cat(f: &str, values: &str, wb: &Workbook, si: usize) -> Option<String> {
507    let (sheet, range) = ref_target(wb, si, f)?;
508    let value_range = ref_target(wb, si, values).map(|(_, r)| r);
509    let levels = category_levels(&range, value_range.as_ref())?;
510    if range.count() > MAX_CACHE {
511        return None;
512    }
513    let sheet = wb.sheet(sheet)?;
514    let (w, h) = (range.width() as usize, range.height() as usize);
515    let text = |row: usize, col: usize| -> String {
516        let v = range.start.offset(row as i64, col as i64).map(|c| sheet.value(c));
517        v.filter(|v| !v.is_empty()).map(|v| v.display()).unwrap_or_default()
518    };
519    // The innermost level (nearest the values), then the outer ones.
520    let (n, lvls): (usize, Vec<Vec<String>>) = if levels.by_columns {
521        (h, (0..=levels.outer as usize).filter_map(|k| (w - 1).checked_sub(k)).map(|c| (0..h).map(|r| text(r, c)).collect()).collect())
522    } else {
523        (w, (0..=levels.outer as usize).filter_map(|k| (h - 1).checked_sub(k)).map(|r| (0..w).map(|c| text(r, c)).collect()).collect())
524    };
525    let mut s =
526        format!("<c:cat><c:multiLvlStrRef><c:f>{}</c:f><c:multiLvlStrCache><c:ptCount val=\"{n}\"/>", esc(&crate::xlsx::fmla::text_to_file(f)));
527    for lvl in &lvls {
528        s.push_str("<c:lvl>");
529        for (i, v) in lvl.iter().enumerate() {
530            if !v.is_empty() {
531                let _ = write!(s, "<c:pt idx=\"{i}\"><c:v>{}</c:v></c:pt>", esc(v));
532            }
533        }
534        s.push_str("</c:lvl>");
535    }
536    s.push_str("</c:multiLvlStrCache></c:multiLvlStrRef></c:cat>");
537    Some(s)
538}

540fn series_xml(s: &Series, idx: usize, g: G, wb: &Workbook, si: usize) -> String {
541    let mut x = format!("<c:ser><c:idx val=\"{idx}\"/><c:order val=\"{idx}\"/>");
542    if let Some(n) = &s.name {
543        if is_ref(n) || is_defined_name(wb, si, n) {
544            x.push_str(&str_data("tx", n, wb, si));
545        } else {
546            let lit = n.trim().strip_prefix('"').and_then(|t| t.strip_suffix('"')).unwrap_or(n);
547            let _ = write!(x, "<c:tx><c:v>{}</c:v></c:tx>", esc(lit));
548        }
549    }
550    let color = s.color.as_ref().and_then(|c: &Color| srgb(c, &wb.theme));
551    let line_like = matches!(g, G::Line { .. } | G::Radar | G::Scatter { lines: true });
552    match (&color, line_like) {
553        (Some(c), true) => {
554            let _ = write!(x, "<c:spPr><a:ln w=\"28575\" cap=\"rnd\"><a:solidFill>{c}</a:solidFill><a:round/></a:ln></c:spPr>");
555        }
556        (Some(c), false) if g == (G::Scatter { lines: false }) => {
557            let _ = write!(
558                x,
559                "<c:spPr><a:ln w=\"19050\"><a:noFill/></a:ln></c:spPr><c:marker><c:symbol val=\"circle\"/><c:size val=\"5\"/><c:spPr><a:solidFill>{c}</a:solidFill></c:spPr></c:marker>"
560            );
561        }
562        (Some(c), false) => {
563            let _ = write!(x, "<c:spPr><a:solidFill>{c}</a:solidFill></c:spPr>");
564        }
565        (None, _) if g == (G::Scatter { lines: false }) || g == G::Stock => x.push_str("<c:spPr><a:ln w=\"19050\"><a:noFill/></a:ln></c:spPr>"),
566        _ => {}
567    }
568    match g {
569        G::Bar { .. } => x.push_str("<c:invertIfNegative val=\"0\"/>"),
570        G::Line { markers: false, .. } | G::Scatter { lines: true } | G::Stock => x.push_str("<c:marker><c:symbol val=\"none\"/></c:marker>"),
571        G::Bubble => x.push_str("<c:invertIfNegative val=\"0\"/>"),
572        _ => {}
573    }
574    // Trendlines sit between dLbls and the data in CT_LineSer, CT_BarSer, CT_AreaSer,
575    // CT_ScatterSer and CT_BubbleSer; pie and radar series have none.
576    if !matches!(g, G::Pie | G::Doughnut | G::Radar) {
577        for t in &s.trendlines {
578            x.push_str(&trendline_xml(t, wb));
579        }
580    }
581    match g {
582        G::Scatter { .. } | G::Bubble => {
583            if let Some(c) = &s.categories {
584                x.push_str(&num_data("xVal", Some(c), wb, si));
585            }
586            x.push_str(&num_data("yVal", Some(&s.values), wb, si));
587            if g == G::Bubble {
588                x.push_str(&num_data("bubbleSize", s.bubble_sizes.as_deref().or(Some(&s.values)), wb, si));
589                x.push_str("<c:bubble3D val=\"0\"/>");
590            } else {
591                let _ = write!(x, "<c:smooth val=\"{}\"/>", s.smooth as u8);
592            }
593        }
594        _ => {
595            if let Some(c) = &s.categories {
596                x.push_str(&cat_data(c, &s.values, wb, si));
597            }
598            x.push_str(&num_data("val", Some(&s.values), wb, si));
599            if matches!(g, G::Line { .. }) {
600                let _ = write!(x, "<c:smooth val=\"{}\"/>", s.smooth as u8);
601            }
602        }
603    }
604    x.push_str("</c:ser>");
605    x
606}

608/// `<c:trendline>` with children in schema order: name, spPr, trendlineType, order, period,
609/// forward, backward, intercept, dispRSqr, dispEq, trendlineLbl.
610fn trendline_xml(t: &Trendline, wb: &Workbook) -> String {
611    let mut x = String::from("<c:trendline>");
612    if let Some(n) = &t.name {
613        let _ = write!(x, "<c:name>{}</c:name>", esc(n));
614    }
615    let color = t.color.as_ref().and_then(|c| srgb(c, &wb.theme));
616    if color.is_some() || t.width.is_some() || t.dash.is_some() {
617        x.push_str("<c:spPr><a:ln");
618        if let Some(w) = t.width.filter(|w| w.is_finite()) {
619            // ST_LineWidth: 0..=20116800 EMU.
620            let _ = write!(x, " w=\"{}\"", (f64::from(w) * EMU_PER_PT).round().clamp(0.0, 20_116_800.0) as i64);
621        }
622        x.push_str(" cap=\"rnd\">");
623        if let Some(c) = &color {
624            let _ = write!(x, "<a:solidFill>{c}</a:solidFill>");
625        }
626        if let Some(d) = t.dash.and_then(|d| DASHES.iter().find(|(k, _)| *k == d)) {
627            let _ = write!(x, "<a:prstDash val=\"{}\"/>", d.1);
628        }
629        x.push_str("</a:ln></c:spPr>");
630    }
631    let ty = match t.kind {
632        TrendlineKind::Linear => "linear",
633        TrendlineKind::Exponential => "exp",
634        TrendlineKind::Logarithmic => "log",
635        TrendlineKind::Polynomial(_) => "poly",
636        TrendlineKind::Power => "power",
637        TrendlineKind::MovingAverage(_) => "movingAvg",
638    };
639    let _ = write!(x, "<c:trendlineType val=\"{ty}\"/>");
640    match t.kind {
641        TrendlineKind::Polynomial(o) => {
642            let _ = write!(x, "<c:order val=\"{}\"/>", o.clamp(2, 6));
643        }
644        TrendlineKind::MovingAverage(p) => {
645            let _ = write!(x, "<c:period val=\"{}\"/>", p.clamp(2, 255));
646        }
647        _ => {}
648    }
649    for (tag, v) in [("forward", t.forward), ("backward", t.backward)] {
650        if v.is_finite() && v > 0.0 {
651            let _ = write!(x, "<c:{tag} val=\"{}\"/>", num(v));
652        }
653    }
654    if let Some(i) = t.intercept.filter(|i| i.is_finite()) {
655        let _ = write!(x, "<c:intercept val=\"{}\"/>", num(i));
656    }
657    let _ = write!(x, "<c:dispRSqr val=\"{}\"/><c:dispEq val=\"{}\"/>", t.display_r_squared as u8, t.display_equation as u8);
658    if t.display_r_squared || t.display_equation {
659        x.push_str("<c:trendlineLbl><c:numFmt formatCode=\"General\" sourceLinked=\"0\"/></c:trendlineLbl>");
660    }
661    x.push_str("</c:trendline>");
662    x
663}

672fn group_xml(g: G, series: &[(usize, &Series)], ch: &Chart, axes: (u32, u32), wb: &Workbook, si: usize) -> String {
673    let sers: String = series.iter().map(|(i, s)| series_xml(s, *i, g, wb, si)).collect();
674    let dl = dlbls(ch.data_labels);
675    let ax = format!("<c:axId val=\"{}\"/><c:axId val=\"{}\"/>", axes.0, axes.1);
676    match g {
677        G::Bar { bar, grouping } => {
678            let overlap = if grouping == "clustered" { String::new() } else { "<c:overlap val=\"100\"/>".into() };
679            format!(
680                "<c:barChart><c:barDir val=\"{}\"/><c:grouping val=\"{grouping}\"/><c:varyColors val=\"0\"/>{sers}{dl}<c:gapWidth val=\"150\"/>{overlap}{ax}</c:barChart>",
681                if bar { "bar" } else { "col" }
682            )
683        }
684        G::Line { stacked, .. } => format!(
685            "<c:lineChart><c:grouping val=\"{}\"/><c:varyColors val=\"0\"/>{sers}{dl}<c:marker val=\"1\"/>{ax}</c:lineChart>",
686            if stacked { "stacked" } else { "standard" }
687        ),
688        G::Pie => format!("<c:pieChart><c:varyColors val=\"1\"/>{sers}{dl}<c:firstSliceAng val=\"{}\"/></c:pieChart>", ch.first_slice_angle % 360),
689        G::Doughnut => format!(
690            "<c:doughnutChart><c:varyColors val=\"1\"/>{sers}{dl}<c:firstSliceAng val=\"{}\"/><c:holeSize val=\"{}\"/></c:doughnutChart>",
691            ch.first_slice_angle % 360,
692            ch.hole_size.clamp(10, 90)
693        ),
694        G::Area { grouping } => format!("<c:areaChart><c:grouping val=\"{grouping}\"/><c:varyColors val=\"0\"/>{sers}{dl}{ax}</c:areaChart>"),
695        G::Scatter { .. } => format!("<c:scatterChart><c:scatterStyle val=\"lineMarker\"/><c:varyColors val=\"0\"/>{sers}{dl}{ax}</c:scatterChart>"),
696        G::Bubble => {
697            format!("<c:bubbleChart><c:varyColors val=\"0\"/>{sers}{dl}<c:bubbleScale val=\"100\"/><c:showNegBubbles val=\"0\"/>{ax}</c:bubbleChart>")
698        }
699        G::Radar => format!("<c:radarChart><c:radarStyle val=\"marker\"/><c:varyColors val=\"0\"/>{sers}{dl}{ax}</c:radarChart>"),
700        G::Stock => format!("<c:stockChart>{sers}{dl}<c:hiLowLines/>{ax}</c:stockChart>"),
701    }
702}

724fn axis_xml(a: &Ax<'_>) -> String {
725    let mut s = format!(
726        "<c:{0}><c:axId val=\"{1}\"/><c:scaling><c:orientation val=\"minMax\"/></c:scaling><c:delete val=\"{2}\"/><c:axPos val=\"{3}\"/>",
727        if a.cat { "catAx" } else { "valAx" },

734    }
735    if let Some(t) = a.title {

737    }

739        s,
740        "<c:numFmt formatCode=\"General\" sourceLinked=\"1\"/><c:majorTickMark val=\"out\"/><c:minorTickMark val=\"none\"/><c:tickLblPos val=\"nextTo\"/><c:crossAx val=\"{}\"/><c:crosses val=\"{}\"/>",

743    );
744    if a.cat {
745        s.push_str("<c:auto val=\"1\"/><c:lblAlgn val=\"ctr\"/><c:lblOffset val=\"100\"/><c:noMultiLvlLbl val=\"0\"/></c:catAx>");
746    } else {
747        s.push_str("<c:crossBetween val=\"between\"/></c:valAx>");
748    }
749    s
750}

752/// A PivotChart link without the workbook in brackets: `[Book.xlsx]Sheet2!PT` → `Sheet2!PT`.
753fn strip_book(link: &str) -> &str {
754    link.strip_prefix('[').and_then(|r| r.split_once(']')).map_or(link, |(_, rest)| rest)
755}

757/// Chart part XML for a model chart on sheet `si`.
758pub fn write_chart(wb: &Workbook, si: usize, ch: &Chart, book_name: &str) -> String {
759    let mut s = String::from(
760        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<c:chartSpace xmlns:c=\"http://schemas.openxmlformats.org/drawingml/2006/chart\" xmlns:a=\"http://schemas.openxmlformats.org/drawingml/2006/main\" xmlns:r=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships\"><c:roundedCorners val=\"0\"/>",
761    );
762    if (1..=48).contains(&ch.style) {
763        let _ = write!(s, "<c:style val=\"{}\"/>", ch.style);
764    }
765    // A PivotChart names its PivotTable behind the workbook in brackets (`[Book.xlsx]Sheet2!PivotTable1`);
766    // without the brackets Excel shows an ordinary chart.
767    if let Some(link) = ch.pivot.as_deref().map(str::trim).filter(|l| !l.is_empty()) {
768        let link = strip_book(link);
769        let _ = write!(s, "<c:pivotSource><c:name>[{}]{}</c:name><c:fmtId val=\"0\"/></c:pivotSource>", esc(book_name), esc(link));
770    }
771    s.push_str("<c:chart>");
772    match &ch.title {
773        Some(t) if !t.is_empty() => {
774            s.push_str(&title_xml(t));
775            s.push_str("<c:autoTitleDeleted val=\"0\"/>");
776        }
777        _ => s.push_str("<c:autoTitleDeleted val=\"1\"/>"),
778    }
779    s.push_str("<c:plotArea><c:layout/>");
780    let all: Vec<(usize, &Series)> = ch.series.iter().enumerate().collect();
781    let primary_g;
782    let mut secondary: Option<G> = None;
783    if ch.kind == ChartKind::Combo {
784        // Series without a kind of their own: the first is a column, the others lines (as drawn).
785        let kind = |i: usize, x: &Series| x.kind.filter(|k| *k != ChartKind::Combo).or((i > 0).then_some(ChartKind::Line));
786        let is_line = |k: Option<ChartKind>| matches!(k.map(group_for), Some(G::Line { .. } | G::Area { .. } | G::Radar | G::Scatter { .. }));
787        let bars: Vec<(usize, &Series)> = all.iter().filter(|(i, x)| !x.secondary && !is_line(kind(*i, x))).copied().collect();
788        let lines: Vec<(usize, &Series)> = all.iter().filter(|(i, x)| !x.secondary && is_line(kind(*i, x))).copied().collect();
789        let sec: Vec<(usize, &Series)> = all.iter().filter(|(_, x)| x.secondary).copied().collect();
790        primary_g = G::Bar { bar: false, grouping: "clustered" };
791        if !bars.is_empty() || (lines.is_empty() && sec.is_empty()) {
792            s.push_str(&group_xml(primary_g, &bars, ch, (1, 2), wb, si));
793        }
794        if !lines.is_empty() {
795            s.push_str(&group_xml(G::Line { stacked: false, markers: false }, &lines, ch, (1, 2), wb, si));
796        }
797        if !sec.is_empty() {
798            let g = match sec.first().and_then(|(_, x)| x.kind).map(group_for) {
799                Some(b @ G::Bar { .. }) => b,
800                _ => G::Line { stacked: false, markers: false },
801            };
802            s.push_str(&group_xml(g, &sec, ch, (3, 4), wb, si));
803            secondary = Some(g);
804        }
805    } else {
806        primary_g = match group_for(ch.kind) {
807            // A stock chart has 3 or 4 series (high-low-close, open-high-low-close).
808            G::Stock if !(3..=4).contains(&ch.series.len()) => G::Line { stacked: false, markers: false },
809            g => g,
810        };
811        s.push_str(&group_xml(primary_g, &all, ch, (1, 2), wb, si));
812    }
813    match primary_g {
814        G::Pie | G::Doughnut => {}
815        G::Scatter { .. } | G::Bubble => {
816            s.push_str(&axis_xml(&Ax {
817                cat: false,
818                id: 1,
819                cross: 2,
820                pos: "b",
821                delete: false,
822                grid: false,
823                title: ch.x_title.as_deref(),
824                crosses_max: false,
825            }));
826            s.push_str(&axis_xml(&Ax {
827                cat: false,
828                id: 2,
829                cross: 1,
830                pos: "l",
831                delete: false,
832                grid: ch.gridlines,
833                title: ch.y_title.as_deref(),
834                crosses_max: false,
835            }));
836        }
837        G::Bar { bar: true, .. } => {
838            s.push_str(&axis_xml(&Ax {
839                cat: true,
840                id: 1,
841                cross: 2,
842                pos: "l",
843                delete: false,
844                grid: false,
845                title: ch.x_title.as_deref(),
846                crosses_max: false,
847            }));
848            s.push_str(&axis_xml(&Ax {
849                cat: false,
850                id: 2,
851                cross: 1,
852                pos: "b",
853                delete: false,
854                grid: ch.gridlines,
855                title: ch.y_title.as_deref(),
856                crosses_max: false,
857            }));
858        }
859        _ => {
860            s.push_str(&axis_xml(&Ax {
861                cat: true,
862                id: 1,
863                cross: 2,
864                pos: "b",
865                delete: false,
866                grid: false,
867                title: ch.x_title.as_deref(),
868                crosses_max: false,
869            }));
870            s.push_str(&axis_xml(&Ax {
871                cat: false,
872                id: 2,
873                cross: 1,
874                pos: "l",
875                delete: false,
876                grid: ch.gridlines,
877                title: ch.y_title.as_deref(),
878                crosses_max: false,
879            }));
880        }
881    }
882    if secondary.is_some() {
883        s.push_str(&axis_xml(&Ax { cat: true, id: 3, cross: 4, pos: "b", delete: true, grid: false, title: None, crosses_max: false }));
884        s.push_str(&axis_xml(&Ax { cat: false, id: 4, cross: 3, pos: "r", delete: false, grid: false, title: None, crosses_max: true }));
885    }
886    if let Some(t) = &ch.data_table {
887        let _ = write!(
888            s,
889            "<c:dTable><c:showHorzBorder val=\"{}\"/><c:showVertBorder val=\"{}\"/><c:showOutline val=\"{}\"/><c:showKeys val=\"{}\"/></c:dTable>",
890            u8::from(t.horizontal_border),
891            u8::from(t.vertical_border),
892            u8::from(t.outline),
893            u8::from(t.show_keys)
894        );
895    }
896    s.push_str("</c:plotArea>");
897    let pos = match ch.legend {
898        LegendPos::None => None,
899        LegendPos::Bottom => Some("b"),
900        LegendPos::Top => Some("t"),
901        LegendPos::Left => Some("l"),
902        LegendPos::Right => Some("r"),
903    };
904    if let Some(p) = pos {
905        let _ = write!(s, "<c:legend><c:legendPos val=\"{p}\"/><c:overlay val=\"0\"/></c:legend>");
906    }
907    s.push_str("<c:plotVisOnly val=\"1\"/><c:dispBlanksAs val=\"gap\"/></c:chart></c:chartSpace>");
908    s
909}
