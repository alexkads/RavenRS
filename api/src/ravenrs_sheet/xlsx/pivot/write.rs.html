3#[derive(Default)]
4struct Stats {
5    blank: bool,
6    text: bool,
7    number: bool,
8    non_int: bool,
9    boolean: bool,
10    error: bool,
11    long: bool,
12    min: f64,
13    max: f64,
14}
15
16impl Stats {
17    fn add(&mut self, v: &Value) {
18        match v {
19            Value::Empty => self.blank = true,
20            Value::Number(n) => {
21                if !self.number {
22                    self.min = *n;
23                    self.max = *n;
24                }
25                self.number = true;
26                self.min = self.min.min(*n);
27                self.max = self.max.max(*n);
28                if n.fract() != 0.0 {
29                    self.non_int = true;
30                }
31            }
32            Value::Text(t) => {
33                self.text = true;
34                if t.chars().count() > 255 {
35                    self.long = true;
36                }
37            }
38            Value::Bool(_) => self.boolean = true,
39            Value::Error(_) => self.error = true,
40            Value::Array(_) => self.add(&v.scalar()),
41        }
42    }
43    /// Excel counts booleans and errors as strings in the cache field flags.
44    fn strings(&self) -> bool {
45        self.text || self.boolean || self.error
46    }
47    /// Kinds for `containsMixedTypes`: strings (text, booleans, errors) and numbers.
48    fn kinds(&self) -> usize {
49        usize::from(self.strings()) + usize::from(self.number)
50    }
51}

53/// Date grouping of a cache field.
54struct Group {
55    by: PivotDateGroup,
56    /// The database field the grouping applies to.
57    base: usize,
58    /// The next (coarser) grouping field of the same base.
59    par: Option<usize>,
60    start: f64,
61    end: f64,
62    /// Group item captions, including the `<start` and `>end` items.
63    items: Vec<String>,
64}

66struct CField {
67    name: String,
68    /// Column offset in the source (database fields only).
69    col: Option<u32>,
70    stats: Stats,
71    /// Shared items (written when `itemized`).
72    items: Vec<Value>,
73    index: HashMap<Key, u32>,
74    itemized: bool,
75    group: Option<Group>,
76    /// A calculated field's formula (without `=`); such fields aren't database fields.
77    formula: Option<String>,
78}

80impl CField {
81    /// Item captions the PivotTable's `<item x>` indices refer to.
82    fn item_labels(&self) -> Vec<String> {
83        match &self.group {
84            Some(g) => g.items.clone(),
85            None => self.items.iter().map(label).collect(),
86        }
87    }

117impl PivotWriter {
118    /// Writes the caches of a sheet's PivotTables and adds the sheet's relationships to its
119    /// PivotTable parts. Pivots whose source can't be resolved are not written.
120    pub fn add_sheet(&mut self, wb: &Workbook, si: usize, out: &mut Out, rels: &mut Rels) {
121        let Some(sheet) = wb.sheet(si) else { return };
122        for (idx, pt) in sheet.pivots.iter().enumerate() {
123            let Some(cache) = build_cache(wb, pt, si) else { continue };
124            let n = self.caches.len() as u32 + 1;
125            let cache_part = format!("xl/pivotCache/pivotCacheDefinition{n}.xml");
126            let rec_rid = if let Some((count, xml)) = &cache.records {
127                let mut r = Rels::default();
128                let rid = r.add("pivotCacheRecords", &format!("pivotCacheRecords{n}.xml"));
129                out.part(&format!("xl/pivotCache/_rels/pivotCacheDefinition{n}.xml.rels"), None, r.xml().into_bytes());
130                let s = format!("{XML_DECL}<pivotCacheRecords xmlns=\"{NS_MAIN}\" xmlns:r=\"{NS_REL}\" count=\"{count}\">{xml}</pivotCacheRecords>");
131                out.part(&format!("xl/pivotCache/pivotCacheRecords{n}.xml"), Some(CT_PIVOT_CACHE_REC), s.into_bytes());
132                Some(rid)
133            } else {
134                None
135            };
136            out.part(&cache_part, Some(CT_PIVOT_CACHE_DEF), cache_xml(&cache, rec_rid.as_deref(), wb.date_system).into_bytes());
137            self.caches.push((n, format!("pivotCache/pivotCacheDefinition{n}.xml")));
138            self.tables += 1;
139            let t = self.tables;
140            rels.add("pivotTable", &format!("../pivotTables/pivotTable{t}.xml"));
141            let mut tr = Rels::default();
142            tr.add("pivotCacheDefinition", &format!("../pivotCache/pivotCacheDefinition{n}.xml"));
143            out.part(&format!("xl/pivotTables/_rels/pivotTable{t}.xml.rels"), None, tr.xml().into_bytes());
144            self.pending.push(Pending { part: format!("xl/pivotTables/pivotTable{t}.xml"), sheet: si, idx, cache_id: n, cache });
145        }
146    }
147
148    /// Adds the workbook relationships to the caches; returns the `<pivotCaches>` element (empty
149    /// when there are none).
150    pub fn workbook_xml(&self, wb_rels: &mut Rels) -> String {
151        if self.caches.is_empty() {
152            return String::new();
153        }
154        let mut s = String::from("<pivotCaches>");
155        for (id, target) in &self.caches {
156            let rid = wb_rels.add("pivotCacheDefinition", target);
157            let _ = write!(s, "<pivotCache cacheId=\"{id}\" r:id=\"{rid}\"/>");
158        }
159        s.push_str("</pivotCaches>");
160        s
161    }
162
163    /// Number formats used by value fields (to be registered in the styles part).
164    pub fn number_formats(&self, wb: &Workbook) -> Vec<String> {
165        let mut v = Vec::new();
166        for p in &self.pending {
167            if let Some(pt) = wb.sheet(p.sheet).and_then(|s| s.pivots.get(p.idx)) {
168                for val in &pt.values {
169                    if let Some(f) = &val.number_format
170                        && !v.contains(f)
171                    {
172                        v.push(f.clone());
173                    }
174                }
175            }
176        }
177        v
178    }
179
180    /// Writes the PivotTable parts. `fmt_id` maps a number format code to its styles id.
181    pub fn finish(self, wb: &Workbook, out: &mut Out, fmt_id: &dyn Fn(&str) -> Option<u32>) {
182        for p in self.pending {
183            let Some(pt) = wb.sheet(p.sheet).and_then(|s| s.pivots.get(p.idx)) else { continue };
184            let xml = table_xml(pt, p.idx, p.cache_id, &p.cache, fmt_id);
185            out.part(&p.part, Some(CT_PIVOT_TABLE), xml.into_bytes());
186        }
187    }

190/// Resolves a pivot source to (sheet, range including the header row, worksheetSource XML).
191fn resolve_source(wb: &Workbook, text: &str, default_sheet: usize, depth: u32) -> Option<(usize, RangeRef, String)> {
192    let t = text.trim().trim_start_matches('=').trim();
193    if t.is_empty() || depth > 4 {
194        return None;
195    }
196    if let Some((si, ti)) = wb.table(t)
197        && let Some(sh) = wb.sheet(si)
198        && let Some(tb) = sh.tables.get(ti)
199    {
200        if !tb.header_row {
201            return None;
202        }
203        let end_row = tb.range.end.row.saturating_sub(tb.totals_row as u32).max(tb.range.start.row);
204        let r = RangeRef::new(tb.range.start, CellRef::new(end_row, tb.range.end.col));
205        // Table names are sanitized when written; refer by range when this one would change.
206        let valid = !tb.name.is_empty()
207            && tb.name.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '.')
208            && tb.name.starts_with(|c: char| c.is_alphabetic() || c == '_')
209            && CellRef::parse(&tb.name).is_none();
210        let xml = if valid { format!("<worksheetSource name=\"{}\"/>", esc_attr(&tb.name)) } else { range_source_xml(&sh.name, r) };
211        return Some((si, r, xml));
212    }
213    let (sheet, body) = crate::base::split_sheet(t);
214    if let Some(r) = RangeRef::parse(body) {
215        let si = match &sheet {
216            Some(n) => wb.sheet_index(n)?,
217            None => default_sheet,
218        };
219        let sh = wb.sheet(si)?;
220        let r = if r.is_full_cols() || r.is_full_rows() { r.intersection(&sh.used_range()?)? } else { r };
221        return Some((si, r, range_source_xml(&sh.name, r)));
222    }
223    if sheet.is_none()
224        && let Some(n) = wb.name(t, default_sheet)
225    {
226        let (si, r, _) = resolve_source(wb, &n.formula.clone(), default_sheet, depth + 1)?;
227        let xml = if n.scope.is_none() {
228            format!("<worksheetSource name=\"{}\"/>", esc_attr(&n.name))
229        } else {
230            wb.sheet(si).map(|s| range_source_xml(&s.name, r))?
231        };
232        return Some((si, r, xml));
233    }
234    None
235}

237fn range_source_xml(sheet: &str, r: RangeRef) -> String {
238    let name: String = sheet.chars().take(31).collect();
239    format!("<worksheetSource ref=\"{}\" sheet=\"{}\"/>", r.a1(), esc_attr(&name))

249fn cache_fields(sh: &crate::model::Sheet, r: RangeRef) -> Vec<CField> {
250    let mut fields: Vec<CField> = Vec::new();
251    // Lower-case names taken, and the next suffix for each header (linear for repeated headers).
252    let mut taken = std::collections::HashSet::new();
253    let mut next_suffix: HashMap<String, usize> = HashMap::new();
254    for (k, c) in (r.start.col..=r.end.col).enumerate() {
255        let mut h = header_text(&sh.value(CellRef::new(r.start.row, c)).scalar());
256        if h.is_empty() {
257            h = format!("Column{}", k + 1);
258        }
259        let mut n = h.clone();
260        let mut i = next_suffix.get(&h.to_lowercase()).copied().unwrap_or(2);
261        while taken.contains(&n.to_lowercase()) {
262            n = format!("{h}{i}");
263            i += 1;
264        }
265        next_suffix.insert(h.to_lowercase(), i);
266        taken.insert(n.to_lowercase());
267        fields.push(CField {
268            name: n,
269            col: Some(k as u32),
270            stats: Stats::default(),
271            items: vec![],
272            index: HashMap::new(),
273            itemized: false,
274            group: None,
275            formula: None,
276        });
277    }
278    fields
279}

281fn group_cache_fields(fields: &mut Vec<CField>, axis: &[&PivotField], date_system: DateSystem) {
282    let find = |fields: &[CField], name: &str| fields.iter().position(|f| f.col.is_some() && same(&f.name, name));
283    let groupable = |f: &CField| {
284        let s = &f.stats;
285        s.number
286            && !s.text
287            && !s.boolean
288            && !s.error
289            && datetime_from_serial(date_system, s.min).is_some()
290            && datetime_from_serial(date_system, s.max).is_some()
291    };
292    // (base column field, groupings fine → coarse)
293    let mut groupings: Vec<(usize, Vec<PivotDateGroup>)> = Vec::new();
294    for pf in axis {
295        let Some(b) = find(fields, &pf.source_col) else { continue };
296        if pf.date_group == PivotDateGroup::None || !fields.get(b).is_some_and(groupable) {
297            continue;
298        }
299        match groupings.iter_mut().find(|(x, _)| *x == b) {
300            Some((_, v)) => {
301                if !v.contains(&pf.date_group) {
302                    v.push(pf.date_group);
303                }
304            }
305            None => groupings.push((b, vec![pf.date_group])),
306        }
307    }
308    let fineness = |g: &PivotDateGroup| match g {
309        PivotDateGroup::Days => 0,
310        PivotDateGroup::Months => 1,
311        PivotDateGroup::Quarters => 2,
312        PivotDateGroup::Years => 3,
313        PivotDateGroup::None => 4,
314    };
315    for (b, list) in groupings.iter_mut() {
316        list.sort_by_key(fineness);
317        let (min, max) = fields.get(*b).map(|f| (f.stats.min, f.stats.max)).unwrap_or((0.0, 0.0));
318        let (start, end) = (min.floor(), max.floor() + 1.0);
319        let make = |by: PivotDateGroup| Group { by, base: *b, par: None, start, end, items: group_items(date_system, by, start, end) };
320        let mut prev = *b;
321        for (k, by) in list.iter().enumerate() {
322            if k == 0 {
323                if let Some(f) = fields.get_mut(*b) {
324                    f.group = Some(make(*by));
325                }
326                continue;
327            }
328            let new = fields.len();
329            let caption = group_caption(*by);
330            let mut n = caption.to_string();
331            let mut i = 2;
332            while fields.iter().any(|f| same(&f.name, &n)) {
333                n = format!("{caption}{i}");
334                i += 1;
335            }
336            if let Some(g) = fields.get_mut(prev).and_then(|f| f.group.as_mut()) {
337                g.par = Some(new);
338            }
339            fields.push(CField {
340                name: n,
341                col: None,
342                stats: Stats::default(),
343                items: vec![],
344                index: HashMap::new(),
345                itemized: false,
346                group: Some(make(*by)),
347                formula: None,
348            });
349            prev = new;
350        }
351    }
352}

354fn build_cache(wb: &Workbook, pt: &PivotTable, pivot_sheet: usize) -> Option<Cache> {
355    let (si, r, source_xml) = resolve_source(wb, &pt.source, pivot_sheet, 0)?;
356    if r.count() > MAX_SOURCE_CELLS {
357        return None;
358    }
359    let sh = wb.sheet(si)?;
360    let mut fields = cache_fields(sh, r);
361    let find = |fields: &[CField], name: &str| fields.iter().position(|f| f.col.is_some() && same(&f.name, name));
362
363    // Records (rows that aren't entirely blank), read once.
364    let mut rows: Vec<u32> = Vec::new();
365    if r.end.row > r.start.row {
366        for row in r.start.row + 1..=r.end.row {
367            let blank = (r.start.col..=r.end.col).all(|c| sh.value(CellRef::new(row, c)).scalar().is_empty());
368            if !blank {
369                rows.push(row);
370            }
371        }
372    }
373    for &row in &rows {
374        for f in fields.iter_mut() {
375            let c = r.start.col + f.col.unwrap_or(0);
376            f.stats.add(&sh.value(CellRef::new(row, c)).scalar());
377        }
378    }
379
380    // Which fields are on an axis, and how dates are grouped: per base column, the base field
381    // takes the finest grouping and coarser ones become extra (non-database) fields chained
382    // through `par`, as Excel lays them out.
383    let axis: Vec<&PivotField> = pt.rows.iter().chain(pt.columns.iter()).collect();
384    group_cache_fields(&mut fields, &axis, wb.date_system);
385    let mut axis_map: Vec<Option<usize>> = Vec::with_capacity(axis.len());
386    for pf in &axis {
387        let Some(b) = find(&fields, &pf.source_col) else {
388            axis_map.push(None);
389            continue;
390        };
391        let grouped = fields.iter().position(|f| f.group.as_ref().is_some_and(|g| g.base == b && g.by == pf.date_group));
392        match grouped {
393            Some(i) if pf.date_group != PivotDateGroup::None => axis_map.push(Some(i)),
394            _ => {
395                if let Some(f) = fields.get_mut(b)
396                    && f.group.is_none()
397                {
398                    f.itemized = true;
399                }
400                axis_map.push(Some(b));
401            }
402        }
403    }
404    // Page fields (filters not on an axis) need items too.
405    for f in &pt.filters {
406        let on_axis = axis.iter().any(|a| same(&a.source_col, &f.source_col));
407        if !on_axis
408            && let Some(i) = find(&fields, &f.source_col)
409            && let Some(cf) = fields.get_mut(i)
410        {
411            cf.itemized = true;
412        }
413    }
414    for f in fields.iter_mut() {
415        if f.group.is_some() {
416            f.itemized = false;
417        } else if f.col.is_some() && (f.stats.text || f.stats.boolean || f.stats.error) {
418            f.itemized = true;
419        }
420    }
421    // Shared items, in order of first appearance.
422    for &row in &rows {
423        for f in fields.iter_mut().filter(|f| f.itemized) {
424            let c = r.start.col + f.col.unwrap_or(0);
425            let v = sh.value(CellRef::new(row, c)).scalar();
426            let k = key(&v);
427            if !f.index.contains_key(&k) && f.items.len() < MAX_ITEMS {
428                f.index.insert(k, f.items.len() as u32);
429                f.items.push(v);
430            }
431        }
432    }
433    // Records snapshot.
434    let db_fields = fields.iter().filter(|f| f.col.is_some()).count() as u64;
435    let records = if (rows.len() as u64).saturating_mul(db_fields) <= MAX_RECORD_CELLS {
436        let mut s = String::new();
437        for &row in &rows {
438            s.push_str("<r>");
439            for f in fields.iter().filter(|f| f.col.is_some()) {
440                let c = r.start.col + f.col.unwrap_or(0);
441                let v = sh.value(CellRef::new(row, c)).scalar();
442                if f.itemized {
443                    match f.index.get(&key(&v)) {
444                        Some(i) => {
445                            let _ = write!(s, "<x v=\"{i}\"/>");
446                        }
447                        None => s.push_str(&item_xml(&v, wb.date_system, false)),
448                    }
449                } else {
450                    s.push_str(&item_xml(&v, wb.date_system, f.group.is_some()));
451                }
452            }
453            s.push_str("</r>");
454        }
455        Some((rows.len() as u64, s))
456    } else {
457        None
458    };
459    // Calculated fields come after the database and grouping fields.
460    for cf in &pt.calculated_fields {
461        let f = cf.formula.trim();
462        let f = f.strip_prefix('=').unwrap_or(f).trim();
463        if cf.name.trim().is_empty() || f.is_empty() || fields.iter().any(|x| same(&x.name, &cf.name)) {
464            continue;
465        }
466        fields.push(CField {
467            name: cf.name.clone(),
468            col: None,
469            stats: Stats::default(),
470            items: vec![],
471            index: HashMap::new(),
472            itemized: false,
473            group: None,
474            formula: Some(f.to_string()),
475        });
476    }
477    Some(Cache { fields, axis_map, source_xml, records })
478}

480fn group_items(sys: DateSystem, by: PivotDateGroup, start: f64, end: f64) -> Vec<String> {
481    let mut v = vec![format!("<{}", mdy(sys, start))];
482    match by {
483        PivotDateGroup::Years => {
484            let y0 = datetime_from_serial(sys, start).map(|d| d.year).unwrap_or(1900);
485            let y1 = datetime_from_serial(sys, end - 1.0).map(|d| d.year).unwrap_or(y0).max(y0);
486            for y in y0..=y1.min(y0 + 10_000) {
487                v.push(y.to_string());
488            }
489        }
490        PivotDateGroup::Quarters => v.extend((1..=4).map(|q| format!("Qtr{q}"))),
491        PivotDateGroup::Months => v.extend(MONTHS.iter().map(ToString::to_string)),
492        PivotDateGroup::Days | PivotDateGroup::None => {
493            const DAYS: [u32; 12] = [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
494            for (m, n) in MONTHS.iter().zip(DAYS) {
495                for d in 1..=n {
496                    v.push(format!("{d}-{m}"));
497                }
498            }
499        }
500    }
501    v.push(format!(">{}", mdy(sys, end)));
502    v
503}

505/// A cache value element (`<n>`, `<s>`, `<m>`…); `date` writes numbers as `<d>`.
506fn item_xml(v: &Value, sys: DateSystem, date: bool) -> String {
507    match v {
508        Value::Empty => "<m/>".into(),
509        Value::Number(n) => {
510            if date && let Some(d) = iso(sys, *n) {
511                return format!("<d v=\"{d}\"/>");
512            }
513            format!("<n v=\"{}\"/>", crate::xlsx::xml::num(*n))
514        }
515        Value::Text(t) => format!("<s v=\"{}\"/>", esc_attr(t)),
516        Value::Bool(b) => format!("<b v=\"{}\"/>", u8::from(*b)),
517        Value::Error(e) => format!("<e v=\"{}\"/>", esc_attr(e.as_str())),
518        Value::Array(_) => item_xml(&v.scalar(), sys, date),
519    }
520}

522fn cache_xml(c: &Cache, records_rid: Option<&str>, sys: DateSystem) -> String {
523    let mut s = format!("{XML_DECL}<pivotCacheDefinition xmlns=\"{NS_MAIN}\" xmlns:r=\"{NS_REL}\"");
524    match (records_rid, &c.records) {
525        (Some(rid), Some((count, _))) => {
526            let _ = write!(s, " r:id=\"{rid}\" recordCount=\"{count}\"");
527        }
528        _ => s.push_str(" saveData=\"0\""),
529    }
530    s.push_str(" refreshOnLoad=\"1\" refreshedBy=\"GridCraft\" createdVersion=\"6\" refreshedVersion=\"6\" minRefreshableVersion=\"3\">");
531    let _ = write!(s, "<cacheSource type=\"worksheet\">{}</cacheSource>", c.source_xml);
532    let _ = write!(s, "<cacheFields count=\"{}\">", c.fields.len());
533    for f in &c.fields {
534        let fmt = if f.group.is_some() { 14 } else { 0 };
535        let _ = write!(s, "<cacheField name=\"{}\" numFmtId=\"{fmt}\"", esc_attr(&f.name));
536        if let Some(formula) = &f.formula {
537            let _ = write!(s, " formula=\"{}\"", esc_attr(formula));
538        }
539        if f.col.is_none() {
540            s.push_str(" databaseField=\"0\"");
541        }
542        s.push('>');
543        let st = &f.stats;
544        if f.col.is_some() {
545            s.push_str("<sharedItems");
546            if let Some(g) = &f.group {
547                s.push_str(" containsSemiMixedTypes=\"0\" containsNonDate=\"0\" containsDate=\"1\" containsString=\"0\"");
548                if st.blank {
549                    s.push_str(" containsBlank=\"1\"");
550                }
551                if let (Some(a), Some(b)) = (iso(sys, st.min), iso(sys, st.max)) {
552                    let _ = write!(s, " minDate=\"{a}\" maxDate=\"{b}\"");
553                }
554                let _ = g;
555            } else {
556                if !st.strings() && !st.blank {
557                    s.push_str(" containsSemiMixedTypes=\"0\"");
558                }
559                if !st.strings() {
560                    s.push_str(" containsString=\"0\"");
561                }
562                if st.blank {
563                    s.push_str(" containsBlank=\"1\"");
564                }
565                if st.kinds() > 1 {
566                    s.push_str(" containsMixedTypes=\"1\"");
567                }
568                if st.number {
569                    s.push_str(" containsNumber=\"1\"");
570                    if !st.non_int {
571                        s.push_str(" containsInteger=\"1\"");
572                    }
573                    let _ = write!(s, " minValue=\"{}\" maxValue=\"{}\"", crate::xlsx::xml::num(st.min), crate::xlsx::xml::num(st.max));
574                }
575                if st.long {
576                    s.push_str(" longText=\"1\"");
577                }
578            }
579            if f.itemized {
580                let _ = write!(s, " count=\"{}\">", f.items.len());
581                for v in &f.items {
582                    s.push_str(&item_xml(v, sys, false));
583                }
584                s.push_str("</sharedItems>");
585            } else {
586                s.push_str("/>");
587            }
588        }
589        if let Some(g) = &f.group {
590            s.push_str("<fieldGroup");
591            if let Some(p) = g.par {
592                let _ = write!(s, " par=\"{p}\"");
593            }
594            let _ = write!(s, " base=\"{}\">", g.base);
595            let _ = write!(s, "<rangePr groupBy=\"{}\"", group_by_name(g.by));
596            if let (Some(a), Some(b)) = (iso(sys, g.start), iso(sys, g.end)) {
597                let _ = write!(s, " startDate=\"{a}\" endDate=\"{b}\"");
598            }
599            let _ = write!(s, "/><groupItems count=\"{}\">", g.items.len());
600            for it in &g.items {
601                let _ = write!(s, "<s v=\"{}\"/>", esc_attr(it));
602            }
603            s.push_str("</groupItems></fieldGroup>");
604        }
605        s.push_str("</cacheField>");
606    }
607    s.push_str("</cacheFields></pivotCacheDefinition>");
608    s
609}

611struct PivotTableLayout<'a> {
612    row_fields: Vec<usize>,
613    column_fields: Vec<usize>,
614    hidden_fields: HashMap<usize, &'a PivotFilter>,
615    page_fields: Vec<(usize, &'a PivotFilter)>,
616    values: Vec<(usize, &'a PivotValue)>,
617}
618
619fn table_xml(pt: &PivotTable, idx: usize, cache_id: u32, c: &Cache, fmt_id: &dyn Fn(&str) -> Option<u32>) -> String {
620    let find_db = |name: &str| c.fields.iter().position(|f| f.col.is_some() && same(&f.name, name));
621    let row_fields: Vec<usize> = c.axis_map.iter().take(pt.rows.len()).filter_map(|x| *x).collect();
622    let column_fields: Vec<usize> = c.axis_map.iter().skip(pt.rows.len()).filter_map(|x| *x).collect();
623    // Filters: on an axis field → hidden items there; otherwise a page field.
624    let mut hidden: HashMap<usize, &PivotFilter> = HashMap::new();
625    let mut pages: Vec<(usize, &PivotFilter)> = Vec::new();
626    for f in &pt.filters {
627        let axis_pos = pt.rows.iter().chain(pt.columns.iter()).position(|a| same(&a.source_col, &f.source_col));
628        match axis_pos {
629            Some(p) => {
630                if let Some(Some(i)) = c.axis_map.get(p) {
631                    hidden.insert(*i, f);
632                }
633            }
634            None => {
635                if let Some(i) = find_db(&f.source_col)
636                    && !pages.iter().any(|(j, _)| *j == i)
637                {
638                    pages.push((i, f));
639                    hidden.insert(i, f);
640                }
641            }
642        }
643    }
644    let find_calc = |name: &str| c.fields.iter().position(|f| f.formula.is_some() && same(&f.name, name));
645    let values: Vec<(usize, &PivotValue)> =
646        pt.values.iter().filter_map(|v| find_db(&v.source_col).or_else(|| find_calc(&v.source_col)).map(|i| (i, v))).collect();
647    let layout = PivotTableLayout { row_fields, column_fields, hidden_fields: hidden, page_fields: pages, values };
648
649    let compact = pt.layout == PivotLayout::Compact;
650    let name = if pt.name.trim().is_empty() { format!("PivotTable{}", idx + 1) } else { pt.name.clone() };
651    let mut s = format!(
652        "{XML_DECL}<pivotTableDefinition xmlns=\"{NS_MAIN}\" name=\"{}\" cacheId=\"{cache_id}\" applyNumberFormats=\"0\" applyBorderFormats=\"0\" applyFontFormats=\"0\" applyPatternFormats=\"0\" applyAlignmentFormats=\"0\" applyWidthHeightFormats=\"1\" dataCaption=\"Values\" updatedVersion=\"6\" minRefreshableVersion=\"3\" useAutoFormatting=\"1\" itemPrintTitles=\"1\" createdVersion=\"6\" indent=\"0\"",
653        esc_attr(&name)
654    );
655    match pt.layout {
656        PivotLayout::Compact => s.push_str(" outline=\"1\" outlineData=\"1\""),
657        PivotLayout::Outline => s.push_str(" compact=\"0\" compactData=\"0\" outline=\"1\" outlineData=\"1\""),
658        PivotLayout::Tabular => s.push_str(" compact=\"0\" compactData=\"0\""),
659    }
660    if pt.values_on_rows {
661        s.push_str(" dataOnRows=\"1\"");
662    }
663    if !pt.show_headers {
664        s.push_str(" showHeaders=\"0\"");
665    }
666    if !pt.enable_show_details {
667        s.push_str(" enableDrill=\"0\"");
668    }
669    // `rowGrandTotals` is the grand total row below the report (Excel's "On for Columns"),
670    // `colGrandTotals` the grand total column at its right ("On for Rows").
671    if !pt.grand_totals_cols {
672        s.push_str(" rowGrandTotals=\"0\"");
673    }
674    if !pt.grand_totals_rows {
675        s.push_str(" colGrandTotals=\"0\"");
676    }
677    s.push_str(" multipleFieldFilters=\"0\">");
678
679    // Location: the body (report filters sit above it).
680    let a = pt.anchor;
681    let end = pt.last_range.map(|r| r.end).unwrap_or(a);
682    let loc = RangeRef::new(a, CellRef::new(end.row.max(a.row), end.col.max(a.col)));
683    let k = layout.values.len();
684    // Σ Values (two or more value fields): its index among the written fields of its axis.
685    let (row_map, _) = c.axis_map.split_at(pt.rows.len().min(c.axis_map.len()));
686    let values_at = |map: &[Option<usize>], n: usize| pt.values_position.map_or(n, |p| map.iter().take(p).filter(|x| x.is_some()).count());
687    let row_values = (k > 1 && pt.values_on_rows).then(|| values_at(row_map, layout.row_fields.len()));
688    // In Columns Σ Values comes after the last field: the column layout doesn't place it elsewhere.
689    let col_values = (k > 1 && !pt.values_on_rows).then_some(layout.column_fields.len());
690    let first_data_row = if layout.column_fields.is_empty() { 1 } else { 1 + layout.column_fields.len() + usize::from(col_values.is_some()) };
691    let row_levels = layout.row_fields.len() + usize::from(row_values.is_some());
692    let first_data_col = if row_levels == 0 && layout.column_fields.is_empty() {
693        0
694    } else if compact {
695        1
696    } else {
697        row_levels.max(1)
698    };
699    // Two or more value fields in Columns alone: Excel 365 would add a `Values` caption row.
700    let hide_values_row = col_values.is_some() && layout.column_fields.is_empty();
701    let first_header_row = usize::from(!hide_values_row);
702    let _ = write!(
703        s,
704        "<location ref=\"{}\" firstHeaderRow=\"{first_header_row}\" firstDataRow=\"{first_data_row}\" firstDataCol=\"{first_data_col}\"",
705        loc.a1()
706    );
707    if !layout.page_fields.is_empty() {
708        let _ = write!(s, " rowPageCount=\"{}\" colPageCount=\"1\"", layout.page_fields.len());
709    }
710    s.push_str("/>");
711
712    write_pivot_fields(&mut s, pt, c, &layout);
713    axis_fields_xml(&mut s, "rowFields", &layout.row_fields, row_values);
714    axis_fields_xml(&mut s, "colFields", &layout.column_fields, col_values);
715    write_page_fields(&mut s, c, &layout);
716    write_data_fields(&mut s, c, &layout, fmt_id);
717    s.push_str("<pivotTableStyleInfo");
718    if !pt.style.is_empty() {
719        let _ = write!(s, " name=\"{}\"", esc_attr(&pt.style));
720    }
721    s.push_str(" showRowHeaders=\"1\" showColHeaders=\"1\" showRowStripes=\"0\" showColStripes=\"0\" showLastColumn=\"1\"/>");
722    if hide_values_row {
723        let _ = write!(
724            s,
725            "<extLst><ext uri=\"{EXT_X14_PIVOT_TABLE}\" xmlns:x14=\"{NS_X14}\"><x14:pivotTableDefinition hideValuesRow=\"1\"/></ext></extLst>"
726        );
727    }
728    s.push_str("</pivotTableDefinition>");
729    s
730}

732fn axis_field<'a>(pt: &'a PivotTable, c: &Cache, i: usize) -> Option<&'a PivotField> {
733    let p = c.axis_map.iter().position(|x| *x == Some(i))?;
734    pt.rows.iter().chain(pt.columns.iter()).nth(p)
735}

737fn write_pivot_fields(s: &mut String, pt: &PivotTable, c: &Cache, layout: &PivotTableLayout<'_>) {
738    let _ = write!(s, "<pivotFields count=\"{}\">", c.fields.len());
739    for (i, f) in c.fields.iter().enumerate() {
740        s.push_str("<pivotField");
741        let axis = if layout.row_fields.contains(&i) {
742            Some("axisRow")
743        } else if layout.column_fields.contains(&i) {
744            Some("axisCol")
745        } else if layout.page_fields.iter().any(|(j, _)| *j == i) {
746            Some("axisPage")
747        } else {
748            None
749        };
750        if let Some(ax) = axis {
751            let _ = write!(s, " axis=\"{ax}\"");
752        }
753        if layout.values.iter().any(|(j, _)| *j == i) {
754            s.push_str(" dataField=\"1\"");
755        }
756        if f.formula.is_some() {
757            s.push_str(" dragToRow=\"0\" dragToCol=\"0\" dragToPage=\"0\"");
758        }
759        match pt.layout {
760            PivotLayout::Compact => {}
761            PivotLayout::Outline => s.push_str(" compact=\"0\""),
762            PivotLayout::Tabular => s.push_str(" compact=\"0\" outline=\"0\""),
763        }
764        if f.group.is_some() {
765            s.push_str(" numFmtId=\"14\"");
766        }
767        let filter = layout.hidden_fields.get(&i).copied();
768        let labels = f.item_labels();
769        let is_page = axis == Some("axisPage");
770        // Hidden items; never hide every item (a field needs one visible item).
771        let mut hide: Vec<bool> = match filter.and_then(|f| f.selected.as_ref()) {
772            Some(sel) if !(is_page && sel.len() == 1) => labels.iter().map(|l| !sel.iter().any(|x| same(x, l))).collect(),
773            _ => vec![false; labels.len()],
774        };
775        if !hide.is_empty() && hide.iter().all(|h| *h) {
776            hide.iter_mut().for_each(|h| *h = false);
777        }
778        if is_page && hide.iter().any(|h| *h) {
779            s.push_str(" multipleItemSelectionAllowed=\"1\"");
780        }
781        s.push_str(" showAll=\"0\"");
782        if let Some(af) = axis_field(pt, c, i) {
783            match af.sort {
784                PivotSort::Asc => s.push_str(" sortType=\"ascending\""),
785                PivotSort::Desc => s.push_str(" sortType=\"descending\""),
786                PivotSort::None => {}
787            }
788        }
789        if !pt.subtotals {
790            s.push_str(" defaultSubtotal=\"0\"");
791        }
792        if !pt.subtotals_top {
793            s.push_str(" subtotalTop=\"0\"");
794        }
795        if axis.is_some() {
796            let collapsed: &[String] = axis_field(pt, c, i).map(|f| f.collapsed_items.as_slice()).unwrap_or(&[]);
797            let count = labels.len() + usize::from(pt.subtotals);
798            let _ = write!(s, "><items count=\"{count}\">");
799            for (x, l) in labels.iter().enumerate() {
800                let _ = write!(s, "<item x=\"{x}\"");
801                if hide.get(x).copied().unwrap_or(false) {
802                    s.push_str(" h=\"1\"");
803                }
804                if collapsed.iter().any(|cl| same(cl, l)) {
805                    s.push_str(" sd=\"0\"");
806                }
807                s.push_str("/>");
808            }
809            if pt.subtotals {
810                s.push_str("<item t=\"default\"/>");
811            }
812            s.push_str("</items></pivotField>");
813        } else {
814            s.push_str("/>");
815        }
816    }
817    s.push_str("</pivotFields>");
818}

820fn write_page_fields(s: &mut String, c: &Cache, layout: &PivotTableLayout<'_>) {
821    if layout.page_fields.is_empty() {
822        return;
823    }
824    let _ = write!(s, "<pageFields count=\"{}\">", layout.page_fields.len());
825    for (i, filter) in &layout.page_fields {
826        let _ = write!(s, "<pageField fld=\"{i}\"");
827        if let Some(selected) = &filter.selected
828            && selected.len() == 1
829            && let Some(one) = selected.first()
830            && let Some(x) = c.fields.get(*i).and_then(|field| field.item_labels().iter().position(|label| same(label, one)))
831        {
832            let _ = write!(s, " item=\"{x}\"");
833        }
834        s.push_str(" hier=\"-1\"/>");
835    }
836    s.push_str("</pageFields>");
837}

839fn write_data_fields(s: &mut String, c: &Cache, layout: &PivotTableLayout<'_>, fmt_id: &dyn Fn(&str) -> Option<u32>) {
840    if layout.values.is_empty() {
841        return;
842    }
843    let _ = write!(s, "<dataFields count=\"{}\">", layout.values.len());
844    let mut used: Vec<String> = Vec::new();
845    for (i, value) in &layout.values {
846        let base = if value.name.trim().is_empty() {
847            format!("{} of {}", func_caption(value.func), c.fields.get(*i).map(|f| f.name.as_str()).unwrap_or(""))
848        } else {
849            value.name.clone()
850        };
851        let mut name = base.clone();
852        let mut suffix = 2;
853        while used.iter().any(|existing| same(existing, &name)) {
854            name = format!("{base}{suffix}");
855            suffix += 1;
856        }
857        used.push(name.clone());
858        let _ = write!(s, "<dataField name=\"{}\" fld=\"{i}\"", esc_attr(&name));
859        if value.func != PivotFunc::Sum {
860            let _ = write!(s, " subtotal=\"{}\"", func_name(value.func));
861        }
862        let base_field = layout.row_fields.first().or(layout.column_fields.first()).copied().unwrap_or(0);
863        match value.show_as {
864            PivotShowAs::Normal => s.push_str(" baseField=\"0\" baseItem=\"0\""),
865            PivotShowAs::PercentOfGrandTotal => s.push_str(" showDataAs=\"percentOfTotal\" baseField=\"0\" baseItem=\"0\""),
866            PivotShowAs::PercentOfColumnTotal => s.push_str(" showDataAs=\"percentOfCol\" baseField=\"0\" baseItem=\"0\""),
867            PivotShowAs::PercentOfRowTotal => s.push_str(" showDataAs=\"percentOfRow\" baseField=\"0\" baseItem=\"0\""),
868            PivotShowAs::RunningTotal => {
869                let _ = write!(s, " showDataAs=\"runTotal\" baseField=\"{base_field}\" baseItem=\"0\"");
870            }
871            PivotShowAs::Rank => {
872                let _ = write!(s, " baseField=\"{base_field}\" baseItem=\"0\"");
873            }
874        }
875        if let Some(id) = value.number_format.as_deref().and_then(fmt_id) {
876            let _ = write!(s, " numFmtId=\"{id}\"");
877        }
878        if value.show_as == PivotShowAs::Rank {
879            let _ = write!(
880                s,
881                "><extLst><ext uri=\"{EXT_X14_DATA_FIELD}\" xmlns:x14=\"{NS_X14}\"><x14:dataField pivotShowAs=\"rankDescending\"/></ext></extLst></dataField>"
882            );
883        } else {
884            s.push_str("/>");
885        }
886    }
887    s.push_str("</dataFields>");
888}

890/// `<rowFields>`/`<colFields>`: the axis's cache field indices, with Σ Values (`-2`) at `values_at`.
891fn axis_fields_xml(s: &mut String, tag: &str, idx: &[usize], values_at: Option<usize>) {
892    let n = idx.len() + usize::from(values_at.is_some());
893    if n == 0 {
894        return;
895    }
896    let _ = write!(s, "<{tag} count=\"{n}\">");
897    for (j, i) in idx.iter().enumerate() {
898        if values_at == Some(j) {
899            s.push_str("<field x=\"-2\"/>");
900        }
901        let _ = write!(s, "<field x=\"{i}\"/>");
902    }
903    if values_at.is_some_and(|p| p >= idx.len()) {
904        s.push_str("<field x=\"-2\"/>");
905    }
906    let _ = write!(s, "</{tag}>");
