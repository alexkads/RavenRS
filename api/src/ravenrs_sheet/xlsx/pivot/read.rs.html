8/// A cache field as read.
9#[derive(Clone, Debug, Default)]
10struct FieldIn {
11    name: String,
12    database: bool,
13    /// Shared item captions.
14    items: Vec<String>,
15    /// (base field, grouping, group item captions); grouping `None` = unsupported grouping.
16    group: Option<(Option<usize>, Option<PivotDateGroup>, Vec<String>)>,
17    /// A calculated field's formula.
18    formula: Option<String>,
19}
20
21impl FieldIn {
22    fn labels(&self) -> &[String] {
23        match &self.group {
24            Some((_, _, items)) if !items.is_empty() => items,
25            _ => &self.items,
26        }
27    }
28}

36/// Workbook-level pivot cache lookup, filled from `<pivotCaches>`; caches are parsed once.
37#[derive(Default)]
38pub struct CachesIn {
39    by_id: HashMap<u32, String>,
40    parsed: HashMap<String, Option<CacheIn>>,
41    next_id: u32,
42}
43
44impl CachesIn {
45    pub fn new(wb_xml: &El, wb_rels: &[crate::xlsx::package::Rel]) -> CachesIn {
46        let mut by_id = HashMap::new();
47        if let Some(pc) = wb_xml.child("pivotCaches") {
48            for c in pc.kids("pivotCache") {
49                if let (Some(id), Some(rid)) = (c.attr_u32("cacheId"), c.attr("id"))
50                    && let Some(r) = wb_rels.iter().find(|r| r.id == rid)
51                {
52                    by_id.insert(id, r.target.clone());
53                }
54            }
55        }
56        CachesIn { by_id, parsed: HashMap::new(), next_id: 1 }
57    }
58}

60fn item_label_el(e: &El, sys: DateSystem) -> String {
61    let v = e.attr("v").unwrap_or("");
62    match e.name.as_str() {
63        "m" => "(blank)".into(),
64        "n" => v.trim().parse::<f64>().ok().filter(|n| n.is_finite()).map(number_to_text).unwrap_or_else(|| v.to_string()),
65        "b" => if matches!(v.trim(), "1" | "true") { "TRUE" } else { "FALSE" }.into(),
66        "d" => crate::xlsx::sheet_read::parse_iso_datetime(v, sys).map(number_to_text).unwrap_or_else(|| v.to_string()),
67        _ => v.to_string(),
68    }
69}

71fn read_cache(cx: &mut Ctx<'_>, part: &str, sheet_name: &str) -> Option<CacheIn> {
72    let x = cx.optional_xml(part).ok().flatten()?;
73    if x.name != "pivotCacheDefinition" {
74        cx.warn(format!("{part} is not a pivot cache definition; its PivotTable was kept as plain cells"));
75        return None;
76    }
77    let src = x.child("cacheSource");
78    let ty = src.and_then(|s| s.attr("type")).unwrap_or("worksheet");
79    if ty != "worksheet" {
80        cx.warn(format!("PivotTables with a {ty} data source are not supported; kept as plain cells"));
81        return None;
82    }
83    let ws = src.and_then(|s| s.child("worksheetSource"));
84    let source = match ws {
85        Some(w) if w.attr("name").is_some_and(|n| !n.trim().is_empty()) => w.attr("name").unwrap_or("").trim().to_string(),
86        Some(w) => {
87            if w.attr("id").is_some() {
88                cx.warn("a PivotTable's source is in another workbook; its source reference may not resolve");
89            }
90            let r = w.attr("ref").map(|r| r.replace('$', "")).and_then(|r| RangeRef::parse(&r));
91            let Some(r) = r else {
92                cx.warn("a PivotTable's source reference is missing or invalid; kept as plain cells");
93                return None;
94            };
95            let sheet = w.attr("sheet").filter(|s| !s.is_empty()).unwrap_or(sheet_name);
96            format!("{}!{}", quote_sheet(sheet), abs_range(&r))
97        }
98        None => {
99            cx.warn("a PivotTable's cache has no worksheet source; kept as plain cells");
100            return None;
101        }
102    };
103    let sys = cx.date_system;
104    let mut fields = Vec::new();
105    if let Some(cf) = x.child("cacheFields") {
106        for f in cf.kids("cacheField").take(MAX_FIELDS) {
107            let items = f.child("sharedItems").map(|si| si.children.iter().map(|e| item_label_el(e, sys)).collect()).unwrap_or_default();
108            let group = f.child("fieldGroup").map(|g| {
109                let base = g.attr_u32("base").map(|b| b as usize);
110                let by = match g.child("rangePr").map(|r| r.attr("groupBy").unwrap_or("range")) {
111                    Some("years") => Some(PivotDateGroup::Years),
112                    Some("quarters") => Some(PivotDateGroup::Quarters),
113                    Some("months") => Some(PivotDateGroup::Months),
114                    Some("days") => Some(PivotDateGroup::Days),
115                    _ => None,
116                };
117                let gi = g.child("groupItems").map(|gi| gi.children.iter().map(|e| item_label_el(e, sys)).collect()).unwrap_or_default();
118                (base, by, gi)
119            });
120            fields.push(FieldIn {
121                name: f.attr("name").unwrap_or("").to_string(),
122                database: f.flag("databaseField", true),
123                items,
124                group,
125                formula: f.attr("formula").map(str::to_string),
126            });
127        }
128    }
129    if fields.is_empty() {
130        cx.warn("a PivotTable's cache has no fields; kept as plain cells");
131        return None;
132    }
133    Some(CacheIn { source, fields })
134}

136/// Reads the PivotTables related to a worksheet part.
137pub fn read_pivots(cx: &mut Ctx<'_>, caches: &mut CachesIn, sheet_part: &str, sheet_name: &str) -> Result<Vec<PivotTable>, IoError> {
138    let rels = cx.pkg.rels(sheet_part)?;
139    let mut out = Vec::new();
140    for rel in rels.iter().filter(|r| r.kind == "pivotTable" && !r.external) {
141        let Some(x) = cx.optional_xml(&rel.target)? else { continue };
142        if x.name != "pivotTableDefinition" {
143            cx.warn(format!("{} is not a PivotTable definition; skipped", rel.target));
144            continue;
145        }
146        // The cache: the table's own relationship, else the workbook's cache id.
147        let trels = cx.pkg.rels(&rel.target)?;
148        let cache_part = trels
149            .iter()
150            .find(|r| r.kind == "pivotCacheDefinition" && !r.external)
151            .map(|r| r.target.clone())
152            .or_else(|| x.attr_u32("cacheId").and_then(|id| caches.by_id.get(&id).cloned()));
153        let Some(cache_part) = cache_part else {
154            cx.warn(format!("PivotTable in {} has no cache; kept as plain cells", rel.target));
155            continue;
156        };
157        let cache = match caches.parsed.get(&cache_part) {
158            Some(c) => c.clone(),
159            None => {
160                let c = read_cache(cx, &cache_part, sheet_name);
161                caches.parsed.insert(cache_part.clone(), c.clone());
162                c
163            }
164        };
165        let Some(cache) = cache else { continue };
166        if let Some(pt) = read_table(cx, &x, &cache, caches.next_id) {
167            caches.next_id = caches.next_id.saturating_add(1);
168            out.push(pt);
169        }
170    }
171    Ok(out)
172}

174/// A pivot field's data items (not subtotal or grand-total items).
175fn items_of(e: &El) -> Vec<&El> {
176    e.child("items").map(|it| it.kids("item").filter(|i| i.attr("t").is_none_or(|t| t == "data")).collect()).unwrap_or_default()
177}

179/// (source column name, date grouping) of a cache field used on an axis.
180fn field_ref(cx: &mut Ctx<'_>, cache: &CacheIn, i: usize) -> Option<(String, PivotDateGroup)> {
181    let f = cache.fields.get(i)?;
182    if f.formula.is_some() {
183        cx.warn("calculated PivotTable fields are not supported");
184        return None;
185    }
186    match &f.group {
187        Some((base, by, _)) => {
188            let by = match by {
189                Some(b) => *b,
190                None => {
191                    cx.warn(format!("PivotTable grouping of \"{}\" is not supported; the field is ungrouped", f.name));
192                    PivotDateGroup::None
193                }
194            };
195            let base_name = if f.database { f.name.clone() } else { base.and_then(|b| cache.fields.get(b))?.name.clone() };
196            Some((base_name, by))
197        }
198        None if !f.database => {
199            cx.warn(format!("PivotTable field \"{}\" (grouped items) is not supported", f.name));
200            None
201        }
202        None => Some((f.name.clone(), PivotDateGroup::None)),
203    }
204}

206fn read_table(cx: &mut Ctx<'_>, x: &El, cache: &CacheIn, id: u32) -> Option<PivotTable> {
207    let mut pt = PivotTable { id, ..PivotTable::default() };
208    if let Some(n) = x.attr("name").filter(|n| !n.trim().is_empty()) {
209        pt.name = n.to_string();
210    } else {
211        pt.name = format!("PivotTable{id}");
212    }
213    pt.source = cache.source.clone();
214    let Some(loc) = x.child("location").and_then(|l| l.attr("ref")).and_then(|r| RangeRef::parse(&r.replace('$', ""))) else {
215        cx.warn(format!("PivotTable \"{}\" has no valid location; kept as plain cells", pt.name));
216        return None;
217    };
218    let pfs: Vec<&El> = x.child("pivotFields").map(|p| p.kids("pivotField").collect()).unwrap_or_default();
219    let pf = |i: usize| pfs.get(i).copied();
220    let idx_list = |name: &str, attr: &str| -> Vec<i64> {
221        x.child(name).map(|e| e.children.iter().filter_map(|f| f.attr_i64(attr)).collect()).unwrap_or_default()
222    };
223    let row_x = idx_list("rowFields", "x");
224    let col_x = idx_list("colFields", "x");
225    let pages: Vec<&El> = x.child("pageFields").map(|p| p.kids("pageField").collect()).unwrap_or_default();
226
227    // Layout from the first axis field (pivotField defaults: compact and outline on).
228    let first_axis = row_x.iter().chain(col_x.iter()).find(|i| **i >= 0).and_then(|i| pf(*i as usize));
229    let (compact, outline) = match first_axis {
230        Some(f) => (f.flag("compact", true), f.flag("outline", true)),
231        None => (x.flag("compact", true), x.flag("outline", false)),
232    };
233    pt.layout = if compact {
234        PivotLayout::Compact
235    } else if outline {
236        PivotLayout::Outline
237    } else {
238        PivotLayout::Tabular
239    };
240    if let Some(f) = first_axis {
241        pt.subtotals = f.flag("defaultSubtotal", true);
242        pt.subtotals_top = f.flag("subtotalTop", true);
243    }
244    pt.grand_totals_rows = x.flag("colGrandTotals", true);
245    pt.grand_totals_cols = x.flag("rowGrandTotals", true);
246    pt.show_headers = x.flag("showHeaders", true);
247    pt.enable_show_details = x.flag("enableDrill", true);
248    pt.style = x.child("pivotTableStyleInfo").and_then(|s| s.attr("name")).unwrap_or("").to_string();
249    if x.child("calculatedItems").is_some() || x.path(&["filters", "filter"]).is_some() {
250        cx.warn(format!("PivotTable \"{}\": calculated items and label/value filters are not supported", pt.name));
251    }
252
253    let mut filters: Vec<PivotFilter> = Vec::new();
254
255    // (fields, position of Σ Values (`-2`) among them)
256    let axis = |cx: &mut Ctx<'_>, list: &[i64], filters: &mut Vec<PivotFilter>| -> (Vec<PivotField>, Option<usize>) {
257        let mut v = Vec::new();
258        let mut values_at = None;
259        for &i in list {
260            if i == -2 {
261                values_at = Some(v.len());
262                continue;
263            }
264            if i < 0 {
265                continue;
266            }
267            let i = i as usize;
268            let Some((name, group)) = field_ref(cx, cache, i) else { continue };
269            let labels = cache.fields.get(i).map(|f| f.labels().to_vec()).unwrap_or_default();
270            let mut f = PivotField { source_col: name.clone(), date_group: group, sort: PivotSort::None, collapsed_items: vec![] };
271            if let Some(e) = pf(i) {
272                f.sort = match e.attr("sortType") {
273                    Some("ascending") => PivotSort::Asc,
274                    Some("descending") => PivotSort::Desc,
275                    _ => PivotSort::None,
276                };
277                let items = items_of(e);
278                let lab = |it: &El| it.attr_u32("x").and_then(|k| labels.get(k as usize)).cloned();
279                f.collapsed_items = items.iter().filter(|it| !it.flag("sd", true)).filter_map(|it| lab(it)).collect();
280                if items.iter().any(|it| it.flag("h", false)) {
281                    let shown: Vec<String> = items.iter().filter(|it| !it.flag("h", false)).filter_map(|it| lab(it)).collect();
282                    if !filters.iter().any(|ff| same(&ff.source_col, &name)) {
283                        filters.push(PivotFilter { source_col: name.clone(), selected: Some(shown) });
284                    }
285                }
286            }
287            v.push(f);
288        }
289        (v, values_at)
290    };
291    let mut page_filters = Vec::new();
292    for p in &pages {
293        let Some(i) = p.attr_i64("fld").filter(|i| *i >= 0).map(|i| i as usize) else { continue };
294        let Some((name, _)) = field_ref(cx, cache, i) else { continue };
295        let labels = cache.fields.get(i).map(|f| f.labels().to_vec()).unwrap_or_default();
296        let selected = if let Some(k) = p.attr_u32("item") {
297            labels.get(k as usize).map(|l| vec![l.clone()])
298        } else {
299            pf(i).and_then(|e| {
300                let items = items_of(e);
301                items.iter().any(|it| it.flag("h", false)).then(|| {
302                    items
303                        .iter()
304                        .filter(|it| !it.flag("h", false))
305                        .filter_map(|it| it.attr_u32("x").and_then(|k| labels.get(k as usize)).cloned())
306                        .collect()
307                })
308            })
309        };
310        page_filters.push(PivotFilter { source_col: name, selected });
311    }
312    let mut axis_filters = Vec::new();
313    let (rows, rows_values) = axis(cx, &row_x, &mut axis_filters);
314    let (columns, cols_values) = axis(cx, &col_x, &mut axis_filters);
315    pt.rows = rows;
316    pt.columns = columns;
317    // Σ Values: where its `-2` entry is, else `dataOnRows` and `dataPosition`; after the last
318    // field by default. In Columns it always comes last, as the column layout puts it there.
319    pt.values_on_rows = rows_values.is_some() || (cols_values.is_none() && x.flag("dataOnRows", false));
320    pt.values_position =
321        if pt.values_on_rows { rows_values.or_else(|| x.attr_u32("dataPosition").map(|p| p as usize)).filter(|p| *p < pt.rows.len()) } else { None };
322    filters.extend(page_filters);
323    filters.extend(axis_filters);
324    pt.filters = filters;
325
326    // Values.
327    if let Some(df) = x.child("dataFields") {
328        for d in df.kids("dataField") {
329            let Some(i) = d.attr_u32("fld").map(|i| i as usize) else { continue };
330            let Some(f) = cache.fields.get(i) else { continue };
331            let func = match d.attr("subtotal") {
332                None => PivotFunc::Sum,
333                Some(s) => func_from(s).unwrap_or_else(|| {
334                    cx.warn(format!("PivotTable function {s} is not supported; using Sum"));
335                    PivotFunc::Sum
336                }),
337            };
338            let ext = d
339                .child("extLst")
340                .into_iter()
341                .flat_map(|l| l.kids("ext"))
342                .filter_map(|e| e.child("dataField"))
343                .find_map(|e| e.attr("pivotShowAs").map(str::to_string));
344            let show_as = match (ext.as_deref(), d.attr("showDataAs")) {
345                (Some("rankDescending"), _) => PivotShowAs::Rank,
346                (Some(other), _) => {
347                    cx.warn(format!("PivotTable show-values-as {other} is not supported"));
348                    PivotShowAs::Normal
349                }
350                (None, None | Some("normal")) => PivotShowAs::Normal,
351                (None, Some("percentOfTotal")) => PivotShowAs::PercentOfGrandTotal,
352                (None, Some("percentOfCol")) => PivotShowAs::PercentOfColumnTotal,
353                (None, Some("percentOfRow")) => PivotShowAs::PercentOfRowTotal,
354                (None, Some("runTotal")) => PivotShowAs::RunningTotal,
355                (None, Some(other)) => {
356                    cx.warn(format!("PivotTable show-values-as {other} is not supported"));
357                    PivotShowAs::Normal
358                }
359            };
360            let number_format = d
361                .attr_u32("numFmtId")
362                .filter(|id| *id != 0)
363                .and_then(|id| crate::xlsx::tables::builtin_format(id).map(str::to_string).or_else(|| cx.num_fmts.get(&id).cloned()));
364            let name = d.attr("name").map(str::to_string).unwrap_or_else(|| format!("{} of {}", func_caption(func), f.name));
365            pt.values.push(PivotValue { source_col: f.name.clone(), func, name, show_as, number_format });
366        }
367    }
368
369    // Calculated fields of the cache (each PivotTable on it can use them).
370    for f in &cache.fields {
371        if let Some(formula) = &f.formula
372            && !f.name.trim().is_empty()
373            && !pt.calculated_fields.iter().any(|c| same(&c.name, &f.name))
374        {
375            let formula = formula.trim();
376            let formula = if formula.starts_with('=') { formula.to_string() } else { format!("={formula}") };
377            pt.calculated_fields.push(PivotCalcField { name: f.name.clone(), formula });
378        }
379    }
380
381    pt.anchor = loc.start;
382    let above = if pages.is_empty() { 0 } else { pages.len() as u32 + 1 };
383    pt.last_range = Some(RangeRef::new(CellRef::new(loc.start.row.saturating_sub(above), loc.start.col), loc.end));
384    Some(pt)
385}
