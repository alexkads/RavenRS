1//! Equations in drawings: Office Math Markup (ECMA-376 Part 1 §22.1) read back into LaTeX for the
2//! common subset, the inverse of `crate::math::to_omml`. A construct outside the subset makes
3//! the whole equation unreadable as maths (`None`), and the caller keeps it as a text box.
4
5use crate::xlsx::xml::El;
6
7/// Deepest OMML nesting converted.
8const MAX_DEPTH: usize = 64;
9/// Longest LaTeX the conversion produces before it gives up. `crate::math::check` refuses a
10/// longer source anyway, so converting the rest of a wide or long equation would only spend time
11/// and memory on a hostile file.
12const MAX_OUT_BYTES: usize = crate::math::MAX_SOURCE_BYTES;
13
14/// The LaTeX for an `m:oMath` (or `m:oMathPara`) element, if every part is in the subset and the
15/// result lays out.
16pub fn to_latex(e: &El) -> Option<String> {
17    let out = convert(e)?;
18    (!out.is_empty() && crate::math::check(&out).is_ok()).then_some(out)
19}
20
21/// The conversion itself, before the final check: `None` for a construct outside the subset, an
22/// equation nested past [`MAX_DEPTH`] or one whose LaTeX exceeds [`MAX_OUT_BYTES`].
23fn convert(e: &El) -> Option<String> {
24    let mut out = String::new();
25    match e.name.as_str() {
26        "oMathPara" => {
27            let maths: Vec<&El> = e.kids("oMath").collect();
28            for (i, m) in maths.iter().enumerate() {
29                if i > 0 {
30                    out.push_str(" \\\\ ");
31                }
32                seq(m, &mut out, 0)?;
33            }
34        }
35        "oMath" => seq(e, &mut out, 0)?,
36        _ => return None,
37    }
38    Some(out.trim().to_string())
39}

41/// The first `m:oMath`/`m:oMathPara` under `e` (an `a14:m` element or a paragraph), searched to a
42/// bounded depth.
43pub fn find_math(e: &El, depth: usize) -> Option<&El> {
44    if depth > 8 {
45        return None;
46    }
47    if e.name == "oMathPara" || e.name == "oMath" {
48        return Some(e);
49    }
50    e.children.iter().find_map(|c| find_math(c, depth + 1))
51}

53/// The LaTeX of a DrawingML text body (`a:txBody`) that holds one equation and nothing else: no
54/// text in its runs or fields, and no second `m:oMath`/`m:oMathPara`. A body that mixes text and
55/// maths, or holds several equations, is `None`, so the caller keeps it as a text box (or reads
56/// the writer's fallback) instead of dropping the rest.
57pub fn equation_source(tb: &El) -> Option<String> {
58    let has_text = tb
59        .kids("p")
60        .any(|p| p.children.iter().any(|c| matches!(c.name.as_str(), "r" | "fld") && c.child("t").is_some_and(|t| !t.text.trim().is_empty())));
61    if has_text || count_math(tb, 0) != 1 {
62        return None;
63    }
64    find_math(tb, 0).and_then(to_latex)
65}

67/// How many `m:oMath`/`m:oMathPara` elements are under `e` (not counting those inside one), to
68/// the same bounded depth as [`find_math`], stopping at two.
69fn count_math(e: &El, depth: usize) -> usize {
70    if depth > 8 {
71        return 0;
72    }
73    if e.name == "oMathPara" || e.name == "oMath" {
74        return 1;
75    }
76    let mut n = 0;
77    for c in &e.children {
78        n += count_math(c, depth + 1);
79        if n > 1 {
80            break;
81        }
82    }
83    n
84}

86/// A marker for the equation a text body holds: 16 hex digits of an FNV-1a 64-bit hash over what
87/// carries the equation's meaning, the OMML element names in document order, the text of the
88/// runs, and the properties the converter reads ([`PROPERTIES`]: a run's `sty`, a delimiter's
89/// characters, an operator's limits and so on). The writer puts it in the shape's description
90/// beside the LaTeX source; the reader takes the source only while the marker still matches the
91/// body, so an equation whose Office Math another application edited (Excel leaves the
92/// description alone) is converted from that Office Math instead of coming back as the old
93/// source. What a re-serialisation may change without changing the equation is left out: every
94/// other property (`ctrlPr`, DrawingML run properties such as size and colour), extension lists,
95/// the DrawingML containers (`txBody`, `p`, `r`, `fld`, `lstStyle`), attributes, namespace
96/// prefixes and whitespace between elements; adjacent runs without read properties are joined,
97/// so splitting or merging them changes nothing, while a run with one is a unit of its own, as
98/// the converter converts it (it recognises function names per run). A text change, an added
99/// run, a structural change, a change to a property the converter reads or a styled run split in
100/// two does.
101pub fn body_marker(tb: &El) -> String {
102    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
103    let mut text = String::new();
104    mark(tb, &mut h, &mut text, 0);
105    mark_text(&mut h, &mut text);
106    format!("{h:016x}")
107}
108
109fn mark(e: &El, h: &mut u64, text: &mut String, depth: usize) {
110    if depth > MAX_DEPTH {
111        return;
112    }
113    for c in &e.children {
114        if c.name.ends_with("Pr") {
115            // Only the properties the converter reads, each as `name=val`, in a fixed order; they
116            // close the run text gathered so far, so a property moving between runs is a change.
117            let read: Vec<(&str, &str)> =
118                read_properties(&c.name).iter().filter_map(|n| c.child(n).map(|p| (*n, p.attr("val").unwrap_or("")))).collect();
119            if !read.is_empty() {
120                mark_text(h, text);
121                mark_bytes(h, b"[");
122                mark_bytes(h, c.name.as_bytes());
123                for (n, v) in read {
124                    mark_bytes(h, b" ");
125                    mark_bytes(h, n.as_bytes());
126                    mark_bytes(h, b"=");
127                    mark_bytes(h, v.as_bytes());
128                }
129                mark_bytes(h, b"]");
130            }
131            continue;
132        }
133        if c.name == "extLst" {
134            continue;
135        }
136        match c.name.as_str() {
137            "r" => {
138                // A run with a property the converter reads is converted on its own (a function
139                // name such as `sin` is recognised per run), so its text is hashed as a unit: the
140                // property closes the text before it, and the run's end closes its own. Runs
141                // without such properties convert character by character and stay joined.
142                let styled = c.children.iter().any(|p| p.name.ends_with("Pr") && read_properties(&p.name).iter().any(|n| p.child(n).is_some()));
143                mark(c, h, text, depth + 1);
144                if styled {
145                    mark_text(h, text);
146                }
147            }
148            "txBody" | "p" | "fld" | "lstStyle" => mark(c, h, text, depth + 1),
149            "t" => text.push_str(&c.text),
150            "br" => text.push('\n'),
151            name => {
152                mark_text(h, text);
153                mark_bytes(h, b"<");
154                mark_bytes(h, name.as_bytes());
155                mark(c, h, text, depth + 1);
156                mark_text(h, text);
157                mark_bytes(h, b">");
158            }
159        }
160    }
161}

163/// Hashes the run text gathered so far, if any, and clears it.
164fn mark_text(h: &mut u64, text: &mut String) {
165    if !text.is_empty() {
166        mark_bytes(h, b"\"");
167        mark_bytes(h, text.as_bytes());
168        mark_bytes(h, b"\"");
169        text.clear();
170    }
171}
172
173fn mark_bytes(h: &mut u64, bytes: &[u8]) {
174    for b in bytes {
175        *h ^= u64::from(*b);
176        *h = h.wrapping_mul(0x0000_0100_0000_01b3);
177    }
178}

180/// Children in order, skipping property elements.
181fn seq(e: &El, out: &mut String, depth: usize) -> Option<()> {
182    if depth > MAX_DEPTH {
183        return None;
184    }
185    for c in &e.children {
186        node(c, out, depth + 1)?;
187        if out.len() > MAX_OUT_BYTES {
188            return None;
189        }
190    }
191    Some(())
192}

194/// `{…}` around a child's contents (empty group when the child is missing).
195fn group(e: Option<&El>, out: &mut String, depth: usize) -> Option<()> {
196    out.push('{');
197    if let Some(e) = e {
198        seq(e, out, depth)?;
199    }
200    out.push('}');
201    Some(())
202}

204/// The property elements and the properties in them that this converter reads, so the ones
205/// that decide what an equation says beyond its structure and text. [`body_marker`] hashes
206/// exactly these, and [`val`] and [`on`] read only these, so the two cannot drift: a property
207/// read here without a row in this table reads as absent, which the conversion tests show.
208const PROPERTIES: &[(&str, &[&str])] = &[
209    ("rPr", &["nor", "sty", "scr"]),
210    ("fPr", &["type"]),
211    ("radPr", &["degHide"]),
212    ("naryPr", &["chr", "limLoc", "subHide", "supHide"]),
213    ("dPr", &["begChr", "sepChr", "endChr"]),
214    ("accPr", &["chr"]),
215    ("barPr", &["pos"]),
216    ("groupChrPr", &["chr", "pos"]),
217];
218
219/// The properties of the property element `pr` that the converter reads.
220fn read_properties(pr: &str) -> &'static [&'static str] {
221    PROPERTIES.iter().find(|(name, _)| *name == pr).map_or(&[], |(_, names)| names)
222}

224/// The `val` of property `name` in the property element `props`, when the converter reads it.
225fn val<'a>(props: Option<&'a El>, name: &str) -> Option<&'a str> {
226    props.filter(|p| read_properties(&p.name).contains(&name)).and_then(|p| p.child(name)).and_then(|c| c.attr("val"))
227}

229/// Whether the switch `name` in the property element `props` is on, when the converter reads it.
230fn on(props: Option<&El>, name: &str) -> bool {
231    props
232        .filter(|p| read_properties(&p.name).contains(&name))
233        .and_then(|p| p.child(name))
234        .is_some_and(|c| c.attr("val").is_none_or(|v| v == "1" || v == "on" || v == "true"))
235}

328/// Appends `s` escaped for `\text{…}`; `None` once the LaTeX exceeds [`MAX_OUT_BYTES`], checked
329/// after every character so that a long run never expands in full.
330fn escape_text(s: &str, out: &mut String) -> Option<()> {
331    for c in s.chars() {
332        match c {
333            '{' | '}' | '%' | '#' | '&' | '$' | '_' => {
334                out.push('\\');
335                out.push(c);
336            }
337            '\\' => out.push_str("\\textbackslash "),
338            '^' | '~' => out.push(' '),
339            c => out.push(c),
340        }
341        if out.len() > MAX_OUT_BYTES {
342            return None;
343        }
344    }
345    Some(())
346}

348/// Known operator names written upright in a run (`sin`, `log`, `lim`).
349const FUNCTIONS: &[&str] = &[
350    "sin", "cos", "tan", "cot", "sec", "csc", "arcsin", "arccos", "arctan", "sinh", "cosh", "tanh", "log", "ln", "lg", "exp", "lim", "max", "min",
351    "sup", "inf", "det", "gcd", "deg", "dim", "ker", "arg", "Pr", "liminf", "limsup",
352];
353
354fn run(r: &El, out: &mut String) -> Option<()> {
355    let text = run_text(r);
356    if text.is_empty() {
357        return Some(());
358    }
359    let rpr = r.child("rPr");
360    if on(rpr, "nor") {
361        out.push_str("\\text{");
362        escape_text(&text, out)?;
363        out.push('}');
364        return Some(());
365    }
366    let plain = val(rpr, "sty") == Some("p");
367    if plain && FUNCTIONS.contains(&text.trim()) {
368        out.push('\\');
369        out.push_str(text.trim());
370        out.push(' ');
371        return Some(());
372    }
373    let wrap = match (val(rpr, "scr"), val(rpr, "sty")) {
374        (Some("double-struck"), _) => Some("\\mathbb{"),
375        (Some("script"), _) => Some("\\mathcal{"),
376        (Some("fraktur"), _) => Some("\\mathfrak{"),
377        (Some("sans-serif"), _) => Some("\\mathsf{"),
378        (Some("monospace"), _) => Some("\\mathtt{"),
379        (_, Some("b")) => Some("\\mathbf{"),
380        (_, Some("bi")) => Some("\\boldsymbol{"),
381        (_, Some("p")) if text.chars().any(|c| c.is_alphabetic()) => Some("\\mathrm{"),
382        _ => None,
383    };
384    if let Some(w) = wrap {
385        out.push_str(w);
386    }
387    for c in text.chars() {
388        match char_latex(c) {
389            Some(l) => out.push_str(l),
390            None => out.push(c),
391        }
392        // Checked per character, not from the run's byte length: a character may map to a
393        // longer macro or a shorter sign, so only the LaTeX produced counts.
394        if out.len() > MAX_OUT_BYTES {
395            return None;
396        }
397    }
398    if wrap.is_some() {
399        out.push('}');
400    }
401    Some(())
402}

404/// A delimiter character as a `\left`/`\right` argument.
405fn delim(c: Option<&str>, default: &str) -> String {
406    let c = c.unwrap_or(default);
407    match c {
408        "" => ".".into(),
409        "{" => "\\{".into(),
410        "}" => "\\}".into(),
411        "⟨" | "〈" => "\\langle".into(),
412        "⟩" | "〉" => "\\rangle".into(),
413        "⌊" => "\\lfloor".into(),
414        "⌋" => "\\rfloor".into(),
415        "⌈" => "\\lceil".into(),
416        "⌉" => "\\rceil".into(),
417        "‖" => "\\|".into(),
418        "∣" => "|".into(),
419        other => other.into(),
420    }
421}

460fn node(e: &El, out: &mut String, depth: usize) -> Option<()> {
461    if depth > MAX_DEPTH {
462        return None;
463    }
464    match e.name.as_str() {
465        // Properties and run-level DrawingML are not content.
466        n if n.ends_with("Pr") => {}
467        "r" => run(e, out)?,
468        "f" => fraction(e, out, depth)?,
469        "rad" => radical(e, out, depth)?,
470        "sSup" | "sSub" | "sSubSup" => scripts(e, out, depth)?,
471        "nary" => nary(e, out, depth)?,
472        "d" => delimiters(e, out, depth)?,
473        "m" => matrix(e, out, depth)?,
474        "eqArr" => equation_array(e, out, depth)?,
475        "acc" => {
476            out.push_str(accent(val(e.child("accPr"), "chr").unwrap_or("\u{0302}"))?);
477            group(e.child("e"), out, depth)?;
478        }
479        "bar" => {
480            out.push_str(if val(e.child("barPr"), "pos") == Some("top") { "\\overline" } else { "\\underline" });
481            group(e.child("e"), out, depth)?;
482        }
483        "groupChr" => group_char(e, out, depth)?,
484        "limLow" | "limUpp" => limit(e, out, depth)?,
485        "func" => {
486            seq(e.child("fName")?, out, depth)?;
487            out.push(' ');
488            group(e.child("e"), out, depth)?;
489        }
490        "borderBox" => {
491            out.push_str("\\boxed");
492            group(e.child("e"), out, depth)?;
493        }
494        "phant" => {
495            out.push_str("\\phantom");
496            group(e.child("e"), out, depth)?;
497        }
498        "box" => group(e.child("e"), out, depth)?,
499        _ => return None,
500    }
501    Some(())
502}

504/// `m:f`: a fraction, or a binomial-style stack without a bar.
505fn fraction(e: &El, out: &mut String, depth: usize) -> Option<()> {
506    if val(e.child("fPr"), "type") == Some("noBar") {
507        out.push_str("\\genfrac{}{}{0pt}{}");
508    } else {
509        out.push_str("\\frac");
510    }
511    group(e.child("num"), out, depth)?;
512    group(e.child("den"), out, depth)
513}

515/// `m:rad`: a square root, or a root with a degree.
516fn radical(e: &El, out: &mut String, depth: usize) -> Option<()> {
517    let hide = on(e.child("radPr"), "degHide");
518    match e.child("deg").filter(|d| !hide && !d.children.is_empty()) {
519        Some(d) => {
520            out.push_str("\\sqrt[");
521            seq(d, out, depth)?;
522            out.push(']');
523        }
524        None => out.push_str("\\sqrt"),
525    }
526    group(e.child("e"), out, depth)
527}

529/// `m:sSup`, `m:sSub`, `m:sSubSup`: a base with scripts.
530fn scripts(e: &El, out: &mut String, depth: usize) -> Option<()> {
531    group(e.child("e"), out, depth)?;
532    if let Some(sub) = e.child("sub") {
533        out.push('_');
534        group(Some(sub), out, depth)?;
535    }
536    if let Some(sup) = e.child("sup") {
537        out.push('^');
538        group(Some(sup), out, depth)?;
539    }
540    Some(())
541}

543/// `m:nary`: a large operator with its limits and operand.
544fn nary(e: &El, out: &mut String, depth: usize) -> Option<()> {
545    let p = e.child("naryPr");
546    let op = nary_op(val(p, "chr").unwrap_or("∫"))?;
547    out.push_str(op);
548    if val(p, "limLoc") == Some("undOvr") && op.contains("int") {
549        out.push_str("\\limits");
550    }
551    if !on(p, "subHide") {
552        out.push('_');
553        group(e.child("sub"), out, depth)?;
554    }
555    if !on(p, "supHide") {
556        out.push('^');
557        group(e.child("sup"), out, depth)?;
558    }
559    out.push(' ');
560    group(e.child("e"), out, depth)
561}

563/// `m:d`: delimiters around one or more parts, with a separator between them.
564fn delimiters(e: &El, out: &mut String, depth: usize) -> Option<()> {
565    let p = e.child("dPr");
566    out.push_str("\\left");
567    out.push_str(&delim(val(p, "begChr"), "("));
568    out.push(' ');
569    let sep = delim(val(p, "sepChr"), "|");
570    for (i, part) in e.kids("e").enumerate() {
571        if i > 0 {
572            out.push_str(" \\middle");
573            out.push_str(&sep);
574            out.push(' ');
575        }
576        seq(part, out, depth)?;
577    }
578    out.push_str(" \\right");
579    out.push_str(&delim(val(p, "endChr"), ")"));
580    Some(())
581}

583/// `m:m`: a matrix, row by row.
584fn matrix(e: &El, out: &mut String, depth: usize) -> Option<()> {
585    out.push_str("\\begin{matrix}");
586    for (i, row) in e.kids("mr").enumerate() {
587        if i > 0 {
588            out.push_str(" \\\\ ");
589        }
590        for (k, cell) in row.kids("e").enumerate() {
591            if k > 0 {
592                out.push_str(" & ");
593            }
594            seq(cell, out, depth)?;
595        }
596    }
597    out.push_str("\\end{matrix}");
598    Some(())
599}

601/// `m:eqArr`: equations stacked one per row.
602fn equation_array(e: &El, out: &mut String, depth: usize) -> Option<()> {
603    out.push_str("\\begin{gathered}");
604    for (i, row) in e.kids("e").enumerate() {
605        if i > 0 {
606            out.push_str(" \\\\ ");
607        }
608        seq(row, out, depth)?;
609    }
610    out.push_str("\\end{gathered}");
611    Some(())
612}

614/// Effective brace orientation, including the character default and its precedence over position.
615fn group_char_is_over(e: &El) -> bool {
616    let p = e.child("groupChrPr");
617    match val(p, "chr").unwrap_or("\u{23DF}") {
618        "\u{23DE}" => true,
619        "\u{23DF}" => false,
620        _ => val(p, "pos") == Some("top"),
621    }
622}
623
624/// `m:groupChr`: a brace over or under its content.
625fn group_char(e: &El, out: &mut String, depth: usize) -> Option<()> {
626    out.push_str(if group_char_is_over(e) { "\\overbrace" } else { "\\underbrace" });
627    group(e.child("e"), out, depth)
628}
629
630/// `m:limLow`, `m:limUpp`: a limit under or over its base.
631fn limit(e: &El, out: &mut String, depth: usize) -> Option<()> {
632    let low = e.name == "limLow";
633    let base = e.child("e");
634    // A function name with a limit (lim, max) takes it as a subscript with \limits.
635    let base_is_fn = base.is_some_and(|b| {
636        let t: String = b.kids("r").map(run_text).collect();
637        b.children.iter().all(|c| c.name == "r") && FUNCTIONS.contains(&t.trim())
638    });
639    // A brace alone under the limit is a labelled brace, as `crate::math::to_omml` writes
640    // `\underbrace{x}_{n}` and `\overbrace{x}^{n}`: the label is the brace's own script, below
641    // for an underbrace and above for an overbrace. An opposite-side limit keeps its placement.
642    let brace = base
643        .and_then(|b| {
644            let mut content = b.children.iter().filter(|c| !c.name.ends_with("Pr"));
645            content.next().filter(|c| c.name == "groupChr" && content.next().is_none())
646        })
647        .filter(|brace| low != group_char_is_over(brace));
648    if base_is_fn {
649        seq(base?, out, depth)?;
650        out.push_str(if low { "\\limits_" } else { "\\limits^" });
651        group(e.child("lim"), out, depth)?;
652    } else if let Some(brace) = brace {
653        group_char(brace, out, depth + 1)?;
654        out.push(if low { '_' } else { '^' });
655        group(e.child("lim"), out, depth)?;
656    } else {
657        out.push_str(if low { "\\underset" } else { "\\overset" });
658        group(e.child("lim"), out, depth)?;
659        group(base, out, depth)?;
660    }
661    Some(())
662}
