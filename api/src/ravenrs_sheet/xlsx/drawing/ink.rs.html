12const MAX_POINTS: usize = 20_000;
13const MAX_COORD: f64 = 1e7;

15fn bounded(v: f64) -> bool {
16    v.is_finite() && v.abs() <= MAX_COORD
17}

19fn only_children(e: &El, names: &[&str]) -> bool {
20    e.children.iter().all(|c| names.contains(&c.name.as_str()))
21}
22
23fn only_attrs(e: &El, names: &[&str]) -> bool {
24    e.attrs.iter().all(|(k, _)| names.contains(&k.as_str()))
25}

27fn one<'a>(e: &'a El, name: &str) -> Option<&'a El> {
28    let mut found = e.children.iter().filter(|c| c.name == name);
29    let first = found.next()?;
30    found.next().is_none().then_some(first)
31}

33fn validate_shape_obj(obj: &El) -> Option<&El> {
34    if obj.name != "sp"
35        || !only_children(obj, &["nvSpPr", "spPr"])
36        || !only_attrs(obj, &["macro", "textlink"])
37        || obj.attrs.iter().any(|(_, v)| !v.is_empty())
38        || obj.path(&["nvSpPr", "cNvSpPr"]).is_some_and(|e| e.flag("txBox", false))
39    {
40        return None;
41    }
42    let sp = one(obj, "spPr")?;
43    if !only_children(sp, &["xfrm", "custGeom", "noFill", "ln"]) || !sp.attrs.is_empty() {
44        return None;
45    }
46    let no_fill = one(sp, "noFill")?;
47    if !no_fill.attrs.is_empty() || !no_fill.children.is_empty() {
48        return None;
49    }
50    Some(sp)
51}

53fn validate_xfrm(sp: &El, anchor: &Anchor) -> Option<()> {
54    let xf = one(sp, "xfrm")?;
55    if !xf.attrs.is_empty() || !only_children(xf, &["off", "ext"]) {
56        return None;
57    }
58    let off = one(xf, "off")?;
59    if off.attr_i64("x")? != 0 || off.attr_i64("y")? != 0 {
60        return None;
61    }
62    let ext = one(xf, "ext")?;
63    let ew = ext.attr_i64("cx")?;
64    let eh = ext.attr_i64("cy")?;
65    if ew <= 0 || eh <= 0 || anchor.width <= 0.0 || anchor.height <= 0.0 || !bounded(anchor.width as f64) || !bounded(anchor.height as f64) {
66        return None;
67    }
68    let same_extent =
69        |emu: i64, extent: f32| (emu as f64 / EMU_PER_PX - extent as f64).abs() <= 0.001_f64.max(extent as f64 * f32::EPSILON as f64 * 2.0);
70    if !same_extent(ew, anchor.width) || !same_extent(eh, anchor.height) {
71        return None;
72    }
73    Some(())
74}

76fn validate_cust_geom(sp: &El) -> Option<(&El, i64, i64)> {
77    let geom = one(sp, "custGeom")?;
78    if !geom.attrs.is_empty() || !only_children(geom, &["avLst", "gdLst", "ahLst", "cxnLst", "rect", "pathLst"]) {
79        return None;
80    }
81    // Text bounds have no effect on this unfilled geometry, which has no caption.
82    if geom.children.iter().any(|c| !matches!(c.name.as_str(), "pathLst" | "rect") && (!c.children.is_empty() || !c.attrs.is_empty())) {
83        return None;
84    }
85    if geom.kids("rect").count() > 1 || geom.kids("rect").any(|r| !r.children.is_empty() || !only_attrs(r, &["l", "t", "r", "b"])) {
86        return None;
87    }
88    let paths = one(geom, "pathLst")?;
89    if paths.children.len() != 1 {
90        return None;
91    }
92    let path = one(paths, "path")?;
93    let w = path.attr_i64("w")?;
94    let h = path.attr_i64("h")?;
95    if w <= 0
96        || h <= 0
97        || path.attr("fill") != Some("none")
98        || !only_attrs(path, &["w", "h", "fill", "stroke", "extrusionOk"])
99        || path.attr("stroke").is_some_and(|_| path.attr_bool("stroke") != Some(true))
100        || path.attr("extrusionOk").is_some_and(|_| path.attr_bool("extrusionOk") != Some(false))
101        || !(2..=MAX_POINTS).contains(&path.children.len())
102    {
103        return None;
104    }
105    Some((path, w, h))
106}

108fn validate_line(sp: &El) -> Option<(i64, Color)> {
109    let ln = one(sp, "ln")?;
110    if !only_attrs(ln, &["w", "cap", "cmpd", "algn"])
111        || ln.attr("cap") != Some("flat")
112        || ln.attr("cmpd").is_some_and(|v| v != "sng")
113        || ln.attr("algn").is_some_and(|v| v != "ctr")
114        || !only_children(ln, &["solidFill", "miter"])
115    {
116        return None;
117    }
118    let width_emu = ln.attr_i64("w")?;
119    let unit = EMU_PER_PX as i64;
120    if width_emu % unit != 0 || !(1..=40).contains(&(width_emu / unit)) {
121        return None;
122    }
123    let width = width_emu / unit;
124    let join = one(ln, "miter")?;
125    if !join.attrs.is_empty() || !join.children.is_empty() {
126        return None;
127    }
128    let fill = one(ln, "solidFill")?;
129    if fill.children.len() != 1 {
130        return None;
131    }
132    let color = one(fill, "srgbClr")?;
133    if !only_attrs(color, &["val"]) || !only_children(color, &["alpha"]) {
134        return None;
135    }
136    let expected_alpha = if width > 8 { 45_000 } else { 100_000 };
137    if color.children.is_empty() {
138        if expected_alpha != 100_000 {
139            return None;
140        }
141    } else {
142        let alpha = one(color, "alpha")?;
143        if !only_attrs(alpha, &["val"]) || !alpha.children.is_empty() || alpha.attr_i64("val")? != expected_alpha {
144            return None;
145        }
146    }
147    let line = Color::from_hex(color.attr("val")?)?;
148    Some((width, line))
149}

151fn parse_points(path: &El, anchor: &Anchor, w: i64, h: i64) -> Option<String> {
152    let mut text = String::new();
153    for (i, command) in path.children.iter().enumerate() {
154        if command.name != if i == 0 { "moveTo" } else { "lnTo" } || !command.attrs.is_empty() || command.children.len() != 1 {
155            return None;
156        }
157        let pt = one(command, "pt")?;
158        if !only_attrs(pt, &["x", "y"]) || !pt.children.is_empty() {
159            return None;
160        }
161        let x = pt.attr_i64("x")? as f64 * anchor.width as f64 / w as f64;
162        let y = pt.attr_i64("y")? as f64 * anchor.height as f64 / h as f64;
163        if !bounded(x) || !bounded(y) {
164            return None;
165        }
166        if i != 0 {
167            text.push(';');
168        }
169        let _ = write!(text, "{},{}", x as f32, y as f32);
170    }
171    Some(text)
172}

174/// Do not partially import a path: losing a curve, subpath, transform or line effect changes
175/// the drawing. Accept only the explicit subset emitted below, including its line styling.
176pub(super) fn read(obj: &El, anchor: Anchor) -> Option<Shape> {
177    let sp = validate_shape_obj(obj)?;
178    validate_xfrm(sp, &anchor)?;
179    let (path, w, h) = validate_cust_geom(sp)?;
180    let (width, line) = validate_line(sp)?;
181    let text = parse_points(path, &anchor, w, h)?;
182    Some(Shape {
183        id: 0,
184        kind: ShapeKind::Ink,
185        anchor,
186        fill: Color::Rgb((width as u32) << 16),
187        line,
188        text,
189        format: Default::default(),
190        math: Default::default(),
191    })
192}

194/// Invalid/native out-of-range data uses the existing rectangle fallback. A successful
195/// result is complete before it is appended, so malformed points never yield half a stroke.
196pub(super) fn write(shape: &Shape, theme: &Theme, id: usize) -> Option<String> {
197    if shape.kind != ShapeKind::Ink || shape.format.no_line {
198        return None;
199    }
200    let width = match shape.fill {
201        Color::Rgb(rgb) => ((rgb >> 16) & 0xff).max(1),
202        _ => 2,
203    };
204    if !(1..=40).contains(&width) {
205        return None;
206    }
207    let a = &shape.anchor;
208    if !bounded(a.width as f64) || !bounded(a.height as f64) || a.width <= 0.0 || a.height <= 0.0 || !a.dx.is_finite() || !a.dy.is_finite() {
209        return None;
210    }
211    let w = (a.width as f64 * EMU_PER_PX).round().max(1.0) as i64;
212    let h = (a.height as f64 * EMU_PER_PX).round().max(1.0) as i64;
213    let mut points = Vec::new();
214    for point in shape.text.split(';') {
215        if points.len() == MAX_POINTS {
216            return None;
217        }
218        let (x, y) = point.split_once(',')?;
219        let (x, y) = (x.parse::<f64>().ok()?, y.parse::<f64>().ok()?);
220        if !bounded(x) || !bounded(y) {
221            return None;
222        }
223        points.push(((x * EMU_PER_PX).round() as i64, (y * EMU_PER_PX).round() as i64));
224    }
225    if points.len() < 2 {
226        return None;
227    }
228    let [r, g, b] = shape.line.resolve(theme)?;
229    let mut xml = format!(
230        "<xdr:sp macro=\"\" textlink=\"\"><xdr:nvSpPr><xdr:cNvPr id=\"{id}\" name=\"Ink {id}\"/><xdr:cNvSpPr/></xdr:nvSpPr><xdr:spPr>{}<a:custGeom><a:avLst/><a:gdLst/><a:ahLst/><a:cxnLst/><a:pathLst><a:path w=\"{w}\" h=\"{h}\" fill=\"none\">",
231        xfrm(a, "a:xfrm")
232    );
233    for (i, (x, y)) in points.iter().enumerate() {
234        let tag = if i == 0 { "moveTo" } else { "lnTo" };
235        let _ = write!(xml, "<a:{tag}><a:pt x=\"{x}\" y=\"{y}\"/></a:{tag}>");
236    }
237    let _ = write!(
238        xml,
239        "</a:path></a:pathLst></a:custGeom><a:noFill/><a:ln w=\"{}\" cap=\"flat\"><a:solidFill><a:srgbClr val=\"{r:02X}{g:02X}{b:02X}\">",
240        width as i64 * EMU_PER_PX as i64
241    );
242    if width > 8 {
243        xml.push_str("<a:alpha val=\"45000\"/>");
244    }
245    xml.push_str("</a:srgbClr></a:solidFill><a:miter/></a:ln></xdr:spPr></xdr:sp>");
246    Some(xml)
247}
