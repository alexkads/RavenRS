31/// A part's relationships.
32#[derive(Default)]
33pub struct Rels {
34    items: Vec<(String, String, String, bool)>,
35}
36
37impl Rels {
38    /// Adds a relationship of type `kind` (last segment, e.g. `worksheet`); returns its id.
39    pub fn add(&mut self, kind: &str, target: &str) -> String {
40        let id = format!("rId{}", self.items.len() + 1);
41        let ty = match kind {
42            "core-properties" => "http://schemas.openxmlformats.org/package/2006/relationships/metadata/core-properties".to_string(),
43            // The VBA project relationship lives in the Microsoft-documented namespace, not the
44            // OpenXML one; Excel rejects the file if this type is wrong.
45            "vbaProject" => "http://schemas.microsoft.com/office/2006/relationships/vbaProject".to_string(),
46            k => format!("{NS_REL}/{k}"),
47        };
48        self.items.push((id.clone(), ty, target.to_string(), false));
49        id
50    }
51    pub fn add_external(&mut self, kind: &str, target: &str) -> String {
52        let id = format!("rId{}", self.items.len() + 1);
53        self.items.push((id.clone(), format!("{NS_REL}/{kind}"), target.to_string(), true));
54        id
55    }
56    /// Adds a Microsoft extension relationship with its complete namespace URI.
57    pub fn add_uri(&mut self, ty: &str, target: &str) -> String {
58        let id = format!("rId{}", self.items.len() + 1);
59        self.items.push((id.clone(), ty.to_string(), target.to_string(), false));
60        id
61    }
62    /// Adds a relationship with its complete type URI, internal or external (kept content).
63    pub fn add_raw(&mut self, ty: &str, target: &str, external: bool) -> String {
64        let id = format!("rId{}", self.items.len() + 1);
65        self.items.push((id.clone(), crate::xlsx::xml::esc_attr(ty), target.to_string(), external));
66        id
67    }
68    pub fn is_empty(&self) -> bool {
69        self.items.is_empty()
70    }
71    pub fn xml(&self) -> String {
72        let mut s = format!("{XML_DECL}<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">");
73        for (id, ty, target, ext) in &self.items {
74            let _ = write!(
75                s,
76                "<Relationship Id=\"{id}\" Type=\"{ty}\" Target=\"{}\"{}/>",
77                esc_attr(target),
78                if *ext { " TargetMode=\"External\"" } else { "" }
79            );
80        }
81        s.push_str("</Relationships>");
82        s
83    }
84}

86/// Shared string table being built.
87#[derive(Default)]
88pub struct Sst {
89    list: Vec<String>,
90    map: HashMap<String, u32>,
91    pub count: u64,
92}
93
94impl Sst {
95    pub fn index(&mut self, s: &str) -> u32 {
96        self.count += 1;
97        if let Some(&i) = self.map.get(s) {
98            return i;
99        }
100        let i = self.list.len() as u32;
101        self.list.push(s.to_string());
102        self.map.insert(s.to_string(), i);
103        i
104    }
105}

139impl Out {
140    pub fn part(&mut self, name: &str, content_type: Option<&str>, data: Vec<u8>) {
141        if let Some(ct) = content_type {
142            self.overrides.push((format!("/{name}"), ct.to_string()));
143        }
144        self.parts.push((name.to_string(), data));
145    }
146    /// The `xl/media` file name (`imageN.ext`) holding `data`. A picture used on several sheets,
147    /// or twice on one, is written once and every drawing points to that part, as in Excel.
148    pub fn image_part(&mut self, ext: &str, data: &[u8]) -> String {
149        let existing = self.parts.iter().find(|(name, bytes)| {
150            name.strip_prefix("xl/media/image").is_some_and(|rest| rest.ends_with(&format!(".{ext}"))) && bytes.as_slice() == data
151        });
152        if let Some((name, _)) = existing {
153            return name.trim_start_matches("xl/media/").to_string();
154        }
155        self.images += 1;
156        let file = format!("image{}.{ext}", self.images);
157        self.part(&format!("xl/media/{file}"), None, data.to_vec());
158        file
159    }
160    pub fn default_type(&mut self, ext: &str, ct: &str) {
161        self.defaults.entry(ext.to_string()).or_insert_with(|| ct.to_string());
162    }
163}

171/// `$A$1:$D$20` (or `$1:$3` / `$A:$C`).
172pub fn abs_range(r: &RangeRef) -> String {
173    let c = |col: u32| crate::base::col_to_letters(col);
174    if r.is_full_rows() && !r.is_full_cols() {
175        return format!("${}:${}", r.start.row + 1, r.end.row + 1);
176    }
177    if r.is_full_cols() && !r.is_full_rows() {
178        return format!("${}:${}", c(r.start.col), c(r.end.col));
179    }
180    if r.is_single() {
181        return format!("${}${}", c(r.start.col), r.start.row + 1);
182    }
183    format!("${}${}:${}${}", c(r.start.col), r.start.row + 1, c(r.end.col), r.end.row + 1)
184}

186/// Refuses sheet names XLSX cannot hold. The reader repairs such names (cuts, strips, renames
187/// duplicates) but formulas keep the old spelling, so the saved file would turn into `#REF!`.
188fn check_sheet_names(wb: &Workbook) -> Result<(), IoError> {
189    let mut seen = std::collections::HashSet::new();
190    for sheet in &wb.sheets {
191        let name = &sheet.name;
192        let bad = if name.is_empty() {
193            Some("is blank")
194        } else if name.chars().count() > 31 {
195            Some("is longer than 31 characters")
196        } else if name.chars().any(|c| matches!(c, ':' | '\\' | '/' | '?' | '*' | '[' | ']')) {
197            Some("contains one of : \\ / ? * [ ]")
198        } else if !seen.insert(name.to_ascii_lowercase()) {
199            Some("is used by another sheet")
200        } else {
201            None
202        };
203        if let Some(why) = bad {
204            return Err(IoError::Format(format!("sheet name \"{name}\" {why}; rename the sheet before saving as XLSX")));
205        }
206    }
207    Ok(())
208}

217/// The `<workbookView>` attributes beyond `activeTab`: the window's display settings
218/// that differ from the format's defaults. An ordinary workbook gets none, so its file
219/// is unchanged; a workbook Excel saved with the sheet tabs or scroll bars hidden
220/// keeps them that way (#3057).
221fn workbook_view_attrs(wb: &Workbook) -> String {
222    let mut attrs = String::new();
223    if !wb.view.show_sheet_tabs {
224        attrs.push_str(" showSheetTabs=\"0\"");
225    }
226    if !wb.view.show_horizontal_scroll {
227        attrs.push_str(" showHorizontalScroll=\"0\"");
228    }
229    if !wb.view.show_vertical_scroll {
230        attrs.push_str(" showVerticalScroll=\"0\"");
231    }
232    let tab_ratio = wb.view.tab_ratio.min(WorkbookView::MAX_TAB_RATIO);
233    if tab_ratio != 600 {
234        let _ = write!(attrs, " tabRatio=\"{tab_ratio}\"");
235    }
236    attrs
237}

239/// Writes the workbook for the file `book_name` (a bare file name). A PivotChart is bound to its
240/// PivotTable in Excel only when its link starts with the workbook in brackets.
241pub fn write_xlsx_named(wb: &Workbook, kind: XlsxKind, book_name: Option<&str>) -> Result<Vec<u8>, IoError> {
242    if wb.sheets.is_empty() {
243        return Err(IoError::Format("a workbook needs at least one sheet".into()));
244    }
245    if wb.sheets.iter().any(|sheet| sheet.cell_pictures.keys().any(|c| sheet.cell(*c).is_some_and(|cell| cell.formula.is_some()))) {
246        return Err(IoError::Format("a static cell picture cannot also contain a formula".into()));
247    }
248    check_sheet_names(wb)?;
249    let mut out = Out {
250        parts: vec![],
251        overrides: vec![],
252        defaults: BTreeMap::new(),
253        dxfs: vec![],
254        sst: Sst::default(),
255        images: 0,
256        charts: 0,
257        book_name: book_name.filter(|n| !n.is_empty() && !n.contains(['[', ']'])).unwrap_or(DEFAULT_BOOK_NAME).to_string(),
258        chart_names: vec![],
259        drawings: 0,
260        tables: 0,
261        comments: 0,
262        vmls: 0,
263        vml_block: 0,
264        ctrl_props: 0,
265        persons: vec![],
266        table_id: 0,
267        dynamic_arrays: false,
268        needs_calc: false,
269        cell_pictures: Default::default(),
270        kept: Default::default(),
271        python_scripts: vec![],
272    };
273    out.kept.reserve_vml_blocks(wb);
274    out.default_type("rels", "application/vnd.openxmlformats-package.relationships+xml");
275    out.default_type("xml", "application/xml");
276
277    // Which sheet is active (must be visible), and at least one visible sheet.
278    let active = if wb.sheets.get(wb.active_sheet()).is_some_and(|s| s.visibility == Visibility::Visible) {
279        wb.active_sheet()
280    } else {
281        wb.sheets.iter().position(|s| s.visibility == Visibility::Visible).unwrap_or(0)
282    };
283
284    let mut wb_rels = Rels::default();
285    let (sheets_xml, kept, pivots) = write_sheets(wb, &mut out, active, &mut wb_rels);
286
287    let names = write_defined_names(wb, &kept, &out.chart_names);
288
289    let mut w = format!("{XML_DECL}<workbook xmlns=\"{NS_MAIN}\" xmlns:r=\"{NS_REL}\">");
290    // A workbook read from a file keeps its <fileVersion> as it was: Excel trusts the saved values
291    // only when appName is "xl" (#799). One without names GridCraft (#1699), and Excel recalculates it.
292    let file_version = crate::xlsx::passthrough::workbook_elements(wb, "fileVersion");
293    w.push_str(if file_version.is_empty() { "<fileVersion appName=\"gridcraft\"/>" } else { &file_version });
294    w.push_str("<workbookPr");
295    match wb.show_objects {
296        ShowObjects::All => {}
297        ShowObjects::Placeholders => w.push_str(" showObjects=\"placeholders\""),
298        ShowObjects::None => w.push_str(" showObjects=\"none\""),
299    }
300    if wb.date_system == crate::base::DateSystem::D1904 {
301        w.push_str(" date1904=\"1\"");
302    }
303    if wb.remove_personal_info {
304        w.push_str(" filterPrivacy=\"1\"");
305    }
306    w.push_str("/>");
307    if wb.protected_structure {
308        let pw = wb.structure_password.as_ref().map(|h| crate::xlsx::sheet_write::password_attrs(h, "workbook")).unwrap_or_default();
309        let _ = write!(w, "<workbookProtection{pw} lockStructure=\"1\"/>");
310    }
311    let _ = write!(w, "<bookViews><workbookView activeTab=\"{}\"", kept.file_index(active));
312    w.push_str(&workbook_view_attrs(wb));
313    let _ = write!(w, "/></bookViews><sheets>{sheets_xml}</sheets>");
314    w.push_str(&kept.elements);
315    if !names.is_empty() {
316        let _ = write!(w, "<definedNames>{names}</definedNames>");
317    }
318    w.push_str("<calcPr calcId=\"191029\"");
319    match wb.calc.mode {
320        CalcMode::Automatic => {}
321        CalcMode::Manual => w.push_str(" calcMode=\"manual\""),
322        CalcMode::AutomaticExceptTables => w.push_str(" calcMode=\"autoNoTable\""),
323    }
324    // CT_CalcPr defaults: iterate off, 100 iterations, 0.001 maximum change. The limits are kept
325    // while iteration is off, as Options keeps them.
326    let calc = &wb.calc;
327    if calc.iterative {
328        w.push_str(" iterate=\"1\"");
329    }
330    if calc.iterative || calc.max_iterations != CalcSettings::DEFAULT_ITERATIONS {
331        let _ = write!(w, " iterateCount=\"{}\"", calc.max_iterations);
332    }
333    if calc.iterative || calc.max_change != CalcSettings::DEFAULT_CHANGE {
334        let _ = write!(w, " iterateDelta=\"{}\"", crate::xlsx::xml::num(calc.max_change));
335    }
336    if wb.calc.precision_as_displayed {
337        w.push_str(" fullPrecision=\"0\"");
338    }
339    if wb.calc.r1c1 {
340        w.push_str(" refMode=\"R1C1\"");
341    }
342    if out.needs_calc {
343        w.push_str(" fullCalcOnLoad=\"1\"");
344    }
345    w.push_str("/>");
346    w.push_str(&pivots.workbook_xml(&mut wb_rels));
347    let exts = crate::xlsx::passthrough::workbook_elements(wb, "ext");
348    if !exts.is_empty() {
349        let _ = write!(w, "<extLst>{exts}</extLst>");
350    }
351    w.push_str("</workbook>");
352
353    // Workbook-level parts.
354    let (styles, fmt_ids) = crate::xlsx::styles::write_styles(wb, &out.dxfs, &pivots.number_formats(wb));
355    pivots.finish(wb, &mut out, &|code: &str| crate::xlsx::tables::builtin_id(code).or_else(|| fmt_ids.get(code).copied()));
356    out.part("xl/styles.xml", Some("application/vnd.openxmlformats-officedocument.spreadsheetml.styles+xml"), styles.into_bytes());
357    wb_rels.add("styles", "styles.xml");
358    if !out.python_scripts.is_empty() {
359        use crate::xlsx::python_part as py;
360        let part = py::write(&out.python_scripts);
361        out.part(py::PART, Some(py::CONTENT_TYPE), part.into_bytes());
362        wb_rels.add_uri(py::REL, "python.xml");
363    }
364    if wb.styles.iter().any(|(_, st)| st.num_fmt.as_str() == "checkbox") {
365        use crate::xlsx::feature_bags as fpb;
366        out.part(fpb::PART, Some(fpb::CONTENT_TYPE), fpb::checkbox_part().into_bytes());
367        wb_rels.add_uri(fpb::REL, "featurePropertyBag/featurePropertyBag.xml");
368    }
369    out.part(
370        "xl/theme/theme1.xml",
371        Some("application/vnd.openxmlformats-officedocument.theme+xml"),
372        crate::xlsx::theme::write_theme(&wb.theme).into_bytes(),
373    );
374    wb_rels.add("theme", "theme/theme1.xml");
375    // Phonetic (furigana) data is kept per string text: the table below holds each text once, so
376    // the runs of the entry with that text are written next to it (`Workbook::phonetics`).
377    let phonetics: HashMap<&str, &PhoneticString> = wb.phonetics.iter().map(|p| (p.text.as_str(), p)).collect();
378    let mut sst = format!("{XML_DECL}<sst xmlns=\"{NS_MAIN}\" count=\"{}\" uniqueCount=\"{}\">", out.sst.count, out.sst.list.len());
379    for t in &out.sst.list {
380        let _ = write!(sst, "<si>{}</si>", crate::xlsx::xml::si_body(t, phonetics.get(t.as_str()).copied()));
381    }
382    sst.push_str("</sst>");
383    out.part("xl/sharedStrings.xml", Some("application/vnd.openxmlformats-officedocument.spreadsheetml.sharedStrings+xml"), sst.into_bytes());
384    wb_rels.add("sharedStrings", "sharedStrings.xml");
385    if !out.cell_pictures.is_empty() {
386        crate::xlsx::richdata::write(&mut out, &mut wb_rels)?;
387    } else if out.dynamic_arrays {
388        out.part(
389            "xl/metadata.xml",
390            Some("application/vnd.openxmlformats-officedocument.spreadsheetml.sheetMetadata+xml"),
391            METADATA.as_bytes().to_vec(),
392        );
393        wb_rels.add("sheetMetadata", "metadata.xml");
394    }
395    if !out.persons.is_empty() {
396        let mut x = format!("{XML_DECL}<personList xmlns=\"{NS_THREADED}\" xmlns:x=\"{NS_MAIN}\">");
397        for (i, name) in out.persons.iter().enumerate() {
398            let n = esc_attr(name);
399            let _ = write!(x, "<person displayName=\"{n}\" id=\"{}\" userId=\"{n}\" providerId=\"None\"/>", person_guid(i));
400        }
401        x.push_str("</personList>");
402        out.part("xl/persons/person.xml", Some(CT_PERSONS), x.into_bytes());
403        wb_rels.add_uri(REL_PERSON, "persons/person.xml");
404    }
405    out.part("xl/workbook.xml", Some(kind.main_content_type()), w.into_bytes());
406    // A macro-free package can't hold a VBA project (Excel refuses such a file).
407    if let Some(data) = wb.vba_project.as_ref().filter(|_| kind.is_macro_enabled()) {
408        wb_rels.add("vbaProject", "vbaProject.bin");
409        out.part("xl/vbaProject.bin", Some(CT_VBA), data.to_vec());
410    }
411    out.part("xl/_rels/workbook.xml.rels", None, wb_rels.xml().into_bytes());
412
413    // Document properties.
414    let p = &wb.props;
415    // "Remove personal information on save": the author and company are not saved (#3089).
416    let private = |v: &String| if wb.remove_personal_info { String::new() } else { v.clone() };
417    let (author, company) = (private(&p.author), private(&p.company));
418    let mut core = format!(
419        "{XML_DECL}<cp:coreProperties xmlns:cp=\"http://schemas.openxmlformats.org/package/2006/metadata/core-properties\" xmlns:dc=\"http://purl.org/dc/elements/1.1/\" xmlns:dcterms=\"http://purl.org/dc/terms/\" xmlns:dcmitype=\"http://purl.org/dc/dcmitype/\" xmlns:xsi=\"http://www.w3.org/2001/XMLSchema-instance\">"
420    );
421    for (tag, v) in [
422        ("dc:title", &p.title),
423        ("dc:subject", &p.subject),
424        ("dc:creator", &author),
425        ("cp:keywords", &p.keywords),
426        ("dc:description", &p.description),
427    ] {
428        if !v.is_empty() {
429            let _ = write!(core, "<{tag}>{}</{tag}>", esc(v));
430        }
431    }
432    core.push_str("</cp:coreProperties>");
433    out.part("docProps/core.xml", Some("application/vnd.openxmlformats-package.core-properties+xml"), core.into_bytes());
434    let mut app = String::from(XML_DECL);
435    app.push_str("<Properties xmlns=\"http://schemas.openxmlformats.org/officeDocument/2006/extended-properties\" xmlns:vt=\"http://schemas.openxmlformats.org/officeDocument/2006/docPropsVTypes\"><Application>GridCraft</Application>");
436    if !company.is_empty() {
437        let _ = write!(app, "<Company>{}</Company>", esc(&company));
438    }
439    app.push_str("</Properties>");
440    out.part("docProps/app.xml", Some("application/vnd.openxmlformats-officedocument.extended-properties+xml"), app.into_bytes());
441    let mut root = Rels::default();
442    root.add("officeDocument", "xl/workbook.xml");
443    root.add("core-properties", "docProps/core.xml");
444    root.add("extended-properties", "docProps/app.xml");
445    crate::xlsx::passthrough::write_package_rels(&mut root, &mut out);
446    out.part("_rels/.rels", None, root.xml().into_bytes());
447
448    // Content types first, then the parts in creation order.
449    let mut ct = format!("{XML_DECL}<Types xmlns=\"http://schemas.openxmlformats.org/package/2006/content-types\">");
450    for (ext, t) in &out.defaults {
451        // Kept parts bring their content types (and extensions) from the input file: escape them.
452        let _ = write!(ct, "<Default Extension=\"{}\" ContentType=\"{}\"/>", esc_attr(ext), esc_attr(t));
453    }
454    for (part, t) in &out.overrides {
455        let _ = write!(ct, "<Override PartName=\"{}\" ContentType=\"{}\"/>", esc_attr(part), esc_attr(t));
456    }
457    ct.push_str("</Types>");
458    let mut zip = ZipOut::new();
459    zip.add("[Content_Types].xml", ct.as_bytes())?;
460    // Excel's order: rels, workbook rels, workbook, the rest, document properties last. libmagic (PHP
461    // `finfo`, `file`) recognises an OOXML package only when an `xl/` entry name appears within the
462    // first few KB, so no user-sized part (document properties, a long description) may precede it.
463    let rank = |n: &str| match n {
464        "_rels/.rels" => 0,
465        "xl/_rels/workbook.xml.rels" => 1,
466        "xl/workbook.xml" => 2,
467        n if n.starts_with("docProps/") => 4,
468        _ => 3,
469    };
470    let mut parts = std::mem::take(&mut out.parts);
471    parts.sort_by_key(|(n, _)| rank(n));
472    for (name, data) in &parts {
473        zip.add(name, data)?;
474    }
475    zip.finish()
476}

478fn write_sheets(
479    wb: &Workbook,
480    out: &mut Out,
481    active: usize,
482    wb_rels: &mut Rels,
483) -> (String, crate::xlsx::passthrough::WorkbookKept, crate::xlsx::pivot::PivotWriter) {
484    // Kept content first: the chart sheets among the worksheets need their relationship ids.
485    let kept = crate::xlsx::passthrough::write_workbook(wb, wb_rels, out);
486    let mut pivots = crate::xlsx::pivot::PivotWriter::default();
487    let mut sheets_xml = String::new();
488    let any_visible = wb.sheets.iter().any(|s| s.visibility == Visibility::Visible);
489    let mut ids = crate::xlsx::passthrough::SheetIds::new(wb, &kept);
490    for (i, sheet) in wb.sheets.iter().enumerate() {
491        sheets_xml.push_str(&kept.sheet_entries(i, false, &mut ids));
492        let (xml, mut rels) = crate::xlsx::sheet_write::write_sheet(wb, i, i == active, out);
493        pivots.add_sheet(wb, i, out, &mut rels);
494        let name = format!("xl/worksheets/sheet{}.xml", i + 1);
495        out.part(&name, Some(CT_WORKSHEET), xml.into_bytes());
496        if !rels.is_empty() {
497            out.part(&format!("xl/worksheets/_rels/sheet{}.xml.rels", i + 1), None, rels.xml().into_bytes());
498        }
499        let rid = wb_rels.add("worksheet", &format!("worksheets/sheet{}.xml", i + 1));
500        let state = match sheet.visibility {
501            Visibility::Visible => "",
502            _ if !any_visible && i == active => "",
503            Visibility::Hidden => " state=\"hidden\"",
504            Visibility::VeryHidden => " state=\"veryHidden\"",
505        };
506        let name: String = sheet.name.chars().take(31).collect();
507        let sheet_id = ids.take(sheet.file_id);
508        let _ = write!(sheets_xml, "<sheet name=\"{}\" sheetId=\"{sheet_id}\"{state} r:id=\"{rid}\"/>", esc_attr(&name));
509    }
510    sheets_xml.push_str(&kept.sheet_entries(wb.sheets.len(), true, &mut ids));
511
512    (sheets_xml, kept, pivots)
513}

515fn write_defined_names(wb: &Workbook, kept: &crate::xlsx::passthrough::WorkbookKept, chart_names: &[String]) -> String {
516    // Defined names, then print areas and titles.
517    let mut names = String::new();
518    let mut seen = std::collections::HashSet::new();
519    for n in &wb.names {
520        if n.name.is_empty() || !seen.insert((n.name.to_ascii_lowercase(), n.scope)) {
521            continue;
522        }
523        if let Some(sc) = n.scope
524            && sc >= wb.sheets.len()
525        {
526            continue;
527        }
528        let _ = write!(names, "<definedName name=\"{}\"", esc_attr(&n.name));
529        if let Some(sc) = n.scope {
530            let _ = write!(names, " localSheetId=\"{}\"", kept.file_index(sc));
531        }
532        if n.hidden {
533            names.push_str(" hidden=\"1\"");
534        }
535        if !n.comment.is_empty() {
536            let _ = write!(names, " comment=\"{}\"", esc_attr(&n.comment));
537        }
538        let _ = write!(names, ">{}</definedName>", esc(&crate::xlsx::fmla::text_to_file(&n.formula)));
539    }
540    for (i, f) in chart_names.iter().enumerate() {
541        let _ = write!(
542            names,
543            "<definedName name=\"{}{i}\" hidden=\"1\">{}</definedName>",
544            crate::xlsx::chartex::NAME_PREFIX,
545            esc(&crate::xlsx::fmla::text_to_file(f))
546        );
547    }
548    for (i, sheet) in wb.sheets.iter().enumerate() {
549        let q = quote_sheet(&sheet.name);
550        if let Some(pa) = &sheet.print.print_area {
551            let _ = write!(
552                names,
553                "<definedName name=\"_xlnm.Print_Area\" localSheetId=\"{}\">{}!{}</definedName>",
554                kept.file_index(i),
555                esc(&q),
556                abs_range(pa)
557            );
558        }
559        let mut titles = vec![];
560        if let Some((c0, c1)) = sheet.print.title_cols {
561            titles.push(format!("{q}!{}", abs_range(&RangeRef::cols(c0.min(c1), c0.max(c1)))));
562        }
563        if let Some((r0, r1)) = sheet.print.title_rows {
564            titles.push(format!("{q}!{}", abs_range(&RangeRef::rows(r0.min(r1), r0.max(r1)))));
565        }
566        if !titles.is_empty() {
567            let _ = write!(
568                names,
569                "<definedName name=\"_xlnm.Print_Titles\" localSheetId=\"{}\">{}</definedName>",
570                kept.file_index(i),
571                esc(&titles.join(","))
572            );
573        }
574    }
575
576    names
577}
