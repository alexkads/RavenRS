19/// [`crate::craft::csv::parse_records`] with a record kept to the grid's [`MAX_COLS`] fields.
20fn parse_records(text: &str, delim: char, quote: char, on_record: impl FnMut(Vec<String>) -> bool) {
21    crate::craft::csv::parse_records(text, delim, quote, MAX_COLS as usize, on_record);
22}

24/// How many rows and bytes of a text file are read to decide each column's date order.
25const DATE_SAMPLE_ROWS: usize = 200;
26const DATE_SAMPLE_BYTES: usize = 1 << 20;
27
28/// What a column's numeric dates say about the order of day and month.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30enum DateRule {
31    /// No `d/m/y` or `d-m-y` value in the sample: read as typed input does.
32    NoDates,
33    /// A value reads only month first (`12/31/2026`), none reads only day first.
34    MonthFirst,
35    /// A value reads only day first (`31/05/2026`), none reads only month first.
36    DayFirst,
37    /// Every date is ambiguous (`03/04/2026`): keep the input parser's month-first reading.
38    Ambiguous,
39    /// Values read only day first and values read only month first: the dates stay text.
40    Conflict,
41}
42
43impl DateRule {
44    fn of(day_first: usize, month_first: usize, ambiguous: usize) -> Self {
45        match (day_first > 0, month_first > 0, ambiguous > 0) {
46            (true, true, _) => Self::Conflict,
47            (true, false, _) => Self::DayFirst,
48            (false, true, _) => Self::MonthFirst,
49            (false, false, true) => Self::Ambiguous,
50            (false, false, false) => Self::NoDates,
51        }
52    }
53}

55/// The evidence a column's sample gives for its date order.
56#[derive(Clone, Default)]
57struct DateEvidence {
58    day_first: usize,
59    month_first: usize,
60    ambiguous: usize,
61    day_first_example: Option<String>,
62    month_first_example: Option<String>,
63    ambiguous_example: Option<String>,
64}

66/// A numeric `d/m/y` or `d-m-y` field, with the serial it has read each way and its time of day.
67struct NumericDate {
68    day_first: Option<f64>,
69    month_first: Option<f64>,
70    time: Option<f64>,
71}
72
73impl NumericDate {
74    /// The field under the column's rule: a date when that reading is valid, else text.
75    fn read(&self, field: &str, rule: DateRule) -> Parsed {
76        let serial = match rule {
77            DateRule::DayFirst => self.day_first,
78            DateRule::MonthFirst | DateRule::Ambiguous => self.month_first,
79            DateRule::Conflict | DateRule::NoDates => None,
80        };
81        match serial {
82            Some(serial) => Parsed {
83                value: Value::number(serial + self.time.unwrap_or(0.0)),
84                format: Some(if self.time.is_some() { "m/d/yyyy h:mm" } else { "m/d/yyyy" }),
85            },
86            None => Parsed { value: Value::text(field), format: None },
87        }
88    }
89}

91/// A numeric date written with `/` or `-`, with the serial it has read day first and month first.
92/// `None` when the field isn't such a date (including ISO `yyyy-mm-dd`, month names and dots,
93/// which read one way and keep the input parser).
94fn numeric_date(field: &str, sys: DateSystem) -> Option<NumericDate> {
95    let text = field.trim();
96    let (date, time) = split_time(text, sys);
97    // Typed input reads a trailing comma after a date (`3/4/2026,`); do the same here.
98    let date = date.trim_end_matches(',');
99    let mut parts = date.split(['/', '-']);
100    let (a, b, c) = (parts.next()?, parts.next()?, parts.next()?);
101    if parts.next().is_some() {
102        return None;
103    }
104    let (a, b, c) = (a.trim(), b.trim(), c.trim());
105    // ISO dates are unambiguous; the input parser reads them. A three-digit year is not a
106    // date to typed input either.
107    if a.len() == 4 || c.len() == 3 {
108        return None;
109    }
110    let a = parse_digits(a)?;
111    let b = parse_digits(b)?;
112    let year = expand_year(c, parse_digits(c)?);
113    let day_first = valid_date(sys, year, b, a);
114    let month_first = valid_date(sys, year, a, b);
115    if day_first.is_none() && month_first.is_none() {
116        return None;
117    }
118    Some(NumericDate { day_first, month_first, time })
119}

121/// Splits a field into its date and a time of day, when the tail reads as a time. Both the first
122/// and last space are tried, so `03/04/2026 10:30 AM` and `03 / 04 / 2026 10:30` both split.
123fn split_time(text: &str, sys: DateSystem) -> (&str, Option<f64>) {
124    for at in [text.find(' '), text.rfind(' ')].into_iter().flatten() {
125        let (date, time) = text.split_at(at);
126        let time = time.trim();
127        // Times are short; a long tail is not one, and parsing it would be wasted work.
128        if time.len() > 40 {
129            continue;
130        }
131        if let Some(fraction) = time_of_day(time, sys) {
132            return (date.trim(), Some(fraction));
133        }
134    }
135    (text, None)
136}

138fn time_of_day(text: &str, sys: DateSystem) -> Option<f64> {
139    match parse_input(text, sys) {
140        Parsed { value: Value::Number(fraction), format: Some(code) } if code.contains('h') => Some(fraction),
141        _ => None,
142    }
143}

145fn parse_digits(s: &str) -> Option<i64> {
146    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
147        return None;
148    }
149    s.parse().ok()
150}

152/// Expands a one- or two-digit year the way typed input does (`26` is 2026, `30` is 1930).
153fn expand_year(text: &str, year: i64) -> i64 {
154    if text.len() <= 2 { if year < 30 { 2000 + year } else { 1900 + year } } else { year }
155}

157/// The serial for a date the input parser accepts, or `None`.
158fn valid_date(sys: DateSystem, year: i64, month: i64, day: i64) -> Option<f64> {
159    if !(1..=12).contains(&month) || day < 1 || year < 1900 {
160        return None;
161    }
162    let year = i32::try_from(year).ok()?;
163    let month = u32::try_from(month).ok()?;
164    if day > i64::from(days_in_month_in(sys, year, month)) {
165        return None;
166    }
167    serial_from_ymd(sys, i64::from(year), i64::from(month), day)
168}

170/// The first records of the file, bounded by rows and bytes, for date-order inference.
171fn sample_records(text: &str, delim: char, quote: char) -> Vec<Vec<String>> {
172    let mut cut = text.len().min(DATE_SAMPLE_BYTES);
173    while !text.is_char_boundary(cut) {
174        cut -= 1;
175    }
176    let truncated = cut < text.len();
177    let sample = text.get(..cut).unwrap_or_default();
178    let mut rows = Vec::new();
179    parse_records(sample, delim, quote, |record| {
180        rows.push(record);
181        rows.len() < DATE_SAMPLE_ROWS
182    });
183    // A cut through the last record leaves a partial field; it says nothing reliable.
184    if truncated {
185        rows.pop();
186    }
187    rows
188}

190/// The date rule for each column, from the sample.
191fn infer_date_rules(rows: &[Vec<String>], sys: DateSystem) -> Vec<(DateRule, DateEvidence)> {
192    let width = rows.iter().map(Vec::len).max().unwrap_or(0).min(MAX_COLS as usize);
193    let mut columns = vec![DateEvidence::default(); width];
194    for row in rows {
195        for (col, field) in row.iter().enumerate() {
196            let Some(column) = columns.get_mut(col) else { continue };
197            let Some(date) = numeric_date(field, sys) else { continue };
198            let example = || field.trim().chars().take(40).collect::<String>();
199            match (date.day_first.is_some(), date.month_first.is_some()) {
200                (true, true) => {
201                    column.ambiguous += 1;
202                    column.ambiguous_example.get_or_insert_with(example);
203                }
204                (true, false) => {
205                    column.day_first += 1;
206                    column.day_first_example.get_or_insert_with(example);
207                }
208                (false, true) => {
209                    column.month_first += 1;
210                    column.month_first_example.get_or_insert_with(example);
211                }
212                (false, false) => {}
213            }
214        }
215    }
216    columns.into_iter().map(|c| (DateRule::of(c.day_first, c.month_first, c.ambiguous), c)).collect()
217}

219/// Warns when the sample could not choose a date order: an all-ambiguous column keeps the
220/// month-first reading, and a column with both kinds keeps its dates as text.
221fn date_warnings(columns: &[(DateRule, DateEvidence)], report: &mut ReadReport) {
222    for (col, (rule, evidence)) in columns.iter().enumerate() {
223        match rule {
224            DateRule::Ambiguous => report.warn(format!(
225                "column {}: every date is ambiguous, so it was read month first ({})",
226                col_to_letters(col as u32),
227                evidence.ambiguous_example.as_deref().unwrap_or("a date")
228            )),
229            DateRule::Conflict => report.warn(format!(
230                "column {}: it holds both day-first and month-first dates ({} and {}), so its dates stay text",
231                col_to_letters(col as u32),
232                evidence.day_first_example.as_deref().unwrap_or("a day-first date"),
233                evidence.month_first_example.as_deref().unwrap_or("a month-first date")
234            )),
235            DateRule::NoDates | DateRule::MonthFirst | DateRule::DayFirst => {}
236        }
237    }
238}

240/// Reads one field with the column's date rule; other values go through typed input.
241fn read_field(field: &str, sys: DateSystem, rule: DateRule) -> Parsed {
242    match numeric_date(field, sys) {
243        Some(date) if rule != DateRule::NoDates => date.read(field, rule),
244        _ => parse_input(field, sys),
245    }
246}

249    read_csv_with_report(bytes, opts).map(|(wb, _)| wb)
250}

252/// [`read_csv`], with the warnings an open shows: a column whose date order had to be guessed,
253/// or whose dates stayed text because it holds both orders.
254pub fn read_csv_with_report(bytes: &[u8], opts: &CsvOptions) -> Result<(Workbook, ReadReport), IoError> {
255    let text = decode_text(bytes);
256    let quote = if opts.quote == 0 { '"' } else { opts.quote as char };
257    let delim = if opts.delimiter == 0 { sniff_delimiter(&text, quote) } else { opts.delimiter } as char;
258    if delim == quote {
259        return Err(IoError::Format("the delimiter and quote character must differ".into()));
260    }
261    let date_columns = if opts.parse_values { infer_date_rules(&sample_records(&text, delim, quote), opts.date_system) } else { vec![] };
262    let mut report = ReadReport::default();
263    date_warnings(&date_columns, &mut report);
264    let mut wb = Workbook::new();
265    wb.date_system = opts.date_system;
266    wb.csv_delimiter = u8::try_from(delim).ok();
267    let mut sheet = Sheet::new("Sheet1");
268    let mut styles = std::mem::take(&mut wb.styles);
269    let mut fmt_cache: Vec<(&'static str, StyleId)> = vec![];
270    let mut row: u32 = 0;
271    // The bounding box of the non-empty cells, which is the sheet's used range.
272    let mut used: Option<(u32, u32, u32, u32)> = None;
273    let mut too_many = false;
274    parse_records(&text, delim, quote, |rec| {
275        if row >= MAX_ROWS {
276            return false;
277        }
278        // A blank line is an empty row.
279        for (col, f) in rec.into_iter().enumerate() {
280            if col as u32 >= MAX_COLS {
281                break;
282            }
283            if f.is_empty() {
284                continue;
285            }
286            let pos = CellRef::new(row, col as u32);
287            let (r0, c0, c1) = used.map_or((row, col as u32, col as u32), |(r0, c0, _, c1)| (r0, c0.min(col as u32), c1.max(col as u32)));
288            used = Some((r0, c0, row, c1));
289            let cell = if opts.parse_values {
290                let rule = date_columns.get(col).map_or(DateRule::NoDates, |(rule, _)| *rule);
291                let p = read_field(&f, opts.date_system, rule);
292                let style = match p.format {
293                    Some(code) => match fmt_cache.iter().find(|(c, _)| *c == code) {
294                        Some((_, id)) => *id,
295                        None => {
296                            let id = styles.derive(StyleId::DEFAULT, |s| s.num_fmt = NumFmt::new(code));
297                            fmt_cache.push((code, id));
298                            id
299                        }
300                    },
301                    None => StyleId::DEFAULT,
302                };
303                Cell { value: p.value, formula: None, style, py: None }
304            } else {
305                Cell::value(Value::Text(Arc::from(f)))
306            };
307            sheet.cells.set(pos, cell);
308        }
309        // Stop at the first record that takes the used range past the limit, before more is stored.
310        if let (Some(max), Some((r0, c0, r1, c1))) = (opts.max_used_cells, used)
311            && u64::from(r1 - r0 + 1) * u64::from(c1 - c0 + 1) > max
312        {
313            too_many = true;
314            return false;
315        }
316        row += 1;
317        true
318    });
319    if too_many {
320        return Err(IoError::TooLarge(format!("the CSV's used range is over the limit of {} cells", opts.max_used_cells.unwrap_or(0))));
321    }
322    wb.styles = styles;
323    wb.sheets = vec![Arc::new(sheet)];
324    Ok((wb, report))
325}

327/// Writes the sheet as CSV from A1 to the end of its used range, so leading empty columns and rows
328/// stay as empty fields and every cell keeps its position (CRLF line ends, UTF-8 without BOM). `cell_text` gives
329/// each cell's field text: the caller decides how values look (Save As CSV writes them as
330/// displayed, number formats applied); quoting is done here.
331///
332/// The output is capped at [`MAX_CSV_BYTES`]: a cell far from A1 makes the file as big as the whole
333/// rectangle between them (`A1` and `XFD1048576` are 16 GiB of commas).
334/// An export that would pass the cap is refused before anything is allocated, or at the first
335/// field or line end that takes it past the cap (a field is measured before it is written), with
336/// [`IoError::TooLarge`]; accepted exports keep every field in its place.
337pub fn write_csv(sheet: &Sheet, delimiter: u8, cell_text: impl FnMut(CellRef) -> String) -> Result<Vec<u8>, IoError> {
338    write_csv_limited(sheet, delimiter, MAX_CSV_BYTES, cell_text)
339}

341/// [`write_csv`] with the output cap `limit` (bytes) given.
342pub fn write_csv_limited(sheet: &Sheet, delimiter: u8, limit: usize, mut cell_text: impl FnMut(CellRef) -> String) -> Result<Vec<u8>, IoError> {
343    let delim = if delimiter == 0 { ',' } else { delimiter as char };
344    let mut out = String::new();
345    let Some(r) = sheet.used_range() else { return Ok(Vec::new()) };
346    let too_large =
347        |what: String| IoError::TooLarge(format!("the CSV would be {what}, over the limit of {} MiB; export a smaller range", limit >> 20));
348    // Every row has a delimiter between its fields and a CRLF, whatever the cells hold: a lower
349    // bound on the output that needs no cell text.
350    let (rows, cols) = (u64::from(r.end.row) + 1, u64::from(r.end.col) + 1);
351    let per_row = (cols - 1).saturating_mul(delim.len_utf8() as u64).saturating_add(2);
352    let least = rows.saturating_mul(per_row);
353    if least > limit as u64 {
354        return Err(too_large(format!("at least {} MiB ({rows} rows by {cols} columns)", least >> 20)));
355    }
356    for row in 0..=r.end.row {
357        for col in 0..=r.end.col {
358            if col > 0 {
359                out.push(delim);
360            }
361            let t = cell_text(CellRef::new(row, col));
362            // The exact size of the field as written (the delimiter was pushed above), so a row
363            // never grows past the cap unseen.
364            if out.len().saturating_add(crate::craft::csv::field_len(&t, delim)) > limit {
365                return Err(too_large(format!("more than {} MiB (stopped at row {})", limit >> 20, row + 1)));
366            }
367            crate::craft::csv::push_field(&mut out, &t, delim);
368        }
369        out.push_str("\r\n");
370        if out.len() > limit {
371            return Err(too_large(format!("more than {} MiB (stopped at row {})", limit >> 20, row + 1)));
372        }
373    }
374    Ok(out.into_bytes())
