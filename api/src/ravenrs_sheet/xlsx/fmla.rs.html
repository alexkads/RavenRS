1//! Formula text conversion between the model (canonical text, no `=`) and files (`_xlfn.`
2//! prefixes for newer functions).

7/// Formula text for a file.
8pub fn to_file(f: &Formula) -> String {
9    match f.expr_arc() {
10        // Most formulas have nothing the file spells differently: print them without copying.
11        Some(e) if !needs_file_spelling(&e) => crate::formula::print(&e),
12        Some(e) => expr_to_file(Expr::clone(&e)),
13        None => f.text.clone(),
14    }
15}
16
17/// Whether [`expr_to_file`] would change anything in `e`: a LET or LAMBDA (parameter prefixes),
18/// a function with a file prefix, a spill, `@` or trim operator, or a structured reference.
19fn needs_file_spelling(e: &Expr) -> bool {
20    let mut found = false;
21    e.walk(&mut |x| {
22        found |= match x {
23            Expr::Call(name, _) => name == "LET" || name == "LAMBDA" || crate::xlsx::tables::file_function_name(name).is_some(),
24            Expr::Unary(crate::formula::UnOp::Spill | crate::formula::UnOp::At | crate::formula::UnOp::Trim(_), _) => true,
25            Expr::Struct(_) => true,
26            _ => false,
27        }
28    });
29    found
30}

32/// Arbitrary formula text (names, CF, validation) for a file. Unparseable text is kept.
33pub fn text_to_file(text: &str) -> String {
34    let body = text.strip_prefix('=').unwrap_or(text);
35    match crate::formula::parse(body) {
36        Ok(e) => expr_to_file(e),
37        Err(_) => body.to_string(),
38    }
39}

41const PARAM_PREFIX: &str = "_xlpm.";
42
43/// Adds Excel's `_xlpm.` prefix to LET/LAMBDA parameter names, at the declaration and at every use in scope.
44fn prefix_params(e: Expr, scope: &[String]) -> Expr {
45    let sub = |x: Box<Expr>, sc: &[String]| Box::new(prefix_params(*x, sc));
46    match e {
47        Expr::Name(n) if scope.iter().any(|s| s.eq_ignore_ascii_case(&n)) => Expr::Name(format!("{PARAM_PREFIX}{n}")),
48        Expr::Unary(op, x) => Expr::Unary(op, sub(x, scope)),
49        Expr::Paren(x) => Expr::Paren(sub(x, scope)),
50        Expr::Binary(op, a, b) => {
51            // A long chain such as A1+A2+… nests on its left: follow it in a loop, not one
52            // level of recursion per operand.
53            let mut spine = vec![(op, b)];
54            let mut left = *a;
55            while let Expr::Binary(op, a, b) = left {
56                spine.push((op, b));
57                left = *a;
58            }
59            let mut acc = prefix_params(left, scope);
60            while let Some((op, b)) = spine.pop() {
61                acc = Expr::Binary(op, Box::new(acc), sub(b, scope));
62            }
63            acc
64        }
65        Expr::Invoke(c, args) => Expr::Invoke(sub(c, scope), args.into_iter().map(|a| prefix_params(a, scope)).collect()),
66        Expr::Call(name, args) if name == "LET" || name == "LAMBDA" => {
67            let last = args.len().saturating_sub(1);
68            let is_let = name == "LET";
69            let mut sc = scope.to_vec();
70            let mut pending: Option<String> = None;
71            let mut out = Vec::with_capacity(args.len());
72            for (i, a) in args.into_iter().enumerate() {
73                let declares = i < last && (!is_let || i % 2 == 0);
74                match a {
75                    Expr::Name(n) if declares => {
76                        out.push(Expr::Name(format!("{PARAM_PREFIX}{n}")));
77                        if is_let {
78                            pending = Some(n);
79                        } else {
80                            sc.push(n);
81                        }
82                    }
83                    other => {
84                        out.push(prefix_params(other, &sc));
85                        // A LET name is in scope only after its value expression.
86                        if let Some(n) = pending.take() {
87                            sc.push(n);
88                        }
89                    }
90                }
91            }
92            Expr::Call(name, out)
93        }
94        // A LET-bound LAMBDA called by name, e.g. `f(3)`.
95        Expr::Call(name, args) if scope.iter().any(|s| s.eq_ignore_ascii_case(&name)) => {
96            Expr::Call(format!("{PARAM_PREFIX}{name}"), args.into_iter().map(|a| prefix_params(a, scope)).collect())
97        }
98        Expr::Call(name, args) => Expr::Call(name, args.into_iter().map(|a| prefix_params(a, scope)).collect()),
99        other => other,
100    }
101}

103fn expr_to_file(e: Expr) -> String {
104    let e = prefix_params(e, &[]);
105    let e = e.map(&mut |x| match x {
106        Expr::Call(name, args) => match crate::xlsx::tables::file_function_name(&name) {
107            Some(n) => Expr::Call(n, args),
108            None => Expr::Call(name, args),
109        },
110        // Files have no `#` operator: Excel writes `A1#` as `_xlfn.ANCHORARRAY(A1)`, and a spill
111        // reference that became `#REF!` (its cell was deleted) as plain `#REF!`.
112        Expr::Unary(crate::formula::UnOp::Spill, r) if matches!(*r, Expr::Error(_)) => *r,
113        Expr::Unary(crate::formula::UnOp::Spill, r) => Expr::Call("_xlfn.ANCHORARRAY".into(), vec![*r]),
114        // Nor an `@` operator: Excel writes `@x` as `_xlfn.SINGLE(x)`.
115        Expr::Unary(crate::formula::UnOp::At, x) => Expr::Call("_xlfn.SINGLE".into(), vec![*x]),
116        // Nor the `[@Col]` shorthand: files spell it `[[#This Row],[Col]]`, or Excel repairs
117        // the file and drops the table. Carried as a name, which prints verbatim.
118        Expr::Struct(st) => Expr::Name(crate::formula::printer::struct_ref_file(&st)),
119        // Nor trim-reference operators: written as the equivalent `TRIMRANGE(range,n,n)`, which
120        // every Excel with trim references reads. (Excel's own file spelling of `A1:.E10` is
121        // not publicly documented; the `_xlfn._TRO_*` call forms are accepted on read.)
122        Expr::Unary(crate::formula::UnOp::Trim(mode), x) => {
123            let code = Expr::Number(f64::from(mode.code()));
124            Expr::Call("_xlfn.TRIMRANGE".into(), vec![*x, code.clone(), code])
125        }
126        other => other,
127    });
128    crate::formula::print(&e)
129}

131/// `_xlpm.x` → `x` (any case, as the parser upper-cases function names).
132fn strip_param_prefix(n: String) -> String {
133    match (n.get(..PARAM_PREFIX.len()), n.get(PARAM_PREFIX.len()..)) {
134        (Some(p), Some(rest)) if p.eq_ignore_ascii_case(PARAM_PREFIX) => rest.to_string(),
135        _ => n,
136    }
137}

139/// Formula text read from a file → model text (strips prefixes when it parses).
140pub fn from_file(text: &str) -> String {
141    let body = text.strip_prefix('=').unwrap_or(text);
142    match crate::formula::parse(body) {
143        Ok(e) => crate::formula::print(&strip_param_prefixes(e)),
144        Err(_) => body.to_string(),
145    }
146}
147
148fn strip_param_prefixes(e: Expr) -> Expr {
149    e.map(&mut |x| match x {
150        Expr::Name(n) => Expr::Name(strip_param_prefix(n)),
151        Expr::Call(n, args) => Expr::Call(strip_param_prefix(n), args),
152        other => other,
153    })
154}

156/// The function Google Sheets' XLSX export wraps formulas in when it thinks Excel lacks one of
157/// their functions.
158const DUMMY_FUNCTION: &str = "__xludf.DUMMYFUNCTION";
159
160/// Whether `text` may hold Google Sheets' export wrapper: a cheap test before parsing.
161fn mentions_dummy_function(text: &str) -> bool {
162    const NEEDLE: &[u8] = b"DUMMYFUNCTION";
163    text.as_bytes().windows(NEEDLE.len()).any(|w| w.eq_ignore_ascii_case(NEEDLE))
164}

166/// The model text of `text` (a formula as read, see [`from_file`]) with its Google Sheets export
167/// wrappers replaced (see [`crate::xlsx::read_xlsx_with`]), or `None` when nothing changes.
168pub(crate) fn unwrap_google_sheets(text: &str, known: &dyn Fn(&str) -> bool) -> Option<String> {
169    if !mentions_dummy_function(text) {
170        return None;
171    }
172    let e = crate::formula::parse(text.strip_prefix('=').unwrap_or(text)).ok()?;
173    let unwrapped = unwrap_dummy_functions(e, known)?;
174    Some(crate::formula::print(&strip_param_prefixes(unwrapped)))
175}

177/// The original formula inside Google Sheets' export wrapper
178/// `IFERROR(__xludf.DUMMYFUNCTION("<formula, quotes doubled>"),<cached value>)`, if `e` is that
179/// wrapper, the text parses and every function it calls is `known`. The parser has already undone
180/// the doubled quotes. The cached value is also the cell's `<v>`, which the reader keeps.
181fn dummy_function_inner(e: &Expr, known: &dyn Fn(&str) -> bool) -> Option<Expr> {
182    let Expr::Call(name, args) = e else { return None };
183    // The parser has already dropped an `_xlfn.` prefix from IFERROR.
184    if !name.eq_ignore_ascii_case("IFERROR") {
185        return None;
186    }
187    let [Expr::Call(dummy, inner), _fallback] = args.as_slice() else { return None };
188    if !dummy.eq_ignore_ascii_case(DUMMY_FUNCTION) {
189        return None;
190    }
191    let [Expr::Text(src)] = inner.as_slice() else { return None };
192    // The text came out of a formula within the length limit, so it is within it too; check
193    // anyway rather than rely on that.
194    if src.len() > crate::formula::MAX_FORMULA_LEN && src.chars().count() > crate::formula::MAX_FORMULA_LEN {
195        return None;
196    }
197    let parsed = crate::formula::parse(src).ok()?;
198    // A function GridCraft can't calculate would show `#NAME?` where Excel shows the cached
199    // value; a wrapper inside the text is one too, so only one level is ever unwrapped.
200    let mut all_known = true;
201    parsed.walk(&mut |x| {
202        if let Expr::Call(n, _) = x {
203            all_known &= known(n);
204        }
205    });
206    all_known.then_some(parsed)
207}

209/// `e` with each Google Sheets export wrapper (see [`dummy_function_inner`]) replaced by the
210/// formula it holds, for the whole formula or anywhere inside it, or `None` when none is. If the
211/// result no longer parses (nesting beyond the parser's limit), nothing is replaced.
212fn unwrap_dummy_functions(e: Expr, known: &dyn Fn(&str) -> bool) -> Option<Expr> {
213    // The whole formula: no parentheses needed.
214    if let Some(inner) = dummy_function_inner(&e, known) {
215        return Some(inner);
216    }
217    let mut found = false;
218    let rewritten = e.map(&mut |x| match dummy_function_inner(&x, known) {
219        // The printer writes no parentheses of its own, so keep the inner formula's precedence
220        // with explicit ones unless it is a single operand.
221        Some(
222            inner @ (Expr::Call(..)
223            | Expr::Ref(_)
224            | Expr::Name(_)
225            | Expr::Number(_)
226            | Expr::Text(_)
227            | Expr::Bool(_)
228            | Expr::Error(_)
229            | Expr::Paren(_)
230            | Expr::Array(_)
231            | Expr::Struct(_)),
232        ) => {
233            found = true;
234            inner
235        }
236        Some(inner) => {
237            found = true;
238            Expr::Paren(Box::new(inner))
239        }
240        None => x,
241    });
242    if !found {
243        return None;
244    }
245    crate::formula::parse(&crate::formula::print(&rewritten)).ok()?;
246    Some(rewritten)
247}

442            "__xludf.DUMMYFUNCTION(\"A1\")",
443        ] {
444            let kept = crate::formula::print(&crate::formula::parse(file).unwrap());
445            assert_eq!(read(file), kept, "{file}");
446            assert_eq!(unwrap_google_sheets(&from_file(file), &test_known_function), None, "{file}");
447        }
448        // Inner text over the formula length limit.
449        let long = format!("IFERROR(__xludf.DUMMYFUNCTION(\"{}\"),5)", "1".repeat(crate::formula::MAX_FORMULA_LEN + 1));
450        assert!(read(&long).contains("DUMMYFUNCTION"));
451        // Nesting that would go past the parser's depth limit once unwrapped is kept as well.
452        let deep = |n: usize, mid: &str| format!("{}{mid}{}", "(".repeat(n), ")".repeat(n));
453        let inner = deep(200, "1").replace('"', "\"\"");
454        let file = deep(200, &format!("1+IFERROR(__xludf.DUMMYFUNCTION(\"{inner}\"),5)"));
455        assert!(crate::formula::parse(&file).is_ok());
456        let read = read(&file);
457        assert!(read.contains("DUMMYFUNCTION"));
458        assert!(crate::formula::parse(&read).is_ok());
459    }

462    fn let_lambda_params() {
463        assert_eq!(to_file(&Formula::new("LET(x,1,x+1)")), "_xlfn.LET(_xlpm.x,1,_xlpm.x+1)");
464        assert_eq!(to_file(&Formula::new("LET(x,A1*2,y,x+1,x*y)")), "_xlfn.LET(_xlpm.x,A1*2,_xlpm.y,_xlpm.x+1,_xlpm.x*_xlpm.y)");
465        assert_eq!(to_file(&Formula::new("LAMBDA(a,b,a^2+b)(3,4)")), "_xlfn.LAMBDA(_xlpm.a,_xlpm.b,_xlpm.a^2+_xlpm.b)(3,4)");
466        // A name outside the scope stays a plain defined name.
467        assert_eq!(to_file(&Formula::new("LET(x,1,x)+x")), "_xlfn.LET(_xlpm.x,1,_xlpm.x)+x");
468        assert_eq!(from_file("_xlfn.LET(_xlpm.x,1,_xlpm.x+1)"), "LET(x,1,x+1)");
469        assert_eq!(from_file("_xlfn.LAMBDA(_xlpm.a,_xlpm.a*2)(3)"), "LAMBDA(a,a*2)(3)");
470        // A LET-bound LAMBDA called by name.
471        assert_eq!(to_file(&Formula::new("LET(f,LAMBDA(x,x*2),f(3))")), "_xlfn.LET(_xlpm.f,_xlfn.LAMBDA(_xlpm.x,_xlpm.x*2),_xlpm.F(3))");
472        assert_eq!(from_file("_xlfn.LET(_xlpm.f,_xlfn.LAMBDA(_xlpm.x,_xlpm.x*2),_xlpm.f(3))"), "LET(f,LAMBDA(x,x*2),F(3))");
473    }
