1//! Export-time implicit-intersection analysis, without evaluating the workbook.
2//!
3//! A range is harmless in SUM, but not in SQRT or arithmetic. Keep the result's shape separate
4//! from affected operations inside it: SUM(range * range) returns a scalar yet needs array
5//! evaluation. Legacy array-aware contexts such as SUMPRODUCT already preserve those operations.
6
7use crate::base::CellRef;
8use crate::formula::{BinOp, Expr, SheetSel, StructItem, UnOp};
9use crate::model::{Formula, Workbook};
10
11#[derive(Clone, Copy, Default)]
12struct Shape {
13    array: bool,
14    affected: bool,
15}
16
17impl Shape {
18    fn array() -> Self {
19        Self { array: true, affected: false }
20    }
21}
22
23pub(crate) fn needs_array(wb: &Workbook, sheet: usize, at: CellRef, formula: &Formula) -> bool {
24    let Some(expr) = formula.expr_arc() else { return false };
25    // The budget covers every node of a formula as long as Excel allows (8,192 characters make
26    // at most about 8,200 nodes), so only names that expand a lot fall back to array evaluation.
27    let mut scan = Scan { wb, at, remaining: 16_384, locals: Vec::new() };
28    let shape = scan.expr(&expr, sheet, false, 0);
29    // An explicit `@` is written as `_xlfn.SINGLE`, which only dynamic-array formulas carry.
30    let mut at = false;
31    expr.walk(&mut |e| at |= matches!(e, Expr::Unary(UnOp::At, _)));
32    shape.array || shape.affected || at
33}
34
35struct Scan<'a> {
36    wb: &'a Workbook,
37    at: CellRef,
38    remaining: usize,
39    locals: Vec<(String, Shape)>,
40}
41
42impl Scan<'_> {
43    fn expr(&mut self, e: &Expr, sheet: usize, array_context: bool, depth: usize) -> Shape {
44        // Names can be cyclic, and a short chain of repeated names can expand exponentially.
45        // Preserve array evaluation when analysis cannot safely finish.
46        if depth >= 64 || self.remaining == 0 {
47            return Shape { array: true, affected: true };
48        }
49        self.remaining -= 1;
50        match e {
51            Expr::Ref(r) => Shape { array: !r.range().is_single() || matches!(r.sheet, SheetSel::Span(_, _)), affected: false },
52            Expr::Array(rows) => Shape { array: rows.len() > 1 || rows.first().is_some_and(|r| r.len() > 1), affected: false },
53            Expr::Name(n) => {
54                if let Some((_, shape)) = self.locals.iter().rev().find(|(key, _)| key.eq_ignore_ascii_case(n)) {
55                    return *shape;
56                }
57                let (scope, bare) = match n.rsplit_once('!') {
58                    Some((s, name)) => (self.wb.sheet_index(&crate::base::unquote_sheet(s)).unwrap_or(sheet), name),
59                    None => (sheet, n.as_str()),
60                };
61                if let Some(def) = self.wb.name(bare, scope) {
62                    return match crate::formula::parse_name(def.formula.trim_start_matches('=')) {
63                        Ok(expr) => self.expr(&expr, def.scope.unwrap_or(scope), array_context, depth + 1),
64                        Err(_) => Shape::array(),
65                    };
66                }
67                Shape { array: self.wb.table(bare).is_some(), affected: false }
68            }
69            Expr::Struct(r) => {
70                let table = if r.table.is_empty() {
71                    self.wb.sheet(sheet).and_then(|s| s.table_at(self.at))
72                } else {
73                    self.wb.table(&r.table).and_then(|(si, ti)| self.wb.sheet(si).and_then(|s| s.tables.get(ti)))
74                };
75                let one_row = r.specifiers.iter().any(|s| matches!(s, StructItem::ThisRow | StructItem::Headers | StructItem::Totals));
76                let one_col = match (&r.col_start, &r.col_end) {
77                    (Some(a), Some(b)) => a.eq_ignore_ascii_case(b),
78                    (Some(_), None) => true,
79                    _ => table.is_some_and(|t| t.columns.len() == 1),
80                };
81                Shape { array: !(one_row && one_col), affected: false }
82            }
83            Expr::Paren(x) => self.expr(x, sheet, array_context, depth + 1),
84            Expr::Unary(op, x) => {
85                let child = self.expr(x, sheet, array_context, depth + 1);
86                match op {
87                    UnOp::At => Shape { array: false, affected: child.affected },
88                    // A trim reference is a reference operator, like `:`.
89                    UnOp::Spill | UnOp::Trim(_) => Shape { array: true, affected: child.affected },
90                    _ => Shape { array: child.array, affected: child.affected || (child.array && !array_context) },
91                }
92            }
93            Expr::Binary(op, a, b) => {
94                // A flat chain such as A1+A2+… nests one level per operator on its left: follow
95                // it in a loop, so a long chain is neither deep recursion nor past the depth
96                // limit (which would write a plain sum as an array formula).
97                let mut spine = vec![(*op, b.as_ref())];
98                let mut left = a.as_ref();
99                while let Expr::Binary(op, a, b) = left {
100                    if self.remaining == 0 {
101                        return Shape { array: true, affected: true };
102                    }
103                    self.remaining -= 1;
104                    spine.push((*op, b.as_ref()));
105                    left = a.as_ref();
106                }
107                let mut acc = self.expr(left, sheet, array_context, depth + 1);
108                while let Some((op, b)) = spine.pop() {
109                    let b = self.expr(b, sheet, array_context, depth + 1);
110                    let reference = matches!(op, BinOp::Range | BinOp::Union | BinOp::Intersect);
111                    let array = reference || acc.array || b.array;
112                    acc = Shape { array, affected: acc.affected || b.affected || (!reference && array && !array_context) };
113                }
114                acc
115            }
116            Expr::Call(name, args) => self.call(name, args, sheet, array_context, depth + 1),
117            Expr::Invoke(_, _) => Shape { array: true, affected: false },
118            _ => Shape::default(),
119        }
120    }
121
122    fn call(&mut self, name: &str, args: &[Expr], sheet: usize, array_context: bool, depth: usize) -> Shape {
123        if name == "LET"
124            && args.len() >= 3
125            && let Some((last, bindings)) = args.split_last()
126        {
127            let saved = self.locals.len();
128            let mut affected = false;
129            for pair in bindings.as_chunks::<2>().0 {
130                if let [Expr::Name(n), value] = pair {
131                    let shape = self.expr(value, sheet, array_context, depth);
132                    affected |= shape.affected;
133                    self.locals.push((n.clone(), shape));
134                }
135            }
136            let mut result = self.expr(last, sheet, array_context, depth);
137            result.affected |= affected;
138            self.locals.truncate(saved);
139            return result;
140        }
141        let legacy_array =
142            matches!(name, "SUMPRODUCT" | "MMULT" | "MINVERSE" | "MDETERM" | "TRANSPOSE" | "FREQUENCY" | "LINEST" | "LOGEST" | "TREND" | "GROWTH");
143        let mut affected = false;
144        let mut lifted = false;
145        let mut shapes = Vec::with_capacity(args.len().min(256));
146        if args.len() > 255 {
147            return Shape { array: true, affected: true };
148        }
149        for (i, arg) in args.iter().enumerate() {
150            let context = array_context || legacy_array || (name == "INDEX" && i == 0);
151            let shape = self.expr(arg, sheet, context, depth);
152            affected |= shape.affected;
153            if shape.array && scalar_arg(name, i) {
154                lifted = true;
155                affected |= !context;
156            }
157            shapes.push(shape);
158        }
159        let first_array = shapes.first().is_some_and(|s| s.array);
160        let array = match name {
161            "INDIRECT" => match args.first() {
162                Some(Expr::Text(text)) if !matches!(args.get(1), Some(Expr::Bool(false))) => {
163                    crate::formula::parse(text).map(|e| self.expr(&e, sheet, array_context, depth).array).unwrap_or(true)
164                }
165                _ => true,
166            },
167            "OFFSET" => {
168                // Omitted dimensions retain the source reference's shape; unknown dimensions
169                // might become multiple cells on a later calculation.
170                if args.len() <= 3 { first_array } else { !args.get(3).is_some_and(one) || args.get(4).map_or(first_array, |e| !one(e)) }
171            }
172            "INDEX" => {
173                let row = args.get(1);
174                let col = args.get(2);
175                let row_positive = row.is_some_and(|e| positive_index(e, shapes.get(1).copied()));
176                let col_positive = col.is_some_and(|e| positive_index(e, shapes.get(2).copied()));
177                !row_positive || col.is_some_and(|_| !col_positive) || (col.is_none() && !args.first().is_some_and(single_column))
178            }
179            "ROW" | "COLUMN" => {
180                affected |= first_array && !array_context;
181                first_array
182            }
183            "IF" | "IFERROR" | "IFNA" | "IFS" | "SWITCH" | "CHOOSE" => shapes.iter().any(|s| s.array),
184            "XLOOKUP" => {
185                lifted
186                    || match (args.get(1), args.get(2)) {
187                        (Some(Expr::Ref(lookup)), Some(Expr::Ref(result))) => {
188                            if lookup.range().width() == 1 {
189                                result.range().width() > 1
190                            } else {
191                                result.range().height() > 1
192                            }
193                        }
194                        _ => true,
195                    }
196            }
197            "FILTER" | "SORT" | "SORTBY" | "UNIQUE" | "SEQUENCE" | "RANDARRAY" | "VSTACK" | "HSTACK" | "TOROW" | "TOCOL" | "WRAPROWS"
198            | "WRAPCOLS" | "TAKE" | "DROP" | "CHOOSEROWS" | "CHOOSECOLS" | "EXPAND" | "GROUPBY" | "PIVOTBY" | "TRANSPOSE" | "MMULT" | "MINVERSE"
199            | "FREQUENCY" | "LINEST" | "LOGEST" | "TREND" | "GROWTH" | "MAP" | "SCAN" | "BYROW" | "BYCOL" | "MAKEARRAY" | "ANCHORARRAY" => true,
200            _ => lifted || self.wb.name(name, sheet).is_some(),
201        };
202        Shape { array, affected }
203    }
204}

206fn positive(e: &Expr) -> bool {
207    matches!(e, Expr::Number(n) if *n >= 1.0)
208}
209
210fn positive_index(e: &Expr, shape: Option<Shape>) -> bool {
211    // MATCH returns a one-based position or an error, never INDEX's zero "whole row" selector.
212    positive(e) || (matches!(e, Expr::Call(name, _) if matches!(name.as_str(), "MATCH" | "XMATCH")) && shape.is_some_and(|s| !s.array))
213}
214
215fn one(e: &Expr) -> bool {
216    matches!(e, Expr::Number(n) if *n == 1.0)
217}
218
219fn single_column(e: &Expr) -> bool {
220    matches!(e, Expr::Ref(r) if r.range().width() == 1)
221}

223/// Legacy reference/array-taking parameters. Other parameters consume a single value.
224/// This is file-format policy, not a second evaluator: unknown results are never calculated.
225fn scalar_arg(name: &str, i: usize) -> bool {
226    match name {
227        "SUM" | "AVERAGE" | "AVERAGEA" | "MIN" | "MINA" | "MAX" | "MAXA" | "COUNT" | "COUNTA" | "COUNTBLANK" | "PRODUCT" | "SUMSQ" | "MEDIAN"
228        | "MODE" | "MODE.SNGL" | "STDEV" | "STDEV.S" | "STDEV.P" | "STDEVP" | "VAR" | "VAR.S" | "VAR.P" | "VARP" | "AVEDEV" | "DEVSQ" | "GEOMEAN"
229        | "HARMEAN" | "KURT" | "SKEW" | "SKEW.P" | "AND" | "OR" | "XOR" | "SUMPRODUCT" | "SUMX2MY2" | "SUMX2PY2" | "SUMXMY2" | "CORREL" | "COVAR"
230        | "COVARIANCE.P" | "COVARIANCE.S" | "PEARSON" | "RSQ" | "SLOPE" | "INTERCEPT" | "STEYX" | "ROWS" | "COLUMNS" | "AREAS" | "ROW" | "COLUMN"
231        | "TRANSPOSE" | "ISREF" | "MMULT" | "MINVERSE" | "MDETERM" | "FREQUENCY" | "CONCAT" | "VSTACK" | "HSTACK" | "ANCHORARRAY" => false,
232        "SUMIF" | "AVERAGEIF" => i == 1,
233        "COUNTIF" => i != 0,
234        "SUMIFS" | "AVERAGEIFS" | "MINIFS" | "MAXIFS" => i > 0 && i.is_multiple_of(2),
235        "COUNTIFS" => i % 2 == 1,
236        "INDEX" | "LARGE" | "SMALL" | "PERCENTILE" | "PERCENTILE.INC" | "PERCENTILE.EXC" | "QUARTILE" | "QUARTILE.INC" | "QUARTILE.EXC" | "IRR"
237        | "MIRR" | "RANK" | "RANK.EQ" | "RANK.AVG" => i != if name.starts_with("RANK") { 1 } else { 0 },
238        "VLOOKUP" | "HLOOKUP" | "MATCH" | "XMATCH" => i != 1,
239        "LOOKUP" | "NPV" | "XNPV" | "SUBTOTAL" => i == 0,
240        "XLOOKUP" => i != 1 && i != 2,
241        "XIRR" => i > 1,
242        "OFFSET" => i != 0,
243        "AGGREGATE" | "TEXTJOIN" => i < 2,
244        "FILTER" | "SORTBY" => i > 1,
245        "SORT" | "UNIQUE" | "TAKE" | "DROP" | "CHOOSECOLS" | "CHOOSEROWS" | "TOROW" | "TOCOL" | "WRAPROWS" | "WRAPCOLS" | "EXPAND" => i != 0,
246        "LINEST" | "LOGEST" | "TREND" | "GROWTH" => i > 1,
247        "FORECAST" | "FORECAST.LINEAR" => i == 0,
248        "FORECAST.ETS" | "FORECAST.ETS.CONFINT" => i != 1 && i != 2,
249        "FORECAST.ETS.SEASONALITY" | "FORECAST.ETS.STAT" => i > 1,
250        _ => true,
251    }
