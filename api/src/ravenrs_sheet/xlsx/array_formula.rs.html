1//! Legacy (pre-dynamic-array) cell formulas and implicit intersection, without evaluating.
2//!
3//! Excel evaluates a cell formula saved without dynamic-array metadata (no `cm` attribute) the
4//! legacy way: a reference that may cover several cells, used where one value is expected, is
5//! intersected with the formula's row or column, and a multi-valued result shows its top-left
6//! value. Dynamic-array Excel displays such a formula with the implicit intersection operator `@`
7//! inserted where that happens (Microsoft support, "Implicit intersection operator: @"). The
8//! reader does the same ([`apply_legacy`]), so the dynamic-array evaluator computes the legacy
9//! result; the writer ([`export`]) saves a formula that is the legacy reading of some text as
10//! that text, without the implied `@` (an `@` of a single value stays, as `_xlfn.SINGLE`), and
11//! gives every other formula dynamic-array metadata.
12//!
13//! A range is harmless in SUM, but not in SQRT or arithmetic: `SUM(A1:A3*B1:B3)` intersects both
14//! operands. Legacy array-aware parameters such as SUMPRODUCT's evaluate what is inside as
15//! arrays, so nothing there is intersected, except IF's condition and IFERROR's and IFNA's
16//! arguments ([`Param::Single`]). This is file-format policy, not a second evaluator: unknown
17//! results are never calculated.
18

28/// What an expression produces, as far as implicit intersection is concerned.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30enum Kind {
31    /// One value, or a reference to one cell.
32    Scalar,
33    /// A reference that may cover several cells.
34    Ref,
35    /// Several values that are not a reference.
36    Array,
37}
38
39impl Kind {
40    fn join(self, other: Kind) -> Kind {
41        match (self, other) {
42            (Kind::Array, _) | (_, Kind::Array) => Kind::Array,
43            (Kind::Ref, _) | (_, Kind::Ref) => Kind::Ref,
44            _ => Kind::Scalar,
45        }
46    }
47}

49/// How a position consumes its operand in a legacy formula.
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51enum Ctx {
52    /// One value: a multi-cell reference is intersected.
53    Value,
54    /// A reference or array as such (SUM's arguments); nested operations still take values.
55    Ref,
56    /// Array evaluation of everything inside (SUMPRODUCT's arguments): nothing is intersected.
57    Array,
58}
59/// A function parameter's legacy evaluation.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61enum Param {
62    Value,
63    /// One value even inside array evaluation: IF's condition and IFERROR's and IFNA's
64    /// arguments. Legacy Excel evaluates them as arrays inside SUMPRODUCT only in a
65    /// Ctrl+Shift+Enter formula, which this analysis never sees. A name's definition is still
66    /// evaluated as an array throughout ([`Legacy::kind_of`]).
67    Single,
68    Ref,
69    Array,
70    /// The parameter is evaluated like the call itself (IF's and CHOOSE's results).
71    Pass,
72    /// A value read from the top-left cell of a reference, never intersected (T: Excel caches
73    /// `T(H12:J12)` as T of H12 in a formula outside columns H:J). The function lifts over the
74    /// cells, so the legacy result is the top-left one.
75    First,
76    /// An array parameter of a dynamic-array-era function: whether Excel evaluates a legacy
77    /// formula there as an array is not established, so such formulas are written as dynamic.
78    Uncertain,
79}

81/// How the writer saves a formula that neither spills nor is a legacy CSE array.
82pub(crate) enum Export {
83    /// As it is, without dynamic-array metadata.
84    Unchanged,
85    /// As this expression (the formula without its implicit `@`), without dynamic-array metadata.
86    Legacy(Expr),
87    /// With dynamic-array metadata: legacy evaluation would differ.
88    Dynamic,
89}

91pub(crate) fn export(wb: &Workbook, sheet: usize, at: CellRef, formula: &Formula, functions: FunctionInfo, names: &mut NameCache) -> Export {
92    let Some(expr) = formula.expr_arc() else { return Export::Unchanged };
93    let mut explicit = false;
94    expr.walk(&mut |e| explicit |= matches!(e, Expr::Unary(UnOp::At, _)));
95    let mut legacy = Legacy::new(wb, at, functions, names);
96    if !explicit {
97        legacy.formula(expr.as_ref().clone(), sheet);
98        return if !legacy.certain {
99            Export::Dynamic
100        } else if !legacy.complete || legacy.inserted == 0 {
101            // A formula that does not spill and needs no `@`. One the analysis cannot finish
102            // reads back with at most a whole-formula `@`, which keeps its one value.
103            Export::Unchanged
104        } else {
105            Export::Dynamic
106        };
107    }
108    // The legacy text of the formula: each `@` legacy evaluation implies at its position is
109    // dropped (the reader puts it back), and so is one of a range an edit made one cell; any other
110    // `@` of a single value stays, written as `_xlfn.SINGLE` in the legacy formula as Excel does.
111    legacy.explicit = true;
112    let out = legacy.formula(expr.as_ref().clone(), sheet);
113    if !legacy.complete || !legacy.certain {
114        // A formula this analysis cannot finish or whose semantics it does not know, but that is
115        // the reader's reading of some text: saved as that text. The reader adds a whole-formula
116        // `@` over such formulas, or `@` where it intersects.
117        let mut candidates = vec![remove_at(expr.as_ref().clone())];
118        if let Expr::Unary(UnOp::At, x) = expr.as_ref() {
119            candidates.insert(0, unwrap_at(x.as_ref().clone()));
120        }
121        for text in candidates {
122            let read = self::legacy(wb, sheet, at, &Formula::from_expr(text.clone()), functions, names);
123            if read.map_or(text == *expr, |f| f.expr_arc().is_some_and(|e| *e == *expr)) {
124                return reads_back(text).map_or(Export::Dynamic, Export::Legacy);
125            }
126        }
127        return Export::Dynamic;
128    }
129    if !legacy.foreign && legacy.inserted == 0 { reads_back(out).map_or(Export::Dynamic, Export::Legacy) } else { Export::Dynamic }
130}

132/// `e` without any `@` and the parentheses [`at`] adds with one.
133fn remove_at(e: Expr) -> Expr {
134    e.map(&mut |x| match x {
135        Expr::Unary(UnOp::At, x) => unwrap_at(*x),
136        other => other,
137    })
138}

140/// Rewrites the legacy cell formulas at `cells` (model sheet index, positions) to their
141/// dynamic-array reading, with `@` where Excel intersects. Runs once names and tables are loaded.
142pub(crate) fn apply_legacy(wb: &mut Workbook, cells: Vec<(usize, Vec<CellRef>)>, functions: FunctionInfo) {
143    // Defined names do not change during the pass: each is analysed once for all the cells.
144    let mut names = NameCache::default();
145    for (si, positions) in cells {
146        let Some(sheet) = wb.sheet(si) else { continue };
147        let mut changed = Vec::new();
148        for at in positions {
149            let Some(formula) = sheet.cell(at).and_then(|c| c.formula.as_deref()) else { continue };
150            if let Some(f) = legacy(wb, si, at, formula, functions, &mut names) {
151                changed.push((at, f));
152            }
153        }
154        if changed.is_empty() {
155            continue;
156        }
157        let Some(sheet) = wb.sheet_mut(si) else { continue };
158        for (at, f) in changed {
159            if let Some(cell) = sheet.cells.get_mut(at) {
160                cell.formula = Some(Arc::new(f));
161            }
162        }
163    }
164}

166/// The dynamic-array reading of a legacy formula, when it differs.
167fn legacy(wb: &Workbook, sheet: usize, at: CellRef, formula: &Formula, functions: FunctionInfo, names: &mut NameCache) -> Option<Formula> {
168    if formula.array.is_some() {
169        return None;
170    }
171    let expr = formula.expr_arc()?;
172    let mut legacy = Legacy::new(wb, at, functions, names);
173    let out = legacy.formula(expr.as_ref().clone(), sheet);
174    if legacy.certain && legacy.complete {
175        if legacy.inserted == 0 {
176            return None;
177        }
178        // Text round trip: the stored expression is exactly what the text parses to.
179        if let Some(out) = reads_back(out) {
180            return Some(Formula::new(&crate::formula::print(&out)));
181        }
182        // The reading's `@` take it past the parser's nesting limit: handled as unfinished.
183    }
184    // An analysis that stopped at its limits, or met semantics it does not know (the array
185    // arguments of dynamic-array functions, LAMBDA), leaves the formula as it was read, with only
186    // the whole-formula `@` when the result may be several values: legacy Excel shows one value,
187    // so the formula must not spill. When that text does not parse either, the formula stays as
188    // it was read.
189    let whole = matches!(expr.as_ref(), Expr::Unary(UnOp::At, _));
190    let multi = legacy.top != Kind::Scalar || (legacy.certain && legacy.complete);
191    (multi && !whole).then(|| self::at(expr.as_ref().clone())).and_then(reads_back).map(|e| Formula::new(&crate::formula::print(&e)))
192}

194/// `e` when its text (the printer writes only the parentheses the expression has) reads back as
195/// the same expression, so a dropped `@` never takes parentheses the text needs with it. An `@`
196/// of a bare union has no such text, so its formula is saved as a dynamic one.
197fn reads_back(e: Expr) -> Option<Expr> {
198    crate::formula::parse(&crate::formula::print(&e)).is_ok_and(|p| p == e).then_some(e)
199}

201/// The operand of an `@` without the parentheses [`at`] adds.
202fn unwrap_at(e: Expr) -> Expr {
203    match e {
204        Expr::Paren(p) if !bare(&p) => *p,
205        other => other,
206    }
207}

209/// `@e`, parenthesized unless `e` prints as one operand. An operand already in parentheses (the
210/// file's own) gets a second pair, so [`unwrap_at`] can tell them from the ones added here.
211fn at(e: Expr) -> Expr {
212    let e = if bare(&e) { e } else { Expr::Paren(Box::new(e)) };
213    Expr::Unary(UnOp::At, Box::new(e))
214}

216/// Prints as one operand of `@`: Excel shows a spill reference's `@` without parentheses
217/// (`=@F4#`, `=@F5#%`).
218fn bare(e: &Expr) -> bool {
219    atomic(e) || matches!(e, Expr::Unary(UnOp::Spill, r) if matches!(**r, Expr::Ref(_)))
220}
221
222fn atomic(e: &Expr) -> bool {
223    matches!(e, Expr::Ref(_) | Expr::Name(_) | Expr::Struct(_) | Expr::Call(..) | Expr::Array(_))
224}

226/// Nesting the analysis follows: the formula parser's own limit. Functions, parentheses and
227/// prefix operators count one level each here as there, so any formula that parses is analysed
228/// in full unless names, or levels the parser does not count (postfix `%` and `#`), take it
229/// deeper; one that is not keeps its legacy reading ([`legacy`]). Operator chains do not count
230/// (see [`Legacy::binary`]).
231const MAX_DEPTH: usize = 256;

233/// What the analysis of one defined name leaves for a later formula that refers to it.
234#[derive(Clone)]
235struct NameInfo {
236    kind: Kind,
237    /// The definition reached a one-cell reference written as a range.
238    one_cell_range: bool,
239    /// The analysis covered only established legacy semantics.
240    certain: bool,
241    /// Nodes of the per-formula budget the analysis used.
242    cost: usize,
243    /// Nesting below the reference that the analysis reached.
244    depth: usize,
245    /// Every name the analysis read, with its result (itself last): a hit records them all, as a
246    /// fresh analysis does, so a later reference to one of them in the formula costs nothing.
247    closure: Vec<(NameKey, Kind, bool)>,
248}

250/// A defined name as [`Legacy::names`] keys it: upper-case name, scope, collapsed.
251type NameKey = (String, usize, bool);
252
253/// Defined names analysed so far, shared by the formulas of one pass (reading a workbook, or
254/// writing one) so that a name is parsed and analysed once, not once per formula that refers
255/// to it. Keyed like [`Legacy::names`]. Only analyses that finished within their limits, did not
256/// depend on the formula's own position and did not reuse a name the formula had already read
257/// (or meet a cycle) are kept. A hit charges the budget and nesting the analysis used and records
258/// every name it read, and is taken only when the formula can afford it, so a formula reads the
259/// same with the cache as without it.
260#[derive(Default)]
261pub(crate) struct NameCache {
262    map: HashMap<NameKey, NameInfo>,
263    /// How many names were analysed (not found in the cache).
264    pub(crate) analyses: usize,
265}

267/// Names kept at most: the cache is bounded even when a workbook defines a great many.
268const MAX_CACHED_NAMES: usize = 65_536;

270struct Legacy<'a> {
271    wb: &'a Workbook,
272    /// Name analyses shared with the other formulas of this pass.
273    shared: &'a mut NameCache,
274    /// The analysis read a structured reference without a table name, which depends on `at`.
275    position_dependent: bool,
276    /// The analysis reused a name this formula had already read, or met a cycle: its result
277    /// leaves out what that name cost.
278    reused: bool,
279    /// The names read so far in this formula, in order, with their results.
280    read: Vec<(NameKey, Kind, bool)>,
281    /// The deepest nesting reached.
282    deepest: usize,
283    at: CellRef,
284    functions: FunctionInfo,
285    remaining: usize,
286    locals: Vec<(String, Kind)>,
287    /// Kinds of the names analysed so far, by upper-case name, scope and whether one-cell ranges
288    /// counted as collapsed ones, with whether the definition reached a one-cell range written as
289    /// a range; `None` while a name's definition is being analysed (a cycle).
290    names: HashMap<(String, usize, bool), Option<(Kind, bool)>>,
291    /// How many `@` were inserted.
292    inserted: usize,
293    /// The analysis covered only established legacy semantics.
294    certain: bool,
295    /// The analysis finished within its nesting and size limits.
296    complete: bool,
297    /// Analysing a formula that has its own `@` (see [`Legacy::explicit_at`]).
298    explicit: bool,
299    /// A one-cell reference written as a range (`A1:A1`, what an edit leaves of `A1:A3`) counts
300    /// as one that may cover several cells, so an `@` the range had before the edit is implied.
301    collapsed: bool,
302    /// The analysis reached a one-cell reference written as a range, which `collapsed` changes.
303    one_cell_range: bool,
304    /// An explicit `@` is neither implied by legacy evaluation nor of a single value.
305    foreign: bool,
306    /// Analysing an expression whose rewrite is discarded (see [`Legacy::kind_of`]).
307    discarded: bool,
308    /// Analysing a union that is a function's single-value argument (see [`Legacy::expr`]).
309    union_arg: bool,
310    /// The kind of the whole formula, before any whole-formula `@`.
311    top: Kind,
312}

314impl<'a> Legacy<'a> {
315    fn new(wb: &'a Workbook, at: CellRef, functions: FunctionInfo, shared: &'a mut NameCache) -> Self {
316        Legacy {
317            wb,
318            shared,
319            position_dependent: false,
320            reused: false,
321            read: Vec::new(),
322            deepest: 0,
323            at,
324            functions,
325            remaining: 16384,
326            locals: Vec::new(),
327            names: HashMap::new(),
328            inserted: 0,
329            certain: true,
330            complete: true,
331            explicit: false,
332            collapsed: false,
333            one_cell_range: false,
334            foreign: false,
335            discarded: false,
336            union_arg: false,
337            top: Kind::Scalar,
338        }
339    }

341    /// The kind of an expression analysed only for its kind (a name's definition, INDIRECT's
342    /// text), evaluated as legacy Excel evaluates a name: as an array throughout, IF's condition
343    /// included. Its rewrite is discarded, so it inserts no `@` into the formula.
344    fn kind_of(&mut self, e: Expr, sheet: usize, depth: usize) -> Kind {
345        let outer = (std::mem::replace(&mut self.discarded, true), std::mem::replace(&mut self.explicit, false), self.inserted);
346        let (_, kind) = self.expr(e, sheet, Ctx::Array, depth);
347        (self.discarded, self.explicit, self.inserted) = outer;
348        kind
349    }

351    /// The whole cell formula: a multi-valued result shows its top-left value.
352    fn formula(&mut self, e: Expr, sheet: usize) -> Expr {
353        let e = match e {
354            Expr::Unary(UnOp::At, x) if self.explicit => return self.explicit_at(*x, sheet, Ctx::Value, 0, true).0,
355            e => e,
356        };
357        let (e, kind) = self.expr(e, sheet, Ctx::Value, 0);
358        self.top = kind;
359        if kind == Kind::Array && self.certain {
360            self.inserted += 1;
361            at(e)
362        } else {
363            e
364        }
365    }
366
367    /// An `@` already in the formula, at a position the analysis reads in `ctx`: analysed as its
368    /// operand alone. The `@` is the one legacy evaluation implies when the analysis puts it back
369    /// (that insertion is then not counted, and the `@` is dropped from the legacy text). An `@`
370    /// of a single value stays, as Excel keeps it (`_xlfn.SINGLE` in a legacy formula); any other
371    /// `@` makes the formula a dynamic-array one (`foreign`).
372    fn explicit_at(&mut self, x: Expr, sheet: usize, ctx: Ctx, depth: usize, top: bool) -> (Expr, Kind) {
373        let added = matches!(&x, Expr::Paren(p) if !bare(p));
374        let operand = unwrap_at(x);
375        let nested = matches!(operand, Expr::Unary(UnOp::At, _));
376        let before = self.inserted;
377        // A one-cell range the `@` applies to, directly or through what a name returns, may be
378        // what an edit left of a larger one: when the `@` is otherwise of a single value, it is
379        // implied if it is with such ranges counted as several cells. The retry's analysis is
380        // kept only when it implies the `@`, and runs only when the first pass reached such a
381        // range (otherwise it cannot differ).
382        let retry = (!nested && !self.collapsed && matches!(operand, Expr::Ref(_) | Expr::Name(_))).then(|| operand.clone());
383        let seen = std::mem::replace(&mut self.one_cell_range, false);
384        let (mut y, mut kind) = self.expr(operand, sheet, ctx, depth + 1);
385        let mut implied = !nested && self.inserted > before;
386        let shrinkable = self.one_cell_range;
387        self.one_cell_range |= seen;
388        if let Some(operand) = retry.filter(|_| !implied && kind == Kind::Scalar && shrinkable) {
389            let state = (self.remaining, self.certain, self.complete, self.foreign);
390            self.collapsed = true;
391            let (y2, kind2) = self.expr(operand, sheet, ctx, depth + 1);
392            self.collapsed = false;
393            if self.inserted > before {
394                (y, kind, implied) = (y2, kind2, true);
395            } else {
396                (self.remaining, self.certain, self.complete, self.foreign) = state;
397            }
398        }
399        match y {
400            Expr::Unary(UnOp::At, inner) if implied => {
401                self.inserted -= 1;
402                (unwrap_at(*inner), Kind::Scalar)
403            }
404            y if top && kind == Kind::Array => {
405                // The whole-formula `@` the reader puts back over a multi-valued result.
406                self.top = kind;
407                (y, Kind::Scalar)
408            }
409            y => {
410                if kind != Kind::Scalar {
411                    self.foreign = true;
412                }
413                let y = if added { Expr::Paren(Box::new(y)) } else { y };
414                (Expr::Unary(UnOp::At, Box::new(y)), Kind::Scalar)
415            }
416        }
417    }
418
419    fn expr(&mut self, e: Expr, sheet: usize, ctx: Ctx, depth: usize) -> (Expr, Kind) {
420        // Names can be cyclic or nest deeply; their kinds are remembered, so repeated names
421        // cost one analysis each.
422        self.deepest = self.deepest.max(depth);
423        if depth >= MAX_DEPTH || self.remaining == 0 {
424            self.complete = false;
425            return (e, Kind::Array);
426        }
427        self.remaining -= 1;
428        // The flag reaches the union through parentheses and `@`, and not its operands.
429        let union_arg = std::mem::take(&mut self.union_arg);
430        self.union_arg = union_arg && matches!(e, Expr::Paren(_) | Expr::Unary(UnOp::At, _));
431        let (e, kind) = self.node(e, sheet, ctx, depth);
432        self.union_arg = false;
433        if ctx == Ctx::Value && kind == Kind::Ref {
434            if matches!(e, Expr::Binary(BinOp::Union, ..)) && !union_arg {
435                // A union, which only stands in parentheses, as an operand: Excel has no legacy
436                // text for its `@` (it shows `(F25,F26)%`), and the operator gives #VALUE! either
437                // way. A function's argument is intersected, as the function would spill it.
438                return (e, Kind::Scalar);
439            }
440            self.inserted += 1;
441            (at(e), Kind::Scalar)
442        } else {
443            (e, kind)
444        }
445    }
446
447    fn node(&mut self, e: Expr, sheet: usize, ctx: Ctx, depth: usize) -> (Expr, Kind) {
448        let inner = |wanted: Ctx| if ctx == Ctx::Array { Ctx::Array } else { wanted };
449        match e {
450            Expr::Ref(r) => {
451                let one_cell_range = r.range().is_single() && !matches!(r.sheet, SheetSel::Span(_, _)) && matches!(r.kind, RefKind::Range(..));
452                self.one_cell_range |= one_cell_range;
453                let single = r.range().is_single() && !matches!(r.sheet, SheetSel::Span(_, _)) && !(self.collapsed && one_cell_range);
454                (Expr::Ref(r), if single { Kind::Scalar } else { Kind::Ref })
455            }
456            Expr::Array(rows) => {
457                let kind = if rows.len() > 1 || rows.first().is_some_and(|r| r.len() > 1) { Kind::Array } else { Kind::Scalar };
458                (Expr::Array(rows), kind)
459            }
460            Expr::Name(n) => {
461                let kind = self.name(&n, sheet, depth);
462                (Expr::Name(n), kind)
463            }
464            Expr::Struct(r) => {
465                let table = if r.table.is_empty() {
466                    self.position_dependent = true;
467                    self.wb.sheet(sheet).and_then(|s| s.table_at(self.at))
468                } else {
469                    self.wb.table(&r.table).and_then(|(si, ti)| self.wb.sheet(si).and_then(|s| s.tables.get(ti)))
470                };
471                let one_row = r.specifiers.iter().any(|s| matches!(s, StructItem::ThisRow | StructItem::Headers | StructItem::Totals));
472                let one_col = match (&r.col_start, &r.col_end) {
473                    (Some(a), Some(b)) => a.eq_ignore_ascii_case(b),
474                    (Some(_), None) => true,
475                    _ => table.is_some_and(|t| t.columns.len() == 1),
476                };
477                (Expr::Struct(r), if one_row && one_col { Kind::Scalar } else { Kind::Ref })
478            }
479            Expr::Paren(x) => {
480                let (x, kind) = self.expr(*x, sheet, ctx, depth + 1);
481                (Expr::Paren(Box::new(x)), kind)
482            }
483            Expr::Unary(UnOp::At, x) if self.explicit => self.explicit_at(*x, sheet, ctx, depth, false),
484            Expr::Unary(UnOp::At, x) => {
485                let (x, _) = self.expr(*x, sheet, inner(Ctx::Ref), depth + 1);
486                (Expr::Unary(UnOp::At, Box::new(x)), Kind::Scalar)
487            }
488            // A trim reference is a reference operator, like `:`.
489            Expr::Unary(op @ (UnOp::Spill | UnOp::Trim(_)), x) => {
490                let (x, _) = self.expr(*x, sheet, inner(Ctx::Ref), depth + 1);
491                (Expr::Unary(op, Box::new(x)), Kind::Ref)
492            }
493            Expr::Unary(op, x) => {
494                let (x, kind) = self.expr(*x, sheet, inner(Ctx::Value), depth + 1);
495                (Expr::Unary(op, Box::new(x)), if kind == Kind::Scalar { Kind::Scalar } else { Kind::Array })
496            }
497            Expr::Binary(..) => self.binary(e, sheet, ctx, depth),
498            Expr::Call(name, args) => self.call(name, args, sheet, ctx, depth + 1),
499            Expr::Invoke(f, args) => {
500                self.certain = false;
501                (Expr::Invoke(f, args), Kind::Array)
502            }
503            other => (other, Kind::Scalar),
504        }
505    }

507    fn name(&mut self, n: &str, sheet: usize, depth: usize) -> Kind {
508        if let Some((_, kind)) = self.locals.iter().rev().find(|(key, _)| key.eq_ignore_ascii_case(n)) {
509            return *kind;
510        }
511        let (scope, bare) = match n.rsplit_once('!') {
512            Some((s, name)) => (self.wb.sheet_index(&crate::base::unquote_sheet(s)).unwrap_or(sheet), name),
513            None => (sheet, n),
514        };
515        if let Some(def) = self.wb.name(bare, scope) {
516            let key = (bare.to_ascii_uppercase(), scope, self.collapsed);
517            match self.names.get(&key) {
518                Some(Some((kind, one_cell_range))) => {
519                    self.reused = true;
520                    self.one_cell_range |= *one_cell_range;
521                    return *kind;
522                }
523                Some(None) => {
524                    // A name defined in terms of itself has no legacy value to keep.
525                    self.certain = false;
526                    self.reused = true;
527                    return Kind::Array;
528                }
529                None => {}
530            }
531            let reached = depth.saturating_add(1);
532            // Taken only when the formula can afford it and has read none of the names it read:
533            // otherwise the fresh analysis below differs (cut short, or reusing those names).
534            let hit = self.shared.map.get(&key).filter(|i| {
535                reached + i.depth < MAX_DEPTH && i.cost <= self.remaining && i.closure.iter().all(|(k, _, _)| !self.names.contains_key(k))
536            });
537            if let Some(info) = hit {
538                // Charged and recorded as the analysis was: the formula reads as a fresh one does.
539                self.deepest = self.deepest.max(reached + info.depth);
540                self.certain &= info.certain;
541                self.one_cell_range |= info.one_cell_range;
542                self.remaining -= info.cost;
543                for (k, kind, one_cell_range) in &info.closure {
544                    self.names.insert(k.clone(), Some((*kind, *one_cell_range)));
545                }
546                self.read.extend(info.closure.iter().cloned());
547                return info.kind;
548            }
549            let Ok(expr) = crate::formula::parse_name(def.formula.trim_start_matches('=')) else { return Kind::Array };
550            self.shared.analyses += 1;
551            self.names.insert(key.clone(), None);
552            // A name's own formula does not see the formula's LET variables.
553            let locals = std::mem::take(&mut self.locals);
554            let seen = std::mem::replace(&mut self.one_cell_range, false);
555            let before = self.remaining;
556            let first_read = self.read.len();
557            let outer = (
558                std::mem::replace(&mut self.certain, true),
559                std::mem::replace(&mut self.complete, true),
560                std::mem::replace(&mut self.position_dependent, false),
561                std::mem::replace(&mut self.reused, false),
562                std::mem::replace(&mut self.deepest, 0),
563            );
564            let kind = self.kind_of(expr, def.scope.unwrap_or(scope), depth + 1);
565            let one_cell_range = std::mem::replace(&mut self.one_cell_range, seen);
566            self.read.push((key.clone(), kind, one_cell_range));
567            let cacheable = self.complete && !self.position_dependent && !self.reused;
568            let info = cacheable.then(|| NameInfo {
569                kind,
570                one_cell_range,
571                certain: self.certain,
572                cost: before.saturating_sub(self.remaining),
573                depth: self.deepest.saturating_sub(depth + 1),
574                closure: self.read[first_read..].to_vec(),
575            });
576            self.certain &= outer.0;
577            self.complete &= outer.1;
578            self.position_dependent |= outer.2;
579            self.reused |= outer.3;
580            self.deepest = self.deepest.max(outer.4);
581            self.one_cell_range |= one_cell_range;
582            self.locals = locals;
583            if let Some(info) = info.filter(|_| self.shared.map.len() < MAX_CACHED_NAMES) {
584                self.shared.map.insert(key.clone(), info);
585            }
586            self.names.insert(key, Some((kind, one_cell_range)));
587            return kind;
588        }
589        if self.wb.table(bare).is_some() { Kind::Ref } else { Kind::Scalar }
590    }
591
592    fn call(&mut self, name: String, args: Vec<Expr>, sheet: usize, ctx: Ctx, depth: usize) -> (Expr, Kind) {
593        if name == "LET" && args.len() >= 3 {
594            return self.let_(args, sheet, ctx, depth);
595        }
596        let lambda_name = !(self.functions.is_builtin)(&name) && !host_function(&name) && self.wb.name(&name, sheet).is_some();
597        if args.len() > 255
598            || lambda_name
599            || matches!(name.as_str(), "LAMBDA" | "MAP" | "REDUCE" | "SCAN" | "BYROW" | "BYCOL" | "MAKEARRAY" | "ISOMITTED")
600        {
601            self.certain = false;
602            return (Expr::Call(name, args), Kind::Array);
603        }
604        let mut lifted = false;
605        let mut kinds = Vec::with_capacity(args.len());
606        let mut out = Vec::with_capacity(args.len());
607        for (i, arg) in args.into_iter().enumerate() {
608            let param = param(&name, i, self.functions);
609            let arg_ctx = match param {
610                Param::Value | Param::Single => Ctx::Value,
611                Param::Ref | Param::First => Ctx::Ref,
612                Param::Pass => ctx,
613                Param::Array => Ctx::Array,
614                Param::Uncertain => {
615                    self.certain = false;
616                    Ctx::Array
617                }
618            };
619            let arg_ctx = if ctx == Ctx::Array && (param != Param::Single || self.discarded) { Ctx::Array } else { arg_ctx };
620            // A union argument, under the file's parentheses and any `@` the reader put on it.
621            let mut operand = &arg;
622            while let Expr::Paren(x) | Expr::Unary(UnOp::At, x) = operand {
623                operand = x;
624            }
625            self.union_arg = arg_ctx == Ctx::Value && matches!(operand, Expr::Binary(BinOp::Union, ..));
626            let (arg, kind) = self.expr(arg, sheet, arg_ctx, depth);
627            // A value parameter given several values evaluates the function once per value.
628            lifted |= match param {
629                Param::Value | Param::Single => kind == Kind::Array || (kind == Kind::Ref && arg_ctx == Ctx::Array),
630                Param::First => kind != Kind::Scalar,
631                _ => false,
632            };
633            kinds.push(kind);
634            out.push(arg);
635        }
636        let first = kinds.first().copied().unwrap_or(Kind::Scalar);
637        let kind = match name.as_str() {
638            "IF" | "CHOOSE" if !lifted => kinds.iter().skip(1).fold(Kind::Scalar, |k, x| k.join(*x)),
639            "ROW" | "COLUMN" => {
640                if first == Kind::Scalar {
641                    Kind::Scalar
642                } else if ctx == Ctx::Array {
643                    Kind::Array
644                } else {
645                    // Outside array evaluation, ROW and COLUMN of a range give its first row or column.
646                    self.inserted += 1;
647                    return (at(Expr::Call(name, out)), Kind::Scalar);
648                }
649            }
650            "INDIRECT" => match out.first() {
651                Some(Expr::Text(text)) if !matches!(out.get(1), Some(Expr::Bool(false))) => match crate::formula::parse(text) {
652                    Ok(e) => {
653                        let kind = self.kind_of(e, sheet, depth);
654                        if kind == Kind::Scalar { Kind::Scalar } else { Kind::Ref }
655                    }
656                    Err(_) => Kind::Ref,
657                },
658                _ => Kind::Ref,
659            },
660            "OFFSET" => offset_kind(&out, first, lifted),
661            "INDEX" => index_kind(&out, &kinds, first, lifted),
662            "XLOOKUP" if !lifted => match (out.get(1), out.get(2)) {
663                (Some(Expr::Ref(lookup)), Some(Expr::Ref(result))) => {
664                    let multi = if lookup.range().width() == 1 { result.range().width() > 1 } else { result.range().height() > 1 };
665                    if multi { Kind::Ref } else { Kind::Scalar }
666                }
667                _ => Kind::Ref,
668            },
669            "ANCHORARRAY" | "TRIMRANGE" => Kind::Ref,
670            "FILTER" | "SORT" | "SORTBY" | "UNIQUE" | "SEQUENCE" | "RANDARRAY" | "VSTACK" | "HSTACK" | "TOROW" | "TOCOL" | "WRAPROWS"
671            | "WRAPCOLS" | "TAKE" | "DROP" | "CHOOSEROWS" | "CHOOSECOLS" | "EXPAND" | "GROUPBY" | "PIVOTBY" | "TRANSPOSE" | "MMULT" | "MINVERSE"
672            | "MUNIT" | "FREQUENCY" | "LINEST" | "LOGEST" | "TREND" | "GROWTH" | "MODE.MULT" | "TEXTSPLIT" | "FILTERXML" => Kind::Array,
673            // Every match (mode 1) or the capture groups (mode 2); the first match by default.
674            "REGEXEXTRACT" => match out.get(2) {
675                None | Some(Expr::Missing) => Kind::Scalar,
676                Some(Expr::Number(n)) if *n == 0.0 => Kind::Scalar,
677                _ => Kind::Array,
678            },
679            // Google Sheets' export placeholder fails with #NAME? so an outer IFERROR keeps the cached scalar.
680            "__XLUDF.DUMMYFUNCTION" => Kind::Scalar,
681            // A user-defined function may return a range or an array.
682            n if user_function(n, self.functions) => Kind::Array,
683            _ if lifted => Kind::Array,
684            _ => Kind::Scalar,
685        };
686        (Expr::Call(name, out), kind)
687    }
688}
689
690fn offset_kind(out: &[Expr], first: Kind, lifted: bool) -> Kind {
691    // Omitted dimensions keep the source reference's shape; unknown dimensions might
692    // become several cells on a later calculation.
693    let first_multi = first != Kind::Scalar;
694    let multi = if out.len() <= 3 { first_multi } else { !out.get(3).is_some_and(one) || out.get(4).map_or(first_multi, |e| !one(e)) };
695    if lifted {
696        Kind::Array
697    } else if multi {
698        Kind::Ref
699    } else {
700        Kind::Scalar
701    }
702}
703
704fn index_kind(out: &[Expr], kinds: &[Kind], first: Kind, lifted: bool) -> Kind {
705    let row = out.get(1);
706    let col = out.get(2);
707    let row_positive = row.is_some_and(|e| positive_index(e, kinds.get(1).copied()));
708    let col_positive = col.is_some_and(|e| positive_index(e, kinds.get(2).copied()));
709    let multi = !row_positive || col.is_some_and(|_| !col_positive) || (col.is_none() && !out.first().is_some_and(one_line));
710    if multi {
711        if first == Kind::Array { Kind::Array } else { Kind::Ref }
712    } else if lifted {
713        Kind::Array
714    } else {
715        Kind::Scalar
716    }
717}

719impl Legacy<'_> {
720    /// A binary operation. `A1+A2+…+A70` parses as a left-nested chain as deep as it is long, so
721    /// the chain's left operands are walked iteratively, at the same depth; each operation still
722    /// costs one node of the budget.
723    fn binary(&mut self, e: Expr, sheet: usize, ctx: Ctx, depth: usize) -> (Expr, Kind) {
724        let operand = |op: BinOp, node: Ctx| {
725            if node == Ctx::Array {
726                Ctx::Array
727            } else if matches!(op, BinOp::Range | BinOp::Union | BinOp::Intersect) {
728                Ctx::Ref
729            } else {
730                Ctx::Value
731            }
732        };
733        // (operator, right operand, the operation's context), outermost first.
734        let mut spine = Vec::new();
735        let mut cur = e;
736        let mut cur_ctx = ctx;
737        while let Expr::Binary(op, a, b) = cur {
738            if !spine.is_empty() {
739                // The caller's `expr` counted the outermost operation.
740                if self.remaining == 0 {
741                    self.complete = false;
742                } else {
743                    self.remaining -= 1;
744                }
745            }
746            spine.push((op, *b, cur_ctx));
747            cur_ctx = operand(op, cur_ctx);
748            cur = *a;
749        }
750        let (mut acc, mut kind) = self.expr(cur, sheet, cur_ctx, depth);
751        for (i, (op, b, node_ctx)) in spine.into_iter().enumerate().rev() {
752            let (b, kb) = self.expr(b, sheet, operand(op, node_ctx), depth);
753            let node = Expr::Binary(op, Box::new(acc), Box::new(b));
754            let k = if matches!(op, BinOp::Range | BinOp::Union | BinOp::Intersect) {
755                Kind::Ref
756            } else if kind == Kind::Scalar && kb == Kind::Scalar {
757                Kind::Scalar
758            } else {
759                Kind::Array
760            };
761            // An inner operation is an operand: `expr` would intersect it there.
762            (acc, kind) = if i > 0 && node_ctx == Ctx::Value && k == Kind::Ref {
763                self.inserted += 1;
764                (at(node), Kind::Scalar)
765            } else {
766                (node, k)
767            };
768        }
769        (acc, kind)
770    }
771
772    fn let_(&mut self, args: Vec<Expr>, sheet: usize, ctx: Ctx, depth: usize) -> (Expr, Kind) {
773        let saved = self.locals.len();
774        let count = args.len();
775        let mut out = Vec::with_capacity(count);
776        let mut pending: Option<String> = None;
777        let mut kind = Kind::Scalar;
778        for (i, arg) in args.into_iter().enumerate() {
779            if i + 1 == count {
780                let (body, k) = self.expr(arg, sheet, ctx, depth);
781                out.push(body);
782                kind = k;
783            } else if i % 2 == 0 {
784                pending = match &arg {
785                    Expr::Name(n) => Some(n.clone()),
786                    _ => None,
787                };
788                out.push(arg);
789            } else {
790                let value_ctx = if ctx == Ctx::Array { Ctx::Array } else { Ctx::Ref };
791                let (value, k) = self.expr(arg, sheet, value_ctx, depth);
792                if let Some(n) = pending.take() {
793                    self.locals.push((n, k));
794                }
795                out.push(value);
796            }
797        }
798        self.locals.truncate(saved);
799        (Expr::Call("LET".into(), out), kind)
800    }
801}

803fn positive(e: &Expr) -> bool {
804    matches!(e, Expr::Number(n) if *n >= 1.0)
805}
806fn positive_index(e: &Expr, kind: Option<Kind>) -> bool {
807    // MATCH returns a one-based position or an error, never INDEX's zero whole-row selector.
808    positive(e) || (matches!(e, Expr::Call(name, _) if matches!(name.as_str(), "MATCH" | "XMATCH")) && kind.is_some_and(|k| k != Kind::Ref))
809}

811fn one(e: &Expr) -> bool {
812    matches!(e, Expr::Number(n) if *n == 1.0)
813}

815/// A one-column or one-row reference: INDEX with a single index picks one cell of it.

817    matches!(e, Expr::Ref(r) if r.range().width() == 1 || r.range().height() == 1)
818}

867fn param(name: &str, i: usize, functions: FunctionInfo) -> Param {
868    match name {
869        "IF" if i == 0 => Param::Single,
870        "IFERROR" | "IFNA" => Param::Single,
871        "CHOOSE" if i == 0 => Param::Value,
872        "IF" | "CHOOSE" => Param::Pass,
873        "SUMPRODUCT" | "MMULT" | "MINVERSE" | "MDETERM" | "TRANSPOSE" | "FREQUENCY" | "LINEST" | "LOGEST" | "TREND" | "GROWTH" => Param::Array,
874        "INDEX" if i == 0 => Param::Array,
875        // LOOKUP's vectors and AGGREGATE's array form evaluate their arguments as arrays.
876        "LOOKUP" if i > 0 => Param::Array,
877        "AGGREGATE" if i == 2 => Param::Array,
878        "ROW" | "COLUMN" | "ROWS" | "COLUMNS" | "AREAS" | "ISREF" | "ISFORMULA" | "FORMULATEXT" | "SHEET" | "SHEETS" | "TRIMRANGE"
879        | "ANCHORARRAY" => Param::Ref,
880        "CELL" | "GETPIVOTDATA" if i == 1 => Param::Ref,
881        "T" => Param::First,
882        "N" if i == 0 => Param::Value,
883        "FILTER" | "SORT" | "SORTBY" | "UNIQUE" | "VSTACK" | "HSTACK" | "TOROW" | "TOCOL" | "WRAPROWS" | "WRAPCOLS" | "TAKE" | "DROP"
884        | "CHOOSEROWS" | "CHOOSECOLS" | "EXPAND" | "GROUPBY" | "PIVOTBY"
885            if !scalar_arg(name, i) =>
886        {
887            Param::Uncertain
888        }
889        n if user_function(n, functions) => Param::Ref,
890        n if scalar_arg(n, i) && (host_function(n) || (functions.is_scalar_arg)(n, i)) => Param::Value,
891        _ => Param::Ref,
892    }
893}

895/// Legacy reference/array-taking parameters. Other parameters consume a single value; the
896/// function library's lifting metadata refines this for the functions it implements.
897fn scalar_arg(name: &str, i: usize) -> bool {
898    match name {
899        "SUM" | "AVERAGE" | "AVERAGEA" | "MIN" | "MINA" | "MAX" | "MAXA" | "COUNT" | "COUNTA" | "COUNTBLANK" | "PRODUCT" | "SUMSQ" | "MEDIAN"
900        | "MODE" | "MODE.SNGL" | "STDEV" | "STDEV.S" | "STDEV.P" | "STDEVP" | "VAR" | "VAR.S" | "VAR.P" | "VARP" | "AVEDEV" | "DEVSQ" | "GEOMEAN"
901        | "HARMEAN" | "KURT" | "SKEW" | "SKEW.P" | "AND" | "OR" | "XOR" | "SUMPRODUCT" | "SUMX2MY2" | "SUMX2PY2" | "SUMXMY2" | "CORREL" | "COVAR"
902        | "COVARIANCE.P" | "COVARIANCE.S" | "PEARSON" | "RSQ" | "SLOPE" | "INTERCEPT" | "STEYX" | "ROWS" | "COLUMNS" | "AREAS" | "ROW" | "COLUMN"
903        | "TRANSPOSE" | "ISREF" | "MMULT" | "MINVERSE" | "MDETERM" | "FREQUENCY" | "CONCAT" | "VSTACK" | "HSTACK" | "ANCHORARRAY" => false,
904        "SUMIF" | "AVERAGEIF" => i == 1,
905        "COUNTIF" => i != 0,
906        "SUMIFS" | "AVERAGEIFS" | "MINIFS" | "MAXIFS" => i > 0 && i.is_multiple_of(2),
907        "COUNTIFS" => i % 2 == 1,
908        "INDEX" | "LARGE" | "SMALL" | "PERCENTILE" | "PERCENTILE.INC" | "PERCENTILE.EXC" | "QUARTILE" | "QUARTILE.INC" | "QUARTILE.EXC" | "IRR"
909        | "MIRR" | "RANK" | "RANK.EQ" | "RANK.AVG" => i != if name.starts_with("RANK") { 1 } else { 0 },
910        "VLOOKUP" | "HLOOKUP" | "MATCH" | "XMATCH" => i != 1,
911        "LOOKUP" | "NPV" | "XNPV" | "SUBTOTAL" => i == 0,
912        "XLOOKUP" => i != 1 && i != 2,
913        "XIRR" => i > 1,
914        "OFFSET" => i != 0,
915        "AGGREGATE" | "TEXTJOIN" => i < 2,
916        "FILTER" | "SORTBY" => i > 1,
917        "SORT" | "UNIQUE" | "TAKE" | "DROP" | "CHOOSECOLS" | "CHOOSEROWS" | "TOROW" | "TOCOL" | "WRAPROWS" | "WRAPCOLS" | "EXPAND" => i != 0,
918        "LINEST" | "LOGEST" | "TREND" | "GROWTH" => i > 1,
919        "FORECAST" | "FORECAST.LINEAR" => i == 0,
920        "FORECAST.ETS" | "FORECAST.ETS.CONFINT" => i != 1 && i != 2,
921        "FORECAST.ETS.SEASONALITY" | "FORECAST.ETS.STAT" => i > 1,
922        _ => true,
923    }
