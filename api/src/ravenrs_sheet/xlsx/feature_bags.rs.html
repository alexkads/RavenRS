1//! Feature property bags ([MS-XLSX] 2.1.22, 2.3.9): how Excel stores cell checkboxes.
2//!
3//! A checkbox is a cell format (`xf`) whose `extLst` carries an `xfComplement i="n"` element
4//! ([MS-XLSX] 2.4.99, styles extension `{C7286773-470A-42A8-94C5-96B5CB345126}`). `n` indexes the
5//! `MappedFeaturePropertyBags` array of the `XFComplements` bag; that entry is an `XFComplement`
6//! bag whose `XFControls` key names an `XFControls` bag, whose `CellControl` key names a
7//! `Checkbox` bag. Bag references are zero-based indexes into the part's list of `bag` elements.
8//!
9//! GridCraft keeps a checkbox as the `checkbox` number format in its model, so reading maps each
10//! `xfComplement` that resolves to a `Checkbox` bag onto that format, and writing emits one chain
11//! that every checkbox `xf` points at.
12
13use crate::xlsx::xml::El;
14
15/// Namespace of the part and of the `xfComplement` element.
16pub const NS: &str = "http://schemas.microsoft.com/office/spreadsheetml/2022/featurepropertybag";
17/// Relationship type from the workbook part.
18pub const REL: &str = "http://schemas.microsoft.com/office/2022/11/relationships/FeaturePropertyBag";
19/// Content type of the part.
20pub const CONTENT_TYPE: &str = "application/vnd.ms-excel.featurepropertybag+xml";
21/// Part name we write.
22pub const PART: &str = "xl/featurePropertyBag/featurePropertyBag.xml";
23/// `ext` URI of the `xfComplement` extension on an `xf`.
24pub const XF_EXT_URI: &str = "{C7286773-470A-42A8-94C5-96B5CB345126}";
25
26/// The bags for one checkbox cell control, referenced as `xfComplement i="0"`: Checkbox (0),
27/// XFControls → 0 (1), XFComplement → 1 (2), XFComplements mapping index 0 → 2 (3). Every
28/// reference points to an earlier bag, as [MS-XLSX] 2.6.230 requires. The XFComplements bag is
29/// the one `xfComplement` indexes into, and Excel finds it by its `extRef`: without
30/// `extRef="XFComplementsMapperExtRef"` it repairs the workbook ("Format from /xl/styles.xml")
31/// and drops every checkbox (#2630).
32pub fn checkbox_part() -> String {
33    format!(
34        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n<FeaturePropertyBags xmlns=\"{NS}\" count=\"4\"><bag type=\"Checkbox\"/><bag type=\"XFControls\"><bagId k=\"CellControl\">0</bagId></bag><bag type=\"XFComplement\"><bagId k=\"XFControls\">1</bagId></bag><bag type=\"XFComplements\" extRef=\"XFComplementsMapperExtRef\"><a k=\"MappedFeaturePropertyBags\"><bagId>2</bagId></a></bag></FeaturePropertyBags>"
35    )
36}
37
38/// The `extLst` that marks an `xf` as a checkbox (complement index 0 of [`checkbox_part`]).
39pub fn checkbox_xf_ext() -> String {
40    format!("<extLst><ext uri=\"{XF_EXT_URI}\" xmlns:xfpb=\"{NS}\"><xfpb:xfComplement i=\"0\"/></ext></extLst>")
41}

43/// For each `xfComplement` index, whether it resolves to a Checkbox cell control. Unknown,
44/// malformed or out-of-range references simply resolve to `false`.
45pub fn checkbox_complements(root: &El) -> Vec<bool> {
46    let bags: Vec<&El> = root.kids("bag").collect();
47    let bag = |i: Option<u32>, ty: &str| i.and_then(|i| bags.get(i as usize).copied()).filter(|b| b.attr("type") == Some(ty));
48    let key = |b: &El, k: &str| b.kids("bagId").find(|e| e.attr("k") == Some(k)).and_then(|e| e.text.trim().parse::<u32>().ok());
49    let Some(map) = bags.iter().find(|b| b.attr("type") == Some("XFComplements")) else { return Vec::new() };
50    let Some(arr) = map.kids("a").find(|a| a.attr("k") == Some("MappedFeaturePropertyBags")) else { return Vec::new() };
51    arr.kids("bagId")
52        .take(100_000)
53        .map(|e| {
54            let complement = bag(e.text.trim().parse::<u32>().ok(), "XFComplement");
55            let controls = complement.and_then(|c| bag(key(c, "XFControls"), "XFControls"));
56            controls.and_then(|c| bag(key(c, "CellControl"), "Checkbox")).is_some()
57        })
58        .collect()
59}

61/// Whether an `xf` element carries an `xfComplement` that resolves to a checkbox.
62pub fn xf_is_checkbox(xf: &El, complements: &[bool]) -> bool {
63    let Some(ext) = xf.child("extLst") else { return false };
64    ext.kids("ext")
65        .filter(|e| e.attr("uri").is_some_and(|u| u.eq_ignore_ascii_case(XF_EXT_URI)))
66        .filter_map(|e| e.child("xfComplement"))
67        .filter_map(|c| c.attr_u32("i"))
68        .any(|i| complements.get(i as usize).copied().unwrap_or(false))
69}

71#[cfg(test)]
72mod tests {
73    use super::*;
74
75    #[test]
76    fn written_chain_resolves_and_hostile_chains_do_not() {
77        let root = crate::xlsx::xml::parse(checkbox_part().as_bytes()).unwrap_or_default();
78        assert_eq!(checkbox_complements(&root), vec![true]);
79        // #2630: Excel repairs the file unless the mapping bag carries this extRef.
80        let mapper = root.kids("bag").find(|b| b.attr("type") == Some("XFComplements"));
81        assert_eq!(mapper.and_then(|b| b.attr("extRef")), Some("XFComplementsMapperExtRef"));
82        // Out-of-range, wrong-type and self references resolve to false without panicking.
83        let bad = format!(
84            "<FeaturePropertyBags xmlns=\"{NS}\"><bag type=\"XFControls\"><bagId k=\"CellControl\">99</bagId></bag><bag type=\"XFComplement\"><bagId k=\"XFControls\">1</bagId></bag><bag type=\"XFComplements\"><a k=\"MappedFeaturePropertyBags\"><bagId>1</bagId><bagId>4294967295</bagId><bagId>x</bagId></a></bag></FeaturePropertyBags>"
85        );
86        let root = crate::xlsx::xml::parse(bad.as_bytes()).unwrap_or_default();
87        assert_eq!(checkbox_complements(&root), vec![false, false, false]);
88    }
