1//! Legacy Excel 97-2003 `.xls` (BIFF8) import. `craft-xls` reads the file (the compound file,
2//! the Workbook stream, cells, formulas as text, formats, layout and names); this module maps
3//! its workbook onto GridCraft's model: formulas parsed, Excel's implicit `@` for legacy
4//! formulas, widths and heights in pixels, formats interned as styles, sheet-name rules, the
5//! sheet the workbook opens on, print areas and titles, and the warnings worded for GridCraft.
6//!
7//! Every formula in the file is a legacy (pre-dynamic-array) one, so it is read with Excel's
8//! implicit `@` (see `array_formula.rs`), like an `.xlsx` formula saved without `cm`.
9

38/// Reads an Excel 97-2003 workbook, with the default [`FunctionInfo`]. Inspect the report's
39/// warnings for what was skipped.
40pub fn read_xls(bytes: &[u8]) -> Result<(Workbook, ReadReport), IoError> {
41    read_xls_budgeted(bytes, x::Limits::default(), FunctionInfo::default())
42}

50fn error(e: x::Error) -> IoError {
51    match e {
52        x::Error::Encrypted => invalid(ENCRYPTED),
53        x::Error::Biff5 => invalid("Excel 5.0/95 workbooks (BIFF5) are not supported; save the file as .xlsx in Excel and try again."),
54        x::Error::TooLarge(why) => IoError::TooLarge(format!("XLS: {why}")),
55        x::Error::Invalid(why) => invalid(why),
56        e => invalid(e.to_string()),
57    }
58}

60/// GridCraft's wording of what the read skipped.
61fn warning(w: &x::Warning) -> String {
62    use x::Warning as W;
63    match w {
64        W::ShortFormatting => "Some cell formatting records were shorter than expected; the formatting present was kept.".into(),
65        W::ShortNumberFormat => "Some number formats were shorter than their declared length; the characters present were kept.".into(),
66        W::ChartSheet => "Chart sheets in .xls files are not imported yet and were skipped.".into(),
67        W::MacroSheet => "Excel 4.0 macro sheets were skipped.".into(),
68        W::OtherSheet => "VBA modules and other non-worksheet sheets were skipped.".into(),
69        W::DialogSheet => "Dialog sheets and unreadable sheets were skipped.".into(),
70        W::MissingSharedString => "Some cells refer to missing shared strings and were left empty.".into(),
71        W::DataTable => "What-if data tables are not supported; their values were kept.".into(),
72        W::DamagedObjects => "Some comments or drawing objects in .xls files could not be read and were skipped.".into(),
73        W::Charts => "Charts in .xls files are not imported yet.".into(),
74        W::PicturesSkipped => "Some pictures in .xls files were skipped.".into(),
75        W::ConditionalFormattingSkipped | W::ConditionalFormatting12 => "Some conditional formatting in .xls files was not imported.".into(),
76        W::DataValidationSkipped => "Some data validation rules in .xls files could not be read and were skipped.".into(),
77        W::DataValidationIgnored => "Some data validation records in .xls files were ignored, as Excel ignores them: they have no DVal record before them, or come after the number of rules it gives.".into(),
78        W::Hyperlinks => "Hyperlinks in .xls files are not imported yet.".into(),
79        W::PivotTables => "PivotTables in .xls files are not imported yet; their values were kept.".into(),
80        W::CellsOutsideGrid(n) => format!("{n} cells outside the grid were skipped."),
81        W::FormulaKept(why) => format!("Some formulas use {why}; their saved results were kept as values."),
82        W::NameSkipped(why) => format!("Some defined names use {why} and were skipped."),
83        W::MacroNames => "Macro names were skipped.".into(),
84        // craft-xls may add warnings; as a module of this crate it cannot yet, so the arm stays
85        // for the next sync.
86        #[allow(unreachable_patterns)]
87        w => format!("{w}."),
88    }
89}

91/// [`read_xls_with`] with explicit limits.
92pub(crate) fn read_xls_budgeted(bytes: &[u8], limits: x::Limits, functions: FunctionInfo) -> Result<(Workbook, ReadReport), IoError> {
93    let read = x::read_with(bytes, limits).map_err(error)?;
94    let book = read.workbook;
95    let mut report = ReadReport::default();
96    for w in &read.warnings {
97        report.warn(warning(w));
98    }
99
100    let mut wb = Workbook::new();
101    wb.sheets.clear();
102    wb.date_system = if book.date_1904 { DateSystem::D1904 } else { DateSystem::D1900 };
103    // One model style per XF, in file order, as the cells refer to them.
104    if let Some(font) = &book.normal_font {
105        wb.styles = StyleTable::with_default(Style { font: font_of(font), ..Style::default() });
106    }
107    let styles: Vec<StyleId> = book.formats.iter().map(|f| wb.styles.intern(style_of(f))).collect();
108    wb.cell_styles = book.cell_styles.iter().map(|(name, f)| (name.clone(), style_of(f))).collect();
109
110    let mut names: Vec<&str> = Vec::new();
111    for s in &book.sheets {
112        wb.check_sheet_name(&s.name, None).map_err(invalid)?;
113        if names.iter().any(|n| n.eq_ignore_ascii_case(&s.name)) {
114            return Err(invalid("That name is already taken. Try a different one."));
115        }
116        names.push(&s.name);
117    }
118    let mut unparsed = false;
119    for s in &book.sheets {
120        // Report rules craft-xls kept that the adapter cannot represent. craft-xls already warns
121        // about records it could not decode.
122        for (dropped, skipped) in [
123            (!s.conditional_formats.is_empty(), x::Warning::ConditionalFormattingSkipped),
124            (s.validations.iter().any(|v| validation_of(v).is_none()), x::Warning::DataValidationSkipped),
125        ] {
126            if dropped {
127                report.warn(warning(&skipped));
128            }
129        }
130        let sheet = sheet_of(s, &styles, &mut unparsed);
131        wb.sheets.push(Arc::new(sheet));
132    }
133    if unparsed {
134        report.warn("Some formulas use syntax GridCraft can't parse yet; their saved results were kept as values.");
135    }
136    wb.calc = calc_of(&book.calc);
137    wb.calc.precision_as_displayed = book.precision_as_displayed;
138    for n in &book.names {
139        wb.names.push(DefinedName { name: n.name.clone(), scope: n.scope, formula: n.formula.clone(), comment: String::new(), hidden: n.hidden });
140    }
141    // Every .xls formula is a legacy (pre-dynamic-array) one: read with Excel's implicit `@`, like
142    // an .xlsx formula without `cm`. Ctrl+Shift+Enter arrays are read as they are.
143    let legacy = wb.sheets.iter().enumerate().map(|(si, s)| (si, legacy_cells(s))).collect();
144    crate::xlsx::array_formula::apply_legacy(&mut wb, legacy, functions);
145    // The sheet the file was saved on, as Excel opens it, when it was read and is visible; else the
146    // first visible sheet.
147    let saved = book.active_sheet.filter(|&i| wb.sheets.get(i).is_some_and(|s| s.visibility == Visibility::Visible));
148    wb.set_active_sheet(saved.or_else(|| wb.sheets.iter().position(|s| s.visibility == Visibility::Visible)).unwrap_or(0));
149    // The overview comes first.
150    report.warnings.insert(
151        0,
152        "Imported from an Excel 97-2003 (.xls) workbook: values, formulas, cell formatting, merged cells, column widths, row heights, defined names and data validation were read. Charts, pictures, comments and other features are not imported yet. Save as .xlsx; saving .xls is not supported.".into(),
153    );
154    Ok((wb, report))
155}

157/// The cells holding a formula that is not a Ctrl+Shift+Enter array.
158fn legacy_cells(sheet: &Sheet) -> Vec<CellRef> {
159    sheet.cells.iter().filter(|(_, c)| c.formula.as_ref().is_some_and(|f| f.array.is_none())).map(|(at, _)| at).collect()
160}

171    match v {
172        x::Value::Number(n) => Value::number(*n),
173        x::Value::Text(s) => Value::Text(s.clone()),
174        x::Value::Bool(b) => Value::Bool(*b),

180fn calc_of(c: &x::Calc) -> CalcSettings {
181    let mut calc = CalcSettings::default();
182    if let Some(mode) = c.mode {
183        calc.mode = match mode {
184            x::CalcMode::Manual => CalcMode::Manual,
185            x::CalcMode::AutomaticExceptTables => CalcMode::AutomaticExceptTables,
186            _ => CalcMode::Automatic,
187        };
188    }
189    if let Some(n) = c.max_iterations {
190        calc.set_max_iterations(u32::from(n));
191    }
192    if let Some(on) = c.iterative {
193        calc.iterative = on;
194    }
195    if let Some(d) = c.max_change {
196        calc.set_max_change(d);
197    }
198    calc
199}

201/// One worksheet. `unparsed` is set when a formula's text is one GridCraft can't parse: the cell
202/// keeps its saved result.
203fn sheet_of(s: &x::Sheet, styles: &[StyleId], unparsed: &mut bool) -> Sheet {
204    let style = |f: Option<u16>| f.and_then(|i| styles.get(usize::from(i)).copied()).unwrap_or(StyleId::DEFAULT);
205    let mut sheet = Sheet::new(&s.name);
206    sheet.visibility = match s.visibility {
207        x::Visibility::Hidden => Visibility::Hidden,
208        x::Visibility::VeryHidden => Visibility::VeryHidden,
209        _ => Visibility::Visible,
210    };
211    for (at, c) in &s.cells {
212        let formula = c.formula.as_ref().and_then(|f| {
213            let mut parsed = Formula::new(&f.text);
214            if parsed.expr.is_none() {
215                *unparsed = true;
216                return None;
217            }
218            // A one-cell Ctrl+Shift+Enter array keeps its range, as when an .xlsx is read: it
219            // evaluates as an array, not with implicit intersection.
220            parsed.array = f.array.map(range);
221            Some(Arc::new(parsed))
222        });
223        sheet.cells.set(cell_ref(*at), Cell { value: value(&c.value), formula, style: style(c.format), py: None });
224    }
225    sheet.merges = s.merges.iter().map(|r| range(*r)).collect();
226    if let Some(h) = s.default_row_height {
227        sheet.default_row_height = pt_to_px(f64::from(h) / 20.0);
228    }
229    for (&col, c) in &s.columns {
230        let px = col_width_to_px(f64::from(c.width) / 256.0);
231        let info = LineInfo {
232            size: (c.custom_width || px != sheet.default_col_width).then_some(px),
233            hidden: c.hidden,
234            outline: c.outline,
235            collapsed: c.collapsed,
236            style: c.format.map(|f| style(Some(f))).filter(|&id| id != StyleId::DEFAULT),
237            ..LineInfo::default()
238        };
239        if info != LineInfo::default() {
240            sheet.cols.insert(u32::from(col), info);
241        }
242    }
243    for (&r, row) in &s.rows {
244        let px = pt_to_px(f64::from(row.height) / 20.0);
245        let info = LineInfo {
246            custom: row.custom_height,
247            hidden: row.hidden,
248            size: (row.custom_height || (px - sheet.default_row_height).abs() > 0.5).then_some(px),
249            outline: row.outline,
250            collapsed: row.collapsed,
251            style: row.format.map(|f| style(Some(f))).filter(|&id| id != StyleId::DEFAULT),
252        };
253        if info != LineInfo::default() {
254            sheet.rows.insert(r, info);
255        }
256    }
257    if let Some(v) = s.view {
258        sheet.show_formulas = v.show_formulas;
259        sheet.show_gridlines = v.show_gridlines;
260        sheet.show_headings = v.show_headings;
261        sheet.show_zeros = v.show_zeros;
262        sheet.right_to_left = v.right_to_left;
263    }
264    sheet.validations = s.validations.iter().filter_map(validation_of).collect();
265    if let Some(text) = &s.print_area {
266        sheet.print.print_area = crate::xlsx::read::parse_print_area(text);
267    }
268    if let Some(text) = &s.print_titles {
269        let (rows, cols) = crate::xlsx::read::parse_print_titles(text);
270        sheet.print.title_rows = rows;
271        sheet.print.title_cols = cols;
272    }
273    sheet
274}

276/// A data validation rule, as `sheet_read` maps a SpreadsheetML `dataValidation`: a list that
277/// is not a literal refers to cells (`=…`).
278fn validation_of(v: &x::DataValidation) -> Option<Validation> {
279    let ranges: Vec<RangeRef> = v.ranges.iter().map(|r| range(*r)).collect();
280    if ranges.is_empty() {
281        return None;
282    }
283    let kind = match v.kind {
284        x::ValidationType::Whole => ValidationKind::Whole,
285        x::ValidationType::Decimal => ValidationKind::Decimal,
286        x::ValidationType::List => ValidationKind::List,
287        x::ValidationType::Date => ValidationKind::Date,
288        x::ValidationType::Time => ValidationKind::Time,
289        x::ValidationType::TextLength => ValidationKind::TextLength,
290        x::ValidationType::Custom => ValidationKind::Custom,
291        _ => ValidationKind::Any,
292    };
293    let f1 = if kind == ValidationKind::List && !v.formula1.is_empty() && !v.formula1.starts_with('"') {
294        format!("={}", v.formula1)
295    } else {
296        v.formula1.clone()
297    };
298    Some(Validation {
299        ranges,
300        kind,
301        op: match v.operator {
302            x::Operator::NotBetween => CfOperator::NotBetween,
303            x::Operator::Equal => CfOperator::Equal,
304            x::Operator::NotEqual => CfOperator::NotEqual,
305            x::Operator::Greater => CfOperator::Greater,
306            x::Operator::Less => CfOperator::Less,
307            x::Operator::GreaterOrEqual => CfOperator::GreaterOrEqual,
308            x::Operator::LessOrEqual => CfOperator::LessOrEqual,
309            _ => CfOperator::Between,
310        },
311        f1,
312        f2: v.formula2.clone(),
313        allow_blank: v.allow_blank,
314        in_cell_dropdown: v.in_cell_dropdown,
315        input_title: v.prompt_title.clone(),
316        input_message: v.prompt.clone(),
317        show_input: v.show_prompt,
318        error_title: v.error_title.clone(),
319        error_message: v.error.clone(),
320        error_style: match v.error_style {
321            x::ErrorStyle::Warning => ErrorStyle::Warning,
322            x::ErrorStyle::Information => ErrorStyle::Information,
323            _ => ErrorStyle::Stop,
324        },
325        show_error: v.show_error,
326    })
327}

336fn font_of(f: &x::Font) -> Font {
337    let mut font = Font::default();
338    if let Some(name) = &f.name {
339        font.name = name.clone();
340    }
341    if let Some(size) = f.size {
342        font.size = size;
343    }
344    font.bold = f.bold;
345    font.italic = f.italic;
346    font.strike = f.strike;
347    font.underline = match f.underline {
348        x::Underline::Single => Underline::Single,
349        x::Underline::Double => Underline::Double,
350        x::Underline::SingleAccounting => Underline::SingleAccounting,
351        x::Underline::DoubleAccounting => Underline::DoubleAccounting,
352        _ => Underline::None,
353    };
354    font.vert = match f.script {
355        x::Script::Superscript => VertAlign::Superscript,
356        x::Script::Subscript => VertAlign::Subscript,
357        _ => VertAlign::Baseline,
358    };
359    font.color = color(f.color);
360    font.family = f.family;
361    font.charset = f.charset;
362    font
363}

418/// The model style of one XF.
419fn style_of(f: &x::CellFormat) -> Style {
420    let a = f.align;
421    let b = f.borders;
422    Style {
423        font: font_of(&f.font),
424        num_fmt: if f.number_format.eq_ignore_ascii_case("General") { NumFmt::default() } else { NumFmt::new(&f.number_format) },
425        protection: Protection { locked: f.locked, hidden: f.hidden },
426        align: Alignment {
427            h: match a.h {
428                x::HAlign::Left => HAlign::Left,
429                x::HAlign::Center => HAlign::Center,
430                x::HAlign::Right => HAlign::Right,
431                x::HAlign::Fill => HAlign::Fill,
432                x::HAlign::Justify => HAlign::Justify,
433                x::HAlign::CenterAcross => HAlign::CenterAcross,
434                x::HAlign::Distributed => HAlign::Distributed,
435                _ => HAlign::General,
436            },
437            v: match a.v {
438                x::VAlign::Top => VAlign::Top,
439                x::VAlign::Center => VAlign::Center,
440                x::VAlign::Justify => VAlign::Justify,
441                x::VAlign::Distributed => VAlign::Distributed,
442                _ => VAlign::Bottom,
443            },
444            wrap: a.wrap,
445            shrink: a.shrink,
446            indent: a.indent,
447            rotation: a.rotation,
448            reading_order: None,
449        },
450        borders: Borders {
451            left: line(b.left),
452            right: line(b.right),
453            top: line(b.top),
454            bottom: line(b.bottom),
455            diag_down: line(b.diag_down),
456            diag_up: line(b.diag_up),
457        },
458        fill: match pattern(f.fill.pattern) {
459            PatternType::None => Fill::default(),
460            PatternType::Solid => Fill::solid(color(f.fill.fg)),
461            p => Fill { pattern: p, fg: color(f.fill.fg), bg: color(f.fill.bg) },
462        },
463        ..Style::default()
464    }
