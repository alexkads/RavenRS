6fn cf_op(s: Option<&str>) -> CfOperator {
7    match s.unwrap_or("between") {
8        "notBetween" => CfOperator::NotBetween,
9        "equal" => CfOperator::Equal,
10        "notEqual" => CfOperator::NotEqual,
11        "greaterThan" => CfOperator::Greater,
12        "lessThan" => CfOperator::Less,
13        "greaterThanOrEqual" => CfOperator::GreaterOrEqual,
14        "lessThanOrEqual" => CfOperator::LessOrEqual,
15        _ => CfOperator::Between,
16    }
17}

19fn cfvo(e: &El) -> CfValueKind {
20    let val = e.attr("val").unwrap_or("");
21    let n = val.trim().parse::<f64>().ok().filter(|v| v.is_finite());
22    match e.attr("type").unwrap_or("num") {
23        "min" | "autoMin" => CfValueKind::Min,
24        "max" | "autoMax" => CfValueKind::Max,
25        "percent" => n.map(CfValueKind::Percent).unwrap_or_else(|| CfValueKind::Formula(from_file(val))),
26        "percentile" => n.map(CfValueKind::Percentile).unwrap_or_else(|| CfValueKind::Formula(from_file(val))),
27        "formula" => CfValueKind::Formula(from_file(val)),
28        _ => n.map(CfValueKind::Number).unwrap_or_else(|| CfValueKind::Formula(from_file(val))),
29    }
30}

32/// `cfvo` elements: plain children, or x14 ones with the value in an `xm:f` child.
33fn cfvos(e: &El) -> Vec<CfValueKind> {
34    e.kids("cfvo")
35        .map(|c| {
36            if c.attr("val").is_none()
37                && let Some(f) = c.child("f")
38            {
39                let mut c2 = c.clone();
40                c2.attrs.push(("val".into(), f.text.clone()));
41                return cfvo(&c2);
42            }
43            cfvo(c)
44        })
45        .collect()
46}

48/// One `conditionalFormatting` block. Every rule keeps its own copy of the block's ranges, so
49/// `kept_cells` counts the range cells the sheet's retained rules hold so far, across blocks; a
50/// rule that would take it past [`MAX_COND_CELLS`] is dropped. Unsupported rules cost nothing.
51pub(super) fn read_cf(cx: &mut Ctx<'_>, e: &El, sheet: &mut Sheet, kept_cells: &mut u64) {
52    let ranges = parse_sqref(e.attr("sqref"));
53    if ranges.is_empty() {
54        return;
55    }
56    let block_cells = ranges.iter().map(RangeRef::count).fold(0u64, u64::saturating_add);
57    for r in e.kids("cfRule") {
58        let style = Box::new(r.attr_u32("dxfId").and_then(|i| cx.dxfs.get(i as usize)).cloned().unwrap_or_default());
59        let formulas: Vec<String> = r.kids("formula").map(|f| from_file(f.text.trim())).collect();
60        let f0 = formulas.first().cloned().unwrap_or_default();
61        let text = r.attr("text").unwrap_or("").to_string();
62        // The x14 id that links a data bar to its Excel 2010 extension (solid fill lives there).
63        let ext_id =
64            r.child("extLst").and_then(|l| l.kids("ext").find_map(|x| x.child("id"))).map(|i| i.text.trim().to_string()).filter(|i| !i.is_empty());
65        let rule = match r.attr("type").unwrap_or("") {
66            "cellIs" => CfRule::CellIs { op: cf_op(r.attr("operator")), a: f0, b: formulas.get(1).cloned(), style },
67            "expression" => CfRule::Expression { formula: f0, style },
68            "containsText" => CfRule::ContainsText { text, style },
69            "notContainsText" => CfRule::NotContainsText { text, style },
70            "beginsWith" => CfRule::BeginsWith { text, style },
71            "endsWith" => CfRule::EndsWith { text, style },
72            "containsBlanks" => CfRule::Blanks { style },
73            "notContainsBlanks" => CfRule::NoBlanks { style },
74            "containsErrors" => CfRule::Errors { style },
75            "notContainsErrors" => CfRule::NoErrors { style },
76            "duplicateValues" => CfRule::Duplicate { style },
77            "uniqueValues" => CfRule::Unique { style },
78            "top10" => {
79                CfRule::Top10 { bottom: r.flag("bottom", false), percent: r.flag("percent", false), rank: r.attr_u32("rank").unwrap_or(10), style }
80            }
81            "aboveAverage" => CfRule::AboveAverage {
82                below: !r.flag("aboveAverage", true),
83                equal: r.flag("equalAverage", false),
84                std_dev: r.attr_u32("stdDev").unwrap_or(0).min(3) as u8,
85                style,
86            },
87            "timePeriod" => CfRule::TimePeriod { period: r.attr("timePeriod").unwrap_or("today").to_string(), style },
88            "colorScale" => {
89                let Some(cs) = r.child("colorScale") else { continue };
90                let colors = cs.kids("color").map(|c| read_color(c, &cx.palette));
91                CfRule::ColorScale { stops: cfvos(cs).into_iter().zip(colors).collect() }
92            }
93            "dataBar" => {
94                let Some(db) = r.child("dataBar") else { continue };
95                let v = cfvos(db);
96                CfRule::DataBar {
97                    min: v.first().cloned().unwrap_or(CfValueKind::Min),
98                    max: v.get(1).cloned().unwrap_or(CfValueKind::Max),
99                    color: db.child("color").map(|c| read_color(c, &cx.palette)).unwrap_or_default(),
100                    gradient: true,
101                    show_value: db.flag("showValue", true),
102                }
103            }
104            "iconSet" => {
105                let Some(is) = r.child("iconSet") else { continue };
106                CfRule::IconSet {
107                    set: is.attr("iconSet").unwrap_or("3TrafficLights1").to_string(),
108                    thresholds: cfvos(is),
109                    reverse: is.flag("reverse", false),
110                    show_value: is.flag("showValue", true),
111                }
112            }
113            other => {
114                cx.warn(format!("conditional format type \"{other}\" is not supported"));
115                continue;
116            }
117        };
118        let total = kept_cells.saturating_add(block_cells);
119        if total > MAX_COND_CELLS {
120            cx.warn(format!("conditional formats past {MAX_COND_CELLS} kept range cells were dropped"));
121            break;
122        }
123        *kept_cells = total;
124        if let Some(id) = ext_id {
125            cx.cf_ext_ids.push((id, sheet.cond_formats.len()));
126        }
127        sheet.cond_formats.push(CondFormat {
128            ranges: ranges.clone(),
129            rule,
130            priority: r.attr_u32("priority").unwrap_or(0),
131            stop_if_true: r.flag("stopIfTrue", false),
132        });
133    }
134}

136/// A `dataValidation` (main namespace, or x14 with `xm:f` formulas and an `xm:sqref` child).
137pub(super) fn read_validation(v: &El) -> Option<Validation> {
138    let sq = v.attr("sqref").map(str::to_string).or_else(|| v.child("sqref").map(|s| s.text.clone()));
139    let ranges = parse_sqref(sq.as_deref());
140    if ranges.is_empty() {
141        return None;
142    }
143    let formula = |n: &str| -> Option<String> {
144        let f = v.child(n)?;
145        let t = f.child("f").map(|x| x.text.as_str()).unwrap_or(f.text.as_str()).trim();
146        if t.is_empty() { None } else { Some(t.to_string()) }
147    };
148    let kind = match v.attr("type").unwrap_or("none") {
149        "whole" => ValidationKind::Whole,
150        "decimal" => ValidationKind::Decimal,
151        "list" => ValidationKind::List,
152        "date" => ValidationKind::Date,
153        "time" => ValidationKind::Time,
154        "textLength" => ValidationKind::TextLength,
155        "custom" => ValidationKind::Custom,
156        _ => ValidationKind::Any,
157    };
158    let mut f1 = formula("formula1").unwrap_or_default();
159    if kind == ValidationKind::List && !f1.is_empty() && !f1.starts_with('"') {
160        f1 = format!("={}", from_file(&f1));
161    } else if !f1.starts_with('"') && !f1.is_empty() {
162        f1 = from_file(&f1);
163    }
164    Some(Validation {
165        ranges,
166        kind,
167        op: cf_op(v.attr("operator")),
168        f1,
169        f2: formula("formula2").map(|f| from_file(&f)),
170        allow_blank: v.flag("allowBlank", false),
171        // `showDropDown="1"` *hides* the in-cell list arrow.
172        in_cell_dropdown: !v.flag("showDropDown", false),
173        input_title: v.attr("promptTitle").unwrap_or("").to_string(),
174        input_message: v.attr("prompt").unwrap_or("").to_string(),
175        show_input: v.flag("showInputMessage", false),
176        error_title: v.attr("errorTitle").unwrap_or("").to_string(),
177        error_message: v.attr("error").unwrap_or("").to_string(),
178        error_style: match v.attr("errorStyle") {
179            Some("warning") => ErrorStyle::Warning,
180            Some("information") => ErrorStyle::Information,
181            _ => ErrorStyle::Stop,
182        },
183        show_error: v.flag("showErrorMessage", false),
184    })
