3fn cf_op_name(op: CfOperator) -> &'static str {
4    match op {
5        CfOperator::Between => "between",
6        CfOperator::NotBetween => "notBetween",
7        CfOperator::Equal => "equal",
8        CfOperator::NotEqual => "notEqual",
9        CfOperator::Greater => "greaterThan",
10        CfOperator::Less => "lessThan",
11        CfOperator::GreaterOrEqual => "greaterThanOrEqual",
12        CfOperator::LessOrEqual => "lessThanOrEqual",
13    }
14}

16fn cfvo_xml(v: &CfValueKind) -> String {
17    match v {
18        CfValueKind::Min => "<cfvo type=\"min\"/>".into(),
19        CfValueKind::Max => "<cfvo type=\"max\"/>".into(),
20        CfValueKind::Number(n) => format!("<cfvo type=\"num\" val=\"{}\"/>", num(*n)),
21        CfValueKind::Percent(n) => format!("<cfvo type=\"percent\" val=\"{}\"/>", num(*n)),
22        CfValueKind::Percentile(n) => format!("<cfvo type=\"percentile\" val=\"{}\"/>", num(*n)),
23        CfValueKind::Formula(f) => format!("<cfvo type=\"formula\" val=\"{}\"/>", esc_attr(&text_to_file(f))),
24    }
25}

37fn solid_bar_ext(priority: u32) -> String {

39    format!("<extLst><ext uri=\"{{B025F937-C7B1-47D3-B67F-A62EFF666E3E}}\" xmlns:x14=\"{X14_NS}\"><x14:id>{id}</x14:id></ext></extLst>")

42/// An x14 `cfvo`: unlike the 2006 one it has no `val` attribute, the value is an `xm:f` child.
43pub(super) fn x14_cfvo(v: &CfValueKind) -> String {
44    let (ty, val) = match v {
45        CfValueKind::Min => return "<x14:cfvo type=\"min\"/>".into(),
46        CfValueKind::Max => return "<x14:cfvo type=\"max\"/>".into(),
47        CfValueKind::Number(n) => ("num", num(*n)),
48        CfValueKind::Percent(n) => ("percent", num(*n)),
49        CfValueKind::Percentile(n) => ("percentile", num(*n)),
50        CfValueKind::Formula(f) => ("formula", text_to_file(f)),
51    };
52    format!("<x14:cfvo type=\"{ty}\"><xm:f>{}</xm:f></x14:cfvo>", esc(&val))
53}

55/// One `cfRule` element. The formulas generated for text, blank, error and date rules refer to the
56/// top-left cell of all the rule's areas, the cell Excel reads relative references from.
57pub(super) fn cf_rule_xml(cf: &CondFormat, priority: u32, out: &mut Out) -> String {
58    let tl = RangeRef::top_left(&cf.ranges).map(|c| c.a1()).unwrap_or_else(|| "A1".into());
59    let stop = if cf.stop_if_true { " stopIfTrue=\"1\"" } else { "" };
60    let mut dxf = |st: &Style| dxf_id(&mut out.dxfs, st);
61    let f = |x: &str| format!("<formula>{}</formula>", esc(&text_to_file(x)));
62    let simple = |ty: &str, st: &Style, extra: &str, body: String, dxf: &mut dyn FnMut(&Style) -> u32| {
63        format!("<cfRule type=\"{ty}\" dxfId=\"{}\" priority=\"{priority}\"{stop}{extra}>{body}</cfRule>", dxf(st))
64    };
65    match &cf.rule {
66        CfRule::CellIs { op, a, b, style } => {
67            let mut body = f(a);
68            if let Some(b) = b {
69                body.push_str(&f(b));
70            }
71            simple("cellIs", style, &format!(" operator=\"{}\"", cf_op_name(*op)), body, &mut dxf)
72        }
73        CfRule::Expression { formula, style } => simple("expression", style, "", f(formula), &mut dxf),
74        CfRule::ContainsText { text, style } => simple(
75            "containsText",
76            style,
77            &format!(" operator=\"containsText\" text=\"{}\"", esc_attr(text)),
78            f(&format!("NOT(ISERROR(SEARCH({},{tl})))", quote(text))),
79            &mut dxf,
80        ),
81        CfRule::NotContainsText { text, style } => simple(
82            "notContainsText",
83            style,
84            &format!(" operator=\"notContains\" text=\"{}\"", esc_attr(text)),
85            f(&format!("ISERROR(SEARCH({},{tl}))", quote(text))),
86            &mut dxf,
87        ),
88        CfRule::BeginsWith { text, style } => simple(
89            "beginsWith",
90            style,
91            &format!(" operator=\"beginsWith\" text=\"{}\"", esc_attr(text)),
92            f(&format!("LEFT({tl},LEN({0}))={0}", quote(text))),
93            &mut dxf,
94        ),
95        CfRule::EndsWith { text, style } => simple(
96            "endsWith",
97            style,
98            &format!(" operator=\"endsWith\" text=\"{}\"", esc_attr(text)),
99            f(&format!("RIGHT({tl},LEN({0}))={0}", quote(text))),
100            &mut dxf,
101        ),
102        CfRule::Blanks { style } => simple("containsBlanks", style, "", f(&format!("LEN(TRIM({tl}))=0")), &mut dxf),
103        CfRule::NoBlanks { style } => simple("notContainsBlanks", style, "", f(&format!("LEN(TRIM({tl}))>0")), &mut dxf),
104        CfRule::Errors { style } => simple("containsErrors", style, "", f(&format!("ISERROR({tl})")), &mut dxf),
105        CfRule::NoErrors { style } => simple("notContainsErrors", style, "", f(&format!("NOT(ISERROR({tl}))")), &mut dxf),
106        CfRule::Duplicate { style } => simple("duplicateValues", style, "", String::new(), &mut dxf),
107        CfRule::Unique { style } => simple("uniqueValues", style, "", String::new(), &mut dxf),
108        CfRule::Top10 { bottom, percent, rank, style } => {
109            let extra = format!("{}{} rank=\"{}\"", if *percent { " percent=\"1\"" } else { "" }, if *bottom { " bottom=\"1\"" } else { "" }, rank);
110            simple("top10", style, &extra, String::new(), &mut dxf)
111        }
112        CfRule::AboveAverage { below, equal, std_dev, style } => {
113            let mut extra = String::new();
114            if *below {
115                extra.push_str(" aboveAverage=\"0\"");
116            }
117            if *equal {
118                extra.push_str(" equalAverage=\"1\"");
119            }
120            if *std_dev > 0 {
121                let _ = write!(extra, " stdDev=\"{std_dev}\"");
122            }
123            simple("aboveAverage", style, &extra, String::new(), &mut dxf)
124        }
125        CfRule::TimePeriod { period, style } => {
126            let d = format!("FLOOR({tl},1)");
127            let body = match period.as_str() {
128                "yesterday" => format!("{d}=TODAY()-1"),
129                "tomorrow" => format!("{d}=TODAY()+1"),
130                "last7Days" => format!("AND(TODAY()-{d}<=6,{d}<=TODAY())"),
131                "thisMonth" => format!("AND(MONTH({tl})=MONTH(TODAY()),YEAR({tl})=YEAR(TODAY()))"),
132                "lastMonth" => format!("AND(MONTH({tl})=MONTH(EDATE(TODAY(),0-1)),YEAR({tl})=YEAR(EDATE(TODAY(),0-1)))"),
133                "nextMonth" => format!("AND(MONTH({tl})=MONTH(EDATE(TODAY(),0+1)),YEAR({tl})=YEAR(EDATE(TODAY(),0+1)))"),
134                "thisWeek" => format!("AND(TODAY()-ROUNDDOWN({tl},0)<=WEEKDAY(TODAY())-1,ROUNDDOWN({tl},0)-TODAY()<=7-WEEKDAY(TODAY()))"),
135                "lastWeek" => format!("AND(TODAY()-ROUNDDOWN({tl},0)>=(WEEKDAY(TODAY())),TODAY()-ROUNDDOWN({tl},0)<(WEEKDAY(TODAY())+7))"),
136                "nextWeek" => format!("AND(ROUNDDOWN({tl},0)-TODAY()>(7-WEEKDAY(TODAY())),ROUNDDOWN({tl},0)-TODAY()<(15-WEEKDAY(TODAY())))"),
137                _ => format!("{d}=TODAY()"),
138            };
139            simple("timePeriod", style, &format!(" timePeriod=\"{}\"", esc_attr(period)), f(&body), &mut dxf)
140        }
141        CfRule::ColorScale { stops } => {
142            let mut b = String::from("<colorScale>");
143            for (v, _) in stops {
144                b.push_str(&cfvo_xml(v));
145            }
146            for (_, c) in stops {
147                let _ = write!(b, "<color {}/>", color_attrs(c));
148            }
149            b.push_str("</colorScale>");
150            format!("<cfRule type=\"colorScale\" priority=\"{priority}\"{stop}>{b}</cfRule>")
151        }
152        CfRule::DataBar { min, max, color, show_value, gradient } => format!(
153            "<cfRule type=\"dataBar\" priority=\"{priority}\"{stop}><dataBar{}>{}{}<color {}/></dataBar>{}</cfRule>",
154            if *show_value { "" } else { " showValue=\"0\"" },
155            cfvo_xml(min),
156            cfvo_xml(max),
157            if *color == Color::Auto { "rgb=\"FF638EC6\"".to_string() } else { color_attrs(color) },
158            if *gradient { String::new() } else { solid_bar_ext(priority) }
159        ),
160        CfRule::IconSet { set, thresholds, reverse, show_value } => {
161            let mut b = format!("<iconSet iconSet=\"{}\"", esc_attr(set));
162            if *reverse {
163                b.push_str(" reverse=\"1\"");
164            }
165            if !*show_value {
166                b.push_str(" showValue=\"0\"");
167            }
168            b.push('>');
169            // CT_IconSet needs one cfvo per icon (minOccurs 2); the first is the lower bound.
170            let k = icon_count(set);
171            let defaults = default_icon_thresholds(set);
172            let cfvos: Vec<&CfValueKind> = match thresholds.len() {
173                n if n == k => thresholds.iter().collect(),
174                n if n + 1 == k => defaults.iter().take(1).chain(thresholds).collect(),
175                _ => defaults.iter().collect(),
176            };
177            for t in cfvos {
178                b.push_str(&cfvo_xml(t));
179            }
180            b.push_str("</iconSet>");
181            format!("<cfRule type=\"iconSet\" priority=\"{priority}\"{stop}>{b}</cfRule>")
182        }
183    }
184}

186fn validation_formula(f: &str) -> String {
187    let f = f.trim();
188    if f.starts_with('"') {
189        return esc(f);
190    }
191    esc(&text_to_file(f.strip_prefix('=').unwrap_or(f)))
192}

194pub(super) fn validation_xml(v: &Validation) -> String {
195    let ty = match v.kind {
196        ValidationKind::Any => "none",
197        ValidationKind::Whole => "whole",
198        ValidationKind::Decimal => "decimal",
199        ValidationKind::List => "list",
200        ValidationKind::Date => "date",
201        ValidationKind::Time => "time",
202        ValidationKind::TextLength => "textLength",
203        ValidationKind::Custom => "custom",
204    };
205    let mut s = format!("<dataValidation type=\"{ty}\"");
206    match v.error_style {
207        ErrorStyle::Stop => {}
208        ErrorStyle::Warning => s.push_str(" errorStyle=\"warning\""),
209        ErrorStyle::Information => s.push_str(" errorStyle=\"information\""),
210    }
211    if v.op != CfOperator::Between && !matches!(v.kind, ValidationKind::List | ValidationKind::Custom | ValidationKind::Any) {
212        let _ = write!(s, " operator=\"{}\"", cf_op_name(v.op));
213    }
214    if v.allow_blank {
215        s.push_str(" allowBlank=\"1\"");
216    }
217    if !v.in_cell_dropdown {
218        s.push_str(" showDropDown=\"1\"");
219    }
220    if v.show_input {
221        s.push_str(" showInputMessage=\"1\"");
222    }
223    if v.show_error {
224        s.push_str(" showErrorMessage=\"1\"");
225    }
226    for (a, t) in [("errorTitle", &v.error_title), ("error", &v.error_message), ("promptTitle", &v.input_title), ("prompt", &v.input_message)] {
227        if !t.is_empty() {
228            let _ = write!(s, " {a}=\"{}\"", esc_attr(t));
229        }
230    }
231    let _ = write!(s, " sqref=\"{}\">", sqref(&v.ranges));
232    if !v.f1.is_empty() {
233        let _ = write!(s, "<formula1>{}</formula1>", validation_formula(&v.f1));
234    }
235    if let Some(f2) = v.f2.as_ref().filter(|f| !f.is_empty()) {
236        let _ = write!(s, "<formula2>{}</formula2>", validation_formula(f2));
237    }
238    s.push_str("</dataValidation>");
239    s
240}
