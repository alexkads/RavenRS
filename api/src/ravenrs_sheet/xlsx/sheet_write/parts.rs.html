5/// Valid table name: letters, digits, `_` and `.`, starting with a letter, `_` or `\`
6/// (Microsoft "Names in formulas" rules).
7pub(super) fn table_name(name: &str, used: &mut HashSet<String>, n: u32) -> String {
8    let mut s: String =
9        name.chars().enumerate().map(|(i, c)| if c.is_alphanumeric() || c == '_' || c == '.' || (i == 0 && c == '\\') { c } else { '_' }).collect();
10    if s.is_empty() || !s.starts_with(|c: char| c.is_alphabetic() || c == '_' || c == '\\') || crate::base::CellRef::parse(&s).is_some() {
11        s = format!("Table{n}");
12    }
13    let mut out = s.clone();
14    let mut k = 2;
15    while !used.insert(out.to_ascii_lowercase()) {
16        out = format!("{s}{k}");
17        k += 1;
18    }
19    out
20}

22/// Column names for a table: one per column of its range, unique and non-empty.
23pub(super) fn table_columns(t: &Table) -> Vec<String> {
24    let w = t.range.width() as usize;
25    let mut used = HashSet::new();
26    // The next suffix to try per lowercased base, so repeated headers don't retry every suffix
27    // already handed out.
28    let mut next: HashMap<String, u32> = HashMap::new();
29    (0..w)
30        .map(|i| {
31            let base = t.columns.get(i).map(|c| c.name.clone()).filter(|n| !n.trim().is_empty()).unwrap_or_else(|| format!("Column{}", i + 1));
32            let lower = base.to_lowercase();
33            if used.insert(lower.clone()) {
34                return base;
35            }
36            let k = next.entry(lower).or_insert(2);
37            loop {
38                let name = format!("{base}{k}");
39                *k += 1;
40                if used.insert(name.to_lowercase()) {
41                    return name;
42                }
43            }
44        })
45        .collect()
46}

48fn filter_operator(op: &str) -> &'static str {
49    match op.trim() {
50        "<" => " operator=\"lessThan\"",
51        "<=" => " operator=\"lessThanOrEqual\"",
52        "<>" => " operator=\"notEqual\"",
53        ">=" => " operator=\"greaterThanOrEqual\"",
54        ">" => " operator=\"greaterThan\"",
55        _ => "",
56    }
57}

59pub(super) fn autofilter_xml(range: RangeRef, criteria: &[(u32, FilterCriterion)], out: &mut Out) -> String {
60    let mut s = format!("<autoFilter ref=\"{}\"", range.a1());
61    if criteria.is_empty() {
62        s.push_str("/>");
63        return s;
64    }
65    s.push('>');
66    for (col, c) in criteria {
67        let _ = write!(s, "<filterColumn colId=\"{col}\">");
68        match c {
69            FilterCriterion::Values { values, blanks } => {
70                let _ = write!(s, "<filters{}>", if *blanks { " blank=\"1\"" } else { "" });
71                for v in values {
72                    let _ = write!(s, "<filter val=\"{}\"/>", esc_attr(v));
73                }
74                s.push_str("</filters>");
75            }
76            FilterCriterion::Custom { a, b, and } => {
77                let _ = write!(s, "<customFilters{}>", if *and { " and=\"1\"" } else { "" });
78                for (op, v) in std::iter::once(a).chain(b.iter()) {
79                    let _ = write!(s, "<customFilter{} val=\"{}\"/>", filter_operator(op), esc_attr(v));
80                }
81                s.push_str("</customFilters>");
82            }
83            FilterCriterion::Top10 { bottom, percent, count } => {
84                let _ =
85                    write!(s, "<top10{}{} val=\"{count}\"/>", if *bottom { " top=\"0\"" } else { "" }, if *percent { " percent=\"1\"" } else { "" });
86            }
87            FilterCriterion::AboveAverage(above) => {
88                let _ = write!(s, "<dynamicFilter type=\"{}\"/>", if *above { "aboveAverage" } else { "belowAverage" });
89            }
90            FilterCriterion::FillColor(c) => {
91                let st = Style { fill: crate::model::Fill::solid(*c), ..Default::default() };
92                let _ = write!(s, "<colorFilter dxfId=\"{}\"/>", dxf_id(&mut out.dxfs, &st));
93            }
94            FilterCriterion::FontColor(c) => {
95                let mut st = Style::default();
96                st.font.color = *c;
97                let _ = write!(s, "<colorFilter dxfId=\"{}\" cellColor=\"0\"/>", dxf_id(&mut out.dxfs, &st));
98            }
99            FilterCriterion::Icon { set, id } => {
100                let _ = write!(s, "<iconFilter iconSet=\"{}\"", esc_attr(set));
101                if let Some(id) = id {
102                    let _ = write!(s, " iconId=\"{id}\"");
103                }
104                s.push_str("/>");
105            }
106        }
107        s.push_str("</filterColumn>");
108    }
109    s.push_str("</autoFilter>");
110    s
111}

113pub(super) fn table_xml(t: &Table, cols: &[String], name: &str, id: u32, out: &mut Out) -> String {
114    let mut s =
115        format!("{XML_DECL}<table xmlns=\"{NS_MAIN}\" id=\"{id}\" name=\"{0}\" displayName=\"{0}\" ref=\"{1}\"", esc_attr(name), t.range.a1());
116    if !t.header_row {
117        s.push_str(" headerRowCount=\"0\"");
118    }
119    if t.totals_row {
120        s.push_str(" totalsRowCount=\"1\"");
121    } else {
122        s.push_str(" totalsRowShown=\"0\"");
123    }
124    if let Some(ty) = &t.table_type {
125        let _ = write!(s, " tableType=\"{}\"", esc_attr(ty));
126    }
127    if let Some(c) = t.connection_id {
128        let _ = write!(s, " connectionId=\"{c}\"");
129    }
130    s.push('>');
131    if t.header_row && t.filter_button {
132        let end_row = if t.totals_row { t.range.end.row.saturating_sub(1).max(t.range.start.row) } else { t.range.end.row };
133        let r = RangeRef::new(t.range.start, CellRef::new(end_row, t.range.end.col));
134        s.push_str(&autofilter_xml(r, &t.filter, out));
135    }
136    let _ = write!(s, "<tableColumns count=\"{}\">", cols.len());
137    for (i, name) in cols.iter().enumerate() {
138        let col = t.columns.get(i);
139        let _ = write!(s, "<tableColumn id=\"{}\"", i + 1);
140        if let Some(u) = col.and_then(|c| c.unique_name.as_ref()) {
141            let _ = write!(s, " uniqueName=\"{}\"", esc_attr(u));
142        }
143        let _ = write!(s, " name=\"{}\"", esc_attr(&encode_xstring(name)));
144        let totals = col.map(|c| c.totals).unwrap_or_default();
145        if t.totals_row {
146            let f = match totals {
147                TotalsFn::None => None,
148                TotalsFn::Average => Some("average"),
149                TotalsFn::Count => Some("count"),
150                TotalsFn::CountNums => Some("countNums"),
151                TotalsFn::Max => Some("max"),
152                TotalsFn::Min => Some("min"),
153                TotalsFn::Sum => Some("sum"),
154                TotalsFn::StdDev => Some("stdDev"),
155                TotalsFn::Var => Some("var"),
156                TotalsFn::Custom => Some("custom"),
157            };
158            if let Some(f) = f {
159                let _ = write!(s, " totalsRowFunction=\"{f}\"");
160            } else if let Some(l) = col.and_then(|c| c.totals_label.as_ref()) {
161                let _ = write!(s, " totalsRowLabel=\"{}\"", esc_attr(l));
162            }
163        }
164        let calc = col.and_then(|c| c.formula.as_ref()).filter(|f| !f.is_empty());
165        let custom = if t.totals_row && totals == TotalsFn::Custom { col.and_then(|c| c.totals_label.as_ref()) } else { None };
166        let xml = col.and_then(|c| c.xml.as_ref());
167        if calc.is_none() && custom.is_none() && xml.is_none() {
168            s.push_str("/>");
169            continue;
170        }
171        s.push('>');
172        if let Some(f) = calc {
173            let _ = write!(s, "<calculatedColumnFormula>{}</calculatedColumnFormula>", esc(&text_to_file(f)));
174        }
175        if let Some(f) = custom {
176            let _ = write!(s, "<totalsRowFormula>{}</totalsRowFormula>", esc(&text_to_file(f)));
177        }
178        if let Some(x) = xml {
179            let _ = write!(s, "<xmlColumnPr mapId=\"{}\" xpath=\"{}\"", x.map_id, esc_attr(&x.xpath));
180            if x.denormalized {
181                s.push_str(" denormalized=\"1\"");
182            }
183            let _ = write!(s, " xmlDataType=\"{}\"/>", esc_attr(&x.data_type));
184        }
185        s.push_str("</tableColumn>");
186    }
187    s.push_str("</tableColumns>");
188    let style = if t.style.is_empty() { "TableStyleMedium2" } else { &t.style };
189    let _ = write!(
190        s,
191        "<tableStyleInfo name=\"{}\" showFirstColumn=\"{}\" showLastColumn=\"{}\" showRowStripes=\"{}\" showColumnStripes=\"{}\"/></table>",
192        esc_attr(style),
193        t.first_col as u8,
194        t.last_col as u8,
195        t.banded_rows as u8,
196        t.banded_cols as u8
197    );
198    s
199}

201/// Ids of the threaded comments of sheet `si`: the thread's id, then one per reply. GUID-shaped
202/// and deterministic; they only need to be unique within the file.
203pub(super) fn thread_ids(sheet: &Sheet, si: usize) -> BTreeMap<CellRef, Vec<String>> {
204    let mut seq: u64 = 0;
205    let mut ids = BTreeMap::new();
206    for (cell, c) in sheet.comments.iter().filter(|(_, c)| c.threaded) {
207        let v: Vec<String> = (0..=c.replies.len())
208            .map(|_| {
209                seq += 1;
210                format!("{{{:08X}-0000-4000-8001-{:012X}}}", si as u64 & 0xFFFF_FFFF, seq & 0xFFFF_FFFF_FFFF)
211            })
212            .collect();
213        ids.insert(*cell, v);
214    }
215    ids
216}

218pub(super) fn comments_xml(sheet: &Sheet, ids: &BTreeMap<CellRef, Vec<String>>) -> String {
219    // A threaded comment's legacy note is authored by `tc={thread id}`.
220    let author_of = |cell: &CellRef, c: &crate::model::Comment| match ids.get(cell).and_then(|v| v.first()) {
221        Some(id) => format!("tc={id}"),
222        None => c.author.clone(),
223    };
224    let mut authors: Vec<String> = vec![];
225    for (cell, c) in &sheet.comments {
226        let a = author_of(cell, c);
227        if !authors.contains(&a) {
228            authors.push(a);
229        }
230    }
231    let mut s = if ids.is_empty() {
232        format!("{XML_DECL}<comments xmlns=\"{NS_MAIN}\"><authors>")
233    } else {
234        format!(
235            "{XML_DECL}<comments xmlns=\"{NS_MAIN}\" xmlns:mc=\"http://schemas.openxmlformats.org/markup-compatibility/2006\" mc:Ignorable=\"xr\" xmlns:xr=\"http://schemas.microsoft.com/office/spreadsheetml/2014/revision\"><authors>"
236        )
237    };
238    for a in &authors {
239        let _ = write!(s, "<author>{}</author>", esc(a));
240    }
241    s.push_str("</authors><commentList>");
242    for (cell, c) in &sheet.comments {
243        let a = author_of(cell, c);
244        let aid = authors.iter().position(|x| *x == a).unwrap_or(0);
245        let mut text = c.text.clone();
246        // Threaded replies are kept as text lines.
247        for (who, reply) in &c.replies {
248            let _ = write!(text, "\n{who}: {reply}");
249        }
250        let link = match ids.get(cell).and_then(|v| v.first()) {
251            Some(id) => format!(" shapeId=\"0\" xr:uid=\"{id}\""),
252            None => String::new(),
253        };
254        let _ = write!(s, "<comment ref=\"{}\" authorId=\"{aid}\"{link}><text><r>{}</r></text></comment>", cell.a1(), crate::xlsx::xml::t_el(&text));
255    }
256    s.push_str("</commentList></comments>");
257    s
258}

260/// `2024-05-01T09:30:00` with optional fraction and zone: what `dT` (xsd:dateTime) accepts.
261/// Anything else is left out rather than written into the file.
262fn iso_datetime(d: &str) -> bool {
263    let b = d.as_bytes();
264    let digits = |r: std::ops::Range<usize>| b.get(r).is_some_and(|s| s.iter().all(u8::is_ascii_digit));
265    d.len() <= 40
266        && digits(0..4)
267        && b.get(4) == Some(&b'-')
268        && digits(5..7)
269        && b.get(7) == Some(&b'-')
270        && digits(8..10)
271        && b.get(10) == Some(&b'T')
272        && digits(11..13)
273        && b.get(13) == Some(&b':')
274        && digits(14..16)
275        && b.get(16) == Some(&b':')
276        && digits(17..19)
277        && b.get(19..).is_some_and(|rest| rest.iter().all(|c| c.is_ascii_digit() || b".:+-Z".contains(c)))
278}

280/// `xl/threadedComments/threadedCommentN.xml`: each thread, then its replies (`parentId`).
281pub(super) fn threaded_xml(sheet: &Sheet, ids: &BTreeMap<CellRef, Vec<String>>, persons: &mut Vec<String>) -> String {
282    let mut person = |name: &str| {
283        let i = persons.iter().position(|p| p == name).unwrap_or_else(|| {
284            persons.push(name.to_string());
285            persons.len() - 1
286        });
287        crate::xlsx::write::person_guid(i)
288    };
289    let mut s = format!("{XML_DECL}<ThreadedComments xmlns=\"{NS_THREADED}\" xmlns:x=\"{NS_MAIN}\">");
290    for (cell, id) in ids {
291        let Some(c) = sheet.comments.get(cell) else { continue };
292        let entries = std::iter::once((&c.author, &c.text)).chain(c.replies.iter().map(|(a, t)| (a, t)));
293        for (k, ((author, text), own)) in entries.zip(id).enumerate() {
294            let date = match c.dates.get(k).filter(|d| iso_datetime(d)) {
295                Some(d) => format!(" dT=\"{}\"", esc_attr(d)),
296                None => String::new(),
297            };
298            let tail = match (k, id.first()) {
299                (0, _) if c.resolved => " done=\"1\"".to_string(),
300                (0, _) | (_, None) => String::new(),
301                (_, Some(parent)) => format!(" parentId=\"{parent}\""),
302            };
303            let _ = write!(
304                s,
305                "<threadedComment ref=\"{}\"{date} personId=\"{}\" id=\"{own}\"{tail}><text>{}</text></threadedComment>",
306                cell.a1(),
307                person(author),
308                esc(text)
309            );
310        }
311    }
312    s.push_str("</ThreadedComments>");
313    s
314}

316/// A protection password hash as attributes, each name preceded by `prefix` (see
317/// [`crate::xlsx::sheet_read::read_password`]).
318pub(crate) fn password_attrs(h: &crate::model::PasswordHash, prefix: &str) -> String {
319    let name = |base: &str| crate::xlsx::sheet_read::password_attr(prefix, base);
320    match h {
321        crate::model::PasswordHash::Legacy(h) => format!(" {}=\"{h:04X}\"", name("password")),
322        crate::model::PasswordHash::Iterated { algorithm, hash, salt, spin_count } => format!(
323            " {}=\"{}\" {}=\"{}\" {}=\"{}\" {}=\"{spin_count}\"",
324            name("algorithmName"),
325            esc_attr(algorithm),
326            name("hashValue"),
327            esc_attr(hash),
328            name("saltValue"),
329            esc_attr(salt),
330            name("spinCount")
331        ),
332    }
333}

335/// Allow Edit Ranges: `<protectedRanges>` (after `sheetProtection`).
336pub(super) fn protected_ranges_xml(sheet: &Sheet) -> String {
337    let mut s = String::new();
338    for r in sheet.edit_ranges.iter().filter(|r| !r.ranges.is_empty()) {
339        let sqref: Vec<String> = r.ranges.iter().map(|x| x.a1()).collect();
340        let _ = write!(s, "<protectedRange name=\"{}\" sqref=\"{}\"", esc_attr(&r.title), sqref.join(" "));
341        if let Some(h) = &r.password {
342            s.push_str(&password_attrs(h, ""));
343        }
344        if let Some(sd) = &r.security_descriptor {
345            let _ = write!(s, " securityDescriptor=\"{}\"", esc_attr(sd));
346        }
347        s.push_str("/>");
348    }
349    if s.is_empty() { s } else { format!("<protectedRanges>{s}</protectedRanges>") }
350}

352/// A note box's `<x:Anchor>` text and its width and height in points: the box read from the file
353/// or [`DEFAULT_NOTE_BOX`], its corners moved with the note's `cell` and kept on the sheet.
354fn note_anchor(cell: CellRef, b: Option<&NoteBox>) -> (String, (f64, f64)) {
355    let b = b.unwrap_or(&DEFAULT_NOTE_BOX);
356    let at = |base: u32, d: i64, max: u32| (i64::from(base) + d).clamp(0, i64::from(max) - 1);
357    let corner = |k: &NoteCorner| format!("{}, {}, {}, {}", at(cell.col, k.col, MAX_COLS), k.col_px, at(cell.row, k.row, MAX_ROWS), k.row_px);
358    (format!("{}, {}", corner(&b.from), corner(&b.to)), b.size_pt.unwrap_or(DEFAULT_NOTE_SIZE_PT))
359}

361/// Legacy VML shapes Excel uses to display notes (numbered from `base + 1`), followed by the
362/// form control shapes `controls`. `idmap` lists the shape blocks the drawing uses.
363pub(super) fn vml_xml(sheet: &Sheet, idmap: &str, base: u64, controls: &str) -> String {
364    let mut s = format!(
365        "<xml xmlns:v=\"urn:schemas-microsoft-com:vml\" xmlns:o=\"urn:schemas-microsoft-com:office:office\" xmlns:x=\"urn:schemas-microsoft-com:office:excel\"><o:shapelayout v:ext=\"edit\"><o:idmap v:ext=\"edit\" data=\"{idmap}\"/></o:shapelayout><v:shapetype id=\"_x0000_t202\" coordsize=\"21600,21600\" o:spt=\"202\" path=\"m,l,21600r21600,l21600,xe\"><v:stroke joinstyle=\"miter\"/><v:path gradientshapeok=\"t\" o:connecttype=\"rect\"/></v:shapetype>"
366    );
367    for (i, (cell, c)) in sheet.comments.iter().enumerate() {
368        let id = base + 1 + i as u64;
369        let vis = if c.visible { "visible" } else { "hidden" };
370        let (anchor, (w, h)) = note_anchor(*cell, c.note_box.as_ref());
371        let _ = write!(
372            s,
373            "<v:shape id=\"_x0000_s{id}\" type=\"#_x0000_t202\" style=\"position:absolute;margin-left:59.25pt;margin-top:1.5pt;width:{}pt;height:{}pt;z-index:{};visibility:{vis}\" fillcolor=\"#ffffe1\" o:insetmode=\"auto\"><v:fill color2=\"#ffffe1\"/><v:shadow on=\"t\" color=\"black\" obscured=\"t\"/><v:path o:connecttype=\"none\"/><v:textbox style=\"mso-direction-alt:auto\"><div style=\"text-align:left\"></div></v:textbox><x:ClientData ObjectType=\"Note\"><x:MoveWithCells/><x:SizeWithCells/><x:Anchor>{anchor}</x:Anchor><x:AutoFill>False</x:AutoFill><x:Row>{}</x:Row><x:Column>{}</x:Column>{}</x:ClientData></v:shape>",
374            num(w),
375            num(h),
376            i + 1,
377            cell.row,
378            cell.col,
379            if c.visible { "<x:Visible/>" } else { "" }
380        );
381    }
382    s.push_str(controls);
383    s.push_str("</xml>");
384    s
385}
