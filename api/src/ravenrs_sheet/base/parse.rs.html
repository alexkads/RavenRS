1//! Parsing what people type into a cell (and numeric text in formulas).
2//!
3//! Recognises numbers with thousands separators, percentages, currency, scientific notation,
4//! fractions (`1 1/2`), dates (`3/4/2026`, `2026-03-04`, `4-Mar-2026`, `Mar 4, 2026`), times
5//! (`10:30`, `10:30:15 PM`) and date-times. Locale: en-US.
6
7use crate::base::date::{DateSystem, MONTHS, serial_from_ymd, time_fraction};
8use crate::base::value::{CellError, Value};
9
10/// A parsed entry: its value and the number format Excel would apply automatically.
11#[derive(Clone, Debug, PartialEq)]
12pub struct Parsed {
13    pub value: Value,
14    pub format: Option<&'static str>,
15}
16
17/// Interprets typed cell input (not formulas). Text that isn't a number, date, boolean or error
18/// stays text. A leading apostrophe forces text.
19pub fn parse_input(s: &str, sys: DateSystem) -> Parsed {
20    if s.is_empty() {
21        return Parsed { value: Value::Empty, format: None };
22    }
23    if let Some(rest) = s.strip_prefix('\'') {
24        return Parsed { value: Value::text(rest), format: None };
25    }
26    let t = s.trim();
27    if t.eq_ignore_ascii_case("TRUE") {
28        return Parsed { value: Value::Bool(true), format: None };
29    }
30    if t.eq_ignore_ascii_case("FALSE") {
31        return Parsed { value: Value::Bool(false), format: None };
32    }
33    if let Some(e) = CellError::parse(t) {
34        return Parsed { value: Value::Error(e), format: None };
35    }
36    if let Some((n, f)) = parse_numeric(t, sys) {
37        return Parsed { value: Value::number(n), format: f };
38    }
39    Parsed { value: Value::text(s), format: None }
40}

42/// Numeric text as arithmetic coerces it ("1,000", "50%", "$3", "1/2/2020", "10:00").

52/// Text that Excel's "Number Stored as Text" check flags: it would be a number if typed (signs,
53/// thousands separators, currency, percent, exponent, a parenthesised negative). Dates, times and
54/// fractions kept as text are not flagged, as in Excel.
55pub fn number_stored_as_text(s: &str) -> Option<f64> {
56    let t = s.trim();
57    // The same parentheses typed entry reads, accounting forms included: `$(5)`, `(5)%`.
58    match parenthesized(t) {
59        Some(inner) if inner.strip_prefix('$').map_or(inner.as_str(), str::trim_start).starts_with(['-', '+', '(']) => None,
60        Some(inner) => parse_plain(&inner).map(|(n, _)| -n),
61        None => parse_plain(t).map(|(n, _)| n),
62    }
63}

65fn parse_numeric(t: &str, sys: DateSystem) -> Option<(f64, Option<&'static str>)> {
66    // Parenthesised negatives: (100) = -100, and the accounting forms `$(100)` and `(5)%`.
67    if let Some(inner) = parenthesized(t) {
68        // Excel reads no sign and no nested parentheses inside: `(-5)`, `($-5)`, `((5))`.
69        if inner.strip_prefix('$').map_or(inner.as_str(), str::trim_start).starts_with(['-', '+', '(']) {
70            return None;
71        }
72        let r = parse_plain(&inner).or_else(|| parse_fraction(&inner).map(|n| (n, Some("# ?/?"))));
73        return r.map(|(n, f)| (-n, f));
74    }
75    parse_unparenthesised(t, sys)
76}
77
78fn parse_unparenthesised(t: &str, sys: DateSystem) -> Option<(f64, Option<&'static str>)> {
79    if t.is_empty() {
80        return None;
81    }
82    if let Some(r) = parse_plain(t) {
83        return Some(r);
84    }
85    if let Some(r) = parse_fraction(t) {
86        return Some((r, Some("# ?/?")));
87    }
88    if let Some(r) = parse_datetime(t, sys) {
89        return Some(r);
90    }
91    None
92}

94/// The number inside accounting parentheses, as plain text: `(5)` gives `5`. A dollar sign may
95/// stand before them (`$(5)` reads as `($5)`) and a percent sign after them (`(5)%` reads as
96/// `(5%)`), with spaces around either. The caller reads only a plain number or a fraction from it.
97fn parenthesized(t: &str) -> Option<String> {
98    let (currency, rest) = match t.strip_prefix('$') {
99        Some(r) => ("$", r.trim_start()),
100        None => ("", t),
101    };
102    let rest = rest.strip_prefix('(')?;
103    let (body, percent) = match rest.strip_suffix(')') {
104        Some(b) => (b, ""),
105        None => (rest.strip_suffix('%')?.trim_end().strip_suffix(')')?, "%"),
106    };
107    Some(format!("{currency}{}{percent}", body.trim()))
108}

110/// Signs, currency, thousands separators, percent, exponent.
111fn parse_plain(t: &str) -> Option<(f64, Option<&'static str>)> {
112    let mut s = t;
113    let mut neg = false;
114    let mut currency = false;
115    let mut percent = false;
116    if let Some(r) = s.strip_prefix('-') {
117        neg = true;
118        s = r.trim_start();
119    } else if let Some(r) = s.strip_prefix('+') {
120        s = r.trim_start();
121    }
122    if let Some(r) = s.strip_prefix('$') {
123        currency = true;
124        s = r.trim_start();
125        if !neg && let Some(r2) = s.strip_prefix('-') {
126            neg = true;
127            s = r2;
128        }
129    }
130    if let Some(r) = s.strip_suffix('%') {
131        percent = true;
132        s = r.trim_end();
133    }
134    // Excel reads `$5` and `5%` but not `$5%`.
135    if s.is_empty() || currency && percent {
136        return None;
137    }
138    let has_sep = s.contains(',');
139    let digits: String = if has_sep {
140        // Thousands separators must be in groups of three before the decimal point.
141        let (int, frac) = s.split_once('.').map_or((s, None), |(a, b)| (a, Some(b)));
142        let groups: Vec<&str> = int.split(',').collect();
143        let first = groups.first()?;
144        if first.is_empty() || first.len() > 3 || groups.iter().skip(1).any(|g| g.len() != 3) {
145            return None;
146        }
147        let mut d = groups.concat();
148        if let Some(f) = frac {
149            d.push('.');
150            d.push_str(f);
151        }
152        d
153    } else {
154        s.to_string()
155    };
156    let b = digits.as_bytes();
157    if !b.iter().all(|c| c.is_ascii_digit() || matches!(c, b'.' | b'e' | b'E' | b'+' | b'-')) {
158        return None;
159    }
160    if !b.first().is_some_and(|c| c.is_ascii_digit() || *c == b'.') {
161        return None;
162    }
163    let mut n: f64 = digits.parse().ok()?;
164    if !n.is_finite() {
165        return None;
166    }
167    let sci = digits.contains(['e', 'E']);
168    if percent {
169        n /= 100.0;
170    }
171    if neg {
172        n = -n;
173    }
174    let fmt = if currency {
175        Some(if digits.contains('.') { "\"$\"#,##0.00" } else { "\"$\"#,##0" })
176    } else if percent {
177        Some(if digits.contains('.') { "0.00%" } else { "0%" })
178    } else if sci {
179        Some("0.00E+00")
180    } else if has_sep {
181        Some(if digits.contains('.') { "#,##0.00" } else { "#,##0" })
182    } else {
183        None
184    };
185    Some((n, fmt))
186}

188/// A plain ASCII number: digits, with an optional `.digits` fraction when `frac` is set. Rejects
189/// `nan`, `inf`, signs and exponents that `f64::from_str` would accept, and digits too many for a
190/// finite number (#2163).
191fn plain_number(s: &str, frac: bool) -> Option<f64> {
192    let (int, dec) = match s.split_once('.') {
193        Some((i, d)) if frac => (i, Some(d)),
194        _ => (s, None),
195    };
196    let digits = |x: &str| !x.is_empty() && x.bytes().all(|c| c.is_ascii_digit());
197    if !digits(int) || dec.is_some_and(|d| !digits(d)) {
198        return None;
199    }
200    s.parse().ok().filter(|x: &f64| x.is_finite())
201}

203/// `1/2` (only with a whole part, else it's a date), `1 1/2`, `0 3/4`.
204fn parse_fraction(t: &str) -> Option<f64> {
205    let (whole, frac) = t.split_once(' ')?;
206    let neg = whole.starts_with('-');
207    let w = plain_number(whole.trim_start_matches('-'), false)?;
208    let (a, b) = frac.trim().split_once('/')?;
209    let a = plain_number(a, false)?;
210    let b = plain_number(b, false)?;
211    if b == 0.0 || !w.is_finite() {
212        return None;
213    }
214    let v = w + a / b;
215    if !v.is_finite() {
216        return None;
217    }
218    Some(if neg { -v } else { v })
219}

221fn month_from_name(s: &str) -> Option<u32> {
222    let l = s.to_ascii_lowercase();
223    if l.len() < 3 {
224        return None;
225    }
226    MONTHS.iter().position(|m| m.to_ascii_lowercase().starts_with(&l)).map(|i| i as u32 + 1)
227}

229/// Time part: `h:mm`, `h:mm:ss`, `h:mm:ss.000`, `h AM`, with optional AM/PM.
230fn parse_time(t: &str) -> Option<(f64, bool, bool)> {
231    let t = t.trim();
232    let lower = t.to_ascii_lowercase();
233    let (body, ampm) = if let Some(b) = lower.strip_suffix("am").or_else(|| lower.strip_suffix('a')) {
234        (b.trim().to_string(), Some(false))
235    } else if let Some(b) = lower.strip_suffix("pm").or_else(|| lower.strip_suffix('p')) {
236        (b.trim().to_string(), Some(true))
237    } else {
238        (lower.clone(), None)
239    };
240    let parts: Vec<&str> = body.split(':').collect();
241    if parts.len() == 1 && ampm.is_none() {
242        return None;
243    }
244    if parts.len() > 3 || parts.iter().any(|p| p.is_empty() || p.starts_with('+')) {
245        return None;
246    }
247    let mut h = plain_number(parts.first()?, false)?;
248    let m = parts.get(1).map_or(Some(0.0), |p| plain_number(p, false))?;
249    let s = parts.get(2).map_or(Some(0.0), |p| plain_number(p, true))?;
250    if m >= 60.0 || s >= 60.0 || h < 0.0 || m < 0.0 || s < 0.0 {
251        return None;
252    }
253    if let Some(pm) = ampm {
254        if !(0.0..=12.0).contains(&h) {
255            return None;
256        }
257        if h == 12.0 {
258            h = 0.0;
259        }
260        if pm {
261            h += 12.0;
262        }
263    } else if h >= 10000.0 {
264        return None;
265    }
266    Some((time_fraction(h, m, s), parts.len() == 3, ampm.is_some()))
267}

269fn parse_date(t: &str, sys: DateSystem) -> Option<(f64, &'static str)> {
270    let t = t.trim().trim_end_matches(',');
271    let num = |s: &str| {
272        let s = s.trim();
273        if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
274            return None;
275        }
276        s.parse::<i64>().ok()
277    };
278    // ISO yyyy-mm-dd or yyyy/mm/dd
279    let seps: &[char] = &['/', '-', '.'];
280    let parts: Vec<&str> = t.split(seps).collect();
281    if parts.len() == 3 {
282        let (a, b, c) = (parts[0].trim(), parts[1].trim(), parts[2].trim());
283        if a.len() == 4
284            && let (Some(y), Some(m), Some(d)) = (num(a), num(b), num(c))
285        {
286            return valid_ymd(sys, y, m, d).map(|s| (s, "yyyy-mm-dd"));
287        }
288        if let (Some(m), Some(d), Some(y)) = (num(a), num(b), num(c)) {
289            let y = if c.len() <= 2 { if y < 30 { 2000 + y } else { 1900 + y } } else { y };
290            return valid_ymd(sys, y, m, d).map(|s| (s, "m/d/yyyy"));
291        }
292        // 4-Mar-2026
293        if let (Some(d), Some(m), Some(y)) = (num(a), month_from_name(b), num(c)) {
294            let y = if c.len() <= 2 { if y < 30 { 2000 + y } else { 1900 + y } } else { y };
295            return valid_ymd(sys, y, m as i64, d).map(|s| (s, "d-mmm-yy"));
296        }
297    }
298    if parts.len() == 2 {
299        let (a, b) = (parts[0].trim(), parts[1].trim());
300        // m/d in the current year is ambiguous without a clock; use month/day of 2026? Excel uses
301        // the current year. We accept d-mmm / mmm-d only with month names, and m/d → current year.
302        if let (Some(m), Some(d)) = (num(a), num(b)) {
303            let y = current_year();
304            if (1..=12).contains(&m) && (1..=31).contains(&d) {
305                return valid_ymd(sys, y, m, d).map(|s| (s, "d-mmm"));
306            }
307            // m/yyyy
308            if (1..=12).contains(&m) && b.len() == 4 {
309                return valid_ymd(sys, d, m, 1).map(|s| (s, "mmm-yy"));
310            }
311        }
312        if let (Some(d), Some(m)) = (num(a), month_from_name(b)) {
313            return valid_ymd(sys, current_year(), m as i64, d).map(|s| (s, "d-mmm"));
314        }
315        if let (Some(m), Some(y)) = (month_from_name(a), num(b)) {
316            let y = if b.len() <= 2 { if y < 30 { 2000 + y } else { 1900 + y } } else { y };
317            return valid_ymd(sys, y, m as i64, 1).map(|s| (s, "mmm-yy"));
318        }
319    }
320    // "Mar 4, 2026" / "March 4 2026" / "4 March 2026"
321    let words: Vec<&str> = t.split([' ', ',']).filter(|w| !w.is_empty()).collect();
322    if words.len() == 3 {
323        // Two-digit years follow the same 2029 rule as the dashed forms above.
324        let expand = |w: &str, y: i64| if w.len() <= 2 { if y < 30 { 2000 + y } else { 1900 + y } } else { y };
325        if let (Some(m), Some(d), Some(y)) = (month_from_name(words[0]), num(words[1]), num(words[2])) {
326            let y = expand(words[2], y);
327            return valid_ymd(sys, y, m as i64, d).map(|s| (s, "d-mmm-yy"));
328        }
329        if let (Some(d), Some(m), Some(y)) = (num(words[0]), month_from_name(words[1]), num(words[2])) {
330            let y = expand(words[2], y);
331            return valid_ymd(sys, y, m as i64, d).map(|s| (s, "d-mmm-yy"));
332        }
333    }
334    None
335}

337fn valid_ymd(sys: DateSystem, y: i64, m: i64, d: i64) -> Option<f64> {
338    // Dates before 1900 stay text (`DATE()` would add 1900 to such a year).
339    if !(1..=12).contains(&m) || d < 1 || y < 1900 {
340        return None;
341    }
342    let max = crate::base::date::days_in_month_in(sys, y as i32, m as u32) as i64;
343    if d > max {
344        return None;
345    }
346    serial_from_ymd(sys, y, m, d)
347}

349/// The year used for dates typed without one, from the system clock (browser clock on wasm).
350/// Falls back to 2026 only when no clock is available.
351pub fn current_year() -> i64 {
352    if let Some(ms) = crate::base::date::unix_now_ms() {
353        let days = (ms / 86_400_000.0).floor() as i64;
354        return crate::base::date::civil_from_days(days).0 as i64;
355    }
356    2026
357}

359fn parse_datetime(t: &str, sys: DateSystem) -> Option<(f64, Option<&'static str>)> {
360    if let Some((tf, secs, ampm)) = parse_time(t) {
361        let f = match (secs, ampm) {
362            // 24 hours or more: elapsed hours, so 25:00 shows as 25:00:00, not 1:00.
363            _ if tf >= 1.0 => "[h]:mm:ss",
364            (false, false) => "h:mm",
365            (true, false) => "h:mm:ss",
366            (false, true) => "h:mm AM/PM",
367            (true, true) => "h:mm:ss AM/PM",
368        };
369        return Some((tf, Some(f)));
370    }
371    if let Some((d, f)) = parse_date(t, sys) {
372        return Some((d, Some(f)));
373    }
374    // date + time: split at the last space(s) before a time-looking tail. Only splits where the
375    // tail could be a time are tried: a long tail can't parse, so this stays linear in the text
376    // instead of reparsing every prefix for every space (one long text field quadratic).
377    for (i, _) in t.match_indices(' ') {
378        let Some(b) = t.get(i + 1..) else { continue };
379        let bytes = b.as_bytes();
380        if b.len() > 36 || !bytes.first().is_some_and(u8::is_ascii_digit) || !b.contains(':') {
381            continue;
382        }
383        if let (Some((d, _)), Some((tf, _, _))) = (t.get(..i).and_then(|a| parse_date(a, sys)), parse_time(b)) {
384            return Some((d + tf, Some("m/d/yyyy h:mm")));
385        }
386    }
387    None
388}

551    #[test]
552    fn deeply_nested_parentheses_do_not_overflow() {
553        // #2180: coercion used to recurse once per parenthesis and overflow the stack. Excel reads
554        // one pair only, so nesting is text now, and it is rejected without recursing.
555        let depth = 100_000;
556        let s = format!("{}1{}", "(".repeat(depth), ")".repeat(depth));
557        assert_eq!(parse_number_text(&s), None);
558        assert_eq!(p(&s).value, Value::text(s.as_str()));
559        let odd = format!("{}1{}", "(".repeat(depth + 1), ")".repeat(depth + 1));
560        assert_eq!(parse_number_text(&odd), None);
561        let unbalanced = format!("{}1{}", "(".repeat(depth), ")".repeat(depth - 1));
562        assert_eq!(parse_number_text(&unbalanced), None);
563        // A single pair still reads.
564        assert_eq!(parse_number_text("(100)"), Some(-100.0));
565        assert_eq!(parse_number_text("( $1,200.50 )"), Some(-1200.5));
566        assert_eq!(parse_number_text("()"), None);
567    }
