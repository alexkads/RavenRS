18/// Whether two LET/LAMBDA names are the same: case-insensitive in any script, as in Excel
19/// (`dług` = `DŁUG`, `ß` = `ẞ`). Close to Unicode's default case folding: lower case, plus the
20/// folds lower-casing misses for letters a name may hold (`ß`/`ẞ` -> `ss`, `ſ` -> `s`, `ς` -> `σ`).
21/// Dotless `ı` stays distinct from `i`, as outside Turkic languages.
22pub fn same_name(a: &str, b: &str) -> bool {
23    if a.is_ascii() && b.is_ascii() {
24        return a.eq_ignore_ascii_case(b);
25    }
26    a == b || fold_name(a) == fold_name(b)
27}
28
29/// The case-folded form of a name: two names are [`same_name`] when their folded forms are equal.
30pub fn fold_name(s: &str) -> String {
31    s.chars()
32        .flat_map(char::to_lowercase)
33        .flat_map(|c| match c {
34            'ß' => ['s', 's'].into_iter().take(2),
35            'ſ' => ['s', 's'].into_iter().take(1),
36            'ς' => ['σ', 'σ'].into_iter().take(1),
37            c => [c, c].into_iter().take(1),
38        })
39        .collect()
40}
