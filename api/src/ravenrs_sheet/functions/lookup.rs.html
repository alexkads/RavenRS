17/// Lookup needle: the scalar to look for. Blank looks for 0; errors propagate.
18fn needle(args: &[Arg], i: usize) -> R<Value> {
19    match arg(args, i)?.value.scalar() {
20        Value::Error(e) => Err(e),
21        Value::Empty => Ok(Value::Number(0.0)),
22        v => Ok(v),
23    }
24}

26/// Order of two values for the approximate and binary searches: [`compare`], except that numbers
27/// compare exactly. Excel's `=` treats numbers equal to 15 significant digits as equal, but its
28/// lookups don't. Observed in Excel 365 with A1 holding
29/// `=0.1+0.2` (0.30000000000000004): `A1=0.3` is TRUE, yet `MATCH(0.3,A1,1)` is #N/A because
30/// A1 is larger, and in {0.1;0.2;0.3;A1;0.4} `MATCH(0.3,…,1)` is 3 and `MATCH(A1,…,1)` is 4.
31/// The same holds for `=100+1E-14` against 100, and in every XLOOKUP / XMATCH mode.
32fn lookup_order(a: &Value, b: &Value) -> Ordering {
33    match (a, b) {
34        (Value::Number(x), Value::Number(y)) => x.partial_cmp(y).unwrap_or_else(|| compare(a, b)),
35        _ => compare(a, b),
36    }
37}

39fn same_type(a: &Value, b: &Value) -> bool {
40    matches!((a, b), (Value::Number(_), Value::Number(_)) | (Value::Text(_), Value::Text(_)) | (Value::Bool(_), Value::Bool(_)))
41}

43/// Exact search (first match, or last when `reverse`). Text needles match wildcards when `wild`.
44fn find_exact(x: &Value, hay: &[Value], wild: bool, reverse: bool) -> Option<usize> {
45    let wild = wild && matches!(x, Value::Text(t) if has_wildcards(t));
46    if reverse { hay.iter().rposition(|v| lookup_equal(x, v, wild)) } else { hay.iter().position(|v| lookup_equal(x, v, wild)) }
47}

49/// Approximate match of VLOOKUP, HLOOKUP, LOOKUP and MATCH: the position of the largest entry
50/// `<= x` in data sorted ascending (`desc = false`), or of the smallest entry `>= x` in data
51/// sorted descending. Like Excel this is a bisection over the whole vector, so on unsorted data
52/// it answers what Excel answers rather than a linear "best" entry:
53/// - the probe is the middle entry of the remaining window; a probe whose type differs from the
54///   needle (text vs number vs logical, blanks, errors) is replaced by the nearest entry of the
55///   needle's type below it in the window; when there is none the window moves up past it, so
56///   entries of other types are skipped (`LOOKUP(9.9E+307, range)` finds the last number);
57/// - a probe equal to the needle ends the search at once (the run of equal entries that follows
58///   it is taken, so sorted duplicates give the last one).
59///
60/// Observed in Excel: `VLOOKUP(534, {"INTEGRAL";0;1;2;534;9999999999;-9999999999}, …)` finds 534,
61/// `HLOOKUP("STRING", {"INTEGRAL","DOUBLE","BLANK","STRING","REF","AREA (rows)",<blank>}, …)`
62/// finds "STRING" and `LOOKUP("DOUBLE", {"DOUBLE","BLANK","STRING","REF","AREA (rows)"}, …)`
63/// finds "DOUBLE". Work is linear in the vector length at worst.
64fn bsearch_last(x: &Value, hay: &[Value], desc: bool) -> Option<usize> {
65    let mut hi = hay.len().checked_sub(1)?;
66    let mut lo = 0usize;
67    let mut best = None;
68    while lo <= hi {
69        let mid = lo + (hi - lo) / 2;
70        let Some(j) = (lo..=mid).rev().find(|&j| hay.get(j).is_some_and(|v| same_type(x, v))) else {
71            lo = mid + 1;
72            continue;
73        };
74        let v = hay.get(j)?;
75        let o = lookup_order(v, x);
76        if o == Ordering::Equal {
77            let mut k = j;
78            while hay.get(k + 1).is_some_and(|w| same_type(x, w) && lookup_order(w, x) == Ordering::Equal) {
79                k += 1;
80            }
81            return Some(k);
82        }
83        let before = if desc { o == Ordering::Greater } else { o == Ordering::Less };
84        if before {
85            best = Some(j);
86            lo = mid + 1;
87        } else if j == 0 {
88            break;
89        } else {
90            hi = j - 1;
91        }
92    }
93    best
94}

96/// Order of an entry relative to the needle for XLOOKUP / XMATCH binary search (search mode ±2):
97/// Excel's sort order over all types (numbers < text < logicals < errors), reversed for
98/// descending data, with blanks always last like a sort leaves them. Unlike the legacy
99/// approximate match no entry is skipped (observed: `XLOOKUP(46523, {9700;"xyz";"xyz";160726;
100/// 204100;510300}, …, 0, 1, 2)` returns the row of the first "xyz").
101fn binary_order(v: &Value, x: &Value, desc: bool) -> Ordering {
102    match v {
103        Value::Empty => Ordering::Greater,
104        _ if desc => lookup_order(v, x).reverse(),
105        _ => lookup_order(v, x),
106    }
107}

109/// Binary search for the first entry not ordered before the needle (`>= x` ascending, `<= x`
110/// descending).
111fn bsearch_first(x: &Value, hay: &[Value], desc: bool) -> Option<usize> {
112    let p = hay.partition_point(|v| binary_order(v, x, desc) == Ordering::Less);
113    (p < hay.len()).then_some(p)
114}
115
116/// Binary search for the last entry not ordered after the needle (`<= x` ascending, `>= x`
117/// descending).
118fn bsearch_upto(x: &Value, hay: &[Value], desc: bool) -> Option<usize> {
119    hay.partition_point(|v| binary_order(v, x, desc) != Ordering::Greater).checked_sub(1)
120}

122/// A 1-D view of an array (row or column), `None` when it is 2-D.
123fn vector(a: &Array) -> Option<Vec<Value>> {
124    if a.rows == 1 || a.cols == 1 { Some(a.data.clone()) } else { None }

143fn vh_lookup(args: &[Arg], vertical: bool) -> R<Value> {
144    let x = needle(args, 0)?;
145    let table = as_array(&arg(args, 1)?.value);
146    if let Value::Error(e) = &arg(args, 1)?.value {
147        return Err(*e);
148    }
149    let idx = num(args, 2)?;
150    // NaN too: it would pass `idx < 1.0` and become index 0 (#2163).
151    if idx.is_nan() || idx < 1.0 {
152        return Err(CellError::Value);
153    }
154    let idx = (idx.trunc() as usize).saturating_sub(1);
155    let approx = opt_bool(args, 3, true)?;
156    let (keys, len) = if vertical { (column(&table, 0), table.cols) } else { (row(&table, 0), table.rows) };
157    if idx >= len {
158        return Err(CellError::Ref);
159    }
160    let pos = if approx { bsearch_last(&x, &keys, false) } else { find_exact(&x, &keys, true, false) };
161    let pos = pos.ok_or(CellError::NA)?;
162    let v = if vertical { table.get(pos, idx) } else { table.get(idx, pos) };
163    Ok(cell_result(v.cloned().unwrap_or(Value::Empty)))
164}

174fn lookup(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
175    let x = needle(args, 0)?;
176    let lv = &arg(args, 1)?.value;
177    if let Value::Error(e) = lv {
178        return Err(*e);
179    }
180    let la = as_array(lv);
181    if args.len() >= 3 {
182        let keys = if la.rows == 1 || la.cols == 1 { la.data.clone() } else { column(&la, 0) };
183        let pos = bsearch_last(&x, &keys, false).ok_or(CellError::NA)?;
184        let rv = &arg(args, 2)?.value;
185        if let Value::Error(e) = rv {
186            return Err(*e);
187        }
188        let ra = as_array(rv);
189        let v = if ra.rows == 1 || ra.cols == 1 {
190            ra.data.get(pos).cloned()
191        } else if ra.rows >= keys.len() {
192            ra.get(pos, 0).cloned()
193        } else {
194            None
195        };
196        return v.map(cell_result).ok_or(CellError::NA);
197    }
198    // Array form: search the first row of a wide array, else the first column; return the
199    // matching entry from the last row / column.
200    if la.cols > la.rows {
201        let keys = row(&la, 0);
202        let pos = bsearch_last(&x, &keys, false).ok_or(CellError::NA)?;
203        Ok(cell_result(la.get(la.rows - 1, pos).cloned().unwrap_or(Value::Empty)))
204    } else {
205        let keys = column(&la, 0);
206        let pos = bsearch_last(&x, &keys, false).ok_or(CellError::NA)?;
207        Ok(cell_result(la.get(pos, la.cols - 1).cloned().unwrap_or(Value::Empty)))
208    }
209}

211fn match_fn(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
212    let x = needle(args, 0)?;
213    let av = &arg(args, 1)?.value;
214    if let Value::Error(e) = av {
215        return Err(*e);
216    }
217    let hay = vector(&as_array(av)).ok_or(CellError::NA)?;
218    let mt = opt_num(args, 2, 1.0)?;
219    let pos = if mt == 0.0 {
220        find_exact(&x, &hay, true, false)
221    } else if mt > 0.0 {
222        bsearch_last(&x, &hay, false)
223    } else {
224        bsearch_last(&x, &hay, true)
225    };
226    pos.map(|p| Value::Number((p + 1) as f64)).ok_or(CellError::NA)
227}

232/// Core XMATCH search: 0-based position in `hay`.
233fn xsearch(x: &Value, hay: &[Value], match_mode: i64, search_mode: i64) -> R<Option<usize>> {
234    if !matches!(match_mode, -1..=2) || !matches!(search_mode, -2 | -1 | 1 | 2) {
235        return Err(CellError::Value);
236    }
237    if search_mode.abs() == 2 {
238        let desc = search_mode == -2;
239        if match_mode == 2 {
240            // Wildcards are not supported with binary search: Excel falls back to exact.
241            return Ok(bsearch_exact(x, hay, desc));
242        }
243        return Ok(match match_mode {
244            0 => bsearch_exact(x, hay, desc),
245            // Next smaller: last <= x (ascending) / first <= x (descending).
246            -1 => {
247                if desc {
248                    bsearch_first(x, hay, true)
249                } else {
250                    bsearch_upto(x, hay, false)
251                }
252            }
253            // Next larger: first >= x (ascending) / last >= x (descending).
254            _ => {
255                if desc {
256                    bsearch_upto(x, hay, true)
257                } else {
258                    bsearch_first(x, hay, false)
259                }
260            }
261        });
262    }
263    let reverse = search_mode == -1;
264    if let Some(p) = find_exact(x, hay, match_mode == 2, reverse) {
265        return Ok(Some(p));
266    }
267    if match_mode == 0 || match_mode == 2 {
268        return Ok(None);
269    }
270    // Best candidate: the largest value below (mode -1) or smallest above (mode 1).
271    let order: Box<dyn Iterator<Item = usize>> = if reverse { Box::new((0..hay.len()).rev()) } else { Box::new(0..hay.len()) };
272    let mut best: Option<usize> = None;
273    for i in order {
274        let Some(v) = hay.get(i) else { continue };
275        if !same_type(x, v) {
276            continue;
277        }
278        let o = lookup_order(v, x);
279        let candidate = if match_mode == -1 { o == Ordering::Less } else { o == Ordering::Greater };
280        if !candidate {
281            continue;
282        }
283        let better = match best.and_then(|b| hay.get(b)) {
284            None => true,
285            Some(bv) => {
286                let c = lookup_order(v, bv);
287                if match_mode == -1 { c == Ordering::Greater } else { c == Ordering::Less }
288            }
289        };
290        if better {
291            best = Some(i);
292        }
293    }
294    Ok(best)
295}

297fn bsearch_exact(x: &Value, hay: &[Value], desc: bool) -> Option<usize> {
298    let p = bsearch_first(x, hay, desc)?;
299    let v = hay.get(p)?;
300    if binary_order(v, x, desc) == Ordering::Equal { Some(p) } else { None }
301}

303fn xlookup(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
304    let x = needle(args, 0)?;
305    let lv = &arg(args, 1)?.value;
306    if let Value::Error(e) = lv {
307        return Err(*e);
308    }
309    let rv = &arg(args, 2)?.value;
310    if let Value::Error(e) = rv {
311        return Err(*e);
312    }
313    let la = as_array(lv);
314    let ra = as_array(rv);
315    let hay = vector(&la).ok_or(CellError::Value)?;
316    // Orientation: a column lookup returns rows of the return array; a row lookup returns columns.
317    let vertical = if la.rows == 1 && la.cols == 1 { ra.rows == 1 || ra.cols != 1 } else { la.cols == 1 };
318    if vertical && ra.rows != la.rows || !vertical && ra.cols != la.cols {
319        return Err(CellError::Value);
320    }
321    let mm = if has(args, 4) { opt_int(args, 4, 0)? } else { 0 };
322    let sm = if has(args, 5) { opt_int(args, 5, 1)? } else { 1 };
323    match xsearch(&x, &hay, mm, sm)? {
324        Some(p) => {
325            if vertical {
326                let r = row(&ra, p);
327                if r.len() == 1 {
328                    return Ok(cell_result(r.into_iter().next().unwrap_or(Value::Empty)));
329                }
330                array_val(1, r.len(), r)
331            } else {
332                let c = column(&ra, p);
333                if c.len() == 1 {
334                    return Ok(cell_result(c.into_iter().next().unwrap_or(Value::Empty)));
335                }
336                array_val(c.len(), 1, c)
337            }
338        }
339        None => {
340            if has(args, 3) {
341                Ok(arg(args, 3)?.value.clone())
342            } else {
343                Err(CellError::NA)
344            }
345        }
346    }
347}

349fn xmatch(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
350    let x = needle(args, 0)?;
351    let lv = &arg(args, 1)?.value;
352    if let Value::Error(e) = lv {
353        return Err(*e);
354    }
355    let hay = vector(&as_array(lv)).ok_or(CellError::Value)?;
356    let mm = if has(args, 2) { int(args, 2)? } else { 0 };
357    let sm = if has(args, 3) { int(args, 3)? } else { 1 };
358    xsearch(&x, &hay, mm, sm)?.map(|p| Value::Number((p + 1) as f64)).ok_or(CellError::NA)
359}

364fn index(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
365    let av = &arg(args, 0)?.value;
366    if let Value::Error(e) = av {
367        return Err(*e);
368    }
369    let a = as_array(av);
370    let r = if has(args, 1) { num(args, 1)? } else { 0.0 };
371    let c = if has(args, 2) { num(args, 2)? } else { 0.0 };
372    if has(args, 3) && num(args, 3)?.trunc() != 1.0 {
373        return Err(CellError::Ref);
374    }
375    if r < 0.0 || c < 0.0 {
376        return Err(CellError::Value);
377    }
378    let (mut r, mut c) = (r.trunc() as usize, c.trunc() as usize);
379    // A single row with only one index: the index selects the column.
380    if a.rows == 1 && args.len() == 2 && a.cols > 1 {
381        c = r;
382        r = 1;
383    }
384    if r > a.rows || c > a.cols {
385        return Err(CellError::Ref);
386    }
387    match (r, c) {
388        (0, 0) => Ok(Value::Array(std::sync::Arc::new(a.into_owned()))),
389        (0, c) => {
390            let v = column(&a, c - 1);
391            if v.len() == 1 {
392                return Ok(v.into_iter().next().unwrap_or(Value::Empty));
393            }
394            array_val(v.len(), 1, v)
395        }
396        (r, 0) => {
397            let v = row(&a, r - 1);
398            if v.len() == 1 {
399                return Ok(v.into_iter().next().unwrap_or(Value::Empty));
400            }
401            array_val(1, v.len(), v)
402        }
403        (r, c) => a.get(r - 1, c - 1).cloned().ok_or(CellError::Ref),
404    }
405}

430/// Sheet text for ADDRESS, quoted when Excel quotes it. A `[Book]Sheet` prefix is judged by its
431/// parts (Microsoft's ADDRESS examples: `"[Book1]Sheet1"` stays bare, `"EXCEL SHEET"` becomes
432/// `'EXCEL SHEET'`). Unlike in a formula, `TRUE` and `FALSE` stay bare.
433fn quote_sheet(s: &str) -> String {
434    let bare = |p: &str| p.eq_ignore_ascii_case("TRUE") || p.eq_ignore_ascii_case("FALSE") || !crate::base::sheet_needs_quotes(p);
435    let word = |p: &str| !p.is_empty() && p.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '.');
436    let plain = match s.strip_prefix('[').and_then(|r| r.split_once(']')) {
437        Some((book, sheet)) => word(book) && bare(sheet),
438        None => bare(s),
439    };
440    if plain { s.to_string() } else { format!("'{}'", s.replace('\'', "''")) }
441}

443fn address(args: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
444    let r = num(args, 0)?.trunc();
445    let c = num(args, 1)?.trunc();
446    let abs = if has(args, 2) { int(args, 2)? } else { 1 };
447    let a1 = if has(args, 3) { opt_bool(args, 3, true)? } else { true };
448    if !(1.0..=1_048_576.0).contains(&r) || !(1.0..=16_384.0).contains(&c) || !(1..=4).contains(&abs) {
449        return Err(CellError::Value);
450    }
451    let (ri, ci) = (r as u32, c as u32);
452    let (row_abs, col_abs) = match abs {
453        1 => (true, true),
454        2 => (true, false),
455        3 => (false, true),
456        _ => (false, false),
457    };
458    let cell = if a1 {
459        format!("{}{}{}{}", if col_abs { "$" } else { "" }, col_to_letters(ci - 1), if row_abs { "$" } else { "" }, ri)
460    } else {
461        let rp = if row_abs { format!("R{ri}") } else { format!("R[{ri}]") };
462        let cp = if col_abs { format!("C{ci}") } else { format!("C[{ci}]") };
463        format!("{rp}{cp}")
464    };
465    if has(args, 4) {
466        let sheet = text(args, 4)?;
467        if sheet.is_empty() {
468            return text_val(format!("!{cell}"));
469        }
470        return text_val(format!("{}!{cell}", quote_sheet(&sheet)));
471    }
472    text_val(cell)
473}
