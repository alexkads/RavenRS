7const LANCZOS_G: f64 = 7.0;
8const LANCZOS: [f64; 9] = [
9    0.999_999_999_999_809_9,
10    676.520_368_121_885_1,
11    -1_259.139_216_722_402_8,
12    771.323_428_777_653_1,
13    -176.615_029_162_140_6,
14    12.507_343_278_686_905,
15    -0.138_571_095_265_720_12,
16    9.984_369_578_019_572e-6,
17    1.505_632_735_149_311_6e-7,
18];

20/// ln |Γ(x)|. NaN at non-positive integers.
21pub(crate) fn ln_gamma(x: f64) -> f64 {
22    if !x.is_finite() {
23        return f64::NAN;
24    }
25    if x <= 0.0 && x == x.floor() {
26        return f64::NAN;
27    }
28    if x < 0.5 {
29        // Reflection.
30        let s = (PI * x).sin().abs();
31        return (PI / s).ln() - ln_gamma(1.0 - x);
32    }
33    if x < 15.0 && x == x.floor() {
34        // Exact for small integers.
35        let mut f = 1.0f64;
36        let mut i = 2.0;
37        while i < x {
38            f *= i;
39            i += 1.0;
40        }
41        return f.ln();
42    }
43    let x = x - 1.0;
44    let mut a = LANCZOS[0];
45    let t = x + LANCZOS_G + 0.5;
46    for (i, c) in LANCZOS.iter().enumerate().skip(1) {
47        a += c / (x + i as f64);
48    }
49    0.5 * (2.0 * PI).ln() + (x + 0.5) * t.ln() - t + a.ln()
50}

52/// Γ(x). NaN at non-positive integers, infinite on overflow.
53pub(crate) fn gamma(x: f64) -> f64 {
54    if !x.is_finite() || (x <= 0.0 && x == x.floor()) {
55        return f64::NAN;
56    }
57    if x == x.floor() && x <= 171.0 {
58        let mut f = 1.0f64;
59        let mut i = 2.0;
60        while i < x {
61            f *= i;
62            i += 1.0;
63        }
64        return f;
65    }
66    if x < 0.5 {
67        return PI / ((PI * x).sin() * gamma(1.0 - x));
68    }
69    if x > 171.7 {
70        return f64::INFINITY;
71    }
72    ln_gamma(x).exp()
73}

75/// ln B(a, b).
76pub(crate) fn ln_beta(a: f64, b: f64) -> f64 {
77    ln_gamma(a) + ln_gamma(b) - ln_gamma(a + b)
78}

80/// Regularised lower incomplete gamma P(a, x).
81pub(crate) fn reg_gamma_p(a: f64, x: f64) -> f64 {
82    if a.is_nan() || a <= 0.0 || x.is_nan() || x < 0.0 || !a.is_finite() {
83        return f64::NAN;
84    }
85    if x == 0.0 {
86        return 0.0;
87    }
88    if x.is_infinite() {
89        return 1.0;
90    }
91    if x < a + 1.0 { gamma_series(a, x) } else { 1.0 - gamma_cf(a, x) }
92}

94/// Regularised upper incomplete gamma Q(a, x) = 1 − P(a, x).
95pub(crate) fn reg_gamma_q(a: f64, x: f64) -> f64 {
96    if a.is_nan() || a <= 0.0 || x.is_nan() || x < 0.0 || !a.is_finite() {
97        return f64::NAN;
98    }
99    if x == 0.0 {
100        return 1.0;
101    }
102    if x.is_infinite() {
103        return 0.0;
104    }
105    if x < a + 1.0 { 1.0 - gamma_series(a, x) } else { gamma_cf(a, x) }
106}

108fn gamma_series(a: f64, x: f64) -> f64 {
109    let mut ap = a;
110    let mut sum = 1.0 / a;
111    let mut del = sum;
112    for _ in 0..10_000 {
113        ap += 1.0;
114        del *= x / ap;
115        sum += del;
116        if del.abs() < sum.abs() * 1e-17 {
117            break;
118        }
119    }
120    sum * (-x + a * x.ln() - ln_gamma(a)).exp()
121}

123fn gamma_cf(a: f64, x: f64) -> f64 {
124    let tiny = 1e-300;
125    let mut b = x + 1.0 - a;
126    let mut c = 1.0 / tiny;
127    let mut d = 1.0 / b;
128    let mut h = d;
129    for i in 1..10_000 {
130        let an = -(i as f64) * (i as f64 - a);
131        b += 2.0;
132        d = an * d + b;
133        if d.abs() < tiny {
134            d = tiny;
135        }
136        c = b + an / c;
137        if c.abs() < tiny {
138            c = tiny;
139        }
140        d = 1.0 / d;
141        let del = d * c;
142        h *= del;
143        if (del - 1.0).abs() < 1e-17 {
144            break;
145        }
146    }
147    (-x + a * x.ln() - ln_gamma(a)).exp() * h
148}

150/// Error function.
151pub(crate) fn erf(x: f64) -> f64 {
152    if x.is_nan() {
153        return f64::NAN;
154    }
155    if x == 0.0 {
156        return 0.0;
157    }
158    if x.abs() < 0.5 {
159        // Taylor series: erf x = 2/√π Σ (-1)^n x^(2n+1) / (n! (2n+1)).
160        let mut sum = x;
161        let mut term = x;
162        let x2 = x * x;
163        for n in 1..60 {
164            term *= -x2 / n as f64;
165            let add = term / (2 * n + 1) as f64;
166            sum += add;
167            if add.abs() < 1e-18 * sum.abs() {
168                break;
169            }
170        }
171        return sum * 2.0 / PI.sqrt();
172    }
173    let p = reg_gamma_p(0.5, x * x);
174    if x < 0.0 { -p } else { p }
175}

177/// Complementary error function, accurate in the tails.
178pub(crate) fn erfc(x: f64) -> f64 {
179    if x.is_nan() {
180        return f64::NAN;
181    }
182    if x.abs() < 0.5 {
183        return 1.0 - erf(x);
184    }
185    if x > 0.0 { reg_gamma_q(0.5, x * x) } else { 1.0 + reg_gamma_p(0.5, x * x) }
186}

188/// Regularised incomplete beta I_x(a, b).
189pub(crate) fn reg_inc_beta(x: f64, a: f64, b: f64) -> f64 {
190    if a.is_nan() || a <= 0.0 || b.is_nan() || b <= 0.0 || !(0.0..=1.0).contains(&x) {
191        return f64::NAN;
192    }
193    if x == 0.0 {
194        return 0.0;
195    }
196    if x == 1.0 {
197        return 1.0;
198    }
199    let ln_bt = -ln_beta(a, b) + a * x.ln() + b * (1.0 - x).ln();
200    let bt = ln_bt.exp();
201    if x < (a + 1.0) / (a + b + 2.0) { bt * beta_cf(a, b, x) / a } else { 1.0 - bt * beta_cf(b, a, 1.0 - x) / b }
202}

204fn beta_cf(a: f64, b: f64, x: f64) -> f64 {
205    let tiny = 1e-300;
206    let qab = a + b;
207    let qap = a + 1.0;
208    let qam = a - 1.0;
209    let mut c = 1.0;
210    let mut d = 1.0 - qab * x / qap;
211    if d.abs() < tiny {
212        d = tiny;
213    }
214    d = 1.0 / d;
215    let mut h = d;
216    for m in 1..10_000 {
217        let m = m as f64;
218        let m2 = 2.0 * m;
219        let aa = m * (b - m) * x / ((qam + m2) * (a + m2));
220        d = 1.0 + aa * d;
221        if d.abs() < tiny {
222            d = tiny;
223        }
224        c = 1.0 + aa / c;
225        if c.abs() < tiny {
226            c = tiny;
227        }
228        d = 1.0 / d;
229        h *= d * c;
230        let aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2));
231        d = 1.0 + aa * d;
232        if d.abs() < tiny {
233            d = tiny;
234        }
235        c = 1.0 + aa / c;
236        if c.abs() < tiny {
237            c = tiny;
238        }
239        d = 1.0 / d;
240        let del = d * c;
241        h *= del;
242        if (del - 1.0).abs() < 1e-16 {
243            break;
244        }
245    }
246    h
247}

249/// Standard normal density.
250pub(crate) fn norm_pdf(z: f64) -> f64 {
251    (-0.5 * z * z).exp() / (2.0 * PI).sqrt()
252}
253
254/// Standard normal cumulative distribution.
255pub(crate) fn norm_cdf(z: f64) -> f64 {
256    0.5 * erfc(-z / std::f64::consts::SQRT_2)
257}

259/// Inverse of the standard normal CDF (NaN outside (0, 1)).
260pub(crate) fn norm_inv(p: f64) -> f64 {
261    if !(p > 0.0 && p < 1.0) {
262        return f64::NAN;
263    }
264    // Acklam's rational approximation, then one Halley refinement step.
265    const A: [f64; 6] = [
266        -3.969_683_028_665_376e1,
267        2.209_460_984_245_205e2,
268        -2.759_285_104_469_687e2,
269        1.383_577_518_672_69e2,
270        -3.066_479_806_614_716e1,
271        2.506_628_277_459_239,
272    ];
273    const B: [f64; 5] =
274        [-5.447_609_879_822_406e1, 1.615_858_368_580_409e2, -1.556_989_798_598_866e2, 6.680_131_188_771_972e1, -1.328_068_155_288_572e1];
275    const C: [f64; 6] = [
276        -7.784_894_002_430_293e-3,
277        -3.223_964_580_411_365e-1,
278        -2.400_758_277_161_838,
279        -2.549_732_539_343_734,
280        4.374_664_141_464_968,
281        2.938_163_982_698_783,
282    ];
283    const D: [f64; 4] = [7.784_695_709_041_462e-3, 3.224_671_290_700_398e-1, 2.445_134_137_142_996, 3.754_408_661_907_416];
284    let plow = 0.02425;
285    let x = if p < plow {
286        let q = (-2.0 * p.ln()).sqrt();
287        (((((C[0] * q + C[1]) * q + C[2]) * q + C[3]) * q + C[4]) * q + C[5]) / ((((D[0] * q + D[1]) * q + D[2]) * q + D[3]) * q + 1.0)
288    } else if p <= 1.0 - plow {
289        let q = p - 0.5;
290        let r = q * q;
291        (((((A[0] * r + A[1]) * r + A[2]) * r + A[3]) * r + A[4]) * r + A[5]) * q
292            / (((((B[0] * r + B[1]) * r + B[2]) * r + B[3]) * r + B[4]) * r + 1.0)
293    } else {
294        let q = (-2.0 * (1.0 - p).ln()).sqrt();
295        -(((((C[0] * q + C[1]) * q + C[2]) * q + C[3]) * q + C[4]) * q + C[5]) / ((((D[0] * q + D[1]) * q + D[2]) * q + D[3]) * q + 1.0)
296    };
297    let mut x = x;
298    for _ in 0..2 {
299        let e = norm_cdf(x) - p;
300        let u = e * (2.0 * PI).sqrt() * (x * x / 2.0).exp();
301        let nx = x - u / (1.0 + x * u / 2.0);
302        if !nx.is_finite() {
303            break;
304        }
305        x = nx;
306    }
307    x
308}

310/// Finds `x` in `[lo, hi]` with `f(x) = target` for a monotone `f` by bisection (expanding `hi`
311/// up to `hi_max` while the target is not bracketed). Returns NaN when no root is bracketed.
312pub(crate) fn invert(f: impl Fn(f64) -> f64, target: f64, lo: f64, hi: f64, hi_max: f64) -> f64 {
313    let (mut lo, mut hi) = (lo, hi);
314    let flo = f(lo) - target;
315    let mut fhi = f(hi) - target;
316    let mut guard = 0;
317    while flo * fhi > 0.0 && hi < hi_max && guard < 200 {
318        lo = hi;
319        hi = (hi * 2.0).min(hi_max);
320        fhi = f(hi) - target;
321        guard += 1;
322    }
323    let flo = f(lo) - target;
324    if flo == 0.0 {
325        return lo;
326    }
327    if fhi == 0.0 {
328        return hi;
329    }
330    if flo * fhi > 0.0 || flo.is_nan() || fhi.is_nan() {
331        return f64::NAN;
332    }
333    let lo_neg = flo < 0.0;
334    for _ in 0..300 {
335        let mid = 0.5 * (lo + hi);
336        if mid <= lo || mid >= hi {
337            break;
338        }
339        let fm = f(mid) - target;
340        if fm == 0.0 {
341            return mid;
342        }
343        if (fm < 0.0) == lo_neg {
344            lo = mid;
345        } else {
346            hi = mid;
347        }
348        if (hi - lo).abs() <= 1e-15 * mid.abs().max(1e-300) {
349            break;
350        }
351    }
352    0.5 * (lo + hi)
353}
