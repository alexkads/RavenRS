1//! Database functions (DSUM, DCOUNT, DGET…): aggregate one field of the records of a table
2//! that satisfy a criteria table.
3//!
4//! The database and the criteria are arrays whose first row holds column labels. Each criteria
5//! row is a set of conditions that must all hold (AND); a record matches when any criteria row
6//! holds (OR). An empty criteria cell matches anything. Plain text conditions match values that
7//! begin with the text, as in Excel's advanced filter, so an empty string matches any text. A
8//! formula under a label that names no field is a computed criterion, which the host evaluates
9//! for each record ([`Ctx::computed_criterion`]).
10

19/// Column index of a label in the header row.
20fn find_column(db: &Array, label: &str) -> Option<usize> {
21    let label = label.trim();
22    (0..db.cols).find(|&c| match db.get(0, c) {
23        Some(Value::Text(t)) => compare_text(t.trim(), label).is_eq(),
24        Some(Value::Number(n)) => crate::base::number_to_text(*n) == label,
25        _ => false,
26    })
27}
28
29/// Resolves the field argument: a label or a 1-based column number. `None` when omitted.
30fn field_index(db: &Array, a: &[Arg]) -> R<Option<usize>> {
31    if !has(a, 1) {
32        return Ok(None);
33    }
34    let v = a.get(1).map(|x| x.value.scalar()).unwrap_or(Value::Empty);
35    match v {
36        Value::Number(n) => {
37            let i = n.trunc();
38            // NaN too: it would pass both comparisons and become index 0 (#2163).
39            if i.is_nan() || i < 1.0 || i > db.cols as f64 {
40                return Err(CellError::Value);
41            }
42            Ok(Some((i as usize).saturating_sub(1)))
43        }
44        Value::Text(t) => find_column(db, &t).map(Some).ok_or(CellError::Value),
45        Value::Bool(_) => Err(CellError::Value),
46        Value::Error(e) => Err(e),
47        _ => Ok(None),
48    }
49}

51/// One criteria cell as a matcher. Plain text becomes a "begins with" pattern.
52fn cell_criterion(v: &Value) -> Option<Criterion> {
53    match v {
54        Value::Empty => None,
55        Value::Text(t) => {
56            let s: &str = t;
57            let has_op = s.starts_with(['=', '<', '>']);
58            let plain_text = !has_op
59                && crate::base::parse::parse_number_text(s).is_none()
60                && !s.eq_ignore_ascii_case("TRUE")
61                && !s.eq_ignore_ascii_case("FALSE")
62                && CellError::parse(s).is_none();
63            if plain_text && !s.ends_with('*') { Some(Criterion::parse(&Value::from(format!("{s}*")))) } else { Some(Criterion::parse(v)) }
64        }
65        other => Some(Criterion::parse(other)),
66    }
67}

69/// A criteria row: its index and its conditions (criteria column, database column or `None`
70/// when the label is unknown, criterion).
71type CriteriaRow = (usize, Vec<(usize, Option<usize>, Criterion)>);
72
73/// Rows (1-based within the database, excluding the header) of matching records.
74fn matching_records(db: &Array, crit: &Array, ctx: &mut dyn Ctx) -> Vec<usize> {
75    // Per criteria row: its index and a list of (criteria column, database column or None when
76    // the label is unknown, criterion).
77    let mut rows: Vec<CriteriaRow> = Vec::new();
78    for r in 1..crit.rows {
79        let mut conds = Vec::new();
80        for c in 0..crit.cols {
81            let Some(cell) = crit.get(r, c) else { continue };
82            let Some(cr) = cell_criterion(cell) else { continue };
83            let col = match crit.get(0, c) {
84                Some(Value::Text(t)) => find_column(db, t),
85                Some(Value::Number(n)) => find_column(db, &crate::base::number_to_text(*n)),
86                _ => None,
87            };
88            conds.push((c, col, cr));
89        }
90        rows.push((r, conds));
91    }
92    let all_match = rows.is_empty();
93    (1..db.rows)
94        .filter(|&rec| {
95            all_match
96                || rows.iter().any(|(r, conds)| {
97                    conds.iter().all(|(c, col, cr)| match col {
98                        Some(f) => cr.matches(db.get(rec, *f).unwrap_or(&Value::Empty)),
99                        // Under a label that names no field, only a formula (a computed
100                        // criterion) can hold.
101                        None => ctx.computed_criterion(*r, *c, db, rec).unwrap_or(false),
102                    })
103                })
104        })
105        .collect()
106}

108struct Selection<'a> {
109    db: Cow<'a, Array>,
110    field: Option<usize>,
111    records: Vec<usize>,
112}
113
114fn select<'a>(a: &'a [Arg], ctx: &mut dyn Ctx) -> R<Selection<'a>> {
115    let dbv = &a.first().ok_or(CellError::Value)?.value;
116    let critv = &a.get(2).ok_or(CellError::Value)?.value;
117    if let Value::Error(e) = dbv {
118        return Err(*e);
119    }
120    if let Value::Error(e) = critv {
121        return Err(*e);
122    }
123    let db = as_array(dbv);
124    let crit = as_array(critv);
125    let field = field_index(&db, a)?;
126    let records = matching_records(&db, &crit, ctx);
127    Ok(Selection { db, field, records })
128}

130/// Numbers of the field among matching records (field required).
131fn field_numbers(a: &[Arg], c: &mut dyn Ctx) -> R<Vec<f64>> {
132    let s = select(a, c)?;
133    let f = s.field.ok_or(CellError::Value)?;
134    Ok(s.records.iter().filter_map(|&r| s.db.get(r, f).and_then(Value::as_f64)).collect())
135}

137fn dsum(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
138    num_val(field_numbers(a, c)?.iter().sum())
139}
140
141fn daverage(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
142    let v = field_numbers(a, c)?;
143    if v.is_empty() {
144        return Err(CellError::Div0);
145    }
146    num_val(v.iter().sum::<f64>() / v.len() as f64)
147}
148
149fn dcount(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
150    let s = select(a, c)?;
151    let n = match s.field {
152        Some(f) => s.records.iter().filter(|&&r| matches!(s.db.get(r, f), Some(Value::Number(_)))).count(),
153        None => s.records.len(),
154    };
155    num_val(n as f64)
156}
157
158fn dcounta(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
159    let s = select(a, c)?;
160    let n = match s.field {
161        Some(f) => s.records.iter().filter(|&&r| !matches!(s.db.get(r, f), None | Some(Value::Empty))).count(),
162        None => s.records.len(),
163    };
164    num_val(n as f64)
165}
166
167fn dget(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
168    let s = select(a, c)?;
169    let f = s.field.ok_or(CellError::Value)?;
170    // Records whose field is blank (or "") don't count, as in Excel.
171    let mut found =
172        s.records.iter().filter_map(|&r| s.db.get(r, f)).filter(|v| !matches!(v, Value::Empty) && !matches!(v, Value::Text(t) if t.is_empty()));
173    match (found.next(), found.next()) {
174        (None, _) => Err(CellError::Value),
175        (Some(v), None) => Ok(v.clone()),
176        _ => Err(CellError::Num),
177    }
178}

180fn dmax(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
181    let v = field_numbers(a, c)?;
182    num_val(v.iter().copied().reduce(f64::max).unwrap_or(0.0))
183}
184
185fn dmin(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
186    let v = field_numbers(a, c)?;
187    num_val(v.iter().copied().reduce(f64::min).unwrap_or(0.0))
188}
189
190fn dproduct(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
191    let v = field_numbers(a, c)?;
192    if v.is_empty() {
193        return num_val(0.0);
194    }
195    num_val(v.iter().product())
196}

198fn variance(v: &[f64], sample: bool) -> R<f64> {
199    let n = v.len() as f64;
200    let min = if sample { 2.0 } else { 1.0 };
201    if n < min {
202        return Err(CellError::Div0);
203    }
204    let mean = v.iter().sum::<f64>() / n;
205    let ss: f64 = v.iter().map(|x| (x - mean) * (x - mean)).sum();
206    Ok(ss / if sample { n - 1.0 } else { n })
207}
208
209fn dvar(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
210    num_val(variance(&field_numbers(a, c)?, true)?)
211}
212
213fn dvarp(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
214    num_val(variance(&field_numbers(a, c)?, false)?)
215}
216
217fn dstdev(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
218    num_val(variance(&field_numbers(a, c)?, true)?.sqrt())
219}
220
221fn dstdevp(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
222    num_val(variance(&field_numbers(a, c)?, false)?.sqrt())
223}
