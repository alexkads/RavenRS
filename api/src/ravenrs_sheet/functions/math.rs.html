25const fn alt_table(first_crit: usize) -> [bool; 256] {
26    // Criteria positions: first_crit, first_crit + 2, …
27    let mut t = [false; 256];
28    let mut i = first_crit;
29    while i < 256 {
30        t[i] = true;
31        i += 2;
32    }
33    t
34}
35/// COUNTIFS(range, crit, range, crit, …)
36static COUNTIFS_SCAL: [bool; 256] = alt_table(1);
37/// SUMIFS(sum_range, range, crit, …)
38static SUMIFS_SCAL: [bool; 256] = alt_table(2);
39const IF2: &[bool] = &[false, true, false];

68/// Rounds |x| scaled by 10^digits with `op` (ceil / floor), keeping sign.
69fn round_dir(x: f64, digits: i32, up: bool) -> f64 {
70    if x == 0.0 || !x.is_finite() {
71        return x;
72    }
73    let ax = clean(x.abs());
74    let (scaled, f, neg) = if digits >= 0 {
75        let f = 10f64.powi(digits);
76        (ax * f, f, false)
77    } else {
78        let f = 10f64.powi(-digits);
79        (ax / f, f, true)
80    };
81    if !scaled.is_finite() || scaled >= 1e17 {
82        return x;
83    }
84    let s = clean(scaled);
85    let r = if up { s.ceil() } else { s.floor() };
86    // Toward zero, r == 0 with an infinite scale would be 0 * inf = NaN (#NUM!); the result is simply
87    // zero. Away from zero the true result (10^309 or more) is not a number, so ROUNDUP stays #NUM!.
88    let out = if !neg {
89        clean(r / f)
90    } else if r == 0.0 && !up {
91        0.0
92    } else {
93        r * f
94    };
95    if x < 0.0 { -out } else { out }
96}

128/// SUMPRODUCT: non-numeric entries of arrays and multi-cell ranges count as 0, but a scalar
129/// argument holding text (a literal or a single-cell reference) is `#VALUE!` (observed:
130/// `SUMPRODUCT(G1330:G1330,I1330:I1330)` with G1330 = the text "4" is `#VALUE!`).
131fn sumproduct(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
132    let arrays: Vec<_> = a.iter().map(|x| (as_array(&x.value), !matches!(x.value, Value::Array(_)))).collect();
133    let Some((first, _)) = arrays.first() else { return Err(CellError::Value) };
134    let (rows, cols) = (first.rows, first.cols);
135    if arrays.iter().any(|(x, _)| x.rows != rows || x.cols != cols) {
136        return Err(CellError::Value);
137    }
138    let mut total = 0.0;
139    for i in 0..rows.saturating_mul(cols) {
140        let mut p = 1.0;
141        for (arr, scalar) in &arrays {
142            match arr.data.get(i) {
143                Some(Value::Number(n)) => p *= n,
144                Some(Value::Error(e)) => return Err(*e),
145                Some(Value::Text(_)) if *scalar => return Err(CellError::Value),
146                _ => p = 0.0,
147            }
148        }
149        total += p;
150    }
151    num_val(total)
152}

154fn pair_sum(a: &[Arg], f: fn(f64, f64) -> f64) -> R<Value> {
155    let x = as_array(&crate::functions::util::arg(a, 0)?.value);
156    let y = as_array(&crate::functions::util::arg(a, 1)?.value);
157    if x.data.len() != y.data.len() {
158        return Err(CellError::NA);
159    }
160    let mut total = 0.0;
161    let mut any = false;
162    for (u, v) in x.data.iter().zip(y.data.iter()) {
163        if let Value::Error(e) = u {
164            return Err(*e);
165        }
166        if let Value::Error(e) = v {
167            return Err(*e);
168        }
169        if let (Value::Number(p), Value::Number(q)) = (u, v) {
170            total += f(*p, *q);
171            any = true;
172        }
173    }
174    if !any {
175        return Err(CellError::Div0);
176    }
177    num_val(total)
178}

248/// CEILING.MATH / FLOOR.MATH / *.PRECISE: `up` picks the direction for positive numbers;
249/// negative numbers round away from zero when `away`, toward zero otherwise.
250fn ceil_floor_math(x: f64, s: f64, up: bool, away: bool) -> R<Value> {
251    let s = s.abs();
252    if s == 0.0 || x == 0.0 {
253        return Ok(Value::Number(0.0));
254    }
255    let q = clean(x / s);
256    let r = if x >= 0.0 {
257        if up { q.ceil() } else { q.floor() }
258    } else if away {
259        q.floor()
260    } else {
261        q.ceil()
262    };
263    num_val(clean(r * s))
264}

290fn even_odd(x: f64, odd: bool) -> f64 {
291    let sign = if x < 0.0 { -1.0 } else { 1.0 };
292    let ax = clean(x.abs());
293    let mut v = ax.ceil();
294    if odd {
295        if v % 2.0 == 0.0 {
296            v += 1.0;
297        }
298    } else if v % 2.0 != 0.0 {
299        v += 1.0;
300    }
301    sign * v
302}

313fn modulo(x: f64, d: f64) -> R<f64> {
314    if d == 0.0 {
315        return Err(CellError::Div0);
316    }
317    // Excel returns #NUM! once the quotient magnitude reaches 1.1259E12 (checked against Excel 16).
318    if (x / d).abs() >= 1.1259e12 {
319        return Err(CellError::Num);
320    }
321    // `%` is an exact remainder (truncated), so |r| < |d| even for tiny divisors.
322    let r = x % d;
323    // Keep the sign of the divisor and avoid returning |d| from rounding.
324    let r = if r != 0.0 && (r < 0.0) != (d < 0.0) { r + d } else { r };
325    if r.abs() >= d.abs() {
326        return Ok(0.0);
327    }
328    Ok(clean(r))
329}

358fn power_of(x: f64, y: f64) -> R<f64> {
359    if x == 0.0 {

369        return Err(CellError::Num);
370    }
371    let r = x.powf(y);

373}

424fn fact_of(n: f64) -> R<f64> {
425    let n = n.trunc();
426    if n < 0.0 {
427        return Err(CellError::Num);
428    }
429    if n > 170.0 {
430        return Err(CellError::Num);
431    }
432    let mut f = 1.0f64;
433    let mut i = 2.0;
434    while i <= n {
435        f *= i;
436        i += 1.0;
437    }
438    Ok(f)
439}

460fn combin_of(n: f64, k: f64) -> R<f64> {
461    let (n, k) = (n.trunc(), k.trunc());
462    if n < 0.0 || k < 0.0 || k > n {
463        return Err(CellError::Num);
464    }
465    let k = k.min(n - k);
466    if k > 1e7 {
467        let r = (crate::functions::special::ln_gamma(n + 1.0)
468            - crate::functions::special::ln_gamma(k + 1.0)
469            - crate::functions::special::ln_gamma(n - k + 1.0))
470        .exp();
471        return if r.is_finite() { Ok(r) } else { Err(CellError::Num) };
472    }
473    let mut r = 1.0f64;
474    let mut i = 1.0;
475    while i <= k {
476        r = r * (n - k + i) / i;
477        if !r.is_finite() {
478            return Err(CellError::Num);
479        }
480        i += 1.0;
481    }
482    Ok(if r < 9e15 { r.round() } else { r })
483}

530fn gcd2(mut a: u64, mut b: u64) -> u64 {
531    while b != 0 {
532        let t = a % b;
533        a = b;
534        b = t;
535    }
536    a
537}

542fn lcm(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
543    let v = integer_args(a)?;
544    let mut acc = 1f64;
545    for &x in &v {
546        if x == 0 {
547            return Ok(Value::Number(0.0));
548        }
549        let accu = acc as u64;
550        let g = gcd2(accu, x);
551        acc = acc / g as f64 * x as f64;
552        if acc >= 9.007_199_254_740_992e15 {
553            return Err(CellError::Num);
554        }
555    }
556    num_val(acc)
557}
558fn multinomial(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
559    let v = numbers(a)?;
560    let mut total = 0.0;
561    let mut r = 1.0f64;
562    for x in v {
563        let k = x.trunc();
564        if k < 0.0 {
565            return Err(CellError::Num);
566        }
567        total += k;
568        r *= combin_of(total, k)?;
569        if !r.is_finite() {
570            return Err(CellError::Num);
571        }
572    }
573    num_val(r)
574}

597fn atan2(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
598    let (x, y) = (num(a, 0)?, num(a, 1)?);
599    if x == 0.0 && y == 0.0 {
600        return Err(CellError::Div0);
601    }
602    num_val(y.atan2(x))
603}

642}
643fn acoth(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
644    unary(a, |x| if x.abs() <= 1.0 { Err(CellError::Num) } else { Ok(0.5 * ((x + 1.0) / (x - 1.0)).ln()) })
645}

647    unary(a, |x| Ok(x * 180.0 / PI))
648}

650    unary(a, |x| Ok(x * PI / 180.0))
651}

653fn seriessum(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
654    let (x, n, m) = (num(a, 0)?, num(a, 1)?, num(a, 2)?);
655    let coeffs = as_array(&crate::functions::util::arg(a, 3)?.value);
656    let mut total = 0.0;
657    for (i, c) in coeffs.iter().enumerate() {
658        let c = match c {
659            Value::Number(v) => *v,
660            Value::Error(e) => return Err(*e),
661            _ => return Err(CellError::Value),
662        };
663        total += c * power_of(x, n + i as f64 * m)?;
664    }
665    num_val(total)
666}
