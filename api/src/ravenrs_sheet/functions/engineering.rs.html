15/// Source text of a base-conversion argument (numbers are written out as integers).
16fn base_source(a: &[Arg]) -> R<String> {
17    let v = a.first().map(scalar).unwrap_or(Value::Empty);
18    match v {
19        Value::Error(e) => Err(e),
20        Value::Bool(_) => Err(CellError::Value),
21        Value::Number(n) => {
22            if n < 0.0 || n != n.trunc() {
23                return Err(CellError::Num);
24            }
25            Ok(number_to_text(n))
26        }
27        Value::Empty => Ok(String::new()),
28        v => Ok(v.to_text()?.to_string()),
29    }
30}

32/// Parses a 10-digit two's-complement number in `radix`.
33fn parse_base(s: &str, radix: u32) -> R<i64> {
34    if s.len() > 10 {
35        return Err(CellError::Num);
36    }
37    if s.is_empty() {
38        return Ok(0);
39    }
40    let v = i64::from_str_radix(s, radix).map_err(|_| CellError::Num)?;
41    if s.starts_with(['+', '-']) {
42        return Err(CellError::Num);
43    }
44    let bits = match radix {
45        2 => 10,
46        8 => 30,
47        _ => 40,
48    };
49    // A full 10-digit number with the top bit set is negative.
50    if s.len() == 10 && v >= 1i64 << (bits - 1) { Ok(v - (1i64 << bits)) } else { Ok(v) }
51}

53/// Writes `v` in `radix`, using 10-digit two's complement for negatives and padding to `places`.
54fn format_base(v: i64, radix: u32, places: Option<f64>) -> R<Value> {
55    let bits: u32 = match radix {
56        2 => 10,
57        8 => 30,
58        _ => 40,
59    };
60    let min = -(1i64 << (bits - 1));
61    let max = (1i64 << (bits - 1)) - 1;
62    if v < min || v > max {
63        return Err(CellError::Num);
64    }
65    let places = places
66        .map(|p| {
67            let p = p.trunc();
68            if !(1.0..=10.0).contains(&p) {
69                return Err(CellError::Num);
70            }
71            Ok(p as usize)
72        })
73        .transpose()?;
74    if v < 0 {
75        let u = v + (1i64 << bits);
76        return text_val(to_radix(u, radix));
77    }
78    let s = to_radix(v, radix);
79    match places {
80        None => text_val(s),
81        Some(p) => {
82            if s.len() > p {
83                return Err(CellError::Num);
84            }
85            text_val(format!("{}{}", "0".repeat(p - s.len()), s))
86        }
87    }
88}

90fn to_radix(mut v: i64, radix: u32) -> String {
91    if v == 0 {
92        return "0".into();
93    }
94    let mut out = Vec::new();
95    while v > 0 {
96        let d = (v % radix as i64) as u32;
97        out.push(char::from_digit(d, radix).unwrap_or('0').to_ascii_uppercase());
98        v /= radix as i64;
99    }
100    out.iter().rev().collect()
101}

103fn places_arg(a: &[Arg]) -> R<Option<f64>> {
104    if !has(a, 1) {
105        return Ok(None);
106    }
107    if matches!(a.get(1).map(scalar), Some(Value::Bool(_))) {
108        return Err(CellError::Value);
109    }
110    Ok(Some(num(a, 1)?))
111}

113fn convert_base(a: &[Arg], from: u32, to: u32) -> R<Value> {
114    let s = base_source(a)?;
115    let v = parse_base(&s, from)?;
116    let places = places_arg(a)?;
117    if to == 10 {
118        return Ok(Value::Number(v as f64));
119    }
120    format_base(v, to, places)
121}
122
123fn dec_to(a: &[Arg], to: u32) -> R<Value> {
124    let v = a.first().map(scalar).unwrap_or(Value::Empty);
125    if matches!(v, Value::Bool(_)) {
126        return Err(CellError::Value);
127    }
128    let n = v.to_number()?.trunc();
129    if n.abs() > 1e15 {
130        return Err(CellError::Num);
131    }
132    format_base(n as i64, to, places_arg(a)?)
133}

165const BIT_LIMIT: f64 = 281_474_976_710_656.0; // 2^48
166
167fn bit_operand(a: &[Arg], i: usize) -> R<u64> {
168    let n = num(a, i)?;
169    if !(0.0..BIT_LIMIT).contains(&n) || n != n.trunc() {
170        return Err(CellError::Num);
171    }
172    Ok(n as u64)
173}
174
175fn bitand(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
176    Ok(Value::Number((bit_operand(a, 0)? & bit_operand(a, 1)?) as f64))
177}
178fn bitor(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
179    Ok(Value::Number((bit_operand(a, 0)? | bit_operand(a, 1)?) as f64))
180}
181fn bitxor(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
182    Ok(Value::Number((bit_operand(a, 0)? ^ bit_operand(a, 1)?) as f64))
183}
184
185fn shift(a: &[Arg], left: bool) -> R<Value> {
186    let n = bit_operand(a, 0)?;
187    let s = num(a, 1)?.trunc();
188    if n == 0 {
189        return Ok(Value::Number(0.0));
190    }
191    if s.abs() > 53.0 {
192        return Err(CellError::Num);
193    }
194    let s = if left { s as i64 } else { -(s as i64) };
195    let r = if s >= 0 {
196        let r = (n as u128) << s;
197        if r >= BIT_LIMIT as u128 {
198            return Err(CellError::Num);
199        }
200        r as u64
201    } else {
202        n >> (-s)
203    };
204    Ok(Value::Number(r as f64))
205}
206fn bitlshift(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
207    shift(a, true)
208}
209fn bitrshift(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
210    shift(a, false)
211}

216fn delta(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
217    let x = num(a, 0)?;
218    let y = opt_num(a, 1, 0.0)?;
219    Ok(Value::Number(if x == y { 1.0 } else { 0.0 }))
220}
221fn gestep(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
222    let x = num(a, 0)?;
223    let y = opt_num(a, 1, 0.0)?;
224    Ok(Value::Number(if x >= y { 1.0 } else { 0.0 }))
225}
226fn erf_fn(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
227    let lo = num(a, 0)?;
228    if has(a, 1) {
229        let hi = num(a, 1)?;
230        return num_val(erf(hi) - erf(lo));
231    }
232    num_val(erf(lo))
233}
234fn erf_precise(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
235    num_val(erf(num(a, 0)?))
236}
237fn erfc_fn(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
238    num_val(erfc(num(a, 0)?))
239}

264// Excel takes the polar route, so IMSQRT("-4") is 1.22514845490862E-16+2i, not 2i. Its cosine
265// reduces its argument with a 64-bit-mantissa pi, which is PI/2 + 2.5e-20 and shows up as the
266// stray real part. `+ 0.0` turns a -0 imaginary part into +0, so the angle stays +pi.
267// shortcut: parts above ~1E154 or below ~1E-155 keep the exact result (Excel gives #NUM! or 0), no known use.
268const HALF_PI_EXCESS: f64 = 2.508_278_806_33e-20;
269im1!(imsqrt, |z: Cx| {
270    let im = z.im + 0.0;
271    let theta = match z.re {
272        0.0 => {
273            if im == 0.0 {
274                0.0
275            } else {
276                std::f64::consts::FRAC_PI_2.copysign(im)
277            }
278        }
279        x if x > 0.0 => (im / x).atan(),
280        x => (im / x).atan() + if im < 0.0 { -std::f64::consts::PI } else { std::f64::consts::PI },
281    };
282    let (half, root) = (theta / 2.0, z.abs().sqrt());
283    Ok(Cx::new(root * (half.cos() + HALF_PI_EXCESS * half.abs().sin()), root * half.sin()))
284});

641        close_tol(ev("BESSELI", vec![n(1.0), n(0.0)]), 1.266_065_877_752_008_4, 1e-12);

645        is_err(ev("BESSELJ", vec![n(1.0), n(-1.0)]), CellError::Num);
646        is_err(ev("BESSELY", vec![n(0.0), n(1.0)]), CellError::Num);
647        is_err(ev("BESSELK", vec![n(-1.0), n(1.0)]), CellError::Num);
648        is_err(ev("BESSELJ", vec![t("x"), n(1.0)]), CellError::Value);
649        // Wronskian J_{1}Y_0 − J_0Y_1 = 2/(πx), checking series and asymptotic regimes.
650        for x in [0.5, 7.0, 15.0, 24.0, 30.0, 80.0] {
