58fn textjoin(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
59    let delims: Vec<String> = as_array(&arg(a, 0)?.value).iter().map(|v| v.to_text()).collect::<R<_>>()?;
60    let ignore = boolean(a, 1)?;
61    let mut items = Vec::new();
62    for x in a.iter().skip(2) {
63        for v in as_array(&x.value).iter() {
64            let s = v.to_text()?;
65            if ignore && s.is_empty() {
66                continue;
67            }
68            items.push(s);
69        }
70    }
71    let mut out = String::new();
72    for (i, s) in items.iter().enumerate() {
73        if i > 0 && !delims.is_empty() {
74            push_capped(&mut out, &delims[(i - 1) % delims.len()])?;
75        }
76        push_capped(&mut out, s)?;
77    }
78    check_len(out)
79}

172fn substitute(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
173    let s = text(a, 0)?;
174    let old = text(a, 1)?;
175    let new = text(a, 2)?;
176    if old.is_empty() {
177        return text_val(s);
178    }
179    if has(a, 3) {
180        let inst = num(a, 3)?;
181        if inst.is_nan() || inst < 1.0 {
182            return Err(CellError::Value);
183        }
184        let inst = inst.trunc() as usize;
185        match s.match_indices(old.as_str()).nth(inst - 1) {
186            Some((pos, _)) => {
187                let mut out = String::with_capacity(s.len() + new.len());
188                out.push_str(&s[..pos]);
189                out.push_str(&new);
190                out.push_str(&s[pos + old.len()..]);
191                check_len(out)
192            }
193            None => text_val(s),
194        }
195    } else {
196        let count = s.matches(old.as_str()).count();
197        if new.len() > old.len() && s.len() + count.saturating_mul(new.len() - old.len()) > MAX_TEXT * 4 {
198            return Err(CellError::Value);
199        }
200        check_len(s.replace(old.as_str(), &new))
201    }
202}

234fn find_impl(a: &[Arg], wild: bool) -> R<Value> {
235    let needle = text(a, 0)?;
236    let hay = text(a, 1)?;
237    let start = opt_num(a, 2, 1.0)?;
238    let hc = chars(&hay);
239    if (start.is_nan() || start < 1.0) || start > hc.len().max(1) as f64 {
240        return Err(CellError::Value);
241    }
242    let start = start as usize - 1;
243    if needle.is_empty() {
244        return num_val((start + 1) as f64);
245    }
246    let pos = if wild {
247        wildcard_find(&needle, &hay, start)
248    } else {
249        let nc = chars(&needle);
250        (start..hc.len()).find(|&i| hc.get(i..i + nc.len()).is_some_and(|w| w == nc.as_slice()))
251    };
252    match pos {
253        Some(p) => num_val((p + 1) as f64),
254        None => Err(CellError::Value),
255    }
256}

333fn numbervalue(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
334    let s = text(a, 0)?;
335    let dec = if a.len() > 1 { text(a, 1)?.chars().next().ok_or(CellError::Value)? } else { '.' };
336    let grp = if a.len() > 2 { text(a, 2)?.chars().next().ok_or(CellError::Value)? } else { ',' };
337    if dec == grp {
338        return Err(CellError::Value);
339    }
340    let mut body: String = s.chars().filter(|c| !c.is_whitespace()).collect();
341    if body.is_empty() {
342        return num_val(0.0);
343    }
344    // Count the trailing percent signs and cut them off in place: copying the rest once per sign
345    // made a long run of them quadratic.
346    let kept = body.trim_end_matches('%').len();
347    let pct = body.len() - kept;
348    body.truncate(kept);
349    let mut seen_dec = false;
350    let mut out = String::new();
351    for c in body.chars() {
352        if c == dec {
353            if seen_dec {
354                return Err(CellError::Value);
355            }
356            seen_dec = true;
357            out.push('.');
358        } else if c == grp {
359            if seen_dec {
360                return Err(CellError::Value);
361            }
362        } else if c == '.' || c == ',' {
363            return Err(CellError::Value);
364        } else {
365            out.push(c);
366        }
367    }
368    let mut n = parse_number_text_in(&out, c.date_system()).ok_or(CellError::Value)?;
369    for _ in 0..pct {
370        n /= 100.0;
371    }
372    num_val(n)
373}

375/// Formats `x` (already rounded) with `decimals` places and optional thousands separators.
376fn format_fixed(x: f64, decimals: i64, commas: bool) -> String {
377    let d = decimals.max(0) as usize;
378    let s = x.abs().to_string();
379    let (int_part, frac) = match s.split_once('.') {
380        Some((i, f)) => (i, f),
381        None => (s.as_str(), ""),
382    };
383    let int_part = if commas {
384        let b = int_part.as_bytes();
385        let mut o = String::new();
386        for (i, ch) in b.iter().enumerate() {
387            if i > 0 && (b.len() - i) % 3 == 0 {
388                o.push(',');
389            }
390            o.push(char::from(*ch));
391        }
392        o
393    } else {
394        int_part.to_string()
395    };
396    let mut out = int_part;
397    if d > 0 {
398        let frac_len = frac.len().min(d);
399        out.push('.');
400        out.push_str(&frac[..frac_len]);
401        out.extend(std::iter::repeat_n('0', d - frac_len));
402    }
403    out
404}

406fn rounded(a: &[Arg], default_dec: f64) -> R<(f64, i64)> {
407    let x = num(a, 0)?;
408    let d = opt_num(a, 1, default_dec)?.trunc();
409    if d > 127.0 {
410        return Err(CellError::Value);
411    }
412    let d = d.max(-308.0) as i64;
413    let r = round_half_away(x, d as i32);
414    if !r.is_finite() || r.abs() >= 1e300 {
415        return Err(CellError::Num);
416    }
417    Ok((r, d))
418}

420fn fixed(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
421    let (r, d) = rounded(a, 2.0)?;
422    let no_commas = opt_bool(a, 2, false)?;

424    if body.len() + if r < 0.0 { 1 } else { 0 } > 255 {

448/// Thai words for a group of up to six decimal digits (most significant first); empty for zero.
449/// One in the tens is สิบ, two is ยี่สิบ, and a final one after higher digits is เอ็ด; `after`
450/// says the number already has a non-zero digit in a higher group (so 1,000,001 ends in เอ็ด).
451fn thai_group(digits: &[u8], after: bool) -> String {
452    let n = digits.len();
453    let higher = after || digits.iter().take(n.saturating_sub(1)).any(|d| *d != 0);
454    let mut out = String::new();
455    for (i, &d) in digits.iter().enumerate() {
456        let place = n - 1 - i;
457        match (place, d) {
458            (_, 0) => {}
459            (0, 1) if higher => out.push_str("เอ็ด"),
460            (1, 1) => out.push_str("สิบ"),
461            (1, 2) => out.push_str("ยี่สิบ"),
462            _ => {
463                out.push_str(THAI_DIGITS.get(d as usize).copied().unwrap_or_default());
464                out.push_str(THAI_PLACES.get(place).copied().unwrap_or_default());
465            }
466        }
467    }
468    out
469}

471/// Thai words for a whole number written in decimal digits: groups of six digits, each but the
472/// last followed by ล้าน (so 10^12 is หนึ่งล้านล้าน).
473fn thai_number(digits: &str) -> String {
474    let d: Vec<u8> = digits.bytes().filter(u8::is_ascii_digit).map(|b| b - b'0').skip_while(|x| *x == 0).collect();
475    if d.is_empty() {
476        return "ศูนย์".to_string();
477    }
478    let first = match d.len() % 6 {
479        0 => 6,
480        r => r,
481    };
482    let mut out = String::new();
483    let mut start = 0;
484    let mut len = first;
485    while let Some(group) = d.get(start..start + len) {
486        out.push_str(&thai_group(group, start > 0));
487        start += len;
488        len = 6;
489        if start < d.len() {
490            out.push_str("ล้าน");
491        }
492    }
493    out
494}

496/// BAHTTEXT(number): the amount in Thai words with บาท, rounded to satang (สตางค์); a whole
497/// number of baht ends in ถ้วน and a negative amount starts with ลบ.
498fn bahttext(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
499    let x = round_half_away(num(a, 0)?, 2);
500    if !x.is_finite() {
501        return Err(CellError::Num);
502    }
503    let fixed = format!("{:.2}", x.abs());
504    let (whole, frac) = fixed.split_once('.').unwrap_or((fixed.as_str(), "00"));
505    let baht = whole.bytes().any(|b| b != b'0');
506    let satang = frac.bytes().any(|b| b != b'0');
507    let mut out = String::new();
508    if x < 0.0 && (baht || satang) {
509        out.push_str("ลบ");
510    }
511    if baht || !satang {
512        out.push_str(&thai_number(whole));
513        out.push_str("บาท");
514    }
515    if satang {
516        out.push_str(&thai_number(frac));
517        out.push_str("สตางค์");
518    } else {
519        out.push_str("ถ้วน");
520    }
521    text_val(out)
522}

546/// Most delimiter characters compared in one search: text length × delimiters is the formula's
547/// to choose, so the search gives up with `#VALUE!` rather than run on.
548const MAX_MATCH_WORK: u64 = 100_000_000;
549
550/// Non-overlapping delimiter matches (start, end) in char positions, in ascending order. Scans left
551/// to right preferring the longest delimiter starting at each position, or (`from_end`) right to
552/// left preferring the longest delimiter ending at each position. Empty delimiters are ignored unless all
553/// delimiters are empty, in which case every position matches with zero length. Repeated
554/// delimiters are tried once, and only those whose first (or, from the end, last) character is the
555/// one at hand; a search past [`MAX_MATCH_WORK`] is `#VALUE!`.
556fn find_matches(text: &[char], delims: &[Vec<char>], insensitive: bool, from_end: bool) -> R<Vec<(usize, usize)>> {
557    let t = if insensitive { fold(text) } else { text.to_vec() };
558    let mut ds: Vec<Vec<char>> = delims.iter().filter(|d| !d.is_empty()).map(|d| if insensitive { fold(d) } else { d.clone() }).collect();
559    if ds.is_empty() {
560        return Ok((0..=t.len()).map(|i| (i, i)).collect());
561    }
562    ds.sort_unstable();
563    ds.dedup();
564    // Delimiters by the character a match is anchored on.
565    let mut by_char: HashMap<char, Vec<&[char]>> = HashMap::new();
566    for d in &ds {
567        let anchor = if from_end { d.last() } else { d.first() };
568        if let Some(&c) = anchor {
569            by_char.entry(c).or_default().push(d);
570        }
571    }
572    let mut work = 0u64;
573    let mut longest = |candidates: Option<&Vec<&[char]>>, fits: &dyn Fn(&[char]) -> bool| -> R<Option<usize>> {
574        let mut best = None;
575        for d in candidates.into_iter().flatten() {
576            work = work.saturating_add(d.len() as u64);
577            if work > MAX_MATCH_WORK {
578                return Err(CellError::Value);
579            }
580            if fits(d) {
581                best = best.max(Some(d.len()));
582            }
583        }
584        Ok(best)
585    };
586    let mut out = Vec::new();
587    if from_end {
588        let mut j = t.len();
589        while j > 0 {
590            let at = t.get(j - 1).and_then(|c| by_char.get(c));
591            match longest(at, &|d| j.checked_sub(d.len()).and_then(|st| t.get(st..j)).is_some_and(|w| w == d))? {
592                Some(l) => {
593                    out.push((j - l, j));
594                    j -= l;
595                }
596                None => j -= 1,
597            }
598        }
599        out.reverse();
600        return Ok(out);
601    }
602    let mut i = 0;
603    while i < t.len() {
604        let at = t.get(i).and_then(|c| by_char.get(c));
605        match longest(at, &|d| t.get(i..i + d.len()).is_some_and(|w| w == d))? {
606            Some(l) => {
607                out.push((i, i + l));
608                i += l;
609            }
610            None => i += 1,
611        }
612    }
613    Ok(out)
614}

616fn text_before_after(a: &[Arg], before: bool) -> R<Value> {
617    let s = chars(&text(a, 0)?);
618    let delims = delimiters(&arg(a, 1)?.value)?;
619    let inst = if has(a, 2) { int(a, 2)? } else { 1 };
620    let mode = opt_int(a, 3, 0)?;
621    let match_end = opt_bool(a, 4, false)?;
622    if !(0..=1).contains(&mode) || inst == 0 || inst.unsigned_abs() as usize > s.len().max(1) {
623        return Err(CellError::Value);
624    }
625    let mut m = find_matches(&s, &delims, mode == 1, inst < 0)?;
626    if match_end {
627        if inst > 0 {
628            if m.last().is_none_or(|x| x.0 != s.len()) {
629                m.push((s.len(), s.len()));
630            }
631        } else if m.first().is_none_or(|x| x.0 != 0) {
632            m.insert(0, (0, 0));
633        }
634    }
635    let k = inst.unsigned_abs() as usize;
636    let hit = if inst > 0 { m.get(k - 1) } else { m.len().checked_sub(k).and_then(|i| m.get(i)) };
637    match hit {
638        Some(&(st, en)) => text_val(if before { s[..st].iter().collect::<String>() } else { s[en..].iter().collect::<String>() }),
639        None => match a.get(5) {
640            Some(x) => Ok(scalar(x)),
641            None => Err(CellError::NA),
642        },
643    }
644}

654fn split_by(s: &[char], delims: &[Vec<char>], insensitive: bool) -> R<Vec<Vec<char>>> {
655    if delims.iter().all(Vec::is_empty) {
656        return Ok(vec![s.to_vec()]);
657    }
658    let mut out = Vec::new();
659    let mut last = 0;
660    for (st, en) in find_matches(s, delims, insensitive, false)? {
661        out.push(s[last..st].to_vec());
662        last = en;
663    }
664    out.push(s[last..].to_vec());
665    Ok(out)
666}

668fn textsplit(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
669    let s = chars(&text(a, 0)?);
670    let col_d = if has(a, 1) { delimiters(&arg(a, 1)?.value)? } else { vec![] };
671    let row_d = if has(a, 2) { delimiters(&arg(a, 2)?.value)? } else { vec![] };
672    if col_d.iter().all(|d| d.is_empty()) && row_d.iter().all(|d| d.is_empty()) {
673        return Err(CellError::Value);
674    }
675    let ignore = opt_bool(a, 3, false)?;
676    let mode = opt_int(a, 4, 0)?;
677    if !(0..=1).contains(&mode) {
678        return Err(CellError::Value);
679    }
680    let pad = if has(a, 5) { scalar(arg(a, 5)?) } else { Value::Error(CellError::NA) };
681    let mut rows: Vec<Vec<String>> = Vec::new();
682    for r in split_by(&s, &row_d, mode == 1)? {
683        if ignore && r.is_empty() {
684            continue;
685        }
686        let cells: Vec<String> =
687            split_by(&r, &col_d, mode == 1)?.into_iter().filter(|c| !(ignore && c.is_empty())).map(|c| c.into_iter().collect()).collect();
688        if ignore && cells.is_empty() {
689            continue;
690        }
691        rows.push(cells);
692    }
693    if rows.is_empty() {
694        return Err(CellError::Calc);
695    }
696    let ncols = rows.iter().map(Vec::len).max().unwrap_or(1).max(1);
697    let nrows = rows.len();
698    if nrows.saturating_mul(ncols) > MAX_CELLS {
699        return Err(CellError::Num);
700    }
701    let mut data = Vec::with_capacity(nrows * ncols);
702    for r in rows {
703        let n = r.len();
704        data.extend(r.into_iter().map(Value::from));
705        data.extend(std::iter::repeat_n(pad.clone(), ncols - n));
706    }
707    array_val(nrows, ncols, data)
708}

783fn regexextract(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
784    let s = text(a, 0)?;

786    let re = build_regex(&text(a, 1)?, case_arg(a, 3)?)?;

789        1 => {

791            if all.is_empty() {
792                return Err(CellError::NA);
793            }
794            array_val(all.len(), 1, all)
795        }
796        2 => {
797            let caps = re.captures(&s).ok_or(CellError::NA)?;
798            if caps.len() <= 1 {
799                return Ok(Value::from(caps.get(0).map_or("", |m| m.as_str())));
800            }
801            let groups: Vec<Value> = (1..caps.len()).map(|i| Value::from(caps.get(i).map_or("", |m| m.as_str()))).collect();
802            array_val(1, groups.len(), groups)
803        }
804        _ => Err(CellError::Value),
805    }
806}

808/// Converts Excel-style `$1` references to the regex crate's unambiguous `${1}` form.
809fn convert_replacement(r: &str) -> String {
810    let mut out = String::with_capacity(r.len() + 4);
811    let c: Vec<char> = r.chars().collect();
812    let mut i = 0;
813    while i < c.len() {
814        let ch = c[i];
815        if ch == '$' && c.get(i + 1).is_some_and(char::is_ascii_digit) {
816            let mut j = i + 1;
817            while c.get(j).is_some_and(char::is_ascii_digit) {
818                j += 1;
819            }
820            out.push_str("${");
821            out.extend(c[i + 1..j].iter());
822            out.push('}');
823            i = j;
824        } else {
825            out.push(ch);
826            i += 1;
827        }
828    }
829    out
830}

832/// A piece of a replacement: literal text or a capture group (by number or name).
833enum Piece {
834    Text(String),
835    Index(usize),
836    Name(String),
837}
838
839/// A replacement split into pieces as `regex::Captures::expand` reads it: `$$` is `$`, `$name` takes
840/// the longest run of `[_0-9A-Za-z]`, `${name}` ends at `}` (`${}` is empty), a name of digits is a
841/// group number, and a `$` that starts no reference is itself.
842fn template(rep: &str) -> Vec<Piece> {
843    let mut out = Vec::new();
844    let mut lit = String::new();
845    let mut rest = rep;
846    while let Some(i) = rest.find('$') {
847        lit.push_str(&rest[..i]);
848        rest = &rest[i + 1..];
849        if let Some(r) = rest.strip_prefix('$') {
850            lit.push('$');
851            rest = r;
852            continue;
853        }
854        // A braced name is a reference even when empty (it expands to nothing).
855        let braced = rest.strip_prefix('{').and_then(|b| b.find('}').map(|j| (&b[..j], &b[j + 1..])));
856        let (name, after) = braced.unwrap_or_else(|| {
857            let j = rest.find(|c: char| !(c == '_' || c.is_ascii_alphanumeric())).unwrap_or(rest.len());
858            (&rest[..j], &rest[j..])
859        });
860        if braced.is_none() && name.is_empty() {
861            lit.push('$');
862            continue;
863        }
864        if !lit.is_empty() {
865            out.push(Piece::Text(std::mem::take(&mut lit)));
866        }
867        out.push(match name.parse::<usize>() {
868            Ok(n) => Piece::Index(n),
869            Err(_) => Piece::Name(name.to_string()),
870        });
871        rest = after;
872    }
873    lit.push_str(rest);
874    if !lit.is_empty() {
875        out.push(Piece::Text(lit));
876    }
877    out
878}

880/// Append a match's replacement to `out`, piece by piece; `None` once `out` would pass `limit`
881/// bytes, before anything that long is built (#2162).
882fn expand_capped(cap: &regex::Captures, pieces: &[Piece], out: &mut String, limit: usize) -> Option<()> {
883    for p in pieces {
884        let s = match p {
885            Piece::Text(t) => t.as_str(),
886            Piece::Index(i) => cap.get(*i).map_or("", |m| m.as_str()),
887            Piece::Name(n) => cap.name(n).map_or("", |m| m.as_str()),
888        };
889        if out.len().saturating_add(s.len()) > limit {
890            return None;
891        }
892        out.push_str(s);
893    }
894    Some(())
895}

897fn regexreplace(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
898    let s = text(a, 0)?;
899    let rep = template(&convert_replacement(&text(a, 2)?));
900    let occ = opt_int(a, 3, 0)?;
901    let re = build_regex(&text(a, 1)?, case_arg(a, 4)?)?;
902    // Built match by match within a bound past which the text can only be too long (a character
903    // is at most four bytes), instead of building every replacement first (#2162).
904    let limit = MAX_TEXT * 4;
905    let mut out = String::new();
906    if occ == 0 {
907        let mut last = 0;
908        for cap in re.captures_iter(&s) {
909            let Some(m) = cap.get(0) else { continue };
910            out.push_str(s.get(last..m.start()).unwrap_or_default());
911            expand_capped(&cap, &rep, &mut out, limit).ok_or(CellError::Value)?;
912            last = m.end();
913        }
914        out.push_str(s.get(last..).unwrap_or_default());
915        return check_len(out);
916    }
917    let caps: Vec<regex::Captures> = re.captures_iter(&s).collect();
918    let k = occ.unsigned_abs() as usize;
919    let idx = if occ > 0 { k.checked_sub(1) } else { caps.len().checked_sub(k) };
920    let Some(cap) = idx.and_then(|i| caps.get(i)) else { return text_val(s) };
921    let Some(m) = cap.get(0) else { return text_val(s) };
922    out.push_str(s.get(..m.start()).unwrap_or_default());
923    expand_capped(cap, &rep, &mut out, limit).ok_or(CellError::Value)?;
924    out.push_str(s.get(m.end()..).unwrap_or_default());
925    check_len(out)
926}
