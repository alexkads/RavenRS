23/// How a unit accepts prefixes: none, SI (raised to `power` for areas/volumes) or SI + binary.

214fn find_unit(name: &str) -> Option<&'static Unit> {
215    UNITS.iter().find(|u| u.names.contains(&name))
216}
217
218/// Resolves a unit name to (unit, prefix multiplier). Exact names win over prefixed ones.
219fn resolve(name: &str) -> Option<(&'static Unit, f64)> {
220    let name = name.replace("^2", "2").replace("^3", "3");
221    if let Some(u) = find_unit(&name) {
222        return Some((u, 1.0));
223    }
224    for (p, f) in BIN_PREFIXES {
225        if let Some(rest) = name.strip_prefix(p)
226            && let Some(u) = find_unit(rest)
227            && u.pre == Pre::Bin
228        {
229            return Some((u, *f));
230        }
231    }
232    for (p, f) in SI_PREFIXES {
233        if let Some(rest) = name.strip_prefix(p)
234            && let Some(u) = find_unit(rest)
235        {
236            match u.pre {
237                Pre::Si(pow) => return Some((u, f.powi(pow))),
238                Pre::Bin => return Some((u, *f)),
239                Pre::No => {}
240            }
241        }
242    }
243    None
244}

246fn to_kelvin(name: &str, x: f64) -> f64 {
247    match name {
248        "C" | "cel" => x + 273.15,
249        "F" | "fah" => (x - 32.0) * 5.0 / 9.0 + 273.15,
250        "Rank" => x * 5.0 / 9.0,
251        "Reau" => x * 1.25 + 273.15,
252        _ => x,
253    }
254}
255
256fn from_kelvin(name: &str, k: f64) -> f64 {
257    match name {
258        "C" | "cel" => k - 273.15,
259        "F" | "fah" => (k - 273.15) * 9.0 / 5.0 + 32.0,
260        "Rank" => k * 9.0 / 5.0,
261        "Reau" => (k - 273.15) / 1.25,
262        _ => k,
263    }
264}

266pub(super) fn convert(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
267    let x = num(a, 0)?;
268    let from = text(a, 1)?;
269    let to = text(a, 2)?;
270    let (fu, fp) = resolve(&from).ok_or(CellError::NA)?;
271    let (tu, tp) = resolve(&to).ok_or(CellError::NA)?;
272    if fu.dim != tu.dim {
273        return Err(CellError::NA);
274    }
275    if fu.dim == Dim::Temperature {
276        let fname = fu.names.first().copied().unwrap_or("K");
277        let tname = tu.names.first().copied().unwrap_or("K");
278        if fname == tname && fp == tp {
279            return num_val(x);
280        }
281        let k = to_kelvin(fname, x * fp);
282        return num_val(from_kelvin(tname, k) / tp);
283    }
284    let base = x * fu.factor * fp;
285    num_val(base / (tu.factor * tp))
286}
