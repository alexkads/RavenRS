15/// A date serial argument (truncated to whole days when `whole`). Out of range → `#NUM!`.
16fn serial_arg(a: &[Arg], i: usize, sys: DateSystem) -> R<f64> {
17    let n = scalar(arg(a, i)?).to_number_in(sys)?;
18    if !(0.0..MAX_SERIAL + 1.0).contains(&n) {
19        return Err(CellError::Num);
20    }
21    Ok(n)
22}

36/// Weekday of a serial, 0 = Sunday.
37fn weekday0(sys: DateSystem, serial: i64) -> usize {
38    match sys {
39        DateSystem::D1900 => (serial + 6).rem_euclid(7) as usize,
40        DateSystem::D1904 => (serial + 5).rem_euclid(7) as usize,
41    }
42}

44fn date(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
45    let y = to_int(num(a, 0)?)?;
46    let m = to_int(num(a, 1)?)?;
47    let d = to_int(num(a, 2)?)?;
48    if !(0..10000).contains(&y) || m.abs() > 1_000_000 || d.abs() > 100_000_000 {
49        return Err(CellError::Num);
50    }
51    num_val(ymd_serial(c.date_system(), y, m, d)? as f64)
52}

54fn time(a: &[Arg], _c: &mut dyn Ctx) -> R<Value> {
55    let h = num(a, 0)?.trunc();
56    let m = num(a, 1)?.trunc();
57    let s = num(a, 2)?.trunc();
58    if h > 32767.0 || m > 32767.0 || s > 32767.0 {
59        return Err(CellError::Num);
60    }
61    let total = h * 3600.0 + m * 60.0 + s;
62    if total < 0.0 {
63        return Err(CellError::Num);
64    }
65    num_val(total.rem_euclid(86400.0) / 86400.0)
66}

108fn weekday(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
109    let sys = c.date_system();
110    let s = day_arg(a, 0, sys)?;
111    let t = to_int(opt_num(a, 1, 1.0)?)?;
112    let wd = weekday0(sys, s) as i64;
113    let r = match t {
114        1 | 17 => wd + 1,
115        2 | 11 => (wd + 6) % 7 + 1,
116        3 => (wd + 6) % 7,
117        12..=16 => (wd + 7 - (t - 10)) % 7 + 1,
118        _ => return Err(CellError::Num),
119    };
120    num_val(r as f64)
121}

123fn isoweek(sys: DateSystem, s: i64) -> R<i64> {
124    let iso_wd = (weekday0(sys, s) + 6) % 7 + 1; // Monday = 1
125    let thursday = s - iso_wd as i64 + 4;
126    let first = match sys {
127        DateSystem::D1900 => 1,
128        DateSystem::D1904 => 0,
129    };
130    let jan1 = if thursday < first {
131        // The week belongs to the year before the calendar starts.
132        first - 365
133    } else {
134        ymd_serial(sys, dt(sys, thursday as f64)?.year as i64, 1, 1)?
135    };
136    Ok((thursday - jan1).div_euclid(7) + 1)
137}

139fn weeknum(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
140    let sys = c.date_system();
141    let s = day_arg(a, 0, sys)?;
142    let t = to_int(opt_num(a, 1, 1.0)?)?;
143    let start = match t {
144        1 | 17 => 0,
145        2 | 11 => 1,
146        12..=16 => t - 10,
147        21 => return num_val(isoweek(sys, s)? as f64),
148        _ => return Err(CellError::Num),
149    } as usize;
150    let y = dt(sys, s as f64)?.year as i64;
151    let jan1 = ymd_serial(sys, y, 1, 1).unwrap_or(s.min(1));
152    let offset = (weekday0(sys, jan1) + 7 - start) % 7;
153    num_val(((s - jan1 + offset as i64).div_euclid(7) + 1) as f64)
154}

169fn edate(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
170    let sys = c.date_system();
171    let d = dt(sys, day_arg(a, 0, sys)? as f64)?;
172    let m = months_arg(a, 1)?;
173    let total = d.year as i64 * 12 + d.month as i64 - 1 + m;
174    let (y, mo) = (total.div_euclid(12), total.rem_euclid(12) + 1);
175    if !(1900..=9999).contains(&y) {
176        return Err(CellError::Num);
177    }
178    // Excel keeps serial 0's day 0 and gives February 1900 its real 28 days here, though the
179    // sheet's calendar has a 29 February 1900 (serial 60).
180    let dd = (d.day as i64).min(days_in_month(y as i32, mo as u32) as i64);
181    num_val(ymd_serial(sys, y, mo, dd)? as f64)
182}

184fn eomonth(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
185    let sys = c.date_system();
186    let d = dt(sys, day_arg(a, 0, sys)? as f64)?;
187    let m = months_arg(a, 1)?;
188    let total = d.year as i64 * 12 + d.month as i64 - 1 + m;
189    let (y, mo) = (total.div_euclid(12), total.rem_euclid(12) + 1);
190    if !(1900..=9999).contains(&y) {
191        return Err(CellError::Num);
192    }
193    // Like EDATE, February 1900 ends on the 28th (serial 59).
194    num_val(ymd_serial(sys, y, mo, days_in_month(y as i32, mo as u32) as i64)? as f64)
195}

197fn datedif(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
198    let sys = c.date_system();
199    let s = day_arg(a, 0, sys)?;
200    let e = day_arg(a, 1, sys)?;
201    let unit = text(a, 2)?.to_ascii_uppercase();
202    if s > e {
203        return Err(CellError::Num);
204    }
205    let d1 = dt(sys, s as f64)?;
206    let d2 = dt(sys, e as f64)?;
207    let months = (d2.year as i64 - d1.year as i64) * 12 + d2.month as i64 - d1.month as i64 - i64::from(d2.day < d1.day);
208    let r = match unit.as_str() {
209        "D" => e - s,
210        "M" => months,
211        "Y" => months / 12,
212        "YM" => months % 12,
213        "MD" => {
214            if d2.day >= d1.day {
215                (d2.day - d1.day) as i64
216            } else {
217                let (py, pm) = if d2.month == 1 { (d2.year - 1, 12) } else { (d2.year, d2.month - 1) };
218                days_in_month(py, pm) as i64 - d1.day as i64 + d2.day as i64
219            }
220        }
221        "YD" => {
222            let later = (d1.month, d1.day) > (d2.month, d2.day);
223            let y = if later { d2.year - 1 } else { d2.year };
224            let dd = d1.day.min(days_in_month(y, d1.month).max(1));
225            let shifted = ymd_serial(sys, y as i64, d1.month as i64, dd as i64)?;
226            e - shifted
227        }
228        _ => return Err(CellError::Num),
229    };
230    num_val(r as f64)
231}

233fn parsed_text(a: &[Arg], c: &mut dyn Ctx) -> R<(f64, &'static str)> {
234    let v = scalar(arg(a, 0)?);
235    let Value::Text(t) = v else {
236        return Err(match v {
237            Value::Error(e) => e,

243        (Value::Number(n), Some(f)) if f.contains(['d', 'y', 'h']) => Ok((n, f)),
244        _ => Err(CellError::Value),
245    }

278fn days360_core(d1: &DateTime, d2: &DateTime, european: bool) -> i64 {
279    let (mut a, mut b) = (d1.day as i64, d2.day as i64);
280    if european {
281        a = a.min(30);
282        b = b.min(30);
283    } else {
284        if a == 31 || last_of_feb(d1) {
285            a = 30;
286        }
287        if b == 31 && a >= 30 {
288            b = 30;
289        }
290    }
291    (d2.year as i64 - d1.year as i64) * 360 + (d2.month as i64 - d1.month as i64) * 30 + b - a
292}

294/// Weekend mask indexed by weekday (0 = Sunday).
295fn weekend_mask(v: Option<&Arg>) -> R<[bool; 7]> {
296    let Some(arg) = v else { return Ok([true, false, false, false, false, false, true]) };
297    match scalar(arg) {
298        Value::Empty if !arg.from_ref => Ok([true, false, false, false, false, false, true]),
299        Value::Text(t) => {
300            let b = t.as_bytes();
301            // All seven days marked as the weekend is a valid string: Microsoft documents
302            // NETWORKDAYS.INTL as "Using 1111111 will always return 0" and WORKDAY.INTL as
303            // rejecting it, which `workday_core`'s `per_week == 0` guard already implements.
304            if b.len() != 7 || !b.iter().all(|c| *c == b'0' || *c == b'1') {
305                return Err(CellError::Value);
306            }
307            // String order is Monday..Sunday.
308            let mut m = [false; 7];
309            for (i, c) in b.iter().enumerate() {
310                m[(i + 1) % 7] = *c == b'1';
311            }
312            Ok(m)
313        }
314        Value::Error(e) => Err(e),
315        other => {
316            let k = to_int(other.to_number()?)?;
317            let mut m = [false; 7];
318            match k {
319                1..=7 => {
320                    m[((k + 5) % 7) as usize] = true;
321                    m[((k + 6) % 7) as usize] = true;
322                }
323                11..=17 => m[(k - 11) as usize] = true,
324                _ => return Err(CellError::Num),
325            }
326            Ok(m)
327        }
328    }
329}

331fn holidays(v: Option<&Arg>, sys: DateSystem) -> R<Vec<i64>> {
332    let mut out = Vec::new();
333    if let Some(a) = v {
334        for x in as_array(&a.value).iter() {
335            match x {
336                Value::Empty => {}
337                Value::Error(e) => return Err(*e),
338                other => {
339                    let n = other.to_number_in(sys)?;
340                    if !(0.0..MAX_SERIAL + 1.0).contains(&n) {
341                        return Err(CellError::Num);
342                    }
343                    out.push(n.floor() as i64);
344                }
345            }
346        }
347    }
348    out.sort_unstable();
349    out.dedup();
350    Ok(out)
351}

353/// Working days in [a, b] (a ≤ b) ignoring holidays.
354fn count_workdays(sys: DateSystem, a: i64, b: i64, mask: &[bool; 7]) -> i64 {
355    let per_week = mask.iter().filter(|w| !**w).count() as i64;
356    let total = b - a + 1;
357    let weeks = total / 7;
358    let mut n = weeks * per_week;
359    let mut d = a + weeks * 7;
360    while d <= b {
361        if !mask[weekday0(sys, d)] {
362            n += 1;
363        }
364        d += 1;
365    }
366    n
367}

369fn networkdays_core(a: &[Arg], c: &mut dyn Ctx, mask: [bool; 7], hol: Option<&Arg>) -> R<Value> {
370    let sys = c.date_system();
371    let s = day_arg(a, 0, sys)?;
372    let e = day_arg(a, 1, sys)?;
373    let hs = holidays(hol, sys)?;
374    let (lo, hi, sign) = if s <= e { (s, e, 1) } else { (e, s, -1) };
375    let mut n = count_workdays(sys, lo, hi, &mask);
376    n -= hs.iter().filter(|&&h| h >= lo && h <= hi && !mask[weekday0(sys, h)]).count() as i64;
377    num_val((n * sign) as f64)
378}

389/// Moves `n` (> 0) working days from `start` in direction `dir`, ignoring holidays.
390fn advance(sys: DateSystem, start: i64, n: i64, dir: i64, mask: &[bool; 7], per_week: i64) -> i64 {
391    let weeks = (n - 1) / per_week;
392    let mut cur = start + dir * 7 * weeks;
393    let mut rem = n - weeks * per_week;
394    while rem > 0 {
395        cur += dir;
396        if !mask[weekday0(sys, cur)] {
397            rem -= 1;
398        }
399    }
400    cur
401}

403fn workday_core(a: &[Arg], c: &mut dyn Ctx, mask: [bool; 7], hol: Option<&Arg>) -> R<Value> {
404    let sys = c.date_system();
405    let start = day_arg(a, 0, sys)?;
406    let days = num(a, 1)?.trunc();
407    if days.abs() > 1e7 {
408        return Err(CellError::Num);
409    }
410    let days = days as i64;
411    let hs = holidays(hol, sys)?;
412    let per_week = mask.iter().filter(|w| !**w).count() as i64;
413    if per_week == 0 {
414        return Err(CellError::Value);
415    }
416    if days == 0 {
417        return num_val(start as f64);
418    }
419    let dir = days.signum();
420    let mut cur = start;
421    let mut rem = days.abs();
422    while rem > 0 {
423        let next = advance(sys, cur, rem, dir, &mask, per_week);
424        let (lo, hi) = if dir > 0 { (cur + 1, next) } else { (next, cur - 1) };
425        rem = hs.iter().filter(|&&h| h >= lo && h <= hi && !mask[weekday0(sys, h)]).count() as i64;
426        cur = next;
427        if !(0..=MAX_SERIAL as i64).contains(&cur) {
428            return Err(CellError::Num);
429        }
430    }
431    num_val(cur as f64)
432}

443/// Excel's YEARFRAC for a basis (0–4) between two whole-day serials.
444pub(crate) fn yearfrac_core(sys: DateSystem, s: i64, e: i64, basis: i64) -> R<f64> {
445    let (s, e) = if s <= e { (s, e) } else { (e, s) };
446    let d1 = dt(sys, s as f64)?;
447    let d2 = dt(sys, e as f64)?;
448    let days = (e - s) as f64;
449    Ok(match basis {
450        0 => {
451            let (mut a, mut b) = (d1.day as i64, d2.day as i64);
452            // Excel tests the end-31 rule against the original start day, before a February end moves to 30.
453            let start_30 = a >= 30;
454            if last_of_feb(&d1) && last_of_feb(&d2) {
455                b = 30;
456            }
457            if last_of_feb(&d1) || a == 31 {
458                a = 30;
459            }
460            if start_30 && b == 31 {
461                b = 30;
462            }
463            ((d2.year as i64 - d1.year as i64) * 360 + (d2.month as i64 - d1.month as i64) * 30 + b - a) as f64 / 360.0
464        }
465        1 => {
466            let within_year = d1.year == d2.year || (d2.year == d1.year + 1 && (d1.month > d2.month || (d1.month == d2.month && d1.day >= d2.day)));
467            if within_year {
468                let year_len = if (d1.year == d2.year && is_leap(d1.year)) || feb29_between(sys, s, e, &d1, &d2) || (d2.month == 2 && d2.day == 29) {
469                    366.0
470                } else {
471                    365.0
472                };
473                days / year_len
474            } else {
475                let years = (d2.year - d1.year + 1) as f64;
476                let span = ymd_serial(sys, d2.year as i64 + 1, 1, 1).unwrap_or(e + 1) - ymd_serial(sys, d1.year as i64, 1, 1).unwrap_or(s);
477                days / (span as f64 / years)
478            }
479        }
480        2 => days / 360.0,
481        3 => days / 365.0,
482        4 => days360_core(&d1, &d2, true) as f64 / 360.0,
483        _ => return Err(CellError::Num),
484    })
485}

487fn feb29_between(sys: DateSystem, s: i64, e: i64, d1: &DateTime, d2: &DateTime) -> bool {
488    (d1.year..=d2.year).any(|y| is_leap(y) && serial_from_ymd(sys, y as i64, 2, 29).is_some_and(|f| (f as i64) >= s && (f as i64) <= e))
489}

491fn yearfrac(a: &[Arg], c: &mut dyn Ctx) -> R<Value> {
492    let sys = c.date_system();

839    fn system_1904() {
840        let mut ctx = TestCtx { sys: crate::base::DateSystem::D1904, ..TestCtx::default() };
841        let spec = crate::functions::lookup("DATE").unwrap();
842        close(crate::functions::call(spec, &[n(2000.0), n(1.0), n(1.0)], &mut ctx), 35064.0);
843        let spec = crate::functions::lookup("WEEKDAY").unwrap();
844        close(crate::functions::call(spec, &[n(0.0)], &mut ctx), 6.0);
845        let spec = crate::functions::lookup("YEAR").unwrap();
846        close(crate::functions::call(spec, &[n(0.0)], &mut ctx), 1904.0);
847        // The clock is 2026-10-07 12:00; in the 1904 system that is serial 44840.5.
848        close(crate::functions::call(crate::functions::lookup("NOW").unwrap(), &[], &mut ctx), 44840.5);
849        let today = crate::functions::call(crate::functions::lookup("TODAY").unwrap(), &[], &mut ctx);
850        close(today.clone(), 44840.0);
851        close(crate::functions::call(spec, &[crate::functions::Arg::val(today)], &mut ctx), 2026.0);
852        // No fictitious day here: 1904 is a real leap year and 1900 is before the epoch.
853        let spec = crate::functions::lookup("DATE").unwrap();
854        close(crate::functions::call(spec, &[n(1904.0), n(2.0), n(29.0)], &mut ctx), 59.0);
855        close(crate::functions::call(spec, &[n(1904.0), n(3.0), n(0.0)], &mut ctx), 59.0);
856        close(crate::functions::call(spec, &[n(1904.0), n(3.0), n(1.0)], &mut ctx), 60.0);
857        is_err(crate::functions::call(spec, &[n(1900.0), n(2.0), n(29.0)], &mut ctx), CellError::Num);
858        let spec = crate::functions::lookup("EOMONTH").unwrap();
859        close(crate::functions::call(spec, &[n(31.0), n(0.0)], &mut ctx), 59.0);
860        let spec = crate::functions::lookup("DATEVALUE").unwrap();
861        close(crate::functions::call(spec, &[t("2/29/1904")], &mut ctx), 59.0);
862        is_err(crate::functions::call(spec, &[t("2/29/1900")], &mut ctx), CellError::Value);
863    }
