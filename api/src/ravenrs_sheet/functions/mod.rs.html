1//! GridCraft worksheet function library.
2//!
3//! Every function is described by a [`FnSpec`] (name, arity, category, which parameters take
4//! scalars, a signature and a one-line description) and implemented as a plain Rust function over
5//! already-evaluated arguments. The calculation engine looks functions up by name with
6//! [`lookup`] and invokes them through [`call`], which applies Excel's automatic array lifting:
7//! passing an array where a scalar is expected evaluates the function once per element.
8//!
9//! Special forms that need references, laziness or hidden-row information (ROW, OFFSET,
10//! INDIRECT, LET, LAMBDA, SUBTOTAL, AGGREGATE, TEXT…) are evaluated by the host; this crate
11//! exports [`aggregate_values`] for SUBTOTAL/AGGREGATE.
12#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
13#![forbid(unsafe_code)]

20/// Builds a [`FnSpec`] whose implementation returns `Result<Value, CellError>`.
21///
22/// `f!(NAME, min, max, Category, scalar_args, "SIG", "description", path::to::impl)`; prefix the
23/// arguments with `volatile;` for volatile functions.
24macro_rules! f {
25    (volatile; $name:literal, $min:expr, $max:expr, $cat:ident, $scal:expr, $sig:literal, $desc:literal, $imp:expr) => {
26        $crate::functions::FnSpec {
27            name: $name,
28            min_args: $min,
29            max_args: $max,
30            category: $crate::functions::Category::$cat,
31            volatile: true,
32            scalar_args: $scal,
33            value_args: &[],
34            signature: $sig,
35            description: $desc,
36            imp: {
37                fn w(a: &[$crate::functions::Arg], c: &mut dyn $crate::functions::Ctx) -> crate::base::Value {
38                    $crate::functions::util::finish(($imp)(a, c))
39                }
40                w
41            },
42        }
43    };
44    ($name:literal, $min:expr, $max:expr, $cat:ident, $scal:expr, $sig:literal, $desc:literal, $imp:expr) => {
45        $crate::functions::FnSpec {
46            name: $name,
47            min_args: $min,
48            max_args: $max,
49            category: $crate::functions::Category::$cat,
50            volatile: false,
51            scalar_args: $scal,
52            value_args: &[],
53            signature: $sig,
54            description: $desc,
55            imp: {
56                fn w(a: &[$crate::functions::Arg], c: &mut dyn $crate::functions::Ctx) -> crate::base::Value {
57                    $crate::functions::util::finish(($imp)(a, c))
58                }
59                w
60            },
61        }
62    };
63}

84/// An evaluated argument. Ranges arrive as `Value::Array` (blank cells = `Value::Empty`) with
85/// `from_ref = true`.
86#[derive(Clone, Debug)]
87pub struct Arg {
88    pub value: Value,
89    pub from_ref: bool,
90}
91
92impl Arg {
93    /// A literal (non-reference) argument.
94    pub fn val(value: impl Into<Value>) -> Arg {
95        Arg { value: value.into(), from_ref: false }
96    }
97    /// An argument that came from a cell or range reference.
98    pub fn reference(value: impl Into<Value>) -> Arg {
99        Arg { value: value.into(), from_ref: true }
100    }
101}
102
103/// Services a function may need from the host.
104pub trait Ctx {
105    fn date_system(&self) -> DateSystem;
106    /// Current local date-time serial.
107    fn now_serial(&self) -> f64;
108    /// Uniform in [0, 1).
109    fn random(&mut self) -> f64;
110    /// For the database functions: whether the formula in the criteria cell at (`row`, `col`) holds
111    /// for `record` (a row of `db` below its labels). `None` when that cell holds no formula or the
112    /// criteria didn't come from a reference.
113    fn computed_criterion(&mut self, _row: usize, _col: usize, _db: &Array, _record: usize) -> Option<bool> {
114        None
115    }
116    /// For GROUPBY/PIVOTBY: whether their function argument is a LAMBDA (its value is then not used).
117    fn has_lambda(&self) -> bool {
118        false
119    }
120    /// Calls that LAMBDA with `arg` (one group's values) and returns its single result; `None` when
121    /// there is no lambda.
122    fn call_lambda(&mut self, _arg: Value) -> Option<Value> {
123        None
124    }
125}

127pub type FnImpl = fn(&[Arg], &mut dyn Ctx) -> Value;

148pub struct FnSpec {
149    /// Upper-case, without any `_xlfn.` prefix.
150    pub name: &'static str,
151    pub min_args: usize,
152    /// `usize::MAX` for variadic functions (Excel allows 255 arguments).
153    pub max_args: usize,
154    pub category: Category,
155    pub volatile: bool,
156    /// For automatic array lifting: true for each parameter position that expects a scalar. When
157    /// an argument at a scalar position is an array, the function is applied element-wise
158    /// (broadcast like Excel's dynamic arrays). The last entry repeats for variadic tails; an
159    /// empty slice means no lifting.
160    pub scalar_args: &'static [bool],
161    /// Parameter positions that take a single value and reject a reference to more than one
162    /// cell with `#VALUE!` (the Analysis ToolPak functions: EOMONTH, WORKDAY, COUPDAYS, BIN2DEC…),
163    /// where other scalar parameters lift the range. Arrays that aren't references still lift.
164    /// Same layout as `scalar_args`; filled at registration from [`lift::value_args`].
165    pub value_args: &'static [bool],
166    pub signature: &'static str,
167    pub description: &'static str,
168    pub imp: FnImpl,
169}

171impl std::fmt::Debug for FnSpec {
172    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
173        f.debug_struct("FnSpec").field("name", &self.name).field("min_args", &self.min_args).field("max_args", &self.max_args).finish()
174    }
175}
176
177impl FnSpec {
178    /// Whether parameter `i` expects a scalar (and is therefore lifted over arrays).
179    pub fn is_scalar_arg(&self, i: usize) -> bool {
180        match self.scalar_args.get(i) {
181            Some(b) => *b,
182            None => self.scalar_args.last().copied().unwrap_or(false),
183        }
184    }
185
186    /// Whether parameter `i` takes a single value, so a multi-cell reference there is `#VALUE!`.
187    pub fn is_value_arg(&self, i: usize) -> bool {
188        match self.value_args.get(i) {
189            Some(b) => *b,
190            None => self.value_args.last().copied().unwrap_or(false),
191        }
192    }
193
194    /// Whether a call with `n` arguments is valid (Excel allows at most 255).
195    pub fn accepts(&self, n: usize) -> bool {
196        (self.min_args..=self.max_args).contains(&n) && n <= 255
197    }
198}

203fn registry() -> &'static (Vec<FnSpec>, HashMap<&'static str, usize>) {
204    static REG: OnceLock<(Vec<FnSpec>, HashMap<&str, usize>)> = OnceLock::new();
205    REG.get_or_init(|| {
206        let mut v: Vec<FnSpec> = Vec::new();
207        for list in [
208            math::specs(),
209            stat::specs(),
210            ets::specs(),
211            text::specs(),
212            date::specs(),
213            lookup::specs(),
214            dynamic::specs(),
215            financial::specs(),
216            info::specs(),
217            engineering::specs(),
218            database::specs(),
219        ] {
220            v.extend(list);
221        }
222        for s in &mut v {
223            s.value_args = lift::value_args(s.name);
224        }
225        let mut map = HashMap::with_capacity(v.len());
226        for (i, s) in v.iter().enumerate() {
227            map.entry(s.name).or_insert(i);
228        }
229        (v, map)
230    })
231}

233/// Looks a function up by name (case-insensitive; `_xlfn.`- and `_xlws.`-free names). A
234/// `_xludf.` name is user-defined and never found.
235pub fn lookup(name: &str) -> Option<&'static FnSpec> {
236    let mut n = name.trim().to_ascii_uppercase();
237    loop {
238        if let Some(r) = n.strip_prefix("_XLFN.") {
239            n = r.to_string();
240        } else if let Some(r) = n.strip_prefix("_XLWS.") {
241            n = r.to_string();
242        } else {
243            break;
244        }
245    }
246    let (v, map) = registry();
247    map.get(n.as_str()).and_then(|&i| v.get(i))
248}
249
250/// Every registered function.
251pub fn all() -> &'static [FnSpec] {
252    &registry().0
253}
254
255/// Calls a function with array lifting applied. Wrong arity gives `#VALUE!`.
256pub fn call(spec: &FnSpec, args: &[Arg], ctx: &mut dyn Ctx) -> Value {
257    if !spec.accepts(args.len()) {
258        return Value::Error(CellError::Value);
259    }
260    lift::call_lifted(spec, args, ctx)
261}

267    #[test]
268    fn registry_is_consistent() {
269        let mut seen = std::collections::HashSet::new();
270        for s in all() {
271            assert!(seen.insert(s.name), "duplicate function {}", s.name);
272            assert_eq!(s.name, s.name.to_ascii_uppercase(), "{} not upper-case", s.name);
273            assert!(s.min_args <= s.max_args, "{} arity", s.name);
274            assert!(!s.description.is_empty() && !s.signature.is_empty(), "{} docs", s.name);
275        }
276        println!("registered functions: {}", all().len());
277        assert!(all().len() >= 400);
278        assert!(lookup("_xlfn._xlws.sort").is_some_and(|s| s.name == "SORT"));
279        assert!(lookup("nosuchfn").is_none());
280        for s in all() {
281            assert!(lookup(&format!("_xlfn.{}", s.name.to_ascii_lowercase())).is_some_and(|x| x.name == s.name));
282        }
283    }
