41/// The value-parameter mask of `name` (empty for functions without value parameters).
42pub(crate) fn value_args(name: &str) -> &'static [bool] {
43    VALUE_ARGS.iter().find(|(names, _)| names.split_ascii_whitespace().any(|n| n == name)).map_or(&[], |(_, mask)| mask)
44}

46/// A reference to more than one cell at a value parameter becomes a scalar `#VALUE!`, so it adds
47/// no shape: `EOMONTH(A1:A3, {1;2})` is a 2x1 array of `#VALUE!`, as in Excel.
48fn reject_multi_cell_refs(spec: &FnSpec, args: &[Arg]) -> Option<Vec<Arg>> {
49    let multi = |i: usize, a: &Arg| a.from_ref && spec.is_value_arg(i) && matches!(&a.value, Value::Array(arr) if arr.data.len() > 1);
50    if !args.iter().enumerate().any(|(i, a)| multi(i, a)) {
51        return None;
52    }
53    Some(args.iter().enumerate().map(|(i, a)| if multi(i, a) { Arg::val(Value::Error(CellError::Value)) } else { a.clone() }).collect())
54}

56/// Lookups whose every call searches a lookup vector from its start, so a call lifted over `n`
57/// needles does `n` x (vector length) work. Their search arguments don't lift.
58const SCANS: &[&str] = &["VLOOKUP", "HLOOKUP", "LOOKUP", "MATCH", "XLOOKUP", "XMATCH"];
59
60/// Work budget of a lifted [`SCANS`] lookup: needle cells times the length of the longest vector
61/// it searches (the longest dimension of its largest non-lifted array). Each needle re-reads and
62/// scans the vector (about 8 ns per entry), so without a cap a one-cell formula such as
63/// `XMATCH(SEQUENCE(1048576), -SEQUENCE(1048576))` runs for hours and hangs opening the file.
64/// Past the budget the call is `#NUM!`, as MMULT is past 2e9 multiply-adds. 5e8 keeps the worst
65/// case to a few seconds and still admits 5,000 needles over a 100,000-row table.
66///
67/// The same budget covers all the work of a lifted call added up across its elements: criteria
68/// aggregates rescanning their ranges, period loops, and array results built per element.
69const LIFTED_SCAN_BUDGET: usize = 500_000_000;
70
71/// Criteria aggregates: every call tests each cell of its ranges against its criteria, so a call
72/// lifted over `n` criteria does `n` x (range cells) work, like the [`SCANS`] lookups.
73const CRITERIA: &[&str] = &["COUNTIF", "COUNTIFS", "SUMIF", "SUMIFS", "AVERAGEIF", "AVERAGEIFS", "MAXIFS", "MINIFS"];
74
75/// Cumulative TVM functions that step through every period from `start_period` (argument 3) to
76/// `end_period` (argument 4) on each call.
77const PERIOD_LOOPS: &[&str] = &["CUMIPMT", "CUMPRINC"];
78
79/// Budget units one period of a [`PERIOD_LOOPS`] call costs: each period computes a payment and a
80/// balance with powers, many times a scanned cell.
81const PERIOD_COST: usize = 16;
82
83/// The work every lifted call of `spec` repeats over its non-lifted arguments: the longest vector
84/// a [`SCANS`] lookup searches, or every cell a [`CRITERIA`] aggregate tests. 0 for other
85/// functions.
86fn scan_len(spec: &FnSpec, args: &[Arg]) -> usize {
87    let scans = SCANS.contains(&spec.name);
88    if !scans && !CRITERIA.contains(&spec.name) {
89        return 0;
90    }
91    // SUMIF and AVERAGEIF read the optional third range only at the coordinates of the first.
92    let first = args.first().and_then(|a| if let Value::Array(arr) = &a.value { Some(arr.data.len()) } else { None });
93    let lens = args.iter().enumerate().filter(|(i, _)| !spec.is_scalar_arg(*i)).filter_map(|(i, a)| match &a.value {
94        Value::Array(arr) if scans => Some(arr.rows.max(arr.cols)),
95        Value::Array(arr) if i == 2 && matches!(spec.name, "SUMIF" | "AVERAGEIF") => Some(arr.data.len().min(first.unwrap_or(usize::MAX))),
96        Value::Array(arr) => Some(arr.data.len()),
97        _ => None,
98    });
99    if scans { lens.max().unwrap_or(0) } else { lens.fold(0, usize::saturating_add) }
100}

102/// The work of one lifted call that depends on its own (lifted) scalar arguments: the number of
103/// periods a [`PERIOD_LOOPS`] function steps through. 0 for other functions.
104fn call_cost(spec: &FnSpec, cell_args: &[Arg]) -> usize {
105    if !PERIOD_LOOPS.contains(&spec.name) {
106        return 0;
107    }
108    // Coercing numeric text reads all of it, even in a call the function then refuses, so each
109    // character is charged as well: of the arguments it coerces in order, up to the first that fails.
110    let mut parse = 0usize;
111    for (i, a) in cell_args.iter().enumerate().take(6) {
112        if let Value::Text(t) = &a.value {
113            parse = parse.saturating_add(t.len());
114        }
115        if crate::functions::util::num(cell_args, i).is_err() {
116            break;
117        }
118    }
119    // The function's own validation and coercion (text like "10000000" counts), so a call it
120    // refuses before its loop costs nothing and no loop goes uncharged.
121    let periods = crate::functions::financial::cumulative_args(cell_args).map_or(0, |(.., start, end, _)| (end - start + 1.0) as usize);
122    parse.saturating_add(periods.saturating_mul(PERIOD_COST))
123}

125pub(crate) fn call_lifted(spec: &FnSpec, args: &[Arg], ctx: &mut dyn Ctx) -> Value {
126    call_lifted_within(spec, args, ctx, LIFTED_SCAN_BUDGET)
127}

129/// Sets the lifted arguments of `cell_args` to their elements at (`r`, `c`).
130fn element_args(spec: &FnSpec, args: &[Arg], r: usize, c: usize, cell_args: &mut [Arg]) {
131    for (i, a) in args.iter().enumerate() {
132        if spec.is_scalar_arg(i)
133            && let Value::Array(arr) = &a.value
134            && let Some(slot) = cell_args.get_mut(i)
135        {
136            let v = arr.get_broadcast(r, c);
137            slot.value = if let Value::Array(inner) = &v { inner.data.first().cloned().unwrap_or(Value::Empty) } else { v };
138            // A blank element of a written array argument is given, as a scalar blank is.
139            slot.blank_value = !slot.from_ref && matches!(slot.value, Value::Empty);
140        }
141    }
142}

144/// [`call_lifted`] with every call's work added up against `budget` (see [`LIFTED_SCAN_BUDGET`]).
145fn call_lifted_within(spec: &FnSpec, args: &[Arg], ctx: &mut dyn Ctx, budget: usize) -> Value {
146    let rejected = if spec.value_args.is_empty() { None } else { reject_multi_cell_refs(spec, args) };
147    let args = rejected.as_deref().unwrap_or(args);
148    let mut rows = 0usize;
149    let mut cols = 0usize;
150    let mut any = false;
151    for (i, a) in args.iter().enumerate() {
152        if spec.is_scalar_arg(i)
153            && let Value::Array(arr) = &a.value
154        {
155            any = true;
156            rows = rows.max(arr.rows);
157            cols = cols.max(arr.cols);
158        }
159    }
160    if !any {
161        return (spec.imp)(args, ctx);
162    }
163    let needles = rows.saturating_mul(cols);
164    // Every call is charged its repeated scan (at least 1), its own period loop, and the cells of
165    // an array result (built in full, though only its first cell is kept), so lifting a scalar
166    // argument can't multiply what one call may do into hours of work:
167    // `COUNTIF(big_range, SEQUENCE(1e4))`, `CUMIPMT(0.1, 1e7, 1, 1, SEQUENCE(1e4,,1e7,0), 0)`,
168    // `MUNIT(SEQUENCE(1e4,,3000,0))`.
169    let per_call = scan_len(spec, args).max(1);
170    let mut cell_args: Vec<Arg> = args.to_vec();
171    let mut spent = needles.saturating_mul(per_call);
172    if needles > MAX_CELLS || spent > budget {
173        return Value::Error(CellError::Num);
174    }
175    // Period loops are known before any call runs, so a call over budget does none of them.
176    if PERIOD_LOOPS.contains(&spec.name) {
177        for r in 0..rows {
178            for c in 0..cols {
179                element_args(spec, args, r, c, &mut cell_args);
180                spent = spent.saturating_add(call_cost(spec, &cell_args));
181                if spent > budget {
182                    return Value::Error(CellError::Num);
183                }
184            }
185        }
186    }
187    let mut data = Vec::with_capacity(rows * cols);
188    for r in 0..rows {
189        for c in 0..cols {
190            element_args(spec, args, r, c, &mut cell_args);
191            let out = (spec.imp)(&cell_args, ctx);
192            data.push(match out {
193                Value::Array(a) => {
194                    spent = spent.saturating_add(a.data.len());
195                    if spent > budget {
196                        return Value::Error(CellError::Num);
197                    }
198                    a.data.first().cloned().unwrap_or(Value::Empty)
199                }
200                v => v,
201            });
202        }
203    }
204    if rows == 1 && cols == 1 {
205        return data.pop().unwrap_or(Value::Empty);
206    }
207    Array::new(rows, cols, data).map(Value::from).unwrap_or(Value::Error(CellError::Calc))
208}

254    /// Security: a lifted lookup searches its whole lookup vector once per needle, so needles x
255    /// vector length past `LIFTED_SCAN_BUDGET` is `#NUM!` at once.
256    /// `XMATCH(SEQUENCE(1048576), -SEQUENCE(1048576))` used to run for hours (and hung opening
257    /// the 1.6 KB file holding it).
258    #[test]
259    fn lifted_lookups_stop_past_their_work_budget() {
260        use crate::base::Array;
261        let tall = |n: usize| av(Array::filled(n, 1, nv(-1.0)).into());
262        let wide = |n: usize| av(Array::filled(1, n, nv(-1.0)).into());
263        is_err(ev("XMATCH", vec![tall(1_048_576), tall(1_048_576)]), CellError::Num);
264        // Just past the budget, for every lookup: 501 needles over 1,000,000 entries.
265        assert_eq!(super::LIFTED_SCAN_BUDGET, 500 * 1_000_000);
266        is_err(ev("XMATCH", vec![tall(501), tall(1_000_000)]), CellError::Num);
267        is_err(ev("MATCH", vec![tall(501), tall(1_000_000), n(0.0)]), CellError::Num);
268        is_err(ev("XLOOKUP", vec![tall(501), tall(1_000_000), tall(1_000_000)]), CellError::Num);
269        is_err(ev("VLOOKUP", vec![tall(501), tall(1_000_000), n(1.0), b(false)]), CellError::Num);
270        is_err(ev("HLOOKUP", vec![tall(501), wide(1_000_000), n(1.0), b(false)]), CellError::Num);
271        is_err(ev("LOOKUP", vec![tall(501), tall(1_000_000)]), CellError::Num);
272        // Within the budget lifted lookups work as before.
273        let keys = || av(col(&[10.0, 20.0, 30.0]));
274        let pos = ev("XMATCH", vec![av(col(&[30.0, 10.0, 99.0])), keys()]);
275        assert_eq!(rows_of(&pos), vec![vec![nv(3.0)], vec![nv(1.0)], vec![Value::Error(CellError::NA)]]);

277        assert_eq!(rows_of(&found), vec![vec![nv(2.0), nv(3.0)]]);
278        // Other lifted functions aren't metered: INDEX reads one entry per position.
279        let picks: Vec<f64> = (1..=1000u32).map(f64::from).collect();

281        assert_eq!(rows_of(&out), vec![vec![nv(-1.0)]; 1000]);
282    }

397    fn value_arg_table_names_registered_scalar_positions() {
398        for (names, mask) in super::VALUE_ARGS {
399            assert!(!mask.is_empty());
400            for name in names.split_ascii_whitespace() {
401                let Some(s) = crate::functions::lookup(name) else { panic!("{name} isn't registered") };
402                assert_eq!(s.value_args, *mask, "{name}");
403                for i in 0..s.max_args.min(12) {
404                    assert!(!s.is_value_arg(i) || s.is_scalar_arg(i), "{name} argument {i} is a value parameter but doesn't lift");
405                }
406            }
407        }
408        assert!(crate::functions::lookup("WORKDAY").is_some_and(|s| s.is_value_arg(1) && !s.is_value_arg(2)));
409        assert!(crate::functions::lookup("WORKDAY.INTL").is_some_and(|s| !s.is_value_arg(2) && s.is_scalar_arg(2)));
410        assert!(crate::functions::lookup("SUM").is_some_and(|s| s.value_args.is_empty()));
411        assert!(crate::functions::lookup("ISOWEEKNUM").is_some_and(|s| s.value_args.is_empty()));
