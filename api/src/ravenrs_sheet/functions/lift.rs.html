41/// The value-parameter mask of `name` (empty for functions without value parameters).
42pub(crate) fn value_args(name: &str) -> &'static [bool] {
43    VALUE_ARGS.iter().find(|(names, _)| names.split_ascii_whitespace().any(|n| n == name)).map_or(&[], |(_, mask)| mask)
44}

46/// A reference to more than one cell at a value parameter becomes a scalar `#VALUE!`, so it adds
47/// no shape: `EOMONTH(A1:A3, {1;2})` is a 2x1 array of `#VALUE!`, as in Excel.
48fn reject_multi_cell_refs(spec: &FnSpec, args: &[Arg]) -> Option<Vec<Arg>> {
49    let multi = |i: usize, a: &Arg| a.from_ref && spec.is_value_arg(i) && matches!(&a.value, Value::Array(arr) if arr.data.len() > 1);
50    if !args.iter().enumerate().any(|(i, a)| multi(i, a)) {
51        return None;
52    }
53    Some(
54        args.iter()
55            .enumerate()
56            .map(|(i, a)| if multi(i, a) { Arg { value: Value::Error(CellError::Value), from_ref: false } } else { a.clone() })
57            .collect(),
58    )
59}

61/// Lookups whose every call searches a lookup vector from its start, so a call lifted over `n`
62/// needles does `n` x (vector length) work. Their search arguments don't lift.
63const SCANS: &[&str] = &["VLOOKUP", "HLOOKUP", "LOOKUP", "MATCH", "XLOOKUP", "XMATCH"];
64
65/// Work budget of a lifted [`SCANS`] lookup: needle cells times the length of the longest vector
66/// it searches (the longest dimension of its largest non-lifted array). Each needle re-reads and
67/// scans the vector (about 8 ns per entry), so without a cap a one-cell formula such as
68/// `XMATCH(SEQUENCE(1048576), -SEQUENCE(1048576))` runs for hours and hangs opening the file.
69/// Past the budget the call is `#NUM!`, as MMULT is past 2e9 multiply-adds. 5e8 keeps the worst
70/// case to a few seconds and still admits 5,000 needles over a 100,000-row table.
71const LIFTED_SCAN_BUDGET: usize = 500_000_000;
72
73/// The length of the longest vector a lifted call of `spec` searches per needle: 0 unless `spec`
74/// is one of the [`SCANS`] lookups and a non-lifted argument is an array.
75fn scan_len(spec: &FnSpec, args: &[Arg]) -> usize {
76    if !SCANS.contains(&spec.name) {
77        return 0;
78    }
79    args.iter()
80        .enumerate()
81        .filter(|(i, _)| !spec.is_scalar_arg(*i))
82        .filter_map(|(_, a)| match &a.value {
83            Value::Array(arr) => Some(arr.rows.max(arr.cols)),
84            _ => None,
85        })
86        .max()
87        .unwrap_or(0)
88}

90pub(crate) fn call_lifted(spec: &FnSpec, args: &[Arg], ctx: &mut dyn Ctx) -> Value {
91    let rejected = if spec.value_args.is_empty() { None } else { reject_multi_cell_refs(spec, args) };
92    let args = rejected.as_deref().unwrap_or(args);
93    let mut rows = 0usize;
94    let mut cols = 0usize;
95    let mut any = false;
96    for (i, a) in args.iter().enumerate() {
97        if spec.is_scalar_arg(i)
98            && let Value::Array(arr) = &a.value
99        {
100            any = true;
101            rows = rows.max(arr.rows);
102            cols = cols.max(arr.cols);
103        }
104    }
105    if !any {
106        return (spec.imp)(args, ctx);
107    }
108    let needles = rows.saturating_mul(cols);
109    if needles > MAX_CELLS || needles.saturating_mul(scan_len(spec, args)) > LIFTED_SCAN_BUDGET {
110        return Value::Error(CellError::Num);
111    }
112    let mut cell_args: Vec<Arg> = args.to_vec();
113    let mut data = Vec::with_capacity(rows * cols);
114    for r in 0..rows {
115        for c in 0..cols {
116            for (i, a) in args.iter().enumerate() {
117                if spec.is_scalar_arg(i)
118                    && let Value::Array(arr) = &a.value
119                    && let Some(slot) = cell_args.get_mut(i)
120                {
121                    let v = arr.get_broadcast(r, c);
122                    slot.value = if let Value::Array(inner) = &v { inner.data.first().cloned().unwrap_or(Value::Empty) } else { v };
123                }
124            }
125            let out = (spec.imp)(&cell_args, ctx);
126            data.push(match out {
127                Value::Array(a) => a.data.first().cloned().unwrap_or(Value::Empty),
128                v => v,
129            });
130        }
131    }
132    if rows == 1 && cols == 1 {
133        return data.pop().unwrap_or(Value::Empty);
134    }
135    Array::new(rows, cols, data).map(Value::from).unwrap_or(Value::Error(CellError::Calc))
136}

182    /// Security: a lifted lookup searches its whole lookup vector once per needle, so needles x
183    /// vector length past `LIFTED_SCAN_BUDGET` is `#NUM!` at once.
184    /// `XMATCH(SEQUENCE(1048576), -SEQUENCE(1048576))` used to run for hours (and hung opening
185    /// the 1.6 KB file holding it).
186    #[test]
187    fn lifted_lookups_stop_past_their_work_budget() {
188        use crate::base::Array;
189        let tall = |n: usize| av(Array::filled(n, 1, nv(-1.0)).into());
190        let wide = |n: usize| av(Array::filled(1, n, nv(-1.0)).into());
191        is_err(ev("XMATCH", vec![tall(1_048_576), tall(1_048_576)]), CellError::Num);
192        // Just past the budget, for every lookup: 501 needles over 1,000,000 entries.
193        assert_eq!(super::LIFTED_SCAN_BUDGET, 500 * 1_000_000);
194        is_err(ev("XMATCH", vec![tall(501), tall(1_000_000)]), CellError::Num);
195        is_err(ev("MATCH", vec![tall(501), tall(1_000_000), n(0.0)]), CellError::Num);
196        is_err(ev("XLOOKUP", vec![tall(501), tall(1_000_000), tall(1_000_000)]), CellError::Num);
197        is_err(ev("VLOOKUP", vec![tall(501), tall(1_000_000), n(1.0), b(false)]), CellError::Num);
198        is_err(ev("HLOOKUP", vec![tall(501), wide(1_000_000), n(1.0), b(false)]), CellError::Num);
199        is_err(ev("LOOKUP", vec![tall(501), tall(1_000_000)]), CellError::Num);
200        // Within the budget lifted lookups work as before.
201        let keys = || av(col(&[10.0, 20.0, 30.0]));
202        let pos = ev("XMATCH", vec![av(col(&[30.0, 10.0, 99.0])), keys()]);
203        assert_eq!(rows_of(&pos), vec![vec![nv(3.0)], vec![nv(1.0)], vec![Value::Error(CellError::NA)]]);
204        let found = ev("XLOOKUP", vec![av(row(&[20.0, 30.0])), keys(), av(col(&[1.0, 2.0, 3.0]))]);
205        assert_eq!(rows_of(&found), vec![vec![nv(2.0), nv(3.0)]]);
206        // Other lifted functions aren't metered: INDEX reads one entry per position.
207        let picks: Vec<f64> = (1..=1000u32).map(f64::from).collect();
208        let out = ev("INDEX", vec![tall(1_048_576), av(col(&picks))]);
209        assert_eq!(rows_of(&out), vec![vec![nv(-1.0)]; 1000]);
210    }

212    #[test]
213    fn value_params_reject_multi_cell_references() {
214        let s = FnSpec { value_args: &[true, false], ..spec() };
215        let mut ctx = TestCtx::default();
216        // A multi-cell reference is #VALUE!; a one-cell reference and a computed array are fine.
217        is_err(crate::functions::call(&s, &[rf(col(&[1.0, 2.0, 3.0])), n(1.0)], &mut ctx), CellError::Value);
218        assert_eq!(crate::functions::call(&s, &[rf(col(&[4.0])), n(1.0)], &mut ctx), nv(5.0));
219        assert_eq!(rows_of(&crate::functions::call(&s, &[av(col(&[1.0, 2.0])), n(1.0)], &mut ctx)), vec![vec![nv(2.0)], vec![nv(3.0)]]);
220        // The rejected reference adds no shape: the other argument still lifts.
221        let v = Value::Error(CellError::Value);
222        assert_eq!(
223            rows_of(&crate::functions::call(&s, &[rf(col(&[1.0, 2.0, 3.0])), av(col(&[1.0, 2.0]))], &mut ctx)),
224            vec![vec![v.clone()], vec![v]]
225        );
226        // A position that isn't a value parameter still lifts the reference.
227        assert_eq!(rows_of(&crate::functions::call(&s, &[n(1.0), rf(row(&[1.0, 2.0]))], &mut ctx)), vec![vec![nv(2.0), nv(3.0)]]);
228    }

230    #[test]
231    fn value_arg_table_names_registered_scalar_positions() {
232        for (names, mask) in super::VALUE_ARGS {
233            assert!(!mask.is_empty());
234            for name in names.split_ascii_whitespace() {
235                let Some(s) = crate::functions::lookup(name) else { panic!("{name} isn't registered") };
236                assert_eq!(s.value_args, *mask, "{name}");
237                for i in 0..s.max_args.min(12) {
238                    assert!(!s.is_value_arg(i) || s.is_scalar_arg(i), "{name} argument {i} is a value parameter but doesn't lift");
239                }
240            }
241        }
242        assert!(crate::functions::lookup("WORKDAY").is_some_and(|s| s.is_value_arg(1) && !s.is_value_arg(2)));
243        assert!(crate::functions::lookup("WORKDAY.INTL").is_some_and(|s| !s.is_value_arg(2) && s.is_scalar_arg(2)));
244        assert!(crate::functions::lookup("SUM").is_some_and(|s| s.value_args.is_empty()));
245        assert!(crate::functions::lookup("ISOWEEKNUM").is_some_and(|s| s.value_args.is_empty()));
246    }
