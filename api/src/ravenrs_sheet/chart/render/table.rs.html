7pub(super) struct Layout {
8    top: f32,
9    row_height: f32,
10    headers: usize,
11    stub: f32,
12    columns: usize,
13}

15pub(super) fn category_axis<'a>(layout: &Option<Layout>, labels: &'a [String], outer: &'a [Vec<String>]) -> CatAxis<'a> {
16    match layout {
17        Some(t) => CatAxis::Table(labels, t.stub),
18        None => CatAxis::Labels(labels, outer),
19    }
20}

22/// Reserve at most half the available height. A table that cannot fit is explicitly labelled
23/// instead of dropping rows or allocating a series × point-sized matrix.
24pub(super) fn reserve(ctx: &mut Ctx, chart: &Chart, data: &ChartData, mut area: Bx, columns: usize) -> (Bx, Option<Layout>) {
25    if chart.data_table.is_none() || !chart.supports_data_table() {
26        return (area, None);
27    }
28    let headers = data.category_levels.len() + 1;
29    let rows = headers + data.series.len();
30    let row_height = LABEL_SIZE * 1.4 + 4.0;
31    let height = row_height * rows as f32;
32    let key_width = if chart.data_table.as_ref().is_some_and(|t| t.show_keys) { 22.0 } else { 0.0 };
33    let names = data.series.iter().take(32).map(|s| ctx.tw(&s.name, LABEL_SIZE)).fold(0.0f32, f32::max);
34    let stub = (names + key_width + 10.0).clamp(24.0, (area.w * 0.3).max(24.0));
35    if columns == 0 || columns > 128 || data.series.len() > 32 || height > area.h * 0.5 || (area.w * 0.6) / (columns as f32) < 18.0 {
36        let message = ctx.fit("Data table needs more space", LABEL_SIZE, area.w);
37        ctx.text(area.cx(), area.bottom(), &message, LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Bottom));
38        area.h -= row_height;
39        return (area, None);
40    }
41    area.h -= height;
42    (area, Some(Layout { top: area.bottom(), row_height, headers, stub, columns }))
43}

45pub(super) fn draw(ctx: &mut Ctx, chart: &Chart, data: &ChartData, plot: Bx, layout: Option<&Layout>) {
46    let Some(layout) = layout else { return };
47    let Some(settings) = &chart.data_table else { return };
48    let x0 = plot.x - layout.stub;
49    let top = layout.top;
50    let rows = layout.headers + data.series.len();
51    let bottom = top + layout.row_height * rows as f32;
52    let cw = plot.w / layout.columns as f32;
53    // Outermost category level first, then the individual categories.
54    for (i, level) in data.category_levels.iter().rev().enumerate() {
55        for (first, end) in level_groups(level, layout.columns) {
56            let text = level.get(first).map(String::as_str).unwrap_or("");
57            let text = ctx.fit(text, LABEL_SIZE, (end - first) as f32 * cw - 4.0);
58            let x = plot.x + (first + end) as f32 * cw / 2.0;
59            ctx.text(x, top + (i as f32 + 0.5) * layout.row_height, &text, LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Middle));
60        }
61    }
62    let header_y = top + (layout.headers as f32 - 0.5) * layout.row_height;
63    let n = data.categories.len().max(data.series.iter().map(|s| s.values.len()).max().unwrap_or(0));
64    for j in 0..layout.columns {
65        let text = if j < n { cat_label(data, j) } else { String::new() };
66        let text = ctx.fit(&text, LABEL_SIZE, cw - 4.0);
67        ctx.text(plot.x + (j as f32 + 0.5) * cw, header_y, &text, LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Middle));
68    }
69    for (i, series) in data.series.iter().enumerate() {
70        let y = top + (layout.headers as f32 + i as f32 + 0.5) * layout.row_height;
71        let name_x = if settings.show_keys { x0 + 25.0 } else { x0 + 4.0 };
72        if settings.show_keys {
73            let swatch = match plot_kind(chart, series, i) {
74                ChartKind::Line | ChartKind::LineStacked => Swatch::Line,
75                ChartKind::LineMarkers => Swatch::LineMarker,
76                _ => Swatch::Box,
77            };
78            draw_swatch(ctx, swatch, x0 + 4.0, y, 7.0, series.color);
79        }
80        let name = ctx.fit(&series.name, LABEL_SIZE, plot.x - name_x - 4.0);
81        ctx.text(name_x, y, &name, LABEL_SIZE, TEXT, (HAlign::Left, VAlign::Middle));
82        let fmt = Fmt::new(&series.number_format);
83        for j in 0..layout.columns {
84            let value = series.values.get(j).copied().flatten().map(|v| fmt.format(v, None)).unwrap_or_default();
85            let text = ctx.fit(&value, LABEL_SIZE, cw - 4.0);
86            ctx.text(plot.x + (j as f32 + 0.5) * cw, y, &text, LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Middle));
87        }
88    }
89    if settings.horizontal_border {
90        for row in 1..rows {
91            ctx.hline(x0, plot.right(), top + row as f32 * layout.row_height, AXIS, 0.75);
92        }
93    }
94    if settings.vertical_border {
95        ctx.vline(plot.x, top, bottom, AXIS, 0.75);
96        // The outer headers span groups; category dividers begin at the innermost header.
97        for j in 1..layout.columns {
98            ctx.vline(plot.x + j as f32 * cw, top + (layout.headers - 1) as f32 * layout.row_height, bottom, AXIS, 0.75);
99        }
100    }
101    if settings.outline {
102        ctx.hline(x0, plot.right(), top, AXIS, 0.75);
103        ctx.hline(x0, plot.right(), bottom, AXIS, 0.75);
104        ctx.vline(x0, top, bottom, AXIS, 0.75);
105        ctx.vline(plot.right(), top, bottom, AXIS, 0.75);
106    }
107}
