10pub(super) fn histogram(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
11    let Some(s) = data.series.first() else { return };
12    let vals: Vec<f64> = s.values.iter().flatten().copied().collect();
13    let Some((lo, hi)) = value_range(vals.iter().copied()) else { return };
14    let n = vals.len();
15    let span = hi - lo;
16    let mut bins = ((n as f64).sqrt().ceil() as usize).clamp(1, 100);
17    if !(span.is_finite() && span > 0.0) {
18        bins = 1;
19    }
20    let width = if bins == 1 && span.partial_cmp(&0.0) != Some(std::cmp::Ordering::Greater) { 1.0 } else { span / bins as f64 };
21    // Upper edge of each bin, exactly as shown in the labels below. Bins are right-inclusive (the first is closed on both sides), so a value
22    // equal to an edge belongs to the bin that ends there.
23    let edges: Vec<f64> = (0..bins).map(|b| if b + 1 == bins { hi.max(lo + b as f64 * width) } else { lo + (b + 1) as f64 * width }).collect();
24    let mut counts = vec![0usize; bins];
25    for v in &vals {
26        let b = edges.partition_point(|e| e < v).min(bins - 1);
27        if let Some(c) = counts.get_mut(b) {
28            *c += 1;
29        }
30    }
31    let f = Fmt::new(&s.number_format);
32    let labels: Vec<String> = (0..bins)
33        .map(|b| {
34            let a = lo + b as f64 * width;
35            let e = edges.get(b).copied().unwrap_or(a);
36            let open = if b == 0 { '[' } else { '(' };
37            format!("{open}{}, {}]", f.format(a, Some(width / 10.0)), f.format(e, Some(width / 10.0)))
38        })
39        .collect();
40    let maxc = counts.iter().copied().max().unwrap_or(0) as f64;
41    let ys = nice_scale(0.0, maxc, true);
42    let fc = Fmt::new("General");
43    let plot = frame(ctx, area, CatAxis::Labels(&labels, &[]), &ValAxis { scale: ys, fmt: &fc }, None, false, chart.gridlines);
44    if !plot.ok() {
45        return;
46    }
47    let cw = plot.w / bins as f32;
48    for (b, c) in counts.iter().enumerate() {
49        let top = plot.bottom() - ys.frac(*c as f64) * plot.h;
50        let x = plot.x + b as f32 * cw;
51        ctx.rect(x, top, cw, plot.bottom() - top, s.color, Some((WHITE, 0.75)));
52        if chart.data_labels {
53            ctx.text(x + cw / 2.0, top - 2.0, &c.to_string(), LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Bottom));
54        }
55    }
56}

60pub(super) fn waterfall(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
61    let Some(s) = data.series.first() else { return };
62    let mut ncat = data.categories.len().max(s.values.len()).min(MAX_CATS);
63    if ncat == 0 {
64        return;
65    }
66    let mut labels: Vec<String> = (0..ncat).map(|j| cat_label(data, j)).collect();
67    let mut vals: Vec<f64> = (0..ncat).map(|j| s.values.get(j).copied().flatten().unwrap_or(0.0)).collect();
68    let mut is_total = vec![false; ncat];
69    let last_total = labels.last().is_some_and(|l| l.to_lowercase().contains("total"));
70    if let Some(t) = is_total.last_mut() {
71        *t = last_total;
72    }
73    if !last_total {
74        labels.push("Total".into());
75        vals.push(0.0);
76        is_total.push(true);
77        ncat += 1;
78    }
79    let mut bars = Vec::with_capacity(ncat);
80    let mut cum = 0.0f64;
81    for (j, v) in vals.iter().enumerate() {
82        if is_total.get(j).copied().unwrap_or(false) {
83            let total = if last_total && j + 1 == ncat && *v != 0.0 { *v } else { cum };
84            bars.push((0.0, total, 0u8));
85            cum = total;
86        } else {
87            let top = clean(cum + v);
88            bars.push((cum, top, if *v >= 0.0 { 1 } else { 2 }));
89            cum = top;
90        }
91    }
92    let (lo, hi) = value_range(bars.iter().flat_map(|(b, t, _)| [*b, *t])).unwrap_or((0.0, 0.0));
93    let ys = nice_scale(lo, hi, true);
94    let f = Fmt::new(&s.number_format);
95    let plot = frame(ctx, area, CatAxis::Labels(&labels, &[]), &ValAxis { scale: ys, fmt: &f }, None, false, chart.gridlines);
96    if !plot.ok() {
97        return;
98    }
99    let cw = plot.w / ncat as f32;
100    let bw = cw / 1.5;
101    let vy = |v: f64| plot.bottom() - ys.frac(v) * plot.h;
102    for (j, (b, t, kind)) in bars.iter().enumerate() {
103        let color = match kind {
104            1 => UP,
105            2 => DOWN,
106            _ => s.color,
107        };
108        let x = plot.x + (j as f32 + 0.5) * cw - bw / 2.0;
109        let (yb, yt) = (vy(*b), vy(*t));
110        ctx.rect(x, yb.min(yt), bw, (yb - yt).abs().max(0.5), color, None);
111        if j + 1 < bars.len() && ncat <= 500 {
112            ctx.out.push(Prim::Line { pts: vec![[x + bw, yt], [x + cw, yt]], color: AXIS, width: 0.75, dash: true });
113        }
114        if chart.data_labels && ncat <= MAX_LABELS {
115            let v = if *kind == 0 { *t } else { t - b };
116            ctx.text(x + bw / 2.0, yb.min(yt) - 2.0, &f.format(v, None), LABEL_SIZE, TEXT, (HAlign::Center, VAlign::Bottom));
117        }
118    }
119}

123pub(super) fn funnel(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
124    let Some(s) = data.series.first() else { return };
125    let n = data.categories.len().max(s.values.len()).min(MAX_CATS);
126    if n == 0 {
127        return;
128    }
129    let vals: Vec<f64> = (0..n).map(|j| s.values.get(j).copied().flatten().unwrap_or(0.0).abs()).collect();
130    let max = vals.iter().copied().fold(0.0f64, f64::max);
131    if max.is_nan() || max <= 0.0 {
132        return;
133    }
134    let labels: Vec<String> = (0..n).map(|j| cat_label(data, j)).collect();
135    let lw = (labels.iter().take(2000).map(|t| ctx.tw(t, LABEL_SIZE)).fold(0.0f32, f32::max) + 8.0).min(area.w * 0.35);
136    let plot = Bx { x: area.x + lw, w: area.w - lw, ..area };
137    if !plot.ok() {
138        return;
139    }
140    let slot = plot.h / n as f32;
141    let bh = slot * 0.85;
142    let f = Fmt::new(&s.number_format);
143    let show = slot >= LABEL_SIZE * 0.9;
144    for (j, v) in vals.iter().enumerate() {
145        let bw = (*v / max) as f32 * plot.w;
146        let x = plot.x + (plot.w - bw) / 2.0;
147        let y = plot.y + j as f32 * slot + (slot - bh) / 2.0;
148        if bw > 0.0 {
149            ctx.rect(x, y, bw, bh, s.color, None);
150        }
151        if show {
152            let yc = y + bh / 2.0;
153            if chart.data_labels {
154                let t = f.format(s.values.get(j).copied().flatten().unwrap_or(0.0), None);
155                let color = if ctx.tw(&t, LABEL_SIZE) + 4.0 <= bw { on(s.color) } else { TEXT };
156                ctx.text(plot.cx(), yc, &t, LABEL_SIZE, color, (HAlign::Center, VAlign::Middle));
157            }
158            let l = ctx.fit(labels.get(j).map(String::as_str).unwrap_or(""), LABEL_SIZE, lw - 8.0);
159            ctx.text(plot.x - 6.0, yc, &l, LABEL_SIZE, TEXT, (HAlign::Right, VAlign::Middle));
160        }
161    }
162}

166/// At most `cap` of the largest values, in category order. When a ring has more categories than
167/// its share of the wedge budget allows, the big slices survive rather than the first ones.
168fn largest_slices(vals: &[f64], cap: usize) -> Vec<(usize, f64)> {
169    let mut idx: Vec<usize> = (0..vals.len()).collect();
170    if idx.len() > cap {
171        // Partition descends by value, so the first `cap` entries are the largest ones.
172        idx.select_nth_unstable_by(cap - 1, |a, b| vals[*b].total_cmp(&vals[*a]));
173        idx.truncate(cap);
174        idx.sort_unstable();
175    }
176    idx.into_iter().map(|j| (j, vals[j])).collect()
177}

179/// One ring's geometry and separator stroke: the slices of a ring differ only in angle and colour.
180struct Ring {
181    cx: f32,
182    cy: f32,
183    outer: f32,
184    inner: f32,
185    stroke: Option<(Rgba, f32)>,
186}
187
188/// A wedge of `ring`, split at 12 o'clock when it crosses the end of the circle.
189fn wedge(ctx: &mut Ctx, ring: &Ring, a0: f32, sweep: f32, fill: Rgba) {
190    let Ring { cx, cy, outer, inner, stroke } = *ring;
191    if sweep <= 1e-4 {
192        return;
193    }
194    // A slice that starts past 12 o'clock still has to be split against 12 o'clock, not against
195    // its unwrapped start: otherwise the half after TAU is drawn from 0 over its neighbours.
196    let a0 = a0.rem_euclid(TAU);
197    let a1 = a0 + sweep;
198    if a1 <= TAU {
199        ctx.out.push(Prim::Wedge { cx, cy, r_outer: outer, r_inner: inner, a0, a1, fill, stroke });
200        return;
201    }
202    // The halves are filled without a stroke: each would outline its own radial edge at 12
203    // o'clock, painting the white separator straight through the slice.
204    ctx.out.push(Prim::Wedge { cx, cy, r_outer: outer, r_inner: inner, a0, a1: TAU, fill, stroke: None });
205    ctx.out.push(Prim::Wedge { cx, cy, r_outer: outer, r_inner: inner, a0: 0.0, a1: a1 - TAU, fill, stroke: None });
206    if let Some((color, width)) = stroke {
207        // One outline for the whole slice: the outer and inner arcs run through 12 o'clock, and
208        // the two radial edges at the slice's own start and end close the path.
209        let arc = |r: f32, from: f32, to: f32| -> Vec<[f32; 2]> {
210            let steps = ((r.abs() * (to - from).abs() / 2.0).ceil() as usize).clamp(4, 512);
211            let at = |a: f32| [cx + r * a.sin(), cy - r * a.cos()];
212            (0..=steps).map(|k| at(from + (to - from) * k as f32 / steps as f32)).collect()
213        };
214        let mut pts = arc(outer, a0, a1);
215        if inner > 0.0 {
216            pts.extend(arc(inner, a1, a0));
217        } else {
218            pts.push([cx, cy]);
219        }
220        let start = pts[0];
221        pts.push(start);
222        ctx.out.push(Prim::Line { pts, color, width, dash: false });
223    }
224}

226pub(super) fn pie(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx, inner: f32, names: bool) {
227    let Some(s) = data.series.first() else { return };
228    let vals: Vec<f64> = s.values.iter().take(MAX_CATS).map(|v| v.map(f64::abs).unwrap_or(0.0)).collect();
229    let total: f64 = vals.iter().sum();
230    if !(total.is_finite() && total > 0.0) {
231        return;
232    }
233    let (cx, cy) = (area.cx(), area.cy());
234    let r = area.w.min(area.h) / 2.0 * 0.92;
235    if r < 2.0 {
236        return;
237    }
238    let ri = r * inner;
239    let ring = Ring { cx, cy, outer: r, inner: ri, stroke: Some((WHITE, 1.5)) };
240    let f = Fmt::new(&s.number_format);
241    let mut a = (chart.first_slice_angle % 360) as f32 * TAU / 360.0;
242    let mut labels = Vec::new();
243    for (j, v) in vals.iter().enumerate() {
244        let sweep = (*v / total) as f32 * TAU;
245        if sweep > 1e-4 {
246            let fill = point_color(data, j);
247            wedge(ctx, &ring, a, sweep, fill);
248            if (chart.data_labels || names) && vals.len() <= MAX_LABELS {
249                let t = if names { cat_label(data, j) } else { f.format(s.values.get(j).copied().flatten().unwrap_or(0.0), None) };
250                let mid = a + sweep / 2.0;
251                let rr = if ri > 0.0 { (r + ri) / 2.0 } else { r * 0.65 };
252                let room = sweep * rr;
253                if ctx.tw(&t, LABEL_SIZE) <= room.max(r - ri) && room >= LABEL_SIZE {
254                    labels.push((cx + rr * mid.sin(), cy - rr * mid.cos(), t, on(fill)));
255                }
256            }
257        }
258        a += sweep;
259    }
260    for (x, y, t, c) in labels {
261        ctx.text(x, y, &t, LABEL_SIZE, c, (HAlign::Center, VAlign::Middle));
262    }
263}

265/// Excel's doughnut: every series is one concentric ring, the first series innermost (measured in
266/// Excel: ring outer radii ≈0.83, 0.92 and 1.0 of the plot radius for series 1, 2 and 3). The hole
267/// size and first slice angle come from the chart; each ring is scaled to its own total.
268pub(super) fn doughnut(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
269    let nser = data.series.len().min(MAX_SERIES);
270    if nser == 0 {
271        return;
272    }
273    let (cx, cy) = (area.cx(), area.cy());
274    let r = area.w.min(area.h) / 2.0 * 0.92;
275    if r < 2.0 {
276        return;
277    }
278    let hole = f32::from(chart.hole_size.clamp(10, 90)) / 100.0;
279    let ri = r * hole;
280    let thick = (r - ri) / nser as f32;
281    // A ring thinner than a pixel would vanish. Rather than suppressing the whole chart when there
282    // are more series than the annulus has pixels, keep the outer rings at one pixel each and drop
283    // the innermost series that no longer fit: Excel's order makes the first series the innermost,
284    // so the outermost rings are the ones worth keeping.
285    let band = thick.max(1.0);
286    let start = (chart.first_slice_angle % 360) as f32 * TAU / 360.0;
287    // Only rings with a pixel of band are drawn, and the wedge budget is shared across them in
288    // drawing order: the innermost ring (the first series) gets its slices first, and the outer
289    // rings are capped instead of the series × category product growing without bound.
290    let rings: Vec<(&SeriesData, f32, f32)> = data
291        .series
292        .iter()
293        .take(nser)
294        .enumerate()
295        .filter_map(|(si, s)| {
296            let outer = (r - (nser - 1 - si) as f32 * band).min(r);
297            let inner = (r - (nser - si) as f32 * band).max(ri);
298            (outer - inner >= 1.0).then_some((s, outer, inner))
299        })
300        .collect();
301    let mut budget = MAX_WEDGES;
302    let mut labels = Vec::new();
303    for (k, (s, outer, inner)) in rings.iter().enumerate() {
304        // The white separator is 1.5 units wide, wider than a one-pixel band: on a thin ring it
305        // would paint over the fill of this ring and its neighbour, so such rings keep only fill.
306        let stroke = if outer - inner > 1.0 { Some((WHITE, 1.5)) } else { None };
307        let ring = Ring { cx, cy, outer: *outer, inner: *inner, stroke };
308        let vals: Vec<f64> = s.values.iter().take(MAX_CATS).map(|v| v.map(f64::abs).unwrap_or(0.0)).collect();
309        let cap = (budget / (rings.len() - k)).max(1);
310        let slices = largest_slices(&vals, cap);
311        let total: f64 = slices.iter().map(|(_, v)| *v).sum();
312        if !(total.is_finite() && total > 0.0) {
313            continue;
314        }
315        let n = slices.len();
316        budget = budget.saturating_sub(n);
317        let f = Fmt::new(&s.number_format);
318        let mut a = start;
319        for (j, v) in slices {
320            let sweep = (v / total) as f32 * TAU;
321            if sweep > 1e-4 {
322                let fill = point_color(data, j);
323                wedge(ctx, &ring, a, sweep, fill);
324                if chart.data_labels && n <= MAX_LABELS {
325                    let t = f.format(s.values.get(j).copied().flatten().unwrap_or(0.0), None);
326                    let mid = a + sweep / 2.0;
327                    let rr = (outer + inner) / 2.0;
328                    if ctx.tw(&t, LABEL_SIZE) <= (sweep * rr).max(outer - inner) && sweep * rr >= LABEL_SIZE {
329                        labels.push((cx + rr * mid.sin(), cy - rr * mid.cos(), t, on(fill)));
330                    }
331                }
332            }
333            a += sweep;
334        }
335    }
336    for (x, y, t, c) in labels {
337        ctx.text(x, y, &t, LABEL_SIZE, c, (HAlign::Center, VAlign::Middle));
338    }
339}

343/// Worst aspect ratio of a row with the given sum, smallest and largest area.
344fn worst(s: f64, mn: f64, mx: f64, side: f64) -> f64 {
345    if s <= 0.0 || side <= 0.0 {
346        return f64::INFINITY;
347    }
348    let s2 = s * s;
349    let w2 = side * side;
350    (w2 * mx / s2).max(s2 / (w2 * mn))
351}

353/// Squarified treemap layout of (index, area) items, sorted descending, into `r`.
354fn squarify(items: &[(usize, f64)], r: Bx) -> Vec<(usize, Bx)> {
355    let total: f64 = items.iter().map(|i| i.1).sum();
356    let area = (r.w as f64) * (r.h as f64);
357    if !(total > 0.0 && area > 0.0) {
358        return vec![];
359    }
360    let scale = area / total;
361    let mut out = Vec::with_capacity(items.len());
362    let mut rem = r;
363    let mut row: Vec<(usize, f64)> = Vec::new();
364    // The row's sum, smallest and largest area, kept as items join so each step is O(1).
365    let (mut sum, mut mn, mut mx) = (0.0f64, f64::INFINITY, 0.0f64);
366    let mut i = 0;
367    let layout = |row: &[(usize, f64)], rem: &mut Bx, out: &mut Vec<(usize, Bx)>| {
368        let s: f64 = row.iter().map(|x| x.1).sum();
369        if s <= 0.0 {
370            return;
371        }
372        if rem.w >= rem.h {
373            let cw = (s / rem.h.max(1e-6) as f64) as f32;
374            let mut y = rem.y;
375            for (j, a) in row {
376                let h = (*a / cw.max(1e-6) as f64) as f32;
377                out.push((*j, Bx { x: rem.x, y, w: cw, h }));
378                y += h;
379            }
380            rem.x += cw;
381            rem.w = (rem.w - cw).max(0.0);
382        } else {
383            let rh = (s / rem.w.max(1e-6) as f64) as f32;
384            let mut x = rem.x;
385            for (j, a) in row {
386                let w = (*a / rh.max(1e-6) as f64) as f32;
387                out.push((*j, Bx { x, y: rem.y, w, h: rh }));
388                x += w;
389            }
390            rem.y += rh;
391            rem.h = (rem.h - rh).max(0.0);
392        }
393    };
394    while let Some(&(j, v)) = items.get(i) {
395        let a = v * scale;
396        let side = rem.w.min(rem.h) as f64;
397        if row.is_empty() || worst(sum + a, mn.min(a), mx.max(a), side) <= worst(sum, mn, mx, side) {
398            row.push((j, a));
399            (sum, mn, mx) = (sum + a, mn.min(a), mx.max(a));
400            i += 1;
401        } else {
402            layout(&row, &mut rem, &mut out);
403            row.clear();
404            (sum, mn, mx) = (0.0, f64::INFINITY, 0.0);
405        }
406    }
407    if !row.is_empty() {
408        layout(&row, &mut rem, &mut out);
409    }
410    out
411}

413pub(super) fn treemap(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
414    let Some(s) = data.series.first() else { return };
415    let mut items: Vec<(usize, f64)> =
416        s.values.iter().take(MAX_MARKERS).enumerate().filter_map(|(j, v)| v.filter(|v| *v > 0.0).map(|v| (j, v))).collect();
417    items.sort_by(|a, b| b.1.total_cmp(&a.1));
418    let f = Fmt::new(&s.number_format);
419    let value_of: std::collections::HashMap<usize, f64> = items.iter().copied().collect();
420    for (j, b) in squarify(&items, area) {
421        let fill = point_color(data, j);
422        ctx.rect(b.x, b.y, b.w, b.h, fill, Some((WHITE, 1.5)));
423        if b.w > 24.0 && b.h > LABEL_SIZE * 1.6 {
424            let name = ctx.fit(&cat_label(data, j), LABEL_SIZE, b.w - 8.0);
425            ctx.text(b.x + 4.0, b.y + 4.0, &name, LABEL_SIZE, on(fill), (HAlign::Left, VAlign::Top));
426            if chart.data_labels && b.h > LABEL_SIZE * 3.2 {
427                let v = value_of.get(&j).copied().unwrap_or(0.0);
428                let t = ctx.fit(&f.format(v, None), LABEL_SIZE, b.w - 8.0);
429                ctx.text(b.x + 4.0, b.y + 4.0 + LABEL_SIZE * 1.4, &t, LABEL_SIZE, on(fill), (HAlign::Left, VAlign::Top));
430            }
431        }
432    }
433}

437pub(super) fn radar(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
438    let ncat = data.categories.len().max(data.series.iter().map(|s| s.values.len()).max().unwrap_or(0)).min(1000);
439    if ncat == 0 {
440        return;
441    }
442    let labels: Vec<String> = (0..ncat).map(|j| cat_label(data, j)).collect();
443    let lw = labels.iter().map(|t| ctx.tw(t, LABEL_SIZE)).fold(0.0f32, f32::max).min(area.w * 0.2);
444    let r = ((area.w - 2.0 * (lw + 8.0)).min(area.h - 2.0 * (LABEL_SIZE * 1.4 + 6.0)) / 2.0).max(0.0);
445    if r < 5.0 {
446        return;
447    }
448    let (cx, cy) = (area.cx(), area.cy());
449    let (lo, hi) = value_range(data.series.iter().flat_map(|s| s.values.iter().flatten().copied())).unwrap_or((0.0, 0.0));
450    let sc = nice_scale(lo, hi, true);
451    let f = Fmt::new(data.series.first().map(|s| s.number_format.as_str()).unwrap_or("General"));
452    let ang = |j: usize| j as f32 / ncat as f32 * TAU;
453    let pt = |a: f32, rr: f32| [cx + rr * a.sin(), cy - rr * a.cos()];
454    let ticks = sc.ticks();
455    let spacing = r / ticks.len().saturating_sub(1).max(1) as f32;
456    let every = ((LABEL_SIZE * 1.3 / spacing.max(1e-3)).ceil() as usize).max(1);
457    for (ti, t) in ticks.into_iter().enumerate() {
458        let rr = sc.frac(t) * r;
459        if rr > 0.5 && chart.gridlines {
460            let mut ring: Vec<[f32; 2]> = (0..ncat).map(|j| pt(ang(j), rr)).collect();
461            if ncat < 3 {
462                ring = (0..=48).map(|k| pt(k as f32 / 48.0 * TAU, rr)).collect();
463            } else if let Some(p) = ring.first().copied() {
464                ring.push(p);
465            }
466            ctx.line(ring, GRID, 0.75);
467        }
468        if ti % every == 0 {
469            ctx.text(cx - 3.0, cy - rr, &f.format(t, Some(sc.step)), LABEL_SIZE * 0.9, TEXT, (HAlign::Right, VAlign::Middle));
470        }
471    }
472    let step = ncat.div_ceil(72).max(1);
473    for j in (0..ncat).step_by(step) {
474        let a = ang(j);
475        ctx.line(vec![[cx, cy], pt(a, r)], GRID, 0.75);
476        let p = pt(a, r + 6.0);
477        let (sa, ca) = (a.sin(), a.cos());
478        let align = if sa > 0.1 {
479            HAlign::Left
480        } else if sa < -0.1 {
481            HAlign::Right
482        } else {
483            HAlign::Center
484        };
485        let valign = if ca > 0.1 {
486            VAlign::Bottom
487        } else if ca < -0.1 {
488            VAlign::Top
489        } else {
490            VAlign::Middle
491        };
492        let t = ctx.fit(labels.get(j).map(String::as_str).unwrap_or(""), LABEL_SIZE, lw.max(20.0));
493        ctx.text(p[0], p[1], &t, LABEL_SIZE, TEXT, (align, valign));
494    }
495    for s in &data.series {
496        let mut pts: Vec<[f32; 2]> = (0..ncat).map(|j| pt(ang(j), sc.frac(s.values.get(j).copied().flatten().unwrap_or(sc.min)) * r)).collect();
497        if let Some(p) = pts.first().copied() {
498            pts.push(p);
499        }
500        ctx.line(pts, s.color, 2.0);
501    }
502}

504// ---------------------------------------------------------------- box & whisker
505
506fn quantile(sorted: &[f64], q: f64, method: QuartileMethod) -> f64 {
507    let n = sorted.len();
508    if n == 0 {
509        return 0.0;
510    }
511    // Clamp exclusive ranks for short samples so one or two points remain drawable.
512    let pos = match method {
513        QuartileMethod::Inclusive => q * (n - 1) as f64,
514        QuartileMethod::Exclusive => (q * (n as f64 + 1.0) - 1.0).clamp(0.0, (n - 1) as f64),
515    };
516    let i = pos.floor() as usize;
517    let frac = pos - i as f64;
518    let a = sorted.get(i).copied().unwrap_or(0.0);
519    let b = sorted.get(i + 1).copied().unwrap_or(a);
520    a + (b - a) * frac
521}

523pub(super) fn box_whisker(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
524    let labels: Vec<String> = data.series.iter().map(|s| s.name.clone()).collect();
525    let Some((lo, hi)) = value_range(data.series.iter().flat_map(|s| s.values.iter().flatten().copied())) else { return };
526    let ys = nice_scale(lo, hi, false);
527    let f = Fmt::new(data.series.first().map(|s| s.number_format.as_str()).unwrap_or("General"));
528    let plot = frame(ctx, area, CatAxis::Labels(&labels, &[]), &ValAxis { scale: ys, fmt: &f }, None, false, chart.gridlines);
529    if !plot.ok() {
530        return;
531    }
532    let n = data.series.len().max(1);
533    let cw = plot.w / n as f32;
534    let bw = (cw * 0.4).min(80.0);
535    let vy = |v: f64| plot.bottom() - ys.frac(v) * plot.h;
536    for (i, s) in data.series.iter().enumerate() {
537        let mut v: Vec<f64> = s.values.iter().flatten().copied().collect();
538        if v.is_empty() {
539            continue;
540        }
541        v.sort_by(f64::total_cmp);
542        let method = chart.series.get(i).map(|s| s.quartiles).unwrap_or_default();
543        let (q1, med, q3) = (quantile(&v, 0.25, method), quantile(&v, 0.5, method), quantile(&v, 0.75, method));
544        let iqr = q3 - q1;
545        let (flo, fhi) = (q1 - 1.5 * iqr, q3 + 1.5 * iqr);
546        let wlo = v.iter().copied().find(|x| *x >= flo).unwrap_or(q1);
547        let whi = v.iter().rev().copied().find(|x| *x <= fhi).unwrap_or(q3);
548        let mean = v.iter().sum::<f64>() / v.len() as f64;
549        let c = plot.x + (i as f32 + 0.5) * cw;
550        ctx.vline(c, vy(whi), vy(q3), s.color, 1.25);
551        ctx.vline(c, vy(q1), vy(wlo), s.color, 1.25);
552        ctx.hline(c - bw / 4.0, c + bw / 4.0, vy(whi), s.color, 1.25);
553        ctx.hline(c - bw / 4.0, c + bw / 4.0, vy(wlo), s.color, 1.25);
554        let (yt, yb) = (vy(q3), vy(q1));
555        ctx.rect(c - bw / 2.0, yt, bw, (yb - yt).max(0.5), s.color, None);
556        ctx.hline(c - bw / 2.0, c + bw / 2.0, vy(med), WHITE, 1.5);
557        let (mx, my, d) = (c, vy(mean), 3.0);
558        ctx.line(vec![[mx - d, my - d], [mx + d, my + d]], WHITE, 1.25);
559        ctx.line(vec![[mx - d, my + d], [mx + d, my - d]], WHITE, 1.25);
560        for o in v.iter().filter(|x| **x < flo || **x > fhi).take(MAX_MARKERS) {
561            ctx.out.push(Prim::Circle { cx: c, cy: vy(*o), r: 2.5, fill: None, stroke: Some((s.color, 1.0)) });
562        }
563    }
564}

568pub(super) fn stock(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
569    let ncat = data.categories.len().max(data.series.iter().map(|s| s.values.len()).max().unwrap_or(0)).min(MAX_CATS);
570    if ncat == 0 {
571        return;
572    }
573    let labels: Vec<String> = (0..ncat).map(|j| cat_label(data, j)).collect();
574    let Some((lo, hi)) = value_range(data.series.iter().flat_map(|s| s.values.iter().flatten().copied())) else { return };
575    let ys = nice_scale(lo, hi, false);
576    let f = Fmt::new(data.series.first().map(|s| s.number_format.as_str()).unwrap_or("General"));
577    let plot = frame(ctx, area, CatAxis::Labels(&labels, &[]), &ValAxis { scale: ys, fmt: &f }, None, false, chart.gridlines);
578    if !plot.ok() {
579        return;
580    }
581    let ser = |k: usize| data.series.get(k);
582    let (open, high, low, close) = match data.series.len() {
583        0 => return,
584        1 => (None, None, None, ser(0)),
585        2 => (None, ser(0), ser(1), None),
586        3 => (None, ser(0), ser(1), ser(2)),
587        _ => (ser(0), ser(1), ser(2), ser(3)),
588    };
589    let get = |s: Option<&SeriesData>, j: usize| s.and_then(|s| s.values.get(j).copied().flatten());
590    let cw = plot.w / ncat as f32;
591    let tick = (cw * 0.25).max(1.0);
592    let vy = |v: f64| plot.bottom() - ys.frac(v) * plot.h;
593    for j in 0..ncat {
594        let c = plot.x + (j as f32 + 0.5) * cw;
595        if let (Some(h), Some(l)) = (get(high, j), get(low, j)) {
596            ctx.vline(c, vy(h), vy(l), STOCK, 1.25);
597        }
598        match (get(open, j), get(close, j)) {
599            (Some(o), Some(cl)) => {
600                let (yo, yc) = (vy(o), vy(cl));
601                let bw = (cw * 0.5).max(1.0);
602                let fill = if cl >= o { WHITE } else { STOCK };
603                ctx.rect(c - bw / 2.0, yo.min(yc), bw, (yo - yc).abs().max(0.75), fill, Some((STOCK, 1.0)));
604            }
605            (None, Some(cl)) => {
606                if high.is_some() {
607                    ctx.hline(c, c + tick, vy(cl), STOCK, 1.25);
608                } else {
609                    ctx.out.push(Prim::Circle { cx: c, cy: vy(cl), r: 2.5, fill: Some(STOCK), stroke: None });
610                }
611            }
612            _ => {}
613        }
614    }
615}

621    #[test]

633            (vec![1., 2., 3., 4.], (1.25, 3.75), (1.75, 3.25)),
