1//! Trendline maths: least-squares fits behind chart trendlines and moving averages.
2//!
3//! Exponential, logarithmic and power trendlines are linear fits of transformed data
4//! (`ln y` against `x`, `y` against `ln x`, `ln y` against `ln x`), the way Excel documents them;
5//! points where the transform is undefined (non-positive values) are skipped and R² is reported
6//! for the transformed fit. Polynomials solve the normal equations on centred, scaled x.
7
8use crate::model::{Trendline, TrendlineKind};
9
10/// A fitted regression trendline.
11#[derive(Clone, Debug, PartialEq)]
12pub(crate) struct TrendFit {
13    kind: TrendlineKind,
14    /// Coefficients `c0..=cd` of `Y = Σ cₖ·Xᵏ` in the transformed variables, for the equation.
15    pub coef: Vec<f64>,
16    /// The same polynomial in `u = (X - center) / scale`, for stable evaluation.
17    center: f64,
18    scale: f64,
19    a: Vec<f64>,
20    /// Coefficient of determination; `None` when undefined (all Y equal).
21    pub r2: Option<f64>,
22}
23
24/// Polynomial degree in the transformed variable; `None` for moving averages.
25fn degree(k: TrendlineKind) -> Option<usize> {
26    match k {
27        TrendlineKind::Linear | TrendlineKind::Exponential | TrendlineKind::Logarithmic | TrendlineKind::Power => Some(1),
28        TrendlineKind::Polynomial(o) => Some(usize::from(o.clamp(2, 6))),
29        TrendlineKind::MovingAverage(_) => None,
30    }
31}
32
33fn log_x(k: TrendlineKind) -> bool {
34    matches!(k, TrendlineKind::Logarithmic | TrendlineKind::Power)
35}
36
37fn log_y(k: TrendlineKind) -> bool {
38    matches!(k, TrendlineKind::Exponential | TrendlineKind::Power)
39}

41/// Fits `t` to the points `(x, y)`. `None` for moving averages and degenerate data (too few
42/// usable points, all x equal, a singular system, non-finite results).
43pub(crate) fn fit_trendline(t: &Trendline, pts: &[(f64, f64)]) -> Option<TrendFit> {
44    let kind = t.kind;
45    let d = degree(kind)?;
46    let data: Vec<(f64, f64)> = pts
47        .iter()
48        .filter(|(x, y)| x.is_finite() && y.is_finite() && (!log_x(kind) || *x > 0.0) && (!log_y(kind) || *y > 0.0))
49        .map(|&(x, y)| (if log_x(kind) { x.ln() } else { x }, if log_y(kind) { y.ln() } else { y }))
50        .collect();
51    if data.len() < 2 || data.len() <= d {
52        return None;
53    }
54    // Set Intercept applies to linear, exponential and polynomial trendlines; it fixes Y at x = 0.
55    let fixed = match kind {
56        TrendlineKind::Linear | TrendlineKind::Polynomial(_) => t.intercept,
57        TrendlineKind::Exponential => t.intercept.filter(|b| *b > 0.0).map(f64::ln),
58        _ => None,
59    }
60    .filter(|b| b.is_finite());
61    let n = data.len() as f64;
62    let center = if fixed.is_some() { 0.0 } else { data.iter().map(|p| p.0).sum::<f64>() / n };
63    let scale = data.iter().map(|p| (p.0 - center).abs()).fold(0.0f64, f64::max);
64    if !(scale.is_finite() && scale > 0.0) {
65        return None;
66    }
67    let first = usize::from(fixed.is_some());
68    let b0 = fixed.unwrap_or(0.0);
69    // Normal equations over the unknown powers `first..=d` of u.
70    let m = d + 1 - first;
71    let mut ata = vec![vec![0.0f64; m]; m];
72    let mut aty = vec![0.0f64; m];
73    for &(x, y) in &data {
74        let u = (x - center) / scale;
75        // `d` is at most 6, so the powers `first..=d` always fit the array (no allocation per point).
76        let mut buf = [0.0f64; 7];
77        for (slot, k) in buf.iter_mut().zip(first..=d) {
78            *slot = u.powi(k as i32);
79        }
80        let pows = buf.get(..m)?;
81        for (i, pi) in pows.iter().enumerate() {
82            if let Some(v) = aty.get_mut(i) {
83                *v += pi * (y - b0);
84            }
85            for (j, pj) in pows.iter().enumerate() {
86                if let Some(v) = ata.get_mut(i).and_then(|r| r.get_mut(j)) {
87                    *v += pi * pj;
88                }
89            }
90        }
91    }
92    let sol = solve(ata, aty)?;
93    let mut a = vec![b0; d + 1];
94    for (k, v) in sol.into_iter().enumerate() {
95        if let Some(slot) = a.get_mut(k + first) {
96            *slot = v;
97        }
98    }
99    let coef = expand(&a, center, scale)?;
100    let fit = TrendFit { kind, coef, center, scale, a, r2: None };
101    let mean = data.iter().map(|p| p.1).sum::<f64>() / n;
102    let (mut ss_res, mut ss_tot) = (0.0f64, 0.0f64);
103    for &(x, y) in &data {
104        ss_res += (y - fit.poly(x)).powi(2);
105        // A set intercept is LINEST(y - b, x, FALSE): the total sum of squares is not centred.
106        ss_tot += if fixed.is_some() { (y - b0).powi(2) } else { (y - mean).powi(2) };
107    }
108    let r2 = (ss_tot > 0.0 && ss_tot.is_finite() && ss_res.is_finite()).then(|| 1.0 - ss_res / ss_tot);
109    Some(TrendFit { r2, ..fit })
110}

112/// Gaussian elimination with partial pivoting; `None` when (nearly) singular.
113fn solve(mut a: Vec<Vec<f64>>, mut b: Vec<f64>) -> Option<Vec<f64>> {
114    let n = b.len();
115    let big = a.iter().flatten().fold(0.0f64, |m, v| m.max(v.abs()));
116    if !(big.is_finite() && big > 0.0) {
117        return None;
118    }
119    for col in 0..n {
120        let piv = (col..n).max_by(|&i, &j| {
121            let vi = a.get(i).and_then(|r| r.get(col)).map_or(0.0, |v| v.abs());
122            let vj = a.get(j).and_then(|r| r.get(col)).map_or(0.0, |v| v.abs());
123            vi.total_cmp(&vj)
124        })?;
125        a.swap(col, piv);
126        b.swap(col, piv);
127        let p = *a.get(col)?.get(col)?;
128        if !p.is_finite() || p.abs() <= big * 1e-13 {
129            return None;
130        }
131        let prow = a.get(col)?.clone();
132        let pb = *b.get(col)?;
133        for r in col + 1..n {
134            let row = a.get_mut(r)?;
135            let f = *row.get(col)? / p;
136            for (v, pv) in row.iter_mut().zip(&prow).skip(col) {
137                *v -= f * pv;
138            }
139            *b.get_mut(r)? -= f * pb;
140        }
141    }
142    let mut x = vec![0.0f64; n];
143    for r in (0..n).rev() {
144        let row = a.get(r)?;
145        let s: f64 = (r + 1..n).map(|c| row.get(c).copied().unwrap_or(0.0) * x.get(c).copied().unwrap_or(0.0)).sum();
146        let v = (b.get(r)? - s) / row.get(r)?;
147        *x.get_mut(r)? = v;
148    }
149    x.iter().all(|v| v.is_finite()).then_some(x)
150}

152/// Coefficients in X of `Σ aₖ·((X - c) / s)ᵏ` (binomial expansion).
153fn expand(a: &[f64], c: f64, s: f64) -> Option<Vec<f64>> {
154    let mut out = vec![0.0f64; a.len()];
155    for (k, ak) in a.iter().enumerate() {
156        let lead = ak / s.powi(k as i32);
157        let mut binom = 1.0f64;
158        for j in 0..=k {
159            // C(k, j) · (-c)^(k-j) · X^j
160            if let Some(o) = out.get_mut(j) {
161                *o += lead * binom * (-c).powi((k - j) as i32);
162            }
163            binom = binom * (k - j) as f64 / (j + 1) as f64;
164        }
165    }
166    out.iter().all(|v| v.is_finite()).then_some(out)
167}

169impl TrendFit {
170    /// The fitted polynomial at transformed `x` (Horner in u).
171    fn poly(&self, x: f64) -> f64 {
172        let u = (x - self.center) / self.scale;
173        self.a.iter().rev().fold(0.0, |acc, c| acc * u + c)
174    }
175
176    /// The trendline's y at `x`; `None` outside its domain (x ≤ 0 for log/power) or when not finite.
177    pub(crate) fn eval(&self, x: f64) -> Option<f64> {
178        let xx = if log_x(self.kind) { if x > 0.0 { x.ln() } else { return None } } else { x };
179        let p = self.poly(xx);
180        let y = if log_y(self.kind) { p.exp() } else { p };
181        y.is_finite().then_some(y)
182    }
183
184    /// Equation label text, e.g. `y = 2x + 1`, `y = 0.5x² - 2x + 1`, `y = 3e^(0.2x)`.
185    pub(crate) fn equation(&self) -> String {
186        let c0 = self.coef.first().copied().unwrap_or(0.0);
187        let c1 = self.coef.get(1).copied().unwrap_or(0.0);
188        match self.kind {
189            TrendlineKind::Exponential => format!("y = {}e^({}x)", num_text(c0.exp()), num_text(c1)),
190            TrendlineKind::Power => format!("y = {}x^{}", num_text(c0.exp()), num_text(c1)),
191            TrendlineKind::Logarithmic => terms(&[(c1, "ln(x)"), (c0, "")]),
192            _ => {
193                let parts: Vec<(f64, &str)> = self.coef.iter().enumerate().rev().map(|(k, c)| (*c, POWERS.get(k).copied().unwrap_or("x"))).collect();
194                terms(&parts)
195            }
196        }
197    }
198
199    /// R² label text, e.g. `R² = 0.9963`.
200    pub(crate) fn r_squared(&self) -> String {
201        match self.r2 {
202            Some(r) => format!("R² = {}", num_text(r)),
203            None => "R² = #N/A".into(),
204        }
205    }
206}

208const POWERS: [&str; 7] = ["", "x", "x²", "x³", "x⁴", "x⁵", "x⁶"];

210/// `y = a·t1 ± b·t2 …`, skipping terms whose coefficient shows as 0.
211fn terms(parts: &[(f64, &str)]) -> String {
212    let mut s = String::from("y = ");
213    let mut first = true;
214    for (c, t) in parts {
215        let txt = num_text(c.abs());
216        if txt == "0" {
217            continue;
218        }
219        let neg = *c < 0.0;
220        if first {
221            if neg {
222                s.push('-');
223            }
224        } else {
225            s.push_str(if neg { " - " } else { " + " });
226        }
227        // A unit coefficient on a power of x is left out: `y = x² - x + 1`.
228        if txt != "1" || t.is_empty() {
229            s.push_str(&txt);
230        }
231        s.push_str(t);
232        first = false;
233    }
234    if first {
235        s.push('0');
236    }
237    s
238}

240/// A coefficient as trendline labels show it: up to four decimals, scientific when tiny or huge.
241pub(crate) fn num_text(v: f64) -> String {
242    if !v.is_finite() {
243        return "#NUM!".into();
244    }
245    let a = v.abs();
246    if a == 0.0 {
247        return "0".into();
248    }
249    if !(1e-4..1e11).contains(&a) {
250        let s = format!("{v:.4E}");
251        let (mant, exp) = s.split_once('E').unwrap_or((s.as_str(), "0"));
252        let mant = trim_zeros(mant);
253        let e: i32 = exp.parse().unwrap_or(0);
254        return format!("{mant}E{}{:02}", if e < 0 { '-' } else { '+' }, e.unsigned_abs());
255    }
256    let s = trim_zeros(&format!("{v:.4}"));
257    if s == "-0" { "0".into() } else { s }
258}
259
260fn trim_zeros(s: &str) -> String {
261    if s.contains('.') { s.trim_end_matches('0').trim_end_matches('.').to_string() } else { s.to_string() }
262}

264/// Moving average of `period` consecutive points in plotted order: point `i` (from
265/// `period - 1`) is `(xᵢ, mean(yᵢ₋ₚ₊₁..=yᵢ))`. Empty when there are fewer points than `period`.
266///
267/// Linear in the number of points: the points are cut into blocks of `period`, and a window is
268/// the tail of one block plus the head of the next, taken from running sums that only ever add
269/// (a running sum that also subtracts would lose small values after a huge one passed through).
270pub(crate) fn moving_average(pts: &[(f64, f64)], period: u32) -> Vec<(f64, f64)> {
271    let p = period.clamp(2, 255) as usize;
272    let pts: Vec<(f64, f64)> = pts.iter().copied().filter(|(x, y)| x.is_finite() && y.is_finite()).collect();
273    let n = pts.len();
274    if n < p {
275        return vec![];
276    }
277    // `head[i]`: sum of the block's points up to and including `i`; `tail[i]`: from `i` to the block's end.
278    let mut head = vec![0.0f64; n];
279    let mut tail = vec![0.0f64; n];
280    for (b, block) in pts.chunks(p).enumerate() {
281        let (mut h, mut t) = (0.0f64, 0.0f64);
282        for k in 0..block.len() {
283            h += block.get(k).map_or(0.0, |q| q.1);
284            if let Some(v) = head.get_mut(b * p + k) {
285                *v = h;
286            }
287            let r = block.len() - 1 - k;
288            t += block.get(r).map_or(0.0, |q| q.1);
289            if let Some(v) = tail.get_mut(b * p + r) {
290                *v = t;
291            }
292        }
293    }
294    (p - 1..n)
295        .filter_map(|i| {
296            let s = i + 1 - p;
297            let mut sum = tail.get(s)? + if s.is_multiple_of(p) { 0.0 } else { *head.get(i)? };
298            if !sum.is_finite() {
299                // A block sum overflowed although the window itself may not: sum it directly.
300                sum = pts.get(s..=i)?.iter().map(|q| q.1).sum();
301            }
302            Some((pts.get(i)?.0, sum / p as f64))
303        })
304        .collect()
305}

307/// Excel's automatic trendline name for the legend, e.g. `Linear (Sales)`.
308pub(crate) fn auto_name(kind: TrendlineKind, series: &str) -> String {
309    let k = match kind {
310        TrendlineKind::Linear => "Linear".to_string(),
311        TrendlineKind::Exponential => "Expon.".into(),
312        TrendlineKind::Logarithmic => "Log.".into(),
313        TrendlineKind::Polynomial(_) => "Poly.".into(),
314        TrendlineKind::Power => "Power".into(),
315        TrendlineKind::MovingAverage(p) => format!("{} per. Mov. Avg.", p.clamp(2, 255)),
316    };
317    format!("{k} ({series})")
318}

327    impl Rng {
328        fn next(&mut self) -> u64 {
329            self.0 ^= self.0 << 13;
330            self.0 ^= self.0 >> 7;
331            self.0 ^= self.0 << 17;
332            self.0
333        }
334        fn unit(&mut self) -> f64 {
335            (self.next() % 1_000_000) as f64 / 1_000_000.0 - 0.5
336        }
337    }

339    /// The window-by-window moving average this module used to compute.
340    fn moving_average_naive(pts: &[(f64, f64)], period: u32) -> Vec<(f64, f64)> {
341        let p = period.clamp(2, 255) as usize;
342        let pts: Vec<(f64, f64)> = pts.iter().copied().filter(|(x, y)| x.is_finite() && y.is_finite()).collect();
343        pts.windows(p).filter_map(|w| w.last().map(|(x, _)| (*x, w.iter().map(|q| q.1).sum::<f64>() / p as f64))).collect()
344    }

347    fn linear_moving_average_matches_the_window_sums() {
348        let mut r = Rng(0x9E37_79B9_7F4A_7C15);
349        for case in 0..60 {
350            let n = (r.next() % 700) as usize;
351            let pts: Vec<(f64, f64)> = (0..n)

356                        // A huge value passing through must not leave rounding residue behind.
357                        2 => {
358                            if r.next().is_multiple_of(50) {
359                                1e300 * r.unit().signum()
360                            } else {
361                                r.unit()
362                            }
363                        }
364                        3 => {
365                            if i % 2 == 0 {
366                                1e15
367                            } else {
368                                -1e15 + r.unit()
369                            }
370                        }

372                            if r.next().is_multiple_of(9) {

377                        }

392                }

406    /// The trendline fit as it was written with a `Vec` of powers per data point.
407    fn fit_reference(t: &Trendline, pts: &[(f64, f64)]) -> Option<TrendFit> {
408        let kind = t.kind;
409        let d = degree(kind)?;
410        let data: Vec<(f64, f64)> = pts
411            .iter()
412            .filter(|(x, y)| x.is_finite() && y.is_finite() && (!log_x(kind) || *x > 0.0) && (!log_y(kind) || *y > 0.0))
413            .map(|&(x, y)| (if log_x(kind) { x.ln() } else { x }, if log_y(kind) { y.ln() } else { y }))
414            .collect();
415        if data.len() < 2 || data.len() <= d {
416            return None;
417        }
418        let fixed = match kind {
419            TrendlineKind::Linear | TrendlineKind::Polynomial(_) => t.intercept,
420            TrendlineKind::Exponential => t.intercept.filter(|b| *b > 0.0).map(f64::ln),
421            _ => None,
422        }
423        .filter(|b| b.is_finite());
424        let n = data.len() as f64;
425        let center = if fixed.is_some() { 0.0 } else { data.iter().map(|p| p.0).sum::<f64>() / n };
426        let scale = data.iter().map(|p| (p.0 - center).abs()).fold(0.0f64, f64::max);
427        if !(scale.is_finite() && scale > 0.0) {
428            return None;
429        }
430        let first = usize::from(fixed.is_some());
431        let b0 = fixed.unwrap_or(0.0);
432        let m = d + 1 - first;
433        let mut ata = vec![vec![0.0f64; m]; m];
434        let mut aty = vec![0.0f64; m];
435        for &(x, y) in &data {
436            let u = (x - center) / scale;
437            let pows: Vec<f64> = (first..=d).map(|k| u.powi(k as i32)).collect();
438            for (i, pi) in pows.iter().enumerate() {
439                aty[i] += pi * (y - b0);
440                for (j, pj) in pows.iter().enumerate() {
441                    ata[i][j] += pi * pj;
442                }
443            }
444        }
445        let sol = solve(ata, aty)?;
446        let mut a = vec![b0; d + 1];
447        for (k, v) in sol.into_iter().enumerate() {
448            a[k + first] = v;
449        }
450        let coef = expand(&a, center, scale)?;
451        let fit = TrendFit { kind, coef, center, scale, a, r2: None };
452        let mean = data.iter().map(|p| p.1).sum::<f64>() / n;
453        let (mut ss_res, mut ss_tot) = (0.0f64, 0.0f64);
454        for &(x, y) in &data {
455            ss_res += (y - fit.poly(x)).powi(2);
456            ss_tot += if fixed.is_some() { (y - b0).powi(2) } else { (y - mean).powi(2) };
457        }
458        let r2 = (ss_tot > 0.0 && ss_tot.is_finite() && ss_res.is_finite()).then(|| 1.0 - ss_res / ss_tot);
459        Some(TrendFit { r2, ..fit })
460    }
