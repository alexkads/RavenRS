65impl Ctx<'_> {
66    fn tw(&self, s: &str, size: f32) -> f32 {
67        let w = self.m.text_width(s, size, false);
68        if w.is_finite() { w.max(0.0) } else { 0.0 }
69    }
70    /// `s` shortened with an ellipsis to fit `maxw`; empty when not even one character fits.
71    ///
72    /// Binary search over the number of kept characters (at most 400 are tried), about nine
73    /// `Measure` calls instead of one per character. It assumes the width of `prefix + "…"` does not
74    /// shrink as the prefix grows, which holds for per-character advance sums (`ApproxMeasure`, the
75    /// PDF metrics) and for the UI's text layout up to kerning of a few hundredths of a point; with
76    /// such a non-monotone measure the result may differ from the old linear scan by a character.
77    /// A label with a line break is not monotone (the ellipsis moves to a new line and the widest
78    /// line can get shorter), so it takes the linear scan, longest prefix first.
79    fn fit(&self, s: &str, size: f32, maxw: f32) -> String {
80        if self.tw(s, size) <= maxw {
81            return s.to_string();
82        }
83        // Byte offset where each of the first 400 characters starts; a prefix of `k` characters
84        // (k < number of characters, so the offset exists) ends at `starts[k]`.
85        let starts: Vec<usize> = s.char_indices().map(|(i, _)| i).take(400).collect();
86        let top = starts.len().saturating_sub(1);
87        let fits = |k: usize| -> bool {
88            let end = starts.get(k).copied().unwrap_or(s.len());
89            let mut t = String::with_capacity(end + 3);
90            t.push_str(s.get(..end).unwrap_or(""));
91            t.push('…');
92            self.tw(&t, size) <= maxw
93        };
94        if s.contains('\n') {
95            let k = (1..=top).rev().find(|&k| fits(k));
96            return k.map_or_else(String::new, |k| format!("{}…", s.get(..starts.get(k).copied().unwrap_or(s.len())).unwrap_or("")));
97        }
98        let (mut lo, mut hi) = (0usize, top);
99        while lo < hi {
100            let mid = lo + (hi - lo).div_ceil(2);
101            if fits(mid) {
102                lo = mid;
103            } else {
104                hi = mid - 1;
105            }
106        }
107        if lo == 0 { String::new() } else { format!("{}…", s.get(..starts.get(lo).copied().unwrap_or(s.len())).unwrap_or("")) }
108    }
109    fn text(&mut self, x: f32, y: f32, s: &str, size: f32, color: Rgba, (align, valign): (HAlign, VAlign)) {
110        if s.is_empty() {
111            return;
112        }
113        self.out.push(Prim::Text { x, y, text: s.to_string(), size, color, bold: false, align, valign, rotation: 0.0 });
114    }
115    fn rect(&mut self, x: f32, y: f32, w: f32, h: f32, fill: Rgba, stroke: Option<(Rgba, f32)>) {
116        self.out.push(Prim::Rect { x, y, w, h, fill: Some(fill), stroke, radius: 0.0 });
117    }
118    fn line(&mut self, pts: Vec<[f32; 2]>, color: Rgba, width: f32) {
119        if pts.len() >= 2 {
120            self.out.push(Prim::Line { pts, color, width, dash: false });
121        }
122    }
123    fn hline(&mut self, x0: f32, x1: f32, y: f32, color: Rgba, width: f32) {
124        self.line(vec![[x0, y], [x1, y]], color, width);
125    }
126    fn vline(&mut self, x: f32, y0: f32, y1: f32, color: Rgba, width: f32) {
127        self.line(vec![[x, y0], [x, y1]], color, width);
128    }
129}

131fn sanitize_vals(v: &[Option<f64>]) -> Vec<Option<f64>> {
132    v.iter().take(MAX_POINTS).map(|x| x.filter(|n| n.is_finite()).map(clean)).collect()
133}

135pub(crate) fn sanitize(d: &ChartData) -> ChartData {
136    ChartData {
137        palette: d.palette.iter().take(MAX_POINTS).copied().collect(),
138        categories: d.categories.iter().take(MAX_POINTS).cloned().collect(),
139        category_levels: d.category_levels.iter().take(3).map(|l| l.iter().take(MAX_POINTS).cloned().collect()).collect(),
140        series: d
141            .series
142            .iter()
143            .take(MAX_SERIES)
144            .map(|s| SeriesData {
145                name: s.name.clone(),
146                values: sanitize_vals(&s.values),
147                x: s.x.as_deref().map(sanitize_vals),
148                sizes: s.sizes.as_deref().map(sanitize_vals),
149                color: s.color,
150                kind: s.kind,
151                secondary: s.secondary,
152                smooth: s.smooth,
153                trendlines: s.trendlines.iter().take(MAX_TRENDLINES).cloned().collect(),
154                number_format: s.number_format.clone(),
155            })
156            .collect(),
157    }
158}

160/// Whether `v` survives `sanitize_vals` unchanged.
161fn is_clean(v: &Option<f64>) -> bool {
162    v.is_none_or(|n| n.is_finite() && clean(n) == n)
163}

165/// Whether [`sanitize`] and [`apply_budget`] would change `d`.
166fn needs_sanitizing(d: &ChartData) -> bool {
167    let cap = points_per_series(d.series.len());
168    let long = |n: usize| n > cap;
169    let mut trendlines = 0usize;
170    d.palette.len() > MAX_POINTS
171        || long(d.categories.len())
172        || d.category_levels.len() > 3
173        || d.category_levels.iter().any(|l| l.len() > MAX_POINTS)
174        || d.series.len() > MAX_SERIES
175        || d.series.iter().any(|s| {
176            trendlines += s.trendlines.len();
177            long(s.values.len())
178                || !s.values.iter().all(is_clean)
179                || s.x.as_ref().is_some_and(|x| long(x.len()) || !x.iter().all(is_clean))
180                || s.sizes.as_ref().is_some_and(|x| long(x.len()) || !x.iter().all(is_clean))
181                || s.trendlines.len() > MAX_TRENDLINES
182        })
183        || trendlines > MAX_TRENDLINES
184}

186/// Enforces the chart's budget of points (`MAX_TOTAL_POINTS` = series x points: every series and
187/// the category list keep their first `MAX_TOTAL_POINTS / series` points) and of trendlines
188/// (`MAX_TRENDLINES` per chart, in series order).
189fn apply_budget(mut d: ChartData) -> ChartData {
190    let cap = points_per_series(d.series.len());
191    d.categories.truncate(cap);
192    let mut trendlines = MAX_TRENDLINES;
193    for s in &mut d.series {
194        s.values.truncate(cap);
195        for v in [&mut s.x, &mut s.sizes].into_iter().flatten() {
196            v.truncate(cap);
197        }
198        s.trendlines.truncate(trendlines);
199        trendlines = trendlines.saturating_sub(s.trendlines.len());
200    }
201    d
202}

204/// The data `render` draws: `d` itself when it is already clean and within budget (the usual
205/// case: no copy), else a cleaned, truncated copy.
206pub(crate) fn sanitized(d: &ChartData) -> Cow<'_, ChartData> {
207    if needs_sanitizing(d) { Cow::Owned(apply_budget(sanitize(d))) } else { Cow::Borrowed(d) }
208}

210/// Lays out and draws a chart into a `w`×`h` box at the origin. Empty for a degenerate box.
211pub fn render(chart: &Chart, data: &ChartData, w: f32, h: f32, m: &dyn Measure) -> Vec<Prim> {
212    if !(w.is_finite() && h.is_finite()) || w < 1.0 || h < 1.0 || w > 1e7 || h > 1e7 {
213        return vec![];
214    }
215    let data = sanitized(data);
216    let mut ctx = Ctx { out: Vec::new(), m };
217    ctx.out.push(Prim::Rect { x: 0.0, y: 0.0, w, h, fill: Some(WHITE), stroke: Some((GRID, 1.0)), radius: 0.0 });
218    let pad = (w.min(h) * 0.04).clamp(2.0, 10.0);
219    let mut area = Bx { x: pad, y: pad, w: w - 2.0 * pad, h: h - 2.0 * pad };
220    if let Some(t) = chart_title(chart, &data) {
221        let size = TITLE_SIZE.min(h * 0.12).max(4.0);
222        let t = ctx.fit(&t, size, area.w);
223        ctx.text(w / 2.0, area.y, &t, size, TEXT, (HAlign::Center, VAlign::Top));
224        let used = size * 1.3 + 4.0;
225        area.y += used;
226        area.h -= used;
227    }
228    let entries = legend_entries(chart, &data);
229    if !entries.is_empty() && area.ok() {
230        area = draw_legend(&mut ctx, &entries, chart.legend, area);
231    }
232    if area.ok() && !data.series.is_empty() {
233        draw_kind(&mut ctx, chart, &data, area);
234    }
235    finalize(ctx.out, w, h)
236}

238fn draw_kind(ctx: &mut Ctx, chart: &Chart, data: &ChartData, area: Bx) {
239    use ChartKind::*;
240    match chart.kind {
241        Pie => pie(ctx, chart, data, area, 0.0, false),
242        Doughnut => doughnut(ctx, chart, data, area),
243        Sunburst => pie(ctx, chart, data, area, 0.2, true),
244        Treemap => treemap(ctx, chart, data, area),
245        Funnel => funnel(ctx, chart, data, area),
246        Radar => radar(ctx, chart, data, area),
247        _ => {
248            let area = axis_titles(ctx, chart, area);
249            if !area.ok() {
250                return;
251            }
252            match chart.kind {
253                BarClustered | BarStacked | BarStacked100 => cartesian(ctx, chart, data, area, true),
254                Scatter | ScatterLines | Bubble => scatter(ctx, chart, data, area),
255                Histogram => histogram(ctx, chart, data, area),
256                Waterfall => waterfall(ctx, chart, data, area),
257                BoxWhisker => box_whisker(ctx, chart, data, area),
258                Stock => stock(ctx, chart, data, area),
259                _ => cartesian(ctx, chart, data, area, false),
260            }
261        }
262    }
263}

271fn clamp_stroke(s: Option<(Rgba, f32)>) -> Option<(Rgba, f32)> {
272    s.filter(|(_, w)| fin(*w) && *w > 0.0).map(|(c, w)| (c, w.min(50.0)))

275/// Drops non-finite primitives and clamps every coordinate into the `w`×`h` box.
276pub(crate) fn finalize(prims: Vec<Prim>, w: f32, h: f32) -> Vec<Prim> {
277    let cx = |v: f32| v.clamp(0.0, w);
278    let cy = |v: f32| v.clamp(0.0, h);
279    let pts_ok = |pts: &[[f32; 2]]| pts.iter().all(|p| fin(p[0]) && fin(p[1]));
280    prims
281        .into_iter()
282        .filter_map(|p| match p {
283            Prim::Rect { x, y, w: rw, h: rh, fill, stroke, radius } => {
284                if !(fin(x) && fin(y) && fin(rw) && fin(rh)) {
285                    return None;
286                }
287                let (x0, x1) = (cx(x.min(x + rw)), cx(x.max(x + rw)));
288                let (y0, y1) = (cy(y.min(y + rh)), cy(y.max(y + rh)));
289                if x1 - x0 <= 0.0 || y1 - y0 <= 0.0 {
290                    return None;
291                }
292                let radius = if fin(radius) { radius.clamp(0.0, (x1 - x0).min(y1 - y0) / 2.0) } else { 0.0 };
293                Some(Prim::Rect { x: x0, y: y0, w: x1 - x0, h: y1 - y0, fill, stroke: clamp_stroke(stroke), radius })
294            }
295            Prim::Line { pts, color, width, dash } => {
296                if !pts_ok(&pts) || pts.len() < 2 || !fin(width) {
297                    return None;
298                }
299                let pts = pts.into_iter().map(|p| [cx(p[0]), cy(p[1])]).collect();
300                Some(Prim::Line { pts, color, width: width.clamp(0.1, 50.0), dash })
301            }
302            Prim::Polygon { pts, fill, stroke } => {
303                if !pts_ok(&pts) || pts.len() < 3 {
304                    return None;
305                }
306                let pts = pts.into_iter().map(|p| [cx(p[0]), cy(p[1])]).collect();
307                Some(Prim::Polygon { pts, fill, stroke: clamp_stroke(stroke) })
308            }
309            Prim::Wedge { cx: x, cy: y, r_outer, r_inner, a0, a1, fill, stroke } => {
310                if ![x, y, r_outer, r_inner, a0, a1].iter().all(|v| fin(*v)) {
311                    return None;
312                }
313                let (x, y) = (cx(x), cy(y));
314                let maxr = x.min(w - x).min(y).min(h - y);
315                let ro = r_outer.min(maxr);
316                if ro <= 0.0 {
317                    return None;
318                }
319                Some(Prim::Wedge { cx: x, cy: y, r_outer: ro, r_inner: r_inner.clamp(0.0, ro), a0, a1, fill, stroke: clamp_stroke(stroke) })
320            }
321            Prim::Circle { cx: x, cy: y, r, fill, stroke } => {
322                if !(fin(x) && fin(y) && fin(r)) {
323                    return None;
324                }
325                let (x, y) = (cx(x), cy(y));
326                let r = r.min(x.min(w - x).min(y).min(h - y));
327                if r <= 0.0 {
328                    return None;
329                }
330                Some(Prim::Circle { cx: x, cy: y, r, fill, stroke: clamp_stroke(stroke) })
331            }
332            Prim::Text { x, y, text, size, color, bold, align, valign, rotation } => {
333                if !(fin(x) && fin(y) && fin(size) && fin(rotation)) || text.is_empty() {
334                    return None;
335                }
336                Some(Prim::Text { x: cx(x), y: cy(y), text, size: size.clamp(1.0, 200.0), color, bold, align, valign, rotation })
337            }
338        })
339        .collect()
