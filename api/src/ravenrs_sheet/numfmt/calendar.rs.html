1//! Calendars a format section can show dates in: the `B1` (Gregorian) and `B2` (Hijri) prefixes.
2
3use crate::base::date::{DateTime, MONTHS, WEEKDAYS, days_from_civil, days_in_month};
4
5/// The calendar a section's year, month and day parts count in.
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
7pub(crate) enum Calendar {
8    #[default]
9    Gregorian,
10    /// The tabular (arithmetic) Islamic calendar.
11    Hijri,
12}
13
14/// Arabic names of the Hijri months, Muharram first. Two-word names are joined by a no-break
15/// space (U+00A0), as Excel writes them.
16const HIJRI_MONTHS: [&str; 12] = [
17    "محرم",
18    "صفر",
19    "ربيع\u{a0}الأول",
20    "ربيع\u{a0}الثاني",
21    "جمادى\u{a0}الأولى",
22    "جمادى\u{a0}الثانية",
23    "رجب",
24    "شعبان",
25    "رمضان",
26    "شوال",
27    "ذو\u{a0}القعدة",
28    "ذو\u{a0}الحجة",
29];
30
31/// Arabic weekday names, Sunday first, shown for `ddd` and `dddd` under `B2`.
32const ARABIC_WEEKDAYS: [&str; 7] = ["الأحد", "الإثنين", "الثلاثاء", "الأربعاء", "الخميس", "الجمعة", "السبت"];
33
34/// Year, month and day of a date in a section's calendar, with the month and weekday names.
35pub(crate) struct CalendarDate {
36    pub year: i64,
37    pub month: u32,
38    pub day: u32,
39    /// `mmmm`.
40    pub month_name: &'static str,
41    /// `mmm`.
42    pub month_short: &'static str,
43    /// `dddd`.
44    pub day_name: &'static str,
45    /// `ddd`.
46    pub day_short: &'static str,
47}
48
49impl Calendar {
50    /// The date of a day in this calendar, `None` when it has none: under `B2`, the 1900 system's
51    /// fictitious 29 February 1900 (serial 60, which Excel shows as `#VALUE!`) and days before
52    /// 1 Muharram 1 AH.
53    pub(crate) fn date(self, dt: &DateTime) -> Option<CalendarDate> {
54        match self {
55            Calendar::Hijri => hijri_date(dt),
56            Calendar::Gregorian => Some(gregorian_date(dt)),
57        }
58    }
59}
60
61fn gregorian_date(dt: &DateTime) -> CalendarDate {
62    let month_name = MONTHS.get((dt.month as usize).wrapping_sub(1)).copied().unwrap_or("");
63    let day_name = WEEKDAYS.get(dt.weekday as usize).copied().unwrap_or("");
64    CalendarDate {
65        year: dt.year.into(),
66        month: dt.month,
67        day: dt.day,
68        month_name,
69        month_short: month_name.get(..3).unwrap_or(month_name),
70        day_name,
71        day_short: day_name.get(..3).unwrap_or(day_name),
72    }
73}

75/// Days from 1 Muharram 1 AH to the first of a Hijri month. Leap years (30 days in the last
76/// month) are years 2, 5, 7, 10, 13, 15, 18, 21, 24, 26 and 29 of each 30-year cycle (10,631
77/// days), which is what Excel shows; odd months have 30 days and even months 29.
78fn days_before_month(year: i64, month: i64) -> i64 {
79    // `(4 + 11 * year) / 30` is the number of leap years before `year`.
80    (year - 1) * 354 + (4 + 11 * year).div_euclid(30) + 29 * (month - 1) + month / 2
81}
82
83/// Hijri year, month and day of the day `days` after 1 Muharram 1 AH (`days >= 0`).
84fn hijri_from_days(days: i64) -> Option<(i64, i64, i64)> {
85    if days < 0 {
86        return None;
87    }
88    // The estimate is the mean year length rounded; it can be one year off at a year boundary.
89    // Estimate and `days_before_month` both move by 30 years per 10,631 days, so one step is
90    // enough everywhere if it is within one cycle (`every_day_round_trips` checks many).
91    let mut year = (30 * days + 10646) / 10631;
92    if days < days_before_month(year, 1) {
93        year -= 1;
94    } else if days >= days_before_month(year + 1, 1) {
95        year += 1;
96    }
97    let month = ((11 * (days - days_before_month(year, 1)) + 330) / 325).clamp(1, 12);
98    let day = days - days_before_month(year, month) + 1;
99    Some((year, month, day))
100}
101
102/// 1 Muharram 1 AH is 18 July 622 (proleptic Gregorian), so that 1 January 2001 is 6 Shawwal 1421.
103fn hijri_date(dt: &DateTime) -> Option<CalendarDate> {
104    // 29 February 1900 (serial 60) is no real day and has no Hijri date. The 1900 system's day 0
105    // (`1900-01-00`, serial 0) shows as 1 January 1900, as Excel shows it.
106    if dt.day > days_in_month(dt.year, dt.month) {
107        return None;
108    }
109    let (year, month, day) = hijri_from_days(days_from_civil(dt.year, dt.month, dt.day.max(1)) - days_from_civil(622, 7, 18))?;
110    let month_name = HIJRI_MONTHS.get(usize::try_from(month - 1).ok()?).copied()?;
111    // Excel shows the full Arabic name for `ddd` as well as `dddd`, as it does for `mmm`.
112    let day_name = ARABIC_WEEKDAYS.get(dt.weekday as usize).copied()?;
113    Some(CalendarDate {
114        year,
115        month: u32::try_from(month).ok()?,
116        day: u32::try_from(day).ok()?,
117        month_name,
118        month_short: month_name,
119        day_name,
120        day_short: day_name,
121    })
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127    use crate::base::DateSystem;
128    use crate::base::date::{datetime_from_serial, max_serial};
129
130    /// Leap years of the 30-year cycle (Excel's set, per the reviewer's 43,512-serial comparison).
131    const LEAP: [i64; 11] = [2, 5, 7, 10, 13, 15, 18, 21, 24, 26, 29];
132
133    fn is_leap(year: i64) -> bool {
134        LEAP.contains(&((year - 1).rem_euclid(30) + 1))
135    }
136
137    fn month_len(year: i64, month: i64) -> i64 {
138        if month == 12 { 29 + i64::from(is_leap(year)) } else { 30 - (month + 1) % 2 }
139    }
140
141    /// Years 1 to 10,000 (every cycle up to and past 31 December 9999, which is 9666 AH): year,
142    /// month and cycle lengths follow the leap-year set.
143    #[test]
144    fn year_and_month_lengths_follow_the_leap_set() {
145        assert_eq!(days_before_month(1, 1), 0);
146        for year in 1..=10_000 {
147            let start = days_before_month(year, 1);
148            let len = days_before_month(year + 1, 1) - start;
149            assert_eq!(len, if is_leap(year) { 355 } else { 354 }, "year {year}");
150            for month in 1..=12 {
151                let next = if month == 12 { days_before_month(year + 1, 1) } else { days_before_month(year, month + 1) };
152                assert_eq!(next - days_before_month(year, month), month_len(year, month), "{year}-{month}");
153            }
154            assert_eq!(days_before_month(year + 30, 1) - start, 10_631, "cycle from {year}");
155        }
156        // One cycle: exactly the eleven leap years, 10,631 days.
157        let cycle: Vec<i64> = (1..=30).filter(|&y| days_before_month(y + 1, 1) - days_before_month(y, 1) == 355).collect();
158        assert_eq!(cycle, LEAP);
159    }
160
161    /// Every day from 1 Muharram 1 AH to one cycle past 31 December 9999: day -> Hijri -> day
162    /// round-trips, the day is within its month, and consecutive days follow each other.
163    #[test]
164    fn every_day_round_trips() {
165        let last = days_from_civil(9999, 12, 31) - days_from_civil(622, 7, 18) + 10_631;
166        let mut prev: Option<(i64, i64, i64)> = None;
167        for days in 0..=last {
168            let (y, m, d) = hijri_from_days(days).unwrap();
169            assert!((1..=12).contains(&m) && (1..=month_len(y, m)).contains(&d), "{days}: {y}-{m}-{d}");
170            assert_eq!(days_before_month(y, m) + d - 1, days, "{y}-{m}-{d}");
171            if let Some((py, pm, pd)) = prev {
172                let expected = if pd < month_len(py, pm) {
173                    (py, pm, pd + 1)
174                } else if pm < 12 {
175                    (py, pm + 1, 1)
176                } else {
177                    (py + 1, 1, 1)
178                };
179                assert_eq!((y, m, d), expected, "after {py}-{pm}-{pd}");
180            }
181            prev = Some((y, m, d));
182        }
183        assert_eq!(hijri_from_days(-1), None);
184    }
185
186    /// Every serial of the 1900 system: the Hijri date names the serial's day, one day apart.
187    #[test]
188    fn every_serial_has_the_right_hijri_day() {
189        let epoch = days_from_civil(622, 7, 18);
190        let mut prev = None;
191        for serial in 1..=max_serial(DateSystem::D1900) {
192            if serial == 60 {
193                continue;
194            }
195            let dt = datetime_from_serial(DateSystem::D1900, serial as f64).unwrap();
196            let h = Calendar::Hijri.date(&dt).unwrap();
197            assert!((1..=month_len(h.year, i64::from(h.month))).contains(&i64::from(h.day)), "serial {serial}");
198            let days = days_before_month(h.year, i64::from(h.month)) + i64::from(h.day) - 1;
199            assert_eq!(days, days_from_civil(dt.year, dt.month, dt.day) - epoch, "serial {serial}");
200            if let Some(p) = prev {
201                assert_eq!(days, p + 1, "serial {serial}");
202            }
203            prev = Some(days);
204        }
205    }
