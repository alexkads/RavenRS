9/// Text being built plus the first repeat-fill position.
10#[derive(Default)]
11pub(crate) struct Out {
12    pub text: String,
13    pub fill: Option<(char, usize)>,
14    /// True when every digit shown is zero (a number section that rounded to zero).
15    pub zero: bool,
16}
17
18impl Out {
19    fn push(&mut self, s: &str) {
20        self.text.push_str(s);
21    }
22    fn fill(&mut self, c: char) {
23        if self.fill.is_none() {
24            self.fill = Some((c, self.text.len()));
25        }
26    }
27    pub(crate) fn prepend(&mut self, s: &str) {
28        self.text.insert_str(0, s);
29        if let Some((_, p)) = &mut self.fill {
30            *p += s.len();
31        }
32    }
33}

35/// Renders a text value through a text section (`@` is replaced by the text).
36pub(crate) fn render_text(sec: &Section, text: &str) -> Out {
37    let mut out = Out::default();
38    for t in &sec.toks {
39        match t {
40            Tok::At => out.push(text),
41            Tok::Fill(c) => out.fill(*c),
42            other => out.push(&other.literal_text()),
43        }
44    }
45    out
46}

48/// Renders a number through a General section (literals around the General keyword).
49pub(crate) fn render_general(sec: &Section, v: f64, width: usize) -> Out {
50    let mut out = Out::default();
51    for t in &sec.toks {
52        match t {
53            Tok::General => out.push(&format_general_fit(v, width).unwrap_or_else(|| "#".repeat(width.clamp(1, 11)))),
54            Tok::Fill(c) => out.fill(*c),
55            other => out.push(&other.literal_text()),
56        }
57    }
58    out
59}

61/// Renders a date/time. `None` when the serial is outside the supported range.
62pub(crate) fn render_date(sec: &Section, v: f64, sys: DateSystem) -> Option<Out> {
63    if !v.is_finite() || !(0.0..max_serial(sys) as f64 + 1.0).contains(&v) {
64        return None;
65    }
66    let sub_digits = sec.toks.iter().filter_map(|t| if let Tok::Date(DatePart::SubSec(n)) = t { Some(*n) } else { None }).max().unwrap_or(0).min(3);
67    let unit: i64 = 10i64.pow(3 - sub_digits as u32);
68    let mut day = v.floor() as i64;
69    let mut ms = (((v - v.floor()) * 86_400_000.0) / unit as f64).round() as i64 * unit;
70    if ms >= 86_400_000 {
71        ms -= 86_400_000;
72        day += 1;
73    }
74    let total_ms = day.saturating_mul(86_400_000).saturating_add(ms);
75    let needs_date = sec.toks.iter().any(|t| matches!(t, Tok::Date(d) if d.is_calendar()));
76    let dt = if needs_date { datetime_from_serial(sys, day as f64)? } else { datetime_from_serial(sys, 0.0).unwrap_or_default() };
77    let ampm = sec.toks.iter().any(|t| matches!(t, Tok::Date(DatePart::AmPm | DatePart::AP(..))));
78    let hour = ms / 3_600_000;
79    let minute = (ms / 60_000) % 60;
80    let second = (ms / 1000) % 60;
81    let milli = ms % 1000;
82    let pad = |n: i64, w: u8| if w >= 2 { format!("{:0w$}", n, w = w.min(20) as usize) } else { n.to_string() };
83    let month_name = MONTHS.get((dt.month as usize).wrapping_sub(1)).copied().unwrap_or("");
84    let day_name = WEEKDAYS.get(dt.weekday as usize).copied().unwrap_or("");
85    let mut out = Out::default();
86    for t in &sec.toks {
87        match t {
88            Tok::Date(d) => {
89                let s = match *d {
90                    DatePart::Year(n) => {
91                        if n <= 2 {
92                            format!("{:02}", dt.year.rem_euclid(100))
93                        } else {
94                            format!("{:04}", dt.year)
95                        }
96                    }
97                    DatePart::Month(n) => match n {
98                        1 | 2 => pad(dt.month as i64, n),
99                        3 => month_name.chars().take(3).collect(),
100                        5 => month_name.chars().take(1).collect(),
101                        _ => month_name.to_string(),
102                    },
103                    DatePart::Day(n) => match n {
104                        1 | 2 => pad(dt.day as i64, n),
105                        3 => day_name.chars().take(3).collect(),
106                        _ => day_name.to_string(),
107                    },
108                    DatePart::Hour(n) => {
109                        let h = if ampm {
110                            let h = hour % 12;
111                            if h == 0 { 12 } else { h }
112                        } else {
113                            hour
114                        };
115                        pad(h, n)
116                    }
117                    DatePart::Minute(n) => pad(minute, n),
118                    DatePart::Second(n) => pad(second, n),
119                    DatePart::SubSec(n) => {
120                        let mut s = format!(".{:03}", milli);
121                        s.truncate(1 + (n as usize).min(3));
122                        if n > 3 {
123                            s.push_str(&"0".repeat(n as usize - 3));
124                        }
125                        s
126                    }
127                    DatePart::ElapsedH(n) => pad(total_ms / 3_600_000, n),
128                    DatePart::ElapsedM(n) => pad(total_ms / 60_000, n),
129                    DatePart::ElapsedS(n) => pad(total_ms / 1000, n),
130                    DatePart::AmPm => (if hour >= 12 { "PM" } else { "AM" }).to_string(),
131                    DatePart::AP(a, p) => (if hour >= 12 { p } else { a }).to_string(),
132                };
133                out.push(&s);
134            }
135            Tok::Fill(c) => out.fill(*c),
136            other => out.push(&other.literal_text()),
137        }
138    }
139    Some(out)
140}

142/// Assigns integer digits right-to-left to the placeholders at `idxs`; the leftmost placeholder
143/// takes all remaining digits. `force_zero` shows `0` in the units placeholder when there are no
144/// digits. A blank `?` pads with a space, and so does the thousands separator it would carry.
145fn fill_int(toks: &[Tok], idxs: &[usize], digits: &str, thousands: bool, force_zero: bool, pieces: &mut [String]) {
146    let ds: Vec<char> = digits.chars().collect();
147    let mut rem = ds.len();
148    let mut pos = 0usize;
149    let count = idxs.len();
150    for (k, &ti) in idxs.iter().rev().enumerate() {
151        let leftmost = k + 1 == count;
152        let ph = match toks.get(ti) {
153            Some(Tok::Digit(c)) => *c,
154            _ => '0',
155        };
156        let mut rev = String::new();
157        let push_digit = |c: char, rev: &mut String, pos: &mut usize| {
158            if thousands && *pos > 0 && pos.is_multiple_of(3) {
159                rev.push(',');
160            }
161            rev.push(c);
162            *pos += 1;
163        };
164        let take = if leftmost { rem } else { rem.min(1) };
165        if take == 0 {
166            match ph {
167                '0' => push_digit('0', &mut rev, &mut pos),
168                _ if force_zero && k == 0 => push_digit('0', &mut rev, &mut pos),
169                '?' => {
170                    if thousands && pos > 0 && pos.is_multiple_of(3) {
171                        rev.push(' ');
172                    }
173                    rev.push(' ');
174                    pos += 1;
175                }
176                _ => {}
177            }
178        } else {
179            for _ in 0..take {
180                rem -= 1;
181                push_digit(ds.get(rem).copied().unwrap_or('0'), &mut rev, &mut pos);
182            }
183        }
184        if let Some(p) = pieces.get_mut(ti) {
185            *p = rev.chars().rev().collect();
186        }
187    }
188}

190fn with_commas(digits: &str, thousands: bool) -> String {
191    if !thousands {
192        return digits.to_string();
193    }
194    let n = digits.chars().count();
195    let mut s = String::with_capacity(n + n / 3);
196    for (i, c) in digits.chars().enumerate() {
197        if i > 0 && (n - i).is_multiple_of(3) {
198            s.push(',');
199        }
200        s.push(c);
201    }
202    s
203}

205/// Fraction-digit placeholders: trailing zeros show as `0`, nothing (`#`) or a space (`?`).
206fn fill_frac(toks: &[Tok], idxs: &[usize], frac: &str, pieces: &mut [String]) {
207    let fd: Vec<char> = frac.chars().collect();
208    let last_sig = fd.iter().rposition(|&c| c != '0');
209    for (i, &ti) in idxs.iter().enumerate() {
210        let ph = match toks.get(ti) {
211            Some(Tok::Digit(c)) => *c,
212            _ => '0',
213        };
214        let s = if last_sig.is_some_and(|l| i <= l) {
215            fd.get(i).copied().unwrap_or('0').to_string()
216        } else {
217            match ph {
218                '0' => "0".into(),
219                '?' => " ".into(),
220                _ => String::new(),
221            }
222        };
223        if let Some(p) = pieces.get_mut(ti) {
224            *p = s;
225        }
226    }
227}

229fn digit_indices(toks: &[Tok], range: std::ops::Range<usize>) -> Vec<usize> {
230    range.filter(|&i| matches!(toks.get(i), Some(Tok::Digit(_)))).collect()
231}

233/// Where a number section's placeholders are: the decimal point, the exponent, and the digit
234/// placeholders before the point (or exponent) and between the point and the exponent.
235struct NumberLayout {
236    point: Option<usize>,
237    exp: Option<usize>,
238    int_idx: Vec<usize>,
239    frac_idx: Vec<usize>,
240}
241
242fn number_layout(toks: &[Tok]) -> NumberLayout {
243    let n = toks.len();
244    let point = toks.iter().position(|t| matches!(t, Tok::Point));
245    let exp = toks.iter().position(|t| matches!(t, Tok::Exp { .. }));
246    let int_end = point.or(exp).unwrap_or(n);
247    let frac_end = exp.unwrap_or(n);
248    let int_idx = digit_indices(toks, 0..int_end);
249    let frac_idx = match point {
250        Some(p) => digit_indices(toks, p + 1..frac_end),
251        None => Vec::new(),
252    };
253    NumberLayout { point, exp, int_idx, frac_idx }
254}

256/// The number a section works on: each `%` multiplies by 100, each scaling comma divides by 1000.
257fn scaled(sec: &Section, mut x: f64) -> f64 {
258    // Observed: `%` in a scientific section is only a literal (`#%e+#` shows 123456.789 as
259    // `1%e+5`).
260    if sec.percent > 0 && !sec.exp {
261        x *= 100f64.powi(sec.percent.min(20));
262    }
263    if sec.scale > 0 {
264        x /= 1000f64.powi(sec.scale.min(110));
265    }
266    x
267}

269/// Scientific notation of a non-negative `x` with `int_count` integer and `frac_count` fraction
270/// placeholders: the exponent (a multiple of the integer placeholder count, engineering style)
271/// and the mantissa's integer and fraction digits.
272fn scientific_parts(x: f64, int_count: usize, frac_count: usize) -> (i32, String, String) {
273    if x == 0.0 {
274        // Observed: zero shows `0` in every integer placeholder, `#` and `?` too
275        // (`#.#E+#` gives `0.E+0`, `####.####E+#` gives `0000.E+0`).
276        return (0, "0".repeat(int_count), "0".repeat(frac_count));
277    }
278    let step = int_count.max(1) as i32;
279    let (digits, xe) = decimal_digits(x);
280    let mut e = xe - xe.rem_euclid(step);
281    let (mut i, mut f) = fixed_from_digits(&digits, xe - e, frac_count);
282    if i.len() > step as usize {
283        e += step;
284        (i, f) = fixed_from_digits(&digits, xe - e, frac_count);
285    }
286    (e, i, f)
287}

289/// The number a number section shows for a non-negative `v`, as a value: what "Set precision as
290/// displayed" stores. `None` when the section shows no digits (`"x"`) or the result isn't finite.
291/// It uses the display's own rounding, so 1.005 with `0.00` gives 1.01, and 0.123456 with
292/// `0.00%` gives 0.1235 (the nearest double to the decimal shown).
293pub(crate) fn shown_value(sec: &Section, v: f64) -> Option<f64> {
294    if !sec.has_digits || !v.is_finite() {
295        return None;
296    }
297    let x = scaled(sec, v.abs());
298    if !x.is_finite() {
299        return None;
300    }
301    if x == 0.0 {
302        return Some(0.0);
303    }
304    // `%` and scaling are undone by moving the decimal point in the digits shown, so the result
305    // is the nearest double to that decimal rather than a product with its own rounding error.
306    let pct = if sec.exp { 0 } else { sec.percent.min(20) };
307    let shift = 3 * sec.scale.clamp(0, 110) - 2 * pct;
308    let shown = if sec.fraction {
309        let lay = fraction_layout(&sec.toks)?;
310        let (whole, num, den) = fraction_parts(&lay, x);
311        let den = den.max(1) as f64;
312        (whole * den + num) / den * 10f64.powi(shift)
313    } else {
314        let NumberLayout { exp, frac_idx, .. } = number_layout(&sec.toks);
315        let (e, int_s, frac_s) = match exp {
316            // Observed: the stored value keeps one digit plus the fraction digits, whatever the
317            // integer placeholders (`##0.0E+0` stores 123456.789 as 120000 though it shows
318            // `123.5E+3`; `###.00E+0` keeps three digits). A `%` in a scientific section stores a
319            // value that matches neither the display nor this rule: left as is.
320            Some(_) if sec.percent > 0 => return None,
321            Some(_) => scientific_parts(x, 1, frac_idx.len()),
322            None => {
323                let (i, f) = fixed(x, frac_idx.len());
324                (0, i, f)
325            }
326        };
327        format!("0{int_s}.{frac_s}0e{}", i64::from(e) + i64::from(shift)).parse::<f64>().ok()?
328    };
329    shown.is_finite().then_some(shown)
330}

332/// Renders a non-negative number through a number section.
333pub(crate) fn render_number(sec: &Section, v: f64) -> Out {
334    let mut x = scaled(sec, v.abs());
335    if !x.is_finite() {
336        x = f64::MAX;
337    }
338    let toks = &sec.toks;
339    let n = toks.len();
340    // Pieces replace the token's own output; `before` is inserted ahead of a token; `lits`
341    // replaces a literal's text (fractions blank the literals of a hidden part).
342    let mut pieces: Vec<String> = vec![String::new(); n];
343    let mut before: Vec<String> = vec![String::new(); n + 1];
344    let mut lits: Vec<Option<String>> = Vec::new();
345    let mut zero = false;
346    if sec.fraction {
347        lits = vec![None; n];
348        render_fraction(sec, x, &mut pieces, &mut lits);
349    } else {
350        let NumberLayout { point, exp, int_idx, frac_idx } = number_layout(toks);
351        let (int_s, frac_s) = if let Some(e_at) = exp {
352            let exp_idx = digit_indices(toks, e_at + 1..n);
353            let (mut e, int_s, frac_s) = scientific_parts(x, int_idx.len(), frac_idx.len());
354            // Exponent text.
355            let (upper, plus) = match toks.get(e_at) {
356                Some(Tok::Exp { upper, plus }) => (*upper, *plus),
357                _ => (true, true),
358            };
359            if int_s.is_empty() && frac_s.chars().all(|c| c == '0') {
360                e = 0;
361            }
362            let head = if upper { "E" } else { "e" };
363            let sign = if e < 0 {
364                "-"
365            } else if plus {
366                "+"
367            } else {
368                ""
369            };
370            let digits = e.unsigned_abs().to_string();
371            // The sign sits right before the exponent digits, after any literal that follows the
372            // `E` (observed: `|0|e+|0|` shows `|1|e|+5|`); `?` pads the digits with spaces.
373            match exp_idx.first() {
374                Some(&first) => {
375                    if let Some(p) = pieces.get_mut(e_at) {
376                        *p = head.to_string();
377                    }
378                    if let Some(b) = before.get_mut(first) {
379                        *b = sign.to_string();
380                    }
381                    fill_int(toks, &exp_idx, &digits, false, false, &mut pieces);
382                }
383                None => {
384                    if let Some(p) = pieces.get_mut(e_at) {
385                        *p = format!("{head}{sign}{digits}");
386                    }
387                }
388            }
389            (int_s, frac_s)
390        } else {
391            fixed(x, frac_idx.len())
392        };
393        zero = int_s.chars().chain(frac_s.chars()).all(|c| c == '0');
394        fill_int(toks, &int_idx, &int_s, sec.thousands, false, &mut pieces);
395        if int_idx.is_empty()
396            && !int_s.is_empty()
397            && let Some(p) = point.or(exp)
398            && let Some(b) = before.get_mut(p)
399        {
400            *b = with_commas(&int_s, sec.thousands);
401        }
402        fill_frac(toks, &frac_idx, &frac_s, &mut pieces);
403        if let Some(p) = point
404            && let Some(s) = pieces.get_mut(p)
405        {
406            *s = ".".into();
407        }
408    }
409    let mut out = Out::default();
410    for (i, t) in toks.iter().enumerate() {
411        if let Some(b) = before.get(i) {
412            out.push(b);
413        }
414        match t {
415            Tok::Lit(s) => out.push(lits.get(i).and_then(Option::as_deref).unwrap_or(s)),
416            Tok::Percent => out.push("%"),
417            Tok::Fill(c) => out.fill(*c),
418            Tok::At | Tok::General | Tok::Date(_) | Tok::Comma => {}
419            _ => out.push(pieces.get(i).map(String::as_str).unwrap_or("")),
420        }
421    }
422    out.zero = zero;
423    out
424}

426/// Best rational approximation of `x` with denominator at most `max_den` (continued fractions).
427fn approximate(x: f64, max_den: u64) -> (u64, u64) {
428    if x.is_nan() || x <= 0.0 || !x.is_finite() {
429        return (0, 1);
430    }
431    if x >= 1e15 {
432        return (x.round() as u64, 1);
433    }
434    let (mut p0, mut q0, mut p1, mut q1) = (0u64, 1u64, 1u64, 0u64);
435    let mut r = x;
436    for _ in 0..64 {
437        let a_f = r.floor();
438        if a_f > 1e15 {
439            break;
440        }
441        let a = a_f as u64;
442        let Some(q2) = a.checked_mul(q1).and_then(|v| v.checked_add(q0)) else { break };
443        if q2 > max_den {
444            break;
445        }
446        let Some(p2) = a.checked_mul(p1).and_then(|v| v.checked_add(p0)) else { break };
447        (p0, q0, p1, q1) = (p1, q1, p2, q2);
448        let frac = r - a_f;
449        if frac < 1e-12 {
450            break;
451        }
452        r = 1.0 / frac;
453    }
454    if q1 == 0 {
455        return (x.round() as u64, 1);
456    }
457    // Semiconvergent bound.
458    let k = (max_den - q0) / q1;
459    let (pb, qb) = (p0.saturating_add(k.saturating_mul(p1)), q0.saturating_add(k.saturating_mul(q1)));
460    let err_a = (p1 as f64 / q1 as f64 - x).abs();
461    let err_b = if qb > 0 { (pb as f64 / qb as f64 - x).abs() } else { f64::INFINITY };
462    if err_b < err_a { (pb, qb) } else { (p1, q1) }
463}

465/// Display width of a token in a fraction section, for blanking a hidden part with spaces.
466fn shown_width(t: &Tok) -> usize {
467    match t {
468        Tok::Lit(s) | Tok::FixedDenom(s) => s.chars().count(),
469        Tok::Digit(_) | Tok::Slash => 1,
470        _ => 0,
471    }

474/// Replaces the literals in `range` with spaces (`spaces`) or nothing; placeholders and the bar
475/// in it get the same treatment through `pieces`.
476fn blank_range(toks: &[Tok], range: std::ops::Range<usize>, spaces: bool, pieces: &mut [String], lits: &mut [Option<String>]) {
477    for i in range {
478        let Some(t) = toks.get(i) else { break };
479        let s = if spaces { " ".repeat(shown_width(t)) } else { String::new() };
480        match t {
481            Tok::Lit(_) => {
482                if let Some(l) = lits.get_mut(i) {
483                    *l = Some(s);
484                }
485            }
486            Tok::Digit(_) | Tok::Slash | Tok::FixedDenom(_) => {
487                if let Some(p) = pieces.get_mut(i) {
488                    *p = s;
489                }
490            }
491            _ => {}
492        }
493    }
494}

496/// Where a fraction section's parts are (see [`render_fraction`]).
497struct FractionLayout {
498    slash: usize,
499    num_start: usize,
500    whole_idx: Vec<usize>,
501    num_idx: Vec<usize>,
502    sep_start: usize,
503    fixed_den: Option<u64>,
504    den_start: usize,
505    den_end: usize,
506    den_idx: Vec<usize>,
507}
508
509fn fraction_layout(toks: &[Tok]) -> Option<FractionLayout> {
510    let is_digit = |i: usize| matches!(toks.get(i), Some(Tok::Digit(_)));
511    let slash = toks.iter().position(|t| matches!(t, Tok::Slash))?;
512    let num_last = (0..slash).rev().find(|&i| is_digit(i))?;
513    let mut num_start = num_last;
514    while num_start > 0 && is_digit(num_start - 1) {
515        num_start -= 1;
516    }
517    let whole_idx = digit_indices(toks, 0..num_start);
518    let num_idx = digit_indices(toks, num_start..num_last + 1);
519    let sep_start = whole_idx.last().map_or(num_start, |&w| w + 1);
520    let (fixed_den, den_start): (Option<u64>, usize) = match toks.get(slash + 1) {
521        Some(Tok::FixedDenom(s)) => (Some(s.parse::<u64>().unwrap_or(u64::MAX).clamp(1, 1_000_000_000)), slash + 1),
522        _ => {
523            let mut i = slash + 1;
524            while matches!(toks.get(i), Some(Tok::Lit(_))) {
525                i += 1;
526            }
527            (None, i)
528        }
529    };
530    let mut den_end = den_start;
531    if fixed_den.is_some() {
532        den_end += 1;
533    } else {
534        while is_digit(den_end) {
535            den_end += 1;
536        }
537    }
538    let den_idx = digit_indices(toks, den_start..den_end);
539    Some(FractionLayout { slash, num_start, whole_idx, num_idx, sep_start, fixed_den, den_start, den_end, den_idx })
540}

542/// The whole part, numerator and denominator a fraction section shows for a non-negative `x`.
543fn fraction_parts(lay: &FractionLayout, x: f64) -> (f64, f64, u64) {
544    let has_whole = !lay.whole_idx.is_empty();
545    let (mut whole, f) = if has_whole { (x.floor(), x - x.floor()) } else { (0.0, x) };
546    // Without a whole part the numerator carries the whole value and can pass u64::MAX, so it
547    // stays an f64 (a float-to-u64 cast saturates).
548    let (mut num, den): (f64, u64) = match lay.fixed_den {
549        Some(d) => ((f * d as f64).round(), d),
550        None if f >= 1e15 => (f.round(), 1),
551        None => {
552            let digits = lay.den_idx.len().clamp(1, 7) as u32;
553            let (n, d) = approximate(f, 10u64.pow(digits) - 1);
554            (n as f64, d)
555        }
556    };
557    if has_whole && num >= den as f64 && den > 0 {
558        // The fractional part is below 1, so the numerator is at most the denominator here.
559        let n = num as u64;
560        whole += (n / den) as f64;
561        num = (n % den) as f64;
562    }
563    (whole, num, den)
564}
565
566/// A fraction section is `[whole] [sep] num [lit] / [lit] den`: the numerator is the run of
567/// placeholders ending at the last one before the bar, the whole part is every placeholder
568/// before it (literals between them stay), `sep` the literals between the two parts.
569///
570/// Observed Excel behaviour: when the whole part shows no digit its separator disappears (or
571/// becomes spaces when the whole part or the numerator has a `?`); when the fraction is zero and
572/// the numerator has no `0` placeholder, the separator, numerator, bar and denominator disappear
573/// (or become spaces when any part has a `?`) and the whole part shows at least `0`.
574fn render_fraction(sec: &Section, x: f64, pieces: &mut [String], lits: &mut [Option<String>]) {
575    let toks = &sec.toks;
576    let Some(lay) = fraction_layout(toks) else { return };
577    let (whole, num, den) = fraction_parts(&lay, x);
578    let FractionLayout { slash, num_start, whole_idx, num_idx, sep_start, fixed_den, den_start, den_end, den_idx } = lay;
579    let has_q = |idx: &[usize]| idx.iter().any(|&i| toks.get(i) == Some(&Tok::Digit('?')));
580    let has_whole = !whole_idx.is_empty();
581    let whole_s = if whole >= 1.0 { fixed(whole, 0).0 } else { String::new() };
582    let num_zero_ph = num_idx.iter().any(|&i| toks.get(i) == Some(&Tok::Digit('0')));
583    if has_whole && num == 0.0 && !num_zero_ph {
584        fill_int(toks, &whole_idx, &whole_s, sec.thousands, true, pieces);
585        let spaces = has_q(&whole_idx) || has_q(&num_idx) || has_q(&den_idx);
586        blank_range(toks, sep_start..den_end, spaces, pieces, lits);
587        return;
588    }
589    if has_whole {
590        // A zero value shows `0` in a `?` or `0` units place, nothing in a `#` one.
591        let units_hash = whole_idx.last().is_some_and(|&i| toks.get(i) == Some(&Tok::Digit('#')));
592        let force = whole_s.is_empty() && num == 0.0 && !units_hash;
593        fill_int(toks, &whole_idx, &whole_s, sec.thousands, force, pieces);
594        let whole_shown = whole_idx.iter().any(|&i| pieces.get(i).is_some_and(|p| p.chars().any(|c| c.is_ascii_digit())));
595        if !whole_shown {
596            let spaces = has_q(&whole_idx) || has_q(&num_idx);
597            blank_range(toks, sep_start..num_start, spaces, pieces, lits);
598        }
599    }
600    let num_s = if num < 1e15 { (num as u64).to_string() } else { fixed(num, 0).0 };
601    fill_int(toks, &num_idx, &num_s, false, true, pieces);
602    if let Some(p) = pieces.get_mut(slash) {
603        *p = "/".into();
604    }
605    let den_s = den.to_string();
606    if fixed_den.is_some() {
607        if let Some(p) = pieces.get_mut(den_start) {
608            *p = den_s;
609        }
610    } else {
611        // Denominator digits are zero padded to the `0` placeholders, then left aligned; `?`
612        // pads with spaces after them.
613        let zeros = den_idx.iter().filter(|&&i| toks.get(i) == Some(&Tok::Digit('0'))).count();
614        // No `{:0>zeros$}`: a format width above u16::MAX panics.
615        let mut padded = "0".repeat(zeros.saturating_sub(den_s.chars().count()));
616        padded.push_str(&den_s);
617        let dc: Vec<char> = padded.chars().collect();
618        let count = den_idx.len();
619        for (k, &ti) in den_idx.iter().enumerate() {
620            let s = if k + 1 == count {
621                dc.get(k..).map(|r| r.iter().collect::<String>()).unwrap_or_default()
622            } else {
623                dc.get(k).map(ToString::to_string).unwrap_or_default()
624            };
625            let s = if s.is_empty() {
626                match toks.get(ti) {
627                    Some(Tok::Digit('?')) => " ".into(),
628                    Some(Tok::Digit('0')) => "0".into(),
629                    _ => String::new(),
630                }
631            } else {
632                s
633            };
634            if let Some(p) = pieces.get_mut(ti) {
635                *p = s;
636            }
637        }
638    }
639}
