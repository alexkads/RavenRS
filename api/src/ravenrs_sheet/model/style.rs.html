17/// A colour: explicit RGB, a theme slot with tint, or automatic (black text / no fill).
18#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Default)]
19pub enum Color {
20    #[default]
21    Auto,
22    /// 0xRRGGBB
23    Rgb(u32),
24    /// Theme index (0 = Background 1 / lt1, 1 = Text 1 / dk1, 2 = lt2, 3 = dk2, 4–9 = Accent 1–6,
25    /// 10 hyperlink, 11 followed) and tint in thousandths (-1000..=1000).
26    Theme(u8, i16),
27}
28
29impl Color {
30    pub fn rgb(r: u8, g: u8, b: u8) -> Color {
31        Color::Rgb(((r as u32) << 16) | ((g as u32) << 8) | b as u32)
32    }
33    pub fn from_hex(s: &str) -> Option<Color> {
34        let h = s.trim().trim_start_matches('#');
35        let h = if h.len() == 8 { h.get(2..)? } else { h };
36        if h.len() != 6 {
37            return None;
38        }
39        u32::from_str_radix(h, 16).ok().map(Color::Rgb)
40    }
41    /// Resolved RGB, `None` for automatic.
42    pub fn resolve(&self, theme: &Theme) -> Option<[u8; 3]> {
43        match *self {
44            Color::Auto => None,
45            Color::Rgb(v) => Some([(v >> 16) as u8, (v >> 8) as u8, v as u8]),
46            Color::Theme(i, tint) => {
47                let base = theme.colors.get(i as usize).copied().unwrap_or(0);
48                Some(apply_tint([(base >> 16) as u8, (base >> 8) as u8, base as u8], tint as f64 / 1000.0))
49            }
50        }
51    }
52    pub fn hex(&self, theme: &Theme) -> Option<String> {
53        self.resolve(theme).map(|[r, g, b]| format!("#{r:02X}{g:02X}{b:02X}"))
54    }
55}

57/// Tint as in ECMA-376: lighten towards white (positive) or darken (negative) in HSL luminance.
58pub fn apply_tint(rgb: [u8; 3], tint: f64) -> [u8; 3] {
59    if tint == 0.0 {
60        return rgb;
61    }
62    let (h, s, l) = rgb_to_hsl(rgb);
63    let l = if tint < 0.0 { l * (1.0 + tint) } else { l * (1.0 - tint) + tint };
64    hsl_to_rgb(h, s, l.clamp(0.0, 1.0))
65}
66
67fn rgb_to_hsl([r, g, b]: [u8; 3]) -> (f64, f64, f64) {
68    let (r, g, b) = (r as f64 / 255.0, g as f64 / 255.0, b as f64 / 255.0);
69    let max = r.max(g).max(b);
70    let min = r.min(g).min(b);
71    let l = (max + min) / 2.0;
72    if max == min {
73        return (0.0, 0.0, l);
74    }
75    let d = max - min;
76    let s = if l > 0.5 { d / (2.0 - max - min) } else { d / (max + min) };
77    let h = if max == r {
78        (g - b) / d + if g < b { 6.0 } else { 0.0 }
79    } else if max == g {
80        (b - r) / d + 2.0
81    } else {
82        (r - g) / d + 4.0
83    };
84    (h / 6.0, s, l)
85}
86
87fn hsl_to_rgb(h: f64, s: f64, l: f64) -> [u8; 3] {
88    if s == 0.0 {
89        let v = (l * 255.0).round() as u8;
90        return [v, v, v];
91    }
92    let q = if l < 0.5 { l * (1.0 + s) } else { l + s - l * s };
93    let p = 2.0 * l - q;
94    let hue = |mut t: f64| {
95        if t < 0.0 {
96            t += 1.0;
97        }
98        if t > 1.0 {
99            t -= 1.0;
100        }
101        if t < 1.0 / 6.0 {
102            p + (q - p) * 6.0 * t
103        } else if t < 0.5 {
104            q
105        } else if t < 2.0 / 3.0 {
106            p + (q - p) * (2.0 / 3.0 - t) * 6.0
107        } else {
108            p
109        }
110    };
111    [(hue(h + 1.0 / 3.0) * 255.0).round() as u8, (hue(h) * 255.0).round() as u8, (hue(h - 1.0 / 3.0) * 255.0).round() as u8]
112}

390/// The parts a differential format states at a value its `Style` can't tell from unset: a dxf's
391/// default parts are unset, so a bold turned off, a face, a size or a protection that matches the
392/// defaults would otherwise read as nothing. The values are the style's own; this records that
393/// the format states them, and only where they match the defaults, so a format built in the app
394/// and the same format read back from a file stay equal.
395#[derive(Clone, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
396pub struct DxfStated {
397    pub bold: bool,
398    pub italic: bool,
399    pub strike: bool,
400    pub underline: bool,
401    pub size: bool,
402    pub name: bool,
403    /// `<scheme val>`: the theme font the face follows (`major`, `minor` or `none`).
404    pub scheme: Option<String>,
405    pub protection: bool,
406}

426/// Interned styles. Index 0 is the default (Normal) style.
427#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
428pub struct StyleTable {
429    styles: Vec<Style>,
430    #[serde(skip)]
431    index: HashMap<Style, StyleId>,
432}
433
434impl Default for StyleTable {
435    fn default() -> Self {
436        let d = Style::default();
437        let mut index = HashMap::new();
438        index.insert(d.clone(), StyleId::DEFAULT);
439        StyleTable { styles: vec![d], index }
440    }
441}
442
443impl StyleTable {
444    /// A table whose default (Normal) style is `normal`, such as a file's own Normal font.
445    pub fn with_default(normal: Style) -> Self {
446        let mut index = HashMap::new();
447        index.insert(normal.clone(), StyleId::DEFAULT);
448        StyleTable { styles: vec![normal], index }
449    }
450    pub fn get(&self, id: StyleId) -> &Style {
451        self.styles.get(id.0 as usize).or_else(|| self.styles.first()).unwrap_or(&DEFAULT_STYLE)
452    }
453    pub fn intern(&mut self, s: Style) -> StyleId {
454        if self.index.is_empty() && !self.styles.is_empty() {
455            self.rebuild_index();
456        }
457        if let Some(id) = self.index.get(&s) {
458            return *id;
459        }
460        let id = StyleId(self.styles.len() as u32);
461        self.index.insert(s.clone(), id);
462        self.styles.push(s);
463        id
464    }
465    /// The style `id` changed by `f`, interned.
466    pub fn derive(&mut self, id: StyleId, f: impl FnOnce(&mut Style)) -> StyleId {
467        let mut s = self.get(id).clone();
468        f(&mut s);
469        self.intern(s)
470    }
471    pub fn len(&self) -> usize {
472        self.styles.len()
473    }
474    pub fn is_empty(&self) -> bool {
475        self.styles.is_empty()
476    }
477    pub fn iter(&self) -> impl Iterator<Item = (StyleId, &Style)> {
478        self.styles.iter().enumerate().map(|(i, s)| (StyleId(i as u32), s))
479    }
480    pub fn rebuild_index(&mut self) {
481        self.index = self.styles.iter().enumerate().map(|(i, s)| (s.clone(), StyleId(i as u32))).collect();
482    }
483}
