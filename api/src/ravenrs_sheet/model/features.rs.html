21    Average,
22    Count,
23    CountNums,
24    Max,

26    Sum,

28    Var,

32impl TotalsFn {
33    /// SUBTOTAL function number.
34    pub fn subtotal_code(&self) -> Option<u32> {
35        Some(match self {
36            TotalsFn::Average => 101,
37            TotalsFn::Count => 103,
38            TotalsFn::CountNums => 102,
39            TotalsFn::Max => 104,
40            TotalsFn::Min => 105,
41            TotalsFn::Sum => 109,
42            TotalsFn::StdDev => 107,
43            TotalsFn::Var => 110,
44            _ => return None,
45        })
46    }
47}

104impl Table {
105    pub fn data_range(&self) -> Option<RangeRef> {
106        let r0 = self.range.start.row + self.header_row as u32;
107        let r1 = self.range.end.row.checked_sub(self.totals_row as u32)?;
108        if r1 < r0 {
109            return None;
110        }
111        Some(RangeRef::new(CellRef::new(r0, self.range.start.col), CellRef::new(r1, self.range.end.col)))
112    }
113    pub fn header_range(&self) -> Option<RangeRef> {
114        self.header_row.then(|| RangeRef::new(self.range.start, CellRef::new(self.range.start.row, self.range.end.col)))
115    }
116    pub fn totals_range(&self) -> Option<RangeRef> {
117        self.totals_row.then(|| RangeRef::new(CellRef::new(self.range.end.row, self.range.start.col), self.range.end))
118    }
119    pub fn column_index(&self, name: &str) -> Option<usize> {
120        self.columns.iter().position(|c| c.name.eq_ignore_ascii_case(name))
121    }
122}

233/// Number of icons in an icon set (`3Arrows` → 3, `5Quarters` → 5), always 3 to 5.
234pub fn icon_count(set: &str) -> usize {
235    set.chars().next().and_then(|c| c.to_digit(10)).unwrap_or(3).clamp(3, 5) as usize
236}

238/// Excel's default icon-set thresholds: one per icon, the first being the lower bound,
239/// then even percent steps (`0, 33, 67` for three icons).
240pub fn default_icon_thresholds(set: &str) -> Vec<CfValueKind> {
241    let k = icon_count(set);
242    (0..k).map(|i| CfValueKind::Percent((i as f64 * 100.0 / k as f64).round())).collect()
243}

596impl Chart {
597    /// Data tables belong to category-axis column, bar, line and area charts.
598    pub fn supports_data_table(&self) -> bool {
599        let eligible = |kind| {
600            matches!(
601                kind,
602                ChartKind::ColumnClustered
603                    | ChartKind::ColumnStacked
604                    | ChartKind::ColumnStacked100
605                    | ChartKind::BarClustered
606                    | ChartKind::BarStacked
607                    | ChartKind::BarStacked100
608                    | ChartKind::Line
609                    | ChartKind::LineMarkers
610                    | ChartKind::LineStacked
611                    | ChartKind::Area
612                    | ChartKind::AreaStacked
613                    | ChartKind::AreaStacked100
614            )
615        };
616        eligible(self.kind) || (self.kind == ChartKind::Combo && self.series.iter().all(|s| eligible(s.kind.unwrap_or(ChartKind::Line))))
617    }

668/// The laid-out form of an equation's LaTeX source. It is derived data: rebuilt by
669/// [`EquationCache::refresh`] whenever the source changes, never saved, and ignored when shapes
670/// are compared. Painters read it and never lay equations out themselves, so nothing in a paint
671/// path can reach the layout engine.
672///
673/// An entry answers only for the source it was built from, compared in full: a hash match alone
674/// is not enough, so two sources whose hashes collide never share a layout. A source over
675/// [`crate::math::MAX_SOURCE_BYTES`] never lays out, so [`EquationCache::get`] answers for it
676/// by its length alone, with [`crate::math::SOURCE_TOO_LONG`], and an entry built from one holds
677/// neither the source nor its hash: a painter asking every frame about a source of megabytes read
678/// from a file does no work on it.
679#[derive(Clone, Debug, Default)]
680pub struct EquationCache {
681    /// A hash of the source the entry was built from (a quick first check).
682    key: u64,
683    /// The source's length in bytes.
684    len: usize,
685    /// The source itself, when it is within the length cap.
686    source: Option<std::sync::Arc<str>>,
687    layout: Option<Result<std::sync::Arc<crate::math::MathList>, crate::math::MathError>>,
688}
689
690impl PartialEq for EquationCache {
691    fn eq(&self, _: &Self) -> bool {
692        true
693    }
694}
695
696fn source_key(src: &str) -> u64 {
697    use std::hash::{Hash, Hasher};
698    #[cfg(test)]
699    equation_cache_tests::HASHES.with(|h| h.set(h.get() + 1));
700    let mut h = std::collections::hash_map::DefaultHasher::new();
701    src.hash(&mut h);
702    h.finish()
703}
704
705impl EquationCache {
706    /// Whether this entry was built from `src`. Over the cap the length is the match (an entry for
707    /// such a source is always the same error); within it, the hash and then the source itself.
708    fn holds(&self, src: &str) -> bool {
709        if self.layout.is_none() || self.len != src.len() {
710            return false;
711        }
712        if src.len() > crate::math::MAX_SOURCE_BYTES {
713            return self.source.is_none();
714        }
715        self.key == source_key(src) && self.source.as_deref() == Some(src)
716    }
717
718    /// Makes this the entry for `src`, with `layout`. A source over the cap is neither copied nor
719    /// hashed.
720    fn store(&mut self, src: &str, layout: Result<std::sync::Arc<crate::math::MathList>, crate::math::MathError>) {
721        let within = src.len() <= crate::math::MAX_SOURCE_BYTES;
722        self.key = if within { source_key(src) } else { 0 };
723        self.len = src.len();
724        self.source = within.then(|| std::sync::Arc::from(src));
725        self.layout = Some(layout);
726    }
727
728    /// The layout of `src`, or why it has none; `None` until [`EquationCache::refresh`] has run
729    /// for this source. A source over [`crate::math::MAX_SOURCE_BYTES`] is answered at once,
730    /// whatever the entry holds, by its length alone.
731    pub fn get(&self, src: &str) -> Option<Result<&crate::math::MathList, &crate::math::MathError>> {
732        if src.len() > crate::math::MAX_SOURCE_BYTES {
733            return Some(Err(&*crate::math::SOURCE_TOO_LONG));
734        }
735        if !self.holds(src) {
736            return None;
737        }
738        self.layout.as_ref().map(|r| r.as_ref().map(|l| l.as_ref()))
739    }
740
741    /// Lays `src` out again unless the cache already holds its layout. True when it did.
742    pub fn refresh(&mut self, src: &str) -> bool {
743        if self.holds(src) {
744            return false;
745        }
746        self.store(src, crate::math::layout(src).map(std::sync::Arc::new));
747        true
748    }
749
750    /// Records that `src` was not laid out, and why, without running the layout engine.
751    pub fn refuse(&mut self, src: &str, message: &str) {
752        self.store(src, Err(crate::math::MathError { span: None, message: message.to_string() }));
753    }
754}

866impl FormControl {
867    /// The name shown for the control: its own, else `Button 3`, `Slider 4`, `Spin Button 5`.
868    pub fn display_name(&self) -> String {
869        if self.name.trim().is_empty() { format!("{} {}", self.kind.label(), self.id) } else { self.name.clone() }
870    }
871
872    /// Checks the numbers of a slider or spin button: 0 ≤ min < max ≤ 30000, 1 ≤ step ≤ max − min,
873    /// and (sliders) page ≥ step. Each rule has its own message.
874    pub fn check_numbers(&self) -> Result<(), String> {
875        if self.kind == ControlKind::Button {
876            return Ok(());
877        }
878        if self.max > CONTROL_MAX {
879            return Err(format!("Maximum value can be at most {CONTROL_MAX}."));
880        }
881        if self.min > CONTROL_MAX {
882            return Err(format!("Minimum value must be a whole number from 0 to {CONTROL_MAX}."));
883        }
884        if self.min >= self.max {
885            return Err(format!("Maximum value ({}) must be greater than Minimum value ({}).", self.max, self.min));
886        }
887        if self.step < 1 {
888            return Err("Incremental change must be at least 1.".into());
889        }
890        let span = self.max - self.min;
891        if self.step > span {
892            return Err(format!("Incremental change ({}) can be at most Maximum minus Minimum ({span}).", self.step));
893        }
894        if self.kind == ControlKind::Slider {
895            if self.page < self.step {
896                return Err(format!("Page change ({}) must be at least the incremental change ({}).", self.page, self.step));
897            }
898            if self.page > CONTROL_MAX {
899                return Err(format!("Page change can be at most {CONTROL_MAX}."));
900            }
901        }
902        Ok(())
903    }
904
905    /// `v` clamped to min..max and snapped to the nearest step from min (max stays reachable).
906    pub fn snap(&self, v: f64) -> u32 {
907        let (min, max) = (self.min.min(CONTROL_MAX), self.max.min(CONTROL_MAX).max(self.min.min(CONTROL_MAX)));
908        if !v.is_finite() || v <= min as f64 {
909            return min;
910        }
911        if v >= max as f64 {
912            return max;
913        }
914        let step = self.step.max(1) as f64;
915        let snapped = (min as f64 + ((v - min as f64) / step).round() * step).clamp(min as f64, max as f64);
916        // When the step doesn't divide the range, the maximum is a stop of its own.
917        if (max as f64 - v).abs() < (snapped - v).abs() { max } else { snapped as u32 }
918    }
919}
