12/// A parsed formula. `text` is the canonical formula text without the leading `=`.
13#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
14#[serde(from = "SavedFormula")]
15pub struct Formula {
16    pub text: String,
17    #[serde(skip)]
18    pub expr: Option<Arc<Expr>>,
19    /// Legacy array formula (Ctrl+Shift+Enter) covering this range; the formula lives in the
20    /// top-left cell.
21    pub array: Option<RangeRef>,
22    /// Read from a file that marks this formula as a dynamic-array formula (cell metadata `cm`).
23    /// The writer keeps the mark, so a formula Excel evaluates as an array is never saved as a
24    /// legacy formula (which Excel evaluates with implicit intersection). Formulas entered or
25    /// rewritten in GridCraft start without it, and the writer decides from their shape.
26    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
27    pub dynamic: bool,
28}
29
30impl Formula {
31    /// Parses `text` (with or without `=`). Unparseable formulas keep their text and evaluate to
32    /// `#NAME?`.
33    pub fn new(text: &str) -> Formula {
34        let body = text.strip_prefix('=').unwrap_or(text);
35        match crate::formula::parse(body) {
36            Ok(e) => Formula { text: crate::formula::print(&e), expr: Some(Arc::new(e)), array: None, dynamic: false },
37            Err(_) => Formula { text: body.to_string(), expr: None, array: None, dynamic: false },
38        }
39    }
40    pub fn from_expr(e: Expr) -> Formula {
41        Formula { text: crate::formula::print(&e), expr: Some(Arc::new(e)), array: None, dynamic: false }
42    }
43    /// The parsed expression, re-parsing when no tree is cached.
44    pub fn expr(&self) -> Option<Expr> {
45        self.expr.as_deref().cloned().or_else(|| crate::formula::parse(&self.text).ok())
46    }
47    /// The parsed expression without copying it (re-parses when no tree is cached).
48    pub fn expr_arc(&self) -> Option<Arc<Expr>> {
49        self.expr.clone().or_else(|| crate::formula::parse(&self.text).ok().map(Arc::new))
50    }
51    /// Runs `f` on the parsed expression, borrowing the cached tree or, when none is cached,
52    /// a freshly parsed one that is dropped afterwards. Unlike [`expr`](Self::expr) it never copies
53    /// the tree, and unlike [`expr_arc`](Self::expr_arc) it needs no reference count or `Arc`
54    /// allocation. `None` when the text does not parse.
55    pub fn with_expr<R>(&self, f: impl FnOnce(&Expr) -> R) -> Option<R> {
56        match &self.expr {
57            Some(e) => Some(f(e)),
58            None => crate::formula::parse(&self.text).ok().map(|e| f(&e)),
59        }
60    }
61}

63/// A formula as a saved workbook holds it: the tree is not saved. Reading one parses its text
64/// once and caches the tree, so a workbook opened from JSON does not parse every formula again on
65/// each recalculation. The text is kept as saved, and text that doesn't parse caches nothing, as
66/// [`Formula::new`] does.
67#[derive(Deserialize)]
68struct SavedFormula {
69    text: String,
70    array: Option<RangeRef>,
71    #[serde(default)]
72    dynamic: bool,
73}
74
75impl From<SavedFormula> for Formula {
76    fn from(f: SavedFormula) -> Formula {
77        let expr = crate::formula::parse(&f.text).ok().map(Arc::new);
78        Formula { text: f.text, expr, array: f.array, dynamic: f.dynamic }
79    }
80}

92#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
93pub struct Cell {
94    /// Constant value, or the cached result of the formula or Python code.
95    pub value: Value,
96    pub formula: Option<Arc<Formula>>,
97    pub style: StyleId,
98    /// Python code, for a Python cell (which has no formula).
99    #[serde(default, skip_serializing_if = "Option::is_none")]
100    pub py: Option<Arc<PyCode>>,
101}
102
103impl Cell {
104    pub fn value(v: Value) -> Cell {
105        Cell { value: v, formula: None, style: StyleId::DEFAULT, py: None }
106    }
107    pub fn formula(f: Formula) -> Cell {
108        Cell { value: Value::Empty, formula: Some(Arc::new(f)), style: StyleId::DEFAULT, py: None }
109    }
110    /// A value cell with this style.
111    pub fn styled(v: Value, style: StyleId) -> Cell {
112        Cell { value: v, formula: None, style, py: None }
113    }
114    /// No value, no formula, no Python code and the default style: storing it is pointless.
115    pub fn is_blank(&self) -> bool {
116        self.value.is_empty() && self.formula.is_none() && self.style == StyleId::DEFAULT && self.py.is_none()
117    }
118    /// Whether the cell holds content (a value, a formula or Python code), formats aside.
119    pub fn has_content(&self) -> bool {
120        !self.value.is_empty() || self.formula.is_some() || self.py.is_some()
121    }
122    /// Removes the content (value, formula and Python code), keeping the style.
123    pub fn clear_content(&mut self) {
124        self.value = Value::Empty;
125        self.formula = None;
126        self.py = None;
127    }
128    /// What the formula bar shows: a formula, a Python cell's code, or the value.
129    pub fn input_text(&self) -> String {
130        if let Some(py) = &self.py {
131            return py.code.clone();
132        }
133        if let Some(f) = &self.formula {
134            return format!("={}", f.text);
135        }
136        match &self.value {
137            Value::Text(t) => {
138                // Text that would parse as something else is shown with its apostrophe by Excel
139                // only when entered that way; we show it plainly.
140                t.to_string()
141            }
142            v => v.display(),
143        }
144    }
145    /// What the formula bar shows when not editing: [`input_text`](Self::input_text), with a
146    /// legacy array formula in braces (`{=A1:A2*2}`).
147    pub fn bar_text(&self) -> String {
148        match &self.formula {
149            Some(f) if f.array.is_some() => format!("{{={}}}", f.text),
150            _ => self.input_text(),
151        }
152    }
153}

160    #[test]
161    fn formulas_read_from_json_keep_their_text_and_cache_the_tree() -> Result<(), serde_json::Error> {
162        let mut array = Formula::new("=A1:B2*2");
163        array.array = Some(RangeRef::new(CellRef::new(0, 2), CellRef::new(1, 3)));
164        array.dynamic = true;
165        for f in [Formula::new("=SUM(A1:A3)+'My Sheet'!B2"), Formula::new("=LET(x,A1*2,x+1)"), array] {
166            let back: Formula = serde_json::from_str(&serde_json::to_string(&f)?)?;
167            assert!(back.expr.is_some(), "{}: the tree is cached after reading", f.text);
168            assert_eq!(back, f);
169        }
170        // Saved text is kept as written, even when the canonical print differs, and its tree is
171        // the one the text parses to.
172        let back: Formula = serde_json::from_str(r#"{"text": "sum( a1 , 2 )", "array": null}"#)?;
173        assert_eq!(back.text, "sum( a1 , 2 )");
174        assert!(!back.dynamic);
175        assert_eq!(back.expr.as_deref(), crate::formula::parse("sum( a1 , 2 )").ok().as_ref());
176        assert!(back.expr.is_some());
177        // Text that doesn't parse caches nothing, as `Formula::new` does.
178        let bad = Formula::new("=SUM(");
179        let back: Formula = serde_json::from_str(&serde_json::to_string(&bad)?)?;
180        assert_eq!(back, bad);
181        assert!(back.expr.is_none());
182        Ok(())
183    }

185    #[test]
186    fn a_workbook_read_from_json_has_every_formula_tree() -> Result<(), Box<dyn std::error::Error>> {
187        let mut wb = crate::model::Workbook::new();
188        let mut array = Formula::new("=A1:A3*2");
189        array.array = Some(RangeRef::new(CellRef::new(0, 3), CellRef::new(2, 3)));
190        let sheet = wb.sheet_mut(0).ok_or("no sheet")?;
191        sheet.cells.set(CellRef::new(0, 0), Cell::value(Value::Number(1.0)));
192        sheet.cells.set(CellRef::new(0, 1), Cell::formula(Formula::new("=SUM(A1:A3)")));
193        sheet.cells.set(CellRef::new(1, 1), Cell::formula(Formula::new("=IF(A1>0,\"yes\",\"no\")")));
194        sheet.cells.set(CellRef::new(2, 1), Cell::formula(Formula::new("=SUM(")));
195        sheet.cells.set(CellRef::new(0, 3), Cell::formula(array));
196        let json = serde_json::to_string(&wb)?;
197        let back: crate::model::Workbook = serde_json::from_str(&json)?;
198        let (before, after) = (wb.sheet(0).ok_or("no sheet")?, back.sheet(0).ok_or("no sheet")?);
199        let mut formulas = 0;
200        for (c, cell) in before.cells.iter() {
201            let (Some(f), Some(read)) = (&cell.formula, after.cell(c).and_then(|cell| cell.formula.as_ref())) else { continue };
202            assert_eq!(read, f, "{c:?}: text, tree and flags as saved");
203            assert_eq!(read.expr.is_some(), crate::formula::parse(&f.text).is_ok(), "{c:?}: every formula that parses has its tree");
204            formulas += 1;
205        }
206        assert_eq!(formulas, 4);
207        Ok(())
208    }
