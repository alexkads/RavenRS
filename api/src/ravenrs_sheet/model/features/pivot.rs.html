6/// How a PivotTable lays out its row fields.
7#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
8pub enum PivotLayout {
9    /// All row fields in one column, inner items indented ("Row Labels" header).
10    #[default]
11    Compact,
12    /// One column per row field; each item on its own row.
13    Outline,
14    /// One column per row field; inner items start on their parent's row, totals below.
15    Tabular,
16}

102/// A PivotTable: its definition plus where its output was last written. The output itself is
103/// static cell values and styles on the sheet, rewritten on every refresh.
104#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
105#[serde(default)]
106pub struct PivotTable {
107    pub id: u32,
108    pub name: String,
109    /// Source: range formula text (`Sheet1!$A$1:$E$200`), a table name or a defined name.
110    pub source: String,
111    /// Top-left of the output body on the pivot's sheet.
112    pub anchor: CellRef,
113    pub rows: Vec<PivotField>,
114    pub columns: Vec<PivotField>,
115    pub values: Vec<PivotValue>,
116    pub filters: Vec<PivotFilter>,
117    pub layout: PivotLayout,
118    /// PivotTable Options › Data › Enable show details: Show Details lists a value cell's
119    /// source records only while this is on.
120    pub enable_show_details: bool,
121    /// Grand total for each row (the "Grand Total" column), like Excel's "On for Rows".
122    pub grand_totals_rows: bool,
123    /// Grand total for each column (the "Grand Total" row), like Excel's "On for Columns".
124    pub grand_totals_cols: bool,
125    pub subtotals: bool,
126    /// Subtotals at the top of each group (Compact/Outline); otherwise below it.
127    pub subtotals_top: bool,
128    pub style: String,
129    pub show_headers: bool,
130    /// Area written by the last refresh.
131    pub last_range: Option<RangeRef>,
132    /// Calculated fields, usable in the Values area by name.
133    pub calculated_fields: Vec<PivotCalcField>,
134    /// The Σ Values pseudo-field (it exists with two or more value fields) sits in Rows rather
135    /// than Columns (Excel's "values on rows").
136    pub values_on_rows: bool,
137    /// Position of Σ Values among the row fields (`None` = after the last). Only used with
138    /// `values_on_rows`: in Columns Σ Values always comes after the last column field.
139    pub values_position: Option<usize>,
140}
