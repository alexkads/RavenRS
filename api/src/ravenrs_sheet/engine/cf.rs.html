37/// Which statistics a rule reads. Scanning the rule's whole range for them costs O(range), so
38/// rules that read none (cell value, formula, text, blanks, errors, dates) never scan.
39#[derive(Clone, Copy, PartialEq, Eq, Debug)]
40enum StatsNeed {
41    /// Sorted numbers, their mean and deviation (top/bottom N, above average, scales, bars, icons).
42    Numbers,
43    /// How often each displayed text occurs (duplicate / unique values).
44    Counts,
45}
46
47fn stats_need(rule: &CfRule) -> Option<StatsNeed> {
48    match rule {
49        CfRule::Duplicate { .. } | CfRule::Unique { .. } => Some(StatsNeed::Counts),
50        // only two- and three-colour scales paint: no statistics for the others
51        CfRule::ColorScale { stops } if !matches!(stops.len(), 2 | 3) => None,
52        CfRule::Top10 { .. } | CfRule::AboveAverage { .. } | CfRule::ColorScale { .. } | CfRule::DataBar { .. } | CfRule::IconSet { .. } => {
53            Some(StatsNeed::Numbers)
54        }
55        _ => None,
56    }
57}

59impl RuleStats {
60    fn build(sh: &Sheet, ranges: &[RangeRef], need: StatsNeed) -> RuleStats {
61        let (sorted, counts) = numbers_in(sh, ranges, need);
62        let n = sorted.len().max(1) as f64;
63        let avg = sorted.iter().sum::<f64>() / n;
64        let std = (sorted.iter().map(|x| (x - avg).powi(2)).sum::<f64>() / n).sqrt();
65        RuleStats { sorted, avg, std, counts }
66    }
67}

69#[cfg(test)]
70thread_local! {
71    /// Cells `numbers_in` visited on this thread (tests count the work a lookup does).
72    static SCANNED: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
73}
74
75/// Most cells one rule's statistics visit, across all of its ranges together: a rule repeating
76/// the same range many times must not multiply the scan.
77const STATS_SCAN_LIMIT: usize = 2_000_000;
78
79/// Steps through the spilled cells inside `r`, row-major: `Some(cell)` for each, and `None` for a
80/// jump past the spills left or right of the columns on a row (one step per row, however many
81/// there are), so unrelated spills between a narrow range's corners cost little.
82fn spill_steps(sh: &Sheet, r: RangeRef) -> impl Iterator<Item = Option<CellRef>> + '_ {
83    let mut cur = Some(r.start);
84    std::iter::from_fn(move || {
85        let at = cur.filter(|at| *at <= r.end)?;
86        let (&c, _) = sh.spill.range(at..=r.end).next()?;
87        if c.col < r.start.col {
88            cur = Some(CellRef::new(c.row, r.start.col));
89            Some(None)
90        } else if c.col > r.end.col {
91            cur = c.row.checked_add(1).map(|row| CellRef::new(row, r.start.col));
92            Some(None)
93        } else {
94            cur = c.col.checked_add(1).map(|col| CellRef::new(c.row, col));
95            Some(Some(c))
96        }
97    })
98}

100fn numbers_in(sh: &Sheet, ranges: &[RangeRef], need: StatsNeed) -> (Vec<f64>, HashMap<String, usize>) {
101    let mut nums = Vec::new();
102    let mut counts = HashMap::new();
103    let mut budget = STATS_SCAN_LIMIT;
104    let mut seen: HashSet<RangeRef> = HashSet::new();
105    let used = sh.used_range();
106    for r in ranges {
107        if budget == 0 {
108            break;
109        }
110        let r = match used.and_then(|u| r.intersection(&u)) {
111            Some(x) => x,
112            None => continue,
113        };
114        // The same range listed twice counts its cells once.
115        if !seen.insert(r) {
116            continue;
117        }
118        // Only stored and spilled cells hold values: blank cells of a sparse range cost nothing,
119        // so a value far down a huge range is still counted. Every cell looked at is charged.
120        let stored = sh.cells.iter_range(r).map(|(c, _)| (Some(c), true));
121        let spilled = spill_steps(sh, r).map(|c| (c, false));
122        for (c, is_stored) in stored.chain(spilled).take(budget) {
123            budget -= 1;
124            // a skip over spills outside the columns (charged, not read), or a spilled value
125            // under a stored cell, which was read with it
126            let Some(c) = c.filter(|c| is_stored || sh.cells.get(*c).is_none()) else { continue };
127            #[cfg(test)]
128            SCANNED.with(|n| n.set(n.get() + 1));
129            let v = sh.value(c);
130            match need {
131                StatsNeed::Numbers => {
132                    if let Value::Number(n) = v {
133                        nums.push(n);
134                    }
135                }
136                StatsNeed::Counts => {
137                    if !v.is_empty() {
138                        *counts.entry(v.display().to_lowercase()).or_insert(0) += 1;
139                    }
140                }
141            }
142        }
143    }
144    nums.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
145    (nums, counts)
146}

148fn percentile(sorted: &[f64], p: f64) -> f64 {
149    if sorted.is_empty() {
150        return 0.0;
151    }
152    let rank = (p / 100.0).clamp(0.0, 1.0) * (sorted.len() - 1) as f64;
153    let lo = rank.floor() as usize;
154    let hi = rank.ceil() as usize;
155    let a = sorted.get(lo).copied().unwrap_or(0.0);
156    let b = sorted.get(hi).copied().unwrap_or(a);
157    a + (b - a) * (rank - lo as f64)
158}

160fn threshold(wb: &Workbook, sheet: usize, at: CellRef, st: &RuleStats, k: &CfValueKind) -> f64 {
161    let min = st.sorted.first().copied().unwrap_or(0.0);
162    let max = st.sorted.last().copied().unwrap_or(0.0);
163    match k {
164        CfValueKind::Min => min,
165        CfValueKind::Max => max,
166        CfValueKind::Number(n) => *n,
167        CfValueKind::Percent(p) => min + (max - min) * p / 100.0,
168        CfValueKind::Percentile(p) => percentile(&st.sorted, *p),
169        CfValueKind::Formula(f) => crate::calc::evaluate(wb, sheet, at, f).to_number().unwrap_or(0.0),
170    }
171}

173fn lerp(a: [u8; 3], b: [u8; 3], t: f64) -> [u8; 3] {
174    let t = t.clamp(0.0, 1.0);
175    [
176        (a[0] as f64 + (b[0] as f64 - a[0] as f64) * t).round() as u8,
177        (a[1] as f64 + (b[1] as f64 - a[1] as f64) * t).round() as u8,
178        (a[2] as f64 + (b[2] as f64 - a[2] as f64) * t).round() as u8,
179    ]
180}

194fn simple_rule_style<'a>(ctx: &RuleEval<'_>, rule: &'a CfRule) -> Option<&'a Style> {
195    let RuleEval { wb, sheet, cell, first, value, text, stats } = ctx;
196    match rule {
197        CfRule::CellIs { op, a, b, style } => {
198            let eval = |f: &str| crate::calc::evaluate(wb, *sheet, *cell, &relative(f, *first, *cell));
199            let av = eval(a);
200            let matches = match op {
201                CfOperator::Between | CfOperator::NotBetween => {
202                    let bv = b.as_deref().map(eval).unwrap_or_default();
203                    let (lo, hi) = if compare(&av, &bv) == std::cmp::Ordering::Greater { (bv, av) } else { (av, bv) };
204                    let within = compare(value, &lo) != std::cmp::Ordering::Less && compare(value, &hi) != std::cmp::Ordering::Greater;
205                    if *op == CfOperator::Between { within } else { !within }
206                }
207                CfOperator::Equal => compare(value, &av) == std::cmp::Ordering::Equal,
208                CfOperator::NotEqual => compare(value, &av) != std::cmp::Ordering::Equal,
209                CfOperator::Greater => compare(value, &av) == std::cmp::Ordering::Greater,
210                CfOperator::Less => compare(value, &av) == std::cmp::Ordering::Less,
211                CfOperator::GreaterOrEqual => compare(value, &av) != std::cmp::Ordering::Less,
212                CfOperator::LessOrEqual => compare(value, &av) != std::cmp::Ordering::Greater,
213            };
214            (matches && !value.is_error()).then_some(style.as_ref())
215        }
216        CfRule::Expression { formula, style } => {
217            crate::calc::evaluate(wb, *sheet, *cell, &relative(formula, *first, *cell)).to_bool().unwrap_or(false).then_some(style.as_ref())
218        }
219        CfRule::ContainsText { text: needle, style } => {
220            (!value.is_error() && text.to_lowercase().contains(&needle.to_lowercase())).then_some(style.as_ref())
221        }
222        CfRule::NotContainsText { text: needle, style } => {
223            (value.is_error() || !text.to_lowercase().contains(&needle.to_lowercase())).then_some(style.as_ref())
224        }
225        CfRule::BeginsWith { text: needle, style } => {
226            (!value.is_error() && text.to_lowercase().starts_with(&needle.to_lowercase())).then_some(style.as_ref())
227        }
228        CfRule::EndsWith { text: needle, style } => {
229            (!value.is_error() && text.to_lowercase().ends_with(&needle.to_lowercase())).then_some(style.as_ref())
230        }
231        CfRule::Blanks { style } => (value.is_empty() || text.trim().is_empty()).then_some(style.as_ref()),
232        CfRule::NoBlanks { style } => (!value.is_empty() && !text.trim().is_empty()).then_some(style.as_ref()),
233        CfRule::Errors { style } => value.is_error().then_some(style.as_ref()),
234        CfRule::NoErrors { style } => (!value.is_error()).then_some(style.as_ref()),
235        CfRule::Duplicate { style } => {
236            (!value.is_empty() && stats.counts.get(&text.to_lowercase()).copied().unwrap_or(0) > 1).then_some(style.as_ref())
237        }
238        CfRule::Unique { style } => {
239            (!value.is_empty() && stats.counts.get(&text.to_lowercase()).copied().unwrap_or(0) == 1).then_some(style.as_ref())
240        }
241        _ => None,
242    }
243}

245fn time_period_matches(wb: &Workbook, period: &str, serial: f64) -> bool {
246    let today = crate::base::date::serial_in_system(wb.date_system, crate::calc::now_serial()).floor();
247    let day = serial.floor();
248    let date = |serial| crate::base::date::datetime_from_serial(wb.date_system, serial);
249    let week_start = today - date(today).map(|value| value.weekday as f64).unwrap_or(0.0);
250    let (year, month) = date(today).map(|value| (value.year, value.month)).unwrap_or((2026, 1));
251    let year_month = date(day).map(|value| (value.year, value.month));
252    let month_matches = |offset: i32| {
253        let shifted_month = month as i32 + offset;
254        let shifted_year = year + (shifted_month - 1).div_euclid(12);
255        let shifted_month = (shifted_month - 1).rem_euclid(12) + 1;
256        year_month == Some((shifted_year, shifted_month as u32))
257    };
258    match period {
259        "yesterday" => day == today - 1.0,
260        "today" => day == today,
261        "tomorrow" => day == today + 1.0,
262        "last7Days" => day > today - 7.0 && day <= today,
263        "lastWeek" => day >= week_start - 7.0 && day < week_start,
264        "thisWeek" => day >= week_start && day < week_start + 7.0,
265        "nextWeek" => day >= week_start + 7.0 && day < week_start + 14.0,
266        "lastMonth" => month_matches(-1),
267        "thisMonth" => month_matches(0),
268        "nextMonth" => month_matches(1),
269        _ => false,
270    }
271}

278    /// Conditional look of `c`, or `None` when no rule applies.
279    pub fn look(&mut self, wb: &Workbook, sheet: usize, c: CellRef) -> Option<CfLook> {
280        let sh = wb.sheet(sheet)?;
281        if sh.cond_formats.is_empty() {
282            return None;
283        }
284        let mut out = CfLook::default();
285        let mut any = false;
286        let mut rules: Vec<(usize, &CondFormat)> =
287            sh.cond_formats.iter().enumerate().filter(|(_, cf)| cf.ranges.iter().any(|r| r.contains(c))).collect();
288        rules.sort_by_key(|(_, cf)| cf.priority);
289        let v = sh.value(c);
290        let no_stats = RuleStats::default();
291        for (idx, cf) in rules {
292            let first = RangeRef::top_left(&cf.ranges).unwrap_or(c);
293            let st: &RuleStats = match stats_need(&cf.rule) {
294                Some(need) => &*self.stats.entry(idx).or_insert_with(|| RuleStats::build(sh, &cf.ranges, need)),
295                None => &no_stats,
296            };
297            let text = v.display();
298            let context = RuleEval { wb, sheet, cell: c, first, value: &v, text: &text, stats: st };
299            let matched_style = simple_rule_style(&context, &cf.rule).or_else(|| match &cf.rule {
300                CfRule::Top10 { bottom, percent, rank, style } => match v.as_f64() {
301                    Some(n) if !st.sorted.is_empty() => {
302                        let k = if *percent { ((st.sorted.len() as f64) * (*rank as f64) / 100.0).floor().max(1.0) as usize } else { *rank as usize };
303                        let k = k.clamp(1, st.sorted.len());
304                        let ok = if *bottom {
305                            n <= st.sorted.get(k - 1).copied().unwrap_or(n)
306                        } else {
307                            n >= st.sorted.get(st.sorted.len() - k).copied().unwrap_or(n)
308                        };
309                        ok.then_some(style.as_ref())
310                    }
311                    _ => None,
312                },
313                CfRule::AboveAverage { below, equal, std_dev, style } => match v.as_f64() {
314                    Some(n) => {
315                        let thr = st.avg + if *below { -1.0 } else { 1.0 } * st.std * *std_dev as f64;
316                        let ok = if *below { n < thr || (*equal && n == thr) } else { n > thr || (*equal && n == thr) };
317                        ok.then_some(style.as_ref())
318                    }
319                    None => None,
320                },
321                CfRule::TimePeriod { period, style } => match v.as_f64() {
322                    Some(n) if time_period_matches(wb, period, n) => Some(style.as_ref()),
323                    _ => None,
324                },
325                CfRule::ColorScale { stops } => {
326                    // Only two- and three-colour scales paint: evaluate no thresholds for others.
327                    if let Some(n) = v.as_f64().filter(|_| matches!(stops.len(), 2 | 3)) {
328                        let pts: Vec<(f64, [u8; 3])> = stops
329                            .iter()
330                            .map(|(k, col)| (threshold(wb, sheet, c, st, k), col.resolve(&wb.theme).unwrap_or([255, 255, 255])))
331                            .collect();
332                        let rgb = match pts.as_slice() {
333                            [(a, ca), (b, cb)] => lerp(*ca, *cb, if b > a { (n - a) / (b - a) } else { 0.0 }),
334                            [(a, ca), (m, cm), (b, cb)] => {
335                                if n <= *m {
336                                    lerp(*ca, *cm, if m > a { (n - a) / (m - a) } else { 0.0 })
337                                } else {
338                                    lerp(*cm, *cb, if b > m { (n - m) / (b - m) } else { 1.0 })
339                                }
340                            }
341                            _ => [255, 255, 255],
342                        };
343                        if out.scale_fill.is_none() {
344                            out.scale_fill = Some(rgb);
345                            any = true;
346                        }
347                    }
348                    None
349                }
350                CfRule::DataBar { min, max, color, gradient, show_value } => {
351                    if let Some(n) = v.as_f64() {
352                        let lo = threshold(wb, sheet, c, st, min);
353                        let hi = threshold(wb, sheet, c, st, max);
354                        let lo = if lo > 0.0 { 0.0 } else { lo };
355                        let frac = if hi > lo { ((n - lo) / (hi - lo)).clamp(0.0, 1.0) } else { 1.0 };
356                        if out.bar.is_none() {
357                            out.bar = Some((
358                                (frac as f32) * 0.9 + 0.1 * (frac > 0.0) as u8 as f32,
359                                color.resolve(&wb.theme).unwrap_or([0x63, 0x8E, 0xC6]),
360                                *gradient,
361                            ));
362                            out.hide_value |= !show_value;
363                            any = true;
364                        }
365                    }
366                    None
367                }
368                CfRule::IconSet { set, thresholds, reverse, show_value } => {
369                    if let Some(n) = v.as_f64() {
370                        let k = icon_count(set);
371                        // One threshold per icon, the first being the lower bound (as in the file);
372                        // also accept the lower bound left out, else Excel's defaults.
373                        let defaults;
374                        let lower = match thresholds.len() {
375                            n if n == k => thresholds.get(1..).unwrap_or_default(),
376                            n if n + 1 == k => thresholds.as_slice(),
377                            _ => {
378                                defaults = default_icon_thresholds(set);
379                                defaults.get(1..).unwrap_or_default()
380                            }
381                        };
382                        let cuts: Vec<f64> = lower.iter().map(|t| threshold(wb, sheet, c, st, t)).collect();
383                        let mut idx = cuts.iter().filter(|cut| n >= **cut).count();
384                        if *reverse {
385                            idx = k - 1 - idx.min(k - 1);
386                        }
387                        if out.icon.is_none() {
388                            out.icon = Some((set.clone(), idx.min(k - 1)));
389                            out.hide_value |= !show_value;
390                            any = true;
391                        }
392                    }
393                    None
394                }
395                _ => None,
396            });
397            if let Some(st) = matched_style {
398                any = true;
399                // Every matching rule applies; where two set the same part, the rule with the
400                // higher priority (seen first) wins.
401                match &mut out.style {
402                    None => out.style = Some(st.clone()),
403                    Some(prev) => layer_under(prev, st),
404                }
405                if cf.stop_if_true {
406                    break;
407                }
408            }
409        }
410        any.then_some(out)
411    }
412}

414/// Adds the parts `lower` sets that `style` leaves unset (differential styles: default parts
415/// are unset).
416fn layer_under(style: &mut Style, lower: &Style) {
417    if style.align.reading_order.is_none() {
418        style.align.reading_order = lower.align.reading_order;
419    }
420    if style.fill.pattern == PatternType::None {
421        style.fill = lower.fill;
422    }
423    if style.font.color == Color::Auto {
424        style.font.color = lower.font.color;
425    }
426    style.font.bold |= lower.font.bold;
427    style.font.italic |= lower.font.italic;
428    style.font.strike |= lower.font.strike;
429    if style.font.underline == Underline::None {
430        style.font.underline = lower.font.underline;
431    }
432    let b = &mut style.borders;
433    for (dst, src) in [
434        (&mut b.left, lower.borders.left),
435        (&mut b.right, lower.borders.right),
436        (&mut b.top, lower.borders.top),
437        (&mut b.bottom, lower.borders.bottom),
438        (&mut b.diag_down, lower.borders.diag_down),
439        (&mut b.diag_up, lower.borders.diag_up),
440    ] {
441        if dst.is_none() {
442            *dst = src;
443        }
444    }
445    if style.num_fmt.as_str() == "General" {
446        style.num_fmt = lower.num_fmt.clone();
447    }
448}

450/// Shifts a rule formula written for the top-left cell of the range to `c`.
451fn relative(f: &str, first: CellRef, c: CellRef) -> String {
452    if first == c {
453        return f.to_string();
454    }
455    match crate::formula::parse(f.trim_start_matches('=')) {
456        Ok(e) => crate::formula::print(&crate::formula::adjust::shift_relative(e, c.row as i64 - first.row as i64, c.col as i64 - first.col as i64)),
457        Err(_) => f.to_string(),
458    }
459}

533    /// The statistics as every rule used to get them: one pass collecting both parts.
534    fn all_stats_before(sh: &Sheet, ranges: &[RangeRef]) -> RuleStats {
535        let mut nums = Vec::new();
536        let mut counts = HashMap::new();
537        for r in ranges {
538            let Some(r) = sh.used_range().and_then(|u| r.intersection(&u)) else { continue };
539            for c in r.iter() {
540                let v = sh.value(c);
541                if let Value::Number(n) = v {
542                    nums.push(n);
543                }
544                if !v.is_empty() {
545                    *counts.entry(v.display().to_lowercase()).or_insert(0) += 1;
546                }
547            }
548        }
549        nums.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
550        let n = nums.len().max(1) as f64;
551        let avg = nums.iter().sum::<f64>() / n;
552        let std = (nums.iter().map(|x| (x - avg).powi(2)).sum::<f64>() / n).sqrt();
553        RuleStats { sorted: nums, avg, std, counts }
554    }
