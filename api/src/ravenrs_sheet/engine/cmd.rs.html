42pub struct CommandSpec {
43    pub id: &'static str,

72impl CommandInfo {
73    /// Format the portable shortcut notation for the runtime operating system.
74    pub fn shortcut_for_platform(&self, mac: bool) -> Option<String> {
75        let keys = if mac {
76            self.shortcut_mac.or(self.shortcut)
77        } else if self.shortcut_mac.is_some() && self.shortcut_mac == self.shortcut {
78            // An identical Mac override marks a Mac-only display shortcut.
79            // The portable field remains unchanged for legacy CLI consumers.
80            None
81        } else {
82            self.shortcut
83        }?;
84        if mac {
85            return Some(keys.to_owned());
86        }
87        Some(keys.split('+').map(|part| if part == "Cmd" { "Ctrl" } else { part }).collect::<Vec<_>>().join("+"))
88    }
89}

107/// Params that hold a secret: `password`, or a name ending in `Password`.
108pub fn is_secret_param(key: &str) -> bool {
109    key == "password" || key.ends_with("Password")
110}
111
112/// The params as the journal records them: secret params are left out, so a password never
113/// reaches the journal dialog, `engine.journal` or a recorded script (saved in the workbook).
114/// Replaying such an entry runs the command without that param.
115pub fn journal_params(params: &Json) -> Json {
116    match params {
117        Json::Object(o) => Json::Object(o.iter().filter(|(k, _)| !is_secret_param(k)).map(|(k, v)| (k.clone(), v.clone())).collect()),
118        other => other.clone(),
119    }
120}

194pub fn find_command(id: &str) -> Option<&'static CommandSpec> {
195    if let Some(spec) = command_specs().iter().find(|c| c.id == id) {
196        return Some(spec);
197    }
198    // Commands registered by UI integration tests (pending keytips go live
199    // the moment their command lands, with no table edit).
200    test_commands().iter().find(|c| c.id == id).copied()

270/// `"moveOnly"`/`"oneCell"`, `"absolute"`/`"don't move or size"` (case-insensitive).

275pub(crate) fn parse_anchor_mode(s: &str) -> Option<crate::model::AnchorMode> {
276    match s.to_ascii_lowercase().replace([' ', '-', '_', '\''], "").as_str() {
277        "moveandsize" | "moveandsizeandcells" | "twocell" => Some(crate::model::AnchorMode::MoveAndSize),
278        "moveonly" | "movebutdontsize" | "movebutdontsizewithcells" | "onecell" => Some(crate::model::AnchorMode::MoveOnly),
279        "absolute" | "dontmoveorsize" | "dontmoveorsizewithcells" | "none" => Some(crate::model::AnchorMode::Absolute),
280        _ => None,
281    }
282}

296/// A range parameter (`"A1:B2"`, `"Sheet2!A1"`, `"B:B"`), defaulting to the current selection's
297/// areas.
298pub(crate) fn target_ranges(s: &Session, p: &Json) -> Result<Vec<RangeRef>> {
299    if let Some(r) = str_param(p, "range").or_else(|| str_param(p, "cell")) {
300        let (_, body) = split_sheet(r);
301        let mut out = Vec::new();
302        for part in body.split(',') {
303            out.push(RangeRef::parse(part).ok_or_else(|| bad("range", format!("not a range: `{part}`")))?);
304        }
305        return Ok(out);
306    }
307    Ok(s.doc()?.selection.ranges.clone())
308}

319/// The sheet a call names (`sheet` param or a `Sheet!` prefix on `range`), else the active one.
320pub(crate) fn target_sheet(s: &Session, p: &Json) -> Result<usize> {
321    let d = s.doc()?;
322    if let Some(v) = p.get("sheet") {
323        if let Some(i) = v.as_u64() {
324            return if (i as usize) < d.wb.sheets.len() { Ok(i as usize) } else { Err(bad("sheet", "no such sheet")) };
325        }
326        if let Some(n) = v.as_str() {
327            return d.wb.sheet_index(n).ok_or_else(|| bad("sheet", format!("no sheet named `{n}`")));
328        }
329    }
330    if let Some(r) = str_param(p, "range").or_else(|| str_param(p, "cell"))
331        && let (Some(sh), _) = split_sheet(r)
332    {
333        return d.wb.sheet_index(&sh).ok_or_else(|| bad("range", format!("no sheet named `{sh}`")));
334    }
335    Ok(d.wb.active_sheet())
336}

340/// A single-cell parameter. Absent or null is `Ok(None)` (callers default it); a value that is not

347/// Like [`cell_param`], for a command that operates on `sheet` (a `Sheet!` prefix must name it).
348pub(crate) fn cell_param_on(s: &Session, p: &Json, key: &str, sheet: usize) -> Result<Option<CellRef>> {
349    let t = match p.get(key) {
350        None | Some(Json::Null) => return Ok(None),
351        Some(Json::String(t)) => t,
352        Some(v) => return Err(bad("cell", format!("invalid cell reference `{v}` for `{key}`"))),
353    };
354    let (sh, body) = split_sheet(t);
355    let c = CellRef::parse(body).ok_or_else(|| bad("cell", format!("invalid cell reference `{t}` for `{key}`")))?;
356    if let Some(name) = sh
357        && s.doc()?.wb.sheet_index(&name) != Some(sheet)
358    {
359        return Err(bad("cell", format!("`{t}` for `{key}` names a sheet other than the one this command works on")));
360    }
361    Ok(Some(c))
362}

364/// Applies an edit to a copy of the active workbook, then recalculates and commits it.
365/// `changed` collects cells whose content changed; set `structural` for row/column/sheet edits
366/// (full recalc).
367pub struct Ctx<'a> {
368    pub wb: Workbook,
369    pub changed: Vec<Key>,
370    pub structural: bool,
371    /// Recalculate every formula without the other effects of `structural` (table headers are not
372    /// resynchronized, array formulas are still checked): for edits that change only what a
373    /// function reads, like sheet protection.
374    pub recalc_all: bool,
375    pub sel: &'a mut Selection,
376    /// Rows whose automatic height should be recomputed (sheet, row).
377    pub fit_rows: Vec<(usize, u32)>,
378    /// The command applied sheet protection itself (row/column inserts and deletes the
379    /// protection allows), so cells it shifts aren't refused as edits of locked cells.
380    pub protection_checked: bool,
381    /// The structural edits the command made, in order (the sheet's name, the edit), as
382    /// references were rewritten for them.
383    pub edits: Vec<(String, crate::formula::adjust::Edit)>,
384}

386pub(crate) fn edit<R>(s: &mut Session, f: impl FnOnce(&mut Ctx) -> Result<R>) -> Result<R> {
387    let d = s.doc_mut()?;
388    commit(d, |cx| {
389        let r = f(cx)?;
390        crate::engine::tables::sync_headers(cx);
391        Ok(r)
392    })
393}

399pub(crate) fn commit<R>(d: &mut DocState, f: impl FnOnce(&mut Ctx) -> Result<R>) -> Result<R> {
400    // `GRIDCRAFT_PROFILE` timing, never on wasm where the clock traps. `elapsed` on the epoch
401    // reads the wall clock without the (disallowed) `now`.
402    let t0 = if cfg!(target_arch = "wasm32") { None } else { std::env::var_os("GRIDCRAFT_PROFILE").map(|_| wall_clock()) };
403    let since = |t0: std::time::Duration| wall_clock().saturating_sub(t0);
404    let mut sel = d.selection.clone();
405    let mut ctx = Ctx {
406        wb: (*d.wb).clone(),
407        changed: Vec::new(),
408        structural: false,
409        recalc_all: false,
410        sel: &mut sel,
411        fit_rows: Vec::new(),
412        protection_checked: false,
413        edits: Vec::new(),
414    };
415    let r = f(&mut ctx)?;
416    let Ctx { mut wb, mut changed, structural, recalc_all, mut fit_rows, protection_checked, edits, .. } = ctx;
417    // Equations whose source changed (or that arrived without a layout, as from a paste) are
418    // laid out here, so painting only ever reads the cache.
419    wb.refresh_equations();
420    if !protection_checked {
421        check_protection(&d.wb, &wb, &d.unlocked_ranges)?;
422    }
423    if !structural {
424        check_arrays(&d.wb, &wb)?;
425    }
426    if wb.calc.precision_as_displayed {
427        round_edited_as_displayed(&d.wb, &mut wb, &mut changed, structural);
428    }
429    if let Some(log) = d.touched.as_mut() {
430        for (si, c) in changed.iter().take(TOUCH_LOG_LIMIT.saturating_sub(log.len())) {
431            if let Some(sh) = wb.sheet(*si) {
432                log.push((sh.name.clone(), *c));
433            }
434        }
435    }
436    if let Some(log) = d.edits.as_mut() {
437        log.extend(edits);
438    }
439    if changed.len() <= 200_000 {
440        fit_rows.extend(changed.iter().map(|(s, c)| (*s, c.row)));
441    }
442    if let Some(t0) = t0 {
443        eprintln!("commit: edit {:?}, {} changed", since(t0), changed.len());
444    }
445    if structural || recalc_all {
446        d.calc.recalc_all(&mut wb);
447    } else if !changed.is_empty() {
448        d.calc.cells_changed(&mut wb, &changed);
449    }
450    controls::sync_linked_values(&mut wb);
451    if let Some(t0) = t0 {
452        eprintln!("commit: + recalc {:?}", since(t0));
453    }
454    fit_rows.sort_unstable();
455    fit_rows.dedup();
456    let mut sweep: Option<(usize, MergeSweep)> = None;
457    let mut left = MERGE_FIT_WORK;
458    for (si, row) in fit_rows {
459        if sweep.as_ref().is_none_or(|(at, _)| *at != si) {
460            if let Some((_, old)) = &sweep {
461                left = old.left();
462            }
463            sweep = Some((si, MergeSweep::new(left)));
464        }
465        if let Some((_, sw)) = sweep.as_mut() {
466            fit_row_height(&mut wb, si, row, sw);
467        }
468    }
469    if let Some(t0) = t0 {
470        eprintln!("commit: + rows {:?}", since(t0));
471    }
472    d.wb = std::sync::Arc::new(wb);
473    d.selection = sel;
474    Ok(r)
475}

477/// "Set precision as displayed": a number typed, pasted or reformatted is stored as its format
478/// shows it (Excel rounds a constant when it is entered and when its cell's format changes,
479/// observed through COM). Only cells the edit touched are visited, so this costs nothing for
480/// the rest of the workbook; formula cells whose format changed are recalculated, and the
481/// recalculation rounds them. A structural edit recalculates everything anyway, so it only
482/// rounds its constants (a sheet the edit created, for one, holds written numbers).
483fn round_edited_as_displayed(old: &Workbook, wb: &mut Workbook, changed: &mut Vec<crate::calc::Key>, structural: bool) {
484    let mut prec = crate::calc::DisplayedPrecision::new();
485    for si in 0..wb.sheets.len() {
486        let Some(after) = wb.sheets.get(si) else { continue };
487        // Sheets are matched by name. A sheet the edit created has no earlier version, so all of
488        // its cells are new.
489        let before = old.sheet_index(&after.name).and_then(|i| old.sheets.get(i));
490        if before.is_some_and(|b| std::sync::Arc::ptr_eq(b, after)) {
491            continue;
492        }
493        let diffs: Vec<CellRef> = match before {
494            Some(before) => before.cells.diff(&after.cells),
495            None => after.cells.iter().map(|(c, _)| c).collect(),
496        };
497        let mut anchors = std::collections::HashSet::new();
498        for &c in &diffs {
499            if !structural && after.cell(c).is_some_and(|cell| cell.formula.is_some()) {
500                changed.push((si, c));
501                anchors.insert(c);
502            }
503        }
504        let updates = displayed_updates(&mut prec, wb, si, diffs.iter().copied());
505        // A spill destination is no stored cell: reformatting it (or its row or column) leaves
506        // no change in the store, so its anchor is recalculated to round the new element.
507        if let (Some(before), false) = (before, structural) {
508            for (&anchor, range) in &after.spill_ranges {
509                if !anchors.contains(&anchor)
510                    && (diffs.iter().any(|d| range.contains(*d))
511                        || line_styles_differ(&before.rows, &after.rows, range.start.row, range.end.row)
512                        || line_styles_differ(&before.cols, &after.cols, range.start.col, range.end.col))
513                {
514                    changed.push((si, anchor));
515                }
516            }
517        }
518        if updates.is_empty() {
519            continue;
520        }
521        let Some(sh) = wb.sheet_mut(si) else { continue };
522        for (c, r) in updates {
523            if let Some(cell) = sh.cells.get_mut(c) {
524                cell.value = crate::base::Value::Number(r);
525                if !structural {
526                    changed.push((si, c));
527                }
528            }
529        }
530    }
531}

533/// Whether any row or column style in `lo..=hi` differs between two sheets' line maps.
534fn line_styles_differ(
535    before: &std::collections::BTreeMap<u32, crate::model::LineInfo>,
536    after: &std::collections::BTreeMap<u32, crate::model::LineInfo>,
537    lo: u32,
538    hi: u32,
539) -> bool {
540    let style = |m: &std::collections::BTreeMap<u32, crate::model::LineInfo>, k: u32| m.get(&k).and_then(|l| l.style);
541    after.range(lo..=hi).any(|(k, l)| l.style != style(before, *k)) || before.range(lo..=hi).any(|(k, l)| l.style != style(after, *k))
542}

544/// Turning "Set precision as displayed" on rounds every number constant in the workbook to its
545/// format, permanently (Excel warns "Data will permanently lose accuracy"). Formulas are rounded
546/// by the recalculation that follows.
547pub(crate) fn round_all_constants_as_displayed(wb: &mut Workbook) {
548    let mut prec = crate::calc::DisplayedPrecision::new();
549    for si in 0..wb.sheets.len() {
550        let Some(sh) = wb.sheets.get(si) else { continue };
551        let updates = displayed_updates(&mut prec, wb, si, sh.cells.iter().map(|(c, _)| c));
552        if updates.is_empty() {
553            continue;
554        }
555        let Some(sh) = wb.sheet_mut(si) else { continue };
556        for (c, r) in updates {
557            if let Some(cell) = sh.cells.get_mut(c) {
558                cell.value = crate::base::Value::Number(r);
559            }
560        }
561    }
562}

564/// The constants among `cells` on sheet `si` whose value the option changes, with their rounded
565/// values. A locked constant on a protected sheet keeps its value: protection keeps its content,
566/// and rounding it by its format would change the content that the sheet protects.
567fn displayed_updates(
568    prec: &mut crate::calc::DisplayedPrecision,
569    wb: &Workbook,
570    si: usize,
571    cells: impl Iterator<Item = CellRef>,
572) -> Vec<(CellRef, f64)> {
573    let Some(sh) = wb.sheet(si) else { return Vec::new() };
574    cells
575        .filter(|&c| !(sh.is_protected() && wb.styles.get(sh.style_id(c)).protection.locked))
576        .filter_map(|c| {
577            let cell = sh.cell(c)?;
578            match cell.value {
579                crate::base::Value::Number(n) if cell.formula.is_none() => {
580                    let r = prec.round(wb, si, c, n);
581                    (r.to_bits() != n.to_bits()).then_some((c, r))
582                }
583                _ => None,
584            }
585        })
586        .collect()
587}

592/// Excel's message for an edit of a locked cell on a protected sheet.
593pub(crate) const PROTECTED: &str = "The cell or chart you're trying to change is on a protected sheet.";
594
595/// Refuses an edit that changed what a locked cell on a protected sheet holds (its value or
596/// formula), or that unlocked it, whichever command made it. Sheets are matched by name, so
597/// moving, renaming or deleting a sheet isn't an edit of its cells. A sort or duplicate removal
598/// that moves an unlocked cell with equal content onto a locked one changes no content but would
599/// leave the cell editable, so the effective lock (row and column formats included) is compared
600/// as well.
601fn check_protection(old: &Workbook, new: &Workbook, unlocked: &[(String, String)]) -> Result<()> {
602    for sh in &old.sheets {
603        if !sh.is_protected() {
604            continue;
605        }
606        let Some(after) = new.sheet_index(&sh.name).and_then(|i| new.sheets.get(i)) else { continue };
607        if std::sync::Arc::ptr_eq(sh, after) || !after.is_protected() {
608            continue;
609        }
610        for c in sh.cells.diff(&after.cells) {
611            let unlocked_now = !new.styles.get(after.style_id(c)).protection.locked;
612            if (unlocked_now || !same_content(sh.cell(c), after.cell(c)))
613                && old.styles.get(sh.style_id(c)).protection.locked
614                && let Some(m) = locked_cell_error(sh, c, unlocked)
615            {
616                return Err(EngineError::Other(m));
617            }
618        }
619    }
620    Ok(())
621}

623/// Refuses an action a protected sheet does not allow, by the tick box for it in Review ▸ Protect
624/// Sheet. `what` names the action in the message. An unprotected sheet allows everything.
625///
626/// Cell edits are guarded separately, by whether the cell is locked (`locked_cell_error`): these
627/// permissions are the ones that are not about one cell's contents.
628pub(crate) fn refuse_unless_allowed(s: &Session, sheet: usize, what: &str, pick: fn(&crate::model::SheetProtection) -> bool) -> Result<()> {
629    if s.doc()?.wb.sheet(sheet).is_some_and(|sh| sh.protection.as_ref().is_some_and(|pr| !pick(pr))) {
630        return Err(EngineError::Other(format!("{what} isn't allowed on this protected sheet (Review › Unprotect Sheet).")));
631    }
632    Ok(())
633}

635/// Whether a locked cell on a protected sheet stays locked: `None` when it's in an Allow Edit
636/// Range without a password or one unlocked in this session, else the message to show.
637pub(crate) fn locked_cell_error(sh: &crate::model::Sheet, c: CellRef, unlocked: &[(String, String)]) -> Option<String> {
638    let mut needs_password = None;
639    for r in sh.edit_ranges.iter().filter(|r| r.contains(c)) {
640        // A range limited to Windows users or groups (a security descriptor) can't be checked
641        // here, so without a password it stays locked rather than opening to every editor.
642        if (r.password.is_none() && r.security_descriptor.is_none()) || unlocked.iter().any(|(s, t)| *s == sh.name && *t == r.title) {
643            return None;
644        }
645        if r.password.is_some() {
646            needs_password.get_or_insert(r.title.as_str());
647        }
648    }
649    Some(match needs_password {
650        Some(t) => format!("The range \"{t}\" needs its password before it can be changed (Review › Allow Edit Ranges)."),
651        None => PROTECTED.into(),
652    })
653}

655/// Refuses an edit that put something into a cell of a multi-cell legacy array other than its
656/// formula cell, while the array stays (Excel: "You can't change part of an array."). The array
657/// would show its value there while the file kept the new content. Cells shifted by row and
658/// column edits aren't edits of their content, so structural commits aren't checked.
659fn check_arrays(old: &Workbook, new: &Workbook) -> Result<()> {
660    for sh in &old.sheets {
661        if sh.spill.is_empty() {
662            continue;
663        }
664        let Some(after) = new.sheet_index(&sh.name).and_then(|i| new.sheets.get(i)) else { continue };
665        if std::sync::Arc::ptr_eq(sh, after) {
666            continue;
667        }
668        for c in sh.cells.diff(&after.cells) {
669            let filled = after.cell(c).is_some_and(crate::model::Cell::has_content);
670            if !filled || !sh.spill.contains_key(&c) || same_content(sh.cell(c), after.cell(c)) {
671                continue;
672            }
673            let in_array = sh.spill_ranges.keys().any(|&anchor| {
674                anchor != c
675                    && after.cell(anchor).and_then(|x| x.formula.as_ref()).and_then(|f| f.array).is_some_and(|a| a.start == anchor && a.contains(c))
676            });
677            if in_array {
678                return Err(EngineError::Other("You can't change part of an array.".into()));
679            }
680        }
681    }
682    Ok(())
683}

685/// Two cells hold the same value, formula or Python code (formats aside; a Python cell's result
686/// is not something the person edits).
687fn same_content(a: Option<&crate::model::Cell>, b: Option<&crate::model::Cell>) -> bool {
688    let empty = |c: Option<&crate::model::Cell>| c.is_none_or(|c| !c.has_content());
689    match (a, b) {
690        (Some(a), Some(b)) => match (&a.formula, &b.formula) {
691            (Some(x), Some(y)) => x.text == y.text && x.array == y.array,
692            (None, None) => match (&a.py, &b.py) {
693                (Some(x), Some(y)) => x.code == y.code && x.object == y.object,
694                (None, None) => a.value == b.value,
695                _ => false,
696            },
697            _ => false,
698        },
699        _ => empty(a) && empty(b),
700    }
701}

703/// Lines a single text line needs when word-wrapped into `width` points. The grid wraps whole
704/// words (egui's LayoutJob), so dividing the total width by the column width undercounts: each
705/// drawn line ends early at the last word that fits. A word wider than the column breaks
706/// mid-word, the way egui does. (#1969)
707fn wrapped_line_count(line: &str, width: f32, size: f32, bold: bool) -> f32 {
708    let word_w = |w: &str| crate::engine::cmd::format::approx_text_width(w, size, bold);
709    let space_w = word_w(" ");
710    let mut lines = 1.0f32;
711    let mut used = 0.0f32;
712    for word in line.split_whitespace() {
713        let w = word_w(word);
714        if w > width {
715            // wider than the column: egui fills the current line, then breaks it across lines
716            if used > 0.0 {
717                lines += 1.0;
718            }
719            let chunks = (w / width).ceil();
720            lines += chunks - 1.0;
721            used = w - (chunks - 1.0) * width;
722            continue;
723        }
724        let gap = if used > 0.0 { space_w } else { 0.0 };
725        if used + gap + w > width {
726            lines += 1.0;
727            used = w;
728        } else {
729            used += gap + w;
730        }
731    }
732    lines
733}

735/// Recomputes an automatic row height from its content: the largest font, and wrapped text.
736/// Rows with a user-set height keep it.
737///
738/// Line breaks in a cell's text count only when the cell wraps, as in Excel: with Wrap Text off
739/// a multi-line value shows on one line and AutoFit gives a one-line row.
740pub fn auto_row_height(wb: &mut Workbook, si: usize, row: u32) {
741    let mut sweep = MergeSweep::new(MERGE_FIT_WORK);
742    fit_row_height(wb, si, row, &mut sweep);
743}

745/// Most merge comparisons one auto-fit pass (a commit, or opening a workbook) makes. Past it the
746/// remaining rows keep the heights they have rather than stall on a sheet whose merges overlap
747/// the fitted rows many times over.
748pub(crate) const MERGE_FIT_WORK: usize = 50_000_000;
749
750/// The merged areas of one sheet that touch a row, for rows visited in ascending order: the
751/// merges are sorted by first row once and swept, so fitting many rows never rescans every merge
752/// per row. All comparisons count against [`MERGE_FIT_WORK`].
753pub(crate) struct MergeSweep {
754    /// Built on the first row that needs merge checks, so rows that return early never pay for it.
755    by_start: Option<Vec<crate::base::RangeRef>>,
756    next: usize,
757    active: Vec<crate::base::RangeRef>,
758    budget: usize,
759    /// The first row was answered by a plain scan: a pass that fits one row never pays for a sort.
760    scanned: bool,
761}
762
763impl MergeSweep {
764    /// A sweep for one sheet's merges with `budget` comparisons left for the pass (carry
765    /// [`MergeSweep::left`] of the previous sheet's sweep into the next). Every call to
766    /// [`MergeSweep::at`] must pass that same sheet.
767    pub(crate) fn new(budget: usize) -> Self {
768        Self { by_start: None, next: 0, active: Vec::new(), budget, scanned: false }
769    }
770
771    /// Comparisons left for the rest of the pass.
772    pub(crate) fn left(&self) -> usize {
773        self.budget
774    }
775
776    /// Takes `n` comparisons from the budget; `false` once it is spent.
777    fn charge(&mut self, n: usize) -> bool {
778        match self.budget.checked_sub(n) {
779            Some(rest) => {
780                self.budget = rest;
781                true
782            }
783            None => {
784                self.budget = 0;
785                false
786            }
787        }
788    }
789
790    /// The merges covering `row`, or `None` when the budget is spent. Rows must not decrease
791    /// between calls (skipping rows is fine).
792    fn at(&mut self, sh: &crate::model::Sheet, row: u32) -> Option<Vec<crate::base::RangeRef>> {
793        if self.by_start.is_none() && !self.scanned {
794            self.scanned = true;
795            if !self.charge(sh.merges.len()) {
796                return None;
797            }
798            return Some(sh.merges.iter().copied().filter(|m| !m.is_single() && m.start.row <= row && row <= m.end.row).collect());
799        }
800        if self.by_start.is_none() {
801            // Sorting is about n log2 n comparisons, charged before it starts.
802            let n = sh.merges.len();
803            if !self.charge(n.saturating_mul((usize::BITS - n.leading_zeros()) as usize)) {
804                return None;
805            }
806            let mut v: Vec<crate::base::RangeRef> = sh.merges.iter().copied().filter(|m| !m.is_single()).collect();
807            v.sort_unstable_by_key(|m| m.start.row);
808            self.by_start = Some(v);
809        }
810        let by_start = self.by_start.as_ref()?;
811        while let Some(m) = by_start.get(self.next).filter(|m| m.start.row <= row) {
812            self.active.push(*m);
813            self.next += 1;
814        }
815        if !self.charge(self.active.len()) {
816            return None;
817        }
818        self.active.retain(|m| m.end.row >= row);
819        Some(self.active.clone())
820    }
821}

823/// [`auto_row_height`] with the sheet's merges taken from `sweep`.
824pub(crate) fn fit_row_height(wb: &mut Workbook, si: usize, row: u32, sweep: &mut MergeSweep) {
825    // Out of budget: the row keeps the height it has.
826    if sweep.left() == 0 {
827        return;
828    }
829    let Some(sh) = wb.sheet(si) else { return };
830    if sh.rows.get(&row).is_some_and(|i| i.custom) {
831        return;
832    }
833    let default = sh.default_row_height;
834    // Fast path: rows of default-size, unwrapped content keep the default height.
835    let plain = sh.cells.row(row, 0, crate::base::MAX_COLS - 1).all(|(_, cell)| {
836        let st = wb.styles.get(cell.style);
837        st.font.size <= crate::model::DEFAULT_FONT_SIZE && !st.align.wrap
838    });
839    if plain {
840        if sh.rows.get(&row).is_some_and(|i| i.size.is_some())
841            && let Some(shm) = wb.sheet_mut(si)
842            && let Some(e) = shm.rows.get_mut(&row)
843        {
844            {
845                e.size = None;
846                if *e == crate::model::LineInfo::default() {
847                    shm.rows.remove(&row);
848                }
849            }
850        }
851        return;
852    }
853    let mut need: f32 = default;
854    // The cell loop must not rescan every merge per cell: the merges touching this row are
855    // picked out once, and per cell the candidates are re-checked (normally none or a couple;
856    // a sheet with many merges and many saved columns would otherwise scan all of them per
857    // cell, rows × merges × columns).
858    // Rows are fitted in ascending order, so the merges touching this one come from a sweep over
859    // the sheet's merges sorted once, not from a scan of all of them. Out of budget: the row keeps
860    // the height it has.
861    let Some(row_merges) = sweep.at(sh, row) else { return };
862    for (c, cell) in sh.cells.row(row, 0, crate::base::MAX_COLS - 1) {
863        if !sweep.charge(row_merges.len()) {
864            return;
865        }
866        // Merged cells never auto-fit their rows, like Excel (column autofit skips them too).
867        if row_merges.iter().any(|m| m.contains(CellRef::new(row, c))) {
868            continue;
869        }
870        let st = wb.styles.get(cell.style);
871        let size = st.font.size;
872        // Scaled from an 11pt row's height, not the sheet's default: a sheet whose default
873        // already fits a bigger font (21pt rows for 16pt text) would otherwise scale it twice.
874        let line = (size * 1.8).round().max(crate::model::DEFAULT_ROW_HEIGHT * size / crate::model::DEFAULT_FONT_SIZE);
875        let mut lines = 1.0f32;
876        if !cell.value.is_empty() || cell.formula.is_some() {
877            // Only a wrapping cell is laid out over several lines: each explicit line break
878            // starts one, and each line folds at the column width.
879            if st.align.wrap {
880                let text = crate::engine::display::cell_text(wb, sh, CellRef::new(row, c));
881                let w = sh.col_width(c).max(1.0) - 6.0;
882                lines = text.lines().map(|l| wrapped_line_count(l, w.max(1.0), size, st.font.bold)).sum::<f32>().max(1.0);
883            }
884        } else if st.font.size <= crate::model::DEFAULT_FONT_SIZE {
885            continue;
886        }
887        let h = if lines > 1.0 { (size * 96.0 / 72.0 * 1.22 * lines + 5.0).ceil() } else { line };
888        need = need.max(h.min(546.0));
889    }
890    let Some(shm) = wb.sheet_mut(si) else { return };
891    let e = shm.rows.entry(row).or_default();
892    e.size = if (need - default).abs() < 0.5 { None } else { Some(need) };
893    if *e == crate::model::LineInfo::default() {
894        shm.rows.remove(&row);
895    }
896}
