14/// Show Details: the source list and the indices of its records behind the value cell at
15/// (`row`, `col`) of the report grid (relative to its top-left), in source order. `None` when
16/// that cell holds no value (a label, a header, a blank). An error, for any cell, when the report
17/// on the sheet is no longer what the source computes: the cell's records would be another cell's.
18pub fn detail_records(wb: &Workbook, pt: &PivotTable, pivot_sheet: usize, row: u32, col: u32) -> Result<Option<(Source, Vec<usize>)>, String> {
19    let src = read_source(wb, &pt.source, pivot_sheet)?;
20    let mut probe = Probe { at: Some((row as usize, col as usize)), hit: None, paths: Some(Vec::new()), cells: None };
21    let out = compute_from(wb, pt, &src, Some(&mut probe))?;
22    // Checked first: in a stale report even a cell that computes as a label may show a value.
23    check_current(wb, pt, pivot_sheet, &out)?;
24    let Some((rp, cp)) = probe.hit else { return Ok(None) };
25    let records =
26        probe.paths.unwrap_or_default().into_iter().filter(|(_, r, c)| r.starts_with(&rp) && c.starts_with(&cp)).map(|(i, _, _)| i).collect();
27    Ok(Some((src, records)))
28}

30/// The cells of the report grid (relative to its top-left) where [`detail_records`] finds
31/// records: its value, subtotal and grand total cells, from one computation of the report. The
32/// same errors as [`detail_records`].
33pub fn detail_cells(wb: &Workbook, pt: &PivotTable, pivot_sheet: usize) -> Result<HashSet<(u32, u32)>, String> {
34    let src = read_source(wb, &pt.source, pivot_sheet)?;
35    let mut probe = Probe { at: None, hit: None, paths: None, cells: Some(HashSet::new()) };
36    let out = compute_from(wb, pt, &src, Some(&mut probe))?;
37    check_current(wb, pt, pivot_sheet, &out)?;
38    Ok(probe.cells.unwrap_or_default().into_iter().map(|(r, c)| (r as u32, c as u32)).collect())
39}

41/// An error when the report on the sheet is not what `out` computes. Excel reads its PivotTable
42/// cache; without one, the report written by the last refresh must still be what the source gives.
43fn check_current(wb: &Workbook, pt: &PivotTable, pivot_sheet: usize, out: &Output) -> Result<(), String> {
44    let sh = wb.sheet(pivot_sheet).ok_or("There's no such sheet.")?;
45    let current = pt.last_range.is_some_and(|r| {
46        r.height() == out.height().max(1)
47            && r.width() == out.width.max(1)
48            && out.grid.iter().enumerate().all(|(ri, line)| {
49                line.iter().enumerate().all(|(ci, oc)| {
50                    sh.value_ref(CellRef::new(r.start.row + ri as u32, r.start.col + ci as u32)).unwrap_or(&Value::Empty) == &oc.value
51                })
52            })
53    });
54    if !current {
55        return Err("The PivotTable report is out of date. Refresh it first.".into());
56    }
57    Ok(())
58}

99/// Most accumulators (groups times value columns) one report may hold: twice what the 5 million
100/// group limit allowed at one value field.
101const MAX_ACCUMULATORS: usize = 16_000_000;
102
103fn aggregate_record(
104    input: &AggregateInput<'_>,
105    data: &mut AggregateData,
106    probe: &mut Option<&mut Probe>,
107    ri: usize,
108    rec: &[Value],
109) -> Result<(), String> {
110    for (((c, _, group, _), fmt), selected) in input.filters.iter().zip(input.filter_fmts).zip(input.filter_sets) {
111        if let Some(selected) = selected {
112            let label = item_label(input.wb, rec.get(*c).unwrap_or(&Value::Empty), *group, fmt);
113            if !selected.contains(&label.to_ascii_lowercase()) {
114                return Ok(());
115            }
116        }
117    }
118    if input.empty {
119        return Ok(());
120    }
121    let mut row_path = Vec::with_capacity(input.rows.len());
122    for (((column, field), items), fmt) in input.rows.iter().zip(data.row_items.iter_mut()).zip(input.row_fmts) {
123        row_path.push(items.intern(input.wb, rec.get(*column).unwrap_or(&Value::Empty), field.date_group, fmt, ri));
124    }
125    let mut col_path = Vec::with_capacity(input.cols.len());
126    for (((column, field), items), fmt) in input.cols.iter().zip(data.col_items.iter_mut()).zip(input.col_fmts) {
127        col_path.push(items.intern(input.wb, rec.get(*column).unwrap_or(&Value::Empty), field.date_group, fmt, ri));
128    }
129    for row_depth in 0..=row_path.len() {
130        for col_depth in 0..=col_path.len() {
131            let accs = data
132                .aggs
133                .entry((row_path[..row_depth].to_vec(), col_path[..col_depth].to_vec()))
134                .or_insert_with(|| vec![Acc::default(); input.vals.len() + input.extra.len()]);
135            for (acc, (source, _)) in accs.iter_mut().zip(input.vals) {
136                if let ValSrc::Col(column) = source {
137                    acc.add(rec.get(*column).unwrap_or(&Value::Empty));
138                }
139            }
140            for (acc, column) in accs.iter_mut().skip(input.vals.len()).zip(input.extra) {
141                acc.add(rec.get(*column).unwrap_or(&Value::Empty));
142            }
143        }
144    }
145    let shown_depth = input
146        .cols
147        .iter()
148        .zip(data.col_items.iter())
149        .zip(col_path.iter())
150        .position(|(((_, field), items), &item)| !field.collapsed_items.is_empty() && is_collapsed(field, &items.label(item)))
151        .map_or(col_path.len(), |depth| depth + 1);
152    data.col_leaves.insert(col_path[..shown_depth].to_vec());
153    if let Some(paths) = probe.as_deref_mut().and_then(|value| value.paths.as_mut()) {
154        paths.push((ri, row_path.clone(), col_path.clone()));
155    }
156    data.row_set.insert(row_path);
157    data.col_set.insert(col_path);
158    if !input.cols.is_empty() && data_cols_exceed_sheet(data.col_leaves.len(), input.pt.grand_totals_rows, input.vals.len()) {
159        return Err("The PivotTable report will not fit on the sheet.".into());
160    }
161    // Each entry holds one accumulator per value field and calculated-field source column, so the
162    // width counts too: a handful of groups over hundreds of fields is large as well.
163    if data.aggs.len() > 5_000_000 || data.aggs.len().saturating_mul(input.vals.len() + input.extra.len()) > MAX_ACCUMULATORS {
164        return Err("The PivotTable report is too large.".into());
165    }
166    Ok(())
167}

169fn aggregate_source(input: AggregateInput<'_>, probe: &mut Option<&mut Probe>) -> Result<AggregateData, String> {
170    let mut data = AggregateData {
171        row_items: input.rows.iter().map(|_| FieldItems::default()).collect(),
172        col_items: input.cols.iter().map(|_| FieldItems::default()).collect(),
173        ..AggregateData::default()
174    };
175    for (ri, record) in input.src.rows.iter().enumerate() {
176        aggregate_record(&input, &mut data, probe, ri, record)?;
177    }
178    Ok(data)
179}

198fn write_headers(g: &mut Grid, input: HeaderInput<'_>) -> Result<(), String> {
199    let HeaderInput { vals, row_fields, col_fields, col_items, col_fmts, vpos, data_cols, specs, row_levels, compact, on_rows, k, l, headers_on } =
200        input;
201    let val_name = |vi: usize| vals.get(vi).map(|(_, v)| v.name.clone()).unwrap_or_default();
202    let row_header = |g: &mut Grid, r: usize| {
203        if row_levels == 0 || !headers_on {
204            return;
205        }
206        if compact {
207            g.set(r, 0, Value::text("Row Labels"), CellKind::Header, 0, None);
208        } else {
209            // One column per row field, Σ Values ("Values") at its place.
210            let mut names: Vec<&str> = row_fields.iter().map(|f| f.source_col.as_str()).collect();
211            if let Some(p) = vpos {
212                names.insert(p.min(names.len()), "Values");
213            }
214            for (i, name) in names.into_iter().enumerate() {
215                g.set(r, i, Value::text(name), CellKind::Header, 0, None);
216            }
217        }
218    };
219
220    // ---- header rows
221    if col_fields.is_empty() {
222        let r = g.push_row(RowKind::Header)?;
223        row_header(g, r);
224        for (j, (_, vi)) in data_cols.iter().enumerate() {
225            match vi {
226                Some(vi) => g.set(r, l + j, Value::text(val_name(*vi)), CellKind::Header, 0, None),
227                // Values on rows, no column field: the one data column is the grand total.
228                None if on_rows => g.set(r, l + j, Value::text("Total"), CellKind::Header, 0, None),
229                None => {}
230            }
231        }
232    } else {
233        let cap = g.push_row(RowKind::Header)?;
234        if l > 0 && k == 1 {
235            g.set(cap, 0, Value::text(val_name(0)), CellKind::Header, 0, None);
236        }
237        if headers_on {
238            let caption = if compact { "Column Labels".to_string() } else { col_fields.first().map(|f| f.source_col.clone()).unwrap_or_default() };
239            g.set(cap, l, Value::text(caption), CellKind::Header, 0, None);
240        }
241        let levels = col_fields.len() + usize::from(k > 1 && !on_rows);
242        let first_level = g.rows.len();
243        for _ in 0..levels {
244            g.push_row(RowKind::Header)?;
245        }
246        let mut prev: Option<&ColSpec> = None;
247        for (j, (si, vi)) in data_cols.iter().enumerate() {
248            let Some(spec) = specs.get(*si) else { continue };
249            let c = l + j;
250            match spec.kind {
251                ColKind::Leaf => {
252                    for (lv, &it) in spec.path.iter().enumerate() {
253                        let same = prev.is_some_and(|p| p.kind != ColKind::Grand && p.path.get(..=lv) == spec.path.get(..=lv));
254                        if !same {
255                            let Some(item) = col_items.get(lv).and_then(|fi| fi.get(it)) else { continue };
256                            let fmt = if item.formatted { col_fmts.get(lv).cloned() } else { None };
257                            g.set(first_level + lv, c, item.value.clone(), CellKind::ColLabel, 0, fmt);
258                        }
259                    }
260                    if k > 1
261                        && let Some(vi) = vi
262                    {
263                        g.set(first_level + col_fields.len(), c, Value::text(val_name(*vi)), CellKind::ColLabel, 0, None);
264                    }
265                }
266                ColKind::Subtotal => {
267                    let lv = spec.path.len().saturating_sub(1);
268                    let label = spec.path.last().and_then(|it| col_items.get(lv).map(|fi| fi.label(*it))).unwrap_or_default();
269                    let text = match vi {
270                        Some(vi) if k > 1 => caption(&label, &val_name(*vi)),
271                        _ => caption(&label, "Total"),
272                    };
273                    g.set(first_level + lv, c, Value::text(text), CellKind::ColLabel, 0, None);
274                }
275                ColKind::Grand => {
276                    let text = match vi {
277                        Some(vi) if k > 1 => format!("Total {}", val_name(*vi)),
278                        _ => "Grand Total".to_string(),
279                    };
280                    g.set(first_level, c, Value::text(text), CellKind::ColLabel, 0, None);
281                }
282            }
283            prev = Some(spec);
284        }
285        let last = g.rows.len() - 1;
286        row_header(g, last);
287    }
288
289    Ok(())
290}

292struct ValueLookup<'a> {
293    wb: &'a Workbook,
294    source: &'a Source,
295    calc: &'a CalcSet,
296    vals: &'a [(ValSrc, &'a PivotValue)],
297    extra: &'a [usize],
298    aggs: &'a AggMap,
299    value_count: usize,
300}
301
302impl ValueLookup<'_> {
303    fn get(&self, row_path: &[u32], col_path: &[u32], value_index: usize) -> Option<Value> {
304        let (source, definition) = self.vals.get(value_index)?;
305        let accumulators = self.aggs.get(&(row_path.to_vec(), col_path.to_vec()))?;
306        match source {
307            ValSrc::Col(_) => accumulators.get(value_index).map(|acc| acc.result(definition.func)),
308            ValSrc::Calc(index) => {
309                let sum = |name: &str| -> Value {
310                    self.source
311                        .col(name)
312                        .and_then(|column| self.extra.iter().position(|extra_column| *extra_column == column))
313                        .and_then(|index| accumulators.get(self.value_count + index))
314                        .map(|acc| acc.result(PivotFunc::Sum))
315                        .unwrap_or(Value::Number(0.0))
316                };
317                Some(self.calc.value(self.wb, *index, &sum))
318            }
319        }
320    }
321}

336fn ratio(a: Value, b: Option<Value>) -> Value {
337    match (a, b) {
338        (Value::Error(error), _) => Value::Error(error),
339        (_, Some(Value::Error(error))) => Value::Error(error),
340        (Value::Number(_), Some(Value::Number(0.0))) => Value::Error(CellError::Div0),
341        (Value::Number(x), Some(Value::Number(y))) => Value::number(x / y),
342        (value, _) => value,
343    }
344}

346/// What Running Total and Rank need from other cells, computed once per sibling set instead of
347/// once per displayed value (which rescanned the whole set every time).
348type ShowKey = (Vec<u32>, Vec<u32>, usize);
349/// Cumulative values by top-level item, and the total of all of them.
350type RunningTotals = (HashMap<u32, f64>, f64);
351/// Top-level items by (rest of a row path, column path).
352type ItemsByRest = HashMap<(Vec<u32>, Vec<u32>), Vec<u32>>;
353
354#[derive(Default)]
355struct ShowCache {
356    /// Ranked values' sibling numbers, sorted ascending without repeats, by (parent, column, value).
357    ranks: std::cell::RefCell<HashMap<ShowKey, Vec<f64>>>,
358    /// Running totals by top-level item, with the total of all of them, by (rest of path, column, value).
359    running: std::cell::RefCell<HashMap<ShowKey, RunningTotals>>,
360    /// The top-level items that have a row path ending in `rest` (with column path `col`), in
361    /// top-level order, by (`rest`, `col`): built once from the report's cells, so a running total
362    /// visits the items that can hold a number, not every top-level item per distinct `rest`.
363    items_by_rest: std::cell::RefCell<Option<ItemsByRest>>,
364}
365
366fn running_total(raw: Value, lookup: &ValueLookup<'_>, context: &ShowContext<'_>) -> Value {
367    let Some((first, rest)) = context.row_path.split_first() else { return raw };
368    if !matches!(raw, Value::Number(_)) {
369        return raw;
370    }
371    let mut index = context.cache.items_by_rest.borrow_mut();
372    let items_by_rest = index.get_or_insert_with(|| {
373        let position: HashMap<u32, usize> = context.level0.iter().enumerate().rev().map(|(at, item)| (*item, at)).collect();
374        let mut map = ItemsByRest::new();
375        for (row, col) in lookup.aggs.keys() {
376            if let Some((item, rest)) = row.split_first()
377                && position.contains_key(item)
378            {
379                map.entry((rest.to_vec(), col.clone())).or_default().push(*item);
380            }
381        }
382        for items in map.values_mut() {
383            items.sort_by_key(|item| position.get(item).copied().unwrap_or(usize::MAX));
384        }
385        map
386    });
387    let mut cache = context.cache.running.borrow_mut();
388    let (by_item, all) = cache.entry((rest.to_vec(), context.col_path.to_vec(), context.value_index)).or_insert_with(|| {
389        let mut total = 0.0;
390        let mut by_item = HashMap::new();
391        for item in items_by_rest.get(&(rest.to_vec(), context.col_path.to_vec())).into_iter().flatten() {
392            let mut path = Vec::with_capacity(rest.len() + 1);
393            path.push(*item);
394            path.extend_from_slice(rest);
395            // An item with no number here adds nothing and is not stored (the shown cell always has one).
396            let Some(value) = number(&lookup.get(&path, context.col_path, context.value_index)) else { continue };
397            total += value;
398            by_item.entry(*item).or_insert(total);
399        }
400        (by_item, total)
401    });
402    Value::number(by_item.get(first).copied().unwrap_or(*all))
403}

405fn ranked_value(raw: Value, lookup: &ValueLookup<'_>, context: &ShowContext<'_>) -> Value {
406    let Some((_, parent)) = context.row_path.split_last() else { return Value::Empty };
407    let Some(me) = raw.as_f64() else { return raw };
408    let mut cache = context.cache.ranks.borrow_mut();
409    let sorted = cache.entry((parent.to_vec(), context.col_path.to_vec(), context.value_index)).or_insert_with(|| {
410        let mut all: Vec<f64> = context
411            .siblings
412            .get(parent)
413            .into_iter()
414            .flatten()
415            .filter_map(|sibling| {
416                let mut path = parent.to_vec();
417                path.push(*sibling);
418                number(&lookup.get(&path, context.col_path, context.value_index))
419            })
420            .collect();
421        all.sort_by(f64::total_cmp);
422        all.dedup();
423        all
424    });
425    // Excel's ranks are dense: ties share a rank and the next value takes the next one
426    // (5, 4, 3, 3, 4 rank 1, 2, 3, 3, 2), so count the distinct larger values.
427    let larger = sorted.len() - sorted.partition_point(|value| *value <= me);
428    Value::number(larger as f64 + 1.0)
429}

431fn show_value(definition: &PivotValue, raw: Value, lookup: &ValueLookup<'_>, context: &ShowContext<'_>) -> Value {
432    match definition.show_as {
433        PivotShowAs::Normal => raw,
434        PivotShowAs::PercentOfGrandTotal => ratio(raw, lookup.get(&[], &[], context.value_index)),
435        PivotShowAs::PercentOfColumnTotal => ratio(raw, lookup.get(&[], context.col_path, context.value_index)),
436        PivotShowAs::PercentOfRowTotal => ratio(raw, lookup.get(context.row_path, &[], context.value_index)),
437        PivotShowAs::RunningTotal => running_total(raw, lookup, context),
438        PivotShowAs::Rank => ranked_value(raw, lookup, context),
439    }
440}

444fn report_filters<'a>(pt: &'a PivotTable, src: &Source) -> (Vec<ReportFilter<'a>>, Vec<ReportFilter<'a>>) {
445    let axis_group =
446        |name: &str| pt.rows.iter().chain(pt.columns.iter()).find(|field| field.source_col.eq_ignore_ascii_case(name)).map(|field| field.date_group);
447    let filters: Vec<_> = pt
448        .filters
449        .iter()
450        .filter_map(|filter| {
451            let column = src.col(&filter.source_col)?;
452            let group = axis_group(&filter.source_col);
453            Some((column, filter, group.unwrap_or_default(), group.is_none()))
454        })
455        .collect();
456    let shown = filters.iter().copied().filter(|filter| filter.3).collect();
457    (filters, shown)
458}

460fn values_position(pt: &PivotTable, src: &Source, value_count: usize) -> Option<usize> {
461    if !pt.values_on_rows || value_count <= 1 {

464    let position = pt.values_position.unwrap_or(pt.rows.len());
465    Some(pt.rows.iter().take(position).filter(|field| src.col(&field.source_col).is_some()).count())
466}

468fn calculated_source_columns(vals: &[(ValSrc, &PivotValue)], calc: &CalcSet, src: &Source) -> Vec<usize> {
469    let mut columns = Vec::new();
470    for (source, _) in vals {
471        let ValSrc::Calc(index) = source else { continue };
472        for name in calc.source_refs(*index) {
473            if let Some(column) = src.col(&name)
474                && !columns.contains(&column)
475            {
476                columns.push(column);
477            }
478        }
479    }
480    columns
481}

483fn data_columns(specs: &[ColSpec], has_column_fields: bool, on_rows: bool, value_count: usize) -> Result<Vec<(usize, Option<usize>)>, String> {
484    let mut columns = Vec::new();
485    for (spec_index, _) in specs.iter().enumerate() {
486        if on_rows {
487            columns.push((spec_index, None));
488        } else if value_count == 0 {
489            if has_column_fields {
490                columns.push((spec_index, None));
491            }
492        } else {
493            columns.extend((0..value_count).map(|value_index| (spec_index, Some(value_index))));
494        }
495        if columns.len() > MAX_COLS as usize {
496            return Err("The PivotTable report will not fit on the sheet.".into());
497        }
498    }
499    Ok(columns)
500}

502pub(super) fn compute_from(wb: &Workbook, pt: &PivotTable, src: &Source, mut probe: Option<&mut Probe>) -> Result<Output, String> {
503    let fmt_of = |c: usize| src.formats.get(c).cloned().unwrap_or_else(|| "General".into());
504    let rows: Vec<(usize, &PivotField)> = pt.rows.iter().filter_map(|f| src.col(&f.source_col).map(|c| (c, f))).collect();
505    let cols: Vec<(usize, &PivotField)> = pt.columns.iter().filter_map(|f| src.col(&f.source_col).map(|c| (c, f))).collect();
506    // Value fields: a source column, or a calculated field evaluated on sums (see `pivot_calc`).
507    let calc = CalcSet::new(pt, &src.headers);
508    let vals: Vec<(ValSrc, &PivotValue)> = pt
509        .values
510        .iter()
511        .filter_map(|v| match src.col(&v.source_col) {
512            Some(c) => Some((ValSrc::Col(c), v)),
513            None => calc.index(&v.source_col).map(|i| (ValSrc::Calc(i), v)),
514        })
515        .collect();
516    // Source columns summed for the calculated fields, accumulated after the value fields.
517    let extra = calculated_source_columns(&vals, &calc, src);
518    if rows.len() + cols.len() > MAX_AXIS_FIELDS {
519        return Err(format!("A PivotTable can have at most {MAX_AXIS_FIELDS} row and column fields."));
520    }
521    if vals.len() > MAX_VALUE_FIELDS {
522        return Err(format!("A PivotTable can have at most {MAX_VALUE_FIELDS} value fields."));
523    }
524    let (filters, shown_filters) = report_filters(pt, src);
525
526    let k = vals.len();
527    let empty = rows.is_empty() && cols.is_empty() && vals.is_empty();
528    // Σ Values in Rows (it exists with two or more value fields): its depth among the row fields
529    // found in the source.
530    let vpos = values_position(pt, src, k);
531    let on_rows = vpos.is_some();
532    let row_levels = rows.len() + usize::from(on_rows);
533
534    // ---- aggregate
535    let filter_fmts: Vec<String> = filters.iter().map(|f| fmt_of(f.0)).collect();
536    let filter_sets: Vec<Option<HashSet<String>>> =
537        filters.iter().map(|(_, field, _, _)| field.selected.as_ref().map(|s| s.iter().map(|v| v.to_ascii_lowercase()).collect())).collect();
538    let row_fmts: Vec<String> = rows.iter().map(|(c, _)| fmt_of(*c)).collect();
539    let col_fmts: Vec<String> = cols.iter().map(|(c, _)| fmt_of(*c)).collect();
540    let aggregate = aggregate_source(
541        AggregateInput {
542            wb,
543            pt,
544            src,
545            rows: &rows,
546            cols: &cols,
547            vals: &vals,
548            extra: &extra,
549            filters: &filters,
550            filter_fmts: &filter_fmts,
551            filter_sets: &filter_sets,
552            row_fmts: &row_fmts,
553            col_fmts: &col_fmts,
554            empty,
555        },
556        &mut probe,
557    )?;
558    let AggregateData { row_items, col_items, aggs, row_set, col_set, col_leaves: _ } = aggregate;
559
560    // ---- axes
561    let row_fields: Vec<&PivotField> = rows.iter().map(|(_, f)| *f).collect();
562    let col_fields: Vec<&PivotField> = cols.iter().map(|(_, f)| *f).collect();
563    let row_paths: Vec<&[u32]> = row_set.iter().map(Vec::as_slice).collect();
564    let col_paths: Vec<&[u32]> = col_set.iter().map(Vec::as_slice).collect();
565    let row_tree = build_tree(&row_paths, 0, &row_fields, &row_items);
566    let col_tree = build_tree(&col_paths, 0, &col_fields, &col_items);
567    let mut siblings: HashMap<Vec<u32>, Vec<u32>> = HashMap::new();
568    collect_siblings(&row_tree, &mut Vec::new(), &mut siblings);
569    let level0: Vec<u32> = row_tree.iter().map(|n| n.item).collect();
570
571    let val_names: Vec<String> = vals.iter().map(|(_, v)| v.name.clone()).collect();
572    let ctx = Ctx { pt, rows: row_fields.clone(), row_items, row_fmts: row_fmts.clone(), vpos, val_names: val_names.clone() };
573    let mut lines: Vec<Line> = Vec::new();
574    if row_levels == 0 {
575        lines.push(Line { kind: LineKind::Item(false), leaf: true, path: vec![], values: true, labels: vec![], vi: None });
576    } else {
577        if pt.layout == PivotLayout::Tabular {
578            ctx.flatten_tabular(&row_tree, 0, None, &mut Vec::new(), &[], &mut lines);
579        } else {
580            ctx.flatten(&row_tree, 0, None, &mut Vec::new(), &mut lines);
581        }
582        // With only Σ Values in Rows its items are the totals already.
583        if pt.grand_totals_cols && !rows.is_empty() {
584            let grand = |text: String, vi: Option<usize>| Line {
585                kind: LineKind::Grand,
586                leaf: false,
587                path: vec![],
588                values: true,
589                labels: vec![(0, Value::text(text), 0, None)],
590                vi,
591            };
592            if on_rows {
593                lines.extend(val_names.iter().enumerate().map(|(vi, name)| grand(format!("Total {name}"), Some(vi))));
594            } else {
595                lines.push(grand("Grand Total".into(), None));
596            }
597        }
598    }
599    let mut specs: Vec<ColSpec> = Vec::new();
600    if cols.is_empty() {
601        specs.push(ColSpec { kind: ColKind::Leaf, path: vec![] });
602    } else {
603        flatten_cols(&col_tree, 0, &col_fields, &col_items, pt.subtotals, &mut Vec::new(), &mut specs);
604        if pt.grand_totals_rows {
605            specs.push(ColSpec { kind: ColKind::Grand, path: vec![] });
606        }
607    }
608    let data_cols = data_columns(&specs, !cols.is_empty(), on_rows, k)?;
609
610    let compact = pt.layout == PivotLayout::Compact;
611    let l = if row_levels == 0 && cols.is_empty() {
612        0
613    } else if compact {
614        1
615    } else {
616        row_levels.max(1)
617    };
618    let mut width = l + data_cols.len();
619    if !shown_filters.is_empty() {
620        width = width.max(2);
621    }
622    if empty {
623        width = width.max(1);
624    }
625    let width = width.max(1);
626    let mut g = Grid { rows: vec![], kinds: vec![], width };
627
628    // ---- report filters
629    for (c, f, _, _) in &shown_filters {
630        let r = g.push_row(RowKind::Filter)?;
631        let name = src.headers.get(*c).cloned().unwrap_or_default();
632        let shown = match &f.selected {
633            None => "(All)".to_string(),
634            Some(v) if v.len() == 1 => v.first().cloned().unwrap_or_default(),
635            Some(_) => "(Multiple Items)".to_string(),
636        };
637        g.set(r, 0, Value::text(name), CellKind::Filter, 0, None);
638        g.set(r, 1, Value::text(shown), CellKind::Filter, 0, None);
639    }
640    if !shown_filters.is_empty() {
641        g.push_row(RowKind::Blank)?;
642    }
643    let body_offset = g.rows.len() as u32;
644
645    if empty {
646        let r = g.push_row(RowKind::Placeholder)?;
647        g.set(r, 0, Value::text(format!("{} — choose fields", pt.name)), CellKind::Placeholder, 0, None);
648        return Ok(Output {
649            width: g.width as u32,
650            grid: g.rows,
651            row_kinds: g.kinds,
652            body_offset,
653            chart_rows: vec![],
654            chart_cols: vec![],
655            label_col: None,
656        });
657    }
658
659    write_headers(
660        &mut g,
661        HeaderInput {
662            vals: &vals,
663            row_fields: &row_fields,
664            col_fields: &col_fields,
665            col_items: &col_items,
666            col_fmts: &col_fmts,
667            vpos,
668            data_cols: &data_cols,
669            specs: &specs,
670            row_levels,
671            compact,
672            on_rows,
673            k,
674            l,
675            headers_on: pt.show_headers,
676        },
677    )?;
678    let val_name = |vi: usize| vals.get(vi).map(|(_, value)| value.name.clone()).unwrap_or_default();
679
680    // ---- body
681    let data_fmt: Vec<Option<String>> = vals
682        .iter()
683        .map(|(vs, v)| match v.show_as {
684            PivotShowAs::PercentOfGrandTotal | PivotShowAs::PercentOfColumnTotal | PivotShowAs::PercentOfRowTotal => {
685                Some(v.number_format.clone().unwrap_or_else(|| "0.00%".into()))
686            }
687            PivotShowAs::Rank => v.number_format.clone(),
688            _ => v.number_format.clone().or_else(|| match v.func {
689                PivotFunc::Count | PivotFunc::CountNumbers => None,
690                _ => {
691                    let f = match vs {
692                        ValSrc::Col(c) => fmt_of(*c),
693                        ValSrc::Calc(_) => "General".to_string(),
694                    };
695                    (f != "General").then_some(f)
696                }
697            }),
698        })
699        .collect();
700    let lookup = ValueLookup { wb, source: src, calc: &calc, vals: &vals, extra: &extra, aggs: &aggs, value_count: k };
701    let show_cache = ShowCache::default();
702    let mut chart_rows: Vec<u32> = Vec::new();
703    let chart_cols: Vec<(u32, String)> = data_cols
704        .iter()
705        .enumerate()
706        .filter_map(|(j, (si, vi))| {
707            let spec = specs.get(*si)?;
708            if spec.kind != ColKind::Leaf {
709                return None;
710            }
711            let mut parts: Vec<String> =
712                spec.path.iter().enumerate().map(|(lv, it)| col_items.get(lv).map(|fi| fi.label(*it)).unwrap_or_default()).collect();
713            if let Some(vi) = vi
714                && (k > 1 || parts.is_empty())
715            {
716                parts.push(val_name(*vi));
717            } else if parts.is_empty() {
718                parts.push("Total".into());
719            }
720            Some(((l + j) as u32, parts.join(" - ")))
721        })
722        .filter(|_| k > 0)
723        .collect();
724    let label_col = if row_levels == 0 {
725        None
726    } else if compact {
727        Some(0)
728    } else {
729        Some(row_levels.saturating_sub(1) as u32)
730    };
731    for line in &lines {
732        let rk = match line.kind {
733            LineKind::Item(b) => RowKind::Item(b),
734            LineKind::Subtotal => RowKind::Subtotal,
735            LineKind::Grand => RowKind::Grand,
736        };
737        let ck = match line.kind {
738            LineKind::Item(_) => (CellKind::RowLabel, CellKind::Data),
739            LineKind::Subtotal => (CellKind::Subtotal, CellKind::Subtotal),
740            LineKind::Grand => (CellKind::GrandTotal, CellKind::GrandTotal),
741        };
742        let r = g.push_row(rk)?;
743        if line.leaf && k > 0 {
744            chart_rows.push(r as u32);
745        }
746        for (c, v, indent, fmt) in &line.labels {
747            g.set(r, *c, v.clone(), ck.0, *indent, fmt.clone());
748        }
749        if !line.values {
750            continue;
751        }
752        let rp = line.path.as_slice();
753        for (j, (si, col_vi)) in data_cols.iter().enumerate() {
754            let line_vi = col_vi.or(line.vi);
755            let (Some(spec), Some(vi)) = (specs.get(*si), &line_vi) else { continue };
756            let Some((_, vdef)) = vals.get(*vi) else { continue };
757            let cp = spec.path.as_slice();
758            let Some(raw) = lookup.get(rp, cp, *vi) else { continue };
759            let context = ShowContext { row_path: rp, col_path: cp, value_index: *vi, level0: &level0, siblings: &siblings, cache: &show_cache };
760            let shown = show_value(vdef, raw, &lookup, &context);
761            if let Some(pr) = probe.as_deref_mut() {
762                if pr.at == Some((r, l + j)) {
763                    pr.hit = Some((rp.to_vec(), cp.to_vec()));
764                }
765                if let Some(cells) = pr.cells.as_mut() {
766                    cells.insert((r, l + j));
767                }
768            }
769            g.set(r, l + j, shown, ck.1, 0, data_fmt.get(*vi).cloned().flatten());
770        }
771    }
772    Ok(Output { width: g.width as u32, grid: g.rows, row_kinds: g.kinds, body_offset, chart_rows, chart_cols, label_col })
773}
