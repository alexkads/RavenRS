79fn map_variables(n: usize, lo: &[f64], hi: &[f64]) -> VariableMap {
80    let mut columns = Vec::new();
81    let mut offset = vec![0.0; n];
82    let mut bound_rows = Vec::new();
83    for j in 0..n {
84        let lower = lo.get(j).copied().unwrap_or(f64::NEG_INFINITY);
85        let upper = hi.get(j).copied().unwrap_or(f64::INFINITY);
86        if lower.is_finite() {
87            if let Some(value) = offset.get_mut(j) {
88                *value = lower;
89            }
90            columns.push(Map { var: j, sign: 1.0 });
91            if upper.is_finite() {
92                bound_rows.push((vec![(columns.len() - 1, 1.0)], Sense::Le, upper - lower));
93            }
94        } else if upper.is_finite() {
95            if let Some(value) = offset.get_mut(j) {
96                *value = upper;
97            }
98            columns.push(Map { var: j, sign: -1.0 });
99        } else {
100            columns.push(Map { var: j, sign: 1.0 });
101            columns.push(Map { var: j, sign: -1.0 });
102        }
103    }
104    VariableMap { columns, offset, bound_rows }
105}

107/// The rows over the tableau's columns, each with a non-negative right side, and per row
108/// whether it was negated to get there.
109fn normalize_rows(rows: &[(Vec<f64>, Sense, f64)], bound_rows: Vec<BoundRow>, columns: &[Map], offset: &[f64]) -> (Vec<LpRow>, Vec<bool>) {
110    let width = columns.len();
111    let mut transformed = Vec::with_capacity(rows.len() + bound_rows.len());
112    for (coefficients, sense, rhs) in rows {
113        let mut row = vec![0.0; width];
114        for (index, mapping) in columns.iter().enumerate() {
115            if let Some(value) = row.get_mut(index) {
116                *value = coefficients.get(mapping.var).copied().unwrap_or(0.0) * mapping.sign;
117            }
118        }
119        let shift: f64 = coefficients.iter().zip(offset).map(|(coefficient, base)| coefficient * base).sum();
120        transformed.push((row, *sense, rhs - shift));
121    }
122    for (entries, sense, rhs) in bound_rows {
123        let mut row = vec![0.0; width];
124        for (index, value) in entries {
125            if let Some(slot) = row.get_mut(index) {
126                *slot = value;
127            }
128        }
129        transformed.push((row, sense, rhs));
130    }
131    // Rows with a negative right side are negated so every right side is >= 0.
132    let flipped = transformed.iter().map(|row| row.2 < 0.0).collect();
133    let normalized = transformed
134        .into_iter()
135        .map(|(row, sense, rhs)| {
136            if rhs < 0.0 {
137                let sense = match sense {
138                    Sense::Le => Sense::Ge,
139                    Sense::Ge => Sense::Le,
140                    Sense::Eq => Sense::Eq,
141                };
142                (row.iter().map(|v| -v).collect(), sense, -rhs)
143            } else {
144                (row, sense, rhs)
145            }
146        })
147        .collect();
148    (normalized, flipped)
149}

151fn build_tableau(normalized: &[(Vec<f64>, Sense, f64)], decision_cols: usize, slack_cols: usize, artificial_cols: usize) -> Tableau {
152    let width = decision_cols + slack_cols + artificial_cols;
153    let mut rows = Vec::with_capacity(normalized.len());
154    let mut basis = vec![0usize; normalized.len()];
155    let (mut slack, mut artificial) = (decision_cols, decision_cols + slack_cols);
156    for (i, (coefficients, sense, rhs)) in normalized.iter().enumerate() {
157        let mut row = vec![0.0; width + 1];
158        for (value, coefficient) in row.iter_mut().zip(coefficients) {
159            *value = *coefficient;
160        }
161        let basic = match sense {
162            Sense::Le => {
163                if let Some(value) = row.get_mut(slack) {
164                    *value = 1.0;
165                }
166                let basic = slack;
167                slack += 1;
168                basic
169            }
170            Sense::Ge => {
171                if let Some(value) = row.get_mut(slack) {
172                    *value = -1.0;
173                }
174                slack += 1;
175                if let Some(value) = row.get_mut(artificial) {
176                    *value = 1.0;
177                }
178                let basic = artificial;
179                artificial += 1;
180                basic
181            }
182            Sense::Eq => {
183                if let Some(value) = row.get_mut(artificial) {
184                    *value = 1.0;
185                }
186                let basic = artificial;
187                artificial += 1;
188                basic
189            }
190        };
191        if let Some(value) = row.get_mut(width) {
192            *value = *rhs;
193        }
194        if let Some(slot) = basis.get_mut(i) {
195            *slot = basic;
196        }
197        rows.push(row);
198    }
199    Tableau { rows, basis, width, decision_cols, slack_cols, artificial_cols }
200}

202fn phase_one(tableau: &mut Tableau, normalized: &[(Vec<f64>, Sense, f64)], budget: &mut u64) -> Result<(), Outcome> {
203    if tableau.artificial_cols == 0 {
204        return Ok(());
205    }
206    let mut objective = vec![0.0; tableau.width + 1];
207    for column in tableau.decision_cols + tableau.slack_cols..tableau.width {
208        if let Some(value) = objective.get_mut(column) {
209            *value = 1.0;
210        }
211    }
212    for (i, row) in tableau.rows.iter().enumerate() {
213        if tableau.basis.get(i).is_some_and(|basic| *basic >= tableau.decision_cols + tableau.slack_cols) {
214            for (value, coefficient) in objective.iter_mut().zip(row) {
215                *value -= coefficient;
216            }
217        }
218    }
219    let all = vec![true; tableau.width];
220    if matches!(simplex(&mut tableau.rows, &mut tableau.basis, &mut objective, &all, budget), Step::Limit) {
221        return Err(Outcome::Limit);
222    }
223    let scale = normalized.iter().map(|row| row.2.abs()).fold(1.0f64, f64::max);
224    if -objective.get(tableau.width).copied().unwrap_or(0.0) > 1e-7 * scale {
225        return Err(Outcome::Infeasible);
226    }
227    for i in 0..tableau.rows.len() {
228        if tableau.basis.get(i).is_none_or(|basic| *basic < tableau.decision_cols + tableau.slack_cols) {
229            continue;
230        }
231        let pivot_col = tableau
232            .rows
233            .get(i)
234            .and_then(|row| (0..tableau.decision_cols + tableau.slack_cols).find(|&column| row.get(column).is_some_and(|value| value.abs() > EPS)));
235        if let Some(column) = pivot_col {
236            pivot(&mut tableau.rows, &mut objective, i, column);
237            if let Some(basic) = tableau.basis.get_mut(i) {
238                *basic = column;
239            }
240        }
241    }
242    Ok(())
243}

245/// Depth-first branch and bound over a relaxation solver; returns the best integer point.
246pub(super) fn branch_and_bound(
247    lo: &[f64],
248    hi: &[f64],
249    integer: &[bool],
250    max_nodes: usize,
251    gap: f64,
252    relax: &mut dyn FnMut(&[f64], &[f64]) -> Outcome,
253) -> (Outcome, usize) {
254    let mut stack: Vec<(Vec<f64>, Vec<f64>)> = vec![(lo.to_vec(), hi.to_vec())];
255    let mut best: Option<(Vec<f64>, f64)> = None;
256    let mut nodes = 0usize;
257    let mut limited = false;
258    let mut root = true;
259    while let Some((l, h)) = stack.pop() {
260        if nodes >= max_nodes {
261            limited = true;
262            break;
263        }
264        nodes += 1;
265        let out = relax(&l, &h);
266        let was_root = std::mem::replace(&mut root, false);
267        let (x, obj, converged) = match out {
268            Outcome::Solved(x, obj, c) => (x, obj, c),
269            Outcome::Unbounded if was_root => return (Outcome::Unbounded, nodes),
270            Outcome::Limit => {
271                limited = true;
272                break;
273            }
274            _ => continue,
275        };
276        if !converged {
277            limited = true;
278        }
279        if let Some((_, b)) = &best
280            && obj >= b - gap * b.abs().max(1e-12) - 1e-9 * (1.0 + b.abs())
281        {
282            continue;
283        }
284        // The most fractional integer variable.
285        let frac = x
286            .iter()
287            .zip(integer)
288            .enumerate()
289            .filter(|(_, (_, int))| **int)
290            .map(|(j, (v, _))| (j, (v - v.round()).abs()))
291            .filter(|(_, f)| *f > 1e-6)
292            .max_by(|a, b| a.1.total_cmp(&b.1));
293        match frac {
294            None => {
295                let snapped: Vec<f64> = x.iter().zip(integer).map(|(v, int)| if *int { v.round() } else { *v }).collect();
296                best = Some((snapped, obj));
297            }
298            Some((j, _)) => {
299                let v = x.get(j).copied().unwrap_or(0.0);
300                let (mut down_h, mut up_l) = (h.clone(), l.clone());
301                if let Some(e) = down_h.get_mut(j) {
302                    *e = e.min(v.floor());
303                }
304                if let Some(e) = up_l.get_mut(j) {
305                    *e = e.max(v.ceil());
306                }
307                // Explore the nearer side first (pushed last).
308                if v - v.floor() < 0.5 {
309                    stack.push((up_l, h));
310                    stack.push((l, down_h));
311                } else {
312                    stack.push((l, down_h));
313                    stack.push((up_l, h));
314                }
315            }
316        }
317    }
318    match best {
319        Some((x, obj)) => (Outcome::Solved(x, obj, !limited), nodes),
320        None if limited => (Outcome::Limit, nodes),
321        None => (Outcome::Infeasible, nodes),
322    }
323}

327/// Pivots the tableau on (r, j), updating the objective row.
328fn pivot(t: &mut [Vec<f64>], obj: &mut [f64], r: usize, j: usize) {
329    let Some(prow) = t.get_mut(r) else { return };
330    let pv = prow.get(j).copied().unwrap_or(0.0);
331    if pv == 0.0 {
332        return;
333    }
334    for v in prow.iter_mut() {
335        *v /= pv;
336    }
337    let prow = prow.clone();
338    for (i, row) in t.iter_mut().enumerate() {
339        if i == r {
340            continue;
341        }
342        let f = row.get(j).copied().unwrap_or(0.0);
343        if f != 0.0 {
344            for (v, p) in row.iter_mut().zip(&prow) {
345                *v -= f * p;
346            }
347        }
348    }
349    let f = obj.get(j).copied().unwrap_or(0.0);
350    if f != 0.0 {
351        for (v, p) in obj.iter_mut().zip(&prow) {
352            *v -= f * p;
353        }
354    }
355}

357enum Step {
358    Optimal,
359    Unbounded,
360    Limit,
361}
362
363/// Minimises with the reduced costs in `obj` (last entry: −objective). Columns not `allowed`
364/// never enter.
365fn simplex(t: &mut [Vec<f64>], basis: &mut [usize], obj: &mut [f64], allowed: &[bool], budget: &mut u64) -> Step {
366    let n = obj.len().saturating_sub(1);
367    let cost = (t.len() as u64 + 1).saturating_mul(n as u64 + 1).max(1);
368    let mut stalled = 0u32;
369    loop {
370        if *budget < cost {
371            return Step::Limit;
372        }
373        *budget -= cost;
374        let bland = stalled > 50;
375        let mut enter = None;
376        let mut best = -EPS;
377        for (j, d) in obj.iter().take(n).enumerate() {
378            if !allowed.get(j).copied().unwrap_or(false) || *d >= -EPS {
379                continue;
380            }
381            if bland {
382                enter = Some(j);
383                break;
384            }
385            if *d < best {
386                best = *d;
387                enter = Some(j);
388            }
389        }
390        let Some(j) = enter else { return Step::Optimal };
391        let mut leave: Option<(usize, f64)> = None;
392        for (i, row) in t.iter().enumerate() {
393            let a = row.get(j).copied().unwrap_or(0.0);
394            if a <= EPS {
395                continue;
396            }
397            let ratio = row.get(n).copied().unwrap_or(0.0) / a;
398            let better = match leave {
399                None => true,
400                Some((li, lr)) => ratio < lr - 1e-12 || (ratio <= lr + 1e-12 && basis.get(i) < basis.get(li)),
401            };
402            if better {
403                leave = Some((i, ratio));
404            }
405        }
406        let Some((r, ratio)) = leave else { return Step::Unbounded };
407        stalled = if ratio <= EPS { stalled + 1 } else { 0 };
408        pivot(t, obj, r, j);
409        if let Some(b) = basis.get_mut(r) {
410            *b = j;
411        }
412    }
413}

420/// [`solve_lp`], with the final tableau when it is optimal.
421pub(super) fn solve_lp_tableau(
422    c: &[f64],
423    rows: &[(Vec<f64>, Sense, f64)],
424    lo: &[f64],
425    hi: &[f64],
426    budget: &mut u64,
427) -> (Outcome, Option<FinalTableau>) {
428    if lo.iter().zip(hi).any(|(l, h)| l > h) {
429        return (Outcome::Infeasible, None);
430    }
431    // x_j = offset + Σ sign·y_k with y ≥ 0.
432    let mapping = map_variables(c.len(), lo, hi);
433    let VariableMap { columns: cols, offset, bound_rows: extra } = mapping;
434    let ny = cols.len();
435    // Rows over y: (coefficients, sense, rhs).
436    let (normalized, flipped) = normalize_rows(rows, extra, &cols, &offset);
437    let n_slack = normalized.iter().filter(|row| row.1 != Sense::Eq).count();
438    let n_art = normalized.iter().filter(|row| row.1 != Sense::Le).count();
439    let mut tableau = build_tableau(&normalized, ny, n_slack, n_art);
440    // Each row's starting basic column is its unit column.
441    let ident = tableau.basis.clone();
442    if let Err(outcome) = phase_one(&mut tableau, &normalized, budget) {
443        return (outcome, None);
444    }
445    let Tableau { rows: mut t, mut basis, width, decision_cols: ny, slack_cols: n_slack, .. } = tableau;
446    let is_art = |column: usize| column >= ny + n_slack && column < width;
447    // Phase 2: the real costs; artificials may not re-enter.
448    let mut cost = vec![0.0; width + 1];
449    for (k, mp) in cols.iter().enumerate() {
450        if let Some(v) = cost.get_mut(k) {
451            *v = c.get(mp.var).copied().unwrap_or(0.0) * mp.sign;
452        }
453    }
454    let mut obj = cost.clone();
455    for (i, row) in t.iter().enumerate() {
456        let cb = basis.get(i).and_then(|b| cost.get(*b)).copied().unwrap_or(0.0);
457        if cb != 0.0 {
458            for (v, x) in obj.iter_mut().zip(row) {
459                *v -= cb * x;
460            }
461        }
462    }
463    let allowed: Vec<bool> = (0..width).map(|j| !is_art(j)).collect();
464    match simplex(&mut t, &mut basis, &mut obj, &allowed, budget) {
465        Step::Limit => return (Outcome::Limit, None),
466        Step::Unbounded => return (Outcome::Unbounded, None),
467        Step::Optimal => {}
468    }
469    let mut y = vec![0.0; width];
470    for (i, row) in t.iter().enumerate() {
471        if let (Some(b), Some(v)) = (basis.get(i), row.get(width))
472            && let Some(e) = y.get_mut(*b)
473        {
474            *e = *v;
475        }
476    }
477    let mut x = offset;
478    for (k, mp) in cols.iter().enumerate() {
479        if let Some(e) = x.get_mut(mp.var) {
480            *e += mp.sign * y.get(k).copied().unwrap_or(0.0);
481        }
482    }
483    let val: f64 = c.iter().zip(&x).map(|(a, b)| a * b).sum();
484    let tableau = FinalTableau { t, basis, obj, width, first_art: ny + n_slack, cols, ident, flipped };
485    (Outcome::Solved(x, val, true), Some(tableau))
486}

488/// The LP to minimise for `goal`, from the linearised model: the costs, and the rows
489/// (coefficients, sense, right side) in the order of `model_rows`.
490pub(super) fn linear_program(goal: Goal, cf: &[f64], f_const: f64, lin: &[(Vec<f64>, f64)], model_rows: &[Row]) -> (Vec<f64>, Vec<LpRow>) {
491    let mut rows: Vec<LpRow> = lin.iter().zip(model_rows).map(|((a, k), r)| (a.clone(), r.sense, -k)).collect();
492    let c = match goal {
493        Goal::Max => cf.iter().map(|v| -v).collect(),
494        Goal::Min => cf.to_vec(),
495        Goal::Value(v) => {
496            rows.push((cf.to_vec(), Sense::Eq, v - f_const));
497            vec![0.0; cf.len()]
498        }
499    };
500    (c, rows)
501}

503/// The linear program Simplex LP solves: the model linearised at `x0`, its objective
504/// coefficients, the costs and rows for the goal, and the work limit for its size.
505pub(super) struct SimplexProgram {
506    pub(super) cf: Vec<f64>,
507    pub(super) c: Vec<f64>,
508    pub(super) rows: Vec<LpRow>,
509    pub(super) work: u64,
510}
511
512/// Builds the [`SimplexProgram`] at `x0`. The solve and the Sensitivity report both build it
513/// here from the start values, so the report's solve repeats the solve's pivots and returns the
514/// same vertex: linearising at another point changes the coefficients in their last bits, and
515/// that can break a tie between pivots the other way.
516pub(super) fn simplex_program(ev: &mut Evaluator, goal: Goal, x0: &[f64]) -> std::result::Result<SimplexProgram, &'static str> {
517    let ((cf, f_const), lin) = linearize(ev, x0)?;
518    let (c, rows) = linear_program(goal, &cf, f_const, &lin, &ev.rows);
519    let size = (rows.len() as u64 + 2 * x0.len() as u64 + 1) * (4 * x0.len() as u64 + 2 * rows.len() as u64 + 1);
520    Ok(SimplexProgram { cf, c, rows, work: LP_WORK.max(size) })
521}

523/// Linearises the model at `x0` (n + 1 evaluations) and checks it at one more point. Returns the
524/// objective gradient and constant, and per row the gradient and constant of `lhs − rhs`.
525pub(super) fn linearize(ev: &mut Evaluator, x0: &[f64]) -> std::result::Result<Linearized, &'static str> {
526    let n = x0.len();
527    let p0 = ev.eval(x0).ok_or("budget")?;
528    if !p0.f.is_finite() || p0.g.iter().any(|v| !v.is_finite()) {
529        return Err("error");
530    }
531    let mut cf = vec![0.0; n];
532    let mut ga = vec![vec![0.0; n]; p0.g.len()];
533    for j in 0..n {
534        let mut x = x0.to_vec();
535        if let Some(v) = x.get_mut(j) {
536            *v += 1.0;
537        }
538        let p = ev.eval(&x).ok_or("budget")?;
539        if let Some(v) = cf.get_mut(j) {
540            *v = p.f - p0.f;
541        }
542        for (row, (gi, g0)) in ga.iter_mut().zip(p.g.iter().zip(&p0.g)) {
543            if let Some(v) = row.get_mut(j) {
544                *v = gi - g0;
545            }
546        }
547    }
548    let dot = |a: &[f64], x: &[f64]| a.iter().zip(x).map(|(p, q)| p * q).sum::<f64>();
549    let f_const = p0.f - dot(&cf, x0);
550    let rows: Vec<(Vec<f64>, f64)> = ga.into_iter().zip(&p0.g).map(|(a, g0)| (a.clone(), g0 - dot(&a, x0))).collect();
551    // Linearity check at a spread-out point.
552    let xt: Vec<f64> =
553        x0.iter().enumerate().map(|(j, v)| v + if j % 2 == 0 { 1.7 + 0.3 * (j % 5) as f64 } else { -0.6 - 0.2 * (j % 3) as f64 }).collect();
554    let pt = ev.eval(&xt).ok_or("budget")?;
555    let close = |actual: f64, pred: f64, terms: f64| actual.is_finite() && (actual - pred).abs() <= 1e-6 * (1.0 + actual.abs() + terms);
556    let terms = |a: &[f64]| a.iter().zip(&xt).map(|(p, q)| (p * q).abs()).sum::<f64>();
557    if !close(pt.f, f_const + dot(&cf, &xt), terms(&cf)) {
558        return Err("nonlinear");
559    }
