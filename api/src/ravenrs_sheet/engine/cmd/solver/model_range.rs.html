12/// Most cells a model written by Save Model takes: the objective, the variables, the options
13/// and one per constraint.
14const MAX_MODEL_ROWS: usize = MAX_CONSTRAINTS + 3;
15
16fn is_options_row(formula: &str) -> bool {
17    formula.trim().trim_start_matches('=').trim_start().starts_with('{')
18}
19
20/// The formulas from `start` down: to the first cell without a formula, or through the options
21/// row (`{…}`) that ends a model written by Save Model; at most [`MAX_MODEL_ROWS`] cells.
22fn formulas_down(sheet: &Sheet, start: CellRef) -> Vec<(CellRef, String)> {
23    let mut out = Vec::new();
24    for k in 0..MAX_MODEL_ROWS as u32 {
25        let Some(c) = start.row.checked_add(k).map(|row| CellRef::new(row, start.col)) else { break };
26        let Some(f) = sheet.cell(c).and_then(|x| x.formula.as_ref()).map(|f| f.text.clone()) else { break };
27        let last = is_options_row(&f);
28        out.push((c, f));
29        if last {
30            break;
31        }
32    }
33    out
34}

36/// The cells of a model written by Save Model that starts at `start` (none if the cells there
37/// don't hold one).
38fn saved_model_cells(sheet: &Sheet, start: CellRef) -> Vec<CellRef> {
39    let old = formulas_down(sheet, start);
40    let rows: Vec<String> = old.iter().map(|(_, f)| f.clone()).collect();
41    if rows.last().is_some_and(|f| is_options_row(f)) && parse_model_formulas(&rows).is_ok() {
42        old.into_iter().map(|(c, _)| c).collect()
43    } else {
44        Vec::new()
45    }
46}

48/// How many cells the options row of Save Model (`={…}`, one row) spans.
49fn options_width(formula: &str) -> u32 {
50    u32::try_from(formula.split(',').count()).unwrap_or(u32::MAX)
51}
52
53/// The options row is an array constant that spills `width - 1` cells to the right of `anchor`.
54/// Refused, before anything is written, when the spill would run off the sheet or be blocked
55/// (the anchor would show `#SPILL!`): a cell there holds content or another formula's spilled
56/// values, or a merged area or a table meets the row. `ours` are the cells Save Model
57/// writes or clears (a model saved at the same place before, whose options spill may be
58/// replaced); `used` are the model's own cells, which the spill must not cover.
59fn check_options_spill(sheet: &Sheet, anchor: CellRef, width: u32, ours: &[CellRef], used: &[CellRef]) -> Result<()> {
60    if width <= 1 {
61        return Ok(());
62    }
63    let end = anchor.col.checked_add(width - 1).filter(|c| *c < crate::base::MAX_COLS).ok_or_else(|| {
64        EngineError::Other(format!(
65            "The saved model's options need {width} cells to the right from {}; choose another place to save it.",
66            anchor.a1()
67        ))
68    })?;
69    let full = RangeRef::new(anchor, CellRef::new(anchor.row, end));
70    let spill = RangeRef::new(CellRef::new(anchor.row, anchor.col.saturating_add(1)), CellRef::new(anchor.row, end));
71    let blocked =
72        |why: String| EngineError::Other(format!("The saved model's options fill {}, but {why}; choose another place to save it.", spill.a1()));
73    if let Some(c) = used.iter().find(|c| spill.contains(**c)) {
74        return Err(blocked(format!("{} is a cell of the model", c.a1())));
75    }
76    for c in spill.iter() {
77        if sheet.cells.has(c) {
78            return Err(blocked(format!("{} isn't empty", c.a1())));
79        }
80        if sheet.spill.contains_key(&c) && sheet.spill_anchor(c).is_none_or(|a| !ours.contains(&a)) {
81            return Err(blocked(format!("{} holds the values of another formula", c.a1())));
82        }
83    }
84    if sheet.merges.iter().any(|m| m.intersects(&full)) {
85        return Err(blocked("merged cells are in the way".into()));
86    }
87    // A dynamic array can't spill into a table.
88    if sheet.tables.iter().any(|t| t.range.intersects(&full)) {
89        return Err(blocked("a table is in the way".into()));
90    }
91    Ok(())
92}

94/// Refused when a cell of the model's column holds another formula's spilled values: the
95/// formula written there would block that spill (the other formula would show `#SPILL!`).
96/// `ours` are as in [`check_options_spill`]: a model saved at the same place before may be
97/// replaced.
98fn check_column_spills(sheet: &Sheet, column: RangeRef, ours: &[CellRef]) -> Result<()> {
99    for c in column.iter() {
100        if sheet.spill.contains_key(&c) && sheet.spill_anchor(c).is_none_or(|a| !ours.contains(&a)) {
101            return Err(EngineError::Other(format!(
102                "The saved model would cover {}, which holds the values of another formula; choose another place to save it.",
103                c.a1()
104            )));
105        }
106    }
107    Ok(())
108}
109
110/// Refused, unless `overwrite`, when a cell of the model's column holds a value, a formula or
111/// Python code (a model saved there before included): Save Model then asks before it replaces
112/// data, as Excel does. The message carries [`OVERWRITE_ERROR`], which the dialog turns into its
113/// question.
114fn check_column_empty(sheet: &Sheet, column: RangeRef, overwrite: bool) -> Result<()> {
115    let taken = column.iter().find(|c| sheet.cell(*c).is_some_and(Cell::has_content));
116    match taken {
117        Some(c) if !overwrite => Err(EngineError::Other(format!(
118            "The range {} {OVERWRITE_ERROR} ({} isn't empty); pass overwrite: true to replace it.",
119            column.a1(),
120            c.a1()
121        ))),
122        _ => Ok(()),
123    }
124}

126/// `data.solverSaveModel`: writes the model on cells of its sheet, in one step. Saving over a
127/// longer model saved at the same cell clears the rest of it, so Load Model reads the new one.
128pub(super) fn save_model_cells(s: &mut Session, p: &Json) -> Result<Json> {
129    let d = s.doc()?;
130    let params = match p.get("model") {
131        Some(m @ Json::Object(_)) => m.clone(),
132        Some(Json::Null) | None => {
133            saved_model(&d.wb, d.wb.active_sheet()).ok_or_else(|| EngineError::Other("This sheet has no saved Solver model.".into()))?
134        }
135        Some(_) => return Err(bad(SAVE_CMD, "`model` must be data.solver params")),
136    };
137    let model = parse_model(s, &params)?;
138    let to = cell_param_on(s, p, "to", model.sheet)?.ok_or_else(|| bad(SAVE_CMD, "missing `to` (the first cell of the saved model)"))?;
139    let rows = model_formulas(&model)?;
140    let count = u32::try_from(rows.len()).unwrap_or(u32::MAX);
141    let last = to
142        .row
143        .checked_add(count.saturating_sub(1))
144        .filter(|r| *r < crate::base::MAX_ROWS)
145        .ok_or_else(|| EngineError::Other("The saved model doesn't fit below that cell.".into()))?;
146    let range = RangeRef::new(to, CellRef::new(last, to.col));
147    // The model's own cells are never overwritten.
148    let mut used = vec![model.objective];
149    used.extend(&model.vars);
150    for c in &model.cons {
151        used.extend(&c.lhs);
152        if let Rhs::Cells(r) = &c.rhs {
153            used.extend(r);
154        }
155    }
156    if used.iter().any(|c| range.contains(*c)) {
157        return Err(EngineError::Other(format!("{} overlaps the model's cells; choose another place to save it.", range.a1())));
158    }
159    let sheet = model.sheet;
160    let sh = s.doc()?.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
161    // What is left of a model saved here before; the new model's own cells stay.
162    let stale: Vec<CellRef> = saved_model_cells(sh, to).into_iter().filter(|c| !range.contains(*c) && !used.contains(c)).collect();
163    let ours: Vec<CellRef> = range.iter().chain(stale.iter().copied()).collect();
164    let width = rows.last().map_or(1, |f| options_width(f));
165    check_column_spills(sh, range, &ours)?;
166    check_options_spill(sh, CellRef::new(last, to.col), width, &ours, &used)?;
167    // Last, so the question is asked only when replacing the data would succeed.
168    check_column_empty(sh, range, bool_param(p, "overwrite").unwrap_or(false))?;
169    edit(s, |cx| {
170        let sh = cx.sheet_mut(sheet)?;
171        let cells: Vec<CellRef> = range.iter().collect();
172        for (c, f) in cells.iter().zip(&rows) {
173            let mut cell = Cell::formula(crate::model::Formula::new(f));
174            cell.style = sh.style_id(*c);
175            sh.set_cell(*c, cell);
176        }
177        for c in &stale {
178            let style = sh.style_id(*c);
179            sh.set_cell(*c, Cell { value: Value::Empty, formula: None, style, py: None });
180        }
181        for c in cells.into_iter().chain(stale) {
182            cx.touch(sheet, c);
183        }
184        Ok(())
185    })?;
186    Ok(json!({"range": range.a1()}))
187}

189/// The formulas of a model range, in order. Every cell up to the last formula must hold one: a
190/// cell without a formula there (a constraint overwritten by a constant, say) would shift the
191/// rows below it and drop a constraint or the options. Empty cells after the last formula are
192/// allowed.
193fn cells_with_formulas(r: RangeRef, formula: impl Fn(CellRef) -> Option<String>) -> Result<Vec<String>> {
194    let cells: Vec<(CellRef, Option<String>)> = r.iter().map(|c| (c, formula(c))).collect();
195    let end = cells.iter().rposition(|(_, f)| f.is_some()).map_or(0, |i| i + 1);
196    let mut rows = Vec::with_capacity(end);
197    for (c, f) in cells.into_iter().take(end) {
198        let f = f.ok_or_else(|| EngineError::Other(format!("{} holds no formula, so the range isn't a model saved with Save Model.", c.a1())))?;
199        rows.push(f);
200    }
201    Ok(rows)
202}

204/// A model read down from `start` must end with its options row, which Save Model always
205/// writes last: a cell without a formula before it (a constraint cleared, say) would otherwise
206/// end the read early and load the rows above it as the whole model, with default options.
207fn complete_from_cell(start: CellRef, down: &[(CellRef, String)]) -> Result<()> {
208    match down.last() {
209        Some((_, f)) if is_options_row(f) => Ok(()),
210        _ if down.len() >= MAX_MODEL_ROWS => Err(EngineError::Other(format!(
211            "No options row ends the model within {MAX_MODEL_ROWS} cells of {}, so it isn't a model saved with Save Model.",
212            start.a1()
213        ))),
214        last => {
215            let gap = last.map_or(Some(start), |(c, _)| c.row.checked_add(1).map(|row| CellRef::new(row, c.col)));
216            Err(EngineError::Other(match gap {
217                Some(gap) => format!(
218                    "{} holds no formula before the model's options row, so {} doesn't start a complete model saved with Save Model.",
219                    gap.a1(),
220                    start.a1()
221                ),
222                None => format!("{} doesn't start a complete model saved with Save Model.", start.a1()),
223            }))
224        }
225    }
226}

228/// `data.solverLoadModel`: makes a model written by Save Model the active sheet's model.
229pub(super) fn load_model_cells(s: &mut Session, p: &Json) -> Result<Json> {
230    let d = s.doc()?;
231    let active = d.wb.active_sheet();
232    let text = str_param(p, "from").ok_or_else(|| bad(LOAD_CMD, "missing `from` (the saved model's range)"))?;
233    let (sh, body) = split_sheet(text);
234    if sh.is_some_and(|n| d.wb.sheet_index(&n) != Some(active)) {
235        return Err(EngineError::Other("Load Model reads a model saved on the active sheet.".into()));
236    }
237    let r = RangeRef::parse(body.trim()).ok_or_else(|| bad(LOAD_CMD, format!("not a range: `{text}`")))?;
238    let cap = MAX_MODEL_ROWS;
239    let sheet = d.wb.sheet(active).ok_or(EngineError::NoDocument)?;
240    let formula = |c: CellRef| sheet.cell(c).and_then(|x| x.formula.as_ref()).map(|f| f.text.clone());
241    let rows: Vec<String> = if r.is_single() {
242        let down = formulas_down(sheet, r.start);
243        complete_from_cell(r.start, &down)?;
244        down.into_iter().map(|(_, f)| f).collect()
245    } else {
246        if r.count() > cap as u64 {
247            return Err(EngineError::Other(format!("The range is too large for a saved model (at most {cap} cells).")));
248        }
249        let rows = cells_with_formulas(r, formula)?;
250        if !rows.last().is_some_and(|f| is_options_row(f)) {
251            return Err(EngineError::Other(format!(
252                "{} doesn't end with the model's options row, so it isn't a complete model saved with Save Model.",
253                r.a1()
254            )));
255        }
256        rows
257    };
258    let params = parse_model_formulas(&rows)?;
259    let model = parse_model(s, &params)?;
260    if model.sheet != active {
261        return Err(EngineError::Other("Load Model reads a model of the active sheet.".into()));
262    }
263    edit(s, |cx| {
264        save_model(&mut cx.wb, &model);
265        Ok(())
266    })?;
267    Ok(json!({"model": saved_model(&s.doc()?.wb, active)}))
268}

270/// Whether a defined name belongs to the saved model of `sheet`.
271fn is_sheet_model_name(n: &DefinedName, sheet: usize) -> bool {
272    n.scope == Some(sheet) && is_model_name(&n.name)
273}

275/// `data.solverReset`: Reset All removes the active sheet's model. A model the sheet reads from
276/// workbook-level names (which other sheets may read too) is left in place and masked by an
277/// empty sheet-level `solver_opt`, so no other sheet's model changes.
278pub(super) fn reset_model(s: &mut Session, _p: &Json) -> Result<Json> {
279    let d = s.doc()?;
280    let active = d.wb.active_sheet();
281    let removed = d.wb.names.iter().filter(|n| is_sheet_model_name(n, active)).count();
282    let mut after = d.wb.names.clone();
283    after.retain(|n| !is_sheet_model_name(n, active));
284    let masked = {
285        let mut probe = (*d.wb).clone();
286        probe.names = after;
287        saved_model(&probe, active).is_some()
288    };
289    if removed > 0 || masked {
290        edit(s, |cx| {
291            cx.wb.names.retain(|n| !is_sheet_model_name(n, active));
292            if masked {
293                cx.wb.names.push(DefinedName {
294                    name: "solver_opt".into(),
295                    scope: Some(active),
296                    formula: "\"\"".into(),
297                    comment: String::new(),
298                    hidden: true,
299                });
300            }
301            Ok(())
302        })?;
303    }
304    Ok(json!({"removed": removed, "masked": masked}))
