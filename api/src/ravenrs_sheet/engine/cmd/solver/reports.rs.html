29/// The `reports` list (`None` when absent).
30pub(super) fn report_kinds(cmd: &'static str, p: &Json) -> Result<Option<Vec<ReportKind>>> {
31    let Some(list) = p.get("reports") else { return Ok(None) };
32    let list = list.as_array().ok_or_else(|| bad(cmd, "`reports` must be an array of answer, sensitivity and limits"))?;
33    let mut kinds = Vec::new();
34    for k in list {
35        let kind = match k.as_str().map(|t| t.trim().to_ascii_lowercase()).as_deref() {
36            Some("answer") => ReportKind::Answer,
37            Some("sensitivity") => ReportKind::Sensitivity,
38            Some("limits") => ReportKind::Limits,
39            _ => return Err(bad(cmd, format!("unknown report {k} (answer, sensitivity or limits)"))),
40        };
41        if !kinds.contains(&kind) {
42            kinds.push(kind);
43        }
44    }
45    Ok(Some(kinds))
46}

55/// A cell's name in the reports: the nearest text to its left and the nearest text above it.
56pub(super) fn cell_name(sh: &Sheet, c: CellRef) -> String {
57    let text = |c: CellRef| match sh.value(c) {
58        Value::Text(t) if !t.trim().is_empty() => Some(t.to_string()),
59        _ => None,
60    };
61    let left = (0..c.col).rev().find_map(|col| text(CellRef::new(c.row, col)));
62    let top = (0..c.row).rev().find_map(|row| text(CellRef::new(row, c.col)));
63    [left, top].into_iter().flatten().collect::<Vec<_>>().join(" ")
64}

74/// Header labels in bold from column 1, then the rows below them; returns the next free row.
75pub(super) fn table(out: &mut Lines, row: u32, headers: &[&str], rows: Vec<Vec<Value>>) -> u32 {
76    for (k, h) in headers.iter().enumerate() {
77        out.push((row, 1 + k as u32, Value::text(*h), true));
78    }
79    let mut r = row;
80    for cells in rows {
81        r = r.saturating_add(1);
82        for (k, v) in cells.into_iter().enumerate() {
83            out.push((r, 1 + k as u32, v, false));
84        }
85    }
86    r.saturating_add(2)
87}

105fn format_constraint_row(sh: &Sheet, c: &Constraint, cell: CellRef, v: f64, rhs: &[f64], k: usize) -> Vec<Value> {
106    let (formula, status, slack) = match (c.rel, &c.rhs) {
107        (Rel::Int, _) => (format!("{}=Integer", abs_ref(cell)), "Binding", 0.0),
108        (Rel::Bin, _) => (format!("{}=Binary", abs_ref(cell)), "Binding", 0.0),
109        (rel, rhs_cells) => {
110            let b = rhs.get(k).or_else(|| rhs.first()).copied().unwrap_or(f64::NAN);
111            let rhs_text = match rhs_cells {
112                Rhs::Cells(cells) => cells.get(k).or_else(|| cells.first()).map(|c| abs_ref(*c)).unwrap_or_default(),
113                _ => crate::base::number_to_text(b),
114            };
115            let slack = (v - b).abs();
116            let binding = rel == Rel::Eq || slack <= PRECISION * (1.0 + b.abs());
117            let slack = if binding { 0.0 } else { slack };
118            (format!("{}{}{rhs_text}", abs_ref(cell), rel.text()), if binding { "Binding" } else { "Not Binding" }, slack)
119        }
120    };
121    let num = |v: f64| if v.is_finite() { Value::number(v) } else { Value::Error(crate::base::CellError::Value) };
122    vec![Value::text(abs_ref(cell)), Value::text(cell_name(sh, cell)), num(v), Value::text(formula), Value::text(status), num(slack)]
123}

125fn answer_report(d: &DocState, sh: &Sheet, inp: &ReportInput) -> Lines {
126    let m = inp.model;
127    let mut trial = Trial::new(d);
128    let read = |trial: &mut Trial, x: &[f64], cells: &[CellRef]| -> Vec<f64> {
129        let sets: Vec<(usize, CellRef, Value)> = m.vars.iter().zip(x).map(|(c, v)| (m.sheet, *c, Value::number(*v))).collect();
130        let reads: Vec<(usize, CellRef)> = cells.iter().map(|c| (m.sheet, *c)).collect();
131        trial.run(&sets, &reads).iter().map(as_num).collect()
132    };
133    let f0 = read(&mut trial, &inp.original, &[m.objective]).first().copied().unwrap_or(f64::NAN);
134    // Every cell the constraints read, at the solution.
135    let mut cells = vec![m.objective];
136    for c in &m.cons {
137        cells.extend(&c.lhs);
138        if let Rhs::Cells(r) = &c.rhs {
139            cells.extend(r);
140        }
141    }
142    let vals = read(&mut trial, &inp.solution, &cells);
143    let mut at = vals.iter().copied();
144    let f1 = at.next().unwrap_or(f64::NAN);
145    let num = |v: f64| if v.is_finite() { Value::number(v) } else { Value::Error(crate::base::CellError::Value) };
146    let mut out = Lines::new();
147    let mut r = title(&mut out, ReportKind::Answer, &sh.name);
148    out.push((r, 0, Value::text(format!("Result: {}", inp.message)), false));
149    out.push((r + 1, 0, Value::text(format!("Engine: {}", method_name(m.method))), false));
150    r += 3;
151    let goal = match m.goal {
152        Goal::Max => "Max".to_string(),
153        Goal::Min => "Min".to_string(),
154        Goal::Value(v) => format!("Value Of {v}"),
155    };
156    out.push((r, 0, Value::text(format!("Objective Cell ({goal})")), true));
157    let obj = vec![vec![Value::text(abs_ref(m.objective)), Value::text(cell_name(sh, m.objective)), num(f0), num(f1)]];
158    r = table(&mut out, r + 1, &["Cell", "Name", "Original Value", "Final Value"], obj);
159    out.push((r, 0, Value::text("Variable Cells"), true));
160    let kind = |j: usize| {
161        let bin = m.cons.iter().any(|c| c.rel == Rel::Bin && m.vars.get(j).is_some_and(|v| c.lhs.contains(v)));
162        match (bin, inp.prob.integer.get(j).copied().unwrap_or(false)) {
163            (true, _) => "Binary",
164            (false, true) => "Integer",
165            _ => "Contin",
166        }
167    };
168    let vars = m
169        .vars
170        .iter()
171        .enumerate()
172        .map(|(j, c)| {
173            let (a, b) = (inp.original.get(j).copied().unwrap_or(f64::NAN), inp.solution.get(j).copied().unwrap_or(f64::NAN));
174            vec![Value::text(abs_ref(*c)), Value::text(cell_name(sh, *c)), num(a), num(b), Value::text(kind(j))]
175        })
176        .collect();
177    r = table(&mut out, r + 1, &["Cell", "Name", "Original Value", "Final Value", "Integer"], vars);
178    out.push((r, 0, Value::text("Constraints"), true));
179    let mut rows = Vec::new();
180    for c in &m.cons {
181        let lhs: Vec<f64> = c.lhs.iter().map(|_| at.next().unwrap_or(f64::NAN)).collect();
182        let rhs: Vec<f64> = match &c.rhs {
183            Rhs::Cells(cells) => cells.iter().map(|_| at.next().unwrap_or(f64::NAN)).collect(),
184            Rhs::Const(v) => vec![*v],
185            Rhs::None => vec![],
186        };
187        for (k, (cell, v)) in c.lhs.iter().zip(&lhs).enumerate() {
188            rows.push(format_constraint_row(sh, c, *cell, *v, &rhs, k));
189        }
190    }
191    table(&mut out, r + 1, &["Cell", "Name", "Cell Value", "Formula", "Status", "Slack"], rows);
192    out
193}

195fn limits_report(d: &DocState, sh: &Sheet, inp: &ReportInput) -> Result<Lines> {
196    let m = inp.model;
197    if inp.prob.integer.iter().any(|b| *b) {
198        return Err(EngineError::Other("The Limits report doesn't apply to models with integer or binary constraints.".into()));
199    }
200    if m.method == Method::Evolutionary {
201        return Err(EngineError::Other("The Limits report is available for Simplex LP and GRG Nonlinear.".into()));
202    }
203    let n = m.vars.len();
204    let x = &inp.solution;
205    // Each variable is minimised and maximised with the others held at the solution.
206    let per = (inp.budget / (2 * n as u64).max(1)).max(n as u64 + 3);
207    let mut ev = Evaluator {
208        trial: Trial::new(d),
209        sheet: m.sheet,
210        vars: m.vars.clone(),
211        objective: m.objective,
212        rows: inp.prob.rows.clone(),
213        evals: 0,
214        budget: 0,
215        target: None,
216    };
217    let objective_at = |ev: &mut Evaluator, x: &[f64]| {
218        ev.target = None;
219        ev.budget = ev.evals.saturating_add(1);
220        ev.eval(x).map(|p| p.f).unwrap_or(f64::NAN)
221    };
222    let num = |v: f64| if v.is_finite() { Value::number(v) } else { Value::Error(crate::base::CellError::Value) };
223    let f = objective_at(&mut ev, x);
224    let mut rows = Vec::new();
225    for (j, c) in m.vars.iter().enumerate() {
226        let (mut lo, mut hi) = (x.clone(), x.clone());
227        if let (Some(l), Some(h), Some(pl), Some(ph)) = (lo.get_mut(j), hi.get_mut(j), inp.prob.lo.get(j), inp.prob.hi.get(j)) {
228            *l = *pl;
229            *h = *ph;
230        }
231        let mut ends = Vec::new();
232        for goal in [Goal::Min, Goal::Max] {
233            ev.target = Some(j);
234            ev.evals = 0;
235            ev.budget = per;
236            let solve = optimise(&mut ev, m, goal, x, super::Bounds { lo: &lo, hi: &hi, integer: &inp.prob.integer }, 0.0);
237            ends.push(match solve.outcome {
238                Outcome::Solved(p, _, _) => {
239                    let v = p.get(j).copied().unwrap_or(f64::NAN);
240                    (num(v), num(objective_at(&mut ev, &p)))
241                }
242                Outcome::Unbounded => (Value::text("Unbounded"), Value::Empty),
243                _ => (Value::Error(crate::base::CellError::NA), Value::Empty),
244            });
245        }
246        let mut row = vec![Value::text(abs_ref(*c)), Value::text(cell_name(sh, *c)), num(x.get(j).copied().unwrap_or(f64::NAN))];
247        for (v, o) in ends {
248            row.push(v);
249            row.push(o);
250        }
251        rows.push(row);
252    }
253    let mut out = Lines::new();
254    let r = title(&mut out, ReportKind::Limits, &sh.name);
255    let obj = vec![vec![Value::text(abs_ref(m.objective)), Value::text(cell_name(sh, m.objective)), num(f)]];
256    let r = table(&mut out, r, &["Objective Cell", "Name", "Value"], obj);
257    table(&mut out, r, &["Variable Cell", "Name", "Value", "Lower Limit", "Objective Result", "Upper Limit", "Objective Result"], rows);
258    Ok(out)
259}

261/// Builds every report before the solution or saved model can change.
262/// What happens to a Sensitivity report the model allows but the solution doesn't: `Refuse`
263/// fails the call before anything changes; `Skip` leaves the report out, with its reason, and
264/// writes the others. `optimum` is false when the solve stopped before an optimum, so the
265/// report isn't attempted.
266#[derive(Clone, Copy, Debug, PartialEq)]
267pub(super) enum NotOptimal {
268    Refuse,
269    Skip { optimum: bool },
270}
271
272/// The reports left out under [`NotOptimal::Skip`], as `{report, reason}`.
273pub(super) type Skipped = Vec<Json>;
274
275pub(super) fn prepare_reports(
276    d: &DocState,
277    kinds: &[ReportKind],
278    inp: &ReportInput,
279    not_optimal: NotOptimal,
280) -> Result<(Vec<(String, Lines)>, Skipped)> {
281    if kinds.is_empty() {
282        return Ok((Vec::new(), Vec::new()));
283    }
284    // A request that must write a sheet is refused before any report is built (a Limits report
285    // runs two optimisations per variable); only a Sensitivity report that may be skipped waits
286    // for the check after the loop.
287    let must_write = kinds.iter().any(|k| *k != ReportKind::Sensitivity || not_optimal == NotOptimal::Refuse);
288    if must_write && d.wb.protected_structure {
289        return Err(protected());
290    }
291    // A model that has no Sensitivity report (integer, Evolutionary, Value Of) is refused in
292    // both modes before any report is built: a Limits report listed first would otherwise run
293    // its two optimisations per variable for an error. Only the solution's own refusal (not
294    // optimal) waits for the loop, where `Skip` may leave the report out.
295    if kinds.contains(&ReportKind::Sensitivity)
296        && let Some(why) = sensitivity_refusal(inp.model, inp.prob)
297    {
298        return Err(EngineError::Other(why.into()));
299    }
300    let sh = d.wb.sheet(inp.model.sheet).ok_or(EngineError::NoDocument)?;
301    let mut built = Vec::new();
302    let mut skipped = Vec::new();
303    for k in kinds {
304        let lines = match k {
305            ReportKind::Answer => answer_report(d, sh, inp),
306            ReportKind::Sensitivity => match not_optimal {
307                NotOptimal::Skip { optimum } => {
308                    let got = if optimum {
309                        sensitivity_report(d, sh, inp).map_err(|e| e.to_string())
310                    } else {
311                        Err("Solver did not reach an optimal solution, so the Sensitivity report was not written.".to_string())
312                    };
313                    match got {
314                        Ok(lines) => lines,
315                        Err(reason) => {
316                            skipped.push(json!({"report": "sensitivity", "reason": reason}));
317                            continue;
318                        }
319                    }
320                }
321                NotOptimal::Refuse => sensitivity_report(d, sh, inp)?,
322            },
323            ReportKind::Limits => limits_report(d, sh, inp)?,
324        };
325        let name = (1..=10_000u32)
326            .map(|i| format!("{} Report {i}", k.title()))
327            .find(|n| d.wb.sheet_index(n).is_none() && !built.iter().any(|(b, _): &(String, Lines)| b == n))
328            .ok_or_else(|| EngineError::Other("Too many report sheets.".into()))?;
329        built.push((name, lines));
330    }
331    // Only a report sheet changes the workbook's structure, so a request whose reports were all
332    // skipped still succeeds in a protected workbook.
333    if !built.is_empty() && d.wb.protected_structure {
334        return Err(protected());
335    }
336    Ok((built, skipped))
337}

339fn protected() -> EngineError {
340    EngineError::Other("The workbook is protected and cannot be changed.".into())
341}

343/// Inserts prepared reports before the model's sheet, which stays active.
344pub(super) fn write_reports(cx: &mut Ctx, at: usize, built: Vec<(String, Lines)>) -> Vec<String> {
345    let names: Vec<String> = built.iter().map(|(n, _)| n.clone()).collect();
346    let count = built.len();
347    if count == 0 {
348        return names;
349    }
350    let bold = cx.wb.styles.derive(crate::model::StyleId::DEFAULT, |st| st.font.bold = true);
351    for (i, (name, lines)) in built.into_iter().enumerate() {
352        let mut sheet = Sheet::new(name);
353        for (r, c, v, b) in lines {
354            let style = if b { bold } else { crate::model::StyleId::DEFAULT };
355            sheet.set_cell(CellRef::new(r, c), Cell { value: v, formula: None, style, py: None });
356        }
357        for (c, w) in [(0u32, 14.0f32), (1, 110.0), (2, 140.0), (3, 120.0), (4, 120.0), (5, 155.0), (6, 140.0), (7, 140.0)] {
358            sheet.cols.insert(c, crate::model::LineInfo { size: Some(w), ..Default::default() });
359        }
360        cx.wb.sheets.insert(at + i, std::sync::Arc::new(sheet));
361    }
362    for nm in cx.wb.names.iter_mut() {
363        if let Some(sc) = nm.scope.as_mut()
364            && *sc >= at
365        {
366            *sc += count;
367        }
368    }
369    if cx.wb.active_sheet() >= at {
370        cx.wb.set_active_sheet(cx.wb.active_sheet().saturating_add(count));
371    }
372    cx.structural = true;
373    names
374}

384/// `data.solverReport`: reports for a solution of the sheet's saved model.
385pub(super) fn report(s: &mut Session, p: &Json) -> Result<Json> {
386    let kinds = report_kinds(REPORT_CMD, p)?.ok_or_else(|| bad(REPORT_CMD, "missing `reports` (answer, sensitivity, limits)"))?;
387    if kinds.is_empty() {
388        return Err(bad(REPORT_CMD, "`reports` is empty (answer, sensitivity, limits)"));
389    }
390    let d = s.doc()?;
391    // The given model as is, else the saved model with any params given here over it.
392    let params = match p.get("model") {
393        Some(m @ Json::Object(_)) => m.clone(),
394        Some(Json::Null) | None => {
395            let mut params = saved_model(&d.wb, d.wb.active_sheet()).unwrap_or_else(|| json!({}));
396            if let (Some(base), Some(over)) = (params.as_object_mut(), p.as_object()) {
397                for (k, v) in over {
398                    base.insert(k.clone(), v.clone());
399                }
400            }
401            params
402        }
403        Some(_) => return Err(bad(REPORT_CMD, "`model` must be data.solver params")),
404    };
405    let model = parse_model(s, &params)?;
406    let prob = build_problem(&model);
407    let d = s.doc()?;
408    let sh = d.wb.sheet(model.sheet).ok_or(EngineError::NoDocument)?;
409    let current: Vec<f64> = model.vars.iter().map(|c| as_num(&sh.value(*c))).collect();
410    // `{cell: number}` maps; cells they leave out keep their current values.
411    let point = |key: &str| -> Result<Vec<f64>> {
412        let mut x = current.clone();
413        let Some(map) = p.get(key) else { return Ok(x) };
414        let map = map.as_object().ok_or_else(|| bad(REPORT_CMD, format!("`{key}` must map cells to numbers")))?;
415        for (cell, v) in map {
416            let c = CellRef::parse(&cell.replace('$', "")).ok_or_else(|| bad(REPORT_CMD, format!("not a cell: `{cell}`")))?;
417            let j = model.vars.iter().position(|v| *v == c).ok_or_else(|| bad(REPORT_CMD, format!("`{cell}` is not a variable cell")))?;
418            let v = v.as_f64().filter(|v| v.is_finite()).ok_or_else(|| bad(REPORT_CMD, format!("`{key}.{cell}` must be a number")))?;
419            if let Some(e) = x.get_mut(j) {
420                *e = v;
421            }
422        }
423        Ok(x)
424    };
425    let solution = point("values")?;
426    let original = if p.get("original").is_some() { point("original")? } else { solution.clone() };
427    let budget = model.max_evals;
428    let message = str_param(p, "message").unwrap_or("").chars().take(500).collect();
429    let inp = ReportInput { model: &model, prob: &prob, original, solution, message, budget };
430    let not_optimal = if bool_param(p, "skipNotOptimal").unwrap_or(false) { NotOptimal::Skip { optimum: true } } else { NotOptimal::Refuse };
431    let (built, skipped) = prepare_reports(s.doc()?, &kinds, &inp, not_optimal)?;
432    // Nothing left to write (every report skipped): no edit, so no Undo step.
433    let sheets = if built.is_empty() { Vec::new() } else { edit(s, |cx| Ok(write_reports(cx, model.sheet, built)))? };
434    insert_report_selections(s, model.sheet, sheets.len());
435    let mut out = json!({"sheets": sheets});
436    if !skipped.is_empty() {
437        out["skippedReports"] = json!(skipped);
438    }
439    Ok(out)
