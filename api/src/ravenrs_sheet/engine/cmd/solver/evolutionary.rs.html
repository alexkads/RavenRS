9/// A small deterministic generator (SplitMix64), so a seed reproduces a run on every platform.
10struct Rng(u64);
11
12impl Rng {
13    fn new(seed: u64) -> Rng {
14        Rng(seed ^ 0x2545_F491_4F6C_DD1D)
15    }
16    fn next(&mut self) -> u64 {
17        self.0 = self.0.wrapping_add(0x9E37_79B9_7F4A_7C15);
18        let mut z = self.0;
19        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
20        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
21        z ^ (z >> 31)
22    }
23    /// Uniform in [0, 1).
24    fn uniform(&mut self) -> f64 {
25        (self.next() >> 11) as f64 / (1u64 << 53) as f64
26    }
27    fn between(&mut self, a: f64, b: f64) -> f64 {
28        a + (b - a) * self.uniform()
29    }
30    fn below(&mut self, n: usize) -> usize {
31        if n == 0 { 0 } else { usize::try_from(self.next() % n as u64).unwrap_or(0) }
32    }
33    /// Standard normal (Box–Muller).
34    fn normal(&mut self) -> f64 {
35        let u = self.uniform().max(f64::MIN_POSITIVE);
36        (-2.0 * u.ln()).sqrt() * (std::f64::consts::TAU * self.uniform()).cos()
37    }
38}

40/// A candidate: its point, the objective to minimise and its constraint violation.
41#[derive(Clone, Debug)]
42struct Member {
43    x: Vec<f64>,
44    phi: f64,
45    viol: f64,
46}
47
48impl Member {
49    /// Feasible beats infeasible; then the lower objective, or the smaller violation.
50    fn better(&self, o: &Member, tol: f64) -> bool {
51        match (self.viol <= tol, o.viol <= tol) {
52            (true, false) => true,
53            (false, true) => false,
54            (true, true) => self.phi < o.phi,
55            (false, false) => self.viol < o.viol,
56        }
57    }
58}

60/// Why the Evolutionary search stopped.
61#[derive(Clone, Copy, Debug, PartialEq)]
62pub(super) enum EvoEnd {
63    /// Almost all of the population is within Convergence of the best member.
64    Converged,
65    /// The best member didn't improve for the set number of generations.
66    Stalled,
67    /// The evaluation budget ran out.
68    Budget,
69}

71/// The search box, integer flags and scoring shared by the genetic and pattern phases.
72struct EvoSearch<'a> {
73    ev: &'a mut Evaluator,
74    rows: Vec<Row>,
75    senses: Vec<Sense>,
76    goal: Goal,
77    lo: Vec<f64>,
78    hi: Vec<f64>,
79    integer: &'a [bool],
80    tol: f64,
81}
82
83impl EvoSearch<'_> {
84    /// Clamps into the box and rounds integer variables.
85    fn repair(&self, x: &mut [f64]) {
86        for (j, v) in x.iter_mut().enumerate() {
87            let (l, h) = (self.lo.get(j).copied().unwrap_or(0.0), self.hi.get(j).copied().unwrap_or(0.0));
88            *v = if v.is_finite() { v.clamp(l, h) } else { l };
89            if self.integer.get(j).copied().unwrap_or(false) {
90                *v = v.round().clamp(l.ceil(), h.floor());
91            }
92        }
93    }
94
95    /// Scores `x`; `None` once the budget is spent.
96    fn score(&mut self, x: Vec<f64>) -> Option<Member> {
97        let p = self.ev.eval(&x)?;
98        let (phi, cons) = shape(&p, &self.rows, self.goal);
99        let viol = violation(&cons, &self.senses);
100        let phi = if phi.is_finite() { phi } else { f64::INFINITY };
101        let viol = if viol.is_finite() && cons.iter().all(|c| c.is_finite()) { viol } else { f64::INFINITY };
102        Some(Member { x, phi, viol })
103    }
104}

106/// The Evolutionary method within `lo..hi` from `x0`.
107pub(super) fn evolutionary(
108    ev: &mut Evaluator,
109    x0: &[f64],
110    lo: &[f64],
111    hi: &[f64],
112    integer: &[bool],
113    goal: Goal,
114    o: &EvoOptions,
115) -> (Outcome, EvoEnd) {
116    let rows = ev.rows.clone();
117    let senses: Vec<Sense> = rows.iter().map(|r| r.sense).collect();
118    let scale = rows.iter().map(|r| r.rhs_const.abs()).fold(1.0, f64::max);
119    let mut start = x0.to_vec();
120    project(&mut start, lo, hi);
121    // Without a bound, search within 100 of the start.
122    let blo: Vec<f64> = lo.iter().zip(&start).map(|(l, s)| if l.is_finite() { *l } else { s - 100.0 }).collect();
123    let bhi: Vec<f64> = hi.iter().zip(&start).map(|(h, s)| if h.is_finite() { *h } else { s + 100.0 }).collect();
124    let no_integer = blo.iter().zip(&bhi).zip(integer).any(|((l, h), int)| *int && l.ceil() > h.floor());
125    if blo.iter().zip(&bhi).any(|(l, h)| l > h) || no_integer {
126        return (Outcome::Infeasible, EvoEnd::Converged);
127    }
128    let span: Vec<f64> = blo.iter().zip(&bhi).map(|(l, h)| (h - l).max(1e-9)).collect();
129    let mut se = EvoSearch { ev, rows, senses, goal, lo: blo, hi: bhi, integer, tol: PRECISION * scale };
130    let mut rng = Rng::new(o.seed);
131    let size = o.population.clamp(MIN_POPULATION, MAX_POPULATION);
132    let tol = se.tol;
133    let mut end = EvoEnd::Budget;
134    // The first generation: the start and random points in the box.
135    let mut pop: Vec<Member> = Vec::with_capacity(size);
136    se.repair(&mut start);
137    let mut next_x = Some(start);
138    while pop.len() < size {
139        let x = next_x.take().unwrap_or_else(|| {
140            let mut x: Vec<f64> = se.lo.iter().zip(&se.hi).map(|(l, h)| rng.between(*l, *h)).collect();
141            se.repair(&mut x);
142            x
143        });
144        match se.score(x) {
145            Some(m) => pop.push(m),
146            None => break,
147        }
148    }
149    let mut best: Option<Member> = None;
150    for m in &pop {
151        if best.as_ref().is_none_or(|b| m.better(b, tol)) {
152            best = Some(m.clone());
153        }
154    }
155    let mut last_improved = 0u64;
156    let mut generation = 0u64;
157    'search: while pop.len() == size {
158        let Some(b) = best.clone() else { break };
159        // Converged: 99% of the population is feasible and within Convergence of the best.
160        let close = pop.iter().filter(|m| m.viol <= tol && (m.phi - b.phi).abs() <= o.convergence * b.phi.abs().max(1.0)).count();
161        if b.viol <= tol && generation >= 5 && close * 100 >= size * 99 {
162            end = EvoEnd::Converged;
163            break;
164        }
165        if generation.saturating_sub(last_improved) >= o.stall {
166            end = EvoEnd::Stalled;
167            break;
168        }
169        // The next generation: the best survives, the rest are children of two tournaments.
170        let mut next = Vec::with_capacity(size);
171        next.push(b.clone());
172        while next.len() < size {
173            let mut parent = || {
174                let (i, k) = (rng.below(size), rng.below(size));
175                match (pop.get(i), pop.get(k)) {
176                    (Some(p), Some(q)) => Some(if q.better(p, tol) { q.x.clone() } else { p.x.clone() }),
177                    _ => None,
178                }
179            };
180            let (Some(pa), Some(pb)) = (parent(), parent()) else { break 'search };
181            let mut x: Vec<f64> = pa
182                .iter()
183                .zip(&pb)
184                .zip(&span)
185                .map(|((a, c), w)| {
186                    let mut v = a + rng.between(-0.25, 1.25) * (c - a);
187                    if rng.uniform() < o.mutation {
188                        v += 0.1 * w * rng.normal();
189                    }
190                    v
191                })
192                .collect();
193            se.repair(&mut x);
194            match se.score(x) {
195                Some(m) => next.push(m),
196                None => break 'search,
197            }
198        }
199        generation += 1;
200        for m in &next {
201            let Some(cur) = &best else { continue };
202            if m.better(cur, tol) {
203                let improved = cur.viol > tol || cur.phi - m.phi > 1e-9 * cur.phi.abs().max(1.0);
204                best = Some(m.clone());
205                if improved {
206                    last_improved = generation;
207                }
208            }
209        }
210        pop = next;
211    }
212    // Polish: a compass search on the continuous variables around the best member.
213    if let Some(mut b) = best.clone() {
214        let mut step: Vec<f64> = span.iter().map(|w| 0.01 * w).collect();
215        let n = b.x.len();
216        'polish: for _ in 0..10_000 {
217            if step.iter().zip(&span).zip(integer).all(|((s, w), int)| *int || *s <= 1e-9 * w.max(1.0)) {
218                break;
219            }
220            let mut moved = false;
221            for j in 0..n {
222                if integer.get(j).copied().unwrap_or(false) {
223                    continue;
224                }
225                for dir in [1.0, -1.0] {
226                    let mut x = b.x.clone();
227                    if let (Some(v), Some(s)) = (x.get_mut(j), step.get(j)) {
228                        *v += dir * s;
229                    }
230                    se.repair(&mut x);
231                    if x == b.x {
232                        continue;
233                    }
234                    let Some(m) = se.score(x) else { break 'polish };
235                    if m.better(&b, tol) {
236                        b = m;
237                        moved = true;
238                        break;
239                    }
240                }
241            }
242            if !moved {
243                for s in &mut step {
244                    *s *= 0.5;
245                }
246            }
247        }
248        best = Some(b);
249    }
250    match best {
251        Some(b) if b.viol <= tol => (Outcome::Solved(b.x, b.phi, end != EvoEnd::Budget), end),
252        _ if end == EvoEnd::Budget => (Outcome::Limit, end),
253        _ => (Outcome::Infeasible, end),
254    }
255}
