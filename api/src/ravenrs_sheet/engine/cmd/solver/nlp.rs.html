8/// Augmented-Lagrangian value of a point: objective `phi` plus the constraint terms.
9fn aug(phi: f64, cons: &[f64], senses: &[Sense], lam: &[f64], mu: f64) -> f64 {
10    let mut v = phi;
11    for ((c, s), l) in cons.iter().zip(senses).zip(lam) {
12        v += match s {
13            Sense::Eq => l * c + 0.5 * mu * c * c,
14            _ => {
15                let t = (l + mu * c).max(0.0);
16                (t * t - l * l) / (2.0 * mu)
17            }
18        };
19    }
20    v
21}

23/// Constraint values in the `≤ 0` / `= 0` form, and the objective to minimise.
24pub(super) fn shape(p: &Point, rows: &[Row], goal: Goal) -> (f64, Vec<f64>) {
25    let phi = match goal {
26        Goal::Max => -p.f,
27        Goal::Min => p.f,
28        Goal::Value(v) => (p.f - v) * (p.f - v),
29    };
30    let cons = p.g.iter().zip(rows).map(|(g, r)| if r.sense == Sense::Ge { -g } else { *g }).collect();
31    (phi, cons)
32}

34pub(super) fn violation(cons: &[f64], senses: &[Sense]) -> f64 {
35    cons.iter().zip(senses).map(|(c, s)| if *s == Sense::Eq { c.abs() } else { c.max(0.0) }).fold(0.0, f64::max)
36}

38pub(super) fn project(x: &mut [f64], lo: &[f64], hi: &[f64]) {
39    for ((v, l), h) in x.iter_mut().zip(lo).zip(hi) {
40        if *v < *l {
41            *v = *l;
42        }
43        if *v > *h {
44            *v = *h;
45        }
46    }
47}

49/// Local minimisation of the model within bounds `lo..hi` from `x0`.
50pub(super) fn nlp(ev: &mut Evaluator, x0: &[f64], lo: &[f64], hi: &[f64], goal: Goal) -> Outcome {
51    let n = x0.len();
52    let rows = ev.rows.clone();
53    let senses: Vec<Sense> = rows.iter().map(|r| r.sense).collect();
54    let mut x = x0.to_vec();
55    project(&mut x, lo, hi);
56    let mut lam = vec![0.0; rows.len()];
57    let mut mu = 10.0;
58    let Some(p) = ev.eval(&x) else { return Outcome::Limit };
59    let (mut phi, mut cons) = shape(&p, &rows, goal);
60    if !phi.is_finite() {
61        return Outcome::Infeasible;
62    }
63    let phi0 = phi;
64    let mut prev_viol = violation(&cons, &senses);
65    let mut converged = false;
66    let mut out_of_budget = false;
67    let mut carried: Option<f64> = None;
68    for _outer in 0..40 {
69        // Inner: projected gradient on the augmented Lagrangian.
70        let lval = |phi: f64, cons: &[f64], lam: &[f64], mu: f64| aug(phi, cons, &senses, lam, mu);
71        let grad = |ev: &mut Evaluator, x: &[f64], lam: &[f64], mu: f64| -> Option<Vec<f64>> {
72            let mut gr = vec![0.0; n];
73            for j in 0..n {
74                let xj = x.get(j).copied().unwrap_or(0.0);
75                let h = 1e-6 * xj.abs().max(1.0);
76                let (l, u) = (lo.get(j).copied().unwrap_or(f64::NEG_INFINITY), hi.get(j).copied().unwrap_or(f64::INFINITY));
77                let (a, b) = (if xj - h >= l { xj - h } else { xj }, if xj + h <= u { xj + h } else { xj });
78                if b <= a {
79                    continue;
80                }
81                let mut xa = x.to_vec();
82                let mut xb = x.to_vec();
83                if let Some(v) = xa.get_mut(j) {
84                    *v = a;
85                }
86                if let Some(v) = xb.get_mut(j) {
87                    *v = b;
88                }
89                let (pa, ca) = shape(&ev.eval(&xa)?, &rows, goal);
90                let (pb, cb) = shape(&ev.eval(&xb)?, &rows, goal);
91                let d = (lval(pb, &cb, lam, mu) - lval(pa, &ca, lam, mu)) / (b - a);
92                if let Some(v) = gr.get_mut(j) {
93                    *v = if d.is_finite() { d } else { 0.0 };
94                }
95            }
96            Some(gr)
97        };
98        let mut l_cur = lval(phi, &cons, &lam, mu);
99        let Some(mut g) = grad(ev, &x, &lam, mu) else {
100            out_of_budget = true;
101            break;
102        };
103        // The step length carries over between outer iterations once known.
104        let gnorm = g.iter().map(|v| v.abs()).fold(0.0, f64::max);
105        let mut alpha = carried.unwrap_or(if gnorm > 0.0 { 1.0 / gnorm } else { 1.0 });
106        let mut inner_done = false;
107        let mut quiet = 0;
108        for _inner in 0..500 {
109            let mut accepted = None;
110            let mut a = alpha;
111            for _ in 0..40 {
112                let mut xn: Vec<f64> = x.iter().zip(&g).map(|(v, d)| v - a * d).collect();
113                project(&mut xn, lo, hi);
114                let Some(pn) = ev.eval(&xn) else { break };
115                let (phn, cn) = shape(&pn, &rows, goal);
116                let ln = lval(phn, &cn, &lam, mu);
117                let decrease: f64 = g.iter().zip(x.iter().zip(&xn)).map(|(d, (u, v))| d * (u - v)).sum();
118                if ln.is_finite() && ln <= l_cur - 1e-4 * decrease {
119                    accepted = Some((xn, phn, cn, ln));
120                    break;
121                }
122                a *= 0.25;
123            }
124            if ev.exhausted() {
125                out_of_budget = true;
126            }
127            let Some((xn, phn, cn, ln)) = accepted else {
128                inner_done = true;
129                break;
130            };
131            // Variables or objective running off towards infinity: unbounded.
132            if phn.abs() > 1e15 * (1.0 + phi0.abs()) || xn.iter().zip(x0).any(|(v, s)| v.abs() > 1e15 * (1.0 + s.abs())) {
133                return Outcome::Unbounded;
134            }
135            let s: Vec<f64> = xn.iter().zip(&x).map(|(a, b)| a - b).collect();
136            let step = s.iter().map(|v| v.abs()).fold(0.0, f64::max);
137            let xmax = xn.iter().map(|v| v.abs()).fold(1.0, f64::max);
138            let small = step <= 1e-10 * xmax || (l_cur - ln).abs() <= 1e-13 * (1.0 + l_cur.abs());
139            quiet = if small { quiet + 1 } else { 0 };
140            x = xn;
141            phi = phn;
142            cons = cn;
143            l_cur = ln;
144            if quiet >= 3 {
145                inner_done = true;
146                break;
147            }
148            let Some(gn) = grad(ev, &x, &lam, mu) else {
149                out_of_budget = true;
150                break;
151            };
152            // Barzilai–Borwein step.
153            let yv: Vec<f64> = gn.iter().zip(&g).map(|(a, b)| a - b).collect();
154            let sy: f64 = s.iter().zip(&yv).map(|(a, b)| a * b).sum();
155            let ss: f64 = s.iter().map(|v| v * v).sum();
156            alpha = if sy > 0.0 { (ss / sy).clamp(1e-12, 1e12) } else { (a * 4.0).min(1e12) };
157            g = gn;
158            if g.iter().all(|v| *v == 0.0) {
159                inner_done = true;
160                break;
161            }
162        }
163        carried = Some(alpha);
164        if out_of_budget {
165            break;
166        }
167        let viol = violation(&cons, &senses);
168        let scale = rows.iter().map(|r| r.rhs_const.abs()).fold(1.0, f64::max);
169        if viol <= PRECISION * scale && inner_done {
170            converged = true;
171            if rows.is_empty() {
172                break;
173            }
174        }
175        // Multiplier and penalty updates.
176        for ((l, c), s) in lam.iter_mut().zip(&cons).zip(&senses) {
177            *l = match s {
178                Sense::Eq => *l + mu * c,
179                _ => (*l + mu * c).max(0.0),
180            };
181        }
182        if viol > 0.25 * prev_viol {
183            mu = (mu * 10.0).min(1e10);
184        }
185        if converged && viol <= PRECISION * scale * 1e-2 {
186            break;
187        }
188        prev_viol = viol;
189        converged = false;
190    }
191    let viol = violation(&cons, &senses);
192    let scale = rows.iter().map(|r| r.rhs_const.abs()).fold(1.0, f64::max);
193    if viol > PRECISION * scale * 10.0 {
194        return if out_of_budget { Outcome::Limit } else { Outcome::Infeasible };
195    }
196    Outcome::Solved(x, phi, (converged || viol <= PRECISION * scale) && !out_of_budget)
197}
