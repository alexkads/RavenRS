1//! The Sensitivity report. Simplex LP: each variable's reduced cost and the range of its
2//! objective coefficient over which the basis stays optimal, and each constraint's shadow price
3//! and the range of its right side over which the shadow price holds, from the final tableau of
4//! the LP linearised at the solution. GRG Nonlinear: the reduced gradients and the Lagrange
5//! multipliers of the binding constraints at the solution, from difference gradients.

13/// Excel prints an unlimited allowable change as 1E+30.
14const NO_LIMIT: f64 = 1e30;
15
16/// Most arithmetic the Lagrange multipliers of a nonlinear model may take (about a second): a
17/// pass over k binding rows and f free variables costs about 3·k²·f (the normal equations and
18/// each multiplier's noise), and each pass drops at most one row, so a model at the constraint
19/// limit could otherwise take minutes.
20#[cfg(not(test))]
21const MULTIPLIER_WORK: u64 = 1_000_000_000;
22/// A low limit in tests, so the refusal is reachable with a small model.
23#[cfg(test)]
24const MULTIPLIER_WORK: u64 = 20_000;

26/// Why a model has no Sensitivity report (`None` when it has one).
27pub(super) fn sensitivity_refusal(m: &Model, prob: &Problem) -> Option<&'static str> {
28    if prob.integer.iter().any(|b| *b) {
29        Some("The Sensitivity report doesn't apply to models with integer or binary constraints.")
30    } else if m.method == Method::Evolutionary {
31        Some("The Sensitivity report is available for Simplex LP and GRG Nonlinear.")
32    } else if matches!(m.goal, Goal::Value(_)) {
33        Some("The Sensitivity report needs an objective to maximise or minimise.")
34    } else {
35        None
36    }
37}

39/// A sensitivity figure: values within `noise` of zero (rounding) print as 0, unlimited ones
40/// as 1E+30. The noise level is local to the figure: it comes from the terms the figure is a
41/// sum of (the rounding error of a sum scales with the sum of its terms' sizes), never from
42/// the model's largest coefficient or gradient component, so a small price or gradient keeps
43/// its value beside a large unrelated one, and never from the objective's value, so adding a
44/// constant to the objective changes nothing the report shows.
45fn sens_num(v: f64, noise: f64) -> Value {
46    if !v.is_finite() || v >= NO_LIMIT {
47        Value::number(NO_LIMIT)
48    } else if v.abs() <= noise {
49        Value::number(0.0)
50    } else {
51        Value::number(v)
52    }
53}

55fn num(v: f64) -> Value {
56    if v.is_finite() { Value::number(v) } else { Value::Error(crate::base::CellError::Value) }

59/// The smallest ratio `num / den` over the `den` above zero (the step to the first bound).
60fn min_ratio(pairs: impl Iterator<Item = (f64, f64)>) -> f64 {
61    pairs.filter(|(_, den)| *den > EPS).map(|(num, den)| num.max(0.0) / den).fold(f64::INFINITY, f64::min)
62}

64/// Solves the square system `a·x = b` by Gaussian elimination with partial pivoting; unknowns
65/// of a singular system that have no pivot are 0.
66fn solve_dense(mut a: Vec<Vec<f64>>, mut b: Vec<f64>) -> Vec<f64> {
67    let n = b.len();
68    let scale = a.iter().flatten().fold(0.0f64, |m, v| m.max(v.abs())).max(f64::MIN_POSITIVE);
69    let mut pivots: Vec<Option<usize>> = vec![None; n];
70    let mut row = 0;
71    for col in 0..n {
72        let Some(p) = (row..n).max_by(|i, k| {
73            let v = |r: &usize| a.get(*r).and_then(|x| x.get(col)).map_or(0.0, |v| v.abs());
74            v(i).total_cmp(&v(k))
75        }) else {
76            break;
77        };
78        if a.get(p).and_then(|x| x.get(col)).is_none_or(|v| v.abs() <= 1e-12 * scale) {
79            continue;
80        }
81        a.swap(row, p);
82        b.swap(row, p);
83        let (Some(prow), Some(pb)) = (a.get(row).cloned(), b.get(row).copied()) else { break };
84        let pv = prow.get(col).copied().unwrap_or(1.0);
85        for (i, (r, bi)) in a.iter_mut().zip(b.iter_mut()).enumerate() {
86            if i == row {
87                continue;
88            }
89            let f = r.get(col).copied().unwrap_or(0.0) / pv;
90            if f != 0.0 {
91                for (v, p) in r.iter_mut().zip(&prow) {
92                    *v -= f * p;
93                }
94                *bi -= f * pb;
95            }
96        }
97        if let Some(e) = pivots.get_mut(col) {
98            *e = Some(row);
99        }
100        row += 1;
101    }
102    pivots
103        .iter()
104        .enumerate()
105        .map(|(col, p)| match p {
106            Some(r) => b.get(*r).copied().unwrap_or(0.0) / a.get(*r).and_then(|x| x.get(col)).copied().unwrap_or(1.0),
107            None => 0.0,
108        })
109        .collect()
110}

120/// What both report bodies share: the model, the solution, its objective value and the rows'
121/// final values and right sides.
122struct At<'a> {
123    inp: &'a ReportInput<'a>,
124    sh: &'a Sheet,
125    point: Point,
126    finals: Vec<(f64, f64)>,
127}
128
129impl At<'_> {
130    fn var_head(&self, j: usize, c: CellRef) -> Vec<Value> {
131        vec![Value::text(abs_ref(c)), Value::text(cell_name(self.sh, c)), num(self.inp.solution.get(j).copied().unwrap_or(f64::NAN))]
132    }
133    fn row_head(&self, i: usize, r: &Row) -> Vec<Value> {
134        let fin = self.finals.get(i).map_or(f64::NAN, |f| f.0);
135        vec![Value::text(abs_ref(r.lhs)), Value::text(cell_name(self.sh, r.lhs)), num(fin)]
136    }
137}

142pub(super) fn sensitivity_report(d: &DocState, sh: &Sheet, inp: &ReportInput) -> Result<Lines> {
143    let m = inp.model;
144    if let Some(why) = sensitivity_refusal(m, inp.prob) {
145        return Err(EngineError::Other(why.into()));
146    }
147    let n = m.vars.len();
148    let x = &inp.solution;
149    let mut ev = Evaluator {
150        trial: Trial::new(d),
151        sheet: m.sheet,
152        vars: m.vars.clone(),
153        objective: m.objective,
154        rows: inp.prob.rows.clone(),
155        evals: 0,
156        // Room for the gradients (GRG) or two linearisations (Simplex LP) at the least.
157        budget: inp.budget.max(3 * n as u64 + 4),
158        target: None,
159    };
160    let point = ev.eval(x).ok_or_else(out_of_evaluations)?;
161    // The rows' final values and right sides at the solution.
162    let mut cells = Vec::new();
163    for r in &inp.prob.rows {
164        cells.push(r.lhs);
165        cells.extend(r.rhs);
166    }
167    let mut trial = Trial::new(d);
168    let sets: Vec<(usize, CellRef, Value)> = m.vars.iter().zip(x).map(|(c, v)| (m.sheet, *c, Value::number(*v))).collect();
169    let reads: Vec<(usize, CellRef)> = cells.iter().map(|c| (m.sheet, *c)).collect();
170    let mut vals = trial.run(&sets, &reads).into_iter().map(|v| as_num(&v));
171    let finals: Vec<(f64, f64)> = inp
172        .prob
173        .rows
174        .iter()
175        .map(|r| {
176            let l = vals.next().unwrap_or(f64::NAN);
177            (l, if r.rhs.is_some() { vals.next().unwrap_or(f64::NAN) } else { r.rhs_const })
178        })
179        .collect();
180    let at = At { inp, sh, point, finals };
181    let (var_rows, con_rows, var_headers, con_headers) =
182        if m.method == Method::Simplex { simplex_body(&mut ev, &at)? } else { nonlinear_body(&mut ev, &at)? };
183    let mut out = Lines::new();
184    let r = title(&mut out, ReportKind::Sensitivity, &sh.name);
185    out.push((r, 0, Value::text("Variable Cells"), true));
186    let r = table(&mut out, r + 1, &var_headers, var_rows);
187    out.push((r, 0, Value::text("Constraints"), true));
188    table(&mut out, r + 1, &con_headers, con_rows);
189    Ok(out)
190}

192/// Simplex LP: reduced costs, shadow prices and their ranges from the final tableau.
193fn simplex_body(ev: &mut Evaluator, at: &At) -> Result<Body> {
194    let (inp, m) = (at.inp, at.inp.model);
195    // The program the solve built, from the same start values (data.solver's x0, which Solver
196    // Results passes back as `original`), so the solve below repeats its pivots. Start values
197    // the model can't be evaluated at (given only to the report) fall back to the solution.
198    let start: Vec<f64> = inp.original.iter().map(|v| if v.is_finite() { *v } else { 0.0 }).collect();
199    let built = match simplex_program(ev, m.goal, &start) {
200        Err("nonlinear" | "error") if start != inp.solution => simplex_program(ev, m.goal, &inp.solution),
201        other => other,
202    };
203    let SimplexProgram { cf, c, rows, mut work } = built.map_err(|e| {
204        if e == "budget" {
205            out_of_evaluations()
206        } else {
207            EngineError::Other("The model isn't linear in the variable cells, so Simplex LP can't report its sensitivity.".into())
208        }
209    })?;
210    let (outcome, tab) = solve_lp_tableau(&c, &rows, &inp.prob.lo, &inp.prob.hi, &mut work);
211    let (Outcome::Solved(vertex, best, _), Some(tab)) = (outcome, tab) else {
212        return Err(EngineError::Other("Solver could not find an optimal basis for the Sensitivity report.".into()));
213    };
214    // The report describes the optimum, so the values it shows must be one.
215    if !optimal_point(&c, &rows, (&inp.prob.lo, &inp.prob.hi), &inp.solution, best) {
216        return Err(not_optimal());
217    }
218    // The prices and ranges come from the tableau of `vertex`, so the values shown must be that
219    // vertex. Another optimum with the same objective (when the optimum is not unique) has other
220    // ranges, so it is refused too, although it is optimal.
221    let near = |a: f64, b: f64| (a - b).abs() <= super::PRECISION * (1.0 + a.abs().max(b.abs()));
222    if vertex.len() != inp.solution.len() || !vertex.iter().zip(&inp.solution).all(|(a, b)| near(*a, *b)) {
223        return Err(EngineError::Other(
224            "The Sensitivity report needs the values at the Simplex LP solution: these values are another optimal solution, whose ranges differ. Solve the model with Simplex LP first."
225                .into(),
226        ));
227    }
228    // Minimising `sign · objective`.
229    let sign = if m.goal == Goal::Max { -1.0 } else { 1.0 };
230    // A shadow price is the dual yᵢ = Σᵣ c_B(r)·B⁻¹(r, i): the row's unit column of the final
231    // tableau holds B⁻¹'s column i, and c_B(r) is the cost of the column basic in row r (a
232    // slack or artificial costs nothing). The rounding error of a sum scales with the sum of
233    // its terms' sizes, so a price below 1E-9 of Σᵣ |c_B(r)·B⁻¹(r, i)| is noise. The bound is
234    // the row's own: a small price keeps its value beside a large coefficient that plays no
235    // part in it, and a constant in the objective (no coefficient) changes no bound. Allowable
236    // changes to a coefficient or a right side are ratios the tableau's own pivot tolerance
237    // (`EPS`) already bounds.
238    let cost_size = |b: usize| tab.cols.get(b).map_or(0.0, |mp| cf.get(mp.var).copied().unwrap_or(0.0).abs());
239    let price_terms = |col: usize| (0..tab.t.len()).map(|r| tab.basis.get(r).map_or(0.0, |b| cost_size(*b)) * entry(&tab, r, col).abs()).sum::<f64>();
240    let (shadow, terms): (Vec<f64>, Vec<f64>) = (0..rows.len())
241        .map(|i| {
242            let col = tab.ident.get(i).copied().unwrap_or(usize::MAX);
243            (sign * flip(&tab, i) * -tab.obj.get(col).copied().unwrap_or(0.0), price_terms(col))
244        })
245        .unzip();
246    let noise = |terms: f64| 1e-9 * terms.max(f64::MIN_POSITIVE);
247    let con_rows = ev
248        .rows
249        .iter()
250        .enumerate()
251        .map(|(i, r)| {
252            // The basic values move along the row's unit column as its right side changes.
253            let col = tab.ident.get(i).copied().unwrap_or(usize::MAX);
254            let dir: Vec<(f64, f64)> =
255                (0..tab.t.len()).filter(|r| real_basic(&tab, *r)).map(|r| (entry(&tab, r, tab.width), flip(&tab, i) * entry(&tab, r, col))).collect();
256            // An artificial still basic after phase one stands for a row the others make
257            // redundant, and it is zero there. When this right side moves it, any change alone
258            // leaves no solution, so both ranges are 0.
259            let pinned = (0..tab.t.len()).any(|r| !real_basic(&tab, r) && entry(&tab, r, col).abs() > EPS);
260            let (increase, decrease) =
261                if pinned { (0.0, 0.0) } else { (min_ratio(dir.iter().map(|(v, d)| (*v, -d))), min_ratio(dir.iter().map(|(v, d)| (*v, *d)))) };
262            let b = at.finals.get(i).map_or(f64::NAN, |f| f.1);
263            let mut row = at.row_head(i, r);
264            let price = sens_num(shadow.get(i).copied().unwrap_or(0.0), noise(terms.get(i).copied().unwrap_or(0.0)));
265            row.extend([price, num(b), sens_num(increase, EPS), sens_num(decrease, EPS)]);
266            row
267        })
268        .collect();
269    let nonbasic: Vec<usize> = (0..tab.first_art).filter(|l| !tab.basis.contains(l)).collect();
270    let var_rows = m
271        .vars
272        .iter()
273        .enumerate()
274        .map(|(j, c)| {
275            let coef = cf.get(j).copied().unwrap_or(0.0);
276            let used: f64 = rows.iter().zip(&shadow).map(|((a, _, _), p)| a.get(j).copied().unwrap_or(0.0) * p).sum();
277            // The reduced cost cf_j − Σᵢ a_ij·yᵢ is a sum of cf_j and the prices' terms times
278            // a_ij: the prices' terms rather than the prices, since a price that is a small
279            // difference of large terms carries their rounding error into every reduced cost
280            // it is used in.
281            let used_terms: f64 = rows.iter().zip(&terms).map(|((a, _, _), t)| a.get(j).copied().unwrap_or(0.0).abs() * t).sum();
282            // A change Δ in the coefficient changes each nonbasic reduced cost by Δ·e.
283            let own = |col: usize| tab.cols.get(col).filter(|mp| mp.var == j).map_or(0.0, |mp| mp.sign);
284            let moves: Vec<(f64, f64)> = nonbasic
285                .iter()
286                .map(|&l| {
287                    let through_basis: f64 = (0..tab.t.len()).map(|r| tab.basis.get(r).map_or(0.0, |b| own(*b)) * entry(&tab, r, l)).sum();
288                    (tab.obj.get(l).copied().unwrap_or(0.0), sign * (own(l) - through_basis))
289                })
290                .collect();
291            let increase = min_ratio(moves.iter().map(|(d, e)| (*d, -e)));
292            let decrease = min_ratio(moves.iter().map(|(d, e)| (*d, *e)));
293            let mut row = at.var_head(j, *c);
294            row.extend([sens_num(coef - used, noise(coef.abs() + used_terms)), num(coef), sens_num(increase, EPS), sens_num(decrease, EPS)]);
295            row
296        })
297        .collect();
298    Ok((
299        var_rows,
300        con_rows,
301        vec!["Cell", "Name", "Final Value", "Reduced Cost", "Objective Coefficient", "Allowable Increase", "Allowable Decrease"],
302        vec!["Cell", "Name", "Final Value", "Shadow Price", "Constraint R.H. Side", "Allowable Increase", "Allowable Decrease"],
303    ))
304}

306/// Whether `x` satisfies the LP (rows and bounds) and reaches its optimum `best` (minimising
307/// `c·x`), within the constraint precision.
308fn optimal_point(c: &[f64], rows: &[LpRow], (lo, hi): (&[f64], &[f64]), x: &[f64], best: f64) -> bool {
309    let near = |v: f64, size: f64| v <= super::PRECISION * (1.0 + size);
310    let dot = |a: &[f64]| a.iter().zip(x).fold((0.0, 0.0), |(s, m), (p, q)| (s + p * q, m + (p * q).abs()));
311    let rows_hold = rows.iter().all(|(a, sense, b)| {
312        let (lhs, size) = dot(a);
313        let size = size + b.abs();
314        match sense {
315            Sense::Le => near(lhs - b, size),
316            Sense::Ge => near(b - lhs, size),
317            Sense::Eq => near((lhs - b).abs(), size),
318        }
319    });
320    let bounds_hold = x.iter().enumerate().all(|(j, v)| {
321        let (l, h) = (lo.get(j).copied().unwrap_or(f64::NEG_INFINITY), hi.get(j).copied().unwrap_or(f64::INFINITY));
322        v.is_finite() && near(l - v, v.abs()) && near(v - h, v.abs())
323    });
324    let (value, size) = dot(c);
325    // Two-sided: a point just outside a bound can beat the optimum by far more than the
326    // precision when its costs are large.
327    rows_hold && bounds_hold && near((value - best).abs(), size + best.abs())
328}

343/// Central-difference gradients, inside the bounds, of the objective and of each row's
344/// `lhs − rhs` at the solution.
345fn gradients(ev: &mut Evaluator, inp: &ReportInput) -> Result<(Vec<f64>, Vec<Vec<f64>>)> {
346    let x = &inp.solution;
347    let n = x.len();
348    let mut df = vec![0.0; n];
349    let mut dg = vec![vec![0.0; n]; ev.rows.len()];
350    for j in 0..n {
351        let xj = x.get(j).copied().unwrap_or(0.0);
352        let (l, u) = (inp.prob.lo.get(j).copied().unwrap_or(f64::NEG_INFINITY), inp.prob.hi.get(j).copied().unwrap_or(f64::INFINITY));
353        let h = 1e-6 * xj.abs().max(1.0);
354        let (a, b) = ((xj - h).max(l), (xj + h).min(u));
355        if b <= a {
356            continue;
357        }
358        let mut point = |v: f64| {
359            let mut p = x.clone();
360            if let Some(e) = p.get_mut(j) {
361                *e = v;
362            }
363            ev.eval(&p).ok_or_else(out_of_evaluations)
364        };
365        let (pa, pb) = (point(a)?, point(b)?);
366        if let Some(e) = df.get_mut(j) {
367            *e = (pb.f - pa.f) / (b - a);
368        }
369        for (row, (ga, gb)) in dg.iter_mut().zip(pa.g.iter().zip(&pb.g)) {
370            if let Some(e) = row.get_mut(j) {
371                *e = (gb - ga) / (b - a);
372            }
373        }
374    }
375    Ok((df, dg))
376}

378/// The rounding noise in the Lagrange multiplier of row `i`, one of the `active` rows whose
379/// multipliers `lambda` solve the normal equations of ∇f = Σ λₖ ∇gₖ over the `free` variables.
380/// Row i's equation gives λᵢ = (Σⱼ gᵢⱼ·dfⱼ − Σ_{k≠i} λₖ·Σⱼ gᵢⱼ·gₖⱼ) / Σⱼ gᵢⱼ²; the difference
381/// gradients are good to about 1E-6 of each component, and the rounding error of a sum scales
382/// with the sum of its terms' sizes, so the noise is 1E-6 of the terms' sizes over the row's
383/// own squared norm. The bound is local to the row: another row enters only through its
384/// coupling term with this one, so a row whose gradient is orthogonal to the others has a bound
385/// from its own gradient alone, and a large component of ∇f that the row's gradient doesn't
386/// touch hides nothing. Nearly parallel active gradients make the normal equations
387/// ill-conditioned and amplify every multiplier's error beyond any bound local to a row; such
388/// multipliers are ill-determined whatever the bound, and they print as computed.
389fn multiplier_noise(i: usize, active: &[usize], lambda: &[f64], free: &[usize], (df, dg): (&[f64], &[Vec<f64>])) -> f64 {
390    let grad = |i: usize, j: usize| dg.get(i).and_then(|r| r.get(j)).copied().unwrap_or(0.0);
391    let own: f64 = free.iter().map(|&j| grad(i, j) * grad(i, j)).sum();
392    let objective: f64 = free.iter().map(|&j| (grad(i, j) * df.get(j).copied().unwrap_or(0.0)).abs()).sum();
393    let coupling: f64 = active
394        .iter()
395        .zip(lambda)
396        .filter(|(k, _)| **k != i)
397        .map(|(&k, l)| free.iter().map(|&j| (grad(i, j) * grad(k, j) * l).abs()).sum::<f64>())
398        .sum();
399    1e-6 * (objective + coupling) / own.max(f64::MIN_POSITIVE)
400}

402/// Least squares on the free variables: ∇f = Σ λᵢ ∇gᵢ over the `active` rows. An inequality's
403/// multiplier has the sign that tightening it costs (λ ≥ 0 for `lhs ≤ rhs` when maximising); a
404/// row whose multiplier has the other sign beyond its own noise ([`multiplier_noise`], the
405/// criterion the report rounds it with) isn't binding, so it leaves and the rest are solved
406/// again; the objective's value plays no part. Returns each row's multiplier and its noise (0
407/// and 0 for a row that doesn't bind). The work is bounded by `budget` (see
408/// [`MULTIPLIER_WORK`]); past it the report is refused.
409fn multipliers(
410    rows: &[Row],
411    goal: Goal,
412    mut active: Vec<usize>,
413    free: &[usize],
414    (df, dg): (&[f64], &[Vec<f64>]),
415    budget: u64,
416) -> Result<(Vec<f64>, Vec<f64>)> {
417    let grad = |i: usize, j: usize| dg.get(i).and_then(|r| r.get(j)).copied().unwrap_or(0.0);
418    let goal_sign = if goal == Goal::Max { 1.0 } else { -1.0 };
419    let mut lambda = vec![0.0; rows.len()];
420    let mut noise = vec![0.0; rows.len()];
421    let mut work = 0u64;
422    while !active.is_empty() {
423        let (k, f) = (active.len() as u64, (free.len() as u64).max(1));
424        work = work.saturating_add(3u64.saturating_mul(k).saturating_mul(k).saturating_mul(f));
425        if work > budget {
426            return Err(EngineError::Other(
427                "The Sensitivity report of this nonlinear model has too many binding constraints to work out in reasonable time.".into(),
428            ));
429        }
430        let normal: Vec<Vec<f64>> =
431            active.iter().map(|&i| active.iter().map(|&k| free.iter().map(|&j| grad(i, j) * grad(k, j)).sum()).collect()).collect();
432        let rhs: Vec<f64> = active.iter().map(|&i| free.iter().map(|&j| grad(i, j) * df.get(j).copied().unwrap_or(0.0)).sum()).collect();
433        let solved = solve_dense(normal, rhs);
434        let bounds: Vec<f64> = active.iter().map(|&i| multiplier_noise(i, &active, &solved, free, (df, dg))).collect();
435        let wrong = active
436            .iter()
437            .zip(&solved)
438            .zip(&bounds)
439            .enumerate()
440            .filter_map(|(k, ((&i, l), bound))| {
441                let want = match rows.get(i).map_or(Sense::Eq, |r| r.sense) {
442                    Sense::Le => goal_sign,
443                    Sense::Ge => -goal_sign,
444                    Sense::Eq => return None,
445                };
446                (want * l < -bound).then_some((k, want * l))
447            })
448            .min_by(|a, b| a.1.total_cmp(&b.1));
449        match wrong {
450            Some((k, _)) => {
451                active.remove(k);
452            }
453            None => {
454                for ((i, l), bound) in active.iter().zip(solved).zip(bounds) {
455                    if let (Some(e), Some(n)) = (lambda.get_mut(*i), noise.get_mut(*i)) {
456                        *e = l;
457                        *n = bound;
458                    }
459                }
460                break;
461            }
462        }
463    }
464    Ok((lambda, noise))
465}

467/// Whether the solution is a Karush-Kuhn-Tucker point of the model, from the gradients the
468/// report computed: every row and bound holds, and each reduced gradient (the objective's
469/// gradient less what the binding rows' multipliers explain) is near zero relative to `terms`,
470/// the sizes of the terms it is made of, or, for a variable at a bound, points only out of that
471/// bound in the objective's direction. A binding row whose multiplier had the wrong sign was
472/// dropped by [`multipliers`], so the part of the gradient it would have explained remains in
473/// the reduced gradients and the point is refused here. Every tolerance is local to the row or
474/// variable it tests: a large right side or gradient component elsewhere in the model excuses
475/// nothing here.
476fn kkt_point(at: &At, rows: &[Row], terms: &[f64], reduced: &[f64]) -> bool {
477    let (inp, x) = (at.inp, &at.inp.solution);
478    // Feasibility: each row within ten times the constraint precision of its own right side.
479    // The nonlinear method accepts a solution within that of the model's largest constant right
480    // side, which lets a small row go by its own size; a row over by more than its own
481    // precision isn't satisfied, whatever the other rows' sizes, so the report refuses it.
482    let rows_hold = rows.iter().enumerate().all(|(i, r)| {
483        let (g, b) = (at.point.g.get(i).copied().unwrap_or(f64::NAN), at.finals.get(i).map_or(f64::NAN, |f| f.1));
484        let over = match r.sense {
485            Sense::Le => g,
486            Sense::Ge => -g,
487            Sense::Eq => g.abs(),
488        };
489        over <= 10.0 * super::PRECISION * (1.0 + b.abs())
490    });
491    if !rows_hold || !at.point.f.is_finite() {
492        return false;
493    }
494    // Stationarity: each reduced gradient relative to the terms it is made of (the objective's
495    // component and the multipliers' parts, at least 1). At optima the nonlinear method solves,
496    // a reduced gradient is below about 1E-6 of those terms; 1E-3 leaves a wide margin for the
497    // method's stopping test and for rounding in the difference gradients of badly scaled
498    // models, and still refuses a point whose objective improves visibly along a direction the
499    // constraints allow. The bound is the variable's own: with a bound from the gradient's
500    // largest component, max 1E9·x + y at (1, 0.5) passed, although raising y gains 1 per unit
501    // against a tolerance of 1E6.
502    // Moving up a variable improves a maximum by +r and a minimum by −r per unit.
503    let sign = if inp.model.goal == Goal::Max { 1.0 } else { -1.0 };
504    x.iter().enumerate().all(|(j, &v)| {
505        let (l, h) = (inp.prob.lo.get(j).copied().unwrap_or(f64::NEG_INFINITY), inp.prob.hi.get(j).copied().unwrap_or(f64::INFINITY));
506        let near = super::PRECISION * (1.0 + v.abs());
507        if !v.is_finite() || v < l - near || v > h + near {
508            return false;
509        }
510        // The same test for "at a bound" as the choice of free variables above.
511        let edge = 1e-6 * v.abs().max(1.0);
512        let (at_lo, at_hi) = (v <= l + edge, v >= h - edge);
513        let gain = reduced.get(j).map_or(f64::NAN, |r| sign * r);
514        let tol = 1e-3 * terms.get(j).copied().unwrap_or(0.0).max(1.0);
515        gain.is_finite() && (at_hi || gain <= tol) && (at_lo || gain >= -tol)
516    })
517}

519/// GRG Nonlinear: reduced gradients and the binding rows' Lagrange multipliers.
520fn nonlinear_body(ev: &mut Evaluator, at: &At) -> Result<Body> {
521    let (inp, m) = (at.inp, at.inp.model);
522    let x = &inp.solution;
523    let (df, dg) = gradients(ev, inp)?;
524    // Binding rows (within the nonlinear method's accuracy) and the variables strictly inside
525    // their bounds.
526    let active: Vec<usize> = ev
527        .rows
528        .iter()
529        .enumerate()
530        .filter(|(i, r)| {
531            let (g, b) = (at.point.g.get(*i).copied().unwrap_or(f64::NAN), at.finals.get(*i).map_or(0.0, |f| f.1));
532            r.sense == Sense::Eq || g.abs() <= 1e-4 * (1.0 + b.abs())
533        })
534        .map(|(i, _)| i)
535        .collect();
536    let free: Vec<usize> = (0..x.len())
537        .filter(|&j| {
538            let v = x.get(j).copied().unwrap_or(0.0);
539            let tol = 1e-6 * v.abs().max(1.0);
540            v > inp.prob.lo.get(j).copied().unwrap_or(f64::NEG_INFINITY) + tol && v < inp.prob.hi.get(j).copied().unwrap_or(f64::INFINITY) - tol
541        })
542        .collect();
543    let (lambda, lambda_noise) = multipliers(&ev.rows, m.goal, active, &free, (&df, &dg), MULTIPLIER_WORK)?;
544    let grad = |i: usize, j: usize| dg.get(i).and_then(|r| r.get(j)).copied().unwrap_or(0.0);
545    let reduced: Vec<f64> =
546        (0..x.len()).map(|j| df.get(j).copied().unwrap_or(0.0) - lambda.iter().enumerate().map(|(i, l)| l * grad(i, j)).sum::<f64>()).collect();
547    // A reduced gradient dfⱼ − Σᵢ λᵢ·gᵢⱼ is a sum of the gradient's own component and the
548    // multipliers' terms; the sizes of those terms bound its rounding error and scale the test
549    // of whether it is zero.
550    let terms: Vec<f64> = (0..x.len())
551        .map(|j| df.get(j).copied().unwrap_or(0.0).abs() + lambda.iter().enumerate().map(|(i, l)| (l * grad(i, j)).abs()).sum::<f64>())
552        .collect();
553    // The report describes the optimum, so the values it shows must be one.
554    if !kkt_point(at, &ev.rows, &terms, &reduced) {
555        return Err(not_optimal());
556    }
557    // Difference gradients are good to about 1E-6 of each component, so a reduced gradient's
558    // noise is 1E-6 of its terms' sizes, plus each multiplier's own noise
559    // ([`multiplier_noise`]) times |gᵢⱼ|. The bound is the variable's own: a small gradient
560    // keeps its value beside a large component of another variable.
561    let reduced_noise = |j: usize| {
562        let carried: f64 = lambda_noise.iter().enumerate().map(|(i, n)| if grad(i, j) == 0.0 { 0.0 } else { n * grad(i, j).abs() }).sum();
563        1e-6 * terms.get(j).copied().unwrap_or(0.0) + carried
564    };
565    let var_rows = m
566        .vars
567        .iter()
568        .enumerate()
569        .map(|(j, c)| {
570            let mut row = at.var_head(j, *c);
571            row.push(sens_num(reduced.get(j).copied().unwrap_or(f64::NAN), reduced_noise(j)));
572            row
573        })
574        .collect();
575    let con_rows = ev
576        .rows
577        .iter()
578        .enumerate()
579        .map(|(i, r)| {
580            let mut row = at.row_head(i, r);
581            row.push(sens_num(lambda.get(i).copied().unwrap_or(0.0), lambda_noise.get(i).copied().unwrap_or(0.0)));
582            row
583        })
584        .collect();
585    Ok((var_rows, con_rows, vec!["Cell", "Name", "Final Value", "Reduced Gradient"], vec!["Cell", "Name", "Final Value", "Lagrange Multiplier"]))
586}

588#[cfg(test)]
589mod tests {
590    use super::{Goal, MULTIPLIER_WORK, Row, Sense, multipliers};
591    use crate::base::CellRef;
592
593    #[test]
594    fn multipliers_at_the_constraint_limit_stop_at_the_work_limit() {
595        // 1,000 binding `<=` rows over 200 free variables, each multiplier with the wrong sign:
596        // without a limit every pass would drop one row and solve again, about 4·10⁸ operations
597        // a pass for up to 1,000 passes. The limit (lowered in tests) refuses it at once.
598        let (k, f) = (1000usize, 200usize);
599        let rows: Vec<Row> = (0..k).map(|_| Row { lhs: CellRef::new(0, 0), sense: Sense::Le, rhs: None, rhs_const: 0.0 }).collect();
600        let dg: Vec<Vec<f64>> = (0..k).map(|i| (0..f).map(|j| if j == i % f { 1.0 } else { 0.0 }).collect()).collect();
601        let df = vec![-1.0; f];
602        let free: Vec<usize> = (0..f).collect();
603        let got = multipliers(&rows, Goal::Max, (0..k).collect(), &free, (&df, &dg), MULTIPLIER_WORK);
604        assert!(got.is_err());
605        // A small model within the limit still gets its multipliers: ∇f = (1, 2) on two
606        // binding rows with unit gradients.
607        let dg2 = vec![vec![1.0, 0.0], vec![0.0, 1.0]];
608        let (got, noise) = multipliers(&rows[..2], Goal::Max, vec![0, 1], &free[..2], (&[1.0, 2.0], &dg2), MULTIPLIER_WORK).unwrap();
609        assert_eq!(got, vec![1.0, 2.0]);
610        // Each multiplier's noise is 1E-6 of its own row's term (1 · 1 and 1 · 2 over a unit
611        // norm): the rows are orthogonal, so neither enters the other's bound.
612        assert_eq!(noise, vec![1e-6, 2e-6]);
613    }
