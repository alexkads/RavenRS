231fn block(s: &Session, p: &Json) -> Result<RangeRef> {
232    if p.get("range").is_some() {
233        return target_range(s, p);
234    }
235    let d = s.doc()?;
236    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
237    let sel = d.selection.current();
238    if !sel.is_single() {
239        return Ok(sel);
240    }
241    Ok(current_region(sh, d.selection.active))
242}

244fn chart_areas(s: &Session, p: &Json) -> Result<Vec<RangeRef>> {
245    let mut areas = if p.get("range").is_some() {
246        target_ranges(s, p)?
247    } else if s.doc()?.selection.ranges.len() > 1 {
248        s.doc()?.selection.ranges.clone()
249    } else {
250        vec![block(s, p)?]
251    };
252    // Whole columns/rows span a million cells: clip their far edges to the cells in use, so only
253    // a block that really is too large gets refused. Bounded areas keep their extent as given.
254    let sheet = target_sheet(s, p)?;
255    if let Some(used) = s.doc()?.wb.sheet(sheet).and_then(|sh| sh.used_range()) {
256        // Areas that start outside the used range keep their extent as given.
257        for r in &mut areas {
258            let whole_cols = r.start.row == 0 && r.end.row >= MAX_ROWS - 1;
259            let whole_rows = r.start.col == 0 && r.end.col >= MAX_COLS - 1;
260            if whole_cols && r.start.row <= used.end.row {
261                r.end.row = r.end.row.min(used.end.row);
262            }
263            if whole_rows && r.start.col <= used.end.col {
264                r.end.col = r.end.col.min(used.end.col);
265            }
266        }
267    }
268    Ok(areas)
269}

273fn insert_table(s: &mut Session, p: &Json) -> Result<Json> {
274    let r = block(s, p)?;
275    let sheet = target_sheet(s, p)?;
276    let d = s.doc()?;
277    if d.wb.sheet(sheet).is_some_and(|sh| sh.tables.iter().any(|t| t.range.intersects(&r))) {
278        return Err(EngineError::Other("A table can't overlap another table.".into()));
279    }
280    let arrays = |sh: &crate::model::Sheet| {
281        sh.cells.iter().any(|(_, x)| x.formula.as_ref().and_then(|f| f.array).is_some_and(|a| !a.is_single() && a.intersects(&r)))
282    };
283    if d.wb.sheet(sheet).is_some_and(arrays) {
284        return Err(EngineError::Other("Multi-cell array formulas aren't allowed in tables.".into()));
285    }
286    if r.count() > 50_000_000 {
287        return Err(bad("insert.table", "range too large"));
288    }
289    let header = bool_param(p, "header").unwrap_or_else(|| {
290        let sh = d.wb.sheet(sheet);
291        sh.is_some_and(|sh| (r.start.col..=r.end.col).all(|c| sh.value(CellRef::new(r.start.row, c)).is_text()) && r.height() > 1)
292    });
293    let name = str_param(p, "name")
294        .map(str::to_string)
295        .unwrap_or_else(|| (1..).map(|n| format!("Table{n}")).find(|n| d.wb.table(n).is_none()).unwrap_or_else(|| "Table".into()));
296    if !valid_table_name(&name) {
297        return Err(EngineError::Other("The name that you entered is not valid.".into()));
298    }
299    if d.wb.table(&name).is_some() || d.wb.names.iter().any(|n| n.name.eq_ignore_ascii_case(&name)) {
300        return Err(EngineError::Other("The name entered already exists. Enter a unique name.".into()));
301    }
302    // Without a header row, the data moves down a row to make room for one: never a table with it.
303    let moves_a_table = |sh: &crate::model::Sheet| {
304        sh.tables.iter().any(|t| t.range.end.row >= r.start.row && t.range.start.col <= r.end.col && t.range.end.col >= r.start.col)
305    };
306    if !header && d.wb.sheet(sheet).is_some_and(moves_a_table) {
307        return Err(EngineError::Other("This operation isn't allowed: it would shift cells in a table on your worksheet.".into()));
308    }
309    let style = str_param(p, "style").unwrap_or("TableStyleMedium2").to_string();
310    let id = d.wb.next_object_id();
311    edit(s, |cx| {
312        // Without a header row, insert one with generic names.
313        let r = if header {
314            r
315        } else {
316            let sh = cx.sheet_mut(sheet)?;
317            // The header row pushes the data down one row: never off the end of the sheet.
318            let last = RangeRef::new(CellRef::new(MAX_ROWS - 1, r.start.col), CellRef::new(MAX_ROWS - 1, r.end.col));
319            if sh.cells.iter_range(last).any(|(_, x)| x.has_content()) {
320                return Err(EngineError::Other("To prevent possible loss of data, cells can't be pushed off the end of the worksheet.".into()));
321            }
322            sh.cells.shift_rows_in_cols(r.start.col, r.end.col, r.start.row, 1);
323            // Like Insert Cells > Shift down: move merges, conditional formats, notes, links and pictures, and re-point every formula
324            // reference (on any sheet) at the cells that moved.
325            let e = Edit::InsertCells { axis: Axis::Rows, range: RangeRef::new(r.start, CellRef::new(r.start.row, r.end.col)) };
326            shift_cell_features(sh, &e);
327            let name = sh.name.clone();
328            rewrite(cx, &name, &e)?;
329            let sh = cx.sheet_mut(sheet)?;
330            for (i, c) in (r.start.col..=r.end.col).enumerate() {
331                sh.set_value(CellRef::new(r.start.row, c), crate::base::Value::text(format!("Column{}", i + 1)));
332            }
333            cx.structural = true;
334            RangeRef::new(r.start, CellRef::new((r.end.row + 1).min(MAX_ROWS - 1), r.end.col))
335        };
336        let sh = cx.sheet_mut(sheet)?;
337        let mut names: Vec<String> = Vec::new();
338        // Lower-case names taken, and the next suffix for each header, so that thousands of equal
339        // headers don't take cubic time.
340        let mut taken = std::collections::HashSet::new();
341        let mut next_suffix = std::collections::HashMap::new();
342        for c in r.start.col..=r.end.col {
343            let mut n = sh.value(CellRef::new(r.start.row, c)).display();
344            if n.is_empty() {
345                n = format!("Column{}", c - r.start.col + 1);
346            }
347            let base = n.clone();
348            let key = base.to_ascii_lowercase();
349            let mut k = next_suffix.get(&key).copied().unwrap_or(2);
350            while taken.contains(&n.to_ascii_lowercase()) {
351                n = format!("{base}{k}");
352                k += 1;
353            }
354            next_suffix.insert(key, k);
355            taken.insert(n.to_ascii_lowercase());
356            // Headers are text.
357            sh.set_value(CellRef::new(r.start.row, c), crate::base::Value::text(n.as_str()));
358            names.push(n);
359        }
360        sh.tables.push(Table {
361            id,
362            name: name.clone(),
363            range: r,
364            header_row: true,
365            totals_row: false,
366            columns: names
367                .into_iter()
368                .map(|n| TableColumn { name: n, totals: TotalsFn::None, totals_label: None, formula: None, unique_name: None, xml: None })
369                .collect(),
370            style,
371            banded_rows: true,
372            banded_cols: false,
373            first_col: false,
374            last_col: false,
375            filter_button: true,
376            filter: vec![],
377            table_type: None,
378            connection_id: None,
379        });
380        cx.structural = true;
381        Ok(json!({"table": name, "range": r.a1()}))
382    })
383}

385/// Format as Table inside a table restyles that table, as in Excel; elsewhere it makes a new one.
386fn format_as_table(s: &mut Session, p: &Json) -> Result<Json> {
387    // Without a range, the active area decides (as in insert_table), not the first one.
388    let r = if p.get("range").is_some() { target_range(s, p)? } else { s.doc()?.selection.current() };
389    let sheet = target_sheet(s, p)?;
390    let inside = s.doc()?.wb.sheet(sheet).and_then(|sh| sh.tables.iter().find(|t| t.range.contains_range(&r))).map(|t| (t.name.clone(), t.range));
391    match inside {
392        Some((table, range)) => {
393            table_style(s, &with(p, "table", json!(table)))?;
394            Ok(json!({"table": table, "range": range.a1()}))
395        }
396        None => insert_table(s, p),
397    }
398}

400fn find_table(s: &Session, p: &Json) -> Result<(usize, usize)> {
401    let d = s.doc()?;
402    if let Some(n) = str_param(p, "table") {
403        return d.wb.table(n).ok_or_else(|| bad("table", format!("no table named `{n}`")));
404    }
405    let sheet = d.wb.active_sheet();
406    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
407    let ti = sh
408        .tables
409        .iter()
410        .position(|t| t.range.contains(d.selection.active))
411        .ok_or_else(|| EngineError::Disabled("table".into(), "select a cell in a table".into()))?;
412    Ok((sheet, ti))
413}

415fn table_flag(s: &mut Session, p: &Json, which: &str) -> Result<Json> {
416    let (si, ti) = find_table(s, p)?;
417    let on = bool_param(p, "on");
418    edit(s, |cx| {
419        let sh = cx.sheet_mut(si)?;
420        let Some(t) = sh.tables.get_mut(ti) else { return Ok(Json::Null) };
421        match which {
422            "totals" => {
423                let v = on.unwrap_or(!t.totals_row);
424                if v != t.totals_row {
425                    if v && t.range.end.row.saturating_add(1) >= MAX_ROWS {
426                        return Err(EngineError::Other("There's no room below the table for the total row.".into()));
427                    }
428                    t.totals_row = v;
429                    if v {
430                        t.range.end.row += 1;
431                        if let Some(first) = t.columns.first_mut() {
432                            first.totals_label = Some("Total".into());
433                        }
434                        if t.columns.len() > 1
435                            && let Some(last) = t.columns.last_mut()
436                            && last.totals == TotalsFn::None
437                        {
438                            last.totals = TotalsFn::Sum;
439                        }
440                    } else {
441                        let row = t.range.end.row;
442                        t.range.end.row = t.range.end.row.saturating_sub(1);
443                        let (c0, c1) = (t.range.start.col, t.range.end.col);
444                        for c in c0..=c1 {
445                            sh.remove_cell(CellRef::new(row, c));
446                        }
447                    }
448                    write_totals(sh, ti);
449                    cx.structural = true;
450                }
451            }
452            "bandedRows" => t.banded_rows = on.unwrap_or(!t.banded_rows),
453            "bandedCols" => t.banded_cols = on.unwrap_or(!t.banded_cols),
454            "firstCol" => t.first_col = on.unwrap_or(!t.first_col),
455            "lastCol" => t.last_col = on.unwrap_or(!t.last_col),
456            "header" => {
457                t.filter_button = on.unwrap_or(!t.filter_button);
458                if !t.filter_button {
459                    t.filter.clear();
460                }
461            }
462            _ => {}
463        }
464        Ok(Json::Null)
465    })
466}

468/// Writes the totals row cells (labels and SUBTOTAL formulas).
469fn write_totals(sh: &mut Sheet, ti: usize) {
470    let Some(t) = sh.tables.get(ti).cloned() else { return };
471    if !t.totals_row {
472        return;
473    }
474    let row = t.range.end.row;
475    for (i, col) in t.columns.iter().enumerate() {
476        let c = CellRef::new(row, t.range.start.col + i as u32);
477        if let Some(code) = col.totals.subtotal_code() {
478            let f = format!("=SUBTOTAL({code},{}[{}])", t.name, col.name);
479            sh.set_cell(
480                c,
481                Cell { formula: Some(std::sync::Arc::new(Formula::new(&f))), value: crate::base::Value::Empty, style: sh.style_id(c), py: None },
482            );
483        } else if let Some(l) = &col.totals_label {
484            sh.set_value(c, crate::base::Value::text(l.as_str()));
485        } else {
486            sh.remove_cell(c);
487        }
488    }
489}

491fn table_total_fn(s: &mut Session, p: &Json) -> Result<Json> {
492    let (si, ti) = find_table(s, p)?;
493    let col = str_param(p, "column").ok_or_else(|| bad("table.totalFunction", "missing `column`"))?.to_string();

497        "countNums" => TotalsFn::CountNums,

530/// Table names follow the Microsoft name rules, as the xlsx writer does: letters, digits, `_` and `.`,
531/// starting with a letter, `_` or `\`, and at most 255 characters. Anything else (a hyphen, say) would
532/// read as an operator inside a structured reference, and the writer would rename the table on save.
533fn valid_table_name(name: &str) -> bool {
534    name.chars().count() <= 255
535        && name.starts_with(|c: char| c.is_alphabetic() || c == '_' || c == '\\')
536        && CellRef::parse(name).is_none()
537        && !super::formulas::is_r1c1(name)
538        && name.chars().enumerate().all(|(i, c)| c.is_alphanumeric() || c == '_' || c == '.' || (i == 0 && c == '\\'))
539}

541fn table_rename(s: &mut Session, p: &Json) -> Result<Json> {
542    let (si, ti) = find_table(s, p)?;
543    let name = str_param(p, "name").ok_or_else(|| bad("table.rename", "missing `name`"))?.to_string();
544    if !valid_table_name(&name) {
545        return Err(EngineError::Other("The name that you entered is not valid.".into()));
546    }
547    let wb = &s.doc()?.wb;
548    if wb.table(&name).is_some_and(|x| x != (si, ti)) || wb.names.iter().any(|n| n.name.eq_ignore_ascii_case(&name)) {
549        return Err(EngineError::Other("The name entered already exists. Enter a unique name.".into()));
550    }
551    edit(s, |cx| {
552        let old = cx.wb.sheet(si).and_then(|sh| sh.tables.get(ti)).map(|t| t.name.clone()).unwrap_or_default();
553        if let Some(t) = cx.sheet_mut(si)?.tables.get_mut(ti) {
554            t.name = name.clone();
555        }
556        // Structured references follow the rename.
557        for i in 0..cx.wb.sheets.len() {
558            let keys: Vec<(CellRef, std::sync::Arc<Formula>)> =
559                cx.wb.sheets.get(i).map(|s| s.cells.iter().filter_map(|(c, x)| x.formula.clone().map(|f| (c, f))).collect()).unwrap_or_default();
560            // A bare name that is also a defined name resolves to the defined name, so it is not the table.
561            let shadowed = cx.wb.name(&old, i).is_some();
562            let Some(sh) = cx.wb.sheet_mut(i) else { continue };
563            for (c, f) in keys {
564                let Some(e) = f.expr() else { continue };
565                let ne = e.map(&mut |x| match x {
566                    crate::formula::Expr::Struct(mut st) if st.table.eq_ignore_ascii_case(&old) => {
567                        st.table = name.clone();
568                        crate::formula::Expr::Struct(st)
569                    }
570                    crate::formula::Expr::Name(n) if !shadowed && n.eq_ignore_ascii_case(&old) => crate::formula::Expr::Name(name.clone()),
571                    o => o,
572                });
573                if let Some(cell) = sh.cells.get_mut(c) {
574                    cell.formula = Some(std::sync::Arc::new(Formula::from_expr(ne)));
575                }
576            }
577        }
578        cx.structural = true;
579        Ok(Json::Null)
580    })
581}

583fn table_convert(s: &mut Session, p: &Json) -> Result<Json> {
584    let (si, ti) = find_table(s, p)?;
585    edit(s, |cx| {
586        // Apply the table look as plain formatting, then drop the table.
587        let wb = &mut cx.wb;
588        let Some(t) = wb.sheet(si).and_then(|sh| sh.tables.get(ti)).cloned() else { return Ok(Json::Null) };
589        let looks: Vec<(CellRef, Style)> = t.range.iter().map(|c| (c, crate::engine::tables::table_cell_style(wb, si, &t, c))).collect();

595        }

597            sh.tables.remove(ti);
598        }
599        cx.structural = true;
600        Ok(Json::Null)
601    })

637/// Builds series from a block: first row/column as names/categories when they're text.
638pub fn series_from_range(sh: &Sheet, r: RangeRef, by_rows: bool) -> Vec<Series> {
639    let q = crate::formula::quote_sheet(&sh.name);
640    let abs = |r: RangeRef| {
641        format!("{q}!${}${}:${}${}", crate::base::col_to_letters(r.start.col), r.start.row + 1, crate::base::col_to_letters(r.end.col), r.end.row + 1)
642    };
643    let abs1 = |c: CellRef| format!("{q}!${}${}", crate::base::col_to_letters(c.col), c.row + 1);
644    let is_text = |c: CellRef| sh.value(c).is_text() || sh.value(c).is_empty();
645    let header_row = r.height() > 1
646        && (r.start.col..=r.end.col).any(|c| sh.value(CellRef::new(r.start.row, c)).is_text())
647        && (r.start.col..=r.end.col).filter(|c| *c != r.start.col).all(|c| is_text(CellRef::new(r.start.row, c)));
648    // A blank top-left cell above a header row marks the first column as the category labels even
649    // when it holds numbers (years, X values), the usual spreadsheet convention, so it isn't
650    // plotted as a series of its own (#287).
651    let blank_corner = header_row && sh.value(r.start).is_empty();
652    let label_col =
653        r.width() > 1 && (blank_corner || (r.start.row + header_row as u32..=r.end.row).all(|row| is_text(CellRef::new(row, r.start.col))));
654    let data = RangeRef::new(CellRef::new(r.start.row + header_row as u32, r.start.col + label_col as u32), r.end);
655    let mut out = Vec::new();
656    if !by_rows {
657        for c in data.start.col..=data.end.col {
658            out.push(Series {
659                name: header_row.then(|| abs1(CellRef::new(r.start.row, c))),
660                categories: label_col.then(|| abs(RangeRef::new(CellRef::new(data.start.row, r.start.col), CellRef::new(data.end.row, r.start.col)))),
661                values: abs(RangeRef::new(CellRef::new(data.start.row, c), CellRef::new(data.end.row, c))),
662                bubble_sizes: None,
663                color: None,
664                secondary: false,
665                kind: None,
666                quartiles: Default::default(),
667                smooth: false,
668                trendlines: Vec::new(),
669            });
670        }
671    } else {
672        for row in data.start.row..=data.end.row {
673            out.push(Series {
674                name: label_col.then(|| abs1(CellRef::new(row, r.start.col))),
675                categories: header_row
676                    .then(|| abs(RangeRef::new(CellRef::new(r.start.row, data.start.col), CellRef::new(r.start.row, data.end.col)))),
677                values: abs(RangeRef::new(CellRef::new(row, data.start.col), CellRef::new(row, data.end.col))),
678                bubble_sizes: None,
679                color: None,
680                secondary: false,
681                kind: None,
682                quartiles: Default::default(),
683                smooth: false,
684                trendlines: Vec::new(),
685            });
686        }
687    }
688    out
689}

691fn series_from_areas(sh: &Sheet, areas: &[RangeRef]) -> Result<(RangeRef, bool, Vec<Series>)> {
692    let Some(first) = areas.first().copied() else { return Err(EngineError::Other("Select data for the chart first.".into())) };
693    let bounds = areas.iter().skip(1).fold(first, |r, area| r.union(area));
694    if areas.len() == 1 {
695        let by_rows = first.width() > first.height() + 1;
696        return Ok((bounds, by_rows, series_from_range(sh, first, by_rows)));
697    }
698    let valid_shape = first.width() == 1
699        && first.height() > 1
700        && areas.iter().skip(1).all(|area| area.width() == 1 && area.height() == first.height())
701        && first.iter().all(|cell| sh.value(cell).is_text() || sh.value(cell).is_empty());
702    if !valid_shape {
703        return Err(bad("insert.chart", "multi-area charts require a text category column followed by equal-height series columns"));
704    }
705    let category = series_from_range(sh, first, false).into_iter().next().map(|series| series.values);
706    let mut series = Vec::with_capacity(areas.len() - 1);
707    for area in &areas[1..] {
708        let mut area_series = series_from_range(sh, *area, false);
709        let Some(mut item) = area_series.pop() else {
710            return Err(EngineError::Other("Select data for the chart first.".into()));
711        };
712        if item.name.is_none() {
713            return Err(bad("insert.chart", "multi-area series columns require headers"));
714        }
715        item.categories = category.clone();
716        series.push(item);
717    }
718    Ok((bounds, false, series))
719}

721/// A scatter plot of a block without category labels takes its first column (or row) as the X
722/// values of the others, as Excel does: two plain columns make one series of points, not two (#428).
723pub fn scatter_x_from_first(kind: ChartKind, series: &mut Vec<Series>) {
724    if matches!(kind, ChartKind::Scatter | ChartKind::ScatterLines) && series.len() > 1 && series[0].categories.is_none() {
725        let x = series.remove(0).values;
726        for se in series.iter_mut() {
727            se.categories = Some(x.clone());
728        }
729    }

732/// Most cells a chart's source block may span (bounds the series it builds).
733const MAX_CHART_CELLS: u64 = 1_000_000;
734
735/// Chart types that suit the shape of the data, best first, as `(type, subtype, label)` for
736/// `insert.chart`. Our own heuristic from the shape of the block (series count, points per
737/// series, text categories, negative values), not a copy of Excel's ranking: two numeric columns
738/// without labels → scatter; one short series → column, pie, bar; long series → line; several
739/// series → clustered and stacked columns and bars, line.
740pub fn recommend_charts(s: &Session, p: &Json) -> Result<Vec<(&'static str, &'static str, &'static str)>> {
741    let areas = chart_areas(s, p)?;
742    if areas.iter().fold(0_u64, |count, area| count.saturating_add(area.count())) > MAX_CHART_CELLS {
743        return Err(bad("insert.chart", "range too large for a chart"));
744    }
745    let sheet = target_sheet(s, p)?;
746    let d = s.doc()?;
747    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
748    let (_, by_rows, series) = series_from_areas(sh, &areas)?;
749    let ranges: Vec<RangeRef> = series
750        .iter()
751        .filter_map(|x| match crate::formula::parse(&x.values) {
752            Ok(crate::formula::Expr::Ref(rf)) => Some(rf.range()),
753            _ => None,
754        })
755        .collect();
756    let points = ranges.first().map_or(0, RangeRef::count);
757    let (mut any_number, mut non_negative) = (false, true);
758    for n in ranges.iter().flat_map(|rr| rr.iter()).filter_map(|c| sh.value(c).as_f64()) {
759        any_number = true;
760        non_negative &= n >= 0.0;
761    }
762    if series.is_empty() || !any_number {
763        return Err(EngineError::Other("Select data for the chart first.".into()));
764    }
765    let labelled = series.first().is_some_and(|x| x.categories.is_some());
766    const COLUMN: (&str, &str, &str) = ("column", "", "Clustered Column");
767    const BAR: (&str, &str, &str) = ("bar", "", "Clustered Bar");
768    const LINE: (&str, &str, &str) = ("line", "", "Line");
769    const LINE_MARKERS: (&str, &str, &str) = ("line", "markers", "Line with Markers");
770    const PIE: (&str, &str, &str) = ("pie", "", "Pie");
771    const DOUGHNUT: (&str, &str, &str) = ("doughnut", "", "Doughnut");
772    const AREA: (&str, &str, &str) = ("area", "", "Area");
773    const SCATTER: (&str, &str, &str) = ("scatter", "", "Scatter");
774    const STACKED_COLUMN: (&str, &str, &str) = ("column", "stacked", "Stacked Column");
775    const STACKED_BAR: (&str, &str, &str) = ("bar", "stacked", "Stacked Bar");
776    let mut out = Vec::new();
777    if series.len() == 2 && !labelled && !by_rows {
778        out.extend([SCATTER, LINE_MARKERS, COLUMN]);
779    } else if series.len() == 1 {
780        if points > 12 {
781            out.extend([LINE, COLUMN, AREA]);
782        } else {
783            out.extend([COLUMN, BAR]);
784            if non_negative && points >= 2 {
785                out.extend([PIE, DOUGHNUT]);
786            }
787            out.push(LINE_MARKERS);
788        }
789    } else if points > 12 {
790        out.extend([LINE, COLUMN, AREA, STACKED_COLUMN]);
791    } else {
792        out.extend([COLUMN, BAR, STACKED_COLUMN, LINE_MARKERS, STACKED_BAR]);
793    }
794    Ok(out)
795}

797/// The chart `insert.chart` would insert for these params, without inserting it: the sheet it
798/// goes on and the chart (previews in the Recommended Charts dialog).
799pub fn build_chart(s: &Session, p: &Json) -> Result<(usize, Chart)> {
800    let areas = chart_areas(s, p)?;
801    if areas.iter().fold(0_u64, |count, area| count.saturating_add(area.count())) > MAX_CHART_CELLS {
802        return Err(bad("insert.chart", "range too large for a chart"));
803    }
804    let mut t = str_param(p, "type").unwrap_or("column").to_string();
805    let mut sub = str_param(p, "subtype").unwrap_or("").to_string();
806    if t == "auto" {
807        let first = recommend_charts(s, p)?.into_iter().next().unwrap_or(("column", "", ""));
808        t = first.0.into();
809        sub = first.1.into();
810    }
811    let sheet = target_sheet(s, p)?;
812    let d = s.doc()?;
813    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
814    let kind = chart_kind(&t, &sub);
815    let (bounds, by_rows, mut series) = series_from_areas(sh, &areas)?;
816    if series.is_empty() {
817        return Err(EngineError::Other("Select data for the chart first.".into()));
818    }
819    scatter_x_from_first(kind, &mut series);
820    let id = d.wb.next_object_id();
821    let at = cell_param_on(s, p, "at", sheet)?.unwrap_or_else(|| CellRef::new(bounds.start.row, bounds.end.col + 2));
822    let title = str_param(p, "title").map(str::to_string).or_else(|| {
823        if series.len() == 1 {
824            series[0].name.as_ref().map(|n| crate::calc::evaluate(&d.wb, sheet, at, n).display())
825        } else {
826            Some("Chart Title".into())
827        }
828    });
829    let chart = Chart {
830        id,
831        kind,
832        anchor: Anchor {
833            cell: at,
834            dx: 0.0,
835            dy: 0.0,
836            width: f64_param(p, "width").unwrap_or(480.0) as f32,
837            height: f64_param(p, "height").unwrap_or(288.0) as f32,
838            mode: anchor_mode_param(p).unwrap_or_default(),
839        },
840        title,
841        series,
842        legend: if matches!(kind, ChartKind::Pie | ChartKind::Doughnut) { LegendPos::Right } else { LegendPos::Bottom },
843        data_labels: false,
844        data_table: None,
845        gridlines: !matches!(kind, ChartKind::Pie | ChartKind::Doughnut | ChartKind::Treemap | ChartKind::Sunburst | ChartKind::Funnel),
846        style: 1,
847        hole_size: 75,
848        first_slice_angle: 0,
849        x_title: None,
850        y_title: None,
851        source: (areas.len() == 1).then(|| format!("{}!{}", crate::formula::quote_sheet(&sh.name), bounds.a1())),
852        by_rows,
853        pivot: None,
854    };
855    Ok((sheet, chart))
856}

956/// Shared by Chart Properties and Add Chart Element; no separate UI state.
957pub(super) fn update_data_table(c: &mut Chart, p: &Json, on: Option<bool>) -> Result<()> {
958    if p.get("dataTable").is_some_and(|v| !v.is_boolean()) {
959        return Err(bad("chart", "`dataTable` must be a boolean"));
960    }
961    if on == Some(false) {
962        c.data_table = None;
963        return Ok(());
964    }
965    let fields = ["dataTableKeys", "dataTableHorizontalBorder", "dataTableVerticalBorder", "dataTableOutline"];
966    if on == Some(true) || fields.iter().any(|k| p.get(k).is_some()) {
967        if !c.supports_data_table() {
968            return Err(bad("chart", "data tables need a column, bar, line, area or compatible combo chart"));
969        }
970        let table = c.data_table.get_or_insert_with(Default::default);
971        for (key, value) in
972            fields.into_iter().zip([&mut table.show_keys, &mut table.horizontal_border, &mut table.vertical_border, &mut table.outline])
973        {
974            if let Some(v) = p.get(key) {

1014/// The block a chart's series come from, and whether each series is a row of it: the bounding
1015/// range of the series names, categories and values, when they're all references to one sheet.
1016fn series_block(series: &[Series]) -> Option<(String, bool)> {
1017    let mut sheet: Option<String> = None;
1018    let mut bounds: Option<RangeRef> = None;
1019    let mut rows = 0;
1020    for s in series {
1021        for (i, f) in [Some(&s.values), s.categories.as_ref(), s.name.as_ref()].into_iter().flatten().enumerate() {
1022            let Ok(crate::formula::Expr::Ref(r)) = crate::formula::parse(f) else { return None };
1023            let crate::formula::SheetSel::Named(n) = &r.sheet else { return None };
1024            if !sheet.get_or_insert_with(|| n.clone()).eq_ignore_ascii_case(n) {
1025                return None;
1026            }
1027            let range = r.range();
1028            if i == 0 && range.height() == 1 && range.width() > 1 {
1029                rows += 1;
1030            }
1031            bounds = Some(bounds.map_or(range, |b| b.union(&range)));
1032        }
1033    }
1034    Some((format!("{}!{}", crate::formula::quote_sheet(&sheet?), bounds?.a1()), rows > 0 && rows == series.len()))
1035}

1134/// Validate before mutating the workbook; header dimensions alone do not prove the pixels decode.
1135fn validate_cell_picture(data: &[u8]) -> Result<&'static str> {
1136    if data.len() > 16 * 1024 * 1024 {
1137        return Err(bad("insert.picture", "cell pictures must be at most 16 MiB"));
1138    }
1139    let format = image::guess_format(data).map_err(|_| bad("insert.picture", "choose a valid PNG or JPEG picture"))?;
1140    let mime = match format {
1141        image::ImageFormat::Png => "image/png",
1142        image::ImageFormat::Jpeg => "image/jpeg",
1143        _ => return Err(bad("insert.picture", "cell pictures support PNG and JPEG")),
1144    };
1145    let (width, height) = image::ImageReader::with_format(std::io::Cursor::new(data), format)
1146        .into_dimensions()
1147        .map_err(|_| bad("insert.picture", "invalid image dimensions"))?;
1148    if width == 0 || height == 0 || width > 8192 || height > 8192 || u64::from(width) * u64::from(height) > 16_000_000 {
1149        return Err(bad("insert.picture", "cell pictures are limited to 8192 pixels per side and 16 million pixels"));
1150    }
1151    let mut reader = image::ImageReader::with_format(std::io::Cursor::new(data), format);
1152    let mut limits = image::Limits::default();
1153    limits.max_image_width = Some(8192);
1154    limits.max_image_height = Some(8192);
1155    limits.max_alloc = Some(64 * 1024 * 1024);
1156    reader.limits(limits);
1157    reader.decode().map_err(|_| bad("insert.picture", "picture cannot be decoded within the 64 MiB limit"))?;
1158    Ok(mime)
1159}

1261/// Largest offset (points) `object.move` accepts from the anchor cell.
1262const MAX_OBJECT_OFFSET: f64 = 100_000.0;
1263
1264/// Largest width or height (points) `object.move` and `chart.set` give an object that is not
1265/// already larger: see [`size_cap`].
1266const MAX_OBJECT_SIZE: f64 = 10_000.0;
1267
1268/// The most a command may make a dimension that is `current` now: [`MAX_OBJECT_SIZE`], or the
1269/// current size when a file made the object larger (the XLSX reader accepts anchors far past the
1270/// limit). Such an object keeps its size when it is moved and shrinks by what a resize asks, but
1271/// it never grows further, so hostile input is still capped. Never NaN: `f64::max` drops one.
1272fn size_cap(current: f32) -> f64 {
1273    MAX_OBJECT_SIZE.max(f64::from(current))
1274}

1284        if let Some(v) = f64_param(p, "dx") {

1378/// Sorts a link target as the two pages of Excel's Insert Hyperlink dialog do: "Place in This
1379/// Document" is a sheet reference, a defined name or `#…`; anything else ("Existing File or Web
1380/// Page") is outside the workbook and saved as a relationship.
1381fn links_outside(wb: &Workbook, sheet: usize, target: &str) -> bool {
1382    let t = target.trim();
1383    if t.starts_with('#') || wb.name(t, sheet).is_some() || wb.table(t).is_some() {
1384        return false;
1385    }
1386    let (on, body) = split_sheet(t);
1387    // Sheet names can't hold `: \ / ? * [ ]`, so `https://…!A1` or `C:\…!A1` is no sheet reference,
1388    // and `Book.xlsx#Sheet1!A1` is a place in another file unless a sheet has that name.
1389    let on_sheet = on.is_none_or(|name| {
1390        wb.sheets.iter().any(|s| s.name.eq_ignore_ascii_case(&name))
1391            || (!name.is_empty() && !name.contains([':', '\\', '/', '?', '*', '[', ']', '#']))
1392    });
1393    !(on_sheet && RangeRef::parse(body).is_some())
1394}

1396fn comment(s: &mut Session, p: &Json, threaded: bool) -> Result<Json> {

1398    let at = cell_param_on(s, p, "cell", sheet)?.unwrap_or(s.doc()?.selection.active);
1399    let text = str_param(p, "text").unwrap_or("").to_string();
1400    let named = str_param(p, "author").map(str::to_string);
1401    let author = named.clone().unwrap_or_else(|| s.prefs.user_name.clone());
1402    edit(s, |cx| {
1403        let comments = &mut cx.sheet_mut(sheet)?.comments;
1404        // Writing a note on a cell that has one edits it, as Excel's Edit Note does: the box, the
1405        // replies, the dates and who wrote it stay. A note replaces a comment and vice versa.

1514/// A conditional format's `style` as the model's `Style` JSON (`font`, `fill`, `borders`, `align`,
1515/// `num_fmt`, `protection`, merged over the default by the caller), where a colour may also be
1516/// written as `format_range` takes it (`"#RRGGBB"`, `{theme, tint}`, `"auto"`) and `fill` may be a
1517/// colour alone (a solid fill) or `"none"` (no fill). An unknown key is an error rather than
1518/// dropped: dropped, it left a rule with an empty format that matches and shows nothing.
1519fn cf_style_json(st: &Json) -> Result<Json> {
1520    const KEYS: [&str; 6] = ["font", "fill", "borders", "align", "num_fmt", "protection"];
1521    let err = |m: String| bad("home.conditionalFormat", m);
1522    let Some(obj) = st.as_object() else { return Err(err("`style` must be an object, e.g. {\"fill\": \"#FFC7CE\"}".into())) };
1523    let mut st = obj.clone();
1524    if let Some(k) = st.keys().find(|k| !KEYS.contains(&k.as_str())) {
1525        let hint = if k == "preset" { "; a preset goes in `rule.preset`" } else { "" };
1526        return Err(err(format!("unknown key `{k}` in `style` (known: {}){hint}", KEYS.join(", "))));
1527    }
1528    // A colour written the way the other formatting commands take it, as the model stores it.
1529    let color = |v: &mut Json, what: &str| -> Result<()> {
1530        if v.is_string() || v.get("theme").is_some() {
1531            let c = super::format::color_param(Some(v)).ok_or_else(|| err(format!("invalid colour {v} for `{what}`")))?;
1532            *v = serde_json::to_value(c).map_err(|e| EngineError::Other(e.to_string()))?;
1533        }
1534        Ok(())
1535    };
1536    if let Some(f) = st.get_mut("font").and_then(Json::as_object_mut)
1537        && let Some(c) = f.get_mut("color")
1538    {
1539        color(c, "font.color")?;
1540    }
1541    if let Some(fill) = st.get_mut("fill") {
1542        // `"none"` (or `"auto"`) is no fill, as `home.fillColor` reads it, not a solid automatic one.
1543        if fill.is_string() && super::format::color_param(Some(fill)) == Some(Color::Auto) {
1544            *fill = serde_json::to_value(Fill::default()).map_err(|e| EngineError::Other(e.to_string()))?;
1545        } else if fill.is_string() || fill.get("theme").is_some() {
1546            *fill = json!({"pattern": "Solid", "fg": fill.clone()});
1547        }
1548        if let Some(f) = fill.as_object_mut() {
1549            for k in ["fg", "bg"] {
1550                if let Some(c) = f.get_mut(k) {
1551                    color(c, &format!("fill.{k}"))?;
1552                }
1553            }
1554            // A colour with no pattern is a solid fill, not an invisible one.
1555            if f.contains_key("fg") && !f.contains_key("pattern") {
1556                f.insert("pattern".into(), json!("Solid"));
1557            }
1558        }
1559    }
1560    if let Some(b) = st.get_mut("borders").and_then(Json::as_object_mut) {
1561        for (side, line) in b.iter_mut() {
1562            if let Some(c) = line.get_mut("color") {
1563                color(c, &format!("borders.{side}.color"))?;
1564            }
1565        }
1566    }
1567    Ok(Json::Object(st))
1568}

1570/// #1999: a formula that doesn't parse must never reach the workbook — Excel refuses to open
1571/// files carrying it, even in repair mode. Validated here so the API and the New Rule dialog
1572/// (which dispatches this command) share one gate, the way Excel's object model refuses.
1573fn check_cf_formula(cmd: &str, what: &str, formula: &str) -> Result<()> {
1574    // Strip exactly one leading '=', the way the XLSX writer does: a value like '===A1'
1575    // only parses after several strips and would still be saved unparseable. (#1999 review)
1576    if crate::formula::parse(formula.strip_prefix('=').unwrap_or(formula)).is_err() {
1577        return Err(bad(cmd, format!("invalid {what} formula `{formula}` — it does not parse; Excel would reject the saved file")));
1578    }
1579    Ok(())
1580}

1618        return ok();

1630    let value_kind = |v: Option<&Json>, default: CfValueKind| -> CfValueKind {
1631        match v {

1665                // in the file and in openpyxl's `CellIsRule`) is kept; other text is quoted.

1690            rank: rule.get("rank").and_then(Json::as_u64).unwrap_or(10) as u32,

1693        "aboveAverage" => CfRule::AboveAverage { below: rule.get("below").and_then(Json::as_bool).unwrap_or(false), equal: false, std_dev: 0, style },
1694        "timePeriod" => CfRule::TimePeriod { period: g("period").unwrap_or_else(|| "today".into()), style },

1773        _ => {}

1789    }
1790    edit(s, |cx| {

1792        // The count is taken again after each step: a delete shortens the list.
1793        if let Some(i) = u32_param(p, "delete").map(|v| v as usize)
1794            && i < sh.cond_formats.len()
1795        {
1796            sh.cond_formats.remove(i);
1797        }
1798        if let Some(i) = u32_param(p, "moveUp").map(|v| v as usize)
1799            && i > 0
1800            && i < sh.cond_formats.len()
1801        {
1802            sh.cond_formats.swap(i, i - 1);
1803        }
1804        if let Some(i) = u32_param(p, "moveDown").map(|v| v as usize)
1805            && i + 1 < sh.cond_formats.len()
1806        {
1807            sh.cond_formats.swap(i, i + 1);
1808        }
1809        for (k, cf) in sh.cond_formats.iter_mut().enumerate() {
1810            cf.priority = k as u32 + 1;
1811        }
1812        Ok(Json::Null)
1813    })
