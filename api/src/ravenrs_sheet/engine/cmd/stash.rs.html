1//! GridCraft data kept in hidden workbook-scope defined names, so that it travels inside the xlsx
2//! file.
3//!
4//! An item `key` is stored as names `_gc_<key>_0`, `_gc_<key>_1`, …, each a formula of string
5//! literals joined with `&`: `"…"&"…"`. Excel limits a string literal to 255 characters and a
6//! formula to 8,192, so a long value is split over as many names as it needs and the file stays
7//! one that Excel accepts. Older items (custom views, scripts, the watch list and scenarios) keep
8//! one name each, `_sc_<key>`; reading accepts that form as well, so those items can move to this
9//! format later, and a write of the same key replaces it.
10//!
11//! XML can't carry every character (controls other than tab, line feed and carriage return, and
12//! U+FFFE and U+FFFF), and the xlsx writer drops those. In `_gc_` items each of them, and the
13//! marker itself, is written as [`ESCAPE`] followed by its code as four upper-case hex digits
14//! (`U+0001` becomes `\u{FDD0}0001`), and reading turns them back. The marker is a Unicode
15//! noncharacter, which XML carries but real text does not use, so the escape costs nothing on
16//! ordinary text. Legacy `_sc_` items are read as they are.
17

26const PREFIX: &str = "_gc_";
27const LEGACY_PREFIX: &str = "_sc_";
28/// Characters (UTF-16 units, as Excel counts them) in one string literal.
29const LITERAL_UNITS: usize = 255;
30/// Characters in one name's formula.
31const FORMULA_UNITS: usize = 8_192;
32/// The longest key: `_gc_<key>_<k>` stays well inside Excel's 255-character names.
33const MAX_KEY: usize = 200;
34/// The longest item: 4,194,304 characters.
35pub const MAX_ITEM_CHARS: usize = 4 * 1024 * 1024;
36/// The most bytes the stored (escaped) form of an item can take: an escaped one-byte character
37/// takes seven (the three-byte marker and four digits).
38const MAX_STORED_BYTES: usize = MAX_ITEM_CHARS * 7;
39/// Starts an escaped character in a `_gc_` item: U+FDD0, a noncharacter.
40pub const ESCAPE: char = '\u{FDD0}';
41/// The most items [`items`] and [`keys`] return: the first this many keys in sorted order. A
42/// file can carry any number of hidden names; the rest are not listed.
43pub const MAX_ITEMS: usize = 10_000;
44
45/// The names that hold one item: its `_gc_` chunks by number (the first of each number) and its
46/// legacy `_sc_` name.
47#[derive(Default)]
48struct Parts<'a> {
49    chunks: BTreeMap<u64, &'a str>,
50    legacy: Option<&'a str>,
51}
52
53impl<'a> Parts<'a> {
54    fn add(&mut self, chunk: Option<u64>, formula: &'a str) {
55        match chunk {
56            Some(i) => {
57                self.chunks.entry(i).or_insert(formula);
58            }
59            None => self.legacy = self.legacy.or(Some(formula)),
60        }
61    }
62
63    /// Whether the names make an item: a later chunk on its own does not.
64    fn is_item(&self) -> bool {
65        self.chunks.contains_key(&0) || self.legacy.is_some()
66    }
67}

71    /// Names [`group`] has looked at on this thread, so a test can bound the work of a listing.

75/// The workbook's items whose lower-cased key starts with `prefix` (lower-cased), with the names
76/// that hold them, in one pass over the names.
77fn group<'a>(wb: &'a Workbook, prefix: &str) -> BTreeMap<String, Parts<'a>> {
78    let prefix = prefix.to_ascii_lowercase();
79    let mut items: BTreeMap<String, Parts<'a>> = BTreeMap::new();
80    for n in wb.names.iter() {
81        #[cfg(test)]
82        NAME_VISITS.with(|v| v.set(v.get().saturating_add(1)));
83        let Some((k, chunk)) = item_of(n) else { continue };
84        let k = k.to_ascii_lowercase();
85        if k.starts_with(&prefix) {
86            items.entry(k).or_default().add(chunk, &n.formula);
87        }
88    }
89    items
90}

92/// The item text stored under `key`, or `None` when there is none (or it is over the size limit).
93pub fn get(wb: &Workbook, key: &str) -> Option<String> {
94    let mut parts = Parts::default();
95    for n in wb.names.iter() {
96        if let Some((k, chunk)) = item_of(n)
97            && k.eq_ignore_ascii_case(key)
98        {
99            parts.add(chunk, &n.formula);
100        }
101    }
102    assemble(&parts)
103}

105/// What a listing found: the first [`MAX_ITEMS`] items or keys, and how many more there were.
106#[derive(Clone, Debug, Default, PartialEq, Eq)]
107pub struct Listing<T> {
108    pub found: T,
109    /// Items past the cap, not listed. A caller says so rather than letting them vanish.
110    pub more: usize,
111}

113/// The items whose key starts with `prefix` (`"macro_"`), by lower-cased key, read in one pass
114/// over the names: the first [`MAX_ITEMS`] items that can be read, and the count of the readable
115/// ones past them. An item over the size limit is neither listed nor counted, so it takes no
116/// place another item could have. ([`keys`] lists keys without reading the items.)
117pub fn items(wb: &Workbook, prefix: &str) -> Listing<BTreeMap<String, String>> {
118    let mut listing: Listing<BTreeMap<String, String>> = Listing::default();
119    for (k, parts) in group(wb, prefix) {
120        let Some(text) = parts.is_item().then(|| assemble(&parts)).flatten() else { continue };
121        if listing.found.len() < MAX_ITEMS {
122            listing.found.insert(k, text);
123        } else {
124            listing.more = listing.more.saturating_add(1);
125        }
126    }
127    listing
128}

130/// The text of the item held by `parts`, as [`get`] reads it.
131fn assemble(parts: &Parts) -> Option<String> {
132    let chunks = &parts.chunks;
133    let mut out = String::new();
134    if !chunks.contains_key(&0) {
135        out = string_literals(parts.legacy?);
136    } else {
137        // Chunks run 0, 1, 2, …; anything after a gap isn't part of the item.
138        for (want, (i, f)) in chunks.iter().enumerate() {
139            if *i != want as u64 {
140                break;
141            }
142            out.push_str(&string_literals(f));
143            if out.len() > MAX_STORED_BYTES {
144                return None;
145            }
146        }
147        out = unescape(&out);
148    }
149    (out.chars().count() <= MAX_ITEM_CHARS).then_some(out)
150}

152/// Whether `c` is stored escaped: XML can't carry it, or it is the marker.
153fn needs_escape(c: char) -> bool {
154    c == ESCAPE || !matches!(c, '\t' | '\n' | '\r' | '\u{20}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..='\u{10FFFF}')
155}
156
157/// Turns [`ESCAPE`] and four hex digits back into the character. A marker not followed by four
158/// hex digits naming a character (a hand-made name) stays as it is.
159fn unescape(s: &str) -> String {
160    if !s.contains(ESCAPE) {
161        return s.to_string();
162    }
163    let mut out = String::with_capacity(s.len());
164    let mut rest = s;
165    while let Some(at) = rest.find(ESCAPE) {
166        out.push_str(rest.get(..at).unwrap_or(""));
167        let after = rest.get(at + ESCAPE.len_utf8()..).unwrap_or("");
168        let code = after.get(..4).filter(|h| h.bytes().all(|b| b.is_ascii_hexdigit())).and_then(|h| u32::from_str_radix(h, 16).ok());
169        match code.and_then(char::from_u32) {
170            Some(c) => {
171                out.push(c);
172                rest = after.get(4..).unwrap_or("");
173            }
174            None => {
175                out.push(ESCAPE);
176                rest = after;
177            }
178        }
179    }
180    out.push_str(rest);
181    out
182}

184/// Stores `text` under `key`, replacing what was there (in either format). Empty text removes it.
185pub fn set(wb: &mut Workbook, key: &str, text: &str) -> Result<()> {
186    set_many(wb, &[(key.to_string(), text.to_string())])
187}

189/// Stores every `(key, text)` of `items` as [`set`] would, with the checks done before anything
190/// changes and one pass over the names for all of them, so writing a thousand items into a
191/// workbook with many thousands of names stays near-linear.
192pub fn set_many(wb: &mut Workbook, items: &[(String, String)]) -> Result<()> {
193    let mut planned = Vec::new();
194    let mut keys: HashSet<String> = HashSet::new();
195    for (key, text) in items {
196        check_key(key)?;
197        if text.chars().count() > MAX_ITEM_CHARS {
198            return Err(EngineError::Other(format!(
199                "The data is too large to keep in the workbook ({} MB at most).",
200                MAX_ITEM_CHARS / (1024 * 1024)
201            )));
202        }
203        // Two items under one key (names compare without case) would leave two hidden names of
204        // one name, of which an xlsx file keeps one: a caller's mistake, refused.
205        if !keys.insert(key.to_ascii_lowercase()) {
206            return Err(EngineError::Other(format!("`{key}` is given twice.")));
207        }
208        for (i, formula) in chunk_formulas(text).into_iter().enumerate() {
209            planned.push(DefinedName { name: format!("{PREFIX}{key}_{i}"), scope: None, formula, comment: "GridCraft data".into(), hidden: true });
210        }
211    }
212    // A visible name of an item's shape is someone's own (see `item_of`); a hidden name beside
213    // it with the same name would be the one an xlsx file drops. Refuse before changing anything.
214    let names: HashSet<String> = planned.iter().map(|n| n.name.to_ascii_lowercase()).collect();
215    if let Some(n) = wb.names.iter().find(|n| n.scope.is_none() && !n.hidden && names.contains(&n.name.to_ascii_lowercase())) {
216        return Err(EngineError::Other(format!(
217            "The workbook has a defined name '{}', which GridCraft needs for its own data. Rename it first.",
218            n.name
219        )));
220    }
221    // What was there under these keys goes, in either format, then the new chunks come.
222    wb.names.retain(|n| !item_of(n).is_some_and(|(k, _)| keys.contains(&k.to_ascii_lowercase())));
223    wb.names.extend(planned);
224    Ok(())
225}

241/// The keys of the items stored in the workbook that start with `prefix` (`"macro_"`), sorted
242/// and lower-cased: the first [`MAX_ITEMS`] of them. To read them all, [`items`] takes one pass
243/// where a [`get`] per key would take one each.
244pub fn keys(wb: &Workbook, prefix: &str) -> Listing<Vec<String>> {
245    let mut found: Vec<String> = group(wb, prefix).into_iter().filter(|(_, parts)| parts.is_item()).map(|(k, _)| k).collect();
246    let more = found.len().saturating_sub(MAX_ITEMS);
247    found.truncate(MAX_ITEMS);
248    Listing { found, more }
249}

251/// Whether the workbook holds an item under `name` (compared without case): a hidden workbook-scope
252/// `_gc_<key>_<k>` name, or a workbook-scope legacy `_sc_<key>` name, which is read whether or not
253/// it is hidden. A visible `_gc_` name is the person's own.
254pub fn holds_item(wb: &Workbook, name: &str) -> bool {
255    wb.names.iter().any(|n| n.name.eq_ignore_ascii_case(name) && item_of(n).is_some())
256}

258/// The item a hidden name belongs to, lower-cased: `_gc_<key>_<k>` and `_sc_<key>` give `<key>`.
259pub fn item_key(name: &str) -> Option<String> {
260    parse_name(name).map(|(k, _)| k.to_ascii_lowercase())
261}

263/// The item a name holds a part of: a hidden workbook-scope `_gc_<key>_<k>` name, or a legacy
264/// `_sc_<key>` one. A visible name of the `_gc_` shape is someone's own name, not an item: it is
265/// neither read, nor listed, nor removed on write.
266fn item_of(n: &DefinedName) -> Option<(&str, Option<u64>)> {
267    if n.scope.is_some() {
268        return None;
269    }
270    parse_name(&n.name).filter(|(_, chunk)| chunk.is_none() || n.hidden)
271}

273/// `_gc_<key>_<k>` → (`key`, `Some(k)`); `_sc_<key>` → (`key`, `None`).
274fn parse_name(name: &str) -> Option<(&str, Option<u64>)> {
275    if let Some(rest) = strip_prefix_ci(name, PREFIX) {
276        let (key, k) = rest.rsplit_once('_')?;
277        // Only the chunk number as `set` writes it: `_gc_watch_00` is someone else's name, not
278        // chunk 0 under another spelling, so it is neither read nor in the way of `_gc_watch_0`.
279        let canonical = !k.is_empty() && k.len() <= 9 && k.bytes().all(|b| b.is_ascii_digit()) && (k == "0" || !k.starts_with('0'));
280        if key.is_empty() || !canonical {
281            return None;
282        }
283        return k.parse().ok().map(|i| (key, Some(i)));
284    }
285    strip_prefix_ci(name, LEGACY_PREFIX).filter(|k| !k.is_empty()).map(|k| (k, None))
286}
287
288fn strip_prefix_ci<'a>(s: &'a str, prefix: &str) -> Option<&'a str> {
289    let head = s.get(..prefix.len())?;
290    head.eq_ignore_ascii_case(prefix).then(|| s.get(prefix.len()..)).flatten()
291}

293fn check_key(key: &str) -> Result<()> {
294    let ok = !key.is_empty() && key.len() <= MAX_KEY && key.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'.');
295    if ok { Ok(()) } else { Err(EngineError::Other(format!("`{key}` can't name workbook data: use letters, digits, `_` and `.`"))) }
296}

298/// `text` as name formulas: string literals of at most 255 characters joined with `&`, each formula
299/// at most 8,192 characters, with the characters XML can't carry escaped (see [`ESCAPE`]).
300/// Concatenating [`string_literals`] of each and turning the escapes back gives `text` back.
301pub fn chunk_formulas(text: &str) -> Vec<String> {
302    let mut formulas = Vec::new();
303    let (mut formula, mut units) = (String::new(), 0usize);
304    let (mut literal, mut lit_units) = (String::from("\""), 0usize);
305    let mut flush_literal = |literal: &mut String, lit_units: &mut usize, formula: &mut String, units: &mut usize| {
306        literal.push('"');
307        let cost = literal.encode_utf16().count() + usize::from(!formula.is_empty());
308        if *units + cost > FORMULA_UNITS && !formula.is_empty() {
309            formulas.push(std::mem::take(formula));
310            *units = 0;
311        }
312        if !formula.is_empty() {
313            formula.push('&');
314            *units += 1;
315        }
316        *units += literal.encode_utf16().count();
317        formula.push_str(literal);
318        *literal = String::from("\"");
319        *lit_units = 0;
320    };
321    for ch in text.chars() {
322        // An escaped character is the marker and four digits, kept in one literal.
323        let escaped = needs_escape(ch);
324        let cost = if escaped { ESCAPE.len_utf16() + 4 } else { ch.len_utf16() };
325        if lit_units + cost > LITERAL_UNITS {
326            flush_literal(&mut literal, &mut lit_units, &mut formula, &mut units);
327        }
328        lit_units += cost;
329        if escaped {
330            literal.push(ESCAPE);
331            let _ = write!(literal, "{:04X}", u32::from(ch));
332        } else {
333            if ch == '"' {
334                literal.push('"');
335            }
336            literal.push(ch);
337        }
338    }
339    if lit_units > 0 {
340        flush_literal(&mut literal, &mut lit_units, &mut formula, &mut units);
341    }
342    if !formula.is_empty() {
343        formulas.push(formula);
344    }
345    formulas
346}

348/// Concatenates the string literals of a `"a"&"b""c"` formula (anything outside them is ignored).
349pub fn string_literals(f: &str) -> String {
350    let mut out = String::new();
351    let mut it = f.chars().peekable();
352    let mut in_str = false;
353    while let Some(c) = it.next() {
354        if !in_str {
355            if c == '"' {
356                in_str = true;
357            }
358            continue;
359        }
360        if c == '"' {
361            if it.peek() == Some(&'"') {
362                it.next();
363                out.push('"');
364            } else {
365                in_str = false;
366            }
367        } else {
368            out.push(c);
369        }
370    }
371    out
372}

559    #[test]
560    fn listing_many_items_takes_one_pass_and_is_capped() {
561        // 20,000 hidden items, as a file can carry: a `get` per key looked at every name for each
562        // key (400 million visits); `items` looks at each name once, and lists at most MAX_ITEMS.
563        let mut wb = Workbook::new();
564        let total = 20_000;
565        for i in 0..total {
566            wb.names.push(name(&format!("_gc_macro_{i:05}_0"), &format!("\"m{i}\"")));
567        }
568        set(&mut wb, "macro_big", &"b".repeat(9_000)).unwrap();
569        wb.names.push(name("_sc_macro_legacy", "\"old\""));
570        wb.names.push(name("_gc_macro_orphan_1", "\"no chunk 0\""));
571        wb.names.push(name("_gc_other_0", "\"not a macro\""));
572        NAME_VISITS.with(|v| v.set(0));
573        let listing = items(&wb, "MACRO_");
574        let all = &listing.found;
575        assert_eq!(NAME_VISITS.with(|v| v.get()), wb.names.len(), "one pass over the names");
576        assert_eq!(all.len(), MAX_ITEMS);
577        let first: Vec<&String> = all.keys().take(2).collect();
578        assert_eq!(first, ["macro_00000", "macro_00001"]);
579        assert_eq!(all.get("macro_09999").map(String::as_str), Some("m9999"));
580        assert!(!all.contains_key("macro_10000"), "past the cap");
581        NAME_VISITS.with(|v| v.set(0));
582        let listed = keys(&wb, "macro_");
583        assert_eq!(NAME_VISITS.with(|v| v.get()), wb.names.len());
584        assert_eq!(listed.found, all.keys().cloned().collect::<Vec<_>>(), "keys and items agree");
585        assert_eq!((listing.more, listed.more), (10_002, 10_002), "the rest are counted, not lost");
586
587        // Under the cap every item is there, read as `get` reads it.
588        wb.names.retain(|n| !n.name.starts_with("_gc_macro_0") && !n.name.starts_with("_gc_macro_1"));
589        let few = items(&wb, "macro_").found;
590        let want: BTreeMap<String, String> =
591            [("macro_big", "b".repeat(9_000)), ("macro_legacy", "old".into())].into_iter().map(|(k, t)| (k.to_string(), t)).collect();
592        assert_eq!(few, want);
593        for (k, text) in &few {
594            assert_eq!(get(&wb, k).as_ref(), Some(text));
595        }
596        assert_eq!(keys(&wb, "macro_").found, ["macro_big", "macro_legacy"]);
597        assert_eq!((items(&wb, "").found.len(), items(&wb, "").more), (3, 0), "the other item too, none past the cap");
598    }

621    #[test]
622    fn items_round_trip_through_xlsx() {
623        let mut wb = Workbook::new();
624        let text: String = (0..20_000).map(|i| ['x', '"', 'ü', '\u{1D11E}', ' '][i % 5]).collect();
625        set(&mut wb, "blob", &text).unwrap();
626        wb.names.push(name("_sc_watch", "\"[\"\"A1\"\"]\""));
627        let bytes = crate::xlsx::write_xlsx(&wb).unwrap();
628        let (back, warnings) = crate::engine::io::open_bytes("data.xlsx", &bytes).unwrap();
629        assert!(warnings.is_empty(), "{warnings:?}");
630        assert_eq!(get(&back, "blob").as_deref(), Some(text.as_str()));
631        assert_eq!(get_json(&back, "watch"), Some(serde_json::json!(["A1"])));
632
633        // Characters XML can't carry, and the escape marker itself, come back too, in text and
634        // in JSON (which leaves U+FFFE and U+FFFF unescaped).
635        let mut wb = Workbook::new();
636        let odd: String = (0u32..0x20).filter_map(char::from_u32).chain(['\u{FFFE}', '\u{FFFF}', ESCAPE, 'a', '\u{7F}', '\u{FDD1}']).collect();
637        let text: String = odd.chars().cycle().take(odd.chars().count() * 300).collect();
638        set(&mut wb, "odd", &text).unwrap();
639        excel_limits_hold(&wb);
640        let json = serde_json::json!({"s": "x\u{FFFF}y\u{1}z\u{FDD0}0041"});
641        set_json(&mut wb, "json", &json).unwrap();
642        // All escapes: each literal holds whole escapes, within the limits.
643        let controls = "\u{1}".repeat(9_000);
644        set(&mut wb, "controls", &controls).unwrap();
645        excel_limits_hold(&wb);
646        let bytes = crate::xlsx::write_xlsx(&wb).unwrap();
647        let (back, warnings) = crate::engine::io::open_bytes("data.xlsx", &bytes).unwrap();
648        assert!(warnings.is_empty(), "{warnings:?}");
649        assert_eq!(get(&back, "odd").as_deref(), Some(text.as_str()));
650        assert_eq!(get_json(&back, "json"), Some(json));
651        assert_eq!(get(&back, "controls").as_deref(), Some(controls.as_str()));
652    }

654    #[test]
655    fn escapes_count_against_the_limits_and_odd_markers_read_as_they_are() {
656        // Excel's limits hold however many characters are escaped, and the longest item of
657        // characters that all need escaping still reads.
658        for len in [1, 50, 51, 52, 1_638, 1_639, 5_000] {
659            let text = "\u{2}".repeat(len);
660            let fs = chunk_formulas(&text);
661            assert_eq!(unescape(&fs.iter().map(|f| string_literals(f)).collect::<String>()), text, "{len}");
662            assert!(fs.iter().all(|f| f.encode_utf16().count() <= FORMULA_UNITS), "{len}");
663        }
664        let mut wb = Workbook::new();
665        let longest = "\u{0}".repeat(MAX_ITEM_CHARS);
666        set(&mut wb, "longest", &longest).unwrap();
667        excel_limits_hold(&wb);
668        assert_eq!(get(&wb, "longest").map(|t| t.len()), Some(MAX_ITEM_CHARS));
669        // A marker in a hand-made name that is not an escape stays as it is.
670        let mut wb = Workbook::new();
671        wb.names.push(name("_gc_hand_0", "\"a\u{FDD0}zz\u{FDD0}D800\u{FDD0}004\u{FDD0}0041\u{FDD0}\""));
672        assert_eq!(get(&wb, "hand").as_deref(), Some("a\u{FDD0}zz\u{FDD0}D800\u{FDD0}004A\u{FDD0}"));
673    }
