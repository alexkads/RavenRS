15fn blend(fg: [u8; 4]) -> Rgb {
16    let a = fg[3] as u16;
17    let mix = |c: u8| ((c as u16 * a + 255 * (255 - a)) / 255) as u8;
18    [mix(fg[0]), mix(fg[1]), mix(fg[2])]
19}

28fn cell_look(wb: &Workbook, si: usize, sh: &Sheet, c: CellRef, cf: &mut Option<crate::engine::cf::CfCache>) -> Look {
29    let mut style = match sh.table_at(c) {
30        Some(t) => crate::engine::tables::table_cell_style(wb, si, t, c),
31        None => wb.styles.get(sh.style_id(c)).clone(),
32    };
33    let cfl = cf.as_mut().and_then(|cache| cache.look(wb, si, c));
34    if let Some(st) = cfl.as_ref().and_then(|l| l.style.as_ref()) {
35        if st.fill.pattern != PatternType::None {
36            style.fill = st.fill;
37        }
38        if st.font.color != Color::Auto {
39            style.font.color = st.font.color;
40        }
41        style.font.bold |= st.font.bold;
42        style.font.italic |= st.font.italic;
43        style.font.strike |= st.font.strike;
44        if st.font.underline != Underline::None {
45            style.font.underline = st.font.underline;
46        }
47        // In a differential style General means "not set" (#1163).
48        if st.num_fmt.as_str() != "General" {
49            style.num_fmt = st.num_fmt.clone();
50        }
51        for (dst, src) in [
52            (&mut style.borders.left, st.borders.left),
53            (&mut style.borders.right, st.borders.right),
54            (&mut style.borders.top, st.borders.top),
55            (&mut style.borders.bottom, st.borders.bottom),
56        ] {
57            if !src.is_none() {
58                *dst = src;
59            }
60        }
61    }
62    let mut fill = match style.fill.pattern {
63        PatternType::None => None,
64        PatternType::Solid => rgb_of(style.fill.fg, wb),
65        _ => {
66            let fg = rgb_of(style.fill.fg, wb).unwrap_or([0, 0, 0]);
67            let bg = rgb_of(style.fill.bg, wb).unwrap_or([255, 255, 255]);
68            Some([0, 1, 2].map(|i| ((fg[i] as u16 + bg[i] as u16 * 3) / 4) as u8))
69        }
70    };
71    if let Some(rgb) = cfl.as_ref().and_then(|l| l.scale_fill) {
72        fill = Some(rgb);
73    }
74    Look { style, fill, cf: cfl }
75}

90    pub(super) oy: f32,
91    pub(super) k: f32,

94impl Xf {
95    fn p(&self, x: f32, y: f32) -> (f32, f32) {
96        (self.ox + x * self.k, self.oy + y * self.k)
97    }
98    fn pts(&self, pts: &[[f32; 2]]) -> Vec<[f32; 2]> {
99        pts.iter().map(|q| [self.ox + q[0] * self.k, self.oy + q[1] * self.k]).collect()
100    }

103/// Draws chart primitives onto a page.
104pub(super) fn draw_prims(pg: &mut Page, prims: &[crate::chart::Prim], xf: Xf) {
105    use crate::chart::{HAlign as H, Prim, VAlign as V};
106    let k = xf.k;
107    for p in prims {
108        match p {
109            Prim::Rect { x, y, w, h, fill, stroke, .. } => {
110                let (px, py) = xf.p(*x, *y);
111                if let Some(f) = fill {
112                    pg.fill_rect(px, py, w * k, h * k, blend(*f));
113                }
114                if let Some((c, lw)) = stroke {
115                    pg.rect_stroke(px, py, w * k, h * k, (lw * k).max(0.1), blend(*c));
116                }
117            }
118            Prim::Line { pts, color, width, dash } => {
119                let d = if *dash { vec![3.0 * k, 2.0 * k] } else { vec![] };
120                pg.polyline(&xf.pts(pts), (width * k).max(0.1), blend(*color), &d);
121            }
122            Prim::Polygon { pts, fill, stroke } => {
123                pg.polygon(&xf.pts(pts), Some(blend(*fill)), stroke.map(|(c, w)| (blend(c), (w * k).max(0.1))));
124            }
125            Prim::Wedge { cx, cy, r_outer, r_inner, a0, a1, fill, stroke } => {
126                let span = (a1 - a0).abs().min(std::f32::consts::TAU);
127                let n = ((span / (std::f32::consts::PI / 36.0)).ceil() as usize).clamp(2, 400);
128                let at = |a: f32, r: f32| [cx + r * a.sin(), cy - r * a.cos()];
129                let mut pts: Vec<[f32; 2]> = (0..=n).map(|i| at(a0 + (a1 - a0) * i as f32 / n as f32, *r_outer)).collect();
130                if *r_inner > 0.0 {
131                    pts.extend((0..=n).rev().map(|i| at(a0 + (a1 - a0) * i as f32 / n as f32, *r_inner)));
132                } else {
133                    pts.push([*cx, *cy]);
134                }
135                pg.polygon(&xf.pts(&pts), Some(blend(*fill)), stroke.map(|(c, w)| (blend(c), (w * k).max(0.1))));
136            }
137            Prim::Circle { cx, cy, r, fill, stroke } => {
138                let (px, py) = xf.p(*cx, *cy);
139                pg.circle(px, py, r * k, fill.map(blend), stroke.map(|(c, w)| (blend(c), (w * k).max(0.1))));
140            }
141            Prim::Text { x, y, text, size, color, bold, align, valign, rotation } => {
142                let font = if *bold { Font::HelveticaBold } else { Font::Helvetica };
143                let size = size * k;
144                let w = crate::pdf::text_width(font, size, text);
145                let dx = match align {
146                    H::Left => 0.0,
147                    H::Center => -w / 2.0,
148                    H::Right => -w,
149                };
150                let dy = match valign {
151                    V::Top => size * 0.78,
152                    V::Middle => size * 0.35,
153                    V::Bottom => -size * 0.21,
154                };
155                let (sn, cs) = if rotation.is_finite() { rotation.sin_cos() } else { (0.0, 1.0) };
156                let (ax, ay) = xf.p(*x, *y);
157                pg.text_rotated(Point::new(ax + dx * cs - dy * sn, ay + dx * sn + dy * cs), size, font, blend(*color), text, *rotation);
158            }
159        }
160    }
161}

163/// Everything needed to draw one page block (a run of rows × a run of columns).
164pub(super) struct Block<'a> {
165    pub(super) wb: &'a Workbook,
166    si: usize,
167    pub(super) sh: &'a Sheet,
168    rows: (u32, u32),
169    cols: (u32, u32),
170    /// Page x of each column's left edge, plus the right edge of the last.
171    pub(super) xs: Vec<f32>,
172    pub(super) ys: Vec<f32>,
173    pub(super) k: f32,
174    pub(super) scale: f32,
175}
176
177impl Block<'_> {
178    fn x(&self, col: u32) -> f32 {
179        if col >= self.cols.0 && col <= self.cols.1 + 1 {
180            return self.xs.get((col - self.cols.0) as usize).copied().unwrap_or(0.0);
181        }
182        let x0 = self.xs.first().copied().unwrap_or(0.0);
183        x0 + ((self.sh.col_left(col) - self.sh.col_left(self.cols.0)) as f32) * self.k
184    }
185    fn y(&self, row: u32) -> f32 {
186        if row >= self.rows.0 && row <= self.rows.1 + 1 {
187            return self.ys.get((row - self.rows.0) as usize).copied().unwrap_or(0.0);
188        }
189        let y0 = self.ys.first().copied().unwrap_or(0.0);
190        y0 + ((self.sh.row_top(row) - self.sh.row_top(self.rows.0)) as f32) * self.k
191    }
192    pub(super) fn rect(&self, r: RangeRef) -> (f32, f32, f32, f32) {
193        let (x0, y0) = (self.x(r.start.col), self.y(r.start.row));
194        (x0, y0, self.x(r.end.col.saturating_add(1)) - x0, self.y(r.end.row.saturating_add(1)) - y0)
195    }
196    pub(super) fn range(&self) -> RangeRef {
197        RangeRef::new(CellRef::new(self.rows.0, self.cols.0), CellRef::new(self.rows.1, self.cols.1))
198    }
199}

201fn border_dash(style: BorderStyle, k: f32) -> Vec<f32> {
202    let u = k.max(0.2);
203    match style {
204        BorderStyle::Dashed | BorderStyle::MediumDashed => vec![4.0 * u, 2.0 * u],
205        BorderStyle::Dotted => vec![1.0 * u, 1.0 * u],
206        BorderStyle::Hair => vec![0.75 * u, 0.75 * u],
207        BorderStyle::DashDot | BorderStyle::MediumDashDot | BorderStyle::SlantDashDot => vec![4.0 * u, 1.5 * u, 1.0 * u, 1.5 * u],
208        BorderStyle::DashDotDot | BorderStyle::MediumDashDotDot => vec![4.0 * u, 1.5 * u, 1.0 * u, 1.5 * u, 1.0 * u, 1.5 * u],
209        _ => vec![],
210    }
211}

213fn draw_border(pg: &mut Page, b: &BorderLine, wb: &Workbook, start: Point, end: Point, k: f32) {
214    if b.is_none() {
215        return;
216    }
217    let (x0, y0, x1, y1) = (start.x, start.y, end.x, end.y);
218    let color = rgb_of(b.color, wb).unwrap_or([0, 0, 0]);
219    if b.style == BorderStyle::Double {
220        let off = 1.0 * k;
221        let lw = (0.75 * k).max(0.2);
222        let (dx, dy) = if (y0 - y1).abs() < 0.001 { (0.0, off) } else { (off, 0.0) };
223        pg.stroke_line(x0 - dx, y0 - dy, x1 - dx, y1 - dy, lw, color);
224        pg.stroke_line(x0 + dx, y0 + dy, x1 + dx, y1 + dy, lw, color);
225        return;
226    }
227    let lw = (b.style.width() * k).clamp(0.25, 6.0);
228    pg.stroke_line_dashed(Point::new(x0, y0), Point::new(x1, y1), lw, color, &border_dash(b.style, k));
229}

231/// Draws the cells of a block: fills, gridlines, data bars, borders and text.
232pub(super) fn draw_block(pg: &mut Page, b: &Block, gridlines: bool, cf: &mut Option<crate::engine::cf::CfCache>) {
233    let (bx, by) = (b.xs.first().copied().unwrap_or(0.0), b.ys.first().copied().unwrap_or(0.0));
234    let (bw, bh) = (b.xs.last().copied().unwrap_or(bx) - bx, b.ys.last().copied().unwrap_or(by) - by);
235    if bw <= 0.0 || bh <= 0.0 {
236        return;
237    }
238    let block = b.range();
239    let sh = b.sh;
240    let merges: Vec<RangeRef> = sh.merges.iter().filter(|m| m.intersects(&block)).copied().collect();
241    let covered = |c: CellRef| merges.iter().any(|m| m.contains(c));
242    // Cells worth visiting.
243    let cells: Vec<CellRef> = if block.count() <= DENSE_BLOCK {
244        block.iter().filter(|c| !covered(*c)).collect()
245    } else {
246        let mut set: Vec<CellRef> = sh.cells.iter_range(block).map(|(c, _)| c).collect();
247        set.extend(sh.spill.range(block.start..=block.end).map(|(c, _)| *c).filter(|c| block.contains(*c)));
248        set.sort_unstable();
249        set.dedup();
250        set.retain(|c| !covered(*c));
251        set
252    };
253    pg.save();
254    pg.clip_rect(bx, by, bw, bh);
255    if gridlines {
256        let grey = [0xB0, 0xB0, 0xB0];
257        let lw = (0.35 * b.scale).clamp(0.1, 0.5);
258        for x in &b.xs {
259            pg.stroke_line(*x, by, *x, by + bh, lw, grey);
260        }
261        for y in &b.ys {
262            pg.stroke_line(bx, *y, bx + bw, *y, lw, grey);
263        }
264    }
265    // Cheap pre-filter: blank, unstyled cells outside tables and conditional formats draw nothing.
266    let skip = |c: &CellRef| {
267        sh.cond_formats.is_empty()
268            && sh.value_ref(*c).is_none_or(|v| v.is_empty())
269            && sh.style_id(*c) == StyleId::DEFAULT
270            && (sh.tables.is_empty() || sh.table_at(*c).is_none())
271    };
272    let looks: Vec<(CellRef, Look)> = cells
273        .iter()
274        .filter(|c| !skip(c))
275        .chain(merges.iter().map(|m| &m.start))
276        .map(|c| (*c, cell_look(b.wb, b.si, sh, *c, cf)))
277        .filter(|(c, l)| {
278            l.fill.is_some()
279                || l.cf.is_some()
280                || merges.iter().any(|m| m.start == *c)
281                || !sh.value(*c).is_empty()
282                || l.style.borders != Borders::default()
283                || l.style.num_fmt.as_str() == "checkbox"
284        })
285        .collect();
286    let rect_of = |c: CellRef| -> (f32, f32, f32, f32, bool) {
287        match merges.iter().find(|m| m.start == c) {
288            Some(m) => {
289                let (x, y, w, h) = b.rect(*m);
290                (x, y, w, h, true)
291            }
292            None => {
293                let (x, y, w, h) = b.rect(RangeRef::cell(c));
294                (x, y, w, h, false)
295            }
296        }
297    };
298    // Fills (merged blocks hide the gridlines inside them).
299    for (c, l) in &looks {
300        let (x, y, w, h, merged) = rect_of(*c);
301        if w <= 0.0 || h <= 0.0 {
302            continue;
303        }
304        if let Some(f) = l.fill {
305            pg.fill_rect(x, y, w, h, f);
306        } else if merged && gridlines {
307            pg.fill_rect(x, y, w, h, [255, 255, 255]);
308        }
309        if merged && gridlines {
310            pg.rect_stroke(x, y, w, h, (0.35 * b.scale).clamp(0.1, 0.5), [0xB0, 0xB0, 0xB0]);
311        }
312        if let Some((frac, rgb, _)) = l.cf.as_ref().and_then(|cf| cf.bar) {
313            let pad = 1.5 * b.k;
314            let bwid = ((w - 2.0 * pad) * frac.clamp(0.0, 1.0)).max(0.0);
315            pg.fill_rect(x + pad, y + pad, bwid, (h - 2.0 * pad).max(0.0), rgb);
316        }
317    }
318    // Borders.
319    for (c, l) in &looks {
320        let (x, y, w, h, merged) = rect_of(*c);
321        if w <= 0.0 || h <= 0.0 {
322            continue;
323        }
324        let bd = &l.style.borders;
325        let (right, bottom) = match merges.iter().find(|m| m.start == *c).filter(|_| merged) {
326            Some(m) => (
327                b.wb.styles.get(sh.style_id(CellRef::new(m.start.row, m.end.col))).borders.right,
328                b.wb.styles.get(sh.style_id(CellRef::new(m.end.row, m.start.col))).borders.bottom,
329            ),
330            None => (bd.right, bd.bottom),
331        };
332        draw_border(pg, &bd.top, b.wb, Point::new(x, y), Point::new(x + w, y), b.k);
333        draw_border(pg, &bottom, b.wb, Point::new(x, y + h), Point::new(x + w, y + h), b.k);
334        draw_border(pg, &bd.left, b.wb, Point::new(x, y), Point::new(x, y + h), b.k);
335        draw_border(pg, &right, b.wb, Point::new(x + w, y), Point::new(x + w, y + h), b.k);
336        draw_border(pg, &bd.diag_down, b.wb, Point::new(x, y), Point::new(x + w, y + h), b.k);
337        draw_border(pg, &bd.diag_up, b.wb, Point::new(x, y + h), Point::new(x + w, y), b.k);
338    }
339    // Text.
340    for (c, l) in &looks {
341        let (x, y, w, h, merged) = rect_of(*c);
342        if w <= 0.0 || h <= 0.0 {
343            continue;
344        }
345        if let Some(picture) = sh.cell_pictures.get(c) {
346            draw_cell_picture(pg, &picture.data, (x, y, w, h), b.k);
347        } else if let Some(on) = checkbox_state(l, &sh.value(*c)) {
348            draw_checkbox(pg, (x, y, w, h), b.k, on);
349        } else {
350            draw_text(pg, b, TextPlacement { cell: *c, look: l, rect: (x, y, w, h), merged, merges: &merges, limits: (bx, bx + bw) });
351        }
352    }
353    pg.restore();
354}

356/// A checkbox cell prints TRUE/FALSE (and empty) as a box, as on screen (#2972); other values
357/// print as text.
358fn checkbox_state(l: &Look, v: &Value) -> Option<bool> {
359    if l.style.num_fmt.as_str() != "checkbox" || l.cf.as_ref().is_some_and(|cf| cf.hide_value) {
360        return None;
361    }
362    match v {
363        Value::Bool(on) => Some(*on),
364        Value::Empty => Some(false),
365        _ => None,
366    }
367}

369/// A checkbox centred in the cell: white with a grey outline, or green with a white tick.
370fn draw_checkbox(pg: &mut Page, (x, y, w, h): (f32, f32, f32, f32), k: f32, on: bool) {
371    let s = (13.2 * k).min(w - 2.0 * k).min(h - 2.0 * k);
372    if s <= 0.0 {
373        return;
374    }
375    let (bx, by) = (x + (w - s) / 2.0, y + (h - s) / 2.0);
376    let corners = [[bx, by], [bx + s, by], [bx + s, by + s], [bx, by + s]];
377    if on {
378        let green = [0x10, 0x7C, 0x41];
379        pg.polygon(&corners, Some(green), Some((green, 0.75 * k)));
380        let tick = [[bx + 0.22 * s, by + 0.52 * s], [bx + 0.42 * s, by + 0.72 * s], [bx + 0.78 * s, by + 0.30 * s]];
381        pg.polyline(&tick, 0.14 * s, [0xFF, 0xFF, 0xFF], &[]);
382    } else {
383        pg.polygon(&corners, Some([0xFF, 0xFF, 0xFF]), Some(([110, 110, 110], 0.75 * k)));
384    }
385}

404/// PDF annotations are not affected by graphics clipping, so intersect each cell with its page
405/// block before adding a clickable rectangle. A repeated print-title cell may appear on several
406/// pages and receives one annotation on each of them.
407pub(super) fn draw_block_links(pg: &mut Page, b: &Block, page: usize, pending: &mut Vec<PendingPdfLink>) {
408    let range = b.range();
409    let (Some(&bx), Some(&by), Some(&right), Some(&bottom)) = (b.xs.first(), b.ys.first(), b.xs.last(), b.ys.last()) else { return };
410    for (cell, link) in b.sh.hyperlinks.range(range.start..=range.end) {
411        if !range.contains(*cell) {
412            continue;
413        }
414        let cell_range = b.sh.merges.iter().find(|m| m.start == *cell).copied().unwrap_or(RangeRef::cell(*cell));
415        let (x, y, w, h) = b.rect(cell_range);
416        let (x0, y0, x1, y1) = (x.max(bx), y.max(by), (x + w).min(right), (y + h).min(bottom));
417        if x1 <= x0 || y1 <= y0 {
418            continue;
419        }
420        let rect = (x0, y0, x1 - x0, y1 - y0);
421        match crate::engine::links::classify(&link.target) {
422            crate::engine::links::LinkKind::External(uri) => pg.link_uri(rect.0, rect.1, rect.2, rect.3, &uri),
423            crate::engine::links::LinkKind::Internal(target) if !link.external => {
424                pending.push(PendingPdfLink { page, sheet: b.si, rect, target });
425            }
426            _ => {}
427        }
428    }
429}

431pub(super) fn printed_target(wb: &Workbook, blocks: &[PrintedBlock], source_sheet: usize, target: &str) -> Option<(usize, f32, f32)> {
432    let target = target.trim().trim_start_matches('#').trim_start_matches('=');
433    let (sheet_name, range) = split_sheet(target);
434    let sheet = sheet_name.map_or(Some(source_sheet), |name| wb.sheet_index(&name))?;
435    let cell = RangeRef::parse(&range.replace('$', ""))?.start;
436    let sh = wb.sheet(sheet)?;
437    let block = blocks.iter().find(|b| b.sheet == sheet && (b.rows.0..=b.rows.1).contains(&cell.row) && (b.cols.0..=b.cols.1).contains(&cell.col))?;
438    let x = block.x + (sh.col_left(cell.col) - sh.col_left(block.cols.0)) as f32 * block.k;
439    let y = block.y + (sh.row_top(cell.row) - sh.row_top(block.rows.0)) as f32 * block.k;
440    Some((block.page, x, y))
441}

443pub(super) fn draw_headings(pg: &mut Page, b: &Block, col_heads: bool, row_heads: bool, at: (f32, f32), size: (f32, f32)) {
444    let fill = [0xF2, 0xF2, 0xF2];
445    let line = [0xA0, 0xA0, 0xA0];
446    let fsize = 9.0 * b.scale.max(0.1);
447    let lw = (0.35 * b.scale).clamp(0.1, 0.5);
448    if col_heads {
449        let y = at.1;
450        for col in b.cols.0..=b.cols.1 {
451            let (x0, x1) = (b.x(col), b.x(col + 1));
452            if x1 - x0 <= 0.0 {
453                continue;
454            }
455            pg.fill_rect(x0, y, x1 - x0, size.1, fill);
456            pg.rect_stroke(x0, y, x1 - x0, size.1, lw, line);
457            let t = crate::base::col_to_letters(col);
458            let tw = crate::pdf::text_width(Font::Helvetica, fsize, &t);
459            pg.text(x0 + (x1 - x0 - tw) / 2.0, y + size.1 / 2.0 + fsize * 0.35, fsize, Font::Helvetica, [0x40, 0x40, 0x40], &t);
460        }
461    }
462    if row_heads {
463        let x = at.0;
464        for row in b.rows.0..=b.rows.1 {
465            let (y0, y1) = (b.y(row), b.y(row + 1));
466            if y1 - y0 <= 0.0 {
467                continue;
468            }
469            pg.fill_rect(x, y0, size.0, y1 - y0, fill);
470            pg.rect_stroke(x, y0, size.0, y1 - y0, lw, line);
471            let t = (row + 1).to_string();
472            let tw = crate::pdf::text_width(Font::Helvetica, fsize, &t);
473            pg.text(x + (size.0 - tw) / 2.0, y0 + (y1 - y0) / 2.0 + fsize * 0.35, fsize, Font::Helvetica, [0x40, 0x40, 0x40], &t);
474        }
475    }
476}

478/// Builds a block for rows × cols positioned at (x, y) on the page.
479pub(super) fn make_block<'a>(
480    wb: &'a Workbook,
481    si: usize,
482    sh: &'a Sheet,
483    rows: (u32, u32),
484    cols: (u32, u32),
485    origin: (f32, f32),
486    lay: &SheetLayout,
487) -> Block<'a> {
488    let (x, y) = origin;
489    let k = lay.k();
490    let mut xs = Vec::with_capacity((cols.1 - cols.0 + 2) as usize);
491    let mut acc = x;
492    xs.push(acc);
493    for c in cols.0..=cols.1 {
494        acc += sh.col_width(c) * k;
495        xs.push(acc);
496    }
497    let mut ys = Vec::with_capacity((rows.1 - rows.0 + 2) as usize);
498    let mut acc = y;
499    ys.push(acc);
500    for r in rows.0..=rows.1 {
501        acc += sh.row_height(r) * k;
502        ys.push(acc);
503    }
504    Block { wb, si, sh, rows, cols, xs, ys, k, scale: lay.scale }
505}
