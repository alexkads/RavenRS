5/// Cell pictures fit the current cell (or merged-cell) rectangle without changing aspect ratio.
6pub(super) fn draw_cell_picture(pg: &mut Page, data: &[u8], rect: (f32, f32, f32, f32), k: f32) {
7    let (x, y, w, h) = rect;
8    let pad = 2.0 * k;
9    let (w, h) = ((w - 2.0 * pad).max(0.0), (h - 2.0 * pad).max(0.0));
10    if w <= 0.0 || h <= 0.0 || data.len() > 16 * 1024 * 1024 {
11        return;
12    }
13    let Some((iw, ih)) = crate::engine::io::image_size(data) else { return };
14    if iw == 0 || ih == 0 || iw > 8192 || ih > 8192 || u64::from(iw) * u64::from(ih) > 16_000_000 {
15        return;
16    }
17    let Ok(mut reader) = image::ImageReader::new(std::io::Cursor::new(data)).with_guessed_format() else { return };
18    let mut limits = image::Limits::default();
19    limits.max_image_width = Some(8192);
20    limits.max_image_height = Some(8192);
21    limits.max_alloc = Some(64 * 1024 * 1024);
22    reader.limits(limits);
23    let Ok(image) = reader.decode() else { return };
24    let image =
25        image.thumbnail(((w / 72.0) * 150.0).ceil().clamp(1.0, 4000.0) as u32, ((h / 72.0) * 150.0).ceil().clamp(1.0, 4000.0) as u32).to_rgba8();
26    let fit = (w / iw as f32).min(h / ih as f32);
27    let (pw, ph) = (iw as f32 * fit, ih as f32 * fit);
28    pg.image_rgba(Rect::new(x + pad + (w - pw) / 2.0, y + pad + (h - ph) / 2.0, pw, ph), image.as_raw(), [image.width(), image.height()]);
29}

33pub(super) const MAX_FALLBACK_LINES: usize = 500;
34
35/// The lines of `text` printed in place of an equation that has no layout: those of its first
36/// `MAX_SOURCE_BYTES` (a longer source never lays out, and a file can carry megabytes) that fit a
37/// frame `h` points tall at `line_h` points per line (at least the first), and never more than
38/// [`MAX_FALLBACK_LINES`]. A single `pg.text` call would print every newline as `?`.
39pub(super) fn fallback_lines(text: &str, h: f32, line_h: f32) -> impl Iterator<Item = &str> {
40    let text = crate::engine::cmd::equation::source_excerpt(text);
41    let fit = h / line_h;
42    let n = if fit >= MAX_FALLBACK_LINES as f32 {
43        MAX_FALLBACK_LINES
44    } else if fit >= 1.0 {
45        fit as usize
46    } else {
47        1
48    };
49    text.lines().take(n)
50}

52fn anchor_rect(sh: &Sheet, a: &Anchor) -> (f64, f64, f64, f64) {
53    (sh.col_left(a.cell.col) + a.dx as f64, sh.row_top(a.cell.row) + a.dy as f64, a.width.max(0.0) as f64, a.height.max(0.0) as f64)
54}

56/// Decodes an embedded picture to RGBA, downscaled to at most `max_w`×`max_h` pixels.
57fn decode_image(data: &[u8], max_w: u32, max_h: u32) -> Option<(Vec<u8>, u32, u32)> {
58    let (w, h) = crate::engine::io::image_size(data)?;
59    if w == 0 || h == 0 || w as u64 * h as u64 > 60_000_000 {
60        return None;
61    }
62    let img = image::load_from_memory(data).ok()?;
63    let img = if img.width() > max_w.max(1) || img.height() > max_h.max(1) { img.thumbnail(max_w.max(1), max_h.max(1)) } else { img };
64    let rgba = img.to_rgba8();
65    let (pw, ph) = (rgba.width(), rgba.height());
66    Some((rgba.into_raw(), pw, ph))
67}

69/// Draws charts, pictures, shapes and sparklines overlapping the body block of a page.
70/// `pics` memoizes decoded pictures for the whole render (one page at a time draws each
71/// picture again, but the decode happens once).
72pub(super) fn draw_objects(pg: &mut Page, wb: &Workbook, si: usize, sh: &Sheet, b: &Block, pics: &mut super::PicCache) {
73    let block = b.range();
74    let (sx0, sy0) = (sh.col_left(block.start.col), sh.row_top(block.start.row));
75    let (sx1, sy1) = (sh.col_left(block.end.col.saturating_add(1)), sh.row_top(block.end.row.saturating_add(1)));
76    let (bx, by) = (b.xs.first().copied().unwrap_or(0.0), b.ys.first().copied().unwrap_or(0.0));
77    let (bw, bh) = (b.xs.last().copied().unwrap_or(bx) - bx, b.ys.last().copied().unwrap_or(by) - by);
78    let visible = |r: (f64, f64, f64, f64)| r.2 > 0.0 && r.3 > 0.0 && r.0 < sx1 && r.0 + r.2 > sx0 && r.1 < sy1 && r.1 + r.3 > sy0;
79    let place = |r: (f64, f64, f64, f64)| (bx + ((r.0 - sx0) as f32) * b.k, by + ((r.1 - sy0) as f32) * b.k, r.2 as f32 * b.k, r.3 as f32 * b.k);
80    pg.save();
81    pg.clip_rect(bx, by, bw, bh);
82    draw_sparklines(pg, wb, si, sh, b, block);
83    for (imi, im) in sh.images.iter().enumerate() {
84        let r = anchor_rect(sh, &im.anchor);
85        if !visible(r) {
86            continue;
87        }
88        let (x, y, w, h) = place(r);
89        // About 150 dpi is plenty for print.
90        let (mw, mh) = (((w / 72.0) * 150.0).ceil().clamp(1.0, 4000.0) as u32, ((h / 72.0) * 150.0).ceil().clamp(1.0, 4000.0) as u32);
91        let decoded = pics.entry((si, imi)).or_insert_with(|| decode_image(&im.data, mw, mh));
92        if let Some((px, pw, ph)) = decoded.as_ref() {
93            pg.image_rgba(Rect::new(x, y, w, h), px, [*pw, *ph]);
94        }
95    }
96    for shp in &sh.shapes {
97        let r = anchor_rect(sh, &shp.anchor);
98        if !visible(r) {
99            continue;
100        }
101        let (x, y, w, h) = place(r);
102        if shp.kind == ShapeKind::Equation {
103            let ink = rgb_of(shp.line, wb).unwrap_or([0, 0, 0]);
104            match shp.math.get(&shp.text) {
105                Some(Ok(list)) => crate::engine::cmd::equation::draw_pdf(pg, list, (x, y, w, h), crate::engine::cmd::equation::PADDING * b.k, ink),
106                _ => {
107                    // No layout (the source does not parse, or the file was past the layout
108                    // budget): the source, one text run per line, as many as the frame holds.
109                    let line_h = 12.0 * b.scale;
110                    // A long line stays inside the frame; this clip nests in the block's.
111                    pg.save();
112                    pg.clip_rect(x, y, w, h);
113                    for (i, line) in fallback_lines(&shp.text, h, line_h).enumerate() {
114                        pg.text(x + 2.0 * b.k, y + line_h * (i + 1) as f32, 10.0 * b.scale, Font::Courier, ink, line);
115                    }
116                    pg.restore();
117                }
118            }
119            continue;
120        }
121        draw_shape(pg, wb, shp, b, (x, y, w, h));
122    }
123    for ch in &sh.charts {
124        let r = anchor_rect(sh, &ch.anchor);
125        if !visible(r) {
126            continue;
127        }
128        let (x, y, _, _) = place(r);
129        let data = crate::chart::resolve(wb, si, ch);
130        let prims = crate::chart::render(ch, &data, r.2 as f32, r.3 as f32, &PdfMeasure);
131        draw_prims(pg, &prims, Xf { ox: x, oy: y, k: b.k });
132    }
133    pg.restore();
134}

136fn draw_sparklines(pg: &mut Page, wb: &Workbook, si: usize, sh: &Sheet, b: &Block, block: RangeRef) {
137    for sp in &sh.sparklines {
138        if !block.contains(sp.cell) {
139            continue;
140        }
141        let vals: Vec<Option<f64>> = match crate::calc::evaluate(wb, si, sp.cell, &sp.source) {
142            Value::Array(a) => a.iter().take(10_000).map(Value::as_f64).collect(),
143            v => vec![v.as_f64()],
144        };
145        let color = rgb_of(sp.color, wb).unwrap_or([0x15, 0x60, 0x82]);
146        let (x, y, w, h) = b.rect(RangeRef::cell(sp.cell));
147        let (wpx, hpx) = (w / b.k, h / b.k);
148        let prims = crate::chart::render_sparkline(sp.kind, &vals, [color[0], color[1], color[2], 255], sp.markers, wpx, hpx);
149        draw_prims(pg, &prims, Xf { ox: x, oy: y, k: b.k });
150    }
151}

153fn draw_shape(pg: &mut Page, wb: &Workbook, shp: &crate::model::Shape, b: &Block, (x, y, w, h): (f32, f32, f32, f32)) {
154    let fill = rgb_of(shp.fill, wb);
155    let line = rgb_of(shp.line, wb).filter(|_| !shp.format.no_line).map(|c| (c, (0.75 * b.k).max(0.25)));
156    match shp.kind {
157        ShapeKind::Ellipse => {
158            let pts: Vec<[f32; 2]> = (0..72)
159                .map(|i| {
160                    let a = i as f32 / 72.0 * std::f32::consts::TAU;
161                    [x + w / 2.0 + w / 2.0 * a.cos(), y + h / 2.0 + h / 2.0 * a.sin()]
162                })
163                .collect();
164            pg.polygon(&pts, fill, line);
165        }
166        ShapeKind::Triangle => pg.polygon(&[[x + w / 2.0, y], [x + w, y + h], [x, y + h]], fill, line),
167        ShapeKind::Line | ShapeKind::Arrow => {
168            let c = line.map(|l| l.0).or(fill).unwrap_or([0, 0, 0]);
169            pg.stroke_line(x, y, x + w, y + h, (1.0 * b.k).max(0.3), c);
170            if shp.kind == ShapeKind::Arrow {
171                let ang = h.atan2(w);
172                let al = 8.0 * b.k;
173                let p = |da: f32| [x + w - al * (ang + da).cos(), y + h - al * (ang + da).sin()];
174                pg.polygon(&[[x + w, y + h], p(0.4), p(-0.4)], Some(c), None);
175            }
176        }
177        _ => {
178            pg.polygon(&[[x, y], [x + w, y], [x + w, y + h], [x, y + h]], fill, line);
179        }
180    }
181    if shp.text.is_empty() {
182        return;
183    }
184    use crate::model::{ShapeTextAlign, ShapeTextAnchor};
185    let fm = &shp.format;
186    let text_box = shp.kind == ShapeKind::TextBox;
187    let size = fm.font_size.unwrap_or(11.0).clamp(1.0, 4000.0) * b.scale;
188    let font = Font::styled(Font::family_for(fm.font.as_deref().unwrap_or("")), fm.bold, fm.italic);
189    let color = fm.text_color.and_then(|c| rgb_of(c, wb)).unwrap_or([0, 0, 0]);
190    let align = fm.align.unwrap_or(if text_box { ShapeTextAlign::Left } else { ShapeTextAlign::Center });
191    let anchor = fm.anchor.unwrap_or(if text_box { ShapeTextAnchor::Top } else { ShapeTextAnchor::Middle });
192    let lines = wrap_lines(&shp.text, font, size, (w - 4.0 * b.k).max(size));
193    let block_h = lines.len().min(500) as f32 * size * 1.2;
194    let mut yy = match anchor {
195        ShapeTextAnchor::Top => y + 2.0 * b.k,
196        ShapeTextAnchor::Middle => y + (h - block_h) / 2.0,
197        ShapeTextAnchor::Bottom => y + h - 2.0 * b.k - block_h,
198    } + size * 0.8;
199    for l in lines.iter().take(500) {
200        let lw = crate::pdf::text_width(font, size, l);
201        let lx = match align {
202            ShapeTextAlign::Left => x + 2.0 * b.k,
203            ShapeTextAlign::Center => x + (w - lw) / 2.0,
204            ShapeTextAlign::Right => x + w - 2.0 * b.k - lw,
205        };
206        pg.text(lx, yy, size, font, color, l);
207        yy += size * 1.2;
208    }
