7#[derive(Clone, Copy, Debug, Default)]
8struct Agg {
9    count: u64,
10    nums: u64,
11    sum: f64,
12    product: f64,
13    min: f64,
14    max: f64,
15}
16
17impl Agg {
18    fn add(&mut self, v: &Value) {
19        if v.is_empty() {
20            return;
21        }
22        self.count += 1;
23        if let Value::Number(n) = v {
24            if self.nums == 0 {
25                self.min = *n;
26                self.max = *n;
27                self.product = 1.0;
28            }
29            self.nums += 1;
30            self.sum += n;
31            self.product *= n;
32            self.min = self.min.min(*n);
33            self.max = self.max.max(*n);
34        }
35    }
36    fn result(&self, func: &str) -> Value {
37        match func {
38            "count" => Value::number(self.count as f64),
39            "countNums" | "countnums" => Value::number(self.nums as f64),
40            "average" => {
41                if self.nums == 0 {
42                    Value::Error(crate::base::CellError::Div0)
43                } else {
44                    Value::number(self.sum / self.nums as f64)
45                }
46            }
47            "max" => Value::number(if self.nums == 0 { 0.0 } else { self.max }),
48            "min" => Value::number(if self.nums == 0 { 0.0 } else { self.min }),
49            "product" => Value::number(if self.nums == 0 { 0.0 } else { self.product }),
50            _ => Value::number(self.sum),
51        }
52    }
53}

55/// A row/column key: a position or a label (matched case-insensitively).
56#[derive(Clone, Debug, PartialEq, Eq, Hash)]
57enum Key {
58    Pos(u32),
59    Label(String),
60}
61
62const MAX_CONSOLIDATE_CELLS: u64 = 5_000_000;
63
64pub(super) fn consolidate(s: &mut Session, p: &Json) -> Result<Json> {
65    const CMD: &str = "data.consolidate";
66    let func = str_param(p, "function").unwrap_or("sum").to_string();
67    if !["sum", "count", "countNums", "countnums", "average", "max", "min", "product"].contains(&func.as_str()) {
68        return Err(bad(CMD, format!("unknown function `{func}`")));
69    }
70    let top = bool_param(p, "topRow").unwrap_or(false);
71    let left = bool_param(p, "leftColumn").unwrap_or(false);
72    let d = s.doc()?;
73    let srcs: Vec<String> = p
74        .get("sources")
75        .and_then(Json::as_array)
76        .map(|a| a.iter().filter_map(|x| x.as_str().map(str::to_string)).collect())
77        .or_else(|| str_param(p, "sources").map(|t| t.split(';').map(str::to_string).collect()))
78        .unwrap_or_default();
79    if srcs.is_empty() {
80        return Err(bad(CMD, "give at least one source range in `sources`"));
81    }
82    if srcs.len() > 255 {
83        return Err(bad(CMD, "too many sources (255 max)"));
84    }
85    let (dsi, dest) = match str_param(p, "destination") {
86        Some(t) => {
87            let (si, r) = parse_range(&d.wb, t, CMD)?;
88            (si, r.start)
89        }
90        None => (d.wb.active_sheet(), d.selection.active),
91    };
92    let mut row_keys: Vec<Key> = Vec::new();
93    let mut col_keys: Vec<Key> = Vec::new();
94    let mut row_label: HashMap<Key, String> = HashMap::new();
95    let mut col_label: HashMap<Key, String> = HashMap::new();
96    let mut aggs: HashMap<(usize, usize), Agg> = HashMap::new();
97    let mut total: u64 = 0;
98    let index_of = |keys: &mut Vec<Key>, k: Key| -> usize {
99        match keys.iter().position(|x| *x == k) {
100            Some(i) => i,
101            None => {
102                keys.push(k);
103                keys.len() - 1
104            }
105        }
106    };
107    for t in &srcs {
108        let (si, r) = parse_range(&d.wb, t, CMD)?;
109        total = total.saturating_add(r.count());
110        if total > MAX_CONSOLIDATE_CELLS {
111            return Err(bad(CMD, "the sources are too large"));
112        }
113        let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
114        let r0 = r.start.row + top as u32;
115        let c0 = r.start.col + left as u32;
116        if r0 > r.end.row || c0 > r.end.col {
117            continue;
118        }
119        let label = |c: CellRef| crate::engine::display::cell_text(&d.wb, sh, c).trim().to_string();
120        // Column keys of this source.
121        let mut cols: Vec<(u32, usize)> = Vec::new();
122        for col in c0..=r.end.col {
123            let k = if top {
124                let l = label(CellRef::new(r.start.row, col));
125                if l.is_empty() {
126                    continue;
127                }
128                let k = Key::Label(l.to_lowercase());
129                col_label.entry(k.clone()).or_insert(l);
130                k
131            } else {
132                Key::Pos(col - c0)
133            };
134            cols.push((col, index_of(&mut col_keys, k)));
135        }
136        for row in r0..=r.end.row {
137            let k = if left {
138                let l = label(CellRef::new(row, r.start.col));
139                if l.is_empty() {
140                    continue;
141                }
142                let k = Key::Label(l.to_lowercase());
143                row_label.entry(k.clone()).or_insert(l);
144                k
145            } else {
146                Key::Pos(row - r0)
147            };
148            let ri = index_of(&mut row_keys, k);
149            for (col, ci) in &cols {
150                aggs.entry((ri, *ci)).or_default().add(&sh.value(CellRef::new(row, *col)));
151            }
152        }
153    }
154    let (out_r0, out_c0) = (dest.row + top as u32, dest.col + left as u32);
155    if out_r0 as u64 + row_keys.len() as u64 > crate::base::MAX_ROWS as u64 || out_c0 as u64 + col_keys.len() as u64 > crate::base::MAX_COLS as u64 {
156        return Err(EngineError::Other("The consolidated data doesn't fit below and right of the destination.".into()));
157    }
158    let mut cells: Vec<(CellRef, Value)> = Vec::new();
159    if top {
160        for (ci, k) in col_keys.iter().enumerate() {
161            cells.push((CellRef::new(dest.row, out_c0 + ci as u32), Value::text(col_label.get(k).cloned().unwrap_or_default())));
162        }
163    }
164    if left {
165        for (ri, k) in row_keys.iter().enumerate() {
166            cells.push((CellRef::new(out_r0 + ri as u32, dest.col), Value::text(row_label.get(k).cloned().unwrap_or_default())));
167        }
168    }
169    for ri in 0..row_keys.len() {
170        for ci in 0..col_keys.len() {
171            let v = aggs.get(&(ri, ci)).map(|a| if a.count == 0 { Value::Empty } else { a.result(&func) }).unwrap_or_default();
172            cells.push((CellRef::new(out_r0 + ri as u32, out_c0 + ci as u32), v));
173        }
174    }
175    let end = CellRef::new(
176        (out_r0 + row_keys.len() as u32).saturating_sub(1).max(dest.row),
177        (out_c0 + col_keys.len() as u32).saturating_sub(1).max(dest.col),
178    );
179    edit(s, |cx| {
180        for (c, v) in cells {
181            let sh = cx.sheet_mut(dsi)?;
182            let style = sh.style_id(c);
183            if v.is_empty() && style == StyleId::DEFAULT {
184                sh.remove_cell(c);
185            } else {
186                sh.set_cell(c, Cell { value: v, formula: None, style, py: None });
187            }
188            cx.touch(dsi, c);
189        }
190        Ok(())
191    })?;
192    Ok(json!({"range": RangeRef::new(dest, end).a1(), "rows": row_keys.len(), "columns": col_keys.len()}))
193}
