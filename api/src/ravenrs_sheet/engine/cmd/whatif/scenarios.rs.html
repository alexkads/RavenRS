8/// Hidden workbook-scope name holding the scenarios as a quoted JSON string.
9pub(crate) const SCENARIO_NAME: &str = "_sc_scenarios";

11const MAX_SCENARIO_CELLS: usize = 32;

13const MAX_SCENARIOS: usize = 1000;

18    pub sheet: String,
19    pub changing: String,

25pub(crate) fn load_scenarios(wb: &Workbook) -> Vec<Scenario> {
26    let Some(n) = wb.names.iter().find(|n| n.scope.is_none() && n.name.eq_ignore_ascii_case(SCENARIO_NAME)) else { return vec![] };
27    serde_json::from_str(&string_literals(&n.formula)).unwrap_or_default()
28}

30/// Concatenates the string literals of a `"a"&"b""c"` formula.
31fn string_literals(f: &str) -> String {
32    let mut out = String::new();
33    let mut it = f.chars().peekable();
34    let mut in_str = false;
35    while let Some(c) = it.next() {
36        if !in_str {
37            if c == '"' {
38                in_str = true;
39            }
40            continue;
41        }
42        if c == '"' {
43            if it.peek() == Some(&'"') {
44                it.next();
45                out.push('"');
46            } else {
47                in_str = false;
48            }
49        } else {
50            out.push(c);
51        }
52    }
53    out
54}

56fn store_scenarios(wb: &mut Workbook, list: &[Scenario]) {
57    wb.names.retain(|n| !(n.scope.is_none() && n.name.eq_ignore_ascii_case(SCENARIO_NAME)));
58    if list.is_empty() {
59        return;
60    }
61    let text = serde_json::to_string(list).unwrap_or_else(|_| "[]".into());
62    wb.names.push(DefinedName {
63        name: SCENARIO_NAME.into(),
64        scope: None,
65        // Excel caps string literals at 255 characters: store `"…"&"…"` chunks.
66        formula: text
67            .chars()
68            .collect::<Vec<char>>()
69            .chunks(250)
70            .map(|c| format!("\"{}\"", c.iter().collect::<String>().replace('"', "\"\"")))
71            .collect::<Vec<_>>()
72            .join("&"),
73        comment: "GridCraft Scenario Manager data".into(),
74        hidden: true,
75    });
76}

82pub(super) fn scenario_list(s: &mut Session, p: &Json) -> Result<Json> {
83    let d = s.doc()?;
84    let sheet = str_param(p, "sheet").map(str::to_string).unwrap_or_else(|| d.wb.active().map(|s| s.name.clone()).unwrap_or_default());
85    let list: Vec<Json> =
86        load_scenarios(&d.wb).iter().filter(|sc| sheet == "*" || sc.sheet.eq_ignore_ascii_case(&sheet)).map(scenario_json).collect();
87    Ok(json!({"scenarios": list}))
88}

90pub(super) fn scenario_add(s: &mut Session, p: &Json) -> Result<Json> {
91    const CMD: &str = "data.scenarioAdd";
92    let name = str_param(p, "name").map(str::trim).filter(|n| !n.is_empty()).ok_or_else(|| bad(CMD, "missing `name`"))?.to_string();
93    if name.chars().count() > 255 {
94        return Err(bad(CMD, "the name is too long (255 characters max)"));
95    }
96    let changing = str_param(p, "changing").ok_or_else(|| bad(CMD, "missing `changing`"))?.to_string();
97    let d = s.doc()?;
98    let (si, cells) = cell_list(&d.wb, &changing, CMD, MAX_SCENARIO_CELLS)?;
99    if cells.is_empty() {
100        return Err(bad(CMD, "no changing cells"));
101    }
102    let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
103    let values: Vec<Json> = match p.get("values").and_then(Json::as_array) {
104        Some(a) => {
105            if a.len() != cells.len() {
106                return Err(bad(CMD, format!("{} values for {} changing cells", a.len(), cells.len())));
107            }
108            a.clone()
109        }
110        None => cells.iter().map(|c| value_json(&sh.value(*c))).collect(),
111    };
112    let changing = cells.iter().map(|c| c.a1()).collect::<Vec<_>>().join(",");
113    let sc = Scenario { name: name.clone(), sheet: sh.name.clone(), changing, values, comment: str_param(p, "comment").unwrap_or("").to_string() };
114    let replace = bool_param(p, "replace").unwrap_or(false);
115    let mut list = load_scenarios(&d.wb);
116    match list.iter().position(|x| x.name.eq_ignore_ascii_case(&name) && x.sheet.eq_ignore_ascii_case(&sc.sheet)) {
117        Some(i) if replace => {
118            if let Some(slot) = list.get_mut(i) {
119                *slot = sc.clone();
120            }
121        }
122        Some(_) => return Err(EngineError::Other(format!("A scenario named '{name}' already exists on this sheet."))),
123        None => {
124            if list.len() >= MAX_SCENARIOS {
125                return Err(bad(CMD, "too many scenarios"));
126            }
127            list.push(sc.clone());
128        }
129    }
130    edit(s, |cx| {
131        store_scenarios(&mut cx.wb, &list);
132        Ok(())
133    })?;
134    Ok(scenario_json(&sc))
135}

137fn find_scenario(wb: &Workbook, p: &Json, cmd: &str) -> Result<(Vec<Scenario>, usize)> {
138    let name = str_param(p, "name").ok_or_else(|| bad(cmd, "missing `name`"))?;
139    let list = load_scenarios(wb);
140    let active = wb.active().map(|s| s.name.clone()).unwrap_or_default();
141    let i = list
142        .iter()
143        .position(|x| x.name.eq_ignore_ascii_case(name) && x.sheet.eq_ignore_ascii_case(&active))
144        .or_else(|| list.iter().position(|x| x.name.eq_ignore_ascii_case(name)))
145        .ok_or_else(|| bad(cmd, format!("no scenario named `{name}`")))?;
146    Ok((list, i))
147}

149pub(super) fn scenario_show(s: &mut Session, p: &Json) -> Result<Json> {
150    const CMD: &str = "data.scenarioShow";
151    let d = s.doc()?;
152    let (list, i) = find_scenario(&d.wb, p, CMD)?;
153    let sc = list.get(i).cloned().ok_or_else(|| bad(CMD, "no such scenario"))?;
154    let si = d.wb.sheet_index(&sc.sheet).ok_or_else(|| bad(CMD, format!("the scenario's sheet `{}` no longer exists", sc.sheet)))?;
155    let (_, cells) = cell_list(&d.wb, &sc.changing, CMD, MAX_SCENARIO_CELLS)?;
156    edit(s, |cx| {
157        for (c, v) in cells.iter().zip(sc.values.iter()) {
158            // Stored numbers are values, not typed text: entering them through the parser would
159            // apply percent entry (9 -> 9%) to a Percentage-formatted cell.
160            let cell = match v.as_f64() {
161                Some(n) => {
162                    let style = cx.wb.sheet(si).map(|sh| sh.style_id(*c)).unwrap_or_default();
163                    Some(Cell { value: Value::Number(n), formula: None, style, py: None })
164                }
165                None => crate::engine::cmd::edit::input_to_cell(&crate::engine::cmd::edit::json_to_input(v), si, *c, &mut cx.wb)?,
166            };
167            let sh = cx.sheet_mut(si)?;
168            match cell {
169                Some(cell) => sh.set_cell(*c, cell),
170                None => {
171                    sh.remove_cell(*c);
172                }
173            }
174            cx.touch(si, *c);
175        }
176        Ok(())
177    })?;
178    Ok(scenario_json(&sc))
179}

181pub(super) fn scenario_delete(s: &mut Session, p: &Json) -> Result<Json> {
182    let d = s.doc()?;
183    let (mut list, i) = find_scenario(&d.wb, p, "data.scenarioDelete")?;
184    let removed = list.remove(i);

187        Ok(())
188    })?;
189    Ok(json!({"deleted": removed.name}))

192pub(super) fn scenario_summary(s: &mut Session, p: &Json) -> Result<Json> {
193    const CMD: &str = "data.scenarioSummary";
194    let d = s.doc()?;
195    let si = d.wb.active_sheet();
196    let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
197    let scenarios: Vec<Scenario> = load_scenarios(&d.wb).into_iter().filter(|x| x.sheet.eq_ignore_ascii_case(&sh.name)).collect();
198    if scenarios.is_empty() {
199        return Err(EngineError::Other("There are no scenarios on this sheet.".into()));
200    }
201    let results: Vec<CellRef> = match str_param(p, "resultCells").map(str::trim).filter(|t| !t.is_empty()) {
202        Some(t) => {
203            let (rsi, cells) = cell_list(&d.wb, t, CMD, 255)?;
204            if rsi != si {
205                return Err(bad(CMD, "result cells must be on the scenarios' sheet"));
206            }
207            cells
208        }
209        None => vec![],
210    };
211    // Changing cells of every scenario, first-seen order; per scenario its values.
212    let mut changing: Vec<CellRef> = Vec::new();
213    let mut per: Vec<HashMap<CellRef, Value>> = Vec::new();
214    for sc in &scenarios {
215        let (_, cells) = cell_list(&d.wb, &sc.changing, CMD, MAX_SCENARIO_CELLS)?;
216        let mut m = HashMap::new();
217        for (c, v) in cells.iter().zip(sc.values.iter()) {
218            if !changing.contains(c) {
219                changing.push(*c);
220            }
221            m.insert(*c, json_value(v, &d.wb));
222        }
223        per.push(m);
224    }
225    let current: Vec<Value> = changing.iter().map(|c| sh.value(*c)).collect();
226    let current_results: Vec<Value> = results.iter().map(|c| sh.value(*c)).collect();
227    let mut trial = Trial::new(d);
228    let reads: Vec<(usize, CellRef)> = results.iter().map(|c| (si, *c)).collect();
229    let scenario_results: Vec<Vec<Value>> = per
230        .iter()
231        .map(|m| {
232            let sets: Vec<(usize, CellRef, Value)> = m.iter().map(|(c, v)| (si, *c, v.clone())).collect();
233            trial.run(&sets, &reads)
234        })
235        .collect();
236    let fmt_of = |c: CellRef| d.wb.styles.get(sh.style_id(c)).num_fmt.clone();
237    let changing_fmts: Vec<_> = changing.iter().map(|c| fmt_of(*c)).collect();
238    let result_fmts: Vec<_> = results.iter().map(|c| fmt_of(*c)).collect();
239    let name = (1..)
240        .map(|n| if n == 1 { "Scenario Summary".to_string() } else { format!("Scenario Summary {n}") })
241        .find(|n| d.wb.sheet_index(n).is_none())
242        .unwrap_or_else(|| "Scenario Summary".into());
243    let names: Vec<String> = scenarios.iter().map(|x| x.name.clone()).collect();
244    let new_index = edit(s, |cx| {
245        let bold = cx.wb.styles.derive(StyleId::DEFAULT, |st| st.font.bold = true);
246        let mut sheet = Sheet::new(name.clone());
247        let put = |sheet: &mut Sheet, row: u32, col: u32, v: Value, style: StyleId| {
248            sheet.set_cell(CellRef::new(row, col), Cell { value: v, formula: None, style, py: None });
249        };
250        put(&mut sheet, 1, 1, Value::text("Scenario Summary"), bold);
251        put(&mut sheet, 2, 2, Value::text("Current Values:"), bold);
252        for (k, n) in names.iter().enumerate() {
253            put(&mut sheet, 2, 3 + k as u32, Value::text(n.as_str()), bold);
254        }
255        put(&mut sheet, 3, 1, Value::text("Changing Cells:"), bold);
256        let mut row = 4u32;
257        for (i, c) in changing.iter().enumerate() {
258            let fmt = changing_fmts.get(i).cloned().unwrap_or_default();
259            let st = cx.wb.styles.derive(StyleId::DEFAULT, |st| st.num_fmt = fmt);
260            put(&mut sheet, row, 1, Value::text(format!("${}${}", crate::base::col_to_letters(c.col), c.row + 1)), StyleId::DEFAULT);
261            let cur = current.get(i).cloned().unwrap_or_default();
262            put(&mut sheet, row, 2, cur.clone(), st);
263            for (k, m) in per.iter().enumerate() {
264                put(&mut sheet, row, 3 + k as u32, m.get(c).cloned().unwrap_or_else(|| cur.clone()), st);
265            }
266            row += 1;
267        }
268        if !results.is_empty() {
269            put(&mut sheet, row, 1, Value::text("Result Cells:"), bold);
270            row += 1;
271            for (i, c) in results.iter().enumerate() {
272                let fmt = result_fmts.get(i).cloned().unwrap_or_default();
273                let st = cx.wb.styles.derive(StyleId::DEFAULT, |st| st.num_fmt = fmt);
274                put(&mut sheet, row, 1, Value::text(format!("${}${}", crate::base::col_to_letters(c.col), c.row + 1)), StyleId::DEFAULT);
275                put(&mut sheet, row, 2, current_results.get(i).cloned().unwrap_or_default(), st);
276                for (k, r) in scenario_results.iter().enumerate() {
277                    put(&mut sheet, row, 3 + k as u32, r.get(i).cloned().unwrap_or_default(), st);
278                }
279                row += 1;
280            }
281        }
282        put(
283            &mut sheet,
284            row + 1,
285            1,
286            Value::text("Notes: Current Values column represents values of changing cells at time Scenario Summary Report was created."),
287            StyleId::DEFAULT,
288        );
289        sheet.cols.insert(1, crate::model::LineInfo { size: Some(110.0), ..Default::default() });
290        sheet.cols.insert(2, crate::model::LineInfo { size: Some(100.0), ..Default::default() });
291        cx.wb.sheets.push(std::sync::Arc::new(sheet));
292        cx.structural = true;
293        Ok(cx.wb.sheets.len() - 1)
294    })?;
295    s.doc_mut()?.wb_switch_sheet(new_index);
296    Ok(json!({"sheet": name, "scenarios": names.len(), "changingCells": changing.len(), "resultCells": results.len()}))
297}
