120/// Totals for a multi-cell selection as `(cell, formula)` writes. `None` when the sheet has no data.
121fn selection_totals(sh: &crate::model::sheet::Sheet, sel: RangeRef, func: &str) -> Result<Option<Vec<(CellRef, String)>>> {
122    // A selection: totals go in its empty last row or column, as Excel's do, and otherwise
123    // past it: to the right of a single row, below each column of anything taller. Columns
124    // past the sheet's data get no total (a whole-row selection doesn't write 16,384
125    // formulas), nor do rows past it, and all totals are one edit (one undo step).
126    let Some(used) = sh.cells.used_range() else { return Ok(None) };
127    if sel.start.row == sel.end.row { single_row_totals(sh, sel, used, func).map(Some) } else { block_totals(sh, sel, used, func).map(Some) }
128}

130/// One row: its total goes in its last cell when that is empty, else just after it.
131fn single_row_totals(sh: &crate::model::sheet::Sheet, sel: RangeRef, used: RangeRef, func: &str) -> Result<Vec<(CellRef, String)>> {
132    let row = sel.start.row;
133    let last_col = used.end.col.min(sel.end.col);
134    let end_col = if sel.is_full_rows() { last_col } else { sel.end.col };
135    let (data_end, out) = if end_col > sel.start.col && blank(sh, row, end_col) { (end_col - 1, end_col) } else { (end_col, end_col + 1) };
136    if out >= crate::base::MAX_COLS {
137        return Err(EngineError::Other("There's no room to the right of the selection for the total.".into()));
138    }
139    Ok(vec![(CellRef::new(row, out), sum_formula(func, CellRef::new(row, sel.start.col), CellRef::new(row, data_end)))])
140}

142/// A block of rows: each column's total below it, and each row's total in an empty last column.
143fn block_totals(sh: &crate::model::sheet::Sheet, sel: RangeRef, used: RangeRef, func: &str) -> Result<Vec<(CellRef, String)>> {
144    let mut writes: Vec<(CellRef, String)> = Vec::new();
145    let last_col = used.end.col.min(sel.end.col);
146    // An empty last column takes each row's total; rows past the data get none.
147    let rows_end = sel.end.row.min(used.end.row);
148    let right_blank = !sel.is_full_rows() && sel.end.col > sel.start.col && (sel.start.row..=rows_end).all(|r| blank(sh, r, sel.end.col));
149    let data_end_col = if right_blank { sel.end.col - 1 } else { last_col };
150    let last_row_blank = (sel.start.col..=last_col).all(|c| blank(sh, sel.end.row, c));
151    let (data_end_row, out_row) = if last_row_blank { (sel.end.row.saturating_sub(1), sel.end.row) } else { (sel.end.row, sel.end.row + 1) };
152    // Each column's total, below it: unless only the last column is there to take totals.
153    if last_row_blank || !right_blank {
154        if out_row >= crate::base::MAX_ROWS {
155            return Err(EngineError::Other("There's no room below the selection for the totals.".into()));
156        }
157        for c in sel.start.col..=data_end_col.min(last_col) {
158            writes.push((CellRef::new(out_row, c), sum_formula(func, CellRef::new(sel.start.row, c), CellRef::new(data_end_row, c))));
159        }
160    }
161    if right_blank {
162        for r in sel.start.row..=data_end_row.min(rows_end) {
163            writes.push((CellRef::new(r, sel.end.col), sum_formula(func, CellRef::new(r, sel.start.col), CellRef::new(r, data_end_col))));
164        }
165        // Both empty: the corner takes the total of the whole block.
166        if last_row_blank {
167            writes.push((
168                CellRef::new(sel.end.row, sel.end.col),
169                sum_formula(func, CellRef::new(sel.start.row, sel.start.col), CellRef::new(data_end_row, data_end_col)),
170            ));
171        }
172    }
173    Ok(writes)
174}

176/// The range a single cell's AutoSum guesses: the numbers above it, else the numbers to its left.
177fn guessed_range(sh: &crate::model::sheet::Sheet, at: CellRef) -> Option<RangeRef> {
178    let numeric = |c: CellRef| sh.value(c).is_number();
179    let mut range = None;
180    if at.row > 0 && (numeric(CellRef::new(at.row - 1, at.col)) || at.col == 0 || !numeric(CellRef::new(at.row, at.col - 1))) {
181        let mut top = at.row - 1;
182        while top > 0 && numeric(CellRef::new(top - 1, at.col)) {
183            top -= 1;
184        }
185        if numeric(CellRef::new(at.row - 1, at.col)) {
186            range = Some(RangeRef::new(CellRef::new(top, at.col), CellRef::new(at.row - 1, at.col)));
187        }
188    }
189    if range.is_none() && at.col > 0 && numeric(CellRef::new(at.row, at.col - 1)) {
190        let mut left = at.col - 1;
191        while left > 0 && numeric(CellRef::new(at.row, left - 1)) {
192            left -= 1;
193        }
194        range = Some(RangeRef::new(CellRef::new(at.row, left), CellRef::new(at.row, at.col - 1)));
195    }
196    range
197}

199fn auto_sum(s: &mut Session, p: &Json) -> Result<Json> {
200    let func = str_param(p, "function").unwrap_or("SUM").to_ascii_uppercase();
201    let d = s.doc()?;
202    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
203    let sel = match str_param(p, "range") {
204        Some(_) => target_range(s, p)?,
205        None => d.selection.current(),
206    };
207    if !sel.is_single() {
208        let Some(writes) = selection_totals(sh, sel, &func)? else { return ok() };
209        let sheet = d.wb.active_sheet();
210        edit(s, |cx| {
211            for (c, f) in &writes {
212                let cell = super::edit::input_to_cell(f, sheet, *c, &mut cx.wb)?.unwrap_or_default();
213                cx.sheet_mut(sheet)?.set_cell(*c, cell);
214                cx.touch(sheet, *c);
215            }
216            Ok(())
217        })?;
218        return ok();
219    }
220    // Single cell: guess the range above, else to the left.
221    let at = sel.start;
222    let text = match guessed_range(sh, at) {
223        Some(r) => format!("={func}({})", r.a1()),
224        None => {
225            s.ui_requests.push(crate::engine::UiRequest::EditCell(Some(format!("={func}()"))));
226            return ok();
227        }
228    };
229    if p.get("enter").and_then(Json::as_bool) == Some(false) {
230        s.ui_requests.push(crate::engine::UiRequest::EditCell(Some(text.clone())));
231        return Ok(json!({"formula": text}));
232    }
233    s.execute("cell.set", json!({"cell": at.a1(), "input": text}))?;
234    Ok(json!({"formula": text}))
235}

320fn valid_name(n: &str) -> bool {
321    let mut chars = n.chars();
322    let Some(first) = chars.next() else { return false };
323    if !(first.is_alphabetic() || first == '_' || first == '\\') {
324        return false;
325    }
326    if n.chars().count() > 255 || CellRef::parse(n).is_some() || is_r1c1(n) {
327        return false;
328    }
329    n.chars().all(|c| c.is_alphanumeric() || matches!(c, '_' | '.' | '\\' | '?'))
330}

332/// Looks like an R1C1 reference (`R`, `C`, `RC`, `R2`, `C3`, `R1C1`), which Excel doesn't
333/// accept as a name.
334pub(crate) fn is_r1c1(n: &str) -> bool {
335    let digits = |s: &str| s.bytes().take_while(u8::is_ascii_digit).count();
336    let mut rest = n;
337    if let Some(r) = rest.strip_prefix(['R', 'r']) {
338        rest = r.get(digits(r)..).unwrap_or("");
339    }
340    if let Some(c) = rest.strip_prefix(['C', 'c']) {
341        rest = c.get(digits(c)..).unwrap_or("");
342    }
343    rest.is_empty() && !n.is_empty()
344}

346fn define_name(s: &mut Session, p: &Json) -> Result<Json> {
347    let Some(name) = str_param(p, "name").map(str::to_string) else {
348        s.ui_requests.push(crate::engine::UiRequest::Dialog("defineName".into(), json!({})));
349        return ok();
350    };
351    if !valid_name(&name) {
352        return Err(EngineError::Other("The name that you entered is not valid.".into()));
353    }
354    for n in [Some(name.as_str()), str_param(p, "oldName")].into_iter().flatten() {
355        refuse_stash_name(&s.doc()?.wb, p, n)?;
356    }
357    let d = s.doc()?;
358    let scope = match str_param(p, "scope") {
359        None | Some("Workbook") => None,
360        Some(sh) => Some(d.wb.sheet_index(sh).ok_or_else(|| bad("formulas.defineName", "no such sheet"))?),
361    };
362    let refers = match str_param(p, "refersTo") {
363        // As in Excel, relative references are relative to the active cell; names keep them
364        // relative to A1, as files do.
365        Some(r) => match crate::formula::parse_name(r.trim_start_matches('=')) {
366            Ok(e) if crate::formula::adjust::has_relative_refs(&e) => {
367                let at = d.selection.active;
368                crate::formula::print(&crate::formula::adjust::shift_relative_wrapping(e, -(at.row as i64), -(at.col as i64)))
369            }
370            _ => r.trim_start_matches('=').to_string(),
371        },
372        None => {
373            let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
374            let r = d.selection.current();
375            let abs = |c: CellRef| format!("${}${}", crate::base::col_to_letters(c.col), c.row + 1);
376            let body = if r.is_single() { abs(r.start) } else { format!("{}:{}", abs(r.start), abs(r.end)) };
377            format!("{}!{}", crate::formula::quote_sheet(&sh.name), body)
378        }
379    };
380    if crate::formula::parse_name(&refers).is_err() {
381        return Err(EngineError::Other("There's a problem with this formula.".into()));
382    }
383    let comment = str_param(p, "comment").unwrap_or("").to_string();
384    // Editing an existing name (Name Manager › Edit…): it must exist, and its new name must not
385    // collide with another name in the target scope.
386    let old = match str_param(p, "oldName") {
387        Some(old_name) => {
388            let old_scope = match str_param(p, "oldScope") {
389                None | Some("Workbook") => None,
390                Some(sh) => Some(d.wb.sheet_index(sh).ok_or_else(|| bad("formulas.defineName", "no such sheet"))?),
391            };
392            if !d.wb.names.iter().any(|n| n.name.eq_ignore_ascii_case(old_name) && n.scope == old_scope) {
393                return Err(bad("formulas.defineName", "no such name to edit"));
394            }
395            let same = old_name.eq_ignore_ascii_case(&name) && old_scope == scope;
396            if !same && d.wb.names.iter().any(|n| n.name.eq_ignore_ascii_case(&name) && n.scope == scope) {
397                return Err(EngineError::Other("The name entered already exists. Enter a unique name.".into()));
398            }
399            Some((old_name.to_string(), old_scope))
400        }
401        None => None,
402    };
403    edit(s, |cx| {
404        if let Some((old_name, old_scope)) = &old {
405            cx.wb.names.retain(|n| !(n.name.eq_ignore_ascii_case(old_name) && n.scope == *old_scope));
406            if !old_name.eq_ignore_ascii_case(&name) {
407                rename_name_in_formulas(&mut cx.wb, old_name, *old_scope, &name);
408            }
409        }
410        cx.wb.names.retain(|n| !(n.name.eq_ignore_ascii_case(&name) && n.scope == scope));
411        cx.wb.names.push(DefinedName { name: name.clone(), scope, formula: refers.clone(), comment: comment.clone(), hidden: false });
412        cx.structural = true;
413        Ok(json!({"name": name, "refersTo": format!("={refers}")}))
414    })
415}

417/// Formulas that use a renamed name follow it: cell formulas and other names' definitions, on
418/// the sheets where `old` resolves to the renamed name (its own sheet for a sheet-scoped name;
419/// every sheet without a same-named sheet-level name for a workbook-level one).
420fn rename_name_in_formulas(wb: &mut crate::model::Workbook, old: &str, old_scope: Option<usize>, new: &str) {
421    let shadowed: Vec<usize> = wb.names.iter().filter(|n| n.name.eq_ignore_ascii_case(old)).filter_map(|n| n.scope).collect();
422    let applies = |sheet: usize| match old_scope {
423        Some(s) => s == sheet,
424        None => !shadowed.contains(&sheet),
425    };
426    // The renamed expression, or `None` when the name doesn't occur in it.
427    let rename = |e: crate::formula::Expr| {
428        let mut hit = false;
429        let out = e.map(&mut |x| match x {
430            crate::formula::Expr::Name(n) if n.eq_ignore_ascii_case(old) => {
431                hit = true;
432                crate::formula::Expr::Name(new.to_string())
433            }
434            o => o,
435        });
436        hit.then_some(out)
437    };
438    for i in 0..wb.sheets.len() {
439        if !applies(i) {
440            continue;
441        }
442        let Some(sh) = wb.sheet_mut(i) else { continue };
443        let formulas: Vec<(CellRef, std::sync::Arc<Formula>)> = sh.cells.iter().filter_map(|(c, x)| x.formula.clone().map(|f| (c, f))).collect();
444        for (c, f) in formulas {
445            let Some(e) = f.expr().and_then(rename) else { continue };
446            let mut nf = Formula::from_expr(e);
447            nf.array = f.array;
448            nf.dynamic = f.dynamic;
449            if let Some(cell) = sh.cells.get_mut(c) {
450                cell.formula = Some(std::sync::Arc::new(nf));
451            }
452        }
453    }
454    for n in wb.names.iter_mut() {
455        if n.scope.is_none_or(applies)
456            && let Some(e) = crate::formula::parse_name(&n.formula).ok().and_then(rename)
457        {
458            n.formula = crate::formula::print(&e);
459        }
460    }
461}

520fn create_from_selection(s: &mut Session, p: &Json) -> Result<Json> {
521    let r = target_range(s, p)?;
522    let top = bool_param(p, "top").unwrap_or(true);
523    let left = bool_param(p, "left").unwrap_or(false);
524    let bottom = bool_param(p, "bottom").unwrap_or(false);
525    let right = bool_param(p, "right").unwrap_or(false);
526    let replace = bool_param(p, "replace").unwrap_or(false);
527    let d = s.doc()?;
528    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
529    let q = crate::formula::quote_sheet(&sh.name);
530    let abs = |r: RangeRef| {
531        format!("{q}!${}${}:${}${}", crate::base::col_to_letters(r.start.col), r.start.row + 1, crate::base::col_to_letters(r.end.col), r.end.row + 1)
532    };
533    let clean = |t: String| -> String {
534        let mut n: String = t.trim().chars().map(|c| if c.is_alphanumeric() || c == '.' { c } else { '_' }).collect();
535        if !n.is_empty() && !valid_name(&n) {
536            n.insert(0, '_');
537        }
538        if valid_name(&n) { n } else { String::new() }
539    };
540    // The labels in the chosen edge rows/columns name the cells between them.
541    let (r0, r1) = (r.start.row + top as u32, r.end.row.saturating_sub(bottom as u32));
542    let (c0, c1) = (r.start.col + left as u32, r.end.col.saturating_sub(right as u32));
543    let mut made = Vec::new();
544    if r0 <= r1 && c0 <= c1 {
545        for (on, row) in [(top, r.start.row), (bottom, r.end.row)] {
546            for c in (c0..=c1).filter(|_| on) {
547                let name = clean(sh.value(CellRef::new(row, c)).display());
548                if !name.is_empty() {
549                    made.push((name, abs(RangeRef::new(CellRef::new(r0, c), CellRef::new(r1, c)))));
550                }
551            }
552        }
553        for (on, col) in [(left, r.start.col), (right, r.end.col)] {
554            for row in (r0..=r1).filter(|_| on) {
555                let name = clean(sh.value(CellRef::new(row, col)).display());
556                if !name.is_empty() {
557                    made.push((name, abs(RangeRef::new(CellRef::new(row, c0), CellRef::new(row, c1)))));
558                }
559            }
560        }
561    }
562    // Excel asks before replacing an existing name; here it's kept unless `replace` is set, and
563    // reported.
564    let exists = |n: &str| d.wb.names.iter().any(|x| x.name.eq_ignore_ascii_case(n) && x.scope.is_none());
565    // A label that names one of GridCraft's own hidden data names would replace part of an item
566    // (see `refuse_stash_name`).
567    let force = bool_param(p, "force").unwrap_or(false);
568    let ours = |n: &str| !force && holds_gridcraft_data(&d.wb, n);
569    let (made, skipped): (Vec<_>, Vec<_>) = made.into_iter().partition(|(n, _)| !ours(n) && (replace || !exists(n)));
570    let skipped: Vec<String> = skipped.into_iter().map(|(n, _)| n).collect();
571    let n = made.len();
572    edit(s, |cx| {
573        for (name, f) in made {
574            cx.wb.names.retain(|x| !(x.name.eq_ignore_ascii_case(&name) && x.scope.is_none()));
575            cx.wb.names.push(DefinedName { name, scope: None, formula: f, comment: String::new(), hidden: false });
576        }
577        cx.structural = true;
578        Ok(json!({"created": n, "skipped": skipped}))
579    })
580}

582/// The names Paste Names lists, as (name, refers-to text with `=`): the workbook's names and the
583/// active sheet's own, without hidden names, in alphabetical order. A sheet's name hides the
584/// workbook name it shares.
585pub fn pasteable_names(wb: &Workbook) -> Vec<(String, String)> {
586    let sheet = wb.active_sheet();
587    let shown = |local: bool| wb.names.iter().filter(move |n| !n.hidden && n.scope.is_some() == local && n.scope.is_none_or(|i| i == sheet));
588    let local: std::collections::HashSet<String> = shown(true).map(|n| crate::formula::fold_name(&n.name)).collect();
589    let mut out: Vec<(String, String)> = shown(true).map(|n| (n.name.clone(), format!("={}", n.formula))).collect();
590    for n in shown(false) {
591        if !local.contains(&crate::formula::fold_name(&n.name)) {
592            out.push((n.name.clone(), format!("={}", n.formula)));
593        }
594    }
595    out.sort_by_cached_key(|(n, _)| n.to_lowercase());
596    out
597}

599fn paste_name(s: &mut Session, p: &Json) -> Result<Json> {
600    let d = s.doc()?;
601    let names = pasteable_names(&d.wb);
602    let listed: Vec<Json> = names.iter().map(|(n, r)| json!({"name": n, "refersTo": r})).collect();
603    if !bool_param(p, "list").unwrap_or(false) {
604        if let Some(name) = str_param(p, "name").map(str::trim).filter(|n| !n.is_empty()) {
605            let Some((listed_name, _)) = names.iter().find(|(n, _)| crate::formula::same_name(n, name)) else {
606                return Err(EngineError::Other(format!("There's no name \"{name}\" to paste.")));
607            };
608            s.ui_requests.push(crate::engine::UiRequest::InsertName(listed_name.clone()));
609        }
610        return Ok(json!({"names": listed}));
611    }
612    if names.is_empty() {
613        return Err(EngineError::Other("This workbook has no names to paste.".into()));
614    }
615    let at = cell_param(s, p, "at")?.unwrap_or(d.selection.active);
616    let rows = u32::try_from(names.len()).unwrap_or(u32::MAX);
617    if at.row.saturating_add(rows) > crate::base::MAX_ROWS || at.col.saturating_add(2) > crate::base::MAX_COLS {
618        return Err(EngineError::Other("There isn't room on the sheet for the list of names.".into()));
619    }
620    let sheet = d.wb.active_sheet();
621    let range = RangeRef::new(at, CellRef::new(at.row + rows - 1, at.col + 1));
622    edit(s, |cx| {
623        for (row, (name, refers)) in (at.row..).zip(&names) {

625                let c = CellRef::new(row, col);
626                let sh = cx.sheet_mut(sheet)?;
627                let style = sh.style_id(c);
628                // The second column holds the refers-to text as a text value, with no formula.
629                sh.set_cell(c, Cell { value: Value::text(text.as_str()), formula: None, style, py: None });
630                cx.touch(sheet, c);
631            }
632        }
633        Ok(json!({"names": listed, "range": range.a1()}))
634    })
635}

656/// Trace Error goes back at most this many links; a longer chain stops there.
657const TRACE_ERROR_LINKS: usize = 10_000;
658
659/// Trace Error: from the cell, the first reference in its formula that shows the same error
660/// (in a range, the range's first cell showing it, row by row), then that cell's, and so on to the
661/// cell the error starts in: one none of whose references shows it, or one without references.
662/// Arrows link the cells on the way, and lead from the source's own references into it.
663fn trace_error(s: &mut Session, p: &Json) -> Result<Json> {
664    use std::collections::HashSet;
665    let d = s.doc()?;
666    let at = cell_param(s, p, "cell")?.unwrap_or(d.selection.active);
667    let sheet = d.wb.active_sheet();
668    let name = |si: usize| d.wb.sheet(si).map(|sh| sh.name.clone());
669    let cell = |(si, c): (usize, CellRef)| json!({"sheet": name(si), "cell": c.a1()});
670    let Some(error) = d.wb.sheet(sheet).and_then(|sh| sh.value(at).as_error()) else {
671        return Ok(json!({"sheet": name(sheet), "cell": at.a1(), "error": null, "source": null, "arrows": [], "truncated": false}));
672    };
673    // A value spilled by a dynamic array comes from its anchor's formula.
674    let precedents = |(si, c): (usize, CellRef)| {
675        let anchor = d.wb.sheet(si).filter(|sh| sh.cell(c).is_none_or(|x| x.formula.is_none())).and_then(|sh| sh.spill_anchor(c));
676        d.calc.graph.precedents_of((si, anchor.unwrap_or(c)))
677    };
678    // The walk keeps to cells it hasn't been to, so a circular reference ends it.
679    let mut chain = vec![(sheet, at)];
680    let mut seen: HashSet<(usize, CellRef)> = HashSet::from([(sheet, at)]);
681    let mut truncated = false;
682    let mut last = (sheet, at);
683    while let Some(next) = precedents(last).iter().find_map(|a| first_showing(&d.wb, a, error)) {
684        if !seen.insert(next) {
685            break;
686        }
687        if chain.len() > TRACE_ERROR_LINKS {
688            truncated = true;
689            break;
690        }
691        chain.push(next);
692        last = next;
693    }
694    let mut arrows = Vec::new();
695    // A chain cut short hasn't reached its source.
696    if !truncated {
697        let mut drawn = HashSet::new();
698        for a in precedents(last) {
699            if drawn.insert(a) {
700                arrows.push(json!({"from": {"sheet": name(a.sheet), "range": a.range.a1()}, "to": cell(last), "kind": "precedent"}));
701            }
702        }
703    }
704    // From the source outwards, as the error travels.
705    for pair in chain.windows(2).rev() {
706        if let [to, from] = pair {
707            arrows.push(json!({"from": {"sheet": name(from.0), "range": from.1.a1()}, "to": cell(*to), "kind": "error"}));
708        }
709    }
710    let source = if truncated { Json::Null } else { cell(last) };
711    Ok(json!({"sheet": name(sheet), "cell": at.a1(), "error": error.as_str(), "source": source, "arrows": arrows, "truncated": truncated}))
712}

714/// The first cell of an area, row by row, that shows `error`: a stored cell or a value spilled
715/// into the area by a dynamic array.
716fn first_showing(wb: &Workbook, a: &crate::calc::Area, error: CellError) -> Option<(usize, CellRef)> {
717    let sh = wb.sheet(a.sheet)?;
718    let holds = |c: &Cell| !c.value.is_empty() || c.formula.is_some();
719    let stored = sh.cells.iter_range(a.range).find(|(_, c)| holds(c) && c.value.as_error() == Some(error)).map(|(at, _)| at);
720    let spilled = sh
721        .spill
722        .range(a.range.start..)
723        .take_while(|(at, _)| **at <= a.range.end)
724        .find(|(at, v)| a.range.contains(**at) && v.as_error() == Some(error) && !sh.cell(**at).is_some_and(holds))
725        .map(|(at, _)| *at);
726    stored.into_iter().chain(spilled).min().map(|c| (a.sheet, c))
727}

729/// Excel's Ctrl+[ / Ctrl+]: select the precedent (or dependent) cells. When they all live on
730/// another sheet, jump there. `all` walks every level (Ctrl+Shift+[ / Ctrl+Shift+]).
731fn select_trace_cells(s: &mut Session, p: &Json, precedents: bool) -> Result<Json> {
732    use std::collections::BTreeSet;
733    let d = s.doc()?;
734    let at = cell_param(s, p, "cell")?.unwrap_or(d.selection.active);
735    let sheet = d.wb.active_sheet();
736    let all = bool_param(p, "all").unwrap_or(false);
737    // (sheet, a1) pairs to select; direct level keeps whole precedent ranges, transitive cells.
738    let mut found: BTreeSet<(usize, String)> = BTreeSet::new();
739    let mut seen: BTreeSet<(usize, CellRef)> = BTreeSet::new();
740    let mut frontier = vec![(sheet, at)];
741    seen.insert((sheet, at));
742    while let Some((ks, kc)) = frontier.pop() {
743        if precedents {
744            for a in d.calc.graph.precedents_of((ks, kc)) {
745                if all {
746                    // Walk only the stored cells inside the precedent area (sparse
747                    // range iterator, not the whole sheet): blanks are never selected
748                    // and only formula cells extend the frontier, so a formula late in
749                    // a large range is not skipped.
750                    let Some(sh) = d.wb.sheet(a.sheet) else { continue };
751                    for (c, cell) in sh.cells.iter_range(a.range) {
752                        if matches!(cell.value, Value::Empty) {
753                            continue;
754                        }
755                        if seen.insert((a.sheet, c)) {
756                            found.insert((a.sheet, c.a1()));
757                            if cell.formula.is_some() {
758                                frontier.push((a.sheet, c));
759                            }
760                        }
761                    }
762                } else {
763                    found.insert((a.sheet, a.range.a1()));
764                }
765            }
766        } else {
767            let mut deps = Vec::new();
768            d.calc.graph.dependents(ks, kc, &mut deps);
769            for k2 in deps {
770                if all {
771                    if seen.insert(k2) {
772                        found.insert((k2.0, k2.1.a1()));
773                        frontier.push(k2);
774                    }
775                } else {
776                    found.insert((k2.0, k2.1.a1()));
777                }
778            }
779        }
780        if !all {
781            break;
782        }
783    }
784    found.remove(&(sheet, at.a1()));
785    let name = if precedents { "precedent" } else { "dependent" };
786    let Some(si) = found.iter().find(|(si, _)| *si == sheet).or_else(|| found.first()).map(|(si, _)| *si) else {
787        return Err(bad("cell", format!("No {name} cells.")));
788    };
789    let list: Vec<String> = found.iter().filter(|(s, _)| *s == si).map(|(_, a)| a.clone()).collect();
790    let mut params = json!({ "range": list.join(",") });
791    if si != sheet {
792        params["sheet"] = json!(si);
793    }
794    s.execute("selection.set", params)
795}

850    let d = s.doc()?;
851    let mut out = Vec::new();
852    for sh in &d.wb.sheets {
853        for (c, cell) in sh.cells.iter() {
854            if let Some(e) = cell.value.as_error() {

856                    json!({"sheet": sh.name, "cell": c.a1(), "error": e.as_str(), "formula": cell.formula.as_ref().map(|f| format!("={}", f.text))}),

858            }

861            }
862        }
863    }
864    let circ: Vec<String> = d
865        .calc
866        .circular
867        .iter()
868        .map(|(si, c)| format!("{}!{}", d.wb.sheet(*si).map(|s| crate::formula::quote_sheet(&s.name)).unwrap_or_default(), c.a1()))
869        .collect();
870    Ok(json!({"errors": out, "circular": circ}))
871}

873fn evaluate_formula(s: &mut Session, p: &Json) -> Result<Json> {
874    let d = s.doc()?;
875    let at = cell_param(s, p, "cell")?.unwrap_or(d.selection.active);
876    let sheet = d.wb.active_sheet();
877    let text = match str_param(p, "formula") {
878        Some(f) => f.trim_start_matches('=').to_string(),
879        None => {
880            d.wb.sheet(sheet)
881                .and_then(|sh| sh.cell(at))
882                .and_then(|c| c.formula.as_ref())
883                .map(|f| f.text.clone())
884                .ok_or_else(|| EngineError::Other("The cell has no formula.".into()))?
885        }
886    };
887    let expr = crate::formula::parse(&text).map_err(|e| EngineError::Other(e.to_string()))?;
888    // Steps: each sub-expression (innermost first) with its value.
889    let mut steps = Vec::new();
890    let mut nodes = Vec::new();
891    expr.walk(&mut |e| {
892        // Borrowed, not cloned: copying every operator of a long chain A1+A2+… copies the
893        // rest of the chain each time (quadratic memory).
894        if matches!(e, crate::formula::Expr::Call(..) | crate::formula::Expr::Binary(..) | crate::formula::Expr::Ref(_)) {
895            nodes.push(e);
896        }
897    });
898    for e in nodes.into_iter().rev().take(200) {
899        let v = crate::calc::recalc::evaluate_expr(&d.wb, sheet, at, e);
900        steps.push(json!({"expression": crate::formula::print(e), "value": crate::engine::cmd::inspect::value_json(&v)}));
901    }
902    let result = crate::calc::recalc::evaluate_expr(&d.wb, sheet, at, &expr);
903    Ok(json!({"formula": format!("={text}"), "steps": steps, "result": crate::engine::cmd::inspect::value_json(&result)}))
904}

906/// Evaluates what typing `formula` into a cell would show. A formula `cell.set` would refuse (it
907/// doesn't parse, or calls a function with the wrong number of arguments) is an error that says
908/// why, not `#NAME?`; an unknown function name still evaluates to `#NAME?`, as in Excel.
909fn evaluate(s: &mut Session, p: &Json) -> Result<Json> {
910    let d = s.doc()?;
911    let f = str_param(p, "formula").ok_or_else(|| bad("formulas.evaluate", "missing `formula`"))?;
912    let at = cell_param(s, p, "cell")?.unwrap_or(d.selection.active);
913    let expr = crate::engine::cmd::edit::parse_typed_formula(f.strip_prefix('=').unwrap_or(f))?;
914    crate::engine::cmd::edit::check_arg_counts(&expr)?;
915    let v = crate::calc::recalc::evaluate_expr(&d.wb, d.wb.active_sheet(), at, &expr);
916    Ok(crate::engine::cmd::inspect::value_json(&v))
917}

927fn calc_options(s: &mut Session, p: &Json) -> Result<Json> {
928    let mode = match str_param(p, "mode") {
929        Some("manual") => Some(CalcMode::Manual),
930        Some("automaticExceptTables") => Some(CalcMode::AutomaticExceptTables),
931        Some("automatic") => Some(CalcMode::Automatic),
932        _ => None,
933    };
934    edit(s, |cx| {
935        let before = cx.wb.calc.clone();
936        if let Some(m) = mode {
937            cx.wb.calc.mode = m;
938        }
939        if let Some(v) = bool_param(p, "iterative") {
940            cx.wb.calc.iterative = v;
941        }
942        if let Some(v) = u32_param(p, "maxIterations") {
943            cx.wb.calc.set_max_iterations(v);
944        }
945        if let Some(v) = f64_param(p, "maxChange") {
946            cx.wb.calc.set_max_change(v);
947        }
948        let precision = bool_param(p, "precisionAsDisplayed");
949        if let Some(v) = precision {
950            cx.wb.calc.precision_as_displayed = v;
951            if v && !before.precision_as_displayed {
952                crate::engine::cmd::round_all_constants_as_displayed(&mut cx.wb);
953            }
954        }
955        let c = &cx.wb.calc;
956        // Switching to automatic, or changing how circular references are calculated outside
957        // manual mode, recalculates the workbook.
958        let iteration_changed = (c.iterative, c.max_iterations, c.max_change) != (before.iterative, before.max_iterations, before.max_change);
959        // Switching precision as displayed recalculates the workbook, in manual mode too, so the
960        // values change as soon as it is switched on or off.
961        let precision_changed = c.precision_as_displayed != before.precision_as_displayed;
962        if mode == Some(CalcMode::Automatic) || (iteration_changed && c.mode != CalcMode::Manual) || precision_changed {
963            cx.structural = true;
964        }
965        Ok(json!({
966            "mode": format!("{:?}", c.mode),
967            "iterative": c.iterative,
968            "maxIterations": c.max_iterations,
969            "maxChange": c.max_change,
970            "precisionAsDisplayed": c.precision_as_displayed,
971        }))
972    })
973}
