3pub(super) fn table_resize(s: &mut Session, p: &Json) -> Result<Json> {
4    let r = str_param(p, "range").and_then(RangeRef::parse).ok_or_else(|| bad("table.resize", "missing `range`"))?;
5    let d = s.doc()?;
6    let si = d.wb.active_sheet();
7    let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
8    let ti = match str_param(p, "table") {
9        Some(n) => sh.tables.iter().position(|t| t.name.eq_ignore_ascii_case(n)),
10        None => sh.tables.iter().position(|t| t.range.contains(d.selection.active)),
11    }
12    .ok_or_else(|| bad("table.resize", "select a cell in a table"))?;
13    edit(s, |cx| {
14        let sheet = cx.sheet_mut(si)?;
15        let headers: Vec<String> = (r.start.col..=r.end.col).map(|c| sheet.value(CellRef::new(r.start.row, c)).display()).collect();
16        if sheet.tables.iter().enumerate().any(|(i, o)| i != ti && o.range.intersects(&r)) {
17            return Err(EngineError::Other("A table can't overlap another table.".into()));
18        }
19        let t = sheet.tables.get_mut(ti).ok_or_else(|| bad("table", "gone"))?;
20        if r.start.row != t.range.start.row || !r.intersects(&t.range) {
21            return Err(EngineError::Other(
22                "The headers must remain in the same row, and the resulting table range must overlap the original table range.".into(),
23            ));
24        }
25        let mut cols = Vec::new();
26        for (i, h) in headers.iter().enumerate() {
27            let name = if h.is_empty() { format!("Column{}", i + 1) } else { h.clone() };
28            let existing = t.columns.iter().find(|c| c.name.eq_ignore_ascii_case(&name)).cloned();
29            cols.push(existing.unwrap_or(TableColumn {
30                name,
31                totals: TotalsFn::None,
32                totals_label: None,
33                formula: None,
34                unique_name: None,
35                xml: None,
36            }));
37        }
38        t.columns = cols;
39        t.range = r;
40        cx.structural = true;
41        Ok(Json::Null)
42    })
43}

45pub(super) fn from_csv(s: &mut Session, p: &Json) -> Result<Json> {
46    let bytes = if let Some(t) = str_param(p, "text") {
47        t.as_bytes().to_vec()
48    } else if let Some(path) = str_param(p, "path") {
49        crate::engine::io::read_file(path)?
50    } else {
51        s.ui_requests.push(crate::engine::UiRequest::Dialog("open".into(), json!({})));
52        return ok();
53    };
54    let text = String::from_utf8_lossy(&bytes);
55    // JSON array of objects → rows.
56    let rows: Vec<Vec<Json>> =
57        if text.trim_start().starts_with('[') {
58            let arr: Vec<Json> = serde_json::from_str(&text).map_err(|e| bad("data.getData", e.to_string()))?;
59            let mut keys: Vec<String> = Vec::new();
60            for o in &arr {
61                if let Some(m) = o.as_object() {
62                    for k in m.keys() {
63                        if !keys.contains(k) {
64                            keys.push(k.clone());
65                        }
66                    }
67                }
68            }
69            let mut out = vec![keys.iter().map(|k| json!(k)).collect::<Vec<_>>()];
70            for o in &arr {
71                out.push(keys.iter().map(|k| o.get(k).cloned().unwrap_or(Json::Null)).collect());
72            }
73            out
74        } else {
75            let delim = str_param(p, "delimiter")
76                .and_then(|d| d.chars().next())
77                .unwrap_or(if text.lines().next().is_some_and(|l| l.contains('\t')) { '\t' } else { ',' });
78            let opts = crate::xlsx::CsvOptions { delimiter: delim as u8, ..Default::default() };
79            let wb = crate::xlsx::read_csv(&bytes, &opts).map_err(|e| EngineError::Other(e.to_string()))?;
80            let sh = wb.sheet(0).ok_or(EngineError::NoDocument)?;
81            let Some(u) = sh.used_range() else { return Err(EngineError::Other("The file is empty.".into())) };
82            (u.start.row..=u.end.row)
83                .map(|r| {
84                    (u.start.col..=u.end.col)
85                        .map(|c| match sh.value(CellRef::new(r, c)) {
86                            Value::Number(n) => json!(n),
87                            Value::Bool(b) => json!(b),
88                            Value::Empty => Json::Null,
89                            v => json!(v.display()),
90                        })
91                        .collect()
92                })
93                .collect()
94        };
95    if rows.is_empty() || rows.len() > 1_048_576 {
96        return Err(EngineError::Other("Nothing to import.".into()));
97    }
98    let name = str_param(p, "sheet").map(str::to_string).or_else(|| {
99        str_param(p, "path").and_then(|x| std::path::Path::new(x).file_stem().and_then(std::ffi::OsStr::to_str).map(|f| f.chars().take(31).collect()))
100    });
101    let r = s.execute("home.insertSheet", json!({"name": name})).or_else(|_| s.execute("home.insertSheet", json!({})))?;
102    let w = rows.iter().map(Vec::len).max().unwrap_or(1).max(1);
103    s.execute("range.setValues", json!({"range": "A1", "values": rows}))?;
104    let range = RangeRef::new(CellRef::new(0, 0), CellRef::new(rows.len() as u32 - 1, w as u32 - 1));
105    let _ = s.execute("insert.table", json!({"range": range.a1(), "header": true}));
106    let _ = s.execute("home.autofitColumnWidth", json!({"cols": RangeRef::cols(0, w as u32 - 1).a1()}));
107    Ok(json!({"sheet": r["name"], "rows": rows.len(), "cols": w}))
108}

110pub(super) fn from_table_range(s: &mut Session, p: &Json) -> Result<Json> {
111    let r = target_range(s, p)?;
112    let d = s.doc()?;
113    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
114    let r = if r.is_single() { crate::engine::selection::current_region(sh, r.start) } else { r };
115    let text = crate::engine::display::range_text(&d.wb, d.wb.active_sheet(), r);
116    from_csv(s, &json!({"text": text, "delimiter": "\t", "sheet": format!("{} (2)", sh.name).chars().take(31).collect::<String>()}))
117}

119pub(super) fn queries(s: &mut Session, _: &Json) -> Result<Json> {
120    let d = s.doc()?;
121    Ok(Json::Array(
122        d.wb.sheets
123            .iter()
124            .flat_map(|sh| {
125                sh.tables
126                    .iter()
127                    .map(move |t| json!({"name": t.name, "sheet": sh.name, "range": t.range.a1(), "rows": t.range.height().saturating_sub(1)}))
128            })
129            .collect(),
130    ))
