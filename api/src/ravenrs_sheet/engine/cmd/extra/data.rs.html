3pub(super) fn table_resize(s: &mut Session, p: &Json) -> Result<Json> {
4    let r = str_param(p, "range").and_then(RangeRef::parse).ok_or_else(|| bad("table.resize", "missing `range`"))?;
5    let d = s.doc()?;
6    let si = d.wb.active_sheet();
7    let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
8    let ti = match str_param(p, "table") {
9        Some(n) => sh.tables.iter().position(|t| t.name.eq_ignore_ascii_case(n)),
10        None => sh.tables.iter().position(|t| t.range.contains(d.selection.active)),
11    }
12    .ok_or_else(|| bad("table.resize", "select a cell in a table"))?;
13    edit(s, |cx| {
14        let sheet = cx.sheet_mut(si)?;
15        let headers: Vec<String> = (r.start.col..=r.end.col).map(|c| sheet.value(CellRef::new(r.start.row, c)).display()).collect();
16        if sheet.tables.iter().enumerate().any(|(i, o)| i != ti && o.range.intersects(&r)) {
17            return Err(EngineError::Other("A table can't overlap another table.".into()));
18        }
19        let t = sheet.tables.get_mut(ti).ok_or_else(|| bad("table", "gone"))?;
20        if r.start.row != t.range.start.row || !r.intersects(&t.range) {
21            return Err(EngineError::Other(
22                "The headers must remain in the same row, and the resulting table range must overlap the original table range.".into(),
23            ));
24        }
25        let mut cols = Vec::new();
26        for (i, h) in headers.iter().enumerate() {
27            let name = if h.is_empty() { format!("Column{}", i + 1) } else { h.clone() };
28            let existing = t.columns.iter().find(|c| c.name.eq_ignore_ascii_case(&name)).cloned();
29            cols.push(existing.unwrap_or(TableColumn {
30                name,
31                totals: TotalsFn::None,
32                totals_label: None,
33                formula: None,
34                unique_name: None,
35                xml: None,
36            }));
37        }
38        t.columns = cols;
39        t.range = r;
40        cx.structural = true;
41        Ok(Json::Null)
42    })
43}

45/// Most cells (rows × columns, blanks included) Get Data imports. The rows are built densely
46/// before they are written, so a few sparse records or one far-away CSV cell must not expand
47/// into a rectangle of billions of slots.
48const MAX_IMPORT_CELLS: u64 = 10_000_000;
49
50fn too_many_cells() -> EngineError {
51    EngineError::Other(format!("The data is too large to import ({MAX_IMPORT_CELLS} cells at most)."))
52}

54pub(super) fn from_csv(s: &mut Session, p: &Json) -> Result<Json> {
55    let bytes = if let Some(t) = str_param(p, "text") {
56        t.as_bytes().to_vec()
57    } else if let Some(path) = str_param(p, "path") {
58        crate::engine::io::read_file(path)?
59    } else {
60        s.ui_requests.push(crate::engine::UiRequest::Dialog("open".into(), json!({})));
61        return ok();
62    };
63    let text = String::from_utf8_lossy(&bytes);
64    // JSON array of objects → rows.
65    let rows: Vec<Vec<Json>> =
66        if text.trim_start().starts_with('[') {
67            let arr: Vec<Json> = serde_json::from_str(&text).map_err(|e| bad("data.getData", e.to_string()))?;
68            let mut keys: Vec<String> = Vec::new();
69            let mut seen: std::collections::HashSet<&str> = std::collections::HashSet::new();
70            for o in &arr {
71                if let Some(m) = o.as_object() {
72                    for k in m.keys() {
73                        if !seen.contains(k.as_str()) {
74                            // Stop collecting at the column limit.
75                            if keys.len() >= crate::base::MAX_COLS as usize {
76                                return Err(EngineError::Other("The data has more fields than a sheet has columns.".into()));
77                            }
78                            seen.insert(k.as_str());
79                            keys.push(k.clone());
80                        }
81                    }
82                }
83            }
84            if arr.len() as u64 + 1 > u64::from(crate::base::MAX_ROWS) {
85                return Err(EngineError::Other("The data has more rows than a sheet has.".into()));
86            }
87            if (arr.len() as u64 + 1).saturating_mul(keys.len().max(1) as u64) > MAX_IMPORT_CELLS {
88                return Err(too_many_cells());
89            }
90            let mut out = vec![keys.iter().map(|k| json!(k)).collect::<Vec<_>>()];
91            for o in &arr {
92                out.push(keys.iter().map(|k| o.get(k).cloned().unwrap_or(Json::Null)).collect());
93            }
94            out
95        } else {
96            let delim = str_param(p, "delimiter")
97                .and_then(|d| d.chars().next())
98                .unwrap_or(if text.lines().next().is_some_and(|l| l.contains('\t')) { '\t' } else { ',' });
99            let opts = crate::xlsx::CsvOptions { delimiter: delim as u8, max_used_cells: Some(MAX_IMPORT_CELLS), ..Default::default() };
100            // The cell limit is enforced while parsing, before an oversized file is stored.
101            let wb = crate::xlsx::read_csv(&bytes, &opts).map_err(|e| match e {
102                crate::xlsx::IoError::TooLarge(_) => too_many_cells(),
103                e => EngineError::Other(e.to_string()),
104            })?;
105            let sh = wb.sheet(0).ok_or(EngineError::NoDocument)?;
106            let Some(u) = sh.used_range() else { return Err(EngineError::Other("The file is empty.".into())) };
107            if u.count() > MAX_IMPORT_CELLS {
108                return Err(too_many_cells());
109            }
110            (u.start.row..=u.end.row)
111                .map(|r| {
112                    (u.start.col..=u.end.col)
113                        .map(|c| match sh.value(CellRef::new(r, c)) {
114                            Value::Number(n) => json!(n),
115                            Value::Bool(b) => json!(b),
116                            Value::Empty => Json::Null,
117                            v => json!(v.display()),
118                        })
119                        .collect()
120                })
121                .collect()
122        };
123    if rows.is_empty() || rows.len() > 1_048_576 {
124        return Err(EngineError::Other("Nothing to import.".into()));
125    }
126    let name = str_param(p, "sheet").map(str::to_string).or_else(|| {
127        str_param(p, "path").and_then(|x| std::path::Path::new(x).file_stem().and_then(std::ffi::OsStr::to_str).map(|f| f.chars().take(31).collect()))
128    });
129    let r = s.execute("home.insertSheet", json!({"name": name})).or_else(|_| s.execute("home.insertSheet", json!({})))?;
130    let w = rows.iter().map(Vec::len).max().unwrap_or(1).max(1);
131    s.execute("range.setValues", json!({"range": "A1", "values": rows}))?;
132    let range = RangeRef::new(CellRef::new(0, 0), CellRef::new(rows.len() as u32 - 1, w as u32 - 1));
133    let _ = s.execute("insert.table", json!({"range": range.a1(), "header": true}));
134    let _ = s.execute("home.autofitColumnWidth", json!({"cols": RangeRef::cols(0, w as u32 - 1).a1()}));
135    Ok(json!({"sheet": r["name"], "rows": rows.len(), "cols": w}))
136}

138pub(super) fn from_table_range(s: &mut Session, p: &Json) -> Result<Json> {
139    let r = target_range(s, p)?;
140    let d = s.doc()?;
141    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
142    let r = if r.is_single() { crate::engine::selection::current_region(sh, r.start) } else { r };
143    let text = crate::engine::display::range_text(&d.wb, d.wb.active_sheet(), r);
144    from_csv(s, &json!({"text": text, "delimiter": "\t", "sheet": format!("{} (2)", sh.name).chars().take(31).collect::<String>()}))
145}

147pub(super) fn queries(s: &mut Session, _: &Json) -> Result<Json> {
148    let d = s.doc()?;
149    Ok(Json::Array(
150        d.wb.sheets
151            .iter()
152            .flat_map(|sh| {
153                sh.tables
154                    .iter()
155                    .map(move |t| json!({"name": t.name, "sheet": sh.name, "range": t.range.a1(), "rows": t.range.height().saturating_sub(1)}))
156            })
157            .collect(),
158    ))
159}

162mod tests {
163    use serde_json::json;

165    use crate::engine::Session;

167    fn s() -> Session {
168        let mut s = Session::new();
169        s.new_workbook();
170        s

172
173    #[test]
174    fn sparse_imports_are_refused_before_their_rows_are_built() {
175        let mut s = s();
176        let sheets = s.doc().unwrap().wb.sheets.len();
177        // 5,000 records with one field each: 5,001 × 5,000 slots once every record gets every field.
178        let records: Vec<_> = (0..5_000).map(|i| json!({ format!("k{i}"): i })).collect();
179        let text = serde_json::to_string(&records).unwrap();
180        let err = s.execute("data.getData", json!({"text": text})).expect_err("too many cells");
181        assert!(err.to_string().contains("too large"), "{err}");
182        // A CSV whose last line reaches 16,001 columns, 1,000 lines down.
183        let text = format!("1{}{}x", "\n".repeat(999), ",".repeat(16_000));
184        let err = s.execute("data.fromTextCsv", json!({"text": text})).expect_err("too many cells");
185        assert!(err.to_string().contains("too large"), "{err}");
186        assert_eq!(s.doc().unwrap().wb.sheets.len(), sheets);
187        // Small imports still work.
188        let out = s.execute("data.getData", json!({"text": r#"[{"a": 1}, {"b": 2}]"#})).unwrap();
189        assert_eq!((out["rows"].as_u64(), out["cols"].as_u64()), (Some(3), Some(2)), "{out}");
190    }
