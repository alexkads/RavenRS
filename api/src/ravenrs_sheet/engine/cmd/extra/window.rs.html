6/// Switches a sheet's view. Like zoom, the view is saved with the file but is not an edit: it
7/// neither records an undo step nor marks a clean workbook as changed.
8pub(super) fn set_view(s: &mut Session, p: &Json, mode: SheetViewMode) -> Result<Json> {
9    let si = target_sheet(s, p)?;
10    let d = s.doc_mut()?;
11    let dirty = d.is_dirty();
12    let wb = std::sync::Arc::make_mut(&mut d.wb);
13    if let Some(sh) = wb.sheet_mut(si) {
14        sh.view = mode;
15    }
16    if !dirty {
17        d.saved = d.wb.clone();
18    }
19    Ok(json!({"view": mode.as_str()}))
20}

28/// The Watch Window list isn't an edit (Excel doesn't undo it): undo and redo keep the current
29/// list when they swap in another snapshot of the workbook.
30pub(crate) fn keep_watch_list(from: &Workbook, to: &mut std::sync::Arc<Workbook>) {
31    let find = |wb: &Workbook| wb.names.iter().find(|n| n.name == WATCH_NAME).cloned();
32    let keep = find(from);
33    if keep == find(to) {
34        return;
35    }
36    let wb = std::sync::Arc::make_mut(to);
37    wb.names.retain(|n| n.name != WATCH_NAME);
38    wb.names.extend(keep);
39}

41/// Normal, Page Layout and Page Break Preview belong to the window, not to the edits: Excel doesn't
42/// undo a switch between them, and `set_view` is no undo step. So undo and redo keep each sheet's
43/// current view when they swap in another snapshot of the workbook, and a view kept this way
44/// doesn't make a saved workbook dirty.
45pub(crate) fn keep_view_modes(d: &mut crate::engine::DocState, now: &Workbook) {
46    // Sheets have no stable id. A name missing from `now` is a rename being undone or redone, which
47    // keeps the sheet count and position: the sheet is the one at the same index, under a name the
48    // restored snapshot doesn't have.
49    let renamed = |i: usize| {
50        (now.sheets.len() == d.wb.sheets.len()).then(|| now.sheets.get(i)).flatten().filter(|sh| !d.wb.sheets.iter().any(|s| s.name == sh.name))
51    };
52    let current = |i: usize, name: &str| now.sheets.iter().find(|sh| sh.name == name).or_else(|| renamed(i)).map(|sh| sh.view);
53    let changed: Vec<(usize, SheetViewMode)> =
54        d.wb.sheets.iter().enumerate().filter_map(|(i, sh)| current(i, &sh.name).filter(|v| *v != sh.view).map(|v| (i, v))).collect();
55    if changed.is_empty() {
56        return;
57    }
58    let dirty = d.is_dirty();
59    let wb = std::sync::Arc::make_mut(&mut d.wb);
60    for (i, v) in changed {
61        if let Some(sh) = wb.sheet_mut(i) {
62            sh.view = v;
63        }
64    }
65    if !dirty {
66        d.saved = d.wb.clone();
67    }
68}

70pub(super) fn custom_views(s: &mut Session, p: &Json) -> Result<Json> {
71    let d = s.doc()?;
72    let mut views = stash_map(&d.wb, VIEWS_NAME);
73    if let Some(name) = str_param(p, "save") {
74        let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
75        views[name] = json!({"sheet": d.wb.active_sheet(), "selection": d.selection.a1(), "zoom": sh.zoom, "freeze": sh.freeze});
76        let v = views.clone();
77        edit(s, |cx| {
78            stash_set(&mut cx.wb, VIEWS_NAME, &v);
79            Ok(())
80        })?;
81    } else if let Some(name) = str_param(p, "show") {
82        let v = views.get(name).cloned().ok_or_else(|| bad("view.customViews", "no such view"))?;
83        if let Some(si) = v["sheet"].as_u64() {
84            s.execute("sheet.activate", json!({"sheet": si}))?;
85        }
86        if let Some(sel) = v["selection"].as_str() {
87            s.execute("selection.set", json!({"range": sel}))?;
88        }
89        if let Some(z) = v["zoom"].as_u64() {
90            s.execute("view.zoom", json!({"percent": z}))?;
91        }
92        if let Some(f) = v.get("freeze") {
93            let freeze = match (f[0].as_u64(), f[1].as_u64()) {
94                (Some(r), Some(c)) => u32::try_from(r).ok().zip(u32::try_from(c).ok()),
95                _ => None,
96            };
97            edit(s, |cx| {
98                let si = cx.sheet_index();
99                cx.sheet_mut(si)?.freeze = freeze;
100                Ok(())
101            })?;
102        }
103    } else if let Some(name) = str_param(p, "delete") {
104        if let Some(o) = views.as_object_mut() {
105            o.remove(name);
106        }
107        let v = views.clone();
108        edit(s, |cx| {
109            stash_set(&mut cx.wb, VIEWS_NAME, &v);
110            Ok(())
111        })?;
112    }
113    Ok(stash_get(&s.doc()?.wb, VIEWS_NAME))
114}

116pub(super) fn split(s: &mut Session, _: &Json) -> Result<Json> {
117    let has = s.doc()?.wb.active().is_some_and(|sh| sh.freeze.is_some());
118    if has { s.execute("view.unfreezePanes", json!({})) } else { s.execute("view.freezePanes", json!({})) }
119}

127/// The window a window command names: `window` (an id), else `doc` (a workbook id: its most
128/// recent window), else the active window.
129pub(crate) fn window_param(s: &Session, cmd: &str, p: &Json) -> Result<u64> {
130    let t = crate::engine::Target::from_json(p).map_err(|e| bad(cmd, e.to_string()))?;
131    s.target_window(t).map_err(|e| bad(cmd, e.to_string()))
132}

134/// View › Arrange All. The engine has no window geometry: it picks the windows and the layout,
135/// and the interface moves its native windows ([`Session::window_list`] order, active first).
136pub(super) fn arrange_all(s: &mut Session, p: &Json) -> Result<Json> {
137    let arrange = match str_param(p, "arrange").unwrap_or("tiled") {
138        a @ ("tiled" | "horizontal" | "vertical" | "cascade") => a,
139        other => return Err(bad("view.arrangeAll", format!("`arrange` must be tiled, horizontal, vertical or cascade, not `{other}`"))),
140    };
141    let active = s.active_window();
142    let doc = active.and_then(|w| s.window(w)).map(|w| w.doc_uid);
143    let only_active = p.get("activeWorkbook").and_then(Json::as_bool).unwrap_or(false);
144    let mut windows: Vec<u64> = s.windows().iter().filter(|w| !w.hidden && (!only_active || Some(w.doc_uid) == doc)).map(|w| w.id).collect();
145    if let Some(at) = windows.iter().position(|w| Some(*w) == active)
146        && let Some(first) = windows.get_mut(..=at)
147    {
148        first.rotate_right(1);
149    }
150    Ok(json!({"arrange": arrange, "windows": windows}))
151}

159pub(super) fn unhide_window(s: &mut Session, p: &Json) -> Result<Json> {
160    let id = if p.get("window").is_some_and(|v| !v.is_null()) || p.get("doc").is_some_and(|v| !v.is_null()) {
161        Some(window_param(s, "view.unhideWindow", p)?)
162    } else if let Some(i) = u32_param(p, "index") {
163        // A workbook's position in the document list.
164        let uid = s.documents().get(i as usize).map(|d| d.uid).ok_or_else(|| bad("view.unhideWindow", "no such window"))?;
165        s.windows().iter().filter(|w| w.doc_uid == uid && w.hidden).map(|w| w.id).next_back().or_else(|| s.doc_window(uid))
166    } else {
167        s.last_hidden_window()
168    };
169    if let Some(id) = id {
170        s.unhide_window(id);
171    }
172    ok()
173}

183pub(super) fn watch_window(s: &mut Session, p: &Json) -> Result<Json> {
184    let mut list: Vec<String> =
185        stash_get(&s.doc()?.wb, WATCH_NAME).as_array().map(|a| a.iter().filter_map(|x| x.as_str().map(str::to_string)).collect()).unwrap_or_default();
186    let mut changed = false;
187    if let Some(a) = str_param(p, "add") {
188        if !list.iter().any(|x| x == a) {
189            list.push(a.to_string());
190        }
191        changed = true;
192    }
193    if let Some(r) = str_param(p, "remove") {
194        list.retain(|x| x != r);
195        changed = true;
196    }
197    if changed {
198        let v = json!(list.clone());
199        let d = s.doc_mut()?;
200        // The watch list is view state: don't add an undo step.
201        let wb = std::sync::Arc::make_mut(&mut d.wb);
202        stash_set(wb, WATCH_NAME, &v);
203    }
204    let d = s.doc()?;
205    let rows: Vec<Json> = list
206        .iter()
207        .map(|w| {
208            let (sheet, body) = split_sheet(w);
209            let si = sheet.and_then(|n| d.wb.sheet_index(&n)).unwrap_or(d.wb.active_sheet());
210            let c = CellRef::parse(body).unwrap_or_default();
211            let sh = d.wb.sheet(si);
212            json!({"cell": w, "value": sh.map(|s| s.value(c).display()), "formula": sh.and_then(|s| s.cell(c)).and_then(|x| x.formula.as_ref()).map(|f| format!("={}", f.text))})
213        })
214        .collect();
215    Ok(Json::Array(rows))
216}
