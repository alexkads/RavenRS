6/// Whether `c` has checkbox formatting.
7pub(crate) fn is_checkbox_cell(wb: &crate::model::Workbook, sh: &crate::model::Sheet, c: CellRef) -> bool {
8    wb.styles.get(sh.style_id(c)).num_fmt.as_str() == "checkbox"
9}

11pub(super) fn checkbox(s: &mut Session, p: &Json) -> Result<Json> {
12    let ranges = target_ranges(s, p)?;
13    let si = target_sheet(s, p)?;
14    edit(s, |cx| {
15        for r in &ranges {
16            if r.count() > 100_000 {
17                return Err(bad("insert.checkbox", "range too large"));
18            }
19            for c in r.iter() {
20                let st = cx.wb.sheet(si).map(|sh| sh.style_id(c)).unwrap_or_default();
21                let new = cx.wb.styles.derive(st, |s| {
22                    s.num_fmt = NumFmt::new("checkbox");
23                    s.align.h = HAlign::Center;
24                });
25                let sh = cx.sheet_mut(si)?;
26                // Values and formulas already there stay (only TRUE/FALSE draw as a box); an empty
27                // cell starts unchecked.
28                let mut cell = sh.cell(c).cloned().unwrap_or_default();
29                if cell.value.is_empty() && cell.formula.is_none() {
30                    cell.value = Value::Bool(false);
31                }
32                cell.style = new;
33                sh.set_cell(c, cell);
34                cx.touch(si, c);
35            }
36        }
37        Ok(Json::Null)
38    })
39}

41pub(super) fn toggle_checkbox(s: &mut Session, p: &Json) -> Result<Json> {
42    let si = target_sheet(s, p)?;
43    let d = s.doc()?;
44    let cells: Vec<CellRef> = match cell_param_on(s, p, "cell", si)? {
45        Some(c) => vec![c],
46        None => d.selection.ranges.iter().flat_map(|r| r.iter().take(100_000).collect::<Vec<_>>()).collect(),
47    };
48    edit(s, |cx| {
49        for c in cells {
50            let checkbox = cx.wb.sheet(si).is_some_and(|sh| is_checkbox_cell(&cx.wb, sh, c));
51            let sh = cx.sheet_mut(si)?;
52            if let Some(cell) = sh.cells.get_mut(c)
53                && cell.formula.is_none()
54                && cell.py.is_none()
55            {
56                // An empty checkbox cell counts as FALSE.
57                let next = match cell.value {
58                    Value::Bool(b) if checkbox => Value::Bool(!b),
59                    Value::Empty if checkbox => Value::Bool(true),
60                    _ => continue,
61                };
62                cell.value = next;
63                cx.changed.push((si, c));
64            }
65        }
66        Ok(Json::Null)
67    })
68}

70pub(super) fn theme_colors(s: &mut Session, p: &Json) -> Result<Json> {

72    if params.get("name").is_none() {

78pub(super) fn theme_fonts(s: &mut Session, p: &Json) -> Result<Json> {
79    let name = s.doc()?.wb.theme.name.clone();
80    let colors: Vec<String> = s.doc()?.wb.theme.colors.iter().map(|c| format!("#{c:06X}")).collect();
81    s.execute("pageLayout.theme", json!({"name": name, "colors": colors, "majorFont": p.get("major"), "minorFont": p.get("minor")}))
82}

84pub(super) fn background(s: &mut Session, p: &Json) -> Result<Json> {
85    if bool_param(p, "clear").unwrap_or(false) {
86        let si = s.doc()?.wb.active_sheet();
87        return edit(s, |cx| {
88            cx.sheet_mut(si)?.images.retain(|i| i.alt != "Sheet background");
89            Ok(Json::Null)
90        });
91    }
92    let mut params = object_params(p);
93    params["alt"] = json!("Sheet background");
94    params["at"] = json!("A1");
95    s.execute("insert.picture", params)
96}

98pub(super) fn draw_border(s: &mut Session, p: &Json) -> Result<Json> {
99    let edge = str_param(p, "edge").unwrap_or("grid");
100    let preset = match edge {
101        "top" => "top",
102        "bottom" => "bottom",
103        "left" => "left",
104        "right" => "right",
105        "outline" => "outside",
106        _ => "all",
107    };
108    let mut params = object_params(p);
109    params["preset"] = json!(preset);
110    s.execute("home.borders", params)
111}

113pub(super) fn new_cell_style(s: &mut Session, p: &Json) -> Result<Json> {
114    let name = str_param(p, "name").ok_or_else(|| bad("home.newCellStyle", "missing `name`"))?.to_string();
115    let d = s.doc()?;
116    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
117    let c = cell_param(s, p, "fromCell")?.unwrap_or(d.selection.active);
118    let style = d.wb.styles.get(sh.style_id(c)).clone();
119    edit(s, |cx| {
120        cx.wb.cell_styles.retain(|(n, _)| !n.eq_ignore_ascii_case(&name));
121        cx.wb.cell_styles.push((name.clone(), style.clone()));
122        Ok(json!({"name": name}))
123    })
124}
