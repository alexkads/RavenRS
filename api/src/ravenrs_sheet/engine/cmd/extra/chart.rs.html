6fn find_chart(s: &Session, p: &Json) -> Result<(usize, usize)> {
7    let d = s.doc()?;
8    let si = d.wb.active_sheet();
9    let sh = d.wb.sheet(si).ok_or(EngineError::NoDocument)?;
10    let idx = match u32_param(p, "chart") {
11        Some(id) => sh.charts.iter().position(|c| c.id == id),
12        None => sh.charts.len().checked_sub(1),
13    };
14    idx.map(|i| (si, i)).ok_or_else(|| bad("chart", "no chart on this sheet"))
15}

17fn with_chart(s: &mut Session, p: &Json, f: impl FnOnce(&mut Chart) -> Result<()>) -> Result<Json> {
18    let (si, ci) = find_chart(s, p)?;
19    edit(s, |cx| {
20        let c = cx.sheet_mut(si)?.charts.get_mut(ci).ok_or_else(|| bad("chart", "no such chart"))?;
21        f(c)?;
22        Ok(json!({"chart": c.id}))
23    })
24}

26pub(super) fn add_element(s: &mut Session, p: &Json) -> Result<Json> {
27    let el = str_param(p, "element").unwrap_or("title").to_string();
28    let on = bool_param(p, "on").unwrap_or(true);
29    let pos = str_param(p, "position").map(str::to_string);
30    with_chart(s, p, |c| {
31        match el.as_str() {
32            "title" => c.title = if on { c.title.clone().filter(|t| !t.is_empty()).or(Some("Chart Title".into())) } else { Some(String::new()) },
33            "legend" => {
34                c.legend = if !on {
35                    LegendPos::None
36                } else {
37                    match pos.as_deref() {
38                        Some("top") => LegendPos::Top,
39                        Some("left") => LegendPos::Left,
40                        Some("right") => LegendPos::Right,
41                        _ => LegendPos::Bottom,
42                    }
43                }
44            }
45            "dataLabels" => c.data_labels = on,
46            "dataTable" => crate::engine::cmd::insert::update_data_table(c, p, Some(on))?,
47            "gridlines" => c.gridlines = on,
48            "axisTitles" => {
49                c.x_title = on.then(|| c.x_title.clone().unwrap_or_else(|| "Axis Title".into()));
50                c.y_title = on.then(|| c.y_title.clone().unwrap_or_else(|| "Axis Title".into()));
51            }
52            "smooth" => {
53                for s in &mut c.series {
54                    s.smooth = on;
55                }
56            }
57            "trendline" => {
58                let only = u32_param(p, "series").map(|i| i as usize);
59                if only.is_some_and(|i| i >= c.series.len()) {
60                    return Err(bad("chart.addElement", "no such series"));
61                }
62                let targets: Vec<usize> = (0..c.series.len()).filter(|i| only.is_none_or(|o| o == *i)).collect();
63                if !on {
64                    for i in targets {
65                        if let Some(se) = c.series.get_mut(i) {
66                            se.trendlines.clear();
67                        }
68                    }
69                    return Ok(());
70                }
71                let t = trendline_param(p)?;
72                let eligible: Vec<usize> =
73                    targets.into_iter().filter(|i| crate::chart::supports_trendline(crate::chart::series_plot_kind(c, *i))).collect();
74                if eligible.is_empty() {
75                    return Err(bad("chart.addElement", "trendlines need a 2-D column, bar, line, area, scatter or bubble series"));
76                }
77                for i in eligible {
78                    if let Some(se) = c.series.get_mut(i) {
79                        se.trendlines.push(t.clone());
80                    }
81                }
82            }
83            other => return Err(bad("chart.addElement", format!("unknown element `{other}`"))),
84        }
85        Ok(())
86    })
87}

89/// The trendline described by `chart.addElement`'s trendline parameters (Excel's Add Chart
90/// Element ▸ Trendline choices, plus the Format Trendline options).
91fn trendline_param(p: &Json) -> Result<Trendline> {
92    let num = |k: &str| Trendline::clamp_forecast(f64_param(p, k).unwrap_or(0.0));
93    let mut t = Trendline {
94        forward: num("forward"),
95        backward: num("backward"),
96        display_equation: bool_param(p, "displayEquation").unwrap_or(false),
97        display_r_squared: bool_param(p, "displayRSquared").unwrap_or(false),
98        ..Trendline::default()
99    };
100    t.kind = match str_param(p, "type").unwrap_or("linear") {
101        "linear" => TrendlineKind::Linear,
102        // Excel's Linear Forecast: a linear trendline projected two periods ahead.
103        "linearForecast" => {
104            if t.forward == 0.0 {
105                t.forward = 2.0;
106            }
107            TrendlineKind::Linear
108        }
109        "exponential" => TrendlineKind::Exponential,
110        "logarithmic" => TrendlineKind::Logarithmic,
111        "power" => TrendlineKind::Power,
112        "polynomial" => TrendlineKind::Polynomial(u32_param(p, "order").unwrap_or(2).clamp(2, 6) as u8),
113        "movingAverage" => TrendlineKind::MovingAverage(u32_param(p, "period").unwrap_or(2).clamp(2, 255)),
114        other => return Err(bad("chart.addElement", format!("unknown trendline type `{other}`"))),
115    };
116    Ok(t)
117}

119pub(super) fn quick_layout(s: &mut Session, p: &Json) -> Result<Json> {
120    let n = u32_param(p, "layout").unwrap_or(1);
121    with_chart(s, p, |c| {
122        let (legend, labels, grid, axes) = match n {
123            2 => (LegendPos::Top, true, false, false),
124            3 => (LegendPos::Bottom, false, true, false),
125            4 => (LegendPos::Bottom, true, false, false),
126            5 => (LegendPos::None, false, true, true),
127            6 => (LegendPos::Right, false, true, true),
128            _ => (LegendPos::Right, false, true, false),
129        };
130        c.legend = legend;
131        c.data_labels = labels;
132        c.gridlines = grid;
133        if axes {
134            c.x_title.get_or_insert_with(|| "Axis Title".into());
135            c.y_title.get_or_insert_with(|| "Axis Title".into());
136        } else {
137            c.x_title = None;
138            c.y_title = None;
139        }
140        Ok(())
141    })
142}

144pub(super) fn change_colors(s: &mut Session, p: &Json) -> Result<Json> {
145    let explicit: Option<Vec<Color>> =
146        p.get("colors").and_then(Json::as_array).map(|a| a.iter().filter_map(|c| c.as_str().and_then(Color::from_hex)).collect());
147    let palette = str_param(p, "palette").unwrap_or("colorful").to_string();
148    with_chart(s, p, |c| {
149        for (i, se) in c.series.iter_mut().enumerate() {
150            se.color = match &explicit {
151                Some(cols) if !cols.is_empty() => cols.get(i % cols.len()).copied(),
152                _ => match palette.as_str() {
153                    "monochrome" => Some(Color::Theme(4, (-400 + (i as i16 * 200)).clamp(-800, 800))),
154                    p if p.starts_with("accent") => {
155                        let k: u8 = p.trim_start_matches("accent").parse().unwrap_or(1);
156                        Some(Color::Theme(3 + k.clamp(1, 6), (-500 + i as i16 * 250).clamp(-800, 800)))
157                    }
158                    _ => None,
159                },
160            };
161        }
162        Ok(())
163    })
164}

166pub(super) fn select_data(s: &mut Session, p: &Json) -> Result<Json> {
167    let d = s.doc()?;
168    // A new data range rebuilds the series, as in Excel's Select Data › Chart data range.
169    let rebuilt = match str_param(p, "range") {
170        Some(range) => {
171            let range = range.trim_start_matches('=');
172            let (sheet_name, body) = split_sheet(range);
173            let src_si = match sheet_name {
174                Some(n) => d.wb.sheet_index(&n).ok_or_else(|| bad("chart.selectData", "no such sheet"))?,
175                None => d.wb.active_sheet(),
176            };
177            let r = RangeRef::parse(body).ok_or_else(|| bad("chart.selectData", "not a range"))?;
178            if r.count() > 1_000_000 {
179                return Err(bad("chart.selectData", "range too large for a chart"));
180            }
181            let src = d.wb.sheet(src_si).ok_or(EngineError::NoDocument)?;
182            let by_rows = bool_param(p, "byRows").unwrap_or(r.width() > r.height() + 1);
183            let series = crate::engine::cmd::insert::series_from_range(src, r, by_rows);
184            Some((series, format!("{}!{}", crate::formula::quote_sheet(&src.name), r.a1()), by_rows))
185        }
186        None => None,
187    };
188    // Horizontal (Category) Axis Labels › Edit: one labels range for every series ("" clears).
189    let categories = match str_param(p, "categories") {
190        None => None,
191        Some(t) if t.trim().trim_start_matches('=').is_empty() => Some(None),
192        Some(t) => {
193            let t = t.trim().trim_start_matches('=');
194            let (sheet_name, body) = split_sheet(t);
195            let si = match sheet_name {
196                Some(n) => d.wb.sheet_index(&n).ok_or_else(|| bad("chart.selectData", "no such sheet"))?,
197                None => d.wb.active_sheet(),
198            };
199            let r = RangeRef::parse(body).ok_or_else(|| bad("chart.selectData", "`categories` is not a range"))?;
200            let name = d.wb.sheet(si).map(|sh| sh.name.clone()).ok_or(EngineError::NoDocument)?;
201            let abs = |c: CellRef| format!("${}${}", crate::base::col_to_letters(c.col), c.row + 1);
202            Some(Some(format!("{}!{}:{}", crate::formula::quote_sheet(&name), abs(r.start), abs(r.end))))
203        }
204    };
205    let mut remove: Vec<usize> = p
206        .get("removeSeries")
207        .and_then(Json::as_array)
208        .map(|a| a.iter().filter_map(Json::as_u64).filter_map(|i| usize::try_from(i).ok()).collect())
209        .unwrap_or_default();
210    remove.sort_unstable();
211    remove.dedup();
212    if rebuilt.is_none() && categories.is_none() && remove.is_empty() {
213        return Err(bad("chart.selectData", "missing `range`, `categories` or `removeSeries`"));
214    }
215    with_chart(s, p, move |c| {
216        if let Some((series, source, by_rows)) = rebuilt {
217            c.series = series;
218            crate::engine::cmd::insert::scatter_x_from_first(c.kind, &mut c.series);
219            c.source = Some(source);
220            c.by_rows = by_rows;
221        }
222        for i in remove.iter().rev() {
223            if *i < c.series.len() && c.series.len() > 1 {
224                c.series.remove(*i);
225            }
226        }
227        if let Some(cats) = categories {
228            for se in &mut c.series {
229                se.categories = cats.clone();
230            }
231        }
232        Ok(())
233    })
234}

236pub(super) fn move_chart(s: &mut Session, p: &Json) -> Result<Json> {
237    let (si, ci) = find_chart(s, p)?;
238    let target = match p.get("sheet").and_then(Json::as_str) {
239        Some(n) => Some(s.doc()?.wb.sheet_index(n).ok_or_else(|| bad("chart.move", "no such sheet"))?),
240        None => None,
241    };
242    let at = cell_param_on(s, p, "at", target.unwrap_or(si))?;
243    edit(s, |cx| {
244        let mut chart = cx.sheet_mut(si)?.charts.remove(ci);
245        if let Some(a) = at {
246            chart.anchor.cell = a;
247        }
248        let dest = target.unwrap_or(si);
249        cx.sheet_mut(dest)?.charts.push(chart);
250        Ok(Json::Null)
251    })
252}

254pub(super) fn format_series(s: &mut Session, p: &Json) -> Result<Json> {
255    let idx = u32_param(p, "series").unwrap_or(0) as usize;
256    let color = str_param(p, "color").and_then(Color::from_hex);
257    with_chart(s, p, |c| {
258        if let Some(se) = c.series.get_mut(idx) {
259            se.color = color;
260        }
261        Ok(())
262    })
