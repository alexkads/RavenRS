8pub(super) fn remove_duplicates(s: &mut Session, p: &Json) -> Result<Json> {
9    let (sheet, r) = data_target(s, p, "data.removeDuplicates")?;
10    let d = s.doc()?;
11    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
12    // Rows and columns past the data are blank: only the part with data is compared (whole
13    // columns compared a million blank rows and reported them as removed duplicates).
14    let r = sh.used_range().and_then(|u| r.intersection(&u)).unwrap_or(RangeRef::cell(r.start));
15    let header = bool_param(p, "header").unwrap_or_else(|| guess_header(&d.wb, sh, r));
16    let cols: Vec<u32> = p
17        .get("columns")
18        .and_then(Json::as_array)
19        .map(|a| a.iter().filter_map(|c| col_param(Some(c), r.start.col)).collect())
20        .unwrap_or_else(|| (r.start.col..=r.end.col).collect());
21    // With no column to compare, every row key is empty: the first row would survive and every
22    // other row would be deleted as a duplicate. A column outside the range compares data the
23    // user didn't select. Excel refuses both.
24    if cols.is_empty() || cols.iter().any(|c| !(r.start.col..=r.end.col).contains(c)) {
25        return Err(bad("data.removeDuplicates", "`columns` must name at least one column of the range"));
26    }
27    let start = r.start.row + header as u32;
28    let mut removed = 0;
29    edit(s, |cx| {
30        let Some(sh) = cx.wb.sheet(sheet) else { return Ok(()) };
31        let mut seen = std::collections::HashSet::new();
32        let mut keep: Vec<u32> = Vec::new();
33        // Cells are compared by the text they show, as in Excel: the same date in two formats
34        // isn't a duplicate, while 1.01 and 1.02 shown as 1.0 are.
35        let shown = |c: CellRef| crate::engine::display::format(&sh.value(c), cx.wb.styles.get(sh.style_id(c)).num_fmt.as_str(), &cx.wb).text;
36        for row in start..=r.end.row {
37            let key: Vec<String> = cols.iter().map(|c| shown(CellRef::new(row, *c)).to_lowercase()).collect();
38            if seen.insert(key) {
39                keep.push(row);
40            } else {
41                removed += 1;
42            }
43        }
44        let pictures: Vec<_> = keep
45            .iter()
46            .enumerate()
47            .flat_map(|(i, row)| {
48                (r.start.col..=r.end.col)
49                    .filter_map(move |col| sh.cell_pictures.get(&CellRef::new(*row, col)).map(|p| (CellRef::new(start + i as u32, col), p.clone())))
50            })
51            .collect();
52        // Rows a filter keeps are compacted to `start + i`, so the notes and hyperlinks that
53        // describe them move with them; the ones on removed rows are dropped with the row.
54        let notes: Vec<_> = keep
55            .iter()
56            .enumerate()
57            .flat_map(|(i, row)| {
58                (r.start.col..=r.end.col)
59                    .filter_map(move |col| sh.comments.get(&CellRef::new(*row, col)).map(|cm| (CellRef::new(start + i as u32, col), cm.clone())))
60            })
61            .collect();
62        let links: Vec<_> = keep
63            .iter()
64            .enumerate()
65            .flat_map(|(i, row)| {
66                (r.start.col..=r.end.col)
67                    .filter_map(move |col| sh.hyperlinks.get(&CellRef::new(*row, col)).map(|h| (CellRef::new(start + i as u32, col), h.clone())))
68            })
69            .collect();
70        let rows: Vec<Vec<Option<Cell>>> =
71            keep.iter().map(|row| (r.start.col..=r.end.col).map(|c| sh.cell(CellRef::new(*row, c)).cloned()).collect()).collect();
72        let shm = cx.sheet_mut(sheet)?;
73        for row in start..=r.end.row {
74            for c in r.start.col..=r.end.col {
75                shm.remove_cell(CellRef::new(row, c));
76            }
77        }
78        for row in start..=r.end.row {
79            for c in r.start.col..=r.end.col {
80                let at = CellRef::new(row, c);
81                shm.comments.remove(&at);
82                shm.hyperlinks.remove(&at);
83            }
84        }
85        for (i, cells) in rows.into_iter().enumerate() {
86            for (j, cell) in cells.into_iter().enumerate() {
87                if let Some(cell) = cell {
88                    shm.set_cell(CellRef::new(start + i as u32, r.start.col + j as u32), cell);
89                }
90            }
91        }
92        shm.cell_pictures.extend(pictures);
93        for (at, cm) in notes {
94            shm.comments.insert(at, cm);
95        }
96        for (at, h) in links {
97            shm.hyperlinks.insert(at, h);
98        }
99        for c in RangeRef::new(CellRef::new(start, r.start.col), r.end).iter() {
100            cx.changed.push((sheet, c));
101        }
102        Ok(())
103    })?;
104    let remaining = (r.end.row + 1 - start) as usize - removed;
105    Ok(json!({"removed": removed, "remaining": remaining}))
106}

108pub(super) fn text_to_columns(s: &mut Session, p: &Json) -> Result<Json> {
109    let r = target_range(s, p)?;
110    let sheet = target_sheet(s, p)?;
111    let mut delims: Vec<char> = p
112        .get("delimiters")
113        .and_then(Json::as_array)
114        .map(|a| a.iter().filter_map(|d| d.as_str().and_then(|x| x.chars().next())).collect())
115        .unwrap_or_else(|| vec!['\t']);
116    if let Some(o) = str_param(p, "other").and_then(|o| o.chars().next()) {
117        delims.push(o);
118    }
119    let fixed: Option<Vec<usize>> =
120        p.get("fixedWidths").and_then(Json::as_array).map(|a| a.iter().filter_map(|x| x.as_u64().map(|v| v as usize)).collect());
121    let consecutive = bool_param(p, "treatConsecutive").unwrap_or(false);
122    let quote = str_param(p, "textQualifier").and_then(|q| q.chars().next()).unwrap_or('"');
123    let dest = cell_param_on(s, p, "destination", sheet)?.unwrap_or(r.start);
124    edit(s, |cx| {
125        let Some(sh) = cx.wb.sheet(sheet) else { return Ok(()) };
126        let mut out: Vec<(CellRef, String)> = Vec::new();
127        for (i, row) in (r.start.row..=r.end.row).enumerate() {
128            let text = sh.value(CellRef::new(row, r.start.col)).display();
129            let parts: Vec<String> = match &fixed {
130                Some(w) => {
131                    let chars: Vec<char> = text.chars().collect();
132                    let mut cuts = vec![0];
133                    let mut breaks: Vec<usize> = w.iter().copied().filter(|x| *x > 0 && *x < chars.len()).collect();
134                    breaks.sort_unstable();
135                    breaks.dedup();
136                    cuts.extend(breaks);
137                    cuts.push(chars.len());
138                    cuts.windows(2)
139                        .map(|p| chars.get(p[0]..p[1]).map(|s| s.iter().collect::<String>().trim().to_string()).unwrap_or_default())
140                        .collect()
141                }
142                None => split_delimited(&text, &delims, consecutive, quote),
143            };
144            for (j, part) in parts.into_iter().enumerate() {
145                out.push((CellRef::new(dest.row + i as u32, dest.col + j as u32), part));
146            }
147        }
148        for (c, t) in out {
149            let cell = match crate::engine::cmd::edit::input_to_cell(&t, sheet, c, &mut cx.wb) {
150                Ok(cell) => cell,
151                // `=PY(…)` doesn't make a Python cell (that takes python.set): the piece is kept
152                // as text rather than lost.
153                Err(_) if crate::model::python::is_py_formula(&t) => {
154                    let style = cx.wb.sheet(sheet).map(|sh| sh.style_id(c)).unwrap_or_default();
155                    Some(Cell { value: crate::base::Value::text(t.as_str()), formula: None, style, py: None })
156                }
157                Err(_) => None,
158            };
159            let shm = cx.sheet_mut(sheet)?;
160            match cell {
161                Some(x) => shm.set_cell(c, x),
162                None => {
163                    shm.remove_cell(c);
164                }
165            }
166            cx.changed.push((sheet, c));
167        }
168        Ok(())
169    })?;
170    ok()
171}

173pub(super) fn split_delimited(text: &str, delims: &[char], consecutive: bool, quote: char) -> Vec<String> {
174    let mut out = Vec::new();
175    let mut cur = String::new();
176    let mut in_q = false;
177    let mut last_delim = false;
178    let mut chars = text.chars().peekable();
179    while let Some(c) = chars.next() {
180        if c == quote {
181            if in_q && chars.peek() == Some(&quote) {
182                cur.push(quote);
183                chars.next();
184            } else {
185                in_q = !in_q;
186            }
187            continue;
188        }
189        if !in_q && delims.contains(&c) {
190            if consecutive && last_delim {
191                continue;
192            }
193            out.push(std::mem::take(&mut cur));
194            last_delim = true;
195            continue;
196        }
197        last_delim = false;
198        cur.push(c);
199    }
200    out.push(cur);
201    out
202}
