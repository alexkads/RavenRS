14/// The sheet and range a sort works on: the sheet `sheet` or a `Sheet!` prefix on `range` names,
15/// else the active one. The selection belongs to the active sheet, so another sheet needs `range`.
16fn sort_target(s: &Session, p: &Json) -> Result<(usize, RangeRef)> {
17    let sheet = target_sheet(s, p)?;
18    let d = s.doc()?;
19    if let Some(t) = str_param(p, "range")
20        && let (Some(name), _) = split_sheet(t)
21        && d.wb.sheet_index(&name) != Some(sheet)
22    {
23        return Err(bad("data.sort", format!("`{t}` names a sheet other than `sheet`")));
24    }
25    if sheet != d.wb.active_sheet() && p.get("range").is_none() {
26        return Err(bad("data.sort", "missing `range`: the selection is on the active sheet"));
27    }
28    Ok((sheet, data_range(s, p)?))
29}

31pub(super) fn quick_sort(s: &mut Session, p: &Json, asc: bool) -> Result<Json> {
32    let (sheet, r) = sort_target(s, p)?;
33    let col = col_param(p.get("column"), r.start.col).unwrap_or(s.doc()?.selection.active.col).clamp(r.start.col, r.end.col);
34    let d = s.doc()?;
35    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
36    let header = guess_header(&d.wb, sh, r);
37    do_sort(s, sheet, r, header, vec![SortKey { col, desc: !asc, by: "values".into(), list: None }], false, false)
38}

40pub(super) fn sort(s: &mut Session, p: &Json) -> Result<Json> {
41    let (sheet, r) = sort_target(s, p)?;
42    let d = s.doc()?;
43    let sh = d.wb.sheet(sheet).ok_or(EngineError::NoDocument)?;
44    let header = bool_param(p, "header").unwrap_or_else(|| guess_header(&d.wb, sh, r));
45    let by_cols = str_param(p, "orientation") == Some("columns");
46    let case = bool_param(p, "matchCase").unwrap_or(false);
47    let mut keys: Vec<SortKey> = Vec::new();
48    for k in p.get("keys").and_then(Json::as_array).map(Vec::as_slice).unwrap_or_default() {
49        let col = if by_cols {
50            let row = match k.get("row").and_then(Json::as_u64) {
51                // Worksheet rows are 1-based; 0 (or a value past u32) has no coordinate.
52                Some(n) => {
53                    let row = u32::try_from(n).ok().and_then(|n| n.checked_sub(1));
54                    Some(row.ok_or_else(|| bad("data.sort", "sort key row must be 1 or greater"))?)
55                }
56                None => None,
57            };
58            row.or_else(|| col_param(k.get("column"), r.start.row))
59        } else {
60            col_param(k.get("column"), r.start.col)
61        };
62        let Some(col) = col else { continue };
63        keys.push(SortKey {
64            col,
65            desc: k.get("order").and_then(Json::as_str) == Some("desc"),
66            by: k.get("by").and_then(Json::as_str).unwrap_or("values").to_string(),
67            list: k.get("customList").and_then(Json::as_array).map(|l| l.iter().filter_map(|x| x.as_str().map(str::to_string)).collect()),
68        });
69    }
70    if keys.is_empty() {
71        s.ui_requests.push(crate::engine::UiRequest::Dialog("sort".into(), json!({"range": r.a1(), "header": header})));
72        return ok();
73    }
74    if keys.len() > 64 {
75        return Err(bad("data.sort", "too many sort levels (64 max)"));
76    }
77    do_sort(s, sheet, r, header, keys, by_cols, case)
78}

80fn do_sort(s: &mut Session, sheet: usize, r: RangeRef, header: bool, keys: Vec<SortKey>, by_cols: bool, case: bool) -> Result<Json> {
81    // Protect Sheet ▸ Sort. Both Sort… and the A-to-Z buttons come through here.
82    crate::engine::cmd::refuse_unless_allowed(s, sheet, "Sorting", |p| p.sort)?;
83    let body = if by_cols {
84        if header && r.width() > 1 { RangeRef::new(CellRef::new(r.start.row, r.start.col + 1), r.end) } else { r }
85    } else if header && r.height() > 1 {
86        RangeRef::new(CellRef::new(r.start.row + 1, r.start.col), r.end)
87    } else {
88        r
89    };
90    if body.count() > 20_000_000 {
91        return Err(bad("data.sort", "range too large"));
92    }
93    edit(s, |cx| {
94        let Some(sh) = cx.wb.sheet(sheet) else { return Ok(Json::Null) };
95        if sh.merges.iter().any(|m| m.intersects(&body) && !m.is_single()) {
96            return Err(EngineError::Other("To do this, all the merged cells need to be the same size.".into()));
97        }
98        let n = if by_cols { body.width() } else { body.height() };
99        let line = |i: u32, k: u32| if by_cols { CellRef::new(k, body.start.col + i) } else { CellRef::new(body.start.row + i, k) };
100        let theme = cx.wb.theme.clone();
101        let styles = &cx.wb.styles;
102        let key_val = |i: u32, k: &SortKey| -> (Value, u32) {
103            let c = line(i, k.col);
104            let v = sh.value(c);
105            let color = match k.by.as_str() {
106                "cellColor" => {
107                    styles.get(sh.style_id(c)).fill.fg.resolve(&theme).map(|[r, g, b]| u32::from_be_bytes([0, r, g, b])).unwrap_or(u32::MAX)
108                }
109                "fontColor" => {
110                    styles.get(sh.style_id(c)).font.color.resolve(&theme).map(|[r, g, b]| u32::from_be_bytes([0, r, g, b])).unwrap_or(u32::MAX)
111                }
112                _ => 0,
113            };
114            (v, color)
115        };
116        let mut order: Vec<u32> = (0..n).collect();
117        let keyed: Vec<Vec<(Value, u32)>> = (0..n).map(|i| keys.iter().map(|k| key_val(i, k)).collect()).collect();
118        order.sort_by(|&a, &b| {
119            for (ki, k) in keys.iter().enumerate() {
120                let (va, ca) = &keyed[a as usize][ki];
121                let (vb, cb) = &keyed[b as usize][ki];
122                let o = if k.by != "values" {
123                    ca.cmp(cb)
124                } else if let Some(list) = &k.list {
125                    let pos = |v: &Value| v.as_text().and_then(|t| list.iter().position(|x| x.eq_ignore_ascii_case(t))).unwrap_or(usize::MAX);
126                    pos(va).cmp(&pos(vb))
127                } else if case && va.is_text() && vb.is_text() {
128                    sort_compare(va, vb).then_with(|| case_order(va.as_text().unwrap_or(""), vb.as_text().unwrap_or("")))
129                } else {
130                    sort_compare(va, vb)
131                };
132                // Blanks always sort last, whatever the order.
133                let o = if k.desc && !(va.is_empty() || vb.is_empty()) { o.reverse() } else { o };
134                if o != Ordering::Equal {
135                    return o;
136                }
137            }
138            Ordering::Equal
139        });
140        if order.iter().enumerate().all(|(i, &o)| i as u32 == o) {
141            return Ok(Json::Null);
142        }
143        // Move whole lines (cells with formulas adjusted for relative references).
144        let lanes: Vec<u32> = if by_cols { (body.start.row..=body.end.row).collect() } else { (body.start.col..=body.end.col).collect() };
145        // Notes and hyperlinks hang off a cell, not a row, so they travel with the row they
146        // describe — the way cell pictures and row heights do. Collect them before inserting,
147        // so a swap can't resurrect one from a destination that is also a source.
148        let side: Vec<(CellRef, Comment)> = order
149            .iter()
150            .enumerate()
151            .flat_map(|(dst_i, &src_i)| lanes.iter().map(move |&k| (src_i, dst_i as u32, k)))
152            .filter_map(|(src_i, dst_i, k)| sh.comments.get(&line(src_i, k)).map(|cm| (line(dst_i, k), cm.clone())))
153            .collect();
154        let links: Vec<(CellRef, Hyperlink)> = order
155            .iter()
156            .enumerate()
157            .flat_map(|(dst_i, &src_i)| lanes.iter().map(move |&k| (src_i, dst_i as u32, k)))
158            .filter_map(|(src_i, dst_i, k)| sh.hyperlinks.get(&line(src_i, k)).map(|h| (line(dst_i, k), h.clone())))
159            .collect();
160        let mut moved: Vec<(CellRef, Option<Cell>)> = Vec::with_capacity(body.count() as usize);
161        let mut pictures = Vec::new();
162        for (dst_i, &src_i) in order.iter().enumerate() {
163            for &k in &lanes {
164                let src = line(src_i, k);
165                let dst = line(dst_i as u32, k);
166                let mut cell = sh.cell(src).cloned();
167                if let Some(c) = cell.as_mut()
168                    && let Some(f) = &c.formula
169                    && let Some(e) = f.expr()
170                {
171                    let shifted = crate::formula::adjust::shift_relative(e, dst.row as i64 - src.row as i64, dst.col as i64 - src.col as i64);
172                    c.formula = Some(Arc::new(Formula::from_expr(shifted)));
173                }
174                if let Some(picture) = sh.cell_pictures.get(&src) {
175                    pictures.push((dst, picture.clone()));
176                }
177                moved.push((dst, cell));
178            }
179        }
180        // Row heights travel with rows.
181        let heights: Vec<Option<LineInfo>> =
182            if by_cols { vec![] } else { order.iter().map(|&i| sh.rows.get(&(body.start.row + i)).copied()).collect() };
183        // Floating objects anchored inside the sorted body travel with their lines, unless they
184        // are pinned absolutely ("Don't move or size with cells", `editAs="absolute"`). Reordering
185        // is the inverse of `order`: `dst_of[src] = dst`.
186        let dst_of: Vec<u32> = {
187            let mut v = vec![0u32; order.len()];
188            for (dst_i, &src_i) in order.iter().enumerate() {
189                if let Some(slot) = v.get_mut(src_i as usize) {
190                    *slot = dst_i as u32;
191                }
192            }
193            v
194        };
195        // The object's own geometry (its top-left anchor cell, offset and size) is read from the
196        // pre-sort `sh`; a two-cell object's bottom-right cell is derived from its size here so it
197        // can be re-fitted to the lines it spans after the sort.
198        let objs: Vec<Anchor> = sh.object_anchors().copied().collect();
199        let descs: Vec<(u32, u32, f32, f32)> = objs
200            .iter()
201            .map(|a| {
202                let x2 = sh.col_left(a.cell.col) + a.dx as f64 + a.width.max(0.0) as f64;
203                let y2 = sh.row_top(a.cell.row) + a.dy as f64 + a.height.max(0.0) as f64;
204                let tc = sh.col_at(x2);
205                let tr = sh.row_at(y2);
206                (tc, tr, (x2 - sh.col_left(tc)) as f32, (y2 - sh.row_top(tr)) as f32)
207            })
208            .collect();
209        let shm = cx.sheet_mut(sheet)?;
210        for (c, cell) in &moved {
211            match cell {
212                Some(x) => shm.set_cell(*c, x.clone()),
213                None => {
214                    shm.remove_cell(*c);
215                }
216            }
217        }
218        // Every cell of the body is the destination of exactly one source cell, so clearing the
219        // body and inserting the permuted entries is the same permutation the cells went through.
220        for c in body.iter() {
221            shm.comments.remove(&c);
222            shm.hyperlinks.remove(&c);
223        }
224        for (at, cm) in side {
225            shm.comments.insert(at, cm);
226        }
227        for (at, h) in links {
228            shm.hyperlinks.insert(at, h);
229        }
230        shm.cell_pictures.extend(pictures);
231        for (i, h) in heights.into_iter().enumerate() {
232            let row = body.start.row + i as u32;
233            match h {
234                Some(info) => {
235                    shm.rows.insert(row, LineInfo { hidden: shm.rows.get(&row).is_some_and(|x| x.hidden), ..info });
236                }
237                None => {
238                    if let Some(e) = shm.rows.get_mut(&row) {
239                        e.size = None;
240                    }
241                }
242            }
243        }
244        // Re-anchor floating objects now that the row heights are final. `objs` was collected in
245        // `object_anchors` order, which `object_anchors_mut` repeats.
246        // Only objects anchored inside the sorted block move: within its lines on the sort axis
247        // and within its span on the other axis (a picture beside the block stays put).
248        let (b0, b1) = if by_cols { (body.start.col, body.end.col) } else { (body.start.row, body.end.row) };
249        let (o0, o1) = if by_cols { (body.start.row, body.end.row) } else { (body.start.col, body.end.col) };
250        let new_anchors: Vec<Anchor> = objs
251            .into_iter()
252            .zip(descs)
253            .map(|(a, desc)| {
254                let other = if by_cols { a.cell.row } else { a.cell.col };
255                if other < o0 || other > o1 { a } else { sorted_anchor(shm, a, desc, by_cols, b0, b1, &dst_of) }
256            })
257            .collect();
258        for (a, n) in shm.object_anchors_mut().zip(new_anchors) {
259            *a = n;
260        }
261        for (c, _) in moved {
262            cx.changed.push((sheet, c));
263        }
264        cx.structural = true;
265        Ok(Json::Null)
266    })?;
267    Ok(json!({"sorted": body.a1()}))
268}

270/// Re-anchor one floating object for a sort. `bottom` is the object's bottom-right line and the
271/// offset of that corner inside it, measured before the sort. `map` (via `dst_of`) translates a
272/// line index on the moving axis and is the identity outside `b0..=b1`. `sh` is the post-sort
273/// grid (row heights already reordered), so re-fitting uses the heights the object lands on.
274/// Objects pinned absolutely ("Don't move or size with cells") never move; "move but don't size"
275/// keeps its size; "move and size" is re-fitted when its whole extent stays inside the body, and
276/// keeps its size when the sort turns its lines around (its bottom line now sorts before its top
277/// line), since the lines then enclose nothing to stretch it between.
278fn sorted_anchor(sh: &Sheet, a: Anchor, bottom: (u32, u32, f32, f32), by_cols: bool, b0: u32, b1: u32, dst_of: &[u32]) -> Anchor {
279    if a.mode == AnchorMode::Absolute {
280        return a;
281    }
282    let (tc, tr, tdx, tdy) = bottom;
283    let map = |v: u32| -> u32 { if v >= b0 && v <= b1 { dst_of.get((v - b0) as usize).map_or(v, |&d| b0 + d) } else { v } };
284    let (axis_start, axis_end) = if by_cols { (a.cell.col, tc) } else { (a.cell.row, tr) };
285    if axis_start < b0 || axis_start > b1 {
286        return a;
287    }
288    let (nr, nc) = if by_cols { (a.cell.row, map(a.cell.col)) } else { (map(a.cell.row), a.cell.col) };
289    if a.mode == AnchorMode::MoveAndSize && axis_end >= b0 && axis_end <= b1 {
290        let (nbr, nbc) = if by_cols { (tr, map(tc)) } else { (map(tr), tc) };
291        let x1 = sh.col_left(nc) + a.dx as f64;
292        let y1 = sh.row_top(nr) + a.dy as f64;
293        let x2 = sh.col_left(nbc) + tdx as f64;
294        let y2 = sh.row_top(nbr) + tdy as f64;
295        // An extent that is not positive means the sort put the bottom-right line before the
296        // top-left one on that axis: the object keeps its old size there.
297        let fit = |extent: f64, old: f32| if extent > 0.0 { extent as f32 } else { old.max(0.0) };
298        Anchor { cell: CellRef::new(nr, nc), dx: a.dx, dy: a.dy, width: fit(x2 - x1, a.width), height: fit(y2 - y1, a.height), mode: a.mode }
299    } else {
300        Anchor { cell: CellRef::new(nr, nc), ..a }
301    }
302}

304/// Case-sensitive tie-break for text that is equal ignoring case: as in Excel, lowercase sorts
305/// before uppercase at the first letter that differs in case.
306fn case_order(a: &str, b: &str) -> Ordering {
307    a.chars().zip(b.chars()).find(|(x, y)| x != y).map_or(Ordering::Equal, |(x, y)| y.is_lowercase().cmp(&x.is_lowercase()))
308}

310pub(super) fn sort_key(v: &Value) -> (u8, i64, String) {
311    match v {
312        Value::Number(n) => (0, (n * 1e6) as i64, String::new()),
313        Value::Text(t) => (1, 0, t.to_lowercase()),
314        Value::Bool(b) => (2, *b as i64, String::new()),
315        Value::Empty => (9, 0, String::new()),
316        _ => (3, 0, v.display()),
317    }
