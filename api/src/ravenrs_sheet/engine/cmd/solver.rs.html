1//! Solver (Data › Analysis › Solver): finds values for variable cells that maximise, minimise
2//! or hit a value in an objective cell, subject to constraints.
3//!
4//! The workbook is the model: every evaluation sets the variable cells on a copy of the
5//! workbook and recalculates their dependents ([`Trial`]), so any formula works. Two methods:
6//!
7//! - **Simplex LP**: the model is linearised from `n + 1` evaluations and checked for linearity
8//!   at one more point; then a dense two-phase simplex (Dantzig's rule, Bland's rule when it
9//!   stalls) solves it exactly. Integer and binary constraints use depth-first branch and bound.
10//! - **GRG Nonlinear** (the name Excel's dialog uses): GridCraft's method is *not* a generalised
11//!   reduced gradient. It is an augmented-Lagrangian method (Powell–Hestenes–Rockafellar) whose
12//!   inner problems are solved by projected gradient descent with Barzilai–Borwein steps and an
13//!   Armijo line search, on central-difference gradients. Variable bounds are kept by
14//!   projection. It finds a local optimum of a smooth model; integer constraints use a small
15//!   branch and bound on top.
16//!
17//! - **Evolutionary**: a genetic search for models that aren't smooth (IF, lookups, rounding),
18//!   with tournament selection, blend crossover and Gaussian mutation inside the variable bounds,
19//!   feasibility-first ranking and integer rounding, followed by a compass (pattern) search around
20//!   the best member. It stops when the population has converged, when the best member stops
21//!   improving, or at the evaluation budget.
22//!
23//! Work is bounded by an evaluation budget (no clock: `Instant` traps on wasm). The model is
24//! saved as hidden sheet-scoped names (`solver_opt`, `solver_adj`, `solver_lhs1`…), so it
25//! travels with the workbook through XLSX.

125/// What a solve ended with, and why Simplex LP or Evolutionary stopped.
126struct Solve {
127    outcome: Outcome,
128    not_linear: bool,
129    evo_end: Option<EvoEnd>,
130}
131
132/// Variable bounds and integrality flags.
133#[derive(Clone, Copy)]
134struct Bounds<'a> {
135    lo: &'a [f64],
136    hi: &'a [f64],
137    integer: &'a [bool],
138}
139
140/// Runs the model's method for `goal` within `lo..hi` from `x0`.
141fn optimise(ev: &mut Evaluator, m: &Model, goal: Goal, x0: &[f64], bounds: Bounds, gap: f64) -> Solve {
142    let Bounds { lo, hi, integer } = bounds;
143    let any_int = integer.iter().any(|b| *b);
144    let mut not_linear = false;
145    let mut evo_end = None;
146    let outcome = if lo.iter().zip(hi).any(|(l, h)| l > h) {
147        Outcome::Infeasible
148    } else {
149        match m.method {
150            Method::Simplex => match simplex_program(ev, goal, x0) {
151                Err("nonlinear") | Err("error") => {
152                    not_linear = true;
153                    Outcome::Infeasible
154                }
155                Err(_) => Outcome::Limit,
156                Ok(SimplexProgram { c, rows, mut work, .. }) => {
157                    let mut relax = |l: &[f64], h: &[f64]| solve_lp(&c, &rows, l, h, &mut work);
158                    if any_int { branch_and_bound(lo, hi, integer, MAX_NODES_LP, gap, &mut relax).0 } else { relax(lo, hi) }
159                }
160            },
161            Method::Grg => {
162                if any_int {
163                    let mut relax = |l: &[f64], h: &[f64]| {
164                        let mut start = x0.to_vec();
165                        project(&mut start, l, h);
166                        nlp(ev, &start, l, h, goal)
167                    };
168                    branch_and_bound(lo, hi, integer, MAX_NODES_NLP, gap, &mut relax).0
169                } else {
170                    nlp(ev, x0, lo, hi, goal)
171                }
172            }
173            Method::Evolutionary => {
174                let (out, end) = evolutionary(ev, x0, lo, hi, integer, goal, &m.evo);
175                evo_end = Some(end);
176                out
177            }
178        }
179    };
180    Solve { outcome, not_linear, evo_end }
181}

185fn run(s: &mut Session, p: &Json) -> Result<Json> {
186    let d = s.doc()?;
187    let active = d.wb.active_sheet();
188    let mut params = p.clone();
189    if bool_param(p, "saved").unwrap_or(false) {
190        let saved = saved_model(&d.wb, active).ok_or_else(|| EngineError::Other("This sheet has no saved Solver model.".into()))?;
191        if let (Some(base), Some(over)) = (saved.as_object(), p.as_object()) {
192            let mut merged = base.clone();
193            for (k, v) in over {
194                merged.insert(k.clone(), v.clone());
195            }
196            params = Json::Object(merged);
197        }
198    }
199    let has_objective = ["objective", "setObjective", "set"].iter().any(|k| str_param(&params, k).is_some_and(|t| !t.trim().is_empty()));
200    if !has_objective {
201        let model = saved_model(&d.wb, active);
202        s.ui_requests.push(crate::engine::UiRequest::Dialog("solver".into(), json!({})));
203        return Ok(json!({"model": model}));
204    }
205    let model = parse_model(s, &params)?;
206    let (budget, gap) = (model.max_evals, model.gap);
207    let keep = bool_param(&params, "keep").unwrap_or(true);
208    let save = bool_param(&params, "saveModel").unwrap_or(true);
209    let reports = report_kinds(CMD, &params)?;
210    let prob = build_problem(&model);
211    // A Sensitivity report the model has none of (integer, Evolutionary, Value Of) fails the
212    // call, so it is refused before the solve runs rather than after it.
213    if let Some(kinds) = &reports
214        && kinds.contains(&reports::ReportKind::Sensitivity)
215        && let Some(why) = sensitivity::sensitivity_refusal(&model, &prob)
216    {
217        return Err(EngineError::Other(why.into()));
218    }
219    let d = s.doc()?;
220    let sh = d.wb.sheet(model.sheet).ok_or(EngineError::NoDocument)?;
221    let x0: Vec<f64> = model.vars.iter().map(|c| as_num(&sh.value(*c))).map(|v| if v.is_finite() { v } else { 0.0 }).collect();
222    let mut ev = Evaluator {
223        trial: Trial::new(d),
224        sheet: model.sheet,
225        vars: model.vars.clone(),
226        objective: model.objective,
227        rows: prob.rows.clone(),
228        evals: 0,
229        budget,
230        target: None,
231    };
232    if model.method == Method::Evolutionary && model.evo.require_bounds && prob.lo.iter().chain(&prob.hi).any(|b| !b.is_finite()) {
233        return Err(EngineError::Other(
234            "The Evolutionary method needs a lower and an upper bound on every variable cell (constraints such as \"A1:A3 <= 10\"), or requireBounds: false."
235                .into(),
236        ));
237    }
238    let Solve { outcome, not_linear, evo_end } =
239        optimise(&mut ev, &model, model.goal, &x0, Bounds { lo: &prob.lo, hi: &prob.hi, integer: &prob.integer }, gap);
240    let (status, message, solution) = match outcome {
241        _ if not_linear => ("notLinear", "The model isn't linear in the variable cells, so Simplex LP can't solve it. Try GRG Nonlinear.", None),
242        Outcome::Solved(x, _, true) if model.method == Method::Evolutionary => match evo_end {
243            Some(EvoEnd::Stalled) => {
244                ("cannotImprove", "Solver stopped when the best solution stopped improving. All constraints are satisfied.", Some(x))
245            }
246            _ => ("converged", "The population converged. The best solution found satisfies all constraints.", Some(x)),
247        },
248        Outcome::Solved(x, _, true) => (
249            if model.method == Method::Simplex { "optimal" } else { "converged" },
250            if model.method == Method::Simplex {
251                "Solver found an optimal solution. All constraints are satisfied."
252            } else {
253                "Solver converged to a locally optimal solution. All constraints are satisfied."
254            },
255            Some(x),
256        ),
257        Outcome::Solved(x, _, false) => {
258            ("iterationLimit", "Solver stopped at its evaluation limit; the best solution found satisfies the constraints.", Some(x))
259        }
260        Outcome::Limit => ("iterationLimit", "Solver stopped at its evaluation limit without finding a feasible solution.", None),
261        Outcome::Infeasible => ("infeasible", "Solver could not find a solution that satisfies all constraints.", None),
262        Outcome::Unbounded => ("unbounded", "The objective can grow without limit: the problem is unbounded.", None),
263    };
264    let evaluations = ev.evals;
265    // The objective's value at the solution.
266    let objective = solution.as_ref().and_then(|x| {
267        ev.budget = ev.budget.saturating_add(1);
268        ev.eval(x).map(|p| p.f)
269    });
270    let mut values = serde_json::Map::new();
271    if let Some(x) = &solution {
272        for (c, v) in model.vars.iter().zip(x) {
273            values.insert(c.a1(), json!(v));
274        }
275    }
276    let original: serde_json::Map<String, Json> = model.vars.iter().zip(&x0).map(|(c, v)| (c.a1(), json!(v))).collect();
277    let changed = keep && solution.is_some();
278    // A Sensitivity report the solution doesn't support is left out (with its reason), so the
279    // solve and the other reports still apply.
280    let (built, skipped) = match (&reports, &solution) {
281        (Some(kinds), Some(x)) => {
282            let inp = ReportInput { model: &model, prob: &prob, original: x0, solution: x.clone(), message: message.to_string(), budget };
283            prepare_reports(s.doc()?, kinds, &inp, NotOptimal::Skip { optimum: sensitivity_status(status) })?
284        }
285        _ => (Vec::new(), Vec::new()),
286    };
287    let count = built.len();
288    let before = s.doc()?.wb.clone();
289    let mut out = json!({
290        "status": status,
291        "message": message,
292        "objective": objective,
293        "values": values,
294        "original": original,
295        "model": model_params(&before, &model),
296        "evaluations": evaluations,
297        "changed": changed,
298    });
299    let names = if changed || save || count > 0 {
300        let sheet = model.sheet;
301        edit(s, |cx| {
302            if changed && let Some(x) = &solution {
303                let sh = cx.sheet_mut(sheet)?;
304                for (c, v) in model.vars.iter().zip(x) {
305                    let style = sh.style_id(*c);
306                    sh.set_cell(*c, Cell { value: Value::number(*v), formula: None, style, py: None });
307                }
308                for c in &model.vars {
309                    cx.touch(sheet + count, *c);
310                }
311            }
312            if save {
313                save_model(&mut cx.wb, &model);
314            }
315            if count > 0 {
316                // Inserting sheets makes this structural, so retain the variable edit's array check.
317                check_arrays(&before, &cx.wb)?;
318            }
319            Ok(write_reports(cx, sheet, built))
320        })?
321    } else {
322        Vec::new()
323    };
324    insert_report_selections(s, model.sheet, count);
325    if reports.is_some() && solution.is_some() {
326        out["reports"] = json!(names);
327    }
328    if !skipped.is_empty() {
329        out["skippedReports"] = json!(skipped);
330    }
331    Ok(out)
332}
