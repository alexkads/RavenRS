1//! Printing: Page Setup, pagination and PDF output (File › Print, File › Export › PDF).
2//!
3//! Sheets are paginated like Excel: the print area (or the used range plus drawing objects) is
4//! split into bands of rows and columns that fit the printable area at the page scale, honouring
5//! manual breaks, repeated title rows/columns, headings, Fit To and centring, and the pages follow
6//! the sheet's page order ("down, then over" by default, or "over, then down"). Each page is
7//! drawn into a PDF with `gridcraft-pdf`: fills, gridlines, borders, formatted text (fonts mapped
8//! to the standard 14 PDF fonts), merged cells, wrapping, overflow into empty neighbours,
9//! conditional formats, table styles, charts, images, shapes and sparklines. Headers and footers
10//! support `&L`/`&C`/`&R` sections and the `&P &N &D &T &A &F &Z` codes.

37/// What one PDF render shares across the pages an object spans.
38#[derive(Default)]
39struct PicCache {
40    /// Decoded pictures (bytes, width, height) keyed by (sheet, picture); `None` = undecodable.
41    decoded: std::collections::HashMap<(usize, usize), Option<DecodedPicture>>,
42    /// Laid-out charts (with their [`chart_cost`]) keyed by (sheet, chart), rendered once however
43    /// many pages they touch. Only charts that can still be drawn are kept.
44    charts: std::collections::HashMap<(usize, usize), (Vec<crate::chart::Prim>, usize)>,
45    /// Charts too big for what was left of the budget: never drawn, so not rendered again.
46    rejected_charts: std::collections::HashSet<(usize, usize)>,
47    /// Bytes of picture pixels embedded so far, and chart cost units emitted so far. Each page an
48    /// object touches gets its own copy in the PDF, so a workbook-controlled anchor over thousands
49    /// of pages would multiply them without bound.
50    image_bytes: usize,
51    chart_prims: usize,
52}
53
54impl PicCache {
55    /// The chart's primitives for one more page, or `None` once they no longer fit
56    /// [`MAX_CHART_PRIMS`]. `render` runs at most once per chart; a chart that doesn't fit is
57    /// dropped at once and remembered, so its vectors are never retained.
58    fn chart(&mut self, key: (usize, usize), render: impl FnOnce() -> Vec<crate::chart::Prim>) -> Option<&[crate::chart::Prim]> {
59        if self.rejected_charts.contains(&key) {
60            return None;
61        }
62        let (prims, cost) = match self.charts.remove(&key) {
63            Some(cached) => cached,
64            None => {
65                let prims = render();
66                let cost = chart_cost(&prims);
67                (prims, cost)
68            }
69        };
70        if self.chart_prims.saturating_add(cost) > MAX_CHART_PRIMS {
71            self.rejected_charts.insert(key);
72            return None;
73        }
74        self.chart_prims += cost;
75        Some(&self.charts.entry(key).or_insert((prims, cost)).0)
76    }
77}

79/// What a chart's drawing costs in the PDF: one unit per primitive plus one per path point (a
80/// smoothed line holds thousands of points, each written out) and per point of a wedge's outline.
81fn chart_cost(prims: &[crate::chart::Prim]) -> usize {
82    use crate::chart::Prim;
83    prims
84        .iter()
85        .map(|p| match p {
86            Prim::Line { pts, .. } | Prim::Polygon { pts, .. } => pts.len().saturating_add(1),
87            // At most 400 steps along each of the outer and inner arcs (see `draw_prims`).
88            Prim::Wedge { .. } => 803,
89            _ => 1,
90        })
91        .fold(0, usize::saturating_add)
92}

94/// Most picture pixel bytes (RGBA, before PDF packing) one render embeds across all pages.
95const MAX_EMBEDDED_IMAGE_BYTES: usize = 512 * 1024 * 1024;
96/// Most chart cost units (primitives and path points, see [`chart_cost`]) one render draws across all pages.
97const MAX_CHART_PRIMS: usize = 4_000_000;

154/// Renders laid-out sheets to a PDF.
155pub fn render_pdf(wb: &Workbook, layouts: &[SheetLayout], file: &str, path: &str) -> (Vec<u8>, usize) {
156    let mut doc = PdfDoc::new();
157    doc.info.title = if wb.props.title.is_empty() { file.to_string() } else { wb.props.title.clone() };
158    doc.info.author = wb.props.author.clone();
159    let total: usize = layouts.iter().map(|l| l.pages.len()).sum();
160    let (date, time) = hf_date_time(wb);
161    let mut page_no = 0usize;
162    let mut pending_links = Vec::new();
163    let mut printed_blocks = Vec::new();
164    // Decoded pictures are reused across the pages their anchor covers: a floating picture
165    // spanning 40 pages was decoded (and pushed into the PDF) 40 times (#2389).
166    let mut pics = PicCache::default();
167    for lay in layouts {
168        let Some(sh) = wb.sheet(lay.sheet) else { continue };
169        let mut cf = if sh.cond_formats.is_empty() { None } else { Some(crate::engine::cf::CfCache::new()) };
170        let k = lay.k();
171        let (aw, ah) = lay.avail();
172        for (pi, (ri, ci)) in lay.pages.iter().enumerate() {
173            page_no += 1;
174            let (Some(rband), Some(cband)) = (lay.row_bands.get(*ri).copied(), lay.col_bands.get(*ci).copied()) else { continue };
175            let rsegs = segments(lay.title_rows, rband);
176            let csegs = segments(lay.title_cols, cband);
177            let content_w = (lay.heading_w + csegs.iter().map(|s| seg_size(sh, *s, false)).sum::<f32>()) * k;
178            let content_h = (lay.heading_h + rsegs.iter().map(|s| seg_size(sh, *s, true)).sum::<f32>()) * k;
179            let ox = lay.margins[0] + if lay.settings.center_h { ((aw - content_w) / 2.0).max(0.0) } else { 0.0 };
180            let oy = lay.margins[2] + if lay.settings.center_v { ((ah - content_h) / 2.0).max(0.0) } else { 0.0 };
181            let page_index = doc.page_count();
182            let pg = doc.add_page(lay.page_w, lay.page_h);
183            let mut y = oy + lay.heading_h * k;
184            let mut body: Option<Block> = None;
185            for (rsi, rs) in rsegs.iter().enumerate() {
186                let mut x = ox + lay.heading_w * k;
187                for (csi, cs) in csegs.iter().enumerate() {
188                    let b = make_block(wb, lay.sheet, sh, *rs, *cs, (x, y), lay);
189                    draw_block(pg, &b, lay.settings.gridlines, &mut cf);
190                    draw_block_links(pg, &b, page_index, &mut pending_links);
191                    printed_blocks.push(PrintedBlock { page: page_index, sheet: lay.sheet, rows: *rs, cols: *cs, x, y, k });
192                    if lay.settings.headings {
193                        draw_headings(pg, &b, rsi == 0, csi == 0, (ox, oy), (lay.heading_w * k, lay.heading_h * k));
194                    }
195                    x += seg_size(sh, *cs, false) * k;
196                    if rsi + 1 == rsegs.len() && csi + 1 == csegs.len() {
197                        body = Some(b);
198                    }
199                }
200                y += seg_size(sh, *rs, true) * k;
201            }
202            if let Some(b) = &body {
203                draw_objects(pg, wb, lay.sheet, sh, b, &mut pics);
204            }
205            if lay.settings.headings {
206                let fill = [0xF2, 0xF2, 0xF2];
207                pg.fill_rect(ox, oy, lay.heading_w * k, lay.heading_h * k, fill);
208                pg.rect_stroke(ox, oy, lay.heading_w * k, lay.heading_h * k, 0.3, [0xA0, 0xA0, 0xA0]);
209            }
210            let cx = HfContext {
211                page: page_no,
212                pages: total,
213                sheet: sh.name.clone(),
214                file: file.to_string(),
215                path: path.to_string(),
216                date: date.clone(),
217                time: time.clone(),
218            };
219            draw_header_footer(pg, lay, lay.settings.hf_page(pi == 0, page_no), &cx);
220        }
221    }
222    for link in pending_links {
223        if let Some((target_page, x, y)) = printed_target(wb, &printed_blocks, link.sheet, &link.target)
224            && let Some(page) = doc.page_mut(link.page)
225        {
226            let (lx, ly, w, h) = link.rect;
227            page.link_page(Rect::new(lx, ly, w, h), target_page, Point::new(x, y));
228        }
229    }
230    let n = doc.page_count();
231    (doc.finish(), n)
232}

236fn build_pdf(s: &Session, p: &Json, cmd: &str) -> Result<(Vec<u8>, usize)> {
237    let opts = PrintOptions::from_params(p, cmd)?;
238    let d = s.doc()?;
239    let lays = layouts(&d.wb, &opts);
240    let path = d.path.clone().unwrap_or_default();
241    Ok(render_pdf(&d.wb, &lays, &d.display_title(), &path))
242}

244pub(crate) fn export_pdf(s: &mut Session, p: &Json) -> Result<Json> {
245    let (bytes, pages) = build_pdf(s, p, "file.exportPdf")?;
246    match str_param(p, "path").filter(|x| !x.is_empty()) {
247        Some(path) => {
248            crate::engine::io::write_file(path, &bytes)?;
249            Ok(json!({"pages": pages, "bytes": bytes.len(), "path": path}))
250        }
251        None => Ok(json!({"pages": pages, "bytes": bytes.len(), "base64": crate::engine::io::base64_encode(&bytes)})),
252    }
253}

255fn print(s: &mut Session, p: &Json) -> Result<Json> {
256    let (bytes, pages) = build_pdf(s, p, "file.print")?;
257    let mut out = json!({"pages": pages, "bytes": bytes.len(), "base64": crate::engine::io::base64_encode(&bytes)});
258    if let Some(path) = str_param(p, "path").filter(|x| !x.is_empty()) {
259        crate::engine::io::write_file(path, &bytes)?;
260        out["path"] = json!(path);
261    }
262    Ok(out)
263}

265fn print_preview(s: &mut Session, p: &Json) -> Result<Json> {
266    let opts = PrintOptions::from_params(p, "file.printPreview")?;
267    let d = s.doc()?;
268    let lays = layouts(&d.wb, &opts);
269    let mut list = Vec::new();
270    let mut n = 0usize;
271    for lay in &lays {
272        let name = d.wb.sheet(lay.sheet).map(|s| s.name.clone()).unwrap_or_default();
273        for (ri, ci) in &lay.pages {
274            n += 1;
275            let (Some(rb), Some(cb)) = (lay.row_bands.get(*ri), lay.col_bands.get(*ci)) else { continue };
276            let r = RangeRef::new(CellRef::new(rb.0, cb.0), CellRef::new(rb.1, cb.1));
277            list.push(json!({
278                "page": n,
279                "sheet": name,
280                "range": r.a1(),
281                "rows": format!("{}:{}", rb.0 + 1, rb.1 + 1),
282                "columns": format!("{}:{}", crate::base::col_to_letters(cb.0), crate::base::col_to_letters(cb.1)),
283                "scale": (lay.scale * 1000.0).round() / 10.0,
284                "paper": lay.settings.paper,
285                "orientation": if lay.settings.orientation == Orientation::Landscape { "landscape" } else { "portrait" },
286                "pageOrder": lay.settings.page_order.id(),
287                "width": lay.page_w,
288                "height": lay.page_h,
289            }));
290        }
291    }
292    if let Some(page) = u32_param(p, "page") {
293        let one =
294            list.get((page as usize).saturating_sub(1)).cloned().ok_or_else(|| bad("file.printPreview", format!("there are only {n} pages")))?;
295        return Ok(json!({"pages": n, "page": one}));
296    }
297    Ok(json!({"pages": n, "pageList": list}))
298}

300fn page_setup(s: &mut Session, p: &Json) -> Result<Json> {
301    const CMD: &str = "pageLayout.pageSetup";
302    let si = target_sheet(s, p)?;
303    let orientation = match str_param(p, "orientation") {
304        None => None,
305        Some("portrait") => Some(Orientation::Portrait),
306        Some("landscape") => Some(Orientation::Landscape),
307        Some(o) => return Err(bad(CMD, format!("orientation must be portrait or landscape, not `{o}`"))),
308    };
309    let page_order = match p.get("pageOrder") {
310        None => None,
311        Some(v) => match v.as_str().and_then(PageOrder::parse) {
312            Some(o) => Some(o),
313            None => {
314                let shown = v.as_str().map_or_else(|| v.to_string(), str::to_string);
315                return Err(bad(CMD, format!("pageOrder must be downThenOver or overThenDown, not `{shown}`")));
316            }
317        },
318    };
319    let paper = match str_param(p, "paper") {
320        None => None,
321        Some(name) => {
322            known_paper(name).ok_or_else(|| bad(CMD, format!("unknown paper size `{name}`")))?;
323            Some(name.trim().to_string())
324        }
325    };
326    let margin_src = p.get("margins").filter(|m| m.is_object()).unwrap_or(p);
327    let mut margins: [Option<f32>; 6] = [None; 6];
328    for (i, key) in ["left", "right", "top", "bottom", "header", "footer"].iter().enumerate() {
329        if let Some(v) = f64_param(margin_src, key).or_else(|| f64_param(p, key)) {
330            if !v.is_finite() || v < 0.0 {
331                return Err(bad(CMD, format!("margin `{key}` must be a non-negative number of inches")));
332            }
333            margins[i] = Some(v.min(20.0) as f32);
334        }
335    }
336    let parse_rows = |key: &str| -> Result<Option<Option<(u32, u32)>>> {
337        match p.get(key) {
338            None => Ok(None),
339            Some(Json::Null) => Ok(Some(None)),
340            Some(Json::String(t)) if t.trim().is_empty() => Ok(Some(None)),
341            Some(Json::String(t)) => {
342                let r = RangeRef::parse(split_sheet(t).1).ok_or_else(|| bad(CMD, format!("`{key}` is not a row range: `{t}`")))?;
343                Ok(Some(Some((r.start.row, r.end.row))))
344            }
345            Some(_) => Err(bad(CMD, format!("`{key}` must be text like \"1:2\""))),
346        }
347    };
348    let parse_cols = |key: &str| -> Result<Option<Option<(u32, u32)>>> {
349        match p.get(key) {
350            None => Ok(None),
351            Some(Json::Null) => Ok(Some(None)),
352            Some(Json::String(t)) if t.trim().is_empty() => Ok(Some(None)),
353            Some(Json::String(t)) => {
354                let r = RangeRef::parse(split_sheet(t).1).ok_or_else(|| bad(CMD, format!("`{key}` is not a column range: `{t}`")))?;
355                Ok(Some(Some((r.start.col, r.end.col))))
356            }
357            Some(_) => Err(bad(CMD, format!("`{key}` must be text like \"A:B\""))),
358        }
359    };
360    let title_rows = parse_rows("titleRows")?;
361    let title_cols = parse_cols("titleCols")?;
362    let print_area = match p.get("printArea") {
363        None => None,
364        Some(Json::Null) => Some(None),
365        Some(Json::String(t)) if t.trim().is_empty() => Some(None),
366        Some(Json::String(t)) => Some(Some(RangeRef::parse(split_sheet(t).1).ok_or_else(|| bad(CMD, format!("`printArea` is not a range: `{t}`")))?)),
367        Some(_) => return Err(bad(CMD, "`printArea` must be a range like \"A1:F50\"")),
368    };
369    let row_breaks: Option<Vec<u32>> = p.get("rowBreaks").and_then(Json::as_array).map(|a| {
370        let mut v: Vec<u32> = a
371            .iter()
372            .filter_map(|x| x.as_u64().or_else(|| x.as_str().and_then(|t| t.trim().parse().ok())))
373            .filter(|r| *r >= 2 && *r <= MAX_ROWS as u64)
374            .map(|r| r as u32 - 1)
375            .take(1026)
376            .collect();
377        v.sort_unstable();
378        v.dedup();
379        v
380    });
381    let col_breaks: Option<Vec<u32>> = p.get("colBreaks").and_then(Json::as_array).map(|a| {
382        let mut v: Vec<u32> = a
383            .iter()
384            .filter_map(|x| match x {
385                Json::String(t) => crate::base::letters_to_col(t.trim().trim_start_matches('$')),
386                Json::Number(n) => n.as_u64().filter(|c| *c >= 2 && *c <= MAX_COLS as u64).map(|c| c as u32 - 1),
387                _ => None,
388            })
389            .filter(|c| *c >= 1)
390            .take(1026)
391            .collect();
392        v.sort_unstable();
393        v.dedup();
394        v
395    });
396    let scale = f64_param(p, "scale").filter(|v| v.is_finite()).map(|v| v.clamp(10.0, 400.0).round() as u16);
397    let fit_w = p.get("fitWidth").map(|v| v.as_u64().unwrap_or(0).min(32767) as u16);
398    let fit_h = p.get("fitHeight").map(|v| v.as_u64().unwrap_or(0).min(32767) as u16);
399    let fit_page = bool_param(p, "fitToPage");
400    let header = str_param(p, "header").map(|t| t.chars().take(255).collect::<String>());
401    let footer = str_param(p, "footer").map(|t| t.chars().take(255).collect::<String>());
402    let flags = [("centerH", 0usize), ("centerV", 1), ("gridlines", 2), ("headings", 3)].map(|(k, i)| (i, bool_param(p, k)));
403    edit(s, |cx| {
404        let pr = &mut cx.sheet_mut(si)?.print;
405        if let Some(o) = orientation {
406            pr.orientation = o;
407        }
408        if let Some(pp) = &paper {
409            pr.paper = pp.clone();
410        }
411        for (i, m) in margins.iter().enumerate() {
412            if let (Some(v), Some(slot)) = (m, pr.margins.get_mut(i)) {
413                *slot = *v;
414            }
415        }
416        if let Some(sc) = scale {
417            pr.scale = sc;
418            pr.fit_to = None;
419        }
420        if fit_w.is_some() || fit_h.is_some() {
421            let (cw, ch) = pr.fit_to.unwrap_or((1, 1));
422            pr.fit_to = Some((fit_w.unwrap_or(cw), fit_h.unwrap_or(ch)));
423        }
424        match fit_page {
425            Some(true) => pr.fit_to = Some((1, 1)),
426            Some(false) => pr.fit_to = None,
427            None => {}
428        }
429        for (i, v) in flags {
430            if let Some(v) = v {
431                match i {
432                    0 => pr.center_h = v,
433                    1 => pr.center_v = v,
434                    2 => pr.gridlines = v,
435                    _ => pr.headings = v,
436                }
437            }
438        }
439        if let Some(h) = &header {
440            pr.header = h.clone();
441        }
442        if let Some(f) = &footer {
443            pr.footer = f.clone();
444        }
445        if let Some(a) = print_area {
446            pr.print_area = a;
447        }
448        if let Some(t) = title_rows {
449            pr.title_rows = t;
450        }
451        if let Some(t) = title_cols {
452            pr.title_cols = t;
453        }
454        if let Some(b) = &row_breaks {
455            pr.row_breaks = b.clone();
456        }
457        if let Some(b) = &col_breaks {
458            pr.col_breaks = b.clone();
459        }
460        if let Some(o) = page_order {
461            pr.page_order = o;
462        }
463        Ok(())
464    })?;
465    let pr = s.doc()?.wb.sheet(si).map(|sh| sh.print.clone()).unwrap_or_default();
466    Ok(serde_json::to_value(&pr).unwrap_or(Json::Null))
467}
