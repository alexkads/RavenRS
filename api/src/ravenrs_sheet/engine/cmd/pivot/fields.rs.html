10pub(super) fn in_axes(pt: &PivotTable, field: &str) -> bool {
11    pt.rows.iter().chain(pt.columns.iter()).any(|f| f.source_col.eq_ignore_ascii_case(field))
12}

14/// Name of the Σ Values pseudo-field (where the value fields sit when there are two or more).
15const VALUES_FIELD: &str = "Σ Values";

17/// Whether `field` names Σ Values: `Σ Values`, or `Values` unless a source field has that name.
18fn is_values_field(src: &pv::Source, field: &str) -> bool {
19    let f = field.trim();
20    f == VALUES_FIELD || (f.eq_ignore_ascii_case("values") && src.col(f).is_none())
21}

23/// Keeps Σ Values in place after a field left (`inserted` false) or arrived at index `i` of
24/// Rows (`rows`) or Columns.
25pub(super) fn shift_values(pt: &mut PivotTable, rows: bool, i: usize, inserted: bool) {
26    if pt.values_on_rows != rows {
27        return;
28    }
29    let Some(p) = pt.values_position else { return };
30    // Σ Values stays just ahead of the field it preceded.
31    let p = match inserted {
32        true if i <= p => p + 1,
33        false if i < p => p - 1,
34        _ => p,
35    };
36    let n = if rows { pt.rows.len() } else { pt.columns.len() };
37    pt.values_position = (p < n).then_some(p);
38}

40/// Takes `field` out of Rows and Columns (Σ Values keeps its place); returns its first entry.
41fn take_from_axes(pt: &mut PivotTable, field: &str) -> Option<PivotField> {
42    let mut first = None;
43    for rows in [true, false] {
44        loop {
45            let list = if rows { &mut pt.rows } else { &mut pt.columns };
46            let Some(i) = list.iter().position(|f| f.source_col.eq_ignore_ascii_case(field)) else { break };
47            let f = list.remove(i);
48            shift_values(pt, rows, i, false);
49            first.get_or_insert(f);
50        }
51    }
52    first
53}

55/// Inserts `f` into Rows (`rows`) or Columns at `position` (default: the end).
56fn insert_axis_field(pt: &mut PivotTable, rows: bool, f: PivotField, position: Option<usize>) {
57    let list = if rows { &mut pt.rows } else { &mut pt.columns };
58    let at = position.unwrap_or(list.len()).min(list.len());
59    list.insert(at, f);
60    shift_values(pt, rows, at, true);
61}

63/// Moves Σ Values between Rows and Columns, or within Rows, to `position` (default: the end).
64/// In Columns it always comes after the last column field: the column layout doesn't place it
65/// anywhere else yet.
66fn move_values_field(cmd: &str, pt: &mut PivotTable, from: &str, to: &str, position: Option<usize>) -> Result<Json> {
67    let current = if pt.values_on_rows { "rows" } else { "columns" };
68    if from != current {
69        return Err(bad(cmd, format!("`{VALUES_FIELD}` isn't in the {from} area")));
70    }
71    if !matches!(to, "rows" | "columns") {
72        return Err(other("The field you are moving cannot be placed in that PivotTable area."));
73    }
74    pt.values_on_rows = to == "rows";
75    pt.values_position = position.filter(|p| pt.values_on_rows && *p < pt.rows.len());
76    Ok(json!({"field": VALUES_FIELD, "to": to}))
77}

79/// The canonical source field name for `field`.
80pub(super) fn canonical(cmd: &str, src: &pv::Source, field: &str) -> Result<String> {
81    src.col(field).and_then(|c| src.headers.get(c).cloned()).ok_or_else(|| bad(cmd, format!("no field `{field}` in the source data")))
82}

84/// The calculated field named `field`, if any.
85fn calc_field<'a>(pt: &'a PivotTable, field: &str) -> Option<&'a PivotCalcField> {
86    pt.calculated_fields.iter().find(|c| c.name.eq_ignore_ascii_case(field.trim()))
87}

89/// The canonical name of a source or calculated field.
90fn canonical_any(cmd: &str, pt: &PivotTable, src: &pv::Source, field: &str) -> Result<String> {
91    match calc_field(pt, field) {
92        Some(c) if src.col(field).is_none() => Ok(c.name.clone()),
93        _ => canonical(cmd, src, field),
94    }

97fn unique_value_name(pt: &PivotTable, base: &str, except: Option<usize>) -> String {
98    let taken = |n: &str| pt.values.iter().enumerate().any(|(i, v)| Some(i) != except && v.name.eq_ignore_ascii_case(n));
99    if !taken(base) {
100        return base.to_string();
101    }
102    (2..10_000).map(|k| format!("{base}{k}")).find(|n| !taken(n)).unwrap_or_else(|| base.to_string())
103}

109/// Adds `field` to an area (moving it out of the other axis areas).
110pub(super) struct AreaPlacement<'a> {
111    pub cmd: &'a str,
112    pub field: &'a str,
113    pub area: &'a str,
114    pub func: Option<PivotFunc>,
115    pub name: Option<&'a str>,
116    pub position: Option<usize>,
117}

119pub(super) fn add_to_area(pt: &mut PivotTable, src: &pv::Source, placement: AreaPlacement<'_>) -> Result<()> {
120    let AreaPlacement { cmd, field, area, func, name, position } = placement;
121    let is_calc = src.col(field).is_none() && calc_field(pt, field).is_some();
122    if is_calc && area != "values" {
123        return Err(other("A calculated field can only be placed in the Values area."));
124    }
125    let field = canonical_any(cmd, pt, src, field)?;
126    match area {
127        "rows" | "columns" => {
128            if pt.rows.len() + pt.columns.len() >= pv::MAX_AXIS_FIELDS && !in_axes(pt, &field) {
129                return Err(other(format!("A PivotTable can have at most {} row and column fields.", pv::MAX_AXIS_FIELDS)));
130            }
131            let existing = take_from_axes(pt, &field).unwrap_or_else(|| PivotField { source_col: field.clone(), ..Default::default() });
132            // A report filter that selects nothing in particular goes away; a selection stays
133            // as the field's hidden items.
134            pt.filters.retain(|f| !(f.source_col.eq_ignore_ascii_case(&field) && f.selected.is_none()));
135            insert_axis_field(pt, area == "rows", existing, position);
136        }
137        "values" => {
138            if pt.values.len() >= pv::MAX_VALUE_FIELDS {
139                return Err(other(format!("A PivotTable can have at most {} value fields.", pv::MAX_VALUE_FIELDS)));
140            }
141            let numeric = is_calc || src.col(&field).and_then(|c| src.numeric.get(c).copied()).unwrap_or(false);
142            let f = func.unwrap_or(if numeric { PivotFunc::Sum } else { PivotFunc::Count });
143            let base = match name {
144                Some(n) if !n.trim().is_empty() => {
145                    if src.col(n).is_some() {
146                        return Err(other("PivotTable field name already exists."));
147                    }
148                    n.to_string()
149                }
150                _ => auto_name(f, &field),
151            };
152            let nm = unique_value_name(pt, &base, None);
153            let v = PivotValue { source_col: field, func: f, name: nm, show_as: PivotShowAs::Normal, number_format: None };
154            let at = position.unwrap_or(pt.values.len()).min(pt.values.len());
155            pt.values.insert(at, v);
156        }
157        _ => {
158            take_from_axes(pt, &field);
159            let existing = pt.filters.iter().position(|f| f.source_col.eq_ignore_ascii_case(&field));
160            let entry = match existing {
161                Some(i) => pt.filters.remove(i),
162                None => PivotFilter { source_col: field, selected: None },
163            };
164            let at = position.unwrap_or(pt.filters.len()).min(pt.filters.len());
165            pt.filters.insert(at, entry);
166        }
167    }
168    Ok(())
169}

171/// Removes `field` from an area; returns the source field it referred to.
172fn remove_from_area(cmd: &str, pt: &mut PivotTable, field: &str, area: &str) -> Result<String> {
173    match area {
174        "rows" | "columns" => {
175            let rows = area == "rows";
176            let list = if rows { &mut pt.rows } else { &mut pt.columns };
177            let i = list
178                .iter()
179                .position(|f| f.source_col.eq_ignore_ascii_case(field))
180                .ok_or_else(|| bad(cmd, format!("`{field}` isn't in the {area} area")))?;
181            let f = list.remove(i);
182            shift_values(pt, rows, i, false);
183            pt.filters.retain(|x| !x.source_col.eq_ignore_ascii_case(&f.source_col));
184            Ok(f.source_col)
185        }
186        "values" => {
187            let i = pt
188                .values
189                .iter()
190                .position(|v| v.name.eq_ignore_ascii_case(field))
191                .or_else(|| pt.values.iter().position(|v| v.source_col.eq_ignore_ascii_case(field)))
192                .ok_or_else(|| bad(cmd, format!("`{field}` isn't in the values area")))?;
193            Ok(pt.values.remove(i).source_col)
194        }
195        _ => {
196            let i = pt
197                .filters
198                .iter()
199                .position(|f| f.source_col.eq_ignore_ascii_case(field) && !in_axes(pt, &f.source_col))
200                .ok_or_else(|| bad(cmd, format!("`{field}` isn't in the filters area")))?;
201            Ok(pt.filters.remove(i).source_col)
202        }
203    }
204}

208/// Applies `f` to the targeted pivot and re-renders it, as one undo step.
209pub(super) fn update(s: &mut Session, p: &Json, f: impl FnOnce(&mut PivotTable, &pv::Source, &Workbook) -> Result<Json>) -> Result<Json> {
210    let (si, pi) = find_pivot(s, p)?;
211    let replace = bool_param(p, "replace").unwrap_or(false);
212    edit(s, |cx| {
213        let pt = cx.wb.sheet(si).and_then(|sh| sh.pivots.get(pi)).cloned().ok_or_else(|| other("There's no such PivotTable."))?;
214        let src = pv::read_source(&cx.wb, &pt.source, si).map_err(other)?;
215        let mut new = pt.clone();
216        let mut out = f(&mut new, &src, &cx.wb)?;
217        if let Some(slot) = cx.sheet_mut(si)?.pivots.get_mut(pi) {
218            *slot = new;
219        }
220        let area = pv::refresh(&mut cx.wb, si, pi, replace).map_err(other)?;
221        cx.structural = true;
222        if out.is_null() {
223            out = json!({});
224        }
225        if let Some(m) = out.as_object_mut() {
226            m.insert("range".into(), json!(area.a1()));
227        }
228        Ok(out)
229    })
230}

232pub(super) fn fields(s: &mut Session, p: &Json) -> Result<Json> {
233    let (si, pi) = find_pivot(s, p)?;
234    let d = s.doc()?;
235    let pt = d.wb.sheet(si).and_then(|sh| sh.pivots.get(pi)).ok_or_else(|| other("There's no such PivotTable."))?;
236    let mut out = pivot_json(&d.wb, si, pt);
237    let src = pv::read_source(&d.wb, &pt.source, si);
238    let list: Vec<Json> = match &src {
239        Ok(src) => src
240            .headers
241            .iter()
242            .enumerate()
243            .map(|(c, h)| {
244                let mut areas = vec![];
245                if pt.rows.iter().any(|f| f.source_col.eq_ignore_ascii_case(h)) {
246                    areas.push("rows");
247                }
248                if pt.columns.iter().any(|f| f.source_col.eq_ignore_ascii_case(h)) {
249                    areas.push("columns");
250                }
251                if pt.values.iter().any(|f| f.source_col.eq_ignore_ascii_case(h)) {
252                    areas.push("values");
253                }
254                if pt.filters.iter().any(|f| f.source_col.eq_ignore_ascii_case(h)) && !in_axes(pt, h) {
255                    areas.push("filters");
256                }
257                json!({
258                    "name": h,
259                    "numeric": src.numeric.get(c).copied().unwrap_or(false),
260                    "date": src.dates.get(c).copied().unwrap_or(false),
261                    "samples": pv::distinct_labels(&d.wb, src, c, PivotDateGroup::None, 5),
262                    "areas": areas,
263                })
264            })
265            .collect(),
266        Err(_) => vec![],
267    };
268    let mut list = list;
269    for c in &pt.calculated_fields {
270        let areas: Vec<&str> = if pt.values.iter().any(|v| v.source_col.eq_ignore_ascii_case(&c.name)) { vec!["values"] } else { vec![] };
271        list.push(json!({"name": c.name, "numeric": true, "date": false, "samples": [], "areas": areas, "calculated": true, "formula": c.formula}));
272    }
273    if let Some(m) = out.as_object_mut() {
274        m.insert("fields".into(), Json::Array(list));
275        if let Ok(src) = &src {
276            m.insert("records".into(), json!(src.rows.len()));
277        }
278        if let Err(e) = src {
279            m.insert("error".into(), json!(e));
280        }
281    }
282    Ok(out)
283}

296pub(super) fn add_field(s: &mut Session, p: &Json) -> Result<Json> {
297    const C: &str = "pivot.addField";
298    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
299    let area = area_param(C, str_param(p, "area"))?;
300    let func = match str_param(p, "func") {
301        Some(f) => Some(func_param(Some(f)).ok_or_else(|| bad(C, format!("unknown function `{f}`")))?),
302        None => None,
303    };
304    let name = str_param(p, "name").map(str::to_string);
305    let pos = position_param(p);
306    update(s, p, |pt, src, _| {
307        add_to_area(pt, src, AreaPlacement { cmd: C, field: &field, area, func, name: name.as_deref(), position: pos })?;
308        Ok(json!({"field": canonical_any(C, pt, src, &field)?, "area": area}))
309    })
310}

312pub(super) fn calculated_field(s: &mut Session, p: &Json) -> Result<Json> {
313    const C: &str = "pivot.calculatedField";
314    let Some(name) = str_param(p, "name").map(|n| n.trim().to_string()) else {
315        // Menu-style call: the Insert Calculated Field dialog.
316        let (si, pi) = find_pivot(s, p)?;
317        let d = s.doc()?;
318        let pt = d.wb.sheet(si).and_then(|sh| sh.pivots.get(pi)).ok_or_else(|| other("There's no such PivotTable."))?;
319        let n = (1..).map(|k| format!("Field{k}")).find(|n| calc_field(pt, n).is_none()).unwrap_or_else(|| "Field1".into());
320        let pivot = pt.name.clone();
321        s.ui_requests.push(crate::engine::UiRequest::Dialog("pivotCalculatedField".into(), json!({"pivot": pivot, "name": n, "formula": "= 0"})));
322        return ok();
323    };
324    if name.is_empty() || name.chars().count() > 255 {
325        return Err(bad(C, "`name` must be 1 to 255 characters"));
326    }
327    let delete = bool_param(p, "delete").unwrap_or(false);
328    let formula = str_param(p, "formula").map(|f| {
329        let f = f.trim();
330        if f.starts_with('=') { f.to_string() } else { format!("={f}") }
331    });
332    let to_values = bool_param(p, "values").unwrap_or(true);
333    if !delete && formula.is_none() {
334        return Err(bad(C, "missing `formula`"));
335    }
336    update(s, p, |pt, src, _| {
337        let existing = pt.calculated_fields.iter().position(|c| c.name.eq_ignore_ascii_case(&name));
338        if delete {
339            let i = existing.ok_or_else(|| bad(C, format!("no calculated field `{name}`")))?;
340            let gone = pt.calculated_fields.remove(i);
341            pt.values.retain(|v| !v.source_col.eq_ignore_ascii_case(&gone.name));
342            return Ok(json!({"name": gone.name, "deleted": true}));
343        }
344        let formula = formula.clone().unwrap_or_default();
345        if src.col(&name).is_some() {
346            return Err(other("PivotTable field name already exists."));
347        }
348        let canon = match existing {
349            Some(i) => {
350                let c = pt.calculated_fields.get_mut(i).ok_or_else(|| other("There's no such calculated field."))?;
351                c.formula = formula.clone();
352                c.name.clone()
353            }
354            None => {
355                if pt.calculated_fields.len() >= crate::engine::pivot_calc::MAX_CALC_FIELDS {
356                    return Err(other(format!("A PivotTable can have at most {} calculated fields.", crate::engine::pivot_calc::MAX_CALC_FIELDS)));
357                }
358                pt.calculated_fields.push(PivotCalcField { name: name.clone(), formula: formula.clone() });
359                name.clone()
360            }
361        };
362        let set = crate::engine::pivot_calc::CalcSet::new(pt, &src.headers);
363        let i = set.index(&canon).ok_or_else(|| other("There's no such calculated field."))?;
364        if let Some(e) = set.error(i) {
365            return Err(bad(C, e.to_string()));
366        }
367        if set.is_cyclic(i) {
368            return Err(bad(C, format!("`{canon}` refers to itself through its formula")));
369        }
370        if existing.is_none() && to_values && !pt.values.iter().any(|v| v.source_col.eq_ignore_ascii_case(&canon)) {
371            add_to_area(pt, src, AreaPlacement { cmd: C, field: &canon, area: "values", func: Some(PivotFunc::Sum), name: None, position: None })?;
372        }
373        Ok(json!({
374            "name": canon,
375            "formula": formula,
376            "calculatedFields": pt.calculated_fields.iter().map(|c| json!({"name": c.name, "formula": c.formula})).collect::<Vec<_>>(),
377        }))
378    })
379}

381pub(super) fn remove_field(s: &mut Session, p: &Json) -> Result<Json> {
382    const C: &str = "pivot.removeField";
383    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
384    let area = match str_param(p, "area") {
385        Some(a) => Some(area_param(C, Some(a))?),
386        None => None,
387    };
388    update(s, p, |pt, _, _| {
389        match area {
390            Some(a) => {
391                remove_from_area(C, pt, &field, a)?;
392            }
393            None => {
394                let mut any = false;
395                for a in ["rows", "columns", "values", "filters"] {
396                    any |= remove_from_area(C, pt, &field, a).is_ok();
397                }
398                if !any {
399                    return Err(bad(C, format!("`{field}` isn't in the PivotTable")));
400                }
401            }
402        }
403        Ok(json!({}))
404    })
405}

407pub(super) fn move_field(s: &mut Session, p: &Json) -> Result<Json> {
408    const C: &str = "pivot.moveField";
409    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
410    let from = area_param(C, str_param(p, "from"))?;
411    let to = area_param(C, str_param(p, "to"))?;
412    let pos = position_param(p);
413    update(s, p, |pt, src, _| {
414        if is_values_field(src, &field) {
415            return move_values_field(C, pt, from, to, pos);
416        }
417        if from == to {
418            // Reorder within the area.
419            match from {
420                "rows" | "columns" => {
421                    let rows = from == "rows";
422                    let list = if rows { &mut pt.rows } else { &mut pt.columns };
423                    let i = list
424                        .iter()
425                        .position(|f| f.source_col.eq_ignore_ascii_case(&field))
426                        .ok_or_else(|| bad(C, format!("`{field}` isn't in the {from} area")))?;
427                    let f = list.remove(i);
428                    shift_values(pt, rows, i, false);
429                    insert_axis_field(pt, rows, f, pos);
430                }
431                "values" => {
432                    let i = pt
433                        .values
434                        .iter()
435                        .position(|v| v.name.eq_ignore_ascii_case(&field))
436                        .or_else(|| pt.values.iter().position(|v| v.source_col.eq_ignore_ascii_case(&field)))
437                        .ok_or_else(|| bad(C, format!("`{field}` isn't in the values area")))?;
438                    let v = pt.values.remove(i);
439                    let at = pos.unwrap_or(pt.values.len()).min(pt.values.len());
440                    pt.values.insert(at, v);
441                }
442                _ => {
443                    let i = pt
444                        .filters
445                        .iter()
446                        .position(|f| f.source_col.eq_ignore_ascii_case(&field))
447                        .ok_or_else(|| bad(C, format!("`{field}` isn't in the filters area")))?;
448                    let f = pt.filters.remove(i);
449                    let at = pos.unwrap_or(pt.filters.len()).min(pt.filters.len());
450                    pt.filters.insert(at, f);
451                }
452            }
453            return Ok(json!({}));
454        }
455        // Keep axis settings (sort, grouping) when moving between rows and columns.
456        let kept = pt.rows.iter().chain(pt.columns.iter()).find(|f| f.source_col.eq_ignore_ascii_case(&field)).cloned();
457        let hidden = pt.filters.iter().find(|f| f.source_col.eq_ignore_ascii_case(&field)).cloned();
458        let src_field = remove_from_area(C, pt, &field, from)?;
459        add_to_area(pt, src, AreaPlacement { cmd: C, field: &src_field, area: to, func: None, name: None, position: pos })?;
460        if matches!(to, "rows" | "columns") {
461            let list = if to == "rows" { &mut pt.rows } else { &mut pt.columns };
462            if let (Some(k), Some(slot)) = (kept, list.iter_mut().find(|f| f.source_col.eq_ignore_ascii_case(&src_field))) {
463                *slot = k;
464            }
465            if let Some(h) = hidden.filter(|h| h.selected.is_some())
466                && !pt.filters.iter().any(|f| f.source_col.eq_ignore_ascii_case(&src_field))
467            {
468                pt.filters.push(h);
469            }
470        }
471        Ok(json!({"field": src_field, "to": to}))
472    })
473}

475pub(super) fn value_settings(s: &mut Session, p: &Json) -> Result<Json> {
476    const C: &str = "pivot.valueSettings";
477    let field = str_param(p, "field").map(str::to_string);
478    let index = u32_param(p, "index").map(|v| v as usize);
479    let func = match str_param(p, "func") {
480        Some(f) => Some(func_param(Some(f)).ok_or_else(|| bad(C, format!("unknown function `{f}`")))?),
481        None => None,
482    };
483    let show = match str_param(p, "showAs") {
484        Some(v) => Some(show_as_param(v).ok_or_else(|| bad(C, format!("unknown showAs `{v}`")))?),
485        None => None,
486    };
487    let name = str_param(p, "name").map(str::to_string);
488    let fmt = p.get("numberFormat").cloned();
489    update(s, p, |pt, src, _| {
490        let i = match (index, &field) {
491            (Some(i), _) => i,
492            (None, Some(f)) => pt
493                .values
494                .iter()
495                .position(|v| v.name.eq_ignore_ascii_case(f))
496                .or_else(|| pt.values.iter().position(|v| v.source_col.eq_ignore_ascii_case(f)))
497                .ok_or_else(|| bad(C, format!("`{f}` isn't in the values area")))?,
498            (None, None) => {
499                if pt.values.len() == 1 {
500                    0
501                } else {
502                    return Err(bad(C, "pass `field` or `index`"));
503                }
504            }
505        };
506        let cur = pt.values.get(i).cloned().ok_or_else(|| bad(C, "no value field at that index"))?;
507        let mut v = cur.clone();
508        if let Some(f) = func {
509            v.func = f;
510            // An automatic caption follows the function. Count Numbers was captioned "Count
511            // Numbers of …" before it took Excel's "Count of …", and saved workbooks still carry it.
512            let is_auto = |base: &str| {
513                cur.name.eq_ignore_ascii_case(base) || cur.name.strip_prefix(base).is_some_and(|r| r.chars().all(|c| c.is_ascii_digit()))
514            };
515            let legacy = cur.func == PivotFunc::CountNumbers && is_auto(&format!("Count Numbers of {}", cur.source_col));
516            if is_auto(&auto_name(cur.func, &cur.source_col)) || legacy {
517                v.name = unique_value_name(pt, &auto_name(f, &cur.source_col), Some(i));
518            }
519        }
520        if let Some(n) = &name {
521            let n = n.trim();
522            if n.is_empty() {
523                return Err(other("The name isn't valid."));
524            }
525            if src.col(n).is_some() {
526                return Err(other("PivotTable field name already exists."));
527            }
528            if pt.values.iter().enumerate().any(|(j, x)| j != i && x.name.eq_ignore_ascii_case(n)) {
529                return Err(other("PivotTable field name already exists."));
530            }
531            v.name = n.to_string();
532        }
533        if let Some(sa) = show {
534            v.show_as = sa;
535        }
536        match &fmt {
537            Some(Json::String(f)) if !f.is_empty() && f.len() <= 255 => v.number_format = Some(f.clone()),
538            Some(Json::Null) => v.number_format = None,
539            Some(Json::String(f)) if f.is_empty() => v.number_format = None,
540            Some(_) => return Err(bad(C, "`numberFormat` is a format code or null")),
541            None => {}
542        }
543        let name = v.name.clone();
544        if let Some(slot) = pt.values.get_mut(i) {
545            *slot = v;
546        }
547        Ok(json!({"name": name}))
548    })
549}
