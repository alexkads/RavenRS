3pub(super) fn area_param(cmd: &str, v: Option<&str>) -> Result<&'static str> {
4    match v.map(str::to_ascii_lowercase) {
5        Some(a) if a == "rows" || a == "row" => Ok("rows"),
6        Some(a) if a == "columns" || a == "column" || a == "cols" => Ok("columns"),
7        Some(a) if a == "values" || a == "value" || a == "data" => Ok("values"),
8        Some(a) if a == "filters" || a == "filter" => Ok("filters"),
9        Some(a) => Err(bad(cmd, format!("unknown area `{a}` (rows|columns|values|filters)"))),
10        None => Err(bad(cmd, "missing `area` (rows|columns|values|filters)")),
11    }
12}

14pub(super) fn func_param(v: Option<&str>) -> Option<PivotFunc> {
15    Some(match v?.to_ascii_lowercase().as_str() {
16        "sum" => PivotFunc::Sum,
17        "count" | "counta" => PivotFunc::Count,
18        "average" | "avg" | "mean" => PivotFunc::Average,
19        "max" => PivotFunc::Max,
20        "min" => PivotFunc::Min,
21        "product" => PivotFunc::Product,
22        "countnumbers" | "countnums" | "count numbers" => PivotFunc::CountNumbers,
23        "stddev" => PivotFunc::StdDev,
24        "stddevp" => PivotFunc::StdDevP,
25        "var" => PivotFunc::Var,
26        "varp" => PivotFunc::VarP,
27        _ => return None,
28    })
29}
