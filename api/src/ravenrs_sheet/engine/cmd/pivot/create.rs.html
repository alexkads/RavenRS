10/// The default source: the table at the selection, else the selected block / current region.
11fn default_source(s: &Session) -> Result<String> {
12    let d = s.doc()?;
13    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
14    if let Some(t) = sh.table_at(d.selection.active) {
15        return Ok(t.name.clone());
16    }
17    let sel = d.selection.current();
18    let r = if sel.is_single() {
19        current_region(sh, d.selection.active)
20    } else {
21        let used = sh.used_range().unwrap_or(sel);
22        sel.intersection(&used).unwrap_or(sel)
23    };
24    if r.is_single() && sh.value(r.start).is_empty() {
25        return Err(other("The reference isn't valid. Select a cell in a range with labeled columns, or pass `source`."));
26    }
27    Ok(pv::range_text(&sh.name, r))
28}

30pub(super) fn source_param(s: &Session, p: &Json) -> Result<String> {
31    match str_param(p, "source").or_else(|| str_param(p, "range")) {
32        Some(t) if !t.trim().is_empty() => {
33            // Bare ranges are on the active sheet; store them with the sheet name.
34            let d = s.doc()?;
35            let t = t.trim().trim_start_matches('=');
36            if !t.contains('!')
37                && RangeRef::parse(t).is_some()
38                && let Some(sh) = d.wb.active()
39                && let Some(r) = RangeRef::parse(t)
40            {
41                return Ok(pv::range_text(&sh.name, r));
42            }
43            Ok(t.to_string())
44        }
45        _ => default_source(s),
46    }
47}

49fn next_pivot_name(wb: &Workbook) -> String {
50    (1..)
51        .map(|n| format!("PivotTable{n}"))
52        .find(|n| !wb.sheets.iter().any(|s| s.pivots.iter().any(|p| p.name.eq_ignore_ascii_case(n))))
53        .unwrap_or_default()
54}

56fn next_pivot_id(wb: &Workbook) -> u32 {
57    let m = wb.sheets.iter().flat_map(|s| s.pivots.iter().map(|p| p.id)).max().unwrap_or(0);
58    wb.next_object_id().max(m.saturating_add(1))
59}

61fn field_list(v: Option<&Json>) -> Vec<Json> {
62    match v {
63        Some(Json::Array(a)) => a.iter().take(512).cloned().collect(),
64        Some(Json::String(s)) => vec![Json::String(s.clone())],
65        _ => vec![],
66    }
67}

69pub(super) fn insert_pivot(s: &mut Session, p: &Json) -> Result<Json> {
70    let source = source_param(s, p)?;
71    let active = s.doc()?.wb.active_sheet();
72    // Validate the source now (header names, size).
73    let src = pv::read_source(&s.doc()?.wb, &source, active).map_err(other)?;
74    let name = match str_param(p, "name") {
75        Some(n) if !n.trim().is_empty() => {
76            let n = n.trim().to_string();
77            if s.doc()?.wb.sheets.iter().any(|sh| sh.pivots.iter().any(|x| x.name.eq_ignore_ascii_case(&n))) {
78                return Err(other("A PivotTable with that name already exists."));
79            }
80            n
81        }
82        _ => next_pivot_name(&s.doc()?.wb),
83    };
84    let dest = str_param(p, "destination").or_else(|| str_param(p, "location")).unwrap_or("new").trim().to_string();
85    // Existing sheet destination: (sheet index, anchor).
86    let target: Option<(usize, CellRef)> = if dest.eq_ignore_ascii_case("new") || dest.is_empty() {
87        None
88    } else {
89        let d = s.doc()?;
90        let (sh, body) = crate::engine::cmd::split_sheet(&dest);
91        let si = match sh {
92            Some(n) => d.wb.sheet_index(&n).ok_or_else(|| bad("insert.pivotTable", format!("no sheet named `{n}`")))?,
93            None => active,
94        };
95        let at = RangeRef::parse(body).map(|r| r.start).ok_or_else(|| bad("insert.pivotTable", format!("not a cell: `{dest}`")))?;
96        Some((si, at))
97    };
98    // Decided on the resolved destination: `New`, ` new ` and an empty one all make a sheet.
99    if target.is_none() && s.doc()?.wb.protected_structure {
100        return Err(other("The workbook is protected and cannot be changed."));
101    }
102    let replace = bool_param(p, "replace").unwrap_or(false);
103    let rows = field_list(p.get("rows"));
104    let columns = field_list(p.get("columns"));
105    let values = field_list(p.get("values"));
106    let filters = field_list(p.get("filters"));
107    let mut new_sheet_at: Option<usize> = None;
108    let res = edit(s, |cx| {
109        let id = next_pivot_id(&cx.wb);
110        let (si, anchor) = match target {
111            Some(t) => t,
112            None => {
113                let at = insert_sheet_before_active(cx);
114                new_sheet_at = Some(at);
115                (at, CellRef::new(2, 0))
116            }
117        };
118        let mut pt = PivotTable { id, name: name.clone(), source: source.clone(), anchor, ..Default::default() };
119        let add = |pt: &mut PivotTable, list: &[Json], area: &str| -> Result<()> {
120            for f in list {
121                match f {
122                    Json::String(n) => {
123                        add_to_area(pt, &src, AreaPlacement { cmd: "insert.pivotTable", field: n, area, func: None, name: None, position: None })?
124                    }
125                    Json::Object(_) => {
126                        let n = str_param(f, "field").ok_or_else(|| bad("insert.pivotTable", "a field object needs `field`"))?;
127                        let func = match str_param(f, "func") {
128                            Some(fname) => {
129                                Some(func_param(Some(fname)).ok_or_else(|| bad("insert.pivotTable", format!("unknown function `{fname}`")))?)
130                            }
131                            None => None,
132                        };
133                        add_to_area(
134                            pt,
135                            &src,
136                            AreaPlacement { cmd: "insert.pivotTable", field: n, area, func, name: str_param(f, "name"), position: None },
137                        )?;
138                        let by = group_param(str_param(f, "group"));
139                        if let Some(by) = by.filter(|b| *b != PivotDateGroup::None)
140                            && matches!(area, "rows" | "columns")
141                            && src.col(n).and_then(|c| src.numeric.get(c).copied()).unwrap_or(false)
142                            && let Some(af) = pt.rows.iter_mut().chain(pt.columns.iter_mut()).find(|x| x.source_col.eq_ignore_ascii_case(n))
143                        {
144                            af.date_group = by;
145                        }
146                    }
147                    _ => return Err(bad("insert.pivotTable", "fields are names or {field, func?, name?}")),
148                }
149            }
150            Ok(())
151        };
152        add(&mut pt, &rows, "rows")?;
153        add(&mut pt, &columns, "columns")?;
154        add(&mut pt, &values, "values")?;
155        add(&mut pt, &filters, "filters")?;
156        let sh = cx.sheet_mut(si)?;
157        sh.pivots.push(pt);
158        let pi = sh.pivots.len() - 1;
159        let area = pv::refresh(&mut cx.wb, si, pi, replace).map_err(other)?;
160        cx.wb.set_active_sheet(si);
161        *cx.sel = crate::engine::Selection::at(anchor);
162        cx.structural = true;
163        let sheet_name = cx.wb.sheet(si).map(|s| s.name.clone()).unwrap_or_default();
164        Ok(json!({
165            "id": id,
166            "name": name,
167            "sheet": sheet_name,
168            "anchor": anchor.a1(),
169            "range": area.a1(),
170            "source": source,
171            "fields": src.headers,
172        }))
173    })?;
174    if let Some(at) = new_sheet_at
175        && let Ok(d) = s.doc_mut()
176    {
177        d.sheet_selections.insert(at.min(d.sheet_selections.len()), crate::engine::Selection::default());
178    }
179    Ok(res)
180}

182/// Inserts a new, empty sheet before the active one (Excel's place for a new PivotTable or a
183/// Show Details sheet); returns its index.
184fn insert_sheet_before_active(cx: &mut Ctx) -> usize {
185    let at = cx.wb.active_sheet().min(cx.wb.sheets.len());
186    let nm = cx.wb.next_sheet_name();
187    cx.wb.sheets.insert(at, Arc::new(Sheet::new(nm)));
188    for n in cx.wb.names.iter_mut() {
189        if let Some(sc) = n.scope.as_mut()
190            && *sc >= at
191        {
192            *sc += 1;
193        }
194    }
195    at
196}

198/// Show Details (double-click a value cell): the source records behind it, as values with their
199/// column's number format, under the source headers on a new sheet.
200pub(super) fn show_details(s: &mut Session, p: &Json) -> Result<Json> {
201    let d = s.doc()?;
202    let at = cell_param(s, p, "cell")?.unwrap_or(d.selection.active);
203    let active = d.wb.active_sheet();
204    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
205    let pt = sh
206        .pivots
207        .iter()
208        .find(|pt| pt.last_range.is_some_and(|r| r.contains(at)))
209        .ok_or_else(|| EngineError::Disabled("pivot.showDetails".into(), pv::NOT_A_VALUE_CELL.into()))?;
210    let origin = pt.last_range.map(|r| r.start).unwrap_or_default();
211    let found = pv::detail_records(&d.wb, pt, active, at.row.saturating_sub(origin.row), at.col.saturating_sub(origin.col));
212    // With the option off, a report that is out of date or a source that cannot be read is
213    // reported as the option; only a cell that is not a value cell keeps its own message.
214    if !pt.enable_show_details && !matches!(found, Ok(None)) {
215        return Err(other(SHOW_DETAILS_OFF));
216    }
217    let (src, records) = found.map_err(other)?.ok_or_else(|| EngineError::Disabled("pivot.showDetails".into(), pv::NOT_A_VALUE_CELL.into()))?;
218    if d.wb.protected_structure {
219        return Err(other("The workbook is protected and cannot be changed."));
220    }
221    let count = records.len();
222    let mut new_sheet_at: Option<usize> = None;
223    let res = edit(s, |cx| {
224        let si = insert_sheet_before_active(cx);
225        new_sheet_at = Some(si);
226        let base = cx.wb.styles.get(StyleId::default()).clone();
227        let fmts: Vec<StyleId> = src
228            .formats
229            .iter()
230            .map(|f| {
231                let mut st = base.clone();
232                st.num_fmt = NumFmt::new(f);
233                cx.wb.styles.intern(st)
234            })
235            .collect();
236        // Columns wide enough for their text, as the PivotTable report's are (dates would show
237        // as ####); the first rows are enough to judge a column.
238        let font = base.font.clone();
239        let widths: Vec<f32> = src
240            .headers
241            .iter()
242            .enumerate()
243            .map(|(c, h)| {
244                let fmt = src.formats.get(c).map(String::as_str).unwrap_or("General");
245                let texts = records.iter().take(1_000).filter_map(|i| src.rows.get(*i).and_then(|r| r.get(c)));
246                let widest = texts
247                    .map(|v| crate::engine::cmd::format::approx_text_width(&crate::engine::display::format(v, fmt, &cx.wb).text, font.size, false))
248                    .fold(crate::engine::cmd::format::approx_text_width(h, font.size, false), f32::max);
249                (widest + 14.0).min(320.0)
250            })
251            .collect();
252        let sh = cx.sheet_mut(si)?;
253        for (c, w) in widths.into_iter().enumerate() {
254            if w > sh.col_width(c as u32) {
255                sh.cols.entry(c as u32).or_default().size = Some(w);
256            }
257        }
258        for (c, h) in src.headers.iter().enumerate() {
259            sh.set_cell(
260                CellRef::new(0, c as u32),
261                Cell { value: crate::base::Value::text(h.as_str()), formula: None, style: StyleId::default(), py: None },
262            );
263        }
264        for (r, i) in records.iter().enumerate() {
265            let Some(rec) = src.rows.get(*i) else { continue };
266            for (c, v) in rec.iter().enumerate() {
267                let style = fmts.get(c).copied().unwrap_or_default();
268                if v.is_empty() && style == StyleId::default() {
269                    continue;
270                }
271                sh.set_cell(CellRef::new(r as u32 + 1, c as u32), Cell { value: v.clone(), formula: None, style, py: None });
272            }
273        }
274        let name = sh.name.clone();
275        cx.wb.set_active_sheet(si);
276        *cx.sel = crate::engine::Selection::at(CellRef::new(0, 0));
277        cx.structural = true;
278        Ok(json!({"sheet": name, "records": count}))
279    })?;
280    if let Some(at) = new_sheet_at
281        && let Ok(d) = s.doc_mut()
282    {
283        d.sheet_selections.insert(at.min(d.sheet_selections.len()), crate::engine::Selection::default());
284    }
285    Ok(res)
286}

291/// PivotTable Options: Enable show details (Data). Changes only the option; the report is not
292/// refreshed. Without params it returns the current setting.
293pub(super) fn options(s: &mut Session, p: &Json) -> Result<Json> {
294    let (si, pi) = find_pivot(s, p)?;
295    let enable = bool_param(p, "enableShowDetails");
296    if p.get("enableShowDetails").is_some_and(|v| !v.is_null()) && enable.is_none() {
297        return Err(bad("pivot.options", "`enableShowDetails` must be true or false"));
298    }
299    let current = |s: &Session| -> Result<Json> {
300        let pt = s.doc()?.wb.sheet(si).and_then(|sh| sh.pivots.get(pi)).ok_or_else(|| other("There's no such PivotTable."))?;
301        Ok(json!({"enableShowDetails": pt.enable_show_details}))
302    };
303    let Some(on) = enable else { return current(s) };
304    if current(s)?["enableShowDetails"] == json!(on) {
305        return current(s);
306    }
307    edit(s, |cx| {
308        let pt = cx.sheet_mut(si)?.pivots.get_mut(pi).ok_or_else(|| other("There's no such PivotTable."))?;
309        pt.enable_show_details = on;
310        Ok(json!({"enableShowDetails": on}))
311    })
312}

314pub(super) fn summarize_table(s: &mut Session, p: &Json) -> Result<Json> {
315    let d = s.doc()?;
316    let name = match str_param(p, "table") {
317        Some(n) => d.wb.table(n).and_then(|(si, ti)| d.wb.sheet(si).and_then(|sh| sh.tables.get(ti))).map(|t| t.name.clone()),
318        None => d.wb.active().and_then(|sh| sh.table_at(d.selection.active)).map(|t| t.name.clone()),
319    }
320    .ok_or_else(|| EngineError::Disabled("table.summarizePivot".into(), "select a cell in a table".into()))?;
321    let mut q = with(p, "source", json!(name));
322    if let Some(m) = q.as_object_mut() {
323        m.remove("table");
324    }
325    insert_pivot(s, &q)
326}

330pub(super) fn pivot_json(wb: &Workbook, si: usize, pt: &PivotTable) -> Json {
331    json!({
332        "id": pt.id,
333        "name": pt.name,
334        "sheet": wb.sheet(si).map(|s| s.name.clone()).unwrap_or_default(),
335        "anchor": pt.anchor.a1(),
336        "range": pt.last_range.map(|r| r.a1()),
337        "source": pt.source,
338        "rows": pt.rows.iter().map(|f| json!({"field": f.source_col, "sort": sort_name(f.sort), "group": group_name(f.date_group), "collapsed": f.collapsed_items})).collect::<Vec<_>>(),
339        "columns": pt.columns.iter().map(|f| json!({"field": f.source_col, "sort": sort_name(f.sort), "group": group_name(f.date_group), "collapsed": f.collapsed_items})).collect::<Vec<_>>(),
340        "values": pt.values.iter().map(|v| json!({"field": v.source_col, "func": func_name(v.func), "name": v.name, "showAs": show_as_name(v.show_as), "numberFormat": v.number_format})).collect::<Vec<_>>(),
341        "filters": pt.filters.iter().map(|f| json!({"field": f.source_col, "selected": f.selected, "report": !in_axes(pt, &f.source_col)})).collect::<Vec<_>>(),
342        "layout": layout_name(pt.layout),
343        "grandTotalsRows": pt.grand_totals_rows,
344        "grandTotalsCols": pt.grand_totals_cols,
345        "subtotals": if !pt.subtotals { "none" } else if pt.subtotals_top { "top" } else { "bottom" },
346        "style": pt.style,
347        "showHeaders": pt.show_headers,
348        "calculatedFields": pt.calculated_fields.iter().map(|c| json!({"name": c.name, "formula": c.formula})).collect::<Vec<_>>(),
349        "valuesOnRows": pt.values_on_rows,
350        "valuesPosition": pt.values_position,
351        "enableShowDetails": pt.enable_show_details,
352    })
