9pub(super) fn filter(s: &mut Session, p: &Json) -> Result<Json> {
10    const C: &str = "pivot.filter";
11    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
12    let selected: Option<Vec<String>> = match p.get("selected") {
13        None | Some(Json::Null) => None,
14        Some(Json::Array(a)) => {
15            if a.len() > 100_000 {
16                return Err(bad(C, "too many items"));
17            }
18            let v: Vec<String> = a
19                .iter()
20                .map(|x| match x {
21                    Json::String(s) => s.clone(),
22                    Json::Null => "(blank)".into(),
23                    other => other.to_string(),
24                })
25                .collect();
26            if v.is_empty() {
27                return Err(other("You must select at least one item."));
28            }
29            Some(v)
30        }
31        Some(Json::String(s)) => Some(vec![s.clone()]),
32        Some(_) => return Err(bad(C, "`selected` is a list of item labels or null")),
33    };
34    update(s, p, |pt, src, _| {
35        let field = canonical(C, src, &field)?;
36        let i = pt.filters.iter().position(|f| f.source_col.eq_ignore_ascii_case(&field));
37        match (i, selected) {
38            (Some(i), None) => {
39                if in_axes(pt, &field) {
40                    pt.filters.remove(i);
41                } else if let Some(f) = pt.filters.get_mut(i) {
42                    f.selected = None;
43                }
44            }
45            (Some(i), Some(sel)) => {
46                if let Some(f) = pt.filters.get_mut(i) {
47                    f.selected = Some(sel);
48                }
49            }
50            (None, Some(sel)) => pt.filters.push(PivotFilter { source_col: field.clone(), selected: Some(sel) }),
51            (None, None) => {}
52        }
53        Ok(json!({"field": field}))
54    })
55}

57fn axis_field<'a>(cmd: &str, pt: &'a mut PivotTable, field: &str) -> Result<&'a mut PivotField> {
58    pt.rows
59        .iter_mut()
60        .chain(pt.columns.iter_mut())
61        .find(|f| f.source_col.eq_ignore_ascii_case(field))
62        .ok_or_else(|| bad(cmd, format!("`{field}` isn't in the rows or columns area")))
63}

65pub(super) fn sort(s: &mut Session, p: &Json) -> Result<Json> {
66    const C: &str = "pivot.sort";
67    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
68    let order = match str_param(p, "order").unwrap_or("asc").to_ascii_lowercase().as_str() {
69        "asc" | "ascending" | "a-z" => PivotSort::Asc,
70        "desc" | "descending" | "z-a" => PivotSort::Desc,
71        "none" | "manual" | "source" => PivotSort::None,
72        o => return Err(bad(C, format!("unknown order `{o}` (asc|desc|none)"))),
73    };
74    update(s, p, |pt, _, _| {
75        axis_field(C, pt, &field)?.sort = order;
76        Ok(json!({}))
77    })
78}

80pub(super) fn group(s: &mut Session, p: &Json) -> Result<Json> {
81    const C: &str = "pivot.group";
82    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
83    let raw = str_param(p, "by").unwrap_or("months");
84    let by = group_param(Some(raw)).ok_or_else(|| bad(C, format!("unknown grouping `{raw}` (years|quarters|months|days|none)")))?;
85    update(s, p, |pt, src, _| {
86        if by != PivotDateGroup::None {
87            let c = src.col(&field).ok_or_else(|| bad(C, format!("no field `{field}` in the source data")))?;
88            if !src.numeric.get(c).copied().unwrap_or(false) {
89                return Err(other("Cannot group that selection."));
90            }
91        }
92        let f = axis_field(C, pt, &field)?;
93        f.date_group = by;
94        f.collapsed_items.clear();
95        Ok(json!({}))
96    })
97}

99pub(super) fn collapse(s: &mut Session, p: &Json) -> Result<Json> {
100    const C: &str = "pivot.collapse";
101    let field = str_param(p, "field").ok_or_else(|| bad(C, "missing `field`"))?.to_string();
102    let on = bool_param(p, "collapse").unwrap_or(true);
103    let items: Option<Vec<String>> =
104        p.get("items").and_then(Json::as_array).map(|a| a.iter().take(100_000).filter_map(|x| x.as_str().map(str::to_string)).collect());
105    update(s, p, |pt, src, wb| {
106        let all = match &items {
107            Some(v) => v.clone(),
108            None => {
109                let f = pt.rows.iter().chain(pt.columns.iter()).find(|f| f.source_col.eq_ignore_ascii_case(&field));
110                let g = f.map(|f| f.date_group).unwrap_or_default();
111                match src.col(&field) {
112                    Some(c) => pv::distinct_labels(wb, src, c, g, 100_000),
113                    None => vec![],
114                }
115            }
116        };
117        let f = axis_field(C, pt, &field)?;
118        if on {
119            for it in all {
120                if !f.collapsed_items.iter().any(|x| x.eq_ignore_ascii_case(&it)) {
121                    f.collapsed_items.push(it);
122                }
123            }
124        } else if items.is_none() {
125            f.collapsed_items.clear();
126        } else {
127            f.collapsed_items.retain(|x| !all.iter().any(|a| a.eq_ignore_ascii_case(x)));
128        }
129        Ok(json!({"collapsed": f.collapsed_items}))
130    })
131}

133pub(super) fn layout(s: &mut Session, p: &Json) -> Result<Json> {
134    const C: &str = "pivot.layout";
135    let lay = match str_param(p, "layout").map(|x| x.to_ascii_lowercase()) {
136        Some(l) if l == "compact" => Some(PivotLayout::Compact),
137        Some(l) if l == "outline" => Some(PivotLayout::Outline),
138        Some(l) if l == "tabular" => Some(PivotLayout::Tabular),
139        Some(l) => return Err(bad(C, format!("unknown layout `{l}` (compact|outline|tabular)"))),
140        None => None,
141    };
142    let subtotals: Option<(bool, Option<bool>)> = match p.get("subtotals") {
143        Some(Json::Bool(b)) => Some((*b, None)),
144        Some(Json::String(m)) => match m.to_ascii_lowercase().as_str() {
145            "top" => Some((true, Some(true))),
146            "bottom" => Some((true, Some(false))),
147            "none" | "off" => Some((false, None)),
148            o => return Err(bad(C, format!("unknown subtotals `{o}` (top|bottom|none)"))),
149        },
150        _ => None,
151    };
152    let style = str_param(p, "style").map(str::to_string);
153    if style.as_ref().is_some_and(|s| s.len() > 64) {
154        return Err(bad(C, "style name too long"));
155    }
156    let gr = bool_param(p, "grandTotalsRows");
157    let gc = bool_param(p, "grandTotalsCols");
158    let hd = bool_param(p, "showHeaders");
159    update(s, p, |pt, _, _| {
160        if let Some(l) = lay {
161            pt.layout = l;
162            if l == PivotLayout::Tabular {
163                pt.subtotals_top = false;
164            }
165        }
166        if let Some((on, top)) = subtotals {
167            pt.subtotals = on;
168            if let Some(t) = top {
169                pt.subtotals_top = t;
170            }
171        }
172        if let Some(st) = &style {
173            pt.style = st.clone();
174        }
175        if let Some(v) = gr {
176            pt.grand_totals_rows = v;
177        }
178        if let Some(v) = gc {
179            pt.grand_totals_cols = v;
180        }
181        if let Some(v) = hd {
182            pt.show_headers = v;
183        }
184        Ok(json!({"layout": layout_name(pt.layout)}))
185    })
186}

188pub(super) fn grand_totals(s: &mut Session, p: &Json) -> Result<Json> {
189    let (r, c) = match str_param(p, "mode").map(|m| m.to_ascii_lowercase()) {
190        Some(m) if m == "off" => (Some(false), Some(false)),
191        Some(m) if m == "on" => (Some(true), Some(true)),
192        Some(m) if m == "rows" => (Some(true), Some(false)),
193        Some(m) if m == "columns" || m == "cols" => (Some(false), Some(true)),
194        Some(m) => return Err(bad("pivot.grandTotals", format!("unknown mode `{m}` (off|on|rows|columns)"))),
195        None => (bool_param(p, "rows"), bool_param(p, "cols")),
196    };
197    let mut q = p.as_object().cloned().unwrap_or_default();
198    if let Some(v) = r {
199        q.insert("grandTotalsRows".into(), json!(v));
200    }
201    if let Some(v) = c {
202        q.insert("grandTotalsCols".into(), json!(v));
203    }
204    layout(s, &Json::Object(q))
205}

211pub(super) fn refresh_all(s: &mut Session, p: &Json) -> Result<Json> {
212    let replace = bool_param(p, "replace").unwrap_or(false);
213    edit(s, |cx| {
214        let mut n = 0;
215        let mut errors = vec![];
216        for si in 0..cx.wb.sheets.len() {
217            let count = cx.wb.sheet(si).map(|sh| sh.pivots.len()).unwrap_or(0);
218            for pi in 0..count {
219                let name = cx.wb.sheet(si).and_then(|sh| sh.pivots.get(pi)).map(|p| p.name.clone()).unwrap_or_default();
220                match pv::refresh(&mut cx.wb, si, pi, replace) {
221                    Ok(_) => n += 1,
222                    Err(e) => errors.push(json!({"pivot": name, "error": e})),
223                }
224            }
225        }
226        cx.structural = n > 0;
227        Ok(json!({"refreshed": n, "errors": errors}))
228    })
229}

231pub(super) fn change_source(s: &mut Session, p: &Json) -> Result<Json> {
232    const C: &str = "pivot.changeSource";
233    if str_param(p, "source").or_else(|| str_param(p, "range")).is_none() {
234        return Err(bad(C, "missing `source`"));
235    }
236    let source = source_param(s, p)?;
237    let (si, _) = find_pivot(s, p)?;
238    let src = pv::read_source(&s.doc()?.wb, &source, si).map_err(other)?;
239    update(s, p, move |pt, _, _| {
240        pt.source = source.clone();
241        let mut dropped: Vec<String> = vec![];
242        let keep = |name: &str, dropped: &mut Vec<String>| {
243            let ok = src.col(name).is_some();
244            if !ok {
245                dropped.push(name.to_string());
246            }
247            ok
248        };
249        // Axis fields leave one by one so Σ Values keeps its place among the rest.
250        for rows in [true, false] {
251            let mut i = 0;
252            loop {
253                let list = if rows { &mut pt.rows } else { &mut pt.columns };
254                let Some(f) = list.get(i) else { break };
255                if keep(&f.source_col, &mut dropped) {
256                    i += 1;
257                } else {
258                    list.remove(i);
259                    shift_values(pt, rows, i, false);
260                }
261            }
262        }
263        pt.values.retain(|f| keep(&f.source_col, &mut dropped));
264        pt.filters.retain(|f| keep(&f.source_col, &mut dropped));
265        Ok(json!({"source": source, "dropped": dropped}))
266    })
267}

269pub(super) fn delete(s: &mut Session, p: &Json) -> Result<Json> {
270    let (si, pi) = find_pivot(s, p)?;
271    edit(s, |cx| {
272        // Its PivotCharts become ordinary charts.
273        let link = cx.wb.sheet(si).and_then(|sh| sh.pivots.get(pi).map(|pt| (sh.name.clone(), pt.name.clone())));
274        if let Some((sheet, name)) = link {
275            for i in 0..cx.wb.sheets.len() {
276                let linked = |c: &Chart| c.pivot.as_deref().is_some_and(|l| pv::links_to(l, &sheet, &name));
277                if cx.wb.sheet(i).is_some_and(|sh| sh.charts.iter().any(linked)) {
278                    for c in cx.sheet_mut(i)?.charts.iter_mut().filter(|c| linked(c)) {
279                        c.pivot = None;
280                    }
281                }
282            }
283        }
284        pv::clear_output(&mut cx.wb, si, pi);
285        let sh = cx.sheet_mut(si)?;
286        if pi < sh.pivots.len() {
287            sh.pivots.remove(pi);
288        }
289        cx.structural = true;
290        Ok(json!({}))
291    })
