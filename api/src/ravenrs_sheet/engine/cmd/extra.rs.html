338/// Workbook-level JSON stored in a hidden defined name (round-trips through XLSX).
339fn stash_get(wb: &Workbook, key: &str) -> Json {
340    wb.names
341        .iter()
342        .find(|n| n.name == key)
343        .and_then(|n| n.formula.strip_prefix('"').and_then(|t| t.strip_suffix('"')).map(|t| t.replace("\"\"", "\"")))
344        .and_then(|t| serde_json::from_str(&t).ok())
345        .unwrap_or(json!({}))
346}
347
348/// Like `stash_get`, for the name-keyed lists (views, scripts): a stored value that is not a JSON object is treated as empty.
349fn stash_map(wb: &Workbook, key: &str) -> Json {
350    let v = stash_get(wb, key);
351    if v.is_object() { v } else { json!({}) }
352}
353
354fn stash_set(wb: &mut Workbook, key: &str, v: &Json) {
355    let text = v.to_string();
356    let formula = format!("\"{}\"", text.replace('"', "\"\""));
357    wb.names.retain(|n| n.name != key);
358    wb.names.push(DefinedName { name: key.into(), scope: None, formula, comment: String::new(), hidden: true });
359}

364/// Copy of the params that is always a JSON object, so keys can be assigned; any other JSON value counts as no params (like null).
365fn object_params(p: &Json) -> Json {
366    Json::Object(p.as_object().cloned().unwrap_or_default())
