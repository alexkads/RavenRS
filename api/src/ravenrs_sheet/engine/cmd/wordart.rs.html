106fn style_param(command: &str, p: &Json) -> Result<Option<&'static WordArtStyle>> {
107    match p.get("style") {
108        None | Some(Json::Null) => Ok(None),
109        Some(Json::Number(n)) => n
110            .as_u64()
111            .and_then(|i| STYLES.get(usize::try_from(i).ok()?))
112            .map(Some)
113            .ok_or_else(|| bad(command, format!("`style` index must be 0..{}", STYLES.len() - 1))),
114        Some(Json::String(name)) => STYLES
115            .iter()
116            .find(|s| s.name.eq_ignore_ascii_case(name.trim()))
117            .map(Some)
118            .ok_or_else(|| bad(command, format!("unknown WordArt style `{name}`"))),
119        Some(_) => Err(bad(command, "`style` must be a preset name or index")),
120    }
121}
122
123fn text_param(command: &str, p: &Json) -> Result<Option<String>> {
124    match p.get("text") {
125        None | Some(Json::Null) => Ok(None),
126        Some(Json::String(t)) if t.is_empty() => Err(bad(command, "`text` must not be empty")),
127        Some(Json::String(t)) if t.chars().nth(MAX_CHARS).is_some() => Err(bad(command, format!("`text` is longer than {MAX_CHARS} characters"))),
128        Some(Json::String(t)) => Ok(Some(t.clone())),
129        Some(_) => Err(bad(command, "`text` must be a string")),
130    }
131}

133/// Applies a preset, then explicit overrides, to a shape's format and fill.
134fn apply(command: &str, shape: &mut Shape, style: Option<&WordArtStyle>, p: &Json) -> Result<()> {
135    if let Some(st) = style {
136        shape.format.text_color = Some(st.fill);
137        shape.format.text_outline = st.outline;
138        shape.format.text_shadow = st.shadow;
139        shape.format.bold = st.bold;
140        shape.format.italic = st.italic;
141        shape.format.font_size = Some(st.size);
142    }
143    let f = &mut shape.format;
144    if let Some(c) = p.get("fill").filter(|v| !v.is_null()) {
145        f.text_color = Some(super::format::color_param(Some(c)).ok_or_else(|| bad(command, "`fill` must be a hex colour"))?);
146    }
147    match p.get("outline") {
148        None | Some(Json::Null) => {}
149        Some(Json::Bool(false)) => f.text_outline = None,
150        Some(v) => {
151            let c = super::format::color_param(Some(v)).ok_or_else(|| bad(command, "`outline` must be a hex colour or false"))?;
152            let w = f64_param(p, "outlineWidth").map_or(f.text_outline.map_or(1.5, |o| o.1), |w| w.clamp(0.25, 20.0) as f32);
153            f.text_outline = Some((c, w));
154        }
155    }
156    if let (Some(w), Some((c, _))) = (f64_param(p, "outlineWidth"), f.text_outline) {
157        f.text_outline = Some((c, w.clamp(0.25, 20.0) as f32));
158    }
159    match p.get("shadow") {
160        None | Some(Json::Null) => {}
161        Some(Json::Bool(false)) => f.text_shadow = None,
162        Some(Json::Bool(true)) => f.text_shadow = Some(f.text_shadow.unwrap_or(rgb(0x808080))),
163        Some(v) => {
164            f.text_shadow = Some(super::format::color_param(Some(v)).ok_or_else(|| bad(command, "`shadow` must be a hex colour or a boolean"))?)
165        }
166    }
167    if let Some(b) = bool_param(p, "bold") {
168        f.bold = b;
169    }
170    if let Some(b) = bool_param(p, "italic") {
171        f.italic = b;
172    }
173    if let Some(sz) = f64_param(p, "fontSize") {
174        f.font_size = Some(sz.clamp(1.0, MAX_POINTS) as f32);
175    }
176    Ok(())
177}

179fn insert_word_art(s: &mut Session, p: &Json) -> Result<Json> {
180    let style = style_param("insert.wordArt", p)?.unwrap_or(&STYLES[0]);
181    let text = text_param("insert.wordArt", p)?.unwrap_or_else(|| DEFAULT_TEXT.to_string());
182    let d = s.doc()?;
183    if d.wb.active().is_some_and(|sh| sh.is_protected()) {
184        return Err(EngineError::Other(PROTECTED.into()));
185    }
186    let id = d.wb.next_object_id();
187    let at = cell_param(s, p, "at")?.unwrap_or(d.selection.active);
188    let mut shape = Shape {
189        id,
190        kind: ShapeKind::WordArt,
191        anchor: Anchor { cell: at, dx: 0.0, dy: 0.0, width: 0.0, height: 0.0, mode: anchor_mode_param(p).unwrap_or_default() },
192        fill: Color::Auto,
193        line: Color::Auto,
194        text: text.clone(),
195        format: ShapeFormat { no_line: true, align: Some(ShapeTextAlign::Center), anchor: Some(ShapeTextAnchor::Middle), ..Default::default() },
196        math: Default::default(),
197    };
198    apply("insert.wordArt", &mut shape, Some(style), p)?;
199    // Default box: the longest line at the font size (about 0.62 em per character) plus padding.
200    let pt = f64::from(shape.format.font_size.unwrap_or(36.0));
201    let longest = text.split('\n').map(|l| l.chars().count()).max().unwrap_or(1) as f64;
202    let lines = text.split('\n').count() as f64;
203    let width = f64_param(p, "width").unwrap_or(longest * pt * 0.62 * 96.0 / 72.0 + 24.0).clamp(8.0, MAX_SIDE);
204    let height = f64_param(p, "height").unwrap_or(lines * pt * 1.35 * 96.0 / 72.0 + 12.0).clamp(8.0, MAX_SIDE);
205    shape.anchor.width = width as f32;
206    shape.anchor.height = height as f32;
207    let sheet = d.wb.active_sheet();
208    edit(s, |cx| {
209        cx.sheet_mut(sheet)?.shapes.push(shape);
210        Ok(json!({"shape": id}))
211    })
212}

214fn set_word_art(s: &mut Session, p: &Json) -> Result<Json> {
215    let id = u32_param(p, "id").ok_or_else(|| bad("wordArt.set", "`id` must be a WordArt id"))?;
216    let style = style_param("wordArt.set", p)?;
217    let text = text_param("wordArt.set", p)?;
218    let d = s.doc()?;
219    let sheet = d.wb.active_sheet();
220    let sh = d.wb.active().ok_or_else(|| bad("wordArt.set", "no such WordArt"))?;
221    let current = sh.shapes.iter().find(|x| x.id == id).ok_or_else(|| bad("wordArt.set", "no such WordArt"))?;
222    if current.kind != ShapeKind::WordArt {
223        return Err(bad("wordArt.set", "the shape is not WordArt"));
224    }
225    if sh.is_protected() {
226        return Err(EngineError::Other(PROTECTED.into()));
227    }
228    let mut updated = current.clone();
229    apply("wordArt.set", &mut updated, style, p)?;
230    if let Some(t) = text {
231        updated.text = t;
232    }
233    if &updated == current {
234        return ok();
235    }
236    edit(s, |cx| {
237        let shape = cx.sheet_mut(sheet)?.shapes.iter_mut().find(|x| x.id == id).ok_or_else(|| bad("wordArt.set", "no such WordArt"))?;
238        *shape = updated;
239        Ok(Json::Null)
240    })
