10fn copy_impl(s: &mut Session, p: &Json, cut: bool) -> Result<Json> {
11    let multiple = || EngineError::Other("This action won't work on multiple selections.".into());
12    let sheet = target_sheet(s, p)?;
13    let ranges = target_ranges(s, p)?;
14    if ranges.len() > 1 && cut {
15        return Err(multiple());
16    }
17    // The selection as made (untrimmed), for Insert Copied Cells.
18    let selected = ranges.first().copied().unwrap_or_default();
19    let d = s.doc()?;
20    let Some(sh) = d.wb.sheet(sheet) else { return Err(EngineError::NoDocument) };
21    let trim = |range: RangeRef| {
22        if range.is_full_cols() || range.is_full_rows() {
23            let used = sh.used_range().unwrap_or(RangeRef::cell(range.start));
24            range.intersection(&used.union(&RangeRef::cell(range.start))).unwrap_or(range)
25        } else {
26            range
27        }
28    };
29    let (wb, range, areas) = if ranges.len() > 1 {
30        let areas: Vec<RangeRef> = ranges.iter().map(|r| trim(*r)).collect();
31        let (wb, block) = slide_together(&d.wb, sheet, &areas).ok_or_else(multiple)?;
32        (std::sync::Arc::new(wb), block, areas)
33    } else {
34        (d.wb.clone(), trim(ranges.first().copied().unwrap_or_default()), Vec::new())
35    };
36    let text = crate::engine::display::range_text(&wb, sheet, range);
37    // The HTML rendering is for the host clipboard (UI); it can reach 4 MB, so agents and
38    // scripts get it only when they ask for it.
39    let html = if p.get("html").and_then(Json::as_bool) == Some(true) { crate::engine::io::range_html(&wb, sheet, range, &text) } else { None };
40    let clip = Clipboard { wb, sheet, range, cut, doc_uid: d.uid, text: text.clone(), areas, selected, trust: d.trust };
41    s.clipboard = Some(clip);
42    let mut result = json!({"range": range.a1(), "text": text});
43    if let Some(html) = html {
44        result["html"] = Json::String(html);
45    }
46    Ok(result)
47}

49/// Excel copies a multi-area selection only when the areas line up: all cover the same rows (they
50/// slide together side by side, left to right) or all cover the same columns (stacked, top to
51/// bottom), without overlapping. Returns a copy of the workbook whose sheet holds the areas packed
52/// into one block at the first area's place, and that block. Formulas travel as their values, as
53/// in Excel. `None` when the areas don't line up.
54fn slide_together(wb: &crate::model::Workbook, sheet: usize, areas: &[RangeRef]) -> Option<(crate::model::Workbook, RangeRef)> {
55    let first = *areas.first()?;
56    let across = areas.iter().all(|a| a.start.row == first.start.row && a.end.row == first.end.row);
57    let down = areas.iter().all(|a| a.start.col == first.start.col && a.end.col == first.end.col);
58    let mut sorted = areas.to_vec();
59    if across {
60        sorted.sort_by_key(|a| a.start.col);
61    } else if down {
62        sorted.sort_by_key(|a| a.start.row);
63    } else {
64        return None;
65    }
66    if sorted.windows(2).any(|w| matches!(w, [a, b] if a.intersects(b))) {
67        return None;
68    }
69    let anchor = sorted.first()?.start;
70    let (h, w) = if across {
71        (first.height(), sorted.iter().map(RangeRef::width).fold(0u32, u32::saturating_add))
72    } else {
73        (sorted.iter().map(RangeRef::height).fold(0u32, u32::saturating_add), first.width())
74    };
75    let block = RangeRef::new(anchor, anchor.offset(i64::from(h) - 1, i64::from(w) - 1)?);
76    let src = wb.sheet(sheet)?;
77    let mut out = wb.clone();
78    let dst = out.sheet_mut(sheet)?;
79    dst.take_cells(block);
80    dst.spill.retain(|c, _| !block.contains(*c));
81    dst.spill_ranges.retain(|c, _| !block.contains(*c));
82    dst.comments.retain(|c, _| !block.contains(*c));
83    dst.merges.retain(|m| !m.intersects(&block));
84    if across {
85        dst.cols.retain(|c, _| !(block.start.col..=block.end.col).contains(c));
86    } else {
87        dst.rows.retain(|r, _| !(block.start.row..=block.end.row).contains(r));
88    }
89    let mut next = if across { anchor.col } else { anchor.row };
90    for a in &sorted {
91        let (dr, dc) = if across { (0, i64::from(next) - i64::from(a.start.col)) } else { (i64::from(next) - i64::from(a.start.row), 0) };
92        let cells = src.cells.iter_range(*a).map(|(c, _)| c).chain(src.spill.keys().copied().filter(|c| a.contains(*c)));
93        for c in cells {
94            let Some(to) = c.offset(dr, dc) else { continue };
95            let style = src.cell(c).map(|x| x.style).unwrap_or_default();
96            dst.set_cell(to, Cell { value: src.value(c), formula: None, style, py: None });
97            if let Some(picture) = src.cell_pictures.get(&c) {
98                dst.cell_pictures.insert(to, picture.clone());
99            }
100        }
101        for (c, comment) in src.comments.iter().filter(|(c, _)| a.contains(**c)) {
102            if let Some(to) = c.offset(dr, dc) {
103                dst.comments.insert(to, comment.clone());
104            }
105        }
106        for m in src.merges.iter().filter(|m| a.contains_range(m)) {
107            if let (Some(s), Some(e)) = (m.start.offset(dr, dc), m.end.offset(dr, dc)) {
108                dst.merges.push(RangeRef::new(s, e));
109            }
110        }
111        // Column widths (side by side) or row heights and hidden rows (stacked) follow their cells.
112        if across {
113            for (c, info) in src.cols.range(a.start.col..=a.end.col) {
114                dst.cols.insert(next.saturating_add(c.saturating_sub(a.start.col)), *info);
115            }
116        } else {
117            for (r, info) in src.rows.range(a.start.row..=a.end.row) {
118                dst.rows.insert(next.saturating_add(r.saturating_sub(a.start.row)), *info);
119            }
120        }
121        next = next.saturating_add(if across { a.width() } else { a.height() });
122    }
123    Some((out, block))
124}

154pub(super) fn paste(s: &mut Session, p: &Json) -> Result<Json> {
155    // External text (from the system clipboard) when it differs from ours.
156    if let Some(text) = str_param(p, "text")
157        && s.clipboard.as_ref().is_none_or(|c| c.text.trim_end_matches(['\r', '\n']) != text.trim_end_matches(['\r', '\n']))
158    {
159        return paste_text(s, p, text);
160    }
161    if s.clipboard.is_none() {
162        return Err(EngineError::Other("Nothing to paste.".into()));
163    }
164    paste_special(s, &json!({"what": "all", "at": p.get("at").cloned().unwrap_or(Json::Null)}))
165}

167/// Drag and drop (dragging the selection's border): moves `range` so its top-left cell lands on
168/// `to`, as Cut and Paste would. Like Excel, it asks before replacing cells that hold data or show a
169/// spilled value (other than the moved cells themselves); run it again with `overwrite: true` to replace them. A
170/// refused or failed move leaves the clipboard as it was.
171pub(super) fn range_move(s: &mut Session, p: &Json) -> Result<Json> {
172    let sheet = target_sheet(s, p)?;
173    let to = cell_param_on(s, p, "to", sheet)?.ok_or_else(|| bad("range.move", "missing `to`"))?;
174    let kept = s.clipboard.take();
175    let moved = cut(s, p).and_then(|_| {
176        let src = s.clipboard.as_ref().map(|c| c.range).unwrap_or_default();
177        let dest = to.offset(i64::from(src.height()) - 1, i64::from(src.width()) - 1).map(|end| RangeRef::new(to, end));
178        if bool_param(p, "overwrite") != Some(true)
179            && let Some(dest) = dest
180            && let Some(sh) = s.doc()?.wb.sheet(sheet)
181            && (sh.cells.iter_range(dest).any(|(c, cell)| !src.contains(c) && cell.has_content())
182                || sh.spill.range(dest.start..=dest.end).any(|(c, v)| dest.contains(*c) && !src.contains(*c) && *v != Value::Empty))
183        {
184            return Err(EngineError::Confirm("There's already data here. Do you want to replace it?".into()));
185        }
186        // On the range's sheet, which `paste` would leave to the active one. The selection
187        // belongs to the active sheet: it follows the moved cells only when they are on it.
188        let selection = s.doc()?.selection.clone();
189        let pasted = paste_special(s, &json!({"what": "all", "at": to.a1(), "sheet": sheet}))?;
190        if sheet != s.doc()?.wb.active_sheet() {
191            s.doc_mut()?.selection = selection;
192        }
193        Ok(pasted)
194    });
195    if moved.is_err() {
196        s.clipboard = kept;
197    }
198    moved
199}

201fn paste_text(s: &mut Session, p: &Json, text: &str) -> Result<Json> {
202    let sheet = target_sheet(s, p)?;
203    let at = cell_param_on(s, p, "at", sheet)?.unwrap_or(s.doc()?.selection.current().start);
204    let rows: Vec<Vec<String>> = parse_tsv(text);
205    let h = rows.len();
206    let w = rows.iter().map(Vec::len).max().unwrap_or(0);
207    // The HTML flavour, when the host passed it: the number behind each cell's shown text. Read
208    // only as far as the text's own shape, so spans in the markup can't expand without bound.
209    let html = str_param(p, "html").map(|html| crate::engine::clip_html::table_cells(html, h, w)).unwrap_or_default();
210    edit(s, |cx| {
211        for (ri, row) in rows.iter().enumerate() {
212            for (ci, v) in row.iter().enumerate() {
213                let Some(c) = at.offset(ri as i64, ci as i64) else { continue };
214                let mut cell = input_to_cell(v, sheet, c, &mut cx.wb).unwrap_or_else(|_| Some(Cell::value(Value::text(v.as_str()))));
215                if let Some(cl) = cell.as_mut() {
216                    numeric_from_html(cl, v, html.get(ri).and_then(|r| r.get(ci)), &mut cx.wb);
217                }
218                let sh = cx.sheet_mut(sheet)?;
219                match cell {
220                    Some(cell) => sh.set_cell(c, cell),
221                    None => {
222                        sh.remove_cell(c);
223                    }
224                }
225                cx.touch(sheet, c);
226            }
227        }
228        if h > 0 && w > 0 {
229            let end = at.offset_clamped(h as i64 - 1, w as i64 - 1);
230            *cx.sel = crate::engine::Selection { active: at, anchor: at, ranges: vec![RangeRef::new(at, end)] };
231        }
232        Ok(())
233    })?;
234    Ok(json!({"rows": h, "cols": w}))
235}

237/// A pasted cell that came out as text ("Rp10,000.00" isn't en-US input) becomes the number the
238/// source recorded for it in its HTML, with a format that shows the same text when one can be
239/// inferred and the cell has none. The HTML cell must show the same text as the tab-separated
240/// field, so HTML that belongs to another copy (or a table laid out differently) changes nothing.
241fn numeric_from_html(cell: &mut Cell, field: &str, html: Option<&crate::engine::clip_html::HtmlCell>, wb: &mut crate::model::Workbook) {
242    let Some(hc) = html else { return };
243    let Some(n) = hc.num else { return };
244    if cell.formula.is_some() || !matches!(cell.value, Value::Text(_)) || wb.styles.get(cell.style).num_fmt.as_str() == "@" {
245        return;
246    }
247    if crate::engine::clip_html::normalized(&hc.text) != crate::engine::clip_html::normalized(field) {
248        return;
249    }
250    cell.value = Value::number(n);
251    if wb.styles.get(cell.style).num_fmt.as_str() == "General"
252        && let Some(code) = crate::engine::clip_html::format_for(field, n)
253    {
254        cell.style = wb.styles.derive(cell.style, |s| s.num_fmt = crate::model::NumFmt::new(&code));
255    }
256}

258/// Tab-separated text with Excel-style quoting.
259pub(crate) fn parse_tsv(text: &str) -> Vec<Vec<String>> {
260    let mut rows = Vec::new();
261    let mut row = Vec::new();
262    let mut field = String::new();
263    let mut chars = text.chars().peekable();
264    // Text read again as unquoted, ahead of the rest of the input: a field that opened with a
265    // quote but turned out not to be quoted.
266    let mut replay: std::collections::VecDeque<char> = std::collections::VecDeque::new();
267    // Text is replayed at most as much as the input is long; past that, quotes are text, so
268    // hostile input full of fields that only look quoted reads in linear time.
269    let mut replay_budget = text.len();
270    let mut quoted = false;
271    let mut at_start = true;
272    // A quoted field's text as it stands in the input, for a field that turns out not to be one.
273    let mut raw = String::new();
274    loop {
275        let c = match replay.pop_front().or_else(|| chars.next()) {
276            Some(c) => c,
277            // A quote that never closes: the field is its text as written.
278            None if quoted => {
279                quoted = false;
280                field.clear();
281                field.push('"');
282                replay_budget = replay_budget.saturating_sub(raw.len());
283                replay.extend(raw.chars().skip(1));
284                raw.clear();
285                continue;
286            }
287            None => break,
288        };
289        if quoted {
290            raw.push(c);
291            if c == '"' {
292                let next = replay.front().copied().or_else(|| chars.peek().copied());
293                if next == Some('"') {
294                    field.push('"');
295                    raw.push('"');
296                    if replay.pop_front().is_none() {
297                        chars.next();
298                    }
299                } else {
300                    quoted = false;
301                    // A field is quoted only when its closing quote ends it, as Excel reads it:
302                    // `"hi" there` is that text, quotes included, and a tab or line break inside
303                    // it still ends the cell, as if the leading quote weren't there.
304                    if !matches!(next, None | Some('\t' | '\n' | '\r')) {
305                        field.clear();
306                        field.push('"');
307                        replay_budget = replay_budget.saturating_sub(raw.len());
308                        let rest: Vec<char> = std::mem::take(&mut raw).chars().skip(1).collect();
309                        for ch in rest.into_iter().rev() {
310                            replay.push_front(ch);
311                        }
312                    }
313                }
314            } else {
315                field.push(c);
316            }
317            continue;
318        }
319        match c {
320            '"' if at_start && replay_budget > 0 => {
321                quoted = true;
322                at_start = false;
323                raw = String::from('"');
324            }
325            '\t' => {
326                row.push(std::mem::take(&mut field));
327                at_start = true;
328            }
329            '\r' => {}
330            '\n' => {
331                row.push(std::mem::take(&mut field));
332                rows.push(std::mem::take(&mut row));
333                at_start = true;
334            }
335            _ => {
336                field.push(c);
337                at_start = false;
338            }
339        }
340    }
341    if !field.is_empty() || !row.is_empty() {
342        row.push(field);
343        rows.push(row);
344    }
345    rows
