10fn copy_impl(s: &mut Session, p: &Json, cut: bool) -> Result<Json> {
11    let multiple = || EngineError::Other("This action won't work on multiple selections.".into());
12    let sheet = target_sheet(s, p)?;
13    let ranges = target_ranges(s, p)?;
14    if ranges.len() > 1 && cut {
15        return Err(multiple());
16    }
17    // The selection as made (untrimmed), for Insert Copied Cells.
18    let mut selected = ranges.first().copied().unwrap_or_default();
19    let d = s.doc()?;
20    let Some(sh) = d.wb.sheet(sheet) else { return Err(EngineError::NoDocument) };
21    let trim = |range: RangeRef| {
22        if range.is_full_cols() || range.is_full_rows() {
23            let used = sh.used_range().unwrap_or(RangeRef::cell(range.start));
24            range.intersection(&used.union(&RangeRef::cell(range.start))).unwrap_or(range)
25        } else {
26            range
27        }
28    };
29    let (wb, range, areas) = if ranges.len() > 1 {
30        let areas: Vec<RangeRef> = ranges.iter().map(|r| trim(*r)).collect();
31        let (wb, block) = slide_together(&d.wb, sheet, &areas).ok_or_else(multiple)?;
32        (std::sync::Arc::new(wb), block, areas)
33    } else {
34        let range = trim(ranges.first().copied().unwrap_or_default());
35        // A copied range with rows a filter hides takes only the rows showing: they are copied
36        // like a multi-area selection stacked top to bottom (formulas as values), as in Excel.
37        // A cut still takes every row.
38        let hidden = if cut { Default::default() } else { super::filtered_rows(sh, range) };
39        let visible = if hidden.is_empty() { Vec::new() } else { visible_row_runs(range, &hidden) };
40        let any_visible = !visible.is_empty();
41        let packed = match visible.as_slice() {
42            [] => (d.wb.clone(), range, Vec::new()),
43            [only] => (d.wb.clone(), *only, Vec::new()),
44            _ => {
45                let (wb, block) = slide_together(&d.wb, sheet, &visible).ok_or_else(multiple)?;
46                (std::sync::Arc::new(wb), block, visible)
47            }
48        };
49        // Whole rows copied with some hidden by a filter come as the block of rows showing, so
50        // Insert Copied Cells makes room for those rows, not for the hidden ones too.
51        if !hidden.is_empty() && any_visible && selected.is_full_rows() {
52            selected = RangeRef::rows(packed.1.start.row, packed.1.end.row);
53        }
54        packed
55    };
56    let text = crate::engine::display::range_text(&wb, sheet, range);
57    // The HTML rendering is for the host clipboard (UI); it can reach 4 MB, so agents and
58    // scripts get it only when they ask for it.
59    let html = if p.get("html").and_then(Json::as_bool) == Some(true) { crate::engine::io::range_html(&wb, sheet, range, &text) } else { None };
60    let clip = Clipboard { wb, sheet, range, cut, doc_uid: d.uid, text: text.clone(), areas, selected, trust: d.trust };
61    s.clipboard = Some(clip);
62    let mut result = json!({"range": range.a1(), "text": text});
63    if let Some(html) = html {
64        result["html"] = Json::String(html);
65    }
66    Ok(result)
67}

69/// The runs of consecutive rows of `range` not in `hidden`, as ranges of `range`'s columns.
70fn visible_row_runs(range: RangeRef, hidden: &std::collections::BTreeSet<u32>) -> Vec<RangeRef> {
71    let mut runs = Vec::new();
72    let mut start = None;
73    for row in range.start.row..=range.end.row {
74        match (hidden.contains(&row), start) {
75            (false, None) => start = Some(row),
76            (true, Some(first)) => {
77                runs.push(RangeRef::new(CellRef::new(first, range.start.col), CellRef::new(row - 1, range.end.col)));
78                start = None;
79            }
80            _ => {}
81        }
82    }
83    if let Some(first) = start {
84        runs.push(RangeRef::new(CellRef::new(first, range.start.col), CellRef::new(range.end.row, range.end.col)));
85    }
86    runs
87}

89/// The validation rules of `src` cut into the areas of a packed block: one rule per area a rule
90/// touches, over that area's part moved by the area's shift. `sorted` areas run in order.
91fn pack_validations(src: &crate::model::Sheet, sorted: &[RangeRef], shifts: &[(i64, i64)], across: bool) -> Vec<crate::model::Validation> {
92    let key = |c: CellRef| if across { c.col } else { c.row };
93    let mut packed_rules = Vec::new();
94    for dv in &src.validations {
95        let Some(from) = dv.ranges.first().map(|r| r.start) else { continue };
96        // The areas each of the rule's ranges touches lie side by side in the sorted list.
97        let end_key = |a: &RangeRef| if across { a.end.col } else { a.end.row };
98        let mut by_area: std::collections::BTreeMap<usize, Vec<RangeRef>> = Default::default();
99        for r in &dv.ranges {
100            let first = sorted.partition_point(|a| end_key(a) < key(r.start));
101            for (i, a) in sorted.iter().enumerate().skip(first).take_while(|(_, a)| key(a.start) <= end_key(r)) {
102                let (dr, dc) = shifts[i];
103                let moved = r.intersection(a).and_then(|x| Some(RangeRef::new(x.start.offset(dr, dc)?, x.end.offset(dr, dc)?)));
104                by_area.entry(i).or_default().extend(moved);
105            }
106        }
107        for ranges in by_area.into_values() {
108            let Some(to) = ranges.first().map(|r| r.start) else { continue };
109            let mut rule = dv.clone();
110            rule.f1 = super::super::data::move_rule_formula(&dv.f1, from, to);
111            rule.f2 = dv.f2.as_ref().map(|f| super::super::data::move_rule_formula(f, from, to));
112            rule.ranges = ranges;
113            packed_rules.push(rule);
114        }
115    }
116    packed_rules
117}

119/// Excel copies a multi-area selection only when the areas line up: all cover the same rows (they
120/// slide together side by side, left to right) or all cover the same columns (stacked, top to
121/// bottom), without overlapping. Returns a copy of the workbook whose sheet holds the areas packed
122/// into one block at the first area's place, and that block. Formulas travel as their values, as
123/// in Excel. `None` when the areas don't line up.
124fn slide_together(wb: &crate::model::Workbook, sheet: usize, areas: &[RangeRef]) -> Option<(crate::model::Workbook, RangeRef)> {
125    let first = *areas.first()?;
126    let across = areas.iter().all(|a| a.start.row == first.start.row && a.end.row == first.end.row);
127    let down = areas.iter().all(|a| a.start.col == first.start.col && a.end.col == first.end.col);
128    let mut sorted = areas.to_vec();
129    if across {
130        sorted.sort_by_key(|a| a.start.col);
131    } else if down {
132        sorted.sort_by_key(|a| a.start.row);
133    } else {
134        return None;
135    }
136    if sorted.windows(2).any(|w| matches!(w, [a, b] if a.intersects(b))) {
137        return None;
138    }
139    let anchor = sorted.first()?.start;
140    let (h, w) = if across {
141        (first.height(), sorted.iter().map(RangeRef::width).fold(0u32, u32::saturating_add))
142    } else {
143        (sorted.iter().map(RangeRef::height).fold(0u32, u32::saturating_add), first.width())
144    };
145    let block = RangeRef::new(anchor, anchor.offset(i64::from(h) - 1, i64::from(w) - 1)?);
146    let src = wb.sheet(sheet)?;
147    let mut out = wb.clone();
148    let dst = out.sheet_mut(sheet)?;
149    dst.take_cells(block);
150    dst.spill.retain(|c, _| !block.contains(*c));
151    dst.spill_ranges.retain(|c, _| !block.contains(*c));
152    dst.comments.retain(|c, _| !block.contains(*c));
153    dst.hyperlinks.retain(|c, _| !block.contains(*c));
154    super::super::data::remove_validation_cells(&mut dst.validations, &[block]);
155    dst.merges.retain(|m| !m.intersects(&block));
156    if across {
157        dst.cols.retain(|c, _| !(block.start.col..=block.end.col).contains(c));
158    } else {
159        dst.rows.retain(|r, _| !(block.start.row..=block.end.row).contains(r));
160    }
161    // Where each area lands: the shift of its cells, and its first row or column in the block.
162    let key = |c: CellRef| if across { c.col } else { c.row };
163    let mut next = if across { anchor.col } else { anchor.row };
164    let mut shifts: Vec<(i64, i64)> = Vec::with_capacity(sorted.len());
165    for a in &sorted {
166        let to = i64::from(next);
167        shifts.push(if across { (0, to - i64::from(a.start.col)) } else { (to - i64::from(a.start.row), 0) });
168        next = next.saturating_add(if across { a.width() } else { a.height() });
169    }
170    // The areas never overlap and run in order, so the area holding a cell is found by search:
171    // each map below is read once, whatever the number of areas.
172    let area_of = |c: CellRef| -> Option<usize> {
173        let i = sorted.partition_point(|a| key(a.start) <= key(c)).checked_sub(1)?;
174        sorted.get(i)?.contains(c).then_some(i)
175    };
176    let moved = |c: CellRef| -> Option<CellRef> {
177        let (dr, dc) = shifts[area_of(c)?];
178        c.offset(dr, dc)
179    };
180    for (a, &(dr, dc)) in sorted.iter().zip(&shifts) {
181        for (c, _) in src.cells.iter_range(*a) {
182            let Some(to) = c.offset(dr, dc) else { continue };
183            let style = src.cell(c).map(|x| x.style).unwrap_or_default();
184            dst.set_cell(to, Cell { value: src.value(c), formula: None, style, py: None });
185            if let Some(picture) = src.cell_pictures.get(&c) {
186                dst.cell_pictures.insert(to, picture.clone());
187            }
188        }
189    }
190    for c in src.spill.keys().copied() {
191        let Some(to) = moved(c) else { continue };
192        let style = src.cell(c).map(|x| x.style).unwrap_or_default();
193        dst.set_cell(to, Cell { value: src.value(c), formula: None, style, py: None });
194        if let Some(picture) = src.cell_pictures.get(&c) {
195            dst.cell_pictures.insert(to, picture.clone());
196        }
197    }
198    // Links and validation rules follow their cells too, so a row the filter hides leaves its own
199    // behind.
200    for (c, link) in &src.hyperlinks {
201        if let Some(to) = moved(*c) {
202            dst.hyperlinks.insert(to, link.clone());
203        }
204    }
205    let packed_rules = pack_validations(src, &sorted, &shifts, across);
206    for (c, comment) in &src.comments {
207        if let Some(to) = moved(*c) {
208            dst.comments.insert(to, comment.clone());
209        }
210    }
211    for m in &src.merges {
212        let Some(i) = area_of(m.start).filter(|&i| sorted[i].contains_range(m)) else { continue };
213        let (dr, dc) = shifts[i];
214        if let (Some(s), Some(e)) = (m.start.offset(dr, dc), m.end.offset(dr, dc)) {
215            dst.merges.push(RangeRef::new(s, e));
216        }
217    }
218    // Column widths (side by side) or row heights and hidden rows (stacked) follow their cells.
219    let mut next = if across { anchor.col } else { anchor.row };
220    for a in &sorted {
221        if across {
222            for (c, info) in src.cols.range(a.start.col..=a.end.col) {
223                dst.cols.insert(next.saturating_add(c.saturating_sub(a.start.col)), *info);
224            }
225        } else {
226            for (r, info) in src.rows.range(a.start.row..=a.end.row) {
227                dst.rows.insert(next.saturating_add(r.saturating_sub(a.start.row)), *info);
228            }
229        }
230        next = next.saturating_add(if across { a.width() } else { a.height() });
231    }
232    dst.validations.extend(packed_rules);
233    Some((out, block))
234}

264pub(super) fn paste(s: &mut Session, p: &Json) -> Result<Json> {
265    // External text (from the system clipboard) when it differs from ours.
266    if let Some(text) = str_param(p, "text")
267        && s.clipboard.as_ref().is_none_or(|c| c.text.trim_end_matches(['\r', '\n']) != text.trim_end_matches(['\r', '\n']))
268    {
269        return paste_text(s, p, text);
270    }
271    if s.clipboard.is_none() {
272        return Err(EngineError::Other("Nothing to paste.".into()));
273    }
274    paste_special(s, &json!({"what": "all", "at": p.get("at").cloned().unwrap_or(Json::Null)}))
275}

277/// Drag and drop (dragging the selection's border): moves `range` so its top-left cell lands on
278/// `to`, as Cut and Paste would. Like Excel, it asks before replacing cells that hold data or show a
279/// spilled value (other than the moved cells themselves); run it again with `overwrite: true` to replace them. A
280/// refused or failed move leaves the clipboard as it was.
281pub(super) fn range_move(s: &mut Session, p: &Json) -> Result<Json> {
282    let sheet = target_sheet(s, p)?;
283    let to = cell_param_on(s, p, "to", sheet)?.ok_or_else(|| bad("range.move", "missing `to`"))?;
284    let kept = s.clipboard.take();
285    let moved = cut(s, p).and_then(|_| {
286        let src = s.clipboard.as_ref().map(|c| c.range).unwrap_or_default();
287        let dest = to.offset(i64::from(src.height()) - 1, i64::from(src.width()) - 1).map(|end| RangeRef::new(to, end));
288        if bool_param(p, "overwrite") != Some(true)
289            && let Some(dest) = dest
290            && let Some(sh) = s.doc()?.wb.sheet(sheet)
291            && (sh.cells.iter_range(dest).any(|(c, cell)| !src.contains(c) && cell.has_content())
292                || sh.spill.range(dest.start..=dest.end).any(|(c, v)| dest.contains(*c) && !src.contains(*c) && *v != Value::Empty))
293        {
294            return Err(EngineError::Confirm("There's already data here. Do you want to replace it?".into()));
295        }
296        // On the range's sheet, which `paste` would leave to the active one. The selection
297        // belongs to the active sheet: it follows the moved cells only when they are on it.
298        let selection = s.doc()?.selection.clone();
299        let pasted = paste_special(s, &json!({"what": "all", "at": to.a1(), "sheet": sheet}))?;
300        if sheet != s.doc()?.wb.active_sheet() {
301            s.doc_mut()?.selection = selection;
302        }
303        Ok(pasted)
304    });
305    if moved.is_err() {
306        s.clipboard = kept;
307    }
308    moved
309}

311fn paste_text(s: &mut Session, p: &Json, text: &str) -> Result<Json> {
312    let sheet = target_sheet(s, p)?;
313    let at = cell_param_on(s, p, "at", sheet)?.unwrap_or(s.doc()?.selection.current().start);
314    let rows: Vec<Vec<String>> = parse_tsv(text);
315    let h = rows.len();
316    let w = rows.iter().map(Vec::len).max().unwrap_or(0);
317    // The HTML flavour, when the host passed it: the number behind each cell's shown text. Read
318    // only as far as the text's own shape, so spans in the markup can't expand without bound.
319    let html = str_param(p, "html").map(|html| crate::engine::clip_html::table_cells(html, h, w)).unwrap_or_default();
320    edit(s, |cx| {
321        for (ri, row) in rows.iter().enumerate() {
322            for (ci, v) in row.iter().enumerate() {
323                let Some(c) = at.offset(ri as i64, ci as i64) else { continue };
324                let mut cell = input_to_cell(v, sheet, c, &mut cx.wb).unwrap_or_else(|_| Some(Cell::value(Value::text(v.as_str()))));
325                if let Some(cl) = cell.as_mut() {
326                    numeric_from_html(cl, v, html.get(ri).and_then(|r| r.get(ci)), &mut cx.wb);
327                }
328                let sh = cx.sheet_mut(sheet)?;
329                match cell {
330                    Some(cell) => sh.set_cell(c, cell),
331                    None => {
332                        sh.remove_cell(c);
333                    }
334                }
335                cx.touch(sheet, c);
336            }
337        }
338        if h > 0 && w > 0 {
339            let end = at.offset_clamped(h as i64 - 1, w as i64 - 1);
340            *cx.sel = crate::engine::Selection { active: at, anchor: at, ranges: vec![RangeRef::new(at, end)] };
341        }
342        Ok(())
343    })?;
344    Ok(json!({"rows": h, "cols": w}))
345}

347/// A pasted cell that came out as text ("Rp10,000.00" isn't en-US input) becomes the number the
348/// source recorded for it in its HTML, with a format that shows the same text when one can be
349/// inferred and the cell has none. The HTML cell must show the same text as the tab-separated
350/// field, so HTML that belongs to another copy (or a table laid out differently) changes nothing.
351fn numeric_from_html(cell: &mut Cell, field: &str, html: Option<&crate::engine::clip_html::HtmlCell>, wb: &mut crate::model::Workbook) {
352    let Some(hc) = html else { return };
353    let Some(n) = hc.num else { return };
354    if cell.formula.is_some() || !matches!(cell.value, Value::Text(_)) || wb.styles.get(cell.style).num_fmt.as_str() == "@" {
355        return;
356    }
357    if crate::engine::clip_html::normalized(&hc.text) != crate::engine::clip_html::normalized(field) {
358        return;
359    }
360    cell.value = Value::number(n);
361    if wb.styles.get(cell.style).num_fmt.as_str() == "General"
362        && let Some(code) = crate::engine::clip_html::format_for(field, n)
363    {
364        cell.style = wb.styles.derive(cell.style, |s| s.num_fmt = crate::model::NumFmt::new(&code));
365    }
366}

368/// Tab-separated text with Excel-style quoting.
369pub(crate) fn parse_tsv(text: &str) -> Vec<Vec<String>> {
370    let mut rows = Vec::new();
371    let mut row = Vec::new();
372    let mut field = String::new();
373    let mut chars = text.chars().peekable();
374    // Text read again as unquoted, ahead of the rest of the input: a field that opened with a
375    // quote but turned out not to be quoted.
376    let mut replay: std::collections::VecDeque<char> = std::collections::VecDeque::new();
377    // Text is replayed at most as much as the input is long; past that, quotes are text, so
378    // hostile input full of fields that only look quoted reads in linear time.
379    let mut replay_budget = text.len();
380    let mut quoted = false;
381    let mut at_start = true;
382    // A quoted field's text as it stands in the input, for a field that turns out not to be one.
383    let mut raw = String::new();
384    loop {
385        let c = match replay.pop_front().or_else(|| chars.next()) {
386            Some(c) => c,
387            // A quote that never closes: the field is its text as written.
388            None if quoted => {
389                quoted = false;
390                field.clear();
391                field.push('"');
392                replay_budget = replay_budget.saturating_sub(raw.len());
393                replay.extend(raw.chars().skip(1));
394                raw.clear();
395                continue;
396            }
397            None => break,
398        };
399        if quoted {
400            raw.push(c);
401            if c == '"' {
402                let next = replay.front().copied().or_else(|| chars.peek().copied());
403                if next == Some('"') {
404                    field.push('"');
405                    raw.push('"');
406                    if replay.pop_front().is_none() {
407                        chars.next();
408                    }
409                } else {
410                    quoted = false;
411                    // A field is quoted only when its closing quote ends it, as Excel reads it:
412                    // `"hi" there` is that text, quotes included, and a tab or line break inside
413                    // it still ends the cell, as if the leading quote weren't there.
414                    if !matches!(next, None | Some('\t' | '\n' | '\r')) {
415                        field.clear();
416                        field.push('"');
417                        replay_budget = replay_budget.saturating_sub(raw.len());
418                        let rest: Vec<char> = std::mem::take(&mut raw).chars().skip(1).collect();
419                        for ch in rest.into_iter().rev() {
420                            replay.push_front(ch);
421                        }
422                    }
423                }
424            } else {
425                field.push(c);
426            }
427            continue;
428        }
429        match c {
430            '"' if at_start && replay_budget > 0 => {
431                quoted = true;
432                at_start = false;
433                raw = String::from('"');
434            }
435            '\t' => {
436                row.push(std::mem::take(&mut field));
437                at_start = true;
438            }
439            '\r' => {}
440            '\n' => {
441                row.push(std::mem::take(&mut field));
442                rows.push(std::mem::take(&mut row));
443                at_start = true;
444            }
445            _ => {
446                field.push(c);
447                at_start = false;
448            }
449        }
450    }
451    if !field.is_empty() || !row.is_empty() {
452        row.push(field);
453        rows.push(row);
454    }
455    rows
456}
