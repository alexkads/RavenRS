6fn matches(hay: &str, what: &str, case: bool, whole: bool) -> bool {
7    let (h, w) = if case { (hay.to_string(), what.to_string()) } else { (hay.to_lowercase(), what.to_lowercase()) };
8    let has_wild = w.contains(['*', '?']);
9    if has_wild {
10        let pat = if whole { w } else { format!("*{w}*") };
11        return wildcard(&h, &pat);
12    }
13    if whole { h == w } else { h.contains(&w) }
14}

16/// `*`/`?` wildcard match with `~` escapes.
17///
18/// The pattern runs as a bit-parallel automaton over the text: one pass, each character costing
19/// one step per 64 pattern positions. Backtracking to the last `*` after a mismatch rescans the
20/// text and costs text × pattern comparisons on a long, repetitive cell, for every cell a
21/// filter or Find visits.
22pub(crate) fn wildcard(text: &str, pat: &str) -> bool {
23    // The pattern as (character, escaped) items, runs of `*` folded into one.
24    let mut p: Vec<(char, bool)> = Vec::new();
25    let mut it = pat.chars();
26    while let Some(c) = it.next() {
27        if c == '~' {
28            if let Some(n) = it.next() {
29                p.push((n, true));
30            }
31        } else if c != '*' || p.last() != Some(&('*', false)) {
32            p.push((c, false));
33        }
34    }
35    let words = (p.len() + 1).div_ceil(64);
36    let bit = |set: &mut [u64], i: usize| set[i / 64] |= 1 << (i % 64);
37    // Positions holding a `*` and a `?`, and where each literal character sits in the pattern.
38    let mut stars = vec![0u64; words];
39    let mut any = vec![0u64; words];
40    let mut at: std::collections::HashMap<char, Vec<usize>> = std::collections::HashMap::new();
41    for (i, item) in p.iter().enumerate() {
42        match item {
43            ('*', false) => bit(&mut stars, i),
44            ('?', false) => bit(&mut any, i),
45            (c, _) => at.entry(*c).or_default().push(i),
46        }
47    }
48    // A literal's bit mask is built when the text first reads it, and kept for at most
49    // `MAX_CACHED_MASKS` literals: a long pattern of mostly distinct characters would otherwise
50    // allocate a mask of the pattern's width for each one up front. When there are more literals
51    // than that, the ones kept are those at the most pattern positions (the dearest to rebuild);
52    // each of the rest has at most pattern length / `MAX_CACHED_MASKS` positions, so rebuilding
53    // its mask in `scratch` costs less than one step of the scan.
54    const MAX_CACHED_MASKS: usize = 256;
55    let cacheable: Option<std::collections::HashSet<char>> = (at.len() > MAX_CACHED_MASKS).then(|| {
56        let mut by_count: Vec<(&char, usize)> = at.iter().map(|(c, v)| (c, v.len())).collect();
57        by_count.sort_unstable_by_key(|&(_, n)| std::cmp::Reverse(n));
58        by_count.into_iter().take(MAX_CACHED_MASKS).map(|(c, _)| *c).collect()
59    });
60    let mut masks: std::collections::HashMap<char, Vec<u64>> = std::collections::HashMap::new();
61    let mut scratch = vec![0u64; words];
62    // A set bit i: the text read so far matches the first i pattern items. A `*` matches nothing
63    // as well, so a state at a `*` is also at the item after it.
64    let close = |set: &mut Vec<u64>| {
65        let mut carry = 0u64;
66        for w in 0..words {
67            let s = set[w] & stars[w];
68            let next = (s << 1) | carry;
69            carry = s >> 63;
70            set[w] |= next;
71        }
72    };
73    let mut state = vec![0u64; words];
74    bit(&mut state, 0);
75    close(&mut state);
76    let mut next = vec![0u64; words];
77    for ch in text.chars() {
78        let lit: Option<&[u64]> = match at.get(&ch) {
79            None => None,
80            Some(positions) if cacheable.as_ref().is_none_or(|k| k.contains(&ch)) => {
81                let mask = masks.entry(ch).or_insert_with(|| {
82                    let mut m = vec![0u64; words];
83                    positions.iter().for_each(|&i| bit(&mut m, i));
84                    m
85                });
86                Some(mask)
87            }
88            Some(positions) => {
89                scratch.fill(0);
90                positions.iter().for_each(|&i| bit(&mut scratch, i));
91                Some(&scratch)
92            }
93        };
94        let mut carry = 0u64;
95        let mut alive = 0u64;
96        for w in 0..words {
97            // Items that read `ch` advance; a `*` reads it and stays.
98            let m = state[w] & (any[w] | lit.map_or(0, |l| l[w]));
99            next[w] = (m << 1) | carry | (state[w] & stars[w]);
100            carry = m >> 63;
101            alive |= next[w];
102        }
103        if alive == 0 {
104            return false;
105        }
106        std::mem::swap(&mut state, &mut next);
107        close(&mut state);
108    }
109    state[p.len() / 64] & (1 << (p.len() % 64)) != 0
110}

112pub(super) fn find(s: &mut Session, p: &Json) -> Result<Json> {
113    let what = str_param(p, "what").unwrap_or("").to_string();
114    if what.is_empty() {
115        return Err(bad("edit.find", "missing `what`"));
116    }
117    let case = bool_param(p, "matchCase").unwrap_or(false);
118    let whole = bool_param(p, "wholeCell").unwrap_or(false);
119    let values = str_param(p, "lookIn") == Some("values");
120    let by_cols = bool_param(p, "byColumns").unwrap_or(false);
121    let all = bool_param(p, "all").unwrap_or(false);
122    let workbook = str_param(p, "within") == Some("workbook");
123    // `selection`: only the selected cells of the active sheet, as Excel searches a selected range.
124    let in_selection = str_param(p, "within") == Some("selection");
125    s.last_find = Some(p.clone());
126    let d = s.doc()?;
127    let sheets: Vec<usize> = if workbook { (0..d.wb.sheets.len()).collect() } else { vec![d.wb.active_sheet()] };
128    let mut hits: Vec<(usize, CellRef, String)> = Vec::new();
129    for si in sheets {
130        let Some(sh) = d.wb.sheet(si) else { continue };
131        let mut cells: Vec<(CellRef, String)> = sh
132            .cells
133            .iter()
134            .filter(|(c, _)| !in_selection || d.selection.contains(*c))
135            .filter_map(|(c, cell)| {
136                let text = if values || cell.formula.is_none() || crate::engine::display::formula_hidden(&d.wb, sh, c) {
137                    crate::engine::display::cell_text(&d.wb, sh, c)
138                } else {
139                    cell.input_text()
140                };
141                matches(&text, &what, case, whole).then_some((c, text))
142            })
143            .collect();
144        if by_cols {
145            cells.sort_by_key(|(c, _)| (c.col, c.row));
146        }
147        hits.extend(cells.into_iter().map(|(c, t)| (si, c, t)));
148    }
149    if hits.is_empty() {
150        return Err(EngineError::Other(format!("We couldn't find what you were looking for: {what}")));
151    }
152    if all {
153        let list: Vec<Json> =
154            hits.iter().map(|(si, c, t)| json!({"sheet": d.wb.sheet(*si).map(|s| s.name.clone()), "cell": c.a1(), "value": t})).collect();
155        return Ok(json!({"count": list.len(), "results": list}));
156    }
157    // Next hit after the active cell.
158    let cur = (d.wb.active_sheet(), d.selection.active);
159    let key = |si: usize, c: CellRef| if by_cols { (si, c.col, c.row) } else { (si, c.row, c.col) };
160    let next = hits.iter().find(|(si, c, _)| key(*si, *c) > key(cur.0, cur.1)).or_else(|| hits.first()).cloned();
161    let Some((si, c, t)) = next else { return Err(EngineError::Other("not found".into())) };
162    let dd = s.doc_mut()?;
163    dd.wb_switch_sheet(si);
164    if in_selection {
165        // The range stays selected and the active cell moves through it, as in Excel.
166        dd.selection.active = c;
167    } else {
168        dd.selection = crate::engine::Selection::at(c);
169    }
170    Ok(json!({"cell": c.a1(), "sheet": si, "value": t, "count": hits.len()}))
171}

173/// Most characters a cell holds (Excel's limit).
174const REPLACE_CELL_MAX_CHARS: usize = 32_767;
175/// Longest text one replacement may give a cell, in bytes: the character limit, each up to four
176/// bytes. Every occurrence copies the replacement text, so a short cell full of matches and a
177/// long replacement would otherwise expand to gigabytes. The character limit is checked on what
178/// fits in this.
179const REPLACE_CELL_MAX_BYTES: usize = REPLACE_CELL_MAX_CHARS * 4;
180/// Most bytes of replaced text one Replace All builds across all its cells.
181const REPLACE_TOTAL_MAX_BYTES: usize = 256 * 1024 * 1024;
182
183pub(super) fn replace(s: &mut Session, p: &Json) -> Result<Json> {
184    let what = str_param(p, "what").unwrap_or("").to_string();
185    let with = str_param(p, "with").unwrap_or("");
186    if what.is_empty() {
187        return Err(bad("edit.replace", "missing `what`"));
188    }
189    let case = bool_param(p, "matchCase").unwrap_or(false);
190    let whole = bool_param(p, "wholeCell").unwrap_or(false);
191    let all = bool_param(p, "all").unwrap_or(true);
192    let workbook = str_param(p, "within") == Some("workbook");
193    let selection = (str_param(p, "within") == Some("selection")).then(|| s.doc().map(|d| d.selection.clone())).transpose()?;
194    let active = s.doc()?.selection.active;
195    let mut skipped = 0usize;
196    let n = edit(s, |cx| {
197        let sheets: Vec<usize> = if workbook { (0..cx.wb.sheets.len()).collect() } else { vec![cx.wb.active_sheet()] };
198        let mut n = 0;
199        let mut total = 0usize;
200        for si in sheets {
201            let cells: Vec<(CellRef, String)> =
202                // A hidden formula's source is not searched: a count or error would reveal its text.
203                cx.wb
204                    .sheet(si)
205                    .map(|sh| sh.cells.iter().filter(|(c, cell)| cell.formula.is_none() && cell.py.is_none() || !crate::engine::display::formula_hidden(&cx.wb, sh, *c)).map(|(c, _)| (c, sh.input_text(c))).collect())
206                    .unwrap_or_default();
207            for (c, text) in cells {
208                if (!all && c != active) || selection.as_ref().is_some_and(|sel| !sel.contains(c)) {
209                    continue;
210                }
211                if !matches(&text, &what, case, whole) {
212                    continue;
213                }
214                let new = if whole {
215                    (with.len() <= REPLACE_CELL_MAX_BYTES).then(|| with.to_string())
216                } else if case {
217                    replace_cs_within(&text, &what, with, REPLACE_CELL_MAX_BYTES)
218                } else {
219                    replace_ci_within(&text, &what, with, REPLACE_CELL_MAX_BYTES)
220                };
221                let new = new.filter(|n| n.chars().count() <= REPLACE_CELL_MAX_CHARS);
222                let Some(new) = new else {
223                    return Err(EngineError::Other(format!("The replacement makes the text of {} too long.", c.a1())));
224                };
225                total = total.saturating_add(new.len());
226                if total > REPLACE_TOTAL_MAX_BYTES {
227                    return Err(EngineError::Other("The replacements make too much text. Replace in a smaller range.".into()));
228                }
229                // A Python cell's text is its code: the replacement edits the code and the cell
230                // stays a Python cell. Code that would no longer fit in a PY formula is left as it was.
231                let old_py = cx.wb.sheet(si).and_then(|sh| sh.cell(c)).and_then(|x| x.py.clone());
232                if let Some(old) = old_py {
233                    let py = crate::model::PyCode { object: old.object, ..crate::model::PyCode::new(&new) };
234                    if py.code == old.code {
235                        continue;
236                    }
237                    if !py.fits() {
238                        skipped += 1;
239                        continue;
240                    }
241                    // Its old spill is left to the recalculation, which removes it and
242                    // recalculates what read it.
243                    let sh = cx.sheet_mut(si)?;
244                    let style = sh.style_id(c);
245                    sh.set_cell(
246                        c,
247                        Cell { value: Value::Error(crate::base::CellError::Blocked), formula: None, style, py: Some(std::sync::Arc::new(py)) },
248                    );
249                    cx.touch(si, c);
250                    n += 1;
251                    continue;
252                }
253                let cell = input_to_cell(&new, si, c, &mut cx.wb).unwrap_or_else(|_| Some(Cell::value(Value::text(new.as_str()))));
254                let sh = cx.sheet_mut(si)?;
255                match cell {
256                    Some(cell) => sh.set_cell(c, cell),
257                    None => {
258                        sh.remove_cell(c);
259                    }
260                }
261                cx.touch(si, c);
262                n += 1;
263            }
264        }
265        Ok(n)
266    })?;
267    if skipped > 0 {
268        return Ok(json!({"replaced": n, "skipped": skipped}));
269    }
270    Ok(json!({"replaced": n}))
271}

273/// Whether case-folding `s` maps every char to exactly one char of the same UTF-8 length, so
274/// byte positions in the folded text still line up with the original's character boundaries.
275fn folds_in_place(s: &str) -> bool {
276    s.chars().all(|c| {
277        let mut folded = c.to_lowercase();
278        let first = folded.next();
279        folded.next().is_none() && first.is_some_and(|l| l.len_utf8() == c.len_utf8())
280    })
281}

288/// `text` with every `what` replaced by `with`, or `None` once the result passes `limit` bytes
289/// (checked before each copy, so nothing larger is allocated).
290pub(super) fn replace_cs_within(text: &str, what: &str, with: &str, limit: usize) -> Option<String> {
291    if what.is_empty() {
292        return (text.len() <= limit).then(|| text.to_string());
293    }
294    let mut out = String::new();
295    let mut i = 0;
296    for (pos, m) in text.match_indices(what) {
297        if out.len() + (pos - i) + with.len() > limit {
298            return None;
299        }
300        out.push_str(text.get(i..pos).unwrap_or(""));
301        out.push_str(with);
302        i = pos + m.len();
303    }
304    let rest = text.get(i..).unwrap_or("");
305    if out.len() + rest.len() > limit {
306        return None;
307    }
308    out.push_str(rest);
309    Some(out)
310}

312/// Case-insensitive [`replace_cs_within`].
313pub(super) fn replace_ci_within(text: &str, what: &str, with: &str, limit: usize) -> Option<String> {
314    let lower = text.to_lowercase();
315    let w = what.to_lowercase();
316    // Slicing `text` at positions found in `lower` is only sound when folding kept every byte
317    // boundary: a length check alone misses foldings that keep the total length while shifting
318    // boundaries (U+212A KELVIN SIGN shrinks 3→1 bytes, U+0130 grows 2→3), where the tail used
319    // to be dropped. Those fall back to the case-sensitive replace, as length changes already do.
320    if lower.len() != text.len() || w.is_empty() || !folds_in_place(text) || !folds_in_place(what) {
321        return replace_cs_within(text, what, with, limit);
322    }
323    let mut out = String::new();
324    let mut i = 0;
325    while let Some(pos) = lower.get(i..).and_then(|s| s.find(&w)) {
326        if out.len() + pos + with.len() > limit {
327            return None;
328        }
329        out.push_str(text.get(i..i + pos).unwrap_or(""));
330        out.push_str(with);
331        i += pos + w.len();
332    }
333    let rest = text.get(i..).unwrap_or("");
334    if out.len() + rest.len() > limit {
335        return None;
336    }
337    out.push_str(rest);
338    Some(out)
