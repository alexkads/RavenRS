4/// How many references the Go To dialog keeps for going back to.
5pub(super) const GO_TO_RECENT: usize = 10;
6
7/// The most areas one Go To selects.
8pub(super) const GO_TO_MAX_AREAS: usize = 2000;
9
10/// How deep a name that refers to other names is followed.
11const GO_TO_NAME_DEPTH: usize = 8;
12
13pub(in super::super) fn go_to(s: &mut Session, p: &Json) -> Result<Json> {
14    if bool_param(p, "list") == Some(true) {
15        let d = s.doc()?;
16        return Ok(json!({"names": go_to_names(&d.wb, d.wb.active_sheet()), "recent": d.go_to_recent}));
17    }
18    let Some(reference) = str_param(p, "reference").or_else(|| str_param(p, "range")) else { return Err(bad("edit.goTo", "missing `reference`")) };
19    let d = s.doc()?;
20    let mut names = Resolver::new(&d.wb);
21    let mut target: Option<usize> = None;
22    let mut areas: Vec<RangeRef> = Vec::new();
23    let mut named = false;
24    for part in reference_parts(reference) {
25        let mut found = Vec::new();
26        let Some((sheet, _, by_name)) = names.target(d.wb.active_sheet(), part, 0, Some(&mut found)) else {
27            return Err(EngineError::Other("Reference isn't valid.".into()));
28        };
29        // One selection holds the areas of one sheet.
30        if target.is_some_and(|t| t != sheet) {
31            return Err(EngineError::Other("Reference isn't valid.".into()));
32        }
33        target = Some(sheet);
34        named |= by_name;
35        areas.extend(found);
36        if areas.len() > GO_TO_MAX_AREAS {
37            return Err(EngineError::Other(format!("Go To selects at most {GO_TO_MAX_AREAS} areas.")));
38        }
39    }
40    let Some(sheet) = target else { return Err(EngineError::Other("Reference isn't valid.".into())) };
41    let range = areas.iter().map(RangeRef::a1).collect::<Vec<_>>().join(",");
42    let out = selection_set(s, &json!({"sheet": sheet, "range": range}))?;
43    if bool_param(p, "record") == Some(true) && !named {
44        let d = s.doc_mut()?;
45        let prefix = crate::base::quote_sheet(&d.wb.sheet(sheet).map(|sh| sh.name.clone()).unwrap_or_default());
46        let entry = areas.iter().map(|r| format!("{prefix}!{}", r.a1())).collect::<Vec<_>>().join(",");
47        d.go_to_recent.retain(|r| !r.eq_ignore_ascii_case(&entry));
48        d.go_to_recent.insert(0, entry);
49        d.go_to_recent.truncate(GO_TO_RECENT);
50    }
51    Ok(out)
52}

54/// The parts of a Go To reference, split at the commas outside quoted sheet names and
55/// parentheses: `Price, Type` or `B14:C22,F19:G30`. The parts are read one at a time, so a caller
56/// that stops at an invalid part or at `GO_TO_MAX_AREAS` does not split the rest of the text.
57pub(super) fn reference_parts(text: &str) -> impl Iterator<Item = &str> {
58    let (mut depth, mut quoted, mut start, mut done) = (0usize, false, 0, false);
59    let mut chars = text.char_indices();
60    std::iter::from_fn(move || {
61        if done {
62            return None;
63        }
64        for (i, c) in chars.by_ref() {
65            match c {
66                '\'' => quoted = !quoted,
67                '(' if !quoted => depth += 1,
68                ')' if !quoted => depth = depth.saturating_sub(1),
69                ',' if !quoted && depth == 0 => {
70                    let part = text.get(start..i).unwrap_or("").trim();
71                    start = i + 1;
72                    return Some(part);
73                }
74                _ => {}
75            }
76        }
77        done = true;
78        Some(text.get(start..).unwrap_or("").trim())
79    })
80}

82/// Resolves Go To references against one workbook. Defined names are found through an index
83/// built once, and whether a name refers to a reference is remembered: listing the names of a
84/// workbook full of names that alias other names would otherwise scan every name for every alias
85/// of every name.
86struct Resolver<'a> {
87    wb: &'a crate::model::Workbook,
88    /// Indexes into `wb.names` by scope and folded name, in workbook order.
89    by_name: std::collections::HashMap<(Option<usize>, String), Vec<usize>>,
90    /// [`Resolver::name_areas`] without areas, by name index, sheet and depth: the target sheet
91    /// and the number of areas.
92    memo: std::collections::HashMap<(usize, usize, usize), Option<(usize, usize)>>,
93}
94
95impl<'a> Resolver<'a> {
96    fn new(wb: &'a crate::model::Workbook) -> Self {
97        let mut by_name: std::collections::HashMap<(Option<usize>, String), Vec<usize>> = std::collections::HashMap::new();
98        for (i, n) in wb.names.iter().enumerate() {
99            by_name.entry((n.scope, crate::formula::fold_name(&n.name))).or_default().push(i);
100        }
101        Self { wb, by_name, memo: std::collections::HashMap::new() }
102    }
103
104    /// The first name of scope `scope` that `same` accepts, as a scan of `wb.names` finds it.
105    /// Names `same` accepts fold alike, so only that bucket is checked.
106    fn find(&self, scope: Option<usize>, body: &str, same: impl Fn(&str, &str) -> bool) -> Option<usize> {
107        self.by_name
108            .get(&(scope, crate::formula::fold_name(body)))?
109            .iter()
110            .copied()
111            .find(|&i| self.wb.names.get(i).is_some_and(|n| same(&n.name, body)))
112    }
113
114    /// Where one part of a Go To reference points, seen from sheet `sheet`: the sheet, how many
115    /// areas (appended to `areas` when given), and whether a defined name or a table gave them.
116    /// `Sheet2!Name` is the name scoped to Sheet2. A name counts when what it refers to is a
117    /// reference; a constant or a formula is not one.
118    fn target(&mut self, sheet: usize, part: &str, depth: usize, areas: Option<&mut Vec<RangeRef>>) -> Option<(usize, usize, bool)> {
119        let wb = self.wb;
120        if part.is_empty() || depth > GO_TO_NAME_DEPTH {
121            return None;
122        }
123        let (prefix, body) = split_sheet(part);
124        let on = match &prefix {
125            Some(n) => wb.sheet_index(n)?,
126            None => sheet,
127        };
128        if let Some(r) = RangeRef::parse(&body.replace('$', "")) {
129            if let Some(a) = areas {
130                a.push(r);
131            }
132            return Some((on, 1, false));
133        }
134        // As `Workbook::name`: the sheet's own name, else the workbook's.
135        let name = match prefix {
136            Some(_) => self.find(Some(on), body, crate::formula::same_name),
137            None => self.find(Some(sheet), body, str::eq_ignore_ascii_case).or_else(|| self.find(None, body, str::eq_ignore_ascii_case)),
138        };
139        if let Some(i) = name {
140            return self.name_areas(i, sheet, depth, areas).map(|(t, n)| (t, n, true));
141        }
142        if prefix.is_none()
143            && let Some((si, ti)) = wb.table(body)
144        {
145            // As in Excel, a table's name stands for its data, without the header and totals rows.
146            let t = wb.sheet(si)?.tables.get(ti)?;
147            if let Some(a) = areas {
148                a.push(t.data_range().unwrap_or(t.range));
149            }
150            return Some((si, 1, true));
151        }
152        None
153    }
154
155    /// The sheet and number of areas defined name `wb.names[i]` refers to, seen from sheet
156    /// `sheet`, when that is a reference; the areas are appended to `areas` when given.
157    fn name_areas(&mut self, i: usize, sheet: usize, depth: usize, mut areas: Option<&mut Vec<RangeRef>>) -> Option<(usize, usize)> {
158        let key = (i, sheet, depth);
159        if areas.is_none()
160            && let Some(known) = self.memo.get(&key)
161        {
162            return *known;
163        }
164        let out = self.name_areas_uncached(i, sheet, depth, areas.as_deref_mut());
165        if areas.is_none() {
166            self.memo.insert(key, out);
167        }
168        out
169    }
170
171    fn name_areas_uncached(&mut self, i: usize, sheet: usize, depth: usize, mut areas: Option<&mut Vec<RangeRef>>) -> Option<(usize, usize)> {
172        let n = self.wb.names.get(i)?;
173        let from = n.scope.unwrap_or(sheet);
174        let formula = n.formula.trim().trim_start_matches('=');
175        let mut target: Option<usize> = None;
176        let mut count = 0usize;
177        for p in reference_parts(formula) {
178            let (s, found, _) = self.target(from, p, depth + 1, areas.as_deref_mut())?;
179            if target.is_some_and(|t| t != s) {
180                return None;
181            }
182            target = Some(s);
183            count += found;
184            if count > GO_TO_MAX_AREAS {
185                return None;
186            }
187        }
188        target.map(|t| (t, count))
189    }
190}

192/// The names the Go To dialog lists, seen from sheet `sheet`: the named ranges of the workbook and
193/// of every sheet, in alphabetical order. A name scoped to another sheet is written `Sheet2!Name`,
194/// and the active sheet's own name hides the workbook name it shares. Hidden names and names that
195/// do not refer to a range are left out.
196pub fn go_to_names(wb: &crate::model::Workbook, sheet: usize) -> Vec<String> {
197    let own: std::collections::HashSet<String> =
198        wb.names.iter().filter(|n| n.scope == Some(sheet)).map(|n| crate::formula::fold_name(&n.name)).collect();
199    let mut names = Resolver::new(wb);
200    let mut out: Vec<String> = wb
201        .names
202        .iter()
203        .enumerate()
204        .filter(|(i, n)| !n.hidden && names.name_areas(*i, sheet, 0, None).is_some())
205        .filter_map(|(_, n)| match n.scope {
206            None if own.contains(&crate::formula::fold_name(&n.name)) => None,
207            None => Some(n.name.clone()),
208            Some(s) if s == sheet => Some(n.name.clone()),
209            Some(s) => Some(format!("{}!{}", crate::base::quote_sheet(&wb.sheet(s)?.name), n.name)),
210        })
211        .collect();
212    out.sort_by_cached_key(|n| crate::formula::fold_name(n));
213    out.dedup_by(|a, b| crate::formula::same_name(a, b));
214    out
215}

217pub(super) fn go_to_special(s: &mut Session, p: &Json) -> Result<Json> {
218    let kind = str_param(p, "kind").unwrap_or("blanks");
219    let d = s.doc()?;
220    let Some(sheet) = d.wb.active() else { return Err(EngineError::NoDocument) };
221    let scope = if d.selection.is_single_cell() { sheet.used_range().unwrap_or(RangeRef::cell(d.selection.active)) } else { d.selection.bounds() };
222    if kind == "lastCell" {
223        let c = sheet.used_range().map(|r| r.end).unwrap_or_default();
224        return selection_set(s, &json!({"cell": c.a1()}));
225    }
226    if kind == "currentRegion" {
227        return select_region(s, &json!({}));
228    }
229    let mut hits: Vec<CellRef> = Vec::new();
230    if scope.count() > 4_000_000 {
231        return Err(bad("edit.goToSpecial", "range too large"));
232    }
233    for c in scope.iter() {
234        let cell = sheet.cell(c);
235        let v = sheet.value(c);
236        let ok = match kind {
237            "blanks" => v.is_empty(),
238            "constants" => cell.is_some_and(|x| x.formula.is_none() && x.py.is_none() && !x.value.is_empty()),
239            "formulas" => cell.is_some_and(|x| x.formula.is_some() || x.py.is_some()),
240            "comments" | "notes" => sheet.comments.contains_key(&c),
241            "errors" => v.is_error(),
242            "numbers" => v.is_number(),
243            "text" => v.is_text(),
244            "visible" => !sheet.is_row_hidden(c.row) && !sheet.is_col_hidden(c.col),
245            "conditionalFormats" => sheet.cond_formats.iter().any(|cf| cf.ranges.iter().any(|r| r.contains(c))),
246            "dataValidation" => sheet.validations.iter().any(|dv| dv.ranges.iter().any(|r| r.contains(c))),
247            _ => return Err(bad("edit.goToSpecial", format!("unknown kind `{kind}`"))),
248        };
249        if ok {
250            hits.push(c);
251        }
252    }
253    if hits.is_empty() {
254        return Err(EngineError::Other("No cells were found.".into()));
255    }
256    let ranges = coalesce(&hits);
257    let d = s.doc_mut()?;
258    d.selection = crate::engine::Selection { active: hits[0], anchor: hits[0], ranges };
259    Ok(json!({"selection": d.selection.a1(), "count": hits.len()}))
260}
