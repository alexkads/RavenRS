7/// The sheet and the row (or column) spans a size or visibility command works on. The lines are
8/// `rows`/`cols` (`"2:5"`, `"B:B,D:F"`, optionally `Sheet!`-prefixed), else the selection's; the
9/// sheet is `sheet`, else the prefix, else the active one. The selection belongs to the active
10/// sheet, so another sheet needs explicit lines.
11fn line_targets(s: &Session, p: &Json, rows: bool) -> Result<(usize, Vec<(u32, u32)>)> {
12    let (sheet, lines) = resolve_lines(s, p, rows)?;
13    // Sizes and visibility are formatting of the line, which a protected sheet must allow.
14    if rows {
15        crate::engine::cmd::refuse_unless_allowed(s, sheet, "Formatting rows", |p| p.format_rows)?;
16    } else {
17        crate::engine::cmd::refuse_unless_allowed(s, sheet, "Formatting columns", |p| p.format_columns)?;
18    }
19    Ok((sheet, lines))
20}

22fn resolve_lines(s: &Session, p: &Json, rows: bool) -> Result<(usize, Vec<(u32, u32)>)> {
23    let key = if rows { "rows" } else { "cols" };
24    let mut sheet = target_sheet(s, p)?;
25    if let Some(t) = str_param(p, key) {
26        let (prefix, body) = split_sheet(t);
27        if let Some(name) = prefix {
28            let named = s.doc()?.wb.sheet_index(&name).ok_or_else(|| bad(key, format!("no sheet named `{name}`")))?;
29            if p.get("sheet").is_some_and(|v| !v.is_null()) && named != sheet {
30                return Err(bad(key, format!("`{t}` names a sheet other than `sheet`")));
31            }
32            sheet = named;
33        }
34        // A comma-separated list ("B:B,D:F") targets several areas, like a multi-area selection.
35        let lines = body
36            .split(',')
37            .map(|part| {
38                let r = parse_lines(part).ok_or_else(|| bad(key, format!("not a range: {part}")))?;
39                Ok(if rows { (r.start.row, r.end.row) } else { (r.start.col, r.end.col) })
40            })
41            .collect::<Result<Vec<_>>>()?;
42        return Ok((sheet, lines));
43    }
44    let d = s.doc()?;
45    if sheet != d.wb.active_sheet() {
46        return Err(bad(key, format!("missing `{key}`: the selection is on the active sheet")));
47    }
48    Ok((sheet, d.selection.ranges.iter().map(|r| if rows { (r.start.row, r.end.row) } else { (r.start.col, r.end.col) }).collect()))
49}

51/// Sizes are pixels at 96 dpi, the model's unit (header drags send pixels). `points` (row height)
52/// and `chars` (column width) take Excel's units, those of its Row Height and Column Width dialogs
53/// and of the file (`ht` is in points).
54pub(super) fn row_height(s: &mut Session, p: &Json) -> Result<Json> {
55    let h = match (f64_param(p, "height"), f64_param(p, "points")) {
56        (Some(h), _) => {
57            if !(0.0..=546.0).contains(&h) {
58                return Err(EngineError::Other("Row height must be between 0 and 546 pixels (409.5 points).".into()));
59            }
60            h
61        }
62        (None, Some(pt)) => {
63            if !(0.0..=409.5).contains(&pt) {
64                return Err(EngineError::Other("Row height must be between 0 and 409.5 points.".into()));
65            }
66            pt * 4.0 / 3.0
67        }
68        _ => return Err(bad("home.rowHeight", "missing `height` (pixels) or `points`")),
69    };
70    let (si, lines) = line_targets(s, p, true)?;
71    edit(s, |cx| {
72        let sh = cx.sheet_mut(si)?;
73        for (a, b) in lines {
74            for r in a..=b.min(a.saturating_add(1_048_575)) {
75                let e = sh.rows.entry(r).or_default();
76                e.size = Some(h as f32);
77                e.hidden = h == 0.0;
78                e.custom = true;
79            }
80        }
81        cx.structural = true; // setting a height shows or hides rows: SUBTOTAL(10x) depends on it
82        Ok(Json::Null)
83    })
84}

86pub(super) fn col_width(s: &mut Session, p: &Json) -> Result<Json> {
87    let w = match (f64_param(p, "width"), f64_param(p, "chars")) {
88        (Some(w), _) => {
89            if !(0.0..=1800.0).contains(&w) {
90                return Err(EngineError::Other("Column width must be between 0 and 1800 pixels.".into()));
91            }
92            w
93        }
94        (None, Some(c)) => {
95            if !(0.0..=255.0).contains(&c) {
96                return Err(EngineError::Other("Column width must be between 0 and 255 characters.".into()));
97            }
98            chars_to_points(c)
99        }
100        _ => return Err(bad("home.columnWidth", "missing `width` (pixels) or `chars`")),
101    };
102    let (si, lines) = line_targets(s, p, false)?;
103    edit(s, |cx| {
104        let sh = cx.sheet_mut(si)?;
105        for (a, b) in lines {
106            for c in a..=b {
107                let e = sh.cols.entry(c).or_default();
108                e.size = Some(w as f32);
109                e.hidden = w == 0.0;
110            }
111        }
112        Ok(Json::Null)
113    })
114}

116/// Excel column width in characters (of the default font's digit width, 7 px) → pixels.
117pub fn chars_to_points(c: f64) -> f64 {
118    if c <= 0.0 {
119        0.0
120    } else if c < 1.0 {
121        (c * 12.0).round()
122    } else {
123        (c * 7.0 + 5.0).round()
124    }
125}

127/// Approximate text width in points for autofit (no font metrics in the engine: average glyph
128/// widths for the default font; a front end with fonts measures instead, see `Session::text_width`).
129pub fn approx_text_width(text: &str, size: f32, bold: bool) -> f32 {
130    let mut w = 0.0;
131    for c in text.chars() {
132        w += match c {
133            'i' | 'l' | 'j' | '\'' | '.' | ',' | ':' | ';' | '!' | '|' => 0.28,
134            'f' | 't' | 'r' | 'I' | ' ' | '(' | ')' | '-' => 0.38,
135            'm' | 'w' | 'M' | 'W' => 0.85,
136            c if c.is_ascii_uppercase() => 0.65,
137            c if c.is_ascii_digit() => 0.56,
138            c if c.is_ascii() => 0.52,
139            c if is_zero_width(c) => 0.0,
140            c => arabic_em(c).unwrap_or(1.0),
141        };
142    }
143    w * size * if bold { 1.07 } else { 1.0 } * 96.0 / 72.0
144}

146/// Arabic vowel and Quranic marks, which sit on the previous letter, and invisible format
147/// characters such as direction marks and zero-width joiners.
148fn is_zero_width(c: char) -> bool {
149    matches!(
150        c,
151        '\u{0610}'..='\u{061A}'
152            | '\u{061C}'
153            | '\u{064B}'..='\u{065F}'
154            | '\u{0670}'
155            | '\u{06D6}'..='\u{06DC}'
156            | '\u{06DF}'..='\u{06E4}'
157            | '\u{06E7}'..='\u{06E8}'
158            | '\u{06EA}'..='\u{06ED}'
159            | '\u{0897}'..='\u{089F}'
160            | '\u{08CA}'..='\u{08E1}'
161            | '\u{08E3}'..='\u{08FF}'
162            | '\u{200B}'..='\u{200F}'
163            | '\u{202A}'..='\u{202E}'
164            | '\u{2060}'..='\u{2069}'
165            | '\u{FE20}'..='\u{FE2F}'
166            | '\u{FEFF}'
167    )
168}

170/// Width in ems of an Arabic-script character, measured from shaped Noto Sans Arabic, Noto Naskh
171/// Arabic and DejaVu Sans: joined letters average about half an em, word ligatures such as ﷺ and the
172/// rial sign about 1.2 em, the honorific ligatures (U+FD40–FD4B) about 1.6 em, and the bismillah
173/// ligature about 11 em. The later honorific ligatures (U+FBC3–FBD2, U+FD4C–FD4F, U+FD90–FD91,
174/// U+FDC8–FDCF) are not measured and get the 1.6 em of their siblings rather than the letter width.
175/// Arabic-Indic digits are as wide as ASCII digits (0.53–0.57 em in Noto Sans Arabic, Arial, Tahoma
176/// and Times New Roman); Arabic punctuation (0.2–0.5 em in the same fonts) keeps the letter width.
177fn arabic_em(c: char) -> Option<f32> {
178    match c {
179        '\u{FDFD}' => Some(11.0),
180        '\u{FBC3}'..='\u{FBD2}' | '\u{FD40}'..='\u{FD4F}' | '\u{FD90}'..='\u{FD91}' | '\u{FDC8}'..='\u{FDCF}' => Some(1.6),
181        '\u{FDF0}'..='\u{FDFC}' | '\u{FDFE}'..='\u{FDFF}' => Some(1.2),
182        '\u{0660}'..='\u{0669}' | '\u{06F0}'..='\u{06F9}' => Some(0.56),
183        '\u{0600}'..='\u{06FF}' | '\u{0750}'..='\u{077F}' | '\u{0870}'..='\u{08FF}' | '\u{FB50}'..='\u{FDFF}' | '\u{FE70}'..='\u{FEFC}' => Some(0.48),
184        _ => None,
185    }
186}

188/// Distinct texts AutoFit measures exactly; past this many, `Session::text_width` is asked for a
189/// quick measure so a huge column doesn't lay out every value.
190const EXACT_TEXT_WIDTHS: usize = 2_000;
191
192pub(super) fn autofit_cols(s: &mut Session, p: &Json) -> Result<Json> {
193    let (si, lines) = line_targets(s, p, false)?;
194    let measure = s.text_width.clone();
195    edit(s, |cx| {
196        let Some(sh) = cx.wb.sheet(si) else { return Ok(Json::Null) };
197        let mut measured: std::collections::HashMap<StyleId, std::collections::HashMap<String, f32>> = Default::default();
198        let mut distinct = 0usize;
199        // Selected columns, then one pass over the populated cells (the store is row-oriented, so a
200        // scan per column would revisit every row for each selected column, empty or not).
201        let cols: std::collections::BTreeSet<u32> = lines.iter().flat_map(|&(a, b)| a..=b.min(a + 16383)).collect();
202        let mut maxes: std::collections::BTreeMap<u32, f32> = Default::default();
203        if let Some(u) = sh.used_range() {
204            // One pass per run of adjacent selected columns, so columns between distant selections
205            // (A and XFD) are never visited.
206            let mut runs: Vec<(u32, u32)> = Vec::new();
207            for &c in &cols {
208                match runs.last_mut() {
209                    Some((_, end)) if *end + 1 == c => *end = c,
210                    _ => runs.push((c, c)),
211                }
212            }
213            // Every run is a pass over all the stored rows: many runs (alternating columns) are one
214            // pass over their whole span instead, the unselected columns skipped.
215            if runs.len() > 4
216                && let (Some(&(first, _)), Some(&(_, last))) = (runs.first(), runs.last())
217            {
218                runs = vec![(first, last)];
219            }
220            for (first, last) in runs {
221                let span = RangeRef::new(CellRef::new(u.start.row, first), CellRef::new(u.end.row, last));
222                for (cell, _) in sh.cells.iter_range(span) {
223                    if !cols.contains(&cell.col) {
224                        continue;
225                    }
226                    if sh.merge_at(cell).is_some_and(|m| !m.is_single()) {
227                        continue;
228                    }
229                    let id = sh.style_id(cell);
230                    let st = cx.wb.styles.get(id);
231                    let text = crate::engine::display::cell_text(&cx.wb, sh, cell);
232                    for line in text.lines() {
233                        let w = match &measure {
234                            Some(m) => {
235                                let known = measured.entry(id).or_default();
236                                match known.get(line) {
237                                    Some(w) => *w,
238                                    None => {
239                                        let w = m(line, &st.font, distinct < EXACT_TEXT_WIDTHS);
240                                        known.insert(line.to_owned(), w);
241                                        distinct += 1;
242                                        w
243                                    }
244                                }
245                            }
246                            None => approx_text_width(line, st.font.size, st.font.bold),
247                        };
248                        let w = w + 6.0 + st.align.indent as f32 * 9.0 + sh.filter_button_space(cell);
249                        let max = maxes.entry(cell.col).or_insert(0.0);
250                        *max = max.max(w);
251                    }
252                }
253            }
254        }
255        let widths: Vec<(u32, f32)> = maxes.into_iter().filter(|&(_, m)| m > 0.0).map(|(c, m)| (c, m.clamp(8.0, 1785.0))).collect();
256        let sh = cx.sheet_mut(si)?;
257        for (c, w) in widths {
258            let e = sh.cols.entry(c).or_default();
259            e.size = Some(w.ceil());
260            e.hidden = false;
261        }
262        Ok(Json::Null)
263    })
264}

266pub(super) fn autofit_rows(s: &mut Session, p: &Json) -> Result<Json> {
267    let (si, lines) = line_targets(s, p, true)?;
268    edit(s, |cx| {
269        for (a, b) in lines {
270            let b = b.min(a + 100_000);
271            if let Some(sh) = cx.wb.sheet_mut(si) {
272                for r in a..=b {
273                    if let Some(e) = sh.rows.get_mut(&r) {
274                        e.custom = false;
275                        e.hidden = false;
276                    }
277                }
278            }
279            cx.fit_rows.extend((a..=b).map(|r| (si, r)));
280        }
281        cx.structural = true; // autofit reveals hidden rows: SUBTOTAL(10x) depends on it
282        Ok(Json::Null)
283    })
284}

286pub(super) fn default_width(s: &mut Session, p: &Json) -> Result<Json> {
287    let w = f64_param(p, "width").ok_or_else(|| bad("home.defaultWidth", "missing `width`"))?;
288    let si = target_sheet(s, p)?;
289    edit(s, |cx| {
290        cx.sheet_mut(si)?.default_col_width = w.clamp(0.0, 1800.0) as f32;
291        Ok(Json::Null)
292    })

295pub(super) fn hide(s: &mut Session, p: &Json, rows: bool, on: bool) -> Result<Json> {
296    let (si, lines) = line_targets(s, p, rows)?;
297    edit(s, |cx| {
298        let sh = cx.sheet_mut(si)?;
299        for (a, b) in lines {
300            // Unhiding a single selected line also reveals hidden neighbours (Excel needs the
301            // lines on both sides selected; we're forgiving).
302            let (a, b) = if !on && a == b { (a.saturating_sub(1), b + 1) } else { (a, b) };
303            for i in a..=b.min(a.saturating_add(1_048_575)) {
304                let map = if rows { &mut sh.rows } else { &mut sh.cols };
305                if on {
306                    map.entry(i).or_default().hidden = true;
307                } else if let Some(e) = map.get_mut(&i) {
308                    e.hidden = false;
309                    if e.size == Some(0.0) {
310                        e.size = None;
311                    }
312                }
313            }
314        }
315        cx.structural = true; // SUBTOTAL(10x) depends on hidden rows
316        Ok(Json::Null)
317    })
