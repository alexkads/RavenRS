229fn with(p: &Json, k: &str, v: Json) -> Json {
230    let mut m = p.as_object().cloned().unwrap_or_default();
231    m.insert(k.into(), v);
232    Json::Object(m)
233}

241/// (sheet, index) of the pivot a call targets.
242fn find_pivot(s: &Session, p: &Json) -> Result<(usize, usize)> {
243    let d = s.doc()?;
244    let active = d.wb.active_sheet();
245    if let Some(v) = p.get("pivot").filter(|v| !v.is_null()) {
246        let mut order: Vec<usize> = vec![active];
247        order.extend((0..d.wb.sheets.len()).filter(|i| *i != active));
248        for si in order {
249            let Some(sh) = d.wb.sheet(si) else { continue };
250            let hit = sh.pivots.iter().position(|pt| match v {
251                Json::String(n) => pt.name.eq_ignore_ascii_case(n),
252                Json::Number(n) => n.as_u64() == Some(pt.id as u64),
253                _ => false,
254            });
255            if let Some(i) = hit {
256                return Ok((si, i));
257            }
258        }
259        return Err(bad("pivot", format!("no PivotTable `{}`", v.as_str().map(str::to_string).unwrap_or_else(|| v.to_string()))));
260    }
261    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
262    let at = d.selection.active;
263    if let Some(i) = sh.pivots.iter().position(|pt| pt.last_range.is_some_and(|r| r.contains(at))) {
264        return Ok((active, i));
265    }
266    if !sh.pivots.is_empty() {
267        return Ok((active, sh.pivots.len() - 1));
268    }
269    Err(EngineError::Disabled("pivot".into(), "select a cell in a PivotTable (or pass `pivot`)".into()))
270}
