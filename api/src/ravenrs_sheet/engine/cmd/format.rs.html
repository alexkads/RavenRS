218/// Accounting Number Format with the currency symbol to use (Excel's own button takes the system
219/// currency; `symbol` is how the control channel picks another one, e.g. `"L"` for lempiras).
220fn accounting(s: &mut Session, p: &Json) -> Result<Json> {
221    let symbol = str_param(p, "symbol").unwrap_or("$");
222    // The symbol goes into a quoted literal: a quote or a backslash would end it, and a control
223    // character would reach the format's text as-is.
224    if symbol.contains(['"', '\\']) || symbol.chars().any(char::is_control) {
225        return Err(bad("home.accounting", "the symbol can't contain a quote, a backslash or a control character"));
226    }
227    set_fmt(s, p, &accounting_format(symbol))
228}

230fn accounting_format(symbol: &str) -> String {
231    let symbol = format!("\"{symbol}\"");
232    format!("_({symbol}* #,##0.00_);_({symbol}* \\(#,##0.00\\);_({symbol}* \"-\"??_);_(@_)")
233}

236    let mut m = p.as_object().cloned().unwrap_or_default();

241/// Applies `f` to the style of every targeted cell (whole rows/columns set the line format too).
242pub(crate) fn apply_style(s: &mut Session, p: &Json, f: impl Fn(&mut Style)) -> Result<Json> {
243    let sheet = target_sheet(s, p)?;
244    let ranges = target_ranges(s, p)?;
245    let guarded = s.doc()?.wb.sheet(sheet).is_some_and(|sh| sh.protection.is_some());
246    if s.doc()?.wb.sheet(sheet).is_some_and(|sh| sh.protection.as_ref().is_some_and(|pr| !pr.format_cells)) {
247        return Err(EngineError::Other("The cell or chart you're trying to change is on a protected sheet.".into()));
248    }
249    edit(s, |cx| {
250        // Formatting on a protected sheet keeps the cell's permission flags: a format call
251        // that flips `locked` from the format permission would authorize content edits that
252        // the password should still gate, so those flags may not change (Excel keeps that
253        // checkbox disabled while the sheet is protected).
254        const FLAG_GUARD: &str = "You can't change cell protection on a protected sheet. Unprotect it first.";
255        macro_rules! guard_flags {
256            ($old:expr, $new:expr) => {
257                if guarded && cx.wb.styles.get($old).protection != cx.wb.styles.get($new).protection {
258                    return Err(EngineError::Other(FLAG_GUARD.into()));
259                }
260            };
261        }
262        let mut cache: std::collections::HashMap<StyleId, StyleId> = std::collections::HashMap::new();
263        for r in &ranges {
264            let full_rows = r.is_full_rows();
265            let full_cols = r.is_full_cols();
266            if full_rows || full_cols {
267                // Line formats.
268                let lines: Vec<u32> = if full_rows && !full_cols { (r.start.row..=r.end.row).collect() } else { (r.start.col..=r.end.col).collect() };
269                for line in lines {
270                    let cur = {
271                        let sh = cx.wb.sheet(sheet);
272                        let info = if full_rows && !full_cols { sh.and_then(|s| s.rows.get(&line)) } else { sh.and_then(|s| s.cols.get(&line)) };
273                        info.and_then(|i| i.style).unwrap_or_default()
274                    };
275                    let new = *cache.entry(cur).or_insert_with(|| cx.wb.styles.derive(cur, &f));
276                    guard_flags!(cur, new);
277                    let sh = cx.sheet_mut(sheet)?;
278                    let map = if full_rows && !full_cols { &mut sh.rows } else { &mut sh.cols };
279                    map.entry(line).or_default().style = Some(new);
280                }
281                // Existing cells in those lines.
282                let cells: Vec<(CellRef, StyleId)> =
283                    cx.wb.sheet(sheet).map(|sh| sh.cells.iter_range(*r).map(|(c, x)| (c, x.style)).collect()).unwrap_or_default();
284                for (c, st) in &cells {
285                    let new = *cache.entry(*st).or_insert_with(|| cx.wb.styles.derive(*st, &f));
286                    guard_flags!(*st, new);
287                    cx.sheet_mut(sheet)?.set_style(*c, new);
288                }
289                // Formatting touched these cells: re-fit their rows (Excel auto-fits).
290                cx.fit_rows.extend(cells.iter().map(|(c, _)| (sheet, c.row)));
291                continue;
292            }
293            if r.count() > 4_000_000 {
294                return Err(bad("format", "selection too large to format cell by cell"));
295            }
296            if r.height() <= 100_000 {
297                cx.fit_rows.extend((r.start.row..=r.end.row).map(|row| (sheet, row)));
298            }
299            for c in r.iter() {
300                let st = cx.wb.sheet(sheet).map(|sh| sh.style_id(c)).unwrap_or_default();
301                let new = *cache.entry(st).or_insert_with(|| cx.wb.styles.derive(st, &f));
302                if new != st {
303                    guard_flags!(st, new);
304                    cx.sheet_mut(sheet)?.set_style(c, new);
305                }
306            }
307        }
308        Ok(Json::Null)
309    })
310}

312/// The style of the active cell (what toggles read).
313pub(crate) fn active_style(s: &Session) -> Result<Style> {
314    let d = s.doc()?;
315    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
316    Ok(d.wb.styles.get(sh.style_id(d.selection.active)).clone())
317}

319fn toggle(s: &mut Session, p: &Json, get: fn(&Style) -> bool, set: fn(&mut Style, bool)) -> Result<Json> {
320    let on = match bool_param(p, "on") {
321        Some(v) => v,
322        None => !get(&active_style(s)?),
323    };
324    apply_style(s, p, |st| set(st, on))?;
325    Ok(json!({"on": on}))
326}

328fn underline(s: &mut Session, p: &Json) -> Result<Json> {
329    let style = match str_param(p, "style").unwrap_or("single") {
330        "double" => Underline::Double,
331        "singleAccounting" => Underline::SingleAccounting,
332        "doubleAccounting" => Underline::DoubleAccounting,
333        "none" => Underline::None,
334        _ => Underline::Single,
335    };
336    let cur = active_style(s)?.font.underline;
337    let on = bool_param(p, "on").unwrap_or(cur != style);
338    let u = if on { style } else { Underline::None };
339    apply_style(s, p, |st| st.font.underline = u)?;
340    Ok(json!({"on": on}))
341}

352fn font_size(s: &mut Session, p: &Json) -> Result<Json> {
353    let size = f64_param(p, "size").ok_or_else(|| bad("home.fontSize", "missing `size`"))?;
354    if !(1.0..=409.0).contains(&size) {
355        return Err(EngineError::Other("The number must be between 1 and 409.".into()));
356    }
357    let size = ((size * 2.0).round() / 2.0) as f32;
358    apply_style(s, p, |st| st.font.size = size)
359}

361const SIZES: [f32; 16] = [8.0, 9.0, 10.0, 11.0, 12.0, 14.0, 16.0, 18.0, 20.0, 24.0, 28.0, 36.0, 48.0, 72.0, 96.0, 409.0];

363fn step_font(s: &mut Session, p: &Json, up: bool) -> Result<Json> {
364    let cur = active_style(s)?.font.size;
365    let next =
366        if up { SIZES.iter().copied().find(|x| *x > cur).unwrap_or(409.0) } else { SIZES.iter().rev().copied().find(|x| *x < cur).unwrap_or(1.0) };
367    apply_style(s, p, |st| st.font.size = next)?;
368    Ok(json!({"size": next}))
369}

371pub(crate) fn color_param(v: Option<&Json>) -> Option<Color> {
372    match v? {
373        Json::String(s) if s.eq_ignore_ascii_case("auto") || s.eq_ignore_ascii_case("automatic") || s.eq_ignore_ascii_case("none") => {
374            Some(Color::Auto)
375        }
376        Json::String(s) => Color::from_hex(s),
377        Json::Object(o) => {
378            let theme = o.get("theme")?.as_u64()? as u8;
379            let tint = o.get("tint").and_then(Json::as_f64).unwrap_or(0.0);
380            Some(Color::Theme(theme.min(11), (tint.clamp(-1.0, 1.0) * 1000.0) as i16))
381        }
382        _ => None,
383    }
384}

414fn borders(s: &mut Session, p: &Json) -> Result<Json> {
415    let preset = str_param(p, "preset").unwrap_or("bottom").to_string();
416    let color = color_param(p.get("color")).unwrap_or(Color::Auto);
417    let base = border_style(str_param(p, "style").unwrap_or("thin"));
418    let sheet = target_sheet(s, p)?;
419    let ranges = target_ranges(s, p)?;
420    if s.doc()?.wb.sheet(sheet).is_some_and(|sh| sh.protection.as_ref().is_some_and(|pr| !pr.format_cells)) {
421        return Err(EngineError::Other("The cell or chart you're trying to change is on a protected sheet.".into()));
422    }
423    edit(s, |cx| {
424        for r in &ranges {
425            if r.count() > 2_000_000 {
426                return Err(bad("home.borders", "selection too large"));
427            }
428            for c in r.iter() {
429                let (top, bottom, left, right) = (c.row == r.start.row, c.row == r.end.row, c.col == r.start.col, c.col == r.end.col);
430                let st = cx.wb.sheet(sheet).map(|sh| sh.style_id(c)).unwrap_or_default();
431                let line = |style: BorderStyle| BorderLine { style, color };
432                let new = cx.wb.styles.derive(st, |s| {
433                    let b = &mut s.borders;
434                    match preset.as_str() {
435                        "none" => *b = Borders::default(),
436                        "all" => {
437                            b.top = line(base);
438                            b.bottom = line(base);
439                            b.left = line(base);
440                            b.right = line(base);
441                        }
442                        "outside" | "thickOutside" => {
443                            let st = if preset == "thickOutside" { BorderStyle::Thick } else { base };
444                            if top {
445                                b.top = line(st);
446                            }
447                            if bottom {
448                                b.bottom = line(st);
449                            }
450                            if left {
451                                b.left = line(st);
452                            }
453                            if right {
454                                b.right = line(st);
455                            }
456                        }
457                        "inside" | "insideHorizontal" | "insideVertical" => {
458                            if preset != "insideVertical" {
459                                if !top {
460                                    b.top = line(base);
461                                }
462                                if !bottom {
463                                    b.bottom = line(base);
464                                }
465                            }
466                            if preset != "insideHorizontal" {
467                                if !left {
468                                    b.left = line(base);
469                                }
470                                if !right {
471                                    b.right = line(base);
472                                }
473                            }
474                        }
475                        "top" if top => b.top = line(base),
476                        "bottom" if bottom => b.bottom = line(base),
477                        "left" if left => b.left = line(base),
478                        "right" if right => b.right = line(base),
479                        "thickBottom" if bottom => b.bottom = line(BorderStyle::Thick),
480                        "doubleBottom" if bottom => b.bottom = line(BorderStyle::Double),
481                        "topBottom" => {
482                            if top {
483                                b.top = line(base);
484                            }
485                            if bottom {
486                                b.bottom = line(base);
487                            }
488                        }
489                        "topThickBottom" => {
490                            if top {
491                                b.top = line(base);
492                            }
493                            if bottom {
494                                b.bottom = line(BorderStyle::Thick);
495                            }
496                        }
497                        "topDoubleBottom" => {
498                            if top {
499                                b.top = line(base);
500                            }
501                            if bottom {
502                                b.bottom = line(BorderStyle::Double);
503                            }
504                        }
505                        "diagonalDown" => b.diag_down = line(base),
506                        "diagonalUp" => b.diag_up = line(base),
507                        _ => {}
508                    }
509                });
510                cx.sheet_mut(sheet)?.set_style(c, new);
511                if preset == "none" {
512                    // A shared line can be stored on either cell. Selected cells are handled
513                    // by this loop; remove only facing edges outside this range.
514                    for (boundary, dr, dc) in [(top, -1, 0), (bottom, 1, 0), (left, 0, -1), (right, 0, 1)] {
515                        if !boundary {
516                            continue;
517                        }
518                        let Some(neighbor) = c.offset(dr, dc) else { continue };
519                        let st = cx.wb.sheet(sheet).map(|sh| sh.style_id(neighbor)).unwrap_or_default();
520                        let new = cx.wb.styles.derive(st, |s| {
521                            let edge = match (dr, dc) {
522                                (-1, _) => &mut s.borders.bottom,
523                                (1, _) => &mut s.borders.top,
524                                (_, -1) => &mut s.borders.right,
525                                _ => &mut s.borders.left,
526                            };
527                            *edge = BorderLine::default();
528                        });
529                        if new != st {
530                            cx.sheet_mut(sheet)?.set_style(neighbor, new);
531                        }
532                    }
533                }
534            }
535        }
536        Ok(Json::Null)
537    })
538}

540fn format_cells(s: &mut Session, p: &Json) -> Result<Json> {
541    let Some(patch) = p.get("style").map(snake_case_keys) else {
542        // Menu invocation without params: the UI opens the dialog.
543        s.ui_requests.push(crate::engine::UiRequest::Dialog("formatCells".into(), json!({})));
544        return ok();
545    };
546    // Validate by merging into the active style once.
547    let base = serde_json::to_value(active_style(s)?).map_err(|e| EngineError::Other(e.to_string()))?;
548    let merged = merge_json(base, &patch);
549    let _: Style = serde_json::from_value(merged).map_err(|e| bad("home.formatCells", e.to_string()))?;
550    apply_style(s, p, |st| {
551        if let Ok(cur) = serde_json::to_value(&*st)
552            && let Ok(new) = serde_json::from_value::<Style>(merge_json(cur, &patch))
553        {
554            *st = new;
555        }
556    })
557}

559/// The style patch with camelCase keys (`numFmt`, `diagDown`, as documented) renamed to the
560/// style's own snake_case (`num_fmt`, `diag_down`). Keys starting with a capital (`Rgb`) stay.
561fn snake_case_keys(v: &Json) -> Json {
562    match v {
563        Json::Object(m) => Json::Object(
564            m.iter()
565                .map(|(k, v)| {
566                    let key = if k.starts_with(|c: char| c.is_ascii_lowercase()) {
567                        k.chars().fold(String::new(), |mut out, c| {
568                            if c.is_ascii_uppercase() {
569                                out.push('_');
570                            }
571                            out.push(c.to_ascii_lowercase());
572                            out
573                        })
574                    } else {
575                        k.clone()
576                    };
577                    (key, snake_case_keys(v))
578                })
579                .collect(),
580        ),
581        other => other.clone(),
582    }
583}

585/// Deep-merges `patch` into `base` (objects merge, everything else replaces).
586pub(crate) fn merge_json(mut base: Json, patch: &Json) -> Json {
587    match (&mut base, patch) {
588        (Json::Object(b), Json::Object(p)) => {
589            for (k, v) in p {
590                let cur = b.remove(k).unwrap_or(Json::Null);
591                b.insert(k.clone(), merge_json(cur, v));
592            }
593            base
594        }
595        (_, p) => p.clone(),
596    }
597}

635fn merge(s: &mut Session, p: &Json, how: &str) -> Result<Json> {
636    let sheet = target_sheet(s, p)?;
637    let ranges = target_ranges(s, p)?;
638    let how =
639        if how == "center" && s.doc()?.wb.sheet(sheet).is_some_and(|sh| ranges.iter().all(|r| sh.merges.contains(r))) { "unmerge" } else { how };
640    edit(s, |cx| {
641        let mut touched = Vec::new();
642        // Areas whose merging changes: a merged cell blocks a dynamic array's spill.
643        let mut areas: Vec<RangeRef> = ranges.clone();
644        for r in &ranges {
645            let sh = cx.sheet_mut(sheet)?;
646            areas.extend(sh.merges.iter().filter(|m| m.intersects(r)));
647            sh.merges.retain(|m| !m.intersects(r));
648            match how {
649                "unmerge" => {}
650                "across" => {
651                    for row in r.start.row..=r.end.row {
652                        let m = RangeRef::new(CellRef::new(row, r.start.col), CellRef::new(row, r.end.col));
653                        if !m.is_single() {
654                            sh.merges.push(m);
655                            // As below: only the first value of each row survives.
656                            let others: Vec<CellRef> = sh.cells.iter_range(m).map(|(c, _)| c).filter(|c| *c != m.start).collect();
657                            for c in others {
658                                if let Some(mut cell) = sh.cells.get(c).cloned() {
659                                    cell.clear_content();
660                                    sh.set_cell(c, cell);
661                                }
662                                touched.push((sheet, c));
663                            }
664                        }
665                    }
666                }
667                _ => {
668                    if !r.is_single() {
669                        sh.merges.push(*r);
670                    }
671                    // Only the upper-left value survives (Excel warns about this).
672                    if r.count() <= 1_000_000 {
673                        let others: Vec<CellRef> = sh.cells.iter_range(*r).map(|(c, _)| c).filter(|c| *c != r.start).collect();
674                        for c in others {
675                            if let Some(mut cell) = sh.cells.get(c).cloned() {
676                                cell.clear_content();
677                                sh.set_cell(c, cell);
678                            }
679                            touched.push((sheet, c));
680                        }
681                    }
682                }
683            }
684        }
685        // Spills over the areas, and blocked ones that could reach them, are evaluated again.
686        let sh = cx.sheet_mut(sheet)?;
687        for a in &areas {
688            touched.extend(sh.spill_ranges.iter().filter(|(_, r)| r.intersects(a)).map(|(anchor, _)| (sheet, *anchor)));
689            let reach = RangeRef::new(CellRef::new(0, 0), a.end);
690            let blocked = sh
691                .cells
692                .iter_range(reach)
693                // A Python cell's result is blocked and unblocked like a formula's.
694                .filter(|(_, x)| (x.formula.is_some() || x.py.is_some()) && x.value == crate::base::Value::Error(crate::base::CellError::Spill));
695            touched.extend(blocked.map(|(c, _)| (sheet, c)));
696        }
697        cx.changed.extend(touched);
698        Ok(())
699    })?;
700    if how == "center" {
701        apply_style(s, p, |st| st.align.h = HAlign::Center)?;
702    }
703    ok()
704}

738/// Adds or removes one decimal place in the number format (Increase/Decrease Decimal).
739pub fn adjust_decimals(code: &str, value: &crate::base::Value, delta: i32) -> String {
740    if code == "General" {
741        // Start from the displayed number of decimals.
742        let text = value.as_f64().map(crate::base::number_to_text).unwrap_or_default();
743        let dec = text.split_once('.').map_or(0, |(_, d)| d.len()) as i32;
744        let n = (dec + delta).max(0) as usize;
745        return if n == 0 { "0".into() } else { format!("0.{}", "0".repeat(n)) };
746    }
747    let sections: Vec<String> = code
748        .split(';')
749        .map(|sec| {
750            // Find the last digit placeholder run before any exponent / percent etc.
751            let chars: Vec<char> = sec.chars().collect();
752            let mut in_quote = false;
753            let mut in_bracket = false;
754            let mut last_digit = None;
755            let mut point = None;
756            for (i, c) in chars.iter().enumerate() {
757                if *c == '"' {
758                    in_quote = !in_quote;
759                }
760                if *c == '[' && !in_quote {
761                    in_bracket = true;
762                }
763                if *c == ']' && !in_quote {
764                    in_bracket = false;
765                    continue;
766                }
767                if in_quote || in_bracket {
768                    continue;
769                }
770                if matches!(c, '0' | '#' | '?') {
771                    last_digit = Some(i);
772                }
773                if *c == '.' && point.is_none() {
774                    point = Some(i);
775                }
776                if matches!(c, 'E' | 'e') {
777                    break;
778                }
779            }
780            let Some(ld) = last_digit else { return sec.to_string() };
781            let mut out: Vec<char> = chars.clone();
782            if delta > 0 {
783                match point {
784                    Some(_) => out.insert(ld + 1, '0'),
785                    None => {
786                        out.insert(ld + 1, '0');
787                        out.insert(ld + 1, '.');
788                    }
789                }
790            } else if let Some(pt) = point
791                && ld > pt
792            {
793                out.remove(ld);
794                if ld == pt + 1 {
795                    out.remove(pt);
796                }
797            }
798            out.into_iter().collect()
799        })
800        .collect();
801    sections.join(";")
802}

804fn decimals(s: &mut Session, p: &Json, delta: i32) -> Result<Json> {
805    let d = s.doc()?;
806    let sh = d.wb.active().ok_or(EngineError::NoDocument)?;
807    let a = d.selection.active;
808    let code = d.wb.styles.get(sh.style_id(a)).num_fmt.as_str().to_string();
809    let new = adjust_decimals(&code, &sh.value(a), delta);
810    set_fmt(s, p, &new)
811}
