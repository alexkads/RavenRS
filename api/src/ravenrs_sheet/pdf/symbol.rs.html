191/// The advance width of code `code` in 1/1000 em, or `None` for an empty slot.
192pub fn symbol_width(code: u8) -> Option<u16> {
193    let w = *WIDTHS.get(usize::from(code).checked_sub(0x20)?)?;
194    (w > 0).then_some(w)
195}

197#[cfg(test)]
198mod tests {
199    use super::*;
200
201    #[test]
202    fn maths_symbols_are_encoded_with_their_widths() {
203        for (c, code, w) in [
204            ('α', 0x61, 631),
205            ('π', 0x70, 549),
206            ('Σ', 0x53, 592),
207            ('∑', 0xE5, 713),
208            ('∫', 0xF2, 274),
209            ('√', 0xD6, 549),
210            ('≤', 0xA3, 549),
211            ('≥', 0xB3, 549),
212            ('±', 0xB1, 549),
213            ('×', 0xB4, 549),
214            ('÷', 0xB8, 549),
215            ('∞', 0xA5, 713),
216            ('∂', 0xB6, 494),
217            ('−', 0x2D, 549),
218            ('=', 0x3D, 549),
219            ('2', 0x32, 500),
220        ] {
221            assert_eq!(symbol_code(c), Some(code), "{c}");
222            assert_eq!(symbol_width(code), Some(w), "{c}");
223        }
224        assert_eq!(symbol_code('A'), None, "Latin letters belong to the text faces");
225        assert_eq!(symbol_code('ℝ'), None);
226        assert_eq!(symbol_width(0x7F), None);
227        assert_eq!(symbol_width(0x10), None);
228        assert_eq!(symbol_width(0xFF), None);
229    }
230
231    #[test]
232    fn every_encoded_character_has_a_width() {
233        for u in 0..0x3000u32 {
234            if let Some(code) = char::from_u32(u).and_then(symbol_code) {
235                assert!(symbol_width(code).is_some(), "U+{u:04X} -> {code:#x}");
236            }
237        }
238    }
