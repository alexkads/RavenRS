10//! is not modified."). Only the numeric widths are reproduced here, typed in from the public
11//! metrics; no AFM file is included. Oblique/italic faces reuse the upright widths (an

67/// Maps a non-ASCII character to an ASCII character of similar width, or `None` to use the
68/// font's default width.
69pub(crate) fn fold(c: char) -> Option<char> {
70    Some(match c {
71        'À'..='Å' => 'A',
72        'Æ' => 'W',
73        'Ç' => 'C',
74        'È'..='Ë' => 'E',
75        'Ì'..='Ï' => 'I',
76        'Ð' => 'D',
77        'Ñ' => 'N',
78        'Ò'..='Ö' | 'Ø' => 'O',
79        '×' => '+',
80        'Ù'..='Ü' => 'U',
81        'Ý' | 'Ÿ' => 'Y',
82        'Þ' => 'P',
83        'ß' => 'b',
84        'à'..='å' => 'a',
85        'æ' => 'm',
86        'ç' => 'c',
87        'è'..='ë' => 'e',
88        'ì'..='ï' => 'i',
89        'ð' => 'o',
90        'ñ' => 'n',
91        'ò'..='ö' | 'ø' => 'o',
92        '÷' => '+',
93        'ù'..='ü' => 'u',
94        'ý' | 'ÿ' => 'y',
95        'þ' => 'p',
96        'Š' => 'S',
97        'š' => 's',
98        'Ž' => 'Z',
99        'ž' => 'z',
100        'Œ' => 'W',
101        'œ' => 'm',
102        '‘'..='‚' => '\'',
103        '“'..='„' => '"',
104        '–' | '€' | '£' | '¥' | '¢' | '¤' => '0',
105        '\u{a0}' => ' ',
106        '·' | '•' => '.',
107        '«' | '»' | '‹' | '›' => '<',
108        '¡' => '!',
109        '¿' => '?',
110        '¦' => '|',
111        '°' | '¹' | '²' | '³' | 'ª' | 'º' => '*',
112        _ => return None,
113    })
114}
