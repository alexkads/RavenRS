1//! Addressing the bands of a report in design order.
2
3use ravenrs_report::{Band, Report};
4
5/// Which band of the report.
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
7pub enum BandRef {
8    ReportHeader,
9    PageHeader,
10    GroupHeader(usize),
11    Detail,
12    DetailReportHeader(usize),
13    DetailReportDetail(usize),
14    DetailReportFooter(usize),
15    GroupFooter(usize),
16    ReportFooter,
17    PageFooter,
18}
19
20impl BandRef {
21    /// The bands present in `report`, top to bottom as the designer shows them.
22    pub fn all(report: &Report) -> Vec<BandRef> {
23        let mut out = Vec::new();
24        if report.report_header.is_some() {
25            out.push(Self::ReportHeader);
26        }
27        if report.page_header.is_some() {
28            out.push(Self::PageHeader);
29        }
30        for (i, g) in report.groups.iter().enumerate() {
31            if g.header.is_some() {
32                out.push(Self::GroupHeader(i));
33            }
34        }
35        out.push(Self::Detail);
36        for (i, d) in report.detail_reports.iter().enumerate() {
37            if d.header.is_some() {
38                out.push(Self::DetailReportHeader(i));
39            }
40            out.push(Self::DetailReportDetail(i));
41            if d.footer.is_some() {
42                out.push(Self::DetailReportFooter(i));
43            }
44        }
45        for (i, g) in report.groups.iter().enumerate().rev() {
46            if g.footer.is_some() {
47                out.push(Self::GroupFooter(i));
48            }
49        }
50        if report.report_footer.is_some() {
51            out.push(Self::ReportFooter);
52        }
53        if report.page_footer.is_some() {
54            out.push(Self::PageFooter);
55        }
56        out
57    }
58
59    /// The band in `report`.
60    pub fn get(self, report: &Report) -> Option<&Band> {
61        match self {
62            Self::ReportHeader => report.report_header.as_ref(),
63            Self::PageHeader => report.page_header.as_ref(),
64            Self::GroupHeader(i) => report.groups.get(i)?.header.as_ref(),
65            Self::GroupFooter(i) => report.groups.get(i)?.footer.as_ref(),
66            Self::Detail => Some(&report.detail),
67            Self::DetailReportHeader(i) => report.detail_reports.get(i)?.header.as_ref(),
68            Self::DetailReportDetail(i) => Some(&report.detail_reports.get(i)?.detail),
69            Self::DetailReportFooter(i) => report.detail_reports.get(i)?.footer.as_ref(),
70            Self::ReportFooter => report.report_footer.as_ref(),
71            Self::PageFooter => report.page_footer.as_ref(),
72        }
73    }
74
75    /// The band in `report`, for editing.
76    pub fn get_mut(self, report: &mut Report) -> Option<&mut Band> {
77        match self {
78            Self::ReportHeader => report.report_header.as_mut(),
79            Self::PageHeader => report.page_header.as_mut(),
80            Self::GroupHeader(i) => report.groups.get_mut(i)?.header.as_mut(),
81            Self::GroupFooter(i) => report.groups.get_mut(i)?.footer.as_mut(),
82            Self::Detail => Some(&mut report.detail),
83            Self::DetailReportHeader(i) => report.detail_reports.get_mut(i)?.header.as_mut(),
84            Self::DetailReportDetail(i) => Some(&mut report.detail_reports.get_mut(i)?.detail),
85            Self::DetailReportFooter(i) => report.detail_reports.get_mut(i)?.footer.as_mut(),
86            Self::ReportFooter => report.report_footer.as_mut(),
87            Self::PageFooter => report.page_footer.as_mut(),
88        }
89    }
90
91    /// The band's name, as the designer shows it.
92    pub fn title(self, report: &Report) -> String {
93        match self {
94            Self::ReportHeader => "Cabeçalho do relatório".into(),
95            Self::PageHeader => "Cabeçalho da página".into(),
96            Self::GroupHeader(i) => format!("Cabeçalho do grupo · {}", group_expr(report, i)),
97            Self::GroupFooter(i) => format!("Rodapé do grupo · {}", group_expr(report, i)),
98            Self::Detail => "Detalhe".into(),
99            Self::DetailReportHeader(i) => format!("Cabeçalho · {}", detail_member(report, i)),
100            Self::DetailReportDetail(i) => format!("Detalhe · {}", detail_member(report, i)),
101            Self::DetailReportFooter(i) => format!("Rodapé · {}", detail_member(report, i)),
102            Self::ReportFooter => "Rodapé do relatório".into(),
103            Self::PageFooter => "Rodapé da página".into(),
104        }
105    }
106
107    /// A stable number for widget keys.
108    pub fn key(self) -> u64 {
109        match self {
110            Self::ReportHeader => 1,
111            Self::PageHeader => 2,
112            Self::Detail => 3,
113            Self::ReportFooter => 4,
114            Self::PageFooter => 5,
115            Self::GroupHeader(i) => 100 + i as u64,
116            Self::GroupFooter(i) => 200 + i as u64,
117            Self::DetailReportHeader(i) => 300 + i as u64,
118            Self::DetailReportDetail(i) => 400 + i as u64,
119            Self::DetailReportFooter(i) => 500 + i as u64,
120        }
121    }
122}
123
124fn group_expr(report: &Report, i: usize) -> String {
125    report.groups.get(i).map(|g| g.expression.clone()).unwrap_or_default()
126}
127
128fn detail_member(report: &Report, i: usize) -> String {
129    report.detail_reports.get(i).map(|d| d.data_member.clone()).unwrap_or_default()
130}
