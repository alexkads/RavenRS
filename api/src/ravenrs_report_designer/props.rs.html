37/// An expression's field with the button that opens it in the expression editor.

56fn number_field(
57    e: &Editor,
58    kind: &'static str,
59    get: fn(&ravenrs_report::Control) -> f32,
60    set: fn(&mut ravenrs_report::Control, f32),
61) -> TextField {
62    let (read, write) = (e.clone(), e.clone());
63    TextField::new(Binding::new(
64        move || read.selected_control().map(|(_, _, c)| format!("{:.0}", get(&c))).unwrap_or_default(),
65        move |s: String| {
66            if let Ok(v) = s.trim().replace(',', ".").parse::<f32>() {
67                write.edit_control(kind, |c| set(c, v.max(0.0)));
68            }
69        },
70    ))
71    .compact()
72}

74struct Swatches {
75    editor: Editor,
76    background: bool,
77}
78
79impl View for Swatches {
80    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
81        let current = self
82            .editor
83            .selected_control()
84            .and_then(|(_, _, c)| if self.background { c.style.background } else { c.style.color });
85        let pick = |hex: Option<&'static str>| {
86            let (e, background) = (self.editor.clone(), self.background);
87            move || {
88                let value = hex.map(str::to_owned);
89                e.edit_control(if background { "bg" } else { "fg" }, |c| {
90                    if background {
91                        c.style.background = value;
92                    } else {
93                        c.style.color = value;
94                    }
95                });
96            }
97        };
98        let mut items: Vec<AnyView> =
99            vec![Box::new(ColorSwatch::none("Padrão").selected(current.is_none()).on_click(pick(None)))];
100        for (hex, name) in PALETTE {
101            let color = crate::surface::color(Some(hex)).unwrap_or(Color::TRANSPARENT);
102            items.push(Box::new(
103                ColorSwatch::new(color, name)
104                    .selected(current.as_deref() == Some(hex))
105                    .on_click(pick(Some(hex))),
106            ));
107        }
108        cx.child(&HStack::new(items).spacing(0.0).wrap())
109    }
110}

136/// A text field over part of the selected control.
137fn control_text(
138    e: &Editor,
139    kind: &'static str,
140    read: impl Fn(&ControlKind) -> String + 'static,
141    write: impl Fn(&mut ControlKind, String) + 'static,
142) -> TextField {
143    let (reader, writer) = (e.clone(), e.clone());
144    TextField::new(Binding::new(
145        move || reader.selected_control().map(|(_, _, c)| read(&c.kind)).unwrap_or_default(),
146        move |s: String| writer.edit_control(kind, |c| write(&mut c.kind, s)),
147    ))
148    .compact()
149}

151fn picture_props(e: &Editor, picture: &ravenrs_report::Picture, rows: &mut Vec<AnyView>) {
152    let chooser = e.clone();
153    rows.push(Box::new(
154        Button::new(if picture.source.is_empty() { "Escolher imagem…" } else { "Trocar imagem…" })
155            .icon(IconName::Image)
156            .full_width()
157            .on_click(move || {
158                let e = chooser.clone();
159                crate::preview::pick_image(move |source| {
160                    e.edit_control("picture", |c| {
161                        if let ControlKind::Picture(p) = &mut c.kind {
162                            p.source = source;
163                        }
164                    });
165                });
166            }),
167    ));
168    let expression = control_text(
169        e,
170        "picture-field",
171        |k| match k {
172            ControlKind::Picture(p) => p.expression.clone().unwrap_or_default(),
173            _ => String::new(),
174        },
175        |k, s| {
176            if let ControlKind::Picture(p) = k {
177                p.expression = (!s.trim().is_empty()).then_some(s);
178            }
179        },
180    );
181    rows.push(Box::new(
182        FormField::new("Campo com a imagem", expression)
183            .hint("Opcional. Ex.: [Foto], em base64 ou endereço data:"),
184    ));
185    let index = PICTURE_MODES.iter().position(|(_, m)| *m == picture.mode).unwrap_or(0);
186    let write = e.clone();
187    let mode = Select::new(
188        Binding::new(
189            move || index,
190            move |i: usize| {
191                write.edit_control("picture-mode", |c| {
192                    if let ControlKind::Picture(p) = &mut c.kind {
193                        p.mode = PICTURE_MODES[i.min(PICTURE_MODES.len() - 1)].1;
194                    }
195                });
196            },
197        ),
198        PICTURE_MODES.iter().map(|(name, _)| *name),
199    )
200    .width(268.0)
201    .label("Encaixe");
202    rows.push(Box::new(FormField::new("Encaixe", mode)));
203}

205fn link_props(e: &Editor, rows: &mut Vec<AnyView>) {
206    let text = control_text(
207        e,
208        "link-text",
209        |k| match k {
210            ControlKind::Link(l) => l.label.expression.clone().unwrap_or_else(|| l.label.text.clone()),
211            _ => String::new(),
212        },
213        |k, s| {
214            if let ControlKind::Link(l) = k {
215                if l.label.expression.is_some() {
216                    l.label.expression = Some(s);
217                } else {
218                    l.label.text = s;
219                }
220            }
221        },
222    );
223    rows.push(Box::new(FormField::new("Texto", text)));
224    let url = control_text(
225        e,
226        "link-url",
227        |k| match k {
228            ControlKind::Link(l) => {
229                l.url_expression.clone().map_or_else(|| l.url.clone(), |x| format!("={x}"))
230            }
231            _ => String::new(),
232        },
233        |k, s| {
234            if let ControlKind::Link(l) = k {
235                match s.strip_prefix('=') {
236                    Some(expression) => l.url_expression = Some(expression.to_owned()),
237                    None => {
238                        l.url_expression = None;
239                        l.url = s;
240                    }
241                }
242            }
243        },
244    );
245    rows.push(Box::new(
246        FormField::new("Endereço", url)
247            .hint("https:, http: ou mailto:. Comece com = para usar uma expressão"),
248    ));
249}

418fn control_props(e: &Editor) -> impl View + use<> {
419    let Some((band, _, control)) = e.selected_control() else { return Box::new(Text::new("")) as AnyView };
420    let report = e.report.get();
421    let kind = match &control.kind {
422        ControlKind::Label(l) if l.expression.is_some() => "Campo",
423        ControlKind::Label(_) => "Rótulo",
424        ControlKind::Table(_) => "Tabela",
425        ControlKind::CheckBox(_) => "Caixa de seleção",
426        ControlKind::Line(_) => "Linha",
427        ControlKind::Shape(_) => "Forma",
428        ControlKind::Panel(_) => "Painel",
429        ControlKind::Picture(_) => "Imagem",
430        ControlKind::Barcode(_) => "Código de barras",
431        ControlKind::Chart(_) => "Gráfico",
432        ControlKind::Link(_) => "Link",
433        ControlKind::Bookmark(_) => "Marcador",
434        _ => "Controle",
435    };
436    let mut rows: Vec<AnyView> = vec![Box::new(
437        HStack::new((
438            Text::new(kind).style(TextStyle::Headline),
439            Spacer::new(),
440            Badge::new(band.title(&report)).tone(Tone::Accent),
441        ))
442        .align(Align::Center),
443    )];
444    if let ControlKind::Label(label) = &control.kind {
445        let is_field = label.expression.is_some();
446        let (read, write) = (e.clone(), e.clone());
447        let text = TextField::new(Binding::new(
448            move || {
449                read.selected_control()
450                    .map(|(_, _, c)| match c.kind {
451                        ControlKind::Label(l) => l.expression.unwrap_or(l.text),
452                        _ => String::new(),
453                    })
454                    .unwrap_or_default()
455            },
456            move |s: String| {
457                write.edit_control("text", |c| {
458                    if let ControlKind::Label(l) = &mut c.kind {
459                        if l.expression.is_some() {
460                            l.expression = Some(s);
461                        } else {
462                            l.text = s;
463                        }
464                    }
465                })
466            },
467        ))
468        .compact();
469        let text: AnyView =
470            if is_field { Box::new(with_editor(e, text, ExpressionTarget::Control)) } else { Box::new(text) };
471        rows.push(Box::new(FormField::new(if is_field { "Expressão" } else { "Texto" }, text).hint(
472            if is_field {
473                "Ex.: [Preco] * [Qtd], Sum([Total]), Iif([Pago], 'Sim', 'Não')"
474            } else {
475                "Texto fixo exibido no relatório"
476            },
477        )));
478        let current = label.format.clone().unwrap_or_default();
479        let index = FORMATS.iter().position(|(_, f)| *f == current).unwrap_or(0);
480        let write = e.clone();
481        let picker = Select::new(
482            Binding::new(
483                move || index,
484                move |i: usize| {
485                    let f = FORMATS[i.min(FORMATS.len() - 1)].1;
486                    write.edit_control("format", |c| {
487                        if let ControlKind::Label(l) = &mut c.kind {
488                            l.format = (!f.is_empty()).then(|| f.to_owned());
489                        }
490                    });
491                },
492            ),
493            FORMATS.iter().map(|(name, _)| *name),
494        )
495        .width(268.0)
496        .label("Formato");
497        rows.push(Box::new(FormField::new("Formato", picker)));
498    }
499    match &control.kind {
500        ControlKind::Picture(picture) => picture_props(e, picture, &mut rows),
501        ControlKind::Link(_) => link_props(e, &mut rows),
502        ControlKind::Bookmark(_) => bookmark_props(e, &mut rows),
503        ControlKind::Barcode(barcode) => barcode_props(e, barcode, &mut rows),
504        ControlKind::Chart(chart) => chart_props(e, chart, &mut rows),
505        _ => {}
506    }
507    rows.push(Box::new(section("Posição e tamanho (pt)")));
508    rows.push(Box::new(
509        Grid::new(
510            2,
511            (
512                FormField::new("X", number_field(e, "x", |c| c.x, |c, v| c.x = v)),
513                FormField::new("Y", number_field(e, "y", |c| c.y, |c, v| c.y = v)),
514                FormField::new("Largura", number_field(e, "w", |c| c.width, |c, v| c.width = v.max(4.0))),
515                FormField::new("Altura", number_field(e, "h", |c| c.height, |c, v| c.height = v.max(2.0))),
516            ),
517        )
518        .spacing(SPACING.s3),
519    ));
520    rows.push(Box::new(section("Aparência")));
521    let size_e = e.clone();
522    let size = State::new(control.style.font_size.unwrap_or(10.0).round() as i32);
523    let size_binding = {
524        let (read, write) = (e.clone(), size_e);
525        Binding::new(
526            move || {
527                read.selected_control()
528                    .map_or(10, |(_, _, c)| c.style.font_size.unwrap_or(10.0).round() as i32)
529            },
530            move |v: i32| write.edit_control("size", |c| c.style.font_size = Some(v.clamp(5, 72) as f32)),
531        )
532    };
533    let _ = size;
534    let (bread, bwrite) = (e.clone(), e.clone());
535    let bold = ToggleButton::new(
536        Binding::new(
537            move || bread.selected_control().is_some_and(|(_, _, c)| c.style.bold == Some(true)),
538            move |v: bool| bwrite.edit_control("bold", |c| c.style.bold = Some(v)),
539        ),
540        "Negrito",
541    )
542    .icon(IconName::Bold);
543    rows.push(Box::new(
544        HStack::new((
545            FormField::new("Fonte", Stepper::new(size_binding).range(5, 72).label("Tamanho da fonte")),
546            Spacer::new(),
547            bold,
548        ))
549        .align(Align::End),
550    ));
551    let (aread, awrite) = (e.clone(), e.clone());
552    let align = SegmentedControl::with_icons(
553        Binding::new(
554            move || {
555                aread.selected_control().map_or(0, |(_, _, c)| match c.style.text_align {
556                    Some(TextAlign::Center) => 1,
557                    Some(TextAlign::Right) => 2,
558                    _ => 0,
559                })
560            },
561            move |i: usize| {
562                awrite.edit_control("align", |c| {
563                    c.style.text_align =
564                        Some([TextAlign::Left, TextAlign::Center, TextAlign::Right][i.min(2)]);
565                })
566            },
567        ),
568        [(IconName::AlignLeft, "Esq."), (IconName::AlignCenter, "Centro"), (IconName::AlignRight, "Dir.")],
569    )
570    .label("Alinhamento");
571    rows.push(Box::new(FormField::new("Alinhamento", align)));
572    rows.push(Box::new(FormField::new("Cor do texto", Swatches { editor: e.clone(), background: false })));
573    rows.push(Box::new(FormField::new("Fundo", Swatches { editor: e.clone(), background: true })));
574    let (bread, bwrite) = (e.clone(), e.clone());
575    rows.push(Box::new(
576        Toggle::new(Binding::new(
577            move || {
578                bread.selected_control().is_some_and(|(_, _, c)| c.style.border_width.unwrap_or(0.0) > 0.0)
579            },
580            move |on: bool| {
581                bwrite.edit_control("border", |c| {
582                    if on {
583                        c.style = c.style.clone().merged(&Style::new().border("all", 0.75, "#dadde5"));
584                    } else {
585                        c.style.border_width = Some(0.0);
586                    }
587                })
588            },
589        ))
590        .label("Borda"),
591    ));
592    let del = e.clone();
593    rows.push(Box::new(
594        Button::new("Excluir controle")
595            .icon(IconName::Trash2)
596            .style(ButtonStyle::Danger)
597            .full_width()
598            .on_click(move || del.delete_selected()),
599    ));
600    Box::new(VStack::new(rows).spacing(SPACING.s4).align(Align::Stretch)) as AnyView
601}

603fn band_props(e: &Editor, band: BandRef) -> impl View + use<> {
604    let report = e.report.get();
605    let mut rows: Vec<AnyView> = vec![Box::new(Text::new(band.title(&report)).style(TextStyle::Headline))];
606    let (read, write) = (e.clone(), e.clone());
607    rows.push(Box::new(FormField::new(
608        "Altura (pt)",
609        TextField::new(Binding::new(
610            move || read.report.with(|r| band.get(r).map(|b| format!("{:.0}", b.height)).unwrap_or_default()),
611            move |s: String| {
612                if let Ok(v) = s.trim().parse::<f32>() {
613                    write.edit_as("band-height", |r| {
614                        if let Some(b) = band.get_mut(r) {
615                            b.height = v.max(4.0);
616                        }
617                    });
618                }
619            },
620        ))
621        .compact(),
622    )));
623    if let BandRef::GroupHeader(i) | BandRef::GroupFooter(i) = band {
624        let (read, write) = (e.clone(), e.clone());
625        rows.push(Box::new(
626            FormField::new(
627                "Agrupar por",
628                with_editor(
629                    e,
630                    TextField::new(Binding::new(
631                        move || {
632                            read.report
633                                .with(|r| r.groups.get(i).map(|g| g.expression.clone()).unwrap_or_default())
634                        },
635                        move |s: String| {
636                            write.edit_as("group", |r| {
637                                if let Some(g) = r.groups.get_mut(i) {
638                                    g.expression = s;
639                                }
640                            })
641                        },
642                    ))
643                    .compact(),
644                    ExpressionTarget::Group(i),
645                ),
646            )
647            .hint("Uma expressão, como [Regiao]"),
648        ));
649        let (read, write) = (e.clone(), e.clone());
650        rows.push(Box::new(
651            Toggle::new(Binding::new(
652                move || read.report.with(|r| r.groups.get(i).is_some_and(|g| g.page_break_before)),
653                move |v: bool| {
654                    write.edit(|r| {
655                        if let Some(g) = r.groups.get_mut(i) {
656                            g.page_break_before = v;
657                        }
658                    })
659                },
660            ))
661            .label("Cada grupo em uma nova página"),
662        ));
663    }
664    if band == BandRef::Detail {
665        let (read, write) = (e.clone(), e.clone());
666        rows.push(Box::new(
667            Toggle::new(Binding::new(
668                move || read.report.with(|r| r.detail.odd_style.is_some()),
669                move |on: bool| {
670                    write.edit(|r| {
671                        if on {
672                            r.styles
673                                .entry("odd".into())
674                                .or_insert_with(|| Style::new().background("#ffffff"));
675                            r.styles
676                                .entry("even".into())
677                                .or_insert_with(|| Style::new().background("#f4f5f8"));
678                            r.detail.odd_style = Some("odd".into());
679                            r.detail.even_style = Some("even".into());
680                        } else {
681                            r.detail.odd_style = None;
682                            r.detail.even_style = None;
683                        }
684                    })
685                },
686            ))
687            .label("Linhas zebradas"),
688        ));
689    }
690    rows.push(Box::new(
691        Text::new("Arraste campos da lista para esta banda, ou use Inserir na faixa de opções.")
692            .style(TextStyle::Callout)
693            .foreground(Semantic::InkMuted),
694    ));
695    VStack::new(rows).spacing(SPACING.s4).align(Align::Stretch)
696}

698fn report_props(e: &Editor) -> impl View + use<> {
699    let (read, write) = (e.clone(), e.clone());
700    let name = TextField::new(Binding::new(
701        move || read.report.with(|r| r.name.clone()),
702        move |s: String| write.edit_as("name", |r| r.name = s),
703    ))
704    .compact();
705    let (read, write) = (e.clone(), e.clone());
706    let culture = SegmentedControl::new(
707        Binding::new(
708            move || read.report.with(|r| usize::from(!r.culture.to_lowercase().starts_with("pt"))),
709            move |i: usize| write.edit(|r| r.culture = if i == 0 { "pt-BR".into() } else { "en-US".into() }),
710        ),
711        ["Português", "English"],
712    )
713    .label("Idioma dos formatos");
714    let (read, write) = (e.clone(), e.clone());
715    let paper = SegmentedControl::new(
716        Binding::new(
717            move || read.report.with(|r| usize::from(r.page.width > r.page.height)),
718            move |i: usize| {
719                write.edit(|r| {
720                    let base = if (r.page.width.min(r.page.height) - 612.0).abs() < 1.0 {
721                        PageSettings::letter()
722                    } else {
723                        PageSettings::a4()
724                    };
725                    r.page = if i == 1 { base.landscape() } else { base };
726                })
727            },
728        ),
729        ["Retrato", "Paisagem"],
730    )
731    .label("Orientação");
732    let (read, write) = (e.clone(), e.clone());
733    let filter = with_editor(
734        e,
735        TextField::new(Binding::new(
736            move || read.report.with(|r| r.filter.clone().unwrap_or_default()),
737            move |s: String| write.edit_as("filter", |r| r.filter = (!s.trim().is_empty()).then_some(s)),
738        ))
739        .compact(),
740        ExpressionTarget::Filter,
741    );
742    VStack::new((
743        Text::new("Relatório").style(TextStyle::Headline),
744        FormField::new("Nome", name),
745        FormField::new("Idioma dos formatos", culture),
746        FormField::new("Orientação", paper),
747        FormField::new("Filtro", filter)
748            .hint("Linhas em que a expressão é falsa ficam de fora, ex.: [Total] >= ?MinTotal"),
749        Text::new("Clique numa banda ou num controle para editar as propriedades dele.")
750            .style(TextStyle::Callout)
751            .foreground(Semantic::InkMuted),
752    ))
753    .spacing(SPACING.s4)
754    .align(Align::Stretch)
755}
