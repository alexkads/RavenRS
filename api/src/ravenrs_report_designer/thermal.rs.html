27/// Where the report goes, as it is kept or as it starts.
28fn settings_of(editor: &Editor) -> PrinterSettings {
29    editor.report.with(|report| report.printer.clone().unwrap_or_default())
30}
31
32fn change(editor: &Editor, what: impl FnOnce(&mut PrinterSettings)) {
33    let mut settings = settings_of(editor);
34    what(&mut settings);
35    editor.edit_as("printer", |report| report.printer = Some(settings));
36}

38/// What it would print and in how many bytes, or why it cannot.
39fn prepared(dialog: &ThermalDialog, settings: &PrinterSettings) -> Result<(Vec<u8>, Vec<Vec<u8>>), String> {
40    let language = Language::from_name(&settings.language)
41        .ok_or_else(|| format!("linguagem desconhecida: {}", settings.language))?;
42    let document = preview::document(&dialog.editor, &dialog.cache, &dialog.params.get());
43    if let Some(problem) = document.errors.first() {
44        return Err(problem.clone());
45    }
46    let mut job = LabelJob::new(language, settings.dpi);
47    if let Some(preset) = presets().iter().find(|preset| Some(&preset.driver_id) == settings.driver.as_ref())
48        && preset.languages.contains(&language)
49    {
50        job.ajustes = preset.ajustes;
51    }
52    let bytes = compile(&document, &job).map_err(|error| error.to_string())?;
53    let mut engine = ReportEngine::new();
54    let mut pictures = Vec::new();
55    for page in document.pages.iter().take(3) {
56        let bitmap =
57            rasterize_page(page, settings.dpi, engine.text_system()).map_err(|error| error.to_string())?;
58        pictures.push(bitmap_to_png(&bitmap));
59    }
60    Ok((bytes, pictures))
61}

63impl View for ThermalDialog {
64    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
65        let e = &self.editor;
66        if !e.thermal_open.get() {
67            return cx.child(&VStack::new(Vec::<AnyView>::new()));
68        }
69        let settings = settings_of(e);
70        let result: State<Option<Result<String, String>>> = cx.use_state(|| None);
71        // The printer's model only fills in its language, which stays free to change.
72        let (read, write) = (e.clone(), e.clone());
73        let languages = Select::new(
74            Binding::new(
75                move || {
76                    let name = settings_of(&read).language;
77                    let found = Language::from_name(&name);
78                    Language::all().iter().position(|language| Some(*language) == found).unwrap_or(0)
79                },
80                move |i: usize| {
81                    let language = Language::all()[i.min(Language::all().len() - 1)];
82                    change(&write, |s| s.language = language.name().to_owned());
83                },
84            ),
85            Language::all().iter().map(|language| language.name().to_owned()).collect::<Vec<_>>(),
86        )
87        .width(220.0)
88        .label("Linguagem");
89        let write = e.clone();
90        let chosen = State::new(0usize);
91        let (chosen_read, chosen_write) = (chosen.clone(), chosen);
92        let models = Select::new(
93            Binding::new(
94                move || chosen_read.get(),
95                move |i: usize| {
96                    chosen_write.set(i);
97                    if let Some(preset) = presets().get(i) {
98                        let driver = preset.driver_id.clone();
99                        let language = preset.languages.first().copied();
100                        change(&write, |s| {
101                            s.driver = Some(driver);
102                            if let Some(language) = language {
103                                s.language = language.name().to_owned();
104                            }
105                        });
106                    }
107                },
108            ),
109            presets().iter().map(|preset| preset.name.to_owned()).collect::<Vec<_>>(),
110        )
111        .width(220.0)
112        .label("Modelo");
113        let (read, write) = (e.clone(), e.clone());
114        let dpi = Select::new(
115            Binding::new(
116                move || usize::from(settings_of(&read).dpi == 300),
117                move |i: usize| change(&write, |s| s.dpi = if i == 1 { 300 } else { 203 }),
118            ),
119            ["203 dpi", "300 dpi"],
120        )
121        .width(120.0)
122        .label("Resolução");
123        let (read, write) = (e.clone(), e.clone());
124        let connection = TextField::new(Binding::new(
125            move || settings_of(&read).connection,
126            move |value: String| change(&write, |s| s.connection = value),
127        ))
128        .placeholder("192.168.0.50:9100, cups:termica ou /caminho/arquivo.prn");
129
130        let prepared = prepared(self, &settings);
131        let mut parts: Vec<AnyView> = vec![Box::new(
132            HStack::new((
133                FormField::new("Modelo", models),
134                FormField::new("Linguagem", languages),
135                FormField::new("Resolução", dpi),
136            ))
137            .spacing(SPACING.s3),
138        )];
139        parts.push(Box::new(FormField::new("Conexão", connection)));
140        let mut bytes_to_send = None;
141        match prepared {
142            Ok((bytes, pictures)) => {
143                parts.push(Box::new(
144                    Text::new(format!("Prévia em pontos, {} dpi. {} bytes.", settings.dpi, bytes.len()))
145                        .style(TextStyle::Callout)
146                        .foreground(Semantic::InkMuted),
147                ));
148                let shown: Vec<AnyView> = pictures
149                    .iter()
150                    .filter_map(|png| Image::from_bytes(png).ok())
151                    .map(|image| Box::new(image) as AnyView)
152                    .collect();
153                parts.push(Box::new(ScrollView::new(shown).frame(None, Some(360.0))));
154                bytes_to_send = Some(bytes);
155            }
156            Err(problem) => {
157                parts.push(Box::new(Alert::new("Não dá para imprimir assim", problem).tone(Tone::Danger)))
158            }
159        }
160        if let Some(outcome) = result.get() {
161            parts.push(Box::new(match outcome {
162                Ok(what) => Alert::new("Pronto", what).tone(Tone::Success),
163                Err(why) => Alert::new("Não saiu", why).tone(Tone::Danger),
164            }));
165        }
166        let picker = cx.file_picker();
167        let name = e.report.with(|report| report.name.clone());
168        let to_file = bytes_to_send.clone();
169        let save = Button::new("Salvar arquivo .prn").disabled(to_file.is_none()).on_click(move || {
170            if let Some(bytes) = &to_file {
171                picker.save(&format!("{name}.prn"), bytes);
172            }
173        });
174        let send = send_button(bytes_to_send, settings.connection.clone(), result, cx);
175        cx.child(
176            &Dialog::new(e.thermal_open.binding(), "Impressora direta")
177                .description(
178                    "Etiquetas e cupons na linguagem da impressora, com o mesmo layout deste relatório.",
179                )
180                .width(720.0)
181                .close_label("Fechar")
182                .content(VStack::new(parts).spacing(SPACING.s4).align(ravenrs_layout::Align::Stretch))
183                .action(HStack::new((save, send)).spacing(SPACING.s2)),
184        )
185    }
186}

188/// How a send that is under way comes back.
189#[cfg(not(target_arch = "wasm32"))]
190type Answer = std::sync::mpsc::Receiver<Result<(), String>>;
191
192/// Sends what the printer takes, out of the way of the screen.
193#[cfg(not(target_arch = "wasm32"))]
194fn send_button(
195    bytes: Option<Vec<u8>>,
196    connection: String,
197    result: State<Option<Result<String, String>>>,
198    cx: &mut BuildCx<'_>,
199) -> Button {
200    let inbox: State<Option<Answer>> = cx.use_state(|| None);
201    let answered = inbox.peek().as_ref().and_then(|receiver| receiver.try_recv().ok());
202    if let Some(answer) = answered {
203        inbox.set(None);
204        result.set(Some(answer.map(|()| "Enviado para a impressora.".to_owned())));
205    }
206    let busy = inbox.peek().is_some();
207    let waker = cx.ui_waker();
208    Button::new("Imprimir")
209        .style(ButtonStyle::Primary)
210        .loading(busy)
211        .disabled(bytes.is_none() || busy)
212        .on_click(move || {
213            let Some(bytes) = bytes.clone() else { return };
214            let (connection, waker) = (connection.clone(), waker.clone());
215            let (answer, receiver) = std::sync::mpsc::channel();
216            inbox.set(Some(receiver));
217            std::thread::spawn(move || {
218                let sent = ravenrs_report_label::output_for(&connection)
219                    .and_then(|output| output.send(&bytes))
220                    .map_err(|error| error.to_string());
221                let _ = answer.send(sent);
222                waker.wake();
223            });
224        })
225}
