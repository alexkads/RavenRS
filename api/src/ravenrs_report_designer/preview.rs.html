11/// The last render, reused while the report and parameters do not change.
12#[derive(Default)]
13pub struct RenderCache {
14    key: Option<(u64, String)>,
15    pub doc: Option<Rc<Document>>,
16    pub millis: f64,
17}
18
19/// Milliseconds from a monotonic clock, on desktop and in the browser.
20pub fn now_ms() -> f64 {
21    #[cfg(target_arch = "wasm32")]
22    {
23        web_sys::window().and_then(|w| w.performance()).map_or(0.0, |p| p.now())
24    }
25    #[cfg(not(target_arch = "wasm32"))]
26    {
27        use std::sync::OnceLock;
28        static START: OnceLock<std::time::Instant> = OnceLock::new();
29        START.get_or_init(std::time::Instant::now).elapsed().as_secs_f64() * 1000.0
30    }
31}

33/// Renders the report if it changed since the last call.
34pub fn document(
35    editor: &Editor,
36    cache: &Rc<RefCell<RenderCache>>,
37    params: &HashMap<String, serde_json::Value>,
38) -> Rc<Document> {
39    let version = editor.version.get();
40    let mut sorted: Vec<_> = params.iter().collect();
41    sorted.sort_by_key(|(name, _)| *name);
42    let key = (version, format!("{sorted:?}"));
43    if let Some(doc) = cache.borrow().doc.clone().filter(|_| cache.borrow().key.as_ref() == Some(&key)) {
44        return doc;
45    }
46    let report = editor.report.get();
47    let data = report.embedded_data().cloned().unwrap_or(serde_json::Value::Null);
48    let options = RenderOptions { parameters: params.clone(), ..RenderOptions::default() };
49    let start = now_ms();
50    let doc = Rc::new(editor.engine.borrow_mut().render_with(&report, &data, &options));
51    let mut c = cache.borrow_mut();
52    c.millis = now_ms() - start;
53    c.key = Some(key);
54    c.doc = Some(doc.clone());
55    doc
56}

60/// Exports the document and hands the file to the user: a download on the web,
61/// or the program that opens the format on the desktop.
62pub fn export(document: &Document, format: Export) {
63    match ravenrs_report_export::export(document, format, None) {
64        Ok(file) => {
65            #[cfg(target_arch = "wasm32")]
66            download(&file);
67            #[cfg(not(target_arch = "wasm32"))]
68            open_on_desktop(&file);
69        }
70        Err(e) => tracing::warn!(error = %e, "could not export the report"),
71    }
72}

74#[cfg(target_arch = "wasm32")]
75fn download(file: &ExportedFile) {
76    use wasm_bindgen::JsCast;
77
78    let Some(page) = web_sys::window().and_then(|w| w.document()) else { return };
79    let parts = js_sys::Array::of1(&js_sys::Uint8Array::from(file.bytes.as_slice()));
80    let options = web_sys::BlobPropertyBag::new();
81    options.set_type(file.mime);
82    let Ok(blob) = web_sys::Blob::new_with_u8_array_sequence_and_options(&parts, &options) else { return };
83    let Ok(url) = web_sys::Url::create_object_url_with_blob(&blob) else { return };
84    if let Some(link) =
85        page.create_element("a").ok().and_then(|e| e.dyn_into::<web_sys::HtmlAnchorElement>().ok())
86    {
87        link.set_href(&url);
88        link.set_download(&file.name);
89        link.click();
90    }
91    web_sys::Url::revoke_object_url(&url).ok();
92}
93
94#[cfg(not(target_arch = "wasm32"))]
95fn open_on_desktop(file: &ExportedFile) {
96    let path = std::env::temp_dir().join(&file.name);
97    if let Err(e) = std::fs::write(&path, &file.bytes) {
98        tracing::warn!(error = %e, "could not write the exported file");
99        return;
100    }
101    if let Err(e) = ravenrs_platform_native::open_with_system(&path) {
102        tracing::warn!(error = %e, "could not open the file");
103    }
104}

106/// Asks the user for a PNG or JPEG and hands it over as a `data:` address. In
107/// the browser this is the file dialog; the desktop build has no file dialog
108/// yet, so a picture there comes from a field instead.
109pub fn pick_image(on_pick: impl FnOnce(String) + 'static) {
110    #[cfg(target_arch = "wasm32")]
111    pick_image_in_browser(on_pick);
112    #[cfg(not(target_arch = "wasm32"))]
113    {
114        let _ = on_pick;
115        tracing::warn!("choosing an image from a file is only available in the browser");
116    }
117}
118
119#[cfg(target_arch = "wasm32")]
120fn pick_image_in_browser(on_pick: impl FnOnce(String) + 'static) {
121    use wasm_bindgen::JsCast;
122    use wasm_bindgen::closure::Closure;
123
124    let Some(page) = web_sys::window().and_then(|w| w.document()) else { return };
125    let Some(input) =
126        page.create_element("input").ok().and_then(|e| e.dyn_into::<web_sys::HtmlInputElement>().ok())
127    else {
128        return;
129    };
130    input.set_type("file");
131    input.set_accept("image/png,image/jpeg");
132    let chosen = input.clone();
133    let on_change = Closure::once_into_js(move || {
134        let Some(file) = chosen.files().and_then(|files| files.get(0)) else { return };
135        let Ok(reader) = web_sys::FileReader::new() else { return };
136        let done = reader.clone();
137        let on_load = Closure::once_into_js(move || {
138            if let Some(source) = done.result().ok().and_then(|r| r.as_string()) {
139                on_pick(source);
140            }
141        });
142        reader.set_onload(Some(on_load.unchecked_ref()));
143        reader.read_as_data_url(&file).ok();
144    });
145    input.set_onchange(Some(on_change.unchecked_ref()));
146    input.click();
147}
