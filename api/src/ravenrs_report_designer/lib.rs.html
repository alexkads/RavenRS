107impl Designer {
108    /// A designer of this report inside an application, with no gallery or wizard: the layout is
109    /// what it is about, with the data embedded in it for the field list and the preview.
110    /// `on_save` gets the report as it stands, and `on_close` is for leaving.
111    pub fn embedded(
112        report: Report,
113        save_label: impl Into<String>,
114        close_label: impl Into<String>,
115        on_save: impl Fn(Report) + 'static,
116        on_close: impl Fn() + 'static,
117    ) -> Self {
118        let mut designer = Self::new();
119        designer.editor = Editor::new(report);
120        designer.editor.gallery_open.set(false);
121        designer.embedded = Some(Embedded {
122            save_label: save_label.into(),
123            close_label: close_label.into(),
124            on_save: Rc::new(on_save),
125            on_close: Rc::new(on_close),
126        });
127        designer
128    }
129
130    fn embedded_bar(&self) -> Option<AnyView> {
131        let embedded = self.embedded.clone()?;
132        let (editor, on_save, on_close) = (self.editor.clone(), embedded.on_save, embedded.on_close);
133        Some(Box::new(
134            HStack::new((
135                Button::new(embedded.save_label).size(ButtonSize::Small).on_click(move || {
136                    on_save(editor.report.get());
137                }),
138                Button::new(embedded.close_label)
139                    .style(ButtonStyle::Ghost)
140                    .size(ButtonSize::Small)
141                    .on_click(move || on_close()),
142            ))
143            .spacing(SPACING.s2)
144            .align(Align::Center)
145            .padding(SPACING.s2),
146        ))
147    }

491/// The data source's fields; drag one onto a band, or click to add it to the selected band.
492struct FieldList(Editor);
493
494impl View for FieldList {
495    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
496        let e = &self.0;
497        let report = e.report.get();
498        let source = report.data_source.clone().unwrap_or_else(|| "Dados".into());
499        let data = report.embedded_data().cloned().unwrap_or_default();
500        let mut sections: Vec<AnyView> = vec![Box::new(
501            HStack::new((
502                Icon::new(IconName::Database).size(14.0),
503                Text::new(source).font_weight(Weight::SEMIBOLD),
504            ))
505            .spacing(SPACING.s2)
506            .align(Align::Center),
507        )];
508        let fields = wizard::fields(&data, report.data_member.as_deref());
509        let mut rows: Vec<AnyView> = Vec::new();
510        for f in &fields {
511            rows.push(Box::new(FieldItem { editor: e.clone(), field: f.clone(), collection: None }));
512        }
513        for f in report.calculated_fields.iter() {
514            rows.push(Box::new(FieldItem {
515                editor: e.clone(),
516                field: FieldInfo { path: f.name.clone(), kind: FieldKind::Number },
517                collection: None,
518            }));
519        }
520        // Fields of nested collections (invoice items) bind inside their detail band.
521        for f in fields.iter().filter(|f| f.kind == FieldKind::Collection) {
522            let items = wizard::rows_of(&data, report.data_member.as_deref())
523                .first()
524                .and_then(|row| row.get(&f.path))
525                .cloned()
526                .unwrap_or_default();
527            for sub in wizard::fields(&items, None) {
528                rows.push(Box::new(FieldItem {
529                    editor: e.clone(),
530                    field: sub,
531                    collection: Some(f.path.clone()),
532                }));
533            }
534        }
535        sections.push(Box::new(VStack::new(rows).spacing(SPACING.s1).align(Align::Stretch)));
536        sections.push(Box::new(
537            Text::new("Arraste um campo até uma banda, ou clique para inserir na banda selecionada.")
538                .style(TextStyle::Callout)
539                .foreground(Semantic::InkMuted),
540        ));
541        cx.child(&VStack::new(sections).spacing(SPACING.s3).align(Align::Stretch))
542    }
543}

545struct FieldItem {
546    editor: Editor,
547    field: FieldInfo,
548    collection: Option<String>,
549}
550
551impl View for FieldItem {
552    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
553        let icon = match self.field.kind {
554            FieldKind::Number => IconName::Sigma,
555            FieldKind::Date => IconName::Calendar,
556            FieldKind::Boolean => IconName::CircleCheck,
557            FieldKind::Collection => IconName::List,
558            _ => IconName::Type,
559        };
560        let label = match &self.collection {
561            Some(c) => format!("{c} › {}", self.field.path),
562            None => self.field.path.clone(),
563        };
564        let item = SidebarItem::new(icon, label).compact();
565        if self.field.kind == FieldKind::Collection {
566            return cx.child(&item);
567        }
568        let (kind, path, collection) = (self.field.kind, self.field.path.clone(), self.collection.clone());
569        // A click, or Enter while focused: into the selected band, or the band that lists this collection.
570        let dragged = Rc::new(std::cell::Cell::new(false));
571        let (e, was_dragged) = (self.editor.clone(), dragged.clone());
572        let add_path = path.clone();
573        let item = item.on_click(move || {
574            if was_dragged.replace(false) {
575                return;
576            }
577            let band = match &collection {
578                Some(c) => e
579                    .report
580                    .with(|r| r.detail_reports.iter().position(|d| &d.data_member == c))
581                    .map_or(e.target_band(), BandRef::DetailReportDetail),
582                None => e.target_band(),
583            };
584            let (x, y) = e.free_spot(band, 18.0);
585            e.add_control(band, field_control(&add_path, kind).at(x, y, 140.0, 18.0));
586            e.tab.set(0);
587        });
588        let row = cx.child(&item);
589        let e = self.editor.clone();
590        cx.on_drag(
591            row,
592            Rc::new(move |d| match d.phase {
593                DragPhase::Start => {
594                    e.drag.set(Some(FieldDrag { path: path.clone(), start: d.position, at: d.position }))
595                }
596                DragPhase::Move => e.drag.update(|g| {
597                    if let Some(g) = g {
598                        g.at = d.position;
599                    }
600                }),
601                _ => {
602                    let Some(g) = e.drag.get() else { return };
603                    e.drag.set(None);
604                    if (g.at.x - g.start.x).hypot(g.at.y - g.start.y) <= 4.0 {
605                        return;
606                    }
607                    // The pointer may come up on this same row: that is no click.
608                    dragged.set(true);
609                    if let Some((band, at)) = e.band_at(d.position) {
610                        dragged.set(false);
611                        let c = field_control(&path, kind).at(
612                            editor::snap(at.x.max(0.0)),
613                            editor::snap((at.y - 9.0).max(0.0)),
614                            140.0,
615                            18.0,
616                        );
617                        e.add_control(band, c);
618                        e.tab.set(0);
619                    }
620                }
621            }),
622        );
623        row
624    }

627/// A new field control with a sensible format for its kind.
628fn field_control(path: &str, kind: FieldKind) -> Control {
629    let c = Control::field(format!("[{path}]"));
630    match kind {
631        FieldKind::Date => c.format("dd/MM/yyyy"),
632        FieldKind::Number
633            if path.to_lowercase().contains("price") || path.to_lowercase().contains("total") =>
634        {
635            c.format("C2").style(Style::new().align(ravenrs_report::TextAlign::Right))
636        }
637        FieldKind::Number => c.style(Style::new().align(ravenrs_report::TextAlign::Right)),
638        _ => c,
639    }
640}
