12/// What is selected on the design surface.
13#[derive(Clone, Copy, Debug, PartialEq)]
14pub enum Selection {
15    None,
16    Band(BandRef),
17    Control(BandRef, usize),
18}

20/// A field being dragged from the field list.
21#[derive(Clone, Debug, PartialEq)]
22pub struct FieldDrag {

24    pub start: Point,
25    pub at: Point,

28/// An expression the expression editor can open.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum ExpressionTarget {
31    /// The selected control's expression: a field, a barcode's or a bookmark's.
32    Control,
33    /// The report's filter.
34    Filter,
35    /// What a group is grouped by.
36    Group(usize),
37}

39/// Everything panels share. Cheap to clone.
40#[derive(Clone)]
41pub struct Editor {
42    pub report: State<Report>,
43    pub selection: State<Selection>,
44    pub tab: State<usize>,
45    pub zoom: State<f32>,
46    pub version: State<u64>,
47    pub page: State<usize>,
48    pub gallery_open: State<bool>,
49    pub wizard_open: State<bool>,
50    pub drag: State<Option<FieldDrag>>,
51    /// The control last copied or cut, for Paste.
52    pub copied: State<Option<Control>>,
53    /// The expression open in the expression editor, and the text being edited.
54    pub expression: State<Option<ExpressionTarget>>,
55    pub expression_draft: State<String>,
56    pub engine: Rc<RefCell<ReportEngine>>,
57    /// Frames of the band bodies in the last paint, for drops.
58    pub band_frames: Rc<RefCell<Vec<(BandRef, Rect)>>>,
59    history: Rc<RefCell<(Vec<Report>, Vec<Report>)>>,
60    /// The kind of the last edit, so typing in one field is one undo step.
61    last_edit: Rc<Cell<Option<&'static str>>>,
62}
63
64impl Editor {
65    pub fn new(report: Report) -> Self {
66        Self {
67            report: State::new(report),
68            selection: State::new(Selection::None),
69            tab: State::new(0),
70            zoom: State::new(1.25),
71            version: State::new(0),
72            page: State::new(0),
73            gallery_open: State::new(true),
74            wizard_open: State::new(false),
75            drag: State::new(None),
76            copied: State::new(None),
77            expression: State::new(None),
78            expression_draft: State::new(String::new()),
79            engine: Rc::new(RefCell::new(ReportEngine::new())),
80            band_frames: Rc::new(RefCell::new(Vec::new())),
81            history: Rc::new(RefCell::new((Vec::new(), Vec::new()))),
82            last_edit: Rc::new(Cell::new(None)),
83        }
84    }
85
86    /// Changes the report as one undoable step.
87    pub fn edit(&self, f: impl FnOnce(&mut Report)) {
88        self.last_edit.set(None);
89        self.checkpoint();
90        self.apply(f);
91    }
92
93    /// Changes the report; consecutive edits of the same `kind` (typing in one
94    /// property) form one undo step.
95    pub fn edit_as(&self, kind: &'static str, f: impl FnOnce(&mut Report)) {
96        if self.last_edit.get() != Some(kind) {
97            self.checkpoint();
98            self.last_edit.set(Some(kind));
99        }
100        self.apply(f);
101    }
102
103    /// Records the current report for undo (start of a drag).
104    pub fn checkpoint(&self) {
105        let mut h = self.history.borrow_mut();
106        h.0.push(self.report.get());
107        if h.0.len() > 200 {
108            h.0.remove(0);
109        }
110        h.1.clear();
111    }
112
113    /// Changes the report without recording an undo step (during a drag).
114    pub fn apply(&self, f: impl FnOnce(&mut Report)) {
115        self.report.update(f);
116        self.version.update(|v| *v += 1);
117    }
118
119    /// Replaces the whole report (a template, a loaded file).
120    pub fn load(&self, report: Report) {
121        self.checkpoint();
122        self.last_edit.set(None);
123        self.report.set(report);
124        self.selection.set(Selection::None);
125        self.page.set(0);
126        self.version.update(|v| *v += 1);
127    }
128
129    pub fn undo(&self) {
130        let previous = self.history.borrow_mut().0.pop();
131        if let Some(prev) = previous {
132            let current = self.report.get();
133            self.history.borrow_mut().1.push(current);
134            self.report.set(prev);
135            self.after_history();
136        }
137    }
138
139    pub fn redo(&self) {
140        let next = self.history.borrow_mut().1.pop();
141        if let Some(next) = next {
142            let current = self.report.get();
143            self.history.borrow_mut().0.push(current);
144            self.report.set(next);
145            self.after_history();
146        }
147    }
148
149    fn after_history(&self) {
150        self.last_edit.set(None);
151        self.version.update(|v| *v += 1);
152        // The selection may point past the end of a band now.
153        let valid = self.report.with(|r| match self.selection.get() {
154            Selection::Control(b, i) => b.get(r).is_some_and(|band| i < band.controls.len()),
155            Selection::Band(b) => b.get(r).is_some(),
156            Selection::None => true,
157        });
158        if !valid {
159            self.selection.set(Selection::None);
160        }
161    }
162
163    /// The selected control, if any.
164    pub fn selected_control(&self) -> Option<(BandRef, usize, Control)> {
165        match self.selection.get() {
166            Selection::Control(b, i) => self
167                .report
168                .with(|r| b.get(r).and_then(|band| band.controls.get(i).cloned()))
169                .map(|c| (b, i, c)),
170            _ => None,
171        }
172    }
173
174    /// Changes the selected control.
175    pub fn edit_control(&self, kind: &'static str, f: impl FnOnce(&mut Control)) {
176        if let Selection::Control(b, i) = self.selection.get() {
177            self.edit_as(kind, |r| {
178                if let Some(c) = b.get_mut(r).and_then(|band| band.controls.get_mut(i)) {
179                    f(c);
180                }
181            });
182        }
183    }
184
185    /// Adds a control to `band` and selects it.
186    pub fn add_control(&self, band: BandRef, control: Control) {
187        let mut index = None;
188        self.edit(|r| {
189            if let Some(b) = band.get_mut(r) {
190                // Grow the band if the control would stick out of it.
191                b.height = b.height.max(control.y + control.height);
192                b.controls.push(control);
193                index = Some(b.controls.len() - 1);
194            }
195        });
196        if let Some(i) = index {
197            self.selection.set(Selection::Control(band, i));
198        }
199    }
200
201    /// Deletes the selected control.
202    pub fn delete_selected(&self) {
203        if let Selection::Control(b, i) = self.selection.get() {
204            self.edit(|r| {
205                if let Some(band) = b.get_mut(r)
206                    && i < band.controls.len()
207                {
208                    band.controls.remove(i);
209                }
210            });
211            self.selection.set(Selection::Band(b));
212        }
213    }
214
215    /// The text of an expression, empty if there is none.
216    pub fn expression_of(&self, target: ExpressionTarget) -> String {
217        match target {
218            ExpressionTarget::Control => self
219                .selected_control()
220                .and_then(|(_, _, c)| match c.kind {
221                    ControlKind::Label(l) => l.expression,
222                    ControlKind::Barcode(b) => b.expression,
223                    ControlKind::Bookmark(b) => b.expression,
224                    _ => None,
225                })
226                .unwrap_or_default(),
227            ExpressionTarget::Filter => self.report.with(|r| r.filter.clone().unwrap_or_default()),
228            ExpressionTarget::Group(i) => {
229                self.report.with(|r| r.groups.get(i).map(|g| g.expression.clone()).unwrap_or_default())
230            }
231        }
232    }
233
234    /// Whether the selected control has an expression to edit.
235    pub fn control_has_expression(&self) -> bool {
236        self.selected_control().is_some_and(|(_, _, c)| match c.kind {
237            ControlKind::Label(l) => l.expression.is_some(),
238            ControlKind::Barcode(b) => b.expression.is_some(),
239            ControlKind::Bookmark(b) => b.expression.is_some(),
240            _ => false,
241        })
242    }
243
244    /// Opens the expression editor on `target`.
245    pub fn open_expression(&self, target: ExpressionTarget) {
246        self.expression_draft.set(self.expression_of(target));
247        self.expression.set(Some(target));
248    }
249
250    /// Writes the expression being edited and closes the editor.
251    pub fn apply_expression(&self) {
252        let Some(target) = self.expression.get() else { return };
253        let text = self.expression_draft.get();
254        match target {
255            ExpressionTarget::Control => self.edit_control("expression", |c| match &mut c.kind {
256                ControlKind::Label(l) => l.expression = Some(text),
257                ControlKind::Barcode(b) => b.expression = Some(text),
258                ControlKind::Bookmark(b) => b.expression = Some(text),
259                _ => {}
260            }),
261            ExpressionTarget::Filter => {
262                self.edit(|r| r.filter = (!text.trim().is_empty()).then_some(text));
263            }
264            ExpressionTarget::Group(i) => self.edit(|r| {
265                if let Some(g) = r.groups.get_mut(i) {
266                    g.expression = text;
267                }
268            }),
269        }
270        self.expression.set(None);
271    }
272
273    /// Keeps a copy of the selected control for Paste.
274    pub fn copy_selected(&self) {
275        if let Some((_, _, c)) = self.selected_control() {
276            self.copied.set(Some(c));
277        }
278    }
279
280    /// Copies the selected control and deletes it.
281    pub fn cut_selected(&self) {
282        if self.selected_control().is_some() {
283            self.copy_selected();
284            self.delete_selected();
285        }
286    }

288    /// Adds the copied control to the band in use, a little below and to the right
289    /// of where it was, or at a free spot if it came from another band.
290    pub fn paste(&self) {
291        let Some(mut c) = self.copied.get() else { return };
292        let band = self.target_band();
293        let width = self.report.with(|r| r.page.content_width());
294        c.x = snap((c.x + 8.0).min((width - c.width).max(0.0)));
295        c.y = snap(c.y + 8.0);
296        self.add_control(band, c.clone());
297        // The next paste lands below this one, as in a stack of copies.
298        self.copied.set(Some(c));
299    }
300
301    /// Moves the selected control over the others in its band (`front`), or under them.
302    pub fn reorder_selected(&self, front: bool) {
303        let Selection::Control(b, i) = self.selection.get() else { return };
304        let mut to = None;
305        self.edit(|r| {
306            if let Some(band) = b.get_mut(r)
307                && i < band.controls.len()
308            {
309                let c = band.controls.remove(i);
310                let at = if front { band.controls.len() } else { 0 };
311                band.controls.insert(at, c);
312                to = Some(at);
313            }
314        });
315        if let Some(at) = to {
316            self.selection.set(Selection::Control(b, at));
317        }
318    }
319
320    /// The band to insert new controls into: the selected one, else the detail.
321    pub fn target_band(&self) -> BandRef {
322        match self.selection.get() {
323            Selection::Band(b) | Selection::Control(b, _) => b,
324            Selection::None => BandRef::Detail,
325        }
326    }
327
328    /// A free spot for a new control in `band`: below its lowest control.
329    pub fn free_spot(&self, band: BandRef, height: f32) -> (f32, f32) {
330        self.report.with(|r| {
331            let Some(b) = band.get(r) else { return (0.0, 0.0) };
332            // Next to the right-most control on the first row, if there is room.
333            let width = r.page.content_width();
334            let right = b.controls.iter().filter(|c| c.y < height).map(|c| c.x + c.width).fold(0.0, f32::max);
335            if right + 110.0 <= width {
336                (right, 0.0)
337            } else {
338                (0.0, b.controls.iter().map(|c| c.y + c.height).fold(0.0, f32::max))
339            }
340        })
341    }
342
343    /// The band under `point` (logical pixels), and the point in band coordinates.
344    pub fn band_at(&self, point: Point) -> Option<(BandRef, Point)> {
345        let zoom = self.zoom.get();
346        self.band_frames
347            .borrow()
348            .iter()
349            .find(|(_, f)| f.contains(point))
350            .map(|(b, f)| (*b, Point::new((point.x - f.x()) / zoom, (point.y - f.y()) / zoom)))
351    }
352}

354/// Rounds to the 2-point design grid.
355pub fn snap(v: f32) -> f32 {
356    (v / 2.0).round() * 2.0
