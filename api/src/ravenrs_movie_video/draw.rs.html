1//! Draws a view into pixels, with no window.
2
3use ravenrs_core::Size;
4use ravenrs_render::{GpuContext, Renderer, wgpu};
5use ravenrs_style::Theme;
6use ravenrs_widgets::{Ui, View};
7
8const FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::Rgba8Unorm;
9
10/// A GPU, a renderer and a texture of the size of the video.
11pub(crate) struct Canvas {
12    gpu: GpuContext,
13    renderer: Renderer,
14    texture: wgpu::Texture,
15    buffer: wgpu::Buffer,
16    padded: u32,
17    size: (u32, u32),
18}
19
20impl Canvas {
21    pub(crate) fn new(size: (u32, u32)) -> Option<Self> {
22        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::new_without_display_handle_from_env());
23        let gpu = pollster::block_on(GpuContext::new(&instance, None)).ok()?;
24        let renderer = Renderer::new(&gpu.device, &gpu.queue, FORMAT);
25        let texture = gpu.device.create_texture(&wgpu::TextureDescriptor {
26            label: Some("video"),
27            size: wgpu::Extent3d { width: size.0, height: size.1, depth_or_array_layers: 1 },
28            mip_level_count: 1,
29            sample_count: 1,
30            dimension: wgpu::TextureDimension::D2,
31            format: FORMAT,
32            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::COPY_SRC,
33            view_formats: &[],
34        });
35        let padded =
36            (size.0 * 4).div_ceil(wgpu::COPY_BYTES_PER_ROW_ALIGNMENT) * wgpu::COPY_BYTES_PER_ROW_ALIGNMENT;
37        let buffer = gpu.device.create_buffer(&wgpu::BufferDescriptor {
38            label: Some("video readback"),
39            size: u64::from(padded * size.1),
40            usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
41            mapped_at_creation: false,
42        });
43        Some(Self { gpu, renderer, texture, buffer, padded, size })
44    }
45
46    /// What `ui` draws now, as rows of RGBA.
47    pub(crate) fn draw(&mut self, ui: &Ui, scale: f32) -> Vec<u8> {
48        let (w, h) = self.size;
49        let view = self.texture.create_view(&wgpu::TextureViewDescriptor::default());
50        self.renderer.render(ui.scene(), &view, (w, h), scale);
51        let mut encoder = self.gpu.device.create_command_encoder(&wgpu::CommandEncoderDescriptor::default());
52        encoder.copy_texture_to_buffer(
53            self.texture.as_image_copy(),
54            wgpu::TexelCopyBufferInfo {
55                buffer: &self.buffer,
56                layout: wgpu::TexelCopyBufferLayout {
57                    offset: 0,
58                    bytes_per_row: Some(self.padded),
59                    rows_per_image: Some(h),
60                },
61            },
62            wgpu::Extent3d { width: w, height: h, depth_or_array_layers: 1 },
63        );
64        self.gpu.queue.submit(Some(encoder.finish()));
65        let slice = self.buffer.slice(..);
66        slice.map_async(wgpu::MapMode::Read, |_| {});
67        let _ = self.gpu.device.poll(wgpu::PollType::wait_indefinitely());
68        let mut pixels = Vec::with_capacity((w * h * 4) as usize);
69        if let Ok(data) = slice.get_mapped_range() {
70            for row in 0..h {
71                let start = (row * self.padded) as usize;
72                pixels.extend_from_slice(&data[start..start + (w * 4) as usize]);
73            }
74        }
75        self.buffer.unmap();
76        pixels
77    }
78}

80/// A Ui for `root` on a screen of `logical` size.
81pub(crate) fn ui_for(root: impl View + 'static, theme: Theme, logical: Size, scale: f32) -> Ui {
82    let mut ui = Ui::new(root, theme);
83    ui.set_viewport(logical, scale);
84    ui.set_time(0.0);
85    ui.update();
86    ui
