1//! A QuickTime file with one H.264 picture track and one track of plain sound, the
2//! simplest container that video sites, players and editors all read.
3
4use std::io::{self, Seek, SeekFrom, Write};
5
6/// One picture, as length-prefixed NAL units.
7pub(crate) struct Picture {
8    pub(crate) offset: u64,
9    pub(crate) size: u32,
10    pub(crate) key: bool,
11}
12
13/// Writes the pictures and the sound as they come, and the index at the end.
14pub(crate) struct MovWriter<W: Write + Seek> {
15    out: W,
16    mdat_start: u64,
17    pictures: Vec<Picture>,
18    /// Where each piece of sound went and how many samples it holds.
19    sound: Vec<(u64, u32)>,
20    sps: Vec<u8>,
21    pps: Vec<u8>,
22}
23
24impl<W: Write + Seek> MovWriter<W> {
25    pub(crate) fn new(mut out: W) -> io::Result<Self> {
26        let mut ftyp = Vec::new();
27        ftyp.extend_from_slice(b"qt  ");
28        ftyp.extend_from_slice(&0x2005_0300_u32.to_be_bytes());
29        ftyp.extend_from_slice(b"qt  ");
30        write_box(&mut out, b"ftyp", &ftyp)?;
31        let mdat_start = out.stream_position()?;
32        // A 64-bit size, filled in at the end.
33        out.write_all(&1_u32.to_be_bytes())?;
34        out.write_all(b"mdat")?;
35        out.write_all(&0_u64.to_be_bytes())?;
36        Ok(Self {
37            out,
38            mdat_start,
39            pictures: Vec::new(),
40            sound: Vec::new(),
41            sps: Vec::new(),
42            pps: Vec::new(),
43        })
44    }
45
46    /// Adds a picture from its NAL units, without their start codes. The parameter
47    /// sets go to the index, the rest to the file.
48    pub(crate) fn picture(&mut self, nals: &[&[u8]]) -> io::Result<()> {
49        let offset = self.out.stream_position()?;
50        let (mut size, mut key) = (0_u32, false);
51        for nal in nals {
52            let Some(header) = nal.first() else { continue };
53            match header & 0x1F {
54                7 => self.sps = nal.to_vec(),
55                8 => self.pps = nal.to_vec(),
56                kind => {
57                    key |= kind == 5;
58                    self.out.write_all(&(nal.len() as u32).to_be_bytes())?;
59                    self.out.write_all(nal)?;
60                    size += 4 + nal.len() as u32;
61                }
62            }
63        }
64        self.pictures.push(Picture { offset, size, key });
65        Ok(())
66    }
67
68    /// Adds a piece of sound, 16-bit little endian, one channel.
69    pub(crate) fn sound(&mut self, samples: &[i16]) -> io::Result<()> {
70        if samples.is_empty() {
71            return Ok(());
72        }
73        let offset = self.out.stream_position()?;
74        let mut bytes = Vec::with_capacity(samples.len() * 2);
75        for sample in samples {
76            bytes.extend_from_slice(&sample.to_le_bytes());
77        }
78        self.out.write_all(&bytes)?;
79        self.sound.push((offset, samples.len() as u32));
80        Ok(())
81    }
82
83    /// Writes the index: `size` of the pictures, `fps` of them a second, sound at `rate`.
84    pub(crate) fn finish(mut self, size: (u32, u32), fps: u32, rate: u32) -> io::Result<W> {
85        let end = self.out.stream_position()?;
86        self.out.seek(SeekFrom::Start(self.mdat_start + 8))?;
87        self.out.write_all(&(end - self.mdat_start).to_be_bytes())?;
88        self.out.seek(SeekFrom::Start(end))?;
89        let frames = self.pictures.len() as u64;
90        let sound_samples: u64 = self.sound.iter().map(|(_, n)| u64::from(*n)).sum();
91        let movie_ms = (frames * 1000).div_ceil(u64::from(fps.max(1)));
92        let mut moov = Vec::new();
93        moov.extend(boxed(b"mvhd", &mvhd(movie_ms, 3)));
94        moov.extend(self.video_track(size, fps, movie_ms));
95        if sound_samples > 0 {
96            moov.extend(self.sound_track(rate, sound_samples));
97        }
98        write_box(&mut self.out, b"moov", &moov)?;
99        Ok(self.out)
100    }

102    fn video_track(&self, (width, height): (u32, u32), fps: u32, movie_ms: u64) -> Vec<u8> {
103        let scale = fps * 100;
104        let frames = self.pictures.len() as u32;
105        let mut entry = Vec::new();
106        entry.extend_from_slice(&[0; 6]);
107        entry.extend_from_slice(&1_u16.to_be_bytes());
108        entry.extend_from_slice(&[0; 16]);
109        entry.extend_from_slice(&(width as u16).to_be_bytes());
110        entry.extend_from_slice(&(height as u16).to_be_bytes());
111        entry.extend_from_slice(&0x0048_0000_u32.to_be_bytes());
112        entry.extend_from_slice(&0x0048_0000_u32.to_be_bytes());
113        entry.extend_from_slice(&0_u32.to_be_bytes());
114        entry.extend_from_slice(&1_u16.to_be_bytes());
115        let mut name = [0_u8; 32];
116        let label = b"H.264";
117        name[0] = label.len() as u8;
118        name[1..=label.len()].copy_from_slice(label);
119        entry.extend_from_slice(&name);
120        entry.extend_from_slice(&24_u16.to_be_bytes());
121        entry.extend_from_slice(&(-1_i16).to_be_bytes());
122        entry.extend(boxed(b"avcC", &self.avcc()));
123        let stsd = full(0, 0, &[&1_u32.to_be_bytes()[..], &boxed(b"avc1", &entry)].concat());
124        let stts = full(0, 0, &[1_u32.to_be_bytes(), frames.to_be_bytes(), 100_u32.to_be_bytes()].concat());
125        let keys: Vec<u32> =
126            self.pictures.iter().enumerate().filter(|(_, p)| p.key).map(|(i, _)| i as u32 + 1).collect();
127        let mut stss = (keys.len() as u32).to_be_bytes().to_vec();
128        keys.iter().for_each(|k| stss.extend_from_slice(&k.to_be_bytes()));
129        let stsc = full(
130            0,
131            0,
132            &[1_u32.to_be_bytes(), 1_u32.to_be_bytes(), 1_u32.to_be_bytes(), 1_u32.to_be_bytes()].concat(),
133        );
134        let mut stsz = [0_u32.to_be_bytes(), frames.to_be_bytes()].concat();
135        self.pictures.iter().for_each(|p| stsz.extend_from_slice(&p.size.to_be_bytes()));
136        let offsets: Vec<u64> = self.pictures.iter().map(|p| p.offset).collect();
137        let stbl = [
138            boxed(b"stsd", &stsd),
139            boxed(b"stts", &stts),
140            boxed(b"stss", &full(0, 0, &stss)),
141            boxed(b"stsc", &stsc),
142            boxed(b"stsz", &full(0, 0, &stsz)),
143            chunk_offsets(&offsets),
144        ]
145        .concat();
146        let vmhd = boxed(b"vmhd", &full(0, 1, &[0_u8; 8]));
147        let duration = u64::from(frames) * 100;
148        track(1, (width, height), movie_ms, scale, duration, b"vide", vmhd, stbl)
149    }

151    fn sound_track(&self, rate: u32, samples: u64) -> Vec<u8> {
152        let mut entry = Vec::new();
153        entry.extend_from_slice(&[0; 6]);
154        entry.extend_from_slice(&1_u16.to_be_bytes());
155        // Version 0 of the sound description: one channel of 16 bits.
156        entry.extend_from_slice(&[0; 8]);
157        entry.extend_from_slice(&1_u16.to_be_bytes());
158        entry.extend_from_slice(&16_u16.to_be_bytes());
159        entry.extend_from_slice(&0_u16.to_be_bytes());
160        entry.extend_from_slice(&0_u16.to_be_bytes());
161        entry.extend_from_slice(&(rate << 16).to_be_bytes());
162        let stsd = full(0, 0, &[&1_u32.to_be_bytes()[..], &boxed(b"sowt", &entry)].concat());
163        let stts =
164            full(0, 0, &[1_u32.to_be_bytes(), (samples as u32).to_be_bytes(), 1_u32.to_be_bytes()].concat());
165        let mut runs: Vec<(u32, u32)> = Vec::new();
166        for (index, (_, count)) in self.sound.iter().enumerate() {
167            if runs.last().is_none_or(|(_, last)| last != count) {
168                runs.push((index as u32 + 1, *count));
169            }
170        }
171        let mut stsc = (runs.len() as u32).to_be_bytes().to_vec();
172        for (first, count) in runs {
173            stsc.extend_from_slice(&first.to_be_bytes());
174            stsc.extend_from_slice(&count.to_be_bytes());
175            stsc.extend_from_slice(&1_u32.to_be_bytes());
176        }
177        let stsz = full(0, 0, &[2_u32.to_be_bytes(), (samples as u32).to_be_bytes()].concat());
178        let offsets: Vec<u64> = self.sound.iter().map(|(o, _)| *o).collect();
179        let stbl = [
180            boxed(b"stsd", &stsd),
181            boxed(b"stts", &stts),
182            boxed(b"stsc", &full(0, 0, &stsc)),
183            boxed(b"stsz", &stsz),
184            chunk_offsets(&offsets),
185        ]
186        .concat();
187        let smhd = boxed(b"smhd", &full(0, 0, &[0_u8; 4]));
188        let movie_ms = samples * 1000 / u64::from(rate.max(1));
189        track(2, (0, 0), movie_ms, rate, samples, b"soun", smhd, stbl)
190    }

192    fn avcc(&self) -> Vec<u8> {
193        let sps = &self.sps;
194        let mut out = vec![
195            1,
196            *sps.get(1).unwrap_or(&66),
197            *sps.get(2).unwrap_or(&0),
198            *sps.get(3).unwrap_or(&40),
199            0xFF,
200            0xE1,
201        ];
202        out.extend_from_slice(&(sps.len() as u16).to_be_bytes());
203        out.extend_from_slice(sps);
204        out.push(1);
205        out.extend_from_slice(&(self.pps.len() as u16).to_be_bytes());
206        out.extend_from_slice(&self.pps);
207        out
208    }
209}

211fn chunk_offsets(offsets: &[u64]) -> Vec<u8> {
212    let large = offsets.iter().any(|o| *o > u64::from(u32::MAX));
213    let mut body = (offsets.len() as u32).to_be_bytes().to_vec();
214    for offset in offsets {
215        if large {
216            body.extend_from_slice(&offset.to_be_bytes());
217        } else {
218            body.extend_from_slice(&(*offset as u32).to_be_bytes());
219        }
220    }
221    boxed(if large { b"co64" } else { b"stco" }, &full(0, 0, &body))
222}

224#[allow(clippy::too_many_arguments, reason = "the fields of one track")]
225fn track(
226    id: u32,
227    (width, height): (u32, u32),
228    movie_ms: u64,
229    scale: u32,
230    duration: u64,
231    kind: &[u8; 4],
232    header: Vec<u8>,
233    stbl: Vec<u8>,
234) -> Vec<u8> {
235    let mut tkhd = Vec::new();
236    tkhd.extend_from_slice(&[0; 8]);
237    tkhd.extend_from_slice(&id.to_be_bytes());
238    tkhd.extend_from_slice(&[0; 4]);
239    tkhd.extend_from_slice(&(movie_ms as u32).to_be_bytes());
240    tkhd.extend_from_slice(&[0; 8]);
241    tkhd.extend_from_slice(&0_u16.to_be_bytes());
242    tkhd.extend_from_slice(&0_u16.to_be_bytes());
243    tkhd.extend_from_slice(&(if kind == b"soun" { 0x0100_u16 } else { 0 }).to_be_bytes());
244    tkhd.extend_from_slice(&[0; 2]);
245    tkhd.extend(matrix());
246    tkhd.extend_from_slice(&(width << 16).to_be_bytes());
247    tkhd.extend_from_slice(&(height << 16).to_be_bytes());
248    let mut mdhd = Vec::new();
249    mdhd.extend_from_slice(&[0; 8]);
250    mdhd.extend_from_slice(&scale.to_be_bytes());
251    mdhd.extend_from_slice(&(duration as u32).to_be_bytes());
252    mdhd.extend_from_slice(&0_u16.to_be_bytes());
253    mdhd.extend_from_slice(&0_u16.to_be_bytes());
254    let handler = |class: &[u8; 4], sub: &[u8; 4], name: &str| {
255        let mut body = Vec::new();
256        body.extend_from_slice(class);
257        body.extend_from_slice(sub);
258        body.extend_from_slice(&[0; 12]);
259        body.push(name.len() as u8);
260        body.extend_from_slice(name.as_bytes());
261        boxed(b"hdlr", &full(0, 0, &body))
262    };
263    let dref = full(0, 0, &[&1_u32.to_be_bytes()[..], &boxed(b"alis", &full(0, 1, &[]))].concat());
264    let dinf = boxed(b"dinf", &boxed(b"dref", &dref));
265    let minf = [header, handler(b"dhlr", b"alis", "RavenRS"), dinf, boxed(b"stbl", &stbl)].concat();
266    let mdia = [boxed(b"mdhd", &full(0, 0, &mdhd)), handler(b"mhlr", kind, "RavenRS"), boxed(b"minf", &minf)]
267        .concat();
268    boxed(b"trak", &[boxed(b"tkhd", &full(0, 0x3, &tkhd)), boxed(b"mdia", &mdia)].concat())
269}

271fn mvhd(duration_ms: u64, next_track: u32) -> Vec<u8> {
272    let mut body = Vec::new();
273    body.extend_from_slice(&[0; 8]);
274    body.extend_from_slice(&1000_u32.to_be_bytes());
275    body.extend_from_slice(&(duration_ms as u32).to_be_bytes());
276    body.extend_from_slice(&0x0001_0000_u32.to_be_bytes());
277    body.extend_from_slice(&0x0100_u16.to_be_bytes());
278    body.extend_from_slice(&[0; 10]);
279    body.extend(matrix());
280    body.extend_from_slice(&[0; 24]);
281    body.extend_from_slice(&next_track.to_be_bytes());
282    full(0, 0, &body)
283}

285fn matrix() -> Vec<u8> {
286    [0x0001_0000_u32, 0, 0, 0, 0x0001_0000, 0, 0, 0, 0x4000_0000]
287        .iter()
288        .flat_map(|v| v.to_be_bytes())
289        .collect()
290}

292fn full(version: u8, flags: u32, body: &[u8]) -> Vec<u8> {
293    let mut out = vec![version];
294    out.extend_from_slice(&flags.to_be_bytes()[1..]);
295    out.extend_from_slice(body);
296    out
297}
298
299fn boxed(kind: &[u8; 4], body: &[u8]) -> Vec<u8> {
300    let mut out = ((body.len() + 8) as u32).to_be_bytes().to_vec();
301    out.extend_from_slice(kind);
302    out.extend_from_slice(body);
303    out
304}
305
306fn write_box(out: &mut impl Write, kind: &[u8; 4], body: &[u8]) -> io::Result<()> {
307    out.write_all(&boxed(kind, body))
