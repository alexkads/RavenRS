3use std::collections::BTreeMap;
4
5use ravenrs_app::{FrameContext, ViewInfo, ViewKind};
6use ravenrs_reactive::State;
7use ravenrs_widgets::{AnyView, BuildCx, Keyed, NodeId, Text, Tile, TileLayout, TileSpot, View, ViewExt};
8
9use crate::{ListView, Session};
10
11/// How many cells wide a piece starts, and how tall.
12const ITEM_COLUMNS: u16 = 2;
13const ITEM_ROWS: u16 = 2;
14
15/// Several views together, each in a frame of its own, as the model says. A dashboard is bound to
16/// no records, and people can move and resize its pieces.
17#[derive(Debug)]
18pub struct DashboardView {
19    session: Session,
20    view: ViewInfo,
21}
22
23impl DashboardView {
24    /// The dashboard of this view in the model.
25    pub fn new(session: Session, view: ViewInfo) -> Self {
26        Self { session, view }
27    }
28}
29
30impl View for DashboardView {
31    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
32        // The one people changed it to, starting over, since pieces are kept by position.
33        self.session.revision();
34        let view = self.session.dashboard_shown(&self.view.id);
35        let key = view.id.bytes().fold(0_u64, |h, b| (h ^ u64::from(b)).wrapping_mul(0x100_0000_01b3));
36        cx.child(&Keyed::new(key, DashboardItems { session: self.session.clone(), view }))
37    }
38}
39
40/// The pieces of one dashboard.
41struct DashboardItems {
42    session: Session,
43    view: ViewInfo,
44}
45
46impl View for DashboardItems {
47    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
48        let model = self.session.model();
49        let Some(items) = model.at(&["Views", &self.view.id, "Items"]) else {
50            return cx.child(&Text::new(self.view.id.clone()));
51        };
52        let order = items.in_order();
53        let frames: State<BTreeMap<String, Session>> = cx.use_state(BTreeMap::new);
54        let spots: State<Vec<TileSpot>> = cx.use_state(|| {
55            order.iter().map(|item| TileSpot::new(item.id.clone(), ITEM_COLUMNS, ITEM_ROWS)).collect()
56        });
57        let alone = order.len() == 1;
58        let mut tiles = TileLayout::new(spots.binding());
59        for item in order {
60            let caption = item.text("Caption").unwrap_or(&item.id).to_owned();
61            let content: AnyView = if let Some(name) = item.text("Control") {
62                match self.session.control(name) {
63                    Some(control) => control(),
64                    None => Box::new(Text::new(name.to_owned())),
65                }
66            } else if let Some(info) = item.text("View").and_then(|id| ViewInfo::from_model(&model, id).ok())
67            {
68                let kept = frames.peek().get(&item.id).cloned();
69                let session = kept.unwrap_or_else(|| {
70                    let session = self.session.child(FrameContext::Nested);
71                    session.frame().set_view(Some(info.clone()));
72                    frames.update(|all| {
73                        all.insert(item.id.clone(), session.clone());
74                    });
75                    session
76                });
77                match info.kind {
78                    ViewKind::List => {
79                        let opener = self.session.clone();
80                        let entity = info.entity.clone().unwrap_or_default();
81                        Box::new(
82                            ListView::new(session.clone(), session.frame(), info)
83                                .plain_toolbar()
84                                .on_open(std::rc::Rc::new(move |key: &ravenrs_app::Value| {
85                                    opener.open_detail(&entity, key);
86                                }))
87                                .grid_shape(|grid| grid.toolbar(false).filter_row(false))
88                                .stretch()
89                                .grow(1.0),
90                        )
91                    }
92                    _ => Box::new(Text::new(info.id)),
93                }
94            } else {
95                Box::new(Text::new(item.id.clone()))
96            };
97            // One piece has nowhere to go and nothing to tell apart from.
98            if alone {
99                return cx.child(&content);
100            }
101            tiles = tiles.tile(Tile::new(item.id.clone(), caption, content));
102        }
103        cx.child(&tiles)
104    }
105}
