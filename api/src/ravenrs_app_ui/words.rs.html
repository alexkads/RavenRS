3use ravenrs_app::RuleWords;

6/// The words the generated screens use.
7#[derive(Clone, Debug)]
8#[non_exhaustive]
9#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
10pub struct Words {
11    /// Lists the keys of what can be done.
12    pub shortcuts: String,
13    /// Gets back to a locked window.
14    pub unlock: String,
15    /// What was typed on a locked window is not it.
16    pub unlock_wrong: String,
17    /// Creates a record.
18    pub new: String,
19    /// Removes what is selected.
20    pub delete: String,
21    /// Keeps the changes.
22    pub save: String,
23    /// Keeps the changes and leaves.
24    pub save_and_close: String,
25    /// Keeps the changes and starts another.
26    pub save_and_new: String,
27    /// Throws away the changes and leaves.
28    pub cancel: String,
29    /// The record before.
30    pub previous: String,
31    /// The record after.
32    pub next: String,
33    /// Reads again.
34    pub refresh: String,
35    /// Goes to the screen before.
36    pub back: String,
37    /// The question before removing.
38    pub delete_question: String,
39    /// What the removal does.
40    pub delete_detail: String,
41    /// What the question before any other action says besides: neutral, since it can be changed
42    /// again.
43    pub confirm_detail: String,
44    /// The question before losing changes.
45    pub discard_question: String,
46    /// What losing changes does.
47    pub discard_detail: String,
48    /// Agrees to lose them.
49    pub discard: String,
50    /// A record being created.
51    pub new_record: String,
52    /// The menu of the ways to see a list.
53    pub variant: String,
54    /// Puts an existing record on a list inside a form.
55    pub link: String,
56    /// Takes a record off a list inside a form, without removing it.
57    pub unlink: String,
58    /// Where a list inside a form would be, while its record is not kept yet.
59    pub save_first: String,
60    /// The menu of the filters ready on a list.
61    pub filter: String,
62    /// Makes a new record from the one selected.
63    pub clone: String,
64    /// Why a record cannot be cloned while it has changes.
65    pub clone_unsaved: String,
66    /// Takes a file out, to wherever people say.
67    pub save_file: String,
68    /// The question when rules only warn.
69    pub warning_question: String,
70    /// Goes on in spite of what rules warn about.
71    pub warning_answer: String,
72    /// The question when someone else changed the same fields, where `{0}` is which ones.
73    pub conflict_question: String,
74    /// What keeping one's own changes does, where `{0}` is which fields.
75    pub conflict_detail: String,
76    /// Keeps one's own changes over the other person's.
77    pub conflict_answer: String,
78    /// What a way of showing a list says when something it needs is missing.
79    pub list_editor_needs: String,
80    /// A row of a timeline for records that point to no resource.
81    pub no_resource: String,
82    /// The series of a chart that counts records.
83    pub chart_count: String,
84    /// Things to be reminded of.
85    pub reminders: String,
86    /// Background work that cannot say what it is doing.
87    pub working: String,
88    /// Background work asked to stop, until it does.
89    pub stopping: String,
90    /// Reminds again later.
91    pub snooze: String,
92    /// Stops reminding.
93    pub dismiss: String,
94    /// Opens what a reminder is about.
95    pub open: String,
96    /// A field that is on.
97    pub yes: String,
98    /// A field that is off.
99    pub no: String,
100    /// What people are told when something is refused.
101    pub access_denied: String,
102    /// What shows in place of a field the person cannot read.
103    pub protected_value: String,
104    /// Enters the application.
105    pub sign_in: String,
106    /// Leaves the application.
107    pub sign_out: String,
108    /// Says signing in did not work, without saying why.
109    pub sign_in_failed: String,
110    /// The name someone signs in with.
111    pub user_name: String,
112    /// Their password.
113    pub password: String,
114    /// The password they have now.
115    pub current_password: String,
116    /// The password they want.
117    pub new_password: String,
118    /// The same again.
119    pub repeat_password: String,
120    /// Signing in could not be tried.
121    pub sign_in_unavailable: String,
122    /// The two are not the same.
123    pub passwords_differ: String,
124    /// It is too short.
125    pub password_too_short: String,
126    /// The current one is not right.
127    pub password_wrong_current: String,
128    /// When the new password is the one they already have.
129    pub password_same: String,
130    /// It was changed.
131    pub password_saved: String,
132    /// Keeps the new password.
133    pub save_password: String,
134    /// Changes one's own password.
135    pub change_password: String,
136    /// Gives someone a password.
137    pub set_password: String,
138    /// Why they have to choose one now.
139    pub must_change_password: String,
140    /// What the first screen of a new installation says.
141    pub first_user: String,
142    /// Creates the first person.
143    pub create_user: String,
144    /// The editor of the application's model.
145    pub model_editor: String,
146    /// Looks for a node of the model.
147    pub search_model: String,
148    /// What shows before a node is picked.
149    pub pick_model_node: String,
150    /// Applies what was typed.
151    pub apply: String,
152    /// Throws away the person's own changes to the model.
153    pub restore_model: String,
154    /// Why what is on a form cannot be kept any more.
155    pub record_gone: String,
156    /// What fails a rule.
157    pub rules: RuleWords,
158    /// In front of money.
159    pub money_prefix: String,
160    /// Between thousands.
161    pub thousands: char,
162    /// Before decimals.
163    pub decimal: char,
164    /// How a date shows in lists, such as `dd/MM/yyyy`.
165    pub date_format: String,
166    /// How a date with a time shows in lists, such as `dd/MM/yyyy HH:mm`.
167    pub date_time_format: String,
168}
