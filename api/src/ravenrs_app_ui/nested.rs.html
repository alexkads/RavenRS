21/// How wide a window of something shown over the screens is.
22pub(crate) const PRESENTED_WIDTH: f32 = 960.0;

24/// How tall a list inside a form is.
25const NESTED_HEIGHT: f32 = 240.0;

27/// Which record a list inside a form belongs to, for the actions that link and unlink.
28pub(crate) struct NestedOwner {
29    /// The entity of the records listed.
30    pub target: String,
31    /// Their field that points to the owner.
32    pub field: String,
33    /// The owner's key.
34    pub key: Value,
35    /// What the list to pick from has to meet.
36    pub link_filter: Option<Criteria>,
37}

39/// A list inside a form, in a frame of its own. What links and unlinks is kept right away.
40pub(crate) struct NestedList {
41    parent: Session,
42    owner: Option<Value>,
43    target: String,
44    field: Option<String>,
45    link_filter: Option<String>,
46    source: Option<ListSource>,
47}
48
49#[cfg_attr(feature = "live", ravenrs_live::expose)]
50impl NestedList {
51    /// A list of records no storage keeps, given by this. There is nothing to link or unlink.
52    pub fn given(parent: Session, target: &str, source: ListSource) -> Self {
53        Self {
54            parent,
55            owner: None,
56            target: target.to_owned(),
57            field: None,
58            link_filter: None,
59            source: Some(source),
60        }
61    }
62
63    pub fn new(
64        parent: Session,
65        owner: &EntityInfo,
66        key: Option<Value>,
67        target: &str,
68        named: Option<&str>,
69        link_filter: Option<&str>,
70    ) -> Self {
71        let field = back_reference(&parent, owner, target, named);
72        Self {
73            parent,
74            owner: key,
75            target: target.to_owned(),
76            field,
77            link_filter: link_filter.map(str::to_owned),
78            source: None,
79        }
80    }
81}

83impl View for NestedList {
84    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
85        let words = self.parent.words();
86        let id = format!("{}_ListView", self.target);
87        let Ok(view) = ViewInfo::from_model(&self.parent.model(), &id) else {
88            return cx.child(&Text::new(id));
89        };
90        let held: State<Option<Session>> = cx.use_state(|| None);
91        if let Some(source) = &self.source {
92            let kept = held.peek().clone();
93            let child = kept.unwrap_or_else(|| {
94                let child = self.parent.child(FrameContext::Nested);
95                child.frame().set_view(Some(view.clone()));
96                held.set(Some(child.clone()));
97                child
98            });
99            let list = ListView::new(child.clone(), child.frame(), view)
100                .source(source.clone())
101                .plain_toolbar()
102                .grid_shape(|grid: DataGrid| grid.toolbar(false).filter_row(false).footer(false));
103            return cx.child(&list.height(NESTED_HEIGHT));
104        }
105        let (Some(key), Some(field)) = (self.owner.clone(), self.field.clone()) else {
106            return cx.child(&Text::new(words.save_first.clone()));
107        };
108        let kept = held.peek().clone();
109        let child = kept.unwrap_or_else(|| {
110            let child = self.parent.child(FrameContext::Nested);
111            child.frame().set_view(Some(view.clone()));
112            held.set(Some(child.clone()));
113            child
114        });
115        child.frame().set_extension(Rc::new(NestedOwner {
116            target: self.target.clone(),
117            field: field.clone(),
118            key: key.clone(),
119            link_filter: self
120                .link_filter
121                .as_deref()
122                .and_then(|text| crate::editors::filter_for(text, self.parent.form_record().as_ref())),
123        }));
124        let opener = self.parent.clone();
125        let target = self.target.clone();
126        let list = ListView::new(child.clone(), child.frame(), view)
127            .filter(Some(Criteria::property(&field).equals(Criteria::value(key))))
128            .implied(field.clone())
129            .plain_toolbar()
130            .on_open(Rc::new(move |key: &Value| opener.open_detail(&target, key)))
131            .grid_shape(|grid: DataGrid| grid.toolbar(false).filter_row(false).footer(false));
132        cx.child(
133            &VStack::new((
134                ActionContainer::new(child.clone(), []),
135                list.height(NESTED_HEIGHT),
136                child.confirmations(),
137                PopupHost::new(child),
138            ))
139            .spacing(8.0)
140            .align(Align::Stretch),
141        )
142    }
143}

145/// A list in a window of its own, for an action that asks for a record.
146struct PopupList {
147    session: Session,
148    action: String,
149    view: ViewInfo,
150    frame: Frame,
151}
152
153impl View for PopupList {
154    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
155        let main = self.session.frame();
156        let open = popup_open(&main);
157        let key_of = {
158            let session = self.session.clone();
159            move |frame: &Frame| {
160                frame.selected().first().and_then(|object| {
161                    let entity = session.entity(object.entity())?;
162                    entity.key_field().map(|field| object.get(&field.name))
163                })
164            }
165        };
166        let accept: Rc<dyn Fn(Option<Value>)> = {
167            let (session, main) = (self.session.clone(), main.clone());
168            Rc::new(move |key| {
169                let input = key.map_or(ActionInput::None, ActionInput::Parameter);
170                if let Err(error) = main.accept_popup(input) {
171                    session.set_error(Some(error.to_string()));
172                }
173            })
174        };
175        let (ok, frame) = (accept.clone(), self.frame.clone());
176        let on_open = accept;
177        let list = ListView::new(self.session.clone(), self.frame.clone(), self.view.clone())
178            .filter(self.session.popup_filter())
179            .on_open(Rc::new(move |key: &Value| on_open(Some(key.clone()))));
180        let words = self.session.words();
181        let cancel_frame = main;
182        cx.child(
183            &Dialog::new(open, self.action.clone())
184                .content(list.height(420.0))
185                .action(
186                    Button::new(words.cancel.clone())
187                        .style(ButtonStyle::Secondary)
188                        .on_click(move || cancel_frame.cancel_popup()),
189                )
190                .action(Button::new("OK").on_click(move || ok(key_of(&frame))))
191                .width(720.0),
192        )
193    }
194}

196/// A form in a window of its own, for an action that asks for a record to be filled in. It runs
197/// with the record, once that meets its rules.
198struct PopupForm {
199    /// Where the action is.
200    owner: Session,
201    /// Where the form is.
202    form: Session,
203    action: String,
204    view: ViewInfo,
205}
206
207impl View for PopupForm {
208    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
209        let main = self.owner.frame();
210        let open = popup_open(&main);
211        let (form, owner_frame) = (self.form.clone(), main.clone());
212        let words = self.owner.words();
213        let cancel_frame = main;
214        let key = self.form.form().map_or(0, |form| Rc::as_ptr(&form) as usize as u64);
215        let content = VStack::new((
216            self.form.error().map(|error| Alert::new(self.action.clone(), error).tone(Tone::Danger)),
217            Keyed::new(key, DetailView::new(self.form.clone(), self.view.clone())),
218        ))
219        .spacing(12.0)
220        .align(Align::Stretch);
221        cx.child(
222            &Dialog::new(open, self.action.clone())
223                .width(480.0)
224                .content(content)
225                .action(
226                    Button::new(words.cancel.clone())
227                        .style(ButtonStyle::Secondary)
228                        .on_click(move || cancel_frame.cancel_popup()),
229                )
230                .action(Button::new("OK").on_click(move || {
231                    let Some((record, _)) = form.check_form() else { return };
232                    if let Err(error) = owner_frame.accept_popup(ActionInput::Record(record)) {
233                        form.set_error(Some(error.to_string()));
234                    }
235                })),
236        )
237    }
238}

247/// The windows of a frame's actions: what they ask for, and what they show.
248pub(crate) struct PopupHost {
249    session: Session,
250}
251
252#[cfg_attr(feature = "live", ravenrs_live::expose)]
253impl PopupHost {
254    pub(crate) fn new(session: Session) -> Self {
255        Self { session }
256    }
257
258    fn parts_for(&self, info: &ViewInfo) -> PopupParts {
259        if info.kind != ViewKind::Detail {
260            let frame = self.session.new_frame(FrameContext::Popup);
261            frame.set_view(Some(info.clone()));
262            return PopupParts::List(frame);
263        }
264        let form = self.session.child(FrameContext::Popup);
265        if let Some(entity) = info.entity.as_deref().and_then(|name| self.session.entity(name)) {
266            let mut record = entity.new_object();
267            self.session.frame().prepare_popup(&mut record);
268            form.open_object(&entity, record, true);
269        }
270        PopupParts::Form(form)
271    }
272}
273
274impl View for PopupHost {
275    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
276        let session = &self.session;
277        crate::jobs::keep(cx, session);
278        let mut parts = Vec::new();
279        // Made once for each time it opens.
280        let held: State<Option<(String, PopupParts)>> = cx.use_state(|| None);
281        match session.frame().open_popup() {
282            Some((action, target)) => {
283                if let Ok(info) = ViewInfo::from_model(&session.model(), &target) {
284                    let kept = held.peek().clone().filter(|(id, _)| *id == action);
285                    let made = kept.map_or_else(
286                        || {
287                            let made = self.parts_for(&info);
288                            held.set(Some((action.clone(), made.clone())));
289                            made
290                        },
291                        |(_, made)| made,
292                    );
293                    let caption = session
294                        .frame()
295                        .action(&action)
296                        .map_or_else(|| action.clone(), |a| a.caption().to_owned());
297                    parts.push(match made {
298                        PopupParts::List(frame) => cx.child(&PopupList {
299                            session: session.clone(),
300                            action: caption,
301                            view: info,
302                            frame,
303                        }),
304                        PopupParts::Form(form) => {
305                            cx.child(&PopupForm { owner: session.clone(), form, action: caption, view: info })
306                        }
307                    });
308                }
309            }
310            None => {
311                if held.peek().is_some() {
312                    held.set(None);
313                }
314            }
315        }
316        parts.push(cx.child(&crate::reminders::Reminders::new(session.clone())));
317        if let Some((title, content)) = session.presented().get() {
318            let (read, write) = (session.presented().clone(), session.presented().clone());
319            let open = Binding::new(
320                move || read.get().is_some(),
321                move |open: bool| {
322                    if !open {
323                        write.set(None);
324                    }
325                },
326            );
327            parts.push(cx.child(&Dialog::new(open, title).width(PRESENTED_WIDTH).content(content())));
328        }
329        cx.container(LayoutStyle::stack(Axis::Vertical), &parts, Paint::None)
330    }
331}

333/// What closes the popup of a frame, when the window closes.
334fn popup_open(main: &Frame) -> Binding<bool> {
335    let (closing, canceling) = (main.clone(), main.clone());
336    Binding::new(
337        move || closing.open_popup().is_some(),
338        move |open: bool| {
339            if !open {
340                canceling.cancel_popup();
341            }
342        },
343    )
