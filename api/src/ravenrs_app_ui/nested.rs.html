14/// How tall a list inside a form is.
15const NESTED_HEIGHT: f32 = 240.0;
16
17/// Which record a list inside a form belongs to, for the actions that link and unlink.
18pub(crate) struct NestedOwner {
19    /// The entity of the records listed.
20    pub target: String,
21    /// Their field that points to the owner.
22    pub field: String,
23    /// The owner's key.
24    pub key: Value,
25    /// What the list to pick from has to meet.
26    pub link_filter: Option<Criteria>,
27}
28
29/// The field of `target` that points to `owner`: the one named, or the first that does.
30pub(crate) fn back_reference(
31    session: &Session,
32    owner: &EntityInfo,
33    target: &str,
34    named: Option<&str>,
35) -> Option<String> {
36    let target = session.entity(target)?;
37    let points_back = |kind: &FieldKind| matches!(kind, FieldKind::Reference(to) if *to == owner.name);
38    target
39        .fields
40        .iter()
41        .find(|field| named.map_or(points_back(&field.kind), |name| field.name == name))
42        .map(|field| field.name.clone())
43}

45/// A list inside a form, in a frame of its own. What links and unlinks is kept right away.
46pub(crate) struct NestedList {
47    parent: Session,
48    owner: Option<Value>,
49    target: String,
50    field: Option<String>,
51    link_filter: Option<String>,
52    source: Option<ListSource>,
53}
54
55#[cfg_attr(feature = "live", ravenrs_live::expose)]
56impl NestedList {
57    /// A list of records no storage keeps, given by this. There is nothing to link or unlink.
58    pub fn given(parent: Session, target: &str, source: ListSource) -> Self {
59        Self {
60            parent,
61            owner: None,
62            target: target.to_owned(),
63            field: None,
64            link_filter: None,
65            source: Some(source),
66        }
67    }
68
69    pub fn new(
70        parent: Session,
71        owner: &EntityInfo,
72        key: Option<Value>,
73        target: &str,
74        named: Option<&str>,
75        link_filter: Option<&str>,
76    ) -> Self {
77        let field = back_reference(&parent, owner, target, named);
78        Self {
79            parent,
80            owner: key,
81            target: target.to_owned(),
82            field,
83            link_filter: link_filter.map(str::to_owned),
84            source: None,
85        }
86    }
87}

89impl View for NestedList {
90    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
91        let words = self.parent.words();
92        let id = format!("{}_ListView", self.target);
93        let Ok(view) = ViewInfo::from_model(&self.parent.model(), &id) else {
94            return cx.child(&Text::new(id));
95        };
96        let held: State<Option<Session>> = cx.use_state(|| None);
97        if let Some(source) = &self.source {
98            let kept = held.peek().clone();
99            let child = kept.unwrap_or_else(|| {
100                let child = self.parent.child(FrameContext::Nested);
101                child.frame().set_view(Some(view.clone()));
102                held.set(Some(child.clone()));
103                child
104            });
105            let list = ListView::new(child.clone(), child.frame(), view)
106                .source(source.clone())
107                .plain_toolbar()
108                .grid_shape(|grid: DataGrid| grid.toolbar(false).filter_row(false).footer(false));
109            return cx.child(&list.height(NESTED_HEIGHT));
110        }
111        let (Some(key), Some(field)) = (self.owner.clone(), self.field.clone()) else {
112            return cx.child(&Text::new(words.save_first.clone()));
113        };
114        let kept = held.peek().clone();
115        let child = kept.unwrap_or_else(|| {
116            let child = self.parent.child(FrameContext::Nested);
117            child.frame().set_view(Some(view.clone()));
118            held.set(Some(child.clone()));
119            child
120        });
121        child.frame().set_extension(Rc::new(NestedOwner {
122            target: self.target.clone(),
123            field: field.clone(),
124            key: key.clone(),
125            link_filter: self
126                .link_filter
127                .as_deref()
128                .and_then(|text| crate::editors::filter_for(text, self.parent.form_record().as_ref())),
129        }));
130        let opener = self.parent.clone();
131        let target = self.target.clone();
132        let list = ListView::new(child.clone(), child.frame(), view)
133            .filter(Some(Criteria::property(&field).equals(Criteria::value(key))))
134            .implied(field.clone())
135            .plain_toolbar()
136            .on_open(Rc::new(move |key: &Value| opener.open_detail(&target, key)))
137            .grid_shape(|grid: DataGrid| grid.toolbar(false).filter_row(false).footer(false));
138        cx.child(
139            &VStack::new((
140                ActionContainer::new(child.clone(), []),
141                list.height(NESTED_HEIGHT),
142                child.confirmations(),
143                PopupHost::new(child),
144            ))
145            .spacing(8.0)
146            .align(Align::Stretch),
147        )
148    }
149}
