16/// The language of the screens and of everything inside them.
17#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
18#[non_exhaustive]
19pub enum Language {
20    /// English.
21    #[default]
22    English,
23    /// Brazilian Portuguese.
24    BrazilianPortuguese,
25}
26
27impl Language {
28    pub(crate) fn words(self) -> Words {
29        match self {
30            Self::English => Words::default(),
31            Self::BrazilianPortuguese => Words::pt_br(),
32        }
33    }
34
35    /// What a translation of the model is known by.
36    fn code(self) -> &'static str {
37        match self {
38            Self::English => "en",
39            Self::BrazilianPortuguese => "pt-BR",
40        }
41    }
42
43    pub(crate) fn labels(self) -> Labels {
44        match self {
45            Self::English => Labels::default(),
46            Self::BrazilianPortuguese => Labels::pt_br(),
47        }
48    }
49}

55/// An application from its records: the module with the actions every application needs,
56/// the screens, and records in memory unless told otherwise.
57///
58/// ```
59/// use ravenrs_app::{BusinessObject, ObjectSpaceExt};
60/// use ravenrs_app_ui::{App, Language};
61///
62/// #[derive(BusinessObject)]
63/// #[object(caption = "Contas", navigation = "Cadastros")]
64/// struct Account {
65///     #[field(key)]
66///     id: i64,
67///     #[field(caption = "Nome")]
68///     name: String,
69/// }
70///
71/// let window = App::new("Raven Contas")
72///     .language(Language::BrazilianPortuguese)
73///     .entity::<Account>()
74///     .seed(|space| {
75///         space.add(&Account { id: 0, name: "Carteira".into() }).map_err(|e| e.to_string())
76///     })
77///     .build()?;
78/// # let _ = window;
79/// # Ok::<(), ravenrs_app::ModelError>(())
80/// ```
81pub struct App {
82    builder: ApplicationBuilder,
83    language: Language,
84    modules: Vec<Box<dyn FnOnce(ApplicationBuilder) -> ApplicationBuilder>>,
85    editors: EditorRegistry,
86    seed: Option<Seed>,
87    storage: Option<Storage>,
88    list_editors: Vec<(String, crate::list_view::ListEditorFactory)>,
89    attachment_store: Option<Rc<dyn AttachmentStore>>,
90    model_store: Option<Rc<dyn ModelStore>>,
91    steps: Vec<SessionStep>,
92    administrator: Option<String>,
93    security: Option<(Rc<ravenrs_app::SecurityContext>, ravenrs_app::SpaceMaker)>,
94}
95
96impl std::fmt::Debug for App {
97    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
98        f.debug_struct("App").field("language", &self.language).finish_non_exhaustive()
99    }
100}

228    /// Keeps records elsewhere than in memory: this makes the unit of work of each view.
229    pub fn storage<S: ObjectSpace + 'static>(mut self, space: impl Fn() -> S + 'static) -> Self {
230        self.storage = Some(Box::new(move |_| {

236    /// Keeps records elsewhere than in memory, in something made from the entities as the model
237    /// has them, with the fields its layers added: this makes what makes the unit of work of each view.
238    pub fn storage_for<S, F>(mut self, make: impl FnOnce(&[ravenrs_app::EntityInfo]) -> F + 'static) -> Self
239    where
240        S: ObjectSpace + 'static,
241        F: Fn() -> S + 'static,
242    {
243        self.storage = Some(Box::new(move |application| {
244            let space = make(application.entities());
245            Rc::new(move || Rc::new(RefCell::new(space())) as Rc<RefCell<dyn ObjectSpace>>)
246        }));
247        self
248    }

262    /// Says who is signed in on the window, and how they leave.
263    pub(crate) fn signed_in(mut self, name: String, sign_out: impl Fn() + 'static) -> Self {
264        let sign_out: Rc<dyn Fn()> = Rc::new(sign_out);
265        self.steps.push(Box::new(move |session| {
266            session.use_account(crate::session::Account { name, sign_out });
267            session.register_control(crate::security::MODEL_EDITOR, crate::security::model_editor(session));
268        }));
269        self
270    }

278    /// Puts it together, or says what is wrong with how it was declared.
279    pub fn build(self) -> Result<AnyView, ModelError> {
280        let words = self.language.words();
281        let mut builder = self.builder.module(SystemModule::with_words(words.clone()));
282        for add in self.modules {
283            builder = add(builder);
284        }
285        let mut application = builder.build()?;
286        if let Some(text) = &self.administrator {
287            let difference = ravenrs_app::Diff::parse(text)?;
288            application.layers_mut().put(&ravenrs_app::ModelLayer::Administrator, difference);
289        }
290        // Before storage is made, for it to know the fields the layers add.
291        if let Some(store) = &self.model_store {
292            application.layers_mut().load(&ravenrs_app::ModelLayer::User, &**store)?;
293        }
294        application.set_language(self.language.code())?;
295        let unsecured = self.security.as_ref().map(|(_, unsecured)| unsecured.clone());
296        if let Some((context, _)) = &self.security {
297            application.set_security(context.clone());
298        }
299        let factory: SpaceFactory = match self.storage {
300            Some(storage) => storage(&application),
301            None => {
302                let database = MemoryDatabase::new(application.entities().to_vec())?;
303                Rc::new(move || Rc::new(RefCell::new(database.space())) as Rc<RefCell<dyn ObjectSpace>>)
304            }
305        };
306        match unsecured {
307            Some(unsecured) => {
308                let mut space = unsecured();
309                application.run_initial_records(&mut *space).map_err(ModelError)?;
310            }
311            None => {
312                let space = factory();
313                let mut space = space.borrow_mut();
314                application.run_initial_records(&mut *space).map_err(ModelError)?;
315            }
316        }
317        if let Some(seed) = self.seed {
318            let space = factory();
319            let mut space = space.borrow_mut();
320            seed(&mut *space).map_err(ModelError)?;
321            space.commit().map_err(|e| ModelError(e.to_string()))?;
322        }
323        let shell = AppShell::from_factory(application, factory, words, self.editors);
324        let session = shell.session();
325        for (name, editor) in self.list_editors {
326            session.register_list_editor(name, move |request| editor(request));
327        }
328        if let Some(store) = self.attachment_store {
329            session.use_attachment_store(store);
330        }
331        for step in self.steps {
332            step(&session);
333        }
334        if let Some(store) = self.model_store {
335            session.use_model_store(store).map_err(ModelError)?;
336        }
337        Ok(Box::new(Localized::new(self.language.labels(), shell)))
338    }

340    /// Puts it together, showing on the screen what is wrong with how it was declared.
341    pub fn into_view(self) -> AnyView {
342        self.build().unwrap_or_else(|error| Box::new(Alert::new("RavenRS", error.0).tone(Tone::Danger)))
343    }
