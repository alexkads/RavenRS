17/// Texts given as JSON, by the code of their language, to put over the ones that come with it.
18pub(crate) type TextPacks = Vec<(String, String)>;
19
20/// The texts that come with the framework with these over them, in this language.
21pub(crate) fn texts_of(language: Language, packs: &[(String, String)]) -> Result<Texts, ModelError> {
22    let mut catalog = Catalog::builtin();
23    for (code, json) in packs {
24        catalog.merge_json(code, json).map_err(|error| ModelError(error.0))?;
25    }
26    Ok(catalog.texts(language))
27}

33/// An application from its records: the module with the actions every application needs,
34/// the screens, and records in memory unless told otherwise.
35///
36/// ```
37/// use ravenrs_app::{BusinessObject, ObjectSpaceExt};
38/// use ravenrs_app_ui::{App, Language};
39///
40/// #[derive(BusinessObject)]
41/// #[object(caption = "Contas", navigation = "Cadastros")]
42/// struct Account {
43///     #[field(key)]
44///     id: i64,
45///     #[field(caption = "Nome")]
46///     name: String,
47/// }
48///
49/// let window = App::new("Raven Contas")
50///     .language(Language::BrazilianPortuguese)
51///     .entity::<Account>()
52///     .seed(|space| {
53///         space.add(&Account { id: 0, name: "Carteira".into() }).map_err(|e| e.to_string())
54///     })
55///     .build()?;
56/// # let _ = window;
57/// # Ok::<(), ravenrs_app::ModelError>(())
58/// ```
59#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
60pub struct App {
61    builder: ApplicationBuilder,
62    language: Language,
63    texts: TextPacks,
64    modules: Vec<Box<dyn FnOnce(ApplicationBuilder) -> ApplicationBuilder>>,
65    editors: EditorRegistry,
66    seed: Option<Seed>,
67    storage: Option<Storage>,
68    kept: Option<Rc<dyn ravenrs_app::Storage>>,
69    list_editors: Vec<(String, crate::list_view::ListEditorFactory)>,
70    attachment_store: Option<Rc<dyn AttachmentStore>>,
71    model_store: Option<Rc<dyn ModelStore>>,
72    steps: Vec<SessionStep>,
73    administrator: Option<String>,
74    security: Option<(Rc<ravenrs_app::SecurityContext>, ravenrs_app::SpaceMaker)>,
75    actions: Vec<ActionMaker>,
76}

292        self.kept = Some(storage);

304    /// Keeps records elsewhere than in memory, in something made from the entities as the model

306    pub fn storage_for<S, F>(mut self, make: impl FnOnce(&[ravenrs_app::EntityInfo]) -> F + 'static) -> Self
307    where
308        S: ObjectSpace + 'static,
309        F: Fn() -> S + 'static,
310    {
311        self.storage = Some(Box::new(move |application| {

313            Rc::new(move || Rc::new(RefCell::new(space())) as Rc<RefCell<dyn ObjectSpace>>)
314        }));

316    }

346    /// Puts it together, or says what is wrong with how it was declared.
347    pub fn build(self) -> Result<AnyView, ModelError> {
348        let texts = texts_of(self.language, &self.texts)?;
349        let words = Words::from_texts(&texts);
350        let mut builder = self.builder.module(SystemModule::with_words(words.clone()));
351        for add in self.modules {
352            builder = add(builder);
353        }
354        if !self.actions.is_empty() {
355            builder = builder.module(OwnActions(self.actions));
356        }
357        let mut application = builder.build()?;
358        if let Some(text) = &self.administrator {
359            let difference = ravenrs_app::Diff::parse(text)?;
360            application.layers_mut().put(&ravenrs_app::ModelLayer::Administrator, difference);
361        }
362        // Before storage is made, for it to know the fields the layers add.
363        if let Some(store) = &self.model_store {
364            application.layers_mut().load(&ravenrs_app::ModelLayer::User, &**store)?;
365        }
366        application.set_language(self.language.code())?;
367        let unsecured = self.security.as_ref().map(|(_, unsecured)| unsecured.clone());
368        if let Some((context, _)) = &self.security {
369            application.set_security(context.clone());
370        }
371        let mut restored = false;
372        let factory: SpaceFactory = match (self.storage, self.kept) {
373            (Some(storage), _) => storage(&application),
374            (None, Some(kept)) => {
375                let database = MemoryDatabase::new(application.entities().to_vec())?
376                    .stored_in(kept, "records")
377                    .map_err(|e| ModelError(e.to_string()))?;
378                restored = database.was_restored();
379                Rc::new(move || Rc::new(RefCell::new(database.space())) as Rc<RefCell<dyn ObjectSpace>>)
380            }
381            (None, None) => {
382                let database = MemoryDatabase::new(application.entities().to_vec())?;
383                Rc::new(move || Rc::new(RefCell::new(database.space())) as Rc<RefCell<dyn ObjectSpace>>)
384            }
385        };
386        match unsecured {
387            Some(unsecured) => {
388                let mut space = unsecured();
389                application.run_initial_records(&mut *space).map_err(ModelError)?;
390            }
391            None => {
392                let space = factory();
393                let mut space = space.borrow_mut();
394                application.run_initial_records(&mut *space).map_err(ModelError)?;
395            }
396        }
397        if let Some(seed) = self.seed.filter(|_| !restored) {
398            let space = factory();
399            let mut space = space.borrow_mut();
400            seed(&mut *space).map_err(ModelError)?;
401            space.commit().map_err(|e| ModelError(e.to_string()))?;
402        }
403        let shell = AppShell::from_factory(application, factory, words, self.editors);
404        let session = shell.session();
405        session.use_texts(texts.clone());
406        session.set_access_words(ravenrs_app_a11y::AccessWords::for_language(self.language.code()));
407        for (name, editor) in self.list_editors {
408            session.register_list_editor(name, move |request| editor(request));
409        }
410        if let Some(store) = self.attachment_store {
411            session.use_attachment_store(store);
412        }
413        for step in self.steps {
414            step(&session);
415        }
416        if let Some(store) = self.model_store {
417            session.use_model_store(store).map_err(ModelError)?;
418        }
419        Ok(Box::new(Localized::new(Labels::from_texts(&texts), shell)))
420    }
