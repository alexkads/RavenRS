17/// Texts given as JSON, by the code of their language, to put over the ones that come with it.
18pub(crate) type TextPacks = Vec<(String, String)>;
19
20/// The texts that come with the framework with these over them, in this language.
21pub(crate) fn texts_of(language: Language, packs: &[(String, String)]) -> Result<Texts, ModelError> {
22    let mut catalog = Catalog::builtin();
23    for (code, json) in packs {
24        catalog.merge_json(code, json).map_err(|error| ModelError(error.0))?;
25    }
26    Ok(catalog.texts(language))
27}

33/// An application from its records: the module with the actions every application needs,
34/// the screens, and records in memory unless told otherwise.
35///
36/// ```
37/// use ravenrs_app::{BusinessObject, ObjectSpaceExt};
38/// use ravenrs_app_ui::{App, Language};
39///
40/// #[derive(BusinessObject)]
41/// #[object(caption = "Contas", navigation = "Cadastros")]
42/// struct Account {
43///     #[field(key)]
44///     id: i64,
45///     #[field(caption = "Nome")]
46///     name: String,
47/// }
48///
49/// let window = App::new("Raven Contas")
50///     .language(Language::BrazilianPortuguese)
51///     .entity::<Account>()
52///     .seed(|space| {
53///         space.add(&Account { id: 0, name: "Carteira".into() }).map_err(|e| e.to_string())
54///     })
55///     .build()?;
56/// # let _ = window;
57/// # Ok::<(), ravenrs_app::ModelError>(())
58/// ```
59#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
60pub struct App {
61    builder: ApplicationBuilder,
62    language: Language,
63    texts: TextPacks,
64    modules: Vec<Box<dyn FnOnce(ApplicationBuilder) -> ApplicationBuilder>>,
65    editors: EditorRegistry,
66    seed: Option<Seed>,
67    storage: Option<Storage>,
68    kept: Option<Rc<dyn ravenrs_app::Storage>>,
69    list_editors: Vec<(String, crate::session::list_types::ListEditorFactory)>,
70    attachment_store: Option<Rc<dyn AttachmentStore>>,
71    model_store: Option<Rc<dyn ModelStore>>,
72    steps: Vec<SessionStep>,
73    administrator: Option<String>,
74    security: Option<(Rc<ravenrs_app::SecurityContext>, ravenrs_app::SpaceMaker)>,
75    theme: Option<ravenrs_style::Theme>,
76    themes: Vec<ravenrs_style::Theme>,
77    brand: Option<Rc<ravenrs_widgets::AnyView>>,
78    brand_subtitle: Option<String>,
79    counts: std::collections::HashMap<String, crate::navigation::CountOf>,
80    counter: bool,
81    actions: Vec<ActionMaker>,
82}

429    /// Puts it together, or says what is wrong with how it was declared.
430    pub fn build(self) -> Result<AnyView, ModelError> {
431        let texts = texts_of(self.language, &self.texts)?;
432        let words = Words::from_texts(&texts);
433        let mut builder = self.builder.module(SystemModule::with_words(words.clone()));
434        for add in self.modules {
435            builder = add(builder);
436        }
437        if !self.actions.is_empty() {
438            builder = builder.module(OwnActions(self.actions));
439        }
440        let mut application = builder.build()?;
441        if let Some(text) = &self.administrator {
442            let difference = ravenrs_app::Diff::parse(text)?;
443            application.layers_mut().put(&ravenrs_app::ModelLayer::Administrator, difference);
444        }
445        // Before storage is made, for it to know the fields the layers add.
446        if let Some(store) = &self.model_store {
447            application.layers_mut().load(&ravenrs_app::ModelLayer::User, &**store)?;
448        }
449        application.set_language(self.language.code())?;
450        let unsecured = self.security.as_ref().map(|(_, unsecured)| unsecured.clone());
451        if let Some((context, _)) = &self.security {
452            application.set_security(context.clone());
453        }
454        let mut restored = false;
455        let factory: SpaceFactory = match (self.storage, self.kept) {
456            (Some(storage), _) => storage(&application),
457            (None, Some(kept)) => {
458                let database = MemoryDatabase::new(application.entities().to_vec())?
459                    .stored_in(kept, "records")
460                    .map_err(|e| ModelError(e.to_string()))?;
461                restored = database.was_restored();
462                Rc::new(move || Rc::new(RefCell::new(database.space())) as Rc<RefCell<dyn ObjectSpace>>)
463            }
464            (None, None) => {
465                let database = MemoryDatabase::new(application.entities().to_vec())?;
466                Rc::new(move || Rc::new(RefCell::new(database.space())) as Rc<RefCell<dyn ObjectSpace>>)
467            }
468        };
469        match unsecured {
470            Some(unsecured) => {
471                let mut space = unsecured();
472                application.run_initial_records(&mut *space).map_err(ModelError)?;
473            }
474            None => {
475                let space = factory();
476                let mut space = space.borrow_mut();
477                application.run_initial_records(&mut *space).map_err(ModelError)?;
478            }
479        }
480        if let Some(seed) = self.seed.filter(|_| !restored) {
481            let space = factory();
482            let mut space = space.borrow_mut();
483            seed(&mut *space).map_err(ModelError)?;
484            space.commit().map_err(|e| ModelError(e.to_string()))?;
485        }
486        let mut shell = AppShell::from_factory(application, factory, words, self.editors);
487        if let Some(theme) = self.theme {
488            shell = shell.theme(theme);
489        }
490        if !self.themes.is_empty() {
491            shell = shell.themes(self.themes);
492        }
493        if let Some(mark) = self.brand {
494            shell = shell.brand(SharedMark(mark));
495            if let Some(subtitle) = self.brand_subtitle {
496                shell = shell.brand_subtitle(subtitle);
497            }
498        }
499        shell = shell.counts(Rc::new(self.counts));
500        if self.counter {
501            shell = shell.template(crate::CounterTemplate);
502        }
503        let session = shell.session();
504        session.use_texts(texts.clone());
505        session.set_access_words(ravenrs_app_a11y::AccessWords::for_language(self.language.code()));
506        for (name, editor) in self.list_editors {
507            session.register_list_editor(name, move |request| editor(request));
508        }
509        if let Some(store) = self.attachment_store {
510            session.use_attachment_store(store);
511        }
512        for step in self.steps {
513            step(&session);
514        }
515        if let Some(store) = self.model_store {
516            session.use_model_store(store).map_err(ModelError)?;
517        }
518        Ok(Box::new(Localized::new(Labels::from_texts(&texts), shell)))
519    }

521    /// Puts it together, showing on the screen what is wrong with how it was declared.
522    pub fn into_view(self) -> AnyView {
523        self.build().unwrap_or_else(|error| Box::new(Alert::new("RavenRS", error.0).tone(Tone::Danger)))
524    }
