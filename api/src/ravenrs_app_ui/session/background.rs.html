1//! Work that takes long, out of the way of the screens: it runs on its own thread, and what it
2//! brings comes back on the thread of the screens. In the browser, which has no threads to spare,
3//! it runs right away.
4

16/// How a piece of work in the background is going.
17#[derive(Clone, Debug, PartialEq)]
18#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
19pub struct JobInfo {
20    /// What it is known by, to cancel it.
21    pub id: u64,
22    /// What it is doing, for people.
23    pub title: String,
24    /// How much of it is done, from 0 to 1, if it says.
25    pub fraction: Option<f32>,
26    /// The step it is on, if it says.
27    pub note: Option<String>,
28}
29
30enum JobEvent {
31    Progress(Option<f32>, Option<String>),
32    Done,
33}
34
35/// What a piece of work has, on its thread, to say how it is going and to hear it was canceled.
36#[derive(Clone, Debug)]
37#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
38pub struct JobProgress {
39    sender: Option<Sender<JobEvent>>,
40    waker: Option<UiWaker>,
41    canceled: Arc<AtomicBool>,
42}
43
44impl std::fmt::Debug for JobEvent {
45    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        f.write_str("JobEvent")
47    }
48}
49
50#[cfg_attr(feature = "live", ravenrs_live::expose)]
51impl JobProgress {
52    /// Says how much is done, from 0 to 1, and what it is doing.
53    pub fn report(&self, fraction: f32, note: impl Into<String>) {
54        if let Some(sender) = &self.sender {
55            let _ = sender.send(JobEvent::Progress(Some(fraction.clamp(0.0, 1.0)), Some(note.into())));
56        }
57        if let Some(waker) = &self.waker {
58            waker.wake();
59        }
60    }
61
62    /// Whether someone asked it to stop. It stops when it next looks.
63    pub fn is_canceled(&self) -> bool {
64        self.canceled.load(Ordering::Acquire)
65    }
66}
67
68type Finish = Box<dyn FnOnce(&Session)>;
69
70struct Running {
71    id: u64,
72    canceled: Arc<AtomicBool>,
73    events: Receiver<JobEvent>,
74    /// What it brought, once it did.
75    result: Box<dyn FnMut() -> Option<Finish>>,
76}
77
78/// What runs for a session.
79pub(crate) struct Jobs {
80    running: RefCell<Vec<Running>>,
81    next: std::cell::Cell<u64>,
82    infos: State<Vec<JobInfo>>,
83    waker: RefCell<Option<UiWaker>>,
84}
85
86impl Default for Jobs {
87    fn default() -> Self {
88        Self {
89            running: RefCell::new(Vec::new()),
90            next: std::cell::Cell::new(1),
91            infos: State::new(Vec::new()),
92            waker: RefCell::new(None),
93        }
94    }
95}
96
97impl std::fmt::Debug for Jobs {
98    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
99        f.debug_struct("Jobs").finish_non_exhaustive()
100    }
101}

104impl Jobs {
105    pub(crate) fn connect(&self, waker: UiWaker) {
106        *self.waker.borrow_mut() = Some(waker);
107    }

110#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
111impl Session {
112    /// Runs `work` out of the way of the screens, then `done` with what it brought, on the thread of
113    /// the screens. Gives what it is known by, to cancel it.
114    pub fn run_background<R: Send + 'static>(
115        &self,
116        title: impl Into<String>,
117        work: impl FnOnce(&JobProgress) -> R + Send + 'static,
118        done: impl FnOnce(&Session, R) + 'static,
119    ) -> u64 {
120        let jobs = self.jobs();
121        let id = jobs.next.get();
122        jobs.next.set(id + 1);
123        if cfg!(target_arch = "wasm32") {
124            let _ = title;
125            let progress =
126                JobProgress { sender: None, waker: None, canceled: Arc::new(AtomicBool::new(false)) };
127            done(self, work(&progress));
128            return id;
129        }
130        let (events, heard) = channel();
131        let (result_out, result_in) = channel::<R>();
132        let canceled = Arc::new(AtomicBool::new(false));
133        let waker = jobs.waker.borrow().clone();
134        let progress =
135            JobProgress { sender: Some(events.clone()), waker: waker.clone(), canceled: canceled.clone() };
136        std::thread::spawn(move || {
137            let brought = work(&progress);
138            let _ = result_out.send(brought);
139            let _ = events.send(JobEvent::Done);
140            if let Some(waker) = waker {
141                waker.wake();
142            }
143        });
144        let mut done = Some(done);
145        let result: Box<dyn FnMut() -> Option<Finish>> = Box::new(move || {
146            let brought = result_in.try_recv().ok()?;
147            let done = done.take()?;
148            Some(Box::new(move |session: &Session| done(session, brought)) as Finish)
149        });
150        jobs.running.borrow_mut().push(Running { id, canceled, events: heard, result });
151        jobs.infos.update(|infos| {
152            infos.push(JobInfo { id, title: title.into(), fraction: None, note: None });
153        });
154        id
155    }
156
157    /// Asks a piece of work to stop. What it brings when it does still comes.
158    pub fn cancel_background(&self, id: u64) {
159        if let Some(job) = self.jobs().running.borrow().iter().find(|job| job.id == id) {
160            job.canceled.store(true, Ordering::Release);
161        }
162    }
163
164    /// What is running now.
165    pub fn background_jobs(&self) -> Vec<JobInfo> {
166        self.jobs().infos.get()
167    }
168
169    /// Takes in what the work running has to say, and finishes what ended. The window does this
170    /// by itself; a test without one calls it.
171    pub fn poll_background(&self) {
172        let jobs = self.jobs();
173        let mut finished: Vec<Finish> = Vec::new();
174        let mut changes: Vec<(u64, Option<f32>, Option<String>)> = Vec::new();
175        let mut gone: Vec<u64> = Vec::new();
176        jobs.running.borrow_mut().retain_mut(|job| {
177            let mut over = false;
178            loop {
179                match job.events.try_recv() {
180                    Ok(JobEvent::Progress(fraction, note)) => changes.push((job.id, fraction, note)),
181                    Ok(JobEvent::Done) | Err(TryRecvError::Disconnected) => {
182                        over = true;
183                        break;
184                    }
185                    Err(TryRecvError::Empty) => break,
186                }
187            }
188            if over {
189                if let Some(finish) = (job.result)() {
190                    finished.push(finish);
191                }
192                gone.push(job.id);
193            }
194            !over
195        });
196        if !changes.is_empty() || !gone.is_empty() {
197            jobs.infos.update(|infos| {
198                for (id, fraction, note) in changes {
199                    if let Some(info) = infos.iter_mut().find(|info| info.id == id) {
200                        info.fraction = fraction;
201                        info.note = note;
202                    }
203                }
204                infos.retain(|info| !gone.contains(&info.id));
205            });
206        }
207        for finish in finished {
208            finish(self);
209        }
210    }
211
212    /// Waits for everything running and finishes it, for a test.
213    #[doc(hidden)]
214    pub fn finish_background(&self) {
215        let deadline = std::time::Instant::now() + std::time::Duration::from_secs(60);
216        while !self.background_jobs().is_empty() && std::time::Instant::now() < deadline {
217            self.poll_background();
218            std::thread::sleep(std::time::Duration::from_millis(2));
219        }
220    }
221}
