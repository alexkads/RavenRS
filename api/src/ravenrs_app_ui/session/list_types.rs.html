1//! What a list hands around: its source of records, its rows as a table, and the request a list
2//! editor gets. The session keeps them, so they live below the views.
3
4use std::rc::Rc;
5
6use ravenrs_app::{EntityInfo, FieldInfo, Object, Value, ViewInfo};
7use ravenrs_widgets::AnyView;
8use ravenrs_widgets::grid::GridTable;
9
10use crate::session::Session;
11
12/// What opening a row of a list does, with its key.
13pub type OpenHandler = Rc<dyn Fn(&Value)>;
14
15/// What gives a list its records when no storage keeps them.
16pub type ListSource = Rc<dyn Fn() -> Vec<Object>>;
17
18/// What a list on screen shows now, as a table, for a report to take the same rows.
19pub(crate) type ShownList = Rc<dyn Fn() -> Option<GridTable>>;
20
21/// What a list editor is asked to show: the records as the list reads them, and how to act on them.
22#[non_exhaustive]
23#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
24pub struct ListEditorRequest {
25    /// The application it lives in.
26    pub session: Session,
27    /// The view it shows.
28    pub view: ViewInfo,
29    /// The entity of the records.
30    pub entity: EntityInfo,
31    /// The fields the list shows, with what their columns say.
32    pub columns: Vec<(FieldInfo, String)>,
33    /// The records, in the list's order.
34    pub objects: Rc<Vec<Object>>,
35    /// Opens a record, by its key.
36    pub on_open: OpenHandler,
37    /// Says which records are selected, which turns actions on and off.
38    pub on_select: Rc<dyn Fn(Vec<Object>)>,
39}
40
41impl std::fmt::Debug for ListEditorRequest {
42    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
43        f.debug_struct("ListEditorRequest").field("view", &self.view).finish_non_exhaustive()
44    }
45}
46
47/// Shows a list some other way than in a grid.
48pub(crate) type ListEditorFactory = Rc<dyn Fn(&ListEditorRequest) -> AnyView>;
