12const SECONDS_IN_A_DAY: i64 = 86_400;

14/// A record's date as the day people pick.
15pub(crate) fn date_of(value: &Value) -> Option<Date> {
16    match value {
17        Value::Date(moment) => Some(Date::from_days(moment.days)),
18        _ => None,
19    }
20}

22pub(crate) fn value_of_date(date: Option<Date>) -> Value {
23    date.map_or(Value::Null, |date| Value::Date(Moment { days: date.to_days(), seconds: 0 }))
24}

26pub(crate) fn date_time_of(value: &Value) -> Option<DateTime> {
27    let Value::Date(moment) = value else { return None };
28    let seconds = i64::from(moment.seconds).rem_euclid(SECONDS_IN_A_DAY);
29    let time = Time::new((seconds / 3600) as u8, ((seconds % 3600) / 60) as u8)?;
30    Some(DateTime { date: Date::from_days(moment.days), time })
31}
32
33pub(crate) fn value_of_date_time(moment: Option<DateTime>) -> Value {
34    moment.map_or(Value::Null, |moment| {
35        let seconds = i64::from(moment.time.hour()) * 3600 + i64::from(moment.time.minute()) * 60;
36        Value::Date(Moment { days: moment.date.to_days(), seconds: seconds as _ })
37    })
38}

40/// A record keeps a file as text: its name on the first line and after it its content, or the key
41/// its content is under when a store keeps it, which `@` tells apart from base64.
42pub(crate) fn value_of_file(
43    file: Option<&PickedFile>,
44    store: Option<&dyn AttachmentStore>,
45) -> Result<Value, String> {
46    let Some(file) = file else { return Ok(Value::Null) };
47    Ok(Value::Text(match store {
48        Some(store) => format!("{}\n@{}", file.name, store.put(&file.name, &file.bytes)?),
49        None => format!("{}\n{}", file.name, encode(&file.bytes)),
50    }))
51}

53/// A file from what a record keeps.
54pub(crate) fn file_of(value: &Value, store: Option<&dyn AttachmentStore>) -> Option<PickedFile> {
55    let Value::Text(text) = value else { return None };
56    let (name, content) = text.split_once('\n')?;
57    let bytes = match content.strip_prefix('@') {
58        Some(key) => store?.get(key).ok()??,
59        None => decode(content)?,
60    };
61    Some(PickedFile::new(name, bytes))
62}

64/// The name of a file a record keeps, without reading its content.
65pub(crate) fn file_name(value: &Value) -> Option<String> {
66    let Value::Text(text) = value else { return None };
67    text.split_once('\n').map(|(name, _)| name.to_owned())
68}

70/// What a `display_format` says around the value, and what it says about the value.
71fn format_parts(format: &str) -> Option<(&str, &str, &str)> {
72    let start = format.find('{')?;
73    let end = start + format[start..].find('}')?;
74    Some((&format[..start], format[start + 1..end].trim_start_matches(':'), &format[end + 1..]))
75}

77/// The decimals of `.2`.
78fn decimals_of(spec: &str) -> Option<usize> {
79    spec.strip_prefix('.')?.parse().ok()
80}

82/// A moment by a pattern such as `dd/MM/yyyy HH:mm`.
83fn moment_text(moment: &Moment, pattern: &str) -> String {
84    let date = Date::from_days(moment.days);
85    let seconds = i64::from(moment.seconds).rem_euclid(SECONDS_IN_A_DAY);
86    pattern
87        .replace("yyyy", &format!("{:04}", date.year()))
88        .replace("MM", &format!("{:02}", date.month()))
89        .replace("dd", &format!("{:02}", date.day()))
90        .replace("HH", &format!("{:02}", seconds / 3600))
91        .replace("mm", &format!("{:02}", (seconds % 3600) / 60))
92        .replace("ss", &format!("{:02}", seconds % 60))
93}

95/// A value as its field's `display_format` asks.
96fn formatted(field: &FieldInfo, value: &Value) -> Option<String> {
97    let (before, spec, after) = format_parts(field.display_format.as_deref()?)?;
98    let middle = match value {
99        Value::Null => return Some(String::new()),
100        Value::Date(moment) if !spec.is_empty() => moment_text(moment, spec),
101        Value::Number(number) => match decimals_of(spec) {
102            Some(decimals) => {
103                let number = if matches!(field.kind, FieldKind::Money) { number / 100.0 } else { *number };
104                format!("{number:.decimals$}")
105            }
106            None => plain_text(field, value),
107        },
108        _ => plain_text(field, value),
109    };
110    Some(format!("{before}{middle}{after}"))
111}
112
113/// The prefix, decimals and suffix a `display_format` gives a number.
114fn number_shape(field: &FieldInfo) -> Option<(String, usize, String)> {
115    let (before, spec, after) = format_parts(field.display_format.as_deref()?)?;
116    Some((before.to_owned(), decimals_of(spec).unwrap_or(0), after.to_owned()))
117}

119/// A value as words, with what its field knows about it.
120pub(crate) fn display(field: &FieldInfo, value: &Value) -> String {
121    formatted(field, value).unwrap_or_else(|| plain_text(field, value))
122}
123
124fn plain_text(field: &FieldInfo, value: &Value) -> String {
125    match (&field.kind, value) {
126        (_, Value::Null) => String::new(),
127        (FieldKind::Tags(options), Value::Text(codes)) => codes
128            .split(',')
129            .map(|code| {
130                options
131                    .iter()
132                    .find(|option| option.code == code)
133                    .map_or(code, |option| option.caption.as_str())
134            })
135            .collect::<Vec<_>>()
136            .join(", "),
137        // Its name, and never its content.
138        (FieldKind::Image | FieldKind::File, value) => file_name(value).unwrap_or_default(),
139        (FieldKind::Enum(options), Value::Text(code)) => options
140            .iter()
141            .find(|option| option.code == *code)
142            .map_or_else(|| code.clone(), |option| option.caption.clone()),
143        (FieldKind::Date, value) => date_of(value).map_or_else(String::new, |date| date.to_string()),
144        // Year first, so a column of these sorts as it reads.
145        (FieldKind::DateTime, Value::Date(when)) => {
146            let (year, month, day) = when.ymd();
147            let (hour, minute) = (when.seconds / 3600, when.seconds % 3600 / 60);
148            format!("{year:04}-{month:02}-{day:02} {hour:02}:{minute:02}")
149        }
150        (_, Value::Text(text)) => text.clone(),
151        (_, Value::Number(number)) if number.fract() == 0.0 => format!("{}", *number as i64),
152        (_, Value::Number(number)) => number.to_string(),
153        (_, Value::Bool(on)) => on.to_string(),
154        (_, other) => format!("{other:?}"),
155    }
156}

158/// The cell of a list. Money shows in whole units, though it is kept in cents.
159pub(crate) fn cell_of(field: &FieldInfo, value: &Value) -> Cell {
160    match (&field.kind, value) {
161        (_, Value::Null) => Cell::Empty,
162        (FieldKind::Money, Value::Number(cents)) => Cell::Number(cents / 100.0),
163        (_, Value::Number(number)) => Cell::Number(*number),
164        // A pattern of its own gives up on sorting as a date, to show as it was asked.
165        (FieldKind::Date, value) if field.display_format.is_some() => Cell::Text(display(field, value)),
166        (FieldKind::Date, value) => date_of(value).map_or(Cell::Empty, Cell::Date),
167        (
168            FieldKind::Enum(_)
169            | FieldKind::Tags(_)
170            | FieldKind::DateTime
171            | FieldKind::Image
172            | FieldKind::File,
173            value,
174        ) => Cell::Text(display(field, value)),
175        (_, Value::Text(_)) if field.display_format.is_some() => Cell::Text(display(field, value)),
176        (_, Value::Bool(on)) => Cell::Bool(*on),
177        (_, Value::Text(text)) => Cell::Text(text.clone()),
178        (_, other) => Cell::Text(format!("{other:?}")),
179    }
180}

182/// The column of a list.
183pub(crate) fn column_of(field: &FieldInfo, caption: &str, words: &Words) -> GridColumn {
184    let number = |decimals: usize, prefix: &str| NumberFormat {
185        decimals,
186        prefix: prefix.to_owned(),
187        thousands: words.thousands,
188        decimal: words.decimal,
189        ..NumberFormat::default()
190    };
191    let column = match &field.kind {
192        FieldKind::Integer | FieldKind::Decimal(_) | FieldKind::Money if field.display_format.is_some() => {
193            let (prefix, decimals, suffix) = number_shape(field).unwrap_or_default();
194            GridColumn::number(caption).number_format(NumberFormat { suffix, ..number(decimals, &prefix) })
195        }
196        FieldKind::Integer => GridColumn::number(caption).number_format(number(0, "")),
197        FieldKind::Decimal(places) => {
198            GridColumn::number(caption).number_format(number(usize::from(*places), ""))
199        }
200        FieldKind::Money => GridColumn::number(caption).number_format(number(2, &words.money_prefix)),
201        FieldKind::Boolean => {
202            let (yes, no) = (words.yes.clone(), words.no.clone());
203            GridColumn::boolean(caption).format(move |value| match value {
204                ravenrs_widgets::grid::Value::Bool(true) => yes.clone(),
205                ravenrs_widgets::grid::Value::Bool(false) => no.clone(),
206                _ => String::new(),
207            })
208        }
209        FieldKind::Date => GridColumn::date(caption),
210        _ => GridColumn::text(caption),
211    };
212    column.field(field.name.clone())
213}

215/// Whether a field can be changed right on a list.
216pub(crate) fn editable_in_list(field: &FieldInfo) -> bool {
217    !field.is_key
218        && !field.read_only
219        && field.calculated.is_none()
220        && matches!(
221            field.kind,
222            FieldKind::Text
223                | FieldKind::Integer
224                | FieldKind::Decimal(_)
225                | FieldKind::Money
226                | FieldKind::Boolean
227                | FieldKind::Date
228                | FieldKind::Enum(_)
229        )
230}

232/// What a cell edited on a list is, in what the record holds; nothing for what cannot be read.
233pub(crate) fn value_of_cell(field: &FieldInfo, cell: &Cell) -> Option<Value> {
234    Some(match (&field.kind, cell) {
235        (_, Cell::Empty) => Value::Null,
236        (FieldKind::Money, Cell::Number(units)) => Value::Number((units * 100.0).round()),
237        (FieldKind::Integer, Cell::Number(number)) => Value::Number(number.trunc()),
238        (_, Cell::Number(number)) => Value::Number(*number),
239        (_, Cell::Bool(on)) => Value::Bool(*on),
240        (FieldKind::Date, Cell::Date(date)) => value_of_date(Some(*date)),
241        (FieldKind::Enum(options), Cell::Text(text)) => {
242            let found = options.iter().find(|option| option.caption == *text || option.code == *text)?;
243            Value::Text(found.code.clone())
244        }
245        (FieldKind::Text | FieldKind::LongText, Cell::Text(text)) => {
246            if text.is_empty() {
247                Value::Null
248            } else {
249                Value::Text(text.clone())
250            }
251        }
252        _ => return None,
253    })
254}

256/// The color of a tint, in the theme.
257pub(crate) fn semantic_of(tint: Tint) -> Semantic {
258    match tint {
259        Tint::Danger => Semantic::Danger,
260        Tint::Warning => Semantic::Warning,
261        Tint::Success => Semantic::Success,
262        Tint::Accent => Semantic::Accent,
263        _ => Semantic::InkMuted,
264    }
265}

267/// Text that identifies a key, the same as the grid gives its cell.
268pub(crate) fn key_text(entity: &EntityInfo, key: &Value) -> String {
269    match entity.key_field() {
270        Some(field) => cell_of(field, key).key_text(),
271        None => String::new(),
272    }
273}

275/// A key back from its text.
276pub(crate) fn key_from_text(entity: &EntityInfo, text: &str) -> Value {
277    match entity.key_field().map(|field| &field.kind) {
278        Some(FieldKind::Integer | FieldKind::Decimal(_)) => text.parse().map_or(Value::Null, Value::Number),
279        _ => Value::Text(text.to_owned()),
280    }

308        let file = PickedFile::new("nota.pdf", vec![1_u8, 2, 250]);
309        let kept = value_of_file(Some(&file), None).expect("kept");
310        let back = file_of(&kept, None).expect("a file");
311        assert_eq!(back.name, "nota.pdf");

313        assert_eq!(value_of_file(None, None), Ok(Value::Null));
314        assert_eq!(file_of(&Value::Text("not a file".into()), None).map(|f| f.name), None);
315        assert_eq!(display(&FieldInfo::new("f", FieldKind::File), &kept), "nota.pdf");
316    }

331        let percent = FieldInfo::new("p", FieldKind::Integer).display_format("{} %");
