323/// The record a form is editing: what it was, and one state per field.
324#[derive(Debug)]
325pub(crate) struct FormState {
326    pub entity: EntityInfo,
327    pub original: Object,
328    pub is_new: bool,
329    pub fields: BTreeMap<String, State<Value>>,
330}
331
332impl FormState {
333    fn new(entity: EntityInfo, original: Object, is_new: bool) -> Self {
334        let fields = entity
335            .fields
336            .iter()
337            .filter(|f| f.calculated.is_none())
338            .map(|f| (f.name.clone(), State::new(original.get(&f.name))))
339            .collect();
340        Self { entity, original, is_new, fields }
341    }
342
343    /// The record with what the fields hold now, without reading them for the build.
344    pub fn current(&self) -> Object {
345        let mut object = self.original.clone();
346        for (name, value) in &self.fields {
347            object.set(name, value.peek().clone());
348        }
349        object
350    }
351
352    /// The same, reading the fields for the build, so what depends on them follows.
353    pub fn watched(&self) -> Object {
354        let mut object = self.original.clone();
355        for (name, value) in &self.fields {
356            object.set(name, value.get());
357        }
358        object
359    }
360
361    pub fn is_dirty(&self) -> bool {
362        self.fields.iter().any(|(name, value)| *value.peek() != self.original.get(name))
363    }
364}

503    fn linked(inner: SessionInner) -> Self {
504        let is_first = inner.list_editors.borrow().is_empty();
505        let inner = Rc::new(inner);
506        let frame = inner.frame.clone();
507        frame.set_extension(Rc::new(SessionLink(Rc::downgrade(&inner))));
508        let telling = Rc::downgrade(&inner);
509        frame.set_messenger(move |message| {
510            if let Some(inner) = telling.upgrade() {
511                Session(inner).show_message(message);
512            }
513        });
514        frame.set_validator(|contexts, cx| {
515            let Some(session) = Session::of(cx) else { return Ok(()) };
516            let form = session.form();
517            let records = match (&form, cx.selected.is_empty()) {
518                (Some(form), true) => vec![form.current()],
519                _ => cx.selected.clone(),
520            };
521            let mut messages = Vec::new();
522            for record in &records {
523                for context in contexts {
524                    let found = session.violations_of(record, context);
525                    messages.extend(found.into_iter().filter(Violation::is_error).map(|v| v.message));
526                }
527            }
528            if messages.is_empty() { Ok(()) } else { Err(messages.join("; ")) }
529        });
530        let session = Self(inner);
531        if is_first {
532            crate::list_editors::register(&session);
533        }
534        session
535    }

948    /// A new record with what this one holds; what belongs to it goes in `pending`, pointing to it.
949    fn copy_of(
950        &self,
951        entity: &EntityInfo,
952        source: &Object,
953        pending: &mut Vec<Object>,
954    ) -> Result<Object, String> {
955        let space = self.0.frame.space();
956        let mut copy = space.borrow_mut().new_object(&entity.name).map_err(|e| e.to_string())?;
957        let key_name = entity.key_field().map(|key| key.name.clone()).unwrap_or_default();
958        let copied = entity.fields.iter().filter(|field| {
959            !field.is_key
960                && !field.non_cloneable
961                && field.calculated.is_none()
962                && !matches!(field.kind, FieldKind::Collection(_))
963        });
964        for field in copied {
965            copy.set(&field.name, source.get(&field.name));
966        }
967        for field in entity.fields.iter().filter(|field| field.aggregated) {
968            let FieldKind::Collection(target) = &field.kind else { continue };
969            let Some(child) = self.entity(target) else { continue };
970            let named = field.back_reference.as_deref();
971            let Some(back) = crate::nested::back_reference(self, entity, target, named) else { continue };
972            let filter = Criteria::property(&back).equals(Criteria::value(source.get(&key_name)));
973            let found = space.borrow().query(target, &Query::all().filter(filter));
974            for record in found.map_err(|e| e.to_string())? {
975                let mut child_copy = self.copy_of(&child, &record, pending)?;
976                child_copy.set(&back, copy.get(&key_name));
977                pending.push(child_copy);
978            }
979        }
980        Ok(copy)
981    }

1109    /// What the form fails, once a save has found something. Follows the fields.
1110    pub(crate) fn form_violations(&self) -> Vec<Violation> {
1111        let Some(form) = self.form() else { return Vec::new() };
1112        let found = |record: &Object| self.violations_of(record, CONTEXT_SAVE);
1113        if self.0.checking.get() || self.0.noting.get() {
1114            return found(&form.watched());
1115        }
1116        if !form.entity.checks_on_leave {
1117            return Vec::new();
1118        }
1119        let left = self.0.left.get();
1120        if left.is_empty() {
1121            return Vec::new();
1122        }
1123        // Only what the person already went through: what is still ahead is not an error yet.
1124        let mut all = found(&form.watched());
1125        all.retain(|violation| violation.field.as_ref().is_some_and(|field| left.contains(field)));
1126        all
1127    }

1358    /// Keeps what the form holds and then runs `then`. What only warns asks first, and so does a
1359    /// field someone else changed too, so `then` may run later, when the person answers, or never.
1360    pub(crate) fn save_then(&self, then: impl Fn(&Self) + 'static) -> Result<(), String> {
1361        let Some(form) = self.0.form.peek().clone() else { return Ok(()) };
1362        let object = form.current();
1363        let found = self.violations_of(&object, CONTEXT_SAVE);
1364        let errors: Vec<String> = found.iter().filter(|v| v.is_error()).map(|v| v.message.clone()).collect();
1365        if !errors.is_empty() {
1366            self.0.error.set(None);
1367            self.0.checking.set(true);
1368            return Err(errors.join("; "));
1369        }
1370        self.0.checking.set(false);
1371        let then: Rc<dyn Fn(&Self)> = Rc::new(then);
1372        let warnings: Vec<String> =
1373            found.iter().filter(|v| v.severity == Severity::Warning).map(|v| v.message.clone()).collect();
1374        if warnings.is_empty() {
1375            return self.save_merged(&then);
1376        }
1377        let session = self.clone();
1378        let words = self.words();
1379        self.0.confirm.ask(
1380            words.warning_question.clone(),
1381            warnings.join("\n"),
1382            words.warning_answer.clone(),
1383            move || {
1384                // What fails goes on the screen.
1385                let _ = session.save_merged(&then);
1386            },
1387        );
1388        Ok(())
1389    }

1391    /// Puts together what the form changed and what someone else kept since it opened. Different
1392    /// fields both stand; the same field asks first.
1393    fn save_merged(&self, then: &Rc<dyn Fn(&Self)>) -> Result<(), String> {
1394        let Some(form) = self.0.form.peek().clone() else { return Ok(()) };
1395        let object = form.current();
1396        if form.is_new || form.entity.transient {
1397            return self.keep(&form, object, then);
1398        }
1399        let key = form.entity.key_field().map_or(Value::Null, |field| object.get(&field.name));
1400        let stored = self.0.frame.space().borrow().find(&form.entity.name, &key);
1401        let stored = match stored {
1402            Ok(stored) => stored,
1403            Err(error) => {
1404                let message = error.to_string();
1405                self.0.error.set(Some(message.clone()));
1406                return Err(message);
1407            }
1408        };
1409        let Some(stored) = stored else {
1410            let message = self.words().record_gone.clone();
1411            self.0.error.set(Some(message.clone()));
1412            return Err(message);
1413        };
1414        let merged = merge_changes(&form.entity, &form.original, &object, &stored);
1415        if merged.conflicts.is_empty() {
1416            return self.keep(&form, merged.record, then);
1417        }
1418        let names: Vec<&str> = merged
1419            .conflicts
1420            .iter()
1421            .filter_map(|name| form.entity.field_named(name))
1422            .map(|field| field.caption.as_str())
1423            .collect();
1424        let (session, then, record) = (self.clone(), then.clone(), merged.record);
1425        let (words, kept_form) = (self.words(), form.clone());
1426        self.0.confirm.ask(
1427            words.conflict_question.clone(),
1428            words.conflict_detail.replace("{0}", &names.join(", ")),
1429            words.conflict_answer.clone(),
1430            move || {
1431                // What fails goes on the screen.
1432                let _ = session.keep(&kept_form, record.clone(), &then);
1433            },
1434        );
1435        Ok(())
1436    }

1438    /// Writes the record, with what comes along with it.
1439    fn keep(&self, form: &Rc<FormState>, object: Object, then: &Rc<dyn Fn(&Self)>) -> Result<(), String> {
1440        let has_notes = self.violations_of(&object, CONTEXT_SAVE).iter().any(|v| !v.is_error());
1441        self.0.noting.set(has_notes);
1442        if form.entity.transient {
1443            self.0.error.set(None);
1444            self.0.form.set(Some(Rc::new(FormState::new(form.entity.clone(), object, false))));
1445            self.touch();
1446            then(self);
1447            return Ok(());
1448        }
1449        let space = self.0.frame.space();
1450        let done = {
1451            let mut space = space.borrow_mut();
1452            let mut kept =
1453                if form.is_new { space.insert(object.clone()) } else { space.save(object.clone()) };
1454            // After the record, which they point to.
1455            for record in self.0.pending.borrow_mut().drain(..) {
1456                kept = kept.and_then(|()| space.insert(record));
1457            }
1458            kept.and_then(|()| space.commit())
1459        };
1460        match done {
1461            Ok(()) => {
1462                self.0.error.set(None);
1463                let saved = form.entity.key_field().map(|f| object.get(&f.name));
1464                self.touch();
1465                if let Some(key) = saved {
1466                    // The same record, no longer new or changed.
1467                    self.remember_siblings(&form.entity);
1468                    let found = space.borrow().find(&form.entity.name, &key).ok().flatten();
1469                    let fresh = found.unwrap_or(object);
1470                    self.0.frame.set_selected(vec![fresh.clone()]);
1471                    self.0.form.set(Some(Rc::new(FormState::new(form.entity.clone(), fresh, false))));
1472                    self.update_position(&form.entity);
1473                }
1474                then(self);
1475                Ok(())
1476            }
1477            Err(error) => {
1478                // What it could not keep must not stay waiting for the next try.
1479                space.borrow_mut().rollback();
1480                let message = error.to_string();
1481                self.0.error.set(Some(message.clone()));
1482                Err(message)
1483            }
1484        }
1485    }

1515    /// Takes away these records and keeps that, in the frame's unit of work.
1516    pub(crate) fn delete(&self, objects: &[Object]) -> Result<(), String> {

1518            objects.iter().flat_map(|object| self.violations_of(object, CONTEXT_DELETE)).collect();
1519        let refused: Vec<String> = found.iter().filter(|v| v.is_error()).map(|v| v.message.clone()).collect();
1520        if !refused.is_empty() {

1522        }
1523        let warnings: Vec<String> =
1524            found.iter().filter(|v| v.severity == Severity::Warning).map(|v| v.message.clone()).collect();
1525        if warnings.is_empty() {

1527        }
1528        let (session, objects, words) = (self.clone(), objects.to_vec(), self.words());
1529        self.0.confirm.ask(
1530            words.delete_question.clone(),

1542        let space = self.0.frame.space();
1543        let done = {
1544            let mut space = space.borrow_mut();
1545            let mut result = Ok(());
1546            for object in objects {
1547                let Some(entity) = self.entity(object.entity()) else { continue };
1548                let Some(field) = entity.key_field() else { continue };
1549                result = result.and_then(|()| space.delete(&entity.name, &object.get(&field.name)));
1550            }

1554            Ok(()) => {

1557                if leaving_form {
1558                    self.back_now();
1559                } else {
1560                    self.0.frame.set_selected(Vec::new());
1561                }
1562                Ok(())

1564            Err(error) => {
1565                space.borrow_mut().rollback();
1566                let message = error.to_string();
1567                self.0.error.set(Some(message.clone()));

1569            }

1619    /// Runs an action, asking first about what its contexts only warn of.
1620    fn execute_after_warnings(&self, action: &Action, input: ActionInput) {
1621        let selected = self.0.frame.selected();
1622        let records = match (self.form(), selected.is_empty()) {
1623            (Some(form), true) => vec![form.current()],
1624            _ => selected,
1625        };
1626        let warnings: Vec<String> = records
1627            .iter()
1628            .flat_map(|record| {
1629                action.contexts().iter().flat_map(move |context| self.violations_of(record, context))
1630            })
1631            .filter(|violation| violation.severity == Severity::Warning)
1632            .map(|violation| violation.message)
1633            .collect();
1634        if warnings.is_empty() {
1635            return self.execute(action.id(), input);
1636        }
1637        let (session, id, words) = (self.clone(), action.id().to_owned(), self.words());
1638        self.0.confirm.ask(
1639            words.warning_question.clone(),
1640            warnings.join("\n"),
1641            words.warning_answer.clone(),

1643        );

1657                frame.space().borrow().find(&entity.name, &key).ok().flatten()
1658            })
1659            .collect();
1660        frame.set_selected(fresh);
1661        self.touch();
1662        let form = self.0.form.peek().clone();
1663        if let Some(form) = form.filter(|form| !form.is_new && !form.is_dirty())
1664            && let Some(field) = form.entity.key_field()
