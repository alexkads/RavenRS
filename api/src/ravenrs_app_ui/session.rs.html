261/// The record a form is editing: what it was, and one state per field.
262#[derive(Debug)]
263pub(crate) struct FormState {
264    pub entity: EntityInfo,
265    pub original: Object,
266    pub is_new: bool,
267    pub fields: BTreeMap<String, State<Value>>,
268}
269
270#[cfg_attr(feature = "live", ravenrs_live::expose)]
271impl FormState {
272    fn new(entity: EntityInfo, original: Object, is_new: bool) -> Self {
273        let fields = entity
274            .fields
275            .iter()
276            .filter(|f| f.calculated.is_none())
277            .map(|f| (f.name.clone(), State::new(original.get(&f.name))))
278            .collect();
279        Self { entity, original, is_new, fields }
280    }
281
282    /// The record with what the fields hold now, without reading them for the build.
283    pub fn current(&self) -> Object {
284        let mut object = self.original.clone();
285        for (name, value) in &self.fields {
286            object.set(name, value.peek().clone());
287        }
288        object
289    }
290
291    /// The same, reading the fields for the build, so what depends on them follows.
292    pub fn watched(&self) -> Object {
293        let mut object = self.original.clone();
294        for (name, value) in &self.fields {
295            object.set(name, value.get());
296        }
297        object
298    }
299
300    pub fn is_dirty(&self) -> bool {
301        self.fields.iter().any(|(name, value)| *value.peek() != self.original.get(name))
302    }
303}

451    fn linked(inner: SessionInner) -> Self {
452        let is_first = inner.list_editors.borrow().is_empty();
453        let inner = Rc::new(inner);
454        let frame = inner.frame.clone();
455        frame.set_extension(Rc::new(SessionLink(Rc::downgrade(&inner))));
456        // Declared actions say what is wrong in the application's words.
457        frame.set_extension(Rc::new(inner.words.rules.clone()));
458        let telling = Rc::downgrade(&inner);
459        frame.set_messenger(move |message| {
460            if let Some(inner) = telling.upgrade() {
461                Session(inner).show_message(message);
462            }
463        });
464        frame.set_validator(|contexts, cx| {
465            let Some(session) = Session::of(cx) else { return Ok(()) };
466            let form = session.form();
467            let records = match (&form, cx.selected.is_empty()) {
468                (Some(form), true) => vec![form.current()],
469                _ => cx.selected.clone(),
470            };
471            let mut messages = Vec::new();
472            for record in &records {
473                for context in contexts {
474                    let found = session.violations_of(record, context);
475                    messages.extend(found.into_iter().filter(Violation::is_error).map(|v| v.message));
476                }
477            }
478            if messages.is_empty() { Ok(()) } else { Err(messages.join("; ")) }
479        });
480        let session = Self(inner);
481        if is_first {
482            crate::list_editors::register(&session);
483        }
484        session
485    }

557    /// Tells people something, which goes away by itself.
558    pub fn show_message(&self, message: Message) {
559        let tone = match message.kind {
560            MessageKind::Success => Tone::Success,
561            MessageKind::Warning => Tone::Warning,
562            MessageKind::Error => Tone::Danger,
563            _ => Tone::Accent,
564        };
565        let mut extras = ToastExtras::new();
566        if let Some((label, run)) = message.action {
567            extras = extras.action(label, move || run());
568        }
569        if let Some(run) = message.on_close {
570            extras = extras.on_close(move || run());
571        }
572        self.0.toasts.show_with(self.title(), message.text, tone, message.seconds.unwrap_or(4.0), extras);
573    }

901    /// A new record with what this one holds; what belongs to it goes in `pending`, pointing to it.
902    fn copy_of(
903        &self,
904        entity: &EntityInfo,
905        source: &Object,
906        pending: &mut Vec<Object>,
907    ) -> Result<Object, String> {
908        let space = self.0.frame.space();
909        let mut copy = space.borrow_mut().new_object(&entity.name).map_err(|e| e.to_string())?;
910        let key_name = entity.key_field().map(|key| key.name.clone()).unwrap_or_default();
911        let copied = entity.fields.iter().filter(|field| {
912            !field.is_key
913                && !field.non_cloneable
914                && field.calculated.is_none()
915                && !matches!(field.kind, FieldKind::Collection(_))
916        });
917        for field in copied {
918            copy.set(&field.name, source.get(&field.name));
919        }
920        for field in entity.fields.iter().filter(|field| field.aggregated) {
921            let FieldKind::Collection(target) = &field.kind else { continue };
922            let Some(child) = self.entity(target) else { continue };
923            let named = field.back_reference.as_deref();
924            let Some(back) = crate::nested::back_reference(self, entity, target, named) else { continue };
925            let filter = Criteria::property(&back).equals(Criteria::value(source.get(&key_name)));
926            let found = space.borrow().query(target, &Query::all().filter(filter));
927            for record in found.map_err(|e| e.to_string())? {
928                let mut child_copy = self.copy_of(&child, &record, pending)?;
929                child_copy.set(&back, copy.get(&key_name));
930                pending.push(child_copy);
931            }
932        }
933        Ok(copy)
934    }

1098    /// What the form fails, once a save has found something. Follows the fields.
1099    pub(crate) fn form_violations(&self) -> Vec<Violation> {
1100        let Some(form) = self.form() else { return Vec::new() };
1101        let found = |record: &Object| self.violations_of(record, CONTEXT_SAVE);
1102        if self.0.checking.get() || self.0.noting.get() {
1103            return found(&form.watched());
1104        }
1105        if !form.entity.checks_on_leave {
1106            return Vec::new();
1107        }
1108        let left = self.0.left.get();
1109        if left.is_empty() {
1110            return Vec::new();
1111        }
1112        // Only what the person already went through: what is still ahead is not an error yet.
1113        let mut all = found(&form.watched());
1114        all.retain(|violation| violation.field.as_ref().is_some_and(|field| left.contains(field)));
1115        all
1116    }

1350    /// Keeps what the form holds and then runs `then`. What only warns asks first, and so does a
1351    /// field someone else changed too, so `then` may run later, when the person answers, or never.
1352    pub(crate) fn save_then(&self, then: impl Fn(&Self) + 'static) -> Result<(), String> {
1353        let Some(form) = self.0.form.peek().clone() else { return Ok(()) };
1354        let object = form.current();
1355        let found = self.violations_of(&object, CONTEXT_SAVE);
1356        let errors: Vec<String> = found.iter().filter(|v| v.is_error()).map(|v| v.message.clone()).collect();
1357        if !errors.is_empty() {
1358            self.0.error.set(None);
1359            self.0.checking.set(true);
1360            let why = errors.join("; ");
1361            self.0.announcer.assertive(self.access_words().not_saved(&why));
1362            return Err(why);
1363        }
1364        self.0.checking.set(false);
1365        let then: Rc<dyn Fn(&Self)> = Rc::new(then);
1366        let warnings: Vec<String> =
1367            found.iter().filter(|v| v.severity == Severity::Warning).map(|v| v.message.clone()).collect();
1368        if warnings.is_empty() {
1369            return self.save_merged(&then);
1370        }
1371        let session = self.clone();
1372        let words = self.words();
1373        self.0.confirm.ask(
1374            words.warning_question.clone(),
1375            warnings.join("\n"),
1376            words.warning_answer.clone(),
1377            move || {
1378                // What fails goes on the screen.
1379                let _ = session.save_merged(&then);
1380            },
1381        );
1382        Ok(())
1383    }

1385    /// Puts together what the form changed and what someone else kept since it opened. Different
1386    /// fields both stand; the same field asks first.
1387    fn save_merged(&self, then: &Rc<dyn Fn(&Self)>) -> Result<(), String> {
1388        let Some(form) = self.0.form.peek().clone() else { return Ok(()) };
1389        let object = form.current();
1390        if form.is_new || form.entity.transient {
1391            return self.keep(&form, object, then);
1392        }
1393        let key = form.entity.key_field().map_or(Value::Null, |field| object.get(&field.name));
1394        let stored = self.0.frame.space().borrow().find(&form.entity.name, &key);
1395        let stored = match stored {
1396            Ok(stored) => stored,
1397            Err(error) => {
1398                let message = error.to_string();
1399                self.0.error.set(Some(message.clone()));
1400                return Err(message);
1401            }
1402        };
1403        let Some(stored) = stored else {
1404            let message = self.words().record_gone.clone();
1405            self.0.error.set(Some(message.clone()));
1406            return Err(message);
1407        };
1408        let merged = merge_changes(&form.entity, &form.original, &object, &stored);
1409        if merged.conflicts.is_empty() {
1410            return self.keep(&form, merged.record, then);
1411        }
1412        let names: Vec<&str> = merged
1413            .conflicts
1414            .iter()
1415            .filter_map(|name| form.entity.field_named(name))
1416            .map(|field| field.caption.as_str())
1417            .collect();
1418        let (session, then, record) = (self.clone(), then.clone(), merged.record);
1419        let (words, kept_form) = (self.words(), form.clone());
1420        self.0.confirm.ask(
1421            words.conflict_question.clone(),
1422            words.conflict_detail.replace("{0}", &names.join(", ")),
1423            words.conflict_answer.clone(),
1424            move || {
1425                // What fails goes on the screen.
1426                let _ = session.keep(&kept_form, record.clone(), &then);
1427            },
1428        );
1429        Ok(())
1430    }

1432    /// Writes the record, with what comes along with it.
1433    fn keep(&self, form: &Rc<FormState>, object: Object, then: &Rc<dyn Fn(&Self)>) -> Result<(), String> {
1434        let has_notes = self.violations_of(&object, CONTEXT_SAVE).iter().any(|v| !v.is_error());
1435        self.0.noting.set(has_notes);
1436        if form.entity.transient {
1437            self.0.error.set(None);
1438            self.0.form.set(Some(Rc::new(FormState::new(form.entity.clone(), object, false))));
1439            self.touch();
1440            then(self);
1441            return Ok(());
1442        }
1443        let space = self.0.frame.space();
1444        let done = {
1445            let mut space = space.borrow_mut();
1446            let mut kept =
1447                if form.is_new { space.insert(object.clone()) } else { space.save(object.clone()) };
1448            // After the record, which they point to.
1449            for record in self.0.pending.borrow_mut().drain(..) {
1450                kept = kept.and_then(|()| space.insert(record));
1451            }
1452            kept.and_then(|()| space.commit())
1453        };
1454        match done {
1455            Ok(()) => {
1456                self.0.error.set(None);
1457                self.0.announcer.polite(self.access_words().saved(&Self::spoken_name(&form.entity, &object)));
1458                let saved = form.entity.key_field().map(|f| object.get(&f.name));
1459                self.touch();
1460                if let Some(key) = saved {
1461                    // The same record, no longer new or changed.
1462                    self.remember_siblings(&form.entity);
1463                    let found = space.borrow().find(&form.entity.name, &key).ok().flatten();
1464                    let fresh = found.unwrap_or(object);
1465                    self.0.frame.set_selected(vec![fresh.clone()]);
1466                    self.0.form.set(Some(Rc::new(FormState::new(form.entity.clone(), fresh, false))));
1467                    self.update_position(&form.entity);
1468                }
1469                then(self);
1470                Ok(())
1471            }
1472            Err(error) => {
1473                // What it could not keep must not stay waiting for the next try.
1474                space.borrow_mut().rollback();
1475                let message = error.to_string();
1476                self.0.error.set(Some(message.clone()));
1477                Err(message)
1478            }
1479        }
1480    }

1510    /// Takes away these records and keeps that, in the frame's unit of work.
1511    pub(crate) fn delete(&self, objects: &[Object]) -> Result<(), String> {

1615    /// Runs an action, asking first about what its contexts only warn of.
1616    fn execute_after_warnings(&self, action: &Action, input: ActionInput) {
1617        let selected = self.0.frame.selected();
1618        let records = match (self.form(), selected.is_empty()) {
1619            (Some(form), true) => vec![form.current()],
1620            _ => selected,
1621        };
1622        let warnings: Vec<String> = records
1623            .iter()
1624            .flat_map(|record| {
1625                action.contexts().iter().flat_map(move |context| self.violations_of(record, context))
1626            })
1627            .filter(|violation| violation.severity == Severity::Warning)
1628            .map(|violation| violation.message)
1629            .collect();

1631            return self.execute(action.id(), input);

1633        let (session, id, words) = (self.clone(), action.id().to_owned(), self.words());
1634        self.0.confirm.ask(
