12pub(crate) type Submit = Rc<dyn Fn(Option<&str>, &str) -> Result<(), PasswordError>>;

14/// Asks for a new password, twice, and for the current one when that is what it takes.
15#[derive(Clone)]
16pub(crate) struct PasswordForm {
17    words: Words,
18    current: Option<State<String>>,
19    new: State<String>,
20    repeated: State<String>,
21    error: State<Option<String>>,
22    submit: Submit,
23    done: Rc<dyn Fn()>,
24}
25
26#[cfg_attr(feature = "live", ravenrs_live::expose)]
27impl PasswordForm {
28    pub(crate) fn new(words: Words, asks_current: bool, submit: Submit, done: Rc<dyn Fn()>) -> Self {
29        Self {
30            words,
31            current: asks_current.then(|| State::new(String::new())),
32            new: State::new(String::new()),
33            repeated: State::new(String::new()),
34            error: State::new(None),
35            submit,
36            done,
37        }
38    }
39
40    fn send(&self) {
41        let (new, repeated) = (self.new.peek().clone(), self.repeated.peek().clone());
42        if new != repeated {
43            self.error.set(Some(self.words.passwords_differ.clone()));
44            return;
45        }
46        let current = self.current.as_ref().map(|state| state.peek().clone());
47        match (self.submit)(current.as_deref(), &new) {
48            Ok(()) => {
49                for state in
50                    [Some(&self.new), Some(&self.repeated), self.current.as_ref()].into_iter().flatten()
51                {
52                    state.set(String::new());
53                }
54                self.error.set(None);
55                (self.done)();
56            }
57            Err(error) => self.error.set(Some(message_of(&self.words, &error))),
58        }
59    }
60}
61
62impl View for PasswordForm {
63    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
64        let words = &self.words;
65        let sending = self.clone();
66        let current: Option<AnyView> = self.current.as_ref().map(|state| {
67            Box::new(FormField::new(
68                words.current_password.clone(),
69                TextField::new(text_binding(state)).secure().autofocus(),
70            )) as AnyView
71        });
72        let form = self.clone();
73        let content = VStack::new((
74            self.error.get().map(|error| Alert::new(words.change_password.clone(), error).tone(Tone::Danger)),
75            current,
76            FormField::new(words.new_password.clone(), TextField::new(text_binding(&self.new)).secure()),
77            FormField::new(
78                words.repeat_password.clone(),
79                TextField::new(text_binding(&self.repeated)).secure().on_submit(move |_| form.send()),
80            ),
81            Button::new(words.save_password.clone()).on_click(move || sending.send()),
82        ))
83        .spacing(12.0)
84        .align(Align::Stretch);
85        cx.child(&content)
86    }
87}

89/// Shows the person's own way of changing their password.
90pub(crate) fn show_change_password(session: &Session) {
91    let words = session.words().clone();
92    let (context, shown) = (session.frame().security(), session.clone());
93    let submit: Submit = Rc::new(move |current, password| match &context {
94        Some(context) => context.change_own_password(current.unwrap_or(""), password),
95        None => Err(PasswordError::NotAllowed),
96    });
97    let told = words.password_saved.clone();
98    let done: Rc<dyn Fn()> = Rc::new(move || {
99        shown.close_presented();
100        shown.show_message(Message::new(told.clone(), MessageKind::Success));
101    });
102    let form = PasswordForm::new(words.clone(), true, submit, done);
103    session.present(words.change_password, move || Box::new(form.clone()) as AnyView);
104}

106pub(crate) fn message_of(words: &Words, error: &PasswordError) -> String {
107    match error {
108        PasswordError::TooShort => words.password_too_short.clone(),
109        PasswordError::WrongCurrent => words.password_wrong_current.clone(),
110        PasswordError::SameAsCurrent => words.password_same.clone(),
111        other => other.to_string(),
112    }
113}

115pub(crate) fn text_binding(state: &State<String>) -> Binding<String> {
116    let (read, write) = (state.clone(), state.clone());
117    Binding::new(move || read.get(), move |value| write.set(value))
