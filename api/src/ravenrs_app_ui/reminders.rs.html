2//! to be built again at the next time due, and reads again when records change.

17const SNOOZE_SECONDS: i64 = 15 * 60;
18
19fn seconds_of(when: DateTime) -> i64 {
20    when.days * 86_400 + i64::from(when.seconds)
21}
22
23fn moment_at(seconds: i64) -> DateTime {
24    DateTime { days: seconds.div_euclid(86_400), seconds: seconds.rem_euclid(86_400) as u32 }
25}
26
27/// A record with a time set, as this person reads it.
28#[derive(Clone)]
29struct Pending {
30    entity: EntityInfo,
31    record: Object,
32    key: String,
33    at: i64,
34}
35
36fn pending(session: &Session) -> Vec<Pending> {
37    let mut all = Vec::new();
38    let space = session.frame().space();
39    let space = space.borrow();
40    let security = session.frame().security();
41    for entity in session.entities_with_reminders() {
42        let Some(reminder) = &entity.reminder else { continue };
43        let found = match &security {
44            Some(security) => security.with(|p| reminders_of(&*space, &entity, p.environment())),
45            None => reminders_of(&*space, &entity, &ravenrs_app::environment()),
46        };
47        for record in found {
48            let Value::Date(when) = record.get(&reminder.field) else { continue };
49            let key = entity
50                .key_field()
51                .and_then(|field| ravenrs_app::key_text(&record.get(&field.name)))
52                .unwrap_or_default();
53            all.push(Pending {
54                key: format!("{}:{key}", entity.name),
55                at: seconds_of(when),
56                entity: entity.clone(),
57                record,
58            });
59        }
60    }
61    all
62}

64/// Changes when a record reminds, in a commit of its own. A person who cannot write only stops
65/// seeing it for now.
66fn set_time(session: &Session, item: &Pending, to: Value, hidden: &State<BTreeSet<String>>) {
67    let Some(reminder) = &item.entity.reminder else { return };
68    let Some(field) = item.entity.key_field() else { return };
69    let space = session.frame().space();
70    let mut space = space.borrow_mut();
71    let result = (|| {
72        let key = item.record.get(&field.name);
73        let Some(mut stored) = space.find(&item.entity.name, &key)? else { return Ok(()) };
74        stored.set(&reminder.field, to);
75        space.save(stored)?;
76        space.commit()
77    })();
78    if let Err(error) = result {
79        space.rollback();
80        drop(space);
81        session.show_message(Message::new(error.to_string(), MessageKind::Warning));
82        let key = item.key.clone();
83        hidden.update(|all| {
84            all.insert(key);
85        });
86    } else {
87        drop(space);
88    }
89    session.records_changed();
90}

92/// What is due now, in a window, and when to look again.
93pub(crate) struct Reminders {
94    session: Session,
95}
96
97impl Reminders {
98    pub(crate) fn new(session: Session) -> Self {
99        Self { session }
100    }
101
102    fn row(&self, item: &Pending, hidden: &State<BTreeSet<String>>) -> AnyView {
103        let session = &self.session;
104        let words = session.words();
105        let reminder = item.entity.reminder.as_ref();
106        let field = reminder
107            .and_then(|reminder| reminder.message.as_deref())
108            .and_then(|name| item.entity.field_named(name))
109            .or_else(|| item.entity.default_property());
110        let what = field.map_or_else(String::new, |field| display(field, &item.record.get(&field.name)));
111        let (opener, opened) = (session.clone(), item.clone());
112        let open = Button::new(words.open.clone()).size(ButtonSize::Small).on_click(move || {
113            opener.reminders().update(|(_, open)| *open = false);
114            if let Some(field) = opened.entity.key_field() {
115                opener.open_detail(&opened.entity.name, &opened.record.get(&field.name));
116            }
117        });
118        let (later, snoozed, keep) = (session.clone(), item.clone(), hidden.clone());
119        let snooze = Button::new(words.snooze.clone())
120            .style(ButtonStyle::Secondary)
121            .size(ButtonSize::Small)
122            .on_click(move || {
123                let at = seconds_of(local_now()) + SNOOZE_SECONDS;
124                set_time(&later, &snoozed, Value::Date(moment_at(at)), &keep);
125            });
126        let (ending, ended, keep) = (session.clone(), item.clone(), hidden.clone());
127        let dismiss = Button::new(words.dismiss.clone())
128            .style(ButtonStyle::Ghost)
129            .size(ButtonSize::Small)
130            .on_click(move || set_time(&ending, &ended, Value::Null, &keep));
131        Box::new(
132            HStack::new((
133                VStack::new((Text::new(what), Text::new(item.entity.caption.clone()))).spacing(2.0).grow(1.0),
134                open,
135                snooze,
136                dismiss,
137            ))
138            .spacing(8.0)
139            .align(Align::Center),
140        )
141    }
142}

144impl View for Reminders {
145    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
146        let session = &self.session;
147        let revision = session.revision();
148        let read: State<Option<(u64, Vec<Pending>)>> = cx.use_state(|| None);
149        let hidden: State<BTreeSet<String>> = cx.use_state(BTreeSet::new);
150        let seen: State<BTreeSet<String>> = cx.use_state(BTreeSet::new);
151        let known = read.peek().as_ref().filter(|(at, _)| *at == revision).map(|(_, all)| all.clone());
152        let all = known.unwrap_or_else(|| {
153            let all = pending(session);
154            read.set(Some((revision, all.clone())));
155            all
156        });
157        let now = seconds_of(local_now());
158        let left = hidden.get();
159        let due: Vec<&Pending> =
160            all.iter().filter(|item| item.at <= now && !left.contains(&item.key)).collect();
161        if let Some(next) = all.iter().map(|item| item.at).filter(|at| *at > now).min() {
162            // A whole second later, so it is due when this comes back.
163            cx.wake_at(cx.now() + (next - now) as f64 + 1.0);
164        }
165        let (count, open) = *session.reminders().peek();
166        // Something new comes up by itself; what was closed stays closed.
167        let keys: BTreeSet<String> = due.iter().map(|item| item.key.clone()).collect();
168        let arrived = keys.iter().any(|key| !seen.peek().contains(key));
169        let open = (open || arrived) && !due.is_empty();
170        if count != due.len() || open != session.reminders().peek().1 {
171            session.reminders().set((due.len(), open));
172        }
173        if *seen.peek() != keys {
174            seen.set(keys);
175        }
176        if !open {
177            return cx.empty();
178        }
179        let (reading, writing) = (session.reminders().clone(), session.reminders().clone());
180        let binding = Binding::new(
181            move || reading.get().1,
182            move |open: bool| writing.update(|(_, showing)| *showing = open),
183        );
184        let rows: Vec<AnyView> = due.iter().map(|item| self.row(item, &hidden)).collect();
185        let dialog = Dialog::new(binding, session.words().reminders.clone())
186            .content(VStack::new(rows).spacing(12.0).align(Align::Stretch));
187        cx.child(&dialog)
188    }
189}
