1//! Keys written as text in the model, such as `command+s`, read as a [`Hotkey`].
2
3use ravenrs_core::Key;
4use ravenrs_widgets::Hotkey;
5
6/// Reads keys such as `command+shift+s`, `alt+c` or `f9`.
7pub(crate) fn hotkey_of(text: &str) -> Option<Hotkey> {
8    let mut command = false;
9    let mut shift = false;
10    let mut alt = false;
11    let mut key = None;
12    for part in text.split('+') {
13        match part.trim().to_lowercase().as_str() {
14            "command" | "cmd" | "ctrl" => command = true,
15            "shift" => shift = true,
16            "alt" | "option" => alt = true,
17            "delete" => key = Some(Key::Delete),
18            "enter" => key = Some(Key::Enter),
19            "esc" | "escape" => key = Some(Key::Escape),
20            other => {
21                let function = other.strip_prefix('f').and_then(|n| n.parse::<u8>().ok());
22                key = Some(match function {
23                    Some(number) if (1..=24).contains(&number) => Key::Function(number),
24                    _ => Key::Character(other.to_owned()),
25                });
26            }
27        }
28    }
29    let key = key?;
30    let hotkey = match (&key, command, alt) {
31        (Key::Character(letter), true, _) => Hotkey::command(letter),
32        (Key::Character(letter), false, true) => Hotkey::alt(letter),
33        _ => Hotkey::key(key),
34    };
35    Some(if shift { hotkey.shift() } else { hotkey })
36}

38#[cfg(test)]
39mod tests {
40    use ravenrs_core::Key;
41    use ravenrs_widgets::Hotkey;
42
43    use super::hotkey_of;
44
45    #[test]
46    fn keys_are_read_from_their_words() {
47        let saved = hotkey_of("command+shift+s");
48        assert_eq!(saved, Some(Hotkey::command("s").shift()));
49        assert_eq!(hotkey_of("Delete"), Some(Hotkey::key(Key::Delete)));
50        assert_eq!(hotkey_of("command+s"), Some(Hotkey::command("s")));
51    }
52}
