17/// The form of the record being edited. What each field uses comes from the model's `Editor`.

30    fn editor(
31        &self,
32        form: &Rc<FormState>,
33        node: &ModelNode,
34        field: &str,
35        state: &FormLook,
36        named_around: bool,
37    ) -> Option<AnyView> {
38        let info = form.entity.field_named(field)?;
39        let item = node.at(&["Items", field]);
40        let caption = item.and_then(|n| n.text("Caption")).unwrap_or(&info.caption).to_owned();
41        let caption =
42            self.session.form_caption().and_then(|words| words(field, &state.record)).unwrap_or(caption);
43        let look = form.entity.appearance_of(AppearanceTarget::Item, field, &self.view, Some(&state.record));
44        let around =
45            form.entity.appearance_of(AppearanceTarget::Layout, field, &self.view, Some(&state.record));
46        if look.visible == Some(false) || around.visible == Some(false) {
47            return None;
48        }
49        if let ravenrs_app::FieldKind::Collection(target) = &info.kind {
50            if let Some(source) = self.session.collection_source(&form.entity.name, field) {
51                let record = form.current();
52                let list = crate::nested::NestedList::given(
53                    self.session.clone(),
54                    target,
55                    Rc::new(move || source(&record)),
56                );
57                return Some(if named_around {
58                    Box::new(list) as AnyView
59                } else {
60                    Box::new(FormField::new(caption, list))
61                });
62            }
63            let key = form.entity.key_field().map(|key| form.original.get(&key.name));
64            let list = crate::nested::NestedList::new(
65                self.session.clone(),
66                &form.entity,
67                (!form.is_new).then_some(key).flatten(),
68                target,
69                info.back_reference.as_deref(),
70                info.lookup_filter.as_deref(),
71            );
72            return Some(if named_around {
73                Box::new(list) as AnyView
74            } else {
75                Box::new(FormField::new(caption, list))
76            });
77        }
78        let mut read_only = item.and_then(|n| n.flag("ReadOnly")).unwrap_or(info.read_only);
79        if let Some(security) = self.session.frame().security() {
80            let (name, record) = (&form.entity.name, Some(&state.record));
81            let (can_read, can_write) = security.with(|p| {
82                (
83                    p.can_field(ravenrs_app::Operation::Read, name, field, record),
84                    p.can_field(ravenrs_app::Operation::Write, name, field, record),
85                )
86            });
87            if !can_read && !(form.is_new && can_write) {
88                let shown = Text::new(self.session.words().protected_value.clone());
89                return Some(Box::new(FormField::new(caption, shown)));
90            }
91            read_only |= !can_write;
92        }
93        let value = match form.fields.get(field) {
94            Some(value) => value.clone(),
95            None => {
96                // A calculated field follows the others.
97                let mut current = state.record.clone();
98                form.entity.compute(&mut current, &ravenrs_app::environment());
99                ravenrs_reactive::State::new(current.get(field))
100            }
101        };
102        let request = EditorRequest {
103            field: info.clone(),
104            value,
105            caption: caption.clone(),
106            read_only: read_only || (info.is_key && !form.is_new) || look.enabled == Some(false),
107            session: self.session.clone(),
108        };
109        let name = item.and_then(|n| n.text("Editor")).unwrap_or("");
110        let editor = self.session.editors().editor(name, &request);
111        // A check box says its own name.
112        if matches!(info.kind, ravenrs_app::FieldKind::Boolean) {
113            let (error, note) = (state.error_of(field), state.note_of(field));
114            if error.is_none() && note.is_none() {
115                return Some(editor);
116            }
117            let (note, tone) = note.map_or((None, Tone::Neutral), |(note, tone)| (Some(note), tone));
118            return Some(Box::new(FormField::new(caption, editor).error_text(error).note_text(note, tone)));
119        }
120        let required = form
121            .entity
122            .rules
123            .iter()
124            .any(|rule| rule.field.as_deref() == Some(field) && matches!(rule.kind, RuleKind::Required));
125        let mut wrapped = FormField::new(caption, editor).error_text(state.error_of(field));
126        if let Some((note, tone)) = state.note_of(field) {
127            wrapped = wrapped.note_text(Some(note), tone);
128        }
129        if let Some(hint) = &info.hint {
130            wrapped = wrapped.hint(hint.clone());
131        }
132        if required {
133            wrapped = wrapped.required();
134        }
135        if let Some(tint) = look.font.or(around.font) {
136            wrapped = wrapped.label_ink(semantic_of(tint));
137        }
138        if let Some(tint) = look.back.or(around.back) {
139            wrapped = wrapped.tint(semantic_of(tint));
140        }
141        Some(Box::new(wrapped))
142    }

144    fn group(
145        &self,
146        form: &Rc<FormState>,
147        node: &ModelNode,
148        source: &ModelNode,
149        state: &FormLook,
150        in_tab: bool,
151    ) -> FormGroup {
152        let columns = source.integer("Columns").unwrap_or(1).max(1) as usize;
153        // The tab already says it.
154        let title = source.text("Caption").filter(|_| !in_tab);
155        let mut group = match title {
156            Some(title) => FormGroup::titled(title),
157            None => FormGroup::new(),
158        }
159        .columns(columns);
160        for item in source.in_order() {
161            if item.text("Kind") == Some("ActionContainer") {
162                let name = item.text("Category").unwrap_or(&item.id).to_owned();
163                group = group
164                    .wide(crate::ActionContainer::new(self.session.clone(), [ActionCategory::InForm(name)]));
165                continue;
166            }
167            let field = item.text("Field").unwrap_or(&item.id);
168            let tall = item.flag("Wide") == Some(true)
169                || node.at(&["Items", field]).and_then(|n| n.flag("Wide")) == Some(true)
170                || form.entity.field_named(field).is_some_and(|f| {
171                    matches!(
172                        f.kind,
173                        ravenrs_app::FieldKind::LongText
174                            | ravenrs_app::FieldKind::Image
175                            | ravenrs_app::FieldKind::Collection(_)
176                    )
177                });
178            let named_around =
179                in_tab || form.entity.field_named(field).is_some_and(|f| title == Some(f.caption.as_str()));
180            if let Some(editor) = self.editor(form, node, field, state, named_around) {
181                group = if tall { group.wide(editor) } else { group.field(editor) };
182            }
183        }
184        group
185    }
186
187    fn shows(&self, form: &FormState, part: &ModelNode, state: &FormLook) -> bool {
188        form.entity.appearance_of(AppearanceTarget::Layout, &part.id, &self.view, Some(&state.record)).visible
189            != Some(false)
190    }

193fn tone_of(severity: Severity) -> Tone {
194    if severity == Severity::Warning { Tone::Warning } else { Tone::Accent }
195}

197/// What the rules see in this build of a form.
198struct FormLook {
199    record: Object,
200    violations: Vec<Violation>,
201}
202
203impl FormLook {
204    fn error_of(&self, field: &str) -> Option<String> {
205        self.violations
206            .iter()
207            .find(|v| v.is_error() && v.field.as_deref() == Some(field))
208            .map(|v| v.message.clone())
209    }
210
211    /// What the rules only want known about a field.
212    fn note_of(&self, field: &str) -> Option<(String, Tone)> {
213        self.violations
214            .iter()
215            .find(|v| !v.is_error() && v.field.as_deref() == Some(field))
216            .map(|v| (v.message.clone(), tone_of(v.severity)))
217    }
218}

220impl View for DetailView {
221    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
222        let session = &self.session;
223        // The model can change under a form, as with another language.
224        session.revision();
225        let model = session.model();
226        let Some(node) = model.at(&["Views", &self.view.id]) else {
227            return cx.child(&Text::new(self.view.id.clone()));
228        };
229        let Some(form) = session.form() else {
230            return cx.empty();
231        };
232        let state = FormLook { record: form.watched(), violations: session.form_violations() };
233        let lists_all = session.lists_field_errors();
234        let messages: Vec<String> = state
235            .violations
236            .iter()
237            .filter(|v| v.is_error() && (lists_all || v.field.is_none()))
238            .map(|v| v.message.clone())
239            .collect();
240        let notes: Vec<&Violation> =
241            state.violations.iter().filter(|v| !v.is_error() && v.field.is_none()).collect();
242        let note = (!notes.is_empty()).then(|| {
243            let text: Vec<&str> = notes.iter().map(|v| v.message.as_str()).collect();
244            let warns = notes.iter().any(|v| v.severity == Severity::Warning);
245            Alert::message(text.join("\n")).tone(if warns { Tone::Warning } else { Tone::Accent })
246        });
247        let mut layout = FormLayout::new().errors(messages);
248        if let Some(arrangement) = node.get("Layout") {
249            for part in arrangement.in_order() {
250                if !self.shows(&form, part, &state) {
251                    continue;
252                }
253                if part.text("Kind") == Some("Tabs") {
254                    for tab in part.in_order().into_iter().filter(|tab| self.shows(&form, tab, &state)) {
255                        let title = tab.text("Caption").unwrap_or(&tab.id).to_owned();
256                        layout = layout
257                            .tab(title, FormLayout::new().group(self.group(&form, node, tab, &state, true)));
258                    }
259                } else {
260                    layout = layout.group(self.group(&form, node, part, &state, false));
261                }
262            }
263        }
264        match note {
265            Some(note) => cx.child(&VStack::new((note, layout)).spacing(12.0).align(Align::Stretch)),
266            None => cx.child(&layout),
267        }
268    }
