1//! Dates as `YYYY-MM-DD` text, which compare in the right order as they are. The clock stays outside,
2//! so the checks work the same in tests and in the browser.
3
4/// Whether a text is a date that exists in the calendar.
5pub(crate) fn data_valida(texto: &str) -> bool {
6    let bytes = texto.as_bytes();
7    if bytes.len() != 10 || bytes[4] != b'-' || bytes[7] != b'-' {
8        return false;
9    }
10    let numero = |faixa: std::ops::Range<usize>| {
11        let parte = texto.get(faixa)?;
12        if parte.bytes().all(|b| b.is_ascii_digit()) { parte.parse::<u32>().ok() } else { None }
13    };
14    let (Some(ano), Some(mes), Some(dia)) = (numero(0..4), numero(5..7), numero(8..10)) else {
15        return false;
16    };
17    let bissexto = ano % 4 == 0 && (ano % 100 != 0 || ano % 400 == 0);
18    let dias = match mes {
19        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
20        4 | 6 | 9 | 11 => 30,
21        2 if bissexto => 29,
22        2 => 28,
23        _ => return false,
24    };
25    (1..=dias).contains(&dia)
26}

28/// A count of days since 1970-01-01 as a date.
29#[cfg(any(test, not(target_arch = "wasm32")))]
30fn data_de_dias(dias: i64) -> String {
31    // The civil-from-days algorithm, on 400-year eras.
32    let z = dias + 719_468;
33    let era = z.div_euclid(146_097);
34    let doe = z.rem_euclid(146_097);
35    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
36    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
37    let mp = (5 * doy + 2) / 153;
38    let dia = doy - (153 * mp + 2) / 5 + 1;
39    let mes = if mp < 10 { mp + 3 } else { mp - 9 };
40    let ano = yoe + era * 400 + i64::from(mes <= 2);
41    format!("{ano:04}-{mes:02}-{dia:02}")
42}

44/// Today's date in UTC, from the system clock.
45#[cfg(not(target_arch = "wasm32"))]
46pub fn hoje_utc() -> String {
47    let segundos = std::time::SystemTime::now()
48        .duration_since(std::time::UNIX_EPOCH)
49        .map_or(0, |duracao| duracao.as_secs());
50    data_de_dias((segundos / 86_400) as i64)
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56
57    #[test]
58    fn only_real_dates_pass() {
59        for boa in ["2026-11-05", "2024-02-29", "2000-02-29", "2026-12-31"] {
60            assert!(data_valida(boa), "{boa}");
61        }
62        for ruim in [
63            "",
64            "2026-13-01",
65            "2026-02-29",
66            "1900-02-29",
67            "2026-1-05",
68            "2026/11/05",
69            "abcd-ef-gh",
70            "2026-00-10",
71        ] {
72            assert!(!data_valida(ruim), "{ruim}");
73        }
74    }
75
76    #[test]
77    fn days_become_the_right_date() {
78        assert_eq!(data_de_dias(0), "1970-01-01");
79        assert_eq!(data_de_dias(19_782), "2024-02-29");
80        assert_eq!(data_de_dias(11_016), "2000-02-29");
81        assert!(data_valida(&hoje_utc()));
82    }
