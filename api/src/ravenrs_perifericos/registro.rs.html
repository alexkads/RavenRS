15const FORMATO: u32 = 1;
16/// How many events of a driver are kept.
17const MAXIMO_HISTORICO: usize = 200;

19/// What happened to a driver.
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
21#[serde(rename_all = "snake_case")]
22pub enum AcaoHistorico {
23    /// A version arrived.
24    Instalado,
25    /// A version became the one in use.
26    Ativado,
27    /// The one in use went back to the previous.
28    Voltou,
29}

31/// One line of a driver's history.
32#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
33pub struct Evento {
34    /// What happened.
35    pub acao: AcaoHistorico,
36    /// To which version.
37    pub versao: Version,
38}

54/// Why the registry refused.
55#[derive(Clone, Debug, PartialEq, Eq)]
56#[non_exhaustive]
57pub enum ErroRegistro {
58    /// The manifest is not valid.
59    ManifestoInvalido(ErroManifesto),
60    /// The driver asks for a newer program.
61    ProgramaAntigo {
62        /// The driver.
63        id: String,
64        /// Its version.
65        versao: String,
66        /// What it asks for.
67        exige: String,
68        /// What is running.
69        programa: String,
70    },
71    /// This version is installed with different contents. A version never changes.
72    VersaoConflitante {
73        /// The driver.
74        id: String,
75        /// Its version.
76        versao: Version,
77    },
78    /// No driver with this name.
79    DriverDesconhecido(String),
80    /// This driver does not have this version here.
81    VersaoNaoInstalada {
82        /// The driver.
83        id: String,
84        /// The version.
85        versao: Version,
86    },
87    /// There is no previous version to go back to.
88    SemAnterior(String),
89    /// The catalog does not have what was accepted.
90    ForaDoCatalogo {
91        /// The driver.
92        id: String,
93        /// The version.
94        versao: Version,
95    },
96}

98impl fmt::Display for ErroRegistro {
99    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
100        match self {
101            Self::ManifestoInvalido(erro) => write!(f, "driver inválido: {erro}"),
102            Self::ProgramaAntigo { id, versao, exige, programa } => write!(
103                f,
104                "o driver {id} {versao} precisa do programa {exige} ou mais novo; este é o {programa}. \
105                 Atualize o programa antes."
106            ),
107            Self::VersaoConflitante { id, versao } => {
108                write!(f, "o driver {id} {versao} já está instalado com outro conteúdo")
109            }
110            Self::DriverDesconhecido(id) => write!(f, "o driver {id} não está instalado"),
111            Self::VersaoNaoInstalada { id, versao } => {
112                write!(f, "a versão {versao} do driver {id} não está instalada")
113            }
114            Self::SemAnterior(id) => write!(f, "o driver {id} não tem versão anterior para voltar"),
115            Self::ForaDoCatalogo { id, versao } => {
116                write!(f, "o catálogo não traz o driver {id} {versao}")
117            }
118        }
119    }
120}

124#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
125struct DriverInstalado {
126    versoes: Vec<Manifesto>,
127    /// The versions in the order they became active; the last one is in use.
128    ativacoes: Vec<Version>,
129    historico: Vec<Evento>,
130}
131
132impl DriverInstalado {
133    fn registrar(&mut self, acao: AcaoHistorico, versao: &Version) {
134        self.historico.push(Evento { acao, versao: versao.clone() });
135        if self.historico.len() > MAXIMO_HISTORICO {
136            self.historico.remove(0);
137        }
138    }
139
140    fn versao(&self, versao: &Version) -> Option<&Manifesto> {
141        self.versoes.iter().find(|manifesto| manifesto.versao == *versao)
142    }
143}

145#[derive(Serialize, Deserialize)]
146struct EstadoGravado {
147    formato: u32,
148    ultima_sequencia: u64,
149    drivers: BTreeMap<String, DriverInstalado>,

152/// The drivers on this machine.
153#[derive(Clone, Debug, PartialEq)]
154pub struct Registro {
155    programa: Version,
156    ultima_sequencia: u64,
157    drivers: BTreeMap<String, DriverInstalado>,
158}
159
160impl Registro {
161    /// A registry with nothing installed, for this version of the program.
162    pub fn vazio(programa: Version) -> Self {
163        Self { programa, ultima_sequencia: 0, drivers: BTreeMap::new() }
164    }
165
166    /// A registry with the drivers that come with the program.
167    pub fn novo(programa: Version) -> Self {
168        let mut registro = Self::vazio(programa);
169        registro.trazer_embutidos();
170        registro
171    }
172
173    /// Brings in what came with the program. Each new driver starts active, and a newer version of
174    /// one already here stays only installed, so a program update does not change the one in use.
175    fn trazer_embutidos(&mut self) {
176        for manifesto in manifestos_embutidos() {
177            // A saved copy of the same version stays, and one that needs a newer program waits.
178            let _ = self.instalar(manifesto);
179        }
180    }
181
182    /// The registry saved in `armazem`, plus what came with the program. With nothing saved, the same
183    /// as [`Self::novo`]. Every saved manifest is checked again.
184    pub fn carregar(programa: Version, armazem: &dyn Armazem) -> Result<Self, ErroArmazem> {
185        let mut registro = Self::vazio(programa);
186        if let Some(bytes) = armazem.ler()? {
187            let estado: EstadoGravado =
188                serde_json::from_slice(&bytes).map_err(|erro| ErroArmazem::Conteudo(erro.to_string()))?;
189            if estado.formato != FORMATO {
190                return Err(ErroArmazem::Conteudo(format!("formato {} desconhecido", estado.formato)));
191            }
192            for (id, driver) in &estado.drivers {
193                for manifesto in &driver.versoes {
194                    manifesto.validar().map_err(|erro| ErroArmazem::Conteudo(erro.to_string()))?;
195                    if manifesto.id != *id {
196                        return Err(ErroArmazem::Conteudo(format!("{} guardado como {id}", manifesto.id)));
197                    }
198                }
199                let conhecidas = driver.ativacoes.iter().all(|versao| driver.versao(versao).is_some());
200                if !conhecidas || driver.ativacoes.is_empty() {
201                    return Err(ErroArmazem::Conteudo(format!("ativações de {id} sem versão instalada")));
202                }
203            }
204            registro.ultima_sequencia = estado.ultima_sequencia;
205            registro.drivers = estado.drivers;
206        }
207        registro.trazer_embutidos();
208        Ok(registro)
209    }
210
211    /// Saves everything in `armazem`.
212    pub fn salvar(&self, armazem: &dyn Armazem) -> Result<(), ErroArmazem> {
213        let estado = EstadoGravado {
214            formato: FORMATO,
215            ultima_sequencia: self.ultima_sequencia,
216            drivers: self.drivers.clone(),
217        };
218        let bytes =
219            serde_json::to_vec_pretty(&estado).map_err(|erro| ErroArmazem::Conteudo(erro.to_string()))?;
220        armazem.gravar(&bytes)
221    }

223    /// The program's version it works for.
224    pub fn programa(&self) -> &Version {
225        &self.programa
226    }
227
228    /// The highest catalog sequence seen, so an older catalog is not accepted again.
229    pub fn ultima_sequencia(&self) -> u64 {
230        self.ultima_sequencia
231    }

233    fn exigir_compativel(&self, manifesto: &Manifesto) -> Result<(), ErroRegistro> {
234        match &manifesto.min_versao_programa {
235            Some(exige) if !manifesto.compativel_com(&self.programa) => Err(ErroRegistro::ProgramaAntigo {
236                id: manifesto.id.clone(),
237                versao: manifesto.versao.to_string(),
238                exige: exige.to_string(),
239                programa: self.programa.to_string(),
240            }),
241            _ => Ok(()),
242        }
243    }
244
245    /// Adds a version. The first version of a driver becomes active; later ones wait for
246    /// [`Self::ativar`]. The same version again changes nothing.
247    pub fn instalar(&mut self, manifesto: Manifesto) -> Result<(), ErroRegistro> {
248        manifesto.validar().map_err(ErroRegistro::ManifestoInvalido)?;
249        self.exigir_compativel(&manifesto)?;
250        let driver = self.drivers.entry(manifesto.id.clone()).or_default();
251        if let Some(existente) = driver.versao(&manifesto.versao) {
252            if *existente == manifesto {
253                return Ok(());
254            }
255            return Err(ErroRegistro::VersaoConflitante { id: manifesto.id, versao: manifesto.versao });
256        }
257        driver.registrar(AcaoHistorico::Instalado, &manifesto.versao);
258        if driver.ativacoes.is_empty() {
259            driver.ativacoes.push(manifesto.versao.clone());
260            driver.registrar(AcaoHistorico::Ativado, &manifesto.versao);
261        }
262        driver.versoes.push(manifesto);
263        Ok(())
264    }
265
266    /// Puts an installed version in use.
267    pub fn ativar(&mut self, id: &str, versao: &Version) -> Result<(), ErroRegistro> {
268        let driver = self.drivers.get(id).ok_or_else(|| ErroRegistro::DriverDesconhecido(id.to_owned()))?;
269        let manifesto = driver
270            .versao(versao)
271            .ok_or_else(|| ErroRegistro::VersaoNaoInstalada { id: id.to_owned(), versao: versao.clone() })?;
272        self.exigir_compativel(manifesto)?;
273        let driver =
274            self.drivers.get_mut(id).ok_or_else(|| ErroRegistro::DriverDesconhecido(id.to_owned()))?;
275        if driver.ativacoes.last() != Some(versao) {
276            driver.ativacoes.push(versao.clone());
277            driver.registrar(AcaoHistorico::Ativado, versao);
278        }
279        Ok(())
280    }
281
282    /// Goes back to the version in use before this one, and returns it.
283    pub fn voltar(&mut self, id: &str) -> Result<Version, ErroRegistro> {
284        let driver = self.drivers.get(id).ok_or_else(|| ErroRegistro::DriverDesconhecido(id.to_owned()))?;
285        let [.., anterior, _] = driver.ativacoes.as_slice() else {
286            return Err(ErroRegistro::SemAnterior(id.to_owned()));
287        };
288        let anterior = anterior.clone();
289        let manifesto = driver.versao(&anterior).ok_or_else(|| ErroRegistro::VersaoNaoInstalada {
290            id: id.to_owned(),
291            versao: anterior.clone(),
292        })?;
293        self.exigir_compativel(manifesto)?;
294        let driver =
295            self.drivers.get_mut(id).ok_or_else(|| ErroRegistro::DriverDesconhecido(id.to_owned()))?;
296        driver.ativacoes.pop();
297        driver.registrar(AcaoHistorico::Voltou, &anterior);
298        Ok(anterior)
299    }

301    /// The version in use.
302    pub fn ativo(&self, id: &str) -> Option<&Manifesto> {
303        let driver = self.drivers.get(id)?;
304        driver.versao(driver.ativacoes.last()?)
305    }

322    /// The driver a device runs on: its pinned version, or the active one.
323    pub fn driver_de(&self, dispositivo: &Dispositivo) -> Result<&Manifesto, ErroRegistro> {
324        let id = &dispositivo.driver_id;
325        let manifesto = match &dispositivo.versao_fixada {
326            Some(versao) => {
327                self.versoes(id).iter().find(|manifesto| manifesto.versao == *versao).ok_or_else(|| {
328                    ErroRegistro::VersaoNaoInstalada { id: id.clone(), versao: versao.clone() }
329                })?
330            }
331            None => self.ativo(id).ok_or_else(|| ErroRegistro::DriverDesconhecido(id.clone()))?,
332        };
333        self.exigir_compativel(manifesto)?;
334        Ok(manifesto)
335    }

337    /// Checks a signed catalog against the last one seen, and remembers its sequence.
338    pub fn abrir_catalogo(
339        &mut self,
340        bytes: &[u8],
341        chave: &ChavePublica,
342        hoje: &str,
343    ) -> Result<Catalogo, ErroCatalogo> {
344        let catalogo = Catalogo::abrir(bytes, chave, hoje, self.ultima_sequencia)?;
345        self.ultima_sequencia = catalogo.sequencia();
346        Ok(catalogo)
347    }
348
349    /// Fetches a catalog and does the same as [`Self::abrir_catalogo`].
350    pub fn buscar_catalogo(
351        &mut self,
352        transporte: &dyn TransporteCatalogo,
353        chave: &ChavePublica,
354        hoje: &str,
355    ) -> Result<Catalogo, ErroCatalogo> {
356        let bytes = transporte.baixar().map_err(ErroCatalogo::Transporte)?;
357        self.abrir_catalogo(&bytes, chave, hoje)
358    }

360    /// Installs what a person accepted and puts it in use. [`Self::voltar`] undoes it.
361    pub fn aceitar(&mut self, catalogo: &Catalogo, atualizacao: &Atualizacao) -> Result<(), ErroRegistro> {
362        let (id, versao) = (&atualizacao.id, &atualizacao.para);
363        let item =
364            catalogo.itens().iter().find(|item| item.manifesto.id == *id && item.manifesto.versao == *versao);
365        let item =
366            item.ok_or_else(|| ErroRegistro::ForaDoCatalogo { id: id.clone(), versao: versao.clone() })?;
367        self.instalar(item.manifesto.clone())?;
368        self.ativar(id, versao)
369    }
