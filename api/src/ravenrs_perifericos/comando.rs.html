1//! The few bytes a driver may send, and the check that they are nothing else.
2//!
3//! A manifest cannot carry an arbitrary command. Each slot accepts only whole commands from a short
4//! list, matched to the last byte, so one command cannot hide a second one after it. The list leaves
5//! out everything that writes to the printer's memory or changes its network.
6
7use std::fmt;
8
9use serde::de::Error as _;
10use serde::{Deserialize, Deserializer, Serialize, Serializer};
11
12/// The longest command a manifest can hold, in bytes.
13pub(crate) const TAMANHO_MAXIMO: usize = 16;
14
15const ESC: u8 = 0x1B;
16const GS: u8 = 0x1D;
17
18/// The bytes of one command. In JSON it is hexadecimal text, such as `"1D 56 42 18"`.
19#[derive(Clone, Copy, PartialEq, Eq, Hash)]
20pub struct Comando {
21    bytes: [u8; TAMANHO_MAXIMO],
22    tamanho: u8,
23}
24
25impl Comando {
26    /// A command with these bytes, if they fit. Having one does not make it acceptable: that is
27    /// checked with the manifest it is in.
28    pub fn novo(bytes: &[u8]) -> Option<Self> {
29        if bytes.is_empty() || bytes.len() > TAMANHO_MAXIMO {
30            return None;
31        }
32        let mut guardados = [0; TAMANHO_MAXIMO];
33        guardados[..bytes.len()].copy_from_slice(bytes);
34        Some(Self { bytes: guardados, tamanho: bytes.len() as u8 })
35    }
36
37    /// The bytes to send.
38    pub fn bytes(&self) -> &[u8] {
39        &self.bytes[..usize::from(self.tamanho)]
40    }
41
42    fn de_hex(texto: &str) -> Option<Self> {
43        let digitos: Vec<u8> = texto.bytes().filter(|b| *b != b' ').collect();
44        if !digitos.len().is_multiple_of(2) {
45            return None;
46        }
47        let mut bytes = Vec::with_capacity(digitos.len() / 2);
48        for par in digitos.as_chunks::<2>().0 {
49            bytes.push(u8::from_str_radix(std::str::from_utf8(par).ok()?, 16).ok()?);
50        }
51        Self::novo(&bytes)
52    }
53}
54
55impl fmt::Display for Comando {
56    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
57        for (indice, byte) in self.bytes().iter().enumerate() {
58            if indice > 0 {
59                f.write_str(" ")?;
60            }
61            write!(f, "{byte:02X}")?;
62        }
63        Ok(())
64    }
65}
66
67impl fmt::Debug for Comando {
68    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69        write!(f, "Comando({self})")
70    }
71}
72
73impl Serialize for Comando {
74    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
75        serializer.collect_str(self)
76    }
77}
78
79impl<'de> Deserialize<'de> for Comando {
80    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
81        let texto = String::deserialize(deserializer)?;
82        Self::de_hex(&texto).ok_or_else(|| {
83            D::Error::custom(format!(
84                "\"{texto}\" não é um comando em hexadecimal de até {TAMANHO_MAXIMO} bytes"
85            ))
86        })
87    }
88}

90/// `GS V`: `m` alone, or `m n` for the forms that feed first.
91pub(crate) fn corte_aceito(bytes: &[u8]) -> bool {
92    match bytes {
93        [GS, b'V', 0 | 1 | 48 | 49] | [GS, b'V', 65 | 66, _] => true,
94        // ESC i, ESC m: the cuts of ESC/Bema.
95        [ESC, b'i' | b'm'] => true,
96        _ => false,
97    }
98}
99
100/// `ESC ( A` as Epson and Bematech have it, whose length must be what it says, or `ESC B n t`.
101pub(crate) fn bipe_aceito(bytes: &[u8]) -> bool {
102    match bytes {
103        [ESC, b'(', b'A', baixo, 0, funcao, resto @ ..] => {
104            usize::from(*baixo) == resto.len() + 1 && matches!(funcao, 0x61 | 0x01 | 0x31)
105        }
106        [ESC, b'B', _, _] => true,
107        _ => false,
108    }
109}
110
111/// `ESC p m t1 t2`, or the drawer commands of ESC/Bema: `ESC v n` and `ESC 80h n`.
112pub(crate) fn gaveta_aceita(bytes: &[u8]) -> bool {
113    matches!(bytes, [ESC, b'p', 0 | 1 | 48 | 49, _, _] | [ESC, b'v' | 0x80, _])
114}
115
116/// Only the mode change that is not saved, `GS F9h SP n`. The one that saves (`GS F9h 5 n`) and the
117/// rest of that family, which set the network, stay out.
118pub(crate) fn modo_aceito(bytes: &[u8]) -> bool {
119    matches!(bytes, [GS, 0xF9, 0x20, 0 | 1 | 48 | 49])
120}

122#[cfg(test)]
123mod tests {
124    use super::*;
125
126    fn comando(texto: &str) -> Option<Comando> {
127        Comando::de_hex(texto)
128    }
129
130    #[test]
131    fn hex_goes_both_ways() {
132        let lido = comando("1d 56 42 18");
133        assert_eq!(lido.map(|c| c.to_string()), Some("1D 56 42 18".to_owned()));
134        assert_eq!(comando("1D564218").map(|c| c.bytes().to_vec()), Some(vec![0x1D, 0x56, 0x42, 0x18]));
135        for ruim in ["", "1", "ZZ", "é1", &"00".repeat(17)] {
136            assert_eq!(comando(ruim), None, "{ruim}");
137        }
138    }
139
140    #[test]
141    fn only_whole_cut_commands_pass() {
142        assert!(corte_aceito(&[GS, b'V', 66, 24]));
143        assert!(corte_aceito(&[GS, b'V', 48]));
144        assert!(!corte_aceito(&[GS, b'V', 66, 24, ESC, b'@']), "a second command after the first");
145        assert!(!corte_aceito(&[GS, b'V', 66]));
146        assert!(!corte_aceito(&[ESC, b'@']));
147    }
148
149    #[test]
150    fn a_beep_must_have_the_length_it_declares() {
151        assert!(bipe_aceito(&[ESC, b'(', b'A', 3, 0, 97, 1, 1]));
152        assert!(bipe_aceito(&[ESC, b'(', b'A', 5, 0, 0x61, 0x64, 1, 2, 1]));
153        assert!(!bipe_aceito(&[ESC, b'(', b'A', 3, 0, 97, 1, 1, ESC, b'@']));
154        assert!(!bipe_aceito(&[ESC, b'(', b'A', 3, 0, 0x30, 1, 1]));
155    }
156
157    #[test]
158    fn the_mode_change_that_saves_is_refused() {
159        assert!(modo_aceito(&[GS, 0xF9, 0x20, 1]));
160        assert!(!modo_aceito(&[GS, 0xF9, 0x35, 1]), "saves to the printer's memory");
161        assert!(!modo_aceito(&[GS, 0xF9, b'W', 1]), "the same family sets the network");
162    }
163
164    #[test]
165    fn the_drawer_takes_its_commands_only() {
166        assert!(gaveta_aceita(&[ESC, b'p', 0, 25, 250]));
167        assert!(gaveta_aceita(&[ESC, b'v', 100]));
168        assert!(!gaveta_aceita(&[ESC, b'p', 7, 25, 250]));
169    }
170}
