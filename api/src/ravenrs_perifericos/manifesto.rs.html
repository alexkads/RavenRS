11/// The largest manifest accepted, in bytes.
12pub(crate) const TAMANHO_MAXIMO_MANIFESTO: usize = 16 * 1024;
13const TEXTO_CURTO: usize = 120;
14const TEXTO_LONGO: usize = 1000;
15const MAXIMO_MODELOS: usize = 32;
16const MAXIMO_VARIACOES: usize = 32;
17/// Epson's reference stops at 4607 rows in a `GS v 0`.
18const MAIOR_BLOCO: u32 = 4607;
19/// The most dots across a raster can have.
20const MAIOR_LARGURA: u32 = 4096;
21const DPI_ACEITOS: [u32; 3] = [180, 203, 300];
22const BAUDS_ACEITOS: [u32; 8] = [1200, 2400, 4800, 9600, 19_200, 38_400, 57_600, 115_200];
23/// Names that say nothing about which device it is.
24const NOMES_VAGOS: [&str; 7] = ["generico", "genérico", "qualquer", "todos", "todas", "diversos", "padrao"];

192/// A driver: a version of what is known about one device.

195pub struct Manifesto {
196    /// Its stable name, as `elgin.l42-pro.etiqueta`.
197    pub id: String,
198    /// Its version.
199    pub versao: Version,
200    /// The kind of device.
201    pub tipo: TipoPeriferico,

205    pub modelos: Vec<Modelo>,

216    /// What it can do.
217    #[serde(default)]
218    pub capacidades: Capacidades,
219    /// What differs on this model.

235/// Why a manifest is not accepted.
236#[derive(Clone, Debug, PartialEq, Eq)]
237#[non_exhaustive]
238pub enum ErroManifesto {
239    /// It is larger than allowed.
240    Grande {
241        /// Its size in bytes.
242        tamanho: usize,
243    },
244    /// It is not JSON in the expected shape. Unknown fields fall here.
245    Formato(String),
246    /// A field has a value that is not allowed.
247    Campo {
248        /// Which field.
249        campo: &'static str,
250        /// Why.
251        motivo: String,
252    },
253    /// A command is not one this kind of device may send.
254    ComandoProibido {
255        /// Which field.
256        campo: &'static str,
257        /// The bytes.
258        comando: String,
259    },
260}

262impl fmt::Display for ErroManifesto {
263    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
264        match self {
265            Self::Grande { tamanho } => {
266                write!(f, "manifesto com {tamanho} bytes; o limite é {TAMANHO_MAXIMO_MANIFESTO}")
267            }
268            Self::Formato(motivo) => write!(f, "manifesto mal formado: {motivo}"),
269            Self::Campo { campo, motivo } => write!(f, "campo {campo}: {motivo}"),
270            Self::ComandoProibido { campo, comando } => {
271                write!(f, "campo {campo}: o comando {comando} não é permitido para este tipo de aparelho")
272            }
273        }
274    }
275}

279fn campo(campo: &'static str, motivo: impl Into<String>) -> ErroManifesto {
280    ErroManifesto::Campo { campo, motivo: motivo.into() }
281}

283/// A short text with no way of being read as a path or a control sequence.
284fn texto(nome: &'static str, valor: &str, limite: usize) -> Result<(), ErroManifesto> {
285    if valor.trim().is_empty() {
286        return Err(campo(nome, "está vazio"));
287    }
288    if valor.chars().count() > limite {
289        return Err(campo(nome, format!("passa de {limite} caracteres")));
290    }
291    if valor.chars().any(char::is_control) {
292        return Err(campo(nome, "tem caractere de controle"));
293    }
294    if valor.contains('/') || valor.contains('\\') || valor.contains("..") {
295        return Err(campo(
296            nome,
297            format!("\"{valor}\" parece um caminho, e manifesto não aponta para arquivos"),
298        ));
299    }
300    Ok(())
301}

303/// A sentence for people, where `ESC/POS` is fine and a path or an address is not.
304fn frase(nome: &'static str, valor: &str) -> Result<(), ErroManifesto> {
305    if valor.trim().is_empty() {
306        return Err(campo(nome, "está vazio"));
307    }
308    if valor.chars().count() > TEXTO_LONGO {
309        return Err(campo(nome, format!("passa de {TEXTO_LONGO} caracteres")));
310    }
311    if valor.chars().any(char::is_control) {
312        return Err(campo(nome, "tem caractere de controle"));
313    }
314    let aponta = valor.contains('\\')
315        || valor.contains("..")
316        || valor.contains("://")
317        || valor.split_whitespace().any(|palavra| palavra.starts_with('/'));
318    if aponta {
319        return Err(campo(nome, "tem o que parece um caminho ou endereço, e manifesto não aponta para nada"));
320    }
321    Ok(())
322}

324fn validar_id(id: &str) -> Result<(), ErroManifesto> {
325    let partes: Vec<&str> = id.split('.').collect();
326    let parte_boa = |parte: &&str| {
327        !parte.is_empty()
328            && !parte.starts_with('-')
329            && !parte.ends_with('-')
330            && parte.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
331    };
332    if id.len() > 80 || partes.len() != 3 || !partes.iter().all(parte_boa) {
333        return Err(campo(
334            "id",
335            format!("\"{id}\" não é fabricante.modelo.função, como elgin.l42-pro.etiqueta"),
336        ));
337    }
338    Ok(())
339}

341fn validar_modelo(modelo: &Modelo, fabricante: &str) -> Result<(), ErroManifesto> {
342    texto("modelos.nome", &modelo.nome, TEXTO_CURTO)?;
343    let nome = modelo.nome.trim().to_lowercase();
344    if nome == fabricante.trim().to_lowercase()
345        || NOMES_VAGOS.contains(&nome.as_str())
346        || nome.chars().count() < 3
347    {
348        return Err(campo(
349            "modelos.nome",
350            format!("\"{}\" não diz qual é o aparelho; use o nome exato do modelo", modelo.nome),
351        ));
352    }
353    if modelo.variacoes.len() > MAXIMO_VARIACOES {
354        return Err(campo("modelos.variacoes", format!("mais de {MAXIMO_VARIACOES}")));
355    }
356    for variacao in &modelo.variacoes {
357        texto("modelos.variacoes", variacao, TEXTO_CURTO)?;
358    }
359    if let Some(firmware) = &modelo.firmware {
360        texto("modelos.firmware", firmware, TEXTO_CURTO)?;
361    }
362    Ok(())
363}

365fn protocolo_serve(tipo: TipoPeriferico, nome: NomeProtocolo) -> bool {
366    use NomeProtocolo as P;
367    match tipo {
368        TipoPeriferico::ImpressoraTermica => nome == P::EscPos,
369        TipoPeriferico::Etiquetadora => matches!(nome, P::Zpl | P::Epl | P::Ppla | P::Tspl),
370        TipoPeriferico::Pinpad => nome == P::Abecs,
371        TipoPeriferico::LeitorCodigo => matches!(nome, P::Hid | P::Serial),
372        TipoPeriferico::Balanca => nome == P::Serial,
373        TipoPeriferico::Gaveta => matches!(nome, P::EscPos | P::Serial),
374    }
375}
376
377fn validar_protocolo(tipo: TipoPeriferico, protocolo: &Protocolo) -> Result<(), ErroManifesto> {
378    if !protocolo_serve(tipo, protocolo.nome) {
379        return Err(campo("protocolo", format!("{:?} não serve para {tipo:?}", protocolo.nome)));
380    }
381    if let Some(versao) = &protocolo.versao {
382        texto("protocolo.versao", versao, TEXTO_CURTO)?;
383    } else if protocolo.nome == NomeProtocolo::Abecs {
384        return Err(campo("protocolo.versao", "ABECS precisa dizer a versão"));
385    }
386    Ok(())
387}

389fn validar_capacidades(tipo: TipoPeriferico, capacidades: &Capacidades) -> Result<(), ErroManifesto> {
390    if capacidades.pix_no_pinpad && tipo != TipoPeriferico::Pinpad {
391        return Err(campo("capacidades.pix_no_pinpad", "só pinpad mostra Pix"));
392    }
393    let imprime = matches!(tipo, TipoPeriferico::ImpressoraTermica | TipoPeriferico::Etiquetadora);
394    if !imprime && (capacidades.corta || capacidades.corte_parcial || capacidades.qr_nativo) {
395        return Err(campo("capacidades", "corte e QR são de impressora"));
396    }
397    if capacidades.gaveta && !matches!(tipo, TipoPeriferico::ImpressoraTermica | TipoPeriferico::Gaveta) {
398        return Err(campo("capacidades.gaveta", "só impressora de cupom e gaveta abrem gaveta"));
399    }
400    Ok(())
401}

403fn validar_comando(
404    nome: &'static str,
405    comando: Option<Comando>,
406    aceito: impl Fn(&[u8]) -> bool,
407) -> Result<(), ErroManifesto> {
408    match comando {
409        Some(comando) if !aceito(comando.bytes()) => {
410            Err(ErroManifesto::ComandoProibido { campo: nome, comando: comando.to_string() })
411        }
412        _ => Ok(()),
413    }
414}

416fn validar_ajustes(tipo: TipoPeriferico, ajustes: &Ajustes) -> Result<(), ErroManifesto> {
417    let recibo = tipo == TipoPeriferico::ImpressoraTermica;
418    let gaveta = recibo || tipo == TipoPeriferico::Gaveta;
419    validar_comando("ajustes.corte_total", ajustes.corte_total, |b| recibo && corte_aceito(b))?;
420    validar_comando("ajustes.corte_parcial", ajustes.corte_parcial, |b| recibo && corte_aceito(b))?;
421    validar_comando("ajustes.bipe", ajustes.bipe, |b| recibo && bipe_aceito(b))?;
422    validar_comando("ajustes.gaveta", ajustes.gaveta, |b| gaveta && gaveta_aceita(b))?;
423    validar_comando("ajustes.comando_modo", ajustes.comando_modo, |b| recibo && modo_aceito(b))?;
424    let imprime = recibo || tipo == TipoPeriferico::Etiquetadora;
425    if !imprime
426        && (ajustes.tamanho_bloco.is_some() || ajustes.polaridade_invertida || ajustes.codepage.is_some())
427    {
428        return Err(campo("ajustes", "bloco, polaridade e codepage são de impressora"));
429    }
430    if let Some(bloco) = ajustes.tamanho_bloco
431        && !(1..=MAIOR_BLOCO).contains(&bloco)
432    {
433        return Err(campo("ajustes.tamanho_bloco", format!("{bloco}; vai de 1 a {MAIOR_BLOCO}")));
434    }
435    Ok(())
436}

438fn validar_padroes(padroes: &Padroes) -> Result<(), ErroManifesto> {
439    if let Some(dpi) = padroes.dpi
440        && !DPI_ACEITOS.contains(&dpi)
441    {
442        return Err(campo("padroes.dpi", format!("{dpi}; aceita {DPI_ACEITOS:?}")));
443    }
444    if let Some(largura) = padroes.largura_pontos
445        && !(8..=MAIOR_LARGURA).contains(&largura)
446    {
447        return Err(campo("padroes.largura_pontos", format!("{largura}; vai de 8 a {MAIOR_LARGURA}")));
448    }
449    if let Some(baud) = padroes.baud
450        && !BAUDS_ACEITOS.contains(&baud)
451    {
452        return Err(campo("padroes.baud", format!("{baud}; aceita {BAUDS_ACEITOS:?}")));
453    }
454    Ok(())
455}

457fn validar_situacao(situacao: &Situacao) -> Result<(), ErroManifesto> {
458    match situacao {
459        Situacao::TestadoNoAparelho { data, por, firmware } => {
460            if !data_valida(data) {
461                return Err(campo("situacao.data", format!("\"{data}\" não é uma data AAAA-MM-DD")));
462            }
463            texto("situacao.por", por, TEXTO_CURTO)?;
464            texto("situacao.firmware", firmware, TEXTO_CURTO)
465        }
466        // A source can be an address, the only text that may have slashes. Nothing ever opens it.
467        Situacao::SoPelaDocumentacao { fonte } => {
468            let limpa = !fonte.trim().is_empty()
469                && fonte.chars().count() <= TEXTO_LONGO
470                && !fonte.chars().any(char::is_control)
471                && !fonte.contains('\\')
472                && !fonte.contains("..")
473                && !fonte.starts_with('/');
474            if limpa {
475                Ok(())
476            } else {
477                Err(campo("situacao.fonte", "precisa dizer de onde veio, sem caminho local"))
478            }
479        }
480        Situacao::Presumido => Ok(()),
481    }
482}

484impl Manifesto {
485    /// Reads and checks a manifest.
486    pub fn de_json(bytes: &[u8]) -> Result<Self, ErroManifesto> {
487        if bytes.len() > TAMANHO_MAXIMO_MANIFESTO {
488            return Err(ErroManifesto::Grande { tamanho: bytes.len() });
489        }
490        let manifesto: Self =
491            serde_json::from_slice(bytes).map_err(|erro| ErroManifesto::Formato(erro.to_string()))?;
492        manifesto.validar()?;
493        Ok(manifesto)
494    }
495
496    /// The manifest as JSON.
497    pub fn para_json(&self) -> Result<String, ErroManifesto> {
498        serde_json::to_string_pretty(self).map_err(|erro| ErroManifesto::Formato(erro.to_string()))
499    }
500
501    /// Checks every limit. A manifest from anywhere goes through this before it is used.
502    pub fn validar(&self) -> Result<(), ErroManifesto> {
503        validar_id(&self.id)?;
504        texto("fabricante", &self.fabricante, TEXTO_CURTO)?;
505        if self.modelos.is_empty() {
506            return Err(campo("modelos", "é obrigatório dizer o modelo exato do aparelho"));
507        }
508        if self.modelos.len() > MAXIMO_MODELOS {
509            return Err(campo("modelos", format!("mais de {MAXIMO_MODELOS}")));
510        }
511        for modelo in &self.modelos {
512            validar_modelo(modelo, &self.fabricante)?;
513        }
514        if self.conexoes.is_empty() {
515            return Err(campo("conexoes", "é obrigatório dizer como o aparelho se conecta"));
516        }
517        validar_padroes(&self.padroes)?;
518        validar_protocolo(self.tipo, &self.protocolo)?;
519        if self.protocolos_alternativos.len() > 8 {
520            return Err(campo("protocolos_alternativos", "mais de 8"));
521        }
522        for protocolo in &self.protocolos_alternativos {
523            validar_protocolo(self.tipo, protocolo)?;
524        }
525        validar_capacidades(self.tipo, &self.capacidades)?;
526        validar_ajustes(self.tipo, &self.ajustes)?;
527        validar_situacao(&self.situacao)?;
528        if let Some(instalacao) = &self.instalacao {
529            frase("instalacao", instalacao)?;
530        }
531        if let Some(observacoes) = &self.observacoes {
532            frase("observacoes", observacoes)?;
533        }
534        Ok(())
535    }
536
537    /// Whether it works with this version of the program.
538    pub fn compativel_com(&self, versao_do_programa: &Version) -> bool {
539        self.min_versao_programa.as_ref().is_none_or(|minima| versao_do_programa >= minima)
540    }
541
542    /// The model's name to show first.
543    pub fn nome_do_modelo(&self) -> &str {
544        self.modelos.first().map_or(self.id.as_str(), |modelo| modelo.nome.as_str())
545    }
