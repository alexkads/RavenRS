48fn matrix_of(transform: &Transform, frame: f32) -> Matrix {
49    let position_at = |frame: f32| match &transform.split {
50        Some((x, y)) => (x.first_at(frame, 0.0), y.first_at(frame, 0.0)),
51        None => transform.position.pair_at(frame, (0.0, 0.0)),
52    };
53    let position = position_at(frame);
54    let mut rotation = transform.rotation.first_at(frame, 0.0);
55    if transform.orient {
56        let (before, after) = (position_at(frame - 0.5), position_at(frame + 0.5));
57        let (dx, dy) = (after.0 - before.0, after.1 - before.1);
58        if dx != 0.0 || dy != 0.0 {
59            rotation += dy.atan2(dx).to_degrees();
60        }
61    }
62    Matrix::of_transform(
63        position,
64        transform.anchor.pair_at(frame, (0.0, 0.0)),
65        transform.scale.pair_at(frame, (100.0, 100.0)),
66        rotation,
67        transform.skew.first_at(frame, 0.0),
68        transform.skew_axis.first_at(frame, 0.0),
69    )
70}

72fn color_of(value: &[f32], opacity: f32) -> Color {
73    // Very old files keep colors from 0 to 255.
74    let k = if value.iter().take(3).any(|v| *v > 1.0) { 255.0 } else { 1.0 };
75    let at = |i: usize| value.get(i).copied().unwrap_or(0.0) / k;
76    Color::rgba(at(0), at(1), at(2), opacity.clamp(0.0, 1.0))
77}

87fn stops_of(paint: &GradientPaint, frame: f32) -> Arc<[ColorStop]> {
88    let raw = paint.stops.at(frame);
89    let count = if paint.count > 0 { paint.count } else { raw.len() / 4 };
90    let colors = raw.get(..count * 4).unwrap_or(&raw);
91    let alphas = raw.get(count * 4..).unwrap_or(&[]);
92    // Alpha has stops of its own, at offsets of their own.
93    let alpha_at = |offset: f32| -> f32 {
94        let pairs: Vec<(f32, f32)> = alphas.as_chunks::<2>().0.iter().map(|p| (p[0], p[1])).collect();
95        let Some(first) = pairs.first() else { return 1.0 };
96        if offset <= first.0 {
97            return first.1;
98        }
99        for pair in pairs.windows(2) {
100            if offset <= pair[1].0 {
101                let k = (offset - pair[0].0) / (pair[1].0 - pair[0].0).max(1e-6);
102                return pair[0].1 + (pair[1].1 - pair[0].1) * k;
103            }
104        }
105        pairs.last().map_or(1.0, |p| p.1)
106    };
107    colors
108        .as_chunks::<4>()
109        .0
110        .iter()
111        .map(|c| ColorStop { offset: c[0], color: Color::rgba(c[1], c[2], c[3], alpha_at(c[0])) })
112        .collect()
113}

115fn gradient_of(paint: &GradientPaint, cx: &Context<'_>) -> PathPaint {
116    let start = cx.matrix.apply(paint.start.pair_at(cx.frame, (0.0, 0.0)));
117    let end = cx.matrix.apply(paint.end.pair_at(cx.frame, (0.0, 0.0)));
118    let stops = stops_of(paint, cx.frame);
119    if paint.radial {
120        let radius = ((end.x - start.x).powi(2) + (end.y - start.y).powi(2)).sqrt();
121        PathPaint::Radial { center: start, radius, stops }
122    } else {
123        PathPaint::Linear { start, end, stops }
124    }
125}

148fn stroked(contours: &[Contour], look: &StrokeLook, cx: &Context<'_>) -> Vec<PathElement> {
149    let scale = cx.matrix.mean_scale();
150    let width = look.width.first_at(cx.frame, 0.0) * scale;
151    let style = StrokeStyle { width, cap: look.cap, join: look.join, miter_limit: look.miter_limit };
152    let (mut pattern, mut offset) = (Vec::new(), 0.0);
153    for (name, value) in &look.dashes {
154        let v = value.first_at(cx.frame, 0.0) * scale;
155        if name == "o" {
156            offset = v;
157        } else {
158            pattern.push(v);
159        }
160    }
161    if pattern.len() % 2 == 1 {
162        pattern.extend_from_within(..);
163    }
164    if pattern.iter().sum::<f32>() > 0.0 {
165        stroke(&dash(contours, &pattern, offset, cx.tolerance), style, cx.tolerance)
166    } else {
167        stroke(contours, style, cx.tolerance)
168    }
169}

171fn rounded_rect(center: (f32, f32), size: (f32, f32), radius: f32, m: Matrix) -> Contour {
172    let (hw, hh) = (size.0 / 2.0, size.1 / 2.0);
173    let r = radius.min(hw).min(hh).max(0.0);
174    let (x0, y0, x1, y1) = (center.0 - hw, center.1 - hh, center.0 + hw, center.1 + hh);
175    if r <= 0.0 {
176        return Contour::polygon(
177            &[m.apply((x1, y0)), m.apply((x1, y1)), m.apply((x0, y1)), m.apply((x0, y0))],
178            true,
179        );
180    }
181    let k = r * (1.0 - KAPPA);
182    let line = |a: (f32, f32), b: (f32, f32)| {
183        let at = |t: f32| m.apply((a.0 + (b.0 - a.0) * t, a.1 + (b.1 - a.1) * t));
184        [at(1.0 / 3.0), at(2.0 / 3.0), m.apply(b)]
185    };
186    let curves = vec![
187        line((x1, y0 + r), (x1, y1 - r)),
188        [m.apply((x1, y1 - k)), m.apply((x1 - k, y1)), m.apply((x1 - r, y1))],
189        line((x1 - r, y1), (x0 + r, y1)),
190        [m.apply((x0 + k, y1)), m.apply((x0, y1 - k)), m.apply((x0, y1 - r))],
191        line((x0, y1 - r), (x0, y0 + r)),
192        [m.apply((x0, y0 + k)), m.apply((x0 + k, y0)), m.apply((x0 + r, y0))],
193        line((x0 + r, y0), (x1 - r, y0)),
194        [m.apply((x1 - k, y0)), m.apply((x1, y0 + k)), m.apply((x1, y0 + r))],
195    ];
196    Contour { start: m.apply((x1, y0 + r)), curves, closed: true }
197}

199fn ellipse(center: (f32, f32), size: (f32, f32), m: Matrix) -> Contour {
200    let (rx, ry) = (size.0 / 2.0, size.1 / 2.0);
201    let (kx, ky) = (rx * KAPPA, ry * KAPPA);
202    let (x, y) = center;
203    let curves = vec![
204        [m.apply((x + kx, y - ry)), m.apply((x + rx, y - ky)), m.apply((x + rx, y))],
205        [m.apply((x + rx, y + ky)), m.apply((x + kx, y + ry)), m.apply((x, y + ry))],
206        [m.apply((x - kx, y + ry)), m.apply((x - rx, y + ky)), m.apply((x - rx, y))],
207        [m.apply((x - rx, y - ky)), m.apply((x - kx, y - ry)), m.apply((x, y - ry))],
208    ];
209    Contour { start: m.apply((x, y - ry)), curves, closed: true }
210}

212/// A star or polygon, with `rounds` the part of a stretch of its circle each kind of
213/// corner bends along: the outer ones first, then the inner ones.
214fn star(
215    center: (f32, f32),
216    count: usize,
217    turn: f32,
218    radii: (f32, f32),
219    rounds: (f32, f32),
220    polygon: bool,
221    m: Matrix,
222) -> Contour {
223    let corners = if polygon { count } else { count * 2 };
224    let stretches = if polygon { corners * 4 } else { corners * 2 } as f32;
225    // Each corner, with the points its curves leave from and arrive by.
226    let shape: Vec<[Point; 3]> = (0..corners)
227        .map(|i| {
228            let outer = polygon || i % 2 == 0;
229            let (radius, round) = if outer { (radii.0, rounds.0) } else { (radii.1, rounds.1) };
230            let angle =
231                turn - std::f32::consts::FRAC_PI_2 + i as f32 * std::f32::consts::TAU / corners as f32;
232            let (sin, cos) = angle.sin_cos();
233            let (x, y) = (center.0 + cos * radius, center.1 + sin * radius);
234            let reach = std::f32::consts::TAU * radius / stretches * round;
235            [
236                m.apply((x + sin * reach, y - cos * reach)),
237                m.apply((x, y)),
238                m.apply((x - sin * reach, y + cos * reach)),
239            ]
240        })
241        .collect();
242    let curves = (0..corners)
243        .map(|i| {
244            let next = &shape[(i + 1) % corners];
245            [shape[i][2], next[0], next[1]]
246        })
247        .collect();
248    Contour { start: shape[0][1], curves, closed: true }
249}

251/// Rounds the corners between straight sides. Curved sides stay as they are.
252fn rounded_corners(contour: &Contour, radius: f32) -> Contour {
253    let mut vertices = vec![contour.start];
254    vertices.extend(contour.curves.iter().map(|c| c[2]));
255    if contour.closed && vertices.len() > 2 && vertices.first() == vertices.last() {
256        vertices.pop();
257    }
258    let straight = contour.curves.iter().scan(contour.start, |from, c| {
259        let to = c[2];
260        let expected = |t: f32| Point::new(from.x + (to.x - from.x) * t, from.y + (to.y - from.y) * t);
261        let near = |p: Point, q: Point| (p.x - q.x).abs() + (p.y - q.y).abs() < 0.5;
262        let is_line = (near(c[0], expected(1.0 / 3.0)) && near(c[1], expected(2.0 / 3.0)))
263            || (near(c[0], *from) && near(c[1], to));
264        *from = to;
265        Some(is_line)
266    });
267    if radius <= 0.0 || vertices.len() < 3 || !straight.into_iter().all(|s| s) {
268        return contour.clone();
269    }
270    let count = vertices.len();
271    let toward = |from: Point, to: Point, by: f32| {
272        let length = ((to.x - from.x).powi(2) + (to.y - from.y).powi(2)).sqrt().max(1e-6);
273        let k = (by / length).min(0.5);
274        Point::new(from.x + (to.x - from.x) * k, from.y + (to.y - from.y) * k)
275    };
276    let corner_range: Vec<usize> =
277        if contour.closed { (0..count).collect() } else { (1..count - 1).collect() };
278    let mut out = Contour { closed: contour.closed, ..Contour::default() };
279    let mut pen: Option<Point> = None;
280    if !contour.closed {
281        out.start = vertices[0];
282        pen = Some(vertices[0]);
283    }
284    let line_to = |from: Point, to: Point| {
285        let at = |t: f32| Point::new(from.x + (to.x - from.x) * t, from.y + (to.y - from.y) * t);
286        [at(1.0 / 3.0), at(2.0 / 3.0), to]
287    };
288    for i in corner_range {
289        let at = vertices[i];
290        let (before, after) = (vertices[(i + count - 1) % count], vertices[(i + 1) % count]);
291        let (a, b) = (toward(at, before, radius), toward(at, after, radius));
292        match pen {
293            Some(p) => out.curves.push(line_to(p, a)),
294            None => out.start = a,
295        }
296        let mix = |p: Point, q: Point, k: f32| Point::new(p.x + (q.x - p.x) * k, p.y + (q.y - p.y) * k);
297        out.curves.push([mix(a, at, KAPPA), mix(b, at, KAPPA), b]);
298        pen = Some(b);
299    }
300    if let Some(p) = pen {
301        let end = if contour.closed { out.start } else { vertices[count - 1] };
302        out.curves.push(line_to(p, end));
303    }
304    out
305}

307fn trim_of(item: &Item, frame: f32) -> Option<TrimNow> {
308    let Item::Trim { start, end, offset, together } = item else { return None };
309    Some(TrimNow {
310        start: start.first_at(frame, 0.0) / 100.0,
311        end: end.first_at(frame, 100.0) / 100.0,
312        offset: offset.first_at(frame, 0.0) / 360.0,
313        together: *together,
314    })
315}

317/// What a style sees of the outlines above it: trimmed by what reaches in from outside.
318fn styled(outlines: &[Contour], cx: &Context<'_>) -> Vec<Contour> {
319    let mut seen = outlines.to_vec();
320    for t in &cx.trims {
321        seen = trim(&seen, t.start, t.end, t.offset, t.together, cx.tolerance);
322    }
323    seen

326/// Draws these items, top first, and hands back the outlines for the styles of the
327/// groups around them.
328fn items(list: &[Item], cx: &Context<'_>, out: &mut Vec<PathFill>) -> Vec<Contour> {
329    let mut outlines: Vec<Contour> = Vec::new();
330    let frame = cx.frame;
331    // What is above a repeater is drawn by it, once per copy, and not on its own.
332    let skip = list.iter().rposition(|item| matches!(item, Item::Repeater { .. })).unwrap_or(0);
333    for (at, item) in list.iter().enumerate().skip(skip) {
334        match item {
335            Item::Group(inner, transform) => {
336                let (matrix, opacity) = match transform {
337                    Some(t) => (
338                        cx.matrix.then_inner(matrix_of(t, frame)),
339                        cx.opacity * t.opacity.first_at(frame, 100.0) / 100.0,
340                    ),
341                    None => (cx.matrix, cx.opacity),
342                };
343                let mut inner_cx = cx.with(matrix, opacity);
344                inner_cx.trims.extend(list[at + 1..].iter().filter_map(|later| trim_of(later, frame)));
345                outlines.extend(items(inner, &inner_cx, out));
346            }
347            Item::Path(shape) => outlines.extend(Contour::of_bezier(&shape.at(frame), cx.matrix)),
348            Item::Rect { position, size, radius } => outlines.push(rounded_rect(
349                position.pair_at(frame, (0.0, 0.0)),
350                size.pair_at(frame, (0.0, 0.0)),
351                radius.first_at(frame, 0.0),
352                cx.matrix,
353            )),
354            Item::Ellipse { position, size } => outlines.push(ellipse(
355                position.pair_at(frame, (0.0, 0.0)),
356                size.pair_at(frame, (0.0, 0.0)),
357                cx.matrix,
358            )),
359            Item::Star { polygon, position, rotation, points, inner, outer, inner_round, outer_round } => {
360                let center = position.pair_at(frame, (0.0, 0.0));
361                let count = points.first_at(frame, 5.0).round().max(3.0) as usize;
362                let turn = rotation.first_at(frame, 0.0).to_radians();
363                let radii = (outer.first_at(frame, 0.0), inner.first_at(frame, 0.0));
364                let rounds =
365                    (outer_round.first_at(frame, 0.0) / 100.0, inner_round.first_at(frame, 0.0) / 100.0);
366                outlines.push(star(center, count, turn, radii, rounds, *polygon, cx.matrix));
367            }
368            Item::Fill { color, opacity, even_odd } => {
369                let alpha = opacity.first_at(frame, 100.0) / 100.0 * cx.opacity;
370                let rule = if *even_odd { FillRule::EvenOdd } else { FillRule::NonZero };
371                emit(
372                    out,
373                    elements_of(&styled(&outlines, cx)),
374                    rule,
375                    PathPaint::Solid(color_of(&color.at(frame), 1.0)),
376                    alpha,
377                    cx,
378                );
379            }
380            Item::GradientFill { paint, opacity, even_odd } => {
381                let alpha = opacity.first_at(frame, 100.0) / 100.0 * cx.opacity;
382                let rule = if *even_odd { FillRule::EvenOdd } else { FillRule::NonZero };
383                emit(out, elements_of(&styled(&outlines, cx)), rule, gradient_of(paint, cx), alpha, cx);
384            }
385            Item::Stroke { color, opacity, look } => {
386                let alpha = opacity.first_at(frame, 100.0) / 100.0 * cx.opacity;
387                let elements = stroked(&styled(&outlines, cx), look, cx);
388                emit(
389                    out,
390                    elements,
391                    FillRule::NonZero,
392                    PathPaint::Solid(color_of(&color.at(frame), 1.0)),
393                    alpha,
394                    cx,
395                );
396            }
397            Item::GradientStroke { paint, opacity, look } => {
398                let alpha = opacity.first_at(frame, 100.0) / 100.0 * cx.opacity;
399                let elements = stroked(&styled(&outlines, cx), look, cx);
400                emit(out, elements, FillRule::NonZero, gradient_of(paint, cx), alpha, cx);
401            }
402            Item::Trim { .. } => {
403                if let Some(t) = trim_of(item, frame) {
404                    outlines = trim(&outlines, t.start, t.end, t.offset, t.together, cx.tolerance);
405                }
406            }
407            Item::RoundedCorners(radius) => {
408                let r = radius.first_at(frame, 0.0) * cx.matrix.mean_scale();
409                outlines = outlines.iter().map(|c| rounded_corners(c, r)).collect();
410            }
411            Item::Repeater { copies, offset, transform, start_opacity, end_opacity } => {
412                let count = copies.first_at(frame, 1.0).max(0.0).ceil() as usize;
413                // Here the anchor is only the center of the turn: position moves from it.
414                let (px, py) = transform.position.pair_at(frame, (0.0, 0.0));
415                let anchor = transform.anchor.pair_at(frame, (0.0, 0.0));
416                let step = Matrix::of_transform(
417                    (px + anchor.0, py + anchor.1),
418                    anchor,
419                    transform.scale.pair_at(frame, (100.0, 100.0)),
420                    transform.rotation.first_at(frame, 0.0),
421                    0.0,
422                    0.0,
423                );
424                let shift = offset.first_at(frame, 0.0);
425                let (first, last) = (
426                    start_opacity.first_at(frame, 100.0) / 100.0,
427                    end_opacity.first_at(frame, 100.0) / 100.0,
428                );
429                outlines.clear();
430                // What is above it in the list is drawn again, once per copy.
431                let earlier = &list[..at];
432                for i in (0..count).rev() {
433                    let mut m = cx.matrix;
434                    for _ in 0..(i as f32 + shift).max(0.0).round() as usize {
435                        m = m.then_inner(step);
436                    }
437                    let k = if count > 1 { i as f32 / (count - 1) as f32 } else { 0.0 };
438                    let copy = cx.with(m, cx.opacity * (first + (last - first) * k));
439                    outlines.extend(items(earlier, &copy, out));
440                }
441            }
442        }
443    }
444    outlines
445}

447fn layer_matrix(layer: &Layer, by_index: &HashMap<i64, &Layer>, frame: f32) -> Matrix {
448    let mut matrix = matrix_of(&layer.transform, frame);
449    let mut parent = layer.parent;
450    let mut guard = 0;
451    while let Some(index) = parent {
452        let Some(found) = by_index.get(&index) else { break };
453        matrix = matrix_of(&found.transform, frame).then_inner(matrix);
454        parent = found.parent;
455        guard += 1;
456        if guard > 64 {
457            break;
458        }
459    }
460    matrix
461}

471fn masks_clip(layer: &Layer, frame: f32, matrix: Matrix) -> Option<PathClip> {
472    if layer.masks.is_empty() {
473        return None;
474    }
475    let contours_of = |subtract: bool| -> Vec<PathElement> {
476        let contours: Vec<Contour> = layer
477            .masks
478            .iter()
479            .filter(|mask| mask.subtract == subtract)
480            .filter_map(|mask| Contour::of_bezier(&mask.path.at(frame), matrix))
481            .collect();
482        elements_of(&contours)
483    };
484    let added = contours_of(false);
485    if !added.is_empty() {
486        let inverted = layer.masks.iter().find(|m| !m.subtract).is_some_and(|m| m.inverted);
487        return Some(PathClip { elements: added.into(), rule: FillRule::NonZero, inverted });
488    }
489    Some(PathClip { elements: contours_of(true).into(), rule: FillRule::NonZero, inverted: true })
490}

492fn layer_paths(
493    layer: &Layer,
494    layers: &[Layer],
495    by_index: &HashMap<i64, &Layer>,
496    cx: &Context<'_>,
497    out: &mut Vec<PathFill>,
498) {
499    let frame = cx.frame;
500    if layer.hidden || frame < layer.in_frame || frame >= layer.out_frame {
501        return;
502    }
503    let matrix = cx.matrix.then_inner(layer_matrix(layer, by_index, frame));
504    // The times in a file already count the layer's start. Only what is inside a
505    // nested composition runs on a clock of its own.
506    let opacity = cx.opacity * layer.transform.opacity.first_at(frame, 100.0) / 100.0;
507    if opacity <= 0.0 {
508        return;
509    }
510    let mut own = cx.with(matrix, opacity);
511    if let Some(clip) = masks_clip(layer, frame, matrix) {
512        own.clip = Some(Arc::new(clip));
513    }
514    if layer.matte_mode > 0 {
515        let inverted = matches!(layer.matte_mode, 2 | 4);
516        let source = match layer.matte_index {
517            Some(index) => layers.iter().find(|l| l.index == Some(index)),
518            None => layers
519                .iter()
520                .position(|l| std::ptr::eq(l, layer))
521                .and_then(|at| at.checked_sub(1))
522                .and_then(|at| layers.get(at)),
523        };
524        if let Some(source) = source {
525            let mut matte = Vec::new();
526            let plain = Context { clip: None, ..cx.with(cx.matrix, 1.0) };
527            let visible = Layer { is_matte: false, ..source.clone() };
528            layer_paths(&visible, layers, by_index, &plain, &mut matte);
529            own.clip = Some(Arc::new(union_clip(&matte, inverted)));
530        }
531    }
532    let first = out.len();
533    match &layer.kind {
534        LayerKind::Empty => {}
535        LayerKind::Shape(list) => {
536            items(list, &own, out);
537        }
538        LayerKind::Solid { width, height, color } => {
539            let contour = rounded_rect((width / 2.0, height / 2.0), (*width, *height), 0.0, matrix);
540            emit(
541                out,
542                elements_of(&[contour]),
543                FillRule::NonZero,
544                PathPaint::Solid(Color::rgba(color[0], color[1], color[2], 1.0)),
545                opacity,
546                &own,
547            );
548        }
549        LayerKind::Image { image, width, height } => {
550            let contour = rounded_rect((width / 2.0, height / 2.0), (*width, *height), 0.0, matrix);
551            let origin = matrix.apply((0.0, 0.0));
552            let (across, down) = (matrix.apply((*width, 0.0)), matrix.apply((0.0, *height)));
553            let placement = [
554                across.x - origin.x,
555                across.y - origin.y,
556                down.x - origin.x,
557                down.y - origin.y,
558                origin.x,
559                origin.y,
560            ];
561            let paint = PathPaint::Image { image: image.clone(), placement };
562            emit(out, elements_of(&[contour]), FillRule::NonZero, paint, opacity, &own);
563        }
564        LayerKind::Precomp { reference, width, height, remap } => {
565            let Some(inner) = cx.document.precomps.get(reference) else { return };
566            let inner_frame = match remap {
567                Some(remap) => remap.first_at(frame, 0.0) * cx.document.frame_rate,
568                None => (frame - layer.start) / layer.stretch,
569            };
570            // A composition shows nothing past its edges, unless something already clips it.
571            if own.clip.is_none() && *width > 0.0 && *height > 0.0 {
572                let contour = rounded_rect((width / 2.0, height / 2.0), (*width, *height), 0.0, matrix);
573                own.clip = Some(Arc::new(PathClip {
574                    elements: elements_of(&[contour]).into(),
575                    rule: FillRule::NonZero,
576                    inverted: false,
577                }));
578            }
579            let inner_cx = Context { frame: inner_frame, ..own.with(matrix, opacity) };
580            let inner_cx = Context { clip: own.clip.clone(), ..inner_cx };
581            draw_layers(inner, &inner_cx, out);
582        }
583    }
584    if let Some((color, effect_opacity)) = &layer.fill_effect {
585        let paint = PathPaint::Solid(color_of(&color.at(frame), 1.0));
586        let kept = effect_opacity.first_at(frame, 1.0).clamp(0.0, 1.0);
587        for path in &mut out[first..] {
588            path.paint = paint.clone();
589            path.opacity *= kept;
590        }
591    }
592}
593
594fn draw_layers(layers: &[Layer], cx: &Context<'_>, out: &mut Vec<PathFill>) {
595    let by_index: HashMap<i64, &Layer> =
596        layers.iter().filter_map(|layer| layer.index.map(|index| (index, layer))).collect();
597    for layer in layers {
598        if layer.is_matte {
599            continue;
600        }
601        layer_paths(layer, layers, &by_index, cx, out);
602    }
603}
604
605pub(crate) fn paths_at(document: &Document, seconds: f64, scale: f32) -> Vec<PathFill> {
606    let frame = document.in_frame + (seconds as f32 * document.frame_rate);
607    // A composition's last frame is the one before where it ends, when layers end too.
608    let last = (document.out_frame - 1.0).max(document.in_frame);
609    let frame = if frame >= document.out_frame { last } else { frame.max(document.in_frame) };
610    let cx = Context {
611        document,
612        frame,
613        matrix: Matrix::IDENTITY,
614        opacity: 1.0,
615        tolerance: 0.1 / scale.max(0.01),
616        clip: None,
617        trims: Vec::new(),
618    };
619    let mut out = Vec::new();
620    draw_layers(&document.layers, &cx, &mut out);
621    // The list is written top first, and painting goes bottom first.
622    out.reverse();
623    out
624}
