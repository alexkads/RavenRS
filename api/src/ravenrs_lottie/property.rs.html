5/// How finely a curved motion path is measured.
6const ARC_SAMPLES: usize = 64;

8/// A cubic timing curve from (0, 0) to (1, 1), as CSS has them.
9fn timing(p1: (f32, f32), p2: (f32, f32), x: f32) -> f32 {
10    if x <= 0.0 {
11        return 0.0;
12    }
13    if x >= 1.0 {
14        return 1.0;
15    }
16    let bezier = |a: f32, b: f32, t: f32| {
17        let u = 1.0 - t;
18        3.0 * u * u * t * a + 3.0 * u * t * t * b + t * t * t
19    };
20    let slope = |a: f32, b: f32, t: f32| {
21        let u = 1.0 - t;
22        3.0 * u * u * a + 6.0 * u * t * (b - a) + 3.0 * t * t * (1.0 - b)
23    };
24    let mut t = x;
25    for _ in 0..8 {
26        let error = bezier(p1.0, p2.0, t) - x;
27        if error.abs() < 1e-5 {
28            break;
29        }
30        let d = slope(p1.0, p2.0, t);
31        if d.abs() < 1e-6 {
32            break;
33        }
34        t = (t - error / d).clamp(0.0, 1.0);
35    }
36    // Newton can stall on steep curves: halving always lands.
37    if (bezier(p1.0, p2.0, t) - x).abs() > 1e-3 {
38        let (mut low, mut high) = (0.0, 1.0);
39        for _ in 0..24 {
40            t = (low + high) / 2.0;
41            if bezier(p1.0, p2.0, t) < x {
42                low = t;
43            } else {
44                high = t;
45            }
46        }
47    }
48    bezier(p1.1, p2.1, t)
49}

51pub(crate) fn numbers(value: &Value) -> Vec<f32> {
52    match value {
53        Value::Number(n) => vec![n.as_f64().unwrap_or(0.0) as f32],
54        Value::Array(list) => list.iter().flat_map(numbers).collect(),
55        _ => Vec::new(),
56    }
57}

59/// One dimension's or all dimensions' handle of a keyframe.
60fn handle(value: Option<&Value>, dimension: usize, default: f32) -> f32 {
61    let Some(value) = value else { return default };
62    match value {
63        Value::Array(list) => {
64            list.get(dimension).or_else(|| list.first()).and_then(Value::as_f64).map_or(default, |v| v as f32)
65        }
66        Value::Number(n) => n.as_f64().map_or(default, |v| v as f32),
67        _ => default,
68    }
69}

78#[derive(Clone, Debug)]
79pub(crate) struct Key<T> {
80    frame: f32,
81    value: T,
82    /// Holds until the next keyframe.
83    hold: bool,
84    easing: Vec<Easing>,
85    /// A position moves along a curve between these handles.
86    tangent_out: Option<Vec<f32>>,
87    tangent_in: Option<Vec<f32>>,
88}

97fn parse_keys<T: Clone>(list: &[Value], value_of: impl Fn(&Value) -> Option<T>) -> Vec<Key<T>> {
98    let mut keys: Vec<Key<T>> = Vec::with_capacity(list.len());
99    for (i, key) in list.iter().enumerate() {
100        let frame = key.get("t").and_then(Value::as_f64).unwrap_or(0.0) as f32;
101        let start = key.get("s").and_then(&value_of);
102        // Old files keep the end of each stretch in `e` and leave the last key empty.
103        let value = match start {
104            Some(value) => value,
105            None => {
106                let before = i.checked_sub(1).and_then(|p| list.get(p));
107                match before.and_then(|k| k.get("e")).and_then(&value_of) {
108                    Some(value) => value,
109                    None => match keys.last() {
110                        Some(last) => last.value.clone(),
111                        None => continue,
112                    },
113                }
114            }
115        };
116        let (o, i_handle) = (key.get("o"), key.get("i"));
117        let dimensions = [o, i_handle]
118            .iter()
119            .filter_map(|h| h.and_then(|h| h.get("x")).and_then(Value::as_array).map(Vec::len))
120            .max()
121            .unwrap_or(1);
122        let easing = (0..dimensions)
123            .map(|d| Easing {
124                out: (handle(o.and_then(|h| h.get("x")), d, 0.0), handle(o.and_then(|h| h.get("y")), d, 0.0)),
125                into: (
126                    handle(i_handle.and_then(|h| h.get("x")), d, 1.0),
127                    handle(i_handle.and_then(|h| h.get("y")), d, 1.0),
128                ),
129            })
130            .collect();
131        keys.push(Key {
132            frame,
133            value,
134            hold: key.get("h").and_then(Value::as_i64) == Some(1),
135            easing,
136            tangent_out: key.get("to").map(numbers).filter(|v| !v.is_empty()),
137            tangent_in: key.get("ti").map(numbers).filter(|v| !v.is_empty()),
138        });
139    }
140    keys
141}

143/// Which two keys `frame` falls between, and how far along.
144fn locate<T>(keys: &[Key<T>], frame: f32) -> Option<(&Key<T>, Option<&Key<T>>)> {
145    let first = keys.first()?;
146    if frame <= first.frame || keys.len() == 1 {
147        return Some((first, None));
148    }
149    for pair in keys.windows(2) {
150        if frame < pair[1].frame {
151            return Some((&pair[0], Some(&pair[1])));
152        }
153    }
154    keys.last().map(|last| (last, None))
155}

157impl Numbers {
158    pub(crate) fn parse(property: Option<&Value>) -> Self {
159        let Some(property) = property else { return Self::Fixed(Vec::new()) };
160        let Some(k) = property.get("k") else {
161            return Self::Fixed(numbers(property));
162        };
163        if let Some(list) = k.as_array()
164            && list.first().is_some_and(|first| first.get("t").is_some())
165        {
166            let keys = parse_keys(list, |v| Some(numbers(v)).filter(|n| !n.is_empty()));
167            if !keys.is_empty() {
168                return Self::Keys(keys);
169            }
170        }
171        Self::Fixed(numbers(k))
172    }
173
174    pub(crate) fn at(&self, frame: f32) -> Vec<f32> {
175        match self {
176            Self::Fixed(value) => value.clone(),
177            Self::Keys(keys) => {
178                let Some((a, next)) = locate(keys, frame) else { return Vec::new() };
179                let Some(b) = next.filter(|_| !a.hold) else { return a.value.clone() };
180                let span = (b.frame - a.frame).max(1e-6);
181                let x = (frame - a.frame) / span;
182                // A curve through space, eased as a whole.
183                if let (Some(to), Some(ti)) = (
184                    &a.tangent_out,
185                    &a.tangent_in.as_ref().or(b.tangent_in.as_ref()).or(a.tangent_in.as_ref()),
186                ) && a.value.len() >= 2
187                    && b.value.len() >= 2
188                    && to.len() >= 2
189                    && ti.len() >= 2
190                    && let Some(e) = a.easing.first()
191                {
192                    let progress = timing(e.out, e.into, x);
193                    let dimensions = a.value.len().min(b.value.len());
194                    let point = |t: f32| -> Vec<f32> {
195                        let u = 1.0 - t;
196                        (0..dimensions)
197                            .map(|d| {
198                                let (p0, p3) = (a.value[d], b.value[d]);
199                                let p1 = p0 + to.get(d).copied().unwrap_or(0.0);
200                                let p2 = p3 + ti.get(d).copied().unwrap_or(0.0);
201                                u * u * u * p0 + 3.0 * u * u * t * p1 + 3.0 * u * t * t * p2 + t * t * t * p3
202                            })
203                            .collect()
204                    };
205                    // The easing runs along the length of the curve, not its parameter.
206                    let samples: Vec<Vec<f32>> =
207                        (0..=ARC_SAMPLES).map(|i| point(i as f32 / ARC_SAMPLES as f32)).collect();
208                    let lengths: Vec<f32> = samples
209                        .windows(2)
210                        .map(|p| p[0].iter().zip(&p[1]).map(|(q, r)| (q - r).powi(2)).sum::<f32>().sqrt())
211                        .collect();
212                    let total: f32 = lengths.iter().sum();
213                    if total <= 0.0 {
214                        return point(progress);
215                    }
216                    let mut left = progress.clamp(0.0, 1.0) * total;
217                    for (i, length) in lengths.iter().enumerate() {
218                        if left <= *length {
219                            let k = if *length > 0.0 { left / length } else { 0.0 };
220                            return point((i as f32 + k) / ARC_SAMPLES as f32);
221                        }
222                        left -= length;
223                    }
224                    return point(1.0);
225                }
226                (0..a.value.len().min(b.value.len()))
227                    .map(|d| {
228                        let e = a.easing.get(d).or_else(|| a.easing.first());
229                        let t = e.map_or(x, |e| timing(e.out, e.into, x));
230                        a.value[d] + (b.value[d] - a.value[d]) * t
231                    })
232                    .collect()
233            }
234        }
235    }
236
237    pub(crate) fn first_at(&self, frame: f32, default: f32) -> f32 {
238        self.at(frame).first().copied().unwrap_or(default)
239    }
240
241    pub(crate) fn pair_at(&self, frame: f32, default: (f32, f32)) -> (f32, f32) {
242        let value = self.at(frame);
243        (value.first().copied().unwrap_or(default.0), value.get(1).copied().unwrap_or(default.1))
244    }
245}

247/// The points of a path at a keyframe: vertices with their handles.
248#[derive(Clone, Debug, Default)]
249pub(crate) struct Bezier {
250    pub closed: bool,
251    pub vertices: Vec<(f32, f32)>,
252    pub tangents_in: Vec<(f32, f32)>,
253    pub tangents_out: Vec<(f32, f32)>,
254}
255
256fn pairs(value: Option<&Value>) -> Vec<(f32, f32)> {
257    value
258        .and_then(Value::as_array)
259        .map(|list| {
260            list.iter()
261                .map(|pair| {
262                    let n = numbers(pair);
263                    (n.first().copied().unwrap_or(0.0), n.get(1).copied().unwrap_or(0.0))
264                })
265                .collect()
266        })
267        .unwrap_or_default()
268}
269
270fn parse_bezier(value: &Value) -> Option<Bezier> {
271    // Keyframed shapes wrap each value in a list of one.
272    let value = value.as_array().and_then(|list| list.first()).unwrap_or(value);
273    let vertices = pairs(value.get("v"));
274    if vertices.is_empty() {
275        return None;
276    }
277    Some(Bezier {
278        closed: value.get("c").and_then(Value::as_bool).unwrap_or(false),
279        tangents_in: pairs(value.get("i")),
280        tangents_out: pairs(value.get("o")),
281        vertices,
282    })
283}

285/// A property with a path for a value.
286#[derive(Clone, Debug)]
287pub(crate) enum Shape {
288    Fixed(Bezier),
289    Keys(Vec<Key<Bezier>>),
290}
291
292impl Shape {
293    pub(crate) fn parse(property: Option<&Value>) -> Self {
294        let k = property.and_then(|p| p.get("k"));
295        if let Some(list) = k.and_then(Value::as_array)
296            && list.first().is_some_and(|first| first.get("t").is_some())
297        {
298            let keys = parse_keys(list, parse_bezier);
299            if !keys.is_empty() {
300                return Self::Keys(keys);
301            }
302        }
303        Self::Fixed(k.and_then(parse_bezier).unwrap_or_default())
304    }
305
306    pub(crate) fn at(&self, frame: f32) -> Bezier {
307        match self {
308            Self::Fixed(path) => path.clone(),
309            Self::Keys(keys) => {
310                let Some((a, next)) = locate(keys, frame) else { return Bezier::default() };
311                let Some(b) = next.filter(|_| !a.hold) else { return a.value.clone() };
312                let x = (frame - a.frame) / (b.frame - a.frame).max(1e-6);
313                let t = a.easing.first().map_or(x, |e| timing(e.out, e.into, x));
314                let mix = |from: &[(f32, f32)], to: &[(f32, f32)]| -> Vec<(f32, f32)> {
315                    from.iter().zip(to).map(|(p, q)| (p.0 + (q.0 - p.0) * t, p.1 + (q.1 - p.1) * t)).collect()
316                };
317                // Paths of different lengths cannot blend: the first holds.
318                if a.value.vertices.len() != b.value.vertices.len() {
319                    return a.value.clone();
320                }
321                Bezier {
322                    closed: a.value.closed,
323                    vertices: mix(&a.value.vertices, &b.value.vertices),
324                    tangents_in: mix(&a.value.tangents_in, &b.value.tangents_in),
325                    tangents_out: mix(&a.value.tangents_out, &b.value.tangents_out),
326                }
327            }
328        }
329    }
330}

339    fn a_fixed_value_stays_and_a_number_is_a_value_of_one() {

345    #[test]
346    fn keys_ease_between_them_and_hold_outside() {
347        let keys = json!({"a": 1, "k": [
348            {"t": 0, "s": [0], "o": {"x": [0], "y": [0]}, "i": {"x": [1], "y": [1]}},
349            {"t": 10, "s": [100]}
350        ]});
351        let value = Numbers::parse(Some(&keys));
352        assert_eq!(value.first_at(-1.0, 0.0), 0.0);
353        assert!((value.first_at(5.0, 0.0) - 50.0).abs() < 0.5, "a straight line here");
354        assert_eq!(value.first_at(11.0, 0.0), 100.0);
355        let eased = json!({"a": 1, "k": [
356            {"t": 0, "s": [0], "o": {"x": 0.8, "y": 0}, "i": {"x": 1, "y": 1}},
357            {"t": 10, "s": [100]}
358        ]});
359        assert!(Numbers::parse(Some(&eased)).first_at(5.0, 0.0) < 40.0, "a slow start");
360    }
