19/// How long a worker holds a run before another may take it, unless it renews: it renews at
20/// every step.
21pub const LEASE_MS: u64 = 60_000;
22
23/// Where a run is.
24#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
25#[serde(rename_all = "snake_case")]
26#[non_exhaustive]
27pub enum RunStatus {
28    /// Asked for, not taken yet.
29    Queued,
30    /// A worker holds it.
31    Running,
32    /// It waits for a time or an event.
33    Waiting,
34    /// Every node that could run, ran.
35    Finished,
36    /// It stopped at a node that failed.
37    Stopped,
38    /// It was cancelled.
39    Cancelled,
40    /// It took longer than its time limit.
41    TimedOut,
42}
43
44impl RunStatus {
45    fn as_str(self) -> &'static str {
46        match self {
47            Self::Queued => "queued",
48            Self::Running => "running",
49            Self::Waiting => "waiting",
50            Self::Finished => "finished",
51            Self::Stopped => "stopped",
52            Self::Cancelled => "cancelled",
53            Self::TimedOut => "timed_out",
54        }
55    }
56
57    fn parse(text: &str) -> Result<Self, StoreError> {
58        Ok(match text {
59            "queued" => Self::Queued,
60            "running" => Self::Running,
61            "waiting" => Self::Waiting,
62            "finished" => Self::Finished,
63            "stopped" => Self::Stopped,
64            "cancelled" => Self::Cancelled,
65            "timed_out" => Self::TimedOut,
66            other => return Err(StoreError::Unreadable(format!("a run status {other}"))),
67        })
68    }
69
70    /// Whether the run is over and will not run again by itself.
71    pub fn ended(self) -> bool {
72        matches!(self, Self::Finished | Self::Stopped | Self::Cancelled | Self::TimedOut)
73    }
74
75    fn of(outcome: &RunOutcome) -> Self {
76        match outcome {
77            RunOutcome::Finished => Self::Finished,
78            RunOutcome::Cancelled { .. } => Self::Cancelled,
79            RunOutcome::TimedOut { .. } => Self::TimedOut,
80            RunOutcome::Waiting { .. } => Self::Waiting,
81            _ => Self::Stopped,
82        }
83    }
84}

183/// Durable runs of flows, kept in a SQLite file that several processes may share: each claims a
184/// run with a lease, runs it writing a checkpoint after every node, and a run left by a worker
185/// that died goes back to the queue when its lease runs out.
186pub struct Runtime {
187    conn: Mutex<Connection>,
188    catalog: Catalog,
189    clock: Arc<dyn Clock>,
190    lease_ms: u64,
191}

241    fn insert(&self, request: &Request, origin: Option<&str>) -> Result<String, StoreError> {
242        let now = self.now();
243        let id = store::new_id("run", now);
244        let mode = if request.mode == Mode::Test { "test" } else { "active" };
245        self.db().execute(
246            "INSERT INTO runs (id, flow_id, revision, definition, mode, actor, start_node, input, status, origin, created_ms)
247             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, 'queued', ?9, ?10)",
248            params![
249                id,
250                request.flow.id,
251                request.flow.revision,
252                request.flow.to_json(),
253                mode,
254                request.actor,
255                request.start_node,
256                serde_json::to_string(&request.input)?,
257                origin,
258                now
259            ],
260        )?;
261        Ok(id)
262    }

264    /// Takes the run due first, for `worker`: one asked for, one whose worker's lease ran out,
265    /// or one whose wait is over. Two workers never take the same run.
266    pub fn claim(&self, worker: &str) -> Result<Option<String>, StoreError> {
267        let now = self.now();
268        store::claim(&mut self.db(), worker, now, now.saturating_add(self.lease_ms as i64))
269    }

271    /// Takes the run due first and runs it, as `worker`; its id, or `None` when nothing was due.
272    pub fn work(&self, worker: &str) -> Result<Option<String>, StoreError> {
273        match self.claim(worker)? {
274            Some(id) => {
275                self.execute(&id, worker)?;
276                Ok(Some(id))
277            }
278            None => Ok(None),
279        }
280    }

282    /// Runs whatever is due, one after another, until nothing is; how many it ran.
283    pub fn work_until_idle(&self, worker: &str) -> Result<usize, StoreError> {
284        let mut ran = 0;
285        while self.work(worker)?.is_some() {
286            ran += 1;
287        }
288        Ok(ran)
289    }

291    /// Runs `id`, which `worker` holds: from its checkpoint when it ran before. Each step is
292    /// written as it happens, and the lease renewed.
293    pub fn execute(&self, id: &str, worker: &str) -> Result<RunStatus, StoreError> {
294        let row = self.row(id)?;
295        let flow =
296            FlowDefinition::from_json(&row.definition).map_err(|e| StoreError::Unreadable(e.to_string()))?;
297        let plan = match Plan::compile(&flow, &self.catalog) {
298            Ok(plan) => plan,
299            Err(errors) => {
300                let reason = errors.iter().map(ToString::to_string).collect::<Vec<_>>().join("; ");
301                let outcome = RunOutcome::Stopped { node: String::new(), reason };
302                return self.end(id, worker, &outcome);
303            }
304        };
305        let started = match row.started_ms {
306            Some(ms) => ms,
307            None => {
308                let now = self.now();
309                self.db().execute("UPDATE runs SET started_ms = ?2 WHERE id = ?1", params![id, now])?;
310                now
311            }
312        };
313        let Kept { checkpoint, in_progress, woken } = self.checkpoint(id, &plan, started)?;
314        let mut events = BTreeMap::new();
315        for (name, items) in store::events(&self.db(), id)? {
316            events.insert(name, serde_json::from_str::<Vec<Item>>(&items)?);
317        }
318        let failure: RefCell<Option<StoreError>> = RefCell::new(None);
319        let lost = RefCell::new(false);
320        let mut executor = Executor::new(&plan, &self.catalog)
321            .mode(row.mode)
322            .shared_clock(self.clock.clone())
323            .events(events)
324            .woken(woken)
325            .on_step(|step| {
326                if let Err(e) = self.write_step(id, worker, step, &lost) {
327                    failure.borrow_mut().get_or_insert(e);
328                }
329            });
330        if let Some(node) = &row.start_node {
331            executor = executor.start_with(node.clone(), row.input.clone());
332        }
333        let run = executor.resume(&checkpoint, &in_progress);
334        drop(executor);
335        if let Some(e) = failure.into_inner() {
336            return Err(e);
337        }
338        if lost.into_inner() {
339            return Ok(RunStatus::Running);
340        }
341        self.end(id, worker, &run.outcome)
342    }
343
344    fn write_step(
345        &self,
346        id: &str,
347        worker: &str,
348        step: Step<'_>,
349        lost: &RefCell<bool>,
350    ) -> Result<(), StoreError> {
351        let db = self.db();
352        match step {
353            Step::Started { node } => store::started(&db, id, node)?,
354            Step::Settled { node, part } => store::settled(&db, id, node, &serde_json::to_string(part)?)?,
355            Step::LoopDone { head } => store::loop_done(&db, id, head)?,
356            _ => {}
357        }
358        let until = self.now().saturating_add(self.lease_ms as i64);
359        if !store::renew(&db, id, worker, until)? {
360            *lost.borrow_mut() = true;
361        }
362        Ok(())
363    }

365    /// What `id` kept: the parts of the nodes that settled (a loop only once it ended), the
366    /// nodes it was in when it stopped, and how each waiting node waits.
367    fn checkpoint(&self, id: &str, plan: &Plan, started: i64) -> Result<Kept, StoreError> {
368        let db = self.db();
369        let ended: BTreeSet<String> = store::loops(&db, id)?.into_iter().collect();
370        let mut unfinished: BTreeSet<&str> = BTreeSet::new();
371        for head in plan.order().iter().filter(|n| !plan.loop_body(n).is_empty()) {
372            if !ended.contains(head) {
373                unfinished.insert(head);
374                unfinished.extend(plan.loop_body(head).iter().map(String::as_str));
375            }
376        }
377        let mut run = Run::default();
378        run.started_ms = started;
379        let (mut busy, mut woken) = (BTreeSet::new(), BTreeMap::new());
380        for step in store::steps(&db, id)? {
381            if unfinished.contains(step.node.as_str()) {
382                continue;
383            }
384            match (step.settled, step.part) {
385                (true, Some(json)) => {
386                    let part: NodeRun = serde_json::from_str(&json)?;
387                    if part.status == NodeStatus::Waiting {
388                        woken.insert(step.node.clone(), part.wake.clone().unwrap_or_default());
389                    }
390                    run.order.push(step.node.clone());
391                    run.nodes.insert(step.node, part);
392                }
393                _ => {
394                    busy.insert(step.node);
395                }
396            }
397        }
398        Ok(Kept { checkpoint: run, in_progress: busy, woken })
399    }

401    fn end(&self, id: &str, worker: &str, outcome: &RunOutcome) -> Result<RunStatus, StoreError> {
402        let status = RunStatus::of(outcome);
403        let (wake_at, wake_event) = match outcome {
404            RunOutcome::Waiting { wake, .. } => (wake.until_ms, wake.event.clone()),
405            _ => (None, None),
406        };
407        let ended = status.ended().then(|| self.now());
408        self.db().execute(
409            "UPDATE runs SET status = ?3, outcome = ?4, wake_at = ?5, wake_event = ?6, ended_ms = ?7, lease_until = NULL
410             WHERE id = ?1 AND worker = ?2",
411            params![id, worker, status.as_str(), serde_json::to_string(outcome)?, wake_at, wake_event, ended],
412        )?;
413        Ok(status)
414    }

416    /// Tells a run waiting for `event` that it came, with `items`; it is due at once.
417    pub fn signal(&self, id: &str, event: &str, items: Vec<Item>) -> Result<(), StoreError> {
418        let db = self.db();
419        db.execute(
420            "INSERT INTO events (run_id, name, items) VALUES (?1, ?2, ?3)
421             ON CONFLICT (run_id, name) DO UPDATE SET items = excluded.items",
422            params![id, event, serde_json::to_string(&items)?],
423        )?;
424        db.execute(
425            "UPDATE runs SET wake_at = ?3 WHERE id = ?1 AND status = 'waiting' AND wake_event = ?2",
426            params![id, event, self.now()],
427        )?;
428        Ok(())
429    }

431    /// Asks for `id` to run again, with the same input, on the flow `revision` says; the new
432    /// run's id, which keeps where it came from.
433    pub fn rerun(&self, id: &str, revision: Revision) -> Result<String, StoreError> {
434        let row = self.row(id)?;
435        let flow = match revision {
436            Revision::Current(flow) => flow,
437            _ => FlowDefinition::from_json(&row.definition)
438                .map_err(|e| StoreError::Unreadable(e.to_string()))?,
439        };
440        let request =
441            Request { flow, start_node: row.start_node, input: row.input, actor: row.actor, mode: row.mode };
442        self.insert(&request, Some(id))
443    }

486        let ids: Vec<String> = {

526    /// Forgets the ended runs `retention` says, with their steps, events and files.
527    pub fn prune(&self, retention: &Retention) -> Result<Pruned, StoreError> {
528        let now = self.now();
529        let candidates: Vec<(String, String, String, i64)> = {
530            let db = self.db();
531            let mut stmt = db.prepare(
532                "SELECT id, flow_id, status, ended_ms FROM runs WHERE ended_ms IS NOT NULL ORDER BY flow_id, ended_ms DESC, id DESC",
533            )?;
534            let rows = stmt.query_map([], |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?, row.get(3)?)))?;
535            rows.collect::<Result<_, _>>()?
536        };
537        let mut seen: BTreeMap<String, usize> = BTreeMap::new();
538        let mut pruned = Pruned::default();
539        for (id, flow, status, ended) in candidates {
540            let place = seen.entry(flow).or_default();
541            *place += 1;
542            if retention.keep_last.is_some_and(|keep| *place <= keep) {
543                continue;
544            }
545            if retention.older_than_ms.is_some_and(|age| now - ended < age as i64) {
546                continue;
547            }
548            let status = RunStatus::parse(&status)?;
549            if !retention.statuses.is_empty() && !retention.statuses.contains(&status) {
550                continue;
551            }
552            pruned.blobs += store::delete(&self.db(), &id)?;
553            pruned.runs += 1;
554        }
555        Ok(pruned)
556    }

558    fn row(&self, id: &str) -> Result<RunRow, StoreError> {
559        let raw = self
560            .db()
561            .query_row(
562                "SELECT definition, mode, actor, start_node, input, started_ms FROM runs WHERE id = ?1",
563                params![id],
564                |row| {
565                    Ok((
566                        row.get::<_, String>(0)?,
567                        row.get::<_, String>(1)?,
568                        row.get::<_, Option<String>>(2)?,
569                        row.get::<_, Option<String>>(3)?,
570                        row.get::<_, String>(4)?,
571                        row.get::<_, Option<i64>>(5)?,
572                    ))
573                },
574            )
575            .optional()?
576            .ok_or_else(|| StoreError::NoRun(id.to_owned()))?;
577        let (definition, mode, actor, start_node, input, started_ms) = raw;
578        Ok(RunRow {
579            definition,
580            mode: if mode == "test" { Mode::Test } else { Mode::Active },
581            actor,
582            start_node,
583            input: serde_json::from_str(&input)?,
584            started_ms,
585        })
586    }
