1//! Synthetic data for testing a reusable `ravenrs-app` module, as `docs/APP-CONTRACTS.md` describes
2//! it: two organizations, T1 and T2, that use **the same local keys** on purpose, so that mixing one
3//! with the other shows; units and people in each; example roles; and a clock that moves only when
4//! a test says so; and a destination for events that answers as the test asks.
5//!
6//! Each organization keeps its records apart, in memory or in a SQLite file of its own. Nothing here
7//! belongs to a sector: the module under test brings its own entities, and the roles are an example
8//! policy of the test, not profiles built into RavenRS.
9//!
10//! ```
11//! use ravenrs_app::{Identity, ObjectSpace, Query};
12//! use ravenrs_app_testkit::{Fixture, PERSON, roles};
13//!
14//! let fixture = Fixture::memory(Vec::new()).unwrap();
15//! let registrar = Identity::new("op", "op", vec![roles::registrar("U1")]);
16//! let space = fixture.t1.space_as(registrar);
17//! let people = space.query(PERSON, &Query::all()).unwrap();
18//! assert_eq!(people.len(), 1); // only the person of U1
19//! ```

35/// The entity of units, the scope of access inside an organization.
36pub const UNIT: &str = "Unit";
37/// The entity of people.
38pub const PERSON: &str = "Person";

40/// A unit: a key such as `U1` and a name.
41pub fn unit_entity() -> EntityInfo {
42    EntityInfo::new(UNIT)
43        .field(FieldInfo::new("id", FieldKind::Text).key())
44        .field(FieldInfo::new("name", FieldKind::Text).default_property())
45}
46
47/// A person: a number for a key, the unit they belong to, a name, a contact and a reserved note
48/// that only some roles may read.
49pub fn person_entity() -> EntityInfo {
50    EntityInfo::new(PERSON)
51        .field(FieldInfo::new("id", FieldKind::Integer).key())
52        .field(FieldInfo::new("unit", FieldKind::Text))
53        .field(FieldInfo::new("name", FieldKind::Text).default_property())
54        .field(FieldInfo::new("contact", FieldKind::Text))
55        .field(FieldInfo::new("note", FieldKind::LongText))
56}

58/// The entities every organization of the fixture has: units, people, events waiting to be sent
59/// and those of security.
60pub fn fixture_entities() -> Vec<EntityInfo> {
61    let mut all = vec![unit_entity(), person_entity(), outbox_entity()];
62    all.extend(security_entities());
63    all

66/// One organization, with its records kept apart from every other.
67pub struct Organization {
68    name: String,
69    entities: Vec<EntityInfo>,
70    maker: SpaceMaker,
71    directory: UserDirectory,
72}
73
74impl fmt::Debug for Organization {
75    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
76        f.debug_struct("Organization").field("name", &self.name).finish_non_exhaustive()
77    }
78}
79
80impl Organization {
81    /// An organization kept in memory, with the fixture's entities and these others.
82    pub fn memory(name: impl Into<String>, extra: Vec<EntityInfo>) -> Result<Self, String> {
83        let entities = with_fixture(extra);
84        let database = MemoryDatabase::new(entities.clone()).map_err(|e| e.to_string())?;
85        let maker: SpaceMaker = Rc::new(move || Box::new(database.space()) as Box<dyn ObjectSpace>);
86        Self::over(name.into(), entities, maker)
87    }
88
89    /// An organization kept in a SQLite file, created when it is not there. Opening the same file
90    /// again is how a test restarts an application.
91    pub fn sqlite(name: impl Into<String>, file: &Path, extra: Vec<EntityInfo>) -> Result<Self, String> {
92        let entities = with_fixture(extra);
93        let connection = Rc::new(rusqlite::Connection::open(file).map_err(|e| e.to_string())?);
94        let mut builder = SqliteDatabase::builder(connection);
95        for entity in &entities {
96            let table = entity.name.to_lowercase();
97            builder = builder.managed(entity.clone(), TableMap::new(table)).map_err(|e| e.to_string())?;
98        }
99        let database = builder.build();
100        let maker: SpaceMaker = Rc::new(move || Box::new(database.space()) as Box<dyn ObjectSpace>);
101        Self::over(name.into(), entities, maker)
102    }
103
104    fn over(name: String, entities: Vec<EntityInfo>, maker: SpaceMaker) -> Result<Self, String> {
105        let directory =
106            UserDirectory::new(maker.clone(), vec![roles::events_worker()]).map_err(|e| e.to_string())?;
107        Ok(Self { name, entities, maker, directory })
108    }
109
110    /// What it is called, which is also what tells its keys apart from another organization's.
111    pub fn name(&self) -> &str {
112        &self.name
113    }
114
115    /// A unit of work with no security, for putting records in place and for checking what was kept.
116    pub fn raw(&self) -> Box<dyn ObjectSpace> {
117        (self.maker)()
118    }
119
120    /// A unit of work as this person, every read and write checked by their roles.
121    pub fn space_as(&self, identity: Identity) -> SecuredObjectSpace {
122        SecuredObjectSpace::new(self.raw(), self.context_of(identity))
123    }
124
125    /// What the security system knows of this person here, to be shared by their units of work.
126    pub fn context_of(&self, identity: Identity) -> Rc<SecurityContext> {
127        Rc::new(SecurityContext::new(
128            identity,
129            self.directory.clone(),
130            self.entities.clone(),
131            RoleMerge::AnyRole,
132        ))
133    }
134
135    /// Puts in place someone who can sign in, with a role kept as a record, such as
136    /// [`roles::events_worker`], and says who they are.
137    pub fn add_user(&self, name: &str, role: &str) -> Result<Identity, SpaceError> {
138        let mut space = self.raw();
139        let found = space
140            .query(SECURITY_ROLE, &Query::all())?
141            .into_iter()
142            .find(|record| record.get("name") == Value::Text(role.to_owned()))
143            .ok_or_else(|| SpaceError::Refused(format!("no role named {role}")))?;
144        let mut user = space.new_object(SECURITY_USER)?;
145        user.set("name", name);
146        let key = user.get("id");
147        space.save(user)?;
148        let mut link = space.new_object(SECURITY_USER_ROLE)?;
149        link.set("user_id", key);
150        link.set("role_id", found.get("id"));
151        space.save(link)?;
152        space.commit()?;
153        self.directory.identity_of(name)?.ok_or_else(|| SpaceError::Refused(format!("{name} cannot sign in")))
154    }
155
156    /// Turns someone off: from their next reload on, they can do nothing.
157    pub fn turn_off(&self, name: &str) -> Result<(), SpaceError> {
158        let mut space = self.raw();
159        for mut user in space.query(SECURITY_USER, &Query::all())? {
160            if user.get("name") == Value::Text(name.to_owned()) {
161                user.set("active", false);
162                space.save(user)?;
163            }
164        }
165        space.commit()
166    }
167
168    /// Puts a unit in place, once.
169    pub fn add_unit(&self, id: &str, name: &str) -> Result<(), SpaceError> {
170        let mut space = self.raw();
171        if space.find(UNIT, &Value::Text(id.to_owned()))?.is_none() {
172            let mut unit = space.new_object(UNIT)?;
173            unit.set("id", id);
174            unit.set("name", name);
175            space.save(unit)?;
176            space.commit()?;
177        }
178        Ok(())
179    }
180
181    /// Puts a person in place, once.
182    pub fn add_person(&self, id: i64, unit: &str, name: &str, note: &str) -> Result<(), SpaceError> {
183        let mut space = self.raw();
184        if space.find(PERSON, &Value::Number(id as f64))?.is_none() {
185            let mut person = space.new_object(PERSON)?;
186            person.set("id", id as f64);
187            person.set("unit", unit);
188            person.set("name", name);
189            person.set("contact", format!("{name} contact"));
190            person.set("note", note);
191            space.save(person)?;
192            space.commit()?;
193        }
194        Ok(())
195    }
196}

198fn with_fixture(extra: Vec<EntityInfo>) -> Vec<EntityInfo> {
199    let mut entities = fixture_entities();
200    entities.extend(extra);
201    entities
202}

204/// The two organizations of the fixture.
205///
206/// T1 has units U1 and U2, person 1 in U1 and person 2 in U2. T2 has unit U1 and its own person 1,
207/// with the same key as the person 1 of T1.
208#[derive(Debug)]
209pub struct Fixture {
210    /// The first organization.
211    pub t1: Organization,
212    /// The second, with the same local keys as the first.
213    pub t2: Organization,
214}
215
216impl Fixture {
217    /// Both kept in memory, with these entities besides the fixture's.
218    pub fn memory(extra: Vec<EntityInfo>) -> Result<Self, String> {
219        let fixture =
220            Self { t1: Organization::memory("T1", extra.clone())?, t2: Organization::memory("T2", extra)? };
221        fixture.seed().map_err(|e| e.to_string())?;
222        Ok(fixture)
223    }
224
225    /// Each kept in its own SQLite file inside this folder. Opening the same folder again finds what
226    /// was kept, as a restart would.
227    pub fn sqlite(folder: &Path, extra: Vec<EntityInfo>) -> Result<Self, String> {
228        let fixture = Self {
229            t1: Organization::sqlite("T1", &folder.join("t1.sqlite"), extra.clone())?,
230            t2: Organization::sqlite("T2", &folder.join("t2.sqlite"), extra)?,
231        };
232        fixture.seed().map_err(|e| e.to_string())?;
233        Ok(fixture)
234    }
235
236    fn seed(&self) -> Result<(), SpaceError> {
237        self.t1.add_unit("U1", "Unit 1 of T1")?;
238        self.t1.add_unit("U2", "Unit 2 of T1")?;
239        self.t1.add_person(1, "U1", "P1", "reserved note of P1")?;
240        self.t1.add_person(2, "U2", "P2", "reserved note of P2")?;
241        self.t2.add_unit("U1", "Unit 1 of T2")?;
242        self.t2.add_person(1, "U1", "P1 of T2", "reserved note of the P1 of T2")
243    }
244}

246/// Roles of the example policy. They exist for tests; an application writes its own.
247pub mod roles {
248    use ravenrs_app::{Access, Operation, Role};
249
250    use ravenrs_app::OUTBOX_EVENT;
251
252    use super::{PERSON, UNIT};
253
254    /// The role of the worker that sends events: it reads and changes the events and nothing else.
255    /// Every organization of the fixture keeps it as a record, for [`add_user`](super::Organization::add_user).
256    pub fn events_worker() -> Role {
257        Role::new("Events worker").type_permission(
258            OUTBOX_EVENT,
259            [Operation::Read, Operation::Write],
260            Access::Allow,
261        )
262    }
263
264    /// Administers everything of the organization it is used in.
265    pub fn administrator() -> Role {
266        Role::new("Administrator").administrative()
267    }
268
269    /// Reads, creates and changes the people of one unit, never their reserved note, and reads units.
270    pub fn registrar(unit: &str) -> Role {
271        let mine = format!("[unit] = '{unit}'");
272        Role::new(format!("Registrar of {unit}"))
273            .type_permission(UNIT, [Operation::Read], Access::Allow)
274            .object_permission(
275                PERSON,
276                &mine,
277                [Operation::Read, Operation::Write, Operation::Create],
278                Access::Allow,
279            )
280            .member_permission(PERSON, ["note"], None, [Operation::Read, Operation::Write], Access::Deny)
281    }
282
283    /// Reads and writes only these entities, as a worker granted one job would.
284    pub fn worker(entities: &[&str]) -> Role {
285        entities.iter().fold(Role::new("Worker"), |role, entity| {
286            role.type_permission(
287                *entity,
288                [Operation::Read, Operation::Write, Operation::Create],
289                Access::Allow,
290            )
291        })
292    }
293}

295/// A clock that moves only when told to, in whole seconds since the Unix epoch.
296#[derive(Clone, Debug)]
297pub struct Clock(Rc<Cell<i64>>);
298
299impl Clock {
300    /// A clock stopped at this instant.
301    pub fn at(seconds: i64) -> Self {
302        Self(Rc::new(Cell::new(seconds)))
303    }
304
305    /// What it says now.
306    pub fn now(&self) -> i64 {
307        self.0.get()
308    }
309
310    /// What it says now, as a date and time.
311    pub fn date_time(&self) -> ravenrs_criteria::DateTime {
312        let now = self.now();
313        ravenrs_criteria::DateTime { days: now.div_euclid(86_400), seconds: now.rem_euclid(86_400) as u32 }
314    }
315
316    /// Moves it forward.
317    pub fn advance(&self, seconds: i64) {
318        self.0.set(self.0.get() + seconds);
319    }
320}
321
322impl Default for Clock {
323    /// 12 November 2026, 00:00 UTC.
324    fn default() -> Self {
325        Self::at(1_794_441_600)
326    }
327}

329/// How [`FakeSink`] answers the next send.
330#[derive(Clone, Debug, PartialEq, Eq)]
331pub enum Behavior {
332    /// Keeps it and says so.
333    Accept,
334    /// Refuses it for good.
335    Refuse,
336    /// Cannot be reached this time; nothing is kept.
337    Down,
338    /// Keeps it, and the answer is lost on the way back.
339    LoseReply,
340}
341
342#[derive(Debug, Default)]
343struct SinkState {
344    script: std::collections::VecDeque<Behavior>,
345    kept: std::collections::BTreeSet<String>,
346    sends: usize,
347    blind: bool,
348}
349
350/// A destination for events that behaves as a test says, with no network and no account.
351///
352/// It keeps one effect per key, as a destination honoring idempotency keys does, and counts every
353/// send, so a test sees what was sent again. Clones share what it kept.
354#[derive(Clone, Debug, Default)]
355pub struct FakeSink(Rc<std::cell::RefCell<SinkState>>);
356
357impl FakeSink {
358    /// One that accepts everything and can tell what it received.
359    pub fn new() -> Self {
360        Self::default()
361    }
362
363    /// One that cannot tell what it received: it declares no [`CAN_TELL_STATUS`].
364    pub fn without_status() -> Self {
365        let sink = Self::default();
366        sink.0.borrow_mut().blind = true;
367        sink
368    }
369
370    /// Answers the next send this way; once the script runs out, it accepts.
371    pub fn then(&self, behavior: Behavior) -> &Self {
372        self.0.borrow_mut().script.push_back(behavior);
373        self
374    }
375
376    /// How many effects it kept: one per key.
377    pub fn effects(&self) -> usize {
378        self.0.borrow().kept.len()
379    }
380
381    /// How many times something was sent to it, kept or not.
382    pub fn sends(&self) -> usize {
383        self.0.borrow().sends
384    }
385
386    /// The keys it kept, in order.
387    pub fn keys(&self) -> Vec<String> {
388        self.0.borrow().kept.iter().cloned().collect()
389    }
390}
391
392impl Provider for FakeSink {
393    fn name(&self) -> &str {
394        "fake destination"
395    }
396
397    fn capabilities(&self) -> Vec<String> {
398        if self.0.borrow().blind { Vec::new() } else { vec![CAN_TELL_STATUS.to_owned()] }
399    }
400}
401
402impl EventSink for FakeSink {
403    fn send(&self, key: &str, _payload: &str) -> Result<(), SpaceError> {
404        let mut state = self.0.borrow_mut();
405        state.sends += 1;
406        match state.script.pop_front().unwrap_or(Behavior::Accept) {
407            Behavior::Accept => {
408                state.kept.insert(key.to_owned());
409                Ok(())
410            }
411            Behavior::Refuse => Err(SpaceError::Refused("refused by the fake destination".to_owned())),
412            Behavior::Down => Err(SpaceError::Temporary("the fake destination is out of reach".to_owned())),
413            Behavior::LoseReply => {
414                state.kept.insert(key.to_owned());
415                Err(SpaceError::Unknown("the answer of the fake destination was lost".to_owned()))
416            }
417        }
418    }
419
420    fn status(&self, key: &str) -> Result<bool, SpaceError> {
421        let state = self.0.borrow();
422        if state.blind {
423            return Err(SpaceError::Unavailable("the fake destination cannot tell".to_owned()));
424        }
425        Ok(state.kept.contains(key))
426    }
