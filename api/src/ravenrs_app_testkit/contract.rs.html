1//! The contract every [`ObjectSpace`] adapter keeps, as a battery any adapter runs with one call:
2//! SQLite, PostgreSQL, PostgREST and the ones to come behave the same before they are called
3//! supported.
4//!
5//! An adapter says how to make fresh storage for a case, and how to reach the same storage a second
6//! time, as another program would. Each case makes its own entities and storage and does not depend
7//! on the order of the others. What an adapter does not guarantee is declared in its
8//! [`Capabilities`], and the case then checks that it fails the declared way, never that it passed.
9//!
10//! ```
11//! use ravenrs_app::{EntityInfo, MemoryDatabase, ObjectSpace, SpaceMaker};
12//! use ravenrs_app_testkit::contract::{self, Adapter, Capabilities, Store};
13//! use std::rc::Rc;
14//!
15//! struct Memory;
16//! impl Adapter for Memory {
17//!     fn name(&self) -> &str {
18//!         "memory"
19//!     }
20//!     fn capabilities(&self) -> Capabilities {
21//!         Capabilities::MEMORY
22//!     }
23//!     fn open(&self, _case: &str, entities: &[EntityInfo]) -> Result<Store, String> {
24//!         let database = MemoryDatabase::new(entities.to_vec()).map_err(|e| e.to_string())?;
25//!         let space: SpaceMaker = Rc::new(move || Box::new(database.space()) as Box<dyn ObjectSpace>);
26//!         Ok(Store { space: space.clone(), again: space })
27//!     }
28//! }
29//!
30//! let report = contract::run(&Memory);
31//! assert!(report.passed.len() > 10);
32//! ```

47/// What happens when two programs make a record at once on the same storage, each numbering keys
48/// on its own.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub enum KeyRace {
51    /// They get different keys: the storage hands them out.
52    Distinct,
53    /// Both take the same next key, and the second commit fails as a whole, with nothing written:
54    /// one program at a time should create records there (#241).
55    SecondCommitFails,
56}
57
58/// What an adapter guarantees where adapters differ. A case whose guarantee is missing checks that
59/// the adapter fails or behaves the declared way, and is reported as a limitation.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub struct Capabilities {
62    /// Two programs creating records at once.
63    pub keys: KeyRace,
64    /// A change to a record another program changed since it was read is refused as a
65    /// [`SpaceError::Conflict`]; without it, the last commit wins.
66    pub lost_updates_detected: bool,
67}
68
69impl Capabilities {
70    /// `MemoryDatabase`: one program, so its spaces share the numbering.
71    pub const MEMORY: Self = Self { keys: KeyRace::Distinct, lost_updates_detected: false };
72    /// `ravenrs-app-sqlite`.
73    pub const SQLITE: Self = Self { keys: KeyRace::SecondCommitFails, lost_updates_detected: false };
74    /// `ravenrs-app-postgres`.
75    pub const POSTGRES: Self = Self { keys: KeyRace::SecondCommitFails, lost_updates_detected: false };
76    /// `ravenrs-app-postgrest`.
77    pub const POSTGREST: Self = Self { keys: KeyRace::SecondCommitFails, lost_updates_detected: false };
78
79    /// Every adapter that comes with RavenRS, by the name `docs/APP-PARITY.md` gives it.
80    pub const ALL: [(&'static str, Self); 4] = [
81        ("MemoryDatabase", Self::MEMORY),
82        ("ravenrs-app-sqlite", Self::SQLITE),
83        ("ravenrs-app-postgres", Self::POSTGRES),
84        ("ravenrs-app-postgrest", Self::POSTGREST),
85    ];
86
87    /// The two cells this adapter's row has in the table of `docs/APP-PARITY.md`.
88    pub fn cells(self) -> [&'static str; 2] {
89        [
90            match self.keys {
91                KeyRace::Distinct => "different keys",
92                KeyRace::SecondCommitFails => "the second commit fails",
93            },
94            if self.lost_updates_detected { "conflict" } else { "the last commit wins" },
95        ]
96    }
97}

99/// Storage made for one case.
100pub struct Store {

102    pub space: SpaceMaker,
103    /// Units of work over the same records as another program would have them: another connection,
104    /// with its own numbering of keys. An adapter that only lives in one program gives `space` again.
105    pub again: SpaceMaker,
106}

114/// An adapter under the contract.
115pub trait Adapter {
116    /// What reports call it.
117    fn name(&self) -> &str;
118    /// What it guarantees where adapters differ.
119    fn capabilities(&self) -> Capabilities;
120    /// Fresh, empty storage for these entities, apart from every other case's. `case` is short,
121    /// lowercase and fit for a table or schema name.
122    fn open(&self, case: &str, entities: &[EntityInfo]) -> Result<Store, String>;
123}

125/// What a run found.
126#[derive(Debug, Default)]
127pub struct Report {
128    /// The cases that passed in full.
129    pub passed: Vec<&'static str>,
130    /// The cases that met a limitation the adapter declared, and behaved as declared.
131    pub limited: Vec<(&'static str, &'static str)>,
132}
133
134type Case = fn(&dyn Adapter) -> Result<Outcome, String>;
135
136/// How a case ended when it did not fail.
137enum Outcome {
138    Passed,
139    Limited(&'static str),
140}
141
142/// Every case, by name.
143const CASES: [(&str, Case); 15] = [
144    ("types", types),
145    ("filters_and_order", filters_and_order),
146    ("pages_only_and_calculated", pages_only_and_calculated),
147    ("uncommitted_is_seen_and_rollback", uncommitted_is_seen_and_rollback),
148    ("all_or_nothing", all_or_nothing),
149    ("gone_record_conflicts", gone_record_conflicts),
150    ("state_machine", state_machine),
151    ("rules_and_watchers", rules_and_watchers),
152    ("audit_in_the_same_commit", audit_in_the_same_commit),
153    ("permission_denied", permission_denied),
154    ("keys_from_two_programs", keys_from_two_programs),
155    ("update_from_two_programs", update_from_two_programs),
156    ("delete_what_another_changed", delete_what_another_changed),
157    ("delete", delete),
158    ("decimals", decimals),
159];
160
161/// The names of the cases, in the order they run, for [`run_only`].
162pub fn cases() -> Vec<&'static str> {
163    CASES.iter().map(|(name, _)| *name).collect()
164}
165
166/// Runs every case on the adapter, each on storage of its own, and says what passed and what met a
167/// declared limitation. Panics with every case that failed and why, after running them all.
168pub fn run(adapter: &dyn Adapter) -> Report {
169    run_only(adapter, |_| true)
170}
171
172/// As [`run`], with only the cases whose name `which` accepts.
173pub fn run_only(adapter: &dyn Adapter, which: impl Fn(&str) -> bool) -> Report {
174    let mut report = Report::default();
175    let mut failed = Vec::new();
176    for (name, case) in CASES {
177        if !which(name) {
178            continue;
179        }
180        match catch_unwind(AssertUnwindSafe(|| case(adapter))) {
181            Ok(Ok(Outcome::Passed)) => report.passed.push(name),
182            Ok(Ok(Outcome::Limited(why))) => report.limited.push((name, why)),
183            Ok(Err(why)) => failed.push(format!("{name}: {why}")),
184            Err(panic) => {
185                let why = panic
186                    .downcast_ref::<String>()
187                    .cloned()
188                    .or_else(|| panic.downcast_ref::<&str>().map(|text| (*text).to_owned()))
189                    .unwrap_or_default();
190                failed.push(format!("{name}: {why}"));
191            }
192        }
193    }
194    for (name, why) in &report.limited {
195        eprintln!("{}: {name} is a declared limitation: {why}", adapter.name());
196    }
197    assert!(failed.is_empty(), "{} broke the contract:\n{}", adapter.name(), failed.join("\n"));
198    report
199}

201/// Fails the case with this, unless it holds.
202macro_rules! ensure {
203    ($holds:expr, $($why:tt)+) => {
204        if !$holds {
205            return Err(format!($($why)+));
206        }
207    };
208}
209
210/// Fails the case unless the two are equal.
211macro_rules! same {
212    ($left:expr, $right:expr) => {{
213        let (left, right) = (&$left, &$right);
214        if left != right {
215            return Err(format!("{} is {left:?}, not {right:?}", stringify!($left)));
216        }
217    }};
218    ($left:expr, $right:expr, $what:expr) => {{
219        let (left, right) = (&$left, &$right);
220        if left != right {
221            return Err(format!("{}: {left:?}, not {right:?}", $what));
222        }
223    }};
224}

234fn add(space: &mut dyn ObjectSpace, entity: &str, values: &[(&str, Value)]) -> Result<Value, String> {
235    let mut object = space.new_object(entity).map_err(wrong)?;
236    for (field, value) in values {
237        object.set(field, value.clone());
238    }
239    let key = object.get("id");
240    space.save(object).map_err(wrong)?;
241    Ok(key)
242}

248fn names(space: &dyn ObjectSpace, entity: &str, query: &Query) -> Result<Vec<String>, String> {
249    let found = space.query(entity, query).map_err(wrong)?;
250    Ok(found
251        .iter()
252        .map(|object| match object.get("name") {
253            Value::Text(text) => text,
254            other => format!("{other:?}"),
255        })
256        .collect())
257}

274fn three_plain(space: &mut dyn ObjectSpace) -> Result<(), String> {
275    for (name, seats, rate, active) in
276        [("Alfa", 1.0, 0.5, true), ("Beta", 2.0, 2.5, false), ("Gama", 3.0, 3.0, true)]
277    {
278        add(
279            space,
280            "Plain",
281            &[
282                ("name", name.into()),
283                ("seats", Value::Number(seats)),
284                ("rate", Value::Number(rate)),
285                ("active", Value::Bool(active)),
286            ],
287        )?;
288    }
289    space.commit().map_err(wrong)
290}

292fn types(adapter: &dyn Adapter) -> Result<Outcome, String> {
293    let entity = EntityInfo::new("Kinds")
294        .field(FieldInfo::new("id", FieldKind::Integer).key())
295        .field(FieldInfo::new("whole", FieldKind::Integer))
296        .field(FieldInfo::new("rate", FieldKind::Decimal(2)))
297        .field(FieldInfo::new("cost", FieldKind::Money))
298        .field(FieldInfo::new("on", FieldKind::Boolean))
299        .field(FieldInfo::new("day", FieldKind::Date))
300        .field(FieldInfo::new("moment", FieldKind::DateTime))
301        .field(FieldInfo::new("text", FieldKind::Text));
302    let store = store(adapter, "types", &[entity])?;
303    let day = DateTime::parse("2026-11-13").ok_or("a date")?;
304    let moment = DateTime::parse("2026-11-13 14:35:20").ok_or("a moment")?;
305    let full = [
306        ("whole", Value::Number(-42.0)),
307        ("rate", Value::Number(16.47)),
308        ("cost", Value::Number(1_234_567.0)),
309        ("on", Value::Bool(true)),
310        ("day", Value::Date(day)),
311        ("moment", Value::Date(moment)),
312        ("text", "Ação, 日本語, emoji 🦀, aspas \" e ' e vírgula, (parênteses)".into()),
313    ];
314    let mut space = (store.space)();
315    let kept = add(&mut *space, "Kinds", &full)?;
316    let empty = add(&mut *space, "Kinds", &[])?;
317    let off = add(&mut *space, "Kinds", &[("on", Value::Bool(false)), ("whole", Value::Number(0.0))])?;
318    space.commit().map_err(wrong)?;
319    let reader = (store.space)();
320    let found = reader.find("Kinds", &kept).map_err(wrong)?.ok_or("the record is not found by its key")?;
321    for (field, value) in &full {
322        same!(found.get(field), *value, field.to_string());
323    }
324    let nothing = reader.find("Kinds", &empty).map_err(wrong)?.ok_or("the empty record is not found")?;
325    for (field, _) in &full {
326        same!(nothing.get(field), Value::Null, format!("{field} left empty"));
327    }
328    let zero = reader.find("Kinds", &off).map_err(wrong)?.ok_or("the third record is not found")?;
329    same!(zero.get("on"), Value::Bool(false));
330    same!(zero.get("whole"), Value::Number(0.0));
331    same!(reader.find("Kinds", &Value::Number(999.0)).map_err(wrong)?, None, "a key nobody has");
332    Ok(Outcome::Passed)
333}

335fn filters_and_order(adapter: &dyn Adapter) -> Result<Outcome, String> {
336    let store = store(adapter, "filters", &[plain()])?;
337    three_plain(&mut *(store.space)())?;
338    let space = (store.space)();
339    let run = |text: &str| names(&*space, "Plain", &filtered(text));
340    same!(run("[seats] > 1")?, ["Beta", "Gama"]);
341    same!(run("[rate] > 2")?, ["Beta", "Gama"], "a whole number against a decimal");
342    same!(run("[rate] = 2.5")?, ["Beta"]);
343    same!(run("[active] = True")?, ["Alfa", "Gama"]);
344    same!(run("[name] Like 'B%'")?, ["Beta"]);
345    same!(run("[name] In ('Alfa', 'Gama')")?, ["Alfa", "Gama"]);
346    same!(run("[seats] Between(2, 3)")?, ["Beta", "Gama"]);
347    same!(run("[seats] >= 2 And [active] = False")?, ["Beta"]);
348    same!(run("[seats] = 1 Or [seats] = 3")?, ["Alfa", "Gama"]);
349    same!(run("Not [active]")?, ["Beta"]);
350    same!(run("[name] = 'Ninguém'")?, Vec::<String>::new());
351    let unknown = space.query("Plain", &filtered("[nope] = 1"));
352    ensure!(
353        matches!(unknown, Err(SpaceError::BadQuery(_))),
354        "a filter on a field nobody has gave {unknown:?}"
355    );
356    let unknown = space.query("Plain", &Query::all().sort("nope", true));
357    ensure!(matches!(unknown, Err(SpaceError::BadQuery(_))), "a sort on a field nobody has gave {unknown:?}");
358    same!(names(&*space, "Plain", &Query::all().sort("seats", false))?, ["Gama", "Beta", "Alfa"]);
359    same!(
360        names(&*space, "Plain", &Query::all().sort("active", true).sort("name", false))?,
361        ["Beta", "Gama", "Alfa"]
362    );
363    let mut space_w = (store.space)();
364    add(
365        &mut *space_w,
366        "Plain",
367        &[("name", "B'; DROP TABLE plain; --".into()), ("seats", Value::Number(9.0))],
368    )?;
369    space_w.commit().map_err(wrong)?;
370    same!(run("[name] = 'B''; DROP TABLE plain; --'")?, ["B'; DROP TABLE plain; --"], "data stays data");
371    same!(count(&*space, "Plain")?, 4);
372    Ok(Outcome::Passed)
373}

375fn pages_only_and_calculated(adapter: &dyn Adapter) -> Result<Outcome, String> {
376    let store = store(adapter, "pages", &[plain()])?;
377    three_plain(&mut *(store.space)())?;
378    let space = (store.space)();
379    let page = Query { limit: Some(1), offset: 1, ..Query::all() };
380    same!(names(&*space, "Plain", &page)?, ["Beta"]);
381    same!(names(&*space, "Plain", &Query { offset: 2, ..Query::all() })?, ["Gama"]);
382    same!(names(&*space, "Plain", &Query { limit: Some(2), ..Query::all() })?, ["Alfa", "Beta"]);
383    same!(names(&*space, "Plain", &Query { offset: 5, ..Query::all() })?, Vec::<String>::new());
384    let only = Query { only: Some(vec!["name".into()]), ..Query::all() };
385    let few = space.query("Plain", &only).map_err(wrong)?;
386    same!(few.len(), 3);
387    same!(few[0].get("name"), Value::Text("Alfa".into()));
388    ensure!(!matches!(few[0].get("id"), Value::Null), "the key comes with only: {:?}", few[0]);
389    same!(few[0].get("seats"), Value::Null, "a field left out by only");
390    let all = space.query("Plain", &Query::all()).map_err(wrong)?;
391    same!(all[2].get("twice"), Value::Number(6.0), "calculated");
392    same!(
393        names(&*space, "Plain", &filtered("[twice] > 2"))?,
394        ["Beta", "Gama"],
395        "filtered on a calculated field"
396    );
397    Ok(Outcome::Passed)
398}

400fn uncommitted_is_seen_and_rollback(adapter: &dyn Adapter) -> Result<Outcome, String> {
401    let store = store(adapter, "pending", &[plain()])?;
402    three_plain(&mut *(store.space)())?;
403    let mut space = (store.space)();
404    add(&mut *space, "Plain", &[("name", "Delta".into()), ("seats", Value::Number(4.0))])?;
405    same!(names(&*space, "Plain", &filtered("[seats] > 2"))?, ["Gama", "Delta"], "a space sees what waits");
406    same!(count(&*(store.space)(), "Plain")?, 3, "another space does not");
407    let alfa = space.query("Plain", &filtered("[name] = 'Alfa'")).map_err(wrong)?;
408    let key = alfa.first().ok_or("Alfa")?.get("id");
409    space.delete("Plain", &key).map_err(wrong)?;
410    let mut beta = space.query("Plain", &filtered("[name] = 'Beta'")).map_err(wrong)?.remove(0);
411    beta.set("name", "Beta 2");
412    space.save(beta).map_err(wrong)?;
413    same!(
414        names(&*space, "Plain", &Query::all())?,
415        ["Beta 2", "Gama", "Delta"],
416        "new, changed and deleted, waiting"
417    );
418    ensure!(space.is_modified(), "a space with changes says so");
419    space.rollback();
420    ensure!(!space.is_modified(), "after a rollback nothing waits");
421    same!(names(&*space, "Plain", &Query::all())?, ["Alfa", "Beta", "Gama"], "all thrown away");
422    same!(count(&*(store.space)(), "Plain")?, 3, "nothing reached the storage");
423    Ok(Outcome::Passed)
424}

426/// A commit with a new record and an update of a record another program deletes after the update
427/// was saved and before the commit, so the update finds nothing to change.
428fn half_good(store: &Store, space: &mut dyn ObjectSpace) -> Result<(), String> {
429    let gone = space.query("Plain", &filtered("[name] = 'Alfa'")).map_err(wrong)?;
430    let mut gone = gone.into_iter().next().ok_or("Alfa")?;
431    let key = gone.get("id");
432    add(space, "Plain", &[("name", "Nova".into())])?;
433    gone.set("seats", Value::Number(10.0));
434    space.save(gone).map_err(wrong)?;
435    let mut other = (store.again)();
436    other.delete("Plain", &key).map_err(wrong)?;
437    other.commit().map_err(wrong)
438}

440fn all_or_nothing(adapter: &dyn Adapter) -> Result<Outcome, String> {
441    let store = store(adapter, "allornothing", &[plain()])?;
442    three_plain(&mut *(store.space)())?;
443    let mut space = (store.space)();
444    half_good(&store, &mut *space)?;
445    let failed = space.commit();
446    ensure!(failed.is_err(), "a commit whose second change fails went through");
447    same!(
448        names(&*(store.space)(), "Plain", &Query::all())?,
449        ["Beta", "Gama"],
450        "nothing of the commit stayed"
451    );
452    space.rollback();
453    add(&mut *space, "Plain", &[("name", "Depois".into())])?;
454    space.commit().map_err(|e| format!("the storage does not work after a failed commit: {e}"))?;
455    same!(count(&*(store.space)(), "Plain")?, 3);
456    Ok(Outcome::Passed)
457}

459fn gone_record_conflicts(adapter: &dyn Adapter) -> Result<Outcome, String> {
460    let store = store(adapter, "gone", &[plain()])?;
461    three_plain(&mut *(store.space)())?;
462    let mut space = (store.space)();
463    let mut beta = space.query("Plain", &filtered("[name] = 'Beta'")).map_err(wrong)?.remove(0);
464    let key = beta.get("id");
465    beta.set("seats", Value::Number(9.0));
466    space.save(beta).map_err(wrong)?;
467    // Deleted after the update was saved: `save` decides between a new record and an update when
468    // it is called, so a record already gone by then would be made again.
469    let mut other = (store.again)();
470    other.delete("Plain", &key).map_err(wrong)?;
471    other.commit().map_err(wrong)?;
472    let error = space.commit();
473    ensure!(matches!(error, Err(SpaceError::Conflict(_))), "an update of a deleted record gave {error:?}");
474    same!(
475        names(&*(store.space)(), "Plain", &Query::all())?,
476        ["Alfa", "Gama"],
477        "the deleted record stays deleted"
478    );
479    Ok(Outcome::Passed)
480}

500fn state_machine(adapter: &dyn Adapter) -> Result<Outcome, String> {
501    let store = store(adapter, "states", &[order()])?;
502    let mut space = (store.space)();
503    let key = add(&mut *space, "Order", &[("name", "Um".into()), ("state", "o".into())])?;
504    space.commit().map_err(wrong)?;
505    let mut space = (store.space)();
506    let mut found = space.find("Order", &key).map_err(wrong)?.ok_or("the order")?;
507    found.set("state", "c");
508    space.save(found).map_err(wrong)?;
509    space.commit().map_err(|e| format!("a transition the machine has was refused: {e}"))?;
510    let mut space = (store.space)();
511    let mut found = space.find("Order", &key).map_err(wrong)?.ok_or("the order")?;
512    found.set("state", "o");
513    let refused = space.save(found).and_then(|()| space.commit());
514    ensure!(refused.is_err(), "a transition the machine does not have went through");
515    let kept = (store.space)().find("Order", &key).map_err(wrong)?.ok_or("the order")?;
516    same!(kept.get("state"), Value::Text("c".into()));
517    Ok(Outcome::Passed)
518}

520/// Rules do not run by themselves: forms, the web API and actions call `validate` before keeping,
521/// and a watcher on the commit can refuse it. What the adapter owes them is reading other records
522/// for a rule such as `unique`, and keeping nothing when a watcher refuses.
523fn rules_and_watchers(adapter: &dyn Adapter) -> Result<Outcome, String> {
524    let entity = order().rule(Rule::unique("name"));
525    let store = store(adapter, "rules", std::slice::from_ref(&entity))?;
526    let mut space = (store.space)();
527    add(&mut *space, "Order", &[("name", "Um".into()), ("state", "o".into())])?;
528    space.commit().map_err(wrong)?;
529    let words = RuleWords::default();
530    let mut space = (store.space)();
531    let mut empty = space.new_object("Order").map_err(wrong)?;
532    empty.set("state", "o");
533    same!(validate(&entity, &empty, CONTEXT_SAVE, &*space, &words).len(), 1, "a missing name");
534    let mut twin = space.new_object("Order").map_err(wrong)?;
535    twin.set("name", "Um");
536    twin.set("state", "o");
537    same!(validate(&entity, &twin, CONTEXT_SAVE, &*space, &words).len(), 1, "a name another record has");
538    twin.set("name", "Dois");
539    same!(validate(&entity, &twin, CONTEXT_SAVE, &*space, &words).len(), 0, "a good record");
540    space.on_event(Box::new(|event| match event {
541        SpaceEvent::Committing(_) => Err("não hoje".to_owned()),
542        _ => Ok(()),
543    }));
544    space.save(twin).map_err(wrong)?;
545    same!(space.commit(), Err(SpaceError::Refused("não hoje".into())), "a watcher refusing");
546    same!(count(&*(store.space)(), "Order")?, 1, "nothing of the refused commit stayed");
547    Ok(Outcome::Passed)
548}

574fn permission_denied(adapter: &dyn Adapter) -> Result<Outcome, String> {
575    let entities = with_security(plain());
576    let store = store(adapter, "denied", &entities)?;
577    three_plain(&mut *(store.space)())?;
578    let reader = Role::new("Leitura").policy(PermissionPolicy::ReadOnlyAll);
579    let hidden = Role::new("Sem custo").policy(PermissionPolicy::AllowAll).type_permission(
580        "Plain",
581        [Operation::Delete],
582        Access::Deny,
583    );
584    let directory = UserDirectory::new(store.space.clone(), Vec::new()).map_err(wrong)?;
585    let as_role = |role: Role| {
586        let identity = Identity::new("u", "u", vec![role]);
587        let context = SecurityContext::new(identity, directory.clone(), entities.clone(), RoleMerge::AnyRole);
588        SecuredObjectSpace::new((store.space)(), Rc::new(context))
589    };
590    let mut space = as_role(reader);
591    same!(count(&space, "Plain")?, 3, "a reader reads");
592    let tried = space.new_object("Plain").and_then(|object| space.save(object)).and_then(|()| space.commit());
593    ensure!(matches!(tried, Err(SpaceError::Denied(_))), "a reader writing gave {tried:?}");
594    let mut space = as_role(hidden);
595    let first = space.query("Plain", &Query::all()).map_err(wrong)?.remove(0).get("id");
596    let tried = space.delete("Plain", &first).and_then(|()| space.commit());
597    ensure!(matches!(tried, Err(SpaceError::Denied(_))), "a delete the role denies gave {tried:?}");
598    same!(count(&*(store.space)(), "Plain")?, 3, "nothing was written");
599    Ok(Outcome::Passed)
600}

602fn keys_from_two_programs(adapter: &dyn Adapter) -> Result<Outcome, String> {
603    let store = store(adapter, "keys", &[plain()])?;
604    three_plain(&mut *(store.space)())?;
605    // Two units of work of one program never get the same key.
606    let (mut a, mut b) = ((store.space)(), (store.space)());
607    let (x, y) =
608        (add(&mut *a, "Plain", &[("name", "X".into())])?, add(&mut *b, "Plain", &[("name", "Y".into())])?);
609    ensure!(x != y, "two spaces of one program got the key {x:?}");
610    a.commit().map_err(wrong)?;
611    b.commit().map_err(wrong)?;
612    let mut tidy = (store.space)();
613    tidy.delete("Plain", &x).map_err(wrong)?;
614    tidy.delete("Plain", &y).map_err(wrong)?;
615    tidy.commit().map_err(wrong)?;
616    let mut one = (store.space)();
617    let mut two = (store.again)();
618    // Each asks for the next key before the other commits.
619    let first = add(&mut *one, "Plain", &[("name", "Um".into())])?;
620    let second = add(&mut *two, "Plain", &[("name", "Dois".into())])?;
621    let also = add(&mut *two, "Plain", &[("name", "Tres".into())])?;
622    one.commit().map_err(wrong)?;
623    let committed = two.commit();
624    let kept = names(&*(store.space)(), "Plain", &Query::all())?;
625    match adapter.capabilities().keys {
626        KeyRace::Distinct => {
627            ensure!(first != second && first != also, "two programs got the same key {first:?}");
628            committed.map_err(wrong)?;
629            same!(kept.len(), 6);
630            Ok(Outcome::Passed)
631        }
632        KeyRace::SecondCommitFails => {
633            if first != second {
634                committed.map_err(wrong)?;
635                same!(kept.len(), 6);
636                return Ok(Outcome::Passed);
637            }
638            ensure!(committed.is_err(), "two records with the key {first:?}: the second commit went through");
639            same!(
640                kept,
641                ["Alfa", "Beta", "Gama", "Um"],
642                "nothing of the second commit stayed, and the first is whole"
643            );
644            Ok(Outcome::Limited(
645                "two programs creating records at once: the second commit fails, whole (#241)",
646            ))
647        }
648    }
649}

651fn update_from_two_programs(adapter: &dyn Adapter) -> Result<Outcome, String> {
652    let store = store(adapter, "updates", &[plain()])?;
653    three_plain(&mut *(store.space)())?;
654    let mut one = (store.space)();
655    let mut two = (store.again)();
656    let mut mine = one.query("Plain", &filtered("[name] = 'Beta'")).map_err(wrong)?.remove(0);
657    let mut theirs = two.query("Plain", &filtered("[name] = 'Beta'")).map_err(wrong)?.remove(0);
658    let key = mine.get("id");
659    theirs.set("seats", Value::Number(20.0));
660    two.save(theirs).map_err(wrong)?;
661    two.commit().map_err(wrong)?;
662    mine.set("seats", Value::Number(10.0));
663    one.save(mine).map_err(wrong)?;
664    let committed = one.commit();
665    let kept = (store.space)().find("Plain", &key).map_err(wrong)?.ok_or("Beta")?.get("seats");
666    if adapter.capabilities().lost_updates_detected {
667        ensure!(matches!(committed, Err(SpaceError::Conflict(_))), "a lost update gave {committed:?}");
668        same!(kept, Value::Number(20.0));
669        return Ok(Outcome::Passed);
670    }
671    committed.map_err(wrong)?;
672    same!(kept, Value::Number(10.0), "the last commit wins");
673    Ok(Outcome::Limited("an update over another program's: the last commit wins (#241)"))
674}

719fn decimals(adapter: &dyn Adapter) -> Result<Outcome, String> {
720    let entity = EntityInfo::new("Rate")
721        .field(FieldInfo::new("id", FieldKind::Integer).key())
722        .field(FieldInfo::new("two", FieldKind::Decimal(2)))
723        .field(FieldInfo::new("six", FieldKind::Decimal(6)));
724    let store = store(adapter, "decimals", &[entity])?;
725    let exact =
726        [(0.1, "two"), (5.49, "two"), (16.47, "two"), (123_456_789.123_456, "six"), (0.000_001, "six")];
727    let mut space = (store.space)();
728    for (number, field) in exact {
729        add(&mut *space, "Rate", &[(field, Value::Number(number))])?;
730    }
731    add(&mut *space, "Rate", &[("two", Value::Number(0.1 + 0.2))])?;
732    space.commit().map_err(wrong)?;
733    let reader = (store.space)();
734    let found = reader.query("Rate", &Query::all()).map_err(wrong)?;
735    for (at, (number, field)) in exact.iter().enumerate() {
736        same!(found[at].get(field), Value::Number(*number), format!("{number}"));
737    }
738    same!(found[5].get("two"), Value::Number(0.3), "rounded to its places");
739    let hits = |text: &str| reader.query("Rate", &filtered(text)).map(|found| found.len()).map_err(wrong);
740    same!(hits("[two] = 16.47")?, 1);
741    same!(hits("[two] > 5.49")?, 1);
742    same!(hits("[six] = 0.000001")?, 1);
743    let mut space = (store.space)();
744    add(&mut *space, "Rate", &[("six", Value::Number(1_234_567_890.123_456))])?;
745    let refused = space.commit();
746    ensure!(
747        matches!(refused, Err(SpaceError::Refused(_))),
748        "more digits than a number keeps gave {refused:?}"
749    );
750    Ok(Outcome::Passed)
751}
