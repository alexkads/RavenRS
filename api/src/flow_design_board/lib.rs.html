1//! A design board, made only with `FlowCanvas`, `FlowToolbox` and `FlowInspector`: frames
2//! hold the shapes of a screen, shapes are placed freely, resized, turned and lined up with
3//! the guides, and arrows between frames say where a tap leads, as in a clickable mock-up.
4//!
5//! It validates the component as a board of this kind: everything here is the library's,
6//! and what it could not do is written in the README.
7

13/// The kinds of piece, with the word the inspector shows for each.
14pub const KINDS: [(&str, &str); 3] = [("shape", "Forma"), ("text", "Texto"), ("screen", "Tela")];
15
16fn piece(id: &str, title: &str, shape: FlowShape, w: f32, h: f32) -> FlowNode {
17    FlowNode::new(id, "shape", title, 0.0, 0.0).shape(shape).size(w, h)
18}
19
20fn at(node: FlowNode, x: f32, y: f32, frame: &str) -> FlowNode {
21    FlowNode { x, y, ..node }.parent(frame)
22}

24/// Two screens of a sign-in, their pieces inside each frame, and where the button leads.
25pub fn sample() -> FlowGraph {
26    let login = Rect::new(0.0, 0.0, 360.0, 600.0);
27    let home = Rect::new(480.0, 0.0, 360.0, 600.0);
28    FlowGraph::new()
29        .node(FlowNode::new_group("login", "Entrar", login))
30        .node(at(piece("logo", "Logo", FlowShape::Ellipse, 96.0, 96.0), 132.0, 72.0, "login"))
31        .node(at(piece("email", "E-mail", FlowShape::Rectangle, 264.0, 48.0), 48.0, 240.0, "login"))
32        .node(at(piece("password", "Senha", FlowShape::Rectangle, 264.0, 48.0), 48.0, 312.0, "login"))
33        .node(
34            at(piece("enter", "Entrar", FlowShape::RoundedRectangle, 264.0, 48.0), 48.0, 408.0, "login")
35                .port(FlowPort::output("tap").side(PortSide::Right)),
36        )
37        .node(
38            FlowNode::new("heading", "text", "Fluxo de entrada", 0.0, -96.0)
39                .shape(FlowShape::Text)
40                .size(480.0, 48.0),
41        )
42        .node(at(
43            FlowNode::new("photo", "image", "Foto do produto", 0.0, 0.0).size(312.0, 160.0),
44            504.0,
45            288.0,
46            "home",
47        ))
48        // A note beside the screen, outside its frame.
49        .node(
50            FlowNode::new("contrast", "text", "Revisar o contraste do botão", -240.0, 408.0)
51                .shape(FlowShape::Note)
52                .size(192.0, 96.0),
53        )
54        .node(FlowNode::new_group("home", "Início", home).port(FlowPort::input("in").side(PortSide::Left)))
55        .node(at(piece("header", "Cabeçalho", FlowShape::Rectangle, 312.0, 64.0), 504.0, 48.0, "home"))
56        .node(at(piece("card", "Cartão do pedido", FlowShape::Card, 312.0, 120.0), 504.0, 144.0, "home"))
57        .node(at(
58            piece("badge", "Novo", FlowShape::Ellipse, 64.0, 32.0).rotation(-12.0),
59            744.0,
60            120.0,
61            "home",
62        ))
63        .edge(
64            FlowEdge::new("tap", FlowEndpoint::new("enter", "tap"), FlowEndpoint::new("home", "in"))
65                .label("Toque")
66                .shape(EdgeShape::Curve),
67        )
68}

96/// A picture for the image pieces of the sample: a soft gradient, made here so that the example
97/// needs no file.
98pub fn sample_picture() -> Option<ImageData> {
99    ImageData::from_fn(240, 140, |x, y| {
100        let (u, v) = (x as f32 / 240.0, y as f32 / 140.0);
101        Color::rgba(0.25 + 0.5 * u, 0.4 + 0.3 * v, 0.85 - 0.3 * u, 1.0)
102    })
103}

105/// The board: pieces on the left, the board in the middle, the selected piece's size, place
106/// and words on the right.
107pub struct Board {
108    /// Where the diagrams are kept between runs.
109    storage: Rc<dyn Storage>,
110}
111
112impl std::fmt::Debug for Board {
113    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
114        f.debug_struct("Board").finish_non_exhaustive()
115    }
116}
117
118impl View for Board {
119    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
120        let graph = cx.use_state(sample);
121        let selection = cx.use_state(FlowSelection::default);
122        let history = cx.use_state(FlowHistory::new);
123        let arrivals = cx.use_state(|| None::<FlowNode>);
124        let view = cx.use_state(FlowViewport::fit);
125        let canvas = FlowCanvas::new(graph.binding())
126            .view(view.binding())
127            .ports_on_hover(true)
128            .label("Quadro")
129            .selection(selection.binding())
130            .history(history.binding())
131            .arrivals(arrivals.binding())
132            // A board is drawn by hand: no arrows to make the next piece, a fine grid.
133            .quick_create(false)
134            .grid(8.0)
135            .kind("shape", FlowKind::new(IconName::Shapes, "Forma"))
136            .kind("text", FlowKind::new(IconName::Type, "Texto"))
137            .kind("image", FlowKind::new(IconName::Image, "Imagem"))
138            // An image piece shows its picture: here one drawn for the sample, in an app the file.
139            .content(|node| {
140                if node.kind == "image" {
141                    sample_picture().map(|p| Box::new(Image::new(p)) as AnyView)
142                } else {
143                    None
144                }
145            })
146            .on_export(ravenrs_flow_export::saver(cx.file_picker(), "Quadro"));
147        let mut inspector =
148            FlowInspector::new(graph.binding(), selection.binding()).history(history.binding());
149        for (kind, caption) in KINDS {
150            inspector = inspector.kind(kind, caption);
151        }
152        cx.child(
153            &HStack::new((
154                VStack::new((
155                    FlowLibrary::new(graph.binding())
156                        .title("Quadros")
157                        .key("boards")
158                        .history(history.binding())
159                        .selection(selection.binding())
160                        .view(view.binding())
161                        .storage(self.storage.clone()),
162                    toolbox().title("Peças").arrivals(arrivals.binding()),
163                    FlowLayers::new(graph.binding(), selection.binding()).history(history.binding()),
164                ))
165                .spacing(0.0)
166                .align(Align::Stretch),
167                canvas,
168                inspector,
169            ))
170            .align(Align::Stretch)
171            .grow(1.0),
172        )
173    }
174}

176/// The whole app.
177pub fn app() -> Localized<Board> {
178    app_kept_in(Rc::new(MemoryStorage::default()))
179}

181/// The whole app, keeping its diagrams in `storage`.
182pub fn app_kept_in(storage: Rc<dyn Storage>) -> Localized<Board> {
183    Localized::new(Labels::pt_br(), Board { storage })
184}

186#[cfg(target_arch = "wasm32")]
187mod web_entry {
188    use std::cell::RefCell;
189
190    use wasm_bindgen::prelude::*;
191
192    thread_local! {
193        static APP: RefCell<Option<ravenrs::web::WebApp>> = const { RefCell::new(None) };
194    }
195
196    /// Runs when the module is instantiated: the diagrams are kept in the browser's storage.
197    #[wasm_bindgen(start)]
198    pub fn start() {
199        console_error_panic_hook::set_once();
200        wasm_bindgen_futures::spawn_local(async {
201            let storage = match ravenrs::web::IndexedDbStorage::open("ravenrs-flow-design-board").await {
202                Ok(storage) => storage,
203                Err(e) => {
204                    web_sys::console::error_1(&e.to_string().into());
205                    return;
206                }
207            };
208            let app = super::app_kept_in(std::rc::Rc::new(storage));
209            match ravenrs::web::start(app, ravenrs::web::WebOptions::new("raven-canvas")).await {
210                Ok(app) => APP.with(|a| *a.borrow_mut() = Some(app)),
211                Err(e) => web_sys::console::error_1(&e.to_string().into()),
212            }
213        });
214    }
