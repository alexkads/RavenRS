21#[derive(Clone)]
22enum Stage {
23    SignIn,
24    NewPassword,
25    Loading,
26    Ready(Rc<AnyView>),
27    Broken(String),
28}

30struct Inner {
31    title: String,
32    words: Words,
33    client: RemoteClient,
34    entities: Vec<EntityInfo>,
35    make: Box<dyn Fn(&RemoteDatabase) -> App>,
36    stage: State<Stage>,
37    name: State<String>,
38    password: State<String>,
39    new_password: State<String>,
40    error: State<Option<String>>,
41    /// Asking the server who this is, so pressing again does not ask twice.
42    asking: Cell<bool>,
43    /// The theme at the door, for the next person: the one before may have chosen another.
44    door_theme: Cell<Option<Theme>>,
45    /// The records of whoever is inside.
46    inside: RefCell<Option<RemoteDatabase>>,
47}

49/// An application whose records are on a server: it asks who is there, reads what they can see, and
50/// opens what `make` puts together over those records.
51#[derive(Clone)]
52pub struct RemoteApp(Rc<Inner>);

54impl std::fmt::Debug for RemoteApp {
55    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
56        f.debug_struct("RemoteApp").field("title", &self.0.title).finish_non_exhaustive()
57    }
58}

60impl RemoteApp {
61    /// `make` gets the records and gives the application, with those records as its storage.
62    pub fn view(
63        title: impl Into<String>,
64        language: Language,
65        client: RemoteClient,
66        entities: Vec<EntityInfo>,
67        make: impl Fn(&RemoteDatabase) -> App + 'static,
68    ) -> Result<impl View, ModelError> {
69        let texts = Texts::new(language);
70        let mine = Self(Rc::new(Inner {
71            title: title.into(),
72            words: Words::from_texts(&texts),
73            client,
74            entities,
75            make: Box::new(make),
76            stage: State::new(Stage::SignIn),
77            name: State::new(String::new()),
78            password: State::new(String::new()),
79            new_password: State::new(String::new()),
80            error: State::new(None),
81            asking: Cell::new(false),
82            door_theme: Cell::new(None),
83            inside: RefCell::new(None),
84        }));
85        Ok(Localized::new(Labels::from_texts(&texts), mine))
86    }
87
88    fn sign_in(&self) {
89        let inner = &self.0;
90        if inner.asking.replace(true) {
91            return;
92        }
93        let (name, password) = (inner.name.peek().clone(), inner.password.peek().clone());
94        let mine = self.clone();
95        inner.client.sign_in(&name, &password, move |result| {
96            mine.0.asking.set(false);
97            match result {
98                Ok(signed) if signed.must_change_password => {
99                    mine.0.error.set(None);
100                    mine.0.password.set(String::new());
101                    mine.0.stage.set(Stage::NewPassword);
102                }
103                Ok(_) => mine.open(),
104                Err(SignInError::Rejected) => mine.0.error.set(Some(mine.0.words.sign_in_failed.clone())),
105                Err(SignInError::Other(why)) => mine.0.error.set(Some(why)),
106            }
107        });
108    }
109
110    fn change_password(&self) {
111        let inner = &self.0;
112        let (current, new) = (inner.password.peek().clone(), inner.new_password.peek().clone());
113        let mine = self.clone();
114        inner.client.change_password(&current, &new, move |result| match result {
115            Ok(()) => mine.open(),
116            Err(why) => mine.0.error.set(Some(why)),
117        });
118    }
119
120    fn open(&self) {
121        let inner = &self.0;
122        inner.error.set(None);
123        // What was typed has no reason to stay around.
124        inner.password.set(String::new());
125        inner.new_password.set(String::new());
126        inner.stage.set(Stage::Loading);
127        let database = match RemoteDatabase::new(inner.client.clone(), inner.entities.clone()) {
128            Ok(database) => database,
129            Err(error) => {
130                inner.stage.set(Stage::Broken(error.to_string()));
131                return;
132            }
133        };
134        database.denied_message(inner.words.access_denied.clone());
135        let (mine, loaded) = (self.clone(), database.clone());
136        database.load(move |result| {
137            if let Err(why) = result {
138                mine.0.stage.set(Stage::Broken(why));
139                return;
140            }
141            let settings = RemoteSettings::new(mine.0.client.clone());
142            let (keeping, reading) = (settings.clone(), mine.clone());
143            settings.load(vec!["theme".to_owned()], Box::new(move || reading.enter(&loaded, keeping)));
144        });
145    }
146
147    fn enter(&self, loaded: &RemoteDatabase, settings: RemoteSettings) {
148        *self.0.inside.borrow_mut() = Some(loaded.clone());
149        let (client, back) = (self.0.client.clone(), self.clone());
150        let watching = loaded.clone();
151        let app = (self.0.make)(loaded).settings(settings).signed_in_remotely(
152            self.0.name.peek().clone(),
153            move || {
154                client.sign_out();
155                if let Some(records) = back.0.inside.borrow_mut().take() {
156                    records.forget();
157                }
158                back.0.name.set(String::new());
159                back.0.stage.set(Stage::SignIn);
160            },
161        );
162        let app = app.with_session(move |session: &Session| {
163            let changed = session.clone();
164            watching.on_changed(move || changed.records_changed());
165            let told = session.clone();
166            watching.on_problem(move |why| told.show_message(Message::new(why, MessageKind::Error)));
167        });
168        match app.build() {
169            Ok(view) => self.0.stage.set(Stage::Ready(Rc::new(view))),
170            Err(error) => self.0.stage.set(Stage::Broken(error.to_string())),
171        }
172    }
173
174    fn form(&self, content: impl View + 'static) -> impl View {
175        centered(Card::new(content).padding(24.0).width(FORM_WIDTH))
176    }
177}
178
179fn centered(content: impl View + 'static) -> impl View {
180    HStack::new((content,)).justify(Justify::Center).align(Align::Center).grow(1.0)
181}

183fn binding(state: &State<String>) -> Binding<String> {
184    let (read, write) = (state.clone(), state.clone());
185    Binding::new(move || read.get(), move |value| write.set(value))
186}

188impl View for RemoteApp {
189    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
190        let inner = &self.0;
191        let at_door = matches!(inner.stage.get(), Stage::SignIn);
192        match inner.door_theme.get() {
193            None => inner.door_theme.set(Some(*cx.theme())),
194            Some(door) if at_door && *cx.theme() != door => {
195                cx.theme_state().set(door);
196            }
197            Some(_) => {}
198        }
199        let words = &inner.words;
200        let title = || Text::new(inner.title.clone()).style(TextStyle::Title);
201        let error = inner.error.get().map(|why| Alert::message(why).tone(Tone::Danger));
202        match inner.stage.get() {
203            Stage::Ready(view) => view.build(cx),
204            Stage::Loading => cx.child(&centered(Spinner::new())),
205            Stage::Broken(why) => cx.child(
206                &self.form(
207                    VStack::new((title(), Alert::message(why).tone(Tone::Danger)))
208                        .spacing(12.0)
209                        .align(Align::Stretch),
210                ),
211            ),
212            Stage::SignIn => {
213                let (by_key, by_button) = (self.clone(), self.clone());
214                cx.child(
215                    &self.form(
216                        VStack::new((
217                            title(),
218                            error,
219                            FormField::new(
220                                words.user_name.clone(),
221                                TextField::new(binding(&inner.name)).autofocus(),
222                            ),
223                            FormField::new(
224                                words.password.clone(),
225                                TextField::new(binding(&inner.password))
226                                    .secure()
227                                    .on_submit(move |_| by_key.sign_in()),
228                            ),
229                            Button::new(words.sign_in.clone())
230                                .full_width()
231                                .on_click(move || by_button.sign_in()),
232                        ))
233                        .spacing(12.0)
234                        .align(Align::Stretch),
235                    ),
236                )
237            }
238            Stage::NewPassword => {
239                let by_button = self.clone();
240                cx.child(
241                    &self.form(
242                        VStack::new((
243                            title(),
244                            Text::new(words.change_password.clone()).foreground(Semantic::InkMuted),
245                            error,
246                            FormField::new(
247                                words.current_password.clone(),
248                                TextField::new(binding(&inner.password)).secure(),
249                            ),
250                            FormField::new(
251                                words.new_password.clone(),
252                                TextField::new(binding(&inner.new_password)).secure(),
253                            ),
254                            Button::new(words.change_password.clone())
255                                .full_width()
256                                .on_click(move || by_button.change_password()),
257                        ))
258                        .spacing(12.0)
259                        .align(Align::Stretch),
260                    ),
261                )
262            }
263        }
264    }
