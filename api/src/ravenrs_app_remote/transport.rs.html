1//! How a request gets to the server.
2
3/// A request on its way.
4#[derive(Clone, Debug)]
5#[non_exhaustive]
6pub struct RemoteRequest {
7    /// `GET`, `POST` and so on.
8    pub method: String,
9    /// The whole address.
10    pub url: String,
11    /// The token to present.
12    pub bearer: Option<String>,
13    /// The JSON to send.
14    pub body: Option<String>,
15}
16
17/// What came back.
18#[derive(Clone, Debug)]
19#[non_exhaustive]
20pub struct RemoteResponse {
21    /// The status.
22    pub status: u16,
23    /// The body.
24    pub body: String,
25}
26
27impl RemoteResponse {
28    /// With this status and body.
29    pub fn new(status: u16, body: impl Into<String>) -> Self {
30        Self { status, body: body.into() }
31    }
32}
33
34/// Carries requests. Calling `done` before returning makes commits wait for the server; calling it
35/// later makes them go on at once, as they have to in a browser. Failing means nothing came back.
36pub trait Transport {
37    /// Sends it, and calls `done` with the answer.
38    fn send(&self, request: RemoteRequest, done: Box<dyn FnOnce(Result<RemoteResponse, String>)>);
39}
40
41/// HTTP: on the desktop it waits for the answer, and in the browser it does not.
42#[derive(Clone, Copy, Debug, Default)]
43pub struct HttpTransport;
44
45#[cfg(not(target_arch = "wasm32"))]
46impl Transport for HttpTransport {
47    fn send(&self, request: RemoteRequest, done: Box<dyn FnOnce(Result<RemoteResponse, String>)>) {
48        let agent = ureq::Agent::config_builder().http_status_as_error(false).build().new_agent();
49        let mut built = ureq::http::Request::builder().method(request.method.as_str()).uri(&request.url);
50        if let Some(token) = &request.bearer {
51            built = built.header("Authorization", format!("Bearer {token}"));
52        }
53        built = built.header("Content-Type", "application/json");
54        let answered = built
55            .body(request.body.unwrap_or_default())
56            .map_err(|error| error.to_string())
57            .and_then(|asked| agent.run(asked).map_err(|error| error.to_string()))
58            .and_then(|mut response| {
59                let status = response.status().as_u16();
60                let body = response.body_mut().read_to_string().map_err(|error| error.to_string())?;
61                Ok(RemoteResponse::new(status, body))
62            });
63        done(answered);
64    }
65}

67#[cfg(target_arch = "wasm32")]
68impl Transport for HttpTransport {
69    fn send(&self, request: RemoteRequest, done: Box<dyn FnOnce(Result<RemoteResponse, String>)>) {
70        use wasm_bindgen::{JsCast, JsValue};
71        use wasm_bindgen_futures::JsFuture;
72
73        async fn fetch(request: RemoteRequest) -> Result<RemoteResponse, JsValue> {
74            let init = web_sys::RequestInit::new();
75            init.set_method(&request.method);
76            if let Some(body) = &request.body {
77                init.set_body(&JsValue::from_str(body));
78            }
79            let asked = web_sys::Request::new_with_str_and_init(&request.url, &init)?;
80            asked.headers().set("Content-Type", "application/json")?;
81            if let Some(token) = &request.bearer {
82                asked.headers().set("Authorization", &format!("Bearer {token}"))?;
83            }
84            let window = web_sys::window().ok_or_else(|| JsValue::from_str("no window"))?;
85            let response: web_sys::Response =
86                JsFuture::from(window.fetch_with_request(&asked)).await?.dyn_into()?;
87            let body = JsFuture::from(response.text()?).await?.as_string().unwrap_or_default();
88            Ok(RemoteResponse::new(response.status(), body))
89        }
90
91        wasm_bindgen_futures::spawn_local(async move {
92            done(fetch(request).await.map_err(|error| format!("{error:?}")));
93        });
94    }
95}
