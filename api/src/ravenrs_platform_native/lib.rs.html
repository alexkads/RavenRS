1//! Desktop backend for RavenRS: one winit window per app, rendered with wgpu
2//! (Metal on macOS, DX12/Vulkan on Windows, Vulkan/GL on Linux).
3//!
4//! Frames are produced on demand: the event loop sleeps until input arrives or
5//! reactive state changes, so an idle app uses no CPU or GPU time.
6//!
7//! Screen readers see the UI through AccessKit: the accessibility tree is
8//! built only while one is running.
9

123/// Failure to start or run the desktop app.
124#[derive(Debug)]
125pub enum PlatformError {
126    /// The event loop could not be created or exited with an error.
127    EventLoop(String),
128    /// The window could not be created.
129    Window(String),
130    /// GPU setup or surface configuration failed.
131    Gpu(GpuError),
132}
133
134impl fmt::Display for PlatformError {
135    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
136        match self {
137            Self::EventLoop(e) => write!(f, "event loop error: {e}"),
138            Self::Window(e) => write!(f, "window could not be created: {e}"),
139            Self::Gpu(e) => write!(f, "{e}"),
140        }
141    }
142}
143
144impl std::error::Error for PlatformError {}
145
146impl From<GpuError> for PlatformError {
147    fn from(e: GpuError) -> Self {
148        Self::Gpu(e)
149    }

157/// Who the app is to the system, for finding its icon and its menu entry.
158///
159/// On Linux the id is what ties the window to the app's `.desktop` file. Wayland
160/// has no way for a window to carry an icon: GNOME shows the one in the
161/// `.desktop` file whose name, without the extension, is the window's id. On
162/// X11 the id must match the file's `StartupWMClass`.
163///
164/// ```
165/// use ravenrs_platform_native::AppIdentity;
166///
167/// // For a file named `notes.desktop`.
168/// let identity = AppIdentity::new("notes");
169/// ```
170#[derive(Clone, Debug)]
171pub struct AppIdentity {
172    id: String,
173    icon: Option<ImageData>,
174    labels: Option<Labels>,
175    diagnostics: Option<Diagnostics>,
176    new_issue_url: Option<String>,
177}

181    fn anonymous(icon: Option<ImageData>) -> Self {
182        Self { id: String::new(), icon, labels: None, diagnostics: None, new_issue_url: None }
183    }
184
185    /// The app known as `id`, which is the name of its `.desktop` file on Linux.
186    pub fn new(id: impl Into<String>) -> Self {
187        Self { id: id.into(), icon: None, labels: None, diagnostics: None, new_issue_url: None }
188    }
189
190    /// Sets the words of the whole UI, including the diagnostics panel.
191    pub fn labels(mut self, labels: Labels) -> Self {

194    }

196    /// Shares `diagnostics` with the app, for a button of its own that opens the

237fn run_window(
238    root: impl View + 'static,
239    options: WindowOptions,
240    identity: AppIdentity,
241) -> Result<(), PlatformError> {
242    let AppIdentity { id, icon, labels, diagnostics, new_issue_url } = identity;
243    let app_id = (!id.is_empty()).then_some(id);
244    let event_loop = EventLoop::<UserEvent>::with_user_event()
245        .build()
246        .map_err(|e| PlatformError::EventLoop(e.to_string()))?;
247    event_loop.set_control_flow(ControlFlow::Wait);
248    let diagnostics = diagnostics.unwrap_or_default();
249    diagnostics.set_open(std::env::var_os(DIAGNOSTICS_VARIABLE).is_some());
250    if let Some(url) = new_issue_url {
251        diagnostics.set_bug_reporter(url, |url| open_url(url).map_err(|e| e.to_string()));
252    }
253    let root = DiagnosticsHost::new(root, diagnostics.clone());
254    let mut ui = match labels {
255        Some(labels) => Ui::new(Localized::new(labels, root), options.theme),
256        None => Ui::new(root, options.theme),
257    };
258    let wake = event_loop.create_proxy();
259    ui.set_remote_waker(move || {
260        // It only fails once the loop is gone, when there is nothing left to wake.
261        let _ = wake.send_event(UserEvent::Wake);
262    });
263    match arboard::Clipboard::new() {
264        Ok(clipboard) => ui.set_clipboard(SystemClipboard(std::cell::RefCell::new(clipboard))),
265        Err(e) => tracing::warn!(error = %e, "system clipboard unavailable; using an in-app clipboard"),
266    }
267    ui.set_sound(sound::SystemSound::default());
268    let mut app = App {
269        display: event_loop.owned_display_handle(),
270        diagnostics,
271        proxy: event_loop.create_proxy(),
272        access: None,
273        ui,
274        options,
275        icon,
276        app_id,
277        window: None,
278        pointer: None,
279        modifiers: Modifiers::default(),
280        last_click: None,
281        started: std::time::Instant::now(),
282        error: None,
283    };
284    event_loop.run_app(&mut app).map_err(|e| PlatformError::EventLoop(e.to_string()))?;
285    match app.error {
286        Some(e) => Err(e),
287        None => Ok(()),
288    }
289}

356impl App {
357    fn create_window(&mut self, event_loop: &ActiveEventLoop) -> Result<WindowState, PlatformError> {
358        // Hidden until the accessibility adapter is attached, as AccessKit requires.
359        let mut attributes = Window::default_attributes()
360            .with_visible(false)
361            .with_title(self.options.title.clone())
362            .with_inner_size(LogicalSize::new(self.options.size.width, self.options.size.height));
363        if let Some(min) = self.options.min_size {
364            attributes = attributes.with_min_inner_size(LogicalSize::new(min.width, min.height));
365        }
366        if let Some(id) = &self.app_id {
367            attributes = with_app_id(attributes, id);
368        }
369        if let Some(image) = &self.icon {
370            match Icon::from_rgba(image.pixels().to_vec(), image.width(), image.height()) {
371                Ok(icon) => attributes = attributes.with_window_icon(Some(icon)),
372                Err(e) => tracing::warn!(error = %e, "could not use the window icon"),
373            }
374        }
375        let window =
376            Arc::new(event_loop.create_window(attributes).map_err(|e| PlatformError::Window(e.to_string()))?);
377        let adapter =
378            accesskit_winit::Adapter::with_event_loop_proxy(event_loop, &window, self.proxy.clone());
379        self.access = Some(Access { adapter, active: false, last: None });
380        window.set_visible(true);
381        self.init_gpu(window)
382    }
383
384    /// Sends the accessibility tree to the screen reader if one is listening
385    /// and the tree changed.
386    fn update_access(&mut self) {
387        let Some(access) = &mut self.access else { return };
388        if !access.active {
389            return;
390        }
391        let tree = self.ui.access_tree();
392        if access.last.as_ref() == Some(&tree) {
393            return;
394        }
395        let scale = f64::from(self.ui.scale());
396        access.adapter.update_if_active(|| access::tree_update(&tree, scale));
397        access.last = Some(tree);
398    }
399
400    fn init_gpu(&mut self, window: Arc<Window>) -> Result<WindowState, PlatformError> {
401        let instance = diagnostics::new_instance(Some(&self.display), false);
402        let surface =
403            instance.create_surface(window.clone()).map_err(|e| GpuError::Surface(e.to_string()))?;
404        let gpu = pollster::block_on(GpuContext::new(&instance, Some(&surface)))?;
405        let caps = surface.get_capabilities(&gpu.adapter);
406        let format = preferred_surface_format(&caps.formats)
407            .ok_or_else(|| GpuError::Surface("the surface reports no supported formats".into()))?;
408        let size = window.inner_size();
409        let config = wgpu::SurfaceConfiguration {
410            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
411            format,
412            width: size.width.max(1),
413            height: size.height.max(1),
414            present_mode: wgpu::PresentMode::AutoVsync,
415            desired_maximum_frame_latency: 2,
416            alpha_mode: caps.alpha_modes.first().copied().unwrap_or(wgpu::CompositeAlphaMode::Auto),
417            view_formats: vec![],
418            color_space: wgpu::SurfaceColorSpace::Auto,
419        };
420        surface.configure(&gpu.device, &config);
421        let described = surface_report(&caps, format, config.present_mode);
422        let (display, adapter) = (self.display.clone(), gpu.adapter.clone());
423        self.diagnostics.set_report_source(move || {
424            diagnostics::gather(Some(&display), &adapter, Some(described.clone()))
425        });
426        let device_lost = Arc::new(AtomicBool::new(false));
427        let (lost, lost_window) = (device_lost.clone(), window.clone());
428        gpu.device.set_device_lost_callback(move |reason, message| {
429            tracing::error!(?reason, message, "GPU device lost; recreating it");
430            lost.store(true, Ordering::SeqCst);
431            lost_window.request_redraw();
432        });
433        let renderer = Renderer::new(&gpu.device, &gpu.queue, format);
434
435        let redraw = window.clone();
436        self.ui.set_waker(move || redraw.request_redraw());
437        self.update_viewport(&window, size);
438        window.request_redraw();
439        tracing::info!(?format, width = size.width, height = size.height, "window ready");
440        Ok(WindowState { surface, config, gpu, renderer, device_lost, cursor: CursorIcon::Default, window })
441    }
442
443    fn update_viewport(&mut self, window: &Window, size: PhysicalSize<u32>) {
444        let scale = window.scale_factor() as f32;
445        let logical = Size::new(size.width as f32 / scale, size.height as f32 / scale);
446        self.ui.set_viewport(logical, scale);
447    }
448
449    fn dispatch(&mut self, event: Event) {
450        self.ui.set_time(self.started.elapsed().as_secs_f64());
451        let redraw = self.ui.handle_event(&event);
452        if let Some(state) = &mut self.window {
453            let cursor = self.ui.cursor();
454            if cursor != state.cursor {
455                state.window.set_cursor(match cursor {
456                    CursorIcon::Default => winit::window::CursorIcon::Default,
457                    CursorIcon::Pointer => winit::window::CursorIcon::Pointer,
458                    CursorIcon::Text => winit::window::CursorIcon::Text,
459                    CursorIcon::ColResize => winit::window::CursorIcon::ColResize,
460                    CursorIcon::RowResize => winit::window::CursorIcon::RowResize,
461                    _ => winit::window::CursorIcon::Default,
462                });
463                state.cursor = cursor;
464            }
465            if redraw {
466                state.window.request_redraw();
467            }
468        }
469    }
470
471    fn redraw(&mut self) {
472        let lost = self.window.as_ref().is_some_and(|s| s.device_lost.load(Ordering::SeqCst));
473        if lost && let Some(old) = self.window.take() {
474            let window = old.window.clone();
475            drop(old);
476            match self.init_gpu(window) {
477                Ok(state) => self.window = Some(state),
478                Err(e) => tracing::error!(error = %e, "GPU recovery failed"),
479            }
480        }
481        self.ui.set_time(self.started.elapsed().as_secs_f64());
482        let Some(state) = &mut self.window else { return };
483        let scale = self.ui.scale();
484        let started = std::time::Instant::now();
485        let scene = self.ui.update();
486        let updated = started.elapsed();
487        let frame = match state.surface.get_current_texture() {
488            wgpu::CurrentSurfaceTexture::Success(frame) => frame,
489            wgpu::CurrentSurfaceTexture::Suboptimal(frame) => {
490                state.surface.configure(&state.gpu.device, &state.config);
491                frame
492            }
493            wgpu::CurrentSurfaceTexture::Outdated | wgpu::CurrentSurfaceTexture::Lost => {
494                state.surface.configure(&state.gpu.device, &state.config);
495                state.window.request_redraw();
496                return;
497            }
498            wgpu::CurrentSurfaceTexture::Timeout | wgpu::CurrentSurfaceTexture::Occluded => return,
499            other => {
500                tracing::error!(?other, "could not acquire a frame");
501                return;
502            }
503        };
504        let acquired = started.elapsed();
505        let view = frame.texture.create_view(&wgpu::TextureViewDescriptor::default());
506        let stats = state.renderer.render(scene, &view, (state.config.width, state.config.height), scale);
507        let rendered = started.elapsed();
508        state.window.pre_present_notify();
509        state.gpu.queue.present(frame);
510        tracing::debug!(
511            update = ?updated,
512            acquire = ?acquired - updated,
513            render = ?rendered - acquired,
514            present = ?started.elapsed() - rendered,
515            ?stats,
516            "frame rendered"
517        );
518        let total = started.elapsed();
519        self.diagnostics.record(FrameSample {
520            update: updated,
521            acquire: acquired - updated,
522            render: rendered - acquired,
523            present: total - rendered,
524            quads: stats.quads,
525            glyphs: stats.glyphs,
526            draw_calls: stats.draw_calls,
527            size: (state.config.width, state.config.height),
528            scale,
529        });
530        if self.ui.needs_redraw() {
531            state.window.request_redraw();
532        }
533        self.update_access();
534    }
535}

537/// Ctrl+Shift+D, or Cmd+Shift+D on a Mac, opens the diagnostics panel.
538fn is_diagnostics_shortcut(key: &WinitKey, modifiers: Modifiers) -> bool {
539    (modifiers.ctrl || modifiers.meta)
540        && modifiers.shift
541        && matches!(key, WinitKey::Character(c) if c.eq_ignore_ascii_case("d"))
542}

544fn map_key(key: &WinitKey) -> Key {
545    match key {
546        WinitKey::Named(NamedKey::Enter) => Key::Enter,
547        WinitKey::Named(NamedKey::Space) => Key::Space,
548        WinitKey::Named(NamedKey::Tab) => Key::Tab,
549        WinitKey::Named(NamedKey::Escape) => Key::Escape,
550        WinitKey::Named(NamedKey::Backspace) => Key::Backspace,
551        WinitKey::Named(NamedKey::Delete) => Key::Delete,
552        WinitKey::Named(NamedKey::Home) => Key::Home,
553        WinitKey::Named(NamedKey::End) => Key::End,
554        WinitKey::Named(NamedKey::PageUp) => Key::PageUp,
555        WinitKey::Named(NamedKey::PageDown) => Key::PageDown,
556        WinitKey::Named(NamedKey::ArrowLeft) => Key::ArrowLeft,
557        WinitKey::Named(NamedKey::ArrowRight) => Key::ArrowRight,
558        WinitKey::Named(NamedKey::ArrowUp) => Key::ArrowUp,
559        WinitKey::Named(NamedKey::ArrowDown) => Key::ArrowDown,
560        WinitKey::Character(c) if c.as_str() == " " => Key::Space,
561        WinitKey::Character(c) => Key::Character(c.to_string()),
562        other => Key::Other(format!("{other:?}")),
563    }
564}

577impl ApplicationHandler<UserEvent> for App {
578    fn new_events(&mut self, _event_loop: &ActiveEventLoop, cause: StartCause) {
579        if let StartCause::ResumeTimeReached { .. } = cause
580            && let Some(state) = &self.window
581        {
582            state.window.request_redraw();
583        }
584    }
585
586    /// Sleeps until input arrives or until the UI's next timed wake-up.
587    fn about_to_wait(&mut self, event_loop: &ActiveEventLoop) {
588        match self.ui.next_wake() {
589            Some(at) => {
590                let at = self.started + std::time::Duration::from_secs_f64(at.max(0.0));
591                event_loop.set_control_flow(ControlFlow::WaitUntil(at));
592            }
593            None => event_loop.set_control_flow(ControlFlow::Wait),
594        }
595    }
596
597    fn resumed(&mut self, event_loop: &ActiveEventLoop) {
598        if self.window.is_some() {
599            return;
600        }
601        match self.create_window(event_loop) {
602            Ok(state) => self.window = Some(state),
603            Err(e) => {
604                tracing::error!(error = %e, "could not start");
605                self.error = Some(e);
606                event_loop.exit();
607            }
608        }
609    }
610
611    fn user_event(&mut self, _event_loop: &ActiveEventLoop, event: UserEvent) {
612        use accesskit_winit::WindowEvent as A;
613        let event = match event {
614            UserEvent::Access(event) => event,
615            UserEvent::Wake => {
616                tracing::trace!("woken from another thread");
617                if let Some(state) = &self.window {
618                    state.window.request_redraw();
619                }
620                return;
621            }
622        };
623        match event.window_event {
624            A::InitialTreeRequested => {
625                if let Some(access) = &mut self.access {
626                    access.active = true;
627                    access.last = None;
628                }
629                self.ui.update();
630                self.update_access();
631            }
632            A::ActionRequested(request) => {
633                let Some(action) = access::action(request.action) else { return };
634                self.ui.set_time(self.started.elapsed().as_secs_f64());
635                if self.ui.perform_action(request.target_node.0, action)
636                    && let Some(state) = &self.window
637                {
638                    state.window.request_redraw();
639                }
640            }
641            A::AccessibilityDeactivated => {
642                if let Some(access) = &mut self.access {
643                    access.active = false;
644                    access.last = None;
645                }
646            }
647        }
648    }
649
650    fn window_event(&mut self, event_loop: &ActiveEventLoop, _id: WindowId, event: WindowEvent) {
651        if let (Some(access), Some(state)) = (&mut self.access, &self.window) {
652            access.adapter.process_event(&state.window, &event);
653        }
654        match event {
655            WindowEvent::CloseRequested => event_loop.exit(),
656            WindowEvent::Resized(size) => {
657                if let Some(state) = &mut self.window {
658                    state.config.width = size.width.max(1);
659                    state.config.height = size.height.max(1);
660                    state.surface.configure(&state.gpu.device, &state.config);
661                    let window = state.window.clone();
662                    self.update_viewport(&window, size);
663                    window.request_redraw();
664                }
665            }
666            WindowEvent::ScaleFactorChanged { .. } => {
667                if let Some(state) = &self.window {
668                    let window = state.window.clone();
669                    self.update_viewport(&window, window.inner_size());
670                    window.request_redraw();
671                }
672            }
673            WindowEvent::CursorMoved { position, .. } => {
674                let scale = self.ui.scale().max(f32::EPSILON) as f64;
675                let PhysicalPosition { x, y } = position;
676                let p = Point::new((x / scale) as f32, (y / scale) as f32);
677                self.pointer = Some(p);
678                self.dispatch(Event::PointerMove { position: p });
679            }
680            WindowEvent::CursorLeft { .. } => {
681                self.pointer = None;
682                self.dispatch(Event::PointerLeave);
683            }
684            WindowEvent::MouseInput { state, button, .. } => {
685                if let Some(position) = self.pointer {
686                    let button = map_button(button);
687                    let event = match state {
688                        ElementState::Pressed => Event::PointerDown { position, button },
689                        ElementState::Released => Event::PointerUp { position, button },
690                    };
691                    self.dispatch(event);
692                    if state == ElementState::Released && button == PointerButton::Primary {
693                        let now = std::time::Instant::now();
694                        let double = self.last_click.is_some_and(|(t, p)| {
695                            now.duration_since(t).as_millis() < 450
696                                && (p.x - position.x).hypot(p.y - position.y) < 5.0
697                        });
698                        if double {
699                            self.last_click = None;
700                            self.dispatch(Event::DoubleClick { position });
701                        } else {
702                            self.last_click = Some((now, position));
703                        }
704                    }
705                }
706            }
707            WindowEvent::MouseWheel { delta, .. } => {
708                if let Some(position) = self.pointer {
709                    let delta = match delta {
710                        // One notch scrolls three 16px lines, as browsers do.
711                        MouseScrollDelta::LineDelta(x, y) => Point::new(x * 48.0, y * 48.0),
712                        MouseScrollDelta::PixelDelta(p) => {
713                            let scale = self.ui.scale().max(f32::EPSILON) as f64;
714                            Point::new((p.x / scale) as f32, (p.y / scale) as f32)
715                        }
716                    };
717                    if self.modifiers.ctrl {
718                        // How touchpads on Windows and Linux send a pinch, and how a mouse asks to zoom.
719                        self.dispatch(Event::Pinch { position, delta: delta.y / WHEEL_PER_ZOOM });
720                    } else {
721                        self.dispatch(Event::Scroll { position, delta });
722                    }
723                }
724            }
725            WindowEvent::PinchGesture { delta, .. } => {
726                if let Some(position) = self.pointer {
727                    self.dispatch(Event::Pinch { position, delta: delta as f32 });
728                }
729            }
730            WindowEvent::ModifiersChanged(m) => {
731                let s: ModifiersState = m.state();
732                self.modifiers = Modifiers {
733                    shift: s.shift_key(),
734                    ctrl: s.control_key(),
735                    alt: s.alt_key(),
736                    meta: s.super_key(),
737                };
738            }
739            WindowEvent::KeyboardInput { event, .. } => {
740                if is_diagnostics_shortcut(&event.logical_key, self.modifiers) {
741                    if event.state == ElementState::Pressed && !event.repeat {
742                        self.diagnostics.toggle();
743                    }
744                    return;
745                }
746                let key = map_key(&event.logical_key);
747                let modifiers = self.modifiers;
748                self.dispatch(match event.state {
749                    ElementState::Pressed => Event::KeyDown { key, modifiers },
750                    ElementState::Released => Event::KeyUp { key, modifiers },
751                });
752            }
753            WindowEvent::Focused(false) => self.dispatch(Event::FocusLost),
754            WindowEvent::RedrawRequested => self.redraw(),
755            _ => {}
756        }
757    }
