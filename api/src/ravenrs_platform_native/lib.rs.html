58/// What reaches the event loop from outside it.
59#[derive(Debug)]
60enum UserEvent {
61    /// From the screen reader.
62    Access(accesskit_winit::Event),
63    /// A [`UiWaker`](ravenrs_widgets::UiWaker) fired, on some other thread.
64    Wake,
65    /// A touchpad pinch on Wayland, which winit does not deliver.
66    Pinch(f32),
67    /// From the icon beside the clock, or from opening the app a second time.
68    Tray(tray::TrayAction),
69    /// The app changed the entries of the menu beside the clock.
70    TrayMenu,
71    /// The [`UiDriver`] of [`run_driven`] asked for a turn.
72    Drive,
73}

76    fn from(event: accesskit_winit::Event) -> Self {
77        Self::Access(event)
78    }

81/// How the main window opens: its title, size and starting theme.
82///
83/// Sizes are in logical pixels, which the system scales for the screen, so 800 is the
84/// same apparent width on a regular and on a high-density display. The defaults are a
85/// 960 × 640 window that the person cannot shrink below 320 × 240, with the default theme.
86#[derive(Clone, Debug)]
87pub struct WindowOptions {
88    /// Window title.
89    pub title: String,
90    /// Initial inner size in logical pixels.
91    pub size: Size,
92    /// Smallest inner size the user can resize to, in logical pixels.
93    pub min_size: Option<Size>,
94    /// Theme the UI starts with. The view can change it later, as with a light and dark
95    /// switch; this is only the first frame's.
96    pub theme: Theme,
97    /// Whether the theme follows the system between Raven Dark and Raven Light.
98    pub follow_system_theme: bool,
99}

147/// Why the app could not open its window or stopped with an error.
148///
149/// Returned by [`run`], [`run_as`] and [`run_with_icon`]. Once the window is on screen,
150/// trouble that the app can survive is logged instead: a lost GPU is recreated, and
151/// a missing clipboard, icon or voice is done without. When the window will not open,
152/// [`diagnose`] says which GPU and drivers the machine has. The crate documentation has
153/// a section on what to try for each of these.
154#[derive(Debug)]
155pub enum PlatformError {
156    /// The window system could not be reached, or the event loop stopped with an error.
157    /// On Linux this is usually a session with neither `DISPLAY` nor `WAYLAND_DISPLAY`,
158    /// as over SSH. It also happens when the app tries to run a second event loop:
159    /// a process has one, so call `run` once.
160    EventLoop(String),
161    /// The window system refused to create the window; the message carries its reason.
162    /// It can also come later, when a window closed to the tray on Wayland could not be
163    /// created again.
164    Window(String),
165    /// No GPU adapter, device or drawing surface could be created: an old or missing
166    /// driver, or a virtual machine without 3D acceleration. On Linux, `WGPU_BACKEND=gl`
167    /// tries OpenGL instead of Vulkan, and `LIBGL_ALWAYS_SOFTWARE=1` with it draws on the CPU.
168    Gpu(GpuError),
169}
170
171impl fmt::Display for PlatformError {
172    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
173        match self {
174            Self::EventLoop(e) => write!(f, "event loop error: {e}"),
175            Self::Window(e) => write!(f, "window could not be created: {e}"),
176            Self::Gpu(e) => write!(f, "{e}"),
177        }
178    }
179}
180
181impl std::error::Error for PlatformError {}
182
183impl From<GpuError> for PlatformError {
184    fn from(e: GpuError) -> Self {
185        Self::Gpu(e)
186    }
187}

189/// Opens a window showing `root` and runs until the window is closed.
190///
191/// This takes over the calling thread: it sets up the window, the GPU and the system's
192/// services, then runs the event loop, drawing a frame only when something changed.
193/// It returns `Ok` when the person closes the window, and an error only when the app
194/// could not start or the loop failed. Call it once, from `main`, on the main thread,
195/// which macOS requires.
196///
197/// The window has the system's generic icon, and Linux cannot tie it to a `.desktop`
198/// file. For an installed app, use [`run_as`]; for just an icon, [`run_with_icon`].
199///
200/// ```no_run
201/// use ravenrs_platform_native::{WindowOptions, run};
202/// use ravenrs_widgets::Text;
203///
204/// run(Text::new("Hello"), WindowOptions::new("Hello"))?;
205/// # Ok::<(), ravenrs_platform_native::PlatformError>(())
206/// ```
207pub fn run(root: impl View + 'static, options: WindowOptions) -> Result<(), PlatformError> {
208    run_window(root, options, AppIdentity::anonymous(None), None)
209}

211/// Like [`run`], with the window's interface driven by `driver`, as a test tool drives it to run
212/// a scenario where the person can watch (`--headed` in Raven Test Studio).
213///
214/// The driver gets the window's own [`Ui`] on this thread whenever it asks for a turn, so what
215/// it does is what the window shows. While it is in control ([`Driving::Driven`]) the window
216/// draws, resizes and animates, but the person's pointer, keyboard and focus changes do not
217/// reach the interface; once it lets go ([`Driving::Released`]) the window is an ordinary one;
218/// when it is done ([`Driving::Done`]) the window closes and this returns. Closing the window
219/// while the driver is in control tells it ([`UiDriver::window_closed`]) and ends the app.
220///
221/// The app runs with no identity: no tray, no single copy and no app id, so a test window never
222/// reaches a copy of the app the person has open.
223pub fn run_driven(
224    root: impl View + 'static,
225    options: WindowOptions,
226    driver: impl UiDriver + 'static,
227) -> Result<(), PlatformError> {
228    run_window(root, options, AppIdentity::anonymous(None), Some(Box::new(driver)))
229}

231/// Who the app is to the system, and what it asks of the desktop: given to [`run_as`].
232///
233/// The id is how the system recognizes the window as belonging to the installed app,
234/// to show its icon and group it with its launcher. On top of it, the identity carries
235/// what an installed app usually wants: the words of the UI ([`labels`](Self::labels)),
236/// a diagnostics panel shared with the app ([`diagnostics`](Self::diagnostics)), a page for
237/// bug reports ([`new_issue_url`](Self::new_issue_url)), a window icon ([`icon`](Self::icon))
238/// and closing to the tray ([`close_to_tray`](Self::close_to_tray)).
239///
240/// On Linux the id is what ties the window to the app's `.desktop` file. Wayland
241/// has no way for a window to carry an icon: GNOME shows the one in the
242/// `.desktop` file whose name, without the extension, is the window's id. On
243/// X11 the id must match the file's `StartupWMClass`.
244///
245/// ```
246/// use ravenrs_platform_native::AppIdentity;
247///
248/// // For a file named `notes.desktop`.
249/// let identity = AppIdentity::new("notes");
250/// ```
251#[derive(Clone, Debug)]
252pub struct AppIdentity {
253    id: String,
254    icon: Option<ImageData>,
255    labels: Option<Labels>,
256    diagnostics: Option<Diagnostics>,
257    new_issue_url: Option<String>,
258    tray: Option<SystemTray>,
259}

313/// Like [`run`], for an installed app that the system should recognize as itself.
314///
315/// The [`AppIdentity`] ties the window to the app's `.desktop` file on Linux, so its icon
316/// and launcher are found, and turns on what an installed app needs: the language of the

328/// ```
329pub fn run_as(
330    root: impl View + 'static,
331    options: WindowOptions,
332    identity: AppIdentity,
333) -> Result<(), PlatformError> {
334    run_window(root, options, identity, None)
335}

350fn run_window(
351    root: impl View + 'static,
352    options: WindowOptions,
353    identity: AppIdentity,
354    driver: Option<Box<dyn UiDriver>>,
355) -> Result<(), PlatformError> {
356    let AppIdentity { id, icon, labels, diagnostics, new_issue_url, tray } = identity;
357    let app_id = (!id.is_empty()).then_some(id);
358    let single_copy = app_id.as_deref().filter(|_| tray.is_some());
359    if single_copy.is_some_and(tray::ask_running_copy) {
360        tracing::info!("the app was already running, and opened its window");
361        return Ok(());
362    }
363    let event_loop = EventLoop::<UserEvent>::with_user_event()
364        .build()
365        .map_err(|e| PlatformError::EventLoop(e.to_string()))?;
366    event_loop.set_control_flow(ControlFlow::Wait);
367    let diagnostics = diagnostics.unwrap_or_default();
368    diagnostics.set_open(std::env::var_os(DIAGNOSTICS_VARIABLE).is_some());
369    if let Some(url) = new_issue_url {
370        diagnostics.set_bug_reporter(url, |url| open_url(url).map_err(|e| e.to_string()));
371    }
372    let root = DiagnosticsHost::new(root, diagnostics.clone());
373    let mut ui = match labels {
374        Some(labels) => Ui::new(Localized::new(labels, root), options.theme),
375        None => Ui::new(root, options.theme),
376    };
377    if options.follow_system_theme {
378        ui.follow_system_theme(true);
379    }
380    let wake = event_loop.create_proxy();
381    ui.set_remote_waker(move || {
382        // It only fails once the loop is gone, when there is nothing left to wake.
383        let _ = wake.send_event(UserEvent::Wake);
384    });
385    match arboard::Clipboard::new() {
386        Ok(clipboard) => ui.set_clipboard(SystemClipboard(std::cell::RefCell::new(clipboard))),
387        Err(e) => tracing::warn!(error = %e, "system clipboard unavailable; using an in-app clipboard"),
388    }
389    ui.set_sound(sound::SystemSound::default());
390    ui.set_file_picker(files::SystemFilePicker);
391    ui.set_links(SystemLinks);
392    let screen_window = ScreenWindow::default();
393    ui.set_full_screen(SystemFullScreen(screen_window.clone()));
394    let answer = event_loop.create_proxy();
395    let tray_notify: tray::Notify = Arc::new(move |action| {
396        // It only fails once the loop is gone, when there is nothing left to tell.
397        let _ = answer.send_event(UserEvent::Tray(action));
398    });
399    if let Some(tray) = &tray {
400        let changed = event_loop.create_proxy();
401        tray.entries().on_change(move || {
402            let _ = changed.send_event(UserEvent::TrayMenu);
403        });
404    }
405    if let Some(id) = single_copy
406        && let Err(e) = tray::listen_for_other_copies(id, tray_notify.clone())
407    {
408        tracing::warn!(error = %e, "opening the app again will start a second copy");
409    }
410    let mut app = App {
411        display: event_loop.owned_display_handle(),
412        diagnostics,
413        proxy: event_loop.create_proxy(),
414        access: None,
415        ui,
416        options,
417        icon,
418        app_id,
419        window: None,
420        screen_window,
421        tray,
422        tray_notify,
423        in_tray: None,
424        away_in_tray: false,
425        place: None,
426        maximized: false,
427        pointer: None,
428        modifiers: Modifiers::default(),
429        last_click: None,
430        started: std::time::Instant::now(),
431        looking: false,
432        driving: if driver.is_some() { Driving::Driven } else { Driving::Released },
433        driver,
434        error: None,
435    };
436    event_loop.run_app(&mut app).map_err(|e| PlatformError::EventLoop(e.to_string()))?;
437    match app.error {
438        Some(e) => Err(e),
439        None => Ok(()),
440    }
441}

537impl App {
538    fn create_window(&mut self, event_loop: &ActiveEventLoop) -> Result<WindowState, PlatformError> {
539        // Hidden until `show_window`, so nobody waits looking at an empty window.
540        // AccessKit also requires it until the adapter is attached.
541        let mut attributes = Window::default_attributes()
542            .with_visible(false)
543            .with_title(self.options.title.clone())
544            .with_inner_size(LogicalSize::new(self.options.size.width, self.options.size.height));
545        if let Some(place) = self.place {
546            attributes = attributes.with_position(place);
547        }
548        attributes = attributes.with_maximized(self.maximized);
549        if let Some(min) = self.options.min_size {
550            attributes = attributes.with_min_inner_size(LogicalSize::new(min.width, min.height));
551        }
552        if let Some(id) = &self.app_id {
553            attributes = with_app_id(attributes, id);
554        }
555        if let Some(image) = &self.icon {
556            match Icon::from_rgba(image.pixels().to_vec(), image.width(), image.height()) {
557                Ok(icon) => attributes = attributes.with_window_icon(Some(icon)),
558                Err(e) => tracing::warn!(error = %e, "could not use the window icon"),
559            }
560        }
561        let window =
562            Arc::new(event_loop.create_window(attributes).map_err(|e| PlatformError::Window(e.to_string()))?);
563        let adapter =
564            accesskit_winit::Adapter::with_event_loop_proxy(event_loop, &window, self.proxy.clone());
565        self.access = Some(Access { adapter, active: false, last: None });
566        *self.screen_window.borrow_mut() = Arc::downgrade(&window);
567        if let Some(theme) = window.theme() {
568            self.ui.set_system_dark(theme == winit::window::Theme::Dark);
569        }
570        self.init_gpu(window)
571    }
572
573    /// Closes the window behind the icon, keeping the UI and its state. `false` if the
574    /// system shows no icon, and only closing the app is left.
575    ///
576    /// The window is only hidden, so that it comes back at once, with its last frame.
577    /// Wayland cannot hide one, so there it is destroyed, and that system shows a new
578    /// window only once it has something to show.
579    fn close_to_tray(&mut self, event_loop: &ActiveEventLoop) -> bool {
580        let (Some(tray), Some(icon), Some(state)) = (&self.tray, &self.icon, &self.window) else {
581            return false;
582        };
583        let id = self.app_id.as_deref().unwrap_or_default();
584        match tray::show(tray, id, icon, self.tray_notify.clone()) {
585            Ok(shown) => self.in_tray = Some(shown),
586            Err(e) => {
587                tracing::info!(error = %e, "no icon beside the clock, so closing the window closes the app");
588                return false;
589            }
590        }
591        let window = &state.window;
592        if !self.on_wayland() {
593            window.set_visible(false);
594            self.away_in_tray = false;
595            leave_the_front(event_loop);
596            return true;
597        }
598        let size = window.inner_size().to_logical::<f32>(window.scale_factor());
599        self.options.size = Size::new(size.width, size.height);
600        self.place = window.outer_position().ok();
601        self.maximized = window.is_maximized();
602        // These hold the window, which only closes when the last of them lets go.
603        state.gpu.device.set_device_lost_callback(|_, _| {});
604        self.ui.set_waker(|| {});
605        self.access = None;
606        self.pointer = None;
607        self.window = None;
608        true
609    }

611    /// Brings the window back, or to the front if it never left.
612    /// An entry of the app's menu beside the clock was chosen: one whose id starts with
613    /// `open:` opens the window first, and the app hears the id on this, the interface's thread.
614    fn tray_choice(&mut self, event_loop: &ActiveEventLoop, id: &str) {
615        if id.starts_with(ravenrs_core::OPENS_WINDOW) {
616            self.open_window(event_loop);
617        }
618        if let Some(tray) = self.tray.clone() {
619            tray.select(id);
620        }
621        // The app may keep the choice for its next build, which runs now even with no window.
622        self.ui.ui_waker().wake();
623        let clock = self.clock();
624        self.ui.set_time(clock);
625        self.ui.update();
626        if let Some(state) = &self.window {
627            state.window.request_redraw();
628        }
629    }
630
631    fn open_window(&mut self, event_loop: &ActiveEventLoop) {
632        self.in_tray = None;
633        if let Some(state) = &self.window {
634            state.window.set_visible(true);
635            state.window.set_minimized(false);
636            state.window.focus_window();
637            state.window.request_redraw();
638            return;
639        }
640        match self.create_window(event_loop) {
641            Ok(state) => self.show_window(state),
642            Err(e) => {
643                tracing::error!(error = %e, "could not open the window again");
644                self.error = Some(e);
645                event_loop.exit();
646            }
647        }
648    }

650    /// Puts a new window on screen. On macOS its first frame, the slow one with the
651    /// shaders and the glyphs, is drawn while hidden, so that nobody sees it black.
652    fn show_window(&mut self, state: WindowState) {
653        let window = state.window.clone();
654        self.window = Some(state);
655        if cfg!(target_os = "macos") {
656            self.redraw();
657        }
658        window.set_visible(true);
659        window.focus_window();
660        // What was drawn while hidden does not count for the system.
661        window.request_redraw();
662    }

664    /// Sends the accessibility tree to the screen reader if one is listening
665    /// and the tree changed.
666    fn update_access(&mut self) {
667        let Some(access) = &mut self.access else { return };
668        if !access.active {
669            return;
670        }
671        let tree = self.ui.access_tree();
672        if access.last.as_ref() == Some(&tree) {
673            return;
674        }
675        let scale = f64::from(self.ui.scale());
676        access.adapter.update_if_active(|| access::tree_update(&tree, scale));
677        access.last = Some(tree);
678    }

680    fn init_gpu(&mut self, window: Arc<Window>) -> Result<WindowState, PlatformError> {
681        let instance = diagnostics::new_instance(Some(&self.display), false);
682        let surface =
683            instance.create_surface(window.clone()).map_err(|e| GpuError::Surface(e.to_string()))?;
684        let gpu = pollster::block_on(GpuContext::new(&instance, Some(&surface)))?;
685        let caps = surface.get_capabilities(&gpu.adapter);
686        let format = preferred_surface_format(&caps.formats)
687            .ok_or_else(|| GpuError::Surface("the surface reports no supported formats".into()))?;
688        let size = window.inner_size();
689        let config = wgpu::SurfaceConfiguration {
690            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
691            format,
692            width: size.width.max(1),
693            height: size.height.max(1),
694            present_mode: wgpu::PresentMode::AutoVsync,
695            desired_maximum_frame_latency: 2,
696            alpha_mode: caps.alpha_modes.first().copied().unwrap_or(wgpu::CompositeAlphaMode::Auto),
697            view_formats: vec![],
698            color_space: wgpu::SurfaceColorSpace::Auto,
699        };
700        surface.configure(&gpu.device, &config);
701        let described = surface_report(&caps, format, config.present_mode);
702        let (display, adapter) = (self.display.clone(), gpu.adapter.clone());
703        self.diagnostics.set_report_source(move || {
704            diagnostics::gather(Some(&display), &adapter, Some(described.clone()))
705        });
706        let device_lost = Arc::new(AtomicBool::new(false));
707        let (lost, lost_window) = (device_lost.clone(), window.clone());
708        gpu.device.set_device_lost_callback(move |reason, message| {
709            tracing::error!(?reason, message, "GPU device lost; recreating it");
710            lost.store(true, Ordering::SeqCst);
711            lost_window.request_redraw();
712        });
713        let mut renderer = Renderer::new(&gpu.device, &gpu.queue, format);
714        // Cables swing on the GPU: one compute dispatch a frame for all of them.
715        if let Some(cables) = ravenrs_render::GpuCables::new(&gpu.device, &gpu.queue) {
716            renderer.set_cables(Some(cables.drawer()));
717            self.ui.set_cable_world(Box::new(cables));
718        }
719
720        let redraw = window.clone();
721        self.ui.set_waker(move || redraw.request_redraw());
722        self.update_viewport(&window, size);
723        window.request_redraw();
724        tracing::info!(?format, width = size.width, height = size.height, "window ready");
725        Ok(WindowState { surface, config, gpu, renderer, device_lost, cursor: CursorIcon::Default, window })
726    }

728    fn update_viewport(&mut self, window: &Window, size: PhysicalSize<u32>) {
729        let scale = window.scale_factor() as f32;
730        let logical = Size::new(size.width as f32 / scale, size.height as f32 / scale);
731        self.ui.set_viewport(logical, scale);
732    }

734    fn dispatch(&mut self, event: Event) {
735        let clock = self.clock();
736        self.ui.set_time(clock);
737        let redraw = self.ui.handle_event(&event);
738        if let Some(state) = &mut self.window {
739            let cursor = self.ui.cursor();
740            if cursor != state.cursor {
741                state.window.set_cursor(match cursor {
742                    CursorIcon::Default => winit::window::CursorIcon::Default,
743                    CursorIcon::Pointer => winit::window::CursorIcon::Pointer,
744                    CursorIcon::Text => winit::window::CursorIcon::Text,
745                    CursorIcon::ColResize => winit::window::CursorIcon::ColResize,
746                    CursorIcon::RowResize => winit::window::CursorIcon::RowResize,
747                    CursorIcon::NwseResize => winit::window::CursorIcon::NwseResize,
748                    CursorIcon::NeswResize => winit::window::CursorIcon::NeswResize,
749                    CursorIcon::Move => winit::window::CursorIcon::Move,
750                    _ => winit::window::CursorIcon::Default,
751                });
752                state.cursor = cursor;
753            }
754            if redraw {
755                state.window.request_redraw();
756            }
757        }
758    }
759
760    fn redraw(&mut self) {
761        // Some systems still ask for frames of a hidden window. Coming back draws what changed.
762        if self.in_tray.is_some() {
763            return;
764        }
765        let lost = self.window.as_ref().is_some_and(|s| s.device_lost.load(Ordering::SeqCst));
766        if lost && let Some(old) = self.window.take() {
767            let window = old.window.clone();
768            drop(old);
769            match self.init_gpu(window) {
770                Ok(state) => self.window = Some(state),
771                Err(e) => tracing::error!(error = %e, "GPU recovery failed"),
772            }
773        }
774        let clock = self.clock();
775        self.ui.set_time(clock);
776        let Some(state) = &mut self.window else { return };
777        let scale = self.ui.scale();
778        // Looking costs a little, so it only happens while the panel is open.
779        let looking = self.diagnostics.is_open();
780        if looking != self.looking {
781            self.looking = looking;
782            let start = self.started;
783            let clock: Option<Rc<dyn Fn() -> f64>> =
784                looking.then(|| Rc::new(move || start.elapsed().as_secs_f64()) as Rc<dyn Fn() -> f64>);
785            self.ui.set_phase_clock(clock);
786            self.ui.set_view_profiling(looking);
787            state.renderer.set_gpu_timing(looking);
788        }
789        let started = std::time::Instant::now();
790        let scene = self.ui.update();
791        let updated = started.elapsed();
792        let frame = match state.surface.get_current_texture() {
793            wgpu::CurrentSurfaceTexture::Success(frame) => frame,
794            wgpu::CurrentSurfaceTexture::Suboptimal(frame) => {
795                state.surface.configure(&state.gpu.device, &state.config);
796                frame
797            }
798            wgpu::CurrentSurfaceTexture::Outdated | wgpu::CurrentSurfaceTexture::Lost => {
799                state.surface.configure(&state.gpu.device, &state.config);
800                state.window.request_redraw();
801                return;
802            }
803            wgpu::CurrentSurfaceTexture::Timeout | wgpu::CurrentSurfaceTexture::Occluded => return,
804            other => {
805                tracing::error!(?other, "could not acquire a frame");
806                return;
807            }
808        };
809        let acquired = started.elapsed();
810        let view = frame.texture.create_view(&wgpu::TextureViewDescriptor::default());
811        let stats = state.renderer.render(scene, &view, (state.config.width, state.config.height), scale);
812        let rendered = started.elapsed();
813        state.window.pre_present_notify();
814        state.gpu.queue.present(frame);
815        tracing::debug!(
816            update = ?updated,
817            acquire = ?acquired - updated,
818            render = ?rendered - acquired,
819            present = ?started.elapsed() - rendered,
820            ?stats,
821            "frame rendered"
822        );
823        let total = started.elapsed();
824        if looking {
825            self.diagnostics.set_view_costs(self.ui.view_costs());
826        }
827        self.diagnostics.record(FrameSample {
828            work: self.ui.take_frame_work(),
829            render_work: stats.work(),
830            update: updated,
831            acquire: acquired - updated,
832            render: rendered - acquired,
833            present: total - rendered,
834            quads: stats.quads,
835            glyphs: stats.glyphs,
836            draw_calls: stats.draw_calls,
837            size: (state.config.width, state.config.height),
838            scale,
839        });
840        if self.ui.needs_redraw() {
841            state.window.request_redraw();
842        }
843        self.update_access();
844    }
845}

847impl App {
848    /// Hands the driver of [`run_driven`] its wake function, once the window is on screen, and
849    /// gives it its first turn.
850    fn start_driver(&mut self, event_loop: &ActiveEventLoop) {
851        let Some(driver) = &mut self.driver else { return };
852        let proxy = self.proxy.clone();
853        // It only fails once the loop is gone, when there is nothing left to drive.
854        driver.start(Box::new(move || {
855            let _ = proxy.send_event(UserEvent::Drive);
856        }));
857        self.drive_turn(event_loop);
858    }
859
860    /// The interface's clock, in seconds since the app started: the real one, or the one the
861    /// driver of [`run_driven`] keeps while it is in control ([`UiDriver::time`]).
862    fn clock(&mut self) -> f64 {
863        let now = self.started.elapsed().as_secs_f64();
864        match &mut self.driver {
865            Some(driver) if self.driving == Driving::Driven => driver.time(now),
866            _ => now,
867        }
868    }
869
870    /// A turn of the driver of [`run_driven`] on the window's interface, then a frame.
871    fn drive_turn(&mut self, event_loop: &ActiveEventLoop) {
872        if self.driving == Driving::Done {
873            return;
874        }
875        let Some(driver) = &mut self.driver else { return };
876        let now = self.started.elapsed().as_secs_f64();
877        let now = driver.time(now);
878        self.driving = driver.drive(&mut self.ui, now);
879        if self.driving == Driving::Done {
880            event_loop.exit();
881            return;
882        }
883        if let Some(state) = &self.window {
884            state.window.request_redraw();
885        }
886    }
887}

962    /// On Wayland, winit does not deliver a touchpad's pinch, so it is heard apart.
963    fn on_wayland(&self) -> bool {
964        use winit::raw_window_handle::{HasDisplayHandle, RawDisplayHandle};
965        self.display
966            .display_handle()
967            .is_ok_and(|handle| matches!(handle.as_raw(), RawDisplayHandle::Wayland(_)))
968    }
969
970    fn listen_for_wayland_pinches(&self) {
971        use winit::raw_window_handle::{HasDisplayHandle, RawDisplayHandle};
972        let Ok(handle) = self.display.display_handle() else { return };
973        let RawDisplayHandle::Wayland(wayland) = handle.as_raw() else { return };
974        let proxy = self.proxy.clone();
975        let heard = ravenrs_wayland_gestures::listen(wayland.display, move |delta| {
976            let _ = proxy.send_event(UserEvent::Pinch(delta));
977        });
978        if let Err(e) = heard {
979            tracing::info!(error = %e, "no touchpad pinches on this Wayland");
980        }
981    }

984impl ApplicationHandler<UserEvent> for App {
985    fn new_events(&mut self, _event_loop: &ActiveEventLoop, cause: StartCause) {
986        if let StartCause::ResumeTimeReached { .. } = cause
987            && let Some(state) = &self.window
988        {
989            state.window.request_redraw();
990        }
991    }
992
993    /// Sleeps until input arrives or until the UI's next timed wake-up.
994    fn about_to_wait(&mut self, event_loop: &ActiveEventLoop) {
995        // A view closed the window once its question was answered.
996        if self.ui.take_close() && !self.close_to_tray(event_loop) {
997            event_loop.exit();
998            return;
999        }
1000        if self.in_tray.is_some()
1001            && let Some(active) = tray::app_is_active()
1002        {
1003            if !active {
1004                self.away_in_tray = true;
1005            } else if self.away_in_tray {
1006                self.open_window(event_loop);
1007            }
1008        }
1009        match self.ui.next_wake().filter(|_| self.in_tray.is_none()) {
1010            Some(at) => {
1011                // The interface's clock may run behind the real one while a driver holds it.
1012                let behind = self.started.elapsed().as_secs_f64() - self.clock();
1013                let at = self.started + std::time::Duration::from_secs_f64((at + behind).max(0.0));
1014                event_loop.set_control_flow(ControlFlow::WaitUntil(at));
1015            }
1016            None => event_loop.set_control_flow(ControlFlow::Wait),
1017        }
1018    }
1019
1020    fn resumed(&mut self, event_loop: &ActiveEventLoop) {
1021        if self.window.is_some() {
1022            return;
1023        }
1024        match self.create_window(event_loop) {
1025            Ok(state) => {
1026                self.show_window(state);
1027                self.listen_for_wayland_pinches();
1028                self.start_driver(event_loop);
1029            }
1030            Err(e) => {
1031                tracing::error!(error = %e, "could not start");
1032                self.error = Some(e);
1033                event_loop.exit();
1034            }
1035        }
1036    }
1037
1038    fn user_event(&mut self, event_loop: &ActiveEventLoop, event: UserEvent) {
1039        use accesskit_winit::WindowEvent as A;
1040        let event = match event {
1041            UserEvent::Access(event) => event,
1042            UserEvent::Tray(tray::TrayAction::Open) => return self.open_window(event_loop),
1043            UserEvent::Tray(tray::TrayAction::Quit) => return event_loop.exit(),
1044            UserEvent::Tray(tray::TrayAction::Select(id)) => return self.tray_choice(event_loop, &id),
1045            UserEvent::TrayMenu => {
1046                if let (Some(shown), Some(tray)) = (&self.in_tray, &self.tray) {
1047                    shown.refresh(tray);
1048                }
1049                return;
1050            }
1051            UserEvent::Pinch(delta) => {
1052                if let Some(position) = self.pointer.filter(|_| self.driving != Driving::Driven) {
1053                    self.dispatch(Event::Pinch { position, delta });
1054                }
1055                return;
1056            }
1057            UserEvent::Drive => return self.drive_turn(event_loop),
1058            UserEvent::Wake => {
1059                tracing::trace!("woken from another thread");
1060                if self.in_tray.is_some() {
1061                    // A hidden window draws nothing, but the screen still builds: what it reads
1062                    // (an update that finished, a run that ended) reaches the menu of the icon.
1063                    let clock = self.clock();
1064                    self.ui.set_time(clock);
1065                    self.ui.update();
1066                } else if let Some(state) = &self.window {
1067                    state.window.request_redraw();
1068                }
1069                return;
1070            }
1071        };
1072        match event.window_event {
1073            A::InitialTreeRequested => {
1074                if let Some(access) = &mut self.access {
1075                    access.active = true;
1076                    access.last = None;
1077                }
1078                self.ui.update();
1079                self.update_access();
1080            }
1081            A::ActionRequested(request) => {
1082                let Some(action) = access::action(request.action) else { return };
1083                let clock = self.clock();
1084                self.ui.set_time(clock);
1085                if self.ui.perform_action(request.target_node.0, action)
1086                    && let Some(state) = &self.window
1087                {
1088                    state.window.request_redraw();
1089                }
1090            }
1091            A::AccessibilityDeactivated => {
1092                if let Some(access) = &mut self.access {
1093                    access.active = false;
1094                    access.last = None;
1095                }
1096            }
1097        }
1098    }
1099
1100    fn window_event(&mut self, event_loop: &ActiveEventLoop, _id: WindowId, event: WindowEvent) {
1101        if let (Some(access), Some(state)) = (&mut self.access, &self.window) {
1102            access.adapter.process_event(&state.window, &event);
1103        }
1104        if self.driving == Driving::Driven {
1105            if is_person_input(&event) {
1106                return;
1107            }
1108            if matches!(event, WindowEvent::CloseRequested) {
1109                if let Some(driver) = &mut self.driver {
1110                    driver.window_closed();
1111                }
1112                event_loop.exit();
1113                return;
1114            }
1115        }
1116        match event {
1117            WindowEvent::ThemeChanged(theme) => {
1118                self.ui.set_system_dark(theme == winit::window::Theme::Dark);
1119                if let Some(state) = &self.window {
1120                    state.window.request_redraw();
1121                }
1122            }
1123            WindowEvent::CloseRequested => {
1124                // A view that holds the window asks its own question first.
1125                if !self.ui.may_close() {
1126                    if let Some(state) = &self.window {
1127                        state.window.request_redraw();
1128                    }
1129                    return;
1130                }
1131                if !self.close_to_tray(event_loop) {
1132                    event_loop.exit();
1133                }
1134            }
1135            WindowEvent::Resized(size) => {
1136                if let Some(state) = &mut self.window {
1137                    state.config.width = size.width.max(1);
1138                    state.config.height = size.height.max(1);
1139                    state.surface.configure(&state.gpu.device, &state.config);
1140                    let window = state.window.clone();
1141                    self.update_viewport(&window, size);
1142                    window.request_redraw();
1143                }
1144            }
1145            WindowEvent::ScaleFactorChanged { .. } => {
1146                if let Some(state) = &self.window {
1147                    let window = state.window.clone();
1148                    self.update_viewport(&window, window.inner_size());
1149                    window.request_redraw();
1150                }
1151            }
1152            WindowEvent::CursorMoved { position, .. } => {
1153                let scale = self.ui.scale().max(f32::EPSILON) as f64;
1154                let PhysicalPosition { x, y } = position;
1155                let p = Point::new((x / scale) as f32, (y / scale) as f32);
1156                self.pointer = Some(p);
1157                self.dispatch(Event::PointerMove { position: p });
1158            }
1159            WindowEvent::HoveredFile(_) => self.ui.set_files_over(true),
1160            WindowEvent::HoveredFileCancelled => self.ui.set_files_over(false),
1161            WindowEvent::DroppedFile(path) => {
1162                // Nothing here knows how large a place will take; this keeps a drop
1163                // from filling the memory with something no place would keep.
1164                if let Some(file) = files::read_picked(&path, Some(files::MOST_DROPPED_BYTES)) {
1165                    self.ui.drop_files(self.pointer, vec![file]);
1166                }
1167            }
1168            WindowEvent::CursorLeft { .. } => {
1169                self.pointer = None;
1170                self.dispatch(Event::PointerLeave);
1171            }
1172            WindowEvent::MouseInput { state, button, .. } => {
1173                if let Some(position) = self.pointer {
1174                    let button = map_button(button);
1175                    let event = match state {
1176                        ElementState::Pressed => Event::PointerDown { position, button },
1177                        ElementState::Released => Event::PointerUp { position, button },
1178                    };
1179                    self.dispatch(event);
1180                    if state == ElementState::Released && button == PointerButton::Primary {
1181                        let now = std::time::Instant::now();
1182                        let double = self.last_click.is_some_and(|(t, p)| {
1183                            now.duration_since(t).as_millis() < 450
1184                                && (p.x - position.x).hypot(p.y - position.y) < 5.0
1185                        });
1186                        if double {
1187                            self.last_click = None;
1188                            self.dispatch(Event::DoubleClick { position });
1189                        } else {
1190                            self.last_click = Some((now, position));
1191                        }
1192                    }
1193                }
1194            }
1195            WindowEvent::MouseWheel { delta, .. } => {
1196                if let Some(position) = self.pointer {
1197                    let delta = match delta {
1198                        // One notch scrolls three 16px lines, as browsers do.
1199                        MouseScrollDelta::LineDelta(x, y) => Point::new(x * 48.0, y * 48.0),
1200                        MouseScrollDelta::PixelDelta(p) => {
1201                            let scale = self.ui.scale().max(f32::EPSILON) as f64;
1202                            Point::new((p.x / scale) as f32, (p.y / scale) as f32)
1203                        }
1204                    };
1205                    // Shift turns a wheel that only goes up and down sideways, as browsers do.
1206                    let delta =
1207                        if self.modifiers.shift && delta.x == 0.0 { Point::new(delta.y, 0.0) } else { delta };
1208                    if self.modifiers.ctrl {
1209                        // How touchpads on Windows send a pinch, and how a mouse asks to zoom.
1210                        self.dispatch(Event::Pinch { position, delta: delta.y / WHEEL_PER_ZOOM });
1211                    } else {
1212                        self.dispatch(Event::Scroll { position, delta });
1213                    }
1214                }
1215            }
1216            WindowEvent::PinchGesture { delta, .. } => {
1217                if let Some(position) = self.pointer {
1218                    self.dispatch(Event::Pinch { position, delta: delta as f32 });
1219                }
1220            }
1221            WindowEvent::ModifiersChanged(m) => {
1222                let s: ModifiersState = m.state();
1223                self.modifiers = Modifiers {
1224                    shift: s.shift_key(),
1225                    ctrl: s.control_key(),
1226                    alt: s.alt_key(),
1227                    meta: s.super_key(),
1228                    base_key: None,
1229                };
1230            }
1231            WindowEvent::KeyboardInput { event, .. } => {
1232                if is_diagnostics_shortcut(&event.logical_key, self.modifiers) {
1233                    if event.state == ElementState::Pressed && !event.repeat {
1234                        self.diagnostics.toggle();
1235                    }
1236                    return;
1237                }
1238                let key = map_key(&event.logical_key);
1239                let mut modifiers = self.modifiers;
1240                if modifiers.alt
1241                    && let WinitKey::Character(base) = event.key_without_modifiers()
1242                {
1243                    modifiers.base_key = base.chars().next();
1244                }
1245                self.dispatch(match event.state {
1246                    ElementState::Pressed => Event::KeyDown { key, modifiers },
1247                    ElementState::Released => Event::KeyUp { key, modifiers },
1248                });
1249            }
1250            WindowEvent::Focused(false) => self.dispatch(Event::FocusLost),
1251            WindowEvent::RedrawRequested => self.redraw(),
1252            _ => {}
1253        }
1254    }
1255}

1257/// The window full screen applies to. Weak, so a window closed behind the icon still goes.
1258type ScreenWindow = std::rc::Rc<std::cell::RefCell<std::sync::Weak<Window>>>;
1259
1260/// Full screen as a borderless window over the whole monitor it is on.
1261struct SystemFullScreen(ScreenWindow);
1262
1263impl ravenrs_widgets::FullScreenProvider for SystemFullScreen {
1264    fn is_on(&self) -> bool {
1265        self.0.borrow().upgrade().is_some_and(|window| window.fullscreen().is_some())
1266    }
1267
1268    fn set(&self, on: bool) {
1269        if let Some(window) = self.0.borrow().upgrade() {
1270            window.set_fullscreen(on.then_some(winit::window::Fullscreen::Borderless(None)));
1271        }
1272    }
1273}
