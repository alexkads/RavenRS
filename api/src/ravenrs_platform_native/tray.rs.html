1//! The icon beside the clock that keeps an app running with its window closed.
2//!
3//! Each system has its own way of showing it. macOS and Windows use `tray-icon`, on
4//! the thread of the event loop. Linux uses `ksni`, a StatusNotifierItem in pure Rust,
5//! so that building an app asks for no GTK; there the icon only appears if something
6//! hosts it, which plain GNOME does not. That is why [`show`] can fail, and why
7//! whoever calls it then closes the app: an app hidden behind an icon nobody draws
8//! could never be closed again.
9

19/// Keeps the app running beside the clock when its window is closed. Give it to
20/// [`AppIdentity::close_to_tray`](crate::AppIdentity::close_to_tray).
21///
22/// The menu starts with an item that opens the window again, in the state it was left, and
23/// ends with one that really closes the app. Between them go the app's own entries, a
24/// [`TrayMenu`]: its state at the top, things to do, choices with a check mark, submenus and
25/// separators, and an update that is ready ([#643](https://github.com/alexkads/RavenRS/issues/643)).
26/// The app sets the menu whenever its state changes, from any thread, and the icon shows the
27/// new one at once. A choice reaches [`Self::on_select`] on the thread of the interface; an id
28/// that starts with [`OPENS_WINDOW`](ravenrs_core::OPENS_WINDOW) opens the window first.
29///
30/// Where the system shows no such icon, such as plain GNOME, closing the window closes the app
31/// as usual: an app hidden behind an icon that nobody draws could never be closed.
32///
33/// With it, the app also runs as a single copy: launching it again opens the window of
34/// the copy already running, which keeps two copies from working on the same data.
35///
36/// Needs the `tray` feature. Without it, the type still exists, so the app compiles the
37/// same, and closing the window closes the app.
38///
39/// ```
40/// use ravenrs_core::{TrayEntry, TrayMenu};
41/// use ravenrs_platform_native::SystemTray;
42///
43/// let menu = TrayMenu::new(vec![
44///     TrayEntry::status("Tudo em dia", None),
45///     TrayEntry::item("open:new", "Nova nota").shortcut("CmdOrCtrl+N"),
46/// ]);
47/// let tray = SystemTray::new("Notes")
48///     .open("Open Notes")
49///     .quit("Quit")
50///     .menu(menu.clone())
51///     .on_select(|id| println!("chose {id}"));
52/// ```
53#[derive(Clone)]
54#[cfg_attr(not(feature = "tray"), allow(dead_code, reason = "no icon to show them on"))]
55pub struct SystemTray {
56    name: String,
57    open: String,
58    quit: String,
59    menu: TrayMenu,
60    on_select: Option<OnSelect>,
61}
62
63/// What hears a choice of the menu.
64type OnSelect = Rc<dyn Fn(&str)>;

97    pub fn quit(mut self, label: impl Into<String>) -> Self {
98        self.quit = label.into();
99        self

103    /// [`TrayMenu::set`] it when the app's state changes; the icon redraws its menu.

106        self

112        self.on_select = Some(Rc::new(handler));
113        self
114    }

117    pub fn entries(&self) -> &TrayMenu {

122    pub(crate) fn select(&self, id: &str) {
123        if let Some(handler) = &self.on_select {
124            handler(id);
125        }
126    }

135/// What the person asked for, from the icon or from opening the app a second time.
136#[derive(Clone, Debug, PartialEq, Eq)]
137#[cfg_attr(not(feature = "tray"), allow(dead_code, reason = "only the icon offers to quit"))]
138pub(crate) enum TrayAction {
139    Open,
140    Quit,
141    /// An entry of the app's menu, by its id.
142    Select(String),
143}

145/// One line of the menu as the system draws it.
146#[derive(Clone, Debug, PartialEq, Eq)]
147#[cfg_attr(not(feature = "tray"), allow(dead_code, reason = "no icon to draw them on"))]
148pub(crate) enum Row {
149    Item { id: String, label: String, enabled: bool, shortcut: Option<String> },
150    Check { id: String, label: String, checked: bool },
151    Sub { label: String, rows: Vec<Row> },
152    Separator,
153}
154
155/// The id of the item that opens the window, and of the one that quits.
156pub(crate) const OPEN: &str = "ravenrs:open";
157pub(crate) const QUIT: &str = "ravenrs:quit";
158/// The id of lines that cannot be chosen.
159const QUIET: &str = "ravenrs:quiet";
160
161fn quiet(label: &str) -> Row {
162    Row::Item { id: QUIET.to_owned(), label: label.to_owned(), enabled: false, shortcut: None }
163}

165/// The menu as the system draws it: the app's state first, then "open", the app's entries and
166/// "quit", with separators between the groups; a status or a detail is a line that cannot be
167/// chosen.
168pub(crate) fn rows(open: &str, entries: &[TrayEntry], quit: &str) -> Vec<Row> {
169    let open = Row::Item { id: OPEN.to_owned(), label: open.to_owned(), enabled: true, shortcut: None };
170    let quit = Row::Item { id: QUIT.to_owned(), label: quit.to_owned(), enabled: true, shortcut: None };
171    // The state reads first, as on the menus of the system's own apps.
172    let leading = entries.iter().take_while(|e| matches!(e, TrayEntry::Status { .. })).count();
173    let mut out = level(&entries[..leading]);
174    if !out.is_empty() {
175        out.push(Row::Separator);
176    }
177    out.push(open);
178    let rest = level(&entries[leading..]);
179    if !rest.is_empty() {
180        out.push(Row::Separator);
181        out.extend(rest);
182    }
183    out.push(Row::Separator);
184    out.push(quit);
185    out
186}
187
188fn level(entries: &[TrayEntry]) -> Vec<Row> {
189    let mut out = Vec::new();
190    for entry in entries {
191        match entry {
192            TrayEntry::Status { title, detail } => {
193                out.push(quiet(title));
194                if let Some(detail) = detail {
195                    out.push(quiet(detail));
196                }
197            }
198            TrayEntry::Item { id, label, detail, enabled, shortcut } => {
199                out.push(Row::Item {
200                    id: id.clone(),
201                    label: label.clone(),
202                    enabled: *enabled,
203                    shortcut: shortcut.clone(),
204                });
205                if let Some(detail) = detail {
206                    out.push(quiet(detail));
207                }
208            }
209            TrayEntry::Check { id, label, checked } => {
210                out.push(Row::Check { id: id.clone(), label: label.clone(), checked: *checked });
211            }
212            TrayEntry::Submenu { label, entries } => {
213                out.push(Row::Sub { label: label.clone(), rows: level(entries) });
214            }
215            TrayEntry::Separator => {
216                // Never two in a row, nor one at the start of a group.
217                if !matches!(out.last(), None | Some(Row::Separator)) {
218                    out.push(Row::Separator);
219                }
220            }
221        }
222    }
223    if matches!(out.last(), Some(Row::Separator)) {
224        out.pop();
225    }
226    out
227}

229/// What a chosen id means.
230#[cfg_attr(not(feature = "tray"), allow(dead_code, reason = "nothing is chosen"))]
231pub(crate) fn action_of(id: &str) -> Option<TrayAction> {
232    match id {
233        OPEN => Some(TrayAction::Open),
234        QUIT => Some(TrayAction::Quit),
235        QUIET => None,
236        other => Some(TrayAction::Select(other.to_owned())),
237    }
238}

261mod system {
262    use std::str::FromStr as _;
263
264    use tray_icon::menu::accelerator::Accelerator;
265    use tray_icon::menu::{
266        CheckMenuItem, IsMenuItem, Menu, MenuEvent, MenuItem, PredefinedMenuItem, Submenu,
267    };
268    use tray_icon::{TrayIcon, TrayIconBuilder};
269
270    use super::{ImageData, Notify, Row, SystemTray, action_of};
271
272    pub(super) type Icon = TrayIcon;
273
274    pub(super) fn show(
275        tray: &SystemTray,
276        _id: &str,
277        icon: &ImageData,
278        notify: Notify,
279    ) -> Result<Icon, String> {
280        // Only the first one stays, and they all say the same.
281        MenuEvent::set_event_handler(Some(move |event: MenuEvent| {
282            if let Some(action) = action_of(event.id.as_ref()) {
283                notify(action);
284            }
285        }));
286        let image = tray_icon::Icon::from_rgba(icon.pixels().to_vec(), icon.width(), icon.height())
287            .map_err(|e| e.to_string())?;
288        TrayIconBuilder::new()
289            .with_icon(image)
290            .with_tooltip(&tray.name)
291            .with_menu(Box::new(menu(&tray.rows())?))
292            .with_menu_on_left_click(true)
293            .build()
294            .map_err(|e| e.to_string())
295    }
296
297    pub(super) fn refresh(icon: &Icon, tray: &SystemTray) -> Result<(), String> {
298        icon.set_menu(Some(Box::new(menu(&tray.rows())?)));
299        Ok(())
300    }
301
302    fn menu(rows: &[Row]) -> Result<Menu, String> {
303        let menu = Menu::new();
304        for item in items(rows)? {
305            menu.append(item.as_ref()).map_err(|e| e.to_string())?;
306        }
307        Ok(menu)
308    }
309
310    fn items(rows: &[Row]) -> Result<Vec<Box<dyn IsMenuItem>>, String> {
311        let mut out: Vec<Box<dyn IsMenuItem>> = Vec::new();
312        for row in rows {
313            out.push(match row {
314                Row::Item { id, label, enabled, shortcut } => {
315                    let keys = shortcut.as_deref().and_then(|s| Accelerator::from_str(s).ok());
316                    Box::new(MenuItem::with_id(id.as_str(), label, *enabled, keys))
317                }
318                Row::Check { id, label, checked } => {
319                    Box::new(CheckMenuItem::with_id(id.as_str(), label, true, *checked, None))
320                }
321                Row::Sub { label, rows } => {
322                    let sub = Submenu::new(label, true);
323                    for item in items(rows)? {
324                        sub.append(item.as_ref()).map_err(|e| e.to_string())?;
325                    }
326                    Box::new(sub)
327                }
328                Row::Separator => Box::new(PredefinedMenuItem::separator()),
329            });
330        }
331        Ok(out)
332    }
333}

367/// How long one running copy of the app takes to answer another, at most.
368const ANSWER_TIME: Duration = Duration::from_millis(500);
369
370/// Tells a copy of the app already running to open its window. `true` if there is one.
371pub(crate) fn ask_running_copy(id: &str) -> bool {
372    let (address, token) = (instance_address(id), instance_token(id));
373    let Ok(mut stream) = TcpStream::connect_timeout(&address, ANSWER_TIME) else {
374        return false;
375    };
376    let _ = stream.set_read_timeout(Some(ANSWER_TIME));
377    let _ = stream.set_write_timeout(Some(ANSWER_TIME));
378    if stream.write_all(token.as_bytes()).is_err() {
379        return false;
380    }
381    let mut answer = vec![0; token.len()];
382    // Something else on this port does not know the token, and is none of our business.
383    stream.read_exact(&mut answer).is_ok() && answer == token.as_bytes()
384}
385
386/// Listens for other copies of the app for as long as it runs: with a window closed
387/// behind an icon, opening the app again must bring that window back, and not start
388/// a second app over the same data.
389pub(crate) fn listen_for_other_copies(id: &str, notify: Notify) -> Result<(), String> {
390    let listener = TcpListener::bind(instance_address(id)).map_err(|e| e.to_string())?;
391    let token = instance_token(id);
392    std::thread::Builder::new()
393        .name("ravenrs-instance".to_owned())
394        .spawn(move || {
395            for stream in listener.incoming().flatten() {
396                if answer_copy(stream, &token) {
397                    notify(TrayAction::Open);
398                }
399            }
400        })
401        .map(|_| ())
402        .map_err(|e| e.to_string())
403}
404
405fn answer_copy(mut stream: TcpStream, token: &str) -> bool {
406    let _ = stream.set_read_timeout(Some(ANSWER_TIME));
407    let _ = stream.set_write_timeout(Some(ANSWER_TIME));
408    let mut asked = vec![0; token.len()];
409    stream.read_exact(&mut asked).is_ok()
410        && asked == token.as_bytes()
411        && stream.write_all(token.as_bytes()).is_ok()
412}

414/// Told apart by person too, so that two people on one computer each have their app.
415fn instance_token(id: &str) -> String {
416    let person = std::env::var("USER").or_else(|_| std::env::var("USERNAME")).unwrap_or_default();
417    format!("ravenrs-instance:{id}:{person}\n")
418}
419
420/// A port of the dynamic range, the same for every copy of this app of this person.
421fn instance_address(id: &str) -> SocketAddr {
422    const FIRST_PORT: u32 = 49152;
423    const PORTS: u32 = 16384;
424    let hash = instance_token(id)
425        .bytes()
426        .fold(0x811c_9dc5_u32, |hash, byte| (hash ^ u32::from(byte)).wrapping_mul(0x0100_0193));
427    let port = u16::try_from(FIRST_PORT + hash % PORTS).unwrap_or(u16::MAX);
428    SocketAddr::from((Ipv4Addr::LOCALHOST, port))
429}
