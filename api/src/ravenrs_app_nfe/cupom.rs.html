56/// The entity of the lines of a receipt.
57pub fn cupom_entity() -> EntityInfo {
58    CAMPOS.iter().fold(
59        EntityInfo::new(CUPOM_LINHA)
60            .caption("Linhas do cupom")
61            .field(FieldInfo::new("id", FieldKind::Guid).key()),
62        |entity, (name, caption)| entity.field(FieldInfo::new(*name, FieldKind::Text).caption(*caption)),
63    )
64}

70/// The receipt as it comes, for this roll. Everything in it can change in the designer.
71pub fn cupom_report(stock: &LabelStock, name: impl Into<String>) -> Report {
72    let width = stock.width;
73    let base = if stock.width < 180.0 { 7.0 } else { 8.0 };
74    let plain = || Style::new().font_size(base);
75    let center = || Style::new().font_size(base).align(TextAlign::Center);
76    let strong = || Style::new().font_size(base + 1.0).bold().align(TextAlign::Center);
77    let right = || Style::new().font_size(base).align(TextAlign::Right);
78    let line = base * 1.5;
79    let code = width.min(120.0);
80    let mut report = stock.report(name);
81    report.culture = "pt-BR".to_owned();
82    report.report_header = Some(
83        Band::new(line * 6.0)
84            .control(text("[emitente]", 0.0, 0.0, width, line, strong()))
85            .control(text("'CNPJ ' + [cnpj] + '  IE ' + [ie]", 0.0, line, width, line, center()))
86            .control(text("[endereco]", 0.0, line * 2.0, width, line, center()))
87            .control(text("[titulo]", 0.0, line * 3.0, width, line * 2.0, strong()))
88            .control(text("[aviso]", 0.0, line * 5.0, width, line, strong())),
89    );
90    report.detail = Band::new(line * 2.0)
91        .control(text("[ordem] + ' ' + [codigo] + ' ' + [descricao]", 0.0, 0.0, width, line, plain()))
92        .control(text(
93            "[quantidade] + ' ' + [unidade] + ' x ' + [unitario]",
94            0.0,
95            line,
96            width * 0.6,
97            line,
98            plain(),
99        ))
100        .control(text("[valor]", width * 0.6, line, width * 0.4, line, right()));
101    let mut y = 0.0;
102    let mut footer = Band::new(0.0);
103    let rows = [
104        ("'Qtde. total de itens: ' + [itens]", line, right()),
105        ("'TOTAL R$ ' + [total]", line * 1.3, strong()),
106        ("[pagamento] + ' R$ ' + [recebido]", line, right()),
107        ("'Troco R$ ' + [troco]", line, right()),
108        ("'Consulte pela chave de acesso em sua SEFAZ'", line, center()),
109        ("[chave]", line * 2.0, center()),
110        ("[consumidor]", line, center()),
111        ("[numero] + '  ' + [emitido]", line, center()),
112        ("'Protocolo: ' + [protocolo]", line, center()),
113    ];
114    for (expression, height, style) in rows {
115        footer = footer.control(text(expression, 0.0, y, width, height, style));
116        y += height;
117    }
118    footer =
119        footer.control(Control::barcode(Symbology::Qr, "[qr_code]").at((width - code) / 2.0, y, code, code));
120    y += code + 2.0;
121    footer = footer.control(text("[observacao]", 0.0, y, width, line * 2.0, center()));
122    y += line * 2.0;
123    footer = footer.control(Control::printer_command(PrinterAction::Cut).at(0.0, y, width, 12.0));
124    footer.height = y;
125    report.report_footer = Some(footer);
126    report
127}

142/// The DANFE of a sale to a company, on a sheet of A4. Everything in it can change in the designer.
143pub fn danfe_report(name: impl Into<String>) -> Report {
144    let page = PageSettings::a4();
145    let width = page.content_width();
146    let plain = || Style::new().font_size(8.0);
147    let strong = || Style::new().font_size(9.0).bold();
148    let center = || Style::new().font_size(12.0).bold().align(TextAlign::Center);
149    let right = || Style::new().font_size(8.0).align(TextAlign::Right);
150    let mut report = Report::new(name).page(page);
151    report.culture = "pt-BR".to_owned();
152    let line = 13.0;
153    report.report_header = Some(
154        Band::new(line * 15.0)
155            .control(text("[aviso]", 0.0, 0.0, width, line, center()))
156            .control(text(
157                "'DANFE - Documento Auxiliar da Nota Fiscal Eletrônica'",
158                0.0,
159                line,
160                width,
161                line * 1.5,
162                center(),
163            ))
164            .control(text("[numero] + '   ' + [emitido]", 0.0, line * 2.5, width, line, plain()))
165            .control(text("[emitente]", 0.0, line * 3.5, width, line, strong()))
166            .control(text("'CNPJ ' + [cnpj] + '  IE ' + [ie]", 0.0, line * 4.5, width, line, plain()))
167            .control(text("[endereco]", 0.0, line * 5.5, width, line, plain()))
168            .control(Control::barcode(Symbology::Code128, "[chave_digitos]").at(
169                0.0,
170                line * 6.7,
171                width * 0.7,
172                line * 2.6,
173            ))
174            .control(text("'Chave de acesso: ' + [chave]", 0.0, line * 9.4, width, line, plain()))
175            .control(text(
176                "'Protocolo de autorização: ' + [protocolo]",
177                0.0,
178                line * 10.4,
179                width,
180                line,
181                plain(),
182            ))
183            .control(text("'DESTINATÁRIO: ' + [dest_nome]", 0.0, line * 11.6, width, line, strong()))
184            .control(text("[dest_documento] + '  IE ' + [dest_ie]", 0.0, line * 12.6, width, line, plain()))
185            .control(text("[dest_endereco]", 0.0, line * 13.6, width, line, plain())),
186    );
187    report.detail = Band::new(line)
188        .control(text("[ordem]", 0.0, 0.0, 24.0, line, plain()))
189        .control(text("[codigo]", 24.0, 0.0, 70.0, line, plain()))
190        .control(text("[descricao]", 94.0, 0.0, width - 94.0 - 210.0, line, plain()))
191        .control(text("[quantidade] + ' ' + [unidade]", width - 210.0, 0.0, 70.0, line, right()))
192        .control(text("[unitario]", width - 140.0, 0.0, 70.0, line, right()))
193        .control(text("[valor]", width - 70.0, 0.0, 70.0, line, right()));
194    report.report_footer = Some(
195        Band::new(line * 5.0)
196            .control(text(
197                "'TOTAL DA NOTA R$ ' + [total]",
198                0.0,
199                line,
200                width,
201                line * 1.3,
202                Style::new().font_size(11.0).bold().align(TextAlign::Right),
203            ))
204            .control(text("[pagamento] + ' R$ ' + [recebido]", 0.0, line * 2.4, width, line, right()))
205            .control(text("[observacao]", 0.0, line * 3.5, width, line * 1.5, plain())),
206    );
207    report
208}

215/// What a layout of this document model cannot do without, and left out.
216fn faltando_no_modelo(report: &Report, modelo: &str) -> Vec<&'static str> {
217    let written = report.to_json();
218    let exigidos: &[&'static str] =
219        if modelo == "55" { &["chave", "numero", "protocolo", "dest_nome"] } else { &CAMPOS_OBRIGATORIOS };
220    exigidos.iter().copied().filter(|field| !written.contains(&format!("[{field}]"))).collect()
221}
222
223/// What makes a receipt fiscal and this layout left out.
224pub fn campos_faltando(report: &Report) -> Vec<&'static str> {
225    faltando_no_modelo(report, "65")
226}

228/// Puts the notice of the environment back, for a layout that took it out.
229fn com_aviso(report: &mut Report) {
230    if report.to_json().contains("[aviso]") {
231        return;
232    }
233    let width = report.page.content_width();
234    let notice = Control::field("[aviso]")
235        .at(0.0, 0.0, width, 12.0)
236        .style(Style::new().font_size(8.0).bold().align(TextAlign::Center));
237    let header = report.report_header.get_or_insert_with(|| Band::new(0.0));
238    for control in &mut header.controls {
239        control.y += 12.0;
240    }
241    header.height += 12.0;
242    header.controls.insert(0, notice);
243}

245/// A stored receipt over the lines of a document, or what keeps it from being printed.
246pub fn renderizar_cupom(
247    session: &Session,
248    modelo: &Object,
249    linhas: &[Object],
250    words: &ReportWords,
251) -> Result<Document, String> {
252    let rows: Vec<(Object, u32)> = linhas.iter().map(|line| (line.clone(), 1)).collect();
253    let do_documento = linhas.first().map_or_else(
254        || "65".to_owned(),
255        |line| match line.get("modelo") {
256            Value::Text(text) => text,
257            _ => "65".to_owned(),
258        },
259    );
260    render_stored_report_with(session, modelo, &rows, words, |report| {
261        let missing = faltando_no_modelo(report, &do_documento);
262        if !missing.is_empty() {
263            return Err(format!(
264                "Este layout não traz o que o documento fiscal precisa: {}. Abra-o no editor e inclua.",
265                missing.join(", ")
266            ));
267        }
268        com_aviso(report);
269        Ok(())
270    })
271}

273/// The lines of the receipt of a document, in order.
274pub fn linhas_do_documento(session: &Session, documento: &str) -> Vec<Object> {
275    let space = session.frame().space();
276    let found = space.borrow().query(CUPOM_LINHA, &ravenrs_app::Query::all()).unwrap_or_default();
277    let mut lines: Vec<Object> =
278        found.into_iter().filter(|line| line.get("documento") == Value::Text(documento.to_owned())).collect();
279    lines.sort_by_key(|line| match line.get("ordem") {
280        Value::Text(order) => order.parse::<u32>().unwrap_or(0),
281        _ => 0,
282    });
283    lines
284}

293    fn linhas(aviso: &str) -> serde_json::Value {
294        let mut rows = Vec::new();
295        for n in 1..=3 {
296            let mut row = serde_json::Map::new();
297            for (name, _) in CAMPOS {
298                row.insert(name.to_owned(), serde_json::json!(format!("{name}{n}")));
299            }
300            row.insert("qr_code".to_owned(), serde_json::json!("https://exemplo/?p=1|3|2"));
301            row.insert("aviso".to_owned(), serde_json::json!(aviso));
302            rows.push(serde_json::Value::Object(row));
303        }
304        serde_json::Value::Array(rows)
305    }

308    fn the_models_have_what_a_receipt_needs_and_fit_their_roll() {
309        let models = modelos_de_cupom();
310        assert_eq!(models.len(), 2);
311        for (name, report) in models {
312            assert!(campos_faltando(&report).is_empty(), "{name}");
313            let document = ReportEngine::new().render(&report, &linhas("SEM VALOR FISCAL"));
314            assert!(document.errors.is_empty(), "{name}: {:?}", document.errors);
315            assert_eq!(document.pages.len(), 1, "{name}");
316            let page = &document.pages[0];
317            assert!(page.height < 2000.0, "{name}: it ends where it prints");
318            assert!(
319                page.items.iter().any(|item| matches!(item, ravenrs_report::Item::BarcodeMark { .. })),
320                "{name}: QR Code"

322            assert!(
323                page.items.iter().any(|item| matches!(item, ravenrs_report::Item::PrinterCommand { .. })),
324                "{name}: cuts at the end"
