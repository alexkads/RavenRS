1//! Saves what a [`FlowCanvas`](ravenrs_widgets::FlowCanvas) exports as a file: a PNG picture at
2//! twice the diagram's size, an SVG drawing, or a PDF page through the report engine's
3//! exporter.
4//!
5//! ```
6//! use ravenrs_flow_export::export;
7//! use ravenrs_widgets::{FlowColors, FlowExportFormat, FlowGraph, FlowNode};
8//!
9//! let graph = FlowGraph::new().node(FlowNode::new("a", "step", "Start", 0.0, 0.0));
10//! let picture = graph.picture(None, FlowColors::paper()).expect("a node to draw");
11//! let file = export(&picture, FlowExportFormat::Svg, "Order flow")?;
12//! assert_eq!(file.name, "Order flow.svg");
13//! assert!(String::from_utf8_lossy(&file.bytes).contains("Start"));
14//! # Ok::<(), String>(())
15//! ```

28/// How many pixels a PNG gets per unit of the diagram.
29pub const PNG_SCALE: f32 = 2.0;
30
31/// The family SVG names for its words: the one the screen draws them in.
32const SVG_FAMILY: &str = "Inter, system-ui, sans-serif";
33
34/// A diagram written as a file.
35#[derive(Clone, Debug, PartialEq, Eq)]
36pub struct FlowFile {
37    /// The file's name: the title, with what file systems refuse taken out, and the extension.
38    pub name: String,
39    /// The media type.
40    pub mime: &'static str,
41    /// The content.
42    pub bytes: Vec<u8>,
43}
44
45/// Writes `picture` in `format`, named after `title`.
46pub fn export(picture: &FlowPicture, format: FlowExportFormat, title: &str) -> Result<FlowFile, String> {
47    let bytes = match format {
48        FlowExportFormat::Svg => picture.to_svg(SVG_FAMILY).into_bytes(),
49        FlowExportFormat::Png => png(picture, PNG_SCALE, &mut TextSystem::new())?,
50        FlowExportFormat::Pdf => pdf(picture, title, &mut TextSystem::new())?,
51        _ => return Err(format!("{format:?} is not written yet")),
52    };
53    Ok(FlowFile { name: file_name(title, format), mime: format.mime(), bytes })
54}

56/// What [`FlowCanvas::on_export`](ravenrs_widgets::FlowCanvas::on_export) takes to hand each
57/// export to the person as a file, through the platform's way of saving one.
58pub fn saver(
59    picker: FilePicker,
60    title: impl Into<String>,
61) -> impl Fn(FlowExportFormat, FlowPicture) + 'static {
62    let title = title.into();
63    move |format, picture| {
64        if let Ok(file) = export(&picture, format, &title) {
65            picker.save(&file.name, &file.bytes);
66        }
67    }
68}

70/// A title as a file name, keeping accents and dropping what file systems refuse.
71pub fn file_name(title: &str, format: FlowExportFormat) -> String {
72    let kept: String = title
73        .chars()
74        .filter(|c| !matches!(c, '/' | '\\' | ':' | '*' | '?' | '"' | '<' | '>' | '|') && !c.is_control())
75        .collect();
76    let kept = kept.trim();
77    let stem = if kept.is_empty() { "diagram" } else { kept };
78    format!("{stem}.{}", format.extension())
79}

81/// Where one run of words lands, in units of the diagram from the picture's corner: its
82/// left, top, baseline and width, and the words that fit.
83struct Placed {
84    text: String,
85    left: f32,
86    top: f32,
87    baseline: f32,
88    width: f32,
89    height: f32,
90}

92fn attrs(t: &FlowText, scale: f32) -> TextAttrs {
93    TextAttrs {
94        size: t.size * scale,
95        line_height: t.size * 1.3 * scale,
96        weight: ravenrs_report::text_weight(t.bold),
97        letter_spacing: 0.0,
98        family: None,
99    }
100}

102/// The words of `t` that fit its width, ended with `…` when some are cut.
103fn fitted(t: &FlowText, text: &mut TextSystem) -> String {
104    let fits = |words: &str, text: &mut TextSystem| {
105        let mut layout = text.layout(words, &attrs(t, 1.0));
106        layout.measure(None).width <= t.width
107    };
108    if fits(&t.text, text) {
109        return t.text.clone();
110    }
111    let mut chars: Vec<char> = t.text.chars().collect();
112    while !chars.is_empty() {
113        chars.pop();
114        let words = format!("{}…", chars.iter().collect::<String>().trim_end());
115        if fits(&words, text) {
116            return words;
117        }
118    }
119    String::new()
120}

122/// Lays `t` out, cut to its width, and says where it sits from `origin`.
123fn place(t: &FlowText, origin: Point, text: &mut TextSystem) -> Option<Placed> {
124    let shown = fitted(t, text);
125    let mut layout = text.layout(&shown, &attrs(t, 1.0));
126    layout.measure(None);
127    let line = layout.lines().into_iter().next()?;
128    let left = if t.centered { t.at.x - line.width / 2.0 } else { t.at.x } - origin.x;
129    let top = t.at.y - line.height / 2.0 - origin.y;
130    Some(Placed {
131        text: shown,
132        left,
133        top,
134        baseline: top + line.baseline,
135        width: line.width,
136        height: line.height,
137    })
138}

140/// The picture in pixels, `scale` per unit, encoded as PNG.
141fn png(picture: &FlowPicture, scale: f32, text: &mut TextSystem) -> Result<Vec<u8>, String> {
142    let image = vector::rasterize_layers(&picture.layers, picture.bounds, scale)
143        .ok_or_else(|| "the diagram is too large for a picture".to_owned())?;
144    let (width, height) = (image.width(), image.height());
145    let mut pixels = image.pixels().to_vec();
146    let mut glyphs = ScaleContext::new();
147    let origin = picture.bounds.origin;
148    for t in &picture.texts {
149        let mut layout = text.layout_ellipsized(&t.text, &attrs(t, scale), (t.width * scale).max(1.0));
150        layout.measure(None);
151        let Some(line) = layout.lines().into_iter().next() else { continue };
152        let left = if t.centered { t.at.x * scale - line.width / 2.0 } else { t.at.x * scale };
153        let top = t.at.y * scale - line.height / 2.0;
154        let mut scene = Scene::new();
155        layout.paint(Point::new(left - origin.x * scale, top - origin.y * scale), t.color, &mut scene);
156        for primitive in scene.primitives() {
157            if let Primitive::Glyphs(run) = primitive {
158                draw_run(&mut pixels, (width, height), run, t.color, &mut glyphs);
159            }
160        }
161    }
162    let buffer = image::RgbaImage::from_raw(width, height, pixels).ok_or_else(|| "bad picture".to_owned())?;
163    let mut out = Cursor::new(Vec::new());
164    buffer.write_to(&mut out, image::ImageFormat::Png).map_err(|e| e.to_string())?;
165    Ok(out.into_inner())
166}

168/// Lays a run of glyphs over straight-alpha RGBA pixels in `color`.
169fn draw_run(
170    pixels: &mut [u8],
171    (width, height): (u32, u32),
172    run: &ravenrs_core::GlyphRun,
173    color: Color,
174    context: &mut ScaleContext,
175) {
176    let Some(font) = swash::FontRef::from_index(run.font.bytes(), run.font.index as usize) else { return };
177    let mut scaler = context.builder(font).size(run.size).hint(false).build();
178    let [r, g, b, a] = color.to_array();
179    for glyph in &run.glyphs {
180        let Ok(id) = u16::try_from(glyph.id) else { continue };
181        let pen = glyph.x.floor();
182        let Some(mask) = Render::new(&[Source::Outline])
183            .format(Format::Alpha)
184            .offset(Vector::new(glyph.x - pen, 0.0))
185            .render(&mut scaler, id)
186        else {
187            continue;
188        };
189        let left = pen as i64 + i64::from(mask.placement.left);
190        let top = glyph.y.round() as i64 - i64::from(mask.placement.top);
191        let w = mask.placement.width.max(1) as usize;
192        for (row, coverage) in mask.data.chunks_exact(w).enumerate() {
193            for (column, &cover) in coverage.iter().enumerate() {
194                let (x, y) = (left + column as i64, top + row as i64);
195                if cover == 0 || x < 0 || y < 0 || x >= i64::from(width) || y >= i64::from(height) {
196                    continue;
197                }
198                let i = ((y as usize) * width as usize + x as usize) * 4;
199                let alpha = f32::from(cover) / 255.0 * a;
200                let under = f32::from(pixels[i + 3]) / 255.0;
201                let out = alpha + under * (1.0 - alpha);
202                if out <= 0.0 {
203                    continue;
204                }
205                for (k, ink) in [r, g, b].into_iter().enumerate() {
206                    let below = f32::from(pixels[i + k]) / 255.0;
207                    let mixed = (ink * alpha + below * under * (1.0 - alpha)) / out;
208                    pixels[i + k] = (mixed * 255.0).round().clamp(0.0, 255.0) as u8;
209                }
210                pixels[i + 3] = (out * 255.0).round().clamp(0.0, 255.0) as u8;
211            }
212        }
213    }
214}

216/// The picture as one page of a document, the size of the diagram, written by the report
217/// engine's PDF exporter: shapes as vectors, words as text.
218fn pdf(picture: &FlowPicture, title: &str, text: &mut TextSystem) -> Result<Vec<u8>, String> {
219    // Laid out as a report's diagram control is, on a page the size of the diagram.
220    let (w, h) = (picture.bounds.width(), picture.bounds.height());
221    let page = Rect::new(0.0, 0.0, w, h);
222    let fitted = ravenrs_report::diagram::fit(picture, page);
223    let mut items = vec![Item::Vector { rect: page, shapes: fitted.shapes }];
224    for label in fitted.labels {
225        let words = FlowText {
226            text: label.text,
227            at: if label.centered {
228                label.rect.center()
229            } else {
230                Point::new(label.rect.x(), label.rect.center().y)
231            },
232            width: label.rect.width(),
233            size: label.size,
234            bold: label.bold,
235            centered: label.centered,
236            color: label.color,
237        };
238        let Some(placed) = place(&words, Point::new(0.0, 0.0), text) else { continue };
239        items.push(Item::Text(TextBlock {
240            clip: Rect::new(placed.left, placed.top, placed.width + 2.0, placed.height),
241            lines: vec![PlacedLine {
242                text: placed.text,
243                x: placed.left,
244                top: placed.top,
245                baseline: placed.baseline,
246                width: placed.width,
247            }],
248            size: words.size,
249            line_height: placed.height,
250            bold: words.bold,
251            color: words.color,
252            align: TextAlign::Left,
253            value: None,
254            format: None,
255        }));
256    }
257    let document = Document {
258        title: title.to_owned(),
259        culture: "en-US".to_owned(),
260        pages: vec![Page { width: w, height: h, items, ..Page::default() }],
261        errors: Vec::new(),
262    };
263    ravenrs_report_export::export(&document, ExportFormat::Pdf, None).map(|file| file.bytes)
264}
