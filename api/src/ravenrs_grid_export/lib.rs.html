1//! Turns what a [`DataGrid`](ravenrs_widgets::DataGrid) shows into a file. The rows go
2//! through the report engine and its exporters, so a grid exports like a report does.
3//!
4//! ```
5//! use ravenrs_grid_export::export;
6//! use ravenrs_widgets::grid::{GridQuery, to_table};
7//! use ravenrs_widgets::{GridColumn, GridExportFormat};
8//!
9//! let columns = [GridColumn::text("Name"), GridColumn::number("Total")];
10//! let rows = vec![vec!["Ana".into(), 12.5.into()]];
11//! let table = to_table("Sales", "en-US", &columns, &rows, &GridQuery::default(), &[0]);
12//! let file = export(&table, GridExportFormat::Xlsx, Some("2026-11-02"))?;
13//! assert_eq!(file.name, "Sales 2026-11-02.xlsx");
14//! # Ok::<(), String>(())
15//! ```
16

24/// A field name that nothing in a title can break.
25fn field(column: usize) -> String {
26    format!("c{column}")
27}
28
29fn json_of(value: &Value) -> Json {
30    match value {
31        Value::Text(text) => Json::String(text.to_string()),
32        Value::Number(number) => serde_json::Number::from_f64(*number).map_or(Json::Null, Json::Number),
33        Value::Bool(on) => Json::Bool(*on),
34        Value::Date(date) => {
35            Json::String(format!("{:04}-{:02}-{:02}", date.year(), date.month(), date.day()))
36        }
37        _ => Json::Null,
38    }
39}

41/// The format of a number: its decimals, with its text around them.
42fn number_pattern(decimals: usize, prefix: &str, suffix: &str) -> String {
43    let digits = if decimals == 0 { "#,##0".to_owned() } else { format!("#,##0.{}", "0".repeat(decimals)) };
44    format!("{prefix}{digits}{suffix}")
45}

47/// What reads well on paper of `theme`: its accent, with its own text, for the headers, and its
48/// text where that is dark enough for white paper. A dark theme's light text stays out, as paper is
49/// white; then the colors of RavenRS take its place.
50pub fn paper_colors(theme: &Theme, options: WizardOptions) -> WizardOptions {
51    let hex = |color: ravenrs_core::Color| {
52        let byte = |channel: f32| (channel.clamp(0.0, 1.0) * 255.0).round() as u8;
53        format!("#{:02x}{:02x}{:02x}", byte(color.r), byte(color.g), byte(color.b))
54    };
55    let palette = &theme.palette;
56    let on_paper =
57        |color: ravenrs_core::Color| color.with_alpha(1.0).contrast_ratio(ravenrs_core::Color::WHITE) >= 4.5;
58    let light = on_paper(palette.ink);
59    WizardOptions {
60        accent: hex(palette.accent),
61        on_accent: hex(palette.on_accent),
62        ink: if light { hex(palette.ink) } else { options.ink.clone() },
63        muted: if light && on_paper(palette.ink_muted) {
64            hex(palette.ink_muted)
65        } else {
66            options.muted.clone()
67        },
68        zebra: if light { hex(palette.bg_300) } else { options.zebra.clone() },
69        ..options
70    }
71}

73/// `table` laid out as a document, the same one the files are made from, in the colors of
74/// RavenRS. [`document_of_with`] uses those of a theme.
75pub fn document_of(table: &GridTable) -> Document {
76    document_with(table, WizardOptions::default())
77}
78
79/// `table` laid out as a document in the colors of `theme`, as [`paper_colors`] puts them.
80pub fn document_of_with(table: &GridTable, theme: &Theme) -> Document {
81    document_with(table, paper_colors(theme, WizardOptions::default()))
82}
83
84fn document_with(table: &GridTable, colors: WizardOptions) -> Document {
85    let rows: Vec<Json> = table
86        .rows
87        .iter()
88        .map(|row| {
89            let cells = row.iter().enumerate().map(|(c, value)| (field(c), json_of(value)));
90            Json::Object(cells.collect::<Map<_, _>>())
91        })
92        .collect();
93    let mut columns = Vec::with_capacity(table.columns.len());
94    let mut words = Vec::with_capacity(table.columns.len());
95    let mut totals = Vec::new();
96    for (c, column) in table.columns.iter().enumerate() {
97        let (kind, pattern) = match &column.kind {
98            GridTableKind::Number { decimals, prefix, suffix } => {
99                (FieldKind::Number, Some(number_pattern(*decimals, prefix, suffix)))
100            }
101            GridTableKind::Date => (FieldKind::Date, None),
102            GridTableKind::Boolean => (FieldKind::Boolean, None),
103            _ => (FieldKind::Text, None),
104        };
105        columns.push(FieldInfo { path: field(c), kind });
106        words.push(ColumnWords { caption: Some(column.title.clone()), format: pattern });
107        if column.total {
108            totals.push(field(c));
109        }
110    }
111    let options = WizardOptions {
112        title: table.title.clone(),
113        culture: table.culture.clone(),
114        columns,
115        group_by: table.group_by.map(field),
116        totals,
117        ..colors
118    };
119    let data = Json::Array(rows);
120    let source = DataSource { name: "rows".into(), kind: DataSourceKind::Json { data: data.clone() } };
121    let report = table_report_with(source, &options, &words);
122    ReportEngine::new().render(&report, &data)
123}

125/// Writes `table` in `format`, in the colors of `theme`, as [`export`] does.
126pub fn export_with(
127    table: &GridTable,
128    format: GridExportFormat,
129    suffix: Option<&str>,
130    theme: &Theme,
131) -> Result<ExportedFile, String> {
132    file_of(&document_of_with(table, theme), format, suffix)
133}
134
135fn file_of(
136    document: &Document,
137    format: GridExportFormat,
138    suffix: Option<&str>,
139) -> Result<ExportedFile, String> {
140    let format = match format {
141        GridExportFormat::Xlsx => ExportFormat::Xlsx,
142        GridExportFormat::Pdf => ExportFormat::Pdf,
143        _ => ExportFormat::Csv,
144    };
145    ravenrs_report_export::export(document, format, suffix)
146}
147
148/// Writes `table` in `format`. `suffix`, such as a date, goes after the title in the name.
149pub fn export(
150    table: &GridTable,
151    format: GridExportFormat,
152    suffix: Option<&str>,
153) -> Result<ExportedFile, String> {
154    file_of(&document_of(table), format, suffix)
155}

157#[cfg(test)]
158#[allow(clippy::expect_used)]
159mod tests {
160    use ravenrs_widgets::grid::{GridQuery, to_table};
161    use ravenrs_widgets::{Date, GridColumn, Summary};
162
163    use super::*;
164
165    fn sample() -> GridTable {
166        let columns = [
167            GridColumn::text("Conta. Principal"),
168            GridColumn::currency("Valor", "R$ ", 2).summary(Summary::Sum),
169            GridColumn::date("Data"),
170            GridColumn::text("Oculta").hidden(),
171        ];
172        let date = Date::new(2026, 11, 2).expect("a date");
173        let rows = vec![
174            vec!["Ana".into(), 1234.5.into(), Value::Date(date), "x".into()],
175            vec!["Bia".into(), 10.0.into(), Value::Date(date), "y".into()],
176        ];
177        let query = GridQuery {
178            hidden: columns.iter().map(|c| c.title() == "Oculta").collect(),
179            ..GridQuery::default()
180        };
181        to_table("Contas", "pt-BR", &columns, &rows, &query, &[1, 0])
182    }
183
184    #[test]
185    fn the_table_is_what_the_grid_shows() {
186        let table = sample();
187        assert_eq!(table.columns.len(), 3, "the hidden column stays out");
188        assert_eq!(table.rows[0][0], Value::from("Bia"), "in the grid's order");
189        assert!(table.columns[1].total);
190        assert_eq!(table.columns[2].kind, GridTableKind::Date);
191    }
192
193    #[test]
194    fn numbers_keep_their_text_around_them() {
195        assert_eq!(number_pattern(2, "R$ ", ""), "R$ #,##0.00");
196        assert_eq!(number_pattern(0, "", " un"), "#,##0 un");
197    }
198
199    fn fills(document: &Document) -> Vec<ravenrs_core::Color> {
200        let items = document.pages.iter().flat_map(|page| page.items.iter());
201        items
202            .filter_map(|item| if let ravenrs_report::Item::Rect { fill, .. } = item { *fill } else { None })
203            .collect()
204    }
205
206    #[test]
207    fn the_headers_take_the_theme_s_accent() {
208        let accent = ravenrs_core::Color::hex(0x0f5e66);
209        let mut theme = Theme { name: "Morning Star", ..Theme::raven_light() };
210        theme.palette.accent = accent;
211        let close = |a: ravenrs_core::Color, b: ravenrs_core::Color| {
212            (a.r - b.r).abs() < 0.01 && (a.g - b.g).abs() < 0.01 && (a.b - b.b).abs() < 0.01
213        };
214        assert!(fills(&document_of_with(&sample(), &theme)).iter().any(|fill| close(*fill, accent)));
215        let raven = ravenrs_core::Color::hex(0x3f57db);
216        assert!(
217            fills(&document_of(&sample())).iter().any(|fill| close(*fill, raven)),
218            "without a theme, as before"
219        );
220    }
221
222    #[test]
223    fn a_dark_theme_keeps_dark_text_on_white_paper() {
224        let colors = paper_colors(&Theme::raven_dark(), WizardOptions::default());
225        assert_eq!(colors.ink, WizardOptions::default().ink);
226        assert_eq!(colors.zebra, WizardOptions::default().zebra);
227        let light = paper_colors(&Theme::raven_light(), WizardOptions::default());
228        assert_ne!(light.zebra, WizardOptions::default().zebra, "a light theme's own ground");
229        assert_eq!(light.on_accent, "#ffffff");
230    }
231
232    #[test]
233    fn every_format_makes_a_file() {
234        let table = sample();
235        let pdf = export(&table, GridExportFormat::Pdf, None).expect("a pdf");
236        assert_eq!(pdf.name, "Contas.pdf");
237        assert!(pdf.bytes.starts_with(b"%PDF"));
238        let xlsx = export(&table, GridExportFormat::Xlsx, None).expect("a workbook");
239        assert!(xlsx.bytes.starts_with(b"PK"), "a workbook is a zip");
240        let csv = export(&table, GridExportFormat::Csv, None).expect("a csv");
241        let text = String::from_utf8(csv.bytes).expect("text");
242        assert!(text.contains("CONTA. PRINCIPAL"), "a title with a dot is a caption: {text}");
243        assert!(text.contains("Bia"));
244        assert!(text.contains("R$ 1.234,50"), "the column's format, in pt-BR: {text}");
245    }
