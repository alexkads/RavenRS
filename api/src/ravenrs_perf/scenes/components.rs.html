14fn common(interaction: &str, a: Point, b: Point) -> Step {
15    match interaction {
16        "hover" => hover_between(a, b),
17        "scroll" => scroll_at(Point::new(640.0, 400.0)),
18        "resize" => resize(),
19        _ => nothing(),
20    }
21}

42fn scheduler(interaction: &str) -> Setup {
43    let Some(start) = Date::new(2026, 11, 1) else { panic!("a valid date") };
44    let axis = State::new(TimeAxis::days(start, 30));
45    let resources: Vec<Resource> = (0..200).map(|i| Resource::new(i, format!("Resource {i}"))).collect();
46    let events: Vec<SchedulerEvent> = (0..5_000u64)
47        .map(|i| {
48            let from = TimePoint::from_date(start).plus_days((i % 28) as i64);
49            SchedulerEvent::new(i, i % 200, from, from.plus_days(1), format!("Event {i}"))
50        })
51        .collect();
52    let view = Scheduler::new(axis.binding()).resources(resources).events(events);
53    Setup {
54        root: Box::new(view),
55        step: common(interaction, Point::new(500.0, 200.0), Point::new(700.0, 300.0)),
56    }
57}

59fn chart_values(shift: u32) -> Vec<f64> {
60    (0..5000).map(|i| f64::from((i + shift) % 97) + f64::from(i) * 0.02).collect()

63/// The window dragged narrower, a new width each frame, as nothing kept can answer for.
64fn chart_drag() -> Step {
65    Box::new(|ui, round| {
66        let width = VIEWPORT.width - (round % 400) as f32;
67        ui.set_viewport(Size::new(width, VIEWPORT.height), SCALE);
68        true
69    })
70}

73    match interaction {

77        }),
78        "resize" => chart_drag(),

220/// A drag that goes back and forth, from the first round on.
221fn dragging(from: Point, to: Point) -> Step {
222    Box::new(move |ui, round| {
223        if round == 0 {
224            ui.handle_event(&Event::PointerMove { position: from });
225            return ui.handle_event(&Event::PointerDown { position: from, button: PointerButton::Primary });
226        }
227        let position = if round % 2 == 0 { from } else { to };
228        ui.handle_event(&Event::PointerMove { position })
229    })
230}

248/// The instruments together, around a stage that follows the same view: every drag
249/// rebuilds this screen, which is what the counters answer for.
250fn image_tools(interaction: &str) -> Setup {
251    let Some(picture) =
252        ImageData::from_fn(256, 256, |x, y| Color::rgb(x as f32 / 256.0, y as f32 / 256.0, 0.4))
253    else {
254        panic!("a valid image")
255    };
256    let hill = |center: f32| -> Vec<f32> {
257        (0..256).map(|i| 500.0 * (-((i as f32 - center) / 40.0).powi(2)).exp() + 10.0).collect()
258    };
259    let channels = [
260        HistogramChannel::new("Red", Color::rgb(0.9, 0.2, 0.2), hill(90.0)),
261        HistogramChannel::new("Green", Color::rgb(0.2, 0.8, 0.2), hill(130.0)),
262        HistogramChannel::new("Blue", Color::rgb(0.2, 0.3, 0.9), hill(170.0)),
263    ];
264    let view = State::new(ZoomView::at(1.0));
265    let points =
266        State::new(vec![Point::ZERO, Point::new(0.3, 0.2), Point::new(0.7, 0.8), Point::new(1.0, 1.0)]);
267    let tint = State::new(HueSaturation::new(30.0, 0.5));
268    let panel = VStack::new((
269        Histogram::new(channels.clone()).height(120.0),
270        CurveEditor::new(points.binding()).size(320.0, 220.0).histogram(channels, HistogramScale::Linear),
271        ColorWheel::new(tint.binding()).diameter(180.0),
272        ZoomControl::new(view.binding()),
273    ))
274    .spacing(8.0)
275    .width(320.0);
276    let stage = VStack::new((
277        Ruler::horizontal().scale(view.get().scale).whole_units(),
278        HStack::new((
279            ZoomPanView::image(view.binding(), picture.clone()),
280            Minimap::image(view.binding(), picture),
281        ))
282        .grow(1.0),
283    ))
284    .grow(1.0);
285    let step: Step = match interaction {
286        // The handle of the second point, and the disc.
287        "curve" => dragging(Point::new(102.0, 287.0), Point::new(120.0, 260.0)),
288        "wheel" => dragging(Point::new(130.0, 470.0), Point::new(90.0, 440.0)),
289        other => common(other, Point::new(100.0, 60.0), Point::new(700.0, 300.0)),
290    };
291    Setup { root: Box::new(HStack::new((panel, stage)).grow(1.0)), step }
292}

294/// What is drawn over a picture, each dragged alone: nothing but the paint may run, except a
295/// protractor, which writes its value as it goes. Idle and hover see them all together.
296fn picture_overlays(interaction: &str) -> Setup {
297    let picture = |seed: f32| {
298        ImageData::from_fn(1024, 640, |x, y| Color::rgb(x as f32 / 1024.0, y as f32 / 640.0, seed))
299    };
300    let (Some(first), Some(second)) = (picture(0.3), picture(0.6)) else { panic!("valid images") };
301    let area = Rect::new(200.0, 120.0, 500.0, 300.0);
302    let stage = || Image::new(first.clone()).size(1024.0, 640.0);
303    let (from, to) = (Point::new(300.0, 200.0), Point::new(360.0, 250.0));
304    let selection = State::new(Some(area));
305    let band = RubberBand::new(selection.binding());
306    let held = State::new(BoxTransform::new(area).rotated(12.0));
307    let handles = TransformHandles::new(held.binding());
308    let cropped = State::new(CropArea::new(area).rotated(-6.0));
309    let crop = CropOverlay::new(cropped.binding());
310    let angle = State::new(5.0_f32);
311    let dial = Protractor::new(angle.binding());
312    let compare = CompareView::new(first.clone(), second.clone()).mode(CompareMode::Divider);
313    let (root, step): (AnyView, Step) = match interaction {
314        "band" => (Box::new(band.stage(stage())), dragging(from, to)),
315        "handles" => (Box::new(handles.stage(stage())), dragging(from, to)),
316        "crop" => (Box::new(crop.stage(stage())), dragging(from, to)),
317        "divider" => (Box::new(compare), dragging(from, to)),
318        "protractor" => (Box::new(dial), dragging(Point::new(60.0, 20.0), Point::new(200.0, 20.0))),
319        other => {
320            let all = ZStack::new((
321                stage(),
322                RuleOfThirdsGrid::new().size(1024.0, 640.0),
323                Marquee::rect(area).animated(false).size(1024.0, 640.0),
324                handles.size(1024.0, 640.0),
325            ));
326            (Box::new(VStack::new((all, dial)).spacing(8.0)), common(other, from, Point::new(900.0, 600.0)))
327        }
328    };
329    Setup { root, step }
330}

332/// Every place of a dock area in use, with splits inside: a tab changes, and a bar is dragged.
333fn docking(interaction: &str) -> Setup {
334    const IDS: [(&str, &str); 8] = [
335        ("photo", "Photo"),
336        ("grid", "Grid"),
337        ("props", "Properties"),
338        ("folders", "Folders"),
339        ("search", "Search"),
340        ("layers", "Layers"),
341        ("history", "History"),
342        ("log", "Log"),
343    ];
344    let panels = IDS.iter().map(|(id, title)| DockPanel::new(*id, *title)).collect();
345    let layout = State::new(
346        DockLayout::new()
347            .center(DockNode::horizontal([DockNode::tabs(["photo", "grid"]), DockNode::tabs(["props"])]))
348            .left(DockNode::tabs(["folders", "search"]), 280.0)
349            .right(DockNode::vertical([DockNode::tabs(["layers"]), DockNode::tabs(["history"])]), 280.0)
350            .bottom(DockNode::tabs(["log"]), 160.0),
351    );
352    let area = DockArea::new(layout.binding(), panels, |id| -> AnyView {
353        let lines: Vec<Text> = (0..12).map(|line| Text::new(format!("{id}, line {line}"))).collect();
354        Box::new(VStack::new(lines).spacing(4.0))
355    });
356    // The bar of the left dock: past its strip and its 280 logical pixels.
357    let (bar, further) = (Point::new(297.0, 300.0), Point::new(340.0, 300.0));
358    let step: Step = match interaction {
359        "tab" => clicking("Grid", "Photo"),
360        "divider" => dragging(bar, further),
361        other => common(other, Point::new(120.0, 16.0), Point::new(600.0, 16.0)),
362    };
363    Setup { root: Box::new(area), step }
364}

452/// Clicks two buttons in turn, as a person using the screen does.
453fn clicking(first: &'static str, second: &'static str) -> Step {
454    Box::new(move |ui, round| {
455        let name = if round % 2 == 0 { first } else { second };
456        let tree = ui.access_tree();
457        let found = tree.nodes.iter().find(|node| node.name.as_deref() == Some(name));
458        let Some(position) = found.map(|node| node.bounds.center()) else {
459            panic!("no button {name} on screen");
460        };
461        ui.handle_event(&Event::PointerMove { position });
462        ui.handle_event(&Event::PointerDown { position, button: PointerButton::Primary });
463        ui.handle_event(&Event::PointerUp { position, button: PointerButton::Primary })
464    })
465}

478pub(super) fn scenes() -> Vec<PerfScene> {
479    let scene = |name, covers, class, interactions, make, ceilings| PerfScene {
480        name,
481        covers,
482        class,
483        interactions,
484        make,
485        ceilings,
486    };
487    let lists = &["hover", "scroll", "idle", "resize"][..];
488    const CHART: &[&str] = &["hover", "data", "idle", "resize"];
489    vec![
490        scene(
491            "data-grid",
492            &["DataGrid"][..],
493            Class::Virtualized,
494            lists,
495            data_grid as fn(&str) -> Setup,
496            VIRTUALIZED,
497        ),
498        scene("scheduler", &["Scheduler"], Class::Virtualized, lists, scheduler, VIRTUALIZED),
499        scene("virtual-list", &["VirtualList"], Class::Virtualized, lists, virtual_list, VIRTUALIZED),
500        scene("filmstrip", &["Filmstrip"], Class::Virtualized, lists, filmstrip, VIRTUALIZED),
501        scene("chart", &["Chart"], Class::Picture, CHART, chart, STILL),
502        scene("chart-bars", &["Chart", "BarChart"], Class::Picture, CHART, chart_bars, STILL),
