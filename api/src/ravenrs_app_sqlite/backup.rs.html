1//! Backups of a whole database, and the copies an application keeps by itself.
2//!
3//! A backup is a file of its own, written while the database is in use and checked before it is
4//! kept. Each one carries a stamp: the application, the tenant and the migrations it was made
5//! with. Restoring reads the stamp first and refuses another application, another tenant or other
6//! migrations, then keeps a copy of what is there now, then replaces everything at once. If
7//! anything fails along the way, the database stays as it was.
8//!
9//! ```
10//! use std::rc::Rc;
11//!
12//! use ravenrs_app::{EntityInfo, FieldInfo, FieldKind, ObjectSpace};
13//! use ravenrs_app_sqlite::{BackupStamp, SqliteDatabase, TableMap, SqliteBackups};
14//!
15//! let member = EntityInfo::new("Member")
16//!     .field(FieldInfo::new("id", FieldKind::Integer).key())
17//!     .field(FieldInfo::new("name", FieldKind::Text));
18//! let connection = Rc::new(rusqlite::Connection::open_in_memory()?);
19//! let database = SqliteDatabase::builder(connection).managed(member, TableMap::new("members"))?.build();
20//! let folder = std::env::temp_dir().join(format!("ravenrs-backup-doc-{}", std::process::id()));
21//! # let _ = std::fs::remove_dir_all(&folder);
22//! let backups = database.backups(BackupStamp::new("Church").tenant("central")).copies_in(folder.join("copies"));
23//!
24//! let file = folder.join("today.db");
25//! backups.save(&file)?;
26//! let what = backups.inspect(&file)?;
27//! assert_eq!(what.stamp.tenant.as_deref(), Some("central"));
28//! backups.restore(&file)?;
29//! # std::fs::remove_dir_all(&folder)?;
30//! # Ok::<(), Box<dyn std::error::Error>>(())
31//! ```

41/// The table in a backup that says what it is. Only backups have it.

44/// How many copies stay in the folder unless said otherwise; the oldest go first.
45pub const COPIES_KEPT: usize = 14;

47/// What a backup says about where it came from.
48#[derive(Clone, Debug, PartialEq, Eq)]
49#[non_exhaustive]
50pub struct BackupStamp {
51    /// The application, which a backup has to be of to be restored.
52    pub application: String,
53    /// The tenant, such as one organization of many, which a backup has to be of too.
54    pub tenant: Option<String>,
55    /// The version of the application that made it, to show; it is not compared.
56    pub version: Option<String>,
57    /// The migrations the database had, in order. Filled in when a backup is made.
58    pub migrations: Vec<String>,
59}
60
61impl BackupStamp {
62    /// Backups of this application.
63    pub fn new(application: impl Into<String>) -> Self {
64        Self { application: application.into(), tenant: None, version: None, migrations: Vec::new() }
65    }
66
67    /// Of this tenant only: one of another is refused.
68    pub fn tenant(mut self, tenant: impl Into<String>) -> Self {
69        self.tenant = Some(tenant.into());
70        self
71    }
72
73    /// Made by this version of the application.
74    pub fn version(mut self, version: impl Into<String>) -> Self {
75        self.version = Some(version.into());
76        self
77    }
78}

80/// What a backup holds, read without changing it.
81#[derive(Clone, Debug, PartialEq, Eq)]
82#[non_exhaustive]
83pub struct BackupInfo {
84    /// Where it came from.
85    pub stamp: BackupStamp,
86    /// When it was made, in seconds since 1970.
87    pub made: u64,
88    /// How many rows each table has, by table, without its own bookkeeping.
89    pub rows: Vec<(String, i64)>,
90}

92/// Why a backup was not made, read or restored.
93#[derive(Clone, Debug, PartialEq, Eq)]
94#[non_exhaustive]
95pub enum BackupError {
96    /// The file is the database in use.
97    SameFile,
98    /// Something is already there, and a backup does not overwrite it.
99    Exists(PathBuf),
100    /// The file is not a backup, or not one this library made.
101    NotABackup,
102    /// It is a backup of another application.
103    OtherApplication(String),
104    /// It is a backup of another tenant, or of none when one is expected.
105    OtherTenant(Option<String>),
106    /// It was made with other migrations: an older or a newer version of the application.
107    OtherMigrations,
108    /// A table of this database is missing from it.
109    MissingTable(String),
110    /// It is damaged.
111    Damaged,
112    /// Its rows do not hold together: a reference points at nothing.
113    Broken,
114    /// Reading or writing failed.
115    Storage(String),
116}
117
118impl fmt::Display for BackupError {
119    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
120        match self {
121            Self::SameFile => write!(f, "that is the database in use"),
122            Self::Exists(path) => write!(f, "there is already a file at {}", path.display()),
123            Self::NotABackup => write!(f, "that file is not a backup"),
124            Self::OtherApplication(name) => write!(f, "that is a backup of {name}"),
125            Self::OtherTenant(Some(name)) => write!(f, "that is a backup of {name}"),
126            Self::OtherTenant(None) => write!(f, "that backup is of no one in particular"),
127            Self::OtherMigrations => write!(f, "that backup is of another version of the application"),
128            Self::MissingTable(table) => write!(f, "that backup has no table {table}"),
129            Self::Damaged => write!(f, "that backup is damaged"),
130            Self::Broken => write!(f, "the records of that backup do not hold together"),
131            Self::Storage(error) => write!(f, "{error}"),
132        }
133    }
134}

136impl std::error::Error for BackupError {}

139    BackupError::Storage(error.to_string())

171fn now() -> u64 {
172    SystemTime::now().duration_since(UNIX_EPOCH).map_or(0, |elapsed| elapsed.as_secs())

175/// The tables of a database that hold records, without SQLite's and a backup's own.
176fn tables_of(connection: &Connection, schema: &str) -> rusqlite::Result<Vec<String>> {
177    let mut statement = connection.prepare(&format!(
178        "SELECT name FROM {schema}.sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' \
179         AND name <> '{STAMP}' ORDER BY name"
180    ))?;
181    statement.query_map([], |row| row.get(0))?.collect()
182}
183
184fn columns_of(connection: &Connection, schema: &str, table: &str) -> rusqlite::Result<Vec<String>> {
185    let mut statement = connection.prepare(&format!("PRAGMA {schema}.table_info({})", quoted(table)))?;
186    statement.query_map([], |row| row.get(1))?.collect()
187}
188
189fn migrations_of(connection: &Connection, schema: &str) -> rusqlite::Result<Vec<String>> {
190    let has: bool = connection
191        .prepare(&format!("SELECT 1 FROM {schema}.sqlite_master WHERE type = 'table' AND name = ?1"))?
192        .exists([MIGRATIONS])?;
193    if !has {
194        return Ok(Vec::new());
195    }
196    let mut statement = connection.prepare(&format!("SELECT id FROM {schema}.{MIGRATIONS} ORDER BY id"))?;
197    statement.query_map([], |row| row.get(0))?.collect()
198}

200fn remove_database_files(path: &Path) {
201    for end in ["", "-wal", "-shm", "-journal"] {
202        let mut file = path.as_os_str().to_owned();
203        file.push(end);
204        let _ = std::fs::remove_file(file);
205    }
206}

208/// Backups of a SQLite database.
209pub trait SqliteBackups {
210    /// Backups of this database, stamped with this.
211    fn backups(&self, stamp: BackupStamp) -> Backups;
212}
213
214impl SqliteBackups for SqliteDatabase {
215    fn backups(&self, stamp: BackupStamp) -> Backups {
216        Backups { database: self.clone(), stamp, folder: None, keep: COPIES_KEPT }
217    }
218}

220impl Backups {
221    /// Keeps the copies made before a restore, and by [`keep_copy`](Self::keep_copy), in this
222    /// folder. Without one, a restore keeps no copy.
223    pub fn copies_in(mut self, folder: impl Into<PathBuf>) -> Self {
224        self.folder = Some(folder.into());
225        self
226    }
227
228    /// How many copies stay in the folder; the oldest go first. [`COPIES_KEPT`] unless said.
229    pub fn keep(mut self, copies: usize) -> Self {
230        self.keep = copies.max(1);
231        self
232    }
233
234    fn connection(&self) -> &Connection {
235        self.database.backend().connection()
236    }
237
238    /// The file of the database in use, if it is in one.
239    fn own_file(&self) -> Option<PathBuf> {
240        self.connection().path().filter(|path| !path.is_empty()).map(PathBuf::from)
241    }
242
243    fn is_own(&self, path: &Path) -> bool {
244        let own = self.own_file().and_then(|own| std::fs::canonicalize(own).ok());
245        own.is_some() && own == std::fs::canonicalize(path).ok()
246    }
247
248    /// Writes a backup to `target`, consistent even while the database is in use, and checks it
249    /// before it is there. It never overwrites a file.
250    pub fn save(&self, target: &Path) -> Result<BackupInfo, BackupError> {
251        if self.is_own(target) {
252            return Err(BackupError::SameFile);
253        }
254        if target.exists() {
255            return Err(BackupError::Exists(target.to_owned()));
256        }
257        if let Some(folder) = target.parent().filter(|folder| !folder.as_os_str().is_empty()) {
258            std::fs::create_dir_all(folder).map_err(storage)?;
259        }
260        let mut draft = target.as_os_str().to_owned();
261        draft.push(".parcial");
262        let draft = PathBuf::from(draft);
263        remove_database_files(&draft);
264        let written = self.write_draft(&draft).and_then(|()| self.inspect(&draft));
265        let info = match written {
266            Ok(info) => info,
267            Err(error) => {
268                remove_database_files(&draft);
269                return Err(error);
270            }
271        };
272        std::fs::rename(&draft, target).map_err(|error| {
273            remove_database_files(&draft);
274            storage(error)
275        })?;
276        Ok(info)
277    }
278
279    fn write_draft(&self, draft: &Path) -> Result<(), BackupError> {
280        // A copy of what is committed, whatever is being written meanwhile.
281        self.connection().execute("VACUUM INTO ?1", [draft.to_string_lossy()]).map_err(storage)?;
282        let copy = Connection::open(draft).map_err(storage)?;
283        let stamp = &self.stamp;
284        let migrations = migrations_of(&copy, "main").map_err(storage)?;
285        copy.execute_batch(&format!("CREATE TABLE {STAMP} (key TEXT PRIMARY KEY, value TEXT)"))
286            .map_err(storage)?;
287        let rows: [(&str, Option<String>); 5] = [
288            ("application", Some(stamp.application.clone())),
289            ("tenant", stamp.tenant.clone()),
290            ("version", stamp.version.clone()),
291            ("migrations", Some(migrations.join("\n"))),
292            ("made", Some(now().to_string())),
293        ];
294        for (key, value) in rows {
295            copy.execute(&format!("INSERT INTO {STAMP} (key, value) VALUES (?1, ?2)"), (key, value))
296                .map_err(storage)?;
297        }
298        Ok(())
299    }

301    /// Reads what a backup holds and whether this database can take it, without changing either.
302    pub fn inspect(&self, path: &Path) -> Result<BackupInfo, BackupError> {
303        if self.is_own(path) {
304            return Err(BackupError::SameFile);
305        }
306        let file = Connection::open_with_flags(path, OpenFlags::SQLITE_OPEN_READ_ONLY)
307            .map_err(|_| BackupError::NotABackup)?;
308        let check: String = file
309            .query_row("PRAGMA quick_check(1)", [], |row| row.get(0))
310            .map_err(|_| BackupError::NotABackup)?;
311        if check != "ok" {
312            return Err(BackupError::Damaged);
313        }
314        let read = |key: &str| -> Result<Option<String>, BackupError> {
315            file.query_row(&format!("SELECT value FROM {STAMP} WHERE key = ?1"), [key], |row| row.get(0))
316                .optional()
317                .map_err(|_| BackupError::NotABackup)
318                .map(Option::flatten)
319        };
320        let application = read("application")?.ok_or(BackupError::NotABackup)?;
321        let stamp = BackupStamp {
322            application,
323            tenant: read("tenant")?,
324            version: read("version")?,
325            migrations: read("migrations")?
326                .map(|text| text.lines().filter(|line| !line.is_empty()).map(str::to_owned).collect())
327                .unwrap_or_default(),
328        };
329        if stamp.application != self.stamp.application {
330            return Err(BackupError::OtherApplication(stamp.application));
331        }
332        if stamp.tenant != self.stamp.tenant {
333            return Err(BackupError::OtherTenant(stamp.tenant));
334        }
335        if stamp.migrations != migrations_of(self.connection(), "main").map_err(storage)? {
336            return Err(BackupError::OtherMigrations);
337        }
338        let theirs = tables_of(&file, "main").map_err(storage)?;
339        for table in tables_of(self.connection(), "main").map_err(storage)? {
340            if !theirs.contains(&table) {
341                return Err(BackupError::MissingTable(table));
342            }
343        }
344        let rows = theirs
345            .iter()
346            .filter(|table| table.as_str() != MIGRATIONS)
347            .map(|table| {
348                let count = file.query_row(&format!("SELECT COUNT(*) FROM {}", quoted(table)), [], |row| {
349                    row.get::<_, i64>(0)
350                });
351                count.map(|count| (table.clone(), count)).map_err(storage)
352            })
353            .collect::<Result<_, _>>()?;
354        let made = read("made")?.and_then(|text| text.parse().ok()).unwrap_or(0);
355        Ok(BackupInfo { stamp, made, rows })
356    }

358    /// Replaces everything in the database with the backup at `path`, or changes nothing. A copy
359    /// of what is there now is kept first, in the folder of [`copies_in`](Self::copies_in), and
360    /// returned. The backup itself is left as it was.
361    pub fn restore(&self, path: &Path) -> Result<(BackupInfo, Option<PathBuf>), BackupError> {
362        let info = self.inspect(path)?;
363        let copy = match &self.folder {
364            Some(_) => Some(self.keep_copy("before-restore")?),
365            None => None,
366        };
367        self.take_all_from(path)?;
368        // Keys are counted again from what is there now.
369        self.database.forget_next_keys();
370        Ok((info, copy))
371    }
372
373    fn take_all_from(&self, path: &Path) -> Result<(), BackupError> {
374        let connection = self.connection();
375        connection.execute("ATTACH ?1 AS backup", [path.to_string_lossy()]).map_err(storage)?;
376        // Rows arrive in any order, so references are checked once they are all in.
377        let enforced: i64 =
378            connection.pragma_query_value(None, "foreign_keys", |row| row.get(0)).map_err(storage)?;
379        connection.pragma_update(None, "foreign_keys", false).map_err(storage)?;
380        let result = self.replace_all();
381        let _ = connection.pragma_update(None, "foreign_keys", enforced);
382        let _ = connection.execute("DETACH backup", []);
383        result
384    }
385
386    fn replace_all(&self) -> Result<(), BackupError> {
387        let connection = self.connection();
388        let transaction = connection.unchecked_transaction().map_err(storage)?;
389        for table in tables_of(connection, "main").map_err(storage)? {
390            if table == MIGRATIONS {
391                continue;
392            }
393            let theirs = columns_of(connection, "backup", &table).map_err(storage)?;
394            let shared: Vec<String> = columns_of(connection, "main", &table)
395                .map_err(storage)?
396                .into_iter()
397                .filter(|column| theirs.contains(column))
398                .map(|column| quoted(&column))
399                .collect();
400            let name = quoted(&table);
401            connection.execute(&format!("DELETE FROM main.{name}"), []).map_err(storage)?;
402            if !shared.is_empty() {
403                let columns = shared.join(", ");
404                connection
405                    .execute(
406                        &format!("INSERT INTO main.{name} ({columns}) SELECT {columns} FROM backup.{name}"),
407                        [],
408                    )
409                    .map_err(storage)?;
410            }
411        }
412        let broken = connection
413            .prepare("PRAGMA main.foreign_key_check")
414            .and_then(|mut statement| statement.exists([]))
415            .map_err(storage)?;
416        if broken {
417            // Dropping the transaction rolls it back.
418            return Err(BackupError::Broken);
419        }
420        transaction.commit().map_err(storage)
421    }

423    /// Keeps a copy of the database in the folder of [`copies_in`](Self::copies_in), named for
424    /// when and why, then removes the oldest beyond [`keep`](Self::keep). `reason` is a short
425    /// word of letters, digits and dashes, such as `daily` or `before-erase`.
426    pub fn keep_copy(&self, reason: &str) -> Result<PathBuf, BackupError> {
427        let folder = self.folder.clone().ok_or_else(|| storage("no folder for copies"))?;
428        let reason: String =
429            reason.chars().map(|c| if c.is_ascii_alphanumeric() || c == '-' { c } else { '-' }).collect();
430        let made = now();
431        // Copies of the same second keep the order they were made in.
432        let sequence = self
433            .copies()
434            .iter()
435            .filter(|copy| copy.made == made)
436            .map(|copy| copy.sequence + 1)
437            .max()
438            .unwrap_or(0);
439        let target = folder.join(format!("copy-{made:012}-{sequence:04}-{reason}.db"));
440        self.save(&target)?;
441        self.prune();
442        Ok(target)
443    }
444
445    /// The copies in the folder, newest first.
446    pub fn copies(&self) -> Vec<KeptCopy> {
447        let Some(folder) = &self.folder else { return Vec::new() };
448        let Ok(entries) = std::fs::read_dir(folder) else { return Vec::new() };
449        let mut found: Vec<KeptCopy> = entries
450            .flatten()
451            .filter_map(|entry| {
452                let name = entry.file_name().to_str()?.to_owned();
453                let rest = name.strip_prefix("copy-")?.strip_suffix(".db")?;
454                let (made, rest) = rest.split_once('-')?;
455                let (sequence, reason) = rest.split_once('-')?;
456                Some(KeptCopy {
457                    path: entry.path(),
458                    reason: reason.to_owned(),
459                    made: made.parse().ok()?,
460                    sequence: sequence.parse().ok()?,
461                })
462            })
463            .collect();
464        found.sort_by_key(|copy| std::cmp::Reverse((copy.made, copy.sequence)));
465        found
466    }
467
468    fn prune(&self) {
469        for old in self.copies().into_iter().skip(self.keep) {
470            remove_database_files(&old.path);
471        }
472    }
