1//! Records in SQLite, as an [`ObjectSpace`].
2//!
3//! ```
4//! use std::rc::Rc;
5//!
6//! use ravenrs_app::{EntityInfo, FieldInfo, FieldKind, ObjectSpace, Query, Value};
7//! use ravenrs_app_sqlite::{SqliteDatabase, TableMap};
8//!
9//! let account = EntityInfo::new("Account")
10//!     .field(FieldInfo::new("id", FieldKind::Integer).key())
11//!     .field(FieldInfo::new("name", FieldKind::Text));
12//! let connection = Rc::new(rusqlite::Connection::open_in_memory()?);
13//! let database = SqliteDatabase::builder(connection).managed(account, TableMap::new("accounts"))?.build();
14//!
15//! let mut space = database.space();
16//! let mut object = space.new_object("Account")?;
17//! object.set("name", "Corrente");
18//! space.save(object)?;
19//! space.commit()?;
20//! let found = database.space().query("Account", &Query::all())?;
21//! assert_eq!(found[0].get("name"), Value::Text("Corrente".into()));
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```

40/// How an entity lands on a table.
41#[derive(Clone)]
42pub struct TableMap {
43    table: String,
44    columns: BTreeMap<String, String>,
45    computed: BTreeMap<String, String>,
46    before_save: Option<Hook>,
47    after_load: Option<Hook>,
48}
49
50impl fmt::Debug for TableMap {
51    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52        f.debug_struct("TableMap").field("table", &self.table).finish_non_exhaustive()
53    }
54}
55
56impl TableMap {
57    /// On this table, with each field in a column of its name.
58    pub fn new(table: impl Into<String>) -> Self {
59        Self {
60            table: table.into(),
61            columns: BTreeMap::new(),
62            computed: BTreeMap::new(),
63            before_save: None,
64            after_load: None,
65        }
66    }
67
68    /// This field is in a column of another name.
69    pub fn column(mut self, field: impl Into<String>, column: impl Into<String>) -> Self {
70        self.columns.insert(field.into(), column.into());
71        self
72    }
73
74    /// This field is not stored: the database works it out with this SQL expression over the
75    /// table, which is trusted as written and never comes from people.
76    pub fn computed(mut self, field: impl Into<String>, sql: impl Into<String>) -> Self {
77        self.computed.insert(field.into(), sql.into());
78        self
79    }
80
81    /// Runs on a record just before it is written, to fit what the table asks for.
82    pub fn before_save(mut self, hook: impl Fn(&mut Object) + 'static) -> Self {
83        self.before_save = Some(Rc::new(hook));
84        self
85    }
86
87    /// Runs on a record just after it is read.
88    pub fn after_load(mut self, hook: impl Fn(&mut Object) + 'static) -> Self {
89        self.after_load = Some(Rc::new(hook));
90        self
91    }
92
93    fn column_of<'a>(&'a self, field: &'a str) -> &'a str {
94        self.columns.get(field).map_or(field, String::as_str)
95    }
96}

98/// What to say when the database refuses something, where `{0}` is the field's caption.

154/// A column of a table, as SQLite describes it.
155struct Column {
156    name: String,
157    kind: String,
158    required: bool,
159    key: bool,
160}
161
162fn columns_of(connection: &Connection, table: &str) -> rusqlite::Result<Vec<Column>> {
163    let mut statement = connection.prepare(&format!("PRAGMA table_info({})", quoted(table)))?;
164    let rows = statement.query_map([], |row| {
165        let default: Option<String> = row.get(4)?;
166        let not_null: bool = row.get(3)?;
167        Ok(Column {
168            name: row.get(1)?,
169            kind: row.get::<_, String>(2)?.to_uppercase(),
170            required: not_null && default.is_none(),
171            key: row.get::<_, i64>(5)? > 0,
172        })
173    })?;
174    rows.collect()
175}

177/// `order_items` as `OrderItems`.
178fn entity_name(table: &str) -> String {
179    table
180        .split('_')
181        .map(|part| {
182            let mut letters = part.chars();
183            letters.next().map_or_else(String::new, |first| first.to_uppercase().chain(letters).collect())
184        })
185        .collect()
186}

188/// What a column holds, by the type SQLite was told; nothing for what a field cannot show.
189fn kind_of(declared: &str) -> Option<FieldKind> {
190    if declared.contains("BLOB") {
191        None
192    } else if declared.contains("INT") {
193        Some(FieldKind::Integer)
194    } else if ["REAL", "FLOA", "DOUB", "NUMERIC", "DECIMAL"].iter().any(|name| declared.contains(name)) {
195        Some(FieldKind::Decimal(2))
196    } else if declared.contains("DATETIME") || declared.contains("TIMESTAMP") {
197        Some(FieldKind::DateTime)
198    } else if declared.contains("DATE") {
199        Some(FieldKind::Date)
200    } else if declared.contains("BOOL") {
201        Some(FieldKind::Boolean)
202    } else {
203        Some(FieldKind::Text)
204    }
205}

207/// Entities from the tables a database already has, for an application over a database that came
208/// first: a field per column, a reference per foreign key, and a required field per column that
209/// asks for it. A table with no single key is left out, since a record could not be found. What
210/// comes out is a start: captions and rules are for the application to give.
211pub fn entities_from_schema(connection: &Connection) -> Result<Vec<(EntityInfo, TableMap)>, ModelError> {
212    let wrong = |e: rusqlite::Error| ModelError(e.to_string());
213    let mut statement = connection
214        .prepare(
215            "SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name",
216        )
217        .map_err(wrong)?;
218    let tables: Vec<String> =
219        statement.query_map([], |row| row.get(0)).and_then(Iterator::collect).map_err(wrong)?;
220    let mut found = Vec::new();
221    for table in tables.iter().filter(|table| ![MIGRATIONS, SETTINGS, FILES].contains(&table.as_str())) {
222        let columns = columns_of(connection, table).map_err(wrong)?;
223        if columns.iter().filter(|column| column.key).count() != 1 {
224            continue;
225        }
226        let mut statement =
227            connection.prepare(&format!("PRAGMA foreign_key_list({})", quoted(table))).map_err(wrong)?;
228        let foreign: Vec<(String, String)> = statement
229            .query_map([], |row| Ok((row.get(3)?, row.get(2)?)))
230            .and_then(Iterator::collect)
231            .map_err(wrong)?;
232        let names_a_record = columns
233            .iter()
234            .find(|c| ["name", "title", "description"].contains(&c.name.as_str()))
235            .map(|c| c.name.clone());
236        let mut entity = EntityInfo::new(entity_name(table));
237        for column in &columns {
238            let target = foreign.iter().find(|(from, _)| *from == column.name).map(|(_, to)| to);
239            let kind = match target {
240                Some(to) => FieldKind::Reference(entity_name(to)),
241                None => match kind_of(&column.kind) {
242                    Some(kind) => kind,
243                    None => continue,
244                },
245            };
246            let number = matches!(kind, FieldKind::Integer);
247            let field = FieldInfo::new(&column.name, kind);
248            let field = match (column.key, number) {
249                (true, true) => field.key(),
250                (true, false) => field.typed_key(),
251                _ => field,
252            };
253            let field = if names_a_record.as_deref() == Some(column.name.as_str()) {
254                field.default_property()
255            } else {
256                field
257            };
258            entity = entity.field(field);
259            if column.required && !column.key {
260                entity = entity.rule(ravenrs_app::Rule::required(&column.name));
261            }
262        }
263        found.push((entity, TableMap::new(table)));
264    }
265    Ok(found)
266}

268/// Keeps a layer of the model in the database, under a name, as for one person's settings.
269#[derive(Clone, Debug)]
270pub struct SqliteModelStore {
271    connection: Rc<Connection>,
272    key: String,
273}
274
275impl SqliteModelStore {
276    /// A store under this name, creating where it keeps things if needed.
277    pub fn new(connection: Rc<Connection>, key: impl Into<String>) -> Result<Self, ModelError> {
278        connection
279            .execute_batch(&format!(
280                "CREATE TABLE IF NOT EXISTS {SETTINGS} (key TEXT PRIMARY KEY, value TEXT NOT NULL)"
281            ))
282            .map_err(|e| ModelError(e.to_string()))?;
283        Ok(Self { connection, key: key.into() })
284    }
285}
286
287impl ModelStore for SqliteModelStore {
288    fn load(&self) -> Result<Option<Diff>, ModelError> {
289        let sql = format!("SELECT value FROM {SETTINGS} WHERE key = ?1");
290        let text: Option<String> = self
291            .connection
292            .query_row(&sql, [&self.key], |row| row.get(0))
293            .optional()
294            .map_err(|e| ModelError(e.to_string()))?;
295        text.map(|text| Diff::parse(&text)).transpose()
296    }
297
298    fn save(&self, difference: &Diff) -> Result<(), ModelError> {
299        let sql = format!("INSERT OR REPLACE INTO {SETTINGS} (key, value) VALUES (?1, ?2)");
300        self.connection
301            .execute(&sql, [self.key.as_str(), difference.to_json().as_str()])
302            .map(|_| ())
303            .map_err(|e| ModelError(e.to_string()))
304    }
305}

343struct Entity {
344    info: EntityInfo,
345    map: TableMap,
346}
347
348impl Entity {
349    /// The fields that have a column of their own.
350    fn stored(&self) -> impl Iterator<Item = &FieldInfo> {
351        self.info.fields.iter().filter(|f| {
352            f.calculated.is_none()
353                && !matches!(f.kind, FieldKind::Collection(_))
354                && !self.map.computed.contains_key(&f.name)
355        })
356    }
357
358    /// A select with every field under its own name, so filters and order by field work.
359    fn select(&self) -> String {
360        let mut columns: Vec<String> = self
361            .stored()
362            .map(|f| format!("{} AS {}", quoted(self.map.column_of(&f.name)), quoted(&f.name)))
363            .collect();
364        columns.extend(self.map.computed.iter().map(|(field, sql)| format!("({sql}) AS {}", quoted(field))));
365        format!(
366            "SELECT {} FROM {} AS {}",
367            columns.join(", "),
368            quoted(&self.map.table),
369            quoted(&self.map.table)
370        )
371    }
372
373    fn read(&self, row: &rusqlite::Row<'_>, names: &[String]) -> rusqlite::Result<Object> {
374        let mut values = BTreeMap::new();
375        for (at, name) in names.iter().enumerate() {
376            let kind = self.info.field_named(name).map(|f| &f.kind);
377            let value = match (row.get_ref(at)?, kind) {
378                (ValueRef::Null, _) => Value::Null,
379                (ValueRef::Integer(n), Some(FieldKind::Boolean)) => Value::Bool(n != 0),
380                (ValueRef::Integer(n), _) => Value::Number(n as f64),
381                (ValueRef::Real(n), _) => Value::Number(n),
382                (ValueRef::Text(bytes), kind) => {
383                    let text = String::from_utf8_lossy(bytes).into_owned();
384                    match kind {
385                        Some(FieldKind::Date | FieldKind::DateTime) => {
386                            DateTime::parse(&text).map_or(Value::Text(text), Value::Date)
387                        }
388                        _ => Value::Text(text),
389                    }
390                }
391                (ValueRef::Blob(_), _) => Value::Null,
392            };
393            values.insert(name.clone(), value);
394        }
395        Ok(Object::from_values(&self.info.name, values))
396    }
397}
398
399fn bound(value: &Value) -> SqlValue {
400    match value {
401        Value::Bool(on) => SqlValue::Integer(i64::from(*on)),
402        Value::Number(n) if n.fract() == 0.0 && n.abs() < 9.0e15 => SqlValue::Integer(*n as i64),
403        Value::Number(n) => SqlValue::Real(*n),
404        Value::Text(text) => SqlValue::Text(text.clone()),
405        Value::Date(moment) => SqlValue::Text(moment.to_string()),
406        _ => SqlValue::Null,
407    }
408}

441impl SqliteBuilder {
442    /// What to say when the database refuses something.

448    fn checked(entity: &EntityInfo) -> Result<(), ModelError> {
449        entity.check()
450    }
451
452    /// An entity on a table that exists, whose schema is not touched.
453    pub fn mapped(mut self, entity: EntityInfo, map: TableMap) -> Result<Self, ModelError> {
454        Self::checked(&entity)?;
455        self.entities.push(Entity { info: entity, map });
456        Ok(self)
457    }
458
459    /// An entity on a table that is created if it is missing.
460    pub fn managed(self, entity: EntityInfo, map: TableMap) -> Result<Self, ModelError> {
461        let mine = self.mapped(entity, map)?;
462        let Some(added) = mine.entities.last() else { return Ok(mine) };
463        let columns: Vec<String> = added
464            .stored()
465            .map(|field| {
466                let column = quoted(added.map.column_of(&field.name));
467                let key = if field.is_key { " PRIMARY KEY" } else { "" };
468                format!("{column} {}{key}", sql_type(&field.kind))
469            })
470            .collect();
471        let sql = format!("CREATE TABLE IF NOT EXISTS {} ({})", quoted(&added.map.table), columns.join(", "));
472        mine.connection.execute(&sql, []).map_err(|e| ModelError(e.to_string()))?;
473        // A table that was already there gets the fields added since.
474        let wrong = |e: rusqlite::Error| ModelError(e.to_string());
475        let has = columns_of(&mine.connection, &added.map.table).map_err(wrong)?;
476        for field in added.stored() {
477            let column = added.map.column_of(&field.name);
478            if !has.iter().any(|found| found.name == column) {
479                let add = format!(
480                    "ALTER TABLE {} ADD COLUMN {} {}",
481                    quoted(&added.map.table),
482                    quoted(column),
483                    sql_type(&field.kind)
484                );
485                mine.connection.execute(&add, []).map_err(wrong)?;
486            }
487        }
488        Ok(mine)
489    }
490
491    /// Runs this once in the life of the database, for what adding a column cannot do, such as
492    /// renaming one or moving data. A migration that fails leaves everything as it was.
493    pub fn migration(self, id: &str, sql: &str) -> Result<Self, ModelError> {
494        let wrong = |e: rusqlite::Error| ModelError(format!("the migration {id}: {e}"));
495        let connection = &self.connection;
496        connection
497            .execute_batch(&format!("CREATE TABLE IF NOT EXISTS {MIGRATIONS} (id TEXT PRIMARY KEY)"))
498            .map_err(wrong)?;
499        let done = connection
500            .prepare(&format!("SELECT 1 FROM {MIGRATIONS} WHERE id = ?1"))
501            .and_then(|mut statement| statement.exists([id]))
502            .map_err(wrong)?;
503        if !done {
504            let transaction = connection.unchecked_transaction().map_err(wrong)?;
505            connection.execute_batch(sql).map_err(wrong)?;
506            connection.execute(&format!("INSERT INTO {MIGRATIONS} (id) VALUES (?1)"), [id]).map_err(wrong)?;
507            transaction.commit().map_err(wrong)?;
508        }
509        Ok(self)
510    }

545/// A unit of work over a [`SqliteDatabase`]. A commit is one transaction.

547    shared: Rc<Shared>,
548    environment: Environment,
549    changes: ChangeSet,
550    watchers: Vec<SpaceWatcher>,
551}
552
553impl fmt::Debug for SqliteObjectSpace {
554    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
555        f.debug_struct("SqliteObjectSpace").finish_non_exhaustive()
556    }

559impl SqliteObjectSpace {
560    fn entity(&self, name: &str) -> Result<&Entity, SpaceError> {
561        self.shared
562            .entities
563            .iter()
564            .find(|entity| entity.info.name == name)
565            .ok_or_else(|| SpaceError::UnknownEntity(name.to_owned()))
566    }
567
568    /// Says what the database refused, in words for people.
569    fn refused(&self, entity: &Entity, error: &rusqlite::Error) -> SpaceError {
570        let words = &self.shared.words;
571        let text = error.to_string();
572        if error.sqlite_error_code() != Some(ErrorCode::ConstraintViolation) {
573            return SpaceError::Storage(words.failed.replace("{0}", &text));
574        }
575        if text.contains("FOREIGN KEY") {
576            return SpaceError::Refused(words.in_use.clone());
577        }
578        if text.contains("UNIQUE") {
579            // "UNIQUE constraint failed: accounts.name", possibly of several columns.
580            let caption = text
581                .rsplit(": ")
582                .next()
583                .and_then(|columns| columns.split(", ").next())
584                .and_then(|column| column.rsplit('.').next())
585                .and_then(|column| {
586                    entity.info.fields.iter().find(|f| entity.map.column_of(&f.name) == column)
587                })
588                .map_or_else(String::new, |field| field.caption.clone());
589            return SpaceError::Refused(words.unique.replace("{0}", &caption));
590        }
591        SpaceError::Refused(words.rule.clone())
592    }
593
594    fn select(
595        &self,
596        entity: &Entity,
597        tail: &str,
598        parameters: Vec<SqlValue>,
599    ) -> Result<Vec<Object>, SpaceError> {
600        self.select_columns(entity, "*", tail, parameters)
601    }
602
603    fn select_columns(
604        &self,
605        entity: &Entity,
606        columns: &str,
607        tail: &str,
608        parameters: Vec<SqlValue>,
609    ) -> Result<Vec<Object>, SpaceError> {
610        let sql = format!("SELECT {columns} FROM ({}){tail}", entity.select());
611        let storage = |e: rusqlite::Error| self.refused(entity, &e);
612        let mut statement = self.shared.connection.prepare_cached(&sql).map_err(storage)?;
613        let names: Vec<String> = statement.column_names().map(str::to_owned).collect();
614        let rows = statement
615            .query_map(params_from_iter(parameters), |row| entity.read(row, &names))
616            .map_err(storage)?;
617        let mut found = Vec::new();
618        for row in rows {
619            let mut object = row.map_err(storage)?;
620            entity.info.compute(&mut object, &self.environment);
621            if let Some(hook) = &entity.map.after_load {
622                hook(&mut object);
623            }
624            found.push(object);
625        }
626        Ok(found)
627    }
628
629    fn exists(&self, entity: &Entity, key: &str) -> bool {
630        let Some(field) = entity.info.key_field() else { return false };
631        let sql = format!(
632            "SELECT 1 FROM {} WHERE {} = ?1",
633            quoted(&entity.map.table),
634            quoted(entity.map.column_of(&field.name))
635        );
636        let value = match field.kind {
637            FieldKind::Integer | FieldKind::Decimal(_) => key.parse().map_or(SqlValue::Null, SqlValue::Real),
638            _ => SqlValue::Text(key.to_owned()),
639        };
640        self.shared
641            .connection
642            .prepare_cached(&sql)
643            .and_then(|mut statement| statement.exists([value]))
644            .unwrap_or(false)
645    }
646
647    fn next_key(&self, entity: &Entity, field: &FieldInfo) -> Result<i64, SpaceError> {
648        let sql = format!(
649            "SELECT COALESCE(MAX({}), 0) + 1 FROM {}",
650            quoted(entity.map.column_of(&field.name)),
651            quoted(&entity.map.table)
652        );
653        let stored: i64 = self
654            .shared
655            .connection
656            .query_row(&sql, [], |row| row.get(0))
657            .map_err(|e| self.refused(entity, &e))?;
658        let mut keys = self.shared.next_keys.borrow_mut();
659        let next = keys.get(&entity.info.name).copied().unwrap_or(0).max(stored);
660        keys.insert(entity.info.name.clone(), next + 1);
661        Ok(next)
662    }
663
664    fn write(&self, entity: &Entity, change: &Change) -> Result<(), SpaceError> {
665        let failed = |e: rusqlite::Error| self.refused(entity, &e);
666        let table = quoted(&entity.map.table);
667        let key_field =
668            entity.info.key_field().ok_or_else(|| SpaceError::MissingKey(entity.info.name.clone()))?;
669        let key_column = quoted(entity.map.column_of(&key_field.name));
670        let connection = &self.shared.connection;
671        let prepared = |object: &Object| {
672            let mut object = object.clone();
673            if let Some(hook) = &entity.map.before_save {
674                hook(&mut object);
675            }
676            object
677        };
678        match change {
679            Change::New(object) => {
680                let object = prepared(object);
681                let stored: Vec<&FieldInfo> = entity.stored().collect();
682                let columns: Vec<String> =
683                    stored.iter().map(|f| quoted(entity.map.column_of(&f.name))).collect();
684                let slots: Vec<String> = (1..=stored.len()).map(|at| format!("?{at}")).collect();
685                let sql =
686                    format!("INSERT INTO {table} ({}) VALUES ({})", columns.join(", "), slots.join(", "));
687                let values = stored.iter().map(|f| bound(&object.get(&f.name)));
688                connection.execute(&sql, params_from_iter(values)).map_err(failed)?;
689            }
690            Change::Updated(object) => {
691                let object = prepared(object);
692                let stored: Vec<&FieldInfo> = entity.stored().filter(|f| !f.is_key).collect();
693                let sets: Vec<String> = stored
694                    .iter()
695                    .enumerate()
696                    .map(|(at, f)| format!("{} = ?{}", quoted(entity.map.column_of(&f.name)), at + 1))
697                    .collect();
698                let sql = format!(
699                    "UPDATE {table} SET {} WHERE {key_column} = ?{}",
700                    sets.join(", "),
701                    stored.len() + 1
702                );
703                let mut values: Vec<SqlValue> = stored.iter().map(|f| bound(&object.get(&f.name))).collect();
704                values.push(bound(&object.get(&key_field.name)));
705                let changed = connection.execute(&sql, params_from_iter(values)).map_err(failed)?;
706                if changed == 0 {
707                    return Err(SpaceError::Conflict(format!("a record of {} is gone", entity.info.name)));
708                }
709            }
710            Change::Deleted { key, .. } => {
711                let sql = format!("DELETE FROM {table} WHERE {key_column} = ?1");
712                connection.execute(&sql, [bound(key)]).map_err(failed)?;
713            }
714            _ => {}
715        }
716        Ok(())
717    }
718
719    fn tell(&mut self, event: SpaceEvent<'_>) -> Result<(), SpaceError> {
720        for watcher in &mut self.watchers {
721            watcher(event).map_err(SpaceError::Refused)?;
722        }
723        Ok(())
724    }

727impl ObjectSpace for SqliteObjectSpace {
728    fn entities(&self) -> Vec<EntityInfo> {
729        self.shared.entities.iter().map(|entity| entity.info.clone()).collect()
730    }
731
732    fn query(&self, name: &str, query: &Query) -> Result<Vec<Object>, SpaceError> {
733        let entity = self.entity(name)?;
734        // What waits for a commit, or what SQL cannot say, goes through memory.
735        let translated = query.filter.as_ref().map(|filter| to_sql(filter, Dialect::Sqlite));
736        let in_database = !self.changes.touches(name)
737            && !entity.info.fields.iter().any(|f| f.calculated.is_some())
738            && !matches!(translated, Some(Err(_)));
739        if !in_database {
740            let rows = self.select(entity, "", Vec::new())?;
741            let mut by_key: BTreeMap<String, Object> =
742                rows.into_iter().filter_map(|o| key_of(&entity.info, &o).map(|k| (k, o))).collect();
743            self.changes.overlay(&entity.info, &mut by_key);
744            let mut objects: Vec<Object> = by_key.into_values().collect();
745            for object in &mut objects {
746                entity.info.compute(object, &self.environment);
747            }
748            return run_query(&entity.info, objects, query, &self.environment);
749        }
750        ravenrs_app::check_query(&entity.info, query)?;
751        let mut tail = String::new();
752        let mut parameters = Vec::new();
753        if let Some(Ok(sql)) = translated {
754            tail.push_str(" WHERE ");
755            tail.push_str(&sql.text);
756            for parameter in &sql.params {
757                match parameter {
758                    SqlParam::Value(value) => parameters.push(bound(value)),
759                    _ => {
760                        return Err(SpaceError::BadQuery(
761                            "a filter asks for a value it was not given".into(),
762                        ));
763                    }
764                }
765            }
766        }
767        let key = entity.info.key_field().map(|f| f.name.clone()).unwrap_or_default();
768        let mut order: Vec<String> = query
769            .sort
770            .iter()
771            .map(|(field, ascending)| {
772                format!("{} {}", quoted(field), if *ascending { "ASC" } else { "DESC" })
773            })
774            .collect();
775        order.push(quoted(&key));
776        tail.push_str(&format!(" ORDER BY {}", order.join(", ")));
777        if let Some(limit) = query.limit {
778            tail.push_str(&format!(" LIMIT {limit} OFFSET {}", query.offset));
779        } else if query.offset > 0 {
780            tail.push_str(&format!(" LIMIT -1 OFFSET {}", query.offset));
781        }
782        let columns = query.only.as_ref().map_or_else(
783            || "*".to_owned(),
784            |only| {
785                let mut names: Vec<&str> = only.iter().map(String::as_str).collect();
786                if !names.contains(&key.as_str()) {
787                    names.push(&key);
788                }
789                names.iter().map(|name| quoted(name)).collect::<Vec<_>>().join(", ")
790            },
791        );
792        self.select_columns(entity, &columns, &tail, parameters)
793    }
794
795    fn find(&self, name: &str, key: &Value) -> Result<Option<Object>, SpaceError> {
796        let entity = self.entity(name)?;
797        let Some(text) = key_text(key) else { return Ok(None) };
798        let field = entity.info.key_field().ok_or_else(|| SpaceError::MissingKey(name.to_owned()))?;
799        if self.changes.touches(name) {
800            let mut by_key = BTreeMap::new();
801            let tail = format!(" WHERE {} = ?1", quoted(&field.name));
802            for object in self.select(entity, &tail, vec![bound(key)])? {
803                by_key.insert(text.clone(), object);
804            }
805            self.changes.overlay(&entity.info, &mut by_key);
806            return Ok(by_key.remove(&text).map(|mut object| {
807                entity.info.compute(&mut object, &self.environment);
808                object
809            }));
810        }
811        let tail = format!(" WHERE {} = ?1", quoted(&field.name));
812        Ok(self.select(entity, &tail, vec![bound(key)])?.into_iter().next())
813    }
814
815    fn new_object(&mut self, name: &str) -> Result<Object, SpaceError> {
816        let entity = self.entity(name)?;
817        let mut object = entity.info.new_object();
818        if let Some(field) = entity.info.key_field()
819            && matches!(field.kind, FieldKind::Integer)
820            && matches!(object.get(&field.name), Value::Null)
821        {
822            let next = self.next_key(entity, field)?;
823            object.set(&field.name, Value::Number(next as f64));
824        }
825        Ok(object)
826    }
827
828    fn save(&mut self, object: Object) -> Result<(), SpaceError> {
829        let entity = self.entity(object.entity())?;
830        let info = entity.info.clone();
831        let exists = key_of(&info, &object).is_some_and(|key| self.exists(entity, &key));
832        self.changes.save(&info, object, exists)
833    }
834
835    fn insert(&mut self, object: Object) -> Result<(), SpaceError> {
836        let entity = self.entity(object.entity())?;
837        let info = entity.info.clone();
838        let exists = key_of(&info, &object).is_some_and(|key| self.exists(entity, &key));
839        self.changes.insert(&info, object, exists)
840    }
841
842    fn delete(&mut self, name: &str, key: &Value) -> Result<(), SpaceError> {
843        let entity = self.entity(name)?;
844        let info = entity.info.clone();
845        let exists = key_text(key).is_some_and(|text| self.exists(entity, &text));
846        self.changes.delete(&info, key, exists);
847        Ok(())
848    }
849
850    fn changes(&self) -> Vec<Change> {
851        self.changes.list().to_vec()
852    }
853
854    fn commit(&mut self) -> Result<(), SpaceError> {
855        let changes = self.changes.list().to_vec();
856        self.tell(SpaceEvent::Committing(&changes))?;
857        let shared = self.shared.clone();
858        let transaction =
859            shared.connection.unchecked_transaction().map_err(|e| SpaceError::Storage(e.to_string()))?;
860        for change in &changes {
861            let name = match change {
862                Change::New(object) | Change::Updated(object) => object.entity(),
863                Change::Deleted { entity, .. } => entity.as_str(),
864                _ => continue,
865            };
866            let entity = self.entity(name)?;
867            if let Change::New(new) | Change::Updated(new) = change
868                && !entity.info.state_machines.is_empty()
869            {
870                let field = entity.info.key_field().ok_or_else(|| SpaceError::MissingKey(name.to_owned()))?;
871                let tail = format!(" WHERE {} = ?1", quoted(entity.map.column_of(&field.name)));
872                let stored = self.select(entity, &tail, vec![bound(&new.get(&field.name))])?;
873                ravenrs_app::check_transitions(&entity.info, stored.first(), new)?;
874            }
875            self.write(entity, change)?;
876        }
877        transaction.commit().map_err(|e| SpaceError::Storage(e.to_string()))?;
878        self.changes.clear();
879        let _ = self.tell(SpaceEvent::Committed(&changes));
880        Ok(())
881    }
882
883    fn rollback(&mut self) {
884        self.changes.clear();
885        let _ = self.tell(SpaceEvent::RolledBack);
886    }
887
888    fn on_event(&mut self, watcher: SpaceWatcher) {
889        self.watchers.push(watcher);
890    }
