1//! Records in SQLite, as an [`ObjectSpace`](ravenrs_app::ObjectSpace).
2//!
3//! ```
4//! use std::rc::Rc;
5//!
6//! use ravenrs_app::{EntityInfo, FieldInfo, FieldKind, ObjectSpace, Query, Value};
7//! use ravenrs_app_sqlite::{SqliteDatabase, TableMap};
8//!
9//! let account = EntityInfo::new("Account")
10//!     .field(FieldInfo::new("id", FieldKind::Integer).key())
11//!     .field(FieldInfo::new("name", FieldKind::Text));
12//! let connection = Rc::new(rusqlite::Connection::open_in_memory()?);
13//! let database = SqliteDatabase::builder(connection).managed(account, TableMap::new("accounts"))?.build();
14//!
15//! let mut space = database.space();
16//! let mut object = space.new_object("Account")?;
17//! object.set("name", "Corrente");
18//! space.save(object)?;
19//! space.commit()?;
20//! let found = database.space().query("Account", &Query::all())?;
21//! assert_eq!(found[0].get("name"), Value::Text("Corrente".into()));
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```

50/// A column of a table, as SQLite describes it.
51struct Column {
52    name: String,
53    kind: String,
54    required: bool,
55    key: bool,
56}
57
58fn columns_of(connection: &Connection, table: &str) -> rusqlite::Result<Vec<Column>> {
59    let mut statement = connection.prepare(&format!("PRAGMA table_info({})", quoted(table)))?;
60    let rows = statement.query_map([], |row| {
61        let default: Option<String> = row.get(4)?;
62        let not_null: bool = row.get(3)?;
63        Ok(Column {
64            name: row.get(1)?,
65            kind: row.get::<_, String>(2)?.to_uppercase(),
66            required: not_null && default.is_none(),
67            key: row.get::<_, i64>(5)? > 0,
68        })
69    })?;
70    rows.collect()
71}

73/// `order_items` as `OrderItems`.
74fn entity_name(table: &str) -> String {
75    table
76        .split('_')
77        .map(|part| {
78            let mut letters = part.chars();
79            letters.next().map_or_else(String::new, |first| first.to_uppercase().chain(letters).collect())
80        })
81        .collect()
82}

84/// What a column holds, by the type SQLite was told; nothing for what a field cannot show.
85fn kind_of(declared: &str) -> Option<FieldKind> {
86    if declared.contains("BLOB") {
87        None
88    } else if declared.contains("INT") {
89        Some(FieldKind::Integer)
90    } else if ["REAL", "FLOA", "DOUB", "NUMERIC", "DECIMAL"].iter().any(|name| declared.contains(name)) {
91        Some(FieldKind::Decimal(2))
92    } else if declared.contains("DATETIME") || declared.contains("TIMESTAMP") {
93        Some(FieldKind::DateTime)
94    } else if declared.contains("DATE") {
95        Some(FieldKind::Date)
96    } else if declared.contains("BOOL") {
97        Some(FieldKind::Boolean)
98    } else {
99        Some(FieldKind::Text)
100    }
101}

103/// Entities from the tables a database already has, for an application over a database that came
104/// first: a field per column, a reference per foreign key, and a required field per column that
105/// asks for it. A table with no single key is left out, since a record could not be found. What
106/// comes out is a start: captions and rules are for the application to give.
107pub fn entities_from_schema(connection: &Connection) -> Result<Vec<(EntityInfo, TableMap)>, ModelError> {
108    let wrong = |e: rusqlite::Error| ModelError(e.to_string());
109    let mut statement = connection
110        .prepare(
111            "SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name",
112        )
113        .map_err(wrong)?;
114    let tables: Vec<String> =
115        statement.query_map([], |row| row.get(0)).and_then(Iterator::collect).map_err(wrong)?;
116    let mut found = Vec::new();
117    for table in tables.iter().filter(|table| ![MIGRATIONS, SETTINGS, FILES].contains(&table.as_str())) {
118        let columns = columns_of(connection, table).map_err(wrong)?;
119        if columns.iter().filter(|column| column.key).count() != 1 {
120            continue;
121        }
122        let mut statement =
123            connection.prepare(&format!("PRAGMA foreign_key_list({})", quoted(table))).map_err(wrong)?;
124        let foreign: Vec<(String, String)> = statement
125            .query_map([], |row| Ok((row.get(3)?, row.get(2)?)))
126            .and_then(Iterator::collect)
127            .map_err(wrong)?;
128        let names_a_record = columns
129            .iter()
130            .find(|c| ["name", "title", "description"].contains(&c.name.as_str()))
131            .map(|c| c.name.clone());
132        let mut entity = EntityInfo::new(entity_name(table));
133        for column in &columns {
134            let target = foreign.iter().find(|(from, _)| *from == column.name).map(|(_, to)| to);
135            let kind = match target {
136                Some(to) => FieldKind::Reference(entity_name(to)),
137                None => match kind_of(&column.kind) {
138                    Some(kind) => kind,
139                    None => continue,
140                },
141            };
142            let number = matches!(kind, FieldKind::Integer);
143            let field = FieldInfo::new(&column.name, kind);
144            let field = match (column.key, number) {
145                (true, true) => field.key(),
146                (true, false) => field.typed_key(),
147                _ => field,
148            };
149            let field = if names_a_record.as_deref() == Some(column.name.as_str()) {
150                field.default_property()
151            } else {
152                field
153            };
154            entity = entity.field(field);
155            if column.required && !column.key {
156                entity = entity.rule(ravenrs_app::Rule::required(&column.name));
157            }
158        }
159        found.push((entity, TableMap::new(table)));
160    }
161    Ok(found)
162}

164/// Keeps a layer of the model in the database, under a name, as for one person's settings.
165#[derive(Clone, Debug)]
166pub struct SqliteModelStore {
167    connection: Rc<Connection>,
168    key: String,
169}
170
171impl SqliteModelStore {
172    /// A store under this name, creating where it keeps things if needed.
173    pub fn new(connection: Rc<Connection>, key: impl Into<String>) -> Result<Self, ModelError> {
174        connection
175            .execute_batch(&format!(
176                "CREATE TABLE IF NOT EXISTS {SETTINGS} (key TEXT PRIMARY KEY, value TEXT NOT NULL)"
177            ))
178            .map_err(|e| ModelError(e.to_string()))?;
179        Ok(Self { connection, key: key.into() })
180    }
181}
182
183impl ModelStore for SqliteModelStore {
184    fn load(&self) -> Result<Option<Diff>, ModelError> {
185        let sql = format!("SELECT value FROM {SETTINGS} WHERE key = ?1");
186        let text: Option<String> = self
187            .connection
188            .query_row(&sql, [&self.key], |row| row.get(0))
189            .optional()
190            .map_err(|e| ModelError(e.to_string()))?;
191        text.map(|text| Diff::parse(&text)).transpose()
192    }
193
194    fn save(&self, difference: &Diff) -> Result<(), ModelError> {
195        let sql = format!("INSERT OR REPLACE INTO {SETTINGS} (key, value) VALUES (?1, ?2)");
196        self.connection
197            .execute(&sql, [self.key.as_str(), difference.to_json().as_str()])
198            .map(|_| ())
199            .map_err(|e| ModelError(e.to_string()))
200    }
201}

203/// Keeps the content of attached files in the database, in a table apart from the records.
204#[derive(Clone, Debug)]
205pub struct SqliteAttachmentStore {
206    connection: Rc<Connection>,
207}
208
209impl SqliteAttachmentStore {
210    /// A store on this connection, creating where it keeps things if needed.
211    pub fn new(connection: Rc<Connection>) -> Result<Self, ModelError> {
212        connection
213            .execute_batch(&format!(
214                "CREATE TABLE IF NOT EXISTS {FILES} (id INTEGER PRIMARY KEY, name TEXT NOT NULL, content BLOB NOT NULL)"
215            ))
216            .map_err(|e| ModelError(e.to_string()))?;
217        Ok(Self { connection })
218    }
219}
220
221impl AttachmentStore for SqliteAttachmentStore {
222    fn put(&self, name: &str, bytes: &[u8]) -> Result<String, String> {
223        let sql = format!("INSERT INTO {FILES} (name, content) VALUES (?1, ?2)");
224        self.connection.execute(&sql, rusqlite::params![name, bytes]).map_err(|e| e.to_string())?;
225        Ok(self.connection.last_insert_rowid().to_string())
226    }
227
228    fn get(&self, key: &str) -> Result<Option<Vec<u8>>, String> {
229        let sql = format!("SELECT content FROM {FILES} WHERE id = ?1");
230        self.connection.query_row(&sql, [key], |row| row.get(0)).optional().map_err(|e| e.to_string())
231    }
232
233    fn remove(&self, key: &str) -> Result<(), String> {
234        let sql = format!("DELETE FROM {FILES} WHERE id = ?1");
235        self.connection.execute(&sql, [key]).map(|_| ()).map_err(|e| e.to_string())
236    }
237}

239fn bound(value: &Value) -> SqlValue {
240    match value {
241        Value::Bool(on) => SqlValue::Integer(i64::from(*on)),
242        Value::Number(n) if n.fract() == 0.0 && n.abs() < 9.0e15 => SqlValue::Integer(*n as i64),
243        Value::Number(n) => SqlValue::Real(*n),
244        Value::Text(text) => SqlValue::Text(text.clone()),
245        Value::Date(moment) => SqlValue::Text(moment.to_string()),
246        _ => SqlValue::Null,
247    }
248}

250fn sql_error(error: &rusqlite::Error) -> SqlError {
251    let text = error.to_string();
252    if matches!(error.sqlite_error_code(), Some(ErrorCode::DatabaseBusy | ErrorCode::DatabaseLocked)) {
253        return SqlError::Busy(text);
254    }
255    if error.sqlite_error_code() != Some(ErrorCode::ConstraintViolation) {
256        return SqlError::Other(text);
257    }
258    if text.contains("FOREIGN KEY") {
259        return SqlError::ForeignKey;
260    }
261    if text.contains("UNIQUE") {
262        // "UNIQUE constraint failed: accounts.name", possibly of several columns.
263        let column = text
264            .rsplit(": ")
265            .next()
266            .and_then(|columns| columns.split(", ").next())
267            .and_then(|column| column.rsplit('.').next())
268            .map(str::to_owned);
269        return SqlError::Unique(column);
270    }
271    SqlError::Constraint
272}

274/// A connection to SQLite, which can be shared with the rest of an application.
275#[derive(Clone, Debug)]
276pub struct SqliteBackend(Rc<Connection>);
277
278impl SqliteBackend {
279    /// The connection underneath.
280    pub fn connection(&self) -> &Connection {
281        &self.0
282    }
283}
284
285impl From<Rc<Connection>> for SqliteBackend {
286    fn from(connection: Rc<Connection>) -> Self {
287        Self(connection)
288    }
289}
290
291impl SqlBackend for SqliteBackend {
292    fn dialect(&self) -> Dialect {
293        Dialect::Sqlite
294    }
295
296    fn column_type(&self, kind: &FieldKind) -> Cow<'static, str> {
297        match kind {
298            FieldKind::Integer | FieldKind::Money | FieldKind::Boolean => "INTEGER".into(),
299            // The same binary number `Value::Number` is, so nothing changes on the way; the value
300            // is rounded to its places before it is written.
301            FieldKind::Decimal(_) => "REAL".into(),
302            _ => "TEXT".into(),
303        }
304    }
305
306    fn decimal_upgrade(&self, table: &str, column: &str, places: u8) -> Option<String> {
307        let quoted = |name: &str| format!("\"{}\"", name.replace('"', "\"\""));
308        let (table, column) = (quoted(table), quoted(column));
309        Some(format!(
310            "UPDATE {table} SET {column} = round({column}, {places}) \
311             WHERE typeof({column}) = 'real' AND {column} <> round({column}, {places})"
312        ))
313    }
314
315    fn slot(&self, at: usize) -> String {
316        format!("?{at}")
317    }
318
319    fn offset_only(&self, offset: usize) -> String {
320        format!(" LIMIT -1 OFFSET {offset}")
321    }
322
323    fn columns_of(&self, table: &str) -> Result<Vec<String>, SqlError> {
324        let columns = columns_of(&self.0, table).map_err(|e| sql_error(&e))?;
325        Ok(columns.into_iter().map(|column| column.name).collect())
326    }
327
328    fn execute(&self, sql: &str, values: &[Value]) -> Result<u64, SqlError> {
329        let mut statement = self.0.prepare_cached(sql).map_err(|e| sql_error(&e))?;
330        let changed =
331            statement.execute(params_from_iter(values.iter().map(bound))).map_err(|e| sql_error(&e))?;
332        Ok(changed as u64)
333    }
334
335    fn batch(&self, sql: &str) -> Result<(), SqlError> {
336        self.0.execute_batch(sql).map_err(|e| sql_error(&e))
337    }
338
339    fn query(&self, sql: &str, values: &[Value]) -> Result<SqlRows, SqlError> {
340        let mut statement = self.0.prepare_cached(sql).map_err(|e| sql_error(&e))?;
341        let names: Vec<String> = statement.column_names().map(str::to_owned).collect();
342        let width = names.len();
343        let rows = statement
344            .query_map(params_from_iter(values.iter().map(bound)), |row| {
345                (0..width)
346                    .map(|at| {
347                        Ok(match row.get_ref(at)? {
348                            ValueRef::Null | ValueRef::Blob(_) => SqlCell::Null,
349                            ValueRef::Integer(n) => SqlCell::Integer(n),
350                            ValueRef::Real(n) => SqlCell::Real(n),
351                            ValueRef::Text(bytes) => {
352                                SqlCell::Text(String::from_utf8_lossy(bytes).into_owned())
353                            }
354                        })
355                    })
356                    .collect::<rusqlite::Result<Vec<SqlCell>>>()
357            })
358            .map_err(|e| sql_error(&e))?;
359        let rows = rows.collect::<rusqlite::Result<Vec<_>>>().map_err(|e| sql_error(&e))?;
360        Ok(SqlRows::new(names, rows))
361    }
362}
