29/// Something found at a place in the code: what [`CodeEditor::diagnostics`] marks and
30/// what a `Playground` lists under its editor.
31#[derive(Clone, Debug, PartialEq, Eq)]
32#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
33pub struct CodeDiagnostic {
34    /// The line, from 1.
35    pub line: usize,
36    /// The column of the first character, from 1; 0 marks the whole line.
37    pub column: usize,
38    /// How many characters it covers; 0 marks the whole line.
39    pub length: usize,
40    /// What happened.
41    pub message: String,
42    /// How to fix it, when there is a likely way.
43    pub hint: Option<String>,
44    /// [`Tone::Danger`] for errors, [`Tone::Warning`] for warnings; other tones read as notes.
45    pub tone: Tone,
46}
47
48#[cfg_attr(feature = "live", ravenrs_live::expose)]
49impl CodeDiagnostic {
50    /// An error at `line` and `column` (both from 1), covering `length` characters.
51    pub fn error(line: usize, column: usize, length: usize, message: impl Into<String>) -> Self {
52        Self { line, column, length, message: message.into(), hint: None, tone: Tone::Danger }
53    }
54
55    /// A warning at `line` and `column` (both from 1), covering `length` characters.
56    pub fn warning(line: usize, column: usize, length: usize, message: impl Into<String>) -> Self {
57        Self { tone: Tone::Warning, ..Self::error(line, column, length, message) }
58    }
59
60    /// How to fix it.
61    pub fn hint(mut self, hint: impl Into<String>) -> Self {
62        self.hint = Some(hint.into());
63        self
64    }
65}

67#[doc(hidden)]
68pub fn tone_color(tone: Tone) -> Semantic {
69    match tone {
70        Tone::Danger => Semantic::Danger,
71        Tone::Warning => Semantic::Warning,
72        Tone::Success => Semantic::Success,
73        Tone::Accent => Semantic::Accent,
74        Tone::Neutral => Semantic::InkMuted,
75        // The base may add more.
76        _ => Semantic::InkMuted,
77    }
78}

80/// A source code editor: monospaced, no wrapping, line numbers, colors for the
81/// language, Tab and Shift+Tab to indent, and Enter keeping the indentation. Ctrl or Cmd
82/// with Enter is left to shortcuts, so a surrounding `Playground` can run the
83/// code. Diagnostics mark their lines and underline what they cover. Cmd/Ctrl+Z undoes and
84/// Cmd+Shift+Z or Ctrl+Y redoes. Cmd/Ctrl+F opens a find bar over the code, and Ctrl+H the same
85/// bar with replacing: Enter and Shift+Enter walk the matches, which are marked in the code,
86/// and Escape closes it, back in the code at the match.
87///
88/// ```
89/// use ravenrs_reactive::State;
90/// use ravenrs_widgets::{CodeDiagnostic, CodeEditor, Ui};
91///
92/// let source = State::new("fn main() {\n    println!(\"hi\");\n}".to_owned());
93/// let editor = CodeEditor::new(source.binding())
94///     .diagnostics(vec![CodeDiagnostic::error(2, 5, 8, "unknown macro")])
95///     .rows(8);
96/// let mut ui = Ui::new(editor, ravenrs_style::Theme::raven_dark());
97/// ui.update();
98/// ```
99#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
100pub struct CodeEditor {
101    value: Binding<String>,
102    language: CodeLanguage,
103    line_numbers: bool,
104    diagnostics: Vec<CodeDiagnostic>,
105    indent: usize,
106    rows: usize,
107    fill: bool,
108    read_only: bool,
109    caret: Option<State<Cursor>>,
110    focus_request: Option<State<bool>>,
111    history: Option<State<TextHistory>>,
112    follow_end: bool,
113    on_complete: Option<Rc<dyn Fn(usize)>>,
114    completions: Vec<CodeCompletion>,
115    on_hover: Option<Rc<dyn Fn(usize)>>,
116    hover: Option<String>,
117    on_go_to: Option<Rc<dyn Fn(usize)>>,
118}

294/// What the find bar of a [`CodeEditor`] holds.
295#[derive(Clone)]
296struct Finder {
297    open: State<bool>,
298    replacing: State<bool>,
299    query: State<String>,
300    replacement: State<String>,
301    case: State<bool>,
302    word: State<bool>,
303}
304
305/// The match after (or before, `back`) the caret, going round at the ends.
306fn next_match(found: &[(usize, usize)], cursor: Cursor, back: bool) -> Option<(usize, usize)> {
307    let (from, to) = cursor.selection().unwrap_or((cursor.caret, cursor.caret));
308    if back {
309        found.iter().rev().find(|m| m.0 < from).or(found.last()).copied()
310    } else {
311        found.iter().find(|m| m.0 > from || (m.0 == from && to == from)).or(found.first()).copied()
312    }
313}
314
315/// Selects a match, with the caret at its end.
316fn select(cursor: &State<Cursor>, found: (usize, usize)) {
317    cursor.set(Cursor { caret: found.1, anchor: Some(found.0) });
318}

320impl View for CodeEditor {
321    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
322        let own_cursor: State<Cursor> = cx.use_state(|| Cursor::at(usize::MAX));
323        let cursor = self.caret.clone().unwrap_or(own_cursor);
324        let own_request: State<bool> = cx.use_state(|| false);
325        let focus = self.focus_request.clone().unwrap_or(own_request);
326        let own_history: State<TextHistory> = cx.use_state(TextHistory::new);
327        let history = self.history.clone().unwrap_or(own_history);
328        let finder = Finder {
329            open: cx.use_state(|| false),
330            replacing: cx.use_state(|| false),
331            query: cx.use_state(String::new),
332            replacement: cx.use_state(String::new),
333            case: cx.use_state(|| false),
334            word: cx.use_state(|| false),
335        };
336        let writable = !self.read_only;
337        if cx.is_focus_within() {
338            for (letter, replacing) in [("f", false), ("h", true)] {
339                let (finder, cursor, value) = (finder.clone(), cursor.clone(), self.value.clone());
340                cx.shortcut(
341                    Key::Character(letter.into()),
342                    true,
343                    Rc::new(move || {
344                        // A selection on one line is what is looked for.
345                        let picked = cursor
346                            .get()
347                            .selection()
348                            .map(|(a, b)| value.get().chars().skip(a).take(b - a).collect::<String>());
349                        if let Some(picked) = picked.filter(|p| !p.contains('\n')) {
350                            finder.query.set(picked);
351                        }
352                        finder.replacing.set(replacing && writable);
353                        finder.open.set(true);
354                    }),
355                );
356            }
357        }
358
359        let text = self.value.get();
360        let open = finder.open.get();
361        let query = finder.query.get();
362        let found =
363            if open { find_all(&text, &query, finder.case.get(), finder.word.get()) } else { Vec::new() };
364
365        let highlight: Option<Highlighter> = match self.language {
366            CodeLanguage::Plain => None,
367            CodeLanguage::Rust => Some(Rc::new(highlight_rust)),
368        };
369        let marks: Vec<LineMark> = self
370            .diagnostics
371            .iter()
372            .filter(|d| d.line > 0)
373            .map(|d| {
374                let columns = (d.column > 0 && d.length > 0).then(|| (d.column - 1, d.column - 1 + d.length));
375                (d.line - 1, tone_color(d.tone), columns)
376            })
377            .collect();
378        let look = CodeLook {
379            highlight,
380            line_numbers: self.line_numbers,
381            marks: Rc::new(marks),
382            indent: self.indent,
383            fill: self.fill,
384            background: Semantic::Surface,
385            found: Rc::new(found.clone()),
386            follow_caret: open,
387            follow_end: self.follow_end,
388            on_complete: self.on_complete.clone(),
389            suggestions: Rc::new(
390                self.completions
391                    .iter()
392                    .map(|c| (c.label.clone(), c.detail.clone(), c.insert.clone()))
393                    .collect(),
394            ),
395            on_hover: self.on_hover.clone(),
396            hover: self.hover.clone(),
397            on_go_to: self.on_go_to.clone(),
398        };
399        let area = TextArea::new(self.value.clone())
400            .rows(self.rows)
401            .read_only(self.read_only)
402            .code(look)
403            .cursor(cursor.clone())
404            .focus_request(focus.clone())
405            .history(history.clone());
406        let area = cx.keyed_child(1, &area);
407        let mut children = Vec::new();
408        if open {
409            let bar = FindBar {
410                finder: finder.clone(),
411                found: Rc::new(found),
412                value: self.value.clone(),
413                cursor: cursor.clone(),
414                focus: focus.clone(),
415                history: history.clone(),
416                writable,
417            };
418            children.push(cx.keyed_child(2, &bar));
419        }
420        children.push(area);
421        cx.container(
422            LayoutStyle {
423                grow: if self.fill { 1.0 } else { 0.0 },
424                min_height: Some(0.0),
425                align: Align::Stretch,
426                spacing: 4.0,
427                ..LayoutStyle::stack(Axis::Vertical)
428            },
429            &children,
430            Paint::None,
431        )
432    }
433}

435/// The bar a [`CodeEditor`] shows over its code to find, and to replace.
436struct FindBar {
437    finder: Finder,
438    found: Rc<Vec<(usize, usize)>>,
439    value: Binding<String>,
440    cursor: State<Cursor>,
441    focus: State<bool>,
442    history: State<TextHistory>,
443    writable: bool,
444}
445
446#[cfg_attr(feature = "live", ravenrs_live::expose)]
447impl FindBar {
448    /// Selects the next match, or the one before.
449    fn step(&self) -> impl Fn(bool) + 'static {
450        let (found, cursor) = (self.found.clone(), self.cursor.clone());
451        move |back| {
452            if let Some(m) = next_match(&found, cursor.get(), back) {
453                select(&cursor, m);
454            }
455        }
456    }
457
458    /// Replaces the selected match and goes to the next; with `all`, every match at once.
459    fn replace(&self, all: bool) -> impl Fn() + 'static {
460        let (found, cursor, value, history) =
461            (self.found.clone(), self.cursor.clone(), self.value.clone(), self.history.clone());
462        let replacement = self.finder.replacement.clone();
463        move || {
464            let before = value.get();
465            let with = replacement.get();
466            let chosen: Vec<(usize, usize)> = if all {
467                found.to_vec()
468            } else {
469                match cursor.get().selection() {
470                    Some(sel) if found.contains(&sel) => vec![sel],
471                    // Nothing found is selected: go to the next one first.
472                    _ => {
473                        if let Some(m) = next_match(&found, cursor.get(), false) {
474                            select(&cursor, m);
475                        }
476                        return;
477                    }
478                }
479            };
480            if chosen.is_empty() {
481                return;
482            }
483            let mut after = before.clone();
484            for &(a, b) in chosen.iter().rev() {
485                let (from, to) =
486                    (crate::text_field::byte_index(&after, a), crate::text_field::byte_index(&after, b));
487                after.replace_range(from..to, &with);
488            }
489            history.update(|h| h.record_change(&before, &after, cursor.get()));
490            let (a, b) = chosen[0];
491            let shift = with.chars().count() as isize - (b - a) as isize;
492            value.set(after);
493            if all {
494                cursor.set(Cursor::at(a + with.chars().count()));
495            } else {
496                // The next match, counted in the new text.
497                let next = found
498                    .iter()
499                    .find(|m| m.0 >= b)
500                    .map(|m| ((m.0 as isize + shift) as usize, (m.1 as isize + shift) as usize));
501                match next {
502                    Some(m) => select(&cursor, m),
503                    None => cursor.set(Cursor::at(a + with.chars().count())),
504                }
505            }
506        }
507    }
508}

510impl View for FindBar {
511    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
512        let theme = *cx.theme();
513        let labels = cx.labels().clone();
514        let finder = &self.finder;
515        let replacing = finder.replacing.get() && self.writable;
516        let close: Rc<dyn Fn()> = {
517            let (open, focus) = (finder.open.clone(), self.focus.clone());
518            Rc::new(move || {
519                open.set(false);
520                focus.set(true);
521            })
522        };
523        let step = Rc::new(self.step());
524        let keys: crate::KeyHandler = {
525            let step = step.clone();
526            Rc::new(move |key: &Key, modifiers: Modifiers| {
527                if *key == Key::Enter {
528                    step(modifiers.shift);
529                    return true;
530                }
531                false
532            })
533        };
534        let field = TextField::new(finder.query.binding())
535            .placeholder(labels.code_find.clone())
536            .compact()
537            .width(220.0)
538            .autofocus()
539            .select_on_open()
540            .intercept_keys(keys)
541            .on_cancel({
542                let close = close.clone();
543                move || close()
544            });
545        let current = self.cursor.get().selection().and_then(|sel| self.found.iter().position(|m| *m == sel));
546        let count = if self.found.is_empty() {
547            labels.code_no_results.clone()
548        } else {
549            labels
550                .code_find_count
551                .replace("{current}", &current.map_or("?".to_owned(), |i| (i + 1).to_string()))
552                .replace("{total}", &self.found.len().to_string())
553        };
554        let toggle = |cx: &mut BuildCx<'_>, key: u64, glyph: &str, name: &str, on: &State<bool>| {
555            let flip = on.clone();
556            let button = Button::new(glyph)
557                .accessible_name(name)
558                .style(ButtonStyle::Ghost)
559                .size(ButtonSize::Mini)
560                .selected(on.get())
561                .on_click(move || flip.update(|v| *v = !*v));
562            cx.keyed_child(key, &button)
563        };
564        let mut first_row = Vec::new();
565        if self.writable {
566            let flip = finder.replacing.clone();
567            let chevron = if replacing { IconName::ChevronDown } else { IconName::ChevronRight };
568            let show = IconButton::new(chevron)
569                .label(labels.code_show_replace.clone())
570                .style(ButtonStyle::Ghost)
571                .size(ButtonSize::Mini)
572                .selected(replacing)
573                .on_click(move || flip.update(|v| *v = !*v));
574            first_row.push(cx.keyed_child(10, &show));
575        }
576        first_row.push(cx.keyed_child(11, &field));
577        first_row.push(toggle(cx, 12, "Aa", &labels.code_match_case, &finder.case));
578        first_row.push(toggle(cx, 13, "ab", &labels.code_whole_word, &finder.word));
579        let attrs = util::attrs(&theme, TextStyle::Caption, None);
580        let muted = theme.color(Semantic::InkMuted);
581        let counter =
582            cx.text(&count, &attrs, muted, LayoutStyle { min_width: Some(72.0), ..LayoutStyle::leaf() });
583        first_row.push(counter);
584        for (key, icon, name, back) in [
585            (14, IconName::ArrowUp, labels.code_find_previous.clone(), true),
586            (15, IconName::ArrowDown, labels.code_find_next.clone(), false),
587        ] {
588            let step = step.clone();
589            let button = IconButton::new(icon)
590                .label(name)
591                .style(ButtonStyle::Ghost)
592                .size(ButtonSize::Mini)
593                .disabled(self.found.is_empty())
594                .on_click(move || step(back));
595            first_row.push(cx.keyed_child(key, &button));
596        }
597        let shut = IconButton::new(IconName::X)
598            .label(labels.code_close_find.clone())
599            .style(ButtonStyle::Ghost)
600            .size(ButtonSize::Mini)
601            .on_click(move || close());
602        first_row.push(cx.keyed_child(16, &shut));
603        let row = |cx: &mut BuildCx<'_>, items: &[NodeId]| {
604            cx.container(
605                LayoutStyle { spacing: 4.0, align: Align::Center, ..LayoutStyle::stack(Axis::Horizontal) },
606                items,
607                Paint::None,
608            )
609        };
610        let mut rows = vec![row(cx, &first_row)];
611        if replacing {
612            let swap = TextField::new(finder.replacement.binding())
613                .placeholder(labels.code_replace.clone())
614                .compact()
615                .width(220.0);
616            let one = Button::new(labels.code_replace.clone())
617                .style(ButtonStyle::Ghost)

651/// Where `query` is in `text`, as `(from, to)` char indices that do not overlap: with `case`,
652/// only with the same capitals; with `word`, only where it is not part of a longer word.
653pub fn find_all(text: &str, query: &str, case: bool, word: bool) -> Vec<(usize, usize)> {
654    if query.is_empty() {
655        return Vec::new();
656    }
657    let fold = |c: char| if case { c } else { c.to_lowercase().next().unwrap_or(c) };
658    let hay: Vec<char> = text.chars().map(fold).collect();
659    let needle: Vec<char> = query.chars().map(fold).collect();
660    let is_word = |c: char| c.is_alphanumeric() || c == '_';
661    let mut found = Vec::new();
662    let mut i = 0;
663    while i + needle.len() <= hay.len() {
664        if hay[i..i + needle.len()] == needle[..] {
665            let end = i + needle.len();
666            let whole = !word
667                || (i.checked_sub(1).is_none_or(|b| !is_word(hay[b]))
668                    && hay.get(end).is_none_or(|c| !is_word(*c)));
669            if whole {
670                found.push((i, end));
671                i = end;
672                continue;
673            }
674        }
675        i += 1;
676    }
677    found
678}

686/// The line and the column of the character `at` of `text`, both from 1: what an editor's
687/// status line shows. `at` counts characters, as the caret of a text field does.
688pub fn line_and_column(text: &str, at: usize) -> (usize, usize) {
689    let (mut line, mut column) = (1, 1);
690    for c in text.chars().take(at) {
691        if c == '\n' {
692            line += 1;
693            column = 1;
694        } else {
695            column += 1;
696        }
697    }
698    (line, column)
699}

701/// Where `line` and `column`, both from 1, fall in `text`, counted in characters as
702/// [`line_and_column`] counts them: the other way around. Column 0 is the start of the line,
703/// and a place past the end of a line or of the text stops there.
704pub fn offset_of(text: &str, line: usize, column: usize) -> usize {
705    let mut at = 0;
706    for (n, content) in text.split('\n').enumerate() {
707        let length = content.chars().count();
708        if n + 1 == line.max(1) {
709            return at + column.saturating_sub(1).min(length);
710        }
711        at += length + 1;
712    }
713    text.chars().count()
714}

716/// The colored stretches of Rust code, as char indices. It reads tokens, not syntax, so
717/// it never fails: unfinished strings and comments color to the end.
718pub fn highlight_rust(text: &str) -> Vec<(usize, usize, Semantic)> {
719    let chars: Vec<char> = text.chars().collect();
720    let mut spans = Vec::new();
721    let mut i = 0;
722    let mut attribute_depth = 0usize;
723    while i < chars.len() {
724        let c = chars[i];
725        let start = i;
726        let next = chars.get(i + 1).copied();
727        let semantic = if c == '/' && next == Some('/') {
728            while i < chars.len() && chars[i] != '\n' {
729                i += 1;
730            }
731            Some(Semantic::InkMuted)
732        } else if c == '/' && next == Some('*') {
733            i += 2;
734            while i < chars.len() && !(chars[i] == '*' && chars.get(i + 1) == Some(&'/')) {
735                i += 1;
736            }
737            i = (i + 2).min(chars.len());
738            Some(Semantic::InkMuted)
739        } else if c == '"' {
740            i += 1;
741            while i < chars.len() && chars[i] != '"' {
742                i += if chars[i] == '\\' { 2 } else { 1 };
743            }
744            i = (i + 1).min(chars.len());
745            Some(Semantic::Success)
746        } else if c == '\'' && chars.get(i + 2) == Some(&'\'') {
747            i += 3;
748            Some(Semantic::Success)
749        } else if c == '#' && (next == Some('[') || (next == Some('!') && chars.get(i + 2) == Some(&'['))) {
750            i += if next == Some('!') { 3 } else { 2 };
751            attribute_depth = 1;
752            Some(Semantic::Chart4)
753        } else if attribute_depth > 0 && (c == '[' || c == ']') {
754            attribute_depth = if c == '[' { attribute_depth + 1 } else { attribute_depth - 1 };
755            i += 1;
756            Some(Semantic::Chart4)
757        } else if c.is_ascii_digit() {
758            while i < chars.len() && (chars[i].is_ascii_alphanumeric() || chars[i] == '_' || chars[i] == '.')
759            {
760                if chars[i] == '.' && !chars.get(i + 1).is_some_and(char::is_ascii_digit) {
761                    break;
762                }
763                i += 1;
764            }
765            Some(Semantic::Chart3)
766        } else if c.is_alphabetic() || c == '_' {
767            while i < chars.len() && (chars[i].is_alphanumeric() || chars[i] == '_') {
768                i += 1;
769            }
770            let word: String = chars[start..i].iter().collect();
771            if attribute_depth > 0 {
772                Some(Semantic::Chart4)
773            } else if chars.get(i) == Some(&'!') && chars.get(i + 1) != Some(&'=') {
774                i += 1;
775                Some(Semantic::Chart4)
776            } else if RUST_KEYWORDS.contains(&word.as_str()) {
777                Some(Semantic::FocusRing)
778            } else if word.starts_with(char::is_uppercase) {
779                Some(Semantic::Chart2)
780            } else {
781                None
782            }
783        } else {
784            i += 1;
785            (attribute_depth > 0 && !c.is_whitespace()).then_some(Semantic::Chart4)
786        };
787        if let Some(semantic) = semantic {
788            match spans.last_mut() {
789                Some((_, end, last)) if *end == start && *last == semantic => *end = i,
790                _ => spans.push((start, i, semantic)),
791            }
792        }
793    }
794    spans
795}
