28/// What turns the code of a [`Playground`] into something to show.
29///
30/// It runs when the person asks, never on every frame, so it may take its time and
31/// build whatever it needs anew.
32pub trait Runner {
33    /// The view `source` makes, or where and why it cannot make one.
34    fn run(&self, source: &str) -> Result<AnyView, Vec<CodeDiagnostic>>;
35
36    /// What the code printed, and the problems of what it runs later, such as a closure on a
37    /// click, since the last time this was asked. Nothing, unless the runner keeps them.
38    fn take_output(&self) -> RunOutput {
39        RunOutput::default()
40    }
41
42    /// Calls `wake` whenever there is something new for [`Self::take_output`], so the
43    /// playground shows it as it comes. Nothing, unless the runner can tell.
44    fn watch(&self, wake: Rc<dyn Fn()>) {
45        let _ = wake;
46    }
47}

49/// What a [`Runner`] heard from the code since it was last asked.
50#[derive(Clone, Debug, Default, PartialEq)]
51#[non_exhaustive]
52#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
53pub struct RunOutput {
54    /// Lines it printed, in order.
55    pub lines: Vec<String>,
56    /// Problems of what it ran after the run itself, with their places.
57    pub problems: Vec<CodeDiagnostic>,
58}
59
60#[cfg_attr(feature = "live", ravenrs_live::expose)]
61impl RunOutput {
62    /// Output with these lines and problems.
63    pub fn new(lines: Vec<String>, problems: Vec<CodeDiagnostic>) -> Self {
64        Self { lines, problems }
65    }
66
67    /// Whether there is nothing in it.
68    pub fn is_empty(&self) -> bool {
69        self.lines.is_empty() && self.problems.is_empty()
70    }
71}

73impl<F: Fn(&str) -> Result<AnyView, Vec<CodeDiagnostic>>> Runner for F {
74    fn run(&self, source: &str) -> Result<AnyView, Vec<CodeDiagnostic>> {
75        self(source)
76    }
77}

82/// The name of a file of `source`: Rust after its first function, as `slider.rs`.
83fn file_name_of(source: &str, language: CodeLanguage) -> String {
84    if language == CodeLanguage::Rust {
85        let mut words = source.split(|c: char| !(c.is_alphanumeric() || c == '_'));
86        while let Some(word) = words.next() {
87            if word == "fn"
88                && let Some(name) = words.find(|w| !w.is_empty())
89            {
90                return format!("{name}.rs");
91            }
92        }
93    }
94    "main.rs".to_owned()
95}

139/// A place to write code and see what it makes, for documentation, tutorials and tools.
140///
141/// The code is on one side and the result on the other; below 768 pixels they become two
142/// tabs. Run (or Ctrl/Cmd+Enter) hands the code to the [`Runner`]. When it has problems,
143/// they are listed under the editor and marked on their lines, and the last result that
144/// worked stays, labeled as such. Examples fill the editor, Restore puts the chosen example
145/// back, and the app can add buttons that receive the code, such as one that copies a link.
146///
147/// ```
148/// use ravenrs_widgets::{AnyView, CodeDiagnostic, Playground, Text, Ui};
149///
150/// let runner = |source: &str| -> Result<AnyView, Vec<CodeDiagnostic>> {
151///     match source.trim().strip_prefix("say ") {
152///         Some(words) => Ok(Box::new(Text::new(words.to_owned()))),
153///         None => Err(vec![CodeDiagnostic::error(1, 1, 3, "Start with `say`")]),
154///     }
155/// };
156/// let playground = Playground::new(runner).example("Hello", "say hello").height(400.0);
157/// let mut ui = Ui::new(playground, ravenrs_style::Theme::raven_dark());
158/// ui.update();
159/// ```
160#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
161pub struct Playground {
162    runner: Rc<dyn Runner>,
163    examples: Vec<(String, String)>,
164    source: Option<Binding<String>>,
165    example: Option<Binding<usize>>,
166    language: CodeLanguage,
167    file_name: Option<String>,
168    title: Option<String>,
169    note: Option<String>,
170    extras: Vec<Extra>,
171    download: Option<Download>,
172    height: Option<f32>,
173    output_open: bool,
174}
175
176impl fmt::Debug for Playground {
177    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
178        f.debug_struct("Playground").field("examples", &self.examples.len()).finish_non_exhaustive()
179    }
180}

274    }

277/// What the code said since the last run: the lines it printed and the problems of what ran
278/// later. Kept apart from the reactive state, so reading it while building changes nothing.
279#[derive(Default)]
280struct Log {
281    lines: Vec<String>,
282    problems: Vec<CodeDiagnostic>,
283}
284
285/// Whether the playground is building, and what the code printed while it was.
286#[derive(Default)]
287struct Building {
288    now: std::cell::Cell<bool>,
289    printed: std::cell::Cell<bool>,
290    /// The lines the last build printed, to tell a build that prints the same from one that
291    /// has something new.
292    last: RefCell<Vec<String>>,
293    /// Whether the console already said that the same lines come on every build.
294    warned: std::cell::Cell<bool>,
295}
296
297/// The most lines the console keeps, the oldest going first.
298const MOST_LINES: usize = 500;
299
300#[cfg_attr(feature = "live", ravenrs_live::expose)]
301impl Log {
302    fn add(&mut self, output: RunOutput) {
303        self.lines.extend(output.lines);
304        let over = self.lines.len().saturating_sub(MOST_LINES);
305        self.lines.drain(..over);
306        self.problems.extend(output.problems);
307    }
308}

310impl View for Playground {
311    #[allow(clippy::too_many_lines, reason = "one screen, built top to bottom")]
312    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
313        let labels = cx.labels();
314        let first = self.examples.first().map(|(_, code)| code.clone()).unwrap_or_default();
315        let own_source = cx.use_state(|| first);
316        let source = self.source.clone().unwrap_or_else(|| own_source.binding());
317        let own_example = cx.use_state(|| 0usize);
318        let example = self.example.clone().unwrap_or_else(|| own_example.binding());
319        let result: State<Option<Shown>> = cx.use_state(|| None);
320        let version: State<u64> = cx.use_state(|| 0);
321        let problems: State<Vec<CodeDiagnostic>> = cx.use_state(Vec::new);
322        // 0 code, 1 result, 2 output: the tabs of a narrow playground.
323        let tab = cx.use_state(|| 0usize);
324        // 0 console, 1 problems.
325        let output_tab = cx.use_state(|| 0usize);
326        let open_at_first = self.output_open;
327        let output_open = cx.use_state(move || open_at_first);
328        let log: State<Rc<RefCell<Log>>> = cx.use_state(Rc::default);
329        let heard: State<u64> = cx.use_state(|| 0);
330        // While the playground builds, what the code prints is only kept: a `println!` in a
331        // closure that runs on every build must not ask for one more build each time (#167).
332        let building: State<Rc<Building>> = cx.use_state(Rc::default);
333        let building = building.get();
334        building.now.set(true);
335        let started = cx.use_state(|| false);
336        // Bumped on each request, so the bar redraws even where the screen does not resize.
337        let asked_screen = cx.use_state(|| 0u32);
338        asked_screen.get();
339        let log = log.get();
340
341        let run: Rc<dyn Fn()> = {
342            let (runner, source, result, version, problems) =
343                (self.runner.clone(), source.clone(), result.clone(), version.clone(), problems.clone());
344            let (tab, output_tab, output_open, log, heard) =
345                (tab.clone(), output_tab.clone(), output_open.clone(), log.clone(), heard.clone());
346            Rc::new(move || {
347                // What earlier runs left is not this run's.
348                let _ = runner.take_output();
349                *log.borrow_mut() = Log::default();
350                match runner.run(&source.get()) {
351                    Ok(view) => {
352                        result.set(Some(Shown(Rc::from(view))));
353                        version.update(|v| *v += 1);
354                        problems.set(Vec::new());
355                        if tab.get() == 0 {
356                            tab.set(1);
357                        }
358                        if output_tab.get() == 1 {
359                            output_tab.set(0);
360                        }
361                    }
362                    Err(found) => {
363                        problems.set(found);
364                        output_tab.set(1);
365                        output_open.set(true);
366                    }
367                }
368                log.borrow_mut().add(runner.take_output());
369                heard.update(|n| *n += 1);
370            })
371        };
372        if !started.get() {
373            started.set(true);
374            // What the code prints, or gets wrong later, shows as it comes.
375            let (wake, during) = (heard.clone(), building.clone());
376            self.runner.watch(Rc::new(move || {
377                if during.now.get() {
378                    during.printed.set(true);
379                } else {
380                    wake.update(|n| *n += 1);
381                }
382            }));
383            // An empty kept code starts from the first example too.
384            if source.get().is_empty()
385                && let Some((_, code)) = self.examples.first()
386            {
387                source.set(code.clone());
388            }
389            run();
390        }
391        cx.shortcut(Key::Enter, true, run.clone());
392        // Read so that what is heard builds the playground again; then taken in.
393        let _ = heard.get();
394        let fresh = self.runner.take_output();
395        if !fresh.is_empty() {
396            log.borrow_mut().add(fresh);
397        }
398
399        let narrow = cx.viewport().width < TABS_BREAKPOINT;
400        let found = problems.get();
401        let late = log.borrow().problems.clone();
402        let printed = log.borrow().lines.clone();
403        let all_problems: Vec<CodeDiagnostic> = found.iter().chain(late.iter()).cloned().collect();
404
405        // The examples: picking one puts its code in and runs it.
406        let picker: Option<AnyView> = (self.examples.len() > 1).then(|| {
407            let (examples, chosen, fill, go) =
408                (self.examples.clone(), example.clone(), source.clone(), run.clone());
409            let read = example.clone();
410            let pick = Binding::new(
411                move || read.get(),
412                move |i: usize| {
413                    chosen.set(i);
414                    if let Some((_, code)) = examples.get(i) {
415                        fill.set(code.clone());
416                        go();
417                    }
418                },
419            );
420            Box::new(
421                Select::new(pick, self.examples.iter().map(|(name, _)| name.clone()))
422                    .label(labels.playground_example.clone()),
423            ) as AnyView
424        });
425        let restore: Option<Rc<dyn Fn()>> = (!self.examples.is_empty()).then(|| {
426            let (examples, chosen, fill) = (self.examples.clone(), example.clone(), source.clone());
427            Rc::new(move || {
428                if let Some((_, code)) = examples.get(chosen.get()) {
429                    fill.set(code.clone());
430                }
431            }) as Rc<dyn Fn()>
432        });
433        let download: Option<Rc<dyn Fn()>> = self.download.as_ref().map(|make| {
434            let (make, code, problems, picker) =
435                (make.clone(), source.clone(), problems.clone(), cx.file_picker());
436            let (output_tab, output_open) = (output_tab.clone(), output_open.clone());
437            Rc::new(move || match make(&code.get()) {
438                Ok((name, bytes)) => picker.save(&name, &bytes),
439                Err(found) => {
440                    problems.set(found);
441                    output_tab.set(1);
442                    output_open.set(true);
443                }
444            }) as Rc<dyn Fn()>
445        });
446        let extras: Vec<Bound> = self
447            .extras
448            .iter()
449            .map(|(label, icon, then)| {
450                let (then, code) = (then.clone(), source.clone());
451                (label.clone(), *icon, Rc::new(move || then(&code.get())) as Rc<dyn Fn()>)
452            })
453            .collect();
454        // Full screen, where the platform can give it.
455        let screen = cx.full_screen();
456        let toggle_screen: Option<Bound> = screen.available().then(|| {
457            let (icon, label) = if screen.is_on() {
458                (IconName::Minimize2, labels.playground_exit_full_screen.clone())
459            } else {
460                (IconName::Maximize2, labels.playground_full_screen.clone())
461            };
462            let asked = asked_screen.clone();
463            (
464                label,
465                icon,
466                Rc::new(move || {
467                    screen.toggle();
468                    asked.update(|n| *n = n.wrapping_add(1));
469                }) as Rc<dyn Fn()>,
470            )
471        });
472        let run_button = || {
473            let go = run.clone();
474            Button::new(labels.playground_run.clone())
475                .icon(IconName::Play)
476                .shortcut(if ravenrs_widgets_base::apple_keys() { "⌘↵" } else { "Ctrl+↵" })
477                .on_click(move || go())
478        };
479
480        // The bar. Wide, every action is a button; narrow, the ones besides Run go in a menu.
481        let mut bar: Vec<AnyView> = Vec::new();
482        // Narrow, the examples take the title's place.
483        if let Some(title) = self.title.as_ref().filter(|_| !narrow || self.examples.len() < 2) {
484            bar.push(Box::new(Text::new(title.clone()).style(TextStyle::Headline)));
485        }
486        if narrow {
487            if let Some(picker) = picker {
488                bar.push(Box::new(VStack::new((picker,)).align(Align::Stretch).grow(1.0)));
489            } else {
490                bar.push(Box::new(Spacer::new()));
491            }
492            let mut menu = DropdownMenu::new(labels.playground_more.clone())
493                .icon(IconName::Ellipsis)
494                .icon_only()
495                .style(ButtonStyle::Ghost);
496            for (label, icon, then) in &extras {
497                let then = then.clone();
498                menu = menu.item(MenuItem::new(label.clone()).icon(*icon).on_select(move || then()));
499            }
500            if let Some(save) = &download {
501                let save = save.clone();
502                menu = menu.item(
503                    MenuItem::new(labels.playground_download.clone())
504                        .icon(IconName::Download)
505                        .on_select(move || save()),
506                );
507            }
508            if let Some(restore) = &restore {
509                let restore = restore.clone();
510                menu = menu.item(
511                    MenuItem::new(labels.playground_restore.clone())
512                        .icon(IconName::RotateCcw)
513                        .on_select(move || restore()),
514                );
515            }
516            if let Some((label, icon, toggle)) = &toggle_screen {
517                let toggle = toggle.clone();
518                menu = menu.item(MenuItem::new(label.clone()).icon(*icon).on_select(move || toggle()));
519            }
520            bar.push(Box::new(menu));
521        } else {
522            if let Some(picker) = picker {
523                bar.push(picker);
524            }
525            bar.push(Box::new(Spacer::new()));
526            for (label, icon, then) in &extras {
527                let then = then.clone();
528                bar.push(Box::new(
529                    Button::new(label.clone()).icon(*icon).style(ButtonStyle::Ghost).on_click(move || then()),
530                ));
531            }
532            if let Some(restore) = &restore {
533                let restore = restore.clone();
534                bar.push(Box::new(
535                    Button::new(labels.playground_restore.clone())
536                        .icon(IconName::RotateCcw)
537                        .style(ButtonStyle::Ghost)
538                        .on_click(move || restore()),
539                ));
540            }
541            if let Some(save) = &download {
542                let save = save.clone();
543                bar.push(Box::new(
544                    Button::new(labels.playground_download.clone())
545                        .icon(IconName::Download)
546                        .style(ButtonStyle::Ghost)
547                        .on_click(move || save()),
548                ));
549            }
550            if let Some((label, icon, toggle)) = &toggle_screen {
551                let toggle = toggle.clone();
552                bar.push(Box::new(
553                    IconButton::new(*icon)
554                        .label(label.clone())
555                        .style(ButtonStyle::Ghost)
556                        .on_click(move || toggle()),
557                ));
558            }
559            bar.push(Box::new(run_button()));
560        }
561        let bar = HStack::new(bar)
562            .spacing(8.0)
563            .align(Align::Center)
564            .padding((8.0, 12.0))
565            .background(Semantic::Surface);
566
567        // How the code stands, under the editor and in the narrow footer.
568        let status = status_line(&labels, all_problems.len());
569        // Where the caret is, as the status line of an editor says it (#163, #166).
570        let caret: State<crate::text_field::Cursor> = cx.use_state(Default::default);
571        let (line, column) = crate::line_and_column(&source.get(), caret.get().caret);
572        let position = Text::new(
573            labels
574                .playground_position
575                .replace("{line}", &line.to_string())
576                .replace("{column}", &column.to_string()),
577        )
578        .style(TextStyle::Caption)
579        .foreground(Semantic::InkMuted);
580
581        let spaces = Text::new(labels.playground_spaces.replace("{count}", &INDENT.to_string()))
582            .style(TextStyle::Caption)
583            .foreground(Semantic::InkMuted);
584        let focus: State<bool> = cx.use_state(|| false);
585        // The tab over the editor: the file's name and its language, as in an editor.
586        let name = self.file_name.clone().unwrap_or_else(|| file_name_of(&source.get(), self.language));
587        let language = match self.language {
588            CodeLanguage::Rust => "Rust".to_owned(),
589            _ => labels.playground_plain_text.clone(),
590        };
591        let file_tab = HStack::new((
592            Text::new(name).style(TextStyle::Caption).font_weight(ravenrs_style::Weight::MEDIUM),
593            Spacer::new(),
594            Text::new(language).style(TextStyle::Caption).foreground(Semantic::InkMuted),
595        ))
596        .align(Align::Center)
597        .padding((6.0, 12.0));
598        let code = VStack::new((
599            file_tab,
600            CodeEditor::new(source.clone())
601                .language(self.language)
602                .indent(INDENT)
603                .diagnostics(found.clone())
604                .caret(caret.clone())
605                .focus_request(focus.clone())
606                .fill(),
607            HStack::new((status, Spacer::new(), position, spaces))
608                .spacing(16.0)
609                .align(Align::Center)
610                .padding((6.0, 12.0)),
611        ))
612        .align(Align::Stretch)
613        .grow(1.0)
614        .background(Semantic::Surface)
615        .grow(1.0);
616
617        // The result: the last one that worked, faded and labeled when the code has changed since.
618        let shown = result.get();
619        let mut head: Vec<AnyView> = vec![Box::new(
620            Text::new(labels.playground_result.to_uppercase())
621                .style(TextStyle::Caption)
622                .foreground(Semantic::InkMuted),
623        )];
624        if shown.is_some() && !found.is_empty() {
625            head.push(Box::new(Badge::new(labels.playground_last_good.clone()).tone(Tone::Warning)));
626        }
627        let outcome: AnyView = match shown {
628            Some(view) => {
629                let view = ScrollView::new((Keyed::new(version.get(), view),)).grow(1.0);
630                if found.is_empty() { Box::new(view) } else { Box::new(view.opacity(0.45)) }
631            }
632            None => Box::new(Spacer::new()),
633        };
634        let radius = cx.theme().radii.lg;
635        let outcome = VStack::new((
636            HStack::new(head).spacing(8.0).align(Align::Center),
637            VStack::new((outcome,))
638                .align(Align::Stretch)
639                .grow(1.0)
640                .padding(16.0)
641                .background(Semantic::SurfaceElevated)
642                .corner_radius(radius)
643                .grow(1.0),
644        ))
645        .spacing(8.0)
646        .align(Align::Stretch)
647        .grow(1.0)
648        .padding(12.0)
649        .grow(1.0);
650
651        // The output: what the code printed, and its problems with their places.
652        let console: AnyView = if printed.is_empty() {
653            Box::new(
654                Text::new(labels.playground_nothing_printed.clone())
655                    .style(TextStyle::Callout)
656                    .foreground(Semantic::InkMuted),
657            )
658        } else {
659            let text = printed.join("\n");
660            let shown = Binding::new(move || text.clone(), |_: String| {});
661            Box::new(
662                CodeEditor::new(shown)
663                    .language(CodeLanguage::Plain)
664                    .line_numbers(false)
665                    .read_only(true)
666                    .fill(),
667            )
668        };
669        let problem_list: AnyView = if all_problems.is_empty() {
670            Box::new(
671                Text::new(labels.playground_no_problems.clone())
672                    .style(TextStyle::Callout)
673                    .foreground(Semantic::InkMuted),
674            )
675        } else {
676            let rows: Vec<AnyView> = all_problems
677                .iter()
678                .map(|problem| {
679                    let place = format!(
680                        "{} {}, {} {}",
681                        labels.playground_line, problem.line, labels.playground_column, problem.column
682                    );
683                    let text = match &problem.hint {
684                        Some(hint) => format!("{}. {hint}", problem.message),
685                        None => problem.message.clone(),
686                    };
687                    let tone = if problem.tone == Tone::Warning { Tone::Warning } else { Tone::Danger };
688                    let (to, focus, code) = (caret.clone(), focus.clone(), source.clone());
689                    let (line, column) = (problem.line, problem.column);
690                    Box::new(GoesTo {
691                        name: labels.playground_go_to.replace("{place}", &place),
692                        content: Box::new(Alert::new(place, text).tone(tone)),
693                        go: Rc::new(move || {
694                            to.set(crate::text_field::Cursor::at(crate::offset_of(
695                                &code.get(),
696                                line,
697                                column,
698                            )));
699                            focus.set(true);
700                        }),
701                    }) as AnyView
702                })
703                .collect();
704            Box::new(ScrollView::new((VStack::new(rows).spacing(8.0).align(Align::Stretch),)).grow(1.0))
705        };
706        let output_tabs = SegmentedControl::new(
707            output_tab.binding(),
708            [
709                format!("{} {}", labels.playground_console, printed.len()),
710                format!("{} {}", labels.playground_problems, all_problems.len()),
711            ],
712        )
713        .label(labels.playground_output.clone());
714        let output_body: AnyView = if output_tab.get() == 0 { console } else { problem_list };
715
716        let clear = {
717            let (log, problems, heard) = (log.clone(), problems.clone(), heard.clone());
718            IconButton::new(IconName::Trash2)
719                .label(labels.playground_clear_output.clone())
720                .style(ButtonStyle::Ghost)
721                .on_click(move || {
722                    *log.borrow_mut() = Log::default();
723                    problems.set(Vec::new());
724                    heard.update(|n| *n += 1);
725                })
726        };
727
728        let body: AnyView = if narrow {
729            let tabs = SegmentedControl::new(
730                tab.binding(),
731                [
732                    labels.playground_code.clone(),
733                    labels.playground_result.clone(),
734                    format!("{} {}", labels.playground_output, printed.len() + all_problems.len()),
735                ],
736            );
737            let page: AnyView = match tab.get() {
738                0 => Box::new(code),
739                1 => Box::new(outcome),
740                _ => Box::new(
741                    VStack::new((output_tabs, output_body))
742                        .spacing(8.0)
743                        .align(Align::Stretch)
744                        .grow(1.0)
745                        .padding(12.0)
746                        .grow(1.0),
747                ),
748            };
749            let footer = HStack::new((
750                VStack::new((status_line(&labels, all_problems.len()),)).grow(1.0),
751                run_button(),
752            ))
753            .spacing(8.0)
754            .align(Align::Center)
755            .padding((8.0, 12.0))
756            .background(Semantic::Surface);
757            Box::new(
758                VStack::new((VStack::new((tabs,)).align(Align::Stretch).padding((8.0, 12.0)), page, footer))
759                    .spacing(8.0)
760                    .align(Align::Stretch)
761                    .grow(1.0),
762            )
763        } else {
764            let open = output_open.get();
765            let toggle = {
766                let open_now = output_open.clone();
767                IconButton::new(if open { IconName::ChevronDown } else { IconName::ChevronUp })
768                    .label(if open {
769                        labels.playground_hide_output.clone()
770                    } else {
771                        labels.playground_show_output.clone()
772                    })
773                    .style(ButtonStyle::Ghost)
774                    .on_click(move || open_now.update(|o| *o = !*o))
775            };
776            let output_head =
777                HStack::new((output_tabs, Spacer::new(), clear, toggle)).spacing(4.0).align(Align::Center);
778            let output: AnyView = if open {
779                Box::new(
780                    VStack::new((output_head, output_body))
781                        .spacing(8.0)
782                        .align(Align::Stretch)
783                        .height(OUTPUT_HEIGHT)
784                        .padding(12.0)
785                        .background(Semantic::Surface),
786                )
787            } else {
788                Box::new(
789                    VStack::new((output_head,))
790                        .align(Align::Stretch)
791                        .padding((6.0, 12.0))
792                        .background(Semantic::Surface),
793                )
794            };
795            let right = VStack::new((outcome, Divider::horizontal(), output)).align(Align::Stretch).grow(1.0);
796            Box::new(
797                SplitPanes::horizontal()
798                    .pane(SplitPane::new(code).range(CODE_MIN_WIDTH, f32::MAX))
799                    .pane(
800                        SplitPane::new(right)
801                            .size((cx.viewport().width * RESULT_SHARE).max(RESULT_MIN_WIDTH))
802                            .range(RESULT_MIN_WIDTH, f32::MAX),
803                    )
804                    .grow(1.0),
805            )
806        };
807
808        let mut parts: Vec<AnyView> = vec![Box::new(bar), Box::new(Divider::horizontal())];
809        if let Some(note) = &self.note {
810            parts.push(Box::new(
811                Text::new(note.clone())
812                    .style(TextStyle::Callout)
813                    .foreground(Semantic::InkMuted)
814                    .padding((8.0, 12.0)),
815            ));
816        }
817        parts.push(body);
818        // The stack fills its box, so the body and the narrow footer reach its bottom.
819        let whole = VStack::new(parts).align(Align::Stretch).grow(1.0);
820        let node = match self.height {
821            Some(height) => cx.child(&whole.background(Semantic::Canvas).height(height)),
822            None => cx.child(&whole.background(Semantic::Canvas).grow(1.0)),
823        };
824        building.now.set(false);
825        // What the build printed shows in the next one. Asked for only when it differs from
826        // what the last build printed: the same lines on every build are printed each frame,
827        // and asking again for them would never stop.
828        if building.printed.replace(false) {
829            let fresh = self.runner.take_output();
830            let lines = fresh.lines.clone();
831            let repeated = *building.last.borrow() == lines;
832            if !fresh.is_empty() {
833                log.borrow_mut().add(fresh);
834            }
835            // Said once, so whoever wrote it knows what keeps coming; one more build shows it.
836            let warn = repeated && !lines.is_empty() && !building.warned.replace(true);
837            if warn {
838                log.borrow_mut().add(RunOutput::new(vec![labels.printed_every_build.clone()], Vec::new()));
839            }
840            if !repeated || warn {
841                *building.last.borrow_mut() = lines;
842                heard.update(|n| *n += 1);
843            }
844        }
845        node
846    }
847}

849/// The height of the open output panel.
850const OUTPUT_HEIGHT: f32 = 200.0;
851
852fn status_line(labels: &crate::Labels, count: usize) -> AnyView {
853    if count == 0 {
854        Box::new(
855            HStack::new((
856                Dot::new().tone(Tone::Success),
857                Text::new(labels.playground_no_problems.clone()).style(TextStyle::Callout),
858            ))
859            .spacing(6.0)
860            .align(Align::Center),
861        )
862    } else {
863        Box::new(
864            HStack::new((
865                Dot::new().tone(Tone::Danger),
866                Text::new(format!("{}: {count}", labels.playground_problems)).style(TextStyle::Callout),
867            ))
868            .spacing(6.0)
869            .align(Align::Center),
870        )
871    }
