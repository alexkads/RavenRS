1//! Headless PDF engine for RavenRS.
2//!
3//! The layers, from the bottom up:
4//!
5//! - [`geom`]: rectangles, matrices and quadrilaterals in PDF user space.
6//! - [`crypt`]: the standard security handler: RC4 and AES, passwords and permissions.
7//! - [`filters`]: stream filters: Flate, LZW, ASCII85 and hex, run length, CCITT fax and JBIG2.
8//! - [`cos`]: the object layer: parsing, cross-reference repair, incremental and full saves,
9//!   object streams and encryption.
10//! - [`model`]: pages, the page tree and the document catalogue over `cos`.
11//! - [`content`]: content streams parsed into operators.
12//! - [`fonts`]: font metrics, encodings, shaping and subsetting for text written into a PDF.
13//! - [`render`]: pages rasterized to pixels, text extraction and document inspection.
14//! - [`organize`]: merging, splitting, page order and rotation, attachments and metadata.
15//! - [`third_party`]: the renderer's own stack (hayro, vello_cpu, lopdf) at the revisions
16//!   PdfCraft patches; not an API of this crate.
17//!
18//! ```rust
19//! use std::sync::Arc;
20//! use ravenrs_pdf::cos::Document;
21//! use ravenrs_pdf::render::{PageRenderer, RenderConfig, RenderRequest};
22//!
23//! // A one-page PDF drawing a red square; the reader rebuilds the missing cross-reference table.
24//! let bytes = Arc::new(
25//!     b"%PDF-1.7\n1 0 obj << /Type /Catalog /Pages 2 0 R >> endobj\n\
26//!       2 0 obj << /Type /Pages /Kids [3 0 R] /Count 1 >> endobj\n\
27//!       3 0 obj << /Type /Page /Parent 2 0 R /MediaBox [0 0 100 100] /Contents 4 0 R >> endobj\n\
28//!       4 0 obj << /Length 27 >> stream\n1 0 0 rg 25 25 50 50 re f\nendstream endobj\n\
29//!       trailer << /Root 1 0 R >>\n%%EOF\n"
30//!         .to_vec(),
31//! );
32//!
33//! // Pages to pixels: premultiplied RGBA, one point per pixel at scale 1.
34//! let mut renderer = PageRenderer::new(bytes.clone(), RenderConfig::default());
35//! assert_eq!(renderer.page_count(), 1);
36//! let page = renderer.render(RenderRequest { page: 0, scale: 1.0, ..Default::default() });
37//! assert_eq!((page.width, page.height), (100, 100));
38//! let centre = ((50 * 100 + 50) * 4) as usize;
39//! assert_eq!(&page.rgba[centre..centre + 3], &[255, 0, 0]);
40//!
41//! // The same file twice, as one document of two pages.
42//! let doc = Document::open(bytes)?;
43//! let both = ravenrs_pdf::organize::combine(&[("a.pdf", &doc), ("b.pdf", &doc)])?;
44//! assert_eq!(ravenrs_pdf::organize::page_count(&both)?, 2);
45//! # Ok::<(), Box<dyn std::error::Error>>(())
46//! ```
47//!
48//! Nothing here draws to a window, so it also runs on servers and in WebAssembly. The engine comes
49//! from PdfCraft (MIT OR Apache-2.0); see `NOTICE`.
