11/// An axis-aligned rectangle in PDF user space (points).
12#[derive(Clone, Copy, Debug, PartialEq, Default)]
13pub struct PageRect {
14    pub x0: f32,
15    pub y0: f32,
16    pub x1: f32,
17    pub y1: f32,
18}
19
20impl PageRect {
21    pub fn new(x0: f32, y0: f32, x1: f32, y1: f32) -> Self {
22        Self { x0: x0.min(x1), y0: y0.min(y1), x1: x0.max(x1), y1: y0.max(y1) }
23    }
24    pub fn width(&self) -> f32 {
25        self.x1 - self.x0
26    }
27    pub fn height(&self) -> f32 {
28        self.y1 - self.y0
29    }
30}
31
32/// Page rotation in quarter turns, as in the PDF `/Rotate` key (normalised to 0/90/180/270).
33#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
34pub enum Rotation {
35    #[default]
36    R0,
37    R90,
38    R180,
39    R270,
40}
41
42impl Rotation {
43    pub fn from_degrees(deg: i64) -> Self {
44        match deg.rem_euclid(360) {
45            90 => Self::R90,
46            180 => Self::R180,
47            270 => Self::R270,
48            _ => Self::R0,
49        }
50    }
51    pub fn swaps_axes(self) -> bool {
52        matches!(self, Self::R90 | Self::R270)
53    }
54}

62        let r = PageRect::new(10.0, 20.0, 0.0, 5.0);
63        assert_eq!((r.x0, r.y0, r.width(), r.height()), (0.0, 5.0, 10.0, 15.0));
64    }
65
66    #[test]
67    fn rotation_normalises_negative_and_large_values() {
68        assert_eq!(Rotation::from_degrees(-90), Rotation::R270);
69        assert_eq!(Rotation::from_degrees(450), Rotation::R90);
70        assert!(Rotation::R90.swaps_axes());
