1//! The quadrilaterals of a `/QuadPoints` array (ISO 32000-2 §12.5.6.10), read by one rule.
2//!
3//! Redaction apply, the comment summaries and outlines, the inspector and generated appearances
4//! all read `/QuadPoints`; they go through [`usable_quads`] so the area a user sees marked is the
5//! area apply removes.
6
7/// The usable quadrilaterals of a `/QuadPoints` array, given its entries in order (`None` for an
8/// entry that is not a number).
9///
10/// A quadrilateral is usable only when all eight of its entries are present, numbers, and within
11/// the range of a PDF real (finite as an `f32`, Annex C), so the readers that keep `f32` see the
12/// same set as those that keep `f64`. Usable values come back on the `f32` grid (each one put
13/// through `f32` and back), so the readers fold the same numbers too: a value `f32` cannot hold
14/// exactly, like 67108869, is 67108872 for every reader, not just the ones that keep `f32`. An
15/// unusable one is skipped whole: the ones after it keep their coordinates, and it never becomes
16/// a folded box or zeros. A trailing partial group is dropped.
17pub fn usable_quads<I: IntoIterator<Item = Option<f64>>>(entries: I) -> Vec<[f64; 8]> {
18    let entries: Vec<Option<f64>> = entries.into_iter().collect();
19    entries
20        .as_chunks::<8>()
21        .0
22        .iter()
23        .filter_map(|q| {
24            let mut quad = [0.0; 8];
25            for (slot, v) in quad.iter_mut().zip(q) {
26                // NaN and infinities stay non-finite as `f32`; so does a value past its range.
27                *slot = v.filter(|v| (*v as f32).is_finite()).map(|v| (v as f32) as f64)?;
28            }
29            Some(quad)
30        })
31        .collect()
32}

34#[cfg(test)]
35mod tests {
36    use super::usable_quads;
37
38    const A: [f64; 8] = [10.0, 20.0, 50.0, 20.0, 10.0, 40.0, 50.0, 40.0];
39    const B: [f64; 8] = [60.0, 70.0, 90.0, 70.0, 60.0, 80.0, 90.0, 80.0];
40
41    fn some(v: &[f64]) -> Vec<Option<f64>> {
42        v.iter().copied().map(Some).collect()
43    }
44
45    #[test]
46    fn finite_quads_are_all_usable() {
47        assert_eq!(usable_quads(some(&[A, B].concat())), vec![A, B]);
48        assert!(usable_quads(Vec::new()).is_empty());
49    }
50
51    #[test]
52    fn an_all_nan_quad_is_unusable() {
53        assert!(usable_quads(some(&[f64::NAN; 8])).is_empty());
54    }
55
56    #[test]
57    fn one_non_finite_entry_makes_its_quad_unusable() {
58        for bad in [f64::NAN, f64::INFINITY, f64::NEG_INFINITY, 1e300] {
59            let mut q = A;
60            q[3] = bad;
61            assert_eq!(usable_quads(some(&[q, B].concat())), vec![B], "{bad}");
62        }
63    }
64
65    #[test]
66    fn usable_values_come_back_on_the_f32_grid() {
67        let mut q = A;
68        q[0] = 67_108_869.0; // `f32` cannot hold it exactly: it rounds to 67108872 there.
69        q[1] = 0.1;
70        let got = usable_quads(some(&q));
71        assert_eq!(got.len(), 1);
72        assert_eq!(got[0][0], (67_108_869.0f64 as f32) as f64);
73        assert_eq!(got[0][1], (0.1f64 as f32) as f64);
74        assert_ne!(got[0][0], q[0]);
75    }
76
77    #[test]
78    fn a_non_number_drops_its_quad_without_shifting_the_next() {
79        let mut entries = some(&[A, B].concat());
80        entries[2] = None;
81        assert_eq!(usable_quads(entries), vec![B]);
82    }
83
84    #[test]
85    fn a_trailing_partial_group_is_dropped() {
86        assert_eq!(usable_quads(some(&[&A[..], &B[..4]].concat())), vec![A]);
87    }
88}
