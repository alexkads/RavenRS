9/// Inter Regular, the interface font. Typed signatures fall back to it, slanted, for characters
10/// the script font lacks (Dancing Script is Latin and Vietnamese only: no Cyrillic or Greek).
11pub static INTER_REGULAR: &[u8] = include_bytes!("../../assets/assets/fonts/Inter-Regular.ttf");

13/// Forward slant of fallback glyphs in a typed signature (about 12°), so they lean like the script.
14const FALLBACK_SLANT: f64 = 0.21;

16/// Bound signature work while allowing long personal names.
17pub const MAX_SIGNATURE_CHARS: usize = 256;

19/// Outlines of a line of existing text at a font size of 1 (em units).
20#[derive(Clone, Debug, Default, PartialEq)]
21pub struct ScriptOutline {
22    pub contours: Vec<Vec<[f64; 2]>>,
23    pub width: f64,
24    pub ascent: f64,
25    pub descent: f64,
26}
27
28impl ScriptOutline {
29    /// Layout bounds including glyph overhangs, so previews and PDF appearances never clip ink.
30    pub fn bounds(&self) -> [f64; 4] {
31        self.contours.iter().flatten().fold([0.0, self.descent, self.width, self.ascent], |mut b, p| {
32            b[0] = b[0].min(p[0]);
33            b[1] = b[1].min(p[1]);
34            b[2] = b[2].max(p[0]);
35            b[3] = b[3].max(p[1]);
36            b
37        })
38    }
39}

41/// One glyph outline in em units, with the baseline at y = 0.
42#[derive(Clone, Debug, PartialEq)]
43pub struct GlyphOutline {

45    pub width: f64,
46    pub bbox: [f64; 4],

49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub enum GlyphError {
51    /// This build has no Japanese face: it was built without craft-fonts (`CRAFT_FONTS_DIR`).
52    NoFont,
53    Missing,
54    TooComplex,
55}
56
57struct Flatten {
58    contours: Vec<Vec<[f64; 2]>>,
59    cur: Vec<[f64; 2]>,
60    scale: f64,
61    /// Horizontal shear applied to every point (x += slant × y), for slanted fallback glyphs.
62    slant: f64,
63    dx: f64,
64    points: usize,
65    too_complex: bool,
66    max_points: usize,
67}
68
69impl Flatten {
70    fn new(scale: f64) -> Self {
71        Self { contours: Vec::new(), cur: Vec::new(), scale, slant: 0.0, dx: 0.0, points: 0, too_complex: false, max_points: 4096 }
72    }
73
74    fn pt(&self, x: f32, y: f32) -> [f64; 2] {
75        let (x, y) = (x as f64 * self.scale, y as f64 * self.scale);
76        [self.dx + x + self.slant * y, y]
77    }
78
79    fn last(&self) -> [f64; 2] {
80        self.cur.last().copied().unwrap_or([self.dx, 0.0])
81    }
82
83    fn push(&mut self, p: [f64; 2]) {
84        if self.points >= self.max_points {
85            self.too_complex = true;
86            return;
87        }
88        self.points += 1;
89        self.cur.push(p);
90    }
91
92    fn close(&mut self) {
93        if self.cur.len() > 2 {
94            self.contours.push(std::mem::take(&mut self.cur));
95        } else {
96            self.cur.clear();
97        }
98    }
99}
100
101const STEPS: usize = 8;
102
103impl OutlinePen for Flatten {
104    fn move_to(&mut self, x: f32, y: f32) {
105        self.close();
106        self.push(self.pt(x, y));
107    }
108
109    fn line_to(&mut self, x: f32, y: f32) {
110        self.push(self.pt(x, y));
111    }
112
113    fn quad_to(&mut self, cx: f32, cy: f32, x: f32, y: f32) {
114        let (p0, c, p) = (self.last(), self.pt(cx, cy), self.pt(x, y));
115        for i in 1..=STEPS {
116            let t = i as f64 / STEPS as f64;
117            let u = 1.0 - t;
118            self.push([u * u * p0[0] + 2.0 * u * t * c[0] + t * t * p[0], u * u * p0[1] + 2.0 * u * t * c[1] + t * t * p[1]]);
119        }
120    }
121
122    fn curve_to(&mut self, c0x: f32, c0y: f32, c1x: f32, c1y: f32, x: f32, y: f32) {
123        let (p0, c0, c1, p) = (self.last(), self.pt(c0x, c0y), self.pt(c1x, c1y), self.pt(x, y));
124        for i in 1..=STEPS {
125            let t = i as f64 / STEPS as f64;
126            let u = 1.0 - t;
127            let (a, b, c, d) = (u * u * u, 3.0 * u * u * t, 3.0 * u * t * t, t * t * t);
128            self.push([a * p0[0] + b * c0[0] + c * c1[0] + d * p[0], a * p0[1] + b * c0[1] + c * c1[1] + d * p[1]]);
129        }
130    }
131
132    fn close(&mut self) {
133        Flatten::close(self);
134    }
135}

137/// Whether the face has a glyph for `ch` (a cheap character-map probe, no outline work).
138pub fn face_has(bytes: &[u8], ch: char) -> bool {
139    FontRef::new(bytes).is_ok_and(|font| font.charmap().map(ch).is_some())
140}

162fn glyph_in_face(bytes: &[u8], ch: char) -> Result<GlyphOutline, GlyphError> {
163    let Ok(font) = FontRef::new(bytes) else { return Err(GlyphError::Missing) };
164    let loc = LocationRef::default();
165    let metrics = font.metrics(Size::unscaled(), loc);
166    let scale = 1.0 / metrics.units_per_em.max(1) as f64;
167    let Some(gid) = font.charmap().map(ch) else { return Err(GlyphError::Missing) };
168    let advances = font.glyph_metrics(Size::unscaled(), loc);
169    let width = advances.advance_width(gid).unwrap_or(0.0) as f64 * scale;
170    if !width.is_finite() || width <= 0.0 || width > 2.0 {
171        return Err(GlyphError::Missing);
172    }
173    bounded_outline(&font, gid, width)
174}

176/// Glyph `gid` of `font` in em units with advance `width`, bounded like [`japanese_glyph`].
177pub(crate) fn bounded_outline(font: &FontRef, gid: GlyphId, width: f64) -> Result<GlyphOutline, GlyphError> {
178    let loc = LocationRef::default();
179    let scale = 1.0 / font.metrics(Size::unscaled(), loc).units_per_em.max(1) as f64;
180    let Some(glyph) = font.outline_glyphs().get(gid) else { return Err(GlyphError::Missing) };
181    let mut pen = Flatten::new(scale);
182    let _ = glyph.draw(DrawSettings::unhinted(Size::unscaled(), loc), &mut pen);
183    pen.close();
184    if pen.too_complex || pen.contours.len() > 256 {
185        return Err(GlyphError::TooComplex);
186    }
187    let mut bbox = [0.0, 0.0, width, 0.0];
188    let mut any = false;
189    for p in pen.contours.iter().flatten() {
190        if !p[0].is_finite() || !p[1].is_finite() {
191            return Err(GlyphError::Missing);
192        }
193        if !any {
194            bbox = [p[0], p[1], p[0], p[1]];
195            any = true;
196        } else {
197            bbox[0] = bbox[0].min(p[0]);
198            bbox[1] = bbox[1].min(p[1]);
199            bbox[2] = bbox[2].max(p[0]);
200            bbox[3] = bbox[3].max(p[1]);
201        }
202    }
203    Ok(GlyphOutline { contours: pen.contours, width, bbox })
204}

206/// The first character of `text` (other than white space) that neither the script font of typed
207/// signatures nor its Inter fallback has a glyph for: [`script_outline`] would leave it out.
208/// `None` when they can draw all of them.
209pub fn script_missing(text: &str) -> Option<char> {
210    let font = FontRef::new(FONT).ok()?;
211    let charmap = font.charmap();
212    let fallback = FontRef::new(INTER_REGULAR).ok();
213    let fallback_map = fallback.as_ref().map(|f| f.charmap());
214    text.chars().find(|c| !c.is_whitespace() && charmap.map(*c).is_none() && fallback_map.as_ref().and_then(|m| m.map(*c)).is_none())
215}

217/// The outlines of `text` in the script font. Characters it lacks (Cyrillic, Greek, …) are drawn
218/// from Inter, slanted and scaled to the script's capital height; characters neither font has are
219/// skipped (check [`script_missing`] first where nobody sees a preview). Over-limit input returns
220/// an empty outline instead of a silently truncated signature.
221pub fn script_outline(text: &str) -> ScriptOutline {
222    if text.chars().take(MAX_SIGNATURE_CHARS + 1).count() > MAX_SIGNATURE_CHARS {
223        return ScriptOutline::default();
224    }
225    let Ok(font) = FontRef::new(FONT) else { return ScriptOutline::default() };
226    let loc = LocationRef::default();
227    let metrics = font.metrics(Size::unscaled(), loc);
228    let scale = 1.0 / metrics.units_per_em.max(1) as f64;
229    let charmap = font.charmap();
230    let glyphs = font.outline_glyphs();
231    let advances = font.glyph_metrics(Size::unscaled(), loc);
232    // The fallback face, scaled so its capitals are as tall as the script's.
233    let fallback = FontRef::new(INTER_REGULAR).ok().map(|f| {
234        let m = f.metrics(Size::unscaled(), loc);
235        let em = 1.0 / m.units_per_em.max(1) as f64;
236        let ratio = match (metrics.cap_height, m.cap_height) {
237            (Some(ours), Some(theirs)) if ours > 0.0 && theirs > 0.0 => (ours as f64 * scale) / (theirs as f64 * em),
238            _ => 1.0,
239        };
240        let ratio = if ratio.is_finite() { ratio.clamp(0.5, 2.0) } else { 1.0 };
241        (f.charmap(), f.outline_glyphs(), f.glyph_metrics(Size::unscaled(), loc), em * ratio)
242    });
243    let mut pen = Flatten::new(scale);
244    // The 4096-point budget is for one untrusted document glyph (japanese_glyph), not a
245    // whole signature in our bundled font. Keep a separate bounded budget for the line.
246    pen.max_points = 262_144;
247    for ch in text.chars() {
248        if let Some(gid) = charmap.map(ch) {
249            (pen.scale, pen.slant) = (scale, 0.0);
250            if let Some(g) = glyphs.get(gid) {
251                let _ = g.draw(DrawSettings::unhinted(Size::unscaled(), loc), &mut pen);
252                pen.close();
253            }
254            pen.dx += advances.advance_width(gid).unwrap_or(0.0) as f64 * scale;
255        } else if let Some((map, outlines, adv, em)) = &fallback
256            && let Some(gid) = map.map(ch)
257        {
258            (pen.scale, pen.slant) = (*em, FALLBACK_SLANT);
259            if let Some(g) = outlines.get(gid) {
260                let _ = g.draw(DrawSettings::unhinted(Size::unscaled(), loc), &mut pen);
261                pen.close();
262            }
263            pen.dx += adv.advance_width(gid).unwrap_or(0.0) as f64 * em;
264        }
265    }
266    pen.close();
267    if pen.too_complex {
268        return ScriptOutline::default();
269    }
270    ScriptOutline { contours: pen.contours, width: pen.dx, ascent: metrics.ascent as f64 * scale, descent: metrics.descent as f64 * scale }
271}

276    fn long_names_keep_every_glyph_with_bounded_outline_work() {
277        let text = "Alexandria Catherine Elizabeth Montgomery-Wellington";
278        let o = super::script_outline(text);
279        let expected: usize = text.chars().map(|c| super::script_outline(&c.to_string()).contours.len()).sum();

281        let max_x = o.contours.iter().flatten().map(|p| p[0]).fold(0.0, f64::max);

287    /// Issue #846: Cyrillic (and Greek) names come out of the fallback face instead of vanishing.
288    #[test]
289    fn cyrillic_and_greek_names_are_drawn_from_the_fallback_face() {
290        for name in ["Саша Дермановић", "Љубица Ђорђевић", "Ђурђа Џаковић", "Νίκος"] {
291            let o = super::script_outline(name);
292            let letters = name.chars().filter(|c| !c.is_whitespace()).count();
293            assert!(o.contours.len() >= letters, "{name}: {} contours for {letters} letters", o.contours.len());
294            for ch in name.chars().filter(|c| !c.is_whitespace()) {
295                let one = super::script_outline(&ch.to_string());
296                assert!(!one.contours.is_empty() && one.width > 0.0, "{name}: {ch} has no outline");
297            }
298            let [_, bottom, right, top] = o.bounds();
299            assert!(right <= o.width + 0.5 && top <= o.ascent + 0.2 && bottom >= o.descent - 0.2, "{name}: {:?}", o.bounds());
300        }
301        // Capitals from the fallback are about as tall as the script's.
302        let cap = |c: &str| super::script_outline(c).contours.iter().flatten().map(|p| p[1]).fold(0.0, f64::max);
303        let (latin, cyrillic) = (cap("H"), cap("Н"));
304        assert!((cyrillic / latin - 1.0).abs() < 0.2, "Н {cyrillic} vs H {latin}");
305        // Mixed names keep every letter, in order: the Cyrillic follows the Latin.
306        let mixed = super::script_outline("Ana Ана");
307        let latin_only = super::script_outline("Ana ");
308        assert!(mixed.width > latin_only.width + 0.5, "{} vs {}", mixed.width, latin_only.width);
309    }
