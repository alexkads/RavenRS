39/// Advance of `s` in Helvetica at `size` points, using the exact standard-14 metrics.
40///
41/// Kept as a free function because generated appearances overwhelmingly use Helvetica; reach for
42/// [`Std14`] directly for any other face, and never for a scaled approximation of this one.
43pub fn helvetica_width(s: &str, size: f64) -> f64 {
44    Std14::Helvetica.text_width(s, size)
45}
46
47/// Approximate advance of `s` at `size` points in a CJK font: full-width characters take one
48/// em, half-width forms half of one, and everything else is measured as Helvetica.
49pub fn cjk_width(s: &str, size: f64) -> f64 {
50    let mut buf = [0u8; 4];
51    s.chars()
52        .map(|c| match c {
53            '\u{ff61}'..='\u{ffdc}' | '\u{ffe8}'..='\u{ffee}' => size * 0.5,
54            _ if is_wide(c) => size,
55            _ => helvetica_width(c.encode_utf8(&mut buf), size),
56        })
57        .sum()
58}

60/// Approximate advance of `s` in Inter Regular at `size` points.
61pub fn inter_width(s: &str, size: f64) -> f64 {
62    use skrifa::MetadataProvider as _;
63    let Ok(font) = skrifa::FontRef::new(INTER_REGULAR) else { return helvetica_width(s, size) };
64    let loc = skrifa::instance::LocationRef::default();
65    let upem = font.metrics(skrifa::instance::Size::unscaled(), loc).units_per_em.max(1) as f64;
66    let scale = size / upem;
67    let charmap = font.charmap();
68    let advances = font.glyph_metrics(skrifa::instance::Size::unscaled(), loc);
69    s.chars().map(|c| charmap.map(c).and_then(|gid| advances.advance_width(gid)).unwrap_or(0.0) as f64 * scale).sum()
70}

72/// East Asian wide and full-width characters (Unicode UAX #11 W and F, by block).
73fn is_wide(c: char) -> bool {
74    matches!(c,
75        '\u{1100}'..='\u{115f}' // Hangul Jamo initials
76        | '\u{2e80}'..='\u{303e}' // CJK radicals, Kangxi, ideographic description, CJK symbols
77        | '\u{3041}'..='\u{33ff}' // kana, Bopomofo, Hangul compatibility, Kanbun, CJK compatibility
78        | '\u{3400}'..='\u{4dbf}' // CJK extension A
79        | '\u{4e00}'..='\u{9fff}' // CJK unified ideographs
80        | '\u{a000}'..='\u{a4cf}' // Yi
81        | '\u{ac00}'..='\u{d7a3}' // Hangul syllables
82        | '\u{f900}'..='\u{faff}' // CJK compatibility ideographs
83        | '\u{fe30}'..='\u{fe4f}' // CJK compatibility forms
84        | '\u{ff00}'..='\u{ff60}' // full-width forms
85        | '\u{ffe0}'..='\u{ffe6}'
86        | '\u{20000}'..='\u{3fffd}' // supplementary ideographic planes
87    )
88}

90/// Greedy line breaking within `width` points (paragraphs split on line breaks, where CRLF counts as one; words longer
91/// than a line are broken by character).
92pub fn wrap(text: &str, size: f64, width: f64) -> Vec<String> {
93    wrap_with(text, size, width, helvetica_width)
94}
95
96/// [`wrap`], measuring with `measure` (for example [`cjk_width`]).
97///
98/// A line measures as the sum of its words, the spaces between them and the characters of a broken
99/// word, each measured once, so the work stays linear in `text` however much fits on a line.
100pub fn wrap_with(text: &str, size: f64, width: f64, measure: impl Fn(&str, f64) -> f64) -> Vec<String> {
101    let mut lines = Vec::new();
102    let space = measure(" ", size);
103    // A CRLF pair is one break, a lone CR or LF is one break each.
104    for para in text.replace("\r\n", "\n").split(['\n', '\r']) {
105        let mut line = String::new();
106        let mut line_width = 0.0;
107        // No word is on the line yet; an empty word (a leading space) still starts it.
108        let mut fresh = true;
109        for word in para.split(' ') {
110            let word_width = measure(word, size);
111            let candidate = if fresh { word_width } else { line_width + space + word_width };
112            if candidate <= width {
113                if !fresh {
114                    line.push(' ');
115                }
116                line.push_str(word);
117                line_width = candidate;
118                fresh = false;
119                continue;
120            }
121            // Indentation alone is not a line: drop it before an over-long word.
122            if line.trim().is_empty() {
123                line.clear();
124            } else {
125                lines.push(std::mem::take(&mut line));
126            }
127            line_width = 0.0;
128            // Broken between characters, but never before a Thai mark: it stays on its letter.
129            let mut start = 0;
130            let mut units = Vec::new();
131            for (i, ch) in word.char_indices().skip(1) {
132                if !shaped_text::joins_previous(ch) {
133                    units.extend(word.get(start..i));
134                    start = i;
135                }
136            }
137            units.extend(word.get(start..).filter(|u| !u.is_empty()));
138            for unit in units {
139                let unit_width = measure(unit, size);
140                if !line.is_empty() && line_width + unit_width > width {
141                    lines.push(std::mem::take(&mut line));
142                    line_width = 0.0;
143                }
144                line.push_str(unit);
145                line_width += unit_width;
146            }
147            fresh = line.is_empty();
148        }
149        lines.push(line);
150    }
151    lines
152}

154/// The WinAnsiEncoding (ISO 32000-2 Annex D) code for `c`, if the encoding has one.
155///
156/// Codes 0x80–0x9F come from the same table the text extractor decodes with, so encoding and
157/// decoding agree (Š š Ž ž Œ œ Ÿ ƒ † ‡ ˆ ˜ ‰ ‹ › as well as quotes, dashes, € and ™).
158pub fn win_ansi_byte(c: char) -> Option<u8> {
159    match c {
160        '\u{20}'..='\u{7e}' | '\u{a0}'..='\u{ff}' => u8::try_from(u32::from(c)).ok(),
161        _ => {
162            let high = encodings::WIN_ANSI.get(0x80..0xa0)?;
163            let i = high.iter().position(|&u| u != 0 && u == u32::from(c))?;
164            u8::try_from(0x80 + i).ok()
165        }
166    }
167}

169/// Encode text in WinAnsiEncoding (ISO 32000-2 Annex D); tabs become spaces and unmappable
170/// characters become `?`.
171pub fn win_ansi(s: &str) -> Vec<u8> {
172    s.chars().map(|c| if c == '\t' { b' ' } else { win_ansi_byte(c).unwrap_or(b'?') }).collect()
173}

175/// The character encoding of a predefined Unicode CMap (`UniJIS-UTF16-H`, `UniGB-UCS2-H`, …;
176/// ISO 32000-2 §9.7.5.2), which maps Unicode text straight to codes.
177#[derive(Clone, Copy, Debug, PartialEq, Eq)]
178pub enum UnicodeCMap {
179    Ucs2,
180    Utf16,
181    Utf8,
182    Utf32,
183}
184
185impl UnicodeCMap {
186    /// The encoding of the predefined CMap `name`, if it is a horizontal Unicode one. Vertical
187    /// (`-V`) CMaps are refused: field appearances are laid out horizontally, so their glyphs
188    /// would be drawn stacked down the field instead of across it.
189    pub fn from_name(name: &[u8]) -> Option<Self> {
190        let name = std::str::from_utf8(name).ok()?;
191        if !name.starts_with("Uni") || !name.ends_with("-H") {
192            return None;
193        }
194        name.split('-').find_map(|part| match part {
195            "UCS2" => Some(Self::Ucs2),
196            "UTF16" => Some(Self::Utf16),
197            "UTF8" => Some(Self::Utf8),
198            "UTF32" => Some(Self::Utf32),
199            _ => None,
200        })
201    }
202
203    /// The first character of `s` this encoding can't hold, which [`Self::encode`] would write as
204    /// `?`: a character beyond U+FFFF in UCS-2 (`𠀋`). UTF-16, UTF-8 and UTF-32 hold every one.
205    pub fn first_unencodable(self, s: &str) -> Option<char> {
206        match self {
207            Self::Ucs2 => s.chars().find(|c| u16::try_from(u32::from(*c)).is_err()),
208            Self::Utf16 | Self::Utf8 | Self::Utf32 => None,
209        }
210    }
211
212    /// Encode `s`; tabs become spaces, and characters UCS-2 can't hold become `?` (check them
213    /// first with [`Self::first_unencodable`]: PdfCraft refuses such text rather than write `?`).
214    pub fn encode(self, s: &str) -> Vec<u8> {
215        let mut out = Vec::with_capacity(s.len().saturating_mul(2));
216        let mut units = [0u16; 2];
217        let mut buf = [0u8; 4];
218        for c in s.chars() {
219            let c = if c == '\t' { ' ' } else { c };
220            match self {
221                Self::Ucs2 => {
222                    let unit = u16::try_from(u32::from(c)).unwrap_or(u16::from(b'?'));
223                    out.extend_from_slice(&unit.to_be_bytes());
224                }
225                Self::Utf16 => {
226                    for unit in c.encode_utf16(&mut units) {
227                        out.extend_from_slice(&unit.to_be_bytes());
228                    }
229                }
230                Self::Utf8 => out.extend_from_slice(c.encode_utf8(&mut buf).as_bytes()),
231                Self::Utf32 => out.extend_from_slice(&u32::from(c).to_be_bytes()),
232            }
233        }
234        out
235    }
236}

238/// Whether [`win_ansi`] keeps `c` rather than writing `?` (a tab becomes a space).
239pub fn win_ansi_encodable(c: char) -> bool {
240    c == '\t' || win_ansi_byte(c).is_some()
241}

243/// The first character of `s` that [`win_ansi`] can't encode and would write as `?`.
244pub fn first_non_win_ansi(s: &str) -> Option<char> {
245    s.chars().find(|c| !win_ansi_encodable(*c))
246}

248/// The first character of `s` the standard fonts (Helvetica and the rest of the standard 14, in
249/// WinAnsiEncoding) can't draw, so that a generated appearance would show it as `?`. Line breaks
250/// don't count: appearances lay lines out themselves. PdfCraft refuses such text with
251/// [`undrawable_message`] instead of writing `?` (AGENTS.md §4: never lose work silently).
252pub fn first_undrawable(s: &str) -> Option<char> {
253    s.chars().find(|c| !matches!(c, '\n' | '\r') && !win_ansi_encodable(*c))
254}

256/// The error for text the standard fonts can't draw ([`first_undrawable`]): which character, in
257/// what (`"the comment"`, `"the reason"`), and what can be written instead. One wording for every
258/// path that writes text in the standard fonts, so agents and users can act on it the same way.
259pub fn undrawable_message(what: &str, c: char) -> String {
260    format!(
261        "the standard fonts can't draw \"{}\" (U+{:04X}) in {what}; only Western European characters can be written there for now",
262        c.escape_debug(),
263        u32::from(c)
264    )
265}

267/// The private key on an annotation (a comment or a form field's widget) whose appearance PdfCraft
268/// drew with `?` in place of characters its fonts can't draw: a text string holding the first
269/// such character. The text itself (a field's `/V`, a comment's `/Contents`) is stored as typed;
270/// the mark lets flattening refuse to make that `?` permanent, lets an edit report it, and tells
271/// such an appearance apart from one another program drew with a font that has those characters.
272pub const UNDRAWN_MARK: &[u8] = b"PCUndrawn";
273
274/// The character [`UNDRAWN_MARK`] records on `d`, if PdfCraft drew its appearance with `?`.
275pub fn undrawn_mark(d: &crate::cos::Dict) -> Option<char> {
276    d.get(UNDRAWN_MARK).and_then(crate::cos::Object::as_string).and_then(|s| s.to_text().chars().next())
277}
278
279/// Record (`Some`) or clear (`None`) [`UNDRAWN_MARK`] on `d`, for the appearance just drawn.
280pub fn set_undrawn_mark(d: &mut crate::cos::Dict, c: Option<char>) {
281    match c {
282        Some(c) => d.set(UNDRAWN_MARK.to_vec(), crate::cos::PdfString::text(&c.to_string())),
283        None => {
284            d.remove(UNDRAWN_MARK);
285        }
286    }
287}

299/// Bytes as a PDF literal string, `(` … `)`, with delimiters escaped.
300pub fn literal(bytes: &[u8]) -> Vec<u8> {
301    let mut out = Vec::with_capacity(bytes.len() + 2);
302    out.push(b'(');
303    for &b in bytes {
304        match b {
305            b'(' | b')' | b'\\' => out.extend_from_slice(&[b'\\', b]),
306            b'\r' => out.extend_from_slice(b"\\r"),
307            b'\n' => out.extend_from_slice(b"\\n"),
308            _ => out.push(b),
309        }
310    }
311    out.push(b')');
312    out
313}

408    /// WinAnsiEncoding's 0xAD is the `hyphen` glyph and its 0xA0 is `space`, not U+00AD and
409    /// U+00A0; MacRomanEncoding's 0xCA is `space` too.
410    ///
411    /// Annex D.2 gives each of those codes the same glyph name as the ordinary one (0x2D, 0x20),
412    /// so the bytes a Windows codepage calls a soft hyphen and a non-breaking space draw ordinary
413    /// ink and must extract as it. Reading them as the formatting characters hid them: a soft
414    /// hyphen is invisible, so a search for "well-known" missed text that drew exactly that, and
415    /// `redact`'s pattern matching missed it too; a non-breaking space does not match the space a
416    /// reader types. Over the 989-file pdf.js corpus this takes the soft hyphens we extract from
417    /// 117 to 13 (104 of them in `XiaoBiaoSong.pdf` alone) and the non-breaking spaces from 30 to
418    /// 26; what remains comes from `/ToUnicode` maps, which say so explicitly and are not touched.
419    /// pdftotext reads both codes as the ordinary character.

421    fn the_alias_codes_read_as_the_glyph_they_draw() {

423        assert_eq!(encodings::WIN_ANSI.get(0x2D), Some(&u32::from('-')));
424        assert_eq!(encodings::WIN_ANSI.get(0xA0), Some(&u32::from(' ')));
425        assert_eq!(encodings::WIN_ANSI.get(0x20), Some(&u32::from(' ')));
426        assert_eq!(encodings::MAC_ROMAN.get(0xCA), Some(&u32::from(' ')));
427        assert_eq!(encodings::MAC_ROMAN.get(0x20), Some(&u32::from(' ')));

439    #[test]
440    fn win_ansi_covers_the_0x80_to_0x9f_glyphs() {
441        // Every character in the WinAnsiEncoding table encodes to its code and back (#347), except
442        // where two codes share one glyph: 0xAD is `hyphen`, the same glyph as 0x2D, and 0xA0 is
443        // `space`, the same glyph as 0x20, so each decodes to the ordinary character and that
444        // character encodes back to the canonical low code. Round-tripping the *code* is what
445        // cannot hold there; round-tripping the character still does.
446        const ALIASES: &[u8] = &[0xA0, 0xAD];
447        for (code, &u) in encodings::WIN_ANSI.iter().enumerate() {
448            let Some(c) = char::from_u32(u).filter(|_| u != 0) else { continue };
449            if ALIASES.contains(&(code as u8)) {
450                let canonical = win_ansi_byte(c).expect("an aliased glyph still encodes");
451                assert!(canonical < code as u8, "{c:?}: the canonical code is the lower one");
452                assert_eq!(encodings::WIN_ANSI.get(usize::from(canonical)), Some(&u), "both codes mean the same character");
453                continue;
454            }
455            assert_eq!(win_ansi_byte(c), Some(code as u8), "{c:?} (U+{u:04X})");
456            assert_eq!(win_ansi(&c.to_string()), [code as u8]);
457        }
458        assert_eq!(
459            win_ansi("Šta? žaba, œuvre — „ok“ ‰ ƒ † ‡ ˆ ˜ ‹›Ÿ"),
460            b"\x8ata? \x9eaba, \x9cuvre \x97 \x84ok\x93 \x89 \x83 \x86 \x87 \x88 \x98 \x8b\x9b\x9f"
461        );
462        // Not in WinAnsi: č ć still need the fallback font; undefined codes are never produced.
463        assert_eq!(win_ansi("čć\u{81}\u{8d}\t"), b"???? ");
464        assert_eq!(win_ansi_byte('\0'), None);
465    }

495    /// Wrapping measures each word and character a bounded number of times, however many fit on a line.
496    #[test]
497    fn wrapping_measures_each_word_a_few_times() {
498        let measured = std::cell::Cell::new(0usize);
499        let measure = |s: &str, size: f64| {
500            measured.set(measured.get() + s.len());
501            helvetica_width(s, size)
502        };
503        let text = format!("{} {}", "word ".repeat(5_000), "x".repeat(5_000));
504        let lines = wrap_with(&text, 1.0, 13_000.0, measure);
505        assert_eq!(lines.len(), 2, "the words fit on one line, the long word on another");
506        assert!(measured.get() <= 4 * text.len(), "{} bytes measured", measured.get());
507        let narrow = wrap_with(&text, 10.0, 100.0, helvetica_width);
508        assert!(narrow.iter().all(|l| helvetica_width(l, 10.0) <= 100.0), "{narrow:?}");
509    }
