1//! What content editing and redaction need from a PDF font: how to split a string into character
2//! codes, each code's advance width, the glyph height (ascent, descent), and what each code means
3//! (Unicode) — and back, which text the font can show. Widths come from the font dictionary
4//! (`/Widths`, `/W`, `/DW`, `/MissingWidth`); a simple font with no `/Widths` uses its embedded
5//! sfnt program's advances, else the exact published standard-14 metrics ([`crate::fonts::Std14`]).
6//!
7//! Meanings come from `/ToUnicode`, else the encoding (`/Encoding` base and `/Differences` glyph
8//! names, ISO 32000-2 Annex D). A code may deliberately have **no** meaning: where the font's own
9//! built-in encoding governs and we hold no table for it — Symbol, ZapfDingbats, MacExpert — this
10//! module reports nothing rather than the Latin letter at that code. Callers must treat a missing
11//! meaning as "unknown", never as "absent": `redact` matches on this map, and a wrong character
12//! there is worse than no character.

96/// Whether `base_font` (subset tag allowed) names a legacy Indic font family that is laid out in
97/// visual order (Kruti Dev, DevLys, Chanakya, Shusha, Shivaji, Walkman, Akruti, APS-DV, DV-TT…).
98fn legacy_indic_font(base_font: &str) -> bool {
99    let name =
100        base_font.split_once('+').filter(|(tag, _)| tag.len() == 6 && tag.bytes().all(|b| b.is_ascii_uppercase())).map_or(base_font, |(_, n)| n);

102    ["kruti", "devlys", "chanakya", "shusha", "shivaji", "walkman", "akruti"].iter().any(|n| name.contains(n))

104}

106/// A source PDF's sfnt program for an in-memory editor. Never a bundled font or an export
107/// fallback. Type 1/Type 3 programs cannot be consumed by an sfnt text renderer.
108pub fn embedded_sfnt(doc: &Document, font: &Dict) -> Option<Vec<u8>> {
109    // No total: one font's two programs, and the metrics read that may adapt one, are each
110    // bounded by that stream's own limit.
111    embedded_sfnt_within(doc, font, &mut DecodeBudget::new(usize::MAX, 0))
112}

114/// The largest embedded sfnt program read for the editor.
115const MAX_SFNT: usize = 16 << 20;
116
117/// [`embedded_sfnt`], decoding against `budget`, which an enumeration of a page's fonts shares:
118/// each program decoded is charged whether or not it turns out to be usable, and so are the
119/// streams of the metrics read that adapts a program without a Unicode cmap.
120pub fn embedded_sfnt_within(doc: &Document, font: &Dict, budget: &mut DecodeBudget) -> Option<Vec<u8>> {
121    use skrifa::{MetadataProvider as _, raw::TableProvider as _};
122    let descendant = dict(doc, font.get(b"DescendantFonts").and_then(|o| doc.resolve(o).as_array().and_then(|a| a.first()).cloned()).as_ref());
123    let descriptor = dict(doc, descendant.as_ref().unwrap_or(font).get(b"FontDescriptor"))?;
124    for key in [b"FontFile2".as_slice(), b"FontFile3".as_slice()] {
125        let Some(object) = descriptor.get(key) else { continue };
126        let object = doc.resolve(object);
127        let Object::Stream(stream) = object.as_ref() else { continue };
128        // As in `Reads::decode`: each stage within the limit, the program, the last, too, read
129        // through a view of the stream; once the allowance is spent a program with data is refused
130        // before its parameters are resolved.
131        let Ok(mut bytes) = budget.decode_stages_within(doc, stream, MAX_SFNT) else { continue };
132        let Ok(face) = skrifa::FontRef::from_index(&bytes, 0) else { continue };
133        if face.metrics(skrifa::instance::Size::unscaled(), skrifa::instance::LocationRef::default()).units_per_em == 0 {
134            continue;
135        }
136        let own_cmap = face.charmap().mappings().next().is_some();
137        // A legacy THSarabunPSK subset may keep a Latin-only cmap beside its Thai glyphs, which
138        // only the PDF's verified maps reach (`composite_glyph_maps`); other programs keep theirs.
139        let base_font = font.name(b"BaseFont").map(String::from_utf8_lossy).unwrap_or_default();
140        if own_cmap && !base_font.rsplit('+').next().is_some_and(|f| f.starts_with("THSarabunPSK")) {
141            return Some(bytes);
142        }
143        // A composite font's cmap is rebuilt only through Identity-H codes (`composite_glyph_maps`):
144        // with any other `/Encoding` nothing is, so its metrics aren't read, and a program with a
145        // cmap of its own is kept as it is, as when the rebuild finds nothing.
146        if font.name(b"Subtype") == Some(b"Type0") && !identity_h(doc, font) {
147            if own_cmap {
148                return Some(bytes);
149            }
150            continue;
151        }
152        // PDF subsets may remove the Unicode cmap entirely. Rebuild an editor-only
153        // cmap from ToUnicode and the actual CID-to-glyph map; source bytes stay unchanged.
154        // Metrics read past the allowance, its widths' read of the program again included,
155        // adapt no copy.
156        let metrics = Metrics::from_dict_within(doc, font, budget);
157        if metrics.past_allowance {
158            continue;
159        }
160        let mut mappings = std::collections::BTreeMap::new();
161        if metrics.composite {
162            let refused = budget.refused();
163            if let Some((map, _)) = composite_glyph_maps(doc, font, &bytes, &metrics, budget) {
164                mappings = map;
165            } else if budget.refused() > refused {
166                // The map the cmap is rebuilt from was left unread for want of the allowance: no
167                // copy is listed, adapted or as it is, as for metrics read past it.
168                continue;
169            }
170            if own_cmap {
171                let charmap = face.charmap();
172                if mappings.keys().all(|&u| char::from_u32(u).is_some_and(|c| charmap.map(c).is_some())) {
173                    return Some(bytes);
174                }
175                // The program's own mappings stay; the PDF's fill only what they lack.
176                mappings.extend(charmap.mappings().filter(|(_, g)| g.to_u32() != 0).take(65_537).map(|(u, g)| (u, g.to_u32())));
177                let mut merged = bytes.clone();
178                let usable = append_editor_cmap(&mut merged, &mappings).is_some()
179                    && skrifa::FontRef::new(&merged).is_ok_and(|font| font.charmap().mappings().next().is_some());
180                return Some(if usable { merged } else { bytes });
181            }
182        } else if own_cmap {
183            return Some(bytes);
184        } else if let Ok(cmap) = face.cmap() {
185            for code in 0..=255 {
186                let Some(text) = metrics.unicode.text(code) else { continue };
187                let Some(ch) = only_char(&text) else { continue };
188                let Some(glyph) = cmap.map_codepoint(code).or_else(|| cmap.map_codepoint(code + 0xf000)) else { continue };
189                let glyph = glyph.to_u32();
190                if glyph > 0 {
191                    mappings.entry(u32::from(ch)).or_insert(glyph);
192                }
193            }
194        }
195        if !mappings.is_empty()
196            && append_editor_cmap(&mut bytes, &mappings).is_some()
197            && skrifa::FontRef::new(&bytes).is_ok_and(|font| font.charmap().mappings().next().is_some())
198        {
199            return Some(bytes);
200        }
201    }
202    None
203}

205/// The CIDs (below 65 536) of a Type 0 font over an embedded TrueType program whose glyph has
206/// outline data, through `/CIDToGIDMap`; empty when the program or the map can't be read.
207/// Subsetters keep glyph ids and empty the glyphs they drop, so an empty glyph is a missing one.
208fn cids_with_outlines(doc: &Document, font: &Dict, reads: &mut Reads<'_>) -> Vec<u64> {
209    use skrifa::raw::TableProvider as _;
210    let Some(descendant) = dict(doc, font.get(b"DescendantFonts").and_then(|o| doc.resolve(o).as_array().and_then(|a| a.first()).cloned()).as_ref())
211    else {
212        return Vec::new();
213    };
214    let Some(program) = dict(doc, descendant.get(b"FontDescriptor")).and_then(|d| d.get(b"FontFile2").cloned()) else { return Vec::new() };
215    let program = doc.resolve(&program);
216    let Object::Stream(stream) = program.as_ref() else { return Vec::new() };
217    let Some(bytes) = reads.decode(stream, MAX_SFNT) else { return Vec::new() };
218    let Some(loca) = skrifa::FontRef::from_index(&bytes, 0).ok().and_then(|face| face.loca(None).ok()) else { return Vec::new() };
219    let outlined = |gid: u32| loca.get_raw(gid as usize).zip(loca.get_raw(gid as usize + 1)).is_some_and(|(start, end)| end > start);
220    match descendant.get(b"CIDToGIDMap").map(|o| doc.resolve(o)).as_deref() {
221        None | Some(Object::Name(_)) => bit_set_below_64k((0..loca.len().min(0x1_0000) as u32).filter(|&gid| outlined(gid))),
222        Some(Object::Stream(map)) => {
223            let Some(map) = reads.decode(map, MAX_CID_TO_GID) else { return Vec::new() };
224            bit_set_below_64k(
225                map.as_chunks::<2>()
226                    .0
227                    .iter()
228                    .enumerate()
229                    .filter(|(_, gid)| outlined(u32::from(u16::from_be_bytes(**gid))))
230                    .map(|(cid, _)| cid as u32),
231            )
232        }
233        Some(_) => Vec::new(),
234    }
235}

237/// A simple font with no `/Widths` is drawn (and found by text search) with its embedded
238/// program's own advances, so measure with them too; a standard-14 guess from the name can be
239/// wider and leave the end of a line outside a redaction mark (#2130). The glyph is looked up
240/// the way the renderer does: the encoding's character in the Unicode cmap, else the code itself
241/// (as in a symbolic cmap), else `.notdef`. Only sfnt programs are read; others keep the guess.
242fn embedded_widths(doc: &Document, font: &Dict, m: &mut Metrics, reads: &mut Reads<'_>) {
243    use skrifa::instance::{LocationRef, Size};
244    use skrifa::{GlyphId, MetadataProvider as _, raw::TableProvider as _};
245    let Some(descriptor) = dict(doc, font.get(b"FontDescriptor")) else { return };
246    for key in [b"FontFile2".as_slice(), b"FontFile3".as_slice()] {
247        let Some(object) = descriptor.get(key) else { continue };
248        let object = doc.resolve(object);
249        let Object::Stream(stream) = object.as_ref() else { continue };
250        let Some(bytes) = reads.decode(stream, MAX_SFNT) else { continue };
251        let Ok(face) = skrifa::FontRef::from_index(&bytes, 0) else { continue };
252        let upem = f64::from(face.metrics(Size::unscaled(), LocationRef::default()).units_per_em);
253        if upem == 0.0 {
254            continue;
255        }
256        let advances = face.glyph_metrics(Size::unscaled(), LocationRef::default());
257        let (charmap, cmap) = (face.charmap(), face.cmap().ok());
258        let found = |g: Option<GlyphId>| g.filter(|g| *g != GlyphId::NOTDEF);
259        m.widths = (0..=255u32)
260            .map(|code| {
261                let glyph = found(m.unicode.encoding.get(&code).and_then(|t| only_char(t)).and_then(|c| charmap.map(c)))
262                    .or_else(|| [0, 0xF000, 0xF100, 0xF200].into_iter().find_map(|o| found(cmap.as_ref()?.map_codepoint(code + o))));
263                // A subset holds only the glyphs it draws: 0 keeps `has_glyph` false for the rest.
264                match glyph {
265                    None if m.subset => 0.0,
266                    _ => advances.advance_width(glyph.unwrap_or(GlyphId::NOTDEF)).map_or(0.0, |w| f64::from(w) * 1000.0 / upem),
267                }
268            })
269            .collect();
270        m.first = 0;
271        return;
272    }
273}

275/// Add or replace an editor-only cmap with a bounded Unicode format-12 table.
276/// Checked ranges prevent damaged sfnt offsets from indexing outside the source program.
277fn append_editor_cmap(bytes: &mut Vec<u8>, mappings: &std::collections::BTreeMap<u32, u32>) -> Option<()> {
278    if mappings.len() > 65536 {
279        return None;
280    }
281    let count = u16::from_be_bytes(bytes.get(4..6)?.try_into().ok()?) as usize;
282    if count == 0 || count > 256 {
283        return None;
284    }
285    let mut entry = None;
286    let mut head = None;
287    for i in 0..count {
288        let at = 12 + i * 16;
289        let record = bytes.get(at..at + 16)?;
290        if record.get(..4)? == b"cmap" {
291            entry = Some(at);
292        }
293        if record.get(..4)? == b"head" {
294            head = Some(u32::from_be_bytes(record.get(8..12)?.try_into().ok()?) as usize);
295        }
296    }
297    let (entry, head) = match entry {
298        Some(entry) => (entry, head),
299        None => insert_editor_cmap_slot(bytes, count, head)?,
300    };
301    let mut cmap = Vec::with_capacity(28 + mappings.len() * 12);
302    cmap.extend_from_slice(&[0, 0, 0, 1, 0, 3, 0, 10, 0, 0, 0, 12, 0, 12, 0, 0]);
303    cmap.extend_from_slice(&(16_u32 + u32::try_from(mappings.len()).ok()? * 12).to_be_bytes());
304    cmap.extend_from_slice(&0_u32.to_be_bytes());
305    cmap.extend_from_slice(&u32::try_from(mappings.len()).ok()?.to_be_bytes());
306    for (&ch, &gid) in mappings {
307        cmap.extend_from_slice(&ch.to_be_bytes());
308        cmap.extend_from_slice(&ch.to_be_bytes());
309        cmap.extend_from_slice(&gid.to_be_bytes());
310    }
311    while !bytes.len().is_multiple_of(4) {
312        bytes.push(0);
313    }
314    let offset = u32::try_from(bytes.len()).ok()?;
315    let checksum = |data: &[u8]| {
316        data.chunks(4).fold(0_u32, |sum, c| {
317            let mut word = [0; 4];
318            for (a, b) in word.iter_mut().zip(c) {
319                *a = *b;
320            }
321            sum.wrapping_add(u32::from_be_bytes(word))
322        })
323    };
324    bytes.get_mut(entry + 4..entry + 8)?.copy_from_slice(&checksum(&cmap).to_be_bytes());
325    bytes.get_mut(entry + 8..entry + 12)?.copy_from_slice(&offset.to_be_bytes());
326    bytes.get_mut(entry + 12..entry + 16)?.copy_from_slice(&u32::try_from(cmap.len()).ok()?.to_be_bytes());
327    bytes.extend_from_slice(&cmap);
328    // SFNT directories are sorted by tag. Some readers tolerate the unsorted directory of
329    // a PDF subset, but shaping engines may binary-search it and otherwise miss the cmap.
330    // Adding a slot increased the directory count; include the new record in this sort.
331    let tables = usize::from(u16::from_be_bytes(bytes.get(4..6)?.try_into().ok()?));
332    let end = 12usize.checked_add(tables.checked_mul(16)?)?;
333    let mut records: Vec<[u8; 16]> = bytes.get(12..end)?.as_chunks::<16>().0.to_vec();
334    records.sort_by_key(|r| [r[0], r[1], r[2], r[3]]);
335    for (index, record) in records.iter().enumerate() {
336        let at = 12 + index * 16;
337        bytes.get_mut(at..at + 16)?.copy_from_slice(record);
338    }
339    if let Some(head) = head {
340        bytes.get_mut(head.checked_add(8)?..head.checked_add(12)?)?.fill(0);
341        let adjustment = 0xb1b0afba_u32.wrapping_sub(checksum(bytes));
342        bytes.get_mut(head + 8..head + 12)?.copy_from_slice(&adjustment.to_be_bytes());
343    }
344    Some(())
345}

347/// Reserve a new cmap directory slot, retaining every original table payload.
348fn insert_editor_cmap_slot(bytes: &mut Vec<u8>, mut count: usize, mut head: Option<usize>) -> Option<(usize, Option<usize>)> {
349    if count >= 256 {
350        return None;
351    }
352    let end = 12usize.checked_add(count.checked_mul(16)?)?;
353    bytes.get(..end)?;
354    // Keep all original table bytes, shifting them by one directory record.
355    for i in 0..count {
356        let at = 12 + i * 16;
357        let offset = u32::from_be_bytes(bytes.get(at + 8..at + 12)?.try_into().ok()?);
358        let len = u32::from_be_bytes(bytes.get(at + 12..at + 16)?.try_into().ok()?);
359        if (offset as usize) < end {
360            return None;
361        }
362        bytes.get(offset as usize..offset.checked_add(len)? as usize)?;
363    }
364    bytes.splice(end..end, [0; 16]);
365    for i in 0..count {
366        let at = 12 + i * 16;
367        let offset = u32::from_be_bytes(bytes.get(at + 8..at + 12)?.try_into().ok()?).checked_add(16)?;
368        bytes.get_mut(at + 8..at + 12)?.copy_from_slice(&offset.to_be_bytes());
369    }
370    head = head.and_then(|offset| offset.checked_add(16));
371    bytes.get_mut(end..end + 4)?.copy_from_slice(b"cmap");
372    count += 1;
373    let count = u16::try_from(count).ok()?;
374    let selector = count.ilog2() as u16;
375    let range = (1u16.checked_shl(u32::from(selector))?).checked_mul(16)?;
376    bytes.get_mut(4..6)?.copy_from_slice(&count.to_be_bytes());
377    bytes.get_mut(6..8)?.copy_from_slice(&range.to_be_bytes());
378    bytes.get_mut(8..10)?.copy_from_slice(&selector.to_be_bytes());
379    bytes.get_mut(10..12)?.copy_from_slice(&count.checked_mul(16)?.checked_sub(range)?.to_be_bytes());
380    Some((end, head))
381}

388/// Most mappings a CMap may add, where each `bfchar` or array-form entry counts one and each sequential
389/// range counts two (it covers at least two codes). So a CMap with at most this many codes is kept whole;
390/// mappings past the budget are ignored, so a stream of them can't take minutes or gigabytes.
391const MAX_CMAP_ENTRIES: usize = 1 << 20;
392/// Most codespace or CID ranges kept from a CMap; Adobe's largest published CMaps have a few
393/// thousand.
394const MAX_CMAP_RANGES: usize = 1 << 16;
395/// Largest decoded Encoding or ToUnicode CMap stream read; real CMaps stay well under it, and a larger
396/// one is left unread (behind one filter not decoded past it; behind several, see `Reads::decode`).
397const MAX_CMAP_BYTES: usize = 16 << 20;
398/// Largest decoded `/CIDToGIDMap` stream read: two bytes for each CID below 65,536.
399const MAX_CID_TO_GID: usize = 1 << 17;

401/// The streams one reading of a font decodes, against the caller's allowance, and whether one was
402/// left unread because that allowance was spent.
403struct Reads<'a> {
404    doc: &'a Document,
405    budget: &'a mut DecodeBudget,
406    refused: bool,
407}
408
409impl<'a> Reads<'a> {
410    fn new(doc: &'a Document, budget: &'a mut DecodeBudget) -> Self {
411        Self { doc, budget, refused: false }
412    }
413
414    /// `stream` decoded within `max` and the allowance, its `/DecodeParms` read where they are given
415    /// by reference too, through a view of the stream that copies no more of its dictionary than
416    /// its filters and parameters; `None` when it can't be, noting when the allowance is why.
417    ///
418    /// Each of its filter stages is held to `max`, as a decode without an allowance holds it, and
419    /// the stages together to `max` for each of them ([`DecodeBudget::decode_stages_within`]): a
420    /// map of 65,536 CIDs behind ASCIIHex and Flate, its Flate data smaller than the map, makes
421    /// all of `MAX_CID_TO_GID` in its last stage, with the hex stage's output on top. So a stream
422    /// reads here when it reads without an allowance and what is left pays for it. A stream with a
423    /// stage past `max` is unread. When what is left is more than what the stages before that one
424    /// made and `max`, it is charged those two and isn't refused; otherwise, with more left than
425    /// the stages' total (behind one filter, `max`), it is charged that total and isn't refused;
426    /// with no more left than that total, it spends all of it and counts as refused. Only what
427    /// the allowance counts as refused ([`DecodeBudget::refused`]) is noted: a read that fails
428    /// without one, such as an empty stream behind a filter nothing decodes, fails the same with
429    /// more left, and callers that refuse on that count must agree with [`Metrics::past_allowance`].
430    fn decode(&mut self, stream: &Stream, max: usize) -> Option<Vec<u8>> {
431        let refused = self.budget.refused();
432        match self.budget.decode_stages_within(self.doc, stream, max) {
433            Ok(data) => Some(data),
434            Err(_) => {
435                self.refused |= self.budget.refused() > refused;
436                None
437            }
438        }
439    }
440}

442/// Most Type 3 glyph descriptions read for their `d1` box. A simple font shows at most 256 codes.
443const MAX_TYPE3_GLYPHS: usize = 256;
444/// Largest decoded Type 3 glyph description read for its `d1` box; a larger one is skipped.
445const MAX_TYPE3_GLYPH_BYTES: usize = 64 << 10;
446/// Decoded bytes all of one font's glyph descriptions may take together.
447const MAX_TYPE3_FONT_BYTES: usize = 4 << 20;
448/// Farthest a Type 3 glyph box reaches from the baseline, in text space per unit of font size. A
449/// bitmap font drawn in page units at `0.01 Tf` is a few thousand; past this a box is garbage, and
450/// clamping it keeps the run boxes built from it finite.
451const MAX_TYPE3_EXTENT: f64 = 1e6;
452
453/// The union of the boxes a Type 3 font's glyph descriptions declare with `d1` (ISO 32000-2
454/// §9.6.4, Table 111), in glyph space, and whether every description chosen was read; `None` when
455/// no glyph declares one (`d0` glyphs don't). The descriptions the encoding's `/Differences`
456/// names are chosen, since it "shall specify the complete character encoding" (Table 110); only a
457/// font without one has its first descriptions chosen. Reads at most [`MAX_TYPE3_GLYPHS`]
458/// descriptions within [`MAX_TYPE3_FONT_BYTES`], each charged to the reading's allowance too.
459fn type3_glyph_box(doc: &Document, font: &Dict, reads: &mut Reads<'_>) -> Option<([f64; 4], bool)> {
460    let procs = dict(doc, font.get(b"CharProcs"))?;
461    let mut names: Vec<&[u8]> = Vec::new();
462    // Whether every glyph that can be drawn is read; if not, the union may be short of them.
463    let mut complete = true;
464    let encoding = dict(doc, font.get(b"Encoding"));
465    // A base encoding reaches glyphs `/Differences` doesn't name, which Type 3 doesn't provide for (its
466    // `/Differences` "shall specify the complete character encoding", Table 110): those boxes go unread.
467    if encoding.as_ref().is_some_and(|e| e.contains(b"BaseEncoding")) {
468        complete = false;
469    }
470    let diffs = encoding.as_ref().and_then(|e| e.get(b"Differences")).map(|d| doc.resolve(d));
471    for name in diffs.as_deref().and_then(Object::as_array).into_iter().flatten().filter_map(Object::as_name) {
472        if procs.get(name).is_none() || names.contains(&name) {
473            continue;
474        }
475        if names.len() >= MAX_TYPE3_GLYPHS {
476            complete = false;
477            break;
478        }
479        names.push(name);
480    }
481    let chosen: Vec<&Object> = if names.is_empty() {
482        complete = procs.len() <= MAX_TYPE3_GLYPHS;
483        procs.iter().take(MAX_TYPE3_GLYPHS).map(|(_, proc)| proc).collect()
484    } else {
485        names.iter().filter_map(|n| procs.get(n)).collect()
486    };
487    // What is left of the font's own total: the descriptions decode against the reading's
488    // allowance, which charges them, and each within what is left of this.
489    let mut left = MAX_TYPE3_FONT_BYTES;
490    let mut union: Option<[f64; 4]> = None;
491    for proc in chosen {
492        let Object::Stream(s) = &*doc.resolve(proc) else { continue };
493        // Once the reading's allowance is spent the rest go unread, before their parameters are
494        // resolved.
495        if reads.budget.refuses(s) {
496            reads.refused = true;
497            complete = false;
498            break;
499        }
500        let before = reads.budget.remaining();
501        let refused = reads.budget.refused();
502        let read = reads.budget.decode_with_damage(doc, s, MAX_TYPE3_GLYPH_BYTES.min(left)).map(|(data, _)| data);
503        left = left.saturating_sub(before.saturating_sub(reads.budget.remaining()));
504        let Ok(data) = read else {
505            // As in `Reads::decode`, only a refusal the allowance counts: a description that fails
506            // without one (an empty one nothing decodes, once the allowance is spent) leaves the
507            // rest to be read, or refused above.
508            if reads.budget.refused() > refused {
509                // The rest go unread too, for want of the reading's allowance.
510                reads.refused = true;
511                complete = false;
512                break;
513            }
514            if left == 0 {
515                // The rest go unread too.
516                complete = false;
517                break;
518            }
519            continue;
520        };
521        if let Some([llx, lly, urx, ury]) = d1_box(&data) {
522            let r = [llx.min(urx), lly.min(ury), llx.max(urx), lly.max(ury)];
523            union = Some(union.map_or(r, |u| [u[0].min(r[0]), u[1].min(r[1]), u[2].max(r[2]), u[3].max(r[3])]));
524        }
525    }
526    union.map(|u| (u, complete))
527}

529/// The glyph box `llx lly urx ury` of a glyph description that begins with `wx wy llx lly urx ury
530/// d1`; `None` for `d0` or anything else first.
531fn d1_box(data: &[u8]) -> Option<[f64; 4]> {
532    let mut lexer = crate::cos::Lexer::new(data, 0);
533    let mut operands = [0.0; 6];
534    for slot in &mut operands {
535        lexer.skip_ws();
536        *slot = std::str::from_utf8(lexer.token()).ok()?.parse::<f64>().ok().filter(|v| v.is_finite())?;
537    }
538    lexer.skip_ws();
539    (lexer.token() == b"d1").then(|| [operands[2], operands[3], operands[4], operands[5]])
540}

542/// The lowest and highest text-space y of the glyph-space box `r` under a font matrix whose
543/// y column is `b`, `d`, `f` (§9.2.4): a flipped matrix (`d < 0`) turns the box over. `None`
544/// when the box has no height in text space or the numbers overflow; clamped to
545/// [`MAX_TYPE3_EXTENT`].
546fn text_y_extent(r: [f64; 4], b: f64, d: f64, f: f64) -> Option<(f64, f64)> {
547    let ys = [(r[0], r[1]), (r[2], r[1]), (r[0], r[3]), (r[2], r[3])].map(|(x, y)| b * x + d * y + f);
548    let lo = ys.iter().copied().fold(f64::INFINITY, f64::min);
549    let hi = ys.iter().copied().fold(f64::NEG_INFINITY, f64::max);
550    (lo.is_finite() && hi.is_finite() && hi > lo).then_some((lo.max(-MAX_TYPE3_EXTENT), hi.min(MAX_TYPE3_EXTENT)))
551}

553/// Code → Unicode of a font. Explicit mappings and sequential ranges are kept as declared, so a range
554/// costs one record however many codes it covers; where two overlap, the one declared later wins.
555#[derive(Clone, Debug, Default, PartialEq)]
556struct Unicode {
557    /// Simple fonts: the encoding table and `/Differences`, which ToUnicode overrides.
558    encoding: HashMap<u32, String>,
559    /// `bfchar` and array-form entries: the text of each code, with the declaration order of its last one.
560    explicit: HashMap<u32, (u32, String)>,
561    /// Sequential `bfrange` records, in declaration order.
562    ranges: Vec<BfRange>,
563    /// Which record covers each stretch of codes; see [`cover_index`].
564    cover: Vec<(u64, Option<usize>)>,
565    /// The latest-declared range covering each code below 65 536, as an index into `ranges` (`u32::MAX` for
566    /// none); empty without ranges. See [`owners_below_64k`].
567    low: Vec<u32>,
568    /// One bit per code below 65 536 that has an explicit entry; empty without any. A miss then skips the
569    /// hash lookup. See [`bit_set_below_64k`].
570    explicit_low: Vec<u64>,
571}
572
573/// A sequential `bfrange`: code `lo + k` (up to `hi`) maps to `units` with its last unit moved on by `k`.
574#[derive(Clone, Debug, PartialEq)]
575struct BfRange {
576    lo: u32,
577    hi: u32,
578    /// Declaration order, shared with the explicit entries.
579    seq: u32,
580    units: Box<[u16]>,
581}
582
583impl Unicode {
584    /// Whether `code` may have an explicit entry; false is certain for codes below 65 536 without one.
585    #[inline]
586    fn may_be_explicit(&self, code: u32) -> bool {
587        code >= 0x1_0000 || self.explicit_low.get(code as usize / 64).is_some_and(|w| (*w >> (code % 64)) & 1 == 1)
588    }
589
590    /// The latest-declared range that covers `code`.
591    #[inline]
592    fn covering(&self, code: u32) -> Option<&BfRange> {
593        if code < 0x1_0000 {
594            return self.ranges.get(*self.low.get(code as usize)? as usize);
595        }
596        let step = self.cover.partition_point(|&(start, _)| start <= u64::from(code));
597        let &(_, record) = self.cover.get(step.checked_sub(1)?)?;
598        self.ranges.get(record?)
599    }
600
601    /// The text of `code`: its explicit entry, unless a range declared after it covers the code;
602    /// otherwise the covering range, otherwise the encoding.
603    #[inline]
604    fn text(&self, code: u32) -> Option<Cow<'_, str>> {
605        let range = self.covering(code);
606        let explicit = if self.may_be_explicit(code) { self.explicit.get(&code) } else { None };
607        if let Some((seq, text)) = explicit
608            && range.is_none_or(|r| r.seq < *seq)
609        {
610            return Some(Cow::Borrowed(text.as_str()));
611        }
612        match range {
613            Some(r) => Some(Cow::Owned(r.text_at(code))),
614            None => self.encoding.get(&code).map(|t| Cow::Borrowed(t.as_str())),
615        }
616    }
617}
618
619impl BfRange {
620    /// The text of `code`, which lies in this range.
621    fn text_at(&self, code: u32) -> String {
622        // The offset is below 65 536 (the width limit in `parse_to_unicode`), so truncating it is exact.
623        let offset = code.saturating_sub(self.lo) as u16;
624        let last = self.units.len().saturating_sub(1);
625        let units = self.units.iter().enumerate().map(|(i, &u)| if i == last { u.wrapping_add(offset) } else { u });
626        char::decode_utf16(units).map(|r| r.unwrap_or(char::REPLACEMENT_CHARACTER)).collect()
627    }
628
629    /// The lowest code of this range whose own text is `c` and that `valid` accepts. Only a single
630    /// unit (a BMP character, or U+FFFD through a surrogate unit) or a surrogate pair can decode to
631    /// one character `c`; longer destinations always decode to several.
632    fn first_code(&self, c: char, valid: impl Fn(u32) -> bool) -> Option<u32> {
633        let width = self.hi.checked_sub(self.lo)?;
634        match &*self.units {
635            [base] if c == '\u{FFFD}' => {
636                // The offsets whose unit is a surrogate or U+FFFD, in order; each decodes to U+FFFD.
637                let mut cursor = 0u32;
638                loop {
639                    let unit = (u32::from(*base) + cursor) & 0xFFFF;
640                    let offset = cursor.checked_add(match unit {
641                        0..=0xD7FF => 0xD800 - unit,
642                        0xD800..=0xDFFF | 0xFFFD => 0,
643                        0xE000..=0xFFFC => 0xFFFD - unit,
644                        _ => 0x1_0000 - unit + 0xD800,
645                    })?;
646                    if offset > width {
647                        return None;
648                    }
649                    if let Some(code) = self.lo.checked_add(offset)
650                        && valid(code)
651                    {
652                        return Some(code);
653                    }
654                    cursor = offset.checked_add(1)?;
655                }
656            }
657            [base] => {
658                let offset = u32::from(u16::try_from(u32::from(c)).ok()?.wrapping_sub(*base));
659                let code = self.lo.checked_add(offset).filter(|_| offset <= width)?;
660                valid(code).then_some(code)
661            }
662            [lead, last] => {
663                let v = u32::from(c).checked_sub(0x1_0000)?;
664                if u32::from(*lead) != 0xD800 + (v >> 10) {
665                    return None;
666                }
667                let low = u16::try_from(0xDC00 + (v & 0x3FF)).ok()?;
668                let offset = u32::from(low.wrapping_sub(*last));
669                let code = self.lo.checked_add(offset).filter(|_| offset <= width)?;
670                valid(code).then_some(code)
671            }
672            _ => None,
673        }
674    }
675}

677/// A token of a CMap stream (PostScript syntax, ISO 32000-2 §9.10.3).
678#[derive(Debug, PartialEq)]
679enum Tok<'a> {
680    /// A `<…>` string; `None` when it isn't an even number of hex digits.
681    Hex(Option<Vec<u8>>),
682    Open,
683    Close,
684    /// An operator, number or name (with its `/`), a `(…)` string or a `<<`/`>>`.
685    Word(&'a [u8]),
686}
687
688/// Splits a CMap into tokens without assuming whitespace between them (`<01><0041>` is two
689/// strings), skipping `%` comments.
690struct Lexer<'a> {
691    data: &'a [u8],
692    pos: usize,
693}
694
695impl<'a> Lexer<'a> {
696    fn new(data: &'a [u8]) -> Self {
697        Self { data, pos: 0 }
698    }
699
700    fn at(&self, i: usize) -> Option<u8> {
701        self.data.get(i).copied()
702    }
703}
704
705fn is_cmap_space(b: u8) -> bool {
706    matches!(b, 0 | b'\t' | b'\n' | 0x0c | b'\r' | b' ')
707}
708
709fn is_cmap_delimiter(b: u8) -> bool {
710    matches!(b, b'(' | b')' | b'<' | b'>' | b'[' | b']' | b'{' | b'}' | b'/' | b'%')
711}
712
713impl<'a> Iterator for Lexer<'a> {
714    type Item = Tok<'a>;
715
716    fn next(&mut self) -> Option<Tok<'a>> {
717        loop {
718            let b = self.at(self.pos)?;
719            if is_cmap_space(b) {
720                self.pos += 1;
721            } else if b == b'%' {
722                while self.at(self.pos).is_some_and(|c| c != b'\n' && c != b'\r') {
723                    self.pos += 1;
724                }
725            } else {
726                break;
727            }
728        }
729        let start = self.pos;
730        let b = self.at(start)?;
731        self.pos += 1;
732        match b {
733            b'[' => Some(Tok::Open),
734            b']' => Some(Tok::Close),
735            b'<' | b'>' if self.at(self.pos) == Some(b) => {
736                self.pos += 1;
737                Some(Tok::Word(self.data.get(start..self.pos)?))
738            }
739            b'<' => {
740                let mut digits = Vec::new();
741                let mut valid = true;
742                loop {
743                    match self.at(self.pos) {
744                        // Unterminated: what's left is one bad string.
745                        None => return Some(Tok::Hex(None)),
746                        Some(b'>') => break,
747                        Some(c) if is_cmap_space(c) => {}
748                        Some(c) => match (c as char).to_digit(16) {
749                            // A hex digit is below 16, so it fits a u8.
750                            Some(d) => digits.push(d as u8),
751                            None => valid = false,
752                        },
753                    }
754                    self.pos += 1;
755                }
756                self.pos += 1;
757                if !valid || digits.is_empty() || !digits.len().is_multiple_of(2) {
758                    return Some(Tok::Hex(None));
759                }
760                // `chunks` never yields an empty slice, so `p[0]` exists.
761                Some(Tok::Hex(Some(digits.chunks(2).map(|p| p[0] << 4 | p.get(1).copied().unwrap_or(0)).collect())))
762            }
763            b'(' => {
764                // Literal strings nest and escape their parentheses; only skipping them matters.
765                let mut depth = 1usize;
766                while depth > 0 {
767                    match self.at(self.pos) {
768                        None => break,
769                        Some(b'\\') => self.pos += 1,
770                        Some(b'(') => depth += 1,
771                        Some(b')') => depth -= 1,
772                        _ => {}
773                    }
774                    self.pos += 1;
775                }
776                Some(Tok::Word(self.data.get(start..self.pos.min(self.data.len()))?))
777            }
778            _ => {
779                // A name keeps its `/`; other delimiters (`)`, `>`, `{`, `}`) stand alone.
780                if b == b'/' || !is_cmap_delimiter(b) {
781                    while self.at(self.pos).is_some_and(|c| !is_cmap_space(c) && !is_cmap_delimiter(c)) {
782                        self.pos += 1;
783                    }
784                }
785                Some(Tok::Word(self.data.get(start..self.pos)?))
786            }
787        }
788    }
789}

792    b.iter().fold(0u32, |a, x| (a << 8) | u32::from(*x))
793}

804/// The next entry of a `begin… end…` section: `N` tokens, or `None` at its `end` operator or
805/// the end of the stream.
806fn entry<'a, const N: usize>(toks: &mut Lexer<'a>, end: &[u8]) -> Option<[Tok<'a>; N]> {
807    let mut out = Vec::with_capacity(N);
808    for _ in 0..N {
809        match toks.next()? {
810            Tok::Word(w) if w == end => return None,
811            t => out.push(t),
812        }
813    }
814    out.try_into().ok()
815}

818    match t {

838/// Codespace ranges and CID mappings from an embedded CMap stream, at most
839/// [`MAX_CMAP_RANGES`] of each (every glyph searches them).
840#[allow(clippy::type_complexity)]
841fn parse_cmap(data: &[u8]) -> (Vec<(usize, Vec<u8>, Vec<u8>)>, Vec<(u32, u32, u32)>) {
842    let (mut spaces, mut cids) = (Vec::new(), Vec::new());
843    let mut toks = Lexer::new(data);
844    while let Some(t) = toks.next() {
845        match t {
846            Tok::Word(b"begincodespacerange") => {
847                while let Some([lo, hi]) = entry(&mut toks, b"endcodespacerange") {
848                    if let (Tok::Hex(Some(lo)), Tok::Hex(Some(hi))) = (lo, hi)
849                        && lo.len() == hi.len()
850                        && spaces.len() < MAX_CMAP_RANGES
851                    {
852                        spaces.push((lo.len(), lo, hi));
853                    }
854                }
855            }
856            Tok::Word(b"begincidrange") => {
857                while let Some([lo, hi, c]) = entry(&mut toks, b"endcidrange") {
858                    if let (Tok::Hex(Some(lo)), Tok::Hex(Some(hi)), Some(c)) = (&lo, &hi, number(&c))
859                        && cids.len() < MAX_CMAP_RANGES
860                    {
861                        cids.push((be(lo), be(hi), c));
862                    }
863                }
864            }
865            Tok::Word(b"begincidchar") => {
866                while let Some([code, c]) = entry(&mut toks, b"endcidchar") {
867                    if let (Tok::Hex(Some(code)), Some(c)) = (&code, number(&c))
868                        && cids.len() < MAX_CMAP_RANGES
869                    {
870                        cids.push((be(code), be(code), c));
871                    }
872                }
873            }
874            _ => {}
875        }
876    }
877    spaces.sort_by_key(|s| s.0);
878    (spaces, cids)
879}

913    /// every call: a caller that reads many fonts, or one font many times, shares an allowance
914    /// through [`Metrics::from_dict_within`].

917        Self::from_dict_within(doc, font, &mut DecodeBudget::new(usize::MAX, 0))

920    /// [`Metrics::from_dict`], decoding against `budget`, which the caller shares across the fonts
921    /// it reads (a page's, say): each stream pays for what it decodes, usable or not. A stream the
922    /// allowance can't pay for is left unread, and [`Metrics::past_allowance`] says so.
923    pub fn from_dict_within(doc: &Document, font: &Dict, budget: &mut DecodeBudget) -> Self {
924        let mut reads = Reads::new(doc, budget);
925        let mut m = Self::read_metrics(doc, font, &mut reads);
926        let base = base_font_name(doc, font);
927        m.subset = base.len() > 7 && base.as_bytes()[6] == b'+' && base[..6].bytes().all(|b| b.is_ascii_uppercase());

929        m.bold |= bold;
930        m.italic |= italic;
931        m.base_font = base;
932        if m.composite && m.subset {
933            m.outlined_cids = cids_with_outlines(doc, font, &mut reads);
934        }
935        m.code_len = match &m.codes {
936            Codes::One => 1,
937            Codes::Two => 2,
938            Codes::Ranges(r) => r.last().map_or(2, |x| x.0),
939        };
940        m.unicode = unicode_map(doc, font, m.composite, &mut reads);
941        if m.std14.is_some() {
942            embedded_widths(doc, font, &mut m, &mut reads);
943        }
944        m.past_allowance = reads.refused;
945        let mut reverse = HashMap::new();
946        for (&code, text) in m.unicode.explicit.iter().map(|(c, (_, t))| (c, t)).chain(m.unicode.encoding.iter()) {
947            if let Some(ch) = only_char(text)
948                && m.shows(code, ch)
949            {
950                keep(&mut reverse, ch, code);
951            }
952        }
953        m.explicit_reverse = reverse;
954        // Only a font known to be laid out in visual order counts: a missing GSUB proves nothing
955        // (bare CFF and subset fonts often drop it), and the font program is never decoded here.
956        if legacy_indic_font(&m.base_font) {
957            let indic = |c: char| matches!(u32::from(c), 0x0900..=0x0D7F);
958            m.visual_order = m.unicode.explicit.values().any(|(_, t)| t.chars().any(indic))
959                || m.unicode.encoding.values().any(|t| t.chars().any(indic))
960                || m.unicode.ranges.iter().any(|r| r.units.iter().any(|&u| matches!(u, 0x0900..=0x0D7F)));
961        }
962        m
963    }

965    fn read_metrics(doc: &Document, font: &Dict, reads: &mut Reads<'_>) -> Self {
966        let mut m = Metrics::fallback();
967        m.std14 = None;
968        let subtype = font.name(b"Subtype").unwrap_or(b"Type1").to_vec();
969        let descriptor;
970        // A Type 3 font measured from its glyph descriptions' `d1` boxes or its `/FontBBox` (see below).
971        let mut type3_measured = false;
972        if subtype == b"Type0" {
973            m.composite = true;
974            m.default = 1000.0;
975            m.codes = Codes::Two;
976            if let Some(e) = font.get(b"Encoding").map(|e| doc.resolve(e))
977                && let Object::Stream(s) = &*e
978                && let Some(data) = reads.decode(s, MAX_CMAP_BYTES)
979            {
980                let (spaces, cids) = parse_cmap(&data);
981                if !spaces.is_empty() {
982                    m.codes = Codes::Ranges(spaces);
983                }
984                m.cid_map = Spans::new(cids);
985            }
986            let desc = font
987                .get(b"DescendantFonts")
988                .map(|d| doc.resolve(d))
989                .and_then(|a| a.as_array().and_then(|a| a.first().cloned()))
990                .and_then(|d| doc.resolve(&d).as_dict().cloned())
991                .unwrap_or_default();
992            if let Some(dw) = desc.get(b"DW").and_then(|d| doc.resolve(d).as_f64()) {
993                m.default = dw;
994                m.explicit_default = true;
995            }
996            if let Some(w) = desc.get(b"W").map(|w| doc.resolve(w)).and_then(|w| w.as_array().cloned()) {
997                let mut ranges = Vec::new();
998                let mut i = 0;
999                while i < w.len() {
1000                    let Some(c0) = doc.resolve(&w[i]).as_f64().map(|v| v as u32) else { break };
1001                    match w.get(i + 1).map(|o| doc.resolve(o)) {
1002                        Some(a) if a.as_array().is_some() => {
1003                            for (k, x) in a.as_array().into_iter().flatten().enumerate() {
1004                                // A run past the last CID ends there: wrapping would give CID 0 its width.
1005                                let Some(cid) = u32::try_from(k).ok().and_then(|k| c0.checked_add(k)) else { break };
1006                                if let Some(v) = doc.resolve(x).as_f64() {
1007                                    m.cid_widths.insert(cid, v);
1008                                }
1009                            }
1010                            i += 2;
1011                        }
1012                        Some(c1) => {
1013                            if let (Some(c1), Some(v)) = (c1.as_f64(), w.get(i + 2).and_then(|x| doc.resolve(x).as_f64()))
1014                                && ranges.len() < MAX_CMAP_ENTRIES
1015                            {
1016                                ranges.push((c0, c1 as u32, v));
1017                            }
1018                            i += 3;
1019                        }
1020                        None => break,
1021                    }
1022                }
1023                m.cid_ranges = Spans::new(ranges);
1024            }
1025            descriptor = dict(doc, desc.get(b"FontDescriptor"));
1026        } else {
1027            m.first = font.get(b"FirstChar").and_then(|f| doc.resolve(f).as_f64()).unwrap_or(0.0).max(0.0) as u32;
1028            m.widths = nums(doc, font.get(b"Widths"));
1029            descriptor = dict(doc, font.get(b"FontDescriptor"));
1030            m.default = descriptor.as_ref().and_then(|d| d.get(b"MissingWidth")).and_then(|w| doc.resolve(w).as_f64()).unwrap_or(0.0);
1031            if subtype == b"Type3" {
1032                if let [a, b, _, d, _, f] = nums(doc, font.get(b"FontMatrix"))[..] {
1033                    m.scale = a;
1034                    // The glyphs' own boxes first: a bitmap font's `/FontBBox` is often all zero
1035                    // or one unit high, while each `d1` declares the box its glyph fills.
1036                    // Each `d1` box "shall be correct" (Table 111), so their union is used as
1037                    // declared, only widened to reach the baseline: a font of small glyphs gets a
1038                    // small box. A `/FontBBox` is often one unit high, so it keeps the 0.5 / 0.1 em
1039                    // minimum.
1040                    // A union cut short by a limit may miss glyphs that are drawn, so a nonzero
1041                    // `/FontBBox` comes first then; all zero, the partial union is still better
1042                    // than no box. A box with no height in text space gives way to the next.
1043                    let glyphs = type3_glyph_box(doc, font, reads);
1044                    // All zero: no assumption about glyph sizes (Table 110).
1045                    let bbox = <[f64; 4]>::try_from(nums(doc, font.get(b"FontBBox"))).ok().filter(|r| r.iter().any(|&v| v != 0.0));
1046                    let union = |complete: bool| glyphs.filter(|g| g.1 == complete).map(|(r, _)| (r, 0.0, 0.0));
1047                    let candidates = [union(true), bbox.map(|r| (r, 0.5, -0.1)), union(false)];
1048                    if let Some(((lo, hi), min_ascent, max_descent)) =
1049                        candidates.into_iter().flatten().find_map(|(r, up, down)| text_y_extent(r, b, d, f).map(|e| (e, up, down)))
1050                    {
1051                        m.ascent = hi.max(min_ascent);
1052                        m.descent = lo.min(max_descent);
1053                        type3_measured = true;
1054                    }
1055                }
1056            } else if m.widths.is_empty() {
1057                let base = String::from_utf8_lossy(font.name(b"BaseFont").unwrap_or(b"")).into_owned();
1058                // The real face, bold and italic included: the published metrics differ by style,
1059                // so Times-Bold is not Times-Roman and neither is Helvetica scaled.
1060                m.std14 = crate::fonts::Std14::from_base_font(&base).or(Some(crate::fonts::Std14::Helvetica));
1061            }
1062        }
1063        if let Some(d) = descriptor {
1064            let flags = descriptor_flags(doc, &d);
1065            let angle = d.get(b"ItalicAngle").and_then(|v| doc.resolve(v).as_f64()).unwrap_or(0.0);
1066            let weight = d.get(b"FontWeight").and_then(|v| doc.resolve(v).as_f64()).unwrap_or(0.0);
1067            m.italic |= flags & 64 != 0 || angle.abs() > 0.1;
1068            m.bold |= flags & 262_144 != 0 || weight >= 600.0;
1069            // Type 3 descriptors may omit these metrics; the glyph-space FontBBox still applies.
1070            let ascent = d.get(b"Ascent").and_then(|v| doc.resolve(v).as_f64());
1071            let descent = d.get(b"Descent").and_then(|v| doc.resolve(v).as_f64());
1072            // Fonts often claim 0; never shrink the glyph box below a sensible minimum.
1073            // A Type 3 font measured from its glyphs' `d1` boxes or its `/FontBBox` keeps those:
1074            // they are in glyph space and go through its font matrix, while a descriptor's
1075            // /Ascent and /Descent are optional for Type 3 (Table 120) and read here as 1000-unit
1076            // values, whatever the matrix.
1077            if (ascent.is_some() && !type3_measured) || subtype != b"Type3" {
1078                let a = ascent.unwrap_or(0.0) / 1000.0;
1079                m.ascent = if a > 0.3 { a.min(1.5) } else { 0.9 };
1080            }
1081            if (descent.is_some() && !type3_measured) || subtype != b"Type3" {
1082                let de = descent.unwrap_or(0.0) / 1000.0;
1083                m.descent = if de < -0.05 { de.max(-0.6) } else { -0.25 };
1084            }
1085        }
1086        m
1087    }

1091        let mut out = Vec::with_capacity(s.len());

1093        while i < s.len() {

1095                Codes::One => 1,
1096                Codes::Two => 2,
1097                Codes::Ranges(r) => r
1098                    .iter()
1099                    .find(|(len, lo, hi)| s.get(i..i + len).is_some_and(|b| b.iter().zip(lo.iter().zip(hi)).all(|(x, (l, h))| x >= l && x <= h)))
1100                    .map_or(r.first().map_or(1, |x| x.0), |x| x.0),

1102            let len = len.min(s.len() - i).max(1);
1103            out.push((be(&s[i..i + len]), len));
1104            i += len;

1116    /// The advance of `code` in text space per unit of font size.
1117    pub fn width(&self, code: u32) -> f64 {
1118        if self.composite {
1119            let cid = self.cid(code);
1120            let w = self.cid_widths.get(&cid).copied().or_else(|| self.cid_ranges.find(cid).map(|r| r.2));
1121            return w.unwrap_or(self.default) * self.scale;
1122        }
1123        if let Some(w) = code.checked_sub(self.first).and_then(|i| self.widths.get(i as usize)) {
1124            return w * self.scale;

1126        match self.std14 {
1127            // A font with no /Widths is one of the standard 14; use its published metrics.

1129            Some(f) => match u8::try_from(self.std14_code(f, code)).map(|c| f.width(c)).unwrap_or(0.0) {
1130                0.0 => self.default * self.scale,
1131                w => w / 1000.0,
1132            },
1133            None => self.default * self.scale,
1134        }

1166    /// Whether the font has a glyph for `code` (subset fonts lack the glyphs they don't use).
1167    /// Only asked for codes the font's Unicode mapping gives: a subset's `ToUnicode` lists the
1168    /// glyphs it kept, while its `/W` leaves out those of the default width (#1443).
1169    fn has_glyph(&self, code: u32) -> bool {
1170        if self.composite {
1171            let cid = self.cid(code);
1172            // CID 0 is `.notdef` (an unmapped code lands there too), whatever ToUnicode says: Word
1173            // maps it to a space and gives it a width, and typing with it draws boxes (#3223).
1174            if cid == 0 {
1175                return false;
1176            }
1177            // A CID /W leaves out takes /DW, and Skia (Chrome) leaves out every CID of that width: for
1178            // those, only the subset's program says whether the glyph is there.
1179            return self.cid_widths.contains_key(&cid)
1180                || self.cid_ranges.find(cid).is_some()
1181                || !self.subset
1182                || self.outlined_cids.get(cid as usize / 64).is_some_and(|w| (*w >> (cid % 64)) & 1 == 1)
1183                // No readable TrueType program to ask (e.g. CFF or missing): an explicit /DW
1184                // still says /W may leave out glyphs of that width (#1443).
1185                || (self.outlined_cids.is_empty() && self.explicit_default);
1186        }
1187        match code.checked_sub(self.first).and_then(|i| self.widths.get(i as usize)) {
1188            // A full font keeps its space whatever `/Widths` says; a subset that gives its space no width has no
1189            // space glyph (pdfTeX writes 0 for the codes it never uses), and a typed space drawn as code 32 would
1190            // glue the words together (#6131): the edit sets it as a positioning gap instead. Only when code 32
1191            // is a space: another glyph a subset puts there keeps its code.
1192            Some(w) => *w > 0.0 || (code == 32 && !(self.subset && self.text_of(code).is_some_and(|t| t.chars().all(char::is_whitespace)))),
1193            None => self.widths.is_empty() && !self.subset,
1194        }
1195    }

1198    /// Type 0 font over a readable TrueType program, where a dropped glyph is left empty; `None`
1199    /// when the program can't tell, for example a simple or CFF font. [`Metrics::encode`] accepts
1200    /// any code with a `/W` entry (#1443), so a caller that would rather use another subset of the
1201    /// same face than draw nothing asks this too. An empty space glyph is normal.
1202    pub fn has_outline(&self, code: u32) -> Option<bool> {
1203        if !self.composite || self.outlined_cids.is_empty() {
1204            return None;
1205        }

1213    }

1215    /// The bytes that show `text` in this font, or `None` if some character has no code or
1216    /// no glyph in it, or the metrics were read past their allowance (the caller then substitutes
1217    /// another font). Empty text needs no verified glyph and always encodes.
1218    pub fn encode(&self, text: &str) -> Option<Vec<u8>> {
1219        // A refused read cannot establish the mappings or glyphs needed to write new text.
1220        // Empty text needs neither and remains usable for erasing existing text.
1221        if self.past_allowance && !text.is_empty() {
1222            return None;
1223        }
1224        // The lowest code showing each wanted character. Explicit entries and the encoding are
1225        // checked one by one; each range is asked for the wanted characters it can show.
1226        let wanted: HashSet<char> = text.chars().collect();
1227        let mut reverse: HashMap<char, u32> = HashMap::new();
1228        for &ch in &wanted {
1229            if let Some(&code) = self.explicit_reverse.get(&ch) {
1230                keep(&mut reverse, ch, code);
1231            }
1232        }
1233        for range in &self.unicode.ranges {
1234            for &c in &wanted {
1235                if let Some(code) = range.first_code(c, |m| self.shows(m, c)) {
1236                    keep(&mut reverse, c, code);
1237                }
1238            }
1239        }
1240        let mut out = Vec::with_capacity(text.len() * self.code_len);
1241        for ch in text.chars() {
1242            let code = *reverse.get(&ch)?;
1243            let bytes = code.to_be_bytes();
1244            out.extend_from_slice(&bytes[4 - self.code_len.clamp(1, 4)..]);
1245        }
1246        Some(out)
1247    }

1316/// Code spans `(lo, hi, value)` in declaration order, where the first-declared span covering a code
1317/// wins, indexed so that a lookup is one binary search.
1318#[derive(Clone, Debug, Default, PartialEq)]
1319struct Spans<T> {
1320    spans: Vec<(u32, u32, T)>,
1321    /// The span covering each stretch of codes; see [`stretches`].
1322    cover: Vec<(u64, Option<usize>)>,
1323}
1324
1325impl<T> Spans<T> {
1326    fn new(spans: Vec<(u32, u32, T)>) -> Self {
1327        let bounds: Vec<(u32, u32)> = spans.iter().map(|s| (s.0, s.1)).collect();
1328        Spans { cover: stretches(&bounds, false), spans }
1329    }
1330
1331    fn is_empty(&self) -> bool {
1332        self.spans.is_empty()
1333    }
1334
1335    /// The first-declared span covering `code`.
1336    fn find(&self, code: u32) -> Option<&(u32, u32, T)> {
1337        let step = self.cover.partition_point(|&(start, _)| {
1338            count_span_reads();
1339            start <= u64::from(code)
1340        });
1341        let &(_, record) = self.cover.get(step.checked_sub(1)?)?;
1342        self.spans.get(record?)
1343    }
1344}

1346/// For each stretch of codes, from its first code on, the latest-declared range covering it (`None`
1347/// where no range does), sorted by code; a lookup is one binary search over it.
1348fn cover_index(ranges: &[BfRange]) -> Vec<(u64, Option<usize>)> {
1349    let spans: Vec<(u32, u32)> = ranges.iter().map(|r| (r.lo, r.hi)).collect();
1350    stretches(&spans, true)
1351}

1353/// For each stretch of codes, from its first code on, the span `(lo, hi)` covering it (`None` where
1354/// none does), sorted by code: the latest-declared one when `later_wins`, else the first-declared.
1355/// At most `u32::MAX` spans are indexed.
1356fn stretches(spans: &[(u32, u32)], later_wins: bool) -> Vec<(u64, Option<usize>)> {
1357    // Heap keys, largest for the winning record; the mapping is its own inverse.
1358    let key = |record: u32| if later_wins { record } else { !record };
1359    // Openings (first code, record) and closings (one past the last code), each in code order. The cover
1360    // can only change at one of their codes, so the sweep visits those codes in order.
1361    let mut opens: Vec<(u32, u32)> = spans.iter().zip(0..u32::MAX).map(|(&(lo, _), i)| (lo, i)).collect();
1362    opens.sort_unstable();
1363    let mut closes: Vec<u64> = spans.iter().take(opens.len()).map(|&(_, hi)| u64::from(hi) + 1).collect();
1364    closes.sort_unstable();
1365    // Keys of the records that have opened and not yet closed; the top is the winning one.
1366    let mut open: BinaryHeap<u32> = BinaryHeap::new();
1367    let mut cover: Vec<(u64, Option<usize>)> = Vec::new();
1368    let (mut next_open, mut next_close) = (0, 0);
1369    loop {
1370        let p = match (opens.get(next_open), closes.get(next_close)) {
1371            (Some(&(lo, _)), Some(&end)) => u64::from(lo).min(end),
1372            (Some(&(lo, _)), None) => u64::from(lo),
1373            (None, Some(&end)) => end,
1374            (None, None) => break,
1375        };
1376        while let Some(&(lo, record)) = opens.get(next_open)
1377            && u64::from(lo) <= p
1378        {
1379            open.push(key(record));
1380            next_open += 1;
1381        }
1382        while closes.get(next_close).is_some_and(|&end| end <= p) {
1383            next_close += 1;
1384        }
1385        while open.peek().is_some_and(|&top| spans.get(key(top) as usize).is_none_or(|&(_, hi)| u64::from(hi) < p)) {
1386            open.pop();
1387        }
1388        let winner = open.peek().map(|&top| key(top) as usize);
1389        if cover.last().map(|&(_, w)| w) != Some(winner) {
1390            cover.push((p, winner));
1391        }
1392    }
1393    cover
1394}

1396/// One bit for each code below 65 536 in `codes`; empty when there is none.
1397/// Sized to the highest such code, so a font with a few low codes costs a few words, not 8 KB.
1398fn bit_set_below_64k(codes: impl Iterator<Item = u32>) -> Vec<u64> {
1399    let low: Vec<u32> = codes.filter(|&code| code < 0x1_0000).collect();
1400    let Some(&max) = low.iter().max() else { return Vec::new() };
1401    let mut bits = vec![0u64; max as usize / 64 + 1];
1402    for code in low {
1403        if let Some(word) = bits.get_mut(code as usize / 64) {
1404            *word |= 1u64 << (code % 64);
1405        }
1406    }
1407    bits
1408}

1410/// The latest-declared range covering each code below 65 536, read off the stretches of `cover`; empty
1411/// when no range covers such a code. The stretches are disjoint, so this takes at most 65 536 steps
1412/// however many ranges there are.
1413///
1414/// The table ends with the last covered code: a code past it has no covering range, and a font whose
1415/// ranges cover a few low codes costs that many entries, not a fixed 256 KB.
1416fn owners_below_64k(cover: &[(u64, Option<usize>)]) -> Vec<u32> {
1417    let stretch_end = |i: usize| cover.get(i + 1).map_or(0x1_0000, |&(next, _)| next.min(0x1_0000));
1418    let len = cover
1419        .iter()
1420        .enumerate()
1421        .filter(|&(_, &(start, winner))| winner.is_some() && start < 0x1_0000)
1422        .map(|(i, _)| stretch_end(i))
1423        .max()
1424        .unwrap_or(0);
1425    let mut owner = vec![u32::MAX; usize::try_from(len).unwrap_or(0)];
1426    for (i, &(start, winner)) in cover.iter().enumerate() {
1427        let Some(record) = winner else { continue };
1428        // Record indices are below MAX_CMAP_ENTRIES, which fits in a u32.
1429        let record = record as u32;
1430        for code in start..stretch_end(i) {
1431            if let Some(slot) = owner.get_mut(code as usize) {
1432                *slot = record;
1433            }
1434        }
1435    }
1436    owner
1437}

1444/// [`parse_to_unicode`] with `budget` explicit entries, so tests can use a small one.
1445fn parse_to_unicode_within(data: &[u8], out: &mut Unicode, mut budget: usize) {
1446    let mut seq = 0u32;
1447    let mut toks = Lexer::new(data);
1448    'scan: while let Some(t) = toks.next() {
1449        match t {
1450            Tok::Word(b"beginbfchar") => {
1451                while let Some([c, u]) = entry(&mut toks, b"endbfchar") {
1452                    if let (Tok::Hex(Some(c)), Tok::Hex(Some(u))) = (c, u) {
1453                        let added = add_explicit(out, &mut budget, &mut seq, be(&c), utf16(&u));
1454                        if !added {
1455                            break 'scan;
1456                        }
1457                    }
1458                }
1459            }
1460            Tok::Word(b"beginbfrange") => {
1461                while let Some([lo, hi, dst]) = entry(&mut toks, b"endbfrange") {
1462                    let (Tok::Hex(Some(lo)), Tok::Hex(Some(hi))) = (lo, hi) else { break };
1463                    let (lo, hi) = (be(&lo), be(&hi));
1464                    match dst {
1465                        // One destination per code, from `lo` up to `hi`.
1466                        Tok::Open => {
1467                            let mut code = (hi >= lo).then_some(lo);
1468                            for t in toks.by_ref() {
1469                                let Some(c) = code else {
1470                                    if t == Tok::Close {
1471                                        break;
1472                                    }
1473                                    continue;
1474                                };
1475                                match t {
1476                                    Tok::Close => break,
1477                                    Tok::Hex(Some(u)) => {
1478                                        let added = add_explicit(out, &mut budget, &mut seq, c, utf16(&u));
1479                                        if !added {
1480                                            break 'scan;
1481                                        }
1482                                    }
1483                                    // A bad entry still stands for its code.
1484                                    _ => {}
1485                                }
1486                                code = c.checked_add(1).filter(|c| *c <= hi);
1487                            }
1488                        }
1489                        Tok::Hex(Some(u)) if hi >= lo && hi - lo < 65536 => {
1490                            let added = if lo == hi {
1491                                add_explicit(out, &mut budget, &mut seq, lo, utf16(&u))
1492                            } else {
1493                                add_range(out, &mut budget, &mut seq, lo, hi, code_units(&u).into_boxed_slice())
1494                            };
1495                            if !added {
1496                                break 'scan;
1497                            }
1498                        }
1499                        _ => {}
1500                    }
1501                }
1502            }
1503            _ => {}
1504        }
1505    }
1506    out.cover = cover_index(&out.ranges);
1507    out.low = owners_below_64k(&out.cover);
1508    out.explicit_low = bit_set_below_64k(out.explicit.keys().copied());
1509}

1511/// A glyph name's text, by the Adobe Glyph List *algorithm* — the published naming rules, not
1512/// the AGL data table, which this crate does not hold.
1513///
1514/// A name is cut at its first `.`, because a suffix marks a variant of the same character rather
1515/// than a different one (`a.sc` is small-capital `a`, `one.oldstyle` is still `1`). What remains
1516/// is split on `_` into components, so a ligature name yields several characters (`f_i` is "fi").
1517/// Each component is read as an Annex D name, as `uniXXXX` with one or more four-hex-digit values,
1518/// or as `uXXXX` through `uXXXXXX`.
1519///
1520/// Returns `None` rather than a partial answer: a name is either fully understood or not, and half
1521/// a ligature would be worse than admitting we cannot read it.
1522pub fn glyph_text(name: &str) -> Option<String> {
1523    let stem = name.split('.').next().filter(|s| !s.is_empty())?;
1524    let mut out = String::new();
1525    for part in stem.split('_') {
1526        out.push_str(&glyph_component(part)?);
1527    }
1528    Some(out).filter(|s| !s.is_empty())
1529}
1530
1531/// One `_`-separated component of a glyph name.
1532fn glyph_component(part: &str) -> Option<String> {
1533    if let Ok(i) = crate::fonts::encodings::NAMES.binary_search_by(|(n, _)| (*n).cmp(part)) {
1534        return char::from_u32(crate::fonts::encodings::NAMES[i].1).map(String::from);
1535    }
1536    // `from_str_radix` accepts a leading `+` and lowercase digits, and the glyph list
1537    // specification's digits are uppercase only (`uni20ac` names nothing), so they are checked.
1538    let hexish = |h: &str| !h.is_empty() && h.bytes().all(|b| matches!(b, b'0'..=b'9' | b'A'..=b'F'));
1539    if let Some(hex) = part.strip_prefix("uni").filter(|h| h.len() >= 4 && h.len() % 4 == 0 && hexish(h)) {
1540        let mut s = String::new();
1541        for i in (0..hex.len()).step_by(4) {
1542            // A lone surrogate half is not a character; such a name is not usable text.
1543            s.push(char::from_u32(u32::from_str_radix(hex.get(i..i.saturating_add(4))?, 16).ok()?)?);
1544        }
1545        return Some(s);
1546    }
1547    let hex = part.strip_prefix('u').filter(|h| (4..=6).contains(&h.len()) && hexish(h))?;
1548    u32::from_str_radix(hex, 16).ok().and_then(char::from_u32).map(String::from)
1549}

1551/// [`glyph_text`] for names that stand for exactly one character.
1552pub fn glyph_unicode(name: &str) -> Option<char> {
1553    let mut chars = glyph_text(name)?.chars().collect::<Vec<_>>().into_iter();
1554    chars.next().filter(|_| chars.len() == 0)
1555}

1557/// What each code of an embedded TrueType font that isn't flagged non-symbolic and names no base
1558/// encoding shows, by the `post` table name of the glyph its `cmap` selects (the way the renderer
1559/// selects it: the code, then at 0xF000, 0xF100 and 0xF200). Such a font's codes need not be
1560/// Standard or WinAnsi: FrameMaker files distilled with Distiller 3 number them in the Macintosh
1561/// glyph order, 29 below ASCII, so `Solid` read as `6ROLG` and typing set the wrong glyphs (#3962).
1562/// Codes without a glyph, and fonts whose `post` table names no glyphs, are left out (the latter
1563/// to [`symbolic_mac_roman_texts`]). A code whose glyph has no name with a Unicode value reads
1564/// as WinAnsi, as the renderer reads it (vendored hayro-interpret patch (79)), so the editor and
1565/// extraction agree: `€` for code 0x80, which Standard leaves without a character.
1566fn symbolic_glyph_texts(doc: &Document, font: &Dict, reads: &mut Reads<'_>) -> Vec<(u32, String)> {
1567    use skrifa::GlyphId;
1568    use skrifa::raw::{FontData, FontRead as _, TableProvider as _, tables::cmap::PlatformId, tables::post::Post, types::Tag};
1569    let Some(descriptor) = dict(doc, font.get(b"FontDescriptor")) else { return Vec::new() };
1570    // Flag bit 6, Nonsymbolic (ISO 32000-2 Table 121).
1571    if descriptor_flags(doc, &descriptor) & 32 != 0 {
1572        return Vec::new();
1573    }
1574    let Some(object) = descriptor.get(b"FontFile2") else { return Vec::new() };
1575    let object = doc.resolve(object);
1576    let Object::Stream(stream) = object.as_ref() else { return Vec::new() };
1577    let Some(bytes) = reads.decode(stream, MAX_SFNT) else { return Vec::new() };
1578    let Ok(face) = skrifa::FontRef::from_index(&bytes, 0) else { return Vec::new() };
1579    let Ok(cmap) = face.cmap() else { return Vec::new() };
1580    // A `post` version written with its 16-bit halves swapped (`00 00 00 02` for 2.0, as ScanSoft
1581    // PDF Create! writes it) is read as the version it means, as the renderer reads it (#2946,
1582    // vendored hayro-interpret patch (49)); otherwise its names would be lost here alone.
1583    let repaired = face.table_data(Tag::new(b"post")).and_then(|data| match data.as_bytes() {
1584        [0, 0, 0, n @ 1..=3, rest @ ..] => Some([&[0, *n, 0, 0], rest].concat()),
1585        _ => None,
1586    });
1587    let post = match &repaired {
1588        Some(bytes) => Post::read(FontData::new(bytes)),
1589        None => face.post(),
1590    };
1591    let Ok(post) = post else { return Vec::new() };
1592    // A font that names no glyphs (no `post` table of version 1.0, or one of version 2.0 with an
1593    // empty name index) is left to `symbolic_mac_roman_texts` and the encoding's reading.
1594    let names_glyphs = match post.version() {
1595        skrifa::raw::types::Version16Dot16::VERSION_1_0 => true,
1596        skrifa::raw::types::Version16Dot16::VERSION_2_0 => post.glyph_name_index().is_some_and(|index| !index.is_empty()),
1597        _ => false,
1598    };
1599    if !names_glyphs {
1600        return Vec::new();
1601    }
1602    let custom = post_custom_names(&post);
1603    (0..=255u32)
1604        .filter_map(|code| {
1605            // The records in order, as the renderer's `map_code` reads a symbolic font: a Windows
1606            // symbol or Unicode cmap with the code and the code at 0xF000, 0xF100 and 0xF200, a
1607            // Macintosh or Unicode-platform one with the code alone; the first glyph found wins.
1608            let glyph = cmap.encoding_records().iter().find_map(|record| {
1609                let offsets: &[u32] = match (record.platform_id(), record.encoding_id()) {
1610                    (PlatformId::Windows, 0 | 1) => &[0, 0xF000, 0xF100, 0xF200],
1611                    (PlatformId::Macintosh | PlatformId::Unicode, 0) => &[0],
1612                    _ => return None,
1613                };
1614                let subtable = record.subtable(cmap.offset_data()).ok()?;
1615                offsets.iter().find_map(|o| subtable.map_codepoint(code + o).filter(|g| *g != GlyphId::NOTDEF))
1616            })?;
1617            let glyph = u16::try_from(glyph.to_u32()).ok()?;
1618            let winansi = || {
1619                let u = crate::fonts::encodings::WIN_ANSI.get(usize::try_from(code).ok()?).copied().filter(|u| *u != 0)?;
1620                char::from_u32(u).map(String::from)
1621            };
1622            let text = post_name(&post, &custom, glyph).filter(|n| uppercase_hex_forms(n)).and_then(glyph_text);
1623            Some((code, text.or_else(winansi)?))
1624        })
1625        .collect()
1626}

1628/// Whether the `uni` and `u` components of a glyph name spell their hex digits in uppercase, as
1629/// the glyph list specification's forms do (ISO 32000-2 §9.10.2): `uni1eae` names nothing, so
1630/// its glyph's code reads as WinAnsi, as the renderer reads it (vendored hayro-interpret patch
1631/// (79)). A component that is a list name (`a`, `ecircumflex`) is read as such whatever its case.
1632fn uppercase_hex_forms(name: &str) -> bool {
1633    let stem = name.split('.').next().unwrap_or_default();
1634    stem.split('_').all(|part| {
1635        // Outside the list, a component `glyph_text` reads is `u` or `uni` and hex digits, so a
1636        // lowercase `a` to `f` in it is a lowercase digit.
1637        crate::fonts::encodings::NAMES.binary_search_by(|(n, _)| (*n).cmp(part)).is_ok() || !part.bytes().any(|b| matches!(b, b'a'..=b'f'))
1638    })
1639}

1641/// The names a version 2.0 `post` table spells out, in order, read once. `Post::glyph_name` walks
1642/// this data from its start for each glyph it is asked about, which over a font program of up to
1643/// 16 MiB, asked about 256 codes, is work an untrusted file can make large. Reading stops at the
1644/// first name that can't be read, as each lookup would.
1645fn post_custom_names<'a>(post: &skrifa::raw::tables::post::Post<'a>) -> Vec<&'a str> {
1646    post.string_data().map(|names| names.iter().map_while(|name| name.ok().map(|n| n.as_str())).collect()).unwrap_or_default()
1647}

1649/// `glyph`'s name in `post`, as `Post::glyph_name` gives it, with the table's own names from
1650/// [`post_custom_names`].
1651fn post_name<'a>(post: &skrifa::raw::tables::post::Post<'a>, custom: &[&'a str], glyph: u16) -> Option<&'a str> {
1652    use skrifa::raw::{tables::post::DEFAULT_GLYPH_NAMES, types::Version16Dot16};
1653    match post.version() {
1654        Version16Dot16::VERSION_1_0 => DEFAULT_GLYPH_NAMES.get(usize::from(glyph)).copied(),
1655        Version16Dot16::VERSION_2_0 => {
1656            let index = usize::from(post.glyph_name_index()?.get(usize::from(glyph))?.get());
1657            match index.checked_sub(DEFAULT_GLYPH_NAMES.len()) {
1658                None => DEFAULT_GLYPH_NAMES.get(index).copied(),
1659                Some(i) => custom.get(i).copied(),
1660            }
1661        }
1662        _ => None,
1663    }

1666/// Code → Unicode for a font: its ToUnicode CMap (for a composite font, also the name
1667/// `/Identity-H` or `/Identity-V`, read as code = Unicode), else (simple fonts) its encoding.
1668fn unicode_map(doc: &Document, font: &Dict, composite: bool, reads: &mut Reads<'_>) -> Unicode {
1669    let mut out = Unicode::default();
1670    if !composite {
1671        let enc = font.get(b"Encoding").map(|e| doc.resolve(e));
1672        let base_name = match enc.as_deref() {
1673            Some(Object::Name(n)) => Some(n.clone()),
1674            Some(Object::Dict(d)) => d.name(b"BaseEncoding").map(<[u8]>::to_vec),
1675            _ => None,
1676        };
1677        // Symbol and ZapfDingbats carry their own built-in encodings, which are not Latin
1678        // (ISO 32000-2 9.6.6.1): Symbol code 0x61 is `alpha`, not `a`. We do not hold those two
1679        // tables, so with no base encoding named they get no implicit mapping at all. Reporting
1680        // nothing is honest; reporting Latin letters silently corrupts text extraction, find,
1681        // copy and — because `redact` matches on this map — redaction.
1682        //
1683        // Deliberately keyed on the *name*, not on the `/FontDescriptor` symbolic flag. A great
1684        // many ordinary Latin TrueType subsets set that flag and still rely on a Latin encoding,
1685        // so suppressing on the flag would lose real text on a large share of real documents.
1686        // An explicitly named base encoding is a deliberate statement and is always honoured,
1687        // even on Symbol.
1688        let symbolic_builtin = matches!(
1689            font.name(b"BaseFont").map(|b| crate::fonts::Std14::from_base_font(&String::from_utf8_lossy(b))),
1690            Some(Some(crate::fonts::Std14::Symbol | crate::fonts::Std14::ZapfDingbats))
1691        );
1692        let table: Option<&[u32; 256]> = match base_name.as_deref() {
1693            Some(b"WinAnsiEncoding") => Some(&crate::fonts::encodings::WIN_ANSI),
1694            Some(b"MacRomanEncoding") => Some(&crate::fonts::encodings::MAC_ROMAN),
1695            Some(b"StandardEncoding") => Some(&crate::fonts::encodings::STANDARD),
1696            // MacExpert is small caps, oldstyle figures and fractions, not Latin at these codes,
1697            // and we hold no table for it either.
1698            Some(b"MacExpertEncoding") => None,
1699            _ if symbolic_builtin => None,
1700            // Standard is the default for a simple font that names no base encoding.
1701            _ => Some(&crate::fonts::encodings::STANDARD),
1702        };
1703        for (code, u) in table.into_iter().flatten().enumerate() {
1704            if let Some(c) = char::from_u32(*u).filter(|_| *u != 0) {
1705                out.encoding.insert(code as u32, c.to_string());
1706            }
1707        }
1708        // A font named as Symbol or ZapfDingbats keeps reporting nothing: its names may be as
1709        // un-Latin as its codes (WordPerfect's WPTypographicSymbols calls its apostrophe `equal`).
1710        // A font whose `post` table names its glyphs reads by those names; one that names none
1711        // and keeps the Macintosh order reads as Mac OS Roman, over the WinAnsi fallback.
1712        if base_name.is_none() && !symbolic_builtin {
1713            out.encoding.extend(symbolic_glyph_texts(doc, font, reads));
1714            out.encoding.extend(symbolic_mac_roman_texts(doc, font, reads));
1715        }
1716        if let Some(Object::Dict(d)) = enc.as_deref()
1717            && let Some(diffs) = d.get(b"Differences").map(|x| doc.resolve(x)).and_then(|x| x.as_array().cloned())
1718        {
1719            let mut code = 0u32;
1720            for item in diffs {
1721                match &item {
1722                    // Out-of-range codes map nothing a simple font can show.
1723                    Object::Int(n) => code = u32::try_from((*n).max(0)).unwrap_or(u32::MAX),
1724                    Object::Name(n) => {
1725                        // A name we cannot read removes any inherited meaning for that code,
1726                        // rather than leaving the base encoding's letter standing in for it.
1727                        match glyph_text(&String::from_utf8_lossy(n)) {
1728                            Some(t) => out.encoding.insert(code, t),
1729                            None => out.encoding.remove(&code),
1730                        };
1731                        code = code.saturating_add(1);
1732                    }
1733                    _ => {}
1734                }
1735            }
1736        }
1737    }
1738    match font.get(b"ToUnicode").map(|t| doc.resolve(t)).as_deref() {
1739        Some(Object::Stream(s)) => {
1740            // Reads resolves indirect parameters too, matching the redaction proof's text.
1741            if let Some(data) = reads.decode(s, MAX_CMAP_BYTES) {
1742                parse_to_unicode(&data, &mut out);
1743            }
1744        }
1745        // `/ToUnicode /Identity-H` is not a stream, as ISO 32000-2 §9.10.3 requires, but CAD
1746        // exporters (WSCAD among them) write it for composite fonts whose two-byte codes are the
1747        // UTF-16 code units. Acrobat shows them as code = Unicode, and so does the renderer's
1748        // text layer; without it the text could be seen but not edited or found. The same rule
1749        // (code = Unicode, no controls or lone surrogates) is in the vendored renderer,
1750        // `Type0Font::char_code_to_unicode` in vendor/hayro-interpret/src/font/cid.rs: keep both alike.
1751        Some(Object::Name(n)) if composite && matches!(n.as_slice(), b"Identity-H" | b"Identity-V") => {
1752            parse_to_unicode(&identity_to_unicode(), &mut out);
1753        }
1754        _ => {}
1755    }
1756    out
1757}

1759/// A ToUnicode CMap mapping each two-byte code to the same UTF-16 code unit, leaving out the
1760/// surrogates (which are not characters on their own) and the controls (C0, DEL and C1, as the
1761/// renderer's text layer does). No range crosses a last-byte boundary (§9.10.3).
1762fn identity_to_unicode() -> Vec<u8> {
1763    let mut ranges = vec![(0x20u32, 0x7Eu32), (0xA0, 0xFF)];
1764    ranges.extend((0x01u32..=0xFF).filter(|h| !(0xD8..=0xDF).contains(h)).map(|h| (h << 8, (h << 8) | 0xFF)));
1765    let mut cmap = String::from("1 begincodespacerange <0000> <FFFF> endcodespacerange\n");
1766    // At most 100 entries per section (§9.7.5.2 limits for CMap files).
1767    for chunk in ranges.chunks(100) {
1768        cmap.push_str(&format!("{} beginbfrange\n", chunk.len()));
1769        for (lo, hi) in chunk {
1770            cmap.push_str(&format!("<{lo:04X}> <{hi:04X}> <{lo:04X}>\n"));
1771        }
1772        cmap.push_str("endbfrange\n");
1773    }
1774    cmap.into_bytes()
1775}

1777/// What each code of an embedded TrueType font that isn't flagged non-symbolic, names no base
1778/// encoding and doesn't name its glyphs shows when two things say the code is Mac OS Roman: its
1779/// Macintosh Roman (1, 0) `cmap` subtable selects the glyph the code draws (ISO 32000-2 §9.6.5.4:
1780/// single bytes "look up the associated glyph descriptions" there, and a (3, 0) subtable maps "the
1781/// single-byte codes in the (1, 0) subtable" prepended with 0xF0), and the glyph sits where the
1782/// standard Macintosh glyph order (the 258 names of a version 1.0 `post` table) puts that Mac OS
1783/// Roman character. So code 0x88 drawn with glyph 106, `agrave`, is `à`, where Standard has
1784/// nothing and WinAnsi `ˆ`, while a Windows subset in the same order whose codes are WinAnsi (0x92
1785/// drawn with glyph 183, `quoteright`) keeps its reading. The glyph drawn is found the way the
1786/// renderer finds it: the Windows (3, 0) and (3, 1) subtables first, with the code and the code at
1787/// 0xF000, 0xF100 and 0xF200, then the Macintosh and Unicode-platform ones. The renderer's text
1788/// layer reads the same codes the same way (vendored hayro-interpret patch (101)).
1789fn symbolic_mac_roman_texts(doc: &Document, font: &Dict, reads: &mut Reads<'_>) -> Vec<(u32, String)> {
1790    use skrifa::GlyphId;
1791    use skrifa::raw::{TableProvider as _, tables::cmap::PlatformId, tables::post::DEFAULT_GLYPH_NAMES, types::Tag};
1792    let Some(descriptor) = dict(doc, font.get(b"FontDescriptor")) else { return Vec::new() };
1793    // Flag bit 6, Nonsymbolic (ISO 32000-2 Table 121).
1794    if descriptor_flags(doc, &descriptor) & 32 != 0 {
1795        return Vec::new();
1796    }
1797    let Some(object) = descriptor.get(b"FontFile2") else { return Vec::new() };
1798    let object = doc.resolve(object);
1799    let Object::Stream(stream) = object.as_ref() else { return Vec::new() };
1800    let Some(bytes) = reads.decode(stream, MAX_SFNT) else { return Vec::new() };
1801    let Ok(face) = skrifa::FontRef::from_index(&bytes, 0) else { return Vec::new() };
1802    // A CFF program in `/FontFile2` selects its glyphs by name, not through the `cmap`. A `post`
1803    // table of version 1.0 or 2.0 (also with its 16-bit halves swapped, as the renderer reads it)
1804    // names the glyphs, and a font that names them needn't keep the order: a Wingdings subset
1805    // draws code 0x83 with glyph 101, `Eacute` in the order, which it names `threesans`.
1806    let named = face
1807        .table_data(Tag::new(b"post"))
1808        .is_some_and(|data| matches!(data.as_bytes().get(..4), Some([0, 1, 0, 0] | [0, 2, 0, 0] | [0, 0, 0, 1] | [0, 0, 0, 2])));
1809    if face.cff().is_ok() || named {
1810        return Vec::new();
1811    }
1812    let Ok(cmap) = face.cmap() else { return Vec::new() };
1813    let records = cmap.encoding_records();
1814    let lookup = |code: u32, windows: bool| {
1815        records.iter().find_map(|record| {
1816            let offsets: &[u32] = match (record.platform_id(), record.encoding_id()) {
1817                (PlatformId::Windows, 0 | 1) if windows => &[0, 0xF000, 0xF100, 0xF200],
1818                (PlatformId::Macintosh | PlatformId::Unicode, 0) if !windows => &[0],
1819                _ => return None,
1820            };
1821            let subtable = record.subtable(cmap.offset_data()).ok()?;
1822            offsets.iter().find_map(|o| subtable.map_codepoint(code + o).filter(|g| *g != GlyphId::NOTDEF))
1823        })
1824    };
1825    (0..=255u32)
1826        .filter_map(|code| {
1827            let drawn = lookup(code, true).or_else(|| lookup(code, false))?;
1828            let mac = records.iter().find_map(|record| {
1829                (record.platform_id() == PlatformId::Macintosh && record.encoding_id() == 0)
1830                    .then(|| record.subtable(cmap.offset_data()).ok()?.map_codepoint(code))
1831                    .flatten()
1832            })?;
1833            if mac != drawn {
1834                return None;
1835            }
1836            let c = char::from_u32(*crate::fonts::encodings::MAC_ROMAN.get(usize::try_from(code).ok()?)?).filter(|c| *c != '\0')?;
1837            let standard = *DEFAULT_GLYPH_NAMES.get(usize::try_from(drawn.to_u32()).ok()?)?;
1838            // Mac OS Roman 0xCA reads as a space, for the reason WinAnsi 0xA0 does (`MAC_ROMAN`).
1839            let at_its_place = glyph_unicode(standard) == Some(c) || (code == 0xCA && standard == "nonbreakingspace");
1840            at_its_place.then(|| (code, c.to_string()))
1841        })
1842        .collect()
1843}
