1//! The one chain PdfCraft follows when a document does not embed a font program, and a record of
2//! what it chose and why.
3//!
4//! Substitution used to be decided per caller and never written down: `edit` picked a craft face
5//! by the script of the text it was drawing, `Std14::from_base_font` matched Latin names, and the
6//! renderer did its own thing. Nothing told the *user* that the glyphs they are looking at are not
7//! the ones the document asked for, which is the difference between "PdfCraft shows nothing for my
8//! Chinese PDF" being a mystery and being a sentence.
9//!
10//! This module answers the *dictionary* question — "the document does not carry this program; what
11//! will stand in for it, and why?" — which is what inspection and reporting need. It deliberately
12//! does not answer the *text* question ("which face has a glyph for こ?"), because that depends on
13//! the string being drawn and is already handled where the drawing happens. It delegates to the
14//! existing per-script selectors rather than restating their order, so there is one source of
15//! truth for each.
16//!
17//! The chain is tried in order, and the first step that yields a face wins:
18//!
19//! 1. The document embeds the program. Nothing is substituted.
20//! 2. `/BaseFont` names one of the standard 14 (or a well-known alias such as Arial). Exact
21//!    published metrics apply, so this is as close to right as a substitution gets.
22//! 3. craft-fonts holds that exact family. Never another family — see [`crate::fonts::embedding`].
23//! 4. The descendant font's `/CIDSystemInfo /Ordering` names a script, and craft-fonts has a face
24//!    for it. Today that means Adobe-Japan1.
25//! 5. The ordering names a script we have **no** licensed face for (GB1, CNS1, Korea1). There is
26//!    no honest substitute, so the chain reports none: the caller draws nothing and can say why.
27//! 6. Otherwise, the nearest of the standard 14 by `/FontDescriptor /Flags` — serif, fixed pitch,
28//!    bold and italic. This step always succeeds, so a simple font is never left without a face.
29
30use crate::fonts::{CraftFont, Std14};
31
32/// What a font dictionary asks for, as far as the dictionary alone can say.
33#[derive(Clone, Copy, Debug, Default)]
34pub struct Request<'a> {
35    /// `/BaseFont` with any `ABCDEF+` subset tag already stripped.
36    pub base_font: &'a str,
37    /// Whether the descriptor carries `/FontFile`, `/FontFile2` or `/FontFile3` (or it is Type 3).
38    pub embedded: bool,
39    /// `/FontDescriptor /Flags`, or 0 when there is no descriptor.
40    pub flags: u32,
41    /// The descendant font's `/CIDSystemInfo /Ordering`, for a composite font.
42    pub ordering: Option<&'a str>,
43}
44
45/// The face the chain picked.
46///
47/// `Debug`, `PartialEq` and `Eq` are written by hand rather than derived, because [`CraftFont`]
48/// carries a whole font program: deriving would print megabytes and compare font files.
49#[derive(Clone, Copy)]
50pub enum Face {
51    /// The document's own program; nothing was substituted.
52    Embedded,
53    /// One of the standard 14, whose metrics are exact.
54    Standard14(Std14),
55    /// A face from the optional craft-fonts build input.
56    Craft(&'static CraftFont),
57    /// No face at all. The caller must draw nothing rather than draw the wrong glyphs.
58    None,
59}
60
61impl PartialEq for Face {
62    fn eq(&self, other: &Self) -> bool {
63        match (self, other) {
64            (Self::Embedded, Self::Embedded) | (Self::None, Self::None) => true,
65            (Self::Standard14(a), Self::Standard14(b)) => a == b,
66            // craft-fonts entries are unique statics, so identity is the comparison that means
67            // what we want: two styles of one family are different faces and must not be equal.
68            (Self::Craft(a), Self::Craft(b)) => std::ptr::eq(*a, *b),
69            _ => false,
70        }
71    }
72}
73
74impl Eq for Face {}
75
76impl std::fmt::Debug for Face {
77    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
78        match self {
79            Self::Embedded => f.write_str("Embedded"),
80            Self::Standard14(s) => write!(f, "Standard14({})", s.base_font()),
81            // The name, never the bytes.
82            Self::Craft(c) => write!(f, "Craft({})", c.name()),
83            Self::None => f.write_str("None"),
84        }
85    }
86}

88/// Why the chain picked that face. The variants are in the order the chain tries them.
89#[derive(Clone, Copy, Debug, PartialEq, Eq)]
90pub enum Why {
91    /// The document carries the font program.
92    Embedded,
93    /// `/BaseFont` names a standard-14 face or a well-known alias of one.
94    NamedStandard14,
95    /// `/BaseFont` is otherwise unknown, but its name hints at a family ("Serif", "Roman",
96    /// "Mono"). A guess from the name, and reported as one.
97    NameHintedStandard14,
98    /// craft-fonts holds this exact family.
99    ExactCraftFamily,
100    /// A craft-fonts face for the script named by `/CIDSystemInfo /Ordering`.
101    ScriptFace,
102    /// The ordering names a script with no licensed face available.
103    NoFaceForScript,
104    /// The nearest standard-14 face by descriptor flags.
105    StyleMatched,
106}
107
108impl Why {
109    /// A stable sentence for a report, written for whoever has to act on it.
110    pub fn note(self) -> &'static str {
111        match self {
112            Self::Embedded => "embedded in the document; not substituted",
113            Self::NamedStandard14 => "names one of the standard 14 fonts, or a well-known alias of one, whose metrics are exact",
114            Self::NameHintedStandard14 => {
115                "not embedded and not a known family, but its name hints at one (Serif, Roman or Mono),                  so the nearest standard-14 face was chosen from that hint"
116            }
117            Self::ExactCraftFamily => "this exact family is available, so no other family was chosen",
118            Self::ScriptFace => "substituted by a face for the script the font declares",
119            Self::NoFaceForScript => {
120                "no licensed face is available for the script this font declares, so nothing is drawn for it \
121                 rather than the wrong glyphs"
122            }
123            Self::StyleMatched => "not embedded and not a known family, so the nearest standard-14 face was chosen by its descriptor flags",
124        }
125    }
126
127    /// Whether the document's own glyphs are being shown. Only [`Why::Embedded`] is faithful.
128    pub fn is_faithful(self) -> bool {
129        matches!(self, Self::Embedded)
130    }
131}

133/// What the chain chose, and why.
134#[derive(Clone, Copy, Debug, PartialEq, Eq)]
135pub struct Substitution {
136    pub face: Face,
137    pub why: Why,
138}
139
140impl Substitution {
141    /// The chosen face's name, for a report. `None` when nothing was chosen.
142    pub fn face_name(&self) -> Option<String> {
143        match self.face {
144            Face::Embedded => None,
145            Face::Standard14(f) => Some(f.base_font().to_owned()),
146            Face::Craft(f) => Some(f.name()),
147            Face::None => None,
148        }
149    }
150}
151
152/// CID orderings we recognise, and whether a craft-fonts face may exist for them.
153///
154/// `Identity` is not a script: it says the codes are glyph ids, so it tells us nothing about
155/// which face would suit and the chain falls through to the flags.
156fn script_for(ordering: &str) -> Option<&'static str> {
157    match ordering {
158        "Japan1" => Some("Jpan"),
159        // Registered orderings the chain maps to no craft-fonts face.
160        "GB1" | "CNS1" | "Korea1" | "KR" => Some(""),
161        _ => None,
162    }
163}
164
165/// Apply the chain to `req`. Total: every request gets a [`Substitution`].
166///
167/// See the module documentation for the order and the reasoning behind each step.
168pub fn substitute(req: &Request<'_>) -> Substitution {
169    let at = |face, why| Substitution { face, why };
170    if req.embedded {
171        return at(Face::Embedded, Why::Embedded);
172    }
173    if let Some((f, how)) = Std14::recognise(req.base_font) {
174        return match how {
175            crate::fonts::std14::Match::Alias => at(Face::Standard14(f), Why::NamedStandard14),
176            crate::fonts::std14::Match::FamilyHint => at(Face::Standard14(f), Why::NameHintedStandard14),
177        };
178    }
179    if let Some(f) = crate::fonts::embedding::matching_faces(req.base_font).first() {
180        return at(Face::Craft(f), Why::ExactCraftFamily);
181    }
182    if let Some(script) = req.ordering.and_then(script_for) {
183        // A script we hold a face for; otherwise there is no honest substitute to offer.
184        return match crate::fonts::CRAFT_FONTS.iter().find(|f| f.covers(script)) {
185            Some(f) => at(Face::Craft(f), Why::ScriptFace),
186            None => at(Face::None, Why::NoFaceForScript),
187        };
188    }
189    at(Face::Standard14(style_matched(req.flags)), Why::StyleMatched)
190}
191
192/// The nearest standard-14 face for `/FontDescriptor /Flags` (ISO 32000-2 table 121): fixed pitch
193/// is bit 1, serif bit 2, italic bit 7 and ForceBold bit 19.
194fn style_matched(flags: u32) -> Std14 {
195    let (bold, italic) = (flags & 0x4_0000 != 0, flags & 0x40 != 0);
196    match (flags & 1 != 0, flags & 2 != 0, bold, italic) {
197        (true, _, false, false) => Std14::Courier,
198        (true, _, true, false) => Std14::CourierBold,
199        (true, _, false, true) => Std14::CourierOblique,
200        (true, _, true, true) => Std14::CourierBoldOblique,
201        (false, true, false, false) => Std14::TimesRoman,
202        (false, true, true, false) => Std14::TimesBold,
203        (false, true, false, true) => Std14::TimesItalic,
204        (false, true, true, true) => Std14::TimesBoldItalic,
205        (false, false, false, false) => Std14::Helvetica,
206        (false, false, true, false) => Std14::HelveticaBold,
207        (false, false, false, true) => Std14::HelveticaOblique,
208        (false, false, true, true) => Std14::HelveticaBoldOblique,
209    }
210}

216    fn req<'a>(base: &'a str, embedded: bool, flags: u32, ordering: Option<&'a str>) -> Request<'a> {

220    #[test]
221    fn an_embedded_font_is_never_substituted() {
222        // True even when the name and flags would otherwise pick something: embedding wins first.
223        for (base, ordering) in [("Helvetica", None), ("Whatever", Some("GB1")), ("", None)] {
224            let s = substitute(&req(base, true, 0, ordering));
225            assert_eq!(s, Substitution { face: Face::Embedded, why: Why::Embedded });
226            assert!(s.why.is_faithful());
227            assert_eq!(s.face_name(), None, "nothing stands in, so there is no substitute to name");
228        }
229    }

231    #[test]
232    fn a_standard_14_name_wins_over_the_flags() {
233        // Flags claiming a fixed-pitch serif must not override a name we have exact metrics for.
234        let s = substitute(&req("Times-BoldItalic", false, 0b11, None));
235        assert_eq!(s.face, Face::Standard14(Std14::TimesBoldItalic));
236        assert_eq!(s.why, Why::NamedStandard14);
237        assert_eq!(s.face_name().as_deref(), Some("Times-BoldItalic"));
238        // Aliases and subset tags resolve the same way.
239        for name in ["Arial", "ArialMT", "Arial-Bold", "Symbol", "ZapfDingbats", "CourierNew"] {
240            assert_eq!(substitute(&req(name, false, 0, None)).why, Why::NamedStandard14, "{name}");
241        }
242    }

244    #[test]
245    fn a_name_hint_is_reported_as_a_guess_not_as_a_standard_14_name() {
246        // A house font whose name merely contains "Serif" gets Times, which is a sensible guess --
247        // but calling that "names one of the standard 14" would tell the reader something untrue.
248        let hinted = substitute(&req("SomeHouseSerif", false, 0, None));
249        assert_eq!(hinted.face, Face::Standard14(Std14::TimesRoman));
250        assert_eq!(hinted.why, Why::NameHintedStandard14);
251        assert!(hinted.why.note().contains("hints at one"), "{}", hinted.why.note());
252
253        // Style words still apply to the guess.
254        assert_eq!(substitute(&req("SomeHouseSerif-BoldItalic", false, 0, None)).face, Face::Standard14(Std14::TimesBoldItalic));
255        // "Mono" hints at Courier the same way.
256        assert_eq!(substitute(&req("HouseMono", false, 0, None)).why, Why::NameHintedStandard14);
257
258        // A real alias is still reported as the request it is.
259        for alias in ["Arial", "CourierNew", "Times-Roman", "Symbol"] {
260            assert_eq!(substitute(&req(alias, false, 0, None)).why, Why::NamedStandard14, "{alias}");
261        }
262        // A name with no hint at all falls through to the flags instead.
263        assert_eq!(substitute(&req("HouseGrotesk", false, 0b10, None)).why, Why::StyleMatched);
264    }

266    #[test]
267    fn flags_pick_the_nearest_of_the_fourteen() {
268        // Bit 1 fixed pitch, bit 2 serif, bit 7 italic, bit 19 ForceBold.
269        let cases = [
270            (0, Std14::Helvetica),
271            (0b10, Std14::TimesRoman),
272            (0b1, Std14::Courier),
273            (0x40, Std14::HelveticaOblique),
274            (0x4_0000, Std14::HelveticaBold),
275            (0x4_0040, Std14::HelveticaBoldOblique),
276            (0b10 | 0x40, Std14::TimesItalic),
277            (0b10 | 0x4_0000, Std14::TimesBold),
278            (0b10 | 0x4_0040, Std14::TimesBoldItalic),
279            (0b1 | 0x40, Std14::CourierOblique),
280            (0b1 | 0x4_0000, Std14::CourierBold),
281            (0b1 | 0x4_0040, Std14::CourierBoldOblique),
282            // Fixed pitch and serif together: monospace is the stronger signal for layout.
283            (0b11, Std14::Courier),
284        ];
285        for (flags, want) in cases {
286            let s = substitute(&req("SomethingUnknown", false, flags, None));
287            assert_eq!(s.face, Face::Standard14(want), "flags {flags:#x}");
288            assert_eq!(s.why, Why::StyleMatched);
289        }
290    }

306    #[test]
307    fn identity_ordering_is_not_a_script_and_falls_through_to_the_flags() {
308        // /Ordering (Identity) says the codes are glyph ids. It implies nothing about the face,
309        // so treating it as a script would invent a substitution the dictionary never asked for.
310        let s = substitute(&req("SomeSubsetFont", false, 0b10, Some("Identity")));
311        assert_eq!(s.face, Face::Standard14(Std14::TimesRoman));
312        assert_eq!(s.why, Why::StyleMatched);
313        // An unregistered ordering is treated the same way, not guessed at.
314        assert_eq!(substitute(&req("X", false, 0, Some("MadeUp1"))).why, Why::StyleMatched);
315    }

317    #[test]
318    fn japanese_depends_on_craft_fonts_and_says_which_it_got() {
319        // craft-fonts is an optional build input, so both outcomes are correct -- but they must be
320        // distinguishable, and neither may be a silent Latin stand-in.
321        let s = substitute(&req("SomeJapaneseFont", false, 0, Some("Japan1")));
322        if crate::fonts::CRAFT_FONTS.iter().any(|f| f.covers("Jpan")) {
323            assert_eq!(s.why, Why::ScriptFace);
324            assert!(matches!(s.face, Face::Craft(_)));
325            assert!(s.face_name().is_some_and(|n| !n.is_empty()));
326        } else {
327            assert_eq!(s.why, Why::NoFaceForScript);
328            assert_eq!(s.face, Face::None);
329        }
330        assert!(!s.why.is_faithful(), "a substitution is never faithful");
331    }

333    #[test]
334    fn every_why_has_a_distinct_usable_note() {
335        let all = [
336            Why::Embedded,
337            Why::NamedStandard14,
338            Why::NameHintedStandard14,
339            Why::ExactCraftFamily,
340            Why::ScriptFace,
341            Why::NoFaceForScript,
342            Why::StyleMatched,
343        ];
344        let mut notes: Vec<&str> = all.iter().map(|w| w.note()).collect();
345        notes.sort_unstable();
346        let before = notes.len();
347        notes.dedup();
348        assert_eq!(notes.len(), before, "each reason reads differently");
349        assert!(all.iter().all(|w| !w.note().is_empty() && w.note().len() > 20), "notes are sentences, not labels");
350        assert_eq!(all.iter().filter(|w| w.is_faithful()).count(), 1, "only an embedded program is faithful");
351    }

353    #[test]
354    fn hostile_input_is_total_and_never_panics() {
355        // Every field is attacker-controlled: a name is any bytes the document chose.
356        let long = "a".repeat(10_000);
357        let names = ["", " ", "+", "ABCDEF+", "\u{0}\u{1}", long.as_str(), "../../etc/passwd", "Ĥéļvéťïçà"];
358        for name in names {
359            for flags in [0, u32::MAX, 0x4_0040, 1] {
360                for ordering in [None, Some(""), Some("Japan1"), Some("\u{0}")] {
361                    let s = substitute(&req(name, false, flags, ordering));
362                    // Always a decision, and the reason always reads.
363                    assert!(!s.why.note().is_empty());
364                    assert_eq!(s.face == Face::None, s.why == Why::NoFaceForScript, "only one step declines: {name:?}");
365                }
366            }
367        }
368    }
