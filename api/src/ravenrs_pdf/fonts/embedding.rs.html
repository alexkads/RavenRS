14/// Compare PDF subset/PostScript names with the full face's family name.
15pub fn face_key(name: &str) -> String {
16    name.rsplit('+').next().unwrap_or(name).chars().filter(char::is_ascii_alphanumeric).flat_map(char::to_lowercase).collect()
17}

19/// Full, approved faces of the source family. This never silently selects another family.
20pub fn matching_faces(name: &str) -> Vec<&'static crate::fonts::CraftFont> {
21    let key = face_key(name);
22    crate::fonts::CRAFT_FONTS.iter().filter(|f| face_key(f.family) == key || face_key(&f.name()) == key).collect()
23}

25/// Whether a complete font program can show every non-newline character.
26pub fn covers(bytes: &[u8], text: &str) -> bool {
27    FontRef::new(bytes).is_ok_and(|f| text.chars().all(|c| matches!(c, '\n' | '\r') || f.charmap().map(c).is_some_and(|g| g.to_u32() != 0)))
28}

30/// Names come from the program, never a caller's guessed family.
31pub fn names(bytes: &[u8]) -> Result<(String, String), String> {
32    use skrifa::string::StringId;
33    validate(bytes)?;
34    let f = FontRef::new(bytes).map_err(|e| e.to_string())?;
35    let name = |id| {
36        f.localized_strings(id)
37            .find(|s| s.language() == Some("en"))
38            .or_else(|| f.localized_strings(id).next())
39            .map(|s| s.to_string())
40            .filter(|s| !s.is_empty() && s.len() <= 128)
41    };
42    let family = name(StringId::FAMILY_NAME).ok_or("the font has no family name")?;
43    let postscript = name(StringId::POSTSCRIPT_NAME).unwrap_or_else(|| family.replace(' ', ""));
44    Ok((family, postscript))
45}

47/// A selected external font is document input. Require its own open-licence declaration;
48/// never turn a proprietary system-interface fallback into an exported PDF font.
49pub fn validate_document_input(bytes: &[u8]) -> Result<(), String> {
50    use skrifa::string::StringId;
51    let (family, _) = names(bytes)?;
52    let key = face_key(&family);
53    if ["sourcehan", "sourceserif", "sourcesans", "notosanscjk", "notoserifcjk"].iter().any(|n| key.starts_with(n)) {
54        return Err("this font family is excluded by the project's asset policy".into());
55    }
56    let f = FontRef::new(bytes).map_err(|e| e.to_string())?;
57    let attribution: String =
58        [StringId::COPYRIGHT_NOTICE, StringId::MANUFACTURER].into_iter().flat_map(|id| f.localized_strings(id)).map(|s| s.to_string()).collect();
59    if attribution.to_ascii_lowercase().contains("adobe") {
60        return Err("Adobe-authored font assets are excluded by the project's asset policy".into());
61    }
62    let license: String = f.localized_strings(StringId::LICENSE_DESCRIPTION).map(|s| s.to_string()).collect();
63    if !["SIL OPEN FONT", "SIL Open Font", "Apache License", "Ubuntu Font Licence", "Bitstream Vera", "MIT License"]
64        .iter()
65        .any(|l| license.contains(l))
66    {
67        return Err("select an openly licensed font whose name table declares its licence (OFL, Apache, Ubuntu, Bitstream Vera or MIT)".into());
68    }
69    Ok(())
70}

72/// Validate before allocating any PDF objects, so an invalid font cannot partly mutate a doc.
73pub fn validate(bytes: &[u8]) -> Result<(), String> {
74    if bytes.len() > 16 << 20 {
75        return Err("the selected font exceeds 16 MiB".into());
76    }
77    let font = FontRef::new(bytes).map_err(|_| "select a standalone TrueType/OpenType font (not a font collection)")?;
78    if font.fvar().is_ok() {
79        return Err("select a static instance of this variable font before embedding".into());
80    }
81    if (font.glyf().is_err() && font.cff().is_err()) || font.head().is_err() || font.hhea().is_err() || font.hmtx().is_err() {
82        return Err("the selected font has no usable TrueType/CFF outlines and metrics".into());
83    }
84    if font.os2().is_ok_and(|t| t.fs_type() & (0x0002 | 0x0004 | 0x0200) != 0) {
85        return Err("this font's embedding permissions prohibit editable outline embedding".into());
86    }
87    let m = font.metrics(Size::unscaled(), LocationRef::default());
88    if m.units_per_em == 0 || font.charmap().mappings().next().is_none() {
89        return Err("the selected font has no Unicode mapping or valid units per em".into());
90    }
91    if font.charmap().mappings().take(65_535).count() >= 65_535 {
92        return Err("the selected font has too many Unicode mappings for one PDF font".into());
93    }
94    if font.charmap().mappings().any(|(_, g)| g.to_u32() >= u32::from(m.glyph_count)) {
95        return Err("the selected font maps Unicode to a glyph outside its program".into());
96    }
97    Ok(())
98}

100fn stream(doc: &mut Document, bytes: Vec<u8>) -> Object {
101    Object::Ref(doc.add(Object::Stream(Stream::from_raw(Dict::new(), bytes))))
102}

104/// Embed the complete font once and provide separate CIDs for Unicode aliases. /ToUnicode,
105/// /CIDToGIDMap and widths agree, including supplementary characters, so reopening needs no
106/// installed font. Keeping the full program also lets subsequent edits add other glyphs.
107pub fn embed(doc: &mut Document, fonts: &mut Dict, bytes: &[u8], family: &str) -> Result<String, String> {
108    validate(bytes)?;
109    let font = FontRef::new(bytes).map_err(|e| e.to_string())?;
110    let mappings: Vec<_> = font
111        .charmap()
112        .mappings()
113        .filter(|(_, gid)| gid.to_u32() > 0)
114        .filter_map(|(cp, gid)| char::from_u32(cp).map(|c| (c.to_string(), gid)))
115        .collect();
116    embed_mapping(doc, fonts, bytes, family, &mappings)
117}

119/// Explicit glyph mapping for shaped text. A CID can represent a contextual glyph or a
120/// ligature, while its ToUnicode retains the logical cluster. The full source program stays exact.
121pub fn embed_mapping(
122    doc: &mut Document,
123    fonts: &mut Dict,
124    bytes: &[u8],
125    family: &str,
126    mappings: &[(String, skrifa::GlyphId)],
127) -> Result<String, String> {
128    validate(bytes)?;
129    if mappings.len() > 65_534 || mappings.iter().any(|(s, _)| s.len() > 4096) {
130        return Err("shaped font mapping exceeds its budget".into());
131    }
132    let font = FontRef::new(bytes).map_err(|e| e.to_string())?;
133    let cff = font.glyf().is_err();
134    let loc = LocationRef::default();
135    let m = font.metrics(Size::unscaled(), loc);
136    let scale = 1000.0 / f64::from(m.units_per_em);
137    let advances = font.glyph_metrics(Size::unscaled(), loc);
138    let mut cid_encoding = String::from(
139        "/CIDInit /ProcSet findresource begin\n12 dict begin\nbegincmap\n/CMapName /PCGlyphs def\n/CMapType 1 def\n1 begincodespacerange\n<0000> <FFFF>\nendcodespacerange\n",
140    );
141    let mut cff_widths = std::collections::BTreeMap::new();
142    let mut gids = vec![0, 0];
143    let mut widths = Vec::with_capacity(mappings.len());
144    let mut cmap = String::from(
145        "/CIDInit /ProcSet findresource begin\n12 dict begin\nbegincmap\n/CMapName /PCUnicode def\n/CMapType 2 def\n1 begincodespacerange\n<0000> <FFFF>\nendcodespacerange\n",
146    );
147    for (chunk_index, chunk) in mappings.chunks(100).enumerate() {
148        let _ = writeln!(cmap, "{} beginbfchar", chunk.len());
149        let _ = writeln!(cid_encoding, "{} begincidchar", chunk.len());
150        for (offset, (unicode, gid)) in chunk.iter().enumerate() {
151            let cid = u16::try_from(chunk_index.saturating_mul(100).saturating_add(offset).saturating_add(1)).map_err(|_| "font mapping limit")?;
152            let glyph = u16::try_from(gid.to_u32()).map_err(|_| "font glyph limit")?;
153            if glyph == 0 || u32::from(glyph) >= u32::from(m.glyph_count) {
154                return Err("font contains an unavailable shaped glyph".into());
155            }
156            let width = advances.advance_width(*gid).unwrap_or(0.0) as f64 * scale;
157            if !width.is_finite() || !(0.0..=16_000.0).contains(&width) {
158                return Err("font contains an invalid advance".into());
159            }
160            gids.extend_from_slice(&glyph.to_be_bytes());
161            widths.push(Object::Real(width));
162            cff_widths.insert(glyph, width);
163            let _ = writeln!(cid_encoding, "<{cid:04X}> {glyph}");
164            let hex: String = unicode.encode_utf16().map(|u| format!("{u:04X}")).collect();
165            let _ = writeln!(cmap, "<{cid:04X}> <{hex}>");
166        }
167        cmap.push_str("endbfchar\n");
168        cid_encoding.push_str("endcidchar\n");
169    }
170    cmap.push_str("endcmap\nCMapName currentdict /CMap defineresource pop\nend\nend\n");
171    cid_encoding.push_str("endcmap\nCMapName currentdict /CMap defineresource pop\nend\nend\n");
172    let name = family.chars().filter(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_')).collect::<String>();
173    if name.is_empty() || name.len() > 128 {
174        return Err("the font family name is invalid".into());
175    }
176    let glyph_map = if cff { cid_encoding.as_bytes() } else { gids.as_slice() };
177    let mut programs = ProgramMatch::new(bytes);
178    if let Some(resource) = same_resource(doc, fonts, &name, &mut programs, glyph_map, cmap.as_bytes()) {
179        return Ok(resource);
180    }
181    let mut descriptor = Dict::new();
182    descriptor.set(b"Type".to_vec(), Object::name("FontDescriptor"));
183    descriptor.set(b"FontName".to_vec(), Object::name(&name));
184    descriptor.set(b"FontFamily".to_vec(), Object::String(PdfString::literal(family.as_bytes().to_vec())));
185    descriptor.set(b"Flags".to_vec(), Object::Int(4));
186    descriptor.set(b"ItalicAngle".to_vec(), Object::Real(f64::from(m.italic_angle)));
187    descriptor.set(b"Ascent".to_vec(), Object::Real(f64::from(m.ascent) * scale));
188    descriptor.set(b"Descent".to_vec(), Object::Real(f64::from(m.descent) * scale));
189    descriptor.set(b"CapHeight".to_vec(), Object::Real(f64::from(m.cap_height.unwrap_or(m.ascent)) * scale));
190    descriptor.set(b"StemV".to_vec(), Object::Int(80));
191    let bounds = m.bounds.ok_or("font has no outline bounds")?;
192    descriptor.set(
193        b"FontBBox".to_vec(),
194        Object::Array([bounds.x_min, bounds.y_min, bounds.x_max, bounds.y_max].map(|v| Object::Real(f64::from(v) * scale)).to_vec()),
195    );
196    // Reuse the program another resource of this page already has, else one registered on another
197    // page; source subset programs stay untouched.
198    let existing = existing_program(doc, fonts, &mut programs);
199    let program = existing.unwrap_or_else(|| {
200        let mut d = Dict::new();
201        d.set(b"Length1".to_vec(), Object::Int(bytes.len() as i64));
202        if cff {
203            d.set(b"Subtype".to_vec(), Object::name("OpenType"));
204        }
205        d.set(b"PCFullFont".to_vec(), Object::Bool(true));
206        Object::Ref(doc.add(Object::Stream(Stream::flate(d, bytes))))
207    });
208    descriptor.set(if cff { b"FontFile3".to_vec() } else { b"FontFile2".to_vec() }, program);
209    let desc = Object::Ref(doc.add(Object::Dict(descriptor)));
210    let mut system = Dict::new();
211    system.set(b"Registry".to_vec(), Object::String(PdfString::literal(b"Adobe".to_vec())));
212    system.set(b"Ordering".to_vec(), Object::String(PdfString::literal(b"Identity".to_vec())));
213    system.set(b"Supplement".to_vec(), Object::Int(0));
214    let mut cid = Dict::new();
215    cid.set(b"Type".to_vec(), Object::name("Font"));
216    cid.set(b"Subtype".to_vec(), Object::name(if cff { "CIDFontType0" } else { "CIDFontType2" }));
217    cid.set(b"BaseFont".to_vec(), Object::name(&name));
218    cid.set(b"CIDSystemInfo".to_vec(), Object::Dict(system));
219    cid.set(b"FontDescriptor".to_vec(), desc);
220    if cff {
221        cid.set(
222            b"W".to_vec(),
223            Object::Array(cff_widths.into_iter().flat_map(|(gid, w)| [Object::Int(i64::from(gid)), Object::Array(vec![Object::Real(w)])]).collect()),
224        );
225    } else {
226        cid.set(b"CIDToGIDMap".to_vec(), stream(doc, gids));
227        cid.set(b"W".to_vec(), Object::Array(vec![Object::Int(1), Object::Array(widths)]));
228    }
229    let descendant = Object::Ref(doc.add(Object::Dict(cid)));
230    let mut top = Dict::new();
231    top.set(b"Type".to_vec(), Object::name("Font"));
232    top.set(b"Subtype".to_vec(), Object::name("Type0"));
233    top.set(b"BaseFont".to_vec(), Object::name(&name));
234    top.set(b"Encoding".to_vec(), if cff { stream(doc, cid_encoding.into_bytes()) } else { Object::name("Identity-H") });
235    top.set(b"DescendantFonts".to_vec(), Object::Array(vec![descendant]));
236    top.set(b"ToUnicode".to_vec(), stream(doc, cmap.into_bytes()));
237    let mut index = 0usize;
238    while fonts.contains(format!("PCFull{index}").as_bytes()) {
239        index = index.saturating_add(1);
240    }
241    let resource = format!("PCFull{index}");
242    fonts.set(resource.as_bytes().to_vec(), Object::Ref(doc.add(Object::Dict(top))));
243    Ok(resource)
244}

246/// Whether the stream `object` refers to decodes to `expected`.
247fn stream_is(doc: &Document, object: Option<&Object>, expected: &[u8]) -> bool {
248    object.is_some_and(|o| match doc.resolve(o).as_ref() {
249        Object::Stream(s) => s.decoded_within(expected.len()).is_ok_and(|d| d == expected),
250        _ => false,
251    })
252}

254/// Full programs one embedding decodes at most while looking for its own. A real match passes
255/// the cheap checks and decodes once; many candidates that pass them come from a hostile file.
256const MAX_PROGRAM_DECODES: usize = 16;
257
258/// Recognises the full program written for `bytes`: each candidate stream decodes at most once
259/// and within `bytes.len()`, and at most [`MAX_PROGRAM_DECODES`] decode in all.
260struct ProgramMatch<'a> {
261    bytes: &'a [u8],
262    known: HashMap<ObjRef, bool>,
263    decodes: usize,
264}
265
266impl<'a> ProgramMatch<'a> {
267    fn new(bytes: &'a [u8]) -> Self {
268        Self { bytes, known: HashMap::new(), decodes: 0 }
269    }
270
271    /// Whether `r`, holding `object`, is the program.
272    fn is(&mut self, doc: &Document, r: ObjRef, object: &Object) -> bool {
273        if let Some(&known) = self.known.get(&r) {
274            return known;
275        }
276        let found = match object {
277            Object::Stream(s)
278                if s.dict.get(b"PCFullFont").is_some()
279                    && s.dict.get(b"Length1").and_then(|l| doc.resolve(l).as_int()) == i64::try_from(self.bytes.len()).ok()
280                    && self.decodes < MAX_PROGRAM_DECODES =>
281            {
282                self.decodes = self.decodes.saturating_add(1);
283                s.decoded_within(self.bytes.len()).is_ok_and(|data| data == self.bytes)
284            }
285            _ => false,
286        };
287        self.known.insert(r, found);
288        found
289    }
290
291    /// Whether the resource's program is this one.
292    fn written(&mut self, doc: &Document, w: &Written) -> bool {
293        w.program.as_ref().is_some_and(|r| self.is(doc, r, &doc.get(r)))
294    }
295}

297/// The program for `programs.bytes` that this page's `PCFull` resources, or else the first
298/// objects of the document, already have.
299fn existing_program(doc: &Document, fonts: &Dict, programs: &mut ProgramMatch) -> Option<Object> {
300    written(doc, fonts)
301        .find(|w| programs.written(doc, w))
302        .map(|w| w.program)
303        .or_else(|| doc.scan_objects().take(8192).find_map(|(r, o)| programs.is(doc, r, &o).then_some(Object::Ref(r))))
304}

306/// A `PCFull` resource that [`embed_mapping`] wrote.
307struct Written {
308    resource: String,
309    top: Dict,
310    cid: Dict,
311    /// The reference to its full program.
312    program: Object,
313}
314
315/// The `PCFull` resources in `fonts`, in order.
316fn written<'a>(doc: &'a Document, fonts: &'a Dict) -> impl Iterator<Item = Written> + 'a {
317    fonts.iter().filter(|(key, _)| key.starts_with(b"PCFull")).take(256).filter_map(move |(key, value)| {
318        let top = doc.dict(value)?;
319        let cid = doc.dict(doc.resolve(top.get(b"DescendantFonts")?).as_array()?.first()?)?;
320        let descriptor = doc.dict(cid.get(b"FontDescriptor")?)?;
321        let program = descriptor.get(b"FontFile2").or_else(|| descriptor.get(b"FontFile3")).filter(|p| matches!(p, Object::Ref(_)))?.clone();
322        Some(Written { resource: String::from_utf8_lossy(key).into_owned(), top, cid, program })
323    })
324}

326/// A resource in `fonts` that already is this embedding: the same font name, program, glyph map
327/// and Unicode map. Editing the same text again reuses it instead of adding another copy.
328fn same_resource(doc: &Document, fonts: &Dict, name: &str, programs: &mut ProgramMatch, glyph_map: &[u8], unicode: &[u8]) -> Option<String> {
329    written(doc, fonts)
330        .find(|w| {
331            let glyphs = w.cid.get(b"CIDToGIDMap").or_else(|| w.top.get(b"Encoding"));
332            w.top.name(b"BaseFont") == Some(name.as_bytes())
333                && stream_is(doc, w.top.get(b"ToUnicode"), unicode)
334                && stream_is(doc, glyphs, glyph_map)
335                && programs.written(doc, w)
336        })
337        .map(|w| w.resource)
338}

432            let mut descriptor = Dict::new();
433            descriptor.set(b"FontFile2".to_vec(), Object::Ref(program));
434            let mut cid = Dict::new();
435            cid.set(b"FontDescriptor".to_vec(), Object::Dict(descriptor));
436            let mut top = Dict::new();
437            top.set(b"DescendantFonts".to_vec(), Object::Array(vec![Object::Dict(cid)]));
438            doc.add(Object::Dict(top))
439        };
440        // 200 resources sharing one decoy program decode it once.
441        let mut doc = Document::new_empty();
442        let shared = decoy(&mut doc);
443        let mut fonts = Dict::new();
444        for i in 0..200 {
445            let r = resource(&mut doc, shared);
446            fonts.set(format!("PCFull{i}").into_bytes(), Object::Ref(r));
447        }
448        let mut programs = ProgramMatch::new(bytes);
449        assert_eq!(existing_program(&doc, &fonts, &mut programs), None);
450        assert_eq!(programs.decodes, 1);
451        // 100 distinct decoys decode a bounded number of times.
452        let mut doc = Document::new_empty();
453        let mut fonts = Dict::new();
454        for i in 0..100 {
455            let program = decoy(&mut doc);
456            let r = resource(&mut doc, program);
457            fonts.set(format!("PCFull{i}").into_bytes(), Object::Ref(r));
458        }
459        let mut programs = ProgramMatch::new(bytes);
460        assert_eq!(existing_program(&doc, &fonts, &mut programs), None);
461        assert_eq!(programs.decodes, MAX_PROGRAM_DECODES);
462    }

465    fn the_full_program_is_written_compressed_and_another_page_reuses_it() {
466        let bytes = include_bytes!("../../assets/assets/fonts/Inter-Regular.ttf");
467        let mut doc = Document::new_empty();
468        let mut first = Dict::new();
469        embed(&mut doc, &mut first, bytes, "Inter").unwrap();
470        let program = || {
471            doc.scan_objects()
472                .find_map(|(_, o)| match o.as_ref() {
473                    Object::Stream(s) if s.dict.get(b"PCFullFont").is_some() => Some(s.clone()),
474                    _ => None,
475                })
476                .unwrap()
477        };
478        let stream = program();

480        assert!(stream.raw.len() < bytes.len(), "{} of {} bytes", stream.raw.len(), bytes.len());

482        // Another page's resources don't have it, so it is found among the document's objects.
483        let objects = doc.scan_objects().count();
484        let mut second = Dict::new();
485        embed(&mut doc, &mut second, bytes, "Inter").unwrap();
486        let programs = doc.scan_objects().filter(|(_, o)| matches!(o.as_ref(), Object::Stream(s) if s.dict.get(b"PCFullFont").is_some())).count();
487        assert_eq!(programs, 1, "one program for both pages");
488        assert!(doc.scan_objects().count() > objects, "the second page gets its own resource");
