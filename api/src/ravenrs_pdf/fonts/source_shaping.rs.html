19/// Complex scripts/marks require shaping instead of independently encoding characters.
20pub fn required(text: &str) -> bool {
21    text.chars()
22        .any(|c| matches!(u32::from(c), 0x0300..=0x036f | 0x0590..=0x0fff | 0x1780..=0x17ff | 0x200c..=0x200f | 0x202a..=0x202e | 0x2066..=0x2069))
23}

25/// Whether a Thai source program retains its mark positioning.
26/// Character coverage alone does not show that stacked vowels and tones can be laid out.
27/// OpenType Thai requires `mark`/`mkmk` (GPOS); `ccmp` substitution (GSUB) is optional.
28pub fn has_thai_layout(bytes: &[u8]) -> bool {
29    FontRef::new(bytes).is_ok_and(|f| positions_thai_marks(&f))
30}
31
32/// Whether the GPOS features the shaper uses for Thai include `mark` or `mkmk`. The selection is
33/// HarfRust's, through the same read-fonts calls: the `thai` script, else `DFLT`, `dflt` or
34/// `latn` (`ScriptList::select`); then, given no language, a language system tagged `dflt`,
35/// else the script's default one. Marks under a script or language it doesn't select don't
36/// count, nor does a GPOS with only kerning.
37fn positions_thai_marks(face: &FontRef) -> bool {
38    use skrifa::Tag;
39    let Ok(gpos) = face.gpos() else { return false };
40    let (Ok(scripts), Ok(features)) = (gpos.script_list(), gpos.feature_list()) else { return false };
41    let records = features.feature_records();
42    let marks = |index: u16| records.get(usize::from(index)).is_some_and(|r| r.feature_tag() == *b"mark" || r.feature_tag() == *b"mkmk");
43    let Some(selected) = scripts.select(&[Tag::new(b"thai")]) else { return false };
44    let Ok(script) = scripts.get(selected.index) else { return false };
45    let language = match script.lang_sys_index_for_tag(Tag::new(b"dflt")) {
46        Some(index) => script.lang_sys(index).ok().map(|l| l.element),
47        None => script.default_lang_sys().transpose().ok().flatten(),
48    };
49    let Some(language) = language else { return false };
50    language.feature_indices().iter().any(|i| marks(i.get()))
51        || (language.required_feature_index() != 0xFFFF && marks(language.required_feature_index()))
52}

54pub fn shape(bytes: &[u8], text: &str) -> Result<ShapedRun, String> {
55    shape_inner(bytes, text, false)
56}

58/// Only used after the PDF adapter has verified legacy Thai PUA glyph mappings.
59pub(crate) fn shape_legacy_thai(bytes: &[u8], text: &str) -> Result<ShapedRun, String> {
60    shape_inner(bytes, text, true)

63fn shape_inner(bytes: &[u8], text: &str, legacy_thai: bool) -> Result<ShapedRun, String> {
64    if text.len() > 262_144 || text.contains(['\n', '\r']) {
65        return Err("shape one bounded text row at a time".into());
66    }
67    if text.is_empty() {
68        return Ok(ShapedRun { glyphs: Vec::new(), advance: 0.0 });
69    }
70    let face = FontRef::new(bytes).map_err(|e| e.to_string())?;
71    // Only a vowel or tone mark set on its base (or Sara Am, whose Nikhahit is) needs positioning:
72    // digits and bare letters shape the same without it.
73    if !legacy_thai && text.chars().any(|c| matches!(u32::from(c), 0x0e31 | 0x0e33..=0x0e3a | 0x0e47..=0x0e4e)) && !positions_thai_marks(&face) {
74        return Err("the source subset has no Thai mark positioning (GPOS mark/mkmk); select its full font".into());
75    }
76    if text.chars().any(|c| matches!(u32::from(c), 0x0600..=0x06ff | 0x0750..=0x077f | 0x08a0..=0x0dff | 0x0e80..=0x0fff | 0x1780..=0x17ff))
77        && face.gsub().is_err()
78    {
79        return Err("the source subset has no contextual substitution tables; select its full font".into());
80    }
81    let data = ShaperData::new(&face);
82    let shaper = data.shaper(&face).build();
83    let em = f64::from(shaper.units_per_em().max(1));
84    let base = if unicode_bidi::get_base_direction(text) == unicode_bidi::Direction::Rtl { Level::rtl() } else { Level::ltr() };
85    let bidi = ParagraphBidiInfo::new(text, Some(base));
86    let (levels, runs) = bidi.visual_runs(0..text.len());
87    let mut glyphs = Vec::new();
88    let mut x = 0.0;
89    for run in runs {
90        let Some(input) = text.get(run.clone()) else { continue };
91        let rtl = levels.get(run.start).is_some_and(Level::is_rtl);
92        let mut buffer = UnicodeBuffer::new();
93        buffer.push_str(input);
94        buffer.set_cluster_level(BufferClusterLevel::MonotoneGraphemes);
95        buffer.set_direction(if rtl { Direction::RightToLeft } else { Direction::LeftToRight });
96        buffer.guess_segment_properties();
97        let output = shaper.shape(buffer, ShapeOptions::new());
98        let mut starts: Vec<_> = output.glyph_infos().iter().map(|g| g.cluster as usize).collect();
99        starts.sort_unstable();
100        starts.dedup();
101        let mut seen = std::collections::HashSet::new();
102        for (g, p) in output.glyph_infos().iter().zip(output.glyph_positions()) {
103            if g.glyph_id == 0 {
104                let missing = input.get(g.cluster as usize..).and_then(|s| s.chars().next()).unwrap_or('\u{fffd}');
105                return Err(format!(
106                    "the full font lacks a required contextual glyph for {missing} (U+{:04X}); choose the complete source face",
107                    u32::from(missing)
108                ));
109            }
110            let start = g.cluster as usize;
111            let end = starts.get(starts.partition_point(|i| *i <= start)).copied().unwrap_or(input.len());
112            let logical = if seen.insert(start) { input.get(start..end).unwrap_or_default().to_string() } else { String::new() };
113            glyphs.push(ShapedGlyph {
114                id: GlyphId::new(g.glyph_id),
115                position: [x + f64::from(p.x_offset) / em, f64::from(p.y_offset) / em],
116                text: logical,
117            });
118            x += f64::from(p.x_advance) / em;
119        }
120    }
121    if !x.is_finite() || glyphs.len() > 65_534 {
122        return Err("shaped row exceeds its geometry budget".into());
123    }
124    Ok(ShapedRun { glyphs, advance: x })
125}

131    /// Where the GPOS table starts in an sfnt.
132    fn gpos(bytes: &[u8]) -> usize {
133        let count = u16::from_be_bytes([bytes[4], bytes[5]]) as usize;
134        let record = (0..count).map(|i| 12 + i * 16).find(|&r| &bytes[r..r + 4] == b"GPOS").unwrap();
135        u32::from_be_bytes(bytes[record + 8..record + 12].try_into().unwrap()) as usize
136    }
137
138    fn u16_at(bytes: &[u8], at: usize) -> usize {
139        u16::from_be_bytes([bytes[at], bytes[at + 1]]) as usize
140    }
141
142    /// Where the GPOS ScriptList starts.
143    fn script_list(bytes: &[u8]) -> usize {
144        gpos(bytes) + u16_at(bytes, gpos(bytes) + 4)
145    }
146
147    /// Where script `tag`'s table starts.
148    fn script(bytes: &[u8], tag: &[u8; 4]) -> usize {
149        let list = script_list(bytes);
150        let record = (0..u16_at(bytes, list)).map(|i| list + 2 + i * 6).find(|&r| &bytes[r..r + 4] == tag).unwrap();
151        list + u16_at(bytes, record + 4)
152    }
153
154    /// Retag script `from` as `to`, keeping the records sorted by tag as the shaper's binary
155    /// search needs.
156    fn retag_script(bytes: &mut [u8], from: &[u8; 4], to: &[u8; 4]) {
157        let list = script_list(bytes);
158        let count = u16_at(bytes, list);
159        let mut records: Vec<[u8; 6]> = (0..count).map(|i| bytes[list + 2 + i * 6..list + 8 + i * 6].try_into().unwrap()).collect();
160        records.iter_mut().find(|r| &r[..4] == from).unwrap()[..4].copy_from_slice(to);
161        records.sort_by(|a, b| a[..4].cmp(&b[..4]));
162        for (i, record) in records.iter().enumerate() {
163            bytes[list + 2 + i * 6..list + 8 + i * 6].copy_from_slice(record);
164        }
165    }

167    /// A shaper uses a font's `thai` script when it has one: marks the font registers only for
168    /// Latin don't position Thai.
169    #[test]
170    fn thai_layout_counts_only_the_script_the_shaper_selects() {
171        let mut bytes = crate::fonts::script::INTER_REGULAR.to_vec();
172        // Inter's DFLT becomes a `thai` script whose default language system has no features;
173        // its latn script keeps `mark` and `mkmk`.
174        retag_script(&mut bytes, b"DFLT", b"thai");
175        let thai = script(&bytes, b"thai");
176        let language = thai + u16_at(&bytes, thai);
177        assert_ne!(language, thai, "a default language system");
178        bytes[language + 2..language + 6].copy_from_slice(&[0xFF, 0xFF, 0, 0]);
179        assert!(!has_thai_layout(&bytes), "the latn script's marks are not the thai script's");
180    }

182    /// Given no language, the shaper takes a language system tagged `dflt` before the script's
183    /// default one, in both directions.
184    #[test]
185    fn thai_layout_follows_a_language_system_tagged_dflt() {
186        // Without DFLT the shaper falls back to latn, whose default language system has marks and
187        // is shared by its CAT, MOL and ROM records. Cyrillic's default one has only `kern`.
188        let mut base = crate::fonts::script::INTER_REGULAR.to_vec();
189        retag_script(&mut base, b"DFLT", b"zzzz");
190        assert!(has_thai_layout(&base));
191        let latn = script(&base, b"latn");
192        let kern_only = script(&base, b"cyrl") + u16_at(&base, script(&base, b"cyrl"));
193        let kern_only = u16::try_from(kern_only - latn).unwrap().to_be_bytes();
194        let rom = (0..u16_at(&base, latn + 2)).map(|j| latn + 4 + j * 6).find(|&r| &base[r..r + 4] == b"ROM ").unwrap();
195        // ROM becomes `dflt` (still sorted last) and has only kerning: it wins over the default.
196        let mut named_without = base.clone();
197        named_without[rom..rom + 4].copy_from_slice(b"dflt");
198        named_without[rom + 4..rom + 6].copy_from_slice(&kern_only);
199        assert!(!has_thai_layout(&named_without), "the dflt language system has no marks");
200        // The default has only kerning and the `dflt` one keeps the marks: it wins again.
201        let mut default_without = base.clone();
202        default_without[rom..rom + 4].copy_from_slice(b"dflt");
203        default_without[latn..latn + 2].copy_from_slice(&kern_only);
204        assert!(has_thai_layout(&default_without), "the dflt language system has the marks");
205    }

207    /// A GPOS table alone is not mark positioning: with its `mark` and `mkmk` features renamed,
208    /// as in a kerning-only subset, a face can't stack Thai vowels and tones.
209    #[test]
210    fn thai_layout_needs_mark_positioning_features() {
211        let inter = crate::fonts::script::INTER_REGULAR;
212        assert!(has_thai_layout(inter), "Inter positions marks under its default script");
213        let mut bytes = inter.to_vec();
214        let table = gpos(&bytes);
215        let features = table + u16_at(&bytes, table + 6);
216        let mut renamed = 0;
217        for i in 0..u16_at(&bytes, features) {
218            let tag = features + 2 + i * 6;
219            if matches!(&bytes[tag..tag + 4], b"mark" | b"mkmk") {
220                bytes[tag..tag + 4].copy_from_slice(b"kern");
221                renamed += 1;
222            }
223        }
224        assert!(renamed > 0);
225        assert!(FontRef::new(&bytes).unwrap().gpos().is_ok(), "the GPOS table still parses");
226        assert!(!has_thai_layout(&bytes));
227        let error = shape(&bytes, "กิ่").unwrap_err();
228        assert!(error.contains("mark positioning"), "{error}");
229    }

231    #[test]
232    fn thai_positioning_does_not_require_optional_gsub() {
233        let Some(face) = crate::fonts::document_thai_font() else {
234            eprintln!("skipping thai_positioning_does_not_require_optional_gsub: built without craft-fonts (set CRAFT_FONTS_DIR to run it)");
235            return;
236        };
237        let mut bytes = face.bytes.to_vec();
238        // Keep the actual GPOS anchors and outlines while removing optional substitutions.
239        // Table payloads stay at their original offsets; no font fixture is committed.
240        let count = u16::from_be_bytes([bytes[4], bytes[5]]) as usize;
241        let tables: Vec<_> = bytes[12..12 + count * 16].as_chunks::<16>().0.iter().filter(|t| &t[..4] != b"GSUB").map(|t| t.to_vec()).collect();
242        bytes[4..6].copy_from_slice(&(tables.len() as u16).to_be_bytes());
243        for (i, table) in tables.iter().enumerate() {
244            bytes[12 + i * 16..28 + i * 16].copy_from_slice(table);
245        }
246        assert!(FontRef::new(&bytes).unwrap().gsub().is_err());
247        assert!(has_thai_layout(&bytes));
248        let shaped = shape(&bytes, "กิ่ น้ำ").unwrap();
249        assert!(shaped.advance > 0.0);
250        assert!(shaped.glyphs.iter().all(|g| g.position.iter().all(|p| p.is_finite())));
251    }
