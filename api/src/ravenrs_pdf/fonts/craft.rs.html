1//! Fonts from the optional craft-fonts build input (<https://github.com/storytold/craft-fonts>).
2//!
3//! `build.rs` embeds every font in craft-fonts' manifest when PdfCraft is built with
4//! `CRAFT_FONTS_DIR=<checkout>`; otherwise [`CRAFT_FONTS`] is empty and everything here returns
5//! nothing. Callers must work either way.
6
7/// A font from the optional craft-fonts build input (empty unless built with `CRAFT_FONTS_DIR`).
8pub struct CraftFont {
9    pub family: &'static str,
10    pub style: &'static str,
11    /// ISO 15924 scripts the font is for, e.g. `"Jpan"`.
12    pub scripts: &'static [&'static str],
13    pub bytes: &'static [u8],
14}
15
16include!(concat!(env!("OUT_DIR"), "/craft_fonts.rs"));

18impl CraftFont {
19    /// Whether the font is meant for `script` (ISO 15924, e.g. `"Jpan"`).
20    pub fn covers(&self, script: &str) -> bool {
21        self.scripts.contains(&script)
22    }
23
24    /// A unique name for registering the face with a font system ("BIZ UDPGothic Bold").
25    pub fn name(&self) -> String {
26        format!("{} {}", self.family, self.style)
27    }
28}

30/// The `Jpan` craft-fonts faces in the order the interface prefers them: BIZ UDPGothic (a UI
31/// face) first, then the rest in manifest order. Empty without craft-fonts.
32pub fn ui_japanese_fonts() -> Vec<&'static CraftFont> {
33    let mut fonts: Vec<&CraftFont> = CRAFT_FONTS.iter().filter(|f| f.covers("Jpan")).collect();
34    // Stable: manifest order within each group.
35    fonts.sort_by_key(|f| f.family != "BIZ UDPGothic");
36    fonts
37}

105fn hebrew_face(faces: &[CraftFont], bold: bool) -> Option<&CraftFont> {
106    let hebr = || faces.iter().filter(|f| f.covers("Hebr"));
107    let style = if bold { "Bold" } else { "Regular" };
108    hebr().find(|f| f.style == style).or_else(|| hebr().find(|f| f.style == "Regular")).or_else(|| hebr().next())
109}

111/// Interface CJK faces in fallback order for the UI language: Simplified Chinese first when
112/// `prefer_hans`, otherwise Japanese first (the historical default).
113///
114/// The order matters beyond glyph shapes. egui renders each character with the FIRST face that
115/// has its glyph, so with Japanese first a Simplified-only character (e.g. U+6B22 欢, absent
116/// from Japanese faces) lands in a different face than its neighbours; the mixed vertical
117/// metrics then sink it below the line. Preferring the UI language's face keeps one line in
118/// one face with one baseline.
119pub fn ui_cjk_fonts(prefer_hans: bool) -> Vec<&'static CraftFont> {
120    order_cjk(CRAFT_FONTS.iter(), prefer_hans)
121}
122
123fn order_cjk<'a>(faces: impl IntoIterator<Item = &'a CraftFont>, prefer_hans: bool) -> Vec<&'a CraftFont> {
124    let mut out: Vec<&'a CraftFont> = faces.into_iter().filter(|f| f.covers("Hans") || f.covers("Jpan")).collect();
125    let first = if prefer_hans { "Hans" } else { "Jpan" };
126    // The preferred script's faces first (a face covering both counts as preferred); within each
127    // group the Japanese UI face (BIZ UDPGothic, not a serif) leads, otherwise manifest order
128    // (the sort is stable).
129    out.sort_by_key(|f| (!f.covers(first), f.family != "BIZ UDPGothic"));
130    out
131}

173fn document_face(faces: &[CraftFont], serif: bool, bold: bool) -> Option<&CraftFont> {
174    document_faces(faces, serif, bold).into_iter().next()
175}
176
177fn document_faces(faces: &[CraftFont], serif: bool, bold: bool) -> Vec<&CraftFont> {
178    let jpan = || faces.iter().filter(|f| f.covers("Jpan"));
179    let mut preferred = Vec::new();
180    if !serif {
181        let style = if bold { "Bold" } else { "Regular" };
182        preferred.extend(jpan().find(|f| f.family == "BIZ UDPGothic" && f.style == style));
183        preferred.extend(jpan().find(|f| f.family == "BIZ UDPGothic" && f.style == "Regular"));
184    }
185    for family in ["Shippori Mincho", "BIZ UDMincho"] {
186        preferred.extend(jpan().find(|f| f.family == family && f.style == "Regular"));
187    }
188    let mut out: Vec<&CraftFont> = Vec::new();
189    for face in preferred.into_iter().chain(jpan().filter(|f| f.style == "Regular")).chain(jpan()) {
190        if !out.iter().any(|f| std::ptr::eq(*f, face)) {
191            out.push(face);
192        }
193    }
194    out
195}

197const fn str_eq(a: &str, b: &str) -> bool {
198    let (a, b) = (a.as_bytes(), b.as_bytes());
199    if a.len() != b.len() {
200        return false;
201    }
202    let mut i = 0;
203    while i < a.len() {
204        if a[i] != b[i] {
205            return false;
206        }
207        i += 1;
208    }
209    true
210}
211
212const fn find(family: &str, style: &str) -> Option<&'static [u8]> {
213    let mut i = 0;
214    while i < CRAFT_FONTS.len() {
215        let f = &CRAFT_FONTS[i];
216        if str_eq(f.family, family) && str_eq(f.style, style) {
217            return Some(f.bytes);
218        }
219        i += 1;
220    }
221    None
222}

232    #[test]
233    fn cjk_fallback_order_follows_the_ui_language() {
234        // Synthetic faces: order_cjk must not depend on the real build input.
235        static BYTES: &[u8] = b"fake";
236        static LATN: &[&str] = &["Latn"];
237        static JPAN_LATN: &[&str] = &["Jpan", "Latn"];
238        static HANS_LATN: &[&str] = &["Hans", "Latn"];
239        let biz_bold = CraftFont { family: "BIZ UDPGothic", style: "Bold", scripts: JPAN_LATN, bytes: BYTES };
240        let biz = CraftFont { family: "BIZ UDPGothic", style: "Regular", scripts: JPAN_LATN, bytes: BYTES };
241        let hans = CraftFont { family: "FakeHans", style: "Regular", scripts: HANS_LATN, bytes: BYTES };
242        let inter = CraftFont { family: "Inter", style: "Regular", scripts: LATN, bytes: BYTES };
243        let faces = [biz_bold, biz, hans, inter];
244        // Chinese mode: the Hans group first (manifest order), then Japanese.
245        let zh: Vec<&str> = order_cjk([&faces[0], &faces[2], &faces[1]], true).iter().map(|f| f.family).collect();
246        assert_eq!(zh, ["FakeHans", "BIZ UDPGothic", "BIZ UDPGothic"]);
247        // Japanese mode keeps the historical default: the Japanese UI face first.
248        let ja: Vec<&str> = order_cjk([&faces[0], &faces[2], &faces[1]], false).iter().map(|f| f.family).collect();
249        assert_eq!(ja, ["BIZ UDPGothic", "BIZ UDPGothic", "FakeHans"]);
250        // Latin-only faces are never interface CJK fallbacks.
251        assert!(order_cjk(&faces, true).iter().all(|f| f.covers("Hans") || f.covers("Jpan")));
252        // A serif listed first in the manifest never leads the Japanese group, in either mode.
253        let mincho = CraftFont { family: "Shippori Mincho", style: "Regular", scripts: JPAN_LATN, bytes: BYTES };
254        let zh: Vec<&str> = order_cjk([&mincho, &faces[2], &faces[1]], true).iter().map(|f| f.family).collect();
255        assert_eq!(zh, ["FakeHans", "BIZ UDPGothic", "Shippori Mincho"]);
256        let ja: Vec<&str> = order_cjk([&mincho, &faces[2], &faces[1]], false).iter().map(|f| f.family).collect();
257        assert_eq!(ja, ["BIZ UDPGothic", "Shippori Mincho", "FakeHans"]);
258    }

317    /// `build.rs` embeds the Japanese CJK faces whole and any other CJK face only up to
318    /// `MAX_OTHER_CJK_BYTES` (the explicitly pinned native-only LXGW UI fallback has a separate
319    /// 16 MiB cap), so a checkout with a large Chinese or Korean face (craft-fonts has the
320    /// 16.4 MB Noto Sans CJK SC) leaves it out. The four Japanese faces of the pinned craft-fonts
321    /// are 24.1 MB.
322    #[test]
323    fn cjk_faces_follow_the_size_rule() {
324        use crate::fonts::cjk_scripts::{MAX_NATIVE_UI_CJK_BYTES, MAX_OTHER_CJK_BYTES, NATIVE_UI_CJK_FAMILIES, WHOLE_CJK_FAMILIES, is_cjk_script};
325        // A tag can carry a region after the code: craft-fonts tags Noto Sans HK UI `Hant-HK`.
326        assert!(["Hant-HK", "Hans", "Jpan", "Hang"].iter().all(|s| is_cjk_script(s)));
327        assert!(!["Latn", "Arab", "Hebr"].iter().any(|s| is_cjk_script(s)));
328        let cjk = CRAFT_FONTS.iter().filter(|f| f.scripts.iter().any(|s| is_cjk_script(s)));
329        let (whole, other): (Vec<_>, Vec<_>) = cjk.partition(|f| WHOLE_CJK_FAMILIES.contains(&f.family));
330        for face in &other {
331            let len = face.bytes.len();
332            let native_ui = NATIVE_UI_CJK_FAMILIES.contains(&face.family);
333            assert!(!cfg!(target_arch = "wasm32") || !native_ui, "native UI face in web payload");
334            let cap = if native_ui { MAX_NATIVE_UI_CJK_BYTES } else { MAX_OTHER_CJK_BYTES };
335            assert!(u64::try_from(len).is_ok_and(|n| n <= cap), "{} is embedded at {len} bytes", face.name());
336        }
337        let bytes: usize = whole.iter().map(|f| f.bytes.len()).sum();
338        assert!(bytes < 25_000_000, "{bytes} bytes of Japanese faces");
339    }
