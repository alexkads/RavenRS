1//! Recover a narrowly identified legacy Thai exporter map, not arbitrary missing Unicode.
2//!
3//! Some THSarabunPSK Identity-H subsets retain original glyph IDs but write Latin/control
4//! destinations for Thai in ToUnicode (including an accidental 00ff..01ff range). The intact
5//! Thai anchors identify the original nominal Thai order. Only that complete fingerprint is
6//! accepted, and then only when every glyph the subset kept advances as its assumed character
7//! does (a mark takes none); correct maps and unrelated fonts are never changed. No font
8//! artwork is bundled.
9
10use super::{Dict, Document, Metrics, Reads, dict, identity_h, only_char, unicode_map};
11use crate::cos::{DecodeBudget, Object};
12use std::collections::BTreeMap;

14/// Recover the nominal Thai codes and the presentation marks of this damaged legacy export.
15/// The caller must record the repair, and attach the map to a copy of the page's font.
16pub fn recovered_thai_map(doc: &Document, font: &Dict) -> Option<Vec<(u16, char)>> {
17    use skrifa::{MetadataProvider as _, raw::TableProvider as _};
18    let base = String::from_utf8_lossy(font.name(b"BaseFont")?);
19    let family = base.rsplit('+').next()?;
20    if !matches!(family, "THSarabunPSK" | "THSarabunPSK-Bold" | "THSarabunPSK-Italic" | "THSarabunPSK-BoldItalic")
21        || font.name(b"Subtype") != Some(b"Type0")
22        || !identity_h(doc, font)
23    {
24        return None;
25    }
26    let desc = dict(doc, font.get(b"DescendantFonts").and_then(|a| doc.resolve(a).as_array().and_then(|a| a.first()).cloned()).as_ref())?;
27    // A map given by reference is the name it refers to. A missing one is taken as Identity, the
28    // default of PDF 1.x (ISO 32000-1), as `composite_glyph_maps` and the renderer take it; PDF 2.0
29    // gives it no default and requires it with a `/FontFile2`.
30    if desc.get(b"CIDToGIDMap").is_some_and(|m| doc.resolve(m).as_name() != Some(b"Identity".as_slice())) {
31        return None;
32    }
33    let map = unicode_map(doc, font, true, &mut Reads::new(doc, &mut DecodeBudget::new(usize::MAX, 0)));
34    // Five separated consonants and three contextual marks must agree, together with the
35    // corrupt destinations and corrupt range. This is deliberately stricter than font-name
36    // detection: a renamed/subsetted font with different glyph order must not be guessed.
37    let fingerprint = [
38        (217, '\u{0e04}'),
39        (226, '\u{0e0d}'),
40        (250, '\u{0e25}'),
41        (256, '\u{0e2b}'),
42        (258, '\u{0e2d}'),
43        (356, '\u{0e4c}'),
44        (358, '\u{0e31}'),
45        (360, '\u{0e47}'),
46        (214, '\u{000f}'),
47        (240, ')'),
48        (248, '#'),
49        (261, '>'),
50        (263, '@'),
51        (265, '>'),
52        (273, 'N'),
53        (352, 'N'),
54        (353, 'O'),
55    ];
56    let matches = |pairs: &[(u32, char)], last: u16| {
57        pairs.iter().all(|&(code, ch)| map.text(code).is_some_and(|t| t.chars().eq([ch])))
58            && map.ranges.iter().any(|r| r.lo == 255 && r.hi == 511 && r.units.as_ref() == [last])
59    };
60    // The same exporter emitted three distinct subset maps. Each must match its entire
61    // intact-anchor and corrupt-destination fingerprint, including the oversized range.
62    let body = matches(&fingerprint, 0x32);
63    let attachment = matches(
64        &[
65            (217, '\u{0e04}'),
66            (256, '\u{0e2b}'),
67            (258, '\u{0e2d}'),
68            (297, '\u{0e58}'),
69            (347, '\u{0e48}'),
70            (356, '\u{0e4c}'),
71            (358, '\u{0e31}'),
72            (360, '\u{0e47}'),
73            (214, '\u{000f}'),
74            (226, '\u{000f}'),
75            (240, '\u{0027}'),
76            (248, '/'),
77            (250, '1'),
78            (265, 'B'),
79            (273, 'N'),
80            (352, 'N'),
81            (353, 'O'),
82        ],
83        0x32,
84    );
85    let bold = matches(
86        &[
87            (217, '\u{0e04}'),
88            (226, '\u{0e0d}'),
89            (232, '\u{0e13}'),
90            (234, '\u{0e15}'),
91            (236, '\u{0e17}'),
92            (243, '\u{0e1e}'),
93            (250, '\u{0e25}'),
94            (256, '\u{0e2b}'),
95            (258, '\u{0e2d}'),
96            (356, '\u{0e4c}'),
97            (214, '\u{000f}'),
98            (240, '!'),
99            (248, '1'),
100            (263, '@'),
101            (273, 'N'),
102            (352, 'N'),
103            (353, 'O'),
104        ],
105        0x30,
106    );
107    if !(body || attachment || bold) {
108        return None;
109    }
110    let nominal = (214u16..=298).filter_map(|gid| {
111        let unicode = u32::from(gid) + if gid <= 271 { 0xd2b } else { 0xd2f };
112        char::from_u32(unicode).map(|ch| (gid, ch))
113    });
114    // Contextual marks have the same character as the nominal mark, with a moved outline.
115    let contextual = [
116        (343, '\u{0e34}'),
117        (344, '\u{0e35}'),
118        (347, '\u{0e48}'),
119        (351, '\u{0e4c}'),
120        (352, '\u{0e48}'),
121        (353, '\u{0e49}'),
122        (356, '\u{0e4c}'),
123        (358, '\u{0e31}'),
124        (360, '\u{0e47}'),
125    ];
126    // The anchors identify the exporter; the font program confirms each glyph's advance class
127    // (not its identity, which nothing in such a subset records). Every glyph the
128    // subset kept must advance as the character it is given (a vowel or tone mark set on its
129    // base takes no advance, every other character does), else the order differs somewhere and
130    // nothing is mapped. A glyph the subset dropped is left unmapped: no code can draw it, and
131    // its advance is the subsetter's filler, not the glyph's.
132    let program = embedded_truetype(doc, &desc)?;
133    let face = skrifa::FontRef::new(&program).ok()?;
134    let loca = face.loca(None).ok()?;
135    let advances = face.glyph_metrics(skrifa::instance::Size::unscaled(), skrifa::instance::LocationRef::default());
136    let mut out = Vec::new();
137    for (gid, ch) in nominal.chain(contextual) {
138        let at = usize::from(gid);
139        if loca.get_raw(at).zip(loca.get_raw(at + 1)).is_none_or(|(start, end)| end <= start) {
140            continue;
141        }
142        if advances.advance_width(skrifa::GlyphId::new(u32::from(gid))).is_none_or(|w| (w == 0.0) != is_thai_mark(ch)) {
143            return None;
144        }
145        out.push((gid, ch));
146    }
147    Some(out)
148}

150/// The character each legacy Windows Thai presentation form U+F700 + index stands for: ฐ and ญ
151/// without descenders, and the vowels and tone marks moved left, lower or both (the PUA
152/// assignments of <https://linux.thai.net/~thep/th-otf/shaping.html>).
153const PUA_BASES: [char; 27] = ['ฐ', 'ิ', 'ี', 'ึ', 'ื', '่', '้', '๊', '๋', '์', '่', '้', '๊', '๋', '์', 'ญ', 'ั', 'ํ', '็', '่', '้', '๊', '๋', '์', 'ุ', 'ู', 'ฺ'];
154
155/// Whether a Thai character is a vowel or tone mark set above or below its base, with no advance
156/// of its own (U+0E31, U+0E34–U+0E3A, U+0E47–U+0E4E). Sara Am and the other vowels advance.
157fn is_thai_mark(ch: char) -> bool {
158    matches!(u32::from(ch), 0x0e31 | 0x0e34..=0x0e3a | 0x0e47..=0x0e4e)
159}

161/// The glyph a legacy THSarabunPSK program draws Thai character `ch` with: U+0E01–U+0E3A are
162/// glyphs 214–271 and U+0E3F–U+0E5B glyphs 272–298 (the order `recovered_thai_map` assumes).
163fn legacy_glyph(ch: u32) -> Option<u32> {
164    match ch {
165        0x0e01..=0x0e3a => ch.checked_sub(0xd2b),
166        0x0e3f..=0x0e5b => ch.checked_sub(0xd2f),
167        _ => None,
168    }
169}

171/// The TrueType program (`/FontFile2`) of a CIDFont, decoded within the editor's size cap.
172fn embedded_truetype(doc: &Document, descendant: &Dict) -> Option<Vec<u8>> {
173    let descriptor = dict(doc, descendant.get(b"FontDescriptor"))?;
174    let program = doc.resolve(descriptor.get(b"FontFile2")?);
175    let Object::Stream(program) = program.as_ref() else { return None };
176    DecodeBudget::new(super::MAX_SFNT, 0).decode_with_damage(doc, program, super::MAX_SFNT).ok().map(|(data, _)| data)
177}

186/// Unicode-to-glyph and glyph-to-PDF-code maps for a two-byte Identity-H TrueType font.
187/// Used by the editor-only cmap adapter and by direct source-font shaping.
188pub(super) fn composite_glyph_maps(
189    doc: &Document,
190    font: &Dict,
191    bytes: &[u8],
192    metrics: &Metrics,
193    budget: &mut DecodeBudget,
194) -> Option<(BTreeMap<u32, u32>, BTreeMap<u32, u16>)> {
195    use skrifa::{MetadataProvider as _, raw::TableProvider as _};
196    if !identity_h(doc, font) {
197        return None;
198    }
199    let descendant = dict(doc, font.get(b"DescendantFonts").and_then(|a| doc.resolve(a).as_array().and_then(|a| a.first()).cloned()).as_ref())?;
200    let cid_map = match descendant.get(b"CIDToGIDMap").map(|m| doc.resolve(m)).as_deref() {
201        None => None,
202        Some(Object::Name(n)) if n == b"Identity" => None,
203        // Read as the font's other streams are, a map behind several filters too.
204        Some(Object::Stream(s)) => Some(Reads::new(doc, budget).decode(s, super::MAX_CID_TO_GID)?),
205        _ => return None,
206    };
207    let face = skrifa::FontRef::new(bytes).ok()?;
208    let loca = face.loca(None).ok()?;
209    let advances = face.glyph_metrics(skrifa::instance::Size::unscaled(), skrifa::instance::LocationRef::default());
210    let outlined = |gid: u32| loca.get_raw(gid as usize).zip(loca.get_raw(gid as usize + 1)).is_some_and(|(a, b)| b > a);
211    let glyph_of = |cid: u16| -> Option<u32> {
212        match &cid_map {
213            None => Some(u32::from(cid)),
214            Some(map) => {
215                let at = usize::from(cid).checked_mul(2)?;
216                Some(u32::from(u16::from_be_bytes(map.get(at..at + 2)?.try_into().ok()?)))
217            }
218        }
219    };
220    let mut unicode = BTreeMap::new();
221    let mut codes = BTreeMap::new();
222    for code in 0..=u16::MAX {
223        let Some(gid) = glyph_of(code) else { continue };
224        if gid == 0 || gid as usize >= loca.len().saturating_sub(1) {
225            continue;
226        }
227        let text = metrics.text_of(u32::from(code));
228        let ch = text.as_deref().and_then(only_char);
229        let space = ch == Some(' ') && advances.advance_width(skrifa::GlyphId::new(gid)).is_some_and(|w| w > 0.0);
230        if !outlined(gid) && !space {
231            continue;
232        }
233        codes.entry(gid).or_insert(code);
234        if let Some(ch) = ch.filter(|c| !c.is_control()) {
235            unicode.entry(u32::from(ch)).and_modify(|old: &mut u32| *old = (*old).min(gid)).or_insert(gid);
236        }
237    }
238    // The verified THSarabunPSK glyph order is retained by these PDF subsets. Restore the
239    // Windows Thai PUA alternates for HarfRust's legacy shaper, only where outlines survive.
240    // See https://linux.thai.net/~thep/th-otf/shaping.html (legacy PUA assignments).
241    let family = metrics.base_font.rsplit('+').next().unwrap_or(&metrics.base_font);
242    let alternate_meanings = metrics.text_of(352).is_some_and(|t| t == "่") && metrics.text_of(353).is_some_and(|t| t == "้");
243    // Each alternate the subset kept must advance as its form does: U+F700 and U+F70F are ฐ and
244    // ญ without their descenders, the rest marks with no advance of their own. Any that doesn't
245    // means another glyph order, and then none is used: the wrong mark would draw silently.
246    let kept: Vec<(usize, u32)> = (342u32..=368).enumerate().filter(|&(_, gid)| outlined(gid)).collect();
247    let in_order =
248        kept.iter().all(|&(index, gid)| advances.advance_width(skrifa::GlyphId::new(gid)).is_some_and(|w| (w > 0.0) == matches!(index, 0 | 15)));
249    // Where the PDF says what an alternate's code is (an intact ToUnicode gives the character it
250    // stands for), it must be that alternate's base character: two alternates of the same kind
251    // in another order pass the advance check, but not this one.
252    let code_of = |gid: u32| match &cid_map {
253        None => u16::try_from(gid).ok(),
254        Some(_) => codes.get(&gid).copied(),
255    };
256    let meanings_agree = kept.iter().all(|&(index, gid)| {
257        let stated = code_of(gid).and_then(|code| metrics.text_of(u32::from(code))).and_then(|t| only_char(&t));
258        stated.filter(|c| matches!(u32::from(*c), 0x0e01..=0x0e5b)).is_none_or(|c| PUA_BASES.get(index) == Some(&c))
259    });
260    if family.starts_with("THSarabunPSK")
261        && unicode.get(&0x0e01) == Some(&214)
262        && unicode.get(&0x0e2d) == Some(&258)
263        && (alternate_meanings || unicode.contains_key(&0xf70a))
264        && in_order
265        && meanings_agree
266    {
267        for (index, gid) in kept {
268            let code = match &cid_map {
269                None => u16::try_from(gid).ok(),
270                Some(_) => codes.get(&gid).copied(),
271            };
272            if let Some(code) = code {
273                unicode.insert(0xf700 + index as u32, gid);
274                codes.entry(gid).or_insert(code);
275            }
276        }
277    }
278    Some((unicode, codes))
279}

281/// Shape Thai in a THSarabunPSK source font's own outlines and return codes for that same font.
282/// No font program, family, weight or size is replaced. A program with OpenType Thai mark
283/// positioning (a complete face loaded with Load font, say) is shaped by that layout; one
284/// without it must be a verified legacy positional program, shaped with its nominal glyphs and
285/// Windows PUA alternates. A subset missing a glyph the text needs is refused, so the caller can
286/// try another subset of the same face or ask for its full font.
287pub struct LegacyThaiFont {
288    bytes: Vec<u8>,
289    codes: BTreeMap<u32, u16>,
290    space: f64,
291    /// The program positions Thai marks itself (GPOS `mark`/`mkmk`).
292    positioned: bool,
293}
294
295impl LegacyThaiFont {
296    /// Read and verify the source font once for all wrapping/preview measurements. `None` for
297    /// another family, a font with no Thai, and a program without Thai mark positioning whose
298    /// Thai glyphs are not in the legacy order: the nominal and alternate marks this shaper
299    /// assumes can't be trusted there, so it is shaped as an ordinary source font.
300    ///
301    /// Its streams, its metrics' too, are decoded against `budget`, which the caller shares
302    /// between every font it reads: what they keep is the programs decoded and the editor cmaps
303    /// added to them. Metrics read past it adapt nothing: the font is refused.
304    pub fn from_pdf(doc: &Document, font: &Dict, space: f64, budget: &mut DecodeBudget) -> Result<Option<Self>, String> {
305        let m = Metrics::from_dict_within(doc, font, budget);
306        if !m.base_font.rsplit('+').next().unwrap_or(&m.base_font).starts_with("THSarabunPSK") {
307            return Ok(None);
308        }
309        if m.past_allowance {
310            return Err("the source Thai font's metrics are past the edit's allowance".into());
311        }
312        if !space.is_finite() || space < 0.0 {
313            return Err("the source space width is invalid".into());
314        }
315        let mut bytes = super::embedded_sfnt_within(doc, font, budget).ok_or("the source Thai font has no readable glyph program")?;
316        let positioned = crate::fonts::source_shaping::has_thai_layout(&bytes);
317        let (unicode, codes) = composite_glyph_maps(doc, font, &bytes, &m, budget).ok_or("the Thai source glyph codes cannot be recovered safely")?;
318        // A subset keeps only the glyphs its document used, so no particular letter or alternate
319        // is required here: [`Self::shape`] refuses a text whose glyphs, or required alternates
320        // ([`require_pua`]), the subset lacks. Without its own positioning, every Thai glyph it
321        // does keep must be where the legacy order puts that character, or one of the PUA
322        // alternates for a mark (or for ฐ and ญ without their descenders).
323        let mut thai = unicode.range(0x0e01..=0x0e5b).peekable();
324        if thai.peek().is_none() {
325            return Ok(None);
326        }
327        let usable = positioned
328            || thai.all(|(&ch, &gid)| {
329                legacy_glyph(ch) == Some(gid)
330                    || ((342..=368).contains(&gid) && (char::from_u32(ch).is_some_and(is_thai_mark) || matches!(ch, 0x0e0d | 0x0e10)))
331            });
332        if !usable {
333            return Ok(None);
334        }
335        if !positioned {
336            // Glyphs 342–368 are alternates moved for a context. Mapped as the nominal character
337            // (its glyph dropped), one would be drawn out of place where the nominal form is
338            // needed: the shaper reaches it only through its PUA form, and that text is refused.
339            use skrifa::MetadataProvider as _;
340            let alternate = |u: u32, g: skrifa::GlyphId| matches!(u, 0x0e01..=0x0e5b) && (342..=368).contains(&g.to_u32());
341            let charmap = skrifa::FontRef::new(&bytes).map_err(|e| e.to_string())?.charmap();
342            if charmap.mappings().any(|(u, g)| alternate(u, g)) {
343                let nominal: BTreeMap<u32, u32> =
344                    charmap.mappings().filter(|&(u, g)| g.to_u32() != 0 && !alternate(u, g)).take(65_537).map(|(u, g)| (u, g.to_u32())).collect();
345                super::append_editor_cmap(&mut bytes, &nominal).ok_or("the Thai source glyph map cannot be rebuilt safely")?;
346            }
347        }
348        Ok(Some(Self { bytes, codes, space, positioned }))
349    }
350
351    /// Shape into glyphs that are already in this exact source face.
352    pub fn shape(&self, text: &str) -> Result<ExistingThaiRun, String> {
353        use skrifa::MetadataProvider as _;
354        if text.len() > 262_144 || text.contains(['\n', '\r']) {
355            return Err("shape one bounded Thai source row at a time".into());
356        }
357        // Only the legacy shaper needs the PUA alternates; a positioned program moves its marks.
358        if !self.positioned {
359            let face = skrifa::FontRef::new(&self.bytes).map_err(|e| e.to_string())?;
360            require_pua(&face.charmap(), text)?;
361        }
362        let mut shaped = crate::fonts::source_shaping::ShapedRun { glyphs: Vec::new(), advance: 0.0 };
363        let mut codes = Vec::new();
364        // Legacy PDF subsets often omit the space glyph. Keep its measured PDF advance as a
365        // positioning gap, rather than drawing .notdef or changing to another font.
366        for piece in text.split_inclusive(' ') {
367            let word = piece.strip_suffix(' ').unwrap_or(piece);
368            if !word.is_empty() {
369                let run = if self.positioned {
370                    crate::fonts::source_shaping::shape(&self.bytes, word)?
371                } else {
372                    crate::fonts::source_shaping::shape_legacy_thai(&self.bytes, word)?
373                };
374                for mut glyph in run.glyphs {
375                    let code = self
376                        .codes
377                        .get(&glyph.id.to_u32())
378                        .ok_or("the Thai subset lacks a required contextual glyph; load the complete original face")?;
379                    glyph.position[0] += shaped.advance;
380                    codes.push(code.to_be_bytes().to_vec());
381                    shaped.glyphs.push(glyph);
382                }
383                shaped.advance += run.advance;
384            }
385            if piece.ends_with(' ') {
386                shaped.glyphs.push(crate::fonts::source_shaping::ShapedGlyph {
387                    id: skrifa::GlyphId::NOTDEF,
388                    position: [shaped.advance, 0.0],
389                    text: " ".into(),
390                });
391                codes.push(Vec::new());
392                shaped.advance += self.space;
393            }
394            if shaped.glyphs.len() > 65_534 || !shaped.advance.is_finite() {
395                return Err("Thai source row exceeds the shaping budget".into());
396            }
397        }
398        Ok(ExistingThaiRun { shaped, codes })
399    }
400}

402/// Verify every context-dependent PUA alternate before asking the legacy shaper to use it.
403/// Otherwise a subset missing an alternate would silently draw the wrong high/left mark.
404fn require_pua(cmap: &skrifa::charmap::Charmap<'_>, text: &str) -> Result<(), String> {
405    let mut base = 0u32;
406    let mut above = false;
407    let mut chars = text.chars().peekable();
408    while let Some(ch) = chars.next() {
409        let c = u32::from(ch);
410        let ascender = matches!(base, 0xe1b | 0xe1d | 0xe1f);
411        let pua = match c {
412            0xe31 if ascender => Some(0xf710),
413            0xe34..=0xe37 if ascender => Some(0xf701 + c - 0xe34),
414            0xe47 if ascender => Some(0xf712),
415            0xe4d if ascender => Some(0xf711),
416            0xe48..=0xe4c => {
417                let high = above || chars.peek().is_some_and(|c| *c == 'ำ');
418                if ascender {
419                    Some((if high { 0xf713 } else { 0xf705 }) + c - 0xe48)
420                } else if !high {
421                    Some(0xf70a + c - 0xe48)
422                } else {
423                    None
424                }
425            }
426            0xe38..=0xe3a if matches!(base, 0xe0e | 0xe0f) => Some(0xf718 + c - 0xe38),
427            0xe38..=0xe3a if base == 0xe0d => Some(0xf70f),
428            0xe38..=0xe3a if base == 0xe10 => Some(0xf700),
429            0xe33 if ascender => Some(0xf711),
430            _ => None,
431        };
432        if pua.is_some_and(|p| char::from_u32(p).is_none_or(|c| cmap.map(c).is_none())) {
433            return Err(format!(
434                "the Thai subset lacks a required contextual mark for {ch} (U+{c:04X}); load the complete original face or explicitly select a replacement font"
435            ));
436        }
437        if matches!(c, 0xe01..=0xe2e) {
438            base = c;
439            above = false;
440        } else if matches!(c, 0xe31 | 0xe34..=0xe37 | 0xe47 | 0xe4d..=0xe4e) {
441            above = true;
442        } else if !matches!(c, 0xe38..=0xe3a | 0xe48..=0xe4c) {
443            base = 0;
444            above = false;
445        }
446    }
447    Ok(())
448}

456    /// Where table `tag` starts in an sfnt.
457    fn table(program: &[u8], tag: &[u8; 4]) -> usize {
458        let count = u16::from_be_bytes([program[4], program[5]]) as usize;
459        let record = (0..count).map(|i| 12 + i * 16).find(|&r| &program[r..r + 4] == tag).unwrap();
460        u32::from_be_bytes(program[record + 8..record + 12].try_into().unwrap()) as usize
461    }

463    /// Set glyph `gid`'s advance (a long `hmtx` entry).
464    fn set_advance(program: &mut [u8], gid: usize, advance: u16) {
465        let long = u16::from_be_bytes(program[table(program, b"hhea") + 34..][..2].try_into().unwrap()) as usize;
466        assert!(gid < long, "a long hmtx entry");
467        let at = table(program, b"hmtx") + gid * 4;
468        program[at..at + 2].copy_from_slice(&advance.to_be_bytes());
469    }

471    /// Empty glyph `gid`, as a subsetter drops it: its data is left as the previous glyph's padding.
472    fn drop_glyph(program: &mut [u8], gid: usize) {
473        let long = program[table(program, b"head") + 51] == 1;
474        let loca = table(program, b"loca");
475        if long {
476            let next: [u8; 4] = program[loca + (gid + 1) * 4..][..4].try_into().unwrap();
477            program[loca + gid * 4..][..4].copy_from_slice(&next);
478        } else {
479            let next: [u8; 2] = program[loca + (gid + 1) * 2..][..2].try_into().unwrap();
480            program[loca + gid * 2..][..2].copy_from_slice(&next);
481        }
482    }

484    /// Inter's outlines standing in for a THSarabunPSK subset (no proprietary font is used): the
485    /// glyphs where the exporter's order has a Thai mark or a mark-like alternate take no
486    /// advance, the others 500 units.
487    fn thai_ordered_program() -> Vec<u8> {
488        let mut program = crate::fonts::script::INTER_REGULAR.to_vec();
489        for gid in 214..=368usize {
490            let mark = match gid {
491                214..=298 => is_thai_mark(char::from_u32(gid as u32 + if gid <= 271 { 0xd2b } else { 0xd2f }).unwrap()),
492                342..=368 => !matches!(gid, 342 | 357),
493                _ => continue,
494            };
495            set_advance(&mut program, gid, if mark { 0 } else { 500 });
496        }
497        program
498    }

506    /// [`thai_font`] over the `/FontFile2` stream `file` and the `/CIDToGIDMap` `cid_to_gid`.
507    fn thai_font_over(doc: &mut Document, file: Object, cid_to_gid: Object, to_unicode: &[u8]) -> Dict {
508        let tu = doc.add(Object::Stream(Stream::flate(Dict::new(), to_unicode)));
509        let mut descriptor = Dict::new();
510        descriptor.set(b"FontFile2".to_vec(), file);
511        let mut desc = Dict::new();
512        desc.set(b"CIDToGIDMap".to_vec(), cid_to_gid);
513        desc.set(b"FontDescriptor".to_vec(), Object::Dict(descriptor));
514        let mut font = Dict::new();
515        font.set(b"Subtype".to_vec(), Object::name("Type0"));
516        font.set(b"BaseFont".to_vec(), Object::name("ABCDEF+THSarabunPSK"));
517        font.set(b"Encoding".to_vec(), Object::name("Identity-H"));
518        font.set(b"DescendantFonts".to_vec(), Object::Array(vec![Object::Dict(desc)]));
519        font.set(b"ToUnicode".to_vec(), Object::Ref(tu));
520        font
521    }

523    /// `data` as a Flate stream under a PNG predictor whose `/DecodeParms` is given by reference:
524    /// read without that dictionary, each row's tag byte stays in the decoded bytes.
525    fn predicted(doc: &mut Document, data: &[u8]) -> Object {
526        let rows: Vec<u8> = data.iter().flat_map(|byte| [0, *byte]).collect();
527        let mut parms = Dict::new();
528        parms.set(b"Predictor".to_vec(), Object::Int(12));
529        parms.set(b"Columns".to_vec(), Object::Int(1));
530        let parms = doc.add(Object::Dict(parms));
531        let mut stream = Stream::flate(Dict::new(), &rows);
532        stream.dict.set(b"DecodeParms".to_vec(), Object::Ref(parms));
533        Object::Ref(doc.add(Object::Stream(stream)))
534    }

646    /// `program` without the tables a legacy subsetter leaves out.
647    fn without_tables(program: &[u8], tags: &[&[u8; 4]]) -> Vec<u8> {
648        let mut program = program.to_vec();
649        let count = u16::from_be_bytes([program[4], program[5]]) as usize;
650        let records: Vec<Vec<u8>> =
651            program[12..12 + count * 16].chunks(16).filter(|r| !tags.iter().any(|t| &r[..4] == t.as_slice())).map(<[u8]>::to_vec).collect();
652        program[4..6].copy_from_slice(&(records.len() as u16).to_be_bytes());
653        for (i, record) in records.iter().enumerate() {
654            program[12 + i * 16..28 + i * 16].copy_from_slice(record);
655        }
656        program
657    }

794        doc.set(tu, Object::Stream(crate::cos::Stream::flate(Dict::new(), data.as_bytes())));
