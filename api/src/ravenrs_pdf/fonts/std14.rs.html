1//! Standard-14 font metrics (ISO 32000-2 Annex D; the Adobe Core14 AFM metrics), as advance
2//! widths in glyph space — thousandths of an em.
3//!
4//! A PDF may name any of the fourteen standard fonts without embedding a program, and the viewer
5//! has to supply the metrics. These are those metrics, exactly, replacing the character-class
6//! estimate that generated appearances used before: that estimate had Times at 0.9 x Helvetica,
7//! which makes a capital A 600 where the specification says 722.
8//!
9//! **Generated.** The tables are data, not typefaces, and are the item AGENTS.md §1.1 permits as
10//! Adobe-authored non-visual technical data; `ATTRIBUTION.toml` carries the entry with
11//! `adobe_data = true`. Each row is indexed by character **code**, not by Unicode: the twelve text
12//! fonts are indexed by WinAnsiEncoding, which is what PdfCraft writes into the appearance streams
13//! it generates, and Symbol and ZapfDingbats by their own built-in encodings. A zero means the
14//! encoding leaves that code undefined.
15
16/// One of the fourteen standard fonts (ISO 32000-2 §9.6.2.2).
17#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
18pub enum Std14 {
19    Helvetica,
20    HelveticaBold,
21    HelveticaOblique,
22    HelveticaBoldOblique,
23    TimesRoman,
24    TimesBold,
25    TimesItalic,
26    TimesBoldItalic,
27    Courier,
28    CourierBold,
29    CourierOblique,
30    CourierBoldOblique,
31    Symbol,
32    ZapfDingbats,
33}
34
35/// How [`Std14::recognise`] matched a `/BaseFont` name.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub enum Match {
38    /// A standard-14 face or a well-known alias of one (Arial for Helvetica, CourierNew for
39    /// Courier). The document is asking for this face.
40    Alias,
41    /// Only a family hint matched -- "Serif", "Roman" or "Mono" in an otherwise unknown name.
42    /// The face is a guess from the name, not the one the document asked for.
43    FamilyHint,
44}
45
46impl Std14 {
47    /// The font a `/BaseFont` name selects, by the spec's names and the common aliases viewers
48    /// accept (`Arial` for `Helvetica`, `TimesNewRoman` for `Times`, `CourierNew` for `Courier`).
49    /// A subset prefix such as `ABCDEF+Helvetica` is ignored.
50    pub fn from_base_font(base: &str) -> Option<Self> {
51        Self::recognise(base).map(|(face, _)| face)
52    }
53
54    /// [`Std14::from_base_font`], also saying *how* the name matched.
55    ///
56    /// Reporting needs the difference. "Arial" really is a request for Helvetica, but
57    /// "SomeHouseSerif" is a house font whose name merely contains "Serif"; choosing Times for it
58    /// is a reasonable guess from the name, not the face the document asked for, and a substitution
59    /// report that called it a standard-14 name would be telling the reader something untrue.
60    pub fn recognise(base: &str) -> Option<(Self, Match)> {
61        let base = base.rsplit('+').next().unwrap_or(base);
62        let flat: String = base.chars().filter(char::is_ascii_alphanumeric).map(|c| c.to_ascii_lowercase()).collect();
63        let has = |w: &str| flat.contains(w);
64        let (bold, italic) = (has("bold"), has("italic") || has("oblique"));
65        // Two bools index a four-element array, so the subscript is provably 0..=3.
66        let styled = |faces: [Self; 4]| faces[usize::from(bold) | usize::from(italic) << 1];
67        let courier = || styled([Self::Courier, Self::CourierBold, Self::CourierOblique, Self::CourierBoldOblique]);
68        let times = || styled([Self::TimesRoman, Self::TimesBold, Self::TimesItalic, Self::TimesBoldItalic]);
69        if has("symbol") {
70            return Some((Self::Symbol, Match::Alias));
71        }
72        if has("zapf") || has("dingbat") {
73            return Some((Self::ZapfDingbats, Match::Alias));
74        }
75        if has("courier") {
76            return Some((courier(), Match::Alias));
77        }
78        if has("mono") {
79            return Some((courier(), Match::FamilyHint));
80        }
81        if has("times") {
82            return Some((times(), Match::Alias));
83        }
84        if (has("serif") && !has("sans")) || has("roman") {
85            return Some((times(), Match::FamilyHint));
86        }
87        if has("helvetica") || has("arial") {
88            return Some((styled([Self::Helvetica, Self::HelveticaBold, Self::HelveticaOblique, Self::HelveticaBoldOblique]), Match::Alias));
89        }
90        None
91    }
92
93    /// The advance of one character `code`, in thousandths of an em. Zero where the font's
94    /// encoding leaves the code undefined.
95    pub fn width(self, code: u8) -> f64 {
96        f64::from(crate::fonts::std14_metrics::WIDTHS[self as usize][code as usize])
97    }
98
99    /// The advance of `code` at `size` points.
100    pub fn advance(self, code: u8, size: f64) -> f64 {
101        self.width(code) * size / 1000.0
102    }
103
104    /// The advance of `text` at `size` points, encoded as the twelve text fonts are written:
105    /// WinAnsiEncoding, with characters it cannot represent replaced as `win_ansi` replaces them.
106    ///
107    /// Symbol and ZapfDingbats have their own built-in encodings and are not WinAnsi, so this
108    /// measures them as if they were; pass codes to [`Std14::width`] for those.
109    pub fn text_width(self, text: &str, size: f64) -> f64 {
110        crate::fonts::win_ansi(text).iter().map(|&c| self.width(c)).sum::<f64>() * size / 1000.0
111    }
112
113    /// Courier is the only fixed-pitch family of the fourteen.
114    pub fn is_fixed_pitch(self) -> bool {
115        matches!(self, Self::Courier | Self::CourierBold | Self::CourierOblique | Self::CourierBoldOblique)
116    }
117
118    /// The `/BaseFont` name for this font, as it is written into a PDF.
119    pub fn base_font(self) -> &'static str {
120        crate::fonts::std14_metrics::BASE_FONTS[self as usize]
121    }
122}

128    #[test]
129    fn sans_serif_names_are_not_mapped_to_times() {
130        assert!(!matches!(Std14::recognise("SansSerif"), Some((Std14::TimesRoman, _))));
131        assert_eq!(Std14::recognise("Arial-SansSerif"), Some((Std14::Helvetica, Match::Alias)));

134    /// Widths from the published Core14 metrics, typed in here by hand so the table is checked
135    /// against the specification rather than against whatever produced it.
136    #[test]
137    fn widths_match_the_published_metrics() {
138        let cases: &[(Std14, char, u16)] = &[
139            // Helvetica
140            (Std14::Helvetica, ' ', 278),
141            (Std14::Helvetica, 'A', 667),
142            (Std14::Helvetica, 'W', 944),
143            (Std14::Helvetica, 'i', 222),
144            (Std14::Helvetica, 'm', 833),
145            (Std14::Helvetica, '0', 556),
146            (Std14::Helvetica, '@', 1015),
147            (Std14::HelveticaBold, 'A', 722),
148            (Std14::HelveticaBold, 'i', 278),
149            // Times: not Helvetica scaled by 0.9, which is what the old estimate assumed.
150            (Std14::TimesRoman, ' ', 250),
151            (Std14::TimesRoman, 'A', 722),
152            (Std14::TimesRoman, 'W', 944),
153            (Std14::TimesRoman, 'i', 278),
154            (Std14::TimesRoman, 'm', 778),
155            (Std14::TimesBold, 'A', 722),
156            (Std14::TimesItalic, 'A', 611),
157            // Courier is fixed pitch at 600 for every glyph.
158            (Std14::Courier, ' ', 600),
159            (Std14::Courier, 'A', 600),
160            (Std14::Courier, 'm', 600),
161            (Std14::CourierBoldOblique, 'W', 600),
162        ];
163        for &(font, ch, want) in cases {
164            let code = crate::fonts::win_ansi(&ch.to_string());
165            assert_eq!(code.len(), 1, "{ch:?} is one WinAnsi byte");
166            assert_eq!(font.width(code[0]) as u16, want, "{font:?} {ch:?}");
167        }
168    }

170    #[test]
171    fn courier_is_fixed_pitch_everywhere_it_is_defined() {
172        for font in [Std14::Courier, Std14::CourierBold, Std14::CourierOblique, Std14::CourierBoldOblique] {
173            assert!(font.is_fixed_pitch());
174            for code in 0..=255u8 {
175                let w = font.width(code);
176                assert!(w == 0.0 || w == 600.0, "{font:?} code {code} has width {w}");
177            }
178        }
179        assert!(!Std14::Helvetica.is_fixed_pitch());
180    }

182    #[test]
183    fn every_text_font_covers_printable_ascii_and_latin1() {
184        for font in [Std14::Helvetica, Std14::TimesRoman, Std14::Courier, Std14::TimesBoldItalic] {
185            for code in 0x20..=0x7Eu8 {
186                assert!(font.width(code) > 0.0, "{font:?} has no width for ASCII {code}");
187            }
188            for code in 0xA0..=0xFFu8 {
189                assert!(font.width(code) > 0.0, "{font:?} has no width for Latin-1 {code}");
190            }
191        }
192    }

194    #[test]
195    fn symbol_and_dingbats_have_their_own_encodings() {
196        // Symbol's code 0x61 is alpha, not 'a'; ZapfDingbats' is a scissors glyph.
197        assert!(Std14::Symbol.width(0x61) > 0.0);
198        assert!(Std14::ZapfDingbats.width(0x61) > 0.0);
199        // Neither defines the Latin-1 upper half the text fonts do.
200        assert_eq!(Std14::Symbol.width(0xFF), 0.0);
201    }

227    fn base_font_aliases_and_subset_prefixes_resolve() {
228        assert_eq!(Std14::from_base_font("ABCDEF+Helvetica-Bold"), Some(Std14::HelveticaBold));

238    #[test]
239    fn text_width_sums_the_glyphs_and_scales_with_size() {
240        let h = Std14::Helvetica;
241        // "AW" = 667 + 944 = 1611 thousandths of an em.
242        assert!((h.text_width("AW", 1000.0) - 1611.0).abs() < 1e-9);
243        assert!((h.text_width("AW", 12.0) - 1611.0 * 12.0 / 1000.0).abs() < 1e-9);
244        assert_eq!(h.text_width("", 12.0), 0.0);
245        // Courier: every character is 600, so the width is purely a count.
246        assert!((Std14::Courier.text_width("hello", 10.0) - 5.0 * 6.0).abs() < 1e-9);
247    }

249    #[test]
250    fn hostile_text_does_not_panic_and_stays_finite() {
251        for font in [Std14::Helvetica, Std14::Symbol, Std14::Courier] {
252            for s in ["", "\0", "\u{202E}", "💥", &"a".repeat(10_000), "\u{FFFD}"] {
253                let w = font.text_width(s, 12.0);
254                assert!(w.is_finite() && w >= 0.0, "{font:?} {s:?} -> {w}");
255            }
256            assert!(font.advance(0, 12.0).is_finite());
257            assert!(font.advance(255, 0.0).is_finite());
258        }
259    }
