2//! chosen): each comment is copied onto the page with the same number, at the same place, with
3//! its appearance, pop-up and replies.

12/// The annotation subtypes copied as comments: the markup annotations of ISO 32000-2 Table 171,
13/// except redaction marks (`Redact`) and `Projection`.
14const COMMENTS: [&[u8]; 16] = [
15    b"Text",
16    b"FreeText",
17    b"Line",
18    b"Square",
19    b"Circle",
20    b"Polygon",
21    b"PolyLine",
22    b"Highlight",
23    b"Underline",
24    b"Squiggly",
25    b"StrikeOut",
26    b"Stamp",
27    b"Caret",
28    b"Ink",
29    b"FileAttachment",
30    b"Sound",
31];

33/// The Locked annotation flag.
34const LOCKED: i64 = 128;

36/// One comment of the source: its entry in the page's `/Annots` and its dictionary.
37struct Incoming {
38    entry: Object,
39    dict: Dict,
40}

42/// Copy the comments of `src` onto the pages of `dst`: the comments of page 1 onto page 1, those of
43/// page 2 onto page 2, and so on, each at the position it has in `src`. Comments on pages past the
44/// last page of `dst` are left out, and so is a reply whose comment is left out. Each copy keeps its
45/// appearance, its pop-up, its replies and its layer. Links, form fields and redaction marks are
46/// not copied.
47///
48/// A comment named (`/NM`) like one already on its page replaces that one, which keeps its object
49/// so replies to it stay attached. A locked comment, or an annotation of that name that is no
50/// comment (a link, a redaction mark, a form widget), is not replaced: the copy is added beside it
51/// without the name. Returns the number of comments copied.
52pub fn import_comments(dst: &mut Document, src: &Document) -> Result<usize, OrganizeError> {
53    let from = walk(src)?;
54    let to = walk(dst)?;
55    let pages: Vec<(ObjRef, ObjRef)> = from.iter().zip(&to).map(|((s, _), (d, _))| (*s, *d)).collect();
56    let mut incoming: Vec<Vec<Incoming>> = pages.iter().map(|(s, _)| page_comments(src, *s)).collect();
57    let kept = threads_kept(src, &incoming);
58    for list in &mut incoming {
59        list.retain(|c| match irt_parent(src, &c.dict) {
60            None => true,
61            Some(Some(p)) => kept.contains(&p),
62            Some(None) => false,
63        });
64    }
65    let mut copier = Copier::new(src, HashMap::new(), pages.iter().copied().collect());
66    // Every comment's object is chosen before any is copied, so a comment that another one reaches
67    // through `/IRT` is copied into the object chosen for it by name.
68    let mut lists: Vec<Option<PageAnnots>> = Vec::new();
69    let mut planned: Vec<(usize, ObjRef, Dict)> = Vec::new();
70    for (i, ((_, page), list)) in pages.iter().zip(incoming).enumerate() {
71        if list.is_empty() {
72            lists.push(None);
73            continue;
74        }
75        let mut annots = PageAnnots::read(dst, *page);
76        for Incoming { entry, dict } in list {
77            // The same comment listed twice on a page is copied once.
78            if entry.as_ref().is_some_and(|r| copier.map.contains_key(&r)) {
79                continue;
80            }
81            let (target, dict) = reserve(dst, src, &mut annots, dict);
82            if let Some(r) = entry.as_ref() {
83                copier.map.insert(r, target);
84            }
85            annots.push(target);
86            planned.push((i, target, dict));
87        }
88        lists.push(Some(annots));
89    }
90    let mut annot_pages = HashMap::new();
91    for (i, target, dict) in &planned {
92        let copy = copier.copy_value(dst, &Object::Dict(dict.clone()), Some(*target))?;
93        dst.set(*target, copy);
94        annot_pages.insert(*target, pages[*i].1);
95    }
96    for (i, target, _) in &planned {
97        if let Some(popup) = dst.get(*target).as_dict().and_then(|d| d.reference(b"Popup"))
98            && let Some(annots) = lists[*i].as_mut()
99        {
100            annots.push(popup);
101            annot_pages.insert(popup, pages[*i].1);
102        }
103    }
104    // A pop-up that only its `/Parent` ties to a copied comment comes along too.
105    for (i, (from, to)) in pages.iter().enumerate() {
106        let Some(annots) = lists[i].as_mut() else { continue };
107        for (popup, parent) in page_popups(src, *from) {
108            if copier.map.contains_key(&popup) || !copier.map.contains_key(&parent) {
109                continue;
110            }
111            let r = dst.add(Object::Null);
112            copier.map.insert(popup, r);
113            let copy = copier.copy_value(dst, &src.get(popup), Some(r))?;
114            dst.set(r, copy);
115            annots.push(r);
116            annot_pages.insert(r, *to);
117        }
118    }
119    for ((_, page), annots) in pages.iter().zip(lists) {
120        if let Some(annots) = annots {
121            dst.update_dict(*page, |p| p.set(b"Annots".to_vec(), Object::Array(annots.into_list())))?;
122        }
123    }
124    copier.fix_annotations(dst, &annot_pages)?;
125    register_layers(dst, src, &copier.ocgs)?;
126    Ok(planned.len())
127}

129/// Choose the object a comment is copied into: the comment of `annots` with its name, or a new
130/// object. Returns it with the dictionary to copy.
131fn reserve(dst: &mut Document, src: &Document, annots: &mut PageAnnots, mut dict: Dict) -> (ObjRef, Dict) {
132    let nm = dict.get(b"NM").and_then(|n| src.resolve(n).as_string().map(|s| s.to_text()));
133    let same = nm.as_deref().and_then(|nm| annots.named(dst, nm));
134    // A second incoming comment with the name of one added earlier is added without the name.
135    let taken = nm.is_some_and(|nm| !annots.added.insert(nm));
136    let target = match same {
137        None if taken => {
138            dict.remove(b"NM");
139            dst.add(Object::Null)
140        }
141        Some((r, false)) => {
142            annots.claimed.insert(r);
143            // The replaced comment's pop-up goes with it: the one it names, and any whose `/Parent`
144            // is the comment.
145            if let Some(p) = dst.get(r).as_dict().and_then(|d| d.reference(b"Popup")) {
146                annots.remove(p);
147            }
148            for p in annots.popups.get(&r).cloned().unwrap_or_default() {
149                annots.remove(p);
150            }
151            r
152        }
153        Some((_, true)) => {
154            dict.remove(b"NM");
155            dst.add(Object::Null)
156        }
157        None => dst.add(Object::Null),
158    };
159    (target, dict)
160}

162/// Resolve `dict`'s `/IRT` entry: `None` when `/IRT` is absent or resolves to `null`,
163/// `Some(Some(r))` when it references annotation object `r` (directly or through indirect
164/// reference objects), or `Some(None)` for any other non-reference value.
165fn irt_parent(src: &Document, dict: &Dict) -> Option<Option<ObjRef>> {
166    let v = dict.get(b"IRT")?;
167    if matches!(v, Object::Null) {
168        return None;
169    }
170    let Some(mut r) = v.as_ref() else {
171        return Some(None);
172    };
173    for _ in 0..8 {
174        match *src.get(r) {
175            Object::Null => return None,
176            Object::Ref(next) => r = next,
177            _ => break,
178        }
179    }
180    Some(Some(r))
181}

183/// The comments of `incoming` that go: a reply goes only with the comment it answers, and that one
184/// only with the comment it answers, up to the first comment of the thread. A thread whose replies
185/// answer each other in a loop goes, as all of its comments are there. Each comment is visited once.
186fn threads_kept(src: &Document, incoming: &[Vec<Incoming>]) -> HashSet<ObjRef> {
187    // For each comment: no parent, a parent comment, or a parent that is no comment of the source.
188    let parents: HashMap<ObjRef, Option<Option<ObjRef>>> =
189        incoming.iter().flatten().filter_map(|c| Some((c.entry.as_ref()?, irt_parent(src, &c.dict)))).collect();
190    let mut known: HashMap<ObjRef, bool> = HashMap::new();
191    for &start in parents.keys() {
192        let (mut path, mut on_path) = (Vec::new(), HashSet::new());
193        let mut at = start;
194        let kept = loop {
195            if let Some(&k) = known.get(&at) {
196                break k;
197            }
198            if !on_path.insert(at) {
199                break true;
200            }
201            match parents.get(&at) {
202                None => break false,
203                Some(None) => {
204                    path.push(at);
205                    break true;
206                }
207                Some(Some(None)) => {
208                    path.push(at);
209                    break false;
210                }
211                Some(Some(Some(parent))) => {
212                    path.push(at);
213                    at = *parent;
214                }
215            }
216        };
217        for r in path {
218            known.insert(r, kept);
219        }
220    }
221    known.into_iter().filter_map(|(r, k)| k.then_some(r)).collect()
222}

224/// The comments in `page`'s `/Annots`, in order.
225fn page_comments(src: &Document, page: ObjRef) -> Vec<Incoming> {
226    let list = src.get(page).as_dict().and_then(|p| p.get(b"Annots").cloned()).map(|a| src.resolve(&a)).and_then(|a| a.as_array().cloned());
227    list.unwrap_or_default()
228        .into_iter()
229        .filter_map(|entry| {
230            let mut dict = src.resolve(&entry).as_dict().cloned()?;
231            let subtype = dict.get(b"Subtype").map(|s| src.resolve(s)).and_then(|s| s.as_name().map(<[u8]>::to_vec))?;
232            if let Some(Some(r)) = irt_parent(src, &dict) {
233                dict.set(b"IRT".to_vec(), Object::Ref(r));
234            }
235            COMMENTS.contains(&subtype.as_slice()).then_some(Incoming { entry, dict })
236        })
237        .collect()
238}

240/// The pop-ups in `page`'s `/Annots` with the comment each belongs to: (pop-up, its `/Parent`).
241fn page_popups(src: &Document, page: ObjRef) -> Vec<(ObjRef, ObjRef)> {
242    let list = src.get(page).as_dict().and_then(|p| p.get(b"Annots").cloned()).map(|a| src.resolve(&a)).and_then(|a| a.as_array().cloned());
243    list.unwrap_or_default()
244        .iter()
245        .filter_map(|entry| {
246            let r = entry.as_ref()?;
247            let d = src.get(r).as_dict().cloned()?;
248            let is_popup = d.get(b"Subtype").map(|s| src.resolve(s)).as_deref().and_then(Object::as_name) == Some(b"Popup".as_slice());
249            is_popup.then(|| d.reference(b"Parent").map(|p| (r, p)))?
250        })
251        .collect()
252}

254/// The annotation in `annots` named `nm` (compared as text, whatever the encoding), and whether it
255/// must not be replaced: a locked comment, or an annotation that is no comment, such as a link, a
256/// redaction mark or a form widget.
257/// A page's `/Annots` while comments are added to it, indexed so that placing each comment does not
258/// scan the whole list.
259struct PageAnnots {
260    list: Vec<Object>,
261    members: HashSet<ObjRef>,
262    /// The annotations on the page before the import, by `/NM`, in list order.
263    names: HashMap<String, Vec<ObjRef>>,
264    /// The pop-ups on the page before the import, by `/Parent`.
265    popups: HashMap<ObjRef, Vec<ObjRef>>,
266    removed: HashSet<ObjRef>,
267    /// The annotations that an incoming comment has already replaced.
268    claimed: HashSet<ObjRef>,
269    /// The names of the incoming comments placed so far.
270    added: HashSet<String>,
271}
272
273impl PageAnnots {
274    /// The `/Annots` of `page`, with inline annotation dictionaries made indirect objects.
275    fn read(dst: &mut Document, page: ObjRef) -> Self {
276        let list: Vec<Object> = dst
277            .get(page)
278            .as_dict()
279            .and_then(|p| p.get(b"Annots").cloned())
280            .map(|a| dst.resolve(&a))
281            .and_then(|a| a.as_array().cloned())
282            .unwrap_or_default();
283        let list: Vec<Object> = list
284            .into_iter()
285            .map(|a| match a {
286                Object::Dict(_) => Object::Ref(dst.add(a)),
287                a => a,
288            })
289            .collect();
290        let (mut members, mut names, mut popups) = (HashSet::new(), HashMap::<String, Vec<ObjRef>>::new(), HashMap::<ObjRef, Vec<ObjRef>>::new());
291        for r in list.iter().filter_map(Object::as_ref) {
292            members.insert(r);
293            let Some(d) = dst.get(r).as_dict().cloned() else { continue };
294            if let Some(name) = d.get(b"NM").and_then(|n| dst.resolve(n).as_string().map(|s| s.to_text())) {
295                names.entry(name).or_default().push(r);
296            }
297            if d.get(b"Subtype").map(|s| dst.resolve(s)).as_deref().and_then(Object::as_name) == Some(b"Popup".as_slice())
298                && let Some(parent) = d.reference(b"Parent")
299            {
300                popups.entry(parent).or_default().push(r);
301            }
302        }
303        PageAnnots { list, members, names, popups, removed: HashSet::new(), claimed: HashSet::new(), added: HashSet::new() }
304    }
305
306    /// Add `r` at the end, unless the page already lists it.
307    fn push(&mut self, r: ObjRef) {
308        if self.removed.remove(&r) {
309            self.list.retain(|a| a.as_ref() != Some(r));
310        }
311        if self.members.insert(r) {
312            self.list.push(Object::Ref(r));
313        }
314    }
315
316    /// Take every entry for `r` off the page.
317    fn remove(&mut self, r: ObjRef) {
318        if self.members.remove(&r) {
319            self.removed.insert(r);
320        }
321    }
322
323    /// The first annotation on the page named `nm`, and whether it must be kept: it is locked, or
324    /// it is no comment.
325    fn named(&self, dst: &Document, nm: &str) -> Option<(ObjRef, bool)> {
326        let mut listed = self.names.get(nm)?.iter().filter(|r| self.members.contains(r));
327        let first = *listed.clone().next()?;
328        // A second incoming comment with the name of one already replaced is added beside it.
329        let Some(&r) = listed.find(|r| !self.claimed.contains(r)) else { return Some((first, true)) };
330        let d = dst.get(r).as_dict().cloned()?;
331        let flags = d.get(b"F").and_then(|f| dst.resolve(f).as_int()).unwrap_or(0);
332        let subtype = d.get(b"Subtype").map(|s| dst.resolve(s));
333        let comment = subtype.as_deref().and_then(Object::as_name).is_some_and(|s| COMMENTS.contains(&s));
334        Some((r, flags & LOCKED != 0 || !comment || d.contains(b"FT")))
335    }
336
337    /// The new `/Annots`.
338    fn into_list(self) -> Vec<Object> {
339        let removed = self.removed;
340        self.list.into_iter().filter(|a| a.as_ref().is_none_or(|r| !removed.contains(&r))).collect()
341    }
342}

352    fn build(bodies: &[&str]) -> Document {
353        let mut out = b"%PDF-1.7\n".to_vec();
354        let mut offsets = Vec::new();
355        for (i, b) in bodies.iter().enumerate() {
356            offsets.push(out.len());
357            out.extend_from_slice(format!("{} 0 obj\n{b}\nendobj\n", i + 1).as_bytes());
358        }
359        let xref = out.len();
360        out.extend_from_slice(format!("xref\n0 {}\n0000000000 65535 f \n", bodies.len() + 1).as_bytes());
361        for o in offsets {
362            out.extend_from_slice(format!("{o:010} 00000 n \n").as_bytes());
363        }
364        out.extend_from_slice(format!("trailer\n<< /Size {} /Root 1 0 R >>\nstartxref\n{xref}\n%%EOF\n", bodies.len() + 1).as_bytes());
365        Document::open(Arc::new(out)).expect("opens")
366    }

420    fn comments_of_another_pdf_land_on_the_pages_with_the_same_numbers() {
421        let mut doc = original();
422        let src = reviewed();
423        assert_eq!(import_comments(&mut doc, &src).unwrap(), 5, "four comments on page 1 and the highlight on page 2");
424        let pages: Vec<ObjRef> = walk(&doc).unwrap().into_iter().map(|(p, _)| p).collect();
425        let one = page_annots(&doc, 0);
426        for (_, d) in &one {

428        }
429
430        // The note named like one already there replaces it in place, with its new pop-up.
431        let (note, d) = by_contents(&one, "from the reviewer");
432        assert_eq!(*note, ObjRef::new(5, 0), "the replaced comment keeps its object");
433        let popup = d.reference(b"Popup").unwrap();
434        assert_ne!(popup, ObjRef::new(8, 0));
435        assert_eq!(doc.get(popup).as_dict().unwrap().reference(b"Parent"), Some(*note));
436        let popups: Vec<ObjRef> = one.iter().filter(|(_, d)| d.name(b"Subtype") == Some(b"Popup".as_slice())).map(|(r, _)| *r).collect();
437        assert_eq!(popups, [popup], "the replaced note's pop-up goes, the copy's is listed once");
438        assert_eq!(by_contents(&one, "a reply").1.reference(b"IRT"), Some(*note), "the reply answers the note");
439
440        // The locked comment of that name stays; the copy is added beside it, unnamed.
441        assert_eq!(text(&by_contents(&one, "mine").1, b"NM").as_deref(), Some("locked"));
442        assert_eq!(text(&by_contents(&one, "the reviewer's").1, b"NM"), None);
443        assert!(one.iter().any(|(_, d)| text(d, b"NM").as_deref() == Some("keep")), "the document's own comments stay");
444
445        // The stamp keeps its appearance; the redaction mark is not copied.

447        let ap = doc.resolve(stamp.get(b"AP").unwrap()).as_dict().cloned().unwrap();

449        assert!(one.iter().all(|(_, d)| d.name(b"Subtype") != Some(b"Redact".as_slice())));
450
451        // Page 3 is past the document's last page: its square and the reply to it are left out.
452        assert!(one.iter().all(|(_, d)| text(d, b"Contents").as_deref() != Some("about page 3")));
453        assert_eq!(one.len(), 7, "three of the document's own, three new copies and the note's new pop-up");

458        assert_eq!(two[0].1.name(b"Subtype"), Some(b"Highlight".as_slice()));
459        assert_eq!(two[0].1.reference(b"P"), Some(pages[1]));
460
461        let saved = Document::open(Arc::new(write_full(&doc, &SaveOptions::default()).unwrap())).unwrap();
462        assert_eq!((page_annots(&saved, 0).len(), page_annots(&saved, 1).len()), (7, 1), "a saved copy reads back the same");
463    }

544    #[test]
545    fn a_long_thread_goes_with_its_first_comment_and_a_loop_of_replies_stays() {
546        // Page 1: replies 6, 7, … each answering the next, the last answering the square on page 3,
547        // which the document has no page for. Then two replies that answer each other.
548        let n = 20_000;
549        let first = 6;
550        let mut bodies: Vec<String> =
551            vec!["<< /Type /Catalog /Pages 2 0 R >>".into(), "<< /Type /Pages /Kids [3 0 R 4 0 R 5 0 R] /Count 3 /MediaBox [0 0 300 400] >>".into()];
552        let list: String = (first..first + n + 2).map(|o| format!("{o} 0 R ")).collect();
553        bodies.push(format!("<< /Type /Page /Parent 2 0 R /Annots [{list}] >>"));
554        bodies.push("<< /Type /Page /Parent 2 0 R >>".into());
555        let square = first + n + 2;
556        bodies.push(format!("<< /Type /Page /Parent 2 0 R /Annots [{square} 0 R] >>"));
557        for o in first..first + n {
558            let parent = if o + 1 == first + n { square } else { o + 1 };
559            bodies.push(format!("<< /Type /Annot /Subtype /Text /IRT {parent} 0 R /Rect [0 0 10 10] /P 3 0 R >>"));
560        }
561        let (a, b) = (first + n, first + n + 1);
562        bodies.push(format!("<< /Type /Annot /Subtype /Text /IRT {b} 0 R /Contents (one) /Rect [0 0 10 10] /P 3 0 R >>"));
563        bodies.push(format!("<< /Type /Annot /Subtype /Text /IRT {a} 0 R /Contents (two) /Rect [0 0 10 10] /P 3 0 R >>"));
564        bodies.push("<< /Type /Annot /Subtype /Square /Rect [0 0 100 100] /P 5 0 R >>".into());
565        let src = build(&bodies.iter().map(String::as_str).collect::<Vec<_>>());
566        let mut doc = original();
567        let started = std::time::Instant::now();
568        assert_eq!(import_comments(&mut doc, &src).unwrap(), 2, "only the two replies that answer each other");
569        assert!(started.elapsed() < std::time::Duration::from_secs(10), "{:?}", started.elapsed());
570        let one = page_annots(&doc, 0);
571        let (one_ref, one_dict) = by_contents(&one, "one");
572        let (two_ref, two_dict) = by_contents(&one, "two");
573        assert_eq!((one_dict.reference(b"IRT"), two_dict.reference(b"IRT")), (Some(*two_ref), Some(*one_ref)));
574    }
