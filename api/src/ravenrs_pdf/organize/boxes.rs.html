1//! Page boxes (ISO 32000-2 §14.11.2): Acrobat's Set Page Boxes and Crop, execution plan M4.4.
2//!
3//! The crop, bleed, trim and art boxes default to their "parent" box (crop → media; the others
4//! → crop) and are clipped to the media box. Margins are measured inward from the media box in
5//! default user space (unrotated), as Acrobat's dialog does.
6
7use crate::cos::{Document, Object};
8
9use crate::organize::{OrganizeError, check, walk};
10
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum PageBox {
13    Media,
14    Crop,
15    Bleed,
16    Trim,
17    Art,
18}
19
20impl PageBox {
21    pub fn key(self) -> &'static [u8] {
22        match self {
23            PageBox::Media => b"MediaBox",
24            PageBox::Crop => b"CropBox",
25            PageBox::Bleed => b"BleedBox",
26            PageBox::Trim => b"TrimBox",
27            PageBox::Art => b"ArtBox",
28        }
29    }
30
31    pub fn from_name(s: &str) -> Option<Self> {
32        let lower = s.trim().to_ascii_lowercase();
33        match lower.strip_suffix("box").unwrap_or(&lower) {
34            "media" => Some(Self::Media),
35            "crop" => Some(Self::Crop),
36            "bleed" => Some(Self::Bleed),
37            "trim" => Some(Self::Trim),
38            "art" => Some(Self::Art),
39            _ => None,
40        }
41    }
42}

44/// What to set a box to.
45#[derive(Clone, Copy, Debug, PartialEq)]
46pub enum BoxSpec {
47    /// Back to its default (the parent box).
48    Remove,
49    /// An absolute rectangle in default user space.
50    Rect([f64; 4]),
51    /// Margins inward from the media box: left, bottom, right, top (points).
52    Margins([f64; 4]),
53}
54
55/// Parses a 4-number PDF rectangle `[x0, y0, x1, y1]` into `[xmin, ymin, xmax, ymax]` when it has
56/// finite coordinates and positive width and height.
57fn rect_of(o: Option<&Object>, doc: &Document) -> Option<[f64; 4]> {
58    let o = doc.resolve(o?);
59    let v: Vec<f64> = o.as_array()?.iter().filter_map(|x| doc.resolve(x).as_f64()).collect();
60    if v.len() != 4 || !v.iter().all(|x| x.is_finite()) {
61        return None;
62    }
63    let r = [v[0].min(v[2]), v[1].min(v[3]), v[0].max(v[2]), v[1].max(v[3])];
64    let (w, h) = (r[2] - r[0], r[3] - r[1]);
65    (w > 0.0 && h > 0.0 && w.is_finite() && h.is_finite()).then_some(r)
66}
67
68/// Returns the coordinate-wise intersection of two normalized `[xmin, ymin, xmax, ymax]` rectangles.
69fn intersect(a: [f64; 4], b: [f64; 4]) -> [f64; 4] {
70    [a[0].max(b[0]), a[1].max(b[1]), a[2].min(b[2]), a[3].min(b[3])]
71}
72
73/// Clips rectangle `a` to `b`, returning `None` when their intersection has no positive area.
74fn clip(a: [f64; 4], b: [f64; 4]) -> Option<[f64; 4]> {
75    let r = intersect(a, b);
76    (r[2] > r[0] && r[3] > r[1]).then_some(r)
77}

79/// The effective boxes of every page: media, crop, bleed, trim, art.
80pub fn page_boxes(doc: &Document) -> Result<Vec<[[f64; 4]; 5]>, OrganizeError> {
81    Ok(walk(doc)?
82        .iter()
83        .map(|(page, inherited)| {
84            let obj = doc.get(*page);
85            let d = obj.as_dict().cloned().unwrap_or_default();
86            let own = |k: &[u8]| rect_of(d.get(k).or_else(|| inherited.get(k)), doc);
87            let media = own(b"MediaBox").unwrap_or([0.0, 0.0, 612.0, 792.0]);
88            let crop = own(b"CropBox").and_then(|c| clip(c, media)).unwrap_or(media);
89            let child = |k: &[u8]| rect_of(d.get(k), doc).and_then(|c| clip(c, media)).unwrap_or(crop);
90            [media, crop, child(b"BleedBox"), child(b"TrimBox"), child(b"ArtBox")]
91        })
92        .collect())
93}

95/// Set a box on `pages` (0-based). Margins apply to each page's own media box.
96pub fn set_page_box(doc: &mut Document, pages: &[usize], which: PageBox, spec: BoxSpec) -> Result<(), OrganizeError> {
97    let all = walk(doc)?;
98    check(pages, all.len())?;
99    let boxes = page_boxes(doc)?;
100    let bad = |m: &str| OrganizeError::InvalidBox(m.to_string());
101    if which == PageBox::Media && spec == BoxSpec::Remove {
102        return Err(bad("the media box can't be removed"));
103    }
104    // Validate every page first so a failure changes nothing.
105    let mut plan = Vec::new();
106    for &i in pages {
107        let media = boxes[i][0];
108        let rect = match spec {
109            BoxSpec::Remove => None,
110            BoxSpec::Rect(r) => {
111                if !r.iter().all(|x| x.is_finite()) {
112                    return Err(bad("the rectangle is not a number"));
113                }
114                let r = [r[0].min(r[2]), r[1].min(r[3]), r[0].max(r[2]), r[1].max(r[3])];
115                Some(if which == PageBox::Media { r } else { intersect(r, media) })
116            }
117            BoxSpec::Margins(m) => {
118                if !m.iter().all(|x| x.is_finite() && *x >= 0.0) {
119                    return Err(bad("margins must be zero or more"));
120                }
121                Some([media[0] + m[0], media[1] + m[1], media[2] - m[2], media[3] - m[3]])
122            }
123        };
124        if let Some(r) = rect {
125            let (w, h) = (r[2] - r[0], r[3] - r[1]);
126            if !w.is_finite() || !h.is_finite() {
127                return Err(bad("the rectangle is not a number"));
128            }
129            if w < 1.0 || h < 1.0 {
130                return Err(bad(&format!("page {} would be smaller than a point", i + 1)));
131            }
132        }
133        plan.push((i, all[i].0, rect, media));
134    }
135    for &(_, page, rect, _) in &plan {
136        doc.update_dict(page, |d| match rect {
137            Some(r) => d.set(which.key().to_vec(), Object::Array(r.iter().map(|v| Object::Real(*v)).collect())),
138            None => {
139                d.remove(which.key());
140            }
141        })?;
142    }
143    if spec == BoxSpec::Remove && which == PageBox::Crop {
144        // An inherited crop box from an ancestor `/Pages` node would come back after removal:
145        // pin the page's media box explicitly only when an ancestor still supplies a valid CropBox.
146        let after = walk(doc)?;
147        for &(i, page, _, media) in &plan {
148            if after.get(i).and_then(|(_, inh)| rect_of(inh.get(b"CropBox"), doc)).is_some() {
149                doc.update_dict(page, |d| d.set(b"CropBox".to_vec(), Object::Array(media.iter().map(|v| Object::Real(*v)).collect())))?;
150            }
151        }
152    }
153    Ok(())
154}
