1//! Keeping the catalog's XMP metadata (`/Root /Metadata`, ISO 32000-2 §14.3.2) in step with the
2//! document information when Document Properties ▸ Description changes it.
3//!
4//! Files that carry XMP state the title and author twice, and viewers that read XMP first show
5//! the packet's value, so changing only `/Info` leaves the file disagreeing with itself. Each
6//! information key maps to the XMP property PDF/A pairs it with (ISO 19005-1 §6.7.3, the same
7//! mapping `pdfcraft-preflight` writes): Title → `dc:title`, Author → `dc:creator`, Subject →
8//! `dc:description`, Keywords → `pdf:Keywords`, Creator → `xmp:CreatorTool`, Producer →
9//! `pdf:Producer`. A custom property (Document Properties ▸ Custom) is kept as `pdfx:<name>`, plain
10//! text in the `http://ns.adobe.com/pdfx/1.3/` schema where Acrobat keeps them, so tools that read
11//! XMP see it too; one whose name can't be an XML element name stays in `/Info` only, and a
12//! PDF/A packet gets no new one (see [`sync_info`]).
13//!
14//! The packet is edited as text, touching only the property that changed: everything else in it
15//! (other schemas, padding, unknown properties) stays byte for byte. A file without XMP gets
16//! none, and a packet this can't edit safely (not UTF-8, no `rdf:RDF`, too large to decode) is
17//! left as it was.

27/// Decoding limit for an XMP packet (they are a few kB; padding rarely takes them past 100 kB).
28const MAX_XMP: usize = 16 << 20;
29
30const DC: &str = "http://purl.org/dc/elements/1.1/";
31const PDF: &str = "http://ns.adobe.com/pdf/1.3/";
32const XMP: &str = "http://ns.adobe.com/xap/1.0/";
33/// Adobe's PDF extension schema: custom document properties, each under its own name.
34const PDFX: &str = "http://ns.adobe.com/pdfx/1.3/";
35
36/// How an XMP property holds its value.
37#[derive(Clone, Copy, Debug, PartialEq)]
38enum Shape {
39    /// A language alternative: `rdf:Alt` with an `x-default` item.
40    LangAlt,
41    /// An ordered array: `rdf:Seq` (the authors).
42    Seq,
43    /// Plain text.
44    Text,
45}
46
47/// The XMP property (qualified name, namespace) that mirrors a document information key.
48fn property(key: &str) -> Option<(&'static str, &'static str, Shape)> {
49    Some(match key {
50        "Title" => ("dc:title", DC, Shape::LangAlt),
51        "Author" => ("dc:creator", DC, Shape::Seq),
52        "Subject" => ("dc:description", DC, Shape::LangAlt),
53        "Keywords" => ("pdf:Keywords", PDF, Shape::Text),
54        "Creator" => ("xmp:CreatorTool", XMP, Shape::Text),
55        "Producer" => ("pdf:Producer", PDF, Shape::Text),
56        _ => return None,
57    })
58}

60/// After the information entry `key` became `value` (empty: removed), make the catalog's XMP
61/// packet say the same: the standard keys' properties, and `pdfx:<key>` for a custom property.
62/// Does nothing for standard keys XMP doesn't mirror (the dates, Trapped), custom names that
63/// aren't XML names, files without XMP, and packets that can't be edited safely.
64///
65/// A custom property is left alone where `pdfx:` would not mean Adobe's schema (a packet binding
66/// the prefix to another namespace keeps its own property), and in a packet that declares PDF/A
67/// one is changed or removed where it stands but never added: PDF/A-1 to -3 allow a schema only
68/// when the packet describes it in an extension schema (ISO 19005-2 §6.6.2.3), so a new `pdfx:`
69/// property would end the file's conformance. `/Info` keeps it either way.
70pub(crate) fn sync_info(doc: &mut Document, key: &str, value: &str) -> Result<(), OrganizeError> {
71    if let Some((name, ns, shape)) = property(key) {
72        return edit_packet(doc, |xmp| set_property(xmp, name, ns, shape, value)).map(|_| ());
73    }
74    if crate::organize::STANDARD_INFO_KEYS.contains(&key) || !is_xml_name(key) {
75        return Ok(());
76    }
77    let name = format!("pdfx:{key}");
78    edit_packet(doc, |xmp| {
79        if binds_prefix_elsewhere(xmp, "pdfx", PDFX) {
80            return None;
81        }
82        // A declaration counts whatever its value says (CDATA, markup): it is the claim that matters.
83        if !has_property(xmp, &name) && has_property(xmp, "pdfaid:part") {
84            return None;
85        }
86        set_property(xmp, &name, PDFX, Shape::Text, value)
87    })
88    .map(|_| ())
89}

91/// Remove the `pdfx:` copies of the custom properties `keys` (see [`sync_info`]), unless the packet
92/// binds `pdfx` to another namespace, whose properties are its own; whether any was there.
93pub(crate) fn remove_custom_properties(doc: &mut Document, keys: &[&str]) -> Result<bool, OrganizeError> {
94    let names: Vec<String> = keys.iter().map(|k| format!("pdfx:{k}")).collect();
95    edit_packet(doc, |xmp| if binds_prefix_elsewhere(xmp, "pdfx", PDFX) { None } else { remove_all(xmp, &names) })
96}

98/// Whether `xmp` has the property `name`, as an element or an attribute, whatever its value.
99fn has_property(xmp: &str, name: &str) -> bool {
100    element_span(xmp, name).is_some() || attribute_span(xmp, name).is_some()
101}

103/// Whether `xmp` declares `prefix` for a namespace other than `ns` anywhere (`xmlns:pdfx="…"` on
104/// any element), so a property written with that prefix may belong to another schema. A
105/// declaration whose value never ends counts as another namespace.
106fn binds_prefix_elsewhere(xmp: &str, prefix: &str, ns: &str) -> bool {
107    let attr = format!("xmlns:{prefix}");
108    // Each declaration is read only up to its closing quote, and the first other value stops the
109    // search, so this is linear in the packet.
110    xmp.match_indices(&attr).any(|(i, _)| {
111        // `xmlns:pdfx` must end there (not `xmlns:pdfxid`), then `=` and a quoted value.
112        let Some(rest) = xmp.get(i + attr.len()..).and_then(|r| r.trim_start().strip_prefix('=')).map(str::trim_start) else {
113            return false;
114        };
115        let Some(quote) = rest.chars().next().filter(|c| matches!(c, '"' | '\'')) else { return false };
116        rest.get(1..).and_then(|v| v.get(..v.find(quote)?)).is_none_or(|value| value != ns)
117    })
118}

120/// Whether `name` can follow `pdfx:` in an element name: an XML name without a colon (XML 1.0
121/// §2.3 `NameStartChar` then `NameChar`s; Namespaces in XML `NCName`). Custom property names may
122/// hold characters XML names can't (`1st`, `Q&A`, `a:b`); those are not mirrored rather than
123/// written under an escaped name of our own.
124fn is_xml_name(name: &str) -> bool {
125    let start = |c: char| {
126        matches!(c,
127            'A'..='Z' | '_' | 'a'..='z' | '\u{C0}'..='\u{D6}' | '\u{D8}'..='\u{F6}' | '\u{F8}'..='\u{2FF}' | '\u{370}'..='\u{37D}'
128            | '\u{37F}'..='\u{1FFF}' | '\u{200C}'..='\u{200D}' | '\u{2070}'..='\u{218F}' | '\u{2C00}'..='\u{2FEF}'
129            | '\u{3001}'..='\u{D7FF}' | '\u{F900}'..='\u{FDCF}' | '\u{FDF0}'..='\u{FFFD}' | '\u{10000}'..='\u{EFFFF}')
130    };
131    let rest = |c: char| start(c) || matches!(c, '-' | '.' | '0'..='9' | '\u{B7}' | '\u{300}'..='\u{36F}' | '\u{203F}'..='\u{2040}');
132    let mut chars = name.chars();
133    chars.next().is_some_and(start) && chars.all(rest)
134}

136/// Remove every occurrence of the properties `names` from the catalog's XMP packet; whether any
137/// was there.
138pub(crate) fn remove_properties(doc: &mut Document, names: &[&str]) -> Result<bool, OrganizeError> {
139    edit_packet(doc, |xmp| remove_all(xmp, names))
140}

142/// `xmp` without any occurrence of the properties `names`; `None` when none was there.
143fn remove_all(xmp: &str, names: &[impl AsRef<str>]) -> Option<String> {
144    let mut out: Option<String> = None;
145    for name in names {
146        // Each removal shortens the packet, so this ends.
147        while let Some(next) = set_property(out.as_deref().unwrap_or(xmp), name.as_ref(), "", Shape::Text, "") {
148            out = Some(next);
149        }
150    }
151    out
152}

154/// Rewrite the catalog's XMP packet with `edit` (`None`: unchanged); whether it changed. Files
155/// without XMP and packets that can't be edited safely are left alone.
156fn edit_packet(doc: &mut Document, edit: impl FnOnce(&str) -> Option<String>) -> Result<bool, OrganizeError> {
157    let Some(root) = doc.root() else { return Ok(false) };
158    // A metadata stream is always indirect (§7.3.8); anything else is left alone.
159    let Some(r) = doc.get(root).as_dict().and_then(|c| c.reference(b"Metadata")) else { return Ok(false) };
160    let obj = doc.get(r);
161    let Object::Stream(stream) = &*obj else { return Ok(false) };
162    let Ok(bytes) = stream.decoded_within(MAX_XMP) else { return Ok(false) };
163    let Ok(xmp) = String::from_utf8(bytes) else { return Ok(false) };
164    let Some(updated) = edit(&xmp) else { return Ok(false) };
165    // Written unfiltered, as PDF/A asks of metadata streams, so other tools can read it.
166    let mut dict = stream.dict.clone();
167    for k in [&b"Filter"[..], b"DecodeParms", b"DL", b"Length"] {
168        dict.remove(k);
169    }
170    doc.set(r, Object::Stream(Stream::from_raw(dict, updated.into_bytes())));
171    Ok(true)
172}

174/// The value the catalog's XMP packet gives the document information key `key`, for files that
175/// state their title, author and so on only there (`/Info` is deprecated in PDF 2.0, and some
176/// writers leave it out). A language alternative gives its `x-default` item (else the first),
177/// several authors are joined with "; ". `None` for keys XMP doesn't mirror, files without XMP,
178/// packets that can't be read safely, and empty values.
179pub(crate) fn info_value(doc: &Document, key: &str) -> Option<String> {
180    let (name, _, shape) = property(key)?;
181    let root = doc.root()?;
182    let r = doc.get(root).as_dict()?.reference(b"Metadata")?;
183    let xmp = decoded_packet(&doc.get(r))?;
184    get_property(&xmp, name, shape)
185}

187/// A decoded packet (`None`: unreadable) and the stream object it came from.
188type Held = (Weak<Object>, Option<Arc<str>>);
189
190thread_local! {
191    /// The packet [`decoded_packet`] last decoded (`None`: unreadable), with the stream object it
192    /// came from. The tab title can come from XMP and is read every frame; a packet decodes to
193    /// up to [`MAX_XMP`] bytes.
194    static PACKET: RefCell<Option<Held>> = const { RefCell::new(None) };
195}
196
197/// The text of the XMP stream `obj`, `None` when it isn't a stream or isn't UTF-8 within
198/// [`MAX_XMP`]. Decoded once while the document keeps handing out the same object: an edit or
199/// another document gives another one. The [`Weak`] keeps its address from being reused.
200fn decoded_packet(obj: &Arc<Object>) -> Option<Arc<str>> {
201    let same = |held: &Held| std::ptr::eq(held.0.as_ptr(), Arc::as_ptr(obj));
202    if let Ok(Some(text)) = PACKET.try_with(|p| p.borrow().as_ref().filter(|held| same(held)).map(|held| held.1.clone())) {
203        return text;
204    }
205    let Object::Stream(stream) = &**obj else { return None };
206    #[cfg(test)]
207    XMP_DECODES.with(|n| n.set(n.get() + 1));
208    let text = stream.decoded_within(MAX_XMP).ok().and_then(|b| String::from_utf8(b).ok()).map(Arc::<str>::from);
209    // Past the thread's end (a destructor reading it) the packet just isn't kept.
210    let _ = PACKET.try_with(|p| *p.borrow_mut() = Some((Arc::downgrade(obj), text.clone())));
211    text
212}

220/// The text of the property `name` in `xmp`, element or attribute form (see [`info_value`]).
221fn get_property(xmp: &str, name: &str, shape: Shape) -> Option<String> {
222    let value = if let Some((start, end)) = element_span(xmp, name) {
223        let element = xmp.get(start..end)?;
224        let open_end = element.find('>')?;
225        // `<dc:title/>`: no value.
226        if element.get(..open_end)?.ends_with('/') {
227            return None;
228        }
229        let inner = element.get(open_end + 1..)?.strip_suffix(&format!("</{name}>"))?;
230        let items = list_items(inner);
231        if items.is_empty() {
232            // An empty `<rdf:Alt/>` or `<rdf:Seq></rdf:Seq>` has no value, not its markup.
233            text_value(inner)?
234        } else if shape == Shape::LangAlt {
235            let (_, text) = items.iter().find(|(lang, _)| lang.as_deref() == Some("x-default")).or(items.first())?;
236            text_value(text)?
237        } else {
238            items.iter().filter_map(|(_, text)| text_value(text)).filter(|t| !t.is_empty()).collect::<Vec<_>>().join("; ")
239        }
240    } else {
241        let (_, value_start, value_end, _) = attribute_span(xmp, name)?;
242        unescape(xmp.get(value_start..value_end)?.trim())
243    };
244    Some(value).filter(|v| !v.trim().is_empty())
245}

247/// Character data `text` as a value, unescaped; `None` when it still holds markup (a `<`, which
248/// a value escapes as `&lt;`), so a structure the reader doesn't take apart never shows as text.
249fn text_value(text: &str) -> Option<String> {
250    let text = text.trim();
251    (!text.contains('<')).then(|| unescape(text))
252}

254/// The `rdf:li` items of a property's content: each one's `xml:lang` (if any) and its text.
255fn list_items(inner: &str) -> Vec<(Option<String>, &str)> {
256    let mut items = Vec::new();
257    let mut rest = inner;
258    while let Some(i) = rest.find("<rdf:li") {
259        let Some(after) = rest.get(i + "<rdf:li".len()..) else { break };
260        // `<rdf:li` must end there: not `<rdf:link`.
261        if !after.starts_with(|c: char| c == '>' || c == '/' || c.is_whitespace()) {
262            rest = after;
263            continue;
264        }
265        let Some(tag_end) = after.find('>') else { break };
266        let tag = after.get(..tag_end).unwrap_or_default();
267        let lang = tag.find("xml:lang=").and_then(|at| {
268            let v = tag.get(at + "xml:lang=".len()..)?;
269            let quote = v.chars().next().filter(|c| matches!(c, '"' | '\''))?;
270            let v = v.get(1..)?;
271            Some(v.get(..v.find(quote)?)?.to_string())
272        });
273        let Some(body) = after.get(tag_end + 1..) else { break };
274        if tag.ends_with('/') {
275            items.push((lang, ""));
276            rest = body;
277            continue;
278        }
279        let Some(close) = body.find("</rdf:li>") else { break };
280        items.push((lang, body.get(..close).unwrap_or_default()));
281        rest = body.get(close..).unwrap_or_default();
282    }
283    items
284}

286/// XML text with its character and entity references resolved (an unknown one is kept as is).
287fn unescape(text: &str) -> String {
288    let mut out = String::with_capacity(text.len());
289    let mut rest = text;
290    while let Some(i) = rest.find('&') {
291        out.push_str(rest.get(..i).unwrap_or_default());
292        let after = rest.get(i..).unwrap_or_default();
293        let Some(semi) = after.find(';').filter(|&n| n <= 12) else {
294            out.push('&');
295            rest = after.get(1..).unwrap_or_default();
296            continue;
297        };
298        let entity = after.get(1..semi).unwrap_or_default();
299        let c = match entity {
300            "amp" => Some('&'),
301            "lt" => Some('<'),
302            "gt" => Some('>'),
303            "quot" => Some('"'),
304            "apos" => Some('\''),
305            _ => entity
306                .strip_prefix("#x")
307                .or_else(|| entity.strip_prefix("#X"))
308                .filter(|h| h.bytes().all(|b| b.is_ascii_hexdigit()))
309                .and_then(|h| u32::from_str_radix(h, 16).ok())
310                .or_else(|| entity.strip_prefix('#').filter(|d| d.bytes().all(|b| b.is_ascii_digit())).and_then(|d| d.parse().ok()))
311                .and_then(char::from_u32),
312        };
313        match c {
314            Some(c) => {
315                out.push(c);
316                rest = after.get(semi + 1..).unwrap_or_default();
317            }
318            None => {
319                out.push('&');
320                rest = after.get(1..).unwrap_or_default();
321            }
322        }
323    }
324    out.push_str(rest);
325    out
326}

328/// `xmp` with the property `name` (namespace `ns`) set to `value`, or removed when `value` is
329/// empty; `None` when nothing changes or the packet has no `rdf:RDF` to add the property to.
330fn set_property(xmp: &str, name: &str, ns: &str, shape: Shape, value: &str) -> Option<String> {
331    let element = (!value.is_empty()).then(|| match shape {
332        Shape::LangAlt => format!("<{name}><rdf:Alt><rdf:li xml:lang=\"x-default\">{}</rdf:li></rdf:Alt></{name}>", esc(value)),
333        Shape::Seq => format!("<{name}><rdf:Seq><rdf:li>{}</rdf:li></rdf:Seq></{name}>", esc(value)),
334        Shape::Text => format!("<{name}>{}</{name}>", esc(value)),
335    });
336    // The element form, `<dc:title>…</dc:title>` (or `<dc:title/>`): replaced where it stands, so
337    // it keeps the namespace declaration of the description around it.
338    if let Some((start, end)) = element_span(xmp, name) {
339        let mut out = String::with_capacity(xmp.len());
340        out.push_str(xmp.get(..start)?);
341        out.push_str(element.as_deref().unwrap_or_default());
342        out.push_str(xmp.get(end..)?);
343        return Some(out);
344    }
345    // The attribute form, `pdf:Keywords="…"` on an `rdf:Description`: only simple text can be
346    // written there; a structured value moves to an element of its own.
347    if let Some((start, value_start, value_end, quote)) = attribute_span(xmp, name) {
348        let mut out = String::with_capacity(xmp.len());
349        if shape == Shape::Text && !value.is_empty() {
350            let escaped = if quote == '"' { esc(value) } else { esc(value).replace('\'', "&apos;") };
351            out.push_str(xmp.get(..value_start)?);
352            out.push_str(&escaped);
353            out.push_str(xmp.get(value_end..)?);
354            return Some(out);
355        }
356        out.push_str(xmp.get(..start)?);
357        out.push_str(xmp.get(value_end + 1..)?);
358        return match element {
359            Some(e) => insert_description(&out, name, ns, &e),
360            None => Some(out),
361        };
362    }
363    insert_description(xmp, name, ns, element.as_deref()?)
364}

366/// `xmp` with a new `rdf:Description` holding `element` (and declaring its namespace) as the last
367/// description of the packet.
368fn insert_description(xmp: &str, name: &str, ns: &str, element: &str) -> Option<String> {
369    let prefix = name.split(':').next()?;
370    let at = xmp.rfind("</rdf:RDF>")?;
371    // Every description of a packet describes the same resource (XMP: one `rdf:about` value).
372    let about = attribute_span(xmp, "rdf:about").and_then(|(_, s, e, _)| xmp.get(s..e)).unwrap_or_default().replace('"', "&quot;");
373    let mut out = String::with_capacity(xmp.len() + element.len() + 128);
374    out.push_str(xmp.get(..at)?);
375    out.push_str(&format!("<rdf:Description rdf:about=\"{about}\" xmlns:{prefix}=\"{ns}\">{element}</rdf:Description>\n"));
376    out.push_str(xmp.get(at..)?);
377    Some(out)
378}

380/// Where the element `name` is in `xmp`, from its `<` to past its end tag (or past `/>`).
381fn element_span(xmp: &str, name: &str) -> Option<(usize, usize)> {
382    let open = format!("<{name}");
383    let mut from = 0;
384    while let Some(i) = xmp.get(from..).and_then(|s| s.find(&open)).map(|i| i + from) {
385        let after = i + open.len();
386        from = after;
387        // `<dc:title` must end there: not `<dc:titles`.
388        match xmp.get(after..).and_then(|s| s.chars().next()) {
389            Some('>' | '/') => {}
390            Some(c) if c.is_whitespace() => {}
391            _ => continue,
392        }
393        let tag_end = after + xmp.get(after..)?.find('>')?;
394        if xmp.get(..tag_end)?.ends_with('/') {
395            return Some((i, tag_end + 1));
396        }
397        let close = format!("</{name}>");
398        let end = tag_end + xmp.get(tag_end..)?.find(&close)? + close.len();
399        return Some((i, end));
400    }
401    None
402}

404/// Where the attribute `name` is in `xmp`: its start (including the whitespace before it), the
405/// start and end of its value, and its quote character. Only matches inside a start tag.
406fn attribute_span(xmp: &str, name: &str) -> Option<(usize, usize, usize, char)> {
407    let mut from = 0;
408    // Track tag and quote state in one pass rather than searching back from every match.
409    let (mut scanned, mut in_tag, mut quote) = (0, false, None);
410    while let Some(i) = xmp.get(from..).and_then(|s| s.find(name)).map(|i| i + from) {
411        from = i + name.len();
412        let before = xmp.get(..i)?;
413        let gap = xmp.get(scanned..i)?;
414        #[cfg(test)]
415        crate::organize::work(gap.len());
416        for c in gap.chars() {
417            if let Some(q) = quote {
418                if c == q {
419                    quote = None;
420                }
421            } else {
422                match c {
423                    '<' => in_tag = true,
424                    '>' if in_tag => in_tag = false,
425                    '\'' | '\"' if in_tag => quote = Some(c),
426                    _ => {}
427                }
428            }
429        }
430        scanned = i;
431        // Preceded by whitespace, and inside a start tag.
432        if !before.ends_with(char::is_whitespace) || !in_tag || quote.is_some() {
433            continue;
434        }
435        let rest = xmp.get(from..)?;
436        let trimmed = rest.trim_start();
437        let Some(after_eq) = trimmed.strip_prefix('=') else { continue };
438        let after_eq = after_eq.trim_start();
439        let Some(quote) = after_eq.chars().next().filter(|c| matches!(c, '"' | '\'')) else { continue };
440        let value_start = xmp.len() - after_eq.len() + 1;
441        let value_end = value_start + xmp.get(value_start..)?.find(quote)?;
442        let start = before.trim_end().len();
443        return Some((start, value_start, value_end, quote));
444    }
445    None
446}

469    #[test]
470    fn values_are_read_back_unescaped() {
471        assert_eq!(unescape("a &amp; b &lt;c&gt; &#65;&#x42; &bogus; &#xD800; & end"), "a & b <c> AB &bogus; &#xD800; & end");
472        assert_eq!(get_property(PACKET, "dc:title", Shape::LangAlt).as_deref(), Some("Untitled 1"));
473        assert_eq!(get_property(PACKET, "dc:creator", Shape::Seq).as_deref(), Some("Old Author"));
474        assert_eq!(get_property(PACKET, "pdf:Producer", Shape::Text).as_deref(), Some("Writer 1.0"));
475        assert_eq!(get_property(PACKET, "pdf:Keywords", Shape::Text), None);
476        // No x-default: the first language; empty and self-closed items are skipped.
477        let alt = "<dc:title><rdf:Alt><rdf:li xml:lang='fr'>Rapport</rdf:li><rdf:li xml:lang='en'>Report</rdf:li></rdf:Alt></dc:title>";
478        assert_eq!(get_property(alt, "dc:title", Shape::LangAlt).as_deref(), Some("Rapport"));
479        let seq = "<dc:creator><rdf:Seq><rdf:li/><rdf:li> </rdf:li><rdf:li>Ada</rdf:li></rdf:Seq></dc:creator>";
480        assert_eq!(get_property(seq, "dc:creator", Shape::Seq).as_deref(), Some("Ada"));
481        // Unfinished markup reads as nothing rather than as a wrong value.
482        assert_eq!(get_property("<dc:title><rdf:Alt><rdf:li>Cut", "dc:title", Shape::LangAlt), None);
483        // An empty alternative or list, or an item holding markup, has no value: never the markup.
484        assert_eq!(get_property("<dc:title><rdf:Alt/></dc:title>", "dc:title", Shape::LangAlt), None);
485        assert_eq!(get_property("<dc:title><rdf:Alt></rdf:Alt></dc:title>", "dc:title", Shape::LangAlt), None);
486        assert_eq!(get_property("<dc:creator><rdf:Seq/></dc:creator>", "dc:creator", Shape::Seq), None);
487        let nested = "<dc:creator><rdf:Seq><rdf:li><rdf:Description/></rdf:li><rdf:li>Ada</rdf:li></rdf:Seq></dc:creator>";
488        assert_eq!(get_property(nested, "dc:creator", Shape::Seq).as_deref(), Some("Ada"));
489    }

522    #[test]
523    fn attribute_values_are_rewritten_where_they_stand() {
524        let packet = "<x:xmpmeta xmlns:x=\"adobe:ns:meta/\"><rdf:RDF xmlns:rdf=\"r\"><rdf:Description rdf:about='uuid:1' \
525                      xmlns:xmp=\"http://ns.adobe.com/xap/1.0/\" xmlns:dc=\"http://purl.org/dc/elements/1.1/\" xmp:CreatorTool='Old \"tool\"' dc:title=\"Odd\"/></rdf:RDF></x:xmpmeta>";
526        let out = set_property(packet, "xmp:CreatorTool", XMP, Shape::Text, "Writer's tool").unwrap();
527        assert!(out.contains("xmp:CreatorTool='Writer&apos;s tool'"), "{out}");
528        // A title can't be an attribute: it moves to an element, in a description for the same resource.
529        let out = set_property(packet, "dc:title", DC, Shape::LangAlt, "Guide").unwrap();
530        assert!(!out.contains("dc:title=\"Odd\""), "{out}");
531        assert!(out.contains("<rdf:Description rdf:about=\"uuid:1\" xmlns:dc=\"http://purl.org/dc/elements/1.1/\"><dc:title><rdf:Alt>"), "{out}");
532        let out = set_property(packet, "xmp:CreatorTool", XMP, Shape::Text, "").unwrap();
533        assert!(out.contains("xmlns:dc=\"http://purl.org/dc/elements/1.1/\" dc:title=\"Odd\"/>"), "{out}");
534    }

592    #[test]
593    fn attribute_search_scans_the_packet_once() {
594        // Thousands of attribute-like words in one text node: none is inside a tag.
595        let packet = format!("<x:xmpmeta><rdf:RDF><rdf:Description>{}</rdf:Description></rdf:RDF></x:xmpmeta>", " rdf:about".repeat(5_000));
596        crate::organize::take_work();
597        assert_eq!(attribute_span(&packet, "rdf:about"), None);
598        let work = crate::organize::take_work();
599        assert!(work <= 2 * packet.len(), "{work} bytes scanned in a {}-byte packet", packet.len());
600        // A real attribute after them is still found.
601        let packet = packet.replace("</rdf:Description>", "</rdf:Description><rdf:Description rdf:about='u'/>");
602        let (_, s, e, q) = attribute_span(&packet, "rdf:about").unwrap();
603        assert_eq!((packet.get(s..e), q), (Some("u"), '\''));
604    }
