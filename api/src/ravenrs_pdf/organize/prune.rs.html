1//! Keeps only the XObjects a part's pages actually draw (Split / Extract Pages, #204).
2//!
3//! In a real book all pages share one `/Resources` dictionary, so every split part carried all
4//! ~2,900 image XObjects while its 1,500 pages drew ~300 of them, and the part saved out at the
5//! source's size. This pass follows each part's content streams (`Do`, form XObjects included)
6//! and drops the `/XObject` entries no page names; the garbage-collecting rewrite on save then
7//! drops the objects themselves.
8
9use std::collections::{HashMap, HashSet};
10
11use crate::cos::{CosError, DecodeBudget, Dict, Document, ObjRef, Object};
12
13use crate::organize::OrganizeError;
14
15/// Form XObjects nested deeper than this are a broken file, not a reason to hang.
16const MAX_FORM_DEPTH: u8 = 8;
17/// Count every resource object and direct child; a graph beyond this is left unpruned.
18const MAX_GLYPH_RESOURCE_WORK: usize = 10_000;
19
20/// The dictionary a set of drawn names belongs to.
21#[derive(Clone, Copy, PartialEq, Eq, Hash)]
22enum XKey {
23    /// An indirect `/XObject` dictionary.
24    XObject(ObjRef),
25    /// An inline `/XObject` dictionary inside an indirect resources object.
26    Resources(ObjRef),
27    /// Everything inline on the page itself.
28    Page(ObjRef),
29}
30
31/// Who owns the resources currently being walked, for the key drawn names are recorded under.
32#[derive(Clone, Copy, PartialEq, Eq, Hash)]
33enum Owner {
34    /// An indirect resources object (a page's, or a form XObject's own).
35    Resources(ObjRef),
36    /// Resources inline on the page.
37    Page(ObjRef),
38}
39
40struct Pruner<'a> {
41    doc: &'a Document,
42    used: HashMap<XKey, HashSet<Vec<u8>>>,
43    /// Dictionaries whose drawn names can't be known for certain (undecodable content, Type 3
44    /// glyphs drawing through them). They are left untouched: pruning must never drop an XObject
45    /// something still draws.
46    unsure: HashSet<XKey>,
47    /// An overlarge glyph-resource graph is not safe to prune at all.
48    glyph_budget_exhausted: bool,
49    /// Nor is a file with forms nested past [`MAX_FORM_DEPTH`]: what the forms there draw isn't
50    /// read, and they may draw it through resources shared with content that was.
51    too_deep: bool,
52    glyph_work_left: usize,
53    glyph_seen: HashSet<ObjRef>,
54    /// An indirect font's explicit resources have one owner regardless of which page uses it.
55    glyph_fonts: HashSet<ObjRef>,
56    /// Forms already analysed, with the owner of the resources they drew against: drawing one
57    /// again records nothing new, and re-reading it on every `Do` is exponential in the nesting.
58    forms_seen: HashSet<(ObjRef, Owner)>,
59    /// Owners of resources that content not read whole drew against, whose every form was read
60    /// as drawn ([`Pruner::unread`]): once is enough.
61    partly_read: HashSet<Owner>,
62}
63
64impl Pruner<'_> {
65    /// Mark the `/XObject` dictionary of `resources` as one not to prune.
66    fn distrust(&mut self, resources: &Dict, owner: Owner) {
67        if let Some((_, key)) = self.xobjects(resources, owner) {
68            self.unsure.insert(key);
69        }
70    }
71
72    fn glyph_work(&mut self, count: usize) -> bool {
73        if !self.glyph_budget_exhausted
74            && let Some(left) = self.glyph_work_left.checked_sub(count)
75        {
76            self.glyph_work_left = left;
77            return true;
78        }
79        self.glyph_budget_exhausted = true;
80        false
81    }
82
83    /// Walk one object's direct children without cloning them or recursing. Referenced objects
84    /// go on a separate work list, so their borrowed dictionaries stay alive during this pass.
85    fn protect_glyph_tree(&mut self, object: &Object, referenced: &mut Vec<ObjRef>) {
86        if !self.glyph_work(1) {
87            return;
88        }
89        let mut pending = vec![object];
90        while let Some(object) = pending.pop() {
91            match object {
92                Object::Ref(r) => {
93                    // A glyph's forms, patterns and soft masks can share either the XObject
94                    // dictionary itself or a resource owner with content we do interpret.
95                    self.unsure.insert(XKey::XObject(*r));
96                    self.unsure.insert(XKey::Resources(*r));
97                    self.unsure.insert(XKey::Page(*r));
98                    if self.glyph_seen.insert(*r) {
99                        referenced.push(*r);
100                    }
101                }
102                Object::Array(array) => {
103                    if !self.glyph_work(array.len()) {
104                        return;
105                    }
106                    pending.extend(array);
107                }
108                Object::Dict(_) | Object::Stream(_) => {
109                    if let Some(dict) = object.as_dict() {
110                        if !self.glyph_work(dict.len()) {
111                            return;
112                        }
113                        pending.extend(dict.iter().map(|(_, value)| value));
114                    }
115                }
116                _ => {}
117            }
118        }
119    }
120
121    /// Glyph programs are not interpreted here. Keep every resource owner reachable from their
122    /// resource graph, including nested forms, patterns and soft masks. Data bytes are never
123    /// decoded; references are visited once and all direct children count against the budget.
124    fn protect_glyph_object(&mut self, object: &Object) {
125        let mut referenced = Vec::new();
126        self.protect_glyph_tree(object, &mut referenced);
127        while !self.glyph_budget_exhausted {
128            let Some(r) = referenced.pop() else { break };
129            let object = self.doc.get(r);
130            self.protect_glyph_tree(&object, &mut referenced);
131        }
132    }
133
134    fn protect_type3_resources(&mut self, resources: &Dict, owner: Owner) {
135        if self.glyph_budget_exhausted {
136            return;
137        }
138        let Some(fonts) = resources.get(b"Font").map(|f| self.doc.resolve(f)) else { return };
139        let Some(fonts) = fonts.as_dict() else { return };
140        // Bound discovery too, but don't spend the glyph-graph budget repeatedly examining an
141        // ordinary shared font catalog on every page of a large document.
142        if fonts.len() > MAX_GLYPH_RESOURCE_WORK {
143            self.glyph_budget_exhausted = true;
144            return;
145        }
146        for (_, font) in fonts.iter() {
147            if self.glyph_budget_exhausted {
148                return;
149            }
150            let font_ref = font.as_ref();
151            let resolved = self.doc.resolve(font);
152            let Some(font) = resolved.as_dict().filter(|f| f.name(b"Subtype") == Some(b"Type3")) else { continue };
153            if let Some(glyph_resources) = font.get(b"Resources") {
154                if let Some(r) = font_ref
155                    && !self.glyph_fonts.insert(r)
156                {
157                    continue;
158                }
159                self.protect_glyph_object(glyph_resources);
160            } else {
161                self.distrust(resources, owner);
162                // This root dictionary is already owned by the content caller. Only its
163                // referenced descendants can share additional owners elsewhere in the file.
164                for (_, value) in resources.iter() {
165                    self.protect_glyph_object(value);
166                    if self.glyph_budget_exhausted {
167                        return;
168                    }
169                }
170            }
171        }
172    }
173
174    /// The `/XObject` dictionary of `resources`, with the key its drawn names belong to.
175    fn xobjects(&self, resources: &Dict, owner: Owner) -> Option<(Dict, XKey)> {
176        let x = resources.get(b"XObject")?;
177        if let Some(r) = x.as_ref() {
178            return Some((self.doc.resolve(x).as_dict()?.clone(), XKey::XObject(r)));
179        }
180        let key = match owner {
181            Owner::Resources(r) => XKey::Resources(r),
182            Owner::Page(p) => XKey::Page(p),
183        };
184        Some((self.doc.resolve(x).as_dict()?.clone(), key))
185    }
186
187    /// Record every XObject name `data` draws with `Do`, descending into form XObjects.
188    fn content(&mut self, data: &[u8], resources: &Dict, owner: Owner, depth: u8) {
189        self.protect_type3_resources(resources, owner);
190        let Some((xobjects, key)) = self.xobjects(resources, owner) else {
191            return;
192        };
193        if depth > MAX_FORM_DEPTH {
194            self.too_deep = true;
195            return;
196        }
197        for op in crate::content::parse(data).ops {
198            if op.op.as_slice() != b"Do" {
199                continue;
200            }
201            let Some(name) = op.name(0) else { continue };
202            self.used.entry(key).or_default().insert(name.to_vec());
203            self.form(&xobjects, name, resources, owner, depth);
204        }
205    }
206
207    /// For content that can't be read whole (damaged partway, or not decoding at all): what it
208    /// draws is unknown, so its `/XObject` dictionary is kept whole, and every form there is read
209    /// as drawn, which also reads every form the part before any damage draws. A form whose
210    /// resources are shared keeps what it draws from them too.
211    fn unread(&mut self, resources: &Dict, owner: Owner, depth: u8) {
212        self.protect_type3_resources(resources, owner);
213        let Some((xobjects, key)) = self.xobjects(resources, owner) else { return };
214        self.unsure.insert(key);
215        if depth > MAX_FORM_DEPTH {
216            self.too_deep = true;
217            return;
218        }
219        if !self.partly_read.insert(owner) {
220            return;
221        }
222        for (name, _) in xobjects.iter() {
223            self.form(&xobjects, name, resources, owner, depth);
224        }
225    }
226
227    /// Record what the XObject `name` of `xobjects` draws when it is a form, drawn from content
228    /// with `resources`.
229    fn form(&mut self, xobjects: &Dict, name: &[u8], resources: &Dict, owner: Owner, depth: u8) {
230        let Some(r) = xobjects.get(name).and_then(Object::as_ref) else { return };
231        let Object::Stream(s) = &*self.doc.get(r) else { return };
232        // A `/Subtype` given by reference reads as no form here, as wherever the workspace tells
233        // forms from images (redaction, edit and Optimize among them): resolving it is a change
234        // across all of them.
235        if s.dict.name(b"Subtype") != Some(b"Form") {
236            return;
237        }
238        let (inner, inner_owner) = match s.dict.get(b"Resources") {
239            Some(res) => match res.as_ref() {
240                Some(rr) => match self.doc.get(rr).as_dict().cloned() {
241                    Some(d) => (d, Owner::Resources(rr)),
242                    None => return,
243                },
244                None => match self.doc.resolve(res).as_dict().cloned() {
245                    // The form carries its resources inline: filter them on the form itself.
246                    Some(d) => (d, Owner::Resources(r)),
247                    None => return,
248                },
249            },
250            // A form without its own resources draws against the ones it was invoked with.
251            None => (resources.clone(), owner),
252        };
253        if !self.forms_seen.insert((r, inner_owner)) {
254            return;
255        }
256        #[cfg(test)]
257        crate::organize::work(1);
258        match DecodeBudget::new(usize::MAX, 0).decode_with_damage(self.doc, s, usize::MAX) {
259            Ok((data, false)) => self.content(&data, &inner, inner_owner, depth + 1),
260            Ok((_, true)) | Err(_) => self.unread(&inner, inner_owner, depth + 1),
261        }
262    }
263}

265/// The page's content, concatenated (`/Contents` may be one stream or an array) with a space
266/// between pieces as the renderer joins them, decoded against one page's [`DecodeBudget`], and
267/// whether a stream is damaged partway (its part is then what decoded before the damage). An
268/// error when a stream can't be decoded or the budget runs out: the page may draw names that
269/// can't be seen.
270pub(crate) fn content_bytes(doc: &Document, contents: &Object) -> Result<(Vec<u8>, bool), CosError> {
271    let mut out = Vec::new();
272    let mut damaged = false;
273    for read in DecodeBudget::page().decode_contents_with_damage(doc, contents) {
274        let (data, partial) = read?;
275        damaged |= partial;
276        out.extend(data);
277        out.push(b' ');
278    }
279    Ok((out, damaged))
280}

282/// The resources of a page (as `walk` reports them) with their owner.
283fn resolve_resources(doc: &Document, resources: &Object, page: ObjRef) -> Option<(Dict, Owner)> {
284    match resources {
285        Object::Ref(r) => doc.get(*r).as_dict().cloned().map(|d| (d, Owner::Resources(*r))),
286        Object::Dict(d) => Some((d.clone(), Owner::Page(page))),
287        _ => None,
288    }
289}

291/// Drop the entries `names` never draws. Returns how many were removed.
292fn drop_unused(xobjects: &mut Dict, names: &HashSet<Vec<u8>>) -> usize {
293    let unused: Vec<Vec<u8>> = xobjects.iter().map(|(k, _)| k.clone()).filter(|k| !names.contains(k)).collect();
294    for k in &unused {
295        xobjects.remove(k);
296    }
297    unused.len()
298}

300/// Filter the dictionary `key` names to `names`.
301fn filter_at(doc: &mut Document, key: XKey, names: &HashSet<Vec<u8>>) -> Result<usize, OrganizeError> {
302    match key {
303        XKey::XObject(r) => {
304            let mut removed = 0;
305            doc.update_dict(r, |d| removed = drop_unused(d, names))?;
306            Ok(removed)
307        }
308        XKey::Resources(r) => {
309            let Some(d) = doc.get(r).as_dict().cloned() else { return Ok(0) };
310            let Some(x) = d.get(b"XObject") else { return Ok(0) };
311            if x.as_ref().is_some() {
312                return Ok(0); // recorded as XObject(r) instead
313            }
314            let Some(mut xd) = doc.resolve(x).as_dict().cloned() else { return Ok(0) };
315            let removed = drop_unused(&mut xd, names);
316            if removed > 0 {
317                doc.update_dict(r, |d| d.set(b"XObject".to_vec(), Object::Dict(xd.clone())))?;
318            }
319            Ok(removed)
320        }
321        XKey::Page(p) => {
322            let Some(pd) = doc.get(p).as_dict().cloned() else { return Ok(0) };
323            let Some(res) = pd.get(b"Resources") else { return Ok(0) };
324            if res.as_ref().is_some() {
325                return Ok(0); // recorded as Resources(r) instead
326            }
327            let Some(mut rd) = doc.resolve(res).as_dict().cloned() else { return Ok(0) };
328            let Some(x) = rd.get(b"XObject") else { return Ok(0) };
329            if x.as_ref().is_some() {
330                return Ok(0);
331            }
332            let Some(mut xd) = doc.resolve(x).as_dict().cloned() else { return Ok(0) };
333            let removed = drop_unused(&mut xd, names);
334            if removed > 0 {
335                rd.set(b"XObject".to_vec(), Object::Dict(xd));
336                doc.update_dict(p, |d| d.set(b"Resources".to_vec(), Object::Dict(rd.clone())))?;
337            }
338            Ok(removed)
339        }
340    }
341}

343/// Drop the page-resource entries the document's pages never draw. Returns how many were removed.
344pub(crate) fn prune_unused_xobjects(doc: &mut Document) -> Result<usize, OrganizeError> {
345    let pages = crate::organize::walk(doc)?;
346    let mut pruner = Pruner {
347        doc,
348        used: HashMap::new(),
349        unsure: HashSet::new(),
350        glyph_budget_exhausted: false,
351        too_deep: false,
352        glyph_work_left: MAX_GLYPH_RESOURCE_WORK,
353        glyph_seen: HashSet::new(),
354        glyph_fonts: HashSet::new(),
355        forms_seen: HashSet::new(),
356        partly_read: HashSet::new(),
357    };
358    for (page, attrs) in &pages {
359        let Some(res) = attrs.get(b"Resources") else { continue };
360        let Some((res_dict, owner)) = resolve_resources(pruner.doc, res, *page) else { continue };
361        let Some(pd) = pruner.doc.get(*page).as_dict().cloned() else { continue };
362        let Some(contents) = pd.get(b"Contents").cloned() else { continue };
363        match content_bytes(pruner.doc, &contents) {
364            Ok((data, false)) if data.is_empty() => {}
365            Ok((data, false)) => pruner.content(&data, &res_dict, owner, 0),
366            Ok((_, true)) | Err(_) => pruner.unread(&res_dict, owner, 0),
367        }
368    }
369    let Pruner { used, unsure, glyph_budget_exhausted, too_deep, .. } = pruner;
370    if glyph_budget_exhausted || too_deep {
371        return Ok(0);
372    }
373    let mut removed = 0;
374    for (key, names) in used {
375        if !unsure.contains(&key) {
376            removed += filter_at(doc, key, &names)?;
377        }
378    }
379    Ok(removed)
380}
