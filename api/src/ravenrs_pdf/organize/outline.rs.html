1//! Editing bookmarks (the document outline, ISO 32000-2 §12.3.3), execution plan M4.6.
2//!
3//! Bookmarks are addressed by their **path**: child indices from the top level, so `[0, 2]` is
4//! the third child of the first top-level bookmark. Paths follow `/First`–`/Next` order, which
5//! is also the order viewers (and our inspector) list them in.
6//!
7//! Edits are surgical: they relink only the affected siblings (`/Parent`, `/Prev`, `/Next`,
8//! `/First`, `/Last`) and recompute `/Count` values, and an object is rewritten only when one of
9//! its values changes. Everything else on an item (colour, style, actions, unknown keys) is kept.
10//! Walks are cycle-safe and iterative, so a broken or deep outline is never followed forever and
11//! never overflows the stack. Listing is lazy: [`bookmark_page`] visits only the bookmarks up to
12//! the end of the page it returns.
13

20/// Deepest level a listing enters (the top level is 0). Bookmarks deeper than this are not
21/// listed; edits still reach them by path.
22pub(crate) const LIST_DEPTH: usize = 32;

30/// A bookmark in listing order, with where it sits and where it goes.
31#[derive(Clone, Debug, PartialEq)]
32pub struct Bookmark {
33    pub obj: ObjRef,
34    /// Child indices from the top level (the paths the edit functions take).
35    pub path: Vec<usize>,
36    pub title: String,
37    /// Shown expanded (a positive `/Count`) when it has children.
38    pub open: bool,
39    /// The 0-based page it goes to, when that page is in this document.
40    pub page: Option<usize>,
41    /// Where on that page it goes: the destination's type (`XYZ`, `Fit`, `FitH`…) and the numbers
42    /// after it, in PDF user space (`None` for `null` or anything that isn't a number).
43    pub view: Option<(Vec<u8>, Vec<Option<f64>>)>,
44    /// The title's colour (`/C`, DeviceRGB, ISO 32000-2 §12.3.3 Table 151), each component kept
45    /// within 0–1; `None` for the default black or a value that isn't three numbers. The
46    /// inspector's `OutlineItem::color` reads it the same way.
47    pub color: Option<[f32; 3]>,
48    /// The title's style (`/F`, Table 152): bit 1 italic, bit 2 bold.
49    pub italic: bool,
50    pub bold: bool,
51}

61#[derive(Debug, thiserror::Error, PartialEq)]
62pub enum OutlineError {
63    #[error("there is no bookmark at {0:?}")]
64    NoSuchBookmark(Vec<usize>),
65    #[error("a bookmark cannot be moved inside itself")]
66    IntoItself,
67    #[error("the title must not be empty")]
68    EmptyTitle,
69    #[error("a bookmark's position must be a finite number")]
70    BadPosition,
71    #[error("a bookmark's colour must be three numbers from 0 to 1")]
72    BadColor,
73    #[error("this document has no tagged headings to make bookmarks from")]
74    NoEntries,
75    #[error("there are no bookmarks to add")]
76    NothingToImport,
77    #[error("the bookmark at {0:?} has a view that is not a known destination type with its values")]
78    BadView(Vec<usize>),
79    #[error("bookmarks may nest at most {0} levels deep")]
80    TooDeep(usize),
81    #[error("at most {0} bookmarks can be added at once")]
82    TooManyBookmarks(usize),
83    #[error("{0}")]
84    Organize(#[from] OrganizeError),
85    #[error("{0}")]
86    Cos(#[from] crate::cos::CosError),
87}

91fn outline_root(doc: &Document) -> Option<ObjRef> {

94}

96/// The children of `parent` (an item or the outline root), in order. Cycle-safe.
97fn children_of(doc: &Document, parent: ObjRef, seen: &mut HashSet<ObjRef>) -> Vec<ObjRef> {
98    let mut out = Vec::new();
99    let mut next = doc.get(parent).as_dict().and_then(|d| d.reference(b"First"));
100    while let Some(r) = next {
101        if !seen.insert(r) {
102            break;
103        }
104        let item = doc.get(r);
105        let Some(d) = item.as_dict() else { break };
106        out.push(r);
107        next = d.reference(b"Next");
108    }
109    out
110}

112/// Every bookmark item, parents before their children. Cycle-safe.
113pub(crate) fn items(doc: &Document) -> Vec<ObjRef> {
114    let Some(root) = outline_root(doc) else { return Vec::new() };
115    let (mut seen, mut out, mut stack) = (HashSet::new(), Vec::new(), vec![root]);
116    while let Some(parent) = stack.pop() {
117        let kids = children_of(doc, parent, &mut seen);
118        stack.extend(kids.iter().copied());
119        out.extend(kids);
120    }
121    out
122}

124/// The outline in listing order: each bookmark once, siblings by `/Next`, children right after
125/// their parent. An explicit stack keeps the walk iterative. A repeated or malformed item ends its
126/// run of siblings there, so a cycle or a broken link stops the walk instead of looping.
127struct Visit<'a> {
128    doc: &'a Document,
129    seen: HashSet<ObjRef>,
130    /// One entry per open level: the next sibling to visit, and how many siblings it has visited.
131    levels: Vec<(Option<ObjRef>, usize)>,
132    /// The deepest level the walk enters.
133    max_depth: usize,
134    /// The bookmark just visited: its children are entered by the next step.
135    enter: Option<ObjRef>,
136}
137
138impl<'a> Visit<'a> {
139    fn new(doc: &'a Document, max_depth: usize) -> Option<Self> {
140        let root = outline_root(doc)?;
141        let first = doc.get(root).as_dict().and_then(|d| d.reference(b"First"));
142        Some(Self { doc, seen: HashSet::from([root]), levels: vec![(first, 0)], max_depth, enter: None })
143    }
144
145    /// Visit the next bookmark, or `None` when the listing is done.
146    fn step(&mut self) -> Option<ObjRef> {
147        if let Some(first) = self.enter.take() {
148            self.levels.push((Some(first), 0));
149        }
150        loop {
151            let depth = self.levels.len().checked_sub(1)?;
152            let level = self.levels.get_mut(depth)?;
153            let Some(r) = level.0.take() else {
154                self.levels.pop();
155                continue;
156            };
157            if !self.seen.insert(r) {
158                continue;
159            }
160            let item = self.doc.get(r);
161            let Some(d) = item.as_dict() else { continue };
162            level.0 = d.reference(b"Next");
163            level.1 = level.1.saturating_add(1);
164            self.enter = if depth < self.max_depth { d.reference(b"First") } else { None };
165            return Some(r);
166        }
167    }
168
169    /// Child indices from the top level to the bookmark last visited.
170    fn path(&self) -> Vec<usize> {
171        self.levels.iter().map(|(_, visited)| visited.saturating_sub(1)).collect()
172    }
173}

175/// Page targets for a listing: the page index of each page object, and the named destinations
176/// (read once, the first time one is needed).
177struct Targets {
178    pages: HashMap<ObjRef, usize>,
179    names: Option<HashMap<Vec<u8>, Object>>,
180}
181
182impl Targets {
183    fn new(doc: &Document) -> Self {
184        let pages = walk(doc).unwrap_or_default().into_iter().enumerate().map(|(i, (r, _))| (r, i)).collect();
185        Self { pages, names: None }
186    }
187
188    /// The named destination `key`: its name-tree entry, else its entry in the `/Dests` dictionary.
189    fn named(&mut self, doc: &Document, key: &[u8]) -> Option<Object> {
190        let names = self.names.get_or_insert_with(|| name_tree(doc));
191        names.get(key).cloned().or_else(|| dests_dict(doc, key))
192    }
193}

195/// Every entry of the catalog's `/Names /Dests` name tree, by name. The first entry of a name wins.
196/// Iterative, and nodes deeper than `NAME_TREE_DEPTH` are skipped.
197fn name_tree(doc: &Document) -> HashMap<Vec<u8>, Object> {
198    let mut out = HashMap::new();
199    let Some(root) = doc.root() else { return out };
200    let Some(names) = doc.get(root).as_dict().and_then(|c| c.get(b"Names").cloned()) else { return out };
201    let names = doc.resolve(&names);
202    let Some(tree) = names.as_dict().and_then(|n| n.get(b"Dests").cloned()) else { return out };
203    // Nodes still to read, with their depth. Kids go on the stack in reverse, so they are read in order.
204    let mut stack = vec![(tree, 0usize)];
205    let mut seen: HashSet<ObjRef> = HashSet::new();
206    while let Some((raw, depth)) = stack.pop() {
207        if depth > NAME_TREE_DEPTH {
208            continue;
209        }
210        if let Object::Ref(r) = raw
211            && !seen.insert(r)
212        {
213            continue;
214        }
215        let node = doc.resolve(&raw);
216        let Some(d) = node.as_dict() else { continue };
217        let pairs = d.get(b"Names").map(|n| doc.resolve(n));
218        if let Some(pairs) = pairs.as_deref().and_then(Object::as_array) {
219            for pair in pairs.chunks(2) {
220                if let [k, v] = pair
221                    && let Some(key) = name_key(doc, k)
222                {
223                    out.entry(key).or_insert_with(|| v.clone());
224                }
225            }
226        }
227        let kids = d.get(b"Kids").map(|k| doc.resolve(k));
228        if let Some(kids) = kids.as_deref().and_then(Object::as_array) {
229            for kid in kids.iter().rev() {
230                stack.push((kid.clone(), depth + 1));
231            }
232        }
233    }
234    out
235}

237/// The bytes of a name-tree key, which is a string or a name.
238fn name_key(doc: &Document, key: &Object) -> Option<Vec<u8>> {
239    match &*doc.resolve(key) {
240        Object::String(s) => Some(s.bytes.clone()),
241        Object::Name(n) => Some(n.clone()),
242        _ => None,
243    }
244}

246/// A named destination in the catalog's PDF 1.1 `/Dests` dictionary.
247fn dests_dict(doc: &Document, key: &[u8]) -> Option<Object> {
248    let root = doc.root()?;
249    let dests = doc.get(root).as_dict().and_then(|c| c.get(b"Dests").cloned())?;
250    let dests = doc.resolve(&dests);
251    dests.as_dict()?.get(key).cloned()
252}

254/// The explicit destination array (`[page /XYZ …]`) a destination stands for: itself, a dictionary
255/// whose `/D` is one, or a named destination that resolves to one.
256fn explicit_dest(doc: &Document, targets: &mut Targets, dest: &Object, depth: u8) -> Option<Vec<Object>> {
257    if depth > DEST_DEPTH {
258        return None;
259    }
260    let dest = doc.resolve(dest);
261    match &*dest {
262        Object::Array(a) => Some(a.clone()),
263        Object::Dict(d) => explicit_dest(doc, targets, d.get(b"D")?, depth + 1),
264        Object::String(s) => {
265            let inner = targets.named(doc, &s.bytes)?;
266            explicit_dest(doc, targets, &inner, depth + 1)
267        }
268        Object::Name(n) => {
269            let inner = targets.named(doc, n)?;
270            explicit_dest(doc, targets, &inner, depth + 1)
271        }
272        _ => None,
273    }
274}

276/// The explicit destination array `dest` stands for: itself, a dictionary whose `/D` is one, or a
277/// named destination (a name or string, ISO 32000-2 §12.3.2.4) looked up in the `/Names /Dests`
278/// tree, then the catalog `/Dests`. `None` when a name names nothing.
279pub(crate) fn resolve_dest(doc: &Document, dest: &Object) -> Option<Vec<Object>> {
280    // Names only: the page map is not needed to look one up.
281    let mut targets = Targets { pages: HashMap::new(), names: None };
282    explicit_dest(doc, &mut targets, dest, 0)
283}

285/// The page index an explicit destination array names.
286fn page_of_explicit(targets: &Targets, dest: &[Object]) -> Option<usize> {
287    match dest.first()? {
288        Object::Ref(page) => targets.pages.get(page).copied(),
289        // Some producers write a page number in a remote-style destination.
290        Object::Int(n) => usize::try_from(*n).ok().filter(|&p| p < targets.pages.len()),
291        _ => None,
292    }
293}

295/// The view an explicit destination array asks for: its type and the numbers after it.
296fn view_of_explicit(doc: &Document, dest: &[Object]) -> Option<(Vec<u8>, Vec<Option<f64>>)> {
297    let kind = dest.get(1).map(|k| doc.resolve(k)).and_then(|k| k.as_name().map(<[u8]>::to_vec))?;
298    // At most the four numbers of /FitR.
299    let numbers = dest.iter().skip(2).take(4).map(|n| doc.resolve(n).as_f64().filter(|v| v.is_finite())).collect();
300    Some((kind, numbers))
301}

303/// The explicit destination an item goes to: its `/Dest` when that names a page here, else the
304/// destination of its GoTo action `/A` (or of an action without a type). Another action's `/D` is
305/// not a place here: a remote go-to's names a page or destination of the other file (ISO 32000-2
306/// §12.6.4.3).
307fn item_dest(doc: &Document, targets: &mut Targets, d: &Dict) -> Option<Vec<Object>> {
308    if let Some(dest) = d.get(b"Dest").and_then(|dest| explicit_dest(doc, targets, dest, 0))
309        && page_of_explicit(targets, &dest).is_some()
310    {
311        return Some(dest);
312    }
313    let action = doc.resolve(d.get(b"A")?);
314    let action = action.as_dict().filter(|a| a.get(b"S").is_none_or(|s| doc.resolve(s).as_name() == Some(b"GoTo")))?;
315    explicit_dest(doc, targets, action.get(b"D")?, 0)
316}

318/// A bookmark's title as viewers show it: the decoded text, without NULs or surrounding space.
319fn title_of(doc: &Document, d: &Dict) -> String {
320    let title = d.get(b"Title").map(|t| doc.resolve(t));
321    let text = match title.as_deref() {
322        Some(Object::String(s)) => s.to_text(),
323        Some(Object::Name(n)) => String::from_utf8_lossy(n).into_owned(),
324        _ => return String::new(),
325    };
326    text.trim_matches('\0').trim().to_string()
327}

329fn describe(doc: &Document, targets: &mut Targets, obj: ObjRef, path: Vec<usize>) -> Bookmark {
330    let item = doc.get(obj);
331    let (title, open, dest) = match item.as_dict() {
332        Some(d) => (title_of(doc, d), d.int(b"Count").is_some_and(|c| c > 0), item_dest(doc, targets, d)),
333        None => (String::new(), false, None),
334    };
335    let page = dest.as_deref().and_then(|d| page_of_explicit(targets, d));
336    let view = dest.as_deref().filter(|_| page.is_some()).and_then(|d| view_of_explicit(doc, d));
337    let color = item.as_dict().and_then(|d| d.get(b"C").map(|c| doc.resolve(c))).and_then(|c| title_color(doc, &c));
338    let flags = item.as_dict().and_then(|d| d.get(b"F").and_then(|f| doc.resolve(f).as_int())).unwrap_or(0);
339    Bookmark { obj, path, title, open, page, view, color, italic: flags & 1 != 0, bold: flags & 2 != 0 }
340}

342/// A bookmark title's `/C` as a colour: three numbers, each kept within 0–1 (ISO 32000-2
343/// §12.3.3, Table 151: "An array of three numbers in the range 0.0 to 1.0"); a number past the
344/// range, however large, is kept at its edge. Black, the default, and anything else give `None`.
345fn title_color(doc: &Document, c: &Object) -> Option<[f32; 3]> {
346    let [r, g, b] = c.as_array()?.as_slice() else { return None };
347    let channel = |v: &Object| doc.resolve(v).as_f64().map(|v| v.clamp(0.0, 1.0) as f32);
348    let rgb = [channel(r)?, channel(g)?, channel(b)?];
349    (rgb != [0.0; 3]).then_some(rgb)
350}

352/// Lists up to `limit` bookmarks from position `offset`, entering no level deeper than `max_depth`.
353fn list(doc: &Document, max_depth: usize, offset: usize, limit: usize) -> BookmarkPage {
354    let mut bookmarks = Vec::new();
355    let Some(mut visit) = Visit::new(doc, max_depth) else { return BookmarkPage { bookmarks, next: None } };
356    for _ in 0..offset {
357        if visit.step().is_none() {
358            return BookmarkPage { bookmarks, next: None };
359        }
360    }
361    let mut targets = Targets::new(doc);
362    while bookmarks.len() < limit {
363        let Some(obj) = visit.step() else { break };
364        bookmarks.push(describe(doc, &mut targets, obj, visit.path()));
365    }
366    let next = if bookmarks.len() == limit && visit.step().is_some() { Some(offset.saturating_add(bookmarks.len())) } else { None };
367    BookmarkPage { bookmarks, next }
368}

388/// The object of the bookmark at `path`, and its parent object (the outline root for top-level).
389fn resolve_path(doc: &Document, path: &[usize]) -> Result<(ObjRef, ObjRef)> {
390    let root = outline_root(doc).ok_or_else(|| OutlineError::NoSuchBookmark(path.to_vec()))?;
391    if path.is_empty() {
392        return Err(OutlineError::NoSuchBookmark(Vec::new()));
393    }
394    let mut parent = root;
395    let mut seen = HashSet::from([root]);
396    for (depth, i) in path.iter().enumerate() {
397        let kids = children_of(doc, parent, &mut seen);
398        let r = *kids.get(*i).ok_or_else(|| OutlineError::NoSuchBookmark(path.to_vec()))?;
399        if depth + 1 == path.len() {
400            return Ok((r, parent));
401        }
402        parent = r;
403    }
404    // The loop returns on the last index.
405    Err(OutlineError::NoSuchBookmark(path.to_vec()))
406}

408/// The object that holds children at `parent_path` (`[]` = the outline root, created if needed).
409pub(crate) fn container(doc: &mut Document, parent_path: &[usize]) -> Result<ObjRef> {
410    if parent_path.is_empty() {
411        if let Some(r) = outline_root(doc) {
412            return Ok(r);
413        }
414        let mut o = Dict::new();
415        o.set(b"Type".to_vec(), Object::name("Outlines"));
416        let r = doc.add(Object::Dict(o));
417        let root = doc.root().ok_or(OrganizeError::NoPageTree)?;
418        doc.update_dict(root, |c| c.set(b"Outlines".to_vec(), Object::Ref(r)))?;
419        return Ok(r);
420    }
421    Ok(resolve_path(doc, parent_path)?.0)
422}

424/// Set `key` on the dictionary `r` (or remove it with `None`) only if it changes.
425fn put(doc: &mut Document, r: ObjRef, key: &[u8], value: Option<Object>) -> Result<()> {
426    let current = doc.get(r).as_dict().and_then(|d| d.get(key).cloned());
427    if current == value {
428        return Ok(());
429    }
430    doc.update_dict(r, |d| match value {
431        Some(v) => d.set(key.to_vec(), v),
432        None => {
433            d.remove(key);
434        }
435    })?;
436    Ok(())
437}

460/// Make `kids` the children of `parent`, in order: rewires `/Parent`, `/Prev`, `/Next`,
461/// `/First` and `/Last`.
462pub(crate) fn relink(doc: &mut Document, parent: ObjRef, kids: &[ObjRef]) -> Result<()> {
463    for (i, k) in kids.iter().enumerate() {
464        put(doc, *k, b"Parent", Some(Object::Ref(parent)))?;
465        put(doc, *k, b"Prev", i.checked_sub(1).map(|p| Object::Ref(kids[p])))?;
466        put(doc, *k, b"Next", kids.get(i + 1).map(|n| Object::Ref(*n)))?;
467    }
468    put(doc, parent, b"First", kids.first().map(|k| Object::Ref(*k)))?;
469    put(doc, parent, b"Last", kids.last().map(|k| Object::Ref(*k)))?;
470    Ok(())
471}

473/// A node whose `/Count` [`recount`] is working out.
474struct Counting {
475    node: ObjRef,
476    kids: Vec<ObjRef>,
477    /// The next child to count.
478    next: usize,
479    /// Items visible below the node, so far.
480    visible: i64,
481    is_root: bool,
482}

484/// Recompute every `/Count` (§12.3.3: open items count their visible descendants, closed items
485/// the negative of what opening them would show; the root counts all visible items).
486pub(crate) fn recount(doc: &mut Document) -> Result<()> {
487    let Some(root) = outline_root(doc) else { return Ok(()) };
488    let mut seen = HashSet::from([root]);
489    let kids = children_of(doc, root, &mut seen);
490    let mut stack = vec![Counting { node: root, kids, next: 0, visible: 0, is_root: true }];
491    // A node is counted once its children are, so the walk keeps its own stack instead of recursing.
492    while let Some(top) = stack.last_mut() {
493        if let Some(&kid) = top.kids.get(top.next) {
494            top.next += 1;
495            let kids = children_of(doc, kid, &mut seen);
496            stack.push(Counting { node: kid, kids, next: 0, visible: 0, is_root: false });
497            continue;
498        }
499        let Some(done) = stack.pop() else { break };
500        let count = if done.kids.is_empty() {
501            None
502        } else if done.is_root || doc.get(done.node).as_dict().and_then(|d| d.int(b"Count")).is_some_and(|c| c > 0) {
503            Some(done.visible)
504        } else {
505            Some(-done.visible)
506        };
507        put(doc, done.node, b"Count", count.map(Object::Int))?;
508        if let Some(parent) = stack.last_mut() {
509            let open = doc.get(done.node).as_dict().and_then(|d| d.int(b"Count")).is_some_and(|c| c > 0);
510            parent.visible = parent.visible.saturating_add(1 + if open { done.visible } else { 0 });
511        }
512    }
513    Ok(())
514}

516fn destination(doc: &Document, page: usize) -> Result<Object> {
517    destination_at(doc, page, None, None)
518}
519
520/// A destination on `page` (0-based) at the user-space point (`left`, `top`), either of which
521/// may be left out to keep the reader's position in that direction.
522fn destination_at(doc: &Document, page: usize, left: Option<f64>, top: Option<f64>) -> Result<Object> {
523    if ![left, top].iter().flatten().all(|v| v.is_finite()) {
524        return Err(OutlineError::BadPosition);
525    }
526    let pages = walk(doc)?;
527    let (p, _) = pages.get(page).ok_or(OrganizeError::NoSuchPage(page))?;
528    Ok(go_to_at(*p, left, top))
529}
530
531/// `/XYZ left top null`: the point (`left`, `top`) at the window's upper-left corner, at the
532/// reader's zoom; a coordinate that is left out keeps the reader's position. With neither,
533/// `/XYZ null null null` goes to the page, keeping the reader's zoom (what Acrobat writes for a
534/// new bookmark when the view has no specific position).
535fn go_to_at(page: ObjRef, left: Option<f64>, top: Option<f64>) -> Object {
536    let coordinate = |v: Option<f64>| v.map_or(Object::Null, Object::Real);
537    Object::Array(vec![Object::Ref(page), Object::name("XYZ"), coordinate(left), coordinate(top), Object::Null])
538}

540/// Add a bookmark titled `title` that goes to `page` (0-based), as child `index` of
541/// `parent_path` (`[]` = top level; an index past the end appends). Returns its path.
542pub fn add_bookmark(doc: &mut Document, parent_path: &[usize], index: usize, title: &str, page: usize) -> Result<Vec<usize>> {
543    add_bookmark_at(doc, parent_path, index, title, page, None, None)
544}
545
546/// [`add_bookmark`] going to the user-space point (`left`, `top`) of `page`, as `/XYZ left top null`
547/// (zoom kept). Either coordinate may be left out to keep the reader's position in that
548/// direction; one that is not finite is refused.
549pub fn add_bookmark_at(
550    doc: &mut Document,
551    parent_path: &[usize],
552    index: usize,
553    title: &str,
554    page: usize,
555    left: Option<f64>,
556    top: Option<f64>,
557) -> Result<Vec<usize>> {
558    if title.trim().is_empty() {
559        return Err(OutlineError::EmptyTitle);
560    }
561    let dest = destination_at(doc, page, left, top)?;
562    let parent = container(doc, parent_path)?;
563    let mut kids = children_of(doc, parent, &mut HashSet::from([parent]));
564    let mut d = Dict::new();
565    d.set(b"Title".to_vec(), Object::String(PdfString::text(title)));
566    d.set(b"Dest".to_vec(), dest);
567    let item = doc.add(Object::Dict(d));
568    let at = index.min(kids.len());
569    kids.insert(at, item);
570    relink(doc, parent, &kids)?;
571    show_children(doc, parent, parent_path)?;
572    recount(doc)?;
573    let mut path = parent_path.to_vec();
574    path.push(at);
575    Ok(path)
576}

578/// One bookmark of a generated tree (New Bookmarks from Structure, or from text styles).
579#[derive(Clone, Debug, PartialEq)]
580pub struct OutlineEntry {
581    /// 1 = directly under the new parent; deeper levels nest under the last shallower entry.
582    pub level: u8,
583    pub title: String,
584    pub page: usize,
585    /// The structure element it stands for (`/SE`).
586    pub element: Option<ObjRef>,
587    /// Where on the page it goes: this height (user space) at the top of the window, at the
588    /// reader's zoom. `None` goes to the page, keeping the reader's position.
589    pub top: Option<f64>,
590}
591
592/// Add a new top-level bookmark `parent_title` (first in the list) holding `entries` nested by
593/// level, as Acrobat does for bookmarks made from structure; everything starts expanded.
594/// Returns the parent's path.
595pub fn add_bookmark_tree(doc: &mut Document, parent_title: &str, entries: &[OutlineEntry]) -> Result<Vec<usize>> {
596    if entries.is_empty() {
597        return Err(OutlineError::NoEntries);
598    }
599    let pages = walk(doc)?;
600    let targets = entries
601        .iter()
602        .map(|e| pages.get(e.page).map(|p| p.0).ok_or(OrganizeError::NoSuchPage(e.page)))
603        .collect::<std::result::Result<Vec<_>, _>>()?;
604    let mut d = Dict::new();
605    d.set(b"Title".to_vec(), Object::String(PdfString::text(parent_title)));
606    let parent = doc.add(Object::Dict(d));
607    // (level, item, its children so far); the bottom is the new parent.
608    let mut stack: Vec<(u8, ObjRef, Vec<ObjRef>)> = vec![(0, parent, Vec::new())];
609    let mut done: Vec<(ObjRef, Vec<ObjRef>)> = Vec::new();
610    for (e, page) in entries.iter().zip(targets) {
611        let mut d = Dict::new();
612        d.set(b"Title".to_vec(), Object::String(PdfString::text(&e.title)));
613        d.set(b"Dest".to_vec(), go_to_at(page, None, e.top));
614        if let Some(se) = e.element {
615            d.set(b"SE".to_vec(), Object::Ref(se));
616        }
617        let item = doc.add(Object::Dict(d));
618        while stack.len() > 1 && stack.last().is_some_and(|s| s.0 >= e.level) {
619            done.extend(stack.pop().map(|(_, r, kids)| (r, kids)));
620        }
621        if let Some(top) = stack.last_mut() {
622            top.2.push(item);
623        }
624        stack.push((e.level, item, Vec::new()));
625    }
626    done.extend(stack.into_iter().map(|(_, r, kids)| (r, kids)));
627    for (r, kids) in &done {
628        relink(doc, *r, kids)?;
629        if !kids.is_empty() && *r != parent {
630            put(doc, *r, b"Count", Some(Object::Int(1)))?;
631        }
632    }
633    let root = container(doc, &[])?;
634    let mut top = children_of(doc, root, &mut HashSet::from([root]));
635    top.insert(0, parent);
636    relink(doc, root, &top)?;
637    put(doc, parent, b"Count", Some(Object::Int(1)))?;
638    recount(doc)?;
639    Ok(vec![0])
640}

642/// Change a bookmark's title.
643pub fn rename_bookmark(doc: &mut Document, path: &[usize], title: &str) -> Result<()> {
644    if title.trim().is_empty() {
645        return Err(OutlineError::EmptyTitle);
646    }
647    let (r, _) = resolve_path(doc, path)?;
648    put(doc, r, b"Title", Some(Object::String(PdfString::text(title))))
649}

651/// Set the colour of a bookmark's title (`/C`, ISO 32000-2 §12.3.3, Table 151: "An array of three
652/// numbers in the range 0.0 to 1.0, representing the components in the DeviceRGB colour space").
653/// Black, the default, and `None` remove `/C`; a component outside 0–1 is refused.
654pub fn set_bookmark_color(doc: &mut Document, path: &[usize], color: Option<[f64; 3]>) -> Result<()> {
655    if color.is_some_and(|c| !c.iter().all(|v| (0.0..=1.0).contains(v))) {
656        return Err(OutlineError::BadColor);
657    }
658    let (r, _) = resolve_path(doc, path)?;
659    let value = color.filter(|c| *c != [0.0; 3]).map(|c| Object::Array(c.iter().map(|v| Object::Real(*v)).collect()));
660    put(doc, r, b"C", value)
661}

663/// Make a bookmark's title bold or italic or not (`/F`, ISO 32000-2 §12.3.3, Table 152: bit 1
664/// italic, bit 2 bold); `None` keeps that one as it is, and so are the other bits. When no bit is
665/// left, `/F` is removed, its default being 0.
666pub fn set_bookmark_style(doc: &mut Document, path: &[usize], bold: Option<bool>, italic: Option<bool>) -> Result<()> {
667    let (r, _) = resolve_path(doc, path)?;
668    let old = doc.get(r).as_dict().and_then(|d| d.get(b"F").and_then(|f| doc.resolve(f).as_int())).unwrap_or(0);
669    let bit = |flags: i64, mask: i64, on: Option<bool>| match on {
670        Some(true) => flags | mask,
671        Some(false) => flags & !mask,
672        None => flags,
673    };
674    let flags = bit(bit(old, 1, italic), 2, bold);
675    put(doc, r, b"F", (flags != 0).then_some(Object::Int(flags)))
676}

678/// Point a bookmark at `page` (0-based), replacing its destination or GoTo action.
679pub fn set_bookmark_page(doc: &mut Document, path: &[usize], page: usize) -> Result<()> {
680    let dest = destination(doc, page)?;
681    let (r, _) = resolve_path(doc, path)?;
682    put(doc, r, b"Dest", Some(dest))?;
683    put(doc, r, b"A", None) // `/Dest` and `/A` must not both be present (§12.3.3)
684}

686/// Remove a bookmark and everything under it. The removed items are unlinked (a full save drops
687/// them; an incremental save leaves the old objects unreferenced).
688pub fn delete_bookmark(doc: &mut Document, path: &[usize]) -> Result<()> {
689    let (r, parent) = resolve_path(doc, path)?;
690    let mut kids = children_of(doc, parent, &mut HashSet::from([parent]));
691    kids.retain(|k| *k != r);
692    relink(doc, parent, &kids)?;
693    recount(doc)
694}

696/// Move the bookmark at `from` to child `index` of `to_parent` (counted after removing it from
697/// its old place; past the end appends). Returns its new path.
698pub fn move_bookmark(doc: &mut Document, from: &[usize], to_parent: &[usize], index: usize) -> Result<Vec<usize>> {
699    if to_parent.starts_with(from) {
700        return Err(OutlineError::IntoItself);
701    }
702    let (r, old_parent) = resolve_path(doc, from)?;
703    // Validate the target before touching the old parent, so an error leaves the outline intact.
704    let parent = container(doc, to_parent)?;
705    let mut old = children_of(doc, old_parent, &mut HashSet::from([old_parent]));
706    old.retain(|k| *k != r);
707    relink(doc, old_parent, &old)?;
708    // Removing `from` shifts later siblings: adjust a target path that runs through them.
709    let mut to_parent = to_parent.to_vec();
710    let depth = from.len() - 1;
711    if to_parent.len() > depth && to_parent[..depth] == from[..depth] && to_parent[depth] > from[depth] {
712        to_parent[depth] -= 1;
713    }
714    let mut kids = children_of(doc, parent, &mut HashSet::from([parent]));
715    let at = index.min(kids.len());
716    kids.insert(at, r);
717    relink(doc, parent, &kids)?;
718    if parent != old_parent {
719        show_children(doc, parent, &to_parent)?;
720    }
721    recount(doc)?;
722    to_parent.push(at);
723    Ok(to_parent)
724}

726/// Open `parent` so a bookmark just added or moved into it shows (the root is always open).
727/// Only the sign of `/Count` matters here: [`recount`] sets its value.
728pub(crate) fn show_children(doc: &mut Document, parent: ObjRef, parent_path: &[usize]) -> Result<()> {
729    if !parent_path.is_empty() && doc.get(parent).as_dict().and_then(|d| d.int(b"Count")).is_none_or(|c| c <= 0) {
730        put(doc, parent, b"Count", Some(Object::Int(1)))?;
731    }
732    Ok(())
733}

735/// Expand or collapse a bookmark that has children.
736pub fn set_bookmark_open(doc: &mut Document, path: &[usize], open: bool) -> Result<()> {
737    let (r, _) = resolve_path(doc, path)?;
738    let count = doc.get(r).as_dict().and_then(|d| d.int(b"Count")).unwrap_or(0);
739    if count != 0 && (count > 0) != open {
740        put(doc, r, b"Count", Some(Object::Int(-count)))?;
741        recount(doc)?;
742    }
743    Ok(())
