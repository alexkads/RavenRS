1//! Document-level file attachments (Acrobat: Edit ▸ Attach file; the Attachments panel): the
2//! `/Names /EmbeddedFiles` name tree of file specifications, each with its embedded stream.
3//!
4//! A FileAttachment annotation (a paper clip on a page) is a comment and lives in the annot
5//! crate; this is the document level only.

12/// The keys of the document-level attachments, in name-tree order.
13pub fn attachment_keys(doc: &Document) -> Vec<String> {
14    entries(doc).into_iter().map(|(k, _)| PdfString::literal(k).to_text()).collect()
15}

17/// Attach `bytes` to the document under `name` (a file name; any folder part is dropped). The
18/// name tree's key is the name, made unique with " (2)", " (3)"… when it is already taken, and
19/// is returned so the caller can find the attachment again. `date` is a PDF date string for the
20/// embedded file's `/ModDate`; `mime` its `/Subtype` when known.
21pub fn add_attachment(
22    doc: &mut Document,
23    name: &str,
24    bytes: &[u8],
25    description: Option<&str>,
26    mime: Option<&str>,
27    date: Option<&str>,
28) -> Result<String, OrganizeError> {
29    let name = file_name(name);
30    if name.is_empty() {
31        return Err(OrganizeError::Invalid("the attachment needs a file name".into()));
32    }
33    let mut all = entries(doc);
34    let taken: Vec<String> = all.iter().map(|(k, _)| PdfString::literal(k.clone()).to_text()).collect();
35    let mut key = name.clone();
36    let mut n = 2;
37    while taken.contains(&key) {
38        key = format!("{name} ({n})");
39        n += 1;
40    }
41
42    let mut params = Dict::new();
43    params.set(b"Size".to_vec(), Object::Int(bytes.len() as i64));
44    if let Some(d) = date {
45        params.set(b"ModDate".to_vec(), Object::String(PdfString::literal(d.as_bytes().to_vec())));
46    }
47    let mut ef = Dict::new();
48    ef.set(b"Type".to_vec(), Object::name("EmbeddedFile"));
49    if let Some(m) = mime.map(str::trim).filter(|m| !m.is_empty()) {
50        ef.set(b"Subtype".to_vec(), Object::Name(m.as_bytes().to_vec()));
51    }
52    ef.set(b"Params".to_vec(), Object::Dict(params));
53    let stream = doc.add(Object::Stream(Stream::flate(ef, bytes)));
54
55    let mut spec = Dict::new();
56    spec.set(b"Type".to_vec(), Object::name("Filespec"));
57    spec.set(b"F".to_vec(), Object::String(PdfString::text(&key)));
58    spec.set(b"UF".to_vec(), Object::String(PdfString::text(&key)));
59    if let Some(d) = description.map(str::trim).filter(|d| !d.is_empty()) {
60        spec.set(b"Desc".to_vec(), Object::String(PdfString::text(d)));
61    }
62    let mut files = Dict::new();
63    files.set(b"F".to_vec(), Object::Ref(stream));
64    files.set(b"UF".to_vec(), Object::Ref(stream));
65    spec.set(b"EF".to_vec(), Object::Dict(files));
66    let spec = doc.add(spec);
67
68    all.push((PdfString::text(&key).bytes, Object::Ref(spec)));
69    set_attachments(doc, all)?;
70    Ok(key)
71}

73/// Remove the document-level attachment whose name-tree key is `key`. The embedded stream is
74/// left for the writer's garbage collection, as with every other removal.
75pub fn delete_attachment(doc: &mut Document, key: &str) -> Result<(), OrganizeError> {
76    let mut all = entries(doc);
77    let before = all.len();
78    all.retain(|(k, _)| PdfString::literal(k.clone()).to_text() != key);
79    if all.len() == before {
80        return Err(OrganizeError::Invalid(format!("there is no attachment named {key}")));
81    }
82    if all.is_empty() { clear_embedded_files(doc) } else { set_attachments(doc, all) }
83}

85/// Set or clear the description of a document-level attachment without replacing its file
86/// specification or embedded stream.
87pub fn set_attachment_description(doc: &mut Document, key: &str, description: Option<&str>) -> Result<(), OrganizeError> {
88    let mut all = entries(doc);
89    let Some((_, value)) = all.iter_mut().find(|(name, _)| PdfString::literal(name.clone()).to_text() == key) else {
90        return Err(OrganizeError::Invalid(format!("there is no attachment named {key}")));
91    };
92    let description = description.map(str::trim).filter(|s| !s.is_empty()).map(PdfString::text).map(Object::String);
93    match value {
94        Object::Ref(reference) => {
95            doc.update_dict(*reference, |spec| match description {
96                Some(description) => spec.set(b"Desc".to_vec(), description),
97                None => {
98                    spec.remove(b"Desc");
99                }
100            })?;
101        }
102        Object::Dict(spec) => {
103            match description {
104                Some(description) => spec.set(b"Desc".to_vec(), description),
105                None => {
106                    spec.remove(b"Desc");
107                }
108            }
109            set_attachments_exact(doc, all)?;
110        }
111        _ => return Err(OrganizeError::Invalid(format!("attachment {key} has an invalid file specification"))),
112    }
113    Ok(())
114}

116/// The `/EmbeddedFiles` entries (key, file specification), in tree order.
117fn entries(doc: &Document) -> Vec<(Vec<u8>, Object)> {
118    let Some(tree) = doc
119        .root()
120        .and_then(|r| doc.get(r).as_dict().cloned())
121        .and_then(|c| c.get(b"Names").map(|n| doc.resolve(n)))
122        .and_then(|n| n.as_dict().and_then(|d| d.get(b"EmbeddedFiles").cloned()))
123    else {
124        return Vec::new();
125    };
126    let mut out = Vec::new();
127    name_tree_entries(doc, &tree, 0, &mut std::collections::HashSet::new(), &mut out);
128    out
129}

131/// Drop `/EmbeddedFiles` from the catalog's `/Names` (and `/Names` itself when nothing is left).
132fn clear_embedded_files(doc: &mut Document) -> Result<(), OrganizeError> {
133    let root = doc.root().ok_or(OrganizeError::NoPageTree)?;
134    let names = doc.get(root).as_dict().and_then(|c| c.get(b"Names").cloned());
135    match names {
136        Some(Object::Ref(r)) => {
137            let mut empty = false;
138            doc.update_dict(r, |n| {
139                n.remove(b"EmbeddedFiles");
140                empty = n.is_empty();
141            })?;
142            if empty {
143                doc.update_dict(root, |c| {
144                    c.remove(b"Names");
145                })?;
146            }
147        }
148        Some(Object::Dict(mut n)) => {
149            n.remove(b"EmbeddedFiles");
150            doc.update_dict(root, |c| {
151                if n.is_empty() {
152                    c.remove(b"Names");
153                } else {
154                    c.set(b"Names".to_vec(), Object::Dict(n));
155                }
156            })?;
157        }
158        _ => {}
159    }
160    Ok(())
161}

163/// The last path component of `name`, trimmed: a dragged-in path or a folder part never becomes
164/// the attachment's name.
165fn file_name(name: &str) -> String {
166    name.rsplit(['/', '\\']).next().unwrap_or(name).trim().to_string()
167}

169/// A media type from the file's extension, for the embedded file's `/Subtype`; `None` when it
170/// is not one of the common kinds (the attachment is still fine without it).
171pub fn mime_for_name(name: &str) -> Option<&'static str> {
172    let ext = name.rsplit_once('.').map(|(_, e)| e.to_ascii_lowercase())?;
173    Some(match ext.as_str() {
174        "pdf" => "application/pdf",
175        "txt" | "log" | "md" => "text/plain",
176        "csv" => "text/csv",
177        "htm" | "html" => "text/html",
178        "xml" => "text/xml",
179        "json" => "application/json",
180        "zip" => "application/zip",
181        "png" => "image/png",
182        "jpg" | "jpeg" => "image/jpeg",
183        "gif" => "image/gif",
184        "tif" | "tiff" => "image/tiff",
185        "svg" => "image/svg+xml",
186        "doc" => "application/msword",
187        "docx" => "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
188        "xls" => "application/vnd.ms-excel",
189        "xlsx" => "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
190        "ppt" => "application/vnd.ms-powerpoint",
191        "pptx" => "application/vnd.openxmlformats-officedocument.presentationml.presentation",
192        "rtf" => "application/rtf",
193        "eml" => "message/rfc822",
194        "msg" => "application/vnd.ms-outlook",
195        _ => return None,
196    })
