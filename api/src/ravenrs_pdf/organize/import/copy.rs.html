5/// Keys that must not be followed when copying: they point back up or across the source
6/// document, and following them would drag in unrelated objects.
7pub(super) const SKIP_ON_PAGE: &[&[u8]] = &[b"Parent", b"B", b"StructParents", b"Thumb", b"PieceInfo"];
8
9const SKIP_ON_ANNOT: &[&[u8]] = &[b"StructParent", b"P"];

11/// How deep a copy follows references and nested arrays and dictionaries, counted together.
12/// Real documents stay well under this (a page's resources, fonts and descriptors, nested
13/// form XObjects and field trees are a few levels each); a chain of thousands of indirect
14/// objects, each shallow enough for the parser, would otherwise exhaust the stack (#1800).
15pub(super) const MAX_COPY_DEPTH: usize = 256;

48pub(crate) struct Copier<'a> {
49    pub(super) src: &'a Document,
50    /// Current recursion depth, bounded by [`MAX_COPY_DEPTH`].
51    depth: usize,
52    pub(crate) map: HashMap<ObjRef, ObjRef>,
53    /// Source page → destination page, for rewriting `/P` and destinations.
54    pub(super) pages: HashMap<ObjRef, ObjRef>,
55    /// Destination objects that are annotation dictionaries (fixed up after copying).
56    pub(super) annots: Vec<ObjRef>,
57    /// Top-level form fields reached through widgets.
58    pub(super) fields: Vec<ObjRef>,
59    /// Optional content groups copied: (source, destination).
60    pub(crate) ocgs: Vec<(ObjRef, ObjRef)>,
61    /// When only some pages are copied: the annotations on them and the fields above each, the
62    /// only kids a copied field keeps (see `fields_on`). `None` keeps every kid. Shared, so the
63    /// copiers made for a repeated page or an action don't each copy it.
64    pub(super) kept: Option<Rc<HashSet<ObjRef>>>,
65}

67impl<'a> Copier<'a> {
68    pub(crate) fn new(src: &'a Document, map: HashMap<ObjRef, ObjRef>, pages: HashMap<ObjRef, ObjRef>) -> Self {
69        Copier { src, depth: 0, map, pages, annots: Vec::new(), fields: Vec::new(), ocgs: Vec::new(), kept: None }
70    }
71
72    /// Copy an indirect object (once), returning its destination reference.
73    pub(super) fn copy_ref(&mut self, dst: &mut Document, r: ObjRef) -> Result<ObjRef, OrganizeError> {
74        if let Some(d) = self.map.get(&r) {
75            return Ok(*d);
76        }
77        // Reserve the number first so cycles (field ↔ widget) terminate.
78        let new = dst.add(Object::Null);
79        self.map.insert(r, new);
80        let obj = self.src.get(r);
81        if obj.as_dict().is_some_and(|d| d.name(b"Type") == Some(b"OCG")) {
82            self.ocgs.push((r, new));
83        }
84        let copied = self.copy_value(dst, &obj, Some(new))?;
85        dst.set(new, copied);
86        Ok(new)
87    }
88
89    /// One level deeper into the object graph, or the refusal when there are too many.
90    fn descend(&mut self) -> Result<(), OrganizeError> {
91        if self.depth >= MAX_COPY_DEPTH {
92            return Err(OrganizeError::Invalid(format!("the pages refer to objects nested more than {MAX_COPY_DEPTH} levels deep")));
93        }
94        self.depth += 1;
95        Ok(())
96    }
97
98    pub(crate) fn copy_value(&mut self, dst: &mut Document, o: &Object, this: Option<ObjRef>) -> Result<Object, OrganizeError> {
99        if !matches!(o, Object::Ref(_) | Object::Array(_) | Object::Dict(_) | Object::Stream(_)) {
100            return Ok(o.clone());
101        }
102        if self.acts_on_none(o) {
103            return Ok(Object::Null);
104        }
105        self.descend()?;
106        let copied = match o {
107            Object::Ref(r) => self.copy_ref(dst, *r).map(Object::Ref),
108            Object::Array(a) => a.iter().map(|x| self.copy_value(dst, x, None)).collect::<Result<Vec<_>, _>>().map(Object::Array),
109            Object::Dict(d) => self.copy_dict(dst, d, this).map(Object::Dict),
110            // Opened without the password its embedded files need, the source holds their data
111            // encrypted: copied into another document, it would read as the file.
112            Object::Stream(s) if self.src.reads_as_stored(s) => Err(OrganizeError::Invalid(
113                "an attached file is encrypted and the document was opened without its password: open it with the password to copy it".into(),
114            )),
115            Object::Stream(s) => self.copy_dict(dst, &s.dict, None).map(|dict| {
116                let mut s = s.clone();
117                s.dict = dict;
118                Object::Stream(s)
119            }),
120            other => Ok(other.clone()),
121        };
122        self.depth = self.depth.saturating_sub(1);
123        copied
124    }

126    fn copy_dict(&mut self, dst: &mut Document, d: &Dict, this: Option<ObjRef>) -> Result<Dict, OrganizeError> {
127        let is_annot = d.name(b"Type") == Some(b"Annot") || (d.contains(b"Subtype") && d.contains(b"Rect") && !d.contains(b"Type"));
128        let is_field = d.contains(b"FT") || (d.contains(b"T") && d.contains(b"Kids"));
129        // A form action lists the fields it submits, resets or hides.
130        let lists_fields = matches!(d.name(b"S"), Some(b"SubmitForm" | b"ResetForm" | b"Hide"));
131        let mut out = Dict::new();
132        for (k, v) in d.iter() {
133            if is_annot && SKIP_ON_ANNOT.contains(&k.as_slice()) {
134                continue;
135            }
136            // Destinations are rewritten afterwards (they may target pages copied later).
137            if is_annot && (k.as_slice() == b"Dest" || k.as_slice() == b"A") {
138                out.set(k.clone(), v.clone());
139                continue;
140            }
141            if is_field
142                && k.as_slice() == b"Kids"
143                && let Some(kept) = &self.kept
144            {
145                let kids: Vec<Object> = self
146                    .src
147                    .resolve(v)
148                    .as_array()
149                    .into_iter()
150                    .flatten()
151                    .filter(|kid| kid.as_ref().is_none_or(|r| kept.contains(&r)))
152                    .cloned()
153                    .collect();
154                out.set(k.clone(), self.copy_value(dst, &Object::Array(kids), None)?);
155                continue;
156            }
157            // Of some of the pages, an action lists only the fields on them: one left behind would
158            // come along with its value. Fields named by text stay.
159            if lists_fields
160                && matches!(k.as_slice(), b"Fields" | b"T")
161                && let Some(kept) = &self.kept
162            {
163                let listed = self.src.resolve(v);
164                let entries = listed.as_array().cloned().unwrap_or_else(|| vec![v.clone()]);
165                let entries: Vec<Object> = entries.into_iter().filter(|e| e.as_ref().is_none_or(|r| kept.contains(&r))).collect();
166                out.set(k.clone(), self.copy_value(dst, &Object::Array(entries), None)?);
167                continue;
168            }
169            out.set(k.clone(), self.copy_value(dst, v, None)?);
170        }
171        if let Some(this) = this {
172            if is_annot {
173                self.annots.push(this);
174            }
175            // A field (or widget) without a parent is a top-level field.
176            if is_field && !d.contains(b"Parent") {
177                self.fields.push(this);
178            }
179        }
180        Ok(out)
181    }

183    /// Whether `o` is a reset or submit action that names only fields left behind (see `kept`).
184    /// Its copy would list none, and readers (ours too) may take an empty list for every field,
185    /// which is what an omitted one means (ISO 32000-2 §12.7.6). So it is not copied, nor are the
186    /// actions after it, as with a link to a page left behind. A list of fields to leave out
187    /// (Include/Exclude flag set) loses nothing.
188    fn acts_on_none(&self, o: &Object) -> bool {
189        let Some(kept) = &self.kept else { return false };
190        let held;
191        let action = match o {
192            // One copied already is not such an action, and needn't be fetched again.
193            Object::Ref(r) if !self.map.contains_key(r) => {
194                held = self.src.resolve(o);
195                held.as_dict()
196            }
197            Object::Dict(d) => Some(d),
198            _ => None,
199        };
200        let Some(a) = action.filter(|a| matches!(a.name(b"S"), Some(b"ResetForm" | b"SubmitForm"))) else { return false };
201        if a.get(b"Flags").and_then(|f| self.src.resolve(f).as_int()).is_some_and(|f| f & 1 != 0) {
202            return false;
203        }
204        let Some(v) = a.get(b"Fields") else { return false };
205        let listed = self.src.resolve(v);
206        // As `copy_dict` reads the list: a lone entry stands for a list of one.
207        let entries = listed.as_array().map_or(std::slice::from_ref(v), Vec::as_slice);
208        !entries.is_empty() && entries.iter().all(|e| e.as_ref().is_some_and(|r| !kept.contains(&r)))
209    }

211    /// Map a source destination (explicit array or reference to one) to the destination
212    /// document; `None` if it targets a page that was not copied.
213    pub(super) fn map_dest(&self, dest: &Object) -> Option<Object> {
214        self.map_dest_depth(dest, 0)
215    }
216
217    fn map_dest_depth(&self, dest: &Object, depth: u8) -> Option<Object> {
218        if depth > 4 {
219            return None;
220        }
221        match dest {
222            Object::Name(n) => self.map_dest_depth(&named_dest(self.src, n)?, depth + 1),
223            Object::String(s) => self.map_dest_depth(&named_dest(self.src, &s.bytes)?, depth + 1),
224            Object::Ref(_) => self.map_dest_depth(&self.src.resolve(dest), depth + 1),
225            Object::Dict(d) => self.map_dest_depth(d.get(b"D")?, depth + 1),
226            Object::Array(a) => {
227                let page = a.first()?.as_ref()?;
228                let new_page = *self.pages.get(&page)?;
229                let mut a = a.clone();
230                a[0] = Object::Ref(new_page);
231                Some(Object::Array(a))
232            }
233            _ => None,
234        }
235    }

237    /// Rewrite `/P`, `/Dest` and GoTo actions on copied annotations.
238    pub(crate) fn fix_annotations(&mut self, dst: &mut Document, annot_pages: &HashMap<ObjRef, ObjRef>) -> Result<(), OrganizeError> {
239        for &a in &std::mem::take(&mut self.annots) {
240            let src_dict = dst.get(a).as_dict().cloned();
241            let Some(d) = src_dict else { continue };
242            let mut d = d;
243            if let Some(p) = annot_pages.get(&a) {
244                d.set(b"P".to_vec(), Object::Ref(*p));
245            }
246            if let Some(dest) = d.get(b"Dest").cloned() {
247                let resolved = self.src.resolve(&dest);
248                match self.map_dest(&resolved) {
249                    Some(nd) => d.set(b"Dest".to_vec(), nd),
250                    None => {
251                        d.remove(b"Dest");
252                    }
253                }
254            }
255            if let Some(action) = d.get(b"A").cloned() {
256                let action = self.src.resolve(&action);
257                if let Some(ad) = action.as_dict() {
258                    if ad.name(b"S") == Some(b"GoTo") {
259                        let target = ad.get(b"D").map(|x| self.src.resolve(x));
260                        match target.and_then(|t| self.map_dest(&t)) {
261                            Some(nd) => {
262                                let mut na = Dict::new();
263                                na.set(b"S".to_vec(), Object::name("GoTo"));
264                                na.set(b"D".to_vec(), nd);
265                                d.set(b"A".to_vec(), Object::Dict(na));
266                            }
267                            None => {
268                                d.remove(b"A");
269                            }
270                        }
271                    } else if self.acts_on_none(&action) {
272                        d.remove(b"A");
273                    } else {
274                        // Other actions (URI, Launch, JavaScript…) carry no page references. The
275                        // copier lends its map out and takes it back with the new copies in it,
276                        // so actions sharing an object share its one copy.
277                        let mut c = Copier { kept: self.kept.clone(), ..Copier::new(self.src, std::mem::take(&mut self.map), HashMap::new()) };
278                        let copied = c.copy_value(dst, &Object::Dict(ad.clone()), None);
279                        self.map = c.map;
280                        d.set(b"A".to_vec(), copied?);
281                    }
282                }
283            }
284            dst.set(a, Object::Dict(d));
285        }
286        Ok(())
287    }
288}

290/// Copy `o` from `src` into `dst`, reusing the copies listed in `map` (source → destination), so
291/// an object the copied pages already brought along (an ICC profile) is not stored twice.
292pub(crate) fn copy_object(dst: &mut Document, src: &Document, o: &Object, map: HashMap<ObjRef, ObjRef>) -> Result<Object, OrganizeError> {
293    Copier::new(src, map, HashMap::new()).copy_value(dst, o, None)
