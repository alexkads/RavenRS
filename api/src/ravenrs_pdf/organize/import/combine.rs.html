4/// A new document containing copies of `pages` from `src` (Extract Pages / Split).
5/// Document information (title, author…) is carried over, and so is the print standard the
6/// source declares (PDF/X: its output intents and identification, #263), and the bookmarks that
7/// go to the copied pages.
8pub fn extract_pages(src: &Document, pages: &[usize]) -> Result<Document, OrganizeError> {
9    let mut out = Document::new_empty();
10    let mut labels = Vec::new();
11    let mut has_labels = false;
12    crate::organize::labels::collect_page_labels(src, pages, 0, &mut labels, &mut has_labels)?;
13    // One source: nothing to deduplicate. Its fields keep their names (a source with two
14    // same-named field roots stays one field, as it was), so Extract and Split never rename.
15    let imported = import_pages_mapped(&mut out, src, pages, 0, FieldConflicts::Merge)?;
16    crate::organize::labels::finish_page_labels(&mut out, &labels, has_labels)?;
17    // The bookmarks of these pages come along, mapped to the copies (#4020).
18    let outline = kept_bookmarks(source_outline(src, imported.page_map));
19    if !outline.is_empty() {
20        let root = crate::organize::outline::container(&mut out, &[])?;
21        if let Some((first, last, _)) = write_outline_level(&mut out, root, &outline)? {
22            out.update_dict(root, |d| {
23                d.set(b"First".to_vec(), Object::Ref(first));
24                d.set(b"Last".to_vec(), Object::Ref(last));
25            })?;
26            let top = crate::organize::outline::children(&out, root);
27            crate::organize::outline::relink_and_recount(&mut out, root, &top)?;
28        }
29    }
30    for key in crate::organize::INFO_KEYS {
31        if let Some(v) = crate::organize::info(src, key) {
32            crate::organize::set_info(&mut out, key, &v)?;
33        }
34    }
35    // After the title: the XMP packet repeats it.
36    crate::organize::pdfx::carry(&mut out, Some(src), &crate::organize::pdfx::PrintStandard::of(src), imported.objects)?;
37    // The pages share their source's resource dictionary, so the part would otherwise carry every
38    // XObject the source lists — images and all (#204). Keep only what these pages draw.
39    crate::organize::prune::prune_unused_xobjects(&mut out)?;
40    Ok(out)
41}

43/// How to split a document.
44#[derive(Clone, Debug, PartialEq)]
45pub enum SplitBy {
46    /// Every `n` pages.
47    PageCount(usize),
48    /// Before each of these 0-based page indices (e.g. `[3, 7]` → 0–2, 3–6, 7–end).
49    Before(Vec<usize>),
50}
51
52/// The page ranges (0-based, end-exclusive) a split produces.
53pub fn split_ranges(page_count: usize, by: &SplitBy) -> Vec<std::ops::Range<usize>> {
54    let mut cuts: Vec<usize> = match by {
55        SplitBy::PageCount(n) => (1..).map(|i| i * (*n).max(1)).take_while(|c| *c < page_count).collect(),
56        SplitBy::Before(v) => v.iter().copied().filter(|c| *c > 0 && *c < page_count).collect(),
57    };
58    cuts.sort_unstable();
59    cuts.dedup();
60    let mut ranges = Vec::new();
61    let mut start = 0;
62    for c in cuts.into_iter().chain(std::iter::once(page_count)) {
63        if c > start {
64            ranges.push(start..c);
65            start = c;
66        }
67    }
68    ranges
69}

71/// Split a document into several new documents.
72pub fn split(src: &Document, by: &SplitBy) -> Result<Vec<Document>, OrganizeError> {
73    let n = crate::organize::page_count(src)?;
74    split_ranges(n, by).into_iter().map(|r| extract_pages(src, &r.collect::<Vec<_>>())).collect()
75}

86/// Combine whole documents, in order, into a new document. Each source gets a top-level
87/// bookmark (its `title`) pointing to its first page, like Acrobat's Combine Files.
88/// The source's own bookmarks are nested (collapsed) under its entry, and its document-level
89/// attachments are carried over. When every source declares the same print standard (PDF/X
90/// with the same output intent), so does the result.
91pub fn combine(sources: &[(&str, &Document)]) -> Result<Document, OrganizeError> {
92    combine_with(sources, FieldConflicts::Rename).map(|o| o.doc)
93}

95/// [`combine`] reporting what the conflict resolution renamed.
96pub fn combine_with(sources: &[(&str, &Document)], conflicts: FieldConflicts) -> Result<CombineOutcome, OrganizeError> {
97    let all: Vec<(&str, &Document, Option<&[usize]>)> = sources.iter().map(|(t, d)| (*t, *d, None)).collect();
98    combine_selected_with(&all, conflicts)
99}

101/// Combine Files with chosen pages: each source contributes `pages` (0-based, in that order;
102/// `None` for all of them). A source whose pages are not all used brings only the bookmarks that
103/// lead to its copied pages (and those above them), and none of its attachments.
104pub fn combine_selected(sources: &[(&str, &Document, Option<&[usize]>)]) -> Result<Document, OrganizeError> {
105    combine_selected_with(sources, FieldConflicts::Rename).map(|o| o.doc)
106}

108/// [`combine_selected`] reporting what the conflict resolution renamed.
109pub fn combine_selected_with(sources: &[(&str, &Document, Option<&[usize]>)], conflicts: FieldConflicts) -> Result<CombineOutcome, OrganizeError> {
110    let mut combiner = Combiner::with_field_conflicts(conflicts);
111    for (title, src, chosen) in sources {
112        combiner.add(title, src, *chosen)?;
113    }
114    combiner.finish_with_renames()
115}

117/// Combine Files one source at a time, with the same result as [`combine_selected`]. [`Combiner::add`]
118/// copies a source's pages into the result and keeps only what the rest of the combine needs: its
119/// bookmarks (read and mapped to the copied pages), its attachments (copied), and its document only
120/// when it declares output intents. So each source can be dropped as soon as it is added, and a long
121/// list holds one source at a time rather than all of them.
122pub struct Combiner {
123    out: Document,
124    marks: Vec<Mark>,
125    attachments: Vec<(Vec<u8>, Object)>,
126    labels: Vec<crate::organize::LabelRange>,
127    has_labels: bool,
128    /// The first source's print standard, and what its pages brought along.
129    standard: Option<Standard>,
130    /// Whether every source so far declares the first source's print standard.
131    agreed: bool,
132    /// How colliding field names are resolved as each source is added.
133    field_conflicts: FieldConflicts,
134    /// What adding sources renamed (old → new fully-qualified field name), in order.
135    renamed_fields: Vec<(String, String)>,
136}

138/// The first source's print standard (see [`Combiner`]).
139struct Standard {
140    declared: crate::organize::pdfx::PrintStandard,
141    /// The source objects the copied pages brought along, and their copies.
142    objects: HashMap<ObjRef, ObjRef>,
143    /// The source itself, kept only for its output intents, which are copied from it.
144    source: Option<Document>,
145}

147impl Default for Combiner {
148    fn default() -> Self {
149        Self::new()

153impl Combiner {
154    /// A combine with no sources yet, renaming colliding field names as sources are added.
155    pub fn new() -> Self {
156        Self::with_field_conflicts(FieldConflicts::Rename)
157    }
158
159    /// [`Combiner::new`] resolving colliding field names as `conflicts` says.
160    pub fn with_field_conflicts(field_conflicts: FieldConflicts) -> Self {
161        Self {
162            out: Document::new_empty(),
163            marks: Vec::new(),
164            attachments: Vec::new(),
165            labels: Vec::new(),
166            has_labels: false,
167            standard: None,
168            agreed: true,
169            field_conflicts,
170            renamed_fields: Vec::new(),
171        }
172    }
173
174    /// What adding sources has renamed so far (old → new fully-qualified field name).
175    pub fn renamed_fields(&self) -> &[(String, String)] {
176        &self.renamed_fields
177    }
178
179    /// Add `src`'s `pages` (0-based, in that order; `None` for all of them) after the pages added so far,
180    /// under a bookmark titled `title`. `src` need not outlive the call.
181    pub fn add(&mut self, title: &str, src: &Document, pages: Option<&[usize]>) -> Result<(), OrganizeError> {
182        let n = crate::organize::page_count(src)?;
183        let pages: Vec<usize> = match pages {
184            Some(p) => {
185                if let Some(bad) = p.iter().find(|i| **i >= n) {
186                    return Err(OrganizeError::NoSuchPage(*bad));
187                }
188                p.to_vec()
189            }
190            None => (0..n).collect(),
191        };
192        let at = crate::organize::page_count(&self.out)?;
193        let mut labels = self.labels.clone();
194        let mut has_labels = self.has_labels;
195        crate::organize::labels::collect_page_labels(src, &pages, at, &mut labels, &mut has_labels)?;
196        let imported = import_pages_mapped(&mut self.out, src, &pages, at, self.field_conflicts)?;
197        self.labels = labels;
198        self.has_labels = has_labels;
199        self.renamed_fields.extend(imported.renamed_fields);
200        if let Some(first) = imported.pages.first() {
201            let outline = source_outline(src, imported.page_map);
202            self.marks.push(Mark { title: title.to_string(), page: *first, outline });
203        }
204        // Attachments belong to the whole file: they come along with all of its pages, not with a
205        // selection of them.
206        if pages.iter().collect::<HashSet<_>>().len() == n {
207            collect_attachments(&mut self.out, src, &mut self.attachments)?;
208        }
209        let declared = crate::organize::pdfx::PrintStandard::of(src);
210        match self.standard.as_ref().map(|first| first.declared.same_as(&declared)) {
211            Some(same) => self.agreed &= same,
212            None => {
213                let source = declared.declares_output_intents().then(|| src.clone());
214                self.standard = Some(Standard { declared, objects: imported.objects, source });
215            }
216        }
217        Ok(())
218    }
219
220    /// The combined document: the pages as added, the print standard of the first source when every
221    /// source declares the same one, shared resources stored once, then the bookmarks and attachments.
222    pub fn finish(self) -> Result<Document, OrganizeError> {
223        self.finish_with_renames().map(|o| o.doc)
224    }
225
226    /// [`Combiner::finish`] reporting what the combine renamed.
227    pub fn finish_with_renames(self) -> Result<CombineOutcome, OrganizeError> {
228        let Combiner { mut out, marks, attachments, labels, has_labels, standard, agreed, renamed_fields, .. } = self;
229        if let Some(first) = standard.filter(|_| agreed) {
230            crate::organize::pdfx::carry(&mut out, first.source.as_ref(), &first.declared, first.objects)?;
231        }
232        // Sources often share fonts, images and profiles (or are the same file): store them once.
233        let all: Vec<ObjRef> = out.object_numbers().into_iter().map(|n| ObjRef::new(n, out.generation(n))).collect();
234        crate::organize::dedupe::dedupe_resources(&mut out, &all, false);
235        crate::organize::labels::finish_page_labels(&mut out, &labels, has_labels)?;
236        add_outline(&mut out, &marks)?;
237        set_attachments(&mut out, attachments)?;
238        Ok(CombineOutcome { doc: out, renamed_fields })
239    }
240}

242/// One run of pages for [`combine_grouped`]: the file it comes from (`group`: runs with the same
243/// group are one file split around, and give the same document), its title, the document, and
244/// its pages (0-based, in that order; `None` for all of them).
245pub type Run<'a> = (usize, &'a str, &'a Document, Option<&'a [usize]>);
246
247/// Combine Files where a file's pages may be split into several runs with other files' pages
248/// between them. Each file (group) is copied once, all its pages together, then the pages are put
249/// in the runs' order: its links between its own pages and its form fields stay whole across its
250/// runs, its attachments come along when its runs use all of its pages, and it gets one top-level
251/// bookmark (the title of its first run) at its first page, its own bookmarks nested. Otherwise as [`combine_selected`] (print standard, shared resources).
252pub fn combine_grouped(runs: &[Run<'_>]) -> Result<Document, OrganizeError> {
253    combine_grouped_with(runs, FieldConflicts::Rename).map(|o| o.doc)
254}
255
256/// [`combine_grouped`] reporting what the conflict resolution renamed.
257pub fn combine_grouped_with(runs: &[Run<'_>], conflicts: FieldConflicts) -> Result<CombineOutcome, OrganizeError> {
258    // Each file's pages, in the order its runs list them, and each run's place in that list.
259    struct File<'a> {
260        title: &'a str,
261        src: &'a Document,
262        pages: Vec<usize>,
263    }
264    let mut files: Vec<(usize, File<'_>)> = Vec::new();
265    let mut slices: Vec<(usize, usize, usize)> = Vec::with_capacity(runs.len());
266    for (group, title, src, chosen) in runs {
267        let at = match files.iter().position(|(g, _)| g == group) {
268            Some(at) => at,
269            None => {
270                files.push((*group, File { title, src, pages: Vec::new() }));
271                files.len() - 1
272            }
273        };
274        let Some((_, file)) = files.get_mut(at) else { continue };
275        let n = crate::organize::page_count(file.src)?;
276        let pages: Vec<usize> = match chosen {
277            Some(p) => {
278                if let Some(bad) = p.iter().find(|i| **i >= n) {
279                    return Err(OrganizeError::NoSuchPage(*bad));
280                }
281                p.to_vec()
282            }
283            None => (0..n).collect(),
284        };
285        slices.push((at, file.pages.len(), pages.len()));
286        file.pages.extend(pages);
287    }
288    let mut out = Document::new_empty();
289    let mut marks = Vec::new();
290    let mut attachments = Vec::new();
291    let mut copies: Vec<Vec<ObjRef>> = Vec::with_capacity(files.len());
292    let mut renamed_fields = Vec::new();
293    let mut standard: Option<(crate::organize::pdfx::PrintStandard, &Document, HashMap<ObjRef, ObjRef>)> = None;
294    let mut agreed = true;
295    for (_, file) in &files {
296        let at = crate::organize::page_count(&out)?;
297        let imported = import_pages_mapped(&mut out, file.src, &file.pages, at, conflicts)?;
298        copies.push(imported.pages.clone());
299        renamed_fields.extend(imported.renamed_fields);
300        if let Some(first) = imported.pages.first() {
301            marks.push(Mark { title: file.title.to_string(), page: *first, outline: source_outline(file.src, imported.page_map) });
302        }
303        // Attachments come along with all of a file's pages, not with a selection of them.
304        if file.pages.iter().collect::<HashSet<_>>().len() == crate::organize::page_count(file.src)? {
305            collect_attachments(&mut out, file.src, &mut attachments)?;
306        }
307        let declared = crate::organize::pdfx::PrintStandard::of(file.src);
308        match &standard {
309            None => standard = Some((declared, file.src, imported.objects)),
310            Some((first, _, _)) => agreed &= first.same_as(&declared),
311        }
312    }
313    // The pages in the runs' order. A file's first run comes before its others, and its pages
314    // were copied in run order, so its bookmark (its first copied page) is its first page shown.
315    let mut order: Vec<(ObjRef, Dict)> = Vec::new();
316    let mut labels = Vec::new();
317    let mut has_labels = false;
318    for (file, start, len) in slices {
319        let end = start.saturating_add(len);
320        let at = order.len();
321        let run = copies.get(file).and_then(|c| c.get(start..end)).unwrap_or_default();
322        order.extend(run.iter().map(|r| (*r, Dict::new())));
323        if let Some((_, f)) = files.get(file)
324            && let Some(pages) = f.pages.get(start..end)
325        {
326            crate::organize::labels::collect_page_labels(f.src, pages, at, &mut labels, &mut has_labels)?;
327        }
328    }
329    rebuild(&mut out, &order)?;
330    if let Some((declared, src, objects)) = standard.filter(|_| agreed) {
331        crate::organize::pdfx::carry(&mut out, Some(src), &declared, objects)?;
332    }
333    let all: Vec<ObjRef> = out.object_numbers().into_iter().map(|n| ObjRef::new(n, out.generation(n))).collect();
334    crate::organize::dedupe::dedupe_resources(&mut out, &all, false);
335    crate::organize::labels::finish_page_labels(&mut out, &labels, has_labels)?;
336    add_outline(&mut out, &marks)?;
337    set_attachments(&mut out, attachments)?;
338    Ok(CombineOutcome { doc: out, renamed_fields })
