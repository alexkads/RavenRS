4/// Rename the top-level fields about to be registered in `dst` whose fully-qualified names
5/// (`parent.T.child`) are already taken, so each imported field stays its own field instead of
6/// blurring into the destination's field of the same name (fields of one fully-qualified name
7/// are one field). A colliding node's own `/T` becomes the first free `Name-1`, `Name-2`…
8/// variant, so everything below it keeps its place in the tree. Returns what was renamed
9/// (old fully-qualified name → new one), in order.
10pub(super) fn resolve_field_conflicts(dst: &mut Document, fields: &[ObjRef]) -> Result<Vec<(String, String)>, OrganizeError> {
11    let mut taken = taken_field_names(dst)?;
12    let planned = plan_renames(dst, fields, &mut taken);
13    let mut renamed = Vec::new();
14    for (node, new_t, old, new) in planned {
15        dst.update_dict(node, |d| d.set(b"T".to_vec(), Object::String(crate::cos::PdfString::text(&new_t))))?;
16        renamed.push((old, new));
17    }
18    Ok(renamed)
19}

21/// What importing `src` into `dst` would rename, computed without changing either document:
22/// each fully-qualified name of `src`'s fields that `dst` already uses (or that an earlier
23/// field of `src` itself uses) is renamed to the first free variant, exactly as the import
24/// would. Unreadable structure previews as nothing: the import itself will say why it failed.
25pub fn field_name_conflicts(dst: &Document, src: &Document) -> Vec<(String, String)> {
26    let Ok(mut taken) = taken_field_names(dst) else { return Vec::new() };
27    let Ok(roots) = field_roots(src) else { return Vec::new() };
28    plan_renames(src, &roots, &mut taken).into_iter().map(|(_, _, old, new)| (old, new)).collect()
29}

31/// What combining `sources`, in order, would rename in each one: the whole list is simulated
32/// with the same first-free-variant rule the combine uses, so a later file's preview sees the
33/// earlier files' renames. Unreadable structure previews as nothing (the combine itself will
34/// say why it fails); non-colliding names still count as taken for the files after them.
35pub fn combine_field_conflicts(sources: &[&Document]) -> Vec<Vec<(String, String)>> {
36    let mut taken = std::collections::HashSet::new();
37    sources
38        .iter()
39        .map(|src| match field_roots(src) {
40            Ok(roots) => plan_renames(src, &roots, &mut taken).into_iter().map(|(_, _, old, new)| (old, new)).collect(),
41            Err(_) => Vec::new(),
42        })
43        .collect()
44}

46/// Walk field trees `roots` in `doc` and plan the renames that would keep every name separate
47/// from `taken`: each node's fully-qualified name joins `taken` (so a later field of the same
48/// import collides with an earlier one), and a node already taken is renamed to the first free
49/// variant of its own `/T`. Names are compared and renamed as decoded text, not as stored
50/// bytes, so one name written in two encodings (PDFDocEncoding, UTF-16BE) is one name.
51/// Read-only: returns (node, new `/T`, old name, new name). Cycle- and depth-safe.
52fn plan_renames(doc: &Document, roots: &[ObjRef], taken: &mut std::collections::HashSet<String>) -> Vec<(ObjRef, String, String, String)> {
53    let mut reserved = std::collections::HashSet::new();
54    let mut seen = std::collections::HashSet::new();
55    for &root in roots {
56        collect_names(doc, root, "", &mut seen, 0, &mut reserved);
57    }
58    let mut planned = Vec::new();
59    let mut walk = RenameWalk { reserved, ..RenameWalk::default() };
60    for &root in roots {
61        plan_renames_under(doc, root, "", taken, &mut walk, 0, &mut planned);
62    }
63    planned
64}

66/// What one [`plan_renames`] call remembers: the nodes it visited, the incoming tree's own
67/// names (so an earlier rename does not steal a later sibling's name), and for each colliding
68/// name the suffix to try next.
69#[derive(Default)]
70struct RenameWalk {
71    seen: HashSet<ObjRef>,
72    reserved: HashSet<String>,
73    next: HashMap<String, u32>,
74}
75
76/// One node of [`plan_renames`].
77fn plan_renames_under(
78    doc: &Document,
79    node: ObjRef,
80    prefix: &str,
81    taken: &mut std::collections::HashSet<String>,
82    walk: &mut RenameWalk,
83    depth: u8,
84    planned: &mut Vec<(ObjRef, String, String, String)>,
85) {
86    if depth > 32 || !walk.seen.insert(node) {
87        return;
88    }
89    let Some(d) = doc.get(node).as_dict().cloned() else { return };
90    let mut name = d.get(b"T").and_then(|t| doc.resolve(t).as_string().map(|s| s.to_text())).unwrap_or_default();
91    if !name.is_empty() {
92        let fqn = qualified(prefix, &name);
93        if taken.contains(&fqn) {
94            // `taken` only grows, so every suffix below the last one handed out for this name is
95            // still taken: resume there instead of at 1.
96            let mut n = walk.next.get(&fqn).copied().unwrap_or(1);
97            let mut variant;
98            let mut candidate;
99            while {
100                #[cfg(test)]
101                crate::organize::work(1);
102                variant = format!("{name}-{n}");
103                candidate = qualified(prefix, &variant);
104                taken.contains(&candidate) || walk.reserved.contains(&candidate)
105            } {
106                n = n.saturating_add(1);
107            }
108            walk.next.insert(fqn.clone(), n.saturating_add(1));
109            planned.push((node, variant.clone(), fqn, candidate.clone()));
110            taken.insert(candidate);
111            name = variant;
112        } else {
113            taken.insert(fqn);
114        }
115    }
116    let prefix = qualified_optional(prefix, &name);
117    let kids = d.get(b"Kids").map(|k| doc.resolve(k)).and_then(|k| k.as_array().cloned()).unwrap_or_default();
118    for kid in kids.iter().filter_map(|k| k.as_ref()) {
119        plan_renames_under(doc, kid, &prefix, taken, walk, depth + 1, planned);
120    }
121}

123/// The field names in use in `dst`: every name in its form, and every name hanging only off a
124/// page's widgets — a field the form never lists still owns its name.
125fn taken_field_names(dst: &Document) -> Result<std::collections::HashSet<String>, OrganizeError> {
126    let mut taken = std::collections::HashSet::new();
127    let mut seen = std::collections::HashSet::new();
128    for root in field_roots(dst)? {
129        collect_names(dst, root, "", &mut seen, 0, &mut taken);
130    }
131    Ok(taken)
132}

134/// A document's field trees, wherever they are rooted: the `/AcroForm /Fields` entries, and
135/// the fields behind widgets on pages (climbed to their root, for forms that never registered
136/// them).
137fn field_roots(doc: &Document) -> Result<Vec<ObjRef>, OrganizeError> {
138    let mut roots: Vec<ObjRef> = doc
139        .root()
140        .and_then(|r| doc.get(r).as_dict().cloned())
141        .and_then(|c| c.get(b"AcroForm").map(|f| doc.resolve(f)))
142        .and_then(|f| f.as_dict().cloned())
143        .and_then(|f| f.get(b"Fields").map(|l| doc.resolve(l)))
144        .and_then(|l| l.as_array().cloned())
145        .unwrap_or_default()
146        .iter()
147        .filter_map(|f| f.as_ref())
148        .collect();
149    for (page, _) in walk(doc)? {
150        let Some(d) = doc.get(page).as_dict().cloned() else { continue };
151        let annots = d.get(b"Annots").map(|a| doc.resolve(a)).and_then(|a| a.as_array().cloned()).unwrap_or_default();
152        for a in annots.iter() {
153            let Some(w) = doc.resolve(a).as_dict().cloned() else { continue };
154            if w.get(b"Subtype").map(|s| doc.resolve(s)).as_deref().and_then(Object::as_name) != Some(b"Widget") {
155                continue;
156            }
157            // Walk up to the field the widget belongs to, however deep it sits; a widget that
158            // is its own field is its own root.
159            let Some(mut root) = a.as_ref() else { continue };
160            let mut up = 0;
161            let mut visited = HashSet::from([root]);
162            while let Some(parent) = doc.get(root).as_dict().and_then(|p| p.reference(b"Parent")) {
163                if up >= 32 || !visited.insert(parent) || doc.get(parent).as_dict().is_none() {
164                    break;
165                }
166                root = parent;
167                up += 1;
168            }
169            roots.push(root);
170        }
171    }
172    Ok(roots)
173}

175/// Record the fully-qualified names of one field tree. Cycle- and depth-safe.
176fn collect_names(
177    dst: &Document,
178    node: ObjRef,
179    prefix: &str,
180    seen: &mut std::collections::HashSet<ObjRef>,
181    depth: u8,
182    out: &mut std::collections::HashSet<String>,
183) {
184    if depth > 32 || !seen.insert(node) {
185        return;
186    }
187    let Some(d) = dst.get(node).as_dict().cloned() else { return };
188    let name = d.get(b"T").and_then(|t| dst.resolve(t).as_string().map(|s| s.to_text())).unwrap_or_default();
189    let prefix = qualified_optional(prefix, &name);
190    if !name.is_empty() {
191        out.insert(prefix.clone());
192    }
193    let kids = d.get(b"Kids").map(|k| dst.resolve(k)).and_then(|k| k.as_array().cloned()).unwrap_or_default();
194    for kid in kids.iter().filter_map(|k| k.as_ref()) {
195        collect_names(dst, kid, &prefix, seen, depth + 1, out);
196    }
197}

199/// `prefix.name`, or `name` at the top level.
200fn qualified(prefix: &str, name: &str) -> String {
201    if prefix.is_empty() { name.to_string() } else { format!("{prefix}.{name}") }
202}
203
204/// The prefix a node's children hang under: its own name, when it has one.
205fn qualified_optional(prefix: &str, name: &str) -> String {
206    if name.is_empty() { prefix.to_string() } else { qualified(prefix, name) }
207}

209/// Add `kid` to `field`'s `/Kids`, in the array object itself when `/Kids` is one, so the field
210/// keeps its other widgets.
211pub(super) fn add_kid(dst: &mut Document, field: ObjRef, kid: ObjRef) -> Result<(), OrganizeError> {
212    match dst.get(field).as_dict().and_then(|f| f.get(b"Kids").cloned()) {
213        Some(Object::Ref(list)) => {
214            let mut kids = dst.get(list).as_array().cloned().unwrap_or_default();
215            kids.push(Object::Ref(kid));
216            dst.set(list, Object::Array(kids));
217        }
218        _ => dst.update_dict(field, |f| {
219            let mut kids = f.get(b"Kids").and_then(|k| k.as_array()).cloned().unwrap_or_default();
220            kids.push(Object::Ref(kid));
221            f.set(b"Kids".to_vec(), Object::Array(kids));
222        })?,
223    }
224    Ok(())
225}

227/// PdfCraft patch (#1810): split a widget that doubles as its own field into a parent field
228/// holding the field keys (name, value, flags…) and a widget child holding the annotation keys
229/// (rectangle, appearance, page). The parent takes the widget's place among the top-level fields;
230/// later copies of the page add widget children of it, so every page has its own widget and the
231/// field stays one.
232pub(super) fn split_own_field(dst: &mut Document, fields: &mut [ObjRef], widget: ObjRef) {
233    let Some(d) = dst.get(widget).as_dict().cloned() else { return };
234    let (mut field, mut child) = field_and_widget(dst, &d);
235    let parent = dst.add(Object::Null);
236    field.set(b"Kids".to_vec(), Object::Array(vec![Object::Ref(widget)]));
237    child.set(b"Parent".to_vec(), Object::Ref(parent));
238    dst.set(parent, Object::Dict(field));
239    dst.set(widget, Object::Dict(child));
240    if let Some(entry) = fields.iter_mut().find(|f| **f == widget) {
241        *entry = parent;
242    }
243}

245/// A combined field/widget has two additional-action namespaces (ISO 32000-1 tables 194/196).
246/// Partition them as well as the ordinary entries, including when copying a later widget.
247pub(super) fn field_and_widget(doc: &Document, d: &Dict) -> (Dict, Dict) {
248    let (mut field, mut child) = (Dict::new(), Dict::new());
249    for (k, v) in d.iter() {
250        if k.as_slice() == b"AA"
251            && let Some(actions) = doc.resolve(v).as_dict().cloned()
252        {
253            let (mut field_actions, mut widget_actions) = (Dict::new(), Dict::new());
254            for (trigger, action) in actions.iter() {
255                if [b"E".as_slice(), b"X", b"D", b"U", b"Fo", b"Bl", b"PO", b"PC", b"PV", b"PI"].contains(&trigger.as_slice()) {
256                    widget_actions.set(trigger.clone(), action.clone());
257                } else {
258                    field_actions.set(trigger.clone(), action.clone());
259                }
260            }
261            if field_actions.iter().next().is_some() {
262                field.set(k.clone(), Object::Dict(field_actions));
263            }
264            if widget_actions.iter().next().is_some() {
265                child.set(k.clone(), Object::Dict(widget_actions));
266            }
267        } else if WIDGET_KEYS.contains(&k.as_slice()) {
268            child.set(k.clone(), v.clone());
269        } else {
270            field.set(k.clone(), v.clone());
271        }
272    }
273    (field, child)
274}

276/// Add top-level fields to the destination's `/AcroForm /Fields` (creating the form if needed).
277pub(super) fn register_fields(dst: &mut Document, fields: &[ObjRef]) -> Result<(), OrganizeError> {
278    if fields.is_empty() {
279        return Ok(());
280    }
281    let root = dst.root().ok_or(OrganizeError::NoPageTree)?;
282    let catalog = dst.get(root).as_dict().cloned().ok_or(OrganizeError::NoPageTree)?;
283    let (form_ref, mut form) = match catalog.get(b"AcroForm") {
284        Some(Object::Ref(r)) => (Some(*r), dst.get(*r).as_dict().cloned().unwrap_or_default()),
285        Some(Object::Dict(d)) => (None, d.clone()),
286        _ => (None, Dict::new()),
287    };
288    let mut list = form.get(b"Fields").map(|f| dst.resolve(f)).and_then(|f| f.as_array().cloned()).unwrap_or_default();
289    let mut listed: HashSet<ObjRef> = list.iter().filter_map(Object::as_ref).collect();
290    for f in fields {
291        #[cfg(test)]
292        crate::organize::work(1);
293        if listed.insert(*f) {
294            list.push(Object::Ref(*f));
295        }
296    }
297    form.set(b"Fields".to_vec(), Object::Array(list));
298    match form_ref {
299        Some(r) => dst.set(r, Object::Dict(form)),
300        None => {
301            let r = dst.add(form);
302            dst.update_dict(root, |c| c.set(b"AcroForm".to_vec(), Object::Ref(r)))?;
303        }
304    }
305    Ok(())
306}

314    fn build(bodies: &[&str]) -> Document {
315        let mut out = b"%PDF-1.7\n".to_vec();
316        let mut offsets = Vec::new();
317        for (i, b) in bodies.iter().enumerate() {
318            offsets.push(out.len());
319            out.extend_from_slice(format!("{} 0 obj\n{b}\nendobj\n", i + 1).as_bytes());
320        }
321        let xref = out.len();
322        out.extend_from_slice(format!("xref\n0 {}\n0000000000 65535 f \n", bodies.len() + 1).as_bytes());
323        for o in offsets {
324            out.extend_from_slice(format!("{o:010} 00000 n \n").as_bytes());
325        }
326        out.extend_from_slice(format!("trailer\n<< /Size {} /Root 1 0 R >>\nstartxref\n{xref}\n%%EOF\n", bodies.len() + 1).as_bytes());
327        Document::open(Arc::new(out)).expect("opens")
328    }
