1//! Page transitions and display times (ISO 32000-2 §12.4.4): the `/Trans` dictionary and `/Dur`
2//! entry a full-screen presentation plays, as Acrobat's Organize ▸ Page Transitions sets them.
3//! Reading them back is the renderer's `DocInfo::transitions`.
4
5use crate::cos::{Dict, Document, Object};
6
7use crate::organize::{OrganizeError, check, walk};
8
9/// The transition styles of ISO 32000-2 table 164, as written in `/S` (`R` is "replace": no
10/// effect).
11pub const TRANSITION_STYLES: [&str; 12] = ["Split", "Blinds", "Box", "Wipe", "Dissolve", "Glitter", "R", "Fly", "Push", "Cover", "Uncover", "Fade"];
12
13/// The directions `/Di` allows, in degrees.
14pub const TRANSITION_DIRECTIONS: [u16; 5] = [0, 90, 180, 270, 315];
15
16/// A page transition to write: the `/Trans` dictionary's entries, with the spec's defaults.
17#[derive(Clone, Debug, PartialEq)]
18pub struct Transition {
19    /// One of [`TRANSITION_STYLES`].
20    pub style: String,
21    /// `/D`: seconds the transition takes (default 1).
22    pub duration: f32,
23    /// `/Dm`: horizontal (`/H`, the default) or vertical (`/V`); Split and Blinds.
24    pub horizontal: bool,
25    /// `/M`: inward (`/I`, the default) or outward (`/O`); Split, Box and Fly.
26    pub inward: bool,
27    /// `/Di`: one of [`TRANSITION_DIRECTIONS`] (default 0), or `None` for `/None` (Fly only);
28    /// Wipe, Glitter, Fly, Cover, Uncover and Push.
29    pub direction: Option<u16>,
30    /// `/SS`: the Fly scale (default 1).
31    pub scale: f32,
32    /// `/B`: Fly with an opaque backdrop.
33    pub opaque: bool,
34}
35
36impl Transition {
37    /// A transition of `style` with every other entry at its default.
38    pub fn new(style: &str) -> Self {
39        Self { style: style.to_string(), duration: 1.0, horizontal: true, inward: true, direction: Some(0), scale: 1.0, opaque: false }
40    }
41}
42
43/// What a page gets: its transition (`None` removes `/Trans`) and the seconds it stays before the
44/// presentation advances (`None` removes `/Dur`: the page waits for the presenter).
45#[derive(Clone, Debug, PartialEq)]
46pub struct TransitionSpec {
47    pub transition: Option<Transition>,
48    pub display: Option<f32>,
49}
50
51/// Set (or remove) the transition and display time of `pages` (0-based). Every value is checked
52/// before any page changes: an unknown style, a direction that is not one of the five, a negative
53/// or non-finite duration, scale or display time are refused.
54pub fn set_transition(doc: &mut Document, pages: &[usize], spec: &TransitionSpec) -> Result<(), OrganizeError> {
55    let bad = |m: String| OrganizeError::Invalid(m);
56    if let Some(t) = &spec.transition {
57        if !TRANSITION_STYLES.contains(&t.style.as_str()) {
58            return Err(bad(format!("unknown transition style {:?}; one of {}", t.style, TRANSITION_STYLES.join(", "))));
59        }
60        if !(t.duration.is_finite() && t.duration >= 0.0) {
61            return Err(bad("the transition duration must be zero or more seconds".into()));
62        }
63        if let Some(d) = t.direction
64            && !TRANSITION_DIRECTIONS.contains(&d)
65        {
66            return Err(bad(format!("the direction must be one of {:?} degrees", TRANSITION_DIRECTIONS)));
67        }
68        if !(t.scale.is_finite() && t.scale > 0.0) {
69            return Err(bad("the Fly scale must be more than zero".into()));
70        }
71    }
72    if let Some(s) = spec.display
73        && !(s.is_finite() && s >= 0.0)
74    {
75        return Err(bad("the display time must be zero or more seconds".into()));
76    }
77    let all = walk(doc)?;
78    check(pages, all.len())?;
79    let trans = spec.transition.as_ref().map(trans_dict);
80    for &i in pages {
81        let Some((page, _)) = all.get(i) else { continue };
82        let trans = trans.clone();
83        doc.update_dict(*page, |d| {
84            match trans {
85                Some(t) => d.set(b"Trans".to_vec(), Object::Dict(t)),
86                None => {
87                    d.remove(b"Trans");
88                }
89            }
90            match spec.display {
91                Some(s) => d.set(b"Dur".to_vec(), Object::Real(f64::from(s))),
92                None => {
93                    d.remove(b"Dur");
94                }
95            }
96        })?;
97    }
98    Ok(())
99}

101/// The `/Trans` dictionary for `t`: only the entries that differ from the defaults, and only the
102/// ones the style reads (table 164), so the file says no more than it means.
103fn trans_dict(t: &Transition) -> Dict {
104    let style = t.style.as_str();
105    let mut d = Dict::new();
106    d.set(b"Type".to_vec(), Object::name("Trans"));
107    d.set(b"S".to_vec(), Object::name(style));
108    if t.duration != 1.0 {
109        d.set(b"D".to_vec(), Object::Real(f64::from(t.duration)));
110    }
111    if matches!(style, "Split" | "Blinds") && !t.horizontal {
112        d.set(b"Dm".to_vec(), Object::name("V"));
113    }
114    if matches!(style, "Split" | "Box" | "Fly") && !t.inward {
115        d.set(b"M".to_vec(), Object::name("O"));
116    }
117    if matches!(style, "Wipe" | "Glitter" | "Fly" | "Cover" | "Uncover" | "Push") {
118        match t.direction {
119            None if style == "Fly" => d.set(b"Di".to_vec(), Object::name("None")),
120            Some(deg) if deg != 0 => d.set(b"Di".to_vec(), Object::Int(i64::from(deg))),
121            _ => {}
122        }
123    }
124    if style == "Fly" {
125        if t.scale != 1.0 {
126            d.set(b"SS".to_vec(), Object::Real(f64::from(t.scale)));
127        }
128        if t.opaque {
129            d.set(b"B".to_vec(), Object::Bool(true));
130        }
131    }
132    d
133}
