1//! Page labels (ISO 32000-2 §12.4.2): "i, ii, iii, 1, 2, A-1…", execution plan M4.4.
2//!
3//! Labels are a number tree of ranges: from page `start` on, use a numbering `style`, a
4//! `prefix` and a first number. [`number_pages`] is Acrobat's "Number pages": relabel pages
5//! `from..=to`, and keep every later page showing the label it had before. The tree is written
6//! back as one flat `/Nums` array, with ranges that merely continue the previous one merged.
7
8use std::collections::HashSet;

11use crate::cos::{Dict, Document, ObjRef, Object, PdfString};

13use crate::organize::{OrganizeError, page_count, pages_root};

15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum LabelStyle {
17    /// 1, 2, 3
18    Decimal,
19    /// I, II, III
20    UpperRoman,
21    /// i, ii, iii
22    LowerRoman,
23    /// A … Z, AA … ZZ
24    UpperAlpha,
25    /// a … z, aa … zz
26    LowerAlpha,
27    /// Prefix only (no number).
28    None,
29}
30
31impl LabelStyle {
32    fn from_name(n: &[u8]) -> Self {
33        match n {
34            b"D" => Self::Decimal,
35            b"R" => Self::UpperRoman,
36            b"r" => Self::LowerRoman,
37            b"A" => Self::UpperAlpha,
38            b"a" => Self::LowerAlpha,
39            _ => Self::None,
40        }
41    }
42
43    fn name(self) -> Option<&'static str> {
44        match self {
45            Self::Decimal => Some("D"),
46            Self::UpperRoman => Some("R"),
47            Self::LowerRoman => Some("r"),
48            Self::UpperAlpha => Some("A"),
49            Self::LowerAlpha => Some("a"),
50            Self::None => None,
51        }
52    }
53
54    /// Format `n` (≥ 1) in this style for previews.
55    /// A numeral exceeding 1024 bytes falls back to decimal; editing and document label
56    /// queries use the checked formatter and return an error instead of storing a fallback.
57    pub fn format(self, n: u32) -> String {
58        self.format_checked(n, MAX_LABEL_BYTES).unwrap_or_else(|_| n.to_string())
59    }
60
61    fn format_checked(self, n: u32, max_bytes: usize) -> Result<String, OrganizeError> {
62        let mut out = match self {
63            Self::Decimal => Some(n.to_string()),
64            Self::UpperRoman | Self::LowerRoman => roman(u64::from(n), max_bytes),
65            Self::UpperAlpha | Self::LowerAlpha => alpha(u64::from(n), max_bytes),
66            Self::None => Some(String::new()),
67        }
68        .ok_or_else(|| invalid_label("a label exceeds 1024 UTF-8 bytes"))?;
69        if out.len() > max_bytes {
70            return Err(invalid_label("a label exceeds 1024 UTF-8 bytes"));
71        }
72        if matches!(self, Self::UpperRoman | Self::UpperAlpha) {
73            out.make_ascii_uppercase();
74        }
75        Ok(out)
76    }
77}

79/// One labelling range.
80#[derive(Clone, Debug, PartialEq, Eq)]
81pub struct LabelRange {
82    /// First page (0-based) of the range.
83    pub start: usize,
84    pub style: LabelStyle,
85    pub prefix: String,
86    /// The number of the range's first page (`/St`, at least 1).
87    pub first: u32,
88}
89
90impl LabelRange {
91    fn number(&self, page: usize) -> Result<u32, OrganizeError> {
92        if self.style == LabelStyle::None {
93            return Ok(self.first); // /St is ignored when the range has no numbering style.
94        }
95        let offset = page.checked_sub(self.start).and_then(|n| u32::try_from(n).ok()).ok_or_else(|| invalid_label("a page number is too large"))?;
96        self.first.checked_add(offset).ok_or_else(|| invalid_label("a page number overflows"))
97    }
98
99    fn label(&self, page: usize) -> Result<String, OrganizeError> {
100        let remaining = MAX_LABEL_BYTES.checked_sub(self.prefix.len()).ok_or_else(|| invalid_label("a prefix exceeds 1024 UTF-8 bytes"))?;
101        let number = self.style.format_checked(self.number(page)?, remaining)?;
102        let mut label = self.prefix.clone();
103        label.push_str(&number);
104        Ok(label)
105    }
106}

123// Borrow direct objects instead of cloning complete untrusted dictionaries or arrays.
124fn with_label_object<T>(doc: &Document, object: &Object, f: impl FnOnce(&Object) -> Result<T, OrganizeError>) -> Result<T, OrganizeError> {
125    let Object::Ref(r) = object else { return f(object) };
126    let mut seen = HashSet::from([*r]);
127    let mut object = doc.get(*r);
128    while let Object::Ref(r) = object.as_ref() {
129        if seen.len() >= MAX_LABEL_TREE_DEPTH || !seen.insert(*r) {
130            return Err(invalid_label("a number tree reference is cyclic or too deeply nested"));
131        }
132        object = doc.get(*r);
133    }
134    f(&object)
135}

137fn label_work(left: &mut usize, count: usize) -> Result<(), OrganizeError> {
138    *left = left.checked_sub(count).ok_or_else(|| invalid_label("the number tree has too many entries"))?;
139    Ok(())
140}

142fn read_label_node(
143    doc: &Document,
144    node: &Object,
145    depth: usize,
146    left: &mut usize,
147    seen: &mut HashSet<ObjRef>,
148    entries: &mut Vec<LabelRange>,
149    prefix_left: &mut usize,
150) -> Result<(), OrganizeError> {
151    label_work(left, 1)?;
152    if depth > MAX_LABEL_TREE_DEPTH {
153        return Err(invalid_label("the number tree is too deep"));
154    }
155    // Only indirect nodes can be reached twice; key them by reference, not by the address of
156    // a resolved object (a missing reference resolves to a fresh, short-lived `Null`).
157    if let Object::Ref(r) = node
158        && !seen.insert(*r)
159    {
160        return Err(invalid_label("the number tree repeats a node or contains a cycle"));
161    }
162    with_label_object(doc, node, |node| {
163        let Some(d) = node.as_dict() else { return Ok(()) };
164        if let Some(nums) = d.get(b"Nums") {
165            with_label_object(doc, nums, |nums| {
166                let Some(nums) = nums.as_array() else { return Ok(()) };
167                label_work(left, nums.len().div_ceil(2))?;
168                for pair in nums.as_chunks::<2>().0 {
169                    let Some(start) = pair[0].as_int().and_then(|n| usize::try_from(n).ok()) else { continue };
170                    with_label_object(doc, &pair[1], |spec| {
171                        let Some(spec) = spec.as_dict() else { return Ok(()) };
172                        let prefix = match spec.get(b"P") {
173                            Some(prefix) => with_label_object(doc, prefix, |p| match p.as_string() {
174                                Some(p) if p.bytes.len() > MAX_PREFIX_BYTES => Err(invalid_label("a raw prefix exceeds 2050 bytes")),
175                                Some(p) => Ok(p.to_text()),
176                                None => Ok(String::new()),
177                            })?,
178                            None => String::new(),
179                        };
180                        if prefix.len() > MAX_LABEL_BYTES {
181                            return Err(invalid_label("a decoded prefix exceeds 1024 UTF-8 bytes"));
182                        }
183                        // A prefix object shared by many ranges is copied into each: count every copy.
184                        *prefix_left =
185                            prefix_left.checked_sub(prefix.len()).ok_or_else(|| invalid_label("decoded prefixes exceed 4 MiB in total"))?;
186                        let style = spec.name(b"S").map_or(LabelStyle::None, LabelStyle::from_name);
187                        let first = spec.int(b"St").unwrap_or(1).max(1);
188                        let first = if style == LabelStyle::None {
189                            // /St has no effect without /S; keep the existing compatibility clamp.
190                            first.min(i64::from(u32::MAX)) as u32
191                        } else {
192                            u32::try_from(first).map_err(|_| invalid_label("a first number exceeds the supported integer range"))?
193                        };
194                        entries.push(LabelRange { start, style, prefix, first });
195                        Ok(())
196                    })?;
197                }
198                Ok(())
199            })?;
200        }
201        if let Some(kids) = d.get(b"Kids") {
202            with_label_object(doc, kids, |kids| {
203                let Some(kids) = kids.as_array() else { return Ok(()) };
204                label_work(left, kids.len())?;
205                if !kids.is_empty() && depth == MAX_LABEL_TREE_DEPTH {
206                    return Err(invalid_label("the number tree is too deep"));
207                }
208                // Recursion is bounded before descending, and arrays are borrowed throughout.
209                for kid in kids {
210                    read_label_node(doc, kid, depth + 1, left, seen, entries, prefix_left)?;
211                }
212                Ok(())
213            })?;
214        }
215        Ok(())
216    })
217}

219fn checked_ranges(doc: &Document) -> Result<Vec<LabelRange>, OrganizeError> {
220    let Some(root) = doc.root() else { return Ok(Vec::new()) };
221    let catalog = doc.get(root);
222    let Some(tree) = catalog.as_dict().and_then(|c| c.get(b"PageLabels")) else { return Ok(Vec::new()) };
223    let mut entries = Vec::new();
224    let (mut left, mut prefix_left) = (MAX_LABEL_TREE_WORK, MAX_PREFIX_TOTAL_BYTES);
225    read_label_node(doc, tree, 0, &mut left, &mut HashSet::new(), &mut entries, &mut prefix_left)?;
226    // A start page given twice keeps its later range, as readers that overwrite do (the sort is
227    // stable, so the later one is last among equals).
228    entries.sort_by_key(|r| r.start);
229    let mut unique: Vec<LabelRange> = Vec::with_capacity(entries.len());
230    for r in entries {
231        match unique.last_mut() {
232            Some(last) if last.start == r.start => *last = r,
233            _ => unique.push(r),
234        }
235    }
236    Ok(unique)
237}

239/// A view of every page's label: the sorted ranges and the page count, with no string kept per
240/// page. [`PageLabels::label`] builds one label; [`PageLabels::iter`] builds them one at a time, so
241/// memory follows the ranges, not the pages.
242#[derive(Clone, Debug, PartialEq, Eq)]
243pub struct PageLabels {
244    ranges: Vec<LabelRange>,
245    pages: usize,
246}
247
248impl PageLabels {
249    /// Reads the document's `/PageLabels` and page count. A tree that is cyclic, too deep or too
250    /// large is an error.
251    pub fn new(doc: &Document) -> Result<Self, OrganizeError> {
252        Ok(Self { ranges: checked_ranges(doc)?, pages: page_count(doc)? })
253    }
254
255    /// The number of pages the labels cover.
256    pub fn pages(&self) -> usize {
257        self.pages
258    }
259
260    /// The label of page `page` (0-based), as a viewer shows it: a page with no range, or with an
261    /// empty range label, shows its physical number. Errors when the range's label cannot be
262    /// formatted (over 1,024 UTF-8 bytes, or a number past 32 bits).
263    pub fn label(&self, page: usize) -> Result<String, OrganizeError> {
264        if page >= self.pages {
265            return Err(OrganizeError::NoSuchPage(page));
266        }
267        let covering = self.ranges.partition_point(|r| r.start <= page);
268        let label = match covering.checked_sub(1).and_then(|i| self.ranges.get(i)) {
269            Some(range) => range.label(page)?,
270            None => String::new(),
271        };
272        Ok(if label.is_empty() { (page + 1).to_string() } else { label })
273    }
274
275    /// Every page's label, in page order, one at a time. Reading is lenient: a label that cannot
276    /// be formatted shows its physical page number, and the other ranges keep their labels.
277    pub fn iter(&self) -> impl Iterator<Item = String> + '_ {
278        (0..self.pages).map(|page| self.label(page).unwrap_or_else(|_| (page + 1).to_string()))
279    }
280}

288/// Strict check for an edit: every label that the sorted, unique `ranges` produce over `pages`
289/// pages must fit. A range is checked from its largest number, so the work follows the ranges
290/// (and at most a thousand numbers at each end of a Roman range), not the pages.
291fn check_spans(ranges: &[LabelRange], pages: usize) -> Result<(), OrganizeError> {
292    let Some(last_page) = pages.checked_sub(1) else { return Ok(()) };
293    for (i, r) in ranges.iter().enumerate() {
294        // Prefix-only labels have no number to check; a range past the last page is never shown.
295        if r.start > last_page || r.style == LabelStyle::None {
296            continue;
297        }
298        let end = ranges.get(i + 1).map_or(last_page, |next| next.start.saturating_sub(1).min(last_page));
299        let lo = u64::from(r.first);
300        let hi = lo.saturating_add(u64::try_from(end.saturating_sub(r.start)).unwrap_or(u64::MAX));
301        let prefix = u64::try_from(r.prefix.len()).unwrap_or(u64::MAX);
302        let remaining = (MAX_LABEL_BYTES as u64).checked_sub(prefix).ok_or_else(|| invalid_label("a prefix exceeds 1024 UTF-8 bytes"))?;
303        // Numbers past u32::MAX are never formatted, so a label that is too long comes first.
304        if longest_numeral(r.style, lo, hi.min(u64::from(u32::MAX))) > remaining {
305            return Err(invalid_label("a label exceeds 1024 UTF-8 bytes"));
306        }
307        if hi > u64::from(u32::MAX) {
308            return Err(invalid_label("a page number overflows"));
309        }
310    }
311    Ok(())
312}

314/// The longest numeral `style` writes for the numbers `lo..=hi` (`1 <= lo <= hi`), in bytes.
315fn longest_numeral(style: LabelStyle, lo: u64, hi: u64) -> u64 {
316    match style {
317        // Decimal and alphabetic numerals only grow with the number, so the last one is longest.
318        LabelStyle::Decimal => u64::from(hi.checked_ilog10().unwrap_or(0)) + 1,
319        LabelStyle::UpperAlpha | LabelStyle::LowerAlpha => hi.saturating_sub(1) / 26 + 1,
320        LabelStyle::UpperRoman | LabelStyle::LowerRoman => longest_roman(lo, hi),
321        LabelStyle::None => 0,
322    }
323}

325/// The byte length of [`roman`]'s numeral for `n`: `n / 1000` copies of `m`, then the hundreds,
326/// tens and units, each a table lookup.
327fn roman_len(n: u64) -> u64 {
328    // Lengths of i, ii, iii, iv, v, vi, vii, viii, ix (and the same pattern for tens and hundreds).
329    const UNITS: [u64; 10] = [0, 1, 2, 3, 2, 1, 2, 3, 4, 2];
330    let digit = |place: u64| UNITS.get(usize::try_from(place % 10).unwrap_or_default()).copied().unwrap_or_default();
331    n / 1000 + digit(n / 100) + digit(n / 10) + digit(n)
332}

334/// The longest Roman numeral among `lo..=hi`, in bytes. Within one thousand the longest numeral is
335/// the one ending in 888, and every whole thousand between the ends contains it, so only the
336/// partial thousands at either end are read.
337fn longest_roman(lo: u64, hi: u64) -> u64 {
338    let (first, last) = (lo / 1000, hi / 1000);
339    if first == last {
340        return (lo..=hi).map(roman_len).max().unwrap_or_default();
341    }
342    let head = (lo..=first.saturating_mul(1000).saturating_add(999)).map(roman_len).max().unwrap_or_default();
343    let tail = (last.saturating_mul(1000)..=hi).map(roman_len).max().unwrap_or_default();
344    // Each whole thousand between the ends peaks at its thousand plus 12 (for 888); the last is largest.
345    let middle = if last.saturating_sub(first) >= 2 { last.saturating_sub(1).saturating_add(12) } else { 0 };
346    head.max(tail).max(middle)
347}

349/// Replace all label ranges (an empty list removes `/PageLabels`).
350pub fn set_page_label_ranges(doc: &mut Document, ranges: &[LabelRange]) -> Result<(), OrganizeError> {
351    pages_root(doc)?;
352    let root = doc.root().ok_or(OrganizeError::NoPageTree)?;
353    if ranges.is_empty() {
354        doc.update_dict(root, |c| {
355            c.remove(b"PageLabels");
356        })?;
357        return Ok(());
358    }
359    if ranges.len() >= MAX_LABEL_TREE_WORK {
360        return Err(invalid_label("the number tree has too many entries"));
361    }
362    // Validate every range and all effective labels before changing any document object.
363    let mut prefix_left = MAX_PREFIX_TOTAL_BYTES;
364    for r in ranges {
365        if r.first == 0 || i64::try_from(r.start).is_err() {
366            return Err(invalid_label("range starts and first numbers must be valid positive PDF integers"));
367        }
368        r.label(r.start)?;
369        prefix_left = prefix_left.checked_sub(r.prefix.len()).ok_or_else(|| invalid_label("decoded prefixes exceed 4 MiB in total"))?;
370    }
371    let mut sorted = ranges.to_vec();
372    sorted.sort_by_key(|r| r.start);
373    if sorted.windows(2).any(|pair| pair.first().zip(pair.get(1)).is_some_and(|(a, b)| a.start == b.start)) {
374        return Err(invalid_label("range start pages must be unique"));
375    }
376    check_spans(&sorted, page_count(doc)?)?;
377    let mut nums = Vec::new();
378    for r in &sorted {
379        let mut spec = Dict::new();
380        if let Some(s) = r.style.name() {
381            spec.set(b"S".to_vec(), Object::name(s));
382        }
383        if !r.prefix.is_empty() {
384            spec.set(b"P".to_vec(), Object::String(PdfString::text(&r.prefix)));
385        }
386        if r.first != 1 {
387            spec.set(b"St".to_vec(), Object::Int(i64::from(r.first)));
388        }
389        nums.push(Object::Int(r.start as i64));
390        // Indirect, as Acrobat writes them: a direct dictionary is valid too, but some readers
391        // (pdfannots2json, through unipdf) skip it and show physical page numbers (#1763).
392        nums.push(Object::Ref(doc.add(Object::Dict(spec))));
393    }
394    let mut tree = Dict::new();
395    tree.set(b"Nums".to_vec(), Object::Array(nums));
396    let tree = doc.add(Object::Dict(tree));
397    doc.update_dict(root, |c| c.set(b"PageLabels".to_vec(), Object::Ref(tree)))?;
398    Ok(())
399}

401/// After `deleted` pages (0-based, as they were numbered before the delete) are gone, move the label
402/// ranges with the pages they label: a range starts as many pages earlier as were deleted before
403/// it, and keeps its first number, so numbering continues over the gap as in Acrobat. A range whose
404/// first page went starts on the next page left; one left with no pages gives way to the next.
405/// A tree that can't be read is left as it is: the delete still goes ahead.
406pub(crate) fn drop_deleted_pages(doc: &mut Document, deleted: &[usize]) -> Result<(), OrganizeError> {
407    let Ok(old) = checked_ranges(doc) else { return Ok(()) };
408    let mut deleted = deleted.to_vec();
409    deleted.sort_unstable();
410    deleted.dedup();
411    let mut ranges: Vec<LabelRange> = Vec::with_capacity(old.len());
412    for r in &old {
413        let start = r.start.saturating_sub(deleted.partition_point(|d| *d < r.start));
414        match ranges.last_mut() {
415            // The later range is the one on the first page left; ranges stay sorted by start.
416            Some(last) if last.start == start => *last = LabelRange { start, ..r.clone() },
417            _ => ranges.push(LabelRange { start, ..r.clone() }),
418        }
419    }
420    if ranges == old {
421        return Ok(());
422    }
423    set_page_label_ranges(doc, &ranges)
424}

426/// Acrobat's "Number pages": label pages `from..=to` (0-based) with `style`, `prefix` and
427/// starting number `first`; pages after `to` keep the labels they had.
428pub fn number_pages(doc: &mut Document, from: usize, to: usize, style: LabelStyle, prefix: &str, first: u32) -> Result<(), OrganizeError> {
429    if prefix.len() > MAX_LABEL_BYTES {
430        return Err(invalid_label("a prefix exceeds 1024 UTF-8 bytes"));
431    }
432    let n = page_count(doc)?;
433    if to >= n {
434        return Err(OrganizeError::NoSuchPage(to));
435    }
436    if from > to {
437        return Err(OrganizeError::NoSuchPage(from));
438    }
439    let mut old = checked_ranges(doc)?;
440    if old.first().is_none_or(|r| r.start > 0) {
441        old.insert(0, default_range());
442    }
443    // `old` starts at page 0, so a range is always found.
444    let at = |p: usize| old.iter().rev().find(|r| r.start <= p).cloned().unwrap_or_else(default_range);
445    let mut ranges: Vec<LabelRange> = old.iter().filter(|r| r.start < from).cloned().collect();
446    ranges.push(LabelRange { start: from, style, prefix: prefix.to_string(), first: first.max(1) });
447    if to + 1 < n {
448        let r = at(to + 1);
449        let first = r.number(to + 1)?;
450        ranges.push(LabelRange { start: to + 1, first, ..r });
451        ranges.extend(old.iter().filter(|r| r.start > to + 1).cloned());
452    }
453    // Drop ranges that only continue the previous one.
454    let mut merged: Vec<LabelRange> = Vec::new();
455    let mut prefix_left = MAX_PREFIX_TOTAL_BYTES;
456    for r in ranges {
457        push_merged_range(&mut merged, &mut prefix_left, r)?;
458    }
459    // Plain 1, 2, 3 everywhere is what a document without labels shows: remove them.
460    if merged.len() == 1 && merged[0] == default_range() {
461        merged.clear();
462    }
463    set_page_label_ranges(doc, &merged)
464}

466fn push_merged_range(merged: &mut Vec<LabelRange>, prefix_left: &mut usize, r: LabelRange) -> Result<(), OrganizeError> {
467    let continues = merged.last().is_some_and(|p| {
468        p.style == r.style
469            && p.prefix == r.prefix
470            && match r.style {
471                LabelStyle::None => p.first == r.first,
472                _ => r.start.checked_sub(p.start).and_then(|n| u32::try_from(n).ok()).and_then(|n| p.first.checked_add(n)) == Some(r.first),
473            }
474    });
475    if !continues {
476        if merged.len() + 1 >= MAX_LABEL_TREE_WORK {
477            return Err(invalid_label("the number tree has too many entries"));
478        }
479        *prefix_left = prefix_left.checked_sub(r.prefix.len()).ok_or_else(|| invalid_label("decoded prefixes exceed 4 MiB in total"))?;
480        merged.push(r);
481    }
482    Ok(())
483}

490/// Whether `set_page_label_ranges` can store `r` while it covers `r.start..end` (`end` exclusive).
491/// Entry-count and aggregate-prefix limits stay hard errors in [`push_merged_range`].
492fn range_writable(r: &LabelRange, end: usize) -> bool {
493    r.label(r.start).is_ok() && check_spans(std::slice::from_ref(r), end).is_ok()
494}

497/// `dst_start` in an output document. Sets `has_labels` when `src` has `/PageLabels`.
498pub(crate) fn collect_page_labels(
499    src: &Document,
500    pages: &[usize],
501    dst_start: usize,
502    out: &mut Vec<LabelRange>,
503    has_labels: &mut bool,
504) -> Result<(), OrganizeError> {
505    if pages.is_empty() {
506        return Ok(());
507    }
508    let mut src_ranges = checked_ranges(src).unwrap_or_default();
509    if !src_ranges.is_empty() {
510        *has_labels = true;
511    }
512    for r in &mut src_ranges {
513        // An empty prefix-only range shows each page's physical number in the source.
514        if r.style == LabelStyle::None && r.prefix.is_empty() {
515            r.style = LabelStyle::Decimal;
516            r.first = u32::try_from(r.start.saturating_add(1)).unwrap_or(1);
517        }
518    }
519    if src_ranges.first().is_none_or(|r| r.start > 0) {
520        src_ranges.insert(0, default_range());
521    }
522    let mut prefix_left = MAX_PREFIX_TOTAL_BYTES.saturating_sub(out.iter().map(|r| r.prefix.len()).sum::<usize>());
523    let mut i = 0;
524    while i < pages.len() {
525        let src_start = pages[i];
526        let mut len = 1;
527        while i + len < pages.len() && src_start.checked_add(len) == Some(pages[i + len]) {
528            len += 1;
529        }
530        let run_dst = dst_start + i;
531        let src_end = src_start.saturating_add(len);
532        let run_end = run_dst.saturating_add(len);
533        let idx = src_ranges.partition_point(|r| r.start <= src_start).saturating_sub(1);
534        let span_end = |following: Option<usize>| {
535            following
536                .filter(|start| *start > src_start && *start < src_end)
537                .and_then(|start| start.checked_sub(src_start).and_then(|delta| run_dst.checked_add(delta)))
538                .unwrap_or(run_end)
539        };
540        if let Some(first_range) = src_ranges.get(idx) {
541            let mapped = match first_range.number(src_start) {
542                Ok(first) => LabelRange { start: run_dst, style: first_range.style, prefix: first_range.prefix.clone(), first },
543                Err(_) => source_physical_range(run_dst, src_start),
544            };
545            let mapped = if range_writable(&mapped, span_end(src_ranges.get(idx + 1).map(|r| r.start))) {
546                mapped
547            } else {
548                source_physical_range(run_dst, src_start)
549            };
550            push_merged_range(out, &mut prefix_left, mapped)?;
551        }
552        let tail = src_ranges.get(idx + 1..).unwrap_or_default();
553        for (n, r) in tail.iter().enumerate() {
554            if r.start >= src_end {
555                break;
556            }
557            let Some(delta) = r.start.checked_sub(src_start) else { continue };
558            let Some(start) = run_dst.checked_add(delta) else { continue };
559            let inner = LabelRange { start, style: r.style, prefix: r.prefix.clone(), first: r.first };
560            let following = tail.get(n + 1).map(|next| next.start);
561            let inner = if range_writable(&inner, span_end(following)) { inner } else { source_physical_range(start, r.start) };
562            push_merged_range(out, &mut prefix_left, inner)?;
563        }
564        i += len;
565    }
566    Ok(())
567}
568
569/// Write collected page label ranges to `dst` when at least one source carried `/PageLabels`.
570pub(crate) fn finish_page_labels(dst: &mut Document, ranges: &[LabelRange], has_labels: bool) -> Result<(), OrganizeError> {
571    if has_labels && !ranges.is_empty() {
572        set_page_label_ranges(dst, ranges)?;
573    }
574    Ok(())
575}

581    /// No limit on the whole sequence: 4,097 labels of 1,024 bytes (over 4 MiB in total) are all
582    /// readable, one at a time or by page.
583    #[test]
584    fn label_view_has_no_aggregate_limit() {
585        let range = LabelRange { start: 0, style: LabelStyle::None, prefix: "x".repeat(MAX_LABEL_BYTES), first: u32::MAX };
586        let view = PageLabels { ranges: vec![range], pages: 4_097 };
587        assert_eq!(view.iter().map(|label| label.len()).sum::<usize>(), 4_097 * MAX_LABEL_BYTES);

591    /// The view keeps the ranges, not one label per page: a billion-page document costs no more to
592    /// view than a short one, and a late label is built without the labels before it.
593    #[test]
594    fn label_view_does_not_grow_with_the_page_count() {
595        let prefix = "x".repeat(1000);
596        let range = LabelRange { start: 0, style: LabelStyle::Decimal, prefix: prefix.clone(), first: 1 };
597        let view = PageLabels { ranges: vec![range], pages: 1_000_000_000 };
598        assert_eq!(view.label(999_999_999).unwrap(), format!("{prefix}1000000000"));
599        assert_eq!(view.iter().take(2).collect::<Vec<_>>(), [format!("{prefix}1"), format!("{prefix}2")]);
600        assert_eq!(view.label(1_000_000_000), Err(OrganizeError::NoSuchPage(1_000_000_000)));
601    }

624    /// The edit check accepts exactly the ranges whose every label the per-page check accepts,
625    /// including spans that cross a thousand, numbers past 32 bits and long prefixes.
626    #[test]
627    fn span_check_agrees_with_per_label_checks() {
628        let styles = [LabelStyle::None, LabelStyle::Decimal, LabelStyle::UpperRoman, LabelStyle::LowerAlpha];
629        let firsts = [1, 3, 990, 999, 1_012_880, 1_013_990, 1_014_000, u32::MAX - 30, u32::MAX];
630        let prefixes = [0, 500, 1000, 1010, 1020, 1024];
631        let mut seed = 0x9e37_79b9_7f4a_7c15u64;
632        let mut next = |bound: usize| {
633            seed = seed.wrapping_mul(6_364_136_223_846_793_005).wrapping_add(1_442_695_040_888_963_407);
634            usize::try_from((seed >> 33) % bound as u64).unwrap_or_default()
635        };
636        for _ in 0..3_000 {
637            let pages = next(60);
638            let count = 1 + next(3);
639            let mut starts: Vec<usize> = (0..count).map(|_| next(pages + 3)).collect();
640            starts.sort_unstable();
641            starts.dedup();
642            let mut ranges = Vec::new();

649            let per_label = (0..pages).all(|p| ranges.iter().rev().find(|r| r.start <= p).is_none_or(|r| r.label(p).is_ok()));
650            assert_eq!(check_spans(&ranges, pages).is_ok(), per_label, "{ranges:?} over {pages} pages");
651        }
