1//! Where a link annotation is active, and where its URI action goes.
2//!
3//! - ISO 32000-2 §12.5.6.5, Table 176: a link's `/QuadPoints` are "the region in which the link
4//!   should be activated"; when they are absent, not understood, or any coordinate lies outside
5//!   `/Rect`, the activation region is `/Rect`.
6//! - ISO 32000-2 §12.6.4.8, Table 211: the catalog's `/URI /Base` is "the base URI that shall be
7//!   used in resolving relative URI references" in URI actions.
8
9/// The most quadrilaterals a link's `/QuadPoints` may give before they are not used (the link is
10/// then active over its `/Rect`, as for an array that isn't understood).
11pub const MAX_LINK_QUADS: usize = 1024;
12
13/// The longest `/URI /Base` that is used. A longer base would only produce an address the link
14/// check refuses as too long, so it is ignored.
15pub const MAX_URI_BASE_BYTES: usize = 8192;
16
17/// How far (in points) a quadrilateral's corner may lie outside `/Rect` and still count as inside:
18/// rounding in the writer, not a different region.
19pub(crate) const RECT_SLACK: f32 = 0.01;
20
21/// A link's quadrilaterals from its `/QuadPoints` values (`None` for an entry that isn't a
22/// number), or an empty list when the link is active over its whole `/Rect`: no values, a count
23/// that isn't a multiple of 8, more than [`MAX_LINK_QUADS`], a value that isn't a finite number,
24/// a corner outside `rect` (normalized, `[x0, y0, x1, y1]`), or no quadrilateral that encloses
25/// an area.
26///
27/// A quadrilateral no thicker than [`RECT_SLACK`] (its corners on one line, give or take
28/// rounding) is left out: it adds nothing to the region the others make, and kept alone it would
29/// leave the link active nowhere. When every quadrilateral is like that, the array is not one
30/// that can be used and the link is active over `/Rect`.
31pub fn link_quads(values: &[Option<f32>], rect: [f32; 4]) -> Vec<[f32; 8]> {
32    if values.is_empty() || !values.len().is_multiple_of(8) || values.len() / 8 > MAX_LINK_QUADS {
33        return Vec::new();
34    }
35    let mut numbers = Vec::with_capacity(values.len());
36    for v in values {
37        match v {
38            Some(n) if n.is_finite() => numbers.push(*n),
39            _ => return Vec::new(),
40        }
41    }
42    let [x0, y0, x1, y1] = rect;
43    let inside = numbers
44        .as_chunks::<2>()
45        .0
46        .iter()
47        .all(|[x, y]| *x >= x0 - RECT_SLACK && *x <= x1 + RECT_SLACK && *y >= y0 - RECT_SLACK && *y <= y1 + RECT_SLACK);
48    if !inside {
49        return Vec::new();
50    }
51    numbers.as_chunks::<8>().0.iter().filter(|q| encloses_area(q)).copied().collect()
52}
53
54/// Whether the quadrilateral `q` (`x1 y1 … x4 y4`, finite) is thicker than [`RECT_SLACK`]: one of
55/// the four triangles its corners make (whose union [`quad_contains`] tests) is more than that
56/// high over its longest side. Worked in `f64`, where products of finite `f32` values neither
57/// overflow nor become NaN.
58fn encloses_area(q: &[f32; 8]) -> bool {
59    let p = [[q[0], q[1]], [q[2], q[3]], [q[4], q[5]], [q[6], q[7]]].map(|[x, y]| [f64::from(x), f64::from(y)]);
60    let len = |u: [f64; 2], v: [f64; 2]| (v[0] - u[0]).hypot(v[1] - u[1]);
61    [[0, 1, 2], [0, 1, 3], [0, 2, 3], [1, 2, 3]].iter().any(|&[a, b, c]| match (p.get(a), p.get(b), p.get(c)) {
62        (Some(&a), Some(&b), Some(&c)) => {
63            // Twice the triangle's area is its longest side times its height over that side.
64            let twice_area = ((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])).abs();
65            let longest = len(a, b).max(len(b, c)).max(len(c, a));
66            twice_area > f64::from(RECT_SLACK) * longest
67        }
68        _ => false,
69    })
70}

72/// Whether the point `(x, y)` lies in the quadrilateral `q` (`x1 y1 … x4 y4`), edges included.
73/// Writers disagree on the corners' order (the specification says counterclockwise, text-markup
74/// writers often go top-left, top-right, bottom-left, bottom-right), so the quadrilateral is taken
75/// as the area its four corners enclose: the union of the four triangles they make.
76pub fn quad_contains(q: &[f32; 8], x: f32, y: f32) -> bool {
77    if !(x.is_finite() && y.is_finite()) {
78        return false;
79    }
80    let p = [[q[0], q[1]], [q[2], q[3]], [q[4], q[5]], [q[6], q[7]]];
81    [[0, 1, 2], [0, 1, 3], [0, 2, 3], [1, 2, 3]].iter().any(|&[a, b, c]| match (p.get(a), p.get(b), p.get(c)) {
82        (Some(a), Some(b), Some(c)) => triangle_contains(*a, *b, *c, [x, y]),
83        _ => false,
84    })
85}

87/// Worked in `f64`, like [`encloses_area`]: in `f32` the cross products of far-apart finite
88/// corners overflow to infinity and their differences to NaN, which no sign test rejects.
89fn triangle_contains(a: [f32; 2], b: [f32; 2], c: [f32; 2], p: [f32; 2]) -> bool {
90    let [a, b, c, p] = [a, b, c, p].map(|[x, y]| [f64::from(x), f64::from(y)]);
91    let cross = |o: [f64; 2], u: [f64; 2], v: [f64; 2]| (u[0] - o[0]) * (v[1] - o[1]) - (u[1] - o[1]) * (v[0] - o[0]);
92    // A triangle with no area holds no point (it would otherwise hold its whole line).
93    if cross(a, b, c) == 0.0 {
94        return false;
95    }
96    let (d1, d2, d3) = (cross(a, b, p), cross(b, c, p), cross(c, a, p));
97    let negative = d1 < 0.0 || d2 < 0.0 || d3 < 0.0;
98    let positive = d1 > 0.0 || d2 > 0.0 || d3 > 0.0;
99    !(negative && positive)
100}

102/// Whether `s` begins with an RFC 3986 scheme (`ALPHA *( ALPHA / DIGIT / "+" / "-" / "." ) ":"`),
103/// so it is an absolute URI rather than a relative reference.
104fn has_scheme(s: &str) -> bool {
105    s.split_once(':').is_some_and(|(scheme, _)| {
106        scheme.chars().next().is_some_and(|c| c.is_ascii_alphabetic())
107            && scheme.chars().all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
108    })
109}

111/// The address a URI action goes to when the document's catalog gives `base` (`/URI /Base`):
112/// a relative reference (`page.html`, `../a/`, `?q=1`, `//host/x`) resolved against it by
113/// RFC 3986 §5 (the WHATWG URL rules, which the browser applies too). `None` when the address is
114/// used as written: it already has a scheme (`https:`, `mailto:`, `javascript:`…), there is no
115/// usable base, or the result wouldn't be a web address.
116///
117/// Only a web (`http`, `https`) base is used, and the result must keep the base's scheme, so a
118/// base never turns a relative link into a `file:`, `javascript:` or other address. An address
119/// with control characters or whitespace other than a plain space is left as written, since the
120/// URL rules would drop them and read what is left (`java\nscript:`) differently. Either way the
121/// address still goes through the same check and confirmation as any other link.
122pub fn resolve_uri(base: Option<&str>, raw: &str) -> Option<String> {
123    let reference = raw.trim();
124    let base = base?.trim();
125    if reference.is_empty() || base.len() > MAX_URI_BASE_BYTES || has_scheme(reference) {
126        return None;
127    }
128    let hidden = |s: &str| s.chars().any(|c| c.is_control() || (c.is_whitespace() && c != ' '));
129    // The base must name its host as written (`https://host/…`): the URL rules would also accept
130    // `https:///host/` or `https:host/`, which no link check reads as a web address.
131    let names_host = base.split_once("://").is_some_and(|(_, rest)| !rest.starts_with(['/', '\\']));
132    if hidden(reference) || hidden(base) || !names_host {
133        return None;
134    }
135    let base = url::Url::parse(base).ok()?;
136    if !matches!(base.scheme(), "http" | "https") || base.host().is_none() {
137        return None;
138    }
139    let resolved = base.join(reference).ok()?;
140    (resolved.scheme() == base.scheme() && resolved.host().is_some()).then(|| resolved.to_string())
141}

149    #[test]
150    fn relative_references_resolve_against_the_base() {
151        let r = |raw: &str| resolve_uri(Some(BASE), raw);
152        assert_eq!(r("page.html").as_deref(), Some("https://example.com/base/page.html"));
153        assert_eq!(r("  page.html ").as_deref(), Some("https://example.com/base/page.html"), "trimmed like any link");
154        assert_eq!(r("../up.html").as_deref(), Some("https://example.com/up.html"));
155        assert_eq!(r("/root.html").as_deref(), Some("https://example.com/root.html"));
156        assert_eq!(r("?q=1").as_deref(), Some("https://example.com/base/?q=1"));
157        assert_eq!(r("#part").as_deref(), Some("https://example.com/base/#part"));
158        assert_eq!(r("a/b:c").as_deref(), Some("https://example.com/base/a/b:c"), "a colon after a slash is no scheme");
159        // A network-path reference names its own host; the confirmation shows that host.
160        assert_eq!(r("//other.example/x").as_deref(), Some("https://other.example/x"));
161        assert_eq!(resolve_uri(Some("http://example.com/dir/file.pdf"), "next.pdf").as_deref(), Some("http://example.com/dir/next.pdf"));
162    }

213    #[test]
214    fn quads_that_are_not_understood_fall_back_to_the_rect() {
215        let rect = [10.0, 10.0, 200.0, 30.0];
216        let good = [10.0, 10.0, 60.0, 10.0, 60.0, 30.0, 10.0, 30.0];
217        assert!(link_quads(&[], rect).is_empty());
218        assert!(link_quads(&good[..7].iter().copied().map(Some).collect::<Vec<_>>(), rect).is_empty(), "not a multiple of 8");
219        let mut odd: Vec<Option<f32>> = good.map(Some).to_vec();
220        odd.push(Some(1.0));
221        assert!(link_quads(&odd, rect).is_empty());
222        let mut not_number = good.map(Some);
223        not_number[3] = None;
224        assert!(link_quads(&not_number, rect).is_empty());
225        let mut nan = good.map(Some);
226        nan[0] = Some(f32::NAN);
227        assert!(link_quads(&nan, rect).is_empty());
228        // A /Rect too large for f32 would hold an infinite corner.
229        let mut infinite = good.map(Some);
230        infinite[2] = Some(f32::INFINITY);
231        assert!(link_quads(&infinite, [f32::NEG_INFINITY, f32::NEG_INFINITY, f32::INFINITY, f32::INFINITY]).is_empty());
232        let mut outside = good.map(Some);
233        outside[2] = Some(250.0);
234        assert!(link_quads(&outside, rect).is_empty(), "a corner outside /Rect");
235        let mut below = good.map(Some);
236        below[1] = Some(9.0);
237        assert!(link_quads(&below, rect).is_empty(), "a corner below /Rect");
238        let too_many: Vec<Option<f32>> = std::iter::repeat_n(good.map(Some), MAX_LINK_QUADS + 1).flatten().collect();
239        assert!(link_quads(&too_many, rect).is_empty());
240        let most: Vec<Option<f32>> = std::iter::repeat_n(good.map(Some), MAX_LINK_QUADS).flatten().collect();
241        assert_eq!(link_quads(&most, rect).len(), MAX_LINK_QUADS);
242    }

244    #[test]
245    fn quads_with_no_area_are_left_out() {
246        let rect = [10.0, 10.0, 200.0, 30.0];
247        let good = [10.0, 10.0, 60.0, 10.0, 60.0, 30.0, 10.0, 30.0];
248        // Corners on one line, all at one point, and a sliver thinner than the slack.
249        let line = [10.0, 20.0, 60.0, 20.0, 60.0, 20.0, 10.0, 20.0];
250        let point = [30.0, 20.0, 30.0, 20.0, 30.0, 20.0, 30.0, 20.0];
251        let sliver = [10.0, 20.0, 60.0, 20.0, 60.0, 20.005, 10.0, 20.005];
252        for flat in [line, point, sliver] {
253            assert!(link_quads(&flat.map(Some), rect).is_empty(), "{flat:?} alone: active over /Rect");
254            let mixed: Vec<Option<f32>> = flat.iter().chain(good.iter()).map(|v| Some(*v)).collect();
255            assert_eq!(link_quads(&mixed, rect), vec![good], "{flat:?} beside a usable quad");
256        }
257        // Thicker than the slack is a region, however thin.
258        let thin = [10.0, 20.0, 60.0, 20.0, 60.0, 20.05, 10.0, 20.05];
259        assert_eq!(link_quads(&thin.map(Some), rect), vec![thin]);
260        // A triangle (two corners shared) still encloses an area.
261        let triangle = [10.0, 10.0, 60.0, 10.0, 60.0, 30.0, 60.0, 30.0];
262        assert_eq!(link_quads(&triangle.map(Some), rect), vec![triangle]);
263        // The largest finite values neither overflow nor turn to NaN.
264        let all = [f32::MIN, f32::MIN, f32::MAX, f32::MAX];
265        let huge = [f32::MIN, f32::MIN, f32::MAX, f32::MIN, f32::MAX, f32::MAX, f32::MIN, f32::MAX];
266        assert_eq!(link_quads(&huge.map(Some), all), vec![huge]);
267        let huge_line = [f32::MIN, f32::MIN, f32::MAX, f32::MAX, f32::MAX, f32::MAX, f32::MIN, f32::MIN];
268        assert!(link_quads(&huge_line.map(Some), all).is_empty());
269    }

271    #[test]
272    fn a_quad_holds_its_area_in_any_corner_order() {
273        // Counterclockwise from the bottom left, as the specification writes them.
274        let ccw = [10.0, 10.0, 60.0, 10.0, 60.0, 30.0, 10.0, 30.0];
275        // Top-left, top-right, bottom-left, bottom-right, as text-markup writers often do.
276        let z = [10.0, 30.0, 60.0, 30.0, 10.0, 10.0, 60.0, 10.0];
277        for q in [ccw, z] {
278            assert!(quad_contains(&q, 30.0, 20.0));
279            assert!(quad_contains(&q, 10.0, 10.0), "corners count");
280            assert!(quad_contains(&q, 60.0, 20.0), "edges count");
281            assert!(!quad_contains(&q, 100.0, 20.0), "right of the quad");
282            assert!(!quad_contains(&q, 30.0, 31.0), "above the quad");
283            assert!(!quad_contains(&q, 9.0, 20.0), "left of the quad");
284            assert!(!quad_contains(&q, 30.0, 9.0), "below the quad");
285        }
286        // A tilted quad: the corner of its bounding box is outside it.
287        let diamond = [50.0, 0.0, 100.0, 50.0, 50.0, 100.0, 0.0, 50.0];
288        assert!(quad_contains(&diamond, 50.0, 50.0));
289        assert!(!quad_contains(&diamond, 5.0, 5.0));
290        // A quad with no area holds nothing.
291        let flat = [0.0, 0.0, 10.0, 0.0, 20.0, 0.0, 30.0, 0.0];
292        assert!(!quad_contains(&flat, 5.0, 0.0));
293        assert!(!quad_contains(&ccw, f32::NAN, 20.0), "a point that isn't a number is nowhere");
294    }

296    #[test]
297    fn a_quad_with_extreme_corners_holds_only_its_area() {
298        // In f32 the cross products overflow (inf, and inf - inf = NaN), which passed every sign
299        // test and put (1.5e20, 1.5e20), outside the triangle, inside it.
300        let q = [0.0, 0.0, 2e20, 0.0, 0.0, 2e20, 0.0, 0.0];
301        assert_eq!(link_quads(&q.map(Some), [0.0, 0.0, 3e20, 3e20]), vec![q], "kept by the area check");
302        assert!(!quad_contains(&q, 1.5e20, 1.5e20), "beyond the long edge");
303        assert!(!quad_contains(&q, 3e20, 1.0), "right of the triangle");
304        assert!(quad_contains(&q, 0.5e20, 0.5e20));
305        assert!(quad_contains(&q, 1.0, 1.0));
306        let max = [0.0, 0.0, f32::MAX, 0.0, 0.0, f32::MAX, 0.0, 0.0];
307        assert!(!quad_contains(&max, f32::MAX, f32::MAX));
308        assert!(quad_contains(&max, 1.0, 1.0));
