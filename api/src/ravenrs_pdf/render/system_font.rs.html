1//! Installed CJK faces for CID fonts a PDF doesn't embed.
2//!
3//! Acrobat and other tools often write Chinese, Japanese or Korean text with one of the Adobe
4//! collections (Adobe-Japan1, GB1, CNS1, Korea1) and a font name such as `KozGoPr6N-Medium` or
5//! `MSung-Light` without embedding the font. hayro maps those CIDs to Unicode itself, but it needs
6//! a face to draw them with; without one the text is blank. The craft-fonts Japanese faces come
7//! first (see `japanese_fallback` in `raster.rs`); this module is the last resort: one face
8//! already installed on this machine, used for the viewer only (`RenderConfig::system_fonts`, off
9//! for exports, printing and OCR). It is read at runtime and never embedded or shipped
10//! (AGENTS.md §1.4), Noto CJK and Source Han are left out on purpose (§1.1), and
11//! `PDFCRAFT_SYSTEM_FONTS=0` turns it off, as for the interface font.
12

23/// Larger files are not read: a font path is still untrusted input. CJK system faces are big
24/// collections (PingFang is tens of megabytes).
25const MAX_BYTES: u64 = 96 << 20;
26/// Faces tried in a collection (`.ttc`).
27const MAX_FACES: u32 = 32;
28/// Downloaded font assets looked at per kind on macOS.
29const MAX_ASSETS: usize = 512;
30
31/// The writing system a character collection is for.
32#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
33enum Script {
34    Japanese,
35    SimplifiedChinese,
36    TraditionalChinese,
37    Korean,
38}
39
40impl Script {
41    /// Map the four supported Adobe CJK collections to their writing systems. Identity and
42    /// custom collections provide no regional hint and cannot select an installed face here.
43    fn of(family: &CidFamily) -> Option<Self> {
44        match family {
45            CidFamily::AdobeJapan1 => Some(Self::Japanese),
46            CidFamily::AdobeGB1 => Some(Self::SimplifiedChinese),
47            CidFamily::AdobeCNS1 => Some(Self::TraditionalChinese),
48            CidFamily::AdobeKorea1 => Some(Self::Korean),
49            CidFamily::AdobeIdentity | CidFamily::Custom { .. } => None,
50        }
51    }
52
53    /// A character the face must have to be worth using. Each is common and not shared by the
54    /// other scripts' faces: 物 (Japanese; also Chinese, so it only rules out Latin-only faces),
55    /// 欢 (Simplified only), 說 (Traditional only), 한 (Hangul).
56    fn probe(self) -> char {
57        match self {
58            Self::Japanese => '\u{7269}',
59            Self::SimplifiedChinese => '\u{6B22}',
60            Self::TraditionalChinese => '\u{8AAA}',
61            Self::Korean => '\u{D55C}',
62        }
63    }
64}

66/// One installed file and the words (in a face's family name) that pick the face for the right
67/// region out of a collection that holds several (PingFang SC, TC and HK share one file).
68struct Candidate {
69    path: PathBuf,
70    region: &'static [&'static str],
71}
72
73/// Keep named Windows body, heading, calligraphic and FangSong faces distinct. A cache keyed only
74/// by script and weight would make the first Chinese font on a page stand in for all of them.
75#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
76enum ChineseFamily {
77    Default,
78    Song,
79    Hei,
80    Kai,
81    FangSong,
82}
83
84impl ChineseFamily {
85    /// The family a font name asks for: the Windows and Founder (方正) names that Chinese PDFs,
86    /// CNKI papers in particular, use without embedding the font. FangSong is tested first, as
87    /// its names contain "song" too. Founder PostScript names are matched by prefix (`FZSSK--GBK1-0`
88    /// is 方正书宋, `FZXBSK--GBK1-0` 方正小标宋, `FZHTK--GBK1-0` 方正黑体, `FZKTK--GBK1-0` 方正楷体,
89    /// `FZFSK--GBK1-0` 方正仿宋).
90    fn of(name: &str) -> Self {
91        // PDF subset names prepend six uppercase ASCII letters and '+'. Match the actual
92        // PostScript name so Founder prefixes work for subsets too.
93        let name = match name.split_once('+') {
94            Some((tag, family)) if tag.len() == 6 && tag.bytes().all(|b| b.is_ascii_uppercase()) => family,
95            _ => name,
96        };
97        let name = name.to_ascii_lowercase();
98        let has = |hints: &[&str]| hints.iter().any(|hint| name.contains(hint));
99        let starts = |prefixes: &[&str]| prefixes.iter().any(|p| name.starts_with(p));
100        if has(&["fangsong", "simfang", "仿宋"]) || starts(&["fzfs"]) {
101            Self::FangSong
102        } else if has(&["simsun", "stsong", "songti", "song", "宋"]) || starts(&["fzss", "fzbs", "fzxbs", "fzcs"]) {
103            Self::Song
104        } else if has(&["simhei", "stheiti", "stxihei", "heiti", "黑"]) || starts(&["fzht", "fzhei", "fzlth", "fzdh"]) {
105            Self::Hei
106        } else if has(&["kaiti", "stkaiti", "kai", "楷"]) || starts(&["fzkt", "fzkai"]) {
107            Self::Kai
108        } else {
109            Self::Default
110        }
111    }
112}

114/// The installed face to draw a CID font of `query`'s collection with, as `(data, face index)`.
115/// `None` when it is turned off, the collection isn't one of the four, or nothing fits.
116pub(crate) fn fallback(query: &FallbackFontQuery) -> Option<(FontData, u32)> {
117    if cfg!(target_arch = "wasm32") || std::env::var_os("PDFCRAFT_SYSTEM_FONTS").is_some_and(|v| v == "0") {
118        return None;
119    }
120    let script = Script::of(&query.character_collection.as_ref()?.family)?;
121    let bold = query.is_bold || query.font_weight >= 600;
122    let family = ChineseFamily::of(query.post_script_name.as_deref().unwrap_or_default());
123    /// A face file and the index of the chosen face in it, or `None` when nothing fits.
124    type Face = Option<(Arc<Vec<u8>>, u32)>;
125    type Faces = HashMap<(Script, bool, ChineseFamily), Face>;
126    static CACHE: Mutex<Option<Faces>> = Mutex::new(None);
127    let mut cache = CACHE.lock().unwrap_or_else(PoisonError::into_inner);
128    let found = cache.get_or_insert_with(HashMap::new).entry((script, bold, family)).or_insert_with(|| {
129        candidates(script, family, bold).iter().find_map(|c| {
130            read(c, script.probe(), bold).inspect(|(_, index)| {
131                log::info!("CJK font fallback for {script:?}/{family:?}: {} (face {index})", c.path.display());
132            })
133        })
134    });
135    found.as_ref().map(|(bytes, index)| (Arc::clone(bytes) as FontData, *index))
136}

138/// Read a candidate through the bounded file cache and select a face that maps `probe`,
139/// preferring the requested region and weight. Unreadable files and unsuitable faces return
140/// `None`, allowing the caller to try the next candidate without changing the PDF.
141fn read(candidate: &Candidate, probe: char, bold: bool) -> Option<(Arc<Vec<u8>>, u32)> {
142    let bytes = file_bytes(&candidate.path)?;
143    let index = pick_face(&bytes, probe, candidate.region, bold)?;
144    Some((bytes, index))
145}

147/// The bytes of an installed font file, read once for the whole process and shared: a collection
148/// such as `PingFang.ttc` is tens of megabytes and is the candidate of several scripts and both
149/// weights, so the per-script cache keeps only a handle and a face index. A file that cannot be
150/// used is remembered too, so it is not tried again for every script or family.
151fn file_bytes(path: &Path) -> Option<Arc<Vec<u8>>> {
152    /// The bytes of each file tried, or `None` for one that cannot be used.
153    type Files = HashMap<PathBuf, Option<Arc<Vec<u8>>>>;
154    static FILES: Mutex<Option<Files>> = Mutex::new(None);
155    let mut files = FILES.lock().unwrap_or_else(PoisonError::into_inner);
156    files.get_or_insert_with(HashMap::new).entry(path.to_path_buf()).or_insert_with(|| read_file(path, MAX_BYTES).map(Arc::new)).clone()
157}

159/// A regular file of at most `max` bytes, or `None`: a font path is untrusted input.
160fn read_file(path: &Path, max: u64) -> Option<Vec<u8>> {
161    // Reject special files before opening them (in particular a FIFO can block on open).
162    let meta = std::fs::metadata(path).ok()?;
163    if !meta.is_file() || meta.len() > max {
164        return None;
165    }
166    let file = std::fs::File::open(path).ok()?;
167    let meta = file.metadata().ok()?;
168    if !meta.is_file() || meta.len() > max {
169        return None;
170    }
171    // Bound the actual read too: the file can grow after the metadata check.
172    let mut bytes = Vec::new();
173    file.take(max.saturating_add(1)).read_to_end(&mut bytes).ok()?;
174    (bytes.len() as u64 <= max).then_some(bytes)
175}

177/// The best face of the file that parses and maps `probe`: the region's face when the file has
178/// one, then the regular (or, for bold text, the semibold or bold) weight, then the first.
179fn pick_face(bytes: &[u8], probe: char, region: &[&str], bold: bool) -> Option<u32> {
180    let mut best: Option<(u32, u32)> = None;
181    for index in 0..MAX_FACES {
182        let Ok(font) = skrifa::FontRef::from_index(bytes, index) else { continue };
183        if font.charmap().map(probe).is_none() {
184            continue;
185        }
186        let names = |id: StringId| font.localized_strings(id).map(|s| s.chars().collect::<String>()).collect::<Vec<_>>();
187        let family = names(StringId::FAMILY_NAME).join(" ");
188        let style = names(StringId::SUBFAMILY_NAME).join(" ").to_ascii_lowercase();
189        let mut score = 1;
190        if region.iter().any(|r| family.split_whitespace().any(|w| w == *r)) {
191            score += 4;
192        }
193        let weighty = style.contains("bold") || style.contains("semibold") || style.contains("heavy");
194        let plain = style.is_empty() || style.contains("regular") || style.contains("normal") || style.contains("medium") && !bold;
195        if (bold && weighty) || (!bold && plain && !weighty) {
196            score += 2;
197        }
198        if best.is_none_or(|(_, s)| score > s) {
199            best = Some((index, score));
200        }
201    }
202    best.map(|(index, _)| index)
203}

205/// List installed-font paths in preference order, with regional face-name hints for collections.
206/// Windows Simplified Chinese first tries the named document family; its generic list also
207/// prefers a bold face for bold requests. macOS includes bounded downloaded-font locations.
208/// Only fixed platform filenames are used: a PDF font name never becomes a filesystem path.
209fn candidates(script: Script, family: ChineseFamily, bold: bool) -> Vec<Candidate> {
210    let region: &'static [&'static str] = match script {
211        Script::Japanese => &["JP", "Japanese"],
212        Script::SimplifiedChinese => &["SC", "GB"],
213        Script::TraditionalChinese => &["TC", "HK", "CNS"],
214        Script::Korean => &["KR", "Korean"],
215    };
216    let files: Vec<PathBuf> = if cfg!(windows) {
217        let dir = windows_fonts();
218        let names: &[&str] = match script {
219            Script::Japanese => &["YuGothR.ttc", "meiryo.ttc", "msgothic.ttc"],
220            Script::SimplifiedChinese => match family {
221                ChineseFamily::Song => &["simsun.ttc", "msyh.ttc", "msjh.ttc"],
222                ChineseFamily::Hei => &["simhei.ttf", "msyh.ttc", "simsun.ttc", "msjh.ttc"],
223                ChineseFamily::Kai => &["simkai.ttf", "simsun.ttc", "msyh.ttc", "msjh.ttc"],
224                ChineseFamily::FangSong => &["simfang.ttf", "simsun.ttc", "msyh.ttc", "msjh.ttc"],
225                ChineseFamily::Default if bold => &["msyhbd.ttc", "msyh.ttc", "simsun.ttc", "msjh.ttc"],
226                ChineseFamily::Default => &["msyh.ttc", "simsun.ttc", "msjh.ttc"],
227            },
228            Script::TraditionalChinese => &["msjh.ttc", "mingliu.ttc", "msyh.ttc"],
229            Script::Korean => &["malgun.ttf", "gulim.ttc"],
230        };
231        names.iter().map(|f| dir.join(f)).collect()
232    } else if cfg!(target_os = "macos") {
233        let names: &[&str] = match script {
234            Script::Japanese => &["ヒラギノ角ゴシック W3.ttc", "Hiragino Sans GB.ttc", "PingFang.ttc"],
235            Script::SimplifiedChinese => &["PingFang.ttc", "Hiragino Sans GB.ttc", "STHeiti Light.ttc"],
236            Script::TraditionalChinese => &["PingFang.ttc", "STHeiti Light.ttc", "Hiragino Sans GB.ttc"],
237            Script::Korean => &["AppleSDGothicNeo.ttc", "AppleGothic.ttf", "PingFang.ttc"],
238        };
239        // The system folder first, then the downloaded font assets: on recent macOS a face such as
240        // PingFang lives in `/System/Library/AssetsV2/com_apple_MobileAsset_Font*/<id>.asset/AssetData`.
241        let mut dirs = vec![PathBuf::from("/System/Library/Fonts")];
242        dirs.extend(macos_font_asset_dirs().iter().cloned());
243        names.iter().flat_map(|f| dirs.iter().map(move |d| d.join(f))).collect()
244    } else {
245        let names: &[&str] = match script {
246            Script::Korean => {
247                &["truetype/nanum/NanumGothic.ttf", "nanum/NanumGothic.ttf", "truetype/baekmuk/gulim.ttf", "truetype/unfonts-core/UnDotum.ttf"]
248            }
249            Script::Japanese => &[
250                "truetype/fonts-japanese-gothic.ttf",
251                "ipa-gothic/ipag.ttf",
252                "truetype/takao-gothic/TakaoGothic.ttf",
253                "truetype/vlgothic/VL-Gothic-Regular.ttf",
254            ],
255            Script::SimplifiedChinese | Script::TraditionalChinese => &[
256                "truetype/wqy/wqy-microhei.ttc",
257                "wenquanyi/wqy-microhei/wqy-microhei.ttc",
258                "wqy-microhei/wqy-microhei.ttc",
259                "truetype/wqy/wqy-zenhei.ttc",
260                "wenquanyi/wqy-zenhei/wqy-zenhei.ttc",
261                "wqy-zenhei/wqy-zenhei.ttc",
262                "truetype/droid/DroidSansFallbackFull.ttf",
263                "google-droid-sans-fonts/DroidSansFallbackFull.ttf",
264                "truetype/arphic/uming.ttc",
265                "arphic-uming/uming.ttc",
266            ],
267        };
268        ["/usr/share/fonts", "/usr/local/share/fonts"].iter().flat_map(|dir| names.iter().map(move |f| Path::new(dir).join(f))).collect()
269    };
270    files.into_iter().map(|path| Candidate { path, region }).collect()
271}

273/// The `AssetData` folders of the downloaded font assets of macOS, found once. The ids in their
274/// names change with each asset version, so they are listed rather than named; at most
275/// [`MAX_ASSETS`] are looked at.
276fn macos_font_asset_dirs() -> &'static [PathBuf] {
277    static DIRS: OnceLock<Vec<PathBuf>> = OnceLock::new();
278    DIRS.get_or_init(|| {
279        let root = Path::new("/System/Library/AssetsV2");
280        let Ok(kinds) = std::fs::read_dir(root) else { return Vec::new() };
281        let mut dirs = Vec::new();
282        for kind in kinds.flatten().filter(|k| k.file_name().to_string_lossy().starts_with("com_apple_MobileAsset_Font")) {
283            let Ok(assets) = std::fs::read_dir(kind.path()) else { continue };
284            for asset in assets.flatten().take(MAX_ASSETS) {
285                let data = asset.path().join("AssetData");
286                if data.is_dir() {
287                    dirs.push(data);
288                }
289            }
290        }
291        dirs
292    })
293}

295/// Locate the Windows Fonts directory using `WINDIR`, then `SystemRoot`, then `C:\Windows`.
296/// This only constructs the path; candidate reads still enforce regular-file and size limits.
297fn windows_fonts() -> PathBuf {
298    let dir = std::env::var_os("WINDIR").or_else(|| std::env::var_os("SystemRoot")).map_or_else(|| PathBuf::from(r"C:\Windows"), PathBuf::from);
299    dir.join("Fonts")
300}

439            ("SimHei", "SimHei", "simhei.ttf"),
440            ("KaiTi", "KaiTi", "simkai.ttf"),
441            ("FangSong", "FangSong", "simfang.ttf"),
442            ("ABCDEF+FZSSK--GBK1-0", "SimSun", "simsun.ttc"),
443            ("ABCDEF+FZHTK--GBK1-0", "SimHei", "simhei.ttf"),
444            ("ABCDEF+FZKTK--GBK1-0", "KaiTi", "simkai.ttf"),
445            ("ABCDEF+FZFSK--GBK1-0", "FangSong", "simfang.ttf"),
446        ] {
447            if !windows_fonts().join(file).is_file() {
448                continue;
449            }
450            let query = FallbackFontQuery {
451                post_script_name: Some(name.into()),
452                character_collection: Some(CharacterCollection { family: CidFamily::AdobeGB1, supplement: 4 }),
453                ..FallbackFontQuery::default()
454            };
455            let (bytes, index) = fallback(&query).expect("installed Chinese font");
456            let font = skrifa::FontRef::from_index(bytes.as_ref().as_ref(), index).unwrap();
457            let names: Vec<String> = font.localized_strings(StringId::POSTSCRIPT_NAME).map(|n| n.chars().collect()).collect();
458            assert!(names.iter().any(|n| n.eq_ignore_ascii_case(expected)), "{name} resolved to {names:?}, expected {expected}");
459        }
460    }

462    /// On a Mac the face for Traditional Chinese is the TC one of PingFang, not the first of the
463    /// collection (which is another region's), and it maps a character only Traditional has.
464    #[cfg(target_os = "macos")]
465    #[test]
466    fn macos_picks_the_regional_pingfang_face() {
467        // Found where this macOS keeps it: the system folder, or the downloaded font assets.
468        let path = candidates(Script::TraditionalChinese, ChineseFamily::Default, false)
469            .into_iter()
470            .map(|c| c.path)
471            .find(|p| p.file_name().is_some_and(|n| n == "PingFang.ttc") && p.is_file());
472        let Some(bytes) = path.and_then(|p| std::fs::read(p).ok()) else {
473            eprintln!("PingFang is not installed here: skipped");
474            return;
475        };
476        for (script, region) in [(Script::TraditionalChinese, "TC"), (Script::SimplifiedChinese, "SC")] {
477            let index = pick_face(&bytes, script.probe(), &[region], false).expect("PingFang covers it");
478            let font = skrifa::FontRef::from_index(&bytes, index).unwrap();
479            let family: String = font.localized_strings(StringId::FAMILY_NAME).next().unwrap().chars().collect();
480            assert!(family.contains(region), "{script:?} got {family:?}");
481        }

483    /// A face that this macOS keeps only in its downloaded font assets (PingFang, on recent
484    /// versions) is among the candidates. The assets are searched here on their own, not by the
485    /// code under test; where there are none, nothing is checked.
486    #[cfg(target_os = "macos")]
487    #[test]
488    fn macos_candidates_include_a_pingfang_that_lives_in_the_font_assets() {
489        /// Independently locate an installed asset by filename within the test's depth limit,
490        /// providing an oracle for the production candidate list without using its discovery.
491        fn find(dir: &Path, name: &str, depth: u32) -> Option<PathBuf> {
492            for entry in std::fs::read_dir(dir).ok()?.flatten() {
493                let path = entry.path();
494                if path.file_name().is_some_and(|n| n == name) && path.is_file() {
495                    return Some(path);
496                }
497                if depth > 0
498                    && path.is_dir()
499                    && !path.is_symlink()
500                    && let Some(found) = find(&path, name, depth - 1)
501                {
502                    return Some(found);
503                }
504            }
505            None
506        }
507        let Some(asset) = find(Path::new("/System/Library/AssetsV2"), "PingFang.ttc", 3) else {
508            eprintln!("no PingFang in the font assets here: skipped");
509            return;
510        };
511        for script in [Script::SimplifiedChinese, Script::TraditionalChinese] {
512            assert!(
513                candidates(script, ChineseFamily::Default, false).iter().any(|c| c.path == asset),
514                "{script:?}: {} is not a candidate",
515                asset.display()
516            );
517        }
518    }
