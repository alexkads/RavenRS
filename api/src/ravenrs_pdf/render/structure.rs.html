2//! inspector's lopdf object vocabulary while loading only the objects its queries use.

13pub(crate) trait Structure {
14    fn trailer(&self) -> &Dictionary;
15    fn get_object(&self, id: ObjectId) -> Result<&Object>;
16    fn encrypted(&self) -> bool;
17
18    fn dereference<'a>(&'a self, mut object: &'a Object) -> Result<(Option<ObjectId>, &'a Object)> {
19        let mut id = None;
20        for _ in 0..128 {
21            match object {
22                Object::Reference(reference) => {
23                    id = Some(*reference);
24                    object = self.get_object(*reference)?;
25                }
26                _ => return Ok((id, object)),
27            }
28        }
29        Err(Error::ReferenceLimit)
30    }
31
32    fn get_dictionary(&self, id: ObjectId) -> Result<&Dictionary> {
33        self.get_object(id).and_then(|o| self.dereference(o)).and_then(|(_, o)| o.as_dict())
34    }
35
36    fn catalog(&self) -> Result<&Dictionary> {
37        self.trailer().get(b"Root").and_then(Object::as_reference).and_then(|id| self.get_dictionary(id))
38    }
39
40    fn page_ids(&self) -> Vec<ObjectId> {
41        let mut pages = Vec::new();
42        let Ok(root) = self.catalog().and_then(|c| c.get(b"Pages")).and_then(Object::as_reference) else { return pages };
43        let mut pending = vec![(root, 0usize)];
44        let mut seen = HashSet::new();
45        while let Some((id, depth)) = pending.pop() {
46            if depth > 256 || !seen.insert(id) {
47                continue;
48            }
49            let Ok(dict) = self.get_dictionary(id) else { continue };
50            match dict.get_type() {
51                Ok(b"Page") => {
52                    // Keep the effective u32 page-count bound without trusting /Count.
53                    if u32::try_from(pages.len()).ok().and_then(|n| n.checked_add(1)).is_none() {
54                        return pages;
55                    }
56                    pages.push(id);
57                }
58                Ok(b"Pages") => {
59                    if let Ok(kids) = dict.get(b"Kids").and_then(|o| self.dereference(o)).and_then(|(_, o)| o.as_array()) {
60                        pending.extend(kids.iter().rev().filter_map(|k| k.as_reference().ok()).map(|id| (id, depth + 1)));
61                    }
62                }
63                _ => {}
64            }
65        }
66        pages
67    }
68
69    // Match lopdf's font-resource precedence: a page's direct resources, then its
70    // indirect resources and ancestors. Stop cycles and excessive inheritance depth.
71    fn get_page_fonts(&self, page: ObjectId) -> Result<BTreeMap<Vec<u8>, &Dictionary>> {
72        let mut resources = Vec::new();
73        let mut current = Some(page);
74        let mut seen = HashSet::new();
75        while let Some(id) = current {
76            if !seen.insert(id) {
77                return Err(Error::ReferenceCycle(id));
78            }
79            if seen.len() > 100 {
80                return Err(Error::RecursionLimit);
81            }
82            let dict = self.get_dictionary(id)?;
83            if let Ok(resource) = dict.get(b"Resources") {
84                match resource {
85                    Object::Dictionary(dict) if id == page => resources.push(dict),
86                    Object::Reference(id) => {
87                        if let Ok(dict) = self.get_dictionary(*id) {
88                            resources.push(dict);
89                        }
90                    }
91                    _ => {}
92                }
93            }
94            current = dict.get(b"Parent").and_then(Object::as_reference).ok();
95        }
96        let mut fonts = BTreeMap::new();
97        for resources in resources {
98            if let Ok(dict) = resources.get(b"Font").and_then(|o| self.dereference(o)).and_then(|(_, o)| o.as_dict()) {
99                for (name, font) in dict.iter() {
100                    if let Ok(font) = self.dereference(font).and_then(|(_, o)| o.as_dict()) {
101                        fonts.entry(name.clone()).or_insert(font);
102                    }
103                }
104            }
105        }
106        Ok(fonts)
107    }
108}

110impl Structure for crate::third_party::lopdf::Document {
111    fn trailer(&self) -> &Dictionary {
112        &self.trailer
113    }
114    fn get_object(&self, id: ObjectId) -> Result<&Object> {
115        self.get_object(id)
116    }
117    // lopdf's `get_object` already follows an indirect object that holds a reference, so the
118    // trait's walk would report the first object of a chain. lopdf's own walk reports the last.
119    fn dereference<'a>(&'a self, object: &'a Object) -> Result<(Option<ObjectId>, &'a Object)> {
120        crate::third_party::lopdf::Document::dereference(self, object)
121    }
122    fn encrypted(&self) -> bool {
123        self.is_encrypted() || self.was_encrypted() || self.trailer.has(b"Encrypt")
124    }
125    // `page_ids` is not overridden: lopdf's own page-tree walk trusts /Count and pre-allocates
126    // by it, so a hostile /Count (a fuzz finding: 4294967295) aborted the process with a 48 GiB
127    // allocation. The trait's bounded default walk counts the real pages instead.
128    fn get_page_fonts(&self, page: ObjectId) -> Result<BTreeMap<Vec<u8>, &Dictionary>> {
129        self.get_page_fonts(page)
130    }
131}

133type Slot = (u32, OnceLock<Option<Box<Object>>>);
134
135pub(crate) struct LazyStructure {
136    source: Document,
137    trailer: Dictionary,
138    // Sorted slots keep returned references stable, without allocating by the largest
139    // input object number (which can be sparse or hostile). Only queried objects fill slots.
140    objects: Vec<Slot>,
141    failed: AtomicBool,
142}
143
144impl LazyStructure {
145    pub(crate) fn new(bytes: Arc<Vec<u8>>, password: Option<&str>) -> std::result::Result<Self, String> {
146        let source = Document::open_with_stream_limit(bytes, password, 256 << 20).map_err(|e| e.to_string())?;
147        // A file rebuilt because its last update has no `/Prev` is read here as rebuilt: the
148        // renderer rebuilds it too, and the fallback reader would follow the cut-off chain.
149        if !source.repair_log().is_empty() && !source.rebuilt_unlinked_update() {
150            return Err("the structure requires compatibility repair".into());
151        }
152        let trailer = convert_dict(source.trailer(), 0).map_err(|e| e.to_string())?;
153        let objects = source.object_numbers().into_iter().map(|n| (n, OnceLock::new())).collect();
154        Ok(Self { source, trailer, objects, failed: AtomicBool::new(false) })
155    }
156
157    pub(crate) fn failed(&self) -> bool {
158        self.failed.load(Ordering::Relaxed)
159    }
160}
161
162impl Structure for LazyStructure {
163    fn trailer(&self) -> &Dictionary {
164        &self.trailer
165    }
166    fn encrypted(&self) -> bool {
167        self.source.security().is_some() || self.trailer.has(b"Encrypt")
168    }
169    /// The object `id`, read from the lazy source and converted once, then kept; a failed
170    /// conversion marks the structure so the caller can fall back to the eager loader.
171    fn get_object(&self, id: ObjectId) -> Result<&Object> {
172        let missing = || Error::ObjectNotFound(id);
173        let slot = self.objects.binary_search_by_key(&id.0, |(n, _)| *n).ok().and_then(|i| self.objects.get(i)).ok_or_else(missing)?;
174        if self.source.generation(id.0) != id.1 {
175            return Err(missing());
176        }
177        slot.1
178            .get_or_init(|| {
179                let result = self.source.try_get(id.0).ok().and_then(|o| convert(&o, 0).ok()).map(Box::new);
180                if result.is_none() {
181                    self.failed.store(true, Ordering::Relaxed);
182                }
183                result
184            })
185            .as_deref()
186            .ok_or_else(missing)
187    }
188}

190/// Nesting `convert` accepts: the same as `pdfcraft_cos`'s parser (each array or dictionary
191/// one level), so every object cos parsed converts. A lower limit would turn deep but valid
192/// nesting into a lazy-reader failure and force the eager lopdf fallback.
193const CONVERT_DEPTH: usize = 128;
194
195/// Convert a cos dictionary to lopdf's, without `/PieceInfo`. `depth` is the nesting level of
196/// the dictionary itself; its values sit one level deeper.
197fn convert_dict(dict: &crate::cos::Dict, depth: usize) -> Result<Dictionary> {
198    if depth > CONVERT_DEPTH {
199        return Err(Error::RecursionLimit);
200    }
201    dict.iter()
202        .filter(|(key, _)| key.as_slice() != PRIVATE_DATA)
203        .map(|(key, value)| Ok((key.clone(), convert(value, depth + 1)?)))
204        .collect::<Result<Vec<_>>>()
205        .map(|items| items.into_iter().collect())
206}
207
208/// `/PieceInfo`: an application's private data about a page, form or document (ISO 32000-1
209/// §14.5), left out of what inspection keeps. Nothing here reads it, and every object read is
210/// kept converted for the inspection's lifetime: InDesign writes kilobytes of it into every page,
211/// and a 1,412-page book held over a gigabyte of it (#5873). The file and the editing document are
212/// untouched; this is the read-only view inspection walks.
213const PRIVATE_DATA: &[u8] = b"PieceInfo";
214
215/// Convert a cos object to lopdf's, failing past [`CONVERT_DEPTH`] levels of arrays and
216/// dictionaries.
217fn convert(object: &CosObject, depth: usize) -> Result<Object> {
218    if depth > CONVERT_DEPTH {
219        return Err(Error::RecursionLimit);
220    }
221    Ok(match object {
222        CosObject::Null => Object::Null,
223        CosObject::Bool(value) => Object::Boolean(*value),
224        CosObject::Int(value) => Object::Integer(*value),
225        CosObject::Real(value) => Object::Real(*value as f32),
226        CosObject::Name(value) => Object::Name(value.clone()),
227        CosObject::String(value) => Object::String(
228            value.bytes.clone(),
229            if value.hex { crate::third_party::lopdf::StringFormat::Hexadecimal } else { crate::third_party::lopdf::StringFormat::Literal },
230        ),
231        CosObject::Ref(value) => Object::Reference((value.num, value.generation)),
232        CosObject::Array(value) => Object::Array(value.iter().map(|o| convert(o, depth + 1)).collect::<Result<_>>()?),
233        CosObject::Dict(value) => Object::Dictionary(convert_dict(value, depth)?),
234        CosObject::Stream(value) => Object::Stream(crate::third_party::lopdf::Stream::new(convert_dict(&value.dict, depth)?, value.raw.to_vec())),
235    })
236}

242    /// A hostile page-tree /Count must not abort the process: lopdf's own `get_pages`
243    /// pre-allocates by it (fuzz finding: /Count 4294967295 -> a 48 GiB allocation),
244    /// the bounded default walk counts the real pages instead.
245    #[test]
246    fn a_hostile_page_count_does_not_abort_inspection() {
247        let pdf = b"%PDF-1.7
2481 0 obj
249<< /Type /Catalog /Pages 2 0 R >>
250endobj
2512 0 obj
252<< /Type /Pages /Kids [3 0 R] /Count 4294967295 >>
253endobj
2543 0 obj
255<< /Type /Page /Parent 2 0 R /MediaBox [0 0 200 200] >>
256endobj
257trailer
258<< /Size 4 /Root 1 0 R >>
259%%EOF";
260        let doc = crate::third_party::lopdf::Document::load_mem(&pdf[..]).unwrap();
261        let pages = Structure::page_ids(&doc);
262        assert_eq!(pages, [(3, 0)], "the walk counts the one real page, not /Count");
263    }

265    fn nested_sparse_page_tree(repeated_and_cyclic: bool) -> crate::third_party::lopdf::Document {
266        let mut doc = crate::third_party::lopdf::Document::with_version("1.7");
267        let mut catalog = Dictionary::new();
268        catalog.set("Type", Object::Name(b"Catalog".to_vec()));
269        catalog.set("Pages", Object::Reference((5, 0)));
270        doc.objects.insert((17, 0), Object::Dictionary(catalog));
271
272        let mut pages = Dictionary::new();
273        pages.set("Type", Object::Name(b"Pages".to_vec()));
274        pages.set("Kids", vec![Object::Reference((31, 0)), Object::Reference((88, 0))]);
275        pages.set("Count", Object::Integer(if repeated_and_cyclic { 5 } else { 3 }));
276        doc.objects.insert((5, 0), Object::Dictionary(pages));
277
278        let mut nested = Dictionary::new();
279        nested.set("Type", Object::Name(b"Pages".to_vec()));
280        nested.set("Parent", Object::Reference((5, 0)));
281        let kids = if repeated_and_cyclic {
282            vec![Object::Reference((12, 0)), Object::Reference((5, 0)), Object::Reference((31, 0)), Object::Reference((42, 0))]
283        } else {
284            vec![Object::Reference((12, 0)), Object::Reference((42, 0))]
285        };
286        nested.set("Kids", kids);
287        nested.set("Count", Object::Integer(if repeated_and_cyclic { 4 } else { 2 }));
288        doc.objects.insert((88, 0), Object::Dictionary(nested));
289
290        for (id, parent) in [(31, 5), (12, 88), (42, 88)] {
291            let mut page = Dictionary::new();
292            page.set("Type", Object::Name(b"Page".to_vec()));
293            page.set("Parent", Object::Reference((parent, 0)));
294            doc.objects.insert((id, 0), Object::Dictionary(page));
295        }
296        doc.max_id = 88;
297        doc.trailer.set("Root", Object::Reference((17, 0)));
298        doc
299    }

319    #[test]
320    fn inspection_loads_only_requested_objects_in_a_compressed_document() {
321        let mut source = crate::third_party::lopdf::Document::with_version("1.7");
322        let catalog = source.add_object(crate::third_party::lopdf::dictionary! { "Type" => "Catalog", "Pages" => (2, 0) });
323        source.add_object(crate::third_party::lopdf::dictionary! { "Type" => "Pages", "Kids" => vec![Object::Reference((3, 0))], "Count" => 1 });
324        source
325            .add_object(crate::third_party::lopdf::dictionary! { "Type" => "Page", "Parent" => (2, 0), "MediaBox" => vec![0.into(), 0.into(), 100.into(), 100.into()] });
326        source.trailer.set("Root", catalog);
327        let unused: Vec<_> =
328            (0..300).map(|i| Object::Reference(source.add_object(crate::third_party::lopdf::dictionary! { "Unused" => i }))).collect();
329        // Keep these objects reachable when the full writer removes orphan objects,
330        // through a catalog extension that the panel inspector has no reason to read.
331        source.get_dictionary_mut(catalog).unwrap().set("Uninspected", unused);
332        let mut bytes = Vec::new();
333        source.save_to(&mut bytes).unwrap();
334        let cos = Document::open(Arc::new(bytes)).unwrap();
335        let compressed = crate::cos::write_full(&cos, &crate::cos::SaveOptions::default()).unwrap();
336        let eager = crate::third_party::lopdf::Document::load_mem(&compressed).unwrap();
337        let expected = Structure::page_ids(&eager);
338        let view = LazyStructure::new(Arc::new(compressed), None).unwrap();
339        assert_eq!(view.page_ids().len(), 1);
340        assert_eq!(view.page_ids(), expected);
341        assert!(!view.failed());
342        assert!(
343            view.objects.iter().filter(|(_, slot)| slot.get().is_some()).count() <= 3,
344            "unused objects must not be materialized to inspect a page tree"
345        );
346        let unused = view.catalog().unwrap().get(b"Uninspected").unwrap().as_array().unwrap();
347        let last = unused.last().unwrap().as_reference().unwrap();
348        assert!(matches!(view.get_dictionary(last).unwrap().get(b"Unused"), Ok(Object::Integer(299))));
349    }
