18    /// The destination `/OpenAction` opens at (ISO 32000-2 §7.7.2 Table 29): an explicit
19    /// destination array, a `/GoTo` action's `/D`, or (leniently, since Table 29 asks for an array
20    /// or an action) a destination name or string; a name or string is looked up as a named
21    /// destination (§12.3.2.4). Any other action (a script, a remote go-to) has none here, and
22    /// neither has a page number past the end of the document.
23    pub(super) fn open_dest(&self, action: &Object) -> Option<(usize, DestView)> {
24        let dest = match self.resolve(action) {
25            Object::Dictionary(d) => {
26                if d.get(b"S").ok().and_then(|o| self.resolve(o).as_name().ok()) != Some(&b"GoTo"[..]) {
27                    return None;
28                }
29                d.get(b"D").ok()?
30            }
31            other => other,
32        };
33        self.dest(dest, 0).filter(|(page, _)| *page < self.page_ids.len())
34    }

36    pub(super) fn page_of(&self, o: &Object) -> Option<usize> {
37        match o {
38            Object::Reference(id) => self.page_index.get(id).copied(),
39            // Some producers write page *numbers* in remote-style destinations.
40            Object::Integer(n) => usize::try_from(*n).ok(),
41            _ => None,
42        }
43    }

47    /// The page and view of a destination: an explicit array (`[page /XYZ left top zoom]`, …),
48    /// a dictionary with `/D`, or a named destination that resolves to one.
49    pub(super) fn dest(&self, dest: &Object, depth: u32) -> Option<(usize, DestView)> {
50        if depth > 8 {
51            return None;
52        }
53        match self.resolve(dest) {
54            Object::Array(a) => {
55                let page = a.first().and_then(|p| self.page_of(p))?;
56                let kind = a.get(1).map(|k| self.resolve(k)).and_then(|k| k.as_name().ok());
57                let operands: Vec<Option<f32>> = a.iter().skip(2).take(4).map(|o| self.resolve(o).as_float().ok()).collect();
58                Some((page, DestView::from_operands(kind, &operands)))
59            }
60            Object::Dictionary(d) => d.get(b"D").ok().and_then(|d| self.dest(d, depth + 1)),
61            Object::String(key, _) | Object::Name(key) => self.named_dest(key).and_then(|d| self.dest(d, depth + 1)),
62            _ => None,
63        }
64    }

66    fn named_dest(&self, key: &[u8]) -> Option<&'a Object> {
67        let catalog = self.doc.catalog().ok()?;
68        if let Some(v) = self.named.get(key) {
69            return Some(v);
70        }
71        // PDF 1.1 style /Dests dictionary.
72        catalog.get(b"Dests").ok().and_then(|o| self.dict(o)).and_then(|d| d.get(key).ok())
73    }

75    /// The destination whose key, read as a text string (UTF-16 or PDFDocEncoding), is `name`:
76    /// a name tree key, then a key of the PDF 1.1 `/Dests` dictionary, whose names older files
77    /// write in PDFDocEncoding rather than UTF-8.
78    fn named_dest_by_text(&self, name: &str) -> Option<&'a Object> {
79        if let Some((_, dest)) = self.named.iter().find(|(key, _)| text_string(key) == name) {
80            return Some(dest);
81        }
82        let dests = self.doc.catalog().ok()?.get(b"Dests").ok().and_then(|o| self.dict(o))?;
83        dests.iter().find(|(key, _)| text_string(key) == name).map(|(_, dest)| dest)
84    }

86    /// A name tree's leaves keyed by their raw key bytes (first occurrence wins).
87    pub(super) fn name_tree_raw(
88        &self,
89        node: &'a Dictionary,
90        seen: &mut HashSet<*const Dictionary>,
91        depth: u32,
92        out: &mut HashMap<Vec<u8>, &'a Object>,
93        key_budget: &mut usize,
94    ) {
95        if depth > 32 || !seen.insert(node as *const _) {
96            return;
97        }
98        if let Ok(Object::Array(pairs)) = node.get(b"Names").map(|o| self.resolve(o)) {
99            for pair in pairs.chunks(2) {
100                if let [k, v] = pair {
101                    let key = match self.resolve(k) {
102                        Object::String(s, _) => s,
103                        Object::Name(n) => n,
104                        _ => continue,
105                    };
106                    // Every key is hashed, so its bytes are spent from one allowance for the whole
107                    // tree (a key repeated by reference is charged each time); one past it is
108                    // not indexed. Only a first occurrence is copied.
109                    let Some(left) = key_budget.checked_sub(key.len()) else { continue };
110                    *key_budget = left;
111                    if !out.contains_key(key.as_slice()) {
112                        out.insert(key.clone(), v);
113                    }
114                }
115            }
116        }
117        if let Ok(Object::Array(kids)) = node.get(b"Kids").map(|o| self.resolve(o)) {
118            for k in kids {
119                if let Some(d) = self.dict(k) {
120                    self.name_tree_raw(d, seen, depth + 1, out, key_budget);
121                }
122            }
123        }
124    }

126    /// The entries of an embedded-files name tree. Each string key is spent from `text_budget`
127    /// (see [`MAX_LINK_ATTACHMENT_TEXT_BYTES`]), and an entry whose key does not fit is left out.
128    pub(super) fn name_tree(
129        &self,
130        node: &'a Dictionary,
131        seen: &mut HashSet<*const Dictionary>,
132        depth: u32,
133        text_budget: &mut usize,
134    ) -> Vec<(String, &'a Object)> {
135        let mut out = Vec::new();
136        if depth > 32 || !seen.insert(node as *const _) {
137            return out;
138        }
139        if let Ok(Object::Array(pairs)) = node.get(b"Names").map(|o| self.resolve(o)) {
140            for pair in pairs.chunks(2) {
141                if let [k, v] = pair {
142                    let key = match self.resolve(k) {
143                        Object::String(bytes, _) => match spend(bytes, text_budget, text_string) {
144                            Some(key) => key,
145                            None => continue,
146                        },
147                        _ => String::new(),
148                    };
149                    out.push((key, v));
150                }
151            }
152        }
153        if let Ok(Object::Array(kids)) = node.get(b"Kids").map(|o| self.resolve(o)) {
154            for k in kids {
155                if let Some(d) = self.dict(k) {
156                    out.extend(self.name_tree(d, seen, depth + 1, text_budget));
157                }
158            }
159        }
160        out
161    }

163    /// The first entry of an embedded-files name tree whose key is `key`, in the order
164    /// [`Self::name_tree`] lists them, with no key budget. It keeps no key, and converts a key
165    /// object or `/Names` array that several places reference only once (a second visit cannot
166    /// match, since the first would have returned).
167    pub(super) fn name_tree_find(&self, node: &'a Dictionary, key: &str, seen: &mut NameTreeSeen, depth: u32) -> Option<&'a Object> {
168        if depth > 32 || !seen.nodes.insert(node as *const _) {
169            return None;
170        }
171        let names = node.get(b"Names").ok();
172        if let Some(names) = names
173            && self.resolved_id(names).is_none_or(|id| seen.objects.insert(id))
174            && let Object::Array(pairs) = self.resolve(names)
175        {
176            for pair in pairs.chunks(2) {
177                if let [k, v] = pair {
178                    if self.resolved_id(k).is_some_and(|id| !seen.objects.insert(id)) {
179                        continue;
180                    }
181                    let found = match self.resolve(k) {
182                        Object::String(bytes, _) => text_string(bytes) == key,
183                        _ => key.is_empty(),
184                    };
185                    if found {
186                        return Some(v);
187                    }
188                }
189            }
190        }
191        if let Ok(Object::Array(kids)) = node.get(b"Kids").map(|o| self.resolve(o)) {
192            for k in kids {
193                if let Some(d) = self.dict(k)
194                    && let Some(v) = self.name_tree_find(d, key, seen, depth + 1)
195                {
196                    return Some(v);
197                }
198            }
199        }
200        None
201    }

205    /// The window `DocInfo::outline` keeps: the first `OUTLINE_WINDOW` bookmarks in outline order,
206    /// nested no deeper than `OUTLINE_DEPTH`, and whether any bookmark was left out. Iterative, and
207    /// each bookmark is read once, so the work and memory stay within the window. Web addresses are
208    /// spent from `text_budget`, as links' are.
209    pub(super) fn outline_window(&self, first: &'a Object, text_budget: &mut usize) -> (Vec<OutlineItem>, bool) {
210        let mut seen = HashSet::new();
211        // The listed bookmarks in outline order, with their depth.
212        let mut listed: Vec<(usize, OutlineItem)> = Vec::new();
213        // One entry per open level: its next sibling, or `None` once that run of siblings has ended.
214        let mut levels: Vec<Option<&'a Object>> = vec![Some(first)];
215        let mut more = false;
216        while let Some(depth) = levels.len().checked_sub(1) {
217            let Some(slot) = levels.get_mut(depth) else { break };
218            let Some(o) = slot.take() else {
219                levels.pop();
220                continue;
221            };
222            if let Object::Reference(id) = o
223                && !seen.insert(*id)
224            {
225                continue; // cycle: this run of siblings ends here
226            }
227            let Some(d) = self.dict(o) else { continue };
228            if listed.len() >= OUTLINE_WINDOW {
229                more = true;
230                break;
231            }
232            *slot = d.get(b"Next").ok();
233            // A bookmark goes to its `/Dest`, else does what its action `/A` does, as a link would.
234            // Only a go-to's `/D` is a page here: a remote go-to's names a page of the other file.
235            // An action without a type is still read as the go-to its `/D` suggests.
236            let action = match d.get(b"Dest").ok().and_then(|dest| self.dest(dest, 0)) {
237                Some((page, view)) => Some(LinkTarget::Page(page, view)),
238                None => d.get(b"A").ok().and_then(|a| self.dict(a)).and_then(|a| match a.get(b"S") {
239                    Ok(_) => self.action_target(a, text_budget),
240                    Err(_) => a.get(b"D").ok().and_then(|dest| self.dest(dest, 0)).map(|(page, view)| LinkTarget::Page(page, view)),
241                }),
242            };
243            let ((page, view), action) = match action {
244                Some(LinkTarget::Page(page, view)) => ((Some(page), view), None),
245                other => ((None, DestView::Top), other),
246            };
247            match d.get(b"First").ok() {
248                Some(child) if depth < OUTLINE_DEPTH => levels.push(Some(child)),
249                Some(_) => more = true,
250                None => {}
251            }
252            let open = d.get(b"Count").ok().and_then(|c| c.as_i64().ok()).is_some_and(|c| c > 0);
253            let title = self.field_text(d, b"Title").unwrap_or_default();
254            let color = match d.get(b"C").map(|o| self.resolve(o)) {
255                Ok(Object::Array(c)) => outline_color(c.iter().map(|v| self.resolve(v).as_float().ok())),
256                _ => None,
257            };
258            let flags = d.get(b"F").ok().and_then(|f| self.resolve(f).as_i64().ok()).unwrap_or(0);
259            let (italic, bold) = (flags & 1 != 0, flags & 2 != 0);
260            listed.push((depth, OutlineItem { title, page, view, action, children: Vec::new(), open, color, italic, bold }));
261        }
262        (nest_outline(listed), more)
263    }

279    /// Apply `/PageLabels` to `pages`. `Err`: the tree is unusable and no label is applied.
280    /// `Ok(notes)`: labels applied, with what was repaired on the way (a duplicated start page,
281    /// a range whose labels are too long to show).
282    pub(super) fn page_labels(&self, catalog: &Dictionary, pages: &mut [PageInfo]) -> Result<Vec<String>, &'static str> {
283        let mut notes = Vec::new();
284        let Ok(tree) = catalog.get(b"PageLabels") else { return Ok(notes) };
285        let Some(tree) = self.label_dict(tree)? else { return Ok(notes) };
286        let mut ranges: BTreeMap<usize, &Dictionary> = BTreeMap::new();
287        let mut stack = vec![(tree, 0usize)];
288        let mut seen = HashSet::new();
289        let mut left = MAX_LABEL_TREE_WORK;
290        while let Some((node, depth)) = stack.pop() {
291            label_work(&mut left, 1)?;
292            if depth > MAX_LABEL_TREE_DEPTH {
293                return Err("the number tree is too deep");
294            }
295            if !seen.insert(node as *const Dictionary) {
296                return Err("the number tree repeats a node or contains a cycle");
297            }
298            if let Ok(nums) = node.get(b"Nums")
299                && let Object::Array(nums) = self.label_object(nums)?
300            {
301                label_work(&mut left, nums.len().div_ceil(2))?;
302                for pair in nums.chunks(2) {
303                    if let [k, v] = pair
304                        && let (Ok(start), Some(d)) = (self.label_object(k)?.as_i64(), self.label_dict(v)?)
305                        && let Ok(start) = usize::try_from(start)
306                        && ranges.insert(start, d).is_some()
307                    {
308                        // The later range wins, as in readers that overwrite.
309                        notes.push(format!("a range starts on page {} twice; the later one is used", start.saturating_add(1)));
310                    }
311                }
312            }
313            if let Ok(kids) = node.get(b"Kids")
314                && let Object::Array(kids) = self.label_object(kids)?
315            {
316                label_work(&mut left, kids.len())?;
317                if kids.len() > MAX_LABEL_TREE_WORK.saturating_sub(stack.len()) {
318                    return Err("the number tree has too many pending nodes");
319                }
320                if !kids.is_empty() && depth == MAX_LABEL_TREE_DEPTH {
321                    return Err("the number tree is too deep");
322                }
323                for kid in kids {
324                    if let Some(d) = self.label_dict(kid)? {
325                        stack.push((d, depth + 1));
326                    }
327                }
328            }
329        }
330        // Decode each bounded prefix once, rather than once for every page in its range.
331        let mut specs = BTreeMap::new();
332        let mut prefix_total = 0usize;
333        for (start, d) in ranges {
334            let first = d.get(b"St").ok().and_then(|o| self.resolve(o).as_i64().ok()).unwrap_or(1).max(1) as u64;
335            let prefix = match d.get(b"P").ok().map(|o| self.resolve(o)) {
336                Some(Object::String(raw, _)) => {
337                    if raw.len() > MAX_PREFIX_BYTES {
338                        return Err("a raw prefix exceeds 2050 bytes");
339                    }
340                    // A label prefix is literal text, not trimmed document metadata.
341                    text_string(raw).trim_matches('\0').to_string()
342                }
343                Some(Object::Name(raw)) => {
344                    if raw.len() > MAX_PREFIX_BYTES {
345                        return Err("a raw prefix exceeds 2050 bytes");
346                    }
347                    // Retain the existing name compatibility and boundary-NUL handling.
348                    String::from_utf8_lossy(raw).trim_matches('\0').to_string()
349                }
350                _ => String::new(),
351            };
352            if prefix.len() > MAX_LABEL_BYTES {
353                return Err("a decoded prefix exceeds 1024 UTF-8 bytes");
354            }
355            // A prefix object shared by many ranges is decoded once per range: count every copy.
356            prefix_total = prefix_total.saturating_add(prefix.len());
357            if prefix_total > MAX_PREFIX_TOTAL_BYTES {
358                return Err("decoded prefixes exceed 4 MiB in total");
359            }
360            let style = d.get(b"S").ok().and_then(|o| self.resolve(o).as_name().ok());
361            specs.insert(start, (first, prefix, style));
362        }
363        let mut staged = Vec::new();
364        let mut total = 0usize;
365        for (i, _) in pages.iter().enumerate() {
366            let Some((&start, (first, prefix, style))) = specs.range(..=i).next_back() else { continue };
367            let n = if matches!(*style, Some(b"D" | b"R" | b"r" | b"A" | b"a")) {
368                let offset = u64::try_from(i - start).map_err(|_| "a page number is too large")?;
369                first.checked_add(offset).ok_or("a page number overflows")?
370            } else {
371                0 // /St is unused when the label consists only of its prefix.
372            };
373            let remaining = MAX_LABEL_BYTES - prefix.len();
374            let number = match *style {
375                Some(b"D") => Some(n.to_string()),
376                Some(b"R" | b"r") => roman(n, remaining),
377                Some(b"A" | b"a") => alpha(n, remaining),
378                _ => Some(String::new()),
379            }
380            .filter(|number| number.len() <= remaining);
381            // A label too long to show keeps that page's physical number; other ranges stay.
382            let Some(mut number) = number else {
383                let note = format!(
384                    "labels from page {} exceed {MAX_LABEL_BYTES} UTF-8 bytes; those pages show their physical page numbers",
385                    start.saturating_add(1)
386                );
387                if !notes.contains(&note) {
388                    notes.push(note);
389                }
390                continue;
391            };
392            if matches!(*style, Some(b"R" | b"A")) {
393                number.make_ascii_uppercase();
394            }
395            let mut label = prefix.clone();
396            label.push_str(&number);
397            if label.is_empty() {
398                continue; // Keep the existing physical number for an empty custom label.
399            }
400            total = total.checked_add(label.len()).ok_or("the total label size overflows")?;
401            if total > MAX_LABEL_TOTAL_BYTES {
402                return Err("custom labels exceed 4 MiB in total");
403            }
404            staged.push((i, label));
405        }
406        // A malformed later range must not leave a partially applied custom label sequence.
407        for (i, label) in staged {
408            if let Some(page) = pages.get_mut(i) {
409                page.label = label;
410            }
411        }
412        Ok(notes)
413    }

430/// A bookmark title's colour from its `/C` entries, each kept within 0–1 (ISO 32000-2 §12.3.3,
431/// Table 151: "An array of three numbers in the range 0.0 to 1.0, representing the components in
432/// the DeviceRGB colour space"). Black, the default, and anything that isn't three numbers give
433/// `None`.
434pub(crate) fn outline_color(values: impl Iterator<Item = Option<f32>>) -> Option<[f32; 3]> {
435    let v: Vec<Option<f32>> = values.take(4).collect();
436    let [Some(r), Some(g), Some(b)] = v.as_slice() else { return None };
437    // A number past the range, however large, is kept at its edge.
438    let rgb = [*r, *g, *b].map(|c| c.clamp(0.0, 1.0));
439    (rgb != [0.0; 3]).then_some(rgb)
440}

442/// Nests a pre-order list of `(depth, item)` into the outline tree.
443fn nest_outline(listed: Vec<(usize, OutlineItem)>) -> Vec<OutlineItem> {
444    let mut roots = Vec::new();
445    // The open ancestors, outermost first; each collects its children until it closes.
446    let mut open: Vec<(usize, OutlineItem)> = Vec::new();
447    for (depth, item) in listed {
448        while open.last().is_some_and(|(d, _)| *d >= depth) {
449            close_outline(&mut open, &mut roots);
450        }
451        open.push((depth, item));
452    }
453    while !open.is_empty() {
454        close_outline(&mut open, &mut roots);
455    }
456    roots
457}

459/// Closes the innermost open bookmark: it joins its parent's children, or the top level.
460fn close_outline(open: &mut Vec<(usize, OutlineItem)>, roots: &mut Vec<OutlineItem>) {
461    if let Some((_, item)) = open.pop() {
462        match open.last_mut() {
463            Some((_, parent)) => parent.children.push(item),
464            None => roots.push(item),
465        }
466    }
467}

474/// The (0-based page, view) of the named destination `name` (ISO 32000-2 §12.3.2.4): a key of
475/// the `/Names /Dests` name tree or of the PDF 1.1 catalog `/Dests` dictionary, as a link's or
476/// `/OpenAction`'s name resolves. `None` when the document can't be read, has no such name, or
477/// the name leads to no page of it. A key matches `name` written in UTF-8 or in PDFDocEncoding,
478/// or as a text string (UTF-16 or PDFDocEncoding) that reads as `name`, the way a link's string
479/// destination is read.
480/// Read like [`inspect`](super::inspect): the lazy reader first, lopdf's tolerant one when the lazy reader fails.
481pub fn named_destination(bytes: Arc<Vec<u8>>, password: Option<&str>, name: &str) -> Option<(usize, DestView)> {
482    // PDFDocEncoding bytes that are also UTF-8 read as UTF-8, so only this finds "Ã©" written C3 A9.
483    let pdfdoc: Option<Vec<u8>> = name.chars().map(crate::cos::pdfdoc_byte).collect();
484    let find = |doc: &dyn Structure| {
485        let insp = Inspector::new(doc);
486        let dest = insp
487            .named_dest(name.as_bytes())
488            .or_else(|| pdfdoc.as_deref().and_then(|key| insp.named_dest(key)))
489            .or_else(|| insp.named_dest_by_text(name))?;
490        insp.dest(dest, 1).filter(|(page, _)| *page < insp.page_ids.len())
491    };
492    let lazy = catch_unwind(AssertUnwindSafe(|| {
493        let doc = LazyStructure::new(bytes.clone(), password).ok()?;
494        let found = find(&doc);
495        (!doc.failed()).then_some(found)
496    }))
497    .ok()
498    .flatten();
499    if let Some(found) = lazy {
500        return found;
501    }
502    catch_unwind(AssertUnwindSafe(|| find(&Document::load_mem_with_options(&bytes, load_options(password)).ok()?))).ok().flatten()
503}

511    /// Three 612×792 pages (objects 3, 4, 5); `catalog` is added to the catalog dictionary and
512    /// `extra` become objects 6, 7, ….
513    fn pdf(catalog: &str, extra: &[String]) -> Vec<u8> {
514        let mut objs =
515            vec![format!("<< /Type /Catalog /Pages 2 0 R {catalog} >>"), "<< /Type /Pages /Kids [3 0 R 4 0 R 5 0 R] /Count 3 >>".to_string()];
516        for _ in 0..3 {
517            objs.push("<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] >>".to_string());
518        }
519        objs.extend(extra.iter().cloned());
520        let mut out = b"%PDF-1.7\n".to_vec();
521        let mut offsets = Vec::new();
522        for (i, o) in objs.iter().enumerate() {
523            offsets.push(out.len());
524            out.extend_from_slice(format!("{} 0 obj\n{o}\nendobj\n", i + 1).as_bytes());
525        }
526        let xref = out.len();
527        out.extend_from_slice(format!("xref\n0 {}\n0000000000 65535 f \n", objs.len() + 1).as_bytes());
528        for o in offsets {
529            out.extend_from_slice(format!("{o:010} 00000 n \n").as_bytes());
530        }
531        out.extend_from_slice(format!("trailer\n<< /Size {} /Root 1 0 R >>\nstartxref\n{xref}\n%%EOF\n", objs.len() + 1).as_bytes());
532        out
533    }

535    fn open_dest(catalog: &str, extra: &[String]) -> Option<(usize, DestView)> {
536        inspect(Arc::new(pdf(catalog, extra)), None).expect("inspects").open_dest
537    }

541    #[test]
542    fn an_explicit_destination_opens_at_its_page_and_position() {
543        assert_eq!(open_dest("/OpenAction [5 0 R /XYZ 100 500 2]", &[]), Some((2, XYZ)));
544        assert_eq!(open_dest("/OpenAction [4 0 R /FitH 400]", &[]), Some((1, DestView::FitH { top: Some(400.0) })));
545        // Indirect, and a zoom of 0 keeps the zoom (Table 149).
546        assert_eq!(
547            open_dest("/OpenAction 6 0 R", &["[4 0 R /XYZ 0 792 0]".into()]),
548            Some((1, DestView::Xyz { left: Some(0.0), top: Some(792.0), zoom: None }))
549        );
550    }

552    #[test]
553    fn a_go_to_action_opens_at_its_destination_explicit_or_named() {
554        assert_eq!(open_dest("/OpenAction << /S /GoTo /D [5 0 R /XYZ 100 500 2] >>", &[]), Some((2, XYZ)));
555        // A name, through the PDF 1.1 catalog /Dests dictionary (§12.3.2.4).
556        assert_eq!(open_dest("/OpenAction << /S /GoTo /D /there >> /Dests << /there [5 0 R /XYZ 100 500 2] >>", &[]), Some((2, XYZ)));
557        // A byte string, through the /Names /Dests name tree, whose value is a dictionary with /D.
558        assert_eq!(
559            open_dest(
560                "/OpenAction << /S /GoTo /D (there) >> /Names << /Dests 6 0 R >>",
561                &["<< /Kids [7 0 R] >>".into(), "<< /Names [(here) [3 0 R /Fit] (there) << /D [5 0 R /XYZ 100 500 2] >>] >>".into()]
562            ),
563            Some((2, XYZ))
564        );
565    }

571    #[test]
572    fn a_named_destination_is_looked_up_by_its_name() {
573        // A string key of the /Names /Dests name tree, whose value may be a dictionary with /D.
574        let tree = "/Names << /Dests 6 0 R >>";
575        let nodes = ["<< /Kids [7 0 R] >>".to_string(), "<< /Names [(here) [3 0 R /Fit] (there) << /D [5 0 R /XYZ 100 500 2] >>] >>".to_string()];
576        assert_eq!(named(tree, &nodes, "there"), Some((2, XYZ)));
577        assert_eq!(named(tree, &nodes, "here"), Some((0, DestView::Fit)));
578        // A name key of the PDF 1.1 catalog /Dests dictionary.
579        assert_eq!(named("/Dests << /old [4 0 R /FitH 400] >>", &[], "old"), Some((1, DestView::FitH { top: Some(400.0) })));
580        // A key written as UTF-16 or in PDFDocEncoding matches the text it reads as.
581        assert_eq!(named("/Names << /Dests << /Names [<FEFF0074006800E9> [4 0 R /Fit]] >> >>", &[], "thé"), Some((1, DestView::Fit)));
582        assert_eq!(named("/Names << /Dests << /Names [(th\\351) [5 0 R /Fit]] >> >>", &[], "thé"), Some((2, DestView::Fit)));
583        // So does a catalog /Dests name written in PDFDocEncoding, as older files have them, but
584        // a name written as UTF-8 comes first.
585        assert_eq!(named("/Dests << /th#E9 [4 0 R /Fit] >>", &[], "thé"), Some((1, DestView::Fit)));
586        assert_eq!(named("/Dests << /th#E9 [4 0 R /Fit] /th#C3#A9 [5 0 R /Fit] >>", &[], "thé"), Some((2, DestView::Fit)));
587        // A key whose PDFDocEncoding bytes are also UTF-8 reads as UTF-8 ("é"), yet is found by
588        // the text it holds in PDFDocEncoding ("Ã©") too.
589        assert_eq!(named("/Dests << /#C3#A9 [4 0 R /Fit] >>", &[], "Ã©"), Some((1, DestView::Fit)));
590        assert_eq!(named("/Names << /Dests << /Names [(\\303\\251) [5 0 R /Fit]] >> >>", &[], "Ã©"), Some((2, DestView::Fit)));
591        // A name that isn't there, one past the last page, one that names itself, no names at
592        // all, and bytes that aren't a PDF.
593        assert_eq!(named(tree, &nodes, "missing"), None);
594        assert_eq!(named("/Dests << /far [7 /Fit] >>", &[], "far"), None);
595        assert_eq!(named("/Dests << /loop /loop >>", &[], "loop"), None);
596        assert_eq!(named("", &[], "there"), None);
597        assert_eq!(super::named_destination(Arc::new(b"not a pdf".to_vec()), None, "there"), None);
598    }

629    #[test]
630    fn cyclic_and_deep_named_destinations_end() {
631        // Names that refer to each other.
632        assert_eq!(open_dest("/OpenAction /a /Dests << /a /b /b /a >>", &[]), None);
633        assert_eq!(open_dest("/OpenAction /a /Dests << /a << /D /a >> >>", &[]), None);
634        // A name tree whose node is its own kid.
635        assert_eq!(open_dest("/OpenAction (x) /Names << /Dests 6 0 R >>", &["<< /Kids [6 0 R] >>".into()]), None);
636        // A name tree deeper than the depth bound: the leaf past it isn't read.
637        let depth = 40;
638        let mut nodes: Vec<String> = (0..depth).map(|i| format!("<< /Kids [{} 0 R] >>", 7 + i)).collect();
639        nodes.push("<< /Names [(x) [4 0 R /Fit]] >>".into());
640        assert_eq!(open_dest("/OpenAction (x) /Names << /Dests 6 0 R >>", &nodes), None);
641        // A shallow one is read.
642        let shallow = ["<< /Kids [7 0 R] >>".to_string(), "<< /Kids [8 0 R] >>".into(), "<< /Names [(x) [4 0 R /Fit]] >>".into()];
643        assert_eq!(open_dest("/OpenAction (x) /Names << /Dests 6 0 R >>", &shallow), Some((1, DestView::Fit)));
644    }

646    #[test]
647    fn a_key_repeated_by_reference_is_charged_each_time_against_the_name_tree_allowance() {
648        let big = format!("<{}>", "41".repeat(1 << 20));
649        let tree = |repeats: usize| {
650            let names = format!("{}(x) [4 0 R /Fit]", "7 0 R [4 0 R /Fit] ".repeat(repeats));
651            [format!("<< /Names [{names}] >>"), big.clone()]
652        };
653        let catalog = "/OpenAction (x) /Names << /Dests 6 0 R >>";
654        assert_eq!(open_dest(catalog, &tree(2)), Some((1, DestView::Fit)));
655        // 17 MiB of repeats of one reference spend the 16 MiB allowance before the real key.
656        assert_eq!(open_dest(catalog, &tree(17)), None);
657    }
