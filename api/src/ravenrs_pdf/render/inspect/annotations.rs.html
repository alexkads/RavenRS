30/// A colour array as RGB, read as pdfcraft-annot's appearance builder reads it (`appearance::color`
31/// and `drawn_color`): one entry is grey, three RGB, four CMYK; among four entries a non-number is
32/// 0 but a reference is unreadable, otherwise every entry must be a number. `None` for anything
33/// else, an empty array or a value out of range; each component is clamped to 0..=1.
34fn drawn_color(c: &[Object]) -> Option<[f32; 3]> {
35    let v: Vec<f32> = if c.len() == 4 {
36        c.iter()
37            .map(|o| match o {
38                Object::Reference(_) => None,
39                o => Some(o.as_float().unwrap_or(0.0)),
40            })
41            .collect::<Option<_>>()?
42    } else {
43        c.iter().map(|o| o.as_float().ok()).collect::<Option<_>>()?
44    };
45    let rgb = match v.as_slice() {
46        [g] => [*g; 3],
47        [r, g, b] => [*r, *g, *b],
48        [c, m, y, k] => [(1.0 - c) * (1.0 - k), (1.0 - m) * (1.0 - k), (1.0 - y) * (1.0 - k)],
49        _ => return None,
50    };
51    // pdfcraft-annot's `finite`: finite and below 1e7 in magnitude.
52    rgb.iter().all(|x| x.is_finite() && x.abs() < 1e7).then(|| rgb.map(|x| x.clamp(0.0, 1.0)))
53}

55/// The bytes a listed annotation keeps.
56pub(super) fn retained(a: &Annotation) -> usize {
57    let text = [&a.author, &a.contents, &a.modified, &a.name, &a.in_reply_to, &a.state, &a.intent];
58    text.into_iter()
59        .map(|t| t.as_ref().map_or(0, String::len))
60        .chain([
61            a.subtype.len(),
62            std::mem::size_of_val(a.quads.as_slice()),
63            std::mem::size_of_val(a.points.as_slice()),
64            std::mem::size_of::<Annotation>(),
65        ])
66        .fold(0, usize::saturating_add)
67}

70    /// Whether an annotation is a Fill & Sign mark. Keep in step with
71    /// `pdfcraft_annot::is_fill_sign`, which the engine's comment list uses.
72    fn is_fill_sign(&self, d: &Dictionary, subtype: &str) -> bool {
73        let yes = |key: &[u8]| matches!(d.get(key).map(|o| self.resolve(o)), Ok(Object::Boolean(true)));
74        if yes(b"PCFillSign") {
75            return true;
76        }
77        match subtype {
78            "FreeText" => self.name(d, b"IT").as_deref() == Some("FreeTextTypeWriter"),
79            "Stamp" => match self.name(d, b"Name").as_deref() {
80                Some("PCCheck" | "PCCross" | "PCDot" | "PCLine" | "PCTypedSignature") => true,
81                Some("PCCustomSignature" | "PCCustomInitials") => yes(b"PCPictureImage"),
82                _ => false,
83            },
84            "Ink" => self.text(d, b"Subj").as_deref() == Some("Signature"),
85            _ => false,
86        }
87    }

91    /// `text_budget` is what is left of [`MAX_LINK_ATTACHMENT_TEXT_BYTES`].
92    pub(super) fn annotations(&self, info: &mut DocInfo, text_budget: &mut usize) {
93        // Indirect `/QuadPoints` arrays whose quadrilaterals did not fit what was left of the budget:
94        // the budget only shrinks, so they are not read again (review on #4474).
95        let mut unfit_quads: HashSet<ObjectId> = HashSet::new();
96        let mut pages: Vec<(ObjectId, usize)> = self.page_index.iter().map(|(&id, &page)| (id, page)).collect();
97        pages.sort_unstable_by_key(|&(_, page)| page);
98        // An annotation, or a whole `/Annots` array, that several entries or pages list is on the
99        // first of them (§12.5.2 allows one page per annotation). Both are told apart by the object
100        // a reference resolves to, since an indirect object may itself hold a reference. Where each
101        // annotation is listed is collected before any is read, so a reply's `/IRT` finds its
102        // parent by reference even when the parent comes later.
103        let mut seen_arrays = HashSet::new();
104        let mut slots: HashMap<ObjectId, (usize, usize)> = HashMap::new();
105        let mut entries = Vec::new();
106        for (id, page) in pages {
107            let Ok(page_dict) = self.doc.get_dictionary(id) else { continue };
108            if let Ok(annots) = page_dict.get(b"Annots")
109                && let Some(r) = self.resolved_id(annots)
110                && !seen_arrays.insert(r)
111            {
112                continue;
113            }
114            let Ok(Object::Array(annots)) = page_dict.get(b"Annots").map(|o| self.resolve(o)) else { continue };
115            for (index, a) in annots.iter().enumerate() {
116                if let Some(r) = self.resolved_id(a) {
117                    let Entry::Vacant(slot) = slots.entry(r) else { continue };
118                    slot.insert((page, index));
119                }
120                entries.push((page, index, a));
121            }
122        }
123        let mut kept = 0usize;
124        // Once the comment budget is spent, comments are no longer built, but links and attachments
125        // are still listed.
126        let mut comments_full = false;
127        for (page, index, a) in entries {
128            let Some(d) = self.dict(a) else { continue };
129            // An annotation's `/Type`, when present, is `/Annot` (ISO 32000-2 §12.5.2, Table 166):
130            // a font or other dictionary a page lists among its annotations isn't one, and neither is
131            // one whose `/Type` isn't a name at all. A null or dangling `/Type` counts as absent
132            // (§7.3.9: a null value is the same as leaving the entry out).
133            if let Ok(t) = d.get(b"Type").map(|t| self.resolve(t))
134                && !matches!(t, Object::Null | Object::Reference(_))
135                && t.as_name().ok() != Some(b"Annot".as_slice())
136            {
137                continue;
138            }
139            let Some(subtype) = self.name(d, b"Subtype") else { continue };
140            if subtype == "FileAttachment"
141                && let Ok(fs) = d.get(b"FS")
142            {
143                let fallback = |budget: &mut usize| self.budgeted_text(d, b"Contents", budget);
144                info.attachments.push(self.attachment(fs, AttachmentSource::Annotation { page, index }, text_budget, fallback));
145            }
146            if subtype == "Link" {
147                self.collect_link(info, (page, index), d, text_budget, &mut unfit_quads);
148                continue;
149            }
150            // Not comments: form widgets, pop-ups and non-markup annotations such as the
151            // Screen annotation that drives a LaTeX `animate` player (keep in step with
152            // `pdfcraft_annot::is_comment_subtype`).
153            if matches!(subtype.as_str(), "Widget" | "Popup" | "Screen" | "Movie" | "RichMedia" | "3D" | "PrinterMark" | "TrapNet" | "Watermark") {
154                continue;
155            }
156            if comments_full {
157                continue;
158            }
159            // Text that trims to nothing is not kept, so `retained` never sees it: charge what is
160            // read, so many comments sharing one huge blank string stop being decoded over and over.
161            let raw: usize = [&b"T"[..], b"Contents", b"M", b"NM", b"State", b"DA"]
162                .iter()
163                .filter_map(|k| match d.get(k).map(|o| self.resolve(o)) {
164                    Ok(Object::String(b, _) | Object::Name(b)) => Some(b.len()),
165                    _ => None,
166                })
167                .fold(0, usize::saturating_add);
168            // (charged once: below, by the larger of these raw bytes and what the annotation keeps)
169            if kept.saturating_add(raw) > MAX_ANNOTATION_BYTES {
170                comments_full = true;
171                continue;
172            }
173            let rect = rect4(self.resolve(d.get(b"Rect").unwrap_or(&Object::Null)));
174            let c = match d.get(b"C").map(|o| self.resolve(o)) {
175                Ok(Object::Array(c)) => Some(c),
176                _ => None,
177            };
178            // Three numbers, some given by reference, with a non-number read as 0.
179            let rgb = c.filter(|c| c.len() == 3).map(|c| {
180                let f = |i: usize| c.get(i).map_or(0.0, |o| self.resolve(o).as_float().unwrap_or(0.0));
181                [f(0), f(1), f(2)]
182            });
183            // Other comments list the stroke colour as pdfcraft-annot draws it and lists it
184            // (`summaries`): grey, CMYK, a non-number among four entries read as 0.
185            let mut color = if subtype == "FreeText" { rgb } else { c.and_then(|c| drawn_color(c)).or(rgb) };
186            // A text box's /C is its background; its text colour is in /DA.
187            if subtype == "FreeText"
188                && let Some(da) = self.text(d, b"DA")
189            {
190                let t: Vec<&str> = da.split_whitespace().collect();
191                if let Some(i) = t.iter().position(|x| *x == "rg")
192                    && i >= 3
193                {
194                    let f = |k: usize| t[k].parse::<f32>().unwrap_or(0.0);
195                    color = Some([f(i - 3), f(i - 2), f(i - 1)]);
196                }
197            }
198            // Only usable quads, by the rule redaction apply reads them with (keep in step with
199            // `pdfcraft_annot::summaries`): a malformed one is not drawn as zeros.
200            let quads = match d.get(b"QuadPoints").map(|o| self.resolve(o)) {
201                Ok(Object::Array(q)) => {
202                    let entries = q.iter().map(|o| self.resolve(o).as_float().ok().map(f64::from));
203                    crate::geom::quad_points::usable_quads(entries).into_iter().map(|quad| quad.map(|v| v as f32)).collect()
204                }
205                _ => Vec::new(),
206            };
207            let in_reply_to = d.get(b"IRT").ok().and_then(|o| self.dict(o)).and_then(|p| self.text(p, b"NM"));
208            let parent = d.get(b"IRT").ok().and_then(|o| self.resolved_id(o)).and_then(|r| slots.get(&r).copied());
209            let fill_sign = self.is_fill_sign(d, &subtype);
210            // Keep in step with `pdfcraft_annot::summaries`.
211            let key: &[u8] = if subtype == "Line" { b"L" } else { b"Vertices" };
212            let points = match d.get(key).map(|o| self.resolve(o)) {
213                Ok(Object::Array(v)) if matches!(subtype.as_str(), "Line" | "PolyLine" | "Polygon") => {
214                    let v: Vec<f32> = v.iter().map(|o| o.as_float().unwrap_or(0.0)).collect();
215                    let pts: Vec<[f32; 2]> = v.as_chunks::<2>().0.to_vec();
216                    if subtype == "Line" && pts.len() != 2 { Vec::new() } else { pts }
217                }
218                _ => Vec::new(),
219            };
220            // A callout: its arrow point and knee, and its text box (keep in step with
221            // `pdfcraft_annot::summaries`).
222            let (points, text_box) = match self.callout(d, rect) {
223                Some((lead, text_box)) if subtype == "FreeText" => (lead, Some(text_box)),
224                _ => (points, None),
225            };
226            // Only the comment's own pop-up: a pop-up annotation whose /Parent, if any, is a
227            // reference to this comment (keep in step with `pdfcraft_annot::summaries`).
228            let own = self.resolved_id(a);
229            let is_own = |p: &&Dictionary| {
230                self.name(p, b"Subtype").as_deref() == Some("Popup")
231                    && p.get(b"Parent")
232                        .map_or(true, |parent| parent.as_reference().is_ok() && self.resolved_id(parent).is_some_and(|id| Some(id) == own))
233            };
234            // A pop-up without a `/Rect` of four numbers finite as `f32` is none, as
235            // `pdfcraft_annot::summaries` reads it: both lists of the comments agree. Its corners
236            // may come in any order.
237            let popup = d.get(b"Popup").ok().and_then(|o| self.dict(o)).filter(is_own).and_then(|p| {
238                let rect = self.resolve(p.get(b"Rect").ok()?).as_array().ok()?;
239                let nums: Vec<f32> = rect.iter().filter_map(|v| self.resolve(v).as_float().ok()).collect();
240                let [x0, y0, x1, y1] = nums[..] else { return None };
241                if !nums.iter().all(|v| v.is_finite()) {
242                    return None;
243                }
244                Some(Popup {
245                    rect: [x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)],
246                    open: p.get(b"Open").ok().and_then(|o| self.resolve(o).as_bool().ok()).unwrap_or(false),
247                })
248            });
249            let annotation = Annotation {
250                page,
251                subtype,
252                author: self.text(d, b"T"),
253                // Acrobat ends lines with CR, which the interface would draw as nothing (keep
254                // in step with `pdfcraft_annot::summaries`).
255                contents: self.text(d, b"Contents").map(|s| s.replace("\r\n", "\n").replace('\r', "\n")),
256                modified: self.text(d, b"M").map(|m| pretty_date(&m)),
257                name: self.text(d, b"NM"),
258                in_reply_to,
259                parent,
260                grouped: self.name(d, b"RT").as_deref() == Some("Group"),
261                rect,
262                color,
263                index,
264                state: state_or_default(self.text(d, b"State"), self.text(d, b"StateModel").as_deref()),
265                quads,
266                locked: d.get(b"F").ok().and_then(|f| self.resolve(f).as_i64().ok()).unwrap_or(0) & 128 != 0,
267                intent: self.name(d, b"IT"),
268                fill_sign,
269                no_rotate: d.get(b"F").is_ok_and(|f| draw_flags(self.resolve(f)) & 16 != 0),
270                points,
271                text_box,
272                popup,
273            };
274            kept = kept.saturating_add(retained(&annotation).max(raw));
275            if kept > MAX_ANNOTATION_BYTES {
276                comments_full = true;
277                continue;
278            }
279            info.annotations.push(annotation);
280        }
281        info.annotations.sort_by(|a, b| a.page.cmp(&b.page).then(b.rect[3].total_cmp(&a.rect[3])));
282        link_replies(&mut info.annotations);
283        fill_group_text(&mut info.annotations);
284    }

286    /// A callout's arrow point and knee (`/CL`, four or six numbers, without the end that meets
287    /// the text box) and its text box (`/Rect`, here `rect`, inset by `/RD`); ISO 32000-2
288    /// Table 177. `None` when either is missing or unusable.
289    fn callout(&self, d: &Dictionary, rect: [f32; 4]) -> Option<(Vec<[f32; 2]>, [f32; 4])> {
290        let nums = |key: &[u8]| -> Option<Vec<f32>> {
291            match self.resolve(d.get(key).ok()?) {
292                Object::Array(v) => v.iter().map(|o| self.resolve(o).as_float().ok()).collect(),
293                _ => None,
294            }
295        };
296        let cl = nums(b"CL").filter(|l| matches!(l.len(), 4 | 6) && l.iter().all(|x| x.is_finite()))?;
297        let rd = nums(b"RD").filter(|v| v.len() == 4 && v.iter().all(|x| x.is_finite() && *x >= 0.0))?;
298        let tb = [rect[0] + rd[0], rect[1] + rd[1], rect[2] - rd[2], rect[3] - rd[3]];
299        if tb[2] - tb[0] < 1.0 || tb[3] - tb[1] < 1.0 {
300            return None;
301        }
302        let (_, lead) = cl.as_chunks::<2>().0.split_last()?;
303        Some((lead.to_vec(), tb))
304    }

306    /// The link annotation `d`, at position `index` of page `page`'s `/Annots`: where its `/Dest`
307    /// goes, else what its action `/A` does ([`Self::action_target`]). `None` when it has no
308    /// rectangle, or a destination or action that leads nowhere. `base` is the catalog's
309    /// `/URI /Base`, which a relative URI action is resolved against. `unfit_quads` holds the
310    /// indirect `/QuadPoints` arrays already found too large for the budget.
311    fn link(
312        &self,
313        page: usize,
314        index: usize,
315        d: &Dictionary,
316        base: Option<&str>,
317        text_budget: &mut usize,
318        unfit_quads: &mut HashSet<ObjectId>,
319    ) -> Option<Link> {
320        let rect = rect4(self.resolve(d.get(b"Rect").ok()?));
321        let target = if let Ok(dest) = d.get(b"Dest") {
322            let (page, view) = self.dest(dest, 0)?;
323            LinkTarget::Page(page, view)
324        } else {
325            match self.action_target(d.get(b"A").ok().and_then(|a| self.dict(a))?, text_budget)? {
326                LinkTarget::Uri(uri) => match crate::render::resolve_uri(base, &uri) {
327                    // The raw address is spent already. What the base adds is spent too, or the
328                    // link keeps the raw address.
329                    Some(resolved) if resolved.len().saturating_sub(uri.len()) <= *text_budget => {
330                        *text_budget = text_budget.saturating_sub(resolved.len().saturating_sub(uri.len()));
331                        LinkTarget::Uri(resolved)
332                    }
333                    _ => LinkTarget::Uri(uri),
334                },
335                other => other,
336            }
337        };
338        // Not read at all once no quadrilateral would fit what is left of the budget, or when this
339        // shared array already did not fit: many links can share one long array, and reading it
340        // again for each would cost its length every time (review on #4474).
341        let shared = d.get(b"QuadPoints").ok().and_then(|o| o.as_reference().ok());
342        let readable = *text_budget >= std::mem::size_of::<[f32; 8]>() && shared.is_none_or(|id| !unfit_quads.contains(&id));
343        let quads = match d.get(b"QuadPoints").map(|o| self.resolve(o)) {
344            // Read no further than one value past the cap: a longer array isn't used.
345            Ok(Object::Array(q)) if readable && q.len() <= crate::render::MAX_LINK_QUADS.saturating_mul(8) => {
346                #[cfg(test)]
347                QUAD_ARRAYS_READ.with(|n| n.set(n.get() + 1));
348                let values: Vec<Option<f32>> = q.iter().map(|o| self.resolve(o).as_float().ok()).collect();
349                let quads = crate::render::link_quads(&values, rect);
350                // Many links can share one long array, which each would otherwise copy. Past the
351                // budget a link is active over its whole `/Rect`.
352                let bytes = std::mem::size_of_val(quads.as_slice());
353                if bytes <= *text_budget {
354                    *text_budget = text_budget.saturating_sub(bytes);
355                    quads
356                } else {
357                    if let Some(id) = shared {
358                        unfit_quads.insert(id);
359                    }
360                    Vec::new()
361                }
362            }
363            _ => Vec::new(),
364        };
365        Some(Link { page, index, rect, quads, target })
366    }

368    /// Lists the link annotation `d`, and notes the JavaScript it runs. JavaScript is read from the
369    /// action, not the listed link: once the text budget is spent the action's name is not kept.
370    fn collect_link(&self, info: &mut DocInfo, at: (usize, usize), d: &Dictionary, text_budget: &mut usize, unfit_quads: &mut HashSet<ObjectId>) {
371        info.has_javascript |= self.runs_javascript(d);
372        if let Some(link) = self.link(at.0, at.1, d, info.uri_base.as_deref(), text_budget, unfit_quads) {
373            if matches!(&link.target, LinkTarget::Other(s) if s == "JavaScript") {
374                info.has_javascript = true;
375            }
376            info.links.push(link);
377        }

385    /// What an action does when a link or a bookmark runs it. Only a go-to's `/D` is a place in
386    /// this document; a remote go-to's names a page or destination of another file (ISO 32000-2
387    /// §12.6.4.3). `None` for an action without a type, or a go-to whose destination is not here.
388    /// A web address is spent from `text_budget` (see [`MAX_LINK_ATTACHMENT_TEXT_BYTES`]).
389    pub(super) fn action_target(&self, a: &Dictionary, text_budget: &mut usize) -> Option<LinkTarget> {
390        Some(match self.name(a, b"S").as_deref()? {
391            "GoTo" => {
392                let (page, view) = self.dest(a.get(b"D").ok()?, 0)?;
393                LinkTarget::Page(page, view)
394            }
395            "URI" => {
396                let uri = a.get(b"URI").ok().and_then(|u| self.resolve(u).as_str().ok())?;
397                match spend(uri, text_budget, |b| String::from_utf8_lossy(b).into_owned()) {
398                    Some(uri) => LinkTarget::Uri(uri),
399                    None => LinkTarget::Other("URI".into()),
400                }
401            }
402            "SetOCGState" => self.layer_state(a),
403            "GoToR" => self.remote_go_to(a).unwrap_or_else(|| LinkTarget::Other("GoToR".into())),
404            "Launch" => self.launch_pdf(a).unwrap_or_else(|| LinkTarget::Other("Launch".into())),
405            other => LinkTarget::Other(spend(other.as_bytes(), text_budget, |b| String::from_utf8_lossy(b).into_owned()).unwrap_or_default()),
406        })
407    }

409    fn remote_go_to(&self, action: &Dictionary) -> Option<LinkTarget> {
410        let file = self.action_file(action)?;
411        let destination = match action.get(b"D") {
412            Err(_) => RemoteDestination::Default,
413            Ok(o) => match self.resolve(o) {
414                Object::Array(items) => {
415                    let page = items.first().and_then(|o| self.resolve(o).as_i64().ok()).and_then(|n| usize::try_from(n).ok())?;
416                    let kind = items.get(1).map(|o| self.resolve(o)).and_then(|o| o.as_name().ok());
417                    let operands: [Option<f32>; 4] = std::array::from_fn(|i| items.get(i + 2).and_then(|o| self.resolve(o).as_float().ok()));
418                    RemoteDestination::Page { page, view: DestView::from_operands(kind, &operands) }
419                }
420                Object::Name(name) if name.len() <= MAX_REMOTE_DESTINATION_NAME_BYTES => {
421                    let name = String::from_utf8_lossy(name).into_owned();
422                    if name.is_empty() || name.len() > MAX_REMOTE_DESTINATION_NAME_BYTES {
423                        return None;
424                    }
425                    RemoteDestination::Named(name)
426                }
427                Object::String(bytes, _) if bytes.len() <= MAX_REMOTE_DESTINATION_NAME_BYTES => {
428                    let name = text_string(bytes);
429                    if name.is_empty() || name.len() > MAX_REMOTE_DESTINATION_NAME_BYTES {
430                        return None;
431                    }
432                    RemoteDestination::Named(name)
433                }
434                _ => return None,
435            },
436        };
437        Some(LinkTarget::RemoteGoTo { launch: false, file, destination, new_window: self.new_window(action) })
438    }

440    /// The non-empty local file an action's `/F` names (a string or a file specification), shared
441    /// by GoToR and Launch. A URL file system (`/FS /URL`) names no local file.
442    fn action_file(&self, action: &Dictionary) -> Option<String> {
443        if self.name(action, b"FS").as_deref() == Some("URL") {
444            return None;
445        }
446        let file = match self.resolve(action.get(b"F").ok()?) {
447            Object::String(bytes, _) => limited_pdf_string(bytes, MAX_REMOTE_FILE_SPEC_BYTES)?,
448            Object::Dictionary(spec) => {
449                if self.name(spec, b"FS").as_deref() == Some("URL") {
450                    return None;
451                }
452                spec.get(b"UF")
453                    .ok()
454                    .and_then(|o| self.pdf_text(o, MAX_REMOTE_FILE_SPEC_BYTES))
455                    .or_else(|| spec.get(b"F").ok().and_then(|o| self.pdf_text(o, MAX_REMOTE_FILE_SPEC_BYTES)))?
456            }
457            _ => return None,
458        };
459        (!file.is_empty()).then_some(file)
460    }

462    fn new_window(&self, action: &Dictionary) -> Option<bool> {
463        match action.get(b"NewWindow").map(|o| self.resolve(o)) {
464            Ok(Object::Boolean(value)) => Some(*value),
465            _ => None,
466        }
467    }

469    /// A Launch action (ISO 32000-2 §12.6.4.6, Table 207) whose `/F` names a `.pdf` file, read as
470    /// the remote Go-To to that file's first page that it amounts to (#4168). Launching anything
471    /// else, or with the deprecated `/Win` launch parameters (an application, an operation, its
472    /// arguments), is not followed: a document must not start programs. `\` separators, which
473    /// Windows writers put in such paths, are read as `/` (§7.11.2.1 separates components with
474    /// `/`) when the path has no `/` of its own. Launch has no `/D`, so one present (written
475    /// for GoToR, or junk) is ignored rather than checked as a destination.
476    fn launch_pdf(&self, action: &Dictionary) -> Option<LinkTarget> {
477        if action.has(b"Win") {
478            return None;
479        }
480        let file = self.action_file(action)?;
481        let new_window = self.new_window(action);
482        let file = if file.contains('/') { file } else { file.replace('\\', "/") };
483        let pdf = file.rsplit('/').next().is_some_and(|name| name.len() > 4 && name.to_ascii_lowercase().ends_with(".pdf"));
484        pdf.then_some(LinkTarget::RemoteGoTo { launch: true, file, destination: RemoteDestination::Default, new_window })
485    }

488        match self.resolve(o) {
489            Object::String(bytes, _) => limited_pdf_string(bytes, max_bytes),
490            _ => None,
491        }

494    /// A set-OCG-state action: `/State` is `ON`, `OFF` or `Toggle`, each followed by the groups
495    /// it applies to; `/PreserveRB` is true unless it is `false`.
496    fn layer_state(&self, a: &Dictionary) -> LinkTarget {
497        let mut changes = Vec::new();
498        if let Ok(Object::Array(items)) = a.get(b"State").map(|o| self.resolve(o)) {
499            let mut op = None;
500            for item in items.iter().take(MAX_LAYER_STATE) {
501                match item {
502                    Object::Name(n) => {
503                        op = match n.as_slice() {
504                            b"ON" => Some(LayerOp::On),
505                            b"OFF" => Some(LayerOp::Off),
506                            b"Toggle" => Some(LayerOp::Toggle),
507                            _ => None,
508                        }
509                    }
510                    Object::Reference(id) => changes.extend(op.map(|op| (op, *id))),
511                    _ => {}
512                }
513            }
514        }
515        let preserve_rb = !matches!(a.get(b"PreserveRB").map(|o| self.resolve(o)), Ok(Object::Boolean(false)));
516        LinkTarget::SetLayers { changes, preserve_rb }
517    }

521    /// Whether `o` or one of its descendants under `/Kids` carries an action (`/A` or `/AA`). It
522    /// walks the same tree `field` does, bounded the same way: a field left out of the listing for
523    /// want of text allowance can still have its action on a widget several levels down.
524    pub(super) fn has_action(&self, o: &Object, depth: u32) -> bool {
525        self.has_action_seen(o, depth, &mut HashMap::new())
526    }
527
528    /// The walk behind [`Self::has_action`]. Each indirect descendant is read again only from a
529    /// shallower depth than before (the depth limit may have cut its subtree): a field tree whose
530    /// `/Kids` repeat the same objects would otherwise be walked exponentially, and this reads each
531    /// object at most once per depth.
532    fn has_action_seen(&self, o: &Object, depth: u32, seen: &mut HashMap<ObjectId, u32>) -> bool {
533        if depth > 32 {
534            return false;
535        }
536        if let Object::Reference(id) = o {
537            if seen.get(id).is_some_and(|&first| first <= depth) {
538                return false;
539            }
540            seen.insert(*id, depth);
541        }
542        let Some(d) = self.dict(o) else { return false };
543        if self.carries_action(d) {
544            return true;
545        }
546        let kids = match d.get(b"Kids").map(|k| self.resolve(k)) {
547            Ok(Object::Array(k)) => k,
548            _ => return false,
549        };
550        kids.iter().any(|k| self.has_action_seen(k, depth + 1, seen))
551    }
552
553    /// Whether `/A` or `/AA` here holds an action. An editor can leave an empty dictionary (or
554    /// array) behind, and that is no action: a document with none must not report JavaScript. An
555    /// `/A` is an action when it carries `/S`; an `/AA` is one when an event maps to such an action.
556    fn carries_action(&self, d: &Dictionary) -> bool {
557        let action = |o: &Object| matches!(self.resolve(o), Object::Dictionary(a) if a.get(b"S").is_ok());
558        [&b"A"[..], &b"AA"[..]].iter().any(|k| match d.get(k) {
559            Ok(o) => match self.resolve(o) {
560                Object::Dictionary(x) => x.get(b"S").is_ok() || x.iter().any(|(_, v)| action(v)),
561                Object::Array(a) => a.iter().any(action),
562                _ => false,
563            },
564            Err(_) => false,
565        })
566    }

568    pub(super) fn field(
569        &self,
570        o: &'a Object,
571        parent: Option<(&str, Option<String>, Option<i64>)>,
572        out: &mut Vec<Field>,
573        walk: &mut FieldWalk,
574        depth: u32,
575    ) {
576        if depth > 32 {
577            return;
578        }
579        if let Object::Reference(id) = o
580            && !walk.seen.insert(*id)
581        {
582            return;
583        }
584        let Some(d) = self.dict(o) else { return };
585        let (parent_name, parent_ft, parent_ff) = parent.unwrap_or(("", None, None));
586        // Non-terminal: has kids that are fields (they carry /T). Widgets-only kids are terminal.
587        let mut kids: Vec<Object> = match d.get(b"Kids").map(|k| self.resolve(k)) {
588            Ok(Object::Array(k)) => k.clone(),
589            _ => Vec::new(),
590        };
591        // A field left unlisted for want of text allowance still has its actions counted, through
592        // its whole subtree: an action can sit on a descendant widget, not only on the field.
593        let unlisted = || {
594            if self.has_action(o, 0) {
595                self.field_actions.set(true);
596            }
597        };
598        // A /T that does not fit the allowance lists no field (not one under its parent's name).
599        let Ok(partial) = self.field_text_within(d, b"T") else {
600            unlisted();
601            return;
602        };
603        let full = match (&partial, parent_name.is_empty()) {
604            (Some(t), true) => t.clone(),
605            (Some(t), false) => format!("{parent_name}.{t}"),
606            (None, _) => parent_name.to_string(),
607        };
608        // Every field copies its parent's full name: charge it, and list no more once it is spent.
609        if !self.field_charge(full.len()) {
610            unlisted();
611            return;
612        }
613        let ft = self.name(d, b"FT").or(parent_ft);
614        let ff = d.get(b"Ff").ok().and_then(|x| x.as_i64().ok()).or(parent_ff);
615        let (field_kids, widget_kids): (Vec<Object>, Vec<Object>) =
616            kids.iter().cloned().partition(|k| self.dict(k).is_some_and(|kd| kd.get(b"T").is_ok()));
617        let field_kids: Vec<_> = field_kids.iter().filter(|k| !self.not_a_child(walk, o, k)).collect();
618        if !field_kids.is_empty() {
619            let here = match o {
620                Object::Reference(id) => Some(*id),
621                _ => None,
622            };
623            walk.path.extend(here);
624            for k in field_kids {
625                // A fully qualified name is built up the field's own `/Parent` chain (ISO 32000-2
626                // §12.7.4.2): a kid that doesn't name this field as its `/Parent` is named from its
627                // own chain, as the form reader names it.
628                let parent = match self.dict(k).map(|kd| kd.get(b"Parent")) {
629                    Some(Ok(Object::Reference(p))) => Some(*p),
630                    _ => None,
631                };
632                if parent == here {
633                    self.field(k, Some((&full, ft.clone(), ff)), out, walk, depth + 1);
634                } else {
635                    let own = parent.map(|p| self.qualified_name(p)).unwrap_or_default();
636                    self.field(k, Some((&own, ft.clone(), ff)), out, walk, depth + 1);
637                }
638            }
639            if here.is_some() {
640                walk.path.pop();
641            }
642            // "A field's children in the hierarchy may also include widget annotations" (ISO
643            // 32000-2 §12.7.4): widget annotations listed beside named kids are still this
644            // field's widgets, listed as a field of this name, as the form reader reads them.
645            kids.retain(|k| self.dict(k).is_some_and(|kd| kd.get(b"T").is_err() && self.name(kd, b"Subtype").as_deref() == Some("Widget")));
646            if kids.is_empty() {
647                return;
648            }
649        }
650        let flags = ff.unwrap_or(0);
651        let kind = match ft.as_deref() {
652            Some("Tx") => FieldKind::Text,
653            Some("Btn") if flags & (1 << 16) != 0 => FieldKind::PushButton,
654            Some("Btn") if flags & (1 << 15) != 0 => FieldKind::Radio,
655            Some("Btn") => FieldKind::CheckBox,
656            Some("Ch") if flags & (1 << 17) != 0 => FieldKind::Combo,
657            Some("Ch") => FieldKind::List,
658            Some("Sig") => FieldKind::Signature,
659            _ => FieldKind::Unknown,
660        };
661        let value = match d.get(b"V").map(|v| self.resolve(v)) {
662            Ok(Object::Name(n)) => self.field_spend(n, |n| String::from_utf8_lossy(n).into_owned()),
663            Ok(Object::String(bytes, _)) => self.field_spend(bytes, text_string),
664            Ok(Object::Array(a)) => {
665                let mut shown = Vec::new();
666                let mut dropped = false;
667                for x in a {
668                    let text = match self.resolve(x) {
669                        Object::String(bytes, _) => self.field_spend(bytes, text_string),
670                        // A name entry (`/V [/Opt1 (b)]`) is a selection too, as the forms reader
671                        // takes it; leaving it out reported a partial selection as complete.
672                        Object::Name(n) => self.field_spend(n, |n| String::from_utf8_lossy(n).into_owned()),
673                        _ => continue,
674                    };
675                    // The ", " between selections is charged to the allowance like the text.
676                    match text {
677                        Some(text) if shown.is_empty() || self.field_charge(2) => shown.push(text),
678                        _ => dropped = true,
679                    }
680                }
681                // A selected value the allowance could not hold shows as an omission rather than
682                // as part of a complete selection (its few bytes are charged, fit or not).
683                let mut value = shown.join(", ");
684                if dropped {
685                    self.field_charge(5);
686                    if !value.is_empty() {
687                        value.push_str(", ");
688                    }
689                    value.push('…');
690                }
691                Some(value)
692            }
693            Ok(Object::Dictionary(_)) if kind == FieldKind::Signature => Some("signed".into()),
694            _ => None,
695        };
696        // The widget is this dict (merged field/widget) or its first kid. A merged dict that also
697        // lists widget kids has them as its widgets (ISO 32000-2 §12.7.4), as the form reader has
698        // it. With no widget among its kids, it is the first page widget naming this field as its
699        // `/Parent`: a child names its parent (ISO 32000-2 §12.7.4.1, Table 226), as MuPDF and
700        // PDFium read it.
701        let merged = d.get(b"Rect").is_ok();
702        let named = if merged || !widget_kids.is_empty() { None } else { self.widget_naming(o) };
703        let widget_kid = widget_kids.iter().filter_map(|k| self.dict(k)).find(|kd| self.name(kd, b"Subtype").as_deref() == Some("Widget"));
704        let widget = match widget_kid {
705            Some(w) => Some(w),
706            None if merged => Some(d),
707            None => widget_kids.first().and_then(|k| self.dict(k)).or_else(|| named.and_then(|id| self.doc.get_dictionary(id).ok())),
708        };
709        let page = widget
710            .and_then(|w| w.get(b"P").ok())
711            .and_then(|p| self.page_of(p))
712            .or_else(|| self.find_widget_page(o, &kids))
713            .or_else(|| named.and_then(|id| self.annotation_pages.get_or_init(|| self.annotation_pages()).get(&id).copied()));
714        let has_actions = self.carries_action(d)
715            || widget.is_some_and(|w| self.carries_action(w))
716            || kids.iter().any(|k| self.dict(k).is_some_and(|w| self.carries_action(w)));
717        let rect = widget.and_then(|w| w.get(b"Rect").ok()).map(|r| rect4(self.resolve(r)));
718        out.push(Field { name: full, kind, value, page, tooltip: self.field_text(d, b"TU"), has_actions, rect });
719    }
720
721    /// Whether `k`, listed in the `/Kids` of field `o`, is not a child of `o` after all: `o` itself
722    /// or a field `o` descends from (a loop), or a root field (one `/Fields` lists) that doesn't
723    /// name `o` as its `/Parent`. Root fields have no ancestors (ISO 32000-2 §12.7.3, Table 224),
724    /// and a child names its parent (§12.7.4.1, Table 226).
725    fn not_a_child(&self, walk: &FieldWalk, o: &Object, k: &Object) -> bool {
726        let Object::Reference(k) = k else { return false };
727        let parent = self.dict(&Object::Reference(*k)).and_then(|d| match d.get(b"Parent") {
728            Ok(Object::Reference(p)) => Some(*p),
729            _ => None,
730        });
731        let o = if let Object::Reference(o) = o { Some(*o) } else { None };
732        o == Some(*k) || walk.path.contains(k) || (walk.roots.contains(k) && (o.is_none() || parent != o))
733    }

735    /// The first page widget that names field `o` as its `/Parent` and is not a field of its own
736    /// (no `/T`), in page-annotation order.
737    fn widget_naming(&self, o: &Object) -> Option<ObjectId> {
738        let Object::Reference(id) = o else { return None };
739        self.parent_widgets.get_or_init(|| self.parent_widgets()).get(id).copied()
740    }

742    /// The first page widget (no `/T`) naming each field as its `/Parent`, in page-annotation order.
743    fn parent_widgets(&self) -> HashMap<ObjectId, ObjectId> {
744        let mut first = HashMap::new();
745        for &pid in &self.page_ids {
746            let Some(annots) = self.doc.get_dictionary(pid).ok().and_then(|d| d.get(b"Annots").ok()) else { continue };
747            let Ok(annots) = self.resolve(annots).as_array() else { continue };
748            for a in annots {
749                let Object::Reference(id) = a else { continue };
750                let Ok(w) = self.doc.get_dictionary(*id) else { continue };
751                if self.name(w, b"Subtype").as_deref() != Some("Widget") || w.get(b"T").is_ok() {
752                    continue;
753                }
754                if let Ok(Object::Reference(parent)) = w.get(b"Parent") {
755                    first.entry(*parent).or_insert(*id);
756                }
757            }
758        }
759        first
760    }

762    /// The first page whose `/Annots` lists the field or one of its kids.
763    fn find_widget_page(&self, field: &Object, kids: &[Object]) -> Option<usize> {
764        let pages = self.annotation_pages.get_or_init(|| self.annotation_pages());
765        std::iter::once(field).chain(kids).filter_map(|o| if let Object::Reference(id) = o { pages.get(id).copied() } else { None }).min()
766    }

768    /// Each annotation listed in a page's `/Annots`, with the first page that lists it.
769    fn annotation_pages(&self) -> HashMap<ObjectId, usize> {
770        let mut pages = HashMap::new();
771        for (idx, &pid) in self.page_ids.iter().enumerate() {
772            let Some(annots) = self.doc.get_dictionary(pid).ok().and_then(|d| d.get(b"Annots").ok()) else { continue };
773            let Ok(annots) = self.resolve(annots).as_array() else { continue };
774            for a in annots {
775                if let Object::Reference(id) = a {
776                    pages.entry(*id).or_insert(idx);
777                }
778            }
779        }
780        pages
781    }

783    /// The fully qualified name of the field `id`: the partial names (`/T`) up its `/Parent`
784    /// chain, joined with periods; empty when none has one or the chain reaches an object that
785    /// isn't a dictionary, as the form reader has it. A `/Parent` loop ends the walk.
786    fn qualified_name(&self, id: ObjectId) -> String {
787        let mut parts = Vec::new();
788        let mut visited = HashSet::new();
789        let mut cur = Some(id);
790        while let Some(c) = cur {
791            if visited.len() > 64 || !visited.insert(c) {
792                break;
793            }
794            let o = Object::Reference(c);
795            let Some(d) = self.dict(&o) else { return String::new() };
796            parts.extend(self.text(d, b"T"));
797            cur = match d.get(b"Parent") {
798                Ok(Object::Reference(p)) => Some(*p),
799                _ => None,
800            };
801        }
802        parts.reverse();
803        parts.join(".")
804    }

806    /// The topmost field above widget `id` (itself without a `/Parent` that is a field), or `None` when it or an
807    /// ancestor is already reachable from the `/Fields` tree, or a `/Parent` loop hides the top.
808    pub(super) fn unlisted_field(&self, id: ObjectId, seen: &HashSet<ObjectId>) -> Option<ObjectId> {
809        let mut top = id;
810        let mut visited = HashSet::from([id]);
811        loop {
812            let o = Object::Reference(top);
813            let d = self.dict(&o)?;
814            // A `/Parent` naming an object the file doesn't hold, or no dictionary, reads as
815            // `null`, the same as no parent (ISO 32000-2 §7.3.9).
816            let parent = match d.get(b"Parent") {
817                Ok(Object::Reference(p)) if self.dict(&Object::Reference(*p)).is_some() => Some(*p),
818                _ => None,
819            };
820            let Some(p) = parent else {
821                return if seen.contains(&top) { None } else { Some(top) };
822            };
823            if !visited.insert(p) || visited.len() > 64 {
824                return None;
825            }
826            if seen.contains(&p) {
827                return None;
828            }
829            top = p;
830        }
831    }

843/// An annotation's `/F` as the renderer reads it: a real is truncated, and a value that isn't a
844/// `u32` counts as no flags.
845fn draw_flags(o: &Object) -> u32 {

847        Object::Integer(i) => *i,
848        Object::Real(r) => *r as i64,
849        _ => 0,

852}

854fn rect4(o: &Object) -> [f32; 4] {
855    match o {
856        Object::Array(a) if a.len() == 4 => {
857            let v: Vec<f32> = a.iter().map(|x| x.as_float().unwrap_or(0.0)).collect();
858            [v[0].min(v[2]), v[1].min(v[3]), v[0].max(v[2]), v[1].max(v[3])]
859        }
860        _ => [0.0; 4],
861    }
