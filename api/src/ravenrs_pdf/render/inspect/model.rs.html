98pub struct PageInfo {
99    /// Displayed size in points, after `/Rotate` and `/UserUnit`.
100    pub width: f32,
101    pub height: f32,
102    /// Page label (`/PageLabels`), falling back to the 1-based page number.
103    pub label: String,
104    /// Effective crop box in user space [x0, y0, x1, y1] (the visible region).
105    pub crop: [f32; 4],
106    /// Clockwise page rotation in degrees (0, 90, 180, 270).
107    pub rotation: u16,
108}
109
110impl PageInfo {
111    /// A point in view space (points, y down, after `/Rotate`; what the text layer uses) →
112    /// PDF user space.
113    pub fn view_to_user(&self, x: f32, y: f32) -> [f32; 2] {
114        let [cx0, cy0, cx1, cy1] = self.crop;
115        let (a, b) = (x / self.width.max(1e-3), y / self.height.max(1e-3));
116        let (u, v) = match self.rotation {
117            90 => (b, 1.0 - a),
118            180 => (1.0 - a, 1.0 - b),
119            270 => (1.0 - b, a),
120            _ => (a, b),
121        };
122        [cx0 + u * (cx1 - cx0), cy1 - v * (cy1 - cy0)]
123    }
124
125    /// PDF user space → view space (the inverse of [`Self::view_to_user`]).
126    pub fn user_to_view(&self, x: f32, y: f32) -> [f32; 2] {
127        let [cx0, cy0, cx1, cy1] = self.crop;
128        let (u, v) = ((x - cx0) / (cx1 - cx0).max(1e-3), (cy1 - y) / (cy1 - cy0).max(1e-3));
129        let (a, b) = match self.rotation {
130            90 => (1.0 - v, u),
131            180 => (1.0 - u, 1.0 - v),
132            270 => (v, 1.0 - u),
133            _ => (u, v),
134        };
135        [a * self.width, b * self.height]
136    }
137
138    /// A point as displayed (points from the page's top-left, y down, before any view rotation)
139    /// → the `/XYZ` position `[left, top]` in user space that puts it at the window's top-left.
140    /// Only a coordinate that is given is pinned: a bookmark made at the top of the window
141    /// leaves the horizontal position alone. On a page turned a quarter, the displayed `y` decides
142    /// the user `left` and the displayed `x` the user `top` (the inverse of [`Self::dest_fraction`]).
143    /// A value that is not finite counts as not given.
144    pub fn xyz_for_view(&self, x: Option<f32>, y: Option<f32>) -> [Option<f32>; 2] {
145        let (x, y) = (x.filter(|v| v.is_finite()), y.filter(|v| v.is_finite()));
146        let [ux, uy] = self.view_to_user(x.unwrap_or(0.0), y.unwrap_or(0.0));
147        if self.rotation.is_multiple_of(180) { [x.map(|_| ux), y.map(|_| uy)] } else { [y.map(|_| ux), x.map(|_| uy)] }
148    }
149
150    /// A destination's user-space point (each coordinate possibly unspecified) → the same point
151    /// as a fraction of the displayed page `[x, y]` (0..1 from its top-left corner, y down),
152    /// after `/Rotate` and a further clockwise `view_rotation` (0, 90, 180 or 270). On a page
153    /// turned a quarter, the user `x` decides the displayed `y` and vice versa, so each output
154    /// is `None` when the input it comes from is (or is not finite). Points off the page are
155    /// clamped to its edge.
156    pub fn dest_fraction(&self, x: Option<f32>, y: Option<f32>, view_rotation: u16) -> [Option<f32>; 2] {
157        let [cx0, cy0, cx1, cy1] = self.crop;
158        let frac = |v: f32| if v.is_finite() { Some(v.clamp(0.0, 1.0)) } else { None };
159        // Unrotated fractions across (u, from the crop's left) and down (v, from its top).
160        let u = x.and_then(|x| frac((x - cx0) / (cx1 - cx0).max(1e-3)));
161        let v = y.and_then(|y| frac((cy1 - y) / (cy1 - cy0).max(1e-3)));
162        let flip = |f: Option<f32>| f.map(|f| 1.0 - f);
163        match (u32::from(self.rotation) + u32::from(view_rotation)) % 360 {
164            90 => [flip(v), u],
165            180 => [flip(u), flip(v)],
166            270 => [v, flip(u)],
167            _ => [u, v],
168        }
169    }
170
171    /// A view-space rectangle → the normalized user-space rectangle covering the same area of the
172    /// page (`[x0, y0, x1, y1]`, `x0 <= x1`, `y0 <= y1`). Under `/Rotate` the corners swap roles,
173    /// so a view rectangle's top-left is not in general the user rectangle's `[x0, y1]`.
174    pub fn view_rect_to_user(&self, r: [f32; 4]) -> [f32; 4] {
175        let (a, b) = (self.view_to_user(r[0], r[1]), self.view_to_user(r[2], r[3]));
176        [a[0].min(b[0]), a[1].min(b[1]), a[0].max(b[0]), a[1].max(b[1])]
177    }
178
179    /// A view-space rectangle as a text-markup quad in user space: top-left, top-right,
180    /// bottom-left, bottom-right as read on screen.
181    pub fn view_rect_to_quad(&self, r: [f32; 4]) -> [f64; 8] {
182        let mut q = [0.0; 8];
183        for (i, (x, y)) in [(r[0], r[1]), (r[2], r[1]), (r[0], r[3]), (r[2], r[3])].into_iter().enumerate() {
184            let [ux, uy] = self.view_to_user(x, y);
185            q[2 * i] = ux as f64;
186            q[2 * i + 1] = uy as f64;
187        }
188        q
189    }
190}

192/// A clickable link annotation.
193#[derive(Clone, Debug)]
194pub struct Link {
195    pub page: usize,
196    /// Zero-based position in the page's raw `/Annots` array.
197    pub index: usize,
198    pub rect: [f32; 4],
199    /// Where the link is active when that isn't the whole of `rect`: its `/QuadPoints`, 8 numbers
200    /// each in user space (ISO 32000-2 Table 176). Empty when the link is active over `rect`
201    /// ([`crate::render::link_quads`]).
202    pub quads: Vec<[f32; 8]>,
203    pub target: LinkTarget,
204}
205
206impl Link {
207    /// Whether the user-space point `(x, y)` activates the link: inside one of its quadrilaterals,
208    /// or inside its rectangle when it has none.
209    pub fn contains(&self, x: f32, y: f32) -> bool {
210        let [x0, y0, x1, y1] = self.rect;
211        if self.quads.is_empty() {
212            return x >= x0 && x <= x1 && y >= y0 && y <= y1;
213        }
214        // Every kept quadrilateral lies within `/Rect` give or take the slack `link_quads`
215        // accepts, so a point outside that is in none of them: skip the scan (up to
216        // `MAX_LINK_QUADS` per link, run per link on every pointer move).
217        let s = crate::render::link_area::RECT_SLACK;
218        if !(x >= x0 - s && x <= x1 + s && y >= y0 - s && y <= y1 + s) {
219            return false;
220        }
221        self.quads.iter().any(|q| crate::render::quad_contains(q, x, y))
222    }
223}

267/// Where an explicit destination (ISO 32000-2 §12.3.2.2, Table 149) places its page in the
268/// window. Coordinates are PDF user space; `None` is a `null` (or unusable) operand, which keeps
269/// that part of the view as it is.
270#[derive(Clone, Copy, Debug, Default, PartialEq)]
271pub enum DestView {
272    /// The top of the page at the current zoom: a destination with no position, or one whose
273    /// type is missing or unknown.
274    #[default]
275    Top,
276    /// `/XYZ left top zoom`: the point (left, top) at the window's upper-left corner, at `zoom`
277    /// (1.0 = 100%; 0 or null keeps the current zoom).
278    Xyz { left: Option<f32>, top: Option<f32>, zoom: Option<f32> },
279    /// `/Fit`, and `/FitB` (fit the page's content box; read as the whole page).
280    Fit,
281    /// `/FitH top`, and `/FitBH`: fit the width, with `top` at the window's top edge.
282    FitH { top: Option<f32> },
283    /// `/FitV left`, and `/FitBV`: fit the height, with `left` at the window's left edge.
284    FitV { left: Option<f32> },
285    /// `/FitR left bottom right top`: fit the rectangle `[x0, y0, x1, y1]` (normalized).
286    FitR { rect: [f32; 4] },
287}
288
289impl DestView {
290    /// The view an explicit destination array asks for, from its elements after the page
291    /// (`/XYZ left top zoom`, `/FitH top`, …). Hostile input never fails: a missing or unknown
292    /// type, or a `/FitR` without four usable numbers, is [`DestView::Top`]; a non-numeric,
293    /// non-finite or missing operand is `None`; a zoom that is not positive is `None`.
294    pub fn from_operands(kind: Option<&[u8]>, operands: &[Option<f32>]) -> Self {
295        let n = |i: usize| operands.get(i).copied().flatten().filter(|v| v.is_finite());
296        match kind {
297            Some(b"XYZ") => Self::Xyz { left: n(0), top: n(1), zoom: n(2).filter(|z| *z > 0.0) },
298            Some(b"Fit" | b"FitB") => Self::Fit,
299            Some(b"FitH" | b"FitBH") => Self::FitH { top: n(0) },
300            Some(b"FitV" | b"FitBV") => Self::FitV { left: n(0) },
301            Some(b"FitR") => match (n(0), n(1), n(2), n(3)) {
302                (Some(x0), Some(y0), Some(x1), Some(y1)) => Self::FitR { rect: [x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)] },
303                _ => Self::Top,
304            },
305            _ => Self::Top,
306        }
307    }
308}

383/// How far [`group_primaries`] follows `/IRT` from a group subordinate to its primary.
384pub const MAX_GROUP_DEPTH: usize = 64;
385
386/// For each annotation of `all`, the position in `all` of the primary of the group it is a
387/// subordinate of ([`Annotation::is_group_member`], ISO 32000-2 §12.5.6.2), or `None` when it is
388/// not one. The group's attributes, its comment and its replies are the primary's. A subordinate
389/// of a subordinate leads to the first non-subordinate up the `/IRT` chain. Expects links
390/// [`link_replies`] settled: a subordinate whose chain loops or names nothing listed has lost its
391/// `parent` there and is a comment of its own, as such a reply is. One still on a subordinate
392/// after [`MAX_GROUP_DEPTH`] steps has no primary (`None`) and is listed under no comment.
393pub fn group_primaries(all: &[Annotation]) -> Vec<Option<usize>> {
394    let mut at: HashMap<(usize, usize), usize> = HashMap::new();
395    for (i, a) in all.iter().enumerate() {
396        at.entry((a.page, a.index)).or_insert(i);
397    }
398    let primary = |start: usize| -> Option<usize> {
399        let mut i = start;
400        for _ in 0..MAX_GROUP_DEPTH {
401            let a = all.get(i)?;
402            if !a.is_group_member() {
403                return Some(i);
404            }
405            i = *at.get(&a.parent?)?;
406        }
407        None
408    };
409    all.iter().enumerate().map(|(i, a)| if a.is_group_member() { primary(i) } else { None }).collect()
410}
411
412/// Give a group's StrikeOut primary with no text of its own (`/Contents` missing or empty) the
413/// text of its first Caret subordinate that has some. A reader ignores a subordinate's
414/// `/Contents` (ISO 32000-2 §12.5.6.2), but Replace Text groups written by earlier PdfCraft builds
415/// made the strikeout the primary, with no text, and kept the replacement only on the caret:
416/// without this they would list as a comment with no text. A primary's own text always wins, no
417/// other subordinate lends its text, and no other kind of primary borrows one. Only the listing
418/// changes, not the file.
419pub fn fill_group_text(all: &mut [Annotation]) {
420    if !all.iter().any(Annotation::is_group_member) {
421        return;
422    }
423    let primaries = group_primaries(all);
424    let lacks_text = |p: usize| all.get(p).is_some_and(|a| a.subtype == "StrikeOut" && a.contents.as_deref().is_none_or(str::is_empty));
425    // Keyed by the primary's position: the first caret with text per primary wins (an empty one
426    // lends nothing), and each primary is written once, so the order of the map does not matter.
427    let mut texts: HashMap<usize, String> = HashMap::new();
428    for (a, p) in all.iter().zip(&primaries) {
429        if let (Some(p), Some(text), "Caret") = (*p, a.contents.as_ref(), a.subtype.as_str())
430            && !text.is_empty()
431            && lacks_text(p)
432        {
433            texts.entry(p).or_insert_with(|| text.clone());
434        }
435    }
436    for (p, text) in texts {
437        if let Some(primary) = all.get_mut(p) {
438            primary.contents = Some(text);
439        }
440    }
441}

473    pub fn is_group_member(&self) -> bool {
474        self.grouped && self.is_reply()
475    }

482    /// Where the comment is drawn on a page turned by `rotation` (`PageInfo::rotation`), in user
483    /// space: `rect`, or for a NoRotate comment `rect` turned upright about its upper-left corner
484    /// (see [`Self::no_rotate`]). Hit-testing, selection and anything placed beside the comment
485    /// use this, so they match the drawing.
486    pub fn shown_rect(&self, rotation: u16) -> [f32; 4] {
487        self.shown(self.rect.map(f64::from), rotation).map(|v| v as f32)
488    }
489
490    /// [`Self::shown_rect`] for this comment with its `/Rect` at `rect` (one being moved or
491    /// resized).
492    pub fn shown(&self, rect: [f64; 4], rotation: u16) -> [f64; 4] {
493        if self.no_rotate { upright_rect(rect, rotation) } else { rect }
494    }
495
496    /// The `/Rect` that shows this comment in the user-space box `shown`: the inverse of
497    /// [`Self::shown`], for a box resized as drawn.
498    pub fn rect_for_shown(&self, shown: [f64; 4], rotation: u16) -> [f64; 4] {
499        if self.no_rotate { rect_from_upright(shown, rotation) } else { shown }
500    }
501}

503/// The user-space box a NoRotate annotation with `/Rect` `rect` is drawn in on a page turned
504/// clockwise by `rotation` (ISO 32000-2 §12.5.3): `rect` turned as far counterclockwise about its
505/// upper-left corner `(x0, y1)`, which stays where it is on the page. As displayed, the box's
506/// top-left is that corner, the rectangle's width runs across and its height down, as on an
507/// unturned page. `rect` itself for 0 or anything that isn't a quarter turn. The vendored
508/// renderer draws the appearance there (hayro-interpret patch 37).
509fn upright_rect(rect: [f64; 4], rotation: u16) -> [f64; 4] {
510    let [x0, y0, x1, y1] = [rect[0].min(rect[2]), rect[1].min(rect[3]), rect[0].max(rect[2]), rect[1].max(rect[3])];
511    let (w, h) = (x1 - x0, y1 - y0);
512    match rotation {
513        90 => [x0, y1, x0 + h, y1 + w],
514        180 => [x0 - w, y1, x0, y1 + h],
515        270 => [x0 - h, y1 - w, x0, y1],
516        _ => [x0, y0, x1, y1],
517    }
518}
519
520/// The `/Rect` whose [`upright_rect`] is `shown`: `shown` turned back about the corner that
521/// stays put.
522fn rect_from_upright(shown: [f64; 4], rotation: u16) -> [f64; 4] {
523    let [s0, s1, s2, s3] = [shown[0].min(shown[2]), shown[1].min(shown[3]), shown[0].max(shown[2]), shown[1].max(shown[3])];
524    let (sw, sh) = (s2 - s0, s3 - s1);
525    // The upper-left corner of `/Rect` and the rectangle's width and height.
526    let ((x0, y1), w, h) = match rotation {
527        90 => ((s0, s1), sh, sw),
528        180 => ((s2, s1), sw, sh),
529        270 => ((s2, s3), sh, sw),
530        _ => return [s0, s1, s2, s3],
531    };
532    [x0, y1 - h, x0 + w, y1]
533}

535/// Settle the reply links of a document's annotations: a reply keeps its `parent` only when
536/// following the links reaches a comment of its own. An `/IRT` to an unlisted object, a pop-up,
537/// the annotation itself or a loop leaves it a comment of its own, so it stays in view instead of
538/// hanging off nothing. A chain of any length settles: each reviewer's state changes reply to
539/// their previous one (ISO 32000-2 §12.5.6.3), so a long review history is a long chain. Each
540/// annotation is walked once, with a seen-set against loops. Both comment readers (inspect, and
541/// the engine's list after an edit) run this.
542///
543/// Only comments can be parents: both readers leave links, form widgets, pop-ups and other
544/// non-markup annotations out of `annotations` (`pdfcraft_annot::is_comment_subtype`), so an
545/// `/IRT` to one of those finds no parent here. Pop-ups are skipped again in case a caller's list
546/// has them.
547pub fn link_replies(annotations: &mut [Annotation]) {
548    let at: HashMap<(usize, usize), usize> =
549        annotations.iter().enumerate().filter(|(_, a)| a.subtype != "Popup").map(|(i, a)| ((a.page, a.index), i)).collect();
550    let links: Vec<Option<usize>> =
551        annotations.iter().enumerate().map(|(i, a)| a.parent.and_then(|p| at.get(&p).copied()).filter(|&j| j != i)).collect();
552    // Whether each annotation's chain reaches a comment of its own, filled in along each walk.
553    let mut settled: Vec<Option<bool>> = vec![None; annotations.len()];
554    for start in 0..annotations.len() {
555        let (mut path, mut seen, mut at) = (Vec::new(), HashSet::new(), start);
556        let ends_well = loop {
557            match settled.get(at).copied().flatten() {
558                Some(known) => break known,
559                None if !seen.insert(at) => break false,
560                None => {}
561            }
562            path.push(at);
563            match links.get(at).copied().flatten() {
564                Some(up) => at = up,
565                None => break true,
566            }
567        };
568        for i in path {
569            if let Some(s) = settled.get_mut(i) {
570                *s = Some(ends_well);
571            }
572        }
573    }
574    // A reply keeps its parent when it has a listed one and its chain reaches a comment; one whose
575    // `/IRT` leads nowhere listed (itself, an unlisted object) is a comment of its own.
576    for ((a, ok), link) in annotations.iter_mut().zip(settled).zip(links) {
577        if link.is_none() || ok != Some(true) {
578            a.parent = None;
579        }
580    }
581    // A group is one page's (Acrobat reads it so): a subordinate whose primary is on another page
582    // is a comment of its own, with its own author and text, rather than part of that group or a
583    // reply to it. ISO 32000-2 §12.5.6.2 doesn't limit a group to a page, but Acrobat (desktop and
584    // iPad) and PDF-XChange all delete the primary without such a subordinate, and Acrobat desktop
585    // lists it on its own. Keep in step with
586    // `pdfcraft_annot::summaries`.
587    for a in annotations.iter_mut() {
588        if a.grouped && a.parent.is_some_and(|(page, _)| page != a.page) {
589            a.parent = None;
590            a.in_reply_to = None;
591        }
592    }
593}

595/// The comment each reply's thread starts from, by the reply's page and `/Annots` position: a
596/// reply to a reply files under that comment too. Expects links [`link_replies`] settled; a loop
597/// in an unsettled list ends where it comes back round.
598pub fn thread_tops(annotations: &[Annotation]) -> HashMap<(usize, usize), (usize, usize)> {
599    let parents: HashMap<(usize, usize), (usize, usize)> = annotations.iter().filter_map(|a| Some(((a.page, a.index), a.parent?))).collect();
600    let mut tops: HashMap<(usize, usize), (usize, usize)> = HashMap::new();
601    for &reply in parents.keys() {
602        let (mut path, mut seen, mut at) = (Vec::new(), HashSet::new(), reply);
603        let top = loop {
604            if let Some(&known) = tops.get(&at) {
605                break known;
606            }
607            let Some(&up) = parents.get(&at) else { break at };
608            if !seen.insert(at) {
609                break up;
610            }
611            path.push(at);
612            at = up;
613        };
614        for r in path {
615            tops.insert(r, top);
616        }
617    }
618    tops
619}

621/// The status and checkmark replies of a document's annotations, by the annotation whose state
622/// each one sets, found by object reference (the settled `parent` links of [`link_replies`]).
623///
624/// ISO 32000-2 §12.5.6.3: a state change is a text annotation whose `/IRT` "shall refer to the
625/// original annotation" and whose `/State` and `/StateModel` "update the state of the original
626/// annotation for the specified user" (`/T`); "Additional state changes shall be made by adding
627/// text annotations in reply to the previous reply for a given user". So a state reply sets the
628/// state of the first annotation up its `/IRT` chain that is not itself a state reply: the
629/// comment, or the reply the state was set on (a reply is an annotation with a state of its own,
630/// and is never folded into its thread's comment). There is one status per reviewer, and a
631/// checkmark only for the reviewer who set it.
632pub struct ReviewStates<'a> {
633    by_target: HashMap<(usize, usize), Vec<&'a Annotation>>,
634}
635
636impl<'a> ReviewStates<'a> {
637    pub fn new(all: &'a [Annotation]) -> Self {
638        let at: HashMap<(usize, usize), &Annotation> = all.iter().filter(|a| a.subtype != "Popup").map(|a| ((a.page, a.index), a)).collect();
639        // The annotation each state reply's chain reaches past other state replies, `None` for a
640        // loop or a reference to nothing listed; filled in along each walk, so a long history
641        // costs one walk, not one per reply.
642        let mut reaches: HashMap<(usize, usize), Option<(usize, usize)>> = HashMap::new();
643        let mut by_target: HashMap<(usize, usize), Vec<&Annotation>> = HashMap::new();
644        // A state set on a group subordinate is the group's, its primary's (ISO 32000-2 §12.5.6.2).
645        let primary_of: HashMap<(usize, usize), (usize, usize)> =
646            all.iter().zip(group_primaries(all)).filter_map(|(a, p)| all.get(p?).map(|p| ((a.page, a.index), (p.page, p.index)))).collect();
647        for r in all.iter().filter(|a| a.state.is_some()) {
648            let (mut path, mut seen, mut up) = (vec![(r.page, r.index)], HashSet::from([(r.page, r.index)]), r.parent);
649            let target = loop {
650                let Some(key) = up else { break None };
651                if let Some(&known) = reaches.get(&key) {
652                    break known;
653                }
654                match at.get(&key) {
655                    None => break None,
656                    Some(p) if p.state.is_none() => break Some(key),
657                    Some(_) if !seen.insert(key) => break None,
658                    Some(p) => {
659                        path.push(key);
660                        up = p.parent;
661                    }
662                }
663            };
664            for key in path {
665                reaches.insert(key, target);
666            }
667            if let Some(t) = target.map(|t| primary_of.get(&t).copied().unwrap_or(t)) {
668                by_target.entry(t).or_default().push(r);
669            }
670        }
671        Self { by_target }
672    }
673
674    /// Every status and checkmark reply about `target`, in document order.
675    pub fn history(&self, target: &Annotation) -> &[&'a Annotation] {
676        self.by_target.get(&(target.page, target.index)).map(Vec::as_slice).unwrap_or(&[])
677    }
678
679    /// Each reviewer's review status on `target`: their latest, in the order they first set one.
680    /// Empty when nobody set one; a status set back to "None" stays listed, since None is a state
681    /// (ISO 32000-2 Table 174: "The user has indicated nothing about the change").
682    pub fn statuses(&self, target: &Annotation) -> Vec<(&'a str, &'a str)> {
683        self.by_reviewer(target, false).into_iter().filter_map(|(who, mine)| Some((who, latest(&mine)?.state.as_deref()?))).collect()
684    }
685
686    /// The latest review status set on `target` by anyone, for callers that show one.
687    pub fn latest_status(&self, target: &Annotation) -> Option<&'a str> {
688        self.history(target).iter().rev().filter(|r| !r.is_mark()).find_map(|r| r.state.as_deref())
689    }
690
691    /// Whether `author`'s latest checkmark on `target` is "Marked". Checkmarks are each
692    /// reviewer's own (ISO 32000-2 Table 174: "The annotation has been marked by the user").
693    pub fn marked_by(&self, target: &Annotation, author: &str) -> bool {
694        let mine: Vec<&Annotation> = self.history(target).iter().copied().filter(|r| r.is_mark() && reviewer(r) == author.trim()).collect();
695        latest(&mine).is_some_and(|r| r.state.as_deref() == Some("Marked"))
696    }
697
698    /// The reviewers whose latest checkmark on `target` is "Marked".
699    pub fn marked(&self, target: &Annotation) -> Vec<&'a str> {
700        self.by_reviewer(target, true)
701            .into_iter()
702            .filter(|(_, mine)| latest(mine).is_some_and(|r| r.state.as_deref() == Some("Marked")))
703            .map(|(who, _)| who)
704            .collect()
705    }
706
707    /// The checkmark (`marks`) or status replies about `target`, by reviewer, in the order each
708    /// reviewer first set one.
709    fn by_reviewer(&self, target: &Annotation, marks: bool) -> Vec<(&'a str, Vec<&'a Annotation>)> {
710        let mut out: Vec<(&str, Vec<&Annotation>)> = Vec::new();
711        for &r in self.history(target).iter().filter(|r| r.is_mark() == marks) {
712            match out.iter_mut().find(|(who, _)| *who == reviewer(r)) {
713                Some((_, mine)) => mine.push(r),
714                None => out.push((reviewer(r), vec![r])),
715            }
716        }
717        out
718    }
719}

721/// Of one reviewer's replies in one state model, the latest: the end of their chain, a reply none
722/// of the others replies to (by reference), whatever order the file lists them in. Among several
723/// such (changes that don't chain), the last listed.
724fn latest<'b>(mine: &[&'b Annotation]) -> Option<&'b Annotation> {
725    let answered: HashSet<(usize, usize)> = mine.iter().filter_map(|r| r.parent).collect();
726    mine.iter().rev().find(|r| !answered.contains(&(r.page, r.index))).copied()
727}

1042    /// A long reply chain, as a reviewer's long status history makes (§12.5.6.3: each change
1043    /// replies to the previous), settles, and its replies file under the comment; a loop, an `/IRT`
1044    /// to itself and one to nothing listed leave an annotation a comment of its own.
1045    #[test]
1046    fn reply_links_settle_at_any_depth_but_not_in_a_loop() {
1047        let keys: Vec<String> = (0..200).map(|i| format!("r{i}")).collect();
1048        let mut specs: Vec<Spec<'_>> = vec![("c", None, "Alice", None)];
1049        for (i, k) in keys.iter().enumerate() {
1050            specs.push((k, Some(if i == 0 { "c" } else { keys[i - 1].as_str() }), "Bob", None));
1051        }
1052        specs.extend([
1053            ("x", Some("y"), "Eve", None),
1054            ("y", Some("x"), "Eve", None),
1055            ("z", Some("x"), "Eve", None),
1056            ("me", Some("me"), "Eve", None),
1057            ("lost", Some("unlisted"), "Eve", None),
1058        ]);
1059        // In file order and newest first: the first reply read then walks the whole chain at once.
1060        for newest_first in [false, true] {
1061            if newest_first {
1062                specs.reverse();
1063            }
1064            let all = listed(&specs);
1065            let replies: Vec<&Annotation> = all.iter().filter(|a| a.name.as_deref().is_some_and(|n| n.starts_with('r'))).collect();
1066            assert!(replies.len() == 200 && replies.iter().all(|a| a.is_reply()), "{newest_first}");
1067            let nowhere = all.iter().filter(|a| matches!(a.name.as_deref(), Some("x" | "y" | "z" | "me" | "lost")));
1068            assert!(nowhere.clone().count() == 5 && nowhere.clone().all(|a| !a.is_reply()), "a loop, a chain into one, itself or nothing listed");
1069            let c = by_name(&all, "c");
1070            let tops = thread_tops(&all);
1071            assert_eq!(tops.len(), 200, "{newest_first}");
1072            assert!(tops.values().all(|&t| t == (c.page, c.index)), "{newest_first}");
1073        }
1074    }

1102    /// The inspector reads a `/StateModel`-only reply as the model's default state, so it counts
1103    /// as Bob's status (None) on the note.
1104    #[test]
1105    fn the_inspector_reads_a_state_model_alone_as_its_default() {
1106        let objs: Vec<&[u8]> = vec![
1107            b"<< /Type /Catalog /Pages 2 0 R >>",
1108            b"<< /Type /Pages /Kids [3 0 R] /Count 1 >>",
1109            b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Annots [4 0 R 5 0 R] >>",
1110            b"<< /Type /Annot /Subtype /Text /Rect [10 10 30 30] /NM (note) /T (Ann) >>",
1111            b"<< /Type /Annot /Subtype /Text /Rect [10 10 30 30] /NM (bare) /T (Bob) /IRT 4 0 R /F 30 /StateModel (Review) >>",
1112        ];
1113        let mut out = b"%PDF-1.7\n".to_vec();
1114        let mut offsets = Vec::new();
1115        for (i, o) in objs.iter().enumerate() {
1116            offsets.push(out.len());
1117            out.extend_from_slice(format!("{} 0 obj\n", i + 1).as_bytes());
1118            out.extend_from_slice(o);
1119            out.extend_from_slice(b"\nendobj\n");
1120        }
1121        let xref = out.len();
1122        out.extend_from_slice(format!("xref\n0 {}\n0000000000 65535 f \n", objs.len() + 1).as_bytes());
1123        for o in offsets {
1124            out.extend_from_slice(format!("{o:010} 00000 n \n").as_bytes());
1125        }
1126        out.extend_from_slice(format!("trailer\n<< /Size {} /Root 1 0 R >>\nstartxref\n{xref}\n%%EOF\n", objs.len() + 1).as_bytes());
1127        let info = crate::render::inspect(std::sync::Arc::new(out), None).expect("inspects");
1128        let note = by_name(&info.annotations, "note");
1129        assert_eq!(ReviewStates::new(&info.annotations).statuses(note), [("Bob", "None")]);
1130    }
