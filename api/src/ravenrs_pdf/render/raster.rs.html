1//! Background page rasterization.
2//!
3//! A `RenderPool` parses lazily once and shares that immutable parser between workers. Render
4//! caches stay worker-local. Requests carry a caller tag that is echoed back.
5//!
6//! **Robustness:** every page render runs under `catch_unwind`. A panic inside the renderer on a
7//! malformed page yields an error and permanently retires the pool's shared parser generation.
8//! Workers discard its caches/results and fall back to private parsers. There is no shared
9//! rebuild loop, and a timed-out shared initialization is never started again in that pool.
10

43/// Monotonic-ish timer that is safe on wasm32 (where `std::time::Instant` panics).
44#[derive(Clone, Copy)]
45struct Stopwatch(#[cfg(not(target_arch = "wasm32"))] std::time::Instant);

47impl Stopwatch {
48    fn start() -> Self {
49        #[cfg(not(target_arch = "wasm32"))]
50        return Self(std::time::Instant::now());
51        #[cfg(target_arch = "wasm32")]
52        return Self();
53    }
54    fn millis(self) -> u32 {
55        #[cfg(not(target_arch = "wasm32"))]

58        return 0;
59    }

62/// Per-document rendering configuration.
63#[derive(Clone, Debug, Default)]
64pub struct RenderConfig {
65    /// User or owner password for encrypted documents.
66    pub password: Option<Arc<str>>,
67    /// Layer (optional content group) visibility overrides: (object number, generation, visible).
68    pub layers: Arc<Vec<(i32, i32, bool)>>,
69    /// View ▸ Hide all comments: markup annotations aren't drawn (fields and links still are).
70    pub hide_comments: bool,
71    /// Annotations shown or hidden for viewing in place of their Hidden and NoView flags (object
72    /// number, generation, shown): fields a document's page actions and timers showed or hid
73    /// (an animation's frames). Display only; the document never changes.
74    pub annotations: Arc<Vec<(i32, i32, bool)>>,
75    /// Annotations drawn with a form XObject as their normal appearance (object number and
76    /// generation of the annotation, then of the form): push buttons a document's page actions
77    /// and timers gave another icon (`buttonSetIcon`, an animation's frames). Display only; the
78    /// document never changes.
79    pub appearances: Arc<Vec<(i32, i32, i32, i32)>>,
80    /// Refuse a whole-page raster that would be reduced by the renderer's hard size caps.
81    /// Tiled requests remain available for pages that exceed those caps. Both modes refuse
82    /// a raster whose rounded integer dimensions exceed the final pixel-allocation budget.
83    pub reject_oversize: bool,
84    /// View ▸ Reading colours: solid text and line-art paints are remapped for readability and
85    /// the page clears to the setting's background. Display only; the document never changes.
86    pub reading_colors: ReadingColors,
87    /// The page clears to transparent instead of white (or the reading background): for a page
88    /// holding only what is laid over another raster, such as a comment while it is dragged.
89    pub transparent: bool,
90    /// Draw a CID font of an Adobe CJK collection that the PDF does not embed with one face already
91    /// installed on this machine (see `system_font`). Off by default and on only for on-screen
92    /// viewer and preview rasters: exports, printing and OCR render without it, so nothing drawn with an
93    /// installed font is saved or printed (AGENTS.md §1.2, §1.4).
94    pub system_fonts: bool,
95    /// The render is output for printing (print as image): the configuration's auto state for
96    /// the View event (ISO 32000-2 §8.11.4.4) is not applied, since printing is not viewing.
97    pub for_print: bool,
98    /// Steps of JBIG2 decoding a render may take across the page's images, instead of the usual
99    /// `crate::third_party::hayro_syntax::MAX_PAGE_JBIG2_WORK`. Those past it aren't drawn, like an image that can't
100    /// be decoded.
101    pub jbig2_work: Option<u64>,
102    /// Layer visibility overrides for print-preview rasters ([`PRINT_TAG`]), when the document
103    /// prints its layers differently from how they are shown (the Print event's auto state, ISO
104    /// 32000-2 §8.11.4.4): the preview shows the page as it prints. `None`: previews use `layers`.
105    pub print_layers: Option<Arc<Vec<(i32, i32, bool)>>>,
106}

108impl RenderConfig {
109    /// This configuration for a raster that is only shown on screen (the viewer, Combine
110    /// thumbnails, figure previews): installed CJK faces may draw fonts the PDF doesn't embed.
111    /// Never use it for anything that can be saved, printed, recognised or returned to an agent
112    /// (AGENTS.md §1.2, §1.4).
113    #[must_use]
114    pub fn on_screen(self) -> Self {
115        Self { system_fonts: true, ..self }
116    }
117
118    /// Build interpreter settings with layer and comment preferences and a font resolver that
119    /// tries bundled Japanese faces, then opt-in installed CJK faces, then standard substitutes.
120    /// Passwords belong to parser construction; request-specific preview overrides are applied
121    /// by `render_page`, which also preserves the caller's warning and cancellation hooks.
122    fn settings(&self) -> InterpreterSettings {
123        let defaults = InterpreterSettings::default();
124        let standard = defaults.font_resolver.clone();
125        let system_fonts = self.system_fonts;
126        InterpreterSettings {
127            ocg_overrides: self.layers.clone(),
128            view_auto_state: !self.for_print,
129            hide_comments: self.hide_comments,
130            annotation_overrides: self.annotations.clone(),
131            appearance_overrides: self.appearances.clone(),
132            font_resolver: Arc::new(move |query| {
133                japanese_fallback(query).or_else(|| system_fonts.then(|| system_fallback(query)).flatten()).or_else(|| standard(query))
134            }),
135            jbig2_work: self.jbig2_work.unwrap_or(defaults.jbig2_work),
136            ..defaults
137        }
138    }
139}

141/// A Japanese face from craft-fonts for a CID font of the Adobe-Japan1 collection that the PDF
142/// doesn't embed (`HeiseiMin-W3`, `KozGoPro-Medium`, …). hayro's own substitutes for fonts that
143/// aren't embedded are the Latin standard 14, so such text drew nothing. `None` for every other
144/// font, and when PdfCraft was built without craft-fonts. Only Japanese: no other CJK face is
145/// embedded whole.
146fn japanese_fallback(query: &FontQuery) -> Option<(FontData, u32)> {
147    let FontQuery::Fallback(f) = query else { return None };
148    if f.character_collection.as_ref()?.family != CidFamily::AdobeJapan1 {
149        return None;
150    }
151    let face = match japanese_face(f.post_script_name.as_deref().unwrap_or_default(), f.is_serif, f.is_bold || f.font_weight >= 600) {
152        // BIZ UDMincho before the document face (Shippori Mincho): it covers half-width katakana
153        // (U+FF61–U+FF9F), which Shippori Mincho lacks, and a character the substitute lacks is
154        // drawn as .notdef.
155        JapaneseFace::Mincho => crate::fonts::ui_japanese_fonts()
156            .into_iter()
157            .find(|c| c.family == "BIZ UDMincho" && c.style == "Regular")
158            .or_else(crate::fonts::document_japanese_font),
159        JapaneseFace::Gothic { bold } => {
160            let faces = crate::fonts::ui_japanese_fonts();
161            let style = if bold { "Bold" } else { "Regular" };
162            faces.iter().find(|c| c.family == "BIZ UDPGothic" && c.style == style).or(faces.first()).copied()
163        }
164    }?;
165    Some((Arc::new(face.bytes), 0))
166}

168/// An installed face for a CID font of an Adobe CJK collection that the PDF doesn't embed and
169/// craft-fonts has no face for (Traditional and Simplified Chinese, Korean, and Japanese in a
170/// build without craft-fonts). `None` for every other font.
171fn system_fallback(query: &FontQuery) -> Option<(FontData, u32)> {
172    let FontQuery::Fallback(f) = query else { return None };
173    crate::render::system_font::fallback(f)
174}

176#[derive(Debug, PartialEq)]
177enum JapaneseFace {
178    Mincho,
179    Gothic { bold: bool },
180}
181
182/// Which kind of Japanese face stands in for the font named `name`: Mincho names (`HeiseiMin`,
183/// `KozMin`, `Ryumin`, `MS-Mincho`) a serif Mincho; Gothic names (`…Gothic…`, `HeiseiKakuGo`,
184/// `KozGo`, `…Maru…`) a sans Gothic; any other name by the font descriptor's serif flag.
185fn japanese_face(name: &str, serif: bool, bold: bool) -> JapaneseFace {
186    let name = name.to_ascii_lowercase();
187    if name.contains("min") {
188        JapaneseFace::Mincho
189    } else if ["goth", "kakugo", "kozgo", "kaku", "maru"].iter().any(|k| name.contains(k)) || !serif {
190        JapaneseFace::Gothic { bold }
191    } else {
192        JapaneseFace::Mincho
193    }
194}

236/// The tag bit of a print-preview raster. Such requests always render the document's true
237/// colours, ignoring `RenderConfig::reading_colors`: a preview's job is fidelity to the
238/// printout, and the print itself — imposed from the document — is never recoloured.
239/// Installed-font fallback is also disabled, matching the output's font policy.
240pub const PRINT_TAG: u64 = 1 << 60;

242/// The reading colours a request renders with: print previews ([`PRINT_TAG`]) ignore the
243/// document's setting and render with reading colours off.
244fn request_reading(config: &RenderConfig, req: &RenderRequest) -> ReadingColors {
245    if req.tag & PRINT_TAG != 0 { ReadingColors::default() } else { config.reading_colors }

329impl AtomicRenderStats {
330    fn record_request(&self, req: RenderRequest) {
331        self.render_requests.fetch_add(1, Ordering::Relaxed);
332        if req.tile.is_some() {
333            self.tile_requests.fetch_add(1, Ordering::Relaxed);
334        }
335        match req.kind {
336            RequestKind::Pixels => {
337                self.page_interpretations.fetch_add(1, Ordering::Relaxed);
338                self.rasterizations.fetch_add(1, Ordering::Relaxed);
339            }
340            RequestKind::Text => {
341                self.text_interpretations.fetch_add(1, Ordering::Relaxed);
342            }
343        }
344    }
345
346    fn snapshot(&self) -> RenderStats {
347        RenderStats {
348            parser_builds: self.parser_builds.load(Ordering::Relaxed),
349            page_interpretations: self.page_interpretations.load(Ordering::Relaxed),
350            render_requests: self.render_requests.load(Ordering::Relaxed),
351            tile_requests: self.tile_requests.load(Ordering::Relaxed),
352            text_interpretations: self.text_interpretations.load(Ordering::Relaxed),
353            rasterizations: self.rasterizations.load(Ordering::Relaxed),
354        }
355    }
356}

358/// Clamp a requested scale so the output respects `MAX_SIDE` and `MAX_PIXELS`.
359pub fn effective_scale(width_pt: f32, height_pt: f32, scale: f32) -> f32 {
360    let (w, h) = (width_pt.max(1.0), height_pt.max(1.0));
361    let by_side = MAX_SIDE / w.max(h);
362    let by_area = (MAX_PIXELS / (w * h)).sqrt();
363    // The caps always win: a floor here once let a 934-million-point-wide page (a fuzzed file)
364    // render 9 million pixels wide. Only guard against a zero or non-finite scale.
365    let capped = scale.min(by_side).min(by_area);
366    if capped.is_finite() && capped > 0.0 { capped } else { by_side.min(by_area).max(f32::MIN_POSITIVE) }
367}

369/// Device pixels that cover `pt` points at `scale`: rounded up, as poppler's `pdftoppm` does, so
370/// the last partial row or column of a page is drawn (anti-aliased against the background) rather
371/// than cut off. Float noise up to 1/100 px is not rounded up, so 612 pt at 150 dpi stays 1275 px.
372/// At least 1; saturates instead of overflowing.
373pub fn device_pixels(pt: f32, scale: f32) -> u32 {
374    let px = pt * scale - 0.01;
375    if px.is_finite() { px.ceil().max(1.0) as u32 } else { 1 }
376}

378/// Render one page with a caller-owned parser and cache. Panics inside the renderer are caught
379/// and reported as `Err((message, panicked))`.
380type Output = (u32, u32, Pixels, Option<Arc<crate::render::text::PageText>>);
381
382fn render_page<'a>(
383    pdf: &'a Pdf,
384    cache: &RenderCache<'a>,
385    settings: &InterpreterSettings,
386    config: &RenderConfig,
387    req: RenderRequest,
388) -> Result<Output, (String, bool)> {
389    // Print previews share the viewer pool, but must use the output font policy and the layers
390    // as they print (`print_layers`). Keep the caller's cancellation/warning hooks and annotation
391    // settings, except what a document's scripts show while it is read (`annotations`,
392    // `appearances`), which never prints. A fresh cache is essential: hayro keys cached fonts by PDF dictionary, not by the
393    // font resolver, so the viewer cache could supply an installed face even after changing the
394    // resolver (and a cached missing font from print preview could blank the viewer on switching
395    // back); nothing drawn with the viewer's layers is reused either.
396    let print_settings;
397    let print_cache;
398    let print = req.tag & PRINT_TAG != 0;
399    let print_layers = config.print_layers.as_ref().filter(|_| print);
400    let live = !settings.annotation_overrides.is_empty() || !settings.appearance_overrides.is_empty();
401    let (settings, cache) = if print && (config.system_fonts || print_layers.is_some() || live) {
402        print_settings = InterpreterSettings {
403            font_resolver: RenderConfig { system_fonts: false, ..config.clone() }.settings().font_resolver,
404            ocg_overrides: print_layers.cloned().unwrap_or_else(|| settings.ocg_overrides.clone()),
405            annotation_overrides: Arc::default(),
406            appearance_overrides: Arc::default(),
407            ..settings.clone()
408        };
409        print_cache = RenderCache::new();
410        (&print_settings, &print_cache)
411    } else {
412        (settings, cache)
413    };
414    let reading = request_reading(config, &req);
415    if req.kind == RequestKind::Text {
416        return match catch_unwind(AssertUnwindSafe(|| crate::render::text::extract_page(pdf, req.page, settings))) {
417            Ok(Some(t)) => Ok((0, 0, Pixels::default(), Some(Arc::new(t)))),
418            Ok(None) => Err((format!("page {} does not exist", req.page + 1), false)),
419            Err(panic) => Err((format!("text extraction crashed on page {}: {}", req.page + 1, panic_message(&panic)), true)),
420        };
421    }
422    let result = catch_unwind(AssertUnwindSafe(|| {
423        let pages = pdf.pages();
424        let page = pages.get(req.page).ok_or_else(|| format!("page {} does not exist", req.page + 1))?;
425        let (w, h) = page.render_dimensions();
426        if !(w.is_finite() && h.is_finite()) || w < 0.5 || h < 0.5 {
427            return Err(format!("page {} has an empty or invalid page box ({w}×{h} pt)", req.page + 1));
428        }
429        // Reading colours: the page clears to the setting's background, and solid text and
430        // line-art paints are remapped (images and patterns never are). Off renders as before.
431        let (bg_color, color_replacement) = match reading.params {
432            Some(p) => {
433                (AlphaColor::from_rgb8(p.background[0], p.background[1], p.background[2]), Some(crate::third_party::hayro::ColorReplacement::from(p)))
434            }
435            None => (WHITE, None),
436        };
437        let bg_color = if config.transparent { TRANSPARENT } else { bg_color };
438        let rs = match req.tile {
439            // Tiles are bounded by construction, so the page itself may be arbitrarily large.
440            Some(t) => {
441                let scale = req.scale.clamp(0.01, 400.0);
442                let (tw, th) = (t.w.clamp(1, 4096), t.h.clamp(1, 4096));
443                RenderSettings {
444                    x_scale: scale,
445                    y_scale: scale,
446                    width: Some(tw as u16),
447                    height: Some(th as u16),
448                    x_offset: t.x as f32,
449                    y_offset: t.y as f32,
450                    bg_color,
451                    color_replacement,
452                    ..Default::default()
453                }
454            }
455            None => {
456                let scale = effective_scale(w, h, req.scale);
457                if config.reject_oversize && (!(req.scale.is_finite() && req.scale > 0.0) || req.scale > scale) {
458                    return Err(format!(
459                        "requested page {} raster scale {req_scale} exceeds renderer limits (maximum scale {scale})",
460                        req.page + 1,
461                        req_scale = req.scale
462                    ));
463                }
464                // hayro floors the size when none is given, losing the partial edge pixels. The
465                // scale keeps each side within MAX_SIDE (give or take float noise), so it fits u16.
466                let side = |pt: f32| device_pixels(pt, scale).min(MAX_SIDE as u32) as u16;
467                RenderSettings {
468                    x_scale: scale,
469                    y_scale: scale,
470                    width: Some(side(w)),
471                    height: Some(side(h)),
472                    bg_color,
473                    color_replacement,
474                    ..Default::default()
475                }
476            }
477        };
478        // Rendered straight into a buffer of whole pixels, which the GUI takes over as is.
479        let (w, h) = render_size(page, &rs);
480        if u64::from(w).checked_mul(u64::from(h)).is_none_or(|pixels| pixels > MAX_PIXELS as u64) {
481            return Err(format!("page {} raster size {w}x{h} exceeds renderer pixel limit", req.page + 1));
482        }
483        let mut pixels = Pixels::zeroed(w.into(), h.into()).ok_or_else(|| format!("page {} is too large to render", req.page + 1))?;
484        // Previews render at low detail (tiny images are skipped); finals always draw all.
485        let mut preview_settings = settings.clone();
486        preview_settings.preview = req.preview;
487        // A print preview shows what prints, and printing is not the View event.
488        preview_settings.view_auto_state &= req.tag & PRINT_TAG == 0;
489        render_into(page, cache, &preview_settings, &rs, pixels.bytes_mut());
490        Ok((w.into(), h.into(), pixels, None))
491    }));
492    match result {
493        Ok(Ok(v)) => Ok(v),
494        Ok(Err(e)) => Err((e, false)),
495        Err(panic) => Err((format!("renderer crashed on page {}: {}", req.page + 1, panic_message(&panic)), true)),
496    }
497}

499fn finish(req: RenderRequest, start: Stopwatch, r: Result<Output, (String, bool)>, warnings: Vec<RenderWarning>) -> RenderedPage {
500    let millis = start.millis();
501    match r {
502        Ok((width, height, rgba, text)) => RenderedPage { request: req, width, height, rgba, error: None, text, warnings, millis },
503        Err((e, _)) => RenderedPage { request: req, width: 0, height: 0, rgba: Pixels::default(), error: Some(e), text: None, warnings, millis },
504    }
505}
506
507fn warning_settings(settings: &InterpreterSettings, warnings: Arc<Mutex<Vec<RenderWarning>>>) -> InterpreterSettings {
508    let target = warnings;
509    InterpreterSettings {
510        warning_sink: Arc::new(move |warning| {
511            // Nested forms, Type 3 glyphs and patterns are interpreted again and each reports the
512            // exhausted budget, or the colours that fell back, and several images can share one
513            // malformed colour space: each warning is recorded once per page.
514            let warning = match warning {
515                InterpreterWarning::ContentTruncated => RenderWarning::ContentTruncated,
516                InterpreterWarning::ColorFallback => RenderWarning::ColourFallback,
517                InterpreterWarning::MalformedIndexedImage => RenderWarning::MalformedIndexedImage,
518                InterpreterWarning::Type3WidthsMissing => RenderWarning::Type3WidthsMissing,
519                InterpreterWarning::UnsupportedFont | InterpreterWarning::ImageDecodeFailure => return,
520            };
521            let mut warnings = lock(&target);
522            if !warnings.contains(&warning) {
523                warnings.push(warning);
524            }
525        }),
526        ..settings.clone()
527    }
528}
529
530fn take_warnings(warnings: &Mutex<Vec<RenderWarning>>) -> Vec<RenderWarning> {
531    std::mem::take(&mut *lock(warnings))
532}

534/// A single-threaded renderer over one parsed document (used by the CLI and tests).
535pub struct PageRenderer {
536    bytes: Arc<Vec<u8>>,
537    config: RenderConfig,
538    pdf: Option<Pdf>,
539    settings: InterpreterSettings,
540    stats: RenderStats,
541}
542
543impl PageRenderer {
544    pub fn new(bytes: Arc<Vec<u8>>, config: RenderConfig) -> Self {
545        let pdf = parse(&bytes, config.password.as_deref());
546        let settings = config.settings();
547        Self { bytes, config, pdf, settings, stats: RenderStats { parser_builds: 1, ..RenderStats::default() } }
548    }
549
550    pub fn page_count(&self) -> usize {
551        self.pdf.as_ref().map(|p| p.pages().len()).unwrap_or(0)
552    }
553
554    /// Snapshot the work counters collected by this renderer.
555    pub fn stats(&self) -> RenderStats {
556        self.stats
557    }
558
559    /// Extract one page for statistics, with semantic exclusions. Never rasterizes or panics.
560    pub fn filtered_text(&mut self, page: usize, filter: crate::render::text::TextFilter) -> Result<crate::render::PageText, String> {
561        let pdf = self.pdf.as_ref().ok_or("the document could not be parsed")?;
562        self.stats.text_interpretations = self.stats.text_interpretations.saturating_add(1);
563        catch_unwind(AssertUnwindSafe(|| crate::render::text::extract_page_filtered(pdf, page, &self.settings, filter)))
564            .map_err(|e| format!("text extraction failed: {}", panic_message(&e)))?
565    }
566
567    /// Render one page. Never panics.
568    pub fn render(&mut self, req: RenderRequest) -> RenderedPage {
569        let start = Stopwatch::start();
570        if self.pdf.is_some() {
571            self.stats.render_requests += 1;
572            if req.tile.is_some() {
573                self.stats.tile_requests += 1;
574            }
575            match req.kind {
576                RequestKind::Pixels => {
577                    self.stats.page_interpretations += 1;
578                    self.stats.rasterizations += 1;
579                }
580                RequestKind::Text => self.stats.text_interpretations += 1,
581            }
582        }
583        let Some(pdf) = self.pdf.as_ref() else {
584            return finish(req, start, Err(("the document could not be parsed".into(), false)), Vec::new());
585        };
586        let cache = RenderCache::new();
587        let warnings = Arc::new(Mutex::new(Vec::new()));
588        let settings = warning_settings(&self.settings, warnings.clone());
589        let r = render_page(pdf, &cache, &settings, &self.config, req);
590        let warnings = take_warnings(&warnings);
591        if matches!(r, Err((_, true))) {
592            self.pdf = parse(&self.bytes, self.config.password.as_deref());
593            self.stats.parser_builds += 1;
594        }
595        finish(req, start, r, warnings)
596    }
597}
598
599fn parse(bytes: &Arc<Vec<u8>>, password: Option<&str>) -> Option<Pdf> {
600    catch_unwind(AssertUnwindSafe(|| Pdf::new_with_password(bytes.clone(), password.unwrap_or("")).ok())).ok().flatten()
601}
602
603pub(crate) fn panic_message(p: &Box<dyn std::any::Any + Send>) -> String {
604    p.downcast_ref::<&str>().map(ToString::to_string).or_else(|| p.downcast_ref::<String>().cloned()).unwrap_or_else(|| "unknown panic".into())
605}
