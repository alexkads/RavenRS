1//! Document inspection: everything panels need that is not pixels.
2//!
3//! Page geometry comes from hayro (which resolves inheritance and rotation); the rest comes from
4//! the lazy COS reader through a read-only lopdf object adapter. The original lopdf loader
5//! remains the compatibility fallback. When neither can inspect a renderable file, panels
6//! are empty and `warnings` says why.

52#[derive(Clone, Debug, Default)]
53pub struct DocInfo {
54    pub title: Option<String>,
55    pub author: Option<String>,
56    pub subject: Option<String>,
57    pub keywords: Option<String>,
58    pub creator: Option<String>,
59    pub producer: Option<String>,
60    /// Custom document properties: `/Info` entries with a key the spec does not name and a text
61    /// value, as (name, text), in the dictionary's order (at most 256).
62    pub custom: Vec<(String, String)>,
63    pub pdf_version: String,
64    pub file_size: usize,
65    pub encrypted: bool,
66    pub tagged: bool,
67    pub has_javascript: bool,
68    /// The form is (also) an XFA form, which isn't read yet (`form.xfa-*` in the parity list).
69    pub xfa: Option<Xfa>,
70    pub pages: Vec<PageInfo>,
71    pub outline: Vec<OutlineItem>,
72    /// The outline has bookmarks that `outline` leaves out (see `OUTLINE_WINDOW`, `OUTLINE_DEPTH`).
73    pub outline_more: bool,
74    pub annotations: Vec<Annotation>,
75    pub fields: Vec<Field>,
76    pub links: Vec<Link>,
77    /// The catalog's `/URI /Base` (ISO 32000-2 Table 211), the base for relative URI actions, when
78    /// it is a string of at most [`crate::render::MAX_URI_BASE_BYTES`].
79    pub uri_base: Option<String>,
80    pub layers: Vec<Layer>,
81    /// Radio-button layer groups (`/OCProperties /D /RBGroups`): turning one layer of a group on
82    /// turns the others off.
83    pub layer_groups: Vec<Vec<(u32, u16)>>,
84    /// The Layers panel's rows: the default configuration's `/Order` (ISO 32000-2 Table 99), a
85    /// tree of layers (indices into `layers`) and labels. A layer it doesn't list isn't presented,
86    /// though it still draws in its state; no `/Order` presents none.
87    pub layer_order: Vec<LayerRow>,
88    pub fonts: Vec<FontInfo>,
89    pub attachments: Vec<Attachment>,
90    /// The document is a PDF portfolio (`/Collection`): its documents are `attachments`.
91    pub portfolio: Option<Portfolio>,
92    /// Pages with a presentation transition (`/Trans`) or display time (`/Dur`), in page order.
93    pub transitions: Vec<PageTransition>,
94    /// Where the document opens (`/OpenAction`): the (0-based page, view) of its destination,
95    /// named destinations resolved. `None` without one, for an action that goes nowhere in this
96    /// document (a script, say), or for one that doesn't resolve to a page of it.
97    pub open_dest: Option<(usize, DestView)>,
98    pub warnings: Vec<String>,
99}
