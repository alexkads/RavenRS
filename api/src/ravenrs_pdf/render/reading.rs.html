1//! The "reading colours" view setting (Acrobat's Replace Document Colors): solid text and
2//! line-art paints are remapped for readability while rendering. Display only — the document
3//! itself never changes. The setting flows into [`crate::third_party::hayro::RenderSettings`] through
4//! `RenderConfig`.
5
6/// The reading-colours setting for one document. `params: None` means off.
7#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
8pub struct ReadingColors {
9    /// The colours and flags, or `None` when the setting is off.
10    pub params: Option<ReadingParams>,
11}
12
13/// Colours and flags of the reading-colours mapping.
14#[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
15pub struct ReadingParams {
16    /// The sRGB colour that dark paints are mapped towards.
17    pub text: [u8; 3],
18    /// The sRGB colour that light paints are mapped towards.
19    pub background: [u8; 3],
20    /// When `true`, only near-black paints are mapped; lighter ones keep their colour.
21    pub only_black: bool,
22    /// When `true`, line-art paints are mapped in addition to text paints.
23    pub line_art: bool,
24}
25
26impl ReadingColors {
27    /// Black text on a white page.
28    pub const BLACK_ON_WHITE: Self =
29        Self { params: Some(ReadingParams { text: [0x00, 0x00, 0x00], background: [0xff, 0xff, 0xff], only_black: false, line_art: true }) };
30    /// White text on a black page.
31    pub const WHITE_ON_BLACK: Self =
32        Self { params: Some(ReadingParams { text: [0xff, 0xff, 0xff], background: [0x00, 0x00, 0x00], only_black: false, line_art: true }) };
33    /// Yellow text on a black page.
34    pub const YELLOW_ON_BLACK: Self =
35        Self { params: Some(ReadingParams { text: [0xff, 0xff, 0x00], background: [0x00, 0x00, 0x00], only_black: false, line_art: true }) };
36    /// Green text on a black page.
37    pub const GREEN_ON_BLACK: Self =
38        Self { params: Some(ReadingParams { text: [0x00, 0xff, 0x00], background: [0x00, 0x00, 0x00], only_black: false, line_art: true }) };
39
40    /// Parses tool/settings input. `mode` is one of `off | black-on-white | white-on-black |
41    /// yellow-on-black | green-on-black | custom`; `custom` needs `text` and `background` as
42    /// `#rrggbb`; the flags override the preset defaults when given. `off` ignores colours.
43    /// Errors are actionable strings.
44    pub fn parse(mode: &str, text: Option<&str>, background: Option<&str>, only_black: Option<bool>, line_art: Option<bool>) -> Result<Self, String> {
45        // Every mode accepts the flags: they override the preset defaults, and are the custom
46        // mode's own values.
47        let with_flags = |mut colors: Self| {
48            if let Some(p) = colors.params.as_mut() {
49                if let Some(v) = only_black {
50                    p.only_black = v;
51                }
52                if let Some(v) = line_art {
53                    p.line_art = v;
54                }
55            }
56            colors
57        };
58        match mode {
59            "off" => Ok(Self::default()),
60            "black-on-white" => Ok(with_flags(Self::BLACK_ON_WHITE)),
61            "white-on-black" => Ok(with_flags(Self::WHITE_ON_BLACK)),
62            "yellow-on-black" => Ok(with_flags(Self::YELLOW_ON_BLACK)),
63            "green-on-black" => Ok(with_flags(Self::GREEN_ON_BLACK)),
64            "custom" => match (text, background) {
65                (None, None) => Err("custom needs text and background as #rrggbb".into()),
66                (None, Some(_)) => Err("custom needs text as #rrggbb".into()),
67                (Some(_), None) => Err("custom needs background as #rrggbb".into()),
68                (Some(text), Some(background)) => Ok(with_flags(Self {
69                    params: Some(ReadingParams {
70                        text: parse_hex("text", text)?,
71                        background: parse_hex("background", background)?,
72                        only_black: false,
73                        line_art: true,
74                    }),
75                })),
76            },
77            other => {
78                Err(format!("unknown mode {other:?}; expected one of: off, black-on-white, white-on-black, yellow-on-black, green-on-black, custom"))
79            }
80        }
81    }
82
83    /// The mode name for echoes and display: `"off"`, a preset name, or `"custom"`.
84    pub fn mode_name(&self) -> &'static str {
85        match self.params {
86            None => "off",
87            Some(params) => {
88                if Some(params) == Self::BLACK_ON_WHITE.params {
89                    "black-on-white"
90                } else if Some(params) == Self::WHITE_ON_BLACK.params {
91                    "white-on-black"
92                } else if Some(params) == Self::YELLOW_ON_BLACK.params {
93                    "yellow-on-black"
94                } else if Some(params) == Self::GREEN_ON_BLACK.params {
95                    "green-on-black"
96                } else {
97                    "custom"
98                }
99            }
100        }
101    }
102
103    /// The text colour as lowercase `#rrggbb`, or `None` when off.
104    pub fn text_hex(&self) -> Option<String> {
105        self.params.map(|p| hex(p.text))
106    }
107
108    /// The background colour as lowercase `#rrggbb`, or `None` when off.
109    pub fn background_hex(&self) -> Option<String> {
110        self.params.map(|p| hex(p.background))
111    }
112}

114impl From<ReadingParams> for crate::third_party::hayro::ColorReplacement {

116        crate::third_party::hayro::ColorReplacement { text: p.text, background: p.background, only_black: p.only_black, line_art: p.line_art }
117    }

120/// Parses a colour given as `#rrggbb` — exactly this shape, either case of digits. The error
121/// names the argument and the expected shape.
122fn parse_hex(name: &str, value: &str) -> Result<[u8; 3], String> {
123    let invalid = || format!("invalid {name} colour {value:?}; expected #rrggbb (six hex digits, e.g. #00ff00)");
124    let Some(digits) = value.strip_prefix('#') else { return Err(invalid()) };
125    if digits.len() != 6 || !digits.bytes().all(|b| b.is_ascii_hexdigit()) {
126        return Err(invalid());
127    }
128    let mut rgb = [0u8; 3];
129    for (i, channel) in rgb.iter_mut().enumerate() {
130        // Length and charset are checked above, so the parse cannot actually fail; the mapping
131        // only keeps it total without panicking.
132        *channel = u8::from_str_radix(&digits[i * 2..i * 2 + 2], 16).map_err(|_| invalid())?;
133    }
134    Ok(rgb)
135}
136
137/// Formats sRGB bytes as lowercase `#rrggbb`.
138fn hex(rgb: [u8; 3]) -> String {
139    format!("#{:02x}{:02x}{:02x}", rgb[0], rgb[1], rgb[2])
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn parse_round_trips_every_mode() {
148        let off = ReadingColors::parse("off", None, None, None, None).unwrap();
149        assert_eq!(off, ReadingColors::default());
150        assert_eq!(off.mode_name(), "off");
151        assert_eq!(off.text_hex(), None);
152        assert_eq!(off.background_hex(), None);
153        // `off` ignores colour and flag arguments.
154        assert_eq!(ReadingColors::parse("off", Some("zzz"), Some("#000000"), Some(true), Some(false)).unwrap(), off);
155
156        for (mode, text, background) in [
157            ("black-on-white", "#000000", "#ffffff"),
158            ("white-on-black", "#ffffff", "#000000"),
159            ("yellow-on-black", "#ffff00", "#000000"),
160            ("green-on-black", "#00ff00", "#000000"),
161        ] {
162            let colors = ReadingColors::parse(mode, None, None, None, None).unwrap();
163            assert_eq!(colors.mode_name(), mode, "{mode}");
164            assert_eq!(colors.text_hex().as_deref(), Some(text), "{mode}");
165            assert_eq!(colors.background_hex().as_deref(), Some(background), "{mode}");
166            let params = colors.params.unwrap();
167            assert!(!params.only_black && params.line_art, "{mode}");
168        }
169
170        // The flags override the preset defaults; a changed preset is no longer one.
171        let overridden = ReadingColors::parse("white-on-black", None, None, Some(true), Some(false)).unwrap();
172        let params = overridden.params.unwrap();
173        assert!(params.only_black && !params.line_art);
174        assert_eq!(overridden.mode_name(), "custom");
175
176        // `custom` takes either case of hex and echoes lowercase.
177        let custom = ReadingColors::parse("custom", Some("#AaBbCc"), Some("#123456"), Some(true), None).unwrap();
178        assert_eq!(custom.mode_name(), "custom");
179        assert_eq!(custom.text_hex().as_deref(), Some("#aabbcc"));
180        assert_eq!(custom.background_hex().as_deref(), Some("#123456"));
181        let params = custom.params.unwrap();
182        assert!(params.only_black && params.line_art);
183    }

195        for bad in ["zzz", "000000", "#000", "#0000000", "#00g000", ""] {
196            assert_eq!(
197                ReadingColors::parse("custom", Some(bad), Some("#000000"), None, None),
198                Err(format!("invalid text colour {bad:?}; expected #rrggbb (six hex digits, e.g. #00ff00)")),
199                "{bad}"
200            );
201        }
202        assert_eq!(
203            ReadingColors::parse("custom", Some("#000000"), Some("ffffff"), None, None),
204            Err("invalid background colour \"ffffff\"; expected #rrggbb (six hex digits, e.g. #00ff00)".to_string())
205        );
206    }
