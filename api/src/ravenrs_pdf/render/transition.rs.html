1//! The geometry of a page transition (ISO 32000-2 §12.4.4, table 164) part-way through its run:
2//! which parts of the outgoing and the incoming page a presentation paints, where, and how
3//! opaque. Pure functions in page fractions (0 to 1 across and down the page box on screen), so
4//! the painter only scales them; nothing here touches pixels or time.
5//!
6//! `Effect` is the transition as a presentation plays it (the `/Trans` entries that matter for
7//! each style, with the spec's defaults), `frame` the picture at progress `t`. Every frame starts
8//! with the outgoing page alone (`t = 0`) and ends with the incoming page alone (`t = 1`); the
9//! pieces of a frame never leave the page box.
10
11/// A page transition as it is played: the style with the entries that style reads. Directions
12/// are degrees as `/Di` gives them: 0 left to right, 90 bottom to top, 180 right to left, 270 top
13/// to bottom, 315 top-left to bottom-right (Glitter; Wipe treats it the same way).
14#[derive(Clone, Copy, Debug, PartialEq)]
15pub enum Effect {
16    /// Two halves split along a horizontal (`horizontal`) or vertical line, revealing the new page
17    /// from the edges inward (`inward`) or from the centre outward.
18    Split { horizontal: bool, inward: bool },
19    /// Strips across (`horizontal`) or down the page open together, like Venetian blinds.
20    Blinds { horizontal: bool },
21    /// A rectangle grows from the centre (`inward` false) or the edges close in on the centre.
22    Box { inward: bool },
23    /// A line sweeps across the page in `direction`, revealing the new page behind it.
24    Wipe { direction: u16 },
25    /// Random squares of the new page appear until it is whole.
26    Dissolve,
27    /// Like Dissolve, but the squares appear in a band that sweeps in `direction`.
28    Glitter { direction: u16 },
29    /// No effect: the new page replaces the old one.
30    Replace,
31    /// The new page flies in from off-screen in `direction` (or from the centre, `None`), growing
32    /// from `scale`; `outward` flies the old page off instead. With `opaque` the moving page hides
33    /// what is under it (it always does here: the pieces overlap in paint order).
34    Fly { direction: Option<u16>, inward: bool, scale: f32 },
35    /// The new page pushes the old one off the page in `direction`.
36    Push { direction: u16 },
37    /// The new page slides in over the old one in `direction`.
38    Cover { direction: u16 },
39    /// The old page slides off in `direction`, uncovering the new one beneath.
40    Uncover { direction: u16 },
41    /// The new page fades in over the old one.
42    Fade,
43}
44
45impl Effect {
46    /// The effect for a `/Trans` dictionary read as its parts: the `/S` name, `/Dm` as
47    /// `horizontal`, `/M` as `inward`, `/Di` as `direction` (`None` for `/None`), `/SS` and `/B`.
48    /// An unknown style is `Replace`, as the spec's default; an entry a style does not read is
49    /// ignored.
50    pub fn from_parts(style: &str, horizontal: bool, inward: bool, direction: Option<u16>, scale: f32) -> Self {
51        let dir = direction.filter(|d| matches!(d, 0 | 90 | 180 | 270 | 315)).unwrap_or(0);
52        match style {
53            "Split" => Self::Split { horizontal, inward },
54            "Blinds" => Self::Blinds { horizontal },
55            "Box" => Self::Box { inward },
56            "Wipe" => Self::Wipe { direction: dir },
57            "Dissolve" => Self::Dissolve,
58            "Glitter" => Self::Glitter { direction: dir },
59            "Fly" => Self::Fly {
60                direction: direction.filter(|d| matches!(d, 0 | 90 | 180 | 270)),
61                inward,
62                scale: if scale.is_finite() && scale > 0.0 { scale.min(1.0) } else { 1.0 },
63            },
64            "Push" => Self::Push { direction: dir },
65            "Cover" => Self::Cover { direction: dir },
66            "Uncover" => Self::Uncover { direction: dir },
67            "Fade" => Self::Fade,
68            _ => Self::Replace,
69        }
70    }
71}

73/// A part of a page to paint: `src` is the part of that page, `dst` where it goes, both as
74/// fractions `[x0, y0, x1, y1]` of the page box (y down), `alpha` its opacity.
75#[derive(Clone, Copy, Debug, PartialEq)]
76pub struct Piece {
77    pub src: [f32; 4],
78    pub dst: [f32; 4],
79    pub alpha: f32,
80}
81
82/// One moment of a transition: the pieces of the outgoing page, then of the incoming page, in
83/// paint order (later pieces over earlier ones).
84#[derive(Clone, Debug, Default, PartialEq)]
85pub struct Frame {
86    pub outgoing: Vec<Piece>,
87    pub incoming: Vec<Piece>,
88}
89
90/// Strips in Blinds, and cells per side in Dissolve and Glitter.
91const BLINDS: usize = 8;
92const GRID: usize = 8;
93/// Strips that approximate a diagonal sweep.
94const DIAGONAL_STRIPS: usize = 16;
95const FULL: [f32; 4] = [0.0, 0.0, 1.0, 1.0];
96
97fn whole(alpha: f32) -> Piece {
98    Piece { src: FULL, dst: FULL, alpha }
99}
100
101/// A piece of the page at `rect` (fractions), shown in place.
102fn at(rect: [f32; 4]) -> Piece {
103    Piece { src: rect, dst: rect, alpha: 1.0 }
104}
105
106/// The whole page shifted by `dx, dy` (fractions), clipped to the page box, or `None` when it is
107/// entirely off the page.
108fn shifted(dx: f32, dy: f32) -> Option<Piece> {
109    let dst = [dx.max(0.0), dy.max(0.0), (1.0 + dx).min(1.0), (1.0 + dy).min(1.0)];
110    if dst[2] - dst[0] <= 0.0 || dst[3] - dst[1] <= 0.0 {
111        return None;
112    }
113    let src = [dst[0] - dx, dst[1] - dy, dst[2] - dx, dst[3] - dy];
114    Some(Piece { src, dst, alpha: 1.0 })
115}

117/// Where a page is at `k` of the way along `direction` (0 at the start, 1 at the end of a move
118/// across the whole page), as an offset in fractions.
119fn offset(direction: u16, k: f32) -> (f32, f32) {
120    match direction {
121        90 => (0.0, -k),
122        180 => (-k, 0.0),
123        270 => (0.0, k),
124        _ => (k, 0.0),
125    }
126}
127
128/// A small deterministic hash, so Dissolve and Glitter look random but repeat exactly.
129fn hash(i: u32) -> u32 {
130    let mut x = i.wrapping_mul(0x9E37_79B9).wrapping_add(0x7F4A_7C15);
131    x ^= x >> 15;
132    x = x.wrapping_mul(0x2C1B_3C6D);
133    x ^= x >> 12;
134    x
135}

137/// The cells of the dissolve grid in the order they appear: by `key`, ties by position.
138fn cells_in_order(key: impl Fn(usize, usize) -> f32) -> Vec<[f32; 4]> {
139    let mut cells: Vec<(f32, usize, usize)> = (0..GRID * GRID).map(|i| (key(i % GRID, i / GRID), i / GRID, i % GRID)).collect();
140    cells.sort_by(|a, b| a.0.total_cmp(&b.0).then(a.1.cmp(&b.1)).then(a.2.cmp(&b.2)));
141    let side = 1.0 / GRID as f32;
142    cells.into_iter().map(|(_, row, col)| [col as f32 * side, row as f32 * side, (col + 1) as f32 * side, (row + 1) as f32 * side]).collect()
143}

145/// The pieces of the incoming page at `t` for a diagonal sweep: a point is revealed once its
146/// distance along the diagonal, `(x + y) / 2`, is within `t`, approximated by vertical strips.
147fn diagonal(t: f32) -> Vec<Piece> {
148    let w = 1.0 / DIAGONAL_STRIPS as f32;
149    (0..DIAGONAL_STRIPS)
150        .filter_map(|k| {
151            let x0 = k as f32 * w;
152            let h = (2.0 * t - (x0 + w / 2.0)).clamp(0.0, 1.0);
153            (h > 0.0).then(|| at([x0, 0.0, x0 + w, h]))
154        })
155        .collect()
156}
157
158/// The incoming page revealed by a straight sweep in `direction` at `t`.
159fn swept(direction: u16, t: f32) -> Vec<Piece> {
160    match direction {
161        90 => vec![at([0.0, 1.0 - t, 1.0, 1.0])],
162        180 => vec![at([1.0 - t, 0.0, 1.0, 1.0])],
163        270 => vec![at([0.0, 0.0, 1.0, t])],
164        315 => diagonal(t),
165        _ => vec![at([0.0, 0.0, t, 1.0])],
166    }
167}

169/// The picture of `effect` at progress `t` (0: the outgoing page alone; 1: the incoming page
170/// alone; anything else clamped). Pieces are in paint order and never leave the page box.
171pub fn frame(effect: &Effect, t: f32) -> Frame {
172    let t = if t.is_finite() { t.clamp(0.0, 1.0) } else { 1.0 };
173    if t <= 0.0 {
174        return Frame { outgoing: vec![whole(1.0)], incoming: Vec::new() };
175    }
176    if t >= 1.0 {
177        return Frame { outgoing: Vec::new(), incoming: vec![whole(1.0)] };
178    }
179    let over_old = |incoming: Vec<Piece>| Frame { outgoing: vec![whole(1.0)], incoming };
180    match *effect {
181        Effect::Replace => Frame { outgoing: Vec::new(), incoming: vec![whole(1.0)] },
182        Effect::Fade => Frame { outgoing: vec![whole(1.0)], incoming: vec![whole(t)] },
183        Effect::Wipe { direction } => over_old(swept(direction, t)),
184        Effect::Split { horizontal, inward } => {
185            let h = t / 2.0;
186            let bands = match (horizontal, inward) {
187                (true, true) => vec![at([0.0, 0.0, 1.0, h]), at([0.0, 1.0 - h, 1.0, 1.0])],
188                (true, false) => vec![at([0.0, 0.5 - h, 1.0, 0.5 + h])],
189                (false, true) => vec![at([0.0, 0.0, h, 1.0]), at([1.0 - h, 0.0, 1.0, 1.0])],
190                (false, false) => vec![at([0.5 - h, 0.0, 0.5 + h, 1.0])],
191            };
192            over_old(bands)
193        }
194        Effect::Blinds { horizontal } => {
195            let w = 1.0 / BLINDS as f32;
196            let strips = (0..BLINDS)
197                .map(|k| {
198                    let a = k as f32 * w;
199                    if horizontal { at([0.0, a, 1.0, a + t * w]) } else { at([a, 0.0, a + t * w, 1.0]) }
200                })
201                .collect();
202            over_old(strips)
203        }
204        Effect::Box { inward } => {
205            let h = t / 2.0;
206            let pieces = if inward {
207                vec![at([0.0, 0.0, 1.0, h]), at([0.0, 1.0 - h, 1.0, 1.0]), at([0.0, h, h, 1.0 - h]), at([1.0 - h, h, 1.0, 1.0 - h])]
208            } else {
209                vec![at([0.5 - h, 0.5 - h, 0.5 + h, 0.5 + h])]
210            };
211            over_old(pieces)
212        }
213        Effect::Dissolve => {
214            let cells = cells_in_order(|col, row| hash((row * GRID + col) as u32) as f32);
215            let shown = ((t * cells.len() as f32).round() as usize).min(cells.len());
216            over_old(cells.into_iter().take(shown).map(at).collect())
217        }
218        Effect::Glitter { direction } => {
219            // Ordered by how far along the sweep a cell lies, jittered so the band sparkles.
220            let along = |col: usize, row: usize| -> f32 {
221                let (x, y) = ((col as f32 + 0.5) / GRID as f32, (row as f32 + 0.5) / GRID as f32);
222                match direction {
223                    90 => 1.0 - y,
224                    180 => 1.0 - x,
225                    270 => y,
226                    315 => (x + y) / 2.0,
227                    _ => x,
228                }
229            };
230            let cells = cells_in_order(|col, row| along(col, row) + (hash((row * GRID + col) as u32 + 977) % 1000) as f32 / 4000.0);
231            let shown = ((t * cells.len() as f32).round() as usize).min(cells.len());
232            over_old(cells.into_iter().take(shown).map(at).collect())
233        }
234        Effect::Push { direction } => {
235            let (dx, dy) = offset(direction, t);
236            let (ux, uy) = offset(direction, 1.0);
237            Frame { outgoing: shifted(dx, dy).into_iter().collect(), incoming: shifted(dx - ux, dy - uy).into_iter().collect() }
238        }
239        Effect::Cover { direction } => {
240            let (dx, dy) = offset(direction, t);
241            let (ux, uy) = offset(direction, 1.0);
242            over_old(shifted(dx - ux, dy - uy).into_iter().collect())
243        }
244        Effect::Uncover { direction } => {
245            let (dx, dy) = offset(direction, t);
246            Frame { outgoing: shifted(dx, dy).into_iter().collect(), incoming: vec![whole(1.0)] }
247        }
248        Effect::Fly { direction, inward, scale } => {
249            // The moving page: at `k` of its flight (0 off-screen or small, 1 whole and in place).
250            // `leaving`: the page moves on in `direction` (flying out); else it comes in from the
251            // side it would have left from (flying in).
252            let flight = |k: f32, leaving: bool| -> Option<Piece> {
253                let s = scale + (1.0 - scale) * k;
254                let (dx, dy) = direction.map_or((0.0, 0.0), |d| {
255                    let (ox, oy) = offset(d, 1.0 - k);
256                    if leaving { (ox, oy) } else { (-ox, -oy) }
257                });
258                // The page at scale `s`, centred, shifted by (dx, dy), clipped to the page box.
259                let half = s / 2.0;
260                let dst = [0.5 - half + dx, 0.5 - half + dy, 0.5 + half + dx, 0.5 + half + dy];
261                let clip = [dst[0].max(0.0), dst[1].max(0.0), dst[2].min(1.0), dst[3].min(1.0)];
262                if clip[2] - clip[0] <= 0.0 || clip[3] - clip[1] <= 0.0 || s <= 0.0 {
263                    return None;
264                }
265                let src = [(clip[0] - dst[0]) / s, (clip[1] - dst[1]) / s, (clip[2] - dst[0]) / s, (clip[3] - dst[1]) / s];
266                Some(Piece { src, dst: clip, alpha: 1.0 })
267            };
268            if inward {
269                Frame { outgoing: vec![whole(1.0)], incoming: flight(t, false).into_iter().collect() }
270            } else {
271                Frame { outgoing: flight(1.0 - t, true).into_iter().collect(), incoming: vec![whole(1.0)] }
272            }
273        }
274    }

281    fn area(p: &Piece) -> f32 {
282        (p.dst[2] - p.dst[0]).max(0.0) * (p.dst[3] - p.dst[1]).max(0.0)
283    }

285    fn inside(p: &Piece) -> bool {
286        p.dst.iter().chain(p.src.iter()).all(|v| (-1e-4..=1.0 + 1e-4).contains(v)) && p.dst[0] <= p.dst[2] && p.dst[1] <= p.dst[3]
287    }

339        // For every effect that reveals the new page in place, the revealed area never shrinks.
340        for e in all_effects().into_iter().filter(|e| {
341            !matches!(e, Effect::Replace | Effect::Fade | Effect::Push { .. } | Effect::Cover { .. } | Effect::Uncover { .. } | Effect::Fly { .. })

346                let a: f32 = f.incoming.iter().map(area).sum();

431        assert_eq!(Effect::from_parts("Wipe", true, true, Some(270), 1.0), Effect::Wipe { direction: 270 });
432        assert_eq!(
433            Effect::from_parts("Wipe", true, true, Some(45), 1.0),
434            Effect::Wipe { direction: 0 },
435            "a direction outside the five is the default"
436        );
437        assert_eq!(Effect::from_parts("Split", false, false, Some(90), 1.0), Effect::Split { horizontal: false, inward: false });
438        assert_eq!(Effect::from_parts("Fly", true, true, None, 0.5), Effect::Fly { direction: None, inward: true, scale: 0.5 });
439        assert_eq!(
440            Effect::from_parts("Fly", true, false, Some(315), f32::NAN),
441            Effect::Fly { direction: None, inward: false, scale: 1.0 },
442            "Fly has no diagonal; a bad scale is 1"
443        );
444        assert_eq!(Effect::from_parts("Swirl", true, true, Some(0), 1.0), Effect::Replace);
445        assert_eq!(Effect::from_parts("R", true, true, Some(0), 1.0), Effect::Replace);
446    }
