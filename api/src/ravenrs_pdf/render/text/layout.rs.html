8/// Glyphs in reading order, as indices into the page's glyphs, with their lines and word gaps.
9pub(super) struct Flow {
10    /// The top of the flow's topmost glyph in view space.
11    pub(super) top: f32,
12    /// The quarter turn whose frame the flow was laid out in.
13    pub(super) quarter: Option<u8>,
14    pub(super) ids: Vec<usize>,
15    pub(super) line_of: Vec<u32>,
16    pub(super) space_before: Vec<bool>,
17}

19/// Lay out the page's glyphs `ids`, giving each the box at the same position in `rects`.
20pub(super) fn arrange_flow(glyphs: &[Drawn], ids: Vec<usize>, rects: Vec<[f32; 4]>, quarter: Option<u8>) -> Flow {
21    let top = ids.iter().filter_map(|&i| glyphs.get(i)).map(|g| g.2.rect[1]).fold(f32::INFINITY, f32::min);
22    let framed: Vec<TextGlyph> = ids.iter().zip(rects).filter_map(|(&i, rect)| glyphs.get(i).map(|g| TextGlyph { rect, ..g.2.clone() })).collect();
23    let (order, line_of, space_before) = arrange(&framed);
24    let ids = order.into_iter().filter_map(|k| ids.get(k).copied()).collect();
25    Flow { top, quarter, ids, line_of, space_before }
26}

28/// Boxes for a run of glyphs set along a curve: each glyph's box along its own direction, placed
29/// where the pen moved in the previous glyph's direction. The path straightened out.
30pub(super) fn unrolled<'g>(run: impl Iterator<Item = &'g Drawn>) -> Vec<[f32; 4]> {
31    let mut pen = [0.0, 0.0];
32    let mut previous: Option<([f64; 2], [f64; 4])> = None;
33    run.map(|(_, u, g)| {
34        let b = along_baseline(&g.quad, *u);
35        if let Some((pu, pb)) = previous {
36            let at = along_baseline(&g.quad, pu);
37            pen = [pen[0] + at[0] - pb[0], pen[1] + (at[1] + at[3] - pb[1] - pb[3]) / 2.0];
38        }
39        previous = Some((*u, b));
40        let half = (b[3] - b[1]) / 2.0;
41        [pen[0], pen[1] - half, pen[0] + b[2] - b[0], pen[1] + half].map(|c| c as f32)
42    })
43    .collect()
44}

46/// View-space box → the frame where text runs left to right, for text running `quarter` × 90°
47/// clockwise from that in a `w` × `h` view (1: text runs downwards).
48pub(super) fn to_upright(r: [f32; 4], quarter: u8, w: f32, h: f32) -> [f32; 4] {
49    let [x0, y0, x1, y1] = r;
50    match quarter {
51        1 => [y0, w - x1, y1, w - x0],
52        2 => [w - x1, h - y1, w - x0, h - y0],
53        3 => [h - y1, x0, h - y0, x1],
54        _ => r,
55    }
56}

58/// Whether two writing directions are close enough to share a frame ([`DIRECTION_TOLERANCE`]).
59pub(super) fn compatible(a: [f64; 2], b: [f64; 2]) -> bool {
60    a[0] * b[0] + a[1] * b[1] > 0.0 && (a[0] * b[1] - a[1] * b[0]).abs() <= DIRECTION_TOLERANCE
61}

63/// Whether glyph `g` continues the line of glyph `p` (content order), by the test `layout` uses
64/// for consecutive glyphs, taken in `p`'s own frame. The direction may turn by less than half a
65/// quarter turn, as text set along a curve does.
66pub(super) fn continues(p: &Drawn, g: &Drawn) -> bool {
67    let (pu, gu) = (p.1, g.1);
68    if (pu[0] * gu[1] - pu[1] * gu[0]).abs() >= pu[0] * gu[0] + pu[1] * gu[1] {
69        return false;
70    }
71    let (pb, gb, own) = (along_baseline(&p.2.quad, pu), along_baseline(&g.2.quad, pu), along_baseline(&g.2.quad, gu));
72    let h = (pb[3] - pb[1]).min(own[3] - own[1]).max(0.1);
73    let cy = |b: [f64; 4]| (b[1] + b[3]) / 2.0;
74    (cy(gb) - cy(pb)).abs() < h * 0.5 && gb[0] - pb[2] < h * 3.0 && gb[0] > pb[0] - h * 2.0
75}

77/// A view-space quad → its box in the frame where `u` runs left to right and `u` turned a
78/// quarter clockwise runs downwards. Along `u` it spans the middles of the quad's start and end
79/// sides, so a skewed glyph is no wider than its advance; across, it spans every corner.
80pub(super) fn along_baseline(quad: &[[f32; 2]; 4], u: [f64; 2]) -> [f64; 4] {
81    let p = quad.map(|[x, y]| [f64::from(x) * u[0] + f64::from(y) * u[1], f64::from(y) * u[0] - f64::from(x) * u[1]]);
82    let (a0, a1) = ((p[0][0] + p[3][0]) / 2.0, (p[1][0] + p[2][0]) / 2.0);
83    let (c0, c1) = p.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(lo, hi), q| (lo.min(q[1]), hi.max(q[1])));
84    [a0.min(a1), c0, a0.max(a1), c1]
85}

87/// The inverse of [`to_upright`].
88pub(super) fn from_upright(r: [f32; 4], quarter: u8, w: f32, h: f32) -> [f32; 4] {
89    let [u0, v0, u1, v1] = r;
90    match quarter {
91        1 => [w - v1, u0, w - v0, u1],
92        2 => [w - u1, h - v1, w - u0, h - v0],
93        3 => [v0, h - u1, v1, h - u0],
94        _ => r,
95    }
96}

108/// Whether `text` is only combining marks (Unicode Mn and Me, which take bidi class NSM) of a
109/// left-to-right script: Thai tone marks and above or below vowels, Indic signs such as virama,
110/// Latin combining accents. Right-to-left marks are ordered with their script by [`reverse_rtl`].
111fn is_mark(text: &str) -> bool {
112    !text.is_empty() && text.chars().all(|c| unicode_bidi::bidi_class(c) == unicode_bidi::BidiClass::NSM && !is_rtl(c))
113}

115/// How far from the glyph before it, as a fraction of its height, a combining mark may start and
116/// still ride on it. A zero-advance mark starts at the pen after its base, give or take a small
117/// mark-positioning offset, and its ink is drawn back over the base.
118const MARK_REACH: f32 = 0.3;
119
120/// Whether glyph `m`, drawn right after `p` in one run, is a combining mark on `p`: ISO 32000-2
121/// §9.10 leaves extraction to the processor, and Unicode stores a mark after its base, as the
122/// content stream draws it here. The mark then stays after `p` in reading order however far its
123/// own box sits along the line; sorting by x alone moved a Thai tone mark that starts a hair past
124/// its base's advance after the next letter (#6517).
125fn rides_on(p: &TextGlyph, m: &TextGlyph) -> bool {
126    let reach = MARK_REACH * (m.rect[3] - m.rect[1]).max(0.1);
127    is_mark(&m.text)
128        && p.text.chars().any(|c| !c.is_whitespace())
129        && !p.text.chars().any(is_rtl)
130        && m.rect[0] >= p.rect[0] - reach
131        && m.rect[0] <= p.rect[2] + reach
132}
133
134/// How far, as a fraction of its letter's height, a combining mark's em box may sit above or below the letter's and
135/// still ride on it. A tone mark lifted over an upper vowel can be drawn more than half an em higher, past where a
136/// glyph of its own would start a new line.
137const MARK_RISE: f32 = 1.0;
138
139/// Whether glyph `m`, drawn right after glyph `p` (a letter, or a mark riding on letter `a`), rides on that cluster:
140/// on `p` itself, or, for a mark stacked on another one, on its letter `a`.
141fn rides_on_cluster(glyphs: &[&TextGlyph], p: usize, a: usize, m: &TextGlyph) -> bool {
142    glyphs.get(p).is_some_and(|pg| rides_on(pg, m)) || (a != p && glyphs.get(a).is_some_and(|ag| rides_on(ag, m)))
143}

145/// Glyph `i` continues the run of glyph `i - 1`: when it is a combining mark on that glyph's cluster, it joins it.
146fn attach_mark(glyphs: &[&TextGlyph], anchor: &mut [usize], i: usize) {
147    let Some(p) = i.checked_sub(1) else { return };
148    let Some(&a) = anchor.get(p) else { return };
149    if let Some(mg) = glyphs.get(i)
150        && rides_on_cluster(glyphs, p, a, mg)
151        && let Some(slot) = anchor.get_mut(i)
152    {
153        *slot = a;
154    }
155}

161pub(super) fn union(a: &mut [f32; 4], b: &[f32; 4]) {
162    a[0] = a[0].min(b[0]);
163    a[1] = a[1].min(b[1]);
164    a[2] = a[2].max(b[2]);
165    a[3] = a[3].max(b[3]);
166}

168/// Reading-order layout. Glyphs are reordered so that `glyphs` is in reading order, lines are
169/// numbered in that order, and word gaps are inferred per line.
170///
171/// Heuristics (documented so they can be tuned and tested; plan/architecture.md §8):
172/// 1. *Segments:* consecutive glyphs (content order) on one baseline with no column-sized gap.
173///    A smaller glyph right next to the previous one (a sub- or superscript) stays in the segment.
174///    Segments on the same baseline that nearly touch are merged (text drawn out of order).
175/// 2. *Blocks:* segments stacked with line-sized gaps and overlapping horizontally.
176/// 3. *Block order:* repeatedly take the top-most remaining block, preferring a block to its left
177///    that overlaps it vertically (so columns read left, then right). A column goes on first (#3658):
178///    the nearest block below the last one, overlapping it horizontally and at most 2.5 heights of
179///    its last line below it, comes next, unless an unread block sits above it in its own span, sits
180///    beside it on its left, or shares its first line (the first lines overlap by at least 30% of the
181///    taller one: a table or form row, which reads across, even with labels and values in different
182///    sizes or on shifted baselines).
183///    Known limit: when a paragraph in the next column starts on the same line as the continuation
184///    (text on a baseline grid), it counts as a row and the top-most rule applies, so the column
185///    order of #3658 is lost there. Telling prose from a row in that case needs more evidence.
186/// 4. Within a segment glyphs run left to right; runs of right-to-left script are reversed so the
187///    text comes out in logical order. A block whose first line sits on the baseline of the line read
188///    just before it, to its right and less than [`SAME_LINE_GAP`] heights away, continues that line
189///    after a space (a label and its value, words drawn out of order or in another size). A block of
190///    several lines does so only after a one-line block narrower than [`SAME_LINE_GAP`] heights (a
191///    label): beside a wider one, or the last line of a column, it is the next column.
192pub fn layout(glyphs: Vec<TextGlyph>) -> PageText {
193    let (order, line_of, space_before) = arrange(&glyphs);
194    let mut slots: Vec<Option<TextGlyph>> = glyphs.into_iter().map(Some).collect();
195    // Each glyph is emitted exactly once.
196    let glyphs = order.iter().filter_map(|i| slots.get_mut(*i).and_then(Option::take)).collect();
197    PageText { glyphs, line_of, space_before, truncated: false }
198}

200/// [`layout`] as the reading order of indices into `glyphs`, with each one's line and word gap.
201fn arrange(glyphs: &[TextGlyph]) -> (Vec<usize>, Vec<u32>, Vec<bool>) {
202    // Drop "fake bold" duplicates: the same character redrawn at (almost) the same place. A redraw
203    // also overlaps most of its own width: a narrow letter in condensed text (80 Tz and below) sits
204    // closer than a fifth of its height to its double, as the second "l" of "Alle" (#1467), and
205    // OCR layers squeeze words to 20 Tz and below (#2457). Only a zero-width mark, which has no
206    // width to overlap, is matched by its height alone.
207    let mut keep = vec![true; glyphs.len()];
208    for i in 1..glyphs.len() {
209        let g = &glyphs[i];
210        let h = (g.rect[3] - g.rect[1]).max(0.1);
211        let w = g.rect[2] - g.rect[0];
212        let dx = if w > 0.0 { (h * 0.2).min(w * 0.5) } else { h * 0.2 };
213        for j in (i.saturating_sub(4)..i).rev() {
214            let p = &glyphs[j];
215            if keep[j] && p.text == g.text && (p.rect[0] - g.rect[0]).abs() < dx && (p.rect[1] - g.rect[1]).abs() < h * 0.2 {
216                keep[i] = false;
217                break;
218            }
219        }
220    }
221    let kept: Vec<usize> = (0..glyphs.len()).filter(|i| keep[*i]).collect();
222    let glyphs: Vec<&TextGlyph> = kept.iter().map(|i| &glyphs[*i]).collect();
223    let n = glyphs.len();
224    if n == 0 {
225        return Default::default();
226    }
227    let height = |i: usize| (glyphs[i].rect[3] - glyphs[i].rect[1]).max(0.1);
228    let cy = |i: usize| (glyphs[i].rect[1] + glyphs[i].rect[3]) / 2.0;
229
230    // 1. Segments in content order. Every run is a range of consecutive glyphs.
231    let mut segs: Vec<Seg> = Vec::new();
232    // `anchor[i]`: the glyph a combining mark rides on (the first glyph of its cluster), or `i`.
233    let mut anchor: Vec<usize> = (0..n).collect();
234    let mut run_start_x = 0.0;
235    let mut furthest_progress = 0.0;
236    for i in 0..n {
237        let g = glyphs[i].rect;
238        let hi = height(i);
239        let cont = segs.last().is_some_and(|s| {
240            let Some(last) = s.first.end.checked_sub(1) else { return false };
241            // A glyph after a combining mark continues from the mark's letter, and a mark on that letter's cluster
242            // stays on its line even when drawn raised or lowered by up to an em (`MARK_RISE`).
243            let p = anchor.get(last).copied().unwrap_or(last);
244            if rides_on_cluster(&glyphs, last, p, glyphs[i]) && (cy(i) - cy(p)).abs() < height(p) * MARK_RISE {
245                return true;
246            }
247            let ph = height(p).min(hi);
248            let sh = s.h.min(hi);
249            let gap = g[0] - glyphs[p].rect[2];
250            // A sub- or superscript is a smaller glyph on a shifted baseline: its em box centre sits off the line by
251            // more than half its own height, but by less than half the height of the full-size glyph beside it.
252            let centre_limit = if gap.abs() < ph { height(p).max(hi) * 0.5 } else { ph * 0.5 };
253            let same_line = ((cy(i) - cy(p)).abs() < centre_limit && gap < ph * 3.0)
254                || ((cy(i) - s.cy()).abs() < sh * 0.3 && (s.h / hi - 1.0).abs() < 0.35 && gap < sh * 3.0);
255            let on_seg = gap < s.h * 1.2 && (is_script_on(&s.bbox, s.h, &g, hi) || is_script_on(&g, hi, &s.bbox, s.h));
256            (same_line || on_seg) && g[0] > glyphs[p].rect[0] - sh * 2.0
257        });
258        let reset = cont && segs.last().is_some_and(|s| restarts_run(&glyphs, s.first.start, i, furthest_progress));
259        if cont
260            && !reset
261            && let Some(s) = segs.last_mut()
262        {
263            attach_mark(&glyphs, &mut anchor, i);
264            s.first.end = i + 1;
265            s.extend_box(&g, hi);
266            furthest_progress = furthest_progress.max((g[0] - run_start_x).abs()).max((g[2] - run_start_x).abs());
267        } else {
268            segs.push(Seg::new(Run { start: i, end: i + 1 }, g, hi));
269            run_start_x = g[0];
270            furthest_progress = 0.0;
271        }
272    }
273    // A run that redraws an earlier run at the same place (doubled or shadowed text) is dropped, as a doubled glyph is
274    // above. Runs are keyed by a fingerprint of their glyph texts and a grid cell a fifth of a line high. A key leads to
275    // the latest retained run with it, and `older[k]` links each retained run to the previous one with the same key, so
276    // the earlier candidates are checked newest first, up to `REDRAW_CANDIDATES`. Both exist only for this pass.
277    let retained = {
278        let candidates = segs.iter().filter(|s| s.first.len() >= 2).count();
279        let mut latest: HashMap<(u64, i32, i32), usize> = HashMap::with_capacity(candidates);
280        let mut older = vec![NONE; segs.len()];
281        let mut kept_runs = 0;
282        for i in 0..segs.len() {
283            let run = segs[i].first;
284            if run.len() >= 2 {
285                let tol = 0.2 * segs[i].h;
286                let (cx, cy_) = ((glyphs[run.start].rect[0] / tol).floor() as i32, (glyphs[run.start].rect[1] / tol).floor() as i32);
287                let fp = fingerprint(&glyphs, run);
288                let mut checks = REDRAW_CANDIDATES;
289                let twin = (-1..=1i32).any(|dx| {
290                    (-1..=1i32).any(|dy| {
291                        let mut link = latest.get(&(fp, cx.saturating_add(dx), cy_.saturating_add(dy))).copied();
292                        while let Some(k) = link {
293                            let Some(prev) = segs.get(k) else { break };
294                            if checks == 0 {
295                                return false;
296                            }
297                            checks -= 1;
298                            if redraws(&glyphs, prev.first, run) {
299                                return true;
300                            }
301                            link = older.get(k).copied().filter(|&o| o != NONE);
302                        }
303                        false
304                    })
305                });
306                if twin {
307                    continue;
308                }
309                older[kept_runs] = latest.insert((fp, cx, cy_), kept_runs).unwrap_or(NONE);
310            }
311            segs.swap(kept_runs, i);
312            kept_runs += 1;
313        }
314        kept_runs
315    };
316    segs.truncate(retained);
317    // Merge same-baseline segments that nearly touch, then attach any remaining superscript or
318    // subscript segments to the closest host line segment they sit on.
319    let mut merged = true;
320    while merged {
321        merged = false;
322        'outer: for a in 0..segs.len() {
323            for b in 0..segs.len() {
324                if a == b {
325                    continue;
326                }
327                let (sa, sb) = (&segs[a], &segs[b]);
328                let h = sa.h.min(sb.h);
329                let gap = (sb.bbox[0] - sa.bbox[2]).max(sa.bbox[0] - sb.bbox[2]);
330                if (sa.cy() - sb.cy()).abs() < h * 0.3 && (sa.h / sb.h - 1.0).abs() < 0.35 && gap < h * 1.2 {
331                    let sb = segs.remove(b);
332                    let a = if b < a { a - 1 } else { a };
333                    segs[a].merge(sb);
334                    merged = true;
335                    break 'outer;
336                }
337            }
338        }
339        if !merged {
340            let mut b = 0;
341            while b < segs.len() {
342                let sb = &segs[b];
343                let best_host = (0..segs.len())
344                    .filter(|&a| {
345                        if a == b {
346                            return false;
347                        }
348                        let sa = &segs[a];
349                        let gap = (sb.bbox[0] - sa.bbox[2]).max(sa.bbox[0] - sb.bbox[2]);
350                        sa.bbox[2] - sa.bbox[0] > sb.bbox[2] - sb.bbox[0] && gap < sa.h * 1.2 && is_script_on(&sa.bbox, sa.h, &sb.bbox, sb.h)
351                    })
352                    .min_by(|&a1, &a2| (segs[a1].cy() - sb.cy()).abs().total_cmp(&(segs[a2].cy() - sb.cy()).abs()));
353                if let Some(a) = best_host {
354                    let sb = segs.remove(b);
355                    let a = if b < a { a - 1 } else { a };
356                    segs[a].merge(sb);
357                    merged = true;
358                } else {
359                    b += 1;
360                }
361            }
362        }
363    }
364
365    // 2. Blocks.
366    let mut by_top: Vec<usize> = (0..segs.len()).collect();
367    by_top.sort_by(|a, b| segs[*a].bbox[1].total_cmp(&segs[*b].bbox[1]));
368    let mut blocks: Vec<(Vec<usize>, [f32; 4])> = Vec::new();
369    for si in by_top {
370        let s = &segs[si];
371        let target = blocks.iter().position(|(members, bb)| {
372            let Some(&m) = members.last() else { return false };
373            let last = &segs[m];
374            let vgap = s.bbox[1] - last.bbox[3];
375            let overlap = s.bbox[2].min(bb[2]) - s.bbox[0].max(bb[0]);
376            let minw = (s.bbox[2] - s.bbox[0]).min(bb[2] - bb[0]).max(1.0);
377            vgap > -last.h * 0.5 && vgap < last.h.max(s.h) * 1.1 && overlap > minw * 0.3 && (last.h / s.h - 1.0).abs() < 0.6
378        });
379        match target {
380            Some(b) => {
381                blocks[b].0.push(si);
382                let bb = s.bbox;
383                union(&mut blocks[b].1, &bb);
384            }
385            None => blocks.push((vec![si], s.bbox)),
386        }
387    }
388
389    // 3. Block order.
390    let mut remaining: Vec<usize> = (0..blocks.len()).collect();
391    let mut block_order: Vec<usize> = Vec::with_capacity(blocks.len());
392    // A block to the left of `cb` that shares at least half the height of the shorter one: it reads first.
393    let beside_left = |cb: &[f32; 4], lb: &[f32; 4]| {
394        let v = lb[3].min(cb[3]) - lb[1].max(cb[1]);
395        lb[2] <= cb[0] + 1.0 && v > 0.5 * (lb[3] - lb[1]).min(cb[3] - cb[1])
396    };
397    // The box of a block's first line. Every block is created with one segment, so the block's own box is a fallback
398    // that is never used.
399    let first_line = |b: usize| blocks[b].0.first().and_then(|&s| segs.get(s)).map_or(blocks[b].1, |s| s.bbox);
400    while let Some(&top) = remaining.iter().min_by(|a, b| blocks[**a].1[1].total_cmp(&blocks[**b].1[1])) {
401        // A column goes on past a paragraph gap (#3658): the nearest block below the one just read that overlaps it
402        // horizontally comes next, if the gap above it is at most two and a half heights of the last line read (a
403        // footer, a footnote or a caption under a figure is further) and no other unread block
404        // - sits above it in its own horizontal span (the rest of a column above a full-width heading or footnote),
405        // - is beside it on its left, or
406        // - shares its first line: their first lines overlap by at least 30% of the taller one (a table or form row,
407        //   which reads across, even with labels and values in different sizes or on shifted baselines; a line that
408        //   only touches it, or a small superscript or footnote marker, is not a row).
409        let next_in_column = block_order.last().and_then(|&prev| {
410            let pb = blocks[prev].1;
411            // The gap is measured in the height of the block's last line, which borders it.
412            let pl = blocks[prev].0.last().and_then(|&s| segs.get(s)).map_or(pb, |s| s.bbox);
413            let tol = (pl[3] - pl[1]).max(0.1) * 0.5;
414            let c = remaining
415                .iter()
416                .copied()
417                .filter(|&c| {
418                    let cb = blocks[c].1;
419                    let overlap = cb[2].min(pb[2]) - cb[0].max(pb[0]);
420                    let minw = (cb[2] - cb[0]).min(pb[2] - pb[0]).max(1.0);
421                    cb[1] >= pb[3] - tol && overlap > minw * 0.3
422                })
423                .min_by(|a, b| blocks[*a].1[1].total_cmp(&blocks[*b].1[1]))?;
424            let cb = blocks[c].1;
425            if cb[1] - pb[3] >= tol * 5.0 {
426                return None;
427            }
428            let cl = first_line(c);
429            let blocked = remaining.iter().any(|&r| {
430                if r == c {
431                    return false;
432                }
433                let rb = blocks[r].1;
434                let rl = first_line(r);
435                let shared = rl[3].min(cl[3]) - rl[1].max(cl[1]);
436                let row = shared > 0.3 * (rl[3] - rl[1]).max(cl[3] - cl[1]);
437                (rb[1] < cb[1] && rb[2].min(cb[2]) > rb[0].max(cb[0])) || beside_left(&cb, &rb) || row
438            });
439            (!blocked).then_some(c)
440        });
441        let pick = next_in_column.unwrap_or_else(|| {
442            let tb = blocks[top].1;
443            remaining
444                .iter()
445                .copied()
446                .filter(|c| beside_left(&tb, &blocks[*c].1))
447                .min_by(|a, b| blocks[*a].1[0].total_cmp(&blocks[*b].1[0]))
448                .unwrap_or(top)
449        });
450        remaining.retain(|r| *r != pick);
451        block_order.push(pick);
452    }
453
454    // 4. Emit each segment in block order. Its glyphs are placed here, into scratch that every segment reuses, and
455    // appended straight to the output. Blocks and their order use only segment boxes, so placement can wait until now.
456    let mut order: Vec<usize> = Vec::with_capacity(n);
457    let mut line_of: Vec<u32> = Vec::with_capacity(n);
458    let mut space_before: Vec<bool> = Vec::with_capacity(n);
459    let mut line = 0u32;
460    let mut sc = Scratch { anchor, ..Default::default() };
461    // `spaces[w]`: a word gap left of the w-th glyph of the segment, by position on the line, not by drawing order
462    // (#1062). Reused by every segment.
463    let mut spaces: Vec<bool> = Vec::new();
464    // The segment emitted last, whether it had right-to-left text, and whether its block is that one line.
465    let mut last: Option<(usize, bool, bool)> = None;
466    for b in block_order {
467        for (k, &si) in blocks[b].0.iter().enumerate() {
468            place_glyphs(&glyphs, &height, &segs[si], &mut sc);
469            let h = segs[si].h;
470            measure_cluster_gaps(&glyphs, &mut sc);
471            // The text of the cluster that glyph `i` belongs to: its base's.
472            let cluster_text = |i: usize| sc.anchor.get(i).and_then(|&a| glyphs.get(a)).map_or("", |g| g.text.as_str());
473            // Word gaps relative to this line's typical letter gap (handles tracking).
474            // Long runs of repeated punctuation belong to leaders, not letter tracking: a TOC leader
475            // must not raise the threshold above the word gaps in its title (#3086).
476            sc.gaps.clear();
477            sc.gaps.extend(sc.placed.windows(2).enumerate().filter_map(|(w, pair)| {
478                let &[p, c] = pair else { return None };
479                let gap = (*sc.cluster_gap.get(w + 1)?)?;
480                let (p, c) = (glyphs.get(p)?, glyphs.get(c)?);
481                if p.text == c.text && p.text.chars().all(|ch| !ch.is_alphanumeric() && !ch.is_whitespace()) {
482                    // Keep short tracked punctuation, including an ellipsis with a final period.
483                    // Treat five or more matching glyphs as a leader. Three neighbours on each
484                    // side suffice to classify this pair, bounding work even on long runs.
485                    let same = |&&i: &&usize| glyphs.get(i).is_some_and(|g| g.text == p.text);
486                    let before = sc.placed.get(..w).unwrap_or_default().iter().rev().take(3).take_while(same).count();
487                    let after = sc.placed.get(w.saturating_add(2)..).unwrap_or_default().iter().take(3).take_while(same).count();
488                    if before + after >= 3 {
489                        return None;
490                    }
491                }
492                Some(gap)
493            }));
494            sc.gaps.sort_by(f32::total_cmp);
495            let typical = sc.gaps.get(sc.gaps.len() / 3).copied().unwrap_or(0.0).max(0.0);
496            let threshold = (typical + h * WORD_GAP).max(h * WORD_GAP);
497            spaces.clear();
498            spaces.resize(sc.placed.len(), false);
499            for (w, pair) in sc.placed.windows(2).enumerate() {
500                let &[p, c] = pair else { continue };
501                let Some(&Some(gap)) = sc.cluster_gap.get(w + 1) else { continue };
502                let (pt, ct) = (cluster_text(p), cluster_text(c));
503                let cjk = pt.chars().any(is_cjk) && ct.chars().any(is_cjk) && gap < h * 0.5;
504                // A combining sign that rides on no glyph before it (drawn apart from its base, or in a run of its own)
505                // may sit anywhere over its base, so it never takes a space next to it. Two base letters of these
506                // scripts are measured like any others: a Thai or Indic word gap is narrower than half an em.
507                let loose_sign = |t: &str| is_mark(t) && t.chars().any(is_complex);
508                let tight_cluster = (loose_sign(pt) || loose_sign(ct)) && gap < h * 0.6;
509                let run_start = sc.breaks.binary_search(&c).is_ok();
510                if let Some(space) = spaces.get_mut(w + 1) {
511                    *space = (gap > threshold || run_start)
512                        && !cjk
513                        && !tight_cluster
514                        && !pt.trim().is_empty()
515                        && !ct.trim().is_empty()
516                        && !pt.ends_with(char::is_whitespace)
517                        && !ct.starts_with(char::is_whitespace);
518                }
519            }
520            // Right-to-left spans are reversed (visual → logical) with their gaps, and only then is a second copy made.
521            let has_rtl = sc.placed.iter().any(|&i| glyphs[i].text.chars().any(is_rtl));
522            if has_rtl {
523                sc.logical.clear();
524                sc.logical.extend_from_slice(&sc.placed);
525                reverse_rtl(&glyphs, &sc.anchor, &mut sc.logical, &mut spaces);
526            }
527            let idx: &[usize] = if has_rtl { &sc.logical } else { &sc.placed };
528            let joined = joins_line(&segs, last, si, k == 0, blocks[b].0.len() > 1, has_rtl);
529            let start = order.len();
530            order.extend_from_slice(idx);
531            // `spaces[0]` stays false: the first glyph of a segment never takes a space, unless it joins a line.
532            space_before.extend(spaces.iter().take(idx.len()));
533            space_before.resize(order.len(), false);
534            line = put_on_line(&mut line_of, &mut space_before, start..order.len(), line, &segs[si], joined);
535            last = Some((si, has_rtl, blocks[b].0.len() == 1));
536        }
537    }
538    (order.into_iter().map(|i| kept[i]).collect(), line_of, space_before)
539}

541/// Puts the reading-order glyphs `glyphs` of segment `seg` on their line: `line`, or the line
542/// before it when `seg` continues the line of `joined` (see [`joins_line`]), with a space before
543/// its first glyph when the two are a word gap apart. Returns the line the next segment starts on.
544fn put_on_line(line_of: &mut Vec<u32>, space_before: &mut [bool], glyphs: std::ops::Range<usize>, line: u32, seg: &Seg, joined: Option<&Seg>) -> u32 {
545    let Some(p) = joined else {
546        line_of.resize(glyphs.end, line);
547        return line.saturating_add(1);
548    };
549    line_of.resize(glyphs.end, line.saturating_sub(1));
550    if let Some(space) = space_before.get_mut(glyphs.start) {
551        *space = seg.bbox[0] - p.bbox[2] > WORD_GAP * p.h.min(seg.h);
552    }
553    line
554}

556/// The segment read just before (`last`, with whether it had right-to-left text) whose line segment
557/// `si` goes on, when `si` is the first line of its block (`first`) and sits on that line's baseline
558/// to its right (a label and its value, a word drawn out of order or set in another size).
559/// Same-baseline segments are always in different blocks. A column of several lines is read after
560/// the column to its left ends, so their lines don't meet. A paragraph of several lines that starts
561/// beside a line as wide as [`SAME_LINE_GAP`] heights or more is the next column, not that line's
562/// value, so it starts a line of its own; only a narrower line that is a block of its own (a label)
563/// takes it: the short last line of a column of several lines is no label (review on #4468). One-line
564/// paragraphs of two columns that share a baseline cannot be told from a table row, and are further
565/// apart than [`same_line`] allows unless the left one fills its column. Right-to-left lines are
566/// left alone: their blocks still read left first.
567fn joins_line(segs: &[Seg], last: Option<(usize, bool, bool)>, si: usize, first: bool, paragraph: bool, has_rtl: bool) -> Option<&Seg> {
568    let (p, p_rtl, p_alone) = last?;
569    if !first || p_rtl || has_rtl {
570        return None;
571    }
572    let (p, s) = (segs.get(p)?, segs.get(si)?);
573    let label = p_alone && p.bbox[2] - p.bbox[0] < SAME_LINE_GAP * p.h;
574    (same_line(p, s) && (!paragraph || label)).then_some(p)
575}
