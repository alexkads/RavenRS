7/// Largest sine of the angle between a glyph's writing direction and a flow's for the glyph to be
8/// laid out in that flow's frame. Turning an em box by θ moves its edges along the line by up to
9/// h·sin θ, so below the layout's word-gap tolerance (`WORD_GAP` · h) the boxes still read as they
10/// would unturned.

62/// Grid cell size, in points, for finding the glyphs drawn at a place.
63const CELL: f32 = 8.0;
64
65/// Most grid cells searched around one place: a glyph so tall its tolerance spans more is skipped.
66const MAX_CELLS: i64 = 4096;
67
68/// Most work (grid cells visited plus glyphs compared) the redraw dedupe in `TextDevice::finish`
69/// may do on one page. A hostile page stacking ~10^5 glyphs and ~10^5 replacements at one spot
70/// would otherwise make ~10^10 comparisons; past the budget the remaining replacements keep
71/// their unmarked redraws (the text reads twice, as it would without `/ActualText`).
72const MAX_REDRAW_WORK: usize = 1 << 22;
73
74/// Most replacement text bytes the redraw dedupe in `TextDevice::finish` copies on one page.
75pub(super) const MAX_REDRAW_TEXT: usize = 1 << 22;

83fn cell(v: f32) -> i64 {
84    // Glyph boxes are finite (`draw_glyph` and `merge` drop the rest); `as` saturates.
85    (v / CELL).floor() as i64
86}

88/// How far a redrawn glyph may sit from the one it redraws: a fifth of its height, as in `redraws`.
89fn redraw_tolerance(rect: &[f32; 4]) -> f32 {
90    0.2 * (rect[3] - rect[1]).max(0.1)

93/// One glyph carrying `text`, boxed around `drawn` in their writing direction; `None` when they
94/// span several lines or writing directions (one box would break selection and line grouping).
95fn merge(drawn: &[Drawn], text: String) -> Option<Drawn> {
96    let first = drawn.first()?;
97    let (quarter, u) = (first.0, first.1);
98    // Across the line: from the writing direction a quarter turn towards the line's bottom.
99    let n = [-u[1], u[0]];
100    let along = |p: &[f32; 2]| f64::from(p[0]) * u[0] + f64::from(p[1]) * u[1];
101    let across = |p: &[f32; 2]| f64::from(p[0]) * n[0] + f64::from(p[1]) * n[1];
102    let height = across(&first.2.quad[0]) - across(&first.2.quad[3]);
103    let (mut s0, mut s1, mut t0, mut t1) = (f64::INFINITY, f64::NEG_INFINITY, f64::INFINITY, f64::NEG_INFINITY);
104    for p in drawn.iter().flat_map(|d| d.2.quad.iter()) {
105        (s0, s1) = (s0.min(along(p)), s1.max(along(p)));
106        (t0, t1) = (t0.min(across(p)), t1.max(across(p)));
107    }
108    let one_line = drawn.iter().all(|d| compatible(u, d.1)) && t1 - t0 <= 2.0 * height.abs();
109    if !(one_line && s0.is_finite() && s1.is_finite() && t0.is_finite() && t1.is_finite()) {
110        return None;
111    }
112    // The edge towards the next line is the one the first glyph's quad starts on.
113    let (near, far) = if height >= 0.0 { (t1, t0) } else { (t0, t1) };
114    let point = |s: f64, t: f64| [(s * u[0] + t * n[0]) as f32, (s * u[1] + t * n[1]) as f32];
115    let quad = [point(s0, near), point(s1, near), point(s1, far), point(s0, far)];
116    let rect = quad.iter().fold([f32::INFINITY, f32::INFINITY, f32::NEG_INFINITY, f32::NEG_INFINITY], |r, p| {
117        [r[0].min(p[0]), r[1].min(p[1]), r[2].max(p[0]), r[3].max(p[1])]
118    });
119    Some((quarter, u, TextGlyph { text, rect, quad, direction: first.2.direction }))
120}

122/// The widest or tallest box (view-space points) a glyph, or the paint standing in for one, may
123/// have and still be placed: anything bigger is damage, not text.
124const MAX_GLYPH_EXTENT: f64 = 10_000.0;
125
126/// The quarter turn of direction `(dx, dy)` (view space, y down): 0 right, 1 down, 2 left, 3 up.
127fn quarter_turn(dx: f64, dy: f64) -> usize {
128    if dx.abs() >= dy.abs() {
129        if dx >= 0.0 { 0 } else { 2 }
130    } else if dy > 0.0 {
131        1
132    } else {
133        3
134    }
135}

137/// What an `/ActualText` replacement paints besides glyphs, measured as [`merge`] measures
138/// glyphs: along the writing direction of the first of it (its transform's x axis, `u`) and
139/// across it.
140struct Painted {
141    quarter: usize,
142    u: [f64; 2],
143    along: [f64; 2],
144    across: [f64; 2],
145}
146
147impl Painted {
148    fn new(transform: Affine) -> Self {
149        let [a, b, ..] = transform.as_coeffs();
150        let length = a.hypot(b);
151        let u = if length.is_finite() && length > 0.0 { [a / length, b / length] } else { [1.0, 0.0] };
152        let empty = [f64::INFINITY, f64::NEG_INFINITY];
153        Painted { quarter: quarter_turn(u[0], u[1]), u, along: empty, across: empty }
154    }
155
156    /// Grow the measure by the view-space box `b`.
157    fn add(&mut self, b: Rect) {
158        let n = [-self.u[1], self.u[0]];
159        for (x, y) in [(b.x0, b.y0), (b.x1, b.y0), (b.x1, b.y1), (b.x0, b.y1)] {
160            let (s, t) = (x * self.u[0] + y * self.u[1], x * n[0] + y * n[1]);
161            self.along = [self.along[0].min(s), self.along[1].max(s)];
162            self.across = [self.across[0].min(t), self.across[1].max(t)];
163        }
164    }
165
166    /// A stand-in glyph with no text of its own, boxed around what was painted and read along its
167    /// direction; `None` when that is too big to be the place of words.
168    fn glyph(self) -> Option<Drawn> {
169        let ([s0, s1], [t0, t1]) = (self.along, self.across);
170        // Too big to be the place of words, or NaN.
171        if !(s1 - s0 <= MAX_GLYPH_EXTENT && t1 - t0 <= MAX_GLYPH_EXTENT) {
172            return None;
173        }
174        let (u, n) = (self.u, [-self.u[1], self.u[0]]);
175        let point = |s: f64, t: f64| [(s * u[0] + t * n[0]) as f32, (s * u[1] + t * n[1]) as f32];
176        // The edge towards the next line starts the quad, as a glyph's em box does.
177        let quad = [point(s0, t1), point(s1, t1), point(s1, t0), point(s0, t0)];
178        let rect = quad.iter().fold([f32::INFINITY, f32::INFINITY, f32::NEG_INFINITY, f32::NEG_INFINITY], |r, p| {
179            [r[0].min(p[0]), r[1].min(p[1]), r[2].max(p[0]), r[3].max(p[1])]
180        });
181        rect.iter()
182            .all(|v| v.is_finite())
183            .then(|| (self.quarter, u, TextGlyph { text: String::new(), rect, quad, direction: [u[0] as f32, u[1] as f32] }))
184    }
185}

187/// The part of `b` inside `c`, or `None` when none of it is. A NaN edge of `c` (a damaged clip)
188/// leaves that edge uncut.
189fn cut(b: Rect, c: Rect) -> Option<Rect> {
190    let (x0, y0, x1, y1) = (b.x0.max(c.x0), b.y0.max(c.y0), b.x1.min(c.x1), b.y1.min(c.y1));
191    (x0 <= x1 && y0 <= y1).then(|| Rect::new(x0, y0, x1, y1))
192}

194/// The view-space box of `path` drawn with `transform`, from its control points (as redaction
195/// measures a path), or `None` when it has no finite point.
196fn control_box(path: &BezPath, transform: Affine) -> Option<Rect> {
197    let mut b: Option<Rect> = None;
198    let mut add = |p: kurbo::Point| {
199        let q = transform * p;
200        if q.is_finite() {
201            b = Some(b.map_or_else(|| Rect::from_points(q, q), |r| r.union_pt(q)));
202        }
203    };
204    for el in path.elements() {
205        match *el {
206            kurbo::PathEl::MoveTo(p) | kurbo::PathEl::LineTo(p) => add(p),
207            kurbo::PathEl::QuadTo(p1, p2) => [p1, p2].into_iter().for_each(&mut add),
208            kurbo::PathEl::CurveTo(p1, p2, p3) => [p1, p2, p3].into_iter().for_each(&mut add),
209            kurbo::PathEl::ClosePath => {}
210        }
211    }
212    b
213}

215impl TextDevice {
216    pub(super) fn new() -> Self {
217        TextDevice {
218            glyphs: Vec::new(),
219            depth: 0,
220            actual: None,
221            painted: None,
222            paint_filtered: false,
223            paint_cropped: false,
224            clips: Vec::new(),
225            replaced: Vec::new(),
226            page: Rect::new(f64::NEG_INFINITY, f64::NEG_INFINITY, f64::INFINITY, f64::INFINITY),
227            filter: TextFilter::default(),
228            excluded_depth: None,
229            actual_filtered: false,
230            actual_cropped: false,
231            partial_replacement: false,
232        }
233    }
234
235    /// Ends the open `/ActualText` replacement (ISO 32000-2 §14.9.4): its glyphs become one glyph
236    /// carrying the replacement text, boxed around them all. Chrome (Skia) writes one around
237    /// each Thai, Indic or Arabic cluster whose shaped glyphs (a tone mark moved clear of a vowel,
238    /// a ligature) have no Unicode of their own, so without it those marks are lost and the
239    /// words can't be found.
240    ///
241    /// A replacement spanning several lines or writing directions stays per glyph (each glyph's own
242    /// Unicode). An empty one (a soft hyphen at a line break, say) stands for no text at all. One
243    /// that marks no glyph reads where it paints, as far as that can be seen: words drawn as
244    /// outlines or pictures.
245    fn end_actual_text(&mut self) {
246        let Some((_, text, mut drawn)) = self.actual.take() else { return };
247        let painted = self.painted.take();
248        let paint_filtered = std::mem::take(&mut self.paint_filtered);
249        let paint_cropped = std::mem::take(&mut self.paint_cropped);
250        let cropped = std::mem::take(&mut self.actual_cropped);
251        if std::mem::take(&mut self.actual_filtered) {
252            // ActualText has no mapping from replacement words to individual glyphs. A mixed
253            // span cannot be filtered honestly; a wholly excluded span can simply be omitted.
254            self.partial_replacement |= !drawn.is_empty();
255            return;
256        }
257        // Before the paint: a span whose glyphs the crop box took has none left, yet it marks
258        // glyphs, so its paint doesn't place it.
259        if cropped {
260            self.glyphs.extend(drawn.into_iter().filter(|d| !d.2.text.chars().all(|c| c.is_control())));
261            return;
262        }
263        if drawn.is_empty()
264            && !self.filter.painted
265            && let Some(stand_in) = painted.and_then(Painted::glyph)
266        {
267            // The paint places it, so paint left out mixes it as excluded glyphs would.
268            if paint_filtered {
269                self.partial_replacement = true;
270                return;
271            }
272            // Its words may be among what the crop box hides, as with a glyph cropped away.
273            if paint_cropped {
274                return;
275            }
276            drawn.push(stand_in);
277        }
278        let Some(glyph) = merge(&drawn, text.clone()) else {
279            self.glyphs.extend(drawn.into_iter().filter(|d| !d.2.text.chars().all(char::is_control)));
280            return;
281        };
282        let originals = drawn.iter().filter(|d| !d.2.text.chars().all(char::is_control)).map(|d| (d.2.text.clone(), d.2.rect)).collect();
283        let at = if text.chars().all(char::is_control) {
284            None
285        } else {
286            self.glyphs.push(glyph);
287            Some(self.glyphs.len() - 1)
288        };
289        self.replaced.push(Replaced { at, text, originals });
290    }
291
292    /// Record view-space box `b`, drawn with `transform`, as painted by the open `/ActualText`
293    /// replacement, cut to the visible area (the crop box, and the clip in force); paint wholly
294    /// outside the crop box, or that an exclusion leaves out (`excluded`), only marks the
295    /// replacement.
296    fn paint(&mut self, b: Rect, transform: Affine, excluded: bool) {
297        // Wholly outside the crop box, as a glyph dropped there is: the replacement can't be read.
298        let Some(on_page) = cut(b, self.page) else {
299            self.paint_cropped = true;
300            return;
301        };
302        // Clipped away entirely: nothing of it shows.
303        let Some(seen) = self.clips.last().map_or(Some(on_page), |c| cut(on_page, *c)) else { return };
304        if excluded {
305            self.paint_filtered = true;
306        } else {
307            self.painted.get_or_insert_with(|| Painted::new(transform)).add(seen);
308        }
309    }
310
311    /// Ends the page. Some producers draw text twice, the copy for an outline, shadow or
312    /// glow, and mark only one drawing with `/ActualText` (Canva). The unmarked redraw of a
313    /// replacement's glyphs (the same text at the same places, in content order) takes the same
314    /// replacement, so the two read alike and line layout drops the copy as a redraw.
315    pub(super) fn finish(&mut self) {
316        self.end_actual_text();
317        if self.replaced.is_empty() {
318            return;
319        }
320        let mut taken = vec![false; self.glyphs.len()];
321        for at in self.replaced.iter().filter_map(|r| r.at) {
322            if let Some(t) = taken.get_mut(at) {
323                *t = true;
324            }
325        }
326        let mut grid: HashMap<(i64, i64), Vec<usize>> = HashMap::new();
327        for (i, g) in self.glyphs.iter().enumerate() {
328            if !taken.get(i).copied().unwrap_or(true) {
329                grid.entry((cell(g.2.rect[0]), cell(g.2.rect[1]))).or_default().push(i);
330            }
331        }
332        // Redraws found: the glyph range and the replacement standing for it.
333        let mut redraws: Vec<(std::ops::Range<usize>, usize)> = Vec::new();
334        let mut budget = MAX_REDRAW_WORK;
335        'replacements: for (k, r) in self.replaced.iter().enumerate() {
336            let Some((_, rect)) = r.originals.first() else { continue };
337            let tol = redraw_tolerance(rect);
338            let (x0, x1, y0, y1) = (cell(rect[0] - tol), cell(rect[0] + tol), cell(rect[1] - tol), cell(rect[1] + tol));
339            if x1.saturating_sub(x0).saturating_add(1).saturating_mul(y1.saturating_sub(y0).saturating_add(1)) > MAX_CELLS {
340                continue;
341            }
342            // Checking one candidate walks its run twice (free, then same).
343            let candidate_cost = r.originals.len().saturating_mul(2).saturating_add(1);
344            for key in (x0..=x1).flat_map(|x| (y0..=y1).map(move |y| (x, y))) {
345                let Some(rest) = budget.checked_sub(1) else { break 'replacements };
346                budget = rest;
347                for &i in grid.get(&key).map_or(&[][..], Vec::as_slice) {
348                    let Some(rest) = budget.checked_sub(candidate_cost) else { break 'replacements };
349                    budget = rest;
350                    let Some(range) = i.checked_add(r.originals.len()).map(|end| i..end) else { continue };
351                    let Some(run) = self.glyphs.get(range.clone()) else { continue };
352                    let free = taken.get(range.clone()).is_some_and(|t| t.iter().all(|t| !t));
353                    let same = run.iter().zip(&r.originals).all(|(g, (text, o))| {
354                        let tol = redraw_tolerance(o);
355                        g.2.text == *text && (g.2.rect[0] - o[0]).abs() < tol && (g.2.rect[1] - o[1]).abs() < tol
356                    });
357                    if free && same {
358                        if let Some(t) = taken.get_mut(range.clone()) {
359                            t.fill(true);
360                        }
361                        redraws.push((range, k));
362                    }
363                }
364            }
365        }
366        if redraws.is_empty() {
367            return;
368        }
369        redraws.sort_by_key(|(range, _)| range.start);
370        let glyphs = std::mem::take(&mut self.glyphs);
371        let mut redraws = redraws.into_iter().peekable();
372        let mut text_left = MAX_REDRAW_TEXT;
373        let mut i = 0;
374        while let Some(g) = glyphs.get(i) {
375            let Some((range, k)) = redraws.next_if(|(range, _)| range.start == i) else {
376                self.glyphs.push(g.clone());
377                i += 1;
378                continue;
379            };
380            let run = glyphs.get(range.clone()).unwrap_or_default();
381            let replacement = self.replaced.get(k).map_or("", |r| r.text.as_str());
382            // Each redraw takes its own copy of the replacement, and one long replacement can
383            // have many redraws: past the budget a redraw keeps its glyphs, as past the work budget.
384            let Some(left) = text_left.checked_sub(replacement.len()) else {
385                self.glyphs.extend_from_slice(run);
386                i = range.end.max(i + 1);
387                continue;
388            };
389            text_left = left;
390            let text = replacement.to_owned();
391            #[cfg(test)]
392            REDRAW_TEXT_BYTES.with(|c| c.set(c.get() + text.len()));
393            if !text.chars().all(char::is_control) {
394                match merge(run, text) {
395                    Some(glyph) => self.glyphs.push(glyph),
396                    None => self.glyphs.extend_from_slice(run),
397                }
398            }
399            i = range.end.max(i + 1);
400        }
401    }
402}

404/// `text` with each Latin compatibility ligature (U+FB00–U+FB06: ff, fi, fl, ffi, ffl, long s t,
405/// st) written as the letters it stands for, their compatibility decomposition. A font whose
406/// encoding names a ligature glyph (`/fi`) reads as the ligature character, and copied or searched
407/// text wants the letters, as a ToUnicode map gives them (ISO 32000-2 §9.10.3 maps the ff, fi and
408/// ffl glyphs to "LATIN SMALL LETTER F … I"). Nothing else is normalised. The glyph keeps its one
409/// box: the PDF gives no position for the letters inside it.
410pub(super) fn ligature_letters(text: String) -> String {
411    let letters = |c: char| match c {
412        '\u{FB00}' => Some("ff"),
413        '\u{FB01}' => Some("fi"),
414        '\u{FB02}' => Some("fl"),
415        '\u{FB03}' => Some("ffi"),
416        '\u{FB04}' => Some("ffl"),
417        '\u{FB05}' | '\u{FB06}' => Some("st"),
418        _ => None,
419    };
420    if !text.chars().any(|c| letters(c).is_some()) {
421        return text;
422    }
423    let mut out = String::with_capacity(text.len());
424    for c in text.chars() {
425        match letters(c) {
426            Some(s) => out.push_str(s),
427            None => out.push(c),
428        }
429    }
430    out
431}

440impl<'a> Device<'a> for TextDevice {
441    fn set_soft_mask(&mut self, _: Option<SoftMask<'a>>) {}
442    fn set_blend_mode(&mut self, _: BlendMode) {}
443    fn draw_path(&mut self, path: &BezPath, transform: Affine, _: &Paint<'a>, _: &PathDrawMode) {
444        // A path is no text, but it places an `/ActualText` replacement that marks no glyph.
445        if self.actual.is_some()
446            && let Some(b) = control_box(path, transform)
447        {
448            self.paint(b, transform, self.excluded_depth.is_some());
449        }
450    }
451    fn push_clip_path(&mut self, clip: &ClipPath) {
452        // Clip paths arrive in view space. NaN edges (a damaged clip) leave that edge unclipped.
453        let b = clip.path.bounding_box();
454        let visible = match self.clips.last() {
455            Some(c) => Rect::new(c.x0.max(b.x0), c.y0.max(b.y0), c.x1.min(b.x1), c.y1.min(b.y1)),
456            None => b,
457        };
458        self.clips.push(visible);
459    }
460    fn push_transparency_group(&mut self, _: f32, _: Option<SoftMask<'a>>, _: BlendMode) {}
461    fn draw_glyph(&mut self, glyph: &Glyph<'a>, transform: Affine, glyph_transform: Affine, _: &Paint<'a>, mode: &GlyphDrawMode) {
462        if self.excluded_depth.is_some() || (self.filter.images && matches!(mode, GlyphDrawMode::Invisible)) {
463            self.actual_filtered |= self.actual.is_some();
464            return;
465        }
466        // Inside an `/ActualText` replacement only the glyph's box matters.
467        let replaced = self.actual.is_some();
468        let text = match glyph.as_unicode() {
469            Some(crate::third_party::hayro::hayro_interpret::hayro_cmap::BfString::Char(c)) => c.to_string(),
470            Some(crate::third_party::hayro::hayro_interpret::hayro_cmap::BfString::String(s)) => s,
471            None if replaced => String::new(),
472            None => {
473                if text_debug() {
474                    eprintln!(
475                        "NOUNICODE {} {:?}",
476                        if matches!(glyph, Glyph::Type3(_)) { "t3" } else { "ol" },
477                        (transform * glyph_transform).as_coeffs()
478                    );
479                }
480                return;
481            }
482        };
483        if !replaced && text.chars().all(char::is_control) {
484            return;
485        }
486        let text = ligature_letters(text);
487        // Glyph space → view space. Glyph space uses 1000 units per em for outline glyphs; the em
488        // box spans descender (−200) to ascender (800), the advance gives the width.
489        let t = transform * glyph_transform;
490        if text_debug() {
491            let kind = if matches!(glyph, Glyph::Type3(_)) { "t3" } else { "ol" };
492            eprintln!("{kind} {text:?} transform={:?} glyph_transform={:?}", transform.as_coeffs(), glyph_transform.as_coeffs());
493        }
494        let advance = match glyph {
495            // A vertical font's advance runs down the column; across it, the glyph is as wide as
496            // its horizontal width w0, centred on the column by default (ISO 32000-2 §9.7.4.3). Taking
497            // the outline's width instead gave a column of Latin letters boxes of uneven widths,
498            // off its centre, and a narrow letter cut the line's gap allowance below the spacing
499            // of widely tracked text, so the column fell apart into lines.
500            Glyph::Outline(g) => g
501                .vertical_glyph_width()
502                .or_else(|| g.advance_width())
503                .filter(|a| a.is_finite() && *a > 0.0)
504                .map(f64::from)
505                .unwrap_or_else(|| g.outline().bounding_box().width().max(500.0)),
506            Glyph::Type3(g) => g.advance_width().filter(|a| a.is_finite() && *a > 0.0).map(f64::from).unwrap_or(600.0),
507        };
508        // The baseline direction in view space (y down).
509        let coefficients = t.as_coeffs();
510        let (dx, dy) = if matches!(glyph, Glyph::Outline(g) if g.is_vertical()) {
511            // PDF vertical fonts advance along negative glyph-space y (WMode 1), while
512            // keeping the glyph outline upright. Its x axis is not the writing direction.
513            (-coefficients[2], -coefficients[3])
514        } else {
515            (coefficients[0], coefficients[1])
516        };
517        let dir = quarter_turn(dx, dy);
518        let em = Rect::new(0.0, -200.0, advance, 800.0);
519        let corners = if matches!(glyph, Glyph::Outline(g) if g.is_vertical()) {
520            [(em.x0, em.y1), (em.x0, em.y0), (em.x1, em.y0), (em.x1, em.y1)]
521        } else {
522            [(em.x0, em.y0), (em.x1, em.y0), (em.x1, em.y1), (em.x0, em.y1)]
523        }
524        .map(|(x, y)| t * kurbo::Point::new(x, y));
525        let b = corners.iter().fold(Rect::new(f64::INFINITY, f64::INFINITY, f64::NEG_INFINITY, f64::NEG_INFINITY), |b, p| {
526            Rect::new(b.x0.min(p.x), b.y0.min(p.y), b.x1.max(p.x), b.y1.max(p.y))
527        });
528        let outside_crop = b.x0.max(self.page.x0) >= b.x1.min(self.page.x1) || b.y0.max(self.page.y0) >= b.y1.min(self.page.y1);
529        if !(b.x0.is_finite() && b.y0.is_finite() && b.x1.is_finite() && b.y1.is_finite())
530            || b.width() > MAX_GLYPH_EXTENT
531            || b.height() > MAX_GLYPH_EXTENT
532            || outside_crop
533        {
534            self.actual_cropped |= replaced && outside_crop;
535            return;
536        }
537        let length = dx.hypot(dy);
538        let direction = if length.is_finite() && length > 0.0 { [dx / length, dy / length] } else { [1.0, 0.0] };
539        // A glyph whose Unicode is several characters (a ligature such as "ffi") stays one glyph
540        // with its own box: the PDF gives no position for the characters inside it.
541        let glyph = TextGlyph {
542            text,
543            rect: [b.x0 as f32, b.y0 as f32, b.x1 as f32, b.y1 as f32],
544            quad: corners.map(|p| [p.x as f32, p.y as f32]),
545            direction: direction.map(|d| d as f32),
546        };
547        match &mut self.actual {
548            Some((_, _, drawn)) => drawn.push((dir, direction, glyph)),
549            None => self.glyphs.push((dir, direction, glyph)),
550        }
551    }
552    fn draw_image(&mut self, image: Image<'a, '_>, transform: Affine) {
553        // A picture of words, like an outline, places a replacement that marks no glyph; with
554        // images excluded, it is not counted.
555        if self.actual.is_none() {
556            return;
557        }
558        // The transform places the picture's pixels, not a unit square.
559        let (w, h) = (f64::from(image.width()), f64::from(image.height()));
560        let corners = [(0.0, 0.0), (w, 0.0), (w, h), (0.0, h)].map(|(x, y)| transform * kurbo::Point::new(x, y));
561        let [a, b, c, d] = corners;
562        if corners.iter().all(|p| p.is_finite()) {
563            self.paint(Rect::from_points(a, c).union_pt(b).union_pt(d), transform, self.excluded_depth.is_some() || self.filter.images);
564        }
565    }
566    fn pop_clip_path(&mut self) {
567        self.clips.pop();
568    }
569    fn pop_transparency_group(&mut self) {}
570    fn begin_marked_content(&mut self, tag: &[u8], mcid: Option<i32>) {
571        self.depth = self.depth.saturating_add(1);
572        if self.excluded_depth.is_none() && self.filter.excludes(tag, mcid) {
573            self.excluded_depth = Some(self.depth);
574        }
575    }
576    fn actual_text(&mut self, text: &[u8]) {
577        // A replacement nested in another is already covered by the outer one.
578        if self.actual.is_none() && self.excluded_depth.is_none() {
579            self.actual = Some((self.depth, crate::cos::PdfString::literal(text).to_text(), Vec::new()));
580        }
581    }
582    fn end_marked_content(&mut self) {
583        if self.actual.as_ref().is_some_and(|(depth, _, _)| *depth == self.depth) {
584            self.end_actual_text();
585        }
586        if self.excluded_depth == Some(self.depth) {
587            self.excluded_depth = None;
588        }
589        self.depth = self.depth.saturating_sub(1);
590    }
591    fn begin_annotations(&mut self) {
592        // A replacement the page content never closed ends with it: the annotations drawn next
593        // are not part of it.
594        self.end_actual_text();
595    }
596}

598/// Extract the text of one page. Panics inside the interpreter are the caller's to catch.
599pub(crate) fn extract_page(pdf: &Pdf, page: usize, settings: &InterpreterSettings) -> Option<PageText> {
600    extract_page_filtered(pdf, page, settings, TextFilter::default()).ok()
601}
602
603pub(crate) fn extract_page_filtered(pdf: &Pdf, page: usize, settings: &InterpreterSettings, filter: TextFilter) -> Result<PageText, String> {
604    let pages = pdf.pages();
605    let p = pages.get(page).ok_or("the page could not be read")?;
606    let (w, h) = p.render_dimensions();
607    let cache = InterpreterCache::new();
608    // Annotation appearances are included: form-field values and comment text are searchable,
609    // as in Acrobat, and match what other extractors report.
610    // The interpreter says when the content was cut at its limits (#623): the text is then
611    // marked incomplete, and the caller's own sink still hears it.
612    let cut = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
613    let settings = {
614        let flag = cut.clone();
615        let inner = settings.warning_sink.clone();
616        InterpreterSettings {
617            warning_sink: std::sync::Arc::new(move |warning| {
618                if matches!(warning, InterpreterWarning::ContentTruncated) {
619                    flag.store(true, std::sync::atomic::Ordering::Relaxed);
620                }
621                inner(warning);
622            }),
623            ..settings.clone()
624        }
625    };
626    let initial = p.initial_transform(true).to_kurbo();
627    let c = p.intersected_crop_box();
628    let mut ctx = Context::new(initial, Rect::new(0.0, 0.0, f64::from(w), f64::from(h)), &cache, p.xref(), settings);
629    let mut dev = TextDevice::new();
630    dev.page = initial.transform_rect_bbox(Rect::new(c.x0, c.y0, c.x1, c.y1));
631    dev.filter = filter;
632    interpret_page(p, &mut ctx, &mut dev);
633    // A replacement the content never closed ends with the page.
634    dev.finish();
635    if dev.partial_replacement {
636        return Err("a replacement text span mixes included and excluded content; disable the exclusions to count this page".into());
637    }
638    let glyphs = dev.glyphs;
639    // Text on a curved path: glyphs that each continue the previous one's line while the
640    // direction turns. Each such run is one line.
641    let turned = |u: [f64; 2]| u[0].abs().min(u[1].abs()) > DIRECTION_TOLERANCE;
642    let mut in_curve = vec![false; glyphs.len()];
643    // Turned text: curved runs, and straight flows along one writing direction. The two are kept
644    // apart so finding a glyph's direction scans only the directions (a few dozen at most: each
645    // differs from the others by more than the tolerance), not every curved run.
646    let mut curves: Vec<Vec<usize>> = Vec::new();
647    let mut straight: Vec<([f64; 2], Vec<usize>)> = Vec::new();
648    let mut start = 0;
649    for end in 1..=glyphs.len() {
650        if end < glyphs.len() && continues(&glyphs[end - 1], &glyphs[end]) {
651            continue;
652        }
653        let run = &glyphs[start..end];
654        if run.iter().any(|g| turned(g.1)) && run.iter().any(|g| !compatible(run[0].1, g.1)) {
655            in_curve[start..end].fill(true);
656            curves.push((start..end).collect());
657        }
658        start = end;
659    }
660    let mut quarters: [Vec<usize>; 4] = Default::default();
661    for (i, (quarter, u, _)) in glyphs.iter().enumerate() {
662        if in_curve[i] {
663            continue;
664        }
665        if !turned(*u) {
666            if let Some(q) = quarters.get_mut(*quarter) {
667                q.push(i);
668            }
669        } else if let Some((_, ids)) = straight.iter_mut().find(|(d, _)| compatible(*d, *u)) {
670            ids.push(i);
671        } else {
672            straight.push((*u, vec![i]));
673        }
674    }
675    // A horizontal heading must not force a vertical body back into horizontal line grouping.
676    // Group each writing direction independently, then join flows from their topmost position.
677    let upright = |quarters: [Vec<usize>; 4]| {
678        let mut flows: Vec<Flow> = quarters
679            .into_iter()
680            .enumerate()
681            .filter(|(_, ids)| !ids.is_empty())
682            .map(|(quarter, ids)| {
683                let rects = ids.iter().map(|&i| to_upright(glyphs[i].2.rect, quarter as u8, w, h)).collect();
684                arrange_flow(&glyphs, ids, rects, Some(quarter as u8))
685            })
686            .collect();
687        flows.sort_by(|a, b| a.top.total_cmp(&b.top));
688        flows
689    };
690    let mut flows = upright(quarters);
691    let (main, mut inline, place) = turned_lines_in_place(&glyphs, &mut flows);
692    let mut extra: Vec<Flow> = curves
693        .into_iter()
694        .map(|ids| (None, ids))
695        .chain(straight.into_iter().map(|(u, ids)| (Some(u), ids)))
696        .map(|(u, ids)| {
697            let rects = match u {
698                // Turned text is laid out along its own baseline.
699                Some(u) => ids.iter().map(|&i| along_baseline(&glyphs[i].2.quad, u).map(|c| c as f32)).collect(),
700                None => unrolled(ids.iter().map(|&i| &glyphs[i])),
701            };
702            arrange_flow(&glyphs, ids, rects, None)
703        })
704        .collect();
705    // Turned text takes the place in the reading order where laying it out among the nearest
706    // quarter turn's text puts its first glyph.
707    let mut rank = Vec::new();
708    if !extra.is_empty() {
709        rank = vec![usize::MAX; glyphs.len()];
710        let mut nearest: [Vec<usize>; 4] = Default::default();
711        for (i, g) in glyphs.iter().enumerate() {
712            if let Some(q) = nearest.get_mut(g.0) {
713                q.push(i);
714            }
715        }
716        for (r, i) in upright(nearest).into_iter().flat_map(|f| f.ids).enumerate() {
717            rank[i] = r;
718        }
719    }
720    let first = |ids: &[usize]| ids.iter().filter_map(|&i| rank.get(i).copied()).min().unwrap_or(usize::MAX);
721    extra.sort_by_key(|f| first(&f.ids));
722    let mut extra = extra.into_iter().peekable();
723    let mut slots: Vec<Option<TextGlyph>> = glyphs.into_iter().map(|g| Some(g.2)).collect();
724    let mut text = PageText::default();
725    let mut emit = |text: &mut PageText, ids: &[usize], line_of: &[u32], space_before: &[bool], quarter: Option<u8>| {
726        let offset = text.line_of.last().map_or(0, |line| line.saturating_add(1));
727        for ((&i, &line), &space) in ids.iter().zip(line_of).zip(space_before) {
728            let Some(mut g) = slots.get_mut(i).and_then(Option::take) else { continue };
729            if let Some(quarter) = quarter {
730                g.rect = from_upright(to_upright(g.rect, quarter, w, h), quarter, w, h);
731            }
732            text.glyphs.push(g);
733            text.line_of.push(line.saturating_add(offset));
734            text.space_before.push(space);
735        }
736    };
737    let placed = |ids: &[usize]| ids.iter().filter_map(|&i| place.get(i).copied()).min().unwrap_or(usize::MAX);
738    for (fi, flow) in flows.into_iter().enumerate() {
739        let mut k = 0;
740        while k < flow.ids.len() {
741            let line = flow.line_of[k];
742            let end = flow.line_of[k..].iter().position(|l| *l != line).map_or(flow.ids.len(), |n| k + n);
743            let before = first(&flow.ids[k..end]);
744            while let Some(f) = extra.next_if(|f| first(&f.ids) < before) {
745                emit(&mut text, &f.ids, &f.line_of, &f.space_before, None);
746            }
747            if Some(fi) == main {
748                let at = placed(&flow.ids[k..end]);
749                while let Some(l) = inline.next_if(|l| l.at < at) {
750                    emit(&mut text, &l.ids, &vec![0; l.ids.len()], &l.space_before, l.quarter);
751                }
752            }
753            emit(&mut text, &flow.ids[k..end], &vec![0; end - k], &flow.space_before[k..end], flow.quarter);
754            k = end;
755        }
756        if Some(fi) == main {
757            for l in inline.by_ref() {
758                emit(&mut text, &l.ids, &vec![0; l.ids.len()], &l.space_before, l.quarter);
759            }
760        }
761    }
762    for f in extra {
763        emit(&mut text, &f.ids, &f.line_of, &f.space_before, None);
764    }
765    text.truncated = cut.load(std::sync::atomic::Ordering::Relaxed);
766    Ok(text)
767}

769/// A line turned a quarter from the page's main text, read in place among it.
770struct InPlace {
771    /// Where laying it out with the main text puts its first glyph.
772    at: usize,
773    ids: Vec<usize>,
774    space_before: Vec<bool>,
775    /// The quarter turn of the flow it came from.
776    quarter: Option<u8>,
777}
778
779/// Takes out of `flows` the lines that are turned a quarter from the page's main text (the unturned flow, when it
780/// holds more glyphs than any other) and sit inside that text's area: a label set upwards beside a row, a turned
781/// table heading. Each is placed where laying it out among the main text, by its box on the page, puts its first
782/// glyph, rather than after all the main text. Lines outside the area (a stamp in the margin) and every other flow
783/// keep their place, as does a page whose main text is vertical.
784///
785/// Returns the main flow's index, the lines in place order, and each glyph's place in that layout.
786fn turned_lines_in_place(glyphs: &[Drawn], flows: &mut [Flow]) -> (Option<usize>, std::iter::Peekable<std::vec::IntoIter<InPlace>>, Vec<usize>) {
787    let nothing = || (None, Vec::new().into_iter().peekable(), Vec::new());
788    let Some((main, size)) = flows.iter().enumerate().find(|(_, f)| f.quarter == Some(0)).map(|(i, f)| (i, f.ids.len())) else { return nothing() };
789    if flows.iter().enumerate().any(|(i, f)| i != main && f.ids.len() >= size) {
790        return nothing();
791    }
792    let rect = |i: usize| glyphs.get(i).map_or([0.0; 4], |g| g.2.rect);
793    let bounds = |ids: &[usize]| {
794        let mut b = [f32::INFINITY, f32::INFINITY, f32::NEG_INFINITY, f32::NEG_INFINITY];
795        for &i in ids {
796            union(&mut b, &rect(i));
797        }
798        b
799    };
800    let Some(area) = flows.get(main).map(|f| bounds(&f.ids)) else { return nothing() };
801    let mut lines = Vec::new();
802    for (fi, flow) in flows.iter_mut().enumerate() {
803        // Upside-down text (a half turn) is not turned a quarter, and keeps its own flow.
804        if fi == main || !matches!(flow.quarter, Some(1 | 3)) {
805            continue;
806        }
807        let (mut ids, mut line_of, mut space_before) = (Vec::new(), Vec::new(), Vec::new());
808        let mut k = 0;
809        while let Some(&line) = flow.line_of.get(k) {
810            let end = flow.line_of.get(k..).and_then(|rest| rest.iter().position(|l| *l != line)).map_or(flow.line_of.len(), |n| k + n);
811            let (Some(line_ids), Some(line_spaces)) = (flow.ids.get(k..end), flow.space_before.get(k..end)) else { break };
812            let b = bounds(line_ids);
813            let (cx, cy) = ((b[0] + b[2]) / 2.0, (b[1] + b[3]) / 2.0);
814            if (area[0]..=area[2]).contains(&cx) && (area[1]..=area[3]).contains(&cy) {
815                lines.push(InPlace { at: usize::MAX, ids: line_ids.to_vec(), space_before: line_spaces.to_vec(), quarter: flow.quarter });
816            } else {
817                ids.extend_from_slice(line_ids);
818                line_of.extend(std::iter::repeat_n(line, line_ids.len()));
819                space_before.extend_from_slice(line_spaces);
820            }
821            k = end;
822        }
823        (flow.ids, flow.line_of, flow.space_before) = (ids, line_of, space_before);
824    }
825    if lines.is_empty() {
826        return nothing();
827    }
828    let ids: Vec<usize> =
829        flows.get(main).map_or(&[][..], |f| &f.ids[..]).iter().copied().chain(lines.iter().flat_map(|l| l.ids.iter().copied())).collect();
830    let rects = ids.iter().map(|&i| rect(i)).collect();
831    let mut place = vec![usize::MAX; glyphs.len()];
832    for (r, i) in arrange_flow(glyphs, ids, rects, Some(0)).ids.into_iter().enumerate() {
833        if let Some(p) = place.get_mut(i) {
834            *p = r;
835        }
836    }
837    for l in &mut lines {
838        l.at = l.ids.iter().filter_map(|&i| place.get(i).copied()).min().unwrap_or(usize::MAX);
839    }
840    lines.sort_by_key(|l| l.at);
841    (Some(main), lines.into_iter().peekable(), place)
842}
