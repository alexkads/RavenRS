7/// Attributes a page inherits from its ancestors (ISO 32000-2 §7.7.3.4).
8pub const INHERITABLE: [&[u8]; 4] = [b"Resources", b"MediaBox", b"CropBox", b"Rotate"];

10/// The largest `/UserUnit` honoured, as in Acrobat (a larger value counts as this one).
11pub const MAX_USER_UNIT: f64 = 75_000.0;

13/// A leaf page and its effective (inherited) attributes.
14#[derive(Clone, Debug, PartialEq)]
15pub struct Page {
16    pub obj: ObjRef,
17    /// The page dictionary with inherited attributes filled in.
18    pub dict: Dict,
19}

21/// The side, in points, at or below which a `/MediaBox` has no area and the page falls back to
22/// US Letter (the renderer's `is_nearly_zero`, 1/4096).
23const MIN_BOX_SIDE: f64 = 1.0 / 4096.0;
24
25fn rect(doc: &Document, o: Option<&Object>) -> Option<[f64; 4]> {
26    let o = doc.resolve(o?);
27    // Four entries, or not a rectangle: a longer array (one shared by many pages, say) is not
28    // resolved entry by entry only to be discarded.
29    let [a, b, c, d] = o.as_array()?.as_slice() else { return None };
30    let v = [a, b, c, d].map(|x| doc.resolve(x).as_f64());
31    let [Some(x0), Some(y0), Some(x1), Some(y1)] = v else { return None };
32    [x0, y0, x1, y1].iter().all(|x| x.is_finite()).then(|| [x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)])
33}

35impl Page {
36    /// The visible region (crop box clipped to the media box), in user space. A page with no
37    /// `/MediaBox`, or one without area (a side of at most 1/4096 pt, as the renderer measures
38    /// it), is US Letter, as the renderer and inspector size it.
39    pub fn crop(&self, doc: &Document) -> [f64; 4] {
40        let media = rect(doc, self.dict.get(b"MediaBox"))
41            .filter(|m| m[2] - m[0] > MIN_BOX_SIDE && m[3] - m[1] > MIN_BOX_SIDE)
42            .unwrap_or([0.0, 0.0, 612.0, 792.0]);
43        match rect(doc, self.dict.get(b"CropBox")) {
44            Some(c) => {
45                let r = [c[0].max(media[0]), c[1].max(media[1]), c[2].min(media[2]), c[3].min(media[3])];
46                if r[2] > r[0] && r[3] > r[1] { r } else { media }
47            }
48            None => media,
49        }
50    }
51
52    /// Clockwise rotation when displayed: 0, 90, 180 or 270. A `/Rotate` that isn't a multiple
53    /// of 90 (ISO 32000-2 §7.7.3.3 requires one) counts as 0, as the renderer draws that page.
54    pub fn rotation(&self, doc: &Document) -> i64 {
55        let r = self.dict.get(b"Rotate").and_then(|o| doc.resolve(o).as_int()).unwrap_or(0).rem_euclid(360);
56        if r % 90 == 0 { r } else { 0 }
57    }
58
59    /// The size of the page's user-space unit in points (`/UserUnit`; it isn't inherited), read
60    /// as Acrobat reads it: a number from 1 up, at most [`MAX_USER_UNIT`]; anything else is 1.
61    pub fn user_unit(&self, doc: &Document) -> f64 {
62        self.dict.get(b"UserUnit").and_then(|o| doc.resolve(o).as_f64()).filter(|u| *u >= 1.0).map_or(1.0, |u| u.min(MAX_USER_UNIT))
63    }
64
65    /// The displayed size (width, height) in points, after `/UserUnit` and `/Rotate`.
66    pub fn display_size(&self, doc: &Document) -> (f64, f64) {
67        let c = self.crop(doc);
68        let u = self.user_unit(doc);
69        let (w, h) = ((c[2] - c[0]) * u, (c[3] - c[1]) * u);
70        if self.rotation(doc) % 180 == 0 { (w, h) } else { (h, w) }
71    }
72
73    /// The matrix `[a b c d e f]` from display space (points, origin at the bottom-left of the
74    /// page as shown, y up, after `/UserUnit` and `/Rotate`) to user space.
75    pub fn view_matrix(&self, doc: &Document) -> [f64; 6] {
76        let m = view_matrix_for(self.rotation(doc), self.crop(doc));
77        let u = self.user_unit(doc);
78        if u == 1.0 { m } else { [m[0] / u, m[1] / u, m[2] / u, m[3] / u, m[4], m[5]] }
79    }
80}

82/// [`Page::view_matrix`] for a region `crop` (`[x0, y0, x1, y1]`) shown turned clockwise by
83/// `rotation` (0, 90, 180 or 270): the matrix from the turned region's own display space
84/// (origin at its bottom-left, y up) to the region's space. For a rectangle in user space it
85/// puts an upright picture into that rectangle as the page displays it.
86pub fn view_matrix_for(rotation: i64, crop: [f64; 4]) -> [f64; 6] {
87    let [x0, y0, x1, y1] = crop;
88    match rotation {
89        90 => [0.0, 1.0, -1.0, 0.0, x1, y0],
90        180 => [-1.0, 0.0, 0.0, -1.0, x1, y1],
91        270 => [0.0, -1.0, 1.0, 0.0, x0, y1],
92        _ => [1.0, 0.0, 0.0, 1.0, x0, y0],
93    }
94}

96/// Leaf pages in document order, with inherited attributes resolved. Cycles and absurdly deep
97/// trees are cut off rather than followed, as is a tree whose inherited values would take more
98/// than [`MAX_INHERITED_COPIES`] bytes to copy into its leaves.
99pub fn pages(doc: &Document) -> Vec<Page> {
100    PageTree::new(doc).collect()
101}
102
103/// Approximate bytes of inherited values copied into leaf dictionaries by one page walk.
104pub const MAX_INHERITED_COPIES: usize = 1 << 30;
105
106/// Approximate heap bytes of `o`: one [`Object`] per node plus string, name and key bytes.
107/// Values nested deeper than the parser allows weigh as too large.
108fn weight(o: &Object, depth: usize) -> usize {
109    if depth > 256 {
110        return usize::MAX;
111    }
112    let own = std::mem::size_of::<Object>();
113    match o {
114        Object::String(s) => own.saturating_add(s.bytes.len()),
115        Object::Name(n) => own.saturating_add(n.len()),
116        Object::Array(a) => a.iter().fold(own, |n, v| n.saturating_add(weight(v, depth + 1))),
117        Object::Dict(d) => d.iter().fold(own, |n, (k, v)| n.saturating_add(k.len()).saturating_add(weight(v, depth + 1))),
118        Object::Stream(s) => s.dict.iter().fold(own, |n, (k, v)| n.saturating_add(k.len()).saturating_add(weight(v, depth + 1))),
119        _ => own,
120    }
121}

123// Retain the owner of each inherited value rather than deep-cloning direct Resources on
124// every edge. Only the final Page dictionaries need owned copies of those values. Each owner
125// carries the value's weight, measured once where the value is declared.
126type Inherited = [Option<(std::sync::Arc<Object>, usize)>; 4];
127
128struct Children {
129    owner: std::sync::Arc<Object>,
130    next: usize,
131    inherited: Inherited,
132}
133
134impl Children {
135    fn kids(&self) -> Option<&Vec<Object>> {
136        match &*self.owner {
137            Object::Dict(d) => d.get(b"Kids")?.as_array(),
138            other => other.as_array(),
139        }
140    }
141}

143struct PageTree<'a> {
144    doc: &'a Document,
145    seen: std::collections::HashSet<ObjRef>,
146    // One cursor per ancestor, never one entry per child. This also preserves the first
147    // depth-first occurrence's attributes when malformed trees share a child.
148    pending: Vec<Children>,
149    // Weight left for copying inherited values into leaves. A large direct value on an
150    // ancestor of many leaves would otherwise be copied once per leaf.
151    budget: usize,
152}
153
154impl<'a> PageTree<'a> {
155    fn new(doc: &'a Document) -> Self {
156        Self::with_budget(doc, MAX_INHERITED_COPIES)
157    }
158
159    fn with_budget(doc: &'a Document, budget: usize) -> Self {
160        let top = doc.root().and_then(|r| doc.get(r).as_dict().and_then(|d| d.reference(b"Pages")));
161        let pending = top
162            .into_iter()
163            .map(|r| Children { owner: std::sync::Arc::new(Object::Array(vec![Object::Ref(r)])), next: 0, inherited: Default::default() })
164            .collect();
165        Self { doc, seen: Default::default(), pending, budget }
166    }
167}
168
169impl Iterator for PageTree<'_> {
170    type Item = Page;
171
172    fn next(&mut self) -> Option<Page> {
173        while let Some(parent) = self.pending.last_mut() {
174            let Some(child) = parent.kids().and_then(|kids| kids.get(parent.next)) else {
175                self.pending.pop();
176                continue;
177            };
178            let node = child.as_ref();
179            // A successful get means next is strictly below the allocated array length.
180            parent.next += 1;
181            let Some(node) = node else { continue };
182            if self.seen.len() >= 1_000_000 {
183                self.pending.clear();
184                return None;
185            }
186            if !self.seen.insert(node) {
187                continue;
188            }
189            let obj = self.doc.get(node);
190            let Some(d) = obj.as_dict() else { continue };
191            let mut inherited = parent.inherited.clone();
192            for (key, owner) in INHERITABLE.iter().zip(&mut inherited) {
193                if let Some(value) = d.get(key) {
194                    *owner = Some((obj.clone(), weight(value, 0)));
195                }
196            }
197            let is_pages = d.name(b"Type") == Some(b"Pages") || (d.contains(b"Kids") && d.name(b"Type") != Some(b"Page"));
198            if is_pages {
199                let owner = match d.get(b"Kids") {
200                    Some(Object::Ref(r)) => self.doc.get(*r),
201                    Some(Object::Array(_)) => obj.clone(),
202                    _ => continue,
203                };
204                self.pending.push(Children { owner, next: 0, inherited });
205            } else {
206                let mut dict = d.clone();
207                for (key, owner) in INHERITABLE.iter().zip(&inherited) {
208                    if !dict.contains(key)
209                        && let Some((owner, weight)) = owner
210                        && let Some(value) = owner.as_dict().and_then(|d| d.get(key))
211                    {
212                        // Past the budget the walk is cut off, like a tree too large to follow.
213                        let Some(left) = self.budget.checked_sub(*weight) else {
214                            self.pending.clear();
215                            return None;
216                        };
217                        self.budget = left;
218                        dict.set(key.to_vec(), value.clone());
219                    }
220                }
221                return Some(Page { obj: node, dict });
222            }
223        }
224        None
225    }
226}

248    /// A document whose objects 1, 2, … are `objs`, object 1 its catalog.
249    fn doc_from(objs: &[String]) -> Document {
250        let mut out = b"%PDF-1.7\n".to_vec();
251        let mut offs = Vec::new();
252        for (i, o) in objs.iter().enumerate() {
253            offs.push(out.len());
254            out.extend_from_slice(format!("{} 0 obj\n{o}\nendobj\n", i + 1).as_bytes());
255        }
256        let x = out.len();
257        out.extend_from_slice(format!("xref\n0 {}\n0000000000 65535 f \n", objs.len() + 1).as_bytes());
258        for o in offs {
259            out.extend_from_slice(format!("{o:010} 00000 n \n").as_bytes());
260        }
261        out.extend_from_slice(format!("trailer\n<< /Size {} /Root 1 0 R >>\nstartxref\n{x}\n%%EOF\n", objs.len() + 1).as_bytes());
262        Document::open(Arc::new(out)).unwrap()
263    }

482    /// Display space is in points: `/UserUnit 2` doubles it. As in Acrobat, the value isn't
483    /// inherited, values below 1 or not numbers count as 1, and values over 75,000 as 75,000.
484    #[test]
485    fn user_unit_scales_display_space() {
486        for rotate in [0, 90, 180, 270] {
487            let d = doc_with(rotate, "", "/UserUnit 2");
488            let p = &pages(&d)[0];
489            let (w, h) = p.display_size(&d);
490            assert_eq!((w, h), if rotate % 180 == 0 { (400.0, 600.0) } else { (600.0, 400.0) }, "{rotate}");
491            let plain = doc(rotate);
492            let unscaled = pages(&plain)[0].view_matrix(&plain);
493            for (x, y) in [(0.0, 0.0), (w, 0.0), (0.0, h), (w, h), (37.0, 81.0)] {
494                assert_eq!(apply(p.view_matrix(&d), x, y), apply(unscaled, x / 2.0, y / 2.0), "{rotate}: ({x}, {y})");
495            }
496        }
497        for (pages_attrs, page_attrs, unit) in [
498            ("/UserUnit 2", "", 1.0),
499            ("", "/UserUnit 1.5", 1.5),
500            ("", "/UserUnit 0.5", 1.0),
501            ("", "/UserUnit 0", 1.0),
502            ("", "/UserUnit -2", 1.0),
503            ("", "/UserUnit /Two", 1.0),
504            ("", "/UserUnit 100000", MAX_USER_UNIT),
505        ] {
506            let d = doc_with(0, pages_attrs, page_attrs);
507            assert_eq!(pages(&d)[0].user_unit(&d), unit, "{pages_attrs} {page_attrs}");
508        }
509    }
