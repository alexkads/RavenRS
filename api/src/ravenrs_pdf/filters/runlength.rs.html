1//! RunLengthDecode (ISO 32000-2 §7.4.5): a length byte L of 0–127 copies the next
2//! L + 1 bytes, 129–255 repeats the next byte 257 − L times, 128 is EOD.
3//! A missing EOD is tolerated; a truncated run is corrupt.
4
5use crate::filters::{Failure, Step};
6
7const NAME: &str = "RunLengthDecode";
8
9pub(crate) fn decode(data: &[u8], max: usize) -> Step {
10    let mut out = Vec::with_capacity(data.len().saturating_mul(2).min(max));
11    let mut i = 0;
12    while let Some(&l) = data.get(i) {
13        i += 1;
14        match l {
15            128 => break,
16            0..=127 => {
17                let n = usize::from(l) + 1;
18                let avail = n.min(data.len() - i);
19                if out.len() + avail > max {
20                    return Err(Failure::limit(max));
21                }
22                out.extend_from_slice(&data[i..i + avail]);
23                i += avail;
24                if avail < n {
25                    return Err(Failure::corrupt(NAME, "truncated literal run", out));
26                }
27            }
28            _ => {
29                let Some(&b) = data.get(i) else {
30                    return Err(Failure::corrupt(NAME, "truncated repeat run", out));
31                };
32                i += 1;
33                let n = 257 - usize::from(l);
34                if out.len() + n > max {
35                    return Err(Failure::limit(max));
36                }
37                out.resize(out.len() + n, b);
38            }
39        }
40    }
41    Ok(out)
42}
43
44pub(crate) fn encode(data: &[u8]) -> Vec<u8> {
45    let mut out = Vec::with_capacity(data.len() + data.len() / 128 + 2);
46    let n = data.len();
47    let mut i = 0;
48    while i < n {
49        let mut run = 1;
50        while i + run < n && run < 128 && data[i + run] == data[i] {
51            run += 1;
52        }
53        if run >= 2 {
54            out.push((257 - run) as u8);
55            out.push(data[i]);
56            i += run;
57            continue;
58        }
59        // Literal: extend until a repeat begins or 128 bytes.
60        let start = i;
61        let mut j = i + 1;
62        while j < n && j - start < 128 && !(j + 1 < n && data[j] == data[j + 1]) {
63            j += 1;
64        }
65        out.push((j - start - 1) as u8);
66        out.extend_from_slice(&data[start..j]);
67        i = j;
68    }
69    out.push(128);
70    out
71}

73#[cfg(test)]
74mod tests {
75    use super::*;
76    use crate::filters::FilterError;
77
78    fn dec(d: &[u8]) -> Result<Vec<u8>, FilterError> {
79        decode(d, 10_000).map_err(|f| f.error)
80    }
81
82    #[test]
83    fn vectors() {
84        assert_eq!(dec(&[2, b'a', b'b', b'c', 254, b'x', 128]).unwrap(), b"abcxxx");
85        assert_eq!(dec(&[129, 7, 128]).unwrap(), [7; 128]);
86        assert!(dec(&[127]).is_err());
87        assert_eq!(dec(&[0, b'q']).unwrap(), b"q"); // missing EOD
88        assert_eq!(dec(&[128, 0, b'q']).unwrap(), b""); // data after EOD ignored
89        assert_eq!(dec(&[]).unwrap(), b"");
90    }
91
92    #[test]
93    fn truncated() {
94        let f = decode(&[4, b'a', b'b'], 100).err().unwrap();
95        assert!(matches!(f.error, FilterError::Corrupt { .. }));
96        assert_eq!(f.partial, b"ab");
97        assert!(matches!(dec(&[0, b'a', 200]), Err(FilterError::Corrupt { .. })));
98    }
99
100    #[test]
101    fn limit() {
102        assert!(matches!(decode(&[129, 0, 129, 0], 200).map_err(|f| f.error), Err(FilterError::LimitExceeded(200))));
103        assert!(matches!(decode(&[3, 1, 2, 3, 4], 3).map_err(|f| f.error), Err(FilterError::LimitExceeded(3))));
104        assert_eq!(decode(&[129, 0, 129, 0], 256).ok().unwrap().len(), 256);
105    }
106
107    #[test]
108    fn encoding() {
109        assert_eq!(encode(b"abcxxx"), [2, b'a', b'b', b'c', 254, b'x', 128]);
110        assert_eq!(encode(b""), [128]);
111        assert_eq!(encode(&[5; 300]), [129, 5, 129, 5, 213, 5, 128]);
112        let lit: Vec<u8> = (0..200u8).collect();
113        let e = encode(&lit);
114        assert_eq!((e[0], e[129]), (127, 71));
115        assert_eq!(dec(&e).unwrap(), lit);
116    }
117}
