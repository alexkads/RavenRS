56#[cfg(test)]
57thread_local! {
58    /// Decisions coded on this thread.
59    pub(crate) static DECISIONS: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
60}
61
62/// A context's adaptive state: the index into [`STATES`] (low 7 bits) and the MPS sense (bit 7).
63/// 0, the initial value, is index 0 with MPS 0 (E.2.8).
64pub(crate) type Context = u8;
65
66/// The encoder's registers (E.2.1) and the bytes written so far.
67pub(crate) struct MqEncoder {
68    a: u32,
69    c: u32,
70    ct: u32,
71    /// The output, after the byte before the first one (B at INITENC), which is 0x00 and is
72    /// dropped by [`MqEncoder::finish`]: a carry can't reach it before the first byte is out.
73    out: Vec<u8>,
74}
75
76impl MqEncoder {
77    /// INITENC (Figure E.10). The byte before the first is 0x00, not 0xFF, so CT starts at 12.
78    pub(crate) fn new() -> Self {
79        MqEncoder { a: 0x8000, c: 0, ct: 12, out: vec![0] }
80    }
81
82    /// The byte B, the last one written (or the one before the first).
83    fn b(&self) -> u8 {
84        self.out.last().copied().unwrap_or(0)
85    }
86
87    /// ENCODE (Figures E.3–E.7): code decision `d` in context `cx`, updating its state.
88    pub(crate) fn encode(&mut self, cx: &mut Context, d: bool) {
89        #[cfg(test)]
90        DECISIONS.with(|n| n.set(n.get() + 1));
91        let (index, mps) = (usize::from(*cx & 0x7F), *cx >> 7 == 1);
92        let Some(&(qe, nmps, nlps, switch)) = STATES.get(index) else { return };
93        self.a -= qe;
94        if d == mps {
95            // CODEMPS.
96            if self.a & 0x8000 != 0 {
97                self.c += qe;
98                return;
99            }
100            if self.a < qe {
101                self.a = qe;
102            } else {
103                self.c += qe;
104            }
105            *cx = nmps | (*cx & 0x80);
106        } else {
107            // CODELPS.
108            if self.a < qe {
109                self.c += qe;
110            } else {
111                self.a = qe;
112            }
113            let mps = mps != switch;
114            *cx = nlps | (u8::from(mps) << 7);
115        }
116        self.renorm();
117    }
118
119    /// RENORME (Figure E.8).
120    fn renorm(&mut self) {
121        loop {
122            self.a <<= 1;
123            self.c <<= 1;
124            self.ct -= 1;
125            if self.ct == 0 {
126                self.byte_out();
127            }
128            if self.a & 0x8000 != 0 {
129                break;
130            }
131        }
132    }
133
134    /// BYTEOUT (Figure E.9), with bit stuffing after 0xFF and carry into the byte before.
135    fn byte_out(&mut self) {
136        if self.b() == 0xFF {
137            self.stuffed();
138        } else if self.c < 0x800_0000 {
139            self.plain();
140        } else {
141            let b = self.b().wrapping_add(1);
142            if let Some(last) = self.out.last_mut() {
143                *last = b;
144            }
145            if b == 0xFF {
146                self.c &= 0x7FF_FFFF;
147                self.stuffed();
148            } else {
149                self.plain();
150            }
151        }
152    }
153
154    fn plain(&mut self) {
155        self.out.push(((self.c >> 19) & 0xFF) as u8);
156        self.c &= 0x7_FFFF;
157        self.ct = 8;
158    }
159
160    fn stuffed(&mut self) {
161        self.out.push(((self.c >> 20) & 0xFF) as u8);
162        self.c &= 0xF_FFFF;
163        self.ct = 7;
164    }
165
166    /// FLUSH (Figures E.11 and E.12): the final bytes, then the 0xFF 0xAC marker.
167    pub(crate) fn finish(mut self) -> Vec<u8> {
168        let top = self.c + self.a;
169        self.c |= 0xFFFF;
170        if self.c >= top {
171            self.c -= 0x8000;
172        }
173        self.c <<= self.ct;
174        self.byte_out();
175        self.c <<= self.ct;
176        self.byte_out();
177        if self.b() != 0xFF {
178            self.out.push(0xFF);
179        }
180        self.out.push(0xAC);
181        self.out.split_off(1)
182    }
183}

185#[cfg(test)]
186mod tests {
187    use super::*;
188
189    /// Annex H.2's test sequence: 256 decisions in one context give these 30 bytes.
190    #[test]
191    fn the_annex_h_test_sequence_encodes_to_its_bytes() {
192        let decisions = [
193            0x00, 0x02, 0x00, 0x51, 0x00, 0x00, 0x00, 0xC0, 0x03, 0x52, 0x87, 0x2A, 0xAA, 0xAA, 0xAA, 0xAA, 0x82, 0xC0, 0x20, 0x00, 0xFC, 0xD7, 0x9E,
194            0xF6, 0xBF, 0x7F, 0xED, 0x90, 0x4F, 0x46, 0xA3, 0xBF,
195        ];
196        let mut mq = MqEncoder::new();
197        let mut cx = 0;
198        for byte in decisions {
199            for bit in (0..8).rev() {
200                mq.encode(&mut cx, byte >> bit & 1 == 1);
201            }
202        }
203        let expected = [
204            0x84, 0xC7, 0x3B, 0xFC, 0xE1, 0xA1, 0x43, 0x04, 0x02, 0x20, 0x00, 0x00, 0x41, 0x0D, 0xBB, 0x86, 0xF4, 0x31, 0x7F, 0xFF, 0x88, 0xFF, 0x37,
205            0x47, 0x1A, 0xDB, 0x6A, 0xDF, 0xFF, 0xAC,
206        ];
207        assert_eq!(mq.finish(), expected);
208    }
