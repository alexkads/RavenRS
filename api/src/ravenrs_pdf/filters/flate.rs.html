1//! FlateDecode (ISO 32000-2 §7.4.4): zlib (RFC 1950) wrapping deflate (RFC 1951).
2//!
3//! Real-world quirks handled:
4//! - the Adler-32 trailer is not required (some producers omit or corrupt it), because the
5//!   deflate stream is inflated raw after the 2-byte zlib header;
6//! - streams without a valid zlib header are inflated as raw deflate;
7//! - data after the end of the deflate stream is ignored.
8//!
9//! A back-reference to before the start of the output points at nothing, and is damage, as zlib
10//! has it: read strictly the stream is refused, and read tolerantly what came before it is kept.
11//! (miniz_oxide, flate2's backend here before zlib-rs, copied zeros and went on.) So is a dynamic
12//! block whose code has no end-of-block symbol, and nothing of that block is kept.
13

20const NAME: &str = "FlateDecode";

22/// Checks the RFC 1950 header: CM = 8, CINFO ≤ 7 and the FCHECK multiple-of-31 rule.
23fn zlib_header(data: &[u8]) -> Option<(u8, u8)> {
24    let (&cmf, &flg) = (data.first()?, data.get(1)?);
25    let ok = cmf & 0x0f == 8 && cmf >> 4 <= 7 && (u16::from(cmf) << 8 | u16::from(flg)) % 31 == 0;
26    ok.then_some((cmf, flg))
27}

29/// `data` inflated within `max`, and whether it was read twice: a zlib header that only looked
30/// valid is read again as raw deflate, within `again`, what a caller paying for what is read has
31/// left for it once it has paid for the second reading. `None`, that reading can't be paid for,
32/// and is [`FilterError::Spent`](crate::filters::FilterError::Spent).
33pub(crate) fn decode(data: &[u8], max: usize, again: Option<usize>) -> (Step, bool) {
34    match zlib_header(data) {
35        Some((_, flg)) if flg & 0x20 != 0 => (Err(Failure::corrupt(NAME, "zlib preset dictionary is not allowed in PDF", Vec::new())), false),
36        Some(_) => match inflate_raw(&data[2..], max) {
37            // A header that only looked valid: retry the whole input as raw deflate.
38            Err(f) if f.partial.is_empty() && matches!(f.error, crate::filters::FilterError::Corrupt { .. }) => {
39                let Some(again) = again else {
40                    return (Err(Failure { error: crate::filters::FilterError::Spent, partial: Vec::new(), discarded: 0 }), false);
41                };
42                let step = match inflate_raw(data, again) {
43                    // Keep the original corruption diagnostic only if the retry is corrupt
44                    // too. A limit error must remain fatal even for tolerant decoding. What the
45                    // retry decoded isn't kept, but a caller charging for work has to hear of it.
46                    Err(retry) if matches!(retry.error, crate::filters::FilterError::Corrupt { .. }) => {
47                        Err(Failure { discarded: retry.partial.len(), ..f })
48                    }
49                    retry => retry,
50                };
51                (step, true)
52            }
53            r => (r, false),
54        },
55        None => (inflate_raw(data, max), false),
56    }
57}

60thread_local! {
61    /// Bytes inflating gave back on this thread, its partial output included.
62    pub(crate) static INFLATED: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
63}

65fn inflate_raw(input: &[u8], max: usize) -> Step {
66    let step = inflate(input, max);
67    #[cfg(test)]
68    INFLATED.with(|c| c.set(c.get() + step.as_ref().map_or_else(|f| f.partial.len(), Vec::len)));
69    step
70}

72fn inflate(input: &[u8], max: usize) -> Step {
73    let mut d = Decompress::new(false);
74    let mut out: Vec<u8> = Vec::new();
75    // One byte beyond the limit lets us tell "exactly max" from "more than max".
76    let cap = max.saturating_add(1);
77    loop {
78        if out.len() == out.capacity() {
79            if out.len() >= cap {
80                return Err(Failure::limit(max));
81            }
82            let grow = out.capacity().max(input.len().saturating_mul(2)).max(4096).min(cap - out.len());
83            out.reserve_exact(grow);
84        }
85        let consumed = d.total_in() as usize;
86        let produced = out.len();
87        let res = d.decompress_vec(&input[consumed.min(input.len())..], &mut out, FlushDecompress::None);
88        if out.len() > max {
89            return Err(Failure::limit(max));
90        }
91        match res {
92            Err(e) => return Err(Failure::corrupt(NAME, e.to_string(), out)),
93            Ok(Status::StreamEnd) => return Ok(out),
94            Ok(Status::Ok | Status::BufError) => {
95                let progressed = d.total_in() as usize != consumed || out.len() != produced;
96                let room_left = out.len() < out.capacity();
97                if room_left && (d.total_in() as usize >= input.len() || !progressed) {
98                    return Err(Failure::corrupt(NAME, "unexpected end of deflate data", out));
99                }
100            }
101        }
102    }
103}

105/// zlib at level 6.
106// The documented never-crash exception: writing into a `Vec` cannot fail.
107#[allow(clippy::expect_used)]
108pub(crate) fn encode(data: &[u8]) -> Vec<u8> {
109    let mut e = flate2::write::ZlibEncoder::new(Vec::with_capacity(data.len() / 2 + 64), Compression::new(6));
110    e.write_all(data).expect("in-memory write");
111    e.finish().expect("in-memory write")
112}

206    /// Deflate's bits: header fields and extra bits least significant first, Huffman codes most
207    /// significant first (RFC 1951 §3.1.1).
208    #[derive(Default)]
209    struct Bits {
210        out: Vec<u8>,
211        acc: u32,
212        n: u32,
213    }
214
215    impl Bits {
216        fn put(&mut self, value: u32, len: u32) {
217            for i in 0..len {
218                self.acc |= ((value >> i) & 1) << self.n;
219                self.n += 1;
220                if self.n == 8 {
221                    self.out.push(self.acc as u8);
222                    (self.acc, self.n) = (0, 0);
223                }
224            }
225        }
226
227        fn code(&mut self, code: u32, len: u32) {
228            for i in (0..len).rev() {
229                self.put((code >> i) & 1, 1);
230            }
231        }
232
233        fn finish(mut self) -> Vec<u8> {
234            if self.n > 0 {
235                self.out.push(self.acc as u8);
236            }
237            self.out
238        }
239    }

241    /// A dynamic block whose literal and length code has no end-of-block symbol can't end (RFC 1951
242    /// §3.2.5): it is refused, and nothing of it is kept.
243    #[test]
244    fn a_dynamic_code_without_an_end_is_damage() {
245        let mut bits = Bits::default();
246        bits.put(1, 1); // the final block,
247        bits.put(2, 2); // dynamic Huffman codes (§3.2.7)
248        bits.put(0, 5); // 257 literal and length codes
249        bits.put(0, 5); // 1 distance code
250        bits.put(14, 4); // 18 code length codes, in their order up to the one for 1
251        for symbol in [16, 17, 18, 0, 8, 7, 9, 6, 10, 5, 11, 4, 12, 3, 13, 2, 14, 1] {
252            bits.put(u32::from(symbol == 18 || symbol == 1), 3);
253        }
254        // Code length codes: 1 is 0, 18 is 1. 'A' and 'B' get a bit each, 256 none.
255        let zeros = |bits: &mut Bits, n: u32| {
256            bits.code(1, 1);
257            bits.put(n - 11, 7);
258        };
259        zeros(&mut bits, 65);
260        bits.code(0, 1);
261        bits.code(0, 1);
262        zeros(&mut bits, 138);
263        zeros(&mut bits, 52);
264        bits.code(0, 1); // the distance code: a bit
265        for _ in 0..4 {
266            bits.code(0, 1); // 'A'
267        }
268        let data = bits.finish();
269        assert!(zlib_header(&data).is_none(), "read as raw deflate");
270        match decode(&data, 1 << 20, Some(1 << 20)).0 {
271            Err(f) => {
272                assert!(matches!(f.error, FilterError::Corrupt { .. }), "{}", f.error);
273                assert_eq!(f.partial, b"");
274            }
275            Ok(out) => panic!("read as {out:?}"),
276        }
277    }

279    /// A back-reference to before the start of the output points at nothing: read strictly the
280    /// stream is refused, and what came before it is what a tolerant reading keeps. miniz_oxide,
281    /// flate2's backend here before zlib-rs, copied zeros and went on, so this also shows which
282    /// backend inflates (the price of reading Flate, `FLATE_READ_WORK`, is set for zlib-rs).
283    #[test]
284    fn a_copy_from_before_the_start_is_damage() {
285        let literal = |bits: &mut Bits, b: u8| bits.code(0x30 + u32::from(b), 8);
286        let mut bits = Bits::default();
287        bits.put(1, 1); // the final block,
288        bits.put(1, 2); // fixed Huffman codes (§3.2.6)
289        literal(&mut bits, b'A');
290        literal(&mut bits, b'B');
291        bits.code(1, 7); // length 3
292        bits.code(4, 5); // distance 5 (code 4 and an extra bit of 0), three bytes before the start
293        bits.put(0, 1);
294        literal(&mut bits, b'C');
295        bits.code(0, 7); // end of block
296        let data = bits.finish();
297        assert!(zlib_header(&data).is_none(), "read as raw deflate");
298        match decode(&data, 1 << 20, Some(1 << 20)).0 {
299            Err(f) => {
300                assert!(matches!(f.error, FilterError::Corrupt { .. }), "{}", f.error);
301                assert_eq!(f.partial, b"AB");
302            }
303            Ok(out) => panic!("read as {out:?}"),
304        }
305    }
