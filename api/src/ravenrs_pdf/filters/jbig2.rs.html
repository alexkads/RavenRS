1//! JBIG2 encoding (ITU-T T.88 | ISO/IEC 14492), for `JBIG2Decode` (ISO 32000-2 §7.4.7), lossless:
2//! a generic region, template 0 with its nominal adaptive pixels, typical prediction on, MQ
3//! arithmetic coding ([`crate::filters::mq`]); or the image's marks as symbols ([`symbols`]), which a group
4//! of images may share through `/JBIG2Globals`. Written from the standard's 6.2.5 (coding a generic
5//! region), 7.2 (segment headers), 7.4.1, 7.4.6 and 7.4.8 (the segments) and D.3 (the embedded
6//! organisation PDF uses). Decoding is in [`decode`].

16/// The widest row, and the most rows, [`encode_jbig2_generic`] encodes: what `hayro-jbig2`, the
17/// decoder behind `JBIG2Decode` here, takes on a side. JBIG2 allows 32-bit sizes, but past this
18/// nothing here could read the stream, and the optimizer, which doesn't read back what it stores,
19/// replaced such an image with one that draws nothing.
20pub const MAX_JBIG2_COLUMNS: u32 = 65_535;
21
22/// Template 0's pixels (Figure 3), as (x, y) offsets from the pixel being coded, each a bit of the
23/// context in this order: the four on the current row, the five fixed ones on the row above, the
24/// three on the row above that, then the adaptive pixels A1–A4 at their nominal places (Table 5).
25const TEMPLATE: [(i32, i32); 16] = [
26    (-1, 0),
27    (-2, 0),
28    (-3, 0),
29    (-4, 0),
30    (-2, -1),
31    (-1, -1),
32    (0, -1),
33    (1, -1),
34    (2, -1),
35    (-1, -2),
36    (0, -2),
37    (1, -2),
38    (3, -1),
39    (-3, -1),
40    (2, -2),
41    (-2, -2),
42];
43
44/// The context that codes SLTP when template 0 is used (Figure 8): those pixel values, in
45/// [`TEMPLATE`]'s order. Figure 8 reads, by row from x = -4: `0 1 0 1` on the current row,
46/// `0 1 1 0 0 1 0` (x = -3 to 3) above it, `1 0 0 1 1` (x = -2 to 2) above that.
47const SLTP: [u8; 16] = [1, 0, 1, 0, 1, 1, 0, 0, 1, 0, 0, 1, 0, 0, 1, 1];
48
49/// Columns of padding either side of a row, so every template pixel of a row's pixels is in it.
50const PAD: usize = 4;
51
52/// `rows` rows of `columns` 1-bit PDF samples as lossless JBIG2, the smaller of two codings:
53/// [`encode_jbig2_generic`]'s, or one that stores each distinct mark of the image once, as a
54/// symbol, and places its copies (see [`symbols`]). Either way the stream is in the embedded
55/// organisation and decodes to the same bits without `/JBIG2Globals`.
56///
57/// Errors: as [`encode_jbig2_generic`].
58pub fn encode_jbig2(data: &[u8], columns: u32, rows: u32) -> Result<Vec<u8>, FilterError> {
59    let generic = encode_jbig2_generic(data, columns, rows)?;
60    let samples = rows_to_encode(data, columns, rows).ok();
61    Ok(match samples.and_then(|s| symbols::encode(s, columns, rows)) {
62        Some(coded) if coded.len() < generic.len() => coded,
63        _ => generic,
64    })
65}

67/// `rows` rows of `columns` 1-bit PDF samples (most significant bit first, each row starting on
68/// a new byte), as a JBIG2 stream in the embedded organisation: a page information segment and an
69/// immediate lossless generic region segment, which `JBIG2Decode` decodes to the same bits
70/// without `/JBIG2Globals`.
71///
72/// A 0 sample is coded as a JBIG2 1 (black), since `JBIG2Decode` gives black as 0, as a grey image's
73/// samples have it. Bytes after the last row are ignored, as are the bits that pad a row to a whole
74/// byte.
75///
76/// Errors: no rows or columns, more of either than [`MAX_JBIG2_COLUMNS`], or more samples than
77/// [`MAX_JBIG2_PIXELS`], what the decoder takes ([`FilterError::Unsupported`]); fewer bytes than the
78/// rows need ([`FilterError::Corrupt`]).
79pub fn encode_jbig2_generic(data: &[u8], columns: u32, rows: u32) -> Result<Vec<u8>, FilterError> {
80    let samples = rows_to_encode(data, columns, rows)?;
81    let width = columns as usize;
82    let stride = width.div_ceil(8);
83    let mut contexts: Vec<Context> = vec![0; 1 << 16];
84    let mut mq = MqEncoder::new();
85    let height = samples.len() / stride;
86    code_generic(&mut mq, &mut contexts, width, height, true, |y, row| {
87        let packed = samples.get(y * stride..).unwrap_or_default();
88        for (x, pixel) in row.iter_mut().enumerate() {
89            let sample = packed.get(x / 8).is_some_and(|b| b & (0x80 >> (x % 8)) != 0);
90            *pixel = u8::from(!sample);
91        }
92    });
93    let region = mq.finish();
94    let mut out = Vec::with_capacity(region.len() + 64);
95    segment(&mut out, 0, 48, &[], false, &page_information(columns, rows));
96    segment(&mut out, 1, 39, &[], false, &generic_region(columns, rows, 0, 0, &region));
97    Ok(out)
98}

100/// The samples of `rows` rows of `columns` 1-bit samples in `data`, when the encoders here take
101/// that size; the errors are [`encode_jbig2_generic`]'s.
102fn rows_to_encode(data: &[u8], columns: u32, rows: u32) -> Result<&[u8], FilterError> {
103    let too_large = columns > MAX_JBIG2_COLUMNS || rows > MAX_JBIG2_COLUMNS || u64::from(columns) * u64::from(rows) > MAX_JBIG2_PIXELS;
104    if columns == 0 || rows == 0 || too_large {
105        return Err(FilterError::Unsupported(format!("JBIG2 encoding of {columns} × {rows} samples")));
106    }
107    let stride = (columns as usize).div_ceil(8);
108    usize::try_from(rows)
109        .ok()
110        .and_then(|r| r.checked_mul(stride))
111        .and_then(|n| data.get(..n))
112        .ok_or_else(|| FilterError::Corrupt { filter: "JBIG2Decode", detail: format!("{} bytes for {rows} rows of {columns} samples", data.len()) })
113}

115/// A group of images coded by [`encode_jbig2_shared`]: the data of the `/JBIG2Globals` stream
116/// they share, and each image's own stream, in the order given, or `None` for one that shares
117/// nothing with the others.
118#[derive(Debug)]
119pub struct SharedJbig2 {
120    pub globals: Vec<u8>,
121    pub images: Vec<Option<Vec<u8>>>,
122}
123
124/// Images, each `rows` rows of `columns` 1-bit PDF samples, as lossless JBIG2 sharing one symbol
125/// dictionary: the marks found in more than one of them are stored once, in a symbol dictionary
126/// segment not associated with a page, which is the `/JBIG2Globals` stream's data, and each image's
127/// stream, in the embedded organisation, places them from there, with its other marks coded as
128/// [`encode_jbig2`]'s symbol coding codes them (see [`symbols`]). With the globals, each decodes to
129/// its own bits. An image that holds none of those marks, or that symbol coding doesn't take, has
130/// no stream here; it is stored alone. `None` when the images share fewer than two marks.
131///
132/// What each image's stream saves is left for the caller to weigh against what the globals take,
133/// which the images that use them share.
134///
135/// Errors: as [`encode_jbig2_generic`], for the first image it would refuse.
136pub fn encode_jbig2_shared(images: &[(&[u8], u32, u32)]) -> Result<Option<SharedJbig2>, FilterError> {
137    let images =
138        images.iter().map(|(data, columns, rows)| Ok((rows_to_encode(data, *columns, *rows)?, *columns, *rows))).collect::<Result<Vec<_>, _>>()?;
139    Ok(symbols::encode_shared(&images).map(|(globals, images)| SharedJbig2 { globals, images }))
140}

142/// A page information segment's data (7.4.8): the page's size, resolution unknown, eventually
143/// lossless, default pixel 0 (white), OR combination, not striped.
144fn page_information(columns: u32, rows: u32) -> Vec<u8> {
145    let mut page = Vec::with_capacity(19);
146    page.extend_from_slice(&columns.to_be_bytes());
147    page.extend_from_slice(&rows.to_be_bytes());
148    page.extend_from_slice(&[0; 8]);
149    page.push(0x01);
150    page.extend_from_slice(&[0, 0]);
151    page
152}

154/// A region segment information field (7.4.1): a `width` × `height` region at (`x`, `y`) on the
155/// page, combined with it by OR.
156fn region_information(out: &mut Vec<u8>, width: u32, height: u32, x: u32, y: u32) {
157    for v in [width, height, x, y] {
158        out.extend_from_slice(&v.to_be_bytes());
159    }
160    out.push(0);
161}

163/// An immediate lossless generic region segment's data (7.4.6): its region information, flags
164/// (MMR 0, template 0, TPGDON 1), the AT pixels at their nominal places, then `coded`.
165fn generic_region(width: u32, height: u32, x: u32, y: u32, coded: &[u8]) -> Vec<u8> {
166    let mut data = Vec::with_capacity(26 + coded.len());
167    region_information(&mut data, width, height, x, y);
168    data.push(0x08);
169    at_pixels(&mut data);
170    data.extend_from_slice(coded);
171    data
172}

174/// Template 0's four AT pixels at their nominal places, as AT flags (7.4.6.3, 7.4.2.1.2).
175fn at_pixels(out: &mut Vec<u8>) {
176    for (x, y) in &TEMPLATE[12..] {
177        out.push(*x as i8 as u8);
178        out.push(*y as i8 as u8);
179    }
180}

182/// A segment (7.2) of page 1: its header (number, type, the segments it refers to, the page, data
183/// length), then its data. `retained`: a later segment refers to this one. `refers_to` holds each
184/// referred-to segment's number and whether a segment after this one refers to it too (its
185/// retain bit, 7.2.4). Segment numbers here stay below 256, so each referred-to one takes a byte
186/// (7.2.5); a segment refers to at most four, so their count and retain bits take one byte too.
187fn segment(out: &mut Vec<u8>, number: u32, kind: u8, refers_to: &[(u8, bool)], retained: bool, data: &[u8]) {
188    segment_of(out, 1, number, kind, refers_to, retained, data);
189}
190
191/// [`segment`], of page `page`: 0 for a segment associated with no page (7.2.6), such as the
192/// global ones `/JBIG2Globals` holds, which a page's segments may refer to.
193fn segment_of(out: &mut Vec<u8>, page: u8, number: u32, kind: u8, refers_to: &[(u8, bool)], retained: bool, data: &[u8]) {
194    out.extend_from_slice(&number.to_be_bytes());
195    out.push(kind);
196    // At most four, so the count fits its three bits.
197    let refers_to = refers_to.get(..refers_to.len().min(4)).unwrap_or_default();
198    let kept = refers_to.iter().enumerate().fold(0u8, |bits, (k, (_, kept))| bits | u8::from(*kept) << (k + 1));
199    out.push((refers_to.len() as u8) << 5 | kept | u8::from(retained));
200    out.extend(refers_to.iter().map(|(n, _)| *n));
201    out.push(page);
202    // A segment's data never comes near 4 GiB here: an image is at most 2^28 samples.
203    out.extend_from_slice(&u32::try_from(data.len()).unwrap_or(u32::MAX).to_be_bytes());
204    out.extend_from_slice(data);
205}

207/// Codes a `width` × `height` bitmap as a generic region (6.2.5.7) with template 0 and the nominal
208/// AT pixels, its decisions in `contexts` (2^16 of them), with typical prediction when `typical`
209/// (TPGDON). `fill(y, row)` sets row `y`'s `width` pixels, 1 for black. Pixels outside the bitmap
210/// count as 0 (white), as the decoder takes them.
211fn code_generic(mq: &mut MqEncoder, contexts: &mut [Context], width: usize, height: usize, typical: bool, mut fill: impl FnMut(usize, &mut [u8])) {
212    let sltp = SLTP.iter().enumerate().fold(0usize, |cx, (k, v)| cx | usize::from(*v) << k);
213    // The two rows above the current one and the current one, as JBIG2 pixels (1 black), padded.
214    let blank = vec![0u8; width + 2 * PAD];
215    let (mut above2, mut above1) = (blank.clone(), blank.clone());
216    let mut row = blank;
217    let mut ltp = false;
218    for y in 0..height {
219        if let Some(pixels) = row.get_mut(PAD..PAD + width) {
220            fill(y, pixels);
221        }
222        if typical {
223            // Typical prediction (6.2.5.5): a row the same as the one above is coded as one decision.
224            let same = row == above1;
225            if let Some(cx) = contexts.get_mut(sltp) {
226                mq.encode(cx, same != ltp);
227            }
228            ltp = same;
229        }
230        if !ltp {
231            for x in 0..width {
232                let at = |dx: i32, dy: i32| {
233                    let line = match dy {
234                        0 => &row,
235                        -1 => &above1,
236                        _ => &above2,
237                    };
238                    let i = (PAD + x).checked_add_signed(dx as isize);
239                    i.and_then(|i| line.get(i)).copied().unwrap_or(0)
240                };
241                let cx = TEMPLATE.iter().enumerate().fold(0usize, |cx, (k, (dx, dy))| cx | usize::from(at(*dx, *dy)) << k);
242                if let Some(state) = contexts.get_mut(cx) {
243                    mq.encode(state, row.get(PAD + x) == Some(&1));
244                }
245            }
246        }
247        std::mem::swap(&mut above2, &mut above1);
248        std::mem::swap(&mut above1, &mut row);
249    }
250}

262    /// [`decode`], with `/JBIG2Globals`.
263    pub(super) fn decode_with(data: &[u8], globals: Option<&[u8]>) -> Vec<Vec<bool>> {
264        struct Rows(Vec<Vec<bool>>, Vec<bool>, usize);
265        impl crate::third_party::hayro_jbig2::Decoder for Rows {
266            fn push_pixel(&mut self, black: bool) {
267                self.1.push(!black);
268            }
269            fn push_pixel_chunk(&mut self, black: bool, chunks: u32) {
270                self.1.extend(std::iter::repeat_n(!black, chunks as usize * 8));
271            }
272            fn next_line(&mut self) {
273                let mut row = std::mem::take(&mut self.1);
274                row.truncate(self.2);
275                self.0.push(row);
276            }
277        }
278        let image = crate::third_party::hayro_jbig2::Image::new_embedded(data, globals).unwrap();
279        let mut rows = Rows(Vec::new(), Vec::new(), image.width() as usize);
280        image.decode(&mut rows).unwrap();
281        rows.0
282    }

284    /// The rows of packed samples `data`, true for a 1 bit.
285    pub(super) fn unpack(data: &[u8], columns: u32, rows: u32) -> Vec<Vec<bool>> {
286        let stride = (columns as usize).div_ceil(8);
287        (0..rows as usize).map(|y| (0..columns as usize).map(|x| data[y * stride + x / 8] & (0x80 >> (x % 8)) != 0).collect()).collect()
288    }

290    /// Rows of `columns` samples from `runs`: alternating white and black runs one longer than
291    /// those values (never empty, even when shrinking makes them all 0), until the rows are full;
292    /// or, with `noise`, the bytes of `runs` as they are.
293    fn bitmap(columns: u32, rows: u32, runs: &[u8], noise: bool) -> Vec<u8> {
294        let stride = (columns as usize).div_ceil(8);
295        let mut data = vec![0u8; stride * rows as usize];
296        if runs.is_empty() {
297            return data;
298        }
299        if noise {
300            data.iter_mut().zip(runs.iter().cycle()).for_each(|(d, r)| *d = *r);
301            return data;
302        }
303        let (mut white, mut left, mut next) = (true, 1 + usize::from(runs[0]), 1);
304        for y in 0..rows as usize {
305            for x in 0..columns as usize {
306                if left == 0 {
307                    white = !white;
308                    left = 1 + usize::from(runs[next % runs.len()]);
309                    next += 1;
310                }
311                if white {
312                    data[y * stride + x / 8] |= 0x80 >> (x % 8);
313                }
314                left -= 1;
315            }
316        }
317        data
318    }

351    /// Packed samples from JBIG2 pixels (1 black): a 0 sample for each 1.
352    pub(super) fn samples(pixels: &[Vec<u8>]) -> Vec<u8> {
353        let width = pixels.first().map_or(0, Vec::len);
354        let stride = width.div_ceil(8);
355        let mut data = vec![0xFF; stride * pixels.len()];
356        for (y, row) in pixels.iter().enumerate() {
357            for (x, p) in row.iter().enumerate() {
358                if *p == 1 {
359                    data[y * stride + x / 8] &= !(0x80 >> (x % 8));
360                }
361            }
362        }
363        data
364    }

413    #[test]
414    fn it_encodes_only_what_the_decoder_takes() {
415        let unsupported = |columns, rows| matches!(encode_jbig2_generic(&[], columns, rows), Err(FilterError::Unsupported(_)));
416        assert!(unsupported(MAX_JBIG2_COLUMNS + 1, 1), "too wide");
417        assert!(unsupported(1, MAX_JBIG2_COLUMNS + 1), "too tall");
418        assert!(unsupported(16_385, 16_384), "too many samples, each side within the bound");
419        assert!(unsupported(u32::MAX, u32::MAX), "the product doesn't overflow");
420        // At the bounds what it writes reads back, through the decoder that has to read it.
421        let wide = vec![0x0F; (MAX_JBIG2_COLUMNS as usize).div_ceil(8) * 4];
422        assert_eq!(decode(&encode_jbig2_generic(&wide, MAX_JBIG2_COLUMNS, 4).unwrap()), unpack(&wide, MAX_JBIG2_COLUMNS, 4));
423        let tall = vec![0x7F; MAX_JBIG2_COLUMNS as usize];
424        assert_eq!(decode(&encode_jbig2_generic(&tall, 1, MAX_JBIG2_COLUMNS).unwrap()), unpack(&tall, 1, MAX_JBIG2_COLUMNS));
425        struct Rows(u32, u64);
426        impl crate::third_party::hayro_jbig2::Decoder for Rows {
427            fn push_pixel(&mut self, black: bool) {
428                self.1 += u64::from(black);
429            }
430            fn push_pixel_chunk(&mut self, black: bool, chunks: u32) {
431                self.1 += u64::from(black) * u64::from(chunks) * 8;
432            }
433            fn next_line(&mut self) {
434                self.0 += 1;
435            }
436        }
437        // One black pixel, so a page that came back blank wouldn't pass.
438        let mut largest = vec![0xFF; 2_048 * 16_384];
439        largest[0] = 0x7F;
440        let largest = encode_jbig2_generic(&largest, 16_384, 16_384).unwrap();
441        let mut rows = Rows(0, 0);
442        crate::third_party::hayro_jbig2::Image::new_embedded(&largest, None).unwrap().decode(&mut rows).unwrap();
443        assert_eq!(rows.0, 16_384);
444        assert_eq!(rows.1, 1);
445    }
