6/// The largest JBIG2 page [`decode_jbig2_within`] decodes, in pixels. The decoder decodes a page whole,
7/// and takes pages of up to this many (20,000 × 10,000 decodes). It is also the most
8/// [`encode_jbig2_generic`](crate::filters::encode_jbig2_generic) encodes, so what it stores reads back.
9pub const MAX_JBIG2_PIXELS: u64 = 1 << 28;
10
11/// A `JBIG2Decode` stream's `data`, with what its `/JBIG2Globals` stream decodes to, decoded as the
12/// filter gives it: `rows` rows of `columns` 1-bit samples, most significant bit first, each
13/// starting on a new byte, with black as 0 (JBIG2 codes black as 1).
14///
15/// The JBIG2 page has to be the image's size, `columns` × `rows`: PdfCraft's renderer draws the
16/// whole page into the image, so the rows of a page of another height wouldn't be the image's. The
17/// decoder decodes the page whole, so it is checked before it is decoded, and has to fit in `max`
18/// bytes as such rows and have at most [`MAX_JBIG2_PIXELS`]. Regions inside it are bounded by the
19/// decoder alone: a region may claim more pixels than its page, and is decoded before it is cut to
20/// the page. With no rows asked for, nothing is decoded.
21///
22/// Errors: `data` or `globals` that don't decode, or a page of another size
23/// ([`FilterError::Corrupt`]); a page past `max` ([`FilterError::LimitExceeded`]) or
24/// [`MAX_JBIG2_PIXELS`] ([`FilterError::Unsupported`], or `Corrupt` from the decoder, which has the
25/// same cap and can refuse the page first).
26///
27/// Decoding spends from `work`: what decoding may still cost across a run of images, in the
28/// decoder's count of steps. There is no way in without it, so a run of images can't decode them
29/// without sharing an allowance. Reading `data` and `globals` is charged first
30/// ([`crate::third_party::hayro_jbig2::charge_reading`]), so an image that turns out not to decode has paid for that,
31/// then what decoding takes ([`crate::third_party::hayro_jbig2::Image::decode_within`]). An image takes at most what
32/// one decode may on top of its reading and its page, whatever `work` allows, and one that `work`
33/// can't pay for spends all of it. With nothing left an image isn't read at all, so the images after
34/// that one cost nothing but this check, however large their globals.
35pub fn decode_jbig2_within(data: &[u8], globals: Option<&[u8]>, columns: u32, rows: u32, max: usize, work: &mut u64) -> Result<Vec<u8>, FilterError> {
36    if rows == 0 {
37        return Ok(Vec::new());
38    }
39    if *work == 0 {
40        return Err(FilterError::Spent);
41    }
42    crate::third_party::hayro_jbig2::charge_reading(work, data.len().saturating_add(globals.map_or(0, <[u8]>::len)))
43        .map_err(|_| FilterError::Spent)?;
44    let corrupt = |detail: String| FilterError::Corrupt { filter: "JBIG2Decode", detail };
45    let image = crate::third_party::hayro_jbig2::Image::new_embedded(data, globals).map_err(|e| corrupt(e.to_string()))?;
46    let (width, height) = (image.width(), image.height());
47    if (width, height) != (columns, rows) {
48        return Err(corrupt(format!("a page of {width} × {height} pixels for an image of {columns} × {rows}")));
49    }
50    if u64::from(width) * u64::from(height) > MAX_JBIG2_PIXELS {
51        return Err(FilterError::Unsupported(format!("JBIG2 pages of {width} × {height} pixels")));
52    }
53    // u32 to usize: wasm32 and up have at least 32-bit usize.
54    let (width, height) = (width as usize, height as usize);
55    if width.div_ceil(8).checked_mul(height).is_none_or(|bytes| bytes > max) {
56        return Err(FilterError::LimitExceeded(max));
57    }
58    let mut out = Rows::new(width, height, false);
59    // Stopped by what was left, it isn't a damaged image, whatever the decoder calls it.
60    image.decode_within(&mut out, work).map_err(|e| if *work == 0 { FilterError::Spent } else { corrupt(e.to_string()) })?;
61    Ok(out.out)
62}

64/// JBIG2 codes black as 1, which `JBIG2Decode` gives as 0.
65impl crate::third_party::hayro_jbig2::Decoder for Rows {
66    fn push_pixel(&mut self, black: bool) {
67        self.push(!black, 1);
68    }
69    fn push_pixel_chunk(&mut self, black: bool, chunks: u32) {
70        self.push(!black, (chunks as usize).saturating_mul(8));
71    }
72    fn next_line(&mut self) {
73        self.end_row();
74    }

88    /// `rows` rows of `columns` samples of noise, the bits that pad each row to a byte 0.
89    fn noise(columns: u32, rows: u32) -> Vec<u8> {
90        let stride = (columns as usize).div_ceil(8);
91        let mut seed = 0x2545_F491_4F6C_DD1Du64;
92        let mut data: Vec<u8> = (0..stride * rows as usize)
93            .map(|_| {
94                seed ^= seed << 13;
95                seed ^= seed >> 7;
96                seed ^= seed << 17;
97                seed as u8
98            })
99            .collect();
100        let pad = (stride * 8 - columns as usize) as u32;
101        data.chunks_mut(stride).for_each(|row| *row.last_mut().unwrap() &= 0xFFu8.checked_shl(pad).unwrap_or(0));
102        data
103    }

105    /// The same stream with its page information (the first segment's data, after its 11-byte
106    /// header: width, then height) claiming a page of `width` × `height`.
107    fn page_of(data: &[u8], width: u32, height: u32) -> Vec<u8> {
108        let mut data = data.to_vec();
109        data[11..15].copy_from_slice(&width.to_be_bytes());
110        data[15..19].copy_from_slice(&height.to_be_bytes());
111        data
112    }

114    /// The JBIG2 file header's ID string (ISO 14492 D.4.1).
115    const FILE_HEADER_ID: [u8; 8] = [0x97, 0x4A, 0x42, 0x32, 0x0D, 0x0A, 0x1A, 0x0A];

117    /// `embedded`, segments as [`encode_jbig2_generic`] writes them (11-byte headers: no references,
118    /// a one-byte page association), behind a file header with `flags` and the page count `pages`,
119    /// in the organization `flags` declares (D.4.2, bit 0): 1 sequential, each segment's header
120    /// before its data (D.1); 0 random access, every header first, the last an end of file segment
121    /// numbered 2, which tells where they end, then every segment's data in the same order (D.2).
122    fn with_file_header(embedded: &[u8], flags: u8, pages: Option<u32>) -> Vec<u8> {
123        let mut data = FILE_HEADER_ID.to_vec();
124        data.push(flags);
125        data.extend(pages.map(u32::to_be_bytes).into_iter().flatten());
126        if flags & 0x01 != 0 {
127            data.extend_from_slice(embedded);
128            return data;
129        }
130        let (mut headers, mut bodies, mut rest) = (Vec::new(), Vec::new(), embedded);
131        while !rest.is_empty() {
132            let (header, after) = rest.split_at(11);
133            assert_eq!(header[5], 0, "no references");
134            let (body, after) = after.split_at(u32::from_be_bytes(header[7..11].try_into().unwrap()) as usize);
135            headers.extend_from_slice(header);
136            bodies.extend_from_slice(body);
137            rest = after;
138        }
139        headers.extend_from_slice(&[0, 0, 0, 2, 51, 0, 1, 0, 0, 0, 0]);
140        data.extend(headers);
141        data.extend(bodies);
142        data
143    }

145    /// A stream that starts with a JBIG2 file header (ISO 14492 D.4), which some producers write
146    /// into `JBIG2Decode` data although ISO 32000-1 7.4.7 has the embedded organization there,
147    /// decodes past it as Acrobat and PDFBox draw it (hayro-jbig2 patch 17), in either organization
148    /// the header declares, with the page count and without it. "Bit 0: File organisation type. If
149    /// this bit is 0, the file uses the random-access organisation. If this bit is 1, the file uses
150    /// the sequential organisation." (D.4.2) A header with its reserved flag bits set is an error,
151    /// not a panic.
152    #[test]
153    fn a_stream_with_a_jbig2_file_header_decodes_past_it() {
154        let (w, h) = (203u32, 41u32);
155        let bits = noise(w, h);
156        let max = bits.len();
157        let embedded = encode_jbig2_generic(&bits, w, h).unwrap();
158        let decode = |flags: u8, pages: Option<u32>| decode_jbig2(&with_file_header(&embedded, flags, pages), None, w, h, max);
159        // Sequential organization (bit 0 is 1), one page.
160        assert_eq!(decode(0x01, Some(1)).unwrap(), bits, "sequential, page count given");
161        // Sequential, number of pages unknown (bit 1 is 1): no count follows (D.4.3).
162        assert_eq!(decode(0x03, None).unwrap(), bits, "sequential, page count unknown");
163        // Random-access organization (bit 0 is 0): the segment headers, then their data.
164        assert_ne!(with_file_header(&embedded, 0x00, Some(1))[13..], embedded[..], "laid out otherwise");
165        assert_eq!(decode(0x00, Some(1)).unwrap(), bits, "random access, page count given");
166        assert_eq!(decode(0x02, None).unwrap(), bits, "random access, page count unknown");
167        assert!(decode(0x11, Some(1)).is_err(), "reserved bits");
168    }

170    /// A file header cut short, or the segments after it, is an error and not a panic: the ID string
171    /// alone or in part, a header whose flags promise a page count with fewer than its four bytes,
172    /// a header with no segments after it, and segments cut anywhere in either organization. No
173    /// prefix of a whole stream decodes to its samples.
174    #[test]
175    fn a_truncated_jbig2_file_header_is_an_error() {
176        let (w, h) = (24u32, 8u32);
177        let bits = noise(w, h);
178        let max = bits.len();
179        let embedded = encode_jbig2_generic(&bits, w, h).unwrap();
180        let decode = |data: &[u8]| decode_jbig2(data, None, w, h, max);
181        assert!(decode(&FILE_HEADER_ID[..5]).is_err(), "part of the ID string");
182        assert!(decode(&FILE_HEADER_ID).is_err(), "the ID string, no flags");
183        for count in 0..4 {
184            let mut data = FILE_HEADER_ID.to_vec();
185            data.push(0x01);
186            data.extend_from_slice(&1u32.to_be_bytes()[..count]);
187            assert!(decode(&data).is_err(), "{count} bytes of the page count");
188        }
189        for (flags, pages) in [(0x01, Some(1)), (0x03, None), (0x00, Some(1)), (0x02, None)] {
190            let whole = with_file_header(&embedded, flags, pages);
191            let header = 9 + if pages.is_some() { 4 } else { 0 };
192            assert!(decode(&whole[..header]).is_err(), "flags {flags:#04x}: the header alone");
193            assert!(decode(&whole[..whole.len() - 1]).is_err(), "flags {flags:#04x}: the last byte cut");
194            for end in 0..whole.len() {
195                assert_ne!(decode(&whole[..end]).ok(), Some(bits.clone()), "flags {flags:#04x}: {end} bytes");
196            }
197            assert_eq!(decode(&whole).unwrap(), bits, "flags {flags:#04x}: whole");
198        }

201    /// The segments after a file header count toward hayro-jbig2's cap on the segments of a stream
202    /// together with its `/JBIG2Globals`, in either organization (patches 8 and 17): globals of
203    /// extension segments, which decode to nothing, fill the cap to the image's segments, and one
204    /// more is refused.
205    #[test]
206    fn segments_after_a_jbig2_file_header_count_with_the_globals() {
207        /// hayro-jbig2's `MAX_SEGMENTS`, the most segments of a stream with its globals.
208        const MAX_SEGMENTS: u32 = 1 << 20;
209        let (w, h) = (24u32, 8u32);
210        let bits = noise(w, h);
211        let max = bits.len();
212        let embedded = encode_jbig2_generic(&bits, w, h).unwrap();
213        // Numbered past the image's segments (0 to 2), of no page, with no data.
214        let globals = |count: u32| -> Vec<u8> {
215            (16..16 + count)
216                .flat_map(|number| {
217                    let mut segment = number.to_be_bytes().to_vec();
218                    segment.extend_from_slice(&[62, 0, 0, 0, 0, 0, 0]);
219                    segment
220                })
221                .collect()
222        };
223        // The image's page information and region, and in random access its end of file segment.
224        for (flags, organization, image) in [(0x01, "sequential", 2), (0x00, "random access", 3)] {
225            let data = with_file_header(&embedded, flags, Some(1));
226            let (at_cap, past_cap) = (globals(MAX_SEGMENTS - image), globals(MAX_SEGMENTS - image + 1));
227            assert_eq!(decode_jbig2(&data, Some(&at_cap), w, h, max).unwrap(), bits, "{organization}: at the cap");
228            let refused = decode_jbig2(&data, Some(&past_cap), w, h, max).unwrap_err();
229            assert!(refused.to_string().contains("too many segments"), "{organization}: {refused}");
230        }

404    /// What a page costs is the same however much a run has left. Given exactly that it decodes,
405    /// and less stops it: spending all that's left, and what comes after isn't read. A page that
406    /// fails still spends what it took.
407    #[test]
408    fn a_run_of_images_spends_its_allowance_down_to_the_step() {
409        let (w, h) = (203u32, 41u32);
410        let data = encode_jbig2_generic(&noise(w, h), w, h).unwrap();
411        let whole = decode_jbig2(&data, None, w, h, usize::MAX).unwrap();
412        let mut work = u64::MAX;
413        assert_eq!(decode_jbig2_within(&data, None, w, h, usize::MAX, &mut work).unwrap(), whole);
414        let cost = u64::MAX - work;
415        assert!(cost >= u64::from(w) * u64::from(h), "a pixel at a time: {cost}");
416        let mut work = cost;
417        assert_eq!(decode_jbig2_within(&data, None, w, h, usize::MAX, &mut work).unwrap(), whole, "exactly enough");
418        assert_eq!(work, 0);
419        let mut work = cost + 7;
420        decode_jbig2_within(&data, None, w, h, usize::MAX, &mut work).unwrap();
421        assert_eq!(work, 7, "only what it took");
422        let mut work = cost - 1;
423        assert!(matches!(decode_jbig2_within(&data, None, w, h, usize::MAX, &mut work), Err(FilterError::Spent)), "a step short");
424        assert_eq!(work, 0, "spends all that was left");
425        // Nothing left: not read at all, so this garbage isn't reported as garbage.
426        assert!(matches!(decode_jbig2_within(b"garbage", None, w, h, usize::MAX, &mut work), Err(FilterError::Spent)));
427        // Too little even for its page.
428        let mut work = 3;
429        assert!(decode_jbig2_within(&data, None, w, h, usize::MAX, &mut work).is_err());
430        assert_eq!(work, 0);
431        // A region that can't be read after the one that can: the page fails having spent the
432        // first, and the segment it fails on, and that isn't being out of allowance.
433        let bad_region = [0, 0, 0, 9, 38, 0, 1, 0, 0, 0, 1, 0xFF];
434        let mut work = u64::MAX;
435        assert!(decode_jbig2_within(&[&data[..], &bad_region].concat(), None, w, h, usize::MAX, &mut work).is_err());
436        assert_eq!(u64::MAX - work, cost + 1 + 4 * bad_region.len() as u64, "the first region, a segment more, and reading it");
437        // Not a stream at all: it pays for being read, and globals count with it.
438        let mut work = u64::MAX;
439        assert!(matches!(decode_jbig2_within(b"garbage", Some(b"globals"), w, h, usize::MAX, &mut work), Err(FilterError::Corrupt { .. })));
440        assert_eq!(u64::MAX - work, 4 * 14);
441    }

494    /// The page has to be the image's size, and is bounded before it is decoded, being decoded
495    /// whole: within `max` as rows of one bit, and within [`MAX_JBIG2_PIXELS`]. Globals are given
496    /// to the decoder: ones it skips decode, ones that aren't JBIG2 are an error, as is damaged
497    /// data, which never panics.
498    #[test]
499    fn what_it_cannot_decode_is_an_error() {
500        let (w, h) = (203u32, 41u32);
501        let data = encode_jbig2_generic(&noise(w, h), w, h).unwrap();
502        for (columns, rows) in [(w - 1, h), (w + 1, h), (w, h - 1), (w, h + 1)] {
503            let e = decode_jbig2(&data, None, columns, rows, usize::MAX).unwrap_err();
504            assert!(matches!(e, FilterError::Corrupt { .. }), "{columns} × {rows}: {e}");
505        }
506        let page = 26 * h as usize;
507        assert!(matches!(decode_jbig2(&data, None, w, h, page - 1), Err(FilterError::LimitExceeded(m)) if m == page - 1), "a byte short");
508        assert_eq!(decode_jbig2(&data, None, w, h, page).unwrap().len(), page, "the page within max");
509        // 16,384 × 16,384 is 2^28 pixels, which decodes; a row more is past the cap, here or in the
510        // decoder, which has the same one and can refuse it first.
511        let large = |rows| decode_jbig2(&page_of(&data, 16_384, rows), None, 16_384, rows, usize::MAX);
512        assert!(matches!(large(16_385), Err(FilterError::Unsupported(_) | FilterError::Corrupt { .. })), "{:?}", large(16_385));
513        assert!(large(16_384).is_ok(), "{:?}", large(16_384).map(|page| page.len()));
514
515        // An extension segment (ISO/IEC 14492 7.4.14), which a decoder skips: its 11-byte header
516        // gives the length of the data after it.
517        let skipped = [0, 0, 0, 0, 0x3E, 0, 0, 0, 0, 0, 4, 0, 0, 0, 1];
518        let generic = decode_jbig2(&data, None, w, h, usize::MAX).unwrap();
519        assert_eq!(decode_jbig2(&data, Some(&skipped), w, h, usize::MAX).unwrap(), generic);
520        assert!(matches!(decode_jbig2(&data, Some(b"not JBIG2"), w, h, usize::MAX), Err(FilterError::Corrupt { .. })));
521        for end in 0..data.len() {
522            let _ = decode_jbig2(&data[..end], None, w, h, usize::MAX);
523        }
524    }
