1//! ASCIIHexDecode (ISO 32000-2 §7.4.2): pairs of hex digits, white space ignored,
2//! `>` is EOD, an odd final digit is padded with 0. A missing EOD is tolerated.
3
4use crate::filters::{Failure, Step, is_pdf_whitespace};
5
6const NAME: &str = "ASCIIHexDecode";
7
8fn nibble(b: u8) -> Option<u8> {
9    match b {
10        b'0'..=b'9' => Some(b - b'0'),
11        b'a'..=b'f' => Some(b - b'a' + 10),
12        b'A'..=b'F' => Some(b - b'A' + 10),
13        _ => None,
14    }
15}
16
17pub(crate) fn decode(data: &[u8], max: usize) -> Step {
18    let mut out = Vec::with_capacity((data.len() / 2).min(max));
19    let mut hi: Option<u8> = None;
20    for (i, &b) in data.iter().enumerate() {
21        if is_pdf_whitespace(b) {
22            continue;
23        }
24        if b == b'>' {
25            break;
26        }
27        let Some(n) = nibble(b) else {
28            return Err(Failure::corrupt(NAME, format!("invalid character 0x{b:02x} at offset {i}"), out));
29        };
30        match hi.take() {
31            None => hi = Some(n),
32            Some(h) => {
33                if out.len() == max {
34                    return Err(Failure::limit(max));
35                }
36                out.push(h << 4 | n);
37            }
38        }
39    }
40    if let Some(h) = hi {
41        if out.len() == max {
42            return Err(Failure::limit(max));
43        }
44        out.push(h << 4);
45    }
46    Ok(out)
47}

49/// Upper-case hex, a newline every 64 digits, terminated by `>`.
50pub(crate) fn encode(data: &[u8]) -> Vec<u8> {
51    const HEX: &[u8; 16] = b"0123456789ABCDEF";
52    let mut out = Vec::with_capacity(data.len() * 2 + data.len() / 32 + 1);
53    for (i, &b) in data.iter().enumerate() {
54        if i > 0 && i % 32 == 0 {
55            out.push(b'\n');
56        }
57        out.push(HEX[usize::from(b >> 4)]);
58        out.push(HEX[usize::from(b & 15)]);
59    }
60    out.push(b'>');
61    out
62}

64#[cfg(test)]
65mod tests {
66    use super::*;
67    use crate::filters::FilterError;
68
69    fn dec(d: &[u8]) -> Result<Vec<u8>, FilterError> {
70        decode(d, 1000).map_err(|f| f.error)
71    }
72
73    #[test]
74    fn vectors() {
75        assert_eq!(dec(b"48656c6C6F>").unwrap(), b"Hello");
76        assert_eq!(dec(b" 4 8\n6\r5\t6\x0c c\x006c 6f >").unwrap(), b"Hello");
77        assert_eq!(dec(b"901FA>").unwrap(), [0x90, 0x1f, 0xa0]);
78        assert_eq!(dec(b"901FA").unwrap(), [0x90, 0x1f, 0xa0]); // missing EOD
79        assert_eq!(dec(b"41>zz garbage after EOD").unwrap(), b"A");
80        assert_eq!(dec(b">").unwrap(), b"");
81        assert_eq!(dec(b"").unwrap(), b"");
82    }
83
84    #[test]
85    fn invalid_char() {
86        let f = decode(b"4142G43>", 1000).err().unwrap();
87        assert!(matches!(f.error, FilterError::Corrupt { .. }));
88        assert_eq!(f.partial, b"AB");
89    }
90
91    #[test]
92    fn limit() {
93        assert!(matches!(decode(b"414243", 2).map_err(|f| f.error), Err(FilterError::LimitExceeded(2))));
94        assert!(matches!(decode(b"41424", 2).map_err(|f| f.error), Err(FilterError::LimitExceeded(2))));
95        assert_eq!(decode(b"4142", 2).ok().unwrap(), b"AB");
96    }
97
98    #[test]
99    fn encoding() {
100        assert_eq!(encode(b"Hello"), b"48656C6C6F>");
101        assert_eq!(encode(b""), b">");
102        let long = encode(&[0xab; 40]);
103        assert_eq!(long[64], b'\n');
104        assert_eq!(dec(&long).unwrap(), [0xab; 40]);
105    }
106}
