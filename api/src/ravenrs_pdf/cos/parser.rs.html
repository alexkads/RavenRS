1//! Lexer and object parser (ISO 32000-2 §7.2–7.3), tolerant of common real-world damage.
2//!
3//! The parser works on a byte slice and a position; it never allocates for skipped data and
4//! never panics on malformed input (every index is bounds-checked). Depth is limited to defend
5//! against pathological nesting.
6
7use std::sync::Arc;
8
9use crate::cos::object::{Dict, ObjRef, Object, PdfString, Stream};
10use crate::cos::{Bytes, CosError};
11
12const MAX_DEPTH: usize = 128;
13/// Objects one top-level [`Lexer::object`] call may build. Real arrays (page trees, parent
14/// trees, widths) stay far below; a decompressed object stream of `0 0 0 ...` does not.
15const MAX_NODES: usize = 1 << 22;
16
17/// One of the six PDF whitespace characters (ISO 32000-2 §7.2.3, Table 1).
18pub fn is_whitespace(b: u8) -> bool {
19    matches!(b, b'\0' | b'\t' | b'\n' | b'\x0C' | b'\r' | b' ')
20}
21
22pub(crate) fn is_delimiter(b: u8) -> bool {
23    matches!(b, b'(' | b')' | b'<' | b'>' | b'[' | b']' | b'{' | b'}' | b'/' | b'%')
24}
25
26fn is_regular(b: u8) -> bool {
27    !is_whitespace(b) && !is_delimiter(b)
28}
29
30/// A cursor over PDF bytes.
31pub struct Lexer<'a> {
32    pub data: &'a [u8],
33    pub pos: usize,
34    /// Objects built by the current top-level [`Lexer::object`] call.
35    nodes: usize,
36}
37
38impl<'a> Lexer<'a> {
39    pub fn new(data: &'a [u8], pos: usize) -> Self {
40        Self { data, pos: pos.min(data.len()), nodes: 0 }
41    }
42
43    fn peek(&self) -> Option<u8> {
44        self.data.get(self.pos).copied()
45    }
46
47    /// Skip whitespace and comments.
48    pub fn skip_ws(&mut self) {
49        while let Some(b) = self.peek() {
50            if is_whitespace(b) {
51                self.pos += 1;
52            } else if b == b'%' {
53                while let Some(c) = self.peek() {
54                    if c == b'\n' || c == b'\r' {
55                        break;
56                    }
57                    self.pos += 1;
58                }
59            } else {
60                break;
61            }
62        }
63    }
64
65    /// Read a run of regular characters (a keyword or number).
66    pub fn token(&mut self) -> &'a [u8] {
67        let start = self.pos.min(self.data.len());
68        while self.peek().is_some_and(is_regular) {
69            self.pos += 1;
70        }
71        &self.data[start..self.pos]
72    }
73
74    /// `true` (and consumes it) if the next token is exactly `kw`.
75    pub fn eat_keyword(&mut self, kw: &[u8]) -> bool {
76        self.skip_ws();
77        let end = self.pos + kw.len();
78        if self.data.get(self.pos..end) == Some(kw) && self.data.get(end).is_none_or(|b| !is_regular(*b)) {
79            self.pos = end;
80            true
81        } else {
82            false
83        }
84    }
85
86    fn err(&self, what: &str) -> CosError {
87        CosError::Syntax { offset: self.pos, detail: what.to_string() }
88    }
89
90    /// Parse one direct object. References `N G R` are recognised.
91    pub fn object(&mut self) -> Result<Object, CosError> {
92        self.nodes = 0;
93        self.object_depth(0)
94    }
95
96    fn over_budget(&self) -> bool {
97        self.nodes > MAX_NODES
98    }
99
100    fn object_depth(&mut self, depth: usize) -> Result<Object, CosError> {
101        if depth > MAX_DEPTH {
102            return Err(self.err("nesting too deep"));
103        }
104        self.nodes = self.nodes.saturating_add(1);
105        if self.over_budget() {
106            return Err(self.err("object too large"));
107        }
108        self.skip_ws();
109        let Some(b) = self.peek() else { return Err(self.err("unexpected end of data")) };
110        match b {
111            b'/' => {
112                self.pos += 1;
113                Ok(Object::Name(self.name_body()))
114            }
115            b'(' => {
116                self.pos += 1;
117                Ok(Object::String(PdfString { bytes: self.literal_string()?, hex: false }))
118            }
119            b'<' if self.data.get(self.pos + 1) == Some(&b'<') => {
120                self.pos += 2;
121                Ok(Object::Dict(self.dict_body(depth)?))
122            }
123            b'<' => {
124                self.pos += 1;
125                Ok(Object::String(PdfString { bytes: self.hex_string()?, hex: true }))
126            }
127            b'[' => {
128                self.pos += 1;
129                let mut items = Vec::new();
130                loop {
131                    self.skip_ws();
132                    match self.peek() {
133                        None => return Err(self.err("unterminated array")),
134                        Some(b']') => {
135                            self.pos += 1;
136                            break;
137                        }
138                        _ => {
139                            let before = self.pos;
140                            match self.object_depth(depth + 1) {
141                                Ok(o) => items.push(o),
142                                Err(e) if self.over_budget() => return Err(e),
143                                // Tolerate junk inside arrays by skipping one byte.
144                                Err(_) if self.pos == before => self.pos += 1,
145                                Err(e) => return Err(e),
146                            }
147                        }
148                    }
149                }
150                Ok(Object::Array(items))
151            }
152            b'+' | b'-' | b'.' | b'0'..=b'9' => self.number_or_ref(),
153            _ => {
154                let start = self.pos;
155                let tok = self.token();
156                match tok {
157                    b"true" => Ok(Object::Bool(true)),
158                    b"false" => Ok(Object::Bool(false)),
159                    b"null" => Ok(Object::Null),
160                    b"" => {
161                        self.pos = start + 1;
162                        Err(CosError::Syntax { offset: start, detail: format!("unexpected byte 0x{b:02x}") })
163                    }
164                    other => Err(CosError::Syntax { offset: start, detail: format!("unexpected keyword {:?}", String::from_utf8_lossy(other)) }),
165                }
166            }
167        }
168    }
169
170    fn name_body(&mut self) -> Vec<u8> {
171        let mut out = Vec::new();
172        while let Some(b) = self.peek() {
173            if !is_regular(b) {
174                break;
175            }
176            self.pos += 1;
177            if b == b'#'
178                && let (Some(h), Some(l)) = (self.data.get(self.pos).and_then(|c| hexval(*c)), self.data.get(self.pos + 1).and_then(|c| hexval(*c)))
179            {
180                out.push(h << 4 | l);
181                self.pos += 2;
182                continue;
183            }
184            out.push(b);
185        }
186        out
187    }
188
189    fn literal_string(&mut self) -> Result<Vec<u8>, CosError> {
190        let mut out = Vec::new();
191        let mut depth = 1usize;
192        while let Some(b) = self.peek() {
193            self.pos += 1;
194            match b {
195                b'(' => {
196                    depth += 1;
197                    out.push(b);
198                }
199                b')' => {
200                    depth -= 1;
201                    if depth == 0 {
202                        return Ok(out);
203                    }
204                    out.push(b);
205                }
206                b'\\' => {
207                    let Some(e) = self.peek() else { break };
208                    self.pos += 1;
209                    match e {
210                        b'n' => out.push(b'\n'),
211                        b'r' => out.push(b'\r'),
212                        b't' => out.push(b'\t'),
213                        b'b' => out.push(8),
214                        b'f' => out.push(12),
215                        b'(' | b')' | b'\\' => out.push(e),
216                        b'\r' => {
217                            if self.peek() == Some(b'\n') {
218                                self.pos += 1;
219                            }
220                        }
221                        b'\n' => {}
222                        b'0'..=b'7' => {
223                            let mut v = (e - b'0') as u32;
224                            for _ in 0..2 {
225                                match self.peek() {
226                                    Some(d @ b'0'..=b'7') => {
227                                        v = v * 8 + (d - b'0') as u32;
228                                        self.pos += 1;
229                                    }
230                                    _ => break,
231                                }
232                            }
233                            out.push((v & 0xFF) as u8);
234                        }
235                        other => out.push(other),
236                    }
237                }
238                // End-of-line markers inside strings are normalised to \n (§7.3.4.2).
239                b'\r' => {
240                    if self.peek() == Some(b'\n') {
241                        self.pos += 1;
242                    }
243                    out.push(b'\n');
244                }
245                other => out.push(other),
246            }
247        }
248        Err(self.err("unterminated string"))
249    }
250
251    fn hex_string(&mut self) -> Result<Vec<u8>, CosError> {
252        let mut out = Vec::new();
253        let mut hi: Option<u8> = None;
254        while let Some(b) = self.peek() {
255            self.pos += 1;
256            if b == b'>' {
257                if let Some(h) = hi {
258                    out.push(h << 4);
259                }
260                return Ok(out);
261            }
262            if is_whitespace(b) {
263                continue;
264            }
265            let Some(v) = hexval(b) else { return Err(self.err("invalid hex digit")) };
266            match hi.take() {
267                None => hi = Some(v),
268                Some(h) => out.push(h << 4 | v),
269            }
270        }
271        Err(self.err("unterminated hex string"))
272    }
273
274    fn dict_body(&mut self, depth: usize) -> Result<Dict, CosError> {
275        let mut d = Dict::new();
276        loop {
277            self.skip_ws();
278            match self.peek() {
279                None => return Err(self.err("unterminated dictionary")),
280                Some(b'>') if self.data.get(self.pos + 1) == Some(&b'>') => {
281                    self.pos += 2;
282                    return Ok(d);
283                }
284                Some(b'/') => {
285                    self.pos += 1;
286                    let key = self.name_body();
287                    self.skip_ws();
288                    // A key directly followed by `>>` has no value: treat as null. (A following
289                    // `/` is a name *value*, so it cannot signal a missing value.)
290                    if self.peek() == Some(b'>') && self.data.get(self.pos + 1) == Some(&b'>') {
291                        d.set(key, Object::Null);
292                        continue;
293                    }
294                    let value = self.object_depth(depth + 1)?;
295                    d.set(key, value);
296                }
297                Some(_) => {
298                    // Tolerate garbage between entries (seen in damaged files): skip a token.
299                    let before = self.pos;
300                    let _ = self.object_depth(depth + 1);
301                    if self.over_budget() {
302                        return Err(self.err("object too large"));
303                    }
304                    if self.pos == before {
305                        self.pos += 1;
306                    }
307                }
308            }
309        }
310    }
311
312    fn number_or_ref(&mut self) -> Result<Object, CosError> {
313        let start = self.pos;
314        let first = self.number()?;
315        // Look ahead for "G R" to form a reference.
316        if let Object::Int(n) = first
317            && n >= 0
318        {
319            let save = self.pos;
320            self.skip_ws();
321            if self.peek().is_some_and(|b| b.is_ascii_digit()) {
322                let gstart = self.pos;
323                let g = self.token();
324                if g.iter().all(u8::is_ascii_digit) && self.eat_keyword(b"R") {
325                    let generation = std::str::from_utf8(g).ok().and_then(|s| s.parse::<u32>().ok()).unwrap_or(0);
326                    if n <= u32::MAX as i64 {
327                        return Ok(Object::Ref(ObjRef::new(n as u32, generation.min(u16::MAX as u32) as u16)));
328                    }
329                }
330                let _ = gstart;
331            }
332            self.pos = save;
333        }
334        let _ = start;
335        Ok(first)
336    }
337
338    fn number(&mut self) -> Result<Object, CosError> {
339        let start = self.pos;
340        let tok = self.token();
341        let s = std::str::from_utf8(tok).map_err(|_| self.err("bad number"))?;
342        if s.contains('.') {
343            // Tolerate forms like "--5", "5-" or "1.2.3" by parsing the longest valid prefix.
344            // Only a doubled sign needs a cleaned copy: a page of InDesign `/PieceInfo` is
345            // thousands of plain reals, and building a copy and a substring search for each
346            // was a large part of reading one (#5873).
347            let t = s.trim_start_matches('+');
348            let cleaned = if t.as_bytes().windows(2).any(|w| w == b"--") { std::borrow::Cow::Owned(t.replacen("--", "-", 1)) } else { t.into() };
349            let v = cleaned.get(..real_prefix_len(cleaned.as_bytes())).and_then(|p| parse_counted::<f64>(p).ok());
350            return Ok(Object::Real(v.unwrap_or(0.0)));
351        }
352        // Tolerate a doubled sign ("--280") like the real branch does.
353        let t = s.trim_start_matches('+');
354        let cleaned = t.strip_prefix('-').filter(|r| r.starts_with('-')).unwrap_or(t);
355        match cleaned.parse::<i64>() {
356            Ok(v) => Ok(Object::Int(v)),
357            Err(_) => {
358                // Out-of-range integers are treated as reals (§7.3.3 allows implementation limits).
359                cleaned.parse::<f64>().map(Object::Real).map_err(|_| CosError::Syntax { offset: start, detail: format!("bad number {s:?}") })
360            }
361        }
362    }
363}

365/// Length of the longest prefix of `s` that reads as a decimal number: a sign, digits with at
366/// most one `.`, and an exponent with digits. Zero when that prefix has no digit. Found in one
367/// pass so a long damaged token is converted once, not once per shorter prefix.
368fn real_prefix_len(s: &[u8]) -> usize {
369    let digits = |from: usize| s.get(from..).map_or(0, |r| r.iter().take_while(|b| b.is_ascii_digit()).count());
370    let mut at = usize::from(matches!(s.first(), Some(b'+' | b'-')));
371    let int = digits(at);
372    at += int;
373    let mut frac = 0;
374    if s.get(at) == Some(&b'.') {
375        frac = digits(at + 1);
376        at += 1 + frac;
377    }
378    if int + frac == 0 {
379        return 0;
380    }
381    if matches!(s.get(at), Some(b'e' | b'E')) {
382        let sign = usize::from(matches!(s.get(at + 1), Some(b'+' | b'-')));
383        let exp = digits(at + 1 + sign);
384        if exp > 0 {
385            at += 1 + sign + exp;
386        }
387    }
388    at
389}

399fn parse_counted<T: std::str::FromStr>(s: &str) -> Result<T, T::Err> {
400    #[cfg(test)]
401    NUMBER_BYTES.with(|c| c.set(c.get() + s.len()));
402    s.parse()
403}

405fn hexval(b: u8) -> Option<u8> {
406    match b {
407        b'0'..=b'9' => Some(b - b'0'),
408        b'a'..=b'f' => Some(b - b'a' + 10),
409        b'A'..=b'F' => Some(b - b'A' + 10),
410        _ => None,
411    }
412}

414/// Parse an indirect object at `offset`: `N G obj <object> [stream…endstream] endobj`.
415/// `resolve_length` resolves an indirect `/Length`. Returns the object and its (num, gen).
416pub fn parse_indirect(data: &[u8], offset: usize, resolve_length: &dyn Fn(ObjRef) -> Option<i64>) -> Result<(ObjRef, Object), CosError> {
417    parse_indirect_in(data, None, offset, resolve_length, None)
418}

432/// Where `endstream` occurs in a document's bytes, found once, the first time a stream's
433/// /Length is wrong or missing. Searching the rest of the file for each such stream instead
434/// makes many of them ending at one late `endstream` quadratic to read.
435pub(crate) type EndstreamIndex = std::sync::OnceLock<Vec<usize>>;
436
437fn endstream_offsets(data: &[u8]) -> Vec<usize> {
438    #[cfg(test)]
439    ENDSTREAM_SCAN.with(|c| c.set(c.get() + data.len()));
440    data.windows(9).enumerate().filter(|(i, w)| *w == b"endstream" && endstream_at(data, *i)).map(|(i, _)| i).collect()
441}

443fn parse_indirect_in(
444    data: &[u8],
445    shared: Option<&Arc<Vec<u8>>>,
446    offset: usize,
447    resolve_length: &dyn Fn(ObjRef) -> Option<i64>,
448    endstreams: Option<&EndstreamIndex>,
449) -> Result<(ObjRef, Object), CosError> {
450    let mut lx = Lexer::new(data, offset);
451    lx.skip_ws();
452    let num_tok = lx.token();
453    lx.skip_ws();
454    let gen_tok = lx.token();
455    let parse_u = |t: &[u8]| std::str::from_utf8(t).ok().and_then(|s| s.parse::<u64>().ok());
456    let (Some(num), Some(generation)) = (parse_u(num_tok), parse_u(gen_tok)) else {
457        return Err(CosError::Syntax { offset, detail: "expected object header".into() });
458    };
459    if !lx.eat_keyword(b"obj") {
460        return Err(CosError::Syntax { offset: lx.pos, detail: "expected 'obj'".into() });
461    }
462    let id = ObjRef::new(num.min(u32::MAX as u64) as u32, generation.min(u16::MAX as u64) as u16);
463    let obj = match lx.object() {
464        Ok(o) => o,
465        // "N G obj endobj" (empty) is treated as null.
466        Err(_) if lx.eat_keyword(b"endobj") => return Ok((id, Object::Null)),
467        Err(e) => return Err(e),
468    };
469    lx.skip_ws();
470    if let Object::Dict(dict) = &obj
471        && lx.eat_keyword(b"stream")
472    {
473        // The keyword is followed by CRLF or LF (a lone CR is tolerated). Microsoft Reporting
474        // Services writes a space first (`stream \r\n`); spaces and tabs count only when a line
475        // break follows, so they are never taken from the data.
476        let mut eol = lx.pos;
477        while matches!(data.get(eol), Some(b' ' | b'\t')) {
478            eol += 1;
479        }
480        if matches!(data.get(eol), Some(b'\r' | b'\n')) {
481            lx.pos = eol;
482        }
483        match (data.get(lx.pos), data.get(lx.pos + 1)) {
484            (Some(b'\r'), Some(b'\n')) => lx.pos += 2,
485            (Some(b'\n'), _) | (Some(b'\r'), _) => lx.pos += 1,
486            _ => {}
487        }
488        let start = lx.pos;
489        let declared = match dict.get(b"Length") {
490            Some(Object::Int(n)) => Some(*n),
491            Some(Object::Ref(r)) => resolve_length(*r),
492            _ => None,
493        };
494        // A /Length that fits in the file (not negative, and no wrap on a 32-bit target).
495        let fits = declared.and_then(|l| usize::try_from(l).ok()).filter(|l| start.checked_add(*l).is_some_and(|end| end <= data.len()));
496        // What follows the /Length bytes, white space aside.
497        let after = |len: usize| {
498            let mut after = Lexer::new(data, start.saturating_add(len));
499            after.skip_ws();
500            data.get(after.pos..).unwrap_or_default()
501        };
502        let len = match fits {
503            // The /Length ends at `endstream`, or where it would be: misspelt (`endstrea`), or
504            // left out before `endobj` or the end of the file. The stream is the bytes its /Length
505            // gives, "how many bytes of the PDF file are used for the stream's data" (ISO 32000-2
506            // §7.3.8.2), as other readers take it. A producer that wrote `endstrea` after every
507            // stream lost every font and every page's content, and searching on would run into a
508            // later object's `endstream`.
509            Some(l) if at_stream_end(after(l)) => l,
510            // Wrong or missing /Length: search for "endstream" (§7.3.8.1 recovery). With none
511            // left in the file, a /Length that fits is still the stream's extent.
512            _ => match find_endstream(data, start, endstreams) {
513                Some(end) => end - start,
514                None => fits.ok_or_else(|| CosError::Syntax { offset: start, detail: "stream without endstream".into() })?,
515            },
516        };
517        let range = start..start + len;
518        let raw = match shared {
519            Some(buf) => Bytes::view(buf, range),
520            None => data.get(range).map(|r| Bytes::from(r.to_vec())),
521        }
522        .ok_or_else(|| CosError::Syntax { offset: start, detail: "stream data past the end of the file".into() })?;
523        let mut dict = dict.clone();
524        dict.set(b"Length".to_vec(), Object::Int(raw.len() as i64));
525        return Ok((id, Object::Stream(Stream { dict, raw })));
526    }
527    Ok((id, obj))
528}

530/// Whether `rest`, what follows a stream's /Length bytes with white space skipped, is where the
531/// stream ends. A whole `endstream` token is: a token runs on through regular characters (ISO
532/// 32000-2 §7.2.3), so payload such as `endstreamer` is not one. A misspelt `endstream`
533/// (`endstrea`), an `endobj` with the `endstream` left out, or both, end it only where the object
534/// ends as well, before the end of the data, another object's `N G obj` header, or `xref`,
535/// `trailer` or `startxref`: payload such as `endobj more` or `endobject`, where a wrong /Length
536/// stops, is not, and the search for `endstream` runs instead. So does anything else but the end
537/// of the data.
538fn at_stream_end(rest: &[u8]) -> bool {
539    let mut lx = Lexer::new(rest, 0);
540    if lx.eat_keyword(b"endstream") {
541        return true;
542    }
543    let misspelt = lx.eat_keyword(b"endstrea");
544    let endobj = lx.eat_keyword(b"endobj");
545    lx.skip_ws();
546    if lx.pos >= rest.len() {
547        return true;
548    }
549    if !(misspelt || endobj) {
550        return false;
551    }
552    if [&b"xref"[..], b"trailer", b"startxref"].iter().any(|kw| Lexer::new(rest, lx.pos).eat_keyword(kw)) {
553        return true;
554    }
555    let number = |lx: &mut Lexer| {
556        lx.skip_ws();
557        let t = lx.token();
558        !t.is_empty() && t.iter().all(u8::is_ascii_digit)
559    };
560    number(&mut lx) && number(&mut lx) && lx.eat_keyword(b"obj")
561}

563/// Whether a whole `endstream` token starts at `at` in `data`: one followed by a regular
564/// character (`endstreamer`) is payload (ISO 32000-2 §7.2.3).
565fn endstream_at(data: &[u8], at: usize) -> bool {
566    data.get(at..).is_some_and(|d| d.starts_with(b"endstream")) && !data.get(at.saturating_add(9)).copied().is_some_and(is_regular)
567}

569/// Offset where stream data ends (before the EOL preceding "endstream").
570fn find_endstream(data: &[u8], from: usize, endstreams: Option<&EndstreamIndex>) -> Option<usize> {
571    let at = match endstreams {
572        Some(index) => {
573            let index = index.get_or_init(|| endstream_offsets(data));
574            index.get(index.partition_point(|&p| p < from)).copied()?
575        }
576        None => {
577            let hay = data.get(from..)?;
578            let found = hay.windows(9).enumerate().position(|(i, w)| w == b"endstream" && endstream_at(hay, i));
579            #[cfg(test)]
580            ENDSTREAM_SCAN.with(|c| c.set(c.get() + found.map_or(hay.len(), |p| p + 9)));
581            found? + from
582        }
583    };
584    let mut end = at;
585    if end > from && data[end - 1] == b'\n' {
586        end -= 1;
587    }
588    if end > from && data[end - 1] == b'\r' {
589        end -= 1;
590    }
591    Some(end)
592}
