29/// A string object. `hex` records how it was written, so it is re-emitted the same way.
30#[derive(Clone, Debug, PartialEq)]
31pub struct PdfString {
32    pub bytes: Vec<u8>,
33    pub hex: bool,
34}
35
36impl PdfString {
37    pub fn literal(bytes: impl Into<Vec<u8>>) -> Self {
38        Self { bytes: bytes.into(), hex: false }
39    }
40
41    /// A text string (ISO 32000-2 §7.9.2.2): PDFDocEncoding when possible, else UTF-16BE with BOM.
42    pub fn text(s: &str) -> Self {
43        if s.chars().all(|c| (' '..='~').contains(&c) || c == '\n' || c == '\r' || c == '\t') {
44            return Self::literal(s.as_bytes().to_vec());
45        }
46        let mut bytes = vec![0xFE, 0xFF];
47        for u in s.encode_utf16() {
48            bytes.extend_from_slice(&u.to_be_bytes());
49        }
50        Self { bytes, hex: false }
51    }
52
53    /// Decode a text string: UTF-16BE (BOM), UTF-8 (BOM, PDF 2.0) or PDFDocEncoding.
54    ///
55    /// Lenient beyond the spec, as readers are: invalid sequences after a BOM decode lossily
56    /// instead of dropping the string, a BOM-less string whose bytes are valid UTF-8 with
57    /// non-ASCII bytes (written by some producers) is read as UTF-8 rather than as
58    /// PDFDocEncoding mojibake, and stray leading U+FEFF (doubled BOMs) are removed.
59    pub fn to_text(&self) -> String {
60        self.text_chars().collect()
61    }
62
63    /// The characters [`PdfString::to_text`] decodes, one at a time, without building a string.
64    pub fn text_chars(&self) -> TextChars<'_> {
65        let decode = match self.bytes.as_slice() {
66            [0xFE, 0xFF, rest @ ..] => {
67                let units: fn(&[u8; 2]) -> u16 = |c| u16::from_be_bytes(*c);
68                Decode::Utf16(char::decode_utf16(rest.as_chunks::<2>().0.iter().map(units)))
69            }
70            [0xEF, 0xBB, 0xBF, rest @ ..] => Decode::Utf8Lossy { chunks: rest.utf8_chunks(), valid: "".chars(), replace: false },
71            b => match std::str::from_utf8(b) {
72                // Pure ASCII is identical either way; only non-ASCII UTF-8 needs the switch.
73                Ok(utf8) if !b.is_ascii() => Decode::Utf8(utf8.chars()),
74                _ => Decode::PdfDoc(b.iter()),
75            },
76        };
77        TextChars { decode, leading: true }
78    }
79
80    /// Whether the string decodes to `s` ([`PdfString::to_text`]), compared without allocating.
81    pub fn text_eq(&self, s: &str) -> bool {
82        self.text_chars().eq(s.chars())
83    }
84}

86/// The characters of a text string: [`PdfString::text_chars`].
87pub struct TextChars<'a> {
88    decode: Decode<'a>,
89    /// Still at the start, where stray U+FEFF (doubled BOMs) are dropped.
90    leading: bool,
91}
92
93/// The UTF-16BE code units after a BOM (an odd last byte is left out).
94type Utf16Units<'a> = std::iter::Map<std::slice::Iter<'a, [u8; 2]>, fn(&[u8; 2]) -> u16>;
95
96enum Decode<'a> {
97    Utf16(std::char::DecodeUtf16<Utf16Units<'a>>),
98    /// After a UTF-8 BOM: each invalid sequence is one U+FFFD, as `String::from_utf8_lossy` has it.
99    Utf8Lossy {
100        chunks: std::str::Utf8Chunks<'a>,
101        valid: std::str::Chars<'a>,
102        replace: bool,
103    },
104    Utf8(std::str::Chars<'a>),
105    PdfDoc(std::slice::Iter<'a, u8>),
106}
107
108impl Iterator for Decode<'_> {
109    type Item = char;
110
111    fn next(&mut self) -> Option<char> {
112        match self {
113            Decode::Utf16(units) => units.next().map(|u| u.unwrap_or(char::REPLACEMENT_CHARACTER)),
114            Decode::Utf8Lossy { chunks, valid, replace } => loop {
115                if let Some(c) = valid.next() {
116                    return Some(c);
117                }
118                if *replace {
119                    *replace = false;
120                    return Some(char::REPLACEMENT_CHARACTER);
121                }
122                let chunk = chunks.next()?;
123                *valid = chunk.valid().chars();
124                *replace = !chunk.invalid().is_empty();
125            },
126            Decode::Utf8(chars) => chars.next(),
127            Decode::PdfDoc(bytes) => bytes.next().map(|&c| crate::crypt::pdfdoc_char(c)),
128        }
129    }
130}
131
132impl Iterator for TextChars<'_> {
133    type Item = char;
134
135    fn next(&mut self) -> Option<char> {
136        loop {
137            let c = self.decode.next()?;
138            if self.leading && c == '\u{FEFF}' {
139                continue;
140            }
141            self.leading = false;
142            return Some(c);
143        }
144    }
145}

147/// A name object (without the leading `/`, with `#xx` escapes already decoded).
148pub type Name = Vec<u8>;
149
150#[cfg(test)]
151thread_local! {
152    /// Key comparisons and index lookups made by [`Dict`] on this thread.
153    static KEY_COMPARES: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
154}
155
156/// Entries from which a [`Dict`] keeps a key index. Smaller dictionaries (nearly all of them)
157/// search their entries; a parsed file controls the key count, and a linear search per
158/// inserted key would make a dictionary of many distinct keys quadratic to build.
159const INDEX_FROM: usize = 32;
160
161/// An ordered dictionary.
162#[derive(Clone, Default)]
163pub struct Dict {
164    entries: Vec<(Name, Object)>,
165    /// Key to position in `entries`, kept once the dictionary reaches [`INDEX_FROM`] entries.
166    /// Boxed so that every `Dict`, and so every `Object`, grows by one pointer, not by a map.
167    #[allow(clippy::box_collection)]
168    index: Option<Box<std::collections::HashMap<Name, usize>>>,
169}
170
171impl PartialEq for Dict {
172    fn eq(&self, other: &Self) -> bool {
173        self.entries == other.entries
174    }
175}
176
177impl std::fmt::Debug for Dict {
178    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
179        f.debug_struct("Dict").field("entries", &self.entries).finish()
180    }
181}
182
183impl Dict {
184    pub fn new() -> Self {
185        Self::default()
186    }
187
188    fn position(&self, key: &[u8]) -> Option<usize> {
189        if let Some(index) = &self.index {
190            #[cfg(test)]
191            KEY_COMPARES.with(|c| c.set(c.get() + 1));
192            return index.get(key).copied();
193        }
194        self.entries.iter().position(|(k, _)| {
195            #[cfg(test)]
196            KEY_COMPARES.with(|c| c.set(c.get() + 1));
197            k == key
198        })
199    }
200
201    pub fn get(&self, key: &[u8]) -> Option<&Object> {
202        self.position(key).and_then(|i| self.entries.get(i)).map(|(_, v)| v)
203    }
204
205    pub fn get_mut(&mut self, key: &[u8]) -> Option<&mut Object> {
206        let i = self.position(key)?;
207        self.entries.get_mut(i).map(|(_, v)| v)
208    }
209
210    /// Insert or replace, keeping the position of an existing key.
211    pub fn set(&mut self, key: impl Into<Name>, value: impl Into<Object>) {
212        let key = key.into();
213        let value = value.into();
214        match self.position(&key).and_then(|i| self.entries.get_mut(i)) {
215            Some((_, v)) => *v = value,
216            None => {
217                let at = self.entries.len();
218                if let Some(index) = &mut self.index {
219                    index.insert(key.clone(), at);
220                } else if at + 1 >= INDEX_FROM {
221                    let index = self.entries.iter().enumerate().map(|(i, (k, _))| (k.clone(), i)).chain([(key.clone(), at)]).collect();
222                    self.index = Some(Box::new(index));
223                }
224                self.entries.push((key, value));
225            }
226        }
227    }
228
229    pub fn remove(&mut self, key: &[u8]) -> Option<Object> {
230        let i = self.position(key)?;
231        if let Some(index) = &mut self.index {
232            index.remove(key);
233            index.values_mut().filter(|at| **at > i).for_each(|at| *at -= 1);
234        }
235        Some(self.entries.remove(i).1)
236    }
237
238    pub fn contains(&self, key: &[u8]) -> bool {
239        self.get(key).is_some()
240    }
241
242    pub fn iter(&self) -> impl Iterator<Item = (&Name, &Object)> {
243        self.entries.iter().map(|(k, v)| (k, v))
244    }
245
246    pub fn len(&self) -> usize {
247        self.entries.len()
248    }
249
250    pub fn is_empty(&self) -> bool {
251        self.entries.is_empty()
252    }
253
254    pub fn name(&self, key: &[u8]) -> Option<&[u8]> {
255        self.get(key).and_then(Object::as_name)
256    }
257
258    pub fn int(&self, key: &[u8]) -> Option<i64> {
259        self.get(key).and_then(Object::as_int)
260    }
261
262    pub fn reference(&self, key: &[u8]) -> Option<ObjRef> {
263        self.get(key).and_then(Object::as_ref)
264    }
265}
266
267impl FromIterator<(Name, Object)> for Dict {
268    fn from_iter<T: IntoIterator<Item = (Name, Object)>>(iter: T) -> Self {
269        let mut d = Dict::new();
270        for (k, v) in iter {
271            d.set(k, v);
272        }
273        d
274    }
275}

277/// A stream: dictionary plus its data *as stored* (still filtered).
278#[derive(Clone, Debug, PartialEq)]
279pub struct Stream {
280    pub dict: Dict,
281    pub raw: Bytes,
282}

284/// Upper bound for a single decoded stream (decompression-bomb defence).
285pub const MAX_DECODED: usize = 1 << 30;
286
287/// The most an image's `/JBIG2Globals` stream may decode to: the symbol and pattern dictionaries an
288/// encoder shares between pages, themselves coded, so far less than any stream may decode to
289/// ([`MAX_DECODED`]). They are decoded before the image's JBIG2 page is read, so the page's size
290/// can't bound them: they have a limit of their own.
291pub const MAX_JBIG2_GLOBALS: usize = 32 << 20;
292
293impl Stream {
294    /// A new stream from already-encoded bytes (`/Length` is set at write time).
295    pub fn from_raw(dict: Dict, raw: Vec<u8>) -> Self {
296        Self { dict, raw: raw.into() }
297    }

299    /// A new stream holding `data` compressed with Flate.
300    pub fn flate(mut dict: Dict, data: &[u8]) -> Self {
301        dict.set(b"Filter".to_vec(), Object::Name(b"FlateDecode".to_vec()));
302        dict.remove(b"DecodeParms");
303        Self { dict, raw: crate::filters::encode_flate(data).into() }
304    }

306    /// The filter chain declared in the dictionary.
307    pub fn filters(&self) -> Vec<(Filter, Params)> {
308        let names: Vec<&[u8]> = match self.dict.get(b"Filter") {
309            Some(Object::Name(n)) => vec![n],
310            Some(Object::Array(a)) => a.iter().filter_map(Object::as_name).collect(),
311            _ => Vec::new(),
312        };
313        let parms: Vec<Option<&Dict>> = match self.dict.get(b"DecodeParms") {
314            Some(Object::Dict(d)) => vec![Some(d)],
315            Some(Object::Array(a)) => a.iter().map(Object::as_dict).collect(),
316            _ => Vec::new(),
317        };
318        names
319            .into_iter()
320            .enumerate()
321            // `/Crypt` is applied by the security handler when the object is loaded (§7.4.10).
322            .filter(|(_, n)| *n != b"Crypt")
323            .map(|(i, n)| {
324                let p = parms.get(i).copied().flatten();
325                let get = |k: &[u8], d: i64| p.and_then(|p| p.int(k)).unwrap_or(d);
326                let params = Params {
327                    predictor: get(b"Predictor", 1),
328                    colors: get(b"Colors", 1),
329                    bits_per_component: get(b"BitsPerComponent", 8),
330                    columns: get(b"Columns", 1),
331                    early_change: get(b"EarlyChange", 1),
332                };
333                (Filter::from_name(n), params)
334            })
335            .collect()
336    }
337
338    /// [`Stream::filters`], or an error when `/Filter` is present but is not a name or an
339    /// array of names (an indirect reference, say): the data is not unfiltered, so it must not
340    /// read as decoded.
341    pub(crate) fn readable_filters(&self) -> Result<Vec<(Filter, Params)>, crate::filters::FilterError> {
342        let readable = match self.dict.get(b"Filter") {
343            None | Some(Object::Null | Object::Name(_)) => true,
344            Some(Object::Array(a)) => a.iter().all(|f| f.as_name().is_some()),
345            Some(_) => false,
346        };
347        if !readable {
348            return Err(crate::filters::FilterError::Unsupported("/Filter is not a name or an array of names".into()));
349        }
350        Ok(self.filters())
351    }
352
353    /// Decoded data, tolerating truncated or corrupt encodings the way viewers do (the data
354    /// decoded before the damage is returned). Decoding past
355    /// [`crate::cos::budget::MAX_PAGE_DECODED`] (256 MiB), the most the renderer reads from one
356    /// stream, is an error. Streams with image codecs fail with
357    /// `CosError::Filter` (keep them encoded). Use `decoded_strict` to detect damage.
358    pub fn decoded(&self) -> Result<Vec<u8>, CosError> {
359        self.decoded_within(crate::cos::budget::MAX_PAGE_DECODED)
360    }
361
362    /// Like [`Stream::decoded`], for a stream whose decoded size has a tighter bound than
363    /// [`MAX_DECODED`]: decoding past `max` bytes is an error.
364    pub fn decoded_within(&self, max: usize) -> Result<Vec<u8>, CosError> {
365        let decode =
366            |chain: &[(Filter, Params)], data: &[u8], max| crate::filters::decode_tolerant(chain, data, max).map(|(v, _)| (v, 0)).map_err(|e| (e, 0));
367        self.decode_chain(max, decode).map(|(v, _)| v).map_err(|(e, _)| CosError::Filter(e.to_string()))
368    }
369
370    /// [`Stream::decoded_within`] for a caller charging an allowance: the filter error is kept
371    /// typed, so a caller can tell a spent limit from corrupt data, `max` bounds all filter
372    /// stages together, and the result gives whether the data stopped at damage (it is then what
373    /// decoded before it) and the bytes the stages decoded (the work done, on success and on
374    /// failure).
375    pub(crate) fn decoded_within_counted(&self, max: usize) -> Result<(Vec<u8>, bool, usize), (crate::filters::FilterError, usize)> {
376        let mut damaged = false;
377        let decode = |chain: &[(Filter, Params)], data: &[u8], max| {
378            crate::filters::decode_tolerant_counted(chain, data, max).map(|(v, partial, spent)| {
379                damaged = partial;
380                (v, spent)
381            })
382        };
383        self.decode_chain(max, decode).map(|(v, spent)| (v, damaged, spent))
384    }
385
386    /// [`Stream::decoded_within_counted`] with every filter stage also held to `each`, as
387    /// [`Stream::decoded_within`] holds it, while `max` bounds the stages together
388    /// ([`crate::filters::decode_tolerant_counted_each`]). Without filters the data is held to
389    /// both.
390    pub(crate) fn decoded_within_each_counted(
391        &self,
392        each: usize,
393        max: usize,
394    ) -> Result<(Vec<u8>, bool, usize), (crate::filters::FilterError, usize)> {
395        let mut damaged = false;
396        let decode = |chain: &[(Filter, Params)], data: &[u8], max| {
397            crate::filters::decode_tolerant_counted_each(chain, data, each.min(MAX_DECODED), max).map(|(v, partial, spent)| {
398                damaged = partial;
399                (v, spent)
400            })
401        };
402        let max = if self.readable_filters().is_ok_and(|chain| chain.is_empty()) { max.min(each) } else { max };
403        self.decode_chain(max, decode).map(|(v, spent)| (v, damaged, spent))
404    }
405
406    /// [`Stream::decoded_within`] spending from `work`, what an operation may still do across a
407    /// document's images ([`crate::filters::decode_tolerant_within`]).
408    pub(crate) fn decoded_spending(&self, max: usize, work: &mut u64) -> Result<Vec<u8>, CosError> {
409        let filter = |e: crate::filters::FilterError| CosError::Filter(e.to_string());
410        let chain = self.readable_filters().map_err(filter)?;
411        crate::filters::decode_tolerant_within(&chain, &self.raw, max.min(MAX_DECODED), work).map_err(filter)
412    }
413
414    /// [`Stream::decoded_spending`] for the samples of an image `rows` high (its `/Height`):
415    /// [`crate::filters::decode_image_tolerant_within`].
416    pub(crate) fn decoded_image_rows_spending(&self, max: usize, rows: u32, work: &mut u64) -> Result<Vec<u8>, CosError> {
417        let filter = |e: crate::filters::FilterError| CosError::Filter(e.to_string());
418        let chain = self.readable_filters().map_err(filter)?;
419        crate::filters::decode_image_tolerant_within(&chain, &self.raw, max.min(MAX_DECODED), rows, work).map_err(filter)
420    }

422    /// The stream decoded through its filters by `decode`, or copied when it has none, within
423    /// `max` and [`MAX_DECODED`].
424    fn decode_chain(
425        &self,
426        max: usize,
427        decode: impl FnOnce(&[(Filter, Params)], &[u8], usize) -> Result<(Vec<u8>, usize), (crate::filters::FilterError, usize)>,
428    ) -> Result<(Vec<u8>, usize), (crate::filters::FilterError, usize)> {
429        let max = max.min(MAX_DECODED);
430        let chain = self.readable_filters().map_err(|e| (e, 0))?;
431        if chain.is_empty() {
432            return self
433                .raw_within_typed(max)
434                .map(|v| {
435                    let n = v.len();
436                    (v, n)
437                })
438                .map_err(|e| (e, 0));
439        }
440        decode(&chain, &self.raw, max)
441    }
442
443    /// Decoded data, bounded by [`MAX_DECODED`]; any corruption is an error.
444    pub fn decoded_strict(&self) -> Result<Vec<u8>, CosError> {
445        self.decoded_strict_within(MAX_DECODED)
446    }
447
448    /// Like [`Stream::decoded_strict`], bounded by `max` (and [`MAX_DECODED`]): for data that
449    /// will be copied elsewhere, where a damaged stream must not pass for a shorter one.
450    pub fn decoded_strict_within(&self, max: usize) -> Result<Vec<u8>, CosError> {
451        let decode = |chain: &[(Filter, Params)], data: &[u8], max| crate::filters::decode(chain, data, max).map(|v| (v, 0)).map_err(|e| (e, 0));
452        self.decode_chain(max, decode).map(|(v, _)| v).map_err(|(e, _)| CosError::Filter(e.to_string()))
453    }
454
455    /// [`Stream::decoded_strict`] bounded by `max` and counted like
456    /// [`Stream::decoded_within_counted`].
457    pub(crate) fn decoded_strict_within_counted(&self, max: usize) -> Result<(Vec<u8>, usize), (crate::filters::FilterError, usize)> {
458        self.decode_chain(max, crate::filters::decode_counted)
459    }
460
461    /// Data after the first filter alone, bounded by `max`: the rest of the chain stays encoded.
462    /// Any corruption in that first filter is an error.
463    pub fn decoded_first_filter_strict(&self, max: usize) -> Result<Vec<u8>, CosError> {
464        let mut chain = self.readable_filters().map_err(|e| CosError::Filter(e.to_string()))?;
465        chain.truncate(1);
466        if chain.is_empty() {
467            return self.raw_within(max);
468        }
469        crate::filters::decode(&chain, &self.raw, max.min(MAX_DECODED)).map_err(|e| CosError::Filter(e.to_string()))
470    }
471
472    fn raw_within(&self, max: usize) -> Result<Vec<u8>, CosError> {
473        self.raw_within_typed(max).map_err(|e| CosError::Filter(e.to_string()))
474    }
475
476    fn raw_within_typed(&self, max: usize) -> Result<Vec<u8>, crate::filters::FilterError> {
477        let max = max.min(MAX_DECODED);
478        // An empty decoder chain still produces an owned output buffer: check its limit
479        // before cloning, including streams whose /Crypt filter was applied on load.
480        if self.raw.len() > max {
481            return Err(crate::filters::FilterError::LimitExceeded(max));
482        }
483        Ok(self.raw.to_vec())
484    }

641    #[test]
642    fn parsing_many_distinct_keys_is_not_quadratic() {
643        let n = 20_000;
644        let body: String = (0..n).map(|i| format!("/K{i} {i} ")).collect();
645        let text = format!("<< {body}/K7 true >>");
646        KEY_COMPARES.with(|c| c.set(0));
647        let Ok(Object::Dict(d)) = crate::cos::parser::Lexer::new(text.as_bytes(), 0).object() else { panic!("not a dict") };
648        let compares = KEY_COMPARES.with(|c| c.get());
649        assert!(compares < 8 * n, "{compares} key comparisons for {n} keys");
650        // Order and replace-in-place semantics stay those of a plain ordered dictionary.
651        assert_eq!(d.len(), n);
652        assert_eq!(d.iter().next().map(|(k, _)| k.as_slice()), Some(&b"K0"[..]));
653        assert_eq!(d.get(b"K7"), Some(&Object::Bool(true)));
654        assert_eq!(d.iter().nth(7).map(|(_, v)| v), Some(&Object::Bool(true)));
655        let mut d = d;
656        assert_eq!(d.remove(b"K3"), Some(Object::Int(3)));
657        assert_eq!(d.get(b"K4"), Some(&Object::Int(4)));
658        assert_eq!(d.iter().nth(3).map(|(k, _)| k.as_slice()), Some(&b"K4"[..]));
659        d.set(b"K3".to_vec(), Object::Null);
660        assert_eq!(d.iter().last().map(|(k, _)| k.as_slice()), Some(&b"K3"[..]));
661        assert!(d.get(b"K20000").is_none());
662    }

672    #[test]
673    fn text_string_encodings() {
674        // UTF-16BE with BOM.
675        let mut utf16 = vec![0xFE, 0xFF];
676        utf16.extend("第一章 概述".encode_utf16().flat_map(u16::to_be_bytes));
677        assert_eq!(PdfString::literal(utf16).to_text(), "第一章 概述");
678        // UTF-8 with BOM (PDF 2.0): the BOM is not part of the text.
679        let mut utf8 = vec![0xEF, 0xBB, 0xBF];
680        utf8.extend_from_slice("第一章".as_bytes());
681        assert_eq!(PdfString::literal(utf8).to_text(), "第一章");
682        // Invalid UTF-8 after the BOM decodes lossily instead of vanishing.
683        assert_eq!(PdfString::literal(vec![0xEF, 0xBB, 0xBF, b'A', 0xFF, b'B']).to_text(), "A\u{FFFD}B");
684        // BOM-less raw UTF-8 from producers that skip the BOM.
685        assert_eq!(PdfString::literal("目录".as_bytes()).to_text(), "目录");
686        // PDFDocEncoding Latin-1 (not valid UTF-8) stays PDFDocEncoding.
687        assert_eq!(PdfString::literal(vec![b'C', b'a', b'f', 0xE9]).to_text(), "Café");
688        // Doubled BOMs leave no U+FEFF behind.
689        assert_eq!(PdfString::literal(vec![0xFE, 0xFF, 0xFE, 0xFF, 0x00, b'A']).to_text(), "A");
690        assert_eq!(PdfString::literal(vec![0xEF, 0xBB, 0xBF, 0xEF, 0xBB, 0xBF, b'A']).to_text(), "A");
691        // Odd trailing byte and empty strings never panic.
692        assert_eq!(PdfString::literal(vec![0xFE, 0xFF, 0x00, b'A', 0x00]).to_text(), "A");
693        assert_eq!(PdfString::literal(vec![0xFE, 0xFF]).to_text(), "");
694        assert_eq!(PdfString::literal(Vec::new()).to_text(), "");
695    }

697    #[test]
698    fn text_eq_compares_what_to_text_decodes() {
699        let utf16 = |s: &str| [0xFE, 0xFF].into_iter().chain(s.encode_utf16().flat_map(u16::to_be_bytes)).collect::<Vec<u8>>();
700        // Each invalid UTF-8 sequence is one U+FFFD, as String::from_utf8_lossy has it.
701        let lossy = [0xEF, 0xBB, 0xBF, 0xC3, 0x28, b'x', 0xF0, 0x9F, 0x92, 0xFF];
702        assert_eq!(PdfString::literal(lossy.to_vec()).to_text(), String::from_utf8_lossy(&lossy[3..]));
703        let cases: Vec<Vec<u8>> = vec![
704            Vec::new(),
705            b"Plain".to_vec(),
706            vec![b'C', b'a', b'f', 0xE9, 0x18],
707            "目录".as_bytes().to_vec(),
708            utf16("第一章"),
709            vec![0xFE, 0xFF, 0xD8, 0x00, 0x00, b'A'],
710            vec![0xFE, 0xFF, 0xFE, 0xFF, 0x00, b'A', 0xFE, 0xFF],
711            lossy.to_vec(),
712        ];
713        for bytes in cases {
714            let s = PdfString::literal(bytes);
715            let text = s.to_text();
716            assert!(s.text_eq(&text), "{text:?}");
717            assert!(!s.text_eq(&format!("{text}x")), "{text:?}");
718            assert!(text.is_empty() || !s.text_eq(&text[..text.len() - text.chars().last().map_or(0, char::len_utf8)]), "{text:?}");
719        }
720        // A U+FEFF after the first character is text and stays.
721        assert!(PdfString::literal(utf16("A\u{FEFF}")).text_eq("A\u{FEFF}"));
722    }
