1//! Serialization and saving (ISO 32000-2 §7.5).
2//!
3//! * `write_incremental` appends the edited objects, a cross-reference section in the same style
4//!   as the file's last one (table or stream) and a trailer with `/Prev`. The original bytes are
5//!   never modified, which keeps digital signatures over earlier revisions valid. With
6//!   `SaveOptions::drop_unlinked`, objects added since the file was read that nothing refers to
7//!   any more are left out (written as free).
8//! * `write_full` writes only objects reachable from the trailer, renumbered densely, with a
9//!   cross-reference table or stream — the "Save As" / garbage-collecting path. Unknown trailer
10//!   entries are copied, including objects reachable only from them.
11

21/// Serialize one object. Deterministic: same object → same bytes.
22pub fn serialize(o: &Object, out: &mut Vec<u8>) {
23    match o {
24        Object::Null => out.extend_from_slice(b"null"),
25        Object::Bool(b) => out.extend_from_slice(if *b { b"true" } else { b"false" }),
26        Object::Int(i) => write_int(*i, out),
27        Object::Real(r) => write_real(*r, out),
28        Object::String(s) => write_string(s, out),
29        Object::Name(n) => write_name(n, out),
30        Object::Array(a) => {
31            out.push(b'[');
32            for (i, x) in a.iter().enumerate() {
33                if i > 0 {
34                    out.push(b' ');
35                }
36                serialize(x, out);
37            }
38            out.push(b']');
39        }
40        Object::Dict(d) => write_dict(d, out),
41        Object::Stream(s) => write_stream(s, out),
42        Object::Ref(r) => {
43            write_digits(u64::from(r.num), out);
44            out.push(b' ');
45            write_digits(u64::from(r.generation), out);
46            out.extend_from_slice(b" R");
47        }
48    }
49}

51/// Upper-case hexadecimal digits.
52const HEX: &[u8; 16] = b"0123456789ABCDEF";

54/// The decimal digits of `v`, the same bytes `write!(out, "{v}")` gives, without the formatter.
55fn write_digits(mut v: u64, out: &mut Vec<u8>) {
56    // `u64::MAX` has 20 digits, so `at` never goes below 0.
57    let mut buf = [0u8; 20];
58    let mut at = buf.len();
59    loop {
60        at -= 1;
61        buf[at] = b'0' + (v % 10) as u8;
62        v /= 10;
63        if v == 0 {
64            break;
65        }
66    }
67    out.extend_from_slice(&buf[at..]);
68}

70/// A signed integer in decimal, as `write!(out, "{i}")` writes it.
71fn write_int(i: i64, out: &mut Vec<u8>) {
72    if i < 0 {
73        out.push(b'-');
74    }
75    write_digits(i.unsigned_abs(), out);
76}

78/// `b` as two upper-case hexadecimal digits.
79fn hex_pair(b: u8) -> [u8; 2] {
80    [HEX[usize::from(b >> 4)], HEX[usize::from(b & 15)]]
81}

83fn write_real(r: f64, out: &mut Vec<u8>) {
84    if !r.is_finite() {
85        out.push(b'0');
86        return;
87    }
88    // A whole number is written as an integer only within the integer range Annex C gives
89    // (±2³¹−1), so `4e9` or `1e30` stays a real token for readers with 32-bit integers.
90    if r.fract() == 0.0 && r.abs() <= f64::from(i32::MAX) {
91        write_int(i64::from(r as i32), out);
92        return;
93    }
94    // Plain decimal notation (§7.3.3 has no exponents). Rust's `Display` for f64 prints the
95    // shortest string that parses back to the same value, never in exponent form, so values
96    // round-trip exactly (rounding to a fixed number of places visibly shifted shading colours).
97    let s = format!("{r}");
98    out.extend_from_slice(if s == "-0" { b"0" } else { s.as_bytes() });
99    // A whole number past that range prints without a point; the trailing `.` keeps it a real.
100    if !s.contains('.') && s != "-0" {
101        out.push(b'.');
102    }
103}

105fn write_name(n: &[u8], out: &mut Vec<u8>) {
106    out.push(b'/');
107    for &b in n {
108        let regular = (b'!'..=b'~').contains(&b) && !matches!(b, b'(' | b')' | b'<' | b'>' | b'[' | b']' | b'{' | b'}' | b'/' | b'%' | b'#');
109        if regular {
110            out.push(b);
111        } else {
112            out.push(b'#');
113            out.extend_from_slice(&hex_pair(b));
114        }
115    }
116}

118fn write_string(s: &PdfString, out: &mut Vec<u8>) {
119    let binary = s.bytes.iter().filter(|b| !(b' '..=b'~').contains(*b) && !matches!(b, b'\n' | b'\r' | b'\t')).count();
120    if s.hex || binary * 4 > s.bytes.len().max(1) {
121        out.reserve(s.bytes.len().saturating_mul(2).saturating_add(2));
122        out.push(b'<');
123        for &b in &s.bytes {
124            out.extend_from_slice(&hex_pair(b));
125        }
126        out.push(b'>');
127        return;
128    }
129    out.push(b'(');
130    for &b in &s.bytes {
131        match b {
132            b'(' | b')' | b'\\' => {
133                out.push(b'\\');
134                out.push(b);
135            }
136            b'\n' => out.extend_from_slice(b"\\n"),
137            b'\r' => out.extend_from_slice(b"\\r"),
138            b'\t' => out.extend_from_slice(b"\\t"),
139            0x08 => out.extend_from_slice(b"\\b"),
140            0x0C => out.extend_from_slice(b"\\f"),
141            b' '..=b'~' => out.push(b),
142            _ => {
143                out.extend_from_slice(&[b'\\', b'0' + (b >> 6), b'0' + ((b >> 3) & 7), b'0' + (b & 7)]);
144            }
145        }
146    }
147    out.push(b')');
148}

150fn write_dict(d: &Dict, out: &mut Vec<u8>) {
151    out.extend_from_slice(b"<<");
152    for (k, v) in d.iter() {
153        write_name(k, out);
154        out.push(b' ');
155        serialize(v, out);
156    }
157    out.extend_from_slice(b">>");
158}

160fn write_stream(s: &Stream, out: &mut Vec<u8>) {
161    let mut d = s.dict.clone();
162    d.set(b"Length".to_vec(), Object::Int(s.raw.len() as i64));
163    write_dict(&d, out);
164    out.extend_from_slice(b"\nstream\n");
165    out.extend_from_slice(&s.raw);
166    out.extend_from_slice(b"\nendstream");
167}

169/// A document opened without the password its embedded files need holds their data as stored and
170/// can't encrypt: a save keeps its `/Encrypt` as it is, and writes no edited or added stream that
171/// would have to be encrypted (an attached file put in, say), which would be written in the clear.
172fn keyless_check(doc: &Document) -> Result<(), CosError> {
173    if !doc.embedded_files_locked() {
174        return Ok(());
175    }
176    let refuse =
177        |what: &str| Err(CosError::Security(format!("{what} without the password of the document's embedded files: open it with the password")));
178    if doc.encryption_changed() || doc.trailer().get(b"Encrypt").is_none() {
179        return refuse("the document's security can't change");
180    }
181    if doc.overlay_entries().any(|(_, _, o)| matches!(o.as_deref(), Some(Object::Stream(s)) if doc.reads_as_stored(s))) {
182        return refuse("an edited or added embedded file can't be encrypted");
183    }
184    Ok(())
185}

187/// The object as written: encrypted with its (output) number when the document is encrypted,
188/// except the `/Encrypt` dictionary itself (`source_num` is its number in the source document).
189/// `o` has the document's numbers in its references ([`crate::cos::security::transform`]).
190fn prepared<'a>(doc: &Document, source_num: u32, num: u32, generation: u16, o: &'a Object) -> Cow<'a, Object> {
191    match doc.output_security() {
192        (Some(h), encrypt_num) if Some(source_num) != encrypt_num && !doc.stored_as_is(source_num) => {
193            let pass = if doc.encryption_changed() { crate::cos::security::Pass::EncryptReplacing } else { crate::cos::security::Pass::Encrypt };
194            Cow::Owned(crate::cos::security::transform(doc, h, o, num, generation, pass))
195        }
196        _ => Cow::Borrowed(o),
197    }
198}

200fn write_indirect(num: u32, generation: u16, o: &Object, out: &mut Vec<u8>) {
201    let _ = writeln!(out, "{num} {generation} obj");
202    serialize(o, out);
203    out.extend_from_slice(b"\nendobj\n");
204}

206/// Options for saving.
207#[derive(Clone, Debug)]
208pub struct SaveOptions {
209    /// Replace or add `/Info /ModDate` with this PDF date string (e.g. from `pdf_date`).
210    pub mod_date: Option<String>,
211    /// Seed for a new file identifier when the document has none (deterministic in tests).
212    pub id_seed: u64,
213    /// Full saves: pack non-stream objects into compressed object streams with a cross-reference
214    /// stream (PDF 1.5+, §7.5.7–7.5.8). Off writes a classic table that any reader accepts.
215    pub object_streams: bool,
216    /// Incremental saves: leave out what was added since the file was read and nothing refers to
217    /// any more, such as a comment added and deleted again or a redaction mark added and cleared
218    /// (written as free). Off by default, because a writer may add an object only it finds again
219    /// by scanning, as a standalone document timestamp is; such a writer frees the leftovers
220    /// before adding it, with [`Document::free_unlinked_additions`]. Full saves always leave
221    /// these out.
222    pub drop_unlinked: bool,
223    /// Entropy for the file identifier of a rewrite that asks for a new one (see
224    /// [`Document::require_full_save_with_new_id`]). Given, the identifier depends only on this and
225    /// on the written bytes, so the same save gives the same file; tests use it to pin output.
226    /// Without it, fresh entropy from the host (the operating system on native targets, the
227    /// browser on WebAssembly) is mixed in.
228    pub id_entropy: Option<[u8; 16]>,
229}

231impl Default for SaveOptions {

233        Self { mod_date: None, id_seed: 0x5052_494E_5443_5241, object_streams: true, drop_unlinked: false, id_entropy: None }
234    }

254/// Append the document's edits to its original bytes (incremental update, §7.5.6).
255///
256/// A document that had to be reconstructed (no usable cross-reference chain) cannot be updated
257/// incrementally — other readers would follow the broken chain — so it is rewritten in full,
258/// which also repairs the file (what users expect after "the file was damaged and repaired").
259pub fn write_incremental(doc: &Document, opts: &SaveOptions) -> Result<Vec<u8>, CosError> {
260    if doc.stream_limited() {
261        return Err(CosError::ReadOnlyLimit);
262    }
263    keyless_check(doc)?;
264    // Reconstructed files have no chain to append to; added or removed encryption must
265    // rewrite every object; redaction must not leave the old revision behind. All need a full save.
266    if doc.reconstructed() || doc.encryption_changed() || doc.full_save_required() {
267        return write_full(doc, opts);
268    }
269    let mut doc = doc.clone();
270    stamp_mod_date(&mut doc, opts);
271    let original = doc.bytes().clone();
272    // The revision is written on its own and joined to the original at the end, so the result is
273    // allocated once at its exact size: growing a buffer that holds a large original past a
274    // guessed capacity would copy the whole file again and keep up to as much again unused.
275    let newline = !matches!(original.last(), Some(b'\n' | b'\r'));
276    // A file with bytes before its `%PDF-` header may count its offsets from the header. The new
277    // section then has to as well: readers take `/Prev` and `startxref` in one convention, and a
278    // mix sends them into reconstruction (#2364).
279    let shift = if doc.revisions().last().is_some_and(|r| r.header_relative) { doc.header_offset() } else { 0 };
280    let base = (original.len() + usize::from(newline)).saturating_sub(shift) as u64;
281    let mut out = Vec::new();
282    let mut offsets: BTreeMap<u32, (u64, u16, bool)> = BTreeMap::new(); // num → (offset, gen, in use)
283    let unlinked = if opts.drop_unlinked { unlinked_additions(&doc) } else { HashSet::new() };
284    for (num, generation, obj) in doc.overlay_entries() {
285        match obj {
286            // Written as free, so `/Size` still covers every number this document handed out. The
287            // generation stays: the number was never in use, and an undo can bring it back as is.
288            Some(_) if unlinked.contains(&num) => {
289                offsets.insert(num, (0, generation, false));
290            }
291            Some(o) => {
292                offsets.insert(num, (base + out.len() as u64, generation, true));
293                write_indirect(num, generation, &prepared(&doc, num, num, generation, &o), &mut out);
294            }
295            None => {
296                offsets.insert(num, (0, generation, false));
297            }
298        }
299    }
300    let prev = doc.revisions().last().map(|r| r.xref_offset);
301    let as_stream = doc.revisions().last().is_some_and(|r| r.is_stream);
302    let mut trailer = doc.trailer().clone();
303    if doc.output_security().0.is_none() {
304        ensure_id(&mut trailer, opts, &original); // an encrypted file's ID is part of its key
305    }
306    if let Some(p) = prev {
307        trailer.set(b"Prev".to_vec(), Object::Int(p as i64));
308    }
309    let size = doc.next_num().max(trailer.int(b"Size").map_or(0, |n| n.clamp(0, i64::from(u32::MAX)) as u32));
310    if as_stream {
311        let rows = offsets.iter().map(|(n, (o, g, used))| (*n, if *used { Row::InFile(*o, *g) } else { Row::Free(*g) })).collect();
312        write_xref_stream(&mut out, base, &mut trailer, rows, size);
313    } else {
314        trailer.set(b"Size".to_vec(), Object::Int(size as i64));
315        let xref_at = base + out.len() as u64;
316        write_xref_table(&mut out, &offsets, prev.is_none());
317        out.extend_from_slice(b"trailer\n");
318        write_dict(&trailer, &mut out);
319        let _ = write!(out, "\nstartxref\n{xref_at}\n%%EOF\n");
320    }
321    let mut file = Vec::with_capacity(original.len() + usize::from(newline) + out.len());
322    file.extend_from_slice(&original);
323    if newline {
324        file.push(b'\n');
325    }
326    file.extend_from_slice(&out);
327    Ok(file)
328}

330/// The objects [`write_full`] writes: everything reachable from the trailer's /Root, /Info and
331/// /Encrypt and from its unknown entries, breadth-first, with the new number each gets
332/// (aliases of one slot included) and, in visit order, whether it is a stream and the parsed
333/// object where the budget allowed keeping it.
334type SaveOrder = (HashMap<ObjRef, u32>, Vec<(ObjRef, bool)>, Vec<Option<Arc<Object>>>);
335
336/// Number the objects reachable from the trailer. `retain` keeps parsed objects for the write
337/// pass, up to the budget. A caller that needs only the numbers passes `false`, so the traversal
338/// holds no parsed objects beyond the current one.
339fn full_save_order(reader: &mut ObjectReader, trailer_in: &Dict, retain: bool) -> Result<SaveOrder, CosError> {
340    // Breadth-first walk from the trailer, assigning new numbers in visit order.
341    let mut map: HashMap<ObjRef, u32> = HashMap::new();
342    // Record the type while traversing so classifying packed objects needs no second parse.
343    let mut order: Vec<(ObjRef, bool)> = Vec::new();
344    // The parsed object of each entry in `order`, kept for the write pass while the budget
345    // allows it. Past the budget the write pass reads the object again.
346    let mut kept: Vec<Option<Arc<Object>>> = Vec::new();
347    let mut kept_budget = KeptObjects::default();
348    let mut queue: VecDeque<ObjRef> = VecDeque::new();
349    let visit_refs = |o: &Object, queue: &mut VecDeque<ObjRef>| collect_refs(o, &mut |r| queue.push_back(r));
350    for key in [&b"Root"[..], b"Info", b"Encrypt"] {
351        if let Some(r) = trailer_in.reference(key) {
352            queue.push_back(r);
353        }
354    }
355    // A direct /Encrypt can refer to objects, /CF or /StmF, say.
356    if let Some(encrypt @ Object::Dict(_)) = trailer_in.get(b"Encrypt") {
357        visit_refs(encrypt, &mut queue);
358    }
359    // Unknown trailer entries are document data. Objects they alone reference must be written
360    // too, or a full rewrite would keep the key and drop its value.
361    for (key, value) in trailer_in.iter() {
362        if !trailer_key_rewritten(key) {
363            collect_refs(value, &mut |r| queue.push_back(r));
364        }
365    }
366    while let Some(r) = queue.pop_front() {
367        if map.contains_key(&r) {
368            continue;
369        }
370        let o = reader.get(r);
371        if matches!(*o, Object::Null) {
372            continue; // dangling reference: written as null by renumbering below
373        }
374        // Accepted xref/header aliases denote one slot, including page-tree cycles.
375        // Check for null first: an unrelated or freed generation must remain dangling.
376        let canonical = ObjRef::new(r.num, reader.document.generation(r.num));
377        if let Some(&num) = map.get(&canonical) {
378            map.insert(r, num);
379            continue;
380        }
381        let num = output_number(order.len())?;
382        map.insert(r, num);
383        map.insert(canonical, num);
384        order.push((canonical, matches!(*o, Object::Stream(_))));
385        visit_refs(&o, &mut queue);
386        kept.push((retain && kept_budget.admit(&o, reader.document.bytes())).then_some(o));
387    }
388    Ok((map, order, kept))
389}

391/// The object numbers a full save of `doc` would write: exactly the objects [`write_full`]
392/// reaches from the trailer, object streams included. For a check that something removed from a
393/// document cannot reach the saved file through a reference nobody looked for.
394pub fn reachable_objects(doc: &Document) -> Result<HashSet<u32>, CosError> {
395    let mut reader = doc.object_reader();
396    let trailer_in = reader.document.trailer().clone();
397    let (_, order, _) = full_save_order(&mut reader, &trailer_in, false)?;
398    Ok(order.into_iter().map(|(r, _)| r.num).collect())
399}

401/// Write only reachable objects, renumbered from 1, with a table or cross-reference stream.
402pub fn write_full(doc: &Document, opts: &SaveOptions) -> Result<Vec<u8>, CosError> {
403    if doc.stream_limited() {
404        return Err(CosError::ReadOnlyLimit);
405    }
406    keyless_check(doc)?;
407    // Renumbering breaks a key that mixes in the object number.
408    if doc.security().is_some_and(|h| !h.has_key() && !h.survives_renumbering()) {
409        return Err(CosError::Security(
410            "the embedded files are encrypted and the document was opened without their password: open it with the password to save it in full"
411                .into(),
412        ));
413    }
414    // A full traversal must not fill the editor's shared caches (which undo snapshots also
415    // own). Parsed objects and decoded object streams are temporary working data here.
416    let mut reader = doc.object_reader();
417    stamp_mod_date(&mut reader.document, opts);
418    let trailer_in = reader.document.trailer().clone();
419    let (map, order, mut kept) = full_save_order(&mut reader, &trailer_in, true)?;
420    let version = reader.document.version().chars().take(3).collect::<String>();
421    let version = if opts.object_streams && version.as_str() < "1.5" { "1.5".to_string() } else { version };
422    let mut out = Vec::new();
423    let _ = writeln!(out, "%PDF-{version}");
424    // A binary comment (raw high bytes) marks the file as binary for transfer tools.
425    out.extend_from_slice(&[b'%', 0xE2, 0xE3, 0xCF, 0xD3, b'\n']);
426    // The /Encrypt dictionary and everything it references (e.g. indirect /CF crypt-filter
427    // dictionaries) are needed before anything can be decrypted: never inside object streams.
428    let mut encryption_objects: std::collections::HashSet<u32> = std::collections::HashSet::new();
429    let mut pending: Vec<ObjRef> = Vec::new();
430    if let Some(encrypt) = trailer_in.get(b"Encrypt") {
431        collect_refs(encrypt, &mut |r| pending.push(r));
432    }
433    while let Some(r) = pending.pop() {
434        if let Some(&num) = map.get(&r)
435            && encryption_objects.insert(num)
436        {
437            collect_refs(&reader.get(r), &mut |x| pending.push(x));
438        }
439    }
440    // Acrobat refuses an encrypted file whose catalog sits in an object stream ("the file is
441    // damaged and could not be repaired", before it even asks for the password), while other
442    // readers accept it. qpdf keeps the catalog out of object streams too. Verified with
443    // Acrobat DC on macOS (#774).
444    let root_outside =
445        if reader.document.output_security().0.is_some() { trailer_in.reference(b"Root").and_then(|r| map.get(&r)).copied() } else { None };
446    let mut rows: BTreeMap<u32, Row> = BTreeMap::new();
447    // Keep only references, not renamed copies of the entire object graph. Standalone
448    // objects still precede object streams, preserving the established output order.
449    let mut packed: Vec<(u32, ObjRef, Option<Arc<Object>>)> = Vec::new();
450    for (i, (r, is_stream)) in order.iter().enumerate() {
451        let num = output_number(i)?;
452        let in_stream = opts.object_streams && !is_stream && !encryption_objects.contains(&num) && root_outside != Some(num);
453        if in_stream {
454            packed.push((num, *r, kept[i].take()));
455        } else {
456            let o = written_object(&mut reader, *r, *is_stream, kept[i].take())?;
457            let o = renumber(&prepared(&reader.document, r.num, num, 0, &o), &map);
458            rows.insert(num, Row::InFile(out.len() as u64, 0));
459            write_indirect(num, 0, &o, &mut out);
460        }
461    }
462    // Object streams take the numbers after the last object. Their contents are not encrypted
463    // individually: the whole stream is (§7.5.7, §7.6.2).
464    let mut next = output_number(order.len())?;
465    let mut batch = ObjectStreamBatch::default();
466    for (num, reference, parsed) in packed {
467        let mut encoded = Vec::new();
468        serialize(&renumber(&*written_object(&mut reader, reference, false, parsed)?, &map), &mut encoded);
469        if batch.count > 0 && batch.body.len().saturating_add(encoded.len()).saturating_add(1) > OBJSTM_BYTES {
470            batch.flush(&reader.document, &mut next, &mut rows, &mut out)?;
471        }
472        let _ = write!(batch.header, "{} {} ", num, batch.body.len());
473        batch.body.extend_from_slice(&encoded);
474        batch.body.push(b'\n');
475        drop(encoded);
476        rows.insert(num, Row::InStream(next, batch.count as u32));
477        batch.count += 1;
478        if batch.count == OBJSTM_SIZE || batch.body.len() >= OBJSTM_BYTES {
479            batch.flush(&reader.document, &mut next, &mut rows, &mut out)?;
480        }
481    }
482    batch.flush(&reader.document, &mut next, &mut rows, &mut out)?;
483    let mut trailer = full_save_trailer(&reader.document, &trailer_in, &map, opts, &out);
484    if opts.object_streams {
485        write_xref_stream(&mut out, 0, &mut trailer, rows, next);
486    } else {
487        trailer.set(b"Size".to_vec(), Object::Int(next as i64));
488        let offsets: BTreeMap<u32, (u64, u16, bool)> =
489            rows.iter().filter_map(|(n, r)| if let Row::InFile(o, g) = r { Some((*n, (*o, *g, true))) } else { None }).collect();
490        let xref_at = out.len();
491        write_xref_table(&mut out, &offsets, true);
492        out.extend_from_slice(b"trailer\n");
493        write_dict(&trailer, &mut out);
494        let _ = write!(out, "\nstartxref\n{xref_at}\n%%EOF\n");
495    }
496    // The size isn't known until the end, and the buffer grew by doubling. The result is often
497    // kept as a document's working bytes for as long as it's open: drop the unused capacity.
498    out.shrink_to_fit();
499    Ok(out)
500}

502/// What `write_full` keeps of the objects it parsed while numbering them, measured with
503/// [`held_bytes`] (the parsed tree and any data that is not a view into the file). Beyond the
504/// budget objects are read again when written, which bounds the memory of a save of a huge
505/// document.
506#[derive(Default)]
507struct KeptObjects {
508    bytes: usize,
509}
510
511impl KeptObjects {
512    const BYTES: usize = 64 * 1024 * 1024;
513
514    /// Whether `object` still fits the budget; if so it is counted against it.
515    fn admit(&mut self, object: &Object, data: &Arc<Vec<u8>>) -> bool {
516        let bytes = self.bytes.saturating_add(held_bytes(object, data, 0));
517        if bytes > Self::BYTES {
518            return false;
519        }
520        self.bytes = bytes;
521        true
522    }
523}

528    match kept {

534/// Assemble the full rewrite's trailer and apply its identifier policy.
535fn full_save_trailer(doc: &Document, trailer_in: &Dict, map: &HashMap<ObjRef, u32>, opts: &SaveOptions, out: &[u8]) -> Dict {
536    let mut trailer = Dict::new();
537    for key in [&b"Root"[..], b"Info", b"Encrypt"] {
538        if let Some(r) = trailer_in.reference(key).and_then(|r| map.get(&r)) {
539            trailer.set(key.to_vec(), Object::Ref(ObjRef::new(*r, 0)));
540        }
541    }
542    if let Some(encrypt @ Object::Dict(_)) = trailer_in.get(b"Encrypt") {
543        trailer.set(b"Encrypt".to_vec(), renumber(encrypt, map));
544    }
545    if let Some(id) = trailer_in.get(b"ID") {
546        trailer.set(b"ID".to_vec(), direct_id(doc, id));
547    }
548    for (key, value) in trailer_in.iter() {
549        if !trailer_key_rewritten(key) {
550            trailer.set(key.clone(), renumber(value, map));
551        }
552    }
553    if doc.output_security().0.is_none() {
554        ensure_id(&mut trailer, opts, doc.bytes());
555    }
556    if doc.new_id_required() {
557        regenerate_id(&mut trailer, opts, out, doc.output_security().0.is_some());
558    }
559    trailer
560}

562/// Object `r` read again for writing. The traversal read it once and only kept its number, so
563/// a second read that fails, finds nothing or finds a different kind of object must fail the
564/// save rather than write `null` in its place.
565fn reread(reader: &mut ObjectReader, r: ObjRef, is_stream: bool) -> Result<Arc<Object>, CosError> {
566    let o = reader.try_get(r)?;
567    if matches!(*o, Object::Null) || matches!(*o, Object::Stream(_)) != is_stream {
568        return Err(CosError::Syntax { offset: 0, detail: format!("object {} read differently while saving; nothing was written", r.num) });
569    }
570    Ok(o)
571}

575    header: Vec<u8>,
576    body: Vec<u8>,
577    count: usize,

580impl ObjectStreamBatch {
581    fn flush(&mut self, doc: &Document, next: &mut u32, rows: &mut BTreeMap<u32, Row>, out: &mut Vec<u8>) -> Result<(), CosError> {
582        if self.count == 0 {
583            return Ok(());
584        }
585        let following = next.checked_add(1).ok_or_else(|| CosError::Syntax { offset: 0, detail: "too many objects to save".into() })?;
586        // Release capacity after each batch too: an unusually large object must not leave
587        // a large scratch buffer alive for the rest of the save.
588        let Self { mut header, body, count } = std::mem::take(self);
589        let first = header.len();
590        header.extend_from_slice(&body);
591        drop(body);
592        let mut d = Dict::new();
593        d.set(b"Type".to_vec(), Object::name("ObjStm"));
594        d.set(b"N".to_vec(), Object::Int(count as i64));
595        d.set(b"First".to_vec(), Object::Int(first as i64));
596        let stm = Object::Stream(Stream::flate(d, &header));
597        rows.insert(*next, Row::InFile(out.len() as u64, 0));
598        write_indirect(*next, 0, &prepared(doc, u32::MAX, *next, 0, &stm), out);
599        *next = following;
600        Ok(())
601    }

604/// `/ID` for the trailer, direct, as it has to be with `/Encrypt`: nothing else in the file holds
605/// what it refers to now. Given by reference it is two entries at most, however many references a
606/// file gave, so that one large string isn't copied for each. An entry that isn't a string stays in
607/// its place as an empty one, which is how the key read it.
608fn direct_id(doc: &Document, id: &Object) -> Object {
609    if matches!(id, Object::Array(a) if !a.iter().any(|s| matches!(s, Object::Ref(_)))) {
610        return id.clone();
611    }
612    let entry = |s: &Object| match &*doc.resolve(s) {
613        s @ Object::String(_) => s.clone(),
614        _ => Object::String(crate::cos::PdfString { bytes: Vec::new(), hex: true }),
615    };
616    let held = doc.resolve(id);
617    Object::Array(held.as_array().into_iter().flatten().take(2).map(entry).collect())
618}

633/// Objects added since the file was read (their numbers are not in use in it) that nothing an
634/// incremental update writes refers to: what one edit added and a later one unlinked, such as a
635/// deleted comment or a cleared redaction mark. The file never held them, so leaving them out
636/// loses nothing, while writing them would keep what the user removed. Only the update is walked:
637/// from the trailer and the file's changed objects through additions. An unchanged object of the
638/// file is not followed: a reference it holds to such a number already reads as null in the file
639/// (a damaged file's dangling reference), and it still does.
640pub(crate) fn unlinked_additions(doc: &Document) -> HashSet<u32> {
641    let added = |num: u32| matches!(doc.xref_entry(num), None | Some(XrefEntry::Free { .. }));
642    let mut pending: HashMap<u32, Arc<Object>> =
643        doc.overlay_entries().filter_map(|(num, _, o)| Some((num, o?))).filter(|(num, _)| added(*num)).collect();
644    if pending.is_empty() {
645        return HashSet::new();
646    }
647    let mut stack: Vec<ObjRef> = Vec::new();
648    doc.trailer().iter().for_each(|(_, v)| collect_refs(v, &mut |r| stack.push(r)));
649    for (num, _, o) in doc.overlay_entries() {
650        if let Some(o) = o
651            && !added(num)
652        {
653            collect_refs(&o, &mut |r| stack.push(r));
654        }
655    }
656    while let Some(r) = stack.pop() {
657        if let Some(o) = pending.remove(&r.num) {
658            collect_refs(&o, &mut |r| stack.push(r));
659        }
660    }
661    pending.into_keys().collect()
662}

664pub(crate) fn collect_refs(o: &Object, f: &mut impl FnMut(ObjRef)) {
665    match o {
666        Object::Ref(r) => f(*r),
667        Object::Array(a) => a.iter().for_each(|x| collect_refs(x, f)),
668        Object::Dict(d) => d.iter().for_each(|(_, v)| collect_refs(v, f)),
669        Object::Stream(s) => s.dict.iter().for_each(|(_, v)| collect_refs(v, f)),
670        _ => {}
671    }
672}

674fn renumber(o: &Object, map: &HashMap<ObjRef, u32>) -> Object {
675    match o {
676        Object::Ref(r) => map.get(r).map(|n| Object::Ref(ObjRef::new(*n, 0))).unwrap_or(Object::Null),
677        Object::Array(a) => Object::Array(a.iter().map(|x| renumber(x, map)).collect()),
678        Object::Dict(d) => Object::Dict(d.iter().map(|(k, v)| (k.clone(), renumber(v, map))).collect()),
679        Object::Stream(s) => Object::Stream(Stream { dict: s.dict.iter().map(|(k, v)| (k.clone(), renumber(v, map))).collect(), raw: s.raw.clone() }),
680        other => other.clone(),
681    }
682}

684/// Classic xref table with one subsection per run of consecutive numbers (§7.5.4).
685fn write_xref_table(out: &mut Vec<u8>, offsets: &BTreeMap<u32, (u64, u16, bool)>, include_zero: bool) {
686    out.extend_from_slice(b"xref\n");
687    let mut all: Vec<(u32, (u64, u16, bool))> = offsets.iter().map(|(k, v)| (*k, *v)).collect();
688    if include_zero {
689        all.insert(0, (0, (0, 65535, false)));
690    }
691    // A section is one or more subsections. `xref` followed immediately by `trailer` is not a
692    // table (qpdf: "xref syntax invalid") and is what an incremental save of a classic-table
693    // file used to append when nothing was edited. A zero-length subsection names no objects,
694    // so the free list and every in-use entry stay as they were, while a trailer-only change
695    // (a missing `/ID`, or a caller edit of the trailer) is still written.
696    if all.is_empty() {
697        let _ = writeln!(out, "0 0");
698        return;
699    }
700    let mut i = 0;
701    while i < all.len() {
702        let start = all[i].0;
703        let mut j = i;
704        while j + 1 < all.len() && all[j + 1].0 == all[j].0 + 1 {
705            j += 1;
706        }
707        let _ = writeln!(out, "{} {}", start, j - i + 1);
708        for (_, (off, generation, used)) in &all[i..=j] {
709            let _ = writeln!(out, "{:010} {:05} {} ", off, generation, if *used { 'n' } else { 'f' });
710        }
711        i = j + 1;
712    }
713}

715/// A cross-reference stream section (§7.5.8) that takes object number `num`, ending the file.
716/// `out` holds the file from byte offset `base` on. Rows are Flate-compressed with the PNG Up
717/// predictor, and field widths fit the largest value.
718fn write_xref_stream(out: &mut Vec<u8>, base: u64, trailer: &mut Dict, mut rows: BTreeMap<u32, Row>, num: u32) {
719    let xref_at = base + out.len() as u64;
720    rows.insert(num, Row::InFile(xref_at, 0));
721    if trailer.get(b"Prev").is_none() {
722        rows.entry(0).or_insert(Row::Free(65535));
723    }
724    let fields = |r: &Row| match *r {
725        Row::Free(g) => (0u64, 0u64, u64::from(g)),
726        Row::InFile(o, g) => (1, o, u64::from(g)),
727        Row::InStream(s, i) => (2, u64::from(s), u64::from(i)),
728    };
729    let width = |v: u64| ((64 - v.leading_zeros()).div_ceil(8)).max(1) as usize;
730    let (w2, w3) = rows.values().map(fields).fold((1, 1), |(a, b), (_, x, y)| (width(x).max(a), width(y).max(b)));
731    let cols = 1 + w2 + w3;
732    let mut index = Vec::new();
733    let mut numbers = rows.keys().copied().peekable();
734    while let Some(start) = numbers.next() {
735        let mut last = start;
736        let mut count = 1i64;
737        while numbers.peek().copied().is_some_and(|next| last.checked_add(1) == Some(next)) {
738            if let Some(next) = numbers.next() {
739                last = next;
740                count += 1;
741            }
742        }
743        index.push(Object::Int(i64::from(start)));
744        index.push(Object::Int(count));
745    }
746    let mut data = Vec::with_capacity(rows.len().saturating_mul(cols + 1));
747    // Type is one byte, and each remaining field is at most one u64. Reuse the
748    // predictor row on the stack instead of allocating a Vec for every xref entry.
749    let mut prev = [0u8; 17];
750    for r in rows.values() {
751        let (t, a, b) = fields(r);
752        let a = a.to_be_bytes();
753        let b = b.to_be_bytes();
754        let fields = std::iter::once(t as u8).chain(a.into_iter().skip(8 - w2)).chain(b.into_iter().skip(8 - w3));
755        data.push(2); // PNG "Up"
756        for (value, previous) in fields.zip(&mut prev) {
757            data.push(value.wrapping_sub(*previous));
758            *previous = value;
759        }
760    }
761    let mut d = trailer.clone();
762    d.set(b"Type".to_vec(), Object::name("XRef"));
763    d.set(b"Size".to_vec(), Object::Int(num as i64 + 1));
764    d.set(b"W".to_vec(), Object::Array(vec![Object::Int(1), Object::Int(w2 as i64), Object::Int(w3 as i64)]));
765    d.set(b"Index".to_vec(), Object::Array(index));
766    let mut stream = Stream::flate(d, &data);
767    let mut parms = Dict::new();
768    parms.set(b"Predictor".to_vec(), Object::Int(12));
769    parms.set(b"Columns".to_vec(), Object::Int(cols as i64));
770    stream.dict.set(b"DecodeParms".to_vec(), Object::Dict(parms));
771    write_indirect(num, 0, &Object::Stream(stream), out);
772    let _ = write!(out, "startxref\n{xref_at}\n%%EOF\n");
773    trailer.set(b"Size".to_vec(), Object::Int(num as i64 + 1));
774}

802/// A new file identifier for a file whose bytes are `content` (ISO 32000-1 14.4): an array of
803/// two equal 16-byte strings, the same for the same content and `seed`.
804pub fn new_file_id(content: &[u8], seed: u64) -> Object {
805    // FNV-1a over a sample of the content and the seed: stable, dependency-free.
806    let mut h: u64 = 0xcbf2_9ce4_8422_2325 ^ seed;
807    for b in content.iter().step_by((content.len() / 4096).max(1)).chain(content.iter().rev().take(64)) {
808        h ^= *b as u64;
809        h = h.wrapping_mul(0x0100_0000_01b3);
810    }
811    let mut id = Vec::with_capacity(16);
812    id.extend_from_slice(&h.to_be_bytes());
813    id.extend_from_slice(&h.rotate_left(29).wrapping_mul(0x9E37_79B9_7F4A_7C15).to_be_bytes());
814    let s = Object::String(PdfString { bytes: id, hex: true });
815    Object::Array(vec![s.clone(), s])
816}

818/// Give a rewritten file a new identifier (§14.4), so that a redacted or sanitized copy does not
819/// claim to be its source. An encrypted file keeps the first string, which its key is derived from,
820/// and gets a new second one. When there is no such string (no `/ID`, one entry, or a first entry
821/// that is not a string) the reader takes the empty string as the key input (`Document::first_id`),
822/// so the empty string is written first and the key stays valid.
823fn regenerate_id(trailer: &mut Dict, opts: &SaveOptions, written: &[u8], encrypted: bool) {
824    let fresh = Object::String(PdfString { bytes: new_id_bytes(opts, written), hex: true });
825    let pair = if encrypted {
826        let first = match trailer.get(b"ID") {
827            Some(Object::Array(a)) => match a.first() {
828                Some(Object::String(s)) => Object::String(s.clone()),
829                _ => Object::String(PdfString { bytes: Vec::new(), hex: true }),
830            },
831            _ => Object::String(PdfString { bytes: Vec::new(), hex: true }),
832        };
833        vec![first, fresh]
834    } else {
835        vec![fresh.clone(), fresh]
836    };
837    trailer.set(b"ID".to_vec(), Object::Array(pair));
838}

840/// FNV-1a over 64 bits, with its published offset basis and prime. The identifier is built from it
841/// rather than from `std`'s `DefaultHasher`, whose algorithm the standard library does not specify.
842const FNV_OFFSET: u64 = 0xcbf2_9ce4_8422_2325;
843const FNV_PRIME: u64 = 0x0000_0100_0000_01b3;
844
845fn fnv1a(state: u64, bytes: &[u8]) -> u64 {
846    bytes.iter().fold(state, |h, b| (h ^ u64::from(*b)).wrapping_mul(FNV_PRIME))
847}

849/// Sixteen bytes for a file identifier, one 8-byte half per lane. The bytes are a pure function of
850/// `SaveOptions::id_entropy`, `id_seed` and the written file: the hash is FNV-1a (above) and every
851/// integer is written as a fixed-width big-endian value, so the same input gives the same identifier
852/// on every platform and toolchain. Without entropy only the salt changes (see `id_salt`). FNV is
853/// not collision-resistant against chosen content; the identifier only has to be unique, not secret.
854/// Every written byte is hashed, once, in a single traversal of the output.
855fn new_id_bytes(opts: &SaveOptions, written: &[u8]) -> Vec<u8> {
856    let salt = id_salt(opts);
857    let length = u64::try_from(written.len()).unwrap_or(u64::MAX);
858    let mut halves = [0u8, 1].map(|lane| {
859        let mut h = fnv1a(FNV_OFFSET, &salt.to_be_bytes());
860        h = fnv1a(h, &[lane]);
861        h = fnv1a(h, &opts.id_seed.to_be_bytes());
862        fnv1a(h, &length.to_be_bytes())
863    });
864    for chunk in written.chunks(4096) {
865        for h in &mut halves {
866            *h = fnv1a(*h, chunk);
867        }
868    }
869    halves.iter().flat_map(|h| h.to_be_bytes()).collect()
870}

872/// The salt of an identifier. With caller entropy it is a fixed hash of that entropy (deterministic,
873/// for tests and callers who pin output). Without it the salt is fresh: on WebAssembly from the
874/// browser ([`browser_salt`], so two fresh web apps that write the same bytes cannot share an
875/// identifier), and elsewhere from the standard library's random hash keys — seeded by the operating
876/// system, so separate processes differ — mixed with a process-wide counter, which separates
877/// successive calls in one process.
878fn id_salt(opts: &SaveOptions) -> u64 {
879    use std::hash::BuildHasher;
880    use std::sync::atomic::{AtomicU64, Ordering};
881    static COUNTER: AtomicU64 = AtomicU64::new(0);
882    match &opts.id_entropy {
883        Some(entropy) => fnv1a(FNV_OFFSET, entropy),
884        None => {
885            #[cfg(target_arch = "wasm32")]
886            if let Some(salt) = browser_salt() {
887                return salt;
888            }
889            std::collections::hash_map::RandomState::new().hash_one(COUNTER.fetch_add(1, Ordering::Relaxed))
890        }
891    }
892}

894/// One byte from one `Math.random` draw: the draw (in [0, 1)) scaled to [0, 256), integer part
895/// taken. The quotient is in [0, 256), so a saturating cast never wraps; a NaN or infinite draw
896/// (which a conforming engine never returns) folds in as 0. No input can panic here.
897#[cfg(any(target_arch = "wasm32", test))]
898fn draw_byte(random: f64) -> u8 {
899    (random * 256.0).floor().rem_euclid(256.0) as u8
900}

902/// The `Math.random` fallback's mixing: one salt byte per draw (see [`draw_byte`]), folded through
903/// FNV-1a exactly like the Web Crypto bytes, so all sixteen bytes of the fold carry entropy from
904/// the draws. Pure and platform-independent, so the native test target can exercise it; on wasm
905/// each draw comes from `Math.random`.
906#[cfg(any(target_arch = "wasm32", test))]
907fn salt_from_draws(draws: &[f64; 16]) -> u64 {
908    let mut bytes = [0u8; 16];
909    for (byte, draw) in bytes.iter_mut().zip(draws) {
910        *byte = draw_byte(*draw);
911    }
912    fnv1a(FNV_OFFSET, &bytes)
913}

915/// Fresh salt entropy from the browser, on WebAssembly. `crypto.getRandomValues` (Web Crypto, the
916/// browser's secure generator, available in windows and workers alike) gives sixteen bytes, folded
917/// into one salt. When that API is missing or refuses the call, `Math.random` is mixed instead
918/// ([`salt_from_draws`]: sixteen draws, so the salt's full width carries entropy) — browsers seed
919/// it afresh for every document, so a fresh instance still cannot reproduce another's salt. Only
920/// where neither is reachable does the caller fall back to its `RandomState` salt, whose keys are
921/// fixed at compile time on `wasm32-unknown-unknown` — the state before browser entropy was wired
922/// in.
923#[cfg(target_arch = "wasm32")]
924fn browser_salt() -> Option<u64> {
925    use wasm_bindgen::JsCast;
926    let crypto = js_sys::Reflect::get(&js_sys::global(), &wasm_bindgen::JsValue::from_str("crypto"))
927        .ok()
928        .filter(|crypto| !crypto.is_undefined() && !crypto.is_null());
929    if let Some(crypto) = crypto {
930        let fill = js_sys::Reflect::get(&crypto, &wasm_bindgen::JsValue::from_str("getRandomValues"))
931            .ok()
932            .and_then(|fill| fill.dyn_into::<js_sys::Function>().ok());
933        if let Some(fill) = fill {
934            let bytes = js_sys::Uint8Array::new_with_length(16);
935            if fill.call1(&crypto, &bytes).is_ok() {
936                return Some(fnv1a(FNV_OFFSET, &bytes.to_vec()));
937            }
938        }

941    let mut draws = [0.0; 16];

948/// A PDF date string for "now" given seconds since the Unix epoch (UTC).
949pub fn pdf_date(unix_secs: i64) -> String {
950    let (days, secs) = (unix_secs.div_euclid(86_400), unix_secs.rem_euclid(86_400));
951    let (y, m, d) = civil_from_days(days);
952    // `{y:04}` rather than `%Y`, which writes a `+` before years past 9999 (and a PDF date is a
953    // bare four-plus-digit year, never signed). No year clamp — the old code wrote `y` as-is.
954    format!("D:{y:04}{m:02}{d:02}{:02}{:02}{:02}Z", secs / 3600, secs / 60 % 60, secs % 60)
955}

957/// Days since 1970-01-01 → proleptic Gregorian (year, month, day). Total for any `i64` count:
958/// dates past `NaiveDate`'s ±262143-year range keep their proleptic Gregorian year instead of
959/// falling back to the epoch.
960///
961/// The Gregorian calendar repeats every 400 years (146_097 days), so chrono converts the day
962/// within one cycle and whole cycles go back into the year (an `i128` so the extremes can't
963/// overflow), keeping chrono as the only calendar logic. 1970-01-01 is `num_days_from_ce` 719_163.
964pub fn civil_from_days(days: i64) -> (i64, u32, u32) {
965    use chrono::Datelike;
966    let (cycles, rest) = (days.div_euclid(146_097), days.rem_euclid(146_097));
967    // `rest + 719_163` is 719_163..865_259, always a valid CE day.
968    match i32::try_from(rest + 719_163).ok().and_then(chrono::NaiveDate::from_num_days_from_ce_opt) {
969        Some(d) => {
970            let y = i128::from(d.year()) + i128::from(cycles) * 400;
971            (i64::try_from(y).unwrap_or(if y < 0 { i64::MIN } else { i64::MAX }), d.month(), d.day())
972        }
973        None => (1970, 1, 1), // unreachable
974    }
975}
