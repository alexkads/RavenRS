78    /// save of it, with this document's credentials: an encrypted one must have the same
79    /// `/Encrypt` dictionary and `/ID` as this document, whose file key is then reused, so no
80    /// password is needed or kept; unencrypted data opens as [`open`](Self::open) does. Data

83        let credentials = match &self.security {

90    fn open_as(data: Arc<Vec<u8>>, credentials: Credentials<'_>, stream_limit: usize) -> Result<Self, CosError> {
91        // Viewers accept files whose header is missing or damaged as long as the body looks
92        // like PDF; so do we (a note goes to the repair log).
93        let header = find(&data, b"%PDF-", 0, 1024);
94        let headerless = header.is_none();
95        if headerless && find(&data, b" obj", 0, 4096).is_none() {
96            return Err(CosError::NotPdf);
97        }
98        let header_offset = header.unwrap_or(0);
99        let version = match header {
100            Some(h) => String::from_utf8_lossy(&data[(h + 5).min(data.len())..(h + 8).min(data.len())]).trim().to_string(),
101            None => "1.4".to_string(),
102        };
103        let mut doc = Document {
104            data,
105            endstreams: Arc::default(),
106            source_identity: Arc::new(()),
107            entries: Arc::new(XrefIndex::empty()),
108            trailer: Dict::new(),
109            revisions: Arc::new(Vec::new()),
110            xref_oddities: Arc::default(),
111            located: Arc::default(),
112            repair_log: Arc::new(Vec::new()),
113            unlinked_update: false,
114            cache: Arc::default(),
115            objstms: Arc::default(),
116            unreadable: Arc::default(),
117            jbig2_globals: Arc::default(),
118            stream_limit: stream_limit.min(crate::cos::object::MAX_DECODED),
119            overlay: BTreeMap::new(),
120            next_num: 1,
121            header_offset,
122            version,
123            security: None,
124            encrypt_num: None,
125            encrypt_members: Arc::default(),
126            unfound: Arc::default(),
127            encryption_changed: false,
128            full_save: false,
129            new_id: false,
130            out_security: None,
131            out_encrypt_num: None,
132        };
133        let mut log = Vec::new();
134        if headerless {
135            log.push("the %PDF- header is missing; reading the file as PDF 1.4".into());
136        }
137        let parsed = doc.read_xref_chain(&mut log);
138        let mut authenticated = false;
139        // The newest section cut the page tree off from the chain (see `update_cuts_off_the_pages`).
140        let mut cut_off = false;
141        let ok = match parsed {
142            Ok((entries, trailer, revisions, oddities)) => {
143                doc.entries = Arc::new(entries);
144                doc.trailer = trailer;
145                doc.revisions = Arc::new(revisions);
146                doc.xref_oddities = Arc::new(oddities);
147                doc.fill_xref_gaps(&mut log);
148                // Authenticate before checking the catalog: it may sit in an encrypted object
149                // stream, which reads as garbage until the key is known.
150                if let Some(enc) = doc.trailer.get(b"Encrypt").cloned() {
151                    doc.authenticate(&enc, &credentials)?;
152                    authenticated = true;
153                }
154                let catalog = doc.root_is_catalog();
155                cut_off = catalog && doc.update_cuts_off_the_pages();
156                catalog && !cut_off
157            }
158            Err(e) => {
159                log.push(format!("cross-reference data unreadable ({e}); rebuilding from object headers"));
160                false
161            }
162        };
163        if cut_off {
164            log.push(UNLINKED_UPDATE.into());
165            doc.unlinked_update = true;
166        } else if !ok && !doc.entries.is_empty() {
167            log.push("cross-reference data does not lead to a valid catalog; rebuilding from object headers".into());
168        }
169        if !ok {
170            doc.reconstruct(&mut log)?;
171        }
172        // Header generations are plaintext, including in encrypted files. Inspect only the
173        // bounded headers, not object bodies, and retain the compact xref representation.
174        doc.note_xref_generation_repairs(&mut log);
175        doc.repair_log = Arc::new(log);
176        // Reconstruction may have found a different trailer: (re)authenticate against it.
177        if (!ok || !authenticated)
178            && let Some(enc) = doc.trailer.get(b"Encrypt").cloned()
179        {
180            doc.authenticate(&enc, &credentials)?;
181        }
182        if Arc::make_mut(&mut doc.entries).remove_exhausted() {
183            Arc::make_mut(&mut doc.repair_log).push("skipped exhausted xref object number 4294967295".into());
184        }
185        let max = doc.entries.max_num();
186        let size = u32::try_from(doc.trailer.int(b"Size").unwrap_or(0).max(0)).unwrap_or(u32::MAX);
187        if size == u32::MAX {
188            Arc::make_mut(&mut doc.repair_log).push("exhausted trailer /Size; new objects use unused object numbers".into());
189        }
190        doc.next_num = max.saturating_add(1).max(size).max(1);
191        Ok(doc)
192    }

194    /// Whether the chain of cross-reference sections ends at the newest one (no `/Prev`, or one
195    /// that isn't an offset) although the file holds earlier sections, and the catalog's `/Pages`
196    /// is not among the objects that one section reaches. ISO 32000-2 §7.5.6 and §7.5.8.2 require
197    /// the `/Prev`; without it the earlier revisions, page tree and all, are cut off, so the file
198    /// is rebuilt from its object headers as a damaged one is (the newest definition of each
199    /// object wins), and the repair log says why. (A file whose `/Root` itself is missing is
200    /// rebuilt anyway.)
201    fn update_cuts_off_the_pages(&self) -> bool {
202        if self.revisions.len() != 1 {
203            return false;
204        }
205        let Some(root) = self.root() else { return false };
206        let Ok(catalog) = self.try_get_reference(root) else { return false };
207        let Object::Dict(catalog) = &*catalog else { return false };
208        // The page tree first: the scan for an earlier `startxref` reads the whole file.
209        !catalog.get(b"Pages").is_some_and(|pages| matches!(&*self.resolve(pages), Object::Dict(_))) && earlier_sections(&self.data)
210    }

212    fn authenticate(&mut self, enc: &Object, credentials: &Credentials<'_>) -> Result<(), CosError> {
213        // Authentication can change how the same input is decoded.
214        self.source_identity = Arc::new(());
215        let (num, dict) = match enc {
216            Object::Ref(r) => (Some(r.num), self.get(*r).as_dict().cloned()),
217            Object::Dict(d) => (None, Some(d.clone())),
218            _ => (None, None),
219        };
220        let Some(dict) = dict else {
221            // A dangling /Encrypt: viewers treat the file as unencrypted, and so do we.
222            let mut log = self.repair_log.as_ref().clone();
223            log.push("the trailer names an /Encrypt dictionary that does not exist; reading the file as unencrypted".into());
224            self.repair_log = Arc::new(log);
225            self.trailer.remove(b"Encrypt");
226            return Ok(());
227        };
228        let params = crate::cos::security::encrypt_dict(self, &dict);
229        let id0 = self.first_id();
230        // Read here, while nothing is decrypted yet.
231        let members = self.encryption_members(enc);
232        let handler = match credentials {
233            Credentials::Password(password) => match crate::crypt::SecurityHandler::open(params.clone(), &id0, *password) {
234                Ok(handler) => Arc::new(handler),
235                // No password, or the empty one: what only the embedded files need isn't asked.
236                Err(crate::crypt::CryptError::WrongPassword) => match self.without_password(&dict, params) {
237                    Some(handler) if password.is_none_or(str::is_empty) => Arc::new(handler),
238                    _ if password.is_none() => return Err(CosError::NeedsPassword),
239                    _ => return Err(CosError::WrongPassword),
240                },
241                Err(other) => return Err(CosError::Security(other.to_string())),
242            },
243            // The same encryption as the document whose key this is: that key opens it. A handler
244            // without the key opens only what it would open by itself.
245            Credentials::Handler { handler, id0: known } if *handler.dict() == params && *known == id0 => {
246                if !handler.has_key() && !crate::cos::security::efopen_only(self, &dict) {
247                    return Err(CosError::NeedsPassword);
248                }
249                Arc::clone(handler)
250            }
251            Credentials::Handler { .. } => return Err(CosError::NeedsPassword),
252        };
253        self.security = Some(handler);
254        self.encrypt_num = num;
255        self.encrypt_members = Arc::new(members);
256        // Objects read while locating the catalog were read before decryption was possible.
257        self.cache = Arc::default();
258        self.objstms = Arc::default();
259        self.unreadable = Arc::default();
260        Ok(())
261    }

263    /// The handler for a document opened without a password that only its embedded files need
264    /// (see [`crate::cos::security::efopen_only`]). Not a repair: ISO 32000-2 §7.6.6 has such a
265    /// document open so; [`Document::embedded_files_locked`] tells it.
266    fn without_password(&self, dict: &Dict, params: crate::crypt::EncryptDict) -> Option<crate::crypt::SecurityHandler> {
267        crate::cos::security::efopen_only(self, dict).then(|| crate::crypt::SecurityHandler::without_key(params)).flatten()
268    }

270    /// A document opened without the password its embedded files need takes either password
271    /// later: with the file key its embedded files decrypt, so what was read as stored is read
272    /// again. Refused while an edited stream holds data read as stored, which the key would
273    /// encrypt a second time.
274    fn unlock_embedded_files(&mut self, password: &str) -> Result<(), CosError> {
275        let Some(current) = self.security.as_deref() else { return Ok(()) };
276        if self.overlay_entries().any(|(_, _, o)| matches!(o.as_deref(), Some(Object::Stream(s)) if self.reads_as_stored(s))) {
277            return Err(CosError::Security("an embedded file was edited before the password was given: save or undo that edit first".into()));
278        }
279        let handler = crate::crypt::SecurityHandler::open(current.dict().clone(), &self.first_id(), Some(password)).map_err(|e| match e {
280            crate::crypt::CryptError::WrongPassword => CosError::WrongPassword,
281            other => CosError::Security(other.to_string()),
282        })?;
283        self.security = Some(Arc::new(handler));
284        self.source_identity = Arc::new(());
285        self.cache = Arc::default();
286        self.objstms = Arc::default();
287        Ok(())
288    }

303    /// The first file identifier, part of the encryption key (empty when there is none).
304    fn first_id(&self) -> Vec<u8> {
305        let id = self.trailer.get(b"ID").map(|id| self.resolve(id));
306        let first = id.as_deref().and_then(Object::as_array).and_then(|a| a.first()).map(|s| self.resolve(s));
307        first.as_deref().and_then(Object::as_string).map(|s| s.bytes.clone()).unwrap_or_default()

310    /// The objects `/Encrypt`, given as `enc`, and `/ID` refer to, the ones they refer to, and so
311    /// on, up to [`MAX_ENCRYPT_MEMBERS`]. Their strings aren't encrypted (ISO 32000-2 §7.6.2): the
312    /// key is made from them, so they have to read as stored, and stay so. A number with no object
313    /// behind it, and a stream, which those dictionaries never hold, isn't one: an object added
314    /// under that number later, or a page's content a hostile `/Encrypt` names, is encrypted.
315    fn encryption_members(&self, enc: &Object) -> HashSet<u32> {
316        let mut pending = Vec::new();
317        crate::cos::writer::collect_refs(enc, &mut |r| pending.push(r));
318        if let Some(id) = self.trailer.get(b"ID") {
319            crate::cos::writer::collect_refs(id, &mut |r| pending.push(r));
320        }
321        let mut members = HashSet::new();
322        while let Some(r) = pending.pop() {
323            if members.len() >= MAX_ENCRYPT_MEMBERS || members.contains(&r.num) {
324                continue;
325            }
326            let held = self.get(r);
327            if matches!(&*held, Object::Null | Object::Stream(_)) {
328                continue;
329            }
330            members.insert(r.num);
331            crate::cos::writer::collect_refs(&held, &mut |x| pending.push(x));
332        }
333        members
334    }

343    /// Lift the restrictions of a document opened with its user password by entering the
344    /// owner (permissions) password (§7.6.4). Fails with `WrongPassword` for any other
345    /// password. Does nothing when the document is unencrypted or already opened as owner.
346    /// Both passwords unlock the same file key, so nothing read so far is decoded again.
347    pub fn unlock(&mut self, password: &str) -> Result<(), CosError> {
348        let Some(current) = self.security.as_deref() else { return Ok(()) };
349        if !current.has_key() {
350            return self.unlock_embedded_files(password);
351        }
352        if current.auth() == crate::crypt::Auth::Owner {
353            return Ok(());
354        }
355        let owner = current.authenticate_owner(&self.first_id(), password).map_err(|e| match e {
356            crate::crypt::CryptError::WrongPassword => CosError::WrongPassword,
357            other => CosError::Security(other.to_string()),
358        })?;
359        self.security = Some(Arc::new(owner));
360        Ok(())
361    }

415    /// Protect the document with a password (§7.6.4). Takes effect on the next save, which is
416    /// always a full rewrite. Returns the handler (authenticated as owner).
417    pub fn set_encryption(&mut self, params: &crate::crypt::NewEncryption) -> Result<(), CosError> {
418        // The file identifier is part of the key; make sure it exists and keep it.
419        let id = self.trailer.get(b"ID").map(|id| self.resolve(id));
420        let id0 = match id.as_deref() {
421            Some(Object::Array(a)) if a.len() == 2 => self.first_id(),
422            _ => {
423                let id = crate::crypt::file_id(&params.seed);
424                let s = Object::String(crate::cos::PdfString { bytes: id.clone(), hex: true });
425                self.trailer.set(b"ID".to_vec(), Object::Array(vec![s.clone(), s]));
426                id
427            }
428        };
429        let h = crate::crypt::create(params, &id0).map_err(|e| CosError::Security(e.to_string()))?;
430        let d = h.dict();
431        let s = |b: &[u8]| Object::String(crate::cos::PdfString { bytes: b.to_vec(), hex: true });
432        let mut e = Dict::new();
433        e.set(b"Filter".to_vec(), Object::name("Standard"));
434        e.set(b"V".to_vec(), Object::Int(d.v));
435        e.set(b"R".to_vec(), Object::Int(d.r));
436        e.set(b"Length".to_vec(), Object::Int(d.length_bits));
437        e.set(b"O".to_vec(), s(&d.o));
438        e.set(b"U".to_vec(), s(&d.u));
439        e.set(b"P".to_vec(), Object::Int(i64::from(d.p)));
440        if d.v >= 4 {
441            let mut cf = Dict::new();
442            let mut std_cf = Dict::new();
443            std_cf.set(b"Type".to_vec(), Object::name("CryptFilter"));
444            let (cfm, len) = if d.v >= 5 { ("AESV3", 32) } else { ("AESV2", 16) };
445            std_cf.set(b"CFM".to_vec(), Object::name(cfm));
446            std_cf.set(b"Length".to_vec(), Object::Int(len));
447            std_cf.set(b"AuthEvent".to_vec(), Object::name("DocOpen"));
448            cf.set(b"StdCF".to_vec(), Object::Dict(std_cf));
449            e.set(b"CF".to_vec(), Object::Dict(cf));
450            e.set(b"StmF".to_vec(), Object::name("StdCF"));
451            e.set(b"StrF".to_vec(), Object::name("StdCF"));
452            if !d.encrypt_metadata {
453                e.set(b"EncryptMetadata".to_vec(), Object::Bool(false));
454            }
455        }
456        if d.v >= 5 {
457            e.set(b"OE".to_vec(), s(&d.oe));
458            e.set(b"UE".to_vec(), s(&d.ue));
459            e.set(b"Perms".to_vec(), s(&d.perms));
460        }
461        if let Some(Object::Ref(old)) = self.trailer.get(b"Encrypt").cloned() {
462            self.free(old);
463        }
464        let r = self.add(e);
465        self.trailer.set(b"Encrypt".to_vec(), Object::Ref(r));
466        // Objects are still read with the original handler; saves use the new one.
467        self.out_security = Some(Arc::new(h));
468        self.out_encrypt_num = Some(r.num);
469        self.encryption_changed = true;
470        Ok(())
471    }

473    /// Remove password protection (requires the document to be open — any authenticated
474    /// password; callers should require the owner password, §7.6.4). Applied by a full save.
475    pub fn remove_encryption(&mut self) {
476        if let Some(Object::Ref(r)) = self.trailer.get(b"Encrypt").cloned() {
477            self.free(r);
478        }
479        self.trailer.remove(b"Encrypt");
480        self.out_security = None;
481        self.out_encrypt_num = None;
482        self.encryption_changed = true;
483    }

485    /// Decrypt a freshly parsed indirect object if the document is encrypted.
486    pub(super) fn decrypted(&self, id: ObjRef, o: Object) -> Object {
487        match &self.security {
488            Some(h) if Some(id.num) != self.encrypt_num && !self.encrypt_members.contains(&id.num) => {
489                crate::cos::security::transform(self, h, &o, id.num, id.generation, crate::cos::security::Pass::Decrypt)
490            }
491            _ => o,
492        }
493    }
494}
