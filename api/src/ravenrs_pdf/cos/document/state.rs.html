4thread_local! {
5    /// The objects being loaded on this thread, innermost last (see `Document::try_get`).
6    static LOADING: std::cell::RefCell<Vec<u32>> = const { std::cell::RefCell::new(Vec::new()) };
7    /// Indirect `/Length` values resolved during the current outermost load: a damaged file
8    /// retries a parse several ways, and each retry would resolve the same lengths again
9    /// (exponential in the chain of streams whose lengths point at further streams). Each is
10    /// kept with whether resolving it read unsaved edits.
11    pub(super) static LENGTHS: std::cell::RefCell<HashMap<ObjRef, Length>> = std::cell::RefCell::new(HashMap::new());
12    /// What the loads running on this thread have read ([`ReadFrame`]): how many frames are
13    /// open, the object numbers read inside them, and how often that list was full.
14    static READS: std::cell::RefCell<(u32, Vec<u32>, u64)> = const { std::cell::RefCell::new((0, Vec::new(), 0)) };
15    /// Unsaved edits read on this thread (`Document::try_get`, and a `/Length` in `Document::load`).
16    static OVERLAY_READS: std::cell::Cell<u64> = const { std::cell::Cell::new(0) };
17    /// How deep `Document::scan_for` is nested on this thread, and how many indirect `/Length`
18    /// values the outermost scan and those inside it have resolved ([`ScanLengths`]).
19    static SCAN_LENGTHS: std::cell::Cell<(u32, u32)> = const { std::cell::Cell::new((0, 0)) };
20}

22/// The indirect `/Length` values a scan of the file for an object may resolve, shared with the
23/// scans it leads to. Each header in the file that starts with the object's number is parsed, and
24/// a `/Length` that names an object the cross-reference table cannot place scans the file for that
25/// one: a damaged file with many such headers, each naming another missing object, would read the
26/// whole file once per header. Past the budget a `/Length` is left unresolved, and the stream is
27/// read as before resolution, by its `endstream`.
28pub(super) struct ScanLengths;
29
30impl ScanLengths {
31    /// A real object is defined once per revision, and its `/Length` once per definition.
32    pub(super) const MAX: u32 = 32;
33
34    /// Marks a scan as running; the outermost one starts the budget afresh.
35    pub(super) fn enter() -> Self {
36        SCAN_LENGTHS.with(|c| {
37            let (depth, used) = c.get();
38            c.set((depth.saturating_add(1), if depth == 0 { 0 } else { used }));
39        });
40        ScanLengths
41    }
42
43    /// Whether one more `/Length` may be resolved, counting it if so.
44    pub(super) fn take(&self) -> bool {
45        SCAN_LENGTHS.with(|c| {
46            let (depth, used) = c.get();
47            if used >= Self::MAX {
48                return false;
49            }
50            c.set((depth, used.saturating_add(1)));
51            true
52        })
53    }
54}
55
56impl Drop for ScanLengths {
57    fn drop(&mut self) {
58        SCAN_LENGTHS.with(|c| {
59            let (depth, used) = c.get();
60            c.set((depth.saturating_sub(1), used));
61        });
62    }
63}

69/// The other objects a cached load read: an object stream for a member, an indirect `/Length`,
70/// and what those read in turn. Clones share the caches, so a document whose edits include one of
71/// them reads the object again instead (`Document::try_get`); `None` is a load that read more
72/// than [`Reads::MAX`], which such a document always reads again.
73#[derive(Clone, Debug, Default)]
74pub(super) struct Reads(Option<Arc<[u32]>>);
75
76impl Reads {
77    /// Objects one entry keeps: a member's stream and a length or two, and what they read.
78    const MAX: usize = 64;
79    /// Object numbers the open frames keep between them.
80    const HELD: usize = 4096;
81
82    /// Whether `overlay` (a document's edits) holds any of these objects.
83    pub(super) fn touch(&self, overlay: &BTreeMap<u32, Slot>) -> bool {
84        !overlay.is_empty() && self.0.as_ref().is_none_or(|nums| nums.iter().any(|n| overlay.contains_key(n)))
85    }
86
87    fn bytes(&self) -> usize {
88        self.0.as_ref().map_or(0, |nums| nums.len().saturating_mul(std::mem::size_of::<u32>()))
89    }
90}

92/// Counts `num` as read by the loads open on this thread, with what loading it read (`reads`).
93pub(super) fn note_read(num: u32, reads: Option<&Reads>) {
94    READS.with(|r| {
95        let (depth, held, full) = &mut *r.borrow_mut();
96        if *depth == 0 {
97            return;
98        }
99        let more = reads.map_or(Some(&[][..]), |r| r.0.as_deref());
100        match more {
101            Some(more) if held.len().saturating_add(more.len()) < Reads::HELD => {
102                held.push(num);
103                held.extend_from_slice(more);
104            }
105            _ => *full = full.wrapping_add(1),
106        }
107    });
108}

110/// Collects what one load reads, for [`Reads`]; frames nest, and each read counts for every
111/// frame open around it.
112pub(super) struct ReadFrame {
113    start: usize,
114    full: u64,
115}
116
117impl ReadFrame {
118    pub(super) fn enter() -> Self {
119        READS.with(|r| {
120            let (depth, held, full) = &mut *r.borrow_mut();
121            *depth = depth.saturating_add(1);
122            ReadFrame { start: held.len(), full: *full }
123        })
124    }
125
126    /// What was read since the frame opened.
127    pub(super) fn reads(&self) -> Reads {
128        READS.with(|r| {
129            let (_, held, full) = &*r.borrow();
130            if *full != self.full {
131                return Reads(None);
132            }
133            let mut nums = held.get(self.start..).unwrap_or_default().to_vec();
134            nums.sort_unstable();
135            nums.dedup();
136            Reads((nums.len() <= Reads::MAX).then(|| Arc::from(nums)))
137        })
138    }
139}
140
141impl Drop for ReadFrame {
142    fn drop(&mut self) {
143        READS.with(|r| {
144            let (depth, held, _) = &mut *r.borrow_mut();
145            *depth = depth.saturating_sub(1);
146            if *depth == 0 {
147                held.clear();
148            }
149        });
150    }
151}

162/// Marks an object as being loaded on this thread. Loading can re-enter (indirect stream
163/// `/Length`, object streams): an object already being loaded is a cycle, and the nesting is
164/// bounded as well.
165pub(super) struct LoadGuard;
166
167impl LoadGuard {
168    const MAX: usize = 32;
169
170    pub(super) fn enter(num: u32) -> Option<Self> {
171        LOADING.with(|l| {
172            let mut l = l.borrow_mut();
173            if l.len() >= Self::MAX || l.contains(&num) {
174                return None;
175            }
176            l.push(num);
177            Some(LoadGuard)
178        })
179    }
180}

182/// Whether an object is being loaded on this thread: [`LENGTHS`] holds only while one is, since
183/// it is cleared when the outermost load ends.
184pub(super) fn loading() -> bool {
185    LOADING.with(|l| !l.borrow().is_empty())
186}

191/// The bytes a load result holds of its own ([`held_bytes`]); a failure holds none.
192pub(super) fn loaded_bytes(loaded: &Loaded, data: &Arc<Vec<u8>>) -> usize {
193    loaded.as_ref().map_or(0, |o| held_bytes(o, data, 0))

196impl Drop for LoadGuard {
197    fn drop(&mut self) {
198        let outermost = LOADING.with(|l| {
199            let mut l = l.borrow_mut();
200            l.pop();
201            l.is_empty()
202        });
203        if outermost {
204            LENGTHS.with(|m| m.borrow_mut().clear());
205        }
206    }

220/// Where member `num` of an object stream starts: its `offset` past `/First` (`first`), when the
221/// number is an object number and the start lies inside the decoded `len` bytes. Checked on every
222/// target, so a 32-bit build neither truncates nor wraps a large `/First` or offset (#1815).
223pub(super) fn objstm_member(num: i64, offset: i64, first: Option<usize>, len: usize) -> Option<(u32, usize)> {
224    let num = u32::try_from(num).ok()?;
225    let start = first?.checked_add(usize::try_from(offset).ok()?)?;
226    (start < len).then_some((num, start))
227}

229/// Objects the reader could not read, each with the reason (see `Document::unreadable_objects`).
230/// Bounded: a document records past [`Unreadable::MAX`] of the objects it shows only the fact that
231/// there were more. Entries a clone has edited are hidden from it and do not count, so the whole
232/// list holds at most `MAX` plus the objects edited in clones.
233#[derive(Debug, Default)]
234pub(super) struct Unreadable {
235    pub(super) objects: BTreeMap<u32, (u16, String)>,
236    pub(super) more: bool,
237}
238
239impl Unreadable {
240    pub(super) const MAX: usize = 100;
241    /// Characters of a reason kept: a parser's message can quote the bytes it stopped at.
242    const REASON: usize = 160;
243
244    /// Whether `num` is to be recorded: not listed yet and there is room among the entries the
245    /// asking document still shows (`hidden` are the ones it has edited, as in [`Self::lines`]),
246    /// so a clone's edits never use up its own allowance. A full list notes that there were more,
247    /// so a caller asks this before building the reason.
248    pub(super) fn wants(&mut self, num: u32, hidden: impl Fn(u32) -> bool) -> bool {
249        if self.objects.contains_key(&num) {
250            return false;
251        }
252        if self.objects.keys().filter(|n| !hidden(**n)).count() >= Self::MAX {
253            self.more = true;
254            return false;
255        }
256        true
257    }
258
259    /// Record `num` once; the first reason given is kept.
260    pub(super) fn note(&mut self, num: u32, generation: u16, reason: &str, hidden: impl Fn(u32) -> bool) {
261        if self.wants(num, hidden) {
262            self.objects.insert(num, (generation, reason.chars().take(Self::REASON).collect()));
263        }
264    }
265
266    /// One line per recorded object not `hidden`, then one saying there were more, if there were.
267    pub(super) fn lines(&self, hidden: impl Fn(u32) -> bool) -> Vec<String> {
268        // Clones share the list and each hides its own edits, so the list may hold more than
269        // `MAX` entries in all; each document still shows at most `MAX`.
270        let mut shown = self.objects.iter().filter(|(num, _)| !hidden(**num));
271        let mut lines: Vec<String> = shown
272            .by_ref()
273            .take(Self::MAX)
274            .map(|(num, (generation, reason))| format!("object {num} {generation} could not be read: {reason}"))
275            .collect();
276        if self.more || shown.next().is_some() {
277            lines.push(format!("more objects could not be read; only the first {} are listed", Self::MAX));
278        }
279        lines
280    }
281}

307/// What a document reads as: its source ([`SourceIdentity`]), its trailer and every object changed
308/// since it was opened. A clone, or a snapshot that undo restores, has the same state, and every
309/// change makes a different one. Changed objects are compared as allocations, which the state
310/// keeps alive, so a later object can't take an address one of them had.
311#[derive(Clone)]
312pub struct ReadState {
313    pub(super) source: SourceIdentity,
314    pub(super) trailer: Dict,
315    pub(super) overlay: BTreeMap<u32, Slot>,
316}
317
318impl ReadState {
319    /// Whether `doc` reads as this state.
320    pub fn matches(&self, doc: &Document) -> bool {
321        self.source == doc.source_identity()
322            && self.trailer == doc.trailer
323            && self.overlay.len() == doc.overlay.len()
324            && self.overlay.iter().zip(&doc.overlay).all(|((a, x), (b, y))| {
325                a == b
326                    && match (x, y) {
327                        (Slot::Set(g, o), Slot::Set(h, p)) => g == h && Arc::ptr_eq(o, p),
328                        (Slot::RepairedSet(g, a, o), Slot::RepairedSet(h, b, p)) => g == h && a == b && Arc::ptr_eq(o, p),
329                        (Slot::Freed(g), Slot::Freed(h)) => g == h,
330                        _ => false,
331                    }
332            })
333    }
334}
335
336impl std::fmt::Debug for ReadState {
337    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
338        f.debug_struct("ReadState").field("changed_objects", &self.overlay.len()).finish_non_exhaustive()
339    }
340}

342/// A temporary reader for whole-document work. Its caches never populate the editor or
343/// undo snapshots. Returned objects survive cache eviction; a stream read from the file is a
344/// view into the input bytes (it keeps the whole input alive while held, not a copy).
345pub(crate) struct ObjectReader {
346    pub(crate) document: Document,
347    pub(super) decoded_count: usize,
348}
349
350impl ObjectReader {
351    pub(super) const OBJECTS: usize = 128;
352    pub(super) const STREAM_BYTES: usize = 8 * 1024 * 1024;
353
354    pub(crate) fn get(&mut self, reference: ObjRef) -> Arc<Object> {
355        self.try_get(reference).unwrap_or_else(|_| Arc::new(Object::Null))
356    }
357
358    pub(crate) fn try_get(&mut self, reference: ObjRef) -> Result<Arc<Object>, CosError> {
359        let object = self.document.try_get_reference(reference);
360        // A missed eviction here only costs a re-parse, so never fail the read on a poisoned
361        // lock. The editing-cache evictors propagate instead; they guard user edits.
362        if let Ok(mut streams) = self.document.objstms.lock() {
363            // Only insertion changes this private cache between reads. Most neighboring
364            // objects reuse a decoded stream: do not sum the whole store for every object.
365            if streams.len() != self.decoded_count {
366                let bytes = streams.values().fold(0usize, |bytes, stream| {
367                    bytes
368                        .saturating_add(stream.data.capacity())
369                        .saturating_add(stream.index.capacity().saturating_mul(std::mem::size_of::<Option<(u32, usize)>>()))
370                });
371                if bytes > Self::STREAM_BYTES || streams.len() >= Self::OBJECTS {
372                    // Retain one oversized active stream so its next object does not require
373                    // decoding it again. That stream still obeys the document's decode cap.
374                    let current = match self.document.xref_entry(reference.num) {
375                        Some(XrefEntry::InStream { stream, .. }) => Some(stream),
376                        _ => None,
377                    };
378                    streams.retain(|num, _| Some(*num) == current);
379                }
380                self.decoded_count = streams.len();
381            }
382        }
383        // Parsed objects stay cached (an indirect /Length or a dictionary read again costs a
384        // lookup, not a parse) until the cache passes its object count or holds more than
385        // STREAM_BYTES of data of its own. Streams read from the file are views into the
386        // input bytes, which the document keeps anyway, so they don't count; decrypted
387        // streams and strings do. Checked after failed loads too: resolving a /Length may
388        // have cached large objects.
389        if let Ok(mut cache) = self.document.cache.lock()
390            && (cache.len() >= Self::OBJECTS || cache.owned() > Self::STREAM_BYTES)
391        {
392            cache.clear();
393        }
394        object
395    }
396}

404/// What loading each object gave (a failure as well), by number, each with what it holds ([`held_bytes`]) measured once, when it goes
405/// in, and their sum kept beside them.
406///
407/// Both caches are bounded by those bytes. Summing them on every read walked every cached object
408/// whole each time: a file whose page dictionaries carry kilobytes of private data (InDesign's
409/// `/PieceInfo`) took seconds to open, almost all of it re-measuring objects already measured
410/// (#5873).
411#[derive(Default)]
412pub(crate) struct ObjectCache {
413    entries: HashMap<u32, (Loaded, usize, Reads)>,
414    owned: usize,
415}
416
417impl ObjectCache {
418    /// What loading `num` gave, and what that load read.
419    pub(super) fn get(&self, num: &u32) -> Option<(&Loaded, &Reads)> {
420        self.entries.get(num).map(|(object, _, reads)| (object, reads))
421    }
422
423    #[cfg(test)]
424    pub(crate) fn contains_key(&self, num: &u32) -> bool {
425        self.entries.contains_key(num)
426    }
427
428    /// Keep what loading `num` gave, which holds `owned` bytes ([`loaded_bytes`]), and what it read.
429    pub(super) fn insert(&mut self, num: u32, object: Loaded, owned: usize, reads: Reads) {
430        let owned = owned.saturating_add(reads.bytes());
431        if let Some((_, before, _)) = self.entries.insert(num, (object, owned, reads)) {
432            self.owned = self.owned.saturating_sub(before);
433        }
434        self.owned = self.owned.saturating_add(owned);
435    }
436
437    pub(crate) fn len(&self) -> usize {
438        self.entries.len()
439    }
440
441    #[cfg(test)]
442    pub(crate) fn is_empty(&self) -> bool {
443        self.entries.is_empty()
444    }
445
446    /// The bytes the cached objects hold between them.
447    pub(crate) fn owned(&self) -> usize {
448        self.owned
449    }
450
451    pub(crate) fn clear(&mut self) {
452        self.entries.clear();
453        self.owned = 0;
454    }
455
456    pub(crate) fn retain(&mut self, mut keep: impl FnMut(&u32) -> bool) {
457        self.entries.retain(|num, _| keep(num));
458        self.owned = self.entries.values().fold(0usize, |n, (_, owned, _)| n.saturating_add(*owned));
459    }
460
461    #[cfg(test)]
462    pub(crate) fn values(&self) -> impl Iterator<Item = &Arc<Object>> {
463        self.entries.values().filter_map(|(object, _, _)| object.as_ref().ok())
464    }
465}

467/// Nesting [`held_bytes`] walks: past the parser's own nesting limit, so every parsed object
468/// is measured in full.
469const OWNED_BYTES_DEPTH: usize = 256;
470
471/// What keeping `object` parsed costs: string and stream data that is not a view into `data`
472/// (the input file), and the parsed tree around it -- an [`Object`] per value and a key per
473/// dictionary entry. An object nested deeper than [`OWNED_BYTES_DEPTH`] cannot be measured and
474/// counts as too large to keep.
475///
476/// The tree counts. Measured by its strings and streams alone, a page dictionary carrying
477/// InDesign's `/PieceInfo` -- thousands of small numeric arrays, parsed into tens of thousands of
478/// values -- weighed nothing, so the caches kept a thousand of them: a 35 MB file held 1.5 GB
479/// (#5873).
480pub(crate) fn held_bytes(object: &Object, data: &Arc<Vec<u8>>, depth: usize) -> usize {
481    if depth > OWNED_BYTES_DEPTH {
482        return usize::MAX;
483    }
484    let node = std::mem::size_of::<Object>();
485    // A key, its slot beside the value, and room in the index a large dictionary keeps.
486    let entry = |key: &[u8]| key.len().saturating_add(std::mem::size_of::<(crate::cos::Name, Object)>()).saturating_add(16);
487    let dict = |d: &crate::cos::Dict, start: usize| {
488        d.iter().fold(start, |n, (k, v)| n.saturating_add(entry(k)).saturating_add(held_bytes(v, data, depth + 1)))
489    };
490    match object {
491        Object::String(s) => node.saturating_add(s.bytes.len()),
492        Object::Name(n) => node.saturating_add(n.len()),
493        Object::Stream(s) => dict(&s.dict, node.saturating_add(if s.raw.shares(data) { 0 } else { s.raw.len() })),
494        Object::Array(a) => a.iter().fold(node, |n, v| n.saturating_add(held_bytes(v, data, depth + 1))),
495        Object::Dict(d) => dict(d, node),
496        _ => node,
497    }
498}
