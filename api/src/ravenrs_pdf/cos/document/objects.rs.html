68    /// Where each revision of the file ends (oldest first): the byte after the `%%EOF` (and its
69    /// end-of-line) that closes its cross-reference section. `data[..end]` is that revision as it
70    /// was saved. A linearized file's first-page section belongs to the revision it precedes, so
71    /// ends only ever increase. Empty for a reconstructed file.
72    pub fn revision_ends(&self) -> Vec<usize> {
73        let d = &self.data[..];
74        let mut ends: Vec<usize> = Vec::with_capacity(self.revisions.len());
75        for r in self.revisions.iter() {
76            let from = (r.xref_offset as usize).saturating_add(self.header_offset).min(d.len());
77            let Some(at) = d[from..].windows(5).position(|w| w == b"%%EOF") else { continue };
78            let mut end = from + at + 5;
79            while end < d.len() && matches!(d[end], b'\r' | b'\n') && end - (from + at + 5) < 2 {
80                end += 1;
81            }
82            match ends.last_mut() {
83                Some(last) if end <= *last => {}
84                _ => ends.push(end),
85            }
86        }
87        ends
88    }

113    /// Record `num` as unreadable. `reason` is built only for an object not yet listed, so a file
114    /// whose broken object is asked for again and again pays for it once.
115    fn note_unreadable(&self, num: u32, reason: impl FnOnce() -> String) {
116        // A poisoned lock only means the object goes unlisted; the read itself still fails.
117        let Ok(mut unreadable) = self.unreadable.lock() else { return };
118        let hidden = |n: u32| self.overlay.contains_key(&n);
119        if unreadable.wants(num, hidden) {
120            unreadable.note(num, self.generation(num), &reason(), hidden);
121        }

170    /// Whether the cross-reference entry of `num` locates that object itself: its `num G obj`
171    /// header at exactly the recorded offset, or the member with that number at the recorded
172    /// position of its object stream. When it does not, [`Self::get`] still finds the object by
173    /// searching, but another reader may resolve the number differently. Ignores unsaved edits.
174    pub fn xref_entry_is_exact(&self, num: u32) -> bool {
175        match self.xref_entry(num) {
176            None | Some(XrefEntry::Free { .. }) => true,
177            Some(XrefEntry::InFile { offset, generation }) => {
178                let at = |off: usize| matches!(parse_indirect_shared(&self.data, off, &|_| None, &self.endstreams), Ok((id, _)) if id == ObjRef::new(num, generation));
179                // Offsets may count from a header that does not start the file.
180                usize::try_from(offset).is_ok_and(|off| at(off) || (self.header_offset != 0 && at(off.saturating_add(self.header_offset))))
181            }
182            Some(XrefEntry::InStream { stream, index }) => self
183                .objstm(stream)
184                .is_ok_and(|stm| usize::try_from(index).ok().and_then(|i| stm.index.get(i)).is_some_and(|m| m.is_some_and(|(n, _)| n == num))),
185        }
186    }

206    pub(super) fn reference_matches(&self, r: ObjRef) -> bool {
207        match self.overlay.get(&r.num) {
208            Some(Slot::Set(generation, _)) => r.generation == *generation,
209            Some(Slot::RepairedSet(generation, alias, _)) => r.generation == *generation || r.generation == *alias,
210            Some(Slot::Freed(_)) => false,
211            None => match self.entries.lookup(r.num, &self.data) {
212                Some(XrefEntry::InFile { offset, generation }) => {
213                    r.generation == generation || self.indexed_header_reference(r.num, offset) == Some(r)
214                }
215                Some(XrefEntry::InStream { .. }) => r.generation == 0,
216                None | Some(XrefEntry::Free { .. }) => false,
217            },
218        }
219    }

221    /// The generations a reference to `num` may carry and still resolve (see [`Self::get`]),
222    /// sorted: none for a free or unknown slot, else the slot's generation, and for a damaged
223    /// file slot (or a repair-aware edit of one) also the generation its object header carries.
224    /// Code comparing revisions (signature validation) uses it to tell whether the references
225    /// that reached an object in one revision still reach the same slot in another.
226    pub fn accepted_generations(&self, num: u32) -> Vec<u16> {
227        // Mirrors `reference_matches`, which stays allocation-free for every lookup.
228        let mut generations = match self.overlay.get(&num) {
229            Some(Slot::Set(generation, _)) => vec![*generation],
230            Some(Slot::RepairedSet(generation, alias, _)) => vec![*generation, *alias],
231            Some(Slot::Freed(_)) => Vec::new(),
232            None => match self.entries.lookup(num, &self.data) {
233                Some(XrefEntry::InFile { offset, generation }) => {
234                    std::iter::once(generation).chain(self.indexed_header_reference(num, offset).map(|h| h.generation)).collect()
235                }
236                Some(XrefEntry::InStream { .. }) => vec![0],
237                None | Some(XrefEntry::Free { .. }) => Vec::new(),
238            },
239        };
240        generations.sort_unstable();
241        generations.dedup();
242        generations
243    }

245    pub(super) fn try_get_reference(&self, r: ObjRef) -> Result<Arc<Object>, CosError> {
246        if !self.reference_matches(r) {
247            // An edit of `r.num` can make it match, so a load that read null here read it too;
248            // with `r.num` edited (freed, or set at another generation) the null is the edit's.
249            if self.overlay.contains_key(&r.num) {
250                note_overlay_read();
251            }
252            note_read(r.num, None);
253            return Ok(Arc::new(Object::Null));
254        }
255        self.try_get(r.num)
256    }

258    /// A repair probe, not a whole-file search: only the indexed header (or the same offset
259    /// relative to a shifted PDF header) can establish this additional generation. A small
260    /// window bounds hostile whitespace/tokens; ordinary xref-matching reads are unchanged.
261    pub(super) fn indexed_header_reference(&self, num: u32, offset: u64) -> Option<ObjRef> {
262        let at = |off: usize| {
263            let bytes = self.data.get(off..off.saturating_add(128).min(self.data.len()))?;
264            let mut lexer = Lexer::new(bytes, 0);
265            lexer.skip_ws();
266            let number = std::str::from_utf8(lexer.token()).ok()?.parse::<u32>().ok()?;
267            lexer.skip_ws();
268            let generation = std::str::from_utf8(lexer.token()).ok()?.parse::<u16>().ok()?;
269            (number == num && lexer.eat_keyword(b"obj")).then_some(ObjRef::new(number, generation))
270        };
271        let off = usize::try_from(offset).ok()?;
272        at(off).or_else(|| if self.header_offset == 0 { None } else { off.checked_add(self.header_offset).and_then(at) })
273    }

275    /// Add at most one bounded diagnostic per object. False stops diagnostic probing only;
276    /// parsers must still read the remaining xref rows needed to build their index.
277    pub(super) fn inspect_xref_generation_repair(&self, num: u32, entry: XrefEntry, recorded: &mut HashSet<u32>, log: &mut Vec<String>) -> bool {
278        let XrefEntry::InFile { offset, generation } = entry else { return true };
279        if recorded.contains(&num) {
280            return true;
281        }
282        let Some(header) = self.indexed_header_reference(num, offset) else { return true };
283        if header.generation == generation {
284            return true;
285        }
286        if recorded.len() == 256 {
287            log.push("additional xref/header generation disagreements omitted after 256 objects".into());
288            return false;
289        }
290        recorded.insert(num);
291        log.push(format!("object {num}: xref generation {generation} disagrees with object header generation {}", header.generation));
292        true
293    }

295    /// Diagnose once at open without expanding compact ranges or decoding an xref twice.
296    /// Materialized readers retain bounded notes while they read accepted sections, before
297    /// newer rows hide older ones. Discarded parse attempts/rebuilt indexes carry no notes.
298    pub(super) fn note_xref_generation_repairs(&self, log: &mut Vec<String>) {
299        match &*self.entries {
300            XrefIndex::Materialized(_, notes) => log.extend(notes.iter().cloned()),
301            XrefIndex::Compact { revisions, .. } => {
302                let mut recorded = HashSet::new();
303                for revision in revisions {
304                    for section in std::iter::once(&revision.primary).chain(revision.hybrid.iter()) {
305                        for num in section.numbers() {
306                            if let Some(entry) = section.lookup(num, &self.data)
307                                && !self.inspect_xref_generation_repair(num, entry, &mut recorded, log)
308                            {
309                                return;
310                            }
311                        }
312                    }
313                }
314            }
315        }
316    }

318    /// Follow a reference if `o` is one; otherwise return `o` itself. An object can be nothing but
319    /// a reference to another, which is followed too, up to [`MAX_REFERENCE_HOPS`] of them.
320    pub fn resolve(&self, o: &Object) -> Arc<Object> {
321        let Object::Ref(r) = o else { return Arc::new(o.clone()) };
322        let mut held = self.get(*r);
323        for _ in 0..MAX_REFERENCE_HOPS {
324            let Object::Ref(r) = &*held else { break };
325            held = self.get(*r);
326        }
327        held
328    }

330    /// `s` with the `/DecodeParms` its filters use made direct, so [`Stream::decoded`] reads them:
331    /// an indirect dictionary resolved, and of an array (itself perhaps indirect), the entries at
332    /// the filters' places (`/Crypt` counts, ISO 32000-2 §7.4.1) resolved, the rest dropped. Of
333    /// each dictionary only the entries the decoders read are kept, numbers and flags given by
334    /// reference (`/Predictor 7 0 R`) resolved; `/JBIG2Globals` stays a reference. A `/Crypt`
335    /// entry, which no decoder reads ([`Stream::filters`] leaves it out), is null. `s` itself
336    /// when none of those is indirect, or when the other filters are more than any decoder takes
337    /// ([`crate::filters::MAX_CHAIN`]). Only small values are copied, so a file can't have a
338    /// large object copied once for each key and filter.
339    pub fn with_direct_parms<'s>(&self, s: &'s Stream) -> Cow<'s, Stream> {
340        let Some(made) = self.direct_parms(s) else { return Cow::Borrowed(s) };
341        let mut s = s.clone();
342        s.dict.set(b"DecodeParms".to_vec(), made);
343        Cow::Owned(s)
344    }

346    /// [`Document::with_direct_parms`] for decoding alone: where that copies `s`, this makes a
347    /// stream of `s`'s data, shared, and a dictionary of its `/Filter` and the parameters made
348    /// direct, all its filters read. A stream named again and again, as a page's `/Contents` may
349    /// list one, isn't copied whole each time.
350    pub(crate) fn decoding_view<'s>(&self, s: &'s Stream) -> Cow<'s, Stream> {
351        let Some(made) = self.direct_parms(s) else { return Cow::Borrowed(s) };
352        let mut dict = Dict::new();
353        if let Some(f) = s.dict.get(b"Filter") {
354            dict.set(b"Filter".to_vec(), f.clone());
355        }
356        dict.set(b"DecodeParms".to_vec(), made);
357        Cow::Owned(Stream { dict, raw: s.raw.clone() })
358    }

360    /// The `/DecodeParms` [`Document::with_direct_parms`] gives `s`, or `None` when it keeps `s`.
361    fn direct_parms(&self, s: &Stream) -> Option<Object> {
362        let p = s.dict.get(b"DecodeParms")?;
363        let crypt = |f: &Object| matches!(f.as_name(), Some(b"Crypt"));
364        let filters: Vec<bool> = match s.dict.get(b"Filter") {
365            Some(Object::Array(a)) => a.iter().map(crypt).collect(),
366            Some(f) => vec![crypt(f)],
367            None => Vec::new(),
368        };
369        if filters.iter().filter(|c| !**c).count() > crate::filters::MAX_CHAIN {
370            return None;
371        }
372        let mut changed = matches!(p, Object::Ref(_));
373        // Its numbers and flags kept, resolved; anything else is no filter's parameters (null).
374        let one = |e: &Object, changed: &mut bool| -> Object {
375            *changed |= matches!(e, Object::Ref(_));
376            let Object::Dict(d) = &*self.held(e) else { return Object::Null };
377            let mut made = Dict::new();
378            for key in DECODE_PARMS_KEYS {
379                let Some(v) = d.get(key) else { continue };
380                *changed |= matches!(v, Object::Ref(_));
381                let v = self.held(v);
382                if matches!(*v, Object::Int(_) | Object::Real(_) | Object::Bool(_)) {
383                    made.set(key.to_vec(), (*v).clone());
384                }
385            }
386            if let Some(g @ Object::Ref(_)) = d.get(b"JBIG2Globals") {
387                made.set(b"JBIG2Globals".to_vec(), g.clone());
388            }
389            Object::Dict(made)
390        };
391        let made = match &*self.held(p) {
392            Object::Array(a) => {
393                Object::Array(a.iter().zip(&filters).map(|(e, crypt)| if *crypt { Object::Null } else { one(e, &mut changed) }).collect())
394            }
395            _ => one(p, &mut changed),
396        };
397        changed.then_some(made)
398    }

413    /// The CCITT parameters of image `s` when its last filter is `CCITTFaxDecode`: its
414    /// `/DecodeParms` (resolved; the dictionary of a lone filter, or its entry in an array), with the
415    /// defaults of ISO 32000-2 Table 11 for what that leaves out. `None` for another image, and for
416    /// one whose `/Filter` isn't a name or an array of names.
417    pub fn fax_params(&self, s: &Stream) -> Option<crate::filters::CcittParams> {
418        let chain = s.readable_filters().ok()?;
419        if !matches!(chain.last(), Some((crate::filters::Filter::CcittFax, _))) {
420            return None;
421        }
422        let parms = self.last_filter_parms(s);
423        let int = |key: &[u8], default: i64| parms.get(key).and_then(|v| self.resolve(v).as_int()).unwrap_or(default);
424        let flag = |key: &[u8], default: bool| match parms.get(key).map(|v| self.resolve(v)).as_deref() {
425            Some(Object::Bool(b)) => *b,
426            _ => default,
427        };
428        let d = crate::filters::CcittParams::default();
429        Some(crate::filters::CcittParams {
430            k: int(b"K", d.k),
431            columns: int(b"Columns", d.columns),
432            rows: int(b"Rows", d.rows),
433            end_of_line: flag(b"EndOfLine", d.end_of_line),
434            encoded_byte_align: flag(b"EncodedByteAlign", d.encoded_byte_align),
435            end_of_block: flag(b"EndOfBlock", d.end_of_block),
436            black_is_1: flag(b"BlackIs1", d.black_is_1),
437        })

440    /// The `/DecodeParms` dictionary of stream `s`'s last filter, resolved: that of a lone filter,
441    /// or its entry in an array. Empty when there is none.
442    fn last_filter_parms(&self, s: &Stream) -> Dict {
443        // Its place in /Filter, /Crypt included, pairs it with its /DecodeParms entry (§7.4.1): the
444        // last filter but a /Crypt, which `filters` leaves out wherever it stands.
445        let index = match s.dict.get(b"Filter") {
446            Some(Object::Array(a)) => a.iter().rposition(|f| f.as_name() != Some(b"Crypt".as_slice())).unwrap_or(0),
447            _ => 0,
448        };
449        match s.dict.get(b"DecodeParms").map(|p| self.resolve(p)).as_deref() {
450            Some(Object::Array(a)) => a.get(index).and_then(|p| self.dict(p)),
451            Some(p) if index == 0 => p.as_dict().cloned(),
452            _ => None,
453        }
454        .unwrap_or_default()
455    }

471    /// The samples of image `s`, decoded within `max` bytes ([`MAX_DECODED`] at most), spending from
472    /// `work`: see [`ImageBudget::decode`](crate::cos::ImageBudget::decode), the way in from outside this
473    /// crate::cos, which says what this decodes and how.
474    pub(crate) fn decoded_image_spending(&self, s: &Stream, max: usize, work: &mut u64) -> Result<Vec<u8>, CosError> {
475        let max = max.min(MAX_DECODED);
476        let s = &*self.with_direct_parms(s);
477        let filter = |e: crate::filters::FilterError| CosError::Filter(e.to_string());
478        let chain = s.readable_filters().map_err(filter)?;
479        // The filters before the image codec, such as ASCII85Decode, are undone first.
480        let undo = |before: &[(crate::filters::Filter, crate::filters::Params)], work: &mut u64| match before {
481            [] => Ok(Cow::Borrowed(&s.raw[..])),
482            _ => crate::filters::decode_tolerant_within(before, &s.raw, max, work).map(Cow::Owned).map_err(filter),
483        };
484        // `/Width` and `/Height` are read as given directly, as every caller reads them first: one
485        // given by reference has the image refused there, before it gets here.
486        let height = s.dict.int(b"Height").and_then(|h| u32::try_from(h).ok());
487        if let Some(((crate::filters::Filter::Jbig2, _), before)) = chain.split_last() {
488            let (Some(columns), Some(height)) = (s.dict.int(b"Width").and_then(|w| u32::try_from(w).ok()), height) else {
489                return Err(CosError::Filter("a JBIG2 image's /Width or /Height isn't a size".into()));
490            };
491            // Nothing to read for what can't be decoded: the images of a file share its globals.
492            let globals = if *work == 0 { None } else { self.jbig2_globals(s, work)? };
493            let data = if *work == 0 { Cow::Borrowed(&[][..]) } else { undo(before, work)? };
494            return crate::filters::decode_jbig2_within(&data, globals.as_deref().map(Vec::as_slice), columns, height, max, work).map_err(filter);
495        }
496        let Some(p) = self.fax_params(s) else {
497            // The last filter gives the image's rows: rows without the tags of the PNG predictor
498            // they declare are read as they are (`crate::filters::decode_image_tolerant_within`).
499            return match height {
500                Some(rows) => s.decoded_image_rows_spending(max, rows, work),
501                None => s.decoded_spending(max, work),
502            };
503        };
504        // As many rows as the image has: a /Rows below /Height would cut the picture short, and
505        // rows past /Height aren't drawn.
506        let p = crate::filters::CcittParams { rows: 0, ..p };
507        if s.dict.int(b"Width") != Some(p.columns) {
508            return Err(CosError::Filter("a fax image's /Columns aren't its /Width".into()));
509        }
510        let stride = usize::try_from(p.columns).unwrap_or(usize::MAX).div_ceil(8).max(1);
511        let rows = u32::try_from(max / stride).unwrap_or(u32::MAX).min(height.unwrap_or(0));
512        let data = undo(chain.split_last().map_or(&[][..], |(_, before)| before), work)?;
513        crate::filters::decode_ccitt_within(&data, &p, rows, work).map_err(filter)
514    }

516    /// What the `/JBIG2Globals` stream of JBIG2 image `s` (in its last filter's `/DecodeParms`)
517    /// decodes to, within [`MAX_JBIG2_GLOBALS`]; `None` without one (a reference to no object is
518    /// one to null, ISO 32000-2 §7.3.10, which is an absent entry, §7.3.7). The image is then read
519    /// without them, as the renderer draws it, so an image that needed them mostly fails to decode
520    /// and can decode to other pixels. A stream that is there and can't be read is an error, rather
521    /// than the image decoded without it.
522    ///
523    /// The images of a scan share one globals stream, so a stream of the file, not edited since
524    /// it was read (the same in every clone) and with its `/DecodeParms` direct (by reference they
525    /// are another object, which can change while this one doesn't), has its filters undone once a
526    /// document and is kept, up to [`JBIG2_GLOBALS_HELD`] in all; others are read each time. That
527    /// saves the filters only: the segments it holds are decoded again for every image. Undoing
528    /// the filters spends from `work`, so globals read for every image pay every time.
529    fn jbig2_globals(&self, s: &Stream, work: &mut u64) -> Result<Option<Arc<Vec<u8>>>, CosError> {
530        let Some(entry) = self.last_filter_parms(s).get(b"JBIG2Globals").cloned() else { return Ok(None) };
531        let held = match entry {
532            Object::Ref(r) if !self.overlay.contains_key(&r.num) => Some(r.num),
533            _ => None,
534        };
535        if let Some(hit) = held.and_then(|num| self.jbig2_globals.lock().ok()?.get(&num).cloned()) {
536            return Ok(Some(hit));
537        }
538        let Object::Stream(g) = &*self.resolve(&entry) else { return Ok(None) };
539        let readable = self.with_direct_parms(g);
540        let held = held.filter(|_| matches!(readable, Cow::Borrowed(_)));
541        let mut data = readable.decoded_spending(MAX_JBIG2_GLOBALS, work).map_err(|e| {
542            let detail = match e {
543                CosError::Filter(detail) => detail,
544                e => e.to_string(),
545            };
546            CosError::Filter(format!("the JBIG2 globals can't be read: {detail}"))
547        })?;
548        // What is kept counts by its length, so that's all it may hold on to.
549        data.shrink_to_fit();
550        let data = Arc::new(data);
551        // A poisoned lock only means nothing is kept.
552        if let (Some(num), Ok(mut kept)) = (held, self.jbig2_globals.lock()) {
553            let total = kept.values().map(|g| g.len()).fold(data.len(), usize::saturating_add);
554            if total <= JBIG2_GLOBALS_HELD {
555                kept.insert(num, data.clone());
556            }
557        }
558        Ok(Some(data))
559    }

561    pub fn try_get(&self, num: u32) -> Result<Arc<Object>, CosError> {
562        if let Some(slot) = self.overlay.get(&num) {
563            note_read(num, None);
564            note_overlay_read();
565            return Ok(match slot {
566                Slot::Set(_, o) | Slot::RepairedSet(_, _, o) => o.clone(),
567                Slot::Freed(_) => Arc::new(Object::Null),
568            });
569        }
570        // Clones share the cache: what a load read that this document has edited since (the
571        // member's object stream, a `/Length`) makes the entry another document's.
572        if let Some((loaded, reads)) = self.cache.lock().map_err(|_| CosError::Poisoned)?.get(&num)
573            && !reads.touch(&self.overlay)
574        {
575            note_read(num, Some(reads));
576            return loaded.clone();
577        }
578        note_read(num, None);
579        // Loading can re-enter (indirect stream `/Length`, object streams); damaged files make
580        // cycles such as `6 0 obj << /Length 6 0 R >>`. Bound the nesting on every path.
581        let _guard =
582            LoadGuard::enter(num).ok_or_else(|| CosError::Syntax { offset: 0, detail: format!("reference cycle while loading object {num}") })?;
583        let _scope = crate::cos::security::FollowScope::enter();
584        let edits_read = overlay_reads();
585        let frame = ReadFrame::enter();
586        let loaded = self.load(num, 0).map(Arc::new);
587        let reads = frame.reads();
588        drop(frame);
589        if let Err(e) = &loaded
590            && !matches!(e, CosError::Poisoned)
591        {
592            self.note_unreadable(num, || e.to_string());
593        }
594        // A failure is kept like an object, or every read of a damaged object loads it again. A
595        // poisoned lock says nothing about the object. Clones share the cache, so a load that read
596        // this document's edits (an edited object stream or `/Length`) is not kept.
597        if overlay_reads() == edits_read && !matches!(loaded, Err(CosError::Poisoned)) {
598            // The editing cache is bounded like the scan reader's: re-parsing a file-backed view
599            // is cheap, and returned objects outlive eviction.
600            self.cache.lock().map_err(|_| CosError::Poisoned)?.insert(num, loaded.clone(), loaded_bytes(&loaded, &self.data), reads);
601            self.evict_editor_cache_if_over_budget(num)?;
602        }
603        loaded
604    }

606    /// Bounds for the editing cache: roomier than the scan reader's (128 objects / 8 MB)
607    /// so interactive editing does not thrash, but still capped so touching a large
608    /// document cannot pin its whole object graph. Only owned (decrypted/copied) bytes
609    /// count; file-backed views re-parse for free.
610    ///
611    /// Worst case per document: input bytes + `EDITOR_CACHE_BYTES` (32 MB) of owned object
612    /// data + `EDITOR_OBJSTM_BYTES` (32 MB) + unsaved `overlay` edits. Each object is measured
613    /// once, as it is cached ([`ObjectCache`](super::state::ObjectCache)), so both caps hold
614    /// after every insertion.
615    /// Clones share these caches, so one clone's eviction drops another's hot set;
616    /// that only costs re-parsing, never correctness. Eviction keeps just the newcomer:
617    /// file-backed views re-parse cheaply and editor working sets stay far below 1024.
618    pub(super) const EDITOR_CACHE_OBJECTS: usize = 1024;
619
620    pub(super) const EDITOR_CACHE_BYTES: usize = 32 * 1024 * 1024;
621
622    /// Object streams decode to megabytes each, so fewer are kept than plain objects.
623    const EDITOR_OBJSTM_OBJECTS: usize = 256;
624
625    const EDITOR_OBJSTM_BYTES: usize = 32 * 1024 * 1024;
626
627    fn evict_editor_cache_if_over_budget(&self, fresh_num: u32) -> Result<(), CosError> {
628        let mut cache = self.cache.lock().map_err(|_| CosError::Poisoned)?;
629        if cache.len() <= Self::EDITOR_CACHE_OBJECTS && cache.owned() <= Self::EDITOR_CACHE_BYTES {
630            return Ok(());
631        }
632        // Eviction keeps just the newcomer: file-backed views re-parse cheaply.
633        cache.retain(|n| *n == fresh_num);
634        Ok(())
635    }
636
637    fn evict_editor_streams_if_over_budget(&self, current: u32) -> Result<(), CosError> {
638        // The caller inserts `current` first, so retaining it keeps a live entry.
639        let over = {
640            let streams = self.objstms.lock().map_err(|_| CosError::Poisoned)?;
641            let bytes = streams.values().fold(0usize, |bytes, stream| {
642                bytes
643                    .saturating_add(stream.data.capacity())
644                    .saturating_add(stream.index.capacity().saturating_mul(std::mem::size_of::<Option<(u32, usize)>>()))
645            });
646            bytes > Self::EDITOR_OBJSTM_BYTES || streams.len() > Self::EDITOR_OBJSTM_OBJECTS
647        };
648        if over {
649            self.objstms.lock().map_err(|_| CosError::Poisoned)?.retain(|n, _| *n == current);
650        }
651        Ok(())
652    }

707    fn load(&self, num: u32, depth: u32) -> Result<Object, CosError> {
708        if depth > 16 {
709            return Err(CosError::Syntax { offset: 0, detail: "reference cycle while loading".into() });
710        }
711        match self.entries.lookup(num, &self.data) {
712            None | Some(XrefEntry::Free { .. }) => Ok(Object::Null),
713            Some(XrefEntry::InFile { offset, .. }) => {
714                let resolve = |r: ObjRef| self.stream_length(r);
715                let off = offset as usize;
716                let at = |off: usize| match parse_indirect_shared(&self.data, off, &resolve, &self.endstreams) {
717                    Ok((id, o)) if id.num == num => Some(self.decrypted(id, o)),
718                    _ => None,
719                };
720                // Offsets relative to a shifted header, or simply wrong: try both fixes.
721                at(off)
722                    .or_else(|| if self.header_offset == 0 { None } else { at(off.saturating_add(self.header_offset)) })
723                    .or_else(|| self.located.get(&num).and_then(|(o, _)| usize::try_from(*o).ok()).and_then(at))
724                    .or_else(|| self.scan_for(num).map(|(id, o)| self.decrypted(id, o)))
725                    .ok_or_else(|| {
726                        self.note_unreadable(num, || self.why_unreadable(off));
727                        CosError::MissingObject(num)
728                    })
729            }
730            Some(XrefEntry::InStream { stream, index }) => {
731                let stm = self.objstm(stream)?;
732                // The member at the entry's position, else wherever the stream lists `num`.
733                let at = usize::try_from(index).ok().and_then(|i| stm.index.get(i)).copied().flatten().filter(|(n, _)| *n == num);
734                let (_, off) = at.or_else(|| stm.index.iter().flatten().find(|(n, _)| *n == num).copied()).ok_or(CosError::MissingObject(num))?;
735                Lexer::new(&stm.data, off).object()
736            }
737        }
738    }

740    pub(super) fn objstm(&self, num: u32) -> Result<Arc<ObjStm>, CosError> {
741        // An edited or freed stream is this document's alone: clones share the parsed streams.
742        let edited = self.overlay.contains_key(&num);
743        if !edited
744            && let Some(s) = self.objstms.lock().map_err(|_| CosError::Poisoned)?.get(&num)
745            && !s.reads.touch(&self.overlay)
746        {
747            note_read(num, Some(&s.reads));
748            return Ok(s.clone());
749        }
750        let frame = ReadFrame::enter();
751        // A file entry that cannot place the stream (one that says it is itself compressed
752        // inside another stream, as in #2367, or points nowhere) is tried by the stream's own
753        // header, as a plain object with a wrong offset is. An edited or freed stream, in the
754        // overlay or in a saved revision's table, is what that says it is: the file's copy
755        // must not come back.
756        let edits_read = overlay_reads();
757        let loaded = match self.try_get(num) {
758            Ok(o) if matches!(*o, Object::Stream(_)) => o,
759            _ if self.overlay.contains_key(&num) || matches!(self.xref_entry(num), Some(XrefEntry::Free { .. })) => {
760                return Err(CosError::MissingObject(num));
761            }
762            _ => Arc::new(self.scan_for(num).map(|(id, o)| self.decrypted(id, o)).ok_or(CosError::MissingObject(num))?),
763        };
764        let Object::Stream(s) = &*loaded else { return Err(CosError::MissingObject(num)) };
765        let n = s.dict.int(b"N").unwrap_or(0).clamp(0, 1_000_000) as usize;
766        let first = usize::try_from(s.dict.int(b"First").unwrap_or(0).max(0)).ok();
767        // Its parameters can be given by reference, and can sit in another object stream. A scan of
768        // the file's object streams comes here without a read around it.
769        let _scope = crate::cos::security::FollowScope::enter();
770        let readable = crate::cos::security::following(Cow::Borrowed(s), || self.with_direct_parms(s));
771        #[cfg(test)]
772        super::tests::OBJSTM_DECODES.with(|decodes| decodes.set(decodes.get() + 1));
773        let data = readable.decoded_within(self.stream_limit)?;
774        let mut lx = Lexer::new(&data, 0);
775        let mut index = Vec::with_capacity(n);
776        for _ in 0..n {
777            let (Some(Object::Int(onum)), Some(Object::Int(off))) = (lx.object().ok(), lx.object().ok()) else { break };
778            index.push(objstm_member(onum, off, first, data.len()));
779        }
780        let stm = Arc::new(ObjStm { index, data: Arc::new(data), reads: frame.reads() });
781        // A stream whose load or decoding read this document's edits (its `/Length`, its
782        // `/DecodeParms`) is as much its own as an edited one.
783        if edited || overlay_reads() != edits_read {
784            return Ok(stm);
785        }
786        // Same bound as the object cache: decoded object streams can be megabytes each.
787        // Insert first so the retained `num` is actually in the map.
788        self.objstms.lock().map_err(|_| CosError::Poisoned)?.insert(num, stm.clone());
789        self.evict_editor_streams_if_over_budget(num)?;
790        Ok(stm)
791    }

793    /// An indirect stream `/Length`, read through the loader's guard against cycles. Kept in
794    /// [`LENGTHS`] only during a load, which clears it when it ends: a scan of the file from
795    /// [`Self::objstm`] can come here without one.
796    fn stream_length(&self, r: ObjRef) -> Option<i64> {
797        if self.overlay.contains_key(&r.num) {
798            note_overlay_read();
799        }
800        let cached = loading();
801        // Each reuse counts as a read of the length and of what resolving it read, and as an edit
802        // read when that went through this document's edits.
803        if cached && let Some((known, edits, reads)) = LENGTHS.with(|m| m.borrow().get(&r).cloned()) {
804            if edits {
805                note_overlay_read();
806            }
807            note_read(r.num, Some(&reads));
808            return known;
809        }
810        note_read(r.num, None);
811        let edits_read = overlay_reads();
812        let frame = ReadFrame::enter();
813        let length = self.try_get_reference(r).ok().and_then(|o| o.as_int());
814        if cached {
815            LENGTHS.with(|m| m.borrow_mut().insert(r, (length, overlay_reads() != edits_read, frame.reads())));
816        }
817        length
818    }

820    /// Last-resort lookup: scan the file for `num G obj`. What isn't there is remembered: each miss
821    /// reads the whole file, and a file can refer to such an object from everywhere.
822    fn scan_for(&self, num: u32) -> Option<(ObjRef, Object)> {
823        if self.unfound.lock().is_ok_and(|unfound| unfound.contains(&num)) {
824            return None;
825        }
826        #[cfg(test)]
827        super::tests::FILE_SCANS.with(|scans| scans.set(scans.get() + 1));
828        let needle = format!("{num} ");
829        let data = &self.data;
830        let budget = ScanLengths::enter();
831        let resolve = |r: ObjRef| if budget.take() { self.stream_length(r) } else { None };
832        let mut found = None;
833        let mut i = 0;
834        while let Some(p) = find(data, needle.as_bytes(), i, data.len()) {
835            i = p + 1;
836            if p > 0 && !is_whitespace(data[p - 1]) {
837                continue;
838            }
839            // An indirect `/Length` is resolved, as at the object's offset, or a stream that ends
840            // in a misspelt `endstream` is searched on into a later object's. A header that parses
841            // as far as the `/Length` starts with `num` itself, after white space. A budget shared
842            // with the scans this leads to bounds how often the file is read again ([`ScanLengths`]).
843            if let Ok((id, o)) = parse_indirect_shared(data, p, &resolve, &self.endstreams)
844                && id.num == num
845            {
846                found = Some((id, o)); // keep the last (newest) definition
847            }
848        }
849        if found.is_none()
850            && let Ok(mut unfound) = self.unfound.lock()
851        {
852            unfound.insert(num);
853        }
854        found
855    }

859    /// Stream data still pointing into another document's file would keep that whole file
860    /// alive for as long as this edit exists: give it a buffer of its own (just its bytes).
861    fn adopt(&self, obj: Object) -> Object {
862        match obj {
863            Object::Stream(mut s) if s.raw.borrows_other_than(&self.data) => {
864                s.raw = s.raw.detached();
865                Object::Stream(s)
866            }
867            other => other,
868        }
869    }

871    /// Replace an object (keeping its generation).
872    pub fn set(&mut self, r: ObjRef, obj: impl Into<Object>) {
873        let obj = self.adopt(obj.into());
874        self.overlay.insert(r.num, Slot::Set(r.generation, Arc::new(obj)));
875        if r.num >= self.next_num {
876            self.next_num = r.num.saturating_add(1);
877        }
878    }

880    /// Replace an accepted existing reference without losing a damaged source slot's aliases.
881    /// Unlike [`Self::set`], this preserves both xref/header generations when they disagree.
882    /// Such an edit forces a full rewrite so the output has consistent references, rather than
883    /// appending a new contradictory xref row. Missing, free and stale references are errors.
884    pub fn set_existing(&mut self, r: ObjRef, obj: impl Into<Object>) -> Result<(), CosError> {
885        if !self.reference_matches(r) {
886            return Err(CosError::MissingObject(r.num));
887        }
888        let alias = self.alternate_generation(r);
889        if let Some(alias) = alias {
890            let obj = self.adopt(obj.into());
891            self.overlay.insert(r.num, Slot::RepairedSet(r.generation, alias, Arc::new(obj)));
892        } else {
893            self.set(r, obj);
894        }
895        Ok(())
896    }

898    /// Other accepted generation for a known-live reference to a damaged slot.
899    pub(super) fn alternate_generation(&self, r: ObjRef) -> Option<u16> {
900        match self.overlay.get(&r.num) {
901            Some(Slot::RepairedSet(generation, alias, _)) => Some(if r.generation == *generation { *alias } else { *generation }),
902            Some(_) => None,
903            None => match self.entries.lookup(r.num, &self.data) {
904                Some(XrefEntry::InFile { offset, generation }) => self.indexed_header_reference(r.num, offset).and_then(|header| {
905                    (header.generation != generation).then_some(if r.generation == generation { header.generation } else { generation })
906                }),
907                _ => None,
908            },
909        }
910    }

920    /// Find a never-used number, retaining the high-water mark for incremental /Size.
921    /// Exhausted hints fall back to gaps without replacing source or overlay objects.
922    fn allocate_num(&mut self) -> u32 {
923        let mut num = if self.next_num == u32::MAX { 1 } else { self.next_num.max(1) };
924        while self.entries.lookup(num, &self.data).is_some() || self.overlay.contains_key(&num) {
925            num = if num == u32::MAX - 1 { 1 } else { num + 1 };
926        }
927        self.next_num = self.next_num.max(num.saturating_add(1));
928        num
929    }

931    /// Delete an object (its number becomes free with the next generation).
932    pub fn free(&mut self, r: ObjRef) {
933        self.overlay.insert(r.num, Slot::Freed(r.generation.saturating_add(1)));
934    }

936    /// Free what was added since the file was read and nothing refers to any more: the objects
937    /// [`SaveOptions::drop_unlinked`](crate::cos::SaveOptions::drop_unlinked) leaves out of an
938    /// incremental save, freed as it writes them (the generation stays). For a writer about to
939    /// add an object that nothing refers to either, which that option would leave out too.
940    pub fn free_unlinked_additions(&mut self) {
941        for num in crate::cos::writer::unlinked_additions(self) {
942            let generation = self.generation(num);
943            self.overlay.insert(num, Slot::Freed(generation));
944        }
945    }

947    /// Record `current`'s object graph as unsaved edits of this document.
948    ///
949    /// Used when the working snapshot was restored from an earlier revision and the next
950    /// incremental save must still append to this file (the last saved revision). Objects
951    /// that already match are left untouched, except members of an object stream of this file
952    /// whose stream number `current` frees or reuses: those are written again on their own.
953    /// Reads go through private readers so neither document's editing cache is filled.
954    pub fn overlay_changes_from(&mut self, current: &Document) -> Result<(), CosError> {
955        let saved_nums = self.object_numbers();
956        let current_nums = current.object_numbers();
957        let mut freed = Vec::new();
958        let mut changed = Vec::new();
959        {
960            let mut saved_reader = self.object_reader();
961            let mut current_reader = current.object_reader();
962            for &num in &saved_nums {
963                if current_nums.binary_search(&num).is_err() {
964                    freed.push(num);
965                }
966            }
967            for &num in &current_nums {
968                let generation = current.generation(num);
969                let same = saved_nums.binary_search(&num).is_ok()
970                    && self.generation(num) == generation
971                    && self.alternate_generation(ObjRef::new(num, generation)) == current.alternate_generation(ObjRef::new(num, generation))
972                    && saved_reader.try_get(ObjRef::new(num, generation))? == current_reader.try_get(ObjRef::new(num, generation))?;
973                if !same {
974                    changed.push(num);
975                }
976            }
977        }
978        // An object stream of this file that is freed or replaced takes its members with it, so
979        // a member that is otherwise unchanged is written again on its own.
980        let gone = |n: u32| freed.binary_search(&n).is_ok() || changed.binary_search(&n).is_ok();
981        let orphaned: Vec<u32> = current_nums
982            .iter()
983            .copied()
984            .filter(|&num| changed.binary_search(&num).is_err())
985            .filter(|&num| {
986                let packed_in = match self.xref_entry(num) {
987                    Some(XrefEntry::InStream { stream, .. }) => Some(stream),
988                    _ => None,
989                };
990                packed_in.is_some_and(gone)
991            })
992            .collect();
993        changed.extend(orphaned);
994        for num in freed {
995            let generation = self.generation(num);
996            self.free(ObjRef::new(num, generation));
997        }
998        let mut current_reader = current.object_reader();
999        for num in changed {
1000            let generation = current.generation(num);
1001            let object = current_reader.try_get(ObjRef::new(num, generation))?;
1002            if let Some(alias) = current.alternate_generation(ObjRef::new(num, generation)) {
1003                let object = self.adopt((*object).clone());
1004                self.overlay.insert(num, Slot::RepairedSet(generation, alias, Arc::new(object)));
1005                self.next_num = self.next_num.max(num.saturating_add(1));
1006            } else {
1007                self.set(ObjRef::new(num, generation), (*object).clone());
1008            }
1009        }
1010        *self.trailer_mut() = current.trailer().clone();
1011        Ok(())
1012    }

1022    /// Current generation of an object number.
1023    pub fn generation(&self, num: u32) -> u16 {
1024        match self.overlay.get(&num) {
1025            Some(Slot::Set(g, _) | Slot::RepairedSet(g, _, _)) => *g,
1026            Some(Slot::Freed(g)) => *g,
1027            None => match self.entries.lookup(num, &self.data) {
1028                Some(XrefEntry::InFile { generation, .. }) => generation,
1029                Some(XrefEntry::Free { next_generation }) => next_generation,
1030                _ => 0,
1031            },
1032        }
1033    }

1035    pub(crate) fn overlay_entries(&self) -> impl Iterator<Item = (u32, u16, Option<Arc<Object>>)> + '_ {
1036        self.overlay.iter().map(|(n, s)| match s {
1037            Slot::Set(g, o) | Slot::RepairedSet(g, _, o) => (*n, *g, Some(o.clone())),
1038            Slot::Freed(g) => (*n, *g, None),
1039        })
1040    }

1043        self.next_num
1044    }

1046    /// All object numbers known (file and overlay), excluding freed ones.
1047    pub fn object_numbers(&self) -> Vec<u32> {
1048        let source_numbers = self.entries.object_numbers(&self.data);
1049        let mut numbers = Vec::with_capacity(source_numbers.len().max(self.overlay.len()));
1050        let mut source = source_numbers.into_iter().peekable();
1051        for (&num, slot) in &self.overlay {
1052            while source.peek().is_some_and(|n| *n < num) {
1053                if let Some(n) = source.next() {
1054                    numbers.push(n);
1055                }
1056            }
1057            if source.peek().is_some_and(|n| *n == num) {
1058                source.next();
1059            }
1060            if matches!(slot, Slot::Set(..) | Slot::RepairedSet(..)) {
1061                numbers.push(num);
1062            }
1063        }
1064        numbers.extend(source);
1065        numbers
1066    }

1105/// An object read in place, or the object a reference names as the document holds it: either way
