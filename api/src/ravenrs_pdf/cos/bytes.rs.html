1//! Stream data that can share a larger buffer instead of copying out of it.
2
3use std::ops::{Deref, Range};
4use std::sync::Arc;
5
6/// A stream's bytes: a buffer of their own, or a range of a larger shared buffer. Streams read
7/// from a file point into the file's bytes, which the document keeps anyway, so caching a
8/// parsed stream costs its dictionary and not a second copy of its data.
9#[derive(Clone, Default)]
10pub struct Bytes {
11    buf: Arc<Vec<u8>>,
12    range: Range<usize>,
13}
14
15impl Bytes {
16    /// `range` of `buf`, without copying; `None` when the range is not within `buf`.
17    pub(crate) fn view(buf: &Arc<Vec<u8>>, range: Range<usize>) -> Option<Self> {
18        (range.start <= range.end && range.end <= buf.len()).then(|| Self { buf: buf.clone(), range })
19    }
20
21    /// Whether these bytes keep alive a larger buffer than themselves that isn't `owner`.
22    pub(crate) fn borrows_other_than(&self, owner: &Arc<Vec<u8>>) -> bool {
23        self.range.len() < self.buf.len() && !Arc::ptr_eq(&self.buf, owner)
24    }
25
26    /// Whether these bytes are (a view into) `buf`, holding no memory of their own.
27    pub(crate) fn shares(&self, buf: &Arc<Vec<u8>>) -> bool {
28        Arc::ptr_eq(&self.buf, buf)
29    }
30
31    /// The same bytes, in a buffer of their own.
32    pub(crate) fn detached(&self) -> Self {
33        Self::from(self.to_vec())
34    }
35}
36
37impl From<Vec<u8>> for Bytes {
38    fn from(v: Vec<u8>) -> Self {
39        let range = 0..v.len();
40        Self { buf: Arc::new(v), range }
41    }
42}
43
44impl From<Arc<Vec<u8>>> for Bytes {
45    fn from(buf: Arc<Vec<u8>>) -> Self {
46        let range = 0..buf.len();
47        Self { buf, range }
48    }
49}
50
51impl Deref for Bytes {
52    type Target = [u8];
53
54    fn deref(&self) -> &[u8] {
55        // `view` and `From` only build in-bounds ranges; an empty slice keeps this panic-free.
56        self.buf.get(self.range.clone()).unwrap_or_default()
57    }
58}
59
60impl AsRef<[u8]> for Bytes {
61    fn as_ref(&self) -> &[u8] {
62        self
63    }
64}
65
66impl PartialEq for Bytes {
67    fn eq(&self, other: &Self) -> bool {
68        (Arc::ptr_eq(&self.buf, &other.buf) && self.range == other.range) || **self == **other
69    }
70}
71
72impl Eq for Bytes {}
73
74impl std::fmt::Debug for Bytes {
75    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
76        (**self).fmt(f)
77    }
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    #[test]
85    fn views_share_their_buffer_and_compare_by_content() {
86        let file = Arc::new(b"0123456789".to_vec());
87        let view = Bytes::view(&file, 2..5).expect("in range");
88        assert_eq!(&*view, b"234");
89        assert_eq!(view.as_ptr(), file[2..].as_ptr(), "no copy");
90        assert_eq!(view, Bytes::from(b"234".to_vec()));
91        assert!(Bytes::view(&file, 5..11).is_none() && Bytes::view(&file, Range { start: 6, end: 5 }).is_none());
92        let other = Arc::new(Vec::new());
93        assert!(view.borrows_other_than(&other) && !view.borrows_other_than(&file));
94        let own = view.detached();
95        assert!(!own.borrows_other_than(&other) && own == view && own.as_ptr() != view.as_ptr());
96        assert!(!Bytes::from(file.clone()).borrows_other_than(&other), "a whole buffer is its own");
97    }
