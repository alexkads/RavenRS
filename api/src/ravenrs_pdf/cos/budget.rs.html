9/// The steps of decoding an operation over a document's images may take whatever the file's size.
10/// A step of JBIG2 decoding ([`crate::third_party::hayro_jbig2::Image::decode_within`]) is about a pixel, and takes 6 to
11/// 16 ns, so this is 2 to 5 minutes, and about as long again for what an operation does with the
12/// pixels. A page scanned at 300 dpi takes about 17 million, 8.4 for its pixels and that again to
13/// decode them, so this alone reads about 1,000 of them. A byte into or out of a filter is a step
14/// too, and takes less: 3 ns at most, or up to 19 ns through a predictor on samples of a bit. A
15/// byte Flate reads is [`crate::filters::FLATE_READ_WORK`] steps besides, since inflating blocks
16/// that make nothing takes up to 21 ns a byte: about 7 ns a step.
17const IMAGE_WORK: u64 = 1 << 34;
18/// The steps each byte of the file adds to [`IMAGE_WORK`], so that a long scan is read whole: such a
19/// page pays for itself with about 2 KB of the file, about the smallest a page of text is stored
20/// in, and a stream of any filter this reads can be decoded about 8 times at Flate's best, 1,032 to
21/// one. So what a hostile file can ask for grows with its length, by about 14 minutes for 10 MB.
22const IMAGE_WORK_PER_BYTE: u64 = 1 << 13;
23
24/// Decoded bytes one page's content walk may produce in all: its `/Contents` streams and every
25/// form XObject it enters, each occurrence counted again.
26pub const MAX_PAGE_DECODED: usize = 256 << 20;
27/// Form XObjects one page's content walk may enter, each entry counted again.
28pub const MAX_PAGE_FORMS: usize = 50_000;

30/// The most pixels an image redaction decodes, to clear the ones under a mark or to store again
31/// without the `/JBIG2Globals` it shares with a redacted image (`pdfcraft_redact`). Optimize has
32/// JBIG2 images share a dictionary only within it (`pdfcraft_optimize`), so that redaction can
33/// store each of them again: one it couldn't would refuse the redaction.
34pub const MAX_REDACTED_IMAGE_PIXELS: u64 = 64 << 20;

36/// The steps a sample of a JPEG costs: a few bytes can hold a page of one grey, and decoding,
37/// resampling and storing it took about 20 to 45 ns a sample.
38const JPEG_WORK: u64 = 2;
39/// The steps a byte of a JPEG costs to read: its tables took 24 ns a byte before a size was known,
40/// and images can share one, so reading it again for each has to count.
41const JPEG_READ_WORK: u64 = 4;

43/// A stream decode within a byte bound, with the bytes its filter stages decoded: the data and
44/// whether it stopped at damage, or the error, each with that count.
45type Decoder = fn(&Stream, usize) -> Result<(Vec<u8>, bool, usize), (FilterError, usize)>;

47/// What one operation over a document's images may still spend decoding them: exporting them,
48/// optimizing them, and storing LZW streams with Flate as it does, clearing them for redaction.
49///
50/// An image has its own limit, which doesn't bound a file of thousands of images made to reach it:
51/// a few hundred bytes of JBIG2 can ask for seconds of decoding, 4 KB of Flate inside Flate for a
52/// gigabyte, and an image that shares `/JBIG2Globals` decodes them again. So images share this
53/// allowance, whatever they do with what they decode and whether or not it works, and past it they
54/// aren't decoded. A caller then treats them as it does an image it can't read, and
55/// [`ImageBudget::refused`] says how many there were.
56///
57/// An image also pays for its pixels once decoded, as what reads it goes through them one by one
58/// however few bytes coded them: blank rows of fax or Flate take almost none. It pays for no more
59/// than its data holds, a bit a pixel at least, so an image declaring a size its data doesn't fill
60/// costs what a smaller one would, not the run's allowance. A JPEG pays for its samples when
61/// it is decoded ([`ImageBudget::pay_jpeg`]), and is otherwise not charged, as JPEG 2000 isn't:
62/// their decoders don't run here.
63#[derive(Debug)]
64pub struct ImageBudget {
65    work: u64,
66    refused: usize,
67}

70    /// The allowance for an operation on `doc`, which a longer file makes larger. The streams made
71    /// or changed since it was opened count as the file's bytes do, so a document built in memory
72    /// (combined, created, or given pages of scans) gets what it would once saved and opened again.
73    pub fn new(doc: &Document) -> Self {
74        let length = |n: usize| u64::try_from(n).unwrap_or(u64::MAX);
75        let edited = doc
76            .scan_objects_subset(doc.modified_objects())
77            .filter_map(|(_, object)| match &*object.ok()? {
78                Object::Stream(s) => Some(length(s.raw.len())),
79                _ => None,
80            })
81            .fold(0u64, u64::saturating_add);
82        let bytes = length(doc.bytes().len()).saturating_add(edited);
83        Self::with_work(IMAGE_WORK.saturating_add(bytes.saturating_mul(IMAGE_WORK_PER_BYTE)))
84    }

91    /// The samples of image `s`, decoded and charged to the allowance: the one way to decode an
92    /// image outside the renderer, so that an operation over many images can't decode them without
93    /// one.
94    ///
95    /// They go through its filters ([`Stream::decoded`]), or, for a CCITT fax or JBIG2 image (its
96    /// last filter), through the filters before it and then [`crate::filters::decode_ccitt_within`]
97    /// with its [`Document::fax_params`], or [`crate::filters::decode_jbig2_within`] with its
98    /// `/JBIG2Globals`, as `CCITTFaxDecode` and `JBIG2Decode` give them (ISO 32000-2 §7.4.6,
99    /// §7.4.7). Parameters given by reference count, for every filter
100    /// ([`Document::with_direct_parms`]). A fax image's `/Columns` has to be its `/Width`, and a
101    /// JBIG2 image's page its `/Width` × `/Height`: PdfCraft's renderer draws the whole page into
102    /// the image. Up to its `/Height` rows are decoded, whatever a fax image's `/Rows` says; damaged
103    /// fax data gives fewer, so a caller checks it has the rows it needs. Damaged JBIG2 data, and
104    /// globals that can't be read (within [`MAX_JBIG2_GLOBALS`](crate::cos::MAX_JBIG2_GLOBALS)), are an
105    /// error.
106    ///
107    /// No stage, the filters before a fax or JBIG2 decoder included, may produce more than `max`
108    /// bytes, nor more than [`MAX_DECODED`](crate::cos::MAX_DECODED). A JBIG2 image's `/JBIG2Globals`
109    /// are another stream, shared between images, and are decoded within `MAX_JBIG2_GLOBALS`
110    /// whatever `max` is. Each filter as it runs, the globals when they have to be read, and the fax
111    /// or JBIG2 decoder spend from the allowance, and the image then pays for its pixels; with
112    /// nothing left an image is refused before anything is read, its globals included.
113    ///
114    /// ```
115    /// # let doc = ravenrs_pdf::cos::Document::new_empty();
116    /// # let s = ravenrs_pdf::cos::Stream::from_raw(ravenrs_pdf::cos::Dict::new(), Vec::new());
117    /// let samples = ravenrs_pdf::cos::ImageBudget::new(&doc).decode(&doc, &s, 1 << 20);
118    /// assert_eq!(samples.unwrap(), b"");
119    /// ```
120    ///
121    /// `Document` has no way of its own to decode an image:
122    ///
123    /// ```compile_fail
124    /// # let doc = ravenrs_pdf::cos::Document::new_empty();
125    /// # let s = ravenrs_pdf::cos::Stream::from_raw(ravenrs_pdf::cos::Dict::new(), Vec::new());
126    /// let _ = doc.decoded_image(&s);
127    /// ```
128    ///
129    /// ```compile_fail
130    /// # let doc = ravenrs_pdf::cos::Document::new_empty();
131    /// # let s = ravenrs_pdf::cos::Stream::from_raw(ravenrs_pdf::cos::Dict::new(), Vec::new());
132    /// let _ = doc.decoded_image_within(&s, 1 << 20);
133    /// ```
134    pub fn decode(&mut self, doc: &Document, s: &Stream, max: usize) -> Result<Vec<u8>, CosError> {
135        let decoded = doc.decoded_image_spending(s, max, &mut self.work);
136        self.count_refused(s, &decoded);
137        self.pay_size(s, decoded?)
138    }

140    /// Image `s` through its filters alone, charged to the allowance: [`Stream::decoded_within`]
141    /// with its `/DecodeParms` given by reference read too ([`Document::with_direct_parms`]), for an
142    /// image that has no fax or JBIG2 decoder to go through.
143    pub fn decode_filters(&mut self, doc: &Document, s: &Stream, max: usize) -> Result<Vec<u8>, CosError> {
144        let decoded = doc.with_direct_parms(s).decoded_spending(max, &mut self.work);
145        self.count_refused(s, &decoded);
146        let decoded = decoded?;
147        if s.filters().iter().all(|(f, _)| f.is_decodable()) { self.pay_size(s, decoded) } else { Ok(decoded) }
148    }

161    /// Pays `steps` of other work on decoded data, or spends all there is and says so.
162    pub fn spend(&mut self, steps: u64) -> bool {
163        let left = self.work.checked_sub(steps);
164        self.work = left.unwrap_or(0);
165        left.is_some()
166    }

168    /// Pays for `pixels` a decode goes on to work through, or spends all there is, and the image
169    /// is refused.
170    pub fn pay_pixels(&mut self, pixels: u64) -> bool {
171        let paid = self.spend(pixels);
172        self.refused += usize::from(!paid);
173        paid
174    }

176    /// Pays for reading JPEG `data` and decoding its `samples`, as [`ImageBudget::pay_pixels`] does.
177    /// A caller pays before its tables are parsed, which can be most of the file.
178    pub fn pay_jpeg(&mut self, samples: u64, data: &[u8]) -> bool {
179        let reading = (data.len() as u64).saturating_mul(JPEG_READ_WORK);
180        self.pay_pixels(samples.saturating_mul(JPEG_WORK).saturating_add(reading))
181    }

183    /// Pays for the pixels of image `s` that its `decoded` data holds, those of its `/Width` and
184    /// `/Height` but no more than a bit of the data each, unless it is JBIG2, whose decoder pays for
185    /// its page, or isn't decoded here. What can't be paid for refuses the image.
186    fn pay_size(&mut self, s: &Stream, decoded: Vec<u8>) -> Result<Vec<u8>, CosError> {
187        if !charged(s) || matches!(s.filters().last(), Some((crate::filters::Filter::Jbig2, _))) {
188            return Ok(decoded);
189        }
190        let side = |key: &[u8]| s.dict.int(key).and_then(|v| u64::try_from(v).ok()).unwrap_or(0);
191        let held = u64::try_from(decoded.len()).unwrap_or(u64::MAX).saturating_mul(8);
192        if self.pay_pixels(side(b"Width").saturating_mul(side(b"Height")).min(held)) {
193            Ok(decoded)
194        } else {
195            Err(CosError::Filter(FilterError::Spent.to_string()))
196        }
197    }

199    /// An image that failed with nothing left was refused, unless it has a filter that costs
200    /// nothing here, which fails whatever is left.
201    fn count_refused(&mut self, s: &Stream, decoded: &Result<Vec<u8>, CosError>) {
202        if decoded.is_err() && self.work == 0 && charged(s) {
203            self.refused += 1;
204        }
205    }

223/// Whether `s` can be decoded here and so charged: general filters, a fax or JBIG2 decoder last at
224/// most. Those decoders anywhere else can't run.
225fn charged(s: &Stream) -> bool {
226    use crate::filters::Filter;
227    s.readable_filters().is_ok_and(|chain| {
228        chain.split_last().is_none_or(|((last, _), before)| {
229            (last.is_decodable() || matches!(last, Filter::Jbig2 | Filter::CcittFax)) && before.iter().all(|(f, _)| f.is_decodable())
230        })
231    })
232}

234/// What one operation may still decode and enter across all the streams it reads.
235///
236/// [`MAX_DECODED`](crate::cos::MAX_DECODED) bounds one decode. It does not bound a page whose
237/// `/Contents` names the same compressed stream thousands of times, or a form drawn thousands of
238/// times from other forms: each occurrence would get a fresh allowance. A `DecodeBudget` is
239/// shared by every decode of one operation, so the total decoded bytes and form entries stay
240/// under fixed limits however the file repeats its references. A decode that runs past the
241/// remaining allowance spends all of it, so a bomb referenced many times is inflated once.
242#[derive(Debug)]
243pub struct DecodeBudget {
244    /// The whole allowance, for errors.
245    limit: usize,
246    bytes: usize,
247    forms: usize,
248    seen: HashSet<ObjRef>,
249    /// Decodes refused for lack of allowance ([`DecodeBudget::refused`]).
250    refusals: usize,
251}

257    }

276    /// Whether the allowance is spent and `stream`, which has data, would be refused, noting the
277    /// refusal ([`DecodeBudget::refused`]): a caller asks this before resolving the stream's
278    /// parameters, work the allowance doesn't count. A stream without data still reads.
279    pub fn refuses(&mut self, stream: &Stream) -> bool {

281        self.refusals += usize::from(refuses);
282        refuses
283    }

290    /// `stream` decoded like [`Stream::decoded`], charged to the allowance. Fails, without
291    /// decoding, once the allowance is spent. A stream damaged partway gives what decoded before
292    /// the damage, so code that writes the stream back reads it with
293    /// [`DecodeBudget::decode_with_damage`] instead.
294    pub fn decode(&mut self, stream: &Stream) -> Result<Vec<u8>, CosError> {
295        self.charge(stream, usize::MAX, Stream::decoded_within_counted).map(|(data, _)| data)
296    }

298    /// Like [`DecodeBudget::decode`], for a stream with a tighter bound of its own: decoding
299    /// past `max` bytes is an error that charges `max`, the work it took to find out.
300    pub fn decode_within(&mut self, stream: &Stream, max: usize) -> Result<Vec<u8>, CosError> {
301        self.charge(stream, max, Stream::decoded_within_counted).map(|(data, _)| data)
302    }

304    /// Like [`DecodeBudget::decode_within`], with every filter stage held to `max`, as
305    /// [`Stream::decoded_within`] holds it, and the stages together to `max` for each of them, the
306    /// stream's `/DecodeParms` given by reference read in `doc` through a view of it that shares
307    /// its data and copies no more of its dictionary than its filters and parameters
308    /// ([`DecodeBudget::decode_with_damage`] reads it so). A
309    /// stream so reads here when it reads without an allowance and the allowance can pay for
310    /// what its stages decode, the output of one stage on top of another's. A stage past `max`
311    /// is an error that charges what the stages before it decoded and `max`, the work it took to
312    /// find out; past the stages' total it charges that total, and past what is left it spends
313    /// all of it, as [`DecodeBudget::decode_within`] does.
314    pub fn decode_stages_within(&mut self, doc: &Document, stream: &Stream, max: usize) -> Result<Vec<u8>, CosError> {
315        // Before the parameters are resolved, work the allowance doesn't count.
316        if self.refuses(stream) {
317            return Err(CosError::Filter(FilterError::LimitExceeded(self.limit).to_string()));
318        }
319        let stream = doc.decoding_view(stream);
320        let total = max.saturating_mul(stream.filters().len().max(1));
321        let room = total.min(self.bytes);
322        match stream.decoded_within_each_counted(max, room) {
323            Ok((data, _, spent)) => {
324                self.bytes = self.bytes.saturating_sub(spent);
325                Ok(data)
326            }
327            // A stage stopped at `max`, below what was left of the room.
328            Err((FilterError::LimitExceeded(_), spent)) if room.saturating_sub(spent) > max => {
329                self.bytes = self.bytes.saturating_sub(spent.saturating_add(max));
330                Err(CosError::Filter(FilterError::LimitExceeded(max).to_string()))
331            }
332            // At the stages' total, with more left.
333            Err((FilterError::LimitExceeded(_), _)) if total < self.bytes => {
334                self.bytes = self.bytes.saturating_sub(total);
335                Err(CosError::Filter(FilterError::LimitExceeded(total).to_string()))
336            }
337            Err((FilterError::LimitExceeded(_), _)) => {
338                self.bytes = 0;
339                self.refusals += 1;
340                Err(CosError::Filter(FilterError::LimitExceeded(self.limit).to_string()))
341            }
342            Err((e, spent)) => {
343                self.bytes = self.bytes.saturating_sub(spent);
344                Err(CosError::Filter(e.to_string()))
345            }
346        }
347    }

349    /// [`DecodeBudget::decode_within`] (`usize::MAX` for no bound of its own), with the stream's
350    /// `/DecodeParms` given by reference read in `doc`, also saying whether the stream is damaged:
351    /// `true` when a filter stopped at corrupt data, so the bytes are only what decoded before it.
352    /// A stream written back from them loses everything after the damage, which other readers may
353    /// still read.
354    pub fn decode_with_damage(&mut self, doc: &Document, stream: &Stream, max: usize) -> Result<(Vec<u8>, bool), CosError> {
355        self.charge(&doc.decoding_view(stream), max, Stream::decoded_within_counted)
356    }

358    /// `stream` decoded like [`Stream::decoded_strict`] (any corruption is an error), charged to
359    /// the allowance, with its `/DecodeParms` given by reference read in `doc`.
360    pub fn decode_strict(&mut self, doc: &Document, stream: &Stream) -> Result<Vec<u8>, CosError> {
361        let strict: Decoder = |s, max| s.decoded_strict_within_counted(max).map(|(data, spent)| (data, false, spent));
362        self.charge(&doc.decoding_view(stream), usize::MAX, strict).map(|(data, _)| data)
363    }

365    /// Decodes `stream` within `max` and the allowance, charging what every filter stage
366    /// decoded: a stage's output counts even when a later stage shrinks it, and a corrupt stream
367    /// pays for the bytes decoded before the error, so referencing a stream again and again
368    /// spends the allowance by the work each decode does.
369    fn charge(&mut self, stream: &Stream, max: usize, decode: Decoder) -> Result<(Vec<u8>, bool), CosError> {
370        if self.refuses(stream) {
371            return Err(CosError::Filter(FilterError::LimitExceeded(self.limit).to_string()));
372        }
373        if max < self.bytes {
374            let result = decode(stream, max);
375            let spent = match &result {
376                Ok((_, _, spent)) => *spent,
377                Err((FilterError::LimitExceeded(_), _)) => max,
378                Err((_, spent)) => *spent,
379            };
380            self.bytes = self.bytes.saturating_sub(spent);
381            return result.map(|(data, damaged, _)| (data, damaged)).map_err(|(e, _)| CosError::Filter(e.to_string()));
382        }
383        match decode(stream, self.bytes) {
384            Ok((data, damaged, spent)) => {
385                self.bytes = self.bytes.saturating_sub(spent);
386                Ok((data, damaged))
387            }
388            Err((FilterError::LimitExceeded(_), _)) => {
389                self.bytes = 0;
390                self.refusals += 1;
391                Err(CosError::Filter(FilterError::LimitExceeded(self.limit).to_string()))
392            }
393            Err((e, spent)) => {
394                self.bytes = self.bytes.saturating_sub(spent);
395                Err(CosError::Filter(e.to_string()))
396            }
397        }
398    }

400    /// Counts one form entry. `false` once [`DecodeBudget::new`]'s `forms` entries are spent:

418    /// Each stream `contents` names, decoded against the allowance, in order, its `/DecodeParms`
419    /// given by reference read in `doc`. `contents` is a page's `/Contents`: one stream, or an
420    /// array whose non-stream entries are left out. A stream that fails to decode, or arrives
421    /// after the allowance is spent, is an `Err` in its place, so the caller chooses between
422    /// skipping it and giving up.
423    pub fn decode_contents(&mut self, doc: &Document, contents: &Object) -> Vec<Result<Vec<u8>, CosError>> {
424        self.decode_contents_with_damage(doc, contents).into_iter().map(|read| read.map(|(data, _)| data)).collect()
425    }
426
427    /// [`DecodeBudget::decode_contents`], each stream also saying whether it is damaged partway
428    /// ([`DecodeBudget::decode_with_damage`], which reads `/DecodeParms` given by reference).
429    pub fn decode_contents_with_damage(&mut self, doc: &Document, contents: &Object) -> Vec<Result<(Vec<u8>, bool), CosError>> {
430        match &*doc.resolve(contents) {
431            Object::Stream(s) => vec![self.decode_with_damage(doc, s, usize::MAX)],
432            Object::Array(list) => list
433                .iter()
434                .filter_map(|entry| match &*doc.resolve(entry) {
435                    Object::Stream(s) => Some(self.decode_with_damage(doc, s, usize::MAX)),
436                    _ => None,
437                })
438                .collect(),
439            _ => Vec::new(),
440        }
441    }

1132    /// What `decode_stages_within` charges, and which limit it reports, on each way a read stops:
1133    /// a stage at the stream's limit with more of the room left charges what came before and the
1134    /// limit; at the stages' total, with more of the allowance left, it charges that total; at
1135    /// what is left of the allowance it spends all of it and reports the allowance's limit, as
1136    /// `decode_within` does. Each boundary between them is read on both sides.

1138    fn a_stage_past_its_limit_charges_what_came_before_and_the_limit() {

1140        let hex = |data: &[u8]| data.iter().flat_map(|b| format!("{b:02X}").into_bytes()).chain(*b">").collect::<Vec<u8>>();
1141        let two = |raw: &[u8]| {
1142            let mut dict = Dict::new();
1143            dict.set(b"Filter".to_vec(), Object::Array(vec![Object::name("ASCIIHexDecode"), Object::name("FlateDecode")]));
1144            Stream::from_raw(dict, hex(raw))
1145        };
1146        let says = |read: Result<Vec<u8>, CosError>, limit: usize| {
1147            read.err().map(|e| e.to_string()) == Some(CosError::Filter(FilterError::LimitExceeded(limit).to_string()).to_string())
1148        };
1149        // Flate data making eight times the limit: the hex stage makes `inner` bytes of it, and
1150        // the Flate stage stops at the limit.
1151        let bomb = Stream::flate(Dict::new(), &vec![0u8; 8 * max]).raw.to_vec();

1157        assert_eq!(budget.remaining(), 1);

1161        assert!(says(budget.decode_stages_within(&Document::new_empty(), &two(&bomb), max), inner + max));
1162        assert_eq!(budget.remaining(), 0);
1163        // The same data padded to exactly the limit: the hex stage makes all of it, and the Flate
1164        // stage, at the limit, reaches the stages' total. With more of the allowance left, that
1165        // total is what it charges and reports.
1166        let mut padded = bomb.clone();
