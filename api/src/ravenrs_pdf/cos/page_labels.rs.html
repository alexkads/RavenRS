1//! Bounded numbering for page labels (ISO 32000-2 §12.4.2).
2//! Pure helpers shared by inspection and editing; no object graph or renderer dependency.
3
4/// A complete custom label, including its prefix, may contain this many UTF-8 bytes.
5pub const MAX_LABEL_BYTES: usize = 1024;
6/// Raw prefixes are bounded before decoding; this admits a full UTF-16BE ASCII label.
7pub const MAX_PREFIX_BYTES: usize = 2 * MAX_LABEL_BYTES + 2;
8/// Sum of the UTF-8 bytes of custom labels for one document.
9pub const MAX_LABEL_TOTAL_BYTES: usize = 4 << 20;
10/// Sum of the UTF-8 bytes of the decoded prefixes of one `/PageLabels` tree. Each range decodes
11/// its own prefix, and ranges may share one prefix object, so without this a small file could ask
12/// for a copy per range: [`MAX_LABEL_TREE_WORK`] ranges naming one 1,024-byte prefix would hold
13/// about 200 MiB. Real documents stay far below it.
14pub const MAX_PREFIX_TOTAL_BYTES: usize = 4 << 20;
15/// The most number-tree entries (nodes, kids and range pairs) read from one `/PageLabels`
16/// tree. Scaled for real documents: one range per page on a ten-thousand-page PDF happens
17/// (prefix labels; #307's file has 9,156 pages), so the cap is twenty times that while still
18/// bounding hostile trees.
19pub const MAX_LABEL_TREE_WORK: usize = 200_000;
20/// Maximum nesting of a label number tree.
21pub const MAX_LABEL_TREE_DEPTH: usize = 32;
22
23/// Lowercase Roman numerals, or `None` if the output exceeds the remaining label budget.
24/// The budget is capped at [`MAX_LABEL_BYTES`], including when `n` is hostile.
25pub fn roman(mut n: u64, max_bytes: usize) -> Option<String> {
26    const PARTS: [(u64, &str); 13] = [
27        (1000, "m"),
28        (900, "cm"),
29        (500, "d"),
30        (400, "cd"),
31        (100, "c"),
32        (90, "xc"),
33        (50, "l"),
34        (40, "xl"),
35        (10, "x"),
36        (9, "ix"),
37        (5, "v"),
38        (4, "iv"),
39        (1, "i"),
40    ];
41    let max_bytes = max_bytes.min(MAX_LABEL_BYTES);
42    let mut out = String::new();
43    for (value, part) in PARTS {
44        let count = usize::try_from(n / value).ok()?;
45        let bytes = count.checked_mul(part.len())?;
46        if bytes > max_bytes.saturating_sub(out.len()) {
47            return None;
48        }
49        // The quotient is checked against the tiny byte budget before repeating.
50        out.push_str(&part.repeat(count));
51        n %= value;
52    }
53    Some(out)
54}

56/// a..z, aa..zz, aaa.. (the repeated-letter numbering specified by PDF).
57/// Returns `None` instead of truncating a label that exceeds the remaining byte budget.
58pub fn alpha(n: u64, max_bytes: usize) -> Option<String> {
59    let n = n.max(1) - 1;
60    let count = usize::try_from(n / 26 + 1).ok()?;
61    if count > max_bytes.min(MAX_LABEL_BYTES) {
62        return None;
63    }
64    let letter = char::from(b'a' + (n % 26) as u8);
65    Some(std::iter::repeat_n(letter, count).collect())
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn bounded_page_labels_follow_pdf_numbering() {
74        assert_eq!(roman(4, MAX_LABEL_BYTES).as_deref(), Some("iv"));
75        assert_eq!(roman(1994, MAX_LABEL_BYTES).as_deref(), Some("mcmxciv"));
76        for (n, expected) in [(0, "a"), (1, "a"), (26, "z"), (27, "aa"), (52, "zz"), (53, "aaa")] {
77            assert_eq!(alpha(n, MAX_LABEL_BYTES).as_deref(), Some(expected));
78        }
79    }
80
81    #[test]
82    fn bounded_page_labels_check_lengths_before_repeating() {
83        assert_eq!(roman(1_024_000, MAX_LABEL_BYTES).unwrap().len(), MAX_LABEL_BYTES);
84        assert!(roman(1_024_001, MAX_LABEL_BYTES).is_none());
85        assert_eq!(alpha(26 * 1024, MAX_LABEL_BYTES).unwrap().len(), MAX_LABEL_BYTES);
86        assert!(alpha(26 * 1024 + 1, MAX_LABEL_BYTES).is_none());
87        assert_eq!(roman(4, 2).as_deref(), Some("iv"));
88        assert!(roman(4, 1).is_none());
89        assert_eq!(alpha(27, 2).as_deref(), Some("aa"));
90        assert!(alpha(27, 1).is_none());
91        for max in [0, MAX_LABEL_BYTES, usize::MAX] {
92            assert!(roman(u64::MAX, max).is_none());
93            assert!(alpha(u64::MAX, max).is_none());
94        }
95    }
