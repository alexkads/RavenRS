1//! Document: cross-reference data, lazy object loading, and a copy-on-write overlay of edits.
2//!
3//! `Document` is cheap to clone (the original bytes, xref index and parse caches are shared), so
4//! a clone is a snapshot: undo/redo and background jobs hold clones while the UI edits another.
5//! Edits never touch the original bytes; the writer appends them (incremental save) or rewrites
6//! the file (full save).
7

36#[derive(Clone)]
37pub struct Document {
38    data: Arc<Vec<u8>>,
39    /// Where `endstream` occurs in `data`; see [`EndstreamIndex`].
40    endstreams: Arc<EndstreamIndex>,
41    source_identity: Arc<()>,
42    entries: Arc<XrefIndex>,
43    trailer: Dict,
44    revisions: Arc<Vec<Revision>>,
45    /// See [`Self::xref_oddities`].
46    xref_oddities: Arc<Vec<(u64, u32)>>,
47    /// Where the file's object headers are, as (offset, generation) of the last header of each
48    /// number. Filled only when the cross-reference data proved inconsistent; see
49    /// `fill_xref_gaps`.
50    located: Arc<BTreeMap<u32, (u64, u16)>>,
51    repair_log: Arc<Vec<String>>,
52    /// See [`Self::rebuilt_unlinked_update`].
53    unlinked_update: bool,
54    /// What loading each file object gave, a failure as well: a damaged object is otherwise
55    /// loaded again on every read, and a member of an object stream that can't be decoded
56    /// decodes the stream again (`Document::load`).
57    cache: Arc<Mutex<state::ObjectCache>>,
58    objstms: Arc<Mutex<HashMap<u32, Arc<ObjStm>>>>,
59    /// See [`Self::unreadable_objects`]. Shared by clones, like the caches: it describes the
60    /// source, and is reset wherever they are.
61    unreadable: Arc<Mutex<Unreadable>>,
62    /// What `/JBIG2Globals` streams of the file decode to, by object number: the images of a scan
63    /// share them, and each would otherwise decode them again (see `Document::jbig2_globals`).
64    jbig2_globals: Arc<Mutex<HashMap<u32, Arc<Vec<u8>>>>>,
65    stream_limit: usize,
66    overlay: BTreeMap<u32, Slot>,
67    next_num: u32,
68    /// Position of the `%PDF-` header (some files have junk before it).
69    header_offset: usize,
70    version: String,
71    /// The authenticated security handler of an encrypted document.
72    security: Option<Arc<crate::crypt::SecurityHandler>>,
73    /// Object number of the `/Encrypt` dictionary (never encrypted itself).
74    encrypt_num: Option<u32>,
75    /// What that dictionary and `/ID` refer to, which the key comes from, and so is as stored.
76    encrypt_members: Arc<HashSet<u32>>,
77    /// Object numbers a scan of the file didn't find: the file doesn't change, so it never will.
78    unfound: Arc<Mutex<HashSet<u32>>>,
79    /// Encryption was added, changed or removed since opening: only a full save can apply it.
80    encryption_changed: bool,
81    /// Set by edits that must not leave earlier revisions in the file (redaction).
82    full_save: bool,
83    /// The next full save also gives the file a new identifier (`/ID`), see
84    /// [`Document::require_full_save_with_new_id`].
85    new_id: bool,
86    /// The handler and `/Encrypt` object number that saves use, when encryption changed.
87    out_security: Option<Arc<crate::crypt::SecurityHandler>>,
88    out_encrypt_num: Option<u32>,
89}
90
91impl std::fmt::Debug for Document {
92    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
93        f.debug_struct("Document").field("bytes", &self.data.len()).field("xref", &self.entries).field("edits", &self.overlay.len()).finish()
94    }
95}
