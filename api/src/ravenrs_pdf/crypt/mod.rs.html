1//! pdfcraft-crypt — the PDF standard security handler (ISO 32000-2 §7.6), standalone (L0).
2//!
3//! Supports every revision of the standard handler:
4//! - R2 (40-bit RC4), R3 (RC4 up to 128-bit), R4 (crypt filters: RC4 or AES-128);
5//! - R5 (the deprecated AES-256 extension) and R6 (AES-256, PDF 2.0).
6//!
7//! It authenticates with the user or owner password (owner first, as viewers do) and derives
8//! per-object keys. It decrypts and encrypts strings and streams, and honours crypt filters,
9//! including `Identity` and embedded-file-only encryption. It can also *create* encryption
10//! dictionaries, which security settings (M8) and our own test fixtures use.
11//!
12//! The crate knows nothing about the COS object model. Callers pass in the values of the
13//! `/Encrypt` dictionary and the first element of the trailer `/ID`.
14

86/// Permission flags from `/P`, read the way the handler's revision reads them (ISO 32000-1:2008,
87/// Table 22). Owners have every permission.
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub struct Permissions {
90    pub bits: i32,
91    pub owner: bool,
92    /// The `/R` value of the handler that read `bits`.
93    pub revision: i64,
94}
95
96impl Permissions {
97    fn bit(&self, n: u32) -> bool {
98        self.owner || (self.bits >> (n - 1)) & 1 == 1
99    }
100    /// Revision 3 and later read the full flag list of Table 22; revision 2 has its own.
101    fn is_revision_3_or_later(&self) -> bool {
102        self.revision >= 3
103    }
104    pub fn print(&self) -> bool {
105        self.bit(3)
106    }
107    /// Modify contents other than the operations below.
108    pub fn modify(&self) -> bool {
109        self.bit(4)
110    }
111    pub fn copy(&self) -> bool {
112        self.bit(5)
113    }
114    pub fn annotate(&self) -> bool {
115        self.bit(6)
116    }
117    pub fn fill_forms(&self) -> bool {
118        if self.is_revision_3_or_later() { self.bit(9) || self.bit(6) } else { self.bit(6) }
119    }
120    /// Revision 2 covers accessibility with the copy flag (bit 5); revision 3 and later with bit 10.
121    pub fn extract_for_accessibility(&self) -> bool {
122        if self.is_revision_3_or_later() { self.bit(10) } else { self.bit(5) }
123    }
124    /// Insert, rotate or delete pages; create bookmarks and thumbnails. Revision 3 and later grant
125    /// it by bit 11 alone. Table 22 defines bit 11 for those revisions only, so at revision 2 the
126    /// modify flag (bit 4) covers it.
127    pub fn assemble(&self) -> bool {
128        if self.is_revision_3_or_later() { self.bit(11) } else { self.bit(4) }
129    }
130    /// Take pages out into another document (Extract, Split, a source of Combine or Insert
131    /// Pages): [`Self::assemble`] and copying (bit 5), since the pages carry their text and
132    /// graphics into a new file without the document's protection (#5081). "Any except
133    /// extracting pages" sets bit 4 but not bit 11, so at revision 3 and later it grants neither.
134    pub fn extract_pages(&self) -> bool {
135        self.assemble() && self.copy()
136    }
137    /// Revision 3 and later need bit 3 as well as bit 12. Table 22 defines bit 12 for those
138    /// revisions only: at revision 2 the print flag (bit 3) is all there is.
139    pub fn print_high_quality(&self) -> bool {
140        if self.is_revision_3_or_later() { self.print() && self.bit(12) } else { self.print() }
141    }
142}

164/// The crypt filter methods of a handler, resolved once: a document controls both the number of
165/// filters and the number of strings, so a search of `/CF` per string would multiply them.
166#[derive(Clone, Debug)]
167struct Methods {
168    string: Method,
169    stream: Method,
170    embedded_file: Method,
171    named: std::collections::HashMap<Vec<u8>, Method>,
172}
173
174impl Methods {
175    fn new(dict: &EncryptDict) -> Self {
176        let mut named = std::collections::HashMap::with_capacity(dict.crypt_filters.len());
177        for (name, method) in &dict.crypt_filters {
178            #[cfg(test)]
179            tests::FILTER_COMPARES.with(|c| c.set(c.get() + 1));
180            // The first of two filters with one name wins, as a search would find it.
181            named.entry(name.clone()).or_insert(*method);
182        }
183        let mut methods = Methods { string: Method::Identity, stream: Method::Identity, embedded_file: Method::Identity, named };
184        methods.string = methods.get(&dict.str_f);
185        methods.stream = methods.get(&dict.stm_f);
186        methods.embedded_file = methods.get(if dict.ef_f.is_empty() { &dict.stm_f } else { &dict.ef_f });
187        methods
188    }
189
190    fn get(&self, name: &[u8]) -> Method {
191        if name == b"Identity" || name.is_empty() {
192            return Method::Identity;
193        }
194        #[cfg(test)]
195        tests::FILTER_COMPARES.with(|c| c.set(c.get() + 1));
196        self.named.get(name).copied().unwrap_or(Method::Identity)
197    }
198}

206    /// Authenticate with `password` (tried as owner password, then user password). `None`
207    /// tries the empty password, which opens documents that only have an owner password.
208    pub fn open(dict: EncryptDict, id0: &[u8], password: Option<&str>) -> Result<Self, CryptError> {
209        if !dict.filter.is_empty() && dict.filter != b"Standard" {
210            return Err(CryptError::Unsupported(format!("/Filter /{}", String::from_utf8_lossy(&dict.filter))));
211        }
212        let pw = password.unwrap_or("");
213        let mut recovered_user = None;
214        let (key, auth) = match dict.r {
215            2..=4 => {
216                let mut found = None;
217                for legacy in legacy_candidates(pw) {
218                    if let Some((k, user)) = owner_key_r4(&dict, id0, &legacy) {
219                        recovered_user = Some(user);
220                        found = Some((k, Auth::Owner));
221                    } else if let Some(k) = user_key_r4(&dict, id0, &legacy) {
222                        found = Some((k, Auth::User));
223                    }
224                    if found.is_some() {
225                        break;
226                    }
227                }
228                found.ok_or(CryptError::WrongPassword)?
229            }
230            5 | 6 => {
231                let pw = sasl_password(pw);
232                if let Some(k) = key_r6(&dict, &pw, true)? {
233                    (k, Auth::Owner)
234                } else if let Some(k) = key_r6(&dict, &pw, false)? {
235                    (k, Auth::User)
236                } else {
237                    return Err(CryptError::WrongPassword);
238                }
239            }
240            r => return Err(CryptError::Unsupported(format!("revision {r}"))),
241        };
242        Ok(Self::new(dict, key, auth, recovered_user))
243    }

245    /// A handler with no file key, for a document opened without a password whose strings and
246    /// streams aren't encrypted and whose encrypted crypt filters are all ones the caller found
247    /// authorised only when an embedded file is opened (`/AuthEvent /EFOpen`, ISO 32000-2 §7.6.6,
248    /// Table 25). Data such a filter encrypts is left as stored, read and written alike, so it
249    /// keeps its encrypted bytes; [`SecurityHandler::decrypts`] tells which. `None` when strings or
250    /// streams are encrypted, which needs the key, or the handler is older than crypt filters (V4).
251    pub fn without_key(dict: EncryptDict) -> Option<Self> {
252        if dict.v < 4 {
253            return None;
254        }
255        let methods = Methods::new(&dict);
256        if methods.string != Method::Identity || methods.stream != Method::Identity {
257            return None;
258        }
259        Some(Self { dict, key: Zeroizing::new(Vec::new()), auth: Auth::User, recovered_user: None, methods })
260    }

272    /// Whether data left as stored by a handler without the key still decrypts after its object
273    /// is renumbered: only AES-256 doesn't mix the object number into the key (Algorithm 1.A).
274    pub fn survives_renumbering(&self) -> bool {
275        [self.methods.string, self.methods.stream, self.methods.embedded_file]
276            .into_iter()
277            .chain(self.methods.named.values().copied())
278            .all(|m| matches!(m, Method::Identity | Method::Aes256))
279    }

281    /// Upgrade to owner access with the owner (permissions) password, as when a restricted
282    /// document is unlocked after opening. The user password, or any other, is `WrongPassword`.
283    /// Both passwords unlock the same file key, so what was decrypted so far stays valid; a
284    /// different key is refused too.
285    pub fn authenticate_owner(&self, id0: &[u8], password: &str) -> Result<Self, CryptError> {
286        let owner = Self::open(self.dict.clone(), id0, Some(password))?;
287        if owner.auth != Auth::Owner || owner.key != self.key {
288            return Err(CryptError::WrongPassword);
289        }
290        Ok(owner)
291    }

304    /// For R5/R6, `/P` is stored in clear and again in `/Perms`, encrypted with the file key
305    /// (Algorithm 13). Where the two differ, a permission counts only when both grant it, so an
306    /// edited `/P` grants nothing extra. Without a readable `/Perms`, `/P` applies as before.
307    pub fn permissions(&self) -> Permissions {
308        let bits = self.perms_p().map_or(self.dict.p, |p| self.dict.p & p);
309        Permissions { bits, owner: self.auth == Auth::Owner, revision: self.dict.r }
310    }
311
312    /// `/P` as `/Perms` holds it: AES-256 ECB of one block (= CBC with a zero IV), marked `adb`.
313    fn perms_p(&self) -> Option<i32> {
314        if !matches!(self.dict.r, 5 | 6) || !self.has_key() {
315            return None;
316        }
317        let plain = Zeroizing::new(aes_cbc_decrypt(&self.key, &[0; 16], self.dict.perms.first_chunk::<16>()?, false));
318        let p = i32::from_le_bytes(*plain.first_chunk::<4>()?);
319        (plain.get(9..12) == Some(&b"adb"[..])).then_some(p)
320    }

340        }
341        match stream {

349    /// Algorithm 1 (§7.6.3.2): the key for one object. AES-256 uses the file key directly
350    /// (Algorithm 1.A, §7.6.3.3).
351    ///
352    /// For an RC4 crypt filter under `/V 4`, a file key shorter than 16 bytes is first extended
353    /// to 16 bytes with zeros. The standard doesn't say so (Algorithm 1 takes the n-byte key as
354    /// it is), but that is how Acrobat writes and reads such files: a `/V 4` RC4 crypt filter
355    /// with a 40- or 48-bit key (pdf.js issue 19484). The renderer's `hayro-syntax` reads them
356    /// the same way, so a page it shows is one this crate can read and save. AES-128 keys are
357    /// left as they are: `AESV2` takes a 128-bit key (Table 25), and Acrobat's extension is only
358    /// seen with RC4. Every key PdfCraft writes under `/V 4` is 16 bytes already.
359    fn object_key(&self, method: Method, num: u32, generation: u16) -> Zeroizing<Vec<u8>> {
360        if method == Method::Aes256 {
361            return self.key.clone();
362        }
363        let mut key = self.key.clone();
364        if method == Method::Rc4 && self.dict.v == 4 && key.len() < 16 {
365            key.resize(16, 0);
366        }
367        let mut h = Md5::new();
368        h.update(&key);
369        h.update(&num.to_le_bytes()[..3]);
370        h.update(generation.to_le_bytes());
371        if method == Method::Aes128 {
372            h.update(b"sAlT");
373        }
374        let mut digest = Zeroizing::new(h.finalize().to_vec());
375        digest.truncate((key.len() + 5).min(16));
376        digest
377    }

379    fn apply(&self, method: Method, num: u32, generation: u16, data: &[u8], encrypt: bool) -> Vec<u8> {
380        // Without the key, encrypted data stays as stored (see `without_key`).
381        if !self.has_key() {
382            return data.to_vec();
383        }
384        match method {
385            Method::Identity => data.to_vec(),
386            Method::Rc4 => rc4(&self.object_key(method, num, generation), data),
387            Method::Aes128 | Method::Aes256 => {
388                let key = self.object_key(method, num, generation);
389                if encrypt {
390                    // Deterministic IV derived from the key, object and content: saves are
391                    // reproducible and no IV repeats for different plaintexts.
392                    let mut h = Sha256::new();
393                    h.update(&key);
394                    h.update(num.to_le_bytes());
395                    h.update(generation.to_le_bytes());
396                    h.update(data);
397                    let digest = h.finalize();
398                    let mut iv = [0u8; 16];
399                    for (d, s) in iv.iter_mut().zip(digest.iter()) {
400                        *d = *s;
401                    }
402                    let mut out = iv.to_vec();
403                    out.extend(aes_cbc_encrypt(&key, &iv, data, true));
404                    out
405                } else {
406                    let Some((iv, body)) = data.split_first_chunk::<16>() else {
407                        return Vec::new(); // only (part of) an IV: empty plaintext
408                    };
409                    aes_cbc_decrypt(&key, iv, body, true)
410                }
411            }
412        }
413    }

439/// PDFDocEncoding (ISO 32000-2 Annex D) where it differs from Latin-1: bytes 0x18–0x1F and
440/// 0x80–0xA0 (0x9F is undefined). Here rather than in pdfcraft-cos, which decodes text strings
441/// with it, because R2–R4 passwords need it too and cos depends on this crate.
442const PDFDOC_LOW: [char; 8] = ['˘', 'ˇ', 'ˆ', '˙', '˝', '˛', '˚', '˜'];
443const PDFDOC_HIGH: [char; 33] = [
444    '•', '†', '‡', '…', '—', '–', 'ƒ', '⁄', '‹', '›', '−', '‰', '„', '“', '”', '‘', '’', '‚', '™', 'ﬁ', 'ﬂ', 'Ł', 'Œ', 'Š', 'Ÿ', 'Ž', 'ı', 'ł', 'œ',
445    'š', 'ž', '\u{FFFD}', '€',
446];
447
448/// PDFDocEncoding → Unicode.
449pub fn pdfdoc_char(c: u8) -> char {
450    match c {
451        0x18..=0x1F => PDFDOC_LOW[usize::from(c - 0x18)],
452        0x80..=0xA0 => PDFDOC_HIGH[usize::from(c - 0x80)],
453        _ => char::from(c),
454    }
455}
456
457/// Unicode → PDFDocEncoding, `None` for a character it lacks (ğ, ş, İ, Cyrillic, CJK, …).
458pub fn pdfdoc_byte(c: char) -> Option<u8> {
459    let at = |table: &[char], base: u8| table.iter().position(|&t| t == c && t != '\u{FFFD}').and_then(|i| u8::try_from(i).ok()).map(|i| base + i);
460    at(&PDFDOC_LOW, 0x18).or_else(|| at(&PDFDOC_HIGH, 0x80)).or_else(|| match u32::from(c) {
461        0x18..=0x1F | 0x80..=0xA0 => None,
462        n => u8::try_from(n).ok(),
463    })
464}

466/// Revisions 2–4: the password in PDFDocEncoding (§7.6.4.3.2, Algorithm 2), at most 32 bytes.
467/// Characters PDFDocEncoding lacks are dropped.
468fn legacy_password(pw: &str) -> Zeroizing<Vec<u8>> {
469    Zeroizing::new(pw.chars().filter_map(pdfdoc_byte).take(32).collect())
470}
471
472/// The bytes an R2–R4 password may have been written as: PDFDocEncoding, as the specification
473/// asks; and for a password with characters PDFDocEncoding lacks (ğ, ş, İ, …), which it gives
474/// no bytes, also its UTF-8 bytes, which qpdf and other tools write for such passwords.
475fn legacy_candidates(pw: &str) -> Vec<Zeroizing<Vec<u8>>> {
476    let mut out = vec![legacy_password(pw)];
477    if pw.chars().any(|c| pdfdoc_byte(c).is_none()) {
478        out.push(Zeroizing::new(pw.bytes().take(32).collect()));
479    }
480    out
481}

483/// Revisions 5–6: SASLprep (RFC 4013), UTF-8, at most 127 bytes.
484fn sasl_password(pw: &str) -> Zeroizing<Vec<u8>> {
485    let prepped = Zeroizing::new(stringprep::saslprep(pw).map(|c| c.into_owned()).unwrap_or_else(|_| pw.to_string()));
486    let mut bytes = Zeroizing::new(prepped.as_bytes().to_vec());
487    bytes.truncate(127);
488    bytes
489}

491fn pad(pw: &[u8]) -> Zeroizing<[u8; 32]> {
492    let mut out = Zeroizing::new(PADDING);
493    let n = pw.len().min(32);
494    out[..n].copy_from_slice(&pw[..n]);
495    out[n..].copy_from_slice(&PADDING[..32 - n]);
496    out
497}

499fn key_len(d: &EncryptDict) -> usize {
500    if d.r == 2 { 5 } else { (d.length_bits.clamp(40, 128) / 8) as usize }
501}

505/// Algorithm 2: file key from a (user) password.
506fn file_key_r4(d: &EncryptDict, id0: &[u8], pw: &[u8]) -> Zeroizing<Vec<u8>> {
507    let n = key_len(d);
508    let mut h = Md5::new();
509    h.update(*pad(pw));
510    h.update(&d.o[..d.o.len().min(32)]);
511    h.update(d.p.to_le_bytes());
512    h.update(id0);
513    if d.r >= 4 && !d.encrypt_metadata {
514        h.update([0xFF; 4]);
515    }
516    let mut digest = Zeroizing::new(h.finalize().to_vec());
517    if d.r >= 3 {
518        for _ in 0..50 {
519            digest = Zeroizing::new(Md5::digest(&digest[..n]).to_vec());
520        }
521    }
522    digest.truncate(n);
523    digest
524}

526/// Algorithms 4/5: the `/U` value a key produces.
527fn u_value_r4(d: &EncryptDict, id0: &[u8], key: &[u8]) -> Vec<u8> {
528    if d.r == 2 {
529        return rc4(key, &PADDING);
530    }
531    let mut h = Md5::new();
532    h.update(PADDING);
533    h.update(id0);
534    let mut x = rc4(key, &h.finalize());
535    for i in 1..=19u8 {
536        let k = Zeroizing::new(key.iter().map(|b| b ^ i).collect::<Vec<u8>>());
537        x = rc4(&k, &x);
538    }
539    x.extend_from_slice(&[0u8; 16]);
540    x
541}
542
543/// Algorithm 6: authenticate a user password; returns the file key.
544///
545/// Some producers ignore `/EncryptMetadata false` when deriving the key (they omit the
546/// `FF FF FF FF` of Algorithm 2, step f). Viewers open those files, so if the strict derivation
547/// fails we retry the other way.
548fn user_key_r4(d: &EncryptDict, id0: &[u8], pw: &[u8]) -> Option<Zeroizing<Vec<u8>>> {
549    let check = |d: &EncryptDict| {
550        let key = file_key_r4(d, id0, pw);
551        let u = u_value_r4(d, id0, &key);
552        let n = if d.r == 2 { 32 } else { 16 };
553        (d.u.len() >= n && u[..n] == d.u[..n]).then_some(key)
554    };
555    check(d).or_else(|| (d.r >= 4 && !d.encrypt_metadata).then(|| check(&EncryptDict { encrypt_metadata: true, ..d.clone() })).flatten())
556}
557
558/// The RC4 key derived from an owner password (Algorithm 3, steps a–d).
559fn owner_rc4_key(d: &EncryptDict, owner_pw: &[u8]) -> Zeroizing<Vec<u8>> {
560    let n = key_len(d);
561    let mut digest = Zeroizing::new(Md5::digest(*pad(owner_pw)).to_vec());
562    if d.r >= 3 {
563        for _ in 0..50 {
564            digest = Zeroizing::new(Md5::digest(&*digest).to_vec());
565        }
566    }
567    digest.truncate(n);
568    digest
569}
570
571type SecretBytes = Zeroizing<Vec<u8>>;
572
573/// Algorithm 7: authenticate an owner password (recover the user password from `/O`).
574/// Returns the file key and the recovered user password (padding removed).
575fn owner_key_r4(d: &EncryptDict, id0: &[u8], pw: &[u8]) -> Option<(SecretBytes, SecretBytes)> {
576    let rk = owner_rc4_key(d, pw);
577    let mut user = Zeroizing::new(d.o[..d.o.len().min(32)].to_vec());
578    if d.r == 2 {
579        user = Zeroizing::new(rc4(&rk, &user));
580    } else {
581        for i in (0..=19u8).rev() {
582            let k = Zeroizing::new(rk.iter().map(|b| b ^ i).collect::<Vec<u8>>());
583            user = Zeroizing::new(rc4(&k, &user));
584        }
585    }
586    let key = user_key_r4(d, id0, &user)?;
587    // The recovered value is the padded password: strip the padding suffix.
588    let len = (0..=32).find(|&i| user.len() >= 32 && user[i..32] == PADDING[..32 - i]).unwrap_or(user.len().min(32));
589    user.truncate(len);
590    Some((key, user))
591}

593// ── Revisions 5–6 ──────────────────────────────────────────────────────────────────────────────
594
595/// The hash of a password for R5 (SHA-256) or R6 (Algorithm 2.B).
596fn hash_r6(r: i64, pw: &[u8], salt: &[u8], udata: &[u8]) -> Zeroizing<Vec<u8>> {
597    let mut k = Zeroizing::new({
598        let mut h = Sha256::new();
599        h.update(pw);
600        h.update(salt);
601        h.update(udata);
602        h.finalize().to_vec()
603    });
604    if r == 5 {
605        return k;
606    }
607    let mut round = 0u32;
608    loop {
609        let mut k1 = Zeroizing::new(Vec::with_capacity(64 * (pw.len() + k.len() + udata.len())));
610        for _ in 0..64 {
611            k1.extend_from_slice(pw);
612            k1.extend_from_slice(&k);
613            k1.extend_from_slice(udata);
614        }
615        // `k` is always a SHA-2 digest (32, 48 or 64 bytes): key = its first 16 bytes, IV = the next 16.
616        let Some((key, rest)) = k.split_first_chunk::<16>() else { break };
617        let Some(iv) = rest.first_chunk::<16>() else { break };
618        let e = Zeroizing::new(aes_cbc_encrypt(key, iv, &k1, false));
619        let m = e.iter().take(16).map(|b| u32::from(*b)).sum::<u32>() % 3;
620        k = Zeroizing::new(match m {
621            0 => Sha256::digest(&*e).to_vec(),
622            1 => Sha384::digest(&*e).to_vec(),
623            _ => Sha512::digest(&*e).to_vec(),
624        });
625        round += 1;
626        if round >= 64 && e.last().is_none_or(|b| u32::from(*b) <= round - 32) {
627            break;
628        }
629    }
630    k.truncate(32);
631    k
632}
633
634/// Algorithm 2.A: authenticate as owner (`owner = true`) or user; returns the file key.
635fn key_r6(d: &EncryptDict, pw: &[u8], owner: bool) -> Result<Option<Zeroizing<Vec<u8>>>, CryptError> {
636    if d.u.len() < 48 || d.o.len() < 48 {
637        return Err(CryptError::Malformed("/O and /U must be 48 bytes for R5/R6".into()));
638    }
639    let (hashed, encrypted, udata) = if owner { (&d.o[..48], &d.oe, &d.u[..48]) } else { (&d.u[..48], &d.ue, &[][..]) };
640    if *hash_r6(d.r, pw, &hashed[32..40], udata) != hashed[..32] {
641        return Ok(None);
642    }
643    if encrypted.len() < 32 {
644        return Err(CryptError::Malformed("/OE or /UE must be 32 bytes".into()));
645    }
646    let k = hash_r6(d.r, pw, &hashed[40..48], udata);
647    Ok(Some(Zeroizing::new(aes_cbc_decrypt(&k, &[0; 16], &encrypted[..32], false))))
648}

652/// Which encryption to create.
653#[derive(Clone, Copy, Debug, PartialEq, Eq)]
654pub enum Algorithm {
655    /// R2, 40-bit RC4 (legacy; for tests and compatibility only).
656    Rc4_40,
657    /// R3, 128-bit RC4.
658    Rc4_128,
659    /// R4, AES-128 crypt filters.
660    Aes128,
661    /// R6, AES-256 (PDF 2.0; the default for new documents).
662    Aes256,
663}
664
665/// A password as [`create`] prepares it: for R2–R4 its PDFDocEncoding (characters it lacks
666/// dropped), at most 32 bytes; for R6 its SASLprep (RFC 4013: soft hyphens and the like removed,
667/// other spaces made plain, then NFKC; the password as given if SASLprep refuses it), in UTF-8,
668/// at most 127 bytes.
669fn prepared_password(algorithm: Algorithm, password: &str) -> Zeroizing<Vec<u8>> {
670    match algorithm {
671        Algorithm::Rc4_40 | Algorithm::Rc4_128 | Algorithm::Aes128 => legacy_password(password),
672        Algorithm::Aes256 => sasl_password(password),
673    }
674}
675
676/// The bytes new encryption with `algorithm` derives its keys from for `password`: for R2–R4 its
677/// PDFDocEncoding cut or padded to exactly 32 bytes (Algorithm 2); for R6 its SASLprep (the
678/// password as given if SASLprep refuses it), at most 127 bytes of UTF-8. Two passwords with the
679/// same bytes are the same password to the file, and a password with the bytes of `""` opens it
680/// without asking, so callers can refuse either before encrypting. Different bytes can still give
681/// the same key (RC4-40 keeps 5 bytes of a hash). `create` still uses the user password for an
682/// empty R2–R4 owner password.
683pub fn effective_password(algorithm: Algorithm, password: &str) -> Vec<u8> {
684    let prepared = prepared_password(algorithm, password);
685    match algorithm {
686        Algorithm::Rc4_40 | Algorithm::Rc4_128 | Algorithm::Aes128 => pad(&prepared).to_vec(),
687        Algorithm::Aes256 => prepared.to_vec(),
688    }
689}
690
691/// Parameters for new encryption.
692#[derive(Clone, Debug)]
693pub struct NewEncryption<'a> {
694    pub algorithm: Algorithm,
695    pub user_password: &'a str,
696    pub owner_password: &'a str,
697    pub permissions: i32,
698    pub encrypt_metadata: bool,
699    /// 32 bytes of entropy for keys and salts (the caller supplies randomness, so this crate
700    /// needs no OS access and tests are deterministic).
701    pub seed: [u8; 32],
702}
703
704/// Expand `seed` into `n` bytes (SHA-256 in counter mode).
705fn expand(seed: &[u8; 32], label: &[u8], n: usize) -> Vec<u8> {
706    let mut out = Vec::new();
707    let mut counter = 0u32;
708    while out.len() < n {
709        let mut h = Sha256::new();
710        h.update(seed);
711        h.update(label);
712        h.update(counter.to_le_bytes());
713        out.extend_from_slice(&h.finalize());
714        counter = counter.saturating_add(1);
715    }
716    out.truncate(n);
717    out
718}
719
720/// Sixteen bytes of file identifier from `seed` (SHA-256 in counter mode, label `id`).
721///
722/// `/ID` is written in the clear. Deriving it the same way as the file key keeps that public
723/// value from being an unkeyed hash of the seed.
724pub fn file_id(seed: &[u8; 32]) -> Vec<u8> {
725    expand(seed, b"id", 16)
726}

728/// Build the `/Encrypt` values and an authenticated (owner) handler for a new encryption.
729pub fn create(params: &NewEncryption, id0: &[u8]) -> Result<SecurityHandler, CryptError> {
730    // Bits 7–8 and 13–32 must be 1; bits 1–2 must be 0 (§7.6.4.2).
731    let p = (params.permissions | !0xFFF | 0xC0) & !0x3;
732    let mut d = EncryptDict { filter: b"Standard".to_vec(), p, encrypt_metadata: params.encrypt_metadata, ..Default::default() };
733    let user = prepared_password(params.algorithm, params.user_password);
734    let owner = prepared_password(params.algorithm, params.owner_password);
735    match params.algorithm {
736        Algorithm::Rc4_40 | Algorithm::Rc4_128 | Algorithm::Aes128 => {
737            (d.v, d.r, d.length_bits) = match params.algorithm {
738                Algorithm::Rc4_40 => (1, 2, 40),
739                Algorithm::Rc4_128 => (2, 3, 128),
740                _ => (4, 4, 128),
741            };
742            if params.algorithm == Algorithm::Aes128 {
743                d.crypt_filters = vec![(b"StdCF".to_vec(), Method::Aes128)];
744                d.stm_f = b"StdCF".to_vec();
745                d.str_f = b"StdCF".to_vec();
746            } else {
747                d.encrypt_metadata = true; // V < 4 always encrypts metadata
748            }
749            let owner = if params.owner_password.is_empty() { user.clone() } else { owner };
750            // Algorithm 3: /O
751            let rk = owner_rc4_key(&d, &owner);
752            let mut o = rc4(&rk, &*pad(&user));
753            if d.r >= 3 {
754                for i in 1..=19u8 {
755                    let k = Zeroizing::new(rk.iter().map(|b| b ^ i).collect::<Vec<u8>>());
756                    o = rc4(&k, &o);
757                }
758            }
759            d.o = o;
760            let key = file_key_r4(&d, id0, &user);
761            d.u = u_value_r4(&d, id0, &key);
762            Ok(SecurityHandler::new(d, key, Auth::Owner, Some(user)))
763        }
764        Algorithm::Aes256 => {
765            (d.v, d.r, d.length_bits) = (5, 6, 256);
766            d.crypt_filters = vec![(b"StdCF".to_vec(), Method::Aes256)];
767            d.stm_f = b"StdCF".to_vec();
768            d.str_f = b"StdCF".to_vec();
769            let key = Zeroizing::new(expand(&params.seed, b"file key", 32));
770            let salts = expand(&params.seed, b"salts", 32);
771            // Algorithm 8: /U, /UE
772            let mut u = hash_r6(6, &user, &salts[0..8], &[]).to_vec();
773            u.extend_from_slice(&salts[0..16]);
774            let uk = hash_r6(6, &user, &salts[8..16], &[]);
775            d.ue = aes_cbc_encrypt(&uk, &[0; 16], &key, false);
776            // Algorithm 9: /O, /OE
777            let mut o = hash_r6(6, &owner, &salts[16..24], &u).to_vec();
778            o.extend_from_slice(&salts[16..32]);
779            let ok = hash_r6(6, &owner, &salts[24..32], &u);
780            d.oe = aes_cbc_encrypt(&ok, &[0; 16], &key, false);
781            d.u = u;
782            d.o = o;
783            // Algorithm 10: /Perms
784            let mut perms = [0u8; 16];
785            perms[..4].copy_from_slice(&p.to_le_bytes());
786            perms[4..8].copy_from_slice(&[0xFF; 4]);
787            perms[8] = if d.encrypt_metadata { b'T' } else { b'F' };
788            perms[9..12].copy_from_slice(b"adb");
789            perms[12..16].copy_from_slice(&expand(&params.seed, b"perms", 4));
790            d.perms = aes_cbc_encrypt(&key, &[0; 16], &perms, false);
791            Ok(SecurityHandler::new(d, key, Auth::Owner, None))
792        }
793    }
794}
