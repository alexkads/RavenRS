1//! AES-CBC (128 or 256-bit key, chosen by key length) with optional PKCS#5 padding.
2//!
3//! Decryption is tolerant, as viewers are: a trailing partial block is ignored, and invalid
4//! padding leaves the data unpadded rather than failing.
5
6use aes::cipher::{Array, BlockCipherDecrypt, BlockCipherEncrypt, KeyInit};
7use aes::{Aes128, Aes256};
8
9enum Cipher {
10    A128(Box<Aes128>),
11    A256(Box<Aes256>),
12}
13
14impl Cipher {
15    fn new(key: &[u8]) -> Self {
16        if let Some(k) = key.first_chunk::<32>() {
17            Cipher::A256(Box::new(Aes256::new(&Array::from(*k))))
18        } else {
19            let mut k = [0u8; 16];
20            let n = key.len().min(16);
21            k[..n].copy_from_slice(&key[..n]);
22            Cipher::A128(Box::new(Aes128::new(&Array::from(k))))
23        }
24    }
25
26    fn encrypt(&self, block: &mut [u8; 16]) {
27        let mut b = Array::from(*block);
28        match self {
29            Cipher::A128(c) => c.encrypt_block(&mut b),
30            Cipher::A256(c) => c.encrypt_block(&mut b),
31        }
32        *block = b.into();
33    }
34
35    fn decrypt(&self, block: &mut [u8; 16]) {
36        let mut b = Array::from(*block);
37        match self {
38            Cipher::A128(c) => c.decrypt_block(&mut b),
39            Cipher::A256(c) => c.decrypt_block(&mut b),
40        }
41        *block = b.into();
42    }
43}
44
45/// Encrypt; with `pad`, PKCS#5 padding is added (otherwise `data` must be whole blocks, and a
46/// trailing partial block is zero-padded).
47pub fn aes_cbc_encrypt(key: &[u8], iv: &[u8; 16], data: &[u8], pad: bool) -> Vec<u8> {
48    let c = Cipher::new(key);
49    let mut buf = data.to_vec();
50    if pad {
51        let n = 16 - buf.len() % 16;
52        buf.extend(std::iter::repeat_n(n as u8, n));
53    } else if !buf.len().is_multiple_of(16) {
54        buf.resize(buf.len().div_ceil(16) * 16, 0);
55    }
56    let mut prev = *iv;
57    for b in buf.as_chunks_mut::<16>().0 {
58        for (x, p) in b.iter_mut().zip(prev) {
59            *x ^= p;
60        }
61        c.encrypt(b);
62        prev = *b;
63    }
64    buf
65}
66
67/// Decrypt; with `pad`, PKCS#5 padding is removed when valid.
68pub fn aes_cbc_decrypt(key: &[u8], iv: &[u8; 16], data: &[u8], pad: bool) -> Vec<u8> {
69    let c = Cipher::new(key);
70    let blocks = data.as_chunks::<16>().0;
71    let mut out = Vec::with_capacity(blocks.len() * 16);
72    let mut prev = *iv;
73    for &cipher in blocks {
74        let mut b = cipher;
75        c.decrypt(&mut b);
76        for (x, p) in b.iter_mut().zip(prev) {
77            *x ^= p;
78        }
79        out.extend_from_slice(&b);
80        prev = cipher;
81    }
82    if pad && let Some(&n) = out.last() {
83        let n = n as usize;
84        if (1..=16).contains(&n) && n <= out.len() && out[out.len() - n..].iter().all(|b| *b as usize == n) {
85            out.truncate(out.len() - n);
86        }
87    }
88    out
89}
