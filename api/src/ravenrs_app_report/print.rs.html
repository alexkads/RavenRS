1//! Stored reports as things to hand out: on these records, this many of each, to the printer or to a
2//! file. Labels are the usual case, and a report of any size can go the same way.
3
4use ravenrs_app::{Object, Value};
5use ravenrs_app_ui::Session;
6use ravenrs_report::{Document, Report, ReportEngine};
7use ravenrs_report_label::{LabelJob, compile};
8
9use crate::stored::{REPORT_DEFINITION, ReportWords, design_report, every_record, report_of, rows_of};
10
11/// The reports kept for this entity, by what people call them.
12pub fn stored_reports_for(session: &Session, entity: &str) -> Vec<Object> {
13    every_record(session, REPORT_DEFINITION, None)
14        .into_iter()
15        .filter(|report| report.get("target_entity") == Value::Text(entity.to_owned()))
16        .collect()
17}
18
19/// Opens the designer on a stored report, for changing its layout.
20pub fn design_stored_report(session: &Session, report: &Object, words: &ReportWords) -> Result<(), String> {
21    design_report(session, report, words)
22}
23
24/// Each record as many times as asked, as the rows of a report.
25fn repeated(records: &[(Object, u32)]) -> Vec<Object> {
26    records
27        .iter()
28        .flat_map(|(record, copies)| std::iter::repeat_n(record.clone(), *copies as usize))
29        .collect()
30}
31
32/// A stored report over these records, each as many times as asked.
33pub fn render_stored_report(
34    session: &Session,
35    report: &Object,
36    records: &[(Object, u32)],
37    words: &ReportWords,
38) -> Result<Document, String> {
39    let (layout, entity) = report_of(session, report, words)?;
40    let rows = rows_of(session, &entity, &repeated(records));
41    let document = ReportEngine::new().render(&layout, &rows);
42    match document.errors.first() {
43        Some(problem) => Err(problem.clone()),
44        None => Ok(document),
45    }
46}
47
48/// Like [`render_stored_report`], for a caller with something to check or change in the layout first.
49pub fn render_stored_report_with(
50    session: &Session,
51    report: &Object,
52    records: &[(Object, u32)],
53    words: &ReportWords,
54    prepare: impl FnOnce(&mut Report) -> Result<(), String>,
55) -> Result<Document, String> {
56    let (mut layout, entity) = report_of(session, report, words)?;
57    prepare(&mut layout)?;
58    let rows = rows_of(session, &entity, &repeated(records));
59    let document = ReportEngine::new().render(&layout, &rows);
60    match document.errors.first() {
61        Some(problem) => Err(problem.clone()),
62        None => Ok(document),
63    }
64}

66/// What a label printer takes for this document.
67pub fn label_bytes(document: &Document, job: &LabelJob) -> Result<Vec<u8>, String> {
68    compile(document, job).map_err(|error| error.to_string())
69}

71/// Sends what a printer takes, out of the way of the screens, and says how it went.
72#[cfg(not(target_arch = "wasm32"))]
73pub fn send_to_printer(
74    session: &Session,
75    title: impl Into<String>,
76    bytes: Vec<u8>,
77    output: impl ravenrs_report_label::LabelOutput + Send + 'static,
78    done: impl FnOnce(&Session, Result<(), String>) + 'static,
79) {
80    session.run_background(
81        title,
82        move |_progress| output.send(&bytes).map_err(|error| error.to_string()),
83        done,
84    );
