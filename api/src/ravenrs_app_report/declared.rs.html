1//! Panels written in code: the same tiles of numbers and charts people keep as records, declared
2//! next to the entities, so an application shows its figures without seeding records by hand.
3//!
4//! ```
5//! use ravenrs_app::BusinessObject;
6//! use ravenrs_app_report::{AnalyticsApp, AnalyticsPanel};
7//! use ravenrs_app_ui::App;
8//!
9//! #[derive(BusinessObject)]
10//! struct Sale {
11//!     #[field(key)]
12//!     id: i64,
13//!     item: String,
14//!     amount: f64,
15//! }
16//!
17//! let panel = AnalyticsPanel::new("Sales", "Vendas")
18//!     .navigation("Relatórios")
19//!     .number::<Sale>("Total vendido", "amount:Sum")
20//!     .number::<Sale>("Vendas", "")
21//!     .bar::<Sale>("Por item", "item", "amount:Sum")
22//!     .pie::<Sale>("Divisão", "item", "amount:Sum");
23//! let app = App::new("Loja").entity::<Sale>().analytics_panel(panel);
24//! # let _ = app;
25//! ```
26//!
27//! Each tile reads the records as the person signed in can read them, like a kept panel. The panel
28//! is a [`DashboardInfo`] whose pieces are controls the session registers; without an
29//! [`App`](ravenrs_app_ui::App), add [`AnalyticsPanel::info`] to the builder and call
30//! [`AnalyticsPanel::register`] with the session.
31
32use ravenrs_app::{BusinessObject, DashboardInfo};
33use ravenrs_app_ui::{App, Session};
34use ravenrs_widgets::AnyView;
35
36use crate::analytics::{TileSpec, tile_content};
37
38/// A chart a tile of an [`AnalyticsPanel`] can draw.
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40#[non_exhaustive]
41pub enum PanelChart {
42    /// Bars, one per argument.
43    Bar,
44    /// A line over the arguments.
45    Line,
46    /// A filled line.
47    Area,
48    /// A pie.
49    Pie,
50    /// A pie with a hole.
51    Doughnut,
52}
53
54impl PanelChart {
55    fn kind(self) -> &'static str {
56        match self {
57            Self::Bar => "bar",
58            Self::Line => "line",
59            Self::Area => "area",
60            Self::Pie => "pie",
61            Self::Doughnut => "doughnut",
62        }
63    }
64}
65
66/// A panel of numbers and charts declared in code, shown in the navigation like any view.
67///
68/// `value` is what a kept tile calls its series: `field:Sum`, `field:Average`, `field:Min`,
69/// `field:Max` or `field:Count`, and an empty one counts the records. `argument` is the field
70/// along the axis or around the pie.
71#[derive(Clone, Debug, PartialEq, Eq)]
72pub struct AnalyticsPanel {
73    name: String,
74    caption: String,
75    navigation: Option<String>,
76    icon: Option<String>,
77    tiles: Vec<TileSpec>,
78}
79
80impl AnalyticsPanel {
81    /// A panel with no tiles yet. `name` is what its view's id starts with, as in
82    /// `Sales_DashboardView`; `caption` is what people see.
83    pub fn new(name: impl Into<String>, caption: impl Into<String>) -> Self {
84        Self { name: name.into(), caption: caption.into(), navigation: None, icon: None, tiles: Vec::new() }
85    }
86
87    /// The group it goes under in the navigation.
88    pub fn navigation(mut self, group: impl Into<String>) -> Self {
89        self.navigation = Some(group.into());
90        self
91    }
92
93    /// The name of its icon; `layout-dashboard` when not given.
94    pub fn icon(mut self, icon: impl Into<String>) -> Self {
95        self.icon = Some(icon.into());
96        self
97    }
98
99    /// A number over the records of `T`.
100    pub fn number<T: BusinessObject>(self, caption: impl Into<String>, value: impl Into<String>) -> Self {
101        self.number_in(T::entity_info().name, caption, value)
102    }
103
104    /// Bars over the records of `T`, one per `argument`.
105    pub fn bar<T: BusinessObject>(
106        self,
107        caption: impl Into<String>,
108        argument: impl Into<String>,
109        value: impl Into<String>,
110    ) -> Self {
111        self.chart_in(T::entity_info().name, PanelChart::Bar, caption, argument, value)
112    }
113
114    /// A line over the records of `T`.
115    pub fn line<T: BusinessObject>(
116        self,
117        caption: impl Into<String>,
118        argument: impl Into<String>,
119        value: impl Into<String>,
120    ) -> Self {
121        self.chart_in(T::entity_info().name, PanelChart::Line, caption, argument, value)
122    }
123
124    /// A pie over the records of `T`, one slice per `argument`.
125    pub fn pie<T: BusinessObject>(
126        self,
127        caption: impl Into<String>,
128        argument: impl Into<String>,
129        value: impl Into<String>,
130    ) -> Self {
131        self.chart_in(T::entity_info().name, PanelChart::Pie, caption, argument, value)
132    }
133
134    /// A doughnut over the records of `T`.
135    pub fn doughnut<T: BusinessObject>(
136        self,
137        caption: impl Into<String>,
138        argument: impl Into<String>,
139        value: impl Into<String>,
140    ) -> Self {
141        self.chart_in(T::entity_info().name, PanelChart::Doughnut, caption, argument, value)
142    }
143
144    /// A number over the records of an entity known by name, for entities declared at run time.
145    pub fn number_in(
146        mut self,
147        entity: impl Into<String>,
148        caption: impl Into<String>,
149        value: impl Into<String>,
150    ) -> Self {
151        self.tiles.push(TileSpec {
152            title: caption.into(),
153            target: entity.into(),
154            kind: "number".into(),
155            series: value.into(),
156            ..TileSpec::default()
157        });
158        self
159    }
160
161    /// A chart over the records of an entity known by name.
162    pub fn chart_in(
163        mut self,
164        entity: impl Into<String>,
165        chart: PanelChart,
166        caption: impl Into<String>,
167        argument: impl Into<String>,
168        value: impl Into<String>,
169    ) -> Self {
170        self.tiles.push(TileSpec {
171            title: caption.into(),
172            target: entity.into(),
173            kind: chart.kind().into(),
174            argument: argument.into(),
175            series: value.into(),
176            ..TileSpec::default()
177        });
178        self
179    }
180
181    /// Only the records that match `criteria`, in the language of filters, for the last tile.
182    pub fn filtered(mut self, criteria: impl Into<String>) -> Self {
183        if let Some(tile) = self.tiles.last_mut() {
184            tile.filter = criteria.into();
185        }
186        self
187    }
188
189    /// Dates along the axis of the last tile grouped by `Day`, `Month` or `Year`.
190    pub fn by(mut self, interval: impl Into<String>) -> Self {
191        if let Some(tile) = self.tiles.last_mut() {
192            tile.interval = interval.into();
193        }
194        self
195    }
196
197    fn control(&self, index: usize) -> String {
198        format!("{}.tile{index}", self.name)
199    }
200
201    /// The dashboard of the model: one control per tile, in order.
202    pub fn info(&self) -> DashboardInfo {
203        let mut info = DashboardInfo::new(self.name.clone(), self.caption.clone())
204            .icon(self.icon.clone().unwrap_or_else(|| "layout-dashboard".into()));
205        if let Some(group) = &self.navigation {
206            info = info.navigation(group.clone());
207        }
208        for (index, tile) in self.tiles.iter().enumerate() {
209            info = info.control(tile.title.clone(), self.control(index));
210        }
211        info
212    }
213
214    /// Registers the tiles as controls of `session`, which [`info`](Self::info) names.
215    pub fn register(&self, session: &Session) {
216        for (index, tile) in self.tiles.iter().enumerate() {
217            let (frame, tile) = (session.frame(), tile.clone());
218            // The frame links back to its session weakly, so the control keeps no cycle alive.
219            session.register_control(self.control(index), move || match Session::of_frame(&frame) {
220                Some(session) => {
221                    // Read again when records change.
222                    session.revision();
223                    tile_content(&session, &tile)
224                }
225                None => Box::new(ravenrs_widgets::Text::new(tile.title.clone())) as AnyView,
226            });
227        }
228    }
229}

231/// Adds [`AnalyticsPanel`]s to an [`App`].
232pub trait AnalyticsApp {
233    /// Shows `panel` in the navigation, with its tiles read from the records.
234    #[must_use]
235    fn analytics_panel(self, panel: AnalyticsPanel) -> Self;
236}
237
238impl AnalyticsApp for App {
239    fn analytics_panel(self, panel: AnalyticsPanel) -> Self {
240        self.dashboard(panel.info()).with_session(move |session| panel.register(session))
241    }
