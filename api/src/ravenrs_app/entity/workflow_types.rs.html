1//! What an entity's workflow says: the states of a field and the moves between them, and reminders.
2//! The entity keeps them; the code that runs them lives in its own module.

9/// A way from one state to another.
10#[derive(Clone, Debug, PartialEq, Eq)]
11#[non_exhaustive]
12#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
13pub struct Transition {
14    /// The state it leaves.
15    pub from: String,
16    /// The state it reaches.
17    pub to: String,
18    /// What its action says.
19    pub caption: String,
20    /// An expression the record has to meet, as it would be after the change.
21    pub when: Option<String>,
22}

24/// The states of a record, in one of its fields, and the ways between them.
25///
26/// ```
27/// use ravenrs_app::{EntityInfo, EnumOption, FieldInfo, FieldKind, StateMachine};
28///
29/// let options = vec![EnumOption::new("draft", "Draft"), EnumOption::new("done", "Done")];
30/// let entity = EntityInfo::new("Task")
31///     .field(FieldInfo::new("id", FieldKind::Integer).key())
32///     .field(FieldInfo::new("state", FieldKind::Enum(options)))
33///     .state_machine(
34///         StateMachine::new("state").state("draft", "Draft").state("done", "Done").start("draft").transition(
35///             "draft",
36///             "done",
37///             "Finish",
38///         ),
39///     );
40/// assert!(entity.check().is_ok());
41/// ```
42#[derive(Clone, Debug, Default, PartialEq, Eq)]
43#[non_exhaustive]
44#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
45pub struct StateMachine {
46    /// The field that holds the state: options, text or a reference.
47    pub field: String,
48    /// The states, as stored and as people see them.
49    pub states: Vec<(String, String)>,
50    /// Where a new record has to start; nothing means any state.
51    pub start: Option<String>,
52    /// The ways between states.
53    pub transitions: Vec<Transition>,
54}
55
56/// The field of a record that says when to remind people of it, and what to say.
57#[derive(Clone, Debug, PartialEq, Eq)]
58#[non_exhaustive]
59#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
60pub struct Reminder {
61    /// The date and time it is due; empty means nothing to remind.
62    pub field: String,
63    /// The field that says what it is about; nothing means what names the record.
64    pub message: Option<String>,
65    /// An expression for whom it is, where `?CurrentUserId` and `?CurrentUserName` are the person.
66    pub criteria: Option<String>,
67}

69/// The date and time on this machine, as people read it on a clock.
70pub fn local_now() -> ravenrs_criteria::DateTime {
71    #[cfg(target_arch = "wasm32")]
72    {
73        ravenrs_criteria::DateTime::now()
74    }
75    #[cfg(not(target_arch = "wasm32"))]
76    {
77        let utc = ravenrs_criteria::DateTime::now();
78        let offset = i64::from(jiff::tz::TimeZone::system().to_offset(jiff::Timestamp::now()).seconds());
79        let seconds = i64::from(utc.seconds) + offset;
80        ravenrs_criteria::DateTime {
81            days: utc.days + seconds.div_euclid(86_400),
82            seconds: seconds.rem_euclid(86_400) as u32,
83        }
84    }
85}

87/// Whether a change keeps to the state machines of its entity. Whoever keeps records calls this
88/// with what is stored and what is about to be, before writing, as every space here does.
89pub fn check_transitions(entity: &EntityInfo, old: Option<&Object>, new: &Object) -> Result<(), SpaceError> {
90    entity.state_machines.iter().try_for_each(|machine| machine.check(entity, old, new))
91}
92
93#[cfg_attr(feature = "live", ravenrs_live::expose)]
94impl StateMachine {
95    pub(crate) fn caption_of<'a>(&'a self, code: &'a str) -> &'a str {
96        self.states.iter().find(|(known, _)| known == code).map_or(code, |(_, caption)| caption)
97    }
98
99    /// Whether the record can make this change: to be called with the record as storage has it, or
100    /// nothing if it is new, and as it is about to be.
101    fn check(&self, entity: &EntityInfo, old: Option<&Object>, new: &Object) -> Result<(), SpaceError> {
102        let text = |object: &Object| key_text(&object.get(&self.field)).unwrap_or_default();
103        let to = text(new);
104        let refused = |why: String| Err(SpaceError::Refused(format!("{}: {why}", entity.caption)));
105        let Some(old) = old else {
106            if !self.states.iter().any(|(state, _)| *state == to) {
107                return refused(format!("{to} is not a state"));
108            }
109            return match &self.start {
110                Some(start) if *start != to => {
111                    refused(format!("a new record starts in {}", self.caption_of(start)))
112                }
113                _ => Ok(()),
114            };
115        };
116        let from = text(old);
117        if from == to {
118            return Ok(());
119        }
120        let environment = crate::environment();
121        let mut after = new.clone();
122        entity.compute(&mut after, &environment);
123        let opens = self.transitions.iter().filter(|way| way.from == from && way.to == to).any(|way| {
124            way.when.as_deref().is_none_or(|expression| {
125                // One that does not read, or fails on the record, does not open.
126                Criteria::parse(expression)
127                    .ok()
128                    .is_some_and(|criteria| Scope::new(&environment).row(&after).matches(&criteria))
129            })
130        });
131        if opens {
132            Ok(())
133        } else {
134            refused(format!("it cannot go from {} to {}", self.caption_of(&from), self.caption_of(&to)))
135        }
136    }
137}
