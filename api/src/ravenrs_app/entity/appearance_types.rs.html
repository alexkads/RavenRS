1//! What an appearance rule says: which items, in which context, look how, and the colors it uses.
2//! The entity keeps them; the code that runs them lives in its own module.
3
4use std::rc::Rc;
5
6use crate::Object;
7
8/// A color with a meaning, which every theme has its own version of.
9#[derive(Clone, Copy, Debug, PartialEq, Eq)]
10#[non_exhaustive]
11#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
12pub enum Tint {
13    /// Something wrong.
14    Danger,
15    /// Something to watch.
16    Warning,
17    /// Something right.
18    Success,
19    /// What stands out.
20    Accent,
21    /// What steps back.
22    Muted,
23}
24
25/// What a rule changes the look of.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
27#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
28pub enum AppearanceTarget {
29    /// A field: a cell of a list, an editor on a form.
30    #[default]
31    Item,
32    /// A group or a tab of a form.
33    Layout,
34    /// An action.
35    Action,
36}
37
38/// Which views a rule applies in.
39#[derive(Clone, Debug, Default, PartialEq, Eq)]
40#[non_exhaustive]
41#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
42pub enum AppearanceContext {
43    /// Every view.
44    #[default]
45    Any,
46    /// Only lists.
47    List,
48    /// Only forms.
49    Detail,
50    /// Only the view with this id.
51    View(String),
52}
53
54pub(crate) type Condition = Rc<dyn Fn(&Object) -> bool>;
55
56/// Changes how things look or behave, for records that meet a condition. It is about looks:
57/// it does not protect data, which is what permissions are for.
58#[derive(Clone)]
59#[non_exhaustive]
60#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
61pub struct AppearanceRule {
62    /// What it is known by.
63    pub id: String,
64    /// The kind of thing it changes.
65    pub target: AppearanceTarget,
66    /// The names of those things, where `*` stands for any text. A list that starts with a
67    /// lone `*` means all but the names after it.
68    pub names: Vec<String>,
69    /// Where it applies.
70    pub context: AppearanceContext,
71    /// The expression a record has to meet. With none, and no function, it always applies.
72    pub criteria: Option<String>,
73    /// Whichever is higher wins, for what both change.
74    pub priority: i32,
75    /// The color of the text.
76    pub font: Option<Tint>,
77    /// The color behind it.
78    pub back: Option<Tint>,
79    /// Heavier text.
80    pub bold: Option<bool>,
81    /// Whether it can be used.
82    pub enabled: Option<bool>,
83    /// Whether it shows.
84    pub visible: Option<bool>,
85    pub(crate) function: Option<Condition>,
86}
