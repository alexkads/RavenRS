1//! What a validation rule says: the field, the kind of check, when it applies and how serious it is.
2//! The entity keeps them; the code that runs them lives in its own module.

10/// How a [`RuleKind::Text`] compares.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12#[non_exhaustive]
13#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
14pub enum TextComparison {
15    /// The field holds exactly the text.
16    Equals,
17    /// The field holds something else.
18    NotEquals,
19    /// The text is somewhere in the field.
20    Contains,
21    /// The field starts with the text.
22    StartsWith,
23    /// The field ends with the text.
24    EndsWith,
25}
26
27/// How a [`RuleKind::Compare`] compares.
28#[derive(Clone, Copy, Debug, PartialEq, Eq)]
29#[non_exhaustive]
30#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
31pub enum Comparison {
32    /// The same.
33    Equal,
34    /// Not the same.
35    NotEqual,
36    /// Below.
37    Less,
38    /// Below or the same.
39    LessOrEqual,
40    /// Above.
41    Greater,
42    /// Above or the same.
43    GreaterOrEqual,
44}
45
46/// What a [`RuleKind::Compare`] compares with.
47#[derive(Clone, Debug, PartialEq)]
48#[non_exhaustive]
49#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
50pub enum Operand {
51    /// This value.
52    Value(Value),
53    /// What this other field holds.
54    Field(String),
55}
56
57pub(crate) type CustomRule = Rc<dyn Fn(&Object, &dyn ObjectSpace) -> Result<(), String>>;
58
59/// How much not meeting a rule matters.
60#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
61#[non_exhaustive]
62#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
63pub enum Severity {
64    /// It stops what was being done until it is fixed.
65    #[default]
66    Error,
67    /// People are told, and can go on anyway.
68    Warning,
69    /// It goes on, with a note on the field.
70    Information,
71}

73/// What a [`Rule`] asks for.
74#[derive(Clone)]
75#[non_exhaustive]
76#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
77pub enum RuleKind {
78    /// The field has something in it.
79    Required,
80    /// The field is within these limits.
81    Range {
82        /// The lowest it can be.
83        min: Option<f64>,
84        /// The highest it can be.
85        max: Option<f64>,
86    },
87    /// The field matches this regular expression.
88    Matches(String),
89    /// The field compares with this text.
90    Text {
91        /// How.
92        comparison: TextComparison,
93        /// With what.
94        text: String,
95        /// Whether case counts for nothing.
96        ignore_case: bool,
97    },
98    /// The field compares with a value or another field.
99    Compare {
100        /// How.
101        comparison: Comparison,
102        /// With what.
103        with: Operand,
104    },
105    /// The record meets this expression.
106    Criteria(String),
107    /// The field is true, as with an agreement.
108    IsTrue,
109    /// No other record holds the same in the field.
110    Unique,
111    /// No other record holds the same in all these fields together.
112    UniqueTogether(Vec<String>),
113    /// Some record of this entity meets this expression.
114    Exists {
115        /// The entity.
116        entity: String,
117        /// The expression, or none for any record at all.
118        criteria: Option<String>,
119    },
120    /// Some record of this entity points to this one through this field.
121    Referenced {
122        /// The entity.
123        entity: String,
124        /// Its field that holds this record's key.
125        field: String,
126    },
127    /// Anything else.
128    Custom(CustomRule),
129}

131/// One thing a record has to meet, in some contexts, and perhaps only when it fits.
132#[derive(Clone, Debug)]
133#[non_exhaustive]
134#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
135pub struct Rule {
136    /// What it is known by.
137    pub id: String,
138    /// The field it is about, where the error shows.
139    pub field: Option<String>,
140    /// What it asks for.
141    pub kind: RuleKind,
142    /// When it applies.
143    pub contexts: Vec<String>,
144    /// It applies only to records that meet this.
145    pub target_criteria: Option<String>,
146    /// What to say, instead of the usual words.
147    pub message: Option<String>,
148    /// Whether an empty field passes, as it does for all but [`RuleKind::Required`].
149    pub skip_empty: bool,
150    /// How much not meeting it matters.
151    pub severity: Severity,
152}
