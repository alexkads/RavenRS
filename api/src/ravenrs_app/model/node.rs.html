1//! A node of the application model.
2
3use std::collections::BTreeMap;
4
5use serde_json::Value as Json;
6
7/// A node of the application model: an id, settings and nodes under it.
8#[derive(Clone, Debug, Default, PartialEq)]
9pub struct ModelNode {
10    /// What it is known by among the nodes beside it.
11    pub id: String,
12    /// Its settings.
13    pub properties: BTreeMap<String, Json>,
14    /// The nodes under it, in the order they came.
15    pub children: Vec<ModelNode>,
16}
17
18impl ModelNode {
19    /// A node with no settings and nothing under it.
20    pub fn new(id: impl Into<String>) -> Self {
21        Self { id: id.into(), ..Self::default() }
22    }
23
24    /// With this setting.
25    pub fn with(mut self, name: &str, value: impl Into<Json>) -> Self {
26        self.properties.insert(name.to_owned(), value.into());
27        self
28    }
29
30    /// With this node under it.
31    pub fn child(mut self, child: ModelNode) -> Self {
32        self.children.push(child);
33        self
34    }
35
36    /// The nodes under it, as they are, including those an index hides.
37    pub(crate) fn in_order_all(&self) -> &[ModelNode] {
38        &self.children
39    }
40
41    /// The node under it with this id.
42    pub fn get(&self, id: &str) -> Option<&ModelNode> {
43        self.children.iter().find(|c| c.id == id)
44    }
45
46    pub(crate) fn get_mut(&mut self, id: &str) -> Option<&mut ModelNode> {
47        self.children.iter_mut().find(|c| c.id == id)
48    }
49
50    pub(crate) fn at_mut(&mut self, path: &[&str]) -> Option<&mut ModelNode> {
51        path.iter().try_fold(self, |node, id| node.get_mut(id))
52    }
53
54    /// The node at this path of ids below it.
55    pub fn at(&self, path: &[&str]) -> Option<&ModelNode> {
56        path.iter().try_fold(self, |node, id| node.get(id))
57    }
58
59    /// A setting as text.
60    pub fn text(&self, name: &str) -> Option<&str> {
61        self.properties.get(name).and_then(Json::as_str)
62    }
63
64    /// A setting as it is typed: text as it is, and anything else as JSON.
65    pub fn typed(&self, name: &str) -> Option<String> {
66        self.properties.get(name).map(|value| match value {
67            Json::String(text) => text.clone(),
68            other => other.to_string(),
69        })
70    }
71
72    /// A setting as a whole number.
73    pub fn integer(&self, name: &str) -> Option<i64> {
74        self.properties.get(name).and_then(Json::as_i64)
75    }
76
77    /// A setting as a number.
78    pub fn number(&self, name: &str) -> Option<f64> {
79        self.properties.get(name).and_then(Json::as_f64)
80    }
81
82    /// A setting as yes or no.
83    pub fn flag(&self, name: &str) -> Option<bool> {
84        self.properties.get(name).and_then(Json::as_bool)
85    }
86
87    /// The nodes under it in the order to show them: by `Index`, then as they came.
88    /// A negative index keeps a node out of sight, so it is not here.
89    pub fn in_order(&self) -> Vec<&ModelNode> {
90        let mut shown: Vec<&ModelNode> =
91            self.children.iter().filter(|c| c.integer("Index").is_none_or(|i| i >= 0)).collect();
92        shown.sort_by_key(|c| c.integer("Index").unwrap_or(i64::MAX));
93        shown
94    }
95}
