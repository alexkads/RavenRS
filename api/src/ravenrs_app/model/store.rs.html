1//! Where a layer's difference is kept.
2
3use std::cell::RefCell;
4
5use super::layers::Diff;
6use crate::ModelError;
7
8/// Keeps one layer's difference.
9pub trait ModelStore {
10    /// What it holds; `None` when nothing was kept yet.
11    fn load(&self) -> Result<Option<Diff>, ModelError>;
12    /// Keeps this.
13    fn save(&self, difference: &Diff) -> Result<(), ModelError>;
14}
15
16/// A difference that came inside the program, as a module's does. It cannot change.
17#[derive(Clone, Copy, Debug)]
18pub struct EmbeddedStore(pub &'static str);
19
20impl ModelStore for EmbeddedStore {
21    fn load(&self) -> Result<Option<Diff>, ModelError> {
22        Diff::parse(self.0).map(Some)
23    }
24
25    fn save(&self, _difference: &Diff) -> Result<(), ModelError> {
26        Err(ModelError("a difference inside the program cannot be saved".into()))
27    }
28}
29
30/// A difference that lasts as long as the program, where nothing else is at hand.
31#[derive(Debug, Default)]
32pub struct MemoryModelStore(RefCell<Option<Diff>>);
33
34impl ModelStore for MemoryModelStore {
35    fn load(&self) -> Result<Option<Diff>, ModelError> {
36        Ok(self.0.borrow().clone())
37    }
38
39    fn save(&self, difference: &Diff) -> Result<(), ModelError> {
40        *self.0.borrow_mut() = Some(difference.clone());
41        Ok(())
42    }
43}

45/// A difference in a file. A failure while saving leaves the old file whole.
46#[cfg(not(target_arch = "wasm32"))]
47#[derive(Clone, Debug)]
48pub struct FileStore(pub std::path::PathBuf);
49
50#[cfg(not(target_arch = "wasm32"))]
51impl ModelStore for FileStore {
52    fn load(&self) -> Result<Option<Diff>, ModelError> {
53        match std::fs::read_to_string(&self.0) {
54            Ok(text) => {
55                Diff::parse(&text).map(Some).map_err(|e| ModelError(format!("{}: {e}", self.0.display())))
56            }
57            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
58            Err(e) => Err(ModelError(format!("{}: {e}", self.0.display()))),
59        }
60    }
61
62    fn save(&self, difference: &Diff) -> Result<(), ModelError> {
63        let failed = |e: std::io::Error| ModelError(format!("{}: {e}", self.0.display()));
64        if let Some(folder) = self.0.parent() {
65            std::fs::create_dir_all(folder).map_err(failed)?;
66        }
67        let draft = self.0.with_extension("tmp");
68        std::fs::write(&draft, difference.to_json()).map_err(failed)?;
69        std::fs::rename(&draft, &self.0).map_err(failed)
70    }
71}

73#[cfg(test)]
74mod tests {
75    use super::*;
76    use crate::{ModelLayer, ModelLayers, ModelNode};
77
78    fn node() -> ModelNode {
79        ModelNode::new("Application").child(ModelNode::new("Views"))
80    }
81
82    #[test]
83    fn embedded_differences_load_and_refuse_saving() {
84        let store = EmbeddedStore(r#"{ "Views": { "Caption": "x" } }"#);
85        let mut layers = ModelLayers::new(node());
86        let module = ModelLayer::Module("m".into());
87        layers.load(&module, &store).expect("loads");
88        assert_eq!(layers.merged().root.at(&["Views"]).and_then(|n| n.text("Caption")), Some("x"));
89        assert!(layers.save(&module, &store).is_err());
90    }
91
92    #[test]
93    fn a_memory_store_gives_back_what_it_was_given() {
94        let store = MemoryModelStore::default();
95        let mut layers = ModelLayers::new(node());
96        layers.load(&ModelLayer::User, &store).expect("nothing is fine");
97        layers.set(&ModelLayer::User, &["Views"], "Caption", "x".into());
98        layers.save(&ModelLayer::User, &store).expect("saves");
99        let mut again = ModelLayers::new(node());
100        again.load(&ModelLayer::User, &store).expect("loads");
101        assert_eq!(again, layers);
102    }
103
104    #[cfg(not(target_arch = "wasm32"))]
105    #[test]
106    fn a_file_store_keeps_it_on_disk() {
107        let folder = std::env::temp_dir().join(format!("ravenrs-app-{}", std::process::id()));
108        let store = FileStore(folder.join("deep").join("user.json"));
109        assert_eq!(store.load(), Ok(None));
110        let mut layers = ModelLayers::new(node());
111        layers.set(&ModelLayer::User, &["Views"], "Caption", "x".into());
112        layers.save(&ModelLayer::User, &store).expect("saves");
113        assert_eq!(store.load().expect("loads").as_ref(), layers.difference(&ModelLayer::User));
114        std::fs::write(&store.0, "{ broken").expect("writes");
115        assert!(store.load().is_err_and(|e| e.0.contains("user.json")), "says which file");
116        std::fs::remove_dir_all(folder).expect("cleans up");
117    }
118}
