12#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
13impl StateMachine {
14    /// Over this field.
15    pub fn new(field: impl Into<String>) -> Self {
16        Self { field: field.into(), ..Self::default() }
17    }
18
19    /// A state, as stored and as people see it.
20    pub fn state(mut self, code: impl Into<String>, caption: impl Into<String>) -> Self {
21        self.states.push((code.into(), caption.into()));
22        self
23    }
24
25    /// Where a new record starts, which also is what it starts with.
26    pub fn start(mut self, code: impl Into<String>) -> Self {
27        self.start = Some(code.into());
28        self
29    }
30
31    /// A way between two states, with what its action says.
32    pub fn transition(
33        mut self,
34        from: impl Into<String>,
35        to: impl Into<String>,
36        caption: impl Into<String>,
37    ) -> Self {
38        self.transitions.push(Transition {
39            from: from.into(),
40            to: to.into(),
41            caption: caption.into(),
42            when: None,
43        });
44        self
45    }
46
47    /// The last way declared only opens for records this expression holds for.
48    pub fn when(mut self, expression: impl Into<String>) -> Self {
49        if let Some(last) = self.transitions.last_mut() {
50            last.when = Some(expression.into());
51        }
52        self
53    }
54
55    pub(crate) fn problem(&self) -> Option<String> {
56        let known = |code: &str| self.states.iter().any(|(state, _)| state == code);
57        if self.states.is_empty() {
58            return Some(format!("the states of {} are not declared", self.field));
59        }
60        if let Some(start) = self.start.as_deref().filter(|start| !known(start)) {
61            return Some(format!("{} starts in {start}, which is not one of its states", self.field));
62        }
63        self.transitions.iter().find_map(|transition| {
64            [&transition.from, &transition.to]
65                .into_iter()
66                .find(|state| !known(state))
67                .map(|state| format!("{} goes through {state}, which is not one of its states", self.field))
68                .or_else(|| {
69                    let expression = transition.when.as_deref()?;
70                    Criteria::parse(expression).err().map(|error| format!("{}: {error}", self.field))
71                })
72        })
73    }
74}

110fn action_of(entity: &EntityInfo, machine: &StateMachine, way: &Transition) -> Action {
111    let (field, to) = (machine.field.clone(), way.to.clone());
112    let key = entity.key_field().map(|key| key.name.clone()).unwrap_or_default();
113    let name = entity.name.clone();
114    let id = format!("{}.{}.{}.{}", entity.name, machine.field, way.from, way.to);
115    let mut opens = format!("[{field}] = \'{}\'", way.from.replace('\'', "\'\'"));
116    if let Some(when) = &way.when {
117        opens = format!("({opens}) And ({when})");
118    }
119    Action::simple(id, way.caption.clone())
120        .category(ActionCategory::Edit)
121        .target_entity(entity.name.clone())
122        .selection(Selection::RequireMultiple)
123        .target_criteria(opens, CriteriaMode::TrueForAll)
124        .requires(Operation::Write)
125        .on_execute(move |cx| {
126            let mut space = cx.space.borrow_mut();
127            let result = (|| {
128                for record in &cx.selected {
129                    let mut stored = space
130                        .find(&name, &record.get(&key))?
131                        .ok_or_else(|| SpaceError::Conflict(format!("a record of {name} is gone")))?;
132                    stored.set(&field, Value::Text(to.clone()));
133                    space.save(stored)?;
134                }
135                space.commit()
136            })();
137            if let Err(error) = result {
138                space.rollback();
139                return Err(error.to_string());
140            }
141            drop(space);
142            cx.records_changed();
143            Ok(())
144        })
145}

147/// What gives the entities with state machines their actions. It comes by itself.
148#[derive(Debug)]
149pub(crate) struct StateMachineModule(pub(crate) Vec<EntityInfo>);
150
151impl Module for StateMachineModule {
152    fn name(&self) -> &str {
153        "StateMachines"
154    }
155
156    fn view_controllers(&self) -> Vec<Box<dyn ViewController>> {
157        let actions = self
158            .0
159            .iter()
160            .flat_map(|entity| {
161                entity.state_machines.iter().flat_map(move |machine| {
162                    machine.transitions.iter().map(move |way| action_of(entity, machine, way))
163                })
164            })
165            .collect();
166        vec![Box::new(Transitions { actions })]
167    }
168}

170/// The records of an entity with a reminder that have a time set, for this person, as this space
171/// shows them. What is due by now is up to whoever asks, as they know the time.
172pub fn reminders_of(
173    space: &dyn crate::ObjectSpace,
174    entity: &EntityInfo,
175    environment: &ravenrs_criteria::Environment,
176) -> Vec<Object> {
177    let Some(reminder) = &entity.reminder else { return Vec::new() };
178    let found = space.query(&entity.name, &Query::all()).unwrap_or_default();
179    let criteria = reminder.criteria.as_deref().and_then(|text| Criteria::parse(text).ok());
180    found
181        .into_iter()
182        .filter(|record| matches!(record.get(&reminder.field), Value::Date(_)))
183        .filter(|record| {
184            criteria.as_ref().is_none_or(|criteria| Scope::new(environment).row(record).matches(criteria))
185        })
186        .collect()
187}
