1//! Signing in through an identity provider outside the application, linked to local users by the
2//! provider's issuer and subject, never by e-mail.
3//!
4//! The provider only says who someone is. What they can do always comes from the local user they
5//! are linked to: a claim never grants a role. Unlinking ends what that sign-in reaches, also in a
6//! session already open, on its next [`SecurityContext::reload`](crate::SecurityContext::reload).
7

13    Query, SECURITY_USER, SpaceError, UserDirectory, Value,
14};

16/// The entity of links between an outside identity and a local user.
17pub const SECURITY_EXTERNAL_LOGIN: &str = "SecurityExternalLogin";
18
19pub(crate) fn external_login_entity() -> EntityInfo {
20    EntityInfo::new(SECURITY_EXTERNAL_LOGIN)
21        .caption("External sign-ins")
22        .field(FieldInfo::new("id", FieldKind::Guid).key())
23        .field(FieldInfo::new("user_id", FieldKind::Reference(SECURITY_USER.to_owned())).caption("User"))
24        .field(FieldInfo::new("issuer", FieldKind::Text).read_only())
25        .field(FieldInfo::new("subject", FieldKind::Text).read_only())
26}

28/// Who a provider says someone is.
29#[derive(Clone, Debug, Default, PartialEq, Eq)]
30#[non_exhaustive]
31#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
32pub struct Claims {
33    /// Who vouches for it, such as `https://login.example.org`.
34    pub issuer: String,
35    /// The person's id at that issuer, which never changes.
36    pub subject: String,
37    /// Their e-mail, as the issuer says; never used to link.
38    pub email: Option<String>,
39    /// Their name, as the issuer says.
40    pub name: Option<String>,
41}
42
43#[cfg_attr(feature = "live", ravenrs_live::expose)]
44impl Claims {
45    /// Someone known to this issuer by this subject.
46    pub fn new(issuer: impl Into<String>, subject: impl Into<String>) -> Self {
47        Self { issuer: issuer.into(), subject: subject.into(), email: None, name: None }
48    }
49
50    /// With the e-mail the issuer gives.
51    pub fn email(mut self, email: impl Into<String>) -> Self {
52        self.email = Some(email.into());
53        self
54    }
55}

57/// A provider that checks what someone gave to sign in and says who they are.
58///
59/// When it cannot be reached it answers [`SpaceError::Temporary`] or [`SpaceError::Unavailable`];
60/// anything else it refuses is [`SpaceError::Denied`] or [`SpaceError::Refused`].
61pub trait IdentityProvider: Provider {
62    /// Who these credentials belong to.
63    fn verify(&self, credentials: &Credentials) -> Result<Claims, SpaceError>;
64}
65
66/// Signing in through an [`IdentityProvider`]: only issuers it accepts, only identities linked to
67/// an active local user, with that user's roles.
68#[derive(Clone)]
69#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
70pub struct ExternalSignIn {
71    directory: UserDirectory,
72    provider: Rc<dyn IdentityProvider>,
73    issuers: Vec<String>,
74}

85#[cfg_attr(feature = "live", ravenrs_live::expose)]
86impl ExternalSignIn {
87    /// Through this provider, over the users of this directory. It accepts no issuer until told.
88    pub fn new(directory: UserDirectory, provider: Rc<dyn IdentityProvider>) -> Self {
89        Self { directory, provider, issuers: Vec::new() }
90    }
91
92    /// Accepts identities from this issuer.
93    pub fn issuer(mut self, issuer: impl Into<String>) -> Self {
94        self.issuers.push(issuer.into());
95        self
96    }
97}

99impl Authenticator for ExternalSignIn {
100    fn authenticate(&self, credentials: &Credentials) -> Result<Identity, AuthError> {
101        let claims = self.provider.verify(credentials).map_err(|error| {
102            if error.is_temporary() || matches!(error, SpaceError::Unavailable(_)) {
103                AuthError::Unavailable(format!("{} cannot be reached now", self.provider.name()))
104            } else {
105                AuthError::Rejected
106            }
107        })?;
108        if !self.issuers.contains(&claims.issuer) {
109            return Err(AuthError::Rejected);
110        }
111        let failed = |error: SpaceError| AuthError::Storage(error.to_string());
112        let user = self.directory.linked_user(&claims.issuer, &claims.subject).map_err(failed)?;
113        let user = user.ok_or(AuthError::Rejected)?;
114        self.directory.identity_by_key(&user).map_err(failed)?.ok_or(AuthError::Rejected)
115    }
116}

118#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
119impl UserDirectory {
120    /// The local user an outside identity is linked to, if any.
121    pub fn linked_user(&self, issuer: &str, subject: &str) -> Result<Option<Value>, SpaceError> {
122        let space = self.raw();
123        Ok(find_link(space.as_ref(), issuer, subject)?.map(|link| link.get("user_id")))
124    }
125
126    /// Links an outside identity to a local user. Linking it again to the same user changes
127    /// nothing; one already linked to someone else is a conflict. Who may do this is up to the
128    /// caller.
129    pub fn link_external(&self, user: &Value, issuer: &str, subject: &str) -> Result<(), SpaceError> {
130        let mut space = self.raw();
131        if let Some(link) = find_link(space.as_ref(), issuer, subject)? {
132            return if link.get("user_id") == *user {
133                Ok(())
134            } else {
135                Err(SpaceError::Conflict("This outside identity is linked to another user.".to_owned()))
136            };
137        }
138        let mut link = space.new_object(SECURITY_EXTERNAL_LOGIN)?;
139        link.set("user_id", user.clone());
140        link.set("issuer", issuer);
141        link.set("subject", subject);
142        space.save(link)?;
143        space.commit()
144    }
145
146    /// Unlinks an outside identity: it no longer signs in, and a session it opened loses its
147    /// permissions on its next reload.
148    pub fn unlink_external(&self, issuer: &str, subject: &str) -> Result<(), SpaceError> {
149        let mut space = self.raw();
150        if let Some(link) = find_link(space.as_ref(), issuer, subject)? {
151            space.delete(SECURITY_EXTERNAL_LOGIN, &link.get("id"))?;
152            space.commit()?;
153        }
154        Ok(())
155    }
156}

158/// Compared here, so an issuer or a subject never becomes part of an expression.
159fn find_link(
160    space: &dyn ObjectSpace,
161    issuer: &str,
162    subject: &str,
163) -> Result<Option<crate::Object>, SpaceError> {
164    Ok(space.query(SECURITY_EXTERNAL_LOGIN, &Query::all())?.into_iter().find(|link| {
165        link.get("issuer") == Value::Text(issuer.to_owned())
166            && link.get("subject") == Value::Text(subject.to_owned())
167    }))
