1//! Values and records as JSON, the way the web API and its client agree on.
2
3use serde_json::{Map, Value as Json};
4
5use crate::{EntityInfo, FieldKind, Object, Value};
6
7/// A value as JSON. A date goes as text.
8pub fn value_to_json(value: &Value) -> Json {
9    match value {
10        Value::Bool(on) => Json::Bool(*on),
11        Value::Number(number) if number.fract() == 0.0 && number.abs() < 9e15 => Json::from(*number as i64),
12        Value::Number(number) => serde_json::Number::from_f64(*number).map_or(Json::Null, Json::Number),
13        Value::Text(text) => Json::String(text.clone()),
14        Value::Date(date) => Json::String(date.to_string()),
15        _ => Json::Null,
16    }
17}
18
19/// A record from JSON that came from the API, taking what its fields hold and leaving out the rest.
20pub fn object_from_json(entity: &EntityInfo, sent: &Map<String, Json>) -> Object {
21    let mut object = Object::from_values(&entity.name, std::collections::BTreeMap::new());
22    for field in &entity.fields {
23        let Some(value) = sent.get(&field.name) else { continue };
24        let value = match (value, &field.kind) {
25            (Json::Bool(on), _) => Value::Bool(*on),
26            (Json::Number(number), _) => number.as_f64().map_or(Value::Null, Value::Number),
27            (Json::String(text), FieldKind::Date | FieldKind::DateTime) => {
28                ravenrs_criteria::DateTime::parse(text).map_or_else(|| Value::Text(text.clone()), Value::Date)
29            }
30            (Json::String(text), _) => Value::Text(text.clone()),
31            _ => Value::Null,
32        };
33        object.set(&field.name, value);
34    }
35    object
36}
