1//! What a provider outside the application says it can do, and the smallest contract for sending
2//! events to one. Failures use the meanings of [`SpaceError`]: refused for good, not allowed,
3//! capability unavailable, temporary, or an outcome still unknown.
4
5use crate::SpaceError;
6
7/// The capability of answering [`EventSink::status`].
8pub const CAN_TELL_STATUS: &str = "events.status";
9
10/// Something outside the application that the application talks to: a messenger, a payment
11/// service, a printer, a store of files.
12pub trait Provider {
13    /// What it is called, for messages and logs.
14    fn name(&self) -> &str;
15
16    /// What it can do, by name. The interface shows what is here and never assumes more.
17    fn capabilities(&self) -> Vec<String> {
18        Vec::new()
19    }
20
21    /// Whether it says it can do this.
22    fn can(&self, capability: &str) -> bool {
23        self.capabilities().iter().any(|own| own == capability)
24    }
25}
26
27/// Where events go.
28///
29/// A destination that already has an event under a key keeps one effect. When the answer to a
30/// send is lost, the send returns [`SpaceError::Unknown`]; whoever sends asks [`status`](Self::status)
31/// before sending again, and only when the provider has [`CAN_TELL_STATUS`].
32pub trait EventSink: Provider {
33    /// Sends an event under this key.
34    fn send(&self, key: &str, payload: &str) -> Result<(), SpaceError>;
35
36    /// Whether the destination already has an event under this key. One without
37    /// [`CAN_TELL_STATUS`] answers [`SpaceError::Unavailable`].
38    fn status(&self, key: &str) -> Result<bool, SpaceError> {
39        let _ = key;
40        Err(SpaceError::Unavailable(format!("{} cannot tell what it received", self.name())))
41    }
42}

44#[cfg(test)]
45mod tests {
46    use super::*;
47
48    struct Mute;
49
50    impl Provider for Mute {
51        fn name(&self) -> &str {
52            "mute"
53        }
54    }
55
56    impl EventSink for Mute {
57        fn send(&self, _key: &str, _payload: &str) -> Result<(), SpaceError> {
58            Err(SpaceError::Unknown("no answer".to_owned()))
59        }
60    }
61
62    #[test]
63    fn a_provider_has_only_the_capabilities_it_declares() {
64        assert!(!Mute.can(CAN_TELL_STATUS));
65        assert!(matches!(Mute.status("k"), Err(SpaceError::Unavailable(_))));
66    }
67
68    #[test]
69    fn each_meaning_says_whether_to_try_again() {
70        let lost = Mute.send("k", "").expect_err("no answer");
71        assert!(lost.may_have_happened() && !lost.is_temporary());
72        assert!(SpaceError::Temporary("busy".to_owned()).is_temporary());
73        for settled in [
74            SpaceError::Unavailable("off".to_owned()),
75            SpaceError::Refused("no".to_owned()),
76            SpaceError::Denied("no".to_owned()),
77            SpaceError::Conflict("taken".to_owned()),
78        ] {
79            assert!(!settled.is_temporary() && !settled.may_have_happened(), "{settled:?}");
80            assert!(!settled.to_string().is_empty());
81        }
82    }
83}
