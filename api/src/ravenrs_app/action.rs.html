30/// Why something is on or off: pairs of reason and answer. It is on only if every answer is yes,
31/// so one part of the application turning it off cannot be undone by another.
32#[derive(Clone, Debug, Default, PartialEq, Eq)]
33#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
34pub struct Reasons(BTreeMap<String, bool>);
35
36#[cfg_attr(feature = "live", ravenrs_live::expose)]
37impl Reasons {
38    /// Whether every answer is yes.
39    pub fn all(&self) -> bool {
40        self.0.values().all(|on| *on)
41    }
42
43    /// Gives the answer for a reason.
44    pub fn set(&mut self, reason: &str, on: bool) {
45        self.0.insert(reason.to_owned(), on);
46    }
47
48    /// The answer for a reason, if it was given.
49    pub fn get(&self, reason: &str) -> Option<bool> {
50        self.0.get(reason).copied()
51    }
52
53    /// The reasons that say no.
54    pub fn against(&self) -> impl Iterator<Item = &str> {
55        self.0.iter().filter(|(_, on)| !**on).map(|(reason, _)| reason.as_str())
56    }
57}

134/// Which views an action or a controller is for. Anything left out does not narrow it.
135#[derive(Clone, Debug, Default, PartialEq, Eq)]
136#[non_exhaustive]
137#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
138pub struct TargetScope {
139    /// Only views of this entity.
140    pub entity: Option<String>,
141    /// Only the view with this id.
142    pub view_id: Option<String>,
143    /// Only views of this kind.
144    pub view_kind: Option<ViewKind>,
145    /// Root or nested.
146    pub nesting: Nesting,
147}
148
149#[cfg_attr(feature = "live", ravenrs_live::expose)]
150impl TargetScope {
151    /// Every view.
152    pub fn any() -> Self {
153        Self::default()
154    }
155
156    /// Only views of this entity.
157    pub fn entity(mut self, entity: impl Into<String>) -> Self {
158        self.entity = Some(entity.into());
159        self
160    }
161
162    /// Only the view with this id.
163    pub fn view_id(mut self, id: impl Into<String>) -> Self {
164        self.view_id = Some(id.into());
165        self
166    }
167
168    /// Only views of this kind.
169    pub fn view_kind(mut self, kind: ViewKind) -> Self {
170        self.view_kind = Some(kind);
171        self
172    }
173
174    /// Root or nested.
175    pub fn nesting(mut self, nesting: Nesting) -> Self {
176        self.nesting = nesting;
177        self
178    }
179
180    /// Whether it covers this view, in a frame of this context. No view is covered
181    /// only by what asks nothing of one.
182    pub fn covers(&self, view: Option<&ViewInfo>, context: FrameContext) -> bool {
183        let nesting = match self.nesting {
184            Nesting::Any => true,
185            Nesting::Root => context != FrameContext::Nested,
186            Nesting::Nested => context == FrameContext::Nested,
187        };
188        nesting
189            && match view {
190                Some(view) => {
191                    self.entity.as_ref().is_none_or(|e| view.entity.as_ref() == Some(e))
192                        && self.view_id.as_ref().is_none_or(|id| *id == view.id)
193                        && self.view_kind.is_none_or(|kind| kind == view.kind)
194                }
195                None => self.entity.is_none() && self.view_id.is_none() && self.view_kind.is_none(),
196            }
197    }
198}

411/// Why an action did not run.
412#[derive(Clone, Debug, PartialEq, Eq)]
413#[non_exhaustive]
414#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
415pub enum ActionError {
416    /// No action has this id.
417    Unknown(String),
418    /// It is out of place here, with the reasons.
419    Inactive(Vec<String>),
420    /// It is shown but off, with the reasons.
421    Disabled(Vec<String>),
422    /// Something asked first said no.
423    Canceled,
424    /// What people gave it is not what it asks for.
425    BadInput(String),
426    /// It ran and failed, in words for people.
427    Failed(String),
428}
429
430impl fmt::Display for ActionError {
431    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
432        match self {
433            Self::Unknown(id) => write!(f, "there is no action {id}"),
434            Self::Inactive(why) => write!(f, "the action is out of place: {}", why.join(", ")),
435            Self::Disabled(why) => write!(f, "the action is off: {}", why.join(", ")),
436            Self::Canceled => f.write_str("the action was canceled"),
437            Self::BadInput(why) | Self::Failed(why) => f.write_str(why),
438        }
439    }
440}

478/// Something people can do. A handle: the frame and its controller hold the same action, so a
479/// controller can turn it off by a reason of its own.
480///
481/// ```
482/// use ravenrs_app::{Action, ActionCategory, CriteriaMode, Selection};
483///
484/// let pay = Action::simple("Pay", "Pagar")
485///     .category(ActionCategory::Edit)
486///     .target_entity("Bill")
487///     .selection(Selection::RequireMultiple)
488///     .target_criteria("[Status] = 'Open'", CriteriaMode::TrueForAll)
489///     .on_execute(|cx| {
490///         for bill in &cx.selected {
491///             let mut bill = bill.clone();
492///             bill.set("Status", "Paid");
493///             cx.space.borrow_mut().save(bill).map_err(|e| e.to_string())?;
494///         }
495///         cx.space.borrow_mut().commit().map_err(|e| e.to_string())
496///     });
497/// assert_eq!(pay.caption(), "Pagar");
498/// ```
499#[derive(Clone)]
500#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
501pub struct Action(Rc<ActionInner>);

506    }

546    /// A button whose work runs away from the screen, on a thread of its own: it shows how far it
547    /// got, can be stopped, and what it hands back is kept on the screen's thread. While it runs,
548    /// the action is off, so it never runs twice at once.
549    ///
550    /// ```
551    /// use ravenrs_app::{Action, Selection};
552    ///
553    /// let archive = Action::background("Archive", "Arquivar", |job| {
554    ///     for record in job.selected() {
555    ///         let mut record = record.clone();
556    ///         record.set("Archived", true);
557    ///         job.save(record);
558    ///     }
559    ///     Ok(())
560    /// })
561    /// .selection(Selection::RequireMultiple);
562    /// assert!(archive.is_background());
563    /// ```

565        id: impl Into<String>,
566        caption: impl Into<String>,

568    ) -> Self {

570    }

592    fn change(mut self, f: impl FnOnce(&mut ActionInner)) -> Self {
593        // While it is being declared nothing else holds it, unless a copy was kept on the way, as
594        // the playground's interpreter does: then the change goes to an action of its own, which
595        // starts with the same reasons and does not share them.
596        if Rc::get_mut(&mut self.0).is_none() {
597            let mut copy = (*self.0).clone();
598            copy.active = State::new(self.0.active.get());
599            copy.enabled = State::new(self.0.enabled.get());
600            self.0 = Rc::new(copy);
601        }
602        if let Some(inner) = Rc::get_mut(&mut self.0) {
603            f(inner);
604        }
605        self
606    }

608    /// Runs this work away from the screen once it runs, after `on_execute`, with what people gave

611        self,
612        work: impl Fn(&crate::JobContext) -> Result<(), String> + Send + Sync + 'static,
613    ) -> Self {
614        self.change(|a| a.background = Some(std::sync::Arc::new(work)))
615    }

717    /// Runs as an update: sets fields of every record selected of this entity, as
718    /// `field = expression; ...`, each read from the record as it was; checks them against the
719    /// rules of `Save`, and keeps them all or none. What does not parse, or names a field the
720    /// entity has not, is found when the application is put together.
721    ///
722    /// ```
723    /// use ravenrs_app::{Action, EntityInfo, FieldInfo, FieldKind, Selection};
724    ///
725    /// let task = EntityInfo::new("Task")
726    ///     .field(FieldInfo::new("id", FieldKind::Integer).key())
727    ///     .field(FieldInfo::new("state", FieldKind::Text));
728    /// let finish = Action::simple("Finish", "Concluir")
729    ///     .target_entity("Task")
730    ///     .selection(Selection::RequireMultiple)
731    ///     .assign(&task, "state = 'done'");
732    /// assert_eq!(finish.caption(), "Concluir");
733    /// ```
734    pub fn assign(self, entity: &EntityInfo, set: &str) -> Self {
735        match crate::record_action::assignments(entity, set) {
736            Ok(found) => {
737                let entity = entity.clone();
738                self.on_execute(move |cx| crate::record_action::assign(&entity, &found, cx))
739            }
740            Err(problem) => self.change(|a| a.problem = Some(problem)),
741        }
742    }

866    /// Says what is wrong with how it was declared.
867    pub(crate) fn check(&self) -> Result<(), ModelError> {
868        if let Some(problem) = &self.0.problem {
869            return Err(ModelError(format!("the action {}: {problem}", self.0.id)));
870        }
871        if let Some((text, _)) = &self.0.criteria {
872            Criteria::parse(text).map_err(|e| ModelError(format!("the action {}: {e}", self.0.id)))?;
873        }
874        Ok(())

877    /// Answers the reasons that follow from the frame.
878    pub(crate) fn update(
879        &self,
880        view: Option<&ViewInfo>,
881        context: FrameContext,
882        selected: &[Object],
883        environment: &Environment,
884    ) {
885        self.set_active(REASON_SCOPE, self.0.scope.covers(view, context));
886        self.set_enabled(REASON_SELECTION, self.0.selection.allows(selected.len()));
887        self.set_enabled(REASON_CRITERIA, self.meets_criteria(selected, environment));
888    }

890    fn meets_criteria(&self, selected: &[Object], environment: &Environment) -> bool {
891        let Some((text, mode)) = &self.0.criteria else { return true };
892        if selected.is_empty() {
893            return true;
894        }
895        // `check` refuses what does not parse, when the application is put together.
896        let Ok(criteria) = Criteria::parse(text) else { return false };
897        let meets = |object: &Object| Scope::new(environment).row(object).matches(&criteria);
898        match mode {
899            CriteriaMode::TrueForAll => selected.iter().all(meets),
900            CriteriaMode::TrueAtLeastForOne => selected.iter().any(meets),
901        }
902    }

904    /// Says whether what people gave it is what it asks for.
905    pub(crate) fn check_input(&self, input: &ActionInput) -> Result<(), ActionError> {
906        match (&self.0.kind, input) {
907            (ActionKind::SingleChoice(items), ActionInput::Choice(id)) => {
908                // With no options declared, they come from the view it is on.
909                if items.is_empty() || items.iter().any(|item| item.has(id)) {
910                    Ok(())
911                } else {
912                    Err(ActionError::BadInput(format!("{} has no option {id}", self.0.id)))
913                }
914            }
915            (ActionKind::SingleChoice(_), _) => {
916                Err(ActionError::BadInput(format!("{} needs an option", self.0.id)))
917            }
918            (ActionKind::Parametrized(_), ActionInput::Parameter(_)) => Ok(()),
919            (ActionKind::Parametrized(_), _) => {
920                Err(ActionError::BadInput(format!("{} needs a value", self.0.id)))
921            }
922            _ => Ok(()),
923        }
924    }

947/// Puts a value in an address, where it cannot change what the address means.
948fn encoded(value: &Value) -> String {
949    let text = match value {
950        Value::Null => String::new(),
951        Value::Text(text) => text.clone(),
952        Value::Number(number) if number.fract() == 0.0 => format!("{}", *number as i64),
953        Value::Number(number) => number.to_string(),
954        Value::Bool(on) => on.to_string(),
955        other => format!("{other:?}"),
956    };
957    let mut out = String::with_capacity(text.len());
958    for byte in text.bytes() {
959        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'~') {
960            out.push(char::from(byte));
961        } else {
962            out.push_str(&format!("%{byte:02X}"));
963        }
964    }
965    out
966}

968/// The address of a [`ActionKind::Url`], with `{field}` taken from the record.
969pub(crate) fn fill_url(template: &str, object: Option<&Object>) -> String {
970    let mut out = String::with_capacity(template.len());
971    let mut rest = template;
972    while let Some(start) = rest.find('{') {
973        out.push_str(&rest[..start]);
974        let after = &rest[start + 1..];
975        let Some(end) = after.find('}') else {
976            out.push_str(&rest[start..]);
977            return out;
978        };
979        out.push_str(&encoded(&object.map_or(Value::Null, |o| o.get(&after[..end]))));
980        rest = &after[end + 1..];
981    }
982    out.push_str(rest);
983    out
984}
