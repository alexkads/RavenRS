8/// What a field holds, which decides how it is edited and stored.
9#[derive(Clone, Debug, PartialEq)]
10#[non_exhaustive]
11pub enum FieldKind {
12    /// A line of text.
13    Text,
14    /// Text of any length, edited in several lines.
15    LongText,
16    /// A whole number.
17    Integer,
18    /// A number, with these decimal places.
19    Decimal(u8),
20    /// Money, in whole cents, so it is always exact.
21    Money,
22    /// A date.
23    Date,
24    /// A date and a time.
25    DateTime,
26    /// Yes or no.
27    Boolean,
28    /// One of these options.
29    Enum(Vec<EnumOption>),
30    /// Another record, by the name of its entity. It holds that record's key.
31    Reference(String),
32    /// The records of another entity that point to this one.
33    Collection(String),
34    /// A color, as `#RRGGBB`.
35    Color,
36    /// A picture.
37    Image,
38    /// A file.
39    File,
40    /// An expression of the criteria language.
41    Criteria,
42    /// A key made up when a record is created, the same everywhere: no two machines repeat one.
43    Guid,
44    /// Several of these options at once, kept as their codes separated by commas.
45    Tags(Vec<EnumOption>),
46}
47
48impl FieldKind {
49    /// How expressions see it.
50    fn field_type(&self) -> FieldType {
51        match self {
52            Self::Text
53            | Self::LongText
54            | Self::Color
55            | Self::Criteria
56            | Self::Enum(_)
57            | Self::Guid
58            | Self::Tags(_) => FieldType::Text,
59            Self::Integer | Self::Decimal(_) | Self::Money => FieldType::Number,
60            Self::Date | Self::DateTime => FieldType::Date,
61            Self::Boolean => FieldType::Boolean,
62            Self::Reference(_) | Self::Collection(_) | Self::Image | Self::File => FieldType::Any,
63        }
64    }
65
66    /// Whether its editor takes the height it needs, instead of one line.
67    pub(crate) fn is_tall(&self) -> bool {
68        matches!(self, Self::LongText | Self::Image | Self::File | Self::Criteria)
69    }
70
71    /// Whether it lists other records instead of holding a value.
72    pub(crate) fn is_collection(&self) -> bool {
73        matches!(self, Self::Collection(_))
74    }
75
76    /// Its name in the model.
77    pub(crate) fn name(&self) -> &'static str {
78        match self {
79            Self::Text => "Text",
80            Self::LongText => "LongText",
81            Self::Integer => "Integer",
82            Self::Decimal(_) => "Decimal",
83            Self::Money => "Money",
84            Self::Date => "Date",
85            Self::DateTime => "DateTime",
86            Self::Boolean => "Boolean",
87            Self::Enum(_) => "Enum",
88            Self::Reference(_) => "Reference",
89            Self::Collection(_) => "Collection",
90            Self::Color => "Color",
91            Self::Image => "Image",
92            Self::File => "File",
93            Self::Criteria => "Criteria",
94            Self::Guid => "Guid",
95            Self::Tags(_) => "Tags",
96        }
97    }
98}

181/// `opening_cents` as `Opening cents`.
182fn caption_of(name: &str) -> String {
183    let spaced = name.replace('_', " ");
184    let mut chars = spaced.chars();
185    chars.next().map_or_else(String::new, |first| first.to_uppercase().chain(chars).collect())
186}

408    /// A field added in the model, under an entity's `Members`: `Kind` is that of a field with
409    /// plain values, with `Caption`, `Places` for decimals, `Options` for a list of options and
410    /// `Expression` for what is worked out and not stored.
411    fn from_model(entity: &str, node: &crate::ModelNode) -> Result<Self, ModelError> {
412        let wrong = |what: String| ModelError(format!("{entity}: the field {} {what}", node.id));
413        let options = || {
414            node.get("Options").map_or_else(Vec::new, |all| {
415                all.in_order()
416                    .iter()
417                    .map(|option| EnumOption::new(&option.id, option.text("Caption").unwrap_or(&option.id)))
418                    .collect()
419            })
420        };
421        let kind = match node.text("Kind").unwrap_or("Text") {
422            "Text" => FieldKind::Text,
423            "LongText" => FieldKind::LongText,
424            "Integer" => FieldKind::Integer,
425            "Decimal" => FieldKind::Decimal(node.integer("Places").unwrap_or(2).clamp(0, 12) as u8),
426            "Money" => FieldKind::Money,
427            "Date" => FieldKind::Date,
428            "DateTime" => FieldKind::DateTime,
429            "Boolean" => FieldKind::Boolean,
430            "Color" => FieldKind::Color,
431            "Enum" => FieldKind::Enum(options()),
432            "Tags" => FieldKind::Tags(options()),
433            other => return Err(wrong(format!("cannot be added in the model as {other}"))),
434        };
435        let mut field = Self::new(&node.id, kind);
436        if let Some(caption) = node.text("Caption") {
437            field.caption = caption.to_owned();
438        }
439        if let Some(expression) = node.text("Expression") {
440            field = field.calculated(expression);
441        }
442        Ok(field)
443    }
444
445    /// Whether the model added it: it comes with what it holds, which a declared one does not need.
446    pub(crate) fn is_custom(node: &crate::ModelNode, declared: &EntityInfo) -> bool {
447        declared.field_named(&node.id).is_none()
448    }

451/// How a list gets its records.
452#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
453#[non_exhaustive]
454pub enum DataAccess {
455    /// All of them at once, kept in memory, so grouping and every search work on the spot.
456    #[default]
457    Client,
458    /// As they are needed, a chunk at a time, for lists too long to read at once.
459    Server,
460    /// All of them at once, but only what the columns show, and with no editing on the list.
461    DataView,
462}
463
464impl DataAccess {
465    /// Its name in the model.
466    pub fn name(self) -> &'static str {
467        match self {
468            Self::Client => "Client",
469            Self::Server => "Server",
470            Self::DataView => "DataView",
471        }
472    }
473
474    /// From its name in the model; anything else is the usual.
475    pub fn from_name(name: &str) -> Self {
476        match name {
477            "Server" => Self::Server,
478            "DataView" => Self::DataView,
479            _ => Self::Client,
480        }
481    }
482}

805    /// Says what is wrong with how it was declared.
806    pub fn check(&self) -> Result<(), ModelError> {
807        let wrong = |what: String| Err(ModelError(format!("{}: {what}", self.name)));
808        if self.name.trim().is_empty() {
809            return Err(ModelError("an entity has no name".into()));
810        }
811        match self.fields.iter().filter(|f| f.is_key).count() {
812            1 => {}
813            0 => return wrong("it has no key field".into()),
814            n => return wrong(format!("it has {n} key fields, and only one can be")),
815        }
816        for (i, field) in self.fields.iter().enumerate() {
817            if self.fields[..i].iter().any(|f| f.name == field.name) {
818                return wrong(format!("the field {} is declared twice", field.name));
819            }
820            if let Some(expression) = &field.calculated {
821                let parsed = Criteria::parse(expression)
822                    .map_err(|e| ModelError(format!("{}.{}: {e}", self.name, field.name)))?;
823                if let Some(problem) = ravenrs_criteria::validate(&parsed, &self.schema()).first() {
824                    return wrong(format!("the field {}: {problem}", field.name));
825                }
826            }
827        }
828        for machine in &self.state_machines {
829            if self.field_named(&machine.field).is_none() {
830                return wrong(format!("a state machine is over {}, which is not a field", machine.field));
831            }
832            if let Some(problem) = machine.problem() {
833                return wrong(problem);
834            }
835        }
836        if let Some(reminder) = &self.reminder
837            && !self
838                .field_named(&reminder.field)
839                .is_some_and(|field| matches!(field.kind, FieldKind::Date | FieldKind::DateTime))
840        {
841            return wrong(format!("a reminder needs a date in {}", reminder.field));
842        }
843        let check_expression = |what: String, expression: &str| -> Result<(), ModelError> {
844            let parsed =
845                Criteria::parse(expression).map_err(|e| ModelError(format!("{}, {what}: {e}", self.name)))?;
846            // A function the application added is known only where it runs.
847            let environment = crate::environment();
848            let problems = ravenrs_criteria::validate(&parsed, &self.schema());
849            let found = problems.iter().find(|problem| {
850                !matches!(problem, ravenrs_criteria::Problem::UnknownFunction(name) if environment.has_function(name))
851            });
852            match found {
853                Some(problem) => Err(ModelError(format!("{}: {what}: {problem}", self.name))),
854                None => Ok(()),
855            }
856        };
857        for filter in &self.filters {
858            if let Some(expression) = &filter.criteria {
859                check_expression(format!("the filter {}", filter.id), expression)?;
860            }
861        }
862        if let Some(expression) = &self.list_criteria {
863            check_expression("the list's criteria".to_owned(), expression)?;
864        }
865        for variant in &self.variants {
866            if let Some(gone) = variant.columns.iter().find(|name| self.field_named(name).is_none()) {
867                return wrong(format!("the variant {} shows {gone}, which is not a field", variant.id));
868            }
869            if let Some(expression) = &variant.filter {
870                let parsed = Criteria::parse(expression)
871                    .map_err(|e| ModelError(format!("{}, variant {}: {e}", self.name, variant.id)))?;
872                if let Some(problem) = ravenrs_criteria::validate(&parsed, &self.schema()).first() {
873                    return wrong(format!("the variant {}: {problem}", variant.id));
874                }
875            }
876        }
877        for field in &self.fields {
878            if let Some(expression) = &field.lookup_filter {
879                Criteria::parse(expression)
880                    .map_err(|e| ModelError(format!("{}.{}: {e}", self.name, field.name)))?;
881            }
882        }
883        for rule in &self.rules {
884            rule.check(self)?;
885        }
886        for rule in &self.appearance {
887            rule.check(self)?;
888        }
889        Ok(())
890    }

892    /// A record of this entity with what a new one starts with.
893    pub fn new_object(&self) -> Object {
894        let values = self
895            .fields
896            .iter()
897            .filter(|f| f.calculated.is_none() && !f.kind.is_collection())
898            .map(|f| (f.name.clone(), f.default_value.clone()))
899            .collect();
900        let mut object = Object { entity: self.name.clone(), values };
901        for field in self.fields.iter().filter(|field| matches!(field.kind, FieldKind::Guid)) {
902            if matches!(object.get(&field.name), Value::Null) {
903                object.set(&field.name, uuid::Uuid::new_v4().to_string());
904            }
905        }
906        object
907    }

909    /// This entity as the model has it now: what people see and the rules it added.
910    pub(crate) fn as_in_model(&self, root: &crate::ModelNode) -> Result<Self, ModelError> {
911        let mut entity = self.clone();
912        let Some(node) = root.at(&["Entities", &self.name]) else { return Ok(entity) };
913        if let Some(members) = node.get("Members") {
914            for member in members.children.iter().filter(|member| FieldInfo::is_custom(member, self)) {
915                entity.fields.push(FieldInfo::from_model(&self.name, member)?);
916            }
917        }
918        if let Some(caption) = node.text("Caption") {
919            entity.caption = caption.to_owned();
920        }
921        for field in &mut entity.fields {
922            let Some(member) = node.at(&["Members", &field.name]) else { continue };
923            if let Some(caption) = member.text("Caption") {
924                field.caption = caption.to_owned();
925            }
926            if let FieldKind::Enum(options) = &mut field.kind {
927                for option in options {
928                    let found = member.at(&["Options", &option.code]).and_then(|n| n.text("Caption"));
929                    if let Some(caption) = found {
930                        option.caption = caption.to_owned();
931                    }
932                }
933            }
934        }
935        if let Some(rules) = node.get("Rules") {
936            for rule in rules.in_order() {
937                entity.rules.push(crate::Rule::from_model(rule)?);
938            }
939        }
940        entity.check()?;
941        Ok(entity)
942    }

944    /// Fills in the calculated fields, in the order declared, so one can read another.
945    pub fn compute(&self, object: &mut Object, environment: &Environment) {
946        for field in &self.fields {
947            let Some(expression) = &field.calculated else { continue };
948            // `check` already refused what does not parse.
949            let value = Criteria::parse(expression)
950                .ok()
951                .and_then(|parsed| Scope::new(environment).row(&*object).evaluate(&parsed).ok())
952                .unwrap_or(Value::Null);
953            object.values.insert(field.name.clone(), value);
954        }
955    }

957    /// The object as it is stored, with no calculated values.
958    pub fn strip_calculated(&self, object: &mut Object) {
959        for field in self.fields.iter().filter(|f| f.calculated.is_some()) {
960            object.values.remove(&field.name);
961        }
962    }

965/// A record: its entity's name and a value per field. Dynamic, so filters, rules and
966/// permissions read every entity the same way.
967#[derive(Clone, Debug, PartialEq)]
968pub struct Object {
969    entity: String,
970    values: BTreeMap<String, Value>,
971}
972
973impl Object {
974    /// A record of this entity with these values, as storage hands them over.
975    pub fn from_values(entity: impl Into<String>, values: BTreeMap<String, Value>) -> Self {
976        Self { entity: entity.into(), values }
977    }
978
979    /// The name of its entity.
980    pub fn entity(&self) -> &str {
981        &self.entity
982    }
983
984    /// The value of a field; nothing is `Null`.
985    pub fn get(&self, field: &str) -> Value {
986        self.values.get(field).cloned().unwrap_or(Value::Null)
987    }
988
989    /// Changes the value of a field.
990    pub fn set(&mut self, field: &str, value: impl Into<Value>) {
991        self.values.insert(field.to_owned(), value.into());
992    }
993
994    /// Every value, by field.
995    pub fn values(&self) -> &BTreeMap<String, Value> {
996        &self.values
997    }
998
999    /// Leaves only these fields.
1000    pub fn keep_only(&mut self, fields: &[&str]) {
1001        self.values.retain(|name, _| fields.contains(&name.as_str()));
1002    }
1003}
1004
1005impl Record for Object {
1006    fn field(&self, path: &[String]) -> Option<Value> {
1007        match path {
1008            [name] => self.values.get(name).cloned(),
1009            _ => None,
1010        }
1011    }
1012
1013    fn collection(&self, _path: &[String]) -> Option<Vec<&Self>> {
1014        None
1015    }
1016}
