59/// Refuses what the modules say they cannot work with, naming the module.
60fn check_manifests(modules: &[Box<dyn Module>], host: &Host) -> Result<(), ModelError> {
61    let mut migrations: Vec<(String, String)> = Vec::new();
62    let mut jobs: Vec<(String, String)> = Vec::new();
63    for module in modules {
64        let name = module.name();
65        let manifest = module.manifest();
66        for (other, range) in &manifest.requires {
67            let found = modules
68                .iter()
69                .find(|m| m.name() == other)
70                .ok_or_else(|| ModelError(format!("the module {name} needs {other}, which was not added")))?;
71            match found.manifest().version {
72                Some(version) if range.accepts(version) => {}
73                Some(version) => {
74                    return Err(ModelError(format!(
75                        "the module {name} needs {other} {range}, and {other} is {version}"
76                    )));
77                }
78                None => {
79                    return Err(ModelError(format!(
80                        "the module {name} needs {other} {range}, and {other} says no version"
81                    )));
82                }
83            }
84        }
85        if let Some(target) = host.target
86            && !manifest.targets.is_empty()
87            && !manifest.targets.contains(&target)
88        {
89            return Err(ModelError(format!("the module {name} does not run on {target:?}")));
90        }
91        if let Some(database) = &host.database
92            && !manifest.databases.is_empty()
93            && !manifest.databases.contains(database)
94        {
95            return Err(ModelError(format!("the module {name} does not work on {database}")));
96        }
97        if let Some(missing) = manifest.capabilities.iter().find(|c| !host.capabilities.contains(c)) {
98            return Err(ModelError(format!(
99                "the module {name} needs the capability {missing}, which the host does not offer"
100            )));
101        }
102        for (kind, ids, seen) in
103            [("migration", &manifest.migrations, &mut migrations), ("job", &manifest.jobs, &mut jobs)]
104        {
105            for id in ids {
106                if let Some((_, first)) = seen.iter().find(|(other, _)| other == id) {
107                    return Err(ModelError(format!("the {kind} {id} comes from both {first} and {name}")));
108                }
109                seen.push((id.clone(), name.to_owned()));
110            }
111        }
112    }
113    Ok(())
114}

124/// Puts modules in an order where each comes after what it depends on, keeping the order they
125/// were added in as far as possible.
126fn in_dependency_order(modules: Vec<Box<dyn Module>>) -> Result<Vec<Box<dyn Module>>, ModelError> {
127    let names: Vec<String> = modules.iter().map(|m| m.name().to_owned()).collect();
128    for (i, name) in names.iter().enumerate() {
129        if names[..i].contains(name) {
130            return Err(ModelError(format!("the module {name} was added twice")));
131        }
132    }
133    let mut waiting: Vec<Option<Box<dyn Module>>> = modules.into_iter().map(Some).collect();
134    let mut placed: Vec<String> = Vec::new();
135    let mut ordered = Vec::with_capacity(waiting.len());
136    while ordered.len() < names.len() {
137        let mut progressed = false;
138        for slot in &mut waiting {
139            let Some(module) = slot.as_ref() else { continue };
140            let mut needs = module.dependencies();
141            needs.extend(
142                module
143                    .manifest()
144                    .requires
145                    .into_iter()
146                    .map(|(name, _)| name)
147                    .filter(|n| !module.dependencies().contains(n)),
148            );
149            if let Some(missing) = needs.iter().find(|d| !names.contains(d)) {
150                return Err(ModelError(format!(
151                    "the module {} depends on {missing}, which was not added",
152                    module.name()
153                )));
154            }
155            if needs.iter().all(|d| placed.contains(d)) {
156                placed.push(module.name().to_owned());
157                if let Some(module) = slot.take() {
158                    ordered.push(module);
159                }
160                progressed = true;
161                break;
162            }
163        }
164        if !progressed {
165            let stuck: Vec<&str> = waiting.iter().flatten().map(|m| m.name()).collect();
166            return Err(ModelError(format!("these modules depend on each other: {}", stuck.join(", "))));
167        }
168    }
169    Ok(ordered)
170}

172/// An application put together: its entities, its model and what makes frames. Layers from
173/// the administrator and the user go on with [`layers_mut`](Self::layers_mut).
174#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
175pub struct Application {
176    title: String,
177    modules: Vec<Box<dyn Module>>,
178    /// As they were declared.
179    declared: Vec<EntityInfo>,
180    /// As the model has them now.
181    entities: Vec<EntityInfo>,
182    services: Vec<Rc<dyn Any>>,
183    translations: Vec<(String, Diff)>,
184    layers: ModelLayers,
185    model: Rc<ModelNode>,
186    unusable: Vec<crate::UnusableNode>,
187    security: Option<Rc<crate::SecurityContext>>,
188    switched_off: Vec<String>,
189}
190

193        f.debug_struct("Application").field("title", &self.title).finish_non_exhaustive()
194    }
195}

298    /// Keeps a setting typed as text in the person's own layer, and applies it. It stays what it
299    /// was: a number has to be typed as a number, and yes or no as `true` or `false`.
300    pub fn set_typed_setting(&mut self, path: &[&str], name: &str, text: &str) -> Result<(), ModelError> {
301        let current = self.model.at(path).and_then(|node| node.properties.get(name));
302        let value = match current {
303            Some(serde_json::Value::String(_)) | None => serde_json::Value::String(text.to_owned()),
304            Some(kept) => match serde_json::from_str::<serde_json::Value>(text) {
305                Ok(value) if std::mem::discriminant(&value) == std::mem::discriminant(kept) => value,
306                _ => return Err(ModelError(format!("{name} cannot be {text}"))),
307            },
308        };
309        self.layers.set(&ModelLayer::User, path, name, value);
310        self.refresh_model()
311    }

325    /// Applies the layers again. Frames made after this see the result; views already showing
326    /// have to be shown again.
327    pub fn refresh_model(&mut self) -> Result<(), ModelError> {
328        let MergedModel { mut root, unusable } = self.layers.merged();
329        let problems = check_model(&root);
330        if !problems.is_empty() {
331            return Err(ModelError(problems.join("; ")));
332        }
333        crate::model::add_custom_fields(&mut root, &self.declared);
334        crate::model::follow_captions(&mut root, &self.declared);
335        self.entities =
336            self.declared.iter().map(|entity| entity.as_in_model(&root)).collect::<Result<_, _>>()?;
337        self.model = Rc::new(root);
338        self.unusable = unusable;
339        Ok(())
340    }

347    /// A frame with new controllers from every module, in order, over this unit of work.
348    pub fn new_frame(&self, context: FrameContext, space: Rc<RefCell<dyn ObjectSpace>>) -> Frame {
349        let view = self.modules.iter().flat_map(|m| m.view_controllers()).collect();
350        let window = self.modules.iter().flat_map(|m| m.window_controllers()).collect();
351        let frame = Frame::new(context, space, Some(self.model.clone()), view, window);
352        for service in &self.services {
353            frame.keep_extension(service.clone());
354        }
355        frame.set_entities(Rc::new(self.entities.clone()));
356        if let Some(security) = &self.security {
357            frame.set_security(security.clone());
358        }
359        frame
360    }

448    /// Puts it together: orders the modules, generates the model, lets the modules change it,
449    /// applies their layers and checks the result. What is wrong comes out here, and not when
450    /// someone reaches that screen.
451    pub fn build(self) -> Result<Application, ModelError> {
452        if let Some(unknown) =
453            self.off.iter().find(|name| !self.modules.iter().any(|m| m.name() == name.as_str()))
454        {
455            return Err(ModelError(format!("the module {unknown} cannot be switched off: it was not added")));
456        }
457        let modules: Vec<Box<dyn Module>> = self
458            .modules
459            .into_iter()
460            .map(|module| {
461                if self.off.iter().any(|name| name == module.name()) {
462                    Box::new(SwitchedOff(module)) as Box<dyn Module>
463                } else {
464                    module
465                }
466            })
467            .collect();
468        let mut modules = in_dependency_order(modules)?;
469        check_manifests(&modules, &self.host)?;
470        let switched_off: Vec<String> = modules
471            .iter()
472            .filter(|m| self.off.iter().any(|name| name == m.name()))
473            .flat_map(|m| m.entities())
474            .map(|entity| entity.name)
475            .collect();
476        let mut entities = self.entities;
477        for module in &modules {
478            entities.extend(module.entities());
479        }
480        let with_states: Vec<EntityInfo> =
481            entities.iter().filter(|entity| !entity.state_machines.is_empty()).cloned().collect();
482        if !with_states.is_empty() {
483            modules.push(Box::new(crate::workflow::StateMachineModule(with_states)));
484        }
485        let with_actions: Vec<EntityInfo> =
486            entities.iter().filter(|entity| !entity.actions.is_empty()).cloned().collect();
487        if !with_actions.is_empty() {
488            modules.push(Box::new(crate::record_action::RecordActionModule(with_actions)));
489        }
490        let mut zero = generate_model(&entities)?;
491        for dashboard in &self.dashboards {
492            crate::model::add_dashboard(&mut zero, dashboard);
493        }
494        for module in &modules {
495            module.extend_model(&mut zero);
496        }
497        for module in &modules {
498            module.update_generated(&mut zero);
499        }
500        let mut seen: Vec<String> = Vec::new();
501        for module in &modules {
502            let mut actions: Vec<crate::Action> =
503                module.view_controllers().iter().flat_map(|c| c.actions()).collect();
504            actions.extend(module.window_controllers().iter().flat_map(|c| c.actions()));
505            for action in actions {
506                action.check()?;
507                if seen.iter().any(|id| id == action.id()) {
508                    return Err(ModelError(format!("two actions have the id {}", action.id())));
509                }
510                seen.push(action.id().to_owned());
511            }
512        }
513        let mut layers = ModelLayers::new(zero);
514        for module in &modules {
515            if let Some(text) = module.model_difference() {
516                let difference = Diff::parse(text)
517                    .map_err(|e| ModelError(format!("the module {}: {e}", module.name())))?;
518                layers.put(&ModelLayer::Module(module.name().to_owned()), difference);
519            }
520        }
521        // Over the other modules, whatever comes in, and under the person's own settings.
522        layers.put(&ModelLayer::Module(LANGUAGE_LAYER.to_owned()), Diff::new());
523        let translations = self
524            .translations
525            .into_iter()
526            .map(|(language, text)| {
527                Diff::parse(&text)
528                    .map(|difference| (language.clone(), difference))
529                    .map_err(|e| ModelError(format!("the translation {language}: {e}")))
530            })
531            .collect::<Result<_, _>>()?;
532        let mut application = Application {
533            title: self.title,
534            modules,
535            declared: entities.clone(),
536            entities,
537            services: self.services,
538            translations,
539            layers,
540            model: Rc::new(ModelNode::default()),
541            unusable: Vec::new(),
542            security: None,
543            switched_off,
544        };
545        application.refresh_model()?;
546        Ok(application)
547    }
