1//! LFM3 read from a GGUF file. candle stops at LFM2, and LFM3 changes the layers: some
2//! attention looks only at a sliding window, attention heads have a gate, and each layer's
3//! outputs pass through one more norm.

12/// How much a short convolution reads: the token and the six before it.
13const CONV_WIDTH: usize = 7;

15/// The cosines and sines of a kind of attention, at the positions being read.

28struct Feed {
29    gate: QMatMul,
30    up: QMatMul,
31    down: QMatMul,
32}
33
34impl Feed {
35    fn forward(&self, xs: &Tensor) -> Result<Tensor> {
36        self.down.forward(&(self.gate.forward(xs)?.silu()? * self.up.forward(xs)?)?)
37    }
38}

40struct Attention {
41    query: QMatMul,
42    key: QMatMul,
43    value: QMatMul,
44    out: QMatMul,
45    gate: QMatMul,
46    query_norm: RmsNorm,
47    key_norm: RmsNorm,
48    heads: usize,
49    key_heads: usize,
50    head_size: usize,
51    /// The keys and values to keep, when this one looks only at a sliding window.
52    keep: Option<usize>,
53    /// Whether it reads token by token, keeping what it read.
54    remembers: bool,
55    cache: Option<(Tensor, Tensor)>,
56}
57
58impl Attention {
59    fn forward(&mut self, xs: &Tensor, reach: Reach<'_>) -> Result<Tensor> {
60        let (batch, tokens, size) = xs.dims3()?;
61        let split = |tensor: Tensor, heads: usize| -> Result<Tensor> {
62            tensor.reshape((batch, tokens, heads, self.head_size))?.transpose(1, 2)?.contiguous()
63        };
64        let query = self.query_norm.forward(&split(self.query.forward(xs)?, self.heads)?)?;
65        let key = self.key_norm.forward(&split(self.key.forward(xs)?, self.key_heads)?)?;
66        let value = split(self.value.forward(xs)?, self.key_heads)?;
67        let query = rope(&query, &reach.rotation.cos, &reach.rotation.sin)?;
68        let key = rope(&key, &reach.rotation.cos, &reach.rotation.sin)?;
69
70        let (key, value) = match &self.cache {
71            Some((old_key, old_value)) => {
72                (Tensor::cat(&[old_key, &key], 2)?, Tensor::cat(&[old_value, &value], 2)?)
73            }
74            None => (key, value),
75        };
76        let seen = key.dim(2)?;
77        if self.remembers {
78            self.cache = Some(match self.keep {
79                Some(keep) if seen > keep => {
80                    (key.narrow(2, seen - keep, keep)?, value.narrow(2, seen - keep, keep)?)
81                }
82                _ => (key.clone(), value.clone()),
83            });
84        }
85
86        let groups = self.heads / self.key_heads;
87        let key = candle_transformers::utils::repeat_kv(key, groups)?;
88        let value = candle_transformers::utils::repeat_kv(value, groups)?.contiguous()?;
89        let scores = (query.matmul(&key.t()?)? / (self.head_size as f64).sqrt())?;
90        let scores = match reach.mask {
91            Some(mask) => scores.broadcast_add(mask)?,
92            None => scores,
93        };
94        let read = softmax_last_dim(&scores)?.matmul(&value)?.transpose(1, 2)?;
95        let gate = sigmoid(&self.gate.forward(xs)?)?.unsqueeze(3)?;
96        self.out.forward(&read.broadcast_mul(&gate)?.reshape((batch, tokens, size))?)
97    }
98}

100struct ShortConv {
101    into: QMatMul,
102    out: QMatMul,
103    /// One tensor for each place in the width, oldest token first.
104    weights: Vec<Tensor>,
105    cache: Option<Tensor>,
106}
107
108impl ShortConv {
109    fn forward(&mut self, xs: &Tensor) -> Result<Tensor> {
110        let (batch, tokens, size) = xs.dims3()?;
111        let parts = self.into.forward(xs)?.transpose(1, 2)?;
112        let mix = parts.narrow(1, size, size)?;
113        let read = (parts.narrow(1, 0, size)? * parts.narrow(1, 2 * size, size)?)?;
114        let before = match &self.cache {
115            Some(cache) => cache.clone(),
116            None => Tensor::zeros((batch, size, CONV_WIDTH - 1), read.dtype(), read.device())?,
117        };
118        let line = Tensor::cat(&[&before, &read], 2)?;
119        self.cache = Some(line.narrow(2, tokens, CONV_WIDTH - 1)?);
120        let mut sum = Tensor::zeros((batch, size, tokens), read.dtype(), read.device())?;
121        for (place, weight) in self.weights.iter().enumerate() {
122            sum = (sum + line.narrow(2, place, tokens)?.broadcast_mul(weight)?)?;
123        }
124        self.out.forward(&(mix * sum)?.transpose(1, 2)?.contiguous()?)
125    }
126}

191impl Stack {
192    /// Reads the layers called `{prefix}.{index}.{part}.weight`, of each `kinds`.
193    pub(crate) fn from_gguf<R: std::io::Seek + std::io::Read>(
194        content: &gguf_file::Content,
195        reader: &mut R,
196        device: &Device,
197        prefix: &str,
198        names: &LayerNames,
199        sizes: Heads,
200        kinds: &[LayerKind],
201    ) -> Result<Self> {
202        let eps = sizes.eps;
203        let mut tensor = |name: &str| -> Result<QTensor> { content.tensor(reader, name, device) };
204        let mut layers = Vec::with_capacity(kinds.len());
205        for (layer, kind) in kinds.iter().enumerate() {
206            let mut matrix = |name: &str| -> Result<QMatMul> {
207                QMatMul::from_qtensor(tensor(&format!("{prefix}.{layer}.{name}.weight"))?)
208            };
209            let feed = Feed { gate: matrix("ffn_gate")?, up: matrix("ffn_up")?, down: matrix("ffn_down")? };
210            let operator = if matches!(kind, LayerKind::Conv) {
211                let into = matrix(names.conv_in)?;
212                let out = matrix(names.conv_out)?;
213                let weights =
214                    tensor(&format!("{prefix}.{layer}.{}.weight", names.conv))?.dequantize(device)?;
215                let (_, width) = weights.dims2()?;
216                if width != CONV_WIDTH {
217                    bail!("unexpected convolution of {:?}", weights.dims());
218                }
219                let weights = (0..CONV_WIDTH)
220                    .map(|place| weights.narrow(1, place, 1)?.contiguous())
221                    .collect::<Result<Vec<_>>>()?;
222                Operator::Conv(ShortConv { into, out, weights, cache: None })
223            } else {
224                let query = matrix("attn_q")?;
225                let key = matrix("attn_k")?;
226                let value = matrix("attn_v")?;
227                let out = matrix(names.attention_out)?;
228                let gate = matrix("attn_gate")?;
229                let in_window = matches!(kind, LayerKind::Window);
230                let attention = Attention {
231                    query,
232                    key,
233                    value,
234                    out,
235                    gate,
236                    query_norm: RmsNorm::from_qtensor(
237                        tensor(&format!("{prefix}.{layer}.attn_q_norm.weight"))?,
238                        eps,
239                    )?,
240                    key_norm: RmsNorm::from_qtensor(
241                        tensor(&format!("{prefix}.{layer}.attn_k_norm.weight"))?,
242                        eps,
243                    )?,
244                    heads: sizes.heads,
245                    key_heads: sizes.key_heads,
246                    head_size: sizes.head_size,
247                    keep: in_window.then_some(sizes.keep),
248                    remembers: sizes.remembers,
249                    cache: None,
250                };
251                if in_window { Operator::Window(attention) } else { Operator::Full(attention) }
252            };
253            let mut norm = |name: &str| -> Result<RmsNorm> {
254                RmsNorm::from_qtensor(tensor(&format!("{prefix}.{layer}.{name}.weight"))?, eps)
255            };
256            layers.push(Layer {
257                operator,
258                feed,
259                operator_norm: norm(names.operator_norm)?,
260                after_operator_norm: names.after_operator_norm.map(&mut norm).transpose()?,
261                feed_norm: norm(names.feed_norm)?,
262                after_feed_norm: names.after_feed_norm.map(&mut norm).transpose()?,
263            });
264        }
265        Ok(Self { layers })
266    }
267
268    /// Passes `hidden` through the layers.
269    pub(crate) fn forward(
270        &mut self,
271        mut hidden: Tensor,
272        full: Reach<'_>,
273        in_window: Reach<'_>,
274    ) -> Result<Tensor> {
275        for layer in &mut self.layers {
276            let normed = layer.operator_norm.forward(&hidden)?;
277            let done = match &mut layer.operator {
278                Operator::Conv(conv) => conv.forward(&normed)?,
279                Operator::Full(attention) => attention.forward(&normed, full)?,
280                Operator::Window(attention) => attention.forward(&normed, in_window)?,
281            };
282            let done = match &layer.after_operator_norm {
283                Some(norm) => norm.forward(&done)?,
284                None => done,
285            };
286            hidden = (done + hidden)?;
287            let fed = layer.feed.forward(&layer.feed_norm.forward(&hidden)?)?;
288            let fed = match &layer.after_feed_norm {
289                Some(norm) => norm.forward(&fed)?,
290                None => fed,
291            };
292            hidden = (fed + hidden)?;
293        }
294        Ok(hidden)
295    }
296
297    /// Forgets what it has read.
298    pub(crate) fn clear(&mut self) {
299        for layer in &mut self.layers {
300            match &mut layer.operator {
301                Operator::Conv(conv) => conv.cache = None,
302                Operator::Full(attention) | Operator::Window(attention) => attention.cache = None,
303            }
304        }
305    }
306}

308/// An LFM3 model with its weights, reading one sequence at a time.
309pub(crate) struct ModelWeights {
310    embeddings: Embedding,
311    stack: Stack,
312    norm: RmsNorm,
313    output: QMatMul,
314    full_frequencies: Vec<f32>,
315    window_frequencies: Vec<f32>,
316    window: usize,
317    device: Device,
318}
319
320impl ModelWeights {
321    /// Reads the weights in `content` from `reader`.
322    pub(crate) fn from_gguf<R: std::io::Seek + std::io::Read>(
323        content: &gguf_file::Content,
324        reader: &mut R,
325        device: &Device,
326    ) -> Result<Self> {
327        let value = |name: &str| match content.metadata.get(name) {
328            Some(value) => Ok(value),
329            None => bail!("no {name} in the model"),
330        };
331        let number = |name: &str| value(name)?.to_u32().map(|number| number as usize);
332        let eps = f64::from(value("lfm3.attention.layer_norm_rms_epsilon")?.to_f32()?);
333        let heads = number("lfm3.attention.head_count")?;
334        let head_size = number("lfm3.attention.key_length")?;
335        let window = number("lfm3.attention.sliding_window")?;
336        let full_base = value("lfm3.rope.freq_base")?.to_f32()?;
337        let window_base = value("lfm3.rope.freq_base_swa")?.to_f32()?;
338        let key_heads = value("lfm3.attention.head_count_kv")?
339            .to_vec()?
340            .iter()
341            .map(|heads| heads.to_i32().map(|heads| heads.unsigned_abs() as usize))
342            .collect::<Result<Vec<_>>>()?;
343        let windowed = value("lfm3.attention.sliding_window_pattern")?
344            .to_vec()?
345            .iter()
346            .map(gguf_file::Value::to_bool)
347            .collect::<Result<Vec<_>>>()?;
348
349        let mut tensor = |name: &str| -> Result<QTensor> { content.tensor(reader, name, device) };
350        let token_weights = tensor("token_embd.weight")?;
351        let (_, size) = token_weights.shape().dims2()?;
352        let embeddings = Embedding::new(token_weights.dequantize(device)?, size);
353        let output = QMatMul::from_qtensor(token_weights)?;
354        let norm = RmsNorm::from_qtensor(tensor("output_norm.weight")?, eps)?;
355        let frequencies = |base: f32| -> Vec<f32> {
356            (0..head_size).step_by(2).map(|place| 1.0 / base.powf(place as f32 / head_size as f32)).collect()
357        };
358        // The file divides the frequencies by these: so much that the second half does not rotate.
359        let full_frequencies = frequencies(full_base)
360            .into_iter()
361            .zip(tensor("rope_freqs.weight")?.dequantize(device)?.to_vec1::<f32>()?)
362            .map(|(frequency, divisor)| frequency / divisor)
363            .collect();
364        let window_frequencies = frequencies(window_base);
365
366        let kinds: Vec<LayerKind> = key_heads
367            .iter()
368            .enumerate()
369            .map(|(layer, key_heads)| match (key_heads, windowed.get(layer)) {
370                (0, _) => LayerKind::Conv,
371                (_, Some(true)) => LayerKind::Window,
372                _ => LayerKind::Full,
373            })
374            .collect();
375        let key_heads = key_heads.iter().copied().find(|heads| *heads > 0).unwrap_or(heads);
376        let sizes = Heads { heads, key_heads, head_size, keep: window - 1, remembers: true, eps };
377        let stack = Stack::from_gguf(content, reader, device, "blk", &MODEL_NAMES, sizes, &kinds)?;
378        Ok(Self {
379            embeddings,
380            stack,
381            norm,
382            output,
383            full_frequencies,
384            window_frequencies,
385            window,
386            device: device.clone(),
387        })
388    }
389
390    /// The logits for the token after `input`, which starts at position `offset`.
391    pub(crate) fn forward(&mut self, input: &Tensor, offset: usize) -> Result<Tensor> {
392        let last = self.last_hidden(&self.embed(input)?, offset)?;
393        self.output.forward(&last)
394    }
395
396    /// What the tokens of `input` read as.
397    pub(crate) fn embed(&self, input: &Tensor) -> Result<Tensor> {
398        self.embeddings.forward(input)
399    }
400
401    /// What the model holds after reading `embeds`, which starts at position `offset`.
402    pub(crate) fn last_hidden(&mut self, embeds: &Tensor, offset: usize) -> Result<Tensor> {
403        let (_, tokens, _) = embeds.dims3()?;
404        let full = rotation(&self.full_frequencies, offset, tokens, &self.device)?;
405        let in_window = rotation(&self.window_frequencies, offset, tokens, &self.device)?;
406        // Only the prompt comes in more than one token at a time.
407        let (full_mask, window_mask) = if tokens > 1 {
408            (Some(mask(tokens, tokens, &self.device)?), Some(mask(tokens, self.window, &self.device)?))
409        } else {
410            (None, None)
411        };
412        let full = Reach { rotation: &full, mask: full_mask.as_ref() };
413        let in_window = Reach { rotation: &in_window, mask: window_mask.as_ref() };
414        let hidden = self.stack.forward(embeds.clone(), full, in_window)?;
415        self.norm.forward(&hidden)?.narrow(1, tokens - 1, 1)?.squeeze(1)
416    }
417
418    /// Forgets what it has read.
419    pub(crate) fn clear_kv_cache(&mut self) {
420        self.stack.clear();
421    }

424pub(crate) fn rotation(
425    frequencies: &[f32],
426    offset: usize,
427    tokens: usize,
428    device: &Device,
429) -> Result<Rotation> {
430    let angles: Vec<f32> = (offset..offset + tokens)
431        .flat_map(|position| frequencies.iter().map(move |frequency| position as f32 * frequency))
432        .collect();
433    let angles = Tensor::from_vec(angles, (tokens, frequencies.len()), device)?;
434    Ok(Rotation { cos: angles.cos()?, sin: angles.sin()? })
435}

437/// What to add to the scores of `tokens` tokens, so that each reads only itself and the
438/// ones before it, `reach` tokens at most.
439pub(crate) fn mask(tokens: usize, reach: usize, device: &Device) -> Result<Tensor> {
440    let values: Vec<f32> = (0..tokens)
441        .flat_map(|row| {
442            (0..tokens).map(
443                move |column| {
444                    if column <= row && row - column < reach { 0.0 } else { f32::NEG_INFINITY }
445                },
446            )
447        })
448        .collect();
449    Tensor::from_vec(values, (tokens, tokens), device)
450}

467    fn the_convolution_reads_the_same_token_by_token_or_all_at_once() {
468        let device = Device::Cpu;
469        let identity = Tensor::eye(2, candle_core::DType::F32, &device).expect("eye");
470        let into = Tensor::cat(&[identity.clone(), identity.clone(), identity.clone()], 0).expect("into");
471        let weights: Vec<Tensor> = (0..CONV_WIDTH)
472            .map(|place| Tensor::new(&[[place as f32 + 1.0], [1.0]], &device).expect("weight"))
473            .collect();
474        let conv = || ShortConv {
475            into: QMatMul::Tensor(into.clone()),
476            out: QMatMul::Tensor(identity.clone()),
477            weights: weights.clone(),
478            cache: None,
479        };
480        let line: Vec<f32> = (0..18).map(|value| value as f32 / 10.0).collect();
481        let input = Tensor::from_vec(line, (1, 9, 2), &device).expect("input");
482        let together = conv().forward(&input).expect("together").to_vec3::<f32>().expect("values");
483        let mut parts = conv();
