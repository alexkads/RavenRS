1//! Sound to spectrum and back, as the voices write it: windows of a periodic Hann,
2//! overlapped by a hop.
3
4use rustfft::FftPlanner;
5use rustfft::num_complex::Complex32;
6
7/// A periodic Hann window of `size` samples.
8pub(crate) fn hann(size: usize) -> Vec<f32> {
9    (0..size)
10        .map(|place| {
11            let turn = std::f32::consts::TAU * place as f32 / size as f32;
12            0.5 - 0.5 * turn.cos()
13        })
14        .collect()
15}
16
17/// The magnitude and angle of each bin of each window, the sound padded by reflection
18/// at both ends as `torch.stft` does when centred.
19pub(crate) fn stft(sound: &[f32], size: usize, hop: usize, window: &[f32]) -> (Vec<Vec<f32>>, Vec<Vec<f32>>) {
20    let half = size / 2;
21    let last = sound.len() as isize - 1;
22    let sample = |place: isize| -> f32 {
23        let mut place = place;
24        if place < 0 {
25            place = -place;
26        }
27        if place > last {
28            place = 2 * last - place;
29        }
30        sound[place.clamp(0, last) as usize]
31    };
32    let count = sound.len() / hop + 1;
33    let fft = FftPlanner::new().plan_fft_forward(size);
34    let bins = size / 2 + 1;
35    let (mut magnitudes, mut angles) = (Vec::with_capacity(count), Vec::with_capacity(count));
36    let mut buffer = vec![Complex32::new(0.0, 0.0); size];
37    for frame in 0..count {
38        let start = (frame * hop) as isize - half as isize;
39        for (place, value) in buffer.iter_mut().enumerate() {
40            *value = Complex32::new(sample(start + place as isize) * window[place], 0.0);
41        }
42        fft.process(&mut buffer);
43        magnitudes.push(buffer[..bins].iter().map(|value| value.norm()).collect());
44        angles.push(buffer[..bins].iter().map(|value| value.im.atan2(value.re)).collect());
45    }
46    (magnitudes, angles)
47}

49/// Sound from the magnitude and angle of each bin of each window, overlapped and added
50/// with the window and divided by the windows' sum: `(frames - 1) * hop + size` samples,
51/// which the caller trims as its spectrum was padded.
52pub(crate) fn inverse_stft(
53    magnitudes: &[Vec<f32>],
54    angles: &[Vec<f32>],
55    size: usize,
56    hop: usize,
57    window: &[f32],
58) -> Vec<f32> {
59    if magnitudes.is_empty() {
60        return Vec::new();
61    }
62    let bins = size / 2 + 1;
63    let fft = FftPlanner::new().plan_fft_inverse(size);
64    let length = (magnitudes.len() - 1) * hop + size;
65    let (mut sound, mut envelope) = (vec![0.0_f32; length], vec![0.0_f32; length]);
66    let mut buffer = vec![Complex32::new(0.0, 0.0); size];
67    for (index, (magnitude, angle)) in magnitudes.iter().zip(angles).enumerate() {
68        for bin in 0..bins {
69            let value = Complex32::from_polar(magnitude[bin], angle[bin]);
70            buffer[bin] = value;
71            if bin > 0 && bin < size - bin {
72                buffer[size - bin] = value.conj();
73            }
74        }
75        fft.process(&mut buffer);
76        let start = index * hop;
77        for place in 0..size {
78            sound[start + place] += buffer[place].re / size as f32 * window[place];
79            envelope[start + place] += window[place] * window[place];
80        }
81    }
82    sound
83        .iter()
84        .zip(&envelope)
85        .map(|(sound, envelope)| if *envelope > 1e-11 { sound / envelope } else { 0.0 })
86        .collect()
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    #[test]
94    fn sound_comes_back_from_its_spectrum() {
95        let sound: Vec<f32> = (0..400).map(|place| (place as f32 * 0.07).sin() * 0.5).collect();
96        let window = hann(20);
97        let (magnitudes, angles) = stft(&sound, 20, 5, &window);
98        assert_eq!(magnitudes.len(), 400 / 5 + 1);
99        let back = inverse_stft(&magnitudes, &angles, 20, 5, &window);
100        let back = &back[10..10 + 400];
101        let worst = sound.iter().zip(back).map(|(a, b)| (a - b).abs()).fold(0.0_f32, f32::max);
102        assert!(worst < 1e-4, "{worst}");
103    }
