290/// Endings whose stressed `e` or `o` is open in most words that have them.
291fn opens_by_ending(word: &str) -> bool {
292    const ENDINGS: &[&str] = &[
293        "ola", "olas", "oja", "ojas", "orte", "ortes", "orta", "ortas", "ece", "ecem", "ere", "erem", "erve",
294        "ervem", "ega", "egam", "egue", "eguem", "ol", "el", "éis", "óis", "ova", "ovas", "oca", "ocas",
295        "eja",
296    ];
297    // Those that stay closed though they end the same.
298    const CLOSED: &[&str] = &[
299        "pela", "pelas", "aquela", "aquelas", "esta", "veja", "deseja", "seja", "sejam", "deseje", "planeja",
300    ];
301    !CLOSED.contains(&word)
302        && ENDINGS.iter().any(|ending| word.ends_with(ending) && word.len() > ending.len())
303}

311fn is_vowel(letter: char) -> bool {
312    matches!(base(letter), 'a' | 'e' | 'i' | 'o' | 'u')
313}

315/// The letter without its accent.
316fn base(letter: char) -> char {
317    match letter {
318        'á' | 'à' | 'â' | 'ã' => 'a',
319        'é' | 'ê' => 'e',
320        'í' => 'i',
321        'ó' | 'ô' | 'õ' => 'o',
322        'ú' | 'ü' => 'u',
323        'y' => 'i',
324        other => other,
325    }
326}

328fn marks_stress(letter: char) -> bool {
329    matches!(letter, 'á' | 'à' | 'â' | 'é' | 'ê' | 'í' | 'ó' | 'ô' | 'ú')
330}

332/// Where each syllable's vowels are in the word: start and end, the end not included.
333fn nuclei(letters: &[char]) -> Vec<(usize, usize)> {
334    let mut found = Vec::new();
335    let mut place = 0;
336    while place < letters.len() {
337        let letter = letters[place];
338        // After q and g the u only says w, or nothing.
339        let glide_u = base(letter) == 'u'
340            && place > 0
341            && matches!(letters[place - 1], 'q' | 'g')
342            && place + 1 < letters.len()
343            && is_vowel(letters[place + 1]);
344        if !is_vowel(letter) || glide_u {
345            place += 1;
346            continue;
347        }
348        let start = place;
349        let mut end = place + 1;
350        // A falling diphthong: a vowel, then an unstressed i or u in the same syllable.
351        if end < letters.len() {
352            let next = letters[end];
353            let first = base(letter);
354            let pairs = matches!(
355                (first, next),
356                ('a', 'i' | 'u')
357                    | ('e', 'i' | 'u')
358                    | ('o', 'i' | 'u')
359                    | ('u', 'i')
360                    | ('i', 'u')
361                    | ('ã', 'o' | 'e')
362            ) || matches!((letter, next), ('ã', 'o' | 'e') | ('õ', 'e') | ('é' | 'ó', 'i' | 'u'));
363            let closes = |at: usize| {
364                // "sair", "ruim", "juiz": a consonant that ends the syllable makes a hiatus.
365                at + 1 < letters.len()
366                    && matches!(letters[at + 1], 'r' | 'l' | 'z' | 'm' | 'n')
367                    && (at + 2 == letters.len() || !is_vowel(letters[at + 2]))
368                    && !(letters[at + 1] == 'n' && letters.get(at + 2) == Some(&'h'))
369            };
370            let nh = end + 2 < letters.len() + 1
371                && letters.get(end + 1) == Some(&'n')
372                && letters.get(end + 2) == Some(&'h');
373            let before_glide = next == 'i' && letters.get(end + 1) == Some(&'u');
374            if pairs && !marks_stress(next) && !closes(end) && !nh && !before_glide {
375                end += 1;
376            }
377        }
378        found.push((start, end));
379        place = end;
380    }
381    found
382}

384/// Which syllable takes the stress, by the accent or else by the ending.
385fn stressed(letters: &[char], nuclei: &[(usize, usize)]) -> usize {
386    if nuclei.is_empty() {
387        return 0;
388    }
389    if let Some(index) =
390        nuclei.iter().position(|(start, end)| letters[*start..*end].iter().any(|l| marks_stress(*l)))
391    {
392        return index;
393    }
394    if let Some(index) =
395        nuclei.iter().position(|(start, end)| letters[*start..*end].iter().any(|l| matches!(l, 'ã' | 'õ')))
396    {
397        return index;
398    }
399    let last = nuclei.len() - 1;
400    let word: String = letters.iter().collect();
401    let mut stem = word.as_str();
402    if let Some(cut) = stem.strip_suffix('s') {
403        stem = cut;
404    }
405    let penultimate = stem.ends_with('a')
406        || stem.ends_with('e')
407        || stem.ends_with('o')
408        || stem.ends_with("am")
409        || stem.ends_with("em")
410        || stem.ends_with("en");
411    // An ending diphthong with i or u keeps the stress: "vai", "chapéu", "aconteceu".
412    let (start, end) = nuclei[last];
413    let ends_in_glide = end - start == 2 && end == letters.len() - usize::from(word.ends_with('s'));
414    if penultimate && last > 0 && !ends_in_glide { last - 1 } else { last }
415}

417/// What follows a vowel, for its nasal sound.
418fn nasal_coda(letters: &[char], after: usize) -> Option<bool> {
419    let letter = *letters.get(after)?;
420    if !matches!(letter, 'm' | 'n') {
421        return None;
422    }
423    let next = letters.get(after + 1).copied();
424    match next {
425        None => Some(false),
426        Some('h') if letter == 'n' => None,
427        Some(next) if is_vowel(next) => None,
428        Some(next) => Some(matches!(next, 'p' | 'b')),
429    }
430}

432/// A word by the rules.
433fn word_phonemes(word: &str) -> String {
434    let letters: Vec<char> = word.chars().collect();
435    let nuclei = nuclei(&letters);
436    let stress = stressed(&letters, &nuclei);
437    let open = OPEN.contains(&word) || opens_by_ending(word);
438    let count = nuclei.len();
439    let mut syllable_of = vec![usize::MAX; letters.len()];
440    for (index, (start, end)) in nuclei.iter().enumerate() {
441        for slot in &mut syllable_of[*start..*end] {
442            *slot = index;
443        }
444    }
445    // Secondary stress on the first syllable, and every other one, of a long word.
446    let secondary = |index: usize| {
447        count >= 3 && index.is_multiple_of(2) && index < stress && (index == 0 || index + 2 <= stress)
448    };
449    let mut out = String::new();
450    let mut place = 0;
451    let at = |place: usize| letters.get(place).copied();
452    while place < letters.len() {
453        let letter = letters[place];
454        if let Some(index) = syllable_of.get(place).copied().filter(|index| *index != usize::MAX) {
455            let (start, end) = nuclei[index];
456            if place != start {
457                place += 1;
458                continue;
459            }
460            if index == stress {
461                out.push('ˈ');
462            } else if secondary(index) {
463                out.push('ˌ');
464            }
465            let vowels: Vec<char> = letters[start..end].to_vec();
466            // After the stress, an i before a vowel glides into it: "cópia", "relatório".
467            // So does an e: "vídeo", "área".
468            if index > stress
469                && (vowels == ['i'] || vowels == ['e'])
470                && letters.get(end).is_some_and(|n| is_vowel(*n))
471            {
472                out.push('j');
473                place = end;
474                continue;
475            }
476            let coda = nasal_coda(&letters, end);
477            let is_final = index == count - 1;
478            let final_of_word = end == letters.len() || (end + 1 == letters.len() && letters[end] == 's');
479            out.push_str(&vowel(
480                &vowels,
481                index == stress,
482                index > stress,
483                is_final && final_of_word,
484                coda,
485                open,
486                &letters,
487                end,
488            ));
489            if coda.is_some() {
490                place = end + 1;
491                // "-am" at the end of a verb is a nasal diphthong.
492                continue;
493            }
494            place = end;
495            continue;
496        }
497        let next = at(place + 1);
498        let next_is_front = next.is_some_and(|n| matches!(base(n), 'e' | 'i'));
499        let previous_vowel = place > 0 && is_vowel(letters[place - 1]);
500        let next_vowel = next.is_some_and(is_vowel);
501        match letter {
502            'b' | 'f' | 'k' | 'p' | 'v' | 'w' => out.push(letter),
503            'c' if next == Some('h') => {
504                out.push('ʃ');
505                place += 1;
506            }
507            'c' | 'ç' if letter == 'ç' || next_is_front => out.push('s'),
508            'c' => out.push('k'),
509            'd' => out.push('d'),
510            't' => out.push('t'),
511            'g' if next == Some('u') && at(place + 2).is_some_and(|n| matches!(base(n), 'e' | 'i')) => {
512                out.push('ɡ');
513                place += 1;
514            }
515            'g' if next == Some('u') && at(place + 2).is_some_and(is_vowel) => {
516                out.push_str("ɡw");
517                place += 1;
518            }
519            'g' if next_is_front => out.push('ʒ'),
520            'g' => out.push('ɡ'),
521            'q' if next == Some('u') && at(place + 2).is_some_and(|n| matches!(base(n), 'e' | 'i')) => {
522                out.push('k');
523                place += 1;
524            }
525            'q' if next == Some('u') => {
526                out.push_str("kw");
527                place += 1;
528            }
529            'q' => out.push('k'),
530            'h' => {}
531            'j' => out.push('ʒ'),
532            'l' if next == Some('h') => {
533                out.push_str("lj");
534                place += 1;
535            }
536            'l' if next_vowel => out.push('l'),
537            'l' => {
538                // At the end of a syllable it becomes a vowel.
539                match out.chars().last() {
540                    Some('ɔ') => out.push('l'),
541                    Some('u') if letters[place - 1] == 'ú' => out.push('l'),
542                    Some('a') => {
543                        out.pop();
544                        out.push('W');
545                    }
546                    Some('o') => out.push('w'),
547                    Some('u') => out.push('w'),
548                    _ => out.push('ʊ'),
549                }
550            }
551            'm' | 'n' if next == Some('h') && letter == 'n' => {
552                out.push('ɲ');
553                place += 1;
554            }
555            'm' | 'n' => out.push(letter),
556            'r' => {
557                let doubled = next == Some('r');
558                let initial = place == 0;
559                let after_coda = place > 0 && matches!(letters[place - 1], 'n' | 'l' | 's');
560                let in_cluster = place > 0 && !is_vowel(letters[place - 1]) && !after_coda;
561                if doubled || initial || after_coda {
562                    out.push('x');
563                    if doubled {
564                        place += 1;
565                    }
566                } else if in_cluster {
567                    out.push('r');
568                } else if next_vowel {
569                    out.push('ɾ');
570                } else if next.is_none() {
571                    out.push('r');
572                } else {
573                    out.push_str("ɾə");
574                }
575            }
576            's' if next == Some('s') => {
577                out.push('s');
578                place += 1;
579            }
580            's' if next == Some('c') && at(place + 2).is_some_and(|n| matches!(base(n), 'e' | 'i')) => {
581                out.push('s');
582                place += 1;
583            }
584            's' if previous_vowel && next_vowel => out.push('z'),
585            's' if next
586                .is_some_and(|n| matches!(n, 'b' | 'd' | 'g' | 'l' | 'm' | 'n' | 'r' | 'v' | 'z' | 'j')) =>
587            {
588                out.push('z');
589            }
590            's' => out.push('s'),
591            'x' => {
592                if X_AS_KS.contains(&word) {
593                    out.push_str("ks");
594                } else if X_AS_S.contains(&word) {
595                    out.push('s');
596                } else if place == 1 && letters[0] == 'e' {
597                    if next_vowel {
598                        out.push('z');
599                    } else if next == Some('c') && at(place + 2).is_some_and(|n| matches!(base(n), 'e' | 'i'))
600                    {
601                        out.push('s');
602                        place += 1;
603                    } else {
604                        out.push('s');
605                    }
606                } else if next.is_none() {
607                    out.push_str("ks");
608                } else {
609                    out.push('ʃ');
610                }
611            }
612            'z' if next.is_none() => out.push('s'),
613            'z' => out.push('z'),
614            other if other.is_alphabetic() => out.push(other),
615            _ => {}
616        }
617        place += 1;
618    }
619    palatalize(&out)
620}

622/// The sound of a syllable's vowels.
623#[allow(clippy::too_many_arguments)]
624fn vowel(
625    vowels: &[char],
626    stressed: bool,
627    after_stress: bool,
628    last: bool,
629    coda: Option<bool>,
630    open: bool,
631    letters: &[char],
632    end: usize,
633) -> String {
634    let before_nasal = matches!(letters.get(end), Some('m' | 'n'))
635        && (letters.get(end + 1).is_some_and(|n| is_vowel(*n)) || letters.get(end + 1) == Some(&'h'));
636    if vowels.len() == 2 {
637        let pair: String = vowels.iter().collect();
638        let sound = match pair.as_str() {
639            "ai" | "ái" => "I",
640            "au" | "áu" => "W",
641            "ei" | "éi" => "A",
642            "eu" => "eʊ",
643            "éu" => "ɛʊ",
644            "ou" => "ow",
645            "oi" | "ôi" => "oɪ",
646            "ói" => "ɔɪ",
647            "ui" => "uɪ",
648            "iu" => "iʊ",
649            "ão" => "ɐ̃ʊ̃",
650            "ãe" => "ɐ̃ɪ̃",
651            "õe" => "õɪ̃",
652            _ => "",
653        };
654        if !sound.is_empty() {
655            return sound.to_owned();
656        }
657    }
658    let mut out = String::new();
659    for (index, letter) in vowels.iter().enumerate() {
660        let with_coda = index == vowels.len() - 1 && coda.is_some();
661        let final_vowel = last && index == vowels.len() - 1 && coda.is_none();
662        let sound = if with_coda {
663            let lips = coda == Some(true);
664            let at_end = end + 1 == letters.len();
665            match base(*letter) {
666                'a' if at_end && letters[end] == 'm' => "ɐ̃ʊ̃".to_owned(),
667                'a' => if lips { "ɐ̃m" } else { "ɐ̃ŋ" }.to_owned(),
668                'e' => if lips { "Am" } else { "Aŋ" }.to_owned(),
669                'i' => if lips { "im" } else { "iŋ" }.to_owned(),
670                'o' => if lips { "om" } else { "oŋ" }.to_owned(),
671                _ => if lips { "ũm" } else { "ũŋ" }.to_owned(),
672            }
673        } else {
674            match *letter {
675                'á' | 'à' => "a".to_owned(),
676                'â' => "ɐ".to_owned(),
677                'ã' => "ɐ̃".to_owned(),
678                'é' => "ɛ".to_owned(),
679                'ê' => "e".to_owned(),
680                'í' | 'y' => "i".to_owned(),
681                'ó' => "ɔ".to_owned(),
682                'ô' => "o".to_owned(),
683                'õ' => "õ".to_owned(),
684                'ú' if before_nasal => "ũ".to_owned(),
685                'ú' => "u".to_owned(),
686                'a' if before_nasal && (stressed || letters.get(end + 1) == Some(&'h')) => "ɐ̃".to_owned(),
687                'a' if !stressed && (final_vowel || before_nasal) => "æ".to_owned(),
688                'a' => "a".to_owned(),
689                'e' if stressed && open => "ɛ".to_owned(),
690                'e' if final_vowel && !stressed => "y".to_owned(),
691                'e' => "e".to_owned(),
692                'o' if stressed && open => "ɔ".to_owned(),
693                'o' if !stressed && (final_vowel || after_stress) => "ʊ".to_owned(),
694                'o' => "o".to_owned(),
695                'i' | 'u' => letter.to_string(),
696                other => other.to_string(),
697            }
698        };
699        out.push_str(&sound);
700    }
701    out
702}

704/// `t` and `d` before an `i` sound become `ʧ` and `ʤ`.
705fn palatalize(phonemes: &str) -> String {
706    let chars: Vec<char> = phonemes.chars().collect();
707    let mut out = String::with_capacity(phonemes.len());
708    for (place, sound) in chars.iter().enumerate() {
709        let next = chars[place + 1..].iter().find(|c| !matches!(c, 'ˈ' | 'ˌ')).copied();
710        let front = next.is_some_and(|n| matches!(n, 'i' | 'y' | 'j' | 'ɪ'));
711        match sound {
712            't' if front => out.push('ʧ'),
713            'd' if front => out.push('ʤ'),
714            other => out.push(*other),
715        }
716    }
717    out
718}

720/// Letters with their accents as one character, however the text wrote them.
721fn composed(word: &str) -> String {
722    let mut out = String::with_capacity(word.len());
723    for letter in word.chars() {
724        let joined = match (out.chars().last(), letter) {
725            (Some(base), '\u{301}') => "aeiou".find(base).map(|at| ['á', 'é', 'í', 'ó', 'ú'][at]),
726            (Some('a'), '\u{300}') => Some('à'),
727            (Some(base), '\u{302}') => "aeo".find(base).map(|at| ['â', 'ê', 'ô'][at]),
728            (Some(base), '\u{303}') => "ao".find(base).map(|at| ['ã', 'õ'][at]),
729            (Some('c'), '\u{327}') => Some('ç'),
730            (Some('u'), '\u{308}') => Some('ü'),
731            _ => None,
732        };
733        match joined {
734            Some(joined) => {
735                out.pop();
736                out.push(joined);
737            }
738            None => out.push(letter),
739        }
740    }
741    out
742}

744/// A word as said alone: the list first, then the rules.
745fn unicode_mark(letter: char) -> bool {
746    matches!(letter, '\u{300}'..='\u{36f}')

749/// A word, or two joined by a hyphen: a pronoun after it belongs to the verb,
750/// "encontrá-la", anything else is a word of its own, "segunda-feira".
751fn hyphenated(written: &str) -> String {
752    const CLITICS: &[&str] =
753        &["la", "las", "lo", "los", "lhe", "lhes", "me", "te", "se", "nos", "vos", "o", "a", "os", "as"];
754    let parts: Vec<&str> = written.split('-').filter(|part| !part.is_empty()).collect();
755    let mut words: Vec<String> = Vec::new();
756    for part in parts {
757        let lower = part.to_lowercase();
758        match words.last_mut() {
759            Some(last) if CLITICS.contains(&lower.as_str()) => last.push_str(part),
760            _ => words.push(part.to_owned()),
761        }
762    }
763    words.iter().map(|part| word(part)).collect::<Vec<_>>().join(" ")
764}

766fn word(word: &str) -> String {
767    let lower = composed(&word.to_lowercase().replace('-', ""));
768    SPOKEN
769        .iter()
770        .find(|(written, _)| *written == lower)
771        .map_or_else(|| word_phonemes(&lower), |(_, said)| (*said).to_owned())
772}

774fn starts_with_vowel(phonemes: &str) -> bool {
775    phonemes.chars().find(|c| !matches!(c, 'ˈ' | 'ˌ')).is_some_and(|c| {
776        matches!(
777            c,
778            'a' | 'e' | 'i' | 'o' | 'u' | 'ɛ' | 'ɔ' | 'ɐ' | 'æ' | 'ʊ' | 'A' | 'I' | 'W' | 'O' | 'y' | 'ũ'
779        )
780    })
781}
782
783fn starts_voiced(phonemes: &str) -> bool {
784    starts_with_vowel(phonemes)
785        || phonemes.chars().find(|c| !matches!(c, 'ˈ' | 'ˌ')).is_some_and(|c| {
786            matches!(
787                c,
788                'b' | 'd' | 'ɡ' | 'l' | 'm' | 'n' | 'ɲ' | 'v' | 'z' | 'ʒ' | 'ʤ' | 'x' | 'r' | 'ɾ' | 'w' | 'j'
789            )
790        })
791}

793/// `text` in Brazilian Portuguese as phonemes for Kokoro. Numbers are left out: write
794/// them in words before.
795pub fn phonemes_pt_br(text: &str) -> String {
796    // Words and what lies between them, which keeps the punctuation the model knows.
797    let mut pieces: Vec<(bool, String)> = Vec::new();
798    let mut current = String::new();
799    for letter in text.chars() {
800        if letter.is_alphabetic() || unicode_mark(letter) || (letter == '-' && !current.is_empty()) {
801            current.push(letter);
802            continue;
803        }
804        if !current.is_empty() {
805            pieces.push((true, hyphenated(&current)));
806            current.clear();
807        }
808        let kept = match letter {
809            '.' | ',' | '!' | '?' | ':' | ';' | '…' | '—' | '(' | ')' | '"' | '“' | '”' => {
810                Some(letter)
811            }
812            ' ' | '\n' | '\t' => Some(' '),
813            _ => None,
814        };
815        if let Some(kept) = kept {
816            pieces.push((false, kept.to_string()));
817        }
818    }
819    if !current.is_empty() {
820        pieces.push((true, hyphenated(&current)));
821    }
822    // How a word ends depends on how the next begins, when nothing pauses between them.
823    for index in 0..pieces.len() {
824        if !pieces[index].0 {
825            continue;
826        }
827        let next = pieces[index + 1..]
828            .iter()
829            .take_while(|(is_word, text)| *is_word || text == " ")
830            .find(|(is_word, _)| *is_word)
831            .map(|(_, text)| text.clone());
832        let Some(next) = next else { continue };
833        let said = &mut pieces[index].1;
834        let reducible = |said: &str, last: char| {
835            // Only a vowel that ends the word after a consonant: not "saiu" or "painel".
836            let mut back = said.chars().rev();
837            back.next() == Some(last) && back.next().is_some_and(|c| !"aeiouɛɔɐæAIWOˈˌ\u{303}".contains(c))
838        };
839        if starts_with_vowel(&next) {
840            if reducible(said, 'y') && !matches!(said.as_str(), "ky" | "ˌely") {
841                said.pop();
842                let before = said.chars().last();
843                if matches!(before, Some('r' | 'ɾ' | 'l')) {
844                    said.push('i');
845                } else if next.starts_with("ˈo") || next.starts_with("ˈɔ") {
846                    said.push('y');
847                } else {
848                    said.push('j');
849                }
850            } else if reducible(said, 'ʊ') && (said.contains('ˈ') || said.contains('ˌ')) && said != "kˌomʊ"
851            {
852                said.pop();
853                said.push('w');
854            } else if said.ends_with('r') {
855                said.pop();
856                said.push('ɾ');
857            }
858        }
859        if said.ends_with('s') && starts_voiced(&next) {
860            said.pop();
861            said.push('z');
862        }
863    }
864    // A small word that ends the sentence is said in full.
865    for index in 0..pieces.len() {
866        let ends = pieces.get(index + 1).is_none_or(|(is_word, text)| !is_word && text != " ");
867        let only_secondary = !pieces[index].1.contains('ˈ');
868        if pieces[index].0 && ends && only_secondary && pieces[index].1.contains('ˌ') {
869            pieces[index].1 = pieces[index].1.replacen('ˌ', "ˈ", 1);
870        }
871        if pieces[index].0
872            && ends
873            && only_secondary
874            && let Some(rest) = pieces[index].1.strip_prefix("dˌ")
875        {
876            pieces[index].1 = format!("dˈ{rest}");
877        }
878    }
879    let mut out = String::new();
880    for (_, piece) in pieces {
881        if piece == " " && out.ends_with(' ') {
882            continue;
883        }
884        out.push_str(&piece);
885    }
886    out.trim().to_owned()
887}

967    fn levenshtein(a: &[char], b: &[char]) -> usize {
968        let mut row: Vec<usize> = (0..=b.len()).collect();
969        for (i, x) in a.iter().enumerate() {
970            let mut previous = row[0];
971            row[0] = i + 1;
972            for (j, y) in b.iter().enumerate() {
973                let keep = row[j + 1];
974                row[j + 1] = (previous + usize::from(x != y)).min(row[j] + 1).min(row[j + 1] + 1);
975                previous = keep;
976            }
977        }
978        row[b.len()]
979    }
