17/// How many times a download tells the UI it moved: one per cent is plenty.
18const STEPS: u64 = 100;
19const GIGABYTE: f32 = 1_000_000_000.0;
20/// The file in the folder that says which model is in use.
21const IN_USE_FILE: &str = "in-use";
22/// What the speed test asks, and how long an answer it wants.
23const TEST_SYSTEM: &str = "Answer in one short sentence.";
24const TEST_USER: &str = "Why keep a copy of important files?";
25const TEST_TOKENS: usize = 40;
26/// Tokens per second from which an answer feels instant, and from which it can be
27/// read as it arrives.
28const FAST: f64 = 15.0;
29const USABLE: f64 = 5.0;

107    fn details(&self, model: &LocalModel) -> String {
108        let gigabytes = |bytes: u64| {
109            let size = format!("{:.1} GB", bytes as f32 / GIGABYTE);
110            if self.decimal_comma { size.replace('.', ",") } else { size }
111        };
112        let understood =
113            model.understood.map_or_else(|| self.not_measured.clone(), |percent| format!("{percent}%"));
114        format!(
115            "{} · {}: {} · {}: {understood} · {}: {}",
116            gigabytes(model.bytes()),
117            self.memory,
118            gigabytes(model.memory),
119            self.understood,
120            self.license,
121            model.license
122        )
123    }
124
125    fn result(&self, outcome: Option<&Tested>) -> Option<String> {
126        let tested = outcome?;
127        let verdict = if tested.tokens_per_second >= FAST {
128            &self.fast
129        } else if tested.tokens_per_second >= USABLE {
130            &self.usable
131        } else {
132            &self.slow
133        };
134        let seconds = format!("{:.1} s", tested.seconds);
135        let seconds = if self.decimal_comma { seconds.replace('.', ",") } else { seconds };
136        Some(format!("{} {seconds}. {verdict}", self.answered_in))
137    }
138
139    fn error(&self, error: &ModelError) -> Option<String> {
140        match error {
141            ModelError::Cancelled => None,
142            ModelError::Network(_) => Some(self.network_error.clone()),
143            ModelError::WrongFile => Some(self.wrong_file.clone()),
144            _ => Some(self.disk_error.clone()),
145        }
146    }

149/// A download in progress, shared with its thread.
150#[derive(Debug, Default)]
151struct Download {
152    received: AtomicU64,
153    cancelled: AtomicBool,
154    finished: Mutex<Option<Result<(), ModelError>>>,
155}
156
157/// How a speed test went.
158#[derive(Clone, Copy, Debug, PartialEq)]
159struct Tested {
160    seconds: f64,
161    tokens_per_second: f64,
162}
163
164/// A speed test in progress: what its thread leaves when it ends.
165type TestRun = Arc<Mutex<Option<Option<Tested>>>>;
166
167/// One model on offer.
168#[derive(Debug)]
169struct Entry {
170    model: LocalModel,
171    installed: bool,
172    download: Option<Arc<Download>>,
173    test: Option<TestRun>,
174    tested: Option<Tested>,
175    did_not_run: bool,
176    error: Option<ModelError>,
177}

179/// The models of a list that fit this machine, to download, try, use and remove. A
180/// model that does not fit is not offered, unless it is already installed and still
181/// fits in memory. Build it once and keep it: downloads belong to it.
182///
183/// ```no_run
184/// use ravenrs_ai::{CATALOG, ModelFolder, ModelManager};
185///
186/// let manager = ModelManager::new(CATALOG, ModelFolder::new("models"));
187/// ```
188#[derive(Clone, Debug)]
189pub struct ModelManager {
190    folder: ModelFolder,
191    entries: Rc<RefCell<Vec<Entry>>>,
192    in_use: Rc<RefCell<Option<(LocalModel, LocalAssistant)>>>,
193    changes: State<u64>,
194    labels: ModelManagerLabels,
195}
196
197impl ModelManager {
198    /// The models of `catalog` that fit this machine, kept in `folder`.
199    pub fn new(catalog: &[LocalModel], folder: ModelFolder) -> Self {
200        let machine = Machine::detect(folder.root());
201        Self::for_machine(catalog, folder, &machine)
202    }
203
204    /// The models of `catalog` that fit `machine`.
205    pub fn for_machine(catalog: &[LocalModel], folder: ModelFolder, machine: &Machine) -> Self {
206        let entries = catalog
207            .iter()
208            .filter_map(|model| {
209                let installed = folder.state(model) == ModelState::Installed;
210                model.fits(machine, installed).then_some(Entry {
211                    model: *model,
212                    installed,
213                    download: None,
214                    test: None,
215                    tested: None,
216                    did_not_run: false,
217                    error: None,
218                })
219            })
220            .collect::<Vec<Entry>>();
221        // What was in use last time, if it is still here and still fits.
222        let saved = std::fs::read_to_string(folder.root().join(IN_USE_FILE)).unwrap_or_default();
223        let in_use = entries
224            .iter()
225            .find(|entry| entry.installed && entry.model.id == saved.trim())
226            .map(|entry| (entry.model, LocalAssistant::new(folder.clone(), entry.model)));
227        Self {
228            folder,
229            entries: Rc::new(RefCell::new(entries)),
230            in_use: Rc::new(RefCell::new(in_use)),
231            changes: State::new(0),
232            labels: ModelManagerLabels::default(),
233        }
234    }
235
236    /// The words it shows.
237    #[must_use]
238    pub fn labels(mut self, labels: ModelManagerLabels) -> Self {
239        self.labels = labels;
240        self
241    }
242
243    /// The model in use, ready to be asked, if there is one.
244    pub fn assistant(&self) -> Option<LocalAssistant> {
245        self.in_use.borrow().as_ref().map(|(_, assistant)| assistant.clone())
246    }
247
248    fn is_in_use(&self, model: &LocalModel) -> bool {
249        self.in_use.borrow().as_ref().is_some_and(|(used, _)| used.id == model.id)
250    }
251
252    /// Starts or stops using a model, and remembers it. Letting go of the assistant
253    /// frees its memory.
254    fn toggle_use(&self, index: usize) {
255        let Some(model) = self.entries.borrow().get(index).map(|entry| entry.model) else { return };
256        let next =
257            (!self.is_in_use(&model)).then(|| (model, LocalAssistant::new(self.folder.clone(), model)));
258        self.set_in_use(next);
259    }
260
261    fn set_in_use(&self, next: Option<(LocalModel, LocalAssistant)>) {
262        let saved = next.as_ref().map_or("", |(model, _)| model.id);
263        let _ = std::fs::create_dir_all(self.folder.root());
264        // Remembering is a convenience: without it, the next run starts with no model.
265        let _ = std::fs::write(self.folder.root().join(IN_USE_FILE), saved);
266        *self.in_use.borrow_mut() = next;
267    }
268
269    fn test(&self, index: usize, waker: UiWaker) {
270        let mut entries = self.entries.borrow_mut();
271        let Some(entry) = entries.get_mut(index) else { return };
272        let run: TestRun = Arc::new(Mutex::new(None));
273        entry.test = Some(run.clone());
274        entry.tested = None;
275        entry.did_not_run = false;
276        let (model, folder) = (entry.model, self.folder.clone());
277        std::thread::spawn(move || {
278            let started = Instant::now();
279            let outcome = LocalEngine::load(&folder, &model).ok().and_then(|mut engine| {
280                let generated = engine.generate(TEST_SYSTEM, TEST_USER, TEST_TOKENS, &mut |_| true).ok()?;
281                (generated.tokens > 0).then(|| Tested {
282                    seconds: started.elapsed().as_secs_f64(),
283                    tokens_per_second: generated.tokens_per_second(),
284                })
285            });
286            if let Ok(mut done) = run.lock() {
287                *done = Some(outcome);
288            }
289            waker.wake();
290        });
291    }
292
293    fn changed(&self) {
294        self.changes.update(|n| *n += 1);
295    }
296
297    fn start(&self, index: usize, waker: UiWaker) {
298        let mut entries = self.entries.borrow_mut();
299        let Some(entry) = entries.get_mut(index) else { return };
300        let download = Arc::new(Download::default());
301        entry.download = Some(download.clone());
302        entry.error = None;
303        let (model, folder) = (entry.model, self.folder.clone());
304        std::thread::spawn(move || {
305            let step = (model.bytes() / STEPS).max(1);
306            let told = AtomicU64::new(0);
307            let progress = |received: u64, _total: u64| {
308                download.received.store(received, Ordering::Relaxed);
309                if received / step != told.swap(received / step, Ordering::Relaxed) {
310                    waker.wake();
311                }
312            };
313            let result = folder.download(&model, &progress, &download.cancelled);
314            if let Ok(mut finished) = download.finished.lock() {
315                *finished = Some(result);
316            }
317            waker.wake();
318        });
319    }
320
321    fn cancel(&self, index: usize) {
322        if let Some(download) = self.entries.borrow().get(index).and_then(|entry| entry.download.as_ref()) {
323            download.cancelled.store(true, Ordering::Relaxed);
324        }
325    }
326
327    fn remove(&self, index: usize) {
328        if let Some(entry) = self.entries.borrow_mut().get_mut(index) {
329            if self.is_in_use(&entry.model) {
330                self.set_in_use(None);
331            }
332            match self.folder.remove(&entry.model) {
333                Ok(()) => {
334                    entry.installed = false;
335                    entry.tested = None;
336                }
337                Err(error) => entry.error = Some(error),
338            }
339        }
340    }
341
342    /// Takes in the downloads that ended, and says how each model stands.
343    fn choices(&self) -> Vec<ModelChoice> {
344        let mut entries = self.entries.borrow_mut();
345        entries
346            .iter_mut()
347            .map(|entry| {
348                let ended = entry
349                    .download
350                    .as_ref()
351                    .and_then(|download| download.finished.lock().ok().and_then(|mut done| done.take()));
352                if let Some(result) = ended {
353                    entry.download = None;
354                    match result {
355                        Ok(()) => entry.installed = true,
356                        Err(ModelError::Cancelled) => {}
357                        Err(error) => entry.error = Some(error),
358                    }
359                }
360                let test_ended =
361                    entry.test.as_ref().and_then(|run| run.lock().ok().and_then(|mut done| done.take()));
362                if let Some(outcome) = test_ended {
363                    entry.test = None;
364                    entry.tested = outcome;
365                    entry.did_not_run = outcome.is_none();
366                }
367                let status = if entry.test.is_some() {
368                    ModelStatus::Testing
369                } else if let Some(download) = &entry.download {
370                    let received = download.received.load(Ordering::Relaxed);
371                    ModelStatus::Downloading(received as f32 / entry.model.bytes().max(1) as f32)
372                } else if entry.installed {
373                    ModelStatus::Installed
374                } else {
375                    ModelStatus::Available
376                };
377                ModelChoice {
378                    name: entry.model.name.to_owned(),
379                    details: self.labels.details(&entry.model),
380                    status,
381                    in_use: self.is_in_use(&entry.model),
382                    result: self.labels.result(entry.tested.as_ref()),
383                    error: if entry.did_not_run {
384                        Some(self.labels.did_not_run.clone())
385                    } else {
386                        entry.error.as_ref().and_then(|error| self.labels.error(error))
387                    },
388                }
389            })
390            .collect()
391    }
392}

394impl View for ModelManager {
395    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
396        // Read, so that a click rebuilds this view. A thread wakes it the other way.
397        let _ = self.changes.get();
398        let waker = cx.ui_waker();
399        let (starting, cancelling, removing) = (self.clone(), self.clone(), self.clone());
400        let (testing, using, test_waker) = (self.clone(), self.clone(), waker.clone());
401        let picker = ModelPicker::new(self.choices())
402            .labels(self.labels.picker.clone())
403            .on_download(move |index| {
404                starting.start(index, waker.clone());
405                starting.changed();
406            })
407            .on_cancel(move |index| {
408                cancelling.cancel(index);
409                cancelling.changed();
410            })
411            .on_test(move |index| {
412                testing.test(index, test_waker.clone());
413                testing.changed();
414            })
415            .on_use(move |index| {
416                using.toggle_use(index);
417                using.changed();
418            })
419            .on_remove(move |index| {
420                removing.remove(index);
421                removing.changed();
422            });
423        cx.child(&picker)
424    }
