1//! `FlowNotation`: a notation for diagrams (its elements, how they look, how they join, what
2//! makes a diagram right, the files it is kept in) handed to the diagram components at once.
3
4use std::fmt;
5use std::rc::Rc;
6
7use crate::flow::{
8    FlowConnector, FlowEdge, FlowGraph, FlowKind, FlowNext, FlowNode, FlowValidator, PortSide,
9};
10use crate::flow::{FlowEndpoint, FlowIssue};

12/// Makes a diagram of a file's text.
13type Reader = Rc<dyn Fn(&str) -> Result<FlowGraph, String>>;
14/// Makes the text of a diagram, given its name.
15type Writer = Rc<dyn Fn(&FlowGraph, &str) -> String>;
16
17/// A file format a [`FlowLibrary`] opens and saves diagrams in besides its own JSON, such as
18/// a notation's XML: what it is called, its extension, and how a diagram is read from it and
19/// written to it.
20#[derive(Clone)]
21#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
22pub struct FlowFileFormat {
23    pub(crate) name: String,
24    pub(crate) extension: String,
25    pub(crate) read: Reader,
26    pub(crate) write: Writer,
27}
28
29impl fmt::Debug for FlowFileFormat {
30    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
31        f.debug_struct("FlowFileFormat")
32            .field("name", &self.name)
33            .field("extension", &self.extension)
34            .finish()
35    }
36}

38#[cfg_attr(feature = "live", ravenrs_live::expose)]
39impl FlowFileFormat {
40    /// A format called `name`, in files ending in `.extension`: `read` makes a diagram of a
41    /// file's text, or says why it cannot; `write` makes the text of a diagram, given its name.
42    pub fn new(
43        name: impl Into<String>,
44        extension: impl Into<String>,
45        read: impl Fn(&str) -> Result<FlowGraph, String> + 'static,
46        write: impl Fn(&FlowGraph, &str) -> String + 'static,
47    ) -> Self {
48        Self { name: name.into(), extension: extension.into(), read: Rc::new(read), write: Rc::new(write) }
49    }
50}

52/// Says what is wrong with a diagram, for [`FlowCanvas::checks`](crate::FlowCanvas::checks).

55/// A notation for diagrams, such as a process notation or a flowchart's: the elements it has,
56/// grouped for a palette, how each kind looks and is called, how connections between them are
57/// drawn, which ones may be made, what comes next from the keys, what makes a diagram right and
58/// the files it is kept in.
59///
60/// It is data: give it to [`FlowCanvas::notation`](crate::FlowCanvas::notation),
61/// [`FlowToolbox::notation`](crate::FlowToolbox::notation),
62/// [`FlowInspector::notation`](crate::FlowInspector::notation) and
63/// [`FlowLibrary::notation`](crate::FlowLibrary::notation), and each takes the part it uses.
64///
65/// ```
66/// use ravenrs_widgets::{FlowCanvas, FlowGraph, FlowNode, FlowNotation, FlowShape, FlowToolbox};
67/// use ravenrs_reactive::State;
68///
69/// let notation = FlowNotation::new()
70///     .model("Steps", "Step", FlowNode::new("step", "step", "Step", 0.0, 0.0))
71///     .model("Steps", "Choice", FlowNode::new("choice", "choice", "Choice?", 0.0, 0.0).shape(FlowShape::Diamond))
72///     .kind("step", "Step")
73///     .kind("choice", "Choice");
74/// let graph = State::new(FlowGraph::new());
75/// let canvas = FlowCanvas::new(graph.binding()).notation(&notation);
76/// let palette = FlowToolbox::new().notation(&notation);
77/// ```
78#[derive(Clone, Default)]
79#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
80pub struct FlowNotation {
81    pub(crate) kinds: Vec<(String, String)>,
82    pub(crate) looks: Vec<(String, FlowKind)>,
83    pub(crate) models: Vec<(String, String, FlowNode)>,
84    pub(crate) connector: Option<FlowConnector>,
85    pub(crate) validate: Option<FlowValidator>,
86    pub(crate) next: Option<FlowNext>,
87    pub(crate) checks: Option<FlowChecks>,
88    pub(crate) runs: Option<crate::flow::FlowRunRules>,
89    pub(crate) formats: Vec<FlowFileFormat>,
90    pub(crate) cables: Option<bool>,
91}
92
93impl fmt::Debug for FlowNotation {
94    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
95        f.debug_struct("FlowNotation").field("kinds", &self.kinds).finish_non_exhaustive()
96    }
97}
98
99#[cfg_attr(feature = "live", ravenrs_live::expose)]
100impl FlowNotation {
101    /// A notation with nothing yet.
102    pub fn new() -> Self {
103        Self::default()
104    }
105
106    /// A kind of element, by its name in the diagram and the words shown for it.
107    pub fn kind(mut self, kind: impl Into<String>, caption: impl Into<String>) -> Self {
108        self.kinds.push((kind.into(), caption.into()));
109        self
110    }
111
112    /// How the cards of a kind look: their icon, word and tone.
113    pub fn look(mut self, kind: impl Into<String>, look: FlowKind) -> Self {
114        self.looks.push((kind.into(), look));
115        self
116    }
117
118    /// An element for the palette, in `group`, called `caption`, whose copies are `node`.
119    pub fn model(mut self, group: impl Into<String>, caption: impl Into<String>, node: FlowNode) -> Self {
120        self.models.push((group.into(), caption.into(), node));
121        self
122    }
123
124    /// Shapes each connection made, as [`FlowCanvas::connection`](crate::FlowCanvas::connection).
125    pub fn connection(mut self, shape: impl Fn(&FlowGraph, FlowEdge) -> FlowEdge + 'static) -> Self {
126        self.connector = Some(Rc::new(shape));
127        self
128    }
129
130    /// Which connections may be made, as [`FlowCanvas::validate`](crate::FlowCanvas::validate).
131    pub fn validate(
132        mut self,
133        rule: impl Fn(&FlowGraph, &FlowEndpoint, &FlowEndpoint) -> Result<(), String> + 'static,
134    ) -> Self {
135        self.validate = Some(Rc::new(rule));
136        self
137    }
138
139    /// What comes next from the keys and the arrows, as
140    /// [`FlowCanvas::next_node`](crate::FlowCanvas::next_node).
141    pub fn next_node(mut self, next: impl Fn(&FlowNode, PortSide) -> Option<FlowNode> + 'static) -> Self {
142        self.next = Some(Rc::new(next));
143        self
144    }
145
146    /// What makes a diagram right, as [`FlowCanvas::checks`](crate::FlowCanvas::checks).
147    pub fn checks(mut self, checks: impl Fn(&FlowGraph) -> Vec<FlowIssue> + 'static) -> Self {
148        self.checks = Some(Rc::new(checks));
149        self
150    }
151
152    /// How a diagram of it runs, as [`FlowCanvas::runs`](crate::FlowCanvas::runs).
153    pub fn runs(mut self, rules: crate::flow::FlowRunRules) -> Self {
154        self.runs = Some(rules);
155        self
156    }
157
158    /// Whether its connections are cables, as [`FlowCanvas::cables`](crate::FlowCanvas::cables).
159    pub fn cables(mut self, cables: bool) -> Self {
160        self.cables = Some(cables);
161        self
162    }
163
164    /// A file format its diagrams are kept in, as [`FlowLibrary::format`](crate::FlowLibrary::format).
165    pub fn format(mut self, format: FlowFileFormat) -> Self {
166        self.formats.push(format);
167        self
168    }
169
170    /// The words shown for a kind, if the notation has it.
171    pub fn caption(&self, kind: &str) -> Option<&str> {
172        self.kinds.iter().find(|(k, _)| k == kind).map(|(_, c)| c.as_str())
173    }
174}
