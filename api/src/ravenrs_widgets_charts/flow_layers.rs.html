15/// The layers of a diagram: its nodes from the front one down, what a frame holds under it and
16/// set in. A row selects its node on the canvas sharing the selection; its buttons hide and
17/// show the node (with what it holds and its lines), and bring it forward or send it back,
18/// each a change the shared history undoes.
19///
20/// ```
21/// use ravenrs_reactive::State;
22/// use ravenrs_widgets::{FlowCanvas, FlowGraph, FlowLayers, FlowSelection, HStack};
23///
24/// let (graph, selection) = (State::new(FlowGraph::new()), State::new(FlowSelection::default()));
25/// let board = HStack::new((
26///     FlowLayers::new(graph.binding(), selection.binding()),
27///     FlowCanvas::new(graph.binding()).selection(selection.binding()),
28/// ));
29/// ```
30#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
31pub struct FlowLayers {
32    graph: Binding<FlowGraph>,
33    selection: Binding<FlowSelection>,
34    history: Option<Binding<FlowHistory>>,
35    width: f32,
36}

45impl FlowLayers {
46    /// The layers of `graph`, picking into `selection`, 240 pixels wide.
47    pub fn new(graph: Binding<FlowGraph>, selection: Binding<FlowSelection>) -> Self {
48        Self { graph, selection, history: None, width: 240.0 }
49    }
50
51    /// Keeps its changes in this history, the canvas's.
52    pub fn history(mut self, history: Binding<FlowHistory>) -> Self {
53        self.history = Some(history);
54        self
55    }
56
57    /// Another width, in logical pixels.
58    pub fn width(mut self, width: f32) -> Self {
59        self.width = width;
60        self
61    }
62}

64/// Writes a change to the diagram, remembering it.
65fn write(
66    graph: &Binding<FlowGraph>,
67    history: &Option<Binding<FlowHistory>>,
68    edit: impl FnOnce(&mut FlowGraph),
69) {
70    let before = graph.get();
71    let mut next = before.clone();
72    edit(&mut next);
73    if next != before {
74        if let Some(history) = history {
75            history.update(|h| h.record(before));
76        }
77        graph.set(next);
78    }
79}

81/// Moves a node one place in the drawing order among the nodes of its own frame: forward is
82/// toward the front.
83fn shift(graph: &mut FlowGraph, id: &str, forward: bool) {
84    let Some(i) = graph.nodes.iter().position(|n| n.id == id) else { return };
85    let parent = graph.nodes[i].parent.clone();
86    let mates: Vec<usize> = (0..graph.nodes.len()).filter(|k| graph.nodes[*k].parent == parent).collect();
87    let Some(at) = mates.iter().position(|k| *k == i) else { return };
88    let other = if forward { mates.get(at + 1) } else { at.checked_sub(1).and_then(|a| mates.get(a)) };
89    if let Some(&j) = other {
90        graph.nodes.swap(i, j);
91    }
92}

94impl View for FlowLayers {
95    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
96        let theme = *cx.theme();
97        let labels = cx.labels();
98        let graph = self.graph.get();
99        let selection = self.selection.get();
100        let muted = theme.color(Semantic::InkMuted);
101        let mut rows: Vec<NodeId> = Vec::new();
102        rows.push(util::label(cx, &labels.flow_layers, TextStyle::Caption, Some(Weight::SEMIBOLD), muted));
103        // From the front down, each frame followed by what it holds.
104        let mut order: Vec<(usize, usize)> = Vec::new();
105        fn walk(graph: &FlowGraph, parent: Option<&str>, depth: usize, order: &mut Vec<(usize, usize)>) {
106            for (i, node) in graph.nodes.iter().enumerate().rev() {
107                if node.parent.as_deref() == parent {
108                    order.push((i, depth));
109                    if node.group {
110                        walk(graph, Some(&node.id), depth + 1, order);
111                    }
112                }
113            }
114        }
115        walk(&graph, None, 0, &mut order);
116        let mut list: Vec<NodeId> = Vec::new();
117        for (i, depth) in order {
118            let node = &graph.nodes[i];
119            let name = if node.title.trim().is_empty() { node.id.clone() } else { node.title.clone() };
120            let (pick, id) = (self.selection.clone(), node.id.clone());
121            let item = ListItem::new(name.clone())
122                .icon(if node.group { IconName::Layers } else { IconName::Square })
123                .selected(selection.has_node(&node.id))
124                .on_click(move || pick.set(FlowSelection::node(id.clone())));
125            let row = cx.keyed_child(util::hash_text(&format!("layer-{}", node.id)), &item);
126            let button = |cx: &mut BuildCx<'_>, icon: IconName, words: String, run: Box<dyn Fn()>| {
127                cx.child(
128                    &IconButton::new(icon)
129                        .label(words)
130                        .size(ButtonSize::Small)
131                        .style(ButtonStyle::Ghost)
132                        .on_click(run),
133                )
134            };
135            let (g1, h1, id1, hidden) =
136                (self.graph.clone(), self.history.clone(), node.id.clone(), node.hidden);
137            let eye = button(
138                cx,
139                if hidden { IconName::EyeOff } else { IconName::Eye },
140                if hidden {
141                    labels.flow_layer_show.replace("{0}", &name)
142                } else {
143                    labels.flow_layer_hide.replace("{0}", &name)
144                },
145                Box::new(move || {
146                    write(&g1, &h1, |g| {
147                        if let Some(n) = g.find_node_mut(&id1) {
148                            n.hidden = !n.hidden;
149                        }
150                    });
151                }),
152            );
153            let (g2, h2, id2) = (self.graph.clone(), self.history.clone(), node.id.clone());
154            let up = button(
155                cx,
156                IconName::ChevronUp,
157                labels.flow_layer_up.replace("{0}", &name),
158                Box::new(move || write(&g2, &h2, |g| shift(g, &id2, true))),
159            );
160            let (g3, h3, id3) = (self.graph.clone(), self.history.clone(), node.id.clone());
161            let down = button(
162                cx,
163                IconName::ChevronDown,
164                labels.flow_layer_down.replace("{0}", &name),
165                Box::new(move || write(&g3, &h3, |g| shift(g, &id3, false))),
166            );
167            let grow = cx.container(
168                LayoutStyle { grow: 1.0, ..LayoutStyle::stack(Axis::Vertical) },
169                &[row],
170                Paint::None,
171            );
172            list.push(cx.container(
173                LayoutStyle {
174                    align: Align::Center,
175                    spacing: 2.0,
176                    padding: Insets { left: 16.0 * depth as f32, ..Insets::default() },
177                    ..LayoutStyle::stack(Axis::Horizontal)
178                },
179                &[grow, eye, up, down],
180                Paint::None,
181            ));
182        }
183        let list = cx.container(
184            LayoutStyle { spacing: 2.0, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
185            &list,
186            Paint::None,
187        );
188        let scroll = cx.child(&ScrollView::new((Raw(list),)).spacing(0.0));
189        rows.push(scroll);
190        let root = cx.container(
191            LayoutStyle {
192                width: Some(self.width),
193                padding: Insets::uniform(theme.spacing.s4),
194                spacing: theme.spacing.s2,
195                align: Align::Stretch,
196                ..LayoutStyle::stack(Axis::Vertical)
197            },
198            &rows,
199            Paint::Quad(crate::context::QuadPaint::new(theme.color(Semantic::Surface))),
200        );
201        cx.semantics(root, Semantics::new(Role::Group).name(labels.flow_layers.clone()));
202        root
203    }
204}

215#[cfg(test)]
216mod tests {
217    use super::shift;
218    use crate::flow::{FlowGraph, FlowNode};
219
220    #[test]
221    fn a_layer_moves_among_the_nodes_of_its_own_frame() {
222        let mut g = FlowGraph::new()
223            .node(FlowNode::new("a", "", "A", 0.0, 0.0))
224            .node(FlowNode::new("f", "", "F", 0.0, 0.0))
225            .node(FlowNode::new("in", "", "In", 0.0, 0.0).parent("f"))
226            .node(FlowNode::new("b", "", "B", 0.0, 0.0));
227        shift(&mut g, "a", true);
228        let ids: Vec<&str> = g.nodes.iter().map(|n| n.id.as_str()).collect();
229        assert_eq!(ids, ["f", "a", "in", "b"]);
230        shift(&mut g, "in", true);
231        assert_eq!(g.nodes[2].id, "in", "alone in its frame, it stays");
232        shift(&mut g, "b", false);
233        let ids: Vec<&str> = g.nodes.iter().map(|n| n.id.as_str()).collect();
234        assert_eq!(ids, ["f", "b", "in", "a"]);
235    }
