1//! `FlowCanvas`: a diagram of nodes joined by connections, edited on an endless surface.
2//!
3//! The model and the lines' geometry live in `ravenrs_graphics::flow`, which knows no engine;
4//! here they become nodes that take focus and the pointer, ports that start connections, and
5//! lines, labels, a grid, controls and a map drawn from the same view.
6//!
7//! The runtime has no transform for a laid out subtree, so a node does not scale a view of its
8//! own: each node is a box placed and sized by the zoom, which draws its icon, caption, title
9//! and line at that zoom. Clicks, focus and screen readers find it where it shows.

76/// from, what went wrong and the last things that happened.
77#[allow(clippy::too_many_arguments, reason = "what the box shows")]
78fn debug_box(
79    cx: &mut BuildCx<'_>,
80    flow: &Flow,
81    graph: &FlowGraph,
82    rules: &FlowRunRules,
83    run: Option<&FlowRun>,
84    size: Size,
85    theme: &Theme,
86    labels: &crate::Labels,
87) -> NodeId {
88    let (f1, f2, f3, f4) = (flow.clone(), flow.clone(), flow.clone(), flow.clone());
89    let over = run.is_some_and(FlowRun::is_over);
90    let waiting = run.is_some_and(|r| r.choosing.is_some());
91    let mut rows: Vec<NodeId> = Vec::new();
92    rows.push(util::label(
93        cx,
94        &labels.flow_debug,
95        TextStyle::Caption,
96        Some(Weight::SEMIBOLD),
97        theme.color(Semantic::InkMuted),
98    ));
99    let buttons = crate::HStack::new((
100        crate::Button::new(labels.flow_run_start.clone())
101            .icon(IconName::Play)
102            .size(ButtonSize::Small)
103            .on_click(move || f1.run_start()),
104        crate::Button::new(labels.flow_run_step.clone())
105            .size(ButtonSize::Small)
106            .style(ButtonStyle::Secondary)
107            .disabled(over || waiting)
108            .on_click(move || f2.run_step(None)),
109        crate::Button::new(labels.flow_run_continue.clone())
110            .size(ButtonSize::Small)
111            .style(ButtonStyle::Secondary)
112            .disabled(over || waiting)
113            .on_click(move || f3.run_on()),
114        crate::Button::new(labels.flow_run_stop.clone())
115            .icon(IconName::Square)
116            .size(ButtonSize::Small)
117            .style(ButtonStyle::Ghost)
118            .disabled(run.is_none())
119            .on_click(move || f4.run.set(None)),
120    ))
121    .spacing(4.0);
122    rows.push(cx.child(&buttons));
123    let f5 = flow.clone();
124    rows.push(
125        cx.child(
126            &crate::Button::new(labels.flow_run_breakpoint.clone())
127                .size(ButtonSize::Small)
128                .style(ButtonStyle::Ghost)
129                .disabled(flow.selection.get().nodes.is_empty())
130                .on_click(move || f5.toggle_breakpoints()),
131        ),
132    );
133    if let Some(r) = run {
134        let many = r
135            .choosing
136            .as_ref()
137            .and_then(|n| graph.find_node(n))
138            .is_some_and(|n| (rules.gate)(n) == FlowGate::ChooseMany);
139        if let (Some(node), true) = (&r.choosing, many) {
140            // One or more ways: a box each, then Follow.
141            let name = graph.find_node(node).map_or(node.clone(), |n| n.title.clone());
142            rows.push(util::label(
143                cx,
144                &labels.flow_run_choose_many.replace("{0}", &name),
145                TextStyle::Callout,
146                None,
147                theme.color(Semantic::Ink),
148            ));
149            for e in FlowRun::ways_out(graph, rules, node) {
150                let to = graph.find_node(&e.to.node).map_or(e.to.node.clone(), |n| n.title.clone());
151                let words = e.label.clone().map_or(to.clone(), |l| format!("{l} → {to}"));
152                let (picked, id) = (flow.picked.clone(), e.id.clone());
153                let (reader, writer) = (picked.clone(), id.clone());
154                let ticked = ravenrs_reactive::Binding::new(
155                    move || reader.get().contains(&id),
156                    move |on: bool| {
157                        let mut all = picked.get();
158                        all.retain(|w| *w != writer);
159                        if on {
160                            all.push(writer.clone());
161                        }
162                        picked.set(all);
163                    },
164                );
165                rows.push(cx.child(&crate::Checkbox::new(words, ticked)));
166            }
167            let follow = flow.clone();
168            rows.push(
169                cx.child(
170                    &crate::Button::new(labels.flow_run_follow.clone())
171                        .size(ButtonSize::Small)
172                        .on_click(move || follow.run_ways()),
173                ),
174            );
175            if flow.refused_choice.get() {
176                rows.push(util::label(
177                    cx,
178                    &labels.flow_run_none_chosen,
179                    TextStyle::Caption,
180                    None,
181                    theme.color(Semantic::Danger),
182                ));
183            }
184        } else if let Some(node) = &r.choosing {
185            let ways: Vec<crate::Button> = FlowRun::ways_out(graph, rules, node)
186                .into_iter()
187                .map(|e| {
188                    let to = graph.find_node(&e.to.node).map_or(e.to.node.clone(), |n| n.title.clone());
189                    let words = e.label.clone().map_or(to.clone(), |l| format!("{l} → {to}"));
190                    let (chosen, id) = (flow.clone(), e.id.clone());
191                    crate::Button::new(words)
192                        .size(ButtonSize::Small)
193                        .style(ButtonStyle::Secondary)
194                        .on_click(move || chosen.run_step(Some(&id)))
195                })
196                .collect();
197            let name = graph.find_node(node).map_or(node.clone(), |n| n.title.clone());
198            rows.push(util::label(
199                cx,
200                &labels.flow_run_choose.replace("{0}", &name),
201                TextStyle::Callout,
202                None,
203                theme.color(Semantic::Ink),
204            ));
205            rows.push(cx.child(&crate::VStack::new(ways).spacing(4.0).align(Align::Stretch)));
206        }
207        if let Some(failure) = r.failure() {
208            rows.push(cx.child(&Alert::message(run_words(failure, graph, labels)).tone(crate::Tone::Danger)));
209        } else if r.is_over() {
210            rows.push(cx.child(&Alert::message(labels.flow_run_done.clone()).tone(crate::Tone::Success)));
211        }
212        for event in r.events.iter().rev().take(6).rev() {
213            let color = if event.is_error() {
214                theme.color(Semantic::Danger)
215            } else {
216                theme.color(Semantic::InkMuted)
217            };
218            rows.push(util::label(cx, &run_words(event, graph, labels), TextStyle::Caption, None, color));
219        }
220    } else {
221        rows.push(util::label(
222            cx,
223            &labels.flow_run_hint,
224            TextStyle::Caption,
225            None,
226            theme.color(Semantic::InkMuted),
227        ));
228    }
229    let width = CHECK_WIDTH;
230    let root = cx.container(
231        LayoutStyle {
232            absolute: Some(((size.width - width - theme.spacing.s4).max(0.0), theme.spacing.s4)),
233            width: Some(width),
234            padding: ravenrs_core::Insets::uniform(theme.spacing.s3),
235            spacing: theme.spacing.s2,
236            align: Align::Stretch,
237            ..LayoutStyle::stack(Axis::Vertical)
238        },
239        &rows,
240        Paint::Quad(crate::context::QuadPaint {
241            corner_radius: theme.radii.md,
242            border: Some(ravenrs_core::Border { width: 1.0, color: theme.color(Semantic::Border) }),
243            shadow: Some(theme.elevation.soft),
244            ..crate::context::QuadPaint::new(theme.color(Semantic::Surface))
245        }),
246    );
247    cx.semantics(root, Semantics::new(Role::Group).name(labels.flow_debug.clone()));
248    root
249}

251/// What a run's event says, in the app's words.
252fn run_words(event: &FlowRunEvent, graph: &FlowGraph, labels: &crate::Labels) -> String {
253    let name = |id: &str| {
254        graph
255            .find_node(id)
256            .map(|n| if n.title.trim().is_empty() { n.id.clone() } else { n.title.clone() })
257            .unwrap_or_else(|| id.to_owned())
258    };
259    match event {
260        FlowRunEvent::Moved { from, to, .. } => {
261            labels.flow_run_moved.replace("{0}", &name(from)).replace("{1}", &name(to))
262        }
263        FlowRunEvent::Ended { node } => labels.flow_run_ended.replace("{0}", &name(node)),
264        FlowRunEvent::Choosing { node } => labels.flow_run_choose.replace("{0}", &name(node)),
265        FlowRunEvent::Finished => labels.flow_run_done.clone(),
266        FlowRunEvent::DeadEnd { node } => labels.flow_run_dead_end.replace("{0}", &name(node)),
267        FlowRunEvent::Stuck { node } => labels.flow_run_stuck.replace("{0}", &name(node)),
268        FlowRunEvent::Looping => labels.flow_run_looping.clone(),
269        _ => labels.flow_run_no_start.clone(),
270    }
271}

284/// Which part of the diagram is in view: where its origin lands in the view and how large it
285/// is drawn. [`FlowCanvas`] writes it as people pan and zoom, and anything sharing the binding
286/// can move the view.
287#[derive(Clone, Copy, Debug, PartialEq)]
288#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
289pub struct FlowViewport {
290    /// Where the diagram's point `(0, 0)` is, from the top left corner of the view.
291    pub offset: Point,
292    /// Screen pixels per unit of the diagram: 1 is 100%.
293    pub zoom: f32,
294    /// Fits the whole diagram in view on the next build, then turns itself off.
295    pub fit: bool,
296}
297
298impl Default for FlowViewport {
299    fn default() -> Self {
300        Self::fit()
301    }
302}
303
304#[cfg_attr(feature = "live", ravenrs_live::expose)]
305impl FlowViewport {
306    /// The whole diagram in view.
307    pub const fn fit() -> Self {
308        Self { offset: Point::new(0.0, 0.0), zoom: 1.0, fit: true }
309    }
310
311    /// At this zoom, with the diagram's origin at `offset`.
312    pub const fn at(offset: Point, zoom: f32) -> Self {
313        Self { offset, zoom, fit: false }
314    }
315
316    /// Where a point of the diagram shows, from the top left corner of the view.
317    pub fn to_screen(&self, p: Point) -> Point {
318        Point::new(self.offset.x + p.x * self.zoom, self.offset.y + p.y * self.zoom)
319    }
320
321    /// The point of the diagram under a point of the view.
322    pub fn to_content(&self, p: Point) -> Point {
323        let zoom = self.zoom.max(f32::EPSILON);
324        Point::new((p.x - self.offset.x) / zoom, (p.y - self.offset.y) / zoom)
325    }
326
327    /// At another zoom, with the point under `anchor` (in the view) staying there.
328    pub fn zoomed(self, zoom: f32, anchor: Point) -> Self {
329        let under = self.to_content(anchor);
330        Self { offset: Point::new(anchor.x - under.x * zoom, anchor.y - under.y * zoom), zoom, fit: false }
331    }
332
333    /// The zoom and offset that show `bounds` whole and centered in a view of `size`.
334    pub fn fitting(bounds: Rect, size: Size, least: f32, most: f32) -> Self {
335        let room =
336            Size::new((size.width - FIT_MARGIN * 2.0).max(1.0), (size.height - FIT_MARGIN * 2.0).max(1.0));
337        let zoom = (room.width / bounds.width().max(1.0))
338            .min(room.height / bounds.height().max(1.0))
339            .clamp(least, most);
340        let center = bounds.center();
341        Self {
342            offset: Point::new(size.width / 2.0 - center.x * zoom, size.height / 2.0 - center.y * zoom),
343            zoom,
344            fit: false,
345        }
346    }
347}

550/// A diagram of nodes joined by connections, on an endless surface with a dotted grid.
551///
552/// * Drag the surface or turn the wheel to pan; pinch, or turn the wheel with Cmd/Ctrl, to zoom
553///   around the pointer; `+`, `-` and `0` zoom in, out and to the whole diagram.
554/// * Press a node to select it (Shift adds to the selection), drag it to move every selected
555///   node, and Shift-drag the surface to select with a rectangle. Arrows move what is selected
556///   one grid step (with Shift, four), Delete takes it out and Escape lets it go.
557/// * Drag from a port to another to connect them. While dragging, the ports that would accept
558///   the connection light up; one that refuses turns red, and the reason shows by the pointer.
559///   With the keys, Tab to a port, press Enter, Tab to the other port and press Enter again.
560/// * Select a connection and drag one of its ends, which show as handles, to another port to
561///   move that end.
562/// * Drag the corner of a selected node, or press Alt with the arrows, to resize it. The
563///   arrange button in the controls lays the diagram out in layers ([`FlowLayout`]).
564/// * Cmd/Ctrl+G puts the selected nodes in a group, which moves, copies and goes with what it
565///   holds; Shift+Cmd/Ctrl+G takes the selected groups apart. Dropping a node inside a group
566///   puts it there, and dropping it outside takes it out.
567/// * Double-click a node, or press F2 (or Enter on a selected node), to type its title in
568///   place: Enter keeps it, Escape gives it up.
569/// * Drag the round handle above a selected node to turn it, 15 degrees at a time (with Shift,
570///   freely), or press Alt+Shift with the arrows; Alt+Shift+0 puts it straight. Groups do
571///   not turn.
572/// * With the pointer over a node, arrows show on its sides: a click on one makes the next
573///   node there, joined to it, with its title open to type. Cmd/Ctrl+Enter does it to the right
574///   of the selected node, Cmd/Ctrl+Shift+Enter below ([`FlowCanvas::next_node`]).
575/// * Cmd/Ctrl+Z undoes, Shift+Cmd/Ctrl+Z (or Cmd/Ctrl+Y) redoes. Cmd/Ctrl+C copies the selected
576///   nodes with the connections between them, as JSON, Cmd/Ctrl+X cuts them and Cmd/Ctrl+V

592///         .edge(FlowEdge::new("e1", FlowEndpoint::new("sale", "out"), FlowEndpoint::new("stock", "in"))),
593/// );

595///     .kind("trigger", FlowKind::new(IconName::Zap, "Trigger"))
596///     .kind("action", FlowKind::new(IconName::Package, "Action"));

1199    /// Opens the title of `id` for typing, with all of it selected. Below the zoom at which
1200    /// nodes show words, the view first goes to 100% around the node, and comes back after.
1201    /// The node to make next to `from` on `side`, and the connection that joins them (output
1202    /// to input, in the way the side points), placed a layer away on the grid, lower down
1203    /// while that place is taken.
1204    fn next_to(
1205        &self,
1206        graph: &FlowGraph,
1207        from: &FlowNode,
1208        side: PortSide,
1209    ) -> Option<(FlowNode, Option<FlowEdge>)> {
1210        let made = match &self.next {
1211            Some(next) => next(from, side)?,
1212            None => FlowNode {
1213                status: FlowNodeStatus::Normal,
1214                parent: None,
1215                group: false,
1216                rotation: 0.0,
1217                ..from.clone()
1218            },
1219        };
1220        let gap = 96.0;
1221        let (w, h) = (made.width, made.height);
1222        let (mut x, mut y) = match side {
1223            PortSide::Right => (from.x + from.width + gap, from.y),
1224            PortSide::Left => (from.x - gap - w, from.y),
1225            PortSide::Bottom => (from.x, from.y + from.height + gap),
1226            PortSide::Top => (from.x, from.y - gap - h),
1227        };
1228        (x, y) = (self.snapped(x), self.snapped(y));
1229        let taken = |x: f32, y: f32| {
1230            let r = Rect::new(x, y, w, h);
1231            graph.nodes.iter().any(|n| {
1232                let e = n.extent();
1233                !n.group && e.x() < r.max_x() && r.x() < e.max_x() && e.y() < r.max_y() && r.y() < e.max_y()
1234            })
1235        };
1236        for _ in 0..20 {
1237            if !taken(x, y) {
1238                break;
1239            }
1240            y = self.snapped(y + h + 48.0);
1241        }
1242        let id = graph.fresh_node_id(&made.id);
1243        let made = FlowNode { id: id.clone(), x, y, ..made };
1244        // Forward (right, down): the old node's output to the new one's input; backward the
1245        // other way round. A port on the side faced is preferred.
1246        let forward = matches!(side, PortSide::Right | PortSide::Bottom);
1247        let opposite = match side {
1248            PortSide::Left => PortSide::Right,
1249            PortSide::Right => PortSide::Left,
1250            PortSide::Top => PortSide::Bottom,
1251            PortSide::Bottom => PortSide::Top,
1252        };
1253        let pick = |node: &FlowNode, direction: PortDirection, facing: PortSide| {
1254            let of: Vec<&FlowPort> = node.ports.iter().filter(|p| p.direction == direction).collect();
1255            of.iter()
1256                .find(|p| p.side == facing)
1257                .or(of.first())
1258                .map(|p| FlowEndpoint::new(node.id.clone(), p.id.clone()))
1259        };
1260        let ends = if forward {
1261            pick(from, PortDirection::Output, side).zip(pick(&made, PortDirection::Input, opposite))
1262        } else {
1263            pick(&made, PortDirection::Output, opposite).zip(pick(from, PortDirection::Input, side))
1264        };
1265        let edge = ends.map(|(a, b)| self.shaped(graph, FlowEdge::new(graph.fresh_edge_id("e"), a, b)));
1266        Some((made, edge))
1267    }

1390    /// group its middle is over, or in none.

1392        for id in moved {
1393            let Some(node) = graph.find_node(id) else { continue };
1394            if node.parent.as_ref().is_some_and(|p| moved.contains(p)) {
1395                continue;
1396            }

1398            let inside = graph.descendants(id);
1399            let target = graph
1400                .nodes
1401                .iter()
1402                .filter(|g| g.group && &g.id != id && !inside.contains(&g.id) && g.rect().contains(middle))
1403                .min_by(|a, b| (a.width * a.height).total_cmp(&(b.width * b.height)))
1404                .map(|g| g.id.clone());

1485    /// A drag on the round handle above a node, which turns it around its middle: the angle of
1486    /// the pointer around it, 15 degrees at a time unless `free`.
1487    fn on_turn_drag(&self, id: &str, phase: DragPhase, position: Point, free: bool) {
1488        if self.locked.get() || !self.rotatable {
1489            return;
1490        }
1491        match phase {
1492            DragPhase::Start => {
1493                self.forget_notice();
1494                *self.memory.gesture.borrow_mut() =
1495                    Gesture::Turn { id: id.to_owned(), before: self.graph.get() };
1496            }
1497            DragPhase::Move => {
1498                let Gesture::Turn { id, .. } = self.memory.gesture.borrow().clone() else { return };
1499                let mut graph = self.graph.get();
1500                let Some(node) = graph.find_node_mut(&id) else { return };
1501                let middle = self.view.get().to_screen(node.rect().center());
1502                let at = self.local(position);
1503                // Straight up is no turn: the handle starts above the node.
1504                let mut degrees = (at.y - middle.y).atan2(at.x - middle.x).to_degrees() + 90.0;
1505                if !free {
1506                    degrees = (degrees / TURN_STEP).round() * TURN_STEP;
1507                }
1508                let degrees = degrees.rem_euclid(360.0);
1509                if node.rotation != degrees {
1510                    node.rotation = degrees;
1511                    self.graph.set(graph);
1512                }
1513            }
1514            DragPhase::End => {
1515                let gesture = std::mem::take(&mut *self.memory.gesture.borrow_mut());
1516                if let Gesture::Turn { id, before } = gesture {
1517                    self.settle(before, FlowChange::Rotate { nodes: vec![id] });
1518                }
1519            }
1520            // The base may add more; they act as nothing here.
1521            _ => {}
1522        }
1523    }

1697                if let Some(w) = graph.find_edge_mut(&edge).and_then(|e| e.waypoints.get_mut(index))
1698                    && *w != point
1699                {
1700                    *w = point;
1701                    self.graph.set(graph);
1702                }
1703            }
1704            DragPhase::End => {
1705                let gesture = std::mem::take(&mut *self.memory.gesture.borrow_mut());
1706                let Gesture::Bend { edge, index, before } = gesture else { return };
1707                // A point let go on the straight between its neighbours is not needed: it goes.
1708                let mut graph = self.graph.get();
1709                let zoom = self.view.get().zoom;
1710                let straight = graph.find_edge(&edge).and_then(|e| graph.edge_controls(e)).is_some_and(|c| {
1711                    c.len() > index + 2
1712                        && geometry::distance_to(&[c[index], c[index + 2]], c[index + 1]) * zoom
1713                            <= BEND_STRAIGHT
1714                });
1715                if straight && let Some(e) = graph.find_edge_mut(&edge) {
1716                    e.waypoints.remove(index);
1717                    self.graph.set(graph);
1718                }
1719                self.settle(before, FlowChange::Bend { edge });
1720            }
1721            // The base may add more; they act as nothing here.
1722            _ => {}
1723        }
1724    }

1726    /// The keys on a bend point: the arrows move it a step of the grid (four with Shift),
1727    /// Delete takes it out.
1728    fn bend_key(&self, edge: &str, index: usize, key: &Key, m: Modifiers) -> bool {
1729        if self.locked.get() {
1730            return false;
1731        }

1733        let mut graph = self.graph.get();

1735        let Some(&(x, y)) = e.waypoints.get(index) else { return false };

1743                self.memory.refocus.set(true);
1744            }
1745            _ => return false,
1746        }

1773            return;
1774        }
1775        let at = self.local(position);
1776        match phase {
1777            DragPhase::Start => {
1778                self.forget_notice();
1779                let graph = self.graph.get();
1780                let Some(node) = graph.find_node(id) else { return };
1781                let size = (node.width, node.height);
1782                *self.memory.gesture.borrow_mut() =
1783                    Gesture::Resize { id: id.to_owned(), pressed: at, size, before: graph };

1904    /// The keys every part of the canvas answers.
1905    fn on_key(&self, key: &Key, m: Modifiers) -> bool {
1906        let (shift, command) = (m.shift, m.meta || m.ctrl);
1907        self.forget_notice();
1908        let step = self.grid * if shift { 4.0 } else { 1.0 };
1909        let has_nodes = !self.selection.get().nodes.is_empty();
1910        if *key == Key::Escape && self.opened.get().is_some() {
1911            self.opened.set(None);
1912            return true;
1913        }
1914        if *key == Key::Escape && self.connecting.get().is_some() {
1915            self.connecting.set(None);
1916            return true;
1917        }
1918        if *key == Key::Function(9) && self.runs.is_some() && has_nodes {
1919            self.toggle_breakpoints();
1920            return true;
1921        }
1922        if command && let Key::Character(c) = key {
1923            match c.to_lowercase().as_str() {
1924                "z" => return self.travel(shift),
1925                "y" => return self.travel(true),
1926                "c" => return self.copy(),
1927                "x" => return self.cut(),
1928                "v" => return self.paste(),
1929                "e" if shift && self.export.is_some() => {
1930                    self.export_menu.set(true);
1931                    return true;
1932                }
1933                "g" if shift => return self.ungroup_selection(),
1934                "g" => return self.group_selection(),
1935                "]" | "}" => return self.reorder(true),
1936                "[" | "{" => return self.reorder(false),
1937                _ => {}
1938            }
1939        }
1940        if m.alt && shift {
1941            match key {
1942                Key::ArrowLeft => return self.turn_by(Some(-TURN_STEP)),
1943                Key::ArrowRight => return self.turn_by(Some(TURN_STEP)),
1944                Key::Character(c) if c == "0" || m.base_key == Some('0') => return self.turn_by(None),
1945                _ => {}
1946            }
1947        }
1948        if m.alt {
1949            match key {
1950                Key::ArrowLeft => return self.resize_by(-step, 0.0),
1951                Key::ArrowRight => return self.resize_by(step, 0.0),
1952                Key::ArrowUp => return self.resize_by(0.0, -step),
1953                Key::ArrowDown => return self.resize_by(0.0, step),
1954                _ => {}
1955            }
1956        }
1957        if *key == Key::Enter
1958            && command
1959            && let [only] = self.selection.get().nodes.as_slice()
1960        {
1961            return self.quick_create(only, if shift { PortSide::Bottom } else { PortSide::Right });
1962        }
1963        if *key == Key::Function(2)
1964            && let [only] = self.selection.get().nodes.as_slice()
1965        {
1966            return self.begin_rename(only);
1967        }
1968        match key {
1969            Key::Character(c) => match c.as_str() {
1970                "+" | "=" => self.zoom_by(ZOOM_STEP),
1971                "-" | "_" => self.zoom_by(1.0 / ZOOM_STEP),
1972                "0" => self.fit(),
1973                "a" | "A" if command => {
1974                    let g = self.graph.get();
1975                    self.select(FlowSelection {
1976                        nodes: g.nodes.iter().map(|n| n.id.clone()).collect(),
1977                        edges: g.edges.iter().map(|e| e.id.clone()).collect(),
1978                    });
1979                }
1980                _ => return false,
1981            },
1982            Key::ArrowLeft if has_nodes => return self.nudge(-step, 0.0),
1983            Key::ArrowRight if has_nodes => return self.nudge(step, 0.0),
1984            Key::ArrowUp if has_nodes => return self.nudge(0.0, -step),
1985            Key::ArrowDown if has_nodes => return self.nudge(0.0, step),
1986            Key::ArrowLeft => self.pan_by(step, 0.0),
1987            Key::ArrowRight => self.pan_by(-step, 0.0),
1988            Key::ArrowUp => self.pan_by(0.0, step),
1989            Key::ArrowDown => self.pan_by(0.0, -step),
1990            Key::Delete | Key::Backspace => return self.delete_selection(),
1991            Key::Escape => {
1992                if self.selection.get().is_empty() {
1993                    return false;
1994                }
1995                self.select(FlowSelection::default());
1996            }
1997            _ => return false,
1998        }
1999        true
2000    }

2097    /// A press and drag on a node.
2098    fn on_node_drag(&self, id: &str, phase: DragPhase, position: Point, m: Modifiers) {
2099        let shift = m.shift;
2100        let at = self.local(position);
2101        self.memory.pointer.set(Some(at));
2102        match phase {
2103            DragPhase::Start => {
2104                self.forget_notice();
2105                // A group lies under the lines inside it: a press on one of them picks the line.
2106                let on_group = self.graph.get().find_node(id).is_some_and(|n| n.group);
2107                if on_group && let Some(edge) = self.edge_at(at) {
2108                    self.select(FlowSelection::edge(edge));
2109                    *self.memory.gesture.borrow_mut() = Gesture::Idle;
2110                    return;
2111                }
2112                let mut selection = self.selection.get();
2113                if shift {
2114                    if selection.has_node(id) {
2115                        selection.nodes.retain(|n| n != id);
2116                    } else {
2117                        selection.nodes.push(id.to_owned());
2118                    }
2119                } else if !selection.has_node(id) {
2120                    selection = FlowSelection::node(id);
2121                }
2122                self.select(selection.clone());
2123                if self.locked.get() || !selection.has_node(id) {
2124                    *self.memory.gesture.borrow_mut() = Gesture::Idle;
2125                    return;
2126                }
2127                let graph = self.graph.get();
2128                let moving = graph.with_descendants(&selection.nodes);
2129                let origins = graph
2130                    .nodes
2131                    .iter()
2132                    .filter(|n| moving.contains(&n.id))
2133                    .map(|n| (n.id.clone(), n.x, n.y))
2134                    .collect();
2135                *self.memory.gesture.borrow_mut() =
2136                    Gesture::Move { pressed: at, origins, before: graph, lead: id.to_owned() };
2137            }
2138            DragPhase::Move => {

2140                else {
2141                    return;
2142                };
2143                let zoom = self.view.get().zoom.max(f32::EPSILON);
2144                let (mut dx, mut dy) = ((at.x - pressed.x) / zoom, (at.y - pressed.y) / zoom);
2145                let mut graph = self.graph.get();
2146                // Guides: the node held lines up its edges or middle with another node's, within
2147                // a few pixels on screen, before the grid; Alt does without.
2148                let (mut across, mut down) = (None, None);
2149                if !m.alt
2150                    && let Some((_, x0, y0)) = origins.iter().find(|(id, _, _)| *id == lead)
2151                    && let Some(held) = graph.find_node(&lead)
2152                {
2153                    let moving: Vec<&String> = origins.iter().map(|(id, _, _)| id).collect();
2154                    let others: Vec<Rect> = graph
2155                        .nodes
2156                        .iter()
2157                        // Other nodes, and the frame the node held is in, so it lines up with its edges.
2158                        .filter(|n| {
2159                            !moving.contains(&&n.id)
2160                                && (!n.group || held.parent.as_deref() == Some(n.id.as_str()))
2161                        })
2162                        .map(FlowNode::extent)
2163                        .collect();
2164                    let reach = GUIDE_REACH / zoom;
2165                    let (w, h) = (held.width, held.height);
2166                    let lines = |at: f32, size: f32| [at, at + size / 2.0, at + size];
2167                    let best = |mine: [f32; 3], theirs: &dyn Fn(&Rect) -> [f32; 3]| {
2168                        others
2169                            .iter()
2170                            .flat_map(theirs)
2171                            .flat_map(|t| mine.iter().map(move |m| (t - m, t)))
2172                            .filter(|(d, _)| d.abs() <= reach)
2173                            .min_by(|a, b| a.0.abs().total_cmp(&b.0.abs()))
2174                    };

2176                        dx += d;
2177                        across = Some(t);
2178                    }
2179                    if let Some((d, t)) = best(lines(y0 + dy, h), &|r: &Rect| lines(r.y(), r.height())) {
2180                        dy += d;
2181                        down = Some(t);
2182                    }
2183                }
2184                self.guides.set_if_changed((across, down));
2185                let mut changed = false;
2186                // A line whose two ends move takes its bend points along; with one end, they stay.
2187                if let Gesture::Move { before, .. } = &*self.memory.gesture.borrow()
2188                    && let Some((_, x0, y0)) = origins.iter().find(|(id, _, _)| *id == lead)
2189                {
2190                    let (nx, ny) = (
2191                        if across.is_some() { x0 + dx } else { self.snapped(x0 + dx) },
2192                        if down.is_some() { y0 + dy } else { self.snapped(y0 + dy) },
2193                    );
2194                    let moving = |id: &str| origins.iter().any(|(o, _, _)| o == id);
2195                    for edge in &mut graph.edges {
2196                        if edge.waypoints.is_empty() || !moving(&edge.from.node) || !moving(&edge.to.node) {
2197                            continue;
2198                        }
2199                        let Some(was) = before.find_edge(&edge.id) else { continue };
2200                        let shifted: Vec<(f32, f32)> =
2201                            was.waypoints.iter().map(|(x, y)| (x + nx - x0, y + ny - y0)).collect();
2202                        if shifted != edge.waypoints {
2203                            edge.waypoints = shifted;
2204                            changed = true;
2205                        }
2206                    }
2207                }
2208                for (id, x, y) in &origins {
2209                    let nx = if across.is_some() { x + dx } else { self.snapped(x + dx) };
2210                    let ny = if down.is_some() { y + dy } else { self.snapped(y + dy) };
2211                    if let Some(node) = graph.find_node_mut(id)
2212                        && (node.x != nx || node.y != ny)
2213                    {
2214                        node.x = nx;
2215                        node.y = ny;
2216                        changed = true;
2217                    }
2218                }

2221                }
2222            }

4497#[cfg_attr(feature = "live", ravenrs_live::expose)]
4498impl MiniMap {
4499    /// The area the map shows, in units of the diagram: the nodes and the view together.
4500    fn world(&self) -> Rect {
4501        let seen = Rect::from_origin_size(
4502            self.view.to_content(Point::new(0.0, 0.0)),
4503            Size::new(
4504                self.size.width / self.view.zoom.max(f32::EPSILON),
4505                self.size.height / self.view.zoom.max(f32::EPSILON),
4506            ),
4507        );
4508        let all = self.graph.bounds().map_or(seen, |b| geometry::union(b, seen));
4509        all.inflate(FIT_MARGIN)
4510    }
4511}

4513impl View for MiniMap {

4515        let theme = *cx.theme();
4516        let labels = cx.labels();
4517        let p = self.palette;
4518        let world = self.world();
4519        let scale = (MAP_SIZE.width / world.width().max(1.0)).min(MAP_SIZE.height / world.height().max(1.0));

4522                (MAP_SIZE.width - world.width() * scale) / 2.0,
4523                (MAP_SIZE.height - world.height() * scale) / 2.0,
