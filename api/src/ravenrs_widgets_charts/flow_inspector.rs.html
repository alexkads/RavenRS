40/// The properties of what is selected in a [`FlowCanvas`](crate::FlowCanvas): title, line,
41/// kind, shape, size, place and status of a node; label, shape, ends and motion of a
42/// connection; for several nodes, what they have in common, a value that differs shown as
43/// mixed and, when chosen, given to all of them.
44///
45/// Give it the canvas's diagram and selection, and, so that its changes are undone and refused
46/// with the canvas's, the same [`FlowHistory`] and the same `on_change`. Text is kept when
47/// Enter is pressed or the field is left; a choice or a number, as it is made.
48///
49/// ```
50/// use ravenrs_reactive::State;
51/// use ravenrs_widgets::{FlowCanvas, FlowGraph, FlowHistory, FlowInspector, FlowSelection, HStack};
52///
53/// let (graph, selection, history) =
54///     (State::new(FlowGraph::new()), State::new(FlowSelection::default()), State::new(FlowHistory::new()));
55/// let editor = HStack::new((
56///     FlowCanvas::new(graph.binding()).selection(selection.binding()).history(history.binding()),
57///     FlowInspector::new(graph.binding(), selection.binding()).history(history.binding()),
58/// ));
59/// ```
60#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
61pub struct FlowInspector {
62    graph: Binding<FlowGraph>,
63    selection: Binding<FlowSelection>,
64    history: Option<Binding<FlowHistory>>,
65    guard: Option<FlowGuard>,
66    kinds: Rc<Vec<(String, String)>>,
67    read_only: bool,
68    width: f32,
69}
70
71impl fmt::Debug for FlowInspector {
72    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
73        f.debug_struct("FlowInspector").field("kinds", &self.kinds).finish_non_exhaustive()
74    }
75}

82            graph,
83            selection,
84            history: None,
85            guard: None,
86            kinds: Rc::new(Vec::new()),

88            width: 288.0,

92    /// Keeps its changes in this history, the canvas's, so Cmd/Ctrl+Z there undoes them.
93    pub fn history(mut self, history: Binding<FlowHistory>) -> Self {
94        self.history = Some(history);
95        self
96    }
97
98    /// Asks this before every change, as [`FlowCanvas::on_change`](crate::FlowCanvas::on_change)
99    /// does; a refusal leaves the diagram as it was and shows why under the fields.
100    pub fn on_change(
101        mut self,
102        guard: impl Fn(&FlowGraph, &FlowGraph, &FlowChange) -> Result<(), String> + 'static,
103    ) -> Self {
104        self.guard = Some(Rc::new(guard));
105        self
106    }
107
108    /// A kind a node may have, by its name in the diagram and the words shown for it. Without
109    /// any, the kind is not offered.
110    pub fn kind(mut self, kind: impl Into<String>, caption: impl Into<String>) -> Self {
111        Rc::make_mut(&mut self.kinds).push((kind.into(), caption.into()));
112        self
113    }
114
115    /// The kinds of `notation`, with their words.
116    pub fn notation(mut self, notation: &crate::FlowNotation) -> Self {
117        for (kind, caption) in &notation.kinds {
118            Rc::make_mut(&mut self.kinds).push((kind.clone(), caption.clone()));
119        }
120        self
121    }
122
123    /// Shows the properties without letting them change, as when the canvas is locked.
124    pub fn read_only(mut self, read_only: bool) -> Self {
125        self.read_only = read_only;
126        self
127    }
128
129    /// Another width, in logical pixels.
130    pub fn width(mut self, width: f32) -> Self {
131        self.width = width;
132        self
133    }

136/// How a change typed in the panel reaches the diagram.
137#[derive(Clone)]
138struct Writer {
139    graph: Binding<FlowGraph>,
140    history: Option<Binding<FlowHistory>>,
141    guard: Option<FlowGuard>,
142    notice: State<Option<String>>,
143}
144
145#[cfg_attr(feature = "live", ravenrs_live::expose)]
146impl Writer {
147    /// Writes `change` made by `edit`, when the app lets it, remembering the diagram before.
148    fn write(&self, nodes: &[String], edges: &[String], edit: impl FnOnce(&mut FlowGraph)) {
149        let before = self.graph.get();
150        let mut next = before.clone();
151        edit(&mut next);
152        if next == before {
153            return;
154        }
155        let change = FlowChange::Edit { nodes: nodes.to_vec(), edges: edges.to_vec() };
156        if let Some(guard) = &self.guard
157            && let Err(why) = guard(&before, &next, &change)
158        {
159            self.notice.set(Some(why));
160            return;
161        }
162        self.notice.set_if_changed(None);
163        if let Some(history) = &self.history {
164            history.update(|h| h.record(before));
165        }
166        self.graph.set(next);
167    }
168
169    /// A field bound to one value of every node in `ids`: read from the first, written to all.
170    fn nodes<T: Clone + PartialEq + Default + 'static>(
171        &self,
172        ids: Vec<String>,
173        read: impl Fn(&FlowNode) -> T + 'static,
174        write: impl Fn(&mut FlowNode, T) + 'static,
175    ) -> Binding<T> {
176        let (reader, writer, read_ids) = (self.graph.clone(), self.clone(), ids.clone());
177        Binding::new(
178            move || {
179                let g = reader.get();
180                read_ids.first().and_then(|id| g.find_node(id)).map(&read).unwrap_or_default()
181            },
182            move |value: T| {
183                writer.write(&ids, &[], |g| {
184                    for node in g.nodes.iter_mut().filter(|n| ids.contains(&n.id)) {
185                        write(node, value.clone());
186                    }
187                });
188            },
189        )
190    }
191
192    /// A field bound to one value of the connection `id`.
193    fn edge<T: Clone + Default + 'static>(
194        &self,
195        id: String,
196        read: impl Fn(&FlowEdge) -> T + 'static,
197        write: impl Fn(&mut FlowEdge, T) + 'static,
198    ) -> Binding<T> {
199        let (reader, writer, read_id) = (self.graph.clone(), self.clone(), id.clone());
200        Binding::new(
201            move || reader.get().find_edge(&read_id).map(&read).unwrap_or_default(),
202            move |value: T| {
203                writer.write(&[], std::slice::from_ref(&id), |g| {
204                    if let Some(edge) = g.edges.iter_mut().find(|e| e.id == id) {
205                        write(edge, value);
206                    }
207                });
208            },
209        )
210    }
211}

213/// A text field that keeps what is typed until Enter or leaving, then writes it.
214fn text_field(cx: &mut BuildCx<'_>, label: &str, value: Binding<String>, read_only: bool) -> NodeId {
215    let draft = cx.use_state(|| value.get());
216    let current = value.get();
217    // What the diagram holds wins when it changed elsewhere (undo, the canvas).
218    let seen = cx.use_state(|| current.clone());
219    if seen.get() != current {
220        seen.set(current.clone());
221        draft.set(current);
222    }
223    let (on_enter, on_leave) = ((value.clone(), draft.clone()), (value, draft.clone()));
224    let field = TextField::new(draft.binding())
225        .read_only(read_only)
226        .on_submit(move |text| on_enter.0.set(text))
227        .on_leave(move || on_leave.0.set(on_leave.1.get()));
228    cx.child(&FormField::new(label, field))
229}

231impl View for FlowInspector {
232    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
233        let theme = *cx.theme();
234        let labels = cx.labels();
235        let notice = cx.use_state(|| None::<String>);
236        let writer = Writer {
237            graph: self.graph.clone(),
238            history: self.history.clone(),
239            guard: self.guard.clone(),
240            notice: notice.clone(),
241        };
242        let graph = self.graph.get();
243        let selection = self.selection.get();
244        let nodes: Vec<String> =
245            selection.nodes.iter().filter(|id| graph.find_node(id).is_some()).cloned().collect();
246        let edges: Vec<String> =
247            selection.edges.iter().filter(|id| graph.find_edge(id).is_some()).cloned().collect();
248        let ro = self.read_only;
249        let muted = theme.color(Semantic::InkMuted);
250        let mut rows: Vec<NodeId> = Vec::new();
251        let heading = |cx: &mut BuildCx<'_>, words: &str, aside: Option<&str>| {
252            let title = util::label(cx, words, TextStyle::Caption, Some(Weight::SEMIBOLD), muted);
253            let mut parts = vec![title];
254            if let Some(aside) = aside {
255                parts.push(util::label(cx, aside, TextStyle::Caption, None, muted));
256            }
257            cx.container(
258                LayoutStyle {
259                    justify: ravenrs_layout::Justify::SpaceBetween,
260                    align: Align::Center,
261                    ..LayoutStyle::stack(Axis::Horizontal)
262                },
263                &parts,
264                Paint::None,
265            )
266        };
267
268        if nodes.is_empty() && edges.is_empty() {
269            rows.push(util::label(cx, &labels.flow_inspector_empty, TextStyle::Callout, None, muted));
270        } else if let ([only], []) = (nodes.as_slice(), edges.as_slice()) {
271            let node = graph.find_node(only).cloned().unwrap_or_default();
272            rows.push(heading(cx, &labels.flow_inspector_node, Some(&node.id)));
273            let ids = vec![only.clone()];
274            let title = writer.nodes(
275                ids.clone(),
276                |n| n.title.clone(),
277                |n, t: String| {
278                    if !t.trim().is_empty() {
279                        n.title = t.trim().to_owned();
280                    }
281                },
282            );
283            rows.push(cx.keyed_child(
284                util::hash_text(&format!("title{only}")),
285                &FieldView { label: labels.flow_field_title.clone(), value: title, read_only: ro },
286            ));
287            let line = writer.nodes(
288                ids.clone(),
289                |n| n.subtitle.clone().unwrap_or_default(),
290                |n, t: String| {
291                    n.subtitle = (!t.trim().is_empty()).then(|| t.trim().to_owned());
292                },
293            );
294            rows.push(cx.keyed_child(
295                util::hash_text(&format!("line{only}")),
296                &FieldView { label: labels.flow_field_line.clone(), value: line, read_only: ro },
297            ));
298            rows.extend(self.kind_and_shape(cx, &writer, ids.clone(), &graph, &labels, ro));
299            let number = |cx: &mut BuildCx<'_>,
300                          label: &str,
301                          read: fn(&FlowNode) -> f32,
302                          write: fn(&mut FlowNode, f32)| {
303                let value = writer.nodes(
304                    ids.clone(),
305                    move |n| Some(f64::from(read(n))),
306                    move |n, v: Option<f64>| {
307                        if let Some(v) = v {
308                            write(n, v as f32);
309                        }
310                    },
311                );
312                cx.child(&NumberField::new(value).label(label).disabled(ro))
313            };
314            let w = number(cx, &labels.flow_field_width, |n| n.width, |n, v| n.width = v.max(1.0));
315            let h = number(cx, &labels.flow_field_height, |n| n.height, |n, v| n.height = v.max(1.0));
316            let x = number(cx, &labels.flow_field_x, |n| n.x, |n, v| n.x = v);
317            let y = number(cx, &labels.flow_field_y, |n| n.y, |n, v| n.y = v);
318            rows.push(grid(cx, &[w, h, x, y], theme.spacing.s3));
319            let status_names = labels.flow_status_names.clone();
320            if ro {
321                let index = status_index(&node.status);
322                rows.push(util::label(cx, &status_names[index], TextStyle::Callout, None, muted));
323            } else {
324                let status = writer.nodes(
325                    ids.clone(),
326                    |n| status_index(&n.status),
327                    |n, i: usize| {
328                        n.status = match i {
329                            1 => FlowNodeStatus::Warning(status_words(&n.status)),
330                            2 => FlowNodeStatus::Error(status_words(&n.status)),
331                            3 => FlowNodeStatus::Disabled,
332                            _ => FlowNodeStatus::Normal,
333                        };
334                    },
335                );
336                rows.push(cx.child(
337                    &SegmentedControl::new(status, status_names).label(labels.flow_field_status.clone()),
338                ));
339            }
340            rows.extend(self.appearance(cx, &writer, ids.clone(), &labels, ro));
341        } else if let ([], [only]) = (nodes.as_slice(), edges.as_slice()) {
342            rows.push(heading(cx, &labels.flow_inspector_edge, Some(only)));
343            let label = writer.edge(
344                only.clone(),
345                |e| e.label.clone().unwrap_or_default(),
346                |e, t: String| {
347                    e.label = (!t.trim().is_empty()).then(|| t.trim().to_owned());
348                },
349            );
350            rows.push(cx.keyed_child(
351                util::hash_text(&format!("label{only}")),
352                &FieldView { label: labels.flow_field_label.clone(), value: label, read_only: ro },
353            ));
354            let shape = writer.edge(
355                only.clone(),
356                |e| LINES.iter().position(|s| *s == e.shape).unwrap_or(0),
357                |e, i: usize| e.shape = LINES[i.min(LINES.len() - 1)],
358            );
359            rows.push(
360                cx.child(
361                    &Select::new(shape, labels.flow_line_names.to_vec())
362                        .label(labels.flow_field_shape.clone())
363                        .disabled(ro),
364                ),
365            );
366            let cap = |cx: &mut BuildCx<'_>, label: &str, start: bool| {
367                let value = writer.edge(
368                    only.clone(),
369                    move |e| CAPS.iter().position(|c| *c == if start { e.start } else { e.end }).unwrap_or(0),
370                    move |e, i: usize| {
371                        let cap = CAPS[i.min(CAPS.len() - 1)];
372                        if start { e.start = cap } else { e.end = cap }
373                    },
374                );
375                cx.child(&Select::new(value, labels.flow_cap_names.to_vec()).label(label).disabled(ro))
376            };
377            let (start, end) =
378                (cap(cx, &labels.flow_field_start, true), cap(cx, &labels.flow_field_end, false));
379            rows.push(grid(cx, &[start, end], theme.spacing.s3));
380            let animated = writer.edge(only.clone(), |e| e.animated, |e, on: bool| e.animated = on);
381            rows.push(cx.child(&Checkbox::new(labels.flow_field_animated.clone(), animated).disabled(ro)));
382            let dashed = writer.edge(only.clone(), |e| e.dashed, |e, on: bool| e.dashed = on);
383            rows.push(cx.child(&Checkbox::new(labels.flow_field_dashed.clone(), dashed).disabled(ro)));
384            let color = writer.edge(
385                only.clone(),
386                |e| e.color.as_deref().and_then(parse_color),
387                |e, c: Option<Color>| e.color = c.map(color_hex),
388            );
389            rows.push(
390                cx.child(
391                    &ColorPicker::new(color)
392                        .label(labels.flow_field_stroke.clone())
393                        .width(self.width - 32.0)
394                        .disabled(ro),
395                ),
396            );
397        } else {
398            let count = labels.flow_inspector_nodes.replace("{0}", &nodes.len().to_string());
399            rows.push(heading(cx, &count, None));
400            rows.extend(self.kind_and_shape(cx, &writer, nodes.clone(), &graph, &labels, ro));
401            rows.extend(self.appearance(cx, &writer, nodes.clone(), &labels, ro));
402        }
403        if ro {
404            rows.push(util::label(cx, &labels.flow_inspector_read_only, TextStyle::Caption, None, muted));
405        }
406        if let Some(why) = notice.get() {
407            let words = util::label(cx, &why, TextStyle::Callout, None, theme.color(Semantic::Danger));
408            cx.semantics(words, Semantics::new(Role::Status).name(why));
409            rows.push(words);
410        }
411        let root = cx.container(
412            LayoutStyle {
413                width: Some(self.width),
414                padding: ravenrs_core::Insets::uniform(theme.spacing.s4),
415                spacing: theme.spacing.s3,
416                align: Align::Stretch,
417                ..LayoutStyle::stack(Axis::Vertical)
418            },
419            &rows,
420            Paint::Quad(crate::context::QuadPaint::new(theme.color(Semantic::Surface))),
421        );
422        cx.semantics(root, Semantics::new(Role::Group).name(labels.flow_inspector.clone()));
423        root
424    }
425}

428impl FlowInspector {
429    /// The kind and the shape of the nodes `ids`, shown as mixed where they differ.
430    /// The colors of the nodes in `ids` and their outline's width: each left empty is the
431    /// theme's.
432    fn appearance(
433        &self,
434        cx: &mut BuildCx<'_>,
435        writer: &Writer,
436        ids: Vec<String>,
437        labels: &crate::Labels,
438        ro: bool,
439    ) -> Vec<NodeId> {
440        let theme = *cx.theme();
441        let half = ((self.width - 32.0 - theme.spacing.s3) / 2.0).floor();
442        let color = |cx: &mut BuildCx<'_>,
443                     label: &str,
444                     read: fn(&FlowStyle) -> Option<Color>,
445                     write: fn(&mut FlowStyle, Option<String>)| {
446            let value = writer.nodes(
447                ids.clone(),
448                move |n| read(&n.style),
449                move |n, c: Option<Color>| write(&mut n.style, c.map(color_hex)),
450            );
451            cx.child(&ColorPicker::new(value).label(label).width(half).disabled(ro))
452        };
453        let fill = color(cx, &labels.flow_field_fill, FlowStyle::fill_color, |s, c| s.fill = c);
454        let stroke = color(cx, &labels.flow_field_stroke, FlowStyle::stroke_color, |s, c| s.stroke = c);
455        let text = color(cx, &labels.flow_field_text, FlowStyle::text_color, |s, c| s.text = c);
456        let width = writer.nodes(
457            ids.clone(),
458            |n| n.style.stroke_width.map(f64::from),
459            |n, v: Option<f64>| n.style.stroke_width = v.map(|v| (v as f32).clamp(0.5, 12.0)),
460        );
461        let width =
462            cx.child(&NumberField::new(width).label(labels.flow_field_stroke_width.as_str()).disabled(ro));
463        vec![grid(cx, &[fill, stroke, text, width], theme.spacing.s3)]
464    }
465
466    fn kind_and_shape(
467        &self,
468        cx: &mut BuildCx<'_>,
469        writer: &Writer,
470        ids: Vec<String>,
471        graph: &FlowGraph,
472        labels: &crate::Labels,
473        ro: bool,
474    ) -> Vec<NodeId> {
475        let chosen: Vec<&FlowNode> = ids.iter().filter_map(|id| graph.find_node(id)).collect();
476        let mut out = Vec::new();
477        // A first option "Mixed" stands for values that differ, until one is chosen.
478        let mixed = |same: bool, names: Vec<String>| {
479            if same {
480                (0, names)
481            } else {
482                (1, std::iter::once(labels.flow_inspector_mixed.clone()).chain(names).collect())
483            }
484        };
485        if !self.kinds.is_empty() {
486            let kinds = self.kinds.clone();
487            let same = chosen.windows(2).all(|w| w[0].kind == w[1].kind);
488            let (offset, names) = mixed(same, kinds.iter().map(|(_, c)| c.clone()).collect());
489            let (read_kinds, write_kinds) = (kinds.clone(), kinds);
490            let value = writer.nodes(
491                ids.clone(),
492                move |n| {
493                    if offset == 1 {
494                        0
495                    } else {
496                        read_kinds.iter().position(|(k, _)| *k == n.kind).unwrap_or(0)
497                    }
498                },
499                move |n, i: usize| {
500                    if let Some((kind, _)) = i.checked_sub(offset).and_then(|i| write_kinds.get(i)) {
501                        n.kind = kind.clone();
502                    }
503                },
504            );
505            out.push(cx.child(&Select::new(value, names).label(labels.flow_field_kind.clone()).disabled(ro)));
506        }
507        let same = chosen.windows(2).all(|w| w[0].shape == w[1].shape);
508        let (offset, names) = mixed(same, labels.flow_shape_names.to_vec());
509        let value = writer.nodes(
510            ids,
511            move |n| if offset == 1 { 0 } else { SHAPES.iter().position(|s| *s == n.shape).unwrap_or(0) },
512            move |n, i: usize| {
513                if let Some(shape) = i.checked_sub(offset).and_then(|i| SHAPES.get(i)) {
514                    n.shape = *shape;
515                }
516            },
517        );
518        out.push(cx.child(&Select::new(value, names).label(labels.flow_field_shape.clone()).disabled(ro)));
519        out
520    }
521}

536/// Two fields to a row.
537fn grid(cx: &mut BuildCx<'_>, cells: &[NodeId], gap: f32) -> NodeId {
538    let rows: Vec<NodeId> = cells
539        .chunks(2)
540        .map(|pair| {
541            let cells: Vec<NodeId> = pair
542                .iter()
543                .map(|&c| {
544                    cx.container(
545                        LayoutStyle { grow: 1.0, ..LayoutStyle::stack(Axis::Vertical) },
546                        &[c],
547                        Paint::None,
548                    )
549                })
550                .collect();
551            cx.container(
552                LayoutStyle { spacing: gap, ..LayoutStyle::stack(Axis::Horizontal) },
553                &cells,
554                Paint::None,
555            )
556        })
557        .collect();
558    cx.container(
559        LayoutStyle { spacing: gap, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
560        &rows,
561        Paint::None,
562    )
563}

565fn status_index(status: &FlowNodeStatus) -> usize {
566    match status {
567        FlowNodeStatus::Warning(_) => 1,
568        FlowNodeStatus::Error(_) => 2,
569        FlowNodeStatus::Disabled => 3,
570        _ => 0,
571    }

574/// The words a status keeps when it changes between warning and error.
