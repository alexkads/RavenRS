53/// An error answered to the client. Internal details go to the log, not the answer.
54enum ApiError {
55    BadRequest(String),
56    NotFound,
57    Conflict(String),
58    Internal(String),
59}
60
61impl From<io::Error> for ApiError {
62    fn from(e: io::Error) -> Self {
63        if e.kind() == io::ErrorKind::InvalidInput {
64            Self::BadRequest(e.to_string())
65        } else {
66            Self::Internal(e.to_string())
67        }
68    }
69}
70
71impl IntoResponse for ApiError {
72    fn into_response(self) -> Response {
73        let (status, message) = match self {
74            Self::BadRequest(message) => (StatusCode::BAD_REQUEST, message),
75            Self::NotFound => (StatusCode::NOT_FOUND, "not found".to_owned()),
76            Self::Conflict(message) => (StatusCode::CONFLICT, message),
77            Self::Internal(detail) => {
78                tracing::error!(error = %detail, "request failed");
79                (StatusCode::INTERNAL_SERVER_ERROR, "internal error".to_owned())
80            }
81        };
82        (status, Json(json!({ "error": message }))).into_response()
83    }
84}

86/// The API's routes.
87pub fn router(state: AppState) -> Router {
88    let limit = state.config.max_body_megabytes.saturating_mul(1024 * 1024);
89    let protected = Router::new()
90        .route("/render", post(render_now))
91        .route("/layouts", get(list_layouts))
92        .route("/layouts/{name}", get(get_layout).put(put_layout).delete(delete_layout))
93        .route("/layouts/{name}/versions", get(layout_versions))
94        .route("/jobs", get(list_jobs))
95        .route("/jobs/{id}", get(get_job).put(put_job).delete(delete_job))
96        .route("/jobs/{id}/run", post(run_job))
97        .route("/jobs/{id}/runs", get(list_runs))
98        .route("/jobs/{id}/runs/{run}", get(get_run))
99        .route("/jobs/{id}/runs/{run}/documents/{file}", get(get_document))
100        .route_layer(middleware::from_fn_with_state(state.clone(), require_token));
101    Router::new()
102        .route("/health", get(|| async { Json(json!({ "status": "ok" })) }))
103        .merge(protected)
104        .layer(DefaultBodyLimit::max(limit))
105        .with_state(state)
106}

108/// Compares without stopping at the first difference, so the time an answer
109/// takes says nothing about how much of a guessed token was right.
110fn same_secret(a: &[u8], b: &[u8]) -> bool {
111    if a.len() != b.len() {
112        return false;
113    }
114    a.iter().zip(b).fold(0_u8, |difference, (x, y)| difference | (x ^ y)) == 0
115}

117async fn require_token(
118    State(state): State<AppState>,
119    headers: HeaderMap,
120    request: Request,
121    next: Next,
122) -> Response {
123    let presented = headers
124        .get(header::AUTHORIZATION)
125        .and_then(|value| value.to_str().ok())
126        .and_then(|value| value.strip_prefix("Bearer "));
127    if presented.is_some_and(|token| same_secret(token.as_bytes(), state.token.as_bytes())) {
128        return next.run(request).await;
129    }
130    tracing::warn!(path = %request.uri().path(), "request without a valid token");
131    (
132        StatusCode::UNAUTHORIZED,
133        [(header::WWW_AUTHENTICATE, "Bearer")],
134        Json(json!({ "error": "a valid API token is required" })),
135    )
136        .into_response()
137}

198/// A request to render now.
199#[derive(Deserialize)]
200#[serde(rename_all = "camelCase", deny_unknown_fields)]
201struct RenderRequest {
202    /// A stored layout's name.
203    name: Option<String>,
204    /// A version of it; the latest by default.
205    version: Option<u32>,
206    /// A layout sent with the request, instead of a stored one.
207    layout: Option<JsonValue>,
208    /// Data sent with the request, instead of the layout's data source.
209    data: Option<JsonValue>,
210    #[serde(default)]
211    parameters: BTreeMap<String, ParameterValue>,
212    format: Format,
213    /// The time zone of `Today()`.
214    timezone: Option<String>,
215}
216
217async fn render_now(State(state): State<AppState>, body: Bytes) -> Result<Response, ApiError> {
218    let request: RenderRequest =
219        serde_json::from_slice(&body).map_err(|e| ApiError::BadRequest(format!("bad request: {e}")))?;
220    let report = match (&request.name, &request.layout) {
221        (Some(name), None) => {
222            let (_, text) = state.store.layout(name, request.version)?.ok_or(ApiError::NotFound)?;
223            parse_report(&text)?
224        }
225        (None, Some(layout)) => serde_json::from_value(layout.clone())
226            .map_err(|e| ApiError::BadRequest(format!("not a valid layout: {e}")))?,
227        _ => return Err(ApiError::BadRequest("send either 'name' or 'layout'".to_owned())),
228    };
229    let zone = TimeZone::get(request.timezone.as_deref().unwrap_or("UTC"))
230        .map_err(|_| ApiError::BadRequest("unknown time zone".to_owned()))?;
231    let local = Timestamp::now().to_zoned(zone);
232    let now = local_date_time(&local).ok_or_else(|| ApiError::Internal("date out of range".to_owned()))?;
233    let date = local.strftime("%Y-%m-%d").to_string();
234    let parameters = parameter_values(&request.parameters, now).map_err(ApiError::BadRequest)?;
235    let format = request.format;
236    let config = state.config.clone();
237    let permit =
238        state.runner.permits().acquire_owned().await.map_err(|e| ApiError::Internal(e.to_string()))?;
239    let rendered = tokio::task::spawn_blocking(move || {
240        let _permit = permit;
241        let data = match request.data {
242            Some(data) => data,
243            None => resolve_data(&report, &parameters, &config)?,
244        };
245        render(&report, &data, parameters, now, &[format], &date)
246    })
247    .await
248    .map_err(|e| ApiError::Internal(e.to_string()))?
249    .map_err(ApiError::BadRequest)?;
250    let warnings = rendered.warnings.len();
251    let file = rendered.files.into_iter().next().ok_or_else(|| ApiError::Internal("no file".to_owned()))?;
252    Ok(file_response(&file.name, file.mime, file.bytes, Some(warnings)))
253}

255/// A file as a download. The name is sent in both of the header's forms, so
256/// neither a quote nor an accent in a title can break it.
257fn file_response(name: &str, mime: &str, bytes: Vec<u8>, warnings: Option<usize>) -> Response {
258    let ascii: String =
259        name.chars().map(|c| if c.is_ascii_alphanumeric() || " .-_".contains(c) { c } else { '_' }).collect();
260    let encoded: String = name
261        .bytes()
262        .map(|b| {
263            if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_') {
264                (b as char).to_string()
265            } else {
266                format!("%{b:02X}")
267            }
268        })
269        .collect();
270    let disposition = format!("attachment; filename=\"{ascii}\"; filename*=UTF-8''{encoded}");
271    let mut response = bytes.into_response();
272    let headers = response.headers_mut();
273    if let Ok(value) = HeaderValue::from_str(mime) {
274        headers.insert(header::CONTENT_TYPE, value);
275    }
276    if let Ok(value) = HeaderValue::from_str(&disposition) {
277        headers.insert(header::CONTENT_DISPOSITION, value);
278    }
279    headers.insert(header::X_CONTENT_TYPE_OPTIONS, HeaderValue::from_static("nosniff"));
280    if let Some(warnings) = warnings {
281        headers.insert("x-report-warnings", HeaderValue::from(warnings));
282    }
283    response
284}

286async fn list_jobs(State(state): State<AppState>) -> Result<Json<Vec<JobSummary>>, ApiError> {
287    let mut jobs = Vec::new();
288    for id in state.store.job_ids()? {
289        if let Some(job) = state.store.job(&id)? {
290            let next_run = state.store.job_state(&id)?.next_run;
291            jobs.push(JobSummary { id, job, next_run });
292        }
293    }
294    Ok(Json(jobs))
295}

306async fn put_job(
307    State(state): State<AppState>,
308    Path(id): Path<String>,
309    body: Bytes,
310) -> Result<Json<JsonValue>, ApiError> {
311    if !valid_name(&id) {
312        return Err(ApiError::BadRequest(format!("'{id}' is not a valid name")));
313    }
314    let job: Job =
315        serde_json::from_slice(&body).map_err(|e| ApiError::BadRequest(format!("bad job: {e}")))?;
316    job.validate().map_err(ApiError::BadRequest)?;
317    if state.store.layout(&job.layout, None)?.is_none() {
318        return Err(ApiError::BadRequest(format!("the layout '{}' does not exist", job.layout)));
319    }
320    for delivery in &job.deliveries {
321        match delivery {
322            Delivery::Email { .. } if state.config.smtp.is_none() => {
323                return Err(ApiError::BadRequest("the server has no mail server configured".to_owned()));
324            }
325            Delivery::Folder { .. } if state.config.output_dir.is_none() => {
326                return Err(ApiError::BadRequest("the server has no outputDir configured".to_owned()));
327            }
328            _ => {}
329        }
330    }
331    state.store.save_job(&id, &job)?;
332    let schedule = Schedule::parse(&job.schedule).map_err(ApiError::BadRequest)?;
333    let zone = TimeZone::get(&job.timezone).map_err(|e| ApiError::BadRequest(e.to_string()))?;
334    let next_run = schedule.next_after(&Timestamp::now().to_zoned(zone)).map(|z| z.timestamp());
335    state.store.save_job_state(&id, &crate::store::JobState { next_run })?;
336    Ok(Json(json!({ "id": id, "nextRun": next_run })))
337}

343async fn run_job(State(state): State<AppState>, Path(id): Path<String>) -> Result<Response, ApiError> {
344    let job = state.store.job(&id)?.ok_or(ApiError::NotFound)?;
345    let runner = state.runner.clone();
346    let job_id = id.clone();
347    tokio::spawn(async move { runner.run_now(&job_id, &job).await });
348    Ok((StatusCode::ACCEPTED, Json(json!({ "job": id, "status": "started" }))).into_response())
349}

376async fn get_document(
377    State(state): State<AppState>,
378    Path((id, run, file)): Path<(String, String, String)>,
379) -> Result<Response, ApiError> {
380    let bytes = state.store.document(&id, &run, &file)?.ok_or(ApiError::NotFound)?;
381    let format = match file.rsplit('.').next() {
382        Some("pdf") => Format::Pdf,
383        Some("xlsx") => Format::Xlsx,
384        _ => Format::Csv,
385    };
386    Ok(file_response(&file, format.mime(), bytes, None))
387}
