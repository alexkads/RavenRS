21/// What is sold as a service, with an NFS-e of the national standard.
22pub fn servico_entity() -> EntityInfo {
23    let text = |name: &str, caption: &str| FieldInfo::new(name, FieldKind::Text).caption(caption);
24    EntityInfo::new(SERVICO)
25        .caption("Serviços")
26        .navigation("Cadastros")
27        .icon("wrench")
28        .field(FieldInfo::new("id", FieldKind::Guid).key())
29        .field(text("descricao", "Descrição").default_property())
30        .field(text("codigo", "Código interno"))
31        .field(text("codigo_tributacao_nacional", "Código de tributação nacional (cTribNac)"))
32        .field(FieldInfo::new("preco", FieldKind::Money).caption("Preço").default_value(0.0))
33        .rule(Rule::required("descricao"))
34        .rule(Rule::required("codigo"))
35        .rule(Rule::unique("codigo"))
36        .rule(
37            Rule::matches("codigo_tributacao_nacional", "^[0-9]{6}$")
38                .message("O código de tributação nacional tem 6 dígitos: item, subitem e desdobro."),
39        )
40}

42/// What is sold.
43pub fn produto_entity() -> EntityInfo {
44    let text = |name: &str, caption: &str| FieldInfo::new(name, FieldKind::Text).caption(caption);
45    EntityInfo::new(PRODUTO)
46        .caption("Produtos")
47        .navigation("Cadastros")
48        .icon("package")
49        .field(FieldInfo::new("id", FieldKind::Guid).key())
50        .field(text("descricao", "Descrição").default_property())
51        .field(text("codigo", "Código interno"))
52        .field(text("gtin", "Código de barras (GTIN)").editor("Gtin"))
53        .field(FieldInfo::new("preco", FieldKind::Money).caption("Preço de venda").default_value(0.0))
54        .field(FieldInfo::new("custo", FieldKind::Decimal(4)).caption("Custo médio").default_value(0.0))
55        .field(FieldInfo::new("estoque", FieldKind::Decimal(3)).caption("Estoque").default_value(0.0))
56        .field(text("unidade", "Unidade").default_value("UN"))
57        .field(text("ncm", "NCM"))
58        .field(text("cfop", "CFOP").default_value("5102"))
59        .field(text("cest", "CEST"))
60        .field(
61            FieldInfo::new(
62                "csosn",
63                FieldKind::Enum(vec![
64                    EnumOption::new("102", "102, sem crédito"),
65                    EnumOption::new("500", "500, ICMS por substituição"),
66                ]),
67            )
68            .caption("CSOSN")
69            .default_value("102"),
70        )
71        .rule(Rule::required("descricao"))
72        .rule(Rule::required("codigo"))
73        .rule(Rule::unique("codigo"))
74        .rule(Rule::matches("ncm", "^[0-9]{8}$").message("O NCM tem 8 dígitos."))
75        .rule(Rule::matches("cfop", "^[0-9]{4}$").message("O CFOP tem 4 dígitos."))
76}
