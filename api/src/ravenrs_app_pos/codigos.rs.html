1//! A product can answer to many barcodes: variations of the same goods, or a pack that counts as many.
2
3use ravenrs_app::{FieldValue, Object, ObjectSpace, Query, SpaceError, Value};
4use ravenrs_pos::conferir_codigo;
5use rust_decimal::Decimal;
6
7use crate::caixa::texto;
8use crate::modelo::{CODIGO, PRODUTO};

10/// What a code read at the counter leads to.
11#[derive(Clone, Debug, PartialEq)]
12pub struct Encontrado {
13    /// The product.
14    pub produto: Object,
15    /// How many units it counts as.
16    pub fator: Decimal,
17}
18
19/// The product of a code: its own barcode, its internal code or any of its other barcodes.
20pub fn produto_por_codigo(space: &dyn ObjectSpace, codigo: &str) -> Option<Encontrado> {
21    let produtos = space.query(PRODUTO, &Query::all()).ok()?;
22    let proprio = produtos
23        .iter()
24        .find(|produto| texto(produto, "gtin") == codigo || texto(produto, "codigo") == codigo);
25    if let Some(produto) = proprio {
26        return Some(Encontrado { produto: produto.clone(), fator: Decimal::ONE });
27    }
28    let codigos = space.query(CODIGO, &Query::all()).ok()?;
29    let achado = codigos.iter().find(|linha| texto(linha, "codigo") == codigo)?;
30    let dono = texto(achado, "produto");
31    let produto = produtos.into_iter().find(|produto| texto(produto, "codigo") == dono)?;
32    let fator = Decimal::from_value(achado.get("fator"));
33    Some(Encontrado { produto, fator: if fator > Decimal::ZERO { fator } else { Decimal::ONE } })
34}

36/// Gives a product one more barcode, in the unit of work. Says so when the code cannot be used or
37/// already belongs to another product. Adding one it already has changes nothing.
38pub fn cadastrar_codigo(
39    space: &mut dyn ObjectSpace,
40    produto: &str,
41    codigo: &str,
42    descricao: &str,
43    fator: Decimal,
44) -> Result<(), SpaceError> {
45    let codigo = conferir_codigo(codigo).map_err(SpaceError::Conflict)?;
46    if fator <= Decimal::ZERO {
47        return Err(SpaceError::Conflict("O código precisa valer ao menos uma unidade.".to_owned()));
48    }
49    if let Some(encontrado) = produto_por_codigo(&*space, &codigo) {
50        let dono = texto(&encontrado.produto, "codigo");
51        if dono == produto {
52            return Ok(());
53        }
54        return Err(SpaceError::Conflict(format!("O código {codigo} já é do produto {dono}.")));
55    }
56    let mut linha = space.new_object(CODIGO)?;
57    linha.set("produto", produto.to_owned());
58    linha.set("codigo", codigo);
59    linha.set("descricao", descricao.to_owned());
60    linha.set("fator", Decimal::to_value(&fator));
61    space.save(linha)
62}

64/// The other barcodes of a product.
65pub fn codigos_do_produto(space: &dyn ObjectSpace, produto: &str) -> Vec<(String, Decimal, Value)> {
66    space
67        .query(CODIGO, &Query::all())
68        .unwrap_or_default()
69        .iter()
70        .filter(|linha| texto(linha, "produto") == produto)
71        .map(|linha| (texto(linha, "codigo"), Decimal::from_value(linha.get("fator")), linha.get("id")))
72        .collect()
73}
