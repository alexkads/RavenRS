15/// What the checkout needs from a payment provider, whichever it is.
16pub trait Tef {
17    /// Starts charging.
18    fn iniciar(&self, pedido: PedidoPagamento) -> Result<Passo, String>;
19    /// Goes on with what the operator answered.
20    fn continuar(&self, id: &IdPagamento, entrada: EntradaOperador) -> Result<Passo, String>;
21    /// Says whether the sale finished: confirms the approval, or undoes it.
22    fn concluir(&self, id: &IdPagamento, impressao: Impressao) -> Result<Estado, String>;
23    /// Settles what an earlier session left open.
24    fn recuperar(&self) -> Result<Vec<Recuperacao>, String>;
25}
26
27impl<P: ProvedorPagamento, D: Diario> Tef for RefCell<Pagamentos<P, D>> {
28    fn iniciar(&self, pedido: PedidoPagamento) -> Result<Passo, String> {
29        self.borrow_mut().iniciar(pedido).map_err(|error| error.to_string())
30    }
31
32    fn continuar(&self, id: &IdPagamento, entrada: EntradaOperador) -> Result<Passo, String> {
33        self.borrow_mut().continuar(id, entrada).map_err(|error| error.to_string())
34    }
35
36    fn concluir(&self, id: &IdPagamento, impressao: Impressao) -> Result<Estado, String> {
37        self.borrow_mut().concluir(id, impressao).map_err(|error| error.to_string())
38    }
39
40    fn recuperar(&self) -> Result<Vec<Recuperacao>, String> {
41        self.borrow_mut().recuperar().map_err(|error| error.to_string())
42    }
43}

45/// The provider and its simulator, for a checkout that has no contract yet: nothing is charged.
46pub fn tef_simulado() -> (Rc<dyn Tef>, SimuladorProvedor) {
47    let simulador = SimuladorProvedor::new();
48    let pagamentos = match Pagamentos::new(simulador.clone(), DiarioMemoria::new()) {
49        Ok(pagamentos) => pagamentos,
50        // A journal in memory cannot fail to open; should it, the checkout still sells in cash.
51        Err(_) => return (Rc::new(SemTef), simulador),
52    };
53    (Rc::new(RefCell::new(pagamentos)), simulador)
54}

64        Err("Não há provedor de cartão configurado.".to_owned())

66    fn concluir(&self, _id: &IdPagamento, _impressao: Impressao) -> Result<Estado, String> {
67        Err("Não há provedor de cartão configurado.".to_owned())
68    }

70        Ok(Vec::new())

74/// How a way of paying goes to the provider, or `None` for cash, which never does.
75pub fn tipo_do_provedor(forma: Forma) -> Option<TipoPagamento> {
76    match forma {
77        Forma::Dinheiro => None,
78        Forma::Debito => Some(TipoPagamento::Debito),
79        Forma::Credito => Some(TipoPagamento::Credito { parcelas: 1 }),
80        Forma::Pix => Some(TipoPagamento::Pix),
81    }
82}

84/// The card group of the note, from what the provider approved.
85pub fn cartao_da_aprovacao(dados: &DadosFiscaisPagamento) -> Cartao {
86    let mut cartao = Cartao::nao_integrado();
87    if dados.tp_integra == TpIntegra::Integrado {
88        cartao.integracao = TipoIntegracao::Integrado;
89    }
90    cartao.cnpj = dados.cnpj.clone();
91    cartao.bandeira = dados.t_band.clone();
92    cartao.autorizacao = dados.c_aut.clone();
93    cartao.cnpj_recebedor = dados.cnpj_receb.clone();
94    cartao.terminal = dados.id_term_pag.clone();
95    cartao
96}

98/// A new id for a payment, with nothing personal in it.
99pub fn novo_id(numero_venda: u64) -> Result<IdPagamento, String> {
100    let agora = std::time::SystemTime::now()
101        .duration_since(std::time::UNIX_EPOCH)
102        .map_or(0, |duration| duration.as_millis());
103    IdPagamento::new(format!("pdv-{numero_venda}-{agora}")).map_err(|error| format!("{error:?}"))
104}
