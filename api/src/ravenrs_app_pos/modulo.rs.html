47/// A question in a window, answered with a line of text.
48fn perguntar(
49    session: &Session,
50    titulo: &'static str,
51    atual: String,
52    responder: impl Fn(String) + Clone + 'static,
53) {
54    let fechar = session.clone();
55    let valor = ravenrs_reactive::State::new(atual);
56    session.present(titulo, move || {
57        let (read, write) = (valor.clone(), valor.clone());
58        let (responder, fechar) = (responder.clone(), fechar.clone());
59        Box::new(
60            TextField::new(ravenrs_reactive::Binding::new(
61                move || read.get(),
62                move |text: String| write.set(text),
63            ))
64            .autofocus()
65            .on_submit(move |text: String| {
66                responder(text.trim().to_owned());
67                fechar.close_presented();
68            }),
69        ) as AnyView
70    });
71}

73/// The window with the rest of what can be done.
74pub(crate) fn abrir_outras_funcoes(session: &Session) {
75    let opcoes: [(&str, &str); 8] = [
76        ("Notas emitidas", "NfeDocumento_ListView"),
77        ("NFS-e emitidas", "NfseDocumento_ListView"),
78        ("Serviços", "PosServico_ListView"),
79        ("Emitente", "NfeEmitente_ListView"),
80        ("Modelos de cupom e etiqueta", "ReportDefinition_ListView"),
81        ("Clientes", "PosCliente_ListView"),
82        ("Periféricos e drivers", VIEW_PERIFERICOS),
83        ("Configuração do caixa", "PosConfiguracao_ListView"),
84    ];
85    let aberta = session.clone();
86    session.present("Outras funções", move || {
87        let botoes: Vec<AnyView> = opcoes
88            .iter()
89            .map(|(nome, view)| {
90                let (session, view) = (aberta.clone(), *view);
91                Box::new(Button::new(*nome).full_width().on_click(move || {
92                    session.close_presented();
93                    session.open_view(view);
94                })) as AnyView
95            })
96            .collect();
97        Box::new(VStack::new(botoes).spacing(8.0))
98    });
99}

106impl WindowController for Sempre {
107    fn id(&self) -> &str {
108        "PosSempre"
109    }
110
111    fn actions(&self) -> Vec<Action> {
112        let caixa = self.caixa.clone();
113        vec![
114            Action::simple("Pos.Atalhos", "Atalhos").hotkey("f1").category(ActionCategory::View).on_execute(
115                |cx| {
116                    sessao(cx)?.show_shortcuts();
117                    Ok(())
118                },
119            ),
120            Action::simple("Pos.Voltar", "Início")
121                .hotkey("esc")
122                .category(ActionCategory::Navigation)
123                .on_execute(move |cx| {
124                    let session = sessao(cx)?;
125                    if matches!(*caixa.tela.peek(), crate::Tela::Cobrando { .. }) {
126                        caixa.responder(&session, ravenrs_pagamento::EntradaOperador::Cancelar);
127                    } else if !caixa.voltar() {
128                        session.open_view(VIEW_INICIO);
129                    }
130                    Ok(())
131                }),
132            Action::simple("Pos.Bloquear", "Bloquear")
133                .hotkey("f12")
134                .category(ActionCategory::View)
135                .on_execute(|cx| {
136                    let session = sessao(cx)?;
137                    let pin = configuracao(&session, "pin");
138                    session.lock("Caixa bloqueado. Digite o PIN.", move |typed| typed == pin);
139                    Ok(())
140                }),
141        ]
142    }
143}

207impl ViewController for NoCaixa {
208    fn id(&self) -> &str {
209        "PosCaixa"
210    }
211
212    fn actions(&self) -> Vec<Action> {
213        let grande = self.caixa.clone();
214        let (item, venda, cliente, nota, pendentes, finalizar) = (
215            self.caixa.clone(),
216            self.caixa.clone(),
217            self.caixa.clone(),
218            self.caixa.clone(),
219            self.caixa.clone(),
220            self.caixa.clone(),
221        );
222        let no_caixa = |id: &str, nome: &str, tecla: &str| {
223            Action::simple(id, nome).hotkey(tecla).target_view(VIEW_CAIXA).category(ActionCategory::Edit)
224        };
225        vec![
226            no_caixa("Pos.CancelarItem", "Cancelar item", "f6").on_execute(move |_| {
227                item.cancelar_ultimo();
228                Ok(())
229            }),
230            no_caixa("Pos.Cliente", "Cliente", "alt+c").on_execute(move |cx| {
231                let session = sessao(cx)?;
232                let atual = cliente.carrinho.peek().cliente.clone().unwrap_or_default();
233                let caixa = cliente.clone();
234                perguntar(&session, "CPF ou CNPJ do cliente", atual, move |documento| {
235                    let digitos: String = documento.chars().filter(char::is_ascii_alphanumeric).collect();
236                    caixa.carrinho.update(|carrinho| {
237                        carrinho.cliente = (!digitos.is_empty()).then_some(digitos);
238                    });
239                });
240                Ok(())
241            }),
242            no_caixa("Pos.NotaGrande", "Nota grande", "f8").on_execute(move |_| {
243                grande.nota_grande.update(|ligada| *ligada = !*ligada);
244                Ok(())
245            }),
246            no_caixa("Pos.Observacao", "Observação", "ctrl+o").on_execute(move |cx| {
247                let session = sessao(cx)?;
248                let atual = nota.carrinho.peek().observacao.clone().unwrap_or_default();
249                let caixa = nota.clone();
250                perguntar(&session, "Observação do documento", atual, move |texto| {
251                    caixa
252                        .carrinho
253                        .update(|carrinho| carrinho.observacao = (!texto.is_empty()).then_some(texto));
254                });
255                Ok(())
256            }),
257            no_caixa("Pos.Pendentes", "Pendentes", "f10").on_execute(move |cx| {
258                let session = sessao(cx)?;
259                abrir_pendentes(&session, &pendentes);
260                Ok(())
261            }),
262            no_caixa("Pos.CancelarVenda", "Cancelar venda", "f11").on_execute(move |_| {
263                venda.cancelar_venda();
264                Ok(())
265            }),
266            Action::simple("Pos.Finalizar", "Finalizar")
267                .hotkey("f9")
268                .target_view(VIEW_CAIXA)
269                .category(ActionCategory::Save)
270                .on_execute(move |cx| {
271                    finalizar.avancar(&sessao(cx)?);
272                    Ok(())
273                }),
274        ]

278/// The sales kept: guard this one, or pick one up.
279fn abrir_pendentes(session: &Session, caixa: &Caixa) {
280    let (aberta, caixa) = (session.clone(), caixa.clone());
281    session.present("Vendas pendentes", move || {
282        let mut linhas: Vec<AnyView> = Vec::new();
283        if !caixa.carrinho.peek().vazio() {
284            let (c, s) = (caixa.clone(), aberta.clone());
285            linhas.push(Box::new(Button::new("Guardar a venda aberta").full_width().on_click(move || {
286                s.close_presented();
287                if let Err(why) = c.guardar(&s) {
288                    s.show_message(Message::new(why, MessageKind::Warning));
289                }
290            })));
291        }
292        let kept = caixa.pendentes(&aberta);
293        if kept.is_empty() {
294            linhas.push(Box::new(Text::new("Nenhuma venda guardada.").style(TextStyle::Callout)));
295        }
296        for (id, titulo) in kept {
297            let (c, s) = (caixa.clone(), aberta.clone());
298            linhas.push(Box::new(Button::new(titulo).full_width().on_click(move || {
299                s.close_presented();
300                if let Err(why) = c.retomar(&s, &id) {
301                    s.show_message(Message::new(why, MessageKind::Warning));
302                }
303            })));
304        }
305        Box::new(VStack::new(linhas).spacing(8.0))
306    });
307}

411    /// Adds it all to an application: the records, the screens, the look of a counter and a window
412    /// that opens on its first screen.
413    pub fn instalar(&self, app: App) -> App {
414        let mut relatorios = StoredReportsModule::new(ReportWords::pt_br()).predefined(
415            "Gôndola 60 x 40 mm",
416            PRODUTO,
417            etiqueta_padrao(),
418        );
419        for (nome, modelo) in modelos_de_cupom().into_iter().chain(modelos_de_danfe()) {
420            relatorios = relatorios.predefined(nome, CUPOM_LINHA, modelo);
421        }
422        for (nome, modelo) in ravenrs_app_nfe::modelos_de_danfse() {
423            relatorios = relatorios.predefined(nome, ravenrs_app_nfe::NFSE_LINHA, modelo);
424        }
425        let (caixa, palavras) = (self.caixa.clone(), Rc::new(ReportWords::pt_br()));
426        let entradas = self.entradas.clone();
427        let servicos = self.servicos.clone();
428        let sessao_guardada = self.sessao.clone();
429        app.language(Language::BrazilianPortuguese)
430            .theme(Theme::raven_light())
431            .counter()
432            .module(AppRecordsModule)
433            .module(relatorios)
434            .module(NfeModule::new(self.caixa.service.clone()))
435            .module(PosModule { caixa: self.caixa.clone() })
436            .dashboard(DashboardInfo::new("PosInicio", "Início").control("Início", "pos.inicio"))
437            .dashboard(DashboardInfo::new("PosCaixa", "Caixa").control("Caixa", "pos.caixa"))
438            .dashboard(DashboardInfo::new("PosEtiquetas", "Etiquetas").control("Etiquetas", "pos.etiquetas"))
439            .dashboard(DashboardInfo::new("PosEntrada", "Entrada").control("Entrada", "pos.entrada"))
440            .dashboard(DashboardInfo::new("PosPreco", "Preços").control("Preços", "pos.preco"))
441            .dashboard(
442                DashboardInfo::new("PosPerifericos", "Periféricos").control("Periféricos", "pos.perifericos"),
443            )
444            .dashboard(
445                DashboardInfo::new("PosVendaServico", "Serviços").control("Serviços", "pos.servico"),
446            )
447            .with_session(move |session| {
448                *sessao_guardada.borrow_mut() = Some(session.clone());
449                for aviso in caixa.recuperar_pagamentos() {
450                    session.show_message(Message::new(aviso, MessageKind::Warning));
451                }
452                let sem_resposta = ravenrs_app_nfe::nfses_pendentes(&*session.frame().space().borrow()).len();
453                if sem_resposta > 0 {
454                    session.show_message(Message::new(
455                        format!(
456                            "{sem_resposta} DPS de NFS-e sem resposta certa. Em Vender serviço (F8), envie de novo: vai a mesma DPS."
457                        ),
458                        MessageKind::Warning,
459                    ));
460                }
461                let aqui = session.clone();
462                session.register_control("pos.inicio", move || {
463                    Box::new(TelaInicio { session: aqui.clone() }.grow(1.0)) as AnyView
464                });
465                let (aqui, caixa_aqui) = (session.clone(), caixa.clone());
466                session.register_control("pos.caixa", move || {
467                    Box::new(TelaCaixa { caixa: caixa_aqui.clone(), session: aqui.clone() }.grow(1.0))
468                        as AnyView
469                });
470                let (aqui, copias) = (session.clone(), entradas.etiquetas.clone());
471                session.register_control("pos.etiquetas", move || {
472                    Box::new(
473                        TelaEtiquetas {
474                            session: aqui.clone(),
475                            palavras: palavras.clone(),
476                            copias: copias.clone(),
477                        }
478                        .grow(1.0),
479                    ) as AnyView
480                });
481                let (aqui, entradas_aqui, service) =
482                    (session.clone(), entradas.clone(), caixa.service.clone());
483                session.register_control("pos.entrada", move || {
484                    Box::new(
485                        TelaEntrada {
486                            entradas: entradas_aqui.clone(),
487                            session: aqui.clone(),
488                            service: service.clone(),
489                        }
490                        .grow(1.0),
491                    ) as AnyView
492                });
493                let (aqui, perifericos) = (session.clone(), caixa.perifericos.clone());
494                session.register_control("pos.perifericos", move || {
495                    Box::new(
496                        crate::perifericos::TelaPerifericos {
497                            perifericos: perifericos.clone(),
498                            session: aqui.clone(),
499                        }
500                        .grow(1.0),
501                    ) as AnyView
502                });
503                let (aqui, servicos_aqui, service, palavras_servico) =
504                    (session.clone(), servicos.clone(), caixa.service.clone(), Rc::new(ReportWords::pt_br()));
505                session.register_control("pos.servico", move || {
506                    Box::new(
507                        TelaServico {
508                            servicos: servicos_aqui.clone(),
509                            service: service.clone(),
510                            session: aqui.clone(),
511                            palavras: palavras_servico.clone(),
512                        }
513                        .grow(1.0),
514                    ) as AnyView
515                });
516                let aqui = session.clone();
517                session.register_control("pos.preco", move || {
518                    Box::new(TelaPreco { session: aqui.clone() }.grow(1.0)) as AnyView
519                });
520                session.open_view(VIEW_INICIO);
521            })
522    }
