21fn titulo(text: impl Into<String>) -> Text {
22    Text::new(text).style(TextStyle::Caption).foreground(Semantic::InkMuted)

25fn quantidade_em_texto(quantidade: rust_decimal::Decimal) -> String {
26    format!("{:.3}", quantidade.round_dp(3)).replace('.', ",")

36    fn cabecalho(&self) -> AnyView {
37        let carrinho = self.caixa.carrinho.get();
38        let cliente = carrinho.cliente.clone().map_or_else(
39            || "Consumidor não identificado".to_owned(),
40            |cliente| format!("Consumidor {cliente}"),
41        );
42        let nota = if self.caixa.nota_grande.get() { "Nota grande (NF-e, A4)" } else { "NFC-e" };
43        let operador = configuracao(&self.session, "operador");
44        let numero = configuracao(&self.session, "caixa");
45        Box::new(
46            HStack::new(vec![
47                Box::new(Text::new(format!("Caixa {numero}")).style(TextStyle::Headline)) as AnyView,
48                Box::new(Text::new(format!("Operador: {operador}")).foreground(Semantic::InkMuted)),
49                Box::new(Spacer::new()),
50                Box::new(Text::new(nota).font_weight(Weight::SEMIBOLD)),
51                Box::new(Text::new(cliente).foreground(Semantic::InkMuted)),
52            ])
53            .spacing(24.0)
54            .align(Align::Center)
55            .padding((16.0, 10.0))
56            .background(Semantic::Surface),

60    fn vendendo(&self) -> AnyView {
61        let carrinho = self.caixa.carrinho.get();
62        let mut tabela = Table::new([
63            Column::new("Item").weight(0.4),
64            Column::new("Código").weight(1.2),
65            Column::new("Descrição").weight(3.0),
66            Column::new("Qtd").weight(0.7).numeric(),
67            Column::new("Unit.").weight(0.8).numeric(),
68            Column::new("Total").weight(0.9).numeric(),
69        ]);
70        for (index, linha) in carrinho.linhas.iter().enumerate() {
71            tabela = tabela.row([
72                (index + 1).to_string(),
73                linha.gtin.clone().unwrap_or_else(|| linha.codigo.clone()),
74                linha.descricao.clone(),
75                quantidade_em_texto(linha.quantidade),
76                valor_em_texto(linha.preco),
77                valor_em_texto(linha.total()),
78            ]);
79        }
80        let lista =
81            Card::new(ScrollView::new(vec![Box::new(tabela) as AnyView]).grow(1.0)).grow(1.0).stretch();
82
83        let ultimo: AnyView = match carrinho.ultimo() {
84            Some(linha) => Box::new(
85                VStack::new(vec![
86                    Box::new(titulo("ÚLTIMO ITEM")) as AnyView,
87                    Box::new(Text::new(linha.descricao.clone()).style(TextStyle::Title)),
88                    Box::new(
89                        HStack::new(vec![
90                            Box::new(
91                                Text::new(format!(
92                                    "{} x R$ {}",
93                                    quantidade_em_texto(linha.quantidade),
94                                    valor_em_texto(linha.preco)
95                                ))
96                                .foreground(Semantic::InkMuted),
97                            ) as AnyView,
98                            Box::new(Spacer::new()),
99                            Box::new(
100                                Text::new(format!("R$ {}", valor_em_texto(linha.total())))
101                                    .style(TextStyle::Headline),
102                            ),
103                        ])
104                        .align(Align::Center),
105                    ),
106                ])
107                .spacing(8.0)
108                .align(Align::Stretch),
109            ),
110            None => Box::new(titulo("PASSE O PRIMEIRO PRODUTO")),
111        };
112        let (caixa, session) = (self.caixa.clone(), self.session.clone());
113        let leitura = ScanField::new(move |lido| caixa.ler(&session, &lido.text))
114            .placeholder("Código de barras ou quantidade*código");
115        let mut lado: Vec<AnyView> = vec![
116            Box::new(Card::new(ultimo)),
117            Box::new(total_escuro("TOTAL DA VENDA", carrinho.total(), Some(carrinho.linhas.len())).grow(1.0)),
118        ];
119        if let Some(problema) = self.caixa.problema.get() {
120            lado.push(Box::new(Alert::new("Atenção", problema).tone(Tone::Danger)));
121        }
122        lado.push(Box::new(
123            VStack::new(vec![Box::new(titulo("CÓDIGO DE BARRAS")) as AnyView, Box::new(leitura)])
124                .spacing(4.0)
125                .align(Align::Stretch),
126        ));
127        Box::new(
128            HStack::new(vec![
129                Box::new(lista) as AnyView,
130                Box::new(VStack::new(lado).spacing(16.0).align(Align::Stretch).width(430.0)),
131            ])
132            .spacing(16.0)
133            .align(Align::Stretch)
134            .padding(16.0)
135            .grow(1.0),
136        )
137    }

139    fn pagando(&self, cx: &mut BuildCx<'_>) -> AnyView {
140        let total = self.caixa.carrinho.peek().total();
141        let forma_atual = self.caixa.forma.get();
142        let mut formas: Vec<AnyView> = Vec::new();
143        for (index, forma) in Forma::TODAS.into_iter().enumerate() {
144            let (caixa, recebido) = (self.caixa.clone(), valor_em_texto(total));
145            let escolher = std::rc::Rc::new(move || {
146                caixa.forma.set(forma);
147                if !forma.tem_troco() {
148                    caixa.recebido.set(recebido.clone());
149                }
150            });
151            let tecla = (index + 1).to_string();
152            Hotkey::alt(&tecla).register(cx, escolher.clone());
153            formas.push(Box::new(
154                Button::new(format!("{}  {}", Hotkey::alt(&tecla).label(), forma.nome()))
155                    .size(ButtonSize::Large)
156                    .full_width()
157                    .style(if forma == forma_atual { ButtonStyle::Primary } else { ButtonStyle::Secondary })
158                    .on_click(move || escolher()),
159            ));
160        }
161        let (read, write) = (self.caixa.recebido.clone(), self.caixa.recebido.clone());
162        let (caixa, session) = (self.caixa.clone(), self.session.clone());
163        let campo = TextField::new(Binding::new(move || read.get(), move |value: String| write.set(value)))
164            .autofocus()
165            .select_on_open()
166            .end_aligned()
167            .on_submit(move |_| caixa.avancar(&session));
168        let recebido = ravenrs_pos::ler_valor(&self.caixa.recebido.get()).unwrap_or_default();
169        let mut esquerda: Vec<AnyView> = vec![
170            Box::new(Text::new("Forma de pagamento").style(TextStyle::Title)),
171            Box::new(VStack::new(formas).spacing(12.0).align(Align::Stretch)),
172            Box::new(FormField::new("Valor recebido", campo)),
173        ];
174        if let Some(problema) = self.caixa.problema.get() {
175            esquerda.push(Box::new(Alert::new("Atenção", problema).tone(Tone::Danger)));
176        }
177        let troco = ravenrs_pos::troco(forma_atual, total, recebido).unwrap_or_default();
178        let lado = VStack::new(vec![
179            Box::new(total_escuro("TOTAL", total, None)) as AnyView,
180            Box::new(total_escuro("RECEBIDO", recebido, None)),
181            Box::new(total_escuro("TROCO", troco, None).grow(1.0)),
182        ])
183        .spacing(8.0)
184        .align(Align::Stretch)
185        .width(470.0);
186        Box::new(
187            HStack::new(vec![
188                Box::new(VStack::new(esquerda).spacing(16.0).align(Align::Stretch).grow(1.0)) as AnyView,
189                Box::new(lado),
190            ])
191            .spacing(16.0)
192            .align(Align::Stretch)
193            .padding(16.0)
194            .grow(1.0),
195        )
196    }

198    fn cobrando(&self, pergunta: Option<&Pergunta>, mensagem: &str) -> AnyView {
199        let responder = |entrada: EntradaOperador| {
200            let (caixa, session) = (self.caixa.clone(), self.session.clone());
201            move || caixa.responder(&session, entrada.clone())
202        };
203        let mut linhas: Vec<AnyView> =
204            vec![Box::new(Text::new("Pagamento no cartão ou Pix").style(TextStyle::Title))];
205        match pergunta {
206            None => {
207                linhas.push(Box::new(Spinner::new()));
208                linhas.push(Box::new(Text::new(mensagem.to_owned()).style(TextStyle::Headline)));
209            }
210            Some(Pergunta::Mensagem { texto, exige_ok }) => {
211                linhas.push(Box::new(Text::new(texto.clone()).style(TextStyle::Headline)));
212                if *exige_ok {
213                    linhas.push(Box::new(
214                        Button::new("OK")
215                            .style(ButtonStyle::Primary)
216                            .on_click(responder(EntradaOperador::Ok)),
217                    ));
218                }
219            }
220            Some(Pergunta::Confirmacao { texto }) => {
221                linhas.push(Box::new(Text::new(texto.clone()).style(TextStyle::Headline)));
222                linhas.push(Box::new(
223                    HStack::new(vec![
224                        Box::new(
225                            Button::new("Sim")
226                                .style(ButtonStyle::Primary)
227                                .on_click(responder(EntradaOperador::Sim)),
228                        ) as AnyView,
229                        Box::new(Button::new("Não").on_click(responder(EntradaOperador::Nao))),
230                    ])
231                    .spacing(8.0),
232                ));
233            }
234            Some(Pergunta::Menu { titulo, opcoes }) => {
235                linhas.push(Box::new(Text::new(titulo.clone()).style(TextStyle::Headline)));
236                for (indice, opcao) in opcoes.iter().enumerate() {
237                    linhas.push(Box::new(
238                        Button::new(opcao.texto.clone())
239                            .style(ButtonStyle::Secondary)
240                            .full_width()
241                            .on_click(responder(EntradaOperador::Opcao(indice))),
242                    ));
243                }
244            }
245            Some(Pergunta::Campo { rotulo, tipo, .. }) => {
246                let (caixa, session) = (self.caixa.clone(), self.session.clone());
247                let mut campo =
248                    ScanField::new(move |lido| caixa.responder(&session, EntradaOperador::Texto(lido.text)))
249                        .placeholder(rotulo.clone());
250                if *tipo == TipoCampo::Mascarado {
251                    campo = campo.placeholder(format!("{rotulo} (oculto)"));
252                }
253                linhas.push(Box::new(FormField::new(rotulo.clone(), campo)));
254            }
255            Some(Pergunta::QrCode { conteudo, mensagem }) => {
256                if let Some(texto) = mensagem {
257                    linhas.push(Box::new(Text::new(texto.clone()).style(TextStyle::Headline)));
258                }
259                linhas.push(Box::new(
260                    ravenrs_widgets::Barcode::qr(conteudo.clone())
261                        .alt("QR Code do Pix")
262                        .width(240.0)
263                        .height(240.0),
264                ));
265            }
266        }
267        linhas.push(Box::new(
268            Button::new("Esc  Cancelar o pagamento")
269                .style(ButtonStyle::Secondary)
270                .on_click(responder(EntradaOperador::Cancelar)),
271        ));
272        Box::new(VStack::new(linhas).spacing(16.0).align(Align::Center).justify(Justify::Center).grow(1.0))
273    }

288    fn resultado(&self, fim: &ravenrs_app_nfe::EmissaoFinal) -> AnyView {
289        let (nome, tone) = match (fim.autorizada, fim.tentar_de_novo) {
290            (true, _) => ("Nota autorizada", Tone::Success),
291            (false, true) => ("A nota não saiu: dá para tentar de novo", Tone::Warning),
292            (false, false) => ("Nota rejeitada", Tone::Danger),
293        };
294        let mut linhas: Vec<AnyView> =
295            vec![Box::new(Alert::new(nome, format!("Código {}: {}", fim.c_stat, fim.motivo)).tone(tone))];
296        let dado = |rotulo: &str, valor: String| -> AnyView {
297            Box::new(
298                HStack::new(vec![
299                    Box::new(Text::new(rotulo.to_owned()).foreground(Semantic::InkMuted).width(150.0))
300                        as AnyView,
301                    Box::new(Text::new(valor).font_weight(Weight::SEMIBOLD)),
302                ])
303                .align(Align::Center),
304            )
305        };
306        linhas.push(dado("Número", format!("{:09}", fim.numero)));
307        if !fim.chave.is_empty() {
308            linhas.push(dado("Chave de acesso", fim.chave.clone()));
309        }
310        if !fim.protocolo.is_empty() {
311            linhas.push(dado("Protocolo", fim.protocolo.clone()));
312        }
313        linhas.push(dado(
314            "Total e troco",
315            format!("R$ {}, troco R$ {}", valor_em_texto(fim.total), valor_em_texto(fim.troco)),
316        ));
317        if fim.autorizada {
318            let (caixa, session, chave) = (self.caixa.clone(), self.session.clone(), fim.chave.clone());
319            linhas.push(Box::new(
320                Button::new(if self.caixa.nota_grande.get() {
321                    "Ver o DANFE"
322                } else {
323                    "Imprimir ou ver o cupom"
324                })
325                .icon(ravenrs_widgets::IconName::Printer)
326                .on_click(move || caixa.imprimir_cupom(&session, &chave)),
327            ));
328            if !self.caixa.nota_grande.get() {
329                let (caixa, session) = (self.caixa.clone(), self.session.clone());
330                linhas.push(Box::new(
331                    Button::new("Trocar por nota grande (cancela a NFC-e)")
332                        .style(ButtonStyle::Secondary)
333                        .on_click(move || caixa.trocar_por_nota_grande(&session)),
334                ));
335            }
336            if let Some(problema) = self.caixa.problema.get() {
337                linhas.push(Box::new(Alert::new("Atenção", problema).tone(Tone::Warning)));
338            }
339        }
340        linhas.push(Box::new(
341            Text::new(if fim.autorizada { "F9 começa a próxima venda." } else { "F9 volta ao pagamento." })
342                .style(TextStyle::Callout)
343                .foreground(Semantic::InkMuted),
344        ));
345        let mut direita: Vec<AnyView> = Vec::new();
346        if self.caixa.service.sandbox_ativo().is_some() {
347            direita.push(Box::new(CenariosSandbox { caixa: self.caixa.clone() }));
348        }
349        Box::new(
350            HStack::new(vec![
351                Box::new(Card::new(VStack::new(linhas).spacing(12.0).align(Align::Stretch)).grow(1.0))
352                    as AnyView,
353                Box::new(VStack::new(direita).spacing(16.0).align(Align::Stretch).width(430.0)),
354            ])
355            .spacing(16.0)
356            .align(Align::Stretch)
357            .padding(16.0)
358            .grow(1.0),
359        )
360    }
361}

363/// A sum in large letters on the darkest ground, as the operator reads it from afar.
364fn total_escuro(rotulo: &str, valor: rust_decimal::Decimal, itens: Option<usize>) -> impl View + use<> {
365    let mut linhas: Vec<AnyView> = vec![
366        Box::new(Text::new(rotulo.to_owned()).style(TextStyle::Caption).foreground(Semantic::Canvas)),
367        Box::new(
368            Text::new(format!("R$ {}", valor_em_texto(valor)))
369                .font_size(56.0)
370                .font_weight(Weight::SEMIBOLD)
371                .foreground(Semantic::Canvas),
372        ),
373    ];
374    if let Some(itens) = itens {
375        linhas.push(Box::new(Text::new(format!("{itens} itens")).foreground(Semantic::Canvas)));
376    }
377    VStack::new(linhas).spacing(4.0).justify(Justify::Center).padding(24.0).background(Semantic::Ink)
378}

380/// How the sandbox answers the next document, for practicing at the counter.

385impl View for CenariosSandbox {
386    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
387        let Some(sandbox) = self.caixa.service.sandbox_ativo() else {
388            return cx.child(&Spacer::new());
389        };
390        let mut lines: Vec<AnyView> = vec![
391            Box::new(Text::new("Cenário do sandbox").style(TextStyle::Headline)),
392            Box::new(
393                Text::new("Vale para a próxima emissão, para treinar o caixa.")
394                    .style(TextStyle::Callout)
395                    .foreground(Semantic::InkMuted),
396            ),
397        ];
398        type Cenario = fn(&ravenrs_nfe::sandbox::SandboxRequest) -> ravenrs_nfe::sandbox::Resposta;
399        let opcoes: [(&str, Cenario); 4] = [
400            ("Assinatura inválida (297)", cenarios::assinatura_297),
401            ("Duplicidade (204)", cenarios::duplicidade_204),
402            ("Serviço paralisado (108)", cenarios::paralisado_108),
403            ("Sem resposta (tempo esgotado)", cenarios::timeout),
404        ];
405        if let Some(simulador) = &self.caixa.simulador {
406            lines.push(Box::new(Text::new("Cartão e Pix simulados").style(TextStyle::Headline)));
407            type CenarioTef = fn(
408                &ravenrs_pagamento::simulador::RequisicaoSimulada,
409            ) -> ravenrs_pagamento::simulador::Resposta;
410            let tef: [(&str, CenarioTef); 4] = [
411                ("Cartão negado", tef_cenarios::negado),
412                ("Pix com QR Code", tef_cenarios::qr_pix),
413                ("Pergunta parcelas", tef_cenarios::pergunta_parcelas),
414                ("Sem o código de autorização", tef_cenarios::c_aut_ausente),
415            ];
416            for (nome, cenario) in tef {
417                let simulador = simulador.clone();
418                lines.push(Box::new(
419                    Button::new(nome)
420                        .style(ButtonStyle::Secondary)
421                        .full_width()
422                        .on_click(move || simulador.use_once(tef_iniciar(cenario))),
423                ));
424            }
425            lines.push(Box::new(Text::new("Notas na SEFAZ").style(TextStyle::Headline)));
426        }
427        for (nome, cenario) in opcoes {
428            let sandbox = sandbox.clone();
429            lines.push(Box::new(
430                Button::new(nome)
431                    .style(ButtonStyle::Secondary)
432                    .full_width()
433                    .on_click(move || sandbox.use_once(autorizacao(cenario))),
434            ));
435        }
436        cx.child(&Card::new(VStack::new(lines).spacing(8.0).align(Align::Stretch)))
437    }
438}

440impl View for TelaCaixa {
441    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
442        let corpo = match self.caixa.tela.get() {
443            Tela::Vendendo => self.vendendo(),
444            Tela::Pagando => self.pagando(cx),
445            Tela::Enviando => self.enviando(),
446            Tela::Cobrando { pergunta, mensagem } => self.cobrando(pergunta.as_ref(), &mensagem),
447            Tela::Resultado(fim) => self.resultado(&fim),
448        };
449        cx.child(&VStack::new(vec![self.cabecalho(), corpo]).spacing(0.0).align(Align::Stretch).grow(1.0))
450    }
451}
