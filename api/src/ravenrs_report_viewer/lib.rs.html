1//! Document viewer for RavenRS Reporting.
2//!
3//! * [`DocumentViewer`] shows a rendered [`Document`]: pages, zoom, thumbnails
4//!   and text search that finds words regardless of case and accents.
5//! * [`ParameterPanel`] asks for a report's parameters and applies them together.
6//! * [`ViewerState`] holds what the user has chosen, so it survives rebuilds and
7//!   can be shared with the rest of an app.
8//!
9//! ```
10//! use std::rc::Rc;
11//! use ravenrs_report::{Band, Control, Report, ReportEngine};
12//! use ravenrs_report_viewer::{DocumentViewer, ViewerState};
13//! use serde_json::json;
14//!
15//! let report = Report::new("Sales").detail(Band::new(18.0).control(Control::field("[Name]").at(0.0, 0.0, 200.0, 18.0)));
16//! let document = Rc::new(ReportEngine::new().render(&report, &json!([{ "Name": "Ana" }])));
17//! let state = ViewerState::new();
18//! let viewer = DocumentViewer::new(document, &state);
19//! # let _ = viewer;
20//! ```

46const ZOOM_STEP: f32 = 0.25;

230/// What is worth keeping between builds, for one document.
231#[derive(Default)]
232struct Cache {
233    /// Identifies the document; everything here is cleared when it changes.
234    document: usize,
235    scenes: HashMap<(usize, u32), Rc<Scene>>,
236    query: String,
237    matches: Rc<Vec<Match>>,
238    text: Option<TextSystem>,
239}
240
241impl Cache {
242    fn for_document(&mut self, document: &Rc<Document>) {
243        let id = Rc::as_ptr(document) as usize;
244        if self.document != id {
245            self.document = id;
246            self.scenes.clear();
247            self.query.clear();
248            self.matches = Rc::new(Vec::new());
249        }
250    }
251
252    fn scene(&mut self, document: &Document, page: usize, zoom: f32) -> Rc<Scene> {
253        let key = (page, zoom.to_bits());
254        if let Some(scene) = self.scenes.get(&key) {
255            return scene.clone();
256        }
257        let mut scene = Scene::new();
258        if let Some(sheet) = document.pages.get(page) {
259            paint_page(sheet, self.text.get_or_insert_with(TextSystem::new), &mut scene, Point::ZERO, zoom);
260        }
261        if self.scenes.len() >= SCENES_MAX {
262            self.scenes.clear();
263        }
264        let scene = Rc::new(scene);
265        self.scenes.insert(key, scene.clone());
266        scene
267    }
268
269    fn matches(&mut self, document: &Document, query: &str) -> Rc<Vec<Match>> {
270        if self.query != query {
271            self.query = query.to_owned();
272            self.matches = Rc::new(find_matches(document, query));
273        }
274        self.matches.clone()
275    }
276
277    /// Where a match is on its page, in points.
278    fn match_rect(&mut self, document: &Document, found: &Match) -> Option<Rect> {
279        let Item::Text(block) = document.pages.get(found.page)?.items.get(found.item)? else { return None };
280        let line = block.lines.get(found.line)?;
281        let text = self.text.get_or_insert_with(TextSystem::new);
282        let start = prefix_width(text, block, line.text.get(..found.range.start)?);
283        let end = prefix_width(text, block, line.text.get(..found.range.end)?);
284        Some(Rect::new(line.x + start, line.top, (end - start).max(1.0), block.line_height))
285    }
286}
287
288fn prefix_width(text: &mut TextSystem, block: &TextBlock, prefix: &str) -> f32 {
289    let attrs = TextAttrs {
290        size: block.size,
291        line_height: block.line_height,
292        weight: if block.bold { 600.0 } else { 400.0 },
293        ..TextAttrs::default()
294    };
295    text.layout(prefix, &attrs).measure(None).width
296}

298/// Copies a cached page into the frame being painted, moved to `origin`.
299fn draw_at(source: &Scene, origin: Point, scene: &mut Scene) {
300    let shift = |r: Rect| Rect::from_origin_size(Point::new(r.x() + origin.x, r.y() + origin.y), r.size);
301    for primitive in source.primitives() {
302        match primitive {
303            Primitive::Quad(q) => scene.push_quad(Quad { rect: shift(q.rect), ..*q }),
304            Primitive::Glyphs(run) => {
305                let mut run = run.clone();
306                for glyph in &mut run.glyphs {
307                    glyph.x += origin.x;
308                    glyph.y += origin.y;
309                }
310                scene.push_glyphs(run);
311            }
312            Primitive::Image(image) => {
313                scene.push_image(ravenrs_core::ImageQuad { rect: shift(image.rect), ..image.clone() });
314            }
315            Primitive::PushClip(r) => scene.push_clip(shift(*r)),
316            Primitive::PopClip => scene.pop_clip(),
317            _ => {}
318        }
319    }
320}

378    fn search_bar(&self, matches: &[Match], current: usize) -> impl View + use<> {
379        let labels = &self.labels;
380        let (read, write) = (self.state.clone(), self.state.clone());
381        let document = self.document.clone();
382        let field = TextField::new(Binding::new(
383            move || read.query.get(),
384            move |query: String| {
385                let found = write.cache.borrow_mut().matches(&document, &query);
386                write.query.set(query);
387                write.current_match.set(0);
388                if let Some(first) = found.first() {
389                    write.page.set(first.page);
390                }
391            },
392        ))
393        .placeholder(labels.search.clone())
394        .icon(IconName::Search)
395        .compact()
396        .width(180.0)
397        .on_submit({
398            let step = self.stepper(1);
399            move |_| step()
400        });
401        let mut items: Vec<AnyView> = vec![Box::new(field)];
402        if !self.state.query.with(|q| q.trim().is_empty()) {
403            let status = if matches.is_empty() {
404                labels.no_matches.clone()
405            } else {
406                format!("{} {} {}", current + 1, labels.of, matches.len())
407            };
408            items.push(Box::new(Text::new(status).style(TextStyle::Callout).foreground(Semantic::InkMuted)));
409            if !matches.is_empty() {
410                let (back, forward) = (self.stepper(-1), self.stepper(1));
411                items.push(Box::new(
412                    IconButton::new(IconName::ChevronUp)
413                        .label(labels.previous_match.clone())
414                        .size(ButtonSize::Small)
415                        .on_click(back),
416                ));
417                items.push(Box::new(
418                    IconButton::new(IconName::ChevronDown)
419                        .label(labels.next_match.clone())
420                        .size(ButtonSize::Small)
421                        .on_click(forward),
422                ));
423            }
424        }
425        HStack::new(items).spacing(SPACING.s2).align(Align::Center)
426    }
427
428    /// A step through the search results, going around at the ends.
429    fn stepper(&self, delta: isize) -> impl Fn() + use<> {
430        let state = self.state.clone();
431        let document = self.document.clone();
432        move || {
433            let query = state.query.get();
434            let found = state.cache.borrow_mut().matches(&document, &query);
435            if found.is_empty() {
436                return;
437            }
438            let count = found.len() as isize;
439            let next = (state.current_match.get() as isize + delta).rem_euclid(count) as usize;
440            state.current_match.set(next);
441            state.page.set(found[next].page);
442        }
443    }

445    fn zoom_controls(&self, zoom: f32, fit_width: f32) -> impl View + use<> {
446        let labels = &self.labels;
447        let (out, into, fit) = (self.state.zoom.clone(), self.state.zoom.clone(), self.state.zoom.clone());
448        HStack::new((
449            IconButton::new(IconName::ZoomOut)
450                .label(labels.zoom_out.clone())
451                .size(ButtonSize::Small)
452                .on_click(move || out.update(|z| *z = (*z - ZOOM_STEP).max(ZOOM_MIN))),
453            Text::new(format!("{:.0}%", zoom * 100.0))
454                .style(TextStyle::Callout)
455                .font_weight(Weight::SEMIBOLD),
456            IconButton::new(IconName::ZoomIn)
457                .label(labels.zoom_in.clone())
458                .size(ButtonSize::Small)
459                .on_click(move || into.update(|z| *z = (*z + ZOOM_STEP).min(ZOOM_MAX))),
460            IconButton::new(IconName::Maximize2)
461                .label(labels.fit_width.clone())
462                .size(ButtonSize::Small)
463                .on_click(move || fit.set(fit_width.clamp(ZOOM_MIN, ZOOM_MAX))),
464        ))
465        .spacing(SPACING.s1)
466        .align(Align::Center)
467    }

469    fn thumbnails_column(&self, current: usize, available_height: f32, cx: &mut BuildCx<'_>) -> NodeId {
470        let pages = self.document.pages.len();
471        let tall =
472            self.document.pages.first().map_or(842.0, |p| p.height) * THUMBNAIL_ZOOM + 2.0 * SPACING.s6;
473        let fits = ((available_height / tall) as usize).clamp(1, THUMBNAILS_MAX);
474        let start = current.saturating_sub(fits / 2).min(pages.saturating_sub(fits));
475        let items: Vec<AnyView> = (start..(start + fits).min(pages))
476            .map(|page| {
477                Box::new(Thumbnail {
478                    document: self.document.clone(),
479                    state: self.state.clone(),
480                    page,
481                    current: page == current,
482                    label: format!("{} {}", self.labels.thumbnail_of_page, page + 1),
483                }) as AnyView
484            })
485            .collect();
486        cx.child(
487            &VStack::new(items)
488                .spacing(SPACING.s2)
489                .padding(SPACING.s3)
490                .align(Align::Center)
491                .background(Semantic::Surface),
492        )
493    }

496impl View for DocumentViewer {
497    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
498        let document = &self.document;
499        let state = &self.state;
500        let pages = document.pages.len().max(1);
501        let current = state.page.get().min(pages - 1);
502        let zoom = state.zoom.get();
503        let query = state.query.get();
504        let matches = {
505            let mut cache = state.cache.borrow_mut();
506            cache.for_document(document);
507            cache.matches(document, &query)
508        };
509        let current_match = state.current_match.get().min(matches.len().saturating_sub(1));
510        let page_width = document.pages.get(current).map_or(595.0, |p| p.width);
511        let viewport = cx.viewport();
512        let thumbnails = state.thumbnails.get();
513        let sidebar = if thumbnails { page_width * THUMBNAIL_ZOOM + 2.0 * SPACING.s3 + 1.0 } else { 0.0 };
514        let fit_width = (viewport.width - sidebar - 2.0 * SPACING.s6) / page_width.max(1.0);
515
516        let labels = &self.labels;
517        let toggle = state.thumbnails.clone();
518        let mut bar: Vec<AnyView> = vec![Box::new(
519            IconButton::new(IconName::PanelLeft)
520                .label(labels.thumbnails.clone())
521                .size(ButtonSize::Small)
522                .on_click(move || toggle.update(|shown| *shown = !*shown)),
523        )];
524        if let Some(leading) = &self.leading {
525            bar.extend(leading());
526        }
527        bar.push(Box::new(Pagination::new(state.page.binding(), pages).labels(
528            labels.previous.clone(),
529            labels.next.clone(),
530            labels.page.clone(),
531        )));
532        bar.push(Box::new(Spacer::new()));
533        bar.push(Box::new(self.search_bar(&matches, current_match)));
534        bar.push(Box::new(self.zoom_controls(zoom, fit_width)));
535        if self.print_button {
536            let printed = document.clone();
537            bar.push(Box::new(
538                Button::new(labels.print.clone())
539                    .icon(IconName::Printer)
540                    .style(ButtonStyle::Secondary)
541                    .size(ButtonSize::Small)
542                    .on_click(move || print(&printed)),
543            ));
544        }
545        if let Some(trailing) = &self.trailing {
546            bar.extend(trailing());
547        }
548        let toolbar = cx.child(
549            &HStack::new(bar)
550                .spacing(SPACING.s3)
551                .align(Align::Center)
552                .wrap()
553                .padding(Insets::symmetric(SPACING.s2, SPACING.s4))
554                .background(Semantic::Surface),
555        );
556        let line = cx.child(&Divider::horizontal());
557
558        let theme = *cx.theme();
559        let highlights: Vec<(Rect, bool)> = {
560            let mut cache = state.cache.borrow_mut();
561            matches
562                .iter()
563                .enumerate()
564                .filter(|(_, found)| found.page == current)
565                .filter_map(|(index, found)| {
566                    Some((cache.match_rect(document, found)?, index == current_match))
567                })
568                .collect()
569        };
570        let mut body: Vec<AnyView> = Vec::new();
571        for message in document.errors.iter().take(ERRORS_SHOWN) {
572            body.push(Box::new(Alert::new(labels.report_error.clone(), message.clone()).tone(Tone::Danger)));
573        }
574        if let Some(sheet) = document.pages.get(current) {
575            let scene = state.cache.borrow_mut().scene(document, current, zoom);
576            body.push(Box::new(PageView {
577                scene,
578                highlights,
579                highlight: with_alpha(theme.color(Semantic::Warning), 0.35),
580                highlight_current: with_alpha(theme.color(Semantic::Warning), 0.7),
581                zoom,
582                width: sheet.width * zoom,
583                height: sheet.height * zoom,
584                text: document.page_text(current).join(" "),
585            }));
586        }
587        // At least the width of the page, and otherwise the width of the view,
588        // wherever the app has put it, so the page stays centered.
589        let main = cx.child(
590            &ScrollView::new((PagesColumn { pages: body, min_width: page_width * zoom + 2.0 * SPACING.s6 },))
591                .horizontal()
592                .align(Align::Stretch)
593                .grow(1.0),
594        );
595        let row_children = if thumbnails {
596            // The window less what is usually above the pages: toolbars and a status bar.
597            let column = self.thumbnails_column(current, viewport.height - CHROME_HEIGHT, cx);
598            let separator = cx.child(&Divider::vertical());
599            vec![column, separator, main]
600        } else {
601            vec![main]
602        };
603        let row = cx.container(
604            LayoutStyle {
605                grow: 1.0,
606                min_height: Some(0.0),
607                align: Align::Stretch,
608                overflow_hidden: true,
609                ..LayoutStyle::stack(Axis::Horizontal)
610            },
611            &row_children,
612            Paint::None,
613        );
614
615        let (up, down) = (state.page.clone(), state.page.clone());
616        // Whole percents: each zoom gets its pages drawn again, and a gesture passes through many.
617        let pinched = state.zoom.clone();
618        cx.on_pinch(
619            row,
620            Rc::new(move |delta| {
621                pinched.update(|z| {
622                    let next = (*z * (1.0 + delta) * 100.0).round() / 100.0;
623                    *z = next.clamp(ZOOM_MIN, ZOOM_MAX);
624                });
625            }),
626        );
627        cx.shortcut(Key::PageUp, false, Rc::new(move || up.update(|p| *p = p.saturating_sub(1))));
628        cx.shortcut(Key::PageDown, false, Rc::new(move || down.update(|p| *p = (*p + 1).min(pages - 1))));
629        cx.container(
630            LayoutStyle {
631                grow: 1.0,
632                min_height: Some(0.0),
633                overflow_hidden: true,
634                align: Align::Stretch,
635                ..LayoutStyle::stack(Axis::Vertical)
636            },
637            &[toolbar, line, row],
638            Paint::None,
639        )
640    }
641}

647/// One page, with what a search found on it.
648struct PageView {
649    scene: Rc<Scene>,
650    highlights: Vec<(Rect, bool)>,
651    highlight: Color,
652    highlight_current: Color,
653    zoom: f32,
654    width: f32,
655    height: f32,
656    text: String,
657}
658
659impl View for PageView {
660    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
661        let scene = self.scene.clone();
662        let highlights = self.highlights.clone();
663        let (normal, strong, zoom) = (self.highlight, self.highlight_current, self.zoom);
664        let canvas = Canvas::new(move |frame, out| {
665            draw_at(&scene, frame.origin, out);
666            for (rect, current) in &highlights {
667                let box_on_screen = Rect::new(
668                    frame.x() + rect.x() * zoom,
669                    frame.y() + rect.y() * zoom,
670                    rect.width() * zoom,
671                    rect.height() * zoom,
672                );
673                out.push_quad(Quad::new(box_on_screen, if *current { strong } else { normal }).radius(2.0));
674            }
675        })
676        .size(self.width, self.height);
677        let canvas = cx.child(&canvas);
678        let node = cx.container(LayoutStyle::stack(Axis::Vertical), &[canvas], Paint::None);
679        // Screen readers get the page's text; everyone else sees the drawing.
680        cx.semantics(node, Semantics::new(Role::Image).name(self.text.clone()));
681        node
682    }
683}

685/// A small picture of a page that goes to it when clicked.
686struct Thumbnail {
687    document: Rc<Document>,
688    state: ViewerState,
689    page: usize,
690    current: bool,
691    label: String,
692}
693
694impl View for Thumbnail {
695    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
696        let theme = *cx.theme();
697        let scene = self.state.cache.borrow_mut().scene(&self.document, self.page, THUMBNAIL_ZOOM);
698        let sheet = self.document.pages.get(self.page);
699        let (width, height) =
700            sheet.map_or((0.0, 0.0), |p| (p.width * THUMBNAIL_ZOOM, p.height * THUMBNAIL_ZOOM));
701        let picture = Canvas::new(move |frame, out| draw_at(&scene, frame.origin, out)).size(width, height);
702        let caption = Text::new((self.page + 1).to_string())
703            .style(TextStyle::Caption)
704            .foreground(if self.current { Semantic::Accent } else { Semantic::InkMuted });
705        let content = cx.child(&VStack::new((picture, caption)).spacing(SPACING.s1).align(Align::Center));
706        let border =
707            self.current.then(|| ravenrs_core::Border { width: 2.0, color: theme.color(Semantic::Accent) });
708        let hovered = cx.is_hover_within();
709        let fill = if hovered && !self.current {
710            theme.color(Semantic::SurfaceElevated)
711        } else {
712            Color::TRANSPARENT
713        };
714        let node = cx.container(
715            LayoutStyle { padding: Insets::uniform(SPACING.s1), ..LayoutStyle::stack(Axis::Vertical) },
716            &[content],
717            Paint::Quad(QuadPaint { corner_radius: 6.0, border, ..QuadPaint::new(fill) }),
718        );
719        let (page, target) = (self.state.page.clone(), self.page);
720        cx.on_click(node, Rc::new(move || page.set(target)));
721        cx.focusable(node, 6.0);
722        cx.cursor(node, CursorIcon::Pointer);
723        cx.semantics(node, Semantics::new(Role::Button).name(self.label.clone()).selected(self.current));
724        node
725    }
