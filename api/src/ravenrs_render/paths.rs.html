1//! Shapes made of curves. Each frame every shape gets a place in a coverage atlas,
2//! where its edges add up the exact area they cover in each pixel, in one pass. The
3//! main pass then paints the shapes from it, in the order of the scene. There are no
4//! triangles to fit and no samples to count, so edges are right at any size.
5
6use std::collections::HashMap;
7
8use bytemuck::{Pod, Zeroable};
9use ravenrs_core::{ColorStop, FillRule, ImageData, PathElement, PathFill, PathPaint, Point};

11/// 16 bits are always there, and rounding piles up a little with many edges to a pixel.
12const COVERAGE_FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::R16Float;
13/// 32 bits are exact for what this does, where the device can blend them.
14const FINE_COVERAGE_FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::R32Float;
15const INITIAL_ATLAS: u32 = 1024;
16/// How far a straight piece may stray from its curve, in device pixels.
17const FLATNESS: f32 = 0.08;
18/// More straight pieces than this for one curve add nothing anyone can see.
19const MOST_PIECES: u32 = 128;
20const RAMP_WIDTH: u32 = 256;
21const RAMP_ROWS: u32 = 256;
22/// The side of the texture for the images that paths show, where the device has it.
23const IMAGE_ATLAS: u32 = 4096;
24/// What is left empty between images, so that none reads its neighbor.
25const IMAGE_GAP: u32 = 2;

47/// A vertex buffer that grows as needed and is reused across frames.
48struct Instances {
49    buffer: wgpu::Buffer,
50    capacity: usize,
51    label: &'static str,
52}
53
54impl Instances {
55    fn new(device: &wgpu::Device, label: &'static str, bytes: usize) -> Self {
56        Self { buffer: Self::create(device, label, bytes), capacity: bytes, label }
57    }
58
59    fn create(device: &wgpu::Device, label: &'static str, bytes: usize) -> wgpu::Buffer {
60        device.create_buffer(&wgpu::BufferDescriptor {
61            label: Some(label),
62            size: bytes as u64,
63            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
64            mapped_at_creation: false,
65        })
66    }
67
68    fn upload(&mut self, device: &wgpu::Device, queue: &wgpu::Queue, bytes: &[u8]) {
69        if bytes.len() > self.capacity {
70            self.capacity = bytes.len().next_power_of_two();
71            self.buffer = Self::create(device, self.label, self.capacity);
72        }
73        if !bytes.is_empty() {
74            queue.write_buffer(&self.buffer, 0, bytes);
75        }
76    }
77}

79/// Where a shape goes in the atlas, and its clip if it has one.
80type Places = ((u32, u32), Option<(u32, u32)>);
81
82/// Where shapes go in the atlas this frame: shelves, filled from the top.
83#[derive(Default)]
84struct Shelves {
85    size: u32,
86    x: u32,
87    y: u32,
88    height: u32,
89}
90
91impl Shelves {
92    /// A place of this size, growing the atlas up to `limit`. Places already given stay valid.
93    fn place(&mut self, width: u32, height: u32, limit: u32) -> Option<(u32, u32)> {
94        if width > limit || height > limit {
95            return None;
96        }
97        while width > self.size {
98            self.size = (self.size * 2).min(limit);
99        }
100        if self.x + width > self.size {
101            self.y += self.height;
102            self.x = 0;
103            self.height = 0;
104        }
105        while self.y + height > self.size {
106            if self.size >= limit {
107                return None;
108            }
109            self.size = (self.size * 2).min(limit);
110        }
111        let place = (self.x, self.y);
112        self.x += width;
113        self.height = self.height.max(height);
114        Some(place)
115    }
116}

369    /// Places for a shape and, if it has one, its clip, which have to be on the same page.
370    /// `None` only for what could not fit even in an empty atlas.
371    fn reserve(&mut self, width: u32, height: u32, clipped: bool, limit: u32) -> Option<Places> {
372        for _ in 0..2 {
373            if let Some(shape) = self.shelves.place(width, height, limit) {
374                if !clipped {
375                    return Some((shape, None));
376                }
377                if let Some(clip) = self.shelves.place(width, height, limit) {
378                    return Some((shape, Some(clip)));
379                }
380            }
381            // Full: what is here gets drawn, and the next page starts empty.
382            self.page_ends.push(self.edges.len() as u32);
383            self.shelves = Shelves { size: self.shelves.size, ..Shelves::default() };
384        }
385        None
386    }

401    /// Adds a shape, clipped to `scissor` in device pixels: x, y, width, height.
402    pub(crate) fn push(&mut self, path: &PathFill, scale: f32, scissor: [u32; 4], limit: u32) {
403        let [a, b, c, d, e, f] = path.transform;
404        let device = |p: Point| Point::new((a * p.x + c * p.y + e) * scale, (b * p.x + d * p.y + f) * scale);
405        self.points.clear();
406        let contours = collect_contours(&path.elements, &device, &mut self.points);
407        let Some(bounds) = bounds_of(&self.points) else { return };
408        // Only what the clip lets through takes up the atlas.
409        let left = bounds[0].floor().max(scissor[0] as f32);
410        let top = bounds[1].floor().max(scissor[1] as f32);
411        let right = bounds[2].ceil().min((scissor[0] + scissor[2]) as f32);
412        let bottom = bounds[3].ceil().min((scissor[1] + scissor[3]) as f32);
413        if right <= left || bottom <= top {
414            return;
415        }
416        let (width, height) = ((right - left) as u32, (bottom - top) as u32);
417        let Some(((ax, ay), clip_place)) = self.reserve(width, height, path.clip.is_some(), limit) else {
418            tracing::warn!(width, height, "a shape is larger than the coverage atlas can be");
419            return;
420        };
421        let (shift_x, shift_y) = (ax as f32 - left, ay as f32 - top);
422        let region = [ax as f32, ay as f32, (ax + width) as f32, (ay + height) as f32];
423        add_edges(&self.points, &contours, (shift_x, shift_y), region, &mut self.edges);
424        let clip = match (&path.clip, clip_place) {
425            (Some(clip), Some((cx, cy))) => {
426                let mut points = Vec::new();
427                let contours = collect_contours(&clip.elements, &device, &mut points);
428                let shift = (cx as f32 - left, cy as f32 - top);
429                let region = [cx as f32, cy as f32, (cx + width) as f32, (cy + height) as f32];
430                add_edges(&points, &contours, shift, region, &mut self.edges);
431                let even_odd = if clip.rule == FillRule::EvenOdd { 1.0 } else { 0.0 };
432                [cx as f32, cy as f32, even_odd, if clip.inverted { 1.0 } else { 0.0 }]
433            }
434            _ => [-1.0; 4],
435        };
436        let opacity = path.opacity.clamp(0.0, 1.0);
437        let at = |p: Point| {
438            let q = device(p);
439            [q.x, q.y]
440        };
441        let mut image_row = [0.0; 2];
442        let (kind, color, gradient, row) = match &path.paint {
443            PathPaint::Solid(color) => (0.0, color.to_array(), [0.0; 4], 0.0),
444            PathPaint::Linear { start, end, stops } => {
445                let (s, t) = (at(*start), at(*end));
446                (1.0, [0.0; 4], [s[0], s[1], t[0], t[1]], self.ramp(stops))
447            }
448            PathPaint::Radial { center, radius, stops } => {
449                let s = at(*center);
450                // A transform that does not keep circles is drawn with its mean scale.
451                let k = ((a * d - b * c).abs().sqrt() * radius * scale).max(0.0);
452                (2.0, [0.0; 4], [s[0], s[1], k, 0.0], self.ramp(stops))
453            }
454            PathPaint::Image { image, placement } => {
455                let Some(place) = self.image_place(image) else { return };
456                let [pa, pb, pc, pd, pe, pf] = *placement;
457                // From the image's square to the device, and back for the shader.
458                let (ma, mb) = (scale * (a * pa + c * pb), scale * (b * pa + d * pb));
459                let (mc, md) = (scale * (a * pc + c * pd), scale * (b * pc + d * pd));
460                let (me, mf) = (scale * (a * pe + c * pf + e), scale * (b * pe + d * pf + f));
461                let det = ma * md - mb * mc;
462                if det == 0.0 {
463                    return;
464                }
465                let u = [md / det, -mc / det, (mc * mf - md * me) / det];
466                let v = [-mb / det, ma / det, (mb * me - ma * mf) / det];
467                image_row = [v[1], v[2]];
468                (3.0, place, [u[0], u[1], u[2], v[0]], 0.0)
469            }
470            _ => return,
471        };
472        let even_odd = if path.rule == FillRule::EvenOdd { 1.0 } else { 0.0 };
473        self.shapes.push(ShapeInstance {
474            rect: [left, top, width as f32, height as f32],
475            atlas: [ax as f32, ay as f32, even_odd, kind],
476            color,
477            gradient,
478            params: [row, opacity, image_row[0], image_row[1]],
479            clip,
480        });
481    }

483    /// Where this image is in the atlas, as a texture coordinate and size that stay
484    /// half a pixel inside it, placing it on first use.
485    fn image_place(&mut self, image: &ImageData) -> Option<[f32; 4]> {
486        let (width, height) = (image.width(), image.height());
487        let at = match self.image_places.get(&image.id()) {
488            Some(at) => *at,
489            None => {
490                let side = self.image_side;
491                let fits = |shelves: &mut Shelves| shelves.place(width + IMAGE_GAP, height + IMAGE_GAP, side);
492                let place = fits(&mut self.image_shelves).or_else(|| {
493                    // Full: the oldest images go, and come back as they are used.
494                    self.image_places.clear();
495                    self.image_shelves = Shelves { size: side, ..Shelves::default() };
496                    fits(&mut self.image_shelves)
497                });
498                let Some((x, y)) = place else {
499                    tracing::warn!(width, height, "an image does not fit in the path atlas");
500                    return None;
501                };
502                self.image_places.insert(image.id(), [x, y]);
503                self.image_uploads.push(image.clone());
504                [x, y]
505            }
506        };
507        let side = self.image_side as f32;
508        Some([
509            (at[0] as f32 + 0.5) / side,
510            (at[1] as f32 + 0.5) / side,
511            (width as f32 - 1.0).max(0.0) / side,
512            (height as f32 - 1.0).max(0.0) / side,
513        ])
514    }

516    /// The row of the ramp texture holding these stops, as a texture coordinate.
517    fn ramp(&mut self, stops: &[ColorStop]) -> f32 {
518        let mut pixels = Vec::with_capacity(RAMP_WIDTH as usize * 4);
519        for x in 0..RAMP_WIDTH {
520            let color = color_at(stops, x as f32 / (RAMP_WIDTH - 1) as f32);
521            pixels.extend(color.map(|v| (v.clamp(0.0, 1.0) * 255.0).round() as u8));
522        }
523        let next = self.ramp_rows.len() as u32;
524        let row = match self.ramp_rows.get(&pixels) {
525            Some(row) => *row,
526            None => {
527                // Full: the oldest colors go, and this frame's come back as they are used.
528                let row = if next >= RAMP_ROWS {
529                    self.ramp_rows.clear();
530                    0
531                } else {
532                    next
533                };
534                self.ramp_rows.insert(pixels.clone(), row);
535                self.ramp_pixels.push((row, pixels));
536                row
537            }
538        };
539        (row as f32 + 0.5) / RAMP_ROWS as f32
540    }

542    /// Uploads this frame's shapes and adds up their coverage.
543    /// Uploads what the frame's shapes need, before any page is drawn.
544    pub(crate) fn prepare(&mut self, device: &wgpu::Device, queue: &wgpu::Queue) {
545        if self.shapes.is_empty() {
546            return;
547        }
548        for (row, pixels) in self.ramp_pixels.drain(..) {
549            queue.write_texture(
550                wgpu::TexelCopyTextureInfo {
551                    texture: &self.ramps,
552                    mip_level: 0,
553                    origin: wgpu::Origin3d { x: 0, y: row, z: 0 },
554                    aspect: wgpu::TextureAspect::All,
555                },
556                &pixels,
557                wgpu::TexelCopyBufferLayout {
558                    offset: 0,
559                    bytes_per_row: Some(RAMP_WIDTH * 4),
560                    rows_per_image: Some(1),
561                },
562                wgpu::Extent3d { width: RAMP_WIDTH, height: 1, depth_or_array_layers: 1 },
563            );
564        }
565        for image in self.image_uploads.drain(..) {
566            let Some(at) = self.image_places.get(&image.id()) else { continue };
567            // Premultiplied, so that blending neighbors does not bring in the color
568            // of what is fully transparent.
569            let mut pixels = image.pixels().to_vec();
570            for pixel in pixels.as_chunks_mut::<4>().0 {
571                let alpha = u32::from(pixel[3]);
572                for channel in &mut pixel[..3] {
573                    *channel = ((u32::from(*channel) * alpha + 127) / 255) as u8;
574                }
575            }
576            queue.write_texture(
577                wgpu::TexelCopyTextureInfo {
578                    texture: &self.images,
579                    mip_level: 0,
580                    origin: wgpu::Origin3d { x: at[0], y: at[1], z: 0 },
581                    aspect: wgpu::TextureAspect::All,
582                },
583                &pixels,
584                wgpu::TexelCopyBufferLayout {
585                    offset: 0,
586                    bytes_per_row: Some(image.width() * 4),
587                    rows_per_image: Some(image.height()),
588                },
589                wgpu::Extent3d { width: image.width(), height: image.height(), depth_or_array_layers: 1 },
590            );
591        }
592        let size = self.shelves.size;
593        if self.coverage.as_ref().is_none_or(|(_, current)| *current != size) {
594            let texture = device.create_texture(&wgpu::TextureDescriptor {
595                label: Some("raven path coverage"),
596                size: wgpu::Extent3d { width: size, height: size, depth_or_array_layers: 1 },
597                mip_level_count: 1,
598                sample_count: 1,
599                dimension: wgpu::TextureDimension::D2,
600                format: self.coverage_format,
601                usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
602                view_formats: &[],
603            });
604            let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
605            self.texture_group = Some(device.create_bind_group(&wgpu::BindGroupDescriptor {
606                label: Some("raven path textures"),
607                layout: &self.texture_layout,
608                entries: &[
609                    wgpu::BindGroupEntry { binding: 0, resource: wgpu::BindingResource::TextureView(&view) },
610                    wgpu::BindGroupEntry {
611                        binding: 1,
612                        resource: wgpu::BindingResource::TextureView(&self.ramps_view),
613                    },
614                    wgpu::BindGroupEntry {
615                        binding: 2,
616                        resource: wgpu::BindingResource::Sampler(&self.sampler),
617                    },
618                    wgpu::BindGroupEntry {
619                        binding: 3,
620                        resource: wgpu::BindingResource::TextureView(&self.images_view),
621                    },
622                ],
623            }));
624            self.coverage = Some((view, size));
625        }
626        queue.write_buffer(
627            &self.target_buffer,
628            0,
629            bytemuck::cast_slice(&[size as f32, size as f32, 0.0, 0.0]),
630        );
631        self.edge_buffer.upload(device, queue, bytemuck::cast_slice(&self.edges));
632        self.shape_buffer.upload(device, queue, bytemuck::cast_slice(&self.shapes));
633    }
634
635    /// Draws one page of the atlas, erasing the one before it.
636    pub(crate) fn draw_page(
637        &self,
638        encoder: &mut wgpu::CommandEncoder,
639        page: u32,
640        timestamp_writes: Option<wgpu::RenderPassTimestampWrites<'_>>,
641    ) {
642        if self.shapes.is_empty() {
643            return;
644        }
645        let page = page as usize;
646        let start = if page == 0 { 0 } else { self.page_ends[page - 1] };
647        let end = self.page_ends.get(page).copied().unwrap_or(self.edges.len() as u32);
648        let Some((view, _)) = &self.coverage else { return };
649        let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
650            label: Some("raven path coverage"),
651            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
652                view,
653                depth_slice: None,
654                resolve_target: None,
655                ops: wgpu::Operations {
656                    load: wgpu::LoadOp::Clear(wgpu::Color::TRANSPARENT),
657                    store: wgpu::StoreOp::Store,
658                },
659            })],
660            depth_stencil_attachment: None,
661            timestamp_writes,
662            occlusion_query_set: None,
663            multiview_mask: None,
664        });
665        pass.set_pipeline(&self.edge_pipeline);
666        pass.set_bind_group(0, &self.target_group, &[]);
667        pass.set_vertex_buffer(0, self.edge_buffer.buffer.slice(..));
668        pass.draw(0..4, start..end);
669    }
670
671    /// Readies the main pass to paint shapes.
672    pub(crate) fn bind(&self, pass: &mut wgpu::RenderPass<'_>) {
673        pass.set_pipeline(&self.shape_pipeline);
674        if let Some(group) = &self.texture_group {
675            pass.set_bind_group(1, group, &[]);
676        }
677        pass.set_vertex_buffer(0, self.shape_buffer.buffer.slice(..));
678    }
679}

681/// The outline as closed lists of device points, appended to `points`.
682fn collect_contours(
683    elements: &[PathElement],
684    device: &impl Fn(Point) -> Point,
685    points: &mut Vec<Point>,
686) -> Vec<std::ops::Range<usize>> {
687    let mut contours = Vec::new();
688    let mut start = points.len();
689    let mut pen = Point::new(0.0, 0.0);
690    for element in elements {
691        match *element {
692            PathElement::MoveTo(p) => {
693                if points.len() > start {
694                    contours.push(start..points.len());
695                }
696                start = points.len();
697                pen = device(p);
698                points.push(pen);
699            }
700            PathElement::LineTo(p) => {
701                pen = device(p);
702                points.push(pen);
703            }
704            PathElement::CubicTo(p1, p2, p3) => {
705                let (p1, p2, p3) = (device(p1), device(p2), device(p3));
706                flatten(pen, p1, p2, p3, points);
707                pen = p3;
708            }
709            PathElement::Close => {
710                if points.len() > start {
711                    contours.push(start..points.len());
712                    pen = points[start];
713                }
714                start = points.len();
715            }
716        }
717    }
718    if points.len() > start {
719        contours.push(start..points.len());
720    }
721    contours
722}

724/// Adds a shape's edges for its place in the atlas. Edges left of the place still
725/// count for what is to their right, and those above or below it do not.
726fn add_edges(
727    points: &[Point],
728    contours: &[std::ops::Range<usize>],
729    shift: (f32, f32),
730    region: [f32; 4],
731    edges: &mut Vec<EdgeInstance>,
732) {
733    for contour in contours {
734        let points = &points[contour.clone()];
735        for (i, from) in points.iter().enumerate() {
736            let to = points[(i + 1) % points.len()];
737            if from.y == to.y {
738                continue;
739            }
740            let (y0, y1) = (from.y + shift.1, to.y + shift.1);
741            if y0.max(y1) <= region[1] || y0.min(y1) >= region[3] {
742                continue;
743            }
744            edges.push(EdgeInstance { line: [from.x + shift.0, y0, to.x + shift.0, y1], region });
745        }
746    }
747}

749/// The curve as straight pieces, the number of them from how much it bends.
750fn flatten(p0: Point, p1: Point, p2: Point, p3: Point, out: &mut Vec<Point>) {
751    let ddx = (p0.x - 2.0 * p1.x + p2.x).abs().max((p1.x - 2.0 * p2.x + p3.x).abs());
752    let ddy = (p0.y - 2.0 * p1.y + p2.y).abs().max((p1.y - 2.0 * p2.y + p3.y).abs());
753    let bend = (ddx * ddx + ddy * ddy).sqrt();
754    let pieces = ((0.75 * bend / FLATNESS).sqrt().ceil() as u32).clamp(1, MOST_PIECES);
755    for i in 1..=pieces {
756        let t = i as f32 / pieces as f32;
757        let u = 1.0 - t;
758        let (a, b, c, d) = (u * u * u, 3.0 * u * u * t, 3.0 * u * t * t, t * t * t);
759        out.push(Point::new(
760            a * p0.x + b * p1.x + c * p2.x + d * p3.x,
761            a * p0.y + b * p1.y + c * p2.y + d * p3.y,
762        ));
763    }
764}

766fn bounds_of(points: &[Point]) -> Option<[f32; 4]> {
767    let first = points.first()?;
768    let mut bounds = [first.x, first.y, first.x, first.y];
769    for p in points {
770        bounds[0] = bounds[0].min(p.x);
771        bounds[1] = bounds[1].min(p.y);
772        bounds[2] = bounds[2].max(p.x);
773        bounds[3] = bounds[3].max(p.y);
774    }
775    Some(bounds)
776}

778fn color_at(stops: &[ColorStop], t: f32) -> [f32; 4] {
779    let Some(first) = stops.first() else { return [0.0; 4] };
780    if t <= first.offset {
781        return first.color.to_array();
782    }
783    for pair in stops.windows(2) {
784        let (a, b) = (pair[0], pair[1]);
785        if t <= b.offset {
786            let k = if b.offset > a.offset { (t - a.offset) / (b.offset - a.offset) } else { 1.0 };
787            let (from, to) = (a.color.to_array(), b.color.to_array());
788            return [0, 1, 2, 3].map(|i| from[i] + (to[i] - from[i]) * k);
789        }
790    }
791    stops.last().map_or([0.0; 4], |stop| stop.color.to_array())
792}
