1//! Asks the GPU how long a frame took, without ever waiting for the answer.
2
3use std::sync::Arc;
4use std::sync::atomic::{AtomicU8, Ordering};
5use std::time::Duration;
6
7/// Main pass start and end, then coverage pass start and end. A frame with more than one
8/// page of shapes has more passes, and only the first of each kind is timed.
9const QUERIES: u32 = 4;
10const QUERIES_SIZE: u64 = QUERIES as u64 * 8;
11/// Answers in flight. When all are taken, frames go unmeasured rather than wait.
12const SLOTS: usize = 4;
13
14const FREE: u8 = 0;
15const MAPPING: u8 = 1;
16const READY: u8 = 2;
17
18struct Slot {
19    buffer: wgpu::Buffer,
20    state: Arc<AtomicU8>,
21}
22
23/// Timestamps around the frame's passes. Answers arrive frames later through buffers
24/// mapped in the background.
25pub(crate) struct GpuTimer {
26    device: wgpu::Device,
27    queries: wgpu::QuerySet,
28    resolve: wgpu::Buffer,
29    slots: Vec<Slot>,
30    /// The slot this frame's answer was copied to, to map after the submit.
31    copied: Option<usize>,
32    /// Nanoseconds in one tick of the GPU's clock.
33    period: f64,
34    latest: Option<Duration>,
35}
36
37impl GpuTimer {
38    /// A timer, unless this GPU cannot tell the time.
39    pub fn new(device: &wgpu::Device, queue: &wgpu::Queue) -> Option<Self> {
40        if !device.features().contains(wgpu::Features::TIMESTAMP_QUERY) {
41            return None;
42        }
43        let queries = device.create_query_set(&wgpu::QuerySetDescriptor {
44            label: Some("raven frame timing"),
45            ty: wgpu::QueryType::Timestamp,
46            count: QUERIES,
47        });
48        let resolve = device.create_buffer(&wgpu::BufferDescriptor {
49            label: Some("raven frame timing"),
50            size: QUERIES_SIZE,
51            usage: wgpu::BufferUsages::QUERY_RESOLVE | wgpu::BufferUsages::COPY_SRC,
52            mapped_at_creation: false,
53        });
54        let slots = (0..SLOTS)
55            .map(|_| Slot {
56                buffer: device.create_buffer(&wgpu::BufferDescriptor {
57                    label: Some("raven frame timing answer"),
58                    size: QUERIES_SIZE,
59                    usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
60                    mapped_at_creation: false,
61                }),
62                state: Arc::new(AtomicU8::new(FREE)),
63            })
64            .collect();
65        Some(Self {
66            device: device.clone(),
67            queries,
68            resolve,
69            slots,
70            copied: None,
71            period: f64::from(queue.get_timestamp_period()),
72            latest: None,
73        })
74    }
75
76    pub fn main_pass(&self) -> wgpu::RenderPassTimestampWrites<'_> {
77        wgpu::RenderPassTimestampWrites {
78            query_set: &self.queries,
79            beginning_of_pass_write_index: Some(0),
80            end_of_pass_write_index: Some(1),
81        }
82    }
83
84    pub fn coverage_pass(&self) -> wgpu::RenderPassTimestampWrites<'_> {
85        wgpu::RenderPassTimestampWrites {
86            query_set: &self.queries,
87            beginning_of_pass_write_index: Some(2),
88            end_of_pass_write_index: Some(3),
89        }
90    }
91
92    /// Collects answers that came in and records this frame's question. Returns the
93    /// latest answer known.
94    pub fn finish_frame(&mut self, encoder: &mut wgpu::CommandEncoder) -> Option<Duration> {
95        let _ = self.device.poll(wgpu::PollType::Poll);
96        self.collect();
97        self.copied = self.slots.iter().position(|slot| slot.state.load(Ordering::Acquire) == FREE);
98        encoder.resolve_query_set(&self.queries, 0..QUERIES, &self.resolve, 0);
99        if let Some(i) = self.copied {
100            encoder.copy_buffer_to_buffer(&self.resolve, 0, &self.slots[i].buffer, 0, QUERIES_SIZE);
101        }
102        self.latest
103    }
104
105    /// Asks for the answer of the frame just submitted.
106    pub fn frame_submitted(&mut self) {
107        let Some(i) = self.copied.take() else { return };
108        let slot = &self.slots[i];
109        slot.state.store(MAPPING, Ordering::Release);
110        let state = slot.state.clone();
111        slot.buffer.slice(..).map_async(wgpu::MapMode::Read, move |result| {
112            state.store(if result.is_ok() { READY } else { FREE }, Ordering::Release);
113        });
114    }
115
116    fn collect(&mut self) {
117        let period = self.period;
118        for slot in &self.slots {
119            if slot.state.load(Ordering::Acquire) != READY {
120                continue;
121            }
122            let mut ticks = [0u64; QUERIES as usize];
123            if let Ok(bytes) = slot.buffer.slice(..).get_mapped_range() {
124                let (chunks, _) = bytes.as_chunks::<8>();
125                for (tick, raw) in ticks.iter_mut().zip(chunks) {
126                    *tick = u64::from_le_bytes(*raw);
127                }
128            }
129            slot.buffer.unmap();
130            slot.state.store(FREE, Ordering::Release);
131            let total = ticks[1].saturating_sub(ticks[0]) + ticks[3].saturating_sub(ticks[2]);
132            self.latest = Some(Duration::from_nanos((total as f64 * period) as u64));
133        }
134    }
135}
