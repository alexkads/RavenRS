1//! Drawing a scene into memory, with no window: for tests, documentation and thumbnails.
2
3use ravenrs_core::Scene;
4
5use crate::{GpuContext, Renderer};
6
7const FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::Rgba8Unorm;
8
9/// Draws a scene at this size, and gives its pixels as RGBA, row by row. Fails when the GPU cannot
10/// give the pixels back.
11pub fn render_to_rgba(gpu: &GpuContext, scene: &Scene, width: u32, height: u32) -> Result<Vec<u8>, String> {
12    let mut renderer = Renderer::new(&gpu.device, &gpu.queue, FORMAT);
13    render_with(gpu, &mut renderer, scene, width, height)
14}
15
16/// Like [`render_to_rgba`], with a renderer of the caller's, made for
17/// `Rgba8Unorm`: one that draws the cables of a [`GpuCables`](crate::GpuCables) world, say.
18pub fn render_with(
19    gpu: &GpuContext,
20    renderer: &mut Renderer,
21    scene: &Scene,
22    width: u32,
23    height: u32,
24) -> Result<Vec<u8>, String> {
25    let texture = gpu.device.create_texture(&wgpu::TextureDescriptor {
26        label: Some("offscreen"),
27        size: wgpu::Extent3d { width, height, depth_or_array_layers: 1 },
28        mip_level_count: 1,
29        sample_count: 1,
30        dimension: wgpu::TextureDimension::D2,
31        format: FORMAT,
32        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::COPY_SRC,
33        view_formats: &[],
34    });
35    let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
36    renderer.render(scene, &view, (width, height), 1.0);
37    let padded =
38        (width * 4).div_ceil(wgpu::COPY_BYTES_PER_ROW_ALIGNMENT) * wgpu::COPY_BYTES_PER_ROW_ALIGNMENT;
39    let buffer = gpu.device.create_buffer(&wgpu::BufferDescriptor {
40        label: Some("offscreen readback"),
41        size: u64::from(padded * height),
42        usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
43        mapped_at_creation: false,
44    });
45    let mut encoder = gpu.device.create_command_encoder(&wgpu::CommandEncoderDescriptor::default());
46    encoder.copy_texture_to_buffer(
47        texture.as_image_copy(),
48        wgpu::TexelCopyBufferInfo {
49            buffer: &buffer,
50            layout: wgpu::TexelCopyBufferLayout {
51                offset: 0,
52                bytes_per_row: Some(padded),
53                rows_per_image: Some(height),
54            },
55        },
56        wgpu::Extent3d { width, height, depth_or_array_layers: 1 },
57    );
58    gpu.queue.submit(Some(encoder.finish()));
59    let slice = buffer.slice(..);
60    let (sender, receiver) = std::sync::mpsc::channel();
61    slice.map_async(wgpu::MapMode::Read, move |mapped| {
62        let _ = sender.send(mapped);
63    });
64    gpu.device.poll(wgpu::PollType::wait_indefinitely()).map_err(|error| error.to_string())?;
65    receiver.recv().map_err(|error| error.to_string())?.map_err(|error| error.to_string())?;
66    let data = slice.get_mapped_range().map_err(|error| error.to_string())?;
67    let mut pixels = Vec::with_capacity((width * height * 4) as usize);
68    for row in 0..height {
69        let start = (row * padded) as usize;
70        pixels.extend_from_slice(&data[start..start + (width * 4) as usize]);
71    }
72    Ok(pixels)
73}
