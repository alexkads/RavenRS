145impl InstanceBuffer {
146    fn new(device: &wgpu::Device, label: &'static str, stride: usize) -> Self {
147        let capacity = 256;
148        Self { buffer: Self::create(device, label, stride * capacity), capacity, stride, label }
149    }
150
151    fn create(device: &wgpu::Device, label: &'static str, size: usize) -> wgpu::Buffer {
152        device.create_buffer(&wgpu::BufferDescriptor {
153            label: Some(label),
154            size: size as u64,
155            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
156            mapped_at_creation: false,
157        })
158    }
159
160    fn upload(&mut self, device: &wgpu::Device, queue: &wgpu::Queue, bytes: &[u8]) {
161        let needed = bytes.len() / self.stride;
162        if needed > self.capacity {
163            self.capacity = needed.next_power_of_two();
164            self.buffer = Self::create(device, self.label, self.capacity * self.stride);
165        }
166        if !bytes.is_empty() {
167            queue.write_buffer(&self.buffer, 0, bytes);
168        }
169    }
170}

243pub(crate) fn pipeline(
244    device: &wgpu::Device,
245    label: &str,
246    source: &str,
247    layout: &wgpu::PipelineLayout,
248    instance_layout: wgpu::VertexBufferLayout<'_>,
249    format: wgpu::TextureFormat,
250) -> wgpu::RenderPipeline {
251    let module = device.create_shader_module(wgpu::ShaderModuleDescriptor {
252        label: Some(label),
253        source: wgpu::ShaderSource::Wgsl(source.into()),
254    });
255    device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
256        label: Some(label),
257        layout: Some(layout),
258        vertex: wgpu::VertexState {
259            module: &module,
260            entry_point: Some("vs_main"),
261            compilation_options: Default::default(),
262            buffers: &[Some(instance_layout)],
263        },
264        primitive: wgpu::PrimitiveState {
265            topology: wgpu::PrimitiveTopology::TriangleStrip,
266            ..Default::default()
267        },
268        depth_stencil: None,
269        multisample: wgpu::MultisampleState::default(),
270        fragment: Some(wgpu::FragmentState {
271            module: &module,
272            entry_point: Some("fs_main"),
273            compilation_options: Default::default(),
274            targets: &premultiplied_target(format),
275        }),
276        multiview_mask: None,
277        cache: None,
278    })
279}

512    /// Draws the cables of a GPU world from its own buffer. Without one, cables are
513    /// drawn as shapes from the points the scene carries.
514    pub fn set_cables(&mut self, drawer: Option<CableDrawer>) {
515        self.cables = drawer.filter(|_| self.cable_pipeline.is_some());
516        self.cable_bind = None;
517    }

519    /// Sends this frame's cable looks and makes the bind group again if a buffer changed.
520    fn prepare_cables(&mut self) {
521        let (Some(drawer), Some((_, layout))) = (&self.cables, &self.cable_pipeline) else { return };
522        if self.cable_looks.is_empty() {
523            return;
524        }
525        let bytes: &[u8] = bytemuck::cast_slice(&self.cable_looks);
526        if bytes.len() as u64 > self.cable_look_buffer.size() {
527            self.cable_look_buffer = self.device.create_buffer(&wgpu::BufferDescriptor {
528                label: Some("raven cable looks"),
529                size: (bytes.len() as u64).next_power_of_two(),
530                usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
531                mapped_at_creation: false,
532            });
533        }
534        self.queue.write_buffer(&self.cable_look_buffer, 0, bytes);
535        if self.cable_pictures.upload(&self.device, &self.queue) {
536            self.cable_bind = None;
537        }
538        let (shape, generation) = drawer.shape();
539        let size = self.cable_look_buffer.size();
540        if self.cable_bind.as_ref().is_none_or(|(_, g, s)| *g != generation || *s != size) {
541            let group = self.device.create_bind_group(&wgpu::BindGroupDescriptor {
542                label: Some("raven cables"),
543                layout,
544                entries: &[
545                    wgpu::BindGroupEntry { binding: 0, resource: shape.as_entire_binding() },
546                    wgpu::BindGroupEntry { binding: 1, resource: self.cable_look_buffer.as_entire_binding() },
547                    wgpu::BindGroupEntry {
548                        binding: 2,
549                        resource: self.cable_pictures.buffer_ref().as_entire_binding(),
550                    },
551                ],
552            });
553            self.cable_bind = Some((group, generation, size));
554        }
555    }

594    /// Renders `scene` into `target`, a texture of `size` device pixels.
595    /// `scale` converts the scene's logical pixels to device pixels.
596    pub fn render(
597        &mut self,
598        scene: &Scene,
599        target: &wgpu::TextureView,
600        size: (u32, u32),
601        scale: f32,
602    ) -> RenderStats {
603        let _span = tracing::trace_span!("ravenrs::render", primitives = scene.len()).entered();
604        if size.0 == 0 || size.1 == 0 || scale <= 0.0 {
605            return RenderStats::default();
606        }
607        self.target_size = size;
608        self.image_cache.begin_frame();
609        self.prepare(scene, scale);
610
611        let globals = Globals {
612            viewport: [size.0 as f32 / scale, size.1 as f32 / scale],
613            scale,
614            srgb_target: u32::from(self.srgb_target),
615        };
616        self.queue.write_buffer(&self.globals_buffer, 0, bytemuck::bytes_of(&globals));
617        self.quad_buffer.upload(&self.device, &self.queue, bytemuck::cast_slice(&self.quads));
618        self.glyph_buffer.upload(&self.device, &self.queue, bytemuck::cast_slice(&self.glyphs));
619        self.image_buffer.upload(&self.device, &self.queue, bytemuck::cast_slice(&self.image_instances));
620        self.prepare_cables();
621
622        let clear = self.color_for_target(scene.clear_color());
623        let mut encoder = self
624            .device
625            .create_command_encoder(&wgpu::CommandEncoderDescriptor { label: Some("raven frame") });
626        // A page of shapes has to be drawn before the shapes that read it, and the main pass
627        // cannot stop for that, so a page after the first splits it in two.
628        self.paths.prepare(&self.device, &self.queue);
629        let coverage_timing = self.timer.as_ref().map(GpuTimer::coverage_pass);
630        self.paths.draw_page(&mut encoder, 0, coverage_timing);
631        let mut load = wgpu::LoadOp::Clear(wgpu::Color {
632            r: f64::from(clear[0]),
633            g: f64::from(clear[1]),
634            b: f64::from(clear[2]),
635            a: f64::from(clear[3]),
636        });
637        let (mut from, mut page) = (0, 0);
638        loop {
639            let next = self.batches[from..]
640                .iter()
641                .position(|batch| matches!(batch.kind, BatchKind::Paths(p) if p > page))
642                .map_or(self.batches.len(), |at| from + at);
643            let timing = if from == 0 { self.timer.as_ref().map(GpuTimer::main_pass) } else { None };
644            self.draw_main_pass(&mut encoder, target, size, load, from..next, timing);
645            let Some(BatchKind::Paths(after)) = self.batches.get(next).map(|batch| batch.kind) else { break };
646            self.paths.draw_page(&mut encoder, after, None);
647            load = wgpu::LoadOp::Load;
648            (from, page) = (next, after);
649        }
650        let gpu_time = self.timer.as_mut().and_then(|timer| timer.finish_frame(&mut encoder));
651        self.queue.submit(Some(encoder.finish()));
652        if let Some(timer) = &mut self.timer {
653            timer.frame_submitted();
654        }
655
656        RenderStats {
657            quads: self.quads.len(),
658            glyphs: self.glyphs.len(),
659            draw_calls: self.batches.len() + self.paths.pages(),
660            atlas_glyphs: self.atlas.len(),
661            images: self.image_instances.len(),
662            cached_images: self.image_cache.len(),
663            paths: self.paths.shapes(),
664            path_edges: self.paths.edges(),
665            breaks_kind: self.breaks.kind,
666            breaks_image: self.breaks.image,
667            breaks_clip: self.breaks.clip,
668            uploaded_bytes: std::mem::size_of_val(self.quads.as_slice())
669                + std::mem::size_of_val(self.glyphs.as_slice())
670                + std::mem::size_of_val(self.image_instances.as_slice())
671                + self.paths.uploaded_bytes(),
672            glyphs_rasterized: self.glyphs_rasterized + self.atlas.rasterized(),
673            atlas_resets: self.atlas_resets,
674            atlas_used: (self.atlas.used() * 1000.0) as u32,
675            atlas_size: self.atlas.size,
676            gpu_time,
677        }
678    }

680    /// Draws these batches on `target`.
681    fn draw_main_pass(
682        &self,
683        encoder: &mut wgpu::CommandEncoder,
684        target: &wgpu::TextureView,
685        size: (u32, u32),
686        load: wgpu::LoadOp<wgpu::Color>,
687        batches: std::ops::Range<usize>,
688        timestamp_writes: Option<wgpu::RenderPassTimestampWrites<'_>>,
689    ) {
690        let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
691            label: Some("raven main pass"),
692            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
693                view: target,
694                depth_slice: None,
695                resolve_target: None,
696                ops: wgpu::Operations { load, store: wgpu::StoreOp::Store },
697            })],
698            depth_stencil_attachment: None,
699            timestamp_writes,
700            occlusion_query_set: None,
701            multiview_mask: None,
702        });
703        pass.set_bind_group(0, &self.globals_bind_group, &[]);
704        let mut current = None;
705        let mut scissor = [0, 0, size.0, size.1];
706        for batch in &self.batches[batches] {
707            if batch.scissor != scissor {
708                scissor = batch.scissor;
709                pass.set_scissor_rect(scissor[0], scissor[1], scissor[2], scissor[3]);
710            }
711            if current != Some(batch.kind) {
712                match batch.kind {
713                    BatchKind::Quads => {
714                        pass.set_pipeline(&self.quad_pipeline);
715                        pass.set_vertex_buffer(0, self.quad_buffer.buffer.slice(..));
716                    }
717                    BatchKind::Glyphs => {
718                        pass.set_pipeline(&self.glyph_pipeline);
719                        pass.set_bind_group(1, &self.atlas_bind_group, &[]);
720                        pass.set_vertex_buffer(0, self.glyph_buffer.buffer.slice(..));
721                    }
722                    BatchKind::Image(id) => {
723                        let Some(image) = self.image_cache.get(id) else { continue };
724                        pass.set_pipeline(&self.image_pipeline);
725                        pass.set_bind_group(1, &image.bind_group, &[]);
726                        pass.set_vertex_buffer(0, self.image_buffer.buffer.slice(..));
727                    }
728                    BatchKind::Paths(_) => self.paths.bind(&mut pass),
729                    BatchKind::Cables => {
730                        let (Some((group, _, _)), Some((pipeline, _))) =
731                            (&self.cable_bind, &self.cable_pipeline)
732                        else {
733                            continue;
734                        };
735                        pass.set_pipeline(pipeline);
736                        pass.set_bind_group(1, group, &[]);
737                    }
738                }
739                current = Some(batch.kind);
740            }
741            pass.draw(0..4, batch.start..batch.end);
742        }
743    }

745    fn color_for_target(&self, c: Color) -> [f32; 4] {
746        let [r, g, b, a] = if self.srgb_target { c.to_linear() } else { c.to_array() };
747        [r * a, g * a, b * a, a]
748    }

750    /// Builds instance lists and batches. Grows the atlas and starts over if it fills up.
751    fn prepare(&mut self, scene: &Scene, scale: f32) {
752        let max_atlas = self.device.limits().max_texture_dimension_2d.min(8192);
753        loop {
754            match self.try_prepare(scene, scale) {
755                Ok(()) => return,
756                Err(AtlasFull) if self.atlas.size < max_atlas => {
757                    let size = (self.atlas.size * 2).min(max_atlas);
758                    tracing::debug!(size, "growing glyph atlas");
759                    self.lose_atlas(size);
760                }
761                Err(AtlasFull) => {
762                    // Already at the device limit: start from an empty atlas. Text in
763                    // this frame that still does not fit is skipped and logged.
764                    tracing::warn!("glyph atlas is full at the device limit; clearing it");
765                    self.lose_atlas(self.atlas.size);
766                    let _ = self.try_prepare_lossy(scene, scale);
767                    return;
768                }
769            }
770        }
771    }
772
773    fn try_prepare(&mut self, scene: &Scene, scale: f32) -> Result<(), AtlasFull> {
774        self.build(scene, scale, false)
775    }
776
777    fn try_prepare_lossy(&mut self, scene: &Scene, scale: f32) -> Result<(), AtlasFull> {
778        self.build(scene, scale, true)
779    }
780
781    fn build(&mut self, scene: &Scene, scale: f32, skip_missing: bool) -> Result<(), AtlasFull> {
782        self.quads.clear();
783        self.glyphs.clear();
784        self.image_instances.clear();
785        self.cable_looks.clear();
786        self.cable_pictures.begin_frame();
787        self.batches.clear();
788        self.clips.clear();
789        self.breaks = BatchBreaks::default();
790        self.paths.begin_frame();
791        self.cable_plugs_pending.clear();
792        for primitive in scene.primitives() {
793            if !matches!(primitive, Primitive::Cable(_)) {
794                self.flush_cable_plugs(scale, skip_missing)?;
795            }
796            self.push_primitive(primitive, scale, skip_missing)?;
797        }
798        self.flush_cable_plugs(scale, skip_missing)
799    }
800
801    /// Draws the plugs held back for the run of cables that just ended.
802    fn flush_cable_plugs(&mut self, scale: f32, skip_missing: bool) -> Result<(), AtlasFull> {
803        if self.cable_plugs_pending.is_empty() {
804            return Ok(());
805        }
806        let plugs = std::mem::take(&mut self.cable_plugs_pending);
807        for primitive in &plugs {
808            self.push_primitive(primitive, scale, skip_missing)?;
809        }
810        // Keep the allocation for the next run.
811        self.cable_plugs_pending = plugs;
812        self.cable_plugs_pending.clear();
813        Ok(())
814    }

816    fn push_primitive(
817        &mut self,
818        primitive: &Primitive,
819        scale: f32,
820        skip_missing: bool,
821    ) -> Result<(), AtlasFull> {
822        match primitive {
823            Primitive::PushClip(rect) => {
824                let (w, h) = self.target_size;
825                let x0 = (rect.x() * scale).floor().clamp(0.0, w as f32) as u32;
826                let y0 = (rect.y() * scale).floor().clamp(0.0, h as f32) as u32;
827                let x1 = (rect.max_x() * scale).ceil().clamp(0.0, w as f32) as u32;
828                let y1 = (rect.max_y() * scale).ceil().clamp(0.0, h as f32) as u32;
829                let outer = self.current_scissor();
830                let ix0 = x0.max(outer[0]);
831                let iy0 = y0.max(outer[1]);
832                let ix1 = x1.min(outer[0] + outer[2]).max(ix0);
833                let iy1 = y1.min(outer[1] + outer[3]).max(iy0);
834                self.clips.push([ix0, iy0, ix1 - ix0, iy1 - iy0]);
835            }
836            Primitive::PopClip => {
837                self.clips.pop();
838            }
839            Primitive::Image(image) => self.push_image(image),
840            Primitive::Path(path) => {
841                let start = self.paths.shapes() as u32;
842                let scissor = self.current_scissor();
843                let limit = self.device.limits().max_texture_dimension_2d.min(8192);
844                self.paths.push(path, scale, scissor, limit);
845                self.extend_batch(BatchKind::Paths(self.paths.page()), start, self.paths.shapes() as u32);
846            }
847            Primitive::Quad(quad) => {
848                let start = self.quads.len() as u32;
849                push_quad(&mut self.quads, quad);
850                self.extend_batch(BatchKind::Quads, start, self.quads.len() as u32);
851            }
852            Primitive::Glyphs(run) => {
853                let start = self.glyphs.len() as u32;
854                let size_px = run.size * scale;
855                let size_q = (size_px * 4.0).round().max(1.0) as u32;
856                let color = run.color.to_array();
857                let inv_atlas = 1.0 / self.atlas.size as f32;
858                let (sin, cos) = run.rotation.sin_cos();
859                for g in &run.glyphs {
860                    let device_x = g.x * scale;
861                    let mut ix = device_x.floor();
862                    let mut bin = ((device_x - ix) * SUBPIXEL_BINS).round();
863                    if bin >= SUBPIXEL_BINS {
864                        ix += 1.0;
865                        bin = 0.0;
866                    }
867                    let key = GlyphKey {
868                        font_id: run.font.id,
869                        font_index: run.font.index,
870                        glyph: g.id,
871                        size_q,
872                        subpixel: bin as u8,
873                    };
874                    let entry = match self.atlas.get_or_insert(&self.queue, key, run.font.bytes()) {
875                        Lookup::Hit(entry) => entry,
876                        Lookup::Full if skip_missing => None,
877                        Lookup::Full => return Err(AtlasFull),
878                    };
879                    let Some(a) = entry else { continue };
880                    let device_y = (g.y * scale).round();
881                    self.glyphs.push(GlyphInstance {
882                        turn: [cos, sin, ix, device_y],
883                        rect: [ix + a.left as f32, device_y - a.top as f32, a.width as f32, a.height as f32],
884                        uv: [
885                            a.x as f32 * inv_atlas,
886                            a.y as f32 * inv_atlas,
887                            (a.x + a.width) as f32 * inv_atlas,
888                            (a.y + a.height) as f32 * inv_atlas,
889                        ],
890                        color,
891                    });
892                }
893                self.extend_batch(BatchKind::Glyphs, start, self.glyphs.len() as u32);
894            }
895            Primitive::Cable(cable) => self.push_cable(cable, scale, skip_missing)?,
896            // Primitives from a newer ravenrs-core this renderer does not know yet are skipped.
897            _ => {}
898        }
899        Ok(())
900    }

902    /// A cable the GPU world simulates is drawn from its buffer, a picture on it too;
903    /// any other as shapes.
904    fn push_cable(&mut self, cable: &Cable, scale: f32, skip_missing: bool) -> Result<(), AtlasFull> {
905        let pictures = &mut self.cable_pictures;
906        let gpu = self
907            .cables
908            .as_ref()
909            .and_then(|d| d.slot(cable))
910            .and_then(|(slot, n)| CableLookGpu::of(cable, slot, n, |image| pictures.place(image)));
911        let points: Vec<Point> = match &cable.points {
912            Some(points) => points.to_vec(),
913            None => cable_rest(cable),
914        };
915        let extra = match gpu {
916            Some(look) => {
917                let item = self.cable_looks.len() as u32;
918                self.cable_looks.push(look);
919                self.extend_batch(BatchKind::Cables, item * CABLE_SEGMENTS, (item + 1) * CABLE_SEGMENTS);
920                self.cable_plugs_pending.extend(cable_plugs(cable, &points));
921                return Ok(());
922            }
923            None => {
924                // Drawn as shapes: the plugs of the cables before it go first, in order.
925                self.flush_cable_plugs(scale, skip_missing)?;
926                cable_primitives(cable, &points)
927            }
928        };
929        for primitive in &extra {
930            self.push_primitive(primitive, scale, skip_missing)?;
931        }
932        Ok(())
933    }

935    fn push_image(&mut self, image: &ImageQuad) {
936        let layout = &self.atlas_layout;
937        self.image_cache.prepare(&self.device, &self.queue, layout, &self.image_sampler, &image.image);
938        let start = self.image_instances.len() as u32;
939        let r = image.rect;
940        self.image_instances.push(ImageInstance {
941            rect: [r.x(), r.y(), r.width(), r.height()],
942            uv: image.uv,
943            radii: image.corner_radii,
944            params: {
945                let (sin, cos) = image.rotation.sin_cos();
946                [image.opacity.clamp(0.0, 1.0), cos, sin, 0.0]
947            },
948        });
949        self.extend_batch(BatchKind::Image(image.image.id()), start, start + 1);
950    }
951
952    fn current_scissor(&self) -> [u32; 4] {
953        self.clips.last().copied().unwrap_or([0, 0, self.target_size.0, self.target_size.1])
954    }
955
956    fn extend_batch(&mut self, kind: BatchKind, start: u32, end: u32) {
957        if start == end {
958            return;
959        }
960        let scissor = self.current_scissor();
961        // A fully clipped batch draws nothing; scissor rects must not be empty anyway.
962        if scissor[2] == 0 || scissor[3] == 0 {
963            return;
964        }
965        match self.batches.last_mut() {
966            Some(last) if last.kind == kind && last.end == start && last.scissor == scissor => last.end = end,
967            Some(last) => {
968                if last.scissor != scissor {
969                    self.breaks.clip += 1;
970                } else if matches!((last.kind, kind), (BatchKind::Image(_), BatchKind::Image(_))) {
971                    self.breaks.image += 1;
972                } else {
973                    self.breaks.kind += 1;
974                }
975                self.batches.push(Batch { kind, start, end, scissor });
976            }
977            None => self.batches.push(Batch { kind, start, end, scissor }),
978        }
979    }

984fn push_quad(out: &mut Vec<QuadInstance>, quad: &Quad) {
985    let r = quad.rect;
986    let (sin, cos) = quad.rotation.sin_cos();
987    let turn = [cos, sin, 0.0, 0.0];
988    let rect = [r.x(), r.y(), r.width(), r.height()];
989    if let Some(shadow) = quad.shadow {
990        out.push(QuadInstance {
991            rect: [rect[0] + shadow.offset.x, rect[1] + shadow.offset.y, rect[2], rect[3]],
992            radii: quad.corner_radii,
993            color0: shadow.color.to_array(),
994            color1: [0.0; 4],
995            gradient: [0.0; 4],
996            border_color: [0.0; 4],
997            params: [0.0, shadow.blur * 0.5, 1.0, 0.0],
998            turn,
999        });
1000    }
1001    let (color0, color1, gradient, gradient_on) = match quad.fill {
1002        Brush::Solid(c) => (c.to_array(), c.to_array(), [0.0; 4], 0.0),
1003        Brush::LinearGradient(g) => {
1004            (g.from.to_array(), g.to.to_array(), [g.start.x, g.start.y, g.end.x, g.end.y], 1.0)
1005        }
1006        // Brushes from a newer ravenrs-core are drawn transparent rather than wrong.
1007        _ => ([0.0; 4], [0.0; 4], [0.0; 4], 0.0),
1008    };
1009    let (border_width, border_color) = quad.border.map_or((0.0, [0.0; 4]), |b| (b.width, b.color.to_array()));
1010    out.push(QuadInstance {
1011        rect,
1012        radii: quad.corner_radii,
1013        color0,
1014        color1,
1015        gradient,
1016        border_color,
1017        params: [border_width, 0.0, 0.0, gradient_on],
1018        turn,
1019    });
