1//! Design tokens and themes.
2//!
3//! Every value here mirrors `docs/design-system/tokens.json`, the source of truth
4//! that is also published as the RavenRS design system. Components never hard-code
5//! colors or sizes: they ask the [`Theme`] for a semantic token.

10/// Semantic color roles. Widgets refer to roles, never to raw colors, so every
11/// component follows the active theme.
12#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
13#[non_exhaustive]
14pub enum Semantic {
15    /// `bg-000`: the canvas behind everything.
16    Canvas,
17    /// `bg-100`: panels and sidebars.
18    Surface,
19    /// `bg-200`: cards.
20    SurfaceElevated,
21    /// `bg-300`: control fills.
22    Control,
23    /// `border`: hairlines and control outlines.
24    Border,
25    /// `ink`: primary text and icons.
26    Ink,
27    /// `ink-muted`: secondary text, captions and placeholders.
28    InkMuted,
29    /// `accent`: primary actions and selection.
30    Accent,
31    /// `accent-hover`: accent fill while hovered.
32    AccentHover,
33    /// `accent-pressed`: accent fill while pressed.
34    AccentPressed,
35    /// `on-accent`: text and icons drawn on an accent fill.
36    OnAccent,
37    /// `focus-ring`: keyboard focus indicator.
38    FocusRing,
39    /// `success`: positive status.
40    Success,
41    /// `warning`: cautionary status.
42    Warning,
43    /// `danger`: errors and destructive actions.
44    Danger,
45    /// `scrim`: the dimmed backdrop behind dialogs and sheets, the same in every theme.
46    Scrim,
47}

49/// The color tokens of one theme.
50#[derive(Clone, Copy, Debug, PartialEq)]
51pub struct Palette {
52    /// `bg-000`: the canvas behind everything.
53    pub bg_000: Color,
54    /// `bg-100`: panels and sidebars.
55    pub bg_100: Color,
56    /// `bg-200`: cards.
57    pub bg_200: Color,
58    /// `bg-300`: control fills.
59    pub bg_300: Color,
60    /// `border`: hairlines and control outlines.
61    pub border: Color,
62    /// `ink`: primary text and icons.
63    pub ink: Color,
64    /// `ink-muted`: secondary text, captions and placeholders.
65    pub ink_muted: Color,
66    /// `accent`: primary actions and selection.
67    pub accent: Color,
68    /// `accent-hover`: accent fill while hovered.
69    pub accent_hover: Color,
70    /// `accent-pressed`: accent fill while pressed.
71    pub accent_pressed: Color,
72    /// `on-accent`: text and icons drawn on an accent fill.
73    pub on_accent: Color,
74    /// `focus-ring`: keyboard focus indicator.
75    pub focus_ring: Color,
76    /// `success`: positive status.
77    pub success: Color,
78    /// `warning`: cautionary status.
79    pub warning: Color,
80    /// `danger`: errors and destructive actions.
81    pub danger: Color,
82}
83
84impl Palette {
85    /// Resolves a semantic role to this palette's color.
86    pub fn get(&self, role: Semantic) -> Color {
87        match role {
88            Semantic::Canvas => self.bg_000,
89            Semantic::Surface => self.bg_100,
90            Semantic::SurfaceElevated => self.bg_200,
91            Semantic::Control => self.bg_300,
92            Semantic::Border => self.border,
93            Semantic::Ink => self.ink,
94            Semantic::InkMuted => self.ink_muted,
95            Semantic::Accent => self.accent,
96            Semantic::AccentHover => self.accent_hover,
97            Semantic::AccentPressed => self.accent_pressed,
98            Semantic::OnAccent => self.on_accent,
99            Semantic::FocusRing => self.focus_ring,
100            Semantic::Success => self.success,
101            Semantic::Warning => self.warning,
102            Semantic::Danger => self.danger,
103            Semantic::Scrim => Color::rgba(0.0, 0.0, 0.0, 0.5),
104        }
105    }
106}

109#[derive(Clone, Copy, Debug, PartialEq)]
110pub struct Weight(pub f32);

112impl Weight {
113    /// Regular weight (400).

115    /// Medium weight (500).
116    pub const MEDIUM: Self = Self(500.0);
117    /// Semibold weight (600).
118    pub const SEMIBOLD: Self = Self(600.0);
119    /// Bold weight (700).
120    pub const BOLD: Self = Self(700.0);
121}

123/// One step of the type scale.
124#[derive(Clone, Copy, Debug, PartialEq)]
125pub struct TypeStyle {
126    /// Font size in logical pixels.
127    pub size: f32,
128    /// Line height in logical pixels.
129    pub line_height: f32,
130    /// Font weight.
131    pub weight: Weight,
132    /// Extra space between characters in logical pixels; may be negative.
133    pub letter_spacing: f32,
134}

136/// Named steps of the type scale.
137#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
138#[non_exhaustive]
139pub enum TextStyle {
140    /// Large dashboard numbers (42px).
141    Metric,
142    /// Page and hero headings (32px).
143    Display,
144    /// Section titles (22px).
145    Title,
146    /// Emphasized headings inside content (17px).
147    Headline,
148    /// Default running text (15px).
149    #[default]
150    Body,
151    /// Secondary, denser text (13px).
152    Callout,
153    /// Small labels and metadata (12px).
154    Caption,
155}
156
157/// The type scale: one [`TypeStyle`] per [`TextStyle`].
158#[derive(Clone, Copy, Debug, PartialEq)]
159pub struct Typography {
160    /// Style for [`TextStyle::Metric`].
161    pub metric: TypeStyle,
162    /// Style for [`TextStyle::Display`].
163    pub display: TypeStyle,
164    /// Style for [`TextStyle::Title`].
165    pub title: TypeStyle,
166    /// Style for [`TextStyle::Headline`].
167    pub headline: TypeStyle,
168    /// Style for [`TextStyle::Body`].
169    pub body: TypeStyle,
170    /// Style for [`TextStyle::Callout`].
171    pub callout: TypeStyle,
172    /// Style for [`TextStyle::Caption`].
173    pub caption: TypeStyle,
174}
175
176impl Typography {
177    /// Returns the type style for a named step.
178    pub fn get(&self, style: TextStyle) -> TypeStyle {
179        match style {
180            TextStyle::Metric => self.metric,
181            TextStyle::Display => self.display,
182            TextStyle::Title => self.title,
183            TextStyle::Headline => self.headline,
184            TextStyle::Body => self.body,
185            TextStyle::Callout => self.callout,
186            TextStyle::Caption => self.caption,
187        }
188    }
189}

191const fn ts(size: f32, line_height: f32, weight: Weight, letter_spacing: f32) -> TypeStyle {
192    TypeStyle { size, line_height, weight, letter_spacing }
193}

269/// Interaction sizing and motion preferences that components must respect.
270#[derive(Clone, Copy, Debug, PartialEq)]
271pub struct Interaction {
272    /// Minimum control height for pointer input.
273    pub min_target: f32,
274    /// Width of the keyboard focus ring.
275    pub focus_ring_width: f32,
276    /// Gap between a control and its focus ring.
277    pub focus_ring_offset: f32,
278    /// When true, components skip non-essential motion.
279    pub reduced_motion: bool,
280}

282/// Motion tokens: how long things take and how they move.
283#[derive(Clone, Copy, Debug, PartialEq)]
284pub struct MotionTokens {
285    /// Hover, press and focus color (`duration-fast`).
286    pub fast: Transition,
287    /// Small layout and opacity changes (`duration-base`).
288    pub base: Transition,
289    /// Panels and sheets entering, by time (`duration-slow`).
290    pub slow: Transition,
291    /// Things changing place or size (`spring-snappy`).
292    pub snappy: Transition,
293    /// Panels and sheets, by physics (`spring-gentle`).
294    pub gentle: Transition,
295}

297/// A complete theme: palette plus the shared scales.
298#[derive(Clone, Copy, Debug, PartialEq)]
299pub struct Theme {
300    /// Display name, e.g. `"Raven Dark"`.
301    pub name: &'static str,
302    /// Color tokens.
303    pub palette: Palette,
304    /// Type scale.
305    pub typography: Typography,
306    /// Spacing scale.
307    pub spacing: Spacing,
308    /// Corner radii.
309    pub radii: Radii,
310    /// Shadow tokens.
311    pub elevation: Elevation,
312    /// Surface material.
313    pub material: Material,
314    /// Interaction sizing and motion preferences.
315    pub interaction: Interaction,
316    /// Motion tokens.
317    pub motion: MotionTokens,
318}
319
320const INTERACTION: Interaction =
321    Interaction { min_target: 40.0, focus_ring_width: 2.0, focus_ring_offset: 3.0, reduced_motion: false };
322
323const MOTION: MotionTokens = MotionTokens {
324    fast: Transition::fast(),
325    base: Transition::base(),
326    slow: Transition::slow(),
327    snappy: Transition::snappy(),
328    gentle: Transition::gentle(),
329};

472    fn opaque_over(fg: Color, bg: Color) -> Color {
473        bg.lerp(fg.with_alpha(1.0), fg.a)

476    /// The design system promises 4.5:1 for text on its grounds in every theme.
477    #[test]
478    fn text_contrast_meets_wcag_aa() {
479        for theme in [Theme::raven_dark(), Theme::raven_light(), Theme::raven_glass()] {
480            let p = theme.palette;
481            let card = opaque_over(p.bg_200, p.bg_000);
482            for (name, fg, bg) in [
483                ("ink/canvas", p.ink, p.bg_000),
484                ("ink/card", p.ink, card),
485                ("ink-muted/canvas", p.ink_muted, p.bg_000),
486                ("ink-muted/card", p.ink_muted, card),
487                ("on-accent/accent", p.on_accent, p.accent),
488            ] {
489                let ratio = fg.contrast_ratio(bg);
490                assert!(ratio >= 4.5, "{}: {name} is {ratio:.2}:1", theme.name);
491            }
492            assert!(p.focus_ring.contrast_ratio(p.bg_000) >= 3.0, "{}: focus ring", theme.name);
493        }
494    }

496    #[test]
497    fn semantic_lookup_matches_palette() {
