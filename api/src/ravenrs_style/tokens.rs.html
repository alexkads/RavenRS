8/// The least contrast for a window to count as readable, well under what the design system asks for.
9const LEAST_CONTRAST: f32 = 3.0;
10/// The sizes of text that still fit a window.
11const TEXT_SIZES: std::ops::RangeInclusive<f32> = 9.0..=64.0;
12/// The sizes a radius can take when it comes from text.
13const RADIUS_RANGE: std::ops::RangeInclusive<f64> = 0.0..=64.0;
14/// The sizes text can take when it comes from text.
15const SIZE_RANGE: std::ops::RangeInclusive<f64> = 8.0..=96.0;
16/// The longest name a theme read from text can have, in characters.
17const NAME_MOST: usize = 64;
18
19/// The names of themes read from text, each kept once for the life of the program, so that a
20/// [`Theme`] stays `Copy`. Only distinct names are kept, and each is short.
21static NAMES: std::sync::Mutex<Vec<&'static str>> = std::sync::Mutex::new(Vec::new());
22
23/// `name` as a name a theme can hold.
24fn interned(name: &str) -> &'static str {
25    if let Some(builtin) = Theme::builtin().into_iter().find(|theme| theme.name == name) {
26        return builtin.name;
27    }
28    let mut names = NAMES.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
29    if let Some(known) = names.iter().find(|known| **known == name) {
30        return known;
31    }
32    let kept: &'static str = Box::leak(name.to_owned().into_boxed_str());
33    names.push(kept);
34    kept
35}

165#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
166impl Typography {
167    /// Changes the size of a step, with its line height in the same proportion.
168    pub fn set_size(&mut self, style: TextStyle, size: f32) {
169        let step = match style {
170            TextStyle::Metric => &mut self.metric,
171            TextStyle::Display => &mut self.display,
172            TextStyle::Title => &mut self.title,
173            TextStyle::Headline => &mut self.headline,
174            TextStyle::Body => &mut self.body,
175            TextStyle::Callout => &mut self.callout,
176            TextStyle::Caption => &mut self.caption,
177        };
178        let proportion = step.line_height / step.size.max(1.0);
179        step.size = size;
180        step.line_height = (size * proportion).round();
181    }
182}

184/// A color as `#rrggbb`, or `#rrggbbaa` when it lets something through.
185fn to_hex(color: Color) -> String {
186    let byte = |channel: f32| (channel.clamp(0.0, 1.0) * 255.0).round() as u8;
187    let (r, g, b, a) = (byte(color.r), byte(color.g), byte(color.b), byte(color.a));
188    if a == 255 { format!("#{r:02x}{g:02x}{b:02x}") } else { format!("#{r:02x}{g:02x}{b:02x}{a:02x}") }
189}
190
191fn from_hex(text: &str) -> Option<Color> {
192    let digits = text.strip_prefix('#')?;
193    let value = u32::from_str_radix(digits, 16).ok()?;
194    match digits.len() {
195        6 => Some(Color::hex(value)),
196        8 => Some(Color::hex_alpha(value)),
197        _ => None,
198    }
199}

201#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
202impl Theme {
203    /// The themes that come with RavenRS.
204    pub const fn builtin() -> [Self; 3] {
205        [Self::raven_dark(), Self::raven_light(), Self::raven_glass()]
206    }
207
208    /// The built-in theme this one started from, by its name; itself when it did not.
209    pub fn base(&self) -> Self {
210        let named = |name: &str| Self::builtin().into_iter().find(|theme| theme.name == name);
211        named(self.based_on).or_else(|| named(self.name)).unwrap_or(*self)
212    }
213
214    /// What differs from [`base`](Self::base), in the shape of [`to_tokens_json`](Self::to_tokens_json).
215    fn changes(&self) -> (Map<String, Value>, Map<String, Value>, Map<String, Value>) {
216        let base = self.base();
217        // A series of a chart counts only when the theme chose it: the others follow its text.
218        let chosen = Semantic::CHART_TOKENS.into_iter().filter(|role| self.charts.get(*role).is_some());
219        let colors = Semantic::TOKENS
220            .into_iter()
221            .chain(chosen)
222            .filter(|role| to_hex(self.color(*role)) != to_hex(base.color(*role)))
223            .map(|role| (role.token().to_owned(), json!(to_hex(self.color(role)))))
224            .collect();
225        let radii = Radii::TOKENS
226            .into_iter()
227            .filter(|token| self.radii.get(token) != base.radii.get(token))
228            .map(|token| (token.to_owned(), json!(self.radii.get(token))))
229            .collect();
230        let sizes = TextStyle::TOKENS
231            .into_iter()
232            .filter(|style| self.typography.get(*style).size != base.typography.get(*style).size)
233            .map(|style| (style.token().to_owned(), json!(self.typography.get(style).size)))
234            .collect();
235        (colors, radii, sizes)
236    }
237
238    /// Whether someone can read a window in this theme, and so find their way out of it. A theme
239    /// kept from another time that is not goes unused: a person can make anything while editing,
240    /// and should not open to a blank window because of it.
241    pub fn is_readable(&self) -> bool {
242        let palette = &self.palette;
243        let flat = |color: Color| palette.bg_000.lerp(color.with_alpha(1.0), color.a);
244        let grounds = [palette.bg_000, palette.bg_100, palette.bg_200];
245        let legible = |ink: Color, ground: Color| flat(ink).contrast_ratio(flat(ground)) >= LEAST_CONTRAST;
246        grounds.iter().all(|ground| legible(palette.ink, *ground) && legible(palette.ink_muted, *ground))
247            && legible(palette.on_accent, palette.accent)
248            && TextStyle::TOKENS.iter().all(|style| TEXT_SIZES.contains(&self.typography.get(*style).size))
249    }
250
251    /// How many tokens differ from [`base`](Self::base).
252    pub fn changed_tokens(&self) -> usize {
253        let (colors, radii, sizes) = self.changes();
254        colors.len() + radii.len() + sizes.len()
255    }
256
257    /// What differs from [`base`](Self::base) as JSON, with the same names as
258    /// `docs/design-system/tokens.json`, for keeping or handing over. A theme with a name of its
259    /// own writes it as `"name"`.
260    pub fn to_tokens_json(&self) -> String {
261        let (colors, radii, sizes) = self.changes();
262        let base = self.base().name;
263        let mut tokens = json!({ "base": base, "color": colors, "radius": radii, "type": sizes });
264        if self.name != base
265            && let Some(object) = tokens.as_object_mut()
266        {
267            object.insert("name".to_owned(), json!(self.name));
268        }
269        serde_json::to_string_pretty(&tokens).unwrap_or_default()
270    }
271
272    /// A theme from what [`to_tokens_json`](Self::to_tokens_json) wrote. `"name"` is optional and
273    /// shows in menus and to screen readers; without it the theme has the name of its base.
274    /// Anything it does not recognize, a name that is empty or longer than 64 characters, or a size
275    /// too far out to read, refuses the whole text, which may come from anywhere.
276    pub fn from_tokens_json(text: &str) -> Result<Self, String> {
277        let tokens: Value = serde_json::from_str(text).map_err(|error| error.to_string())?;
278        let name = tokens.get("base").and_then(Value::as_str).ok_or("no base theme")?;
279        let mut theme = Self::builtin()
280            .into_iter()
281            .find(|theme| theme.name == name)
282            .ok_or_else(|| format!("unknown base theme: {name}"))?;
283        if let Some(own) = tokens.get("name") {
284            let own = own.as_str().map(str::trim).ok_or("bad name")?;
285            if own.is_empty() || own.chars().count() > NAME_MOST {
286                return Err("bad name".to_owned());
287            }
288            theme.name = interned(own);
289        }
290        let part = |key: &str| tokens.get(key).and_then(Value::as_object).cloned().unwrap_or_default();
291        for (token, value) in part("color") {
292            let role = Semantic::TOKENS
293                .into_iter()
294                .chain(Semantic::CHART_TOKENS)
295                .find(|role| role.token() == token)
296                .ok_or_else(|| format!("unknown color: {token}"))?;
297            let color = value.as_str().and_then(from_hex).ok_or_else(|| format!("bad color: {token}"))?;
298            theme.set_color(role, color);
299        }
300        for (token, value) in part("radius") {
301            if !Radii::TOKENS.contains(&token.as_str()) {
302                return Err(format!("unknown radius: {token}"));
303            }
304            let radius = value
305                .as_f64()
306                .filter(|r| RADIUS_RANGE.contains(r))
307                .ok_or_else(|| format!("bad radius: {token}"))?;
308            theme.radii.set(&token, radius as f32);
309        }
310        for (token, value) in part("type") {
311            let style = TextStyle::TOKENS
312                .into_iter()
313                .find(|style| style.token() == token)
314                .ok_or_else(|| format!("unknown text size: {token}"))?;
315            let size = value
316                .as_f64()
317                .filter(|s| SIZE_RANGE.contains(s))
318                .ok_or_else(|| format!("bad text size: {token}"))?;
319            theme.typography.set_size(style, size as f32);
320        }
321        Ok(theme)
322    }
323}

329    #[test]
330    fn every_token_is_read_and_changed_by_its_name() {
331        for role in Semantic::TOKENS {
332            let mut palette = Theme::raven_dark().palette;
333            let color = Color::hex(0x123456);
334            palette.set(role, color);
335            assert_eq!(palette.get(role), color, "{}", role.token());
336        }
337        let mut radii = Theme::raven_dark().radii;
338        for token in Radii::TOKENS {
339            radii.set(token, 3.0);
340            assert_eq!(radii.get(token), Some(3.0));
341        }
342    }

344    #[test]
345    fn what_changed_goes_to_text_and_comes_back() {
346        let mut theme = Theme::raven_light();
347        assert_eq!(theme.changed_tokens(), 0);
348        theme.palette.set(Semantic::Accent, Color::hex(0xff0066));
349        theme.palette.set(Semantic::Surface, Color::hex_alpha(0x11223344));
350        theme.radii.set("md", 4.0);
351        theme.typography.set_size(TextStyle::Body, 16.0);
352        assert_eq!(theme.changed_tokens(), 4);
353        let again = Theme::from_tokens_json(&theme.to_tokens_json()).expect("reads");
354        assert_eq!(again.name, "Raven Light");
355        assert_eq!(to_hex(again.palette.accent), "#ff0066");
356        assert_eq!(to_hex(again.palette.bg_100), "#11223344");
357        assert_eq!(again.radii.md, 4.0);
358        assert_eq!(again.typography.body.size, 16.0);
359        assert_eq!(again.changed_tokens(), 4);
360    }

379    #[test]
380    fn the_series_of_a_chart_are_tokens_too() {
381        let mut theme = Theme { name: "Morning Star", ..Theme::raven_light() };
382        let before = theme.color(Semantic::Chart2);
383        theme.set_color(Semantic::Chart2, Color::hex(0xc2410c));

388        assert!(text.contains("chart-2"), "{text}");
389        assert!(!text.contains("chart-3"), "only what changed: {text}");
390        let again = Theme::from_tokens_json(&text).expect("reads");
391        assert_eq!(to_hex(again.color(Semantic::Chart2)), "#c2410c");
392        assert_eq!(again, theme);

394        assert_eq!(to_hex(Theme::from_tokens_json(raw).expect("reads").color(Semantic::Chart5)), "#123456");
