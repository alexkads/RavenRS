1//! Watching a session as it runs: what a film of a test needs (`ravenrs-movie`'s `SessionFilm`),
2//! and nothing a test that does not watch pays for.
3
4use ravenrs_core::{Color, Point, Primitive, Size};
5
6use crate::E2e;
7
8/// Told about what a session draws and does, in order: each frame it draws, where the pointer
9/// goes, each click and key, and each step a test marks. Every method but [`Self::frame`] does
10/// nothing unless implemented.
11pub trait FrameObserver {
12    /// A frame the session drew, as its primitives.
13    fn frame(&mut self, primitives: &[Primitive]);
14    /// The pointer moved to `at`.
15    fn pointer(&mut self, at: Point) {
16        let _ = at;
17    }
18    /// The primary button went down at `at`.
19    fn click(&mut self, at: Point) {
20        let _ = at;
21    }
22    /// A key was pressed.
23    fn key(&mut self) {}
24    /// A test began the step called `title`, as [`E2e::mark_step`] says.
25    fn step(&mut self, title: &str) {
26        let _ = title;
27    }
28}
29
30impl E2e {
31    /// Tells `observer` about everything the session draws and does from now on, starting with
32    /// the frame on the screen; it replaces any other.
33    pub fn observe(&mut self, mut observer: Box<dyn FrameObserver>) {
34        observer.frame(self.ui.scene().primitives());
35        self.observer = Some(observer);
36    }
37
38    /// Stops telling the observer, and gives it back.
39    pub fn stop_observing(&mut self) -> Option<Box<dyn FrameObserver>> {
40        self.observer.take()
41    }
42
43    /// Marks the start of a step called `title`, for an observer: a film makes it a chapter.
44    pub fn mark_step(&mut self, title: &str) {
45        if let Some(observer) = &mut self.observer {
46            observer.step(title);
47        }
48    }
49
50    /// The window's size, in logical pixels.
51    pub fn size(&self) -> Size {
52        self.size
53    }
54
55    /// The color behind everything the session draws.
56    pub fn background(&self) -> Color {
57        self.ui.scene().clear_color()
58    }
59
60    /// The current frame's display list, without advancing time or replacing the observer.
61    /// Protected fields are already drawn hidden, as they are on screen.
62    pub fn primitives(&self) -> &[Primitive] {
63        self.ui.scene().primitives()
64    }
65
66    pub(crate) fn observed_frame(&mut self) {
67        if let Some(observer) = &mut self.observer {
68            observer.frame(self.ui.scene().primitives());
69        }
70    }
71
72    pub(crate) fn observed_pointer(&mut self, at: Point) {
73        if let Some(observer) = &mut self.observer {
74            observer.pointer(at);
75        }
76    }
77
78    pub(crate) fn observed_click(&mut self, at: Point) {
79        if let Some(observer) = &mut self.observer {
80            observer.click(at);
81        }
82    }
83
84    pub(crate) fn observed_key(&mut self) {
85        if let Some(observer) = &mut self.observer {
86            observer.key();
87        }
88    }
89}
90
91/// Input as raw as a platform's, for whoever scripts a session step by step, such as
92/// `ravenrs-movie`'s `Recorder`, which paces a training as a person would: each call sends one
93/// event, tells the observer and draws what follows, with no checks on what is under it.
94impl E2e {
95    /// Moves the pointer to `at`.
96    pub fn pointer_to(&mut self, at: Point) {
97        self.move_to(at);
98    }
99
100    /// Presses the primary button at `at`.
101    pub fn mouse_down(&mut self, at: Point) {
102        self.pointer = at;
103        self.observed_click(at);
104        self.send(&ravenrs_core::Event::PointerDown {
105            position: at,
106            button: ravenrs_core::PointerButton::Primary,
107        });
108    }
109
110    /// Lets go of the primary button at `at`.
111    pub fn mouse_up(&mut self, at: Point) {
112        self.pointer = at;
113        self.send(&ravenrs_core::Event::PointerUp {
114            position: at,
115            button: ravenrs_core::PointerButton::Primary,
116        });
117    }
118
119    /// Presses `key` with `modifiers`, without letting it go.
120    pub fn key_down(&mut self, key: ravenrs_core::Key, modifiers: ravenrs_core::Modifiers) {
121        self.observed_key();
122        self.send(&ravenrs_core::Event::KeyDown { key, modifiers });
123    }
124
125    /// Turns the wheel by `delta` over `at`.
126    pub fn scroll_at(&mut self, at: Point, delta: Point) {
127        self.send(&ravenrs_core::Event::Scroll { position: at, delta });
128    }
129
130    /// Gives the keyboard focus to the node `id` of the tree, as assistive technology does.
131    pub fn focus(&mut self, id: u64) {
132        self.ui.perform_action(id, ravenrs_core::AccessAction::Focus);
133        self.frame();
134    }
135
136    /// Draws one frame at `seconds` on the session's clock, which only moves forward: what a
137    /// script that keeps its own pace calls once per frame of its film.
138    pub fn frame_at(&mut self, seconds: f64) {
139        self.clock = self.clock.max(seconds);
140        self.ui.set_time(self.clock);
141        self.frame();
142    }
143
144    /// Every node `query` finds, in reading order.
145    pub fn find_all(&self, query: &crate::Query) -> crate::E2eResult<Vec<ravenrs_core::AccessNode>> {
146        Ok(self.candidates(query)?.into_iter().cloned().collect())
147    }
148}
