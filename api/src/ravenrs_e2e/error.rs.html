1//! What can go wrong in a test, with a code that tools can rely on.
2
3use std::fmt;
4
5/// Why a step of a test failed. The codes are stable: tools, agents and reports match on them.
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
7#[non_exhaustive]
8pub enum ErrorCode {
9    /// Nothing on the screen matches the query.
10    NotFound,
11    /// More than one thing matches and the query does not say which.
12    Ambiguous,
13    /// The target does not respond to input.
14    Disabled,
15    /// Something else is on top of the target, so input would land there.
16    Covered,
17    /// The target is outside the window or has no size.
18    NotVisible,
19    /// The target's value is secret and is never given out.
20    ProtectedValue,
21    /// A check did not hold.
22    Expectation,
23    /// What a step waited for did not happen in time.
24    Timeout,
25    /// The target has no declared source location, or source instrumentation is disabled.
26    NoSource,
27}
28
29impl ErrorCode {
30    /// The code as reports write it, such as `E2E_NOT_FOUND`.
31    pub fn as_str(self) -> &'static str {
32        match self {
33            Self::NotFound => "E2E_NOT_FOUND",
34            Self::Ambiguous => "E2E_AMBIGUOUS",
35            Self::Disabled => "E2E_DISABLED",
36            Self::Covered => "E2E_COVERED",
37            Self::NotVisible => "E2E_NOT_VISIBLE",
38            Self::ProtectedValue => "E2E_PROTECTED_VALUE",
39            Self::Expectation => "E2E_EXPECTATION",
40            Self::Timeout => "E2E_TIMEOUT",
41            Self::NoSource => "E2E_NO_SOURCE",
42        }
43    }
44}
45
46impl fmt::Display for ErrorCode {
47    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
48        f.write_str(self.as_str())
49    }
50}
51
52/// A failed step: its code, what happened, how to fix it, what came close and the screen as it
53/// was, so the failure explains itself without running the test again.
54#[derive(Clone, Debug, PartialEq)]
55pub struct E2eError {
56    code: ErrorCode,
57    message: String,
58    hint: Option<String>,
59    similar: Vec<String>,
60    screen: String,
61}
62
63impl E2eError {
64    /// Whether waiting could still make it pass: the part may not be there yet, may still be
65    /// covered or disabled, or the check may not hold yet. An ambiguous query or a secret value
66    /// stays so however long one waits.
67    pub fn may_pass_later(&self) -> bool {
68        matches!(
69            self.code,
70            ErrorCode::NotFound
71                | ErrorCode::Covered
72                | ErrorCode::Disabled
73                | ErrorCode::NotVisible
74                | ErrorCode::Expectation
75        )
76    }
77
78    /// The same failure, as a wait that ran out after `waited`.
79    pub(crate) fn timed_out(mut self, waited: std::time::Duration) -> Self {
80        self.message = format!("waited {:.1} s and still: {}", waited.as_secs_f64(), self.message);
81        self.code = ErrorCode::Timeout;
82        if self.hint.is_none() {
83            self.hint = Some("wait longer, or check what the screen is waiting for".to_owned());
84        }
85        self
86    }
87
88    pub(crate) fn new(code: ErrorCode, message: impl Into<String>) -> Self {
89        Self { code, message: message.into(), hint: None, similar: Vec::new(), screen: String::new() }
90    }
91
92    /// With how to fix it, said after the message.
93    pub fn hint(mut self, hint: impl Into<String>) -> Self {
94        self.hint = Some(hint.into());
95        self
96    }
97
98    pub(crate) fn similar(mut self, similar: Vec<String>) -> Self {
99        self.similar = similar;
100        self
101    }
102
103    pub(crate) fn screen(mut self, screen: String) -> Self {
104        self.screen = screen;
105        self
106    }
107
108    /// The stable code.
109    pub fn code(&self) -> ErrorCode {
110        self.code
111    }
112
113    /// What happened.
114    pub fn message(&self) -> &str {
115        &self.message
116    }
117
118    /// How to fix it, when there is an obvious way.
119    pub fn fix(&self) -> Option<&str> {
120        self.hint.as_deref()
121    }
122
123    /// Up to three nodes that came close to what the query asked for, one line each.
124    pub fn close_matches(&self) -> &[String] {
125        &self.similar
126    }
127
128    /// The screen when it failed, as [`E2e::snapshot`](crate::E2e::snapshot) writes it.
129    pub fn screen_text(&self) -> &str {
130        &self.screen
131    }
132}
133
134impl fmt::Display for E2eError {
135    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
136        write!(f, "[{}] {}", self.code, self.message)?;
137        if let Some(hint) = &self.hint {
138            write!(f, "\n  fix: {hint}")?;
139        }
140        if !self.similar.is_empty() {
141            f.write_str("\n  close matches:")?;
142            for line in &self.similar {
143                write!(f, "\n    {line}")?;
144            }
145        }
146        if !self.screen.is_empty() {
147            f.write_str("\n  screen:")?;
148            for line in self.screen.lines() {
149                write!(f, "\n    {line}")?;
150            }
151        }
152        Ok(())
153    }
154}
155
156impl std::error::Error for E2eError {}
157
158/// The result of a test step.
159pub type E2eResult<T = ()> = Result<T, E2eError>;
