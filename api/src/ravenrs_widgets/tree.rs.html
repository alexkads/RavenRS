21/// One item of a [`TreeView`], with the items under it.
22#[derive(Clone, Debug, PartialEq)]
23pub struct TreeNode {
24    key: String,
25    label: String,
26    icon: Option<IconName>,
27    children: Vec<TreeNode>,
28    lazy: bool,
29    disabled: bool,
30}
31
32impl TreeNode {
33    /// An item the app knows by `key`, showing `label`.
34    pub fn new(key: impl Into<String>, label: impl Into<String>) -> Self {
35        Self {
36            key: key.into(),
37            label: label.into(),
38            icon: None,
39            children: Vec::new(),
40            lazy: false,
41            disabled: false,
42        }
43    }
44
45    /// An icon before the label.
46    pub fn icon(mut self, icon: IconName) -> Self {
47        self.icon = Some(icon);
48        self
49    }
50
51    /// Appends an item under this one.
52    pub fn child(mut self, child: TreeNode) -> Self {
53        self.children.push(child);
54        self
55    }
56
57    /// Appends items under this one.
58    pub fn children(mut self, children: impl IntoIterator<Item = TreeNode>) -> Self {
59        self.children.extend(children);
60        self
61    }
62
63    /// It can open although its items are not here yet; [`TreeView::on_expand`] brings them.
64    pub fn lazy(mut self) -> Self {
65        self.lazy = true;
66        self
67    }
68
69    /// It can be seen and opened, but not selected or checked.
70    pub fn disabled(mut self, disabled: bool) -> Self {
71        self.disabled = disabled;
72        self
73    }
74
75    fn can_open(&self) -> bool {
76        self.lazy || !self.children.is_empty()
77    }
78
79    fn matches(&self, query: &str) -> bool {
80        self.label.to_lowercase().contains(query) || self.children.iter().any(|child| child.matches(query))
81    }
82
83    /// The keys at the tips of the branch: what a check really holds.
84    fn collect_tips(&self, out: &mut Vec<String>) {
85        if self.children.is_empty() {
86            if !self.disabled {
87                out.push(self.key.clone());
88            }
89        } else {
90            for child in &self.children {
91                child.collect_tips(out);
92            }
93        }
94    }
95
96    fn check_state(&self, checked: &BTreeSet<String>) -> Toggled {
97        let mut tips = Vec::new();
98        self.collect_tips(&mut tips);
99        let on = tips.iter().filter(|tip| checked.contains(*tip)).count();
100        if on == 0 {
101            Toggled::False
102        } else if on == tips.len() {
103            Toggled::True
104        } else {
105            Toggled::Mixed
106        }
107    }
108}

110/// A line of the tree as it stands: open branches are followed, closed ones are not.
111#[derive(Clone, Debug)]
112struct Line {
113    node: TreeNode,
114    level: u32,
115    parent: Option<usize>,
116    open: bool,
117}
118
119fn lines(nodes: &[TreeNode], expanded: &BTreeSet<String>, query: &str) -> Vec<Line> {
120    fn walk(
121        nodes: &[TreeNode],
122        level: u32,
123        parent: Option<usize>,
124        expanded: &BTreeSet<String>,
125        query: &str,
126        out: &mut Vec<Line>,
127    ) {
128        for node in nodes {
129            if !query.is_empty() && !node.matches(query) {
130                continue;
131            }
132            // A filter has to show what it found, wherever it is.
133            let open = node.can_open()
134                && (expanded.contains(&node.key)
135                    || (!query.is_empty() && node.children.iter().any(|c| c.matches(query))));
136            let here = out.len();
137            out.push(Line { node: node.clone(), level, parent, open });
138            if open {
139                walk(&node.children, level + 1, Some(here), expanded, query, out);
140            }
141        }
142    }
143    let mut out = Vec::new();
144    walk(nodes, 1, None, expanded, query, &mut out);
145    out
146}

148/// A list of items that open to show more items: a chooser, a menu of a module, the
149/// structure of a document. Arrows move and open, Enter opens the item, Space checks it
150/// or selects it. With a [`height`](Self::height), only the visible lines are built.
151///
152/// ```
153/// use ravenrs_reactive::State;
154/// use ravenrs_widgets::{TreeNode, TreeView, Ui};
155///
156/// let selected = State::new(None::<String>);
157/// let tree = TreeView::new(vec![
158///     TreeNode::new("sales", "Sales").child(TreeNode::new("orders", "Orders")),
159///     TreeNode::new("stock", "Stock"),
160/// ])
161/// .selected(selected.binding());
162/// let mut ui = Ui::new(tree, ravenrs_style::Theme::raven_dark());
163/// ui.update();
164/// ```
165pub struct TreeView {
166    nodes: Vec<TreeNode>,
167    selected: Option<Binding<Option<String>>>,
168    expanded: Option<Binding<BTreeSet<String>>>,
169    checked: Option<Binding<BTreeSet<String>>>,
170    query: String,
171    on_open: Option<KeyHandler>,
172    on_expand: Option<KeyHandler>,
173    height: Option<f32>,
174    row_height: f32,
175    label: Option<String>,
176}
177
178impl fmt::Debug for TreeView {
179    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
180        f.debug_struct("TreeView").field("nodes", &self.nodes.len()).finish_non_exhaustive()
181    }
182}

184impl TreeView {
185    /// A tree of these items, all closed.
186    pub fn new(nodes: Vec<TreeNode>) -> Self {
187        Self {
188            nodes,
189            selected: None,
190            expanded: None,
191            checked: None,
192            query: String::new(),
193            on_open: None,
194            on_expand: None,
195            height: None,
196            row_height: 32.0,
197            label: None,
198        }
199    }
200
201    /// The item selected, by key.
202    pub fn selected(mut self, selected: Binding<Option<String>>) -> Self {
203        self.selected = Some(selected);
204        self
205    }
206
207    /// The items open, by key, to keep or to set from outside.
208    pub fn expanded(mut self, expanded: Binding<BTreeSet<String>>) -> Self {
209        self.expanded = Some(expanded);
210        self
211    }
212
213    /// Shows a check box on every line, bound to the keys of the items at the tips.
214    /// Checking a branch checks everything under it, and a branch partly under
215    /// check says so.
216    pub fn checked(mut self, checked: Binding<BTreeSet<String>>) -> Self {
217        self.checked = Some(checked);
218        self
219    }
220
221    /// Shows only the items with this text, and the branches that lead to them.
222    pub fn filter(mut self, text: impl Into<String>) -> Self {
223        self.query = text.into().trim().to_lowercase();
224        self
225    }
226
227    /// Runs with the key on Enter or on a double click.
228    pub fn on_open(mut self, handler: impl Fn(&str) + 'static) -> Self {
229        self.on_open = Some(Rc::new(handler));
230        self
231    }
232
233    /// Runs with the key when an item opens, to bring what is under a [`TreeNode::lazy`] one.
234    pub fn on_expand(mut self, handler: impl Fn(&str) + 'static) -> Self {
235        self.on_expand = Some(Rc::new(handler));
236        self
237    }
238
239    /// Scrolls inside this height, building only the lines in sight.
240    pub fn height(mut self, height: f32) -> Self {
241        self.height = Some(height);
242        self
243    }
244
245    /// Height of a line in logical pixels. 32 by default.
246    pub fn row_height(mut self, height: f32) -> Self {
247        self.row_height = height.max(24.0);
248        self
249    }
250
251    /// The name screen readers announce.
252    pub fn label(mut self, label: impl Into<String>) -> Self {
253        self.label = Some(label.into());
254        self
255    }

258#[derive(Clone)]
259struct TreeActions {
260    selected: Option<Binding<Option<String>>>,
261    expanded: Binding<BTreeSet<String>>,
262    checked: Option<Binding<BTreeSet<String>>>,
263    on_open: Option<KeyHandler>,
264    on_expand: Option<KeyHandler>,
265    active: State<Option<String>>,
266}
267
268impl TreeActions {
269    fn set_open(&self, key: &str, open: bool) {
270        self.expanded.update(|set| {
271            if open {
272                set.insert(key.to_owned());
273            } else {
274                set.remove(key);
275            }
276        });
277        if open && let Some(on_expand) = &self.on_expand {
278            on_expand(key);
279        }
280    }
281
282    fn select(&self, node: &TreeNode) {
283        self.active.set(Some(node.key.clone()));
284        if !node.disabled
285            && let Some(selected) = &self.selected
286        {
287            selected.set(Some(node.key.clone()));
288        }
289    }
290
291    fn toggle_check(&self, node: &TreeNode) {
292        let Some(checked) = &self.checked else { return };
293        if node.disabled {
294            return;
295        }
296        let mut tips = Vec::new();
297        node.collect_tips(&mut tips);
298        let all_on = node.check_state(&checked.get()) == Toggled::True;
299        checked.update(|set| {
300            for tip in tips {
301                if all_on {
302                    set.remove(&tip);
303                } else {
304                    set.insert(tip);
305                }
306            }
307        });
308    }
309
310    fn open_item(&self, node: &TreeNode) {
311        self.select(node);
312        if !node.disabled
313            && let Some(on_open) = &self.on_open
314        {
315            on_open(&node.key);
316        }
317    }
318}

320struct TreeLine<'a> {
321    line: &'a Line,
322    actions: &'a TreeActions,
323    active: bool,
324    focused: bool,
325    selected: bool,
326    check: Option<Toggled>,
327    row_height: f32,
328}
329
330impl View for TreeLine<'_> {
331    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
332        let theme = *cx.theme();
333        let Line { node, level, open, .. } = self.line;
334        let ink = theme.color(Semantic::Ink);
335        let muted = theme.color(Semantic::InkMuted);
336        let tone = |color: Color| if node.disabled { util::faded(color) } else { color };
337        let mut parts = Vec::new();
338        if node.can_open() {
339            let chevron = if *open { IconName::ChevronDown } else { IconName::ChevronRight };
340            let glyph = util::icon(cx, chevron, 16.0, muted);
341            let button = cx.container(
342                LayoutStyle {
343                    width: Some(20.0),
344                    height: Some(self.row_height),
345                    align: Align::Center,
346                    justify: Justify::Center,
347                    ..LayoutStyle::stack(Axis::Horizontal)
348                },
349                &[glyph],
350                Paint::None,
351            );
352            let (actions, key, open) = (self.actions.clone(), node.key.clone(), *open);
353            cx.on_click(button, Rc::new(move || actions.set_open(&key, !open)));
354            cx.semantics(button, Semantics::hidden());
355            parts.push(button);
356        } else {
357            parts.push(cx.container(
358                LayoutStyle { width: Some(20.0), ..LayoutStyle::leaf() },
359                &[],
360                Paint::None,
361            ));
362        }
363        if let Some(check) = self.check {
364            let state = util::EditorState { disabled: node.disabled, read_only: false };
365            let mark = match check {
366                Toggled::True => Some(IconName::Check),
367                Toggled::Mixed => Some(IconName::Minus),
368                _ => None,
369            }
370            .map(|icon| util::icon(cx, icon, 14.0, theme.color(Semantic::OnAccent)));
371            let square =
372                check_box_around(cx, &theme, !matches!(check, Toggled::False), false, state, mark.as_slice());
373            let (actions, item) = (self.actions.clone(), node.clone());
374            cx.on_click(square, Rc::new(move || actions.toggle_check(&item)));
375            cx.semantics(square, Semantics::hidden());
376            parts.push(square);
377        }
378        if let Some(icon) = node.icon {
379            parts.push(util::icon(cx, icon, 16.0, tone(muted)));
380        }
381        let text = cx.text_truncated(
382            &node.label,
383            &util::attrs(&theme, TextStyle::Body, None),
384            tone(ink),
385            LayoutStyle { grow: 1.0, min_width: Some(0.0), ..LayoutStyle::leaf() },
386        );
387        cx.semantics(text, Semantics::hidden());
388        parts.push(text);
389        let fill = if self.selected {
390            theme.color(Semantic::Accent).with_alpha(0.14)
391        } else if cx.is_hovered() {
392            ink.with_alpha(0.05)
393        } else {
394            Color::TRANSPARENT
395        };
396        let border = (self.active && self.focused)
397            .then(|| Border { width: 1.5, color: theme.color(Semantic::Accent) });
398        let row = cx.container(
399            LayoutStyle {
400                height: Some(self.row_height),
401                padding: Insets {
402                    left: 4.0 + (*level as f32 - 1.0) * LEVEL_INDENT,
403                    right: 8.0,
404                    ..Insets::ZERO
405                },
406                spacing: 6.0,
407                align: Align::Center,
408                align_self: Some(Align::Stretch),
409                ..LayoutStyle::stack(Axis::Horizontal)
410            },
411            &parts,
412            Paint::Quad(QuadPaint { corner_radius: theme.radii.sm, border, ..QuadPaint::new(fill) }),
413        );
414        let (select, item) = (self.actions.clone(), node.clone());
415        cx.on_click(row, Rc::new(move || select.select(&item)));
416        let (opener, item) = (self.actions.clone(), node.clone());
417        cx.on_double_click(row, Rc::new(move || opener.open_item(&item)));
418        cx.cursor(row, CursorIcon::Pointer);
419        let mut semantics = Semantics::new(Role::TreeItem)
420            .name(node.label.clone())
421            .level(*level)
422            .selected(self.selected)
423            .disabled(node.disabled)
424            .active(self.active);
425        if node.can_open() {
426            semantics = semantics.expanded(*open);
427        }
428        if let Some(check) = self.check {
429            semantics = semantics.toggled(check);
430        }
431        cx.semantics(row, semantics);
432        row
433    }
434}

436fn key_handler(
437    all: Rc<Vec<Line>>,
438    at: usize,
439    actions: TreeActions,
440    reveal: State<bool>,
441) -> crate::KeyHandler {
442    Rc::new(move |key, _| {
443        let Some(line) = all.get(at) else { return false };
444        let go = |index: usize| {
445            if let Some(line) = all.get(index) {
446                actions.active.set(Some(line.node.key.clone()));
447                reveal.set(true);
448            }
449        };
450        match key {
451            Key::ArrowDown => go((at + 1).min(all.len().saturating_sub(1))),
452            Key::ArrowUp => go(at.saturating_sub(1)),
453            Key::Home => go(0),
454            Key::End => go(all.len().saturating_sub(1)),
455            Key::ArrowRight if line.node.can_open() && !line.open => actions.set_open(&line.node.key, true),
456            Key::ArrowRight if line.open => go(at + 1),
457            Key::ArrowRight => {}
458            Key::ArrowLeft if line.open => actions.set_open(&line.node.key, false),
459            Key::ArrowLeft => {
460                if let Some(parent) = line.parent {
461                    go(parent);
462                }
463            }
464            Key::Enter => actions.open_item(&line.node),
465            Key::Space if actions.checked.is_some() => actions.toggle_check(&line.node),
466            Key::Space => actions.select(&line.node),
467            _ => return false,
468        }
469        true
470    })
471}

473impl View for TreeView {
474    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
475        let local: State<BTreeSet<String>> = cx.use_state(BTreeSet::new);
476        let expanded = self.expanded.clone().unwrap_or_else(|| local.binding());
477        let active: State<Option<String>> = cx.use_state(|| None);
478        let reveal: State<bool> = cx.use_state(|| false);
479        let all = Rc::new(lines(&self.nodes, &expanded.get(), &self.query));
480        let selected = self.selected.as_ref().and_then(Binding::get);
481        let checked = self.checked.as_ref().map(Binding::get);
482        // The keyboard's place: where it was, else the selection, else the top.
483        let wanted = active.get().or_else(|| selected.clone());
484        let at = wanted.and_then(|key| all.iter().position(|l| l.node.key == key)).unwrap_or(0);
485        let actions = TreeActions {
486            selected: self.selected.clone(),
487            expanded,
488            checked: self.checked.clone(),
489            on_open: self.on_open.clone(),
490            on_expand: self.on_expand.clone(),
491            active: active.clone(),
492        };
493        let rh = self.row_height;
494        let focused = cx.is_focused();
495        let offset = cx.scroll_offset();
496        let range = match self.height {
497            Some(height) => {
498                let viewport = cx.scroll_viewport().map_or(height, |size| size.height);
499                let top = at as f32 * rh;
500                if *reveal.peek() {
501                    reveal.set_if_changed(false);
502                    if top < offset.y {
503                        cx.set_scroll_offset(Point::new(0.0, top));
504                    } else if top + rh > offset.y + viewport {
505                        cx.set_scroll_offset(Point::new(0.0, top + rh - viewport));
506                    }
507                }
508                let first = ((offset.y / rh).floor() as usize).saturating_sub(OVERSCAN).min(all.len());
509                let last = (((offset.y + viewport) / rh).ceil() as usize + OVERSCAN).min(all.len());
510                first..last
511            }
512            None => 0..all.len(),
513        };
514        let mut rows = Vec::with_capacity(range.len() + 2);
515        rows.push(cx.container(
516            LayoutStyle { height: Some(range.start as f32 * rh), ..LayoutStyle::leaf() },
517            &[],
518            Paint::None,
519        ));
520        for i in range.clone() {
521            let line = &all[i];
522            let row = TreeLine {
523                line,
524                actions: &actions,
525                active: i == at,
526                focused,
527                selected: selected.as_deref() == Some(line.node.key.as_str()),
528                check: checked.as_ref().map(|set| line.node.check_state(set)),
529                row_height: rh,
530            };
531            rows.push(cx.keyed_child(util::hash_text(&line.node.key), &row));
532        }
533        rows.push(cx.container(
534            LayoutStyle {
535                height: Some(all.len().saturating_sub(range.end) as f32 * rh),
536                ..LayoutStyle::leaf()
537            },
538            &[],
539            Paint::None,
540        ));
541        let content = cx.container(
542            LayoutStyle { align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
543            &rows,
544            Paint::None,
545        );
546        let body = cx.container(
547            LayoutStyle {
548                height: self.height,
549                min_height: Some(0.0),
550                overflow_hidden: self.height.is_some(),
551                align: Align::Stretch,
552                align_self: Some(Align::Stretch),
553                ..LayoutStyle::stack(Axis::Vertical)
554            },
555            &[content],
556            Paint::None,
557        );
558        if self.height.is_some() {
559            cx.scrollable_virtual(body, false);
560        }
561        let name = self.label.clone().unwrap_or_else(|| cx.labels().tree_name.clone());
562        cx.semantics(body, Semantics::new(Role::Tree).name(name));
563        cx.focusable(body, 4.0);
564        cx.on_key(body, key_handler(all, at, actions, reveal));
565        body
566    }
567}

571    use super::*;

573    fn sample() -> Vec<TreeNode> {
574        vec![
575            TreeNode::new("a", "Alpha")
576                .child(TreeNode::new("a1", "One").child(TreeNode::new("deep", "Deep")))
577                .child(TreeNode::new("a2", "Two")),
578            TreeNode::new("b", "Beta"),
579        ]
580    }
581
582    #[test]
583    fn closed_branches_are_not_followed() {
584        let keys = |open: &[&str]| -> Vec<String> {
585            let open = open.iter().map(|s| (*s).to_owned()).collect();
586            lines(&sample(), &open, "").into_iter().map(|l| l.node.key).collect()
587        };
588        assert_eq!(keys(&[]), ["a", "b"]);
589        assert_eq!(keys(&["a"]), ["a", "a1", "a2", "b"]);
590        assert_eq!(keys(&["a", "a1"]), ["a", "a1", "deep", "a2", "b"]);
591        assert_eq!(keys(&["a1"]), ["a", "b"], "a closed parent hides what is open inside");
592    }
593
594    #[test]
595    fn a_filter_opens_the_way_to_what_it_finds() {
596        let found = lines(&sample(), &BTreeSet::new(), "deep");
597        let keys: Vec<_> = found.iter().map(|l| l.node.key.as_str()).collect();
598        assert_eq!(keys, ["a", "a1", "deep"]);
599        assert_eq!(found[2].parent, Some(1));
600        assert_eq!(found[2].level, 3);
601    }
602
603    #[test]
604    fn a_branch_is_checked_by_what_is_under_it() {
605        let root = &sample()[0];
606        let set = |keys: &[&str]| keys.iter().map(|s| (*s).to_owned()).collect::<BTreeSet<_>>();
607        assert_eq!(root.check_state(&set(&[])), Toggled::False);
608        assert_eq!(root.check_state(&set(&["deep"])), Toggled::Mixed);
609        assert_eq!(root.check_state(&set(&["deep", "a2"])), Toggled::True);
610    }
