13/// A [`ReleaseNotes`] drops its list of versions below this window width.
14const LIST_BREAKPOINT: f32 = 640.0;
15const LIST_WIDTH: f32 = 220.0;
16
17/// A line of text to copy, such as a command: shown apart from the prose, with a
18/// button that puts it on the clipboard.
19#[derive(Clone, Debug)]
20pub struct CodeBlock {
21    code: String,
22}
23
24impl CodeBlock {
25    /// A block showing `code`.
26    pub fn new(code: impl Into<String>) -> Self {
27        Self { code: code.into() }
28    }
29}
30
31impl View for CodeBlock {
32    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
33        let labels = cx.labels();
34        let copied = cx.use_state(|| false);
35        let clipboard = cx.clipboard();
36        let code = self.code.clone();
37        let done = copied.clone();
38        let (icon, label) = if copied.get() {
39            (IconName::Check, labels.copied.clone())
40        } else {
41            (IconName::Copy, labels.copy.clone())
42        };
43        let radius = cx.theme().radii.md;
44        let padding = cx.theme().spacing.s2;
45        cx.child(
46            &HStack::new((
47                Text::new(self.code.clone()).style(TextStyle::Caption).grow(1.0),
48                Button::new(label).icon(icon).style(ButtonStyle::Ghost).size(ButtonSize::Mini).on_click(
49                    move || {
50                        clipboard.write(&code);
51                        done.set(true);
52                    },
53                ),
54            ))
55            .spacing(padding)
56            .align(Align::Center)
57            .padding(padding)
58            .background(Semantic::Control)
59            .corner_radius(radius),
60        )
61    }
62}

64#[derive(Clone, Debug)]
65enum Block {
66    Section(String, String),
67    Code(String),
68}
69
70/// What one version brought, for a [`ReleaseNotes`].
71#[derive(Clone, Debug)]
72pub struct ReleaseNote {
73    version: String,
74    date: Option<String>,
75    title: String,
76    changes: Vec<String>,
77    blocks: Vec<Block>,
78    badge: Option<(String, Tone)>,
79}
80
81impl ReleaseNote {
82    /// A note on `version`, as the app words it, with the title of the release.
83    pub fn new(version: impl Into<String>, title: impl Into<String>) -> Self {
84        Self {
85            version: version.into(),
86            date: None,
87            title: title.into(),
88            changes: Vec::new(),
89            blocks: Vec::new(),
90            badge: None,
91        }
92    }
93
94    /// The release date, already formatted.
95    pub fn date(mut self, date: impl Into<String>) -> Self {
96        self.date = Some(date.into());
97        self
98    }
99
100    /// The list of what changed.
101    pub fn changes(mut self, changes: impl IntoIterator<Item = impl Into<String>>) -> Self {
102        self.changes = changes.into_iter().map(Into::into).collect();
103        self
104    }
105
106    /// Adds a block of text with a heading, after what is already there.
107    pub fn section(mut self, heading: impl Into<String>, text: impl Into<String>) -> Self {
108        self.blocks.push(Block::Section(heading.into(), text.into()));
109        self
110    }
111
112    /// Adds a [`CodeBlock`], after what is already there.
113    pub fn code(mut self, code: impl Into<String>) -> Self {
114        self.blocks.push(Block::Code(code.into()));
115        self
116    }
117
118    /// Marks the version, such as "Installed" or "New".
119    pub fn badge(mut self, label: impl Into<String>, tone: Tone) -> Self {
120        self.badge = Some((label.into(), tone));
121        self
122    }
123
124    /// The details on the right of the list.
125    fn details(&self) -> impl View + use<> {
126        let date = self.date.clone().map(|date| format!("{} · {date}", self.version));
127        let changes: Vec<Text> = self
128            .changes
129            .iter()
130            .map(|change| Text::new(format!("• {change}")).style(TextStyle::Callout))
131            .collect();
132        let blocks: Vec<crate::AnyView> = self
133            .blocks
134            .iter()
135            .map(|block| match block {
136                Block::Section(heading, text) => Box::new(
137                    VStack::new((
138                        Text::new(heading.clone()).style(TextStyle::Callout).font_weight(Weight::SEMIBOLD),
139                        Text::new(text.clone()).style(TextStyle::Callout).foreground(Semantic::InkMuted),
140                    ))
141                    .spacing(4.0),
142                ) as crate::AnyView,
143                Block::Code(code) => Box::new(CodeBlock::new(code.clone())) as crate::AnyView,
144            })
145            .collect();
146        VStack::new((
147            HStack::new((
148                Text::new(date.unwrap_or_else(|| self.version.clone()))
149                    .style(TextStyle::Caption)
150                    .foreground(Semantic::InkMuted),
151                self.badge.clone().map(|(label, tone)| Badge::new(label).tone(tone)),
152            ))
153            .spacing(8.0)
154            .align(Align::Center),
155            Text::new(self.title.clone()).style(TextStyle::Headline),
156            VStack::new(changes).spacing(6.0),
157            VStack::new(blocks).spacing(14.0).align(Align::Stretch),
158        ))
159        .spacing(14.0)
160        .align(Align::Stretch)
161    }
162}

164/// An app's versions, newest first: the list on the left, what the chosen one
165/// brought on the right, and buttons that walk from one to the next.
166///
167/// It shows whichever version `selected` points at, so the app opens it on the
168/// one it wants to talk about. It goes inside a [`Dialog`](crate::Dialog), or
169/// anywhere else with room for its height. In a narrow window the list goes away
170/// and the buttons remain.
171///
172/// ```
173/// use ravenrs_reactive::State;
174/// use ravenrs_widgets::{ReleaseNote, ReleaseNotes, Tone};
175///
176/// let selected = State::new(0);
177/// let notes = ReleaseNotes::new(
178///     vec![
179///         ReleaseNote::new("Version 1.3", "Faster start")
180///             .date("Nov 1, 2026")
181///             .badge("New", Tone::Success)
182///             .changes(["Opens in half the time"])
183///             .section("Why update", "Large files no longer stall the window."),
184///         ReleaseNote::new("Version 1.2", "Dark theme").badge("Installed", Tone::Neutral),
185///     ],
186///     selected.binding(),
187/// );
188/// ```
189#[derive(Debug)]
190pub struct ReleaseNotes {
191    notes: Vec<ReleaseNote>,
192    selected: Binding<usize>,
193    height: f32,
194}
195
196impl ReleaseNotes {
197    /// The versions in `notes`, newest first, on the one at `selected`.
198    pub fn new(notes: Vec<ReleaseNote>, selected: Binding<usize>) -> Self {
199        Self { notes, selected, height: 460.0 }
200    }
201
202    /// The height, in logical pixels (460 by default). Longer notes scroll.
203    pub fn height(mut self, height: f32) -> Self {
204        self.height = height;
205        self
206    }
207}
208
209impl View for ReleaseNotes {
210    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
211        let last = self.notes.len().saturating_sub(1);
212        let at = self.selected.get().min(last);
213        let Some(note) = self.notes.get(at) else {
214            return cx.empty();
215        };
216        let labels = cx.labels();
217        let list = (cx.viewport().width >= LIST_BREAKPOINT).then(|| {
218            let rows: Vec<ListItem> = self
219                .notes
220                .iter()
221                .enumerate()
222                .map(|(i, note)| {
223                    let line = [note.badge.as_ref().map(|(label, _)| label.clone()), note.date.clone()];
224                    let line = line.into_iter().flatten().collect::<Vec<_>>().join(" · ");
225                    let selected = self.selected.clone();
226                    let row = ListItem::new(note.version.clone())
227                        .selected(i == at)
228                        .on_click(move || selected.set(i));
229                    if line.is_empty() { row } else { row.subtitle(line) }
230                })
231                .collect();
232            ScrollView::new((List::new(rows),)).frame(Some(LIST_WIDTH), None)
233        });
234        let walk = |to: Option<usize>| {
235            let selected = self.selected.clone();
236            move || {
237                if let Some(to) = to {
238                    selected.set(to);
239                }
240            }
241        };
242        let (newer, older) = (at.checked_sub(1), (at < last).then_some(at + 1));
243        let body = VStack::new((
244            ScrollView::new((note.details(),)),
245            HStack::new((
246                Button::new(labels.release_newer.clone())
247                    .icon(IconName::ChevronLeft)
248                    .style(ButtonStyle::Ghost)
249                    .size(ButtonSize::Small)
250                    .disabled(newer.is_none())
251                    .on_click(walk(newer)),
252                Button::new(labels.release_older.clone())
253                    .icon(IconName::ChevronRight)
254                    .style(ButtonStyle::Ghost)
255                    .size(ButtonSize::Small)
256                    .disabled(older.is_none())
257                    .on_click(walk(older)),
258                Spacer::new(),
259                Text::new(format!("{} {} {}", at + 1, labels.release_of, self.notes.len()))
260                    .style(TextStyle::Caption)
261                    .foreground(Semantic::InkMuted),
262            ))
263            .spacing(4.0)
264            .align(Align::Center),
265        ))
266        .spacing(10.0)
267        .align(Align::Stretch)
268        .grow(1.0);
269        cx.child(&HStack::new((list, body)).spacing(16.0).align(Align::Stretch).height(self.height))
270    }
