20/// A tab bar with one content view per tab; only the selected one is built.

48    roving: util::Roving,
49    label: String,
50    selected: Binding<usize>,

53impl View for TabButton {
54    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
55        let theme = *cx.theme();
56        let on = self.selected.get() == self.roving.index;
57        let color =
58            if on || cx.is_hovered() { theme.color(Semantic::Ink) } else { theme.color(Semantic::InkMuted) };
59        let label = util::label(cx, &self.label, TextStyle::Body, Some(Weight::SEMIBOLD), color);
60        motion::control(cx, label);
61        let label = cx.container(
62            LayoutStyle { padding: Insets::symmetric(0.0, 2.0), ..LayoutStyle::stack(Axis::Horizontal) },
63            &[label],
64            Paint::None,
65        );
66        // The underline is one object that slides from tab to tab.
67        let marker = cx.container(
68            LayoutStyle { height: Some(2.0), align_self: Some(Align::Stretch), ..LayoutStyle::leaf() },
69            &[],
70            if on {
71                let accent = theme.color(Semantic::Accent);
72                Paint::Quad(QuadPaint { corner_radius: 1.0, ..QuadPaint::new(accent) })
73            } else {
74                Paint::None
75            },
76        );
77        if on {
78            cx.motion(marker, Motion::new().shared_layout(self.scope, 0, theme.motion.snappy));
79        }
80        let style = LayoutStyle {
81            padding: Insets { top: 8.0, ..Insets::ZERO },
82            spacing: 10.0,
83            ..LayoutStyle::stack(Axis::Vertical)
84        };
85        let node = cx.container(style, &[label, marker], Paint::None);
86        cx.semantics(node, Semantics::new(Role::Tab).selected(on));
87        let (selected, index) = (self.selected.clone(), self.roving.index);
88        cx.on_click(node, Rc::new(move || selected.set(index)));
89        // Tabs activate as focus moves (WAI-ARIA tabs, automatic activation).
90        let selected = self.selected.clone();
91        self.roving.wire(cx, node, Some(Rc::new(move |i| selected.set(i))));
92        cx.focusable(node, 6.0);
93        cx.cursor(node, CursorIcon::Pointer);
94        node
95    }
96}

98impl View for Tabs {
99    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
100        let border = cx.theme().color(Semantic::Border);
101        let (count, focus_to) = (self.tabs.len(), util::roving_cell(cx, None));
102        let scope = cx.widget_id();
103        let buttons: Vec<NodeId> = self
104            .tabs
105            .iter()
106            .enumerate()
107            .map(|(index, (label, _))| {
108                let roving = util::Roving { index, count, focus_to: focus_to.clone() };
109                cx.child(&TabButton { scope, roving, label: label.clone(), selected: self.selected.clone() })
110            })
111            .collect();
112        let bar = cx.container(
113            LayoutStyle {
114                spacing: 24.0,
115                align_self: Some(Align::Stretch),
116                ..LayoutStyle::stack(Axis::Horizontal)
117            },
118            &buttons,
119            Paint::Custom(Rc::new(move |f: Rect, scene| {
120                scene.push_quad(Quad::new(Rect::new(f.x(), f.max_y() - 1.0, f.width(), 1.0), border));
121            })),
122        );
123        cx.semantics(bar, Semantics::new(Role::TabList));
124        let selected = self.selected.get().min(self.tabs.len().saturating_sub(1));
125        let mut children = vec![bar];
126        if let Some((label, content)) = self.tabs.get(selected) {
127            // Keyed by tab so each tab keeps its own local state.
128            let content = cx.keyed_child(selected as u64, content.as_ref());
129            // A box that fills the space, so content that grows still does.
130            let panel = cx.container(
131                LayoutStyle {
132                    grow: 1.0,
133                    min_height: Some(0.0),
134                    align: Align::Stretch,
135                    ..LayoutStyle::stack(Axis::Vertical)
136                },
137                &[content],
138                Paint::None,
139            );
140            cx.semantics(panel, Semantics::new(Role::TabPanel).name(label.clone()));
141            children.push(panel);
142        }
143        cx.container(
144            LayoutStyle { spacing: 20.0, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
145            &children,
146            Paint::None,
147        )
148    }

152/// where it was when the sidebar is open: padding, row padding, icon, and back.

155/// What a [`Sidebar`] does when it closes.
156#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
157#[non_exhaustive]
158pub enum SidebarCollapse {
159    /// It stays open; there is nothing to toggle.
160    #[default]
161    Fixed,
162    /// It narrows to a rail of icons, with the labels as tooltips.
163    Icons,
164    /// It leaves the screen.
165    Hidden,
166}

168/// How a [`Sidebar`] sits next to the page.
169#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
170#[non_exhaustive]
171pub enum SidebarVariant {
172    /// A full-height column with a hairline toward the page.
173    #[default]
174    Standard,
175    /// A rounded, raised panel away from the window's edges.
176    Floating,
177    /// No surface of its own; the page sits in a [`SidebarInset`] beside it.
178    Inset,
179}

181/// Whether a [`Sidebar`] is open, shared with whatever toggles it.
182///
183/// A sidebar narrower windows show as a sheet keeps that apart: closing the
184/// sheet does not change what the sidebar does when the window grows again.
185#[derive(Clone)]
186pub struct SidebarState {
187    open: Binding<bool>,
188    sheet: State<bool>,
189    narrow: Rc<Cell<bool>>,
190}

192impl fmt::Debug for SidebarState {
193    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
194        f.debug_struct("SidebarState").field("narrow", &self.narrow.get()).finish_non_exhaustive()
195    }

198impl SidebarState {
199    /// A state that starts open or closed.
200    pub fn new(open: bool) -> Self {
201        Self::controlled(State::new(open).binding())
202    }
203
204    /// A state read from and written to `open`, for an app that saves or drives it.
205    pub fn controlled(open: Binding<bool>) -> Self {
206        Self { open, sheet: State::new(false), narrow: Rc::new(Cell::new(false)) }
207    }
208
209    /// Whether the sidebar is showing: its sheet in a narrow window, otherwise
210    /// the column at full width.
211    pub fn is_open(&self) -> bool {
212        if self.narrow.get() { self.sheet.get() } else { self.open.get() }
213    }
214
215    /// Opens or closes it.
216    pub fn set_open(&self, open: bool) {
217        if self.narrow.get() {
218            self.sheet.set(open);
219        } else {
220            self.open.set(open);
221        }
222    }
223
224    /// Opens it if closed, closes it if open.
225    pub fn toggle(&self) {
226        self.set_open(!self.is_open());
227    }
228
229    /// The window is narrow enough for this sidebar to be a sheet; see
230    /// [`Sidebar::narrow_below`]. Known once the sidebar has been built.
231    pub fn is_narrow(&self) -> bool {
232        self.narrow.get()
233    }
234}

238    icons: bool,
239    in_sheet: bool,
240    side: Side,
241    state: SidebarState,
242    focus_to: util::FocusTarget,
243    next: Cell<usize>,
244    count: Rc<Cell<usize>>,

247/// Up, Down, Home and End move between a sidebar's rows; Right and Left open
248/// and close a row that has content below it.
249fn wire_row_keys(
250    cx: &mut BuildCx<'_>,
251    node: NodeId,
252    scope: Option<&SidebarScope>,
253    open: Option<Binding<bool>>,
254) {
255    let place = scope.map(|s| {
256        let index = s.next.get();
257        s.next.set(index + 1);
258        if s.focus_to.take(index) {
259            cx.request_focus_node(node);
260        }
261        (index, s.focus_to.clone(), s.count.clone())
262    });
263    if place.is_none() && open.is_none() {
264        return;
265    }
266    cx.on_key(
267        node,
268        Rc::new(move |key, _| {
269            if let Some(open) = &open {
270                match key {
271                    Key::ArrowRight if !open.get() => {
272                        open.set(true);
273                        return true;
274                    }
275                    Key::ArrowLeft if open.get() => {
276                        open.set(false);
277                        return true;
278                    }
279                    _ => {}
280                }
281            }
282            let Some((index, focus_to, count)) = &place else { return false };
283            let (index, count) = (*index, count.get().max(1));
284            let target = match key {
285                Key::ArrowDown => (index + 1) % count,
286                Key::ArrowUp => (index + count - 1) % count,
287                Key::Home => 0,
288                Key::End => count - 1,
289                _ => return false,
290            };
291            focus_to.request(target);
292            true
293        }),
294    );
295}

297/// A navigation column: `Sidebar::new((SidebarSection::new("Workspace"), SidebarItem::new(...), ...))`.
298///
299/// Its children scroll between an optional header and footer. With
300/// [`Self::collapsible`] it closes to icons or off the screen, by Cmd/Ctrl+B or
301/// a [`SidebarTrigger`]. With [`Self::narrow_below`] small windows show it as a
302/// sheet over the page.
303///
304/// ```
305/// use ravenrs_widgets::*;
306///
307/// let state = SidebarState::new(true);
308/// let sidebar = Sidebar::new((
309///     SidebarGroup::new("Workspace", (
310///         SidebarItem::new(IconName::House, "Home").selected(true),
311///         SidebarItem::new(IconName::Users, "People").badge(Badge::new("3")),
312///     )),
313/// ))
314/// .collapsible(SidebarCollapse::Icons)
315/// .narrow_below(768.0)
316/// .state(state.clone());
317/// let trigger = SidebarTrigger::new(state);
318/// # let _ = (sidebar, trigger);
319/// ```
320pub struct Sidebar<C, P = ()> {
321    children: C,
322    width: f32,
323    collapse: SidebarCollapse,
324    variant: SidebarVariant,
325    side: Side,
326    narrow_below: Option<f32>,
327    shortcut: bool,
328    label: Option<String>,
329    parts: P,
330}

462/// A header or footer: laid out at full width, so closing only covers it, and
463/// on the rail what fits between the rail's margins shows.
464struct SidebarEnd<'a> {
465    content: &'a dyn View,
466    width: f32,
467}

469impl View for SidebarEnd<'_> {
470    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
471        let content = cx.child(self.content);
472        let inner = cx.container(
473            LayoutStyle {
474                width: Some((self.width - 24.0).max(0.0)),
475                align: Align::Stretch,
476                ..LayoutStyle::stack(Axis::Vertical)
477            },
478            &[content],
479            Paint::None,
480        );
481        let window = cx.container(
482            LayoutStyle {
483                grow: 1.0,
484                min_width: Some(0.0),
485                overflow_hidden: true,
486                ..LayoutStyle::stack(Axis::Horizontal)
487            },
488            &[inner],
489            Paint::None,
490        );
491        // Only on the rail: open, this edge would cut focus rings.
492        if cx.sidebar().is_some_and(|s| s.icons) {
493            cx.clip(window);
494        }
495        cx.container(
496            LayoutStyle { padding: Insets::uniform(12.0), ..LayoutStyle::stack(Axis::Horizontal) },
497            &[window],
498            Paint::None,
499        )
500    }

503/// Header, scrolling content and footer, and the scope their parts read.
504struct SidebarColumn<'a, C> {
505    children: &'a C,
506    parts: Option<&'a SidebarParts>,
507    scope: Rc<SidebarScope>,
508    width: f32,
509    /// The width the header and footer lay out at.
510    full_width: f32,
511    height: Option<f32>,
512}
513
514impl<C: ViewSeq> View for SidebarColumn<'_, C> {
515    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
516        let (header, footer) = self.parts.map_or((None, None), |p| (p.header.as_ref(), p.footer.as_ref()));
517        let outer = cx.replace_sidebar(Some(self.scope.clone()));
518        let mut children = Vec::new();
519        if let Some(header) = header {
520            children.push(cx.child(&SidebarEnd { content: header.as_ref(), width: self.full_width }));
521            children.push(cx.child(&Divider::horizontal()));
522        }
523        children.push(
524            cx.child(&ScrollView::new(self.children).spacing(2.0).padding(Insets::symmetric(16.0, 12.0))),
525        );
526        if let Some(footer) = footer {
527            children.push(cx.child(&Divider::horizontal()));
528            children.push(cx.child(&SidebarEnd { content: footer.as_ref(), width: self.full_width }));
529        }
530        cx.replace_sidebar(outer);
531        self.scope.count.set(self.scope.next.get());
532        cx.container(
533            LayoutStyle {
534                width: Some(self.width),
535                height: self.height,
536                align: Align::Stretch,
537                ..LayoutStyle::stack(Axis::Vertical)
538            },
539            &children,
540            Paint::None,
541        )
542    }
543}

557impl<C: ViewSeq, P> Sidebar<C, P> {
558    fn build_with(&self, cx: &mut BuildCx<'_>, parts: Option<&SidebarParts>) -> NodeId {
559        let theme = *cx.theme();
560        let own = cx.use_state(|| SidebarState::new(true));
561        let focus_to = util::roving_cell(cx, None);
562        let count = cx.use_state(|| Rc::new(Cell::new(0_usize))).get();
563        let state = parts.and_then(|p| p.state.clone()).unwrap_or_else(|| own.get());
564        let viewport = cx.viewport();
565        let narrow = self.narrow_below.is_some_and(|below| viewport.width < below);
566        state.narrow.set(narrow);
567        if !narrow && *state.sheet.peek() {
568            state.sheet.set(false);
569        }
570        if self.shortcut && (narrow || self.collapse != SidebarCollapse::Fixed) {
571            let toggle = state.clone();
572            cx.shortcut(Key::Character("b".into()), true, Rc::new(move || toggle.toggle()));
573        }
574        let margin = if self.variant == SidebarVariant::Floating { theme.spacing.s2 } else { 0.0 };
575        let open = state.open.get();
576        let target = match self.collapse {
577            _ if narrow => 0.0,
578            SidebarCollapse::Fixed => self.width,
579            _ if open => self.width,
580            SidebarCollapse::Icons => SIDEBAR_RAIL + 2.0 * margin,
581            SidebarCollapse::Hidden => 0.0,
582        };
583        // Reduced motion makes this instant. Labels stay until the column has
584        // narrowed, and the column's edge uncovers them on the way back.
585        let width = cx.animated(target, theme.motion.gentle);
586        let scope = Rc::new(SidebarScope {
587            icons: self.collapse == SidebarCollapse::Icons && !open && width <= target + 0.5,
588            in_sheet: narrow,
589            side: self.side,
590            state: state.clone(),
591            focus_to,
592            next: Cell::new(0),
593            count,
594        });
595        let label = self.label.clone().unwrap_or_else(|| cx.labels().navigation.clone());
596
597        if narrow {
598            let anchor = cx.empty();
599            if state.sheet.get() {
600                let panel = self.width.min(viewport.width);
601                let column = SidebarColumn {
602                    children: &self.children,
603                    parts,
604                    scope,
605                    width: panel,
606                    full_width: panel,
607                    height: Some(viewport.height),
608                };
609                let layer = SheetLayer {
610                    open: &state.sheet.binding(),
611                    title: &label,
612                    description: None,
613                    content: Some(&column),
614                    side: self.side,
615                    width: panel,
616                    bare: true,
617                };
618                let close = state.sheet.clone();
619                cx.overlay(anchor, Placement::Fill, &layer, Some(Rc::new(move || close.set(false))));
620            }
621            return anchor;
622        }
623        if target == 0.0 && width <= 0.5 {
624            return cx.empty();
625        }
626
627        let inset = margin.min(width / 2.0);
628        let card_width = (width - 2.0 * inset).max(0.0);
629        let full_width = (self.width - 2.0 * margin).max(0.0);
630        let column = SidebarColumn {
631            children: &self.children,
632            parts,
633            scope,
634            // Off-screen closing uncovers a column of constant width; closing to
635            // icons reflows it, so the icons can end up centered.
636            width: if self.collapse == SidebarCollapse::Hidden { full_width } else { card_width },
637            full_width,
638            height: None,
639        };
640        let column = cx.child(&column);
641        let (fill, border) = (theme.color(Semantic::Surface), theme.color(Semantic::Border));
642        let side = self.side;
643        let paint = match self.variant {
644            SidebarVariant::Inset => Paint::None,
645            SidebarVariant::Floating => Paint::Quad(QuadPaint {
646                corner_radius: theme.radii.lg,
647                border: Some(Border { width: 1.0, color: border }),
648                shadow: Some(theme.elevation.soft),
649                ..QuadPaint::new(fill)
650            }),
651            _ => Paint::Custom(Rc::new(move |f: Rect, scene| {
652                let x = if side == Side::Left { f.max_x() - 1.0 } else { f.x() };
653                scene.push_quad(Quad::new(f, fill));
654                scene.push_quad(Quad::new(Rect::new(x, f.y(), 1.0, f.height()), border));
655            })),
656        };
657        let card = cx.container(
658            LayoutStyle {
659                width: Some(card_width),
660                align: Align::Stretch,
661                align_self: Some(Align::Stretch),
662                overflow_hidden: true,
663                ..LayoutStyle::stack(Axis::Horizontal)
664            },
665            &[column],
666            paint,
667        );
668        cx.clip(card);
669        cx.semantics(card, Semantics::new(Role::Navigation).name(label));
670        if margin == 0.0 {
671            return card;
672        }
673        cx.container(
674            LayoutStyle {
675                width: Some(width),
676                padding: Insets::uniform(inset),
677                align: Align::Stretch,
678                align_self: Some(Align::Stretch),
679                ..LayoutStyle::stack(Axis::Horizontal)
680            },
681            &[card],
682            Paint::None,
683        )
684    }
685}

869struct SidebarGroupTitle<'a> {
870    title: &'a str,
871    open: Option<Binding<bool>>,
872}
873
874impl View for SidebarGroupTitle<'_> {
875    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
876        let theme = *cx.theme();
877        let hovered = self.open.is_some() && cx.is_hovered();
878        let color = theme.color(if hovered { Semantic::Ink } else { Semantic::InkMuted });
879        let attrs = util::attrs(&theme, TextStyle::Caption, None);
880        let text = cx.text_truncated(&self.title.to_uppercase(), &attrs, color, LayoutStyle::leaf());
881        let text = cx.container(
882            LayoutStyle { grow: 1.0, min_width: Some(0.0), ..LayoutStyle::stack(Axis::Horizontal) },
883            &[text],
884            Paint::None,
885        );
886        let mut children = vec![text];
887        if let Some(open) = &self.open {
888            let chevron = if open.get() { IconName::ChevronDown } else { IconName::ChevronRight };
889            children.push(util::icon(cx, chevron, 14.0, color));
890        }
891        let node = cx.container(
892            LayoutStyle {
893                grow: 1.0,
894                min_width: Some(0.0),
895                padding: Insets::symmetric(6.0, 10.0),
896                align: Align::Center,
897                spacing: theme.spacing.s2,
898                ..LayoutStyle::stack(Axis::Horizontal)
899            },
900            &children,
901            Paint::Quad(QuadPaint {
902                corner_radius: theme.radii.md,
903                ..QuadPaint::new(if hovered {
904                    theme.color(Semantic::Ink).with_alpha(0.05)
905                } else {
906                    Color::TRANSPARENT
907                })
908            }),
909        );
910        if let Some(open) = &self.open {
911            motion::control(cx, node);
912            cx.semantics(node, Semantics::new(Role::Button).name(self.title).expanded(open.get()));
913            let toggle = open.clone();
914            cx.on_click(node, Rc::new(move || toggle.update(|o| *o = !*o)));
915            cx.focusable(node, theme.radii.md);
916            cx.cursor(node, CursorIcon::Pointer);
917            let scope = cx.sidebar();
918            wire_row_keys(cx, node, scope.as_deref(), Some(open.clone()));
919        } else {
920            // The group around it already carries this name.
921            cx.semantics(node, Semantics::hidden());
922        }
923        node
924    }
925}

927struct SidebarRows<'a>(&'a dyn ViewSeq);

929impl View for SidebarRows<'_> {
930    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
931        let children = cx.children(self.0);
932        cx.container(

940impl<C: ViewSeq> View for SidebarGroup<C> {
941    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
942        let default = self.open_by_default;
943        let own = cx.use_state(move || default);
944        let open = self.open.clone().unwrap_or_else(|| own.binding());
945        let scope = cx.sidebar();
946        let icons = scope.as_ref().is_some_and(|s| s.icons);
947        let mut children = Vec::new();
948        if icons {
949            // Nothing to separate from above the first rows.
950            if scope.is_some_and(|s| s.next.get() > 0) {
951                children.push(rail_separator(cx));
952            }
953        } else {
954            let title = cx.child(&SidebarGroupTitle {
955                title: &self.title,
956                open: self.collapsible.then(|| open.clone()),
957            });
958            let mut parts = vec![title];
959            parts.extend(self.actions.iter().map(|action| cx.child(action.as_ref())));
960            children.push(cx.container(
961                LayoutStyle {
962                    padding: Insets { top: 10.0, ..Insets::ZERO },
963                    align: Align::Center,
964                    spacing: 2.0,
965                    ..LayoutStyle::stack(Axis::Horizontal)
966                },
967                &parts,
968                Paint::None,
969            ));
970        }
971        let unfolded = icons || !self.collapsible || open.get();
972        children.push(motion::collapse(cx, unfolded, &SidebarRows(&self.children)));
973        let node = cx.container(
974            LayoutStyle { spacing: 2.0, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
975            &children,
976            Paint::None,
977        );
978        cx.semantics(node, Semantics::new(Role::Group).name(self.title.clone()));
979        node
980    }
981}

1104/// The focusable part of a [`SidebarItem`].
1105struct SidebarButton<'a> {
1106    item: &'a SidebarItem,
1107    /// Whether the submenu is unfolded, for a row with one.
1108    expanded: Option<Binding<bool>>,
1109    /// Draws the row's fill itself: there is no action sharing it.
1110    filled: bool,
1111}
1112
1113/// A row's fill, shared by its button and the box around its actions.
1114fn row_paint(cx: &BuildCx<'_>, selected: bool, hovered: bool) -> Paint {
1115    let theme = cx.theme();
1116    let fill = if selected {
1117        theme.color(Semantic::Control)
1118    } else if hovered {
1119        theme.color(Semantic::Ink).with_alpha(0.05)
1120    } else {
1121        Color::TRANSPARENT
1122    };
1123    Paint::Quad(QuadPaint { corner_radius: theme.radii.md, ..QuadPaint::new(fill) })
1124}
1125
1126impl View for SidebarButton<'_> {
1127    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1128        let theme = *cx.theme();
1129        let item = self.item;
1130        let scope = cx.sidebar();
1131        let icons = scope.as_ref().is_some_and(|s| s.icons);
1132        let hovered = cx.is_hovered() && !item.disabled;
1133        let (ink, muted) = (theme.color(Semantic::Ink), theme.color(Semantic::InkMuted));
1134        let dim = |color: Color| if item.disabled { color.with_alpha(0.5) } else { color };
1135        let icon_color =
1136            dim(if item.selected { theme.color(Semantic::Accent).lerp(ink, 0.35) } else { muted });
1137        let text_style = if item.compact { TextStyle::Callout } else { TextStyle::Body };
1138        let mut children = Vec::new();
1139        if let Some(icon) = item.icon {
1140            children.push(util::icon(cx, icon, 16.0, icon_color));
1141        } else if icons {
1142            let letter = item.label.chars().next().map(String::from).unwrap_or_default();
1143            children.push(util::label(cx, &letter, TextStyle::Callout, Some(Weight::SEMIBOLD), icon_color));
1144        }
1145        let note = item.count.as_deref().or(item.badge.as_ref().map(Badge::text));
1146        if icons {
1147            if note.is_some() {
1148                children.push(cx.container(
1149                    LayoutStyle {
1150                        width: Some(6.0),
1151                        height: Some(6.0),
1152                        absolute: Some((24.0, 6.0)),
1153                        ..LayoutStyle::leaf()
1154                    },
1155                    &[],
1156                    Paint::Quad(QuadPaint {
1157                        corner_radius: 3.0,
1158                        ..QuadPaint::new(theme.color(Semantic::Accent))
1159                    }),
1160                ));
1161            }
1162        } else {
1163            let attrs = util::attrs(
1164                &theme,
1165                text_style,
1166                Some(if item.selected { Weight::SEMIBOLD } else { Weight::REGULAR }),
1167            );
1168            let color = dim(if item.selected || hovered { ink } else { muted });
1169            let label = cx.text_truncated(&item.label, &attrs, color, LayoutStyle::leaf());
1170            children.push(cx.container(
1171                LayoutStyle { grow: 1.0, min_width: Some(0.0), ..LayoutStyle::stack(Axis::Horizontal) },
1172                &[label],
1173                Paint::None,
1174            ));
1175            if let Some(count) = &item.count {
1176                children.push(util::label(cx, count, TextStyle::Caption, None, dim(muted)));
1177            }
1178            if let Some(badge) = &item.badge {
1179                children.push(cx.child(badge));
1180            }
1181            if let Some(expanded) = &self.expanded {
1182                let chevron = if expanded.get() { IconName::ChevronDown } else { IconName::ChevronRight };
1183                children.push(util::icon(cx, chevron, 14.0, dim(muted)));
1184            }
1185        }
1186        let style = LayoutStyle {
1187            padding: if icons {
1188                Insets::ZERO
1189            } else {
1190                Insets::symmetric(0.0, if item.compact { theme.spacing.s2 } else { 10.0 })
1191            },
1192            width: icons.then_some(SIDEBAR_RAIL - 24.0),
1193            height: Some(if item.compact { 32.0 } else { 36.0 }),
1194            spacing: if item.compact { theme.spacing.s2 } else { 10.0 },
1195            align: Align::Center,
1196            justify: if icons { Justify::Center } else { Justify::Start },
1197            grow: if self.filled { 0.0 } else { 1.0 },
1198            min_width: Some(0.0),
1199            ..LayoutStyle::stack(Axis::Horizontal)
1200        };
1201        let paint = if self.filled { row_paint(cx, item.selected, hovered) } else { Paint::None };
1202        let node = cx.container(style, &children, paint);
1203        if self.filled {
1204            motion::control(cx, node);
1205        }
1206        let mut semantics = Semantics::new(Role::Button)
1207            .name(item.label.clone())
1208            .selected(item.selected)
1209            .disabled(item.disabled);
1210        if let Some(note) = note {
1211            semantics = semantics.description(note);
1212        }
1213        if let Some(expanded) = &self.expanded {
1214            semantics = semantics.expanded(expanded.get());
1215        }
1216        cx.semantics(node, semantics);
1217        cx.cursor(node, if item.disabled { CursorIcon::Default } else { CursorIcon::Pointer });
1218        if !item.disabled {
1219            let handler = item.on_click.clone();
1220            let expanded = self.expanded.clone();
1221            let state = scope.as_ref().filter(|s| icons || s.in_sheet).map(|s| s.state.clone());
1222            if handler.is_some() || expanded.is_some() || state.is_some() {
1223                cx.on_click(
1224                    node,
1225                    Rc::new(move || {
1226                        match (&expanded, &state) {
1227                            // Reduced to icons there is no room to unfold into.
1228                            (Some(expanded), Some(state)) if icons => {
1229                                expanded.set(true);
1230                                state.set_open(true);
1231                            }
1232                            (Some(expanded), _) => expanded.update(|e| *e = !*e),
1233                            // A sheet is in the way of where the item leads.
1234                            (None, Some(state)) if !icons => state.set_open(false),
1235                            _ => {}
1236                        }
1237                        if let Some(handler) = &handler {
1238                            handler();
1239                        }
1240                    }),
1241                );
1242            }
1243            cx.focusable(node, theme.radii.md);
1244            let keys = self.expanded.clone().filter(|_| !icons);
1245            wire_row_keys(cx, node, scope.as_deref(), keys);
1246        }
1247        if let Some(scope) = scope.as_ref().filter(|s| s.icons)
1248            && (cx.is_hovered() || cx.is_focus_visible())
1249        {
1250            let placement =
1251                if scope.side == Side::Left { Placement::RightCenter } else { Placement::LeftCenter };
1252            cx.overlay(node, placement, &Bubble(&item.label), None);
1253        }
1254        node
1255    }
1256}

1261impl View for SidebarSubmenu<'_> {

1263        let border = cx.theme().color(Semantic::Border);
1264        let children: Vec<NodeId> = self.0.iter().map(|item| cx.child(item)).collect();

1283impl View for SidebarItem {
1284    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1285        let default = self.expanded_by_default.unwrap_or_else(|| self.holds_selection());
1286        let own = cx.use_state(move || default);
1287        let expanded = self.expanded.clone().unwrap_or_else(|| own.binding());
1288        let icons = cx.sidebar().is_some_and(|s| s.icons);
1289        let has_submenu = !self.submenu.is_empty();
1290        let filled = icons || self.actions.is_empty();
1291        let button =
1292            cx.child(&SidebarButton { item: self, expanded: has_submenu.then(|| expanded.clone()), filled });
1293        let row = if filled {
1294            button
1295        } else {
1296            let mut parts = vec![button];
1297            parts.extend(self.actions.iter().map(|action| cx.child(action.as_ref())));
1298            let paint = row_paint(cx, self.selected, cx.is_hover_within() && !self.disabled);
1299            let row = cx.container(
1300                LayoutStyle {
1301                    padding: Insets { right: 2.0, ..Insets::ZERO },
1302                    align: Align::Center,
1303                    spacing: 2.0,
1304                    ..LayoutStyle::stack(Axis::Horizontal)
1305                },
1306                &parts,
1307                paint,
1308            );
1309            motion::control(cx, row);
1310            row
1311        };
1312        if !has_submenu || icons {
1313            return row;
1314        }
1315        let submenu = motion::collapse(cx, expanded.get(), &SidebarSubmenu(&self.submenu));
1316        cx.container(
1317            LayoutStyle { align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
1318            &[row, submenu],
1319            Paint::None,
1320        )
1321    }

1324/// Placeholder rows while a sidebar's items load.
1325#[derive(Clone, Copy, Debug)]
1326pub struct SidebarSkeleton {
1327    rows: usize,
1328    icon: bool,
1329}
1330
1331impl SidebarSkeleton {
1332    /// This many placeholder rows.
1333    pub fn new(rows: usize) -> Self {
1334        Self { rows, icon: true }
1335    }
1336
1337    /// Whether the rows have a place for an icon (true by default).
1338    pub fn icon(mut self, icon: bool) -> Self {
1339        self.icon = icon;
1340        self
1341    }
1342}
1343
1344impl View for SidebarSkeleton {
1345    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1346        // Uneven lengths read as text; the same ones on each build stay still.
1347        const LENGTHS: [f32; 4] = [128.0, 88.0, 152.0, 104.0];
1348        let icons = cx.sidebar().is_some_and(|s| s.icons);
1349        let mut rows = Vec::with_capacity(self.rows);
1350        for i in 0..self.rows {
1351            let mut parts = Vec::new();
1352            if self.icon || icons {
1353                parts.push(cx.child(&Skeleton::block(16.0, 16.0)));
1354            }
1355            if !icons {
1356                parts.push(cx.child(&Skeleton::block(LENGTHS[i % LENGTHS.len()], 12.0)));
1357            }
1358            rows.push(cx.container(
1359                LayoutStyle {
1360                    height: Some(36.0),
1361                    padding: Insets::symmetric(0.0, 10.0),
1362                    spacing: 10.0,
1363                    align: Align::Center,
1364                    overflow_hidden: true,
1365                    ..LayoutStyle::stack(Axis::Horizontal)
1366                },
1367                &parts,
1368                Paint::None,
1369            ));
1370        }
1371        let name = cx.labels().loading.clone();
1372        let node = cx.container(
1373            LayoutStyle { spacing: 2.0, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
1374            &rows,
1375            Paint::None,
1376        );
1377        cx.semantics(node, Semantics::new(Role::Group).name(name));
1378        node
1379    }

1382/// A path of locations, the last one being the current page.
1383#[derive(Clone, Debug)]
1384pub struct Breadcrumb(Vec<String>);
1385
1386impl Breadcrumb {
1387    /// A breadcrumb of `items`, from the root to the current page.
1388    pub fn new(items: impl IntoIterator<Item = impl Into<String>>) -> Self {
1389        Self(items.into_iter().map(Into::into).collect())
1390    }
1391}
1392
1393impl View for Breadcrumb {
1394    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1395        let theme = *cx.theme();
1396        let mut children = Vec::new();
1397        for (i, item) in self.0.iter().enumerate() {
1398            let last = i + 1 == self.0.len();
1399            if i > 0 {
1400                children.push(util::icon(cx, IconName::ChevronRight, 14.0, theme.color(Semantic::InkMuted)));
1401            }
1402            let color = if last { theme.color(Semantic::Ink) } else { theme.color(Semantic::InkMuted) };
1403            let weight = if last { Some(Weight::SEMIBOLD) } else { None };
1404            children.push(util::label(cx, item, TextStyle::Callout, weight, color));
1405        }
1406        let node = cx.container(
1407            LayoutStyle { spacing: 6.0, align: Align::Center, ..LayoutStyle::stack(Axis::Horizontal) },
1408            &children,
1409            Paint::None,
1410        );
1411        cx.semantics(node, Semantics::new(Role::Group).name("Breadcrumb"));
1412        node
1413    }

1473        };
