11/// A grouped surface lifted off the canvas: `bg-200` fill, `radius-xl` corners,
12/// `shadow-soft` and a hairline border, padded by `space-6`.
13pub struct Card<V> {
14    content: V,
15    fill: Fill,
16    corner_radius: Option<f32>,
17    shadow: Option<Option<Shadow>>,
18    padding: Option<Insets>,
19    border: bool,
20    spacing: Option<f32>,
21    on_click: Option<Rc<dyn Fn()>>,
22}
23
24impl<V: fmt::Debug> fmt::Debug for Card<V> {
25    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
26        f.debug_struct("Card").field("content", &self.content).finish_non_exhaustive()
27    }
28}
29
30impl<V: View> Card<V> {
31    /// A card around `content` with the theme's surface, radius, shadow and padding.
32    pub fn new(content: V) -> Self {
33        Self {
34            content,
35            fill: Semantic::SurfaceElevated.into(),
36            corner_radius: None,
37            shadow: None,
38            padding: None,
39            border: true,
40            spacing: None,
41            on_click: None,
42        }
43    }
44
45    /// Corner radius in logical pixels (theme `radius-xl` by default).
46    pub fn corner_radius(mut self, radius: f32) -> Self {
47        self.corner_radius = Some(radius);
48        self
49    }
50
51    /// Replaces the theme's `shadow-soft`.
52    pub fn shadow(mut self, shadow: Shadow) -> Self {
53        self.shadow = Some(Some(shadow));
54        self
55    }
56
57    /// Removes the shadow.
58    pub fn no_shadow(mut self) -> Self {
59        self.shadow = Some(None);
60        self
61    }
62
63    /// Replaces the theme's card padding.
64    pub fn padding(mut self, padding: impl Into<Insets>) -> Self {
65        self.padding = Some(padding.into());
66        self
67    }
68
69    /// Background, as a theme role (`SurfaceElevated` by default) or a literal color.
70    pub fn fill(mut self, fill: impl Into<Fill>) -> Self {
71        self.fill = fill.into();
72        self
73    }
74
75    /// Shows or hides the hairline border (shown by default).
76    pub fn border(mut self, visible: bool) -> Self {
77        self.border = visible;
78        self
79    }
80
81    /// Makes the whole card a button: it runs `handler` when clicked, or on
82    /// Enter and Space while focused, and marks hover with the accent.
83    pub fn on_click(mut self, handler: impl Fn() + 'static) -> Self {
84        self.on_click = Some(Rc::new(handler));
85        self
86    }
87
88    fn paint(&self, cx: &BuildCx<'_>, fill: Color) -> QuadPaint {
89        let theme = cx.theme();
90        let active = self.on_click.is_some() && (cx.is_hovered() || cx.is_focus_visible());
91        QuadPaint {
92            fill: fill.into(),
93            corner_radius: self.corner_radius.unwrap_or(theme.radii.xl),
94            border: self.border.then(|| Border {
95                width: 1.0,
96                color: theme.color(if active { Semantic::Accent } else { Semantic::Border }),
97            }),
98            shadow: self.shadow.unwrap_or(Some(theme.elevation.soft)),
99        }
100    }
101
102    fn build_with(&self, cx: &mut BuildCx<'_>, fill: Color) -> NodeId {
103        let paint = self.paint(cx, fill);
104        let style = LayoutStyle {
105            padding: self.padding.unwrap_or_else(|| cx.theme().card_padding()),
106            spacing: self.spacing.unwrap_or(cx.theme().spacing.s2),
107            align: Align::Stretch,
108            ..LayoutStyle::stack(Axis::Vertical)
109        };
110        let corner_radius = paint.corner_radius;
111        let child = cx.child(&self.content);
112        let node = cx.container(style, &[child], Paint::Quad(paint));
113        if self.on_click.is_some() {
114            crate::motion::control(cx, node);
115        }
116        if let Some(handler) = &self.on_click {
117            cx.semantics(node, Semantics::new(Role::Button));
118            cx.on_click(node, handler.clone());
119            cx.focusable(node, corner_radius);
120            cx.cursor(node, CursorIcon::Pointer);
121        }
122        node
123    }
124}
125
126impl<V: View> View for Card<V> {
127    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
128        let fill = self.fill.resolve(cx.theme());
129        self.build_with(cx, fill)
130    }
131}

133/// A translucent card for Raven Glass compositions: a light, see-through fill
134/// and a brighter hairline over whatever is behind it.
135///
136/// Status: the backdrop blur of the glass material needs the renderer's layer
137/// compositor (planned); until then the card renders its translucent fill only.
138#[derive(Debug)]
139pub struct GlassCard<V> {
140    card: Card<V>,
141}
142
143impl<V: View> GlassCard<V> {
144    /// A glass card around `content`.
145    pub fn new(content: V) -> Self {
146        Self { card: Card::new(content).fill(Color::hex_alpha(0xffffff1a)) }
147    }
148
149    /// Corner radius in logical pixels (theme `radius-xl` by default).
150    pub fn corner_radius(mut self, radius: f32) -> Self {
151        self.card = self.card.corner_radius(radius);
152        self
153    }
154
155    /// Replaces the theme's `shadow-soft`.
156    pub fn shadow(mut self, shadow: Shadow) -> Self {
157        self.card = self.card.shadow(shadow);
158        self
159    }
160
161    /// Replaces the theme's card padding.
162    pub fn padding(mut self, padding: impl Into<Insets>) -> Self {
163        self.card = self.card.padding(padding);
164        self
165    }
166}

168impl<V: View> View for GlassCard<V> {

170        let fill = self.card.fill.resolve(cx.theme());
171        self.card.build_with(cx, fill)
