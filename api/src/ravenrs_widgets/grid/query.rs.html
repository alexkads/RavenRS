13/// What the user asked the grid to show.
14#[derive(Clone, Debug, Default, PartialEq)]
15pub struct GridQuery {
16    /// Sort columns in priority order.
17    pub sort: Vec<(usize, SortDirection)>,
18    /// Filter-row text per column (index = column): the short syntax of [`matches_filter`], or
19    /// a [`ravenrs_criteria`] expression over columns named by their titles, such as
20    /// `[Seats] > 10 And [Country] = 'Brazil'`.
21    pub filters: Vec<String>,
22    /// Text searched in every visible column.
23    pub search: String,
24    /// Grouping columns, outermost first.
25    pub group_by: Vec<usize>,
26    /// Keys of collapsed groups (see [`ViewRow::Group::key`]).
27    pub collapsed: BTreeSet<String>,
28    /// Hidden columns are ignored by search.
29    pub hidden: Vec<bool>,
30}
31
32impl GridQuery {
33    /// The filter text of `column` (zero-based), or `""`.
34    pub fn filter(&self, column: usize) -> &str {
35        self.filters.get(column).map_or("", String::as_str)
36    }
37
38    /// Sets the filter text of `column` (zero-based).
39    pub fn set_filter(&mut self, column: usize, text: String) {
40        if self.filters.len() <= column {
41            self.filters.resize(column + 1, String::new());
42        }
43        self.filters[column] = text;
44    }
45
46    /// The column (zero-based) is hidden.
47    pub fn is_hidden(&self, column: usize) -> bool {
48        self.hidden.get(column).copied().unwrap_or(false)
49    }
50
51    /// Plain click: sort by this column only, cycling ascending → descending →
52    /// none. With `add`, the column is added to (or cycled within) a multi-column sort.
53    pub fn toggle_sort(&mut self, column: usize, add: bool) {
54        let current = self.sort.iter().position(|(c, _)| *c == column);
55        let next = match current.map(|i| self.sort[i].1) {
56            None => Some(SortDirection::Ascending),
57            Some(SortDirection::Ascending) => Some(SortDirection::Descending),
58            Some(SortDirection::Descending) => None,
59        };
60        if !add {
61            self.sort.clear();
62            if let Some(d) = next {
63                self.sort.push((column, d));
64            }
65            return;
66        }
67        match (current, next) {
68            (Some(i), Some(d)) => self.sort[i].1 = d,
69            (Some(i), None) => {
70                self.sort.remove(i);
71            }
72            (None, Some(d)) => self.sort.push((column, d)),
73            (None, None) => {}
74        }
75    }
76
77    /// Adds the column to or removes it from the grouping; expands all groups.
78    pub fn toggle_group(&mut self, column: usize) {
79        if let Some(i) = self.group_by.iter().position(|c| *c == column) {
80            self.group_by.remove(i);
81        } else {
82            self.group_by.push(column);
83        }
84        self.collapsed.clear();
85    }
86
87    /// Collapses or expands the group with this key.
88    pub fn toggle_collapsed(&mut self, key: &str) {
89        if !self.collapsed.remove(key) {
90            self.collapsed.insert(key.to_owned());
91        }
92    }
93}

95/// A visible row of the grid: a group header or a data row.
96#[derive(Clone, Debug, PartialEq)]
97pub enum ViewRow {
98    /// A group header.
99    Group {
100        /// Nesting depth, 0 for the outermost groups.
101        level: usize,
102        /// The column grouped by (zero-based).
103        column: usize,
104        /// Display text of the group's value.
105        label: String,
106        /// Unique path of this group, used to collapse it.
107        key: String,
108        /// Number of data rows in the group.
109        count: usize,
110        /// The group's rows are hidden.
111        collapsed: bool,
112        /// Aggregates per column over the group's rows.
113        summaries: Vec<Option<f64>>,
114    },
115    /// Index into the data rows.
116    Data(usize),
117}

130/// Does `value` (shown as `display`) pass a filter-row expression?
131///
132/// Numbers accept `>`, `>=`, `<`, `<=`, `=`, `<>` and ranges `a..b`; everything
133/// else is a case-insensitive "contains". This is the short syntax of one cell; only
134/// [`run_query`] knows the columns that a filter-row expression names.
135pub fn matches_filter(value: &Value, display: &str, filter: &str) -> bool {
136    Filter::parse(filter).matches(value, || Cow::Borrowed(display))
137}
138
139type NumTest = fn(f64, f64) -> bool;
140
141/// A filter-row expression parsed once per query instead of once per row.
142#[derive(Debug)]
143struct Filter {
144    /// The comparison applied to numbers, when the text is one.
145    number: Option<NumberTest>,
146    /// The answer for booleans, when the text is a yes/no word.
147    boolean: Option<bool>,
148    /// The lowercased text for the "contains" fallback.
149    text: String,
150}
151
152#[derive(Debug)]
153enum NumberTest {
154    Range(f64, f64),
155    Compare(NumTest, f64),
156}
157
158impl Filter {
159    fn parse(filter: &str) -> Self {
160        let f = filter.trim();
161        let num = |s: &str| s.trim().replace(',', "").parse::<f64>().ok();
162        let mut number = None;
163        if let Some((a, b)) = f.split_once("..")
164            && let (Some(a), Some(b)) = (num(a), num(b))
165        {
166            number = Some(NumberTest::Range(a, b));
167        }
168        if number.is_none() {
169            let ops: [(&str, NumTest); 7] = [
170                (">=", |n, x| n >= x),
171                ("<=", |n, x| n <= x),
172                ("<>", |n, x| (n - x).abs() > f64::EPSILON),
173                ("!=", |n, x| (n - x).abs() > f64::EPSILON),
174                (">", |n, x| n > x),
175                ("<", |n, x| n < x),
176                ("=", |n, x| (n - x).abs() < 1e-9),
177            ];
178            number = ops.into_iter().find_map(|(op, test)| {
179                f.strip_prefix(op).and_then(num).map(|x| NumberTest::Compare(test, x))
180            });
181        }
182        let text = f.to_lowercase();
183        let boolean = match text.as_str() {
184            "yes" | "true" | "1" | "y" | "sim" => Some(true),
185            "no" | "false" | "0" | "n" | "não" | "nao" => Some(false),
186            _ => None,
187        };
188        Self { number, boolean, text }
189    }
190
191    /// `display` is only called when the value has to be matched as text.
192    fn matches<'a>(&self, value: &Value, display: impl FnOnce() -> Cow<'a, str>) -> bool {
193        if self.text.is_empty() {
194            return true;
195        }
196        match (value, &self.number, self.boolean) {
197            (Value::Number(n), Some(NumberTest::Range(a, b)), _) => *n >= *a && *n <= *b,
198            (Value::Number(n), Some(NumberTest::Compare(test, x)), _) => test(*n, *x),
199            (Value::Bool(b), _, Some(want)) => *b == want,
200            _ => contains_lowercase(&display(), &self.text),
201        }
202    }
203}

205/// The expression in a filter-row text, when it is one: it parses and every field it reads is a
206/// column title. Any other text, such as `[draft]`, stays a short filter, and so does anything a
207/// row cannot answer on its own: aggregates, parameters and functions the grid does not register.
208fn expression_of(filter: &str, titles: &HashMap<&str, usize>) -> Option<Criteria> {
209    if !filter.contains('[') {
210        return None;
211    }
212    let criteria = Criteria::parse(filter).ok()?;
213    let fields = criteria.properties();
214    let answerable = !criteria
215        .any(&|c| matches!(c, Criteria::Aggregate(_) | Criteria::Parameter(_) | Criteria::Custom(..)));
216    (answerable && !fields.is_empty() && fields.iter().all(|f| titles.contains_key(f.as_str())))
217        .then_some(criteria)
218}

220/// A grid row as expressions read it, with raw values rather than the text shown.
221struct GridRecord<'a> {
222    titles: &'a HashMap<&'a str, usize>,
223    row: &'a Row,
224}
225
226impl Record for GridRecord<'_> {
227    fn field(&self, path: &[String]) -> Option<ravenrs_criteria::Value> {
228        use ravenrs_criteria::Value as Field;
229        let column = match path {
230            [title] => self.titles.get(title.as_str()),
231            _ => self.titles.get(path.join(".").as_str()),
232        }?;
233        Some(match self.row.get(*column) {
234            Some(Value::Text(s)) => Field::Text(s.clone()),
235            Some(Value::Number(n)) => Field::Number(*n),
236            Some(Value::Bool(b)) => Field::Bool(*b),
237            Some(Value::Empty) | None => Field::Null,
238        })
239    }
240
241    fn collection(&self, _path: &[String]) -> Option<Vec<&Self>> {
242        None
243    }
244}

246/// Case-insensitive "contains" where `needle` is already lowercase. ASCII text,
247/// the common case, is compared in place without allocating.
248fn contains_lowercase(haystack: &str, needle: &str) -> bool {
249    if needle.is_empty() {
250        return true;
251    }
252    if haystack.is_ascii() && needle.is_ascii() {
253        let (h, n) = (haystack.as_bytes(), needle.as_bytes());
254        return h.windows(n.len()).any(|w| w.iter().zip(n).all(|(a, b)| a.to_ascii_lowercase() == *b));
255    }
256    haystack.to_lowercase().contains(needle)
257}

259/// A precomputed sort key, so sorting does not format or allocate per comparison.
260/// Text becomes its rank among the column's distinct lowercase texts.
261#[derive(Clone, Copy, Debug, PartialEq)]
262enum Key {
263    Empty,
264    Num(f64),
265    Text(u32),
266}
267
268/// Sort keys of `column` for the rows in `indices`, in the same order.
269fn keys_of(column: &GridColumn, c: usize, rows: &[Row], indices: &[usize]) -> Vec<Key> {
270    let empty = Value::Empty;
271    let mut texts: HashMap<Cow<'_, str>, u32> = HashMap::new();
272    let mut keys: Vec<Key> = indices
273        .iter()
274        .map(|&r| match rows[r].get(c).unwrap_or(&empty) {
275            Value::Number(n) => Key::Num(*n),
276            Value::Bool(b) => Key::Num(f64::from(u8::from(*b))),
277            v @ Value::Text(_) => {
278                let next = texts.len() as u32;
279                Key::Text(*texts.entry(column.display_cow(v)).or_insert(next))
280            }
281            _ => Key::Empty,
282        })
283        .collect();
284    if !texts.is_empty() {
285        // Distinct texts sorted case-insensitively; equal lowercase shares a rank.
286        let mut distinct: Vec<(String, u32)> = texts.iter().map(|(t, &id)| (t.to_lowercase(), id)).collect();
287        distinct.sort_unstable();
288        let mut rank_of = vec![0u32; distinct.len()];
289        let mut rank = 0;
290        for (i, (text, id)) in distinct.iter().enumerate() {
291            if i > 0 && *text != distinct[i - 1].0 {
292                rank += 1;
293            }
294            rank_of[*id as usize] = rank;
295        }
296        for key in &mut keys {
297            if let Key::Text(id) = key {
298                *id = rank_of[*id as usize];
299            }
300        }
301    }
302    keys
303}
304
305fn cmp_keys(a: &Key, b: &Key) -> Ordering {
306    match (a, b) {
307        (Key::Empty, Key::Empty) => Ordering::Equal,
308        // Empty values sort last in ascending order.
309        (Key::Empty, _) => Ordering::Greater,
310        (_, Key::Empty) => Ordering::Less,
311        (Key::Num(x), Key::Num(y)) => x.total_cmp(y),
312        (Key::Text(x), Key::Text(y)) => x.cmp(y),
313        (Key::Num(_), Key::Text(_)) => Ordering::Less,
314        (Key::Text(_), Key::Num(_)) => Ordering::Greater,
315    }
316}

318/// False when no number this column shows can contain `needle`, so search can
319/// skip formatting its numbers.
320fn numbers_may_contain(column: &GridColumn, needle: &str) -> bool {
321    if column.format.is_some() {
322        return true;
323    }
324    let (extra, thousands, decimal) = match &column.number {
325        Some(f) => (format!("{}{}", f.prefix, f.suffix).to_lowercase(), f.thousands, f.decimal),
326        None => (String::new(), ',', '.'),
327    };
328    needle.chars().all(|ch| {
329        ch.is_ascii_digit() || "-.infa".contains(ch) || ch == thousands || ch == decimal || extra.contains(ch)
330    })
331}

333fn summarize(summary: Summary, column: usize, rows: &[Row], indices: &[usize]) -> Option<f64> {
334    let values = indices.iter().filter_map(|&i| rows[i].get(column)).filter(|v| !v.is_empty());
335    match summary {
336        Summary::Count => Some(values.count() as f64),
337        Summary::Sum => Some(values.filter_map(Value::as_f64).sum()),
338        Summary::Average => {
339            let (sum, n) = values.filter_map(Value::as_f64).fold((0.0, 0usize), |(s, n), v| (s + v, n + 1));
340            (n > 0).then(|| sum / n as f64)
341        }
342        Summary::Min => values.filter_map(Value::as_f64).reduce(f64::min),
343        Summary::Max => values.filter_map(Value::as_f64).reduce(f64::max),
344    }
345}

347fn summaries(columns: &[GridColumn], rows: &[Row], indices: &[usize]) -> Vec<Option<f64>> {
348    columns
349        .iter()
350        .enumerate()
351        .map(|(c, col)| col.summary.and_then(|s| summarize(s, c, rows, indices)))
352        .collect()
353}

355/// Runs the whole pipeline.
356pub fn run_query(columns: &[GridColumn], rows: &[Row], query: &GridQuery) -> QueryResult {
357    let search = query.search.trim().to_lowercase();
358    // Reversed so that the first of two columns with one title wins.
359    let titles: HashMap<&str, usize> =
360        columns.iter().enumerate().rev().map(|(c, col)| (col.title.as_str(), c)).collect();
361    let mut active_filters: Vec<(usize, Filter)> = Vec::new();
362    let mut expressions: Vec<Criteria> = Vec::new();
363    for (c, f) in (0..columns.len()).map(|c| (c, query.filter(c))) {
364        if f.trim().is_empty() || !columns[c].filterable {
365            continue;
366        }
367        match expression_of(f, &titles) {
368            Some(expression) => expressions.push(expression),
369            None => active_filters.push((c, Filter::parse(f))),
370        }
371    }
372    let env = Environment::new();
373    let searched: Vec<(usize, bool)> = (0..columns.len())
374        .filter(|&c| !query.is_hidden(c))
375        .map(|c| (c, numbers_may_contain(&columns[c], &search)))
376        .collect();
377    let empty = Value::Empty;
378    let cell = |r: usize, c: usize| rows[r].get(c).unwrap_or(&empty);
379
380    let mut indices: Vec<usize> = (0..rows.len())
381        .filter(|&r| {
382            active_filters.iter().all(|(c, f)| {
383                let v = cell(r, *c);
384                f.matches(v, || columns[*c].display_cow(v))
385            })
386        })
387        .filter(|&r| {
388            expressions.iter().all(|expression| {
389                let record = GridRecord { titles: &titles, row: &rows[r] };
390                Scope::new(&env).row(&record).matches(expression)
391            })
392        })
393        .filter(|&r| {
394            search.is_empty()
395                || searched.iter().any(|&(c, numbers)| {
396                    let v = cell(r, c);
397                    (numbers || !matches!(v, Value::Number(_)))
398                        && contains_lowercase(&columns[c].display_cow(v), &search)
399                })
400        })
401        .collect();
402
403    // Group columns sort first (in their sort direction if they have one).
404    let mut order: Vec<(usize, SortDirection)> = query
405        .group_by
406        .iter()
407        .map(|&g| (g, query.sort.iter().find(|(c, _)| *c == g).map_or(SortDirection::Ascending, |s| s.1)))
408        .collect();
409    order.extend(query.sort.iter().filter(|(c, _)| !query.group_by.contains(c)).copied());
410    order.retain(|(c, _)| *c < columns.len());
411    if !order.is_empty() {
412        // Keys only for matching rows, indexed by position in `indices`.
413        let keys: Vec<Vec<Key>> =
414            order.iter().map(|&(c, _)| keys_of(&columns[c], c, rows, &indices)).collect();
415        let mut positions: Vec<usize> = (0..indices.len()).collect();
416        // Ties fall back to the original position, so the result matches a stable sort.
417        positions.sort_unstable_by(|&a, &b| {
418            for (k, (_, dir)) in order.iter().enumerate() {
419                let o = cmp_keys(&keys[k][a], &keys[k][b]);
420                if o != Ordering::Equal {
421                    return dir.apply(o);
422                }
423            }
424            a.cmp(&b)
425        });
426        indices = positions.into_iter().map(|p| indices[p]).collect();
427    }
428
429    let totals = summaries(columns, rows, &indices);
430    let mut out = Vec::with_capacity(indices.len());
431    let groups: Vec<usize> = query.group_by.iter().copied().filter(|&g| g < columns.len()).collect();
432    emit_groups(columns, rows, query, &groups, 0, &indices, "", &mut out);
433    QueryResult { rows: out, matched: indices, totals }
434}

436#[allow(clippy::too_many_arguments)]
437fn emit_groups(
438    columns: &[GridColumn],
439    rows: &[Row],
440    query: &GridQuery,
441    groups: &[usize],
442    level: usize,
443    indices: &[usize],
444    parent_key: &str,
445    out: &mut Vec<ViewRow>,
446) {
447    let Some(&column) = groups.get(level) else {
448        out.extend(indices.iter().map(|&i| ViewRow::Data(i)));
449        return;
450    };
451    let empty = Value::Empty;
452    let label_of = |r: usize| columns[column].display_cow(rows[r].get(column).unwrap_or(&empty));
453    let mut start = 0;
454    while start < indices.len() {
455        let label = label_of(indices[start]).into_owned();
456        let mut end = start + 1;
457        while end < indices.len() && *label_of(indices[end]) == *label {
458            end += 1;
459        }
460        let run = &indices[start..end];
461        let key = format!("{parent_key}\u{1f}{column}:{label}");
462        let collapsed = query.collapsed.contains(&key);
463        out.push(ViewRow::Group {
464            level,
465            column,
466            label: label.clone(),
467            key: key.clone(),
468            count: run.len(),
469            collapsed,
470            summaries: summaries(columns, rows, run),
471        });
472        if !collapsed {
473            emit_groups(columns, rows, query, groups, level + 1, run, &key, out);
474        }
475        start = end;
476    }
477}

479/// Rows as tab-separated text, the way spreadsheets copy and paste: the
480/// shown (formatted) values of the `visible` columns, no header.
481pub fn to_tsv(columns: &[GridColumn], rows: &[Row], visible: &[usize], selected: &[usize]) -> String {
482    let empty = Value::Empty;
483    let clean = |s: String| s.replace(['\t', '\n', '\r'], " ");
484    selected
485        .iter()
486        .filter_map(|&r| rows.get(r))
487        .map(|row| {
488            visible
489                .iter()
490                .map(|&c| clean(columns[c].display(row.get(c).unwrap_or(&empty))))
491                .collect::<Vec<_>>()
492                .join("\t")
493        })
494        .collect::<Vec<_>>()
495        .join("\n")
496}

498/// Matching rows as CSV (RFC 4180), visible columns only, in display order.
499pub fn to_csv(columns: &[GridColumn], rows: &[Row], query: &GridQuery, matched: &[usize]) -> String {
500    let visible: Vec<usize> = (0..columns.len()).filter(|&c| !query.is_hidden(c)).collect();
501    let mut out = String::new();
502    for (i, &c) in visible.iter().enumerate() {
503        if i > 0 {
504            out.push(',');
505        }
506        push_field(&mut out, &columns[c].title, ',');
507    }
508    out.push_str("\r\n");
509    let empty = Value::Empty;
510    for &r in matched {
511        for (i, &c) in visible.iter().enumerate() {
512            if i > 0 {
513                out.push(',');
514            }
515            let v = rows[r].get(c).unwrap_or(&empty);
516            // Raw numbers export unformatted so spreadsheets read them as numbers.
517            match v {
518                Value::Number(n) => out.push_str(&n.to_string()),
519                _ => push_field(&mut out, &columns[c].display_cow(v), ','),
520            }
521        }
522        out.push_str("\r\n");
523    }
524    out
525}
