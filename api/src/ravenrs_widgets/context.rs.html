21/// Stable identity of a widget across rebuilds, derived from its position in the
22/// view tree (or from an explicit key, see [`BuildCx::keyed_child`]). Hover,
23/// press and focus are remembered per `WidgetId`.
24#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
25pub struct WidgetId(u64);
26
27impl WidgetId {
28    pub(crate) const ROOT: Self = Self(0x5241_5645_4e52_5321);
29
30    fn child(self, key: u64) -> Self {
31        let mut h = self.0 ^ key.wrapping_mul(0x9e37_79b9_7f4a_7c15);
32        h = (h ^ (h >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
33        h = (h ^ (h >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
34        Self(h ^ (h >> 31))
35    }
36
37    /// The raw identity, for derived ids such as accessibility node ids.
38    pub(crate) fn raw(self) -> u64 {
39        self.0
40    }
41}

138    /// Which part of the drag this event is.
139    pub phase: DragPhase,
140    /// Pointer position in absolute logical pixels.
141    pub position: Point,

143    pub frame: Rect,
144    /// The frame of the dragged element's parent (e.g. a splitter's container).
145    pub container: Rect,

148impl Drag {
149    /// Horizontal position inside the frame, from 0.0 (left) to 1.0 (right).
150    pub fn fraction_x(&self) -> f32 {
151        if self.frame.width() <= 0.0 {
152            return 0.0;
153        }
154        ((self.position.x - self.frame.x()) / self.frame.width()).clamp(0.0, 1.0)
155    }
156}

230/// How a node appears to screen readers: its [`Role`], name and state.
231///
232/// Built-in components set this themselves. Custom components call
233/// [`BuildCx::semantics`] on the node they return:
234///
235/// ```
236/// use ravenrs_core::Role;
237/// use ravenrs_widgets::Semantics;
238///
239/// let s = Semantics::new(Role::Switch).name("Wi-Fi").toggled(true);
240/// assert_eq!(s.role(), Role::Switch);
241/// ```
242///
243/// Without semantics, a focusable node is announced as a button named by the
244/// text inside it, text is announced as a label, and other boxes are
245/// transparent: their children belong to the nearest described ancestor.
246#[derive(Clone, Debug, PartialEq)]
247pub struct Semantics {
248    pub(crate) role: Role,
249    pub(crate) name: Option<String>,
250    pub(crate) description: Option<String>,
251    pub(crate) value: Option<String>,
252    pub(crate) numeric: Option<NumericValue>,
253    pub(crate) toggled: Option<Toggled>,
254    pub(crate) selected: Option<bool>,
255    pub(crate) expanded: Option<bool>,
256    pub(crate) disabled: bool,
257    pub(crate) read_only: bool,
258    pub(crate) modal: bool,
259    pub(crate) level: Option<u32>,
260    pub(crate) hidden: bool,
261    pub(crate) active: bool,
262    /// The name is a guess (an icon's name) that a tooltip may replace.
263    pub(crate) fallback_name: bool,
264}

510/// Widget-local state that survives rebuilds, keyed by widget identity and
511/// call order (see [`BuildCx::use_state`]). Slots of widgets that were not
512/// built in the latest build are dropped.

515    slots: HashMap<(WidgetId, u32), Box<dyn Any>>,
516    used: HashSet<(WidgetId, u32)>,

519impl LocalStore {
520    pub fn begin_build(&mut self) {
521        self.used.clear();
522    }
523
524    pub fn end_build(&mut self) {
525        let used = &self.used;
526        self.slots.retain(|k, _| used.contains(k));
527    }

655    /// Builds what `build` builds with `color` as its [`Self::foreground`]. Layers opened
656    /// inside it start without one, since they are not painted over this view.
657    pub fn with_foreground<R>(&mut self, color: Color, build: impl FnOnce(&mut Self) -> R) -> R {

749    /// State owned by this view instance that survives rebuilds, like SwiftUI's
750    /// `@State`. Call it unconditionally and in the same order on every build;
751    /// `init` runs only the first time. The state is dropped when the view stops
752    /// being built.
753    pub fn use_state<T: 'static>(&mut self, init: impl FnOnce() -> T) -> State<T> {
754        let id = self.widget_id();
755        let slot = match self.scopes.last_mut() {
756            Some(scope) => {
757                scope.next_state += 1;
758                scope.next_state
759            }
760            None => 0,
761        };
762        let key = (id, slot);
763        self.local.used.insert(key);
764        if let Some(state) = self.local.slots.get(&key).and_then(|b| b.downcast_ref::<State<T>>()) {
765            return state.clone();
766        }
767        let state = State::new(init());
768        self.local.slots.insert(key, Box::new(state.clone()));
769        state
770    }

807    fn build_scoped(&mut self, key: u64, view: &dyn View) -> NodeId {
808        let id = self.widget_id().child(key);
809        self.scopes.push(Scope { id, next_child: 0, next_state: 0, next_motion: 0 });
810        let profile = self.view_profile.clone();
811        if let Some(profile) = &profile {
812            profile.borrow_mut().enter(view.type_name(), self.elements.len(), self.text.stats().shapes);
813        }
814        let node = view.build(self);
815        if let Some(profile) = &profile {
816            profile.borrow_mut().leave(self.elements.len(), self.text.stats().shapes);
817        }
818        self.scopes.pop();
819        node
820    }

916    /// The first described or focusable node at or under `node`, in tree
917    /// order: the control a label or tooltip around it is about.
918    fn control_in(&self, node: NodeId) -> Option<usize> {
919        let mut stack = vec![node];
920        while let Some(n) = stack.pop() {
921            let e = self.elements.get(n.0 as usize)?;
922            if e.semantics.as_deref().is_some_and(|s| !s.hidden) || e.focus_radius.is_some() {
923                return Some(n.0 as usize);
924            }
925            stack.extend(e.children.iter().rev());
926        }
927        None
928    }

930    /// Names the control at or under `node` with `name` unless it already has
931    /// a name of its own, and sets `description` (form fields, tooltips).
932    /// When it has a name, `name` becomes its description instead.
933    pub(crate) fn label_control(&mut self, node: NodeId, name: &str, description: Option<&str>) {
934        let Some(i) = self.control_in(node) else { return };
935        let e = &mut self.elements[i];
936        let s = e.semantics.get_or_insert_with(|| Box::new(Semantics::new(Role::Button)));
937        if s.name.is_none() || s.fallback_name {
938            s.name = Some(name.to_owned());
939            s.fallback_name = false;
940            if let Some(d) = description {
941                s.description = Some(d.to_owned());
942            }
943        } else if s.name.as_deref() != Some(name) {
944            s.description = Some(description.map_or_else(|| name.to_owned(), |d| format!("{name}. {d}")));
945        }
946    }

1036    /// Declares how `node` animates; see [`Motion`]. Call it in the same order on
1037    /// every build, or give the motion a [`Motion::key`].
1038    pub fn motion(&mut self, node: NodeId, motion: Motion) {
1039        let id = self.widget_id();
1040        let key = match motion.shared {
1041            Some((scope, tag)) => MotionKey::Shared(scope, tag),
1042            None => match (motion.key, self.scopes.last_mut()) {
1043                (Some(k), _) => MotionKey::Node(id, k | (1 << 63)),
1044                (None, Some(scope)) => {
1045                    scope.next_motion += 1;
1046                    MotionKey::Node(id, scope.next_motion)
1047                }
1048                (None, None) => MotionKey::Node(id, 0),
1049            },
1050        };
1051        let index = self.motions.len() as u32;
1052        if let Some(e) = self.element_mut(node) {
1053            e.motion = Some(index);
1054            self.motions.push(MotionEntry { node, key, motion });
1055        }
1056    }

1058    /// A value that eases toward `target`, for what [`Motion`] does not cover:
1059    /// numbers that feed layout or custom drawing. Call it in the same order on
1060    /// every build. It rebuilds the UI on each frame while it moves, so prefer
1061    /// [`Motion`] when you can.
1062    pub fn animated<T: Animatable + 'static>(&mut self, target: T, transition: Transition) -> T {
1063        let cell = self.use_state(|| Rc::new(RefCell::new(Animated::new(target))));
1064        let cell = cell.peek().clone();
1065        let mut value = cell.borrow_mut();
1066        let now = self.motion_clock;
1067        let transition =
1068            if !self.motion_enabled || self.reduced_motion { Transition::Instant } else { transition };
1069        value.set_target(target, transition, now);
1070        if value.step(now) {
1071            self.value_moving = true;
1072            self.wake_at(self.now);
1073        }
1074        value.value(now)
1075    }

1235    /// Builds `view` as a floating layer above the rest of the UI, placed relative
1236    /// to `anchor`. `on_dismiss` is called on a click outside the layer and its
1237    /// anchor, or on Escape.
1238    pub fn overlay(
1239        &mut self,
1240        anchor: NodeId,
1241        placement: Placement,
1242        view: &dyn View,
1243        on_dismiss: Option<Rc<dyn Fn()>>,
1244    ) {
1245        // The slot comes before the content: overlays are placed and painted in
1246        // order, so one opened from inside this one must come after it, or it
1247        // would be placed with no anchor yet and painted underneath.
1248        let key = 0x6f76_6572_6c61_7900 + self.overlays.len() as u64;
1249        self.keyed_overlay(key, anchor, placement, view, on_dismiss);
1250    }

1252    /// Like [`Self::overlay`], for a layer that must stay the same one when other
1253    /// layers before it come and go. A plain overlay is told apart by its place among
1254    /// them, so a new one before it makes it start over, and animate in again.
1255    pub fn keyed_overlay(
1256        &mut self,
1257        key: u64,
1258        anchor: NodeId,
1259        placement: Placement,
1260        view: &dyn View,
1261        on_dismiss: Option<Rc<dyn Fn()>>,
1262    ) {
1263        let slot = self.overlays.len();
1264        let gap = self.theme.spacing.s2;
1265        self.overlays.push(Overlay { root: anchor, anchor, placement, gap, on_dismiss, floating: false });
1266        let outer = self.foreground.take();
1267        let root = self.keyed_child(key, view);
1268        self.foreground = outer;
1269        self.overlays[slot].root = root;
1270    }

1272    /// Builds `view` as a layer that stays open beside the UI, like a chat in a
1273    /// corner: it is under every menu and dialog, wherever it is built, keeps its state
1274    /// when they open, and is not closed by a click outside it.
1275    pub fn floating_overlay(&mut self, anchor: NodeId, placement: Placement, view: &dyn View) {
1276        // Its key does not depend on the other layers, which come and go.
1277        let key = 0x666c_6f61_7400_0000 + self.floating_overlays;
1278        self.floating_overlays += 1;
1279        let slot = self.overlays.len();
1280        let gap = self.theme.spacing.s2;
1281        self.overlays.push(Overlay {
1282            root: anchor,
1283            anchor,
1284            placement,
1285            gap,
1286            on_dismiss: None,
1287            floating: true,
1288        });
1289        let outer = self.foreground.take();
1290        let root = self.keyed_child(key, view);
1291        self.foreground = outer;
1292        self.overlays[slot].root = root;
1293    }
