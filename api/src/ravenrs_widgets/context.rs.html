21/// Stable identity of a widget across rebuilds, derived from its position in the
22/// view tree (or from an explicit key, see [`BuildCx::keyed_child`]). Hover,
23/// press and focus are remembered per `WidgetId`.
24#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
25pub struct WidgetId(u64);
26
27impl WidgetId {
28    pub(crate) const ROOT: Self = Self(0x5241_5645_4e52_5321);
29
30    fn child(self, key: u64) -> Self {
31        let mut h = self.0 ^ key.wrapping_mul(0x9e37_79b9_7f4a_7c15);
32        h = (h ^ (h >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
33        h = (h ^ (h >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
34        Self(h ^ (h >> 31))
35    }
36
37    /// The raw identity, for derived ids such as accessibility node ids.
38    pub(crate) fn raw(self) -> u64 {
39        self.0
40    }
41}

148impl Drag {
149    /// Horizontal position inside the frame, from 0.0 (left) to 1.0 (right).
150    pub fn fraction_x(&self) -> f32 {
151        if self.frame.width() <= 0.0 {
152            return 0.0;
153        }
154        ((self.position.x - self.frame.x()) / self.frame.width()).clamp(0.0, 1.0)
155    }

210/// How a node appears to screen readers: its [`Role`], name and state.
211///
212/// Built-in components set this themselves. Custom components call
213/// [`BuildCx::semantics`] on the node they return:

216/// use ravenrs_core::Role;
217/// use ravenrs_widgets::Semantics;
218///
219/// let s = Semantics::new(Role::Switch).name("Wi-Fi").toggled(true);
220/// assert_eq!(s.role(), Role::Switch);
221/// ```
222///
223/// Without semantics, a focusable node is announced as a button named by the
224/// text inside it, text is announced as a label, and other boxes are
225/// transparent: their children belong to the nearest described ancestor.

234    pub(crate) selected: Option<bool>,
235    pub(crate) expanded: Option<bool>,
236    pub(crate) disabled: bool,
237    pub(crate) modal: bool,
238    pub(crate) level: Option<u32>,
239    pub(crate) hidden: bool,
240    pub(crate) active: bool,
241    /// The name is a guess (an icon's name) that a tooltip may replace.
242    pub(crate) fallback_name: bool,
243}

643    /// State owned by this view instance that survives rebuilds, like SwiftUI's
644    /// `@State`. Call it unconditionally and in the same order on every build;
645    /// `init` runs only the first time. The state is dropped when the view stops
646    /// being built.
647    pub fn use_state<T: 'static>(&mut self, init: impl FnOnce() -> T) -> State<T> {
648        let id = self.widget_id();
649        let slot = match self.scopes.last_mut() {
650            Some(scope) => {
651                scope.next_state += 1;
652                scope.next_state
653            }
654            None => 0,
655        };
656        let key = (id, slot);
657        self.local.used.insert(key);
658        if let Some(state) = self.local.slots.get(&key).and_then(|b| b.downcast_ref::<State<T>>()) {
659            return state.clone();
660        }
661        let state = State::new(init());
662        self.local.slots.insert(key, Box::new(state.clone()));
663        state
664    }

683    pub fn child(&mut self, view: &dyn View) -> NodeId {
684        let key = match self.scopes.last_mut() {
685            Some(scope) => {
686                scope.next_child += 1;
687                scope.next_child
688            }
689            None => 0,
690        };
691        self.build_scoped(key, view)
692    }
693
694    /// Builds a child with an explicit key, so it keeps its hover/focus state
695    /// when siblings are inserted or removed (use it for dynamic lists).
696    pub fn keyed_child(&mut self, key: u64, view: &dyn View) -> NodeId {
697        // Keep keyed and positional identities apart.
698        self.build_scoped(key ^ (1 << 63), view)
699    }

701    fn build_scoped(&mut self, key: u64, view: &dyn View) -> NodeId {
702        let id = self.widget_id().child(key);
703        self.scopes.push(Scope { id, next_child: 0, next_state: 0, next_motion: 0 });
704        let node = view.build(self);
705        self.scopes.pop();
706        node
707    }

716    /// Creates a box that lays out `children` and paints `paint` behind them.
717    pub fn container(&mut self, style: LayoutStyle, children: &[NodeId], paint: Paint) -> NodeId {
718        let child_layouts: Vec<LayoutId> =
719            children.iter().filter_map(|c| self.elements.get(c.0 as usize).map(|e| e.layout)).collect();
720        let layout = match self.layout.node(style, &child_layouts) {
721            Ok(id) => id,
722            Err(e) => self.fail(e),
723        };
724        let mut element = Element::new(self.widget_id(), layout);
725        element.children = children.to_vec();
726        element.paint = paint;
727        self.push(element)
728    }

730    /// Creates a text leaf that wraps to the width its parent offers.
731    pub fn text(&mut self, text: &str, attrs: &TextAttrs, color: Color, style: LayoutStyle) -> NodeId {
732        let layout = self.text.layout(text, attrs);
733        let id = match self.layout.measured_leaf(style, TextLeaf { layout, truncate: None }) {
734            Ok(id) => id,
735            Err(e) => self.fail(e),
736        };
737        let mut element = Element::new(self.widget_id(), id);
738        element.text_color = Some(color);
739        element.text = Some(text.into());
740        self.push(element)
741    }

803    /// The first described or focusable node at or under `node`, in tree
804    /// order: the control a label or tooltip around it is about.
805    fn control_in(&self, node: NodeId) -> Option<usize> {
806        let mut stack = vec![node];
807        while let Some(n) = stack.pop() {
808            let e = self.elements.get(n.0 as usize)?;
809            if e.semantics.as_deref().is_some_and(|s| !s.hidden) || e.focus_radius.is_some() {
810                return Some(n.0 as usize);
811            }
812            stack.extend(e.children.iter().rev());
813        }
814        None
815    }

817    /// Names the control at or under `node` with `name` unless it already has
818    /// a name of its own, and sets `description` (form fields, tooltips).
819    /// When it has a name, `name` becomes its description instead.
820    pub(crate) fn label_control(&mut self, node: NodeId, name: &str, description: Option<&str>) {
821        let Some(i) = self.control_in(node) else { return };
822        let e = &mut self.elements[i];
823        let s = e.semantics.get_or_insert_with(|| Box::new(Semantics::new(Role::Button)));
824        if s.name.is_none() || s.fallback_name {
825            s.name = Some(name.to_owned());
826            s.fallback_name = false;
827            if let Some(d) = description {
828                s.description = Some(d.to_owned());
829            }
830        } else if s.name.as_deref() != Some(name) {
831            s.description = Some(description.map_or_else(|| name.to_owned(), |d| format!("{name}. {d}")));
832        }
833    }

909    /// Declares how `node` animates; see [`Motion`]. Call it in the same order on
910    /// every build, or give the motion a [`Motion::key`].
911    pub fn motion(&mut self, node: NodeId, motion: Motion) {
912        let id = self.widget_id();
913        let key = match motion.shared {
914            Some((scope, tag)) => MotionKey::Shared(scope, tag),
915            None => match (motion.key, self.scopes.last_mut()) {
916                (Some(k), _) => MotionKey::Node(id, k | (1 << 63)),
917                (None, Some(scope)) => {
918                    scope.next_motion += 1;
919                    MotionKey::Node(id, scope.next_motion)
920                }
921                (None, None) => MotionKey::Node(id, 0),
922            },
923        };
924        let index = self.motions.len() as u32;
925        if let Some(e) = self.element_mut(node) {
926            e.motion = Some(index);
927            self.motions.push(MotionEntry { node, key, motion });
928        }
929    }

931    /// A value that eases toward `target`, for what [`Motion`] does not cover:
932    /// numbers that feed layout or custom drawing. Call it in the same order on
933    /// every build. It rebuilds the UI on each frame while it moves, so prefer
934    /// [`Motion`] when you can.
935    pub fn animated<T: Animatable + 'static>(&mut self, target: T, transition: Transition) -> T {
936        let cell = self.use_state(|| Rc::new(RefCell::new(Animated::new(target))));
937        let cell = cell.peek().clone();
938        let mut value = cell.borrow_mut();
939        let now = self.motion_clock;
940        let transition =
941            if !self.motion_enabled || self.reduced_motion { Transition::Instant } else { transition };
942        value.set_target(target, transition, now);
943        if value.step(now) {
944            self.value_moving = true;
945            self.wake_at(self.now);
946        }
947        value.value(now)
948    }

1053    /// Builds `view` as a floating layer above the rest of the UI, placed relative
1054    /// to `anchor`. `on_dismiss` is called on a click outside the layer and its
1055    /// anchor, or on Escape.
1056    pub fn overlay(
1057        &mut self,
1058        anchor: NodeId,
1059        placement: Placement,
1060        view: &dyn View,
1061        on_dismiss: Option<Rc<dyn Fn()>>,
1062    ) {
1063        // The slot comes before the content: overlays are placed and painted in
1064        // order, so one opened from inside this one must come after it, or it
1065        // would be placed with no anchor yet and painted underneath.
1066        let slot = self.overlays.len();
1067        let key = 0x6f76_6572_6c61_7900 + slot as u64;
1068        let gap = self.theme.spacing.s2;
1069        self.overlays.push(Overlay { root: anchor, anchor, placement, gap, on_dismiss });
1070        let root = self.keyed_child(key, view);
1071        self.overlays[slot].root = root;
1072    }
