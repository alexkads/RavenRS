1684    /// A rule in words, with its next days: `Every month, 3 times. Next: 05/11/2026, 05/12/2026`.
1685    pub fn recurrence_summary(&self, rule: &RecurrenceRule, start: Date) -> String {
1686        let how = match (rule.frequency, rule.interval.max(1)) {
1687            (Frequency::Daily, 1) => self.recurrence_each[0].clone(),
1688            (Frequency::Weekly, 1) => self.recurrence_each[1].clone(),
1689            (Frequency::Monthly, 1) => self.recurrence_each[2].clone(),
1690            (Frequency::Yearly, 1) => self.recurrence_each[3].clone(),
1691            (Frequency::Daily | Frequency::Weekly | Frequency::Monthly | Frequency::Yearly, n) => {
1692                format!("{} {n} {}", self.recurrence_every, self.recurrence_unit(rule.frequency, n))
1693            }
1694            _ => self.recurrence_frequencies[0].clone(),
1695        };
1696        let until = match rule.end {
1697            RecurrenceEnd::Until(day) => format!(" {} {}", self.recurrence_until, (self.short_date)(day)),
1698            RecurrenceEnd::After(count) => format!(", {count} {}", self.recurrence_times),
1699            _ => String::new(),
1700        };
1701        let next: Vec<String> = rule.days_from(start).take(3).map(self.short_date).collect();
1702        format!("{how}{until}. {}: {}", self.recurrence_next, next.join(", "))
1703    }

1718/// Everything inside uses these words instead of the English defaults: wrap the
1719/// root of an app, or just the part that speaks another language.
1720///
1721/// ```
1722/// use ravenrs_widgets::{DatePicker, Date, Labels, Localized};
1723/// use ravenrs_reactive::State;
1724///
1725/// let day = State::new(Date::new(2026, 11, 15));
1726/// let app = Localized::new(Labels::pt_br(), DatePicker::new(day.binding(), Date::new(2026, 11, 1).unwrap_or(Date::from_days(0))));
1727/// # let _ = app;
1728/// ```
1729#[derive(Debug)]
1730pub struct Localized<V> {
1731    labels: std::rc::Rc<Labels>,
1732    content: V,
1733}
1734
1735impl<V: View> Localized<V> {
1736    /// `content`, using `labels`.
1737    pub fn new(labels: Labels, content: V) -> Self {
1738        Self { labels: std::rc::Rc::new(labels), content }
1739    }
1740}
1741
1742impl<V: View> View for Localized<V> {
1743    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1744        let outer = cx.replace_labels(self.labels.clone());
1745        let node = cx.child(&self.content);
1746        cx.replace_labels(outer);
1747        node
1748    }
