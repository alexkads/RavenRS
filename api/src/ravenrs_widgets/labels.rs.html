213fn pairs<const N: usize>(words: [(&str, &str); N]) -> [(String, String); N] {
214    words.map(|(one, many)| (one.to_owned(), many.to_owned()))
215}

217fn owned<const N: usize>(words: [&str; N]) -> [String; N] {
218    words.map(str::to_owned)

480    /// A month and its year: `November 2026`, `novembro de 2026`.
481    pub fn month_title(&self, date: Date) -> String {
482        let month = &self.months[usize::from(date.month().clamp(1, 12)) - 1];
483        format!("{month}{}{}", self.month_year, date.year())
484    }
485
486    /// A full date, for screen readers: `15 November 2026`, `15 novembro de 2026`.
487    pub fn long_date(&self, date: Date) -> String {
488        format!("{} {}", date.day(), self.month_title(date))
489    }
490
491    /// The word for one period of this frequency; `count` picks singular or plural.
492    pub fn recurrence_unit(&self, frequency: Frequency, count: u32) -> &str {
493        let unit = match frequency {
494            Frequency::Daily => &self.recurrence_units[0],
495            Frequency::Weekly => &self.recurrence_units[1],
496            Frequency::Yearly => &self.recurrence_units[3],
497            _ => &self.recurrence_units[2],
498        };
499        if count > 1 { &unit.1 } else { &unit.0 }
500    }
501
502    /// A rule in words, with its next days: `Every month, 3 times. Next: 05/11/2026, 05/12/2026`.
503    pub fn recurrence_summary(&self, rule: &RecurrenceRule, start: Date) -> String {
504        let how = match (rule.frequency, rule.interval.max(1)) {
505            (Frequency::Daily, 1) => self.recurrence_each[0].clone(),
506            (Frequency::Weekly, 1) => self.recurrence_each[1].clone(),
507            (Frequency::Monthly, 1) => self.recurrence_each[2].clone(),
508            (Frequency::Yearly, 1) => self.recurrence_each[3].clone(),
509            (Frequency::Daily | Frequency::Weekly | Frequency::Monthly | Frequency::Yearly, n) => {
510                format!("{} {n} {}", self.recurrence_every, self.recurrence_unit(rule.frequency, n))
511            }
512            _ => self.recurrence_frequencies[0].clone(),
513        };
514        let until = match rule.end {
515            RecurrenceEnd::Until(day) => format!(" {} {}", self.recurrence_until, (self.short_date)(day)),
516            RecurrenceEnd::After(count) => format!(", {count} {}", self.recurrence_times),
517            _ => String::new(),
518        };
519        let next: Vec<String> = rule.days_from(start).take(3).map(self.short_date).collect();
520        format!("{how}{until}. {}: {}", self.recurrence_next, next.join(", "))
521    }
522
523    /// What a grid's footer calls this summary.
524    pub fn summary(&self, summary: Summary) -> &str {
525        let index = match summary {
526            Summary::Count => 0,
527            Summary::Sum => 1,
528            Summary::Average => 2,
529            Summary::Min => 3,
530            _ => 4,
531        };
532        &self.grid_summaries[index]
533    }

536/// Everything inside uses these words instead of the English defaults: wrap the
537/// root of an app, or just the part that speaks another language.
538///
539/// ```
540/// use ravenrs_widgets::{DatePicker, Date, Labels, Localized};
541/// use ravenrs_reactive::State;
542///
543/// let day = State::new(Date::new(2026, 11, 15));
544/// let app = Localized::new(Labels::pt_br(), DatePicker::new(day.binding(), Date::new(2026, 11, 1).unwrap_or(Date::from_days(0))));
545/// # let _ = app;
546/// ```
547#[derive(Debug)]
548pub struct Localized<V> {
549    labels: std::rc::Rc<Labels>,
550    content: V,
551}
552
553impl<V: View> Localized<V> {
554    /// `content`, using `labels`.
555    pub fn new(labels: Labels, content: V) -> Self {
556        Self { labels: std::rc::Rc::new(labels), content }
557    }
558}
559
560impl<V: View> View for Localized<V> {
561    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
562        let outer = cx.replace_labels(self.labels.clone());
563        let node = cx.child(&self.content);
564        cx.replace_labels(outer);
565        node
566    }
567}
