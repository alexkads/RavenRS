1395    /// A rule in words, with its next days: `Every month, 3 times. Next: 05/11/2026, 05/12/2026`.
1396    pub fn recurrence_summary(&self, rule: &RecurrenceRule, start: Date) -> String {
1397        let how = match (rule.frequency, rule.interval.max(1)) {
1398            (Frequency::Daily, 1) => self.recurrence_each[0].clone(),
1399            (Frequency::Weekly, 1) => self.recurrence_each[1].clone(),
1400            (Frequency::Monthly, 1) => self.recurrence_each[2].clone(),
1401            (Frequency::Yearly, 1) => self.recurrence_each[3].clone(),
1402            (Frequency::Daily | Frequency::Weekly | Frequency::Monthly | Frequency::Yearly, n) => {
1403                format!("{} {n} {}", self.recurrence_every, self.recurrence_unit(rule.frequency, n))
1404            }
1405            _ => self.recurrence_frequencies[0].clone(),
1406        };
1407        let until = match rule.end {
1408            RecurrenceEnd::Until(day) => format!(" {} {}", self.recurrence_until, self.short_date(day)),
1409            RecurrenceEnd::After(count) => format!(", {count} {}", self.recurrence_times),
1410            _ => String::new(),
1411        };
1412        let next: Vec<String> = rule.days_from(start).take(3).map(|day| self.short_date(day)).collect();
1413        format!("{how}{until}. {}: {}", self.recurrence_next, next.join(", "))
1414    }

1437/// let app = Localized::new(Labels::pt_br(), DatePicker::new(day.binding(), Date::new(2026, 11, 1).unwrap_or(Date::from_days(0))));
1438/// # let _ = app;
1439/// ```
1440#[derive(Debug)]
1441pub struct Localized<V> {
1442    labels: std::rc::Rc<Labels>,
1443    content: V,
1444}
1445
1446#[cfg_attr(feature = "live", ravenrs_live::expose)]
1447impl<V: View> Localized<V> {
1448    /// `content`, using `labels`.
1449    pub fn new(labels: Labels, content: V) -> Self {
1450        Self { labels: std::rc::Rc::new(labels), content }
1451    }
1452}
1453
1454impl<V: View> View for Localized<V> {
1455    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1456        let outer = cx.replace_labels(self.labels.clone());
1457        let node = cx.child(&self.content);
1458        cx.replace_labels(outer);
1459        node
1460    }
1461}

1463/// A template with `{0}` and `{1}` filled in with these, each written where it goes.
1464fn two_numbers(template: &str, first: usize, second: usize) -> String {
1465    format_with(template, |index, out| match index {
1466        0 => write!(out, "{first}").is_ok(),
1467        1 => write!(out, "{second}").is_ok(),
1468        _ => false,
1469    })
