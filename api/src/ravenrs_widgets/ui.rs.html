34/// Longest time between frames that animations treat as having really passed.
35const MAX_FRAME_GAP: f64 = 0.25;

37/// How much a longer gap counts for: little enough not to skip the animation,
38/// enough that a machine this slow still gets through it.

41/// Time between paints of animated drawing, in seconds.
42const LIVE_FRAME: f64 = 1.0 / 60.0;

44/// How far past a clip a node still gets painted: what it draws around itself, such as
45/// a shadow, can reach into the clip.
46const PAINT_MARGIN: f32 = 64.0;

48/// Whether anything `frame` draws, with its margin, can land inside `clip`.
49fn touches(frame: Rect, clip: Rect) -> bool {
50    frame.max_x() + PAINT_MARGIN > clip.x()
51        && frame.x() - PAINT_MARGIN < clip.max_x()
52        && frame.max_y() + PAINT_MARGIN > clip.y()
53        && frame.y() - PAINT_MARGIN < clip.max_y()
54}

93#[cfg_attr(feature = "live", ravenrs_live::expose)]
94impl ViewProfile {
95    pub(crate) fn enter(&mut self, name: &'static str, elements: usize, shapes: u64) {
96        let started = (self.clock)();
97        self.open.push(OpenView {
98            name,
99            started,
100            elements,
101            shapes,
102            inner_time: 0.0,
103            inner_elements: 0,
104            inner_shapes: 0,
105        });
106    }
107
108    pub(crate) fn leave(&mut self, elements: usize, shapes: u64) {
109        let Some(view) = self.open.pop() else { return };
110        let total = ((self.clock)() - view.started).max(0.0);
111        let made = elements.saturating_sub(view.elements);
112        let shaped = shapes.saturating_sub(view.shapes);
113        if let Some(outer) = self.open.last_mut() {
114            outer.inner_time += total;
115            outer.inner_elements += made;
116            outer.inner_shapes += shaped;
117        }
118        let cost = self
119            .costs
120            .entry(view.name)
121            .or_insert_with(|| ViewCost { name: view.name, ..ViewCost::default() });
122        cost.count += 1;
123        cost.time += std::time::Duration::from_secs_f64((total - view.inner_time).max(0.0));
124        cost.elements += made.saturating_sub(view.inner_elements) as u32;
125        cost.shapes += shaped.saturating_sub(view.inner_shapes) as u32;
126    }

347    /// What was done since the last call: a platform asks once per frame.
348    pub fn take_frame_work(&mut self) -> FrameWork {
349        let shapes = self.text.stats().shapes;
350        let (phases, shapes_before) = self.frame_start;
351        let now = self.phases;
352        let mut work = std::mem::take(&mut self.frame);
353        work.phases = PhaseTimes {
354            build: now.build - phases.build,
355            layout: now.layout - phases.layout,
356            place: now.place - phases.place,
357            paint: now.paint - phases.paint,
358        };
359        work.shapes = (shapes - shapes_before) as u32;
360        work.elements = self.elements.len() as u32;
361        self.frame_start = (now, shapes);
362        work
363    }

475    /// Hands dropped files to the place under `position`. When no place is there, and the
476    /// screen has only one, it goes there: not every system says where a drop landed.
477    /// Whether anything took them.
478    pub fn drop_files(&mut self, position: Option<Point>, files: Vec<crate::PickedFile>) -> bool {
479        self.files_over.set_if_changed(false);
480        let mut node = position.and_then(|p| self.hit_test(p));
481        while let Some(n) = node {
482            if let Some(handler) = self.elements[n.0 as usize].on_file_drop.clone() {
483                handler(files);
484                return true;
485            }
486            node = self.parents.get(n.0 as usize).copied().flatten();
487        }
488        let mut places = self.elements.iter().filter_map(|e| e.on_file_drop.clone());
489        match (places.next(), places.next()) {
490            (Some(only), None) => {
491                only(files);
492                true
493            }
494            _ => false,
495        }
496    }

576    /// Sets the clock (seconds since the app started). Platforms call it before
577    /// handling events and before each frame. Animations start only once a
578    /// platform drives this clock.
579    ///
580    /// Animations follow their own clock, which skips stalls: after a long pause
581    /// (a hidden window, a slow frame) they continue about where they were
582    /// instead of jumping to the end.
583    pub fn set_time(&mut self, seconds: f64) {
584        let elapsed = seconds - self.time;
585        self.motion.clock += if (0.0..=MAX_FRAME_GAP).contains(&elapsed) { elapsed } else { STALL_STEP };
586        self.motion.enabled = true;
587        self.time = seconds;
588    }

621    pub fn update(&mut self) -> &Scene {
622        let anchor = self.links.take_anchor();
623        if self.needs_rebuild() {
624            self.rebuild();
625            // The tree changed under a stationary pointer: refresh hover once.
626            if self.refresh_hover() {
627                self.rebuild();
628            }
629            // Those views were built for their old size. Their new build can move
630            // others, but a second round of that waits for the next frame.
631            if std::mem::take(&mut self.viewport_moved) {
632                self.rebuild();
633                self.viewport_moved = false;
634            }
635        } else if self.glide.is_some() && anchor.is_none() {
636            self.advance_glide();
637        } else if self.motion.active || self.live_is_due() {
638            // Only animations are due: the tree and its layout are still valid.
639            let _span = tracing::trace_span!("ravenrs::motion_frame").entered();
640            self.paint();
641        }
642        if let Some(name) = anchor {
643            self.start_glide(&name);
644        }
645        &self.scene
646    }

648    /// Starts scrolling the container around the anchor `name` so the anchor lands at
649    /// its offset below the top. Under reduced motion it jumps.
650    fn start_glide(&mut self, name: &str) {
651        let Some(i) = self.elements.iter().position(|e| e.anchor.as_ref().is_some_and(|(n, _)| &**n == name))
652        else {
653            tracing::debug!(name, "no anchor of that name");
654            return;
655        };
656        let offset = self.elements[i].anchor.as_ref().map_or(0.0, |a| a.1);
657        let mut node = self.parents.get(i).copied().flatten();
658        while let Some(n) = node {
659            let k = n.0 as usize;
660            if self.elements[k].scroll {
661                let current = self.clamped_offset(k);
662                let to = (current.y + self.frames[i].y() - self.frames[k].y() - offset)
663                    .clamp(0.0, self.max_offset(k).y);
664                let widget = self.elements[k].widget;
665                if self.reduced_motion() {
666                    self.scroll.offsets.insert(widget, Point::new(current.x, to));
667                    self.place_and_paint();
668                } else {
669                    self.glide = Some((widget, current.y, to, self.time));
670                    self.next_live = Some(self.time);
671                }
672                return;
673            }
674            node = self.parents.get(k).copied().flatten();
675        }
676    }

678    /// One frame of a glide: eases in and out over about half a second.
679    fn advance_glide(&mut self) {
680        let Some((widget, from, to, start)) = self.glide else { return };
681        const GLIDE: f64 = 0.55;
682        let t = ((self.time - start) / GLIDE).clamp(0.0, 1.0) as f32;
683        let eased = ravenrs_animation::Easing::EaseInOut.sample(t);
684        let x = self.scroll.offsets.get(&widget).map_or(0.0, |o| o.x);
685        self.scroll.offsets.insert(widget, Point::new(x, from + (to - from) * eased));
686        self.scrolled_at.insert(widget, self.time);
687        if t >= 1.0 {
688            self.glide = None;
689        } else {
690            self.next_live = Some(self.time + LIVE_FRAME);
691        }
692        self.place_and_paint();
693        self.refresh_hover();
694    }

696    /// Runs the `on_appear` handlers of elements that show for the first time.
697    fn tell_appeared(&mut self) {
698        let mut due = Vec::new();
699        for (i, e) in self.elements.iter().enumerate() {
700            let Some(handler) = &e.on_appear else { continue };
701            if self.appeared.contains(&e.widget) {
702                continue;
703            }
704            let (frame, view) = (self.frames[i], self.views[i]);
705            if frame.max_y() > view.y()
706                && frame.y() < view.max_y()
707                && frame.max_x() > view.x()
708                && frame.x() < view.max_x()
709            {
710                due.push((e.widget, handler.clone()));
711            }
712        }
713        for (widget, handler) in due {
714            self.appeared.insert(widget);
715            handler();
716        }
717    }

735    /// Feeds one input event; returns true when a redraw is needed.
736    pub fn handle_event(&mut self, event: &Event) -> bool {
737        match event {
738            Event::PointerMove { position } => {
739                self.pointer = Some(*position);
740                self.tell_activity(crate::context::UiActivity::PointerMoved(*position));
741                if let Some(widget) = self.dragging {
742                    self.drag(widget, DragPhase::Move, *position);
743                }
744                self.refresh_hover();
745                if let Some(node) = self.hit_test(*position) {
746                    let i = node.0 as usize;
747                    if let Some(handler) = self.elements[i].on_pointer_move.clone() {
748                        handler(*position, self.frames[i]);
749                    }
750                }
751            }
752            Event::PointerDown { position, button: PointerButton::Primary } => {
753                self.pointer = Some(*position);
754                self.tell_activity(crate::context::UiActivity::Clicked(*position));
755                self.reveal_focus = false;
756                self.dismiss_overlays_outside(*position);
757                self.refresh_hover();
758                let target = self.hit_test(*position);
759                let element = target.map(|n| &self.elements[n.0 as usize]);
760                let widget = element.map(|e| e.widget);
761                let draggable = element.is_some_and(|e| e.on_drag.is_some());
762                if self.interaction.pressed != widget {
763                    self.interaction.pressed = widget;
764                    self.ask_build(BuildCause::Press);
765                }
766                // Like the DOM, a click focuses the nearest focusable element at or above the target.
767                let mut node = target;
768                let mut focused = None;
769                while let Some(n) = node {
770                    let e = &self.elements[n.0 as usize];
771                    if e.focus_radius.is_some() {
772                        focused = Some(e.widget);
773                        break;
774                    }
775                    node = self.parents.get(n.0 as usize).copied().flatten();
776                }
777                if self.interaction.focused != focused || self.interaction.focus_visible {
778                    self.interaction.focused = focused;
779                    self.interaction.focus_visible = false;
780                    self.ask_build(BuildCause::Focus);
781                }
782                if let (true, Some(w)) = (draggable, widget) {
783                    self.dragging = Some(w);
784                    self.drag(w, DragPhase::Start, *position);
785                }
786            }
787            Event::PointerUp { position, button: PointerButton::Primary } => {
788                if let Some(w) = self.dragging.take() {
789                    self.drag(w, DragPhase::End, *position);
790                }
791                let pressed = self.interaction.pressed.take();
792                if pressed.is_some() {
793                    self.ask_build(BuildCause::Press);
794                }
795                if let Some(node) = self.hit_test(*position) {
796                    let element = &self.elements[node.0 as usize];
797                    if pressed == Some(element.widget) {
798                        let handler = element.on_click.clone();
799                        if let Some(handler) = handler {
800                            handler();
801                        }
802                    }
803                }
804            }
805            Event::PointerLeave => {
806                self.pointer = None;
807                self.refresh_hover();
808                if self.dragging.is_none() && self.interaction.pressed.take().is_some() {
809                    self.ask_build(BuildCause::Press);
810                }
811            }
812            Event::Scroll { position, delta } => {
813                if self.scroll(*position, *delta) {
814                    if !self.needs_build {
815                        self.place_and_paint();
816                        self.refresh_hover();
817                    }
818                    return true;
819                }
820            }
821            Event::Pinch { position, delta } => {
822                self.pinch(*position, *delta);
823            }
824            Event::PointerDown { position, button: PointerButton::Secondary } => {
825                self.pointer = Some(*position);
826                self.dismiss_overlays_outside(*position);
827                let mut node = self.hit_test(*position);
828                while let Some(n) = node {
829                    if let Some(handler) = self.elements[n.0 as usize].on_context_menu.clone() {
830                        handler(*position);
831                        break;
832                    }
833                    if let Some(handler) = self.elements[n.0 as usize].on_context_menu_in.clone() {
834                        handler(*position, self.frames[n.0 as usize]);
835                        break;
836                    }
837                    node = self.parents.get(n.0 as usize).copied().flatten();
838                }
839            }
840            Event::DoubleClick { position } => {
841                if let Some(node) = self.hit_test(*position) {
842                    let handler = self.elements[node.0 as usize].on_double_click.clone();
843                    if let Some(handler) = handler {
844                        handler();
845                    }
846                }
847            }
848            Event::KeyUp { modifiers, .. } => self.modifiers.set(*modifiers),
849            Event::KeyDown { key, modifiers } => {
850                self.modifiers.set(*modifiers);
851                let handler = self.focused_element().and_then(|e| e.on_key.clone());
852                if let Some(handler) = handler
853                    && handler(key, *modifiers)
854                {
855                    return self.needs_redraw();
856                }
857                let command = modifiers.ctrl || modifiers.meta;
858                // Escape closes what opened last: a menu before anything that listens for it.
859                if *key == Key::Escape
860                    && let Some(dismiss) = self.overlays.iter().rev().find_map(|o| o.on_dismiss.clone())
861                {
862                    dismiss();
863                    return self.needs_redraw();
864                }
865                let shortcut = self.shortcuts.iter().find(|s| {
866                    s.command == command
867                        && (!s.shift || modifiers.shift)
868                        && match (&s.key, key) {
869                            (Key::Character(a), Key::Character(b)) => a.eq_ignore_ascii_case(b),
870                            (a, b) => a == b,
871                        }
872                });
873                if let Some(s) = shortcut.map(|s| s.handler.clone()) {
874                    s();
875                    return self.needs_redraw();
876                }
877                match key {
878                    Key::Tab => self.move_focus(!modifiers.shift),
879                    Key::Enter | Key::Space => {
880                        let handler = self.focused_element().and_then(|e| e.on_click.clone());
881                        if let Some(handler) = handler {
882                            handler();
883                        }
884                    }
885                    Key::Escape => {
886                        let dismiss = self.overlays.iter().rev().find_map(|o| o.on_dismiss.clone());
887                        if let Some(dismiss) = dismiss {
888                            dismiss();
889                        } else if self.interaction.focused.take().is_some() {
890                            self.ask_build(BuildCause::Focus);
891                        }
892                    }
893                    _ => {}
894                }
895            }
896            Event::FocusLost => {
897                self.dragging = None;
898                if self.interaction.pressed.take().is_some() {
899                    self.ask_build(BuildCause::Press);
900                }
901            }
902            _ => {}
903        }
904        self.needs_redraw()
905    }

949    /// Keeps focus inside an open modal layer, moving it in when the layer
950    /// opens and back to where it was when the layer closes.
951    fn trap_focus(&mut self) {
952        match self.modal_root() {
953            Some(root) => {
954                let focused = self.interaction.focused;
955                let layers = self.modal_layers(root);
956                let inside = (0..self.elements.len()).any(|i| {
957                    let e = &self.elements[i];
958                    Some(e.widget) == focused
959                        && e.focus_radius.is_some()
960                        && layers.iter().any(|layer| self.is_inside(NodeId(i as u32), *layer))
961                });
962                if !inside {
963                    self.modal_return.get_or_insert(focused);
964                    // With nothing to focus in the layer, focus still leaves the page behind it.
965                    let first = self.first_focusable(root);
966                    if self.interaction.focused != first {
967                        self.interaction.focused = first;
968                        self.ask_build(BuildCause::Focus);
969                    }
970                }
971            }
972            None => {
973                if let Some(back) = self.modal_return.take() {
974                    self.interaction.focused = back;
975                    self.ask_build(BuildCause::Focus);
976                }
977            }
978        }
979    }

1032    /// Scrolls the innermost scroll container under `p` that can move in the
1033    /// direction of `delta`. Returns true if something scrolled. An open modal
1034    /// layer keeps it from what is behind, even when its own content is at the end.
1035    fn scroll(&mut self, p: Point, delta: Point) -> bool {
1036        let layers = self.modal_root().map(|root| self.modal_layers(root));
1037        for &node in self.paint_order.iter().rev() {
1038            let i = node.0 as usize;
1039            let e = &self.elements[i];
1040            if !(e.scroll || e.scrolls_with_widget) || !self.frames[i].intersect(&self.clips[i]).contains(p) {
1041                continue;
1042            }
1043            if !layers.as_ref().is_none_or(|l| l.iter().any(|root| self.is_inside(node, *root))) {
1044                continue;
1045            }
1046            // Part of a view that scrolls moves what scrolls, such as frozen row names.
1047            let i = if e.scroll {
1048                i
1049            } else {
1050                let widget = e.widget;
1051                match self.elements.iter().position(|other| other.scroll && other.widget == widget) {
1052                    Some(found) => found,
1053                    None => continue,
1054                }
1055            };
1056            let e = &self.elements[i];
1057            let max = self.max_offset(i);
1058            let current = self.scroll.offsets.get(&e.widget).copied().unwrap_or(Point::ZERO);
1059            let delta = if e.wheel_sideways && delta.x == 0.0 { Point::new(delta.y, 0.0) } else { delta };
1060            let next = Point::new(
1061                if e.scroll_x { (current.x - delta.x).clamp(0.0, max.x) } else { 0.0 },
1062                (current.y - delta.y).clamp(0.0, max.y),
1063            );
1064            if (next.x - current.x).abs() > f32::EPSILON || (next.y - current.y).abs() > f32::EPSILON {
1065                self.scroll.offsets.insert(e.widget, next);
1066                self.scrolled_at.insert(e.widget, self.time);
1067                self.scrollbar_showing = true;
1068                self.next_live = Some(self.time);
1069                let e = &self.elements[i];
1070                if e.rebuild_on_scroll {
1071                    self.ask_build(BuildCause::Scroll);
1072                }
1073                return true;
1074            }
1075        }
1076        false
1077    }

1090    fn move_focus(&mut self, forward: bool) {
1091        let layers = self.modal_root().map(|root| self.modal_layers(root));
1092        let focusable: Vec<WidgetId> = self
1093            .paint_order
1094            .iter()
1095            .filter(|n| layers.as_ref().is_none_or(|l| l.iter().any(|root| self.is_inside(**n, *root))))
1096            .map(|n| &self.elements[n.0 as usize])
1097            .filter(|e| e.focus_radius.is_some())
1098            .map(|e| e.widget)
1099            .collect();
1100        if focusable.is_empty() {
1101            return;
1102        }
1103        let current = self.interaction.focused.and_then(|w| focusable.iter().position(|f| *f == w));
1104        let next = match (current, forward) {
1105            (None, true) => 0,
1106            (None, false) => focusable.len() - 1,
1107            (Some(i), true) => (i + 1) % focusable.len(),
1108            (Some(i), false) => (i + focusable.len() - 1) % focusable.len(),
1109        };
1110        self.interaction.focused = Some(focusable[next]);
1111        self.interaction.focus_visible = true;
1112        self.reveal_focus = true;
1113        self.ask_build(BuildCause::Focus);
1114    }

1116    /// Recomputes the hovered widget and cursor; returns true if hover changed.
1117    fn refresh_hover(&mut self) -> bool {
1118        let target = self.pointer.and_then(|p| self.hit_test(p));
1119        let element = target.map(|n| &self.elements[n.0 as usize]);
1120        let hovered = element.map(|e| e.widget);
1121        self.cursor = element.and_then(|e| e.cursor).unwrap_or_default();
1122        let mut chain = Vec::new();
1123        let mut node = target;
1124        while let Some(n) = node {
1125            chain.push(self.elements[n.0 as usize].widget);
1126            node = self.parents.get(n.0 as usize).copied().flatten();
1127        }
1128        if hovered != self.interaction.hovered || chain != self.hover_chain {
1129            // Only a view that asked about the pointer can look different now. Those that stay
1130            // in the chain keep their answer, unless the pointer moved on to or off them.
1131            let readers = self.hover_readers.borrow();
1132            let matters = |widget: &WidgetId| readers.contains(widget);
1133            let affected = chain.iter().filter(|w| !self.hover_chain.contains(w)).any(matters)
1134                || self.hover_chain.iter().filter(|w| !chain.contains(w)).any(matters)
1135                || hovered.iter().chain(&self.interaction.hovered).any(matters);
1136            drop(readers);
1137            self.interaction.hovered = hovered;
1138            self.hover_chain = chain;
1139            if affected {
1140                self.ask_build(BuildCause::Hover);
1141            }
1142            return affected;
1143        }
1144        false
1145    }

1155    fn rebuild(&mut self) {
1156        let _span = tracing::trace_span!("ravenrs::frame").entered();
1157        self.needs_build = false;
1158        let cause = self.take_build_cause();
1159        self.build_cause = Some(cause);
1160        self.state_dirty.set(false);
1161        // Before the build: a wake during it must cause another one.
1162        self.ui_waker.clear();
1163        if let Err(e) = self.build() {
1164            tracing::error!(error = %e, "build failed; keeping the previous frame");
1165            return;
1166        }
1167        if let Err(e) = self.compute_layout() {
1168            tracing::error!(error = %e, "layout failed; keeping the previous frame");
1169            return;
1170        }
1171        self.place_and_paint();
1172        self.forget_stale_frames();
1173    }

1175    /// Drops the frames of elements the latest build did not ask about. Measured rows come
1176    /// and go as a list scrolls, and each one would otherwise stay in this map for good;
1177    /// what the list must remember of a row it keeps in its own track.
1178    fn forget_stale_frames(&mut self) {
1179        let asked = |e: &&Element| e.scroll || e.report_frame;
1180        // Counted first so that an ordinary frame allocates nothing here.

1347    fn solve_layout(&mut self) -> Result<(), ravenrs_layout::LayoutError> {
1348        let Some(root) = self.root_node else { return Ok(()) };
1349        let roots = std::iter::once(root).chain(self.overlays.iter().map(|o| o.root)).collect::<Vec<_>>();
1350        for node in roots {
1351            let layout_id = self.elements[node.0 as usize].layout;
1352            self.layout.compute(layout_id, self.viewport, |leaf, input| {
1353                let limit = input.known_width.or(input.available_width.limit());
1354                // Truncated text never wraps: it takes what it needs, up to the limit.
1355                let one_line = leaf.truncate.is_some();
1356                let size = leaf.layout.measure(if one_line { None } else { limit });
1357                let width = match limit {
1358                    Some(limit) if one_line => size.width.ceil().min(limit),
1359                    _ => size.width.ceil(),
1360                };
1361                Size::new(input.known_width.unwrap_or(width), input.known_height.unwrap_or(size.height))
1362            })?;
1363        }
1364        self.content.clear();
1365        self.content.resize(self.elements.len(), Size::ZERO);
1366        for (i, e) in self.elements.iter().enumerate() {
1367            if e.scroll {
1368                self.content[i] = self.layout.content_size(e.layout)?;
1369            }
1370            if e.measure {
1371                let content = self.layout.content_size(e.layout)?;
1372                let before = self.scroll.contents.insert(e.widget, content);
1373                // What is built from the room inside a scroll has to see that room change.
1374                self.viewport_moved |= e.rebuild_on_scroll && before != Some(content);
1375            }
1376        }
1377        Ok(())
1378    }

1425    /// If focus just moved, scrolls each container around the focused element,
1426    /// innermost first, just far enough to show it with its focus ring. Returns
1427    /// true if something scrolled, which makes the frames stale.
1428    fn scroll_focus_into_view(&mut self) -> bool {
1429        if !std::mem::take(&mut self.reveal_focus) {
1430            return false;
1431        }
1432        let focused = self.interaction.focused;
1433        let is_focused = |e: &Element| Some(e.widget) == focused && e.focus_radius.is_some();
1434        let Some(start) = self.elements.iter().position(is_focused) else { return false };
1435        let ix = self.theme().interaction;
1436        let mut target = self.frames[start].inflate(ix.focus_ring_offset + ix.focus_ring_width);
1437        let mut scrolled = false;
1438        let mut node = NodeId(start as u32);
1439        // A floating layer's parent is its anchor, but moving the page behind it shows nothing.
1440        while !self.overlays.iter().any(|o| o.root == node) {
1441            let Some(parent) = self.parents.get(node.0 as usize).copied().flatten() else { break };
1442            node = parent;
1443            let i = node.0 as usize;
1444            if !self.elements[i].scroll {
1445                continue;
1446            }
1447            let (frame, max, offset) = (self.frames[i], self.max_offset(i), self.clamped_offset(i));
1448            let next = Point::new(
1449                if self.elements[i].scroll_x {
1450                    reveal_offset(offset.x, (target.x(), target.max_x()), (frame.x(), frame.max_x()), max.x)
1451                } else {
1452                    offset.x
1453                },
1454                reveal_offset(offset.y, (target.y(), target.max_y()), (frame.y(), frame.max_y()), max.y),
1455            );
1456            if next != offset {
1457                self.scroll.offsets.insert(self.elements[i].widget, next);
1458                // Virtualized content depends on the offset: build again at the new position.
1459                if self.elements[i].rebuild_on_scroll {
1460                    self.ask_build(BuildCause::Scroll);
1461                }
1462                target = target.translate(offset - next);
1463                scrolled = true;
1464            }
1465        }
1466        scrolled
1467    }

1469    /// Sets absolute frames for `node` and its subtree; `origin` is where the
1470    /// node's own frame starts (its layout location is ignored for roots).
1471    fn place(&mut self, node: NodeId, origin: Point) {
1472        let window = Rect::from_origin_size(Point::ZERO, self.viewport);
1473        let mut stack = vec![(node, origin, true, Point::ZERO, window, window)];
1474        while let Some((node, parent_origin, is_root, mut shift, parent_frame, view)) = stack.pop() {
1475            let i = node.0 as usize;
1476            let Ok(local) = self.layout.frame(self.elements[i].layout) else { continue };
1477            let mut frame = if is_root {
1478                Rect::from_origin_size(parent_origin, local.size)
1479            } else {
1480                local.translate(parent_origin)
1481            };
1482            // Sticky: held below the top of the view, and taken away by the parent's bottom.
1483            if let Some(top) = self.elements[i].pin.filter(|_| !is_root) {
1484                let lowest = (parent_frame.max_y() - frame.height()).max(frame.y());
1485                let y = frame.y().max(view.y() + top).min(lowest);
1486                let moved = Point::new(0.0, y - frame.y());
1487                frame = frame.translate(moved);
1488                shift = shift + moved;
1489            }
1490            self.frames[i] = frame;
1491            self.shifts[i] = shift;
1492            self.views[i] = view;
1493            let mut child_origin = frame.origin + self.elements[i].translate;
1494            let mut child_shift = shift;
1495            let mut child_view = view;
1496            if self.elements[i].scroll {
1497                child_view = frame;
1498                let offset = self.clamped_offset(i);
1499                child_origin = child_origin - offset;
1500                child_shift = child_shift - offset;
1501                let widget = self.elements[i].widget;
1502                let before = self.scroll.viewports.insert(widget, frame.size);
1503                self.viewport_moved |= self.elements[i].rebuild_on_scroll && before != Some(frame.size);
1504                // Store the clamped value so builds see where the content really is.
1505                self.scroll.offsets.insert(widget, offset);
1506            } else if self.elements[i].report_frame {
1507                let before = self.scroll.viewports.insert(self.elements[i].widget, frame.size);
1508                self.viewport_moved |= before != Some(frame.size);
1509            }
1510            let element = &self.elements[i];
1511            for child in &element.children {
1512                stack.push((*child, child_origin, false, child_shift, frame, child_view));
1513            }
1514        }
1515    }

1566    /// How far the scroll is along `range` for element `i`, from 0 to 1.
1567    fn progress_of(&self, i: usize, range: crate::ScrollRange) -> f32 {
1568        let mut scene = None;
1569        if range == crate::ScrollRange::Scene {
1570            let mut node = Some(NodeId(i as u32));
1571            while let Some(n) = node {
1572                let k = n.0 as usize;
1573                if let Some(track) = self.elements[k].scroll_track {
1574                    scene = Some((k, track));
1575                    break;
1576                }
1577                node = self.parents.get(k).copied().flatten();
1578            }
1579        }
1580        match scene {
1581            Some((k, track)) => range.progress(self.frames[i], self.views[k], Some((self.frames[k], track))),
1582            None => range.progress(self.frames[i], self.views[i], None),
1583        }
1584    }

1642    fn paint_node(&mut self, node: NodeId, clip: Rect, theme: &Theme) {
1643        let i = node.0 as usize;
1644        self.paint_order.push(node);
1645        self.clips[i] = clip;
1646        let frame = self.frames[i];
1647        let start = self.scene.len();
1648        let sampled = self.sample_motion(i);
1649        let motion = sampled.as_ref().map(|(entry, _)| *entry);
1650        let (quad, text_color, visual, mut effect) = match sampled {
1651            Some((_, s)) => (s.quad, s.text, s.frame.unwrap_or(frame), s.effect),
1652            None => (None, None, frame, Effect::IDENTITY),
1653        };
1654        if let Some((range, cell)) = self.elements[i].progress_out.clone() {
1655            cell.set(self.progress_of(i, range));
1656        }
1657        let mut linked = self.scroll_effect(i);
1658        if let Some(link) = self.elements[i].time_link.clone() {
1659            let time = if self.reduced_motion() { 0.0 } else { self.time };
1660            self.has_live = true;
1661            linked = linked.then(link.effect_at(time, frame));
1662        }
1663        effect = effect.then(linked);
1664        // Off screen, its pixels would be thrown away: what it draws goes nowhere, though its
1665        // children still get their turn, since they can stand outside it. A node under an
1666        // effect may land elsewhere, one leaving is captured for its way out, and one
1667        // without a size may draw around itself.
1668        let shown = effect != Effect::IDENTITY
1669            || motion.is_some_and(|m| m.motion.exit.is_some())
1670            || frame.is_empty()
1671            || touches(visual, clip);
1672        let underlaid = motion.is_some_and(|m| m.motion.underlay);
1673        if shown && !underlaid {
1674            self.paint_shape(i, visual, quad);
1675        }
1676        let exit = motion.filter(|m| m.motion.exit.is_some());
1677        if exit.is_some() {
1678            self.motion.begin_capture();
1679        }
1680        // Everything from here on travels with the node.
1681        let content_start = self.scene.len();
1682        if let Some(color) = self.elements[i].text_color.filter(|_| shown) {
1683            let color = text_color.unwrap_or(color);
1684            let layout_id = self.elements[i].layout;
1685            if let Some(leaf) = self.layout.context_mut(layout_id) {
1686                if let Some((text, attrs)) = &leaf.truncate {
1687                    let fitted = self.text.layout_ellipsized(text, attrs, frame.width() + 0.5);
1688                    fitted.paint(frame.origin, color, &mut self.scene);
1689                } else {
1690                    // Re-break at the final width; the half pixel absorbs rounding.
1691                    leaf.layout.measure(Some(frame.width() + 0.5));
1692                    leaf.layout.paint(frame.origin, color, &mut self.scene);
1693                }
1694            }
1695        }
1696        let clips_children = self.elements[i].clip;
1697        let child_clip = if clips_children { clip.intersect(&frame) } else { clip };
1698        let pushes_clip = clips_children && shown;
1699        if pushes_clip {
1700            self.scene.push_clip(frame);
1701        }
1702        let children = self.elements[i].children.clone();
1703        for &child in &children {
1704            if let Some(m) = self.elements[child.0 as usize].motion
1705                && self.motions.get(m as usize).is_some_and(|e| e.motion.underlay)
1706            {
1707                let c = child.0 as usize;
1708                if let Some((_, s)) = self.sample_motion(c) {
1709                    self.paint_shape(c, s.frame.unwrap_or(self.frames[c]), s.quad);
1710                }
1711            }
1712        }
1713        for child in children {
1714            self.paint_node(child, child_clip, theme);
1715        }
1716        let outside = self.elements[i].scrollbar_outside;
1717        if shown && self.elements[i].scroll && outside == 0.0 {
1718            self.paint_scrollbar(i, theme, 0.0);
1719        }
1720        if pushes_clip {
1721            self.scene.pop_clip();
1722        }
1723        if shown && self.elements[i].scroll && outside != 0.0 {
1724            self.paint_scrollbar(i, theme, outside);
1725        }
1726        let element = &self.elements[i];
1727        if let Some(radius) = element.focus_radius
1728            && shown
1729            && self.interaction.focus_visible
1730            && self.interaction.focused == Some(element.widget)
1731        {
1732            let ix = theme.interaction;
1733            let grow = ix.focus_ring_offset + ix.focus_ring_width;
1734            let ring = Quad::new(frame.inflate(grow), Color::TRANSPARENT)
1735                .radius(radius + grow)
1736                .border(ix.focus_ring_width, theme.color(Semantic::FocusRing));
1737            self.scene.push_quad(ring);
1738        }
1739        let Some(entry) = motion else {
1740            self.scene.apply_effect(start..self.scene.len(), linked, frame.center());
1741            return;
1742        };
1743        let end = self.scene.len();
1744        if !entry.motion.shape_only && visual.origin != frame.origin {
1745            self.scene.apply_effect(
1746                content_start..end,
1747                {
1748                    let d = visual.origin - frame.origin;
1749                    Effect::moved(d.x, d.y)
1750                },
1751                Point::ZERO,
1752            );
1753        }
1754        let mut pivot = visual.center();
1755        let mut flip = false;
1756        if entry.motion.anchored
1757            && let Some(anchor) = self.anchor_of(node)
1758        {
1759            let target = anchor.center();
1760            pivot = Point::new(
1761                target.x.clamp(visual.x(), visual.max_x()),
1762                target.y.clamp(visual.y(), visual.max_y()),
1763            );
1764            flip = visual.center().y < target.y;
1765            effect = flipped(effect, flip);
1766        }
1767        if exit.is_some() {
1768            let layer = if self.painting_overlays { GhostLayer::Overlay } else { GhostLayer::Main };
1769            let primitives = &self.scene.primitives()[start..end];
1770            self.motion.end_capture(entry.key, primitives, start..end, clip, pivot, flip, layer);
1771        }
1772        self.scene.apply_effect(start..end, effect, pivot);
1773    }

1775    /// Draws the scrollbar around a scroll, and lets it fade. `outside` puts it that far
1776    /// past the container's edge, in the room around what it scrolls.
1777    fn paint_scrollbar(&mut self, i: usize, theme: &Theme, outside: f32) {
1778        let frame = self.frames[i];
1779        let (content, max) = (self.content[i], self.max_offset(i));
1780        let offset = self.clamped_offset(i);
1781        let Some(&moved) = self.scrolled_at.get(&self.elements[i].widget) else { return };
1782        let age = self.time - moved;
1783        if age >= SCROLLBAR_HOLD + SCROLLBAR_FADE {
1784            return;
1785        }
1786        self.scrollbar_showing = true;
1787        let fading = ((age - SCROLLBAR_HOLD) / SCROLLBAR_FADE).clamp(0.0, 1.0) as f32;
1788        let alpha = if self.reduced_motion() { SCROLLBAR_ALPHA } else { SCROLLBAR_ALPHA * (1.0 - fading) };
1789        let color = theme.color(Semantic::InkMuted).with_alpha(alpha);
1790        if max.y > 0.5 && frame.height() > 0.0 {
1791            let track = frame.height() - 8.0;
1792            let thumb = (track * frame.height() / content.height).max(24.0).min(track);
1793            let y = frame.y() + 4.0 + (track - thumb) * (offset.y / max.y);
1794            let rect = Rect::new(frame.max_x() - SCROLLBAR_WIDTH - 3.0 + outside, y, SCROLLBAR_WIDTH, thumb);
1795            self.scene.push_quad(Quad::new(rect, color).radius(SCROLLBAR_WIDTH / 2.0));
1796        }
1797        if self.elements[i].scroll_x && max.x > 0.5 && frame.width() > 0.0 {
1798            let track = frame.width() - 8.0;
1799            let thumb = (track * frame.width() / content.width).max(24.0).min(track);
1800            let x = frame.x() + 4.0 + (track - thumb) * (offset.x / max.x);
1801            let rect = Rect::new(x, frame.max_y() - SCROLLBAR_WIDTH - 3.0, thumb, SCROLLBAR_WIDTH);
1802            self.scene.push_quad(Quad::new(rect, color).radius(SCROLLBAR_WIDTH / 2.0));
1803        }
1804    }
1805}

1807/// Top-left corner of an overlay of `size`, kept inside the viewport.
1808/// The scroll offset along one axis that shows `target` inside `view`, moving
1809/// as little as possible. A target larger than the view shows its start.
1810fn reveal_offset(offset: f32, target: (f32, f32), view: (f32, f32), max: f32) -> f32 {
1811    let next = if target.0 < view.0 || target.1 - target.0 > view.1 - view.0 {
1812        offset + target.0 - view.0
1813    } else if target.1 > view.1 {
1814        offset + target.1 - view.1
1815    } else {
1816        offset
1817    };
1818    next.clamp(0.0, max)
1819}

1821/// How far a layer in a corner of the window keeps from its edges.
1822const VIEWPORT_MARGIN: f32 = 16.0;
1823
1824fn overlay_origin(placement: Placement, anchor: Rect, size: Size, gap: f32, viewport: Size) -> Point {
1825    let mut p = match placement {
1826        Placement::Below => Point::new(anchor.x(), anchor.max_y() + gap),
1827        Placement::Above => Point::new(anchor.x(), anchor.y() - gap - size.height),
1828        Placement::Right => Point::new(anchor.max_x() + gap, anchor.y()),
1829        Placement::AboveCenter => {
1830            Point::new(anchor.center().x - size.width / 2.0, anchor.y() - gap - size.height)
1831        }
1832        Placement::RightCenter => Point::new(anchor.max_x() + gap, anchor.center().y - size.height / 2.0),
1833        Placement::LeftCenter => {
1834            Point::new(anchor.x() - gap - size.width, anchor.center().y - size.height / 2.0)
1835        }
1836        Placement::RightEnd => Point::new(anchor.max_x() + gap, anchor.max_y() - size.height),
1837        Placement::LeftStart => Point::new(anchor.x() - gap - size.width, anchor.y()),
1838        Placement::LeftEnd => Point::new(anchor.x() - gap - size.width, anchor.max_y() - size.height),
1839        Placement::Fill => return Point::ZERO,
1840        Placement::At(p) => p,
1841        Placement::BottomRight => Point::new(
1842            viewport.width - size.width - VIEWPORT_MARGIN,
1843            viewport.height - size.height - VIEWPORT_MARGIN,
1844        ),
1845        Placement::BottomLeft => Point::new(VIEWPORT_MARGIN, viewport.height - size.height - VIEWPORT_MARGIN),
1846        Placement::BottomCenter => {
1847            Point::new((viewport.width - size.width) / 2.0, viewport.height - size.height - VIEWPORT_MARGIN)
1848        }
1849        Placement::TopLeft => Point::new(VIEWPORT_MARGIN, VIEWPORT_MARGIN),
1850        Placement::TopCenter => Point::new((viewport.width - size.width) / 2.0, VIEWPORT_MARGIN),
1851        Placement::TopRight => Point::new(viewport.width - size.width - VIEWPORT_MARGIN, VIEWPORT_MARGIN),
1852    };
1853    // A layer at the side changes sides when only the other one has room (a submenu at
1854    // the window's edge), the same way one below goes above.
1855    let left = anchor.x() - gap - size.width;
1856    if placement == Placement::Right && p.x + size.width > viewport.width && left >= 0.0 {
1857        p.x = left;
1858    }
1859    // Flip vertically when there is no room, then clamp.
1860    if placement == Placement::Below && p.y + size.height > viewport.height {
1861        p.y = anchor.y() - gap - size.height;
1862    }
1863    if matches!(placement, Placement::Above | Placement::AboveCenter) && p.y < 0.0 {
1864        p.y = anchor.max_y() + gap;
1865    }
1866    p.x = p.x.clamp(4.0, (viewport.width - size.width - 4.0).max(4.0));
1867    p.y = p.y.clamp(4.0, (viewport.height - size.height - 4.0).max(4.0));
1868    p
1869}
