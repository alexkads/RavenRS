78    /// Always, even for one panel.
79    #[default]
80    Always,

87/// Panels in a tree of splits and groups of tabs, at the left, the right, the bottom and the center
88/// of a space. Bars between them resize with the pointer or the keyboard, docks collapse from the
89/// strips on their edges, and panels close and come back.
90///
91/// A tab goes anywhere with a drag, with the target marked: to another group, to one side of a
92/// group, or to a border that has no dock showing. Without a pointer, the arrows walk between tabs,
93/// Alt with an arrow moves one within its group, Alt and Shift with an arrow send it to a dock, the
94/// menu of a tab or of the button at the end of the strip has "Move to…", and Delete closes.
95///
96/// The arrangement is a [`DockLayout`] in a [`Binding`], which holds ids alone and can be saved as
97/// text. What a panel shows is asked from the app, by id, for the panel at the front of each group:
98/// that is all a change of tab builds, and what a panel must remember while behind others belongs in
99/// the app. An id in the arrangement with no [`DockPanel`] is left out.
100///
101/// ```
102/// use ravenrs_reactive::State;
103/// use ravenrs_widgets::{AnyView, DockArea, DockLayout, DockNode, DockPanel, Text, Ui};
104///
105/// let saved = None::<String>; // What the app kept the last time.
106/// let start = DockLayout::new()
107///     .center(DockNode::tabs(["photo"]))
108///     .left(DockNode::tabs(["folders", "search"]), 240.0);
109/// let layout = State::new(saved.and_then(|text| DockLayout::parse(&text).ok()).unwrap_or(start));
110///
111/// let area = DockArea::new(
112///     layout.binding(),
113///     vec![
114///         DockPanel::new("photo", "Photo").closable(false),
115///         DockPanel::new("folders", "Folders"),
116///         DockPanel::new("search", "Search"),
117///     ],
118///     |id| -> AnyView { Box::new(Text::new(format!("Panel {id}"))) },
119/// )
120/// .on_end(|now| println!("{}", now.to_text()));
121/// let mut ui = Ui::new(area, ravenrs_style::Theme::raven_dark());
122/// ui.update();
123/// ```
124pub struct DockArea {
125    layout: Binding<DockLayout>,
126    panels: Vec<DockPanel>,
127    content: Rc<dyn Fn(&str) -> AnyView>,
128    height: Option<f32>,
129    metrics: Metrics,
130    movable: bool,
131    tab_bar: DockTabBar,
132    on_end: Option<OnEnd>,
133    label: Option<String>,
134}

254/// A key for something under the area that stays the same wherever it goes.
255fn key_of(kind: u8, text: &str) -> u64 {
256    let mut hash = 0xcbf2_9ce4_8422_2325_u64 ^ u64::from(kind);
257    for byte in text.bytes() {
258        hash = (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3);
259    }
260    hash
261}

293/// What the parts of the area change the arrangement through.
294#[derive(Clone)]
295struct Actions {
296    layout: Binding<DockLayout>,
297    on_end: Option<OnEnd>,
298    spoken: State<String>,
299    /// The panel whose tab takes focus when it is next built.
300    focus: Rc<RefCell<Option<String>>>,
301    labels: Rc<Labels>,
302}
303
304impl Actions {
305    fn apply(&self, change: impl FnOnce(&mut DockLayout) -> bool) -> bool {
306        let mut next = self.layout.get();
307        if !change(&mut next) {
308            return false;
309        }
310        self.layout.set(next.clone());
311        if let Some(on_end) = &self.on_end {
312            on_end(&next);
313        }
314        true
315    }
316
317    fn settled(&self) {
318        if let Some(on_end) = &self.on_end {
319            on_end(&self.layout.get());
320        }
321    }
322
323    /// Sends a panel somewhere, keeping focus on its tab and saying where it went.
324    fn send(&self, id: &str, target: DockDrop) -> bool {
325        let moved = self.apply(|layout| layout.move_panel(id, target));
326        if moved {
327            *self.focus.borrow_mut() = Some(id.to_owned());
328            if let Some(side) = self.layout.get().side_of(id) {
329                self.spoken.set(self.labels.dock_moved_to(side));
330            }
331        }
332        moved
333    }
334
335    fn close(&self, id: &str, next: Option<String>, title: &str) {
336        if self.apply(|layout| layout.close(id)) {
337            *self.focus.borrow_mut() = next;
338            self.spoken.set(self.labels.dock_closed_name(title));
339        }
340    }
341
342    fn reopen(&self, id: &str) {
343        if self.apply(|layout| layout.reopen(id)) {
344            *self.focus.borrow_mut() = Some(id.to_owned());
345        }
346    }
347}

349/// The items of a panel's menu.
350fn panel_items(
351    area: &DockArea,
352    layout: &DockLayout,
353    arranged: &Arranged,
354    actions: &Actions,
355    id: &str,
356) -> Vec<MenuItem> {
357    let labels = &actions.labels;
358    let mut items = Vec::new();
359    if area.movable {
360        // These menus are made on every build, so what each place allows is read off the tree
361        // rather than tried on a copy of it.
362        let own_group = arranged.groups.iter().find(|group| group.panels.iter().any(|panel| panel == id));
363        let mut places = Vec::new();
364        for side in DockSide::ALL_IN_MENU {
365            if layout.side_of(id) != Some(side) {
366                places.push((labels.dock_place(side).to_owned(), DockDrop::Side(side)));
367            }
368        }
369        for group in &arranged.groups {
370            if own_group.is_some_and(|own| std::ptr::eq(own, group)) {
371                continue;
372            }
373            let Some(first) = group.panels.first() else { continue };
374            let shown = group.panels.get(group.active).unwrap_or(first);
375            let target = DockDrop::Tab { beside: first.clone(), at: group.panels.len() };
376            places.push((labels.dock_with_name(area.title_of(shown)), target));
377        }
378        if own_group.is_some_and(|own| own.panels.len() > 1) {
379            let split = |edge| DockDrop::Split { beside: id.to_owned(), edge };
380            places.push((labels.dock_split_right.clone(), split(DockEdge::Right)));
381            places.push((labels.dock_split_below.clone(), split(DockEdge::Bottom)));
382        }
383        if !places.is_empty() {
384            let mut item = MenuItem::new(labels.dock_move_to.clone());
385            for (name, target) in places {
386                let (actions, id) = (actions.clone(), id.to_owned());
387                item = item.item(MenuItem::new(name).on_select(move || {
388                    actions.send(&id, target.clone());
389                }));
390            }
391            items.push(item);
392        }
393    }
394    if area.panel(id).is_some_and(|panel| panel.closable) {
395        let (actions, id, title) = (actions.clone(), id.to_owned(), area.title_of(id).to_owned());
396        items.push(
397            MenuItem::new(labels.dock_close.clone()).on_select(move || actions.close(&id, None, &title)),
398        );
399    }
400    if let Some(reopen) = reopen_item(area, layout, actions) {
401        items.push(reopen);
402    }
403    items
404}

406/// The submenu of what can come back, if anything can.
407fn reopen_item(area: &DockArea, layout: &DockLayout, actions: &Actions) -> Option<MenuItem> {
408    let closed = layout.closed_panels();
409    if closed.is_empty() {
410        return None;
411    }
412    let mut item = MenuItem::new(actions.labels.dock_reopen.clone());
413    for id in closed {
414        let (actions, owned) = (actions.clone(), id.to_owned());
415        item =
416            item.item(MenuItem::new(area.title_of(id).to_owned()).on_select(move || actions.reopen(&owned)));
417    }
418    Some(item)
419}

421/// What a tab's drag does: it aims as it goes, and moves on release.
422struct DragContext {
423    actions: Actions,
424    state: State<Option<TabDrag>>,
425    frames: Rc<Frames>,
426    arranged: Rc<Arranged>,
427    tab_bar: DockTabBar,
428}
429
430impl DragContext {
431    fn aim_at(&self, id: &str, position: Point) -> Option<DropAim> {
432        let root = self.frames.root.get();
433        let at = Point::new(position.x - root.x(), position.y - root.y());
434        let tabs = self.frames.tabs.borrow();
435        let tab_bar = self.tab_bar;
436        let strip = move |group: &GroupPlace| strip_height(tab_bar, group.panels.len());
437        let tab = |panel: &str| {
438            tabs.get(panel).map(|f| Rect::new(f.x() - root.x(), f.y() - root.y(), f.width(), f.height()))
439        };
440        let found = geometry::aim(&self.arranged, at, &strip, &tab)?;
441        // A drop that would leave things as they are marks nothing.
442        self.actions.layout.get().move_panel(id, found.target.clone()).then_some(found)
443    }
444
445    fn handler(self, id: String) -> crate::context::DragHandler {
446        Rc::new(move |drag| match drag.phase {
447            DragPhase::Start => self.state.set(Some(TabDrag {
448                panel: id.clone(),
449                pressed_at: drag.position,
450                moving: false,
451                aim: None,
452            })),
453            DragPhase::Move => {
454                // Nothing here when it was called off with Escape.
455                let Some(mut now) = self.state.get() else { return };
456                let (dx, dy) = (drag.position.x - now.pressed_at.x, drag.position.y - now.pressed_at.y);
457                now.moving |= dx.hypot(dy) > reorder::DRAG_THRESHOLD;
458                if now.moving {
459                    now.aim = self.aim_at(&id, drag.position);
460                }
461                self.state.set_if_changed(Some(now));
462            }
463            DragPhase::End => {
464                if let Some(done) = self.state.get()
465                    && let Some(aim) = done.aim
466                {
467                    self.actions.send(&id, aim.target);
468                }
469                self.state.set(None);
470            }
471        })
472    }
473}

475/// A tab.
476struct TabView<'a> {
477    area: &'a DockArea,
478    panel: &'a DockPanel,
479    group: &'a GroupPlace,
480    index: usize,
481    on: bool,
482    actions: &'a Actions,
483    frames: &'a Rc<Frames>,
484    drag: &'a State<Option<TabDrag>>,
485    arranged: &'a Rc<Arranged>,
486    roving: Roving,
487}
488
489impl View for TabView<'_> {
490    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
491        let theme = *cx.theme();
492        let labels = cx.labels();
493        let (panel, on) = (self.panel, self.on);
494        let hovered = cx.is_hover_within();
495        let dragged = self.drag.get().is_some_and(|d| d.moving && d.panel == panel.id);
496        let ink = theme.color(if on { Semantic::Ink } else { Semantic::InkMuted });
497        let attrs = util::attrs(&theme, TextStyle::Callout, None);
498        let title_width = (cx.measure_text(&panel.title, &attrs).width.ceil() + 1.0).min(TITLE_MAX_WIDTH);
499        let gap = theme.spacing.s2;
500        let right_pad = if panel.closable { theme.spacing.s1 } else { theme.spacing.s3 };
501        let mut width = theme.spacing.s3 + right_pad + title_width;
502        let mut parts = Vec::with_capacity(3);
503        if let Some(icon) = panel.icon {
504            parts.push(util::icon(cx, icon, 14.0, ink));
505            width += 14.0 + gap;
506        }
507        let title = cx.text_truncated(
508            &panel.title,
509            &attrs,
510            ink,
511            LayoutStyle { width: Some(title_width), min_width: Some(0.0), ..LayoutStyle::leaf() },
512        );
513        cx.semantics(title, Semantics::hidden());
514        parts.push(title);
515        let (id, group_panels) = (panel.id.clone(), &self.group.panels);
516        // Whoever is beside it takes focus when it closes from the keyboard.
517        let next = group_panels
518            .get(self.index + 1)
519            .or_else(|| self.index.checked_sub(1).and_then(|i| group_panels.get(i)))
520            .cloned();
521        if panel.closable {
522            let (actions, id, title) = (self.actions.clone(), id.clone(), panel.title.clone());
523            width += ButtonSize::Mini.height() + gap;
524            let button = IconButton::new(IconName::X)
525                .label(labels.dock_close_name(&panel.title))
526                .size(ButtonSize::Mini)
527                .on_click(move || actions.close(&id, None, &title));
528            parts.push(cx.child(&button));
529        }
530        let fill = if dragged {
531            theme.color(Semantic::Accent).with_alpha(0.16)
532        } else if on {
533            theme.color(Semantic::Surface)
534        } else if hovered {
535            theme.color(Semantic::Ink).with_alpha(0.05)
536        } else {
537            Color::TRANSPARENT
538        };
539        let radius = theme.radii.sm;
540        let (frames, recorded) = (self.frames.clone(), panel.id.clone());
541        let node = cx.container(
542            LayoutStyle {
543                height: Some(TAB_HEIGHT),
544                width: Some(width),
545                min_width: Some(width),
546                spacing: gap,
547                padding: Insets { top: 0.0, bottom: 0.0, left: theme.spacing.s3, right: right_pad },
548                align: Align::Center,
549                ..LayoutStyle::stack(Axis::Horizontal)
550            },
551            &parts,
552            Paint::Custom(Rc::new(move |frame, scene| {
553                frames.tabs.borrow_mut().insert(recorded.clone(), frame);
554                if fill.a > 0.0 {
555                    scene.push_quad(Quad::new(frame, fill).radius(radius));
556                }
557            })),
558        );
559        let (actions, to_show) = (self.actions.clone(), id.clone());
560        cx.on_click(
561            node,
562            Rc::new(move || {
563                actions.apply(|layout| !layout.is_active(&to_show) && layout.activate(&to_show));
564            }),
565        );
566        cx.cursor(node, CursorIcon::Pointer);
567        cx.focusable(node, radius);
568        cx.semantics(node, Semantics::new(Role::Tab).name(panel.title.clone()).selected(on));
569        if self.actions.focus.borrow().as_deref() == Some(id.as_str()) {
570            self.actions.focus.borrow_mut().take();
571            cx.request_focus_node(node);
572        }
573        let movable = self.area.movable;
574        if movable {
575            let context = DragContext {
576                actions: self.actions.clone(),
577                state: self.drag.clone(),
578                frames: self.frames.clone(),
579                arranged: self.arranged.clone(),
580                tab_bar: self.area.tab_bar,
581            };
582            cx.on_drag(node, context.handler(id.clone()));
583        }
584        let (index, count) = (self.index, group_panels.len());
585        let first = group_panels.first().cloned().unwrap_or_default();
586        let (actions, called_off, closable) = (self.actions.clone(), self.drag.clone(), panel.closable);
587        let title = panel.title.clone();
588        let key_id = id.clone();
589        let before: crate::KeyHandler = Rc::new(move |key, modifiers| {
590            if *key == Key::Escape && called_off.peek().is_some() {
591                called_off.set(None);
592                return true;
593            }
594            if *key == Key::Delete && closable {
595                actions.close(&key_id, next.clone(), &title);
596                return true;
597            }
598            if !movable {
599                return false;
600            }
601            if let Some(side) = dock_key(key, modifiers) {
602                actions.send(&key_id, DockDrop::Side(side));
603                return true;
604            }
605            if let Some(to) = reorder::key_target(key, modifiers, index, count, false) {
606                // A gap, as the layout counts them: past the tab itself it is one more.
607                let at = if to > index { to + 1 } else { to };
608                if actions.send(&key_id, DockDrop::Tab { beside: first.clone(), at }) {
609                    actions.spoken.set(actions.labels.moved_to_position(to + 1, count));
610                }
611            }
612            reorder::is_move_key(key, modifiers)
613        });
614        let (actions, panels) = (self.actions.clone(), group_panels.clone());
615        let opens: Rc<dyn Fn(usize)> = Rc::new(move |to| {
616            if let Some(id) = panels.get(to) {
617                actions.apply(|layout| layout.activate(id));
618            }
619        });
620        self.roving.wire_with(cx, node, Some(opens), Some(before));
621        node
622    }
623}

625/// Which dock Alt and Shift with an arrow send a panel to.
626fn dock_key(key: &Key, modifiers: Modifiers) -> Option<DockSide> {
627    if !(modifiers.alt && modifiers.shift) {
628        return None;
629    }
630    match key {
631        Key::ArrowLeft => Some(DockSide::Left),
632        Key::ArrowRight => Some(DockSide::Right),
633        Key::ArrowDown => Some(DockSide::Bottom),
634        Key::ArrowUp => Some(DockSide::Center),
635        _ => None,
636    }
637}

646impl View for TabWithMenu<'_> {
647    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {

649        let Some(panel) = strip.group.panels.get(self.index).and_then(|id| strip.area.panel(id)) else {
650            return cx.empty();
651        };
652        let view = TabView {
653            area: strip.area,
654            panel,
655            group: strip.group,
656            index: self.index,
657            on: self.index == strip.group.active,
658            actions: strip.actions,
659            frames: strip.frames,
660            drag: strip.drag,
661            arranged: strip.arranged,
662            roving: Roving {
663                index: self.index,
664                count: strip.group.panels.len(),
665                focus_to: self.focus_to.clone(),
666            },
667        };
668        let mut menu = ContextMenu::new(view);
669        for item in panel_items(strip.area, strip.layout, strip.arranged, strip.actions, &panel.id) {
670            menu = menu.item(item);
671        }
672        menu.build(cx)
673    }

676/// The tabs of a group, and the button with the menu of the one that shows.
677struct Strip<'a> {
678    area: &'a DockArea,
679    group: &'a GroupPlace,
680    layout: &'a DockLayout,
681    actions: &'a Actions,
682    frames: &'a Rc<Frames>,
683    drag: &'a State<Option<TabDrag>>,
684    arranged: &'a Rc<Arranged>,
685}
686
687impl View for Strip<'_> {
688    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
689        let theme = *cx.theme();
690        let labels = cx.labels();
691        let focus_to = util::roving_cell(cx, None);
692        let tabs: Vec<Keyed<TabWithMenu<'_>>> = self
693            .group
694            .panels
695            .iter()
696            .enumerate()
697            .map(|(index, id)| {
698                Keyed::new(
699                    key_of(KEY_PANEL, id),
700                    TabWithMenu { strip: self, index, focus_to: focus_to.clone() },
701                )
702            })
703            .collect();
704        let tabs =
705            cx.child(&ScrollEdgeArrow::new(tabs).spacing(2.0).fade(Semantic::Canvas).height(TAB_HEIGHT));
706        let tabs = cx.container(
707            LayoutStyle {
708                grow: 1.0,
709                min_width: Some(0.0),
710                align: Align::Stretch,
711                ..LayoutStyle::stack(Axis::Vertical)
712            },
713            &[tabs],
714            Paint::None,
715        );
716        cx.semantics(tabs, Semantics::new(Role::TabList).name(labels.dock_tabs_name.clone()));
717        let mut nodes = vec![tabs];
718        let shown = self.group.panels.get(self.group.active).map(String::as_str).unwrap_or_default();
719        let items = panel_items(self.area, self.layout, self.arranged, self.actions, shown);
720        if !items.is_empty() {
721            let entries: Vec<Entry> = items.into_iter().map(Entry::Item).collect();
722            nodes.push(
723                cx.child(&Options { entries, name: labels.dock_options_name(self.area.title_of(shown)) }),
724            );
725        }
726        cx.container(
727            LayoutStyle {
728                height: Some(TAB_HEIGHT),
729                min_width: Some(0.0),
730                align: Align::Center,
731                spacing: theme.spacing.s1,
732                padding: Insets { top: 0.0, bottom: 0.0, left: 0.0, right: theme.spacing.s1 },
733                align_self: Some(Align::Stretch),
734                ..LayoutStyle::stack(Axis::Horizontal)
735            },
736            &nodes,
737            Paint::Quad(QuadPaint::new(theme.color(Semantic::Canvas))),
738        )
739    }
740}

761/// The strip that collapses a dock and brings it back.
762struct EdgeToggle {
763    rect: Rect,
764    side: DockSide,
765    open: bool,
766    actions: Actions,
767}
768
769impl View for EdgeToggle {
770    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
771        let theme = *cx.theme();
772        let labels = cx.labels();
773        let lit = cx.is_hovered() || cx.is_focus_visible();
774        // It points the way the dock will go: out to hide, in to show.
775        let icon = match (self.side, self.open) {
776            (DockSide::Left, true) | (DockSide::Right, false) => IconName::ChevronLeft,
777            (DockSide::Left, false) | (DockSide::Right, true) => IconName::ChevronRight,
778            (_, true) => IconName::ChevronDown,
779            (_, false) => IconName::ChevronUp,
780        };
781        let ink = theme.color(if lit { Semantic::Ink } else { Semantic::InkMuted });
782        let glyph = util::icon(cx, icon, 10.0, ink);
783        cx.semantics(glyph, Semantics::hidden());
784        let fill = if lit { theme.color(Semantic::Ink).with_alpha(0.06) } else { Color::TRANSPARENT };
785        let node = cx.container(
786            LayoutStyle { align: Align::Center, justify: Justify::Center, ..placed(self.rect) },
787            &[glyph],
788            Paint::Quad(QuadPaint::new(fill)),
789        );
790        let (actions, side, open) = (self.actions.clone(), self.side, self.open);
791        cx.on_click(
792            node,
793            Rc::new(move || {
794                actions.apply(|layout| {
795                    layout.set_open(side, !open);
796                    true
797                });
798            }),
799        );
800        cx.cursor(node, CursorIcon::Pointer);
801        cx.focusable(node, 2.0);
802        let name = if open { labels.dock_hide_name(self.side) } else { labels.dock_show_name(self.side) };
803        cx.semantics(node, Semantics::new(Role::Button).name(name).expanded(open));
804        node
805    }
806}

808/// What shows when the center holds nothing: the words, and a way back for each closed panel.
809struct EmptyCenter<'a> {
810    area: &'a DockArea,
811    layout: &'a DockLayout,
812    actions: &'a Actions,
813    rect: Rect,
814}

816impl View for EmptyCenter<'_> {
817    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
818        let theme = *cx.theme();
819        let labels = cx.labels();
820        let attrs = util::attrs(&theme, TextStyle::Callout, None);
821        let words = cx.text(&labels.dock_empty, &attrs, theme.color(Semantic::InkMuted), LayoutStyle::leaf());
822        let mut buttons = Vec::new();
823        for id in self.layout.closed_panels() {
824            let (actions, owned) = (self.actions.clone(), id.to_owned());

826                .style(ButtonStyle::Secondary)

831        let row = cx.container(
832            LayoutStyle {
833                wrap: true,
834                spacing: theme.spacing.s2,
835                justify: Justify::Center,
836                align_self: Some(Align::Stretch),
837                ..LayoutStyle::stack(Axis::Horizontal)
838            },
839            &buttons,
840            Paint::None,
841        );
842        cx.container(
843            LayoutStyle {
844                align: Align::Center,
845                justify: Justify::Center,
846                spacing: theme.spacing.s3,
847                padding: Insets::uniform(theme.spacing.s5),
848                ..placed(self.rect)
849            },
850            &[words, row],
851            Paint::None,
852        )
853    }

856impl View for DockArea {
857    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
858        let theme = *cx.theme();
859        let labels = cx.labels();
860        let drag: State<Option<TabDrag>> = cx.use_state(|| None);
861        let spoken: State<String> = cx.use_state(String::new);
862        let frames: Rc<Frames> = cx.use_state(|| Rc::new(Frames::default())).peek().clone();
863        let focus: Rc<RefCell<Option<String>>> = cx.use_state(|| Rc::new(RefCell::new(None))).peek().clone();
864        let known: Vec<&str> = self.panels.iter().map(|panel| panel.id.as_str()).collect();
865        let mut layout = self.layout.get();
866        // From here on what was saved fits what exists, and it is saved that way next time.
867        if layout.reconcile(&known) {
868            self.layout.set(layout.clone());
869        }
870        let actions = Actions {
871            layout: self.layout.clone(),
872            on_end: self.on_end.clone(),
873            spoken: spoken.clone(),
874            focus,
875            labels: labels.clone(),
876        };
877        // Known from the second build on; until then the window is the best guess there is.
878        let size = cx.scroll_viewport().unwrap_or_else(|| {
879            let window = cx.viewport();
880            Size::new(window.width, self.height.unwrap_or(window.height))
881        });
882        let arranged = Rc::new(geometry::arrange(&layout, size, &self.metrics));
883        let mut nodes = Vec::with_capacity(arranged.groups.len() * 2 + arranged.bars.len() + 5);
884        for group in &arranged.groups {
885            nodes.push(self.build_group(cx, group, &layout, &actions, &frames, &drag, &arranged));
886        }
887        for bar in &arranged.bars {
888            nodes.push(self.build_bar(cx, bar, &actions));
889        }
890        for (side, rect) in &arranged.toggles {
891            let toggle = EdgeToggle {
892                rect: *rect,
893                side: *side,
894                open: layout.is_open(*side),
895                actions: actions.clone(),
896            };
897            nodes.push(cx.keyed_child(key_of(KEY_TOGGLE, labels.dock_place(*side)), &toggle));
898        }
899        if layout.node(DockSide::Center).is_none() {
900            let empty = EmptyCenter { area: self, layout: &layout, actions: &actions, rect: arranged.center };
901            nodes.push(cx.child(&empty));
902        }
903        if let Some(aim) = drag.get().and_then(|d| d.aim) {
904            nodes.push(mark(cx, aim.mark));
905        }
906        let announcement = spoken.get();
907        nodes.push(reorder::announcer(cx, announcement));
908        let recorded = frames.clone();
909        let node = cx.container(
910            LayoutStyle {
911                height: self.height,
912                grow: if self.height.is_none() { 1.0 } else { 0.0 },
913                min_width: Some(0.0),
914                min_height: Some(0.0),
915                align_self: Some(Align::Stretch),
916                overflow_hidden: true,
917                ..LayoutStyle::stack(Axis::Vertical)
918            },
919            &nodes,
920            Paint::Custom(Rc::new(move |frame, scene| {
921                recorded.root.set(frame);
922                scene.push_quad(Quad::new(frame, theme.color(Semantic::Canvas)));
923            })),
924        );
925        cx.report_frame(node);
926        cx.clip(node);
927        let name = self.label.clone().unwrap_or_else(|| labels.dock_area_name.clone());
928        cx.semantics(node, Semantics::new(Role::Group).name(name));
929        node
930    }
931}

933impl DockArea {
934    #[allow(clippy::too_many_arguments, reason = "the parts of one build, private to it")]
935    fn build_group(
936        &self,
937        cx: &mut BuildCx<'_>,
938        group: &GroupPlace,
939        layout: &DockLayout,
940        actions: &Actions,
941        frames: &Rc<Frames>,
942        drag: &State<Option<TabDrag>>,
943        arranged: &Rc<Arranged>,
944    ) -> NodeId {
945        let theme = *cx.theme();
946        let mut parts = Vec::with_capacity(2);
947        let first = group.panels.first().map(String::as_str).unwrap_or_default();
948        if self.strip_height(group.panels.len()) > 0.0 {
949            let strip = Strip { area: self, group, layout, actions, frames, drag, arranged };
950            parts.push(cx.keyed_child(key_of(KEY_STRIP, first), &strip));
951        }
952        let shown = group.panels.get(group.active).map(String::as_str).unwrap_or_default();
953        // Under the area itself and by id: what a panel keeps goes with it to any group.
954        let content = (self.content)(shown);
955        let content = cx.keyed_child(key_of(KEY_PANEL, shown), content.as_ref());
956        let body = cx.container(
957            LayoutStyle {
958                grow: 1.0,
959                height: Some(0.0),
960                min_height: Some(0.0),
961                min_width: Some(0.0),
962                align: Align::Stretch,
963                overflow_hidden: true,
964                ..LayoutStyle::stack(Axis::Vertical)
965            },
966            &[content],
967            Paint::Quad(QuadPaint::new(theme.color(Semantic::Surface))),
968        );
969        cx.clip(body);
970        cx.semantics(body, Semantics::new(Role::TabPanel).name(self.title_of(shown).to_owned()));
971        parts.push(body);
972        let node = cx.container(placed(group.rect), &parts, Paint::None);
973        cx.clip(node);
974        node
975    }
976
977    fn build_bar(&self, cx: &mut BuildCx<'_>, bar: &BarPlace, actions: &Actions) -> NodeId {
978        let labels = cx.labels();
979        let (value, write) = (bar.value, self.layout.clone());
980        let (name, key, inverted) = match &bar.kind {
981            BarKind::Dock(side) => (
982                labels.dock_resize_name(*side),
983                format!("{side:?}"),
984                matches!(side, DockSide::Right | DockSide::Bottom),
985            ),
986            BarKind::Split { side, path, index, names, .. } => (
987                labels.dock_resize_between(self.title_of(&names.0), self.title_of(&names.1)),
988                format!("{side:?}{path:?}{index}"),
989                false,
990            ),
991        };
992        let kind = bar.kind.clone();
993        let binding = Binding::new(
994            move || value,
995            move |next: f32| {
996                let mut layout = write.get();
997                match &kind {
998                    BarKind::Dock(side) => layout.set_size(*side, next),
999                    BarKind::Split { side, path, index, both, .. } => {
1000                        layout.resize_split(*side, path, *index, next, *both);
1001                    }
1002                }
1003                write.set(layout);
1004            },
1005        );
1006        let mut handle = SplitHandle::new(bar.axis, binding).range(bar.min, bar.max).label(name);
1007        if inverted {
1008            handle = handle.inverted();
1009        }
1010        let actions = actions.clone();
1011        let handle = handle.on_end(move || actions.settled());
1012        let handle = cx.keyed_child(key_of(KEY_BAR, &key), &handle);
1013        // The bar stretches across its stack, so the stack goes the way the bar's own axis does.
1014        let style = LayoutStyle { axis: Some(bar.axis), ..placed(bar.rect) };
1015        cx.container(style, &[handle], Paint::None)
1016    }
1017}

1019/// The mark of where a drop lands: a line between tabs, or the room a panel would take.
1020fn mark(cx: &mut BuildCx<'_>, rect: Rect) -> NodeId {
1021    let theme = *cx.theme();
1022    let accent = theme.color(Semantic::Accent);
1023    let line = rect.width() <= 2.5;
1024    let paint = if line {
1025        QuadPaint { corner_radius: 1.0, ..QuadPaint::new(accent) }
1026    } else {
1027        QuadPaint {
1028            corner_radius: theme.radii.sm,
1029            border: Some(Border { width: 2.0, color: accent }),
1030            ..QuadPaint::new(accent.with_alpha(0.18))
1031        }
1032    };
1033    let node = cx.container(placed(rect), &[], Paint::Quad(paint));
1034    cx.semantics(node, Semantics::hidden());
1035    node
