11/// How far inside a group, as a part of it, a drop still means its edge.
12const EDGE_PART: f32 = 0.25;
13/// How close to the border of the area a drop means a dock that is not showing.
14const BORDER_ZONE: f32 = 28.0;
15/// How thick the line that marks a gap between tabs is.
16const GAP_LINE: f32 = 2.0;
17
18/// The measures an arrangement is laid out with.
19#[derive(Clone, Copy, Debug, PartialEq)]
20pub(crate) struct Metrics {
21    /// How thick a bar between parts is.
22    pub bar: f32,
23    /// How thick the strip that collapses a dock is, or zero without one.
24    pub toggle: f32,
25    /// The smallest and the largest the left dock gets.
26    pub left: (f32, f32),
27    /// The smallest and the largest the right dock gets.
28    pub right: (f32, f32),
29    /// The smallest and the largest the bottom dock gets.
30    pub bottom: (f32, f32),
31    /// The room the center keeps, which docks do not take from it.
32    pub center_min: (f32, f32),
33    /// The smallest a group gets along a split.
34    pub pane_min: f32,
35}
36
37impl Metrics {
38    fn limits(&self, side: DockSide) -> (f32, f32) {
39        match side {
40            DockSide::Left => self.left,
41            DockSide::Right => self.right,
42            DockSide::Bottom => self.bottom,
43            _ => (0.0, f32::MAX),
44        }
45    }
46}

91/// The sizes the parts of a split get in `room`: their shares, with each part at least at its
92/// smallest and the difference taken from those that have some to spare.
93pub(crate) fn distribute(room: f32, shares: &[f32], mins: &[f32]) -> Vec<f32> {
94    let mut sizes: Vec<f32> = shares.iter().zip(mins).map(|(share, min)| (share * room).max(*min)).collect();
95    let mut over = sizes.iter().sum::<f32>() - room;
96    if over > 0.0 {
97        let mut spare: f32 = sizes.iter().zip(mins).map(|(size, min)| size - min).sum();
98        if spare > 0.0 {
99            let part = (over / spare).min(1.0);
100            for (size, min) in sizes.iter_mut().zip(mins) {
101                *size -= (*size - min) * part;
102            }
103            spare = over.min(spare);
104            over -= spare;
105        }
106        // Too little room for every smallest: the last parts give way, down to nothing.
107        for size in sizes.iter_mut().rev() {
108            if over <= 0.0 {
109                break;
110            }
111            let taken = size.min(over);
112            *size -= taken;
113            over -= taken;
114        }
115    }
116    sizes
117}

119/// The least a node can be along `axis`.
120fn least(node: &DockNode, axis: Axis, metrics: &Metrics) -> f32 {
121    match node {
122        DockNode::Tabs { .. } => metrics.pane_min,
123        DockNode::Split { axis: own, children } => {
124            let parts = children.iter().map(|c| least(&c.node, axis, metrics));
125            if *own == axis {
126                parts.sum::<f32>() + children.len().saturating_sub(1) as f32 * metrics.bar
127            } else {
128                parts.fold(0.0, f32::max)
129            }
130        }
131    }
132}

134/// The name of a pair: what shows on either side of a bar.
135fn shown_name(node: &DockNode) -> String {
136    match node {
137        DockNode::Tabs { panels, active } => panels.get(*active).cloned().unwrap_or_default(),
138        DockNode::Split { children, .. } => children.first().map(|c| shown_name(&c.node)).unwrap_or_default(),
139    }

142fn place_node(
143    node: &DockNode,
144    rect: Rect,
145    side: DockSide,
146    path: &mut Vec<usize>,
147    metrics: &Metrics,
148    out: &mut Arranged,
149) {
150    match node {
151        DockNode::Tabs { panels, active } => {
152            out.groups.push(GroupPlace { panels: panels.clone(), active: *active, rect });
153        }
154        DockNode::Split { axis, children } => place_split(*axis, children, rect, side, path, metrics, out),
155    }
156}

158fn place_split(
159    axis: Axis,
160    children: &[DockChild],
161    rect: Rect,
162    side: DockSide,
163    path: &mut Vec<usize>,
164    metrics: &Metrics,
165    out: &mut Arranged,
166) {
167    let vertical = axis == Axis::Vertical;
168    let length = if vertical { rect.height() } else { rect.width() };
169    let room = (length - children.len().saturating_sub(1) as f32 * metrics.bar).max(0.0);
170    let shares: Vec<f32> = children.iter().map(|c| c.share).collect();
171    let mins: Vec<f32> = children.iter().map(|c| least(&c.node, axis, metrics)).collect();
172    let sizes = distribute(room, &shares, &mins);
173    let mut at = if vertical { rect.y() } else { rect.x() };
174    for (index, child) in children.iter().enumerate() {
175        let size = sizes.get(index).copied().unwrap_or(0.0);
176        let part = if vertical {
177            Rect::new(rect.x(), at, rect.width(), size)
178        } else {
179            Rect::new(at, rect.y(), size, rect.height())
180        };
181        path.push(index);
182        place_node(&child.node, part, side, path, metrics, out);
183        path.pop();
184        at += size;
185        if let (Some(next), Some(next_size), Some(next_min)) =
186            (children.get(index + 1), sizes.get(index + 1), mins.get(index + 1))
187        {
188            let bar = if vertical {
189                Rect::new(rect.x(), at, rect.width(), metrics.bar)
190            } else {
191                Rect::new(at, rect.y(), metrics.bar, rect.height())
192            };
193            let both = size + next_size;
194            let min = mins[index].min(both);
195            out.bars.push(BarPlace {
196                kind: BarKind::Split {
197                    side,
198                    path: path.clone(),
199                    index,
200                    both,
201                    names: (shown_name(&child.node), shown_name(&next.node)),
202                },
203                axis,
204                rect: bar,
205                value: size,
206                min,
207                max: (both - next_min).max(min),
208            });
209            at += metrics.bar;
210        }
211    }
212}

214/// How large a dock gets out of `room`, and how large it could get while the center and `taken`
215/// keep theirs.
216fn dock_size(
217    layout: &DockLayout,
218    side: DockSide,
219    room: f32,
220    taken: f32,
221    keep: f32,
222    metrics: &Metrics,
223) -> (f32, f32) {
224    let (min, max) = metrics.limits(side);
225    let most = (room - taken - keep - metrics.bar).clamp(min, max.max(min));
226    (layout.size(side).clamp(min, most.max(min)).min((room - taken).max(0.0)), most)
227}

229/// Lays an arrangement out in an area of `size`.
230pub(crate) fn arrange(layout: &DockLayout, size: Size, metrics: &Metrics) -> Arranged {
231    let mut out = Arranged { size, ..Arranged::default() };
232    let has = |side| layout.node(side).is_some();
233    let open = |side| has(side) && layout.is_open(side);
234    let mut x0 = 0.0;
235    let mut x1 = size.width;
236    let mut y1 = size.height;
237    let toggle = metrics.toggle;
238    let toggles = |side| has(side) && toggle > 0.0;
239    if toggles(DockSide::Left) {
240        out.toggles.push((DockSide::Left, Rect::new(0.0, 0.0, toggle, size.height)));
241        x0 += toggle;
242    }
243    if toggles(DockSide::Right) {
244        x1 -= toggle;
245        out.toggles.push((DockSide::Right, Rect::new(x1, 0.0, toggle, size.height)));
246    }
247    let width = (x1 - x0).max(0.0);
248    let (center_w, center_h) = metrics.center_min;
249    // The right gives way first when a narrow window has too little for both.
250    let left_size = open(DockSide::Left).then(|| {
251        let right_least = if open(DockSide::Right) { metrics.right.0 + metrics.bar } else { 0.0 };
252        dock_size(layout, DockSide::Left, width, right_least, center_w, metrics)
253    });
254    let right_size = open(DockSide::Right).then(|| {
255        let taken = left_size.map_or(0.0, |(left, _)| left + metrics.bar);
256        dock_size(layout, DockSide::Right, width, taken, center_w, metrics)
257    });
258    let mut path = Vec::new();
259    if let (Some((left, most)), Some(node)) = (left_size, layout.node(DockSide::Left)) {
260        place_node(node, Rect::new(x0, 0.0, left, size.height), DockSide::Left, &mut path, metrics, &mut out);
261        x0 += left;
262        let (min, _) = metrics.left;
263        out.bars.push(BarPlace {
264            kind: BarKind::Dock(DockSide::Left),
265            axis: Axis::Horizontal,
266            rect: Rect::new(x0, 0.0, metrics.bar, size.height),
267            value: left,
268            min,
269            max: most,
270        });
271        x0 += metrics.bar;
272    } else if !open(DockSide::Left) {
273        let (min, max) = metrics.left;
274        let ghost = layout.size(DockSide::Left).clamp(min, max).min(width / 2.0);
275        out.ghosts.push((DockSide::Left, Rect::new(x0, 0.0, ghost, size.height)));
276    }
277    if let (Some((right, most)), Some(node)) = (right_size, layout.node(DockSide::Right)) {
278        x1 -= right;
279        place_node(
280            node,
281            Rect::new(x1, 0.0, right, size.height),
282            DockSide::Right,
283            &mut path,
284            metrics,
285            &mut out,
286        );
287        x1 -= metrics.bar;
288        let (min, _) = metrics.right;
289        out.bars.push(BarPlace {
290            kind: BarKind::Dock(DockSide::Right),
291            axis: Axis::Horizontal,
292            rect: Rect::new(x1, 0.0, metrics.bar, size.height),
293            value: right,
294            min,
295            max: most,
296        });
297    } else if !open(DockSide::Right) {
298        let (min, max) = metrics.right;
299        let ghost = layout.size(DockSide::Right).clamp(min, max).min(width / 2.0);
300        out.ghosts.push((DockSide::Right, Rect::new(x1 - ghost, 0.0, ghost, size.height)));
301    }
302    let middle = (x1 - x0).max(0.0);
303    if toggles(DockSide::Bottom) {
304        y1 -= toggle;
305        out.toggles.push((DockSide::Bottom, Rect::new(x0, y1, middle, toggle)));
306    }
307    if let (true, Some(node)) = (open(DockSide::Bottom), layout.node(DockSide::Bottom)) {
308        let (bottom, most) = dock_size(layout, DockSide::Bottom, y1, 0.0, center_h, metrics);
309        y1 -= bottom;
310        place_node(node, Rect::new(x0, y1, middle, bottom), DockSide::Bottom, &mut path, metrics, &mut out);
311        y1 -= metrics.bar;
312        let (min, _) = metrics.bottom;
313        out.bars.push(BarPlace {
314            kind: BarKind::Dock(DockSide::Bottom),
315            axis: Axis::Vertical,
316            rect: Rect::new(x0, y1, middle, metrics.bar),
317            value: bottom,
318            min,
319            max: most,
320        });
321    } else if !open(DockSide::Bottom) {
322        let (min, max) = metrics.bottom;
323        let ghost = layout.size(DockSide::Bottom).clamp(min, max).min(y1 / 2.0);
324        out.ghosts.push((DockSide::Bottom, Rect::new(x0, y1 - ghost, middle, ghost)));
325    }
326    out.center = Rect::new(x0, 0.0, middle, y1.max(0.0));
327    if let Some(node) = layout.node(DockSide::Center) {
328        place_node(node, out.center, DockSide::Center, &mut path, metrics, &mut out);
329    }
330    out
331}

333/// What a tab being dragged would do if dropped here.
334#[derive(Clone, Debug, PartialEq)]
335pub(crate) struct DropAim {
336    pub target: DockDrop,
337    /// What to mark, relative to the area.
338    pub mark: Rect,
339}
340
341/// Which edge of `rect` the point counts for, if it is close enough to one.
342fn edge_of(rect: Rect, at: Point) -> Option<DockEdge> {
343    let part = |near: f32, length: f32| if length > 0.0 { near / length } else { 1.0 };
344    let candidates = [
345        (part(at.x - rect.x(), rect.width()), DockEdge::Left),
346        (part(rect.max_x() - at.x, rect.width()), DockEdge::Right),
347        (part(at.y - rect.y(), rect.height()), DockEdge::Top),
348        (part(rect.max_y() - at.y, rect.height()), DockEdge::Bottom),
349    ];
350    candidates
351        .into_iter()
352        .filter(|(near, _)| *near < EDGE_PART)
353        .min_by(|a, b| a.0.total_cmp(&b.0))
354        .map(|(_, edge)| edge)
355}

357/// The half of `rect` a split on `edge` would take.
358fn half(rect: Rect, edge: DockEdge) -> Rect {
359    let (w, h) = (rect.width() / 2.0, rect.height() / 2.0);
360    match edge {
361        DockEdge::Left => Rect::new(rect.x(), rect.y(), w, rect.height()),
362        DockEdge::Right => Rect::new(rect.x() + w, rect.y(), w, rect.height()),
363        DockEdge::Top => Rect::new(rect.x(), rect.y(), rect.width(), h),
364        DockEdge::Bottom => Rect::new(rect.x(), rect.y() + h, rect.width(), h),
365    }
366}

368/// What lies under the pointer at `at` for the panel `dragged`. `strip` says how tall the tabs of
369/// a group are, and `tab` where a tab is on screen now, relative to the area. It finds a place
370/// whether or not dropping there changes anything; the caller tries that.
371pub(crate) fn aim(
372    arranged: &Arranged,
373    at: Point,
374    strip: &dyn Fn(&GroupPlace) -> f32,
375    tab: &dyn Fn(&str) -> Option<Rect>,
376) -> Option<DropAim> {
377    let size = arranged.size;
378    if at.x < 0.0 || at.y < 0.0 || at.x > size.width || at.y > size.height {
379        return None;
380    }
381    // A dock that does not show opens from the border it belongs to.
382    for (side, rect) in &arranged.ghosts {
383        let near = match side {
384            DockSide::Left => at.x < BORDER_ZONE,
385            DockSide::Right => at.x > size.width - BORDER_ZONE,
386            DockSide::Bottom => at.y > size.height - BORDER_ZONE,
387            _ => false,
388        };
389        if near {
390            return Some(DropAim { target: DockDrop::Side(*side), mark: *rect });
391        }
392    }
393    let group = arranged.groups.iter().find(|group| group.rect.contains(at));
394    let Some(group) = group else {
395        return arranged
396            .center
397            .contains(at)
398            .then_some(DropAim { target: DockDrop::Side(DockSide::Center), mark: arranged.center });
399    };
400    let first = group.panels.first()?;
401    let height = strip(group);
402    if at.y < group.rect.y() + height {
403        return Some(aim_tab(group, first, at, height, tab));
404    }
405    let body = Rect::new(
406        group.rect.x(),
407        group.rect.y() + height,
408        group.rect.width(),
409        (group.rect.height() - height).max(0.0),
410    );
411    Some(match edge_of(body, at) {
412        Some(edge) => {
413            DropAim { target: DockDrop::Split { beside: first.clone(), edge }, mark: half(body, edge) }
414        }
415        None => {
416            DropAim { target: DockDrop::Tab { beside: first.clone(), at: group.panels.len() }, mark: body }
417        }
418    })
419}

421fn aim_tab(
422    group: &GroupPlace,
423    first: &str,
424    at: Point,
425    height: f32,
426    tab: &dyn Fn(&str) -> Option<Rect>,
427) -> DropAim {
428    let frames: Vec<Option<Rect>> = group.panels.iter().map(|panel| tab(panel)).collect();
429    let start = frames.first().copied().flatten().map_or(group.rect.x(), |frame| frame.x());
430    let extents: Vec<f32> = frames.iter().map(|frame| frame.map_or(0.0, |f| f.width())).collect();
431    let spacing = match (frames.first().copied().flatten(), frames.get(1).copied().flatten()) {
432        (Some(a), Some(b)) => (b.x() - a.max_x()).max(0.0),
433        _ => 0.0,
434    };
435    let gap = reorder::gap_at(&extents, spacing, at.x - start);
436    let line = frames
437        .get(gap)
438        .copied()
439        .flatten()
440        .map(|frame| frame.x())
441        .or_else(|| frames.last().copied().flatten().map(|frame| frame.max_x()))
442        .unwrap_or(group.rect.x());
443    DropAim {
444        target: DockDrop::Tab { beside: first.to_owned(), at: gap },
445        mark: Rect::new(line - GAP_LINE / 2.0, group.rect.y(), GAP_LINE, height),
446    }
447}
