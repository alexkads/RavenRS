15/// The floating panel surface shared by menus, selects and popovers.
16fn panel_paint(theme: &Theme) -> Paint {
17    Paint::Quad(QuadPaint {
18        corner_radius: theme.radii.lg,
19        border: Some(Border { width: 1.0, color: theme.color(Semantic::Border) }),
20        shadow: Some(theme.elevation.raised),
21        ..QuadPaint::new(theme.color(Semantic::SurfaceElevated))
22    })
23}

25/// Wraps children in the floating panel surface. Clicks on its background stay
26/// in the panel instead of reaching what is underneath.
27pub(crate) fn panel(cx: &mut BuildCx<'_>, children: &[NodeId], padding: f32, min_width: f32) -> NodeId {
28    let theme = *cx.theme();
29    let style = LayoutStyle {
30        padding: Insets::uniform(padding),
31        min_width: Some(min_width),
32        align: Align::Stretch,
33        ..LayoutStyle::stack(Axis::Vertical)
34    };
35    let node = cx.container(style, children, panel_paint(&theme));
36    cx.cursor(node, CursorIcon::Default);
37    motion::popup(cx, node);
38    node
39}

62impl View for Bubble<'_> {
63    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
64        let theme = *cx.theme();
65        // Inverted colors read as a tooltip in every theme.
66        let text =
67            util::label(cx, self.0, TextStyle::Callout, Some(Weight::MEDIUM), theme.color(Semantic::Canvas));
68        let node = cx.container(
69            LayoutStyle { padding: Insets::symmetric(5.0, 10.0), ..LayoutStyle::stack(Axis::Horizontal) },

73        // The text is already the name or description of the anchor.
74        cx.semantics(node, Semantics::hidden());
75        motion::popup(cx, node);
76        node
77    }

255/// A row in a menu or select list. `on_click` runs after `close`. Rows take
256/// focus as the list opens, and arrow keys move between them.
257struct ListRow<'a> {
258    label: &'a str,
259    icon: Option<IconName>,
260    trailing: Option<&'a str>,
261    checked: bool,
262    destructive: bool,
263    close: Rc<dyn Fn()>,
264    on_click: Option<Rc<dyn Fn()>>,
265    roving: util::Roving,
266    /// `MenuItem` in menus, `ListBoxOption` in select lists.
267    role: Role,
268    /// Left (false) or Right (true) in a menubar menu: open the neighbor menu.
269    on_side: Option<Rc<dyn Fn(bool)>>,
270}
271
272impl View for ListRow<'_> {
273    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
274        let theme = *cx.theme();
275        let hovered = cx.is_hovered() || cx.is_focus_visible();
276        let ink = if self.destructive { theme.color(Semantic::Danger) } else { theme.color(Semantic::Ink) };
277        let muted = if self.destructive { ink } else { theme.color(Semantic::InkMuted) };
278        let mut children = Vec::new();
279        if let Some(icon) = self.icon {
280            children.push(util::icon(cx, icon, 16.0, muted));
281        }
282        let weight = self.checked.then_some(Weight::SEMIBOLD);
283        let label = util::label(cx, self.label, TextStyle::Body, weight, ink);
284        children.push(cx.container(
285            LayoutStyle { grow: 1.0, ..LayoutStyle::stack(Axis::Horizontal) },
286            &[label],
287            Paint::None,
288        ));
289        if let Some(t) = self.trailing {
290            children.push(util::label(cx, t, TextStyle::Callout, None, theme.color(Semantic::InkMuted)));
291        }
292        if self.checked {
293            children.push(util::icon(
294                cx,
295                IconName::Check,
296                16.0,
297                theme.color(Semantic::Accent).lerp(ink, 0.3),
298            ));
299        }
300        let fill = if hovered { theme.color(Semantic::Ink).with_alpha(0.07) } else { Color::TRANSPARENT };
301        let style = LayoutStyle {
302            padding: Insets::symmetric(0.0, 10.0),
303            height: Some(34.0),
304            spacing: 10.0,
305            align: Align::Center,
306            ..LayoutStyle::stack(Axis::Horizontal)
307        };
308        let node = cx.container(
309            style,
310            &children,
311            Paint::Quad(QuadPaint { corner_radius: 7.0, ..QuadPaint::new(fill) }),
312        );
313        motion::control(cx, node);
314        let mut semantics = Semantics::new(self.role);
315        if self.role == Role::ListBoxOption {
316            semantics = semantics.selected(self.checked);
317        }
318        if let Some(t) = self.trailing {
319            semantics = semantics.name(self.label).description(t);
320        }
321        cx.semantics(node, semantics);
322        let (action, close) = (self.on_click.clone(), self.close.clone());
323        cx.on_click(
324            node,
325            Rc::new(move || {
326                close();
327                if let Some(a) = &action {
328                    a();
329                }
330            }),
331        );
332        cx.focusable(node, 7.0);
333        let before: Option<crate::KeyHandler> = self.on_side.clone().map(|side| {
334            Rc::new(move |key: &ravenrs_core::Key, _: ravenrs_core::Modifiers| match key {
335                ravenrs_core::Key::ArrowLeft => {
336                    side(false);
337                    true
338                }
339                ravenrs_core::Key::ArrowRight => {
340                    side(true);
341                    true
342                }
343                _ => false,
344            }) as crate::KeyHandler
345        });
346        self.roving.wire_with(cx, node, None, before);
347        cx.cursor(node, CursorIcon::Pointer);
348        node
349    }
350}

357pub(crate) struct MenuList<'a> {
358    pub entries: &'a [Entry],
359    pub close: Rc<dyn Fn()>,
360    /// Set by a menubar: Left and Right open the neighbor menu.
361    pub on_side: Option<Rc<dyn Fn(bool)>>,
362}
363
364impl View for MenuList<'_> {
365    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
366        let mut rows = Vec::new();
367        let count = self.entries.iter().filter(|e| matches!(e, Entry::Item(_))).count();
368        let focus_to = list_focus(cx, 0);
369        let mut item_index = 0;
370        for (i, entry) in self.entries.iter().enumerate() {
371            match entry {
372                Entry::Item(item) => {
373                    let roving = util::Roving { index: item_index, count, focus_to: focus_to.clone() };
374                    item_index += 1;
375                    let row = ListRow {
376                        roving,
377                        label: &item.label,
378                        icon: item.icon,
379                        trailing: item.shortcut.as_deref(),
380                        checked: false,
381                        destructive: item.destructive,
382                        close: self.close.clone(),
383                        on_click: item.on_select.clone(),
384                        role: Role::MenuItem,
385                        on_side: self.on_side.clone(),
386                    };
387                    rows.push(cx.keyed_child(i as u64, &row));
388                }
389                Entry::Separator => {
390                    let line = cx.child(&crate::Divider::horizontal());
391                    rows.push(cx.container(
392                        LayoutStyle {
393                            padding: Insets::symmetric(4.0, 4.0),
394                            align: Align::Stretch,
395                            ..LayoutStyle::stack(Axis::Vertical)
396                        },
397                        &[line],
398                        Paint::None,
399                    ));
400                }
401            }
402        }
403        let node = panel(cx, &rows, 6.0, 220.0);
404        cx.semantics(node, Semantics::new(Role::Menu));
405        node
406    }
407}

409impl View for DropdownMenu {
410    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
411        let open: State<bool> = cx.use_state(|| false);
412        let toggle = open.clone();
413        let mut trigger = Button::new(self.label.clone())
414            .style(self.style)
415            .size(self.size)
416            .trailing_icon(IconName::ChevronDown)
417            .on_click(move || toggle.update(|o| *o = !*o));
418        if let Some(icon) = self.icon {
419            trigger = trigger.icon(icon);
420        }
421        let anchor = cx.child(&trigger);
422        let is_open = open.get();
423        cx.update_semantics(anchor, |s| s.expanded = Some(is_open));
424        let was_open = cx.use_state(|| Rc::new(Cell::new(false))).get();
425        if is_open {
426            was_open.set(true);
427            let close: Rc<dyn Fn()> = {
428                let open = open.clone();
429                Rc::new(move || open.set(false))
430            };
431            cx.overlay(
432                anchor,
433                Placement::Below,
434                &MenuList { entries: &self.entries, close: close.clone(), on_side: None },
435                Some(close),
436            );
437        } else if was_open.replace(false) {
438            // Focus returns to the trigger when the menu closes.
439            cx.request_focus_node(anchor);
440        }
441        anchor
442    }
443}

476struct Options<'a> {
477    options: &'a [String],
478    selected: Binding<usize>,
479    close: Rc<dyn Fn()>,
480    width: f32,
481}
482
483impl View for Options<'_> {
484    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
485        let current = self.selected.get();
486        let (count, focus_to) = (self.options.len(), list_focus(cx, current));
487        let rows: Vec<NodeId> = self
488            .options
489            .iter()
490            .enumerate()
491            .map(|(i, option)| {
492                let selected = self.selected.clone();
493                let row = ListRow {
494                    roving: util::Roving { index: i, count, focus_to: focus_to.clone() },
495                    label: option,
496                    icon: None,
497                    trailing: None,
498                    checked: i == current,
499                    destructive: false,
500                    close: self.close.clone(),
501                    on_click: Some(Rc::new(move || selected.set(i))),
502                    role: Role::ListBoxOption,
503                    on_side: None,
504                };
505                cx.keyed_child(i as u64, &row)
506            })
507            .collect();
508        let node = panel(cx, &rows, 6.0, self.width);
509        cx.semantics(node, Semantics::new(Role::ListBox));
510        node
511    }
512}

514impl View for Select {
515    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
516        let theme = *cx.theme();
517        let open: State<bool> = cx.use_state(|| false);
518        let is_open = open.get();
519        let hovered = cx.is_hovered();
520        let current = self.options.get(self.selected.get()).map_or("", String::as_str);
521        let label = util::label(cx, current, TextStyle::Body, None, theme.color(Semantic::Ink));
522        let label = cx.container(
523            LayoutStyle { grow: 1.0, ..LayoutStyle::stack(Axis::Horizontal) },
524            &[label],
525            Paint::None,
526        );
527        let chevron = util::icon(
528            cx,
529            if is_open { IconName::ChevronUp } else { IconName::ChevronDown },
530            16.0,
531            theme.color(Semantic::InkMuted),
532        );
533        let border = if is_open {
534            Border { width: 1.5, color: theme.color(Semantic::Accent) }
535        } else {
536            let base = theme.color(Semantic::Border);
537            Border {
538                width: 1.0,
539                color: if hovered { base.lerp(theme.color(Semantic::Ink), 0.25) } else { base },
540            }
541        };
542        let style = LayoutStyle {
543            padding: Insets::symmetric(0.0, 12.0),
544            width: Some(self.width),
545            height: Some(40.0),
546            spacing: 8.0,
547            align: Align::Center,
548            align_self: Some(Align::Fit),
549            ..LayoutStyle::stack(Axis::Horizontal)
550        };
551        let paint = QuadPaint {
552            corner_radius: theme.radii.md,
553            border: Some(border),
554            ..QuadPaint::new(theme.color(Semantic::Control))
555        };
556        let node = cx.container(style, &[label, chevron], Paint::Quad(paint));
557        cx.semantics(label, Semantics::hidden());
558        let mut semantics = Semantics::new(Role::ComboBox).value(current).expanded(is_open);
559        if let Some(label) = &self.label {
560            semantics = semantics.name(label.clone());
561        }
562        cx.semantics(node, semantics);
563        let toggle = open.clone();
564        cx.on_click(node, Rc::new(move || toggle.update(|o| *o = !*o)));
565        // Arrow keys open the list too, like a native select.
566        let opener = open.clone();
567        cx.on_key(
568            node,
569            Rc::new(move |key, _| {
570                let opens = matches!(key, ravenrs_core::Key::ArrowDown | ravenrs_core::Key::ArrowUp);
571                if opens {
572                    opener.set(true);
573                }
574                opens
575            }),
576        );
577        cx.focusable(node, theme.radii.md);
578        cx.cursor(node, CursorIcon::Pointer);
579        let was_open = cx.use_state(|| Rc::new(Cell::new(false))).get();
580        if !is_open && was_open.replace(false) {
581            // Focus returns to the field when the list closes.
582            cx.request_focus();
583        }
584        if is_open {
585            was_open.set(true);
586            let close: Rc<dyn Fn()> = {
587                let open = open.clone();
588                Rc::new(move || open.set(false))
589            };
590            let list = Options {
591                options: &self.options,
592                selected: self.selected.clone(),
593                close: close.clone(),
594                width: self.width,
595            };
596            cx.overlay(node, Placement::Below, &list, Some(close));
597        }
598        node
599    }
600}

671    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
672        let theme = *cx.theme();
673        let viewport = cx.viewport();
674        let d = self.dialog;
675        let close = d.open.clone();
676        let title = util::label(cx, &d.title, TextStyle::Title, None, theme.color(Semantic::Ink));
677        let title = cx.container(
678            LayoutStyle { grow: 1.0, ..LayoutStyle::stack(Axis::Horizontal) },
679            &[title],
680            Paint::None,
681        );
682        let x = cx.child(
683            &IconButton::new(IconName::X)
684                .label(d.close_label.clone().unwrap_or_else(|| cx.labels().close.clone()))
685                .size(crate::ButtonSize::Small)
686                .on_click(move || close.set(false)),
687        );
688        cx.semantics(title, Semantics::new(Role::Heading).level(2));
689        let header = cx.container(
690            LayoutStyle { align: Align::Center, spacing: 12.0, ..LayoutStyle::stack(Axis::Horizontal) },
691            &[title, x],
692            Paint::None,
693        );
694        let mut body = vec![header];
695        if let Some(text) = &d.description {
696            body.push(util::label(cx, text, TextStyle::Body, None, theme.color(Semantic::InkMuted)));
697        }
698        if let Some(content) = &d.content {
699            body.push(cx.child(content.as_ref()));
700        }
701        if !d.actions.is_empty() {
702            let actions: Vec<NodeId> = d.actions.iter().map(|a| cx.child(a.as_ref())).collect();
703            body.push(cx.container(
704                LayoutStyle {
705                    spacing: 10.0,
706                    justify: Justify::End,
707                    padding: Insets { top: 8.0, ..Insets::ZERO },
708                    ..LayoutStyle::stack(Axis::Horizontal)
709                },
710                &actions,
711                Paint::None,
712            ));
713        }
714        let card = cx.container(
715            LayoutStyle {
716                width: Some((viewport.width - 32.0).clamp(240.0, d.width.max(240.0))),
717                padding: Insets::uniform(24.0),
718                spacing: 14.0,
719                align: Align::Stretch,
720                ..LayoutStyle::stack(Axis::Vertical)
721            },
722            &body,
723            Paint::Quad(QuadPaint {
724                corner_radius: theme.radii.xl,
725                border: Some(Border { width: 1.0, color: theme.color(Semantic::Border) }),
726                shadow: Some(theme.elevation.raised),
727                ..QuadPaint::new(theme.color(Semantic::SurfaceElevated))
728            }),
729        );
730        let mut semantics = Semantics::new(Role::Dialog).name(d.title.clone()).modal();
731        if let Some(text) = &d.description {
732            semantics = semantics.description(text.clone());
733        }
734        cx.semantics(card, semantics);
735        // The card swallows clicks so only the backdrop closes the dialog.
736        cx.cursor(card, CursorIcon::Default);
737        cx.motion(
738            card,
739            Motion::new()
740                .enter(Effect::moved(0.0, 8.0).with_opacity(0.0).with_scale(0.96), theme.motion.gentle)
741                .exit(Effect::IDENTITY.with_scale(0.98), theme.motion.fast),
742        );
743        let backdrop = cx.container(
744            LayoutStyle {
745                width: Some(viewport.width),
746                height: Some(viewport.height),
747                align: Align::Center,
748                justify: Justify::Center,
749                ..LayoutStyle::stack(Axis::Vertical)
750            },
751            &[card],
752            Paint::Quad(QuadPaint::new(theme.color(Semantic::Scrim))),
753        );
754        let close = d.open.clone();
755        cx.on_click(backdrop, Rc::new(move || close.set(false)));
756        motion::scrim(cx, backdrop);
757        backdrop
758    }
759}
760
761impl View for Dialog {
762    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
763        let anchor = cx.empty();
764        if self.open.get() {
765            let close = self.open.clone();
766            cx.overlay(
767                anchor,
768                Placement::Fill,
769                &DialogLayer { dialog: self },
770                Some(Rc::new(move || close.set(false))),
771            );
772        }
773        anchor
774    }
