2use std::rc::Rc;

4use ravenrs_core::{Rect, Scene};
5use ravenrs_layout::LayoutStyle;

7use crate::View;

10/// A box of a given size that draws whatever its closure appends to the scene.
11/// The closure receives the box's absolute frame in logical pixels.
12///
13/// ```
14/// use ravenrs_core::{Color, Quad};
15/// use ravenrs_widgets::Canvas;
16///
17/// let bars = Canvas::new(|frame, scene| {
18///     for (i, v) in [0.4_f32, 0.9, 0.6].iter().enumerate() {
19///         let h = frame.height() * v;
20///         let x = frame.x() + i as f32 * 24.0;
21///         let bar = ravenrs_core::Rect::new(x, frame.max_y() - h, 16.0, h);
22///         scene.push_quad(Quad::new(bar, Color::hex(0x4c66ee)).radius(4.0));
23///     }
24/// })
25/// .size(72.0, 48.0);
26/// ```
27pub struct Canvas {
28    draw: DrawFn,
29    style: LayoutStyle,
30}
31
32impl fmt::Debug for Canvas {
33    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
34        f.debug_struct("Canvas").field("style", &self.style).finish_non_exhaustive()
35    }
36}
37
38impl Canvas {
39    /// A canvas drawn by `draw`, called on every paint with the absolute frame.
40    pub fn new(draw: impl Fn(Rect, &mut Scene) + 'static) -> Self {
41        Self { draw: Rc::new(draw), style: LayoutStyle::leaf() }
42    }
43
44    /// Fixed size in logical pixels.
45    pub fn size(mut self, width: f32, height: f32) -> Self {
46        self.style.width = Some(width);
47        self.style.height = Some(height);
48        self
49    }
50
51    /// Fills the free space along the parent stack's axis.
52    pub fn grow(mut self, grow: f32) -> Self {
53        self.style.grow = grow;
54        self
55    }
56}
57
58impl View for Canvas {
59    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
60        cx.container(self.style, &[], Paint::Custom(self.draw.clone()))
61    }
62}
