12/// A single-line text input bound to a `String`.
13///
14/// Supports typing, deleting, moving by character, word and line, selecting
15/// with Shift, the mouse (drag, double-click) or Cmd/Ctrl+A, copy, cut and
16/// paste, and Enter (submit). Passwords can be pasted but not copied.
17///
18/// ```
19/// use ravenrs_reactive::State;
20/// use ravenrs_widgets::{IconName, TextField, Ui};
21///
22/// let query = State::new(String::new());
23/// let field = TextField::new(query.binding()).placeholder("Search").icon(IconName::Search).clearable();
24/// let mut ui = Ui::new(field, ravenrs_style::Theme::raven_dark());
25/// ui.update();
26/// ```
27pub struct TextField {
28    value: Binding<String>,
29    placeholder: String,
30    icon: Option<IconName>,
31    secure: bool,
32    width: Option<f32>,
33    size: ButtonSize,
34    prefix: Option<AnyView>,
35    suffix: Option<AnyView>,
36    clearable: bool,
37    flat: bool,
38    end_aligned: bool,
39    autofocus: bool,
40    select_on_open: bool,
41    on_submit: Option<Rc<dyn Fn(String)>>,
42    on_leave: Option<Rc<dyn Fn()>>,
43    on_cancel: Option<Rc<dyn Fn()>>,
44    intercept: Option<crate::KeyHandler>,
45    state: util::EditorState,
46    /// Shown instead of the value while the field does not have focus.
47    pub(crate) idle_text: Option<String>,
48    /// A caret at the end stays there when a change makes the text longer.
49    pub(crate) caret_follows_end: bool,
50}

209/// Caret and selection of a text field, as char indices. The selection runs
210/// from `anchor` to `caret`.
211#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
212pub(crate) struct Cursor {
213    pub caret: usize,
214    pub anchor: Option<usize>,
215}
216
217impl Cursor {
218    /// A caret at `at` with nothing selected.
219    pub fn at(at: usize) -> Self {
220        Self { caret: at, anchor: None }
221    }
222
223    /// The selected range, if it is not empty.
224    pub fn selection(&self) -> Option<(usize, usize)> {
225        let a = self.anchor?;
226        (a != self.caret).then(|| (a.min(self.caret), a.max(self.caret)))
227    }
228
229    /// Moves the caret, extending the selection when `extend` is true.
230    pub(crate) fn move_to(&mut self, to: usize, extend: bool) {
231        if extend {
232            self.anchor.get_or_insert(self.caret);
233        } else {
234            self.anchor = None;
235        }
236        self.caret = to;
237    }
238}

240fn is_word(c: char) -> bool {
241    c.is_alphanumeric() || c == '_'
242}
243
244/// The start of the word before `at` (Alt/Ctrl+Left).
245fn word_start(chars: &[char], at: usize) -> usize {
246    let mut i = at.min(chars.len());
247    while i > 0 && !is_word(chars[i - 1]) {
248        i -= 1;
249    }
250    while i > 0 && is_word(chars[i - 1]) {
251        i -= 1;
252    }
253    i
254}
255
256/// The end of the word after `at` (Alt/Ctrl+Right).
257fn word_end(chars: &[char], at: usize) -> usize {
258    let mut i = at.min(chars.len());
259    while i < chars.len() && !is_word(chars[i]) {
260        i += 1;
261    }
262    while i < chars.len() && is_word(chars[i]) {
263        i += 1;
264    }
265    i
266}

268/// The word around `at`, for double-click selection.
269pub(crate) fn word_at(text: &str, at: usize) -> (usize, usize) {
270    let chars: Vec<char> = text.chars().collect();
271    let at = at.min(chars.len());
272    let on_word = |i: usize| chars.get(i).is_some_and(|c| is_word(*c));
273    if !on_word(at) && !(at > 0 && on_word(at - 1)) {
274        return (at, (at + 1).min(chars.len()));
275    }
276    let mut start = at;
277    while start > 0 && is_word(chars[start - 1]) {
278        start -= 1;
279    }
280    let mut end = at;
281    while end < chars.len() && is_word(chars[end]) {
282        end += 1;
283    }
284    (start, end)
285}

287/// Removes the selection, leaving the caret where it was. Returns false when
288/// nothing was selected.
289fn delete_selection(text: &mut String, cursor: &mut Cursor) -> bool {
290    let Some((a, b)) = cursor.selection() else { return false };
291    text.replace_range(byte_index(text, a)..byte_index(text, b), "");
292    *cursor = Cursor::at(a);
293    true
294}
295
296pub(crate) fn insert(text: &mut String, cursor: &mut Cursor, s: &str) {
297    delete_selection(text, cursor);
298    let at = byte_index(text, cursor.caret);
299    text.insert_str(at, s);
300    *cursor = Cursor::at(cursor.caret + s.chars().count());
301}

303/// How a field treats the clipboard.
304pub(crate) struct ClipboardAccess<'a> {
305    pub clipboard: &'a crate::Clipboard,
306    /// Passwords can be pasted into but never copied out of.
307    pub secure: bool,
308    /// Pasted text keeps its line breaks.
309    pub multiline: bool,
310}

312/// Applies a key to `text` and the cursor: typing, deleting, moving (Shift
313/// extends the selection; Alt or Ctrl moves by word; Cmd moves to the ends),
314/// select all, and copy, cut and paste with Cmd or Ctrl. Returns false for
315/// keys the field does not handle, including a paste the clipboard cannot
316/// read synchronously (the browser then delivers it as typed text).
317pub(crate) fn edit(
318    text: &mut String,
319    cursor: &mut Cursor,
320    key: &Key,
321    modifiers: Modifiers,
322    clipboard: Option<ClipboardAccess<'_>>,
323) -> bool {
324    let chars: Vec<char> = text.chars().collect();
325    let len = chars.len();
326    cursor.caret = cursor.caret.min(len);
327    cursor.anchor = cursor.anchor.map(|a| a.min(len));
328    let shift = modifiers.shift;
329    let word = modifiers.alt || modifiers.ctrl;
330    let command = modifiers.ctrl || modifiers.meta;
331    let multiline = clipboard.as_ref().is_some_and(|access| access.multiline);
332    if command && let Key::Character(c) = key {
333        match c.to_ascii_lowercase().as_str() {
334            "a" => *cursor = Cursor { caret: len, anchor: Some(0) },
335            "c" | "x" => {
336                let Some(access) = clipboard else { return false };
337                let Some((a, b)) = cursor.selection() else { return true };
338                if access.secure {
339                    return true;
340                }
341                access.clipboard.write(&chars[a..b].iter().collect::<String>());
342                if c.eq_ignore_ascii_case("x") {
343                    delete_selection(text, cursor);
344                }
345            }
346            "v" => {
347                let Some(pasted) = clipboard.and_then(|access| access.clipboard.read()) else { return false };
348                if multiline {
349                    insert(text, cursor, &pasted.replace("\r\n", "\n").replace('\r', "\n"));
350                } else {
351                    // A single-line field keeps the first line.
352                    let line = pasted.lines().next().unwrap_or_default();
353                    insert(text, cursor, line);
354                }
355            }
356            _ => return false,
357        }
358        return true;
359    }
360    match key {
361        // Line breaks (from a browser paste, say) become spaces in one line.
362        Key::Character(s) if !command => {
363            insert(text, cursor, &s.replace("\r\n", " ").replace(['\n', '\r'], " "))
364        }
365        Key::Space if !command => insert(text, cursor, " "),
366        Key::Backspace | Key::Delete if delete_selection(text, cursor) => {}
367        Key::Backspace if cursor.caret > 0 => {
368            let from = if modifiers.meta {
369                0
370            } else if word {
371                word_start(&chars, cursor.caret)
372            } else {
373                cursor.caret - 1
374            };
375            text.replace_range(byte_index(text, from)..byte_index(text, cursor.caret), "");
376            *cursor = Cursor::at(from);
377        }
378        Key::Delete if cursor.caret < len => {
379            let to = if word { word_end(&chars, cursor.caret) } else { cursor.caret + 1 };
380            text.replace_range(byte_index(text, cursor.caret)..byte_index(text, to), "");
381            *cursor = Cursor::at(cursor.caret);
382        }
383        Key::Backspace | Key::Delete => {}
384        Key::ArrowLeft | Key::ArrowRight => {
385            let left = *key == Key::ArrowLeft;
386            let to = match (cursor.selection(), shift) {
387                // Without Shift, an arrow collapses the selection to that side.
388                (Some((a, b)), false) if !word && !modifiers.meta => {
389                    if left {
390                        a
391                    } else {
392                        b
393                    }
394                }
395                _ if modifiers.meta => {
396                    if left {
397                        0
398                    } else {
399                        len
400                    }
401                }
402                _ if word => {
403                    if left {
404                        word_start(&chars, cursor.caret)
405                    } else {
406                        word_end(&chars, cursor.caret)
407                    }
408                }
409                _ if left => cursor.caret.saturating_sub(1),
410                _ => (cursor.caret + 1).min(len),
411            };
412            cursor.move_to(to, shift);
413        }
414        Key::Home | Key::ArrowUp => cursor.move_to(0, shift),
415        Key::End | Key::ArrowDown => cursor.move_to(len, shift),
416        _ => return false,
417    }
418    true
419}

421/// The char index nearest to `x` (relative to the text's left edge), given
422/// the x position of every char boundary.
423pub(crate) fn index_at(boundaries: &[f32], x: f32) -> usize {
424    let after = boundaries.partition_point(|b| *b < x);
425    match after {
426        0 => 0,
427        n if n >= boundaries.len() => boundaries.len().saturating_sub(1),
428        n => {
429            if x - boundaries[n - 1] < boundaries[n] - x {
430                n - 1
431            } else {
432                n
433            }
434        }
435    }
436}

438impl View for TextField {
439    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
440        let theme = *cx.theme();
441        let cursor: State<Cursor> = cx.use_state(|| {
442            if self.select_on_open {
443                Cursor { caret: self.value.get().chars().count(), anchor: Some(0) }
444            } else {
445                Cursor::at(usize::MAX)
446            }
447        });
448        let focus_requested: State<bool> = cx.use_state(|| false);
449        let state = self.state;
450        if self.autofocus && !state.disabled && !*focus_requested.peek() {
451            focus_requested.set_if_changed(true);
452            cx.request_focus();
453        }
454        // Set by the clear button, which took the focus with its click.
455        let refocus: State<bool> = cx.use_state(|| false);
456        if *refocus.peek() {
457            refocus.set(false);
458            cx.request_focus();
459        }
460        let focused = cx.is_focused() && !state.disabled;
461        let had_focus: State<bool> = cx.use_state(|| false);
462        if *had_focus.peek() != focused {
463            had_focus.set(focused);
464            if !focused && let Some(left) = &self.on_leave {
465                left();
466            }
467        }
468        let hovered = cx.is_hovered() && !state.disabled;
469        let text = self.value.get();
470        let len = text.chars().count();
471        let cur = cursor.get();
472        let caret_at = cur.caret.min(len);
473        let visible = self.idle_text.as_ref().filter(|_| !focused).unwrap_or(&text);
474        let shown: String = if self.secure { "•".repeat(len) } else { visible.clone() };
475        let dense = matches!(self.size, ButtonSize::Small | ButtonSize::Mini);
476        let attrs = util::attrs(&theme, self.size.text_style(), None);
477
478        let mut children = Vec::new();
479        if let Some(icon) = self.icon {
480            let size = if self.size == ButtonSize::Mini { 12.0 } else { 16.0 };
481            children.push(util::icon(cx, icon, size, state.ink(theme.color(Semantic::InkMuted))));
482        }
483        // Keyed, so that the clear button coming and going does not move their state.
484        if let Some(prefix) = &self.prefix {
485            children.push(cx.keyed_child(1, prefix));
486        }
487        // Where each char boundary falls, for the caret, the selection and clicks.
488        let boundaries: Rc<Vec<f32>> = Rc::new(if shown.chars().count() <= 512 {
489            let chars: Vec<char> = shown.chars().collect();
490            (0..=chars.len())
491                .map(|i| cx.measure_text(&chars[..i].iter().collect::<String>(), &attrs).width)
492                .collect()
493        } else {
494            // Long text: one measurement, spread evenly.
495            let w = cx.measure_text(&shown, &attrs).width;
496            let n = shown.chars().count();
497            (0..=n).map(|i| w * i as f32 / n as f32).collect()
498        });
499        let caret_x = boundaries.get(caret_at).copied().unwrap_or(0.0);
500        let selection = cur.selection().map(|(a, b)| {
501            let at = |i: usize| boundaries.get(i.min(len)).copied().unwrap_or(0.0);
502            (at(a), at(b))
503        });
504        let highlight = theme.color(Semantic::Accent).with_alpha(if focused { 0.32 } else { 0.16 });
505        let content = if shown.is_empty() {
506            cx.text(
507                &self.placeholder,
508                &attrs,
509                state.ink(theme.color(Semantic::InkMuted)),
510                LayoutStyle::leaf(),
511            )
512        } else {
513            cx.text(&shown, &attrs, state.ink(theme.color(Semantic::Ink)), LayoutStyle::leaf())
514        };
515        let accent = theme.color(Semantic::Accent);
516        let line = attrs.line_height;
517        let end_aligned = self.end_aligned;
518        let text_width = boundaries.last().copied().unwrap_or(0.0);
519        // How far the text starts into an area of this width.
520        let inset = move |width: f32| if end_aligned { (width - text_width).max(0.0) } else { 0.0 };
521        let text_area = cx.container(
522            LayoutStyle {
523                grow: 1.0,
524                min_width: Some(0.0),
525                align: Align::Center,
526                justify: if end_aligned { Justify::End } else { Justify::Start },
527                overflow_hidden: true,
528                ..LayoutStyle::stack(Axis::Horizontal)
529            },
530            &[content],
531            Paint::Custom(Rc::new(move |f: Rect, scene| {
532                let y = f.center().y - line / 2.0 + 2.0;
533                let start = f.x() + inset(f.width());
534                if let Some((a, b)) = selection {
535                    scene.push_quad(
536                        Quad::new(Rect::new(start + a, y, b - a, line - 4.0), highlight).radius(2.0),
537                    );
538                }
539                if focused && selection.is_none() && state.accepts_input() {
540                    let x = (start + caret_x).min(f.max_x() - 1.5);
541                    scene.push_quad(Quad::new(Rect::new(x, y, 1.5, line - 4.0), accent).radius(0.75));
542                }
543            })),
544        );
545        cx.clip(text_area);
546        // Press places the caret, dragging selects, Shift-click extends.
547        if !state.disabled {
548            let (drag_cursor, mods, b) = (cursor.clone(), cx.modifiers(), boundaries.clone());
549            cx.on_drag(
550                text_area,
551                Rc::new(move |d| {
552                    let at = index_at(&b, d.position.x - d.frame.x() - inset(d.frame.width()));
553                    match d.phase {
554                        DragPhase::Start => drag_cursor.update(|c| {
555                            if mods.get().shift {
556                                c.move_to(at, true);
557                            } else {
558                                *c = Cursor { caret: at, anchor: Some(at) };
559                            }
560                        }),
561                        DragPhase::Move => drag_cursor.update(|c| c.move_to(at, true)),
562                        _ => {}
563                    }
564                }),
565            );
566            let (word_cursor, word_value) = (cursor.clone(), self.value.clone());
567            cx.on_double_click(
568                text_area,
569                Rc::new(move || {
570                    let caret = word_cursor.get().caret;
571                    let (a, b) = word_at(&word_value.get(), caret);
572                    word_cursor.set(Cursor { caret: b, anchor: Some(a) });
573                }),
574            );
575            cx.cursor(text_area, CursorIcon::Text);
576        }
577        children.push(text_area);
578        if self.clearable && !text.is_empty() && state.accepts_input() {
579            let (value, caret, refocus) = (self.value.clone(), cursor.clone(), refocus.clone());
580            let clear = IconButton::new(IconName::X)
581                .label(cx.labels().clear.clone())
582                .style(ButtonStyle::Ghost)
583                .size(ButtonSize::Mini)
584                .on_click(move || {
585                    value.set(String::new());
586                    caret.set(Cursor::at(usize::MAX));
587                    refocus.set(true);
588                });
589            children.push(cx.keyed_child(2, &clear));
590        }
591        if let Some(suffix) = &self.suffix {
592            children.push(cx.keyed_child(3, suffix));
593        }
594
595        let border = if focused {
596            Border { width: 1.5, color: accent }
597        } else {
598            let base = theme.color(Semantic::Border);
599            Border {
600                width: 1.0,
601                color: state.ink(if hovered { base.lerp(theme.color(Semantic::Ink), 0.25) } else { base }),
602            }
603        };
604        let style = LayoutStyle {
605            padding: Insets::symmetric(0.0, if dense && !self.flat { 8.0 } else { 12.0 }),
606            height: if self.flat { None } else { Some(self.size.height()) },
607            grow: if self.flat { 1.0 } else { 0.0 },
608            width: self.width,
609            min_width: Some(if dense { 0.0 } else { 160.0 }),
610            spacing: 8.0,
611            align: Align::Center,
612            align_self: Some(if self.width.is_some() { Align::Start } else { Align::Stretch }),
613            ..LayoutStyle::stack(Axis::Horizontal)
614        };
615        let paint = if self.flat {
616            let fill = if hovered && !focused {
617                theme.color(Semantic::Ink).with_alpha(0.05)
618            } else {
619                Color::TRANSPARENT
620            };
621            QuadPaint {
622                corner_radius: theme.radii.sm,
623                border: focused.then_some(border),
624                ..QuadPaint::new(fill)
625            }
626        } else {
627            QuadPaint {
628                corner_radius: theme.radii.md,
629                border: Some(border),
630                ..QuadPaint::new(state.fill(theme.color(Semantic::Control)))
631            }
632        };
633        let node = cx.container(style, &children, Paint::Quad(paint));
634        crate::motion::control(cx, node);
635        // The visible text is exposed as the value; the placeholder names the
636        // field until a label (a FormField, say) gives it a real name.
637        cx.semantics(text_area, Semantics::hidden());
638        let mut semantics = Semantics::new(Role::TextInput).value(shown);
639        if !self.placeholder.is_empty() {
640            semantics = semantics.name(self.placeholder.clone());
641            semantics.fallback_name = true;
642        }
643        cx.semantics(node, state.semantics(semantics));
644        if state.disabled {
645            return node;
646        }
647
648        let (value, submit, cancel) = (self.value.clone(), self.on_submit.clone(), self.on_cancel.clone());
649        let (intercept, clipboard, secure) = (self.intercept.clone(), cx.clipboard(), self.secure);
650        let follows_end = self.caret_follows_end;
651        cx.on_key(
652            node,
653            Rc::new(move |key, modifiers| {
654                if let Some(i) = &intercept
655                    && i(key, modifiers)
656                {
657                    return true;
658                }
659                if *key == Key::Escape
660                    && let Some(cancel) = &cancel
661                {
662                    cancel();
663                    return true;
664                }
665                if *key == Key::Enter {
666                    if let (Some(submit), true) = (&submit, state.accepts_input()) {
667                        submit(value.get());
668                    }
669                    return true;
670                }
671                let mut text = value.get();
672                let mut c = cursor.get();
673                let access = ClipboardAccess { clipboard: &clipboard, secure, multiline: false };
674                let handled = edit(&mut text, &mut c, key, modifiers, Some(access));
675                if handled {
676                    let changed = text != value.get();
677                    if changed && !state.accepts_input() {
678                        // Copying still works; the text stays as it was.
679                        return true;
680                    }
681                    if follows_end && c.caret == text.chars().count() {
682                        c.caret = usize::MAX;
683                    }
684                    cursor.set(c);
685                    if changed {
686                        value.set(text);
687                    }
688                }
689                handled
690            }),
691        );
692        cx.focusable(node, theme.radii.md);
693        cx.cursor(node, CursorIcon::Text);
694        node
695    }
696}

711    #[test]
712    fn editing_keys() {

714        assert!(press(&mut text, &mut c, Key::Backspace, NONE));
715        assert_eq!((text.as_str(), c.caret), ("héll", 4));
716        press(&mut text, &mut c, Key::Home, NONE);
717        press(&mut text, &mut c, Key::ArrowRight, NONE);
718        press(&mut text, &mut c, Key::Delete, NONE);

720        press(&mut text, &mut c, Key::Character("é".into()), NONE);
721        press(&mut text, &mut c, Key::Space, NONE);

723        assert!(!press(&mut text, &mut c, Key::Tab, NONE));

771        let clipboard = crate::Clipboard::default();
772        let access = || Some(ClipboardAccess { clipboard: &clipboard, secure: false, multiline: false });
773        let (mut text, mut c) = (String::from("draft"), Cursor::at(0));
