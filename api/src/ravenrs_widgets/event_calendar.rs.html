21/// Something on a [`EventCalendar`]'s day.
22#[derive(Clone, Debug, PartialEq, Eq)]
23pub struct CalendarEvent {
24    /// Given back by [`EventCalendar::on_event`].
25    pub id: u64,
26    /// Its day.
27    pub date: Date,
28    /// What it is.
29    pub title: String,
30    /// Its color; say what it means in the title or next to the scheduler too.
31    pub tone: Tone,
32    /// It already happened, so it stands back.
33    pub done: bool,
34}
35
36impl CalendarEvent {
37    /// An event on this day.
38    pub fn new(id: u64, date: Date, title: impl Into<String>) -> Self {
39        Self { id, date, title: title.into(), tone: Tone::Accent, done: false }
40    }
41
42    /// Sets the color.
43    pub fn tone(mut self, tone: Tone) -> Self {
44        self.tone = tone;
45        self
46    }
47
48    /// Marks it as already done.
49    pub fn done(mut self, done: bool) -> Self {
50        self.done = done;
51        self
52    }
53}

55/// A month with its events inside each day, for agendas and bookings. Use a
56/// [`Calendar`](crate::Calendar) to pick a date and this to see what is on them.
57///
58/// From the keyboard the grid is one Tab stop: arrows move by day and week, Home
59/// and End to the ends of the week, Page Up and Page Down by month, and Enter or
60/// Space picks. A day's name reads out what is on it.
61///
62/// ```
63/// use ravenrs_reactive::State;
64/// use ravenrs_widgets::{Date, CalendarEvent, EventCalendar, Tone, Ui};
65///
66/// let today = Date::new(2026, 11, 15).unwrap();
67/// let (month, day) = (State::new(today.first_of_month()), State::new(Some(today)));
68/// let calendar = EventCalendar::new(month.binding(), day.binding())
69///     .today(today)
70///     .events([CalendarEvent::new(1, today, "Aluguel").tone(Tone::Warning)]);
71/// let mut ui = Ui::new(calendar, ravenrs_style::Theme::raven_light());
72/// ui.update();
73/// ```
74pub struct EventCalendar {
75    month: Binding<Date>,
76    selected: Binding<Option<Date>>,
77    events: Vec<CalendarEvent>,
78    today: Option<Date>,
79    per_day: usize,
80    on_event: Option<Rc<dyn Fn(u64)>>,
81}
82
83impl std::fmt::Debug for EventCalendar {
84    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
85        f.debug_struct("EventCalendar").field("events", &self.events.len()).finish_non_exhaustive()
86    }
87}

89/// The height of a day with room for its events, in logical pixels.
90const DAY_HEIGHT: f32 = 104.0;
91
92impl EventCalendar {
93    /// `month` holds the month shown (any day of it), so the app can load its
94    /// events; `selected` holds the day picked.
95    pub fn new(month: Binding<Date>, selected: Binding<Option<Date>>) -> Self {
96        Self { month, selected, events: Vec::new(), today: None, per_day: 3, on_event: None }
97    }
98
99    /// The events; those outside the days shown are ignored.
100    pub fn events(mut self, events: impl IntoIterator<Item = CalendarEvent>) -> Self {
101        self.events = events.into_iter().collect();
102        self
103    }
104
105    /// Marks this day, and offers a button to come back to it.
106    pub fn today(mut self, today: Date) -> Self {
107        self.today = Some(today);
108        self
109    }
110
111    /// How many events show in a day before the rest become a count (3 by default).
112    pub fn events_per_day(mut self, count: usize) -> Self {
113        self.per_day = count.max(1);
114        self
115    }
116
117    /// Called with an event's `id` when it is clicked, after its day is picked.
118    pub fn on_event(mut self, handler: impl Fn(u64) + 'static) -> Self {
119        self.on_event = Some(Rc::new(handler));
120        self
121    }
122}

124struct EventChip<'a> {
125    event: &'a CalendarEvent,
126    selected: &'a Binding<Option<Date>>,
127    on_event: Option<Rc<dyn Fn(u64)>>,
128}

130impl View for EventChip<'_> {
131    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
132        let theme = *cx.theme();
133        let event = self.event;
134        let hovered = cx.is_hovered();
135        let color = if event.done { theme.color(Semantic::InkMuted) } else { event.tone.color(&theme) };
136        let fill = color.with_alpha(match (event.done, hovered) {
137            (true, false) => 0.08,
138            (true, true) | (false, false) => 0.16,
139            (false, true) => 0.26,
140        });
141        let mut children = Vec::new();
142        if event.done {
143            children.push(util::icon(cx, IconName::Check, 12.0, color));
144        }
145        let ink = if event.done { color } else { theme.color(Semantic::Ink) };
146        let title =
147            cx.child(&Text::new(event.title.clone()).style(TextStyle::Caption).foreground(ink).truncate());
148        children.push(cx.container(
149            LayoutStyle {
150                grow: 1.0,
151                min_width: Some(0.0),
152                overflow_hidden: true,
153                ..LayoutStyle::stack(Axis::Horizontal)
154            },
155            &[title],
156            Paint::None,
157        ));
158        let node = cx.container(
159            LayoutStyle {
160                padding: Insets::symmetric(2.0, 6.0),
161                spacing: 4.0,
162                align: Align::Center,
163                min_width: Some(0.0),
164                overflow_hidden: true,
165                ..LayoutStyle::stack(Axis::Horizontal)
166            },
167            &children,
168            Paint::Quad(QuadPaint { corner_radius: theme.radii.sm, ..QuadPaint::new(fill) }),
169        );
170        cx.semantics(node, Semantics::new(Role::Button).name(event.title.clone()));
171        let (selected, date, id, on_event) =
172            (self.selected.clone(), event.date, event.id, self.on_event.clone());
173        cx.on_click(
174            node,
175            Rc::new(move || {
176                selected.set(Some(date));
177                if let Some(handler) = &on_event {
178                    handler(id);
179                }
180            }),
181        );
182        cx.cursor(node, CursorIcon::Pointer);
183        node
184    }
185}

187struct CalendarDay<'a> {
188    date: Date,
189    in_month: bool,
190    active: bool,
191    events: Vec<&'a CalendarEvent>,
192    scheduler: &'a EventCalendar,
193    keys: &'a GridKeys,
194}
195
196impl View for CalendarDay<'_> {
197    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
198        let theme = *cx.theme();
199        let s = self.scheduler;
200        let labels = cx.labels();
201        let (selected, is_today) = (s.selected.get() == Some(self.date), s.today == Some(self.date));
202        let hovered = cx.is_hovered() || cx.is_focus_visible();
203        let ink = theme.color(Semantic::Ink);
204        let number_color = if is_today {
205            theme.color(Semantic::OnAccent)
206        } else if self.in_month {
207            ink
208        } else {
209            theme.color(Semantic::InkMuted)
210        };
211        let weight = is_today.then_some(Weight::SEMIBOLD);
212        let number = util::label(cx, &self.date.day().to_string(), TextStyle::Caption, weight, number_color);
213        cx.semantics(number, Semantics::hidden());
214        let number = cx.container(
215            LayoutStyle {
216                width: Some(22.0),
217                height: Some(22.0),
218                align: Align::Center,
219                justify: Justify::Center,
220                align_self: Some(Align::Fit),
221                ..LayoutStyle::stack(Axis::Horizontal)
222            },
223            &[number],
224            if is_today {
225                Paint::Quad(QuadPaint {
226                    corner_radius: 11.0,
227                    ..QuadPaint::new(theme.color(Semantic::Accent))
228                })
229            } else {
230                Paint::None
231            },
232        );
233        let mut children = vec![number];
234        for event in self.events.iter().take(s.per_day) {
235            let chip = EventChip { event, selected: &s.selected, on_event: s.on_event.clone() };
236            children.push(cx.keyed_child(event.id, &chip));
237        }
238        let hidden = self.events.len().saturating_sub(s.per_day);
239        if hidden > 0 {
240            let more = format!("+{hidden} {}", labels.calendar_more);
241            let line = util::label(cx, &more, TextStyle::Caption, None, theme.color(Semantic::InkMuted));
242            cx.semantics(line, Semantics::hidden());
243            children.push(line);
244        }
245        let fill = if hovered {
246            ink.with_alpha(0.05)
247        } else if self.in_month {
248            Color::TRANSPARENT
249        } else {
250            ink.with_alpha(0.02)
251        };
252        let border = if selected {
253            Border { width: 1.5, color: theme.color(Semantic::Accent) }
254        } else {
255            Border { width: 1.0, color: theme.color(Semantic::Border) }
256        };
257        let node = cx.container(
258            LayoutStyle {
259                min_height: Some(DAY_HEIGHT),
260                min_width: Some(0.0),
261                padding: Insets::uniform(4.0),
262                spacing: 3.0,
263                align: Align::Stretch,
264                overflow_hidden: true,
265                ..LayoutStyle::stack(Axis::Vertical)
266            },
267            &children,
268            Paint::Quad(QuadPaint {
269                corner_radius: theme.radii.sm,
270                border: Some(border),
271                ..QuadPaint::new(fill)
272            }),
273        );
274        let mut semantics = Semantics::new(Role::Button).name(labels.long_date(self.date)).selected(selected);
275        let mut notes: Vec<String> = Vec::new();
276        if is_today {
277            notes.push(labels.today.clone());
278        }
279        notes.extend(self.events.iter().map(|e| e.title.clone()));
280        if !notes.is_empty() {
281            semantics = semantics.description(notes.join(", "));
282        }
283        cx.semantics(node, semantics);
284        if self.keys.focus_to.get() == Some(self.date) {
285            self.keys.focus_to.set(None);
286            cx.request_focus();
287        }
288        if self.active {
289            cx.focusable(node, theme.radii.sm);
290            cx.on_key(node, self.keys.handler(self.date));
291        }
292        let (pick, date) = (s.selected.clone(), self.date);
293        cx.on_click(node, Rc::new(move || pick.set(Some(date))));
294        cx.cursor(node, CursorIcon::Pointer);
295        node
296    }
297}

299impl View for EventCalendar {
300    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
301        let theme = *cx.theme();
302        let labels = cx.labels();
303        let shown = self.month.get().first_of_month();
304        let (prev, next) = (self.month.clone(), self.month.clone());
305        let mut header = vec![
306            cx.child(
307                &IconButton::new(IconName::ChevronLeft)
308                    .label(labels.previous_month.clone())
309                    .size(ButtonSize::Small)
310                    .on_click(move || prev.set(shown.add_months(-1))),
311            ),
312            cx.child(
313                &IconButton::new(IconName::ChevronRight)
314                    .label(labels.next_month.clone())
315                    .size(ButtonSize::Small)
316                    .on_click(move || next.set(shown.add_months(1))),
317            ),
318            util::label(
319                cx,
320                &labels.month_title(shown),
321                TextStyle::Headline,
322                None,
323                theme.color(Semantic::Ink),
324            ),
325        ];
326        if let Some(today) = self.today {
327            let (month, selected) = (self.month.clone(), self.selected.clone());
328            header.push(
329                cx.child(
330                    &Button::new(labels.today.clone())
331                        .style(ButtonStyle::Ghost)
332                        .size(ButtonSize::Small)
333                        .on_click(move || {
334                            month.set(today.first_of_month());
335                            selected.set(Some(today));
336                        }),
337                ),
338            );
339        }
340        let header = cx.container(
341            LayoutStyle { spacing: 6.0, align: Align::Center, ..LayoutStyle::stack(Axis::Horizontal) },
342            &header,
343            Paint::None,
344        );
345        let mut cells: Vec<NodeId> = labels
346            .weekdays
347            .iter()
348            .map(|d| {
349                let name = util::label(cx, d, TextStyle::Caption, None, theme.color(Semantic::InkMuted));
350                // Each day's name already says the date.
351                cx.semantics(name, Semantics::hidden());
352                name
353            })
354            .collect();
355        let keys = GridKeys {
356            month: self.month.clone(),
357            cursor: cx.use_state(|| None),
358            focus_to: cx.use_state(|| Rc::new(Cell::new(None))).get(),
359        };
360        let in_shown = |d: &Date| d.first_of_month() == shown;
361        let active = [keys.cursor.get(), self.selected.get(), self.today]
362            .into_iter()
363            .flatten()
364            .find(in_shown)
365            .unwrap_or(shown);
366        // As many weeks as the month touches, so a short month leaves no empty row.
367        let start = shown.add_days(-i32::from(shown.weekday()));
368        let last = shown.add_months(1).add_days(-1);
369        let days = i32::try_from(last.to_days() - start.to_days()).unwrap_or(34) / 7 * 7 + 7;
370        for i in 0..days {
371            let date = start.add_days(i);
372            let events = self.events.iter().filter(|e| e.date == date).collect();
373            let day = CalendarDay {
374                date,
375                in_month: in_shown(&date),
376                active: date == active,
377                events,
378                scheduler: self,
379                keys: &keys,
380            };
381            cells.push(cx.keyed_child(date.to_days() as u64, &day));
382        }
383        let grid = cx.container(
384            LayoutStyle {
385                grid_columns: Some(7),
386                spacing: 4.0,
387                align_self: Some(Align::Stretch),
388                ..LayoutStyle::stack(Axis::Vertical)
389            },
390            &cells,
391            Paint::None,
392        );
393        cx.semantics(grid, Semantics::new(Role::Calendar).name(labels.month_title(shown)));
394        cx.container(
395            LayoutStyle { spacing: 10.0, align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
396            &[header, grid],
397            Paint::None,
398        )
399    }
400}

402/// Edits a [`RecurrenceRule`]: how often, every how many, on which weekdays and
403/// until when, and says in words what it chose. `start` is the first day, which
404/// the rule counts from.
405///
406/// ```
407/// use ravenrs_reactive::State;
408/// use ravenrs_widgets::{Date, RecurrenceEditor, RecurrenceRule, Ui};
409///
410/// let rule = State::new(RecurrenceRule::default());
411/// let start = Date::new(2026, 11, 5).unwrap();
412/// let mut ui = Ui::new(RecurrenceEditor::new(rule.binding(), start), ravenrs_style::Theme::raven_light());
413/// ui.update();
414/// ```
415#[derive(Debug)]
416pub struct RecurrenceEditor {
417    rule: Binding<RecurrenceRule>,
418    start: Date,
419}
420
421impl RecurrenceEditor {
422    /// An editor of `rule`, for something first happening on `start`.
423    pub fn new(rule: Binding<RecurrenceRule>, start: Date) -> Self {
424        Self { rule, start }
425    }
426}
427
428/// A binding to one part of the rule.
429fn part<T: 'static>(
430    rule: &Binding<RecurrenceRule>,
431    read: impl Fn(&RecurrenceRule) -> T + 'static,
432    write: impl Fn(&mut RecurrenceRule, T) + 'static,
433) -> Binding<T> {
434    let (from, to) = (rule.clone(), rule.clone());
435    Binding::new(move || read(&from.get()), move |value| to.update(|r| write(r, value)))
436}
437
438impl View for RecurrenceEditor {
439    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
440        let labels = cx.labels();
441        let rule = self.rule.get();
442        let repeats = rule.frequency != Frequency::Once;
443        let frequency = part(
444            &self.rule,
445            |r| Frequency::ALL.iter().position(|f| *f == r.frequency).unwrap_or(0),
446            |r, i: usize| r.frequency = Frequency::ALL[i.min(Frequency::ALL.len() - 1)],
447        );
448        let mut rows: Vec<crate::AnyView> = vec![Box::new(
449            SegmentedControl::new(frequency, labels.recurrence_frequencies.clone())
450                .label(labels.recurrence_name.clone()),
451        )];
452        if repeats {
453            let interval = part(
454                &self.rule,
455                |r| i32::try_from(r.interval.max(1)).unwrap_or(1),
456                |r, n: i32| r.interval = u32::try_from(n.max(1)).unwrap_or(1),
457            );
458            rows.push(Box::new(
459                HStack::new((
460                    Text::new(labels.recurrence_every.clone()).style(TextStyle::Callout),
461                    Stepper::new(interval).range(1, 99).label(labels.recurrence_every.clone()),
462                    Text::new(labels.recurrence_unit(rule.frequency, rule.interval).to_owned())
463                        .style(TextStyle::Callout),
464                ))
465                .spacing(8.0)
466                .align(Align::Center),
467            ));
468        }
469        if rule.frequency == Frequency::Weekly {
470            let own = usize::from(self.start.weekday());
471            let days: Vec<crate::AnyView> = (0..7)
472                .map(|day| {
473                    let chosen = part(
474                        &self.rule,
475                        // With none chosen the rule keeps the first day's weekday, so that one shows on.
476                        move |r| if r.weekdays.contains(&true) { r.weekdays[day] } else { day == own },
477                        move |r, on: bool| {
478                            if !r.weekdays.contains(&true) {
479                                r.weekdays[own] = true;
480                            }
481                            r.weekdays[day] = on;
482                        },
483                    );
484                    Box::new(ToggleButton::new(chosen, labels.weekdays[day].clone())) as crate::AnyView
485                })
486                .collect();
487            rows.push(Box::new(HStack::new(days).spacing(4.0)));
488        }
489        if repeats {
490            let start = self.start;
491            let kind = part(
492                &self.rule,
493                |r| match r.end {
494                    RecurrenceEnd::Until(_) => 1,
495                    RecurrenceEnd::After(_) => 2,
496                    _ => 0,
497                },
498                move |r, i: usize| {
499                    r.end = match i {
500                        1 => RecurrenceEnd::Until(start.add_months(12)),
501                        2 => RecurrenceEnd::After(12),
502                        _ => RecurrenceEnd::Never,
503                    };
504                },
505            );
506            let ends = SegmentedControl::new(kind, labels.recurrence_ends.clone())
507                .label(labels.recurrence_ends_name.clone());
508            let extra: Option<crate::AnyView> = match rule.end {
509                RecurrenceEnd::Until(day) => {
510                    let last = part(
511                        &self.rule,
512                        |r| if let RecurrenceEnd::Until(day) = r.end { Some(day) } else { None },
513                        |r, day: Option<Date>| {
514                            if let Some(day) = day {
515                                r.end = RecurrenceEnd::Until(day);
516                            }
517                        },
518                    );
519                    Some(Box::new(
520                        DatePicker::new(last, day).label(labels.recurrence_ends_name.clone()).width(190.0),
521                    ))
522                }
523                RecurrenceEnd::After(_) => {
524                    let count = part(
525                        &self.rule,
526                        |r| {
527                            if let RecurrenceEnd::After(n) = r.end {
528                                i32::try_from(n).unwrap_or(1)
529                            } else {
530                                1
531                            }
532                        },
533                        |r, n: i32| r.end = RecurrenceEnd::After(u32::try_from(n.max(1)).unwrap_or(1)),
534                    );
535                    Some(Box::new(
536                        HStack::new((
537                            Stepper::new(count).range(1, 999).label(labels.recurrence_times.clone()),
538                            Text::new(labels.recurrence_times.clone()).style(TextStyle::Callout),
539                        ))
540                        .spacing(8.0)
541                        .align(Align::Center),
542                    ))
543                }
544                _ => None,
545            };
546            rows.push(Box::new(FormField::new(
547                labels.recurrence_ends_name.clone(),
548                VStack::new((ends, extra)).spacing(8.0),
549            )));
550        }
551        let summary = labels.recurrence_summary(&rule, self.start);
552        rows.push(Box::new(Text::new(summary).style(TextStyle::Callout).foreground(Semantic::InkMuted)));
553        cx.child(&VStack::new(rows).spacing(10.0).align(Align::Stretch))
554    }
