148/// A timeline of resources: a row for each person, room or account, a time axis
149/// across the top, and events as bars that can be dragged along the time and
150/// between rows, resized by their ends, and drawn on an empty stretch. Weekends
151/// are shaded, and only what is in view gets built, so large plans stay smooth.
152///
153/// The app holds the data: `on_change` reports where an event would go, and the
154/// app stores it and passes the events again. `can_drop` can refuse a place.
155///
156/// From the keyboard, Tab reaches the events. Left and Right move one by a step,
157/// Up and Down to the next row, Shift with Left and Right resize it, Enter opens
158/// it and Delete removes it.
159///
160/// ```
161/// use ravenrs_reactive::State;
162/// use ravenrs_widgets::{Date, Resource, Scheduler, SchedulerEvent, TimeAxis, TimePoint, Ui};
163///
164/// let first = Date::new(2026, 11, 1).unwrap();
165/// let axis = State::new(TimeAxis::days(first, 30));
166/// let start = TimePoint::from_date(first.add_days(4));
167/// let scheduler = Scheduler::new(axis.binding())
168///     .resources([Resource::new(1, "Ana"), Resource::new(2, "Bruno")])
169///     .events([SchedulerEvent::new(7, 1, start, start.plus_days(3), "Férias")])
170///     .on_change(|change| println!("{change:?}"));
171/// let mut ui = Ui::new(scheduler, ravenrs_style::Theme::raven_light());
172/// ui.update();
173/// ```
174pub struct Scheduler {
175    axis: Binding<TimeAxis>,
176    resources: Vec<Resource>,
177    events: Vec<SchedulerEvent>,
178    ranges: Vec<SchedulerRange>,
179    now: Option<TimePoint>,
180    selected: Option<Binding<Option<u64>>>,
181    resource_width: f32,
182    read_only: bool,
183    toolbar: bool,
184    on_change: Option<OnChange>,
185    on_create: Option<OnChange>,
186    on_open: Option<Rc<dyn Fn(u64)>>,
187    on_delete: Option<Rc<dyn Fn(u64)>>,
188    can_drop: Option<CanDrop>,
189}

322/// What a drag is doing to an event.
323#[derive(Clone, Copy, Debug, PartialEq)]
324enum DragKind {
325    /// Along with the pointer; `grab` is how far into the event it was taken.
326    Move {
327        grab: i64,
328    },
329    ResizeStart,
330    ResizeEnd,

333/// Where the event under a drag would land.
334#[derive(Clone, Copy, Debug, PartialEq)]
335struct Preview {
336    change: SchedulerChange,
337    allowed: bool,
338}

340/// What a drag or a key needs, shared by every bar and by the background.
341#[derive(Clone)]
342struct Editing {
343    axis: TimeAxis,
344    rows: Rc<Vec<(u64, f32, f32)>>,
345    preview: State<Option<Preview>>,
346    on_change: Option<OnChange>,
347    on_create: Option<OnChange>,
348    can_drop: Option<CanDrop>,
349}
350
351impl Editing {
352    fn resource_at(&self, y: f32) -> Option<u64> {
353        let found = self.rows.iter().find(|(_, top, height)| y >= *top && y < top + height);
354        found.or_else(|| if y < 0.0 { self.rows.first() } else { self.rows.last() }).map(|r| r.0)
355    }
356
357    fn allowed(&self, change: &SchedulerChange) -> bool {
358        self.can_drop.as_ref().is_none_or(|rule| rule(change))
359    }
360
361    fn propose(&self, change: SchedulerChange) {
362        self.preview.set(Some(Preview { change, allowed: self.allowed(&change) }));
363    }
364
365    /// Applies a finished change if it is allowed and really is one.
366    fn commit(&self, original: Option<SchedulerChange>) {
367        let Some(preview) = *self.preview.peek() else { return };
368        self.preview.set(None);
369        if !preview.allowed || original == Some(preview.change) {
370            return;
371        }
372        let handler = if preview.change.id.is_some() { &self.on_change } else { &self.on_create };
373        if let Some(handler) = handler {
374            handler(preview.change);
375        }
376    }
377
378    /// Follows a drag on an event or on one of its ends.
379    fn follow(&self, event: &SchedulerEvent, kind: DragKind, origin: &Cell<Point>, drag: Drag, bar: Point) {
380        if drag.phase == DragPhase::Start {
381            origin.set(Point::new(drag.container.x() - bar.x, drag.container.y() - bar.y));
382        }
383        let at = Point::new(drag.position.x - origin.get().x, drag.position.y - origin.get().y);
384        let time = self.axis.snap(self.axis.time_at(at.x));
385        let original = SchedulerChange {
386            id: Some(event.id),
387            resource: event.resource,
388            start: event.start,
389            end: event.end,
390        };
391        let step = self.axis.snap_minutes();
392        let change = match kind {
393            DragKind::Move { grab } => {
394                let start = self.axis.snap(self.axis.time_at(at.x).plus_minutes(-grab));
395                let length = event.end.minutes() - event.start.minutes();
396                let resource = self.resource_at(at.y).unwrap_or(event.resource);
397                SchedulerChange { start, end: start.plus_minutes(length), resource, ..original }
398            }
399            DragKind::ResizeStart => {
400                SchedulerChange { start: time.min(event.end.plus_minutes(-step)), ..original }
401            }
402            DragKind::ResizeEnd => {
403                SchedulerChange { end: time.max(event.start.plus_minutes(step)), ..original }
404            }
405        };
406        match drag.phase {
407            DragPhase::End => {
408                self.propose(change);
409                self.commit(Some(original));
410            }
411            _ => self.propose(change),
412        }
413    }
414}

416struct EndHandle {
417    event: SchedulerEvent,
418    kind: DragKind,
419    editing: Editing,
420    bar: Point,
421    width: f32,
422}
423
424impl View for EndHandle {
425    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
426        let at_start = self.kind == DragKind::ResizeStart;
427        let x = if at_start { 0.0 } else { (self.width - HANDLE_WIDTH).max(0.0) };
428        let node = cx.container(
429            LayoutStyle {
430                absolute: Some((x, 0.0)),
431                width: Some(HANDLE_WIDTH),
432                height: Some(LANE_HEIGHT - 4.0),
433                ..LayoutStyle::leaf()
434            },
435            &[],
436            Paint::None,
437        );
438        let origin = cx.use_state(|| Rc::new(Cell::new(Point::ZERO))).get();
439        let (event, kind, editing) = (self.event.clone(), self.kind, self.editing.clone());
440        let bar = self.bar;
441        cx.on_drag(
442            node,
443            // A handle's parent is the bar, so the timeline's origin comes from where the bar is.
444            Rc::new(move |drag: Drag| editing.follow(&event, kind, &origin, drag, bar)),
445        );
446        cx.cursor(node, CursorIcon::ColResize);
447        node
448    }
449}

451struct Bar<'a> {
452    event: &'a SchedulerEvent,
453    /// Where it is drawn, in the timeline's content.
454    frame: Rect,
455    title: String,
456    /// It is the one a drag is moving, and this says if that place is allowed.
457    dragging: Option<bool>,
458    selected: bool,
459    /// The event keyboard users reach with Tab.
460    tab_stop: bool,
461    editable: bool,
462    editing: Editing,
463    /// Row by row, for Up and Down.
464    selection: Option<Binding<Option<u64>>>,
465    on_open: Option<Rc<dyn Fn(u64)>>,
466    on_delete: Option<Rc<dyn Fn(u64)>>,
467}
468
469impl Bar<'_> {
470    fn color(&self, theme: &Theme) -> ravenrs_core::Color {
471        match self.dragging {
472            Some(false) => theme.color(Semantic::Danger),
473            _ if self.event.done => theme.color(Semantic::InkMuted),
474            _ => self.event.tone.color(theme),
475        }
476    }
477
478    fn key_handler(&self) -> crate::KeyHandler {
479        let event = self.event.clone();
480        let (editing, editable) = (self.editing.clone(), self.editable && !event.locked);
481        let (on_open, on_delete) = (self.on_open.clone(), self.on_delete.clone());
482        Rc::new(move |key, modifiers| {
483            let step = editing.axis.snap_minutes();
484            let original = SchedulerChange {
485                id: Some(event.id),
486                resource: event.resource,
487                start: event.start,
488                end: event.end,
489            };
490            let change = match key {
491                Key::Enter => {
492                    if let Some(open) = &on_open {
493                        open(event.id);
494                    }
495                    return on_open.is_some();
496                }
497                Key::Delete | Key::Backspace if editable => {
498                    if let Some(delete) = &on_delete {
499                        delete(event.id);
500                    }
501                    return on_delete.is_some();
502                }
503                Key::ArrowLeft | Key::ArrowRight if editable => {
504                    let by = if *key == Key::ArrowLeft { -step } else { step };
505                    if modifiers.shift && !event.resizable {
506                        return true;
507                    }
508                    if modifiers.shift {
509                        let end = event.end.plus_minutes(by);
510                        SchedulerChange { end: end.max(event.start.plus_minutes(step)), ..original }
511                    } else {
512                        SchedulerChange {
513                            start: event.start.plus_minutes(by),
514                            end: event.end.plus_minutes(by),
515                            ..original
516                        }
517                    }
518                }
519                Key::ArrowUp | Key::ArrowDown if editable => {
520                    let here = editing.rows.iter().position(|r| r.0 == event.resource);
521                    let there =
522                        here.and_then(|i| if *key == Key::ArrowUp { i.checked_sub(1) } else { Some(i + 1) });
523                    let Some(row) = there.and_then(|i| editing.rows.get(i)) else { return true };
524                    SchedulerChange { resource: row.0, ..original }
525                }
526                _ => return false,
527            };
528            editing.propose(change);
529            editing.commit(Some(original));
530            true
531        })
532    }
533}

535impl View for Bar<'_> {
536    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
537        let theme = *cx.theme();
538        let e = self.event;
539        let color = self.color(&theme);
540        let hovered = cx.is_hovered() || cx.is_focus_visible();
541        let solid = !e.done && self.dragging.is_none();
542        let fill = if self.selected || hovered && solid {
543            color
544        } else {
545            color.with_alpha(if e.done { 0.16 } else { 0.8 })
546        };
547        let ink = if e.done { theme.color(Semantic::Ink) } else { theme.color(Semantic::OnAccent) };
548        let mut children = Vec::new();
549        if e.done {
550            children.push(util::icon(cx, IconName::Check, 12.0, color));
551        }
552        let title =
553            cx.child(&Text::new(self.title.clone()).style(TextStyle::Caption).foreground(ink).truncate());
554        children.push(cx.container(
555            LayoutStyle {
556                grow: 1.0,
557                min_width: Some(0.0),
558                overflow_hidden: true,
559                ..LayoutStyle::stack(Axis::Horizontal)
560            },
561            &[title],
562            Paint::None,
563        ));
564        let bar_point = Point::new(self.frame.x(), self.frame.y());
565        if self.editable && !e.locked && e.resizable {
566            for kind in [DragKind::ResizeStart, DragKind::ResizeEnd] {
567                let handle = EndHandle {
568                    event: e.clone(),
569                    kind,
570                    editing: self.editing.clone(),
571                    bar: bar_point,
572                    width: self.frame.width(),
573                };
574                children.push(cx.child(&handle));
575            }
576        }
577        let border = (self.selected || self.dragging.is_some())
578            .then(|| Border { width: 1.5, color: theme.color(Semantic::Ink).with_alpha(0.55) });
579        let node = cx.container(
580            LayoutStyle {
581                absolute: Some((self.frame.x(), self.frame.y())),
582                width: Some(self.frame.width()),
583                height: Some(self.frame.height()),
584                padding: Insets::symmetric(0.0, 8.0),
585                spacing: 4.0,
586                align: Align::Center,
587                overflow_hidden: true,
588                ..LayoutStyle::stack(Axis::Horizontal)
589            },
590            &children,
591            Paint::Quad(QuadPaint { corner_radius: theme.radii.sm, border, ..QuadPaint::new(fill) }),
592        );
593        let labels = cx.labels();
594        let range = time_range(&labels, self.editing.axis.scale, e.start, e.end);
595        cx.semantics(
596            node,
597            Semantics::new(Role::Button).name(e.title.clone()).description(range).selected(self.selected),
598        );
599        if self.tab_stop {
600            cx.focusable(node, theme.radii.sm);
601            cx.on_key(node, self.key_handler());
602        }
603        let (selection, id) = (self.selection.clone(), e.id);
604        cx.on_click(
605            node,
606            Rc::new(move || {
607                if let Some(selection) = &selection {
608                    selection.set(Some(id));
609                }
610            }),
611        );
612        if let Some(open) = self.on_open.clone() {
613            cx.on_double_click(node, Rc::new(move || open(id)));
614        }
615        if self.editable && !e.locked {
616            let origin = cx.use_state(|| Rc::new(Cell::new(Point::ZERO))).get();
617            let grab = cx.use_state(|| Rc::new(Cell::new(0_i64))).get();
618            let (event, editing) = (e.clone(), self.editing.clone());
619            cx.on_drag(
620                node,
621                Rc::new(move |drag: Drag| {
622                    if drag.phase == DragPhase::Start {
623                        // The pointer stays over the same part of the event while it moves.
624                        let inside = drag.position.x - drag.frame.x();
625                        let minutes = (f64::from(inside)
626                            / f64::from(editing.axis.x_of(editing.axis.start.plus_days(1)).max(0.001))
627                            * 1440.0) as i64;
628                        grab.set(minutes);
629                    }
630                    let kind = DragKind::Move { grab: grab.get() };
631                    let content_origin = Point::ZERO;
632                    editing.follow(&event, kind, &origin, drag, content_origin);
633                }),
634            );
635        }
636        cx.cursor(node, CursorIcon::Pointer);
637        node
638    }
639}

641/// The empty part of the timeline: draws a new event, and holds the bars.

645    height: f32,

649impl Lanes {
650    fn build(&self, cx: &mut BuildCx<'_>, s: &Scheduler, editing: &Editing) -> NodeId {
651        let node = cx.container(
652            LayoutStyle {
653                width: Some(self.width),
654                height: Some(self.height),
655                ..LayoutStyle::stack(Axis::Vertical)
656            },
657            &self.children,
658            Paint::None,
659        );
660        if !s.read_only && s.on_create.is_some() {
661            let from: Rc<Cell<Option<(TimePoint, u64)>>> = cx.use_state(|| Rc::new(Cell::new(None))).get();
662            let editor = editing.clone();
663            cx.on_drag(
664                node,
665                Rc::new(move |drag: Drag| {
666                    let at = Point::new(drag.position.x - drag.frame.x(), drag.position.y - drag.frame.y());
667                    let time = editor.axis.snap(editor.axis.time_at(at.x));
668                    if drag.phase == DragPhase::Start
669                        && let Some(resource) = editor.resource_at(at.y)
670                    {
671                        from.set(Some((time, resource)));
672                    }
673                    let Some((begin, resource)) = from.get() else { return };
674                    let step = editor.axis.snap_minutes();
675                    let (start, end) = if time >= begin {
676                        (begin, time.max(begin.plus_minutes(step)))
677                    } else {
678                        (time, begin)
679                    };
680                    editor.propose(SchedulerChange { id: None, resource, start, end });
681                    if drag.phase == DragPhase::End {
682                        from.set(None);
683                        editor.commit(None);
684                    }
685                }),
686            );
687        }
688        node
689    }
690}

692/// A moment as a person reads it on this scale.
693fn time_label(labels: &Labels, scale: TimeScale, time: TimePoint) -> String {
694    let date = (labels.short_date)(time.date());
695    match scale {
696        TimeScale::Hours => {
697            let (hour, minute) = time.hour_minute();
698            format!("{date} {hour:02}:{minute:02}")
699        }
700        _ => date,
701    }
702}
703
704/// A stretch of time as a person reads it. Days are whole, so the last one shown is the last one in it.
705fn time_range(labels: &Labels, scale: TimeScale, start: TimePoint, end: TimePoint) -> String {
706    let last = if scale == TimeScale::Hours { end } else { end.plus_minutes(-1).max(start) };
707    let (from, to) = (time_label(labels, scale, start), time_label(labels, scale, last));
708    if from == to { from } else { format!("{from} – {to}") }
709}
710
711fn tick_text(labels: &Labels, scale: TimeScale, time: TimePoint) -> String {
712    let date = time.date();
713    match scale {
714        TimeScale::Hours => format!("{:02}", time.hour_minute().0),
715        TimeScale::Days => date.day().to_string(),
716        TimeScale::Weeks => format!("{:02}", date.day()),
717        TimeScale::Months => {
718            let name = &labels.months[usize::from(date.month().clamp(1, 12)) - 1];
719            name.chars().take(3).collect()
720        }
721    }
722}
723
724fn group_text(labels: &Labels, scale: TimeScale, time: TimePoint) -> String {
725    let date = time.date();
726    match scale {
727        TimeScale::Hours => (labels.short_date)(date),
728        TimeScale::Days | TimeScale::Weeks => labels.month_title(date),
729        TimeScale::Months => date.year().to_string(),
730    }
731}

773impl Scheduler {
774    /// Groups stay together, in the order their first resource appears.
775    fn rows(&self, lanes_of: &HashMap<u64, usize>, collapsed: &BTreeSet<String>) -> (Vec<Row>, f32) {
776        let mut order: Vec<Option<&str>> = Vec::new();
777        for resource in &self.resources {
778            let group = resource.group.as_deref();
779            if !order.contains(&group) {
780                order.push(group);
781            }
782        }
783        let mut rows = Vec::with_capacity(self.resources.len() + order.len());
784        let mut y = 0.0;
785        for group in order {
786            let members: Vec<usize> =
787                (0..self.resources.len()).filter(|i| self.resources[*i].group.as_deref() == group).collect();
788            let mut hidden = false;
789            if let Some(name) = group {
790                hidden = collapsed.contains(name);
791                rows.push(Row::Group { name: name.to_owned(), count: members.len(), collapsed: hidden, y });
792                y += GROUP_HEIGHT;
793            }
794            if hidden {
795                continue;
796            }
797            for index in members {
798                let lanes = lanes_of.get(&self.resources[index].id).copied().unwrap_or(1);
799                let height = lanes as f32 * LANE_HEIGHT + 2.0 * ROW_PADDING;
800                rows.push(Row::Resource { index, y, height });
801                y += height;
802            }
803        }
804        (rows, y)
805    }

878        let collapsed: State<BTreeSet<String>> = cx.use_state(BTreeSet::new);
879        let own_selection: State<Option<u64>> = cx.use_state(|| None);
880        let selection = self.selected.clone().unwrap_or_else(|| own_selection.binding());
881        let selected = selection.get();
882        let current = preview.get();
883
884        // Where each event is drawn, the one being dragged at its new place.
885        let placed: Vec<(usize, SchedulerChange)> = self
886            .events
887            .iter()
888            .enumerate()
889            .map(|(i, e)| {
890                let here =
891                    SchedulerChange { id: Some(e.id), resource: e.resource, start: e.start, end: e.end };

894            .collect();

897            by_resource.entry(change.resource).or_default().push(slot);
898        }
899        let mut lane_of = vec![0_usize; placed.len()];
900        let mut lanes_of: HashMap<u64, usize> = HashMap::new();
901        for (resource, slots) in &by_resource {
902            let spans: Vec<(TimePoint, TimePoint)> =
903                slots.iter().map(|s| (placed[*s].1.start, placed[*s].1.end)).collect();
904            let (lanes, count) = assign_lanes(&spans);
905            for (slot, lane) in slots.iter().zip(lanes) {

907            }
908            lanes_of.insert(*resource, count);
909        }
910        let (rows, total_height) = self.rows(&lanes_of, &collapsed.get());

1198                width: Some(self.resource_width),
1199                height: Some(total_height),
1200                ..LayoutStyle::stack(Axis::Vertical)
1201            },
1202            &names,
1203            Paint::None,
1204        );

1208                width: Some(self.resource_width),
1209                min_height: Some(0.0),
