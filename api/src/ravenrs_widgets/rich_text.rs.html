14/// How a piece of a line is marked.
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
16struct Marks {
17    bold: bool,
18    italic: bool,
19    strike: bool,
20    code: bool,
21}
22
23impl Marks {
24    fn with(self, mark: char) -> Self {
25        match mark {
26            '*' => Self { bold: true, ..self },
27            '_' => Self { italic: true, ..self },
28            '~' => Self { strike: true, ..self },
29            _ => Self { code: true, ..self },
30        }
31    }
32}

34/// A run of text with one look.
35#[derive(Clone, Debug, PartialEq, Eq)]
36struct Span {
37    text: String,
38    marks: Marks,
39    link: bool,
40}

42/// What a line starts with.
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44enum LineKind {
45    Plain,
46    Item,
47    Quote,
48}

50/// What a text is made of, from top to bottom.
51#[derive(Clone, Debug, PartialEq, Eq)]
52enum Block {
53    Line(LineKind, Vec<Span>),
54    Code(String),
55}
56
57impl Block {
58    fn is_plain(&self) -> bool {
59        match self {
60            Self::Line(LineKind::Plain, spans) => {
61                spans.iter().all(|span| span.marks == Marks::default() && !span.link)
62            }
63            _ => false,
64        }
65    }
66}
67
68const FENCE: &str = "```";
69
70fn parse(text: &str, marks: bool) -> Vec<Block> {
71    let mut blocks = Vec::new();
72    let mut rest = text;
73    if marks {
74        // Fences come first: they cross lines, and nothing inside them is a mark.
75        while let Some(open) = rest.find(FENCE) {
76            let inside = &rest[open + FENCE.len()..];
77            let Some(close) = inside.find(FENCE) else { break };
78            let code = inside[..close].trim_matches('\n');
79            if code.trim().is_empty() {
80                // Six backticks around nothing are six backticks.
81                let end = open + FENCE.len() * 2 + close;
82                parse_lines(&rest[..end], marks, &mut blocks);
83                rest = &rest[end..];
84                continue;
85            }
86            parse_lines(rest[..open].strip_suffix('\n').unwrap_or(&rest[..open]), marks, &mut blocks);
87            blocks.push(Block::Code(code.to_owned()));
88            let after = &inside[close + FENCE.len()..];
89            rest = after.strip_prefix('\n').unwrap_or(after);
90            if rest.is_empty() {
91                return blocks;
92            }
93        }
94    }
95    parse_lines(rest, marks, &mut blocks);
96    blocks
97}
98
99fn parse_lines(text: &str, marks: bool, blocks: &mut Vec<Block>) {
100    if text.is_empty() {
101        return;
102    }
103    for line in text.split('\n') {
104        let filled = |body: &&str| !body.trim().is_empty();
105        let (kind, body) = if !marks {
106            (LineKind::Plain, line)
107        } else if let Some(body) = line.strip_prefix("* ").or_else(|| line.strip_prefix("- ")).filter(filled)
108        {
109            (LineKind::Item, body)
110        } else if let Some(body) = line.strip_prefix("> ").filter(filled) {
111            (LineKind::Quote, body)
112        } else {
113            (LineKind::Plain, line)
114        };
115        let mut spans = Vec::new();
116        parse_inline(body, Marks::default(), marks, &mut spans);
117        blocks.push(Block::Line(kind, spans));
118    }
119}

121/// The length of the link starting `text`, if one does. Punctuation that ends a sentence is
122/// left out, or "see https://x/y." would lead to a page that does not exist. Only web
123/// addresses count, so that a message can never hand the app another kind of address.
124fn link_length(text: &str) -> Option<usize> {
125    let scheme = ["https://", "http://"].into_iter().find(|scheme| text.starts_with(scheme))?;
126    let end =
127        text.find(|c: char| c.is_whitespace() || matches!(c, '<' | '>' | '"' | '\'')).unwrap_or(text.len());
128    let link = text[..end].trim_end_matches(['.', ',', ';', ':', '!', '?', ')', ']', '}']);
129    (link.len() > scheme.len()).then_some(link.len())
130}

132/// A mark opens after the start, a space or punctuation, right against what it marks.
133fn opens(text: &str, at: usize, mark: char) -> bool {
134    let before = text[..at].chars().next_back();
135    let after = text[at + mark.len_utf8()..].chars().next();
136    before.is_none_or(|c| !c.is_alphanumeric() && c != mark)
137        && after.is_some_and(|c| !c.is_whitespace() && c != mark)
138}
139
140/// Where the mark opened at `at` closes: against the text, before the end, a space or
141/// punctuation. A code mark closes at the next backtick, whatever is around it.
142fn closes(text: &str, at: usize, mark: char) -> Option<usize> {
143    let start = at + mark.len_utf8();
144    text[start..].match_indices(mark).map(|(i, _)| start + i).find(|&i| {
145        if i == start {
146            return false;
147        }
148        if mark == '`' {
149            return true;
150        }
151        let before = text[..i].chars().next_back();
152        let after = text[i + mark.len_utf8()..].chars().next();
153        before.is_some_and(|c| !c.is_whitespace()) && after.is_none_or(|c| !c.is_alphanumeric())
154    })
155}

157fn push_span(spans: &mut Vec<Span>, text: &str, marks: Marks, link: bool) {
158    if !text.is_empty() {
159        spans.push(Span { text: text.to_owned(), marks, link });
160    }
161}
162
163fn parse_inline(text: &str, base: Marks, marks: bool, spans: &mut Vec<Span>) {
164    let (mut pending, mut at) = (0, 0);
165    while let Some(c) = text[at..].chars().next() {
166        // The link first and whole: a mark inside an address is not a mark.
167        if c == 'h'
168            && let Some(length) = link_length(&text[at..])
169        {
170            push_span(spans, &text[pending..at], base, false);
171            push_span(spans, &text[at..at + length], base, true);
172            at += length;
173            pending = at;
174            continue;
175        }
176        if marks
177            && matches!(c, '*' | '_' | '~' | '`')
178            && opens(text, at, c)
179            && let Some(close) = closes(text, at, c)
180        {
181            push_span(spans, &text[pending..at], base, false);
182            let inside = &text[at + 1..close];
183            if c == '`' {
184                push_span(spans, inside, base.with(c), false);
185            } else {
186                parse_inline(inside, base.with(c), true, spans);
187            }
188            at = close + 1;
189            pending = at;
190            continue;
191        }
192        at += c.len_utf8();
193    }
194    push_span(spans, &text[pending..], base, false);
195}

197/// The text a reader takes in, marks gone and one line per line.
198fn read_aloud(blocks: &[Block]) -> String {
199    let lines: Vec<String> = blocks
200        .iter()
201        .map(|block| match block {
202            Block::Line(_, spans) => spans.iter().map(|span| span.text.as_str()).collect(),
203            Block::Code(code) => code.clone(),
204        })
205        .collect();
206    lines.join("\n")
207}

209/// Text with a little formatting, written the way chat apps write it: `*bold*`, `_italic_`,
210/// `~struck~`, `` `code` ``, a fenced block of ```` ``` ````, lines that start with `* `, `- `
211/// or `> `, and web addresses, which become links.
212///
213/// It wraps to the width it is offered, between words, and a marked piece stays with the
214/// punctuation around it. Text with nothing to format is drawn as one [`Text`], for the same
215/// cost. Links take the keyboard: Tab reaches them, and Enter or Space opens them. It only
216/// recognizes `http` and `https`, and it hands the address to [`Self::on_link`] instead of
217/// opening anything.
218///
219/// `ravenrs-text` shapes one style per text and embeds neither an italic nor a monospace
220/// font, so italics are told apart by a muted color and code by a chip behind it.
221///
222/// ```
223/// use ravenrs_widgets::{RichText, Ui};
224///
225/// let text = RichText::new("*Ready:* see `report.pdf` at https://example.com/r")
226///     .on_link(|address| println!("open {address}"));
227/// let mut ui = Ui::new(text, ravenrs_style::Theme::raven_dark());
228/// ui.update();
229/// ```
230#[derive(Clone)]
231pub struct RichText {
232    text: String,
233    marks: bool,
234    style: TextStyle,
235    foreground: Fill,
236    on_link: Option<Rc<dyn Fn(String)>>,
237}

297/// One piece of a word: words change look in the middle, as in `(*this*),`.
298struct Piece {
299    text: String,
300    marks: Marks,
301    link: bool,
302    /// The strike goes on over the space after it.
303    strike_on: bool,
304}
305
306/// Cuts a line at the places it can wrap. A chip of code or a link stays in one piece, unless
307/// it is too wide for a whole line, when its own text wraps.
308fn words(spans: &[Span]) -> Vec<Vec<Piece>> {
309    let mut words: Vec<Vec<Piece>> = Vec::new();
310    let mut glued = false;
311    for span in spans {
312        let piece = |text: &str, strike_on| Piece {
313            text: text.to_owned(),
314            marks: span.marks,
315            link: span.link,
316            strike_on,
317        };
318        if span.link || span.marks.code {
319            if glued && let Some(word) = words.last_mut() {
320                word.push(piece(&span.text, false));
321            } else {
322                words.push(vec![piece(&span.text, false)]);
323            }
324            glued = true;
325            continue;
326        }
327        let parts: Vec<&str> = span.text.split_whitespace().collect();
328        if parts.is_empty() {
329            glued = false;
330            continue;
331        }
332        let spaced_before = span.text.starts_with(char::is_whitespace);
333        for (i, part) in parts.iter().enumerate() {
334            let strike_on = span.marks.strike && i + 1 < parts.len();
335            if i == 0
336                && glued
337                && !spaced_before
338                && let Some(word) = words.last_mut()
339            {
340                word.push(piece(part, strike_on));
341            } else {
342                words.push(vec![piece(part, strike_on)]);
343            }
344        }
345        glued = !span.text.ends_with(char::is_whitespace);
346    }
347    words
348}

350/// The line drawn over or under a word, behind its glyphs.
351fn rule(frame: Rect, y: f32, color: Color, scene: &mut Scene) {
352    scene.push_quad(Quad::new(Rect::new(frame.x(), frame.y() + y, frame.width(), 1.0), color));
353}

355/// A link inside a line.
356struct LinkPiece {
357    text: String,
358    attrs: TextAttrs,
359    on_link: Option<Rc<dyn Fn(String)>>,
360}
361
362impl View for LinkPiece {
363    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
364        let theme = *cx.theme();
365        let active = self.on_link.is_some();
366        let role = if !active {
367            Semantic::Accent
368        } else if cx.is_pressed() {
369            Semantic::AccentPressed
370        } else if cx.is_hovered() {
371            Semantic::AccentHover
372        } else {
373            Semantic::Accent
374        };
375        // The accent is made to fill things. Drawn thin on any surface it needs some ink,
376        // which takes it the right way in either kind of theme, to stay readable.
377        let color = theme.color(role).lerp(theme.color(Semantic::Ink), 0.3);
378        let text = cx.text(&self.text, &self.attrs, color, LayoutStyle::leaf());
379        cx.semantics(text, Semantics::hidden());
380        let (words, attrs) = (self.text.clone(), self.attrs.clone());
381        let node = cx.container(
382            LayoutStyle { min_width: Some(0.0), ..LayoutStyle::stack(Axis::Horizontal) },
383            &[text],
384            // Under each row, as wide as the row: an address too long for a line wraps by itself.
385            Paint::Rich(Rc::new(move |frame, paint| {
386                let mut layout = paint.text.layout(&words, &attrs);
387                layout.measure(Some(frame.width() + 0.5));
388                for line in layout.lines() {
389                    let row = Rect::new(frame.x(), frame.y() + line.top, line.width, line.height);
390                    rule(row, line.height - 3.0, color, paint.scene);
391                }
392            })),
393        );
394        if let Some(on_link) = &self.on_link {
395            let (on_link, address) = (on_link.clone(), self.text.clone());
396            cx.on_click(node, Rc::new(move || on_link(address.clone())));
397            cx.focusable(node, theme.radii.sm);
398            cx.cursor(node, CursorIcon::Pointer);
399            cx.semantics(node, Semantics::new(Role::Link).name(self.text.clone()));
400        }
401        node
402    }
403}

405impl RichText {
406    /// One word, made of one or more pieces, with the space that follows it.
407    fn word(
408        &self,
409        cx: &mut BuildCx<'_>,
410        pieces: &[Piece],
411        ink: Color,
412        space: f32,
413        next_link: &mut u64,
414    ) -> NodeId {
415        let theme = *cx.theme();
416        let base = crate::util::attrs(&theme, self.style, None);
417        let mut children = Vec::with_capacity(pieces.len());
418        let mut trailing = space;
419        for piece in pieces {
420            let mut attrs = base.clone();
421            if piece.marks.bold {
422                attrs.weight = Weight::SEMIBOLD.0;
423            }
424            if piece.link {
425                let key = *next_link;
426                *next_link += 1;
427                children.push(cx.keyed_child(
428                    key,
429                    &LinkPiece { text: piece.text.clone(), attrs, on_link: self.on_link.clone() },
430                ));
431                continue;
432            }
433            let color = if piece.marks.italic { theme.color(Semantic::InkMuted) } else { ink };
434            let text = cx.text(&piece.text, &attrs, color, LayoutStyle::leaf());
435            cx.semantics(text, Semantics::hidden());
436            let node = if piece.marks.code {
437                cx.container(
438                    LayoutStyle {
439                        padding: Insets::symmetric(0.0, theme.spacing.s1),
440                        min_width: Some(0.0),
441                        ..LayoutStyle::stack(Axis::Horizontal)
442                    },
443                    &[text],
444                    Paint::Quad(QuadPaint {
445                        corner_radius: theme.radii.sm,
446                        ..QuadPaint::new(theme.color(Semantic::Border))
447                    }),
448                )
449            } else if piece.marks.strike {
450                let middle = base.line_height / 2.0;
451                let reach = if piece.strike_on { space } else { 0.0 };
452                if piece.strike_on {
453                    trailing = 0.0;
454                }
455                cx.container(
456                    LayoutStyle {
457                        padding: Insets { right: reach, ..Insets::ZERO },
458                        min_width: Some(0.0),
459                        ..LayoutStyle::stack(Axis::Horizontal)
460                    },
461                    &[text],
462                    Paint::Custom(Rc::new(move |frame, scene| rule(frame, middle, color, scene))),
463                )
464            } else {
465                text
466            };
467            children.push(node);
468        }
469        cx.container(
470            LayoutStyle {
471                padding: Insets { right: trailing, ..Insets::ZERO },
472                min_width: Some(0.0),
473                ..LayoutStyle::stack(Axis::Horizontal)
474            },
475            &children,
476            Paint::None,
477        )
478    }
479
480    /// The words of a line, wrapping in whatever width it gets.
481    fn flow(&self, cx: &mut BuildCx<'_>, spans: &[Span], ink: Color, next_link: &mut u64) -> NodeId {
482        let attrs = crate::util::attrs(cx.theme(), self.style, None);
483        // Measured between letters, since a space at the end of a text may not count.
484        let one = cx.measure_text("n", &attrs).width;
485        let space = cx.measure_text("n n", &attrs).width - one * 2.0;
486        let words = words(spans);
487        let children: Vec<NodeId> =
488            words.iter().map(|pieces| self.word(cx, pieces, ink, space, next_link)).collect();
489        cx.container(
490            LayoutStyle {
491                wrap: true,
492                grow: 1.0,
493                min_width: Some(0.0),
494                // An empty line still takes its height, as it does in a text.
495                min_height: Some(attrs.line_height),
496                ..LayoutStyle::stack(Axis::Horizontal)
497            },
498            &children,
499            Paint::None,
500        )
501    }
502}

504impl View for RichText {
505    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
506        let blocks = parse(&self.text, self.marks);
507        if blocks.iter().all(Block::is_plain) {
508            return cx.child(&Text::new(self.text.clone()).style(self.style).foreground(self.foreground));
509        }
510        let theme = *cx.theme();
511        let ink = self.foreground.resolve(&theme);
512        let attrs = crate::util::attrs(&theme, self.style, None);
513        let mut next_link = 0;
514        let mut children = Vec::with_capacity(blocks.len());
515        for block in &blocks {
516            let node = match block {
517                Block::Line(LineKind::Plain, spans) => self.flow(cx, spans, ink, &mut next_link),
518                Block::Line(LineKind::Item, spans) => {
519                    let bullet = cx.text("•", &attrs, ink, LayoutStyle::leaf());
520                    cx.semantics(bullet, Semantics::hidden());
521                    let line = self.flow(cx, spans, ink, &mut next_link);
522                    cx.container(
523                        LayoutStyle {
524                            spacing: theme.spacing.s2,
525                            padding: Insets { left: theme.spacing.s1, ..Insets::ZERO },
526                            ..LayoutStyle::stack(Axis::Horizontal)
527                        },
528                        &[bullet, line],
529                        Paint::None,
530                    )
531                }
532                Block::Line(LineKind::Quote, spans) => {
533                    let bar = cx.container(
534                        LayoutStyle {
535                            width: Some(2.0),
536                            align_self: Some(Align::Stretch),
537                            ..LayoutStyle::leaf()
538                        },
539                        &[],
540                        Paint::Quad(QuadPaint {
541                            corner_radius: 1.0,
542                            ..QuadPaint::new(theme.color(Semantic::Border))
543                        }),
544                    );
545                    let line = self.flow(cx, spans, theme.color(Semantic::InkMuted), &mut next_link);
546                    cx.container(
547                        LayoutStyle { spacing: theme.spacing.s2, ..LayoutStyle::stack(Axis::Horizontal) },
548                        &[bar, line],
549                        Paint::None,
550                    )
551                }
552                Block::Code(code) => {
553                    let text = cx.text(code, &attrs, ink, LayoutStyle::leaf());
554                    cx.semantics(text, Semantics::hidden());
555                    cx.container(
556                        LayoutStyle {
557                            padding: Insets::symmetric(theme.spacing.s2, theme.spacing.s3),
558                            align_self: Some(Align::Stretch),
559                            ..LayoutStyle::stack(Axis::Vertical)
560                        },
561                        &[text],
562                        Paint::Quad(QuadPaint {
563                            corner_radius: theme.radii.md,
564                            ..QuadPaint::new(theme.color(Semantic::Border))
565                        }),
566                    )
567                }
568            };
569            children.push(node);
570        }
571        let node = cx.container(
572            LayoutStyle { align: Align::Stretch, min_width: Some(0.0), ..LayoutStyle::stack(Axis::Vertical) },
573            &children,
574            Paint::None,
575        );
576        cx.semantics(node, Semantics::new(Role::Label).name(read_aloud(&blocks)));
577        node
578    }
579}
