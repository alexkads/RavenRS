28/// A folder or file of a [`FileBrowser`]. Nothing here touches the file system: it says what the
29/// app knows, wherever that came from.
30#[derive(Clone, Debug, PartialEq)]
31#[non_exhaustive]
32pub struct FileEntry {
33    /// What the selection keeps. The name by default.
34    pub id: String,
35    /// What it is called.
36    pub name: String,
37    /// It opens into more entries.
38    pub is_folder: bool,
39    /// How many bytes it has, if that is known.
40    pub size: Option<u64>,
41    /// When it changed, as it should read.
42    pub modified: Option<String>,
43    /// What sorts the dates, as a number that grows with time.
44    pub modified_order: i64,
45    /// An icon of its own, instead of the one for a folder or a file.
46    pub icon: Option<IconName>,
47}
48
49impl FileEntry {
50    fn new(name: String, is_folder: bool, size: Option<u64>) -> Self {
51        Self { id: name.clone(), name, is_folder, size, modified: None, modified_order: 0, icon: None }
52    }
53
54    /// A folder called `name`.
55    pub fn folder(name: impl Into<String>) -> Self {
56        Self::new(name.into(), true, None)
57    }
58
59    /// A file called `name`, of `size` bytes.
60    pub fn file(name: impl Into<String>, size: u64) -> Self {
61        Self::new(name.into(), false, Some(size))
62    }
63
64    /// What identifies it in the selection, when names can repeat.
65    pub fn id(mut self, id: impl Into<String>) -> Self {
66        self.id = id.into();
67        self
68    }
69
70    /// When it changed: how it reads, and a number that grows with time to sort it by.
71    pub fn modified(mut self, text: impl Into<String>, order: i64) -> Self {
72        self.modified = Some(text.into());
73        self.modified_order = order;
74        self
75    }
76
77    /// An icon of its own.
78    pub fn icon(mut self, icon: IconName) -> Self {
79        self.icon = Some(icon);
80        self
81    }
82}

84/// What a [`FileBrowser`] sorts by.
85#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
86#[non_exhaustive]
87pub enum FileSortColumn {
88    /// By name, ignoring case.
89    #[default]
90    Name,
91    /// By size.
92    Size,
93    /// By date.
94    Modified,
95}

97/// The order of a [`FileBrowser`]. Folders always come first.
98#[derive(Clone, Copy, Debug, PartialEq, Eq)]
99pub struct FileSort {
100    /// What it sorts by.
101    pub column: FileSortColumn,
102    /// From the least to the most.
103    pub ascending: bool,
104}
105
106impl Default for FileSort {
107    fn default() -> Self {
108        Self { column: FileSortColumn::Name, ascending: true }
109    }
110}

112/// The positions of `entries` in this order, folders first.
113fn sorted(entries: &[FileEntry], sort: FileSort) -> Vec<usize> {
114    let names: Vec<String> = entries.iter().map(|entry| entry.name.to_lowercase()).collect();
115    let mut order: Vec<usize> = (0..entries.len()).collect();
116    order.sort_by(|&a, &b| {
117        let (x, y) = (&entries[a], &entries[b]);
118        let by = match sort.column {
119            FileSortColumn::Name => names[a].cmp(&names[b]),
120            FileSortColumn::Size => x.size.cmp(&y.size),
121            FileSortColumn::Modified => x.modified_order.cmp(&y.modified_order),
122        };
123        let by = if sort.ascending { by } else { by.reverse() };
124        y.is_folder.cmp(&x.is_folder).then(by).then_with(|| names[a].cmp(&names[b]))
125    });
126    order
127}

129/// An explorer of folders and files: a breadcrumb that goes back, a header that sorts, and a
130/// list of any length with icon, name, size and date. The entries are data from the app, which
131/// answers [`on_navigate`](Self::on_navigate) and [`on_open`](Self::on_open) with another
132/// folder; it never touches the file system, so it serves a backup on a server as well as a disk.
133/// A click selects alone, Ctrl or Cmd adds, and Shift takes a range. The arrows, Home and End
134/// move, Space adds, Enter opens, and Backspace goes up.
135///
136/// ```
137/// use ravenrs_widgets::{FileBrowser, FileEntry, Ui};
138///
139/// let browser = FileBrowser::new(
140///     ["Backups", "2026"],
141///     vec![FileEntry::folder("October"), FileEntry::file("notes.txt", 2048)],
142/// )
143/// .on_open(|entry| println!("{}", entry.name))
144/// .on_navigate(|depth| println!("{depth}"));
145/// let mut ui = Ui::new(browser, ravenrs_style::Theme::raven_dark());
146/// ui.update();
147/// ```
148pub struct FileBrowser {
149    path: Vec<String>,
150    entries: Rc<[FileEntry]>,
151    selection: Option<Binding<BTreeSet<String>>>,
152    sort: Option<Binding<FileSort>>,
153    mode: SelectionMode,
154    on_navigate: Option<Rc<dyn Fn(usize)>>,
155    on_open: Option<OpenHandler>,
156    height: Option<f32>,
157    label: Option<String>,
158}
159
160impl fmt::Debug for FileBrowser {
161    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
162        f.debug_struct("FileBrowser").field("path", &self.path).finish_non_exhaustive()
163    }
164}

166impl FileBrowser {
167    /// The folder at `path`, from the root, with `entries` inside it. Keep the entries in an
168    /// `Rc` between builds when they are many: they are sorted again only when they are not the
169    /// same ones.
170    pub fn new(
171        path: impl IntoIterator<Item = impl Into<String>>,
172        entries: impl Into<Rc<[FileEntry]>>,
173    ) -> Self {
174        Self {
175            path: path.into_iter().map(Into::into).collect(),
176            entries: entries.into(),
177            selection: None,
178            sort: None,
179            mode: SelectionMode::Multiple,
180            on_navigate: None,
181            on_open: None,
182            height: None,
183            label: None,
184        }
185    }
186
187    /// The ids of the selected entries, to read and to change from outside.
188    pub fn selection(mut self, selection: Binding<BTreeSet<String>>) -> Self {
189        self.selection = Some(selection);
190        self
191    }
192
193    /// The order, to keep or to set from outside.
194    pub fn sort(mut self, sort: Binding<FileSort>) -> Self {
195        self.sort = Some(sort);
196        self
197    }
198
199    /// How entries can be selected ([`SelectionMode::Multiple`] by default).
200    pub fn mode(mut self, mode: SelectionMode) -> Self {
201        self.mode = mode;
202        self
203    }
204
205    /// Called with the depth of the place to go back to, 0 being the root.
206    pub fn on_navigate(mut self, handler: impl Fn(usize) + 'static) -> Self {
207        self.on_navigate = Some(Rc::new(handler));
208        self
209    }
210
211    /// Called with an entry on a double click or on Enter.
212    pub fn on_open(mut self, handler: impl Fn(&FileEntry) + 'static) -> Self {
213        self.on_open = Some(Rc::new(handler));
214        self
215    }
216
217    /// A fixed height. It takes the height its parent gives it by default.
218    pub fn height(mut self, height: f32) -> Self {
219        self.height = Some(height);
220        self
221    }
222
223    /// What screen readers call it, instead of the name in [`Labels`](crate::Labels).
224    pub fn label(mut self, label: impl Into<String>) -> Self {
225        self.label = Some(label.into());
226        self
227    }

230/// A column title that sorts.
231struct SortHeader {
232    text: String,
233    column: FileSortColumn,
234    sort: Binding<FileSort>,
235    width: Option<f32>,
236    end: bool,
237}

239impl View for SortHeader {
240    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
241        let theme = *cx.theme();
242        let now = self.sort.get();
243        let on = now.column == self.column;
244        let lit = cx.is_hovered() || cx.is_focus_visible();
245        let color = theme.color(if on || lit { Semantic::Ink } else { Semantic::InkMuted });
246        let mut parts = vec![util::label(cx, &self.text, TextStyle::Caption, Some(Weight::SEMIBOLD), color)];
247        if on {
248            let icon = if now.ascending { IconName::ArrowUp } else { IconName::ArrowDown };
249            parts.push(util::icon(cx, icon, 12.0, color));
250        }
251        let node = cx.container(
252            LayoutStyle {
253                width: self.width,
254                grow: if self.width.is_none() { 1.0 } else { 0.0 },
255                min_width: Some(0.0),
256                spacing: theme.spacing.s1,
257                align: Align::Center,
258                justify: if self.end { Justify::End } else { Justify::Start },
259                ..LayoutStyle::stack(Axis::Horizontal)
260            },
261            &parts,
262            Paint::None,
263        );
264        let (sort, column) = (self.sort.clone(), self.column);
265        cx.on_click(
266            node,
267            Rc::new(move || {
268                sort.update(|s| {
269                    s.ascending = if s.column == column { !s.ascending } else { true };
270                    s.column = column;
271                });
272            }),
273        );
274        cx.semantics(node, Semantics::new(Role::ColumnHeader).name(self.text.clone()).selected(on));
275        cx.focusable(node, theme.radii.sm);
276        cx.cursor(node, CursorIcon::Pointer);
277        node
278    }
279}

281/// A line of the list.
282struct FileRow {
283    entry_index: usize,
284    position: usize,
285    entries: Rc<[FileEntry]>,
286    selected: bool,
287    active: bool,
288    keyboard: bool,
289    dates: bool,
290    pick: Rc<dyn Fn(usize)>,
291    open: Option<OpenHandler>,
292}
293
294impl View for FileRow {
295    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
296        let theme = *cx.theme();
297        let labels = cx.labels();
298        let Some(entry) = self.entries.get(self.entry_index) else { return cx.empty() };
299        let (ink, muted) = (theme.color(Semantic::Ink), theme.color(Semantic::InkMuted));
300        let icon = entry.icon.unwrap_or(if entry.is_folder { IconName::Folder } else { IconName::File });
301        let tint = if entry.is_folder { theme.color(Semantic::Accent) } else { muted };
302        let mut parts = vec![util::icon(cx, icon, 16.0, tint)];
303        let attrs = util::attrs(&theme, TextStyle::Callout, None);
304        parts.push(cx.text_truncated(
305            &entry.name,
306            &attrs,
307            ink,
308            LayoutStyle { grow: 1.0, min_width: Some(0.0), ..LayoutStyle::leaf() },
309        ));
310        let size = entry.size.map(size_text).unwrap_or_default();
311        let column = |width: f32, end: bool| LayoutStyle {
312            width: Some(width),
313            align: Align::Center,
314            justify: if end { Justify::End } else { Justify::Start },
315            ..LayoutStyle::stack(Axis::Horizontal)
316        };
317        let number = util::label(cx, &size, TextStyle::Caption, None, muted);
318        parts.push(cx.container(column(SIZE_WIDTH, true), &[number], Paint::None));
319        if self.dates {
320            let date = entry.modified.as_deref().unwrap_or("");
321            let date = util::label(cx, date, TextStyle::Caption, None, muted);
322            parts.push(cx.container(column(MODIFIED_WIDTH, true), &[date], Paint::None));
323        }
324        let accent = theme.color(Semantic::Accent);
325        let fill = if self.selected {
326            accent.with_alpha(0.16)
327        } else if cx.is_hovered() {
328            ink.with_alpha(0.05)
329        } else {
330            Color::TRANSPARENT
331        };
332        let border = (self.active && self.keyboard).then_some(Border { width: 1.5, color: accent });
333        let node = cx.container(
334            LayoutStyle {
335                grow: 1.0,
336                spacing: theme.spacing.s2,
337                padding: Insets::symmetric(0.0, theme.spacing.s3),
338                align: Align::Center,
339                ..LayoutStyle::stack(Axis::Horizontal)
340            },
341            &parts,
342            Paint::Quad(QuadPaint { corner_radius: theme.radii.sm, border, ..QuadPaint::new(fill) }),
343        );
344        let (pick, position) = (self.pick.clone(), self.position);
345        cx.on_click(node, Rc::new(move || pick(position)));
346        if let Some(open) = self.open.clone() {
347            let (entries, index) = (self.entries.clone(), self.entry_index);
348            cx.on_double_click(
349                node,
350                Rc::new(move || {
351                    if let Some(entry) = entries.get(index) {
352                        open(entry);
353                    }
354                }),
355            );
356        }
357        cx.cursor(node, CursorIcon::Pointer);
358        let what = if entry.is_folder { labels.folder_word.clone() } else { size };
359        cx.semantics(
360            node,
361            Semantics::new(Role::ListBoxOption)
362                .name(entry.name.clone())
363                .description(what)
364                .selected(self.selected)
365                .active(self.active),
366        );
367        node
368    }
369}

371impl View for FileBrowser {
372    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
373        let theme = *cx.theme();
374        let labels = cx.labels();
375        let local_selection: State<BTreeSet<String>> = cx.use_state(BTreeSet::new);
376        let selection = self.selection.clone().unwrap_or_else(|| local_selection.binding());
377        let local_sort: State<FileSort> = cx.use_state(FileSort::default);
378        let sort = self.sort.clone().unwrap_or_else(|| local_sort.binding());
379        let commands: State<Option<ScrollCommand>> = cx.use_state(|| None);
380        let cursor: State<usize> = cx.use_state(|| 0);
381        let anchor: State<usize> = cx.use_state(|| 0);
382        type Sorted = (Rc<[FileEntry]>, FileSort, Rc<Vec<usize>>);
383        let cache: Rc<RefCell<Option<Sorted>>> = cx.use_state(|| Rc::new(RefCell::new(None))).peek().clone();
384        let order = sort.get();
385        let sorted_now = {
386            let mut cached = cache.borrow_mut();
387            match cached.as_ref() {
388                Some((entries, by, found)) if Rc::ptr_eq(entries, &self.entries) && *by == order => {
389                    found.clone()
390                }
391                _ => {
392                    let found = Rc::new(sorted(&self.entries, order));
393                    *cached = Some((self.entries.clone(), order, found.clone()));
394                    found
395                }
396            }
397        };
398
399        let count = sorted_now.len();
400        let active = cursor.get().min(count.saturating_sub(1));
401        let selected = Rc::new(selection.get());
402        let keyboard = cx.is_focus_visible();
403        let mode = self.mode;
404        let dates = self.entries.iter().any(|entry| entry.modified.is_some());
405        let id_of = {
406            let (entries, order) = (self.entries.clone(), sorted_now.clone());
407            move |position: usize| {
408                order.get(position).and_then(|i| entries.get(*i)).map_or_else(String::new, |e| e.id.clone())
409            }
410        };
411        let modifiers = cx.modifiers();
412        let apply: Rc<dyn Fn(usize, Modifiers)> = {
413            let (selection, cursor, anchor, id_of) =
414                (selection.clone(), cursor.clone(), anchor.clone(), id_of.clone());
415            Rc::new(move |position, held| {
416                let mut next = anchor.get();
417                selection.update(|set| {
418                    next = selection::pick(set, mode, next, position, held, &id_of);
419                });
420                anchor.set(next);
421                cursor.set(position);
422            })
423        };
424        let pick: Rc<dyn Fn(usize)> = {
425            let apply = apply.clone();
426            let modifiers = modifiers.clone();
427            Rc::new(move |position| apply(position, modifiers.get()))
428        };
429
430        let mut crumbs = Breadcrumb::new(self.path.iter().cloned()).root_icon(IconName::House);
431        if let Some(go) = self.on_navigate.clone() {
432            crumbs = crumbs.on_navigate(move |depth| go(depth));
433        }
434        let crumbs = cx.child(&crumbs);
435        let crumbs = cx.container(
436            LayoutStyle {
437                padding: Insets::symmetric(theme.spacing.s2, theme.spacing.s3),
438                align: Align::Start,
439                ..LayoutStyle::stack(Axis::Vertical)
440            },
441            &[crumbs],
442            Paint::None,
443        );
444        let first_line = cx.child(&Divider::horizontal());
445        let mut titles = vec![cx.child(&SortHeader {
446            text: labels.file_name.clone(),
447            column: FileSortColumn::Name,
448            sort: sort.clone(),
449            width: None,
450            end: false,
451        })];
452        titles.push(cx.child(&SortHeader {
453            text: labels.file_size.clone(),
454            column: FileSortColumn::Size,
455            sort: sort.clone(),
456            width: Some(SIZE_WIDTH),
457            end: true,
458        }));
459        if dates {
460            titles.push(cx.child(&SortHeader {
461                text: labels.file_modified.clone(),
462                column: FileSortColumn::Modified,
463                sort: sort.clone(),
464                width: Some(MODIFIED_WIDTH),
465                end: true,
466            }));
467        }
468        let header = cx.container(
469            LayoutStyle {
470                spacing: theme.spacing.s2,
471                // The icon column of the rows, and their padding, line up the titles.
472                padding: Insets::symmetric(theme.spacing.s1, theme.spacing.s3),
473                align: Align::Center,
474                ..LayoutStyle::stack(Axis::Horizontal)
475            },
476            &titles,
477            Paint::Quad(QuadPaint::new(theme.color(Semantic::Surface))),
478        );
479        let second_line = cx.child(&Divider::horizontal());
480
481        let body = if count == 0 {
482            let words = util::label(
483                cx,
484                &labels.folder_empty,
485                TextStyle::Callout,
486                None,
487                theme.color(Semantic::InkMuted),
488            );
489            cx.container(
490                LayoutStyle {
491                    grow: 1.0,
492                    min_height: Some(0.0),
493                    padding: Insets::uniform(theme.spacing.s6),
494                    align: Align::Center,
495                    justify: Justify::Center,
496                    ..LayoutStyle::stack(Axis::Vertical)
497                },
498                &[words],
499                Paint::None,
500            )
501        } else {
502            let (entries, order_now, open, shown, picker) = (
503                self.entries.clone(),
504                sorted_now.clone(),
505                self.on_open.clone(),
506                selected.clone(),
507                pick.clone(),
508            );
509            let id = id_of.clone();
510            let mut list = VirtualList::new(count, move |position| FileRow {
511                entry_index: order_now.get(position).copied().unwrap_or(0),
512                position,
513                entries: entries.clone(),
514                selected: shown.contains(&id(position)),
515                active: position == active,
516                keyboard,
517                dates,
518                pick: picker.clone(),
519                open: open.clone(),
520            })
521            .row_height(ROW_HEIGHT)
522            .bare()
523            .commands(commands.clone());
524            let (keyed_entries, keyed_order) = (self.entries.clone(), sorted_now.clone());
525            list = list.key(move |position| {
526                keyed_order
527                    .get(position)
528                    .and_then(|i| keyed_entries.get(*i))
529                    .map_or(position as u64, |entry| util::hash_text(&entry.id))
530            });
531            cx.keyed_child(1, &list)
532        };
533        let root = cx.container(
534            LayoutStyle {
535                height: self.height,
536                grow: if self.height.is_some() { 0.0 } else { 1.0 },
537                min_height: Some(0.0),
538                min_width: Some(0.0),
539                overflow_hidden: true,
540                align: Align::Stretch,
541                align_self: Some(Align::Stretch),
542                ..LayoutStyle::stack(Axis::Vertical)
543            },
544            &[crumbs, first_line, header, second_line, body],
545            Paint::None,
546        );
547        let name = self.label.clone().unwrap_or_else(|| labels.file_browser_name.clone());
548        cx.semantics(root, Semantics::new(Role::ListBox).name(name).description(count.to_string()));
549        cx.focusable(root, theme.radii.sm);
550        let (entries, order_now, open, up, depth) = (
551            self.entries.clone(),
552            sorted_now,
553            self.on_open.clone(),
554            self.on_navigate.clone(),
555            self.path.len(),
556        );
557        let select_all = selection;
558        cx.on_key(
559            root,
560            Rc::new(move |key, held| {
561                let to = match key {
562                    Key::ArrowUp => active.saturating_sub(1),
563                    Key::ArrowDown => (active + 1).min(count.saturating_sub(1)),
564                    Key::Home => 0,
565                    Key::End => count.saturating_sub(1),
566                    Key::Enter => {
567                        if let (Some(open), Some(entry)) =
568                            (&open, order_now.get(active).and_then(|i| entries.get(*i)))
569                        {
570                            open(entry);
571                        }
572                        return true;
573                    }
574                    Key::Space => {
575                        if count > 0 {
576                            apply(active, Modifiers { ctrl: true, ..Modifiers::default() });
577                        }
578                        return true;
579                    }
580                    Key::Backspace => {
581                        if let (Some(up), true) = (&up, depth >= 2) {
582                            up(depth - 2);
583                        }
584                        return true;
585                    }
586                    Key::Character(c) if selection::adds(held) && c.eq_ignore_ascii_case("a") => {
587                        if mode != SelectionMode::Multiple {
588                            return false;
589                        }
590                        select_all.set((0..count).map(&id_of).collect());
591                        return true;
592                    }
593                    _ => return false,
594                };
595                if count == 0 {
596                    return false;
597                }
598                if selection::adds(held) {
599                    cursor.set(to);
600                } else {
601                    apply(to, held);
602                }
603                commands.set(Some(ScrollCommand::Reveal(to)));
604                true
605            }),
606        );
607        root
608    }
609}

615    #[test]
616    fn folders_stay_first_in_either_direction() {
617        let entries = [
618            FileEntry::file("b.txt", 10),
619            FileEntry::folder("Z"),
620            FileEntry::file("A.txt", 30),
621            FileEntry::folder("c"),
622        ];
623        let by = |column, ascending| sorted(&entries, FileSort { column, ascending });
624        assert_eq!(by(FileSortColumn::Name, true), vec![3, 1, 2, 0]);
625        assert_eq!(by(FileSortColumn::Name, false), vec![1, 3, 0, 2]);
626        assert_eq!(by(FileSortColumn::Size, false), vec![3, 1, 2, 0]);
627    }
