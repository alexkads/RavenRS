20/// How long an arrow at the end of the strip is.
21const ARROW_LENGTH: f32 = 32.0;
22/// How much of the view an arrow moves.
23const ARROW_STEP: f32 = 0.8;
24
25/// What a cell of a [`Filmstrip`] is told when it is drawn.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27#[non_exhaustive]
28pub struct FilmstripCell {
29    /// Its position in the strip.
30    pub index: usize,
31    /// It is in the selection.
32    pub selected: bool,
33    /// The keyboard is on it.
34    pub active: bool,
35}
36
37/// A horizontal strip of thumbnails that builds only what shows, so thousands cost the same as
38/// a few. Each cell comes from whoever uses it, with its own image, stars or badges; the strip
39/// draws the frame around it. A click selects alone, Ctrl or Cmd adds, and Shift takes a range;
40/// the arrows, Home and End move and keep the selected cell in view, Space adds, Enter
41/// activates. Arrows appear on the ends that have more to show.
42///
43/// ```
44/// use std::collections::BTreeSet;
45///
46/// use ravenrs_reactive::State;
47/// use ravenrs_widgets::{Filmstrip, FilmstripCell, SelectionMode, Text, Ui};
48///
49/// let selected = State::new(BTreeSet::from([0_usize]));
50/// let strip = Filmstrip::new(5_000, |cell: FilmstripCell| Text::new(format!("{}", cell.index + 1)))
51///     .selection(selected.binding())
52///     .mode(SelectionMode::Multiple);
53/// let mut ui = Ui::new(strip, ravenrs_style::Theme::raven_dark());
54/// ui.update();
55/// ```
56pub struct Filmstrip {
57    count: usize,
58    cell: Rc<dyn Fn(FilmstripCell) -> AnyView>,
59    key: Option<Rc<dyn Fn(usize) -> u64>>,
60    selection: Option<Binding<BTreeSet<usize>>>,
61    mode: SelectionMode,
62    on_activate: Option<Rc<dyn Fn(usize)>>,
63    cell_name: Option<Rc<dyn Fn(usize) -> String>>,
64    cell_width: f32,
65    height: f32,
66    spacing: f32,
67    label: Option<String>,
68    fade: Semantic,
69}
70
71impl fmt::Debug for Filmstrip {
72    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
73        f.debug_struct("Filmstrip").field("count", &self.count).finish_non_exhaustive()
74    }
75}

77impl Filmstrip {
78    /// A strip of `count` cells, with `cell` making the view of each one when it shows.
79    pub fn new<V: View + 'static>(count: usize, cell: impl Fn(FilmstripCell) -> V + 'static) -> Self {
80        Self {
81            count,
82            cell: Rc::new(move |info| Box::new(cell(info))),
83            key: None,
84            selection: None,
85            mode: SelectionMode::Single,
86            on_activate: None,
87            cell_name: None,
88            cell_width: 96.0,
89            height: 96.0,
90            spacing: 6.0,
91            label: None,
92            fade: Semantic::Canvas,
93        }
94    }
95
96    /// What identifies a cell, which is its position by default.
97    pub fn key(mut self, key: impl Fn(usize) -> u64 + 'static) -> Self {
98        self.key = Some(Rc::new(key));
99        self
100    }
101
102    /// The selected cells, to read and to change from outside. A single one that changes from
103    /// outside is brought into view.
104    pub fn selection(mut self, selection: Binding<BTreeSet<usize>>) -> Self {
105        self.selection = Some(selection);
106        self
107    }
108
109    /// How cells can be selected ([`SelectionMode::Single`] by default).
110    pub fn mode(mut self, mode: SelectionMode) -> Self {
111        self.mode = mode;
112        self
113    }
114
115    /// Called with a cell on a double click or on Enter.
116    pub fn on_activate(mut self, handler: impl Fn(usize) + 'static) -> Self {
117        self.on_activate = Some(Rc::new(handler));
118        self
119    }
120
121    /// What screen readers call each cell.
122    pub fn cell_name(mut self, name: impl Fn(usize) -> String + 'static) -> Self {
123        self.cell_name = Some(Rc::new(name));
124        self
125    }
126
127    /// The width of each cell (96 by default).
128    pub fn cell_width(mut self, width: f32) -> Self {
129        self.cell_width = width.max(1.0);
130        self
131    }
132
133    /// The height of the strip (96 by default).
134    pub fn height(mut self, height: f32) -> Self {
135        self.height = height.max(1.0);
136        self
137    }
138
139    /// Gap between cells (6 by default).
140    pub fn spacing(mut self, spacing: f32) -> Self {
141        self.spacing = spacing.max(0.0);
142        self
143    }
144
145    /// What screen readers call the strip, instead of the name in [`Labels`](crate::Labels).
146    pub fn label(mut self, label: impl Into<String>) -> Self {
147        self.label = Some(label.into());
148        self
149    }
150
151    /// The color the arrows fade from, which should be what is behind the strip.
152    pub fn fade(mut self, fade: Semantic) -> Self {
153        self.fade = fade;
154        self
155    }

158/// The frame around a cell, and what makes it an option.
159struct FilmCell {
160    index: usize,
161    selected: bool,
162    active: bool,
163    keyboard: bool,
164    body: Rc<dyn Fn(FilmstripCell) -> AnyView>,
165    name: Option<Rc<dyn Fn(usize) -> String>>,
166    pick: Rc<dyn Fn(usize)>,
167    open: Option<Rc<dyn Fn(usize)>>,
168}
169
170impl View for FilmCell {
171    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
172        let theme = *cx.theme();
173        let hovered = cx.is_hovered();
174        let body =
175            (self.body)(FilmstripCell { index: self.index, selected: self.selected, active: self.active });
176        let child = cx.child(&*body);
177        // By its thickness as well as its color: the state is not left to color alone.
178        let border = if self.selected {
179            Border { width: 2.0, color: theme.color(Semantic::Accent) }
180        } else if self.active && self.keyboard {
181            Border { width: 2.0, color: theme.color(Semantic::Ink) }
182        } else if hovered {
183            Border { width: 1.0, color: theme.color(Semantic::Ink).with_alpha(0.4) }
184        } else {
185            Border { width: 1.0, color: theme.color(Semantic::Border) }
186        };
187        let node = cx.container(
188            LayoutStyle {
189                grow: 1.0,
190                overflow_hidden: true,
191                align: Align::Stretch,
192                padding: Insets::uniform(2.0),
193                ..LayoutStyle::stack(Axis::Vertical)
194            },
195            &[child],
196            Paint::Quad(QuadPaint {
197                corner_radius: theme.radii.sm,
198                border: Some(border),
199                ..QuadPaint::new(theme.color(Semantic::Surface))
200            }),
201        );
202        let (pick, index) = (self.pick.clone(), self.index);
203        cx.on_click(node, Rc::new(move || pick(index)));
204        if let Some(open) = self.open.clone() {
205            cx.on_double_click(node, Rc::new(move || open(index)));
206        }
207        cx.cursor(node, CursorIcon::Pointer);
208        let mut semantics = Semantics::new(Role::ListBoxOption).selected(self.selected).active(self.active);
209        if let Some(name) = &self.name {
210            semantics = semantics.name(name(index));
211        }
212        cx.semantics(node, semantics);
213        node
214    }
215}

217impl View for Filmstrip {
218    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
219        let labels = cx.labels();
220        let local: State<BTreeSet<usize>> = cx.use_state(BTreeSet::new);
221        let selection = self.selection.clone().unwrap_or_else(|| local.binding());
222        let commands: State<Option<ScrollCommand>> = cx.use_state(|| None);
223        let cursor: State<usize> = cx.use_state(|| 0);
224        let anchor: State<usize> = cx.use_state(|| 0);
225        let probe = cx.use_state(|| Rc::new(Cell::new(ListProbe::default()))).peek().clone();
226        // What the strip itself last made of the selection: anything else came from outside.
227        let known: Rc<Cell<Option<usize>>> = cx.use_state(|| Rc::new(Cell::new(None))).peek().clone();
228        let arrow_used: State<bool> = cx.use_state(|| false);
229
230        let count = self.count;
231        let selected = Rc::new(selection.get());
232        let single = (selected.len() == 1).then(|| selected.iter().next().copied()).flatten();
233        if single != known.get() {
234            known.set(single);
235            if let Some(index) = single {
236                if *cursor.peek() != index {
237                    cursor.set(index);
238                    anchor.set(index);
239                }
240                commands.set(Some(ScrollCommand::Reveal(index)));
241            }
242        }
243        let active = cursor.get().min(count.saturating_sub(1));
244        let keyboard = cx.is_focus_visible();
245        let mode = self.mode;
246        let modifiers = cx.modifiers();
247
248        let pick: Rc<dyn Fn(usize)> = {
249            let (selection, cursor, anchor, known) =
250                (selection.clone(), cursor.clone(), anchor.clone(), known.clone());
251            Rc::new(move |index| {
252                let mut next = anchor.get();
253                selection.update(|set| {
254                    next = selection::pick(set, mode, next, index, modifiers.get(), |i| i);
255                    known.set((set.len() == 1).then(|| set.iter().next().copied()).flatten());
256                });
257                anchor.set(next);
258                cursor.set(index);
259            })
260        };
261
262        let (body, name, open, picker, shown) = (
263            self.cell.clone(),
264            self.cell_name.clone(),
265            self.on_activate.clone(),
266            pick.clone(),
267            selected.clone(),
268        );
269        let mut list = VirtualList::new(count, move |index| FilmCell {
270            index,
271            selected: shown.contains(&index),
272            active: index == active,
273            keyboard,
274            body: body.clone(),
275            name: name.clone(),
276            pick: picker.clone(),
277            open: open.clone(),
278        })
279        .horizontal()
280        .row_height(self.cell_width)
281        .spacing(self.spacing)
282        .padding(self.spacing)
283        .height(self.height)
284        .bare()
285        .commands(commands.clone())
286        .probe(probe.clone());
287        list.key = self.key.clone();
288        let strip = cx.child(&list);
289
290        let seen = probe.get();
291        let last = (seen.total - seen.view).max(0.0);
292        let step = (seen.view * ARROW_STEP).max(ARROW_LENGTH);
293        let arrow = |forward: bool| {
294            let at = if forward { seen.view - ARROW_LENGTH } else { 0.0 };
295            let (commands, arrow_used) = (commands.clone(), arrow_used.clone());
296            let by = if forward { step } else { -step };
297            EdgeArrow {
298                forward,
299                vertical: false,
300                frame: Rect::new(at, 0.0, ARROW_LENGTH, self.height),
301                fade: self.fade,
302                label: if forward { labels.scroll_forward.clone() } else { labels.scroll_back.clone() },
303                on_click: Rc::new(move || {
304                    commands.set(Some(ScrollCommand::By(by)));
305                    arrow_used.set(true);
306                }),
307            }
308        };
309        let mut nodes = vec![strip];
310        let back = (seen.offset > 0.5).then(|| cx.keyed_child(1, &arrow(false)));
311        let forward = (seen.offset < last - 0.5).then(|| cx.keyed_child(2, &arrow(true)));
312        nodes.extend(back);
313        nodes.extend(forward);
314        // Reaching the end takes away the arrow that had the keyboard.
315        if arrow_used.get() {
316            arrow_used.set(false);
317            match (back, forward) {
318                (Some(other), None) | (None, Some(other)) => cx.request_focus_node(other),
319                _ => {}
320            }
321        }
322        let root = cx.container(
323            LayoutStyle {
324                height: Some(self.height),
325                min_width: Some(0.0),
326                align: Align::Stretch,
327                align_self: Some(Align::Stretch),
328                ..LayoutStyle::stack(Axis::Vertical)
329            },
330            &nodes,
331            Paint::None,
332        );
333        let name = self.label.clone().unwrap_or_else(|| labels.filmstrip_name.clone());
334        cx.semantics(root, Semantics::new(Role::ListBox).name(name).description(count.to_string()));
335        cx.focusable(root, cx.theme().radii.sm);
336        let open = self.on_activate.clone();
337        let select_all = selection.clone();
338        cx.on_key(
339            root,
340            Rc::new(move |key, modifiers| {
341                if count == 0 {
342                    return false;
343                }
344                let to = match key {
345                    Key::ArrowLeft => active.saturating_sub(1),
346                    Key::ArrowRight => (active + 1).min(count - 1),
347                    Key::Home => 0,
348                    Key::End => count - 1,
349                    Key::Space => {
350                        if mode == SelectionMode::Multiple {
351                            let adding = ravenrs_core::Modifiers { ctrl: true, ..Default::default() };
352                            let mut next = anchor.get();
353                            selection.update(|set| {
354                                next = selection::pick(set, mode, next, active, adding, |i| i);
355                            });
356                            anchor.set(next);
357                        } else {
358                            selection.set(BTreeSet::from([active]));
359                        }
360                        return true;
361                    }
362                    Key::Enter => {
363                        if let Some(open) = &open {
364                            open(active);
365                        }
366                        return true;
367                    }
368                    Key::Character(c) if selection::adds(modifiers) && c.eq_ignore_ascii_case("a") => {
369                        if mode != SelectionMode::Multiple {
370                            return false;
371                        }
372                        select_all.set((0..count).collect());
373                        return true;
374                    }
375                    _ => return false,
376                };
377                cursor.set(to);
378                // Ctrl or Cmd moves without changing the selection.
379                if !selection::adds(modifiers) && mode != SelectionMode::None {
380                    let mut next = anchor.get();
381                    selection.update(|set| {
382                        next = selection::pick(set, mode, next, to, modifiers, |i| i);
383                        known.set((set.len() == 1).then(|| set.iter().next().copied()).flatten());
384                    });
385                    anchor.set(next);
386                }
387                commands.set(Some(ScrollCommand::Reveal(to)));
388                true
389            }),
390        );
391        root
392    }
