1//! Looking at JSON items four ways: a table with a column per key, a tree that opens, the JSON
2//! itself and the schema of what the items hold. The same item and the same field stay chosen
3//! from one way to the next, and the field chosen is said as a path (`cliente.nome`,
4//! `itens[0].preço`) the app can turn into an expression.
5//!
6//! A field that is absent and one that is `null` are told apart everywhere. Whole numbers stay
7//! exact, as the JSON had them. A list longer than the limit shows its first items and says so.
8

24/// How many items, and how many entries of a list inside one, are shown before the rest is
25/// left out, unless [`DataInspector::limit`] says otherwise.
26pub const SAMPLE: usize = 1000;
27
28/// The path of a field inside an item: keys joined by dots, rows in brackets, and a key that
29/// is not a plain word in quotes, as `cliente.nome`, `itens[0].preço` or `["nome completo"]`.
30pub fn field_path(steps: &[PathStep]) -> String {
31    let mut out = String::new();
32    for step in steps {
33        match step {
34            PathStep::Row(i) => out.push_str(&format!("[{i}]")),
35            PathStep::Key(k) if plain(k) => {
36                if !out.is_empty() {
37                    out.push('.');
38                }
39                out.push_str(k);
40            }
41            PathStep::Key(k) => out.push_str(&format!("[{}]", Value::String(k.clone()))),
42        }
43    }
44    out
45}
46
47/// A key that reads as it is in a path: letters, digits and `_`, not starting with a digit.
48fn plain(key: &str) -> bool {
49    let mut chars = key.chars();
50    chars.next().is_some_and(|c| c.is_alphabetic() || c == '_')
51        && chars.all(|c| c.is_alphanumeric() || c == '_')
52}

54/// One step of a [`field_path`]: a key of an object or a row of a list.
55#[derive(Clone, Debug, PartialEq, Eq)]
56#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
57pub enum PathStep {
58    /// A key of an object.
59    Key(String),
60    /// A row of a list.
61    Row(usize),
62}
63
64/// Hears the path of a field chosen.
65type Pick = Rc<dyn Fn(&str)>;

67/// Looks at JSON items as a table, a tree, the JSON itself or their schema.
68#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
69pub struct DataInspector {

74    selected: Option<Binding<Option<String>>>,
75    on_pick: Option<Pick>,
76    limit: usize,
77    label: Option<String>,

88    /// The items of the JSON `text`: each entry of a list, or the one object. Text that is not
89    /// JSON shows as no items.
90    pub fn new(text: impl Into<String>) -> Self {
91        let value: Value = serde_json::from_str(&text.into()).unwrap_or(Value::Null);
92        let items = match value {
93            Value::Array(items) => items,
94            Value::Null => Vec::new(),
95            other => vec![other],
96        };
97        Self::of(items)
98    }
99
100    /// The items as the app has them.
101    pub fn of(items: Vec<Value>) -> Self {
102        Self {
103            total: items.len(),
104            items: Rc::new(items),
105            mode: None,
106            item: None,
107            selected: None,
108            on_pick: None,
109            limit: SAMPLE,
110            label: None,
111        }
112    }
113
114    /// Which way it shows them: 0 the table, 1 the tree, 2 the JSON, 3 the schema.
115    pub fn mode(mut self, mode: Binding<usize>) -> Self {
116        self.mode = Some(mode);
117        self
118    }
119
120    /// Which item the tree and the JSON show, from 0.
121    pub fn item(mut self, item: Binding<usize>) -> Self {
122        self.item = Some(item);
123        self
124    }
125
126    /// The path of the field chosen, the same in every way of showing.
127    pub fn selected(mut self, selected: Binding<Option<String>>) -> Self {
128        self.selected = Some(selected);
129        self
130    }
131
132    /// Hears the path of a field chosen in the tree or the schema.
133    pub fn on_pick(mut self, handler: impl Fn(&str) + 'static) -> Self {
134        self.on_pick = Some(Rc::new(handler));
135        self
136    }
137
138    /// How many items, and entries of a list inside one, it shows before saying the rest was
139    /// left out.
140    pub fn limit(mut self, limit: usize) -> Self {
141        self.limit = limit.max(1);
142        self
143    }
144
145    /// What screen readers call it.
146    pub fn label(mut self, label: impl Into<String>) -> Self {
147        self.label = Some(label.into());

149    }

152/// A value as a cell says it: text as it is, numbers exact, a list or an object by its size.
153fn cell(value: Option<&Value>, labels: &crate::Labels) -> String {
154    match value {
155        None => labels.inspector_absent.clone(),
156        Some(Value::Null) => "null".to_owned(),
157        Some(Value::String(s)) => s.clone(),
158        Some(Value::Array(a)) => format!("[{}]", a.len()),
159        Some(Value::Object(o)) => format!("{{{}}}", o.len()),
160        Some(other) => other.to_string(),
161    }
162}

164/// The tree of `value`, its items keyed by their paths, lists cut at `limit`.
165fn tree(
166    steps: &mut Vec<PathStep>,
167    name: String,
168    value: &Value,
169    limit: usize,
170    labels: &crate::Labels,
171) -> TreeNode {
172    let key = field_path(steps);
173    match value {
174        Value::Object(map) => {
175            let mut node = TreeNode::new(key, format!("{name} {{{}}}", map.len()));
176            for (k, v) in map {
177                steps.push(PathStep::Key(k.clone()));
178                node = node.child(tree(steps, k.clone(), v, limit, labels));
179                steps.pop();
180            }
181            node
182        }
183        Value::Array(list) => {
184            let mut node = TreeNode::new(key.clone(), format!("{name} [{}]", list.len()));
185            for (i, v) in list.iter().enumerate().take(limit) {
186                steps.push(PathStep::Row(i));
187                node = node.child(tree(steps, format!("[{i}]"), v, limit, labels));
188                steps.pop();
189            }
190            if list.len() > limit {
191                let left = labels.inspector_more.replace("{0}", &(list.len() - limit).to_string());
192                node = node.child(TreeNode::new(format!("{key}…"), left).disabled(true));
193            }
194            node
195        }
196        leaf => TreeNode::new(key, format!("{name}: {}", cell(Some(leaf), labels))),
197    }
198}

200/// The keys that reach each path of `value`, by the path as people read it.
201fn keys_of(value: &Value, steps: &mut Vec<PathStep>, out: &mut BTreeMap<String, Vec<String>>) {
202    if !steps.is_empty() {
203        let keys = steps
204            .iter()
205            .map(|s| match s {
206                PathStep::Key(k) => k.clone(),
207                PathStep::Row(i) => i.to_string(),
208            })
209            .collect();
210        out.insert(field_path(steps), keys);
211    }
212    match value {
213        Value::Object(map) => {
214            for (k, v) in map {
215                steps.push(PathStep::Key(k.clone()));
216                keys_of(v, steps, out);
217                steps.pop();
218            }
219        }
220        Value::Array(list) => {
221            for (i, v) in list.iter().enumerate().take(SAMPLE) {
222                steps.push(PathStep::Row(i));
223                keys_of(v, steps, out);
224                steps.pop();
225            }
226        }
227        _ => {}
228    }
229}

254/// Every path of the items, with the kinds seen at it and in how many items it is.
255fn schema(items: &[Value]) -> BTreeMap<String, (BTreeSet<String>, usize)> {
256    fn walk(
257        steps: &mut Vec<PathStep>,
258        value: &Value,
259        seen: &mut BTreeMap<String, (BTreeSet<String>, usize)>,
260        first: bool,
261    ) {
262        if !steps.is_empty() {
263            let entry = seen.entry(field_path(steps)).or_default();
264            entry.0.insert(kind_word(value).to_owned());
265            if first {
266                entry.1 += 1;
267            }
268        }
269        match value {
270            Value::Object(map) => {
271                for (k, v) in map {
272                    steps.push(PathStep::Key(k.clone()));
273                    walk(steps, v, seen, first);
274                    steps.pop();
275                }
276            }
277            // The entries of a list share one path, the first entry's, so a list of records
278            // reads as its records.
279            Value::Array(list) => {
280                for (i, v) in list.iter().enumerate() {
281                    steps.push(PathStep::Row(0));
282                    walk(steps, v, seen, first && i == 0);
283                    steps.pop();
284                }
285            }
286            _ => {}
287        }
288    }
289    let mut seen = BTreeMap::new();
290    for item in items {
291        walk(&mut Vec::new(), item, &mut seen, true);
292    }
293    seen
294}

296/// The kind of a value, as the schema keys it before the words.
297fn kind_word(value: &Value) -> &'static str {
298    match value {
299        Value::Null => "null",
300        Value::Bool(_) => "boolean",
301        Value::Number(_) => "number",
302        Value::String(_) => "text",
303        Value::Array(_) => "list",
304        Value::Object(_) => "object",
305    }
306}

308impl View for DataInspector {
309    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
310        let labels = cx.labels();
311        let own_mode = cx.use_state(|| 0_usize);
312        let own_item = cx.use_state(|| 0_usize);
313        let own_selected = cx.use_state(|| None::<String>);
314        let expanded = cx.use_state(BTreeSet::<String>::new);
315        let mode = self.mode.clone().unwrap_or_else(|| own_mode.binding());
316        let item = self.item.clone().unwrap_or_else(|| own_item.binding());
317        let selected = self.selected.clone().unwrap_or_else(|| own_selected.binding());
318
319        let shown = self.items.len().min(self.limit);
320        let at = item.get().min(shown.saturating_sub(1));
321        let mut parts: Vec<AnyView> = Vec::new();
322        parts.push(Box::new(
323            SegmentedControl::new(
324                mode.clone(),
325                [
326                    labels.inspector_table.clone(),
327                    labels.inspector_tree.clone(),
328                    labels.inspector_json.clone(),
329                    labels.inspector_schema.clone(),
330                ],
331            )
332            .label(self.label.clone().unwrap_or_else(|| labels.inspector_mode.clone())),
333        ));
334        if self.total > shown {
335            parts.push(Box::new(
336                Text::new(
337                    labels
338                        .inspector_sampled
339                        .replace("{0}", &shown.to_string())
340                        .replace("{1}", &self.total.to_string()),
341                )
342                .style(TextStyle::Callout)
343                .foreground(Semantic::Warning),
344            ));
345        }
346        if self.items.is_empty() {
347            parts.push(Box::new(Text::new(labels.inspector_empty.clone()).foreground(Semantic::InkMuted)));
348            return cx.child(&VStack::new(parts).spacing(8.0).align(Align::Stretch));
349        }
350        // Tree and JSON show one item at a time.
351        if matches!(mode.get(), 1 | 2) && shown > 1 {
352            let (back, next) = (item.clone(), item.clone());
353            let last = shown - 1;
354            parts.push(Box::new(
355                HStack::new((
356                    IconButton::new(IconName::ChevronLeft)
357                        .label(labels.inspector_previous.clone())
358                        .size(ButtonSize::Small)
359                        .style(ButtonStyle::Ghost)
360                        .disabled(at == 0)
361                        .on_click(move || back.set(at.saturating_sub(1))),
362                    Text::new(
363                        labels
364                            .inspector_item
365                            .replace("{0}", &(at + 1).to_string())
366                            .replace("{1}", &shown.to_string()),
367                    )
368                    .style(TextStyle::Callout),
369                    IconButton::new(IconName::ChevronRight)
370                        .label(labels.inspector_next.clone())
371                        .size(ButtonSize::Small)
372                        .style(ButtonStyle::Ghost)
373                        .disabled(at == last)
374                        .on_click(move || next.set((at + 1).min(last))),
375                    Spacer::new(),
376                ))
377                .spacing(4.0)
378                .align(Align::Center),
379            ));
380        }
381        let items = &self.items[..shown];
382        let body: AnyView = match mode.get() {
383            1 => {
384                let picked = selected.clone();
385                let on_pick = self.on_pick.clone();
386                let roots: Vec<TreeNode> = match &items[at] {
387                    Value::Object(map) => map
388                        .iter()
389                        .map(|(k, v)| {
390                            tree(&mut vec![PathStep::Key(k.clone())], k.clone(), v, self.limit, &labels)
391                        })
392                        .collect(),
393                    other => vec![tree(&mut Vec::new(), "item".to_owned(), other, self.limit, &labels)],
394                };
395                // A pick in the tree is heard as the path chosen.
396                let read = picked.clone();
397                let heard = Binding::new(
398                    move || read.get(),
399                    move |path: Option<String>| {
400                        if let (Some(path), Some(on_pick)) = (&path, &on_pick) {
401                            on_pick(path);
402                        }
403                        picked.set(path);
404                    },
405                );
406                Box::new(
407                    TreeView::new(roots)
408                        .selected(heard)
409                        .expanded(expanded.binding())
410                        .label(self.label.clone().unwrap_or_else(|| labels.inspector_tree.clone())),
411                )
412            }
413            2 => Box::new(ScrollView::new((Text::new(
414                serde_json::to_string_pretty(&items[at]).unwrap_or_default(),
415            )
416            .style(TextStyle::Callout),))),
417            3 => {
418                let mut table = Table::new([
419                    Column::new(labels.inspector_path.clone()),
420                    Column::new(labels.inspector_kind.clone()),
421                    Column::new(labels.inspector_in.clone()),
422                ]);
423                for (path, (kinds, count)) in schema(items) {
424                    let words: Vec<String> = kinds
425                        .iter()
426                        .map(|k| match k.as_str() {
427                            "null" => "null".to_owned(),
428                            "boolean" => labels.inspector_boolean.clone(),
429                            "number" => labels.inspector_number.clone(),
430                            "text" => labels.inspector_text.clone(),
431                            "list" => labels.inspector_list.clone(),
432                            _ => labels.inspector_object.clone(),
433                        })
434                        .collect();
435                    let mark = if selected.get().as_deref() == Some(path.as_str()) { "▸ " } else { "" };
436                    table = table.row([
437                        format!("{mark}{path}"),
438                        words.join(" · "),
439                        labels
440                            .inspector_present
441                            .replace("{0}", &count.to_string())
442                            .replace("{1}", &items.len().to_string()),
443                    ]);
444                }
445                Box::new(table)
446            }
447            _ => {
448                // A column per key, in the order the keys sort (as JSON objects keep them).
449                let mut keys: Vec<String> = Vec::new();
450                for it in items {
451                    if let Value::Object(map) = it {
452                        for k in map.keys() {
453                            if !keys.contains(k) {
454                                keys.push(k.clone());
455                            }
456                        }
457                    }
458                }
459                let mut table = Table::new(keys.iter().map(|k| Column::new(k.clone())));
460                for it in items {
461                    table = table.row(keys.iter().map(|k| cell(it.get(k.as_str()), &labels)));
462                }
463                Box::new(table)
464            }
465        };
466        parts.push(body);
467        if let Some(path) = selected.get() {
468            parts.push(Box::new(
469                Text::new(labels.inspector_chosen.replace("{0}", &path))
470                    .style(TextStyle::Callout)
471                    .foreground(Semantic::InkMuted),
472            ));
473            // The field chosen can be carried to an expression, or copied as one from the keyboard.
474            let mut keys = BTreeMap::new();
475            keys_of(&items[at], &mut Vec::new(), &mut keys);
476            if let Some(found) = keys.get(&path) {
477                let field = FieldDrag::new(path.clone(), crate::field_expression(found));
478                let (clipboard, expression) = (cx.clipboard(), field.expression.clone());
479                parts.push(Box::new(
480                    HStack::new((
481                        Carry { words: labels.inspector_drag.replace("{0}", &path), field },
482                        Button::new(labels.inspector_copy.clone())
483                            .icon(IconName::Copy)
484                            .style(ButtonStyle::Secondary)
485                            .size(ButtonSize::Small)
486                            .on_click(move || clipboard.write(&expression)),
487                    ))
488                    .spacing(8.0)
489                    .align(Align::Center),
490                ));
491            }
492        }
493        cx.child(&VStack::new(parts).spacing(8.0).align(Align::Stretch))
494    }
495}

501    #[test]
502    fn a_path_quotes_what_is_not_a_plain_word() {
503        let steps =
504            [PathStep::Key("cliente".into()), PathStep::Key("nome completo".into()), PathStep::Row(2)];
505        assert_eq!(field_path(&steps), r#"cliente["nome completo"][2]"#);
506        assert_eq!(
507            field_path(&[PathStep::Key("itens".into()), PathStep::Row(0), PathStep::Key("preço".into())]),
508            "itens[0].preço"
509        );
510    }
511
512    #[test]
513    fn the_schema_counts_in_how_many_items_a_field_is() {
514        let items = vec![serde_json::json!({"a": 1, "b": null}), serde_json::json!({"a": "x"})];
515        let seen = schema(&items);
516        assert_eq!(seen["a"].1, 2);
517        assert_eq!(seen["a"].0.len(), 2, "a number and a text");
518        assert_eq!(seen["b"].1, 1, "absent in the second");
519    }
