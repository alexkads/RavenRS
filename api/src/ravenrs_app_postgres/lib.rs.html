32/// PostgreSQL counts from the first day of 2000.
33fn its_epoch() -> i64 {
34    DateTime::from_ymd(2000, 1, 1).map_or(0, |date| date.days)
35}

41fn number_of(value: &Value) -> Result<f64, Boxed> {
42    match value {
43        Value::Number(number) => Ok(*number),
44        Value::Bool(on) => Ok(f64::from(u8::from(*on))),
45        Value::Text(text) => Ok(text.trim().parse()?),
46        other => Err(format!("{other:?} is not a number").into()),
47    }
48}

50fn date_of(value: &Value) -> Result<DateTime, Boxed> {
51    match value {
52        Value::Date(moment) => Ok(*moment),
53        Value::Text(text) => DateTime::parse(text).ok_or_else(|| format!("{text} is not a date").into()),
54        other => Err(format!("{other:?} is not a date").into()),
55    }
56}

58fn text_of(value: &Value) -> String {
59    match value {
60        Value::Text(text) => text.clone(),
61        Value::Date(moment) => moment.to_string(),
62        Value::Bool(on) => on.to_string(),
63        Value::Number(number) if number.fract() == 0.0 && number.abs() < 9e15 => (*number as i64).to_string(),
64        Value::Number(number) => number.to_string(),
65        _ => String::new(),
66    }
67}

73/// A number in the form PostgreSQL has for `numeric`: groups of four digits around the point.
74/// The digits are those of the shortest text that reads back as the number, so 16.47 goes as
75/// 16.47, with as many places as it has.
76fn write_numeric(number: f64, out: &mut BytesMut) -> Result<(), Boxed> {
77    if !number.is_finite() {
78        return Err("a number has to be finite".into());
79    }
80    let text = number.abs().to_string();
81    let (whole, part) = text.split_once('.').unwrap_or((&text, ""));
82    let whole = whole.trim_start_matches('0');
83    let mut groups: Vec<&str> = Vec::new();
84    let lead = whole.len() % 4;
85    if lead > 0 {
86        groups.push(&whole[..lead]);
87    }
88    groups.extend(whole.as_bytes()[lead..].chunks(4).map(|chunk| std::str::from_utf8(chunk).unwrap_or("0")));
89    let weight = groups.len() as i16 - 1;
90    let mut digits: Vec<i16> = groups.iter().map(|group| group.parse()).collect::<Result<_, _>>()?;
91    let padded = format!("{part:0<width$}", width = part.len().div_ceil(4) * 4);
92    for chunk in padded.as_bytes().chunks(4) {
93        digits.push(std::str::from_utf8(chunk)?.parse()?);
94    }
95    out.put_i16(digits.len() as i16);
96    out.put_i16(weight);
97    out.put_u16(if number < 0.0 { 0x4000 } else { 0 });
98    out.put_i16(part.len() as i16);
99    for digit in digits {
100        out.put_i16(digit);
101    }
102    Ok(())
103}

105fn read_numeric(raw: &[u8]) -> Result<f64, Boxed> {
106    let word = |at: usize| -> Result<i16, Boxed> {
107        let bytes = raw.get(at..at + 2).ok_or("a numeric is cut short")?;
108        Ok(i16::from_be_bytes([bytes[0], bytes[1]]))
109    };
110    let (count, weight, sign) = (word(0)? as usize, i32::from(word(2)?), word(4)? as u16);
111    if sign == 0xC000 {
112        return Ok(f64::NAN);
113    }
114    // Written out as text and read once: adding the groups up as numbers would land beside the
115    // value, 16.47 as 16.470000000000002.
116    let mut whole = String::new();
117    let mut part = String::new();
118    for at in 0..count {
119        let group = format!("{:04}", word(8 + at * 2)?);
120        if (at as i32) <= weight { whole.push_str(&group) } else { part.push_str(&group) }
121    }
122    for _ in count as i32..=weight {
123        whole.push_str("0000");
124    }
125    // Groups that start after the point leave zeros between it and them.
126    let gap = (-weight - 1).max(0) as usize;
127    let text = format!(
128        "{}{}.{}{part}",
129        if sign == 0x4000 { "-" } else { "" },
130        if whole.is_empty() { "0" } else { &whole },
131        "0000".repeat(gap)
132    );
133    Ok(text.parse()?)
134}

136impl ToSql for Bound<'_> {
137    fn to_sql(&self, ty: &Type, out: &mut BytesMut) -> Result<IsNull, Boxed> {
138        if matches!(self.0, Value::Null) {
139            return Ok(IsNull::Yes);
140        }
141        match *ty {
142            Type::BOOL => {
143                let on = match self.0 {
144                    Value::Bool(on) => *on,
145                    other => number_of(other)? != 0.0,
146                };
147                out.put_u8(u8::from(on));
148            }
149            Type::INT2 => out.put_i16(number_of(self.0)? as i16),
150            Type::INT4 => out.put_i32(number_of(self.0)? as i32),
151            Type::INT8 => out.put_i64(number_of(self.0)? as i64),
152            Type::FLOAT4 => out.put_f32(number_of(self.0)? as f32),
153            Type::FLOAT8 => out.put_f64(number_of(self.0)?),
154            Type::NUMERIC => write_numeric(number_of(self.0)?, out)?,
155            Type::DATE => out.put_i32((date_of(self.0)?.days - its_epoch()) as i32),
156            Type::TIMESTAMP | Type::TIMESTAMPTZ => {
157                let moment = date_of(self.0)?;
158                let days = moment.days - its_epoch();
159                out.put_i64((days * SECONDS_A_DAY + i64::from(moment.seconds)) * MICROS);
160            }
161            Type::UUID => {
162                let text = text_of(self.0).replace('-', "");
163                if text.len() != 32 {
164                    return Err("not a uuid".into());
165                }
166                for at in (0..32).step_by(2) {
167                    out.put_u8(u8::from_str_radix(&text[at..at + 2], 16)?);
168                }
169            }
170            _ => out.put_slice(text_of(self.0).as_bytes()),
171        }
172        Ok(IsNull::No)
173    }
174
175    fn accepts(_: &Type) -> bool {
176        true
177    }
178
179    /// With no type said, the value goes as text, as it would be written in the statement.
180    fn encode_format(&self, ty: &Type) -> Format {
181        if ty.oid() == 0 { Format::Text } else { Format::Binary }
182    }
183
184    to_sql_checked!();
185}

190impl<'a> FromSql<'a> for Cell {
191    fn from_sql(ty: &Type, raw: &'a [u8]) -> Result<Self, Boxed> {
192        let cell = match *ty {
193            Type::BOOL => SqlCell::Bool(bool::from_sql(ty, raw)?),
194            Type::INT2 => SqlCell::Integer(i64::from(i16::from_sql(ty, raw)?)),
195            Type::INT4 => SqlCell::Integer(i64::from(i32::from_sql(ty, raw)?)),
196            Type::INT8 => SqlCell::Integer(i64::from_sql(ty, raw)?),
197            Type::FLOAT4 => SqlCell::Real(f64::from(f32::from_sql(ty, raw)?)),
198            Type::FLOAT8 => SqlCell::Real(f64::from_sql(ty, raw)?),
199            Type::NUMERIC => {
200                let number = read_numeric(raw)?;
201                if number.fract() == 0.0 && number.abs() < 9e15 {
202                    SqlCell::Integer(number as i64)
203                } else {
204                    SqlCell::Real(number)
205                }
206            }
207            Type::DATE => {
208                let days = i64::from(i32::from_sql(&Type::INT4, raw)?);
209                SqlCell::Date(DateTime { days: days + its_epoch(), seconds: 0 })
210            }
211            Type::TIMESTAMP | Type::TIMESTAMPTZ => {
212                let total = i64::from_sql(&Type::INT8, raw)?.div_euclid(MICROS);
213                SqlCell::Date(DateTime {
214                    days: total.div_euclid(SECONDS_A_DAY) + its_epoch(),
215                    seconds: total.rem_euclid(SECONDS_A_DAY) as u32,
216                })
217            }
218            Type::UUID => {
219                let hex: String = raw.iter().map(|byte| format!("{byte:02x}")).collect();
220                let part = |from: usize, to: usize| hex.get(from..to).unwrap_or("");
221                SqlCell::Text(format!(
222                    "{}-{}-{}-{}-{}",
223                    part(0, 8),
224                    part(8, 12),
225                    part(12, 16),
226                    part(16, 20),
227                    part(20, 32)
228                ))
229            }
230            // Whatever else has no field to show it, as with a blob in SQLite.
231            Type::BYTEA => SqlCell::Null,
232            _ => std::str::from_utf8(raw).map_or(SqlCell::Null, |text| SqlCell::Text(text.to_owned())),
233        };
234        Ok(Self(cell))
235    }
236
237    fn from_sql_null(_: &Type) -> Result<Self, Boxed> {
238        Ok(Self(SqlCell::Null))
239    }
240
241    fn accepts(_: &Type) -> bool {
242        true
243    }
244}

246fn sql_error(error: &postgres::Error) -> SqlError {
247    let Some(database) = error.as_db_error() else { return SqlError::other(error) };
248    let code = database.code();
249    if *code == SqlState::UNIQUE_VIOLATION {
250        // "Key (name)=(x) already exists." The value stays out of it.
251        let column = database
252            .detail()
253            .and_then(|detail| detail.strip_prefix("Key ("))
254            .and_then(|rest| rest.split(')').next())
255            .and_then(|columns| columns.split(", ").next())
256            .map(|column| column.trim_matches('"').to_owned());
257        return SqlError::Unique(column);
258    }
259    if *code == SqlState::FOREIGN_KEY_VIOLATION {
260        return SqlError::ForeignKey;
261    }
262    if code.code().starts_with("23") {
263        return SqlError::Constraint;
264    }
265    SqlError::Other(database.message().to_owned())
266}

268/// A connection to PostgreSQL.
269///
270/// Made from a [`Client`] you opened, or with [`PostgresBackend::connect`] from the address a
271/// database's panel gives. One made from an address is opened again by itself when it dropped
272/// while no transaction was open, as a database that went to sleep leaves it. A statement already
273/// sent is never sent again, unless it only reads.
274pub struct PostgresBackend {
275    client: RefCell<Client>,
276    statements: RefCell<HashMap<String, postgres::Statement>>,
277    mode: Statements,
278    reopen: Option<Reopen>,
279    in_transaction: Flag<bool>,
280}

286    }

304/// What the server says when a parameter has no type it can work out, nor any other.

309fn cells_of(row: &postgres::Row) -> Result<Vec<SqlCell>, postgres::Error> {
310    (0..row.len()).map(|at| row.try_get::<_, Cell>(at).map(|cell| cell.0)).collect()
311}

313/// Whether the connection is gone, rather than the statement refused.
314fn dropped(error: &postgres::Error, client: &Client) -> bool {
315    error.is_closed()
316        || client.is_closed()
317        || error.code().is_some_and(|code| {
318            [SqlState::ADMIN_SHUTDOWN, SqlState::CRASH_SHUTDOWN, SqlState::CANNOT_CONNECT_NOW].contains(code)
319        })
320}

322impl PostgresBackend {
323    /// Opens a connection from an address such as `postgresql://user:password@host/db?sslmode=require`.
324    /// See [`PostgresOptions`] for what TLS and poolers do.
325    ///
326    /// # Errors
327    ///
328    /// When the address cannot be read, or the server cannot be reached or refuses.
329    pub fn connect(address: &str) -> Result<Self, ConnectError> {
330        Self::connect_with(address, PostgresOptions::new())
331    }
332
333    /// [`PostgresBackend::connect`], told more than the address says.
334    ///
335    /// # Errors
336    ///
337    /// As [`PostgresBackend::connect`], and when a certificate authority cannot be read.
338    pub fn connect_with(address: &str, options: PostgresOptions) -> Result<Self, ConnectError> {
339        let (reopen, mode) = connect::prepare(address, &options)?;
340        let client = reopen.open().map_err(ConnectError::Connect)?;
341        Ok(Self { mode, reopen: Some(reopen), ..Self::from(client) })
342    }
343
344    /// The same connection, with statements going this way. For a [`Client`] behind a pooler.
345    #[must_use]
346    pub fn with_statements(mut self, mode: Statements) -> Self {
347        self.mode = mode;
348        self
349    }
350
351    /// How statements go.
352    pub fn statement_mode(&self) -> Statements {
353        self.mode
354    }
355
356    fn statement(&self, sql: &str) -> Result<postgres::Statement, postgres::Error> {
357        if let Some(found) = self.statements.borrow().get(sql) {
358            return Ok(found.clone());
359        }
360        let made = self.client.borrow_mut().prepare(sql)?;
361        self.statements.borrow_mut().insert(sql.to_owned(), made.clone());
362        Ok(made)
363    }
364
365    fn prepared(&self, sql: &str, values: &[Value], reads: bool) -> Result<Ran, postgres::Error> {
366        let statement = self.statement(sql)?;
367        let bound: Vec<Bound<'_>> = values.iter().map(Bound).collect();
368        let parameters: Vec<&(dyn ToSql + Sync)> =
369            bound.iter().map(|value| value as &(dyn ToSql + Sync)).collect();
370        if !reads {
371            let changed = self.client.borrow_mut().execute(&statement, &parameters)?;
372            return Ok((Vec::new(), Vec::new(), changed));
373        }
374        let names: Vec<String> = statement.columns().iter().map(|column| column.name().to_owned()).collect();
375        let rows = self.client.borrow_mut().query(&statement, &parameters)?;
376        let read = rows.iter().map(cells_of).collect::<Result<Vec<_>, _>>()?;
377        Ok((names, read, 0))
378    }
379
380    /// One round trip, and nothing left on the server: the parameters go as text, and the server
381    /// works out their types as it does when it prepares.
382    fn unprepared(&self, sql: &str, values: &[Value]) -> Result<Ran, postgres::Error> {
383        let bound: Vec<Bound<'_>> = values.iter().map(Bound).collect();
384        let mut client = self.client.borrow_mut();
385        let mut rows = client
386            .query_typed_raw(sql, bound.iter().map(|value| (value as &(dyn ToSql + Sync), unspecified())))?;
387        let mut names = Vec::new();
388        let mut read = Vec::new();
389        while let Some(row) = rows.next()? {
390            if names.is_empty() {
391                names = row.columns().iter().map(|column| column.name().to_owned()).collect();
392            }
393            read.push(cells_of(&row)?);
394        }
395        let changed = rows.rows_affected().unwrap_or(0);
396        Ok((names, read, changed))
397    }
398
399    fn run(&self, sql: &str, values: &[Value], reads: bool) -> Result<Ran, postgres::Error> {
400        match self.mode {
401            Statements::Prepared => self.prepared(sql, values, reads),
402            Statements::Unprepared => self.unprepared(sql, values),
403        }
404    }
405
406    /// Opens the connection again when it dropped and no transaction was open on it.
407    fn revive(&self) -> Result<(), SqlError> {
408        let Some(reopen) = &self.reopen else { return Ok(()) };
409        if self.in_transaction.get() || !self.client.borrow().is_closed() {
410            return Ok(());
411        }
412        self.renew(reopen)
413    }
414
415    fn renew(&self, reopen: &Reopen) -> Result<(), SqlError> {
416        let client = reopen.open().map_err(|error| sql_error(&error))?;
417        *self.client.borrow_mut() = client;
418        self.statements.borrow_mut().clear();
419        Ok(())
420    }
421
422    /// Does the work, and once more on a new connection when the old one was gone and doing it
423    /// again changes nothing twice.
424    fn on_connection<T>(
425        &self,
426        again: bool,
427        work: impl Fn(&Self) -> Result<T, postgres::Error>,
428    ) -> Result<T, SqlError> {
429        self.revive()?;
430        match work(self) {
431            Ok(done) => Ok(done),
432            Err(error) => match &self.reopen {
433                Some(reopen)
434                    if again && !self.in_transaction.get() && dropped(&error, &self.client.borrow()) =>
435                {
436                    self.renew(reopen)?;
437                    work(self).map_err(|error| sql_error(&error))
438                }
439                _ => Err(sql_error(&error)),
440            },
441        }
442    }
443}

445impl SqlBackend for PostgresBackend {
446    fn dialect(&self) -> Dialect {
447        Dialect::Postgres
448    }
449
450    fn column_type(&self, kind: &FieldKind) -> Cow<'static, str> {
451        match kind {
452            FieldKind::Integer | FieldKind::Money => "BIGINT".into(),
453            // Exact: 16.47 is 16.47 here, and not the binary number nearest to it.
454            FieldKind::Decimal(places) => format!("NUMERIC(18, {})", places.min(&18)).into(),
455            FieldKind::Boolean => "BOOLEAN".into(),
456            _ => "TEXT".into(),
457        }
458    }
459
460    fn decimal_upgrade(&self, table: &str, column: &str, places: u8) -> Option<String> {
461        let (table, column) = (quoted(table), quoted(column));
462        let places = places.min(18);
463        Some(format!(
464            "ALTER TABLE {table} ALTER COLUMN {column} TYPE NUMERIC(18, {places}) \
465             USING round({column}::numeric, {places})"
466        ))
467    }
468
469    fn slot(&self, at: usize) -> String {
470        format!("${at}")
471    }
472
473    fn offset_only(&self, offset: usize) -> String {
474        format!(" OFFSET {offset}")
475    }
476
477    fn columns_of(&self, table: &str) -> Result<Vec<String>, SqlError> {
478        let sql = "SELECT column_name::text FROM information_schema.columns \
479                   WHERE table_schema = current_schema() AND table_name = $1";
480        let (_, rows, _) =
481            self.on_connection(true, |me| me.run(sql, &[Value::Text(table.to_owned())], true))?;
482        Ok(rows
483            .into_iter()
484            .filter_map(|row| match row.into_iter().next() {
485                Some(SqlCell::Text(name)) => Some(name),
486                _ => None,
487            })
488            .collect())
489    }
490
491    fn execute(&self, sql: &str, values: &[Value]) -> Result<u64, SqlError> {
492        // A write is not sent twice: the first may have landed before the connection dropped.
493        self.on_connection(false, |me| me.run(sql, values, false)).map(|(_, _, changed)| changed)
494    }
495
496    fn batch(&self, sql: &str) -> Result<(), SqlError> {
497        let ends = matches!(sql, "COMMIT" | "ROLLBACK");
498        // What changes the schema can leave statements that no longer fit.
499        if !ends && sql != "BEGIN" {
500            self.statements.borrow_mut().clear();
501        }
502        if ends {
503            // Only on the connection the transaction is on: COMMIT on a new one would find nothing
504            // to commit, and say it went well.
505            let done = self.client.borrow_mut().batch_execute(sql).map_err(|error| sql_error(&error));
506            self.in_transaction.set(false);
507            return done;
508        }
509        // Nothing has happened yet when BEGIN goes, so it can go again on a new connection.
510        let done = self.on_connection(sql == "BEGIN", |me| me.client.borrow_mut().batch_execute(sql));
511        if sql == "BEGIN" && done.is_ok() {
512            self.in_transaction.set(true);
513        }
514        done
515    }
516
517    fn query(&self, sql: &str, values: &[Value]) -> Result<SqlRows, SqlError> {
518        let (names, rows, _) = self.on_connection(true, |me| me.run(sql, values, true))?;
519        Ok(SqlRows::new(names, rows))
520    }
521}
