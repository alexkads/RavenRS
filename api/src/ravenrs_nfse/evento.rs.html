1//! The request for an event: here only the cancellation, `e101101`.
2
3use ravenrs_dfe::xml::escape_text;
4use ravenrs_dfe::{Algoritmo, Signer};
5
6use crate::ambiente::{NS_NFSE, VER_APLIC, VERSAO};
7use crate::chave::ChaveNfse;
8use crate::erro::NfseError;
9use crate::modelo::Inscricao;
10
11/// The code of the cancellation event.
12pub const CANCELAMENTO: &str = "101101";
13
14/// Why an NFS-e is canceled, `cMotivo`.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
16pub enum MotivoCancelamento {
17    /// 1, a mistake in the issue.
18    ErroNaEmissao,
19    /// 2, the service was not provided.
20    ServicoNaoPrestado,
21    /// 9, anything else.
22    Outros,
23}
24
25impl MotivoCancelamento {
26    /// The code.
27    #[must_use]
28    pub const fn codigo(self) -> u8 {
29        match self {
30            Self::ErroNaEmissao => 1,
31            Self::ServicoNaoPrestado => 2,
32            Self::Outros => 9,
33        }
34    }
35}
36
37/// The Id of the request: `PRE` + key (50) + event (6), the pattern `PRE[0-9]{56}`. The annotation
38/// of the XSD also mentions a sequence; this follows what the ACBr does. Test before relying on it.
39#[must_use]
40pub fn id_pedido(chave: &ChaveNfse, evento: &str) -> String {
41    format!("PRE{}{evento}", chave.as_str())
42}
43
44/// Writes and signs the request for cancellation.
45pub fn pedido_cancelamento(
46    chave: &ChaveNfse,
47    autor: &Inscricao,
48    motivo: MotivoCancelamento,
49    justificativa: &str,
50    quando: &str,
51    signer: &dyn Signer,
52    algoritmo: Algoritmo,
53) -> Result<String, NfseError> {
54    let justificativa = justificativa.trim();
55    if justificativa.chars().count() < 15 || justificativa.chars().count() > 255 {
56        return Err(NfseError::validacao("xMotivo", "a justificativa tem de 15 a 255 caracteres"));
57    }
58    if ravenrs_dfe::Instante::de_xml(quando).is_none() {
59        return Err(NfseError::validacao("dhEvento", "use AAAA-MM-DDThh:mm:ss-03:00"));
60    }
61    let id = id_pedido(chave, CANCELAMENTO);
62    let tag = match autor {
63        Inscricao::Cnpj(_) => "CNPJAutor",
64        Inscricao::Cpf(_) => "CPFAutor",
65    };
66    let xml = format!(
67        "<pedRegEvento xmlns=\"{NS_NFSE}\" versao=\"{VERSAO}\"><infPedReg Id=\"{id}\"><tpAmb>2</tpAmb><verAplic>{VER_APLIC}</verAplic><dhEvento>{quando}</dhEvento><{tag}>{autor}</{tag}><chNFSe>{chave}</chNFSe><e{CANCELAMENTO}><xDesc>Cancelamento de NFS-e</xDesc><cMotivo>{motivo}</cMotivo><xMotivo>{justificativa}</xMotivo></e{CANCELAMENTO}></infPedReg></pedRegEvento>",
68        quando = escape_text(quando),
69        autor = autor.numero(),
70        chave = chave.as_str(),
71        motivo = motivo.codigo(),
72        justificativa = escape_text(justificativa),
73    );
74    Ok(ravenrs_dfe::assinatura::assinar_envelopado(&xml, "pedRegEvento", &id, signer, algoritmo)?)
75}
