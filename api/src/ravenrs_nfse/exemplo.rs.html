1//! A sample DPS, for tests, documentation and the sandbox. The CNPJs are the usual test ones.
2
3use rust_decimal::Decimal;
4
5use crate::modelo::{Dps, Inscricao, Prestador, Servico, Tomador};
6
7/// The CNPJ of the sample provider.
8pub const CNPJ_PRESTADOR: &str = "11222333000181";
9/// São Paulo, by its IBGE code.
10pub const MUNICIPIO: &str = "3550308";
11
12/// The sample provider: ME/EPP in São Paulo.
13#[must_use]
14pub fn prestador() -> Prestador {
15    Prestador {
16        inscricao: Inscricao::Cnpj(CNPJ_PRESTADOR.to_owned()),
17        inscricao_municipal: Some("12345678".to_owned()),
18        razao_social: "MERCADO EXEMPLO LTDA".to_owned(),
19        municipio: MUNICIPIO.to_owned(),
20    }
21}
22
23/// A DPS with this number: an installation service of R$ 150,00, to a company.
24#[must_use]
25pub fn dps(numero: u64) -> Dps {
26    Dps {
27        serie: 1,
28        numero,
29        emitida_em: "2026-11-06T10:00:00-03:00".to_owned(),
30        competencia: "2026-11-06".to_owned(),
31        prestador: prestador(),
32        tomador: Some(Tomador {
33            inscricao: Inscricao::Cnpj("11444777000161".to_owned()),
34            nome: "ATACADO EXEMPLO LTDA".to_owned(),
35            email: None,
36        }),
37        servico: Servico {
38            codigo_tributacao_nacional: "140101".to_owned(),
39            descricao: "Instalação e configuração de balança no caixa".to_owned(),
40            municipio_prestacao: MUNICIPIO.to_owned(),
41            valor: Decimal::new(15_000, 2),
42        },
43    }
44}
