17/// The code with which the Sefin says a DPS with this Id already exists. From a third-party
18/// catalog, not the official table: see `docs/NFSE.md`.
19pub const DPS_JA_EXISTE: &str = "E0014";
20
21/// An NFS-e issued.
22#[derive(Clone, Debug, PartialEq, Eq)]
23pub struct Emitida {
24    /// The Id of the DPS that became it.
25    pub id_dps: String,
26    /// Its access key.
27    pub chave: ChaveNfse,
28    /// The XML of the NFS-e, as the Sefin gave it.
29    pub xml: String,
30    /// The alerts that came with it.
31    pub alertas: Vec<Mensagem>,
32    /// Whether it came from a recovery: the answer of the send was lost, and the NFS-e was found
33    /// by the Id of the DPS.
34    pub recuperada: bool,
35}

87    /// The signature algorithm. The vigent one is not confirmed for the national standard: the only
88    /// official text is a historical manual with RSA-SHA1, and projects in use sign with SHA-256.
89    /// The default is SHA-1, like the NF-e, until the first real send settles it.
90    #[must_use]
91    pub fn algoritmo(mut self, algoritmo: Algoritmo) -> Self {
92        self.algoritmo = algoritmo;
93        self
94    }
95
96    /// The emitter, if the configuration can work.
97    pub fn build(self) -> Result<Emissor, NfseError> {
98        let signer = self.signer.ok_or_else(|| NfseError::Configuracao("falta quem assina".to_owned()))?;
99        let transport =
100            self.transport.ok_or_else(|| NfseError::Configuracao("falta o transporte".to_owned()))?;
101        match self.ambiente {
102            Ambiente::Sandbox if !transport.intercepta() => {
103                return Err(NfseError::Configuracao(
104                    "o sandbox só aceita um transporte que não sai do computador".to_owned(),
105                ));
106            }
107            Ambiente::ProducaoRestrita if eh_do_laboratorio(signer.certificate_der()) => {
108                return Err(NfseError::Configuracao(
109                    "o certificado do laboratório não vale fora do laboratório: use um A1 ou A3 ICP-Brasil"
110                        .to_owned(),
111                ));
112            }
113            _ => {}
114        }
115        Ok(Emissor { ambiente: self.ambiente, signer, transport, algoritmo: self.algoritmo })
116    }
117}

136fn cabecalhos_json() -> Vec<(String, String)> {
137    vec![
138        ("Content-Type".to_owned(), "application/json".to_owned()),
139        ("Accept".to_owned(), "application/json".to_owned()),
140    ]
141}

143/// Whether the answer was lost on the way: the DPS may or may not have become an NFS-e.
144fn incerto(erro: &NfseError) -> bool {
145    match erro {
146        NfseError::Transporte(TransportError::Timeout | TransportError::Conexao(_)) => true,
147        NfseError::RetornoInvalido { status, .. } => *status >= 500,
148        _ => false,
149    }
150}

174    fn enviar(&self, pedido: &HttpRequest) -> Result<HttpResponse, NfseError> {
175        Ok(self.transport.send(pedido)?)
176    }

178    /// The DPS as it would go, signed. Useful to look at, and to keep with the sale.
179    pub fn assinar(&self, dps: &Dps) -> Result<String, NfseError> {
180        dps::assinar(dps, self.signer.as_ref(), self.algoritmo)
181    }
182
183    /// Sends the DPS and returns the NFS-e.
184    ///
185    /// If the answer is lost (time out, connection cut, 5xx) or the Sefin says the DPS already
186    /// exists, it looks for the NFS-e by the Id of the DPS before saying anything. When it is not
187    /// found either, the error is [`NfseError::NaoConfirmada`]: send the same DPS again, never with
188    /// another number, or the same service gets two NFS-e.
189    pub fn emitir(&self, dps: &Dps) -> Result<Emitida, NfseError> {
190        let id_dps = dps::id_dps(dps);
191        let assinado = self.assinar(dps)?;
192        let corpo = json!({ "dpsXmlGZipB64": ravenrs_dfe::compactacao::gzip_base64(&assinado)? });
193        let pedido = HttpRequest::post(self.url("/nfse"), cabecalhos_json(), corpo.to_string().into_bytes());
194        let enviado = self.enviar(&pedido).and_then(|resposta| self.ler_emissao(&id_dps, &resposta));
195        match enviado {
196            Ok(emitida) => Ok(emitida),
197            Err(erro) if incerto(&erro) || erro.codigos().contains(&DPS_JA_EXISTE) => {
198                match self.recuperar(&id_dps) {
199                    Ok(Some(emitida)) => Ok(emitida),
200                    Ok(None) if erro.codigos().contains(&DPS_JA_EXISTE) => Err(erro),
201                    Ok(None) => Err(NfseError::NaoConfirmada { id_dps, motivo: erro.to_string() }),
202                    Err(consulta) => Err(NfseError::NaoConfirmada {
203                        id_dps,
204                        motivo: format!("{erro}; a consulta também falhou: {consulta}"),
205                    }),
206                }
207            }
208            Err(erro) => Err(erro),
209        }
210    }
211
212    fn ler_emissao(&self, id_dps: &str, resposta: &HttpResponse) -> Result<Emitida, NfseError> {
213        if let Some(recusa) = retorno::recusa(resposta) {
214            return Err(recusa);
215        }
216        if !(200..300).contains(&resposta.status) {
217            return Err(NfseError::RetornoInvalido {
218                status: resposta.status,
219                motivo: String::from_utf8_lossy(&resposta.body).chars().take(200).collect(),
220            });
221        }
222        let corpo = retorno::json(resposta)?;
223        let xml = retorno::documento(resposta, "nfseXmlGZipB64")?;
224        let chave = match retorno::texto(&corpo, &["chaveAcesso"]) {
225            Some(chave) => ChaveNfse::parse(&chave)?,
226            None => chave_do_xml(&xml)?,
227        };
228        Ok(Emitida {
229            id_dps: id_dps.to_owned(),
230            chave,
231            xml,
232            alertas: retorno::mensagens(&corpo, &["alertas"]),
233            recuperada: false,
234        })
235    }

237    /// Looks for the NFS-e of a DPS by its Id: `HEAD /dps/{id}`, then the key in `GET /dps/{id}`,
238    /// then the document in `GET /nfse/{chave}`. `None` means the DPS did not become an NFS-e.
239    pub fn recuperar(&self, id_dps: &str) -> Result<Option<Emitida>, NfseError> {
240        let caminho = format!("/dps/{id_dps}");
241        let existe = self.enviar(&HttpRequest::head(self.url(&caminho), cabecalhos_json()))?;
242        match existe.status {
243            404 => return Ok(None),
244            200..=299 => {}
245            status => {
246                return Err(NfseError::RetornoInvalido { status, motivo: "HEAD /dps inesperado".to_owned() });
247            }
248        }
249        let dados = self.enviar(&HttpRequest::get(self.url(&caminho), cabecalhos_json()))?;
250        if let Some(recusa) = retorno::recusa(&dados) {
251            return Err(recusa);
252        }
253        let chave = retorno::texto(&retorno::json(&dados)?, &["chaveAcesso"]).ok_or_else(|| {
254            NfseError::RetornoInvalido { status: dados.status, motivo: "chaveAcesso ausente".to_owned() }
255        })?;
256        let chave = ChaveNfse::parse(&chave)?;
257        let xml = self.consultar(&chave)?;
258        Ok(Some(Emitida { id_dps: id_dps.to_owned(), chave, xml, alertas: Vec::new(), recuperada: true }))
259    }

261    /// The XML of an NFS-e, by its key.
262    pub fn consultar(&self, chave: &ChaveNfse) -> Result<String, NfseError> {
263        let resposta =
264            self.enviar(&HttpRequest::get(self.url(&format!("/nfse/{chave}")), cabecalhos_json()))?;
265        if let Some(recusa) = retorno::recusa(&resposta) {
266            return Err(recusa);
267        }
268        retorno::documento(&resposta, "nfseXmlGZipB64")
269    }

271    /// Cancels an NFS-e, and returns the XML of the event registered.
272    pub fn cancelar(
273        &self,
274        chave: &ChaveNfse,
275        autor: &Inscricao,
276        motivo: MotivoCancelamento,
277        justificativa: &str,
278        quando: &str,
279    ) -> Result<String, NfseError> {
280        let pedido = evento::pedido_cancelamento(
281            chave,
282            autor,
283            motivo,
284            justificativa,
285            quando,
286            self.signer.as_ref(),
287            self.algoritmo,
288        )?;
289        let corpo =
290            json!({ "pedidoRegistroEventoXmlGZipB64": ravenrs_dfe::compactacao::gzip_base64(&pedido)? });
291        let resposta = self.enviar(&HttpRequest::post(
292            self.url(&format!("/nfse/{chave}/eventos")),
293            cabecalhos_json(),
294            corpo.to_string().into_bytes(),
295        ))?;
296        if let Some(recusa) = retorno::recusa(&resposta) {
297            return Err(recusa);
298        }
299        retorno::documento(&resposta, "eventoXmlGZipB64")
300    }

302    /// The agreement of a city with the national standard, as the ADN answers it. The shape of the
303    /// answer is not confirmed, so it comes as JSON.
304    pub fn convenio(&self, municipio: &str) -> Result<Value, NfseError> {
305        let url = format!("{}/parametros_municipais/{municipio}/convenio", self.ambiente.base_parametros());
306        let resposta = self.enviar(&HttpRequest::get(url, cabecalhos_json()))?;
307        if let Some(recusa) = retorno::recusa(&resposta) {
308            return Err(recusa);
309        }
310        retorno::json(&resposta)
311    }

314/// The key from the Id of `infNFSe`, when the answer does not bring it apart.
315fn chave_do_xml(xml: &str) -> Result<ChaveNfse, NfseError> {
316    let raiz = ravenrs_dfe::xml::parse(xml)?;
317    let id = raiz
318        .find("infNFSe")
319        .and_then(|info| info.attr("Id"))
320        .ok_or_else(|| NfseError::RetornoInvalido { status: 200, motivo: "NFS-e sem Id".to_owned() })?;
321    ChaveNfse::parse(id)
