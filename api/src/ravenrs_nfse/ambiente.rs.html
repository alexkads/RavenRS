10/// Proof that the code calling the emitter knows it is a beta. Its field is private, so it can
11/// only exist in code that says it out loud.
12///
13/// ```compile_fail,E0423
14/// let ack = ravenrs_nfse::BetaAck(());
15/// ```
16#[derive(Clone, Copy, Debug)]
17pub struct BetaAck(());
18
19impl BetaAck {
20    /// Acknowledges that this is a beta, not homologated and not for real fiscal documents.
21    #[must_use]
22    pub const fn i_understand_this_is_experimental() -> Self {
23        Self(())
24    }
25}

27/// Where the DPS goes. Production does not exist in this enum on purpose: `tpAmb` is always 2.
28#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
29pub enum Ambiente {
30    /// Nothing leaves the computer: the transport must intercept, like [`crate::sandbox::SefinSandbox`].
31    Sandbox,
32    /// The restricted production of the Sefin Nacional, the test environment of the national
33    /// standard. Needs a real ICP-Brasil certificate; the laboratory's is refused.
34    ProducaoRestrita,
35}
36
37impl Ambiente {
38    /// The base of the Sefin Nacional API.
39    ///
40    /// The sources disagree on the prefix (`/SefinNacional`, `/API/SefinNacional`); the routes are
41    /// matched from their end, so only the real one needs to be right. Not confirmed on a real send.
42    #[must_use]
43    pub const fn base_sefin(self) -> &'static str {
44        match self {
45            Self::Sandbox => "https://sefin.sandbox.ravenrs.invalid/SefinNacional",
46            Self::ProducaoRestrita => "https://sefin.producaorestrita.nfse.gov.br/SefinNacional",
47        }
48    }
49
50    /// The base of the parameters of the cities, at the ADN. Not confirmed on a real send.
51    #[must_use]
52    pub const fn base_parametros(self) -> &'static str {
53        match self {
54            Self::Sandbox => "https://adn.sandbox.ravenrs.invalid/parametrizacao",
55            Self::ProducaoRestrita => "https://adn.producaorestrita.nfse.gov.br/parametrizacao",
56        }
57    }
58}
