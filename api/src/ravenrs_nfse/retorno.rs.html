1//! Reading the answers of the Sefin: JSON around documents in GZip + Base64.
2//!
3//! The names of the fields (`chaveAcesso`, `nfseXmlGZipB64`, `eventoXmlGZipB64`, `erros` or `erro`,
4//! with `codigo` or `Codigo`) come from the code of two projects, not from an official source: the
5//! Swagger did not open. They are read with tolerance until a real answer is captured.
6
7use ravenrs_dfe::HttpResponse;
8use serde_json::Value;
9
10use crate::erro::{Mensagem, NfseError};
11
12/// The body of an answer as JSON.
13pub fn json(resposta: &HttpResponse) -> Result<Value, NfseError> {
14    serde_json::from_slice(&resposta.body).map_err(|_| NfseError::RetornoInvalido {
15        status: resposta.status,
16        motivo: "a resposta não é JSON".to_owned(),
17    })
18}
19
20/// A text field, by any of these names.
21#[must_use]
22pub fn texto(valor: &Value, nomes: &[&str]) -> Option<String> {
23    nomes.iter().find_map(|nome| valor.get(*nome)).and_then(Value::as_str).map(str::to_owned)
24}
25
26fn mensagem(valor: &Value) -> Mensagem {
27    Mensagem {
28        codigo: texto(valor, &["codigo", "Codigo"]).unwrap_or_else(|| "?".to_owned()),
29        descricao: texto(valor, &["descricao", "Descricao", "mensagem", "Mensagem"]).unwrap_or_default(),
30        complemento: texto(valor, &["complemento", "Complemento"]),
31    }
32}
33
34/// The messages under `erros`, `erro` or `alertas`.
35#[must_use]
36pub fn mensagens(valor: &Value, nomes: &[&str]) -> Vec<Mensagem> {
37    nomes
38        .iter()
39        .filter_map(|nome| valor.get(*nome))
40        .flat_map(|item| match item {
41            Value::Array(lista) => lista.iter().map(mensagem).collect(),
42            Value::Object(_) => vec![mensagem(item)],
43            _ => Vec::new(),
44        })
45        .collect()
46}
47
48/// The refusal of an answer, if it is one: status 4xx with messages.
49#[must_use]
50pub fn recusa(resposta: &HttpResponse) -> Option<NfseError> {
51    if !(400..500).contains(&resposta.status) {
52        return None;
53    }
54    let corpo = json(resposta).ok()?;
55    let erros = mensagens(&corpo, &["erros", "erro"]);
56    (!erros.is_empty()).then_some(NfseError::Rejeitada(erros))
57}
58
59/// A document in GZip + Base64 under this field.
60pub fn documento(resposta: &HttpResponse, campo: &str) -> Result<String, NfseError> {
61    let corpo = json(resposta)?;
62    let codificado = texto(&corpo, &[campo]).ok_or_else(|| NfseError::RetornoInvalido {
63        status: resposta.status,
64        motivo: format!("{campo} ausente"),
65    })?;
66    Ok(ravenrs_dfe::compactacao::de_gzip_base64(&codificado)?)
67}

69#[cfg(test)]
70mod tests {
71    use super::*;
72
73    fn resposta(status: u16, corpo: &str) -> HttpResponse {
74        HttpResponse { status, headers: Vec::new(), body: corpo.as_bytes().to_vec() }
75    }
76
77    #[test]
78    fn reads_errors_in_either_spelling() {
79        let a = recusa(&resposta(400, r#"{"erros":[{"Codigo":"E0014","Descricao":"DPS já existe"}]}"#));
80        let b = recusa(&resposta(400, r#"{"erro":{"codigo":"E0014","descricao":"DPS já existe"}}"#));
81        assert_eq!(a.as_ref().map(NfseError::codigos), Some(vec!["E0014"]));
82        assert_eq!(a, b);
83        assert_eq!(recusa(&resposta(201, "{}")), None);
84    }
