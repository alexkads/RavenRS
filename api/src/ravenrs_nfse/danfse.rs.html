1//! What the DANFSe has to have, from the NT 008 (v1.02, 14/07/2026): since 03/08/2026 the API of the
2//! DANFSe is suspended and the issuing software draws it. One page, portrait, at least A4, a QR Code
3//! of at least 1.52 cm, and "NFS-e SEM VALIDADE JURÍDICA" when `tpAmb=2`. The layout itself is a
4//! report of the standard designer (`ravenrs-app-nfe`), which anyone can change.
5
6use crate::chave::ChaveNfse;
7
8/// The notice the NT 008 asks for when `tpAmb=2`. This crate only issues with `tpAmb=2`.
9pub const AVISO_SEM_VALIDADE: &str = "NFS-e SEM VALIDADE JURÍDICA";
10
11/// The smallest side of the QR Code, in centimeters.
12pub const QR_MINIMO_CM: f32 = 1.52;
13
14/// The smallest side of the QR Code, in points of 1/72 inch.
15pub const QR_MINIMO_PONTOS: f32 = QR_MINIMO_CM / 2.54 * 72.0;
16
17/// The address the QR Code opens: the public query of the NFS-e by its key.
18///
19/// The NT 008 was read for the rules of the page, not for this address, which comes from the
20/// public query portal and is **not confirmed** for the QR Code. Check it before the first real use.
21#[must_use]
22pub fn conteudo_qr(chave: &ChaveNfse) -> String {
23    format!("https://www.nfse.gov.br/ConsultaPublica/?tpc=1&chave={chave}")
24}
25
26#[cfg(test)]
27mod tests {
28    use super::*;
29
30    #[test]
31    fn the_smallest_qr_is_about_43_points() {
32        assert!((QR_MINIMO_PONTOS - 43.09).abs() < 0.01);
33        let chave = ChaveNfse::parse(&"3".repeat(50)).expect("key");
34        assert!(conteudo_qr(&chave).ends_with(&"3".repeat(50)));
35    }
36}
