8/// A CNPJ or a CPF, as text.
9#[derive(Clone, Debug, PartialEq, Eq, Hash)]
10pub enum Inscricao {
11    /// Fourteen digits.
12    Cnpj(String),
13    /// Eleven digits.
14    Cpf(String),
15}
16
17impl Inscricao {
18    /// The digits.
19    #[must_use]
20    pub fn numero(&self) -> &str {
21        match self {
22            Self::Cnpj(numero) | Self::Cpf(numero) => numero,
23        }
24    }
25
26    /// The tag of the layout: `CNPJ` or `CPF`.
27    #[must_use]
28    pub const fn tag(&self) -> &'static str {
29        match self {
30            Self::Cnpj(_) => "CNPJ",
31            Self::Cpf(_) => "CPF",
32        }
33    }
34
35    /// The kind in the Id of the DPS: 1 for CPF, 2 for CNPJ.
36    #[must_use]
37    pub const fn tipo(&self) -> u8 {
38        match self {
39            Self::Cpf(_) => 1,
40            Self::Cnpj(_) => 2,
41        }
42    }
43
44    fn validar(&self, campo: &str) -> Result<(), NfseError> {
45        let (tamanho, nome) = match self {
46            Self::Cnpj(_) => (14, "CNPJ"),
47            Self::Cpf(_) => (11, "CPF"),
48        };
49        if self.numero().len() != tamanho || !self.numero().bytes().all(|b| b.is_ascii_digit()) {
50            return Err(NfseError::validacao(campo, format!("{nome} precisa ter {tamanho} dígitos")));
51        }
52        Ok(())
53    }
54}

56/// Who provides the service and issues the DPS.
57#[derive(Clone, Debug, PartialEq, Eq)]
58pub struct Prestador {
59    /// CNPJ or CPF.
60    pub inscricao: Inscricao,
61    /// The municipal registration, when the city has one.
62    pub inscricao_municipal: Option<String>,
63    /// The name, for the DANFSe. The DPS does not carry it: the Sefin takes it from the register.
64    pub razao_social: String,
65    /// The IBGE code of the city that issues, `cLocEmi`.
66    pub municipio: String,
67}

69/// Who takes the service. Optional for a consumer who does not identify.
70#[derive(Clone, Debug, PartialEq, Eq)]
71pub struct Tomador {
72    /// CNPJ or CPF.
73    pub inscricao: Inscricao,
74    /// The name.
75    pub nome: String,
76    /// Where the NFS-e is sent, if anywhere.
77    pub email: Option<String>,
78}

80/// The service.
81#[derive(Clone, Debug, PartialEq, Eq)]
82pub struct Servico {
83    /// `cTribNac`: item, subitem and national breakdown of the list of services, six digits.
84    pub codigo_tributacao_nacional: String,
85    /// What was done, up to 2000 characters.
86    pub descricao: String,
87    /// The IBGE code of the city where it was done.
88    pub municipio_prestacao: String,
89    /// The value, in reais.
90    pub valor: Decimal,
91}

93/// A declaration of service provided, before it becomes an NFS-e.
94#[derive(Clone, Debug, PartialEq, Eq)]
95pub struct Dps {
96    /// The series, 1 to 99999.
97    pub serie: u32,
98    /// The number, never repeated in the series. After a lost answer, the same one goes again.
99    pub numero: u64,
100    /// When it was issued: `2026-11-06T10:00:00-03:00`.
101    pub emitida_em: String,
102    /// The month of competence, as its first day or any day of it: `2026-11-06`.
103    pub competencia: String,
104    /// Who provides.
105    pub prestador: Prestador,
106    /// Who takes, if identified.
107    pub tomador: Option<Tomador>,
108    /// The service.
109    pub servico: Servico,
110}

112fn digitos(valor: &str, tamanho: usize) -> bool {
113    valor.len() == tamanho && valor.bytes().all(|b| b.is_ascii_digit())
114}
115
116impl Dps {
117    /// Checks what can be checked before sending.
118    pub fn validar(&self) -> Result<(), NfseError> {
119        if !(1..=99_999).contains(&self.serie) {
120            return Err(NfseError::validacao("serie", "a série vai de 1 a 99999"));
121        }
122        if !(1..=999_999_999_999_999).contains(&self.numero) {
123            return Err(NfseError::validacao("nDPS", "o número vai de 1 a 999999999999999"));
124        }
125        if Instante::de_xml(&self.emitida_em).is_none() {
126            return Err(NfseError::validacao("dhEmi", "use AAAA-MM-DDThh:mm:ss-03:00"));
127        }
128        let competencia = self.competencia.as_bytes();
129        if competencia.len() != 10 || competencia.get(4) != Some(&b'-') || competencia.get(7) != Some(&b'-') {
130            return Err(NfseError::validacao("dCompet", "use AAAA-MM-DD"));
131        }
132        self.prestador.inscricao.validar("prest")?;
133        if !digitos(&self.prestador.municipio, 7) {
134            return Err(NfseError::validacao("cLocEmi", "o código IBGE do município tem 7 dígitos"));
135        }
136        if let Some(im) = &self.prestador.inscricao_municipal
137            && (im.is_empty() || im.len() > 15)
138        {
139            return Err(NfseError::validacao("IM", "a inscrição municipal tem de 1 a 15 caracteres"));
140        }
141        if let Some(tomador) = &self.tomador {
142            tomador.inscricao.validar("toma")?;
143            if tomador.nome.trim().is_empty() || tomador.nome.chars().count() > 300 {
144                return Err(NfseError::validacao("toma/xNome", "o nome tem de 1 a 300 caracteres"));
145            }
146        }
147        let servico = &self.servico;
148        if !digitos(&servico.codigo_tributacao_nacional, 6) {
149            return Err(NfseError::validacao("cTribNac", "o código de tributação nacional tem 6 dígitos"));
150        }
151        if !digitos(&servico.municipio_prestacao, 7) {
152            return Err(NfseError::validacao("cLocPrestacao", "o código IBGE do município tem 7 dígitos"));
153        }
154        let descricao = servico.descricao.trim();
155        if descricao.is_empty() || descricao.chars().count() > 2000 {
156            return Err(NfseError::validacao("xDescServ", "a descrição tem de 1 a 2000 caracteres"));
157        }
158        if servico.valor <= Decimal::ZERO {
159            return Err(NfseError::validacao("vServ", "o valor do serviço precisa ser positivo"));
160        }
161        if servico.valor.round_dp_with_strategy(2, RoundingStrategy::MidpointAwayFromZero) != servico.valor {
162            return Err(NfseError::validacao("vServ", "o valor tem no máximo 2 casas decimais"));
163        }
164        Ok(())
165    }
