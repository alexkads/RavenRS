1//! A sandbox of the Sefin Nacional, in the style of MSW: it implements [`Transport`], so the
2//! emitter builds the same bytes and calls the same addresses, and nothing leaves the computer.
3//!
4//! | MSW | Here |
5//! |---|---|
6//! | `http.post('/nfse', resolver)` | [`post`], [`get`], [`head`] with a [`Rota`] |
7//! | `server.use(handler)` | [`SefinSandbox::use_handler`] |
8//! | `{ once: true }` | [`SefinSandbox::use_once`] |
9//! | `server.resetHandlers()` | [`SefinSandbox::reset_handlers`] |
10//! | `onUnhandledRequest` | [`SefinSandbox::on_unhandled`] |
11//!
12//! With no handler, it answers as the Sefin would, after checking for real: GZip and Base64, the
13//! namespace and version, the Id against the fields it is made of, `tpAmb=2`, the signature with
14//! the hash it was made with, a DPS already used, and the agreement of the city. Each refusal comes
15//! with its code, where the code came from and how sure that is ([`regra`]).
16//!
17//! The NFS-e it gives back is a reduced one: `infNFSe` with the Id, the number, the date of
18//! processing and the DPS received. It does not follow the whole `TCInfNFSe`.
19

147/// What a handler decides.
148#[derive(Clone, Debug)]
149pub enum Resposta {
150    /// Answers this.
151    Http(HttpResponse),
152    /// Behaves as if the time limit ran out, with nothing processed.
153    Timeout,
154    /// Processes the request as usual and then loses the answer: the NFS-e exists, the sender
155    /// does not know. This is what recovery by the Id is for.
156    ProcessarEPerder,
157    /// Fails with this error.
158    Erro(TransportError),
159    /// Leaves it to the next handler, and in the end to the default behavior.
160    Proximo,
161}

172impl PedidoSefin {
173    /// A part of the route, by its name.
174    #[must_use]
175    pub fn parametro(&self, nome: &str) -> Option<&str> {
176        self.parametros.iter().find(|(chave, _)| chave == nome).map(|(_, valor)| valor.as_str())
177    }

180/// A rule that answers a kind of request.
181#[derive(Clone)]
182pub struct Handler {
183    rota: Rota,
184    responder: Arc<dyn Fn(&PedidoSefin) -> Resposta + Send + Sync>,
185}

187impl std::fmt::Debug for Handler {
188    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
189        f.debug_struct("Handler").field("rota", &self.rota.to_string()).finish_non_exhaustive()
190    }
191}

193fn handler(
194    metodo: Metodo,
195    caminho: &str,
196    f: impl Fn(&PedidoSefin) -> Resposta + Send + Sync + 'static,
197) -> Handler {
198    Handler { rota: Rota::new(metodo, caminho), responder: Arc::new(f) }
199}

305fn json_resposta(status: u16, corpo: &Value) -> HttpResponse {
306    HttpResponse {
307        status,
308        headers: vec![("Content-Type".to_owned(), "application/json".to_owned())],
309        body: corpo.to_string().into_bytes(),
310    }
311}
312
313fn recusa(status: u16, codigo: &str, complemento: Option<String>) -> HttpResponse {
314    let descricao = regra(codigo).map_or("Recusada", |regra| regra.descricao);
315    let mut erro = json!({ "Codigo": codigo, "Descricao": format!("{PREFIXO}{descricao}") });
316    if let Some(complemento) = complemento {
317        erro["Complemento"] = Value::String(complemento);
318    }
319    json_resposta(status, &json!({ "tipoAmbiente": 2, "versaoAplicativo": VER_APLIC, "erros": [erro] }))
320}
321
322fn codigos(resposta: &HttpResponse) -> Vec<String> {
323    let Ok(corpo) = serde_json::from_slice::<Value>(&resposta.body) else { return Vec::new() };
324    ["erros", "erro"]
325        .iter()
326        .filter_map(|nome| corpo.get(*nome))
327        .flat_map(|item| match item {
328            Value::Array(lista) => lista.clone(),
329            outro => vec![outro.clone()],
330        })
331        .filter_map(|erro| {
332            erro.get("Codigo").or_else(|| erro.get("codigo")).and_then(Value::as_str).map(str::to_owned)
333        })
334        .collect()
335}

420    fn rodar_handlers(&self, http: &HttpRequest) -> Option<Resposta> {
421        let candidatos = self.lock().handlers.candidatos(|handler| handler.rota.casar(http).is_some());
422        for (id, once, handler) in candidatos {
423            if once && !self.lock().handlers.consumir(id) {
424                continue;
425            }
426            let pedido =
427                PedidoSefin { http: http.clone(), parametros: handler.rota.casar(http).unwrap_or_default() };
428            let resposta = (handler.responder)(&pedido);
429            if !matches!(resposta, Resposta::Proximo) {
430                return Some(resposta);
431            }
432        }
433        None
434    }

436    fn padrao(&self, http: &HttpRequest) -> Option<HttpResponse> {
437        let rotas: [(Metodo, &str); 6] = [
438            (Metodo::Post, "/nfse"),
439            (Metodo::Head, "/dps/{id}"),
440            (Metodo::Get, "/dps/{id}"),
441            (Metodo::Get, "/nfse/{chave}"),
442            (Metodo::Post, "/nfse/{chave}/eventos"),
443            (Metodo::Get, "/parametros_municipais/{municipio}/convenio"),
444        ];
445        let (indice, parametros) = rotas.iter().enumerate().find_map(|(indice, (metodo, caminho))| {
446            Rota::new(*metodo, caminho).casar(http).map(|parametros| (indice, parametros))
447        })?;
448        let parametro = |nome: &str| {
449            parametros
450                .iter()
451                .find(|(chave, _)| chave == nome)
452                .map(|(_, valor)| valor.clone())
453                .unwrap_or_default()
454        };
455        Some(match indice {
456            0 => self.receber_dps(http),
457            1 => {
458                let existe = self.lock().por_dps.contains_key(&parametro("id"));
459                HttpResponse { status: if existe { 200 } else { 404 }, headers: Vec::new(), body: Vec::new() }
460            }
461            2 => match self.lock().por_dps.get(&parametro("id")) {
462                Some(nota) => {
463                    json_resposta(200, &json!({ "idDps": parametro("id"), "chaveAcesso": nota.chave }))
464                }
465                None => recusa(404, "SANDBOX-NAO-ENCONTRADA", None),
466            },
467            3 => self.devolver_nfse(&parametro("chave")),
468            4 => self.receber_evento(http, &parametro("chave")),
469            _ => {
470                let municipio = parametro("municipio");
471                let aderente = !self.lock().sem_convenio.contains(&municipio);
472                json_resposta(
473                    200,
474                    &json!({ "codigoMunicipio": municipio, "aderente": aderente, "origem": VER_APLIC }),
475                )
476            }
477        })
478    }

480    fn devolver_nfse(&self, chave: &str) -> HttpResponse {
481        let estado = self.lock();
482        let nota = estado.dps_da_chave.get(chave).and_then(|id| estado.por_dps.get(id));
483        match nota.and_then(|nota| gzip_base64(&nota.xml).ok().map(|zip| (nota, zip))) {
484            Some((nota, zip)) => {
485                json_resposta(200, &json!({ "chaveAcesso": nota.chave, "nfseXmlGZipB64": zip }))
486            }
487            None => recusa(404, "SANDBOX-NAO-ENCONTRADA", None),
488        }
489    }

491    fn documento(http: &HttpRequest, campo: &str) -> Result<(Element, String), HttpResponse> {
492        let corpo: Value = serde_json::from_slice(&http.body)
493            .map_err(|_| recusa(400, "SANDBOX-CONTEUDO", Some("o corpo não é JSON".to_owned())))?;
494        let zip = corpo
495            .get(campo)
496            .and_then(Value::as_str)
497            .ok_or_else(|| recusa(400, "SANDBOX-CONTEUDO", Some(format!("{campo} ausente"))))?;
498        let texto =
499            de_gzip_base64(zip).map_err(|erro| recusa(400, "SANDBOX-CONTEUDO", Some(erro.to_string())))?;
500        let raiz =
501            xml::parse(&texto).map_err(|erro| recusa(400, "SANDBOX-LEIAUTE", Some(erro.to_string())))?;
502        Ok((raiz, texto))
503    }

505    fn conferir_assinatura(&self, raiz: &Element, id: &str) -> Result<(), HttpResponse> {
506        let aceitos = self.lock().aceitos.clone();
507        verificar_envelopado(raiz, id, &aceitos).map(|_| ()).map_err(|falha| {
508            let motivo = match falha {
509                FalhaAssinatura::ForaDoPadrao(motivo) => format!("fora do padrão: {motivo}"),
510                FalhaAssinatura::Difere(motivo) => format!("não confere: {motivo}"),
511            };
512            recusa(400, "E0714", Some(motivo))
513        })
514    }

516    fn receber_dps(&self, http: &HttpRequest) -> HttpResponse {
517        match self.conferir_dps(http) {
518            Ok(resposta) | Err(resposta) => resposta,
519        }

522    fn conferir_dps(&self, http: &HttpRequest) -> Result<HttpResponse, HttpResponse> {
523        let (raiz, dps_recebida) = Self::documento(http, "dpsXmlGZipB64")?;
524        let leiaute = |motivo: &str| recusa(400, "SANDBOX-LEIAUTE", Some(motivo.to_owned()));
525        if raiz.local_name() != "DPS" || raiz.attr("xmlns") != Some(NS_NFSE) {
526            return Err(leiaute("raiz DPS no namespace do padrão nacional"));
527        }
528        if !matches!(raiz.attr("versao"), Some("1.00" | "1.01")) {
529            return Err(leiaute("versao 1.00 ou 1.01"));
530        }
531        let info = raiz.child("infDPS").ok_or_else(|| leiaute("infDPS ausente"))?;
532        let id = info.attr("Id").unwrap_or_default().to_owned();
533        let campo = |nome: &str| info.child_text(nome).unwrap_or_default();
534        let prest = info.child("prest").ok_or_else(|| leiaute("prest ausente"))?;
535        let (tipo, inscricao) = match (prest.child_text("CNPJ"), prest.child_text("CPF")) {
536            (Some(cnpj), _) => (2, cnpj),
537            (None, Some(cpf)) => (1, cpf),
538            _ => return Err(leiaute("prest sem CNPJ nem CPF")),
539        };
540        let serie: u32 = campo("serie").parse().unwrap_or(0);
541        let numero: u64 = campo("nDPS").parse().unwrap_or(0);
542        let municipio = campo("cLocEmi");
543        let esperado = format!("DPS{municipio}{tipo}{inscricao:0>14}{serie:05}{numero:015}");
544        if id.len() != 45 || id != esperado {
545            return Err(recusa(400, "E0004", Some(format!("esperado {esperado}"))));
546        }
547        if campo("tpAmb") != "2" {
548            return Err(recusa(400, "E0006", Some(format!("tpAmb={}", campo("tpAmb")))));
549        }
550        let agora = self.lock().agora.clone();
551        let (emissao, processamento) =
552            (ravenrs_dfe::Instante::de_xml(&campo("dhEmi")), ravenrs_dfe::Instante::de_xml(&agora));
553        match (emissao, processamento) {
554            (Some(emissao), Some(processamento)) if emissao > processamento => {
555                return Err(recusa(
556                    400,
557                    "E0008",
558                    Some(format!("dhEmi {} depois de {agora}", campo("dhEmi"))),
559                ));
560            }
561            (None, _) => return Err(leiaute("dhEmi fora do formato")),
562            _ => {}
563        }
564        self.conferir_assinatura(&raiz, &id)?;
565        if self.lock().sem_convenio.contains(&municipio) {
566            return Err(recusa(400, "E0037", Some(municipio)));
567        }
568        if self.lock().por_dps.contains_key(&id) {
569            return Err(recusa(400, "E0014", Some(id)));
570        }
571        Ok(self.gerar_nfse(&id, &municipio, tipo, &inscricao, &dps_recebida, &agora))
572    }
573
574    fn gerar_nfse(
575        &self,
576        id_dps: &str,
577        municipio: &str,
578        tipo: u8,
579        inscricao: &str,
580        dps: &str,
581        agora: &str,
582    ) -> HttpResponse {
583        let mut estado = self.lock();
584        let numero = estado.proximo_numero;
585        estado.proximo_numero += 1;
586        let ano_mes = format!("{}{}", agora.get(2..4).unwrap_or("26"), agora.get(5..7).unwrap_or("01"));
587        // Generating environment 2: the Sefin Nacional. Numeric code from the number, for repeatable tests.
588        let sem_dv = format!(
589            "{municipio}2{tipo}{inscricao:0>14}{numero:013}{ano_mes}{:09}",
590            numero * 7919 % 1_000_000_000
591        );
592        let chave = format!("{sem_dv}{}", dv_modulo_11(&sem_dv));
593        // The DPS goes inside as received, without its XML declaration if it had one.
594        let dps = dps.trim_start_matches(|c| c != '<').to_owned();
595        let xml = format!(
596            "<NFSe xmlns=\"{NS_NFSE}\" versao=\"{VERSAO}\"><infNFSe Id=\"NFS{chave}\"><xLocEmi>{}</xLocEmi><ambGer>2</ambGer><nNFSe>{numero}</nNFSe><dhProc>{agora}</dhProc><cStat>100</cStat>{dps}</infNFSe></NFSe>",
597            escape_text("SANDBOX - SEM VALIDADE JURÍDICA"),
598        );
599        let resposta = match gzip_base64(&xml) {
600            Ok(zip) => json_resposta(
601                201,
602                &json!({
603                    "tipoAmbiente": 2,
604                    "versaoAplicativo": VER_APLIC,
605                    "dataHoraProcessamento": agora,
606                    "idDps": id_dps,
607                    "chaveAcesso": chave,
608                    "nfseXmlGZipB64": zip,
609                    "alertas": [{ "Codigo": "SANDBOX", "Descricao": format!("{PREFIXO}NFS-e sem validade jurídica") }],
610                }),
611            ),
612            Err(erro) => recusa(500, "SANDBOX-CONTEUDO", Some(erro.to_string())),
613        };
614        estado.dps_da_chave.insert(chave.clone(), id_dps.to_owned());
615        estado.por_dps.insert(id_dps.to_owned(), Nota { chave, xml, cancelada: false });
616        resposta
617    }

619    fn receber_evento(&self, http: &HttpRequest, chave: &str) -> HttpResponse {
620        let conferido = (|| -> Result<HttpResponse, HttpResponse> {
621            let (raiz, _) = Self::documento(http, "pedidoRegistroEventoXmlGZipB64")?;
622            let leiaute = |motivo: &str| recusa(400, "SANDBOX-LEIAUTE", Some(motivo.to_owned()));
623            if raiz.local_name() != "pedRegEvento" || raiz.attr("xmlns") != Some(NS_NFSE) {
624                return Err(leiaute("raiz pedRegEvento no namespace do padrão nacional"));
625            }
626            let info = raiz.child("infPedReg").ok_or_else(|| leiaute("infPedReg ausente"))?;
627            let id = info.attr("Id").unwrap_or_default().to_owned();
628            if info.child_text("chNFSe").as_deref() != Some(chave) {
629                return Err(leiaute("chNFSe difere da chave do caminho"));
630            }
631            if id != format!("PRE{chave}{CANCELAMENTO}") || info.child(&format!("e{CANCELAMENTO}")).is_none()
632            {
633                return Err(leiaute("só o cancelamento e101101, com Id PRE + chave + 101101"));
634            }
635            self.conferir_assinatura(&raiz, &id)?;
636            let mut estado = self.lock();
637            let Some(id_dps) = estado.dps_da_chave.get(chave).cloned() else {
638                return Err(recusa(404, "SANDBOX-NAO-ENCONTRADA", None));
639            };
640            let agora = estado.agora.clone();
641            let Some(nota) = estado.por_dps.get_mut(&id_dps) else {
642                return Err(recusa(404, "SANDBOX-NAO-ENCONTRADA", None));
643            };
644            if nota.cancelada {
645                return Err(recusa(400, "E0840", None));
646            }
647            nota.cancelada = true;
648            let evento = format!(
649                "<evento xmlns=\"{NS_NFSE}\" versao=\"{VERSAO}\"><infEvento Id=\"EVT{chave}{CANCELAMENTO}001\"><verAplic>{VER_APLIC}</verAplic><ambGer>2</ambGer><nSeqEvento>1</nSeqEvento><dhProc>{agora}</dhProc><chNFSe>{chave}</chNFSe><tpEvento>{CANCELAMENTO}</tpEvento></infEvento></evento>"
650            );
651            let zip = gzip_base64(&evento)
652                .map_err(|erro| recusa(500, "SANDBOX-CONTEUDO", Some(erro.to_string())))?;
653            Ok(json_resposta(
654                201,
655                &json!({ "tipoAmbiente": 2, "versaoAplicativo": VER_APLIC, "eventoXmlGZipB64": zip }),
656            ))
657        })();
658        match conferido {
659            Ok(resposta) | Err(resposta) => resposta,
660        }
661    }
662
663    fn nao_tratado(&self, http: &HttpRequest) -> Result<HttpResponse, TransportError> {
664        let regra = self.lock().on_unhandled.clone();
665        let acao = format!("{} {}", http.metodo.nome(), http.caminho());
666        match regra {
667            OnUnhandled::Error => Err(TransportError::NaoTratado { url: http.url.clone(), acao: Some(acao) }),
668            OnUnhandled::Warn => {
669                self.lock().avisos.push(format!("pedido não tratado: {acao}"));
670                Ok(recusa(404, "SANDBOX-NAO-ENCONTRADA", Some(acao)))
671            }
672            OnUnhandled::Passthrough(transport) => transport.send(http),
673        }
674    }
675
676    fn responder(&self, http: &HttpRequest) -> Result<HttpResponse, TransportError> {
677        match self.rodar_handlers(http) {
678            Some(Resposta::Http(resposta)) => Ok(resposta),
679            Some(Resposta::Timeout) => Err(TransportError::Timeout),
680            Some(Resposta::Erro(erro)) => Err(erro),
681            Some(Resposta::ProcessarEPerder) => {
682                drop(self.padrao(http));
683                Err(TransportError::Timeout)
684            }
685            Some(Resposta::Proximo) | None => match self.padrao(http) {
686                Some(resposta) => Ok(resposta),
687                None => self.nao_tratado(http),
688            },
689        }
690    }
691}
692
693impl Transport for SefinSandbox {
694    fn send(&self, request: &HttpRequest) -> Result<HttpResponse, TransportError> {
695        let resultado = self.responder(request);
696        let registro = Registro {
697            http: request.clone(),
698            status: resultado.as_ref().ok().map(|resposta| resposta.status),
699            codigos: resultado.as_ref().map(codigos).unwrap_or_default(),
700        };
701        self.lock().registros.push(registro);
702        resultado
703    }
704
705    fn intercepta(&self) -> bool {
706        true
707    }
