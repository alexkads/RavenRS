15/// Where the records are kept, unless `RAVEN_PDV` says otherwise.
16pub fn arquivo_padrao() -> PathBuf {
17    let casa = std::env::var_os("HOME")
18        .or_else(|| std::env::var_os("USERPROFILE"))
19        .map_or_else(PathBuf::new, PathBuf::from);
20    casa.join("mini-pdv-beta.sqlite")
21}

34/// The sample records, the first time only.
35fn semear(space: &mut dyn ObjectSpace) -> Result<(), String> {
36    let erro = |error: ravenrs_app::SpaceError| error.to_string();
37    if space.query(PRODUTO, &Query::all()).map_err(erro)?.is_empty() {
38        for (codigo, gtin, descricao, preco, ncm) in PRODUTOS {
39            let mut produto = space.new_object(PRODUTO).map_err(erro)?;
40            produto.set("codigo", codigo.to_owned());
41            produto.set("gtin", gtin.to_owned());
42            produto.set("descricao", descricao.to_owned());
43            produto.set("preco", preco);
44            produto.set("ncm", ncm.to_owned());
45            space.save(produto).map_err(erro)?;
46        }
47    }
48    if space.query(NFE_EMITENTE, &Query::all()).map_err(erro)?.is_empty() {
49        let mut emitente = space.new_object(NFE_EMITENTE).map_err(erro)?;
50        for (campo, valor) in [
51            ("razao_social", "MERCADO EXEMPLO LTDA"),
52            ("nome_fantasia", "Mercado Exemplo"),
53            ("cnpj", "11222333000181"),
54            ("inscricao_estadual", "110042490114"),
55            ("inscricao_municipal", "12345678"),
56            ("logradouro", "Rua de Exemplo"),
57            ("numero", "100"),
58            ("bairro", "Centro"),
59            ("codigo_municipio", "3550308"),
60            ("municipio", "Sao Paulo"),
61            ("uf", "SP"),
62            ("cep", "01001000"),
63        ] {
64            emitente.set(campo, valor.to_owned());
65        }
66        space.save(emitente).map_err(erro)?;
67    }
68    if space.query(SERVICO, &Query::all()).map_err(erro)?.is_empty() {
69        // `cTribNac` of the national list: item, subitem and breakdown. Check them with an accountant.
70        for (codigo, descricao, ctrib, preco) in [
71            ("S1", "Instalação e configuração de balança", "140101", 15_000.0),
72            ("S2", "Manutenção de equipamento de cozinha", "140101", 9_000.0),
73        ] {
74            let mut servico = space.new_object(SERVICO).map_err(erro)?;
75            servico.set("codigo", codigo.to_owned());
76            servico.set("descricao", descricao.to_owned());
77            servico.set("codigo_tributacao_nacional", ctrib.to_owned());
78            servico.set("preco", preco);
79            space.save(servico).map_err(erro)?;
80        }
81    }
82    if space.query(CLIENTE, &Query::all()).map_err(erro)?.is_empty() {
83        let mut cliente = space.new_object(CLIENTE).map_err(erro)?;
84        for (campo, valor) in [
85            ("nome", "RESTAURANTE EXEMPLO LTDA"),
86            ("documento", "11444777000161"),
87            ("logradouro", "Avenida de Exemplo"),
88            ("numero", "200"),
89            ("bairro", "Centro"),
90            ("codigo_municipio", "3550308"),
91            ("municipio", "Sao Paulo"),
92            ("uf", "SP"),
93            ("cep", "01002000"),
94        ] {
95            cliente.set(campo, valor.to_owned());
96        }
97        space.save(cliente).map_err(erro)?;
98    }
99    if space.query(CONFIGURACAO, &Query::all()).map_err(erro)?.is_empty() {
100        let mut configuracao = space.new_object(CONFIGURACAO).map_err(erro)?;
101        configuracao.set("operador", "Maria".to_owned());
102        space.save(configuracao).map_err(erro)?;
103    }
104    space.commit().map_err(|error| error.to_string())
105}

107/// The application over this file, or in memory with none, with its point of sale.
108pub fn app_com_pos(arquivo: Option<&Path>) -> Result<(AnyView, Pos), String> {
109    let service =
110        NfeService::sandbox(BetaAck::i_understand_this_is_experimental()).map_err(|e| e.to_string())?;
111    let pos = Pos::new(service);
112    let mut app = pos.instalar(App::new("Mini-PDV")).seed(semear);
113    if let Some(arquivo) = arquivo {
114        let conexao = rusqlite::Connection::open(arquivo).map_err(|e| e.to_string())?;
115        let mut builder = SqliteDatabase::builder(Rc::new(conexao));
116        for entidade in entidades() {
117            let tabela = TableMap::new(entidade.name.to_lowercase());
118            builder = builder.managed(entidade, tabela).map_err(|error| error.0)?;
119        }
120        let database = builder.build();
121        app = app.storage(move || database.space());
122    }
123    app.build().map(|view| (view, pos)).map_err(|error| error.0)
124}

126/// Puts a purchase from a supplier at the sandbox, addressed to the sample store, so that
127/// "Buscar na SEFAZ" on the entry of goods has something to find. Nothing of it leaves the computer.
128pub fn compra_de_exemplo(pos: &Pos) -> Result<(), String> {
129    use ravenrs_app_nfe::engine::sandbox::{Fornecedor, gerar_nota_de_compra};
130    use ravenrs_app_nfe::engine::{Csosn, Item};
131    let Some(sandbox) = pos.caixa().service.sandbox_ativo() else { return Ok(()) };
132    let item = |codigo: &str, gtin: &str, descricao: &str, ncm: &str, quantidade: &str, valor: &str| {
133        Ok::<_, String>(Item {
134            codigo: codigo.to_owned(),
135            gtin: Some(gtin.to_owned()),
136            descricao: descricao.to_owned(),
137            ncm: ncm.to_owned(),
138            cest: None,
139            cfop: "5102".to_owned(),
140            unidade: "UN".to_owned(),
141            quantidade: quantidade.parse().map_err(|error| format!("{error}"))?,
142            valor_unitario: valor.parse().map_err(|error| format!("{error}"))?,
143            desconto: Default::default(),
144            csosn: Csosn::S102,
145            origem: 0,
146        })
147    };
148    let fornecedor = Fornecedor::novo("ATACADO EXEMPLO LTDA", "11444777000161", "11222333000181");
149    let itens = vec![
150        item("AT-ARROZ", "7891000053508", "ARROZ BRANCO T1 5KG", "10063021", "10", "19.90")?,
151        item("AT-CAFE", "7891000315507", "CAFE TORRADO 500G", "09012100", "12", "12.40")?,
152    ];
153    let xml = gerar_nota_de_compra(&fornecedor, itens).map_err(|error| error.to_string())?;
154    sandbox.colocar_nota_recebida(&xml).map_err(|error| error.to_string())
155}

157/// The window. What goes wrong on the way shows on it.
158pub fn app(arquivo: Option<&Path>) -> AnyView {
159    match app_com_pos(arquivo).and_then(|(view, pos)| compra_de_exemplo(&pos).map(|()| (view, pos))) {
160        Ok((view, _)) => view,
161        Err(why) => Box::new(Alert::new("Mini-PDV", why).tone(Tone::Danger)),
162    }
