1//! An automation builder for chat conversations, made only with `FlowCanvas`, `FlowToolbox`
2//! and `FlowInspector`: a trigger starts the conversation, messages offer buttons, each
3//! button leads somewhere, conditions split the way and actions reach other systems.
4//!
5//! It validates the component as a builder of this kind: everything here is the library's,
6//! and what it could not do is written in the README.
7

13/// The kinds of block, with the icon and the word the canvas shows on each.
14pub const KINDS: [(&str, &str); 4] =
15    [("trigger", "Gatilho"), ("message", "Mensagem"), ("logic", "Lógica"), ("action", "Ação")];
16
17fn trigger(id: &str, title: &str, line: &str) -> FlowNode {
18    FlowNode::new(id, "trigger", title, 0.0, 0.0).subtitle(line).port(FlowPort::output("out"))
19}
20
21fn message(id: &str, title: &str, text: &str) -> FlowNode {
22    FlowNode::new(id, "message", title, 0.0, 0.0)
23        .subtitle(text)
24        .port(FlowPort::input("in"))
25        .port(FlowPort::output("out"))
26}
27
28/// A message with a button per answer: each answer is an output of its own, so each leads
29/// somewhere else.
30fn buttons(id: &str, title: &str, answers: &[&str]) -> FlowNode {
31    let mut node = FlowNode::new(id, "message", title, 0.0, 0.0)
32        .size(240.0, 64.0 + 24.0 * answers.len() as f32)
33        .port(FlowPort::input("in"));
34    for answer in answers {
35        node = node.port(FlowPort::output(answer.to_lowercase()).label(*answer));
36    }
37    node
38}
39
40fn condition(id: &str, title: &str) -> FlowNode {
41    FlowNode::new(id, "logic", title, 0.0, 0.0)
42        .shape(FlowShape::Diamond)
43        .size(176.0, 104.0)
44        .port(FlowPort::input("in").side(PortSide::Left))
45        .port(FlowPort::output("sim").label("Sim").side(PortSide::Right))
46        .port(FlowPort::output("não").label("Não").side(PortSide::Bottom))
47}
48
49fn action(id: &str, title: &str, line: &str) -> FlowNode {
50    FlowNode::new(id, "action", title, 0.0, 0.0).subtitle(line).port(FlowPort::input("in"))
51}
52
53fn at(node: FlowNode, x: f32, y: f32) -> FlowNode {
54    FlowNode { x, y, ..node }
55}
56
57fn link(id: &str, from: (&str, &str), to: &str) -> FlowEdge {
58    FlowEdge::new(id, FlowEndpoint::new(from.0, from.1), FlowEndpoint::new(to, "in"))
59}

61/// A conversation that welcomes, asks what the person wants and takes each answer its way.
62pub fn sample() -> FlowGraph {
63    FlowGraph::new()
64        .node(at(trigger("start", "Palavra-chave", "“oi”, “olá”, “menu”"), 0.0, 120.0))
65        .node(at(buttons("menu", "Boas-vindas", &["Pedidos", "Horários", "Atendente"]), 288.0, 96.0))
66        .node(at(message("hours", "Horários", "Seg a sex, 8h às 18h"), 600.0, 0.0))
67        .node(at(condition("open", "Dentro do horário?"), 600.0, 168.0))
68        .node(at(action("human", "Avisar a equipe", "Canal #atendimento"), 888.0, 168.0))
69        .node(at(message("later", "Fora do horário", "Respondemos amanhã cedo"), 600.0, 360.0))
70        .node(at(action("orders", "Consultar pedidos", "API da loja"), 600.0, -168.0))
71        .edge(link("e1", ("start", "out"), "menu"))
72        .edge(link("e2", ("menu", "pedidos"), "orders"))
73        .edge(link("e3", ("menu", "horários"), "hours"))
74        .edge(link("e4", ("menu", "atendente"), "open"))
75        .edge(link("e5", ("open", "sim"), "human"))
76        .edge(link("e6", ("open", "não"), "later"))
77}

104/// What comes after a block made with the arrow or Cmd/Ctrl+Enter: a message after anything
105/// but an action, which ends the way.
106pub fn next(from: &FlowNode, _side: PortSide) -> Option<FlowNode> {
107    (from.kind != "action").then(|| message("text", "Mensagem", "Escreva a mensagem"))
108}
109
110/// Whether a block may be joined to another: nothing goes into a trigger, and an action ends
111/// the way.
112pub fn validate(graph: &FlowGraph, from: &FlowEndpoint, to: &FlowEndpoint) -> Result<(), String> {
113    let kind = |id: &str| graph.find_node(id).map(|n| n.kind.clone()).unwrap_or_default();
114    if kind(&to.node) == "trigger" {
115        return Err("Um gatilho começa a conversa: nada chega nele".into());
116    }
117    if kind(&from.node) == "action" {
118        return Err("Uma ação termina o caminho".into());
119    }
120    Ok(())
121}

123/// The way a test run takes from the first trigger: each block and the connection it left
124/// by, its first way out (the first answer of a question, "yes" of a condition).
125pub fn run_path(graph: &FlowGraph) -> Vec<(String, Option<String>)> {
126    let mut path = Vec::new();
127    let Some(mut at) = graph.nodes.iter().find(|n| n.kind == "trigger").map(|n| n.id.clone()) else {
128        return path;
129    };
130    while !path.iter().any(|(id, _): &(String, Option<String>)| *id == at) {
131        let out = graph.edges.iter().find(|e| e.from.node == at);
132        path.push((at.clone(), out.map(|e| e.id.clone())));
133        match out {
134            Some(edge) => at = edge.to.node.clone(),
135            None => break,
136        }
137    }
138    path
139}

141/// The diagram as a test run shows it at `step` of `path`: the blocks before it done, the
142/// one at it running, the connections taken moving; `None` puts every block back.
143pub fn marked(graph: &FlowGraph, path: &[(String, Option<String>)], step: Option<usize>) -> FlowGraph {
144    let mut next = graph.clone();
145    for node in &mut next.nodes {
146        if matches!(node.status, FlowNodeStatus::Running | FlowNodeStatus::Success) {
147            node.status = FlowNodeStatus::Normal;
148        }
149    }
150    for edge in &mut next.edges {
151        edge.animated = false;
152    }
153    let Some(step) = step else { return next };
154    for (i, (id, left_by)) in path.iter().enumerate().take(step + 1) {
155        if let Some(node) = next.nodes.iter_mut().find(|n| n.id == *id) {
156            node.status = if i == step { FlowNodeStatus::Running } else { FlowNodeStatus::Success };
157        }
158        if i < step
159            && let Some(edge) = left_by.as_ref().and_then(|e| next.edges.iter_mut().find(|x| x.id == *e))
160        {
161            edge.animated = true;
162        }
163    }
164    next
165}

167fn icon(kind: &str) -> IconName {
168    match kind {
169        "trigger" => IconName::Zap,
170        "message" => IconName::MessageCircle,
171        "logic" => IconName::Split,
172        _ => IconName::Send,
173    }
174}

176/// The builder: blocks on the left, the conversation in the middle, the selected block's
177/// properties on the right.
178pub struct Builder {
179    /// Where the diagrams are kept between runs.
180    storage: Rc<dyn Storage>,
181}
182
183impl std::fmt::Debug for Builder {
184    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
185        f.debug_struct("Builder").finish_non_exhaustive()
186    }
187}

189impl View for Builder {
190    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
191        let graph = cx.use_state(sample);
192        let selection = cx.use_state(FlowSelection::default);
193        let history = cx.use_state(FlowHistory::new);
194        let arrivals = cx.use_state(|| None::<FlowNode>);
195        let view = cx.use_state(FlowViewport::fit);
196        // The connections hang as cables; the switch puts the plain lines back.
197        let cables = cx.use_state(|| true);
198        let mut canvas = FlowCanvas::new(graph.binding())
199            .cables(cables.get())
200            .view(view.binding())
201            .label("Conversa")
202            .selection(selection.binding())
203            .history(history.binding())
204            .arrivals(arrivals.binding())
205            .validate(validate)
206            .next_node(next)
207            .on_export(ravenrs_flow_export::saver(cx.file_picker(), "Conversa"));
208        let mut inspector =
209            FlowInspector::new(graph.binding(), selection.binding()).history(history.binding());
210        for (kind, caption) in KINDS {
211            let tone = if kind == "trigger" { Semantic::Warning } else { Semantic::Accent };
212            canvas = canvas.kind(kind, FlowKind::new(icon(kind), caption).tone(tone));
213            inspector = inspector.kind(kind, caption);
214        }
215        // A test run: the trigger first, then a step at a time along the first way out.
216        let step = cx.use_state(|| None::<usize>);
217        let path = run_path(&graph.get());
218        let (g1, s1, g2, s2, g3, s3) =
219            (graph.clone(), step.clone(), graph.clone(), step.clone(), graph.clone(), step.clone());
220        let last = path.len().saturating_sub(1);
221        let (p1, p2, p3) = (path.clone(), path.clone(), path);
222        let running = step.get().is_some();
223        let bar = HStack::new((
224            Button::new("Testar conversa").icon(IconName::Play).disabled(running).on_click(move || {
225                s1.set(Some(0));
226                g1.set(marked(&g1.get(), &p1, Some(0)));
227            }),
228            Button::new("Próximo passo")
229                .style(ButtonStyle::Secondary)
230                .disabled(!running || step.get() == Some(last))
231                .on_click(move || {
232                    let next = s2.get().map_or(0, |s| (s + 1).min(last));
233                    s2.set(Some(next));
234                    g2.set(marked(&g2.get(), &p2, Some(next)));
235                }),
236            Button::new("Parar").style(ButtonStyle::Ghost).disabled(!running).on_click(move || {
237                s3.set(None);
238                g3.set(marked(&g3.get(), &p3, None));
239            }),
240            Spacer::new(),
241            Toggle::new(cables.binding()).label("Cabos"),
242        ))
243        .spacing(8.0)
244        .padding(8.0);
245        cx.child(
246            &VStack::new((
247                bar,
248                HStack::new((
249                    VStack::new((
250                        FlowLibrary::new(graph.binding())
251                            .title("Conversas")
252                            .key("chats")
253                            .history(history.binding())
254                            .selection(selection.binding())
255                            .view(view.binding())
256                            .storage(self.storage.clone()),
257                        toolbox().title("Blocos").arrivals(arrivals.binding()),
258                    ))
259                    .spacing(0.0)
260                    .align(Align::Stretch),
261                    canvas,
262                    inspector,
263                ))
264                .align(Align::Stretch)
265                .grow(1.0),
266            ))
267            .spacing(0.0)
268            .align(Align::Stretch)
269            .grow(1.0),
270        )
271    }
272}

274/// The whole app.

279/// The whole app, keeping its diagrams in `storage`.

284#[cfg(target_arch = "wasm32")]
285mod web_entry {
286    use std::cell::RefCell;
287
288    use wasm_bindgen::prelude::*;
289
290    thread_local! {
291        static APP: RefCell<Option<ravenrs::web::WebApp>> = const { RefCell::new(None) };
292    }
293
294    /// Runs when the module is instantiated: the diagrams are kept in the browser's storage.
295    #[wasm_bindgen(start)]
296    pub fn start() {
297        console_error_panic_hook::set_once();
298        wasm_bindgen_futures::spawn_local(async {
299            let storage = match ravenrs::web::IndexedDbStorage::open("ravenrs-flow-chat-bot").await {
300                Ok(storage) => storage,
301                Err(e) => {
302                    web_sys::console::error_1(&e.to_string().into());
303                    return;
304                }
305            };
306            let app = super::app_kept_in(std::rc::Rc::new(storage));
307            match ravenrs::web::start(app, ravenrs::web::WebOptions::new("raven-canvas")).await {
308                Ok(app) => APP.with(|a| *a.borrow_mut() = Some(app)),
309                Err(e) => web_sys::console::error_1(&e.to_string().into()),
310            }
311        });
312    }
