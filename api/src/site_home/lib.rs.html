24/// The reports the fan shows, rendered once by the real engine.
25#[derive(Debug)]
26struct Reports {
27    customers: Document,
28    invoice: Document,
29    dashboard: Document,
30    sales: Document,
31}
32
33impl Reports {
34    fn render() -> Self {
35        let mut engine = ReportEngine::new();
36        let mut run = |report: ravenrs_report::Report| {
37            let data = report.embedded_data().cloned().unwrap_or(serde_json::Value::Null);
38            engine.render(&report, &data)
39        };
40        let customers = run(samples::customer_list());
41        let dashboard = run(samples::sales_dashboard());
42        let sales = run(samples::sales_by_region(300));
43        let invoice = engine.render(&samples::invoice(), &samples::invoice_data());
44        Self { customers, invoice, dashboard, sales }
45    }
46}

75impl View for Home {
76    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
77        let links = cx.links();
78        let open = self.palette.clone();
79        cx.shortcut(Key::Character("k".into()), true, Rc::new(move || open.set(!open.get())));
80        let (reports, palette, platform, clicks) =
81            (self.reports.clone(), self.palette.clone(), self.platform.clone(), self.clicks.clone());
82        let page = responsive(move |size| -> AnyView {
83            let narrow = size.width < 760.0;
84
85            Box::new(ZStack::new((
86                ScrollView::new((
87                    // Room for the bar, which floats over the page.
88                    Spacer::new().frame(None, Some(BAR)),
89                    hero(size, narrow, links.clone()),
90                    Anchor::new("demo", demo(links.clone())).offset(BAR),
91                    Anchor::new("pipeline", pipeline(narrow, clicks.clone())).offset(BAR),
92                    Anchor::new("components", components(narrow)).offset(BAR),
93                    Anchor::new("reports", reporting(reports.clone(), narrow, links.clone())).offset(BAR),
94                    Anchor::new("platforms", platforms(narrow, platform.clone())).offset(BAR),
95                    Anchor::new("roadmap", roadmap(narrow)).offset(BAR),
96                    closing(links.clone()),
97                ))
98                .spacing(0.0)
99                .height(size.height),
100                nav(narrow, palette.clone()),
101                command_palette(palette.clone(), links.clone()),
102            )))
103        });
104        cx.child(&page)
105    }
106}

182/// The app screen the floor of squares stands up into: a sidebar, cards and a chart.
183fn screen(u: f32, v: f32, time: f64) -> Color {
184    let theme = Theme::raven_dark();
185    let c = |s: Semantic| theme.color(s);
186    let inside = |a: f32, b: f32, top: f32, bottom: f32| u >= a && u <= b && v >= top && v <= bottom;
187    if v < 0.075 {
188        return if inside(0.012, 0.028, 0.02, 0.06) {
189            c(Semantic::Danger)
190        } else if inside(0.036, 0.052, 0.02, 0.06) {
191            c(Semantic::Warning)
192        } else if inside(0.06, 0.076, 0.02, 0.06) {
193            c(Semantic::Success)
194        } else {
195            c(Semantic::Surface)
196        };
197    }
198    if u < 0.2 {
199        if inside(0.02, 0.18, 0.13, 0.2) {
200            return if inside(0.03, 0.05, 0.14, 0.19) { c(Semantic::Accent) } else { c(Semantic::Control) };
201        }
202        let row = [0.25, 0.33, 0.41, 0.49].iter().any(|t| inside(0.03, 0.15, *t, t + 0.03));
203        return if row { c(Semantic::Border) } else { c(Semantic::Surface) };
204    }
205    if inside(0.24, 0.48, 0.11, 0.15) {
206        return c(Semantic::Ink);
207    }
208    if inside(0.24, 0.42, 0.18, 0.2) {
209        return c(Semantic::InkMuted);
210    }
211    if inside(0.84, 0.96, 0.11, 0.18) {
212        return c(Semantic::Accent);
213    }
214    for (a, b) in [(0.24, 0.47), (0.495, 0.725), (0.75, 0.96)] {
215        if inside(a, b, 0.25, 0.43) {
216            return if inside(a + 0.02, a + 0.14, 0.32, 0.37) {
217                c(Semantic::Ink)
218            } else if inside(a + 0.02, a + 0.1, 0.28, 0.3) {
219                c(Semantic::InkMuted)
220            } else if inside(a + 0.02, a + 0.07, 0.39, 0.41) {
221                c(Semantic::Success)
222            } else {
223                c(Semantic::SurfaceElevated)
224            };
225        }
226    }
227    if inside(0.24, 0.7, 0.48, 0.94) {
228        if inside(0.27, 0.67, 0.53, 0.9) && (u - 0.27) % 0.04 < 0.028 {
229            let bar = ((u - 0.27) / 0.04).floor();
230            let top = 0.9 - 0.37 * (0.35 + 0.55 * (0.5 + 0.5 * (bar * 1.7 + time as f32 * 1.2).sin()));
231            if v >= top {
232                return c(Semantic::Accent);
233            }
234        }
235        return c(Semantic::SurfaceElevated);
236    }
237    if inside(0.725, 0.96, 0.48, 0.94) {
238        if inside(0.745, 0.94, 0.83, 0.9) {
239            return c(Semantic::Accent);
240        }
241        return c(Semantic::SurfaceElevated);
242    }
243    c(Semantic::Canvas)
244}

330/// The share of the scene each step holds, with a short cross-fade into the next.
331fn step_link(i: usize, opacity_off: f32) -> ScrollLink {
332    let n = STEPS.len() as f32;
333    let (from, to) = (i as f32 / n, (i + 1) as f32 / n);
334    let off = Effect::IDENTITY.with_opacity(opacity_off);
335    let mut stops = vec![];
336    if i > 0 {
337        stops.push((from - 0.04, off));
338    }
339    stops.push((from + 0.02, Effect::IDENTITY));
340    stops.push((to - 0.02, Effect::IDENTITY));
341    if i + 1 < STEPS.len() {
342        stops.push((to + 0.04, off));
343    }
344    ScrollLink::keyframes(stops)
345}

399fn pipeline(narrow: bool, clicks: State<u32>) -> impl View {
400    let steps: Vec<AnyView> = STEPS
401        .iter()
402        .enumerate()
403        .map(|(i, (title, text))| {
404            let title =
405                HStack::new((Badge::new((i + 1).to_string()), Text::new(*title).style(TextStyle::Headline)))
406                    .spacing(10.0)
407                    .align(Align::Center);
408            // On a phone the panel and the list share the screen: titles only.
409            let row: AnyView =
410                if narrow { Box::new(title) } else { Box::new(stack((title, lede(text))).spacing(4.0)) };
411            Box::new(ScrollLinked::new(row, step_link(i, 0.35))) as AnyView
412        })
413        .collect();
414    let panes: Vec<AnyView> = (0..STEPS.len())
415        .map(|i| {
416            let pane = stack((
417                stack((step_visual(i, &clicks),)).align(Align::Center).frame(None, Some(240.0)),
418                CodeBlock::new(CODE[i]),
419            ))
420            .spacing(16.0)
421            .align(Align::Stretch);
422            Box::new(ScrollLinked::new(pane, step_link(i, 0.0))) as AnyView
423        })
424        .collect();
425    let text = stack((
426        kicker("From state to pixels"),
427        heading("One frame, six steps, no webview."),
428        lede("Scroll: the section holds while a click travels through the engine."),
429        stack(steps).spacing(14.0),
430    ))
431    .spacing(14.0);
432    let panel = Card::new(ZStack::new(panes)).padding(20.0);
433    let body: AnyView = if narrow {
434        Box::new(stack((panel, text)).spacing(20.0))
435    } else {
436        Box::new(HStack::new((text.grow(1.0), panel.grow(1.0))).spacing(48.0).align(Align::Start))
437    };
438    ScrollScene::new(stack((column(body),)).padding(Insets::symmetric(72.0, 0.0)))
439        .length(if narrow { 1800.0 } else { 2400.0 })
440        .top(BAR)
441}

541/// `1234567.0` as `1,234,567`.
542fn group(v: f64) -> String {
543    let digits = (v.round() as u64).to_string();
544    let mut out = String::new();
545    for (i, c) in digits.chars().enumerate() {
546        if i > 0 && (digits.len() - i).is_multiple_of(3) {
547            out.push(',');
548        }
549        out.push(c);
550    }
551    out
552}

827#[cfg(target_arch = "wasm32")]
828mod web_entry {
829    use std::cell::RefCell;
830
831    use wasm_bindgen::prelude::*;
832
833    thread_local! {
834        static APP: RefCell<Option<ravenrs::web::WebApp>> = const { RefCell::new(None) };
835    }
836
837    #[wasm_bindgen(start)]
838    pub fn start() {
839        console_error_panic_hook::set_once();
840        wasm_bindgen_futures::spawn_local(async {
841            // WebGL2 opens in every browser and phone, with or without WebGPU.
842            let options =
843                ravenrs::web::WebOptions::new("raven-canvas").backend(ravenrs::web::WebBackend::WebGl2);
844            match ravenrs::web::start(super::Home::new(), options).await {
845                Ok(app) => APP.with(|a| *a.borrow_mut() = Some(app)),
846                Err(e) => web_sys_log(&e.to_string()),
847            }
848        });
849    }
850
851    fn web_sys_log(message: &str) {
852        let _ = message;
853    }
854}
