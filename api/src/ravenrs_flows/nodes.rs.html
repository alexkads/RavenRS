27/// The parameter `name` read as an expression; `None` when the node has none.
28fn expression(cx: &NodeContext<'_>, name: &str) -> Result<Option<Criteria>, NodeError> {
29    match cx.parameter(name) {
30        None | Some(Value::Null) => Ok(None),
31        Some(Value::String(text)) => Criteria::parse(text)
32            .map(Some)
33            .map_err(|e| NodeError::new(format!("the parameter {name} is not an expression: {e}"))),
34        Some(_) => Err(NodeError::new(format!("the parameter {name} must be an expression, as text"))),
35    }
36}

38/// The parameter `name` read as an expression the node cannot do without.
39fn required(cx: &NodeContext<'_>, name: &str) -> Result<Criteria, NodeError> {
40    expression(cx, name)?.ok_or_else(|| NodeError::new(format!("the parameter {name} is missing")))
41}

43/// `criteria` worked out over `item`. The criteria language is parsed into a tree and read
44/// field by field; it runs no code, so a flow's expressions cannot reach the machine.
45fn work_out(env: &Environment, criteria: &Criteria, item: &Item) -> Criterion {
46    Scope::new(env).row(&item.json).evaluate(criteria).unwrap_or(Criterion::Null)
47}

49/// Whether a value counts as true: `true`, a number other than 0, text that is not empty, a date.
50fn truthy(value: &Criterion) -> bool {
51    match value {
52        Criterion::Null => false,
53        Criterion::Bool(b) => *b,
54        Criterion::Number(n) => *n != 0.0,
55        Criterion::Text(t) => !t.is_empty(),
56        _ => true,
57    }
58}
59
60/// A value of the criteria language as JSON: whole numbers stay whole.
61fn to_json(value: Criterion) -> Value {
62    match value {
63        Criterion::Null => Value::Null,
64        Criterion::Bool(b) => Value::Bool(b),
65        Criterion::Number(n) if n.fract() == 0.0 && n.abs() < 9.0e15 => json!(n as i64),
66        Criterion::Number(n) => json!(n),
67        Criterion::Text(t) => Value::String(t),
68        Criterion::Date(d) => Value::String(d.to_string()),
69        other => Value::String(format!("{other:?}")),
70    }
71}

73/// A value as the text a case or a key is compared by.
74fn as_text(value: &Value) -> String {
75    match value {
76        Value::Null => String::new(),
77        Value::String(s) => s.clone(),
78        other => other.to_string(),
79    }
80}

82/// The value at a field path written with dots (`address.city`), when the item has it.
83fn field<'a>(json: &'a Value, path: &str) -> Option<&'a Value> {
84    path.split('.').try_fold(json, |at, step| match at {
85        Value::Object(map) => map.get(step),
86        Value::Array(items) => items.get(step.parse::<usize>().ok()?),
87        _ => None,
88    })
89}

96/// JSON values in a total order: missing and null first, then false before true, numbers,
97/// text, and anything else by how it is written.
98fn compare(a: Option<&Value>, b: Option<&Value>) -> Ordering {
99    fn rank(v: Option<&Value>) -> u8 {
100        match v {
101            None | Some(Value::Null) => 0,
102            Some(Value::Bool(_)) => 1,
103            Some(Value::Number(_)) => 2,
104            Some(Value::String(_)) => 3,
105            Some(_) => 4,
106        }
107    }
108    match (a, b) {
109        (Some(Value::Bool(x)), Some(Value::Bool(y))) => x.cmp(y),
110        (Some(Value::Number(x)), Some(Value::Number(y))) => {
111            x.as_f64().unwrap_or(0.0).partial_cmp(&y.as_f64().unwrap_or(0.0)).unwrap_or(Ordering::Equal)
112        }
113        (Some(Value::String(x)), Some(Value::String(y))) => x.cmp(y),
114        (Some(x), Some(y)) if rank(a) == 4 && rank(b) == 4 => x.to_string().cmp(&y.to_string()),
115        _ => rank(a).cmp(&rank(b)),
116    }
117}

119/// Two objects as one; the fields of `over` win.
120fn merged(base: &Value, over: &Value) -> Value {
121    match (base, over) {
122        (Value::Object(a), Value::Object(b)) => {
123            let mut map = a.clone();
124            map.extend(b.iter().map(|(k, v)| (k.clone(), v.clone())));
125            Value::Object(map)
126        }
127        (_, Value::Null) => base.clone(),
128        _ => over.clone(),
129    }
130}

174/// Sends each item to the output named after the text of its `value`, one output per entry of
175/// `cases`; an item matching no case goes to `otherwise`.
176#[derive(Clone, Copy, Debug, Default)]
177pub struct Switch;
178
179impl NodeKind for Switch {
180    fn spec(&self) -> NodeSpec {
181        NodeSpec::new("switch", 1)
182            .title("Switch")
183            .input(MAIN)
184            .output("otherwise")
185            .requires("value")
186            .requires("cases")
187            .chooses()
188    }
189    fn outputs(&self, node: &NodeDef) -> Vec<String> {
190        let mut outputs: Vec<String> = match node.parameters.get("cases") {
191            Some(Value::Array(cases)) => cases.iter().map(as_text).collect(),
192            _ => Vec::new(),
193        };
194        outputs.push("otherwise".to_owned());
195        outputs
196    }
197    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
198        let value = required(cx, "value")?;
199        let cases = self.outputs(cx.node());
200        let env = Environment::new();
201        let mut given: Ports = cases.iter().map(|c| (c.clone(), Vec::new())).collect();
202        for item in cx.input(MAIN) {
203            let said = as_text(&to_json(work_out(&env, &value, item)));
204            let port = if cases[..cases.len() - 1].contains(&said) { said } else { "otherwise".to_owned() };
205            given.entry(port).or_default().push(item.clone());
206        }
207        Ok(given)
208    }
209}

211/// Joins the items of its inputs `a` and `b`, by its parameter `mode`:
212///
213/// - `append` (the default): every item of `a`, then every item of `b`, once both arrived;
214/// - `position`: the items at the same place made one, `b`'s fields winning;
215/// - `key`: each item of `a` made one with each item of `b` whose `key` gives the same value
216///   (`key_a` and `key_b` when the two sides name it differently); `keep_unmatched` keeps the
217///   items of `a` that found none;
218/// - `first`: the items of the first input that has any.
219#[derive(Clone, Copy, Debug, Default)]
220pub struct Merge;
221
222impl NodeKind for Merge {
223    fn spec(&self) -> NodeSpec {
224        NodeSpec::new("merge", 1).title("Merge").input("a").input("b").output(MAIN)
225    }
226    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
227        let (a, b) = (cx.input("a"), cx.input("b"));
228        let items = match text_parameter(cx, "mode", "append") {
229            "append" => a.iter().chain(b).cloned().collect(),
230            "first" => {
231                if a.is_empty() {
232                    b.to_vec()
233                } else {
234                    a.to_vec()
235                }
236            }
237            "position" => (0..a.len().max(b.len()))
238                .map(|i| match (a.get(i), b.get(i)) {
239                    (Some(x), Some(y)) => Item::derive_all([x, y], merged(&x.json, &y.json)),
240                    (Some(x), None) | (None, Some(x)) => x.clone(),
241                    (None, None) => Item::default(),
242                })
243                .collect(),
244            "key" => {
245                let env = Environment::new();
246                let key_a = expression(cx, "key_a")?.map_or_else(|| required(cx, "key"), Ok)?;
247                let key_b = expression(cx, "key_b")?.map_or_else(|| required(cx, "key"), Ok)?;
248                let keep = cx.parameter("keep_unmatched").and_then(Value::as_bool).unwrap_or(false);
249                let mut by_key: BTreeMap<String, Vec<&Item>> = BTreeMap::new();
250                for y in b {
251                    by_key.entry(as_text(&to_json(work_out(&env, &key_b, y)))).or_default().push(y);
252                }
253                let mut out = Vec::new();
254                for x in a {
255                    let key = as_text(&to_json(work_out(&env, &key_a, x)));
256                    match by_key.get(&key) {
257                        Some(matches) if !key.is_empty() => {
258                            out.extend(
259                                matches.iter().map(|y| Item::derive_all([x, *y], merged(&x.json, &y.json))),
260                            );
261                        }
262                        _ if keep => out.push(x.clone()),
263                        _ => {}
264                    }
265                }
266                out
267            }
268            other => {
269                return Err(NodeError::new(format!(
270                    "the merge mode {other} is not append, position, key or first"
271                )));
272            }
273        };
274        Ok(main_port(items))
275    }
276}

278/// A loop over its input in batches of `size` (1 by default). Each turn it sends the next
279/// batch to `batch`, which leads to the loop's body; what the body sends back reaches its input
280/// `back`. When no batch is left, it gives everything that came back on `done`.
281#[derive(Clone, Copy, Debug, Default)]
282pub struct Batches;
283
284impl NodeKind for Batches {
285    fn spec(&self) -> NodeSpec {
286        NodeSpec::new("batches", 1)
287            .title("Loop in batches")
288            .input(MAIN)
289            .input("back")
290            .output("batch")
291            .output("done")
292            .loops()
293    }
294    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
295        let size = match cx.parameter("size") {
296            None => 1,
297            Some(v) => v
298                .as_u64()
299                .filter(|n| *n > 0)
300                .ok_or_else(|| NodeError::new("the batch size must be a whole number above 0"))?
301                as usize,
302        };
303        let all = cx.input(MAIN);
304        let start = cx.iteration() * size;
305        if start < all.len() {
306            let batch = all[start..(start + size).min(all.len())].to_vec();
307            Ok(Ports::from([("batch".to_owned(), batch)]))
308        } else {
309            Ok(Ports::from([("done".to_owned(), cx.input("back").to_vec())]))
310        }
311    }
312}

314/// Sets fields on each item: its parameter `fields` maps each name to an expression worked out
315/// over the item. With `keep_only`, the item keeps only those fields.
316#[derive(Clone, Copy, Debug, Default)]
317pub struct Set;
318
319impl NodeKind for Set {
320    fn spec(&self) -> NodeSpec {
321        NodeSpec::new("set", 1).title("Set fields").input(MAIN).output(MAIN).requires("fields")
322    }
323    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
324        let Some(Value::Object(fields)) = cx.parameter("fields") else {
325            return Err(NodeError::new("the parameter fields must map names to expressions"));
326        };
327        // A field is an expression, or text with expressions inside it (`Olá {{ nome }}`).
328        let mut parsed = Vec::with_capacity(fields.len());
329        let mut texts = Vec::new();
330        for (name, text) in fields {
331            let text = text
332                .as_str()
333                .ok_or_else(|| NodeError::new(format!("the field {name} must be an expression, as text")))?;
334            if text.contains("{{") {
335                let checked = ravenrs_criteria::Template::parse(text)
336                    .map_err(|errors| NodeError::new(format!("the field {name}: {}", errors[0].message)));
337                checked?;
338                texts.push((name.clone(), text.to_owned()));
339                continue;
340            }
341            let criteria =
342                Criteria::parse(text).map_err(|e| NodeError::new(format!("the field {name}: {e}")))?;
343            parsed.push((name.clone(), criteria));
344        }
345        let keep_only = cx.parameter("keep_only").and_then(Value::as_bool).unwrap_or(false);
346        let env = Environment::new();
347        let made = cx
348            .input(MAIN)
349            .iter()
350            .map(|item| {
351                let mut map = match (&item.json, keep_only) {
352                    (Value::Object(map), false) => map.clone(),
353                    _ => Map::new(),
354                };
355                for (name, criteria) in &parsed {
356                    map.insert(name.clone(), to_json(work_out(&env, criteria, item)));
357                }
358                for (name, text) in &texts {
359                    let record = crate::expression::record(
360                        &item.json,
361                        &cx.nodes(),
362                        &Map::new(),
363                        &crate::expression::run_value(cx.now_ms()),
364                    );
365                    let written = crate::expression::render(text, &record)
366                        .map_err(|why| NodeError::new(format!("the field {name}: {why}")))?;
367                    map.insert(name.clone(), Value::String(written));
368                }
369                Ok(item.derive(Value::Object(map)))
370            })
371            .collect::<Result<Vec<_>, NodeError>>()?;
372        Ok(main_port(made))
373    }
374}

398/// Puts the items in order by the fields of its parameter `by`: a list of field paths, or of
399/// `{ "field": …, "descending": true }`. Items that tie keep the order they came in.
400#[derive(Clone, Copy, Debug, Default)]
401pub struct Sort;
402
403impl NodeKind for Sort {
404    fn spec(&self) -> NodeSpec {
405        NodeSpec::new("sort", 1).title("Sort").input(MAIN).output(MAIN).requires("by")
406    }
407    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
408        let Some(Value::Array(by)) = cx.parameter("by") else {
409            return Err(NodeError::new("the parameter by must be a list of fields"));
410        };
411        let mut keys = Vec::new();
412        for key in by {
413            match key {
414                Value::String(path) => keys.push((path.clone(), false)),
415                Value::Object(o) => match o.get("field").and_then(Value::as_str) {
416                    Some(path) => keys.push((
417                        path.to_owned(),
418                        o.get("descending").and_then(Value::as_bool).unwrap_or(false),
419                    )),
420                    None => return Err(NodeError::new("each sort key needs a field")),
421                },
422                _ => return Err(NodeError::new("each sort key is a field or { field, descending }")),
423            }
424        }
425        let mut items = cx.input(MAIN).to_vec();
426        items.sort_by(|x, y| {
427            keys.iter()
428                .map(|(path, down)| {
429                    let order = compare(field(&x.json, path), field(&y.json, path));
430                    if *down { order.reverse() } else { order }
431                })
432                .find(|o| *o != Ordering::Equal)
433                .unwrap_or(Ordering::Equal)
434        });
435        Ok(main_port(items))
436    }
437}

439/// One item per group of items with the same values of the fields in `by` (one group of all
440/// with none), in the order each group first appears. Its parameter `aggregates` lists
441/// `{ "field": …, "op": "sum" | "count" | "min" | "max" | "avg" | "list", "as": … }`.
442#[derive(Clone, Copy, Debug, Default)]
443pub struct Group;
444
445impl NodeKind for Group {
446    fn spec(&self) -> NodeSpec {
447        NodeSpec::new("group", 1).title("Group").input(MAIN).output(MAIN)
448    }
449    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
450        let by: Vec<String> = match cx.parameter("by") {
451            None => Vec::new(),
452            Some(Value::Array(list)) => list.iter().map(as_text).collect(),
453            Some(_) => return Err(NodeError::new("the parameter by must be a list of fields")),
454        };
455        let mut aggregates = Vec::new();
456        if let Some(list) = cx.parameter("aggregates") {
457            let Value::Array(list) = list else {
458                return Err(NodeError::new("the parameter aggregates must be a list"));
459            };
460            for a in list {
461                let op = a.get("op").and_then(Value::as_str).unwrap_or("count");
462                if !["sum", "count", "min", "max", "avg", "list"].contains(&op) {
463                    return Err(NodeError::new(format!(
464                        "the aggregate {op} is not sum, count, min, max, avg or list"
465                    )));
466                }
467                let path = a.get("field").and_then(Value::as_str).unwrap_or_default().to_owned();
468                let name =
469                    a.get("as").and_then(Value::as_str).map_or_else(|| format!("{op}_{path}"), str::to_owned);
470                aggregates.push((op.to_owned(), path, name));
471            }
472        }
473        let mut groups: Vec<(Vec<Value>, Vec<&Item>)> = Vec::new();
474        for item in cx.input(MAIN) {
475            let key: Vec<Value> =
476                by.iter().map(|p| field(&item.json, p).cloned().unwrap_or(Value::Null)).collect();
477            match groups.iter_mut().find(|(k, _)| *k == key) {
478                Some((_, members)) => members.push(item),
479                None => groups.push((key, vec![item])),
480            }
481        }
482        let made = groups
483            .into_iter()
484            .map(|(key, members)| {
485                let mut map = Map::new();
486                for (path, value) in by.iter().zip(key) {
487                    map.insert(path.clone(), value);
488                }
489                for (op, path, name) in &aggregates {
490                    let values: Vec<&Value> = members
491                        .iter()
492                        .filter_map(|m| field(&m.json, path))
493                        .filter(|v| !v.is_null())
494                        .collect();
495                    let numbers: Vec<f64> = values.iter().filter_map(|v| v.as_f64()).collect();
496                    let result = match op.as_str() {
497                        "count" => json!(if path.is_empty() { members.len() } else { values.len() }),
498                        "sum" => to_json(Criterion::Number(numbers.iter().sum())),
499                        "avg" if numbers.is_empty() => Value::Null,
500                        "avg" => json!(numbers.iter().sum::<f64>() / numbers.len() as f64),
501                        "min" => values
502                            .iter()
503                            .copied()
504                            .min_by(|x, y| compare(Some(x), Some(y)))
505                            .cloned()
506                            .unwrap_or(Value::Null),
507                        "max" => values
508                            .iter()
509                            .copied()
510                            .max_by(|x, y| compare(Some(x), Some(y)))
511                            .cloned()
512                            .unwrap_or(Value::Null),
513                        _ => Value::Array(values.into_iter().cloned().collect()),
514                    };
515                    map.insert(name.clone(), result);
516                }
517                Item::derive_all(members, Value::Object(map))
518            })
519            .collect();
520        Ok(main_port(made))
521    }
522}

524/// One item per entry of the list in each item's `field`: an object entry is the new item, any
525/// other is kept under the field's name. An item whose field is not a list goes to the error
526/// output.
527#[derive(Clone, Copy, Debug, Default)]
528pub struct Split;
529
530impl NodeKind for Split {
531    fn spec(&self) -> NodeSpec {
532        NodeSpec::new("split", 1).title("Split a list").input(MAIN).output(MAIN).requires("field")
533    }
534    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
535        let path = text_parameter(cx, "field", "");
536        let name = path.rsplit('.').next().unwrap_or(path);
537        let (mut made, mut failed) = (Vec::new(), Vec::new());
538        for item in cx.input(MAIN) {
539            match field(&item.json, path) {
540                Some(Value::Array(entries)) => made.extend(entries.iter().map(|entry| match entry {
541                    Value::Object(_) => item.derive(entry.clone()),
542                    other => item.derive(json!({ name: other })),
543                })),
544                _ => failed.push(item.failed(&format!("{path} is not a list"))),
545            }
546        }
547        Ok(Ports::from([(MAIN.to_owned(), made), (ERROR_PORT.to_owned(), failed)]))
548    }
549}

551/// One item holding a list of all that came: the whole items, or only their `field`, under the
552/// name `as` (`items` by default).
553#[derive(Clone, Copy, Debug, Default)]
554pub struct Join;
555
556impl NodeKind for Join {
557    fn spec(&self) -> NodeSpec {
558        NodeSpec::new("join", 1).title("Join into a list").input(MAIN).output(MAIN)
559    }
560    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
561        let name = text_parameter(cx, "as", "items").to_owned();
562        let items = cx.input(MAIN);
563        let list: Vec<Value> = match cx.parameter("field").and_then(Value::as_str) {
564            Some(path) => items.iter().filter_map(|i| field(&i.json, path).cloned()).collect(),
565            None => items.iter().map(|i| i.json.clone()).collect(),
566        };
567        Ok(main_port(vec![Item::derive_all(items, json!({ name: list }))]))
568    }
569}

571/// Waits `ms` milliseconds on the run's clock, then passes its items on. It stops early when
572/// the run is cancelled or its time is up. The wait is in memory: a wait that outlives the app
573/// needs a durable run.
574#[derive(Clone, Copy, Debug, Default)]
575pub struct Wait;
576
577impl NodeKind for Wait {
578    fn spec(&self) -> NodeSpec {
579        NodeSpec::new("wait", 1).title("Wait").input(MAIN).output(MAIN).requires("ms")
580    }
581    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
582        let ms = cx
583            .parameter("ms")
584            .and_then(Value::as_u64)
585            .ok_or_else(|| NodeError::new("the wait must be a whole number of ms"))?;
586        cx.sleep(ms)?;
587        Ok(main_port(cx.input(MAIN).to_vec()))
588    }
589}

591/// Runs another flow with the items that came, and gives what it ends with. Its parameter
592/// `flow` names one of the flows it was made with; `input` names the node of that flow that
593/// starts with the items (its first trigger by default) and `output` the node whose first
594/// output it gives (the last to run by default). The flow it calls stops when this one is
595/// cancelled, and has only the time this node has left.
596#[derive(Clone, Debug)]
597pub struct Subflow {
598    flows: Arc<BTreeMap<String, FlowDefinition>>,
599    catalog: Arc<Catalog>,
600}
601
602impl Subflow {
603    /// A subflow node that may call any of `flows`, run with the kinds of `catalog`.
604    pub fn new(flows: impl IntoIterator<Item = FlowDefinition>, catalog: Catalog) -> Self {
605        Self {
606            flows: Arc::new(flows.into_iter().map(|f| (f.id.clone(), f)).collect()),
607            catalog: Arc::new(catalog),
608        }
609    }
610}
611
612impl NodeKind for Subflow {
613    fn spec(&self) -> NodeSpec {
614        NodeSpec::new("subflow", 1).title("Run a flow").input(MAIN).output(MAIN).requires("flow")
615    }
616    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
617        let id = text_parameter(cx, "flow", "");
618        let flow = self.flows.get(id).ok_or_else(|| NodeError::new(format!("there is no flow {id}")))?;
619        let plan = Plan::compile(flow, &self.catalog).map_err(|errors| {
620            let first = errors.first().map(ToString::to_string).unwrap_or_default();
621            NodeError::new(format!("the flow {id} does not compile: {first}"))
622        })?;
623        let input = match cx.parameter("input").and_then(Value::as_str) {
624            Some(node) => node.to_owned(),
625            None => plan
626                .order()
627                .iter()
628                .find(|n| plan.spec(n).is_some_and(|s| s.trigger))
629                .cloned()
630                .ok_or_else(|| NodeError::new(format!("the flow {id} has no trigger to start from")))?,
631        };
632        let output = match cx.parameter("output").and_then(Value::as_str) {
633            Some(node) => node.to_owned(),
634            None => plan.order().last().cloned().unwrap_or_default(),
635        };
636        let mut executor = Executor::new(&plan, &self.catalog)
637            .mode(cx.mode())
638            .shared_clock(cx.clock().clone())
639            .cancel_with(cx.token().child())
640            .start_with(input, cx.input(MAIN).to_vec());
641        if let Some(left) = cx.time_left_ms() {
642            executor = executor.time_limit_ms(left.max(0) as u64);
643        }
644        let run = executor.run();
645        match run.outcome {
646            RunOutcome::Finished => {
647                let port = plan.outputs(&output).first().cloned().unwrap_or_default();
648                Ok(main_port(run.output(&output, &port).to_vec()))
649            }
650            RunOutcome::Cancelled { reason } => Err(NodeError::cancelled(reason)),
651            RunOutcome::TimedOut { reason } => Err(NodeError::timed_out(reason)),
652            RunOutcome::Stopped { node, reason } => {
653                Err(NodeError::new(format!("the flow {id} stopped at {node}: {reason}")))
654            }
655            RunOutcome::Waiting { node, .. } => Err(NodeError::new(format!(
656                "the flow {id} waits at {node}; a subflow cannot wait yet, since the flow inside is not kept"
657            ))),
658        }
659    }
660}

662/// Waits `ms` milliseconds in a way that outlives the app: the run stops at this node and a
663/// durable run starts it again when the time comes, then the items pass on. Run without one,
664/// the run just ends waiting ([`RunOutcome::Waiting`]).
665#[derive(Clone, Copy, Debug, Default)]
666pub struct Delay;
667
668impl NodeKind for Delay {
669    fn spec(&self) -> NodeSpec {
670        NodeSpec::new("delay", 1).title("Wait, even across restarts").input(MAIN).output(MAIN).requires("ms")
671    }
672    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
673        let ms = cx
674            .parameter("ms")
675            .and_then(Value::as_i64)
676            .ok_or_else(|| NodeError::new("the wait must be a whole number of ms"))?;
677        let until = cx.woken().and_then(|w| w.until_ms).unwrap_or_else(|| cx.clock().now_ms() + ms);
678        if cx.clock().now_ms() >= until {
679            Ok(main_port(cx.input(MAIN).to_vec()))
680        } else {
681            Err(NodeError::wait_until(until))
682        }
683    }
684}

686/// Waits for the event named by its parameter `event` (an approval, a reply, a webhook), then
687/// gives the items the event brought. The run stops here until a durable run is told of it.
688///
689/// With `keep` true it is a gate instead: it gives the items that reached it, each with the
690/// first item the event brought under the field `as` (`"event"` when not said), so what waited
691/// for an approval goes on with the approval.
692#[derive(Clone, Copy, Debug, Default)]
693pub struct WaitEvent;
694
695impl NodeKind for WaitEvent {
696    fn spec(&self) -> NodeSpec {
697        NodeSpec::new("wait_event", 1).title("Wait for an event").input(MAIN).output(MAIN).requires("event")
698    }
699    fn execute(&self, cx: &NodeContext<'_>) -> Result<Ports, NodeError> {
700        let event = text_parameter(cx, "event", "");
701        let Some(brought) = cx.event(event) else { return Err(NodeError::wait_for(event)) };
702        if !cx.parameter("keep").and_then(Value::as_bool).unwrap_or(false) {
703            return Ok(main_port(brought.to_vec()));
704        }
705        let field = text_parameter(cx, "as", "event");
706        let news = brought.first().map(|item| item.json.clone()).unwrap_or(Value::Null);
707        let kept = cx
708            .input(MAIN)
709            .iter()
710            .map(|item| {
711                let mut json = item.json.clone();
712                match json.as_object_mut() {
713                    Some(fields) => {
714                        fields.insert(field.to_owned(), news.clone());
715                    }
716                    None => json = serde_json::json!({ "value": json, field: news.clone() }),
717                }
718                item.derive(json)
719            })
720            .collect();
721        Ok(main_port(kept))
722    }
723}

725impl Catalog {
726    /// A catalog with every built-in kind: `manual`, `if`, `switch`, `merge`, `batches`, `set`,
727    /// `filter`, `sort`, `group`, `split`, `join`, `wait`, `delay` and `wait_event`, each at
728    /// version 1. A [`Subflow`]
729    /// is added apart, with the flows it may call.
730    pub fn builtin() -> Self {
731        Self::new()
732            .with(Manual)
733            .with(If)
734            .with(Switch)
735            .with(Merge)
736            .with(Batches)
737            .with(Set)
738            .with(Filter)
739            .with(Sort)
740            .with(Group)
741            .with(Split)
742            .with(Join)
743            .with(Wait)
744            .with(Delay)
745            .with(WaitEvent)
746    }
