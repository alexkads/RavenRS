24/// What a node does when it fails.
25#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
26#[serde(rename_all = "snake_case")]
27#[non_exhaustive]
28pub enum ErrorPolicy {
29    /// The run stops at the node (the default).
30    #[default]
31    Stop,
32    /// The run goes on: the items that failed go to the node's error output
33    /// ([`ERROR_PORT`](crate::ERROR_PORT)), the others where they were going, and a node that
34    /// fails as a whole gives nothing.
35    Continue,
36    /// Like `Continue`, but a node that fails as a whole sends every item that reached it to
37    /// its error output, with the reason.
38    Route,
39}

41/// How a node is tried again when it fails in a way worth it ([`Failure::Retryable`], or a time
42/// limit with `on_timeout`).
43///
44/// [`Failure::Retryable`]: crate::Failure::Retryable
45#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
46pub struct Retry {
47    /// How many tries in all, the first one included.
48    pub tries: u32,
49    /// How long to wait before the second try, in milliseconds.
50    #[serde(default)]
51    pub wait_ms: u64,
52    /// What each wait is multiplied by before the next one (1 keeps them the same).
53    #[serde(default = "one_f64")]
54    pub backoff: f64,
55    /// Whether a try that ran out of time is tried again; off by default, since the call may
56    /// have happened.
57    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
58    pub on_timeout: bool,
59}
60
61fn one_f64() -> f64 {
62    1.0
63}
64
65impl Retry {
66    /// `tries` tries in all, `wait_ms` apart.
67    pub fn new(tries: u32, wait_ms: u64) -> Self {
68        Self { tries: tries.max(1), wait_ms, backoff: 1.0, on_timeout: false }
69    }
70
71    /// With each wait `factor` times the one before.
72    pub fn backoff(mut self, factor: f64) -> Self {
73        self.backoff = factor.max(1.0);
74        self
75    }
76
77    /// Trying again also after running out of time.
78    pub fn on_timeout(mut self) -> Self {
79        self.on_timeout = true;
80        self
81    }
82
83    /// The wait before the try `next` (from 2).
84    pub fn wait_before(&self, next: u32) -> u64 {
85        let times = next.saturating_sub(2) as i32;
86        (self.wait_ms as f64 * self.backoff.powi(times)).round() as u64
87    }
88}

115/// A node of a flow: which kind it is, at which version, and how it is set up.
116#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
117pub struct NodeDef {
118    /// Unique in the flow, and kept when the node is renamed or moved.
119    pub id: String,
120    /// The kind's name in the catalog.
121    pub kind: String,
122    /// The kind's version the node was set up for.
123    #[serde(default = "one")]
124    pub version: u32,
125    /// What the person called it, when not the kind's own title.
126    #[serde(default, skip_serializing_if = "Option::is_none")]
127    pub title: Option<String>,
128    /// Its parameters, by name.
129    #[serde(default, skip_serializing_if = "Map::is_empty")]
130    pub parameters: Map<String, Value>,
131    /// A disabled node passes the items of its first input to its first output unchanged.
132    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
133    pub disabled: bool,
134    /// What it does when it fails.
135    #[serde(default, skip_serializing_if = "is_default")]
136    pub on_error: ErrorPolicy,
137    /// Items fixed on its output ports, used instead of running it while testing.
138    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
139    pub pinned: BTreeMap<String, Vec<Item>>,
140    /// How it is tried again; once, without it.
141    #[serde(default, skip_serializing_if = "Option::is_none")]
142    pub retry: Option<Retry>,
143    /// The most each try may take, in milliseconds.
144    #[serde(default, skip_serializing_if = "Option::is_none")]
145    pub timeout_ms: Option<u64>,
146}
147
148fn is_default(policy: &ErrorPolicy) -> bool {
149    *policy == ErrorPolicy::default()
150}

152impl NodeDef {
153    /// A node `id` of the kind `kind`, version 1, with no parameters.

156            id: id.into(),
157            kind: kind.into(),
158            version: 1,
159            title: None,
160            parameters: Map::new(),
161            disabled: false,
162            on_error: ErrorPolicy::default(),
163            pinned: BTreeMap::new(),
164            retry: None,
165            timeout_ms: None,
166        }
167    }
168
169    /// Tried again as `retry` says.
170    pub fn retry(mut self, retry: Retry) -> Self {
171        self.retry = Some(retry);
172        self
173    }
174
175    /// With each try taking at most `ms` milliseconds.
176    pub fn timeout_ms(mut self, ms: u64) -> Self {
177        self.timeout_ms = Some(ms);
178        self
179    }
180
181    /// Set up for the version `version` of its kind.
182    pub fn version(mut self, version: u32) -> Self {
183        self.version = version;
184        self
185    }
186
187    /// Called `title`.
188    pub fn title(mut self, title: impl Into<String>) -> Self {
189        self.title = Some(title.into());
190        self
191    }
192
193    /// With the parameter `name` set to `value`.
194    pub fn parameter(mut self, name: impl Into<String>, value: Value) -> Self {
195        self.parameters.insert(name.into(), value);
196        self
197    }
198
199    /// Disabled, or not.
200    pub fn disabled(mut self, disabled: bool) -> Self {
201        self.disabled = disabled;
202        self
203    }
204
205    /// Doing `policy` when it fails.
206    pub fn on_error(mut self, policy: ErrorPolicy) -> Self {
207        self.on_error = policy;
208        self
209    }
210
211    /// With `items` fixed on its output port `port`.

214        self
215    }

218/// A flow: its nodes and connections, and the revision they are at.
219#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
220pub struct FlowDefinition {
221    /// The version of the JSON it was read from or will be written as.
222    #[serde(default = "format")]
223    pub format: u32,
224    /// Unique among the app's flows.
225    pub id: String,
226    /// What the person called it.
227    pub name: String,
228    /// Which revision it is; the app raises it on each saved change.
229    #[serde(default)]
230    pub revision: u32,
231    /// The nodes, in the order they were made.
232    #[serde(default)]
233    pub nodes: Vec<NodeDef>,
234    /// The connections, in the order they were made.
235    #[serde(default)]
236    pub connections: Vec<Connection>,
237}
238
239impl FlowDefinition {
240    /// An empty flow `id` called `name`, at revision 0.
241    pub fn new(id: impl Into<String>, name: impl Into<String>) -> Self {
242        Self {
243            format: FORMAT,
244            id: id.into(),
245            name: name.into(),
246            revision: 0,
247            nodes: Vec::new(),
248            connections: Vec::new(),
249        }
250    }
251
252    /// With the node `node` added at the end.
253    pub fn node(mut self, node: NodeDef) -> Self {
254        self.nodes.push(node);
255        self
256    }
257
258    /// With a connection from the output `from_port` of `from` to the input `to_port` of `to`.
259    pub fn connect(mut self, from: &str, from_port: &str, to: &str, to_port: &str) -> Self {
260        self.connections
261            .push(Connection { from: PortRef::new(from, from_port), to: PortRef::new(to, to_port) });
262        self
263    }
264
265    /// The node with the id `id`.
266    pub fn find(&self, id: &str) -> Option<&NodeDef> {
267        self.nodes.iter().find(|n| n.id == id)
268    }
269
270    /// The flow written as JSON, at the version [`FORMAT`].
271    pub fn to_json(&self) -> String {
272        let mut written = self.clone();
273        written.format = FORMAT;
274        serde_json::to_string(&written).unwrap_or_default()
275    }
276
277    /// A flow read from JSON. A version newer than [`FORMAT`] is refused, so an older build
278    /// never drops what it does not know.
279    pub fn from_json(text: &str) -> Result<Self, DefinitionError> {
280        let read: Self =
281            serde_json::from_str(text).map_err(|e| DefinitionError::Unreadable(e.to_string()))?;
282        if read.format > FORMAT {
283            return Err(DefinitionError::Newer { format: read.format });
284        }
285        Ok(Self { format: FORMAT, ..read })
286    }
287}
288
289/// Why a flow could not be read.
290#[derive(Clone, Debug, PartialEq, Eq)]
291#[non_exhaustive]
292pub enum DefinitionError {
293    /// The text is not a flow; the reason is the reader's.
294    Unreadable(String),
295    /// It was written by a newer build, at this version of the format.
296    Newer {
297        /// The version it was written at.
298        format: u32,
299    },
300}
301
302impl fmt::Display for DefinitionError {
303    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
304        match self {
305            Self::Unreadable(why) => write!(f, "not a flow: {why}"),
306            Self::Newer { format } => {
307                write!(f, "written at version {format} of the format; this build reads up to {FORMAT}")
308            }
309        }
310    }
311}
