1//! What passes from node to node: items, each a JSON value with the files it carries and
2//! where it came from. The bottom layer of the crate: it knows no definition, kind or run.
3
4use std::collections::BTreeMap;
5
6use serde::{Deserialize, Serialize};
7use serde_json::Value;
8
9/// Where an item is: the node that gave it, the output port and its place in that port's list.
10#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
11pub struct ItemRef {
12    /// The node's id.
13    pub node: String,
14    /// The output port's id.
15    pub port: String,
16    /// Its place among the items of that port, from 0.
17    pub index: usize,
18}
19
20impl ItemRef {
21    /// The item `index` of the port `port` of the node `node`.
22    pub fn new(node: impl Into<String>, port: impl Into<String>, index: usize) -> Self {
23        Self { node: node.into(), port: port.into(), index }
24    }
25}
26
27/// A file an item carries, kept apart from its JSON: what it is, how big, and where it is.
28#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
29pub struct BinaryRef {
30    /// Its media type, such as `image/png`.
31    pub mime: String,
32    /// Its size in bytes.
33    pub size: u64,
34    /// Where it is kept: a path or a key in a store, as the app decides.
35    pub location: String,
36}
37
38/// One item of a list passing along a connection.
39///
40/// A node that makes an item from another uses [`Item::derive`], so the run can tell where
41/// each item came from; a node that passes an item on unchanged returns it as it came, and the
42/// run says that it came from where it was.
43#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
44pub struct Item {
45    /// What the item says.
46    pub json: Value,
47    /// The files it carries, by name.
48    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
49    pub binary: BTreeMap<String, BinaryRef>,
50    /// Where it is, once a run has put it on a port; `None` while a node is making it.
51    #[serde(default, skip_serializing_if = "Option::is_none")]
52    pub at: Option<ItemRef>,
53    /// The items it was made from, in the order they were given.
54    #[serde(default, skip_serializing_if = "Vec::is_empty")]
55    pub from: Vec<ItemRef>,
56}
57
58impl Item {
59    /// A new item saying `json`, made from nothing that came before.
60    pub fn new(json: Value) -> Self {
61        Self { json, ..Self::default() }
62    }
63
64    /// A new item saying `json`, made from this one: it keeps this one's files and says it came
65    /// from here.
66    pub fn derive(&self, json: Value) -> Self {
67        Self { json, binary: self.binary.clone(), at: None, from: self.at.iter().cloned().collect() }
68    }
69
70    /// A new item saying `json`, made from all of `parents` (a merge, a group).
71    pub fn derive_all<'a>(parents: impl IntoIterator<Item = &'a Item>, json: Value) -> Self {
72        Self { json, from: parents.into_iter().filter_map(|p| p.at.clone()).collect(), ..Self::default() }
73    }
74
75    /// A new item made from this one, saying it failed because of `message`: an object keeps its
76    /// fields and gains `error`; anything else is kept under `value`.
77    pub fn failed(&self, message: &str) -> Self {
78        let json = match &self.json {
79            Value::Object(map) => {
80                let mut map = map.clone();
81                map.insert("error".to_owned(), Value::String(message.to_owned()));
82                Value::Object(map)
83            }
84            other => serde_json::json!({ "error": message, "value": other }),
85        };
86        self.derive(json)
87    }
88
89    /// The same item carrying the file `binary` under `name`.
90    pub fn with_binary(mut self, name: impl Into<String>, binary: BinaryRef) -> Self {
91        self.binary.insert(name.into(), binary);
92        self
93    }
94
95    /// The value at `key`, when the item is a JSON object that has it.
96    pub fn get(&self, key: &str) -> Option<&Value> {
97        self.json.get(key)
98    }
99
100    /// Puts the item on the port `port` of `node` at `index`. An item that was already somewhere
101    /// passed through, and is said to come from there.
102    pub(crate) fn placed(mut self, node: &str, port: &str, index: usize) -> Self {
103        if let Some(previous) = self.at.take() {
104            self.from = vec![previous];
105        }
106        self.at = Some(ItemRef::new(node, port, index));
107        self
108    }
109}
110
111/// The items of a node's ports, by port id.
112pub type Ports = BTreeMap<String, Vec<Item>>;
