1//! What an expression of a flow can read, and how a parameter written as text with
2//! expressions inside it (`Olá {{ nome }}`) is checked, previewed and worked out.
3//!
4//! The expressions are of the criteria language (`ravenrs-criteria`). An item's own fields read
5//! as they are (`[nome]` or `{{ nome }}`), and four names reach further: `item` (the item),
6//! `nodes.<id>` (the first item a node that already ran gave), `vars` (the app's variables)
7//! and `run` (the run: its `started`, in milliseconds since 1970). Each evaluation gets its
8//! own environment, so runs on different threads never share one.
9
10use std::collections::BTreeMap;
11
12use ravenrs_criteria::{Environment, RenderError, Scope, Template, TemplateError};
13use serde_json::{Map, Value, json};
14
15use crate::definition::FlowDefinition;
16
17/// The name of the item itself.
18pub const ITEM: &str = "item";
19/// The name of the nodes that already ran, each by its id.
20pub const NODES: &str = "nodes";
21/// The name of the app's variables.
22pub const VARS: &str = "vars";
23/// The name of the run.
24pub const RUN: &str = "run";
25
26/// How many bytes a preview may grow to before it stops and says so.
27pub const PREVIEW_LIMIT: usize = 64 * 1024;
28
29/// What an expression reads for `item`: the item's own fields, and [`ITEM`], [`NODES`],
30/// [`VARS`] and [`RUN`].
31pub fn record(
32    item: &Value,
33    nodes: &BTreeMap<String, Value>,
34    vars: &Map<String, Value>,
35    run: &Value,
36) -> Value {
37    let mut map = item.as_object().cloned().unwrap_or_default();
38    map.insert(ITEM.into(), item.clone());
39    map.insert(NODES.into(), Value::Object(nodes.iter().map(|(k, v)| (k.clone(), v.clone())).collect()));
40    map.insert(VARS.into(), Value::Object(vars.clone()));
41    map.insert(RUN.into(), run.clone());
42    Value::Object(map)
43}
44
45/// The run as expressions read it.
46pub fn run_value(started_ms: i64) -> Value {
47    json!({ "started": started_ms })
48}

50/// The problems of `text` as a template of `flow`: what does not parse, and each
51/// `nodes.<id>` that names no node of the flow, marked where it is.
52pub fn check(text: &str, flow: &FlowDefinition) -> Vec<TemplateError> {
53    let template = match Template::parse(text) {
54        Ok(t) => t,
55        Err(errors) => return errors,
56    };
57    template
58        .references()
59        .into_iter()
60        .filter_map(|(path, at)| match path.as_slice() {
61            [head, id, ..] if head == NODES && flow.find(id).is_none() => {
62                Some(TemplateError { at, message: format!("there is no node {id} in this flow") })
63            }
64            _ => None,
65        })
66        .collect()
67}

69/// What a preview of a template says.
70#[derive(Clone, Debug, PartialEq, Eq)]
71pub enum Preview {
72    /// The text it works out to.
73    Text(String),
74    /// What is wrong with it, and where; nothing was worked out.
75    Problems(Vec<TemplateError>),
76    /// It parsed, but working it out failed, saying why: never shown as an empty value.
77    Failed(String),
78}
79
80/// `text` worked out over `record` (see [`record`]), checked first against `flow` when there
81/// is one, and stopped past [`PREVIEW_LIMIT`].
82pub fn preview(text: &str, record: &Value, flow: Option<&FlowDefinition>) -> Preview {
83    let problems = match flow {
84        Some(flow) => check(text, flow),
85        None => Template::parse(text).err().unwrap_or_default(),
86    };
87    if !problems.is_empty() {
88        return Preview::Problems(problems);
89    }
90    let Ok(template) = Template::parse(text) else { return Preview::Problems(Vec::new()) };
91    let env = Environment::new();
92    match template.render_within(&Scope::new(&env).row(record), PREVIEW_LIMIT) {
93        Ok(text) => Preview::Text(text),
94        Err(RenderError::TooLong(limit)) => {
95            Preview::Failed(format!("the text passed {limit} bytes, the limit of a preview"))
96        }
97        Err(error) => Preview::Failed(error.to_string()),
98    }
99}

101/// `text` worked out for one item of a run, a failure saying why.
102pub fn render(text: &str, record: &Value) -> Result<String, String> {
103    let template = Template::parse(text)
104        .map_err(|errors| errors.into_iter().map(|e| e.message).collect::<Vec<_>>().join("; "))?;
105    let env = Environment::new();
106    template.render(&Scope::new(&env).row(record)).map_err(|e| e.to_string())
107}
