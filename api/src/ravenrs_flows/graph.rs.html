1//! The bridge to the canvas: a flow drawn as a [`FlowGraph`] for the `FlowCanvas`, the drawing
2//! read back into the flow, and the [`FlowRunRules`] its debugger follows. Each node's setup
3//! rides in its card's `data` and the flow's name in the graph's, so copying, pasting and undoing
4//! on the canvas keep them; the drawing itself reads none of it, so no automation rule goes into
5//! the geometry. Above `definition` and `catalog`.
6

19/// The field of a card's `data`, and of the graph's, that holds what a flow keeps there.
20pub const GRAPH_KEY: &str = "flow";
21
22/// What the drawing puts before the name of an input that has an output of the same name. On
23/// the canvas a port's id is unique within its node, and in a flow an input and an output may
24/// share a name (`main`, in most kinds), so the drawn input is `in:main`.
25const INPUT_MARK: &str = "in:";
26
27/// The flow's name of a port drawn as `id`: the drawn id without the mark an input takes when
28/// an output shares its name.
29pub fn port_name(id: &str) -> &str {
30    id.strip_prefix(INPUT_MARK).unwrap_or(id)
31}
32
33/// The id the drawing gives the input `name` of a kind with these `outputs`.
34fn drawn_input(outputs: &[String], name: &str) -> String {
35    if outputs.iter().any(|o| o == name) { format!("{INPUT_MARK}{name}") } else { name.to_owned() }
36}
37
38/// The card of a node of the kind `spec` says, with its ports: what
39/// [`FlowDefinition::to_graph`] draws for each node, and what an app's toolbox offers to drag
40/// in, so both name the ports alike.
41pub fn card(spec: &NodeSpec, id: &str, title: impl Into<String>) -> FlowNode {
42    let mut card = FlowNode::new(id, &spec.kind, title, 0.0, 0.0);
43    for port in &spec.inputs {
44        card = card.port(FlowPort::input(drawn_input(&spec.outputs, port)));
45    }
46    for port in &spec.outputs {
47        card = card.port(FlowPort::output(port));
48    }
49    card
50}
51
52/// How far apart [`FlowDefinition::to_graph`] puts the nodes it lays out.
53const STEP: (f32, f32) = (260.0, 120.0);
54
55fn title_of(node: &NodeDef, catalog: &Catalog) -> String {
56    node.title.clone().unwrap_or_else(|| match catalog.get(&node.kind, node.version) {
57        Some(kind) => kind.spec().title,
58        None => node.kind.clone(),
59    })
60}
61
62fn edge_id(c: &Connection) -> String {
63    format!("{}.{}>{}.{}", c.from.node, c.from.port, c.to.node, c.to.port)
64}

66impl FlowDefinition {
67    /// The flow drawn for a `FlowCanvas`: a node card per node, with the ports of its kind,
68    /// titled by its title or its kind's, shown as disabled when it is, and the node's whole
69    /// setup in its `data`; one connection per connection; the flow's id, name and revision in
70    /// the graph's `data`. The nodes are laid out in columns by how far they are from a start,
71    /// which the app then moves as the person draws.
72    pub fn to_graph(&self, catalog: &Catalog) -> FlowGraph {
73        let depth = depths(self);
74        let mut rows: BTreeMap<usize, usize> = BTreeMap::new();
75        let mut graph = FlowGraph::new();
76        graph.data = json!({ GRAPH_KEY: { "id": self.id, "name": self.name, "revision": self.revision } });
77        for node in &self.nodes {
78            let col = depth.get(node.id.as_str()).copied().unwrap_or(0);
79            let row = rows.entry(col).or_default();
80            let mut card = match catalog.get(&node.kind, node.version) {
81                Some(kind) => self::card(&kind.spec(), &node.id, title_of(node, catalog)),
82                None => FlowNode::new(&node.id, &node.kind, title_of(node, catalog), 0.0, 0.0),
83            };
84            card.x = col as f32 * STEP.0;
85            card.y = *row as f32 * STEP.1;
86            *row += 1;
87            if node.disabled {
88                card.status = FlowNodeStatus::Disabled;
89            }
90            card.data = json!({ GRAPH_KEY: node });
91            graph.nodes.push(card);
92        }
93        for c in &self.connections {
94            let outputs = self
95                .nodes
96                .iter()
97                .find(|n| n.id == c.to.node)
98                .and_then(|n| catalog.get(&n.kind, n.version))
99                .map(|kind| kind.spec().outputs)
100                .unwrap_or_default();
101            graph.edges.push(FlowEdge::new(
102                edge_id(c),
103                FlowEndpoint::new(&c.from.node, &c.from.port),
104                FlowEndpoint::new(&c.to.node, drawn_input(&outputs, &c.to.port)),
105            ));
106        }
107        graph
108    }
109
110    /// The flow as the person left the drawing: its id, name and revision from the graph's
111    /// `data`, its nodes and connections the graph's, in the graph's order. Each node takes its
112    /// setup from its card's `data`, so a pasted card brings the setup of the node it was copied
113    /// from under its new id; a card with none (drawn from the toolbox) starts with no
114    /// parameters. A title the person typed over the kind's is kept, and the disabled look sets
115    /// `disabled`. Groups are drawing only, and are left out.
116    pub fn from_graph(graph: &FlowGraph, catalog: &Catalog) -> FlowDefinition {
117        let head = graph.data.get(GRAPH_KEY);
118        let text =
119            |key: &str| head.and_then(|h| h.get(key)).and_then(Value::as_str).unwrap_or_default().to_owned();
120        let mut flow = FlowDefinition::new(text("id"), text("name"));
121        flow.format = FORMAT;
122        flow.revision = head.and_then(|h| h.get("revision")).and_then(Value::as_u64).unwrap_or(0) as u32;
123        for card in graph.nodes.iter().filter(|n| !n.group) {
124            let kept = card
125                .data
126                .get(GRAPH_KEY)
127                .and_then(|setup| serde_json::from_value::<NodeDef>(setup.clone()).ok());
128            let mut node = kept.unwrap_or_else(|| NodeDef::new(&card.id, &card.kind));
129            node.id = card.id.clone();
130            node.kind = card.kind.clone();
131            node.title = None;
132            if card.title != title_of(&node, catalog) {
133                node.title = Some(card.title.clone());
134            }
135            node.disabled = card.status == FlowNodeStatus::Disabled;
136            flow.nodes.push(node);
137        }
138        let kept: BTreeSet<&str> = flow.nodes.iter().map(|n| n.id.as_str()).collect();
139        for edge in &graph.edges {
140            if kept.contains(edge.from.node.as_str()) && kept.contains(edge.to.node.as_str()) {
141                flow.connections.push(Connection {
142                    from: PortRef::new(&edge.from.node, &edge.from.port),
143                    to: PortRef::new(&edge.to.node, port_name(&edge.to.port)),
144                });
145            }
146        }
147        flow
148    }
149}

151impl Catalog {
152    /// How the `FlowCanvas` debugger moves through `flow`: a trigger starts, a branch chooses
153    /// one way out, a node that leads nowhere in the flow ends, any other node passes on; a node
154    /// the flow does not have (a note) and connections it does not have are left out.
155    ///
156    /// The rules are taken from the flow as it is now: when the person changes the flow, ask
157    /// for them again.
158    pub fn run_rules(&self, flow: &FlowDefinition) -> FlowRunRules {
159        let leads: BTreeSet<&str> = flow.connections.iter().map(|c| c.from.node.as_str()).collect();
160        let gates: BTreeMap<String, FlowGate> = flow
161            .nodes
162            .iter()
163            .map(|node| {
164                let spec = self.get(&node.kind, node.version).map(|k| k.spec());
165                let gate = match spec {
166                    Some(s) if s.trigger => FlowGate::Start,
167                    _ if !leads.contains(node.id.as_str()) => FlowGate::End,
168                    Some(s) if s.chooses => FlowGate::Choose,
169                    _ => FlowGate::Pass,
170                };
171                (node.id.clone(), gate)
172            })
173            .collect();
174        let edges: BTreeSet<String> = flow.connections.iter().map(edge_id).collect();
175        FlowRunRules {
176            gate: Rc::new(move |node: &FlowNode| gates.get(&node.id).copied().unwrap_or(FlowGate::Outside)),
177            follows: Rc::new(move |edge: &FlowEdge| {
178                edges.contains(&format!(
179                    "{}.{}>{}.{}",
180                    edge.from.node,
181                    edge.from.port,
182                    edge.to.node,
183                    port_name(&edge.to.port)
184                ))
185            }),
186        }
187    }
188}

190/// How many connections each node is from a node nothing feeds, by the longest way, going
191/// round no more than once.
192fn depths(flow: &FlowDefinition) -> BTreeMap<&str, usize> {
193    let mut depth: BTreeMap<&str, usize> = flow.nodes.iter().map(|n| (n.id.as_str(), 0)).collect();
194    for _ in 0..flow.nodes.len() {
195        let mut moved = false;
196        for c in &flow.connections {
197            let from = depth.get(c.from.node.as_str()).copied().unwrap_or(0);
198            if let Some(to) = depth.get_mut(c.to.node.as_str())
199                && *to < from + 1
200                && from + 1 < flow.nodes.len()
201            {
202                *to = from + 1;
203                moved = true;
204            }
205        }
206        if !moved {
207            break;
208        }
209    }
210    depth
211}
