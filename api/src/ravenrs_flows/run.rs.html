1//! Running a plan in memory, one node after another in the plan's order: the whole flow, from a
2//! node with what an earlier run left upstream, or one node alone. A loop goes round its body
3//! while it sends items into it. A node is tried again when it fails in a way worth it, a run
4//! can be cancelled from any thread, and time limits stop a node that takes too long. Same
5//! flow, items and clock, same run. Above `plan`, `catalog` and `time`.
6

21/// The most turns a loop takes, unless [`Executor::loop_limit`] says otherwise, before the run
22/// says it goes round for ever. Each turn the loop node gets everything that came back so far,
23/// so a loop of many turns costs more than its turns.
24pub const LOOP_LIMIT: usize = 10_000;

26/// How a node's part of a run went.
27#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
28#[serde(rename_all = "snake_case")]
29#[non_exhaustive]
30pub enum NodeStatus {
31    /// It ran.
32    Done,
33    /// It gave its pinned items, in a test, or the items the run was started with.
34    Pinned,
35    /// It is disabled and passed its first input on.
36    Disabled,
37    /// No item reached it, so it did not run.
38    Skipped,
39    /// It failed.
40    Failed,
41    /// It did not run again: what it gave is an earlier run's.
42    Reused,
43    /// It was cancelled while it ran.
44    Cancelled,
45    /// It took longer than its time limit, or than what was left of the run's.
46    TimedOut,
47    /// It called something and no one knows whether that happened. It was not tried again.
48    Uncertain,
49    /// It waits for a time or an event ([`NodeRun::wake`]); the run stopped there.
50    Waiting,
51}

53/// A node's part of a run: what reached it, what it gave, and how it went. For a node inside
54/// a loop, these are the last turn's.
55#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
56pub struct NodeRun {
57    /// How it went.
58    pub status: NodeStatus,
59    /// The items that reached each input.
60    pub inputs: Ports,
61    /// The items it gave on each output, placed and traced.
62    pub outputs: Ports,
63    /// Why it failed, when it did.
64    pub error: Option<String>,
65    /// How many times it ran in this run: more than once inside a loop.
66    pub runs: usize,
67    /// How many tries its last run took: more than 1 when it was tried again.
68    pub attempts: u32,
69    /// Whether its result came from asking the kind if a call of unknown result happened.
70    pub reconciled: bool,
71    /// When a waiting node is to be woken.
72    #[serde(default, skip_serializing_if = "Option::is_none")]
73    pub wake: Option<Wake>,
74}

76/// How a run ended.
77#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
78#[serde(rename_all = "snake_case", tag = "end")]
79#[non_exhaustive]
80pub enum RunOutcome {
81    /// Every node that could run, ran (the default of a run not started).
82    #[default]
83    Finished,
84    /// It stopped at a node that failed with [`ErrorPolicy::Stop`].
85    Stopped {
86        /// The node.
87        node: String,
88        /// Why it failed.
89        reason: String,
90    },
91    /// It was cancelled.
92    Cancelled {
93        /// Why, as the token was told.
94        reason: String,
95    },
96    /// The run took longer than its time limit.
97    TimedOut {
98        /// What took too long.
99        reason: String,
100    },
101    /// A node waits; the run goes on when a durable run wakes it.
102    Waiting {
103        /// The node.
104        node: String,
105        /// What wakes it.
106        wake: Wake,
107    },

110/// A step of a run, as [`Executor::on_step`] hears it: what a durable run writes down so that it
111/// can start again where it was.
112#[derive(Clone, Copy, Debug)]
113#[non_exhaustive]
114pub enum Step<'r> {
115    /// A node is about to run.
116    Started {
117        /// The node.
118        node: &'r str,
119    },
120    /// A node's part is over: it ran, was skipped, failed or waits.
121    Settled {
122        /// The node.
123        node: &'r str,
124        /// Its part.
125        part: &'r NodeRun,
126    },
127    /// A loop went round for the last time: it and its body are over.
128    LoopDone {
129        /// The loop node.
130        head: &'r str,
131    },
132}

134/// A run of a flow: each node's part, in the order they ran, and how it ended.
135#[derive(Clone, Debug, Default, PartialEq)]
136pub struct Run {
137    /// Each node's part, by id.
138    pub nodes: BTreeMap<String, NodeRun>,
139    /// The ids of the nodes that took part, in the order they first ran.
140    pub order: Vec<String>,
141    /// The node the run stopped at, when one failed with [`ErrorPolicy::Stop`].
142    pub stopped_at: Option<String>,
143    /// The clock's time when the run started.
144    pub started_ms: i64,
145    /// How it ended.
146    pub outcome: RunOutcome,
147    deadline_ms: Option<i64>,
148}
149
150impl Run {
151    /// The items `node` gave on `port`; none when it gave none or did not run.
152    pub fn output(&self, node: &str, port: &str) -> &[Item] {
153        self.nodes.get(node).and_then(|n| n.outputs.get(port)).map(Vec::as_slice).unwrap_or(&[])
154    }
155
156    /// How `node`'s part went, when it took part.
157    pub fn status(&self, node: &str) -> Option<NodeStatus> {
158        self.nodes.get(node).map(|n| n.status)
159    }
160
161    /// Whether it ended with every node that could run having run.
162    pub fn succeeded(&self) -> bool {
163        self.outcome == RunOutcome::Finished
164    }
165
166    /// Records `part` of `node`, counting one more time it ran.
167    fn record(&mut self, node: &str, mut part: NodeRun) {
168        match self.nodes.get(node) {
169            Some(before) => part.runs = before.runs + 1,
170            None => self.order.push(node.to_owned()),
171        }
172        self.nodes.insert(node.to_owned(), part);
173    }
174}

338    /// Goes on with a run that stopped: the nodes whose part is in `checkpoint` keep it and do
339    /// not run again (a waiting node runs again, to see whether its wait is over), the nodes in
340    /// `in_progress` were running when it stopped and are asked to reconcile first
341    /// ([`NodeKind::reconcile`]), and the rest run. The run keeps the time it first started.
342    pub fn resume(&self, checkpoint: &Run, in_progress: &BTreeSet<String>) -> Run {
343        let all: BTreeSet<String> = self.plan.order().iter().cloned().collect();
344        self.go(&all, &Run::default(), Some((checkpoint, in_progress)))
345    }

347    /// Runs `node` and every node it leads to; the nodes upstream do not run again, and what
348    /// they gave is taken from `previous`. A node inside a loop's body runs that turn once.
349    pub fn run_from(&self, node: &str, previous: &Run) -> Result<Run, RunError> {
350        self.known(node)?;
351        let ahead: BTreeSet<String> = self.plan.downstream(node).into_iter().collect();
352        self.needs(&ahead, previous)?;
353        Ok(self.go(&ahead, previous, None))

365    fn known(&self, node: &str) -> Result<(), RunError> {
366        match self.plan.definition().find(node) {
367            Some(_) => Ok(()),
368            None => Err(RunError::UnknownNode { node: node.to_owned() }),
369        }

372    /// Every node feeding the nodes to run, and not running itself, took part in `previous`.
373    fn needs(&self, running: &BTreeSet<String>, previous: &Run) -> Result<(), RunError> {
374        for id in running {
375            for c in self.plan.feeds(id) {
376                if !running.contains(&c.from.node) && !previous.nodes.contains_key(&c.from.node) {
377                    return Err(RunError::NoPrevious { node: c.from.node.clone() });
378                }
379            }
380        }
381        Ok(())
382    }

384    /// Why the run must stop now, if it must: it was cancelled, or its time is up.
385    fn halted(&self, run: &Run) -> Option<RunOutcome> {
386        if self.token.is_cancelled() {
387            return Some(RunOutcome::Cancelled {
388                reason: self.token.reason().unwrap_or_else(|| "cancelled".to_owned()),
389            });
390        }
391        match (run.deadline_ms, self.time_limit_ms) {
392            (Some(deadline), Some(limit)) if self.clock.now_ms() >= deadline => {
393                Some(RunOutcome::TimedOut { reason: format!("the run took more than {limit} ms") })
394            }
395            _ => None,
396        }
397    }

399    /// Runs the nodes in `running`, in the plan's order, reusing from `previous` the parts of
400    /// the nodes upstream that feed them.
401    fn go(
402        &self,
403        running: &BTreeSet<String>,
404        previous: &Run,
405        resumed: Option<(&Run, &BTreeSet<String>)>,
406    ) -> Run {
407        let started = resumed.map_or_else(|| self.clock.now_ms(), |(checkpoint, _)| checkpoint.started_ms);
408        let mut run = Run {
409            started_ms: started,
410            deadline_ms: self.time_limit_ms.map(|l| started.saturating_add(l as i64)),
411            ..Run::default()
412        };
413        let upstream: BTreeSet<&str> = running
414            .iter()
415            .flat_map(|id| self.plan.feeds(id))
416            .map(|c| c.from.node.as_str())
417            .filter(|n| !running.contains(*n))
418            .collect();
419        let mut gone_round: BTreeSet<&str> = BTreeSet::new();
420        for id in self.plan.order() {
421            if gone_round.contains(id.as_str()) {
422                continue;
423            }
424            if upstream.contains(id.as_str()) {
425                if let Some(earlier) = previous.nodes.get(id) {
426                    run.record(id, NodeRun { status: NodeStatus::Reused, runs: 0, ..earlier.clone() });
427                }
428                continue;
429            }
430            if !running.contains(id) {
431                continue;
432            }
433            if let Some(outcome) = self.halted(&run) {
434                run.outcome = outcome;
435                break;
436            }
437            let body: Vec<&String> =
438                self.plan.loop_body(id).iter().filter(|b| running.contains(*b)).collect();
439            if let Some((checkpoint, busy)) = resumed {
440                if let Some(kept) = checkpoint.nodes.get(id).filter(|k| k.status != NodeStatus::Waiting) {
441                    run.record(id, kept.clone());
442                    for b in &body {
443                        if let Some(part) = checkpoint.nodes.get(*b) {
444                            run.record(b, part.clone());
445                        }
446                    }
447                    gone_round.extend(body.iter().map(|b| b.as_str()));
448                    continue;
449                }
450                if busy.contains(id)
451                    && body.is_empty()
452                    && let Some(next) = self.reconciled(id, &mut run)
453                {
454                    if next == Next::Stop {
455                        break;
456                    }
457                    continue;
458                }
459            }
460            let next =
461                if body.is_empty() { self.once(id, 0, &mut run) } else { self.round(id, &body, &mut run) };
462            gone_round.extend(body.iter().map(|b| b.as_str()));
463            if next == Next::Stop {
464                break;
465            }
466        }
467        run
468    }

470    /// A node that was running when its run stopped, asked whether its call happened: its part
471    /// when the kind knows, `None` when it did not happen and the node is to run.
472    fn reconciled(&self, node: &str, run: &mut Run) -> Option<Next> {
473        let def = self.plan.definition().find(node)?;
474        let kind = self.catalog.shared(&def.kind, def.version)?;
475        let inputs = self.inputs_of(node, run);
476        let cx = self.context(def, &inputs, run.started_ms, 0, 1, None, self.token.child());
477        let part = match kind.reconcile(&cx) {
478            Reconciled::NotDone => return None,
479            Reconciled::Done(given) => self.given(def, inputs.clone(), given, 1, true),
480            _ => {
481                let unknown = NodeError::unknown(
482                    "it was running when the run stopped, and whether it happened is not known",
483                );
484                self.failure(def, inputs.clone(), unknown, 1, NodeStatus::Uncertain)
485            }
486        };
487        Some(self.settle(def, part, run))
488    }

490    /// One part of `node`, at the turn `iteration`, recorded in `run`.
491    fn once(&self, node: &str, iteration: usize, run: &mut Run) -> Next {
492        let Some(def) = self.plan.definition().find(node) else { return Next::Go };
493        self.tell(Step::Started { node });
494        let inputs = self.inputs_of(node, run);
495        self.seen.replace(Arc::new(first_items(run)));
496        let part = self.node(def, inputs, run, iteration);
497        self.settle(def, part, run)
498    }

500    /// Records `part` and says whether the run stops there, and how it ended if it does.
501    fn settle(&self, def: &NodeDef, part: NodeRun, run: &mut Run) -> Next {
502        let reason = part.error.clone().unwrap_or_default();
503        let outcome = match part.status {
504            NodeStatus::Waiting => Some(RunOutcome::Waiting {
505                node: def.id.clone(),
506                wake: part.wake.clone().unwrap_or_default(),
507            }),
508            NodeStatus::Cancelled => Some(RunOutcome::Cancelled { reason }),
509            NodeStatus::TimedOut if self.halted(run).is_some() => self.halted(run),
510            NodeStatus::Failed | NodeStatus::TimedOut | NodeStatus::Uncertain
511                if def.on_error == ErrorPolicy::Stop =>
512            {
513                run.stopped_at = Some(def.id.clone());
514                Some(RunOutcome::Stopped { node: def.id.clone(), reason })
515            }
516            _ => None,
517        };
518        self.tell(Step::Settled { node: &def.id, part: &part });
519        run.record(&def.id, part);
520        match outcome {
521            Some(outcome) => {
522                run.outcome = outcome;
523                Next::Stop
524            }
525            None => Next::Go,
526        }
527    }

529    /// The loop `head` going round `body`: each turn the head runs with what came back so far,
530    /// and the body runs while the head sends it items.
531    fn round(&self, head: &str, body: &[&String], run: &mut Run) -> Next {
532        let Some(def) = self.plan.definition().find(head) else { return Next::Go };
533        let inside: BTreeSet<&str> = body.iter().map(|b| b.as_str()).collect();
534        let into_body: BTreeSet<&str> = body
535            .iter()
536            .flat_map(|b| self.plan.feeds(b))
537            .filter(|c| c.from.node == head)
538            .map(|c| c.from.port.as_str())
539            .collect();
540        let back: Vec<_> = self.plan.back_connections().iter().filter(|c| c.to.node == head).collect();
541        let mut came_back = Ports::new();
542        let limit = self.loop_limit;
543        for iteration in 0..=limit {
544            if iteration == limit {
545                let part = failed(Ports::new(), format!("the loop went round {limit} times without ending"));
546                return self.settle(def, part, run);
547            }
548            if let Some(outcome) = self.halted(run) {
549                run.outcome = outcome;
550                return Next::Stop;
551            }
552            let mut inputs = self.inputs_of(head, run);
553            for (port, items) in &came_back {
554                inputs.entry(port.clone()).or_default().extend(items.iter().cloned());
555            }
556            self.tell(Step::Started { node: head });
557            let part = self.node(def, inputs, run, iteration);
558            let sends = part
559                .outputs
560                .iter()
561                .any(|(port, items)| !items.is_empty() && into_body.contains(port.as_str()));
562            if self.settle(def, part, run) == Next::Stop {
563                return Next::Stop;
564            }
565            if !sends {
566                self.tell(Step::LoopDone { head });
567                return Next::Go;
568            }
569            for b in body {
570                if let Some(outcome) = self.halted(run) {
571                    run.outcome = outcome;
572                    return Next::Stop;
573                }
574                if self.once(b, iteration, run) == Next::Stop {
575                    return Next::Stop;
576                }
577            }
578            for c in &back {
579                if inside.contains(c.from.node.as_str()) {
580                    let items = run.output(&c.from.node, &c.from.port).to_vec();
581                    came_back.entry(c.to.port.clone()).or_default().extend(items);
582                }
583            }
584        }
585        Next::Go
586    }

588    /// The items reaching each input of `node`, from what the nodes feeding it gave in `run`.
589    fn inputs_of(&self, node: &str, run: &Run) -> Ports {
590        let mut inputs = Ports::new();
591        for c in self.plan.feeds(node) {
592            let items = run.output(&c.from.node, &c.from.port);
593            inputs.entry(c.to.port.clone()).or_default().extend(items.iter().cloned());
594        }
595        inputs
596    }

598    /// One node's part: the items it starts with, its pinned items, skipped, disabled, or run,
599    /// tried again as its setup says.
600    fn node(&self, node: &NodeDef, inputs: Ports, run: &Run, iteration: usize) -> NodeRun {
601        let Some(spec) = self.plan.spec(&node.id) else {
602            return failed(inputs, format!("the kind {} is not in the plan", node.kind));
603        };
604        let first_out = self.plan.outputs(&node.id).first().cloned();
605        if let Some(items) = self.starts.get(&node.id) {
606            let given = first_out.map(|p| Ports::from([(p, items.clone())])).unwrap_or_default();
607            return done(NodeStatus::Pinned, inputs, place(&node.id, given));
608        }
609        let fed = !self.plan.feeds(&node.id).is_empty();
610        let nothing_came = inputs.values().all(Vec::is_empty);
611        if self.mode == Mode::Test && !node.pinned.is_empty() {
612            return done(NodeStatus::Pinned, inputs, place(&node.id, node.pinned.clone()));
613        }
614        if !spec.trigger && (!fed || nothing_came) {
615            return done(NodeStatus::Skipped, inputs, Ports::new());
616        }
617        if node.disabled {
618            let first_in = spec.inputs.first().and_then(|p| inputs.get(p)).cloned().unwrap_or_default();
619            let given = first_out.map(|p| Ports::from([(p, first_in)])).unwrap_or_default();
620            return done(NodeStatus::Disabled, inputs, place(&node.id, given));
621        }
622        let Some(kind) = self.catalog.shared(&node.kind, node.version) else {
623            return failed(
624                inputs,
625                format!("the kind {} (version {}) is not in the catalog", node.kind, node.version),
626            );
627        };
628        let retry = node.retry.unwrap_or(Retry::new(1, 0));
629        let mut attempt = 1;
630        loop {
631            let deadline = self.deadline_of(node, run);
632            let result = self.call(&kind, node, &inputs, run.started_ms, iteration, attempt, deadline);
633            let error = match result {
634                Ok(given) => return self.given(node, inputs, given, attempt, false),
635                Err(e) if e.failure == Failure::Suspended => {
636                    return NodeRun {
637                        attempts: attempt,
638                        wake: e.wake,
639                        ..done(NodeStatus::Waiting, inputs, Ports::new())
640                    };
641                }
642                Err(e) => e,
643            };
644            let again = match error.failure {
645                Failure::Retryable => true,
646                Failure::TimedOut => retry.on_timeout && self.halted(run).is_none(),
647                Failure::Unknown => {
648                    let cx = self.context(
649                        node,
650                        &inputs,
651                        run.started_ms,
652                        iteration,
653                        attempt,
654                        deadline,
655                        self.token.child(),
656                    );
657                    match kind.reconcile(&cx) {
658                        Reconciled::Done(given) => return self.given(node, inputs, given, attempt, true),
659                        Reconciled::NotDone => true,
660                        _ => return self.failure(node, inputs, error, attempt, NodeStatus::Uncertain),
661                    }
662                }
663                _ => false,
664            };
665            if !again || attempt >= retry.tries {
666                let status = match error.failure {
667                    Failure::Cancelled => NodeStatus::Cancelled,
668                    Failure::TimedOut => NodeStatus::TimedOut,
669                    _ => NodeStatus::Failed,
670                };
671                return self.failure(node, inputs, error, attempt, status);
672            }
673            attempt += 1;
674            if !self.wait(retry.wait_before(attempt)) {
675                let why = self.token.reason().unwrap_or_else(|| "cancelled".to_owned());
676                return self.failure(
677                    node,
678                    inputs,
679                    NodeError::cancelled(why),
680                    attempt - 1,
681                    NodeStatus::Cancelled,
682                );
683            }
684        }
685    }

687    /// When `node` must end by, on the run's clock: its own limit, or what is left of the run's.
688    fn deadline_of(&self, node: &NodeDef, run: &Run) -> Option<i64> {
689        let own = node.timeout_ms.map(|t| self.clock.now_ms().saturating_add(t as i64));
690        match (own, run.deadline_ms) {
691            (Some(a), Some(b)) => Some(a.min(b)),
692            (a, b) => a.or(b),
693        }
694    }
695
696    /// Waits `ms` on the run's clock, a little at a time; `false` when the run was cancelled.
697    fn wait(&self, ms: u64) -> bool {
698        let mut left = ms;
699        while left > 0 {
700            if self.token.is_cancelled() {
701                return false;
702            }
703            let now = left.min(10);
704            self.clock.sleep_ms(now);
705            left -= now;
706        }
707        !self.token.is_cancelled()
708    }

710    #[allow(clippy::too_many_arguments, reason = "the parts of one node's context")]
711    fn context<'n>(
712        &self,
713        node: &'n NodeDef,
714        inputs: &'n Ports,
715        now_ms: i64,
716        iteration: usize,
717        attempt: u32,
718        deadline: Option<i64>,
719        token: CancelToken,
720    ) -> NodeContext<'n> {
721        NodeContext::new(node, inputs, now_ms, self.mode)
722            .at_iteration(iteration)
723            .at_attempt(attempt)
724            .with_clock(self.clock.clone())
725            .with_token(token)
726            .with_deadline(deadline)
727            .with_events(self.events.clone())
728            .woken_from(self.woken.get(&node.id).cloned())
729            .with_nodes(self.seen.borrow().clone())
730    }

732    /// One try of `kind` on `node`. With a time limit or a token to watch, the try runs on a
733    /// thread of its own: when the run is cancelled or the time is up, the node's token is
734    /// cancelled and the run stops waiting for it.
735    #[allow(clippy::too_many_arguments, reason = "the parts of one node's context")]
736    fn call(
737        &self,
738        kind: &Arc<dyn NodeKind>,
739        node: &NodeDef,
740        inputs: &Ports,
741        now_ms: i64,
742        iteration: usize,
743        attempt: u32,
744        deadline: Option<i64>,
745    ) -> Result<Ports, NodeError> {
746        let token = self.token.child();
747        if !self.watched && deadline.is_none() || cfg!(target_arch = "wasm32") {
748            return kind.execute(&self.context(node, inputs, now_ms, iteration, attempt, deadline, token));
749        }
750        #[cfg(not(target_arch = "wasm32"))]
751        {
752            self.on_thread(kind, node, inputs, now_ms, iteration, attempt, deadline, token)
753        }
754        #[cfg(target_arch = "wasm32")]
755        {
756            unreachable!("a browser runs every node on its own thread")
757        }
758    }
759
760    #[cfg(not(target_arch = "wasm32"))]
761    #[allow(clippy::too_many_arguments, reason = "the parts of one node's context")]
762    fn on_thread(
763        &self,
764        kind: &Arc<dyn NodeKind>,
765        node: &NodeDef,
766        inputs: &Ports,
767        now_ms: i64,
768        iteration: usize,
769        attempt: u32,
770        deadline: Option<i64>,
771        token: CancelToken,
772    ) -> Result<Ports, NodeError> {
773        use std::sync::mpsc::{RecvTimeoutError, channel};
774        use std::time::{Duration, Instant};
775
776        let (tx, rx) = channel();
777        let (kind, node_def, given, clock, mode, own) =
778            (kind.clone(), node.clone(), inputs.clone(), self.clock.clone(), self.mode, token.clone());
779        let (events, woken, seen) =
780            (self.events.clone(), self.woken.get(&node.id).cloned(), self.seen.borrow().clone());
781        std::thread::spawn(move || {
782            let cx = NodeContext::new(&node_def, &given, now_ms, mode)
783                .at_iteration(iteration)
784                .at_attempt(attempt)
785                .with_clock(clock)
786                .with_token(own)
787                .with_deadline(deadline)
788                .with_events(events)
789                .with_nodes(seen)
790                .woken_from(woken);
791            let _ = tx.send(kind.execute(&cx));
792        });
793        let began = Instant::now();
794        let real_limit = deadline.map(|d| Duration::from_millis((d - self.clock.now_ms()).max(0) as u64));
795        loop {
796            match rx.recv_timeout(Duration::from_millis(5)) {
797                Ok(result) => return result,
798                Err(RecvTimeoutError::Disconnected) => {
799                    return Err(NodeError::new("the node stopped without answering"));
800                }
801                Err(RecvTimeoutError::Timeout) => {
802                    if self.token.is_cancelled() {
803                        let why = self.token.reason().unwrap_or_else(|| "cancelled".to_owned());
804                        token.cancel(why.clone());
805                        return Err(NodeError::cancelled(why));
806                    }
807                    let real = real_limit.is_some_and(|l| began.elapsed() >= l);
808                    let on_clock = deadline.is_some_and(|d| self.clock.now_ms() >= d);
809                    if real || on_clock {
810                        token.cancel("the time limit was reached");
811                        return Err(NodeError::timed_out(format!(
812                            "it took more than {} ms",
813                            real_limit.map_or(0, |l| l.as_millis())
814                        )));
815                    }
816                }
817            }
818        }
819    }

821    /// The part of a node that answered, checked against its outputs and its error policy.
822    fn given(&self, node: &NodeDef, inputs: Ports, given: Ports, attempt: u32, reconciled: bool) -> NodeRun {
823        let outputs = self.plan.outputs(&node.id);
824        if let Some(port) = given.keys().find(|p| *p != ERROR_PORT && !outputs.contains(p)) {
825            return NodeRun {
826                attempts: attempt,
827                ..failed(inputs, format!("the kind {} has no output {port}", node.kind))
828            };
829        }
830        let failures = given.get(ERROR_PORT).map_or(0, Vec::len);
831        let mut part = done(NodeStatus::Done, inputs, place(&node.id, given));
832        part.attempts = attempt;
833        part.reconciled = reconciled;
834        if failures > 0 && node.on_error == ErrorPolicy::Stop {
835            let why = part.outputs[ERROR_PORT][0]
836                .get("error")
837                .and_then(|e| e.as_str())
838                .unwrap_or("it failed")
839                .to_owned();
840            part.status = NodeStatus::Failed;
841            part.error = Some(format!("{failures} item(s) failed; the first: {why}"));
842        }
843        part
844    }
845
846    /// The part of a node that failed as a whole: with [`ErrorPolicy::Route`], every item that
847    /// reached it goes to its error output.
848    fn failure(
849        &self,
850        node: &NodeDef,
851        inputs: Ports,
852        error: NodeError,
853        attempt: u32,
854        status: NodeStatus,
855    ) -> NodeRun {
856        let routed = node.on_error == ErrorPolicy::Route && status != NodeStatus::Cancelled;
857        let outputs = if routed {
858            let sent: Vec<Item> = inputs.values().flatten().map(|i| i.failed(&error.message)).collect();
859            place(&node.id, Ports::from([(ERROR_PORT.to_owned(), sent)]))
860        } else {
861            Ports::new()
862        };
863        NodeRun { status, error: Some(error.message), attempts: attempt, ..done(status, inputs, outputs) }
864    }
865}
866
867fn done(status: NodeStatus, inputs: Ports, outputs: Ports) -> NodeRun {
868    NodeRun { status, inputs, outputs, error: None, runs: 1, attempts: 1, reconciled: false, wake: None }
869}
870
871fn failed(inputs: Ports, error: String) -> NodeRun {
872    NodeRun { error: Some(error), ..done(NodeStatus::Failed, inputs, Ports::new()) }
873}

875/// The items of each port put where they are, each saying where it came from.
876fn place(node: &str, given: Ports) -> Ports {
877    given
878        .into_iter()
879        .map(|(port, items)| {
880            let placed = items.into_iter().enumerate().map(|(i, item)| item.placed(node, &port, i)).collect();
881            (port, placed)
882        })
883        .collect()
884}

886/// The first item each node of `run` gave, by id: on `main` when it gave there, otherwise on
887/// the first port that has one.
888fn first_items(run: &Run) -> BTreeMap<String, serde_json::Value> {
889    run.nodes
890        .iter()
891        .filter_map(|(id, part)| {
892            let first = part
893                .outputs
894                .get(crate::nodes::MAIN)
895                .and_then(|items| items.first())
896                .or_else(|| part.outputs.values().find_map(|items| items.first()))?;
897            Some((id.clone(), first.json.clone()))
898        })
899        .collect()
