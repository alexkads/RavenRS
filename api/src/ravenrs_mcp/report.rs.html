18/// Keys of the layout whose text is an expression.
19const EXPRESSION_KEYS: [&str; 7] =
20    ["expression", "filter", "visible", "argument", "value", "urlExpression", "condition"];

22/// A field given by name when there is no sample data.
23#[derive(Debug, Deserialize, schemars::JsonSchema)]
24pub struct FieldInput {
25    /// The path used in expressions, such as `Customer` or `Customer.Name`.
26    pub name: String,
27    /// text, number, date or boolean. Defaults to text.
28    pub kind: Option<String>,
29}

31/// A column of a table layout.
32#[derive(Debug, Deserialize, schemars::JsonSchema)]
33pub struct ColumnInput {
34    /// The field it shows.
35    pub field: String,
36    /// The header; a caption made from the field name when absent.
37    pub caption: Option<String>,
38    /// A .NET-style format, such as `C2`, `N0`, `P1` or `dd/MM/yyyy`.
39    pub format: Option<String>,
40}

42/// What a new table layout is made of.
43#[derive(Debug, Default, Deserialize, schemars::JsonSchema)]
44pub struct CreateInput {
45    /// The title printed at the top, also the layout's name.
46    pub title: String,
47    /// Culture of number and date formats, such as `pt-BR` or `en-US`. Defaults to `en-US`.
48    pub culture: Option<String>,
49    /// Rows of data as a JSON array of objects, embedded in the layout. Its fields are read from it.
50    pub data: Option<Value>,
51    /// Instead of embedded rows: an address the report server fetches when the report runs. Give `fields` with it.
52    pub data_url: Option<String>,
53    /// The fields, when there is no `data` to read them from.
54    pub fields: Option<Vec<FieldInput>>,
55    /// A path to the list of rows inside the data, such as `orders`.
56    pub data_member: Option<String>,
57    /// The columns, in order. Defaults to every field that is not a list.
58    pub columns: Option<Vec<ColumnInput>>,
59    /// Groups rows by this field, with a header and subtotals.
60    pub group_by: Option<String>,
61    /// Number fields that get subtotals and a grand total.
62    pub totals: Option<Vec<String>>,
63    /// Turns the page sideways. More than six columns turn it anyway.
64    pub landscape: Option<bool>,
65    /// The color of the column headers, as `#rrggbb`.
66    pub accent: Option<String>,
67    /// The name of the data source. Defaults to `Data`.
68    pub source_name: Option<String>,
69}

71fn kind_of(name: Option<&str>) -> Result<FieldKind, String> {
72    match name.unwrap_or("text").to_lowercase().as_str() {
73        "text" | "string" => Ok(FieldKind::Text),
74        "number" | "decimal" | "integer" => Ok(FieldKind::Number),
75        "date" | "datetime" => Ok(FieldKind::Date),
76        "boolean" | "bool" => Ok(FieldKind::Boolean),
77        other => Err(format!("unknown field kind '{other}': use text, number, date or boolean")),
78    }
79}

81/// One row with a plausible value for each field, so the layout has something to preview.
82fn sample_row(fields: &[FieldInfo]) -> Value {
83    let mut row = serde_json::Map::new();
84    for f in fields {
85        let value = match f.kind {
86            FieldKind::Number => json!(0),
87            FieldKind::Date => json!("2026-01-31"),
88            FieldKind::Boolean => json!(true),
89            _ => json!(wizard::caption(&f.path)),
90        };
91        row.insert(f.path.clone(), value);
92    }
93    Value::Array(vec![Value::Object(row)])
94}

96/// A table layout built from `input`, with the check of what came out.
97pub fn create(input: &CreateInput) -> Result<Value, String> {
98    if input.title.trim().is_empty() {
99        return Err("give the report a title".into());
100    }
101    let member = input.data_member.as_deref();
102    let (fields, kind) = match (&input.data, &input.fields) {
103        (Some(data), _) => {
104            if !data.is_array() && member.is_none() {
105                return Err("data must be an array of rows, or give data_member to point at one".into());
106            }
107            (wizard::fields(data, member), DataSourceKind::Json { data: data.clone() })
108        }
109        (None, Some(given)) => {
110            let fields = given
111                .iter()
112                .map(|f| Ok(FieldInfo { path: f.name.clone(), kind: kind_of(f.kind.as_deref())? }))
113                .collect::<Result<Vec<_>, String>>()?;
114            let kind = match &input.data_url {
115                Some(url) => DataSourceKind::Url { url: url.clone(), headers: BTreeMap::new() },
116                None => DataSourceKind::Json { data: sample_row(&fields) },
117            };
118            (fields, kind)
119        }
120        (None, None) => return Err("give data (rows to embed) or fields (with or without data_url)".into()),
121    };
122    if fields.is_empty() {
123        return Err("no fields found: the rows must be JSON objects".into());
124    }
125    let find = |name: &str| fields.iter().find(|f| f.path == name).cloned();
126    let unknown = |name: &str| format!("unknown field '{name}'; the fields are: {}", names(&fields));
127    let (columns, words): (Vec<FieldInfo>, Vec<ColumnWords>) = match &input.columns {
128        Some(given) => given
129            .iter()
130            .map(|c| {
131                let field = find(&c.field).ok_or_else(|| unknown(&c.field))?;
132                Ok((field, ColumnWords { caption: c.caption.clone(), format: c.format.clone() }))
133            })
134            .collect::<Result<Vec<_>, String>>()?
135            .into_iter()
136            .unzip(),
137        None => fields
138            .iter()
139            .filter(|f| f.kind != FieldKind::Collection)
140            .map(|f| (f.clone(), ColumnWords::default()))
141            .unzip(),
142    };
143    if let Some(group) = &input.group_by
144        && find(group).is_none()
145    {
146        return Err(unknown(group));
147    }
148    let totals = input.totals.clone().unwrap_or_default();
149    for t in &totals {
150        match find(t) {
151            Some(f) if f.kind == FieldKind::Number => {}
152            Some(_) => return Err(format!("'{t}' is not a number field, so it cannot be totaled")),
153            None => return Err(unknown(t)),
154        }
155    }
156    let page =
157        if input.landscape == Some(true) { PageSettings::a4().landscape() } else { PageSettings::a4() };
158    let defaults = WizardOptions::default();
159    let options = WizardOptions {
160        title: input.title.clone(),
161        culture: input.culture.clone().unwrap_or(defaults.culture),
162        columns,
163        group_by: input.group_by.clone(),
164        totals,
165        page,
166        accent: input.accent.clone().unwrap_or(defaults.accent),
167    };
168    let source = DataSource { name: input.source_name.clone().unwrap_or_else(|| "Data".into()), kind };
169    let mut report = wizard::table_report_with(source, &options, &words);
170    if let Some(member) = member {
171        report = report.data_member(member);
172    }
173    let layout: Value = serde_json::from_str(&report.to_json()).map_err(|e| e.to_string())?;
174    Ok(json!({
175        "file_name": format!("{}.rvreport.json", slug(&input.title)),
176        "fields": fields.iter().map(|f| json!({ "name": f.path, "kind": format!("{:?}", f.kind) })).collect::<Vec<_>>(),
177        "check": check_report(&report, None, &HashMap::new()),
178        "layout": layout,
179    }))
180}

182fn names(fields: &[FieldInfo]) -> String {
183    fields.iter().map(|f| f.path.as_str()).collect::<Vec<_>>().join(", ")

186/// A file name made of the title's letters and digits.
187fn slug(title: &str) -> String {
188    let mut out = String::new();
189    for c in title.to_lowercase().chars() {
190        if c.is_alphanumeric() {
191            out.push(c);
192        } else if !out.ends_with('-') && !out.is_empty() {
193            out.push('-');
194        }
195    }
196    let out = out.trim_end_matches('-').to_owned();
197    if out.is_empty() { "report".into() } else { out }
198}

200/// What expressions in this report may refer to.
201fn schema_of(report: &Report, data: Option<&Value>, rows_of: Option<&str>) -> Schema {
202    let kind = |k: FieldKind| match k {
203        FieldKind::Number => FieldType::Number,
204        FieldKind::Date => FieldType::Date,
205        FieldKind::Boolean => FieldType::Boolean,
206        FieldKind::Text => FieldType::Text,
207        _ => FieldType::Any,
208    };
209    let mut schema = Schema::new();
210    if let Some(data) = data {
211        let member = report.data_member.as_deref();
212        let first = wizard::rows_of(data, member).first().copied().cloned();
213        for f in wizard::fields(data, member) {
214            if f.kind == FieldKind::Collection {
215                let items = first.as_ref().and_then(|row| row.get(&f.path)).cloned().unwrap_or_default();
216                let rows = wizard::fields(&items, None)
217                    .into_iter()
218                    .fold(Schema::new(), |s, sub| s.field(&sub.path, kind(sub.kind)));
219                schema = schema.collection(&f.path, rows);
220                // Inside a detail report the rows are the collection's.
221                if rows_of == Some(f.path.as_str()) {
222                    for sub in wizard::fields(&items, None) {
223                        schema = schema.field(&sub.path, kind(sub.kind));
224                    }
225                }
226            } else {
227                schema = schema.field(&f.path, kind(f.kind));
228            }
229        }
230    }
231    for c in &report.calculated_fields {
232        schema = schema.field(&c.name, FieldType::Any);
233    }
234    for p in &report.parameters {
235        let k = match p.kind {
236            ParameterKind::Number => FieldType::Number,
237            ParameterKind::Date => FieldType::Date,
238            ParameterKind::Boolean => FieldType::Boolean,
239            _ => FieldType::Text,
240        };
241        schema = schema.parameter(&p.name, k);
242    }
243    schema
244}

246/// Every expression in the layout, with where it is.
247fn expressions(value: &Value, path: &str, out: &mut Vec<(String, String)>) {
248    match value {
249        Value::Object(map) => {
250            for (key, v) in map {
251                let here = if path.is_empty() { key.clone() } else { format!("{path}.{key}") };
252                match v {
253                    Value::String(text)
254                        if EXPRESSION_KEYS.contains(&key.as_str()) && !text.trim().is_empty() =>
255                    {
256                        // Calculated fields and groups keep their expression as text; data sources do not.
257                        if !here.starts_with("dataSources") {
258                            out.push((here, text.clone()));
259                        }
260                    }
261                    _ => expressions(v, &here, out),
262                }
263            }
264        }
265        Value::Array(items) => {
266            for (i, v) in items.iter().enumerate() {
267                expressions(v, &format!("{path}[{i}]"), out);
268            }
269        }
270        _ => {}
271    }
272}

274/// The check of a parsed layout: its expressions, then a render over its data.
275fn check_report(report: &Report, data: Option<&Value>, parameters: &HashMap<String, Value>) -> Value {
276    let embedded = report.embedded_data();
277    let data = data.or(embedded);
278    let layout: Value = serde_json::from_str(&report.to_json()).unwrap_or(Value::Null);
279    let mut found = Vec::new();
280    expressions(&layout, "", &mut found);
281    let master = schema_of(report, data, None);
282    let mut problems = Vec::new();
283    let mut warnings = Vec::new();
284    for (path, text) in &found {
285        // A detail report's expressions read its collection's rows.
286        let schema = match path.strip_prefix("detailReports[").and_then(|rest| rest.split(']').next()) {
287            Some(i) => {
288                let member = i
289                    .parse::<usize>()
290                    .ok()
291                    .and_then(|i| report.detail_reports.get(i))
292                    .map(|d| d.data_member.as_str());
293                schema_of(report, data, member)
294            }
295            None => master.clone(),
296        };
297        match Criteria::parse(text) {
298            Err(e) => problems
299                .push(json!({ "at": path, "expression": text, "error": e.message, "offset": e.offset })),
300            Ok(parsed) => {
301                let unknown: Vec<String> =
302                    validate(&parsed, &schema).iter().map(ToString::to_string).collect();
303                // The sample data may lack a field the real data has, so these only warn.
304                if !unknown.is_empty() && data.is_some() {
305                    warnings.push(json!({ "at": path, "expression": text, "unknown": unknown }));
306                }
307            }
308        }
309    }
310    let render = match data {
311        Some(data) => {
312            let options = RenderOptions { parameters: parameters.clone(), ..RenderOptions::default() };
313            let doc = ReportEngine::new().render_with(report, data, &options);
314            json!({ "pages": doc.pages.len(), "errors": doc.errors })
315        }
316        None => json!({ "skipped": "the data comes from outside the layout; pass data to render it" }),
317    };
318    json!({
319        "valid": problems.is_empty() && render["errors"].as_array().is_none_or(Vec::is_empty),
320        "expressions_checked": found.len(),
321        "problems": problems,
322        "warnings": warnings,
323        "render": render,
324    })
325}

327/// Checks a layout given as JSON text or as an object.
328pub fn check(layout: &Value, data: Option<&Value>, parameters: Option<&Value>) -> Result<Value, String> {
329    let text = match layout {
330        Value::String(s) => s.clone(),
331        other => other.to_string(),
332    };
333    let report = match Report::from_json(&text) {
334        Ok(report) => report,
335        Err(e) => {
336            return Ok(json!({
337                "valid": false,
338                "json_error": { "message": e.to_string(), "line": e.line(), "column": e.column() },
339            }));
340        }
341    };
342    let parameters: HashMap<String, Value> = match parameters {
343        Some(Value::Object(map)) => map.clone().into_iter().collect(),
344        Some(_) => return Err("parameters must be an object of name and value".into()),
345        None => HashMap::new(),
346    };
347    let mut answer = check_report(&report, data, &parameters);
348    answer["name"] = json!(report.name);
349    Ok(answer)
350}

352/// A ready layout: its name, what it shows and how it is made.
353type Template = (&'static str, &'static str, fn() -> Report);

355/// The ready layouts, or the layout of one.
356pub fn templates(name: Option<&str>) -> Result<Value, String> {
357    let list: [Template; 4] = [
358        ("invoice", "Issuer, customer, items as master-detail, totals and notes.", samples::invoice),
359        (
360            "sales_by_region",
361            "Nested groups with subtotals, zebra rows, conditional highlight and a minimum value parameter.",
362            || samples::sales_by_region(20),
363        ),
364        ("sales_dashboard", "Area, bar and doughnut charts over the same orders.", samples::sales_dashboard),
365        ("customer_list", "Grouped by initial, with a check box.", samples::customer_list),
366    ];
367    match name {
368        None => Ok(json!({
369            "templates": list.iter().map(|(n, d, _)| json!({ "name": n, "description": d })).collect::<Vec<_>>(),
370        })),
371        Some(wanted) => {
372            let (n, d, make) = list
373                .iter()
374                .find(|(n, _, _)| n.eq_ignore_ascii_case(wanted))
375                .ok_or_else(|| format!("unknown template '{wanted}': use invoice, sales_by_region, sales_dashboard or customer_list"))?;
376            let layout: Value = serde_json::from_str(&make().to_json()).map_err(|e| e.to_string())?;
377            Ok(
378                json!({ "name": n, "description": d, "file_name": format!("{n}.rvreport.json"), "layout": layout }),
379            )
380        }
381    }
382}
