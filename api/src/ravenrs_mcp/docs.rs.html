18/// One row of a Markdown table.
19#[derive(Debug)]
20pub struct Row {
21    pub section: String,
22    pub headers: Vec<String>,
23    pub cells: Vec<String>,
24}
25
26/// Every table row of a Markdown document, skipping code blocks.
27pub fn table_rows(markdown: &str) -> Vec<Row> {
28    let mut rows = Vec::new();
29    let mut section = String::new();
30    let mut headers: Option<Vec<String>> = None;
31    let mut in_code = false;
32    for line in markdown.lines() {
33        let line = line.trim();
34        if line.starts_with("```") {
35            in_code = !in_code;
36        }
37        if in_code {
38            continue;
39        }
40        if line.starts_with('#') {
41            section = line.trim_start_matches('#').trim().to_owned();
42        }
43        if !line.starts_with('|') {
44            headers = None;
45            continue;
46        }
47        let cells = split_row(line);
48        if cells.iter().all(|cell| !cell.is_empty() && cell.chars().all(|c| matches!(c, '-' | ':'))) {
49            continue;
50        }
51        match &headers {
52            None => headers = Some(cells),
53            Some(headers) => {
54                rows.push(Row { section: section.clone(), headers: headers.clone(), cells });
55            }
56        }
57    }
58    rows
59}

61/// Splits a table line on `|`, except inside code spans or after a backslash.
62fn split_row(line: &str) -> Vec<String> {
63    let inner = line.trim_matches('|');
64    let mut cells = vec![String::new()];
65    let mut in_code = false;
66    let mut chars = inner.chars().peekable();
67    while let Some(c) = chars.next() {
68        match c {
69            '\\' if chars.peek() == Some(&'|') => {
70                chars.next();
71                cells.last_mut().into_iter().for_each(|cell| cell.push('|'));
72            }
73            '`' => {
74                in_code = !in_code;
75                cells.last_mut().into_iter().for_each(|cell| cell.push('`'));
76            }
77            '|' if !in_code => cells.push(String::new()),
78            c => cells.last_mut().into_iter().for_each(|cell| cell.push(c)),
79        }
80    }
81    cells.iter().map(|cell| cell.trim().to_owned()).collect()
82}

84/// The group of a component in the README's components table.
85pub fn component_group(name: &str) -> String {
86    let rows = table_rows(README);
87    let row = rows.iter().find(|row| {
88        row.section == "Components"
89            && row.cells.get(1).is_some_and(|cell| backticked_names(cell).contains(&name))
90    });
91    row.and_then(|row| row.cells.first()).map_or_else(|| "Other".to_owned(), Clone::clone)
92}

94/// The identifiers a table cell lists in backticks: `Button` from `` `Button` (4 styles) ``.
95pub fn backticked_names(cell: &str) -> Vec<&str> {
96    cell.split('`')
97        .skip(1)
98        .step_by(2)
99        .filter_map(|code| code.split(|c: char| !c.is_alphanumeric() && c != '_').next())
100        .filter(|name| !name.is_empty())
101        .collect()
102}

104/// The component's row in the audit table of `docs/QUALITY.md`, if it has one.
105pub fn quality_row(name: &str) -> Option<Value> {
106    table_rows(QUALITY).into_iter().find_map(|row| {
107        let listed =
108            row.section == "Component audit" && row.cells.first().is_some_and(|cell| has_word(cell, name));
109        listed.then(|| {
110            let columns = row.headers.iter().zip(&row.cells).skip(1).filter(|(_, cell)| !cell.is_empty());
111            Value::Object(columns.map(|(header, cell)| (header.clone(), json!(cell))).collect())
112        })
113    })
114}

116/// Rows of the parity documents that match every given filter, with counts.
117pub fn parity(
118    document: Option<&str>,
119    query: Option<&str>,
120    status: Option<&str>,
121    limit: usize,
122) -> Result<Value, String> {
123    let documents: &[(&str, &str)] = match document {
124        None => &[("components", PARITY), ("reporting", REPORTING_PARITY)],
125        Some("components") => &[("components", PARITY)],
126        Some("reporting") => &[("reporting", REPORTING_PARITY)],
127        Some(other) => return Err(format!("unknown document '{other}': use 'components' or 'reporting'")),
128    };
129    if let Some(status) = status
130        && !["done", "partial", "planned"].contains(&status)
131    {
132        return Err(format!("unknown status '{status}': use 'done', 'partial' or 'planned'"));
133    }
134    let query = query.map(str::to_lowercase);
135
136    let mut matches = Vec::new();
137    let (mut done, mut partial, mut planned) = (0, 0, 0);
138    for (name, text) in documents {
139        for row in table_rows(text) {
140            let Some(detail) = row.cells.last() else { continue };
141            let Some(row_status) = row_status(detail) else { continue };
142            if status.is_some_and(|status| status != row_status) {
143                continue;
144            }
145            let searchable = format!("{} {}", row.section, row.cells.join(" ")).to_lowercase();
146            if query.as_ref().is_some_and(|query| !searchable.contains(query)) {
147                continue;
148            }
149            match row_status {
150                "done" => done += 1,
151                "partial" => partial += 1,
152                _ => planned += 1,
153            }
154            let middle = row.headers.iter().zip(&row.cells).skip(1).take(row.cells.len().saturating_sub(2));
155            let reference: Map<String, Value> = middle
156                .filter(|(_, cell)| !cell.is_empty())
157                .map(|(header, cell)| (header.clone(), json!(cell)))
158                .collect();
159            matches.push(json!({
160                "document": name,
161                "section": row.section,
162                "feature": row.cells.first(),
163                "status": row_status,
164                "ravenrs": detail,
165                "reference": reference,
166            }));
167        }
168    }
169    let total = matches.len();
170    matches.truncate(limit);
171    Ok(json!({
172        "total": total,
173        "done": done,
174        "partial": partial,
175        "planned": planned,
176        "shown": matches.len(),
177        "rows": matches,
178    }))
179}

181/// The status a parity cell opens with. A row like "✅ basics. Export ⬜" counts as done.
182fn row_status(cell: &str) -> Option<&'static str> {
183    cell.chars().find_map(|c| match c {
184        '✅' => Some("done"),
185        '🟡' => Some("partial"),
186        '⬜' => Some("planned"),
187        _ => None,
188    })
189}

191/// Design tokens, flattened, with the Rust name where one exists.
192pub fn tokens(category: Option<&str>, query: Option<&str>) -> Result<Value, String> {
193    let category = match category {
194        Some("type") => Some("typography"),
195        other => other,
196    };
197    if let Some(category) = category
198        && !TOKEN_CATEGORIES.contains(&category)
199    {
200        return Err(format!("unknown category '{category}': use one of {}", TOKEN_CATEGORIES.join(", ")));
201    }
202    let document: Value = serde_json::from_str(TOKENS).map_err(|error| error.to_string())?;
203    let query = query.map(str::to_lowercase);
204
205    let mut tokens = Vec::new();
206    for name in TOKEN_CATEGORIES {
207        if category.is_some_and(|category| category != name) {
208            continue;
209        }
210        for token in category_tokens(&document, name) {
211            let mut token = token.clone();
212            let token_name = token["name"].as_str().unwrap_or_default().to_owned();
213            if let Some(rust) = rust_name(name, &token_name) {
214                token["rust"] = json!(rust);
215            }
216            token["category"] = json!(name);
217            let matches = query.as_ref().is_none_or(|query| token.to_string().to_lowercase().contains(query));
218            if matches {
219                tokens.push(token);
220            }
221        }
222    }
223    Ok(json!({
224        "themes": document["color"]["themes"],
225        "total": tokens.len(),
226        "tokens": tokens,
227        "note": "In views, use the names in `rust` instead of literal values. Tokens without a `rust` name are read from the Theme, whose API is still evolving; check it with component_api (Theme).",
228    }))
229}

231fn category_tokens<'a>(document: &'a Value, category: &str) -> Vec<&'a Value> {
232    if category == "typography" {
233        let groups = document["type"]["groups"].as_array().map_or(&[][..], Vec::as_slice);
234        groups.iter().flat_map(|group| group["styles"].as_array().map_or(&[][..], Vec::as_slice)).collect()
235    } else {
236        document[category]["tokens"].as_array().map_or(Vec::new(), |tokens| tokens.iter().collect())
237    }
238}

240/// The Rust name of a token, from the docs of `Semantic` and the variants of `TextStyle`.
241fn rust_name(category: &str, token: &str) -> Option<String> {
242    let (enum_name, matches): (&str, &dyn Fn(&crate::api::Member) -> bool) = match category {
243        "color" => ("Semantic", &|variant| variant.docs.starts_with(&format!("`{token}`"))),
244        "typography" => ("TextStyle", &|variant| variant.name.eq_ignore_ascii_case(token)),
245        _ => return None,
246    };
247    let items = Index::get().find(enum_name);
248    let variant = items.first()?.variants.iter().find(|variant| matches(variant))?;
249    Some(format!("{enum_name}::{}", variant.name))
250}
