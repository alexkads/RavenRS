25/// One row of a Markdown table.
26#[derive(Debug)]
27pub struct Row {
28    pub section: String,
29    pub headers: Vec<String>,
30    pub cells: Vec<String>,
31}
32
33/// Every table row of a Markdown document, skipping code blocks.
34pub fn table_rows(markdown: &str) -> Vec<Row> {
35    let mut rows = Vec::new();
36    let mut section = String::new();
37    let mut headers: Option<Vec<String>> = None;
38    let mut in_code = false;
39    for line in markdown.lines() {
40        let line = line.trim();
41        if line.starts_with("```") {
42            in_code = !in_code;
43        }
44        if in_code {
45            continue;
46        }
47        if line.starts_with('#') {
48            section = line.trim_start_matches('#').trim().to_owned();
49        }
50        if !line.starts_with('|') {
51            headers = None;
52            continue;
53        }
54        let cells = split_row(line);
55        if cells.iter().all(|cell| !cell.is_empty() && cell.chars().all(|c| matches!(c, '-' | ':'))) {
56            continue;
57        }
58        match &headers {
59            None => headers = Some(cells),
60            Some(headers) => {
61                rows.push(Row { section: section.clone(), headers: headers.clone(), cells });
62            }
63        }
64    }
65    rows
66}

68/// Splits a table line on `|`, except inside code spans or after a backslash.
69fn split_row(line: &str) -> Vec<String> {
70    let inner = line.trim_matches('|');
71    let mut cells = vec![String::new()];
72    let mut in_code = false;
73    let mut chars = inner.chars().peekable();
74    while let Some(c) = chars.next() {
75        match c {
76            '\\' if chars.peek() == Some(&'|') => {
77                chars.next();
78                cells.last_mut().into_iter().for_each(|cell| cell.push('|'));
79            }
80            '`' => {
81                in_code = !in_code;
82                cells.last_mut().into_iter().for_each(|cell| cell.push('`'));
83            }
84            '|' if !in_code => cells.push(String::new()),
85            c => cells.last_mut().into_iter().for_each(|cell| cell.push(c)),
86        }
87    }
88    cells.iter().map(|cell| cell.trim().to_owned()).collect()
89}

91/// The group of a component in the README's components table. A part of a listed component,
92/// such as `SidebarItem` or `ListItem`, goes in the group of the longest listed name it starts with.
93pub fn component_group(name: &str) -> String {
94    let rows = table_rows(README);
95    let components = || rows.iter().filter(|row| row.section == "Components");
96    fn listed(row: &Row) -> Vec<&str> {
97        row.cells.get(1).map(|cell| backticked_names(cell)).unwrap_or_default()
98    }
99    let row = components().find(|row| listed(row).contains(&name)).or_else(|| {
100        components()
101            .filter_map(|row| {
102                let owner = listed(row)
103                    .into_iter()
104                    .filter(|owner| name.starts_with(owner))
105                    .max_by_key(|o| o.len())?;
106                Some((owner.len(), row))
107            })
108            .max_by_key(|(len, _)| *len)
109            .map(|(_, row)| row)
110    });
111    row.and_then(|row| row.cells.first()).map_or_else(|| "Other".to_owned(), Clone::clone)
112}

114/// The identifiers a table cell lists in backticks: `Button` from `` `Button` (4 styles) ``.
115pub fn backticked_names(cell: &str) -> Vec<&str> {
116    cell.split('`')
117        .skip(1)
118        .step_by(2)
119        .filter_map(|code| code.split(|c: char| !c.is_alphanumeric() && c != '_').next())
120        .filter(|name| !name.is_empty())
121        .collect()
122}

124/// The component's row in the audit table of `docs/QUALITY.md`, if it has one.
125pub fn quality_row(name: &str) -> Option<Value> {
126    table_rows(QUALITY).into_iter().find_map(|row| {
127        let listed =
128            row.section == "Component audit" && row.cells.first().is_some_and(|cell| has_word(cell, name));
129        listed.then(|| {
130            let columns = row.headers.iter().zip(&row.cells).skip(1).filter(|(_, cell)| !cell.is_empty());
131            Value::Object(columns.map(|(header, cell)| (header.clone(), json!(cell))).collect())
132        })
133    })
134}

136/// Rows of the parity documents that match every given filter, with counts.
137pub fn parity(
138    document: Option<&str>,
139    query: Option<&str>,
140    status: Option<&str>,
141    limit: usize,
142) -> Result<Value, String> {
143    let documents: &[(&str, &str)] = match document {
144        // The application tracker is asked for by name, as the tool says.
145        None => &[("components", PARITY), ("reporting", REPORTING_PARITY)],
146        Some("components") => &[("components", PARITY)],
147        Some("reporting") => &[("reporting", REPORTING_PARITY)],
148        Some("application") => &[("application", APP_PARITY)],
149        Some(other) => {
150            return Err(format!(
151                "unknown document '{other}': use 'components', 'reporting' or 'application'"
152            ));
153        }
154    };
155    if let Some(status) = status
156        && !["done", "partial", "planned"].contains(&status)
157    {
158        return Err(format!("unknown status '{status}': use 'done', 'partial' or 'planned'"));
159    }
160    let query = query.map(str::to_lowercase);
161
162    let mut matches = Vec::new();
163    let (mut done, mut partial, mut planned) = (0, 0, 0);
164    for (name, text) in documents {
165        for row in table_rows(text) {
166            // The status is in the last cell that has one: the last column of the component and
167            // reporting trackers, the middle one of the application tracker.
168            let Some((at, row_status)) = row
169                .cells
170                .iter()
171                .enumerate()
172                .skip(1)
173                .rev()
174                .find_map(|(at, cell)| row_status(cell).map(|s| (at, s)))
175            else {
176                continue;
177            };
178            let detail = &row.cells[at];
179            if status.is_some_and(|status| status != row_status) {
180                continue;
181            }
182            let searchable = format!("{} {}", row.section, row.cells.join(" ")).to_lowercase();
183            if query.as_ref().is_some_and(|query| !searchable.contains(query)) {
184                continue;
185            }
186            match row_status {
187                "done" => done += 1,
188                "partial" => partial += 1,
189                _ => planned += 1,
190            }
191            let others = row.headers.iter().zip(&row.cells).enumerate().skip(1).filter(|(i, _)| *i != at);
192            let reference: Map<String, Value> = others
193                .map(|(_, pair)| pair)
194                .filter(|(_, cell)| !cell.is_empty())
195                .map(|(header, cell)| (header.clone(), json!(cell)))
196                .collect();
197            matches.push(json!({
198                "document": name,
199                "section": row.section,
200                "feature": row.cells.first(),
201                "status": row_status,
202                "ravenrs": detail,
203                "reference": reference,
204            }));
205        }
206    }
207    let total = matches.len();
208    matches.truncate(limit);
209    Ok(json!({
210        "total": total,
211        "done": done,
212        "partial": partial,
213        "planned": planned,
214        "shown": matches.len(),
215        "rows": matches,
216    }))
217}

219/// One part of a guide: what is under a heading, up to the next heading.
220struct Section<'a> {
221    heading: &'a str,
222    text: &'a str,
223}
224
225/// The parts of a Markdown document, skipping code blocks when looking for headings.
226fn sections(markdown: &str) -> Vec<Section<'_>> {
227    let mut starts: Vec<(usize, &str)> = Vec::new();
228    let mut at = 0;
229    let mut in_code = false;
230    for line in markdown.split_inclusive('\n') {
231        let trimmed = line.trim_end();
232        if trimmed.trim_start().starts_with("```") {
233            in_code = !in_code;
234        } else if !in_code && trimmed.starts_with('#') {
235            starts.push((at, trimmed.trim_start_matches('#').trim()));
236        }
237        at += line.len();
238    }
239    let mut parts = Vec::with_capacity(starts.len() + 1);
240    if starts.first().is_none_or(|(first, _)| *first > 0) {
241        let end = starts.first().map_or(markdown.len(), |(first, _)| *first);
242        parts.push(Section { heading: "", text: &markdown[..end] });
243    }
244    for (position, (start, heading)) in starts.iter().enumerate() {
245        let end = starts.get(position + 1).map_or(markdown.len(), |(next, _)| *next);
246        parts.push(Section { heading, text: &markdown[*start..end] });
247    }
248    parts
249}

251fn matching_guides(document: Option<&str>) -> Result<Vec<(&'static str, &'static str)>, String> {
252    let Some(document) = document else { return Ok(GUIDES.to_vec()) };
253    let wanted = document.to_lowercase();
254    let found: Vec<_> =
255        GUIDES.iter().copied().filter(|(path, _)| path.to_lowercase().contains(&wanted)).collect();
256    if found.is_empty() {
257        return Err(format!("no guide matches '{document}'. Call guides with no arguments to list them."));
258    }
259    Ok(found)
260}

262/// A piece of `text` around the first of `words`, on whole characters.
263fn excerpt(text: &str, words: &[String]) -> String {
264    let lower = text.to_lowercase();
265    let first = words.iter().filter_map(|word| lower.find(word.as_str())).min().unwrap_or(0);
266    // Lowercasing can change byte lengths, so the place is only a hint, measured in characters.
267    let at = lower[..first].chars().count();
268    let start = at.saturating_sub(MAX_EXCERPT / 3);
269    let piece: String = text.chars().skip(start).take(MAX_EXCERPT).collect();
270    let piece = piece.trim();
271    if start > 0 { format!("...{piece}") } else { piece.to_owned() }
272}

274/// The guides: all of them, the parts of some, the text of a part, or the parts that have some words.
275pub fn guides(
276    document: Option<&str>,
277    section: Option<&str>,
278    query: Option<&str>,
279    limit: usize,
280) -> Result<Value, String> {
281    let chosen = matching_guides(document)?;
282    if let Some(query) = query {
283        let words: Vec<String> = query.to_lowercase().split_whitespace().map(str::to_owned).collect();
284        if words.is_empty() {
285            return Err("the query has no words".to_owned());
286        }
287        let mut found = Vec::new();
288        for (path, text) in &chosen {
289            for part in sections(text) {
290                let lower = part.text.to_lowercase();
291                if !words.iter().all(|word| lower.contains(word.as_str())) {
292                    continue;
293                }
294                let heading = part.heading.to_lowercase();
295                let in_heading = words.iter().filter(|word| heading.contains(word.as_str())).count();
296                let times: usize = words.iter().map(|word| lower.matches(word.as_str()).count()).sum();
297                found.push((in_heading, times, *path, part.heading, excerpt(part.text, &words)));
298            }
299        }
300        found.sort_by_key(|part| std::cmp::Reverse((part.0, part.1)));
301        let total = found.len();
302        found.truncate(limit);
303        let matches: Vec<Value> = found
304            .into_iter()
305            .map(|(_, _, path, heading, text)| json!({ "document": path, "section": heading, "excerpt": text }))
306            .collect();
307        return Ok(json!({ "query": query, "total": total, "shown": matches.len(), "matches": matches }));
308    }
309    if let Some(section) = section {
310        let wanted = section.to_lowercase();
311        let mut found = Vec::new();
312        for (path, text) in &chosen {
313            for part in sections(text) {
314                if part.heading.to_lowercase().contains(&wanted) {
315                    let whole = part.text.len() <= MAX_SECTION;
316                    let shown: String = part.text.chars().take(MAX_SECTION).collect();
317                    found.push(json!({
318                        "document": path,
319                        "section": part.heading,
320                        "text": shown,
321                        "whole": whole,
322                    }));
323                }
324            }
325        }
326        if found.is_empty() {
327            return Err(format!("no section matches '{section}'. Pass only `document` to list them."));
328        }
329        found.truncate(limit);
330        return Ok(json!({ "sections": found }));
331    }
332    if document.is_some() {
333        let listed: Vec<Value> = chosen
334            .iter()
335            .map(|(path, text)| {
336                let headings: Vec<&str> =
337                    sections(text).into_iter().map(|part| part.heading).filter(|h| !h.is_empty()).collect();
338                json!({ "document": path, "sections": headings })
339            })
340            .collect();
341        return Ok(json!({ "documents": listed }));
342    }
343    let listed: Vec<Value> = chosen
344        .iter()
345        .map(|(path, text)| {
346            let title = sections(text).into_iter().map(|part| part.heading).find(|h| !h.is_empty());
347            json!({ "document": path, "title": title.unwrap_or(path) })
348        })
349        .collect();
350    Ok(json!({ "total": listed.len(), "documents": listed }))
351}

353/// The scenarios C01–C19 and the groups M01–M19 of docs/APP-CONTRACTS.md, with the issues that own
354/// them, only those whose text has the query when there is one.
355pub fn app_contracts(query: Option<&str>) -> Value {
356    let query = query.map(str::to_lowercase);
357    let wanted = |cells: &[String]| {
358        query.as_ref().is_none_or(|query| cells.join(" ").to_lowercase().contains(query.as_str()))
359    };
360    let (mut scenarios, mut groups) = (Vec::new(), Vec::new());
361    for row in table_rows(APP_CONTRACTS) {
362        if !wanted(&row.cells) {
363            continue;
364        }
365        let entry: Map<String, Value> = row
366            .headers
367            .iter()
368            .zip(&row.cells)
369            .map(|(header, cell)| (header.to_lowercase(), json!(cell)))
370            .collect();
371        match row.headers.first().map(String::as_str) {
372            Some("Scenario") => scenarios.push(Value::Object(entry)),
373            Some("Group") => groups.push(Value::Object(entry)),
374            _ => {}
375        }
376    }
377    json!({
378        "document": "docs/APP-CONTRACTS.md",
379        "status": "docs/APP-PARITY.md, through parity_status with document 'application'",
380        "scenarios": scenarios,
381        "groups": groups,
382    })
383}

385/// The status a parity cell opens with. A row like "✅ basics. Export ⬜" counts as done.
386fn row_status(cell: &str) -> Option<&'static str> {
387    cell.chars().find_map(|c| match c {
388        '✅' => Some("done"),
389        '🟡' => Some("partial"),
390        '⬜' => Some("planned"),
391        _ => None,
392    })
393}

395/// Design tokens, flattened, with the Rust name where one exists.
396pub fn tokens(category: Option<&str>, query: Option<&str>) -> Result<Value, String> {
397    let category = match category {
398        Some("type") => Some("typography"),
399        other => other,
400    };
401    if let Some(category) = category
402        && !TOKEN_CATEGORIES.contains(&category)
403    {
404        return Err(format!("unknown category '{category}': use one of {}", TOKEN_CATEGORIES.join(", ")));
405    }
406    let document: Value = serde_json::from_str(TOKENS).map_err(|error| error.to_string())?;
407    let query = query.map(str::to_lowercase);
408
409    let mut tokens = Vec::new();
410    for name in TOKEN_CATEGORIES {
411        if category.is_some_and(|category| category != name) {
412            continue;
413        }
414        for token in category_tokens(&document, name) {
415            let mut token = token.clone();
416            let token_name = token["name"].as_str().unwrap_or_default().to_owned();
417            if let Some(rust) = rust_name(name, &token_name) {
418                token["rust"] = json!(rust);
419            }
420            token["category"] = json!(name);
421            let matches = query.as_ref().is_none_or(|query| token.to_string().to_lowercase().contains(query));
422            if matches {
423                tokens.push(token);
424            }
425        }
426    }
427    Ok(json!({
428        "themes": document["color"]["themes"],
429        "total": tokens.len(),
430        "tokens": tokens,
431        "note": "In views, use the names in `rust` instead of literal values. Tokens without a `rust` name are read from the Theme, whose API is still evolving; check it with component_api (Theme).",
432    }))
433}

435fn category_tokens<'a>(document: &'a Value, category: &str) -> Vec<&'a Value> {
436    if category == "typography" {
437        let groups = document["type"]["groups"].as_array().map_or(&[][..], Vec::as_slice);
438        groups.iter().flat_map(|group| group["styles"].as_array().map_or(&[][..], Vec::as_slice)).collect()
439    } else {
440        document[category]["tokens"].as_array().map_or(Vec::new(), |tokens| tokens.iter().collect())
441    }
442}

444/// The Rust name of a token, from the docs of `Semantic` and the variants of `TextStyle`.
445fn rust_name(category: &str, token: &str) -> Option<String> {
446    let (enum_name, matches): (&str, &dyn Fn(&crate::api::Member) -> bool) = match category {
447        "color" => ("Semantic", &|variant| variant.docs.starts_with(&format!("`{token}`"))),
448        "typography" => ("TextStyle", &|variant| variant.name.eq_ignore_ascii_case(token)),
449        _ => return None,
450    };
451    // The theme's own enum: an engine may have a type of the same name (`ravenrs-deck` has a
452    // `TextStyle` struct for its PDF text), which says nothing about design tokens.
453    let items = Index::get().find(enum_name);
454    let theme = items.iter().find(|item| item.krate == "ravenrs-style")?;
455    let variant = theme.variants.iter().find(|variant| matches(variant))?;
456    Some(format!("{enum_name}::{}", variant.name))
457}
