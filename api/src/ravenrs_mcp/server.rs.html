94impl RavenServer {
95    #[tool(
96        description = "List RavenRS components with their group and one-line summary.",
97        annotations(read_only_hint = true, open_world_hint = false)
98    )]
99    fn list_components(&self, Parameters(input): Parameters<ListComponents>) -> CallToolResult {
100        let query = input.query.map(|query| query.to_lowercase());
101        let mut groups = BTreeSet::new();
102        let components: Vec<Value> = Index::get()
103            .components()
104            .into_iter()
105            .map(|item| (docs::component_group(&item.name), item))
106            .inspect(|(group, _)| {
107                groups.insert(group.clone());
108            })
109            .filter(|(group, _)| input.group.as_ref().is_none_or(|wanted| group.eq_ignore_ascii_case(wanted)))
110            .filter(|(_, item)| {
111                query.as_ref().is_none_or(|query| {
112                    item.name.to_lowercase().contains(query)
113                        || summary(&item.docs).to_lowercase().contains(query)
114                })
115            })
116            .map(|(group, item)| {
117                json!({
118                    "name": item.name,
119                    "group": group,
120                    "summary": summary(&item.docs),
121                    "source": location(&item.file, item.line),
122                })
123            })
124            .collect();
125        CallToolResult::structured(json!({
126            "ravenrs_version": env!("CARGO_PKG_VERSION"),
127            "total": components.len(),
128            "groups": groups,
129            "components": components,
130        }))
131    }
132
133    #[tool(
134        description = "Show the public API of a component or any other public RavenRS item: docs, constructors and builder methods with their signatures, fields, enum variants, implemented traits, source location and, for components, the quality audit.",
135        annotations(read_only_hint = true, open_world_hint = false)
136    )]
137    fn component_api(&self, Parameters(input): Parameters<ComponentApi>) -> CallToolResult {
138        let items = Index::get().find(input.name.trim());
139        if items.is_empty() {
140            return not_found(&input.name);
141        }
142        let filter = input.filter.map(|filter| filter.to_lowercase());
143        let matches: Vec<Value> = items.into_iter().map(|item| describe(item, filter.as_deref())).collect();
144        CallToolResult::structured(json!({
145            "name": input.name,
146            "matches": matches,
147            "note": "Every view also takes the ViewExt modifiers, such as padding, background and frame. See component_api with name ViewExt.",
148        }))
149    }
150
151    #[tool(
152        description = "Show real code that uses a component or type, taken from the RavenRS examples, its rustdoc and the getting started guide. Examples are picked to cover as many builder methods as possible.",
153        annotations(read_only_hint = true, open_world_hint = false)
154    )]
155    fn component_examples(&self, Parameters(input): Parameters<ComponentExamples>) -> CallToolResult {
156        let index = Index::get();
157        let items = index.find(input.name.trim());
158        let Some(name) = items.first().map(|item| item.name.as_str()) else {
159            return not_found(&input.name);
160        };
161        let limit = input.limit.unwrap_or(3).clamp(1, 10);
162
163        let usages: Vec<&Usage> = index.usages.iter().filter(|usage| usage.root == name).collect();
164        let found = usages.len();
165        let mut examples: Vec<Value> = pick_varied(usages, limit)
166            .into_iter()
167            .map(|usage| {
168                json!({ "kind": "example", "source": location(&usage.file, usage.line), "code": usage.code })
169            })
170            .collect();
171        for item in &items {
172            for code in code_blocks_with(&item.docs, name) {
173                examples.push(
174                    json!({ "kind": "rustdoc", "source": location(&item.file, item.line), "code": code }),
175                );
176            }
177        }
178        for code in code_blocks_with(docs::GETTING_STARTED, name) {
179            examples.push(json!({ "kind": "guide", "source": "docs/GETTING_STARTED.md", "code": code }));
180        }
181        examples.truncate(limit);
182        CallToolResult::structured(json!({
183            "name": name,
184            "usages_in_examples": found,
185            "examples": examples,
186        }))
187    }
188
189    #[tool(
190        description = "Look up design tokens from docs/design-system/tokens.json: colors per theme, type scale, spacing, radii, shadows, durations and springs, with what each is for and its Rust name, such as Semantic::Ink or TextStyle::Title.",

197    #[tool(
198        description = "Check whether a feature exists: searches the parity trackers against shadcn/ui, DevExpress and Telerik (components) and DevExpress Reporting (reporting), and returns each row's status (done, partial or planned) with counts.",
199        annotations(read_only_hint = true, open_world_hint = false)

202        answer(docs::parity(
203            input.document.as_deref(),
204            input.query.as_deref(),
205            input.status.as_deref(),
206            input.limit.unwrap_or(50).clamp(1, 200),
207        ))
208    }
209
210    #[tool(
211        description = "Check an expression in the ravenrs-criteria language without running it: syntax errors with their offset, the normalized text, the fields and parameters it uses and, given a schema, unknown names.",
212        annotations(read_only_hint = true, open_world_hint = false)
213    )]
214    fn validate_expression(&self, Parameters(input): Parameters<ValidateExpression>) -> CallToolResult {
215        answer(expression::check(&input.expression, input.schema.as_ref()))
216    }

218    #[tool(
219        description = "Search the functions and aggregates of the ravenrs-criteria expression language, with signatures and descriptions.",
220        annotations(read_only_hint = true, open_world_hint = false)
221    )]
222    fn expression_functions(&self, Parameters(input): Parameters<ExpressionFunctions>) -> CallToolResult {
223        CallToolResult::structured(expression::functions(input.category.as_deref(), input.query.as_deref()))
224    }

236fn answer(result: Result<Value, String>) -> CallToolResult {
237    match result {
238        Ok(value) => CallToolResult::structured(value),
239        Err(message) => CallToolResult::structured_error(json!({ "error": message })),
240    }
241}

243fn not_found(name: &str) -> CallToolResult {
244    CallToolResult::structured_error(json!({
245        "error": format!("no public item named '{name}'"),
246        "did_you_mean": Index::get().suggest(name.trim()),
247    }))
248}

250fn location(file: &str, line: usize) -> String {
251    format!("{file}:{line}")
252}

254/// The first paragraph of some docs, on one line.
255fn summary(docs: &str) -> String {
256    let paragraph = docs.split("\n\n").next().unwrap_or_default();
257    paragraph.split_whitespace().collect::<Vec<_>>().join(" ")
258}

260fn describe(item: &Item, filter: Option<&str>) -> Value {
261    let mut value = json!({
262        "crate": item.krate,
263        "kind": item.kind,
264        "signature": item.signature,
265        "source": location(&item.file, item.line),
266        "docs": item.docs,
267    });
268    if item.non_exhaustive {
269        value["non_exhaustive"] = json!(true);
270    }
271    if !item.traits.is_empty() {
272        value["traits"] = json!(item.traits);
273    }
274    if !item.fields.is_empty() {
275        value["fields"] = members(&item.fields, &item.file, filter);
276    }
277    if !item.variants.is_empty() {
278        value["variants"] = members(&item.variants, &item.file, filter);
279    }
280    if !item.methods.is_empty() {
281        value["methods"] = members(&item.methods, &item.file, filter);
282    }
283    if item.krate == "ravenrs-widgets"
284        && let Some(quality) = docs::quality_row(&item.name)
285    {
286        value["quality"] = quality;
287    }
288    value
289}

291fn members(members: &[Member], file: &str, filter: Option<&str>) -> Value {
292    let matching: Vec<&Member> = members
293        .iter()
294        .filter(|member| filter.is_none_or(|filter| member.name.to_lowercase().contains(filter)))
295        .collect();
296    let items: Vec<Value> = matching
297        .iter()
298        .take(MAX_MEMBERS)
299        .map(|member| {
300            json!({
301                "signature": member.signature,
302                "docs": member.docs,
303                "source": location(file, member.line),
304            })
305        })
306        .collect();
307    let mut value = json!({ "total": matching.len(), "items": items });
308    if matching.len() > MAX_MEMBERS {
309        value["hint"] = json!("Too many to list. Pass `filter` to narrow them down.");
310    }
311    value
312}

314/// Picks usages that together show the most methods, preferring short ones.
315fn pick_varied(mut usages: Vec<&Usage>, limit: usize) -> Vec<&Usage> {
316    usages.sort_by_key(|usage| usage.code.len());
317    let mut seen = BTreeSet::new();
318    let mut picked = Vec::new();
319    while picked.len() < limit && !usages.is_empty() {
320        let new_methods =
321            |usage: &&Usage| usage.methods.iter().filter(|method| !seen.contains(*method)).count();
322        // On a tie, `max_by_key` would take the last and longest, so walk from the short end.
323        let best = usages
324            .iter()
325            .enumerate()
326            .rev()
327            .max_by_key(|(_, usage)| new_methods(usage))
328            .map_or(0, |(position, _)| position);
329        let usage = usages.remove(best);
330        seen.extend(usage.methods.iter().cloned());
331        picked.push(usage);
332    }
333    picked
334}
