1//! A form made from a schema of parameters, for what a node, a connector or a credential asks:
2//! text, numbers, yes or no, a choice, options the app loads (that may depend on another field),
3//! named groups and lists of rows that grow and shrink. The values are JSON; the form edits a
4//! copy and gives it back only on Aplicar, and Descartar forgets the copy whole.
5//!
6//! Made of the editors there are (`FormField`, `TextField`, `TextArea`, `NumberField`, `Select`,
7//! `Toggle`, `SegmentedControl`, `Alert`), so it looks and reads like any other form.
8//!
9//! A field that takes expressions can hold one instead of a value: it is kept as an object
10//! with the key [`EXPRESSION_KEY`], `{"$expr": "[valor] > 100"}`, and the field shows the mode
11//! it is in.
12

31/// What a parameter holds, which picks its editor.
32#[derive(Clone, Debug, PartialEq)]
33#[non_exhaustive]
34#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
35pub enum ParamKind {
36    /// A line of text.
37    Text,
38    /// Text of several lines.
39    LongText,
40    /// A number.
41    Number,
42    /// Yes or no.
43    Boolean,
44    /// One of these options, each its value and the words shown.
45    Choice(Vec<(String, String)>),
46    /// One of the options the app loads with [`ParameterForm::on_load`] and gives with
47    /// [`ParameterForm::options`].
48    Loaded,
49    /// A named group of fields, kept as an object.
50    Group(Vec<ParamField>),
51    /// Rows of these fields that the person adds and removes, kept as a list of objects.
52    List(Vec<ParamField>),
53}

55/// A parameter of a schema.
56#[derive(Clone, Debug, PartialEq)]
57#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
58pub struct ParamField {
59    /// Its key in the values.
60    pub name: String,
61    /// What the form calls it.
62    pub label: String,
63    /// What it holds.
64    pub kind: ParamKind,
65    /// Its value when the values have none.
66    pub default: Option<Value>,
67    /// Aplicar asks for a value.

174/// Takes the expression of a field of data carried to a field of the form.
175type Take = Rc<dyn Fn(&str)>;
176
177/// A field of the form as a place a field of data can be carried to: text takes it as an
178/// expression, written where the value ends; anything else refuses it, saying why.
179struct Takes {
180    inner: AnyView,
181    take: Option<Take>,
182    refused: String,
183}
184
185impl View for Takes {
186    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
187        let node = cx.child(&*self.inner);
188        let refused = self.refused.clone();
189        match self.take.clone() {
190            Some(take) => cx.drop_target(
191                node,
192                move |p| if FieldDrag::carried(p).is_some() { Ok(()) } else { Err(refused.clone()) },
193                move |p, _| {
194                    if let Some(field) = FieldDrag::carried(p) {
195                        take(&field.expression);
196                    }
197                },
198            ),
199            None => cx.drop_target(node, move |_| Err(refused.clone()), |_, _| {}),
200        }
201        node
202    }
203}

205/// One step into the values: a key of an object or a row of a list.
206#[derive(Clone, Debug, PartialEq)]
207enum Step {
208    Key(String),
209    Row(usize),
210}
211
212fn read(root: &Value, path: &[Step]) -> Value {
213    let mut at = root;
214    for step in path {
215        let next = match step {
216            Step::Key(k) => at.get(k.as_str()),
217            Step::Row(i) => at.get(*i),
218        };
219        match next {
220            Some(v) => at = v,
221            None => return Value::Null,
222        }
223    }
224    at.clone()
225}
226
227fn write(root: &mut Value, path: &[Step], value: Value) {
228    let Some((last, before)) = path.split_last() else {
229        *root = value;
230        return;
231    };
232    let mut at = root;
233    for step in before {
234        at = match step {
235            Step::Key(k) => {
236                if !at.is_object() {
237                    *at = Value::Object(Map::new());
238                }
239                at.as_object_mut()
240                    .map(|m| m.entry(k.clone()).or_insert(Value::Null))
241                    .unwrap_or_else(|| unreachable!())
242            }
243            Step::Row(i) => {
244                if !at.is_array() {
245                    *at = Value::Array(Vec::new());
246                }
247                let Some(list) = at.as_array_mut() else { return };
248                while list.len() <= *i {
249                    list.push(Value::Object(Map::new()));
250                }
251                &mut list[*i]
252            }
253        };
254    }
255    match last {
256        Step::Key(k) => {
257            if !at.is_object() {
258                *at = Value::Object(Map::new());
259            }
260            if let Some(m) = at.as_object_mut() {
261                m.insert(k.clone(), value);
262            }
263        }
264        Step::Row(i) => {
265            if let Some(list) = at.as_array_mut()
266                && *i < list.len()
267            {
268                list[*i] = value;
269            }
270        }
271    }
272}

274/// `values` with the default of each field it lacks, in groups too.
275fn with_defaults(fields: &[ParamField], mut values: Map<String, Value>) -> Map<String, Value> {
276    for field in fields {
277        let missing = values.get(&field.name).is_none_or(Value::is_null);
278        match &field.kind {
279            ParamKind::Group(inner) => {
280                let own = values.get(&field.name).and_then(Value::as_object).cloned().unwrap_or_default();
281                values.insert(field.name.clone(), Value::Object(with_defaults(inner, own)));
282            }
283            _ if missing => {
284                if let Some(default) = &field.default {
285                    values.insert(field.name.clone(), default.clone());
286                }
287            }
288            _ => {}
289        }
290    }
291    values
292}

294/// Whether a value says nothing: null, empty text or an empty list.
295fn empty(value: &Value) -> bool {
296    match value {
297        Value::Null => true,
298        Value::String(s) => s.trim().is_empty(),
299        Value::Array(a) => a.is_empty(),
300        _ => false,
301    }
302}
303
304/// Whether `field` shows, in the object `here` it belongs to.
305fn shown(field: &ParamField, here: &Value) -> bool {
306    match &field.shown_when {
307        Some((other, value)) => here.get(other.as_str()) == Some(value),
308        None => true,
309    }
310}
311
312/// What is missing among the required fields that show, in `here` and its groups and rows.
313fn missing(fields: &[ParamField], here: &Value, words: &str, out: &mut Vec<String>) {
314    for field in fields.iter().filter(|f| shown(f, here)) {
315        let value = here.get(field.name.as_str()).cloned().unwrap_or(Value::Null);
316        match &field.kind {
317            ParamKind::Group(inner) => missing(inner, &value, words, out),
318            ParamKind::List(inner) => {
319                for row in value.as_array().into_iter().flatten() {
320                    missing(inner, row, words, out);
321                }
322            }
323            _ if field.required && empty(&value) => out.push(words.replace("{0}", &field.label)),
324            _ => {}
325        }
326    }
327}

335/// A form made from a schema of parameters, editing a copy of `values` that Aplicar gives back
336/// and Descartar forgets.

338pub struct ParameterForm {

340    values: Binding<ParamValues>,

378        self
379    }

381    /// Hears Descartar, after the copy went back to the values.

384        self
385    }

388/// What the parts of the form share while it builds.
389struct Parts {
390    draft: State<Value>,
391    options: BTreeMap<String, ParamOptions>,
392    on_load: Option<Load>,
393    labels: Rc<crate::Labels>,
394}
395
396#[cfg_attr(feature = "live", ravenrs_live::expose)]
397impl Parts {
398    fn binding(&self, path: Vec<Step>) -> Binding<Value> {
399        let (a, b, p) = (self.draft.clone(), self.draft.clone(), path.clone());
400        Binding::new(move || read(&a.get(), &path), move |v| b.update(|root| write(root, &p, v)))
401    }
402
403    fn draft_map(&self) -> Map<String, Value> {
404        self.draft.get().as_object().cloned().unwrap_or_default()
405    }
406
407    /// The editor of `field` at `path`, inside the object `here`.
408    fn field(&self, field: &ParamField, path: Vec<Step>) -> AnyView {
409        let value = self.binding(path.clone());
410        let now = value.get();
411        let is_expression = now.get(EXPRESSION_KEY).is_some();
412        let editor: AnyView = if is_expression {
413            let text = value.clone().map(
414                |v| v.get(EXPRESSION_KEY).and_then(Value::as_str).unwrap_or_default().to_owned(),
415                |v, text: String| *v = serde_json::json!({ EXPRESSION_KEY: text }),
416            );
417            Box::new(TextField::new(text).placeholder(self.labels.param_expression.clone()))
418        } else {
419            self.editor(field, value.clone(), &path)
420        };
421        // A field of data carried here: an expression takes its expression at the end, text
422        // takes it as `{{ … }}`; other kinds refuse it.
423        let take: Option<Take> = match (&field.kind, is_expression) {
424            (ParamKind::Group(_) | ParamKind::List(_), _) => None,
425            (_, true) => {
426                let value = value.clone();
427                Some(Rc::new(move |expression: &str| {
428                    value.update(|v| {
429                        let before =
430                            v.get(EXPRESSION_KEY).and_then(Value::as_str).unwrap_or_default().to_owned();
431                        let joined = if before.trim().is_empty() {
432                            expression.to_owned()
433                        } else {
434                            format!("{before} {expression}")
435                        };
436                        *v = serde_json::json!({ EXPRESSION_KEY: joined });
437                    });
438                }))
439            }
440            (ParamKind::Text | ParamKind::LongText, false) => {
441                let value = value.clone();
442                Some(Rc::new(move |expression: &str| {
443                    value.update(|v| {
444                        let before = v.as_str().unwrap_or_default().to_owned();
445                        *v = Value::String(format!("{before}{{{{ {expression} }}}}"));
446                    });
447                }))
448            }
449            _ => None,
450        };
451        let editor: AnyView = match &field.kind {
452            ParamKind::Group(_) | ParamKind::List(_) => editor,
453            _ => Box::new(Takes { inner: editor, take, refused: self.labels.param_no_field.clone() }),
454        };
455        let wrapped: AnyView = match &field.kind {
456            ParamKind::Group(_) | ParamKind::List(_) => return editor,
457            // A switch carries its own label beside it, so a label above would say it twice.
458            ParamKind::Boolean if !is_expression => match &field.hint {
459                Some(hint) => Box::new(
460                    VStack::new((
461                        editor,
462                        Text::new(hint.clone()).style(TextStyle::Caption).foreground(Semantic::InkMuted),
463                    ))
464                    .spacing(4.0)
465                    .align(Align::Start),
466                ),
467                None => editor,
468            },
469            _ => {
470                let mut wrapped = FormField::new(field.label.clone(), editor);
471                if field.required {
472                    wrapped = wrapped.required();
473                }
474                if let Some(hint) = &field.hint {
475                    wrapped = wrapped.hint(hint.clone());
476                }
477                Box::new(wrapped)
478            }
479        };
480        if !field.expressions {
481            return wrapped;
482        }
483        // The switch between value and expression sits above the field, outside it, so the
484        // field's label still names the field.
485        let mode = value.map(
486            |v| usize::from(v.get(EXPRESSION_KEY).is_some()),
487            |v, mode: usize| match (mode, v.get(EXPRESSION_KEY).is_some()) {
488                // The value written becomes the expression's first words.
489                (1, false) => {
490                    let text = match &*v {
491                        Value::Null => String::new(),
492                        Value::String(s) => s.clone(),
493                        other => other.to_string(),
494                    };
495                    *v = serde_json::json!({ EXPRESSION_KEY: text });
496                }
497                (0, true) => *v = Value::Null,
498                _ => {}
499            },
500        );
501        Box::new(
502            VStack::new((
503                HStack::new((
504                    Spacer::new(),
505                    SegmentedControl::new(
506                        mode,
507                        [self.labels.param_fixed.clone(), self.labels.param_expression.clone()],
508                    )
509                    .label(format!("{}: {}", field.label, self.labels.param_mode)),
510                )),
511                wrapped,
512            ))
513            .spacing(4.0)
514            .align(Align::Stretch),
515        )
516    }
517
518    /// The editor of a value of `field`'s kind.
519    fn editor(&self, field: &ParamField, value: Binding<Value>, path: &[Step]) -> AnyView {
520        match &field.kind {
521            ParamKind::Text => Box::new(TextField::new(text_of(value))),
522            ParamKind::LongText => Box::new(TextArea::new(text_of(value))),
523            ParamKind::Number => Box::new(NumberField::new(value.map(
524                |v| v.as_f64(),
525                |v, n: Option<f64>| {
526                    *v = n.and_then(serde_json::Number::from_f64).map_or(Value::Null, Value::Number)
527                },
528            ))),
529            ParamKind::Boolean => Box::new(
530                Toggle::new(value.map(|v| v.as_bool().unwrap_or(false), |v, b: bool| *v = Value::Bool(b)))
531                    .label(field.label.clone()),
532            ),
533            ParamKind::Choice(options) => self.choice(field, value, options),
534            ParamKind::Loaded => match self.options.get(&field.name) {
535                Some(ParamOptions::Ready(options)) => self.choice(field, value, options),
536                Some(ParamOptions::Failed(why)) => {
537                    let (load, name, draft) = (self.on_load.clone(), field.name.clone(), self.draft.clone());
538                    Box::new(
539                        Alert::new(self.labels.param_load_failed.clone(), why.clone())
540                            .tone(Tone::Danger)
541                            .action(self.labels.param_retry.clone(), move || {
542                                if let Some(load) = &load {
543                                    load(
544                                        &name,
545                                        &ParamValues(draft.get().as_object().cloned().unwrap_or_default()),
546                                    );
547                                }
548                            }),
549                    )
550                }
551                _ => Box::new(
552                    HStack::new((
553                        Spinner::new(),
554                        Text::new(self.labels.param_loading.clone())
555                            .style(TextStyle::Callout)
556                            .foreground(Semantic::InkMuted),
557                    ))
558                    .spacing(8.0)
559                    .align(Align::Center),
560                ),
561            },
562            ParamKind::Group(inner) => {
563                let rows: Vec<AnyView> = inner
564                    .iter()
565                    .filter(|f| shown(f, &read(&self.draft.get(), path)))
566                    .map(|f| self.field(f, [path, &[Step::Key(f.name.clone())]].concat()))
567                    .collect();
568                Box::new(
569                    VStack::new((
570                        Text::new(field.label.clone()).style(TextStyle::Headline),
571                        VStack::new(rows).spacing(10.0).align(Align::Stretch),
572                    ))
573                    .spacing(8.0)
574                    .align(Align::Stretch),
575                )
576            }
577            ParamKind::List(inner) => self.list(field, inner, value, path),
578        }
579    }

581    fn choice(&self, field: &ParamField, value: Binding<Value>, options: &[(String, String)]) -> AnyView {
582        let values: Vec<String> = options.iter().map(|(v, _)| v.clone()).collect();
583        let words: Vec<String> = std::iter::once(self.labels.param_none.clone())
584            .chain(options.iter().map(|(_, w)| w.clone()))
585            .collect();
586        let (find, keep) = (values.clone(), values);
587        let selected = value.map(
588            move |v| v.as_str().and_then(|s| find.iter().position(|x| x == s)).map_or(0, |i| i + 1),
589            move |v, i: usize| {
590                *v = if i == 0 {
591                    Value::Null
592                } else {
593                    keep.get(i - 1).cloned().map_or(Value::Null, Value::String)
594                }
595            },
596        );
597        Box::new(Select::new(selected, words).label(field.label.clone()))
598    }

600    fn list(
601        &self,
602        field: &ParamField,
603        inner: &[ParamField],
604        value: Binding<Value>,
605        path: &[Step],
606    ) -> AnyView {
607        let rows = value.get().as_array().map_or(0, Vec::len);
608        let mut views: Vec<AnyView> = Vec::new();
609        for i in 0..rows {
610            let row: Vec<Step> = [path, &[Step::Row(i)]].concat();
611            let cells: Vec<AnyView> = inner
612                .iter()
613                .map(|f| -> AnyView {
614                    Box::new(
615                        VStack::new((self.field(f, [&row[..], &[Step::Key(f.name.clone())]].concat()),))
616                            .grow(1.0),
617                    )
618                })
619                .collect();
620            let gone = value.clone();
621            views.push(Box::new(
622                HStack::new((
623                    HStack::new(cells).spacing(8.0).align(Align::End).grow(1.0),
624                    IconButton::new(IconName::Trash2)
625                        .label(self.labels.param_remove.replace("{0}", &(i + 1).to_string()))
626                        .style(ButtonStyle::Ghost)
627                        .size(ButtonSize::Small)
628                        .on_click(move || {
629                            gone.update(|v| {
630                                if let Some(list) = v.as_array_mut()
631                                    && i < list.len()
632                                {
633                                    list.remove(i);
634                                }
635                            });
636                        }),
637                ))
638                .spacing(8.0)
639                .align(Align::End),
640            ));
641        }
642        let (more, fresh) = (value, Value::Object(with_defaults(inner, Map::new())));
643        Box::new(
644            VStack::new((
645                Text::new(field.label.clone()).style(TextStyle::Headline),
646                VStack::new(views).spacing(8.0).align(Align::Stretch),
647                Button::new(self.labels.param_add.clone())
648                    .icon(IconName::Plus)
649                    .style(ButtonStyle::Secondary)
650                    .size(ButtonSize::Small)
651                    .on_click(move || {
652                        let row = fresh.clone();
653                        more.update(|v| match v.as_array_mut() {
654                            Some(list) => list.push(row),
655                            None => *v = Value::Array(vec![row]),
656                        });
657                    }),
658            ))
659            .spacing(8.0)
660            .align(Align::Start),
661        )
662    }
663}

665/// A value as text: text as it is, a number written, nothing as empty.
666fn text_of(value: Binding<Value>) -> Binding<String> {
667    value.map(
668        |v| match v {
669            Value::String(s) => s.clone(),
670            Value::Null => String::new(),
671            other => other.to_string(),
672        },
673        |v, s: String| *v = Value::String(s),
674    )
675}
676
677/// The fields, in groups too, that load their options, with the path of the object they sit in.
678fn loaded<'a>(fields: &'a [ParamField], out: &mut Vec<&'a ParamField>) {
679    for field in fields {
680        match &field.kind {
681            ParamKind::Loaded => out.push(field),
682            ParamKind::Group(inner) => loaded(inner, out),
683            _ => {}
684        }
685    }
686}

688impl View for ParameterForm {
689    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
690        let labels = cx.labels();
691        let committed = with_defaults(&self.fields, self.values.get().0);
692        let draft = cx.use_state(|| Value::Object(committed.clone()));
693        let seen = cx.use_state(|| committed.clone());
694        let errors = cx.use_state(Vec::<String>::new);
695        let asked = cx.use_state(BTreeMap::<String, Value>::new);
696        // Values changed from outside, as another node chosen: the copy starts again from them.
697        if seen.get() != committed {
698            seen.set(committed.clone());
699            draft.set(Value::Object(committed.clone()));
700            errors.set(Vec::new());
701        }
702        let now = draft.get();
703        let mut with_options = Vec::new();
704        loaded(&self.fields, &mut with_options);
705        for field in with_options {
706            // What its options depend on now: asked again when it changes.
707            let key = Value::Array(
708                field
709                    .depends_on
710                    .iter()
711                    .map(|d| now.get(d.as_str()).cloned().unwrap_or(Value::Null))
712                    .collect(),
713            );
714            if asked.get().get(&field.name) != Some(&key) {
715                asked.update(|a| {
716                    a.insert(field.name.clone(), key.clone());
717                });
718                if let Some(load) = &self.on_load {
719                    load(&field.name, &ParamValues(now.as_object().cloned().unwrap_or_default()));
720                }
721            }
722            // A choice the options no longer have is cleared.
723            if let Some(ParamOptions::Ready(options)) = self.options.get(&field.name)
724                && let Some(chosen) = now.get(field.name.as_str()).and_then(Value::as_str)
725                && !options.iter().any(|(v, _)| v == chosen)
726            {
727                draft.update(|d| write(d, &[Step::Key(field.name.clone())], Value::Null));
728            }
729        }
730
731        let parts = Parts {
732            draft: draft.clone(),
733            options: self.options.clone(),
734            on_load: self.on_load.clone(),
735            labels: labels.clone(),
736        };
737        let now = draft.get();
738        let rows: Vec<AnyView> = self
739            .fields
740            .iter()
741            .filter(|f| shown(f, &now))
742            .map(|f| parts.field(f, vec![Step::Key(f.name.clone())]))
743            .collect();
744        let wrong: Vec<AnyView> = errors
745            .get()
746            .into_iter()
747            .map(|e| -> AnyView {
748                Box::new(Text::new(e).style(TextStyle::Callout).foreground(Semantic::Danger))
749            })
750            .collect();
751        let changed = parts.draft_map() != committed;
752
753        let (fields, values, apply_draft, apply_errors, on_apply) =
754            (self.fields.clone(), self.values.clone(), draft.clone(), errors.clone(), self.on_apply.clone());
755        let required = labels.param_required.clone();
756        let apply = move || {
757            let mut wrong = Vec::new();
758            let now = apply_draft.get();
759            missing(&fields, &now, &required, &mut wrong);
760            if !wrong.is_empty() {
761                apply_errors.set(wrong);
762                return;
763            }
764            apply_errors.set(Vec::new());
765            let given = ParamValues(now.as_object().cloned().unwrap_or_default());
766            values.set(given.clone());
767            if let Some(on_apply) = &on_apply {
768                on_apply(&given);
769            }
770        };
771        let (back, back_errors, back_values, on_discard, back_fields) = (
772            draft.clone(),
773            errors.clone(),
774            self.values.clone(),
775            self.on_discard.clone(),
776            self.fields.clone(),
777        );
778        let discard = move || {
779            back.set(Value::Object(with_defaults(&back_fields, back_values.get().0)));
780            back_errors.set(Vec::new());
781            if let Some(on_discard) = &on_discard {
782                on_discard();
783            }
784        };
785        cx.child(
786            &VStack::new((
787                VStack::new(rows).spacing(12.0).align(Align::Stretch),
788                VStack::new(wrong).spacing(4.0),
789                HStack::new((
790                    Spacer::new(),
791                    Button::new(labels.param_discard.clone())
792                        .style(ButtonStyle::Ghost)
793                        .disabled(!changed)
794                        .on_click(discard),
795                    Button::new(labels.param_apply.clone())
796                        .icon(IconName::Check)
797                        .disabled(!changed)
798                        .on_click(apply),
799                ))
800                .spacing(8.0)
801                .align(Align::Center),
802            ))
803            .spacing(14.0)
804            .align(Align::Stretch),
805        )
806    }
