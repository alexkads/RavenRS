30/// Breaks `text` into lines that fit `width`, as ranges of char indices. Line
31/// breaks in the text always break, the break itself belongs to no line, and
32/// long lines break after a space, or anywhere in a word wider than `width`.
33#[doc(hidden)]
34pub fn wrap_lines(text: &str, width: f32, mut measure: impl FnMut(&str) -> f32) -> Vec<(usize, usize)> {
35    let chars: Vec<char> = text.chars().collect();
36    let mut lines = Vec::new();
37    let mut paragraph = 0;
38    loop {
39        let end = chars[paragraph..].iter().position(|c| *c == '\n').map_or(chars.len(), |n| paragraph + n);
40        wrap_paragraph(&chars, paragraph, end, width, &mut measure, &mut lines);
41        if end == chars.len() {
42            return lines;
43        }
44        paragraph = end + 1;
45    }
46}

48fn wrap_paragraph(
49    chars: &[char],
50    start: usize,
51    end: usize,
52    width: f32,
53    measure: &mut impl FnMut(&str) -> f32,
54    lines: &mut Vec<(usize, usize)>,
55) {
56    let piece = |a: usize, b: usize| chars[a..b].iter().collect::<String>();
57    let mut line = start;
58    let mut used = 0.0;
59    let mut at = start;
60    while at < end {
61        // A word with the spaces after it.
62        let mut next = at;
63        while next < end && !chars[next].is_whitespace() {
64            next += 1;
65        }
66        while next < end && chars[next].is_whitespace() {
67            next += 1;
68        }
69        let w = measure(&piece(at, next));
70        if used + w > width && line < at {
71            lines.push((line, at));
72            line = at;
73            used = 0.0;
74        }
75        if w > width {
76            // Wider than a whole line: cut the word where it stops fitting.
77            let mut from = at;
78            loop {
79                let mut to = from + 1;
80                while to < next && measure(&piece(from, to + 1)) <= width {
81                    to += 1;
82                }
83                if to == next {
84                    line = from;
85                    used = measure(&piece(from, to));
86                    break;
87                }
88                lines.push((from, to));
89                from = to;
90            }
91        } else {
92            used += w;
93        }
94        at = next;
95    }
96    lines.push((line, end));
97}

206impl View for SuggestionList {
207    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
208        let theme = *cx.theme();
209        let mut attrs = util::attrs(&theme, TextStyle::Body, None);
210        attrs.family = Some("monospace".to_owned());
211        attrs.size -= 1.0;
212        let caption = util::attrs(&theme, TextStyle::Caption, None);
213        // Twelve at most, with the chosen one among them.
214        let first = self.chosen.saturating_sub(SHOWN_SUGGESTIONS - 1);
215        let rows: Vec<NodeId> = self
216            .items
217            .iter()
218            .enumerate()
219            .skip(first)
220            .take(SHOWN_SUGGESTIONS)
221            .map(|(i, (label, detail, _))| {
222                let name = cx.text(label, &attrs, theme.color(Semantic::Ink), LayoutStyle::leaf());
223                let mut parts = vec![name];
224                if let Some(detail) = detail {
225                    let muted = theme.color(Semantic::InkMuted);
226                    let style = LayoutStyle { grow: 1.0, min_width: Some(0.0), ..LayoutStyle::leaf() };
227                    parts.push(cx.text_truncated(detail, &caption, muted, style));
228                }
229                let fill = if i == self.chosen {
230                    theme.color(Semantic::Accent).with_alpha(0.22)
231                } else {
232                    ravenrs_core::Color::TRANSPARENT
233                };
234                let row = cx.container(
235                    LayoutStyle {
236                        padding: Insets::symmetric(3.0, 8.0),
237                        spacing: 12.0,
238                        align: Align::Center,
239                        align_self: Some(Align::Stretch),
240                        ..LayoutStyle::stack(Axis::Horizontal)
241                    },
242                    &parts,
243                    Paint::Quad(QuadPaint { corner_radius: theme.radii.sm, ..QuadPaint::new(fill) }),
244                );
245                let pick = self.pick.clone();
246                cx.on_click(row, Rc::new(move || pick(i)));
247                cx.cursor(row, CursorIcon::Pointer);
248                cx.semantics(
249                    row,
250                    Semantics::new(Role::ListBoxOption).name(label.clone()).selected(i == self.chosen),
251                );
252                row
253            })
254            .collect();
255        let list = cx.container(
256            LayoutStyle {
257                padding: Insets::uniform(4.0),
258                width: Some(360.0),
259                overflow_hidden: true,
260                align: Align::Stretch,
261                ..LayoutStyle::stack(Axis::Vertical)
262            },
263            &rows,
264            Paint::Quad(QuadPaint {
265                corner_radius: theme.radii.md,
266                border: Some(Border { width: 1.0, color: theme.color(Semantic::Border) }),
267                ..QuadPaint::new(theme.color(Semantic::Surface))
268            }),
269        );
270        cx.semantics(list, Semantics::new(Role::ListBox).name(cx.labels().code_suggestions.clone()));
271        list
272    }

308/// Where the word that ends at `caret` starts: what a suggestion replaces.

311    while at > 0 && (chars[at - 1].is_alphanumeric() || chars[at - 1] == '_') {
312        at -= 1;
313    }

446/// Moves the caret to the same column of another line.
447fn move_lines(lines: &[(usize, usize)], c: &mut Cursor, down: bool, extend: bool) {
448    let here = line_of(lines, c.caret);
449    let column = c.caret - lines[here].0;
450    let there = if down { here + 1 } else { here.wrapping_sub(1) };
451    let to = match lines.get(there) {
452        Some((start, end)) => (start + column).min(*end),
453        None if down => lines[here].1,
454        None => lines[here].0,
455    };
456    c.move_to(to, extend);
457}

459/// Applies a key to a text area. Returns false for keys it does not handle.
460fn edit_area(
461    text: &mut String,
462    c: &mut Cursor,
463    lines: &[(usize, usize)],
464    key: &Key,
465    modifiers: Modifiers,
466    access: ClipboardAccess<'_>,
467    indent: usize,
468) -> bool {
469    let len = text.chars().count();
470    c.caret = c.caret.min(len);
471    let plain = !modifiers.ctrl && !modifiers.meta;
472    match key {
473        Key::Enter if indent > 0 && !plain => return false,
474        Key::Enter if indent > 0 => {
475            let chars: Vec<char> = text.chars().collect();
476            let start = lines[line_of(lines, c.caret)].0;
477            let before = &chars[start..c.caret.max(start)];
478            let mut spaces = before.iter().take_while(|ch| **ch == ' ').count();
479            if before
480                .iter()
481                .rev()
482                .find(|ch| !ch.is_whitespace())
483                .is_some_and(|ch| matches!(ch, '{' | '(' | '['))
484            {
485                spaces += indent;
486            }
487            insert(text, c, &format!("\n{}", " ".repeat(spaces)));
488        }
489        Key::Tab if indent > 0 && modifiers.shift => {
490            let start = lines[line_of(lines, c.caret)].0;
491            let leading = text.chars().skip(start).take(indent).take_while(|ch| *ch == ' ').count();
492            if leading > 0 {
493                let from = byte_index(text, start);
494                text.replace_range(from..byte_index(text, start + leading), "");
495                c.caret = c.caret.saturating_sub(leading).max(start);
496                c.anchor = None;
497            }
498        }
499        Key::Tab if indent > 0 => {
500            let column = c.caret - lines[line_of(lines, c.caret)].0;
501            insert(text, c, &" ".repeat(indent - column % indent));
502        }
503        Key::Enter => insert(text, c, "\n"),
504        Key::ArrowUp | Key::ArrowDown if plain => {
505            move_lines(lines, c, *key == Key::ArrowDown, modifiers.shift)
506        }
507        Key::Home if plain => c.move_to(lines[line_of(lines, c.caret)].0, modifiers.shift),
508        Key::End if plain => c.move_to(lines[line_of(lines, c.caret)].1, modifiers.shift),
509        Key::Character(s) if plain && s.contains(['\n', '\r']) => {
510            insert(text, c, &s.replace("\r\n", "\n").replace('\r', "\n"));
511        }
512        _ => return edit(text, c, key, modifiers, Some(access)),
513    }
514    true
515}

517impl View for TextArea {
518    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
519        let theme = *cx.theme();
520        let state = self.state;
521        let own: State<Cursor> = cx.use_state(|| Cursor::at(usize::MAX));
522        let cursor = self.cursor.clone().unwrap_or(own);
523        let own_request: State<bool> = cx.use_state(|| false);
524        let focus_request = self.focus_request.clone().unwrap_or(own_request);
525        let own_history: State<TextHistory> = cx.use_state(TextHistory::new);
526        let history = self.history.clone().unwrap_or(own_history);
527        let suggesting: State<bool> = cx.use_state(|| false);
528        let chosen: State<usize> = cx.use_state(|| 0);
529        // Where the pointer is over the code, and where it came to rest and since when.
530        let pointer: State<Option<(usize, Point)>> = cx.use_state(|| None);
531        let resting: State<Option<(usize, f64)>> = cx.use_state(|| None);
532        // The length and height of the text last time, to keep its end in sight.
533        let end_seen: State<(usize, f32)> = cx.use_state(|| (0, 0.0));
534        if focus_request.get() && !state.disabled {
535            focus_request.set(false);
536            cx.request_focus();
537        }
538        let followed: State<usize> = cx.use_state(|| usize::MAX);
539        let focused = cx.is_focused() && !state.disabled;
540        let hovered = cx.is_hovered() && !state.disabled;
541        let text = self.value.get();
542        let chars: Vec<char> = text.chars().collect();
543        let code = self.code.clone();
544        let mut attrs = util::attrs(&theme, TextStyle::Body, None);
545        if code.is_some() {
546            attrs.family = Some("monospace".to_owned());
547            attrs.size -= 1.0;
548        }
549        let attrs = attrs;
550        let line_height = attrs.line_height;
551
552        // The width of the last layout, known from the second build on.
553        let viewport = cx.scroll_viewport();
554        let outer = self.width.or(viewport.map(|v| v.width)).unwrap_or(FIRST_GUESS_WIDTH);
555        let inner = (outer - 2.0 * AREA_PADDING).max(1.0);
556        let lines = if code.is_some() {
557            // Code does not wrap: a line is what lies between line breaks.
558            wrap_lines(&text, f32::INFINITY, |_| 0.0)
559        } else {
560            let attrs = attrs.clone();
561            wrap_lines(&text, inner, |s| cx.measure_text(s, &attrs).width)
562        };
563        let lines = Rc::new(lines);
564        let fill = code.as_ref().is_some_and(|c| c.fill);
565        let rows_shown = if fill {
566            viewport.map_or(self.rows, |v| ((v.height - AREA_PADDING) / line_height).ceil().max(1.0) as usize)
567        } else {
568            self.grows.map_or(self.rows, |(min, max)| lines.len().clamp(min, max))
569        };
570        let height = rows_shown as f32 * line_height + AREA_PADDING;
571        // In a monospaced font a character's place is its column times one advance, which
572        // also counts the spaces a measured piece would leave out at its end.
573        let advance = code.is_some().then(|| cx.measure_text("0000000000", &attrs).width / 10.0);
574        // The line numbers' width, and the room they keep beside the code.
575        let gutter_width = code.as_ref().filter(|c| c.line_numbers).map(|_| {
576            let digits = lines.len().max(1).to_string().len().max(2);
577            digits as f32 * cx.measure_text("0", &attrs).width
578        });
579        let gutter_room = gutter_width.map_or(0.0, |w| w + GUTTER_GAP);
580        let cur = cursor.get();
581        let caret = cur.caret.min(chars.len());
582        let caret_line = line_of(&lines, caret);
583
584        // Keep the caret in view after it moves, and leave the wheel alone otherwise.
585        let scrolled = cx.scroll_offset();
586        let mut offset = scrolled.y;
587        if code.as_ref().is_some_and(|c| c.follow_end) {
588            let total = lines.len() as f32 * line_height;
589            let (seen_len, seen_total) = *end_seen.peek();
590            if seen_len != chars.len() {

592                // Only when the end was in sight: someone who scrolled up to read stays there.
593                if offset + visible >= seen_total - line_height * 1.5 {
594                    offset = (total - visible).max(0.0);
595                    cx.set_scroll_offset(Point::new(scrolled.x, offset));
596                }
597                end_seen.set_if_changed((chars.len(), total));
598            }
599        }
600        // Code scrolls sideways too, since it does not wrap.
601        let mut sideways = if code.is_some() { scrolled.x } else { 0.0 };
602        // Everything fits again once text is deleted, and nothing is left to scroll.
603        if self.grows.is_some() && lines.len() <= rows_shown && offset > 0.0 {
604            offset = 0.0;
605            cx.set_scroll_offset(Point::new(sideways, 0.0));
606        }
607        let follow = focused || code.as_ref().is_some_and(|c| c.follow_caret);
608        if follow && followed.get() != caret {
609            followed.set_if_changed(caret);
610            let top = caret_line as f32 * line_height;
611            // While it grows, the last layout's height is already out of date.
612            let last_height = viewport.filter(|_| self.grows.is_none()).map_or(height, |v| v.height);
613            let visible = last_height - AREA_PADDING;
614            let next = if top < offset {
615                top
616            } else if top + line_height > offset + visible {
617                top + line_height - visible
618            } else {
619                offset
620            };
621            // Sideways, the caret's column stays in view, between the line numbers and the edge.
622            let across = match (advance, viewport) {
623                (Some(advance), Some(v)) => {
624                    let column = caret - lines[caret_line].0;
625                    let x = column as f32 * advance;
626                    let visible = (v.width - 2.0 * AREA_PADDING - gutter_room).max(advance);
627                    if x < sideways {
628                        (x - advance * 4.0).max(0.0)
629                    } else if x + advance > sideways + visible {
630                        x + advance * 4.0 - visible
631                    } else {
632                        sideways
633                    }
634                }
635                _ => sideways,
636            };
637            if next != offset || across != sideways {
638                offset = next;
639                sideways = across;
640                cx.set_scroll_offset(Point::new(across, next));
641            }
642        }
643

645        let first = ((offset / line_height).floor() as usize).min(lines.len().saturating_sub(1));

648        let mut boundaries: Vec<(usize, Vec<f32>)> = Vec::with_capacity(last - first);
649        let mut rows = Vec::with_capacity(last - first + 2);

651            rows.push(cx.container(
652                LayoutStyle { height: Some(first as f32 * line_height), ..LayoutStyle::leaf() },
653                &[],
654                Paint::None,
655            ));
656        }
657        if text.is_empty() {
658            let color = state.ink(theme.color(Semantic::InkMuted));
659            rows.push(cx.text(&self.placeholder, &attrs, color, LayoutStyle::leaf()));
660            boundaries.push((0, vec![0.0]));
661        } else {
662            let spans =
663                code.as_ref().and_then(|c| c.highlight.as_ref()).map(|h| h(&text)).unwrap_or_default();
664            for &(a, b) in &lines[first..last] {
665                let shown = piece(a, b);
666                let color = state.ink(theme.color(Semantic::Ink));
667                let line = LayoutStyle { height: Some(line_height), ..LayoutStyle::leaf() };
668                let xs: Vec<f32> = match advance {
669                    Some(advance) => (a..=b).map(|i| (i - a) as f32 * advance).collect(),
670                    None => (a..=b).map(|i| cx.measure_text(&piece(a, i), &attrs).width).collect(),
671                };
672                rows.push(if shown.is_empty() {
673                    cx.container(line, &[], Paint::None)
674                } else if spans.is_empty() {
675                    cx.text(&shown, &attrs, color, line)
676                } else {
677                    // Each colored stretch sits where its first character falls.
678                    let mut pieces = Vec::new();
679                    let mut at = a;
680                    let mut push = |cx: &mut BuildCx<'_>, from: usize, to: usize, semantic: Semantic| {
681                        let words = piece(from, to);
682                        if words.trim().is_empty() {
683                            return;
684                        }
685                        let place =
686                            LayoutStyle { absolute: Some((xs[from - a], 0.0)), ..LayoutStyle::leaf() };
687                        pieces.push(cx.text(&words, &attrs, state.ink(theme.color(semantic)), place));
688                    };
689                    for &(from, to, semantic) in spans.iter().filter(|(from, to, _)| *to > a && *from < b) {
690                        let (from, to) = (from.max(a), to.min(b));
691                        if from > at {
692                            push(cx, at, from, Semantic::Ink);
693                        }
694                        push(cx, from, to, semantic);
695                        at = to;
696                    }
697                    if at < b {
698                        push(cx, at, b, Semantic::Ink);
699                    }
700                    cx.container(line, &pieces, Paint::None)
701                });
702                boundaries.push((a, xs));
703            }
704        }
705        if last < lines.len() {
706            rows.push(cx.container(
707                LayoutStyle {
708                    height: Some((lines.len() - last) as f32 * line_height),
709                    ..LayoutStyle::leaf()
710                },
711                &[],
712                Paint::None,
713            ));
714        }
715        let boundaries = Rc::new(boundaries);
716
717        // The selection on each visible line, and the caret, under the text.
718        let x_at = |i: usize, line: usize| -> f32 {
719            let Some((start, xs)) = boundaries.get(line.wrapping_sub(first)) else { return 0.0 };
720            xs.get(i.saturating_sub(*start)).copied().unwrap_or(0.0)
721        };
722        let selected: Vec<(f32, f32, f32)> = cur
723            .selection()
724            .map(|(a, b)| {
725                (first..last)
726                    .filter(|&n| lines[n].0 <= b && a <= lines[n].1)
727                    .map(|n| {
728                        let (from, to) = (a.max(lines[n].0), b.min(lines[n].1));
729                        (n as f32 * line_height, x_at(from, n), x_at(to, n).max(x_at(from, n) + 4.0))
730                    })
731                    .collect()
732            })
733            .unwrap_or_default();
734        // What a search found, on the visible lines, under the selection.
735        let in_view: &[(usize, usize)] = &lines;
736        let found_bands: Vec<(f32, f32, f32)> = code
737            .as_ref()
738            .map(|c| {
739                c.found
740                    .iter()
741                    .flat_map(|&(a, b)| {
742                        (first..last)
743                            .filter(move |&n| in_view[n].0 <= b && a <= in_view[n].1 && a < b)
744                            .map(move |n| (n, a.max(in_view[n].0), b.min(in_view[n].1)))
745                    })
746                    .map(|(n, from, to)| {
747                        (n as f32 * line_height, x_at(from, n), x_at(to, n).max(x_at(from, n) + 4.0))
748                    })
749                    .collect()
750            })
751            .unwrap_or_default();
752        let found_color = theme.color(Semantic::Warning).with_alpha(0.28);
753        let caret_spot = (focused && cur.selection().is_none() && state.accepts_input())
754            .then(|| (x_at(caret, caret_line), caret_line as f32 * line_height));

756        // A search's current match is the selection: stronger than the other matches under it.
757        let searching = code.as_ref().is_some_and(|c| c.follow_caret);
758        let highlight = accent.with_alpha(if searching {
759            0.45
760        } else if focused {
761            0.32
762        } else {
763            0.16
764        });

766        let marks: Vec<DrawnMark> = code
767            .as_ref()
768            .map(|c| {
769                c.marks
770                    .iter()
771                    .filter(|(n, _, _)| (first..last).contains(n))
772                    .map(|&(n, semantic, columns)| {
773                        let under = columns.map(|(from, to)| {
774                            let start = lines[n].0;
775                            let (from, to) = ((start + from).min(lines[n].1), (start + to).min(lines[n].1));
776                            (x_at(from, n), x_at(to, n).max(x_at(from, n) + 8.0))
777                        });
778                        (n as f32 * line_height, theme.color(semantic), under)
779                    })
780                    .collect()
781            })
782            .unwrap_or_default();

784        // Code is as wide as its longest line, and the area scrolls to it; room for the caret after.
785        let widest = advance.map(|advance| {
786            let columns = lines.iter().map(|&(a, b)| b - a).max().unwrap_or(0);
787            columns as f32 * advance + 2.0
788        });
789        // The suggestions that fit the word being typed, and a point under its start for them.
790        let prefix_start = word_before(&chars, caret);
791        let typed_word: String = chars[prefix_start..caret].iter().collect::<String>().to_lowercase();
792        let offered: Vec<Suggestion> = match &code {
793            Some(look) if focused && suggesting.get() => look
794                .suggestions
795                .iter()
796                .filter(|(label, _, _)| {

798                })
799                .take(50)
800                .cloned()
801                .collect(),
802            _ => Vec::new(),
803        };
804        let list_anchor = (!offered.is_empty()).then(|| {

806            cx.container(
807                LayoutStyle {
808                    absolute: Some(spot),
809                    width: Some(1.0),
810                    height: Some(1.0),
811                    ..LayoutStyle::leaf()
812                },
813                &[],
814                Paint::None,
815            )
816        });
817        rows.extend(list_anchor);
818        let content = cx.container(
819            LayoutStyle {
820                align: Align::Start,
821                grow: if code.is_some() { 1.0 } else { 0.0 },
822                min_width: widest,
823                ..LayoutStyle::stack(Axis::Vertical)
824            },
825            &rows,
826            Paint::Custom(Rc::new(move |f: Rect, scene| {
827                for (y, color, under) in &marks {
828                    let band = Rect::new(
829                        f.x() - AREA_PADDING / 2.0,
830                        f.y() + y,
831                        f.width() + AREA_PADDING,
832                        line_height,
833                    );
834                    scene.push_quad(Quad::new(band, color.with_alpha(0.14)));
835                    if let Some((a, b)) = under {
836                        let line = Rect::new(f.x() + a, f.y() + y + line_height - 3.0, b - a, 2.0);
837                        scene.push_quad(Quad::new(line, *color).radius(1.0));
838                    }
839                }
840                for (y, a, b) in &found_bands {
841                    let band = Rect::new(f.x() + a, f.y() + y + 2.0, b - a, line_height - 4.0);
842                    scene.push_quad(Quad::new(band, found_color).radius(2.0));
843                }
844                for (y, a, b) in &selected {
845                    let band = Rect::new(f.x() + a, f.y() + y + 2.0, b - a, line_height - 4.0);
846                    scene.push_quad(Quad::new(band, highlight).radius(2.0));
847                }
848                if let Some((x, y)) = caret_spot {
849                    let bar = Rect::new(f.x() + x, f.y() + y + 2.0, 1.5, line_height - 4.0);
850                    scene.push_quad(Quad::new(bar, accent).radius(0.75));
851                }
852            })),
853        );
854        cx.semantics(content, Semantics::hidden());
855
856        // A tip about what the pointer rests on, asked once it has rested a moment.
857        let on_hover = code.as_ref().and_then(|c| c.on_hover.clone());
858        if !cx.is_hover_within() && pointer.peek().is_some() {
859            pointer.set(None);
860            resting.set(None);
861        }
862        let mut tip = None;
863        if let (Some(ask), Some((at, spot))) = (&on_hover, pointer.get()) {
864            let now = cx.now();
865            match resting.get() {
866                Some((rested, since)) if rested == at => {
867                    if now - since >= HOVER_DELAY {
868                        tip = code.as_ref().and_then(|c| c.hover.clone()).map(|text| (text, spot));
869                    } else {
870                        cx.wake_at(since + HOVER_DELAY);
871                    }
872                }
873                _ => {
874                    resting.set(Some((at, now)));
875                    cx.wake_at(now + HOVER_DELAY);
876                }
877            }
878            // Asked once per place: the build after the delay asks, the app answers later.
879            if resting.peek().is_some_and(|(rested, since)| rested == at && now - since >= HOVER_DELAY)
880                && tip.is_none()
881            {
882                ask(at);
883            }
884        }
885        if let Some((text, spot)) = tip {
886            cx.overlay(content, Placement::At(Point::new(spot.x, spot.y + 18.0)), &HoverTip(text), None);
887        }
888
889        // Line numbers, built for the same lines in view, with spacers for the rest.
890        let gutter_fill = state.fill(theme.color(code.as_ref().map_or(Semantic::Control, |c| c.background)));
891        let gutter = gutter_width.map(|width| {
892            let mut numbers = Vec::with_capacity(last - first + 2);
893            if first > 0 {
894                numbers.push(cx.container(
895                    LayoutStyle { height: Some(first as f32 * line_height), ..LayoutStyle::leaf() },
896                    &[],
897                    Paint::None,
898                ));
899            }
900            let marked: Vec<(usize, Semantic)> =
901                code.as_ref().map(|c| c.marks.iter().map(|(n, s, _)| (*n, *s)).collect()).unwrap_or_default();
902            for n in first..last.max(first + 1).min(lines.len().max(1)) {
903                let label = (n + 1).to_string();
904                let semantic = marked.iter().find(|(m, _)| *m == n).map_or(Semantic::InkMuted, |(_, s)| *s);
905                let x = width - cx.measure_text(&label, &attrs).width;
906                let place = LayoutStyle { absolute: Some((x, 0.0)), ..LayoutStyle::leaf() };
907                let number = cx.text(&label, &attrs, state.ink(theme.color(semantic)), place);
908                let row = cx.container(
909                    LayoutStyle { height: Some(line_height), width: Some(width), ..LayoutStyle::leaf() },
910                    &[number],
911                    Paint::None,
912                );
913                // A marked line number takes the caret to the mark, with the focus.
914                let mark = code.as_ref().and_then(|c| c.marks.iter().find(|(m, _, _)| *m == n).cloned());
915                if let (Some((_, _, columns)), Some(&(start, end))) = (mark, lines.get(n))
916                    && !state.disabled
917                {
918                    let (to, focus) = (cursor.clone(), focus_request.clone());
919                    let at = (start + columns.map_or(0, |(a, _)| a)).min(end);
920                    cx.on_click(
921                        row,
922                        Rc::new(move || {
923                            to.set(Cursor::at(at));
924                            focus.set(true);
925                        }),
926                    );
927                    cx.cursor(row, CursorIcon::Pointer);
928                }
929                numbers.push(row);
930            }
931            if last < lines.len() {
932                numbers.push(cx.container(
933                    LayoutStyle {
934                        height: Some((lines.len() - last) as f32 * line_height),
935                        ..LayoutStyle::leaf()
936                    },
937                    &[],
938                    Paint::None,
939                ));
940            }
941            let column = cx.container(
942                LayoutStyle {
943                    width: Some(width + GUTTER_GAP),
944                    // Down to the area's bottom border, even when the code ends higher.
945                    min_height: viewport.map(|v| (v.height - AREA_PADDING / 2.0 - BORDER_ROOM).max(0.0)),
946                    padding: Insets { right: GUTTER_GAP, ..Insets::default() },
947                    ..LayoutStyle::stack(Axis::Vertical)
948                },
949                &numbers,
950                Paint::Custom(Rc::new(move |f: Rect, scene| {
951                    // Over the code that scrolls under it, from inside the area's border to the gap.
952                    let under = Rect::new(
953                        f.x() - AREA_PADDING + BORDER_ROOM,
954                        f.y(),
955                        f.width() + AREA_PADDING - BORDER_ROOM - 4.0,
956                        f.height(),
957                    );
958                    scene.push_quad(Quad::new(under, gutter_fill));
959                    for (y, color) in &gutter_marks {
960                        let bar = Rect::new(f.x() - AREA_PADDING, f.y() + y, 3.0, line_height);
961                        scene.push_quad(Quad::new(bar, *color));
962                    }
963                })),
964            );
965            cx.semantics(column, Semantics::hidden());
966            // Drawn after the code and over it, and moved back by what the code scrolled
967            // sideways, so the numbers stay put.
968            let pinned = cx.container(
969                LayoutStyle { absolute: Some((AREA_PADDING, AREA_PADDING / 2.0)), ..LayoutStyle::leaf() },
970                &[column],
971                Paint::None,
972            );
973            cx.translate(pinned, Point::new(sideways, 0.0));
974            pinned
975        });
976        // What the line numbers take in the row, beside the code.
977        let gutter_space = gutter.map(|_| {
978            cx.container(LayoutStyle { width: Some(gutter_room), ..LayoutStyle::leaf() }, &[], Paint::None)
979        });
980
981        let border = if focused {
982            Border { width: 1.5, color: accent }
983        } else {
984            let base = theme.color(Semantic::Border);
985            Border {
986                width: 1.0,
987                color: state.ink(if hovered { base.lerp(theme.color(Semantic::Ink), 0.25) } else { base }),
988            }
989        };
990        let children: Vec<NodeId> = gutter_space.into_iter().chain([content]).chain(gutter).collect();
991        let background = code.as_ref().map_or(Semantic::Control, |c| c.background);
992        let node = cx.container(
993            LayoutStyle {
994                padding: Insets::symmetric(AREA_PADDING / 2.0, AREA_PADDING),
995                height: if fill { None } else { Some(height) },
996                min_height: fill.then_some(line_height + AREA_PADDING),
997                grow: if fill { 1.0 } else { 0.0 },
998                width: self.width,
999                min_width: Some(160.0),
1000                overflow_hidden: true,
1001                align_self: Some(if self.width.is_some() { Align::Start } else { Align::Stretch }),
1002                ..LayoutStyle::stack(if code.is_some() { Axis::Horizontal } else { Axis::Vertical })
1003            },
1004            &children,
1005            Paint::Quad(QuadPaint {
1006                corner_radius: theme.radii.md,
1007                border: Some(border),
1008                ..QuadPaint::new(state.fill(theme.color(background)))
1009            }),
1010        );
1011        cx.scrollable_virtual(node, code.is_some());
1012        crate::motion::control(cx, node);
1013        let mut semantics = Semantics::new(Role::TextInput).value(text.clone());
1014        if !self.placeholder.is_empty() {
1015            semantics = semantics.name(self.placeholder.clone());
1016            semantics.fallback_name = true;
1017        }
1018        cx.semantics(node, state.semantics(semantics));
1019        if state.disabled {
1020            return node;
1021        }

1023        // Press places the caret, dragging selects, Shift-click extends.

1025            let (value, cursor, history, suggesting) =
1026                (self.value.clone(), cursor.clone(), history.clone(), suggesting.clone());
1027            let offered = offered.clone();

1029                let Some((_, _, put)) = offered.get(i) else { return };
1030                let before = value.get();
1031                let c = cursor.get();
1032                let chars: Vec<char> = before.chars().collect();
1033                let caret = c.caret.min(chars.len());
1034                let start = word_before(&chars, caret);
1035                let mut after = before.clone();
1036                after.replace_range(byte_index(&before, start)..byte_index(&before, caret), put);
1037                history.update(|h| h.record_change(&before, &after, c));
1038                value.set(after);
1039                cursor.set(Cursor::at(start + put.chars().count()));
1040                suggesting.set(false);
1041            })
1042        };
1043        if let Some(anchor) = list_anchor {

1045                let suggesting = suggesting.clone();
1046                Rc::new(move || suggesting.set(false))
1047            };
1048            let list = SuggestionList {
1049                items: offered.clone(),
1050                chosen: chosen.get().min(offered.len() - 1),
1051                pick: accept.clone(),
1052            };
1053            cx.overlay(anchor, Placement::Below, &list, Some(close));
1054        }
1055        let hit = {
1056            let (lines, boundaries) = (lines.clone(), boundaries.clone());
1057            move |p: Point, frame: Rect| -> usize {
1058                let n = (((p.y - frame.y()) / line_height).max(0.0) as usize).min(lines.len() - 1);
1059                let Some((start, xs)) = n.checked_sub(first).and_then(|i| boundaries.get(i)) else {
1060                    return lines[n].0;
1061                };
1062                start + index_at(xs, p.x - frame.x())
1063            }
1064        };
1065        let go_to = code.as_ref().and_then(|c| c.on_go_to.clone());
1066        if on_hover.is_some() {
1067            let (pointer, hit, value) = (pointer.clone(), hit.clone(), self.value.clone());
1068            cx.on_pointer_move(
1069                content,
1070                Rc::new(move |p, frame| {
1071                    // Kept by the word under the pointer, so moving inside it asks nothing new.
1072                    let word = word_at(&value.get(), hit(p, frame)).0;
1073                    if pointer.peek().map(|(at, _)| at) != Some(word) {
1074                        pointer.set(Some((word, p)));
1075                    }
1076                }),
1077            );
1078        }

1080        cx.on_drag(
1081            content,
1082            Rc::new(move |d| {
1083                let at = hit(d.position, d.frame);
1084                let held = mods.get();
1085                if let (DragPhase::Start, Some(go), true) = (d.phase, &click_go, held.meta || held.ctrl) {
1086                    go(at);
1087                    return;
1088                }
1089                match d.phase {
1090                    DragPhase::Start => drag_cursor.update(|c| {
1091                        if mods.get().shift {
1092                            c.move_to(at, true);
1093                        } else {
1094                            *c = Cursor { caret: at, anchor: Some(at) };
1095                        }
1096                    }),
1097                    DragPhase::Move => drag_cursor.update(|c| c.move_to(at, true)),
1098                    _ => {}
1099                }
1100            }),
1101        );
1102        let (word_cursor, word_value) = (cursor.clone(), self.value.clone());
1103        cx.on_double_click(
1104            content,
1105            Rc::new(move || {
1106                let caret = word_cursor.get().caret;
1107                let (a, b) = word_at(&word_value.get(), caret);
1108                word_cursor.set(Cursor { caret: b, anchor: Some(a) });
1109            }),
1110        );
1111        cx.cursor(content, CursorIcon::Text);
1112
1113        let (value, clipboard) = (self.value.clone(), cx.clipboard());
1114        let indent = code.as_ref().map_or(0, |c| c.indent);
1115        let complete = code.as_ref().and_then(|c| c.on_complete.clone());
1116        let offered_count = offered.len();
1117        cx.on_key(
1118            node,
1119            Rc::new(move |key, modifiers| {
1120                let caret_now = cursor.get().caret.min(value.get().chars().count());
1121                if *key == Key::Function(12)
1122                    && let Some(go) = &go_to
1123                {
1124                    go(caret_now);
1125                    return true;
1126                }
1127                if *key == Key::Space
1128                    && modifiers.ctrl
1129                    && let Some(ask) = &complete
1130                {
1131                    ask(caret_now);
1132                    suggesting.set(true);
1133                    chosen.set(0);
1134                    return true;
1135                }
1136                if offered_count > 0 {
1137                    let at = chosen.get().min(offered_count - 1);
1138                    let took = match key {
1139                        Key::ArrowDown => {
1140                            chosen.set((at + 1) % offered_count);
1141                            true
1142                        }
1143                        Key::ArrowUp => {
1144                            chosen.set((at + offered_count - 1) % offered_count);
1145                            true
1146                        }
1147                        Key::Enter | Key::Tab => {
1148                            accept(at);
1149                            true
1150                        }
1151                        Key::Escape => {
1152                            suggesting.set(false);
1153                            true
1154                        }
1155                        _ => false,
1156                    };
1157                    if took {
1158                        return true;
1159                    }
1160                }
1161                let mut text = value.get();
1162                let mut c = cursor.get();
1163                let access = ClipboardAccess { clipboard: &clipboard, secure: false, multiline: true };
1164                let handled = if state.accepts_input() {
1165                    let mut handled = false;
1166                    history.update(|steps| {
1167                        handled = steps.apply(&mut text, &mut c, key, modifiers, |t, c| {
1168                            edit_area(t, c, &lines, key, modifiers, access, indent)
1169                        });
1170                    });
1171                    handled
1172                } else {
1173                    edit_area(&mut text, &mut c, &lines, key, modifiers, access, indent)
1174                };
1175                if handled {
1176                    let changed = text != value.get();
1177                    if changed && !state.accepts_input() {
1178                        return true;
1179                    }
1180                    cursor.set(c);
1181                    // `.` and `::` ask for suggestions; a word keeps them open, anything else closes.
1182                    if let Some(ask) = complete.as_ref().filter(|_| changed) {
1183                        let before = c.caret.checked_sub(2).and_then(|i| text.chars().nth(i));
1184                        match key {
1185                            Key::Character(typed)
1186                                if typed == "." || (typed == ":" && before == Some(':')) =>
1187                            {
1188                                ask(c.caret);
1189                                suggesting.set(true);
1190                                chosen.set(0);
1191                            }
1192                            Key::Character(typed)
1193                                if typed.chars().all(|ch| ch.is_alphanumeric() || ch == '_') =>
1194                            {
1195                                chosen.set(0);
1196                            }
1197                            Key::Backspace => {}
1198                            _ => suggesting.set(false),
1199                        }
1200                    } else if !changed && suggesting.get() {
1201                        suggesting.set(false);
1202                    }
1203                    if changed {
1204                        value.set(text);
1205                    }
1206                }
1207                handled
1208            }),
1209        );
1210        cx.focusable(node, theme.radii.md);
1211        cx.cursor(node, CursorIcon::Text);
1212        node
1213    }
1214}

1218/// Reads a number typed with these separators. Besides the exact format it
1219/// takes the other common way to write a decimal, so "12.5" works where the
1220/// decimal separator is a comma: a lone group separator counts as decimal unless
1221/// exactly three digits follow it.
1222#[doc(hidden)]
1223pub fn parse_number(text: &str, decimal: char, group: char) -> Option<f64> {
1224    let trimmed = text.trim();
1225    let normal: String = if trimmed.contains(decimal) {
1226        trimmed.chars().filter(|c| *c != group).map(|c| if c == decimal { '.' } else { c }).collect()
1227    } else if trimmed.matches(group).count() == 1 {
1228        let (_, after) = trimmed.split_once(group)?;
1229        if after.chars().count() == 3 {
1230            trimmed.chars().filter(|c| *c != group).collect()
1231        } else {
1232            trimmed.replace(group, ".")
1233        }
1234    } else {
1235        trimmed.chars().filter(|c| *c != group).collect()
1236    };
1237    let valid = !normal.is_empty()
1238        && normal.chars().enumerate().all(|(i, c)| c.is_ascii_digit() || c == '.' || (i == 0 && c == '-'));
1239    if !valid {
1240        return None;
1241    }
1242    normal.parse().ok().filter(|n: &f64| n.is_finite())
1243}

1520#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1521enum MaskSlot {
1522    Digit,
1523    Letter,
1524    Any,
1525    Literal(char),
1526}
1527
1528#[cfg_attr(feature = "live", ravenrs_live::expose)]
1529impl MaskSlot {
1530    fn accepts(self, c: char) -> bool {
1531        match self {
1532            Self::Digit => c.is_ascii_digit(),
1533            Self::Letter => c.is_alphabetic(),
1534            Self::Any => c.is_alphanumeric(),
1535            Self::Literal(_) => false,
1536        }
1537    }
1538}

1540/// `0` is a digit, `a` a letter and `*` a letter or digit. `\` makes the next
1541/// character plain, and any other character is shown as it is.
1542fn parse_mask(mask: &str) -> Vec<MaskSlot> {
1543    let mut chars = mask.chars();
1544    let mut slots = Vec::new();
1545    while let Some(c) = chars.next() {
1546        slots.push(match c {
1547            '0' => MaskSlot::Digit,
1548            'a' => MaskSlot::Letter,
1549            '*' => MaskSlot::Any,
1550            '\\' => MaskSlot::Literal(chars.next().unwrap_or('\\')),
1551            other => MaskSlot::Literal(other),
1552        });
1553    }
1554    slots
1555}
1556
1557/// Pulls what the slots accept out of text in the mask's format, or any text.
1558fn mask_extract(slots: &[MaskSlot], text: &str) -> String {
1559    let mut chars = text.chars().peekable();
1560    let mut raw = String::new();
1561    for slot in slots {
1562        match slot {
1563            MaskSlot::Literal(l) => {
1564                if chars.peek() == Some(l) {
1565                    chars.next();
1566                }
1567            }
1568            slot => {
1569                for c in chars.by_ref() {
1570                    if slot.accepts(c) {
1571                        raw.push(c);
1572                        break;
1573                    }
1574                }
1575            }
1576        }
1577    }
1578    raw
1579}
1580
1581/// Puts `raw` in the mask, stopping where it runs out.
1582fn mask_fill(slots: &[MaskSlot], raw: &str) -> String {
1583    let mut chars = raw.chars();
1584    let mut out = String::new();
1585    let mut pending = String::new();
1586    for slot in slots {
1587        match slot {
1588            MaskSlot::Literal(l) => pending.push(*l),
1589            _ => {
1590                let Some(c) = chars.next() else { break };
1591                out.push_str(&pending);
1592                pending.clear();
1593                out.push(c);
1594            }
1595        }
1596    }
1597    out
1598}

1828/// Reads the four digits a time field holds, once they are all there.
1829fn time_from_digits(digits: &str) -> Option<Time> {
1830    if digits.len() != 4 {
1831        return None;
1832    }
1833    Time::new(digits[..2].parse().ok()?, digits[2..].parse().ok()?)
1834}

1836impl View for TimePicker {
1837    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1838        let open: State<bool> = cx.use_state(|| false);
1839        // Unfinished digits live here until they make a time.
1840        let draft: State<Option<String>> = cx.use_state(|| None);
1841        let labels = cx.labels();
1842        let (read, write, target, value) =
1843            (draft.clone(), draft.clone(), self.value.clone(), self.value.clone());
1844        let digits = Binding::new(
1845            move || {
1846                read.get().unwrap_or_else(|| {
1847                    value.get().map_or_else(String::new, |t| format!("{:02}{:02}", t.hour, t.minute))
1848                })
1849            },
1850            move |text: String| {
1851                let time = time_from_digits(&text);
1852                write.set(if time.is_some() || text.is_empty() { None } else { Some(text.clone()) });
1853                if time.is_some() || text.is_empty() {
1854                    target.set(time);
1855                }
1856            },
1857        );
1858        let mut field = MaskedField::new(digits, "00:00")
1859            .width(self.width)
1860            .disabled(self.state.disabled)
1861            .read_only(self.state.read_only);
1862        if let Some(label) = &self.label {
1863            field = field.label(label.clone());
1864        }
1865        let field = cx.child(&field);
1866        let accepts = self.state.accepts_input();
1867        let toggle = open.clone();
1868        let button = cx.child(
1869            &IconButton::new(IconName::Clock)
1870                .label(labels.pick_time.clone())
1871                .disabled(!accepts)
1872                .on_click(move || toggle.update(|o| *o = !*o)),
1873        );
1874        let node = cx.container(
1875            LayoutStyle {
1876                spacing: 4.0,
1877                align: Align::Center,
1878                align_self: Some(Align::Fit),
1879                ..LayoutStyle::stack(Axis::Horizontal)
1880            },
1881            &[field, button],
1882            Paint::None,
1883        );
1884        if open.get() && accepts {
1885            let slots: Vec<Time> = (0..24 * 60)
1886                .step_by(self.step as usize)
1887                .filter_map(|m: u16| Time::new((m / 60) as u8, (m % 60) as u8))
1888                .collect();
1889            let options: Vec<String> = slots.iter().map(Time::to_string).collect();
1890            let chosen = self.value.clone();
1891            let near = self.value.get().map_or(0, |t| slots.iter().rposition(|s| *s <= t).unwrap_or(0));
1892            let (clear, closing) = (draft, open.clone());
1893            let selected = Binding::new(
1894                move || near,
1895                move |i: usize| {
1896                    clear.set(None);
1897                    chosen.set(slots.get(i).copied());
1898                },
1899            );
1900            let close: Rc<dyn Fn()> = Rc::new(move || closing.set(false));
1901            let list = Options { options: &options, selected, close: close.clone(), width: self.width };
1902            cx.overlay(node, Placement::Below, &list, Some(close));
1903        }
1904        node
1905    }
1906}

1953    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1954        let date = self.value.map(
1955            |v| v.map(|v| v.date),
1956            |v, date: Option<Date>| {
1957                *v = date.map(|date| DateTime { date, time: v.map_or_else(Time::default, |old| old.time) });

1961            |v| v.map(|v| v.time),
1962            |v, time: Option<Time>| {
1963                *v = match (*v, time) {
1964                    (Some(old), Some(time)) => Some(DateTime { date: old.date, time }),

1970            .fill()
1971            .disabled(self.state.disabled)
1972            .read_only(self.state.read_only);
1973        let mut hour = TimePicker::new(time)
1974            .step_minutes(self.step)
1975            .disabled(self.state.disabled)
1976            .read_only(self.state.read_only);
1977        if let Some(label) = &self.label {
1978            day = day.label(label.clone());
1979            hour = hour.label(label.clone());
1980        }
1981        let (day, hour) =
1982            (cx.child(&VStack::new((day,)).spacing(0.0).align(Align::Stretch).grow(1.0)), cx.child(&hour));

1984            LayoutStyle {
1985                spacing: 8.0,
1986                align: Align::Center,
1987                align_self: Some(Align::Stretch),
1988                wrap: true,
1989                ..LayoutStyle::stack(Axis::Horizontal)
