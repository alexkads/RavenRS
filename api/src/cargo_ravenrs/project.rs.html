6/// Create a project, optionally initialize Git and AI, check every chosen target and probe MCP.
7pub fn new_project(
8    path: &Path,
9    template: &str,
10    target: &str,
11    git: bool,
12    ai: bool,
13    check: bool,
14    plugin: bool,
15) -> Result<(), String> {
16    let name = path.file_name().and_then(|s| s.to_str()).ok_or("Nome de projeto inválido")?;
17    let files = project_files(name, template, target)?;
18    if path.exists() {
19        // Never overwrite user code, including on repeated bootstrap runs.
20        let expected = format!("# Generated by cargo-ravenrs {VERSION}");
21        if !path.join("Cargo.toml").is_file() || !read(&path.join("Cargo.toml"))?.starts_with(&expected) {
22            return Err(format!("{} já existe e não foi gerado por esta versão.", path.display()));
23        }
24        let state: DocumentMut = read(&path.join(".ravenrs-project.toml"))?
25            .parse()
26            .map_err(|e| format!("Metadados do projeto inválidos: {e}"))?;
27        let previous_template =
28            state.get("template").and_then(Item::as_str).ok_or("Modelo original ausente nos metadados")?;
29        let previous_target =
30            state.get("target").and_then(Item::as_str).ok_or("Alvo original ausente nos metadados")?;
31        if previous_template != template || previous_target != target {
32            return Err(format!(
33                "{} já contém um projeto {previous_template}/{previous_target}; você escolheu {template}/{target}. Use outra pasta para criar esse modelo/alvo. O projeto existente não foi alterado.",
34                path.display()
35            ));
36        }
37        println!("Projeto existente: preservando o código.");
38    } else {
39        std::fs::create_dir_all(path).map_err(|e| e.to_string())?;
40        for file in files {
41            write(path, &file.path, &file.contents)?;
42        }
43        write(
44            path,
45            ".ravenrs-project.toml",
46            &format!(
47                "# cargo-ravenrs {VERSION}\nversion = \"{VERSION}\"\ntemplate = \"{template}\"\ntarget = \"{target}\"\nai = {ai}\n"
48            ),
49        )?;
50    }
51    if git && !path.join(".git").exists() {
52        run(path, "git", &["init"])?;
53    }
54    if ai {
55        configure_ai(path, plugin)?;
56    }
57    if check {
58        if target != "web" {
59            run(path, "cargo", &["check"])?;
60        }
61        if target != "desktop" {
62            run(path, "cargo", &["check", "--target", "wasm32-unknown-unknown"])?;
63        }
64    }
65    crate::mcp::probe("cargo", &["ravenrs".into(), "mcp".into()], path)?;
66    let next = if target == "web" {
67        "cargo check --target wasm32-unknown-unknown  # veja README.md para servir web/"
68    } else {
69        "cargo run"
70    };
71    println!("Pronto.\n  cd {}\n  {next}\n  claude  # ou code .\n  cargo ravenrs doctor", path.display());
72    Ok(())
73}
