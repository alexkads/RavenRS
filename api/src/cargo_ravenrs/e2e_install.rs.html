13/// Prepare and validate the complete plan before writing any generated file.
14pub(crate) fn install(folder: &Path) -> Result<usize, String> {
15    let root = folder.canonicalize().map_err(|error| format!("{}: {error}", folder.display()))?;
16    let files = plan(&root)?;
17    for file in &files {
18        let mut path = root.clone();
19        for component in Path::new(&file.path).components() {
20            path.push(component);
21            if path.is_symlink() {
22                return Err(format!(
23                    "{} é um link; instale manualmente sem redirecionar arquivos.",
24                    path.display()
25                ));
26            }
27        }
28        if path.exists() && !path.is_file() {
29            return Err(format!("{} não é um arquivo; nenhum arquivo foi alterado.", path.display()));
30        }
31    }
32    let changed: Vec<_> = files
33        .into_iter()
34        .filter(|file| std::fs::read(root.join(&file.path)).ok().as_deref() != Some(file.contents.as_bytes()))
35        .collect();
36    for file in &changed {
37        write(&root, &file.path, &file.contents)?;
38    }
39    Ok(changed.len())
40}

42fn plan(root: &Path) -> Result<Vec<ProjectFile>, String> {
43    let original = read(&root.join("Cargo.toml"))?;
44    let mut manifest: DocumentMut = original.parse().map_err(|error| format!("Cargo.toml: {error}"))?;
45    let name = manifest.get("package").and_then(|item| item.get("name")).and_then(Item::as_str)
46        .ok_or("Selecione a pasta de um app gerado com --project; bibliotecas e workspaces virtuais não têm entrada para instalar.")?.to_owned();
47    if manifest.get("bin").is_some()
48        || manifest["package"].get("autobins").and_then(Item::as_bool) == Some(false)
49    {
50        return Err(
51            "Entrada de binário personalizada; configure o canal manualmente. Nenhum arquivo foi alterado."
52                .into(),
53        );
54    }
55    let state: DocumentMut = read(&root.join(".ravenrs-project.toml"))
56        .map_err(|_| "e2e install reconhece projetos de cargo ravenrs new; para outra entrada, configure o canal manualmente. Nenhum arquivo foi alterado.".to_owned())?
57        .parse().map_err(|error| format!(".ravenrs-project.toml: {error}"))?;
58    let kind =
59        state.get("template").and_then(Item::as_str).ok_or("Modelo ausente em .ravenrs-project.toml")?;
60    let target = state.get("target").and_then(Item::as_str).ok_or("Alvo ausente em .ravenrs-project.toml")?;
61    versions::ensure_current(root)?;
62    let generated = template::project_files(&name, kind, target)?;
63    let project = ravenrs_e2e_runner::project::read(root)?;
64    project.check_configs()?;
65    let app = project
66        .found
67        .iter()
68        .find(|found| found.folder == root && found.package == name && found.is_app())
69        .ok_or("Este pacote não é um app RavenRS com entrada nativa reconhecida.")?;
70    if app.binaries != [name.clone()] {
71        return Err("Selecione um app com um único binário de mesmo nome do pacote; outras entradas exigem configuração manual.".into());
72    }
73    for (file, scenario) in &app.scenarios {
74        if let Err(why) = scenario {
75            return Err(format!("{}: {why}", file.display()));
76        }
77    }
78    if !app.config.folder.components().all(|part| matches!(part, Component::Normal(_) | Component::CurDir)) {
79        return Err("A instalação escreve somente dentro do app; escolha uma pasta relativa de cenários sem .. em e2e.toml.".into());
80    }
81    let mut scenario_parts = Vec::new();
82    for part in app.config.folder.components() {
83        if let Component::Normal(name) = part {
84            scenario_parts.push(name.to_str().ok_or("A pasta de cenários precisa ser UTF-8.")?);
85        }
86    }
87    scenario_parts.push("start.raventest");
88    let scenario_path = scenario_parts.join("/");
89    let default_main = artifact(&generated, "src/main.rs")?;
90    let new_main = default_main
91        .replace("\"../tests/e2e/start.raventest\"", &format!("{:?}", format!("../{scenario_path}")));
92    let current_main = read(&root.join("src/main.rs"))?;
93    let legacy_main = template::legacy_main(&name)?;
94    let exact = [&legacy_main, &default_main, &new_main]
95        .iter()
96        .any(|known| known.trim_end() == current_main.trim_end());
97    let (recognized, already_current) = if exact {
98        (true, current_main.trim_end() == new_main.trim_end())
99    } else if let Some(current) = formatted(root, &current_main) {
100        let already = formatted(root, &new_main).as_deref() == Some(current.as_str());
101        (
102            already
103                || [&legacy_main, &default_main]
104                    .iter()
105                    .any(|known| formatted(root, known).as_deref() == Some(current.as_str())),
106            already,
107        )
108    } else {
109        (false, false)
110    };
111    if !recognized {
112        return Err("src/main.rs tem uma entrada personalizada; configure --e2e-serve manualmente antes da janela/persistência. Nenhum arquivo foi alterado.".into());
113    }
114    let expected: DocumentMut =
115        artifact(&generated, "Cargo.toml")?.parse().map_err(|error| format!("Modelo: {error}"))?;
116    if manifest.get("dependencies").and_then(|deps| deps.get("ravenrs")).is_none() {
117        return Err("A dependência ravenrs do modelo foi renomeada; configure a entrada manualmente.".into());
118    }
119    for dependency in ["ravenrs-e2e", "ravenrs-e2e-wire"] {
120        let global = manifest.get("dependencies").is_some_and(|deps| deps.get(dependency).is_some());
121        let deps = if global {
122            table_at(manifest.as_table_mut(), &["dependencies"])?
123        } else {
124            table_at(manifest.as_table_mut(), &["target", NATIVE, "dependencies"])?
125        };
126        if deps.get(dependency).is_none() {
127            deps.insert(dependency, expected["target"][NATIVE]["dependencies"][dependency].clone());
128        }
129        let item = deps.get_mut(dependency).ok_or("Dependência não inserida")?;
130        if item.get("package").and_then(Item::as_str).is_some_and(|package| package != dependency) {
131            return Err(format!("{dependency} aponta para outro pacote; configure manualmente."));
132        }
133        if dependency == "ravenrs-e2e-wire" {
134            dependency_feature(item, "serve")?;
135        }
136    }
137    let metadata = table_at(manifest.as_table_mut(), &["package", "metadata", "ravenrs", "e2e"])?;
138    for key in ["app", "binary"] {
139        if let Some(current) = metadata.get(key) {
140            if current.as_str() != Some(&name) {
141                return Err(format!(
142                    "package.metadata.ravenrs.e2e.{key} difere do app gerado; nenhum arquivo foi alterado."
143                ));
144            }
145        } else {
146            metadata.insert(key, value(&name));
147        }
148    }
149    append_feature(metadata, "features", "e2e-source")?;
150    let features = table_at(manifest.as_table_mut(), &["features"])?;
151    append_feature(features, "e2e-source", "ravenrs/e2e-source")?;
152    let scenario = if root.join(&scenario_path).exists() {
153        read(&root.join(&scenario_path))?
154    } else {
155        artifact(&generated, "tests/e2e/start.raventest")?
156    };
157    ravenrs_e2e_wire::Scenario::parse(&scenario).map_err(|error| format!("{scenario_path}: {error}"))?;
158    let mut files = vec![
159        ProjectFile { path: "Cargo.toml".into(), contents: manifest.to_string() },
160        ProjectFile {
161            path: "src/main.rs".into(),
162            contents: if already_current { current_main } else { new_main },
163        },
164        ProjectFile { path: scenario_path, contents: scenario },
165    ];
166    if kind == "app-sqlite" && root.join("tests/screen.rs").is_file() {
167        let screen = artifact(&generated, "tests/screen.rs")?;
168        let legacy = screen.replace("::tasks().into_view()", "::app()");
169        if read(&root.join("tests/screen.rs"))?.trim_end() == legacy.trim_end() {
170            files.push(ProjectFile { path: "tests/screen.rs".into(), contents: screen });
171        }
172    }
173    Ok(files)
174}

176fn artifact(files: &[ProjectFile], path: &str) -> Result<String, String> {
177    files
178        .iter()
179        .find(|file| file.path == path)
180        .map(|file| file.contents.clone())
181        .ok_or_else(|| format!("Modelo sem {path}"))
182}

184/// Parse/format stdin only: no file is rewritten, compiled or executed. Both inputs use the
185/// project's formatting configuration, so formatting alone never makes its entry "custom".
186fn formatted(root: &Path, source: &str) -> Option<String> {
187    let mut child = Command::new("rustfmt")
188        .args(["--emit", "stdout", "--edition", "2024"])
189        .current_dir(root)
190        .stdin(Stdio::piped())
191        .stdout(Stdio::piped())
192        .stderr(Stdio::null())
193        .spawn()
194        .ok()?;
195    let written = child.stdin.take().is_some_and(|mut input| input.write_all(source.as_bytes()).is_ok());
196    let output = child.wait_with_output().ok()?;
197    if !written || !output.status.success() {
198        return None;
199    }
200    String::from_utf8(output.stdout).ok().map(|text| text.replace("\r\n", "\n"))
201}

203fn table_at<'a>(table: &'a mut dyn TableLike, path: &[&str]) -> Result<&'a mut dyn TableLike, String> {
204    let Some((key, rest)) = path.split_first() else {
205        return Ok(table);
206    };
207    if table.get(key).is_none() {
208        table.insert(key, Item::Table(Table::new()));
209    }
210    let child = table
211        .get_mut(key)
212        .and_then(Item::as_table_like_mut)
213        .ok_or_else(|| format!("{} precisa ser uma tabela em Cargo.toml", path.join(".")))?;
214    table_at(child, rest)
215}

217fn append_feature(table: &mut dyn TableLike, key: &str, feature: &str) -> Result<(), String> {
218    if table.get(key).is_none() {
219        table.insert(key, value(toml_edit::Array::new()));
220    }
221    let list = table
222        .get_mut(key)
223        .and_then(Item::as_array_mut)
224        .ok_or_else(|| format!("{key} precisa ser uma lista de textos"))?;
225    if list.iter().any(|item| item.as_str().is_none_or(|text| text.trim().is_empty())) {
226        return Err(format!("{key} precisa ser uma lista de textos não vazios"));
227    }
228    if !list.iter().any(|item| item.as_str() == Some(feature)) {
229        list.push(feature);
230    }
231    Ok(())
232}

234fn dependency_feature(item: &mut Item, feature: &str) -> Result<(), String> {
235    if let Some(version) = item.as_str() {
236        let mut table = InlineTable::new();
237        table.insert("version", version.into());
238        let mut new = Value::InlineTable(table);
239        if let Some(old) = item.as_value() {
240            *new.decor_mut() = old.decor().clone();
241        }
242        *item = Item::Value(new);
243    }
244    let table =
245        item.as_table_like_mut().ok_or("A dependência ravenrs-e2e-wire precisa ser uma versão ou tabela")?;
246    append_feature(table, "features", feature)
247}
