1//! RavenRS dependency versions as Cargo resolves them, without changing any manifest.
2//!
3//! A project may declare `ravenrs = "=27.1.28"`, rename it (`ui = { package = "ravenrs" }`),
4//! put it under `[target.'cfg(..)'.dependencies]`, or inherit it with `ravenrs.workspace = true`.
5//! For inheritance, the version lives in `[workspace.dependencies]` of the workspace that owns
6//! the manifest: the one named by `package.workspace`, or else the nearest ancestor with a
7//! `[workspace]` table. A virtual workspace root has no `[package]`; there every member is read.
8use crate::read;
9use std::path::{Path, PathBuf};
10use toml_edit::{DocumentMut, Item, TableLike};
11
12/// One RavenRS dependency with the version Cargo would request.
13#[derive(Debug, Clone, PartialEq, Eq)]
14pub struct Declared {
15    /// Where it was declared, for messages: `<manifest>: <table>.<key>`.
16    pub origin: String,
17    /// Real package name (after `package = ".."` renames).
18    pub package: String,
19    /// Requirement as written, such as `=27.1.28` or `^27.1`.
20    pub version: String,
21}
22
23const KINDS: [&str; 3] = ["dependencies", "dev-dependencies", "build-dependencies"];
24
25/// Refuses a framework version different from the CLI before modifying or diagnosing a project.
26pub(crate) fn ensure_current(root: &Path) -> Result<(), String> {
27    let versions = ravenrs_versions(root)?;
28    if versions.is_empty() {
29        return Err("Nenhuma dependência RavenRS com versão no projeto nem no workspace.".into());
30    }
31    for declared in versions {
32        if declared.version.trim_start_matches(['=', '^', '~']) != crate::VERSION {
33            return Err(format!(
34                "{} ({}) usa {}, mas cargo-ravenrs usa {}. Instale a tag correspondente.",
35                declared.package,
36                declared.origin,
37                declared.version,
38                crate::VERSION
39            ));
40        }
41    }
42    Ok(())
43}

45fn manifest(path: &Path) -> Result<DocumentMut, String> {
46    read(path)?.parse().map_err(|e| format!("{}: {e}", path.display()))
47}

49/// Every RavenRS dependency with an explicit or inherited version, for the project at `root`.
50pub fn ravenrs_versions(root: &Path) -> Result<Vec<Declared>, String> {
51    let path = root.join("Cargo.toml");
52    let doc = manifest(&path)?;
53    let mut found = Vec::new();
54    if doc.get("package").is_some() {
55        let workspace = owning_workspace(root, &doc)?;
56        package_versions(&path, &doc, workspace.as_ref(), &mut found)?;
57    }
58    if doc.get("workspace").is_some() {
59        let own = Some((path.clone(), doc.clone()));
60        workspace_table_versions(&path, &doc, &mut found);
61        for member in members(root, &doc)? {
62            let member_path = member.join("Cargo.toml");
63            if member_path == path {
64                continue;
65            }
66            let member_doc = manifest(&member_path)?;
67            package_versions(&member_path, &member_doc, own.as_ref(), &mut found)?;
68        }
69    }
70    Ok(found)
71}

73/// The workspace manifest Cargo would use for a package at `root`.
74fn owning_workspace(root: &Path, doc: &DocumentMut) -> Result<Option<(PathBuf, DocumentMut)>, String> {
75    if doc.get("workspace").is_some() {
76        return Ok(Some((root.join("Cargo.toml"), doc.clone())));
77    }
78    if let Some(explicit) = doc["package"].get("workspace").and_then(Item::as_str) {
79        let path = root.join(explicit).join("Cargo.toml");
80        return Ok(Some((path.clone(), manifest(&path)?)));
81    }
82    let mut dir = root.canonicalize().map_err(|e| format!("{}: {e}", root.display()))?;
83    while dir.pop() {
84        let path = dir.join("Cargo.toml");
85        if path.is_file() {
86            let candidate = manifest(&path)?;
87            if candidate.get("workspace").is_some() {
88                return Ok(Some((path, candidate)));
89            }
90        }
91    }
92    Ok(None)
93}

95fn package_versions(
96    path: &Path,
97    doc: &DocumentMut,
98    workspace: Option<&(PathBuf, DocumentMut)>,
99    found: &mut Vec<Declared>,
100) -> Result<(), String> {
101    let mut tables: Vec<(String, &dyn TableLike)> = Vec::new();
102    for kind in KINDS {
103        if let Some(table) = doc.get(kind).and_then(Item::as_table_like) {
104            tables.push((kind.into(), table));
105        }
106    }
107    if let Some(targets) = doc.get("target").and_then(Item::as_table_like) {
108        for (cfg, target) in targets.iter() {
109            for kind in KINDS {
110                if let Some(table) = target.get(kind).and_then(Item::as_table_like) {
111                    tables.push((format!("target.'{cfg}'.{kind}"), table));
112                }
113            }
114        }
115    }
116    for (name, table) in tables {
117        for (key, item) in table.iter() {
118            let origin = format!("{}: {name}.{key}", path.display());
119            if item.get("workspace").and_then(Item::as_bool) == Some(true) {
120                let Some((ws_path, ws)) = workspace else {
121                    return Err(format!(
122                        "{origin} herda do workspace, mas nenhum [workspace] foi encontrado."
123                    ));
124                };
125                let Some(inherited) =
126                    ws.get("workspace").and_then(|w| w.get("dependencies")).and_then(|d| d.get(key))
127                else {
128                    return Err(format!(
129                        "{origin} herda do workspace, mas {} não declara {key}.",
130                        ws_path.display()
131                    ));
132                };
133                // The member may not rename an inherited dependency; the workspace entry decides.
134                push(found, format!("{origin} → {}", ws_path.display()), key, inherited);
135            } else {
136                push(found, origin, key, item);
137            }
138        }
139    }
140    Ok(())
141}

143fn workspace_table_versions(path: &Path, doc: &DocumentMut, found: &mut Vec<Declared>) {
144    if let Some(table) =
145        doc.get("workspace").and_then(|w| w.get("dependencies")).and_then(Item::as_table_like)
146    {
147        for (key, item) in table.iter() {
148            push(found, format!("{}: workspace.dependencies.{key}", path.display()), key, item);
149        }
150    }
151}

153fn push(found: &mut Vec<Declared>, origin: String, key: &str, item: &Item) {
154    let package = item.get("package").and_then(Item::as_str).unwrap_or(key);
155    if !package.starts_with("ravenrs") {
156        return;
157    }
158    let version = item.as_str().or_else(|| item.get("version").and_then(Item::as_str));
159    if let Some(version) = version {
160        let declared = Declared { origin, package: package.into(), version: version.into() };
161        if !found.iter().any(|d| d.package == declared.package && d.version == declared.version) {
162            found.push(declared);
163        }
164    }
165}

167/// Members of a workspace, expanding `dir/*` globs and honoring `exclude`.
168fn members(root: &Path, doc: &DocumentMut) -> Result<Vec<PathBuf>, String> {
169    let list = |key: &str| -> Vec<String> {
170        doc["workspace"]
171            .get(key)
172            .and_then(Item::as_array)
173            .map(|a| a.iter().filter_map(|v| v.as_str().map(str::to_owned)).collect())
174            .unwrap_or_default()
175    };
176    let exclude: Vec<PathBuf> = list("exclude").iter().map(|e| root.join(e)).collect();
177    let mut out = Vec::new();
178    for pattern in list("members") {
179        let candidates = match pattern.strip_suffix("/*") {
180            Some(parent) => {
181                let mut dirs: Vec<PathBuf> = std::fs::read_dir(root.join(parent))
182                    .map_err(|e| format!("workspace.members {pattern}: {e}"))?
183                    .filter_map(Result::ok)
184                    .map(|e| e.path())
185                    .collect();
186                dirs.sort();
187                dirs
188            }
189            None => vec![root.join(&pattern)],
190        };
191        for dir in candidates {
192            if dir.join("Cargo.toml").is_file() && !exclude.iter().any(|e| dir.starts_with(e)) {
193                out.push(dir);
194            }
195        }
196    }
197    Ok(out)
