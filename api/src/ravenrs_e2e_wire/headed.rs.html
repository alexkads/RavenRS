1//! The app's side of the channel with its real window on screen (`--headed` in the tools).
2//!
3//! The app builds its view as usual, on its test data, and runs it with
4//! `ravenrs_platform_native::run_driven(view, options, Headed::new(hello))`. The window's
5//! platform lends its interface to the [`Headed`] driver on the main thread whenever a command
6//! arrives, and the command runs through the same code as on the channel with no window, on the
7//! interface the window shows. Standard input is read on a thread of its own, so the window keeps
8//! drawing while the tool is quiet: between steps, during a pause or a slowed-down run.
9//!
10//! The interface's clock is the driver's ([`UiDriver::time`]): between two steps the app gets at
11//! most [`SETTLE`] seconds, enough for what a step started (a menu opening, a page sliding) to
12//! finish on screen, and then its time stands still until the next step. A slowed-down run or a
13//! pause therefore never lets a notice expire, or a timer fire, that a run at full speed would
14//! not: the same scenario passes or fails the same way at any pace, with or without the window.
15
16use std::io::{self, BufRead, BufReader, Write};

20use ravenrs_e2e::E2e;

26/// How long the app's clock runs after a step before it stands still until the next one, in
27/// seconds: what the step started finishes on screen, and a pause adds nothing after it.
28pub const SETTLE: f64 = 0.6;
29
30/// A line from the tool, or the end of its input.
31enum Line {
32    Text(String),
33    Closed,
34}
35
36/// Answers a tool on the interface of the app's real window. See the [module](self).
37pub struct Headed {
38    hello: Hello,
39    session: E2e,
40    filming: Option<Box<dyn Filming>>,
41    input: Option<Box<dyn BufRead + Send>>,
42    output: Box<dyn Write>,
43    lines: Option<Receiver<Line>>,
44    released: bool,
45    /// How far the interface's clock is behind the real one: the time the app stood still.
46    behind: f64,
47    /// The interface's clock when the last step ended.
48    last_step: Option<f64>,
49}
50
51impl std::fmt::Debug for Headed {
52    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
53        f.debug_struct("Headed")
54            .field("app", &self.hello.app)
55            .field("released", &self.released)
56            .finish_non_exhaustive()
57    }
58}

60impl Headed {
61    /// Answers over this process's standard input and output, starting with `hello` (which is
62    /// sent with `window: true`) once the window is on screen. Nothing else may write to standard
63    /// output meanwhile.
64    pub fn new(hello: Hello) -> Self {
65        Self::over(hello, BufReader::new(io::stdin()), io::stdout())
66    }
67
68    /// [`Self::new`] over any input and output, for tests and other transports.
69    pub fn over(hello: Hello, input: impl BufRead + Send + 'static, output: impl Write + 'static) -> Self {
70        Self {
71            hello: Hello { window: true, ..hello },
72            session: E2e::for_window(),
73            filming: None,
74            input: Some(Box::new(input)),
75            output: Box::new(output),
76            lines: None,
77            released: false,
78            behind: 0.0,
79            last_step: None,
80        }
81    }
82
83    /// Also films the window's interface when the tool asks, as `serve_filmed` does.
84    pub fn filmed(mut self, filming: impl Filming + 'static) -> Self {
85        self.filming = Some(Box::new(filming));
86        self
87    }
88}

90impl UiDriver for Headed {
91    fn start(&mut self, wake: Box<dyn Fn() + Send + Sync>) {
92        let (send, lines) = channel();
93        if let Some(input) = self.input.take() {
94            std::thread::spawn(move || {
95                for line in input.lines() {
96                    let Ok(line) = line else { break };
97                    if send.send(Line::Text(line)).is_err() {
98                        return;
99                    }
100                    wake();
101                }
102                let _ = send.send(Line::Closed);
103                wake();
104            });
105        }
106        self.lines = Some(lines);
107        if write_line(&mut self.output, &self.hello).is_err() {
108            // Nobody to answer: the tool is gone, and the first turn ends the app.
109            self.lines = None;
110        }
111    }
112
113    fn time(&mut self, now: f64) -> f64 {
114        let clock = now - self.behind;
115        match self.last_step {
116            Some(ended) => clock.min(ended + SETTLE),
117            None => clock,
118        }
119    }
120
121    fn drive(&mut self, ui: &mut Ui, now: f64) -> Driving {
122        // `now` is already the interface's clock; the real one goes on while steps run.
123        let real = now + self.behind;
124        let started = Instant::now();
125        let Some(lines) = &self.lines else {
126            return if self.released { Driving::Released } else { Driving::Done };
127        };
128        loop {
129            let line = match lines.try_recv() {
130                Ok(Line::Text(line)) => line,
131                Err(TryRecvError::Empty) => break,
132                // The tool let go of the window and closed its end: the window stays.
133                Ok(Line::Closed) | Err(TryRecvError::Disconnected) if self.released => break,
134                // The tool is gone without saying bye: nobody will drive the window again.
135                Ok(Line::Closed) | Err(TryRecvError::Disconnected) => return Driving::Done,
136            };
137            if self.released {
138                continue;
139            }
140            let real_now = real + started.elapsed().as_secs_f64();
141            // The time the app stood still since the last step stays behind it.
142            if let Some(ended) = self.last_step {
143                self.behind = self.behind.max(real_now - (ended + SETTLE));
144            }
145            let Self { session, filming, .. } = self;
146            let answered = session.in_window(ui, real_now - self.behind, |session| {
147                answer(&line, session, filming.as_deref_mut())
148            });
149            // A step that waited moved the session's clock with real time; the interface's goes on
150            // from there.
151            let after = real + started.elapsed().as_secs_f64();
152            self.behind = after - self.session.clock();
153            self.last_step = Some(self.session.clock());
154            let Some((reply, end)) = answered else { continue };
155            if write_line(&mut self.output, &reply).is_err() {
156                return Driving::Done;
157            }
158            match end {
159                Some(End::Bye) => return Driving::Done,
160                Some(End::Release) => self.released = true,
161                None => {}
162            }
163        }
164        if self.released { Driving::Released } else { Driving::Driven }
165    }
166
167    fn window_closed(&mut self) {
168        let failure = Failure::of(codes::WINDOW_CLOSED, "the window was closed while the test drove it");
169        let _ = write_line(&mut self.output, &Reply::of(Err(failure)));
170    }
171}
