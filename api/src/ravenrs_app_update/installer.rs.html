10/// What the installer writes, in the app's home.
11pub(crate) const LOG: &str = "atualizacao.log";
12/// The time an installer began, while it runs.
13pub(crate) const LOCK: &str = "atualizacao.trava";
14/// The [`Memory`] of the last update.
15pub(crate) const MEMORY: &str = "atualizacao.json";
16/// The installer as downloaded, which is what runs.
17const DOWNLOADED: &str = "install.cmd";

19/// A lock older than this belongs to an installer that died without removing it.
20pub const LOCK_EXPIRES: Duration = Duration::from_secs(3 * 3600);
21
22/// A running installer writes to its log at least this often; a lock whose log has
23/// been quiet for longer belongs to a window that closed in the middle.
24pub const LOG_QUIET_FOR: Duration = Duration::from_secs(20 * 60);
25
26/// How often the installer's log is read for the step it is on.
27const STAGE_EVERY: Duration = Duration::from_secs(1);

29/// Remembers an update that failed.
30#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
31pub struct Memory {
32    /// The version that failed last.
33    pub failed: Option<String>,
34    /// When, in seconds since 1970.
35    pub when: Option<u64>,
36}

38/// The step the installer announced on a line of its log: `> text`.
39/// What it adds in parentheses is for a terminal: the status bar's sentence has
40/// no room for it, and its end is what must not get cut.
41pub fn installer_stage(line: &str) -> Option<String> {
42    let text = line.trim().strip_prefix("> ")?.trim();
43    let text = text.split(" (").next().unwrap_or(text).trim();
44    (!text.is_empty()).then(|| text.to_owned())
45}
46
47/// The last step announced in the log so far.
48pub(crate) fn last_stage(log: &str) -> Option<String> {
49    log.lines().rev().find_map(installer_stage)
50}

52/// The downloaded file really is the whole installer. An error page, an empty
53/// file or one cut at any point does not pass, because the end of the file has to be there.
54///
55/// Copies already installed check with these same marks, so the template keeps them.
56pub fn installer_is_whole(text: &str) -> bool {
57    let end = text.trim_end();
58    let last_line = end.strip_suffix('}').map(str::trim_end).and_then(|before| before.lines().last());
59    text.starts_with(":<<\"::FIM-DO-CMD\"")
60        && text.contains("\n#==POWERSHELL==")
61        && text.contains("\n#==FIM-POWERSHELL==")
62        && last_line.is_some_and(|line| line.contains("Fim do instalador"))
63}

65/// How to run the installer here: at low priority on Linux and macOS, so
66/// compiling does not compete with whoever is using the machine.
67pub fn installer_command(file: &Path) -> (String, Vec<String>) {
68    let path = file.display().to_string();
69    if cfg!(windows) {
70        ("cmd".to_owned(), vec!["/c".to_owned(), path])
71    } else {
72        ("nice".to_owned(), vec!["-n".to_owned(), "10".to_owned(), "sh".to_owned(), path])
73    }
74}

76/// An installer is running. The lock holds the time it began, and the installer's
77/// log, beside it, must still be getting written: the window that took the lock may
78/// have closed before it could let go.
79pub fn is_locked(lock: &Path, now: u64) -> bool {
80    let Some(began) = std::fs::read_to_string(lock).ok().and_then(|text| text.trim().parse::<u64>().ok())
81    else {
82        return false;
83    };
84    let written = std::fs::metadata(lock.with_file_name(LOG))
85        .and_then(|meta| meta.modified())
86        .ok()
87        .and_then(|when| when.duration_since(std::time::UNIX_EPOCH).ok())
88        .map(|since| since.as_secs());
89    lock_is_live(began, written, now)
90}
91
92/// The rule behind [`is_locked`]: begun less than [`LOCK_EXPIRES`] ago, and its log
93/// written in the last [`LOG_QUIET_FOR`]. A log older than the lock is the last run's.
94pub fn lock_is_live(began: u64, log_written: Option<u64>, now: u64) -> bool {
95    let last_sign = log_written.filter(|&written| written >= began).unwrap_or(began);
96    now.saturating_sub(began) < LOCK_EXPIRES.as_secs()
97        && now.saturating_sub(last_sign) < LOG_QUIET_FOR.as_secs()
98}

100/// Seconds since 1970, by the machine's clock.
101pub(crate) fn seconds_now() -> u64 {
102    std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map_or(0, |d| d.as_secs())
103}

105/// What is remembered about the last update; nothing, when it is missing or unreadable.
106pub fn read_memory(file: &Path) -> Memory {
107    std::fs::read(file).ok().and_then(|bytes| serde_json::from_slice(&bytes).ok()).unwrap_or_default()
108}
109
110fn write_memory(file: &Path, memory: &Memory) {
111    if let Ok(text) = serde_json::to_vec_pretty(memory)
112        && let Err(e) = std::fs::write(file, text)
113    {
114        tracing::warn!(error = %e, "could not remember the failed update");
115    }
116}

118pub(crate) fn download(url: &str) -> Result<String, String> {
119    let config = ureq::Agent::config_builder().timeout_global(Some(Duration::from_secs(20))).build();
120    let agent: ureq::Agent = config.into();
121    let mut response = agent.get(url).call().map_err(|e| e.to_string())?;
122    response.body_mut().read_to_string().map_err(|e| e.to_string())
123}

125/// Where one run of the installer happens. The app's comes from its [`InstalledApp`];
126/// tests give a folder of their own and an installer that does not compile.
127pub(crate) struct Setup<'a> {
128    /// Where the lock, the memory, the log and the downloaded installer go.
129    pub home: PathBuf,
130    /// The installed app, which has to answer `--versao` with the new version.
131    pub binary: Option<PathBuf>,
132    /// The variable that keeps the installer from waiting for a key.
133    pub no_pause_env: String,
134    /// Gets the installer's text.
135    pub fetch: Box<dyn Fn() -> Result<String, String> + 'a>,
136}
137
138impl Setup<'static> {
139    /// The app's own: its home, its installed binary, and the installer from its address.
140    pub(crate) fn of(app: &InstalledApp) -> Self {
141        let url = app.installer_url;
142        Setup {
143            home: app.home(),
144            binary: app.installed_binary(),
145            no_pause_env: app.no_pause_env(),
146            fetch: Box::new(move || download(url)),
147        }
148    }
149}

151/// Runs the installer once, unless another window already is: then follows that one.
152/// A failure is remembered, so that version is not tried again by itself for a day.
153pub(crate) fn run_installer(
154    setup: &Setup<'_>,
155    version: &str,
156    on_stage: &dyn Fn(String),
157) -> Result<(), String> {
158    let home = &setup.home;
159    std::fs::create_dir_all(home).map_err(|e| e.to_string())?;
160    let (lock, memory) = (home.join(LOCK), home.join(MEMORY));
161    let now = seconds_now();
162    if is_locked(&lock, now) {
163        return follow_other_installer(&lock, setup.binary.as_deref(), version, on_stage);
164    }
165    std::fs::write(&lock, now.to_string()).map_err(|e| e.to_string())?;
166    let result = compile_and_check(setup, version, on_stage);
167    // Without this the next start would take it for a running installer until the lock expires.
168    let _ = std::fs::remove_file(&lock);
169    let remembered = match &result {
170        Ok(()) => Memory::default(),
171        Err(_) => Memory { failed: Some(version.to_owned()), when: Some(seconds_now()) },
172    };
173    write_memory(&memory, &remembered);
174    result
175}

177/// Another window is installing: follows its steps until it lets go of the lock,
178/// then asks the installed app which version it is.
179fn follow_other_installer(
180    lock: &Path,
181    binary: Option<&Path>,
182    version: &str,
183    on_stage: &dyn Fn(String),
184) -> Result<(), String> {
185    let log = lock.with_file_name(LOG);
186    let mut told = None;
187    while is_locked(lock, seconds_now()) {
188        let stage = std::fs::read(&log).ok().and_then(|text| last_stage(&String::from_utf8_lossy(&text)));
189        let stage = format!("outra janela está atualizando — {}", stage.as_deref().unwrap_or("começando"));
190        if told.as_ref() != Some(&stage) {
191            told = Some(stage.clone());
192            on_stage(stage);
193        }
194        std::thread::sleep(STAGE_EVERY);
195    }
196    installed_answers(binary, version)
197}

199/// The installed app opens and says it is `version`.
200pub(crate) fn installed_answers(binary: Option<&Path>, version: &str) -> Result<(), String> {
201    let binary = binary.ok_or("não sei onde o app fica neste sistema.")?;
202    let output = std::process::Command::new(binary)
203        .arg("--versao")
204        .output()
205        .map_err(|e| format!("o app novo não abriu: {e}"))?;
206    let answered = String::from_utf8_lossy(&output.stdout);
207    if answered.trim() == version {
208        Ok(())
209    } else {
210        Err(format!("o instalado responde {}, e não {version}.", answered.trim()))
211    }
212}

214/// Runs the installer, telling each step it announces in its log as it gets there.
215fn follow_installer(
216    mut command: std::process::Command,
217    log: &Path,
218    on_stage: &dyn Fn(String),
219) -> Result<std::process::ExitStatus, String> {
220    let mut child = command.spawn().map_err(|e| format!("não consegui rodar o instalador: {e}"))?;
221    let mut told = None;
222    loop {
223        // Not `read_to_string`: a line that is not UTF-8 must not hide the steps.
224        let stage = std::fs::read(log).ok().and_then(|text| last_stage(&String::from_utf8_lossy(&text)));
225        if stage.is_some() && stage != told {
226            told.clone_from(&stage);
227            on_stage(stage.unwrap_or_default());
228        }
229        if let Some(status) = child.try_wait().map_err(|e| e.to_string())? {
230            return Ok(status);
231        }
232        std::thread::sleep(STAGE_EVERY);
233    }
234}

236fn compile_and_check(setup: &Setup<'_>, version: &str, on_stage: &dyn Fn(String)) -> Result<(), String> {
237    let text = (setup.fetch)().map_err(|e| format!("não consegui baixar o instalador: {e}"))?;
238    if !installer_is_whole(&text) {
239        return Err("o instalador baixado veio incompleto; tente de novo mais tarde.".to_owned());
240    }
241    let file: PathBuf = setup.home.join(DOWNLOADED);
242    std::fs::write(&file, text).map_err(|e| e.to_string())?;
243    let (program, arguments) = installer_command(&file);
244    let log_path = setup.home.join(LOG);
245    let log = std::fs::File::create(&log_path).map_err(|e| e.to_string())?;
246    let errors = log.try_clone().map_err(|e| e.to_string())?;
247    let mut command = std::process::Command::new(program);
248    command.args(arguments).env(&setup.no_pause_env, "1").stdout(log).stderr(errors);
249    let status = follow_installer(command, &log_path, on_stage)?;
250    if !status.success() {
251        return Err(format!("a compilação falhou; veja {}.", log_path.display()));
252    }
253    installed_answers(setup.binary.as_deref(), version)
254}
