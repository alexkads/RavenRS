5/// What a version brings, written for the people using the app.
6#[derive(Clone, Debug, PartialEq, Eq, Deserialize)]
7pub struct News {
8    /// The version.
9    pub versao: String,
10    /// The release day; only the history has it.
11    #[serde(default)]
12    pub data: Option<String>,
13    /// One line: what people will notice first.
14    pub titulo: String,
15    /// An update that should not wait.
16    #[serde(default)]
17    pub importante: bool,
18    /// What changed.
19    pub novidades: Vec<String>,
20    /// Why it is worth updating now.
21    pub por_que_atualizar: String,
22    /// The GitHub issues this version closes, by number. Older copies ignore it.
23    #[serde(default)]
24    pub issues: Vec<u32>,
25    /// Who made this version, by name. Older copies ignore it.
26    #[serde(default)]
27    pub contribuidores: Vec<String>,
28}

30/// Reads `novidades.json`. A broken one is no news: better silent than a blank announcement.
31pub fn parse_news(text: &str) -> Option<News> {
32    let news: News = serde_json::from_str(text).ok()?;
33    (version_numbers(&news.versao).is_some() && !news.titulo.trim().is_empty()).then_some(news)
34}
35
36/// Reads a history file, skipping an entry it cannot understand.
37pub fn parse_history(text: &str) -> Vec<News> {
38    let items: Vec<serde_json::Value> = serde_json::from_str(text).unwrap_or_default();
39    items.into_iter().filter_map(|item| serde_json::from_value(item).ok()).collect()
40}

42/// One list of versions, newest first, out of several. When a version is in
43/// more than one, the first list's account of it is the one that stays, with
44/// the date of whichever list has one.
45pub fn join_history(lists: impl IntoIterator<Item = Vec<News>>) -> Vec<News> {
46    let mut all: Vec<News> = Vec::new();
47    for news in lists.into_iter().flatten() {
48        match all.iter_mut().find(|known| known.versao == news.versao) {
49            // The announcement has no date; the history does.
50            Some(known) => known.data = known.data.take().or(news.data),
51            None => all.push(news),
52        }
53    }
54    all.sort_by_key(|news| std::cmp::Reverse(version_numbers(&news.versao)));
55    all
56}

58pub(crate) fn version_numbers(version: &str) -> Option<Vec<u64>> {
59    version.trim().split('.').map(|part| part.parse().ok()).collect()
60}
61
62/// `candidate` is a later version than `current`, comparing numbers (0.1.10 is after 0.1.9).
63pub fn is_newer(candidate: &str, current: &str) -> bool {
64    match (version_numbers(candidate), version_numbers(current)) {
65        (Some(a), Some(b)) => a > b,
66        _ => false,
67    }
68}

70/// Where an update stands, for the status bar and the settings page.
71#[derive(Clone, Debug, PartialEq, Eq, Default)]
72pub enum UpdateStatus {
73    /// Nothing to say yet, or up to date.
74    #[default]
75    Quiet,
76    /// A new version exists and is waiting for a click.
77    Available(News),
78    /// Compiling it now.
79    Compiling(News),
80    /// It is installed; reopening uses it.
81    Ready(News),
82    /// It failed, and the app that is open keeps working.
83    Failed(News, String),
84    /// A new version exists, but this copy updates with `git pull`.
85    Development(News),
86    /// Someone asked, and the answer has not come yet.
87    Checking,
88    /// Someone asked, and there is nothing newer.
89    UpToDate,
90    /// Someone asked, and the network did not answer.
91    NoAnswer(String),
92}
93
94impl UpdateStatus {
95    /// The version it is about, if any.
96    pub fn news(&self) -> Option<&News> {
97        match self {
98            Self::Available(news)
99            | Self::Compiling(news)
100            | Self::Ready(news)
101            | Self::Failed(news, _)
102            | Self::Development(news) => Some(news),
103            Self::Quiet | Self::Checking | Self::UpToDate | Self::NoAnswer(_) => None,
104        }
105    }
106}

108/// What a notice in the status bar offers.
109#[derive(Clone, Copy, Debug, PartialEq, Eq)]
110pub enum NoticeAction {
111    /// Opens the news on the announced version.
112    News,
113    /// Installs the announced version.
114    Update,
115    /// Installs it again, after a failure.
116    Retry,
117    /// The news again, where the steps for doing it by hand are.
118    HowToUpdate,
119    /// Why it failed, the step it stopped at and the end of the installer's log.
120    Details,
121    /// Opens the installed version.
122    Reopen,
123    /// Asks the network again.
124    CheckAgain,
125    /// Hides the notice for this session.
126    Later,
127    /// Hides a notice that has been read.
128    Close,
129}
130
131impl NoticeAction {
132    /// The button's words.
133    pub fn label(self) -> &'static str {
134        match self {
135            Self::News => "Ver novidades",
136            Self::Update => "Atualizar",
137            Self::Retry | Self::CheckAgain => "Tentar de novo",
138            Self::HowToUpdate => "Como atualizar",
139            Self::Details => "Ver detalhes",
140            Self::Reopen => "Reabrir agora",
141            Self::Later => "Depois",
142            Self::Close => "Fechar",
143        }
144    }
145
146    /// The button that moves things forward, drawn as the main one.
147    pub fn is_primary(self) -> bool {
148        matches!(self, Self::Update | Self::Reopen | Self::Retry | Self::CheckAgain)
149    }
150}

152/// What the status bar says about updates, in one line.
153#[derive(Clone, Debug, PartialEq, Eq)]
154pub struct Notice {
155    /// The sentence.
156    pub text: String,
157    /// Something is under way.
158    pub busy: bool,
159    /// Tint this one: an important version that is waiting for a click.
160    pub important: bool,
161    /// Its buttons, in order.
162    pub actions: Vec<NoticeAction>,
163}
164
165/// The whole rule of what the status bar says, with nothing on screen, as
166/// VintageLightbox words it. `dismissed` is the version hidden for this session;
167/// what is under way and what waits for reopening cannot be hidden.
168pub fn notice(
169    status: &UpdateStatus,
170    current: &str,
171    stage: Option<&str>,
172    dismissed: Option<&str>,
173) -> Option<Notice> {
174    let said = |text: String, busy, important, actions: &[NoticeAction]| {
175        Some(Notice { text, busy, important, actions: actions.to_vec() })
176    };
177    use NoticeAction::{CheckAgain, Close, Details, Later, News, Reopen, Retry, Update};
178    match status {
179        UpdateStatus::Quiet => None,
180        UpdateStatus::Checking => said("Procurando versão nova…".to_owned(), true, false, &[]),
181        UpdateStatus::UpToDate => {
182            said(format!("Você está na versão mais recente ({current})."), false, false, &[Close])
183        }
184        UpdateStatus::NoAnswer(reason) => said(
185            format!("Não consegui verificar se há versão nova: {reason}"),
186            false,
187            false,
188            &[CheckAgain, Close],
189        ),
190        UpdateStatus::Available(news) | UpdateStatus::Development(news) | UpdateStatus::Failed(news, _)
191            if dismissed == Some(news.versao.as_str()) =>
192        {
193            None
194        }
195        UpdateStatus::Available(news) => {
196            let opening = if news.importante {
197                format!("Atualização importante: versão {}", news.versao)
198            } else {
199                format!("Versão {} disponível", news.versao)
200            };
201            said(
202                format!("{opening} — {}", news.titulo.trim()),
203                false,
204                news.importante,
205                &[News, Update, Later],
206            )
207        }
208        UpdateStatus::Development(news) => said(
209            format!(
210                "Versão {} no repositório. Esta cópia foi compilada à mão: atualize com git pull.",
211                news.versao
212            ),
213            false,
214            false,
215            &[News, Later],
216        ),
217        UpdateStatus::Compiling(news) => said(
218            format!(
219                "Atualizando para a versão {} em segundo plano — {}. O app continua funcionando.",
220                news.versao,
221                stage.unwrap_or("começando"),
222            ),
223            true,
224            false,
225            &[News],
226        ),
227        UpdateStatus::Ready(news) => said(
228            format!(
229                "Versão {} instalada. Ela entra na próxima vez que o app abrir — ou reabra agora.",
230                news.versao
231            ),
232            false,
233            news.importante,
234            &[Reopen],
235        ),
236        // The reason goes in the details: here it would be cut, and the end is what matters.
237        UpdateStatus::Failed(news, _) => said(
238            format!(
239                "A atualização para a versão {} não deu certo; a {current} continua funcionando.",
240                news.versao
241            ),
242            false,
243            false,
244            &[Retry, Details, Close],
245        ),
246    }
