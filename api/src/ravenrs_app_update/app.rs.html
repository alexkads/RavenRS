7/// An app that installs and updates by compiling on the user's machine.
8///
9/// Everything about one app is here, and nothing about it is anywhere else in
10/// this crate: the installer, the folders and the addresses all come from these
11/// fields. An app keeps one in a `static`:
12///
13/// ```
14/// use ravenrs_app_update::InstalledApp;
15///
16/// pub static APP: InstalledApp = InstalledApp {
17///     name: "Raven Exemplo",
18///     package: "raven-exemplo",
19///     home_dir: ".raven-exemplo",
20///     bundle_id: "dev.ravenrs.raven-exemplo",
21///     icons: "examples/raven-exemplo/assets",
22///     mac_category: "public.app-category.productivity",
23///     linux_categories: "Office;",
24///     source: "https://github.com/alexkads/RavenRS/archive/refs/heads/master.tar.gz",
25///     env_prefix: "RE",
26///     installed_env: "RAVEN_EXEMPLO_INSTALLED",
27///     news_urls: &["https://raw.githubusercontent.com/alexkads/RavenRS/master/examples/raven-exemplo/novidades.json"],
28///     history_urls: &[
29///         "https://raw.githubusercontent.com/alexkads/RavenRS/master/examples/raven-exemplo/historico-de-novidades.json",
30///     ],
31///     installer_url: "https://raw.githubusercontent.com/alexkads/RavenRS/master/examples/raven-exemplo/install.cmd",
32///     issues_url: "https://github.com/alexkads/RavenRS/issues",
33///     version: env!("CARGO_PKG_VERSION"),
34///     installed: option_env!("RAVEN_EXEMPLO_INSTALLED").is_some(),
35///     history: "[]",
36/// };
37///
38/// assert!(APP.home().ends_with(".raven-exemplo"));
39/// assert!(APP.render_installer().is_ok());
40/// ```
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub struct InstalledApp {
43    /// The name people see: in the menus, the shortcut and the macOS bundle.
44    /// Accents are fine; the Windows parts use [`Self::ascii_name`].
45    pub name: &'static str,
46    /// The Cargo package, which is also the binary's name and, on Linux, the
47    /// `.desktop` file, the icon and the window's id: the window has to be opened with this id.
48    pub package: &'static str,
49    /// The folder in the user's home where the installer keeps the code, the
50    /// build and its notes, and where the app may keep its own files. It starts with a dot.
51    pub home_dir: &'static str,
52    /// The macOS bundle identifier.
53    pub bundle_id: &'static str,
54    /// The folder, inside the repository, with `icon.ico`, `icon.icns` and `icon.png`.
55    pub icons: &'static str,
56    /// The macOS category (`LSApplicationCategoryType`).
57    pub mac_category: &'static str,
58    /// The Linux menu categories, each one ending in `;`.
59    pub linux_categories: &'static str,
60    /// The `.tar.gz` of the code the installer compiles.
61    pub source: &'static str,
62    /// What the installer's options start with: `RF` gives `RF_CASA`, `RF_DESTINO` and the rest.
63    pub env_prefix: &'static str,
64    /// The variable the installer sets while it compiles, which [`Self::installed`] reads.
65    pub installed_env: &'static str,
66    /// Where `novidades.json` is read, in order. An installed copy only knows the
67    /// addresses it was compiled with, so addresses are only ever added.
68    pub news_urls: &'static [&'static str],
69    /// Where `historico-de-novidades.json` is read, in order, for skipping more than one version.
70    pub history_urls: &'static [&'static str],
71    /// The installer the app downloads and runs, the same one its README shows.
72    pub installer_url: &'static str,
73    /// Where the issues are read, without the number.
74    pub issues_url: &'static str,
75    /// The running version: `env!("CARGO_PKG_VERSION")`.
76    pub version: &'static str,
77    /// The installer compiled this copy: `option_env!(<installed_env>).is_some()`.
78    pub installed: bool,
79    /// The history compiled into the app: `include_str!("../historico-de-novidades.json")`.
80    pub history: &'static str,
81}

83/// How this copy of the app was installed.
84#[derive(Clone, Copy, Debug, PartialEq, Eq)]
85pub enum InstallKind {
86    /// By the installer, which is the only thing that can update it.
87    Installed,
88    /// `cargo run`: whoever develops updates with `git pull`.
89    Development,
90}
91
92impl InstalledApp {
93    /// The name without accents, for the Windows shortcut and the `cmd` window,
94    /// where the console's code page would garble them.
95    pub fn ascii_name(&self) -> String {
96        self.name.chars().filter_map(ascii_letter).collect()
97    }
98
99    /// The folder under `%LOCALAPPDATA%\Programs` where Windows keeps the app: the ASCII name, without spaces.
100    pub fn windows_dir(&self) -> String {
101        self.ascii_name().split_whitespace().collect()
102    }
103
104    /// How this copy was installed.
105    pub fn install_kind(&self) -> InstallKind {
106        if self.installed { InstallKind::Installed } else { InstallKind::Development }
107    }
108
109    /// `~/<home_dir>`: what the installer leaves, and where the update keeps its notes.
110    pub fn home(&self) -> PathBuf {
111        user_home().unwrap_or_else(|| PathBuf::from(".")).join(self.home_dir)
112    }
113
114    /// Where the installer puts the app; the update checks this one and reopens it.
115    pub fn installed_binary(&self) -> Option<PathBuf> {
116        if cfg!(windows) {
117            std::env::var_os("LOCALAPPDATA").map(|base| {
118                PathBuf::from(base)
119                    .join("Programs")
120                    .join(self.windows_dir())
121                    .join(format!("{}.exe", self.package))
122            })
123        } else if cfg!(target_os = "macos") {
124            // The installer uses /Applications when it can write there.
125            let inside = format!("{}.app/Contents/MacOS/{}", self.name, self.package);
126            let system = PathBuf::from("/Applications").join(&inside);
127            if system.exists() {
128                Some(system)
129            } else {
130                user_home().map(|h| h.join("Applications").join(inside))
131            }
132        } else {
133            user_home().map(|h| h.join(".local/bin").join(self.package))
134        }
135    }
136
137    /// Where the installer keeps the version it replaced: on macOS in
138    /// `~/<home_dir>/anterior`, where it does not show as a second app; elsewhere
139    /// beside the installed one, as `.anterior`.
140    pub fn previous_version(&self) -> Option<PathBuf> {
141        if cfg!(target_os = "macos") {
142            Some(self.home().join("anterior").join(format!("{}.app", self.name)))
143        } else {
144            self.installed_binary().map(|binary| {
145                let mut name = binary.into_os_string();
146                name.push(".anterior");
147                PathBuf::from(name)
148            })
149        }
150    }
151
152    /// Running the installer by hand on this system, for when updating from inside the app fails.
153    pub fn install_command(&self) -> String {
154        self.install_command_on(cfg!(windows))
155    }
156
157    /// Running the installer by hand, in PowerShell on Windows or in a terminal elsewhere.
158    pub fn install_command_on(&self, windows: bool) -> String {
159        let url = self.installer_url;
160        if windows {
161            format!(r#"irm {url} -OutFile "$env:TEMP\install.cmd"; & "$env:TEMP\install.cmd""#)
162        } else {
163            format!("curl -fsSL {url} -o /tmp/install.cmd && sh /tmp/install.cmd")
164        }
165    }
166
167    /// Where an issue is read, by its number.
168    pub fn issue_url(&self, number: u32) -> String {
169        format!("{}/{number}", self.issues_url.trim_end_matches('/'))
170    }
171
172    /// Every version up to this one, newest first, as compiled into the app.
173    pub fn history(&self) -> Vec<News> {
174        parse_history(self.history)
175    }
176
177    /// Where the installer writes what it does.
178    pub fn installer_log(&self) -> PathBuf {
179        self.home().join(crate::installer::LOG)
180    }
181
182    /// The end of the installer's log, for the details of a failure.
183    pub fn installer_log_tail(&self, lines: usize) -> Option<String> {
184        let text = std::fs::read(self.installer_log())
185            .ok()
186            .map(|bytes| String::from_utf8_lossy(&bytes).into_owned())?;
187        let kept: Vec<&str> = text.lines().rev().take(lines).collect();
188        let tail = kept.into_iter().rev().collect::<Vec<_>>().join("\n");
189        (!tail.trim().is_empty()).then_some(tail)
190    }
191
192    /// The variable that keeps the installer from waiting for a key at the end,
193    /// which the app sets when it runs it.
194    pub(crate) fn no_pause_env(&self) -> String {
195        format!("{}_SEM_PAUSA", self.env_prefix)
196    }
197}
198
199fn user_home() -> Option<PathBuf> {
200    std::env::var_os(if cfg!(windows) { "USERPROFILE" } else { "HOME" }).map(PathBuf::from)
201}

203/// The letter without its accent; anything else that is not ASCII goes.
204fn ascii_letter(c: char) -> Option<char> {
205    if c.is_ascii() {
206        return Some(c);
207    }
208    let plain = match c {
209        'á' | 'à' | 'â' | 'ã' | 'ä' => 'a',
210        'Á' | 'À' | 'Â' | 'Ã' | 'Ä' => 'A',
211        'é' | 'è' | 'ê' | 'ë' => 'e',
212        'É' | 'È' | 'Ê' | 'Ë' => 'E',
213        'í' | 'ì' | 'î' | 'ï' => 'i',
214        'Í' | 'Ì' | 'Î' | 'Ï' => 'I',
215        'ó' | 'ò' | 'ô' | 'õ' | 'ö' => 'o',
216        'Ó' | 'Ò' | 'Ô' | 'Õ' | 'Ö' => 'O',
217        'ú' | 'ù' | 'û' | 'ü' => 'u',
218        'Ú' | 'Ù' | 'Û' | 'Ü' => 'U',
219        'ç' => 'c',
220        'Ç' => 'C',
221        'ñ' => 'n',
222        'Ñ' => 'N',
223        _ => return None,
224    };
225    Some(plain)
226}
