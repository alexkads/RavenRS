1//! The installer of every app, made from one template.
2//!
3//! Each app keeps the `install.cmd` made for it, whole, because that file is
4//! what people double-click and what installed copies download: a copy only
5//! knows the address it was compiled with, and `cmd` cannot include another
6//! file. A test in the app compares it with [`InstalledApp::render_installer`],
7//! so it never drifts from the template.
8
9use std::path::Path;
10
11use crate::InstalledApp;
12
13/// The installer with `@NAME@` marks where each app's values go.
14pub const INSTALLER_TEMPLATE: &str = include_str!("../install.cmd.in");
15
16/// Set to `1`, [`assert_installer_is_current`] writes the app's installer
17/// instead of failing when it differs.
18pub const WRITE_INSTALLER_ENV: &str = "RAVENRS_WRITE_INSTALLER";
19
20impl InstalledApp {
21    /// This app's installer: the template with its values, or the field that
22    /// cannot go into a script as it is.
23    pub fn render_installer(&self) -> Result<String, String> {
24        self.check()?;
25        let icons_windows = self.icons.replace('/', "\\");
26        let windows_dir = self.windows_dir();
27        let ascii_name = self.ascii_name();
28        let values: [(&str, &str); 13] = [
29            ("@NOME@", self.name),
30            ("@NOME_ASCII@", &ascii_name),
31            ("@PACOTE@", self.package),
32            ("@CASA@", self.home_dir),
33            ("@PASTA_WINDOWS@", &windows_dir),
34            ("@BUNDLE_ID@", self.bundle_id),
35            ("@ICONES@", self.icons),
36            ("@ICONES_WINDOWS@", &icons_windows),
37            ("@CATEGORIA_MAC@", self.mac_category),
38            ("@CATEGORIAS_LINUX@", self.linux_categories),
39            ("@FONTE@", self.source),
40            ("@PREFIXO@", self.env_prefix),
41            ("@INSTALADO@", self.installed_env),
42        ];
43        let mut text = INSTALLER_TEMPLATE.to_owned();
44        for (mark, value) in values {
45            text = text.replace(mark, value);
46        }
47        Ok(text)
48    }
49
50    /// Every field goes into `cmd`, PowerShell, `sh`, a plist and a `.desktop`
51    /// file. A quote, a `$` or a `%` would change what the installer does, so
52    /// each field takes only what it needs.
53    pub fn check(&self) -> Result<(), String> {
54        let fields = [
55            ("name", self.name, Allowed::Name),
56            ("package", self.package, Allowed::Package),
57            ("home_dir", self.home_dir, Allowed::Path),
58            ("bundle_id", self.bundle_id, Allowed::Path),
59            ("icons", self.icons, Allowed::Path),
60            ("mac_category", self.mac_category, Allowed::Path),
61            ("linux_categories", self.linux_categories, Allowed::Categories),
62            ("source", self.source, Allowed::Url),
63            ("env_prefix", self.env_prefix, Allowed::Variable),
64            ("installed_env", self.installed_env, Allowed::Variable),
65            ("installer_url", self.installer_url, Allowed::Url),
66        ];
67        for (field, value, allowed) in fields {
68            if value.is_empty() || !value.chars().all(|c| allowed.takes(c)) {
69                return Err(format!("{field} = {value:?}: {}", allowed.says()));
70            }
71        }
72        if !self.home_dir.starts_with('.') || self.home_dir.contains('/') {
73            return Err(format!("home_dir = {:?}: one hidden folder, starting with a dot", self.home_dir));
74        }
75        if !self.linux_categories.ends_with(';') {
76            return Err(format!("linux_categories = {:?}: each category ends in ;", self.linux_categories));
77        }
78        if self.windows_dir().is_empty() {
79            return Err(format!("name = {:?}: needs a letter Windows can show", self.name));
80        }
81        Ok(())
82    }
83}

85#[derive(Clone, Copy)]
86enum Allowed {
87    Name,
88    Package,
89    Path,
90    Categories,
91    Url,
92    Variable,
93}
94
95impl Allowed {
96    fn takes(self, c: char) -> bool {
97        match self {
98            Self::Name => c.is_alphanumeric() || c == ' ' || c == '-',
99            Self::Package => c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-',
100            Self::Path => c.is_ascii_alphanumeric() || matches!(c, '.' | '-' | '_' | '/'),
101            Self::Categories => c.is_ascii_alphanumeric() || c == ';',
102            Self::Url => c.is_ascii_alphanumeric() || matches!(c, ':' | '/' | '.' | '-' | '_' | '~'),
103            Self::Variable => c.is_ascii_uppercase() || c.is_ascii_digit() || c == '_',
104        }
105    }
106
107    fn says(self) -> &'static str {
108        match self {
109            Self::Name => "letters, digits, spaces and hyphens only",
110            Self::Package => "lowercase letters, digits and hyphens only, like the Cargo package",
111            Self::Path => "ASCII letters, digits, '.', '-', '_' and '/' only",
112            Self::Categories => "ASCII letters and ';' only",
113            Self::Url => "an https address without spaces, quotes, '%', '?' or '&'",
114            Self::Variable => "uppercase letters, digits and '_' only",
115        }
116    }
117}

119/// The app's committed installer is the one the template makes for it; if not,
120/// says how to write it again. With [`WRITE_INSTALLER_ENV`] set to `1`, writes it.
121pub fn check_installer(app: &InstalledApp, file: &Path) -> Result<(), String> {
122    let made = app.render_installer()?;
123    let committed = std::fs::read_to_string(file).unwrap_or_default();
124    if committed == made {
125        return Ok(());
126    }
127    if std::env::var(WRITE_INSTALLER_ENV).is_ok_and(|value| value == "1") {
128        return std::fs::write(file, made).map_err(|e| format!("{}: {e}", file.display()));
129    }
130    Err(format!(
131        "{} is not what crates/ravenrs-app-update/install.cmd.in makes for {}: change the template, \
132         never the copy, and run the tests again with {WRITE_INSTALLER_ENV}=1 to write it",
133        file.display(),
134        app.name
135    ))
136}
137
138/// [`check_installer`], for an app's tests.
139///
140/// # Panics
141///
142/// When the installer differs from what the template makes, or a field cannot go into it.
143#[track_caller]
144pub fn assert_installer_is_current(app: &InstalledApp, file: &Path) {
145    if let Err(problem) = check_installer(app, file) {
146        panic!("{problem}");
147    }
148}
