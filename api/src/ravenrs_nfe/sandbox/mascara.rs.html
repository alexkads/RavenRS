1//! Masks personal data in whatever gets written down or exported.
2//!
3//! The issuer's CNPJ stays, because it is part of the access key and identifies the case. The
4//! recipient, names, addresses, free text and the certificate do not.
5
6use crate::xml::{self, Element, Node};
7
8/// What goes in place of a certificate.
9const SEM_CERTIFICADO: &str = "[certificado removido]";
10
11/// Tags whose content is personal wherever they are, when an XML cannot be read as a tree.
12const TAGS_SENSIVEIS: [&str; 15] = [
13    "CPF",
14    "CNPJ",
15    "idEstrangeiro",
16    "xNome",
17    "xFant",
18    "xLgr",
19    "nro",
20    "xCpl",
21    "xBairro",
22    "fone",
23    "email",
24    "IE",
25    "infCpl",
26    "xContato",
27    "X509Certificate",
28];
29
30/// Masks a value and keeps enough to recognize it: `111******35`, or `***` when it is short.
31#[must_use]
32pub fn mascarar_valor(valor: &str) -> String {
33    let caracteres: Vec<char> = valor.chars().collect();
34    if caracteres.len() < 8 {
35        return "***".to_string();
36    }
37    let inicio: String = caracteres[..3].iter().collect();
38    let fim: String = caracteres[caracteres.len() - 2..].iter().collect();
39    format!("{inicio}{}{fim}", "*".repeat(caracteres.len() - 5))
40}
41
42/// Whether the content of a tag under these parents is personal.
43fn sensivel(pais: &[&str], tag: &str) -> bool {
44    let dentro = |nome: &str| pais.contains(&nome);
45    if tag == "X509Certificate" {
46        return true;
47    }
48    if dentro("dest") || dentro("retirada") || dentro("entrega") || dentro("autXML") || dentro("transporta") {
49        return true;
50    }
51    if dentro("emit") {
52        return matches!(tag, "CPF" | "xNome" | "xFant" | "xLgr" | "nro" | "xCpl" | "xBairro" | "fone");
53    }
54    if dentro("infRespTec") {
55        return matches!(tag, "xContato" | "email" | "fone");
56    }
57    matches!(tag, "infCpl" | "email")
58}

60/// Masks a value found at this path, if the path is personal. Used for each diagnosis.
61#[must_use]
62pub fn mascarar_em_caminho(caminho: &str, valor: &str) -> String {
63    let partes: Vec<&str> = caminho
64        .split('/')
65        .filter(|parte| !parte.is_empty())
66        .map(|parte| parte.split('[').next().unwrap_or(parte))
67        .collect();
68    let Some((tag, pais)) = partes.split_last() else {
69        return valor.to_string();
70    };
71    if sensivel(pais, tag) { mascarar_valor(valor) } else { valor.to_string() }
72}

74fn mascarar_arvore(element: &mut Element, pais: &mut Vec<String>) {
75    let nome = element.local_name().to_string();
76    let referencias: Vec<&str> = pais.iter().map(String::as_str).collect();
77    let trocar = sensivel(&referencias, &nome);
78    let certificado = nome == "X509Certificate";
79    pais.push(nome);
80    for node in &mut element.children {
81        match node {
82            Node::Element(filho) => mascarar_arvore(filho, pais),
83            Node::Text(texto) if trocar => {
84                *texto = if certificado { SEM_CERTIFICADO.to_string() } else { mascarar_valor(texto) };
85            }
86            _ => {}
87        }
88    }
89    pais.pop();
90}

92/// Masks text that could not be read as XML: known tags by name, then long runs of digits.
93fn mascarar_texto_bruto(texto: &str) -> String {
94    let mut saida = texto.to_string();
95    for tag in TAGS_SENSIVEIS {
96        let abre = format!("<{tag}>");
97        let fecha = format!("</{tag}>");
98        let mut inicio_busca = 0;
99        while let Some(posicao) = saida[inicio_busca..].find(&abre) {
100            let inicio = inicio_busca + posicao + abre.len();
101            let fim = saida[inicio..].find(&fecha).map_or(saida.len(), |f| inicio + f);
102            let troca = if tag == "X509Certificate" {
103                SEM_CERTIFICADO.to_string()
104            } else {
105                mascarar_valor(&saida[inicio..fim])
106            };
107            saida.replace_range(inicio..fim, &troca);
108            inicio_busca = inicio + troca.len();
109        }
110    }
111    mascarar_digitos_soltos(&saida)
112}

114/// Masks runs of exactly 11 or 14 digits, the size of a CPF or a CNPJ.
115fn mascarar_digitos_soltos(texto: &str) -> String {
116    let mut saida = String::with_capacity(texto.len());
117    let mut corrente = String::new();
118    let fechar = |corrente: &mut String, saida: &mut String| {
119        if matches!(corrente.len(), 11 | 14) {
120            saida.push_str(&mascarar_valor(corrente));
121        } else {
122            saida.push_str(corrente);
123        }
124        corrente.clear();
125    };
126    for caractere in texto.chars() {
127        if caractere.is_ascii_digit() {
128            corrente.push(caractere);
129        } else {
130            fechar(&mut corrente, &mut saida);
131            saida.push(caractere);
132        }
133    }
134    fechar(&mut corrente, &mut saida);
135    saida
136}

138/// Masks a report in plain text, where a tax id could only appear as a loose run of digits.
139#[must_use]
140pub fn mascarar_texto_de_relatorio(texto: &str) -> String {
141    mascarar_digitos_soltos(texto)
142}

144/// Masks an XML text. What cannot be read as a tree still goes through the known tags.
145#[must_use]
146pub fn mascarar_xml(texto: &str) -> String {
147    match xml::parse(texto) {
148        Ok(mut raiz) => {
149            mascarar_arvore(&mut raiz, &mut Vec::new());
150            xml::to_string(&raiz)
151        }
152        Err(_) => mascarar_texto_bruto(texto),
153    }
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    #[test]
161    fn value_keeps_only_the_ends() {
162        assert_eq!(mascarar_valor("11144477735"), "111******35");
163        assert_eq!(mascarar_valor("ANA"), "***");
164    }
165
166    #[test]
167    fn recipient_goes_and_issuer_cnpj_stays() {
168        let xml = "<NFe><emit><CNPJ>11222333000181</CNPJ><xNome>MERCADO ABC</xNome></emit><dest><CPF>11144477735</CPF><xNome>ANA SILVA</xNome><enderDest><xLgr>RUA A DA SILVA</xLgr></enderDest></dest><X509Certificate>MIIB</X509Certificate></NFe>";
169        let saida = mascarar_xml(xml);
170        assert!(saida.contains("<CNPJ>11222333000181</CNPJ>"));
171        for segredo in ["11144477735", "ANA SILVA", "MERCADO ABC", "RUA A DA SILVA", "MIIB"] {
172            assert!(!saida.contains(segredo), "{segredo}");
173        }
174        assert!(saida.contains(SEM_CERTIFICADO));
175    }
176
177    #[test]
178    fn broken_xml_is_still_masked() {
179        let xml = "<dest><CPF>11144477735</CPF><xNome>ANA SILVA";
180        let saida = mascarar_xml(xml);
181        assert!(!saida.contains("11144477735"));
182        assert!(!saida.contains("ANA SILVA"));
183        assert_eq!(mascarar_xml("cpf 11144477735 aqui"), "cpf 111******35 aqui");
184    }
185
186    #[test]
187    fn access_key_is_not_a_loose_tax_id() {
188        let chave = "35261111222333000181650010000000011123456786";
189        assert_eq!(mascarar_digitos_soltos(chave), chave);
190    }
191
192    #[test]
193    fn path_rule_matches_the_tree_rule() {
194        assert_eq!(mascarar_em_caminho("/NFe/infNFe/dest/CPF", "11144477735"), "111******35");
195        assert_eq!(mascarar_em_caminho("/NFe/infNFe/emit/CNPJ", "11222333000181"), "11222333000181");
196        assert_eq!(mascarar_em_caminho("/NFe/infNFe/det[1]/prod/vProd", "1.00"), "1.00");
197    }
