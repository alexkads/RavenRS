17/// The prazo of a cancellation, by rule 2P12-14: 24 hours.
18const PRAZO_CANCELAMENTO: i64 = 24 * 3600;
19/// How long to wait after nothing was found, by NT 2014.002, item 3.11.4.
20const ESPERA_DISTRIBUICAO: i64 = 3600;
21/// The tolerance of rule 2P13-14, in seconds.
22const TOLERANCIA_DATA: i64 = 300;
23/// The largest batch of the distribution, by NT 2014.002.
24const MAXIMO_LOTE: usize = 50;

26/// What the sandbox knows about a key.
27#[derive(Clone, Debug, PartialEq, Eq)]
28pub enum EstadoChave {
29    /// The sandbox has never seen it.
30    Desconhecida,
31    /// It was refused and therefore does not exist for the SEFAZ.
32    Rejeitada {
33        /// The last refusal.
34        c_stat: u32,
35    },
36    /// Authorized.
37    Autorizada {
38        /// The protocol.
39        n_prot: String,
40    },
41    /// Authorized, but the sender never got the answer: the dangerous case.
42    AutorizadaSemRetorno {
43        /// The protocol that exists and the sender does not know.
44        n_prot: String,
45    },
46    /// Cancelled.
47    Cancelada {
48        /// The protocol of the authorization.
49        n_prot: String,
50    },
51}

83/// What the sandbox remembers.
84#[derive(Debug, Default)]
85pub(crate) struct Notas {
86    por_chave: HashMap<String, Nota>,
87    /// Issuer, model, series and number to the key that took them.
88    por_numero: HashMap<(String, String, String, String), String>,
89    rejeitadas: HashMap<String, u32>,
90    sem_retorno: HashSet<String>,
91    recebidas: HashMap<String, Recebida>,
92    entradas: Vec<Entrada>,
93    /// For each recipient: the NSU handed over last, and when nothing new came.
94    consultas: HashMap<String, Consulta>,
95}

103impl Notas {
104    pub(super) fn estado(&self, chave: &str) -> EstadoChave {
105        if let Some(nota) = self.por_chave.get(chave) {
106            let n_prot = nota.n_prot.clone();
107            if nota.cancelada {
108                return EstadoChave::Cancelada { n_prot };
109            }
110            if self.sem_retorno.contains(chave) {
111                return EstadoChave::AutorizadaSemRetorno { n_prot };
112            }
113            return EstadoChave::Autorizada { n_prot };
114        }
115        match self.rejeitadas.get(chave) {
116            Some(c_stat) => EstadoChave::Rejeitada { c_stat: *c_stat },
117            None => EstadoChave::Desconhecida,
118        }
119    }
120
121    /// Records that the sender never got the answer about this key.
122    pub(super) fn marcar_sem_retorno(&mut self, chave: &str) {
123        self.sem_retorno.insert(chave.to_string());
124    }
125
126    /// Records that the sender saw an answer about this key.
127    pub(super) fn marcar_com_retorno(&mut self, chave: &str) {
128        self.sem_retorno.remove(chave);
129    }
130
131    pub(super) fn registrar_rejeicao(&mut self, chave: &str, c_stat: u32) {
132        if !self.por_chave.contains_key(chave) {
133            self.rejeitadas.insert(chave.to_string(), c_stat);
134        }
135    }
136
137    fn proximo_nsu(&self) -> u64 {
138        self.entradas.iter().map(|e| e.nsu).max().unwrap_or(0) + 1
139    }
140
141    /// Puts a document in the distribution of its recipient, as a summary.
142    pub(super) fn colocar(&mut self, chave: String, destinatario: String, proc_nfe: String, resumo: String) {
143        let nsu = self.proximo_nsu();
144        self.entradas.push(Entrada {
145            nsu,
146            destinatario: destinatario.clone(),
147            schema: "resNFe_v1.01.xsd".to_string(),
148            xml: resumo.clone(),
149        });
150        self.recebidas.insert(
151            chave,
152            Recebida { destinatario, proc_nfe, resumo, manifestacoes: HashSet::new(), liberada: false },
153        );
154    }
155}

157fn namespace_proprio(element: &Element) -> Option<&str> {
158    match element.name.split_once(':') {
159        Some((prefixo, _)) => element.attr(&format!("xmlns:{prefixo}")),
160        None => element.attr("xmlns"),
161    }

164/// Checks the envelope and gives back the message inside it.
165fn abrir_envelope(pedido: &SandboxRequest, servico: Servico) -> Result<Element, Resposta> {
166    let tipo = pedido.http.header("Content-Type").unwrap_or("");
167    if !tipo.to_ascii_lowercase().starts_with("application/soap+xml") {
168        return Err(Resposta::Http(resposta::fault(415, "Content-Type não é SOAP 1.2")));
169    }
170    let envelope = xml::parse(pedido.body()).map_err(|_| pedido.responder(243))?;
171    if envelope.local_name() != "Envelope" || namespace_proprio(&envelope) != Some(NS_SOAP12) {
172        return Err(Resposta::Http(resposta::fault(400, "Envelope não é SOAP 1.2")));
173    }
174    let corpo =
175        envelope.child("Body").ok_or_else(|| Resposta::Http(resposta::fault(400, "Envelope sem Body")))?;
176    let mensagem = corpo
177        .find("nfeDadosMsg")
178        .ok_or_else(|| Resposta::Http(resposta::fault(400, "Body sem nfeDadosMsg")))?;
179    let namespace = if servico == Servico::Distribuicao {
180        corpo.child(servico.operacao()).and_then(|grupo| grupo.attr("xmlns"))
181    } else {
182        mensagem.attr("xmlns")
183    };
184    if namespace != Some(servico.namespace().as_str()) {
185        return Err(Resposta::Http(resposta::fault(400, "nfeDadosMsg com namespace diferente da ação")));
186    }
187    let mut filhos = mensagem.elements();
188    let dados = filhos.next().ok_or_else(|| pedido.responder(225))?;
189    if filhos.next().is_some() {
190        return Err(pedido.responder(225));
191    }
192    Ok(dados.clone())
193}

195/// Answers the first refusal of an analysis, if there is one, as a server would.

201fn responder_achado(pedido: &SandboxRequest, achado: &Diagnostico) -> Resposta {
202    let mut texto = achado.titulo.clone();
203    if achado.regra == "B24-10" || achado.regra == "J01" {
204        texto.push_str(". O sandbox só aceita tpAmb=2");
205    }
206    if matches!(
207        achado.regra,
208        "E01"
209            | "E02"
210            | "E03"
211            | "E04"
212            | "E06"
213            | "A01"
214            | "F03"
215            | "SBX-FORA-DO-LABORATORIO"
216            | "CERT-CHAVE-FRACA"
217            | "CERT-SHA1"
218    ) {
219        texto.push_str(": ");
220        texto.push_str(achado.explicacao.split(". ").next().unwrap_or(""));
221    }
222    if achado.c_stat == 0 {
223        return Resposta::Http(resposta::fault(400, &achado.explicacao));
224    }
225    pedido.responder_com(achado.c_stat, &texto)
226}

228fn por_regra(pedido: &SandboxRequest, id: &str, detalhe: Option<&str>) -> Resposta {
229    let Some(regra) = regra(id) else {
230        return pedido.responder(999);
231    };
232    match detalhe {
233        Some(detalhe) => pedido.responder_com(regra.c_stat, &format!("{} {detalhe}", regra.titulo())),
234        None => pedido.responder_com(regra.c_stat, regra.titulo()),
235    }

238/// Runs the checks of the service of this request.
239pub(super) fn conferir(pedido: &SandboxRequest, sandbox: &SefazSandbox) -> Resposta {
240    let Some(servico) = pedido.servico else {
241        return Resposta::Proximo;
242    };
243    let dados = match abrir_envelope(pedido, servico) {
244        Ok(dados) => dados,
245        Err(resposta) => return resposta,
246    };
247    if dados.attr("xmlns") != Some(NS_NFE) {
248        return por_regra(pedido, "D01-LOTE", None);
249    }
250    let opcoes = sandbox.opcoes();
251    match servico {
252        Servico::Status => match responder_rejeicao(pedido, &analise::analisar_status(&dados, &opcoes)) {
253            Some((resposta, _)) => resposta,
254            None => pedido.responder(107),
255        },
256        Servico::Autorizacao => conferir_autorizacao(pedido, sandbox, &dados, &opcoes),
257        Servico::Consulta => conferir_consulta(pedido, sandbox, &dados, &opcoes),
258        Servico::Evento => conferir_lote_evento(pedido, sandbox, &dados, &opcoes),
259        Servico::Distribuicao => conferir_distribuicao(pedido, sandbox, &dados, &opcoes),
260    }
261}

263fn conferir_autorizacao(
264    pedido: &SandboxRequest,
265    sandbox: &SefazSandbox,
266    dados: &Element,
267    opcoes: &Opcoes,
268) -> Resposta {
269    let analise = analise::analisar_lote(dados, Some(pedido.body()), opcoes);
270    if let Some((resposta, c_stat)) = responder_rejeicao(pedido, &analise) {
271        if let Some(chave) = &analise.chave {
272            sandbox.lock().notas.registrar_rejeicao(chave, c_stat);
273        }
274        return resposta;
275    }
276    let (Some(chave), Some(nfe)) = (analise.chave.as_deref(), dados.child("NFe")) else {
277        return por_regra(pedido, "D01-LOTE", None);
278    };
279    let Some(inf) = nfe.child("infNFe") else {
280        return por_regra(pedido, "D01-LOTE", None);
281    };
282    registrar_nota(pedido, sandbox, nfe, inf, chave, opcoes)
283}

285fn registrar_nota(
286    pedido: &SandboxRequest,
287    sandbox: &SefazSandbox,
288    nfe: &Element,
289    inf: &Element,
290    chave: &str,
291    opcoes: &Opcoes,
292) -> Resposta {
293    let ide = inf.child("ide");
294    let campo = |tag: &str| ide.and_then(|i| i.child_text(tag)).unwrap_or_default();
295    let cnpj = inf.child("emit").and_then(|e| e.child_text("CNPJ")).unwrap_or_default();
296    let numero_chave = (cnpj, campo("mod"), campo("serie"), campo("nNF"));
297    let dig_val = nfe.find_text("DigestValue").unwrap_or_default();
298    let dh = pedido.dh_recbto.as_str();
299
300    let mut estado = sandbox.lock();
301    if let Some(nota) = estado.notas.por_chave.get(chave) {
302        let motivo = format!("{} [nProt:{}]", texto_de(204), nota.n_prot);
303        drop(estado);
304        return pedido.responder_com(204, &motivo);
305    }
306    if let Some(existente) = estado.notas.por_numero.get(&numero_chave) {
307        let motivo = format!("{} [chNFe:{existente}]", texto_de(539));
308        drop(estado);
309        return pedido.responder_com(539, &motivo);
310    }
311    let n_prot = pedido.proximo_protocolo();
312    estado.notas.por_numero.insert(numero_chave, chave.to_string());
313    estado.notas.rejeitadas.remove(chave);
314    estado.notas.por_chave.insert(
315        chave.to_string(),
316        Nota {
317            n_prot: n_prot.clone(),
318            dig_val: dig_val.clone(),
319            dh: dh.to_string(),
320            emitida_em: Instante::de_xml(&campo("dhEmi")),
321            autorizada_em: opcoes.agora,
322            cancelada: false,
323        },
324    );
325    drop(estado);
326
327    let prot = resposta::prot_nfe(&DadosProtocolo {
328        chave,
329        dh,
330        n_prot: Some(&n_prot),
331        dig_val: Some(&dig_val),
332        c_stat: 100,
333        motivo: texto_de(100),
334    });
335    let retorno = resposta::ret_lote(104, texto_de(104), &campo("cUF"), dh, Some(&prot));
336    Resposta::Http(resposta::soap(Servico::Autorizacao, &retorno))
337}

343fn conferir_consulta(
344    pedido: &SandboxRequest,
345    sandbox: &SefazSandbox,
346    dados: &Element,
347    opcoes: &Opcoes,
348) -> Resposta {
349    let analise = analise::analisar_consulta(dados, opcoes);
350    if let Some((resposta, _)) = responder_rejeicao(pedido, &analise) {
351        return resposta;
352    }
353    let Some(chave) = analise.chave else {
354        return por_regra(pedido, "D01", None);
355    };
356    let nota = sandbox.lock().notas.por_chave.get(&chave).cloned();
357    let Some(nota) = nota else {
358        return por_regra(pedido, "J03", None);
359    };
360    let c_uf = chave.get(..2).unwrap_or("");
361    let (c_stat, texto) = if nota.cancelada { (101, texto_de(101)) } else { (100, texto_de(100)) };
362    let prot = resposta::prot_nfe(&DadosProtocolo {
363        chave: &chave,
364        dh: &nota.dh,
365        n_prot: Some(&nota.n_prot),
366        dig_val: Some(&nota.dig_val),
367        c_stat: 100,
368        motivo: texto_de(100),
369    });
370    let retorno = resposta::ret_consulta(c_stat, texto, c_uf, &pedido.dh_recbto, &chave, Some(&prot));
371    Resposta::Http(resposta::soap(Servico::Consulta, &retorno))
372}

374fn conferir_lote_evento(
375    pedido: &SandboxRequest,
376    sandbox: &SefazSandbox,
377    dados: &Element,
378    opcoes: &Opcoes,
379) -> Resposta {
380    if dados.local_name() != "envEvento" {
381        return por_regra(pedido, "D01a", None);
382    }
383    if dados.attr("versao") != Some("1.00") {
384        return por_regra(pedido, "B06", None);
385    }
386    let eventos: Vec<&Element> = dados.elements().filter(|e| e.local_name() == "evento").collect();
387    let [evento] = eventos.as_slice() else {
388        return pedido.responder_com(215, "O sandbox só aceita lote com um evento");
389    };
390    let analise = analise::analisar_evento(evento, Some(pedido.body()), opcoes);
391    if let Some((resposta, _)) = responder_rejeicao(pedido, &analise) {
392        return resposta;
393    }
394    let Some(inf) = evento.child("infEvento") else {
395        return por_regra(pedido, "D01", None);
396    };
397    let campo = |tag: &str| inf.child_text(tag).unwrap_or_default();
398    let chave = campo("chNFe");
399    let tp_evento = campo("tpEvento");
400    let negada = if tp_evento == TP_EVENTO_CANCELAMENTO {
401        cancelar(sandbox, inf, &chave, opcoes)
402    } else {
403        manifestar(sandbox, &chave, &campo("CNPJ"), &tp_evento)
404    };
405    if let Some(id) = negada {
406        return por_regra(pedido, id, None);
407    }
408    let id_lote = dados.child_text("idLote").unwrap_or_default();
409    let c_orgao = campo("cOrgao");
410    let dh = pedido.dh_recbto.as_str();
411    let n_prot = pedido.proximo_protocolo();
412    let retorno_evento = resposta::ret_evento(&DadosEvento {
413        c_orgao: &c_orgao,
414        c_stat: 135,
415        motivo: texto_de(135),
416        chave: &chave,
417        tp_evento: &tp_evento,
418        sequencia: &campo("nSeqEvento"),
419        dh,
420        n_prot: Some(&n_prot),
421    });
422    let retorno = resposta::ret_lote_evento(&id_lote, 128, texto_de(128), &c_orgao, Some(&retorno_evento));
423    Resposta::Http(resposta::soap(Servico::Evento, &retorno))
424}

426/// The rules of a cancellation that depend on the stored document, in the order of Table 5-38.
427fn cancelar(sandbox: &SefazSandbox, inf: &Element, chave: &str, opcoes: &Opcoes) -> Option<&'static str> {
428    let protocolo = inf.child("detEvento").and_then(|d| d.child_text("nProt")).unwrap_or_default();
429    let momento = inf.child_text("dhEvento").and_then(|d| Instante::de_xml(&d));
430    let mut estado = sandbox.lock();
431    let Some(nota) = estado.notas.por_chave.get_mut(chave) else {
432        return Some("2P12-10");
433    };
434    if let (Some(agora), Some(autorizada)) = (opcoes.agora, nota.autorizada_em)
435        && agora.0 - autorizada.0 > PRAZO_CANCELAMENTO
436        && !nota.cancelada
437    {
438        return Some("2P12-14");
439    }
440    if nota.cancelada {
441        return Some("2P12-22");
442    }
443    if let (Some(momento), Some(emitida)) = (momento, nota.emitida_em)
444        && momento < emitida
445    {
446        return Some("2P13-10");
447    }
448    if let (Some(momento), Some(autorizada)) = (momento, nota.autorizada_em)
449        && momento.0 + TOLERANCIA_DATA < autorizada.0
450    {
451        return Some("2P13-14");
452    }
453    if nota.n_prot != protocolo {
454        return Some("2P23-10");
455    }
456    nota.cancelada = true;
457    None
458}

460/// The recipient's manifestation, which releases the full document.
461fn manifestar(sandbox: &SefazSandbox, chave: &str, autor: &str, tp_evento: &str) -> Option<&'static str> {
462    let mut estado = sandbox.lock();
463    let notas = &mut estado.notas;
464    let nsu = notas.proximo_nsu();
465    let Some(recebida) = notas.recebidas.get_mut(chave) else {
466        return Some("MANIF-494");
467    };
468    if recebida.destinatario.get(..8) != autor.get(..8) {
469        return Some("MANIF-575");
470    }
471    if !recebida.manifestacoes.insert(tp_evento.to_string()) {
472        return Some("3P15-10");
473    }
474    // By NT 2014.002, item 7: every manifestation but "desconhecimento" releases the download.
475    if tp_evento != TipoManifestacao::Desconhecimento.codigo() && !recebida.liberada {
476        recebida.liberada = true;
477        let entrada = Entrada {
478            nsu,
479            destinatario: recebida.destinatario.clone(),
480            schema: "procNFe_v4.00.xsd".to_string(),
481            xml: recebida.proc_nfe.clone(),
482        };
483        notas.entradas.push(entrada);
484    }
485    None
486}

488fn conferir_distribuicao(
489    pedido: &SandboxRequest,
490    sandbox: &SefazSandbox,
491    dados: &Element,
492    opcoes: &Opcoes,
493) -> Resposta {
494    if dados.local_name() != "distDFeInt" {
495        return por_regra(pedido, "D01a", None);
496    }
497    if dados.child_text("tpAmb").as_deref() != opcoes.tp_amb_esperado.as_deref() {
498        return por_regra(pedido, "DIST-252", None);
499    }
500    let cnpj = dados.child_text("CNPJ").unwrap_or_default();
501    if !cnpj_valido(&cnpj) {
502        return por_regra(pedido, "DIST-489", None);
503    }
504    let dh = pedido.dh_recbto.as_str();
505    if let Some(grupo) = dados.child("consChNFe") {
506        return distribuir_por_chave(pedido, sandbox, grupo, &cnpj);
507    }
508    let Some(grupo) = dados.child("distNSU") else {
509        return por_regra(pedido, "D01", None);
510    };
511    let ultimo: u64 = grupo.child_text("ultNSU").and_then(|t| t.parse().ok()).unwrap_or(0);
512
513    let mut estado = sandbox.lock();
514    let notas = &mut estado.notas;
515    let chave_consulta = cnpj.get(..8).unwrap_or("").to_string();
516    let maior = notas
517        .entradas
518        .iter()
519        .filter(|e| e.destinatario.get(..8) == Some(chave_consulta.as_str()))
520        .map(|e| e.nsu)
521        .max()
522        .unwrap_or(0);
523    let consulta = notas.consultas.entry(chave_consulta.clone()).or_default();
524    if ultimo > maior && maior > 0 {
525        drop(estado);
526        return por_regra(pedido, "DIST-589", None);
527    }
528    if ultimo < consulta.ultimo_entregue {
529        let entregue = consulta.ultimo_entregue;
530        drop(estado);
531        let retorno = resposta::ret_distribuicao(
532            656,
533            &format!("{}. Deve ser utilizado o ultNSU nas solicitacoes subsequentes", texto_de(656)),
534            dh,
535            entregue,
536            maior,
537            &[],
538        );
539        return Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno));
540    }
541    let documentos: Vec<(u64, String, String)> = notas
542        .entradas
543        .iter()
544        .filter(|e| e.nsu > ultimo && e.destinatario.get(..8) == Some(chave_consulta.as_str()))
545        .take(MAXIMO_LOTE)
546        .map(|e| (e.nsu, e.schema.clone(), e.xml.clone()))
547        .collect();
548    if documentos.is_empty() {
549        let bloqueado = match (consulta.sem_novidade_em, opcoes.agora) {
550            (Some(Some(antes)), Some(agora)) => agora.0 - antes.0 < ESPERA_DISTRIBUICAO,
551            (Some(_), _) => true,
552            (None, _) => false,
553        };
554        if bloqueado {
555            drop(estado);
556            let retorno = resposta::ret_distribuicao(
557                656,
558                &format!("{}. Aguarde 1 hora quando nao houver mais documentos", texto_de(656)),
559                dh,
560                ultimo,
561                maior,
562                &[],
563            );
564            return Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno));
565        }
566        consulta.sem_novidade_em = Some(opcoes.agora);
567        drop(estado);
568        let retorno = resposta::ret_distribuicao(137, texto_de(137), dh, ultimo.max(maior), maior, &[]);
569        return Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno));
570    }
571    let ult = documentos.last().map_or(ultimo, |d| d.0);
572    consulta.ultimo_entregue = ult;
573    consulta.sem_novidade_em = None;
574    drop(estado);
575    let retorno = resposta::ret_distribuicao(138, texto_de(138), dh, ult, maior, &documentos);
576    Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno))
577}
578
579fn distribuir_por_chave(
580    pedido: &SandboxRequest,
581    sandbox: &SefazSandbox,
582    grupo: &Element,
583    cnpj: &str,
584) -> Resposta {
585    let chave = grupo.child_text("chNFe").unwrap_or_default();
586    if !forma_valida(&chave) || ChaveAcesso::parse(&chave).is_err() {
587        return por_regra(pedido, "DIST-236", None);
588    }
589    let dh = pedido.dh_recbto.as_str();
590    let estado = sandbox.lock();
591    let Some(recebida) = estado.notas.recebidas.get(&chave) else {
592        drop(estado);
593        let retorno = resposta::ret_distribuicao(137, texto_de(137), dh, 0, 0, &[]);
594        return Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno));
595    };
596    if recebida.destinatario.get(..8) != cnpj.get(..8) {
597        drop(estado);
598        return por_regra(pedido, "DIST-640", None);
599    }
600    let documento = if recebida.liberada {
601        (0, "procNFe_v4.00.xsd".to_string(), recebida.proc_nfe.clone())
602    } else {
603        (0, "resNFe_v1.01.xsd".to_string(), recebida.resumo.clone())
604    };
605    drop(estado);
606    let retorno = resposta::ret_distribuicao(138, texto_de(138), dh, 0, 0, &[documento]);
607    Resposta::Http(resposta::soap(Servico::Distribuicao, &retorno))
608}

610/// Puts a purchase in the distribution of its recipient. Fails if the XML has no recipient CNPJ.
611pub(super) fn colocar_nota(sandbox: &SefazSandbox, xml_texto: &str) -> Result<(), crate::NfeError> {
612    let (chave, destinatario, proc_nfe, resumo) = compras::separar(xml_texto)?;
613    sandbox.lock().notas.colocar(chave, destinatario, proc_nfe, resumo);
614    Ok(())
615}
