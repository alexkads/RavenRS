18/// The supplier of a test purchase, and who is buying.
19#[derive(Clone, Debug, PartialEq, Eq)]
20pub struct Fornecedor {
21    /// The supplier's legal name.
22    pub razao_social: String,
23    /// The supplier's CNPJ.
24    pub cnpj: String,
25    /// The supplier's state registration.
26    pub inscricao_estadual: String,
27    /// The supplier's address.
28    pub endereco: Endereco,
29    /// The CNPJ of the buyer.
30    pub comprador_cnpj: String,
31    /// When the purchase was issued.
32    pub emitida_em: String,
33}

62/// Generates a purchase as `nfeProc`: model 55, signed by a certificate generated for the supplier.
63///
64/// The protocol is the sandbox's own, so this is a test document, with `tpAmb=2` and the
65/// homologation texts. Numbers grow on each call.
66pub fn gerar_nota_de_compra(fornecedor: &Fornecedor, itens: Vec<Item>) -> Result<String, NfeError> {
67    let certificado =
68        GeradorCertificado::novo(fornecedor.razao_social.clone(), fornecedor.cnpj.clone()).emitir()?;
69    let emissor = Emissor::builder(BetaAck::i_understand_this_is_experimental())
70        .ambiente(Ambiente::Sandbox)
71        .emitente(Emitente {
72            cnpj: fornecedor.cnpj.clone(),
73            razao_social: fornecedor.razao_social.clone(),
74            nome_fantasia: None,
75            inscricao_estadual: fornecedor.inscricao_estadual.clone(),
76            endereco: fornecedor.endereco.clone(),
77        })
78        .signer(certificado)
79        .transport(SefazSandbox::new())
80        .build()?;
81    let numero = PROXIMO_NUMERO.fetch_add(1, Ordering::Relaxed);
82    let mut rascunho =
83        Rascunho::novo(Modelo::Nfe55, 1, numero, "VENDA DE MERCADORIA", fornecedor.emitida_em.clone())
84            .com_destinatario(Destinatario {
85                documento: DocumentoDestinatario::Cnpj(fornecedor.comprador_cnpj.clone()),
86                nome: Some("COMPRADOR".to_string()),
87                endereco: Some(fornecedor.endereco.clone()),
88            });
89    rascunho.itens = itens;
90    // The payment of a sale between companies is outside what this test looks at.
91    let rascunho = rascunho.com_pagamento(FormaPagamento::SemPagamento, rust_decimal::Decimal::ZERO);
92    let documento = emissor.preparar(&rascunho)?;
93    let dig_val = xml::parse(&documento.xml)?.find_text("DigestValue").unwrap_or_default();
94    let protocolo = format!("2{numero:014}");
95    let chave = documento.chave.as_str();
96    Ok(format!(
97        "<nfeProc xmlns=\"{NS_NFE}\" versao=\"4.00\">{}<protNFe versao=\"4.00\"><infProt><tpAmb>2</tpAmb><verAplic>RAVENRS-SANDBOX</verAplic><chNFe>{chave}</chNFe><dhRecbto>{}</dhRecbto><nProt>{protocolo}</nProt><digVal>{dig_val}</digVal><cStat>100</cStat><xMotivo>[SANDBOX] {}</xMotivo></infProt></protNFe></nfeProc>",
98        documento.xml,
99        escape_text(&fornecedor.emitida_em),
100        escape_text("Autorizado o uso da NF-e"),
101    ))
102}

104fn texto_ou_vazio(element: &Element, tag: &str) -> String {
105    element.child_text(tag).unwrap_or_default()
106}

108/// The key, the recipient's CNPJ, the `nfeProc` and the summary of a document.
109pub(super) fn separar(xml_texto: &str) -> Result<(String, String, String, String), NfeError> {
110    let raiz = xml::parse(xml_texto)?;
111    let invalido = |motivo: &str| NfeError::Xml(motivo.to_string());
112    let nfe = raiz.find("NFe").ok_or_else(|| invalido("o XML não tem NFe"))?;
113    let inf = nfe.child("infNFe").ok_or_else(|| invalido("o XML não tem infNFe"))?;
114    let chave = inf
115        .attr("Id")
116        .and_then(|id| id.strip_prefix("NFe"))
117        .ok_or_else(|| invalido("o XML não tem Id"))?
118        .to_string();
119    let dest = inf.child("dest").ok_or_else(|| invalido("o XML não tem destinatário"))?;
120    let destinatario = dest.child_text("CNPJ").ok_or_else(|| invalido("o destinatário não tem CNPJ"))?;
121    let ide = inf.child("ide").ok_or_else(|| invalido("o XML não tem ide"))?;
122    let emit = inf.child("emit").ok_or_else(|| invalido("o XML não tem emit"))?;
123    let total =
124        inf.child("total").and_then(|t| t.child("ICMSTot")).ok_or_else(|| invalido("o XML não tem total"))?;
125    let protocolo = raiz.find("infProt");
126    let n_prot = protocolo.map_or_else(|| "000000000000000".to_string(), |p| texto_ou_vazio(p, "nProt"));
127    let dh = protocolo.map_or_else(|| texto_ou_vazio(ide, "dhEmi"), |p| texto_ou_vazio(p, "dhRecbto"));
128    let dig_val = nfe.find_text("DigestValue").unwrap_or_default();
129    let resumo = format!(
130        "<resNFe xmlns=\"{NS_NFE}\" versao=\"1.01\"><chNFe>{chave}</chNFe><CNPJ>{}</CNPJ><xNome>{}</xNome><IE>{}</IE><dhEmi>{}</dhEmi><tpNF>{}</tpNF><vNF>{}</vNF><digVal>{dig_val}</digVal><dhRecbto>{dh}</dhRecbto><nProt>{n_prot}</nProt><cSitNFe>1</cSitNFe></resNFe>",
131        texto_ou_vazio(emit, "CNPJ"),
132        escape_text(&texto_ou_vazio(emit, "xNome")),
133        texto_ou_vazio(emit, "IE"),
134        texto_ou_vazio(ide, "dhEmi"),
135        texto_ou_vazio(ide, "tpNF"),
136        texto_ou_vazio(total, "vNF"),
137    );
138    let proc_nfe = if raiz.local_name() == "nfeProc" {
139        xml_texto.to_string()
140    } else {
141        format!("<nfeProc xmlns=\"{NS_NFE}\" versao=\"4.00\">{}</nfeProc>", xml::to_string(nfe))
142    };
143    Ok((chave, destinatario, proc_nfe, resumo))
144}
