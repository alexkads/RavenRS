29/// One finding about a message.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub struct Diagnostico {
32    /// The rule in the catalog.
33    pub regra: &'static str,
34    /// The code the server would answer; 0 when nothing reaches the server.
35    pub c_stat: u32,
36    /// The name of the code.
37    pub titulo: String,
38    /// Whether it stops the document.
39    pub efeito: Efeito,
40    /// The path of the field.
41    pub caminho: String,
42    /// What is there, masked when the field is personal.
43    pub encontrado: Option<String>,
44    /// What should be there.
45    pub esperado: Option<String>,
46    /// What goes wrong.
47    pub explicacao: String,
48    /// What to do.
49    pub como_corrigir: String,
50    /// Where the rule came from.
51    pub fonte: Fonte,
52    /// How sure that is.
53    pub confianca: Confianca,
54}

56impl Diagnostico {
57    /// Builds a finding from a rule of the catalog. Personal values come out masked.
58    pub(crate) fn de(regra: &Regra, caminho: &str, encontrado: Option<&str>, esperado: Option<&str>) -> Self {
59        Self {
60            regra: regra.id,
61            c_stat: regra.c_stat,
62            titulo: regra.titulo().to_string(),
63            efeito: regra.efeito,
64            caminho: caminho.to_string(),
65            encontrado: encontrado.map(|valor| mascarar_em_caminho(caminho, valor)),
66            esperado: esperado.map(str::to_string),
67            explicacao: regra.explicacao.to_string(),
68            como_corrigir: regra.como_corrigir.to_string(),
69            fonte: regra.fonte,
70            confianca: regra.confianca,
71        }
72    }
73}

75/// How to judge a message.
76#[derive(Clone, Debug, Default, PartialEq, Eq)]
77pub struct Opcoes {
78    /// The moment of the receipt. Without it, rules that need a clock are listed as not judged.

105impl Analise {
106    /// The first refusal, which is what a server would answer.
107    #[must_use]
108    pub fn primeira_rejeicao(&self) -> Option<&Diagnostico> {
109        self.diagnosticos.iter().find(|d| d.efeito == Efeito::Rejeicao)

113        Self {
114            diagnosticos: Vec::new(),
115            nao_avaliado: Vec::new(),

118            autopsia: None,
119        }

122    fn achar(&mut self, id: &str, caminho: &str, encontrado: Option<&str>, esperado: Option<&str>) {
123        if let Some(regra) = regra(id) {
124            self.diagnosticos.push(Diagnostico::de(regra, caminho, encontrado, esperado));
125        }
126    }
127
128    fn alertar(&mut self, id: &str, caminho: &str, encontrado: Option<&str>, esperado: Option<&str>) {
129        if let Some(regra) = regra(id) {
130            let mut achado = Diagnostico::de(regra, caminho, encontrado, esperado);
131            achado.efeito = Efeito::Alerta;
132            self.diagnosticos.push(achado);
133        }
134    }

353/// Walks a tree looking for structure problems, with the path of each one.
354fn conferir_estrutura(analise: &mut Analise, element: &Element, caminho: &str) {
355    let nome = element.local_name();
356    let filhos = element.elements().count();
357    if filhos > 0 && element.children.iter().any(|node| matches!(node, Node::Text(_))) {
358        analise.achar("D01e", caminho, Some("texto ou espaço entre as tags"), Some("nada entre as tags"));
359    }
360    if let Some((_, ordem)) = ORDEM.iter().find(|(pai, _)| *pai == nome) {
361        let mut ultimo = 0;
362        let mut contagem = std::collections::HashMap::new();
363        for filho in element.elements() {
364            let tag = filho.local_name();
365            *contagem.entry(tag).or_insert(0_usize) += 1;
366            let caminho_filho = format!("{caminho}/{tag}");
367            match ordem.iter().position(|outro| *outro == tag) {
368                None => analise.achar("XSD-DESCONHECIDA", &caminho_filho, Some(tag), None),
369                Some(posicao) if posicao < ultimo => {
370                    analise.achar(
371                        "XSD-ORDEM",
372                        &caminho_filho,
373                        Some(tag),
374                        Some(&format!("depois de {tag} não pode vir antes de {}", ordem[ultimo])),
375                    );
376                }
377                Some(posicao) => ultimo = posicao,
378            }
379        }
380    }
381    if let Some((_, obrigatorias)) = OBRIGATORIAS.iter().find(|(pai, _)| *pai == nome) {
382        for tag in *obrigatorias {
383            if element.child(tag).is_none() {
384                analise.achar("XSD-OBRIGATORIA", &format!("{caminho}/{tag}"), None, Some(tag));
385            }
386        }
387    }
388    if nome == "Signature" {
389        return;
390    }
391    let mut repeticoes: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
392    for filho in element.elements() {
393        let tag = filho.local_name();
394        let conta = repeticoes.entry(tag).or_insert(0);
395        *conta += 1;
396        let caminho_filho = if matches!(tag, "det" | "detPag") {
397            format!("{caminho}/{tag}[{conta}]")
398        } else {
399            format!("{caminho}/{tag}")
400        };
401        conferir_estrutura(analise, filho, &caminho_filho);
402    }
403}

405fn conferir_namespaces(analise: &mut Analise, element: &Element, caminho: &str) {
406    for (chave, valor) in &element.attrs {
407        if (chave == "xmlns" || chave.starts_with("xmlns:")) && valor != NS_NFE && valor != NS_DSIG {
408            analise.achar("D01d", caminho, Some(&format!("{chave}={valor}")), Some(NS_NFE));
409        }
410    }
411    if element.local_name() == "Signature" {
412        return;
413    }
414    for filho in element.elements() {
415        conferir_namespaces(analise, filho, &format!("{caminho}/{}", filho.local_name()));
416    }
417}

432/// Checks the certificate inside a signature: groups A and E of the MOC.
433fn conferir_certificado(
434    analise: &mut Analise,
435    assinatura: &Element,
436    caminho: &str,
437    momento: Option<Instante>,
438    opcoes: &Opcoes,
439) {
440    let caminho = format!("{caminho}/Signature/KeyInfo/X509Data/X509Certificate");
441    let texto: Option<String> = assinatura
442        .child("KeyInfo")
443        .and_then(|k| k.child("X509Data"))
444        .and_then(|x| x.child_text("X509Certificate"))
445        .map(|t| t.split_whitespace().collect());
446    let der = texto.and_then(|t| {
447        use base64::Engine;
448        base64::engine::general_purpose::STANDARD.decode(t).ok()
449    });
450    let Some(der) = der else {
451        analise.achar("E01", &caminho, Some("ausente ou não é base64"), Some("certificado X.509"));
452        return;
453    };
454    let Some(info) = InfoCertificado::ler(&der) else {
455        analise.achar("E01", &caminho, Some("não foi possível ler o certificado"), Some("certificado X.509"));
456        return;
457    };
458
459    // A real certificate in the laboratory is one refusal, ahead of what else it would fail.
460    if !opcoes.ancoras.is_empty()
461        && !eh_do_laboratorio(&der)
462        && !opcoes.ancoras.iter().any(|spki| info.emitido_por(spki))
463    {
464        let motivo = if cita_icp_brasil(&der) { "certificado ICP-Brasil" } else { "certificado de fora" };
465        analise.achar(
466            "SBX-FORA-DO-LABORATORIO",
467            &caminho,
468            Some(motivo),
469            Some("emitido pela AC de teste do sandbox"),
470        );
471        return;
472    }
473    if opcoes.certificado_transmite && info.autenticacao_cliente == Some(false) {
474        analise.achar(
475            "A01",
476            &caminho,
477            Some("sem Autenticação Cliente"),
478            Some("Extended Key Usage com Autenticação Cliente"),
479        );
480    }
481    if !opcoes.certificado_transmite {
482        analise.nao_avaliado.push(
483            "O certificado de transmissão (grupo A do MOC) não vem no XML e não foi avaliado.".to_string(),
484        );
485    }
486    if info.versao != 3 {
487        analise.achar("E01", &caminho, Some(&format!("versão {}", info.versao)), Some("versão 3"));
488    }
489    if info.eh_ac {
490        analise.achar("E01", &caminho, Some("certificado de AC"), Some("certificado do titular"));
491    }
492    if info.uso_assinatura != Some(true) {
493        analise.achar(
494            "E01",
495            &caminho,
496            Some("uso da chave sem Assinatura Digital e Não Recusa"),
497            Some("Assinatura Digital e Não Recusa"),
498        );
499    }
500    if info.bits_chave.is_some_and(|bits| bits < 2048) {
501        let bits = info.bits_chave.unwrap_or(0).to_string();
502        analise.achar("CERT-CHAVE-FRACA", &caminho, Some(&format!("{bits} bits")), Some("2048 bits ou mais"));
503    }
504    if info.hash == HashCertificado::Sha1 {
505        analise.achar("CERT-SHA1", &caminho, Some("sha1WithRSAEncryption"), Some("sha256WithRSAEncryption"));
506    }
507    match momento {
508        Some(momento) => {
509            let antes = info.valido_de.is_some_and(|de| momento < de);
510            let depois = info.valido_ate.is_some_and(|ate| momento > ate);
511            if antes || depois {
512                let intervalo = format!(
513                    "de {} a {}",
514                    info.valido_de.map(Instante::para_xml).unwrap_or_default(),
515                    info.valido_ate.map(Instante::para_xml).unwrap_or_default()
516                );
517                analise.achar(
518                    "E02",
519                    &caminho,
520                    Some(&intervalo),
521                    Some(&format!("valer em {}", momento.para_xml())),
522                );
523            }
524        }
525        None => analise
526            .nao_avaliado
527            .push("A validade do certificado não foi avaliada: não há data de referência.".to_string()),
528    }
529    if info.cnpj.is_none() && info.cpf.is_none() && !eh_curinga(&info, &der) {
530        analise.achar(
531            "E03",
532            &caminho,
533            Some("sem otherName de CNPJ ou CPF"),
534            Some("OID 2.16.76.1.3.3 ou 2.16.76.1.3.1"),
535        );
536    }
537    if opcoes.ancoras.is_empty() {
538        analise
539            .nao_avaliado
540            .push("A cadeia, a raiz ICP-Brasil e a revogação do certificado não foram avaliadas: precisam da lista da SEFAZ.".to_string());
541    } else {
542        if !opcoes.ancoras.iter().any(|spki| info.emitido_por(spki)) {
543            analise.achar("E04", &caminho, info.emissor_cn.as_deref(), Some("assinado pela AC que diz ser"));
544        }
545        if opcoes.revogados.iter().any(|serial| serial == info.serial_hex()) {
546            analise.achar("E06", &caminho, Some(info.serial_hex()), Some("fora da lista de revogação"));
547        }
548    }
549}

551/// Checks a signature: group F of the MOC.
552fn conferir_assinatura(
553    analise: &mut Analise,
554    pai: &Element,
555    id: &str,
556    caminho: &str,
557    cru: Option<&str>,
558    opcoes: &Opcoes,
559) {
560    const C14N: &str = "http://www.w3.org/TR/2001/REC-xml-c14n-20010315";
561    const ENVELOPED: &str = "http://www.w3.org/2000/09/xmldsig#enveloped-signature";
562    const RSA_SHA1: &str = "http://www.w3.org/2000/09/xmldsig#rsa-sha1";
563    const SHA1: &str = "http://www.w3.org/2000/09/xmldsig#sha1";
564
565    let exame = autopsia::examinar(pai, id, cru, opcoes.agora);
566    let causa_texto = exame.causas.first().map(|c| c.explicacao.clone());
567    let caminho_assinatura = format!("{caminho}/Signature");
568    let mut padrao_ok = true;
569
570    let Some(assinatura) = pai.child("Signature") else {
571        analise.achar("F01", &caminho_assinatura, None, Some("Signature depois do documento"));
572        analise.autopsia = Some(exame);
573        return;
574    };
575    let sha256 = exame.causas.iter().any(|c| c.causa == Causa::Sha256NoLugarDeSha1);
576    if sha256 {
577        padrao_ok = false;
578        analise.achar(
579            "SIG-SHA256",
580            &format!("{caminho_assinatura}/SignedInfo"),
581            exame.algoritmo_assinatura.as_deref(),
582            Some(RSA_SHA1),
583        );
584    }
585    if assinatura.attr("xmlns") != Some(NS_DSIG) {
586        padrao_ok = false;
587        analise.achar("F01", &caminho_assinatura, assinatura.attr("xmlns"), Some(NS_DSIG));
588    }
589    let info = assinatura.child("SignedInfo");
590    let algoritmo = |grupo: Option<&Element>, tag: &str| {
591        grupo.and_then(|g| g.child(tag)).and_then(|t| t.attr("Algorithm")).map(str::to_string)
592    };
593    let c14n = algoritmo(info, "CanonicalizationMethod");
594    if c14n.as_deref() != Some(C14N) {
595        padrao_ok = false;
596        analise.achar(
597            "F01",
598            &format!("{caminho_assinatura}/SignedInfo/CanonicalizationMethod"),
599            c14n.as_deref(),
600            Some(C14N),
601        );
602    }
603    let metodo = algoritmo(info, "SignatureMethod");
604    if metodo.as_deref() != Some(RSA_SHA1) && !sha256 {
605        padrao_ok = false;
606        analise.achar(
607            "F01",
608            &format!("{caminho_assinatura}/SignedInfo/SignatureMethod"),
609            metodo.as_deref(),
610            Some(RSA_SHA1),
611        );
612    }
613    let referencia = info.and_then(|i| i.child("Reference"));
614    let caminho_referencia = format!("{caminho_assinatura}/SignedInfo/Reference");
615    let uri_certa = format!("#{id}");
616    if referencia.and_then(|r| r.attr("URI")) != Some(uri_certa.as_str()) {
617        padrao_ok = false;
618        analise.achar(
619            "F01",
620            &format!("{caminho_referencia}/@URI"),
621            referencia.and_then(|r| r.attr("URI")),
622            Some(&uri_certa),
623        );
624    }
625    let transforms: Vec<&str> = referencia
626        .and_then(|r| r.child("Transforms"))
627        .map(|t| t.elements().filter_map(|e| e.attr("Algorithm")).collect())
628        .unwrap_or_default();
629    if transforms != [ENVELOPED, C14N] {
630        padrao_ok = false;
631        analise.achar(
632            "F01",
633            &format!("{caminho_referencia}/Transforms"),
634            Some(&transforms.join(", ")),
635            Some(&format!("{ENVELOPED}, {C14N}")),
636        );
637    }
638    let digest = algoritmo(referencia, "DigestMethod");
639    if digest.as_deref() != Some(SHA1) && !sha256 {
640        padrao_ok = false;
641        analise.achar("F01", &format!("{caminho_referencia}/DigestMethod"), digest.as_deref(), Some(SHA1));
642    }
643
644    if padrao_ok {
645        if !exame.digest_confere {
646            let posicao = analise.diagnosticos.len();
647            analise.achar(
648                "F02-DIGEST",
649                &format!("{caminho_referencia}/DigestValue"),
650                exame.digest_informado.as_deref(),
651                exame.digest_calculado.as_deref(),
652            );
653            if let (Some(achado), Some(causa)) = (analise.diagnosticos.get_mut(posicao), causa_texto) {
654                achado.explicacao = format!("{} Causa provável: {causa}", achado.explicacao);
655            }
656        } else if exame.assinatura_confere == Some(false) {
657            analise.achar(
658                "F02",
659                &format!("{caminho_assinatura}/SignatureValue"),
660                None,
661                Some("valor que confere com a chave do certificado"),
662            );
663        }
664    }
665    analise.autopsia = Some(exame);
666}

669fn conferir_titular(analise: &mut Analise, emitente: Option<&Element>, caminho: &str) {

679    if let (Some(cpf), Some(certificado)) = (emitente.child_text("CPF"), &info.cpf)
680        && cpf != *certificado
681    {
682        analise.achar("F03A", &format!("{caminho}/CPF"), Some("***"), Some("o CPF do certificado"));
683    }

690fn conferir_ide(analise: &mut Analise, inf: &Element, chave: &str, opcoes: &Opcoes) {
691    const IDE: &str = "/NFe/infNFe/ide";
692    let Some(ide) = inf.child("ide") else { return };
693    let texto = |tag: &str| ide.child_text(tag).unwrap_or_default();
694    let modelo = texto("mod");
695    let emit = inf.child("emit");
696    let cnpj = emit.and_then(|e| e.child_text("CNPJ")).unwrap_or_default();
697
698    // A03-10.
699    let dh = texto("dhEmi");
700    let serie: u32 = texto("serie").parse().unwrap_or(0);
701    let numero: u32 = texto("nNF").parse().unwrap_or(0);
702    let montada = format!(
703        "{}{}{}{}{}{:03}{:09}{}{}{}",
704        texto("cUF"),
705        dh.get(2..4).unwrap_or(""),
706        dh.get(5..7).unwrap_or(""),
707        cnpj,
708        modelo,
709        serie,
710        numero,
711        texto("tpEmis"),
712        texto("cNF"),
713        texto("cDV"),
714    );
715    if montada != chave {
716        analise.achar("A03-10", "/NFe/infNFe/@Id", Some(chave), Some(&montada));
717    }
718    // B03-10.
719    let c_nf = texto("cNF");
720    const PROIBIDOS: [&str; 20] = [
721        "00000000", "11111111", "22222222", "33333333", "44444444", "55555555", "66666666", "77777777",
722        "88888888", "99999999", "12345678", "23456789", "34567890", "45678901", "56789012", "67890123",
723        "78901234", "89012345", "90123456", "01234567",
724    ];
725    if PROIBIDOS.contains(&c_nf.as_str()) || c_nf.parse::<u32>().ok() == Some(numero) {
726        analise.achar(
727            "B03-10",
728            &format!("{IDE}/cNF"),
729            Some(&c_nf),
730            Some("fora da lista, e diferente de nNF"),
731        );
732    }
733    // B09-10 and B09-40.
734    let emissao = Instante::de_xml(&dh);
735    if let (Some(agora), Some(emissao)) = (opcoes.agora, emissao) {
736        if emissao.0 > agora.0 + TOLERANCIA_DATA {
737            analise.achar(
738                "B09-10",
739                &format!("{IDE}/dhEmi"),
740                Some(&dh),
741                Some(&format!("até 5 minutos depois de {}", agora.para_xml())),
742            );
743        }
744        if modelo == "65" && texto("tpEmis") == "1" && agora.0 > emissao.0 + TOLERANCIA_DATA {
745            analise.achar(
746                "B09-40",
747                &format!("{IDE}/dhEmi"),
748                Some(&dh),
749                Some(&format!("até 5 minutos antes de {}", agora.para_xml())),
750            );
751        }
752    } else if opcoes.agora.is_none() {
753        analise.nao_avaliado.push(
754            "O atraso da emissão (703 e 704) não foi avaliado: não há hora de recebimento.".to_string(),
755        );
756    }
757    // Rules of the NFC-e.
758    if modelo == "65" {
759        if texto("idDest") != "1" {
760            analise.achar("B11a-10", &format!("{IDE}/idDest"), Some(&texto("idDest")), Some("1"));
761        }
762        if !matches!(texto("tpImp").as_str(), "4" | "5") {
763            analise.achar("B21-10", &format!("{IDE}/tpImp"), Some(&texto("tpImp")), Some("4 ou 5"));
764        }
765        if texto("finNFe") != "1" {
766            analise.achar("B25-20", &format!("{IDE}/finNFe"), Some(&texto("finNFe")), Some("1"));
767        }
768        if texto("indFinal") == "0" {
769            analise.achar("B25a-10", &format!("{IDE}/indFinal"), Some("0"), Some("1"));
770        }
771        if !matches!(texto("indPres").as_str(), "1" | "4") {
772            analise.achar("B25b-20", &format!("{IDE}/indPres"), Some(&texto("indPres")), Some("1 ou 4"));
773        }
774    } else if matches!(texto("tpImp").as_str(), "4" | "5") {
775        analise.achar("B21-20", &format!("{IDE}/tpImp"), Some(&texto("tpImp")), Some("0, 1, 2 ou 3"));
776    }
777    // B12-20.
778    if texto("cMunFG").get(..2) != texto("cUF").get(..2) {
779        analise.achar(
780            "B12-20",
781            &format!("{IDE}/cMunFG"),
782            Some(&texto("cMunFG")),
783            Some(&format!("começar com {}", texto("cUF"))),
784        );
785    }
786    // B23-10.
787    if forma_valida(chave) && chave.as_bytes()[43].wrapping_sub(b'0') != digito_mod11(&chave[..43]) {
788        analise.achar(
789            "B23-10",
790            &format!("{IDE}/cDV"),
791            chave.get(43..),
792            Some(&digito_mod11(&chave[..43]).to_string()),
793        );
794    }
795    // B24-10.
796    if let Some(esperado) = &opcoes.tp_amb_esperado {
797        if texto("tpAmb") != *esperado {
798            analise.achar("B24-10", &format!("{IDE}/tpAmb"), Some(&texto("tpAmb")), Some(esperado));
799        }
800    } else {
801        analise
802            .nao_avaliado
803            .push("O ambiente (252) não foi avaliado: não se sabe quem recebeu a mensagem.".to_string());
804    }
805    // C02-10.
806    if !cnpj.is_empty() && !cnpj_valido(&cnpj) {
807        analise.achar(
808            "C02-10",
809            "/NFe/infNFe/emit/CNPJ",
810            Some(&cnpj),
811            Some("CNPJ com dígito verificador válido"),
812        );
813    }
814}

875                    Some(&CFOP_NFCE.join(", ")),

877            }
878        }
879        for (id, quantidade, unitario) in [("I11-10", "qCom", "vUnCom"), ("I11-20", "qTrib", "vUnTrib")] {
880            let (Some(q), Some(u), Some(valor)) =
881                (decimal(prod, quantidade), decimal(prod, unitario), decimal(prod, "vProd"))
882            else {
883                continue;
884            };
885            let Some(calculado) = q.checked_mul(u) else { continue };

888                analise.achar(
889                    id,
890                    &format!("{caminho}/vProd"),
891                    Some(&valor.to_string()),
892                    Some(&calculado.to_string()),
893                );
894            }
895        }
896        let tem_500 = det.find("ICMSSN500").is_some();
897        if modelo == "55" && tem_500 && prod.child("CEST").is_none() {
898            analise.achar("N23-10", &format!("{caminho}/CEST"), None, Some("CEST do produto"));
899        }
900    }

971    let tolerancia: Decimal = TOLERANCIA_TOTAL.parse().unwrap_or(Decimal::ZERO);
972    if (calculado - v_nf).abs() > tolerancia {
973        analise.achar(
974            "W16-10",
975            &format!("{TOTAL}/vNF"),
976            Some(&v_nf.to_string()),
977            Some(&calculado.to_string()),
978        );
979    }

983fn conferir_pagamentos(analise: &mut Analise, inf: &Element, modelo: &str) {
984    const PAG: &str = "/NFe/infNFe/pag";
985    const ACEITAM_CARD: [&str; 9] = ["03", "04", "10", "11", "12", "13", "15", "17", "18"];

987    let v_nf = inf
988        .child("total")
989        .and_then(|t| t.child("ICMSTot"))
990        .map_or(Decimal::ZERO, |grupo| decimal_ou_zero(grupo, "vNF"));
991    let c_uf = inf.child("ide").and_then(|i| i.child_text("cUF")).unwrap_or_default();
992    let mut pago = Decimal::ZERO;
993    let mut sem_pagamento = false;
994    for (posicao, det) in pag.elements().filter(|e| e.local_name() == "detPag").enumerate() {
995        let caminho = format!("{PAG}/detPag[{}]", posicao + 1);
996        let t_pag = det.child_text("tPag").unwrap_or_default();
997        let valor = det.child_text("vPag").unwrap_or_default();
998        pago += valor.parse().unwrap_or(Decimal::ZERO);
999        if t_pag == "90" {
1000            sem_pagamento = true;
1001            if modelo == "65" {
1002                analise.achar(
1003                    "YA02-40",
1004                    &format!("{caminho}/tPag"),
1005                    Some("90"),
1006                    Some("a forma de pagamento real"),
1007                );
1008            }
1009            if !eh_zero(&valor) {
1010                analise.achar("YA03-30", &format!("{caminho}/vPag"), Some(&valor), Some("0.00"));
1011            }
1012        }
1013        match det.child("card") {
1014            None if matches!(t_pag.as_str(), "03" | "04" | "17") => {
1015                if modelo == "65" {
1016                    analise.achar("YA04-10", &format!("{caminho}/card"), None, Some("grupo card"));
1017                } else {
1018                    // Announced for the future in model 55 by NT 2025.001.
1019                    analise.alertar(
1020                        "YA04-10",
1021                        &format!("{caminho}/card"),
1022                        None,
1023                        Some("grupo card, em implementação futura no modelo 55"),
1024                    );
1025                }
1026            }
1027            None => {}
1028            Some(card) => {
1029                let caminho_card = format!("{caminho}/card");
1030                if !ACEITAM_CARD.contains(&t_pag.as_str()) {
1031                    analise.achar("YA04-20", &caminho_card, Some(&t_pag), Some(&ACEITAM_CARD.join(", ")));
1032                }
1033                let cnpj = card.child_text("CNPJ");
1034                let autorizacao = card.child_text("cAut");
1035                let integrado = card.child_text("tpIntegra").as_deref() == Some("1");
1036                if integrado && (cnpj.is_none() || autorizacao.is_none()) {
1037                    analise.achar("YA05-10", &caminho_card, None, Some("CNPJ e cAut"));
1038                }
1039                if let Some(cnpj) = &cnpj
1040                    && !cnpj_valido(cnpj)
1041                {
1042                    analise.achar(
1043                        "YA05-20",
1044                        &format!("{caminho_card}/CNPJ"),
1045                        Some(cnpj),
1046                        Some("CNPJ com dígito verificador válido"),
1047                    );
1048                }
1049                if let Some(cnpj) = card.child_text("CNPJReceb")
1050                    && !cnpj_valido(&cnpj)
1051                {
1052                    analise.achar(
1053                        "YA07a",
1054                        &format!("{caminho_card}/CNPJReceb"),
1055                        Some(&cnpj),
1056                        Some("CNPJ com dígito verificador válido"),
1057                    );
1058                }
1059                if c_uf == "43" && integrado && autorizacao.is_some() {
1060                    analise.achar(
1061                        "SBX-RS-CAUT",
1062                        &format!("{caminho_card}/cAut"),
1063                        autorizacao.as_deref(),
1064                        Some("igual ao comprovante do TEF"),
1065                    );
1066                }
1067            }
1068        }
1069    }
1070    let troco = pag.child_text("vTroco");
1071    if modelo == "65" && pago < v_nf && !sem_pagamento {
1072        analise.achar(
1073            "YA03-10",
1074            &format!("{PAG}/detPag/vPag"),
1075            Some(&pago.to_string()),
1076            Some(&v_nf.to_string()),
1077        );
1078    }
1079    match &troco {
1080        None if pago > v_nf => {
1081            analise.achar("YA03-20", &format!("{PAG}/vTroco"), None, Some(&(pago - v_nf).to_string()));
1082        }
1083        Some(texto) => {
1084            let informado: Decimal = texto.parse().unwrap_or(Decimal::ZERO);
1085            if informado != pago - v_nf {
1086                analise.achar(
1087                    "YA09-10",
1088                    &format!("{PAG}/vTroco"),
1089                    Some(texto),
1090                    Some(&(pago - v_nf).to_string()),
1091                );
1092            }
1093        }
1094        None => {}
1095    }

1098fn conferir_qr(analise: &mut Analise, nfe: &Element, chave: &str, modelo: &str, tp_amb: &str) {
1099    if modelo != "65" {
1100        return;
1101    }
1102    const QR: &str = "/NFe/infNFeSupl/qrCode";
1103    let Some(texto) = nfe.child("infNFeSupl").and_then(|s| s.child_text("qrCode")) else {
1104        analise.achar("ZX02-10", QR, None, Some("infNFeSupl/qrCode"));
1105        return;
1106    };
1107    let Some(parametros) = texto.split("?p=").nth(1) else {
1108        analise.nao_avaliado.push("O QR Code não usa o parâmetro p=, então não foi conferido.".to_string());
1109        return;
1110    };
1111    let partes: Vec<&str> = parametros.split('|').collect();
1112    if partes.first() != Some(&chave) {
1113        analise.achar("ZX02-228", QR, partes.first().copied(), Some(chave));
1114    }
1115    if !matches!(partes.get(1).copied(), Some("2" | "3")) {
1116        analise.achar("ZX02-236", QR, partes.get(1).copied(), Some("2 ou 3"));
1117    }
1118    if partes.get(2).copied() != Some(tp_amb) {
1119        analise.achar("ZX02-244", QR, partes.get(2).copied(), Some(tp_amb));
1120    }
1121}

1168            .to_string(),
1169    );

1173/// Judges a batch and the document inside it.
1174pub(crate) fn analisar_lote(lote: &Element, cru: Option<&str>, opcoes: &Opcoes) -> Analise {
1175    let mut prefixo = Analise::nova("enviNFe");
1176    if lote.local_name() != "enviNFe" {
1177        prefixo.achar("D01a-LOTE", "/", Some(lote.local_name()), Some("enviNFe"));
1178        return prefixo;
1179    }
1180    match lote.attr("versao") {
1181        None => prefixo.achar("D01b", "/enviNFe/@versao", None, Some("4.00")),
1182        Some("4.00") => {}
1183        Some(outra) => prefixo.achar("B06", "/enviNFe/@versao", Some(outra), Some("4.00")),
1184    }
1185    if lote.attr("xmlns") != Some(NS_NFE) {
1186        prefixo.achar("D01-LOTE", "/enviNFe/@xmlns", lote.attr("xmlns"), Some(NS_NFE));
1187    }
1188    conferir_cabecalho_lote(&mut prefixo, lote);
1189    let id_lote = lote.child_text("idLote").unwrap_or_default();
1190    if id_lote.is_empty() || id_lote.len() > 15 || !id_lote.bytes().all(|b| b.is_ascii_digit()) {
1191        prefixo.achar("XSD-VALOR", "/enviNFe/idLote", Some(&id_lote), Some("de 1 a 15 dígitos"));
1192    }
1193    match lote.child_text("indSinc").as_deref() {
1194        Some("1") => {}
1195        Some("0") => {
1196            let extinto = opcoes.agora.is_some_and(|agora| {
1197                em_vigor("ASSINCRONO-EXTINTO", opcoes.tp_amb_esperado.as_deref() != Some("1"), agora)
1198            });
1199            let id = if extinto { "GAP03a-6" } else { "GAP03a-3" };
1200            prefixo.achar(id, "/enviNFe/indSinc", Some("0"), Some("1"));
1201        }
1202        outro => prefixo.achar("XSD-VALOR", "/enviNFe/indSinc", outro, Some("1")),
1203    }
1204    let notas: Vec<&Element> = lote.elements().filter(|e| e.local_name() == "NFe").collect();
1205    let [nfe] = notas.as_slice() else {
1206        let modelo_65 = notas.iter().any(|n| n.find_text("mod").as_deref() == Some("65"));
1207        if notas.len() > 1 && modelo_65 {
1208            prefixo.achar("GAP03a-4", "/enviNFe/NFe", Some(&notas.len().to_string()), Some("1"));
1209        } else {
1210            prefixo.achar(
1211                "D01-LOTE",
1212                "/enviNFe/NFe",
1213                Some(&notas.len().to_string()),
1214                Some("uma nota por lote neste sandbox"),
1215            );
1216        }
1217        return prefixo;
1218    };
1219    let mut analise = analisar_nfe(nfe, cru, opcoes);
1220    prefixo.diagnosticos.append(&mut analise.diagnosticos);
1221    analise.diagnosticos = prefixo.diagnosticos;
1222    analise.tipo = "enviNFe".to_string();
1223    analise
1224}

1226fn conferir_cabecalho_lote(analise: &mut Analise, lote: &Element) {
1227    for filho in lote.children.iter() {
1228        if let Node::Text(_) = filho {
1229            analise.achar(
1230                "D01e",
1231                "/enviNFe",
1232                Some("texto ou espaço entre as tags"),
1233                Some("nada entre as tags"),
1234            );
1235            break;
1236        }
1237    }
1238    let ordem = ["idLote", "indSinc", "NFe"];
1239    let mut ultimo = 0;

1242            None => {
1243                analise.achar("XSD-DESCONHECIDA", &format!("/enviNFe/{}", filho.local_name()), None, None)
1244            }
1245            Some(posicao) if posicao < ultimo => {
1246                analise.achar("XSD-ORDEM", &format!("/enviNFe/{}", filho.local_name()), None, None);
1247            }
1248            Some(posicao) => ultimo = posicao,
1249        }
1250    }

1253/// Judges a cancellation event.
1254pub(crate) fn analisar_evento(evento: &Element, cru: Option<&str>, opcoes: &Opcoes) -> Analise {
1255    const INF: &str = "/evento/infEvento";
1256    let mut analise = Analise::nova("evento");
1257    conferir_estrutura(&mut analise, evento, "/evento");
1258    let Some(inf) = evento.child("infEvento") else {
1259        analise.achar("D01", INF, None, Some("infEvento"));
1260        return analise;
1261    };
1262    let campo = |tag: &str| inf.child_text(tag).unwrap_or_default();
1263    let chave = campo("chNFe");
1264    let tp_evento = campo("tpEvento");
1265    let cancelamento = tp_evento == TP_EVENTO_CANCELAMENTO;
1266    let manifestacao = TipoManifestacao::de_codigo(&tp_evento);
1267    if !cancelamento && manifestacao.is_none() {
1268        analise.achar(
1269            "D04",
1270            &format!("{INF}/tpEvento"),
1271            Some(&tp_evento),
1272            Some("110111, 210200, 210210, 210220 ou 210240"),
1273        );
1274    }
1275    if campo("verEvento") != "1.00" {
1276        analise.achar("D05", &format!("{INF}/verEvento"), Some(&campo("verEvento")), Some("1.00"));
1277    }
1278    let detalhe = inf.child("detEvento");
1279    let justificativa = detalhe.and_then(|d| d.child_text("xJust")).unwrap_or_default().chars().count();
1280    let descricao = detalhe.and_then(|d| d.child_text("descEvento"));
1281    if cancelamento {
1282        if descricao.as_deref() != Some("Cancelamento") || !(15..=255).contains(&justificativa) {
1283            analise.achar(
1284                "D06",
1285                &format!("{INF}/detEvento"),
1286                Some(&format!("xJust com {justificativa} caracteres")),
1287                Some("descEvento=Cancelamento e xJust de 15 a 255 caracteres"),
1288            );
1289        }
1290    } else if let Some(tipo) = manifestacao {
1291        if descricao.as_deref() != Some(tipo.descricao()) {
1292            analise.achar(
1293                "D06",
1294                &format!("{INF}/detEvento/descEvento"),
1295                descricao.as_deref(),
1296                Some(tipo.descricao()),
1297            );
1298        }
1299        if tipo == TipoManifestacao::NaoRealizada && !(15..=255).contains(&justificativa) {
1300            analise.achar(
1301                "MANIF-595",
1302                &format!("{INF}/detEvento/xJust"),
1303                None,
1304                Some("de 15 a 255 caracteres"),
1305            );
1306        }
1307    }
1308    if !forma_valida(&chave) {
1309        analise.achar("D01", &format!("{INF}/chNFe"), Some(&chave), Some("44 caracteres"));
1310    } else {
1311        analise.chave = Some(chave.clone());
1312        if ChaveAcesso::parse(&chave).is_err() {
1313            analise.achar("P-DV", &format!("{INF}/chNFe"), Some(&chave), None);
1314        }
1315    }
1316    let sequencia = campo("nSeqEvento");
1317    let numero: u8 = sequencia.parse().unwrap_or(0);
1318    let id = format!("ID{tp_evento}{chave}{numero:02}");
1319    if inf.attr("Id") != Some(id.as_str()) {
1320        analise.achar("D01", &format!("{INF}/@Id"), inf.attr("Id"), Some(&id));
1321    }
1322    let dh = campo("dhEvento");
1323    let momento = opcoes.agora.or_else(|| Instante::de_xml(&dh));
1324    if let Some(assinatura) = evento.child("Signature") {
1325        conferir_certificado(&mut analise, assinatura, "/evento", momento, opcoes);
1326    }
1327    let id_real = inf.attr("Id").unwrap_or_default().to_string();
1328    conferir_assinatura(&mut analise, evento, &id_real, "/evento", cru, opcoes);
1329    conferir_titular(&mut analise, Some(inf), INF);
1330    if let Some(esperado) = &opcoes.tp_amb_esperado
1331        && campo("tpAmb") != *esperado
1332    {
1333        analise.achar("B24-10", &format!("{INF}/tpAmb"), Some(&campo("tpAmb")), Some(esperado));
1334    }
1335    if cancelamento && chave.get(6..20) != Some(campo("CNPJ").as_str()) {
1336        analise.achar("P12-44", &format!("{INF}/CNPJ"), Some(&campo("CNPJ")), chave.get(6..20));
1337    }
1338    if let (Some(agora), Some(evento_em)) = (opcoes.agora, Instante::de_xml(&dh))
1339        && evento_em.0 > agora.0 + TOLERANCIA_DATA
1340    {
1341        analise.achar(
1342            "P13-10",
1343            &format!("{INF}/dhEvento"),
1344            Some(&dh),
1345            Some(&format!("até 5 minutos depois de {}", agora.para_xml())),
1346        );
1347    }
1348    if numero != 1 {
1349        analise.achar("P15-10", &format!("{INF}/nSeqEvento"), Some(&sequencia), Some("1"));
1350    }
1351    analise.nao_avaliado.push(
1352        "Existência, situação e prazo da nota (494, 501, 580, 222, 573) dependem do que a SEFAZ guarda."
1353            .to_string(),
1354    );
1355    analise
1356}

1408/// Judges any message: `NFe`, `nfeProc`, `enviNFe`, `evento`, the queries, or any of them in SOAP.
1409pub fn analisar(xml_texto: &str, opcoes: &Opcoes) -> Analise {
1410    let raiz = match xml::parse(xml_texto) {
1411        Ok(raiz) => raiz,
1412        Err(erro) => {
1413            let mut analise = Analise::nova("ilegível");
1414            analise.achar("B02", "/", Some(&erro.to_string()), Some("XML bem formado"));
1415            return analise;
1416        }
1417    };
1418    let mut analise = if let Some(lote) = raiz.find("enviNFe") {
1419        // Comes before NFe, which is inside it.
1420        analisar_lote(lote, Some(xml_texto), opcoes)
1421    } else if let Some(nfe) = raiz.find("NFe") {
1422        analisar_nfe(nfe, Some(xml_texto), opcoes)
1423    } else if let Some(evento) = raiz.find("evento") {
1424        analisar_evento(evento, Some(xml_texto), opcoes)
1425    } else if let Some(consulta) = raiz.find("consSitNFe") {
1426        analisar_consulta(consulta, opcoes)
1427    } else if let Some(status) = raiz.find("consStatServ") {
1428        analisar_status(status, opcoes)
1429    } else {
1430        let mut analise = Analise::nova(raiz.local_name());
1431        analise.achar("D01a", "/", Some(raiz.local_name()), Some("NFe, enviNFe, evento ou uma consulta"));
1432        analise
1433    };
1434    if xml_texto.starts_with('\u{feff}')
1435        && let Some(regra) = regra("D01e")
1436    {
1437        let achado =
1438            Diagnostico::de(regra, "/", Some("BOM no início do texto"), Some("texto começando em <"));
1439        analise.diagnosticos.insert(0, achado);
1440    }
1441    analise
1442}
