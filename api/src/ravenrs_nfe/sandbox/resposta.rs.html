9/// The codes that stop at the batch, with no `protNFe` or `retEvento` inside.
10const SO_NO_LOTE: [u32; 14] = [108, 109, 126, 129, 215, 225, 239, 243, 452, 516, 517, 565, 587, 588];

12/// The fallback state when a message does not say which one it is from.
13const UF_PADRAO: &str = "35";

15fn motivo(texto: &str) -> String {
16    escape_text(&format!("{PREFIXO_MOTIVO}{texto}"))
17}

19/// Wraps a return in the SOAP 1.2 envelope of a service.
20pub(super) fn soap(servico: Servico, retorno: &str) -> HttpResponse {
21    let corpo = if servico == Servico::Distribuicao {
22        format!(
23            "<nfeDistDFeInteresseResponse xmlns=\"{}\"><nfeDistDFeInteresseResult>{retorno}</nfeDistDFeInteresseResult></nfeDistDFeInteresseResponse>",
24            servico.namespace()
25        )
26    } else {
27        format!("<nfeResultMsg xmlns=\"{}\">{retorno}</nfeResultMsg>", servico.namespace())
28    };
29    let body = format!(
30        "<?xml version=\"1.0\" encoding=\"utf-8\"?><soap:Envelope xmlns:soap=\"{NS_SOAP12}\"><soap:Body>{corpo}</soap:Body></soap:Envelope>"
31    );
32    HttpResponse {
33        status: 200,
34        headers: vec![("Content-Type".to_string(), "application/soap+xml; charset=utf-8".to_string())],
35        body: body.into_bytes(),
36    }
37}
38
39/// A SOAP 1.2 fault, which is how a server refuses an envelope it cannot route.
40pub(super) fn fault(status: u16, texto: &str) -> HttpResponse {
41    let body = format!(
42        "<?xml version=\"1.0\" encoding=\"utf-8\"?><soap:Envelope xmlns:soap=\"{NS_SOAP12}\"><soap:Body><soap:Fault><soap:Code><soap:Value>soap:Sender</soap:Value></soap:Code><soap:Reason><soap:Text xml:lang=\"pt-BR\">{}</soap:Text></soap:Reason></soap:Fault></soap:Body></soap:Envelope>",
43        motivo(texto)
44    );
45    HttpResponse {
46        status,
47        headers: vec![("Content-Type".to_string(), "application/soap+xml; charset=utf-8".to_string())],
48        body: body.into_bytes(),
49    }
50}

53    pub chave: &'a str,
54    pub dh: &'a str,
55    pub n_prot: Option<&'a str>,
56    pub dig_val: Option<&'a str>,
57    pub c_stat: u32,

61pub(super) fn prot_nfe(dados: &DadosProtocolo<'_>) -> String {
62    let n_prot = dados.n_prot.map_or(String::new(), |n| format!("<nProt>{n}</nProt>"));
63    let dig_val = dados.dig_val.map_or(String::new(), |d| format!("<digVal>{d}</digVal>"));
64    format!(
65        "<protNFe versao=\"4.00\"><infProt><tpAmb>2</tpAmb><verAplic>{VER_APLIC}</verAplic><chNFe>{}</chNFe><dhRecbto>{}</dhRecbto>{n_prot}{dig_val}<cStat>{}</cStat><xMotivo>{}</xMotivo></infProt></protNFe>",
66        dados.chave,
67        dados.dh,
68        dados.c_stat,
69        motivo(dados.motivo)
70    )
71}

73pub(super) fn ret_lote(c_stat: u32, texto: &str, c_uf: &str, dh: &str, prot: Option<&str>) -> String {

76        motivo(texto),
77        prot.unwrap_or("")
78    )

103/// The return of the distribution service, each document compressed with GZip.
104pub(super) fn ret_distribuicao(
105    c_stat: u32,
106    texto: &str,
107    dh: &str,
108    ult_nsu: u64,
109    max_nsu: u64,
110    documentos: &[(u64, String, String)],
111) -> String {
112    let mut lote = String::new();
113    if !documentos.is_empty() {
114        lote.push_str("<loteDistDFeInt>");
115        for (nsu, schema, xml) in documentos {
116            let base = ravenrs_dfe::compactacao::gzip_base64(xml).unwrap_or_default();
117            lote.push_str(&format!("<docZip NSU=\"{nsu:015}\" schema=\"{schema}\">{base}</docZip>"));
118        }
119        lote.push_str("</loteDistDFeInt>");
120    }
121    format!(
122        "<retDistDFeInt xmlns=\"{NS_NFE}\" versao=\"1.01\"><tpAmb>2</tpAmb><verAplic>{VER_APLIC}</verAplic><cStat>{c_stat}</cStat><xMotivo>{}</xMotivo><dhResp>{dh}</dhResp><ultNSU>{ult_nsu:015}</ultNSU><maxNSU>{max_nsu:015}</maxNSU>{lote}</retDistDFeInt>",
123        motivo(texto)
124    )
125}

138pub(super) fn ret_evento(dados: &DadosEvento<'_>) -> String {
139    let registro = dados
140        .n_prot
141        .map_or(String::new(), |n| format!("<dhRegEvento>{}</dhRegEvento><nProt>{n}</nProt>", dados.dh));
142    let descricao = if dados.n_prot.is_some() { "<xEvento>Cancelamento registrado</xEvento>" } else { "" };
143    format!(
144        "<retEvento versao=\"1.00\"><infEvento><tpAmb>2</tpAmb><verAplic>{VER_APLIC}</verAplic><cOrgao>{}</cOrgao><cStat>{}</cStat><xMotivo>{}</xMotivo><chNFe>{}</chNFe><tpEvento>{}</tpEvento>{descricao}<nSeqEvento>{}</nSeqEvento>{registro}</infEvento></retEvento>",
145        dados.c_orgao,
146        dados.c_stat,
147        motivo(dados.motivo),
148        dados.chave,
149        dados.tp_evento,
150        dados.sequencia,
151    )
152}

168fn uf_de(dados: Option<&Element>) -> String {
169    let Some(dados) = dados else { return UF_PADRAO.to_string() };
170    dados
171        .find_text("cUF")
172        .or_else(|| dados.find_text("cOrgao"))
173        .or_else(|| dados.find_text("chNFe").and_then(|c| c.get(..2).map(str::to_string)))
174        .unwrap_or_else(|| UF_PADRAO.to_string())
175}
176
177/// The key of a message, or a valid-looking one made of zeros when it has none.
178fn chave_de(dados: Option<&Element>) -> String {
179    let id = dados
180        .and_then(|d| d.find("infNFe"))
181        .and_then(|inf| inf.attr("Id"))
182        .and_then(|id| id.strip_prefix("NFe"))
183        .map(str::to_string);
184    id.or_else(|| dados.and_then(|d| d.find_text("chNFe")))
185        .filter(|chave| crate::chave::forma_valida(chave))
186        .unwrap_or_else(|| "0".repeat(44))
187}

189/// Answers a `cStat` in the format of the service of the request.
190pub(super) fn responder(pedido: &SandboxRequest, c_stat: u32, texto: &str) -> Resposta {
191    let Some(servico) = pedido.servico else {
192        return Resposta::Erro(TransportError::Outro(
193            "um handler só responde um cStat em um dos quatro serviços".to_string(),
194        ));
195    };
196    let dados = pedido.dados();
197    let dados = dados.as_ref();
198    let c_uf = uf_de(dados);
199    let dh = pedido.dh_recbto.as_str();
200    let chave = chave_de(dados);
201    let retorno = match servico {
202        Servico::Status => ret_status(c_stat, texto, &c_uf, dh),
203        Servico::Autorizacao if SO_NO_LOTE.contains(&c_stat) => ret_lote(c_stat, texto, &c_uf, dh, None),
204        Servico::Autorizacao => {
205            let n_prot = matches!(c_stat, 100 | 150).then(|| pedido.proximo_protocolo());
206            let dig_val = dados.and_then(|d| d.find_text("DigestValue"));
207            let prot = prot_nfe(&DadosProtocolo {
208                chave: &chave,
209                dh,
210                n_prot: n_prot.as_deref(),
211                dig_val: dig_val.as_deref(),
212                c_stat,
213                motivo: texto,
214            });
215            ret_lote(104, "Lote processado", &c_uf, dh, Some(&prot))
216        }
217        Servico::Consulta => ret_consulta(c_stat, texto, &c_uf, dh, &chave, None),
218        Servico::Distribuicao => ret_distribuicao(c_stat, texto, dh, 0, 0, &[]),
219        Servico::Evento => {
220            let id_lote = dados.and_then(|d| d.child_text("idLote")).unwrap_or_else(|| "1".to_string());
221            if SO_NO_LOTE.contains(&c_stat) {
222                ret_lote_evento(&id_lote, c_stat, texto, &c_uf, None)
223            } else {
224                let sequencia =
225                    dados.and_then(|d| d.find_text("nSeqEvento")).unwrap_or_else(|| "1".to_string());
226                let n_prot = matches!(c_stat, 135 | 136 | 155).then(|| pedido.proximo_protocolo());
227                let evento = ret_evento(&DadosEvento {
228                    c_orgao: &c_uf,
229                    c_stat,
230                    motivo: texto,
231                    chave: &chave,
232                    tp_evento: &dados
233                        .and_then(|d| d.find_text("tpEvento"))
234                        .unwrap_or_else(|| "110111".to_string()),
235                    sequencia: &sequencia,
236                    dh,
237                    n_prot: n_prot.as_deref(),
238                });
239                ret_lote_evento(&id_lote, 128, "Lote de evento processado", &c_uf, Some(&evento))
240            }
241        }
242    };
243    Resposta::Http(soap(servico, &retorno))
