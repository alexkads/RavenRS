7/// The fields every return has.
8#[derive(Clone, Debug, PartialEq, Eq)]
9pub struct Retorno {
10    /// The status code.
11    pub c_stat: u32,
12    /// The text the server sent.
13    pub x_motivo: String,
14    /// Which application answered. The sandbox says `RAVENRS-SANDBOX`.
15    pub ver_aplic: String,
16    /// The environment the server says it is, `tpAmb`.
17    pub tp_amb: String,
18    /// When the server received the message, when it says.
19    pub dh_recbto: Option<String>,
20}

22impl Retorno {

24    #[must_use]
25    pub fn veio_do_sandbox(&self) -> bool {
26        self.ver_aplic == "RAVENRS-SANDBOX"
27    }

64/// The outcome of a document, read from its `cStat`.
65#[derive(Clone, Copy, Debug, PartialEq, Eq)]
66pub enum Situacao {
67    /// The use was authorized: 100 or 150.
68    Autorizada,
69    /// The use was denied: 110, 205, 301, 302 or 303.
70    Denegada,
71    /// The document already exists with this key or number: 204 or 539.
72    Duplicada,
73    /// The service is down for now: 108 or 109.
74    Paralisado,
75    /// The document was refused.
76    Rejeitada,
77}
78
79/// The situation of a `cStat`.
80#[must_use]
81pub const fn situacao_de(c_stat: u32) -> Situacao {
82    match c_stat {
83        100 | 150 => Situacao::Autorizada,
84        110 | 205 | 301 | 302 | 303 => Situacao::Denegada,
85        204 | 539 => Situacao::Duplicada,
86        108 | 109 => Situacao::Paralisado,
87        _ => Situacao::Rejeitada,
88    }
89}

91/// The answer of a synchronous authorization.
92#[derive(Clone, Debug, PartialEq, Eq)]
93pub struct Autorizacao {
94    /// The status of the batch.
95    pub lote: Retorno,
96    /// The protocol of the document, absent when the batch itself was refused.
97    pub protocolo: Option<Protocolo>,
98    /// The document with its protocol, `nfeProc`, present only when authorized.
99    pub nfe_proc: Option<String>,
100}
101
102impl Autorizacao {
103    /// The status that decides: the document's if it has one, otherwise the batch's.
104    #[must_use]
105    pub fn c_stat(&self) -> u32 {
106        self.protocolo.as_ref().map_or(self.lote.c_stat, |protocolo| protocolo.c_stat)
107    }
108
109    /// The text of [`Self::c_stat`].
110    #[must_use]
111    pub fn x_motivo(&self) -> &str {
112        self.protocolo.as_ref().map_or(self.lote.x_motivo.as_str(), |p| p.x_motivo.as_str())
113    }
114
115    /// The outcome.
116    #[must_use]
117    pub fn situacao(&self) -> Situacao {
118        situacao_de(self.c_stat())
119    }
120}

156/// Finds the return element, or says what came instead: a SOAP fault or nothing usable.
157fn corpo(resposta: &HttpResponse, raiz: &str) -> Result<Element, NfeError> {
158    let invalido = |motivo: String| NfeError::RetornoInvalido { status: resposta.status, motivo };
159    let texto = std::str::from_utf8(&resposta.body)
160        .map_err(|_| invalido("corpo da resposta não é UTF-8".to_string()))?;
161    let documento = xml::parse(texto).map_err(|erro| invalido(erro.to_string()))?;
162    if let Some(retorno) = documento.find(raiz) {
163        return Ok(retorno.clone());
164    }
165    if let Some(fault) = documento.find("Fault") {
166        let motivo = fault
167            .find("Reason")
168            .and_then(|reason| reason.find_text("Text"))
169            .or_else(|| fault.find_text("faultstring"))
170            .unwrap_or_else(|| "SOAP Fault sem texto".to_string());
171        return Err(invalido(motivo));
172    }
173    Err(invalido(format!("{raiz} não encontrado na resposta")))
174}

176fn comum(element: &Element, status: u16) -> Result<Retorno, NfeError> {
177    let c_stat = element
178        .child_text("cStat")
179        .and_then(|text| text.trim().parse().ok())
180        .ok_or_else(|| NfeError::RetornoInvalido { status, motivo: "retorno sem cStat".to_string() })?;
181    Ok(Retorno {
182        c_stat,
183        x_motivo: element.child_text("xMotivo").unwrap_or_default(),
184        ver_aplic: element.child_text("verAplic").unwrap_or_default(),
185        tp_amb: element.child_text("tpAmb").unwrap_or_default(),
186        dh_recbto: element.child_text("dhRecbto"),
187    })
188}
189
190fn protocolo(element: &Element, status: u16) -> Result<Option<Protocolo>, NfeError> {
191    let Some(inf) = element.child("protNFe").and_then(|prot| prot.child("infProt")) else {
192        return Ok(None);
193    };
194    let base = comum(inf, status)?;
195    Ok(Some(Protocolo {
196        ch_nfe: inf.child_text("chNFe").unwrap_or_default(),
197        dh_recbto: base.dh_recbto,
198        n_prot: inf.child_text("nProt"),
199        dig_val: inf.child_text("digVal"),
200        c_stat: base.c_stat,
201        x_motivo: base.x_motivo,
202    }))
203}

211/// Reads the authorization return; `nfe_assinada` becomes `nfeProc` when it was authorized.
212pub fn ler_autorizacao(resposta: &HttpResponse, nfe_assinada: &str) -> Result<Autorizacao, NfeError> {
213    let element = corpo(resposta, "retEnviNFe")?;
214    let lote = comum(&element, resposta.status)?;
215    let protocolo = protocolo(&element, resposta.status)?;
216    let autorizada = protocolo.as_ref().is_some_and(|p| situacao_de(p.c_stat) == Situacao::Autorizada);
217    let nfe_proc = element.child("protNFe").filter(|_| autorizada).map(|prot| {
218        format!(
219            "<nfeProc versao=\"4.00\" xmlns=\"{}\">{nfe_assinada}{}</nfeProc>",
220            crate::soap::NS_NFE,
221            xml::to_string(prot)
222        )
223    });
224    Ok(Autorizacao { lote, protocolo, nfe_proc })
225}

237/// Reads the event return.
238pub fn ler_evento(resposta: &HttpResponse) -> Result<RetornoEvento, NfeError> {
239    let element = corpo(resposta, "retEnvEvento")?;
240    let lote = comum(&element, resposta.status)?;
241    let inf = element.child("retEvento").and_then(|retorno| retorno.child("infEvento"));
242    let (c_stat, x_motivo) = match inf {
243        Some(inf) => {
244            let base = comum(inf, resposta.status)?;
245            (base.c_stat, base.x_motivo)
246        }
247        None => (lote.c_stat, lote.x_motivo.clone()),
248    };
249    Ok(RetornoEvento {
250        c_stat,
251        x_motivo,
252        ch_nfe: inf.and_then(|inf| inf.child_text("chNFe")),
253        n_prot: inf.and_then(|inf| inf.child_text("nProt")),
254        lote,
255    })
256}

376/// One document of a distribution batch, already decompressed.
377#[derive(Clone, Debug, PartialEq, Eq)]
378pub struct DocumentoDfe {
379    /// The NSU; since NT 2014.002 v1.15 the server may leave it out.
380    pub nsu: Option<u64>,
381    /// The schema that names the kind: `resNFe_v1.01.xsd`, `procNFe_v4.00.xsd` and so on.
382    pub schema: String,
383    /// The XML.
384    pub xml: String,
385}
386
387impl DocumentoDfe {
388    /// Whether this is only the summary, which asks for a manifestation before the full XML.
389    #[must_use]
390    pub fn eh_resumo(&self) -> bool {
391        self.schema.starts_with("resNFe")
392    }
393
394    /// Whether this is the full document with its protocol.
395    #[must_use]
396    pub fn eh_nota_completa(&self) -> bool {
397        self.schema.starts_with("procNFe")
398    }
399}

401/// The answer of the distribution service.
402#[derive(Clone, Debug, PartialEq, Eq)]
403pub struct LoteDfe {
404    /// The last NSU handed over: the next query starts here.
405    pub ult_nsu: u64,
406    /// The largest NSU there is.
407    pub max_nsu: u64,
408    /// The documents.
409    pub documentos: Vec<DocumentoDfe>,
410    /// The status.
411    pub c_stat: u32,
412    /// The text of the status.
413    pub x_motivo: String,
414}
415
416impl LoteDfe {
417    /// Whether nothing is left to fetch: the next query should wait one hour (NT 2014.002, 3.11.4).
418    #[must_use]
419    pub fn sem_novidade(&self) -> bool {
420        self.c_stat == 137 || (self.c_stat == 138 && self.ult_nsu >= self.max_nsu)
421    }
422}

424/// Reads the distribution return, decompressing each document with the limit of
425/// [`ravenrs_dfe::compactacao::LIMITE_DOCUMENTO`], so a gzip bomb stops there.
426pub fn ler_distribuicao(resposta: &HttpResponse) -> Result<LoteDfe, NfeError> {
427    let element = corpo(resposta, "retDistDFeInt")?;
428    let retorno = comum(&element, resposta.status)?;
429    let numero = |tag: &str| element.child_text(tag).and_then(|t| t.parse::<u64>().ok()).unwrap_or(0);
430    let mut documentos = Vec::new();
431    if let Some(lote) = element.child("loteDistDFeInt") {
432        for zip in lote.elements().filter(|e| e.local_name() == "docZip") {
433            let invalido = |motivo: &str| NfeError::RetornoInvalido {
434                status: resposta.status,
435                motivo: motivo.to_string(),
436            };
437            let xml = ravenrs_dfe::compactacao::de_gzip_base64(&zip.text())
438                .map_err(|erro| invalido(&format!("docZip: {erro}")))?;
439            documentos.push(DocumentoDfe {
440                nsu: zip.attr("NSU").and_then(|n| n.parse().ok()),
441                schema: zip.attr("schema").unwrap_or_default().to_string(),
442                xml,
443            });
444        }
445    }
446    Ok(LoteDfe {
447        ult_nsu: numero("ultNSU"),
448        max_nsu: numero("maxNSU"),
449        documentos,
450        c_stat: retorno.c_stat,
451        x_motivo: retorno.x_motivo,
452    })
453}
