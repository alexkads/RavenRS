1//! Where a document goes: the environment, the beta acknowledgement, the model and the state.
2
3/// The explicit acknowledgement that this crate is an experimental proof of concept.
4///
5/// It has no public constructor other than the one with the long name, so an emitter
6/// can only exist in code that says it out loud.
7///
8/// ```compile_fail,E0423
9/// let ack = ravenrs_nfe::BetaAck(());
10/// ```
11#[derive(Clone, Copy, Debug)]
12pub struct BetaAck(());
13
14impl BetaAck {
15    /// Acknowledges that this is a beta, not homologated and not for real fiscal documents.
16    #[must_use]
17    pub const fn i_understand_this_is_experimental() -> Self {
18        Self(())
19    }
20}
21
22/// The environment of a document. Production does not exist in this enum on purpose.
23///
24/// ```compile_fail,E0599
25/// let producao = ravenrs_nfe::Ambiente::Producao;
26/// ```
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum Ambiente {
29    /// Everything stays in the local `SefazSandbox`; nothing reaches the SEFAZ.
30    Sandbox,
31    /// The real homologation servers of the SEFAZ, with no fiscal value.
32    Homologacao,
33}
34
35impl Ambiente {
36    /// The value of `tpAmb`, which is `2` in every variant: no path here writes `1`.
37    #[must_use]
38    pub const fn tp_amb(self) -> &'static str {
39        "2"
40    }
41}
42
43/// Who produced a document, kept with it so a test note never passes for a real one.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum Origem {
46    /// Produced for the local sandbox.
47    Sandbox,
48    /// Produced for the homologation servers.
49    Homologacao,
50}
51
52impl From<Ambiente> for Origem {
53    fn from(ambiente: Ambiente) -> Self {
54        match ambiente {
55            Ambiente::Sandbox => Self::Sandbox,
56            Ambiente::Homologacao => Self::Homologacao,
57        }
58    }
59}

61/// The document model.
62#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
63pub enum Modelo {
64    /// NF-e, model 55.
65    Nfe55,
66    /// NFC-e, model 65: what a cash register issues.
67    Nfce65,
68}
69
70impl Modelo {
71    /// The value of `mod`.
72    #[must_use]
73    pub const fn codigo(self) -> &'static str {
74        match self {
75            Self::Nfe55 => "55",
76            Self::Nfce65 => "65",
77        }
78    }
79}
