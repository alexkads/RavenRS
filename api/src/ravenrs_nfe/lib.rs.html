1//! # ravenrs-nfe: BETA, PROOF OF CONCEPT
2//!
3//! **Not homologated. Must not issue real fiscal documents. Production is blocked by design:**
4//! [`Ambiente`] has no production variant and no code path here writes `tpAmb=1`. An emitter only
5//! exists with [`BetaAck::i_understand_this_is_experimental`], and every document carries a
6//! sentence in `infCpl` identifying it as a beta. Nothing about this crate has been validated
7//! against a real SEFAZ server. See `REGRAS.md` for the origin of each rule and what has no source.
8//!
9//! NF-e and NFC-e with no GUI: draft, totals, access key, deterministic XML, inclusive C14N,
10//! XMLDSig, SOAP 1.2 and the reading of the returns. Limits of this POC: only Simples Nacional
11//! (CRT 1, CSOSN 102 and 500), only synchronous authorization, models 55 and 65, no voiding of
12//! number ranges, no contingency, no DANFE. The sandbox also explains failures and reproduces
13//! field problems offline: see [`sandbox`].
14//!
15//! Everything that leaves goes through [`Transport`]. [`sandbox::SefazSandbox`] replaces it with a
16//! local SEFAZ that really checks what arrives; the emitter does not know the difference.
17//!
18//! ```
19//! use std::str::FromStr;
20//! use ravenrs_nfe::sandbox::{SandboxSigner, SefazSandbox};
21//! use ravenrs_nfe::{
22//!     Ambiente, BetaAck, Csosn, Decimal, Emissor, Emitente, Endereco, FormaPagamento, Item,
23//!     Modelo, Rascunho, Situacao, Uf,
24//! };
25//!
26//! # fn main() -> Result<(), ravenrs_nfe::NfeError> {
27//! let emitente = Emitente {
28//!     cnpj: "11222333000181".into(),
29//!     razao_social: "MERCADO DE TESTE LTDA".into(),
30//!     nome_fantasia: None,
31//!     inscricao_estadual: "123456789012".into(),
32//!     endereco: Endereco {
33//!         logradouro: "RUA DAS FLORES".into(),
34//!         numero: "100".into(),
35//!         bairro: "CENTRO".into(),
36//!         codigo_municipio: "3550308".into(),
37//!         municipio: "SAO PAULO".into(),
38//!         uf: Uf::SP,
39//!         cep: "01001000".into(),
40//!     },
41//! };
42//! let server = SefazSandbox::new();
43//! let emissor = Emissor::builder(BetaAck::i_understand_this_is_experimental())
44//!     .ambiente(Ambiente::Sandbox)
45//!     .emitente(emitente)
46//!     .signer(SandboxSigner::new()?)
47//!     .transport(server.clone())
48//!     .build()?;
49//!
50//! let preco = Decimal::from_str("7.90").expect("decimal");
51//! let rascunho = Rascunho::novo(Modelo::Nfce65, 1, 1, "VENDA", "2026-11-05T14:30:00-03:00")
52//!     .com_item(Item {
53//!         codigo: "1".into(),
54//!         gtin: None,
55//!         descricao: "ARROZ".into(),
56//!         ncm: "10063021".into(),
57//!         cest: None,
58//!         cfop: "5102".into(),
59//!         unidade: "UN".into(),
60//!         quantidade: Decimal::ONE,
61//!         valor_unitario: preco,
62//!         desconto: Decimal::ZERO,
63//!         csosn: Csosn::S102,
64//!         origem: 0,
65//!     })
66//!     .com_pagamento(FormaPagamento::Dinheiro, preco);
67//!
68//! let emissao = emissor.emitir(&rascunho)?;
69//! assert_eq!(emissao.autorizacao.situacao(), Situacao::Autorizada);
70//! assert!(emissao.autorizacao.x_motivo().starts_with("[SANDBOX] "));
71//! # Ok(())
72//! # }
73//! ```
