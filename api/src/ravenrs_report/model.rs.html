1//! The report definition: page, bands, groups, controls and styles. It is plain
2//! data (serde), saved as JSON (`*.rvreport.json`) and built in code with the
3//! fluent methods below. Units are points (1/72 inch).
4
5use std::collections::BTreeMap;

9/// A banded report definition.
10///
11/// ```
12/// use ravenrs_report::{Band, Control, Report};
13///
14/// let report = Report::new("Customers")
15///     .page_header(Band::new(24.0).control(Control::label("Name").at(0.0, 0.0, 200.0, 20.0)))
16///     .detail(Band::new(20.0).control(Control::field("[Name]").at(0.0, 0.0, 200.0, 20.0)));
17/// let json = report.to_json();
18/// assert_eq!(ravenrs_report::Report::from_json(&json).expect("round trip"), report);
19/// ```
20#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
21#[serde(rename_all = "camelCase", default)]
22pub struct Report {
23    /// The report's name, also the document title.
24    pub name: String,
25    /// Paper size and margins.
26    pub page: PageSettings,
27    /// Detail rows side by side, as on a sheet of labels.
28    pub columns: Option<Columns>,
29    /// One long page as tall as what it prints, as on a roll of receipt paper. The height of
30    /// [`page`](Self::page) is the most it can reach.
31    pub continuous: bool,
32    /// The printer it goes to directly, when it does.
33    pub printer: Option<PrinterSettings>,
34    /// Culture for numbers and dates, such as `pt-BR` or `en-US`.
35    pub culture: String,
36    /// Where the data comes from, saved with the layout. The engine renders
37    /// whatever data it is given; designers and the report server load it from here.
38    pub data_sources: Vec<DataSource>,
39    /// The data source that feeds the report (the first one when unset).
40    pub data_source: Option<String>,
41    /// Path to the rows in the data (`orders`, `report.lines`). Without it, an
42    /// array of data is the rows and an object is a single row.
43    pub data_member: Option<String>,
44    /// Rows for which this expression is false are left out.
45    pub filter: Option<String>,
46    /// Sorting of detail rows within their groups.
47    pub sort: Vec<SortField>,
48    /// Values the user supplies when running the report (`?Name` in expressions).
49    pub parameters: Vec<Parameter>,
50    /// Fields computed from other fields, usable like data fields.
51    pub calculated_fields: Vec<CalculatedField>,
52    /// Named styles that controls refer to.
53    pub styles: BTreeMap<String, Style>,
54    /// Printed once, at the top of the first page.
55    pub report_header: Option<Band>,
56    /// Printed at the top of every page.
57    pub page_header: Option<Band>,
58    /// Groups, outermost first.
59    pub groups: Vec<Group>,
60    /// Printed once per row.
61    pub detail: Band,
62    /// Nested reports printed after each row's detail (master-detail).
63    pub detail_reports: Vec<DetailReport>,
64    /// Printed once, after the last row.
65    pub report_footer: Option<Band>,
66    /// Printed at the bottom of every page.
67    pub page_footer: Option<Band>,
68}

70/// Which way rows fill the columns of a page.
71#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
72#[serde(rename_all = "camelCase")]
73#[non_exhaustive]
74pub enum ColumnFlow {
75    /// Left to right, then the next line: labels on a sheet.
76    #[default]
77    AcrossThenDown,
78    /// Top to bottom, then the next column: a directory.
79    DownThenAcross,
80}

82/// Detail rows side by side. Only the detail band flows; headers and footers keep the whole width.
83#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
84#[serde(rename_all = "camelCase", default)]
85pub struct Columns {
86    /// How many fit across.
87    pub count: u32,
88    /// Between one column and the next.
89    pub spacing: f32,
90    /// Between one line of rows and the next.
91    pub row_spacing: f32,
92    /// Which way they fill.
93    pub flow: ColumnFlow,
94}
95
96impl Default for Columns {
97    fn default() -> Self {
98        Self { count: 2, spacing: 0.0, row_spacing: 0.0, flow: ColumnFlow::default() }
99    }
100}
101
102impl Columns {
103    /// This many, with no room between them.
104    pub fn new(count: u32) -> Self {
105        Self { count, ..Self::default() }
106    }
107
108    /// With this much room between columns and between lines.
109    pub fn spaced(self, spacing: f32, row_spacing: f32) -> Self {
110        Self { spacing, row_spacing, ..self }
111    }
112
113    /// Filling in this direction.
114    pub fn flow(self, flow: ColumnFlow) -> Self {
115        Self { flow, ..self }
116    }
117
118    /// The width of each column, in the width between the margins.
119    pub fn width_in(&self, content_width: f32) -> f32 {
120        let count = self.count.max(1) as f32;
121        ((content_width - self.spacing * (count - 1.0)) / count).max(0.0)
122    }
123}

136/// How a data source obtains its data.

138#[serde(tag = "type", rename_all = "camelCase")]
139#[non_exhaustive]

142    Json {
143        /// The data.

160    },
161}

169/// Paper size and margins, in points.
170#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
171#[serde(rename_all = "camelCase", default)]
172pub struct PageSettings {
173    /// Paper width.
174    pub width: f32,
175    /// Paper height.
176    pub height: f32,
177    /// Top margin.
178    pub margin_top: f32,
179    /// Right margin.
180    pub margin_right: f32,
181    /// Bottom margin.
182    pub margin_bottom: f32,
183    /// Left margin.
184    pub margin_left: f32,
185}
186
187/// Points in this many millimeters.
188pub fn mm(millimeters: f32) -> f32 {
189    millimeters * 72.0 / 25.4
190}
191
192impl Default for PageSettings {
193    fn default() -> Self {
194        Self::a4()
195    }
196}

198impl PageSettings {
199    /// A4 portrait (210 × 297 mm) with 1.5 cm margins.
200    pub fn a4() -> Self {
201        Self {
202            width: 595.28,
203            height: 841.89,
204            margin_top: 42.52,
205            margin_right: 42.52,
206            margin_bottom: 42.52,
207            margin_left: 42.52,
208        }
209    }
210
211    /// US Letter portrait (8.5 × 11 in) with 0.5 in margins.
212    pub fn letter() -> Self {
213        Self {
214            width: 612.0,
215            height: 792.0,
216            margin_top: 36.0,
217            margin_right: 36.0,
218            margin_bottom: 36.0,
219            margin_left: 36.0,
220        }
221    }
222
223    /// A label of this size, in millimeters, with no margins.
224    pub fn label_mm(width: f32, height: f32) -> Self {
225        Self {
226            width: mm(width),
227            height: mm(height),
228            margin_top: 0.0,
229            margin_right: 0.0,
230            margin_bottom: 0.0,
231            margin_left: 0.0,
232        }
233    }
234
235    /// The same paper turned sideways.
236    pub fn landscape(self) -> Self {
237        Self { width: self.height.max(self.width), height: self.width.min(self.height), ..self }
238    }
239
240    /// Sets all four margins.
241    pub fn margins(self, m: f32) -> Self {
242        Self { margin_top: m, margin_right: m, margin_bottom: m, margin_left: m, ..self }
243    }
244
245    /// The width between the side margins, where bands are laid out.
246    pub fn content_width(&self) -> f32 {
247        (self.width - self.margin_left - self.margin_right).max(0.0)
248    }
249}
