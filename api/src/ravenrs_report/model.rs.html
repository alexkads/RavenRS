1//! The report definition: page, bands, groups, controls and styles. It is plain
2//! data (serde), saved as JSON (`*.rvreport.json`) and built in code with the
3//! fluent methods below. Units are points (1/72 inch).
4
5use std::collections::BTreeMap;

7use serde::{Deserialize, Serialize};

9/// A banded report definition.
10///
11/// ```
12/// use ravenrs_report::{Band, Control, Report};
13///
14/// let report = Report::new("Customers")
15///     .page_header(Band::new(24.0).control(Control::label("Name").at(0.0, 0.0, 200.0, 20.0)))
16///     .detail(Band::new(20.0).control(Control::field("[Name]").at(0.0, 0.0, 200.0, 20.0)));
17/// let json = report.to_json();
18/// assert_eq!(ravenrs_report::Report::from_json(&json).expect("round trip"), report);
19/// ```
20#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
21#[serde(rename_all = "camelCase", default)]
22pub struct Report {
23    /// The report's name, also the document title.
24    pub name: String,
25    /// Paper size and margins.
26    pub page: PageSettings,
27    /// Culture for numbers and dates, such as `pt-BR` or `en-US`.
28    pub culture: String,
29    /// Where the data comes from, saved with the layout. The engine renders
30    /// whatever data it is given; designers and the report server load it from here.
31    pub data_sources: Vec<DataSource>,
32    /// The data source that feeds the report (the first one when unset).
33    pub data_source: Option<String>,
34    /// Path to the rows in the data (`orders`, `report.lines`). Without it, an
35    /// array of data is the rows and an object is a single row.
36    pub data_member: Option<String>,
37    /// Rows for which this expression is false are left out.
38    pub filter: Option<String>,
39    /// Sorting of detail rows within their groups.
40    pub sort: Vec<SortField>,
41    /// Values the user supplies when running the report (`?Name` in expressions).
42    pub parameters: Vec<Parameter>,
43    /// Fields computed from other fields, usable like data fields.
44    pub calculated_fields: Vec<CalculatedField>,
45    /// Named styles that controls refer to.
46    pub styles: BTreeMap<String, Style>,
47    /// Printed once, at the top of the first page.
48    pub report_header: Option<Band>,
49    /// Printed at the top of every page.
50    pub page_header: Option<Band>,
51    /// Groups, outermost first.
52    pub groups: Vec<Group>,
53    /// Printed once per row.
54    pub detail: Band,
55    /// Nested reports printed after each row's detail (master-detail).
56    pub detail_reports: Vec<DetailReport>,
57    /// Printed once, after the last row.
58    pub report_footer: Option<Band>,
59    /// Printed at the bottom of every page.
60    pub page_footer: Option<Band>,
61}

63/// A named source of report data, stored with the layout.

65#[serde(rename_all = "camelCase", default)]
66pub struct DataSource {

71    pub kind: DataSourceKind,

74/// How a data source obtains its data.
75#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
76#[serde(tag = "type", rename_all = "camelCase")]
77#[non_exhaustive]
78pub enum DataSourceKind {
79    /// Data stored in the layout itself (samples, small lookup tables).
80    Json {
81        /// The data.
82        data: serde_json::Value,
83    },
84    /// JSON fetched over HTTP(S) when the report runs.
85    Url {
86        /// The address; `{?Name}` inserts a parameter value.
87        url: String,
88        /// Extra request headers. Secrets belong in the server's configuration, not here.
89        #[serde(default)]
90        headers: BTreeMap<String, String>,
91    },
92    /// A SQL query run by the report server against a named connection.
93    Sql {
94        /// A connection name from the server's configuration (never a connection string with a password).
95        connection: String,
96        /// The query; `@Name` binds a report parameter.
97        query: String,
98    },
99}
100
101impl Default for DataSourceKind {
102    fn default() -> Self {
103        Self::Json { data: serde_json::Value::Null }
104    }
105}

107/// Paper size and margins, in points.
108#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
109#[serde(rename_all = "camelCase", default)]
110pub struct PageSettings {
111    /// Paper width.
112    pub width: f32,
113    /// Paper height.
114    pub height: f32,
115    /// Top margin.
116    pub margin_top: f32,
117    /// Right margin.
118    pub margin_right: f32,
119    /// Bottom margin.
120    pub margin_bottom: f32,
121    /// Left margin.
122    pub margin_left: f32,
123}
124
125impl Default for PageSettings {
126    fn default() -> Self {
127        Self::a4()
128    }
129}

131impl PageSettings {
132    /// A4 portrait (210 × 297 mm) with 1.5 cm margins.
133    pub fn a4() -> Self {
134        Self {
135            width: 595.28,
136            height: 841.89,
137            margin_top: 42.52,
138            margin_right: 42.52,
139            margin_bottom: 42.52,
140            margin_left: 42.52,
141        }
142    }
143
144    /// US Letter portrait (8.5 × 11 in) with 0.5 in margins.
145    pub fn letter() -> Self {
146        Self {
147            width: 612.0,
148            height: 792.0,
149            margin_top: 36.0,
150            margin_right: 36.0,
151            margin_bottom: 36.0,
152            margin_left: 36.0,
153        }
154    }
155
156    /// The same paper turned sideways.
157    pub fn landscape(self) -> Self {
158        Self { width: self.height.max(self.width), height: self.width.min(self.height), ..self }
159    }
160
161    /// Sets all four margins.
162    pub fn margins(self, m: f32) -> Self {
163        Self { margin_top: m, margin_right: m, margin_bottom: m, margin_left: m, ..self }
164    }
165
166    /// The width between the side margins, where bands are laid out.
167    pub fn content_width(&self) -> f32 {
168        (self.width - self.margin_left - self.margin_right).max(0.0)
169    }
170}

639#[serde(rename_all = "camelCase", default)]
640pub struct Style {
641    /// Font size in points.
642    pub font_size: Option<f32>,
643    /// Bold text.
644    pub bold: Option<bool>,
645    /// Text color, `#rrggbb` or `#rrggbbaa`.
646    pub color: Option<String>,
647    /// Fill color.
648    pub background: Option<String>,
649    /// Border color.
650    pub border_color: Option<String>,
651    /// Border width in points; 0 for none.
652    pub border_width: Option<f32>,
653    /// Which sides have a border: any of `top`, `right`, `bottom`, `left`, or `all`.
654    pub borders: Option<String>,
655    /// Inner padding in points.
656    pub padding: Option<f32>,
657    /// Horizontal alignment.
658    pub text_align: Option<TextAlign>,
659    /// Vertical alignment.
660    pub vertical_align: Option<VerticalAlign>,
661    /// Wraps long text onto more lines (true by default).
662    pub word_wrap: Option<bool>,
663}

665impl Style {

668        Self::default()
669    }
670
671    /// Sets the font size.

673        self.font_size = Some(size);
674        self
675    }
676
677    /// Makes the text bold.
678    pub fn bold(mut self) -> Self {
679        self.bold = Some(true);
680        self
681    }

683    /// Sets the text color (`#rrggbb`).
684    pub fn color(mut self, color: &str) -> Self {
