1//! Labels: the sizes they come in, and reports laid out for them.
2//!
3//! A label is a report like any other: each row is a label, in the detail band. A roll has one
4//! label per page; a sheet flows them in columns. So the designer, the preview and every export
5//! work on them, and what is specific to label printers stays out of the engine.
6
7use crate::model::{Band, Columns, Control, PageSettings, Report, Style, TextAlign, mm};
8
9/// A size labels come in.
10#[derive(Clone, Copy, Debug, PartialEq)]
11pub struct LabelStock {
12    /// What it is called.
13    pub name: &'static str,
14    /// The paper: one label for a roll, a whole sheet otherwise.
15    pub page: PageSettings,
16    /// How they sit on the paper. A roll has none.
17    pub columns: Option<Columns>,
18    /// The width of each label, in points.
19    pub width: f32,
20    /// The height of each label, in points.
21    pub height: f32,
22    /// A roll with no gaps, where a page is as long as what it prints.
23    pub continuous: bool,
24}
25
26impl LabelStock {
27    /// A roll of receipt paper, with this much of its width within reach of the printer. Rows take
28    /// the height they need, so its detail band starts small.
29    pub fn receipt(name: &'static str, printable_width: f32) -> Self {
30        Self {
31            name,
32            page: receipt_page(printable_width, 0.0),
33            columns: None,
34            width: mm(printable_width),
35            height: 14.0,
36            continuous: true,
37        }
38    }
39
40    /// One label after another on a roll, of this size in millimeters.
41    pub fn roll(name: &'static str, width: f32, height: f32) -> Self {
42        Self {
43            name,
44            page: PageSettings::label_mm(width, height),
45            columns: None,
46            width: mm(width),
47            height: mm(height),
48            continuous: false,
49        }
50    }
51
52    /// A sheet of A4 with `across` labels on each line, laid out as `sheet` says.
53    pub fn a4_sheet(name: &'static str, across: u32, sheet: SheetLayout) -> Self {
54        Self {
55            name,
56            page: PageSettings {
57                margin_top: mm(sheet.top),
58                margin_bottom: mm(sheet.top),
59                margin_left: mm(sheet.side),
60                margin_right: mm(sheet.side),
61                ..PageSettings::a4()
62            },
63            columns: Some(Columns::new(across).spaced(mm(sheet.gap_across), mm(sheet.gap_down))),
64            width: mm(sheet.width),
65            height: mm(sheet.height),
66            continuous: false,
67        }
68    }
69
70    /// A report with no content yet, its paper and detail band set for this label.
71    pub fn report(&self, name: impl Into<String>) -> Report {
72        let mut report = Report::new(name).page(self.page);
73        report.columns = self.columns;
74        report.continuous = self.continuous;
75        report.detail = Band::new(self.height);
76        report
77    }
78}

80/// A roll of receipt paper, in millimeters of what the printer can reach: 72 on the 80 mm roll, 48 on
81/// the 58 mm one. A report on it is as long as what it prints, up to about two meters.
82pub fn receipt_page(printable_width: f32, margin: f32) -> PageSettings {
83    PageSettings {
84        width: mm(printable_width),
85        height: mm(2000.0),
86        margin_top: mm(margin),
87        margin_bottom: mm(margin),
88        margin_left: mm(margin),
89        margin_right: mm(margin),
90    }
91}

93/// How labels sit on a sheet, in millimeters.
94#[derive(Clone, Copy, Debug, Default, PartialEq)]
95pub struct SheetLayout {
96    /// Each label's width.
97    pub width: f32,
98    /// Each label's height.
99    pub height: f32,
100    /// The margin above and below.
101    pub top: f32,
102    /// The margin on the sides.
103    pub side: f32,
104    /// Between labels across.
105    pub gap_across: f32,
106    /// Between labels down.
107    pub gap_down: f32,
108}

131/// The expressions a shelf label shows, field by field.
132#[derive(Clone, Debug, PartialEq)]
133pub struct ShelfLabelFields {
134    /// What the product is.
135    pub description: String,
136    /// Its price.
137    pub price: String,
138    /// The format of the price.
139    pub price_format: String,
140    /// Its GTIN or other code.
141    pub barcode: String,
142    /// The format of the code.
143    pub symbology: crate::barcode::Symbology,
144}
145
146/// A shelf label for this stock: description on top, the price large, the code below. It is a
147/// starting point, to change in the designer.
148pub fn shelf_label(stock: &LabelStock, name: impl Into<String>, fields: &ShelfLabelFields) -> Report {
149    let (width, height) = (stock.width, stock.height);
150    let inset = (width * 0.04).min(mm(2.0));
151    let inner = width - 2.0 * inset;
152    let description = height * 0.26;
153    let price = height * 0.30;
154    let code = height - description - price - 2.0 * inset;
155    let mut report = stock.report(name);
156    report.detail = Band::new(height)
157        .control(
158            Control::field(fields.description.clone())
159                .at(inset, inset, inner, description)
160                .fixed_height()
161                .style(Style::new().font_size((description * 0.42).clamp(6.0, 14.0)).bold()),
162        )
163        .control(
164            Control::field(fields.price.clone())
165                .at(inset, inset + description, inner, price)
166                .fixed_height()
167                .format(&fields.price_format)
168                .style(
169                    Style::new().font_size((price * 0.72).clamp(9.0, 40.0)).bold().align(TextAlign::Right),
170                ),
171        )
172        .control(
173            Control::barcode(fields.symbology, fields.barcode.clone())
174                .at(inset, inset + description + price, inner, code)
175                .style(Style::new().font_size((code * 0.2).clamp(5.0, 9.0))),
176        );
177    report
178}

180#[cfg(test)]
181mod tests {
182    use super::*;
183
184    #[test]
185    fn every_stock_fits_its_paper() {
186        for stock in common_stocks() {
187            let across = stock.columns.map_or(1, |columns| columns.count) as f32;
188            let spacing = stock.columns.map_or(0.0, |columns| columns.spacing);
189            let used = across * stock.width + (across - 1.0) * spacing;
190            assert!(used <= stock.page.content_width() + 0.5, "{}: {used}", stock.name);
191            assert!(stock.height <= stock.page.height, "{}", stock.name);
192        }
193    }
194
195    #[test]
196    fn a_label_report_keeps_its_stock_through_json() {
197        let stock = common_stocks()[3];
198        let report = stock.report("Etiqueta");
199        let json = report.to_json();
200        let back = Report::from_json(&json).unwrap_or_else(|e| panic!("{e}"));
201        assert_eq!(back.columns, stock.columns);
202        assert_eq!(back.page, stock.page);
203    }
204
205    #[test]
206    fn a_long_description_does_not_push_the_code_off_the_label() {
207        use crate::{Item, ReportEngine};
208        let stock = common_stocks()[1];
209        let report = shelf_label(
210            &stock,
211            "Etiqueta",
212            &ShelfLabelFields {
213                description: "[Name]".into(),
214                price: "[Price]".into(),
215                price_format: "C2".into(),
216                barcode: "[Code]".into(),
217                symbology: crate::barcode::Symbology::Ean13,
218            },
219        );
220        let long = "Sabão em pó concentrado para roupas brancas e coloridas, pacote econômico de 5 kg";
221        let data = serde_json::json!([{ "Name": long, "Price": 5.49, "Code": "7891000100103" }]);
222        let document = ReportEngine::new().render(&report, &data);
223        assert_eq!(document.pages.len(), 1);
224        let bars = document.pages[0]
225            .items
226            .iter()
227            .find_map(|item| match item {
228                Item::BarcodeMark { rect, .. } => Some(*rect),
229                _ => None,
230            })
231            .unwrap_or_else(|| panic!("no code: {:?}", document.errors));
232        assert!(bars.max_y() <= stock.height + 0.5, "{bars:?} on a label of {}", stock.height);
233    }
