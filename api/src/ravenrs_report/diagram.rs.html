1//! A flow diagram fitted into a box: the shapes of its picture moved and scaled, and where each
2//! of its words goes. The report engine prints it as a control, and a diagram exported alone as
3//! a PDF is laid out the same way.
4
5use ravenrs_core::{Color, Point, Rect};
6use ravenrs_graphics::flow::{FlowPicture, FlowText, dashes, flatten};
7use ravenrs_graphics::vector::{Layer, PathSegment, Shape};
8
9/// One run of words of a fitted diagram.
10#[derive(Clone, Debug, PartialEq)]
11pub struct DiagramLabel {
12    /// The words.
13    pub text: String,
14    /// The box they are centred in, vertically, and start at or centred in, across.
15    pub rect: Rect,
16    /// Font size, in points.
17    pub size: f32,
18    /// Semibold.
19    pub bold: bool,
20    /// Centred across rather than from the left.
21    pub centered: bool,
22    /// Their color.
23    pub color: Color,
24}
25
26/// A diagram fitted into a box.
27#[derive(Clone, Debug, Default, PartialEq)]
28pub struct FittedDiagram {
29    /// The shapes, back to front; the picture's background is left out, so the box shows its own,
30    /// and dashed lines come as one short line per dash.
31    pub shapes: Vec<Shape>,
32    /// The words, over the shapes.
33    pub labels: Vec<DiagramLabel>,
34}
35
36/// Words smaller than this, in points, are left out: they could not be read.
37pub const SMALLEST_WORDS: f32 = 2.0;
38
39/// `picture` scaled to fit inside `rect` whole, keeping its proportions, and centred there.
40pub fn fit(picture: &FlowPicture, rect: Rect) -> FittedDiagram {
41    let b = picture.bounds;
42    if b.width() <= 0.0 || b.height() <= 0.0 {
43        return FittedDiagram::default();
44    }
45    let scale = (rect.width() / b.width()).min(rect.height() / b.height()).max(0.0);
46    let (ox, oy) = (
47        rect.x() + (rect.width() - b.width() * scale) / 2.0 - b.x() * scale,
48        rect.y() + (rect.height() - b.height() * scale) / 2.0 - b.y() * scale,
49    );
50    let at = |p: Point| Point::new(ox + p.x * scale, oy + p.y * scale);
51    let moved = |path: &[PathSegment]| -> Vec<PathSegment> {
52        path.iter()
53            .map(|s| match *s {
54                PathSegment::Move(p) => PathSegment::Move(at(p)),
55                PathSegment::Line(p) => PathSegment::Line(at(p)),
56                PathSegment::Cubic(a, b, c) => PathSegment::Cubic(at(a), at(b), at(c)),
57                other => other,
58            })
59            .collect()
60    };
61    let mut shapes = Vec::new();
62    for layer in picture.layers.iter().skip(1) {
63        match layer {
64            // A dashed line is cut into its dashes, each a short line of its own, so it prints
65            // dashed in every output, which know only whole shapes.
66            Layer::Dashed { path, width, dash, color } => {
67                let points: Vec<Point> = flatten(path).into_iter().map(at).collect();
68                let step = (dash * scale).max(0.5);
69                for piece in dashes(&points, step, step, 0.0) {
70                    let line: Vec<PathSegment> = piece
71                        .iter()
72                        .enumerate()
73                        .map(|(i, p)| if i == 0 { PathSegment::Move(*p) } else { PathSegment::Line(*p) })
74                        .collect();
75                    shapes.push(Shape {
76                        path: line,
77                        fill: None,
78                        stroke: Some(((width * scale).max(0.25), *color)),
79                    });
80                }
81            }
82            other => {
83                let shape = other.to_shape();
84                shapes.push(Shape {
85                    path: moved(&shape.path),
86                    fill: shape.fill,
87                    stroke: shape.stroke.map(|(w, c)| ((w * scale).max(0.25), c)),
88                });
89            }
90        }
91    }
92    let labels = picture
93        .texts
94        .iter()
95        .filter(|t| t.size * scale >= SMALLEST_WORDS)
96        .map(|t: &FlowText| {
97            let (size, width) = (t.size * scale, t.width * scale);
98            let height = size * 1.4;
99            let p = at(t.at);
100            let left = if t.centered { p.x - width / 2.0 } else { p.x };
101            DiagramLabel {
102                text: t.text.clone(),
103                rect: Rect::new(left, p.y - height / 2.0, width, height),
104                size,
105                bold: t.bold,
106                centered: t.centered,
107                color: t.color,
108            }
109        })
110        .collect();
111    FittedDiagram { shapes, labels }
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117    use ravenrs_graphics::flow::{FlowColors, FlowGraph, FlowNode};
118
119    #[test]
120    fn a_diagram_fits_its_box_keeping_its_proportions() {
121        let g = FlowGraph::new()
122            .node(FlowNode::new("a", "", "A", 0.0, 0.0))
123            .node(FlowNode::new("b", "", "B", 600.0, 0.0));
124        let picture = g.picture(None, FlowColors::paper()).expect("a picture");
125        let rect = Rect::new(10.0, 20.0, 200.0, 200.0);
126        let fitted = fit(&picture, rect);
127        let points: Vec<Point> = fitted
128            .shapes
129            .iter()
130            .flat_map(|s| s.path.iter())
131            .filter_map(|s| match s {
132                PathSegment::Move(p) | PathSegment::Line(p) => Some(*p),
133                _ => None,
134            })
135            .collect();
136        for p in &points {
137            assert!(rect.contains(*p) || (p.x - rect.max_x()).abs() < 0.01, "{p:?} is inside");
138        }
139        // Wider than tall: the width fills the box, and the drawing is centred down it.
140        let (least, most) = points.iter().fold((f32::MAX, f32::MIN), |(a, b), p| (a.min(p.y), b.max(p.y)));
141        assert!((least - rect.y()) > 10.0 && (rect.max_y() - most) > 10.0);
142        assert_eq!(fitted.labels.len(), 2, "both titles");
143    }
144
145    #[test]
146    fn a_dashed_line_comes_as_its_dashes() {
147        use ravenrs_graphics::flow::{FlowEdge, FlowEndpoint, FlowPort};
148        let g = FlowGraph::new()
149            .node(FlowNode::new("a", "", "A", 0.0, 0.0).port(FlowPort::output("o")))
150            .node(FlowNode::new("b", "", "B", 600.0, 0.0).port(FlowPort::input("i")))
151            .edge(FlowEdge::new("e", FlowEndpoint::new("a", "o"), FlowEndpoint::new("b", "i")).dashed());
152        let picture = g.picture(None, FlowColors::paper()).expect("a picture");
153        let whole = fit(&picture, Rect::new(0.0, 0.0, 800.0, 200.0));
154        let solid = g.clone();
155        let mut solid = solid;
156        solid.edges[0].dashed = false;
157        let plain = fit(
158            &solid.picture(None, FlowColors::paper()).expect("a picture"),
159            Rect::new(0.0, 0.0, 800.0, 200.0),
160        );
161        assert!(whole.shapes.len() > plain.shapes.len() + 5, "many short dashes instead of one line");
162        let open = whole.shapes.iter().filter(|s| s.fill.is_none() && s.path.len() >= 2).count();
163        assert!(open > 5, "each dash a short open line");
164    }
