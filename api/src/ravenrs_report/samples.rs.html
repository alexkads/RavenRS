19/// A small deterministic random generator, so samples are the same everywhere.
20struct Rng(u64);
21
22impl Rng {
23    fn next(&mut self) -> u64 {
24        self.0 ^= self.0 << 13;
25        self.0 ^= self.0 >> 7;
26        self.0 ^= self.0 << 17;
27        self.0
28    }
29
30    fn pick<'a>(&mut self, items: &[&'a str]) -> &'a str {
31        items[(self.next() % items.len() as u64) as usize]
32    }
33
34    fn range(&mut self, lo: u64, hi: u64) -> u64 {
35        lo + self.next() % (hi - lo + 1)
36    }
37}

72/// `count` sales rows: date, region, city, customer, product, quantity, price.
73pub fn sales_data(count: usize) -> Json {
74    let mut rng = Rng(0x5eed_1234_abcd_9876);
75    let rows: Vec<Json> = (0..count)
76        .map(|i| {
77            let (city, region) = CITIES[(rng.next() % CITIES.len() as u64) as usize];
78            let (product, price) = PRODUCTS[(rng.next() % PRODUCTS.len() as u64) as usize];
79            let month = rng.range(1, 12);
80            let day = rng.range(1, 28);
81            json!({
82                "Id": 10_000 + i,
83                "Date": format!("2026-{month:02}-{day:02}"),
84                "Region": region,
85                "City": city,
86                "Customer": format!("{} {}", rng.pick(&FIRST), rng.pick(&LAST)),
87                "Product": product,
88                "Quantity": rng.range(1, 12),
89                "UnitPrice": price,
90                "Paid": !rng.next().is_multiple_of(5),
91            })
92        })
93        .collect();
94    Json::Array(rows)
95}
