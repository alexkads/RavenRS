1//! Lints for applications built with [RavenRS](https://alexkads.github.io/RavenRS/).
2//!
3//! The compiler and Clippy know Rust, not the rules an interface must follow to stay right in every
4//! theme, for screen readers and across rebuilds. `ravenrs-lint` reads the syntax tree of an
5//! application's sources and reports those rules the way the compiler does:
6//!
7//! ```text
8//! error[ravenrs::raw_color]: literal color in the interface
9//!   --> src/ui.rs:12:21
10//!    |
11//! 12 |         .background(Color::rgb(0.1, 0.2, 0.3))
12//!    |                     ^^^^^^^^^^
13//!    |
14//!    = help: use a `Semantic` role, such as `.background(Semantic::Control)`, so it stays right in every theme
15//! ```
16//!
17//! # Running it
18//!
19//! From the command line, after `cargo install ravenrs-lint`:
20//!
21//! ```text
22//! cargo ravenrs-lint            # every package of the workspace that depends on RavenRS
23//! cargo ravenrs-lint --init     # recommended rustc and Clippy lints, the test below and AGENTS.md
24//! ```
25//!
26//! From `cargo test`, so nothing passes while a rule is broken, with `ravenrs-lint` as a
27//! dev-dependency and a file such as `tests/ravenrs_lint.rs`:
28//!
29//! ```no_run
30//! #[test]
31//! fn follows_the_ravenrs_rules() {
32//!     ravenrs_lint::assert_clean();
33//! }
34//! ```
35//!
36//! # Configuring it
37//!
38//! Levels follow Cargo's `[lints]`: `allow`, `warn` or `deny`. They go in the package's or the
39//! workspace's metadata, and the package wins:
40//!
41//! ```toml
42//! [package.metadata.ravenrs-lint]
43//! language = "pt-BR"            # how button captions are checked: "en" (default) or "pt-BR"
44//! extra-verbs = ["Ok"]          # first words accepted as verbs besides the built-in ones
45//!
46//! [package.metadata.ravenrs-lint.lints]
47//! off_grid_spacing = "deny"
48//! ```
49//!
50//! A single place is excused with a comment that says why, on the same line or on its own line
51//! right above:
52//!
53//! ```text
54//! // ravenrs-lint: allow(raw_color) the church's logo keeps its brand color in every theme
55//! ```
56//!
57//! `allow-file(...)` excuses the whole file. Test code (`#[test]`, `#[cfg(test)]`) is not checked.
58//!
59//! The rules are listed in [`Lint`].

77/// One broken rule at one place in a source file.
78#[derive(Clone, Debug, PartialEq)]
79pub struct Diagnostic {
80    /// The rule.
81    pub lint: Lint,
82    /// How it is reported, after the configuration.
83    pub level: Level,
84    /// The file, as given to the check.
85    pub file: PathBuf,
86    /// Line, from 1.
87    pub line: usize,
88    /// Column of the first character, from 1.
89    pub column: usize,
90    /// Column after the last character on the same line, from 1.
91    pub end_column: usize,
92    /// What is wrong.
93    pub message: String,
94    /// How to fix it.
95    pub help: String,
96    /// The text of the line, for showing it.
97    pub snippet: String,
98}

100/// Why a check could not run.
101#[derive(Debug)]
102#[non_exhaustive]
103pub enum Error {
104    /// A file or directory could not be read.
105    Io(PathBuf, std::io::Error),
106    /// A source file is not valid Rust.
107    Parse(PathBuf, syn::Error),
108    /// The configuration names something that does not exist.
109    Config(String),
110    /// `cargo metadata` failed or answered something unexpected.
111    Cargo(String),
112}
113
114impl fmt::Display for Error {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116        match self {
117            Error::Io(path, error) => write!(f, "could not read {}: {error}", path.display()),
118            Error::Parse(path, error) => {
119                let at = error.span().start();
120                write!(f, "{}:{}:{}: {error}", path.display(), at.line, at.column + 1)
121            }
122            Error::Config(message) => write!(f, "invalid ravenrs-lint configuration: {message}"),
123            Error::Cargo(message) => write!(f, "cargo metadata: {message}"),
124        }
125    }
126}
127
128impl std::error::Error for Error {}

130/// Checks one source file. `file` is only used in the diagnostics.
131pub fn check_source(file: &Path, source: &str, config: &Config) -> Result<Vec<Diagnostic>, Error> {
132    let syntax = syn::parse_file(source).map_err(|error| Error::Parse(file.to_owned(), error))?;
133    let mut found = rules::check(&syntax, config, source.lines().count());
134    found.extend(rules::check_allow_comments(source));
135    let allowed = rules::Allowances::read(source);
136    let found = found.into_iter().filter(|finding| !allowed.covers(finding.lint, finding.line));
137    let mut diagnostics = diagnostics_of(file, source, found, config);
138    diagnostics.sort_by_key(|d| (d.line, d.column));
139    Ok(diagnostics)
140}
141
142/// The findings that the configuration does not turn off, with the line they point at.
143fn diagnostics_of(
144    file: &Path,
145    source: &str,
146    found: impl IntoIterator<Item = rules::Finding>,
147    config: &Config,
148) -> Vec<Diagnostic> {
149    let lines: Vec<&str> = source.lines().collect();
150    found
151        .into_iter()
152        .filter_map(|finding| {
153            let level = config.level(finding.lint);
154            (level != Level::Allow).then(|| Diagnostic {
155                lint: finding.lint,
156                level,
157                file: file.to_owned(),
158                line: finding.line,
159                column: finding.column,
160                end_column: finding.end_column,
161                message: finding.message,
162                help: finding.help,
163                snippet: lines.get(finding.line.saturating_sub(1)).copied().unwrap_or_default().to_owned(),
164            })
165        })
166        .collect()
167}

169/// Checks how the top-level modules of the crate whose root is `root` (its `lib.rs` or
170/// `main.rs`) use each other: the `module_cycle` rule, which reads the whole crate. Paths in the
171/// diagnostics are relative to `base` when they are inside it.
172pub fn check_modules(root: &Path, base: &Path, config: &Config) -> Result<Vec<Diagnostic>, Error> {
173    if config.level(Lint::ModuleCycle) == Level::Allow {
174        return Ok(Vec::new());
175    }
176    let mut diagnostics = Vec::new();
177    for (path, finding) in coupling::ModuleGraph::read(root)?.findings(config) {
178        let source = std::fs::read_to_string(&path).map_err(|error| Error::Io(path.clone(), error))?;
179        let shown = path.strip_prefix(base).unwrap_or(&path);
180        diagnostics.extend(diagnostics_of(shown, &source, [finding], config));
181    }
182    Ok(diagnostics)
183}

185/// Checks every `.rs` file under `dir`, in a stable order. Paths in the diagnostics are relative
186/// to `base` when they are inside it.
187pub fn check_dir(dir: &Path, base: &Path, config: &Config) -> Result<Vec<Diagnostic>, Error> {
188    let mut files = Vec::new();
189    collect_rust_files(dir, &mut files)?;
190    files.sort();
191    let mut diagnostics = Vec::new();
192    for path in files {
193        let source = std::fs::read_to_string(&path).map_err(|error| Error::Io(path.clone(), error))?;
194        let shown = path.strip_prefix(base).unwrap_or(&path);
195        diagnostics.extend(check_source(shown, &source, config)?);
196    }
197    Ok(diagnostics)
198}
199
200fn collect_rust_files(dir: &Path, files: &mut Vec<PathBuf>) -> Result<(), Error> {
201    let entries = std::fs::read_dir(dir).map_err(|error| Error::Io(dir.to_owned(), error))?;
202    for entry in entries {
203        let path = entry.map_err(|error| Error::Io(dir.to_owned(), error))?.path();
204        if path.is_dir() {
205            collect_rust_files(&path, files)?;
206        } else if path.extension().is_some_and(|ext| ext == "rs") {
207            files.push(path);
208        }
209    }
210    Ok(())
211}

213/// How the top-level modules of the crate whose root is `root` use each other, as a table: each
214/// module's lines, how many modules it uses and how many use it, then the loops. For whoever is
215/// untangling a crate that `module_cycle` reports.
216pub fn module_report(root: &Path) -> Result<String, Error> {
217    Ok(coupling::ModuleGraph::read(root)?.report())
218}

220/// The roots of the crates in a package: `lib.rs` and `main.rs` in its source directories.
221pub fn crate_roots(package: &Package) -> Vec<PathBuf> {
222    package
223        .source_dirs
224        .iter()
225        .flat_map(|dir| ["lib.rs", "main.rs"].map(|name| dir.join(name)))
226        .filter(|path| path.is_file())
227        .collect()
228}

230/// Checks the sources of one package: the directories of its library and binaries.
231pub fn check_package(package: &Package) -> Result<Vec<Diagnostic>, Error> {
232    let config = &package.config;
233    // The file rules are skipped when all of them are off, as `--only module_cycle` leaves them.
234    let file_rules =
235        Lint::ALL.iter().any(|&lint| lint != Lint::ModuleCycle && config.level(lint) != Level::Allow);
236    let mut diagnostics = Vec::new();
237    if file_rules {
238        for dir in &package.source_dirs {
239            diagnostics.extend(check_dir(dir, &package.root, config)?);
240        }
241    }
242    for root in crate_roots(package) {
243        diagnostics.extend(check_modules(&root, &package.root, config)?);
244    }
245    Ok(diagnostics)
246}

248/// Checks the package whose tests are running and panics with every diagnostic, warnings
249/// included, the way `cargo clippy -- -D warnings` fails. Call it from a `#[test]`.
250///
251/// # Panics
252///
253/// When a rule is broken, and when the package cannot be read.
254#[allow(clippy::panic)]
255pub fn assert_clean() {
256    let Some(dir) = std::env::var_os("CARGO_MANIFEST_DIR") else {
257        panic!("ravenrs_lint::assert_clean runs from `cargo test`, which sets CARGO_MANIFEST_DIR");
258    };
259    let manifest = PathBuf::from(dir).join("Cargo.toml");
260    let result = packages(&manifest).and_then(|found| {
261        let same = |a: &Path, b: &Path| a == b || a.canonicalize().ok() == b.canonicalize().ok();
262        let package = found
263            .into_iter()
264            .find(|p| same(&p.manifest, &manifest))
265            .ok_or_else(|| Error::Cargo(format!("{} is not in its workspace", manifest.display())))?;
266        check_package(&package)
267    });
268    match result {
269        Ok(diagnostics) if diagnostics.is_empty() => {}
270        Ok(diagnostics) => panic!("\n{}", render(&diagnostics, true)),
271        Err(error) => panic!("{error}"),
272    }
