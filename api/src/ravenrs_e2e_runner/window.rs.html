1//! Running with the app's real window on screen (`--headed`, #729): the mode, the pace a person
2//! can follow and the window kept open to look at a failure.
3//!
4//! A run with no window is the default, for build servers and agents. With the window, the
5//! scenario, its blocks, their waits and their codes are the same; the app is opened with
6//! `--e2e-serve --e2e-window`, shows its window on its test data, and each step lands on the
7//! interface the window shows (`ravenrs_e2e_wire::Headed`). On top of that:
8//!
9//! - **slow-mo** waits a moment after each step, in the tool, while the app is idle and its
10//!   window keeps drawing. It is not a step and has no timeout, so it never makes a step fail;
11//!   the app's clock keeps running, as it does for a person.
12//! - **[`Control`]** pauses before the next block, runs exactly one more, or stops the run; the
13//!   window keeps drawing all along, since the app only waits for its next command.
14//! - **[`KeepOpen`]** leaves the window to the person at the end, to look at how the app was
15//!   left; the run waits until they close it, and no process is left behind.
16
17use std::sync::{Arc, Condvar, Mutex};
18use std::time::Duration;

20/// Whether the app shows its window during a run.
21#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
22pub enum WindowMode {
23    /// No window: the default, for build servers and agents.
24    #[default]
25    Headless,
26    /// The app's real window, on screen.
27    Headed,
28}
29
30impl WindowMode {
31    /// The words `e2e.toml` takes for `window`.
32    pub const WORDS: [&'static str; 2] = ["headless", "headed"];
33
34    /// The mode a word says, in English or in Portuguese.
35    pub fn read(word: &str) -> Option<Self> {
36        Some(match word {
37            "headless" | "sem-janela" => Self::Headless,
38            "headed" | "com-janela" => Self::Headed,
39            _ => return None,
40        })
41    }
42
43    /// The word for the mode, as reports write it.
44    pub fn word(self) -> &'static str {
45        match self {
46            Self::Headless => "headless",
47            Self::Headed => "headed",
48        }
49    }
50}

52/// When the window is left open at the end of a run, for the person to look at it.
53#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
54pub enum KeepOpen {
55    /// The window closes with the run: the default, so nothing waits on a person.
56    #[default]
57    Never,
58    /// The window stays when a block failed, showing the screen as the failure left it.
59    OnFailure,
60    /// The window always stays.
61    Always,
62}
63
64impl KeepOpen {
65    /// The words `--keep-open` and `e2e.toml` take.
66    pub const WORDS: [&'static str; 3] = ["never", "on-failure", "always"];
67
68    /// The choice a word says, in English or in Portuguese.
69    pub fn read(word: &str) -> Option<Self> {
70        Some(match word {
71            "never" | "nunca" => Self::Never,
72            "on-failure" | "na-falha" => Self::OnFailure,
73            "always" | "sempre" => Self::Always,
74            _ => return None,
75        })
76    }
77
78    /// Whether a run that `failed` leaves its window open.
79    pub fn keeps(self, failed: bool) -> bool {
80        match self {
81            Self::Never => false,
82            Self::OnFailure => failed,
83            Self::Always => true,
84        }
85    }
86}

88/// How a run shows the app: with or without its window, at what pace, and whether the window
89/// stays at the end.
90#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
91pub struct Window {
92    /// With or without the window.
93    pub mode: WindowMode,
94    /// How long to wait after each step, for a person to follow it; only with the window.
95    pub slow_mo: Duration,
96    /// Whether the window stays at the end; only with the window.
97    pub keep_open: KeepOpen,
98}
99
100impl Window {
101    /// Whether the app shows its window.
102    pub fn headed(&self) -> bool {
103        self.mode == WindowMode::Headed
104    }
105
106    /// Why these choices make no sense together, said to whoever wrote them: a pace or a window
107    /// kept open are only for a run with the window.
108    pub fn check(&self) -> Result<(), String> {
109        if self.headed() {
110            return Ok(());
111        }
112        if !self.slow_mo.is_zero() {
113            return Err("--slow-mo só vale com a janela: acrescente --headed".to_owned());
114        }
115        if self.keep_open != KeepOpen::Never {
116            return Err("--keep-open só vale com a janela: acrescente --headed".to_owned());
117        }
118        Ok(())
119    }
120}

122/// Whether this computer can show a window, or why not. Linux needs a graphical session
123/// (`DISPLAY` or `WAYLAND_DISPLAY`), which a server or an SSH session usually lacks; macOS and
124/// Windows are taken as able, and an app that still cannot open its window says why when it
125/// ends.
126pub fn graphic_session() -> Result<(), String> {
127    graphic_session_in(|name| std::env::var_os(name).is_some_and(|v| !v.is_empty()))
128}
129
130/// [`graphic_session`] with the environment read through `set`, for tests.
131pub fn graphic_session_in(set: impl Fn(&str) -> bool) -> Result<(), String> {
132    if cfg!(any(target_os = "macos", target_os = "windows")) || set("DISPLAY") || set("WAYLAND_DISPLAY") {
133        return Ok(());
134    }
135    Err("este computador não tem sessão gráfica (nem DISPLAY nem WAYLAND_DISPLAY): rode sem --headed, ou numa sessão com tela".to_owned())
136}

171/// What a person watching a run asked for, shared between the run and whoever reads them (a
172/// terminal, a window): pause, one more block, carry on, stop, and the pace.
173#[derive(Clone, Default)]
174pub struct Control(Arc<(Mutex<Pace>, Condvar)>);
175
176#[derive(Debug, Default)]
177struct Pace {
178    paused: bool,
179    /// Blocks allowed to run while paused.
180    steps: u32,
181    stopped: bool,
182    slow_mo: Duration,
183}
184
185impl std::fmt::Debug for Control {
186    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
187        f.debug_tuple("Control").field(&*self.lock()).finish()
188    }
189}
190
191impl PartialEq for Control {
192    fn eq(&self, other: &Self) -> bool {
193        Arc::ptr_eq(&self.0, &other.0)
194    }
195}
196
197impl Eq for Control {}
198
199impl Control {
200    /// Control of a run that waits `slow_mo` after each step and starts running.
201    pub fn new(slow_mo: Duration) -> Self {
202        let control = Self::default();
203        control.set_slow_mo(slow_mo);
204        control
205    }
206
207    fn lock(&self) -> std::sync::MutexGuard<'_, Pace> {
208        self.0.0.lock().unwrap_or_else(std::sync::PoisonError::into_inner)
209    }
210
211    fn changed(&self, change: impl FnOnce(&mut Pace)) {
212        change(&mut self.lock());
213        self.0.1.notify_all();
214    }
215
216    /// Pauses before the next block; the one running finishes.
217    pub fn pause(&self) {
218        self.changed(|p| p.paused = true);
219    }
220
221    /// Carries on with every block left.
222    pub fn resume(&self) {
223        self.changed(|p| {
224            p.paused = false;
225            p.steps = 0;
226        });
227    }
228
229    /// Runs exactly one more block, then stays paused.
230    pub fn next(&self) {
231        self.changed(|p| {
232            p.paused = true;
233            p.steps += 1;
234        });
235    }
236
237    /// Stops the run before the next block; the blocks left are reported as not run.
238    pub fn stop(&self) {
239        self.changed(|p| p.stopped = true);
240    }
241
242    /// Changes how long to wait after each step, even halfway through a run.
243    pub fn set_slow_mo(&self, slow_mo: Duration) {
244        self.changed(|p| p.slow_mo = slow_mo);
245    }
246
247    /// Whether the run is paused.
248    pub fn is_paused(&self) -> bool {
249        self.lock().paused
250    }
251
252    /// Whether the run was stopped.
253    pub fn is_stopped(&self) -> bool {
254        self.lock().stopped
255    }
256
257    /// Waits for the turn of the next block: `true` when it may run, `false` when the run was
258    /// stopped. While paused it waits for [`Self::next`], [`Self::resume`] or [`Self::stop`].
259    pub fn turn(&self) -> bool {
260        let (lock, changed) = &*self.0;
261        let mut pace = lock.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
262        loop {
263            if pace.stopped {
264                return false;
265            }
266            if !pace.paused {
267                return true;
268            }
269            if pace.steps > 0 {
270                pace.steps -= 1;
271                return true;
272            }
273            pace = changed.wait(pace).unwrap_or_else(std::sync::PoisonError::into_inner);
274        }
275    }
276
277    /// Waits the slow-mo after a step, cut short when the run is stopped or paused (a pause
278    /// already waits for the person).
279    pub fn present(&self) {
280        let (lock, changed) = &*self.0;
281        let pace = lock.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
282        let slow_mo = pace.slow_mo;
283        if slow_mo.is_zero() {
284            return;
285        }
286        let _ = changed.wait_timeout_while(pace, slow_mo, |p| !p.stopped && !p.paused);
287    }
288}

290/// What a person types in the terminal to steer a run with the window, one word per line, as
291/// `--step` reads them: Enter runs one more block, `c` carries on, `p` pauses, `q` stops.
292pub fn steer(control: &Control, line: &str) -> Option<&'static str> {
293    Some(match line.trim() {
294        "" | "n" | "proximo" | "próximo" => {
295            control.next();
296            "próximo passo"
297        }
298        "c" | "continuar" => {
299            control.resume();
300            "continuando"
301        }
302        "p" | "pausar" => {
303            control.pause();
304            "pausado"
305        }
306        "q" | "parar" => {
307            control.stop();
308            "parado"
309        }
310        _ => return None,
311    })
312}

314/// What the terminal says a person can type while a run with the window goes.
315pub const KEYS: &str = "Enter = próximo passo, c = continuar, p = pausar, q = parar";
316
317/// Lets the person at the terminal steer a run with the window, when standard input is a
318/// terminal (a pipe is never read): [`steer_from_terminal`], after saying how.
319pub fn steer_when_terminal(control: &Control) {
320    use std::io::IsTerminal as _;
321    if !std::io::stdin().is_terminal() {
322        return;
323    }
324    if control.is_paused() {
325        eprintln!("[e2e] pausado antes do primeiro passo: {KEYS}");
326    } else {
327        eprintln!("[e2e] com a janela do app: {KEYS}");
328    }
329    steer_from_terminal(control.clone());
330}

332/// Reads the terminal on a thread of its own and steers `control` with each line ([`steer`]),
333/// telling on standard error what each did, until the input ends.
334pub fn steer_from_terminal(control: Control) {
335    std::thread::spawn(move || {
336        use std::io::BufRead as _;
337        for line in std::io::stdin().lock().lines() {
338            let Ok(line) = line else { break };
339            match steer(&control, &line) {
340                Some(done) => eprintln!("[e2e] {done}"),
341                None => eprintln!("[e2e] {KEYS}"),
342            }
343        }
344    });
