1//! The report of a suite as one HTML page (T5 of #627): the summary, a table per scenario
2//! with each block, and the failure with its meaning, fix and the app's screen. The page needs
3//! nothing from the network, reads in light and dark, and fits a phone.
4
5use crate::report::Report;
6
7/// `reports` as one self-contained page.
8pub fn page(reports: &[Report]) -> String {
9    render(reports, None)
10}
11
12/// A page kept in `folder`, so films beside it can have portable relative links.
13pub(crate) fn page_in(reports: &[Report], folder: &std::path::Path) -> String {
14    render(reports, Some(folder))
15}

17fn render(reports: &[Report], folder: Option<&std::path::Path>) -> String {
18    let passed = reports.iter().filter(|r| r.passed()).count();
19    let blocks: usize = reports.iter().map(|r| r.total).sum();
20    let ms: u64 = reports.iter().map(|r| r.duration_ms).sum();
21    let verdict = if passed == reports.len() { "passed" } else { "failed" };
22    let mut body = format!(
23        "<header><h1>Relatório e2e</h1><p class=\"verdict {verdict}\">{}</p>\
24         <dl><div><dt>Cenários</dt><dd>{passed} de {} passaram</dd></div>\
25         <div><dt>Passos</dt><dd>{blocks}</dd></div><div><dt>Tempo</dt><dd>{}</dd></div>\
26         <div><dt>Janela</dt><dd>{}</dd></div></dl></header>",
27        if verdict == "passed" { "Passou" } else { "Falhou" },
28        reports.len(),
29        seconds(ms),
30        esc(&window_line(reports))
31    );
32    for report in reports {
33        body.push_str(&scenario(report, folder));
34    }
35    if let Some(report) = reports.first() {
36        body.push_str("<footer><h2>Não verificado</h2><ul>");
37        for item in report.not_verified() {
38            body.push_str(&format!("<li>{}</li>", esc(item)));
39        }
40        body.push_str("</ul></footer>");
41    }
42    format!(
43        "<!doctype html>\n<html lang=\"pt-BR\"><head><meta charset=\"utf-8\">\
44         <meta name=\"viewport\" content=\"width=device-width, initial-scale=1\">\
45         <title>Relatório e2e</title><style>{STYLE}</style></head><body><main>{body}</main></body></html>\n"
46    )
47}

49fn scenario(report: &Report, folder: Option<&std::path::Path>) -> String {
50    let state = if report.passed() { "passed" } else { "failed" };
51    let mut out = format!(
52        "<section class=\"{state}\"><h2>{}</h2><p class=\"summary\">{} · {}{}</p>\
53         <div class=\"scroll\"><table><thead><tr><th>#</th><th>Bloco</th><th>Alvo</th><th>Resultado</th><th>Tempo</th></tr></thead><tbody>",
54        esc(&report.scenario),
55        esc(&crate::session::summary(&report.outcomes, report.total)),
56        seconds(report.duration_ms),
57        if report.cached_at.is_some() { " · do cache" } else { "" }
58    );
59    for (i, o) in report.outcomes.iter().enumerate() {
60        let (class, result) = match (&o.error, &o.code) {
61            (None, _) => ("ok", "passou".to_owned()),
62            (Some(_), Some(code)) => ("bad", format!("falhou · <code>{}</code>", esc(code))),
63            (Some(_), None) => ("bad", "falhou".to_owned()),
64        };
65        out.push_str(&format!(
66            "<tr class=\"{class}\"><td>{}</td><td>{}<small>{}</small></td><td><code>{}</code></td><td>{result}</td><td>{}</td></tr>",
67            i + 1,
68            esc(&o.title),
69            esc(&o.kind),
70            esc(&o.line),
71            seconds(o.ms)
72        ));
73    }
74    for (i, title) in report.not_run.iter().enumerate() {
75        out.push_str(&format!(
76            "<tr class=\"skip\"><td>{}</td><td>{}</td><td></td><td>não rodou</td><td></td></tr>",
77            report.outcomes.len() + i + 1,
78            esc(title)
79        ));
80    }
81    out.push_str("</tbody></table></div>");
82    if let Some(f) = report.failure() {
83        out.push_str("<div class=\"failure\"><h3>Por que falhou</h3>");
84        out.push_str(&format!("<p>{}</p>", esc(f.error.as_deref().unwrap_or_default())));
85        if let Some(fix) = &f.fix {
86            out.push_str(&format!("<p><strong>Como corrigir:</strong> {}</p>", esc(fix)));
87        }
88        if !f.close.is_empty() {
89            out.push_str("<p><strong>Parecidos na tela:</strong></p><ul>");
90            for close in &f.close {
91                out.push_str(&format!("<li><code>{}</code></li>", esc(close)));
92            }
93            out.push_str("</ul>");
94        }
95        if let Some(screen) = &f.screen {
96            out.push_str(&format!(
97                "<details open><summary>Tela do app testado</summary><pre>{}</pre></details>",
98                esc(screen)
99            ));
100        }
101        out.push_str("</div>");
102    }
103    if !report.films.is_empty() || report.film_note.is_some() {
104        out.push_str("<div class=\"films\"><h3>Filme</h3><ul>");
105        for film in &report.films {
106            let name = std::path::Path::new(film)
107                .file_name()
108                .map_or_else(|| film.clone(), |n| n.to_string_lossy().into_owned());
109            match film_href(film, folder) {
110                Some(href) => out.push_str(&format!(
111                    "<li><a href=\"{}\">{}</a> <small>abre no MoviePlayer</small></li>",
112                    esc(&href),
113                    esc(&name)
114                )),
115                None => out.push_str(&format!("<li>{} <small>caminho indisponível</small></li>", esc(film))),
116            }
117        }
118        if let Some(note) = &report.film_note {
119            out.push_str(&format!("<li>{}</li>", esc(note)));
120        }
121        out.push_str("</ul></div>");
122    }
123    out.push_str("</section>");
124    out
125}

127/// How the apps were shown, as the header says it: with the window or not, where, at what pace.
128fn window_line(reports: &[Report]) -> String {
129    let headed = reports.iter().any(|r| r.window == crate::window::WindowMode::Headed);
130    if !headed {
131        return format!("sem janela, em {}", crate::window::this_system());
132    }
133    let mut line = format!("com a janela do app, em {}", crate::window::this_system());
134    if let Some(ms) = reports.iter().map(|r| r.slow_mo_ms).find(|ms| *ms > 0) {
135        line.push_str(&format!(", {ms} ms entre os passos"));
136    }
137    if reports.iter().any(|r| r.stopped) {
138        line.push_str(", parado antes do fim");
139    }
140    line
141}

143fn film_href(film: &str, folder: Option<&std::path::Path>) -> Option<String> {
144    // Existing symlinks resolve to the real file; a missing old artifact still gets a path.
145    let path = std::fs::canonicalize(film).or_else(|_| std::path::absolute(film)).ok()?;
146    let url = url::Url::from_file_path(&path).ok()?;
147    if let Some(folder) = folder
148        && let Ok(folder) = std::fs::canonicalize(folder).or_else(|_| std::path::absolute(folder))
149        && path.parent() == Some(folder.as_path())
150        && let Ok(base) = url::Url::from_directory_path(folder)
151        && let Some(relative) = base.make_relative(&url)
152    {
153        return Some(relative);
154    }
155    Some(url.into())
156}

158/// Text that goes inside an element or an attribute.
159pub(crate) fn esc(text: &str) -> String {
160    let mut out = String::with_capacity(text.len());
161    for c in text.chars() {
162        match c {
163            '&' => out.push_str("&amp;"),
164            '<' => out.push_str("&lt;"),
165            '>' => out.push_str("&gt;"),
166            '"' => out.push_str("&quot;"),
167            '\'' => out.push_str("&#39;"),
168            c => out.push(c),
169        }
170    }
171    out
172}

174fn seconds(ms: u64) -> String {
175    format!("{},{:03} s", ms / 1000, ms % 1000)
176}
