1//! A Rust project the Studio is pointed at: which of its packages are made with RavenRS, which
2//! of them are apps, whether each speaks the test channel, which scenarios it keeps, and which
3//! screens it builds, so a person knows what can be tested there without knowing the project.
4//!
5//! The project is read, never built: `cargo metadata --no-deps` says the packages, and their
6//! sources are searched for the views they build (`impl View for`, `impl Component for`,
7//! functions that return `impl View`) and the test ids they declare (`.test_id("…")`).
8

17/// A package of the project made with RavenRS.
18#[derive(Clone, Debug, PartialEq)]
19pub struct Found {
20    /// The name its scenarios open: the one it declares for the Studio, or its package name.
21    pub name: String,
22    /// Its Cargo package.
23    pub package: String,
24    /// Its folder.
25    pub folder: PathBuf,
26    /// The binaries it builds; none for a library of components.
27    pub binaries: Vec<String>,
28    /// Whether it answers the test channel (`--e2e-serve`).
29    pub channel: bool,
30    /// What it declares for the Studio in `[package.metadata.ravenrs.e2e]`, when it does.
31    pub declared: Option<App>,
32    /// Its scenario files in `tests/e2e/`, with what is wrong with each that does not read.
33    pub scenarios: Vec<(PathBuf, Result<String, String>)>,
34    /// The views it builds.
35    pub screens: Vec<Screen>,
36    /// Its `e2e.toml`, or the defaults; a file that does not read leaves the defaults here and
37    /// says why in `config_error`.
38    pub config: crate::config::Config,
39    /// Why its `e2e.toml` could not be read, when it could not.
40    pub config_error: Option<String>,
41}

65/// What a project has that is made with RavenRS.
66#[derive(Clone, Debug, PartialEq)]
67pub struct Project {
68    /// The folder pointed at.
69    pub root: PathBuf,
70    /// Its packages made with RavenRS, in name order.
71    pub found: Vec<Found>,
72    /// The folder cargo builds it into.
73    pub target: PathBuf,
74}

76/// The crates whose use makes a package one made with RavenRS.
77fn uses_ravenrs(dependency: &str) -> bool {
78    dependency == "ravenrs"
79        || dependency.starts_with("ravenrs-widgets")
80        || dependency.starts_with("ravenrs-app")
81}

83/// Reads the project in `folder` (a package or a workspace, or any folder inside one).
84pub fn read(folder: &Path) -> Result<Project, String> {
85    let root = folder.canonicalize().map_err(|e| format!("{}: {e}", folder.display()))?;
86    let manifest = root.join("Cargo.toml");
87    let mut cargo = Process::new("cargo");
88    cargo.args(["metadata", "--no-deps", "--format-version", "1", "--offline"]).current_dir(&root);
89    if manifest.is_file() {
90        cargo.arg("--manifest-path").arg(&manifest);
91    }
92    let out = cargo.output().map_err(|e| format!("não deu para rodar o cargo: {e}"))?;
93    if !out.status.success() {
94        let why = String::from_utf8_lossy(&out.stderr);
95        return Err(format!("{} não é um projeto Rust que o cargo leia: {}", root.display(), why.trim()));
96    }
97    let metadata: Value = serde_json::from_slice(&out.stdout).map_err(|e| e.to_string())?;
98    Ok(of_metadata(&root, &metadata))
99}

101/// The project as `cargo metadata` describes it, counting only the packages inside `root`.
102pub fn of_metadata(root: &Path, metadata: &Value) -> Project {
103    let mut found: Vec<Found> = metadata["packages"]
104        .as_array()
105        .into_iter()
106        .flatten()
107        .filter_map(|package| {
108            let manifest = PathBuf::from(package["manifest_path"].as_str()?);
109            let folder = manifest.parent()?.to_path_buf();
110            if !folder.starts_with(root) {
111                return None;
112            }
113            let dependencies: Vec<&str> = package["dependencies"]
114                .as_array()
115                .into_iter()
116                .flatten()
117                .filter_map(|d| d["name"].as_str())
118                .collect();
119            if !dependencies.iter().any(|d| uses_ravenrs(d)) {
120                return None;
121            }
122            let binaries: Vec<String> = package["targets"]
123                .as_array()
124                .into_iter()
125                .flatten()
126                .filter(|t| t["kind"].as_array().is_some_and(|k| k.iter().any(|k| k == "bin")))
127                .filter_map(|t| t["name"].as_str().map(str::to_owned))
128                .collect();
129            let declared = App::declared(package);
130            let package_name = package["name"].as_str()?.to_owned();
131            let sources = rust_files(&folder.join("src"));
132            let channel = dependencies.contains(&"ravenrs-e2e-wire")
133                && sources.iter().any(|file| {
134                    std::fs::read_to_string(file).is_ok_and(|text| {
135                        text.contains("e2e-serve")
136                            || text.contains("SERVE_FLAG")
137                            || text.contains("ravenrs_e2e_wire::serve")
138                    })
139                });
140            let (config, config_error) = match crate::config::Config::read(&folder) {
141                Ok(config) => (config, None),
142                Err(why) => (crate::config::Config::default(), Some(why)),
143            };
144            let config_error = App::declaration_error(package)
145                .map(|why| format!("{}: {why}", manifest.display()))
146                .or(config_error);
147            Some(Found {
148                name: declared.as_ref().map_or_else(|| package_name.clone(), |d| d.name.clone()),
149                package: package_name,
150                scenarios: scenarios_in(&folder.join(&config.folder)),
151                screens: screens_in(root, &sources),
152                folder,
153                binaries,
154                channel,
155                declared,
156                config,
157                config_error,
158            })
159        })
160        .collect();
161    found.sort_by(|a, b| a.package.cmp(&b.package));
162    let target = metadata["target_directory"].as_str().map_or_else(|| root.join("target"), PathBuf::from);
163    Project { root: root.to_path_buf(), found, target }
164}

166fn rust_files(folder: &Path) -> Vec<PathBuf> {
167    let mut files = Vec::new();
168    let mut pending = vec![folder.to_path_buf()];
169    while let Some(dir) = pending.pop() {
170        for entry in std::fs::read_dir(&dir).into_iter().flatten().flatten() {
171            let path = entry.path();
172            if path.is_dir() {
173                pending.push(path);
174            } else if path.extension().is_some_and(|e| e == "rs") {
175                files.push(path);
176            }
177        }
178    }
179    files.sort();
180    files
181}

183fn scenarios_in(folder: &Path) -> Vec<(PathBuf, Result<String, String>)> {
184    let mut files: Vec<PathBuf> = std::fs::read_dir(folder)
185        .into_iter()
186        .flatten()
187        .flatten()
188        .map(|e| e.path())
189        .filter(|p| p.extension().is_some_and(|e| e == ravenrs_e2e_wire::EXTENSION))
190        .collect();
191    files.sort();
192    files
193        .into_iter()
194        .map(|file| {
195            let read = std::fs::read_to_string(&file).map_err(|e| e.to_string()).and_then(|text| {
196                ravenrs_e2e_wire::Scenario::parse(&text).map(|s| s.name).map_err(|e| e.to_string())
197            });
198            (file, read)
199        })
200        .collect()
201}

203/// The views the files build, with the test ids each file declares.
204fn screens_in(root: &Path, files: &[PathBuf]) -> Vec<Screen> {
205    let (Ok(implemented), Ok(function), Ok(test_id)) = (
206        Regex::new(r"impl(?:\s*<[^>]*>)?\s+(View|Component)\s+for\s+([A-Za-z_]\w*)"),
207        Regex::new(r"pub\s+fn\s+([a-z_]\w*)\s*(?:<[^>]*>)?\s*\([^)]*\)\s*->\s*impl\s+View\b"),
208        Regex::new(r#"test_id\(\s*"([^"]+)""#),
209    ) else {
210        return Vec::new();
211    };
212    let mut screens = Vec::new();
213    for file in files {
214        let Ok(text) = std::fs::read_to_string(file) else { continue };
215        let ids: Vec<String> = test_id.captures_iter(&text).map(|c| c[1].to_owned()).collect();
216        let relative = file.strip_prefix(root).unwrap_or(file).to_path_buf();
217        for (index, line) in text.lines().enumerate() {
218            let found = implemented
219                .captures(line)
220                .map(|c| (c[2].to_owned(), if &c[1] == "View" { "View" } else { "Component" }))
221                .or_else(|| function.captures(line).map(|c| (c[1].to_owned(), "fn")));
222            if let Some((name, kind)) = found {
223                screens.push(Screen {
224                    name,
225                    kind,
226                    file: relative.clone(),
227                    line: index + 1,
228                    test_ids: ids.clone(),
229                });
230            }
231        }
232    }
233    screens
234}

236impl Project {
237    /// Refuses execution when an app's test configuration is invalid, while discovery can
238    /// still show every error. Both command-line tools use this before opening an app.
239    pub fn check_configs(&self) -> Result<(), String> {
240        let errors: Vec<&str> = self.found.iter().filter_map(|found| found.config_error.as_deref()).collect();
241        if errors.is_empty() { Ok(()) } else { Err(errors.join("\n")) }
242    }
243    /// Where a run of this project keeps what it found: `e2e` inside the folder cargo builds
244    /// this project into, as `cargo metadata` says (the workspace's `target`, or
245    /// `CARGO_TARGET_DIR` and `build.target-dir` when set), so git never sees it.
246    pub fn e2e_folder(&self) -> PathBuf {
247        self.target.join("e2e")
248    }
249
250    /// Makes the apps of the project known by name, its code before any installed copy, so its
251    /// scenarios open them.
252    pub fn learn_apps(&self) {
253        for found in self.found.iter().filter(|f| f.is_app() && f.channel && f.config_error.is_none()) {
254            let declared = found.declared.clone().unwrap_or_default();
255            apps::learn(App {
256                name: found.name.clone(),
257                package: found.package.clone(),
258                binary: Some(declared.binary.clone())
259                    .filter(|b| !b.is_empty() && found.binaries.contains(b))
260                    .unwrap_or_else(|| found.binaries[0].clone()),
261                source: Some(self.root.clone()),
262                source_first: true,
263                binary_path: None,
264                ..declared
265            });
266        }
267    }
268
269    /// The scenario folders of the apps that can run them: with the channel and scenarios.
270    pub fn scenario_folders(&self) -> Vec<PathBuf> {
271        self.found
272            .iter()
273            .filter(|f| f.is_app() && f.channel && !f.scenarios.is_empty())
274            .map(|f| f.folder.join(&f.config.folder))
275            .collect()
276    }

278    /// Everything, for a tool or an AI.
279    pub fn json(&self) -> Value {
280        let found: Vec<Value> = self
281            .found
282            .iter()
283            .map(|f| {
284                json!({
285                    "name": f.name,
286                    "package": f.package,
287                    "kind": if f.is_app() { "app" } else { "library" },
288                    "folder": f.folder.strip_prefix(&self.root).unwrap_or(&f.folder),
289                    "binaries": f.binaries,
290                    "channel": f.channel,
291                    "declared": f.declared.is_some(),
292                    "features": f.declared.as_ref().map(|app| app.features.as_slice()).unwrap_or_default(),
293                    "config_error": f.config_error,
294                    "next": next_step(f),
295                    "scenarios": f.scenarios.iter().map(|(file, read)| match read {
296                        Ok(name) => json!({ "file": file.strip_prefix(&self.root).unwrap_or(file), "name": name }),
297                        Err(why) => json!({ "file": file.strip_prefix(&self.root).unwrap_or(file), "error": why }),
298                    }).collect::<Vec<_>>(),
299                    "screens": f.screens.iter().map(|s| json!({
300                        "name": s.name, "kind": s.kind, "file": s.file, "line": s.line, "test_ids": s.test_ids,
301                    })).collect::<Vec<_>>(),
302                })
303            })
304            .collect();
305        json!({
306            "root": self.root,
307            "apps": self.found.iter().filter(|f| f.is_app()).count(),
308            "libraries": self.found.iter().filter(|f| !f.is_app()).count(),
309            "found": found,
310        })
311    }

313    /// A few lines for a person.
314    pub fn text(&self) -> String {
315        if self.found.is_empty() {
316            return format!("{}: nenhum pacote feito com RavenRS", self.root.display());
317        }
318        let mut out = format!("{}\n", self.root.display());
319        for f in &self.found {
320            let unmarked = f.screens.iter().filter(|s| s.test_ids.is_empty()).count();
321            out.push_str(&format!(
322                "\n{} ({}, {}): {} telas, {} sem marcas para teste; canal de teste: {}; cenários: {}\n",
323                f.name,
324                f.package,
325                if f.is_app() { format!("app {}", f.binaries.join(", ")) } else { "biblioteca".to_owned() },
326                f.screens.len(),
327                unmarked,
328                if f.channel { "sim" } else { "não" },
329                f.scenarios.len(),
330            ));
331            if let Some(app) = &f.declared
332                && !app.features.is_empty()
333            {
334                out.push_str(&format!("  features de teste: {}\n", app.features.join(", ")));
335            }
336            for (file, read) in &f.scenarios {
337                let shown = file.strip_prefix(&self.root).unwrap_or(file).display().to_string();
338                match read {
339                    Ok(name) => out.push_str(&format!("  cenário {shown}: {name}\n")),
340                    Err(why) => out.push_str(&format!("  cenário {shown}: não lê ({why})\n")),
341                }
342            }
343            if let Some(why) = &f.config_error {
344                out.push_str(&format!("  configuração de teste não lê: {why}\n"));
345            }
346            if let Some(next) = next_step(f) {
347                out.push_str(&format!("  próximo passo: {next}\n"));
348            }
349        }
350        out
351    }
352}

354/// What to do so the package can be tested by the Studio, when something is missing.
355fn next_step(f: &Found) -> Option<String> {
356    if !f.is_app() {
357        return Some("é uma biblioteca: teste as telas dela pelo app que as usa".to_owned());
358    }
359    if !f.channel {
360        return Some(format!(
361            "ligue o canal de teste: dependa de ravenrs-e2e-wire e trate {} no main (cargo ravenrs e2e install, F6 da #627, fará isso)",
362            ravenrs_e2e_wire::SERVE_FLAG
363        ));
364    }
365    if f.scenarios.is_empty() {
366        return Some(format!(
367            "escreva cenários .raventest em {}/ do app, ou numa pasta do QA",
368            ravenrs_e2e_wire::FOLDER
369        ));
370    }
371    None
