1//! What a run drives and how each step goes: the app under test (in this process or apart, over
2//! the test channel), the step a block stands for, why a step failed in words a person reads
3//! (Portuguese, by the stable code) and how each block went.
4
5use ravenrs_e2e::E2e;
6use ravenrs_e2e_wire::{Block, Client, Command, Kind};
7
8/// The app a run drives: compiled into the tool, or opened apart in a child process with no
9/// window and driven over the test channel (`ravenrs-e2e-wire`). A step runs the same either way.
10#[derive(Debug)]
11pub enum Session {
12    /// An app compiled into the tool, such as the sample form or another Studio.
13    Here(Box<E2e>),
14    /// An app opened in a child process, such as Raven Finanças; it ends with the run.
15    Apart(Client),
16}
17
18impl Session {
19    /// Runs `command` on the app, as `Command::apply` answers.
20    pub fn run(&mut self, command: &Command) -> Result<Option<String>, ravenrs_e2e_wire::Failure> {
21        match self {
22            Self::Here(e2e) => command.apply(e2e),
23            Self::Apart(client) => client.send(command),
24        }
25    }
26
27    /// The screen of the app as text, when it can still say.
28    pub fn screen(&mut self) -> Option<String> {
29        self.run(&Command::Snapshot).ok().flatten()
30    }
31}
32
33/// Opens the app a scenario's Open block names: `None` when there is no app by that name, the
34/// reason when there is one and it could not be opened.
35pub type Opener = fn(&str) -> Option<Result<Session, Box<Failure>>>;
36
37/// One step a block stands for.
38#[derive(Clone, Debug, PartialEq)]
39pub enum Step {
40    /// Opens the app under test by name, from nothing.
41    Open(String),
42    /// Anything done on the app once it is open: click, type, check or mention, with its target
43    /// written as on the block (`#id` or the text shown).
44    Do(Command),
45}
46
47impl Step {
48    /// The step `block` stands for, or why it cannot run, said to whoever wrote it.
49    pub fn of(block: &Block) -> Result<Self, String> {
50        if block.kind == Kind::Open {
51            return Ok(Self::Open(block.target.clone()));
52        }
53        if block.target.trim().is_empty() {
54            return Err("Diga o alvo: #id ou o texto que aparece".to_owned());
55        }
56        let needs_text = matches!(
57            block.kind,
58            Kind::Type | Kind::Check | Kind::Mention | Kind::Fill | Kind::Select | Kind::Drag
59        );
60        if needs_text && block.text.is_none() {
61            return Err(if block.kind == Kind::Drag {
62                "Diga para onde arrastar: #de = #até".to_owned()
63            } else {
64                "Diga o texto depois do alvo: #alvo = texto".to_owned()
65            });
66        }
67        block.command().map(Self::Do).ok_or_else(|| format!("Bloco desconhecido: {}", block.kind.word()))
68    }
69}

71/// Why a block failed: the message, and from `ravenrs-e2e` its stable code, how to fix it and
72/// what came close on the screen.
73#[derive(Clone, Debug, Default, PartialEq)]
74pub struct Failure {
75    /// What went wrong, starting with the code when there is one: `[E2E_NOT_FOUND] …`.
76    pub message: String,
77    /// The stable code, such as `E2E_NOT_FOUND`.
78    pub code: Option<String>,
79    /// What the failure means, in Portuguese, from its code.
80    pub meaning: Option<String>,
81    /// The detail `ravenrs-e2e` gave: which query, what it found.
82    pub detail: Option<String>,
83    /// How to fix it, when there is an obvious way.
84    pub fix: Option<String>,
85    /// Up to three nodes that came close to what the block asked for.
86    pub close: Vec<String>,
87    /// The screen of the app when it failed, when the failure brought it.
88    pub screen: Option<String>,
89}
90
91impl Failure {
92    /// A failure with only a message, for what is not a step of the app: no app to open, a
93    /// block that cannot run.
94    pub fn plain(message: impl Into<String>) -> Self {
95        Self { message: message.into(), ..Self::default() }
96    }
97}
98
99impl std::fmt::Display for Failure {
100    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
101        f.write_str(&self.message)
102    }
103}
104
105impl std::error::Error for Failure {}

174impl From<ravenrs_e2e_wire::Failure> for Failure {
175    fn from(f: ravenrs_e2e_wire::Failure) -> Self {
176        let (meaning, fix) = in_portuguese(&f.code);
177        Self {
178            message: format!("[{}] {meaning}: {}", f.code, f.message),
179            meaning: Some(meaning.to_owned()),
180            detail: Some(f.message),
181            fix: fix.map(str::to_owned).or(f.fix),
182            close: f.close,
183            screen: (!f.screen.is_empty()).then_some(f.screen),
184            code: Some(f.code),
185        }
186    }
187}

189/// The failure of an Open block that names no app this run knows.
190fn no_app(name: &str) -> Failure {
191    let (meaning, _) = in_portuguese("E2E_APP_START");
192    Failure {
193        message: format!("[E2E_APP_START] {meaning}: não há app chamado {name:?} para abrir"),
194        code: Some("E2E_APP_START".to_owned()),
195        meaning: Some(meaning.to_owned()),
196        detail: Some(format!("não há app chamado {name:?}")),
197        fix: Some(
198            "o nome no bloco abrir é o app que um pacote declara em [package.metadata.ravenrs.e2e]; rode dentro do projeto dele, ou o abra pelo Studio, que tem apps compilados nele"
199                .to_owned(),
200        ),
201        ..Failure::default()
202    }
203}

205/// Runs `step` on the app in `session`, opening apps with `open`.
206pub fn perform(session: &mut Option<Session>, step: &Step, open: Opener) -> Result<(), Box<Failure>> {
207    let command = match step {
208        Step::Open(name) => {
209            // The app a previous Open block started ends before the next one starts.
210            *session = None;
211            let opened = open(name).ok_or_else(|| Box::new(no_app(name)))?;
212            *session = Some(opened?);
213            return Ok(());
214        }
215        Step::Do(command) => command,
216    };
217    let Some(app) = session.as_mut() else {
218        return Err(Box::new(Failure::plain("O app não está aberto: comece o cenário por um bloco Abrir")));
219    };
220    app.run(command).map(|_| ()).map_err(|f| Box::new(Failure::from(f)))
221}

223/// What a step of a run did, for the log and for whoever called the Studio.
224#[derive(Clone, Debug, PartialEq)]
225pub struct Outcome {
226    /// The block's id in the scenario.
227    pub id: String,
228    /// The block's kind: `open`, `click`, `type`, `check` or `mention`.
229    pub kind: String,
230    /// The block's title.
231    pub title: String,
232    /// The block's target and text, as written on it.
233    pub line: String,
234    /// `None` when it passed, else what went wrong, starting with its code when it has one.
235    pub error: Option<String>,
236    /// The stable code of the failure, such as `E2E_NOT_FOUND`, when it has one.
237    pub code: Option<String>,
238    /// What the failure means, in Portuguese, from its code.
239    pub meaning: Option<String>,
240    /// The detail `ravenrs-e2e` gave: which query, what it found.
241    pub detail: Option<String>,
242    /// How to fix the failure, when there is an obvious way.
243    pub fix: Option<String>,
244    /// Up to three nodes that came close to what the block asked for.
245    pub close: Vec<String>,
246    /// The screen of the app under test when the block failed, as `E2e::snapshot` writes it.
247    pub screen: Option<String>,
248    /// How long the block took, in milliseconds, waits included.
249    pub ms: u64,
250}

252/// One line that sums up a run: how many blocks passed out of the path.
253pub fn summary(outcomes: &[Outcome], total: usize) -> String {
254    let passed = outcomes.iter().filter(|o| o.error.is_none()).count();
255    match outcomes.iter().position(|o| o.error.is_some()) {
256        Some(i) => format!("Falhou no passo {} de {total}", i + 1),
257        None if passed == total => format!("{passed} de {total} passos passaram"),
258        None => format!("{passed} de {total} passos até agora"),
259    }
260}
