1//! What a run tells whoever asked for it: a person in a terminal, a build server, or the coding
2//! agent that called the tool.
3//!
4//! The text is a few lines an agent can quote as is. The JSON carries every block with its
5//! result, the stable code of a failure and the screen of the app at that moment, so the agent
6//! can explain or fix the failure without running it again.

37/// A finished run of a scenario.
38#[derive(Clone, Debug, PartialEq)]
39pub struct Report {
40    /// The scenario's name.
41    pub scenario: String,
42    /// How each block went, in order.
43    pub outcomes: Vec<Outcome>,
44    /// How many blocks the scenario's path has.
45    pub total: usize,
46    /// When the run this report came from was kept in the cache, in seconds since 1970;
47    /// `None` when it ran just now.
48    pub cached_at: Option<u64>,
49    /// How long the run took, in milliseconds, measured on the clock of the machine that ran
50    /// it (a report from the cache keeps the time of the run it came from).
51    pub duration_ms: u64,
52    /// Whether the app ran apart, in a child process over the test channel.
53    pub apart: bool,
54    /// The titles of the blocks of the path that did not run, because one before them failed.
55    pub not_run: Vec<String>,
56    /// The films kept of the apps the run drove (`.ravenmovie`), when it filmed.
57    pub films: Vec<String>,
58    /// Why an app was not filmed, when it was asked to be.
59    pub film_note: Option<String>,
60    /// Whether the apps showed their window (`--headed`).
61    pub window: crate::window::WindowMode,
62    /// How long the run waited after each step, for a person to follow it.
63    pub slow_mo_ms: u64,
64    /// Whether the person watching stopped the run before its end.
65    pub stopped: bool,
66}

69    /// A run that could not start: its first block failed with `why`, before any app opened.

74    /// A run refused before its scenarios could start, preserving the structured failure
75    /// from opening the app or reading its embedded scenarios. Use [`Self::refused`] for a
76    /// plain message; neither constructor infers a code from the message's text.
77    pub fn refused_with(scenario: &str, block: &str, failure: runner::Failure) -> Self {
78        let outcome = Outcome {
79            id: "open".to_owned(),
80            kind: "open".to_owned(),
81            title: format!("Abrir o {block}"),
82            line: block.to_owned(),
83            error: Some(failure.message),
84            code: failure.code,
85            meaning: failure.meaning,
86            detail: failure.detail,
87            fix: failure.fix,
88            close: failure.close,
89            screen: failure.screen,
90            ms: 0,
91        };
92        Self {
93            scenario: scenario.to_owned(),
94            outcomes: vec![outcome],
95            total: 1,
96            cached_at: None,
97            duration_ms: 0,
98            apart: true,
99            not_run: Vec::new(),
100            films: Vec::new(),
101            film_note: None,
102            window: crate::app::window(),
103            slow_mo_ms: 0,
104            stopped: false,
105        }
106    }

108    /// What this run could not check, with the systems it did not run on.
109    pub fn not_verified(&self) -> Vec<&'static str> {
110        let mut said = match (self.window, self.apart) {
111            (crate::window::WindowMode::Headed, _) => NOT_VERIFIED_HEADED,
112            (_, true) => NOT_VERIFIED_APART,
113            (_, false) => NOT_VERIFIED,
114        }
115        .to_vec();
116        said[1] = crate::window::other_systems();
117        said
118    }

120    /// A report read back from what [`Report::json`] wrote.
121    pub fn from_json(value: &Value) -> Option<Self> {
122        let text = |v: &Value, key: &str| v.get(key).and_then(Value::as_str).map(str::to_owned);
123        let outcomes = value
124            .get("blocks")?
125            .as_array()?
126            .iter()
127            .map(|b| {
128                Some(Outcome {
129                    id: text(b, "id")?,
130                    kind: text(b, "kind")?,
131                    title: text(b, "title")?,
132                    line: text(b, "line").unwrap_or_default(),
133                    error: text(b, "error"),
134                    code: text(b, "code"),
135                    meaning: text(b, "meaning"),
136                    detail: text(b, "detail"),
137                    fix: text(b, "fix"),
138                    close: b
139                        .get("close")
140                        .and_then(Value::as_array)
141                        .map(|a| a.iter().filter_map(Value::as_str).map(str::to_owned).collect())
142                        .unwrap_or_default(),
143                    screen: text(b, "screen"),
144                    ms: b.get("duration_ms").and_then(Value::as_u64).unwrap_or_default(),
145                })
146            })
147            .collect::<Option<Vec<_>>>()?;
148        Some(Self {
149            scenario: text(value, "scenario")?,
150            outcomes,
151            total: usize::try_from(value.get("total")?.as_u64()?).ok()?,
152            cached_at: None,
153            duration_ms: value.get("duration_ms").and_then(Value::as_u64).unwrap_or_default(),
154            apart: value.get("opened_apart").and_then(Value::as_bool).unwrap_or_default(),
155            not_run: value
156                .get("not_run")
157                .and_then(Value::as_array)
158                .map(|a| a.iter().filter_map(Value::as_str).map(str::to_owned).collect())
159                .unwrap_or_default(),
160            films: value
161                .get("films")
162                .and_then(Value::as_array)
163                .map(|a| a.iter().filter_map(Value::as_str).map(str::to_owned).collect())
164                .unwrap_or_default(),
165            film_note: text(value, "film_note"),
166            window: value
167                .get("window")
168                .and_then(Value::as_str)
169                .and_then(crate::window::WindowMode::read)
170                .unwrap_or_default(),
171            slow_mo_ms: value.get("slow_mo_ms").and_then(Value::as_u64).unwrap_or_default(),
172            stopped: value.get("stopped").and_then(Value::as_bool).unwrap_or_default(),
173        })
174    }

176    /// Whether every block of the path ran and passed.
177    pub fn passed(&self) -> bool {
178        self.outcomes.len() == self.total && self.outcomes.iter().all(|o| o.error.is_none())
179    }
180
181    /// The first block that failed, if any.
182    pub fn failure(&self) -> Option<&Outcome> {
183        self.outcomes.iter().find(|o| o.error.is_some())
184    }
185
186    /// The process exit code: 0 when it passed, 1 when it did not.
187    pub fn exit_code(&self) -> i32 {
188        if self.passed() { 0 } else { 1 }
189    }

191    /// A few lines to read or quote: the verdict, the failure and its fix, what was not checked.
192    pub fn text(&self) -> String {
193        let mut out = format!(
194            "{}: {}\n{}",
195            self.scenario,
196            if self.passed() { "passou" } else { "falhou" },
197            runner::summary(&self.outcomes, self.total)
198        );
199        if self.window == crate::window::WindowMode::Headed {
200            out.push_str(&format!("\ncom a janela do app na tela, em {}", crate::window::this_system()));
201            if self.slow_mo_ms > 0 {
202                out.push_str(&format!(", {} ms entre os passos", self.slow_mo_ms));
203            }
204        }
205        if self.stopped {
206            out.push_str("\nparado por quem acompanhava antes do fim");
207        }
208        if self.cached_at.is_some() {
209            out.push_str(
210                "\ndo cache: nem o cenário nem o build mudaram desde a última vez (--no-cache roda de novo)",
211            );
212        }
213        if let Some(f) = self.failure() {
214            out.push_str(&format!(
215                "\nfalhou em {} ({}: {}): {}",
216                f.title,
217                f.kind,
218                f.line,
219                f.error.as_deref().unwrap_or_default()
220            ));
221            if let Some(fix) = &f.fix {
222                out.push_str(&format!("\ncomo corrigir: {fix}"));
223            }
224            if !f.close.is_empty() {
225                out.push_str("\nparecidos na tela:");
226                for line in &f.close {
227                    out.push_str(&format!("\n  {line}"));
228                }
229            }
230            if let Some(screen) = &f.screen {
231                out.push_str("\ntela do app testado:\n");
232                out.push_str(screen);
233            }
234        }
235        for film in &self.films {
236            out.push_str(&format!("\nfilme: {film}"));
237        }
238        if let Some(note) = &self.film_note {
239            out.push_str(&format!("\nsem filme: {note}"));
240        }
241        out.push_str("\nnão verificado: ");
242        out.push_str(&self.not_verified().join("; "));
243        out
244    }

246    /// Everything, for a tool: the verdict, each block with its result, code and screen.
247    pub fn json(&self) -> Value {
248        let blocks: Vec<Value> = self
249            .outcomes
250            .iter()
251            .enumerate()
252            .map(|(i, o)| {
253                json!({
254                    "step": i + 1,
255                    "id": o.id,
256                    "kind": o.kind,
257                    "title": o.title,
258                    "line": o.line,
259                    "result": if o.error.is_none() { "passed" } else { "failed" },
260                    "code": o.code,
261                    "error": o.error,
262                    "meaning": o.meaning,
263                    "detail": o.detail,
264                    "fix": o.fix,
265                    "close": o.close,
266                    "screen": o.screen,
267                    "duration_ms": o.ms,
268                })
269            })
270            .collect();
271        json!({
272            "scenario": self.scenario,
273            "verdict": if self.passed() { "passed" } else { "failed" },
274            "summary": runner::summary(&self.outcomes, self.total),
275            "total": self.total,
276            "ran": self.outcomes.len(),
277            "blocks": blocks,
278            "duration_ms": self.duration_ms,
279            "from_cache": self.cached_at.is_some(),
280            "cached_at": self.cached_at,
281            "not_verified": self.not_verified(),
282            "opened_apart": self.apart,
283            "not_run": self.not_run,
284            "films": self.films,
285            "film_note": self.film_note,
286            "window": self.window.word(),
287            "slow_mo_ms": self.slow_mo_ms,
288            "platform": crate::window::this_system(),
289            "stopped": self.stopped,
290        })
291    }

294impl Report {
295    /// The run as lines of JSON, one per event: `start`, a `step` for each block that ran, with
296    /// how long it took and how it went, and `end` with the verdict. It is what `trace.jsonl`
297    /// keeps, and what `--format jsonl` prints.
298    pub fn trace(&self) -> String {
299        let mut lines = vec![json!({
300            "event": "start",
301            "scenario": self.scenario,
302            "total": self.total,
303            "opened_apart": self.apart,
304            "from_cache": self.cached_at.is_some(),
305            "window": self.window.word(),
306            "slow_mo_ms": self.slow_mo_ms,
307            "platform": crate::window::this_system(),
308        })];
309        for (i, o) in self.outcomes.iter().enumerate() {
310            lines.push(json!({
311                "event": "step",
312                "step": i + 1,
313                "id": o.id,
314                "kind": o.kind,
315                "title": o.title,
316                "line": o.line,
317                "result": if o.error.is_none() { "passed" } else { "failed" },
318                "code": o.code,
319                "error": o.error,
320                "duration_ms": o.ms,
321            }));
322        }
323        for (i, title) in self.not_run.iter().enumerate() {
324            lines.push(json!({
325                "event": "skipped",
326                "step": self.outcomes.len() + i + 1,
327                "title": title,
328            }));
329        }
330        lines.push(json!({
331            "event": "end",
332            "verdict": if self.passed() { "passed" } else { "failed" },
333            "summary": runner::summary(&self.outcomes, self.total),
334            "duration_ms": self.duration_ms,
335        }));
336        lines.iter().map(Value::to_string).collect::<Vec<_>>().join("\n") + "\n"
337    }

339    /// `reports` in the JUnit XML that build servers and test tools read: a `testsuite` for each
340    /// report and a `testcase` for each block of its path. A failure carries the stable code in
341    /// `type`, and the fix and the screen of the app in its text; the blocks after it are
342    /// `skipped`.
343    pub fn junit(reports: &[Report]) -> String {
344        let mut out = String::from("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n");
345        let tests: usize = reports.iter().map(|r| r.outcomes.len() + r.not_run.len()).sum();
346        let failures = reports.iter().filter(|r| r.failure().is_some()).count();
347        let skipped: usize = reports.iter().map(|r| r.not_run.len()).sum();
348        let seconds: u64 = reports.iter().map(|r| r.duration_ms).sum();
349        out.push_str(&format!(
350            "<testsuites name=\"ravenrs-e2e\" tests=\"{tests}\" failures=\"{failures}\" skipped=\"{skipped}\" time=\"{}\">\n",
351            secs(seconds)
352        ));
353        for report in reports {
354            let suite = xml(&report.scenario);
355            out.push_str(&format!(
356                "  <testsuite name=\"{suite}\" tests=\"{}\" failures=\"{}\" skipped=\"{}\" time=\"{}\">\n",
357                report.outcomes.len() + report.not_run.len(),
358                usize::from(report.failure().is_some()),
359                report.not_run.len(),
360                secs(report.duration_ms)
361            ));
362            for (i, o) in report.outcomes.iter().enumerate() {
363                let name = xml(&format!("{} {}: {}", i + 1, o.title, o.line));
364                let open =
365                    format!("    <testcase classname=\"{suite}\" name=\"{name}\" time=\"{}\"", secs(o.ms));
366                match &o.error {
367                    None => out.push_str(&format!("{open}/>\n")),
368                    Some(error) => {
369                        let mut body = error.clone();
370                        if let Some(fix) = &o.fix {
371                            body.push_str(&format!("\ncomo corrigir: {fix}"));
372                        }
373                        if let Some(screen) = &o.screen {
374                            body.push_str(&format!("\ntela do app testado:\n{screen}"));
375                        }
376                        out.push_str(&format!(
377                            "{open}>\n      <failure type=\"{}\" message=\"{}\">{}</failure>\n    </testcase>\n",
378                            xml(o.code.as_deref().unwrap_or("E2E_FAILED")),
379                            xml(error),
380                            xml(&body)
381                        ));
382                    }
383                }
384            }
385            for (i, title) in report.not_run.iter().enumerate() {
386                let name = xml(&format!("{} {title}", report.outcomes.len() + i + 1));
387                out.push_str(&format!(
388                    "    <testcase classname=\"{suite}\" name=\"{name}\" time=\"0\">\n      <skipped message=\"um passo antes falhou\"/>\n    </testcase>\n"
389                ));
390            }
391            // A build server shows the film next to the suite, as JUnit's attachments say.
392            if !report.films.is_empty() {
393                out.push_str("    <system-out>");
394                for film in &report.films {
395                    out.push_str(&xml(&format!("[[ATTACHMENT|{film}]]\n")));
396                }
397                out.push_str("</system-out>\n");
398            }
399            out.push_str("  </testsuite>\n");
400        }
401        out.push_str("</testsuites>\n");
402        out
403    }

405    /// Keeps the run in `folder`, under a folder named after the scenario: `relatorio.json`,
406    /// `relatorio.html`, `trace.jsonl` and, when it failed and the app's screen came with the failure,
407    /// `arvore.txt`. A run that passes removes an `arvore.txt` left by an earlier failure.
408    /// Says where it kept them.
409    pub fn save(&self, folder: &std::path::Path) -> std::io::Result<std::path::PathBuf> {
410        let dir = folder.join(folder_name(&self.scenario));
411        std::fs::create_dir_all(&dir)?;
412        let json = serde_json::to_string_pretty(&self.json()).unwrap_or_default();
413        std::fs::write(dir.join("relatorio.json"), json + "\n")?;
414        std::fs::write(dir.join("trace.jsonl"), self.trace())?;
415        std::fs::write(dir.join("relatorio.html"), crate::html::page_in(std::slice::from_ref(self), &dir))?;
416        let tree = dir.join("arvore.txt");
417        match self.failure().and_then(|f| f.screen.as_ref()) {
418            Some(screen) => std::fs::write(&tree, format!("{screen}\n"))?,
419            None if tree.exists() => std::fs::remove_file(&tree)?,
420            None => {}
421        }
422        Ok(dir)
423    }
424}

426/// A scenario's name as one folder: letters, digits, `-` and `_` stay, a run of anything else
427/// becomes one `-`, so a `/` or `..` in a name never leaves the folder given.
428pub fn folder_name(scenario: &str) -> String {
429    let mut name = String::new();
430    for c in scenario.chars() {
431        if c.is_alphanumeric() || c == '_' || c == '-' {
432            name.push(c);
433        } else if !name.ends_with('-') {
434            name.push('-');
435        }
436    }
437    let name = name.trim_matches('-');
438    if name.is_empty() { "cenario".to_owned() } else { name.to_owned() }
439}

441/// Text that goes inside an XML attribute or element.
442fn xml(text: &str) -> String {
443    let mut out = String::with_capacity(text.len());
444    for c in text.chars() {
445        match c {
446            '&' => out.push_str("&amp;"),
447            '<' => out.push_str("&lt;"),
448            '>' => out.push_str("&gt;"),
449            '"' => out.push_str("&quot;"),
450            '\'' => out.push_str("&apos;"),
451            // Control characters other than tab and line breaks are not allowed in XML 1.0.
452            c if c.is_control() && !matches!(c, '\t' | '\n' | '\r') => {}
453            c => out.push(c),
454        }
455    }
456    out
457}

459/// Milliseconds as the seconds JUnit writes.
460fn secs(ms: u64) -> String {
461    format!("{}.{:03}", ms / 1000, ms % 1000)
462}

464/// Keeps each report in `out`, when given, and gives them all in `format` (one of [`FORMATS`];
465/// anything else is text). The JSON is a list when `list`, else the one report. Where they were
466/// kept goes to stderr, so the JSON, JSON lines or XML on stdout stays only that.
467pub fn deliver(reports: &[Report], list: bool, format: &str, out: Option<&std::path::Path>) {
468    if let Some(out) = out {
469        for report in reports {
470            match report.save(out) {
471                Ok(dir) => eprintln!("{}: gravado em {}", report.scenario, dir.display()),
472                Err(e) => eprintln!("{}: não gravou em {}: {e}", report.scenario, out.display()),
473            }
474        }
475    }
476    print!("{}", render(reports, list, format));
477}

479/// `reports` in `format`, as [`deliver`] prints them.
480pub fn render(reports: &[Report], list: bool, format: &str) -> String {
481    match format {
482        "json" => {
483            let all: Vec<_> = reports.iter().map(Report::json).collect();
484            let value = match all.as_slice() {
485                [one] if !list => one.clone(),
486                _ => Value::Array(all),
487            };
488            serde_json::to_string_pretty(&value).unwrap_or_default() + "\n"
489        }
490        "jsonl" => reports.iter().map(Report::trace).collect(),
491        "junit" => Report::junit(reports),
492        "html" => crate::html::page(reports),
493        _ => reports.iter().map(|r| format!("{}\n\n", r.text())).collect(),
494    }
495}
