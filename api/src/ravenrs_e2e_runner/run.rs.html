14/// The `.raventest` files at `path`: the file itself, or every one in the folder, in name order.
15pub fn files_at(path: &Path) -> Result<Vec<PathBuf>, String> {
16    let mut files: Vec<PathBuf> = if path.is_dir() {
17        std::fs::read_dir(path)
18            .map_err(|e| format!("{}: {e}", path.display()))?
19            .flatten()
20            .map(|entry| entry.path())
21            .filter(|p| p.extension().is_some_and(|ext| ext == EXTENSION))
22            .collect()
23    } else if path.is_file() {
24        vec![path.to_path_buf()]
25    } else {
26        return Err(format!("{} não existe", path.display()));
27    };
28    files.sort();
29    if files.is_empty() {
30        return Err(format!("{} não tem arquivos .{EXTENSION}", path.display()));
31    }
32    Ok(files)
33}

35/// The scenarios in `files`, read; the first that cannot be read says its file and line.
36pub fn read(files: &[PathBuf]) -> Result<Vec<Scenario>, String> {
37    files
38        .iter()
39        .map(|file| {
40            let text = std::fs::read_to_string(file).map_err(|e| format!("{}: {e}", file.display()))?;
41            Scenario::parse(&text).map_err(|e| format!("{}: {e}", file.display()))
42        })
43        .collect()
44}

46/// How a run goes beyond its blocks: whether it films, and where it keeps what it makes.
47#[derive(Clone, Debug, Default, PartialEq, Eq)]
48pub struct RunOptions {
49    /// When to keep the film of the apps it opens.
50    pub film: FilmMode,
51    /// Where each run is kept, a folder per scenario; films go there too.
52    pub out: Option<PathBuf>,
53    /// With or without the app's window, at what pace, and whether it stays at the end.
54    pub window: Window,
55    /// What the person watching asks for (pause, one more block, stop), when they can ask.
56    pub control: Option<Control>,
57}
58
59impl RunOptions {
60    /// The folder of the run called `title`, where its films go.
61    pub fn folder_of(&self, title: &str) -> Option<PathBuf> {
62        self.out.as_ref().map(|out| out.join(crate::report::folder_name(title)))
63    }
64}

66/// The title of a run of `scenarios` read from `path`: the scenario's name, or how many there
67/// are and where, said from where it was called.
68pub fn title_of(path: &Path, scenarios: &[Scenario]) -> String {
69    match scenarios {
70        [one] => one.name.clone(),
71        many => {
72            let here = std::env::current_dir().ok();
73            let shown = here.as_deref().and_then(|h| path.strip_prefix(h).ok()).unwrap_or(path);
74            format!("{} cenários de {}", many.len(), shown.display())
75        }
76    }
77}

79/// Runs the scenario files at `path` (one file, or every one in a folder), opening apps with
80/// `open`, and reports them all as one run under a title that names them.
81pub fn run_files(path: &Path, open: Opener) -> Result<Report, String> {
82    run_files_with(path, open, &RunOptions::default())
83}
84
85/// [`run_files`] as `options` say: filming, and where to keep the films.
86pub fn run_files_with(path: &Path, open: Opener, options: &RunOptions) -> Result<Report, String> {
87    let scenarios = read(&files_at(path)?)?;
88    let title = title_of(path, &scenarios);
89    Ok(run_with(&title, &scenarios, open, options))
90}

92/// Runs `scenarios` one after the other, each from its first block, and stops at the first
93/// block that fails: the blocks after it are reported as not run. Each scenario's app ends
94/// with it.
95pub fn run(title: &str, scenarios: &[Scenario], open: Opener) -> Report {
96    run_with(title, scenarios, open, &RunOptions::default())
97}

99/// [`run`] as `options` say.
100pub fn run_with(title: &str, scenarios: &[Scenario], open: Opener, options: &RunOptions) -> Report {
101    run_kept(title, scenarios, open, options).0
102}

104/// [`run_files_with`], also giving the window left open at the end, when `options` keep it
105/// ([`crate::window::KeepOpen`]): print the report, then tell the person and wait on it.
106pub fn run_files_kept(
107    path: &Path,
108    open: Opener,
109    options: &RunOptions,
110) -> Result<(Report, Option<Inspection>), String> {
111    let scenarios = read(&files_at(path)?)?;
112    let title = title_of(path, &scenarios);
113    Ok(run_kept(&title, &scenarios, open, options))
114}

116/// [`run_with`], also giving the window left open at the end, when `options` keep it.
117pub fn run_kept(
118    title: &str,
119    scenarios: &[Scenario],
120    open: Opener,
121    options: &RunOptions,
122) -> (Report, Option<Inspection>) {
123    let started = Instant::now();
124    let blocks: Vec<_> = scenarios.iter().flat_map(|s| s.blocks.iter()).collect();
125    let mut driver = Driver::new(open)
126        .filming(options.film, options.folder_of(title))
127        .presenting(options.window, options.control.clone());
128    let mut outcomes: Vec<Outcome> = Vec::new();
129    for block in &blocks {
130        if !driver.turn() {
131            break;
132        }
133        let id = format!("b{}", outcomes.len() + 1);
134        let outcome = driver.step(Step::of(block), &id, block.kind.word(), &block.title, &block.line());
135        let failed = outcome.error.is_some();
136        outcomes.push(outcome);
137        if failed {
138            break;
139        }
140    }
141    let failed = outcomes.iter().any(|o| o.error.is_some());
142    driver.end(failed);
143    let not_run = blocks.iter().skip(outcomes.len()).map(|b| b.title.clone()).collect();
144    let report = Report {
145        scenario: title.to_owned(),
146        outcomes,
147        total: blocks.len(),
148        cached_at: None,
149        duration_ms: u64::try_from(started.elapsed().as_millis()).unwrap_or(u64::MAX),
150        apart: true,
151        not_run,
152        films: driver.films(),
153        film_note: driver.film_note(),
154        window: options.window.mode,
155        slow_mo_ms: u64::try_from(options.window.slow_mo.as_millis()).unwrap_or(u64::MAX),
156        stopped: driver.stopped(),
157    };
158    (report, driver.inspection())
159}
