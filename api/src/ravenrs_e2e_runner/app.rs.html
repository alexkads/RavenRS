1//! The apps a scenario opens apart, in a child process with no window, over the test channel.
2//!
3//! Shared by Raven Test Studio and `cargo ravenrs e2e`. The runner keeps no list of apps of its own. An app declares itself in its `Cargo.toml`, in
4//! `[package.metadata.ravenrs.e2e]`: the name its scenarios open (`app`), the binary, the first
5//! version that speaks the channel (`since`), a variable that points at another copy, and where
6//! its installer puts it. The Studio learns an app from that declaration when a project is opened
7//! with `--project` ([`crate::project`]), and knows Raven Finanças by the declaration in its
8//! own `Cargo.toml`, read when the Studio is built, so the installed Studio can open the installed
9//! app with no checkout.
10//!
11//! An app is found the way a person has it: a path in its variable; for an app of the project
12//! the Studio was pointed at, its code built with Cargo in that project; then the copy the
13//! installer put on this computer; then, inside the RavenRS checkout, its code. An installed copy
14//! from before the test channel is never opened with `--e2e-serve`: it would read the option as
15//! the path of a database and open its window.
16

27/// An app the Studio opens apart, as its `Cargo.toml` declares it.
28#[derive(Clone, Debug, Default, PartialEq, Eq)]
29pub struct App {
30    /// The name on a scenario's Open block.
31    pub name: String,
32    /// Its Cargo package.
33    pub package: String,
34    /// The binary of the package that is the app.
35    pub binary: String,
36    /// The first version that speaks the test channel; empty when any does.
37    pub since: String,
38    /// A variable that points at another copy of its binary.
39    pub variable: String,
40    /// Where its installer puts it, from the home folder.
41    pub installed: Vec<String>,
42    /// Cargo features enabled only when this tool builds the app for a test.
43    pub features: Vec<String>,
44    /// The project to build its code in with Cargo, and whether that comes before an
45    /// installed copy (it does for a project the Studio was pointed at).
46    pub source: Option<PathBuf>,
47    /// Whether its code comes before the installed copy.
48    pub source_first: bool,
49    /// A binary to open, before anything else: a test opens the Studio it was built with.
50    pub binary_path: Option<PathBuf>,
51}

53impl App {
54    /// The app a package declares in `[package.metadata.ravenrs.e2e]`, from the JSON of
55    /// `cargo metadata`; `None` when it declares none.
56    pub fn declared(package: &Value) -> Option<Self> {
57        let e2e = package.pointer("/metadata/ravenrs/e2e")?;
58        let text = |key: &str| e2e.get(key).and_then(Value::as_str).unwrap_or_default().to_owned();
59        let name = package.get("name").and_then(Value::as_str).unwrap_or_default().to_owned();
60        let installed = e2e
61            .get("installed")
62            .and_then(Value::as_array)
63            .map(|list| list.iter().filter_map(Value::as_str).map(str::to_owned).collect())
64            .unwrap_or_default();
65        Some(Self {
66            name: Some(text("app")).filter(|n| !n.is_empty()).unwrap_or_else(|| name.clone()),
67            binary: Some(text("binary")).filter(|b| !b.is_empty()).unwrap_or_else(|| name.clone()),
68            package: name,
69            since: text("since"),
70            variable: text("variable"),
71            installed,
72            features: features_of(e2e).ok()?,
73            source: None,
74            source_first: false,
75            binary_path: None,
76        })
77    }
78
79    /// The same declaration read from the text of a `Cargo.toml`, for the app the Studio knows
80    /// when it is built. TOML syntax is read by the same parser used for suite configuration;
81    /// its declaration is then interpreted exactly as Cargo's JSON metadata.
82    pub fn declared_in(manifest: &str) -> Option<Self> {
83        let document: toml_edit::DocumentMut = manifest.parse().ok()?;
84        let package = document.get("package")?;
85        let name = package.get("name")?.as_str()?;
86        let declaration = package.get("metadata")?.get("ravenrs")?.get("e2e")?.as_table_like()?;
87        let e2e: serde_json::Map<String, Value> = declaration
88            .iter()
89            .map(|(key, item)| {
90                let value = if let Some(text) = item.as_str() {
91                    Value::String(text.to_owned())
92                } else if let Some(array) = item.as_array() {
93                    Value::Array(
94                        array
95                            .iter()
96                            .map(|item| {
97                                item.as_str().map_or(Value::Null, |text| Value::String(text.to_owned()))
98                            })
99                            .collect(),
100                    )
101                } else {
102                    Value::Null
103                };
104                (key.to_owned(), value)
105            })
106            .collect();
107        Self::declared(&serde_json::json!({ "name": name, "metadata": { "ravenrs": { "e2e": e2e } } }))
108    }
109
110    pub(crate) fn declaration_error(package: &Value) -> Option<String> {
111        let e2e = package.pointer("/metadata/ravenrs/e2e")?;
112        features_of(e2e).err()
113    }
114}

116fn features_of(e2e: &Value) -> Result<Vec<String>, String> {
117    let Some(value) = e2e.get("features") else { return Ok(Vec::new()) };
118    let why =
119        || "package.metadata.ravenrs.e2e.features precisa ser uma lista de textos não vazios".to_owned();
120    value
121        .as_array()
122        .ok_or_else(why)?
123        .iter()
124        .map(|feature| {
125            feature.as_str().filter(|text| !text.trim().is_empty()).map(str::to_owned).ok_or_else(why)
126        })
127        .collect()
128}

130/// The apps learned from projects opened in this run, before the ones the Studio knows.
131static LEARNED: Mutex<Vec<App>> = Mutex::new(Vec::new());
132
133/// Makes `app` known by its name, as an app of a project the Studio was pointed at: its code
134/// there comes before any installed copy.
135pub fn learn(app: App) {
136    if let Ok(mut learned) = LEARNED.lock() {
137        learned.retain(|known| known.name != app.name);
138        learned.push(app);
139    }
140}
141
142/// Whether `name` is an app learned from a project the Studio was pointed at.
143pub fn learned(name: &str) -> bool {
144    LEARNED.lock().is_ok_and(|apps| apps.iter().any(|app| app.name == name))
145}
146
147/// Apps a tool knows by itself, after the learned ones: the Studio knows Raven Finanças.
148static KNOWN: Mutex<Vec<App>> = Mutex::new(Vec::new());
149
150/// Makes `app` known by its name, after any app learned from a project: an app a tool brings
151/// the declaration of, such as the Studio with Raven Finanças.
152pub fn know(app: App) {
153    if let Ok(mut known) = KNOWN.lock()
154        && !known.iter().any(|k| k.name == app.name)
155    {
156        known.push(app);
157    }
158}
159
160/// The app named `name`: one learned from a project, else one a tool made known.
161pub fn known(name: &str) -> Option<App> {
162    let find = |apps: &Mutex<Vec<App>>| {
163        apps.lock().ok().and_then(|apps| apps.iter().find(|app| app.name == name).cloned())
164    };
165    find(&LEARNED).or_else(|| find(&KNOWN))
166}

168/// Whether the apps this process opens show their window: set once by the tool from `--headed`
169/// (see [`crate::window`]), read by every Open block of every run.
170static WINDOW: Mutex<crate::window::WindowMode> = Mutex::new(crate::window::WindowMode::Headless);
171
172/// Opens the apps of every run of this process with their window, or without it.
173pub fn set_window(mode: crate::window::WindowMode) {
174    if let Ok(mut window) = WINDOW.lock() {
175        *window = mode;
176    }
177}
178
179/// Whether the apps this process opens show their window.
180pub fn window() -> crate::window::WindowMode {
181    WINDOW.lock().map(|w| *w).unwrap_or_default()
182}

184/// How long an installed app has to say hello.
185pub const READY_TIME: Duration = Duration::from_secs(20);
186/// How long Cargo may build an app before its process group is stopped.
187pub const BUILD_TIME: Duration = Duration::from_secs(900);

189/// The scenarios the app named `name` brings, read from its hello: the `.raventest` files of its
190/// `tests/e2e/`, embedded in the app. `None` when the Studio does not know the app.
191/// Opening failures keep their structured code and diagnostics; absent or malformed scenarios
192/// return a plain failure without inventing a build/start code.
193pub fn scenarios(name: &str) -> Option<Result<Vec<ravenrs_e2e_wire::Scenario>, Box<Failure>>> {
194    let app = known(name)?;
195    Some(start(&app).and_then(|client| {
196        if client.scenarios().is_empty() {
197            return Err(Box::new(Failure::plain(format!(
198                "O {} aberto não traz cenários na pasta tests/e2e; atualize-o, ou rode um arquivo .raventest com --run",
199                app.name
200            ))));
201        }
202        client
203            .scenarios()
204            .iter()
205            .map(|text| ravenrs_e2e_wire::Scenario::parse(text).map_err(|e| Box::new(Failure::plain(format!("cenário do {}: {e}", app.name)))))
206            .collect()
207    }))
208}

210/// Opens the app named `name` apart, if it is one the Studio knows.
211pub fn open(name: &str) -> Option<Result<Session, Box<Failure>>> {
212    let app = known(name)?;
213    Some(start(&app).map(Session::Apart))
214}

216fn start(app: &App) -> Result<Client, Box<Failure>> {
217    let process = match find(app).map_err(start_failure)? {
218        Launch::Binary(path) => Process::new(path),
219        Launch::Source(root) => {
220            let binary = crate::build::compile_with_features(
221                &root,
222                &app.package,
223                &app.binary,
224                &app.features,
225                BUILD_TIME,
226            )?;
227            let mut process = Process::new(binary);
228            process.current_dir(root);
229            process
230        }
231    };
232    spawn(process, &app.name)
233}

235/// Opens `process` on the test channel, with its window when this process asks for it
236/// ([`set_window`]): an app that cannot show it, or a computer with no screen, is refused with
237/// `E2E_NO_WINDOW` before any step, never run with no window in its place.
238fn spawn(mut process: Process, name: &str) -> Result<Client, Box<Failure>> {
239    let headed = window() == crate::window::WindowMode::Headed;
240    if headed {
241        crate::window::graphic_session().map_err(|why| no_window(why, None))?;
242    }
243    process.arg(SERVE_FLAG);
244    if headed {
245        process.arg(WINDOW_FLAG);
246    }
247    let client = Client::spawn(process, READY_TIME).map_err(|failure| Box::new(failure.into()))?;
248    if headed && !client.window() {
249        return Err(no_window(
250            format!("o {name} aberto não mostra a janela no canal de teste"),
251            Some(format!(
252                "atualize o {name}, ou sirva-o com ravenrs_e2e_wire::Headed e ravenrs::native::run_driven; sem --headed ele roda sem janela"
253            )),
254        ));
255    }
256    Ok(client)
257}

259fn no_window(detail: String, fix: Option<String>) -> Box<Failure> {
260    let mut failure = ravenrs_e2e_wire::Failure::of(codes::NO_WINDOW, detail);
261    failure.fix = fix;
262    Box::new(failure.into())
263}

265enum Launch {
266    Binary(PathBuf),
267    Source(PathBuf),
268}

270/// Installed and explicit binaries run directly; source projects compile first.
271fn find(app: &App) -> Result<Launch, String> {
272    if let Some(path) = &app.binary_path {
273        return Ok(Launch::Binary(path.clone()));
274    }
275    if let Some(path) =
276        Some(&app.variable).filter(|v| !v.is_empty()).and_then(std::env::var_os).map(PathBuf::from)
277    {
278        return Ok(Launch::Binary(path));
279    }
280    if app.source_first
281        && let Some(root) = &app.source
282    {
283        return Ok(Launch::Source(root.clone()));
284    }
285    let found = installed(app).into_iter().map(|path| {
286        let version = version_of(&path);
287        (path, version)
288    });
289    let too_old = match pick_installed(found, &app.since) {
290        Ok(path) => return Ok(Launch::Binary(path)),
291        Err(too_old) => too_old,
292    };
293    if let Some(root) = &app.source {
294        return Ok(Launch::Source(root.clone()));
295    }
296    Err(match too_old {
297        Some(v) => format!(
298            "O {} instalado é a versão {v}, e o teste pelo canal começa na {}: atualize o {} e rode de novo",
299            app.name, app.since, app.name
300        ),
301        None => format!(
302            "O {} não está instalado neste computador. Instale-o, rode os cenários dentro do projeto dele (cargo ravenrs e2e run, ou --project no Studio), ou diga onde ele está em {}",
303            app.name,
304            if app.variable.is_empty() { "uma variável de ambiente" } else { &app.variable }
305        ),
306    })
307}

309/// The first installed copy that speaks the test channel (version `since` or later), or the
310/// version of the one that is too old, to tell the person to update it. A copy that does not
311/// say its version is not opened.
312pub fn pick_installed(
313    found: impl IntoIterator<Item = (PathBuf, Option<String>)>,
314    since: &str,
315) -> Result<PathBuf, Option<String>> {
316    let mut too_old = None;
317    for (path, version) in found {
318        match version {
319            Some(v) if !is_older(&v, since) => return Ok(path),
320            Some(v) => too_old = Some(v),
321            None => {}
322        }
323    }
324    Err(too_old)
325}

327/// Opens the app at `binary` apart, as its variable does.
328pub fn open_from(binary: &Path) -> Result<Session, Box<Failure>> {
329    let name = binary.file_stem().map(|s| s.to_string_lossy().into_owned()).unwrap_or_default();
330    spawn(Process::new(binary), &name).map(Session::Apart)

337/// Where the installer puts the app, for this person.
338fn installed(app: &App) -> Vec<PathBuf> {
339    let home = std::env::var_os("HOME").or_else(|| std::env::var_os("USERPROFILE")).map(PathBuf::from);
340    let mut places: Vec<PathBuf> = Vec::new();
341    if cfg!(target_os = "macos") {
342        places.extend(
343            app.installed.iter().filter(|p| p.starts_with("Applications/")).map(|p| Path::new("/").join(p)),
344        );
345    }
346    if let Some(home) = home {
347        places.extend(app.installed.iter().map(|tail| home.join(tail)));
348    }
349    places.into_iter().filter(|p| p.is_file()).collect()
350}

352/// The version the app says with `--versao`, which every installed RavenRS app answers.
353fn version_of(path: &Path) -> Option<String> {
354    let out = Process::new(path).arg("--versao").output().ok()?;
355    let text = String::from_utf8(out.stdout).ok()?;
356    let line = text.lines().next()?.trim().to_owned();
357    (out.status.success() && line.split('.').all(|part| part.parse::<u32>().is_ok())).then_some(line)
358}

360/// Whether version `a` comes before `b`, part by part.
361pub fn is_older(a: &str, b: &str) -> bool {
362    let parts = |v: &str| v.split('.').map(|p| p.parse::<u32>().unwrap_or(0)).collect::<Vec<_>>();
363    parts(a) < parts(b)
364}

366/// The RavenRS checkout this process runs in, when it has a package called `package` among its
367/// examples or tools.
368pub fn checkout_with(package: &str) -> Option<PathBuf> {
369    let here = std::env::current_dir().ok()?;
370    here.ancestors()
371        .find(|dir| {
372            ["examples", "tools"].iter().any(|group| {
373                std::fs::read_dir(dir.join(group)).into_iter().flatten().flatten().any(|entry| {
374                    std::fs::read_to_string(entry.path().join("Cargo.toml"))
375                        .is_ok_and(|text| text.contains(&format!("name = \"{package}\"")))
376                })
377            })
378        })
379        .map(Path::to_path_buf)
