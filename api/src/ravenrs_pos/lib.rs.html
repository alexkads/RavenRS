17/// What was read at the counter: a code, and how many of it.
18#[derive(Clone, Debug, PartialEq, Eq)]
19pub struct Leitura {
20    /// How many.
21    pub quantidade: Decimal,
22    /// The code.
23    pub codigo: String,
24}
25
26/// Reads `code`, `3*code` or `1,5*code`. Says why when it cannot.
27pub fn ler_codigo(texto: &str) -> Result<Leitura, String> {
28    let texto = texto.trim();
29    let (quantidade, codigo) = match texto.split_once('*') {
30        Some((quantidade, codigo)) => {
31            let quantidade: Decimal =
32                quantidade.trim().replace(',', ".").parse().map_err(|_| {
33                    format!("Quantidade \u{201c}{}\u{201d} não é um número.", quantidade.trim())
34                })?;
35            (quantidade, codigo.trim())
36        }
37        None => (Decimal::ONE, texto),
38    };
39    if codigo.is_empty() {
40        return Err("Falta o código do produto.".to_owned());
41    }
42    if quantidade <= Decimal::ZERO {
43        return Err("A quantidade precisa ser maior que zero.".to_owned());
44    }
45    Ok(Leitura { quantidade, codigo: codigo.to_owned() })
46}

48/// One line of a sale, with what a fiscal document asks of it.
49#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
50pub struct Linha {
51    /// The product's own code.
52    pub codigo: String,
53    /// Its barcode, if it has one.
54    pub gtin: Option<String>,
55    /// What it is.
56    pub descricao: String,
57    /// The unit it is sold by.
58    pub unidade: String,
59    /// The price of each.
60    pub preco: Decimal,
61    /// How many.
62    pub quantidade: Decimal,
63    /// NCM.
64    pub ncm: String,
65    /// CFOP.
66    pub cfop: String,
67    /// CEST, for goods under tax substitution.
68    pub cest: Option<String>,
69    /// CSOSN.
70    pub csosn: String,
71}
72
73impl Linha {
74    /// What this line comes to.
75    pub fn total(&self) -> Decimal {
76        (self.preco * self.quantidade)
77            .round_dp_with_strategy(2, rust_decimal::RoundingStrategy::MidpointAwayFromZero)
78    }
79}

81/// A sale being made.
82#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
83pub struct Carrinho {
84    /// Its lines, in the order they came.
85    pub linhas: Vec<Linha>,
86    /// The CPF or CNPJ of who is buying, when they asked for it on the note.
87    pub cliente: Option<String>,
88    /// What goes along on the document.
89    pub observacao: Option<String>,
90}
91
92impl Carrinho {
93    /// Adds a line.
94    pub fn adicionar(&mut self, linha: Linha) {
95        self.linhas.push(linha);
96    }
97
98    /// Takes out the last line, and gives it.
99    pub fn cancelar_ultimo(&mut self) -> Option<Linha> {
100        self.linhas.pop()
101    }
102
103    /// Takes out the line in this position, from 1 as the receipt counts.
104    pub fn cancelar_item(&mut self, numero: usize) -> Option<Linha> {
105        let at = numero.checked_sub(1)?;
106        (at < self.linhas.len()).then(|| self.linhas.remove(at))
107    }
108
109    /// What it all comes to.
110    pub fn total(&self) -> Decimal {
111        self.linhas.iter().map(Linha::total).sum()
112    }
113
114    /// The last line.
115    pub fn ultimo(&self) -> Option<&Linha> {
116        self.linhas.last()
117    }
118
119    /// Whether it has nothing.
120    pub fn vazio(&self) -> bool {
121        self.linhas.is_empty()
122    }
123
124    /// As text, to keep it and pick it up later.
125    pub fn para_texto(&self) -> String {
126        serde_json::to_string(self).unwrap_or_default()
127    }
128
129    /// From what [`para_texto`](Self::para_texto) gave.
130    pub fn de_texto(texto: &str) -> Result<Self, String> {
131        serde_json::from_str(texto).map_err(|error| error.to_string())
132    }
133}

135/// How someone pays.
136#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
137pub enum Forma {
138    /// Cash.
139    #[default]
140    Dinheiro,
141    /// Debit card.
142    Debito,
143    /// Credit card.
144    Credito,
145    /// Pix.
146    Pix,
147}
148
149impl Forma {
150    /// All of them, in the order of their keys.
151    pub const TODAS: [Self; 4] = [Self::Dinheiro, Self::Debito, Self::Credito, Self::Pix];
152
153    /// What people call it.
154    pub fn nome(self) -> &'static str {
155        match self {
156            Self::Dinheiro => "Dinheiro",
157            Self::Debito => "Cartão de débito",
158            Self::Credito => "Cartão de crédito",
159            Self::Pix => "Pix",
160        }
161    }
162
163    /// Whether someone can hand over more than the total and get change.
164    pub fn tem_troco(self) -> bool {
165        self == Self::Dinheiro
166    }
167}

169/// Reads an amount as people type it: `1.234,56`, `1234,56` or `1234.56`.
170pub fn ler_valor(texto: &str) -> Option<Decimal> {
171    let texto = texto.trim().trim_start_matches("R$").trim();
172    let limpo = if texto.contains(',') { texto.replace('.', "").replace(',', ".") } else { texto.to_owned() };
173    limpo.parse().ok()
174}
175
176/// An amount as people read it: `1.234,56`.
177pub fn valor_em_texto(valor: Decimal) -> String {
178    let texto = format!("{:.2}", valor.round_dp(2));
179    let (inteiro, centavos) = texto.split_once('.').unwrap_or((&texto, "00"));
180    let (sinal, digitos) = inteiro.strip_prefix('-').map_or(("", inteiro), |resto| ("-", resto));
181    let mut agrupado = String::new();
182    for (posicao, digito) in digitos.chars().enumerate() {
183        if posicao > 0 && (digitos.len() - posicao) % 3 == 0 {
184            agrupado.push('.');
185        }
186        agrupado.push(digito);
187    }
188    format!("{sinal}{agrupado},{centavos}")
189}
190
191/// What comes of paying `recebido` for `total` in this way, or why it is not enough.
192pub fn troco(forma: Forma, total: Decimal, recebido: Decimal) -> Result<Decimal, String> {
193    if recebido < total {
194        return Err(format!("Faltam R$ {}.", valor_em_texto(total - recebido)));
195    }
196    if recebido > total && !forma.tem_troco() {
197        return Err(format!("{} não tem troco: o valor é R$ {}.", forma.nome(), valor_em_texto(total)));
198    }
199    Ok(recebido - total)
200}

221    #[test]
222    fn a_code_can_come_with_how_many() {
223        assert_eq!(ler_codigo("789").map(|l| l.quantidade), Ok(Decimal::ONE));
224        let leitura = ler_codigo(" 3 * 789 ").unwrap_or_else(|e| panic!("{e}"));
225        assert_eq!((leitura.quantidade, leitura.codigo.as_str()), (Decimal::from(3), "789"));
226        assert_eq!(ler_codigo("1,5*789").map(|l| l.quantidade.to_string()), Ok("1.5".to_owned()));
227        assert!(ler_codigo("x*789").is_err());
228        assert!(ler_codigo("0*789").is_err());
229        assert!(ler_codigo("3*").is_err());
230        assert!(ler_codigo("  ").is_err());
231    }
232
233    #[test]
234    fn totals_do_not_lose_a_cent() {
235        let mut carrinho = Carrinho::default();
236        carrinho.adicionar(linha("0.10", "3"));
237        carrinho.adicionar(linha("5.49", "3"));
238        assert_eq!(carrinho.total().to_string(), "16.77");
239        assert_eq!(carrinho.ultimo().map(Linha::total).map(|t| t.to_string()), Some("16.47".to_owned()));
240    }

242    #[test]
243    fn lines_come_out_by_position_or_the_last() {
244        let mut carrinho = Carrinho::default();
245        carrinho.adicionar(linha("1", "1"));
246        carrinho.adicionar(linha("2", "1"));
247        carrinho.adicionar(linha("3", "1"));
248        assert_eq!(carrinho.cancelar_item(0), None);
249        assert_eq!(carrinho.cancelar_item(9), None);
250        assert_eq!(carrinho.cancelar_item(1).map(|l| l.preco), Some(Decimal::ONE));
251        assert_eq!(carrinho.cancelar_ultimo().map(|l| l.preco), Some(Decimal::from(3)));
252        assert_eq!(carrinho.linhas.len(), 1);
253    }
