1//! What a price has to cover: from a cost, the taxes and the fixed costs to the price and the markup,
2//! and back from a price to what is left of it.
3
4use rust_decimal::{Decimal, RoundingStrategy};
5use serde::{Deserialize, Serialize};
6
7const CEM: Decimal = Decimal::ONE_HUNDRED;
8/// Half a cent goes up, as on a fiscal document.
9const MEIO: RoundingStrategy = RoundingStrategy::MidpointAwayFromZero;
10
11/// What goes into the price of a product. Percentages are of the selling price, as 12.5 for 12.5%.
12#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
13pub struct Composicao {
14    /// What a unit was bought for.
15    pub custo: Decimal,
16    /// Freight, for a unit.
17    pub frete: Decimal,
18    /// Other costs of buying, for a unit, such as tax paid at purchase that cannot be recovered.
19    pub outros_custos: Decimal,
20    /// Taxes on the sale, such as the rate of the Simples Nacional.
21    pub impostos: Decimal,
22    /// The share of fixed costs each sale carries.
23    pub custos_fixos: Decimal,
24    /// Commission and card fees.
25    pub comissao: Decimal,
26    /// The profit wanted.
27    pub margem: Decimal,
28}
29
30/// What comes out of a [`Composicao`], or of a price given.
31#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
32pub struct Formacao {
33    /// What a unit costs, with freight and the rest.
34    pub custo_total: Decimal,
35    /// The price.
36    pub preco: Decimal,
37    /// Price divided by total cost: 2.0 sells for twice the cost.
38    pub markup: Decimal,
39    /// What goes to taxes.
40    pub valor_impostos: Decimal,
41    /// What goes to fixed costs.
42    pub valor_custos_fixos: Decimal,
43    /// What goes to commission and fees.
44    pub valor_comissao: Decimal,
45    /// What is left as profit.
46    pub lucro: Decimal,
47    /// The profit as a share of the price, in percent.
48    pub margem: Decimal,
49    /// The price at which there is no profit and no loss.
50    pub ponto_de_equilibrio: Decimal,
51}
52
53impl Composicao {
54    /// What is taken out of each price, before the profit, in percent.
55    fn despesas(&self) -> Decimal {
56        self.impostos + self.custos_fixos + self.comissao
57    }
58
59    fn custo_total(&self) -> Decimal {
60        self.custo + self.frete + self.outros_custos
61    }
62
63    fn conferir(&self) -> Result<(), String> {
64        let negativos =
65            [self.custo, self.frete, self.outros_custos, self.impostos, self.custos_fixos, self.comissao];
66        if negativos.iter().any(|value| *value < Decimal::ZERO) {
67            return Err("Custos e percentuais não podem ser negativos.".to_owned());
68        }
69        Ok(())
70    }
71
72    /// The price that gives the margin wanted, with what it comes to.
73    pub fn calcular(&self) -> Result<Formacao, String> {
74        self.conferir()?;
75        let soma = self.despesas() + self.margem;
76        if soma >= CEM {
77            return Err(format!(
78                "Impostos, custos fixos, comissão e margem somam {}%. Precisam ficar abaixo de 100%.",
79                soma.normalize()
80            ));
81        }
82        if self.custo_total() <= Decimal::ZERO {
83            return Err("Informe o custo.".to_owned());
84        }
85        let preco = (self.custo_total() / ((CEM - soma) / CEM)).round_dp_with_strategy(2, MEIO);
86        Ok(self.para_o_preco(preco))
87    }
88
89    /// What is left at this price, whatever margin was wanted.
90    pub fn para_o_preco(&self, preco: Decimal) -> Formacao {
91        let custo_total = self.custo_total();
92        let parte = |percentual: Decimal| (preco * percentual / CEM).round_dp_with_strategy(2, MEIO);
93        let (valor_impostos, valor_custos_fixos, valor_comissao) =
94            (parte(self.impostos), parte(self.custos_fixos), parte(self.comissao));
95        let lucro = preco - custo_total - valor_impostos - valor_custos_fixos - valor_comissao;
96        let despesas = self.despesas();
97        Formacao {
98            custo_total,
99            preco,
100            markup: if custo_total > Decimal::ZERO {
101                (preco / custo_total).round_dp(4)
102            } else {
103                Decimal::ZERO
104            },
105            valor_impostos,
106            valor_custos_fixos,
107            valor_comissao,
108            lucro,
109            margem: if preco > Decimal::ZERO {
110                (lucro / preco * CEM).round_dp_with_strategy(2, MEIO)
111            } else {
112                Decimal::ZERO
113            },
114            ponto_de_equilibrio: if despesas < CEM {
115                (custo_total / ((CEM - despesas) / CEM)).round_dp_with_strategy(2, MEIO)
116            } else {
117                Decimal::ZERO
118            },
119        }
120    }
121}

123/// A price rounded up to end in these cents, as a shelf price does: 5,37 with 90 gives 5,90.
124pub fn arredondar_preco(preco: Decimal, centavos: u8) -> Decimal {
125    let inteiro = preco.trunc();
126    let final_desejado = Decimal::new(i64::from(centavos.min(99)), 2);
127    let candidato = inteiro + final_desejado;
128    if candidato >= preco { candidato } else { candidato + Decimal::ONE }
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134
135    fn d(text: &str) -> Decimal {
136        text.parse().unwrap_or_else(|_| panic!("{text}"))
137    }
138
139    fn composicao() -> Composicao {
140        Composicao {
141            custo: d("10"),
142            frete: d("0.50"),
143            outros_custos: d("0"),
144            impostos: d("6"),
145            custos_fixos: d("10"),
146            comissao: d("4"),
147            margem: d("10"),
148        }
149    }
150
151    #[test]
152    fn the_price_covers_it_all_and_gives_the_margin() {
153        // 10,50 / (1 - 0,30) = 15,00
154        let f = composicao().calcular().unwrap_or_else(|e| panic!("{e}"));
155        assert_eq!(f.preco, d("15"));
156        assert_eq!(f.custo_total, d("10.50"));
157        assert_eq!(f.markup, d("1.4286"));
158        assert_eq!(
159            (f.valor_impostos, f.valor_custos_fixos, f.valor_comissao),
160            (d("0.90"), d("1.50"), d("0.60"))
161        );
162        assert_eq!(f.lucro, d("1.50"));
163        assert_eq!(f.margem, d("10"));
164        assert_eq!(f.ponto_de_equilibrio, d("13.13"));
165    }
166
167    #[test]
168    fn a_price_given_says_what_is_left() {
169        let f = composicao().para_o_preco(d("12"));
170        assert_eq!(f.lucro, d("-0.90"), "below break-even it shows the loss");
171        assert_eq!(f.margem, d("-7.5"));
172        assert_eq!(composicao().para_o_preco(d("0")).margem, d("0"));
173    }
174
175    #[test]
176    fn what_cannot_be_says_why() {
177        let tudo = Composicao { margem: d("80"), ..composicao() };
178        assert!(tudo.calcular().unwrap_err().contains("100%"));
179        assert!(Composicao { custo: d("0"), frete: d("0"), ..composicao() }.calcular().is_err());
180        assert!(Composicao { custo: d("-1"), ..composicao() }.calcular().is_err());
181    }
182
183    #[test]
184    fn a_negative_margin_is_a_price_below_cost_on_purpose() {
185        let f = Composicao { margem: d("-5"), ..composicao() }.calcular().unwrap_or_else(|e| panic!("{e}"));
186        assert!((f.margem - d("-5")).abs() < d("0.1"), "{}", f.margem);
187    }
188
189    #[test]
190    fn shelf_prices_round_up() {
191        assert_eq!(arredondar_preco(d("5.37"), 90), d("5.90"));
192        assert_eq!(arredondar_preco(d("5.95"), 90), d("6.90"));
193        assert_eq!(arredondar_preco(d("5.90"), 90), d("5.90"));
194        assert_eq!(arredondar_preco(d("5.01"), 99), d("5.99"));
195    }
