1//! The display list: an ordered list of primitives in logical pixels.
2//!
3//! Widgets append primitives in painter's order (back to front). Renderers may
4//! batch consecutive primitives of the same kind but must preserve this order.

11/// How a shape is filled.
12#[derive(Clone, Copy, Debug, PartialEq)]
13#[non_exhaustive]
14pub enum Brush {
15    /// A single flat color.
16    Solid(Color),
17    /// A two-stop linear gradient.
18    LinearGradient(LinearGradient),
19}
20
21impl From<Color> for Brush {
22    fn from(c: Color) -> Self {
23        Self::Solid(c)
24    }
25}
26
27impl From<LinearGradient> for Brush {
28    fn from(g: LinearGradient) -> Self {
29        Self::LinearGradient(g)
30    }
31}

33/// A two-stop linear gradient. `start` and `end` are relative to the shape's
34/// rectangle: `(0, 0)` is its top-left corner and `(1, 1)` its bottom-right.
35#[derive(Clone, Copy, Debug, PartialEq)]
36pub struct LinearGradient {
37    /// Gradient start, relative to the shape's rectangle.
38    pub start: Point,
39    /// Gradient end, relative to the shape's rectangle.
40    pub end: Point,
41    /// Color at `start`.
42    pub from: Color,
43    /// Color at `end`.
44    pub to: Color,
45}
46
47impl LinearGradient {
48    /// Top-to-bottom gradient from `from` to `to`.
49    pub const fn vertical(from: Color, to: Color) -> Self {
50        Self { start: Point::new(0.0, 0.0), end: Point::new(0.0, 1.0), from, to }
51    }
52}

63/// A soft drop shadow. `blur` is the Gaussian standard deviation times two,
64/// matching what CSS `box-shadow` calls the blur radius.
65#[derive(Clone, Copy, Debug, PartialEq)]
66pub struct Shadow {
67    /// Shadow offset from the shape, in logical pixels.
68    pub offset: Point,
69    /// Blur radius in logical pixels (CSS `box-shadow` semantics).
70    pub blur: f32,
71    /// Shadow color, usually translucent black.
72    pub color: Color,
73}
74
75impl Shadow {
76    /// The `shadow-soft` token of Raven Dark: `0 8px 24px rgba(0,0,0,0.35)`.
77    pub const fn soft() -> Self {
78        Self { offset: Point::new(0.0, 8.0), blur: 24.0, color: Color::rgba(0.0, 0.0, 0.0, 0.35) }
79    }
80
81    /// The `shadow-raised` token of Raven Dark: `0 16px 40px rgba(0,0,0,0.45)`.
82    pub const fn raised() -> Self {
83        Self { offset: Point::new(0.0, 16.0), blur: 40.0, color: Color::rgba(0.0, 0.0, 0.0, 0.45) }
84    }
85}

87/// A rounded rectangle with optional border and shadow. The renderer draws it with
88/// a signed-distance function, so corners stay crisp at any scale factor.
89#[derive(Clone, Copy, Debug, PartialEq)]
90pub struct Quad {
91    /// Bounds of the rectangle, in logical pixels.
92    pub rect: Rect,
93    /// Corner radii: top-left, top-right, bottom-right, bottom-left.
94    pub corner_radii: [f32; 4],
95    /// Interior fill.
96    pub fill: Brush,
97    /// Optional stroke along the edge.
98    pub border: Option<Border>,
99    /// Optional drop shadow drawn behind the shape.
100    pub shadow: Option<Shadow>,
101}
102
103impl Quad {
104    /// Creates a sharp-cornered quad with no border or shadow.
105    ///
106    /// ```
107    /// use ravenrs_core::{Color, Quad, Rect, Scene};
108    ///
109    /// let mut scene = Scene::new();
110    /// scene.push_quad(Quad::new(Rect::new(0.0, 0.0, 120.0, 32.0), Color::hex(0x3b82f6)).radius(6.0));
111    /// assert_eq!(scene.len(), 1);
112    /// ```
113    pub fn new(rect: Rect, fill: impl Into<Brush>) -> Self {
114        Self { rect, corner_radii: [0.0; 4], fill: fill.into(), border: None, shadow: None }
115    }
116
117    /// Sets all four corner radii to `r` logical pixels.
118    pub fn radius(mut self, r: f32) -> Self {
119        self.corner_radii = [r; 4];
120        self
121    }
122
123    /// Adds a border of `width` logical pixels.
124    pub fn border(mut self, width: f32, color: Color) -> Self {
125        self.border = Some(Border { width, color });
126        self
127    }
128
129    /// Adds a drop shadow.
130    pub fn shadow(mut self, shadow: Shadow) -> Self {
131        self.shadow = Some(shadow);
132        self
133    }
134}

136/// Font bytes plus a stable identity used as a cache key by the glyph atlas.
137#[derive(Clone)]
138pub struct FontRef {
139    /// Unique identity of these font bytes; equal ids must mean equal data.
140    pub id: u64,
141    /// Raw font file bytes (TTF, OTF or a collection).
142    pub data: Arc<dyn AsRef<[u8]> + Send + Sync>,
143    /// Index inside a font collection; 0 for single-font files.
144    pub index: u32,
145}
146
147impl FontRef {
148    /// Borrows the raw font file bytes.
149    pub fn bytes(&self) -> &[u8] {
150        (*self.data).as_ref()
151    }
152}
153
154impl fmt::Debug for FontRef {
155    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
156        f.debug_struct("FontRef").field("id", &self.id).field("index", &self.index).finish()
157    }
158}
159
160impl PartialEq for FontRef {
161    fn eq(&self, other: &Self) -> bool {
162        self.id == other.id && self.index == other.index
163    }

190/// Decoded pixels: RGBA8, straight (not premultiplied) alpha, sRGB, row-major.
191/// Cloning is cheap; the renderer caches the GPU texture by `id`.
192#[derive(Clone)]
193pub struct ImageData {
194    id: u64,
195    width: u32,
196    height: u32,
197    pixels: Arc<[u8]>,
198}
199
200impl ImageData {
201    /// Wraps RGBA8 pixels. Returns `None` when the buffer does not match the size.
202    pub fn from_rgba8(width: u32, height: u32, pixels: impl Into<Arc<[u8]>>) -> Option<Self> {
203        let pixels = pixels.into();
204        if width == 0 || height == 0 || pixels.len() != width as usize * height as usize * 4 {
205            return None;
206        }
207        static NEXT_ID: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(1);
208        let id = NEXT_ID.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
209        Some(Self { id, width, height, pixels })
210    }
211
212    /// Builds an image by evaluating `f(x, y)` for every pixel (procedural art, tests).
213    pub fn from_fn(width: u32, height: u32, f: impl Fn(u32, u32) -> Color) -> Option<Self> {
214        let mut pixels = Vec::with_capacity(width as usize * height as usize * 4);
215        for y in 0..height {
216            for x in 0..width {
217                let c = f(x, y);
218                let to8 = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
219                pixels.extend_from_slice(&[to8(c.r), to8(c.g), to8(c.b), to8(c.a)]);
220            }
221        }
222        Self::from_rgba8(width, height, pixels)
223    }
224
225    /// Stable identity used as the GPU cache key.
226    pub fn id(&self) -> u64 {
227        self.id
228    }
229
230    /// Width in pixels.
231    pub fn width(&self) -> u32 {
232        self.width
233    }
234
235    /// Height in pixels.
236    pub fn height(&self) -> u32 {
237        self.height
238    }
239
240    /// RGBA8 pixel bytes, row-major, `width * height * 4` long.
241    pub fn pixels(&self) -> &[u8] {
242        &self.pixels
243    }
244
245    /// Width divided by height.
246    pub fn aspect_ratio(&self) -> f32 {
247        self.width as f32 / self.height as f32
248    }
249}
250
251impl fmt::Debug for ImageData {
252    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
253        f.debug_struct("ImageData").field("id", &self.id).field("size", &(self.width, self.height)).finish()
254    }
255}
256
257impl PartialEq for ImageData {
258    fn eq(&self, other: &Self) -> bool {
259        self.id == other.id
260    }
261}

308/// A visual change applied to painted content without touching layout: what a
309/// compositor would do to a layer, here done to the display list.
310#[derive(Clone, Copy, Debug, PartialEq)]
311pub struct Effect {
312    /// Multiplies alpha, `0.0` to `1.0`. Each primitive fades on its own, so
313    /// shapes that overlap show through each other while partly faded.
314    pub opacity: f32,
315    /// Movement in logical pixels.
316    pub offset: Point,
317    /// Size multiplier around the pivot.
318    pub scale: f32,
319}
320
321impl Effect {
322    /// No change.
323    pub const IDENTITY: Self = Self { opacity: 1.0, offset: Point::new(0.0, 0.0), scale: 1.0 };
324
325    /// Fully transparent.
326    pub const fn hidden() -> Self {
327        Self { opacity: 0.0, ..Self::IDENTITY }
328    }
329
330    /// Moved by `(x, y)` logical pixels.
331    pub const fn moved(x: f32, y: f32) -> Self {
332        Self { offset: Point::new(x, y), ..Self::IDENTITY }
333    }
334
335    /// Scaled around the pivot.
336    pub const fn scaled(scale: f32) -> Self {
337        Self { scale, ..Self::IDENTITY }
338    }
339
340    /// The same effect with this opacity.
341    pub const fn with_opacity(mut self, opacity: f32) -> Self {
342        self.opacity = opacity;
343        self
344    }
345
346    /// The same effect with this scale.
347    pub const fn with_scale(mut self, scale: f32) -> Self {
348        self.scale = scale;
349        self
350    }
351
352    /// True when applying the effect would change nothing.
353    pub fn is_identity(&self) -> bool {
354        *self == Self::IDENTITY
355    }
356
357    fn point(&self, p: Point, pivot: Point) -> Point {
358        Point::new(
359            pivot.x + (p.x - pivot.x) * self.scale + self.offset.x,
360            pivot.y + (p.y - pivot.y) * self.scale + self.offset.y,
361        )
362    }
363
364    fn rect(&self, r: Rect, pivot: Point) -> Rect {
365        Rect::from_origin_size(self.point(r.origin, pivot), r.size.scale(self.scale))
366    }
367
368    fn color(&self, c: Color) -> Color {
369        c.with_alpha(c.a * self.opacity)
370    }
371
372    fn apply(&self, primitive: &mut Primitive, pivot: Point) {
373        match primitive {
374            Primitive::Quad(q) => {
375                q.rect = self.rect(q.rect, pivot);
376                q.corner_radii = q.corner_radii.map(|r| r * self.scale);
377                q.fill = match q.fill {
378                    Brush::Solid(c) => Brush::Solid(self.color(c)),
379                    Brush::LinearGradient(g) => Brush::LinearGradient(LinearGradient {
380                        from: self.color(g.from),
381                        to: self.color(g.to),
382                        ..g
383                    }),
384                };
385                if let Some(b) = &mut q.border {
386                    b.width *= self.scale;
387                    b.color = self.color(b.color);
388                }
389                if let Some(s) = &mut q.shadow {
390                    s.offset = Point::new(s.offset.x * self.scale, s.offset.y * self.scale);
391                    s.blur *= self.scale;
392                    s.color = self.color(s.color);
393                }
394            }
395            Primitive::Glyphs(run) => {
396                run.size *= self.scale;
397                run.color = self.color(run.color);
398                for g in &mut run.glyphs {
399                    let p = self.point(Point::new(g.x, g.y), pivot);
400                    (g.x, g.y) = (p.x, p.y);
401                }
402            }
403            Primitive::Image(image) => {
404                image.rect = self.rect(image.rect, pivot);
405                image.corner_radii = image.corner_radii.map(|r| r * self.scale);
406                image.opacity *= self.opacity;
407            }
408            Primitive::PushClip(rect) => *rect = self.rect(*rect, pivot),
409            Primitive::PopClip => {}
410        }
411    }
412}
413
414impl Default for Effect {
415    fn default() -> Self {
416        Self::IDENTITY
417    }
418}

420/// An ordered display list. Reused between frames to avoid reallocating.
421#[derive(Clone, Debug, Default)]
422pub struct Scene {
423    primitives: Vec<Primitive>,
424    clear_color: Color,
425}
426
427impl Scene {
428    /// Creates an empty display list with a transparent clear color.
429    pub fn new() -> Self {
430        Self::default()
431    }
432
433    /// Empties the list but keeps its allocation.
434    pub fn clear(&mut self) {
435        self.primitives.clear();
436    }
437
438    /// Sets the color the frame is cleared to before drawing.
439    pub fn set_clear_color(&mut self, color: Color) {
440        self.clear_color = color;
441    }
442
443    /// Color the frame is cleared to before drawing.
444    pub fn clear_color(&self) -> Color {
445        self.clear_color
446    }
447
448    /// Appends a rounded rectangle.
449    pub fn push_quad(&mut self, quad: Quad) {
450        self.primitives.push(Primitive::Quad(quad));
451    }
452
453    /// Appends a glyph run; empty runs are skipped.
454    pub fn push_glyphs(&mut self, run: GlyphRun) {
455        if !run.glyphs.is_empty() {
456            self.primitives.push(Primitive::Glyphs(run));
457        }
458    }
459
460    /// Appends an image.
461    pub fn push_image(&mut self, image: ImageQuad) {
462        self.primitives.push(Primitive::Image(image));
463    }
464
465    /// Clips subsequent primitives to `rect` (intersected with the current clip)
466    /// until [`Scene::pop_clip`].
467    pub fn push_clip(&mut self, rect: Rect) {
468        self.primitives.push(Primitive::PushClip(rect));
469    }
470
471    /// Ends the clip started by the matching [`Scene::push_clip`].
472    pub fn pop_clip(&mut self) {
473        self.primitives.push(Primitive::PopClip);
474    }
475
476    /// Applies `effect` to a range of primitives, scaling around `pivot`.
477    ///
478    /// ```
479    /// use ravenrs_core::{Color, Effect, Point, Primitive, Quad, Rect, Scene};
480    ///
481    /// let mut scene = Scene::new();
482    /// scene.push_quad(Quad::new(Rect::new(0.0, 0.0, 10.0, 10.0), Color::WHITE));
483    /// scene.apply_effect(0..1, Effect::moved(5.0, 0.0), Point::ZERO);
484    /// let Primitive::Quad(q) = &scene.primitives()[0] else { unreachable!() };
485    /// assert_eq!(q.rect.x(), 5.0);
486    /// ```
487    pub fn apply_effect(&mut self, range: std::ops::Range<usize>, effect: Effect, pivot: Point) {
488        if effect.is_identity() {
489            return;
490        }
491        if let Some(primitives) = self.primitives.get_mut(range) {
492            for primitive in primitives {
493                effect.apply(primitive, pivot);
494            }
495        }
496    }
497
498    /// Appends primitives recorded earlier.
499    pub fn extend_from_slice(&mut self, primitives: &[Primitive]) {
500        self.primitives.extend_from_slice(primitives);
501    }
502
503    /// All primitives in painter's order.
504    pub fn primitives(&self) -> &[Primitive] {
505        &self.primitives
506    }
507
508    /// Number of primitives.
509    pub fn len(&self) -> usize {
510        self.primitives.len()
511    }
512
513    /// Returns `true` when no primitives have been pushed.
514    pub fn is_empty(&self) -> bool {
515        self.primitives.is_empty()
516    }
517}

523    #[test]
524    fn image_data_validates_size_and_gets_unique_ids() {
525        assert!(ImageData::from_rgba8(2, 2, vec![0u8; 15]).is_none());
526        let a = ImageData::from_rgba8(2, 2, vec![0u8; 16]).unwrap();
527        let b = ImageData::from_fn(2, 1, |x, _| if x == 0 { Color::WHITE } else { Color::BLACK }).unwrap();
528        assert_ne!(a.id(), b.id());
529        assert_eq!(&b.pixels()[..8], &[255, 255, 255, 255, 0, 0, 0, 255]);
530        assert_eq!(a, a.clone());
531    }

533    #[test]
534    fn empty_glyph_runs_are_skipped() {
535        let mut scene = Scene::new();
536        scene.push_glyphs(GlyphRun {
537            font: FontRef { id: 1, data: Arc::new(Vec::<u8>::new()), index: 0 },
