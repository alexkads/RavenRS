1/// A non-premultiplied RGBA color with components in `0.0..=1.0`, in the sRGB
2/// color space (the space design tokens are written in).
3#[derive(Clone, Copy, Debug, Default, PartialEq)]
4pub struct Color {
5    /// Red component, `0.0..=1.0`.
6    pub r: f32,
7    /// Green component, `0.0..=1.0`.
8    pub g: f32,
9    /// Blue component, `0.0..=1.0`.
10    pub b: f32,
11    /// Alpha (opacity), `0.0` transparent to `1.0` opaque.
12    pub a: f32,
13}
14
15impl Color {
16    /// Fully transparent black.
17    pub const TRANSPARENT: Self = Self::rgba(0.0, 0.0, 0.0, 0.0);
18    /// Opaque black.
19    pub const BLACK: Self = Self::rgb(0.0, 0.0, 0.0);
20    /// Opaque white.
21    pub const WHITE: Self = Self::rgb(1.0, 1.0, 1.0);
22
23    /// Builds a color from sRGB components and alpha, each in `0.0..=1.0`.
24    pub const fn rgba(r: f32, g: f32, b: f32, a: f32) -> Self {
25        Self { r, g, b, a }
26    }
27
28    /// Builds an opaque color from sRGB components in `0.0..=1.0`.
29    pub const fn rgb(r: f32, g: f32, b: f32) -> Self {
30        Self::rgba(r, g, b, 1.0)
31    }
32
33    /// Builds an opaque color from `0xRRGGBB`.
34    pub const fn hex(rgb: u32) -> Self {
35        Self::rgb(
36            ((rgb >> 16) & 0xff) as f32 / 255.0,
37            ((rgb >> 8) & 0xff) as f32 / 255.0,
38            (rgb & 0xff) as f32 / 255.0,
39        )
40    }
41
42    /// Builds a color from `0xRRGGBBAA`.
43    pub const fn hex_alpha(rgba: u32) -> Self {
44        Self::rgba(
45            ((rgba >> 24) & 0xff) as f32 / 255.0,
46            ((rgba >> 16) & 0xff) as f32 / 255.0,
47            ((rgba >> 8) & 0xff) as f32 / 255.0,
48            (rgba & 0xff) as f32 / 255.0,
49        )
50    }
51
52    /// Returns the same color, `factor` times as opaque as it already is.
53    /// Use this on a theme's tokens: in a glass theme they are translucent to
54    /// begin with, and [`Self::with_alpha`] would make them more opaque, not less.
55    ///
56    /// ```
57    /// use ravenrs_core::Color;
58    ///
59    /// let glass = Color::rgba(1.0, 1.0, 1.0, 0.2);
60    /// assert_eq!(glass.faded(0.5).a, 0.1);
61    /// assert_eq!(glass.with_alpha(0.5).a, 0.5);
62    /// ```
63    pub const fn faded(self, factor: f32) -> Self {
64        Self::rgba(self.r, self.g, self.b, self.a * factor)
65    }
66
67    /// Returns the same color with alpha replaced by `a` (`0.0..=1.0`).
68    pub const fn with_alpha(self, a: f32) -> Self {
69        Self::rgba(self.r, self.g, self.b, a)
70    }
71
72    /// Component-wise interpolation in sRGB space, `t` clamped to `0..=1`.
73    pub fn lerp(self, other: Self, t: f32) -> Self {
74        let t = t.clamp(0.0, 1.0);
75        Self::rgba(
76            self.r + (other.r - self.r) * t,
77            self.g + (other.g - self.g) * t,
78            self.b + (other.b - self.b) * t,
79            self.a + (other.a - self.a) * t,
80        )
81    }
82
83    /// Converts to linear-light RGB (alpha unchanged), as needed by `*Srgb` render targets.
84    pub fn to_linear(self) -> [f32; 4] {
85        fn channel(c: f32) -> f32 {
86            if c <= 0.04045 { c / 12.92 } else { ((c + 0.055) / 1.055).powf(2.4) }
87        }
88        [channel(self.r), channel(self.g), channel(self.b), self.a]
89    }
90
91    /// Returns the sRGB components as `[r, g, b, a]`, without linearization.
92    pub fn to_array(self) -> [f32; 4] {
93        [self.r, self.g, self.b, self.a]
94    }
95
96    /// WCAG relative luminance.
97    pub fn luminance(self) -> f32 {
98        let [r, g, b, _] = self.to_linear();
99        0.2126 * r + 0.7152 * g + 0.0722 * b
100    }
101
102    /// WCAG contrast ratio between two opaque colors (1.0 to 21.0).
103    pub fn contrast_ratio(self, other: Self) -> f32 {
104        let (a, b) = (self.luminance(), other.luminance());
105        let (hi, lo) = if a > b { (a, b) } else { (b, a) };
106        (hi + 0.05) / (lo + 0.05)
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    #[test]
115    fn hex_parsing() {
116        assert_eq!(Color::hex(0xff0000), Color::rgb(1.0, 0.0, 0.0));
117        let c = Color::hex_alpha(0xffffff80);
118        assert!((c.a - 128.0 / 255.0).abs() < 1e-6);
119    }
120
121    #[test]
122    fn contrast_black_white_is_21() {
123        assert!((Color::BLACK.contrast_ratio(Color::WHITE) - 21.0).abs() < 0.01);
124    }
125
126    #[test]
127    fn lerp_clamps() {
128        assert_eq!(Color::BLACK.lerp(Color::WHITE, 2.0), Color::WHITE);
129    }
