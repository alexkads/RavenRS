1//! A platform-neutral accessibility tree: what screen readers and other
2//! assistive technology see. Widgets describe themselves with a [`Role`] and
3//! state; platform layers translate the [`AccessTree`] to the operating
4//! system's API (AccessKit on desktop, ARIA in the browser).

8/// What kind of control or content a node is, as assistive technology names it.
9#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
10#[non_exhaustive]
11pub enum Role {
12    /// The application window, the root of the tree.
13    Window,
14    /// A container with no meaning of its own.
15    Group,
16    /// Static text.
17    Label,
18    /// A heading; see [`AccessNode::level`].
19    Heading,
20    /// Activates an action.
21    Button,
22    /// Navigates somewhere.
23    Link,
24    /// A two- or three-state check box.
25    CheckBox,
26    /// An on/off switch.
27    Switch,
28    /// A set of radio buttons.
29    RadioGroup,
30    /// One choice of a radio group.
31    RadioButton,
32    /// Editable single-line text.
33    TextInput,
34    /// Editable multi-line text.
35    MultilineTextInput,
36    /// A value picked along a track.
37    Slider,
38    /// A number changed with increment and decrement buttons.
39    SpinButton,
40    /// A field that opens a list of choices.
41    ComboBox,
42    /// A list of selectable options.
43    ListBox,
44    /// One option of a list box.
45    ListBoxOption,
46    /// A popup menu.
47    Menu,
48    /// A horizontal bar of menus.
49    MenuBar,
50    /// One command in a menu.
51    MenuItem,
52    /// A row of tabs.
53    TabList,
54    /// One tab.
55    Tab,
56    /// The content shown for the selected tab.
57    TabPanel,
58    /// A window over the app; see [`AccessNode::modal`].
59    Dialog,
60    /// An urgent message.
61    Alert,
62    /// A status message (toasts).
63    Status,
64    /// A rich tooltip.
65    Tooltip,
66    /// A row of controls.
67    Toolbar,
68    /// A data grid.
69    Grid,
70    /// A static table.
71    Table,
72    /// A row of a grid or table.
73    Row,
74    /// A cell of a grid or table.
75    Cell,
76    /// A column header of a grid or table.
77    ColumnHeader,
78    /// A list of items.
79    List,
80    /// One item of a list.
81    ListItem,
82    /// An image; its name is the alternative text.
83    Image,
84    /// Progress of a task.
85    ProgressIndicator,
86    /// A visual divider.
87    Separator,
88    /// A region that scrolls.
89    ScrollView,
90    /// A date grid.
91    Calendar,
92    /// The band that ends a page: copyright, links, version (a landmark).
93    ContentInfo,
94    /// A set of links or buttons that move around the app (a landmark).
95    Navigation,
96}

98impl Role {
99    /// Roles whose accessible name comes from the text inside them when no
100    /// explicit name is given, as in ARIA's "name from content".
101    pub fn takes_name_from_content(self) -> bool {
102        matches!(
103            self,
104            Self::Button
105                | Self::Link
106                | Self::CheckBox
107                | Self::Switch
108                | Self::RadioButton
109                | Self::Heading
110                | Self::ListBoxOption
111                | Self::MenuItem
112                | Self::Tab
113                | Self::Cell
114                | Self::ColumnHeader
115                | Self::ListItem
116                | Self::Tooltip
117                | Self::Label
118        )
119    }
120}

122/// State of a check box, switch or toggle button.
123#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
124pub enum Toggled {
125    /// Off or unchecked.
126    False,
127    /// On or checked.
128    True,
129    /// Partly checked (a parent of mixed children).
130    Mixed,
131}
132
133impl From<bool> for Toggled {
134    fn from(on: bool) -> Self {
135        if on { Self::True } else { Self::False }
136    }
137}

152/// What assistive technology can ask a node to do.
153#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
154#[non_exhaustive]
155pub enum AccessAction {
156    /// Activate it, as a click or Enter would.
157    Click,
158    /// Move keyboard focus to it.
159    Focus,
160    /// Raise its value one step.
161    Increment,
162    /// Lower its value one step.
163    Decrement,
164}

166/// One node of the accessibility tree.
167#[derive(Clone, Debug, PartialEq)]
168#[non_exhaustive]
169pub struct AccessNode {
170    /// Identity, stable across frames for the same widget.
171    pub id: u64,
172    /// What the node is.
173    pub role: Role,
174    /// The accessible name: a label, or the text inside for roles that take
175    /// their name from content.
176    pub name: Option<String>,
177    /// Extra description, read after the name.
178    pub description: Option<String>,
179    /// A text value (text fields, combo boxes, sliders with formatted values).
180    pub value: Option<String>,
181    /// A numeric value with its range.
182    pub numeric: Option<NumericValue>,
183    /// Checked state.
184    pub toggled: Option<Toggled>,
185    /// Selected state (tabs, options, rows, days).
186    pub selected: Option<bool>,
187    /// Expanded state of something that opens (menus, combo boxes, disclosures).
188    pub expanded: Option<bool>,
189    /// The control does not respond to input.
190    pub disabled: bool,
191    /// Keyboard focus can be on it.
192    pub focusable: bool,
193    /// While it is open, the rest of the app is inert (dialogs).
194    pub modal: bool,
195    /// Heading level, 1 to 6.
196    pub level: Option<u32>,
197    /// The option highlighted inside a focused text field that drives a list
198    /// (combobox pattern): screen readers announce it as if it had focus.
199    pub active_descendant: Option<u64>,
200    /// Absolute frame in logical pixels.
201    pub bounds: Rect,
202    /// Child node ids in reading order.
203    pub children: Vec<u64>,
204    /// Actions assistive technology may request.
205    pub actions: Vec<AccessAction>,
206}
207
208impl AccessNode {
209    /// A node with no name, state, children or actions.
210    pub fn new(id: u64, role: Role, bounds: Rect) -> Self {
211        Self {
212            id,
213            role,
214            name: None,
215            description: None,
216            value: None,
217            numeric: None,
218            toggled: None,
219            selected: None,
220            expanded: None,
221            disabled: false,
222            focusable: false,
223            modal: false,
224            level: None,
225            active_descendant: None,
226            bounds,
227            children: Vec::new(),
228            actions: Vec::new(),
229        }
230    }
231}

233/// The whole accessibility tree of one window, as of the latest frame.
234#[derive(Clone, Debug, Default, PartialEq)]
235pub struct AccessTree {
236    /// Id of the root node (role [`Role::Window`]).
237    pub root: u64,
238    /// Every node, root first, parents before children.
239    pub nodes: Vec<AccessNode>,
240    /// The node with keyboard focus, if any.
241    pub focus: Option<u64>,
242}
243
244impl AccessTree {
245    /// The node with this id.
246    pub fn get(&self, id: u64) -> Option<&AccessNode> {
247        self.nodes.iter().find(|n| n.id == id)
248    }
249
250    /// The first node with this role and name, for tests and tools.
251    pub fn find(&self, role: Role, name: &str) -> Option<&AccessNode> {
252        self.nodes.iter().find(|n| n.role == role && n.name.as_deref() == Some(name))
253    }
254
255    /// The node with keyboard focus.
256    pub fn focused(&self) -> Option<&AccessNode> {
257        self.focus.and_then(|id| self.get(id))
258    }
259}
