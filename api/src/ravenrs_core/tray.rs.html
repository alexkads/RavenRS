1//! What the menu of the icon beside the clock shows: a plain value that crosses threads, so the
2//! screen, a background thread (an update that finished) and the platform all read the same.
3
4use std::sync::{Arc, Mutex, PoisonError};
5
6/// An id that starts with this opens the app's window before the choice reaches the app, for an
7/// entry whose answer is on screen, such as "Novo lançamento" or the news of an update.
8pub const OPENS_WINDOW: &str = "open:";
9
10/// One entry of a [`TrayMenu`].
11#[derive(Clone, Debug, PartialEq, Eq)]
12#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
13pub enum TrayEntry {
14    /// The app's state at the top of the menu: a line that cannot be chosen, and a second,
15    /// quieter one when there is a detail.
16    Status {
17        /// What it says.
18        title: String,
19        /// A second line, such as the last time it ran.
20        detail: Option<String>,
21    },
22    /// Something to do. The app hears its `id` when it is chosen.
23    Item {
24        /// What the app hears; one that starts with [`OPENS_WINDOW`] opens the window first.
25        id: String,
26        /// What it says.
27        label: String,
28        /// A line below it that cannot be chosen, such as "Versão 0.1.8 pronta".
29        detail: Option<String>,
30        /// Whether it can be chosen.
31        enabled: bool,
32        /// A key combination shown beside it, such as `CmdOrCtrl+N`, where the system shows one.
33        shortcut: Option<String>,
34    },
35    /// A choice that is on or off, with a check mark.
36    Check {
37        /// What the app hears.
38        id: String,
39        /// What it says.
40        label: String,
41        /// Whether it is on.
42        checked: bool,
43    },
44    /// Entries one level down, behind an arrow.
45    Submenu {
46        /// What it says.
47        label: String,
48        /// What it holds.
49        entries: Vec<TrayEntry>,
50    },
51    /// A line between groups.
52    Separator,
53}
54
55#[cfg_attr(feature = "live", ravenrs_live::expose)]
56impl TrayEntry {
57    /// The state at the top, with an optional second line.
58    pub fn status(title: impl Into<String>, detail: Option<String>) -> Self {
59        Self::Status { title: title.into(), detail }
60    }
61
62    /// Something to do, enabled, with no detail or shortcut.
63    pub fn item(id: impl Into<String>, label: impl Into<String>) -> Self {
64        Self::Item { id: id.into(), label: label.into(), detail: None, enabled: true, shortcut: None }
65    }
66
67    /// A choice that is on or off.
68    pub fn check(id: impl Into<String>, label: impl Into<String>, checked: bool) -> Self {
69        Self::Check { id: id.into(), label: label.into(), checked }
70    }
71
72    /// Entries one level down.
73    pub fn submenu(label: impl Into<String>, entries: Vec<TrayEntry>) -> Self {
74        Self::Submenu { label: label.into(), entries }
75    }
76
77    /// The same item, enabled or not; other entries are left as they are.
78    pub fn enabled(mut self, on: bool) -> Self {
79        if let Self::Item { enabled, .. } = &mut self {
80            *enabled = on;
81        }
82        self
83    }
84
85    /// The same item, with a line below it.
86    pub fn detail(mut self, text: impl Into<String>) -> Self {
87        if let Self::Item { detail, .. } = &mut self {
88            *detail = Some(text.into());
89        }
90        self
91    }
92
93    /// The same item, with a key combination beside it, such as `CmdOrCtrl+N`.
94    pub fn shortcut(mut self, keys: impl Into<String>) -> Self {
95        if let Self::Item { shortcut, .. } = &mut self {
96            *shortcut = Some(keys.into());
97        }
98        self
99    }
100}

102type Changed = Arc<dyn Fn() + Send + Sync>;
103
104#[derive(Default)]
105struct Shared {
106    entries: Mutex<Vec<TrayEntry>>,
107    changed: Mutex<Option<Changed>>,
108}
109
110/// The app's own entries of the menu beside the clock, between "Abrir" and "Sair", which stay
111/// first and last. Clones share the same menu: the app keeps one and sets it whenever its state
112/// changes, from any thread, and the icon shows the new menu at once.
113///
114/// ```
115/// use ravenrs_core::{TrayEntry, TrayMenu};
116///
117/// let menu = TrayMenu::new(vec![TrayEntry::status("Autoteste: 18 de 18", None)]);
118/// menu.set(vec![TrayEntry::status("Rodando…", None), TrayEntry::item("run", "Rodar autoteste")]);
119/// assert_eq!(menu.entries().len(), 2);
120/// ```
121#[derive(Clone, Default)]
122#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
123pub struct TrayMenu(Arc<Shared>);
124
125impl std::fmt::Debug for TrayMenu {
126    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
127        f.debug_tuple("TrayMenu").field(&self.entries()).finish()
128    }
129}
130
131#[cfg_attr(feature = "live", ravenrs_live::expose)]
132impl TrayMenu {
133    /// A menu with these entries.
134    pub fn new(entries: Vec<TrayEntry>) -> Self {
135        let menu = Self::default();
136        *menu.0.entries.lock().unwrap_or_else(PoisonError::into_inner) = entries;
137        menu
138    }
139
140    /// The entries as they are now.
141    pub fn entries(&self) -> Vec<TrayEntry> {
142        self.0.entries.lock().unwrap_or_else(PoisonError::into_inner).clone()
143    }
144
145    /// Replaces the entries. The icon hears it only when they differ, so a screen may set
146    /// them on every build.
147    pub fn set(&self, entries: Vec<TrayEntry>) {
148        {
149            let mut now = self.0.entries.lock().unwrap_or_else(PoisonError::into_inner);
150            if *now == entries {
151                return;
152            }
153            *now = entries;
154        }
155        let changed = self.0.changed.lock().unwrap_or_else(PoisonError::into_inner).clone();
156        if let Some(changed) = changed {
157            changed();
158        }
159    }
160
161    /// What runs when the entries change: the platform redraws the menu. Only the last one
162    /// given is kept.
163    pub fn on_change(&self, changed: impl Fn() + Send + Sync + 'static) {
164        *self.0.changed.lock().unwrap_or_else(PoisonError::into_inner) = Some(Arc::new(changed));
165    }
166}

168#[cfg(test)]
169mod tests {
170    use std::sync::atomic::{AtomicUsize, Ordering};
171
172    use super::*;
173
174    #[test]
175    fn the_icon_hears_only_a_menu_that_changed() {
176        let menu = TrayMenu::new(vec![TrayEntry::Separator]);
177        let heard = Arc::new(AtomicUsize::new(0));
178        let h = heard.clone();
179        menu.on_change(move || {
180            h.fetch_add(1, Ordering::SeqCst);
181        });
182        menu.set(vec![TrayEntry::Separator]);
183        assert_eq!(heard.load(Ordering::SeqCst), 0, "the same entries say nothing");
184        let other = menu.clone();
185        std::thread::spawn(move || other.set(vec![TrayEntry::item("a", "A").enabled(false)])).join().ok();
186        assert_eq!(heard.load(Ordering::SeqCst), 1);
187        assert_eq!(menu.entries(), [TrayEntry::item("a", "A").enabled(false)]);
188    }
189
190    #[test]
191    fn the_builders_touch_only_items() {
192        let item = TrayEntry::item("x", "X").detail("d").shortcut("CmdOrCtrl+N").enabled(false);
193        assert_eq!(
194            item,
195            TrayEntry::Item {
196                id: "x".into(),
197                label: "X".into(),
198                detail: Some("d".into()),
199                enabled: false,
200                shortcut: Some("CmdOrCtrl+N".into())
201            }
202        );
203        assert_eq!(TrayEntry::Separator.enabled(false).detail("d"), TrayEntry::Separator);
204    }
