1//! Text with expressions inside it: `Olá {{ item.nome }}, o total é {{ [preço] * [qtd] }}`.
2//!
3//! Each `{{ … }}` holds an expression of the criteria language; a field path alone, such as
4//! `item.nome`, needs no brackets. `\{{` writes two braces. Parsing reports every problem at
5//! once, each with where it is in the text, so an editor can mark it. Rendering stops at the
6//! first expression that fails, saying which one and why, so a failure never reads as an
7//! empty value; and it stops when the text grows past a limit, so a preview cannot run away.
8
9use std::ops::Range;
10
11use crate::ast::Criteria;
12use crate::eval::{EvalError, Record, Scope};
13
14/// A text with expressions inside it.
15#[derive(Clone, Debug, PartialEq)]
16pub struct Template {
17    parts: Vec<Part>,
18}
19
20/// A piece of a [`Template`].
21#[derive(Clone, Debug, PartialEq)]
22pub enum Part {
23    /// Text written as it is.
24    Text(String),
25    /// An expression, and where its source is in the template's text.
26    Expression(Criteria, Range<usize>),
27}
28
29/// A problem found in a template's text, and where.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub struct TemplateError {
32    /// The bytes of the template's text it is about.
33    pub at: Range<usize>,
34    /// What is wrong, in English.
35    pub message: String,
36}
37
38/// Why a template could not be rendered.
39#[derive(Clone, Debug, PartialEq)]
40pub enum RenderError {
41    /// The expression at these bytes failed.
42    Failed(Range<usize>, EvalError),
43    /// The text grew past this many bytes.
44    TooLong(usize),
45}
46
47impl std::fmt::Display for RenderError {
48    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
49        match self {
50            Self::Failed(_, error) => write!(f, "{error}"),
51            Self::TooLong(limit) => write!(f, "the text passed {limit} bytes"),
52        }
53    }
54}
55
56/// Whether `text` is a field path alone: words joined by dots.
57fn bare_path(text: &str) -> bool {
58    !text.is_empty()
59        && text.split('.').all(|word| {
60            let mut chars = word.chars();
61            chars.next().is_some_and(|c| c.is_alphabetic() || c == '_')
62                && chars.all(|c| c.is_alphanumeric() || c == '_')
63        })
64}
65
66impl Template {
67    /// Reads `text`, finding every problem at once.
68    pub fn parse(text: &str) -> Result<Self, Vec<TemplateError>> {
69        let mut parts = Vec::new();
70        let mut errors = Vec::new();
71        let mut plain = String::new();
72        let mut i = 0;
73        while i < text.len() {
74            let rest = &text[i..];
75            if rest.starts_with("\\{{") {
76                plain.push_str("{{");
77                i += 3;
78            } else if let Some(open) = rest.strip_prefix("{{") {
79                let Some(end) = open.find("}}") else {
80                    errors.push(TemplateError {
81                        at: i..text.len(),
82                        message: "`{{` is never closed with `}}`".into(),
83                    });
84                    break;
85                };
86                if !plain.is_empty() {
87                    parts.push(Part::Text(std::mem::take(&mut plain)));
88                }
89                let inner = &open[..end];
90                let at = i..i + 2 + end + 2;
91                let source = inner.trim();
92                let code = if bare_path(source) { format!("[{source}]") } else { source.to_owned() };
93                if source.is_empty() {
94                    errors.push(TemplateError {
95                        at: at.clone(),
96                        message: "an expression is missing between `{{` and `}}`".into(),
97                    });
98                } else {
99                    match Criteria::parse(&code) {
100                        Ok(criteria) => parts.push(Part::Expression(criteria, at.clone())),
101                        Err(error) => errors.push(TemplateError { at: at.clone(), message: error.message }),
102                    }
103                }
104                i = at.end;
105            } else {
106                let c = rest.chars().next().unwrap_or_default();
107                plain.push(c);
108                i += c.len_utf8();
109            }
110        }
111        if !plain.is_empty() {
112            parts.push(Part::Text(plain));
113        }
114        if errors.is_empty() { Ok(Self { parts }) } else { Err(errors) }
115    }
116
117    /// Its pieces, in order.
118    pub fn parts(&self) -> &[Part] {
119        &self.parts
120    }
121
122    /// Whether it has no expression: the text is what it renders.
123    pub fn is_plain(&self) -> bool {
124        self.parts.iter().all(|p| matches!(p, Part::Text(_)))
125    }
126
127    /// Every field path its expressions read, with where each expression is.
128    pub fn references(&self) -> Vec<(Vec<String>, Range<usize>)> {
129        let mut out = Vec::new();
130        for part in &self.parts {
131            if let Part::Expression(criteria, at) = part {
132                let mut paths = Vec::new();
133                properties(criteria, &mut paths);
134                out.extend(paths.into_iter().map(|p| (p, at.clone())));
135            }
136        }
137        out
138    }
139
140    /// The text with each expression's value in its place; an empty value writes nothing.
141    pub fn render<R: Record>(&self, scope: &Scope<'_, R>) -> Result<String, RenderError> {
142        self.render_within(scope, usize::MAX)
143    }
144
145    /// Like [`Self::render`], stopping when the text passes `limit` bytes.
146    pub fn render_within<R: Record>(
147        &self,
148        scope: &Scope<'_, R>,
149        limit: usize,
150    ) -> Result<String, RenderError> {
151        let mut out = String::new();
152        for part in &self.parts {
153            match part {
154                Part::Text(text) => out.push_str(text),
155                Part::Expression(criteria, at) => {
156                    let value = scope.evaluate(criteria).map_err(|e| RenderError::Failed(at.clone(), e))?;
157                    out.push_str(&value.to_text());
158                }
159            }
160            if out.len() > limit {
161                return Err(RenderError::TooLong(limit));
162            }
163        }
164        Ok(out)
165    }
166}
167
168/// The field paths `criteria` reads.
169fn properties(criteria: &Criteria, out: &mut Vec<Vec<String>>) {
170    let all = |list: &[Criteria], out: &mut Vec<Vec<String>>| list.iter().for_each(|c| properties(c, out));
171    match criteria {
172        Criteria::Property(path) => out.push(path.clone()),
173        Criteria::Unary(_, a) | Criteria::IsNull(a) => properties(a, out),
174        Criteria::Binary(_, a, b) => {
175            properties(a, out);
176            properties(b, out);
177        }
178        Criteria::Between(a, b, c) => {
179            properties(a, out);
180            properties(b, out);
181            properties(c, out);
182        }
183        Criteria::In(a, list) => {
184            properties(a, out);
185            all(list, out);
186        }
187        Criteria::Group(_, list) | Criteria::Function(_, list) | Criteria::Custom(_, list) => all(list, out),
188        Criteria::Aggregate(call) => {
189            if let Some(path) = &call.collection {
190                out.push(path.clone());
191            }
192            if let Some(condition) = &call.condition {
193                properties(condition, out);
194            }
195            all(&call.args, out);
196        }
197        Criteria::Value(_) | Criteria::Parameter(_) => {}
198    }
199}

201#[cfg(all(test, feature = "json"))]
202mod tests {
203    use super::*;
204    use crate::Environment;
205    use serde_json::json;
206
207    #[test]
208    fn a_field_path_alone_needs_no_brackets_and_text_stays_as_it_is() {
209        let env = Environment::new();
210        let item = json!({ "item": { "nome": "Ana" }, "qtd": 2, "preço": 3.5 });
211        let t = Template::parse("Olá {{ item.nome }}, são {{ [qtd] * [preço] }} reais \\{{não}}")
212            .expect("parses");
213        assert_eq!(t.render(&Scope::new(&env).row(&item)).as_deref(), Ok("Olá Ana, são 7 reais {{não}}"));
214        assert_eq!(t.references().len(), 3);
215        assert!(Template::parse("só texto").expect("parses").is_plain());
216    }
217
218    #[test]
219    fn every_problem_is_found_with_where_it_is() {
220        let errors = Template::parse("a {{ [x] + }} b {{ }} c {{ [y]").expect_err("three problems");
221        assert_eq!(errors.len(), 3);
222        assert_eq!(errors[0].at, 2..13);
223        assert!(errors[2].message.contains("never closed"));
224    }
225
226    #[test]
227    fn a_failure_is_not_an_empty_value_and_a_runaway_text_stops() {
228        let env = Environment::new();
229        let item = json!({ "a": null });
230        let scope = Scope::new(&env).row(&item);
231        assert_eq!(
232            Template::parse("[{{ a }}]").expect("parses").render(&scope).as_deref(),
233            Ok("[]"),
234            "null writes nothing"
235        );
236        let failed = Template::parse("x {{ Nope(1) }}").expect("parses").render(&scope);
237        assert!(matches!(failed, Err(RenderError::Failed(ref at, _)) if *at == (2..15)), "{failed:?}");
238        let long = Template::parse("{{ PadLeft('a', 5000, 'b') }}").expect("parses");
239        assert_eq!(long.render_within(&scope, 100), Err(RenderError::TooLong(100)));
240    }
241
242    #[test]
243    fn two_threads_each_render_with_their_own_environment() {
244        let template = Template::parse("{{ [n] * 2 }}").expect("parses");
245        let handles: Vec<_> = (1..=2)
246            .map(|n| {
247                let t = template.clone();
248                std::thread::spawn(move || {
249                    let env = Environment::new();
250                    let item = json!({ "n": n });
251                    t.render(&Scope::new(&env).row(&item))
252                })
253            })
254            .collect();
255        let got: Vec<_> = handles.into_iter().map(|h| h.join().expect("joins")).collect();
256        assert_eq!(got, [Ok("2".to_owned()), Ok("4".to_owned())]);
257    }
