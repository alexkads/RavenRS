1//! The function catalog: one table that drives parsing, arity checks and the
2//! documentation an expression editor shows.
3
4/// What a function is for, as an expression editor groups them.
5#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
6#[non_exhaustive]
7pub enum Category {
8    /// Conditions and null handling.
9    Logical,
10    /// Text.
11    String,
12    /// Numbers.
13    Math,
14    /// Dates and times.
15    DateTime,
16    /// Formatting and the page or row being printed.
17    Reporting,
18    /// Summaries over rows.
19    Aggregate,
20}
21
22/// A catalog entry.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub struct FunctionInfo {
25    /// The name, as written in expressions.
26    pub name: &'static str,
27    /// The group it belongs to.
28    pub category: Category,
29    /// How it is called, such as `Round(value, digits)`.
30    pub signature: &'static str,
31    /// What it does.
32    pub description: &'static str,
33}
34
35macro_rules! functions {
36    ($($variant:ident $name:literal $min:literal $max:literal $cat:ident $sig:literal $doc:literal;)*) => {
37        /// The built-in scalar functions. Names are case-insensitive in expressions.
38        #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
39        #[non_exhaustive]
40        #[allow(missing_docs, reason = "documented by the catalog, see Function::info")]
41        pub enum Function { $($variant,)* }
42
43        const FUNCTIONS: &[(Function, usize, usize, FunctionInfo)] = &[
44            $((Function::$variant, $min, $max, FunctionInfo {
45                name: $name, category: Category::$cat, signature: $sig, description: $doc,
46            }),)*
47        ];
48    };
49}

191impl Function {
192    /// The catalog entry.
193    pub fn info(self) -> &'static FunctionInfo {
194        // The table holds every variant, in declaration order.
195        &FUNCTIONS[self as usize].3
196    }
197
198    /// The name, as written in expressions.
199    pub fn name(self) -> &'static str {
200        self.info().name
201    }
202
203    /// The fewest and most arguments it takes.
204    pub fn arity(self) -> (usize, usize) {
205        let e = &FUNCTIONS[self as usize];
206        (e.1, e.2)
207    }
208
209    /// The function with this name, in any case. `Min` and `Max` are the
210    /// two-argument forms; with one argument they are aggregates.
211    pub fn from_name(name: &str) -> Option<Self> {
212        FUNCTIONS.iter().find(|e| e.3.name.eq_ignore_ascii_case(name)).map(|e| e.0)
213    }
214
215    /// Every built-in function.
216    pub fn all() -> impl Iterator<Item = Function> {
217        FUNCTIONS.iter().map(|e| e.0)
218    }
219}

388impl Aggregate {
389    fn entry(
390        self,
391    ) -> &'static (Aggregate, &'static str, &'static str, usize, usize, &'static str, &'static str) {
392        &AGGREGATES[self as usize]
393    }
394
395    /// The name, as written in expressions.
396    pub fn name(self) -> &'static str {
397        self.entry().1
398    }
399
400    /// The fewest and most arguments it takes.
401    pub fn arity(self) -> (usize, usize) {
402        (self.entry().3, self.entry().4)
403    }
404
405    /// The catalog entry.
406    pub fn info(self) -> FunctionInfo {
407        let e = self.entry();
408        FunctionInfo { name: e.1, category: Category::Aggregate, signature: e.5, description: e.6 }
409    }
410
411    /// The aggregate with this name or `sum`-prefixed alias, in any case.
412    pub fn from_name(name: &str) -> Option<Self> {
413        AGGREGATES
414            .iter()
415            .find(|e| e.1.eq_ignore_ascii_case(name) || (!e.2.is_empty() && e.2.eq_ignore_ascii_case(name)))
416            .map(|e| e.0)
417    }
418
419    /// Every aggregate.
420    pub fn all() -> impl Iterator<Item = Aggregate> {
421        AGGREGATES.iter().map(|e| e.0)
422    }
423}
424
425/// The whole catalog, for expression editors: every function and aggregate.
426pub fn catalog() -> Vec<FunctionInfo> {
427    Function::all().map(|f| *f.info()).chain(Aggregate::all().map(Aggregate::info)).collect()
428}

430#[cfg(test)]
431mod tests {
432    use super::*;
433
434    #[test]
435    fn tables_match_their_enums() {
436        for (i, f) in Function::all().enumerate() {
437            assert_eq!(f as usize, i, "{f:?}");
438            assert_eq!(Function::from_name(f.name()).map(Function::name), Some(f.name()));
439            assert!(f.info().signature.starts_with(f.name()), "{f:?}");
440        }
441        for (i, a) in Aggregate::all().enumerate() {
442            assert_eq!(a as usize, i, "{a:?}");
443            assert_eq!(Aggregate::from_name(a.name()), Some(a));
444        }
445        assert_eq!(Aggregate::from_name("SUMSUM"), Some(Aggregate::Sum));
446        assert_eq!(Aggregate::from_name("sumWAvg"), Some(Aggregate::WeightedAvg));
447        assert!(catalog().len() > 130);
448    }
