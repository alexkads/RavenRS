22/// Lines kept past the used area and the visible window, so there is always room to scroll.
23const SPARE_ROWS: u32 = 100;
24const SPARE_COLUMNS: u32 = 26;
25/// How close to a column header's edge a drag resizes the column.
26const RESIZE_GRIP: f32 = 4.0;
27/// Narrowest a column gets by dragging.
28const MIN_COLUMN: f32 = 8.0;
29/// Empty cells to the right that left-aligned text may run into.
30const OVERFLOW_CELLS: u32 = 12;
31
32/// Where a point falls on the surface.
33#[derive(Clone, Copy, Debug, PartialEq)]
34enum Hit {
35    Corner,
36    Column(u32),
37    /// Near the right edge of a column's header.
38    ColumnEdge(u32),
39    Row(u32),
40    Cell(CellRef),
41}
42
43/// Rows, columns and scroll, enough to turn a point into a cell.
44#[derive(Clone, Debug)]
45struct Grid {
46    rows: Axis,
47    columns: Axis,
48    offset: Point,
49}
50
51impl Grid {
52    fn hit(&self, local: Point) -> Hit {
53        let in_header = local.y < HEADER_HEIGHT;
54        let in_row_header = local.x < ROW_HEADER_WIDTH;
55        let x = local.x - ROW_HEADER_WIDTH + self.offset.x;
56        let y = local.y - HEADER_HEIGHT + self.offset.y;
57        match (in_header, in_row_header) {
58            (true, true) => Hit::Corner,
59            (true, false) => {
60                let col = self.columns.at(x);
61                let right = self.columns.start(col) + self.columns.size(col);
62                let left = self.columns.start(col);
63                if right - x <= RESIZE_GRIP {
64                    Hit::ColumnEdge(col)
65                } else if x - left <= RESIZE_GRIP && col > 0 {
66                    Hit::ColumnEdge(col - 1)
67                } else {
68                    Hit::Column(col)
69                }
70            }
71            (false, true) => Hit::Row(self.rows.at(y)),
72            (false, false) => Hit::Cell(CellRef::new(self.rows.at(y), self.columns.at(x))),
73        }
74    }
75
76    /// The cell's rectangle relative to the surface.
77    fn cell_rect(&self, c: CellRef) -> Rect {
78        Rect::new(
79            ROW_HEADER_WIDTH + self.columns.start(c.col) - self.offset.x,
80            HEADER_HEIGHT + self.rows.start(c.row) - self.offset.y,
81            self.columns.size(c.col),
82            self.rows.size(c.row),
83        )
84    }
85
86    fn range_rect(&self, r: RangeRef) -> Rect {
87        let a = self.cell_rect(r.start);
88        let b = self.cell_rect(r.end);
89        Rect::new(a.x(), a.y(), b.x() + b.width() - a.x(), b.y() + b.height() - a.y())
90    }
91}

143impl View for Surface {
144    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
145        let _ = self.state.version();
146        let edit = self.state.edit().get();
147        let theme = *cx.theme();
148        let offset = cx.scroll_offset();
149        let window = cx.viewport();
150        let size = cx.scroll_viewport().unwrap_or(Size::new(window.width, window.height * 0.6));
151        let selection = self.state.selection_now();
152
153        let (picture, content) = self.state.read(|wb| {
154            let sheet = wb.active();
155            let empty = Sheet::new(String::from("Sheet1"));
156            let sheet = sheet.unwrap_or(&empty);
157            let grid = Grid { rows: Axis::rows(sheet, 1.0), columns: Axis::columns(sheet, 1.0), offset };
158            picture(wb, sheet, grid, size, &selection, theme)
159        });
160
161        // Keeps the active cell in view when the keyboard moved it.
162        let last_active = cx.use_state(|| selection.active);
163        if *last_active.peek() != selection.active {
164            last_active.set(selection.active);
165            if let Some(to) = scroll_to_show(&picture.grid, selection.active, size) {
166                cx.set_scroll_offset(to);
167            }
168        }
169        // After an edit over the cell ends, typing goes to the grid again.
170        let was_editing = cx.use_state(|| false);
171        let editing_here = edit.as_ref().is_some_and(|e| e.place == EditPlace::Cell);
172        if *was_editing.peek() != editing_here {
173            was_editing.set(editing_here);
174            if !editing_here {
175                cx.request_focus();
176            }
177        }
178
179        let grid = picture.grid.clone();
180        let active_text = self.state.cell_text(&selection.active.a1());
181        let mut children = Vec::new();
182        if let Some(edit) = edit.filter(|e| e.place == EditPlace::Cell) {
183            children.push(self.editor(cx, &grid, edit.cell));
184        }
185        let picture = Rc::new(picture);
186        let surface = cx.container(
187            LayoutStyle {
188                absolute: Some((offset.x, offset.y)),
189                width: Some(size.width),
190                height: Some(size.height),
191                ..LayoutStyle::leaf()
192            },
193            &children,
194            Paint::Rich(Rc::new({
195                let picture = picture.clone();
196                move |frame: Rect, paint: &mut PaintCx<'_>| paint_picture(&picture, frame, paint)
197            })),
198        );
199        cx.focusable(surface, 0.0);
200        cx.semantics(
201            surface,
202            Semantics::new(Role::Grid)
203                .name(self.name.clone())
204                .value(format!("{} {}", selection.active.a1(), active_text))
205                .read_only(self.read_only),
206        );
207        self.pointer(cx, surface, &grid);
208        self.keys(cx, surface);
209
210        let content = cx.container(
211            LayoutStyle { width: Some(content.width), height: Some(content.height), ..LayoutStyle::leaf() },
212            &[surface],
213            Paint::None,
214        );
215        let body = cx.container(
216            LayoutStyle {
217                grow: 1.0,
218                min_height: Some(0.0),
219                overflow_hidden: true,
220                align_self: Some(Align::Stretch),
221                ..LayoutStyle::stack(LayoutAxis::Vertical)
222            },
223            &[content],
224            Paint::None,
225        );
226        cx.scrollable_virtual(body, true);
227        body
228    }
229}

231impl Surface {
232    /// The text field over the cell being edited.
233    fn editor(&self, cx: &mut BuildCx<'_>, grid: &Grid, cell: CellRef) -> NodeId {
234        let rect = grid.cell_rect(cell);
235        let (read, write) = (self.state.clone(), self.state.clone());
236        let (submit, cancel, keys) = (self.state.clone(), self.state.clone(), self.state.clone());
237        let field = TextField::new(Binding::new(
238            move || read.edit().get().map(|e| e.text).unwrap_or_default(),
239            move |text: String| {
240                write.edit().update(|e| {
241                    if let Some(e) = e {
242                        e.text = text
243                    }
244                })
245            },
246        ))
247        .compact()
248        .autofocus()
249        .width(rect.width().max(120.0))
250        .on_submit(move |_| submit.commit_edit(1, 0))
251        .on_cancel(move || cancel.cancel_edit())
252        .intercept_keys(Rc::new(move |key: &Key, modifiers: Modifiers| match key {
253            Key::Tab => {
254                keys.commit_edit(0, if modifiers.shift { -1 } else { 1 });
255                true
256            }
257            Key::Enter if modifiers.shift => {
258                keys.commit_edit(-1, 0);
259                true
260            }
261            _ => false,
262        }));
263        let node = cx.child(&field);
264        cx.container(
265            LayoutStyle { absolute: Some((rect.x().max(0.0), rect.y().max(0.0))), ..LayoutStyle::leaf() },
266            &[node],
267            Paint::None,
268        )
269    }
270
271    fn pointer(&self, cx: &mut BuildCx<'_>, surface: NodeId, grid: &Grid) {
272        // Where the latest press landed, relative to the surface: a drag starts on every press,
273        // so this is where a click or a double click happened.
274        // Both outlive this build: the press itself changes the selection, which rebuilds.
275        let pressed: Rc<Cell<Option<Point>>> = cx.use_state(|| Rc::new(Cell::new(None))).peek().clone();
276        let start: Rc<Cell<Option<Hit>>> = cx.use_state(|| Rc::new(Cell::new(None))).peek().clone();
277        let grid = Rc::new(grid.clone());
278        cx.on_double_click(surface, {
279            let (state, grid, pressed, read_only) =
280                (self.state.clone(), grid.clone(), pressed.clone(), self.read_only);
281            Rc::new(move || match pressed.get().map(|local| grid.hit(local)) {
282                Some(Hit::Cell(_)) if !read_only => state.begin_edit(None, EditPlace::Cell),
283                Some(Hit::ColumnEdge(col)) if !read_only => {
284                    let cols = format!("{0}:{0}", col_to_letters(col));
285                    let _ = state.execute("home.autofitColumnWidth", json!({ "cols": cols }));
286                }
287                _ => {}
288            })
289        });
290        // A drag either picks a range of cells or, from a header's edge, sizes a column.
291        let (state, read_only, drag_modifiers) = (self.state.clone(), self.read_only, cx.modifiers());
292        cx.on_drag(
293            surface,
294            Rc::new(move |drag: Drag| {
295                let local = Point::new(drag.position.x - drag.frame.x(), drag.position.y - drag.frame.y());
296                match drag.phase {
297                    DragPhase::Start => {
298                        pressed.set(Some(local));
299                        let hit = grid.hit(local);
300                        let extend = drag_modifiers.get().shift;
301                        // Shift starts from the active cell, as a Shift-click does.
302                        let anchor = match hit {
303                            Hit::Cell(_) if extend => Hit::Cell(state.selection_now().active),
304                            other => other,
305                        };
306                        start.set(Some(anchor));
307                        if !matches!(hit, Hit::ColumnEdge(_)) {
308                            click(&state, &grid, hit, extend);
309                        }
310                    }
311                    DragPhase::Move | DragPhase::End => match start.get() {
312                        Some(Hit::Cell(from)) => {
313                            if let Hit::Cell(to) = grid.hit(clamped(local, drag.frame.size)) {
314                                state.select_range(RangeRef::new(from, to), from);
315                            }
316                        }
317                        Some(Hit::ColumnEdge(col)) if !read_only && drag.phase == DragPhase::End => {
318                            let left = ROW_HEADER_WIDTH + grid.columns.start(col) - grid.offset.x;
319                            let width = (local.x - left).max(MIN_COLUMN).round();
320                            let cols = format!("{0}:{0}", col_to_letters(col));
321                            let _ =
322                                state.execute("home.columnWidth", json!({ "cols": cols, "width": width }));
323                        }
324                        _ => {}
325                    },
326                    _ => {}
327                }
328                if drag.phase == DragPhase::End {
329                    start.set(None);
330                }
331            }),
332        );
333    }
334
335    fn keys(&self, cx: &mut BuildCx<'_>, surface: NodeId) {
336        let state = self.state.clone();
337        let clipboard = cx.clipboard();
338        let read_only = self.read_only;
339        cx.on_key(
340            surface,
341            Rc::new(move |key: &Key, m: Modifiers| {
342                let command = m.ctrl || m.meta;
343                match key {
344                    Key::ArrowUp => step(&state, -1, 0, m),
345                    Key::ArrowDown => step(&state, 1, 0, m),
346                    Key::ArrowLeft => step(&state, 0, -1, m),
347                    Key::ArrowRight => step(&state, 0, 1, m),
348                    Key::PageUp | Key::PageDown => {
349                        let dr = if *key == Key::PageUp { -1 } else { 1 };
350                        let _ = state.execute(
351                            "selection.move",
352                            json!({ "dr": dr, "dc": 0, "page": true, "extend": m.shift }),
353                        );
354                    }
355                    Key::Enter => state.move_by(if m.shift { -1 } else { 1 }, 0, false),
356                    Key::Tab => state.move_by(0, if m.shift { -1 } else { 1 }, false),
357                    Key::Home => {
358                        let active = state.selection_now().active;
359                        let to = if command { CellRef::new(0, 0) } else { CellRef::new(active.row, 0) };
360                        state.select_range(RangeRef::new(to, to), to);
361                    }
362                    Key::Delete | Key::Backspace if !read_only => {
363                        let _ = state.execute("edit.clearContents", json!({}));
364                    }
365                    Key::Other(name) if name == "F2" && !read_only => state.begin_edit(None, EditPlace::Cell),
366                    Key::Character(text) if command => {
367                        return shortcut(&state, &clipboard, &text.to_lowercase(), m.shift, read_only);
368                    }
369                    Key::Character(text) if !read_only && !text.is_empty() => {
370                        if text.contains(['\t', '\n']) {
371                            let _ = state.execute("edit.paste", json!({ "text": text }));
372                        } else {
373                            state.begin_edit(Some(text.clone()), EditPlace::Cell);
374                        }
375                    }
376                    _ => return false,
377                }
378                true
379            }),
380        );
381    }
382}

384fn step(state: &SpreadsheetState, dr: i64, dc: i64, m: Modifiers) {
385    let jump = m.ctrl || m.meta;
386    let _ = state.execute("selection.move", json!({ "dr": dr, "dc": dc, "extend": m.shift, "jump": jump }));
387}

389/// Cmd or Ctrl with a letter.
390fn shortcut(
391    state: &SpreadsheetState,
392    clipboard: &ravenrs_widgets::Clipboard,
393    key: &str,
394    shift: bool,
395    read_only: bool,
396) -> bool {
397    let run = |command: &str| {
398        let _ = state.execute(command, json!({}));
399    };
400    match key {
401        "c" => copy(state, clipboard, "edit.copy"),
402        "a" => run("edit.selectAll"),
403        _ if read_only => return false,
404        "z" if shift => run("edit.redo"),
405        "z" => run("edit.undo"),
406        "y" => run("edit.redo"),
407        "b" => run("home.bold"),
408        "i" => run("home.italic"),
409        "u" => run("home.underline"),
410        "x" => copy(state, clipboard, "edit.cut"),
411        "v" => {
412            let params = clipboard.read().map_or_else(|| json!({}), |text| json!({ "text": text }));
413            let _ = state.execute("edit.paste", params);
414        }
415        _ => return false,
416    }
417    true
418}

420fn copy(state: &SpreadsheetState, clipboard: &ravenrs_widgets::Clipboard, command: &str) {
421    if let Ok(reply) = state.execute(command, json!({}))
422        && let Some(text) = reply.get("text").and_then(|t| t.as_str())
423    {
424        clipboard.write(text);
425    }
426}

428fn click(state: &SpreadsheetState, grid: &Grid, hit: Hit, extend: bool) {
429    let active = state.selection_now().active;
430    let (range, at) = match hit {
431        Hit::Corner => {
432            let _ = state.execute("edit.selectAll", json!({}));
433            return;
434        }
435        Hit::Column(col) | Hit::ColumnEdge(col) => {
436            let from = if extend { active.col.min(col) } else { col };
437            let to = if extend { active.col.max(col) } else { col };
438            (
439                RangeRef::new(CellRef::new(0, from), CellRef::new(MAX_ROWS - 1, to)),
440                CellRef::new(grid.offset_row(), col),
441            )
442        }
443        Hit::Row(row) => {
444            let from = if extend { active.row.min(row) } else { row };
445            let to = if extend { active.row.max(row) } else { row };
446            (RangeRef::new(CellRef::new(from, 0), CellRef::new(to, MAX_COLS - 1)), CellRef::new(row, 0))
447        }
448        Hit::Cell(c) if extend => (RangeRef::new(active, c), active),
449        Hit::Cell(c) => (RangeRef::new(c, c), c),
450    };
451    let active = if extend { active } else { at };
452    state.select_range(range, active);
453}

470/// The scroll offset that shows `cell`, if it is not in view now.
471fn scroll_to_show(grid: &Grid, cell: CellRef, size: Size) -> Option<Point> {
472    let (x, w) = (grid.columns.start(cell.col), grid.columns.size(cell.col));
473    let (y, h) = (grid.rows.start(cell.row), grid.rows.size(cell.row));
474    let view_w = (size.width - ROW_HEADER_WIDTH).max(0.0);
475    let view_h = (size.height - HEADER_HEIGHT).max(0.0);
476    let mut to = grid.offset;
477    if x < to.x {
478        to.x = x;
479    } else if x + w > to.x + view_w {
480        to.x = (x + w - view_w).max(0.0);
481    }
482    if y < to.y {
483        to.y = y;
484    } else if y + h > to.y + view_h {
485        to.y = (y + h - view_h).max(0.0);
486    }
487    (to != grid.offset).then_some(to)
488}

490/// Works out what the visible window shows, and the size of the whole scrollable area.
491fn picture(
492    wb: &Workbook,
493    sheet: &Sheet,
494    grid: Grid,
495    size: Size,
496    selection: &ravenrs_sheet::engine::Selection,
497    theme: Theme,
498) -> (Picture, Size) {
499    let first_row = grid.rows.at(grid.offset.y);
500    let first_col = grid.columns.at(grid.offset.x);
501    let mut rows = Vec::new();
502    let mut r = first_row;
503    while r < grid.rows.count() && grid.rows.start(r) < grid.offset.y + size.height {
504        if grid.rows.size(r) > 0.0 {
505            rows.push(r);
506        }
507        r += 1;
508    }
509    let mut columns = Vec::new();
510    let mut c = first_col;
511    while c < grid.columns.count() && grid.columns.start(c) < grid.offset.x + size.width {
512        if grid.columns.size(c) > 0.0 {
513            columns.push(c);
514        }
515        c += 1;
516    }
517    let used = sheet.used_range();
518    let extent_rows = (used.map_or(0, |u| u.end.row + 1).max(r) + SPARE_ROWS).min(MAX_ROWS);
519    let extent_cols = (used.map_or(0, |u| u.end.col + 1).max(c) + SPARE_COLUMNS).min(MAX_COLS);
520    let content = Size::new(
521        ROW_HEADER_WIDTH + grid.columns.start(extent_cols),
522        HEADER_HEIGHT + grid.rows.start(extent_rows),
523    );
524
525    let ink = theme.color(Semantic::Ink);
526    let mut cells = Vec::new();
527    let mut merges = Vec::new();
528    for merge in &sheet.merges {
529        let visible =
530            rows.first().zip(rows.last()).is_some_and(|(a, b)| merge.end.row >= *a && merge.start.row <= *b)
531                && columns
532                    .first()
533                    .zip(columns.last())
534                    .is_some_and(|(a, b)| merge.end.col >= *a && merge.start.col <= *b);
535        if visible {
536            let style = wb.styles.get(sheet.style_id(merge.start));
537            merges.push((grid.range_rect(*merge), fill_color(wb, style)));
538            if let Some(look) = cell_look(wb, sheet, &grid, merge.start, Some(*merge), ink) {
539                cells.push(look);
540            }
541        }
542    }
543    for &row in &rows {
544        for &col in &columns {
545            let at = CellRef::new(row, col);
546            if sheet.merge_at(at).is_some() {
547                continue;
548            }
549            if let Some(look) = cell_look(wb, sheet, &grid, at, None, ink) {
550                cells.push(look);
551            }
552        }
553    }
554
555    let selection_rects =
556        selection.ranges.iter().map(|r| grid.range_rect(clip_range(*r, extent_rows, extent_cols))).collect();
557    let active = match sheet.merge_at(selection.active) {
558        Some(m) => grid.range_rect(m),
559        None => grid.cell_rect(selection.active),
560    };
561    let selected_rows = selection.ranges.iter().map(|r| (r.start.row, r.end.row)).collect();
562    let selected_columns = selection.ranges.iter().map(|r| (r.start.col, r.end.col)).collect();
563    let picture = Picture {
564        size,
565        grid,
566        rows,
567        columns,
568        cells,
569        merges,
570        gridlines: sheet.show_gridlines,
571        selection: selection_rects,
572        active,
573        selected_rows,
574        selected_columns,
575        theme,
576    };
577    (picture, content)
578}

580/// A range cut to the area that exists on screen, so a whole column does not make a huge rectangle.
581fn clip_range(r: RangeRef, rows: u32, cols: u32) -> RangeRef {
582    let end = CellRef::new(r.end.row.min(rows.saturating_sub(1)), r.end.col.min(cols.saturating_sub(1)));
583    RangeRef::new(r.start, end)
584}

590fn fill_color(wb: &Workbook, style: &ravenrs_sheet::model::Style) -> Option<Color> {
591    match style.fill.pattern {
592        PatternType::None => None,
593        PatternType::Solid => style.fill.fg.resolve(&wb.theme).map(rgb),
594        _ => {
595            // Patterns show as a blend of their two colors.
596            let fg = style.fill.fg.resolve(&wb.theme).unwrap_or([0, 0, 0]);
597            let bg = style.fill.bg.resolve(&wb.theme).unwrap_or([255, 255, 255]);
598            let mix = |i: usize| ((fg[i] as u16 + bg[i] as u16 * 3) / 4) as u8;
599            Some(rgb([mix(0), mix(1), mix(2)]))
600        }
601    }
602}

604/// Text with no color of its own: the theme's ink, unless the cell's fill makes it hard to read.
605fn automatic_ink(ink: Color, fill: Option<Color>) -> Color {
606    match fill {
607        Some(fill) if ink.contrast_ratio(fill) < 3.0 => {
608            if Color::BLACK.contrast_ratio(fill) >= Color::WHITE.contrast_ratio(fill) {
609                Color::BLACK
610            } else {
611                Color::WHITE
612            }
613        }
614        _ => ink,
615    }
616}

618fn edge(wb: &Workbook, line: &BorderLine, ink: Color) -> Option<Edge> {
619    let width = match line.style {
620        BorderStyle::None => return None,
621        BorderStyle::Medium
622        | BorderStyle::MediumDashed
623        | BorderStyle::MediumDashDot
624        | BorderStyle::MediumDashDotDot => 2.0,
625        BorderStyle::Thick | BorderStyle::Double => 3.0,
626        _ => 1.0,
627    };
628    Some(Edge { width, color: line.color.resolve(&wb.theme).map_or(ink, rgb) })
629}

631/// How one cell looks, or nothing when it has no fill, no borders and no text.
632fn cell_look(
633    wb: &Workbook,
634    sheet: &Sheet,
635    grid: &Grid,
636    at: CellRef,
637    merge: Option<RangeRef>,
638    ink: Color,
639) -> Option<CellLook> {
640    let style = wb.styles.get(sheet.style_id(at));
641    let value = sheet.value(at);
642    let fill = fill_color(wb, style);
643    let b = &style.borders;
644    let edges =
645        [edge(wb, &b.left, ink), edge(wb, &b.top, ink), edge(wb, &b.right, ink), edge(wb, &b.bottom, ink)];
646    if matches!(value, Value::Empty) && fill.is_none() && edges.iter().all(Option::is_none) {
647        return None;
648    }
649    let rect = merge.map_or_else(|| grid.cell_rect(at), |m| grid.range_rect(m));
650    let formatted = display::format(&value, style.num_fmt.as_str(), wb);
651    let text = if sheet.show_formulas {
652        display::cell_text(wb, sheet, at)
653    } else if !sheet.show_zeros && value.as_f64() == Some(0.0) {
654        String::new()
655    } else {
656        formatted.text
657    };
658    let color = formatted
659        .color
660        .and_then(|c| c.rgb())
661        .map(rgb)
662        .or_else(|| style.font.color.resolve(&wb.theme).map(rgb))
663        .unwrap_or_else(|| automatic_ink(ink, fill));
664    let align = match style.align.h {
665        HAlign::General => match value {
666            Value::Number(_) => HAlign::Right,
667            Value::Bool(_) | Value::Error(_) => HAlign::Center,
668            _ if formatted.numeric => HAlign::Right,
669            _ => HAlign::Left,
670        },
671        other => other,
672    };
673    // Left-aligned text runs into the empty cells to its right.
674    let mut clip = rect;
675    if merge.is_none() && align == HAlign::Left && matches!(value, Value::Text(_)) && !style.align.wrap {
676        let mut col = at.col + 1;
677        while col < at.col + OVERFLOW_CELLS
678            && col < MAX_COLS
679            && sheet.value(CellRef::new(at.row, col)) == Value::Empty
680        {
681            col += 1;
682        }
683        let end = grid.cell_rect(CellRef::new(at.row, col - 1));
684        clip = Rect::new(rect.x(), rect.y(), end.x() + end.width() - rect.x(), rect.height());
685    }
686    let size_px = (style.font.size * 96.0 / 72.0).clamp(6.0, 96.0);
687    let attrs = TextAttrs {
688        size: size_px,
689        line_height: (size_px * 1.25).round(),
690        weight: if style.font.bold { 600.0 } else { 400.0 },
691        ..TextAttrs::default()
692    };
693    Some(CellLook {
694        rect,
695        clip,
696        fill,
697        text,
698        color,
699        attrs,
700        align,
701        valign: style.align.v,
702        underline: style.font.underline != Underline::None,
703        strike: style.font.strike,
704        edges,
705    })
706}

708fn quad(paint: &mut PaintCx<'_>, rect: Rect, color: Color) {
709    if rect.width() > 0.0 && rect.height() > 0.0 {
710        paint.scene.push_quad(Quad::new(rect, color));
711    }
712}

714fn at(frame: Rect, r: Rect) -> Rect {

718fn paint_picture(p: &Picture, frame: Rect, paint: &mut PaintCx<'_>) {
719    let theme = &p.theme;
720    let canvas = theme.color(Semantic::Canvas);
721    let line = theme.color(Semantic::Border).with_alpha(0.55);
722    let accent = theme.color(Semantic::Accent);
723    let cells = Rect::new(
724        frame.x() + ROW_HEADER_WIDTH,
725        frame.y() + HEADER_HEIGHT,
726        (p.size.width - ROW_HEADER_WIDTH).max(0.0),
727        (p.size.height - HEADER_HEIGHT).max(0.0),
728    );
729    quad(paint, frame, canvas);
730    paint.scene.push_clip(cells);
731    if p.gridlines {
732        for &col in &p.columns {
733            let r = p.grid.cell_rect(CellRef::new(0, col));
734            quad(paint, Rect::new(frame.x() + r.x() + r.width() - 1.0, cells.y(), 1.0, cells.height()), line);
735        }
736        for &row in &p.rows {
737            let r = p.grid.cell_rect(CellRef::new(row, 0));
738            quad(paint, Rect::new(cells.x(), frame.y() + r.y() + r.height() - 1.0, cells.width(), 1.0), line);
739        }
740    }
741    for (rect, fill) in &p.merges {
742        let inner = at(frame, *rect);
743        quad(
744            paint,
745            Rect::new(inner.x(), inner.y(), inner.width() - 1.0, inner.height() - 1.0),
746            fill.unwrap_or(canvas),
747        );
748    }
749    for cell in &p.cells {
750        if let Some(fill) = cell.fill {
751            quad(paint, at(frame, cell.rect), fill);
752        }
753    }
754    for cell in &p.cells {
755        paint_edges(cell, frame, paint);
756    }
757    for cell in p.cells.iter().filter(|c| !c.text.is_empty()) {
758        paint_text(cell, frame, paint);
759    }
760    // The selection: a tint over the range, and a frame around the active cell.
761    for r in &p.selection {
762        quad(paint, at(frame, *r), accent.with_alpha(0.10));
763        let r = at(frame, *r);
764        paint.scene.push_quad(Quad::new(r, Color::TRANSPARENT).border(1.0, accent));
765    }
766    let active = at(frame, p.active);
767    quad(paint, active, canvas.with_alpha(0.0));
768    paint.scene.push_quad(
769        Quad::new(
770            Rect::new(active.x() - 1.0, active.y() - 1.0, active.width() + 1.0, active.height() + 1.0),
771            Color::TRANSPARENT,
772        )
773        .border(2.0, accent),
774    );
775    paint.scene.pop_clip();
776    paint_headers(p, frame, paint);
777}

779fn paint_edges(cell: &CellLook, frame: Rect, paint: &mut PaintCx<'_>) {
780    let r = at(frame, cell.rect);
781    let [left, top, right, bottom] = cell.edges;
782    if let Some(e) = left {
783        quad(paint, Rect::new(r.x() - 1.0, r.y() - 1.0, e.width, r.height() + 1.0), e.color);
784    }
785    if let Some(e) = top {
786        quad(paint, Rect::new(r.x() - 1.0, r.y() - 1.0, r.width() + 1.0, e.width), e.color);
787    }
788    if let Some(e) = right {
789        quad(paint, Rect::new(r.x() + r.width() - e.width, r.y() - 1.0, e.width, r.height() + 1.0), e.color);
790    }
791    if let Some(e) = bottom {
792        quad(paint, Rect::new(r.x() - 1.0, r.y() + r.height() - e.width, r.width() + 1.0, e.width), e.color);
793    }
794}

796fn paint_text(cell: &CellLook, frame: Rect, paint: &mut PaintCx<'_>) {
797    const PAD: f32 = 3.0;
798    let clip = at(frame, cell.clip);
799    let rect = at(frame, cell.rect);
800    let mut layout = paint.text.layout(&cell.text, &cell.attrs);
801    let measured = layout.measure(None);
802    let x = match cell.align {
803        HAlign::Right => rect.x() + rect.width() - PAD - measured.width,
804        HAlign::Center | HAlign::CenterAcross => rect.x() + (rect.width() - measured.width) / 2.0,
805        _ => rect.x() + PAD,
806    };
807    let y = match cell.valign {
808        VAlign::Top => rect.y() + 1.0,
809        VAlign::Center | VAlign::Justify | VAlign::Distributed => {
810            rect.y() + (rect.height() - measured.height) / 2.0
811        }
812        _ => rect.y() + rect.height() - measured.height - 1.0,
813    };
814    paint.scene.push_clip(Rect::new(
815        clip.x(),
816        clip.y(),
817        (clip.width() - 1.0).max(0.0),
818        (clip.height() - 1.0).max(0.0),
819    ));
820    layout.paint(Point::new(x.round(), y.round()), cell.color, paint.scene);
821    let baseline = y + measured.height * 0.82;
822    if cell.underline {
823        quad(paint, Rect::new(x, baseline.round(), measured.width, 1.0), cell.color);
824    }
825    if cell.strike {
826        quad(paint, Rect::new(x, (y + measured.height * 0.55).round(), measured.width, 1.0), cell.color);
827    }
828    paint.scene.pop_clip();
829}

831fn paint_headers(p: &Picture, frame: Rect, paint: &mut PaintCx<'_>) {
832    let theme = &p.theme;
833    let surface = theme.color(Semantic::Surface);
834    let border = theme.color(Semantic::Border);
835    let muted = theme.color(Semantic::InkMuted);
836    let accent = theme.color(Semantic::Accent);
837    let style = theme.text(TextStyle::Caption);
838    let attrs = TextAttrs {
839        size: style.size,
840        line_height: style.line_height,
841        weight: style.weight.0,
842        ..TextAttrs::default()
843    };
844    let strong = TextAttrs { weight: 600.0, ..attrs.clone() };
845
846    let top = Rect::new(frame.x(), frame.y(), frame.width(), HEADER_HEIGHT);
847    let left = Rect::new(frame.x(), frame.y(), ROW_HEADER_WIDTH, frame.height());
848    quad(paint, top, surface);
849    quad(paint, left, surface);
850
851    paint.scene.push_clip(Rect::new(
852        frame.x() + ROW_HEADER_WIDTH,
853        frame.y(),
854        (frame.width() - ROW_HEADER_WIDTH).max(0.0),
855        HEADER_HEIGHT,
856    ));
857    for &col in &p.columns {
858        let r = p.grid.cell_rect(CellRef::new(0, col));
859        let cell = Rect::new(frame.x() + r.x(), frame.y(), r.width(), HEADER_HEIGHT);
860        let selected = p.selected_columns.iter().any(|&(a, b)| col >= a && col <= b);
861        if selected {
862            quad(paint, cell, accent.with_alpha(0.14));
863            quad(paint, Rect::new(cell.x(), cell.y() + HEADER_HEIGHT - 2.0, cell.width(), 2.0), accent);
864        }
865        quad(paint, Rect::new(cell.x() + cell.width() - 1.0, cell.y(), 1.0, HEADER_HEIGHT), border);
866        let label = col_to_letters(col);
867        let mut layout = paint.text.layout(&label, if selected { &strong } else { &attrs });
868        let m = layout.measure(None);
869        let color = if selected { accent } else { muted };
870        layout.paint(
871            Point::new(
872                (cell.x() + (cell.width() - m.width) / 2.0).round(),
873                (cell.y() + (HEADER_HEIGHT - m.height) / 2.0).round(),
874            ),
875            color,
876            paint.scene,
877        );
878    }
879    paint.scene.pop_clip();
880
881    paint.scene.push_clip(Rect::new(
882        frame.x(),
883        frame.y() + HEADER_HEIGHT,
884        ROW_HEADER_WIDTH,
885        (frame.height() - HEADER_HEIGHT).max(0.0),
886    ));
887    for &row in &p.rows {
888        let r = p.grid.cell_rect(CellRef::new(row, 0));
889        let cell = Rect::new(frame.x(), frame.y() + r.y(), ROW_HEADER_WIDTH, r.height());
890        let selected = p.selected_rows.iter().any(|&(a, b)| row >= a && row <= b);
891        if selected {
892            quad(paint, cell, accent.with_alpha(0.14));
893            quad(paint, Rect::new(cell.x() + ROW_HEADER_WIDTH - 2.0, cell.y(), 2.0, cell.height()), accent);
894        }
895        quad(paint, Rect::new(cell.x(), cell.y() + cell.height() - 1.0, ROW_HEADER_WIDTH, 1.0), border);
896        let label = (row + 1).to_string();
897        let mut layout = paint.text.layout(&label, if selected { &strong } else { &attrs });
898        let m = layout.measure(None);
899        let color = if selected { accent } else { muted };
900        layout.paint(
901            Point::new(
902                (cell.x() + (ROW_HEADER_WIDTH - m.width) / 2.0).round(),
903                (cell.y() + (cell.height() - m.height) / 2.0).round(),
904            ),
905            color,
906            paint.scene,
907        );
908    }
909    paint.scene.pop_clip();
910
911    quad(paint, Rect::new(frame.x(), frame.y() + HEADER_HEIGHT - 1.0, frame.width(), 1.0), border);
912    quad(paint, Rect::new(frame.x() + ROW_HEADER_WIDTH - 1.0, frame.y(), 1.0, frame.height()), border);
913    quad(paint, Rect::new(frame.x(), frame.y(), ROW_HEADER_WIDTH - 1.0, HEADER_HEIGHT - 1.0), surface);
914}
