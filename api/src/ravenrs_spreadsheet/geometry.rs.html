1//! Where rows and columns sit, in logical pixels, with their custom sizes.
2
3use ravenrs_sheet::model::Sheet;
4
5/// Height of the column header strip.
6pub(crate) const HEADER_HEIGHT: f32 = 24.0;
7/// Width of the row header strip.
8pub(crate) const ROW_HEADER_WIDTH: f32 = 48.0;
9
10/// One direction of the grid: every line has the default size except the listed ones.
11#[derive(Clone, Debug)]
12pub(crate) struct Axis {
13    default: f32,
14    /// Lines with their own size (zero when hidden), in order.
15    custom: Vec<(u32, f32)>,
16    /// Lines past the last one do not exist.
17    count: u32,
18}
19
20impl Axis {
21    /// The columns of `sheet`.
22    pub(crate) fn columns(sheet: &Sheet, zoom: f32) -> Self {
23        let custom = sheet.cols.keys().map(|&c| (c, sheet.col_width(c) * zoom)).collect();
24        Self { default: sheet.default_col_width * zoom, custom, count: ravenrs_sheet::base::MAX_COLS }
25    }
26
27    /// The rows of `sheet`.
28    pub(crate) fn rows(sheet: &Sheet, zoom: f32) -> Self {
29        let custom = sheet.rows.keys().map(|&r| (r, sheet.row_height(r) * zoom)).collect();
30        Self { default: sheet.default_row_height * zoom, custom, count: ravenrs_sheet::base::MAX_ROWS }
31    }
32
33    /// Lines in this direction.
34    pub(crate) fn count(&self) -> u32 {
35        self.count
36    }
37
38    /// The size of line `i`.
39    pub(crate) fn size(&self, i: u32) -> f32 {
40        match self.custom.binary_search_by_key(&i, |&(k, _)| k) {
41            Ok(at) => self.custom[at].1,
42            Err(_) => self.default,
43        }
44    }
45
46    /// Where line `i` starts.
47    pub(crate) fn start(&self, i: u32) -> f32 {
48        let mut at = i as f32 * self.default;
49        for &(k, size) in &self.custom {
50            if k >= i {
51                break;
52            }
53            at += size - self.default;
54        }
55        at
56    }
57
58    /// The line under `position`, the last one past the end.
59    pub(crate) fn at(&self, position: f32) -> u32 {
60        let position = position.max(0.0);
61        let (mut line, mut start) = (0u32, 0.0f32);
62        for &(k, size) in &self.custom {
63            let k_start = start + (k - line) as f32 * self.default;
64            if position < k_start {
65                break;
66            }
67            if position < k_start + size {
68                return k;
69            }
70            line = k + 1;
71            start = k_start + size;
72        }
73        if self.default <= 0.0 {
74            return line.min(self.count - 1);
75        }
76        let more = ((position - start) / self.default).floor();
77        (line as f32 + more).min((self.count - 1) as f32) as u32
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::Axis;
84
85    fn axis() -> Axis {
86        Axis { default: 20.0, custom: vec![(2, 50.0), (4, 0.0)], count: 100 }
87    }
88
89    #[test]
90    fn custom_sizes_move_the_lines_after_them() {
91        let a = axis();
92        assert_eq!(a.start(0), 0.0);
93        assert_eq!(a.start(2), 40.0);
94        assert_eq!(a.start(3), 90.0);
95        assert_eq!(a.start(5), 110.0, "a hidden line takes no room");
96        assert_eq!(a.size(2), 50.0);
97        assert_eq!(a.size(7), 20.0);
98    }
99
100    #[test]
101    fn a_position_finds_its_line_and_skips_hidden_ones() {
102        let a = axis();
103        assert_eq!(a.at(0.0), 0);
104        assert_eq!(a.at(39.9), 1);
105        assert_eq!(a.at(40.0), 2);
106        assert_eq!(a.at(89.0), 2);
107        assert_eq!(a.at(90.0), 3);
108        assert_eq!(a.at(110.0), 5);
109        assert_eq!(a.at(1.0e9), 99, "stops at the last line");
110        for i in [0, 1, 2, 3, 5, 6, 40] {
111            assert_eq!(a.at(a.start(i)), i);
112        }
113    }
