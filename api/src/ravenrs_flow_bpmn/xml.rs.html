11fn escaped(text: &str) -> String {
12    text.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
13}

15/// Writes one process: its lanes, its elements and the flows and associations among them, for
16/// the elements `member` keeps.
17fn write_process(
18    out: &mut String,
19    graph: &FlowGraph,
20    id: &str,
21    name: &str,
22    member: &dyn Fn(&FlowNode) -> bool,
23) {
24    out.push_str(&format!(
25        "  <bpmn:process id=\"{}\" name=\"{}\" isExecutable=\"false\">\n",
26        escaped(id),
27        escaped(name)
28    ));
29    let lanes: Vec<&FlowNode> = graph.nodes.iter().filter(|n| n.kind == LANE && member(n)).collect();
30    if !lanes.is_empty() {
31        out.push_str(&format!("    <bpmn:laneSet id=\"LaneSet_{}\">\n", escaped(id)));
32        for lane in &lanes {
33            out.push_str(&format!(
34                "      <bpmn:lane id=\"{}\" name=\"{}\">\n",
35                escaped(&lane.id),
36                escaped(&lane.title)
37            ));
38            for inside in graph.children(&lane.id) {
39                out.push_str(&format!(
40                    "        <bpmn:flowNodeRef>{}</bpmn:flowNodeRef>\n",
41                    escaped(&inside.id)
42                ));
43            }
44            out.push_str("      </bpmn:lane>\n");
45        }
46        out.push_str("    </bpmn:laneSet>\n");
47    }
48    let mine: Vec<&FlowNode> =
49        graph.nodes.iter().filter(|n| n.kind != LANE && n.kind != PARTICIPANT && member(n)).collect();
50    for node in &mine {
51        let kind = if elements::family(&node.kind).is_some() { node.kind.as_str() } else { "task" };
52        if kind == "textAnnotation" {
53            out.push_str(&format!(
54                "    <bpmn:textAnnotation id=\"{}\"><bpmn:text>{}</bpmn:text></bpmn:textAnnotation>\n",
55                escaped(&node.id),
56                escaped(&node.title)
57            ));
58            continue;
59        }
60        out.push_str(&format!(
61            "    <bpmn:{kind} id=\"{}\" name=\"{}\">\n",
62            escaped(&node.id),
63            escaped(&node.title)
64        ));
65        for e in graph.edges.iter().filter(|e| e.to.node == node.id && e.kind.is_empty()) {
66            out.push_str(&format!("      <bpmn:incoming>{}</bpmn:incoming>\n", escaped(&e.id)));
67        }
68        for e in graph.edges.iter().filter(|e| e.from.node == node.id && e.kind.is_empty()) {
69            out.push_str(&format!("      <bpmn:outgoing>{}</bpmn:outgoing>\n", escaped(&e.id)));
70        }
71        if kind == "intermediateCatchEvent" {
72            out.push_str("      <bpmn:timerEventDefinition />\n");
73        }
74        out.push_str(&format!("    </bpmn:{kind}>\n"));
75    }
76    for edge in &graph.edges {
77        let tag = match edge.kind.as_str() {
78            ASSOCIATION => "association",
79            MESSAGE => continue,
80            _ => "sequenceFlow",
81        };
82        if !mine.iter().any(|n| n.id == edge.from.node) {
83            continue;
84        }
85        out.push_str(&format!(
86            "    <bpmn:{tag} id=\"{}\" sourceRef=\"{}\" targetRef=\"{}\"{} />\n",
87            escaped(&edge.id),
88            escaped(&edge.from.node),
89            escaped(&edge.to.node),
90            label(edge)
91        ));
92    }
93    out.push_str("  </bpmn:process>\n");
94}
95
96fn label(edge: &FlowEdge) -> String {
97    edge.label.as_deref().map(|l| format!(" name=\"{}\"", escaped(l))).unwrap_or_default()
98}

100/// The process as BPMN 2.0 XML, with its drawing, named `name`. With participants, a
101/// collaboration holds them, each with a process of its own, and the messages between them.
102pub fn to_xml(graph: &FlowGraph, name: &str) -> String {
103    let mut out = String::from(
104        "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n\
105<bpmn:definitions xmlns:bpmn=\"http://www.omg.org/spec/BPMN/20100524/MODEL\" \
106xmlns:bpmndi=\"http://www.omg.org/spec/BPMN/20100524/DI\" \
107xmlns:dc=\"http://www.omg.org/spec/DD/20100524/DC\" \
108xmlns:di=\"http://www.omg.org/spec/DD/20100524/DI\" \
109id=\"Definitions_1\" targetNamespace=\"http://ravenrs.dev/bpmn\" exporter=\"RavenRS\">\n",
110    );
111    let pools: Vec<&FlowNode> = graph.nodes.iter().filter(|n| n.kind == PARTICIPANT).collect();
112    let plane = if pools.is_empty() {
113        write_process(&mut out, graph, "Process_1", name, &|_| true);
114        "Process_1"
115    } else {
116        out.push_str("  <bpmn:collaboration id=\"Collaboration_1\">\n");
117        for pool in &pools {
118            out.push_str(&format!(
119                "    <bpmn:participant id=\"{0}\" name=\"{1}\" processRef=\"Process_{0}\" />\n",
120                escaped(&pool.id),
121                escaped(&pool.title)
122            ));
123        }
124        for edge in graph.edges.iter().filter(|e| e.kind == MESSAGE) {
125            out.push_str(&format!(
126                "    <bpmn:messageFlow id=\"{}\" sourceRef=\"{}\" targetRef=\"{}\"{} />\n",
127                escaped(&edge.id),
128                escaped(&edge.from.node),
129                escaped(&edge.to.node),
130                label(edge)
131            ));
132        }
133        out.push_str("  </bpmn:collaboration>\n");
134        for pool in &pools {
135            let id = pool.id.clone();
136            let inside =
137                move |n: &FlowNode| elements::participant_of(graph, &n.id).as_deref() == Some(id.as_str());
138            write_process(&mut out, graph, &format!("Process_{}", pool.id), &pool.title, &inside);
139        }
140        // What sits in no participant still belongs to a process.
141        let loose = |n: &FlowNode| n.kind != PARTICIPANT && elements::participant_of(graph, &n.id).is_none();
142        if graph.nodes.iter().any(|n| loose(n) && n.kind != LANE) {
143            write_process(&mut out, graph, "Process_1", name, &loose);
144        }
145        "Collaboration_1"
146    };
147    out.push_str(&format!(
148        "  <bpmndi:BPMNDiagram id=\"BPMNDiagram_1\">\n    <bpmndi:BPMNPlane id=\"BPMNPlane_1\" bpmnElement=\"{plane}\">\n"
149    ));
150    for node in &graph.nodes {
151        let horizontal =
152            if node.kind == LANE || node.kind == PARTICIPANT { " isHorizontal=\"true\"" } else { "" };
153        out.push_str(&format!(
154            "      <bpmndi:BPMNShape id=\"{0}_di\" bpmnElement=\"{0}\"{horizontal}>\n        <dc:Bounds x=\"{1}\" y=\"{2}\" width=\"{3}\" height=\"{4}\" />\n      </bpmndi:BPMNShape>\n",
155            escaped(&node.id),
156            node.x,
157            node.y,
158            node.width,
159            node.height
160        ));
161    }
162    for edge in &graph.edges {
163        let Some(points) = graph.edge_controls(edge) else { continue };
164        out.push_str(&format!(
165            "      <bpmndi:BPMNEdge id=\"{0}_di\" bpmnElement=\"{0}\">\n",
166            escaped(&edge.id)
167        ));
168        for p in points {
169            out.push_str(&format!("        <di:waypoint x=\"{}\" y=\"{}\" />\n", p.x.round(), p.y.round()));
170        }
171        out.push_str("      </bpmndi:BPMNEdge>\n");
172    }
173    out.push_str("    </bpmndi:BPMNPlane>\n  </bpmndi:BPMNDiagram>\n</bpmn:definitions>\n");
174    out
175}

177/// What was read of one element, before it is placed.
178#[derive(Default)]
179struct Read {
180    /// Each element: its id, kind, name and the process it is in.
181    nodes: Vec<(String, String, String, Option<String>)>,
182    /// Each participant: its id, name and the process it holds.
183    participants: Vec<(String, String, Option<String>)>,
184    /// The process each lane is in.
185    lane_process: Vec<(String, Option<String>)>,
186    flows: Vec<(String, String, String, String, Option<String>)>,
187    lanes: Vec<(String, String, Vec<String>)>,
188    bounds: Vec<(String, Rect)>,
189    waypoints: Vec<(String, Vec<Point>)>,
190    process: Option<String>,
191}

193fn attribute(tag: &BytesStart<'_>, name: &str) -> Option<String> {
194    let found = tag.attributes().flatten().find(|a| a.key.local_name().as_ref() == name.as_bytes())?;
195    let raw = String::from_utf8_lossy(&found.value).into_owned();
196    Some(quick_xml::escape::unescape(&raw).map_or(raw.clone(), |text| text.into_owned()))
197}
198
199fn number(tag: &BytesStart<'_>, name: &str) -> f32 {
200    attribute(tag, name).and_then(|v| v.parse().ok()).unwrap_or(0.0)
201}

203/// The process named in the XML, as a diagram, and its name. Elements this editor does not
204/// draw are read as tasks, so nothing of the process is lost; what has no drawing is placed
205/// by arranging.
206pub fn from_xml(text: &str) -> Result<(FlowGraph, String), String> {
207    let mut reader = Reader::from_str(text);
208    reader.config_mut().trim_text(true);
209    let mut read = Read::default();
210    let mut open: Vec<String> = Vec::new();
211    let mut element: Option<String> = None;
212    let mut lane: Option<(String, String, Vec<String>)> = None;
213    let mut in_process: Option<String> = None;
214    let mut text_of: Option<String> = None;
215    loop {
216        let event = reader.read_event().map_err(|e| format!("XML: {e}"))?;
217        match &event {
218            Event::Start(tag) | Event::Empty(tag) => {
219                let local = String::from_utf8_lossy(tag.local_name().as_ref()).into_owned();
220                let id = attribute(tag, "id").unwrap_or_default();
221                match local.as_str() {
222                    "process" => {
223                        if read.process.is_none() {
224                            read.process = attribute(tag, "name");
225                        }
226                        in_process = Some(id.clone());
227                    }
228                    "participant" => read.participants.push((
229                        id,
230                        attribute(tag, "name").unwrap_or_default(),
231                        attribute(tag, "processRef"),
232                    )),
233                    "lane" => {
234                        read.lane_process.push((id.clone(), in_process.clone()));
235                        lane = Some((id, attribute(tag, "name").unwrap_or_default(), Vec::new()));
236                    }
237                    "sequenceFlow" | "association" | "messageFlow" => {
238                        let kind = match local.as_str() {
239                            "association" => ASSOCIATION,
240                            "messageFlow" => MESSAGE,
241                            _ => "",
242                        };
243                        read.flows.push((
244                            id,
245                            kind.to_owned(),
246                            attribute(tag, "sourceRef").unwrap_or_default(),
247                            attribute(tag, "targetRef").unwrap_or_default(),
248                            attribute(tag, "name").filter(|n| !n.is_empty()),
249                        ));
250                    }
251                    "BPMNShape" => {
252                        element = attribute(tag, "bpmnElement");
253                    }
254                    "BPMNEdge" => {
255                        let e = attribute(tag, "bpmnElement").unwrap_or_default();
256                        read.waypoints.push((e.clone(), Vec::new()));
257                        element = Some(e);
258                    }
259                    "Bounds" => {
260                        if let Some(e) = &element {
261                            let r = Rect::new(
262                                number(tag, "x"),
263                                number(tag, "y"),
264                                number(tag, "width"),
265                                number(tag, "height"),
266                            );
267                            read.bounds.push((e.clone(), r));
268                        }
269                    }
270                    "waypoint" => {
271                        if let Some((_, points)) = read.waypoints.last_mut() {
272                            points.push(Point::new(number(tag, "x"), number(tag, "y")));
273                        }
274                    }
275                    "flowNodeRef" => text_of = Some("flowNodeRef".into()),
276                    "text" if open.last().is_some_and(|o| o == "textAnnotation") => {
277                        text_of = Some("text".into())
278                    }
279                    _ if elements::family(&local).is_some_and(|f| f != Family::Lane)
280                        || local.ends_with("Task")
281                        || local.ends_with("Event")
282                        || local.ends_with("Gateway")
283                        || local == "subProcess"
284                        || local == "callActivity" =>
285                    {
286                        let kind = if elements::family(&local).is_some() {
287                            local.clone()
288                        } else if local.ends_with("Event") {
289                            "intermediateCatchEvent".into()
290                        } else if local.ends_with("Gateway") {
291                            "exclusiveGateway".into()
292                        } else {
293                            "task".into()
294                        };
295                        read.nodes.push((
296                            id,
297                            kind,
298                            attribute(tag, "name").unwrap_or_default(),
299                            in_process.clone(),
300                        ));
301                    }
302                    _ => {}
303                }
304                if matches!(event, Event::Start(_)) {
305                    open.push(local);
306                }
307            }
308            Event::Text(t) => {
309                let value = t.decode().map(|v| v.into_owned()).unwrap_or_default();
310                match text_of.take().as_deref() {
311                    Some("flowNodeRef") => {
312                        if let Some((_, _, members)) = &mut lane {
313                            members.push(value.trim().to_owned());
314                        }
315                    }
316                    Some("text") => {
317                        if let Some(last) = read.nodes.last_mut().filter(|n| n.1 == "textAnnotation") {
318                            last.2 = value.trim().to_owned();
319                        }
320                    }
321                    _ => {}
322                }
323            }
324            Event::End(tag) => {
325                let local = String::from_utf8_lossy(tag.local_name().as_ref()).into_owned();
326                if local == "process" {
327                    in_process = None;
328                }
329                if local == "lane" {
330                    read.lanes.extend(lane.take());
331                }
332                if local == "BPMNShape" || local == "BPMNEdge" {
333                    element = None;
334                }
335                open.pop();
336            }
337            Event::Eof => break,
338            _ => {}
339        }
340    }
341    build(read)
342}

344/// The port of `node` of this direction on the side facing `toward`.
345fn facing(node: &FlowNode, direction: PortDirection, toward: Point) -> Option<String> {
346    node.ports
347        .iter()
348        .filter(|p| p.direction == direction)
349        .filter_map(|p| Some((node.port_point(&p.id)?, p.id.clone())))
350        .min_by(|a, b| {
351            let d = |q: Point| (q.x - toward.x).hypot(q.y - toward.y);
352            d(a.0).total_cmp(&d(b.0))
353        })
354        .map(|(_, id)| id)
355}

357fn build(read: Read) -> Result<(FlowGraph, String), String> {
358    if read.nodes.is_empty() {
359        return Err("Nenhum elemento de processo no arquivo".into());
360    }
361    let mut graph = FlowGraph::new();
362    let placed = |id: &str| read.bounds.iter().find(|(e, _)| e == id).map(|(_, r)| *r);
363    // The participant that holds a process.
364    let owner = |process: &Option<String>| {
365        read.participants.iter().find(|(_, _, p)| p.is_some() && p == process).map(|(id, _, _)| id.clone())
366    };
367    for (id, name, _) in &read.participants {
368        let mut pool = elements::element(PARTICIPANT, id, name);
369        if let Some(r) = placed(id) {
370            pool = FlowNode { x: r.x(), y: r.y(), width: r.width(), height: r.height(), ..pool };
371        }
372        graph = graph.node(pool);
373    }
374    for (id, name, _) in &read.lanes {
375        let mut lane = elements::element(LANE, id, name);
376        if let Some(r) = placed(id) {
377            lane = FlowNode { x: r.x(), y: r.y(), width: r.width(), height: r.height(), ..lane };
378        }
379        let process = read.lane_process.iter().find(|(l, _)| l == id).and_then(|(_, p)| p.clone());
380        if let Some(pool) = owner(&process) {
381            lane = lane.parent(pool);
382        }
383        graph = graph.node(lane);
384    }
385    for (id, kind, name, process) in &read.nodes {
386        let mut node = elements::element(kind, id, name);
387        if let Some(r) = placed(id) {
388            node = FlowNode {
389                x: r.x(),
390                y: r.y(),
391                width: r.width().max(1.0),
392                height: r.height().max(1.0),
393                ..node
394            };
395        }
396        if let Some((lane, _, _)) = read.lanes.iter().find(|(_, _, members)| members.contains(id)) {
397            node = node.parent(lane.clone());
398        } else if let Some(pool) = owner(process) {
399            node = node.parent(pool);
400        }
401        graph = graph.node(node);
402    }
403    let all_placed = read.nodes.iter().all(|(id, _, _, _)| placed(id).is_some());
404    for (id, kind, from, to, name) in &read.flows {
405        let (Some(a), Some(b)) = (graph.find_node(from).cloned(), graph.find_node(to).cloned()) else {
406            continue;
407        };
408        let route = read.waypoints.iter().find(|(e, _)| e == id).map(|(_, p)| p.clone()).unwrap_or_default();
409        // The drawing starts and ends on the ports it uses; without one, the sides facing.
410        let first_toward = route.first().copied().unwrap_or(b.rect().center());
411        let last_from = route.last().copied().unwrap_or(a.rect().center());
412        let out = facing(&a, PortDirection::Output, first_toward)
413            .or_else(|| facing(&a, PortDirection::Input, first_toward));
414        let into = facing(&b, PortDirection::Input, last_from)
415            .or_else(|| facing(&b, PortDirection::Output, last_from));
416        let (Some(out), Some(into)) = (out, into) else { continue };
417        let mut edge =
418            elements::connection(id, FlowEndpoint::new(from, out), FlowEndpoint::new(to, into), kind);
419        edge.label = name.clone();
420        // The drawing's own corners, between its ends, become bend points.
421        if route.len() > 2 {
422            edge.waypoints = route[1..route.len() - 1].iter().map(|p| (p.x, p.y)).collect();
423        }
424        graph = graph.edge(edge);
425    }
426    if !all_placed {
427        FlowLayout::new(FlowDirection::Right).apply(&mut graph);
428    }
429    Ok((graph, read.process.unwrap_or_default()))
430}
