1//! BPMN 2.0 for the RavenRS diagram components, as data and functions only: the components
2//! are the library's own, and this crate hands them what BPMN is.
3//!
4//! - [`notation`] is BPMN as a [`FlowNotation`]: its elements grouped for a palette, how
5//!   tasks look, which connections may be made and how each is drawn (sequence, association,
6//!   message), what comes next from the keys, the checks of a process and the BPMN XML
7//!   format. Give it to `FlowCanvas`, `FlowToolbox`, `FlowInspector` and `FlowLibrary`.
8//! - [`xml`] writes a process as BPMN 2.0 XML with its drawing (BPMNDI) and reads what other
9//!   tools write, elements, lanes, labels and bend points included.
10//! - [`rules`] says what is wrong with a process.
11//! - [`plan`] is what a local model is asked, and how its answer becomes a process: the model
12//!   only writes a small plan the app checks, never the XML.
13//!
14//! Words shown are in Brazilian Portuguese.
15//!
16//! ```
17//! use ravenrs_reactive::State;
18//! use ravenrs_widgets::{FlowCanvas, FlowGraph, FlowToolbox};
19//!
20//! let bpmn = ravenrs_flow_bpmn::notation();
21//! let graph = State::new(ravenrs_flow_bpmn::example_process());
22//! let canvas = FlowCanvas::new(graph.binding()).notation(&bpmn);
23//! let palette = FlowToolbox::new().notation(&bpmn);
24//! let file = ravenrs_flow_bpmn::xml::to_xml(&graph.get(), "Compra");
25//! assert!(file.contains("<bpmn:process"));
26//! ```

37/// BPMN 2.0 as a notation for the diagram components.
38pub fn notation() -> FlowNotation {
39    let mut notation = FlowNotation::new();
40    for (group, family) in [
41        ("Eventos", Family::Event),
42        ("Atividades", Family::Activity),
43        ("Gateways", Family::Gateway),
44        ("Dados e anotações", Family::Artifact),
45        ("Raias e participantes", Family::Lane),
46    ] {
47        for (kind, caption, _) in ELEMENTS.iter().filter(|(_, _, f)| *f == family) {
48            notation = notation.model(group, *caption, elements::element(kind, kind, caption));
49        }
50    }
51    for (kind, caption, _) in ELEMENTS {
52        notation = notation.kind(kind, caption);
53    }
54    for (kind, word) in
55        [("task", "Tarefa"), ("userTask", "Usuário"), ("serviceTask", "Serviço"), ("scriptTask", "Script")]
56    {
57        notation = notation.look(kind, FlowKind::new(elements::icon(kind), word));
58    }
59    notation
60        .connection(|graph, edge| {
61            let mut shaped = FlowGraph { edges: vec![edge], ..graph.clone() };
62            elements::normalize(&mut shaped);
63            shaped.edges.pop().unwrap_or_default()
64        })
65        .validate(rules::validate)
66        .next_node(next)
67        .checks(rules::problems)
68        .runs(run_rules())
69        .format(bpmn_format())
70}

72/// How a process runs: a token leaves each start event, passes tasks, chooses one way at an
73/// exclusive gateway and one or more at an inclusive one (whose join waits only for the ways
74/// taken), waits for every way and splits at a parallel one, and stops at an end; it follows
75/// sequence flows only.
76pub fn run_rules() -> FlowRunRules {
77    FlowRunRules {
78        gate: std::rc::Rc::new(|node: &FlowNode| match node.kind.as_str() {
79            "startEvent" => FlowGate::Start,
80            "endEvent" => FlowGate::End,
81            "exclusiveGateway" => FlowGate::Choose,
82            "inclusiveGateway" => FlowGate::ChooseMany,
83            "parallelGateway" => FlowGate::Parallel,
84            kind => match elements::family(kind) {
85                Some(Family::Artifact | Family::Lane) | None
86                    if node.group || elements::family(kind).is_some() =>
87                {
88                    FlowGate::Outside
89                }
90                _ => FlowGate::Pass,
91            },
92        }),
93        follows: std::rc::Rc::new(|edge: &FlowEdge| edge.kind == SEQUENCE),
94    }
95}

97/// BPMN 2.0 XML as a file format of a `FlowLibrary`.
98pub fn bpmn_format() -> FlowFileFormat {
99    FlowFileFormat::new("BPMN 2.0", "bpmn", |text| xml::from_xml(text).map(|(graph, _)| graph), xml::to_xml)
100}

102/// The next element made with the arrow or Cmd/Ctrl+Enter: a task after anything that leads
103/// on, nothing after an end or outside the flow.
104pub fn next(from: &FlowNode, _side: PortSide) -> Option<FlowNode> {
105    match elements::family(&from.kind) {
106        Some(Family::Event | Family::Activity | Family::Gateway) if from.kind != "endEvent" => {
107            Some(elements::element("task", "task", "Nova tarefa"))
108        }
109        _ => None,
110    }
111}

113fn at(node: FlowNode, x: f32, y: f32) -> FlowNode {
114    FlowNode { x, y, ..node }
115}

117fn flow(id: &str, from: (&str, &str), to: (&str, &str)) -> FlowEdge {
118    elements::connection(id, FlowEndpoint::new(from.0, from.1), FlowEndpoint::new(to.0, to.1), SEQUENCE)
119}

121/// A customer and a store, each a participant with its process, an order passing between them
122/// as a message: a collaboration to start from.
123pub fn example_collaboration() -> FlowGraph {
124    use elements::{MESSAGE, PARTICIPANT, connection, element};
125    FlowGraph::new()
126        .node(at(element(PARTICIPANT, "customer", "Cliente"), 0.0, 0.0).size(900.0, 200.0))
127        .node(at(element(PARTICIPANT, "store", "Loja"), 0.0, 260.0).size(900.0, 200.0))
128        .node(at(element("startEvent", "want", "Quer comprar"), 80.0, 80.0).parent("customer"))
129        .node(at(element("userTask", "order", "Fazer pedido"), 180.0, 64.0).parent("customer"))
130        .node(at(element("endEvent", "asked", "Pedido feito"), 420.0, 80.0).parent("customer"))
131        .node(at(element("startEvent", "got", "Pedido recebido"), 80.0, 340.0).parent("store"))
132        .node(at(element("serviceTask", "ship", "Enviar"), 180.0, 324.0).parent("store"))
133        .node(at(element("endEvent", "sent", "Enviado"), 420.0, 340.0).parent("store"))
134        .edge(flow("s1", ("want", "out"), ("order", "in")))
135        .edge(flow("s2", ("order", "out"), ("asked", "in")))
136        .edge(flow("s3", ("got", "out"), ("ship", "in")))
137        .edge(flow("s4", ("ship", "out"), ("sent", "in")))
138        .edge(FlowEdge {
139            label: Some("Pedido".into()),
140            ..connection("m1", FlowEndpoint::new("order", "down"), FlowEndpoint::new("got", "top"), MESSAGE)
141        })
142}
