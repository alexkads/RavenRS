22/// The element's XML name for a type the model wrote.
23fn kind_of(tipo: &str) -> &'static str {
24    match tipo.trim().to_lowercase().as_str() {
25        "inicio" | "início" | "start" => "startEvent",
26        "fim" | "end" => "endEvent",
27        "usuario" | "usuário" | "user" => "userTask",
28        "servico" | "serviço" | "service" => "serviceTask",
29        "script" => "scriptTask",
30        "decisao" | "decisão" | "gateway" | "exclusivo" => "exclusiveGateway",
31        "paralelo" | "parallel" => "parallelGateway",
32        "espera" | "timer" | "aguardar" => "intermediateCatchEvent",
33        _ => "task",
34    }
35}

37/// A process drafted from a model's answer, with what had to be mended on the way.
38#[derive(Clone, Debug, Default, PartialEq)]
39pub struct Drafted {
40    /// The process, laid out from left to right.
41    pub graph: FlowGraph,
42    /// What the answer got wrong and was mended, one sentence each, for the person to see.
43    pub mended: Vec<String>,
44}

46/// The objects of the list after `"key"`, as text, each whole even when the list around them
47/// is broken: small models leave out a comma or close a list twice.
48fn objects<'a>(text: &'a str, key: &str) -> Vec<&'a str> {
49    let Some(at) = text.find(&format!("\"{key}\"")) else { return Vec::new() };
50    let Some(open) = text[at..].find('[').map(|i| at + i) else { return Vec::new() };
51    let mut found = Vec::new();
52    let (mut depth, mut start, mut quoted, mut escaped) = (0usize, None, false, false);
53    for (i, c) in text[open + 1..].char_indices() {
54        let i = open + 1 + i;
55        if quoted {
56            match (escaped, c) {
57                (true, _) => escaped = false,
58                (false, '\\') => escaped = true,
59                (false, '"') => quoted = false,
60                _ => {}
61            }
62            continue;
63        }
64        match c {
65            '"' => quoted = true,
66            '{' => {
67                if depth == 0 {
68                    start = Some(i);
69                }
70                depth += 1;
71            }
72            '}' if depth > 0 => {
73                depth -= 1;
74                if depth == 0
75                    && let Some(from) = start.take()
76                {
77                    found.push(&text[from..=i]);
78                }
79            }
80            ']' if depth == 0 => break,
81            _ => {}
82        }
83    }
84    found
85}

87/// The text after `"name":` in an object, read even when the object is not valid JSON.
88fn field(object: &str, name: &str) -> Option<String> {
89    if let Ok(Value::Object(map)) = serde_json::from_str::<Value>(object) {
90        return map.get(name).and_then(Value::as_str).map(|v| v.trim().to_owned());
91    }
92    let at = object.find(&format!("\"{name}\""))? + name.len() + 2;
93    let rest = object[at..].trim_start().strip_prefix(':')?.trim_start().strip_prefix('"')?;
94    Some(rest[..rest.find('"')?].trim().to_owned())
95}
96
97/// Every quoted text in an object, in order.
98fn texts(object: &str) -> Vec<String> {
99    object.split('"').skip(1).step_by(2).map(str::to_owned).collect()
100}

102/// The process a model's answer plans, laid out from left to right, with what had to be
103/// mended (a step without a type, a missing start or end); why not, when the answer is no plan.
104pub fn plan(answer: &str) -> Result<Drafted, String> {
105    let steps = objects(answer, "etapas");
106    if steps.is_empty() {
107        return Err(if answer.contains('{') {
108            "O plano não tem etapas".into()
109        } else {
110            "A resposta não tem um plano em JSON".into()
111        });
112    }
113    let mut mended = Vec::new();
114    let mut graph = FlowGraph::new();
115    for (i, step) in steps.iter().enumerate() {
116        let id = field(step, "id").filter(|s| !s.is_empty()).unwrap_or_else(|| format!("e{}", i + 1));
117        if graph.find_node(&id).is_some() {
118            mended.push(format!("A etapa {id} aparecia duas vezes; ficou a primeira"));
119            continue;
120        }
121        let tipo = match field(step, "tipo") {
122            Some(tipo) => tipo,
123            // A step called "fim" or "inicio" without a type is that.
124            None => {
125                let guess = if matches!(id.as_str(), "fim" | "inicio" | "início") {
126                    id.clone()
127                } else {
128                    "tarefa".into()
129                };
130                mended.push(format!("A etapa {id} veio sem tipo; ficou {guess}"));
131                guess
132            }
133        };
134        let name = field(step, "nome").or_else(|| texts(step).last().cloned()).unwrap_or_default();
135        graph = graph.node(elements::element(kind_of(&tipo), &id, &name));
136    }
137    let links = objects(answer, "ligacoes");
138    for (i, link) in links.iter().enumerate() {
139        let (from, to) = (field(link, "de").unwrap_or_default(), field(link, "para").unwrap_or_default());
140        let Some(a) = graph.find_node(&from) else {
141            mended.push(format!("A ligação {} sai de {from}, que não existe; ficou de fora", i + 1));
142            continue;
143        };
144        if graph.find_node(&to).is_none() {
145            mended.push(format!("A ligação {} vai para {to}, que não existe; ficou de fora", i + 1));
146            continue;
147        }
148        if a.kind == "endEvent" || from == to {
149            mended.push(format!("A ligação de {from} para {to} não cabe num processo; ficou de fora"));
150            continue;
151        }
152        // A split leaves by its side, then below, then above; the rest leave by the side.
153        let used = graph.edges.iter().filter(|e| e.from.node == from).count();
154        let out = match (a.kind.ends_with("Gateway"), used) {
155            (true, 1) => "down",
156            (true, 2) => "up",
157            _ => "out",
158        };
159        let mut edge = elements::connection(
160            &format!("f{}", i + 1),
161            FlowEndpoint::new(from, out),
162            FlowEndpoint::new(to, "in"),
163            SEQUENCE,
164        );
165        edge.label = field(link, "rotulo").filter(|l| !l.is_empty());
166        graph = graph.edge(edge);
167    }
168    // A process starts and ends: what is missing is added and said.
169    let leads_on = |g: &FlowGraph, id: &str| g.edges.iter().any(|e| e.from.node == id);
170    let reached = |g: &FlowGraph, id: &str| g.edges.iter().any(|e| e.to.node == id);
171    if !graph.nodes.iter().any(|n| n.kind == "startEvent") {
172        let first = graph.nodes.iter().find(|n| !reached(&graph, &n.id)).map(|n| n.id.clone());
173        graph = graph.node(elements::element("startEvent", "inicio_auto", "Início"));
174        if let Some(first) = first {
175            graph = graph.edge(elements::connection(
176                "f_inicio",
177                FlowEndpoint::new("inicio_auto", "out"),
178                FlowEndpoint::new(first, "in"),
179                SEQUENCE,
180            ));
181        }
182        mended.push("Faltava o início; acrescentei um".into());
183    }
184    if !graph.nodes.iter().any(|n| n.kind == "endEvent") {
185        let loose: Vec<String> = graph
186            .nodes
187            .iter()
188            .filter(|n| n.kind != "startEvent" && !leads_on(&graph, &n.id))
189            .map(|n| n.id.clone())
190            .collect();
191        graph = graph.node(elements::element("endEvent", "fim_auto", "Fim"));
192        for (i, id) in loose.iter().enumerate() {
193            graph = graph.edge(elements::connection(
194                &format!("f_fim{i}"),
195                FlowEndpoint::new(id.clone(), "out"),
196                FlowEndpoint::new("fim_auto", "in"),
197                SEQUENCE,
198            ));
199        }
200        mended.push("Faltava o fim; acrescentei um".into());
201    }
202    FlowLayout::new(FlowDirection::Right).gaps(64.0, 48.0).apply(&mut graph);
203    Ok(Drafted { graph, mended })
204}

206/// The process in words, for the model to review.
207pub fn summary(graph: &FlowGraph) -> String {
208    let name = |id: &str| {
209        graph.find_node(id).map_or(id.to_owned(), |n| format!("{} ({})", n.title, elements::caption(&n.kind)))
210    };
211    let mut out = String::from("Etapas:\n");
212    for node in graph.nodes.iter().filter(|n| elements::family(&n.kind).is_some()) {
213        out.push_str(&format!("- {}\n", name(&node.id)));
214    }
215    out.push_str("Ligações:\n");
216    for e in graph.edges.iter().filter(|e| e.kind.is_empty()) {
217        let label = e.label.as_deref().map(|l| format!(" [{l}]")).unwrap_or_default();
218        out.push_str(&format!("- {} -> {}{label}\n", name(&e.from.node), name(&e.to.node)));
219    }
220    out
221}
