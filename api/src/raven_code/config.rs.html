1//! What Raven Code remembers between runs, in the user's folder of settings: the projects opened
2//! last, and how each project was left (the arrangement of the window, the open files and where
3//! their carets were). Nothing is written inside the projects. Tests give it a folder of their
4//! own with [`Config::at`].
5
6use std::fs;
7use std::hash::{Hash, Hasher};
8use std::io;
9use std::path::{Path, PathBuf};
10
11/// How many projects the list of recent ones keeps.
12pub const RECENT: usize = 10;
13
14/// The first line of a saved session, with the version of its format.
15const SESSION_HEADER: &str = "raven-code session 1";
16
17/// Where Raven Code keeps what it remembers.
18#[derive(Clone, Debug, PartialEq, Eq)]
19pub struct Config {
20    dir: PathBuf,
21}
22
23impl Config {
24    /// In this folder, made when something is first written.
25    pub fn at(dir: impl Into<PathBuf>) -> Self {
26        Self { dir: dir.into() }
27    }
28
29    /// In the user's folder of settings: `~/Library/Application Support/raven-code` on a Mac,
30    /// `%APPDATA%\raven-code` on Windows, and `$XDG_CONFIG_HOME/raven-code` (or
31    /// `~/.config/raven-code`) elsewhere. `None` when the system says no home.
32    pub fn of_user() -> Option<Self> {
33        let var = |name: &str| std::env::var_os(name).filter(|v| !v.is_empty()).map(PathBuf::from);
34        let base = if cfg!(target_os = "macos") {
35            var("HOME").map(|home| home.join("Library/Application Support"))
36        } else if cfg!(windows) {
37            var("APPDATA")
38        } else {
39            var("XDG_CONFIG_HOME").or_else(|| var("HOME").map(|home| home.join(".config")))
40        };
41        base.map(|base| Self::at(base.join("raven-code")))
42    }
43
44    /// Its folder.
45    pub fn dir(&self) -> &Path {
46        &self.dir
47    }
48
49    fn recent_file(&self) -> PathBuf {
50        self.dir.join("recent.txt")
51    }
52
53    /// The projects opened last, the latest first.
54    pub fn recent(&self) -> Vec<PathBuf> {
55        fs::read_to_string(self.recent_file())
56            .map(|text| text.lines().filter(|line| !line.is_empty()).map(PathBuf::from).collect())
57            .unwrap_or_default()
58    }
59
60    /// Puts a project at the top of the recent ones, once, keeping ten.
61    pub fn remember(&self, project: &Path) -> io::Result<()> {
62        let mut recent = self.recent();
63        recent.retain(|p| p != project);
64        recent.insert(0, project.to_path_buf());
65        recent.truncate(RECENT);
66        self.write_recent(&recent)
67    }
68
69    /// Takes a project out of the recent ones, as when its folder is gone.
70    pub fn forget(&self, project: &Path) -> io::Result<()> {
71        let mut recent = self.recent();
72        recent.retain(|p| p != project);
73        self.write_recent(&recent)
74    }
75
76    fn write_recent(&self, recent: &[PathBuf]) -> io::Result<()> {
77        fs::create_dir_all(&self.dir)?;
78        let text: String = recent.iter().map(|p| format!("{}\n", p.display())).collect();
79        fs::write(self.recent_file(), text)
80    }
81
82    /// Where the session of a project is kept: its name and a hash of its whole path, so two
83    /// projects of the same name do not share one.
84    fn session_file(&self, project: &Path) -> PathBuf {
85        let mut hasher = std::hash::DefaultHasher::new();
86        project.hash(&mut hasher);
87        let name = project.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
88        let name: String =
89            name.chars().map(|c| if c.is_alphanumeric() || c == '-' { c } else { '_' }).collect();
90        self.dir.join("sessions").join(format!("{name}-{:016x}.txt", hasher.finish()))
91    }
92
93    /// How a project was left, if it was.
94    pub fn session(&self, project: &Path) -> Option<Session> {
95        Session::from_text(&fs::read_to_string(self.session_file(project)).ok()?)
96    }
97
98    /// Keeps how a project is left.
99    pub fn save_session(&self, project: &Path, session: &Session) -> io::Result<()> {
100        let file = self.session_file(project);
101        if let Some(parent) = file.parent() {
102            fs::create_dir_all(parent)?;
103        }
104        fs::write(file, session.to_text())
105    }
106}
107
108/// How a project was left: what the window looked like and what was open in it.
109#[derive(Clone, Debug, Default, PartialEq, Eq)]
110pub struct Session {
111    /// The arrangement of the dock, as `DockLayout::to_text` writes it.
112    pub layout: String,
113    /// The open files, in the order of their tabs, each with where its caret was.
114    pub open: Vec<(String, usize)>,
115    /// The file at the front.
116    pub active: Option<String>,
117}
118
119impl Session {
120    /// As text: a header, the file at the front, one line per open file, then the layout.
121    pub fn to_text(&self) -> String {
122        let mut text = format!("{SESSION_HEADER}\n");
123        if let Some(active) = &self.active {
124            text.push_str(&format!("active {active}\n"));
125        }
126        for (path, caret) in &self.open {
127            text.push_str(&format!("open {caret} {path}\n"));
128        }
129        text.push_str("layout\n");
130        text.push_str(&self.layout);
131        text
132    }
133
134    /// Reads [`to_text`](Self::to_text); `None` for another format or version.
135    pub fn from_text(text: &str) -> Option<Self> {
136        let mut lines = text.lines();
137        if lines.next()? != SESSION_HEADER {
138            return None;
139        }
140        let mut session = Self::default();
141        for line in lines.by_ref() {
142            if line == "layout" {
143                break;
144            } else if let Some(active) = line.strip_prefix("active ") {
145                session.active = Some(active.to_owned());
146            } else if let Some(open) = line.strip_prefix("open ") {
147                let (caret, path) = open.split_once(' ')?;
148                session.open.push((path.to_owned(), caret.parse().ok()?));
149            }
150        }
151        session.layout = lines.collect::<Vec<_>>().join("\n");
152        Some(session)
153    }
154}
