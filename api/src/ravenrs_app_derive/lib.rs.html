10/// Declares a struct as a business object: an entity for the model, and the way to and from
11/// a record.
12///
13/// On the struct, `#[object(caption = "...", navigation = "...", icon = "...",
14/// list_criteria = "...", form_columns = 2, cloneable, transient, checks_on_leave, list_editable,
15/// data_access = "server", form_actions = "name")]`, where `data_access` is `client`, `server` or
16/// `data_view`.
17///
18/// On a field, `#[field(...)]`:
19///
20/// * `key` numbered by the application, or `typed_key` typed by people. Exactly one field has one.
21///   With `guid`, a `key` is made up when a record is created instead of being numbered.
22/// * `caption = "..."` what people see.
23/// * `default_property` what names a record. `read_only`. `no_list`, `no_detail`, `lookup`.
24/// * What it holds, when the type does not say: `money`, `long`, `color`, `date`, `guid`,
25///   `decimals = 2`, `reference = "Entity"`, `collection = "Entity"`, `tags = "a,b"`.
26/// * `calculated = "[a] + [b]"`, in the criteria language, checked here.
27/// * `default = ...` what a new record starts with.
28/// * `required`, `unique`, `min = 1`, `max = 10`, `matches = "^[A-Z]+$"`: what it has to meet on
29///   saving, and `message = "..."` what to say instead of the usual words.
30/// * On a form: `editor = "..."`, `placeholder = "..."`, `hint = "..."`, `wide`, `autofocus`,
31///   `group = "..."` for the titled group it goes in, `edit_mask = "(00) 0000-0000"`,
32///   `predefined = "a,b"` for values to pick from, `criteria_of = "Entity"` for a filter builder.
33/// * `display_format = "{:.2} %"`, how its value shows when it is not being edited.
34/// * On a reference: `lookup_filter = "[flow] = ?flow"`, `none_caption = "..."`. On a collection:
35///   `back_reference = "..."`, `aggregated`. `secret` never leaves a secured space. `non_cloneable` stays behind on cloning, and
36///   `most_bytes = 1024` limits a picture or a file.
37///
38/// On the struct, as many as needed:
39///
40/// * `#[rule(criteria = "[a] <= [b]", message = "...", context = "Pay", target = "[c] > 0")]`, or
41///   `#[rule(unique = "a,b")]`. With `warning` people can go on anyway, and with `information` it
42///   only leaves a note. A message can say `{field}` and `{reference.field}`.
43/// * `#[appearance(names = "a,b*", criteria = "...", font = "danger", back = "warning", bold,
44///   enabled = false, visible = false, priority = 1, context = "list", action)]`. With `layout`,
45///   the names are groups of a form; with `action`, actions. A context is `list`, `detail` or a
46///   view's id. Tints are `danger`, `warning`, `success`, `accent` and `muted`.
47/// * `#[variant(id = "...", caption = "...", filter = "...", columns = "a,b")]` and
48///   `#[list_filter(id = "...", caption = "...", filter = "...")]`, for people to pick on the list.
49#[proc_macro_derive(BusinessObject, attributes(object, field, rule, appearance, variant, list_filter))]
50pub fn derive_business_object(input: TokenStream) -> TokenStream {
51    let input = parse_macro_input!(input as DeriveInput);
52    business_object(&input).unwrap_or_else(Error::into_compile_error).into()
53}

78fn object_attrs(input: &DeriveInput) -> Result<ObjectAttrs> {
79    let mut attrs = ObjectAttrs::default();
80    for attr in input.attrs.iter().filter(|a| a.path().is_ident("object")) {
81        attr.parse_nested_meta(|meta| {
82            if meta.path.is_ident("caption") {
83                attrs.caption = Some(meta.value()?.parse()?);
84            } else if meta.path.is_ident("navigation") {
85                attrs.navigation = Some(meta.value()?.parse()?);
86            } else if meta.path.is_ident("icon") {
87                attrs.icon = Some(meta.value()?.parse()?);
88            } else if meta.path.is_ident("list_criteria") {
89                let text: LitStr = meta.value()?.parse()?;
90                Criteria::parse(&text.value()).map_err(|e| Error::new(text.span(), e.to_string()))?;
91                attrs.list_criteria = Some(text);
92            } else if meta.path.is_ident("form_columns") {
93                attrs.form_columns = Some(meta.value()?.parse()?);
94            } else if meta.path.is_ident("cloneable") {
95                attrs.cloneable = true;
96            } else if meta.path.is_ident("transient") {
97                attrs.transient = true;
98            } else if meta.path.is_ident("checks_on_leave") {
99                attrs.checks_on_leave = true;
100            } else if meta.path.is_ident("list_editable") {
101                attrs.list_editable = true;
102            } else if meta.path.is_ident("data_access") {
103                let text: LitStr = meta.value()?.parse()?;
104                if !["client", "server", "data_view"].contains(&text.value().as_str()) {
105                    return Err(Error::new(text.span(), "use client, server or data_view"));
106                }
107                attrs.data_access = Some(text);
108            } else if meta.path.is_ident("form_actions") {
109                attrs.form_actions.push(meta.value()?.parse()?);
110            } else {
111                return Err(meta.error(
112                    "unknown setting: use caption, navigation, icon, list_criteria, form_columns, cloneable, \
113                     transient, checks_on_leave, list_editable, data_access or form_actions",
114                ));
115            }
116            Ok(())
117        })?;
118    }
119    Ok(attrs)
120}

160fn set_kind(attrs: &mut FieldAttrs, path: &syn::Path, kind: Tokens) -> Result<()> {
161    if attrs.kind.is_some() {
162        return Err(Error::new(path.span(), "a field holds one kind of thing: this says a second"));
163    }
164    attrs.kind = Some(kind);
165    Ok(())
166}

168fn field_attrs(field: &syn::Field) -> Result<FieldAttrs> {
169    let mut attrs = FieldAttrs::default();
170    for attr in field.attrs.iter().filter(|a| a.path().is_ident("field")) {
171        attr.parse_nested_meta(|meta| {
172            let path = &meta.path;
173            let named = |name: &str| path.is_ident(name);
174            if named("key") {
175                attrs.key = true;
176            } else if named("typed_key") {
177                attrs.typed_key = true;
178            } else if named("caption") {
179                attrs.caption = Some(meta.value()?.parse()?);
180            } else if named("default_property") {
181                attrs.default_property = true;
182            } else if named("read_only") {
183                attrs.read_only = true;
184            } else if named("no_list") {
185                attrs.no_list = true;
186            } else if named("no_detail") {
187                attrs.no_detail = true;
188            } else if named("lookup") {
189                attrs.lookup = true;
190            } else if named("money") {
191                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Money))?;
192            } else if named("long") {
193                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::LongText))?;
194            } else if named("color") {
195                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Color))?;
196            } else if named("date") {
197                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Date))?;
198            } else if named("guid") {
199                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Guid))?;
200            } else if named("tags") {
201                let options: LitStr = meta.value()?.parse()?;
202                let names = split_names(&options);
203                set_kind(
204                    &mut attrs,
205                    path,
206                    quote!(::ravenrs_app::FieldKind::Tags(::std::vec![
207                        #(::ravenrs_app::EnumOption::new(#names, #names)),*
208                    ])),
209                )?;
210            } else if named("decimals") {
211                let places: LitInt = meta.value()?.parse()?;
212                let places: u8 = places.base10_parse()?;
213                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Decimal(#places)))?;
214            } else if named("reference") {
215                let target: LitStr = meta.value()?.parse()?;
216                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Reference(#target.to_owned())))?;
217            } else if named("collection") {
218                let target: LitStr = meta.value()?.parse()?;
219                set_kind(&mut attrs, path, quote!(::ravenrs_app::FieldKind::Collection(#target.to_owned())))?;
220            } else if named("calculated") {
221                attrs.calculated = Some(meta.value()?.parse()?);
222            } else if named("default") {
223                attrs.default = Some(meta.value()?.parse()?);
224            } else if named("required") {
225                attrs.required = true;
226            } else if named("unique") {
227                attrs.unique = true;
228            } else if named("min") {
229                attrs.min = Some(meta.value()?.parse()?);
230            } else if named("max") {
231                attrs.max = Some(meta.value()?.parse()?);
232            } else if named("matches") {
233                let pattern: LitStr = meta.value()?.parse()?;
234                regex_lite::Regex::new(&pattern.value())
235                    .map_err(|e| Error::new(pattern.span(), format!("this pattern is not valid: {e}")))?;
236                attrs.matches = Some(pattern);
237            } else if named("message") {
238                attrs.message = Some(meta.value()?.parse()?);
239            } else if named("editor") {
240                attrs.editor = Some(meta.value()?.parse()?);
241            } else if named("placeholder") {
242                attrs.placeholder = Some(meta.value()?.parse()?);
243            } else if named("hint") {
244                attrs.hint = Some(meta.value()?.parse()?);
245            } else if named("lookup_filter") {
246                let text: LitStr = meta.value()?.parse()?;
247                Criteria::parse(&text.value()).map_err(|e| Error::new(text.span(), e.to_string()))?;
248                attrs.lookup_filter = Some(text);
249            } else if named("none_caption") {
250                attrs.none_caption = Some(meta.value()?.parse()?);
251            } else if named("back_reference") {
252                attrs.back_reference = Some(meta.value()?.parse()?);
253            } else if named("most_bytes") {
254                attrs.most_bytes = Some(meta.value()?.parse()?);
255            } else if named("wide") {
256                attrs.wide = true;
257            } else if named("autofocus") {
258                attrs.autofocus = true;
259            } else if named("non_cloneable") {
260                attrs.non_cloneable = true;
261            } else if named("secret") {
262                attrs.secret = true;
263            } else if named("aggregated") {
264                attrs.aggregated = true;
265            } else if named("group") {
266                attrs.group = Some(meta.value()?.parse()?);
267            } else if named("display_format") {
268                attrs.display_format = Some(meta.value()?.parse()?);
269            } else if named("edit_mask") {
270                attrs.edit_mask = Some(meta.value()?.parse()?);
271            } else if named("predefined") {
272                attrs.predefined = Some(meta.value()?.parse()?);
273            } else if named("criteria_of") {
274                attrs.criteria_of = Some(meta.value()?.parse()?);
275            } else {
276                return Err(meta.error(
277                    "unknown setting: use key, typed_key, caption, default_property, read_only, no_list, \
278                     no_detail, lookup, money, long, color, date, guid, tags, decimals, reference, collection, \
279                     calculated, default, required, unique, min, max, matches, message, editor, placeholder, \
280                     hint, lookup_filter, none_caption, back_reference, most_bytes, wide, autofocus, \
281                     non_cloneable, secret, aggregated, group, display_format, edit_mask, predefined or criteria_of",
282                ));
283            }
284            Ok(())
285        })?;
286    }
287    if attrs.key && attrs.typed_key {
288        return Err(Error::new(field.span(), "a field is key or typed_key, not both"));
289    }
290    Ok(attrs)
291}

293fn check_expression(expression: &LitStr, names: &[String]) -> Result<()> {
294    let parsed = Criteria::parse(&expression.value())
295        .map_err(|e| Error::new(expression.span(), format!("this expression does not parse: {e}")))?;
296    for property in parsed.properties() {
297        if !names.contains(&property) {
298            return Err(Error::new(expression.span(), format!("this struct has no field `{property}`")));
299        }
300    }
301    Ok(())
302}

304fn field_rules(ident: &Ident, attrs: &FieldAttrs) -> Vec<Tokens> {
305    let name = ident.to_string();
306    let mut rules = Vec::new();
307    if attrs.required {
308        rules.push(quote!(::ravenrs_app::Rule::required(#name)));
309    }
310    if attrs.unique {
311        rules.push(quote!(::ravenrs_app::Rule::unique(#name)));
312    }
313    if attrs.min.is_some() || attrs.max.is_some() {
314        let limit = |value: &Option<syn::Expr>| match value {
315            Some(value) => quote!(::std::option::Option::Some((#value) as f64)),
316            None => quote!(::std::option::Option::None),
317        };
318        let (min, max) = (limit(&attrs.min), limit(&attrs.max));
319        rules.push(quote!(::ravenrs_app::Rule::range(#name, #min, #max)));
320    }
321    if let Some(pattern) = &attrs.matches {
322        rules.push(quote!(::ravenrs_app::Rule::matches(#name, #pattern)));
323    }
324    if let Some(message) = &attrs.message {
325        for rule in &mut rules {
326            *rule = quote!(#rule.message(#message));
327        }
328    }
329    rules
330}

332fn split_names(text: &LitStr) -> Vec<String> {
333    text.value().split(',').map(|name| name.trim().to_owned()).filter(|name| !name.is_empty()).collect()

347fn struct_rules(input: &DeriveInput, names: &[String]) -> Result<Vec<Tokens>> {
348    let mut rules = Vec::new();
349    for attr in input.attrs.iter().filter(|a| a.path().is_ident("rule")) {
350        let mut found = RuleAttrs::default();
351        attr.parse_nested_meta(|meta| {
352            if meta.path.is_ident("warning") {
353                found.warning = true;
354                return Ok(());
355            }
356            if meta.path.is_ident("information") {
357                found.information = true;
358                return Ok(());
359            }
360            let slot = if meta.path.is_ident("criteria") {
361                &mut found.criteria
362            } else if meta.path.is_ident("unique") {
363                &mut found.unique
364            } else if meta.path.is_ident("message") {
365                &mut found.message
366            } else if meta.path.is_ident("context") {
367                &mut found.context
368            } else if meta.path.is_ident("target") {
369                &mut found.target
370            } else {
371                return Err(meta.error(
372                    "unknown setting: use criteria, unique, message, context, target, warning or information",
373                ));
374            };
375            *slot = Some(meta.value()?.parse()?);
376            Ok(())
377        })?;
378        let id = format!("rule{}", rules.len());
379        let mut rule = match (&found.criteria, &found.unique) {
380            (Some(expression), None) => {
381                check_expression(expression, names)?;
382                quote!(::ravenrs_app::Rule::criteria(#id, #expression))
383            }
384            (None, Some(fields)) => {
385                let fields = split_names(fields);
386                if let Some(missing) = fields.iter().find(|f| !names.contains(f)) {
387                    return Err(Error::new(attr.span(), format!("this struct has no field `{missing}`")));
388                }
389                quote!(::ravenrs_app::Rule::new(
390                    #id,
391                    ::ravenrs_app::RuleKind::UniqueTogether(::std::vec![#(#fields.to_owned()),*]),
392                ))
393            }
394            _ => return Err(Error::new(attr.span(), "a rule says either criteria or unique")),
395        };
396        if let Some(text) = &found.message {
397            rule = quote!(#rule.message(#text));
398        }
399        if let Some(text) = &found.context {
400            rule = quote!(#rule.contexts([#text]));
401        }
402        if let Some(expression) = &found.target {
403            check_expression(expression, names)?;
404            rule = quote!(#rule.target_criteria(#expression));
405        }
406        if found.warning && found.information {
407            return Err(Error::new(attr.span(), "a rule is a warning or information, not both"));
408        }
409        if found.warning {
410            rule = quote!(#rule.warning());
411        }
412        if found.information {
413            rule = quote!(#rule.information());
414        }
415        rules.push(rule);
416    }
417    Ok(rules)
418}

420fn tint(text: &LitStr) -> Result<Tokens> {
421    Ok(match text.value().as_str() {
422        "danger" => quote!(::ravenrs_app::Tint::Danger),
423        "warning" => quote!(::ravenrs_app::Tint::Warning),
424        "success" => quote!(::ravenrs_app::Tint::Success),
425        "accent" => quote!(::ravenrs_app::Tint::Accent),
426        "muted" => quote!(::ravenrs_app::Tint::Muted),
427        _ => return Err(Error::new(text.span(), "use danger, warning, success, accent or muted")),
428    })
429}

431fn struct_appearance(input: &DeriveInput, names: &[String]) -> Result<Vec<Tokens>> {
432    let mut rules = Vec::new();
433    for attr in input.attrs.iter().filter(|a| a.path().is_ident("appearance")) {
434        let mut targets: Option<LitStr> = None;
435        let mut pieces = Vec::new();
436        attr.parse_nested_meta(|meta| {
437            let named = |name: &str| meta.path.is_ident(name);
438            if named("names") {
439                targets = Some(meta.value()?.parse()?);
440            } else if named("criteria") {
441                let expression: LitStr = meta.value()?.parse()?;
442                check_expression(&expression, names)?;
443                pieces.push(quote!(.criteria(#expression)));
444            } else if named("font") {
445                let color = tint(&meta.value()?.parse()?)?;
446                pieces.push(quote!(.font(#color)));
447            } else if named("back") {
448                let color = tint(&meta.value()?.parse()?)?;
449                pieces.push(quote!(.back(#color)));
450            } else if named("bold") {
451                pieces.push(quote!(.bold()));
452            } else if named("enabled") {
453                let on: syn::LitBool = meta.value()?.parse()?;
454                pieces.push(quote!(.enabled(#on)));
455            } else if named("visible") {
456                let on: syn::LitBool = meta.value()?.parse()?;
457                pieces.push(quote!(.visible(#on)));
458            } else if named("priority") {
459                let priority: LitInt = meta.value()?.parse()?;
460                pieces.push(quote!(.priority(#priority)));
461            } else if named("action") {
462                pieces.push(quote!(.target(::ravenrs_app::AppearanceTarget::Action)));
463            } else if named("layout") {
464                pieces.push(quote!(.target(::ravenrs_app::AppearanceTarget::Layout)));
465            } else if named("context") {
466                let context: LitStr = meta.value()?.parse()?;
467                let context = match context.value().as_str() {
468                    "list" => quote!(::ravenrs_app::AppearanceContext::List),
469                    "detail" => quote!(::ravenrs_app::AppearanceContext::Detail),
470                    _ => quote!(::ravenrs_app::AppearanceContext::View(#context.to_owned())),
471                };
472                pieces.push(quote!(.context(#context)));
473            } else {
474                return Err(meta.error(
475                    "unknown setting: use names, criteria, font, back, bold, enabled, visible, priority, \
476                     context, action or layout",
477                ));
478            }
479            Ok(())
480        })?;
481        let Some(targets) = targets else {
482            return Err(Error::new(attr.span(), "an appearance rule needs names = \"...\""));
483        };
484        let targets = split_names(&targets);
485        let id = format!("appearance{}", rules.len());
486        rules.push(quote!(::ravenrs_app::AppearanceRule::new(#id, [#(#targets),*]) #(#pieces)*));
487    }
488    Ok(rules)
489}

491fn field_tokens(ident: &Ident, ty: &syn::Type, attrs: &FieldAttrs) -> Tokens {
492    let name = ident.to_string();
493    let kind = attrs
494        .kind
495        .clone()
496        .unwrap_or_else(|| quote_spanned!(ty.span()=> <#ty as ::ravenrs_app::FieldValue>::kind()));
497    let mut info = quote!(::ravenrs_app::FieldInfo::new(#name, #kind));
498    let mut then = |piece: Tokens| info = quote!(#info #piece);
499    if attrs.key {
500        then(quote!(.key()));
501    }
502    if attrs.typed_key {
503        then(quote!(.typed_key()));
504    }
505    if let Some(caption) = &attrs.caption {
506        then(quote!(.caption(#caption)));
507    }
508    if attrs.default_property {
509        then(quote!(.default_property()));
510    }
511    if attrs.read_only {
512        then(quote!(.read_only()));
513    }
514    if attrs.no_list {
515        then(quote!(.in_list(false)));
516    }
517    if attrs.no_detail {
518        then(quote!(.in_detail(false)));
519    }
520    if attrs.lookup {
521        then(quote!(.in_lookup(true)));
522    }
523    if let Some(expression) = &attrs.calculated {
524        then(quote!(.calculated(#expression)));
525    }
526    if let Some(value) = &attrs.default {
527        then(quote!(.default_value(#value)));
528    }
529    if let Some(editor) = &attrs.editor {
530        then(quote!(.editor(#editor)));
531    }
532    if let Some(text) = &attrs.placeholder {
533        then(quote!(.placeholder(#text)));
534    }
535    if let Some(text) = &attrs.hint {
536        then(quote!(.hint(#text)));
537    }
538    if let Some(text) = &attrs.lookup_filter {
539        then(quote!(.lookup_filter(#text)));
540    }
541    if let Some(text) = &attrs.none_caption {
542        then(quote!(.none_caption(#text)));
543    }
544    if let Some(text) = &attrs.back_reference {
545        then(quote!(.back_reference(#text)));
546    }
547    if let Some(bytes) = &attrs.most_bytes {
548        then(quote!(.most_bytes(#bytes)));
549    }
550    if attrs.wide {
551        then(quote!(.wide()));
552    }
553    if attrs.autofocus {
554        then(quote!(.autofocus()));
555    }
556    if attrs.secret {
557        then(quote!(.secret()));
558    }
559    if attrs.non_cloneable {
560        then(quote!(.non_cloneable()));
561    }
562    if attrs.aggregated {
563        then(quote!(.aggregated()));
564    }
565    if let Some(text) = &attrs.group {
566        then(quote!(.group(#text)));
567    }
568    if let Some(text) = &attrs.display_format {
569        then(quote!(.display_format(#text)));
570    }
571    if let Some(text) = &attrs.edit_mask {
572        then(quote!(.edit_mask(#text)));
573    }
574    if let Some(text) = &attrs.predefined {
575        let values = split_names(text);
576        then(quote!(.predefined([#(#values),*])));
577    }
578    if let Some(text) = &attrs.criteria_of {
579        then(quote!(.criteria_of(#text)));
580    }
581    info
582}

584/// `#[variant(...)]` and `#[list_filter(...)]` on a struct, as calls on its entity.
585fn struct_choices(input: &DeriveInput, names: &[String]) -> Result<Vec<Tokens>> {
586    let mut found = Vec::new();
587    for attr in &input.attrs {
588        let is_variant = attr.path().is_ident("variant");
589        if !is_variant && !attr.path().is_ident("list_filter") {
590            continue;
591        }
592        let (mut id, mut caption, mut filter, mut columns) =
593            (None::<LitStr>, None::<LitStr>, None::<LitStr>, None::<LitStr>);
594        attr.parse_nested_meta(|meta| {
595            let slot = if meta.path.is_ident("id") {
596                &mut id
597            } else if meta.path.is_ident("caption") {
598                &mut caption
599            } else if meta.path.is_ident("filter") {
600                &mut filter
601            } else if is_variant && meta.path.is_ident("columns") {
602                &mut columns
603            } else {
604                return Err(meta.error("unknown setting: use id, caption, filter or, on a variant, columns"));
605            };
606            *slot = Some(meta.value()?.parse()?);
607            Ok(())
608        })?;
609        let (Some(id), Some(caption)) = (id, caption) else {
610            return Err(Error::new(attr.span(), "this needs an id and a caption"));
611        };
612        if let Some(text) = &filter {
613            let parsed =
614                Criteria::parse(&text.value()).map_err(|e| Error::new(text.span(), e.to_string()))?;
615            if let Some(unknown) = parsed.properties().iter().find(|name| !names.contains(name)) {
616                return Err(Error::new(text.span(), format!("there is no field {unknown}")));
617            }
618        }
619        let shown: Vec<String> = columns
620            .iter()
621            .flat_map(|text| text.value().split(',').map(|c| c.trim().to_owned()).collect::<Vec<_>>())
622            .collect();
623        if let (Some(text), Some(unknown)) = (&columns, shown.iter().find(|name| !names.contains(name))) {
624            return Err(Error::new(text.span(), format!("there is no field {unknown}")));
625        }
626        let filter =
627            filter.map(|text| if is_variant { quote!(.filter(#text)) } else { quote!(.criteria(#text)) });
628        found.push(if is_variant {
629            let shown = (!shown.is_empty()).then(|| quote!(.columns([#(#shown),*])));
630            quote!(.variant(::ravenrs_app::ViewVariant::new(#id, #caption) #filter #shown))
631        } else {
632            quote!(.filter(::ravenrs_app::ListFilter::new(#id, #caption) #filter))
633        });
634    }
635    Ok(found)
636}

638fn business_object(input: &DeriveInput) -> Result<Tokens> {
639    let name = &input.ident;
640    if !input.generics.params.is_empty() {
641        return Err(Error::new(input.generics.span(), "a business object cannot be generic"));
642    }
643    let Data::Struct(data) = &input.data else {
644        return Err(Error::new(name.span(), "a business object is a struct with named fields"));
645    };
646    let Fields::Named(fields) = &data.fields else {
647        return Err(Error::new(name.span(), "a business object is a struct with named fields"));
648    };
649    let object = object_attrs(input)?;
650    let names: Vec<String> =
651        fields.named.iter().filter_map(|f| f.ident.as_ref().map(Ident::to_string)).collect();
652    let mut infos = Vec::new();
653    let mut rules = Vec::new();
654    let mut to_values = Vec::new();
655    let mut from_values = Vec::new();
656    let mut keys = 0;
657    for field in &fields.named {
658        let Some(ident) = &field.ident else { continue };
659        let attrs = field_attrs(field)?;
660        if attrs.key || attrs.typed_key {
661            keys += 1;
662            if keys > 1 {
663                return Err(Error::new(
664                    ident.span(),
665                    "only one field can be the key, and another one already is",
666                ));
667            }
668        }
669        if let Some(expression) = &attrs.calculated {
670            check_expression(expression, &names)?;
671        }
672        infos.push(field_tokens(ident, &field.ty, &attrs));
673        rules.extend(field_rules(ident, &attrs));
674        let (text, ty) = (ident.to_string(), &field.ty);
675        to_values.push(quote_spanned!(ty.span()=>
676            (#text.to_owned(), <#ty as ::ravenrs_app::FieldValue>::to_value(&self.#ident))
677        ));
678        from_values.push(quote_spanned!(ty.span()=>
679            #ident: <#ty as ::ravenrs_app::FieldValue>::from_value(object.get(#text))
680        ));
681    }
682    if keys == 0 {
683        return Err(Error::new(
684            name.span(),
685            "a business object needs a key: mark a field with #[field(key)] or #[field(typed_key)]",
686        ));
687    }
688    rules.extend(struct_rules(input, &names)?);
689    let appearance = struct_appearance(input, &names)?;
690    let entity = name.to_string();
691    let mut info = quote!(::ravenrs_app::EntityInfo::new(#entity));
692    if let Some(caption) = &object.caption {
693        info = quote!(#info.caption(#caption));
694    }
695    if let Some(place) = &object.navigation {
696        info = quote!(#info.navigation(#place));
697    }
698    if let Some(icon) = &object.icon {
699        info = quote!(#info.icon(#icon));
700    }
701    if let Some(text) = &object.list_criteria {
702        info = quote!(#info.list_criteria(#text));
703    }
704    if object.list_editable {
705        info = quote!(#info.list_editable());
706    }
707    if let Some(access) = &object.data_access {
708        let access = match access.value().as_str() {
709            "server" => quote!(::ravenrs_app::DataAccess::Server),
710            "data_view" => quote!(::ravenrs_app::DataAccess::DataView),
711            _ => quote!(::ravenrs_app::DataAccess::Client),
712        };
713        info = quote!(#info.data_access(#access));
714    }
715    for name in &object.form_actions {
716        info = quote!(#info.form_actions(#name));
717    }
718    if let Some(columns) = &object.form_columns {
719        info = quote!(#info.form_columns(#columns));
720    }
721    if object.cloneable {
722        info = quote!(#info.cloneable());
723    }
724    if object.transient {
725        info = quote!(#info.transient());
726    }
727    if object.checks_on_leave {
728        info = quote!(#info.checks_on_leave());
729    }
730    let choices = struct_choices(input, &names)?;
731    Ok(quote! {
732        impl ::ravenrs_app::BusinessObject for #name {
733            fn entity_info() -> ::ravenrs_app::EntityInfo {
734                #info #(.field(#infos))* #(.rule(#rules))* #(.appearance(#appearance))* #(#choices)*
735            }
736
737            fn to_object(&self) -> ::ravenrs_app::Object {
738                ::ravenrs_app::Object::from_values(
739                    #entity,
740                    ::std::collections::BTreeMap::from([#(#to_values),*]),
741                )
742            }
743
744            fn from_object(object: &::ravenrs_app::Object) -> Self {
745                Self { #(#from_values),* }
746            }
747        }
748    })
749}

751fn business_enum(input: &DeriveInput) -> Result<Tokens> {
752    let name = &input.ident;
753    let Data::Enum(data) = &input.data else {
754        return Err(Error::new(name.span(), "a business enum is an enum with no data"));
755    };
756    let Some(first) = data.variants.first() else {
757        return Err(Error::new(name.span(), "a business enum needs at least one option"));
758    };
759    let first = &first.ident;
760    let mut options = Vec::new();
761    let mut codes = Vec::new();
762    let mut parses = Vec::new();
763    for variant in &data.variants {
764        if !matches!(variant.fields, Fields::Unit) {
765            return Err(Error::new(variant.ident.span(), "an option of a business enum cannot hold data"));
766        }
767        let ident = &variant.ident;
768        let (mut caption, mut code) = (ident.to_string(), ident.to_string());
769        for attr in variant.attrs.iter().filter(|a| a.path().is_ident("option")) {
770            attr.parse_nested_meta(|meta| {
771                let value: LitStr = if meta.path.is_ident("caption") || meta.path.is_ident("code") {
772                    meta.value()?.parse()?
773                } else {
774                    return Err(meta.error("unknown setting: use caption or code"));
775                };
776                if meta.path.is_ident("caption") {
777                    caption = value.value();
778                } else {
779                    code = value.value();
780                }
781                Ok(())
782            })?;
783        }
784        options.push(quote!(::ravenrs_app::EnumOption::new(#code, #caption)));
785        codes.push(quote!(Self::#ident => #code));
786        parses.push(quote!(#code => Self::#ident));
787    }
788    Ok(quote! {
789        impl ::ravenrs_app::FieldValue for #name {
790            fn kind() -> ::ravenrs_app::FieldKind {
791                ::ravenrs_app::FieldKind::Enum(::std::vec![#(#options),*])
792            }
793
794            fn to_value(&self) -> ::ravenrs_app::Value {
795                ::ravenrs_app::Value::Text(match self { #(#codes),* }.to_owned())
796            }
797
798            fn from_value(value: ::ravenrs_app::Value) -> Self {
799                match value {
800                    ::ravenrs_app::Value::Text(code) => match code.as_str() {
801                        #(#parses,)*
802                        _ => Self::#first,
803                    },
804                    _ => Self::#first,
805                }
806            }
807        }
808    })
