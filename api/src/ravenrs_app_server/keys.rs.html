24fn hash_of(secret: &str) -> String {
25    Sha256::digest(secret.as_bytes()).iter().map(|byte| format!("{byte:02x}")).collect()
26}

28/// Compares without stopping at the first difference.
29fn same(a: &str, b: &str) -> bool {
30    a.len() == b.len() && a.bytes().zip(b.bytes()).fold(0_u8, |difference, (x, y)| difference | (x ^ y)) == 0
31}

33fn new_secret() -> String {
34    format!("{}{}", uuid::Uuid::new_v4().simple(), uuid::Uuid::new_v4().simple())
35}

37fn public(key: &Object) -> Json {
38    let text = |name: &str| match key.get(name) {
39        Value::Text(text) => Json::String(text),
40        Value::Date(date) => Json::String(date.to_string()),
41        Value::Bool(on) => Json::Bool(on),
42        _ => Json::Null,
43    };
44    json!({
45        "id": text("id"),
46        "name": text("name"),
47        "user_id": ravenrs_app::value_to_json(&key.get("user_id")),
48        "roles": text("roles"),
49        "active": text("active"),
50        "expires": text("expires"),
51    })
52}

59    pub(crate) fn session_of_key(&self, token: &str) -> Result<Who, Failure> {
60        let parts = token.strip_prefix(PREFIX).and_then(|rest| rest.split_once('_'));
61        let (id, secret) = parts.ok_or_else(Failure::unauthorized)?;
62        let space = (self.space)();
63        let key = space
64            .find(SECURITY_API_KEY, &Value::Text(id.to_owned()))
65            .map_err(stored)?
66            .ok_or_else(Failure::unauthorized)?;
67        let Value::Text(hash) = key.get("secret_hash") else { return Err(Failure::unauthorized()) };
68        if !same(&hash, &hash_of(secret)) || key.get("active") != Value::Bool(true) {
69            return Err(Failure::unauthorized());
70        }
71        match key.get("expires") {
72            Value::Null => {}
73            Value::Date(expires) => {
74                let now = local_now();
75                if (expires.days, expires.seconds) <= (now.days, now.seconds) {
76                    return Err(Failure::unauthorized());
77                }
78            }
79            // A date nobody can read is not a key that lasts forever.
80            _ => return Err(Failure::unauthorized()),
81        }
82        // Who the user is now: losing a role or being turned off counts at once for their keys.
83        let owner = self
84            .directory
85            .identity_by_key(&key.get("user_id"))
86            .map_err(stored)?
87            .ok_or_else(Failure::unauthorized)?;
88        let identity = narrowed(owner, &key, self.merge);
89        let name = format!("{} ({})", identity.name, text_of(&key, "name"));
90        let mut context =
91            SecurityContext::new(identity, self.directory.clone(), self.entities.clone(), self.merge);
92        let trail = self.audit.clone().map(|settings| {
93            let trail = Rc::new(AuditTrail::new(settings, move || name.clone()));
94            trail.write_with(self.space.clone());
95            trail
96        });
97        if let Some(trail) = &trail {
98            context = context.with_audit_trail(trail.clone());
99        }
100        Ok((Rc::new(context), trail))
101    }

107    /// `name`, and optionally `user`, `roles` and `expires_in_days`. The key shows here and never again.
108    pub(crate) fn make_key(&self, request: &ApiRequest, who: &Who) -> Result<(u16, Json), Failure> {
109        self.only_administrators(who)?;
110        let body = body_of(request)?;
111        let name = body
112            .get("name")
113            .and_then(Json::as_str)
114            .filter(|name| !name.trim().is_empty())
115            .ok_or_else(|| Failure::bad_request("A key has a name."))?;
116        let owner = match body.get("user").and_then(Json::as_str) {
117            Some(user) => self
118                .directory
119                .identity_of(user)
120                .map_err(stored)?
121                .ok_or_else(|| Failure::bad_request("There is no such active user."))?,
122            None => who.0.identity(),
123        };
124        let roles = roles_asked(&body, &owner, self.merge)?;
125        let mut space = (self.space)();
126        let mut key = space.new_object(SECURITY_API_KEY).map_err(stored)?;
127        let id = uuid::Uuid::new_v4().simple().to_string();
128        let secret = new_secret();
129        key.set("id", id.clone());
130        key.set("name", name);
131        key.set("user_id", owner.key.clone());
132        key.set("roles", roles);
133        key.set("active", true);
134        key.set("secret_hash", hash_of(&secret));
135        if let Some(days) = body.get("expires_in_days") {
136            let days = days
137                .as_i64()
138                .filter(|days| (1..=MOST_DAYS).contains(days))
139                .ok_or_else(|| Failure::bad_request("expires_in_days goes from 1 to ten years."))?;
140            let now = local_now();
141            key.set(
142                "expires",
143                Value::Date(ravenrs_criteria::DateTime { days: now.days + days, seconds: now.seconds }),
144            );
145        }
146        space.insert(key.clone()).map_err(stored)?;
147        space.commit().map_err(stored)?;
148        note(who, &key, "API key made");
149        let mut answer = public(&key);
150        if let Some(object) = answer.as_object_mut() {
151            object.insert("key".to_owned(), json!(format!("{PREFIX}{id}_{secret}")));
152        }
153        Ok((201, answer))
154    }

156    pub(crate) fn list_keys(&self, who: &Who) -> Result<(u16, Json), Failure> {
157        self.only_administrators(who)?;
158        let found = (self.space)().query(SECURITY_API_KEY, &Query::all()).map_err(stored)?;
159        Ok((200, json!({ "items": found.iter().map(public).collect::<Vec<_>>() })))
160    }

162    pub(crate) fn rotate_key(&self, id: &str, who: &Who) -> Result<(u16, Json), Failure> {
163        self.only_administrators(who)?;
164        let mut space = (self.space)();
165        let mut key = space
166            .find(SECURITY_API_KEY, &Value::Text(id.to_owned()))
167            .map_err(stored)?
168            .ok_or_else(Failure::not_found)?;
169        let secret = new_secret();
170        key.set("secret_hash", hash_of(&secret));
171        space.save(key.clone()).map_err(stored)?;
172        space.commit().map_err(stored)?;
173        note(who, &key, "API key rotated");
174        let mut answer = public(&key);
175        if let Some(object) = answer.as_object_mut() {
176            object.insert("key".to_owned(), json!(format!("{PREFIX}{id}_{secret}")));
177        }
178        Ok((200, answer))
179    }

181    pub(crate) fn revoke_key(&self, id: &str, who: &Who) -> Result<(u16, Json), Failure> {
182        self.only_administrators(who)?;
183        let mut space = (self.space)();
184        let key = space
185            .find(SECURITY_API_KEY, &Value::Text(id.to_owned()))
186            .map_err(stored)?
187            .ok_or_else(Failure::not_found)?;
188        space.delete(SECURITY_API_KEY, &key.get("id")).map_err(stored)?;
189        space.commit().map_err(stored)?;
190        note(who, &key, "API key revoked");
191        Ok((200, json!({})))
192    }

202/// The roles of the user that the key may use: all of them, if it names none. When every role has
203/// to allow something, leaving one out would take away what it denies, so a key keeps them all.
204fn narrowed(mut owner: Identity, key: &Object, merge: RoleMerge) -> Identity {
205    let wanted = text_of(key, "roles");
206    if wanted.is_empty() || merge != RoleMerge::default() {
207        return owner;
208    }
209    let named: Vec<&str> = wanted.split(',').map(str::trim).filter(|name| !name.is_empty()).collect();
210    owner.roles.retain(|role| named.contains(&role.name()));
211    owner
212}
213
214fn roles_asked(body: &Map<String, Json>, owner: &Identity, merge: RoleMerge) -> Result<String, Failure> {
215    let Some(asked) = body.get("roles") else { return Ok(String::new()) };
216    if merge != RoleMerge::default() {
217        return Err(Failure::bad_request(
218            "Here every role has to allow something, so a key cannot leave any of them out.",
219        ));
220    }
221    let names = asked.as_array().ok_or_else(|| Failure::bad_request("roles is a list of names."))?;
222    let mut kept = Vec::with_capacity(names.len());
223    for name in names {
224        let name = name.as_str().ok_or_else(|| Failure::bad_request("roles is a list of names."))?;
225        // They are kept separated by commas, so one with a comma would read as others.
226        if name.contains(',') || !owner.roles.iter().any(|role| role.name() == name) {
227            return Err(Failure::bad_request(format!("The user does not have the role {name}.")));
228        }
229        kept.push(name);
230    }
231    if kept.is_empty() {
232        return Err(Failure::bad_request("A key with roles names at least one."));
233    }
234    Ok(kept.join(","))
235}

237fn note(who: &Who, key: &Object, what: &str) {
238    if let Some(trail) = &who.1 {
239        trail.custom(SECURITY_API_KEY, &key.get("id"), "secret_hash", what);
240    }
