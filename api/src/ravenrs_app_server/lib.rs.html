1//! A web API over an application's model: the same entities, rules, permissions, states and audit
2//! trail as its screens, for clients that do not use RavenRS.
3//!
4//! [`ApiService`] answers requests with no HTTP around it, and [`serve`] and [`router`] put HTTP
5//! around one. Every write goes through a [`SecuredObjectSpace`], so what a person cannot do on the
6//! screens they cannot do here, and every read leaves as that person can see it.
7//!
8//! Tokens live in memory: restarting the service asks everyone to sign in again. It speaks plain
9//! HTTP, so anything beyond the machine it runs on belongs behind something that does TLS.

61/// What came in, with no HTTP left in it.
62#[derive(Clone, Debug, Default)]
63#[non_exhaustive]
64pub struct ApiRequest {
65    /// `GET`, `POST` and so on.
66    pub method: String,
67    /// The parts of the path.
68    pub path: Vec<String>,
69    /// The query string, decoded.
70    pub query: BTreeMap<String, String>,
71    /// The token presented.
72    pub bearer: Option<String>,
73    /// The address it came from, as far as anyone can tell.
74    pub client: Option<String>,
75    /// The `X-RavenRS-*` headers, with lowercase names, such as a webhook's signature.
76    pub headers: BTreeMap<String, String>,
77    /// The body.
78    pub body: Vec<u8>,
79}

81impl ApiRequest {
82    /// A request with this method and path, such as `/api/data/Order`.
83    pub fn new(method: &str, path: &str) -> Self {
84        Self {
85            method: method.to_owned(),
86            path: path.split('/').filter(|part| !part.is_empty()).map(str::to_owned).collect(),
87            ..Self::default()
88        }
89    }
90
91    /// With this token.
92    pub fn bearer(mut self, token: impl Into<String>) -> Self {
93        self.bearer = Some(token.into());
94        self
95    }
96
97    /// From this address.
98    pub fn client(mut self, client: impl Into<String>) -> Self {
99        self.client = Some(client.into());
100        self
101    }
102
103    /// With this header; its name is kept in lowercase.
104    pub fn header(mut self, name: &str, value: impl Into<String>) -> Self {
105        self.headers.insert(name.to_ascii_lowercase(), value.into());
106        self
107    }
108
109    /// With this body, as it is.
110    pub fn bytes(mut self, body: impl Into<Vec<u8>>) -> Self {
111        self.body = body.into();
112        self
113    }
114
115    /// With this body.
116    pub fn json(mut self, body: &Json) -> Self {
117        self.body = body.to_string().into_bytes();
118        self
119    }
120
121    /// With this in the query string.
122    pub fn query(mut self, name: &str, value: &str) -> Self {
123        self.query.insert(name.to_owned(), value.to_owned());
124        self
125    }
126}

138/// What a request's answer is when it does not work out.
139#[derive(Debug)]
140pub struct Failure {
141    status: u16,
142    code: &'static str,
143    message: String,
144    violations: Option<Json>,
145    /// Which operation of a batch it was.
146    operation: Option<usize>,
147}
148
149impl Failure {
150    fn new(status: u16, code: &'static str, message: impl Into<String>) -> Self {
151        Self { status, code, message: message.into(), violations: None, operation: None }
152    }
153
154    /// What was asked does not fit.
155    pub fn bad_request(message: impl Into<String>) -> Self {
156        Self::new(400, "bad_request", message)
157    }
158
159    /// There is nothing there. It is also what someone who cannot see a record is told.
160    pub fn not_found() -> Self {
161        Self::new(404, "not_found", "There is no such record.")
162    }
163
164    /// This person cannot do this.
165    pub fn denied() -> Self {
166        Self::new(403, "denied", "You are not allowed to do this.")
167    }
168
169    fn unauthorized() -> Self {
170        Self::new(401, "unauthorized", "A valid token is required.")
171    }
172
173    fn into_response(self) -> ApiResponse {
174        let mut body = json!({ "error": self.code, "message": self.message });
175        if let (Some(violations), Some(object)) = (self.violations, body.as_object_mut()) {
176            object.insert("violations".to_owned(), violations);
177        }
178        if let (Some(operation), Some(object)) = (self.operation, body.as_object_mut()) {
179            object.insert("operation".to_owned(), json!(operation));
180        }
181        ApiResponse { status: self.status, body }
182    }
183}

185impl From<SpaceError> for Failure {
186    fn from(error: SpaceError) -> Self {
187        match error {
188            SpaceError::UnknownEntity(_) => Self::not_found(),
189            SpaceError::BadQuery(why) => Self::bad_request(why),
190            SpaceError::MissingKey(why) => Self::bad_request(why),
191            SpaceError::Denied(_) => Self::denied(),
192            SpaceError::Refused(why) => Self::new(422, "refused", why),
193            error @ SpaceError::DuplicateKey { .. } => Self::new(409, "conflict", error.to_string()),
194            SpaceError::Conflict(why) => Self::new(409, "conflict", why),
195            // What went wrong in storage is for the log, and not for whoever is asking.
196            other => {
197                tracing::error!(error = %other, "storage failed");
198                Self::new(500, "internal", "Something went wrong on the server.")
199            }
200        }
201    }
202}

271fn records_in(value: &Json) -> usize {
272    match value {
273        Json::Object(object) => 1 + object.values().map(records_in).sum::<usize>(),
274        Json::Array(items) => items.iter().map(records_in).sum(),
275        _ => 0,
276    }
277}

279fn body_of(request: &ApiRequest) -> Result<Map<String, Json>, Failure> {
280    let value: Json = serde_json::from_slice(&request.body)
281        .map_err(|_| Failure::bad_request("The body has to be a JSON object."))?;
282    if records_in(&value) > MOST_RECORDS {
283        return Err(Failure::bad_request("The body holds too many records."));
284    }
285    match value {
286        Json::Object(object) => Ok(object),
287        _ => Err(Failure::bad_request("The body has to be a JSON object.")),
288    }
289}

414    pub fn handle(&mut self, request: ApiRequest) -> ApiResponse {

417            Err(failure) => failure.into_response(),

421    fn route(&mut self, request: &ApiRequest) -> Result<(u16, Json), Failure> {
422        let path: Vec<&str> = request.path.iter().map(String::as_str).collect();
423        let method = request.method.as_str();
424        match (method, path.as_slice()) {
425            ("POST", ["api", "auth", "token"]) => return self.sign_in(request),
426            ("POST", ["api", "hooks", source]) => {
427                let hook = self.hooks.get(*source).ok_or_else(Failure::not_found)?;
428                return hooks::receive(hook, source, request, (self.space)());
429            }
430            ("POST", ["api", "auth", "revoke"]) => {
431                if let Some(token) = &request.bearer {
432                    self.sessions.remove(token);
433                }
434                return Ok((200, json!({})));
435            }
436            _ => {}
437        }
438        let session = self.session_of(request)?;
439        let by_key = request.bearer.as_deref().is_some_and(keys::is_key);
440        if by_key && path.get(1) == Some(&"auth") {
441            // A key does not get to change passwords, or to make itself more keys.
442            return Err(Failure::denied());
443        }
444        if path.as_slice() == ["api", "auth", "password"] && method == "POST" {
445            return self.change_password(request, &session);
446        }
447        if session.0.identity().must_change_password {
448            return Err(Failure::new(
449                403,
450                "password_change_required",
451                "This person has to choose a new password first, at /api/auth/password.",
452            ));
453        }
454        let mut space = self.space_for(&session);
455        match (method, path.as_slice()) {
456            ("GET", ["api", "openapi.json"]) => Ok((200, self.description(&session))),
457            ("GET", ["api", "me"]) => {
458                let identity = session.0.identity();
459                let roles: Vec<&str> = identity.roles.iter().map(Role::name).collect();
460                Ok((200, json!({ "name": identity.name, "roles": roles })))
461            }
462            ("POST", ["api", "auth", "keys"]) => self.make_key(request, &session),
463            ("GET", ["api", "auth", "keys"]) => self.list_keys(&session),
464            ("POST", ["api", "auth", "keys", id, "rotate"]) => self.rotate_key(id, &session),
465            ("DELETE", ["api", "auth", "keys", id]) => self.revoke_key(id, &session),
466            ("GET" | "PUT" | "DELETE", ["api", "settings", name]) => {
467                self.reach_setting(method, name, request, &session)
468            }
469            ("GET", ["api", "data", entity]) => self.list(&*space, entity, request),
470            ("GET", ["api", "data", entity, key]) => {
471                let info = self.entity(entity)?;
472                let key = key_from_path(&info, key)?;
473                let found = space.find(entity, &key)?.ok_or_else(Failure::not_found)?;
474                let mut record = object_to_json(&info, &found);
475                self.expand(&*space, &info, &key, &mut record, request)?;
476                Ok((200, Json::Object(record)))
477            }
478            ("POST", ["api", "data", entity]) => {
479                let answer = self.create(&mut *space, entity, &body_of(request)?, 0)?;
480                space.commit()?;
481                Ok((201, answer))
482            }
483            ("PATCH", ["api", "data", entity, key]) => {
484                let info = self.entity(entity)?;
485                let key = key_from_path(&info, key)?;
486                let answer = self.change(&mut *space, &info, &key, &body_of(request)?, 0)?;
487                space.commit()?;
488                Ok((200, answer))
489            }
490            ("DELETE", ["api", "data", entity, key]) => {
491                let info = self.entity(entity)?;
492                let key = key_from_path(&info, key)?;
493                self.remove(&mut *space, &info, &key)?;
494                space.commit()?;
495                Ok((200, json!({})))
496            }
497            ("POST", ["api", "validate", entity]) => self.check(entity, &body_of(request)?),
498            ("POST", ["api", "batch"]) => self.batch(&mut *space, &body_of(request)?),
499            ("POST", ["api", "run", name]) => {
500                let endpoint = self.endpoints.get(*name).cloned().ok_or_else(Failure::not_found)?;
501                let input: Json = if request.body.is_empty() {
502                    Json::Null
503                } else {
504                    serde_json::from_slice(&request.body)
505                        .map_err(|_| Failure::bad_request("The body has to be JSON."))?
506                };
507                let held = std::cell::RefCell::new(space);
508                let identity = session.0.identity();
509                let context = EndpointContext { name: &identity.name, space: &held };
510                let answer = endpoint(&context, &input)?;
511                Ok((200, answer))
512            }
513            _ => Err(Failure::new(404, "not_found", "There is no such route.")),
514        }
515    }

517    /// Counts a try that costs the hash of a password, or refuses it. Only what gets this far counts,
518    /// so a body that means nothing does not use anyone's turn.
519    fn count_try(&mut self, request: &ApiRequest) -> Result<(), Failure> {
520        let now = Instant::now();
521        if now.duration_since(self.tries_since) >= MINUTE {
522            self.tries.clear();
523            self.tries_since = now;
524        }
525        let all: u32 = self.tries.values().sum();
526        let client = request.client.clone().unwrap_or_default();
527        let mine = self.tries.entry(client).or_insert(0);
528        if *mine >= self.sign_ins_a_minute || all >= self.sign_ins_a_minute.saturating_mul(ALL_ADDRESSES) {
529            return Err(Failure::new(429, "too_many", "Too many tries. Try again soon."));
530        }
531        *mine += 1;
532        Ok(())
533    }

535    fn sign_in(&mut self, request: &ApiRequest) -> Result<(u16, Json), Failure> {
536        let now = Instant::now();
537        let body = body_of(request)?;
538        let text = |name: &str| body.get(name).and_then(Json::as_str).unwrap_or("").to_owned();
539        let rejected = || Failure::new(401, "unauthorized", "The name or the password is not right.");
540        if text("password").len() > MOST_PASSWORD {
541            return Err(rejected());
542        }
543        self.count_try(request)?;
544        self.sessions.retain(|_, session| session.expires > now);
545        let identity = self
546            .authenticator
547            .authenticate(&Credentials::new(text("name"), text("password")))
548            .map_err(|error| match error {
549                AuthError::Rejected => rejected(),
550                other => {
551                    tracing::error!(error = %other, "signing in failed");
552                    Failure::new(500, "internal", "Something went wrong on the server.")
553                }
554            })?;
555        let must_change_password = identity.must_change_password;
556        let author = identity.name.clone();
557        let mut context =
558            SecurityContext::new(identity, self.directory.clone(), self.entities.clone(), self.merge);
559        let trail = self.audit.clone().map(|settings| {
560            let trail = Rc::new(AuditTrail::new(settings, move || author.clone()));
561            trail.write_with(self.space.clone());
562            trail
563        });
564        if let Some(trail) = &trail {
565            context = context.with_audit_trail(trail.clone());
566        }
567        let token = token();
568        self.sessions.insert(
569            token.clone(),
570            Session { context: Rc::new(context), trail, expires: now + self.lifetime },
571        );
572        Ok((
573            200,
574            json!({
575                "token": token,
576                "expires_in": self.lifetime.as_secs(),
577                "must_change_password": must_change_password,
578            }),
579        ))
580    }

582    fn session_of(
583        &mut self,
584        request: &ApiRequest,
585    ) -> Result<(Rc<SecurityContext>, Option<Rc<AuditTrail>>), Failure> {
586        let token = request.bearer.as_deref().ok_or_else(Failure::unauthorized)?;
587        if keys::is_key(token) {
588            return self.session_of_key(token);
589        }
590        let now = Instant::now();
591        if now.duration_since(self.swept) >= MINUTE {
592            self.sessions.retain(|_, session| session.expires > now);
593            self.swept = now;
594        }
595        let session = self.sessions.get(token).ok_or_else(Failure::unauthorized)?;
596        if session.expires <= now {
597            self.sessions.remove(token);
598            return Err(Failure::unauthorized());
599        }
600        // Who they are is read each time, so a role lost or a user turned off counts now. Someone
601        // who cannot sign in any more has no session either, whatever a route goes on to check.
602        let (context, trail) = (session.context.clone(), session.trail.clone());
603        let key = context.identity().key;
604        if self.directory.identity_by_key(&key).map_err(Failure::from)?.is_none() {
605            self.sessions.remove(token);
606            return Err(Failure::unauthorized());
607        }
608        context.reload().map_err(Failure::from)?;
609        Ok((context, trail))
610    }

612    fn change_password(
613        &mut self,
614        request: &ApiRequest,
615        session: &(Rc<SecurityContext>, Option<Rc<AuditTrail>>),
616    ) -> Result<(u16, Json), Failure> {
617        let body = body_of(request)?;
618        let text = |name: &str| body.get(name).and_then(Json::as_str).unwrap_or("").to_owned();
619        if text("current").len() > MOST_PASSWORD {
620            return Err(Failure::new(403, "denied", PasswordError::WrongCurrent.to_string()));
621        }
622        self.count_try(request)?;
623        session.0.change_own_password(&text("current"), &text("new")).map_err(|error| match error {
624            PasswordError::TooShort | PasswordError::SameAsCurrent => {
625                Failure::new(422, "refused", error.to_string())
626            }
627            PasswordError::WrongCurrent => Failure::new(403, "denied", error.to_string()),
628            PasswordError::NotAllowed | PasswordError::NoSuchUser => Failure::denied(),
629            other => {
630                tracing::error!(error = %other, "changing a password failed");
631                Failure::new(500, "internal", "Something went wrong on the server.")
632            }
633        })?;
634        // The new password ends the other sessions the old one opened. Keys are made apart, and stay.
635        let key = session.0.identity().key;
636        let kept = request.bearer.clone().unwrap_or_default();
637        self.sessions.retain(|token, other| *token == kept || other.context.identity().key != key);
638        Ok((200, json!({})))
639    }

661    fn description(&self, session: &(Rc<SecurityContext>, Option<Rc<AuditTrail>>)) -> Json {
662        // Only what this person can reach is any of their business.
663        let reachable: Vec<EntityInfo> = self

679    fn reach_setting(
680        &self,
681        method: &str,
682        name: &str,
683        request: &ApiRequest,
684        session: &(Rc<SecurityContext>, Option<Rc<AuditTrail>>),
685    ) -> Result<(u16, Json), Failure> {
686        let (Some(access), Some(store)) = (self.declared.get(name), &self.settings) else {
687            return Err(Failure::not_found());
688        };
689        // A key stands for a system, and has no preferences of a person to reach. One with no role
690        // left reaches nothing.
691        let by_key = request.bearer.as_deref().is_some_and(keys::is_key);
692        if session.0.identity().roles.is_empty() || (by_key && *access == SettingAccess::PerPerson) {
693            return Err(Failure::denied());
694        }
695        let scope = match access {
696            SettingAccess::PerPerson => {
697                let key = session.0.identity().key;
698                SettingScope::User(ravenrs_app::key_text(&key).ok_or_else(Failure::denied)?)
699            }
700            SettingAccess::Shared => {
701                if method != "GET" && !session.0.with(ravenrs_app::Permissions::is_administrator) {
702                    return Err(Failure::denied());
703                }
704                SettingScope::Global
705            }
706        };
707        let failed = |why: String| {
708            tracing::error!(error = %why, "a setting failed");
709            Failure::new(500, "internal", "Something went wrong on the server.")
710        };
711        match method {
712            "GET" => {
713                let held = store.get(&scope, name).map_err(failed)?;
714                let value =
715                    held.and_then(|text| serde_json::from_str::<Json>(&text).ok()).unwrap_or(Json::Null);
716                Ok((200, json!({ "value": value })))
717            }
718            "PUT" => {
719                if request.body.len() > MOST_SETTING {
720                    return Err(Failure::bad_request("The setting is too large."));
721                }
722                let value = body_of(request)?.remove("value").unwrap_or(Json::Null);
723                store.set(&scope, name, &value.to_string()).map_err(failed)?;
724                Ok((200, json!({})))
725            }
726            _ => {
727                store.remove(&scope, name).map_err(failed)?;
728                Ok((200, json!({})))
729            }
730        }
731    }

733    fn query_of(&self, info: &EntityInfo, request: &ApiRequest) -> Result<Query, Failure> {
734        let mut query = Query::all();
735        if let Some(filter) = request.query.get("filter") {
736            if filter.len() > MOST_FILTER {
737                return Err(Failure::bad_request("The filter is too long."));
738            }
739            query.filter = Some(
740                Criteria::parse(filter)
741                    .map_err(|error| Failure::bad_request(format!("The filter: {error}")))?,
742            );
743        }
744        if let Some(order) = request.query.get("order") {
745            for part in order.split(',').map(str::trim).filter(|part| !part.is_empty()) {
746                let (name, ascending) = part.strip_prefix('-').map_or((part, true), |name| (name, false));
747                let known = info.field_named(name).is_some_and(exposed);
748                if !known {
749                    return Err(Failure::bad_request(format!("{} has no field {name}", info.name)));
750                }
751                query.sort.push((name.to_owned(), ascending));
752            }
753        }
754        if let Some(filter) = &query.filter
755            && filter.properties().iter().any(|path| {
756                let head = path.split('.').next().unwrap_or(path);
757                info.field_named(head).is_some_and(|field| field.secret)
758            })
759        {
760            return Err(Failure::bad_request("The filter asks about a field that cannot be read."));
761        }
762        query.offset = number_in(request, "skip")?.unwrap_or(0);
763        let most = if request.query.contains_key("expand") { MOST_TOP_EXPANDED } else { MOST_TOP };
764        query.limit = Some(number_in(request, "top")?.unwrap_or(DEFAULT_TOP).min(most));
765        Ok(query)
766    }

768    fn list(
769        &self,
770        space: &dyn ObjectSpace,
771        entity: &str,
772        request: &ApiRequest,
773    ) -> Result<(u16, Json), Failure> {
774        let info = self.entity(entity)?;
775        let query = self.query_of(&info, request)?;
776        let found = space.query(entity, &query)?;
777        let key_name = info.key_field().map(|field| field.name.clone());
778        let mut items = Vec::with_capacity(found.len());
779        for object in &found {
780            let mut record = object_to_json(&info, object);
781            if let Some(name) = &key_name {
782                self.expand(space, &info, &object.get(name), &mut record, request)?;
783            }
784            items.push(Json::Object(record));
785        }
786        let mut answer = json!({ "items": items });
787        if request.query.get("count").is_some_and(|value| value == "true")
788            && let Some(object) = answer.as_object_mut()
789        {
790            let all = Query { filter: query.filter, ..Query::all() };
791            object.insert("count".to_owned(), json!(space.query(entity, &all)?.len()));
792        }
793        Ok((200, answer))
794    }

804    fn expand(
805        &self,
806        space: &dyn ObjectSpace,
807        info: &EntityInfo,
808        key: &Value,
809        record: &mut Map<String, Json>,
810        request: &ApiRequest,
811    ) -> Result<(), Failure> {
812        let Some(asked) = request.query.get("expand") else { return Ok(()) };
813        let names: BTreeSet<&str> = asked.split(',').map(str::trim).filter(|name| !name.is_empty()).collect();
814        if names.len() > MOST_EXPANDED {
815            return Err(Failure::bad_request("It asks to expand too many collections."));
816        }
817        for name in names {
818            let field = info
819                .field_named(name)
820                .filter(|field| matches!(field.kind, FieldKind::Collection(_)))
821                .ok_or_else(|| Failure::bad_request(format!("{} has no collection {name}", info.name)))?;
822            let (child, back) = self
823                .back_reference(info, field)
824                .filter(|(child, _)| !self.hidden.contains(&child.name))
825                .ok_or_else(|| Failure::bad_request(format!("{name} cannot be expanded")))?;
826            let query = Query {
827                filter: Some(Criteria::property(&back).equals(Criteria::value(key.clone()))),
828                limit: Some(MOST_TOP),
829                ..Query::all()
830            };
831            let items: Vec<Json> = space
832                .query(&child.name, &query)?
833                .iter()
834                .map(|object| Json::Object(object_to_json(&child, object)))
835                .collect();
836            record.insert(name.to_owned(), Json::Array(items));
837        }
838        Ok(())
839    }

841    /// The rules look at the record whole, as stored with what was sent over it, and at all the
842    /// others: a required field this person cannot see is still there, and only a whole record can
843    /// tell. What answers has been through the permissions by the time it leaves.
844    fn refuse_broken_rules(&self, info: &EntityInfo, object: &Object, context: &str) -> Result<(), Failure> {
845        let raw = (self.space)();
846        let broken: Vec<Json> = validate(info, object, context, &*raw, &self.words)
847            .iter()
848            .filter(|violation| violation.is_error())
849            .map(|violation| json!({ "field": violation.field, "message": violation.message }))
850            .collect();
851        if broken.is_empty() {
852            return Ok(());
853        }
854        let mut failure = Failure::new(422, "invalid", "The rules of the entity refuse this.");
855        failure.violations = Some(Json::Array(broken));
856        Err(failure)
857    }

859    /// A reference has to be to a record this person can see, of the kind of key it has.
860    fn refuse_broken_references(
861        &self,
862        space: &dyn ObjectSpace,
863        info: &EntityInfo,
864        object: &Object,
865        sent: &Map<String, Json>,
866    ) -> Result<(), Failure> {
867        for name in sent.keys() {
868            let Some(field) = info.field_named(name) else { continue };
869            let FieldKind::Reference(target) = &field.kind else { continue };
870            let value = object.get(name);
871            if value != Value::Null && space.find(target, &value)?.is_none() {
872                return Err(Failure::bad_request(format!("{name} refers to a record that is not there")));
873            }
874        }
875        Ok(())
876    }

878    /// What goes back after a write, as this person can see it.
879    fn seen(
880        &self,
881        space: &dyn ObjectSpace,
882        info: &EntityInfo,
883        key: &Value,
884        object: &Object,
885    ) -> Result<Json, Failure> {
886        let found = space.find(&info.name, key)?;
887        let mut record = object_to_json(info, found.as_ref().unwrap_or(object));
888        if found.is_none() {
889            // Created where they cannot see it: they get to know its key, and nothing else.
890            record.retain(|name, _| info.field_named(name).is_some_and(|field| field.is_key));
891        }
892        Ok(Json::Object(record))
893    }

895    fn create(
896        &self,
897        space: &mut dyn ObjectSpace,
898        entity: &str,
899        sent: &Map<String, Json>,
900        depth: usize,
901    ) -> Result<Json, Failure> {
902        let info = self.entity(entity)?;
903        let mut object = space.new_object(entity)?;
904        apply(&info, &mut object, sent, true)?;
905        info.compute(&mut object, &ravenrs_app::environment());
906        self.refuse_broken_references(&*space, &info, &object, sent)?;
907        self.refuse_broken_rules(&info, &object, CONTEXT_SAVE)?;
908        let key = info.key_field().map(|field| object.get(&field.name)).unwrap_or_default();
909        space.insert(object.clone())?;
910        self.nested(space, &info, &key, sent, depth)?;
911        self.seen(&*space, &info, &key, &object)
912    }

914    fn change(
915        &self,
916        space: &mut dyn ObjectSpace,
917        info: &EntityInfo,
918        key: &Value,
919        sent: &Map<String, Json>,
920        depth: usize,
921    ) -> Result<Json, Failure> {
922        let mut object = space.find(&info.name, key)?.ok_or_else(Failure::not_found)?;
923        apply(info, &mut object, sent, false)?;
924        let mut whole = (self.space)().find(&info.name, key)?.unwrap_or_else(|| object.clone());
925        apply(info, &mut whole, sent, false)
926            .or_else(|failure| if failure.status == 403 { Ok(()) } else { Err(failure) })?;
927        info.compute(&mut whole, &ravenrs_app::environment());
928        self.refuse_broken_references(&*space, info, &object, sent)?;
929        self.refuse_broken_rules(info, &whole, CONTEXT_SAVE)?;
930        space.save(object.clone())?;
931        self.nested(space, info, key, sent, depth)?;
932        self.seen(&*space, info, key, &object)
933    }

935    fn remove(&self, space: &mut dyn ObjectSpace, info: &EntityInfo, key: &Value) -> Result<(), Failure> {
936        space.find(&info.name, key)?.ok_or_else(Failure::not_found)?;
937        let whole = (self.space)().find(&info.name, key)?.ok_or_else(Failure::not_found)?;
938        self.refuse_broken_rules(info, &whole, CONTEXT_DELETE)?;
939        space.delete(&info.name, key)?;
940        Ok(())
941    }

943    /// Records of an aggregated collection that came along. Each has to belong to this owner.
944    fn nested(
945        &self,
946        space: &mut dyn ObjectSpace,
947        info: &EntityInfo,
948        key: &Value,
949        sent: &Map<String, Json>,
950        depth: usize,
951    ) -> Result<(), Failure> {
952        for (name, value) in sent {
953            let Some(field) = info.field_named(name) else { continue };
954            if !matches!(field.kind, FieldKind::Collection(_)) {
955                continue;
956            }
957            if !field.aggregated {
958                return Err(Failure::bad_request(format!("{name} is not written through its owner")));
959            }
960            if depth >= MOST_NESTING {
961                return Err(Failure::bad_request("The records are nested too deep."));
962            }
963            let (child, back) = self
964                .back_reference(info, field)
965                .ok_or_else(|| Failure::bad_request(format!("{name} cannot be written")))?;
966            self.entity(&child.name)?;
967            let items =
968                value.as_array().ok_or_else(|| Failure::bad_request(format!("{name} has to be a list")))?;
969            let child_key = child
970                .key_field()
971                .ok_or_else(|| Failure::bad_request(format!("{} has no key", child.name)))?
972                .clone();
973            for item in items {
974                let item = item
975                    .as_object()
976                    .ok_or_else(|| Failure::bad_request(format!("{name} has to hold records")))?;
977                let mine = |space: &dyn ObjectSpace, wanted: &Value| -> Result<(), Failure> {
978                    let found = space.find(&child.name, wanted)?.ok_or_else(Failure::not_found)?;
979                    if found.get(&back) == *key { Ok(()) } else { Err(Failure::not_found()) }
980                };
981                if let Some(gone) = item.get("$delete") {
982                    let gone = from_json(&child_key, gone)?;
983                    mine(&*space, &gone)?;
984                    self.remove(space, &child, &gone)?;
985                } else if let Some(existing) = item.get(&child_key.name) {
986                    let existing = from_json(&child_key, existing)?;
987                    mine(&*space, &existing)?;
988                    let mut rest = item.clone();
989                    rest.remove(&child_key.name);
990                    if rest.contains_key(&back) {
991                        return Err(Failure::bad_request(format!("{back} comes from the owner")));
992                    }
993                    self.change(space, &child, &existing, &rest, depth + 1)?;
994                } else {
995                    if item.contains_key(&back) {
996                        return Err(Failure::bad_request(format!("{back} comes from the owner")));
997                    }
998                    let mut record = item.clone();
999                    record.insert(back.clone(), ravenrs_app::value_to_json(key));
1000                    self.create(space, &child.name, &record, depth + 1)?;
1001                }
1002            }
1003        }
1004        Ok(())
1005    }

1007    /// What the rules would say of a record, with nothing kept.
1008    fn check(&self, entity: &str, sent: &Map<String, Json>) -> Result<(u16, Json), Failure> {
1009        let info = self.entity(entity)?;
1010        let mut object = info.new_object();
1011        apply(&info, &mut object, sent, true)?;
1012        info.compute(&mut object, &ravenrs_app::environment());
1013        let raw = (self.space)();
1014        let found: Vec<Json> = validate(&info, &object, CONTEXT_SAVE, &*raw, &self.words)
1015            .iter()
1016            .map(|violation| {
1017                json!({
1018                    "field": violation.field,
1019                    "message": violation.message,
1020                    "stops": violation.is_error(),
1021                })
1022            })
1023            .collect();
1024        Ok((200, json!({ "violations": found })))
1025    }

1027    fn batch(&self, space: &mut dyn ObjectSpace, sent: &Map<String, Json>) -> Result<(u16, Json), Failure> {
1028        let operations = sent
1029            .get("operations")
1030            .and_then(Json::as_array)
1031            .ok_or_else(|| Failure::bad_request("A batch has a list of operations."))?;
1032        if operations.len() > MOST_OPERATIONS {
1033            return Err(Failure::bad_request("The batch has too many operations."));
1034        }
1035        let mut results = Vec::with_capacity(operations.len());
1036        for (index, operation) in operations.iter().enumerate() {
1037            let done = self.one_of_batch(space, operation);
1038            match done {
1039                Ok(result) => results.push(result),
1040                Err(mut failure) => {
1041                    space.rollback();
1042                    failure.operation = Some(index);
1043                    return Err(failure);
1044                }
1045            }
1046        }
1047        space.commit()?;
1048        Ok((200, json!({ "results": results })))
1049    }
1050
1051    fn one_of_batch(&self, space: &mut dyn ObjectSpace, operation: &Json) -> Result<Json, Failure> {
1052        let text = |name: &str| operation.get(name).and_then(Json::as_str);
1053        let entity = text("entity").ok_or_else(|| Failure::bad_request("An operation names its entity."))?;
1054        let empty = Map::new();
1055        // No body is an empty one. Anything else that is not a record was a mistake, and taking it
1056        // as empty would quietly write a record of nothing but its defaults.
1057        let body = match operation.get("body") {
1058            None | Some(Json::Null) => &empty,
1059            Some(Json::Object(body)) => body,
1060            Some(_) => return Err(Failure::bad_request("The body of an operation is a record.")),
1061        };
1062        let info = self.entity(entity)?;
1063        let key = || -> Result<Value, Failure> {
1064            let field = info.key_field().ok_or_else(|| Failure::bad_request("It has no key."))?;
1065            let sent =
1066                operation.get("key").ok_or_else(|| Failure::bad_request("An operation names its key."))?;
1067            from_json(field, sent)
1068        };
1069        match text("method") {
1070            Some("POST") => self.create(space, entity, body, 0),
1071            Some("PATCH") => self.change(space, &info, &key()?, body, 0),
1072            Some("DELETE") => {
1073                self.remove(space, &info, &key()?)?;
1074                Ok(json!({}))
1075            }
1076            _ => Err(Failure::bad_request("An operation is POST, PATCH or DELETE.")),
1077        }
1078    }

1081/// The entity of a collection's records and their field that holds the owner's key: the one the
1082/// collection names, or else the first that points to the owner.
1083pub(crate) fn back_reference(
1084    entities: &[EntityInfo],
1085    owner: &EntityInfo,
1086    field: &ravenrs_app::FieldInfo,
1087) -> Option<(EntityInfo, String)> {
1088    let FieldKind::Collection(target) = &field.kind else { return None };
1089    let child = entities.iter().find(|entity| entity.name == *target)?.clone();
1090    let back = field.back_reference.clone().or_else(|| {
1091        child
1092            .fields
1093            .iter()
1094            .find(|other| matches!(&other.kind, FieldKind::Reference(to) if *to == owner.name))
1095            .map(|other| other.name.clone())
1096    })?;
1097    Some((child, back))
1098}
