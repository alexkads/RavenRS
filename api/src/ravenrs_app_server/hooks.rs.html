1//! Notifications a provider pushes in: a payment confirmed, a delivery made. Each source has a
2//! secret per organization, and a request is read only after its organization and signature check
3//! out. Each event is kept by organization, source and the provider's own id, so one delivered twice
4//! has one effect, and one older than what is known about the same subject changes nothing.
5//!
6//! The envelope a provider's adapter sends, signed with HMAC-SHA256 of the body:
7//!
8//! ```text
9//! POST /api/hooks/{source}
10//! X-RavenRS-Organization: T1
11//! X-RavenRS-Signature: sha256=<hex of HMAC-SHA256(secret, body)>
12//!
13//! { "id": "evt_1", "subject": "charge_9", "at": 1731600000000, "data": { .. } }
14//! ```
15//!
16//! `at` is when the fact happened at the provider, in milliseconds; `subject` is what it is about.
17

28/// The entity of the events received, which storage has to know for [`ApiService::webhook`].
29///
30/// [`ApiService::webhook`]: crate::ApiService::webhook
31pub const WEBHOOK_EVENT: &str = "WebhookEvent";
32/// The header with the organization the event is for.
33pub const ORGANIZATION_HEADER: &str = "x-ravenrs-organization";
34/// The header with `sha256=` and the HMAC of the body.
35pub const SIGNATURE_HEADER: &str = "x-ravenrs-signature";
36/// The longest provider id or subject kept.
37const MOST_ID: usize = 200;
38
39/// The events received, kept to know a repeated one and the latest fact of each subject. It keeps
40/// neither the body's signature nor the secret.
41pub fn webhook_event_entity() -> EntityInfo {
42    EntityInfo::new(WEBHOOK_EVENT)
43        .caption("Webhook events")
44        .field(FieldInfo::new("id", FieldKind::Guid).key())
45        .field(FieldInfo::new("organization", FieldKind::Text))
46        .field(FieldInfo::new("source", FieldKind::Text))
47        .field(FieldInfo::new("provider_id", FieldKind::Text).default_property())
48        .field(FieldInfo::new("subject", FieldKind::Text))
49        .field(FieldInfo::new("at", FieldKind::Integer))
50        .field(FieldInfo::new("applied", FieldKind::Boolean))
51}

53/// What a source's handler gets: the event and a unit of work, committed with the record of the
54/// event, so the effect and the mark that it happened go in together or not at all.
55#[non_exhaustive]
56pub struct WebhookContext<'a> {
57    /// The organization it is for.
58    pub organization: &'a str,
59    /// The provider's id of the event.
60    pub id: &'a str,
61    /// What it is about.
62    pub subject: &'a str,
63    /// When it happened at the provider, in milliseconds.
64    pub at: i64,
65    /// The unit of work, with no person behind it: the source is trusted by its secret.
66    pub space: &'a RefCell<Box<dyn ObjectSpace>>,
67}
68
69impl fmt::Debug for WebhookContext<'_> {
70    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
71        f.debug_struct("WebhookContext")
72            .field("id", &self.id)
73            .field("subject", &self.subject)
74            .finish_non_exhaustive()
75    }
76}

78/// What a source does with an event it has not seen and that is newer than what it knows.
79pub type WebhookHandler = Rc<dyn Fn(&WebhookContext<'_>, &Json) -> Result<(), Failure>>;

81/// A source of webhooks: its secret for each organization this service answers for.
82pub struct Webhook {
83    secrets: Vec<(String, Vec<u8>)>,
84    handler: WebhookHandler,
85}
86
87impl fmt::Debug for Webhook {
88    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
89        // The secrets never leave, not even in a debug print.
90        let organizations: Vec<&str> = self.secrets.iter().map(|(o, _)| o.as_str()).collect();
91        f.debug_struct("Webhook").field("organizations", &organizations).finish_non_exhaustive()
92    }
93}
94
95impl Webhook {
96    /// A source whose events `handler` applies.
97    pub fn new(handler: impl Fn(&WebhookContext<'_>, &Json) -> Result<(), Failure> + 'static) -> Self {
98        Self { secrets: Vec::new(), handler: Rc::new(handler) }
99    }
100
101    /// The secret this source signs with for `organization`, kept apart from any payload or log.
102    pub fn secret(mut self, organization: impl Into<String>, secret: impl AsRef<[u8]>) -> Self {
103        self.secrets.push((organization.into(), secret.as_ref().to_vec()));
104        self
105    }
106
107    fn secret_of(&self, organization: &str) -> Option<&[u8]> {
108        self.secrets.iter().find(|(o, _)| o == organization).map(|(_, s)| s.as_slice())
109    }
110}

112/// HMAC-SHA256 of `body` with `secret`, as RFC 2104 puts it.
113pub fn sign(secret: &[u8], body: &[u8]) -> String {
114    const BLOCK: usize = 64;
115    let mut key = if secret.len() > BLOCK { Sha256::digest(secret).to_vec() } else { secret.to_vec() };
116    key.resize(BLOCK, 0);
117    let pad = |byte: u8| key.iter().map(|k| k ^ byte).collect::<Vec<u8>>();
118    let inner = Sha256::new().chain_update(pad(0x36)).chain_update(body).finalize();
119    let outer = Sha256::new().chain_update(pad(0x5c)).chain_update(inner).finalize();
120    format!("sha256={}", outer.iter().map(|b| format!("{b:02x}")).collect::<String>())
121}

123/// Compares without stopping at the first difference, so the time taken says nothing.
124fn same(a: &[u8], b: &[u8]) -> bool {
125    a.len() == b.len() && a.iter().zip(b).fold(0u8, |acc, (x, y)| acc | (x ^ y)) == 0
126}

128/// The one answer to anything wrong before the body is read: no detail of which part failed.
129fn refused() -> Failure {
130    Failure::new(401, "unauthorized", "The notification was not accepted.")
131}

133fn text(record: &Object, name: &str) -> String {
134    match record.get(name) {
135        Value::Text(text) => text,
136        _ => String::new(),
137    }

140/// Answers one notification for `source`.
141pub(crate) fn receive(
142    hook: &Webhook,
143    source: &str,
144    request: &crate::ApiRequest,
145    space: Box<dyn ObjectSpace>,
146) -> Result<(u16, Json), Failure> {
147    let organization = request.headers.get(ORGANIZATION_HEADER).ok_or_else(refused)?;
148    let secret = hook.secret_of(organization).ok_or_else(refused)?;
149    let signature = request.headers.get(SIGNATURE_HEADER).ok_or_else(refused)?;
150    if !same(sign(secret, &request.body).as_bytes(), signature.as_bytes()) {
151        return Err(refused());
152    }
153    // Only now is the body read.
154    let body: Json = serde_json::from_slice(&request.body)
155        .map_err(|_| Failure::bad_request("The body has to be JSON."))?;
156    let field = |name: &str| {
157        body.get(name)
158            .and_then(Json::as_str)
159            .filter(|t| !t.is_empty() && t.len() <= MOST_ID)
160            .map(str::to_owned)
161            .ok_or_else(|| Failure::bad_request(format!("The notification needs {name}.")))
162    };
163    let (id, subject) = (field("id")?, field("subject")?);
164    let at = body
165        .get("at")
166        .and_then(Json::as_i64)
167        .ok_or_else(|| Failure::bad_request("The notification needs at."))?;
168    let data = body.get("data").cloned().unwrap_or(Json::Null);
169
170    let space = RefCell::new(space);
171    let known = space.borrow().query(WEBHOOK_EVENT, &Query::all())?;
172    let ours: Vec<&Object> = known
173        .iter()
174        .filter(|e| text(e, "organization") == *organization && text(e, "source") == source)
175        .collect();
176    if ours.iter().any(|e| text(e, "provider_id") == id) {
177        return Ok((200, json!({ "received": true, "duplicate": true })));
178    }
179    let latest = ours
180        .iter()
181        .filter(|e| text(e, "subject") == subject && e.get("applied") == Value::Bool(true))
182        .filter_map(|e| match e.get("at") {
183            Value::Number(at) => Some(at as i64),
184            _ => None,
185        })
186        .max();
187    // A fact older than one already applied to the same subject is kept, but changes nothing.
188    let applies = latest.is_none_or(|latest| at > latest);
189    if applies {
190        let context = WebhookContext { organization, id: &id, subject: &subject, at, space: &space };
191        (hook.handler)(&context, &data)?;
192    }
193    let mut record = space.borrow_mut().new_object(WEBHOOK_EVENT)?;
194    record.set("organization", organization.as_str());
195    record.set("source", source);
196    record.set("provider_id", id.as_str());
197    record.set("subject", subject.as_str());
198    record.set("at", at as f64);
199    record.set("applied", applies);
200    let mut space = space.into_inner();
201    space.save(record)?;
202    space.commit()?;
203    tracing::info!(source, applied = applies, "webhook received");
204    Ok((200, json!({ "received": true, "applied": applies })))
205}
