1//! What the database needs for commits: one function that applies a whole unit of work in one
2//! transaction. PostgREST runs each call to a function in a transaction of its own, so a failure
3//! anywhere leaves nothing behind.
4//!
5//! It runs as whoever called it (`SECURITY INVOKER`): the grants and row level security that
6//! hold for the tables over REST hold here too. Names go through `format('%I')`, and values
7//! through `jsonb_populate_record`, which reads each one as its column's type.
8
9/// Installs `ravenrs_commit(changes jsonb)` in the schema PostgREST serves. Run it once, as
10/// whoever owns that schema, and then grant `EXECUTE` to the roles that write:
11///
12/// ```sql
13/// SET search_path TO api;  -- the schema PostgREST serves
14/// -- SERVER_SQL here
15/// REVOKE EXECUTE ON FUNCTION ravenrs_commit(jsonb) FROM PUBLIC;
16/// GRANT EXECUTE ON FUNCTION ravenrs_commit(jsonb) TO authenticated;
17/// NOTIFY pgrst, 'reload schema';
18/// ```
19pub const SERVER_SQL: &str = r#"
20CREATE OR REPLACE FUNCTION ravenrs_commit(changes jsonb) RETURNS jsonb
21LANGUAGE plpgsql SECURITY INVOKER AS $ravenrs$
22DECLARE
23  change jsonb;
24  resource text;
25  key_column text;
26  key_value jsonb;
27  columns text;
28  affected bigint;
29BEGIN
30  FOR change IN SELECT value FROM jsonb_array_elements(changes) LOOP
31    resource := change->>'resource';
32    key_column := change->>'key_column';
33    key_value := jsonb_build_object(key_column, change->'key');
34    SELECT string_agg(format('%I', name), ', ') INTO columns
35      FROM jsonb_object_keys(CASE WHEN jsonb_typeof(change->'values') = 'object'
36                                  THEN change->'values' ELSE '{}'::jsonb END) AS name;
37    CASE change->>'op'
38    WHEN 'insert' THEN
39      EXECUTE format('INSERT INTO %I (%s) SELECT %s FROM jsonb_populate_record(NULL::%I, $1)',
40                     resource, columns, columns, resource)
41        USING change->'values';
42    WHEN 'update' THEN
43      IF columns IS NULL THEN
44        EXECUTE format('SELECT 1 FROM %I WHERE %I = (SELECT %I FROM jsonb_populate_record(NULL::%I, $1))',
45                       resource, key_column, key_column, resource)
46          USING key_value;
47      ELSE
48        EXECUTE format('UPDATE %I SET (%s) = (SELECT %s FROM jsonb_populate_record(NULL::%I, $1)) '
49                       'WHERE %I = (SELECT %I FROM jsonb_populate_record(NULL::%I, $2))',
50                       resource, columns, columns, resource, key_column, key_column, resource)
51          USING change->'values', key_value;
52      END IF;
53      GET DIAGNOSTICS affected = ROW_COUNT;
54      IF affected = 0 THEN
55        RAISE EXCEPTION 'a record of % is gone, or is not yours to change', resource
56          USING ERRCODE = 'PT409';
57      END IF;
58    WHEN 'delete' THEN
59      EXECUTE format('DELETE FROM %I WHERE %I = (SELECT %I FROM jsonb_populate_record(NULL::%I, $1))',
60                     resource, key_column, key_column, resource)
61        USING key_value;
62    ELSE
63      RAISE EXCEPTION 'unknown change %', change->>'op' USING ERRCODE = '22023';
64    END CASE;
65  END LOOP;
66  RETURN jsonb_build_object('applied', jsonb_array_length(changes));
67END
68$ravenrs$;
69"#;
