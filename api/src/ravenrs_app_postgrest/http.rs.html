1//! How a request reaches PostgREST. The space only builds requests and reads answers; [`Http`]
2//! carries them, so a test can stand in for the server and lose an answer on purpose.
3
4use std::fmt;
5use std::time::Duration;
6
7/// A request on its way.
8#[derive(Clone, PartialEq, Eq)]
9#[non_exhaustive]
10pub struct HttpRequest {
11    /// `GET` or `POST`.
12    pub method: &'static str,
13    /// The whole address, with its query.
14    pub url: String,
15    /// Headers, the token among them.
16    pub headers: Vec<(String, String)>,
17    /// The JSON sent, if any.
18    pub body: Option<String>,
19}
20
21impl fmt::Debug for HttpRequest {
22    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
23        // The headers carry the token and the key: only their names are shown.
24        let names: Vec<&str> = self.headers.iter().map(|(name, _)| name.as_str()).collect();
25        f.debug_struct("HttpRequest")
26            .field("method", &self.method)
27            .field("url", &self.url)
28            .field("headers", &names)
29            .finish_non_exhaustive()
30    }
31}
32
33impl HttpRequest {
34    /// The value of a header, by its name in any case.
35    pub fn header(&self, name: &str) -> Option<&str> {
36        self.headers
37            .iter()
38            .find(|(named, _)| named.eq_ignore_ascii_case(name))
39            .map(|(_, value)| value.as_str())
40    }
41}
42
43/// What came back.
44#[derive(Clone, Debug, PartialEq, Eq)]
45#[non_exhaustive]
46pub struct HttpResponse {
47    /// The status.
48    pub status: u16,
49    /// The body.
50    pub body: String,
51}
52
53impl HttpResponse {
54    /// With this status and body.
55    pub fn new(status: u16, body: impl Into<String>) -> Self {
56        Self { status, body: body.into() }
57    }
58}
59
60/// Why no answer came.
61#[derive(Clone, Debug, PartialEq, Eq)]
62#[non_exhaustive]
63pub enum HttpError {
64    /// The request never left: no name, no connection, a bad address. Nothing happened.
65    NotSent(String),
66    /// It left, and no answer came back. It may have happened.
67    NoAnswer(String),
68}
69
70/// Carries a request and waits for the answer.
71pub trait Http {
72    /// Sends it.
73    ///
74    /// # Errors
75    ///
76    /// When no answer came, saying whether the request ever left.
77    fn send(&self, request: &HttpRequest) -> Result<HttpResponse, HttpError>;
78}

80/// HTTP and HTTPS through `ureq`, with rustls.
81#[derive(Clone)]
82pub struct UreqHttp {
83    agent: ureq::Agent,
84}
85
86impl fmt::Debug for UreqHttp {
87    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
88        f.debug_struct("UreqHttp").finish_non_exhaustive()
89    }
90}
91
92impl Default for UreqHttp {
93    fn default() -> Self {
94        Self::new(Duration::from_secs(10), Duration::from_secs(60))
95    }
96}
97
98impl UreqHttp {
99    /// Waiting this long to connect, and that long for the whole answer.
100    pub fn new(connect: Duration, whole: Duration) -> Self {
101        let agent = ureq::Agent::config_builder()
102            .http_status_as_error(false)
103            .timeout_connect(Some(connect))
104            .timeout_global(Some(whole))
105            .build()
106            .new_agent();
107        Self { agent }
108    }
109}
110
111/// Whether a failure came before the request left.
112fn before_sending(error: &ureq::Error) -> bool {
113    match error {
114        ureq::Error::HostNotFound
115        | ureq::Error::ConnectionFailed
116        | ureq::Error::BadUri(_)
117        | ureq::Error::Http(_)
118        | ureq::Error::InvalidProxyUrl => true,
119        ureq::Error::Timeout(timeout) => {
120            matches!(timeout, ureq::Timeout::Resolve | ureq::Timeout::Connect)
121        }
122        ureq::Error::Io(io) => {
123            matches!(io.kind(), std::io::ErrorKind::ConnectionRefused | std::io::ErrorKind::AddrNotAvailable)
124        }
125        _ => false,
126    }
127}
128
129impl Http for UreqHttp {
130    fn send(&self, request: &HttpRequest) -> Result<HttpResponse, HttpError> {
131        let mut built = ureq::http::Request::builder().method(request.method).uri(&request.url);
132        for (name, value) in &request.headers {
133            built = built.header(name, value);
134        }
135        let asked = built
136            .body(request.body.clone().unwrap_or_default())
137            .map_err(|error| HttpError::NotSent(error.to_string()))?;
138        let mut response = self.agent.run(asked).map_err(|error| {
139            if before_sending(&error) {
140                HttpError::NotSent(error.to_string())
141            } else {
142                HttpError::NoAnswer(error.to_string())
143            }
144        })?;
145        let status = response.status().as_u16();
146        let body =
147            response.body_mut().read_to_string().map_err(|error| HttpError::NoAnswer(error.to_string()))?;
148        Ok(HttpResponse::new(status, body))
149    }
