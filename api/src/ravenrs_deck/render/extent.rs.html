1//! The pasteboard: what the editor draws around the slide.
2//!
3//! PowerPoint's Normal view draws shapes, pictures and text that sit partly or wholly off the slide
4//! on the area around it, at full strength and with their effects (master and layout graphics
5//! too), so they can be seen and grabbed. Slide shows, exports and thumbnails clip to the slide.
6//!
7//! The editor renders only what it needs: the slide, grown to take in the drawing that spills off
8//! it and is in view ([`pasteboard_region`]). A slide with nothing off it costs what it always did.
9

18/// Largest distance (points) a shape's outline or effects are taken to reach beyond its box:
19/// hostile files can't blow the pasteboard up.
20const MAX_SPILL: f64 = 2000.0;
21
22/// The visible area is rounded out to tiles of this many pixels, so scrolling over a pasteboard
23/// that is wider than the window re-renders only when a new tile comes into view.
24pub const TILE: i64 = 256;
25
26/// How far (points) a shape's outline and effects reach beyond its box: half the outline and its
27/// arrowheads, the outer shadow (offset, blur and scale), the glow and the reflection.
28pub(crate) fn spill(line: Option<&Line>, effects: Option<&Effects>, w: f64, h: f64) -> f64 {
29    let fin = |v: f64| if v.is_finite() { v.abs() } else { 0.0 };
30    let mut s: f64 = 0.0;
31    if let Some(l) = line
32        && l.fill.as_ref().is_some_and(|f| !f.is_none())
33    {
34        // Arrowheads reach up to a few line widths past the end point.
35        s = s.max(fin(l.width.unwrap_or(0.75)) * 3.0 + 1.0);
36    }
37    if let Some(e) = effects {
38        let big = fin(w).max(fin(h));
39        if let Some(sh) = &e.outer_shadow {
40            let grow = (fin(sh.sx).max(fin(sh.sy)) - 1.0).max(0.0) * big;
41            s = s.max(fin(sh.dist) + fin(sh.blur) + grow);
42        }
43        if let Some(g) = &e.glow {
44            s = s.max(fin(g.radius));
45        }
46        if let Some(r) = &e.reflection {
47            s = s.max(fin(h) + fin(r.dist));
48        }
49    }
50    s.min(MAX_SPILL)
51}

53/// Slide-space box a top-level shape draws into, its spill included.
54fn shape_box(rctx: &Ctx, s: &Shape) -> Option<Rect> {
55    if s.hidden {
56        return None;
57    }
58    let x: Xfrm = resolve::xfrm(rctx, s);
59    if !x.is_finite() {
60        return None;
61    }
62    let b = match &s.kind {
63        ShapeKind::Ink { strokes } => {
64            let mut pts = strokes.iter().flat_map(|k| k.points.iter()).filter(|(px, py, _)| px.is_finite() && py.is_finite());
65            let (px, py, _) = pts.next()?;
66            let r = pts.fold(Rect::new(*px, *py, *px, *py), |r, (px, py, _)| r.union_pt(kurbo::Point::new(*px, *py)));
67            let w = strokes.iter().map(|k| k.width).filter(|w| w.is_finite()).fold(0.0_f64, f64::max).min(MAX_SPILL);
68            return Some(r.inflate(w, w));
69        }
70        _ => x.bounds(),
71    };
72    let (line, _) = resolve::line(rctx, s);
73    let (fx, _) = resolve::effects(rctx, s);
74    let d = spill(Some(&line), fx.as_ref(), x.w, x.h);
75    Some(b.inflate(d, d))
76}

78fn union_boxes<'a>(rctx: &Ctx, shapes: impl IntoIterator<Item = &'a Shape>, into: Rect) -> Rect {
79    shapes.into_iter().filter_map(|s| shape_box(rctx, s)).filter(|r| r.is_finite()).fold(into, |a, r| a.union(r))
80}

82fn slide_rect(pres: &Presentation) -> Rect {
83    Rect::new(0.0, 0.0, pres.slide_size.width, pres.slide_size.height)
84}

86/// What slide `index` draws, in slide space: at least the slide, grown by every shape (and master
87/// and layout graphic) that reaches off it. Estimated from the shapes' boxes, outlines and effects;
88/// text spilling out of its box is found while rendering ([`crate::render::render_slide_drawn`]).
89pub fn slide_extent(pres: &Presentation, index: usize) -> Rect {
90    let page = slide_rect(pres);
91    let Some(slide) = pres.slides.get(index) else { return page };
92    let Some(rctx) = Ctx::for_slide(pres, slide) else { return page };
93    let mut r = union_boxes(&rctx, &slide.shapes, page);
94    let (show_layout, show_master) = resolve::show_master_shapes(slide, rctx.layout);
95    if show_master {
96        let mctx = Ctx::for_master(pres, rctx.master);
97        r = union_boxes(&mctx, rctx.master.shapes.iter().filter(|s| s.ph.is_none()), r);
98    }
99    if show_layout && let Some(l) = rctx.layout {
100        let lctx = Ctx::for_layout(pres, rctx.master, l);
101        r = union_boxes(&lctx, l.shapes.iter().filter(|s| s.ph.is_none()), r);
102    }
103    r
104}

106/// What a layout (or the master, `layout: None`) draws in Slide Master view, in slide space.
107pub fn layout_extent(pres: &Presentation, master: usize, layout: Option<usize>) -> Rect {
108    let page = slide_rect(pres);
109    let Some(m) = pres.masters.get(master) else { return page };
110    match layout.and_then(|i| m.layouts.get(i)) {
111        Some(l) => {
112            let mut r = union_boxes(&Ctx::for_layout(pres, m, l), &l.shapes, page);
113            if l.show_master_shapes {
114                r = union_boxes(&Ctx::for_master(pres, m), m.shapes.iter().filter(|s| s.ph.is_none()), r);
115            }
116            r
117        }
118        None => union_boxes(&Ctx::for_master(pres, m), &m.shapes, page),
119    }
120}

122/// A rectangle of pixels on the editor canvas, relative to the slide's top-left corner.
123#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
124pub struct Region {
125    pub x: i32,
126    pub y: i32,
127    pub w: u32,
128    pub h: u32,
129}
130
131impl Region {
132    /// The slide itself, `w` × `h` pixels.
133    pub fn slide(w: u32, h: u32) -> Self {
134        Region { x: 0, y: 0, w: w.max(1), h: h.max(1) }
135    }
136
137    fn edges(&self) -> (i64, i64, i64, i64) {
138        let (x, y) = (self.x as i64, self.y as i64);
139        (x, y, x + self.w as i64, y + self.h as i64)
140    }
141
142    fn from_edges(x0: i64, y0: i64, x1: i64, y1: i64) -> Self {
143        let clamp = |v: i64| v.clamp(i32::MIN as i64, i32::MAX as i64) as i32;
144        let side = |a: i64, b: i64| (b - a).clamp(1, MAX_SIDE as i64) as u32;
145        Region { x: clamp(x0), y: clamp(y0), w: side(x0, x1), h: side(y0, y1) }
146    }
147
148    /// Does this region cover all of `other`?
149    pub fn contains(&self, other: &Region) -> bool {
150        let (a, b) = (self.edges(), other.edges());
151        a.0 <= b.0 && a.1 <= b.1 && a.2 >= b.2 && a.3 >= b.3
152    }
153
154    /// Is the region more than the slide (some of the pasteboard is drawn)?
155    pub fn beyond(&self, slide: (u32, u32)) -> bool {
156        *self != Region::slide(slide.0, slide.1)
157    }
158
159    /// Render options' `offset` that puts the slide's top-left corner where it falls in the region.
160    pub fn offset(&self) -> (f64, f64) {
161        (-(self.x as f64), -(self.y as f64))
162    }
163
164    /// The region's size, for render options' `size`.
165    pub fn size(&self) -> (u32, u32) {
166        (self.w, self.h)
167    }
168}

170/// `r` (slide space) in pixels at `scale` pixels per point, rounded outwards; `None` when not finite.
171fn to_px(r: Rect, scale: f64) -> Option<(i64, i64, i64, i64)> {
172    let lim = i32::MAX as f64 / 2.0;
173    let px = |v: f64| if v.is_finite() { Some(v.clamp(-lim, lim)) } else { None };
174    Some((
175        px((r.x0 * scale).floor())? as i64,
176        px((r.y0 * scale).floor())? as i64,
177        px((r.x1 * scale).ceil())? as i64,
178        px((r.y1 * scale).ceil())? as i64,
179    ))
180}

182/// The pixels the editor renders: the slide (`slide_px`, its size in pixels at `scale` pixels per
183/// point) plus the part of `extent` (slide space, from [`slide_extent`] or what a render drew) that
184/// falls within `visible`, the canvas's area relative to the slide's top-left corner. The visible
185/// area is rounded out to whole [`TILE`]s.
186pub fn pasteboard_region(slide_px: (u32, u32), scale: f64, extent: Rect, visible: Region) -> Region {
187    let page = Region::slide(slide_px.0, slide_px.1);
188    if !(scale.is_finite() && scale > 0.0) {
189        return page;
190    }
191    let Some((ex0, ey0, ex1, ey1)) = to_px(extent, scale) else { return page };
192    let (vx0, vy0, vx1, vy1) = visible.edges();
193    let (vx0, vy0) = (vx0.div_euclid(TILE) * TILE, vy0.div_euclid(TILE) * TILE);
194    let (vx1, vy1) = ((vx1 + TILE - 1).div_euclid(TILE) * TILE, (vy1 + TILE - 1).div_euclid(TILE) * TILE);
195    let (cx0, cy0, cx1, cy1) = (ex0.max(vx0), ey0.max(vy0), ex1.min(vx1), ey1.min(vy1));
196    if cx0 >= cx1 || cy0 >= cy1 {
197        return page;
198    }
199    let (px0, py0, px1, py1) = page.edges();
200    Region::from_edges(cx0.min(px0), cy0.min(py0), cx1.max(px1), cy1.max(py1))
201}

292    #[test]
293    fn the_region_is_the_slide_unless_something_off_it_is_in_view() {
294        let slide = (960, 540);
295        let page = Region::slide(960, 540);
296        let view = Region { x: -100, y: -50, w: 1160, h: 640 };
297        let whole = Rect::new(0.0, 0.0, 960.0, 540.0);
298        assert_eq!(pasteboard_region(slide, 1.0, whole, view), page);
299        assert!(!pasteboard_region(slide, 1.0, whole, view).beyond(slide));
300        // Off to the left, in view: the region reaches it.
301        let r = pasteboard_region(slide, 1.0, Rect::new(-80.0, 0.0, 960.0, 540.0), view);
302        assert_eq!((r.x, r.y, r.w, r.h), (-80, 0, 1040, 540));
303        assert!(r.beyond(slide) && r.contains(&page));
304        assert_eq!(r.offset(), (80.0, 0.0));
305        // Far off and out of view (the view rounds out to tiles): only what's visible.
306        let r = pasteboard_region(slide, 1.0, Rect::new(-5000.0, 0.0, 960.0, 540.0), view);
307        assert_eq!(r.x, -256);
308        // Wholly out of view: the slide.
309        assert_eq!(pasteboard_region(slide, 1.0, Rect::new(-5000.0, 0.0, -4000.0, 540.0).union(whole), Region { x: 0, y: 0, w: 960, h: 540 }), page);
310        // Scale applies; nonsense falls back to the slide.
311        let r = pasteboard_region((480, 270), 0.5, Rect::new(-80.0, 0.0, 960.0, 540.0), view);
312        assert_eq!(r.x, -40);
313        assert_eq!(pasteboard_region(slide, f64::NAN, whole, view), page);
314        assert_eq!(pasteboard_region(slide, 1.0, Rect::new(f64::NAN, 0.0, 1.0, 1.0), view), page);
315        // Huge extents are capped.
316        let r =
317            pasteboard_region(slide, 1.0, Rect::new(-1e12, -1e12, 1e12, 1e12), Region { x: -1_000_000, y: -1_000_000, w: 2_000_000, h: 2_000_000 });
318        assert!(r.w <= MAX_SIDE && r.h <= MAX_SIDE);
319    }
