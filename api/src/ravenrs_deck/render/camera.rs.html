1//! 3-D cameras (`a:scene3d/a:camera`) turning a flat shape, approximated in 2-D.
2//!
3//! A shape seen through a 3-D camera preset (isometric, perspective…) or rotated with
4//! `a:camera/a:rot` is drawn by PowerPoint as its plane turned in space and projected. DeckCraft
5//! turns the shape's corners the same way and draws the shape under the affine transform that best
6//! fits them: exact for orthographic cameras, within about 2% of the shape's size for the
7//! perspective ones (whose foreshortening an affine transform can't show). Depth, bevels' sides and
8//! lighting are not drawn.
9//!
10//! The presets' turns were fitted to PowerPoint's exports (black-box: a picture with coloured
11//! corner markers under each preset, all fits within half a point on a 100 × 75 pt shape):
12//! points relative to the shape's centre are turned `p = Rz(-rev) · Rx(lat) · Ry(lon) · (x, y, 0)`
13//! and, for perspective cameras, scaled by `D / (D - p.z)` with `D = dist · max(w, h)`.
14
15use kurbo::{Affine, Point};

56/// Perspective distance for a perspective camera given its own rotation (fitted on
57/// `perspectiveFront` with `a:rot`, about 11 shape sizes).
58const PERSPECTIVE_DIST: f64 = 11.0;

60/// A camera's turn of the shape's plane.
61#[derive(Clone, Copy, Debug, PartialEq)]
62pub struct Camera {
63    pub lat: f64,
64    pub lon: f64,
65    pub rev: f64,
66    /// Perspective distance in shape sizes; 0 for an orthographic camera.
67    pub dist: f64,
68}

70/// The value of attribute `name` in the first `tag` element of `raw` (XML as kept from the file).
71fn attr<'a>(tag: &'a str, name: &str) -> Option<&'a str> {
72    let key = format!(" {name}=\"");
73    let s = tag.find(&key)? + key.len();
74    let rest = tag.get(s..)?;
75    rest.get(..rest.find('"')?)
76}

78/// The camera of 3-D XML kept from the file, when it turns the shape.
79pub fn from_raw(raw: &str) -> Option<Camera> {
80    let i = raw.find("camera ")?;
81    let rest = raw.get(i..)?;
82    let open_end = rest.find('>')?;
83    let open = rest.get(..open_end)?;
84    let prst = attr(open, "prst").unwrap_or("orthographicFront");
85    let preset = PRESETS.iter().find(|p| p.0 == prst);
86    let perspective = prst.starts_with("perspective");
87    // Its own `a:rot` (inside the camera element, not the light rig's).
88    let self_closing = open.ends_with('/');
89    let own_rot = if self_closing {
90        None
91    } else {
92        let body = rest.get(open_end..rest.find("camera>").unwrap_or(rest.len()))?;
93        body.find("rot ").and_then(|j| {
94            let t = body.get(j..)?;
95            let t = t.get(..t.find('>')?)?;
96            let deg = |n: &str| attr(t, n).and_then(|v| v.parse::<f64>().ok()).map(|v| v / 60000.0).unwrap_or(0.0);
97            Some((deg("lat"), deg("lon"), deg("rev")))
98        })
99    };
100    let cam = match (own_rot, preset) {
101        // PowerPoint turns the other way about the horizontal and vertical axes for `a:rot`.
102        (Some((lat, lon, rev)), p) => {
103            let dist = match p {
104                Some(p) if p.4 > 0.0 => p.4,
105                _ if perspective => PERSPECTIVE_DIST,
106                _ => 0.0,
107            };
108            Camera { lat: -lat, lon: -lon, rev, dist }
109        }
110        (None, Some(p)) => Camera { lat: p.1, lon: p.2, rev: p.3, dist: p.4 },
111        (None, None) => return None,
112    };
113    let flat = [cam.lat, cam.lon, cam.rev].iter().all(|a| (a.rem_euclid(360.0)).min(360.0 - a.rem_euclid(360.0)) < 1e-6);
114    (!flat).then_some(cam)
115}

117/// Where a point of the shape (relative to its centre) lands.
118fn project(c: &Camera, p: Point, size: f64) -> Point {
119    project3(c, p.x, p.y, 0.0, size)
120}

122/// Direction `v` (shape space: x right, y down, z to the viewer) turned by camera `c`, without
123/// perspective: a surface normal as the camera sees it.
124pub fn rotate(c: &Camera, v: [f64; 3]) -> [f64; 3] {
125    let (a, b, r) = (c.lat.to_radians(), c.lon.to_radians(), -c.rev.to_radians());
126    let (x, y, z) = (v[0], v[1], v[2]);
127    let (x1, y1, z1) = (x * b.cos() + z * b.sin(), y, -x * b.sin() + z * b.cos());
128    let (x2, y2, z2) = (x1, y1 * a.cos() - z1 * a.sin(), y1 * a.sin() + z1 * a.cos());
129    [x2 * r.cos() - y2 * r.sin(), x2 * r.sin() + y2 * r.cos(), z2]
130}

132/// Where a point in the shape's space (relative to its centre; z towards the viewer, so an
133/// extrusion runs to negative z) lands.
134pub fn project3(c: &Camera, x: f64, y: f64, z: f64, size: f64) -> Point {
135    let p = Point::new(x, y);
136    let (a, b, r) = (c.lat.to_radians(), c.lon.to_radians(), -c.rev.to_radians());
137    let (x1, y1, z1) = (p.x * b.cos() + z * b.sin(), p.y, -p.x * b.sin() + z * b.cos());
138    let (x2, y2, z2) = (x1, y1 * a.cos() - z1 * a.sin(), y1 * a.sin() + z1 * a.cos());
139    let (x3, y3) = (x2 * r.cos() - y2 * r.sin(), x2 * r.sin() + y2 * r.cos());
140    if c.dist > 0.0 {
141        let d = c.dist * size;
142        let s = d / (d - z2);
143        if s.is_finite() && s > 0.0 {
144            return Point::new(x3 * s, y3 * s);
145        }
146    }
147    Point::new(x3, y3)
148}

150/// The transform (shape space to shape space) that turns a `w` × `h` shape as camera `c` does:
151/// the affine map best fitting its turned corners.
152pub fn affine(c: &Camera, w: f64, h: f64) -> Option<Affine> {
153    if !(w.is_finite() && h.is_finite()) || w <= 0.0 || h <= 0.0 {
154        return None;
155    }
156    let size = w.max(h);
157    let src = [(-w / 2.0, -h / 2.0), (w / 2.0, -h / 2.0), (w / 2.0, h / 2.0), (-w / 2.0, h / 2.0)];
158    let dst: Vec<Point> = src.iter().map(|(x, y)| project(c, Point::new(*x, *y), size)).collect();
159    // Least squares over the four corners. The corners are symmetric about the centre, so the
160    // normal equations are diagonal: x' = a x + b y + e with a = Σ x x' / Σ x², and so on.
161    let (sxx, syy) = (src.iter().map(|p| p.0 * p.0).sum::<f64>(), src.iter().map(|p| p.1 * p.1).sum::<f64>());
162    let dot = |f: &dyn Fn(&(f64, f64), &Point) -> f64| src.iter().zip(&dst).map(|(s, d)| f(s, d)).sum::<f64>();
163    let a = dot(&|s, d| s.0 * d.x) / sxx;
164    let b = dot(&|s, d| s.1 * d.x) / syy;
165    let cc = dot(&|s, d| s.0 * d.y) / sxx;
166    let dd = dot(&|s, d| s.1 * d.y) / syy;
167    let e = dot(&|_, d| d.x) / 4.0;
168    let f = dot(&|_, d| d.y) / 4.0;
169    let local = Affine::new([a, cc, b, dd, e, f]);
170    let m = Affine::translate((w / 2.0, h / 2.0)) * local * Affine::translate((-w / 2.0, -h / 2.0));
171    (m.is_finite() && m.determinant().abs() > 1e-9).then_some(m)
172}

178    fn corners(m: Affine, w: f64, h: f64) -> Vec<Point> {
179        [(0.0, 0.0), (w, 0.0), (w, h), (0.0, h)].iter().map(|(x, y)| m * Point::new(*x, *y)).collect()
180    }

182    #[test]
183    fn turning_about_the_vertical_axis_narrows_the_shape() {
184        // orthographicFront with lon 60: half as wide, as tall (PowerPoint: 0.506 of the width).
185        let c = from_raw(r#"<a:scene3d><a:camera prst="orthographicFront"><a:rot lat="0" lon="3600000" rev="0"/></a:camera><a:lightRig rig="threePt" dir="t"><a:rot lat="0" lon="0" rev="1200000"/></a:lightRig></a:scene3d>"#).unwrap();
186        let m = affine(&c, 100.0, 75.0).unwrap();
187        let k = corners(m, 100.0, 75.0);
188        assert!((k[1].x - k[0].x - 50.0).abs() < 0.01 && (k[3].y - k[0].y - 75.0).abs() < 0.01, "{k:?}");
189        assert!(((k[0].x + k[1].x) / 2.0 - 50.0).abs() < 1e-9, "about the centre");
190    }

192    #[test]
193    fn presets_turn_the_shape_as_powerpoint_does() {
194        // isometricTopUp, 100 × 75: PowerPoint's corners at (-1.247, 0.127), (0.170, -0.955),
195        // (1.237, -0.141), (-0.181, 0.941) of the half sizes from the centre.
196        let c = from_raw(r#"<a:scene3d><a:camera prst="isometricTopUp"/><a:lightRig rig="threePt" dir="t"/></a:scene3d>"#).unwrap();
197        let k = corners(affine(&c, 100.0, 75.0).unwrap(), 100.0, 75.0);
198        for (p, (u, v)) in k.iter().zip([(-1.247, 0.127), (0.170, -0.955), (1.237, -0.141), (-0.181, 0.941)]) {
199            let (x, y) = ((p.x - 50.0) / 50.0, (p.y - 37.5) / 37.5);
200            assert!((x - u).abs() < 0.03 && (y - v).abs() < 0.03, "({x:.3}, {y:.3}) vs ({u}, {v})");
201        }
202        // Front-facing cameras and the oblique ones leave a flat shape as it is.
203        for raw in [r#"<a:scene3d><a:camera prst="orthographicFront"/></a:scene3d>"#, r#"<a:scene3d><a:camera prst="obliqueTopLeft"/></a:scene3d>"#] {
204            assert_eq!(from_raw(raw), None);
205        }
206        assert_eq!(from_raw("<a:sp3d/>"), None);
207    }

217        }
