7/// The prompt shown in an empty placeholder, formatted like its content would be.
8pub(crate) fn prompt_body(rctx: &Ctx, s: &Shape, kind: PhType) -> TextBody {
9    // Layout/master prompt text when the layout supplies a custom one.
10    let (lp, _) = resolve::parents(rctx, s);
11    let custom = lp.and_then(|l| l.ph.as_ref().filter(|p| p.has_custom_prompt).and(l.text.as_ref())).filter(|t| !t.is_empty());
12    if let Some(t) = custom {
13        return t.clone();
14    }
15    if matches!(rctx.owner, resolve::Owner::Layout | resolve::Owner::Master) {
16        return TextBody::default();
17    }
18    let txt = kind.prompt();
19    if txt.is_empty() || kind.is_footer_kind() {
20        return TextBody::default();
21    }
22    let mut t = TextBody::from_text(txt);
23    if matches!(kind, PhType::Picture | PhType::Chart | PhType::Table | PhType::Media | PhType::Diagram | PhType::ClipArt) {
24        t.body.anchor = Some(crate::model::text::Anchor::Bottom);
25        if let Some(p) = t.paragraphs.first_mut() {
26            p.props.align = Some(crate::model::text::Align::Center);
27            p.props.bullet = Some(crate::model::text::Bullet::None);
28            p.props.margin_left = Some(0.0);
29            p.props.indent = Some(0.0);
30            for r in &mut p.runs {
31                r.props.size = Some(14.0);
32            }
33        }
34    }
35    t
36}

38/// Most path elements this thread's glyph cache retains. Its entries are shared with the font
39/// cache, but they stay alive here after that cache evicts them, so they are bounded by elements.
40const GLYPH_CACHE_MAX_ELEMENTS: usize = 1 << 22;
41
42thread_local! {
43    /// The cached outlines, and the path elements they hold.
44    static GLYPH_CACHE: std::cell::RefCell<(HashMap<(u32, u32), Arc<BezPath>>, usize)> = std::cell::RefCell::new((HashMap::new(), 0));
45}
46
47pub(crate) fn glyph(face: &crate::fonts::FontFace, gid: u32) -> Arc<BezPath> {
48    let key = (face.id(), gid);
49    if let Some(p) = GLYPH_CACHE.with(|c| c.borrow().0.get(&key).cloned()) {
50        return p;
51    }
52    let p = FontDb::global().outline(face, gid);
53    GLYPH_CACHE.with(|c| {
54        let mut c = c.borrow_mut();
55        let n = p.elements().len();
56        if c.0.len() > 200_000 || c.1.saturating_add(n) > GLYPH_CACHE_MAX_ELEMENTS {
57            c.0.clear();
58            c.1 = 0;
59        }
60        c.1 += n;
61        c.0.insert(key, p.clone());
62    });
63    p
64}

81/// Where text laid out in text rect `tr` is drawn, for a shape drawn with `m` (shape-local →
82/// output): `m`, without mirroring, then the layout's `rotation` about the rect's centre.
83///
84/// Flips mirror a shape's geometry, never its text. When `m` reflects (flipH or flipV, the shape's
85/// own or a group's) the text rect keeps its mirrored place and the text is mirrored back about
86/// the rect's centre: a flipH shape's text stays readable and a flipV shape's text is turned 180°
87/// (both flips: 180°), as in PowerPoint. Rendering, hit testing and the caret all use this.
88pub fn text_transform(m: Affine, tr: Rect, rotation: f64) -> Affine {
89    let c = tr.center().to_vec2();
90    let mut t = m;
91    if m.determinant() < 0.0 && c.is_finite() {
92        t = t * Affine::translate(c) * Affine::scale_non_uniform(-1.0, 1.0) * Affine::translate(-c);
93    }
94    if rotation != 0.0 && rotation.is_finite() && c.is_finite() {
95        t = t * Affine::translate(c) * Affine::rotate(rotation.to_radians()) * Affine::translate(-c);
96    }
97    t
98}

112/// Draw a laid-out text block with per-paragraph animation states.
113pub fn draw_layout_with(ctx: &mut RenderContext, l: &crate::text::TextLayout, tr: Rect, m: Affine, paras: &[(usize, ParaState)]) {
114    let para = |i: usize| paras.iter().find(|(p, _)| *p == i).map(|(_, s)| *s).unwrap_or_default();
115    let m = text_transform(m, tr, l.rotation);
116    let lines_of = LineIndex::new(l);
117    let deco = |ctx: &mut RenderContext, d: &crate::text::Deco| {
118        let ps = para(d.para);
119        if !ps.visible || ps.opacity <= 0.001 {
120            return;
121        }
122        ctx.set_transform(m * Affine::translate(ps.offset));
123        ctx.set_paint(color(d.color, ps.opacity));
124        fill_deco(ctx, d);
125    };
126    for d in l.decos.iter().filter(|d| d.behind) {
127        deco(ctx, d);
128    }
129    // Text shadows, then glows, under all the text.
130    for pass in [TextEffect::Shadow, TextEffect::Glow] {
131        for run in &l.runs {
132            let ps = para(run.para);
133            if ps.visible && ps.opacity > 0.001 {
134                text_effect(ctx, run, pass, m * Affine::translate(ps.offset), ps.opacity);
135            }
136        }
137    }
138    for run in &l.runs {
139        if (run.alpha <= 0.0 || run.color.a == 0) && run.outline.is_none() {
140            continue;
141        }
142        let ps = para(run.para);
143        if !ps.visible || ps.opacity <= 0.001 {
144            continue;
145        }
146        let m = m * Affine::translate(ps.offset);
147        // A gradient fill starts over on every line and spans that line's text (from its top to
148        // its bottom, from its first to its last character), as in PowerPoint.
149        let mut grads: Vec<(usize, peniko::Gradient)> = Vec::new();
150        for (gid, x, y) in &run.glyphs {
151            let at = m * glyph_place(run, *gid, *x, *y);
152            let Some(g) = run.gradient.as_ref().filter(|g| !g.stops.is_empty()) else {
153                draw_glyph(ctx, run, *gid, at, ps.opacity);
154                continue;
155            };
156            let li = lines_of.line_at(run.para, *y);
157            if !grads.iter().any(|(i, _)| *i == li) {
158                let scheme = crate::color::ColorScheme { name: String::new(), colors: [Rgba::BLACK; 12] };
159                let stops = g.stops.iter().map(|s| (s.pos, s.color.resolve(&scheme, None))).collect();
160                let bounds = l.lines.get(li).and_then(line_bounds).unwrap_or(tr);
161                grads.push((li, paint::gradient(&g.shape, stops, bounds, run.alpha * ps.opacity)));
162            }
163            let fill = grads.iter().find(|(i, _)| *i == li).map(|(_, g)| (g, Affine::translate((-*x, -*y))));
164            draw_glyph_with(ctx, run, *gid, at, ps.opacity, fill);
165        }
166    }
167    for d in l.decos.iter().filter(|d| !d.behind) {
168        deco(ctx, d);
169    }
170}

172/// Where glyph `gid` of `run` with its pen at (`x`, `y`) is drawn, in text space: there, turned a
173/// quarter turn back about its em box's centre when it stands upright in vertical text.
174pub(crate) fn glyph_place(run: &crate::text::GlyphRun, gid: u32, x: f64, y: f64) -> Affine {
175    let at = Affine::translate((x, y));
176    match run.upright_centre(gid, x, y) {
177        Some((cx, cy)) => Affine::translate((cx, cy)) * Affine::rotate(-std::f64::consts::FRAC_PI_2) * Affine::translate((-cx, -cy)) * at,
178        None => at,
179    }
180}

182/// Fill an underline, strikethrough or highlight with the current paint and transform: its box,
183/// its dashes, or its zigzag stroked.
184pub(crate) fn fill_deco(ctx: &mut RenderContext, d: &crate::text::Deco) {
185    if let Some((pts, w)) = d.wave() {
186        let mut path = BezPath::new();
187        for (i, p) in pts.iter().enumerate() {
188            if i == 0 {
189                path.move_to(*p);
190            } else {
191                path.line_to(*p);
192            }
193        }
194        ctx.set_stroke(kurbo::Stroke::new(w.max(0.01)));
195        ctx.stroke_path(&path);
196        return;
197    }
198    for r in d.pieces() {
199        ctx.fill_rect(&r);
200    }
201}

203/// Finds the line of a paragraph holding a glyph on baseline `y` without scanning the paragraph's
204/// lines for every glyph. A paragraph whose lines are ordered top to bottom (every layout the text
205/// crate produces) is searched by bisection; any other falls back to a scan with the same answer.
206pub(crate) struct LineIndex<'a> {
207    l: &'a crate::text::TextLayout,
208    /// Per paragraph: its lines' tops, bottoms and baselines all rise with the line index.
209    ordered: Vec<bool>,
210}
211
212impl<'a> LineIndex<'a> {
213    pub(crate) fn new(l: &'a crate::text::TextLayout) -> Self {
214        let ordered = l
215            .para_lines
216            .iter()
217            .map(|&(first, count)| {
218                let end = first.saturating_add(count).min(l.lines.len());
219                l.lines.get(first..end).is_some_and(|ls| {
220                    ls.iter().all(|a| a.top <= a.bottom)
221                        && ls.windows(2).all(|w| w[0].top <= w[1].top && w[0].bottom <= w[1].bottom && w[0].baseline <= w[1].baseline)
222                })
223            })
224            .collect();
225        LineIndex { l, ordered }
226    }
227
228    /// The nearest line baseline to `y` among `lines` (the first of equals), as an index into `lines`.
229    fn nearest(lines: &[crate::text::LineInfo], y: f64) -> Option<usize> {
230        let i = lines.partition_point(|li| li.baseline < y);
231        let best = match (i.checked_sub(1), lines.get(i)) {
232            (Some(p), Some(n)) => {
233                if (lines[p].baseline - y).abs() <= (n.baseline - y).abs() {
234                    p
235                } else {
236                    i
237                }
238            }
239            (Some(p), None) => p,
240            (None, Some(_)) => i,
241            (None, None) => return None,
242        };
243        let b = lines[best].baseline;
244        Some(lines.partition_point(|li| li.baseline < b))
245    }
246
247    /// The paragraph's lines when they are ordered, with the index of the first.
248    fn ordered_lines(&self, para: usize) -> Option<(usize, &[crate::text::LineInfo])> {
249        let &(first, count) = self.l.para_lines.get(para)?;
250        let end = first.saturating_add(count).min(self.l.lines.len());
251        (self.ordered[para] && first < end).then(|| (first, &self.l.lines[first..end]))
252    }
253
254    /// The line of paragraph `para` holding a glyph on baseline `y` (the nearest baseline).
255    pub(crate) fn line_at(&self, para: usize, y: f64) -> usize {
256        if let Some((first, ls)) = self.ordered_lines(para) {
257            return Self::nearest(ls, y).map_or(0, |i| first + i);
258        }
259        line_at_scan(self.l, para, y)
260    }
261
262    /// The first line of paragraph `para` whose box holds `y`, else the nearest baseline's line.
263    pub(crate) fn line_of(&self, para: usize, y: f64) -> Option<usize> {
264        let Some((first, ls)) = self.ordered_lines(para) else { return crate::render::warp::line_of_scan(self.l, para, y) };
265        let i = ls.partition_point(|li| li.bottom + 0.01 < y);
266        if ls.get(i).is_some_and(|li| y >= li.top - 0.01) {
267            return Some(first + i);
268        }
269        Self::nearest(ls, y).map(|i| first + i)
270    }
271}

273/// The line of paragraph `para` holding a glyph on baseline `y` (the nearest baseline), by scanning.
274fn line_at_scan(l: &crate::text::TextLayout, para: usize, y: f64) -> usize {
275    if let Some(&(first, count)) = l.para_lines.get(para) {
276        if count == 0 {
277            return 0;
278        }
279        let end = (first + count).min(l.lines.len());
280        if first < end {
281            return l.lines[first..end]
282                .iter()
283                .enumerate()
284                .min_by(|a, b| (a.1.baseline - y).abs().total_cmp(&(b.1.baseline - y).abs()))
285                .map(|(i, _)| first + i)
286                .unwrap_or(0);
287        }
288    }
289    l.lines
290        .iter()
291        .enumerate()
292        .filter(|(_, li)| li.para == para)
293        .min_by(|a, b| (a.1.baseline - y).abs().total_cmp(&(b.1.baseline - y).abs()))
294        .map(|(i, _)| i)
295        .unwrap_or(0)
296}

298/// The box a line's text covers: its top and bottom, its first and last caret.
299pub(crate) fn line_bounds(li: &crate::text::LineInfo) -> Option<Rect> {
300    let (x0, x1) = li.caret_x.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), x| (a.min(*x), b.max(*x)));
301    let r = Rect::new(x0, li.top, x1, li.bottom);
302    (r.is_finite() && r.width() > 0.0 && r.height() > 0.0).then_some(r)
303}

311/// Draw `run`'s shadow or glow (if it has one) under its glyphs, placed by `m`.
312fn text_effect(ctx: &mut RenderContext, run: &crate::text::GlyphRun, which: TextEffect, m: Affine, opacity: f64) {
313    let Some(fx) = run.effects else { return };
314    let k = run.size / run.face.upem.max(1.0);
315    let skew = if run.fake_italic { Affine::new([1.0, 0.0, -0.2, 1.0, 0.0, 0.0]) } else { Affine::IDENTITY };
316    let px = scale_of(m);
317    let (c, sigma, offset, grow) = match (which, fx.shadow, fx.glow) {
318        (TextEffect::Shadow, Some(s), _) => (s.color, s.blur * 0.5 * px, Vec2::new(s.dx, s.dy), 0.0),
319        // Measured on PowerPoint's export: a text glow of radius r is solid to about a third of r
320        // past the glyph's edge and fades out by about two thirds of r.
321        (TextEffect::Glow, _, Some((gc, r))) => (gc, r * 0.15 * px, Vec2::ZERO, r * 0.68),
322        _ => return,
323    };
324    let alpha = (c.a as f64 / 255.0) * opacity;
325    if alpha <= 0.0 {
326        return;
327    }
328    ctx.set_transform(Affine::IDENTITY);
329    ctx.push_layer(None, None, Some(alpha as f32), None, Some(blur(sigma)));
330    ctx.set_paint(color(c.with_alpha(255), 1.0));
331    for (gid, x, y) in &run.glyphs {
332        let path = glyph(&run.face, *gid);
333        if path.elements().is_empty() {
334            continue;
335        }
336        // The offset is in slide space: a shadow falls the same way whatever the text's rotation.
337        let at = pre_translate(m * glyph_place(run, *gid, *x, *y), offset * px) * skew * Affine::scale(k);
338        ctx.set_transform(at);
339        ctx.fill_path(&path);
340        if grow > 0.0 {
341            ctx.set_stroke(kurbo::Stroke::new(grow / k.max(1e-9)).with_join(kurbo::Join::Round));
342            ctx.stroke_path(&path);
343        }
344    }
345    ctx.pop_layer();
346}

353/// [`draw_glyph`], filled with `fill` (a gradient laid out in the text's space, and the map from
354/// the glyph's origin to that space) instead of the run's colour.
355fn draw_glyph_with(
356    ctx: &mut RenderContext,
357    run: &crate::text::GlyphRun,
358    gid: u32,
359    at: Affine,
360    opacity: f64,
361    fill: Option<(&peniko::Gradient, Affine)>,
362) {
363    let alpha = run.alpha * opacity;
364    if (alpha <= 0.0 || run.color.a == 0) && run.outline.is_none() {
365        return;
366    }
367    let k = run.size / run.face.upem.max(1.0);
368    // Colour glyphs (emoji) paint their own colours; an outline (text outline effect) still
369    // strokes the plain glyph below.
370    if alpha > 0.0 && run.outline.is_none() && run.face.is_color() && colr::paint(ctx, &run.face, gid, at, k, run.color, alpha) {
371        return;
372    }
373    let path = glyph(&run.face, gid);
374    if path.elements().is_empty() {
375        return;
376    }
377    let skew = if run.fake_italic { Affine::new([1.0, 0.0, -0.2, 1.0, 0.0, 0.0]) } else { Affine::IDENTITY };
378    ctx.set_transform(at * skew * Affine::scale(k));
379    if let Some((g, to_text)) = fill
380        && alpha > 0.0
381    {
382        ctx.set_paint(g.clone());
383        ctx.set_paint_transform((skew * Affine::scale(k)).inverse() * to_text);
384        ctx.fill_path(&path);
385        ctx.reset_paint_transform();
386    } else if alpha > 0.0 && run.color.a > 0 {
387        ctx.set_paint(color(run.color, alpha));
388        ctx.fill_path(&path);
389        if run.fake_bold {
390            ctx.set_stroke(kurbo::Stroke::new(run.face.upem * 0.025).with_join(kurbo::Join::Round));
391            ctx.stroke_path(&path);
392        }
393    }
394    if let Some((oc, ow)) = run.outline {
395        ctx.set_paint(color(oc, opacity));
396        ctx.set_stroke(kurbo::Stroke::new(ow / k.max(1e-9)).with_join(kurbo::Join::Round));
397        ctx.stroke_path(&path);
398    }
399}
