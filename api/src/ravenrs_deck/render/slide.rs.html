6static PANICS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);

8fn note_panic() {
9    PANICS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
10}

12/// Render panics caught (and drawn as blank) since start; soak tests require it to stay 0.
13pub fn panics_caught() -> u64 {

17/// Does anything on this slide (or its layout/master graphics) use blur-based effects?
18pub fn needs_filters(pres: &Presentation, slide: &Slide) -> bool {
19    let Some(rctx) = Ctx::for_slide(pres, slide) else { return false };
20    let mut any = false;
21    let mut check = |shapes: &[Shape]| {
22        crate::model::walk(shapes, &mut |s, _| {
23            if any {
24                return;
25            }
26            let (e, _) = resolve::effects(&rctx, s);
27            if e.is_some_and(|e| !e.is_empty()) || text_has_effects(&rctx, s) {
28                any = true;
29            }
30        });
31    };
32    check(&slide.shapes);
33    if let Some(l) = rctx.layout {
34        check(&l.shapes);
35    }
36    check(&rctx.master.shapes);
37    any
38}

40/// Does the shape's text have a shadow or glow (on a run, in its list style, or in the master's
41/// text styles it inherits)? Such text is blurred, which needs filters.
42fn text_has_effects(rctx: &Ctx, s: &Shape) -> bool {
43    use crate::model::text::{ListStyle, RunProps};
44    let fx = |r: &RunProps| r.shadow.as_ref().is_some_and(|s| !s.inner) || r.glow.as_ref().is_some_and(|g| g.radius > 0.0);
45    let styled = |l: &ListStyle| l.levels.iter().flatten().any(|lv| fx(&lv.run));
46    let Some(t) = &s.text else { return false };
47    t.paragraphs.iter().any(|p| p.runs.iter().any(|r| fx(&r.props)) || fx(&p.end_props))
48        || styled(&t.list_style)
49        || styled(&rctx.master.title_style)
50        || styled(&rctx.master.body_style)
51        || styled(&rctx.master.other_style)
52}

54/// Render slide `index` at `opts.scale` pixels per point. A panic inside the renderer becomes a
55/// blank image (logged), never a crash.
56pub fn render_slide(pres: &Presentation, index: usize, opts: &RenderOpts) -> Image {
57    render_slide_drawn(pres, index, opts).0
58}
59
60/// [`render_slide`], also returning how far the drawing reached, in slide space (at least the
61/// slide; off it, what the shapes, their effects and their text drew, the parts outside the output
62/// included). The editor grows its pasteboard to it ([`pasteboard_region`]).
63pub fn render_slide_drawn(pres: &Presentation, index: usize, opts: &RenderOpts) -> (Image, Rect) {
64    let page = Rect::new(0.0, 0.0, pres.slide_size.width, pres.slide_size.height);
65    let Some(slide) = pres.slides.get(index) else { return (Image::default(), page) };
66    match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
67        RENDERER.with(|r| {
68            let mut r = r.borrow_mut();
69            r.drawn = None;
70            let img = r.slide(pres, slide, index, opts);
71            (img, r.drawn.take())
72        })
73    })) {
74        Ok((img, drawn)) => {
75            let k = opts.scale;
76            let to_slide =
77                |d: Rect| Rect::new((d.x0 - opts.offset.0) / k, (d.y0 - opts.offset.1) / k, (d.x1 - opts.offset.0) / k, (d.y1 - opts.offset.1) / k);
78            let extent = match drawn {
79                Some(d) if k.is_finite() && k > 0.0 && to_slide(d).is_finite() => page.union(to_slide(d)),
80                _ => page,
81            };
82            (img, extent)
83        }
84        Err(_) => {
85            log::error!("rendering slide {index} panicked; showing a blank slide");
86            note_panic();
87            RENDERER.with(|r| *r.borrow_mut() = Renderer::default());
88            let (w, h) = output_size(pres, opts);
89            (Image { width: w as u32, height: h as u32, pixels: vec![255; w as usize * h as usize * 4] }, page)
90        }
91    }
92}

96pub enum SlidePart {

98    Backdrop,
99    /// The slide's own shapes over transparency.

103/// Render one part of slide `index` (transitions that move the contents over the backgrounds).
104/// A panic inside the renderer becomes a transparent image (logged), never a crash.
105pub fn render_slide_part(pres: &Presentation, index: usize, part: SlidePart, opts: &RenderOpts) -> Image {
106    let Some(slide) = pres.slides.get(index) else { return Image::default() };
107    match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| RENDERER.with(|r| r.borrow_mut().slide_part(pres, slide, index, part, opts)))) {
108        Ok(img) => img,
109        Err(_) => {
110            log::error!("rendering part of slide {index} panicked; showing nothing");
111            note_panic();
112            RENDERER.with(|r| *r.borrow_mut() = Renderer::default());
113            let (w, h) = output_size(pres, opts);
114            Image { width: w as u32, height: h as u32, pixels: vec![0; w as usize * h as usize * 4] }
115        }
116    }
117}

119/// Render a frame between slides `old` and `new` (indices into `pres.slides`), e.g. of Morph: the
120/// backdrop (background, master and layout graphics) of `old` cross-fading into that of `new` by
121/// `mix` (0..1), then `shapes` in order, each resolved against the slide it comes from (`true`:
122/// `old`). Pairs of shapes in `text` draw their boxes as usual but their texts morph by words or
123/// characters ([`TextMorph`]), on top of the later shape of the pair. Shapes listed in `paths` draw
124/// an outline morphing from another geometry into their own ([`PathMorph`]). `opts.state` applies
125/// to `shapes` only.
126pub fn render_blend(
127    pres: &Presentation,
128    (old, new): (usize, usize),
129    mix: f64,
130    shapes: &[(Shape, bool)],
131    text: &[TextMorph],
132    paths: &[PathMorph],
133    opts: &RenderOpts,
134) -> Image {
135    let (Some(a), Some(b)) = (pres.slides.get(old), pres.slides.get(new)) else { return Image::default() };
136    match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
137        RENDERER.with(|r| r.borrow_mut().blend(pres, (a, old), (b, new), mix, shapes, text, paths, opts))
138    })) {
139        Ok(img) => img,
140        Err(_) => {
141            log::error!("rendering a frame between slides {old} and {new} panicked; showing the new slide");
142            note_panic();
143            RENDERER.with(|r| *r.borrow_mut() = Renderer::default());
144            render_slide(pres, new, opts)
145        }
146    }
147}

162pub(crate) fn output_size(pres: &Presentation, opts: &RenderOpts) -> (u16, u16) {
163    let (w, h) = opts
164        .size
165        .unwrap_or(((pres.slide_size.width * opts.scale).round().max(1.0) as u32, (pres.slide_size.height * opts.scale).round().max(1.0) as u32));
166    (w.clamp(1, MAX_SIDE) as u16, h.clamp(1, MAX_SIDE) as u16)
167}

169/// Render `ids` together onto a transparent canvas, in their slide positions (copy as a picture).
170pub fn render_shapes(pres: &Presentation, slide: &Slide, ids: &[ShapeId], scale: f64) -> Option<Image> {
171    if ids.is_empty() || !scale.is_finite() || scale <= 0.0 {
172        return None;
173    }
174    let rctx = Ctx::for_slide(pres, slide)?;
175    let mut items = Vec::new();
176    let mut bounds: Option<Rect> = None;
177    for id in ids.iter().take(256) {
178        let Some(s) = slide.shape(*id) else { continue };
179        let b = resolve::xfrm(&rctx, s).bounds();
180        if !b.width().is_finite() || !b.height().is_finite() || b.width() < 0.0 || b.height() < 0.0 {
181            continue;
182        }
183        bounds = Some(match bounds {
184            Some(u) => u.union(b),
185            None => b,
186        });
187        items.push(s);
188    }
189    let b = bounds?.inflate(8.0, 8.0);
190    if !(b.width() > 0.0 && b.height() > 0.0 && b.width().is_finite() && b.height().is_finite()) {
191        return None;
192    }
193    let fit = (2048.0 / b.width().max(b.height())).min(scale);
194    if !(fit.is_finite() && fit > 0.0) {
195        return None;
196    }
197    let w = (b.width() * fit).ceil().clamp(1.0, 2048.0) as u16;
198    let h = (b.height() * fit).ceil().clamp(1.0, 2048.0) as u16;
199    let mut ctx = RenderContext::new_with(w, h, vello_cpu::RenderSettings { num_threads: 0, ..Default::default() });
200    let opts = RenderOpts { scale: fit, ..Default::default() };
201    let f = Frame { pres, opts: &opts, fields: SlideFields::new(1, String::new(), &opts) };
202    let view = Affine::scale(fit) * Affine::translate(-b.origin().to_vec2());
203    RENDERER.with(|r| {
204        let mut r = r.borrow_mut();
205        for s in items {
206            r.shape(&mut ctx, &f, &rctx, s, view, 0);
207        }
208        Some(r.finish(ctx, w, h))
209    })
210}

212/// Render one shape alone onto a transparent canvas (drag previews, copy as picture).
213pub fn render_shape(pres: &Presentation, slide: &Slide, id: ShapeId, scale: f64) -> Option<Image> {
214    let rctx = Ctx::for_slide(pres, slide)?;
215    let s = slide.shape(id)?;
216    let x = resolve::xfrm(&rctx, s);
217    let b = x.bounds().inflate(8.0, 8.0);
218    let w = (b.width() * scale).ceil().clamp(1.0, MAX_SIDE as f64) as u16;
219    let h = (b.height() * scale).ceil().clamp(1.0, MAX_SIDE as f64) as u16;
220    let mut ctx = RenderContext::new_with(w, h, vello_cpu::RenderSettings { num_threads: 0, ..Default::default() });
221    let opts = RenderOpts { scale, ..Default::default() };
222    let f = Frame { pres, opts: &opts, fields: SlideFields::new(1, String::new(), &opts) };
223    let view = Affine::scale(scale) * Affine::translate(-b.origin().to_vec2());
224    RENDERER.with(|r| {
225        let mut r = r.borrow_mut();
226        r.shape(&mut ctx, &f, &rctx, s, view, 0);
227        Some(r.finish(ctx, w, h))
228    })
