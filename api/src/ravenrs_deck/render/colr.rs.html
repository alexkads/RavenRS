1//! Colour glyphs (OpenType `COLR` v0 and v1, colours from `CPAL`), as colour emoji fonts draw
2//! them: layers of glyph outlines filled with solid colours or gradients.
3
4use crate::color::Rgba;
5use crate::fonts::FontFace;
6use kurbo::{Affine, BezPath, Point, Rect};
7use skrifa::color::{Brush, ColorPainter, ColorStop, CompositeMode, Extend, Transform};
8use skrifa::raw::types::BoundingBox;
9use skrifa::{GlyphId, MetadataProvider};
10use vello_cpu::{RenderContext, peniko};
11
12/// Nesting depth of clips and layers past which a glyph opens no more (hostile fonts).
13const MAX_DEPTH: usize = 64;
14/// Fills and gradient stops one colour glyph may paint; the rest of a hostile glyph is skipped.
15const MAX_WORK: usize = 100_000;
16
17/// Paint `gid` of `face` as a colour glyph, its origin mapped by `at` (y down, points) at `scale`
18/// points per font unit. `fg` is the text colour (palette index 0xFFFF) and `alpha` fades it all.
19/// False when the glyph has no colour version, so the caller draws the plain outline.
20pub fn paint(ctx: &mut RenderContext, face: &FontFace, gid: u32, at: Affine, scale: f64, fg: Rgba, alpha: f64) -> bool {
21    let Some(font) = face.skrifa() else { return false };
22    let glyphs = font.color_glyphs();
23    let Some(glyph) = glyphs.get(GlyphId::new(gid)) else { return false };
24    let palette: Vec<Rgba> = font
25        .color_palettes()
26        .get(0)
27        .map(|p| p.colors().iter().map(|c| Rgba { r: c.red, g: c.green, b: c.blue, a: c.alpha }).collect())
28        .unwrap_or_default();
29    // Font units are y up; the slide is y down.
30    let base = at * Affine::scale(scale) * Affine::scale_non_uniform(1.0, -1.0);
31    let mut p = Painter { ctx, face, base, stack: vec![Affine::IDENTITY], palette, fg, opened: Vec::new(), work: 0 };
32    p.ctx.push_layer(None, None, Some(alpha.clamp(0.0, 1.0) as f32), None, None);
33    let ok = glyph.paint(face.location(), &mut p).is_ok();
34    // A malformed glyph may leave layers open: close them all.
35    while let Some(open) = p.opened.pop() {
36        if open {
37            p.ctx.pop_layer();
38        }
39    }
40    p.ctx.pop_layer();
41    ok
42}
43
44struct Painter<'a> {
45    ctx: &'a mut RenderContext,
46    face: &'a FontFace,
47    base: Affine,
48    /// Transforms pushed by the glyph (font units), innermost last.
49    stack: Vec<Affine>,
50    palette: Vec<Rgba>,
51    fg: Rgba,
52    /// Clips and composite groups pushed by the glyph, innermost last: true when a layer was
53    /// opened on the context for it (false past `MAX_DEPTH`, so its pop is skipped too).
54    opened: Vec<bool>,
55    /// Fills plus gradient stops painted so far, against [`MAX_WORK`].
56    work: usize,
57}
58
59fn affine(t: &Transform) -> Affine {
60    Affine::new([t.xx as f64, t.yx as f64, t.xy as f64, t.yy as f64, t.dx as f64, t.dy as f64])
61}
62
63fn pt(p: skrifa::raw::types::Point<f32>) -> Point {
64    Point::new(p.x as f64, p.y as f64)
65}
66
67fn extend(e: Extend) -> peniko::Extend {
68    match e {
69        Extend::Repeat => peniko::Extend::Repeat,
70        Extend::Reflect => peniko::Extend::Reflect,
71        _ => peniko::Extend::Pad,
72    }
73}

75fn blend(m: CompositeMode) -> peniko::BlendMode {
76    use peniko::{Compose, Mix};
77    let (mix, compose) = match m {
78        CompositeMode::Clear => (Mix::Normal, Compose::Clear),
79        CompositeMode::Src => (Mix::Normal, Compose::Copy),
80        CompositeMode::Dest => (Mix::Normal, Compose::Dest),
81        CompositeMode::DestOver => (Mix::Normal, Compose::DestOver),
82        CompositeMode::SrcIn => (Mix::Normal, Compose::SrcIn),
83        CompositeMode::DestIn => (Mix::Normal, Compose::DestIn),
84        CompositeMode::SrcOut => (Mix::Normal, Compose::SrcOut),
85        CompositeMode::DestOut => (Mix::Normal, Compose::DestOut),
86        CompositeMode::SrcAtop => (Mix::Normal, Compose::SrcAtop),
87        CompositeMode::DestAtop => (Mix::Normal, Compose::DestAtop),
88        CompositeMode::Xor => (Mix::Normal, Compose::Xor),
89        CompositeMode::Plus => (Mix::Normal, Compose::Plus),
90        CompositeMode::Screen => (Mix::Screen, Compose::SrcOver),
91        CompositeMode::Overlay => (Mix::Overlay, Compose::SrcOver),
92        CompositeMode::Darken => (Mix::Darken, Compose::SrcOver),
93        CompositeMode::Lighten => (Mix::Lighten, Compose::SrcOver),
94        CompositeMode::ColorDodge => (Mix::ColorDodge, Compose::SrcOver),
95        CompositeMode::ColorBurn => (Mix::ColorBurn, Compose::SrcOver),
96        CompositeMode::HardLight => (Mix::HardLight, Compose::SrcOver),
97        CompositeMode::SoftLight => (Mix::SoftLight, Compose::SrcOver),
98        CompositeMode::Difference => (Mix::Difference, Compose::SrcOver),
99        CompositeMode::Exclusion => (Mix::Exclusion, Compose::SrcOver),
100        CompositeMode::Multiply => (Mix::Multiply, Compose::SrcOver),
101        CompositeMode::HslHue => (Mix::Hue, Compose::SrcOver),
102        CompositeMode::HslSaturation => (Mix::Saturation, Compose::SrcOver),
103        CompositeMode::HslColor => (Mix::Color, Compose::SrcOver),
104        CompositeMode::HslLuminosity => (Mix::Luminosity, Compose::SrcOver),
105        _ => (Mix::Normal, Compose::SrcOver),
106    };
107    peniko::BlendMode::new(mix, compose)
108}

110impl Painter<'_> {
111    fn current(&self) -> Affine {
112        self.base * self.stack.last().copied().unwrap_or(Affine::IDENTITY)
113    }
114
115    fn color(&self, index: u16, alpha: f32) -> peniko::Color {
116        let c = if index == 0xFFFF { self.fg } else { self.palette.get(index as usize).copied().unwrap_or(self.fg) };
117        let a = (c.a as f64 * (alpha as f64).clamp(0.0, 1.0)).round().clamp(0.0, 255.0) as u8;
118        peniko::Color::from_rgba8(c.r, c.g, c.b, a)
119    }
120
121    fn stops(&self, stops: &[ColorStop]) -> Vec<peniko::ColorStop> {
122        stops.iter().map(|s| peniko::ColorStop { offset: s.offset.clamp(0.0, 1.0), color: self.color(s.palette_index, s.alpha).into() }).collect()
123    }
124
125    /// The glyph outline in font units, y up.
126    fn outline(&self, gid: GlyphId) -> BezPath {
127        let p = crate::fonts::FontDb::global().outline(self.face, gid.to_u32());
128        Affine::scale_non_uniform(1.0, -1.0) * (*p).clone()
129    }
130
131    fn push_clip(&mut self, path: &BezPath) {
132        let open = self.opened.len() < MAX_DEPTH;
133        if open {
134            self.ctx.set_transform(self.current());
135            self.ctx.push_clip_layer(path);
136        }
137        self.opened.push(open);
138    }
139
140    fn pop(&mut self) {
141        if self.opened.pop() == Some(true) {
142            self.ctx.pop_layer();
143        }
144    }
145
146    /// Fill the open clip area (a large box in the current transform) with `paint`.
147    fn cover(&mut self, paint: impl Into<vello_cpu::PaintType>) {
148        self.ctx.set_transform(self.current());
149        self.ctx.set_paint(paint);
150        self.ctx.fill_rect(&Rect::new(-1.0e5, -1.0e5, 1.0e5, 1.0e5));
151    }
152}

154impl ColorPainter for Painter<'_> {
155    fn push_transform(&mut self, t: Transform) {
156        let next = self.stack.last().copied().unwrap_or(Affine::IDENTITY) * affine(&t);
157        self.stack.push(next);
158    }
159
160    fn pop_transform(&mut self) {
161        if self.stack.len() > 1 {
162            self.stack.pop();
163        }
164    }
165
166    fn push_clip_glyph(&mut self, gid: GlyphId) {
167        let path = self.outline(gid);
168        self.push_clip(&path);
169    }
170
171    fn push_clip_box(&mut self, b: BoundingBox<f32>) {
172        let r = Rect::new(b.x_min as f64, b.y_min as f64, b.x_max as f64, b.y_max as f64);
173        self.push_clip(&kurbo::Shape::to_path(&r, 0.1));
174    }
175
176    fn pop_clip(&mut self) {
177        self.pop();
178    }
179
180    fn fill(&mut self, brush: Brush<'_>) {
181        let stops = match &brush {
182            Brush::Solid { .. } => 0,
183            Brush::LinearGradient { color_stops, .. } | Brush::RadialGradient { color_stops, .. } | Brush::SweepGradient { color_stops, .. } => {
184                color_stops.len()
185            }
186        };
187        self.work = self.work.saturating_add(1 + stops);
188        if self.work > MAX_WORK {
189            return;
190        }
191        match brush {
192            Brush::Solid { palette_index, alpha } => {
193                let c = self.color(palette_index, alpha);
194                self.cover(c);
195            }
196            Brush::LinearGradient { p0, p1, color_stops, extend: e } => {
197                let g = peniko::Gradient::new_linear(pt(p0), pt(p1)).with_stops(self.stops(color_stops).as_slice()).with_extend(extend(e));
198                self.cover(g);
199            }
200            Brush::RadialGradient { c0, r0, c1, r1, color_stops, extend: e } => {
201                let g = peniko::Gradient::new_two_point_radial(pt(c0), r0.max(0.0), pt(c1), r1.max(0.0))
202                    .with_stops(self.stops(color_stops).as_slice())
203                    .with_extend(extend(e));
204                self.cover(g);
205            }
206            Brush::SweepGradient { c0, start_angle, end_angle, color_stops, extend: e } => {
207                let g = peniko::Gradient::new_sweep(pt(c0), start_angle.to_radians(), end_angle.to_radians())
208                    .with_stops(self.stops(color_stops).as_slice())
209                    .with_extend(extend(e));
210                self.cover(g);
211            }
212        }
213    }
214
215    fn push_layer(&mut self, mode: CompositeMode) {
216        let open = self.opened.len() < MAX_DEPTH;
217        if open {
218            self.ctx.push_blend_layer(blend(mode));
219        }
220        self.opened.push(open);
221    }
222
223    fn pop_layer(&mut self) {
224        self.pop();
225    }
226}
