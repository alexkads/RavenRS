1//! 3-D extrusion (`a:sp3d extrusionH`, `a:extrusionClr`) seen through a turned camera, drawn in
2//! 2-D: quads join the shape's outline on its front face to the same outline on its back face (the
3//! camera's view of the depth axis), and the face is drawn on top as usual.
4//!
5//! Each side (one quad per flattened outline edge) is lit for its own normal by the shape's light
6//! rig and material, and the sides are drawn farthest first. Without a turned camera an extrusion
7//! points at the viewer and nothing of it shows.
8
9use crate::color::{Rgba, SchemeSlot};

12use kurbo::{Affine, BezPath, Vec2};

18/// Outlines with more edges than this (after flattening) are not extruded.
19const MAX_EDGES: usize = 4096;
20
21/// An extrusion: its depth in points and its own colour, if set.
22#[derive(Clone, Debug, PartialEq)]
23pub struct Extrusion {
24    pub depth: f64,
25    pub color: Option<ColorRef>,
26}
27
28fn attr<'a>(tag: &'a str, name: &str) -> Option<&'a str> {
29    let key = format!(" {name}=\"");
30    let s = tag.find(&key)? + key.len();
31    let rest = tag.get(s..)?;
32    rest.get(..rest.find('"')?)
33}
34
35/// The extrusion in 3-D XML kept from the file (`a:sp3d extrusionH="…"` with an `a:scene3d`).
36pub fn from_raw(raw: &str) -> Option<Extrusion> {
37    if !raw.contains("scene3d") {
38        return None;
39    }
40    let i = raw.find("sp3d ")?;
41    let tag = raw.get(i..)?;
42    let tag = tag.get(..tag.find('>')?)?;
43    let depth = attr(tag, "extrusionH")?.parse::<f64>().ok()? / 12700.0;
44    if !(depth.is_finite() && depth > 0.01) {
45        return None;
46    }
47    let color = color_in(raw, "extrusionClr");
48    Some(Extrusion { depth: depth.min(4000.0), color })
49}
50
51/// The colour in the first `<a:{name}>` element of `raw` (an `srgbClr` or a `schemeClr`).
52fn color_in(raw: &str, name: &str) -> Option<ColorRef> {
53    let j = raw.find(&format!("{name}>"))?;
54    let body = raw.get(j..)?;
55    let end = body.find(&format!("/a:{name}")).or_else(|| body.find(&format!("/{name}"))).unwrap_or(body.len());
56    let body = body.get(..end)?;
57    if let Some(k) = body.find("srgbClr ") {
58        let v = attr(body.get(k..)?, "val")?;
59        let n = u32::from_str_radix(v, 16).ok()?;
60        Some(ColorRef::rgb(Rgba::rgb((n >> 16) as u8, (n >> 8) as u8, n as u8)))
61    } else if let Some(k) = body.find("schemeClr ") {
62        SchemeSlot::from_xml(attr(body.get(k..)?, "val")?).map(ColorRef::scheme)
63    } else {
64        None
65    }
66}

68/// A 3-D contour: a band of `width` points around the shape's outline (`a:sp3d contourW`,
69/// `a:contourClr`), drawn behind the face so only its outer half shows past the outline.
70#[derive(Clone, Debug, PartialEq)]
71pub struct Contour {
72    pub width: f64,
73    pub color: Option<ColorRef>,
74}
75
76/// The contour in 3-D XML kept from the file, if it has one.
77pub fn contour_from_raw(raw: &str) -> Option<Contour> {
78    let i = raw.find("sp3d ")?;
79    let tag = raw.get(i..)?;
80    let tag = tag.get(..tag.find('>')?)?;
81    let width = attr(tag, "contourW")?.parse::<f64>().ok()? / 12700.0;
82    if !(width.is_finite() && width > 0.01) {
83        return None;
84    }
85    Some(Contour { width: width.min(2000.0), color: color_in(raw, "contourClr") })
86}

88/// Draw `c` around `outline` (shape space, mapped by `m`), lit as a surface facing the viewer.
89pub fn draw_contour(ctx: &mut RenderContext, outline: &BezPath, m: Affine, width: f64, color: Rgba, lit: &Lighting) {
90    ctx.set_transform(m);
91    ctx.set_paint(crate::render::color(lit.shade(color, [0.0, 0.0, 1.0]), 1.0));
92    ctx.set_stroke(kurbo::Stroke::new(2.0 * width).with_join(kurbo::Join::Round));
93    ctx.stroke_path(outline);
94}

96/// The camera's view of the depth axis: where the back face sits relative to the front face, in
97/// the turned shape's space (the space `camera::affine` maps into).
98pub fn back_offset(c: &Camera, depth: f64, w: f64, h: f64) -> Vec2 {
99    let size = w.max(h);
100    camera::project3(c, 0.0, 0.0, -depth, size) - camera::project3(c, 0.0, 0.0, 0.0, size)
101}
102
103/// The side colour before lighting: the extrusion colour, or the face's.
104pub fn side_color(rctx: &Ctx, e: &Extrusion, face: Option<Rgba>) -> Rgba {
105    e.color.as_ref().map(|c| rctx.color(c, face)).or(face).unwrap_or(Rgba::rgb(0x80, 0x80, 0x80))
106}

108/// Draw the sides of `outline` (shape space, `w` × `h`): one quad per edge between the front face
109/// and the back face, each lit by `lit` for its own normal as camera `cam` sees it, farthest first.
110/// `pre` maps the turned shape's space to the output and `turn` is the camera's turn.
111#[allow(clippy::too_many_arguments)]
112pub fn draw(
113    ctx: &mut RenderContext,
114    outline: &BezPath,
115    w: f64,
116    h: f64,
117    cam: &Camera,
118    depth: f64,
119    pre: Affine,
120    turn: Affine,
121    side: Rgba,
122    lit: &Lighting,
123) {
124    let offset = back_offset(cam, depth, w, h);
125    let px = (pre * offset.to_point() - pre * kurbo::Point::ZERO).hypot();
126    if !px.is_finite() || px < 0.25 {
127        return;
128    }
129    let polys = crate::render::pathgrad::polygons(outline);
130    let size = w.max(h);
131    let mut quads: Vec<(f64, BezPath, Rgba)> = vec![];
132    for poly in &polys {
133        if poly.len() > MAX_EDGES {
134            continue;
135        }
136        let area: f64 = poly.windows(2).map(|q| q[0].x * q[1].y - q[1].x * q[0].y).sum::<f64>() / 2.0;
137        for e in poly.windows(2) {
138            let (a, b) = (e[0], e[1]);
139            let d = b - a;
140            let len = d.hypot();
141            if len < 1e-9 {
142                continue;
143            }
144            // Outward normal in the shape's plane (as in the bevel), turned by the camera.
145            let n = kurbo::Vec2::new(d.y, -d.x) / len;
146            let n = if area > 0.0 { n } else { -n };
147            let vn = camera::rotate(cam, [n.x, n.y, 0.0]);
148            if vn[2] < -0.02 {
149                continue;
150            }
151            let mid = a.midpoint(b);
152            let z = camera::rotate(cam, [mid.x - w / 2.0, mid.y - h / 2.0, -depth / 2.0])[2];
153            let (fa, fb) = (turn * a, turn * b);
154            let mut q = BezPath::new();
155            q.move_to(fa);
156            q.line_to(fb);
157            q.line_to(fb + offset);
158            q.line_to(fa + offset);
159            q.close_path();
160            quads.push((z, q, lit.shade(side, vn)));
161        }
162    }
163    let _ = size;
164    quads.sort_by(|a, b| a.0.total_cmp(&b.0));
165    ctx.set_transform(pre);
166    for (_, q, c) in &quads {
167        ctx.set_paint(crate::render::color(*c, 1.0));
168        ctx.fill_path(q);
169        // Close the hairline seams between neighbouring quads.
170        ctx.set_stroke(kurbo::Stroke::new(0.6 / crate::render::scale_of(pre).max(1e-6)));
171        ctx.stroke_path(q);
172    }
173}

175#[cfg(test)]
176mod tests {
177    use super::*;
178
179    #[test]
180    fn reads_depth_and_colour() {
181        let raw = r#"<a:scene3d><a:camera prst="isometricOffAxis2Left"/><a:lightRig rig="threePt" dir="t"/></a:scene3d><a:sp3d extrusionH="381000"><a:extrusionClr><a:srgbClr val="C00000"/></a:extrusionClr></a:sp3d>"#;
182        let e = from_raw(raw).unwrap();
183        assert!((e.depth - 30.0).abs() < 1e-9);
184        assert_eq!(e.color, Some(ColorRef::rgb(Rgba::rgb(0xC0, 0, 0))));
185        assert!(from_raw(r#"<a:sp3d extrusionH="381000"/>"#).is_none(), "no scene: drawn flat");
186        assert!(from_raw(r#"<a:scene3d/><a:sp3d/>"#).is_none());
187    }
188
189    #[test]
190    fn the_back_face_goes_away_from_the_viewer() {
191        // Turned left about the vertical axis, the back face shows to the right... of the turned
192        // face for a positive longitude, and nothing for a front camera.
193        let c = Camera { lat: 0.0, lon: 30.0, rev: 0.0, dist: 0.0 };
194        let o = back_offset(&c, 20.0, 100.0, 50.0);
195        assert!((o.x.abs() - 10.0).abs() < 1e-9 && o.y.abs() < 1e-9, "{o:?}");
196        let front = Camera { lat: 0.0, lon: 0.0, rev: 0.0, dist: 0.0 };
197        assert!(back_offset(&front, 20.0, 100.0, 50.0).hypot() < 1e-9);
198    }
