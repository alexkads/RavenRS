58/// [`place_slide`] with the render options the slide was drawn with, so date fields take the same
59/// date and time as the raster (`opts.now`).
60pub fn place_slide_with(pres: &Presentation, index: usize, opts: &crate::render::RenderOpts) -> Placed {
61    let mut out = Placed::default();
62    let Some(slide) = pres.slides.get(index) else { return out };
63    let Some(rctx) = Ctx::for_slide(pres, slide) else { return out };
64    let fields = SlideFields::new(pres.slide_number(index), pres.header_footer.footer_text.clone(), opts);
65    let (show_layout, show_master) = resolve::show_master_shapes(slide, rctx.layout);
66    if show_master {
67        let mctx = Ctx::for_master(pres, rctx.master);
68        for s in rctx.master.shapes.iter().filter(|s| s.ph.is_none()) {
69            place(&mut out, &mctx, &fields, s, Affine::IDENTITY, 0, None, false);
70        }
71    }
72    if show_layout && let Some(l) = rctx.layout {
73        let lctx = Ctx::for_layout(pres, rctx.master, l);
74        for s in l.shapes.iter().filter(|s| s.ph.is_none()) {
75            place(&mut out, &lctx, &fields, s, Affine::IDENTITY, 0, None, false);
76        }
77    }
78    for s in &slide.shapes {
79        place(&mut out, &rctx, &fields, s, Affine::IDENTITY, 0, None, true);
80    }
81    out
82}

84/// The header, date, footer and page-number text of a notes or handout `master`, laid out in its
85/// placeholders (master points) with their own formatting and wrapping, for page `n`. The page
86/// number placeholder shows `n` in its first run's formatting.
87pub fn place_master_texts(pres: &Presentation, master: &crate::model::Master, n: usize) -> Vec<PlacedText> {
88    use crate::model::PhType;
89    let mut out = Placed::default();
90    let ctx = Ctx::for_master(pres, master);
91    let fields =
92        SlideFields::new(u32::try_from(n).unwrap_or(u32::MAX), pres.header_footer.footer_text.clone(), &crate::render::RenderOpts::default());
93    for s in master.shapes.iter().filter(|s| matches!(s.ph_type(), Some(PhType::Header | PhType::Date | PhType::Footer | PhType::SlideNum))) {
94        let mut s = s.clone();
95        if s.ph_type() == Some(PhType::SlideNum) {
96            let mut body = s.text.clone().unwrap_or_default();
97            let props =
98                body.paragraphs.first().and_then(|p| p.runs.first().map(|r| r.props.clone()).or(Some(p.end_props.clone()))).unwrap_or_default();
99            let mut para = body.paragraphs.first().cloned().unwrap_or_default();
100            para.runs = vec![crate::model::text::Run::with(n.to_string(), props)];
101            body.paragraphs = vec![para];
102            s.text = Some(body);
103        }
104        place(&mut out, &ctx, &fields, &s, Affine::IDENTITY, 0, None, false);
105    }
106    out.texts
107}

109/// `group` is the index in `out.shapes` of the enclosing group; `own` is true for the slide's own
110/// shapes (recorded in `out.shapes`), false for the layout's and master's.
111#[allow(clippy::too_many_arguments)]
112fn place(out: &mut Placed, rctx: &Ctx, fields: &SlideFields, s: &Shape, parent: Affine, depth: usize, group: Option<usize>, own: bool) {
113    if depth > 32 || s.hidden {
114        return;
115    }
116    let x = resolve::xfrm(rctx, s);
117    if !x.is_finite() {
118        return;
119    }
120    let m = parent * x.affine();
121    let rect = m.transform_rect_bbox(Rect::new(0.0, 0.0, x.w, x.h));
122    if let Some(link) = &s.click {
123        out.links.push(PlacedLink { rect, link: link.clone(), shape: s.id, text: false });
124    }
125    let me = if own {
126        out.shapes.push(PlacedShape { id: s.id, parent: group, is_group: matches!(s.kind, ShapeKind::Group { .. }), rect, link: s.click.clone() });
127        Some(out.shapes.len() - 1)
128    } else {
129        None
130    };
131    match &s.kind {
132        ShapeKind::Group { children, child } => {
133            let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
134            let gm = parent * crate::geom::group_child_affine(&x, ch);
135            for c in children {
136                place(out, rctx, fields, c, gm, depth + 1, me, own);
137            }
138        }
139        ShapeKind::Table(t) => {
140            // Cell text, laid out exactly as the table draws it.
141            // The table's style is looked up once, not once per cell.
142            let def = table::resolve_style(rctx, t);
143            for (r, c, rect) in table::cell_rects(t, &table::row_heights(rctx, t, fields)) {
144                let Some(cell) = t.cell(r, c).filter(|cell| !cell.text.is_empty()) else { continue };
145                let body = table::cell_body_with_def(rctx, def, t, r, c, cell);
146                let tmp = Shape::table_cell(body.clone());
147                place_text(out, rctx, fields, s.id, &tmp, &body, rect, m);
148            }
149        }
150        ShapeKind::Chart(_) | ShapeKind::Ink { .. } => {}
151        _ => {
152            let Some(body) = s.text.as_ref().filter(|b| !b.is_empty()) else { return };
153            let tr = shape_geometry(s, x.w, x.h).text_rect;
154            place_text(out, rctx, fields, s.id, s, body, tr, m);
155        }
156    }
157}

159/// Lay out `body` in text rect `tr` of shape `s` (shape-local → slide `m`) and record it and its hyperlinks.
160#[allow(clippy::too_many_arguments)]
161fn place_text(out: &mut Placed, rctx: &Ctx, fields: &SlideFields, owner: ShapeId, s: &Shape, body: &TextBody, tr: Rect, m: Affine) {
162    let layout = crate::text::layout(rctx, s, body, &Opts { rect: tr, fields, prompt_color: None, no_shrink: false });
163    let m = crate::render::text_transform(m, tr, layout.rotation);
164    // Field runs carry their saved text (`‹#›`); the layout drew their values, so the body that
165    // goes with it (character offsets, text layer) must hold the same values.
166    let mut placed = body.clone();
167    for (pi, para) in body.paragraphs.iter().enumerate() {
168        for (ri, run) in para.runs.iter().enumerate() {
169            let RunKind::Field { field } = &run.kind else { continue };
170            let lang = resolve::run(rctx, s, para, &run.props).lang;
171            if let Some(v) = fields.field_in(field, lang.as_deref())
172                && let Some(out_run) = placed.paragraphs.get_mut(pi).and_then(|p| p.runs.get_mut(ri))
173            {
174                out_run.text = v;
175                out_run.kind = RunKind::Text;
176            }
177        }
178    }
179    // Text hyperlinks: one box per line segment of a linked run.
180    for run in layout.runs.iter().filter(|r| r.link && !r.glyphs.is_empty()) {
181        let Some(link) = run_link(&placed, run.para, run.chars.0) else { continue };
182        let Some(line) = layout.lines.iter().find(|l| l.para == run.para && l.start <= run.chars.0 && run.chars.0 <= l.end) else { continue };
183        // The run's visual extent (right-to-left runs run from right to left).
184        let first = run.glyphs.first().map(|g| g.1).unwrap_or(0.0);
185        let (x0, x1) = line.span(run.chars.0, run.chars.1).unwrap_or((first, first));
186        let r = Rect::new(x0, line.top, x1, line.bottom);
187        out.links.push(PlacedLink { rect: m.transform_rect_bbox(r), link, shape: owner, text: true });
188    }
189    out.texts.push(PlacedText { layout, transform: m, body: placed });
190}

192/// The hyperlink of the model run containing character `at` of paragraph `para`.
193fn run_link(body: &TextBody, para: usize, at: usize) -> Option<Hyperlink> {
194    let p = body.paragraphs.get(para)?;
195    let mut pos = 0;
196    for r in &p.runs {
197        let n = r.char_len();
198        if at < pos + n.max(1) {
199            return r.props.link.clone();
200        }
201        pos += n;
202    }
203    None
204}

206/// The outlines of shape `s`'s text as drawn, in the space `m` maps shape-local points to, and
207/// the colour of its first visible run: what Merge Shapes turns letters into. Empty when the
208/// shape has no text.
209pub fn text_outline(rctx: &Ctx, s: &Shape, m: Affine) -> (kurbo::BezPath, Option<crate::color::Rgba>) {
210    let mut out = kurbo::BezPath::new();
211    let Some(body) = s.text.as_ref().filter(|b| !b.is_empty()) else { return (out, None) };
212    let x = resolve::xfrm(rctx, s);
213    let tr = shape_geometry(s, x.w, x.h).text_rect;
214    let layout = crate::text::layout(rctx, s, body, &Opts { rect: tr, fields: &crate::text::NoFields, prompt_color: None, no_shrink: false });
215    let m = crate::render::text_transform(m, tr, layout.rotation);
216    let mut first = None;
217    for run in &layout.runs {
218        let k = run.size / run.face.upem.max(1.0);
219        let skew = if run.fake_italic { Affine::new([1.0, 0.0, -0.2, 1.0, 0.0, 0.0]) } else { Affine::IDENTITY };
220        for (gid, gx, gy) in &run.glyphs {
221            let path = crate::render::glyph(&run.face, *gid);
222            if path.elements().is_empty() {
223                continue;
224            }
225            if first.is_none() && run.color.a > 0 {
226                first = Some(run.color);
227            }
228            let at = m * crate::render::glyph_place(run, *gid, *gx, *gy) * skew * Affine::scale(k);
229            out.extend(path.elements().iter().map(|e| at * *e));
230        }
231    }
232    (out, first)
