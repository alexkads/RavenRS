4/// Stroke `path` with `line`. A gradient line is laid out over `extent` (the untrimmed outline the
5/// line belongs to; `path` itself when `None`).
6#[allow(clippy::too_many_arguments)]
7pub(crate) fn stroke(
8    ctx: &mut RenderContext,
9    rctx: &Ctx,
10    line: &Line,
11    ph: Option<Rgba>,
12    path: &BezPath,
13    extent: Option<&BezPath>,
14    m: Affine,
15    lw: f64,
16    alpha: f64,
17) {
18    use crate::model::style::{LineCap, LineJoin};
19    let cap = match line.cap.unwrap_or_default() {
20        LineCap::Flat => kurbo::Cap::Butt,
21        LineCap::Round => kurbo::Cap::Round,
22        LineCap::Square => kurbo::Cap::Square,
23    };
24    let join = match line.join.unwrap_or_default() {
25        LineJoin::Round => kurbo::Join::Round,
26        LineJoin::Bevel => kurbo::Join::Bevel,
27        LineJoin::Miter => kurbo::Join::Miter,
28    };
29    // Hairlines stay visible at any zoom.
30    let min_w = 0.5 / scale_of(m).max(1e-6);
31    let sw = lw.max(min_w);
32    // PowerPoint's default miter limit is 8 widths (`a:miter lim="800000"`).
33    let mut sk = kurbo::Stroke::new(sw).with_caps(cap).with_join(join).with_miter_limit(8.0);
34    // Square caps on a dashed line square off only the line's own two ends (where a dash is
35    // drawn): each dash itself ends flat, as PowerPoint draws them.
36    let mut square_ends = None;
37    if let Some(d) = &line.dash {
38        let pat: Vec<f64> = d.pattern().iter().map(|v| v * lw.max(0.75)).collect();
39        let pat = dash_with_caps(pat, cap, sw);
40        let pat = limit_dash_count(path, pat);
41        if !pat.is_empty() && pat.iter().all(|v| *v > 0.0) {
42            if cap == kurbo::Cap::Square {
43                sk = sk.with_caps(kurbo::Cap::Butt);
44                square_ends = Some(square_dash_ends(path, &pat, sw));
45            }
46            sk = sk.with_dashes(0.0, pat);
47        }
48    }
49    ctx.set_transform(m);
50    match &line.fill {
51        Some(Fill::Solid { color: c }) => {
52            ctx.set_paint(color(rctx.color(c, ph), alpha));
53        }
54        Some(Fill::Gradient(g)) if !g.stops.is_empty() => ctx.set_paint(line_gradient(rctx, g, ph, extent.unwrap_or(path), lw, alpha)),
55        _ => ctx.set_paint(color(Rgba::BLACK, alpha)),
56    }
57    // Compound lines: each band stroked along the path moved to its place across the width.
58    let compound = line.compound.unwrap_or_default();
59    if compound != crate::model::style::Compound::Single && lw * scale_of(m) >= 1.5 {
60        let tolerance = 0.1 / scale_of(m).max(1e-6);
61        for &(a, b) in compound::bands(compound) {
62            let d = (-0.5 + (a + b) / 2.0) * lw;
63            let mut band = sk.clone();
64            band.width = (b - a) * lw;
65            ctx.set_stroke(band);
66            ctx.stroke_path(&compound::offset(path, d, tolerance));
67        }
68        return;
69    }
70    ctx.set_stroke(sk);
71    ctx.stroke_path(path);
72    if let Some(ends) = square_ends {
73        ctx.fill_path(&ends);
74    }
75}

77/// The square caps of a dashed line drawn `width` wide with dash `pattern`: a half-width square
78/// past the start of each open sub-path (every dash pattern starts with a dash) and past its end
79/// when the pattern is in a dash there.
80fn square_dash_ends(path: &BezPath, pattern: &[f64], width: f64) -> BezPath {
81    use kurbo::{ParamCurve, ParamCurveArclen};
82    let mut out = BezPath::new();
83    let period: f64 = pattern.iter().sum();
84    let mut cap = |at: Point, dir: Vec2| {
85        let n = Vec2::new(-dir.y, dir.x) * (width / 2.0);
86        let back = dir * (width / 2.0);
87        out.move_to(at + n);
88        out.line_to(at + n + back);
89        out.line_to(at - n + back);
90        out.line_to(at - n);
91        out.close_path();
92    };
93    let dir = |seg: &kurbo::PathSeg, t0: f64, t1: f64| {
94        let d = seg.eval(t1) - seg.eval(t0);
95        let d = if d.hypot() > 1e-9 { d } else { seg.end() - seg.start() };
96        if d.hypot() > 1e-9 { Some(d.normalize()) } else { None }
97    };
98    let mut subs: Vec<(Vec<kurbo::PathSeg>, bool)> = vec![];
99    let mut cur: Vec<kurbo::PathSeg> = vec![];
100    let mut closed = false;
101    let mut last = Point::ZERO;
102    let mut start = Point::ZERO;
103    for el in path.elements() {
104        let seg = match *el {
105            PathEl::MoveTo(p) => {
106                if !cur.is_empty() {
107                    subs.push((std::mem::take(&mut cur), closed));
108                }
109                closed = false;
110                last = p;
111                start = p;
112                continue;
113            }
114            PathEl::LineTo(p) => kurbo::PathSeg::Line(kurbo::Line::new(last, p)),
115            PathEl::QuadTo(a, p) => kurbo::PathSeg::Quad(kurbo::QuadBez::new(last, a, p)),
116            PathEl::CurveTo(a, b, p) => kurbo::PathSeg::Cubic(kurbo::CubicBez::new(last, a, b, p)),
117            PathEl::ClosePath => {
118                closed = true;
119                last = start;
120                continue;
121            }
122        };
123        last = seg.end();
124        cur.push(seg);
125    }
126    if !cur.is_empty() {
127        subs.push((cur, closed));
128    }
129    for segs in subs.iter().filter(|(s, c)| !c && !s.is_empty()).map(|(s, _)| s) {
130        let (Some(first), Some(end)) = (segs.first(), segs.last()) else { continue };
131        if let Some(d) = dir(first, 1e-3, 0.0) {
132            cap(first.start(), d);
133        }
134        let len: f64 = segs.iter().map(|s| s.arclen(1e-3)).sum();
135        if !(period > 0.0 && len.is_finite()) {
136            continue;
137        }
138        let mut pos = len % period;
139        // Ending exactly on a period boundary is the end of a gap.
140        let mut on = pos > period * 1e-6;
141        for v in pattern.iter().cycle().take(pattern.len()) {
142            if pos <= *v {
143                break;
144            }
145            pos -= v;
146            on = !on;
147        }
148        if on && let Some(d) = dir(end, 1.0 - 1e-3, 1.0) {
149            cap(end.end(), d);
150        }
151    }
152    out
153}

155/// A gradient line's paint: laid out, as PowerPoint lays it, over the box the stroke covers (the
156/// path's bounds grown by half the line width), so a horizontal line's vertical gradient runs
157/// across its width.
158fn line_gradient(rctx: &Ctx, g: &crate::model::style::Gradient, ph: Option<Rgba>, path: &BezPath, lw: f64, alpha: f64) -> peniko::Gradient {
159    let bounds = path.bounding_box().inflate(lw / 2.0, lw / 2.0);
160    let bounds = paint::tile_box(bounds, g.tile);
161    let stops: Vec<(f64, Rgba)> = g.stops.iter().map(|s| (s.pos, rctx.color(&s.color, ph))).collect();
162    paint::gradient(&g.shape, stops, bounds, alpha)
163}

165/// A dash pattern (on, off, on, off… lengths) for a line drawn `width` wide with `cap`. As
166/// PowerPoint draws them, round caps sit inside each dash's length rather than adding to it: a
167/// "round dot" (`sysDot`, one width on and one off) is a row of separate circles, not a solid
168/// line. So each dash loses a width (half at each end, where the caps go) and each gap gains it; a
169/// dash no longer than the width becomes just its two caps. (Square-capped dashes end flat; see
170/// `square_dash_ends`.)
171pub(crate) fn dash_with_caps(mut pat: Vec<f64>, cap: kurbo::Cap, width: f64) -> Vec<f64> {
172    if cap != kurbo::Cap::Round || !pat.len().is_multiple_of(2) || !(width.is_finite() && width > 0.0) {
173        return pat;
174    }
175    for (i, v) in pat.iter_mut().enumerate() {
176        *v = if i.is_multiple_of(2) { (*v - width).max(width * 1e-3) } else { *v + width };
177    }
178    pat
179}

181/// The most dashes one stroke is cut into. Stroking expands every dash into its own outline, so a
182/// tiny dash on a huge shape (a 1 EMU-wide `custDash` on a 2147483647 EMU rectangle) would
183/// otherwise need gigabytes and abort the app when the slide is drawn. Past this count the
184/// pattern is scaled up evenly, which looks the same: the dashes are far below a pixel by then.
185pub(crate) const MAX_DASHES: f64 = 20_000.0;
186
187pub(crate) fn limit_dash_count(path: &BezPath, pat: Vec<f64>) -> Vec<f64> {
188    use kurbo::ParamCurveArclen;
189    let period: f64 = pat.iter().sum();
190    if !(period.is_finite() && period > 0.0) {
191        return pat;
192    }
193    let len: f64 = path.segments().map(|s| s.arclen(1e-2)).sum();
194    // Every dash of the pattern counts, not each repeat of it: a custom pattern can hold many.
195    let dashes = len / period * pat.len().div_ceil(2) as f64;
196    if !len.is_finite() {
197        return vec![];
198    }
199    if dashes <= MAX_DASHES {
200        return pat;
201    }
202    let k = dashes / MAX_DASHES;
203    pat.into_iter().map(|v| v * k).collect()
204}

206fn end_points(path: &BezPath) -> Option<((Point, Point), (Point, Point))> {
207    let els: Vec<PathEl> = path.elements().to_vec();
208    let mut pts: Vec<Point> = vec![];
209    for e in &els {
210        match e {
211            PathEl::MoveTo(p) | PathEl::LineTo(p) => pts.push(*p),
212            PathEl::QuadTo(a, b) => {
213                pts.push(*a);
214                pts.push(*b);
215            }
216            PathEl::CurveTo(a, b, c) => {
217                pts.push(*a);
218                pts.push(*b);
219                pts.push(*c);
220            }
221            PathEl::ClosePath => {}
222        }
223    }
224    if pts.len() < 2 {
225        return None;
226    }
227    let first = *pts.first()?;
228    let second = pts.iter().skip(1).find(|p| (**p - first).hypot() > 1e-6).copied().unwrap_or(*pts.get(1)?);
229    let last = *pts.last()?;
230    let prev = pts.iter().rev().skip(1).find(|p| (**p - last).hypot() > 1e-6).copied().unwrap_or(*pts.get(pts.len() - 2)?);
231    Some(((first, second), (last, prev)))
232}

234/// An arrowhead's width and length for a line `lw` wide. PowerPoint sizes them in multiples of
235/// the line width (small 2, medium 3, large 5), but never smaller than for a 2 pt line.
236pub(crate) fn head_size(e: &crate::model::style::LineEnd, lw: f64) -> (f64, f64) {
237    let sz = |s: &str| match s {
238        "sm" => 2.0,
239        "lg" => 5.0,
240        _ => 3.0,
241    };
242    let k = lw.max(2.0);
243    (sz(&e.w) * k, sz(&e.len) * k)
244}

246/// How far back from the tip an open arrowhead's legs (`lw` wide) meet along their centre lines:
247/// PowerPoint puts the outside of the point, not the legs' centre lines, on the line's end.
248fn open_arrow_setback(wl: f64, ll: f64, lw: f64) -> f64 {
249    let half = (wl / 2.0).atan2(ll);
250    if half.sin() > 1e-6 { (lw / 2.0 / half.sin()).min(ll) } else { 0.0 }
251}

253/// How much of a `lw`-wide line to leave off at its start and end, under its arrowheads: up to
254/// a triangle's base, into a stealth arrow's notch far enough for its barbs to cover the line's
255/// corners, and to where an open arrow's legs meet. Ovals and diamonds sit on the end itself.
256pub(crate) fn head_trims(line: &Line, lw: f64) -> (f64, f64) {
257    // A square cap reaches half a width past the line's end: leave that off too.
258    let cap = if line.cap == Some(crate::model::style::LineCap::Square) { lw / 2.0 } else { 0.0 };
259    let trim = |e: &Option<crate::model::style::LineEnd>| {
260        let Some(e) = e else { return 0.0 };
261        let (wl, ll) = head_size(e, lw);
262        match e.kind.as_str() {
263            "triangle" => ll + cap,
264            "stealth" => (ll * (0.6 + 0.4 * lw / wl.max(1e-9))).min(ll) + cap,
265            "arrow" => open_arrow_setback(wl, ll, lw.max(0.75)) + cap,
266            _ => 0.0,
267        }
268    };
269    (trim(&line.head), trim(&line.tail))
270}

272/// `path` with `start` left off the beginning of its first sub-path and `end` off the end of its
273/// last (by arc length), where those are open. A sub-path shorter than its trims disappears.
274pub(crate) fn trim_ends(path: &BezPath, start: f64, end: f64) -> BezPath {
275    use kurbo::{ParamCurve, ParamCurveArclen};
276    const ACC: f64 = 1e-3;
277    let mut subs: Vec<(Vec<kurbo::PathSeg>, bool)> = vec![];
278    let (mut last, mut first) = (Point::ZERO, Point::ZERO);
279    for el in path.elements() {
280        let seg = match *el {
281            PathEl::MoveTo(p) => {
282                subs.push((vec![], false));
283                last = p;
284                first = p;
285                continue;
286            }
287            PathEl::ClosePath => {
288                if let Some(s) = subs.last_mut() {
289                    s.1 = true;
290                }
291                last = first;
292                continue;
293            }
294            PathEl::LineTo(p) => kurbo::PathSeg::Line(kurbo::Line::new(last, p)),
295            PathEl::QuadTo(a, p) => kurbo::PathSeg::Quad(kurbo::QuadBez::new(last, a, p)),
296            PathEl::CurveTo(a, b, p) => kurbo::PathSeg::Cubic(kurbo::CubicBez::new(last, a, b, p)),
297        };
298        last = seg.end();
299        if subs.is_empty() {
300            subs.push((vec![], false));
301        }
302        if let Some(s) = subs.last_mut() {
303            s.0.push(seg);
304        }
305    }
306    if let Some((segs, false)) = subs.first_mut()
307        && start > 0.0
308    {
309        let mut left = start;
310        let mut drop_count = 0;
311        for (i, seg) in segs.iter().enumerate() {
312            let l = seg.arclen(ACC);
313            if l > left {
314                let partial = seg.subsegment(seg.inv_arclen(left, ACC)..1.0);
315                segs.drain(..i);
316                segs[0] = partial;
317                drop_count = 0;
318                break;
319            }
320            left -= l;
321            drop_count = i + 1;
322        }
323        if drop_count > 0 {
324            segs.drain(..drop_count);
325        }
326    }
327    if let Some((segs, false)) = subs.last_mut()
328        && end > 0.0
329    {
330        let mut left = end;
331        while let Some(seg) = segs.last().copied() {
332            let l = seg.arclen(ACC);
333            if l > left {
334                if let Some(s1) = segs.last_mut() {
335                    *s1 = seg.subsegment(0.0..seg.inv_arclen(l - left, ACC));
336                }
337                break;
338            }
339            left -= l;
340            segs.pop();
341        }
342    }
343    let mut out = BezPath::new();
344    for (segs, closed) in subs {
345        let Some(s0) = segs.first() else { continue };
346        out.move_to(s0.start());
347        for seg in &segs {
348            match seg {
349                kurbo::PathSeg::Line(l) => out.line_to(l.p1),
350                kurbo::PathSeg::Quad(q) => out.quad_to(q.p1, q.p2),
351                kurbo::PathSeg::Cubic(c) => out.curve_to(c.p1, c.p2, c.p3),
352            }
353        }
354        if closed {
355            out.close_path();
356        }
357    }
358    out
359}

362pub(crate) fn arrowheads(ctx: &mut RenderContext, rctx: &Ctx, line: &Line, ph: Option<Rgba>, geo: &Geometry, rel: Affine, m: Affine, lw: f64) {
363    let Some(sp) = geo.paths.iter().find(|p| p.stroke) else { return };
364    let Some(((a, a2), (b, b2))) = end_points(&sp.path) else { return };
365    // The outline's ends in the space the arrowheads are sized and drawn in.
366    let (a, a2, b, b2) = (rel * a, rel * a2, rel * b, rel * b2);
367    let col = match &line.fill {
368        Some(Fill::Solid { color: c }) => rctx.color(c, ph),
369        _ => Rgba::BLACK,
370    };
371    // A gradient line's heads take the gradient where they are.
372    let grad = match &line.fill {
373        Some(Fill::Gradient(g)) if !g.stops.is_empty() => {
374            let mut moved = sp.path.clone();
375            moved.apply_affine(rel);
376            Some(line_gradient(rctx, g, ph, &moved, lw, 1.0))
377        }
378        _ => None,
379    };
380    let set_paint = |ctx: &mut RenderContext| match &grad {
381        Some(g) => ctx.set_paint(g.clone()),
382        None => ctx.set_paint(color(col, 1.0)),
383    };
384    for (end, tip, from) in [(&line.head, a, a2), (&line.tail, b, b2)] {
385        let Some(e) = end else { continue };
386        if e.kind == "none" {
387            continue;
388        }
389        let (wl, ll) = head_size(e, lw);
390        let dir = (tip - from).normalize();
391        if !dir.x.is_finite() {
392            continue;
393        }
394        let n = Vec2::new(-dir.y, dir.x);
395        let base = tip - dir * ll;
396        let mut p = BezPath::new();
397        match e.kind.as_str() {
398            "oval" => {
399                let c = kurbo::Ellipse::new(tip, (ll / 2.0, wl / 2.0), dir.y.atan2(dir.x));
400                p = c.to_path(0.05);
401            }
402            "diamond" => {
403                p.move_to(tip + dir * ll / 2.0);
404                p.line_to(tip + n * wl / 2.0);
405                p.line_to(tip - dir * ll / 2.0);
406                p.line_to(tip - n * wl / 2.0);
407                p.close_path();
408            }
409            "stealth" => {
410                p.move_to(tip);
411                p.line_to(base + n * wl / 2.0);
412                p.line_to(tip - dir * ll * 0.6);
413                p.line_to(base - n * wl / 2.0);
414                p.close_path();
415            }
416            "arrow" => {
417                // Legs the line's width with round ends, meeting in a sharp point whose outside is
418                // on the line's end.
419                let leg = lw.max(0.75);
420                let back = dir * open_arrow_setback(wl, ll, leg);
421                let mut o = BezPath::new();
422                o.move_to(base + n * wl / 2.0 - back);
423                o.line_to(tip - back);
424                o.line_to(base - n * wl / 2.0 - back);
425                ctx.set_transform(m);
426                set_paint(ctx);
427                ctx.set_stroke(kurbo::Stroke::new(leg).with_join(kurbo::Join::Miter).with_miter_limit(100.0).with_caps(kurbo::Cap::Round));
428                ctx.stroke_path(&o);
429                continue;
430            }
431            _ => {
432                p.move_to(tip);
433                p.line_to(base + n * wl / 2.0);
434                p.line_to(base - n * wl / 2.0);
435                p.close_path();
436            }
437        }
438        ctx.set_transform(m);
439        set_paint(ctx);
440        ctx.fill_path(&p);
441    }
