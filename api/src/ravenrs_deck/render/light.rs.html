1//! Light rigs (`a:lightRig rig= dir=`) and materials (`a:sp3d prstMaterial=`) for 3-D shapes:
2//! how bright a surface is for its normal.
3//!
4//! The three-point rig with the matte material is fitted to PowerPoint's exports (see
5//! [`crate::render::bevel`]): `c · (0.453 + 0.731 · max(0, n · L)) + 14.7` with the key light from the upper
6//! left. The other rigs and materials are modelled on it, not measured: each rig is a set of
7//! directional lights with an ambient level and a colour tint, each material scales the diffuse
8//! and ambient terms and adds a specular highlight.
9
10use crate::color::Rgba;
11
12/// A light rig and material: what lights a 3-D surface.
13#[derive(Clone, Debug, PartialEq)]
14pub struct Lighting {
15    /// Unit directions from the surface towards each light (x right, y down, z to the viewer)
16    /// and their strengths.
17    pub lights: Vec<([f64; 3], f64)>,
18    pub ambient: f64,
19    pub diffuse: f64,
20    /// Added to every channel (the measured lift of the three-point rig).
21    pub lift: f64,
22    /// Colour of the light, per channel.
23    pub tint: [f64; 3],
24    pub specular: f64,
25    pub shininess: f64,
26    /// `flat`: no shading at all.
27    pub flat: bool,
28}
29
30fn dir(az: f64, el: f64) -> [f64; 3] {
31    let (az, el) = (az.to_radians(), el.to_radians());
32    [el.cos() * az.cos(), el.cos() * az.sin(), el.sin()]
33}

35fn value_after<'a>(raw: &'a str, key: &str) -> Option<&'a str> {
36    let s = raw.find(key)? + key.len();
37    let rest = raw.get(s..)?;
38    rest.get(..rest.find('"')?)
39}

41/// The rig's lights as (azimuth, elevation, strength), its ambient level and tint.
42fn rig(name: &str) -> (Vec<(f64, f64, f64)>, f64, [f64; 3]) {
43    const WHITE: [f64; 3] = [1.0, 1.0, 1.0];
44    match name {
45        "balanced" => (vec![(240.0, 40.0, 0.75), (300.0, 40.0, 0.35), (90.0, 30.0, 0.25)], 0.47, WHITE),
46        "soft" => (vec![(270.0, 60.0, 0.6)], 0.58, WHITE),
47        "harsh" => (vec![(225.0, 30.0, 1.15)], 0.3, WHITE),
48        "flood" => (vec![(270.0, 70.0, 0.8), (240.0, 40.0, 0.35)], 0.5, WHITE),
49        "contrasting" => (vec![(210.0, 30.0, 1.05), (30.0, 20.0, 0.3)], 0.33, WHITE),
50        "morning" => (vec![(240.0, 35.0, 0.9)], 0.45, [1.0, 0.95, 0.85]),
51        "sunrise" => (vec![(200.0, 20.0, 0.95)], 0.42, [1.0, 0.88, 0.75]),
52        "sunset" => (vec![(340.0, 20.0, 0.95)], 0.42, [1.0, 0.82, 0.7]),
53        "chilly" => (vec![(240.0, 45.0, 0.85)], 0.45, [0.88, 0.94, 1.0]),
54        "freezing" => (vec![(240.0, 45.0, 0.85)], 0.45, [0.8, 0.9, 1.0]),
55        "twoPt" => (vec![(225.0, 45.0, 0.8), (315.0, 45.0, 0.45)], 0.42, WHITE),
56        "glow" => (vec![(270.0, 85.0, 0.55)], 0.68, WHITE),
57        "brightRoom" => (vec![(270.0, 60.0, 0.45)], 0.75, WHITE),
58        // threePt and the legacy rigs.
59        _ => (vec![(240.0, 45.0, 1.0)], 0.453, WHITE),
60    }
61}

63/// The rig direction (`dir`) turns the lights about the view axis.
64fn dir_turn(d: &str) -> f64 {
65    match d {
66        "tr" => 45.0,
67        "r" => 90.0,
68        "br" => 135.0,
69        "b" => 180.0,
70        "bl" => 225.0,
71        "l" => 270.0,
72        "tl" => 315.0,
73        _ => 0.0,
74    }
75}

77impl Lighting {
78    /// The lighting of 3-D XML kept from the file; the three-point rig and warm matte material
79    /// when unset.
80    pub fn from_raw(raw: &str) -> Lighting {
81        let rig_name = value_after(raw, "rig=\"").unwrap_or("threePt");
82        let turn = dir_turn(value_after(raw, " dir=\"").unwrap_or("t"));
83        let material = value_after(raw, "prstMaterial=\"").unwrap_or("warmMatte");
84        let (lights, ambient, tint) = rig(rig_name);
85        let (dk, ak, specular, shininess) = match material {
86            "plastic" => (1.0, 1.0, 0.45, 24.0),
87            "metal" => (0.8, 0.9, 0.7, 40.0),
88            "softmetal" => (0.9, 0.95, 0.35, 12.0),
89            "powder" | "translucentPowder" => (0.85, 1.12, 0.0, 1.0),
90            "dkEdge" => (1.12, 0.8, 0.1, 8.0),
91            "softEdge" => (0.9, 1.08, 0.05, 6.0),
92            "clear" => (0.9, 1.0, 0.6, 50.0),
93            _ => (1.0, 1.0, 0.0, 1.0),
94        };
95        Lighting {
96            lights: lights.into_iter().map(|(az, el, s)| (dir(az + turn, el), s)).collect(),
97            ambient: ambient * ak,
98            diffuse: 0.731 * dk,
99            lift: 14.7,
100            tint,
101            specular,
102            shininess,
103            flat: rig_name == "flat" || material == "flat",
104        }
105    }
106
107    /// The diffuse brightness factor for unit normal `n` (the three-point matte rig: 0.453 +
108    /// 0.731 · max(0, n · L)).
109    pub fn factor(&self, n: [f64; 3]) -> f64 {
110        if self.flat {
111            return 1.0;
112        }
113        let dot = |l: &[f64; 3]| (n[0] * l[0] + n[1] * l[1] + n[2] * l[2]).max(0.0);
114        self.ambient + self.diffuse * self.lights.iter().map(|(l, s)| s * dot(l)).sum::<f64>()
115    }
116
117    /// The specular highlight (0–1) for unit normal `n`, seen from the front.
118    pub fn highlight(&self, n: [f64; 3]) -> f64 {
119        if self.flat || self.specular <= 0.0 {
120            return 0.0;
121        }
122        self.lights
123            .iter()
124            .map(|(l, s)| {
125                let h = [l[0], l[1], l[2] + 1.0];
126                let len = (h[0] * h[0] + h[1] * h[1] + h[2] * h[2]).sqrt().max(1e-9);
127                let d = ((n[0] * h[0] + n[1] * h[1] + n[2] * h[2]) / len).max(0.0);
128                s * d.powf(self.shininess)
129            })
130            .sum::<f64>()
131            * self.specular
132    }
133
134    /// Colour `c` lit on a surface with unit normal `n`.
135    pub fn shade(&self, c: Rgba, n: [f64; 3]) -> Rgba {
136        if self.flat {
137            return c;
138        }
139        let f = self.factor(n);
140        let hi = self.highlight(n) * 255.0;
141        let ch = |v: u8, t: f64| (f64::from(v) * f * t + self.lift + hi).round().clamp(0.0, 255.0) as u8;
142        Rgba::rgba(ch(c.r, self.tint[0]), ch(c.g, self.tint[1]), ch(c.b, self.tint[2]), c.a)
143    }
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149
150    #[test]
151    fn three_point_matte_is_the_measured_formula() {
152        let l = Lighting::from_raw(r#"<a:lightRig rig="threePt" dir="t"/>"#);
153        let n = [0.0, 0.0, 1.0];
154        let expect = 0.453 + 0.731 * 45f64.to_radians().sin();
155        assert!((l.factor(n) - expect).abs() < 1e-9);
156        assert_eq!(l.highlight(n), 0.0);
157        let c = l.shade(Rgba::rgb(100, 100, 100), n);
158        assert_eq!(c.r, (100.0 * expect + 14.7_f64).round() as u8);
159    }
160
161    #[test]
162    fn rigs_and_materials_differ_as_named() {
163        let face = [0.0, 0.0, 1.0];
164        let f = |raw: &str| Lighting::from_raw(raw).factor(face);
165        assert!(f(r#"rig="brightRoom""#) > f(r#"rig="threePt""#), "a bright room is brighter");
166        assert!(f(r#"rig="harsh""#) < f(r#"rig="soft""#) + 0.2);
167        assert_eq!(Lighting::from_raw(r#"rig="flat""#).shade(Rgba::rgb(10, 20, 30), face), Rgba::rgb(10, 20, 30));
168        // Metal shines where plastic shines less and matte not at all.
169        let up_left = {
170            let d = dir(240.0, 45.0);
171            let h = [d[0], d[1], d[2] + 1.0];
172            let l = (h[0] * h[0] + h[1] * h[1] + h[2] * h[2]).sqrt();
173            [h[0] / l, h[1] / l, h[2] / l]
174        };
175        let hi = |m: &str| Lighting::from_raw(&format!(r#"prstMaterial="{m}""#)).highlight(up_left);
176        assert!(hi("metal") > hi("plastic") && hi("plastic") > 0.0 && hi("matte") == 0.0);
177        // Turning the rig to the right moves the key light: a right-facing surface gets brighter.
178        let right = [0.7, 0.0, 0.7];
179        assert!(Lighting::from_raw(r#"rig="threePt" dir="r""#).factor(right) > Lighting::from_raw(r#"rig="threePt" dir="t""#).factor(right));
180        // Sunset is warm.
181        let warm = Lighting::from_raw(r#"rig="sunset""#).shade(Rgba::rgb(200, 200, 200), face);
182        assert!(warm.r > warm.b);
183    }
