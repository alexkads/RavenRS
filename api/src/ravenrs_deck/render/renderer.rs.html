207    #[allow(clippy::too_many_arguments)]
208    pub(crate) fn blend(
209        &mut self,
210        pres: &Presentation,
211        old: (&Slide, usize),
212        new: (&Slide, usize),
213        mix: f64,
214        shapes: &[(Shape, bool)],
215        text: &[TextMorph],
216        paths: &[PathMorph],
217        opts: &RenderOpts,
218    ) -> Image {
219        let mix = if mix.is_finite() { mix.clamp(0.0, 1.0) } else { 1.0 };
220        self.render_work = 0;
221        self.shadow_work = 0;
222        self.skip = None;
223        self.in_shadow_pass = false;
224        let (mut ctx, w, h, view) = Self::canvas(pres, opts, needs_filters(pres, old.0) || needs_filters(pres, new.0));
225        let (Some(co), Some(cn)) = (Ctx::for_slide(pres, old.0), Ctx::for_slide(pres, new.0)) else { return self.finish(ctx, w, h) };
226        if mix < 1.0 {
227            let f = Frame { pres, opts, fields: Self::fields(pres, old.1, opts) };
228            self.backdrop(&mut ctx, &f, &co, old.0, view);
229        }
230        if mix > 0.0 {
231            let f = Frame { pres, opts, fields: Self::fields(pres, new.1, opts) };
232            let layered = mix < 1.0;
233            if layered {
234                ctx.push_layer(None, None, Some(mix as f32), None, None);
235            }
236            self.backdrop(&mut ctx, &f, &cn, new.0, view);
237            if layered {
238                ctx.pop_layer();
239            }
240        }
241        let fo = Frame { pres, opts, fields: Self::fields(pres, old.1, opts) };
242        let fnew = Frame { pres, opts, fields: Self::fields(pres, new.1, opts) };
243        let mut drawn: Vec<ShapeId> = Vec::new();
244        // Set only now: the backdrop's ids may clash with the frame's.
245        self.paths = paths.to_vec();
246        for (s, from_old) in shapes {
247            let (f, c) = if *from_old { (&fo, &co) } else { (&fnew, &cn) };
248            let Some(tm) = text.iter().find(|m| m.old == s.id || m.new == s.id) else {
249                self.shape(&mut ctx, f, c, s, view, 0);
250                continue;
251            };
252            self.shape(&mut ctx, f, c, &Shape { text: None, ..s.clone() }, view, 0);
253            drawn.push(s.id);
254            if !(drawn.contains(&tm.old) && drawn.contains(&tm.new)) {
255                continue;
256            }
257            let find = |id: ShapeId| shapes.iter().map(|(s, _)| s).find(|s| s.id == id);
258            if let (Some(a), Some(b)) = (find(tm.old), find(tm.new)) {
259                let now = b.xfrm.unwrap_or_else(|| resolve::xfrm(&cn, b));
260                morph_text::draw(&mut ctx, (&co, a, &fo.fields), (&cn, b, &fnew.fields), tm, now, view);
261            }
262        }
263        self.paths.clear();
264        self.morphs.sweep();
265        self.finish(ctx, w, h)
266    }

268    pub(crate) fn layout(&mut self, pres: &Presentation, mi: usize, li: Option<usize>, opts: &RenderOpts) -> Image {
269        self.render_work = 0;
270        self.shadow_work = 0;
271        self.skip = None;
272        self.in_shadow_pass = false;
273        let (w, h) = output_size(pres, opts);
274        let mut ctx = RenderContext::new_with(w, h, vello_cpu::RenderSettings { num_threads: 0, ..Default::default() });
275        let view = Affine::translate((opts.offset.0, opts.offset.1)) * Affine::scale(opts.scale);
276        let f = Frame { pres, opts, fields: SlideFields::new(1, String::new(), opts) };
277        if let Some(m) = pres.masters.get(mi) {
278            match li.and_then(|i| m.layouts.get(i)) {
279                Some(l) => {
280                    let c = Ctx::for_layout(pres, m, l);
281                    self.background(&mut ctx, &f, &c, None, view);
282                    if l.show_master_shapes {
283                        let mc = Ctx::for_master(pres, m);
284                        for s in m.shapes.iter().filter(|s| s.ph.is_none()) {
285                            self.shape(&mut ctx, &f, &mc, s, view, 0);
286                        }
287                    }
288                    for s in &l.shapes {
289                        self.shape(&mut ctx, &f, &c, s, view, 0);
290                    }
291                }
292                None => {
293                    let c = Ctx::for_master(pres, m);
294                    self.background(&mut ctx, &f, &c, None, view);
295                    for s in &m.shapes {
296                        self.shape(&mut ctx, &f, &c, s, view, 0);
297                    }
298                }
299            }
300        }
301        self.finish(ctx, w, h)
302    }

314        let (fill, ph) = resolve::background(rctx, slide);
315        if f.opts.transparent_background {
316            self.bg = Some((fill, ph, view));
317            return;
318        }
319        let r = Rect::new(0.0, 0.0, f.pres.slide_size.width, f.pres.slide_size.height);
320        ctx.set_transform(view);
321        let path = r.to_path(0.1);
322        // PowerPoint lays every background over white: a picture stretched into part of the
323        // slide, or a see-through colour, leaves white around and under it.
324        ctx.set_paint(color(Rgba::WHITE, 1.0));
325        ctx.fill_path(&path);
326        paint::fill_path(ctx, rctx, &fill, ph, &path, r, view, FillMode::Norm, 1.0, f.pres);
327        self.bg = Some((fill, ph, view));
328    }

340    /// How lines are drawn for a shape under `m` at `depth`: the transform to stroke under and the
341    /// one that takes the shape's outline into it. Inside a group they are the slide-level view and
342    /// the rest of the chain, so the group's child-coordinate scale moves and sizes the outline but
343    /// never the line width; outside a group, `m` itself and no change to the outline.
344    fn line_space(&self, m: Affine, depth: usize) -> (Affine, Affine) {
345        let view = self.view;
346        if depth > 0 && view.determinant().abs() > 1e-12 && view.is_finite() { (view, view.inverse() * m) } else { (m, Affine::IDENTITY) }
347    }

349    /// Shape-space units per point: a group's child coordinates can scale a shape's box (a legacy
350    /// 1360-unit child space makes one unit hundreds of EMUs), but 3-D sizes kept in points
351    /// (contour width, extrusion depth, bevel) stay points, as PowerPoint draws them. 1 outside a
352    /// group.
353    fn per_point(&self, m: Affine, depth: usize) -> f64 {
354        let k = scale_of(self.line_space(m, depth).1);
355        if k.is_finite() && k > 1e-12 { 1.0 / k } else { 1.0 }
356    }

358    /// [`Renderer::per_point`] along the shape's own x and y axes: a group can stretch its child
359    /// space unevenly, and a tiled fill keeps its size in points on the slide on both axes.
360    fn per_point_xy(&self, m: Affine, depth: usize) -> (f64, f64) {
361        let c = self.line_space(m, depth).1.as_coeffs();
362        let inv = |s: f64| if s.is_finite() && s > 1e-12 { 1.0 / s } else { 1.0 };
363        (inv(c[0].hypot(c[1])), inv(c[2].hypot(c[3])))
364    }

366    /// The size and transform a shape's text is set in: inside a group whose child coordinates
367    /// scale the shape, the shape's size in slide points and `m` with that scale taken out, so
368    /// insets and font sizes stay in points; otherwise `w`, `h` and `m` themselves.
369    fn text_space(&self, m: Affine, depth: usize, w: f64, h: f64) -> (f64, f64, Affine) {
370        let (_, rel) = self.line_space(m, depth);
371        let c = rel.as_coeffs();
372        let (sx, sy) = (c[0].hypot(c[1]), c[2].hypot(c[3]));
373        let ok = |k: f64| k.is_finite() && k > 1e-9;
374        if depth == 0 || !ok(sx) || !ok(sy) || ((sx - 1.0).abs() < 1e-9 && (sy - 1.0).abs() < 1e-9) {
375            return (w, h, m);
376        }
377        let (tw, th) = (w * sx, h * sy);
378        if !tw.is_finite() || !th.is_finite() {
379            return (w, h, m);
380        }
381        (tw, th, m * Affine::scale_non_uniform(1.0 / sx, 1.0 / sy))
382    }

384    /// A shape's geometry in its own `w` × `h` space, built at its size on the slide: inside a
385    /// group whose child coordinates scale the shape unevenly, the preset's rounded corners and
386    /// other size-derived features stay as they would be on a plain shape of that size (#1083).
387    fn scaled_geometry(&self, s: &Shape, m: Affine, depth: usize, w: f64, h: f64) -> Geometry {
388        let (tw, th, _) = self.text_space(m, depth, w, h);
389        if (tw, th) == (w, h) {
390            return shape_geometry(s, w, h);
391        }
392        let mut geo = shape_geometry(s, tw, th);
393        let k = |a: f64, b: f64| if b.abs() > 1e-12 { a / b } else { 1.0 };
394        let back = Affine::scale_non_uniform(k(w, tw), k(h, th));
395        for p in &mut geo.paths {
396            p.path.apply_affine(back);
397        }
398        geo.text_rect = back.transform_rect_bbox(geo.text_rect);
399        for p in &mut geo.sites {
400            *p = back * *p;
401        }
402        geo
403    }

405    /// Whether `id` is in `skip`, looked up in a set built once per frame and slice.
406    fn skipped(&mut self, skip: &[ShapeId], id: ShapeId) -> bool {
407        if skip.is_empty() {
408            return false;
409        }
410        let key = (skip.as_ptr() as usize, skip.len());
411        match &self.skip {
412            Some((p, n, set)) if (*p, *n) == key => set.contains(&id),
413            _ => {
414                let set: std::collections::HashSet<ShapeId> = skip.iter().copied().collect();
415                let hit = set.contains(&id);
416                self.skip = Some((key.0, key.1, set));
417                hit
418            }
419        }
420    }

422    fn shape_at(&mut self, ctx: &mut RenderContext, f: &Frame, rctx: &Ctx, s: &Shape, parent: Affine, depth: usize) {
423        if depth > 32 || (s.hidden && !f.opts.show_hidden) || self.skipped(f.opts.skip, s.id) {
424            return;
425        }
426        let work = if self.in_shadow_pass { &mut self.shadow_work } else { &mut self.render_work };
427        *work = work.saturating_add(1);
428        if *work > 100_000 {
429            return;
430        }
431        // Footer placeholders on layouts/masters show only on slides that include them.
432        let st = f.opts.state.map(|g| g(s.id)).unwrap_or_default();
433        if !st.visible || st.opacity <= 0.001 {
434            return;
435        }
436        let x = resolve::xfrm(rctx, s);
437        if !x.is_finite() {
438            return;
439        }
440        let mut m = parent * x.affine();
441        if st.offset != Vec2::ZERO || st.scale != (1.0, 1.0) || st.rotate != 0.0 {
442            let c = Point::new(x.w / 2.0, x.h / 2.0);
443            let around = Affine::translate(c.to_vec2())
444                * Affine::rotate(st.rotate.to_radians())
445                * Affine::scale_non_uniform(st.scale.0, st.scale.1)
446                * Affine::translate(-c.to_vec2());
447            m = parent * Affine::translate(st.offset) * x.affine() * around;
448        }
449        if let Some(a) = camera_turn(rctx, s, x.w, x.h) {
450            m *= a;
451        }
452        let layered = st.opacity < 0.999 || st.clip.is_some();
453        if layered {
454            ctx.set_transform(m);
455            let clip = st.clip.map(|[l, t, r, b]| Rect::new(l * x.w, t * x.h, r * x.w, b * x.h).to_path(0.1));
456            ctx.push_layer(clip.as_ref(), None, Some(st.opacity.clamp(0.0, 1.0) as f32), None, None);
457        }
458        match &s.kind {
459            ShapeKind::Group { children, child } => {
460                let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
461                let gm = parent * crate::geom::group_child_affine(&x, ch);
462                // A group's own fill (a group filled with its parent's passes that on).
463                let own = match &s.fill {
464                    Some(Fill::Group) | None => self.group_fill.clone(),
465                    // A tile is given in points; this group's own space may be scaled by its parents.
466                    Some(fl) => Some((paint::tiled_per_point(fl, self.per_point_xy(m, depth)).into_owned(), m, Rect::new(0.0, 0.0, x.w, x.h))),
467                };
468                let outer = std::mem::replace(&mut self.group_fill, own);
469                // A group's outer shadow is cast by everything in it, as PowerPoint draws it: the
470                // children drawn once more under the shadow's offset, scale and skew, recoloured to
471                // the shadow colour and blurred, below the group.
472                if !self.in_shadow_pass
473                    && let Some(sh) = s.effects.as_ref().and_then(|e| e.outer_shadow.as_ref())
474                {
475                    let c = rctx.color(&sh.color, None);
476                    let k = scale_of(m);
477                    let d = Vec2::new(sh.dir.to_radians().cos(), sh.dir.to_radians().sin()) * sh.dist;
478                    let local = shadow_local(sh, x.w, x.h);
479                    let (own, pre) =
480                        if sh.rotate_with_shape { (Affine::translate(d) * local, Affine::IDENTITY) } else { (local, Affine::translate(d * k)) };
481                    let det = m.determinant();
482                    if det.is_finite() && det.abs() > 1e-12 {
483                        let shift = pre * m * own * m.inverse();
484                        ctx.set_transform(Affine::IDENTITY);
485                        ctx.push_layer(None, None, Some((c.a as f32) / 255.0), None, Some(blur(sh.blur / 3.0 * k)));
486                        self.in_shadow_pass = true;
487                        for ch in children {
488                            self.shape(ctx, f, rctx, ch, shift * gm, depth + 1);
489                        }
490                        self.in_shadow_pass = false;
491                        ctx.set_transform(Affine::IDENTITY);
492                        ctx.push_layer(None, Some(peniko::BlendMode::new(peniko::Mix::Normal, peniko::Compose::SrcIn)), None, None, None);
493                        ctx.set_paint(color(c.with_alpha(255), 1.0));
494                        let (w, h) = (f.opts.scale * f.pres.slide_size.width, f.opts.scale * f.pres.slide_size.height);
495                        ctx.fill_rect(&Rect::new(-w, -h, 3.0 * w, 3.0 * h).inflate(1e4, 1e4));
496                        ctx.pop_layer();
497                        ctx.pop_layer();
498                    }
499                }
500                for c in children {
501                    self.shape(ctx, f, rctx, c, gm, depth + 1);
502                }
503                self.group_fill = outer;
504            }
505            ShapeKind::Table(t) => {
506                // Rows grow with their text: what the cells cover, borders included.
507                let cells = table::draw(ctx, f.pres, rctx, t, m, x.w, x.h, &f.fields, f.opts.edit && f.opts.editing_text == Some(s.id));
508                self.track(m, cells.unwrap_or(Rect::new(0.0, 0.0, x.w, x.h)).inflate(2.0, 2.0));
509            }
510            ShapeKind::Chart(c) => {
511                self.track(m, Rect::new(0.0, 0.0, x.w, x.h));
512                chart::draw(ctx, rctx, c, m, x.w, x.h)
513            }
514            ShapeKind::Ink { strokes } => {
515                for sk in strokes {
516                    let legacy = crate::model::ink_pressure_is_legacy(&sk.points);
517                    let pen = if sk.width.is_finite() { sk.width.abs() } else { 0.0 };
518                    for (px, py, pr) in &sk.points {
519                        let w = crate::model::ink_sample_width(pen, *pr, legacy);
520                        self.track(parent, Rect::new(*px, *py, *px, *py).inflate(w, w));
521                    }
522                    let pts: Vec<(f64, f64, f32)> = sk.points.iter().copied().filter(|p| p.0.is_finite() && p.1.is_finite()).collect();
523                    if pts.len() < 2 {
524                        continue;
525                    }
526                    let (base, rel) = self.line_space(parent, depth);
527                    ctx.set_transform(base);
528                    ctx.set_paint(color(sk.color, if sk.highlighter { 0.4 } else { 1.0 }));
529                    // Uniform-width and highlighter strokes are one path: segments drawn one by one
530                    // overlap at the joins, which darkens see-through highlighter ink in beads.
531                    if legacy || sk.highlighter {
532                        let mut bp = BezPath::new();
533                        for (i, (px, py, _)) in pts.iter().enumerate() {
534                            if i == 0 {
535                                bp.move_to((*px, *py));
536                            } else {
537                                bp.line_to((*px, *py));
538                            }
539                        }
540                        bp.apply_affine(rel);
541                        ctx.set_stroke(kurbo::Stroke::new(pen.max(0.1)).with_caps(kurbo::Cap::Round).with_join(kurbo::Join::Round));
542                        ctx.stroke_path(&bp);
543                        continue;
544                    }
545                    for pair in pts.windows(2) {
546                        let [p0, p1] = pair else { continue };
547                        let mut bp = BezPath::new();
548                        bp.move_to((p0.0, p0.1));
549                        bp.line_to((p1.0, p1.1));
550                        bp.apply_affine(rel);
551                        let a = crate::model::ink_sample_width(pen, p0.2, legacy);
552                        let b = crate::model::ink_sample_width(pen, p1.2, legacy);
553                        ctx.set_stroke(kurbo::Stroke::new(((a + b) * 0.5).max(0.1)).with_caps(kurbo::Cap::Round).with_join(kurbo::Join::Round));
554                        ctx.stroke_path(&bp);
555                    }
556                }
557            }
558            _ => self.geometry_shape(ctx, f, rctx, s, m, x.w, x.h, &st, depth),
559        }
560        if layered {
561            ctx.pop_layer();
562        }
563    }

565    /// Fill the filled sub-paths of `geo` (a `w` × `h` shape placed by `m`) with `fl`, faded per
566    /// sub-path by `fade`: `useBgFill` shows the slide background through the shape and `grpFill`
567    /// the group's fill.
568    #[allow(clippy::too_many_arguments)]
569    fn fill_geometry(
570        &self,
571        ctx: &mut RenderContext,
572        f: &Frame,
573        rctx: &Ctx,
574        fl: &Fill,
575        fill_ph: Option<Rgba>,
576        geo: &Geometry,
577        (w, h): (f64, f64),
578        m: Affine,
579        per_point: (f64, f64),
580        fade: &dyn Fn(usize) -> crate::geom::morph::Fade,
581    ) {
582        for (i, sp) in geo.paths.iter().enumerate() {
583            if sp.fill == FillMode::None || fade(i).fill <= 0.0 {
584                continue;
585            }
586            ctx.set_transform(m);
587            ctx.set_fill_rule(if sp.even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
588            match (fl, &self.bg) {
589                // `useBgFill`: the slide's background, laid out on the slide (not the
590                // shape), shows through the shape.
591                (Fill::Background, Some((bg, ph, view))) => {
592                    let slide = Rect::new(0.0, 0.0, f.pres.slide_size.width, f.pres.slide_size.height);
593                    let path = (view.inverse() * m) * sp.path.clone();
594                    paint::fill_path(ctx, rctx, bg, *ph, &path, slide, *view, sp.fill, fade(i).fill, f.pres);
595                }
596                // `grpFill`: the group's fill, laid out over the whole group.
597                (Fill::Group, _) => {
598                    if let Some((gf, gm, gbox)) = &self.group_fill
599                        && gm.determinant().abs() > 1e-12
600                    {
601                        let path = (gm.inverse() * m) * sp.path.clone();
602                        paint::fill_path(ctx, rctx, gf, None, &path, *gbox, *gm, sp.fill, fade(i).fill, f.pres);
603                    }
604                }
605                _ => {
606                    let fl = paint::tiled_per_point(fl, per_point);
607                    paint::fill_path(ctx, rctx, &fl, fill_ph, &sp.path, Rect::new(0.0, 0.0, w, h), m, sp.fill, fade(i).fill, f.pres)
608                }
609            }
610            ctx.set_fill_rule(peniko::Fill::NonZero);
611        }
612    }

614    #[allow(clippy::too_many_arguments)]
615    fn geometry_shape(
616        &mut self,
617        ctx: &mut RenderContext,
618        f: &Frame,
619        rctx: &Ctx,
620        s: &Shape,
621        m: Affine,
622        w: f64,
623        h: f64,
624        st: &ShapeState,
625        depth: usize,
626    ) {
627        let (geo, fades) = match self.paths.iter().find(|p| p.id == s.id) {
628            Some(pm) => self.morphs.geometry(pm, s, w, h),
629            None => (self.scaled_geometry(s, m, depth, w, h), Vec::new()),
630        };
631        if !drawable(&geo, m, w, h) {
632            log::warn!("render: shape {} skipped: its outline is not finite or far larger than any screen", s.id.0);
633            return;
634        }
635        let (fill, fill_ph) = resolve::fill(rctx, s);
636        let (line, line_ph) = resolve::line(rctx, s);
637        let (effects, fx_ph) = resolve::effects(rctx, s);
638        let reach = extent::spill(Some(&line), effects.as_ref(), w, h);
639        self.track(m, Rect::new(0.0, 0.0, w, h).inflate(reach, reach));
640        let is_pic = matches!(s.kind, ShapeKind::Picture { .. });
641        let outline = geo.outline();
642        // Placeholders in the editor: dashed outline and prompt, nothing else when empty.
643        let empty_ph = s.ph.is_some() && s.text.as_ref().is_none_or(TextBody::is_empty) && !is_pic && !matches!(s.kind, ShapeKind::Media(_));
644        let fill = if let Some(t) = st.tint { Some(Fill::solid(crate::model::ColorRef::rgb(t))) } else { fill };
645        let d = ShapeDraw { f, rctx, s, st, m, w, h, depth, geo, fades, fill, fill_ph, line, line_ph, effects, fx_ph, outline, is_pic, empty_ph };
646        self.cast_effects(ctx, &d);
647        let soft = d.effects.as_ref().and_then(|e| e.soft_edge).filter(|r| *r > 0.0);
648        if let Some(r) = soft {
649            ctx.set_transform(Affine::IDENTITY);
650            ctx.push_layer(None, None, None, None, None);
651            let _ = r;
652        }
653        let pt = self.point_scale(&d);
654        self.draw_extrusion(ctx, &d, pt);
655        self.draw_contour(ctx, &d, pt);
656        self.draw_fill(ctx, &d);
657        self.draw_fill_overlay(ctx, &d);
658        self.draw_bevel(ctx, &d, pt);
659        self.draw_outline(ctx, &d);
660        if let Some(r) = soft {
661            self.draw_feather(ctx, &d, r);
662        }
663        self.draw_inner_shadow(ctx, &d);
664        self.draw_text_body(ctx, &d);
665    }

667    /// Shape-space units per point for the 3-D sizes, measured without the camera's turn (which
668    /// `shape_at` puts in `m`): its foreshortening is not a child-coordinate scale.
669    fn point_scale(&self, d: &ShapeDraw) -> f64 {
670        let turn = d
671            .effects
672            .as_ref()
673            .and_then(|e| e.raw3d.as_deref())
674            .and_then(camera::from_raw)
675            .and_then(|c| camera::affine(&c, d.w, d.h))
676            .filter(|t| t.determinant().abs() > 1e-9);
677        self.per_point(turn.map_or(d.m, |t| d.m * t.inverse()), d.depth)
678    }

680    fn cast_effects(&mut self, ctx: &mut RenderContext, d: &ShapeDraw) {
681        let m = d.m;
682        let w = d.w;
683        let h = d.h;
684        let depth = d.depth;
685        let fill_ph = d.fill_ph;
686        let line_ph = d.line_ph;
687        let fx_ph = d.fx_ph;
688        let is_pic = d.is_pic;
689        let s = d.s;
690        let rctx = d.rctx;
691        let geo = &d.geo;
692        let fill = &d.fill;
693        let effects = &d.effects;
694        let line = &d.line;
695        // Effects below the shape.
696        if let Some(e) = &effects {
697            // A picture (or picture fill) casts its shadow from its pixels, not its frame. A tile
698            // is sized in points on the slide, as the face's is, and so is its shadow and reflection.
699            let content = match (&s.kind, &fill) {
700                (ShapeKind::Picture { fill: pf }, _) | (_, Some(Fill::Picture(pf))) => {
701                    let pf = paint::tiled_picture_per_point(pf, self.per_point_xy(m, depth));
702                    Some(paint::resolved(rctx, &pf, None).into_owned())
703                }
704                _ => None,
705            };
706            // What casts the shadow: the fill as see-through as it is, and the outline.
707            let fill_alpha = fill_opacity(rctx, fill, fill_ph);
708            let outline = line.fill.as_ref().filter(|f| !f.is_none()).map(|lf| {
709                let a = match lf {
710                    Fill::Solid { color: c } => rctx.color(c, line_ph).a as f64 / 255.0,
711                    _ => 1.0,
712                };
713                let (base, rel) = self.line_space(m, depth);
714                let lw = line.width.unwrap_or(0.75).max(0.0);
715                let outside = if border_outside(is_pic, effects.as_ref()) { outside_line(line, lw) } else { None };
716                Caster { line: outside.map_or(std::borrow::Cow::Borrowed(line), std::borrow::Cow::Owned), ph: line_ph, alpha: a, base, rel }
717            });
718            self.effects_below(ctx, rctx, e, fx_ph, FxSource { geo, fill, fill_alpha, outline, content: content.as_ref(), m, w, h });
719        }
720    }

769    fn draw_fill(&mut self, ctx: &mut RenderContext, d: &ShapeDraw) {
770        let m = d.m;
771        let w = d.w;
772        let h = d.h;
773        let fill_ph = d.fill_ph;
774        let empty_ph = d.empty_ph;
775        let s = d.s;
776        let f = d.f;
777        let rctx = d.rctx;
778        let geo = &d.geo;
779        let fill = &d.fill;
780        let outline = &d.outline;
781        let fade = |i: usize| d.fade(i);
782        let pt_xy = self.per_point_xy(m, d.depth);
783        // Fill.
784        match &s.kind {
785            ShapeKind::Picture { fill: pf } => {
786                // The picture's own fill shows behind its transparent pixels, as in PowerPoint (an
787                // emphasis tint recolours it, as it does any shape's fill). A picture that sets
788                // neither a fill nor a style shows the fill its layout or master placeholder sets
789                // (#2068). A style fill alone, its own or inherited, stays undrawn.
790                let explicit = |sh: &crate::model::Shape| sh.fill.as_ref().is_some_and(|f| !f.is_none());
791                let inherited = || {
792                    let (lp, mp) = resolve::parents(rctx, s);
793                    [lp, mp].into_iter().flatten().find(|p| p.fill.is_some() || p.style.is_some()).is_some_and(explicit)
794                };
795                if (explicit(s) || (s.fill.is_none() && s.style.is_none() && inherited()))
796                    && let Some(fl) = &fill
797                {
798                    self.fill_geometry(ctx, f, rctx, fl, fill_ph, geo, (w, h), m, pt_xy, &fade);
799                }
800                ctx.set_transform(m);
801                let bounds = Rect::new(0.0, 0.0, w, h);
802                ctx.push_clip_layer(outline);
803                let t = f.opts.gif_times.iter().find(|g| g.0 == s.id).map(|g| g.1);
804                let pf = paint::tiled_picture_per_point(pf, pt_xy);
805                paint::picture(ctx, f.pres, &paint::resolved(rctx, &pf, None), bounds, m, t);
806                ctx.pop_layer();
807            }
808            // An object DeckCraft doesn't model (an embedded OLE object, an unknown graphic): its
809            // preview picture, stretched over the frame, as PowerPoint draws it.
810            ShapeKind::Opaque { preview: Some(media), .. } => {
811                // An embedded object's fallback picture can carry a fill (old equation objects
812                // drawn on coloured boxes): behind the preview, as for a picture.
813                if s.fill.as_ref().is_some_and(|f| !f.is_none())
814                    && let Some(fl) = &fill
815                {
816                    self.fill_geometry(ctx, f, rctx, fl, fill_ph, geo, (w, h), m, pt_xy, &fade);
817                }
818                ctx.set_transform(m);
819                let pf = crate::model::style::PictureFill { media: *media, ..Default::default() };
820                paint::picture(ctx, f.pres, &pf, Rect::new(0.0, 0.0, w, h), m, None);
821            }
822            ShapeKind::Media(mc) => {
823                ctx.set_transform(m);
824                let bounds = Rect::new(0.0, 0.0, w, h);
825                if let Some(p) = mc.poster {
826                    let pf = crate::model::style::PictureFill { media: p, crop: mc.crop, ..Default::default() };
827                    ctx.push_clip_layer(outline);
828                    paint::picture(ctx, f.pres, &pf, bounds, m, None);
829                    ctx.pop_layer();
830                } else if mc.video {
831                    ctx.set_paint(peniko::Color::from_rgba8(20, 20, 24, 255));
832                    ctx.fill_rect(&bounds);
833                }
834                if !mc.video || mc.poster.is_none() {
835                    paint::media_glyph(ctx, mc.video, bounds);
836                }
837            }
838            _ => {
839                if let Some(fl) = &fill
840                    && !(empty_ph && f.opts.edit && s.fill.is_none() && s.line.is_none() && s.style.is_none())
841                {
842                    self.fill_geometry(ctx, f, rctx, fl, fill_ph, geo, (w, h), m, pt_xy, &fade);
843                }
844            }
845        }
846    }

912    fn draw_outline(&mut self, ctx: &mut RenderContext, d: &ShapeDraw) {
913        let m = d.m;
914        let w = d.w;
915        let h = d.h;
916        let depth = d.depth;
917        let line_ph = d.line_ph;
918        let is_pic = d.is_pic;
919        let s = d.s;
920        let rctx = d.rctx;
921        let geo = &d.geo;
922        let effects = &d.effects;
923        let outline = &d.outline;
924        let line = d.line.clone();
925        let fade = |i: usize| d.fade(i);
926        // Some presets' outlines default to sharp corners (PowerPoint's own per-shape default).
927        let line = match (&s.geom, line.join) {
928            (Geom::Preset { name, .. }, None) if preset::mitred_by_default(name) => Line { join: Some(crate::model::style::LineJoin::Miter), ..line },
929            _ => line,
930        };
931        // Outline.
932        if line.fill.as_ref().is_some_and(|f| !f.is_none()) {
933            let lw = line.width.unwrap_or(0.75).max(0.0);
934            let (base, rel) = self.line_space(m, depth);
935            // Arrowheads go on lines, arcs and open freeforms (filled or not), never on closed
936            // outlines or on presets like brackets.
937            let ends = s.is_line()
938                || geo.is_open()
939                || (matches!(s.geom, Geom::Custom { .. })
940                    && geo.paths.iter().find(|p| p.stroke).is_some_and(|p| !p.path.elements().iter().any(|e| matches!(e, PathEl::ClosePath))));
941            let trims = if ends { head_trims(&line, lw) } else { (0.0, 0.0) };
942            let has_heads = ends && [&line.head, &line.tail].iter().any(|e| e.as_ref().is_some_and(|e| e.kind != "none"));
943            // A see-through line and its arrowheads are one shape in PowerPoint: drawn opaque into
944            // a layer that is then faded, so they don't darken where they meet.
945            let see_through = match &line.fill {
946                Some(Fill::Solid { color: c }) if has_heads => Some(rctx.color(c, line_ph)).filter(|c| c.a < 255),
947                _ => None,
948            };
949            let line = match see_through {
950                Some(c) => {
951                    ctx.set_transform(Affine::IDENTITY);
952                    ctx.push_layer(None, None, Some(c.a as f32 / 255.0), None, None);
953                    std::borrow::Cow::Owned(Line { fill: Some(Fill::solid(crate::model::style::ColorRef::rgb(c.with_alpha(255)))), ..line.clone() })
954                }
955                None => std::borrow::Cow::Borrowed(&line),
956            };
957            // A picture's border lies outside the picture, as PowerPoint draws it: stroked twice
958            // as wide, outside the outline only (see `border_outside`).
959            let outside = border_outside(is_pic, effects.as_ref()).then(|| outside_line(&line, lw)).flatten();
960            let clipped = outside.is_some();
961            let (line, lw) = match outside {
962                Some(o) => {
963                    let mut clip = Rect::new(0.0, 0.0, w, h).inflate(w + h + lw * 8.0, w + h + lw * 8.0).to_path(0.1);
964                    clip.extend(outline.iter());
965                    ctx.set_transform(m);
966                    ctx.set_fill_rule(peniko::Fill::EvenOdd);
967                    ctx.push_clip_layer(&clip);
968                    ctx.set_fill_rule(peniko::Fill::NonZero);
969                    (std::borrow::Cow::Owned(o), lw * 2.0)
970                }
971                None => (line, lw),
972            };
973            let first_stroked = geo.paths.iter().position(|p| p.stroke);
974            for (i, sp) in geo.paths.iter().enumerate().filter(|(i, p)| p.stroke && fade(*i).stroke > 0.0) {
975                let mut path = if rel == Affine::IDENTITY {
976                    std::borrow::Cow::Borrowed(&sp.path)
977                } else {
978                    let mut moved = sp.path.clone();
979                    moved.apply_affine(rel);
980                    std::borrow::Cow::Owned(moved)
981                };
982                // The line stops inside its arrowheads, so a wide line doesn't poke out of them.
983                let whole = path.clone();
984                if Some(i) == first_stroked && trims != (0.0, 0.0) {
985                    path = std::borrow::Cow::Owned(trim_ends(&path, trims.0, trims.1));
986                }
987                stroke(ctx, rctx, &line, line_ph, &path, Some(&whole), base, lw, fade(i).stroke);
988            }
989            if ends {
990                arrowheads(ctx, rctx, &line, line_ph, geo, rel, base, lw);
991            }
992            if clipped {
993                ctx.pop_layer();
994            }
995            if see_through.is_some() {
996                ctx.pop_layer();
997            }
998        }
999    }

1001    fn draw_feather(&mut self, ctx: &mut RenderContext, d: &ShapeDraw, r: f64) {
1002        let m = d.m;
1003        let outline = &d.outline;
1004        // Feather: the shape (fill and line) fades to nothing at its outline. Measured on
1005        // PowerPoint's export: the mask is the outline shrunk by 0.85 r, blurred by a third of
1006        // r, so it's half opaque 0.85 r inside the edge and transparent at the edge itself.
1007        ctx.set_transform(m);
1008        ctx.push_layer(
1009            None,
1010            Some(peniko::BlendMode::new(peniko::Mix::Normal, peniko::Compose::DestIn)),
1011            None,
1012            None,
1013            Some(blur(r / 3.0 * scale_of(m))),
1014        );
1015        ctx.set_paint(peniko::Color::from_rgba8(0, 0, 0, 255));
1016        ctx.fill_path(outline);
1017        ctx.push_blend_layer(peniko::BlendMode::new(peniko::Mix::Normal, peniko::Compose::DestOut));
1018        ctx.set_stroke(kurbo::Stroke::new(r * 1.7).with_join(kurbo::Join::Round));
1019        ctx.stroke_path(outline);
1020        ctx.pop_layer();
1021        ctx.pop_layer();
1022        ctx.pop_layer();
1023    }

1026        let m = d.m;
1027        let w = d.w;
1028        let h = d.h;
1029        let fx_ph = d.fx_ph;
1030        let rctx = d.rctx;
1031        let effects = &d.effects;
1032        let outline = &d.outline;
1033        // Inner shadow.
1034        if let Some(e) = &effects
1035            && let Some(sh) = &e.inner_shadow
1036        {
1037            ctx.set_transform(m);
1038            ctx.push_clip_layer(outline);
1039            let c = rctx.color(&sh.color, fx_ph);
1040            let d = Vec2::new(sh.dir.to_radians().cos(), sh.dir.to_radians().sin()) * sh.dist;
1041            // As PowerPoint draws it (measured on its export): the shadow falls inside along the
1042            // edges it points to (45 degrees: the bottom and right), blurred by a third of the
1043            // blur radius.
1044            ctx.push_layer(None, None, None, None, Some(blur(sh.blur / 3.0 * scale_of(m))));
1045            ctx.set_paint(color(c, 1.0));
1046            let mut ring = Rect::new(0.0, 0.0, w, h).inflate(w.max(h), w.max(h)).to_path(0.1);
1047            let mut hole = outline.clone();
1048            hole.apply_affine(Affine::translate(-d));
1049            ring.extend(hole.iter());
1050            ctx.set_fill_rule(peniko::Fill::EvenOdd);
1051            ctx.fill_path(&ring);
1052            ctx.set_fill_rule(peniko::Fill::NonZero);
1053            ctx.pop_layer();
1054            ctx.pop_layer();
1055        }

1058    fn draw_text_body(&mut self, ctx: &mut RenderContext, d: &ShapeDraw) {
1059        let m = d.m;
1060        let w = d.w;
1061        let h = d.h;
1062        let depth = d.depth;
1063        let empty_ph = d.empty_ph;
1064        let s = d.s;
1065        let f = d.f;
1066        let rctx = d.rctx;
1067        let st = d.st;
1068        let geo = &d.geo;
1069        // Text (or the placeholder prompt in the editor).
1070        let skip_text = f.opts.skip_text == Some(s.id);
1071        if let Some(body) = &s.text
1072            && !skip_text
1073        {
1074            // Text in a group is set at the shape's size on the slide, in points: a group's
1075            // child-coordinate scale sizes and places the box, never the type (#233's twin).
1076            let (tw, th, m) = self.text_space(m, depth, w, h);
1077            let text_geo;
1078            let geo = if (tw, th) == (w, h) {
1079                geo
1080            } else {
1081                text_geo = shape_geometry(s, tw, th);
1082                &text_geo
1083            };
1084            let tr = geo.text_rect;
1085            let editing = f.opts.edit && f.opts.editing_text == Some(s.id);
1086            if !body.is_empty() || editing {
1087                // Imported placeholders can have no paragraph until the first keystroke.
1088                let mut empty;
1089                let body = if editing && body.paragraphs.is_empty() {
1090                    empty = body.clone();
1091                    empty.paragraphs.push(Default::default());
1092                    &empty
1093                } else {
1094                    body
1095                };
1096                let mut l = crate::text::layout(rctx, s, body, &Opts { rect: tr, fields: &f.fields, prompt_color: None, no_shrink: false });
1097                if editing {
1098                    preview_empty_bullets(&mut l);
1099                }
1100                let tm = text_transform(m, tr, l.rotation);
1101                for b in l.lines.iter().filter_map(line_bounds) {
1102                    // Glyphs reach a little past their carets and line box (italics, swashes).
1103                    let pad = b.height() * 0.3;
1104                    self.track(tm, b.inflate(pad, pad));
1105                }
1106                match text_warp(rctx, s, tr).filter(|_| l.rotation == 0.0) {
1107                    Some(w) => warp::draw(ctx, &l, tr, m, &st.paras, &w),
1108                    None => draw_layout_with(ctx, &l, tr, m, &st.paras),
1109                }
1110            } else if f.opts.edit && s.ph.is_some() {
1111                let kind = s.ph_type().unwrap_or(PhType::Body);
1112                let prompt_body = prompt_body(rctx, s, kind);
1113                if !prompt_body.is_empty() {
1114                    let grey = Rgba::rgb(0x59, 0x59, 0x59);
1115                    draw_text(ctx, rctx, s, &prompt_body, m, &Opts { rect: tr, fields: &f.fields, prompt_color: Some(grey), no_shrink: false });
1116                }
1117            }
1118        }
1119        if empty_ph && f.opts.edit {
1120            // Dashed placeholder outline.
1121            ctx.set_transform(m);
1122            let px = 1.0 / scale_of(m).max(1e-6);
1123            ctx.set_paint(peniko::Color::from_rgba8(0xA6, 0xA6, 0xA6, 255));
1124            ctx.set_stroke(kurbo::Stroke::new(px).with_dashes(0.0, [3.0 * px, 2.0 * px]));
1125            ctx.stroke_path(&Rect::new(0.0, 0.0, w, h).to_path(0.1));
1126            if matches!(s.ph_type(), Some(PhType::Picture | PhType::Chart | PhType::Table | PhType::Media | PhType::Diagram)) {
1127                paint::placeholder_icon(ctx, s.ph_type().unwrap_or(PhType::Picture), Rect::new(0.0, 0.0, w, h));
1128            }
1129        }
1130    }

1132    fn effects_below(&mut self, ctx: &mut RenderContext, rctx: &Ctx, e: &Effects, ph: Option<Rgba>, src: FxSource) {
1133        let FxSource { geo, fill, fill_alpha, outline, content, m, w, h } = src;
1134        let silhouette = {
1135            let mut p = BezPath::new();
1136            for sp in &geo.paths {
1137                if sp.fill != FillMode::None || fill.is_none() {
1138                    p.extend(sp.path.iter());
1139                }
1140            }
1141            if p.elements().is_empty() { geo.outline() } else { p }
1142        };
1143        let k = scale_of(m);
1144        if let Some(sh) = &e.outer_shadow {
1145            let c = rctx.color(&sh.color, ph);
1146            let d = Vec2::new(sh.dir.to_radians().cos(), sh.dir.to_radians().sin()) * sh.dist;
1147            let local = shadow_local(sh, w, h);
1148            // Shadow offset is in slide space (doesn't rotate with the shape unless asked).
1149            let (own, pre) = if sh.rotate_with_shape { (Affine::translate(d) * local, Affine::IDENTITY) } else { (local, Affine::translate(d * k)) };
1150            let sm = pre * m * own;
1151            ctx.set_transform(Affine::IDENTITY);
1152            // Blurred by a third of the blur radius, as PowerPoint's export shows.
1153            ctx.push_layer(None, None, Some((c.a as f32) / 255.0), None, Some(blur(sh.blur / 3.0 * k)));
1154            ctx.set_transform(sm);
1155            match content {
1156                // As PowerPoint draws it: the shadow is the picture's own coverage (a logo on a
1157                // transparent background casts the logo's shadow): the picture, clipped to the
1158                // shape, recoloured to the shadow colour.
1159                Some(pf) => {
1160                    ctx.push_clip_layer(&silhouette);
1161                    paint::picture(ctx, rctx.pres, pf, Rect::new(0.0, 0.0, w, h), sm, None);
1162                    // Drawing the picture leaves its own pixel transform set: back to the shape's,
1163                    // or the recolouring below covers only part of a large picture's shadow.
1164                    ctx.set_transform(sm);
1165                    ctx.push_layer(None, Some(peniko::BlendMode::new(peniko::Mix::Normal, peniko::Compose::SrcIn)), None, None, None);
1166                    ctx.set_paint(color(c.with_alpha(255), 1.0));
1167                    ctx.fill_rect(&Rect::new(0.0, 0.0, w, h).inflate(1.0, 1.0));
1168                    ctx.pop_layer();
1169                    ctx.pop_layer();
1170                }
1171                // The shadow is cast by what is drawn, as PowerPoint casts it: the filled part as
1172                // see-through as its fill (none for an outline-only shape) …
1173                None => {
1174                    if let Some(a) = fill_alpha {
1175                        ctx.set_paint(color(c.with_alpha(255), a));
1176                        // Each subpath under its own fill rule, as the shape itself is
1177                        // drawn: one merged path filled NonZero fills the hole of an
1178                        // even-odd shape (a donut, a frame) (#1596).
1179                        for sp in geo.paths.iter().filter(|sp| sp.fill != FillMode::None) {
1180                            ctx.set_fill_rule(if sp.even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
1181                            ctx.fill_path(&sp.path);
1182                        }
1183                        ctx.set_fill_rule(peniko::Fill::NonZero);
1184                    }
1185                }
1186            }
1187            // … and the outline (a picture's border too).
1188            if let Some(o) = &outline {
1189                let cast = Line { fill: Some(Fill::solid(crate::model::style::ColorRef::rgb(c.with_alpha(255)))), ..o.line.as_ref().clone() };
1190                let lw = o.line.width.unwrap_or(0.75).max(0.0);
1191                for sp in geo.paths.iter().filter(|sp| sp.stroke) {
1192                    let mut path = sp.path.clone();
1193                    path.apply_affine(o.rel * own);
1194                    stroke(ctx, rctx, &cast, o.ph, &path, None, pre * o.base, lw, o.alpha);
1195                }
1196            }
1197            ctx.pop_layer();
1198        }
1199        // A shape with no fill and no line (a text box) draws nothing to glow around: its text's
1200        // own glow is all PowerPoint shows, not a halo round the box.
1201        let glows = fill_alpha.is_some() || content.is_some() || outline.is_some();
1202        if let Some(g) = e.glow.as_ref().filter(|_| glows) {
1203            let c = rctx.color(&g.color, ph);
1204            ctx.set_transform(Affine::IDENTITY);
1205            // Measured on PowerPoint's export: a glow of radius r is half strength about half of r
1206            // out from the outline and gone by r.
1207            ctx.push_layer(None, None, Some((c.a as f32) / 255.0), None, Some(blur(g.radius * 0.17 * k)));
1208            ctx.set_transform(m);
1209            ctx.set_paint(color(c.with_alpha(255), 1.0));
1210            ctx.set_stroke(kurbo::Stroke::new(g.radius.max(0.0) * 0.95).with_join(kurbo::Join::Round));
1211            ctx.stroke_path(&silhouette);
1212            // A shape without a fill (an outline, a frame) glows along its line only (#225).
1213            if fill_alpha.is_some() || content.is_some() {
1214                // Per subpath under its own fill rule, as in the shadow above: the
1215                // merged silhouette filled NonZero fills an even-odd hole (#1596).
1216                // When no subpath takes part the silhouette is the outline fallback.
1217                let mut filled_any = false;
1218                for sp in geo.paths.iter().filter(|sp| sp.fill != FillMode::None || fill.is_none()) {
1219                    ctx.set_fill_rule(if sp.even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
1220                    ctx.fill_path(&sp.path);
1221                    filled_any = true;
1222                }
1223                if !filled_any {
1224                    ctx.fill_path(&silhouette);
1225                }
1226                ctx.set_fill_rule(peniko::Fill::NonZero);
1227            }
1228            ctx.pop_layer();
1229        }
1230        if let Some(rf) = &e.reflection {
1231            // Mirror below the shape, fading out.
1232            let mm = m * Affine::translate((0.0, 2.0 * h + rf.dist)) * Affine::scale_non_uniform(1.0, -1.0);
1233            ctx.set_transform(Affine::IDENTITY);
1234            ctx.push_layer(None, None, Some(rf.start_alpha.clamp(0.0, 1.0) as f32 * 0.6), None, Some(blur(rf.blur * 0.5 * k)));
1235            ctx.set_transform(mm);
1236            match content {
1237                // A picture reflects its pixels (it has no fill of its own).
1238                Some(pf) => {
1239                    ctx.push_clip_layer(&silhouette);
1240                    paint::picture(ctx, rctx.pres, pf, Rect::new(0.0, 0.0, w, h), mm, None);
1241                    ctx.pop_layer();
1242                }
1243                None => {
1244                    if let Some(fl) = fill {
1245                        // Per subpath under its own fill rule, as in the shadow above:
1246                        // the merged silhouette filled NonZero fills an even-odd hole
1247                        // (#1596). The outline fallback covers shapes with no
1248                        // participating subpath.
1249                        let mut filled_any = false;
1250                        for sp in geo.paths.iter().filter(|sp| sp.fill != FillMode::None || fill.is_none()) {
1251                            ctx.set_fill_rule(if sp.even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
1252                            paint::fill_path(ctx, rctx, fl, ph, &sp.path, Rect::new(0.0, 0.0, w, h), mm, FillMode::Norm, 1.0, rctx.pres);
1253                            filled_any = true;
1254                        }
1255                        if !filled_any {
1256                            paint::fill_path(ctx, rctx, fl, ph, &silhouette, Rect::new(0.0, 0.0, w, h), mm, FillMode::Norm, 1.0, rctx.pres);
1257                        }
1258                        ctx.set_fill_rule(peniko::Fill::NonZero);
1259                    }
1260                }
1261            }
1262            ctx.set_transform(mm);
1263            // Fade mask.
1264            ctx.push_layer(None, Some(peniko::BlendMode::new(peniko::Mix::Normal, peniko::Compose::DestIn)), None, None, None);
1265            let grad = peniko::Gradient::new_linear(Point::new(0.0, h), Point::new(0.0, h * (1.0 - rf.end_pos.clamp(0.01, 1.0))))
1266                .with_stops([(0.0, peniko::Color::from_rgba8(0, 0, 0, 255)), (1.0, peniko::Color::from_rgba8(0, 0, 0, 0))].as_slice());
1267            ctx.set_paint(grad);
1268            ctx.fill_rect(&Rect::new(0.0, 0.0, w, h).inflate(4.0, 4.0));
1269            ctx.pop_layer();
1270            ctx.pop_layer();
1271        }
1272    }
