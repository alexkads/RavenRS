8pub(crate) fn shade(c: Rgba, m: FillMode) -> Rgba {
9    match m {
10        FillMode::Lighten => c.lerp(Rgba::WHITE.with_alpha(c.a), 0.4),
11        FillMode::LightenLess => c.lerp(Rgba::WHITE.with_alpha(c.a), 0.2),

18/// Geometry of a shape at its size.
19pub fn shape_geometry(shape: &Shape, w: f64, h: f64) -> Geometry {
20    match &shape.geom {
21        Geom::Preset { name, adj } => preset::build(name, w, h, adj).or_else(|| preset::build("rect", w, h, &[])).unwrap_or(Geometry {
22            paths: vec![],
23            text_rect: Rect::new(0.0, 0.0, w, h),
24            handles: vec![],
25            sites: vec![],
26        }),
27        Geom::Custom { paths } => {
28            let mut out = vec![];
29            for p in paths {
30                let (pw, ph) = (if p.w > 0.0 { p.w } else { w.max(1e-9) }, if p.h > 0.0 { p.h } else { h.max(1e-9) });
31                let mut bp = parse_path(&p.d);
32                bp.apply_affine(Affine::scale_non_uniform(w / pw, h / ph));
33                // PowerPoint fills each custom path's sub-paths alternately (even-odd): a sub-path
34                // inside another cuts a hole whichever way it winds, as in icons and outlined text.
35                out.push(SubPath { path: bp, fill: p.fill, stroke: p.stroke, even_odd: true });
36            }
37            Geometry { paths: out, text_rect: Rect::new(0.0, 0.0, w, h), handles: vec![], sites: vec![] }
38        }
39    }
40}

42/// Parse our custom path syntax (`M x y L x y C … Q … Z`).
43pub fn parse_path(d: &str) -> BezPath {
44    let mut bp = BezPath::new();
45    let toks: Vec<&str> = d.split(|c: char| c.is_whitespace() || c == ',').filter(|t| !t.is_empty()).collect();
46    let mut i = 0;
47    let num = |i: usize| toks.get(i).and_then(|t| t.parse::<f64>().ok()).filter(|v| v.is_finite());
48    let mut started = false;
49    while i < toks.len() {
50        let t = toks.get(i).copied().unwrap_or("");
51        match t {
52            "M" => {
53                if let (Some(x), Some(y)) = (num(i + 1), num(i + 2)) {
54                    bp.move_to((x, y));
55                    started = true;
56                }
57                i += 3;
58            }
59            "L" => {
60                if let (Some(x), Some(y)) = (num(i + 1), num(i + 2)) {
61                    if !started {
62                        bp.move_to((x, y));
63                        started = true;
64                    } else {
65                        bp.line_to((x, y));
66                    }
67                }
68                i += 3;
69            }
70            "C" => {
71                if let (Some(a), Some(b), Some(c), Some(d2), Some(e), Some(f)) =
72                    (num(i + 1), num(i + 2), num(i + 3), num(i + 4), num(i + 5), num(i + 6))
73                    && started
74                {
75                    bp.curve_to((a, b), (c, d2), (e, f));
76                }
77                i += 7;
78            }
79            "Q" => {
80                if let (Some(a), Some(b), Some(c), Some(d2)) = (num(i + 1), num(i + 2), num(i + 3), num(i + 4))
81                    && started
82                {
83                    bp.quad_to((a, b), (c, d2));
84                }
85                i += 5;
86            }
87            "Z" | "z" => {
88                if started {
89                    bp.close_path();
90                }
91                i += 1;
92            }
93            _ => i += 1,
94        }
95    }
96    bp
97}

99/// The transform a 3-D camera applies to shape `s` of size `w` x `h` (approximated by the affine
100/// map that fits it best), as drawn; `None` for no camera, and for groups, tables and charts.
101pub fn camera_turn(rctx: &Ctx, s: &Shape, w: f64, h: f64) -> Option<Affine> {
102    if matches!(s.kind, ShapeKind::Group { .. } | ShapeKind::Table(_) | ShapeKind::Chart(_)) {
103        return None;
104    }
105    let (e, _) = resolve::effects(rctx, s);
106    e.as_ref().and_then(|e| e.raw3d.as_deref()).and_then(camera::from_raw).and_then(|c| camera::affine(&c, w, h))
107}

109/// The rectangle a picture's image occupies given its crop (fractions; negative = padding).
110pub fn crop_dest(bounds: Rect, crop: [f64; 4]) -> Rect {
111    let [l, t, r, b] = crop.map(|v| if v.is_finite() { v.clamp(-10.0, 0.99) } else { 0.0 });
112    let vis_w = (1.0 - l - r).max(1e-6);
113    let vis_h = (1.0 - t - b).max(1e-6);
114    let fw = bounds.width() / vis_w;
115    let fh = bounds.height() / vis_h;
116    Rect::new(bounds.x0 - l * fw, bounds.y0 - t * fh, bounds.x0 - l * fw + fw, bounds.y0 - t * fh + fh)
117}

119/// Whether a shape's border is drawn outside its outline: a picture's is, unless the picture is
120/// drawn in 3-D (a bevel seen through a 3-D scene), where PowerPoint centres it on the edge as on

122pub(crate) fn border_outside(is_pic: bool, effects: Option<&Effects>) -> bool {
123    is_pic && !effects.is_some_and(|e| e.bevel.is_some() && e.raw3d.as_deref().is_some_and(|r| r.contains("scene3d")))

126/// The line that draws `line` (`lw` wide) outside a picture: twice as wide, so the half outside
127/// the outline is the border, with dashes kept their length. `None` for compound lines, whose
128/// bands would not survive the doubling (they stay centred on the outline).
129pub(crate) fn outside_line(line: &Line, lw: f64) -> Option<Line> {
130    if line.compound.is_some_and(|c| c != crate::model::style::Compound::Single) {
131        return None;
132    }
133    let dash = line.dash.as_ref().map(|d| {
134        let p = d.pattern();
135        crate::model::style::Dash::Custom {
136            pattern: p.chunks(2).map(|c| (c.first().copied().unwrap_or(0.0) / 2.0, c.get(1).copied().unwrap_or(0.0) / 2.0)).collect(),
137        }
138    });
139    Some(Line { width: Some(lw * 2.0), dash: dash.or_else(|| line.dash.clone()), ..line.clone() })
140}

152/// How opaque a fill is overall (`None` when nothing is filled): a solid colour's alpha, a
153/// gradient's mean stop alpha, otherwise opaque.
154pub(crate) fn fill_opacity(rctx: &Ctx, fill: &Option<Fill>, ph: Option<Rgba>) -> Option<f64> {
155    match fill {
156        None | Some(Fill::None) => None,
157        Some(Fill::Solid { color: c }) => Some(rctx.color(c, ph).a as f64 / 255.0),
158        Some(Fill::Gradient(g)) if !g.stops.is_empty() => {
159            Some(g.stops.iter().map(|s| rctx.color(&s.color, ph).a as f64 / 255.0).sum::<f64>() / g.stops.len() as f64)
160        }
161        Some(_) => Some(1.0),
162    }
163}

165/// A shadow's own scale and skew (`sx`, `sy`, `kx`, `ky`) in the shape's space, about the point
166/// of the shape's box its alignment names (`algn`: `b` is the middle of the bottom edge, `tl` the
167/// top-left corner…), as PowerPoint applies them: skewed, then scaled. A perspective shadow
168/// (`algn="b"`, `sy` 23%, `kx` -20°) so lies flat along the shape's bottom edge, not around its
169/// middle.
170pub(crate) fn shadow_local(sh: &crate::model::style::Shadow, w: f64, h: f64) -> Affine {
171    let (ax, ay) = match sh.align.as_str() {
172        "tl" => (0.0, 0.0),
173        "t" => (0.5, 0.0),
174        "tr" => (1.0, 0.0),
175        "l" => (0.0, 0.5),
176        "r" => (1.0, 0.5),
177        "bl" => (0.0, 1.0),
178        "b" => (0.5, 1.0),
179        "br" => (1.0, 1.0),
180        _ => (0.5, 0.5),
181    };
182    let a = Vec2::new(w * ax, h * ay);
183    let (tx, ty) = (sh.kx.to_radians().tan(), sh.ky.to_radians().tan());
184    let k = Affine::new([sh.sx, sh.sy * ty, sh.sx * tx, sh.sy, 0.0, 0.0]);
185    if !k.is_finite() {
186        return Affine::IDENTITY;
187    }
188    Affine::translate(a) * k * Affine::translate(-a)
189}

203/// The widest a shape may reach on screen, in pixels, and still be drawn: ten thousand screens
204/// across, far past any real deck, and still cheap for vello to flatten (#1165).
205const MAX_DEVICE_EXTENT: f64 = 1e7;
206
207/// Whether a shape's geometry can go to vello. Geometry far beyond any screen (a size at the EMU
208/// limit, group scales compounding, custom-geometry guides) makes vello's flattener size a buffer
209/// by its on-screen length, and the allocation aborts the process (#1165). Each path is checked
210/// for finite points first: a bounding box drops NaN, and a union with the frame would hide it.
211/// The extent is the control box, which always holds the curve: the tight bounding box can miss
212/// a curve's extremum when finding it overflows.
213pub(crate) fn drawable(geo: &Geometry, m: Affine, w: f64, h: f64) -> bool {
214    if !geo.paths.iter().all(|p| p.path.is_finite()) {
215        return false;
216    }
217    let dev = m.transform_rect_bbox(geo.outline().control_box().union(Rect::new(0.0, 0.0, w, h)));
218    dev.is_finite() && dev.width().max(dev.height()) <= MAX_DEVICE_EXTENT
219}
