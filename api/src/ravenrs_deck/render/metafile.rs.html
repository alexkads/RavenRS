1//! Windows Metafile and Enhanced Metafile pictures (`image/x-wmf`, `image/x-emf`): parsed by
2//! `craft-metafile` and drawn as vectors into a bitmap of the size the slide needs, so they go
3//! through the same crop, tile and recolour path as any other picture.
4//!
5//! Text items name the font the file asks for; they are drawn with the face slide text would use
6//! for that family (`FontDb::document_face`, so a missing font falls back the same way), symbol
7//! fonts mapped as bullets map them, and the file's own advances (the dx array) placing each
8//! character.
9

19/// Longest side of a metafile drawn without a target size (export, thumbnails of the picture).
20pub const DEFAULT_SIDE: u32 = 1024;
21/// Largest side drawn.
22const MAX_SIDE: u32 = 8192;
23/// Pattern cells smaller than this (output pixels) are drawn as the tile's average colour: a
24/// sub-pixel pattern would only alias.
25const MIN_CELL_PX: f64 = 1.5;
26/// Size of text whose font gives no height (LOGFONT height 0), in text units.
27const DEFAULT_EM: f64 = 12.0;
28
29/// Is `bytes` a WMF or EMF file?
30pub fn is_metafile(bytes: &[u8]) -> bool {
31    crate::craft::metafile::is_metafile(bytes)
32}

34/// Draw the metafile `bytes` into straight RGBA with its longer side `side` px (transparent where
35/// it draws nothing). `None` when it doesn't parse or has no size.
36pub fn rasterize(bytes: &[u8], side: u32) -> Option<image::RgbaImage> {
37    let mut pic = crate::craft::metafile::parse(bytes).ok()?;
38    if !pic.has_size() {
39        return None;
40    }
41    let side = side.clamp(16, MAX_SIDE) as f64;
42    let (pw, ph) = (pic.width as f64, pic.height as f64);
43    let k = side / pw.max(ph);
44    let w = (pw * k).round().clamp(1.0, MAX_SIDE as f64) as u16;
45    let h = (ph * k).round().clamp(1.0, MAX_SIDE as f64) as u16;
46    let pixmaps: Vec<Option<Arc<Pixmap>>> = pic.items.iter_mut().map(|it| it.take_pixels().and_then(|(w, h, px)| premul(w, h, &px))).collect();
47    // Pattern tiles shared by several fills are converted once.
48    let mut tiles: HashMap<*const Tile, (Option<Arc<Pixmap>>, peniko::Color)> = HashMap::new();
49    let mut ctx = RenderContext::new_with(w, h, vello_cpu::RenderSettings { num_threads: 0, ..Default::default() });
50    let r = Rect::new(0.0, 0.0, w as f64, h as f64);
51    ctx.push_clip_layer(&r.to_path(0.1));
52    for it in pic.place(r, [0.0; 4]) {
53        match it {
54            PlacedItem::Path { path, fill, stroke, line, even_odd, region, clips } => clipped(&mut ctx, region.as_deref(), &clips, |ctx| {
55                if let Some(f) = fill {
56                    ctx.set_paint(rgba(f));
57                    ctx.set_fill_rule(if even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
58                    ctx.fill_path(&path);
59                }
60                // The pen's caps, joins and dashes (GDI's default: round, solid); a cosmetic
61                // (zero-width) pen is one pixel.
62                if let Some((s, sw)) = stroke {
63                    ctx.set_paint(rgba(s));
64                    ctx.set_stroke(pen(sw.max(1.0), &line));
65                    ctx.stroke_path(&path);
66                }
67            }),
68            PlacedItem::Bitmap { index, rect, transform, region, clips } => clipped(&mut ctx, region.as_deref(), &clips, |ctx| {
69                let Some(Some(pm)) = pixmaps.get(index) else { return };
70                let (bw, bh) = (pm.width().max(1) as f64, pm.height().max(1) as f64);
71                ctx.set_paint(vello_cpu::Image {
72                    image: vello_cpu::ImageSource::Pixmap(pm.clone()),
73                    sampler: peniko::ImageSampler::default().with_quality(peniko::ImageQuality::Medium),
74                });
75                let c = transform.as_coeffs();
76                if c[1] == 0.0 && c[2] == 0.0 {
77                    ctx.set_paint_transform(Affine::translate((rect.x0, rect.y0)) * Affine::scale_non_uniform(rect.width() / bw, rect.height() / bh));
78                    ctx.fill_rect(&rect);
79                } else {
80                    // A turned or skewed bitmap: `transform` maps the unit square onto its place.
81                    ctx.set_paint_transform(transform * Affine::scale_non_uniform(1.0 / bw, 1.0 / bh));
82                    ctx.fill_path(&(transform * Rect::new(0.0, 0.0, 1.0, 1.0).to_path(0.1)));
83                }
84                ctx.reset_paint_transform();
85            }),
86            PlacedItem::Pattern { index, path, even_odd, tile, region, clips } => clipped(&mut ctx, region.as_deref(), &clips, |ctx| {
87                let Some(Item::Pattern { tile: t, .. }) = pic.items.get(index) else { return };
88                let key = Arc::as_ptr(t);
89                let (pm, avg) = tiles.entry(key).or_insert_with(|| (premul(t.width, t.height, &t.rgba), average(&t.rgba))).clone();
90                ctx.set_fill_rule(if even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
91                let cell = tile.determinant().abs().sqrt();
92                match pm {
93                    Some(pm) if cell >= MIN_CELL_PX => {
94                        ctx.set_paint(vello_cpu::Image {
95                            image: vello_cpu::ImageSource::Pixmap(pm),
96                            sampler: peniko::ImageSampler {
97                                x_extend: peniko::Extend::Repeat,
98                                y_extend: peniko::Extend::Repeat,
99                                quality: peniko::ImageQuality::Low,
100                                alpha: 1.0,
101                            },
102                        });
103                        ctx.set_paint_transform(tile);
104                        ctx.fill_path(&path);
105                        ctx.reset_paint_transform();
106                    }
107                    _ => {
108                        ctx.set_paint(avg);
109                        ctx.fill_path(&path);
110                    }
111                }
112            }),
113            PlacedItem::Text { index, transform, clip, region, clips } => clipped(&mut ctx, region.as_deref(), &clips, |ctx| {
114                let Some(Item::Text(t)) = pic.items.get(index) else { return };
115                if let Some(c) = clip {
116                    ctx.push_clip_layer(&c.to_path(0.1));
117                }
118                draw_text(ctx, t, transform);
119                if clip.is_some() {
120                    ctx.pop_layer();
121                }
122            }),
123        }
124    }
125    ctx.set_fill_rule(peniko::Fill::NonZero);
126    ctx.pop_layer();
127    ctx.flush();
128    let mut px = vec![0u8; w as usize * h as usize * 4];
129    if let Some(pm) = vello_cpu::PixmapMut::new(w, h, &mut px) {
130        ctx.render(pm, &mut Resources::new());
131    }
132    // Premultiplied to straight.
133    for p in px.as_chunks_mut::<4>().0 {
134        let a = p[3] as u32;
135        if a != 0 && a != 255 {
136            for c in p.iter_mut().take(3) {
137                *c = ((*c as u32 * 255 + a / 2) / a).min(255) as u8;
138            }
139        }
140    }
141    image::RgbaImage::from_raw(w as u32, h as u32, px)
142}

144/// Draws `draw` inside the device-context clip and every clip path. No region is the default clip.
145fn clipped(ctx: &mut RenderContext, region: Option<&BezPath>, clips: &[Arc<PlacedClip>], draw: impl FnOnce(&mut RenderContext)) {
146    if let Some(region) = region {
147        ctx.push_clip_layer(region);
148    }
149    for c in clips {
150        ctx.set_fill_rule(if c.even_odd { peniko::Fill::EvenOdd } else { peniko::Fill::NonZero });
151        ctx.push_clip_layer(&c.path);
152    }
153    if !clips.is_empty() {
154        ctx.set_fill_rule(peniko::Fill::NonZero);
155    }
156    draw(ctx);
157    for _ in 0..clips.len() + usize::from(region.is_some()) {
158        ctx.pop_layer();
159    }
160}

162/// A GDI pen as a stroke `width` wide (target units): its caps, joins, miter limit and dashes.
163fn pen(width: f64, line: &LineStyle) -> kurbo::Stroke {
164    let cap = match line.cap {
165        Cap::Round => kurbo::Cap::Round,
166        Cap::Square => kurbo::Cap::Square,
167        Cap::Flat => kurbo::Cap::Butt,
168    };
169    let join = match line.join {
170        Join::Round => kurbo::Join::Round,
171        Join::Bevel => kurbo::Join::Bevel,
172        Join::Miter => kurbo::Join::Miter,
173    };
174    let stroke = kurbo::Stroke::new(width).with_caps(cap).with_join(join).with_miter_limit(f64::from(line.miter_limit));
175    // A dash pattern with no length would never advance: such a pen draws solid.
176    let dash: Vec<f64> = line.dash.iter().map(|d| f64::from(*d)).collect();
177    if dash.iter().all(|d| d.is_finite() && *d >= 0.0) && dash.iter().sum::<f64>() > 0.0 { stroke.with_dashes(0.0, dash) } else { stroke }
178}

180/// One glyph placed in text space (x along the baseline from the string's start, font units at
181/// `scale` text units each).
182struct Glyph {
183    face: Arc<FontFace>,
184    gid: u32,
185    x: f64,
186    y: f64,
187}

189/// Draws a text item; `m` maps its text space to the output.
190fn draw_text(ctx: &mut RenderContext, t: &Text, m: Affine) {
191    let db = FontDb::global();
192    let font = &t.font;
193    let family = if font.family.trim().is_empty() { "Arial" } else { font.family.trim() };
194    let (bold, italic) = (font.weight >= 600, font.italic);
195    let style = match (bold, italic) {
196        (true, true) => "Bold Italic",
197        (true, false) => "Bold",
198        (false, true) => "Italic",
199        (false, false) => "Regular",
200    };
201    let face = db.document_face(family, style);
202    let upem = face.upem.max(1.0);
203    let (asc, desc) = win_metrics(&face);
204    // LOGFONT height: negative is the em, positive the cell (ascent plus descent).
205    let h = f64::from(font.height);
206    let em = if h < 0.0 {
207        -h
208    } else if h > 0.0 {
209        h * upem / (asc + desc).max(1.0)
210    } else {
211        DEFAULT_EM
212    };
213    let k = em / upem;
214    if !k.is_finite() || k <= 0.0 {
215        return;
216    }
217    // Characters as drawn: symbol fonts (Wingdings…) map like bullets; each character takes the
218    // face that has it.
219    let chars: Vec<char> = t.text.chars().map(|c| crate::text::symbol_char(&face, family, c).unwrap_or(c)).collect();
220    let faces: Vec<Arc<FontFace>> = chars
221        .iter()
222        .map(|c| {
223            if c.is_whitespace() || c.is_control() || face.covers(*c) { face.clone() } else { db.cascade(&[*c], std::slice::from_ref(&face), style) }
224        })
225        .collect();
226    // Shape runs of one face; record each glyph's pen position and its character.
227    let mut shaped: Vec<(Glyph, usize)> = Vec::with_capacity(chars.len());
228    let mut char_pen: Vec<f64> = vec![0.0; chars.len() + 1];
229    let mut pen = 0.0;
230    let mut i = 0;
231    while i < chars.len() {
232        let Some(f) = faces.get(i).cloned() else { break };
233        let mut j = i + 1;
234        while faces.get(j).is_some_and(|g| g.id() == f.id()) {
235            j += 1;
236        }
237        let run: String = chars.get(i..j).unwrap_or(&[]).iter().collect();
238        // Byte offset in the run → character index in the string.
239        let byte_char: Vec<usize> = run.char_indices().map(|(b, _)| b).collect();
240        let ku = f.upem.max(1.0);
241        for g in crate::fonts::shape(&f, &run, &[], |c| c) {
242            let ci = i + byte_char.partition_point(|b| *b < g.cluster).min(j - i - 1);
243            let s = em / ku;
244            shaped.push((Glyph { face: f.clone(), gid: g.gid, x: pen + f64::from(g.x_offset) * s, y: -f64::from(g.y_offset) * s }, ci));
245            pen += f64::from(g.x_advance) * s;
246            if let Some(p) = char_pen.get_mut(ci + 1) {
247                *p = pen;
248            }
249        }
250        i = j;
251    }
252    // Character start pens: a character's start is the previous one's end.
253    for ci in 1..char_pen.len() {
254        let prev = char_pen.get(ci - 1).copied().unwrap_or(0.0);
255        if let Some(p) = char_pen.get_mut(ci)
256            && *p < prev
257        {
258            *p = prev;
259        }
260    }
261    // The file's advances place each character; glyphs keep their offset within their character.
262    let use_dx = !t.dx.is_empty() && t.dx.len() == chars.len();
263    let mut width = pen;
264    let mut x_scale = 1.0;
265    if use_dx {
266        let mut starts = Vec::with_capacity(chars.len() + 1);
267        let mut x = 0.0;
268        for d in &t.dx {
269            starts.push(x);
270            x += f64::from(*d);
271        }
272        width = x;
273        for (g, ci) in &mut shaped {
274            let (start, pen_start) = (starts.get(*ci).copied().unwrap_or(0.0), char_pen.get(*ci).copied().unwrap_or(0.0));
275            g.x = start + (g.x - pen_start);
276        }
277    } else if font.width > 0.0 {
278        // A LOGFONT width narrows or widens the font against its average character width.
279        let avg = avg_char_width(&face) * k;
280        if avg > 0.0 {
281            x_scale = (f64::from(font.width) / avg).clamp(0.05, 20.0);
282            width *= x_scale;
283            for (g, _) in &mut shaped {
284                g.x *= x_scale;
285            }
286        }
287    }
288    let x0 = match t.h_align {
289        HAlign::Left => 0.0,
290        HAlign::Center => -width / 2.0,
291        HAlign::Right => -width,
292    };
293    let (a, d) = (asc * k, desc * k);
294    let base = match t.v_align {
295        VAlign::Baseline => 0.0,
296        VAlign::Top => a,
297        VAlign::Bottom => -d,
298    };
299    let at = m * Affine::translate((x0, base));
300    ctx.set_fill_rule(peniko::Fill::NonZero);
301    if let Some(bg) = t.background
302        && width > 0.0
303    {
304        ctx.set_paint(rgba(bg));
305        ctx.fill_path(&(at * Rect::new(0.0, -a, width, d).to_path(0.1)));
306    }
307    let color = rgba(t.color);
308    ctx.set_paint(color);
309    for (g, _) in &shaped {
310        let path = db.outline(&g.face, g.gid);
311        if path.elements().is_empty() {
312            continue;
313        }
314        let s = em / g.face.upem.max(1.0);
315        let skew = if italic && !g.face.italic { Affine::new([1.0, 0.0, -0.2, 1.0, 0.0, 0.0]) } else { Affine::IDENTITY };
316        let gm = at * Affine::translate((g.x, g.y)) * Affine::scale_non_uniform(x_scale, 1.0) * skew * Affine::scale(s);
317        let sc = scale_of(gm);
318        if !sc.is_finite() || sc > 1e5 {
319            continue;
320        }
321        let p: BezPath = gm * path.as_ref().clone();
322        if p.elements().len() > 10_000 || p.bounding_box().width() > 1e6 || p.bounding_box().height() > 1e6 {
323            continue;
324        }
325        ctx.fill_path(&p);
326        if bold && g.face.weight < 600.0 {
327            ctx.set_stroke(kurbo::Stroke::new(em * 0.025 * scale_of(at)).with_join(kurbo::Join::Round));
328            ctx.stroke_path(&p);
329        }
330    }
331    // Underline and strike-out: the font's own positions and thickness.
332    let metrics = face.skrifa().map(|f| f.metrics(skrifa::instance::Size::unscaled(), face.location()));
333    let deco = |on: bool, d: Option<skrifa::metrics::Decoration>, fallback: (f64, f64)| -> Option<Rect> {
334        if !on || width <= 0.0 {
335            return None;
336        }
337        let (off, th) = d.map_or(fallback, |d| (f64::from(d.offset) * k, f64::from(d.thickness) * k));
338        let th = if th > 0.0 { th } else { em / 20.0 };
339        // Offsets are y-up from the baseline; text space is y-down.
340        Some(Rect::new(0.0, -off - th / 2.0, width, -off + th / 2.0))
341    };
342    let lines = [
343        deco(font.underline, metrics.as_ref().and_then(|m| m.underline), (-em * 0.1, em / 20.0)),
344        deco(font.strike, metrics.as_ref().and_then(|m| m.strikeout), (em * 0.26, em / 20.0)),
345    ];
346    for r in lines.into_iter().flatten() {
347        ctx.fill_path(&(at * r.to_path(0.1)));
348    }
349}

351/// Mean scale of an affine transform's linear part.
352fn scale_of(m: Affine) -> f64 {
353    m.determinant().abs().sqrt()
354}

356/// The face's Windows ascent and descent (OS/2 `usWinAscent`/`usWinDescent`, as GDI measures the
357/// cell), in font units; the typographic ones when the font has no OS/2 table.
358fn win_metrics(face: &FontFace) -> (f64, f64) {
359    use skrifa::raw::TableProvider as _;
360    let os2 = face.skrifa().and_then(|f| f.os2().ok()).map(|o| (f64::from(o.us_win_ascent()), f64::from(o.us_win_descent())));
361    match os2 {
362        Some((a, d)) if a + d > 0.0 => (a, d),
363        _ => (face.ascent.max(0.0), face.descent.max(0.0)),
364    }
365}

367/// OS/2 `xAvgCharWidth` in font units (half an em without it).
368fn avg_char_width(face: &FontFace) -> f64 {
369    use skrifa::raw::TableProvider as _;
370    face.skrifa().and_then(|f| f.os2().ok()).map(|o| f64::from(o.x_avg_char_width())).filter(|w| *w > 0.0).unwrap_or(face.upem * 0.5)
371}

373fn rgba(c: [u8; 4]) -> peniko::Color {
374    peniko::Color::from_rgba8(c[0], c[1], c[2], c[3])
375}

377/// The average colour of straight RGBA pixels (alpha-weighted).
378fn average(px: &[u8]) -> peniko::Color {
379    let (mut r, mut g, mut b, mut a, mut n) = (0u64, 0u64, 0u64, 0u64, 0u64);
380    for p in px.as_chunks::<4>().0 {
381        let w = u64::from(p[3]);
382        r += u64::from(p[0]) * w;
383        g += u64::from(p[1]) * w;
384        b += u64::from(p[2]) * w;
385        a += w;
386        n += 1;
387    }
388    if a == 0 || n == 0 {
389        return peniko::Color::from_rgba8(0, 0, 0, 0);
390    }
391    peniko::Color::from_rgba8((r / a) as u8, (g / a) as u8, (b / a) as u8, (a / n) as u8)
392}

394/// Straight RGBA rows to a premultiplied pixmap.
395fn premul(w: u32, h: u32, px: &[u8]) -> Option<Arc<Pixmap>> {
396    if w == 0 || h == 0 || w > u16::MAX as u32 || h > u16::MAX as u32 || px.len() < w as usize * h as usize * 4 {
397        return None;
398    }
399    let data = px
400        .as_chunks::<4>()
401        .0
402        .iter()
403        .take(w as usize * h as usize)
404        .map(|p| {
405            let a = p[3] as u16;
406            let m = |c: u8| ((c as u16 * a + 127) / 255) as u8;
407            vello_cpu::color::PremulRgba8 { r: m(p[0]), g: m(p[1]), b: m(p[2]), a: p[3] }
408        })
409        .collect();
410    Some(Arc::new(Pixmap::from_parts(data, w as u16, h as u16)))
