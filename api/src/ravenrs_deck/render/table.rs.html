15/// Style colours for a cell: (fill, text colour, bold).
16pub struct CellLook {
17    pub fill: Option<Rgba>,
18    pub text: Option<Rgba>,
19    pub bold: bool,
20    pub border: Option<(Rgba, f64)>,
21    /// Left, right, top and bottom borders from a file-defined style (a fully transparent colour
22    /// means no border); `None` falls back to `border`.
23    pub sides: [Option<(Rgba, f64)>; 4],
24    /// A file-defined style's fill as a paint (gradients, see-through colours), with its
25    /// placeholder colour; drawn instead of `fill`.
26    pub paint: Option<(Fill, Option<Rgba>)>,
27}

29fn accent(ctx: &Ctx, style: &str) -> Rgba {
30    let slot = match style.rsplit('-').next().unwrap_or("accent1") {
31        "accent2" => SchemeSlot::Accent2,
32        "accent3" => SchemeSlot::Accent3,
33        "accent4" => SchemeSlot::Accent4,
34        "accent5" => SchemeSlot::Accent5,
35        "accent6" => SchemeSlot::Accent6,
36        "tx1" => SchemeSlot::Tx1,
37        _ => SchemeSlot::Accent1,
38    };
39    ctx.color(&ColorRef::scheme(slot), None)
40}

42/// The look of cell (`r`, `c`) from a table style the file defines: each part that applies to
43/// the cell (whole table, bands, first and last rows and columns, corners, in that order) sets the
44/// fill, text and the borders on its region's edges (its inside lines within it).
45fn look_def(ctx: &Ctx, def: &TableStyleDef, t: &Table, r: usize, c: usize) -> CellLook {
46    let (rows, cols) = (t.rows.len(), t.cols.len());
47    let (r1, c1) = t.cell(r, c).map(|cell| (r + cell.row_span.max(1) as usize - 1, c + cell.grid_span.max(1) as usize - 1)).unwrap_or((r, c));
48    let (r1, c1) = (r1.min(rows.saturating_sub(1)), c1.min(cols.saturating_sub(1)));
49    let last_r = rows.saturating_sub(1);
50    let last_c = cols.saturating_sub(1);
51    let header = t.first_row && r == 0;
52    let total = t.last_row && r1 == last_r && rows > 1;
53    let first_col = t.first_col && c == 0;
54    let last_col = t.last_col && c1 == last_c && cols > 1;
55    let body_r = r.saturating_sub(usize::from(t.first_row));
56    let body_c = c.saturating_sub(usize::from(t.first_col));
57    let mut look = CellLook { fill: None, text: None, bold: false, border: None, sides: [None; 4], paint: None };
58    for part in TablePart::ALL {
59        // (applies, region rows, region cols)
60        let (on, region) = match part {
61            TablePart::WholeTbl => (true, (0, last_r, 0, last_c)),
62            TablePart::Band1H => (t.band_row && !header && !total && body_r % 2 == 0, (r, r1, 0, last_c)),
63            TablePart::Band2H => (t.band_row && !header && !total && body_r % 2 == 1, (r, r1, 0, last_c)),
64            TablePart::Band1V => (t.band_col && !first_col && !last_col && body_c % 2 == 0, (0, last_r, c, c1)),
65            TablePart::Band2V => (t.band_col && !first_col && !last_col && body_c % 2 == 1, (0, last_r, c, c1)),
66            TablePart::LastCol => (last_col, (0, last_r, c, last_c)),
67            TablePart::FirstCol => (first_col, (0, last_r, 0, c1)),
68            TablePart::LastRow => (total, (r, last_r, 0, last_c)),
69            TablePart::FirstRow => (header, (0, r1, 0, last_c)),
70            TablePart::SeCell => (total && last_col, (r, r1, c, c1)),
71            TablePart::SwCell => (total && first_col, (r, r1, c, c1)),
72            TablePart::NeCell => (header && last_col, (r, r1, c, c1)),
73            TablePart::NwCell => (header && first_col, (r, r1, c, c1)),
74        };
75        if !on {
76            continue;
77        }
78        let Some(ps) = def.part(part) else { continue };
79        if let Some(p) = style_fill(ctx, ps) {
80            look.fill = fill_color(ctx, &p.0, p.1);
81            look.paint = Some(p);
82        }
83        if let Some(tc) = &ps.text_color {
84            look.text = Some(ctx.color(tc, None));
85        }
86        if let Some(b) = ps.bold {
87            look.bold = b;
88        }
89        let (r0, rr1, c0, cc1) = region;
90        // Left, right, top, bottom of the cell: the part's edge or its inside line.
91        let pick = [if c == c0 { 0 } else { 5 }, if c1 == cc1 { 1 } else { 5 }, if r == r0 { 2 } else { 4 }, if r1 == rr1 { 3 } else { 4 }];
92        for (side, i) in pick.iter().enumerate() {
93            if let (Some(Some(b)), Some(slot)) = (ps.borders.get(*i), look.sides.get_mut(side)) {
94                *slot = Some(border_look(ctx, b));
95            }
96        }
97    }
98    look
99}

101/// A style part's fill, a theme fill reference resolved to the theme's fill with its colour as
102/// the placeholder colour.
103fn style_fill(ctx: &Ctx, ps: &TablePartStyle) -> Option<(Fill, Option<Rgba>)> {
104    if let Some((idx, col)) = &ps.fill_ref {
105        let ph = ctx.color(col, None);
106        return Some(match ctx.master.theme.format.fills.get((*idx as usize).saturating_sub(1)) {
107            Some(f) if *idx > 0 => (f.clone(), Some(ph)),
108            _ if *idx == 0 => (Fill::None, None),
109            _ => (Fill::solid(ColorRef::rgb(ph)), None),
110        });
111    }
112    ps.fill.clone().map(|f| (f, None))
113}

115/// A fill's colour for a table cell (gradients and pictures by their first colour).
116fn fill_color(ctx: &Ctx, f: &Fill, ph: Option<Rgba>) -> Option<Rgba> {
117    match f {
118        Fill::None => Some(Rgba::TRANSPARENT),
119        Fill::Solid { color: c } => Some(ctx.color(c, ph)),
120        Fill::Gradient(g) => g.stops.first().map(|s| ctx.color(&s.color, ph)),
121        Fill::Pattern(p) => Some(ctx.color(&p.bg, ph)),
122        _ => None,
123    }
124}

126/// A style border's colour and width (a transparent colour for no line).
127fn border_look(ctx: &Ctx, b: &TableBorder) -> (Rgba, f64) {
128    match b {
129        TableBorder::Line { line } => match &line.fill {
130            Some(Fill::Solid { color: c }) => (ctx.color(c, None), line.width.unwrap_or(0.75)),
131            Some(Fill::Gradient(g)) => (g.stops.first().map(|s| ctx.color(&s.color, None)).unwrap_or(Rgba::BLACK), line.width.unwrap_or(0.75)),
132            _ => (Rgba::TRANSPARENT, 0.0),
133        },
134        TableBorder::Ref { idx, color } => {
135            let w = ctx.master.theme.format.lines.get((*idx as usize).saturating_sub(1)).and_then(|l| l.width).unwrap_or(0.75);
136            if *idx == 0 { (Rgba::TRANSPARENT, 0.0) } else { (ctx.color(color, None), w) }
137        }
138    }
139}

150/// Which parts of the table a cell sits in, for the preset styles.
151#[derive(Clone, Copy)]
152struct PresetPart {
153    header: bool,
154    total: bool,
155    first_col: bool,
156    last_col: bool,
157    banded: bool,
158}
159
160impl PresetPart {
161    /// In a header, total, first or last column: the parts a preset style makes strong.
162    fn strong(self) -> bool {
163        self.header || self.total || self.first_col || self.last_col
164    }
165}

167/// Tint `k` of accent `a`.
168fn tint(a: Rgba, k: i32) -> Rgba {
169    crate::color::apply(a, &[ColorTransform::Tint(k)])
170}

172fn light1_look(a: Rgba, p: PresetPart, on: impl Fn(Rgba) -> Rgba) -> CellLook {
173    let fill = p.banded.then(|| tint(a, 20000));
174    CellLook { fill, text: fill.map(on), bold: p.strong(), border: if p.header { Some((a, 1.0)) } else { None }, sides: [None; 4], paint: None }
175}

177fn light2_look(a: Rgba, p: PresetPart) -> CellLook {
178    CellLook {
179        fill: if p.header { Some(a) } else { None },
180        text: if p.header { Some(Rgba::WHITE) } else { None },
181        bold: p.header || p.total || p.first_col,
182        border: Some((a, 1.0)),
183        sides: [None; 4],
184        paint: None,
185    }
186}

188fn dark1_look(a: Rgba, p: PresetPart) -> CellLook {
189    CellLook {
190        fill: Some(if p.header {
191            Rgba::BLACK
192        } else if p.banded {
193            crate::color::apply(a, &[ColorTransform::Shade(60000)])
194        } else {
195            crate::color::apply(a, &[ColorTransform::Shade(75000)])
196        }),
197        text: Some(Rgba::WHITE),
198        bold: p.header || p.total || p.first_col,
199        border: Some((Rgba::WHITE, 1.0)),
200        sides: [None; 4],
201        paint: None,
202    }
203}

205fn medium4_look(a: Rgba, p: PresetPart, on: impl Fn(Rgba) -> Rgba) -> CellLook {
206    let fill = if p.header || p.banded { tint(a, 40000) } else { tint(a, 20000) };
207    CellLook { fill: Some(fill), text: Some(on(fill)), bold: p.strong(), border: Some((a, 0.75)), sides: [None; 4], paint: None }
208}

210/// medium2: strong header, banded light tints, white grid.
211fn medium2_look(a: Rgba, p: PresetPart, on: impl Fn(Rgba) -> Rgba) -> CellLook {
212    let strong = p.strong();
213    let fill = if strong {
214        a
215    } else if p.banded {
216        tint(a, 40000)
217    } else {
218        tint(a, 20000)
219    };
220    CellLook {
221        fill: Some(fill),
222        text: Some(if strong { Rgba::WHITE } else { on(fill) }),
223        bold: strong,
224        border: Some((Rgba::WHITE, 1.0)),
225        sides: [None; 4],
226        paint: None,
227    }
228}

230pub fn look_with_def(ctx: &Ctx, def: Option<&TableStyleDef>, t: &Table, r: usize, c: usize) -> CellLook {
231    if let Some(def) = def {
232        return look_def(ctx, def, t, r, c);
233    }
234    let a = accent(ctx, &t.style);
235    let kind = t.style.split('-').next().unwrap_or("medium2");
236    let header = t.first_row && r == 0;
237    let total = t.last_row && r + 1 == t.rows.len();
238    let first_col = t.first_col && c == 0;
239    let last_col = t.last_col && c + 1 == t.cols.len();
240    let body_r = if t.first_row { r.wrapping_sub(1) } else { r };
241    let band = t.band_row && !header && !total && body_r % 2 == 0;
242    let band_c = t.band_col && c.is_multiple_of(2);
243    let p = PresetPart { header, total, first_col, last_col, banded: band || band_c };
244    let text_color = ctx.color(&ColorRef::scheme(SchemeSlot::Tx1), None);
245    // Text on a tinted cell never inherits the text colour, which a dark theme makes light and
246    // unreadable on the light tint: PowerPoint's medium styles set it to the theme's dark colour
247    // (dk1). A dark theme may keep that in dk1 and map tx1 away from it, or (Nocturne) put it in
248    // lt1, so of the two the one contrasting more with the cell's fill is used.
249    let dk1 = ctx.color(&ColorRef::scheme(SchemeSlot::Dk1), None);
250    let lt1 = ctx.color(&ColorRef::scheme(SchemeSlot::Lt1), None);
251    let on = |fill: Rgba| if dk1.contrast(fill) >= lt1.contrast(fill) { dk1 } else { lt1 };
252    match kind {
253        "none" => CellLook { fill: None, text: None, bold: false, border: None, sides: [None; 4], paint: None },
254        "grid" => CellLook { fill: None, text: None, bold: false, border: Some((text_color, 0.75)), sides: [None; 4], paint: None },
255        "light1" => light1_look(a, p, on),
256        "light2" => light2_look(a, p),
257        "dark1" => dark1_look(a, p),
258        "medium4" => medium4_look(a, p, on),
259        _ => medium2_look(a, p, on),
260    }
261}

263/// Running edges 0, s0, s0+s1, … of a list of sizes (negative sizes count as 0).
264fn edges(sizes: &[f64]) -> Vec<f64> {
265    std::iter::once(0.0)
266        .chain(sizes.iter().scan(0.0, |a, s| {
267            *a += s.max(0.0);
268            Some(*a)
269        }))
270        .collect()
271}

278pub fn cell_body_with_def(rctx: &Ctx, def: Option<&TableStyleDef>, t: &Table, r: usize, c: usize, cell: &Cell) -> TextBody {
279    let lk = look_with_def(rctx, def, t, r, c);
280    let mut body = cell.text.clone();
281    let [ml, mr, mt, mb] = cell.margins.unwrap_or([7.2, 7.2, 3.6, 3.6]);
282    body.body.inset_l = Some(ml);
283    body.body.inset_r = Some(mr);
284    body.body.inset_t = Some(mt);
285    body.body.inset_b = Some(mb);
286    body.body.anchor = Some(cell.anchor.unwrap_or(Anchor::Top));
287    if cell.vertical.is_some() {
288        body.body.vert = cell.vertical;
289    }
290    for p in &mut body.paragraphs {
291        for run in &mut p.runs {
292            if run.props.fill.is_none()
293                && let Some(tc) = lk.text
294            {
295                run.props.fill = Some(Fill::solid(ColorRef::rgb(tc)));
296            }
297            if lk.bold && run.props.bold.is_none() {
298                run.props.bold = Some(true);
299            }
300        }
301    }
302    body
303}

305/// Heights the rows are drawn at. A stored row height (`a:tr/@h`) is a minimum: the row grows so the
306/// text of each of its cells fits, as in PowerPoint. Files from pandoc or python-pptx store 0.
307/// Cells spanning several rows and vertical text keep the stored height.
308pub fn row_heights(rctx: &Ctx, t: &Table, fields: &dyn Fields) -> Vec<f64> {
309    row_heights_with_def(rctx, resolve_style(rctx, t), t, fields)
310}
311
312/// [`row_heights`] with the table's style already resolved, for a caller that needs it too.
313fn row_heights_with_def(rctx: &Ctx, style_def: Option<&TableStyleDef>, t: &Table, fields: &dyn Fields) -> Vec<f64> {
314    let xs = edges(&t.cols);
315    let mut out = Vec::with_capacity(t.rows.len());
316    for (ri, row) in t.rows.iter().enumerate() {
317        let mut h = row.height.max(0.0);
318        for (ci, cell) in row.cells.iter().enumerate().take(t.cols.len()) {
319            if cell.is_covered() || cell.row_span > 1 || !matches!(cell.vertical, None | Some(TextDir::Horizontal)) {
320                continue;
321            }
322            let c1 = ci.saturating_add(cell.grid_span.max(1) as usize).min(t.cols.len());
323            let (Some(x0), Some(x1)) = (xs.get(ci), xs.get(c1)) else { continue };
324            let body = cell_body_with_def(rctx, style_def, t, ri, ci, cell);
325            let tmp = Shape::table_cell(body.clone());
326            let need = crate::text::fit_height(rctx, &tmp, &body, Rect::new(*x0, 0.0, *x1, h.max(1.0)), fields);
327            if need.is_finite() {
328                h = h.max(need.min(20_000.0));
329            }
330        }
331        out.push(h);
332    }
333    out
334}

336/// Cell rectangles in table-local space, honouring merges: (row, col, rect) for visible cells.
337/// `heights` are the drawn row heights ([`row_heights`]).
338pub fn cell_rects(t: &Table, heights: &[f64]) -> Vec<(usize, usize, Rect)> {
339    let mut out = vec![];
340    let xs = edges(&t.cols);
341    let ys = edges(heights);
342    for (ri, row) in t.rows.iter().enumerate() {
343        for (ci, cell) in row.cells.iter().enumerate().take(t.cols.len()) {
344            if cell.is_covered() {
345                continue;
346            }
347            let c1 = (ci + cell.grid_span.max(1) as usize).min(t.cols.len());
348            let r1 = (ri + cell.row_span.max(1) as usize).min(t.rows.len());
349            let (Some(x0), Some(x1), Some(y0), Some(y1)) = (xs.get(ci), xs.get(c1), ys.get(ri), ys.get(r1)) else { continue };
350            out.push((ri, ci, Rect::new(*x0, *y0, *x1, *y1)));
351        }
352    }
353    out
354}

356/// Can a cell's text be left undrawn because it lands wholly outside the `target` (pixels)? Only
357/// for cells whose text stays inside the cell: its row grew to fit it (one row, horizontal text,
358/// not at the height cap), so nothing overflows. A margin covers italic overhang and the like.
359/// A table of thousands of rows otherwise lays out every cell's text off the slide's bottom.
360fn text_off_target(cell: &Cell, rect: Rect, m: Affine, target: (u16, u16)) -> bool {
361    if cell.row_span > 1 || !matches!(cell.vertical, None | Some(TextDir::Horizontal)) || rect.height() >= 20_000.0 {
362        return false;
363    }
364    let b = m.transform_rect_bbox(rect).inflate(64.0, 64.0);
365    let view = Rect::new(0.0, 0.0, f64::from(target.0), f64::from(target.1));
366    b.is_finite() && (b.x1 < view.x0 || b.y1 < view.y0 || b.x0 > view.x1 || b.y0 > view.y1)
367}

375/// Draws the table; returns the box its cells cover (table space), which grows with their text.
376#[allow(clippy::too_many_arguments)]
377pub fn draw(
378    ctx: &mut RenderContext,
379    pres: &Presentation,
380    rctx: &Ctx,
381    t: &Table,
382    m: Affine,
383    _w: f64,
384    _h: f64,
385    fields: &dyn crate::text::Fields,
386    editing: bool,
387) -> Option<kurbo::Rect> {
388    let _ = pres;
389    let style_def = resolve_style(rctx, t);
390    let rects = cell_rects(t, &row_heights_with_def(rctx, style_def, t, fields));
391    let covered = rects.iter().map(|(_, _, r)| *r).reduce(|a, b| a.union(b));
392    let style_bg = style_def.and_then(|d| d.background.as_ref()).and_then(|b| style_fill(rctx, b));
393    let own_bg = t.background.clone().map(|f| (f, None));
394    if let Some(all) = rects.iter().map(|(_, _, r)| *r).reduce(|a, b| a.union(b)) {
395        for (bg, ph) in style_bg.into_iter().chain(own_bg) {
396            ctx.set_transform(m);
397            crate::render::paint::fill_path(ctx, rctx, &bg, ph, &all.to_path(0.1), all, m, crate::geom::preset::FillMode::Norm, 1.0, rctx.pres);
398        }
399    }
400    for (r, c, rect) in &rects {
401        let Some(cell) = t.cell(*r, *c) else { continue };
402        let lk = look_with_def(rctx, style_def, t, *r, *c);
403        ctx.set_transform(m);
404        match &cell.fill {
405            Some(f) => {
406                crate::render::paint::fill_path(ctx, rctx, f, None, &rect.to_path(0.1), *rect, m, crate::geom::preset::FillMode::Norm, 1.0, rctx.pres)
407            }
408            None => match &lk.paint {
409                Some((f, ph)) => crate::render::paint::fill_path(
410                    ctx,
411                    rctx,
412                    f,
413                    *ph,
414                    &rect.to_path(0.1),
415                    *rect,
416                    m,
417                    crate::geom::preset::FillMode::Norm,
418                    1.0,
419                    rctx.pres,
420                ),
421                None => {
422                    if let Some(fc) = lk.fill.filter(|f| f.a > 0) {
423                        ctx.set_paint(color(fc, 1.0));
424                        ctx.fill_rect(rect);
425                    }
426                }
427            },
428        }
429    }
430    // Diagonal borders, under the edges: drawn like any line (dashes, compound, caps).
431    for (r, c, rect) in &rects {
432        let Some(cell) = t.cell(*r, *c) else { continue };
433        for (line, a, b) in [(&cell.diag_down, (rect.x0, rect.y0), (rect.x1, rect.y1)), (&cell.diag_up, (rect.x0, rect.y1), (rect.x1, rect.y0))] {
434            let Some(l) = line.as_ref().filter(|l| l.fill.as_ref().is_some_and(|f| !f.is_none())) else { continue };
435            let mut p = BezPath::new();
436            p.move_to(a);
437            p.line_to(b);
438            crate::render::stroke(ctx, rctx, l, None, &p, None, m, l.width.unwrap_or(1.0).max(0.0), 1.0);
439        }
440    }
441    // Borders: explicit cell borders, else the style's grid.
442    for (r, c, rect) in &rects {
443        let Some(cell) = t.cell(*r, *c) else { continue };
444        let lk = look_with_def(rctx, style_def, t, *r, *c);
445        let edges = [
446            (rect.x0, rect.y0, rect.x0, rect.y1),
447            (rect.x1, rect.y0, rect.x1, rect.y1),
448            (rect.x0, rect.y0, rect.x1, rect.y0),
449            (rect.x0, rect.y1, rect.x1, rect.y1),
450        ];
451        for (i, e) in edges.iter().enumerate() {
452            let line = cell.borders.get(i).and_then(|b| b.as_ref());
453            let (col, wd) = match line {
454                // An explicit border is drawn like any line: dashed, compound, gradient.
455                Some(l) => {
456                    if l.fill.as_ref().is_some_and(|f| !f.is_none()) {
457                        let mut p = BezPath::new();
458                        p.move_to((e.0, e.1));
459                        p.line_to((e.2, e.3));
460                        crate::render::stroke(ctx, rctx, l, None, &p, None, m, l.width.unwrap_or(1.0).max(0.25), 1.0);
461                    }
462                    continue;
463                }
464                None => match lk.sides.get(i).copied().flatten().or(lk.border) {
465                    Some(b) if b.0.a > 0 && b.1 > 0.0 => b,
466                    _ => continue,
467                },
468            };
469            let mut p = BezPath::new();
470            p.move_to((e.0, e.1));
471            p.line_to((e.2, e.3));
472            ctx.set_transform(m);
473            ctx.set_paint(color(col, 1.0));
474            ctx.set_stroke(kurbo::Stroke::new(wd.max(0.25)));
475            ctx.stroke_path(&p);
476        }
477    }
478    // Text.
479    for (r, c, rect) in &rects {
480        let Some(cell) = t.cell(*r, *c) else { continue };
481        if (cell.text.is_empty() && !editing) || text_off_target(cell, *rect, m, (ctx.width(), ctx.height())) {
482            continue;
483        }
484        let body = cell_body_with_def(rctx, style_def, t, *r, *c, cell);
485        let tmp = Shape::table_cell(body.clone());
486        #[cfg(test)]
487        TEXT_CELLS_DRAWN.with(|n| n.set(n.get() + 1));
488        let mut l = crate::text::layout(rctx, &tmp, &body, &crate::text::Opts { rect: *rect, fields, prompt_color: None, no_shrink: false });
489        if editing {
490            crate::render::preview_empty_bullets(&mut l);
491        }
492        crate::render::draw_layout(ctx, &l, *rect, m);
493        let _ = peniko::Fill::NonZero;
494    }
495    covered
