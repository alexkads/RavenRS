1//! Gradient colour ramps as PowerPoint draws them.
2//!
3//! A gradient of exactly two stops, at 0 % and 100 % (what a two-colour gradient is), is not
4//! blended linearly in sRGB by PowerPoint: it mixes the colours in linear light, and the second
5//! colour's share follows an S-shaped curve of the position. Black to white is 105 (not 64) a
6//! quarter of the way, 186 (not 128) half way and 238 (not 191) three quarters of the way, in any
7//! direction, on shapes and on slide backgrounds. Gradients with three or more stops, or with a
8//! stop moved off an end, blend linearly in sRGB (measured from PowerPoint's own slide export).
9//! The renderer interpolates linearly in sRGB, so a two-colour gradient is expanded into short
10//! pieces whose colours follow PowerPoint's ramp.
11
12use crate::color::Rgba;
13
14/// The next stop's share of the linear-light mix at sixteenths of a segment (0 at the first stop,
15/// 1 at the second). Measured, then made symmetric: `w(1 - t) = 1 - w(t)`.
16const WEIGHT: [f64; 17] =
17    [0.0, 0.0205, 0.0469, 0.0878, 0.1431, 0.2143, 0.2989, 0.3972, 0.5, 0.6028, 0.7011, 0.7857, 0.8569, 0.9122, 0.9531, 0.9795, 1.0];
18
19/// Pieces each segment between two stops is cut into.
20const PIECES: usize = 16;
21
22/// The next stop's share at `t` (0..1) between two stops.
23pub fn weight(t: f64) -> f64 {
24    if !t.is_finite() {
25        return 0.0;
26    }
27    let x = t.clamp(0.0, 1.0) * (WEIGHT.len() - 1) as f64;
28    let i = (x.floor() as usize).min(WEIGHT.len() - 2);
29    let f = x - i as f64;
30    let (a, b) = (WEIGHT.get(i).copied().unwrap_or(0.0), WEIGHT.get(i + 1).copied().unwrap_or(1.0));
31    a + (b - a) * f
32}
33
34fn to_linear(v: u8) -> f64 {
35    let v = v as f64 / 255.0;
36    if v <= 0.04045 { v / 12.92 } else { ((v + 0.055) / 1.055).powf(2.4) }
37}
38
39fn to_srgb(v: f64) -> u8 {
40    let v = v.clamp(0.0, 1.0);
41    let s = if v <= 0.0031308 { v * 12.92 } else { 1.055 * v.powf(1.0 / 2.4) - 0.055 };
42    (s * 255.0).round().clamp(0.0, 255.0) as u8
43}
44
45/// The colour at `t` between stops `a` and `b`. Alpha mixes by the same share.
46pub fn mix(a: Rgba, b: Rgba, t: f64) -> Rgba {
47    let w = weight(t);
48    let ch = |x: u8, y: u8| to_srgb(to_linear(x) + (to_linear(y) - to_linear(x)) * w);
49    Rgba { r: ch(a.r, b.r), g: ch(a.g, b.g), b: ch(a.b, b.b), a: (a.a as f64 + (b.a as f64 - a.a as f64) * w).round().clamp(0.0, 255.0) as u8 }
50}
51
52/// Sorted stops (position 0..1, colour) as a renderer interpolating linearly in sRGB should get
53/// them: a two-colour gradient (two stops, at 0 and 1) expanded to follow PowerPoint's ramp,
54/// anything else unchanged.
55pub fn expand(stops: &[(f64, Rgba)]) -> Vec<(f64, Rgba)> {
56    let two_colour = matches!(stops, [(p, _), (q, _)] if *p <= 1e-6 && *q >= 1.0 - 1e-6);
57    if !two_colour {
58        return stops.to_vec();
59    }
60    let mut out = Vec::with_capacity(stops.len() * PIECES + 1);
61    for (i, &(p, c)) in stops.iter().enumerate() {
62        out.push((p, c));
63        let Some(&(q, d)) = stops.get(i + 1) else { continue };
64        if q - p <= 1e-6 || c == d {
65            continue;
66        }
67        for k in 1..PIECES {
68            let t = k as f64 / PIECES as f64;
69            out.push((p + (q - p) * t, mix(c, d, t)));
70        }
71    }
72    out
73}

75#[cfg(test)]
76mod tests {
77    use super::*;
78
79    #[test]
80    fn black_to_white_follows_powerpoint() {
81        // PowerPoint's export: 105 at a quarter, 186 half way, 238 at three quarters.
82        let (k, w) = (Rgba::rgb(0, 0, 0), Rgba::rgb(255, 255, 255));
83        for (t, want) in [(0.25, 105), (0.5, 186), (0.75, 238)] {
84            let got = mix(k, w, t).r as i32;
85            assert!((got - want).abs() <= 3, "{t}: {got} vs {want}");
86        }
87    }
88
89    #[test]
90    fn colour_pairs_follow_powerpoint() {
91        // Blue (4472C4) to orange (ED7D31) half way: PowerPoint draws (178, 120, 146).
92        let m = mix(Rgba::rgb(68, 114, 196), Rgba::rgb(237, 125, 49), 0.5);
93        assert!((m.r as i32 - 178).abs() <= 3 && (m.g as i32 - 120).abs() <= 3 && (m.b as i32 - 146).abs() <= 3, "{m:?}");
94        // Red to blue a quarter of the way: (238, 0, 105).
95        let m = mix(Rgba::rgb(255, 0, 0), Rgba::rgb(0, 0, 255), 0.25);
96        assert!((m.r as i32 - 238).abs() <= 3 && (m.b as i32 - 105).abs() <= 3, "{m:?}");
97    }
98
99    #[test]
100    fn the_ends_are_the_stops_and_the_weight_is_symmetric() {
101        let (a, b) = (Rgba::rgb(10, 200, 30), Rgba::rgb(240, 20, 90));
102        assert_eq!(mix(a, b, 0.0), a);
103        assert_eq!(mix(a, b, 1.0), b);
104        for k in 0..=32 {
105            let t = k as f64 / 32.0;
106            assert!((weight(t) + weight(1.0 - t) - 1.0).abs() < 1e-9);
107            assert!(weight(t) <= weight((t + 0.01).min(1.0)), "monotonic");
108        }
109        assert_eq!(weight(f64::NAN), 0.0);
110    }
111
112    #[test]
113    fn expand_keeps_stops_and_hard_edges() {
114        let (r, g) = (Rgba::rgb(255, 0, 0), Rgba::rgb(0, 255, 0));
115        // Three or more stops, or a stop off an end, blend linearly in sRGB: unchanged.
116        let e = expand(&[(0.0, r), (0.5, r), (0.5, g), (1.0, g)]);
117        assert_eq!(e, vec![(0.0, r), (0.5, r), (0.5, g), (1.0, g)]);
118        assert_eq!(expand(&[(0.2, r), (1.0, g)]), vec![(0.2, r), (1.0, g)]);
119        assert_eq!(expand(&[(0.0, r), (0.5, g), (1.0, r)]).len(), 3);
120        // Two equal colours add nothing.
121        assert_eq!(expand(&[(0.0, r), (1.0, r)]), vec![(0.0, r), (1.0, r)]);
122        let e = expand(&[(0.0, r), (1.0, g)]);
123        assert_eq!(e.len(), PIECES + 1);
124        assert_eq!(e.first(), Some(&(0.0, r)));
125        assert_eq!(e.last(), Some(&(1.0, g)));
126        assert!(e.windows(2).all(|w| w[0].0 <= w[1].0));
127    }
