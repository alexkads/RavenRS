1//! Compound outlines (`a:ln/@cmpd`): double, thick-thin, thin-thick and triple lines.
2//!
3//! PowerPoint splits the line's width into bands across the path, measured from its left side
4//! (looking along the path) to its right, as these shares of the width (read from PowerPoint's
5//! export of 24 pt lines of each kind):
6//! - double: 1/3 line, 1/3 gap, 1/3 line;
7//! - thin-thick: 1/5 line, 1/5 gap, 3/5 line; thick-thin the mirror;
8//! - triple: 1/6 line, 1/6 gap, 1/3 line, 1/6 gap, 1/6 line.
9//!
10//! On a shape drawn clockwise (all presets) the left side is the outside.
11
12use crate::model::style::Compound;
13use kurbo::{BezPath, CubicBez, ParamCurve, PathEl, PathSeg, Point, Vec2};
14
15/// The bands of `c` as (from, to) shares of the width, left side first.
16pub fn bands(c: Compound) -> &'static [(f64, f64)] {
17    match c {
18        Compound::Single => &[(0.0, 1.0)],
19        Compound::Double => &[(0.0, 1.0 / 3.0), (2.0 / 3.0, 1.0)],
20        Compound::ThinThick => &[(0.0, 0.2), (0.4, 1.0)],
21        Compound::ThickThin => &[(0.0, 0.6), (0.8, 1.0)],
22        Compound::Triple => &[(0.0, 1.0 / 6.0), (2.0 / 6.0, 4.0 / 6.0), (5.0 / 6.0, 1.0)],
23    }
24}

26/// Unit normal pointing to the right of travel (y down), or None for a zero-length tangent.
27fn right(t: Vec2) -> Option<Vec2> {
28    let l = t.hypot();
29    (l > 1e-12 && l.is_finite()).then(|| Vec2::new(-t.y, t.x) / l)
30}

32/// `path` moved `d` to the right of travel (negative: to the left). Line corners meet at their
33/// miter point (bevelled when sharper than about 11°); curves are offset by kurbo; other joins are
34/// bridged by a straight line.
35pub fn offset(path: &BezPath, d: f64, tolerance: f64) -> BezPath {
36    let mut out = BezPath::new();
37    if d.abs() < 1e-9 {
38        return path.clone();
39    }
40    // Each subpath as segments, offset one by one.
41    let mut segs: Vec<PathSeg> = Vec::new();
42    let mut closed = false;
43    let flush = |segs: &mut Vec<PathSeg>, closed: bool, out: &mut BezPath| {
44        let pieces: Vec<(PathSeg, BezPath)> = segs.iter().filter_map(|s| offset_seg(*s, d, tolerance).map(|p| (*s, p))).collect();
45        let begin = out.elements().len();
46        for (i, (seg, piece)) in pieces.iter().enumerate() {
47            let Some(PathEl::MoveTo(p)) = piece.elements().first().copied() else { continue };
48            match i.checked_sub(1).and_then(|j| pieces.get(j)) {
49                None => out.move_to(p),
50                Some((prev, _)) => join(out, *prev, *seg, p, d),
51            }
52            for e in piece.elements().iter().skip(1) {
53                out.push(*e);
54            }
55        }
56        if closed && let (Some((last, _)), Some((first, piece))) = (pieces.last(), pieces.first()) {
57            if let Some(PathEl::MoveTo(p)) = piece.elements().first().copied() {
58                // Lines meeting at the start: both ends move to their miter point.
59                match miter(*last, *first, d) {
60                    Some(m) if is_line(*last) && is_line(*first) => {
61                        set_last(out, m);
62                        if let Some(PathEl::MoveTo(_)) = out.elements().get(begin) {
63                            out.elements_mut()[begin] = PathEl::MoveTo(m);
64                        }
65                    }
66                    _ => out.line_to(p),
67                }
68            }
69            out.close_path();
70        }
71        segs.clear();
72    };
73    let mut start = Point::ZERO;
74    let mut cur = Point::ZERO;
75    for el in path.elements() {
76        match *el {
77            PathEl::MoveTo(p) => {
78                flush(&mut segs, closed, &mut out);
79                closed = false;
80                start = p;
81                cur = p;
82            }
83            PathEl::LineTo(p) => {
84                segs.push(PathSeg::Line(kurbo::Line::new(cur, p)));
85                cur = p;
86            }
87            PathEl::QuadTo(a, b) => {
88                segs.push(PathSeg::Quad(kurbo::QuadBez::new(cur, a, b)));
89                cur = b;
90            }
91            PathEl::CurveTo(a, b, c) => {
92                segs.push(PathSeg::Cubic(CubicBez::new(cur, a, b, c)));
93                cur = c;
94            }
95            PathEl::ClosePath => {
96                if (cur - start).hypot() > 1e-9 {
97                    segs.push(PathSeg::Line(kurbo::Line::new(cur, start)));
98                }
99                closed = true;
100                flush(&mut segs, closed, &mut out);
101                closed = false;
102                cur = start;
103            }
104        }
105    }
106    flush(&mut segs, closed, &mut out);
107    out
108}

110fn offset_seg(s: PathSeg, d: f64, tolerance: f64) -> Option<BezPath> {
111    // Far-off or non-finite geometry (hostile files) is left out rather than offset.
112    let sane = |p: Point| p.is_finite() && p.x.abs() < 1e7 && p.y.abs() < 1e7;
113    let ok = match s {
114        PathSeg::Line(l) => sane(l.p0) && sane(l.p1),
115        PathSeg::Quad(q) => sane(q.p0) && sane(q.p1) && sane(q.p2),
116        PathSeg::Cubic(c) => sane(c.p0) && sane(c.p1) && sane(c.p2) && sane(c.p3),
117    };
118    if !ok || !d.is_finite() {
119        return None;
120    }
121    match s {
122        PathSeg::Line(l) => {
123            let n = right(l.p1 - l.p0)?;
124            let mut p = BezPath::new();
125            p.move_to(l.p0 + n * d);
126            p.line_to(l.p1 + n * d);
127            Some(p)
128        }
129        PathSeg::Quad(q) => offset_seg(PathSeg::Cubic(q.raise()), d, tolerance),
130        PathSeg::Cubic(c) => {
131            // Nearly straight cubics (collinear control points) are offset as lines.
132            let chord = c.p3 - c.p0;
133            let flat = |p: Point| chord.cross(p - c.p0).abs() <= 1e-9 * chord.hypot2().max(1e-18);
134            if flat(c.p1) && flat(c.p2) {
135                return offset_seg(PathSeg::Line(kurbo::Line::new(c.p0, c.p3)), d, tolerance);
136            }
137            let mut p = BezPath::new();
138            kurbo::offset::offset_cubic(c, d, tolerance, &mut p);
139            Some(p)
140        }
141    }
142}

144/// The segment's tangent at its start and at its end (the first non-zero control vector).
145fn tangents(s: PathSeg) -> (Vec2, Vec2) {
146    let first = |v: &[Vec2]| v.iter().copied().find(|v| v.hypot2() > 1e-18).unwrap_or(Vec2::ZERO);
147    match s {
148        PathSeg::Line(l) => (l.p1 - l.p0, l.p1 - l.p0),
149        PathSeg::Quad(q) => (first(&[q.p1 - q.p0, q.p2 - q.p0]), first(&[q.p2 - q.p1, q.p2 - q.p0])),
150        PathSeg::Cubic(c) => (first(&[c.p1 - c.p0, c.p2 - c.p0, c.p3 - c.p0]), first(&[c.p3 - c.p2, c.p3 - c.p1, c.p3 - c.p0])),
151    }
152}

154fn is_line(s: PathSeg) -> bool {
155    matches!(s, PathSeg::Line(_))
156}

158/// Move the path's last point to `p`.
159fn set_last(out: &mut BezPath, p: Point) {
160    if let Some(e) = out.elements_mut().last_mut() {
161        match e {
162            PathEl::MoveTo(q) | PathEl::LineTo(q) => *q = p,
163            PathEl::QuadTo(_, q) => *q = p,
164            PathEl::CurveTo(_, _, q) => *q = p,
165            PathEl::ClosePath => {}
166        }
167    }
168}

170/// Where the offsets of `prev` and `next` meet at their shared corner (their miter point), or
171/// None for a corner sharper than about 11 degrees or a degenerate tangent.
172fn miter(prev: PathSeg, next: PathSeg, d: f64) -> Option<Point> {
173    let (_, t0) = tangents(prev);
174    let (t1, _) = tangents(next);
175    let (n0, n1) = (right(t0)?, right(t1)?);
176    let denom = 1.0 + n0.dot(n1);
177    (denom > 0.02).then(|| prev.end() + (n0 + n1) * (d / denom)).filter(|m| m.is_finite())
178}
179
180/// Join the offset of `prev` (ending at `out`'s current point) to the offset of `next`, which
181/// starts at `p`: two lines meet at their miter point (both ends moved there, so an inner corner
182/// doesn't loop); anything else is bridged straight.
183fn join(out: &mut BezPath, prev: PathSeg, next: PathSeg, p: Point, d: f64) {
184    match miter(prev, next, d) {
185        Some(m) if is_line(prev) && is_line(next) => set_last(out, m),
186        Some(m) => {
187            out.line_to(m);
188            out.line_to(p);
189        }
190        None => out.line_to(p),
191    }
192}

194#[cfg(test)]
195mod tests {
196    use super::*;
197    use kurbo::Shape;

199    #[test]
200    fn bands_split_the_width_like_powerpoint() {
201        for c in [Compound::Single, Compound::Double, Compound::ThinThick, Compound::ThickThin, Compound::Triple] {
202            let b = bands(c);
203            assert_eq!(b.first().map(|x| x.0), Some(0.0));
204            assert_eq!(b.last().map(|x| x.1), Some(1.0));
205            assert!(b.windows(2).all(|w| w[0].1 < w[1].0), "{c:?}: bands in order with gaps");
206        }
207        // Thin-thick: the thin line on the left, thick-thin its mirror.
208        let (tt, th) = (bands(Compound::ThinThick), bands(Compound::ThickThin));
209        assert!((tt[0].1 - tt[0].0) < (tt[1].1 - tt[1].0));
210        assert!((th[0].1 - th[0].0) > (th[1].1 - th[1].0));
211    }
212
213    #[test]
214    fn offsetting_a_rectangle_grows_or_shrinks_it() {
215        // Clockwise in y-down coordinates: the left of travel is the outside.
216        let r = kurbo::Rect::new(0.0, 0.0, 100.0, 50.0).to_path(0.1);
217        let out = offset(&r, -5.0, 0.01).bounding_box();
218        assert!(
219            (out.x0 + 5.0).abs() < 1e-6 && (out.y0 + 5.0).abs() < 1e-6 && (out.x1 - 105.0).abs() < 1e-6 && (out.y1 - 55.0).abs() < 1e-6,
220            "{out:?}"
221        );
222        let inn = offset(&r, 5.0, 0.01).bounding_box();
223        assert!((inn.x0 - 5.0).abs() < 1e-6 && (inn.y1 - 45.0).abs() < 1e-6, "{inn:?}");
224        // Corners meet at their miter points: the area is the offset rectangle's.
225        assert!((offset(&r, -5.0, 0.01).area().abs() - 110.0 * 60.0).abs() < 1e-6);
226    }
227
228    #[test]
229    fn offsetting_a_circle_changes_its_radius() {
230        let c = kurbo::Circle::new((50.0, 50.0), 20.0).to_path(0.01);
231        let b = offset(&c, 4.0, 0.01).bounding_box();
232        let w = b.width() / 2.0;
233        assert!((w - 16.0).abs() < 0.2 || (w - 24.0).abs() < 0.2, "radius {w}");
234        // A hostile path never panics and yields finite output.
235        let mut p = BezPath::new();
236        p.move_to((0.0, 0.0));
237        p.line_to((0.0, 0.0));
238        p.curve_to((f64::MAX, 0.0), (0.0, f64::MIN_POSITIVE), (1.0, 1.0));
239        p.close_path();
240        let _ = offset(&p, 3.0, 0.01);
241    }
