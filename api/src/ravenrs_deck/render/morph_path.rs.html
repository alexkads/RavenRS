2//! circle into a star), with the points aligned by `crate::geom::morph`.

4use crate::geom::Xfrm;
5use crate::geom::morph::{Fade, GeometryMorph};

12/// A frame shape (see [`crate::render::render_blend`]) whose outline morphs from `from` into its own
13/// geometry.
14#[derive(Clone, Debug, PartialEq)]
15pub struct PathMorph {
16    pub id: ShapeId,
17    /// The old shape's geometry.
18    pub from: Geom,
19    /// The boxes of the old and the new shape on their own slides.
20    pub from_box: Xfrm,
21    pub to_box: Xfrm,
22    /// Progress, 0..1 (already eased).
23    pub t: f64,
24}
25
26/// `now / own`, or 1 when that is no usable scale (a line's zero height).
27fn ratio(now: f64, own: f64) -> f64 {
28    let r = now / own;
29    if own.abs() > 1e-9 && r.is_finite() { r } else { 1.0 }
30}
31
32/// What a [`GeometryMorph`] is built from: both outlines at their own sizes. None of it changes
33/// while a transition plays, only `t` and the current box do.
34#[derive(Debug)]
35struct Source {
36    from: Geom,
37    from_wh: (f64, f64),
38    to: Geom,
39    to_wh: (f64, f64),
40}
41
42impl Source {
43    fn of(pm: &PathMorph, s: &Shape) -> Self {
44        Source { from: pm.from.clone(), from_wh: (pm.from_box.w, pm.from_box.h), to: s.geom.clone(), to_wh: (pm.to_box.w, pm.to_box.h) }
45    }
46
47    fn is(&self, pm: &PathMorph, s: &Shape) -> bool {
48        self.from == pm.from && self.from_wh == (pm.from_box.w, pm.from_box.h) && self.to == s.geom && self.to_wh == (pm.to_box.w, pm.to_box.h)
49    }
50
51    fn build(&self) -> GeometryMorph {
52        let geo = |geom: &Geom, (w, h): (f64, f64)| shape_geometry(&Shape { geom: geom.clone(), ..Shape::default() }, w, h);
53        GeometryMorph::new(&geo(&self.from, self.from_wh), &geo(&self.to, self.to_wh))
54    }
55}

57/// The aligned outlines of the morphs being played, kept across frames: aligning is the expensive
58/// part, blending them at `t` is cheap.
59#[derive(Debug, Default)]
60pub(crate) struct Cache {
61    /// With whether the current frame used it.
62    morphs: Vec<(Source, GeometryMorph, bool)>,
63}
64
65impl Cache {
66    /// The geometry of frame shape `s` at its current size `w` x `h`: both outlines are built at
67    /// their own sizes (so the alignment does not change while the box grows), then scaled into the
68    /// box. With a [`Fade`] per sub-path.
69    pub(crate) fn geometry(&mut self, pm: &PathMorph, s: &Shape, w: f64, h: f64) -> (Geometry, Vec<Fade>) {
70        let k = match self.morphs.iter().position(|(src, ..)| src.is(pm, s)) {
71            Some(k) => k,
72            None => {
73                let src = Source::of(pm, s);
74                let m = src.build();
75                self.morphs.push((src, m, false));
76                self.morphs.len() - 1
77            }
78        };
79        let scale = |x: Xfrm| Vec2::new(ratio(w, x.w), ratio(h, x.h));
80        let Some((_, m, used)) = self.morphs.get_mut(k) else {
81            // Not reachable (k was just found or pushed); align afresh rather than panic.
82            return Source::of(pm, s).build().at(pm.t, scale(pm.from_box), scale(pm.to_box));
83        };
84        *used = true;
85        m.at(pm.t, scale(pm.from_box), scale(pm.to_box))
86    }
87
88    /// Ends a frame: forgets the morphs it did not use (their transition is over).
89    pub(crate) fn sweep(&mut self) {
90        self.morphs.retain(|(.., used)| *used);
91        for (.., used) in &mut self.morphs {
92            *used = false;
93        }
94    }
95
96    #[cfg(test)]
97    pub(crate) fn len(&self) -> usize {
98        self.morphs.len()
99    }
100}
