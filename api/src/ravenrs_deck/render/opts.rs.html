3/// Animation state of one shape during a slide show or preview.
4#[derive(Clone, Debug)]
5pub struct ShapeState {
6    pub visible: bool,
7    pub opacity: f64,
8    /// Applied in slide space around the shape's centre: translation (pt), scale, rotation (deg).
9    pub offset: Vec2,
10    pub scale: (f64, f64),
11    pub rotate: f64,
12    /// Reveal clip in shape-local unit space (0..1): l, t, r, b.
13    pub clip: Option<[f64; 4]>,
14    /// Colour override for emphasis effects (fill).
15    pub tint: Option<Rgba>,
16    /// Per-paragraph states of by-paragraph text builds (paragraph index, state).
17    pub paras: Vec<(usize, ParaState)>,
18}

41pub struct RenderOpts<'a> {
42    /// Pixels per point.
43    pub scale: f64,
44    /// Editor view: placeholder prompts and dashed placeholder outlines.
45    pub edit: bool,
46    /// Draw shapes marked hidden (Selection pane) — the editor hides them like the show does.
47    pub show_hidden: bool,
48    /// Per-shape animation state.
49    pub state: Option<&'a dyn Fn(ShapeId) -> ShapeState>,
50    /// Don't draw these shapes (being edited in place, drawn by the UI).
51    pub skip: &'a [ShapeId],
52    /// Hide the text of this shape (the UI draws it while editing).
53    pub skip_text: Option<ShapeId>,
54    /// Text being edited: replace its prompt with gray markers on empty list paragraphs.
55    /// Ignored outside editor rendering.
56    pub editing_text: Option<ShapeId>,
57    /// Animated GIF pictures and how far into their play they are (seconds); others show frame 1.
58    pub gif_times: &'a [(ShapeId, f64)],
59    /// Worker threads (0 = this thread; required when effects use filters).
60    pub threads: u16,
61    /// Translate the slide within the output (pixels) and output size override.
62    pub offset: (f64, f64),
63    pub size: Option<(u32, u32)>,
64    /// Draw a background colour outside the slide (output larger than slide).
65    pub clear: Option<Rgba>,
66    /// Show this text in every date field instead of the date.
67    pub date_text: Option<String>,
68    /// The local date and time date fields show (`None`: the system clock).
69    pub now: Option<DateTime>,
70    /// Leave the slide background out (neither the white underlay nor the background fill), so
71    /// only what is on the slide is drawn over transparency: PNG export with a transparent
72    /// background. Shapes filled with the background (`useBgFill`) still show it.
73    pub transparent_background: bool,
74}

99pub(crate) struct SlideFields {
100    pub(crate) num: u32,
101    /// Text for every date field instead of the date ([`RenderOpts::date_text`]).
102    pub(crate) date: Option<String>,
103    pub(crate) footer: String,
104    /// The local date and time date fields show; `None`: they keep their saved text.
105    pub(crate) now: Option<DateTime>,
106}
107
108impl SlideFields {
109    pub(crate) fn new(num: u32, footer: String, opts: &RenderOpts) -> Self {
110        SlideFields { num, date: opts.date_text.clone(), footer, now: opts.now.or_else(DateTime::now_local) }
111    }
112}

114impl Fields for SlideFields {
115    fn field(&self, kind: &str) -> Option<String> {
116        self.field_in(kind, None)
117    }
118
119    fn field_in(&self, kind: &str, lang: Option<&str>) -> Option<String> {
120        if kind == "slidenum" {
121            return Some(self.num.to_string());
122        }
123        if datetime::is_date_field(kind) {
124            // An automatic date: today's, in the format of its type. (A fixed date is plain text
125            // in the slide's placeholder, not a field.)
126            return self.date.clone().or_else(|| datetime::field_text(kind, lang, self.now?));
127        }
128        if kind == "footer" {
129            return Some(self.footer.clone());
130        }
131        None
132    }
