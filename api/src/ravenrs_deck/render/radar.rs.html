1//! Radar charts: one spoke per category, the value axis running up the first spoke, a polygon
2//! of gridlines per major unit and each series a closed polygon (filled for a filled radar).

14/// The point at `frac` of the radius along spoke `i` of `n` (the first spoke points up).
15fn spoke(cen: Point, r: f64, i: usize, n: usize, frac: f64) -> Point {
16    let a = -std::f64::consts::FRAC_PI_2 + std::f64::consts::TAU * i as f64 / n.max(1) as f64;
17    Point::new(cen.x + r * frac * a.cos(), cen.y + r * frac * a.sin())
18}

20pub fn draw(ctx: &mut RenderContext, rctx: &Ctx, c: &Chart, m: Affine, area: Rect, base: f64, text_col: Rgba) {
21    let n = c.categories.len().max(c.series.iter().map(|s| s.values.len()).max().unwrap_or(0));
22    if n < 3 {
23        // Fewer than three spokes make no polygon; PowerPoint draws a line, we draw nothing.
24        return;
25    }
26    let grid_col = Rgba::rgb(0xD9, 0xD9, 0xD9);
27    // Room for the category labels around the outside.
28    let r = (area.width().min(area.height()) / 2.0 - base * 1.9).max(base);
29    let cen = area.center();
30    let (min, max, step) = c.axis_scale(false, 5).unwrap_or((0.0, 1.0, 0.2));
31    let span = crate::render::chart::positive_span(max - min);
32    let ticks = if step > 0.0 { (((max - min) / step).round() as i64).clamp(1, 20) } else { 1 };
33    ctx.set_transform(m);
34    ctx.set_stroke(kurbo::Stroke::new(0.75));
35    ctx.set_paint(color(grid_col, 1.0));
36    if c.gridlines {
37        for k in 1..=ticks {
38            let frac = k as f64 / ticks as f64;
39            let mut p = BezPath::new();
40            for i in 0..n {
41                let q = spoke(cen, r, i, n, frac);
42                if i == 0 { p.move_to(q) } else { p.line_to(q) }
43            }
44            p.close_path();
45            ctx.stroke_path(&p);
46        }
47    }
48    for i in 0..n {
49        let mut p = BezPath::new();
50        p.move_to(cen);
51        p.line_to(spoke(cen, r, i, n, 1.0));
52        ctx.set_transform(m);
53        ctx.set_paint(color(grid_col, 1.0));
54        ctx.stroke_path(&p);
55        if let Some(name) = c.categories.get(i) {
56            let q = spoke(cen, r + base * 1.3, i, n, 1.0);
57            let w = name.chars().count() as f64 * base * 0.55 + base;
58            let align = if (q.x - cen.x).abs() < base {
59                Align::Center
60            } else if q.x > cen.x {
61                Align::Left
62            } else {
63                Align::Right
64            };
65            let x0 = match align {
66                Align::Left => q.x - base * 0.3,
67                Align::Right => q.x - w + base * 0.3,
68                _ => q.x - w / 2.0,
69            };
70            label(
71                ctx,
72                rctx,
73                m,
74                name,
75                Rect::new(x0, q.y - base * 0.7, x0 + w, q.y + base * 0.7),
76                crate::render::chart::LabelStyle { size: base, align, anchor: Anchor::Middle, col: text_col, bold: false },
77            );
78        }
79    }
80    // Value labels up the first spoke.
81    if !c.y_hidden {
82        for k in 0..=ticks {
83            let v = min + step * k as f64;
84            let q = spoke(cen, r, 0, n, k as f64 / ticks as f64);
85            let t = c.axis_label(false, v);
86            let w = t.chars().count() as f64 * base * 0.55 + base * 0.4;
87            label(
88                ctx,
89                rctx,
90                m,
91                &t,
92                Rect::new(q.x - w - base * 0.3, q.y - base * 0.6, q.x - base * 0.3, q.y + base * 0.6),
93                crate::render::chart::LabelStyle { size: base * 0.9, align: Align::Right, anchor: Anchor::Middle, col: text_col, bold: false },
94            );
95        }
96    }
97    let filled = c.kind == ChartType::FilledRadar;
98    for (si, s) in c.series.iter().enumerate() {
99        let pts: Vec<Point> = (0..n)
100            .map(|i| {
101                let v = s.values.get(i).copied().flatten().unwrap_or(min);
102                spoke(cen, r, i, n, ((v - min) / span).clamp(0.0, 1.2))
103            })
104            .collect();
105        let mut p = BezPath::new();
106        for (i, q) in pts.iter().enumerate() {
107            if i == 0 { p.move_to(*q) } else { p.line_to(*q) }
108        }
109        p.close_path();
110        ctx.set_transform(m);
111        if filled {
112            ctx.set_paint(color(series_fill(rctx, c, si, Some(s)), 0.75));
113            ctx.fill_path(&p);
114        } else {
115            let col = series_line(rctx, c, si, s);
116            let lw = s.line.as_ref().and_then(|l| l.width).filter(|w| w.is_finite() && *w > 0.0).unwrap_or(2.25).clamp(0.25, 20.0);
117            ctx.set_paint(color(col, 1.0));
118            ctx.set_stroke(kurbo::Stroke::new(lw).with_join(kurbo::Join::Round));
119            ctx.stroke_path(&p);
120            if s.marker.is_some() {
121                for q in &pts {
122                    ctx.fill_path(&kurbo::Circle::new(*q, 3.5).to_path(0.1));
123                }
124            }
125        }
126        if c.series_data_labels(s) {
127            for (i, q) in pts.iter().enumerate() {
128                if let Some(v) = s.values.get(i).copied().flatten() {
129                    let t = c.data_label(s, v);
130                    label(
131                        ctx,
132                        rctx,
133                        m,
134                        &t,
135                        Rect::new(q.x - 40.0, q.y - base * 1.4, q.x + 40.0, q.y - base * 0.2),
136                        crate::render::chart::LabelStyle {
137                            size: base * 0.9,
138                            align: Align::Center,
139                            anchor: Anchor::Bottom,
140                            col: text_col,
141                            bold: false,
142                        },
143                    );
144                }
145            }
146        }
147    }
148}

150/// A stock chart's marks at one category: the high-low line, then the up/down bar between the
151/// first (open) and last (close) series, else a dash at the close.
152#[allow(clippy::too_many_arguments)]
153pub fn stock_marks(ctx: &mut RenderContext, rctx: &Ctx, c: &Chart, m: Affine, x: f64, band: f64, i: usize, to_y: &dyn Fn(f64) -> f64) {
154    let vals: Vec<f64> = c.series.iter().filter_map(|s| s.values.get(i).copied().flatten()).collect();
155    if vals.len() < 2 {
156        return;
157    }
158    let hi = vals.iter().copied().fold(f64::MIN, f64::max);
159    let lo = vals.iter().copied().fold(f64::MAX, f64::min);
160    let ink = Rgba::rgb(0x40, 0x40, 0x40);
161    ctx.set_transform(m);
162    ctx.set_paint(color(ink, 1.0));
163    ctx.set_stroke(kurbo::Stroke::new(1.0));
164    let mut p = BezPath::new();
165    p.move_to((x, to_y(hi)));
166    p.line_to((x, to_y(lo)));
167    ctx.stroke_path(&p);
168    let open = c.series.first().and_then(|s| s.values.get(i).copied().flatten());
169    let close = c.series.last().and_then(|s| s.values.get(i).copied().flatten());
170    if c.up_down_bars && c.series.len() >= 4 {
171        let (Some(o), Some(cl)) = (open, close) else { return };
172        let w = band * 0.4;
173        let (y0, y1) = (to_y(o.max(cl)), to_y(o.min(cl)));
174        let r = Rect::new(x - w / 2.0, y0, x + w / 2.0, y1.max(y0 + 1.0));
175        // PowerPoint's default: white when the price rose, dark when it fell.
176        ctx.set_paint(color(if cl >= o { Rgba::WHITE } else { ink }, 1.0));
177        ctx.fill_rect(&r);
178        ctx.set_paint(color(ink, 1.0));
179        ctx.stroke_path(&r.to_path(0.1));
180        return;
181    }
182    if let Some(cl) = close {
183        let col = c.series.last().map(|s| series_line(rctx, c, c.series.len() - 1, s)).unwrap_or(ink);
184        let y = to_y(cl);
185        let mut d = BezPath::new();
186        d.move_to((x - band * 0.12, y));
187        d.line_to((x + band * 0.12, y));
188        ctx.set_paint(color(col, 1.0));
189        ctx.set_stroke(kurbo::Stroke::new(2.5));
190        ctx.stroke_path(&d);
191    }
