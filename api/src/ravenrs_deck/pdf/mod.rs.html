1//! PDF export.
2//!
3//! Each slide is drawn by the DeckCraft renderer at print resolution (so the PDF looks exactly like
4//! the slide, effects included) and overlaid with an invisible layer of real text from the text
5//! layout engine with embedded font subsets, so the text can be searched, selected and copied.
6//! Hyperlinks and shape actions become link annotations, slide titles become bookmarks.
7//!
8//! Layouts: one slide per page, notes pages (slide above its speaker notes) and handouts (1, 2, 3,

37pub enum PdfError {

107/// The page size in points for a paper id from [`PAPERS`] (case-insensitive), turned on its side
108/// when `landscape`. `None` for an unknown id.
109pub fn paper_size(id: &str, landscape: bool) -> Option<(f64, f64)> {
110    let (_, _, (w, h)) = PAPERS.iter().find(|p| p.0.eq_ignore_ascii_case(id.trim()))?;
111    Some(if landscape { (*h, *w) } else { (*w, *h) })
112}

114/// The slide box on a notes page of `paper`: an inch of margin at the sides and the slide at most
115/// half the page tall, so landscape paper still leaves room for the notes.
116pub fn notes_slide_box(paper: (f64, f64), aspect: f64) -> Rect {
117    let (pw, ph) = paper;
118    let mut w = (pw - 2.0 * 72.0).max(1.0);
119    let mut h = w / aspect.max(0.01);
120    let max_h = (ph * 0.5 - 54.0).max(1.0);
121    if h > max_h {
122        h = max_h;
123        w = h * aspect;
124    }
125    let x = (pw - w) / 2.0;
126    Rect::new(x, 54.0, x + w, 54.0 + h)
127}

134fn plan_pages(pres: &Presentation, opts: &PdfOptions, slides: &[usize], sw: f64, sh: f64) -> Vec<Vec<(usize, Rect)>> {
135    match opts.layout {
136        PageLayout::Slides => slides.iter().map(|&i| vec![(i, Rect::new(0.0, 0.0, sw, sh))]).collect(),
137        PageLayout::Notes => {
138            let b = notes_frames(pres, opts.paper).0;
139            slides.iter().map(|&i| vec![(i, b)]).collect()
140        }
141        PageLayout::Handouts { per_page } => {
142            let boxes = handout_boxes(per_page, opts.paper, sw / sh);
143            slides.chunks(boxes.len()).map(|c| c.iter().zip(&boxes).map(|(i, b)| (*i, *b)).collect()).collect()
144        }
145    }
146}

148fn build_outline(pres: &Presentation, pages: &[Vec<(usize, Rect)>]) -> Outline {
149    let mut outline = Outline::new();
150    for (p, v) in pages.iter().enumerate() {
151        for (i, _) in v {
152            let t = pres.slides[*i].title();
153            let t = t.lines().next().unwrap_or("").trim().to_string();
154            let label = if t.is_empty() { format!("Slide {}", i + 1) } else { format!("{}. {t}", i + 1) };
155            outline.push_child(OutlineNode::new(label, XyzDestination::new(p, Point::from_xy(0.0, 0.0))));
156        }
157    }
158    outline
159}

161fn export_inner(pres: &Presentation, opts: &PdfOptions) -> Result<Vec<u8>> {
162    let slides: Vec<usize> = match &opts.slides {
163        Some(v) => v.iter().copied().filter(|i| *i < pres.slides.len()).collect(),
164        None => (0..pres.slides.len()).filter(|i| opts.include_hidden || !pres.slides[*i].hidden).collect(),
165    };
166    if slides.is_empty() {
167        return Err(PdfError::NoPages);
168    }
169    let title = opts.title.clone().unwrap_or_else(|| pres.props.title.clone());
170    let metadata = Metadata {
171        creator: "DeckCraft".into(),
172        producer: "DeckCraft".into(),
173        title: (!title.trim().is_empty()).then_some(title),
174        authors: if pres.props.author.trim().is_empty() { Vec::new() } else { vec![pres.props.author.clone()] },
175        ..Default::default()
176    };
177    let mut writer = Writer::new(Default::default(), metadata);
178    let pdf = writer.document_mut();
179
180    let (sw, sh) = (pres.slide_size.width.max(1.0), pres.slide_size.height.max(1.0));
181    let mut ex = Exporter { pres, opts, fonts: HashMap::new(), backdrops: Backdrops::default() };
182    // Page plan: per page, the slides on it with their boxes (page points).
183    let pages = plan_pages(pres, opts, &slides, sw, sh);
184    // Bookmarks and slide → page lookup for internal links.
185    let page_of: HashMap<usize, usize> = pages.iter().enumerate().flat_map(|(p, v)| v.iter().map(move |(i, _)| (*i, p))).collect();
186    pdf.set_outline(build_outline(pres, &pages));
187
188    for v in &pages {
189        let (pw, ph) = match opts.layout {
190            PageLayout::Slides => (sw, sh),
191            _ => opts.paper,
192        };
193        let size = Size::from_wh(pw as f32, ph as f32).ok_or(PdfError::NoPages)?;
194        let mut page = pdf.start_page_with(PageSettings::new(size));
195        let mut annots = Vec::new();
196        {
197            let mut s = page.surface();
198            for (i, b) in v {
199                ex.slide(&mut s, *i, *b)?;
200                annots.extend(ex.links(*i, *b, &page_of));
201            }
202            match opts.layout {
203                PageLayout::Notes => {
204                    if let Some((i, _)) = v.first() {
205                        let notes = pres.slides[*i].notes_text();
206                        ex.paragraphs(&mut s, &notes, notes_frames(pres, opts.paper).1, 12.0)?;
207                    }
208                    ex.page_texts(&mut s, pres.notes_master.as_deref(), pw, ph, v.first().map(|x| x.0 + 1).unwrap_or(1))?;
209                }
210                PageLayout::Handouts { per_page } => {
211                    if per_page == 3 {
212                        for (_, b) in v {
213                            ex.note_lines(&mut s, Rect::new(b.x1 + 24.0, b.y0, pw - 54.0, b.y1));
214                        }
215                    }
216                    ex.page_texts(&mut s, pres.handout_master.as_deref(), pw, ph, page_of.get(&v[0].0).map(|p| p + 1).unwrap_or(1))?;
217                }
218                PageLayout::Slides => {}
219            }
220            s.finish();
221        }
222        for a in annots {
223            page.add_annotation(a);
224        }
225        page.finish();
226    }
227    writer.finish().map_err(Into::into)
228}

230/// A notes-size master's rect of placeholder `kind`, scaled to a page of `paper` points.
231fn master_rect(pres: &Presentation, m: &crate::model::Master, kind: crate::model::PhType, paper: (f64, f64)) -> Option<Rect> {
232    let x = m.shapes.iter().find(|s| s.ph_type() == Some(kind))?.xfrm?;
233    let (kx, ky) = (paper.0 / pres.notes_size.width.max(1.0), paper.1 / pres.notes_size.height.max(1.0));
234    let r = Rect::new(x.x * kx, x.y * ky, (x.x + x.w) * kx, (x.y + x.h) * ky);
235    (r.width() > 1.0 && r.height() > 1.0).then_some(r)
236}

238/// A notes page's slide box and notes area on `paper`: where the notes master puts its slide image
239/// and body placeholders, or the default layout when it has none.
240pub fn notes_frames(pres: &Presentation, paper: (f64, f64)) -> (Rect, Rect) {
241    let (pw, ph) = paper;
242    let aspect = pres.slide_size.width.max(1.0) / pres.slide_size.height.max(1.0);
243    let nm = pres.notes_master.as_deref();
244    let slide = match nm.and_then(|m| master_rect(pres, m, crate::model::PhType::SlideImage, paper)) {
245        // The slide keeps its own shape, centred in the frame.
246        Some(f) => {
247            let (w, h) = if f.width() / f.height() > aspect { (f.height() * aspect, f.height()) } else { (f.width(), f.width() / aspect) };
248            Rect::from_center_size(f.center(), (w, h))
249        }
250        None => notes_slide_box(paper, aspect),
251    };
252    let notes = nm
253        .and_then(|m| master_rect(pres, m, crate::model::PhType::Body, paper))
254        .unwrap_or_else(|| Rect::new(72.0, slide.y1 + 36.0, pw - 72.0, ph - 54.0));
255    (slide, notes)
256}

269/// The resolved style of placeholder `sh`'s first run (or its paragraph's end properties) on
270/// master `m`: run, list style, master text styles, theme fonts and colours.
271pub fn master_text_style(pres: &Presentation, m: &crate::model::Master, sh: &crate::model::Shape) -> TextStyle {
272    use crate::model::resolve::{self, Ctx};
273    let ctx = Ctx::for_master(pres, m);
274    let para = sh.text.as_ref().and_then(|t| t.paragraphs.first()).cloned().unwrap_or_default();
275    let own = para.runs.first().map(|r| r.props.clone()).unwrap_or_else(|| para.end_props.clone());
276    let r = resolve::run(&ctx, sh, &para, &own);
277    let family = r.font.as_deref().map(|f| if f.starts_with('+') { m.theme.font(f) } else { f.to_string() }).unwrap_or_default();
278    let color = match &r.fill {
279        Some(crate::model::Fill::Solid { color }) => ctx.color(color, None),
280        _ => crate::model::Rgba::rgb(0x60, 0x60, 0x60),
281    };
282    TextStyle { family, size: r.size.unwrap_or(12.0).clamp(1.0, 400.0), bold: r.bold == Some(true), italic: r.italic == Some(true), color }
283}

297/// The header, date, footer and page number of `m` for page `n`, laid out by the text engine
298/// (wrapped, mixed formatting) and scaled to `paper`, for drawing without glyph access (the
299/// Notes Page view).
300pub fn master_text_pieces(pres: &Presentation, m: &crate::model::Master, paper: (f64, f64), n: usize) -> Vec<TextPiece> {
301    let (kx, ky) = (paper.0 / pres.notes_size.width.max(1.0), paper.1 / pres.notes_size.height.max(1.0));
302    let mut out = vec![];
303    for pt in crate::render::place_master_texts(pres, m, n) {
304        let mut para_chars: HashMap<usize, Vec<char>> = HashMap::new();
305        for run in &pt.layout.runs {
306            let Some(g) = run.glyphs.first() else { continue };
307            let chars = para_chars.entry(run.para).or_insert_with(|| {
308                pt.body.paragraphs.get(run.para).map(|p| p.runs.iter().flat_map(|r| r.content().chars()).collect()).unwrap_or_default()
309            });
310            let (a, b) = (run.chars.0.min(chars.len()), run.chars.1.min(chars.len()));
311            let text: String = chars.get(a..b.max(a)).unwrap_or_default().iter().collect();
312            if text.trim().is_empty() {
313                continue;
314            }
315            let p = pt.transform * kurbo::Point::new(g.1, g.2);
316            out.push(TextPiece {
317                at: (p.x * kx, p.y * ky),
318                text,
319                size: run.size * ky,
320                color: run.color,
321                bold: run.fake_bold || run.face.weight >= 600.0,
322                italic: run.fake_italic || run.face.italic,
323            });
324        }
325    }
326    out
327}

329/// The header, date, footer and page-number text a master puts on page `n`, with where it goes,
330/// whether it's right-aligned and its style. Placeholders with no text are left out; the page
331/// number is always `n`.
332pub fn master_texts(pres: &Presentation, m: &crate::model::Master, paper: (f64, f64), n: usize) -> Vec<(Rect, String, bool, TextStyle)> {
333    use crate::model::PhType;
334    let mut out = vec![];
335    for kind in [PhType::Header, PhType::Date, PhType::Footer, PhType::SlideNum] {
336        let Some(r) = master_rect(pres, m, kind, paper) else { continue };
337        let Some(sh) = m.shapes.iter().find(|s| s.ph_type() == Some(kind)) else { continue };
338        let text = if kind == PhType::SlideNum { n.to_string() } else { sh.text.as_ref().map(|t| t.text()).unwrap_or_default().trim().to_string() };
339        if text.is_empty() {
340            continue;
341        }
342        let right = sh
343            .text
344            .as_ref()
345            .and_then(|t| t.paragraphs.first())
346            .and_then(|p| p.props.align)
347            .map_or(matches!(kind, PhType::Date | PhType::SlideNum), |a| a == crate::model::text::Align::Right);
348        let mut style = master_text_style(pres, m, sh);
349        // Sizes scale with the page, as the placeholders do.
350        style.size *= paper.1 / pres.notes_size.height.max(1.0);
351        out.push((r, text, right, style));
352    }
353    out
354}

356/// Slide boxes for a handout page (portrait paper, half-inch-ish margins).
357pub fn handout_boxes(per_page: u8, paper: (f64, f64), aspect: f64) -> Vec<Rect> {
358    let (pw, ph) = paper;
359    let (mx, top, bottom) = (54.0, 54.0, 54.0);
360    let (cols, rows, notes) = match per_page {
361        0 | 1 => (1, 1, false),
362        2 => (1, 2, false),
363        3 => (1, 3, true),
364        4 => (2, 2, false),
365        6 => (2, 3, false),
366        _ => (3, 3, false),
367    };
368    let gap = 18.0;
369    let aw = if notes { (pw - 2.0 * mx) * 0.5 } else { pw - 2.0 * mx };
370    let ah = ph - top - bottom - 18.0;
371    let cw = (aw - gap * (cols as f64 - 1.0)) / cols as f64;
372    let ch = (ah - gap * (rows as f64 - 1.0)) / rows as f64;
373    let (w, h) = if cw / ch > aspect { (ch * aspect, ch) } else { (cw, cw / aspect) };
374    let mut out = Vec::new();
375    for r in 0..rows {
376        for c in 0..cols {
377            let cx = mx + c as f64 * (cw + gap) + (cw - w) / 2.0;
378            let cy = top + r as f64 * (ch + gap) + (ch - h) / 2.0;
379            let cx = if notes { mx } else { cx };
380            out.push(Rect::new(cx, cy, cx + w, cy + h));
381        }
382    }
383    out
384}

406/// Premultiplied RGBA (the renderer's pixels) to the straight RGBA the writer stores. Runs of
407/// pixels that are all fully transparent or fully opaque, most of a slide, are skipped in blocks.
408fn straight(rgba: &mut [u8]) {
409    let (blocks, rest) = rgba.as_chunks_mut::<64>();
410    for block in blocks {
411        if block.as_chunks::<4>().0.iter().fold(false, |m, px| m | (px[3].wrapping_sub(1) < 254)) {
412            straight_px(block);
413        }
414    }
415    straight_px(rest);
416}
417
418fn straight_px(rgba: &mut [u8]) {
419    for px in rgba.as_chunks_mut::<4>().0 {
420        let a = px[3] as u32;
421        if a != 0 && a != 255 {
422            for c in &mut px[..3] {
423                *c = (((*c as u32 * 255 + a / 2) / a).min(255)) as u8;
424            }
425        }
426    }
427}
428
429/// A backdrop raster composited onto white (slides are normally opaque, and opaque blocks are skipped).
430fn opaque(rgba: &mut [u8]) {
431    let (blocks, rest) = rgba.as_chunks_mut::<64>();
432    for block in blocks {
433        if !block.as_chunks::<4>().0.iter().fold(true, |m, px| m & (px[3] == 255)) {
434            opaque_px(block);
435        }
436    }
437    opaque_px(rest);
438}
439
440fn opaque_px(rgba: &mut [u8]) {
441    for px in rgba.as_chunks_mut::<4>().0 {
442        let k = 255 - px[3];
443        px[0] = px[0].saturating_add(k);
444        px[1] = px[1].saturating_add(k);
445        px[2] = px[2].saturating_add(k);
446        px[3] = 255;
447    }
448}

450/// Gives every fully transparent content pixel the backdrop's colour. The renderer leaves them
451/// RGB 0,0,0; a viewer that smooths the image blends colour and mask separately, so the black
452/// shows as a grey hairline around white pictures and shapes (#1455).
453fn fill_clear(front: &mut [u8], back: &[u8]) {
454    for (f, b) in front.as_chunks_mut::<4>().0.iter_mut().zip(back.as_chunks::<4>().0) {
455        if f[3] == 0 {
456            f[..3].copy_from_slice(&b[..3]);
457        }
458    }
459}

461/// Most bytes of opaque backdrop rasters kept to recognise a backdrop seen before.
462const BACKDROP_CACHE_BYTES: usize = 32 << 20;
463
464/// Backdrops already handed to the writer, most recent first, with their pixels. Slides that
465/// share a layout render the same backdrop; reusing its image skips hashing and compressing the
466/// raster again, and the writer stores it once either way (#845).
467#[derive(Default)]
468struct Backdrops(Vec<(Vec<u8>, u32, u32, Image)>);
469
470impl Backdrops {
471    fn image(&mut self, mut img: crate::render::Image) -> Image {
472        opaque(&mut img.pixels);
473        let (w, h) = (img.width, img.height);
474        if let Some(i) = self.0.iter().position(|(px, pw, ph, _)| (*pw, *ph) == (w, h) && *px == img.pixels) {
475            let hit = self.0.remove(i);
476            let image = hit.3.clone();
477            self.0.insert(0, hit);
478            return image;
479        }
480        if img.pixels.len() > BACKDROP_CACHE_BYTES {
481            return Image::from_rgba8(img.pixels, w, h);
482        }
483        let image = Image::from_rgba8(img.pixels.clone(), w, h);
484        self.0.insert(0, (img.pixels, w, h, image.clone()));
485        let mut total = 0;
486        self.0.retain(|(px, ..)| {
487            total += px.len();
488            total <= BACKDROP_CACHE_BYTES
489        });
490        image
491    }
492}

500/// Render scale for a slide of `slide` units drawn `box_width` points wide at `dpi`, with the total pixel count capped.
501fn raster_scale(box_width: f64, slide: (f64, f64), dpi: f64) -> f64 {
502    let scale = (box_width / 72.0 * dpi.clamp(36.0, 600.0)) / slide.0;
503    let pixels = slide.0 * scale * slide.1 * scale;
504    if pixels > MAX_RASTER_PIXELS { scale * (MAX_RASTER_PIXELS / pixels).sqrt() } else { scale }
505}

507impl Exporter<'_> {
508    fn font(&mut self, face: &crate::fonts::FontFace) -> Result<Font> {
509        if let Some(font) = self.fonts.get(&face.id()) {
510            return Ok(font.clone());
511        }
512        let font = write::font(face.data(), face.index(), None)?;
513        self.fonts.insert(face.id(), font.clone());
514        Ok(font)
515    }

517    /// Slide `i` drawn into `b` (page points).
518    fn slide(&mut self, s: &mut Surface, i: usize, b: Rect) -> Result<()> {
519        let pres = self.pres;
520        let (sw, sh) = (pres.slide_size.width.max(1.0), pres.slide_size.height.max(1.0));
521        let k = b.width() / sw;
522        // Raster at the requested dpi relative to the box size on paper.
523        let scale = raster_scale(b.width(), (sw, sh), self.opts.dpi);
524        // One clock reading for the raster and the text layer, so a date field cannot differ between them.
525        let ropts = RenderOpts { scale, threads: 4, now: crate::text::datetime::DateTime::now_local(), ..Default::default() };
526        // Two rasters per slide: the backdrop (background, master and layout graphics) is the same
527        // on every slide that shares its layout, so the writer stores it once, and the slide's own
528        // shapes draw over it (#845).
529        let mut back = crate::render::render_slide_part(pres, i, crate::render::SlidePart::Backdrop, &ropts);
530        opaque(&mut back.pixels);
531        let front = crate::render::render_slide_part(pres, i, crate::render::SlidePart::Content, &ropts);
532        let (w, h) = (back.width, back.height);
533        if w > 0
534            && h > 0
535            && let Some(size) = Size::from_wh(b.width() as f32, b.height() as f32)
536        {
537            s.push_transform(&Transform::from_translate(b.x0 as f32, b.y0 as f32));
538            // Taken, not copied: a print-resolution slide is ~16 MB.
539            let mut rgba = front.pixels;
540            let content = rgba.as_chunks::<4>().0.iter().any(|px| px[3] != 0);
541            if content {
542                fill_clear(&mut rgba, &back.pixels);
543                straight(&mut rgba);
544            }
545            s.draw_image(self.backdrops.image(back), size);
546            if content {
547                s.draw_image(Image::from_rgba8(rgba, w, h), size);
548            }
549            s.pop();
550        }
551        if self.opts.frame_slides && self.opts.layout != PageLayout::Slides {
552            let p = kurbo::Shape::to_path(&b, 0.1);
553            if let Some(path) = to_path(&p) {
554                s.set_fill(None);
555                s.set_stroke(Some(krilla::paint::Stroke {
556                    paint: krilla::color::rgb::Color::new(0x80, 0x80, 0x80).into(),
557                    width: 0.5,
558                    ..Default::default()
559                }));
560                s.draw_path(&path);
561                s.set_stroke(None);
562            }
563        }
564        if self.opts.text_layer {
565            let view = Affine::translate((b.x0, b.y0)) * Affine::scale(k);
566            for pt in crate::render::place_slide_with(pres, i, &ropts).texts {
567                self.text_layer(s, &pt, view, false)?;
568            }
569            let _ = sh;
570        }
571        Ok(())
572    }
573
574    /// Invisible glyphs exactly over the rasterised text.
575    fn text_layer(&mut self, s: &mut Surface, pt: &PlacedText, view: Affine, visible: bool) -> Result<()> {
576        // Each paragraph's characters, gathered once: a long paragraph wraps into many runs.
577        let mut para_chars: HashMap<usize, Vec<char>> = HashMap::new();
578        for run in &pt.layout.runs {
579            if run.glyphs.is_empty() || run.size <= 0.0 {
580                continue;
581            }
582            let chars = para_chars.entry(run.para).or_insert_with(|| {
583                pt.body.paragraphs.get(run.para).map(|p| p.runs.iter().flat_map(|r| r.content().chars()).collect()).unwrap_or_default()
584            });
585            let (a, b) = (run.chars.0.min(chars.len()), run.chars.1.min(chars.len()));
586            let text: String = chars.get(a..b.max(a)).unwrap_or_default().iter().collect();
587            if text.trim().is_empty() {
588                continue;
589            }
590            let font = self.font(&run.face)?;
591            let n = run.glyphs.len();
592            let ranges = glyph_text_ranges(&text, run.chars, &run.cells, n);
593            let (x0, y0) = (run.glyphs[0].1, run.glyphs[0].2);
594            let kg: Vec<KrillaGlyph> = run
595                .glyphs
596                .iter()
597                .enumerate()
598                .map(|(gi, g)| {
599                    let next = run.glyphs.get(gi + 1).map_or(g.1 + run.face.advance(g.0) * run.size / run.face.upem.max(1.0), |nx| nx.1);
600                    let range = ranges.get(gi).cloned().unwrap_or(text.len()..text.len());
601                    KrillaGlyph::new(GlyphId::new(g.0), ((next - g.1) / run.size) as f32, 0.0, ((g.2 - y0) / run.size) as f32, 0.0, range, None)
602                })
603                .collect();
604            s.push_transform(&tf(view * pt.transform * Affine::translate((x0, y0))));
605            s.set_fill(Some(if visible {
606                let c = run.color;
607                Fill {
608                    paint: krilla::color::rgb::Color::new(c.r, c.g, c.b).into(),
609                    opacity: NormalizedF32::new((f64::from(c.a) / 255.0 * run.alpha) as f32).unwrap_or(NormalizedF32::ONE),
610                    ..Default::default()
611                }
612            } else {
613                grey(0, 0.0)
614            }));
615            s.draw_glyphs(Point::from_xy(0.0, 0.0), &kg, font, &text, run.size as f32, false);
616            s.set_fill(None);
617            s.pop();
618        }
619        Ok(())
620    }

622    fn links(&self, i: usize, b: Rect, page_of: &HashMap<usize, usize>) -> Vec<Annotation> {
623        let pres = self.pres;
624        let k = b.width() / pres.slide_size.width.max(1.0);
625        let view = Affine::translate((b.x0, b.y0)) * Affine::scale(k);
626        let mut out = Vec::new();
627        for l in crate::render::place_slide(pres, i).links {
628            let r = view.transform_rect_bbox(l.rect).intersect(b);
629            let Some(rect) = krilla::geom::Rect::from_ltrb(r.x0 as f32, r.y0 as f32, r.x1 as f32, r.y1 as f32) else { continue };
630            let goto = |slide: Option<usize>| {
631                slide.and_then(|s| page_of.get(&s)).map(|p| Target::Destination(XyzDestination::new(*p, Point::from_xy(0.0, 0.0)).into()))
632            };
633            let target = match &l.link.action {
634                // Only the links the slide show opens (web and mail): a reader of the file gets the rest
635                // as plain text.
636                ClickAction::Url { url } if safe_url(url) => Some(Target::Action(Action::Link(LinkAction::new(url.clone())))),
637                ClickAction::Slide { slide } => goto(pres.slides.iter().position(|x| x.id == *slide)),
638                ClickAction::NextSlide => goto(Some(i + 1)),
639                ClickAction::PreviousSlide => goto(i.checked_sub(1)),
640                ClickAction::FirstSlide => goto(Some(0)),
641                ClickAction::LastSlide => goto(pres.slides.len().checked_sub(1)),
642                _ => None,
643            };
644            if let Some(t) = target {
645                let alt = if l.link.tooltip.is_empty() { None } else { Some(l.link.tooltip.clone()) };
646                out.push(Annotation::new_link(LinkAnnotation::new(rect, t), alt));
647            }
648        }
649        out
650    }

658    /// Wrapped plain text (notes).
659    fn paragraphs(&mut self, s: &mut Surface, text: &str, area: Rect, size: f64) -> Result<()> {
660        let (font, face) = self.ui_font()?;
661        let k = size / face.upem.max(1.0);
662        let width = |w: &str| w.chars().map(|c| face.advance(face.glyph_for(c)) * k).sum::<f64>();
663        let lh = size * 1.3;
664        let mut y = area.y0 + size;
665        s.set_fill(Some(grey(0x20, 1.0)));
666        let space = width(" ");
667        'outer: for para in text.split('\n') {
668            // The line's width is kept as words are added, not measured again for each word.
669            let mut line = String::new();
670            let mut line_w = 0.0;
671            for word in para.split(' ') {
672                let word_w = width(word);
673                let cand_w = if line.is_empty() { word_w } else { line_w + space + word_w };
674                if cand_w > area.width() && !line.is_empty() {
675                    if y > area.y1 {
676                        break 'outer;
677                    }
678                    s.draw_text(Point::from_xy(area.x0 as f32, y as f32), font.clone(), size as f32, &line, false, TextDirection::Auto);
679                    y += lh;
680                    line = word.to_string();
681                    line_w = word_w;
682                } else {
683                    if !line.is_empty() {
684                        line.push(' ');
685                    }
686                    line.push_str(word);
687                    line_w = cand_w;
688                }
689            }
690            if y > area.y1 {
691                break;
692            }
693            if !line.is_empty() {
694                s.draw_text(Point::from_xy(area.x0 as f32, y as f32), font.clone(), size as f32, &line, false, TextDirection::Auto);
695            }
696            y += lh;
697        }
698        s.set_fill(None);
699        Ok(())
700    }

702    /// Ruled lines for handwritten notes (3-per-page handouts).
703    fn note_lines(&self, s: &mut Surface, r: Rect) {
704        // Counted, not stepped: at huge coordinates adding 22 leaves y unchanged, and a huge page
705        // would want millions of rules.
706        let first = r.y0 + 18.0;
707        let count = ((r.y1 - first) / 22.0).floor() + 1.0;
708        if count.is_nan() || count < 1.0 {
709            return;
710        }
711        s.set_stroke(Some(krilla::paint::Stroke {
712            paint: krilla::color::rgb::Color::new(0xB0, 0xB0, 0xB0).into(),
713            width: 0.5,
714            ..Default::default()
715        }));
716        for i in 0..(count.min(MAX_NOTE_LINES as f64) as usize) {
717            let y = first + 22.0 * i as f64;
718            let mut p = kurbo::BezPath::new();
719            p.move_to((r.x0, y));
720            p.line_to((r.x1, y));
721            if let Some(path) = to_path(&p) {
722                s.draw_path(&path);
723            }
724        }
725        s.set_stroke(None);
726    }

728    /// A notes or handout page's header, date, footer and page number from its master; just the
729    /// page number when the deck has no such master.
730    fn page_texts(&mut self, s: &mut Surface, master: Option<&crate::model::Master>, pw: f64, ph: f64, n: usize) -> Result<()> {
731        let Some(m) = master else { return self.footer(s, pw, ph, n) };
732        // Laid out by the text engine in the placeholders, scaled from the notes size to the page:
733        // wrapping, alignment and mixed formatting as the master has them.
734        let pres = self.pres;
735        let view = Affine::scale_non_uniform(pw / pres.notes_size.width.max(1.0), ph / pres.notes_size.height.max(1.0));
736        for pt in crate::render::place_master_texts(pres, m, n) {
737            self.text_layer(s, &pt, view, true)?;
738        }
739        Ok(())
740    }
741    fn footer(&mut self, s: &mut Surface, pw: f64, ph: f64, n: usize) -> Result<()> {
742        let (font, _) = self.ui_font()?;
743        s.set_fill(Some(grey(0x60, 1.0)));
744        s.draw_text(Point::from_xy((pw - 72.0) as f32, (ph - 30.0) as f32), font, 9.0, &n.to_string(), false, TextDirection::Auto);
745        s.set_fill(None);
746        Ok(())
747    }

750/// Byte range in `text` (the run's characters `chars`) of each of `n` glyphs, for the ToUnicode map.
751/// `cells` holds each glyph's shaping-cluster head (paragraph character index); a cluster runs to the
752/// next head in the run, so a ligature maps to all its characters and every glyph of a cluster shares
753/// its range (krilla then writes the text once).
754fn glyph_text_ranges(text: &str, chars: (usize, usize), cells: &[(usize, char)], n: usize) -> Vec<std::ops::Range<usize>> {
755    let starts: Vec<usize> = text.char_indices().map(|(b, _)| b).collect();
756    let byte = |ci: usize| ci.checked_sub(chars.0).and_then(|i| starts.get(i).copied()).unwrap_or(text.len());
757    let mut heads: Vec<usize> = cells.iter().map(|c| c.0).collect();
758    heads.sort_unstable();
759    heads.dedup();
760    let end = |ci: usize| heads.get(heads.partition_point(|&h| h <= ci)).copied().unwrap_or(chars.1);
761    (0..n).map(|gi| cells.get(gi).map_or(text.len()..text.len(), |&(ci, _)| byte(ci)..byte(end(ci)))).collect()
762}

764fn to_path(bp: &kurbo::BezPath) -> Option<krilla::geom::Path> {
765    use kurbo::PathEl;
766    let mut pb = krilla::geom::PathBuilder::new();
767    for el in bp.elements() {
768        match *el {
769            PathEl::MoveTo(p) => pb.move_to(p.x as f32, p.y as f32),
770            PathEl::LineTo(p) => pb.line_to(p.x as f32, p.y as f32),
771            PathEl::QuadTo(a, p) => pb.quad_to(a.x as f32, a.y as f32, p.x as f32, p.y as f32),
772            PathEl::CurveTo(a, b, p) => pb.cubic_to(a.x as f32, a.y as f32, b.x as f32, b.y as f32, p.x as f32, p.y as f32),
773            PathEl::ClosePath => pb.close(),
774        }
775    }
776    pb.finish()
777}
