1//! Keep opaque chart payloads out of the JSON manifest and within the media load budget.
2use std::collections::HashSet;
3use std::io::Cursor;
4use std::sync::Arc;

11const PREFIX: &str = "chart-parts/";
12type Payload = (String, String, Vec<u8>);
13
14pub(super) fn separate(p: &mut Presentation) -> Result<Vec<Payload>> {
15    let mut payloads = Vec::new();
16    visit_parts(p, |part| {
17        // The native namespace is independent of the original OPC names, which may repeat
18        // when charts from several decks are combined. No user-supplied path is extracted.
19        let entry = format!("{PREFIX}{}", payloads.len());
20        part.native_entry = Some(entry.clone());
21        payloads.push((entry, part.content_type.clone(), std::mem::take(&mut part.bytes)));
22        Ok(())
23    })?;
24    Ok(payloads)
25}

27pub(super) fn restore(
28    p: &mut Presentation,
29    z: &mut zip::ZipArchive<Cursor<&[u8]>>,
30    left: &mut u64,
31    mut items: usize,
32    max_items: usize,
33    damaged: &mut Vec<String>,
34) -> Result<()> {
35    let mut seen = HashSet::new();
36    visit_parts(p, |part| {
37        if items >= max_items {
38            return Err(FormatError::Damaged(format!("media and chart parts exceed {max_items} items")));
39        }
40        items += 1;
41        if let Some(entry) = part.native_entry.take() {
42            // Accept only names produced by separate(), and inflate each member once.
43            let canonical = entry.strip_prefix(PREFIX).and_then(|s| s.parse::<usize>().ok()).is_some_and(|n| entry == format!("{PREFIX}{n}"));
44            if !canonical || !seen.insert(entry.clone()) || !part.bytes.is_empty() {
45                return Err(FormatError::Damaged("invalid or ambiguous chart part reference".into()));
46            }
47            if z.index_for_name(&entry).is_none() {
48                // Missing members consume no inflation work and can be reported for recovery.
49                damaged.push(format!("{entry}: missing chart part"));
50                return Ok(());
51            }
52            if z.by_name(&entry).is_ok_and(|f| f.size() > *left) {
53                return Err(FormatError::Damaged("embedded media and chart parts are larger than DeckCraft can open".into()));
54            }
55            // A CRC/decode error may already have inflated bytes. Stop on that first error
56            // rather than letting later corrupt members reuse the unchanged byte budget.
57            let bytes = read_entry_max(z, &entry, (*left).min(MAX_ENTRY))?;
58            *left -= bytes.len() as u64;
59            part.bytes = bytes;
60        } else {
61            // Older manifests used JSON byte arrays. Charge those too, after the existing
62            // bounded manifest parser has read them, so resaving migrates them losslessly.
63            let size = part.bytes.len() as u64;
64            if size > (*left).min(MAX_ENTRY) {
65                return Err(FormatError::Damaged("inline chart part is larger than DeckCraft can open".into()));
66            }
67            *left -= size;
68        }
69        Ok(())
70    })
71}

73fn visit_parts(p: &mut Presentation, mut visit: impl FnMut(&mut ChartPart) -> Result<()>) -> Result<()> {
74    // Use an explicit stack for groups. Charts can also live on masters and their layouts,
75    // including the notes and handout masters; none of these owners may leak inline bytes.
76    for slide in &mut p.slides {
77        shapes(&mut Arc::make_mut(slide).shapes, &mut visit)?;
78    }
79    for master in &mut p.masters {
80        master_parts(Arc::make_mut(master), &mut visit)?;
81    }
82    for master in [&mut p.notes_master, &mut p.handout_master].into_iter().flatten() {
83        master_parts(Arc::make_mut(master), &mut visit)?;
84    }
85    Ok(())
86}
87
88fn master_parts(master: &mut Master, visit: &mut impl FnMut(&mut ChartPart) -> Result<()>) -> Result<()> {
89    shapes(&mut master.shapes, visit)?;
90    for layout in &mut master.layouts {
91        shapes(&mut layout.shapes, visit)?;
92    }
93    Ok(())
94}
95
96fn shapes(root: &mut [Shape], visit: &mut impl FnMut(&mut ChartPart) -> Result<()>) -> Result<()> {
97    let mut pending = vec![root];
98    while let Some(shapes) = pending.pop() {
99        for shape in shapes {
100            match &mut shape.kind {
101                ShapeKind::Group { children, .. } => pending.push(children),
102                ShapeKind::Chart(chart) => {
103                    for part in &mut chart.kept.parts {
104                        visit(part)?;
105                    }
106                }
107                _ => {}
108            }
109        }
110    }
111    Ok(())
112}
