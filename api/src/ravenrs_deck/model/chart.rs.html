271/// Count, mean and sample standard deviation of a series' finite values (for error bars).
272#[derive(Clone, Copy, Debug, Default, PartialEq)]
273pub struct SeriesStats {
274    pub n: f64,
275    pub mean: f64,
276    pub sd: f64,
277}
278
279impl SeriesStats {
280    pub fn of(values: &[Option<f64>]) -> Self {
281        let vals = || values.iter().flatten().copied().filter(|v| v.is_finite());
282        let n = vals().count() as f64;
283        let mean = vals().sum::<f64>() / n.max(1.0);
284        let sd = if n > 1.0 { (vals().map(|x| (x - mean).powi(2)).sum::<f64>() / (n - 1.0)).sqrt() } else { 0.0 };
285        SeriesStats { n, mean, sd }
286    }
287}

289impl ErrorBars {
290    /// Where point `i` (value `v`) of a series with `values` has its bar: (low, high) in data units.
291    /// Standard deviation bars stand around the series' mean, as Excel draws them; the others
292    /// around the point. `None` for an amount that isn't a finite number.
293    pub fn range(&self, values: &[Option<f64>], i: usize, v: f64) -> Option<(f64, f64)> {
294        self.range_in(&SeriesStats::of(values), i, v)
295    }
296
297    /// [`ErrorBars::range`] with the series' statistics worked out once ([`SeriesStats::of`]), so
298    /// drawing every point of a long series stays linear.
299    pub fn range_in(&self, stats: &SeriesStats, i: usize, v: f64) -> Option<(f64, f64)> {
300        let SeriesStats { n, mean, sd } = *stats;
301        let (centre, minus, plus) = match self.val_type.as_str() {
302            "percentage" => (v, (v * self.value / 100.0).abs(), (v * self.value / 100.0).abs()),
303            "stdDev" => (mean, sd * self.value, sd * self.value),
304            "stdErr" => (v, sd / n.max(1.0).sqrt(), sd / n.max(1.0).sqrt()),
305            "cust" => (v, self.minus.get(i).copied().flatten().unwrap_or(0.0), self.plus.get(i).copied().flatten().unwrap_or(0.0)),
306            _ => (v, self.value, self.value),
307        };
308        let (lo, hi) = match self.bar_type.as_str() {
309            "plus" => (centre, centre + plus),
310            "minus" => (centre - minus, centre),
311            _ => (centre - minus, centre + plus),
312        };
313        (lo.is_finite() && hi.is_finite()).then_some((lo, hi))
314    }
315}

620        let (lo, hi) = (lo.min(0.0), hi.max(0.0));

624    /// The smallest and largest value plotted against one value axis: plain values as they are,
625    /// stacked series as running sums from zero, 100 % series as 0–1. `(0, 0)` without finite
626    /// values; `None` when no series uses the axis.
627    fn plotted_extent(&self, secondary: bool) -> Option<(f64, f64)> {
628        if self.kind == ChartType::Waterfall {
629            return (!secondary).then(|| self.value_range());
630        }
631        let ser: Vec<&Series> = self.series.iter().filter(|s| self.on_secondary(s) == secondary).collect();
632        if ser.is_empty() {
633            return (!secondary && self.kind != ChartType::Combo).then_some((0.0, 0.0));
634        }
635        let mut ext = (f64::INFINITY, f64::NEG_INFINITY);
636        let has_stacked_or_percent = ser.iter().any(|s| {
637            let k = self.series_kind(s);
638            k.stacked() || k.percent()
639        });
640        let percent = if has_stacked_or_percent {
641            self.fold_stacked(&ser, &mut ext)
642        } else {
643            for s in &ser {
644                for v in s.values.iter().flatten().copied().filter(|v| v.is_finite()) {
645                    widen(&mut ext, v);
646                }
647            }
648            false
649        };
650        if !percent {
651            self.fold_trendlines(&ser, &mut ext);
652        }
653        if percent {
654            ext.0 = ext.0.min(0.0);
655            ext.1 = ext.1.max(1.0);
656        }
657        let (lo, hi) = ext;
658        Some(if lo <= hi { (lo, hi) } else { (0.0, 0.0) })
659    }

661    /// Widens `ext` by the points of the stacked and 100 % series in `ser`; returns whether a
662    /// 100 % series is among them.
663    fn fold_stacked(&self, ser: &[&Series], ext: &mut (f64, f64)) -> bool {
664        let mut percent = false;
665        let n = ser.iter().map(|s| s.values.len()).max().unwrap_or(0);
666        let budget = (ser.len() * n).min(10_000_000);
667        let n_clamped = if ser.is_empty() { 0 } else { n.min(budget / ser.len()) };
668        for i in 0..n_clamped {
669            let (mut p, mut m, mut stacked) = (0.0, 0.0, false);
670            for s in ser {
671                let k = self.series_kind(s);
672                let Some(v) = s.values.get(i).copied().flatten().filter(|v| v.is_finite()) else { continue };
673                if k.percent() {
674                    percent = true;
675                } else if k.stacked() {
676                    stacked = true;
677                    if v >= 0.0 { p += v } else { m += v }
678                } else {
679                    widen(ext, v);
680                }
681            }
682            if stacked {
683                ext.1 = ext.1.max(p);
684                ext.0 = ext.0.min(m);
685            }
686        }
687        percent
688    }

734    /// A logarithmic value axis' scale in exponents — (min, max, unit), so its labels are
735    /// `base^min`, `base^(min+unit)`… — covering the positive values: whole powers, as PowerPoint
736    /// and LibreOffice draw them. `None` when the axis isn't logarithmic or nothing is positive.
737    pub fn log_axis_scale(&self, max_intervals: usize) -> Option<(f64, f64, f64)> {
738        let b = self.y_log_base.filter(|b| b.is_finite() && *b >= 2.0)?;
739        let vals = self.series.iter().filter(|s| !self.on_secondary(s)).flat_map(|s| s.values.iter().flatten().copied());
740        let (lo, hi) = vals
741            .filter(|v| v.is_finite() && *v > 0.0)
742            .fold(None, |r: Option<(f64, f64)>, v| Some(r.map_or((v, v), |(a, z)| (a.min(v), z.max(v)))))?;
743        let e = |v: f64| v.log(b);
744        let lo = self.y_min.filter(|v| *v > 0.0).map_or(e(lo).floor(), e);
745        let hi = self.y_max.filter(|v| *v > 0.0).map_or(e(hi).ceil(), e).max(lo + 1.0);
746        let unit = ((hi - lo) / max_intervals.max(1) as f64).ceil().max(1.0);
747        Some((lo, lo + ((hi - lo) / unit).ceil() * unit, unit))
748    }

838/// The (min, max, major unit) of an axis over the data range [lo, hi]: `fmin`/`fmax`/`major`
839/// where the chart fixes them; otherwise Office's automatic limits, rounded out to the smallest
840/// 1-2-5 unit giving at most `max_intervals` intervals:
841/// The data range [lo, hi] an axis covers, clamped to ±1e100; (0, 1) without finite values.
842///
843/// - **Maximum:** `hi + (hi − lo)/20` (with `headroom`), rounded up to the unit.
844/// - **Minimum:** zero when every value is ≥ 0 and `lo` is below 5/6 of `hi`; otherwise
845///   `lo − (hi − lo)/20`, rounded down to the unit, so values bunched high get an axis that starts
846///   near them.
847/// - All-negative values mirror this: the axis ends at zero when `hi` is above 5/6 of `lo`.
848///   Mixed signs get headroom at both ends.
849fn axis_data_range(lo: f64, hi: f64) -> (f64, f64) {
850    if lo.is_finite() && hi.is_finite() { (lo.min(hi).clamp(-1e100, 1e100), lo.max(hi).clamp(-1e100, 1e100)) } else { (0.0, 1.0) }
851}
852
853/// Office's automatic (min, max) of the data range [lo, hi]. "How chart axis limits are determined",
854/// as Jon Peltier summarises it: 5 % of the data range past the data, and zero when the values don't
855/// sit within the top sixth of their largest magnitude. A single value is measured from zero.
856fn auto_limits(lo: f64, hi: f64, headroom: bool) -> (f64, f64) {
857    let flat = hi - lo <= 1e-12 * lo.abs().max(hi.abs());
858    let pad = if !headroom {
859        0.0
860    } else if flat {
861        hi.abs() / 20.0
862    } else {
863        (hi - lo) / 20.0
864    };
865    let auto_lo = if lo >= 0.0 && (flat || lo < hi * 5.0 / 6.0) { 0.0 } else { lo - pad };
866    let auto_hi = if hi <= 0.0 && (flat || hi > lo * 5.0 / 6.0) { 0.0 } else { hi + pad };
867    (auto_lo, auto_hi)
868}

870/// The ends of a span too small to show: with one fixed limit on the far side of the data, the
871/// automatic end gives way; otherwise the span is one `unit` wide.
872fn flat_span_ends(fmin: Option<f64>, fmax: Option<f64>, lo: f64, hi: f64, unit: f64) -> (f64, f64) {
873    match (fmin, fmax) {
874        (None, Some(_)) => (if hi > 0.0 { 0.0 } else { hi - hi.abs().max(unit) }, hi),
875        (Some(_), None) if lo < 0.0 => (lo, 0.0),
876        _ => (lo, lo + unit),
877    }
878}

880/// The major step: `major` when it gives at most 1000 intervals, else the smallest 1-2-5 step
881/// that fits `n` intervals over [lo, hi].
882fn axis_step(major: Option<f64>, lo: f64, hi: f64, n: f64, unit: f64) -> f64 {
883    match major.filter(|m| *m > 0.0 && (hi - lo) / m <= 1000.0) {
884        Some(m) => m,
885        None => {
886            let mut step = 10f64.powf(((hi - lo) / n).log10().floor());
887            // Subnormal spans underflow to a zero (or non-finite) step: fall back to the unit.
888            if !(step > 0.0 && step.is_finite()) {
889                step = unit;
890            }
891            let mut k = 0;
892            // 1, 2, 5, 10, 20… × the magnitude until the intervals fit.
893            while ((hi / step).ceil() - (lo / step).floor()) > n && k < 30 {
894                step *= if matches!(k % 3, 1) { 2.5 } else { 2.0 };
895                k += 1;
896            }
897            step
898        }
899    }
900}

902pub fn axis_scale(
903    lo: f64,
904    hi: f64,
905    fmin: Option<f64>,
906    fmax: Option<f64>,
907    major: Option<f64>,
908    max_intervals: usize,
909    headroom: bool,
910) -> (f64, f64, f64) {
911    // Values past ±1e100 aren't axis values; clamping keeps the arithmetic finite.
912    let fin = |v: Option<f64>| v.filter(|v| v.is_finite()).map(|v| v.clamp(-1e100, 1e100));
913    let (fmin, fmax) = (fin(fmin), fin(fmax));
914    let (lo, hi) = axis_data_range(lo, hi);
915    let (auto_lo, auto_hi) = auto_limits(lo, hi, headroom);
916    let (mut lo, mut hi) = (fmin.unwrap_or(auto_lo), fmax.unwrap_or(auto_hi));
917    let mag = lo.abs().max(hi.abs());
918    // Relative to the data's magnitude, so tiny values don't count as flat; all-zero falls back to 1.
919    let unit = if mag == 0.0 { 1.0 } else { mag };
920    if hi - lo <= 1e-12 * mag {
921        (lo, hi) = flat_span_ends(fmin, fmax, lo, hi, unit);
922    }
923    let n = max_intervals.clamp(1, 50) as f64;
924    let step = axis_step(fin(major), lo, hi, n, unit);
925    let min = fmin.unwrap_or((lo / step + 1e-9).floor() * step);
926    let max = fmax.unwrap_or((hi / step - 1e-9).ceil() * step).max(min + step);
927    (min, max, step)
928}

930/// "Nice" axis scale: (min, max, major step) covering [lo, hi].
931pub fn nice_scale(lo: f64, hi: f64) -> (f64, f64, f64) {
932    let (lo, hi) = if lo.is_finite() && hi.is_finite() { (lo.min(hi), lo.max(hi)) } else { (0.0, 1.0) };
933    let data_mag = lo.abs().max(hi.abs());
934    let span = if hi - lo <= 1e-12 * data_mag { data_mag } else { hi - lo };
935    let span = if span == 0.0 { 1.0 } else { span };
936    let raw = span / 5.0;
937    let mag = 10f64.powf(raw.log10().floor());
938    let norm = raw / mag;
939    let step = if norm <= 1.0 {
940        1.0
941    } else if norm <= 2.0 {
942        2.0
943    } else if norm <= 2.5 {
944        2.5
945    } else if norm <= 5.0 {
946        5.0
947    } else {
948        10.0
949    } * mag;
950    // Subnormal spans underflow to a zero or NaN step: use the span itself.
951    let step = if step > 0.0 && step.is_finite() { step } else { span };
952    let min = (lo / step).floor() * step;
953    let max = (hi / step).ceil() * step;
954    let max = if (max - min).abs() <= 1e-12 * data_mag { min + step } else { max };
955    (min, max, step)
956}

1237/// One histogram bin: `[lo, hi]` (the first bin is closed on both sides, later ones open on the
1238/// left, as Excel bins) and how many observations fall in it.
1239#[derive(Clone, Copy, Debug, PartialEq)]
1240pub struct Bin {
1241    pub lo: f64,
1242    pub hi: f64,
1243    pub count: usize,
1244}
1245
1246/// Bins a histogram has at most (a hostile bin size cannot make millions).
1247pub const MAX_BINS: usize = 1000;
1248
1249/// A box and whisker summary of one group: quartiles by Excel's exclusive method, whiskers at
1250/// the furthest points within 1.5 IQR of the box, and the points beyond them.
1251#[derive(Clone, Debug, PartialEq)]
1252pub struct BoxStats {
1253    pub low: f64,
1254    pub q1: f64,
1255    pub median: f64,
1256    pub q3: f64,
1257    pub high: f64,
1258    pub mean: f64,
1259    pub outliers: Vec<f64>,
1260}

1262impl Chart {
1263    /// The first series' finite observations.
1264    fn observations(&self) -> Vec<f64> {
1265        self.series.first().map(|s| s.values.iter().flatten().copied().filter(|v| v.is_finite()).collect()).unwrap_or_default()
1266    }
1267
1268    /// The histogram's bins over the first series: by `bin_size`, else `bin_count`, else
1269    /// Scott's rule (3.49·σ·n^(-1/3)). Empty without observations.
1270    pub fn histogram_bins(&self) -> Vec<Bin> {
1271        let mut v = self.observations();
1272        if v.is_empty() {
1273            return vec![];
1274        }
1275        v.sort_by(f64::total_cmp);
1276        let (lo, hi) = (v[0], v[v.len() - 1]);
1277        let span = hi - lo;
1278        let n = v.len() as f64;
1279        let width = if let Some(w) = self.bin_size.filter(|w| w.is_finite() && *w > 0.0) {
1280            w
1281        } else if let Some(k) = self.bin_count.filter(|k| *k > 0) {
1282            span / f64::from(k)
1283        } else {
1284            let mean = v.iter().sum::<f64>() / n;
1285            let sd = (v.iter().map(|x| (x - mean).powi(2)).sum::<f64>() / (n - 1.0).max(1.0)).sqrt();
1286            3.49 * sd / n.cbrt()
1287        };
1288        let fixed = self.bin_count.filter(|k| *k > 0).filter(|_| self.bin_size.filter(|w| w.is_finite() && *w > 0.0).is_none());
1289        let count = if span <= 0.0 || !width.is_finite() || width <= 0.0 {
1290            1
1291        } else if let Some(k) = fixed {
1292            (k as usize).clamp(1, MAX_BINS)
1293        } else {
1294            ((span / width).ceil() as usize).clamp(1, MAX_BINS)
1295        };
1296        let width = if count == 1 && (span <= 0.0 || !width.is_finite() || width <= 0.0) { span.max(1.0) } else { width.max(span / MAX_BINS as f64) };
1297        let mut bins: Vec<Bin> = (0..count).map(|i| Bin { lo: lo + width * i as f64, hi: lo + width * (i + 1) as f64, count: 0 }).collect();
1298        for x in v {
1299            // (lo, hi] bins, the first one also holding the minimum.
1300            let mut i = (((x - lo) / width).ceil() as usize).saturating_sub(1).min(count - 1);
1301            // The division can round across an edge: settle on the bin whose stored bounds hold x.
1302            while i > 0 && x <= bins[i - 1].hi {
1303                i -= 1;
1304            }
1305            while i + 1 < count && x > bins[i].hi {
1306                i += 1;
1307            }
1308            if let Some(b) = bins.get_mut(i) {
1309                b.count += 1;
1310            }
1311        }
1312        bins
1313    }

1316    /// touching columns, labelled by bin range. A Pareto sorts the bins by count and adds the
1317    /// cumulative percentage as a line on a 0–100 % secondary axis.
1318    pub fn histogram_view(&self) -> Chart {
1319        let mut bins = self.histogram_bins();
1320        let pareto = self.kind == ChartType::Pareto;
1321        if pareto {
1322            bins.sort_by_key(|b| std::cmp::Reverse(b.count));
1323        }
1324        let fmt = |v: f64| {
1325            let r = (v * 100.0).round() / 100.0;
1326            if (r - r.round()).abs() < 1e-9 { format!("{}", r.round() as i64) } else { format!("{r}") }
1327        };
1328        let first_lo = bins.iter().map(|b| b.lo).fold(f64::INFINITY, f64::min);
1329        let mut v = Chart {
1330            kind: if pareto { ChartType::Combo } else { ChartType::Column },
1331            categories: bins
1332                .iter()
1333                .map(|b| if b.lo <= first_lo { format!("[{}, {}]", fmt(b.lo), fmt(b.hi)) } else { format!("({}, {}]", fmt(b.lo), fmt(b.hi)) })
1334                .collect(),
1335            series: vec![],
1336            gap_width: 0.0,
1337            raw: None,
1338            verbatim: false,
1339            subtotals: vec![],
1340            ..self.clone()

1342        let name = self.series.first().map(|s| s.name.clone()).unwrap_or_default();
1343        let base = self.series.first().cloned().unwrap_or_default();
1344        v.series.push(Series {

1369            v.y2_major = Some(0.2);

1371        }
1372        v.vary_colors = false;

1376    /// The box chart's groups (distinct category labels in order of appearance; one unnamed
1377    /// group without categories) and, per series, each group's summary (None when it has no
1378    /// points).
1379    ///
1380    /// Each value is put in its group's bucket once (groups are found by name in a map), and the
1381    /// series x groups result is held to [`MAX_BOX_CELLS`]: groups past it are left out.
1382    pub fn box_groups(&self) -> (Vec<String>, Vec<Vec<Option<BoxStats>>>) {
1383        let max_groups = (MAX_BOX_CELLS / self.series.len().max(1)).max(1);
1384        let mut groups: Vec<String> = Vec::new();
1385        let mut index: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
1386        for c in &self.categories {
1387            if !index.contains_key(c.as_str()) && groups.len() < max_groups {
1388                index.insert(c, groups.len());
1389                groups.push(c.clone());
1390            }
1391        }
1392        if groups.is_empty() {
1393            groups.push(String::new());
1394            index.insert("", 0);
1395        }
1396        // A value past the categories belongs to the unnamed group, when there is one.
1397        let past_end = index.get("").copied();
1398        let stats = self
1399            .series
1400            .iter()
1401            .map(|s| {
1402                let mut buckets: Vec<Vec<f64>> = vec![Vec::new(); groups.len()];
1403                for (i, v) in s.values.iter().enumerate() {
1404                    let Some(v) = v.filter(|v| v.is_finite()) else { continue };
1405                    let g = match self.categories.get(i) {
1406                        Some(c) => index.get(c.as_str()).copied(),
1407                        None => past_end,
1408                    };
1409                    if let Some(b) = g.and_then(|g| buckets.get_mut(g)) {
1410                        b.push(v);
1411                    }
1412                }
1413                buckets.into_iter().map(|pts| box_stats_by(pts, self.inclusive_quartiles)).collect()
1414            })
1415            .collect();
1416        (groups, stats)
1417    }

1429/// Excel's QUARTILE.EXC position `p·(n+1)` (1-based) on sorted data, clamped to the ends.
1430fn quantile_exc(v: &[f64], p: f64) -> f64 {
1431    let n = v.len();
1432    let pos = (p * (n as f64 + 1.0)).clamp(1.0, n as f64);
1433    let i = pos.floor() as usize;
1434    let frac = pos - i as f64;
1435    let a = v[i.saturating_sub(1).min(n - 1)];
1436    let b = v[i.min(n - 1)];
1437    a + (b - a) * frac
1438}
1439
1440/// Excel's QUARTILE.INC position `p·(n-1)` (0-based) on sorted data.
1441fn quantile_inc(v: &[f64], p: f64) -> f64 {
1442    let n = v.len();
1443    let pos = (p * (n as f64 - 1.0)).clamp(0.0, (n - 1) as f64);
1444    let i = pos.floor() as usize;
1445    let a = v[i.min(n - 1)];
1446    let b = v[(i + 1).min(n - 1)];
1447    a + (b - a) * (pos - i as f64)
1448}

1455/// The summary of one group's points; `inclusive` picks QUARTILE.INC over QUARTILE.EXC.
1456pub fn box_stats_by(mut pts: Vec<f64>, inclusive: bool) -> Option<BoxStats> {
1457    if pts.is_empty() {
1458        return None;
1459    }
1460    pts.sort_by(f64::total_cmp);
1461    let q = |p: f64| if inclusive { quantile_inc(&pts, p) } else { quantile_exc(&pts, p) };
1462    let (q1, median, q3) = (q(0.25), q(0.5), q(0.75));
1463    let iqr = q3 - q1;
1464    let (lo_fence, hi_fence) = (q1 - 1.5 * iqr, q3 + 1.5 * iqr);
1465    let inside: Vec<f64> = pts.iter().copied().filter(|v| *v >= lo_fence && *v <= hi_fence).collect();
1466    let low = inside.first().copied().unwrap_or(q1).min(q1);
1467    let high = inside.last().copied().unwrap_or(q3).max(q3);
1468    let mean = pts.iter().sum::<f64>() / pts.len() as f64;
1469    let outliers = pts.iter().copied().filter(|v| *v < lo_fence || *v > hi_fence).collect();
1470    Some(BoxStats { low, q1, median, q3, high, mean, outliers })
1471}

1473/// A node of a treemap or sunburst hierarchy: its label, the total of the leaves under it, and
1474/// its children (none for a leaf, which names its category index).
1475#[derive(Clone, Debug, PartialEq)]
1476pub struct TreeNode {
1477    pub name: String,
1478    pub value: f64,
1479    pub leaf: Option<usize>,
1480    pub children: Vec<TreeNode>,
1481}
1482
1483/// Levels of hierarchy a chart may have (a hostile file cannot nest deeper).
1484pub const MAX_LEVELS: usize = 16;
1485
1486impl Chart {
1487    /// The first series as a hierarchy: top-level nodes, each holding the consecutive
1488    /// categories that share its label, down to the leaves. Without outer levels every
1489    /// category is a top-level leaf. Negative and missing values count as zero.
1490    pub fn category_tree(&self) -> Vec<TreeNode> {
1491        let Some(s) = self.series.first() else { return vec![] };
1492        let n = self.categories.len().max(s.values.len());
1493        let levels: Vec<&Vec<String>> = self.category_levels.iter().take(MAX_LEVELS).collect();
1494        // Each leaf's path from the top level down, blanks filled from the row above.
1495        // Labels are shared, not copied: an outer label given once stands for every leaf under it.
1496        use std::rc::Rc;
1497        let empty: Rc<str> = Rc::from("");
1498        let mut last: Vec<Rc<str>> = vec![empty.clone(); levels.len()];
1499        let paths: Vec<Vec<Rc<str>>> = (0..n)
1500            .map(|i| {
1501                let mut path = Vec::with_capacity(levels.len() + 1);
1502                for (d, lv) in levels.iter().enumerate().rev() {
1503                    let l = lv.get(i).map(|x| x.trim()).unwrap_or("");
1504                    if !l.is_empty() {
1505                        last[d] = Rc::from(l);
1506                        // A new outer label starts fresh inner groups.
1507                        for inner in last.iter_mut().take(d) {
1508                            *inner = empty.clone();
1509                        }
1510                    }
1511                    path.push(last[d].clone());
1512                }
1513                path.push(self.categories.get(i).map_or_else(|| empty.clone(), |c| Rc::from(c.as_str())));
1514                path
1515            })
1516            .collect();
1517        let value = |i: usize| s.values.get(i).copied().flatten().filter(|v| v.is_finite()).unwrap_or(0.0).max(0.0);
1518        fn build(rows: &[(usize, &[std::rc::Rc<str>])], value: &dyn Fn(usize) -> f64) -> Vec<TreeNode> {
1519            let mut out: Vec<TreeNode> = Vec::new();
1520            let mut i = 0;
1521            while i < rows.len() {
1522                let (idx, path) = rows[i];
1523                if path.len() <= 1 {
1524                    out.push(TreeNode {
1525                        name: path.first().map(|p| p.to_string()).unwrap_or_default(),
1526                        value: value(idx),
1527                        leaf: Some(idx),
1528                        children: vec![],
1529                    });
1530                    i += 1;
1531                    continue;
1532                }
1533                let head = &path[0];
1534                let mut j = i;
1535                while j < rows.len() && rows[j].1.len() > 1 && &rows[j].1[0] == head {
1536                    j += 1;
1537                }
1538                let sub: Vec<(usize, &[std::rc::Rc<str>])> = rows[i..j].iter().map(|(k, p)| (*k, &p[1..])).collect();
1539                let children = build(&sub, value);
1540                out.push(TreeNode { name: head.to_string(), value: children.iter().map(|c| c.value).sum(), leaf: None, children });
1541                i = j;
1542            }
1543            out
1544        }
1545        let rows: Vec<(usize, &[Rc<str>])> = paths.iter().enumerate().map(|(i, p)| (i, p.as_slice())).collect();
1546        build(&rows, &value)
1547    }
1548}
