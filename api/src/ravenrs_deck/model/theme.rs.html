8#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
9#[serde(default, rename_all = "camelCase")]
10pub struct FontSet {
11    pub latin: String,
12    pub ea: String,
13    pub cs: String,
14    /// Per-script typefaces (`a:font script="Arab" typeface="…"`), in document order.
15    #[serde(skip_serializing_if = "Vec::is_empty")]
16    pub scripts: Vec<ScriptFont>,
17}

19/// A theme font for one script (ISO 15924 tag, e.g. `Arab`).
20#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
21#[serde(default, rename_all = "camelCase")]
22pub struct ScriptFont {
23    pub script: String,
24    pub typeface: String,
25}
26
27impl FontSet {
28    /// The typeface the theme gives `script` (`"Arab"`), if any.
29    pub fn script(&self, script: &str) -> Option<&str> {
30        self.scripts.iter().find(|f| f.script.eq_ignore_ascii_case(script) && !f.typeface.is_empty()).map(|f| f.typeface.as_str())
31    }
32}

56#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]

80impl Theme {
81    /// Resolve a theme font reference (`+mj-lt`, `+mn-ea`…) to a family name.
82    pub fn font(&self, name: &str) -> String {
83        match name {
84            "+mj-lt" => self.fonts.major.latin.clone(),
85            "+mn-lt" => self.fonts.minor.latin.clone(),
86            "+mj-ea" => self.fonts.major.ea.clone(),
87            "+mn-ea" => self.fonts.minor.ea.clone(),
88            "+mj-cs" => self.fonts.major.cs.clone(),
89            "+mn-cs" => self.fonts.minor.cs.clone(),
90            other => other.to_string(),
91        }
92    }
93}

107/// Our default format scheme: flat solid fills, a soft gradient, and a gentle shadow for "intense".
108pub fn default_format() -> FormatScheme {
109    let grad = |a: i32, b: i32| {
110        Fill::Gradient(Gradient {
111            tile: [0.0; 4],
112            stops: vec![
113                GradientStop { pos: 0.0, color: ph().with(ColorTransform::LumMod(a)).with(ColorTransform::SatMod(103000)) },
114                GradientStop { pos: 1.0, color: ph().with(ColorTransform::LumMod(b)) },
115            ],
116            shape: GradientShape::Linear { angle: 90.0, scaled: false },
117            rotate_with_shape: true,
118            flip: None,
119        })
120    };
121    FormatScheme {
122        name: "DeckCraft".into(),
123        fills: vec![Fill::solid(ph()), grad(110000, 92000), grad(105000, 80000)],
124        lines: vec![Line::solid(ph(), 0.75), Line::solid(ph(), 1.0), Line::solid(ph(), 1.5)],
125        effects: vec![
126            Effects::default(),
127            Effects::default(),
128            Effects {
129                outer_shadow: Some(Shadow {
130                    color: ColorRef::rgb(Rgba::BLACK).with(ColorTransform::Alpha(40000)),
131                    blur: 6.0,
132                    dist: 3.0,
133                    dir: 90.0,
134                    inner: false,
135                    sx: 1.0,
136                    sy: 1.0,
137                    kx: 0.0,
138                    ky: 0.0,
139                    align: "t".into(),
140                    rotate_with_shape: false,
141                }),

153/// Design ▸ Variants: four colour variations of `t`, built from its built-in colours when it is
154/// a built-in theme (so picking one variant after another doesn't drift), else from its own.
155/// 1: as designed; 2 and 3: the accents rotated; 4: dark (light and dark colours swapped).
156pub fn theme_variants(t: &Theme) -> Vec<ColorScheme> {
157    let base = builtin_themes().into_iter().find(|b| b.name == t.name).map_or_else(|| t.colors.clone(), |b| b.colors);
158    let rotate = |by: usize| {
159        let mut c = base.clone();
160        let accents: Vec<_> = base.colors.get(4..10).map(<[_]>::to_vec).unwrap_or_default();
161        for (i, slot) in c.colors.iter_mut().skip(4).take(6).enumerate() {
162            if let Some(a) = accents.get((i + by) % accents.len().max(1)) {
163                *slot = *a;
164            }
165        }
166        c
167    };
168    let mut dark = base.clone();
169    dark.colors.swap(0, 1);
170    dark.colors.swap(2, 3);
171    let mut out = vec![base.clone(), rotate(1), rotate(3), dark];
172    for (i, c) in out.iter_mut().enumerate().skip(1) {
173        c.name = format!("{} {}", base.name, i + 1);
174    }
175    out
176}

298fn shadow(blur: f64, dist: f64, alpha: i32) -> Effects {
299    Effects {
300        outer_shadow: Some(Shadow {
301            color: ColorRef::rgb(Rgba::BLACK).with(ColorTransform::Alpha(alpha)),
302            blur,
303            dist,
304            dir: 90.0,
305            inner: false,
306            sx: 1.0,
307            sy: 1.0,
308            kx: 0.0,
309            ky: 0.0,
310            align: "b".into(),
311            rotate_with_shape: false,
312        }),
313        ..Default::default()
314    }
315}

317fn glow(radius: f64) -> Effects {

321/// Named effect schemes for Design ▸ Effects. Each one replaces the theme's format scheme
322/// (subtle, moderate and intense fills, lines and effects) and leaves the colour and font schemes.
323pub fn builtin_format_schemes() -> Vec<FormatScheme> {
324    let mut soft = default_format();
325    soft.name = "Soft".into();
326    soft.lines = vec![Line::solid(ph(), 0.5), Line::solid(ph(), 0.75), Line::solid(ph(), 1.0)];
327    soft.effects = vec![shadow(16.0, 1.0, 18_000), shadow(12.0, 2.0, 22_000), shadow(10.0, 3.0, 28_000)];
328
329    let mut crisp = default_format();
330    crisp.name = "Crisp".into();
331    crisp.fills = vec![Fill::solid(ph()), Fill::solid(ph()), Fill::solid(ph())];
332    crisp.lines = vec![Line::solid(ph(), 1.0), Line::solid(ph(), 1.5), Line::solid(ph(), 2.25)];
333    crisp.effects = vec![Effects::default(), Effects::default(), shadow(0.0, 2.0, 55_000)];
334
335    let mut lift = default_format();
336    lift.name = "Lift".into();
337    lift.effects = vec![Effects::default(), shadow(8.0, 6.0, 28_000), shadow(12.0, 10.0, 35_000)];
338
339    let mut glow_scheme = default_format();
340    glow_scheme.name = "Glow".into();
341    glow_scheme.effects = vec![Effects::default(), glow(6.0), glow(12.0)];
342
343    let mut plain = default_format();
344    plain.name = "Plain".into();

347    plain.effects = vec![Effects::default(), Effects::default(), Effects::default()];

349    vec![default_format(), soft, crisp, lift, glow_scheme, plain]
