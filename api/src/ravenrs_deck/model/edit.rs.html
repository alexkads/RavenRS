8/// (paragraph, char offset in the paragraph).
9pub type Pos = (usize, usize);

11fn clamp(body: &TextBody, p: Pos) -> Pos {
12    if body.paragraphs.is_empty() {
13        return (0, 0);
14    }
15    let pi = p.0.min(body.paragraphs.len() - 1);
16    let n = body.paragraphs.get(pi).map(Paragraph::char_len).unwrap_or(0);
17    (pi, p.1.min(n))
18}

20fn order(a: Pos, b: Pos) -> (Pos, Pos) {
21    if a <= b { (a, b) } else { (b, a) }
22}

24/// Split the paragraph's runs so a run boundary falls at char `at`; returns the index of the
25/// first run starting at or after `at`.
26fn split_at(p: &mut Paragraph, at: usize) -> usize {
27    let mut pos = 0;
28    let mut i = 0;
29    while i < p.runs.len() {
30        let len = p.runs.get(i).map(Run::char_len).unwrap_or(0);
31        if pos == at {
32            return i;
33        }
34        if at < pos + len {
35            let Some(r) = p.runs.get(i) else { return i };
36            if r.kind != RunKind::Text {
37                return i + 1;
38            }
39            let k = at - pos;
40            let byte = r.text.char_indices().nth(k).map(|(b, _)| b).unwrap_or(r.text.len());
41            let (a, b) = r.text.split_at(byte);
42            let second = Run { text: b.to_string(), props: r.props.clone(), kind: RunKind::Text };
43            let first = a.to_string();
44            if let Some(r) = p.runs.get_mut(i) {
45                r.text = first;
46            }
47            p.runs.insert(i + 1, second);
48            return i + 1;
49        }
50        pos += len;
51        i += 1;
52    }
53    p.runs.len()
54}

56/// The explicit run properties in effect at `pos` (the run before the caret, or the end mark).
57pub fn props_at(body: &TextBody, pos: Pos) -> RunProps {
58    let (pi, ch) = clamp(body, pos);
59    let Some(p) = body.paragraphs.get(pi) else { return RunProps::default() };
60    let mut acc = 0;
61    let mut last = None;
62    for r in &p.runs {
63        let len = r.char_len();
64        if ch <= acc + len && ch > acc {
65            return r.props.clone();
66        }
67        if acc + len >= ch && ch == 0 {
68            return r.props.clone();
69        }
70        acc += len;
71        last = Some(&r.props);
72    }
73    last.cloned().unwrap_or_else(|| p.end_props.clone())
74}

76/// Insert `text` at `pos` (with `props`, else the props at `pos`). `\n` starts a new paragraph
77/// (same level and paragraph props); `\u{b}` inserts a line break. Returns the caret after it.
78pub fn insert(body: &mut TextBody, pos: Pos, text: &str, props: Option<RunProps>) -> Pos {
79    if body.paragraphs.is_empty() {
80        body.paragraphs.push(Paragraph::default());
81    }
82    let (pi0, mut ch) = clamp(body, pos);
83    let props = props.unwrap_or_else(|| props_at(body, (pi0, ch)));
84    let text = text.replace("\r\n", "\n").replace('\r', "\n");
85    // Work on the paragraph being edited apart from the body, so each new paragraph is pushed to a
86    // list and the body's vector is spliced once instead of shifted for every `\n`.
87    let Some(slot) = body.paragraphs.get_mut(pi0) else { return (pi0, ch) };
88    let mut cur = std::mem::take(slot);
89    let mut done: Vec<Paragraph> = Vec::new();
90    let mut first = true;
91    for part in text.split('\n') {
92        if !first {
93            // Split the current paragraph at ch.
94            let idx = split_at(&mut cur, ch);
95            let tail: Vec<Run> = cur.runs.drain(idx..).collect();
96            let newp = Paragraph {
97                level: cur.level,
98                props: cur.props.clone(),
99                runs: tail,
100                def_props: cur.def_props.clone(),
101                end_props: cur.end_props.clone(),
102            };
103            cur.end_props = props.clone();
104            done.push(std::mem::replace(&mut cur, newp));
105            ch = 0;
106        }
107        first = false;
108        // The run index just after the last insertion: the next one lands there with no rescan.
109        let mut next: Option<usize> = None;
110        for (k, seg) in part.split('\u{b}').enumerate() {
111            if k > 0 {
112                let idx = next.unwrap_or_else(|| split_at(&mut cur, ch));
113                cur.runs.insert(idx, Run { text: String::new(), props: props.clone(), kind: RunKind::Break });
114                ch += 1;
115                next = Some(idx + 1);
116            }
117            if seg.is_empty() {
118                continue;
119            }
120            let idx = next.unwrap_or_else(|| split_at(&mut cur, ch));
121            cur.runs.insert(idx, Run::with(seg, props.clone()));
122            ch += seg.chars().count();
123            next = Some(idx + 1);
124        }
125        cur.normalize();
126        if cur.runs.is_empty() {
127            cur.end_props = props.clone();
128        }
129    }
130    let pi = pi0 + done.len();
131    done.push(cur);
132    body.paragraphs.splice(pi0..=pi0, done);
133    (pi, ch)
134}

136/// Delete the range between `a` and `b`; returns the caret at the start.
137pub fn delete(body: &mut TextBody, a: Pos, b: Pos) -> Pos {
138    if body.paragraphs.is_empty() {
139        return (0, 0);
140    }
141    let (s, e) = order(clamp(body, a), clamp(body, b));
142    if s == e {
143        return s;
144    }
145    if s.0 == e.0 {
146        if let Some(p) = body.paragraphs.get_mut(s.0) {
147            let end_props = props_of_range(p, s.1);
148            let i0 = split_at(p, s.1);
149            let i1 = split_at(p, e.1);
150            if i1 > i0 {
151                p.runs.drain(i0..i1);
152            }
153            p.normalize();
154            if p.runs.is_empty() {
155                p.end_props = end_props;
156            }
157        }
158        return s;
159    }
160    // Keep the head of s.0 and the tail of e.0, joined; drop paragraphs between.
161    let tail: Vec<Run> = match body.paragraphs.get_mut(e.0) {
162        Some(p) => {
163            let i = split_at(p, e.1);
164            p.runs.drain(i..).collect()
165        }
166        None => vec![],
167    };
168    if let Some(p) = body.paragraphs.get_mut(s.0) {
169        let i = split_at(p, s.1);
170        p.runs.truncate(i);
171        p.runs.extend(tail);
172        p.normalize();
173    }
174    if e.0 > s.0 && e.0 < body.paragraphs.len() {
175        body.paragraphs.drain(s.0 + 1..=e.0);
176    }
177    s
178}

180fn props_of_range(p: &Paragraph, at: usize) -> RunProps {
181    let mut acc = 0;
182    for r in &p.runs {
183        let len = r.char_len();
184        if at < acc + len || (at == acc + len && at > 0) {
185            return r.props.clone();
186        }
187        acc += len;
188    }
189    p.runs.last().map(|r| r.props.clone()).unwrap_or_else(|| p.end_props.clone())
190}

192/// Apply `f` to the run properties of every run in [a, b). An empty range changes the empty
193/// paragraph's end mark (so typing there picks the formatting up).
194pub fn format(body: &mut TextBody, a: Pos, b: Pos, f: &dyn Fn(&mut RunProps)) {
195    if body.paragraphs.is_empty() {
196        return;
197    }
198    let (s, e) = order(clamp(body, a), clamp(body, b));
199    for pi in s.0..=e.0 {
200        let Some(p) = body.paragraphs.get_mut(pi) else { continue };
201        let from = if pi == s.0 { s.1 } else { 0 };
202        let to = if pi == e.0 { e.1 } else { p.char_len() };
203        if p.runs.is_empty() || (from == to && s == e) {
204            f(&mut p.end_props);
205            continue;
206        }
207        let i0 = split_at(p, from);
208        let i1 = split_at(p, to);
209        for r in p.runs.iter_mut().take(i1).skip(i0) {
210            f(&mut r.props);
211        }
212        if to == p.char_len() {
213            f(&mut p.end_props);
214        }
215        p.normalize();
216    }
217}

219/// Apply `f` to every run in the body (and end marks).
220pub fn format_all(body: &mut TextBody, f: &dyn Fn(&mut RunProps)) {
221    for p in &mut body.paragraphs {
222        for r in &mut p.runs {
223            f(&mut r.props);
224        }
225        f(&mut p.end_props);
226        p.normalize();
227    }
228}

230/// The text between two positions (`\n` between paragraphs).
231pub fn text_range(body: &TextBody, a: Pos, b: Pos) -> String {
232    let (s, e) = order(clamp(body, a), clamp(body, b));
233    let mut out = String::new();
234    for pi in s.0..=e.0 {
235        let Some(p) = body.paragraphs.get(pi) else { continue };
236        let t: Vec<char> = p.text().chars().collect();
237        let from = if pi == s.0 { s.1 } else { 0 };
238        let to = if pi == e.0 { e.1 } else { t.len() };
239        out.extend(t.get(from.min(t.len())..to.min(t.len())).unwrap_or(&[]));
240        if pi != e.0 {
241            out.push('\n');
242        }
243    }
244    out
245}

247/// A formatted copy of the range.
248pub fn slice(body: &TextBody, a: Pos, b: Pos) -> TextBody {
249    let (s, e) = order(clamp(body, a), clamp(body, b));
250    let mut out = TextBody { body: body.body.clone(), list_style: body.list_style.clone(), paragraphs: vec![] };
251    for pi in s.0..=e.0 {
252        let Some(p) = body.paragraphs.get(pi) else { continue };
253        let mut p = p.clone();
254        let to = if pi == e.0 { e.1 } else { p.char_len() };
255        let i1 = split_at(&mut p, to);
256        p.runs.truncate(i1);
257        let from = if pi == s.0 { s.1 } else { 0 };
258        let i0 = split_at(&mut p, from);
259        p.runs.drain(..i0);
260        out.paragraphs.push(p);
261    }
262    out
263}

265/// Insert a formatted body at `pos` (paste): first paragraph merges into the current one.
266pub fn insert_body(body: &mut TextBody, pos: Pos, src: &TextBody) -> Pos {
267    let mut caret = clamp(body, pos);
268    // Plain text runs go straight in at the run index after the previous one and the paragraph is
269    // normalized once, not rescanned and merged for every source run.
270    let mut next: Option<usize> = None;
271    let mut dirty = false;
272    for (k, sp) in src.paragraphs.iter().enumerate() {
273        if k > 0 {
274            if let (true, Some(p)) = (dirty, body.paragraphs.get_mut(caret.0)) {
275                p.normalize();
276            }
277            (next, dirty) = (None, false);
278            caret = insert(body, caret, "\n", None);
279            if let Some(p) = body.paragraphs.get_mut(caret.0) {
280                p.level = sp.level;
281                p.props = sp.props.clone();
282            }
283        }
284        for r in &sp.runs {
285            if r.kind == RunKind::Text
286                && !r.text.is_empty()
287                && !r.text.contains(['\n', '\r', '\u{b}'])
288                && let Some(p) = body.paragraphs.get_mut(caret.0)
289            {
290                let idx = next.unwrap_or_else(|| split_at(p, caret.1));
291                p.runs.insert(idx, Run::with(&r.text, r.props.clone()));
292                caret.1 += r.text.chars().count();
293                (next, dirty) = (Some(idx + 1), true);
294                continue;
295            }
296            // Anything else takes the general path, after the pending runs are merged.
297            if let (true, Some(p)) = (dirty, body.paragraphs.get_mut(caret.0)) {
298                p.normalize();
299            }
300            (next, dirty) = (None, false);
301            match r.kind {
302                RunKind::Text => caret = insert(body, caret, &r.text, Some(r.props.clone())),
303                RunKind::Break => caret = insert(body, caret, "\u{b}", Some(r.props.clone())),
304                _ => {
305                    if let Some(p) = body.paragraphs.get_mut(caret.0) {
306                        let idx = split_at(p, caret.1);
307                        p.runs.insert(idx, r.clone());
308                        caret.1 += r.char_len();
309                    }
310                }
311            }
312        }
313    }
314    if let (true, Some(p)) = (dirty, body.paragraphs.get_mut(caret.0)) {
315        p.normalize();
316    }
317    caret
318}

332/// Next/previous word stop (Ctrl+Right/Left on Windows, Option+arrows on the Mac). At the end of
333/// a paragraph Ctrl+Right goes to the start of the next one; at its start Ctrl+Left goes to the
334/// end of the previous one.
335pub fn word_move(body: &TextBody, pos: Pos, forward: bool) -> Pos {
336    let (pi, ch) = clamp(body, pos);
337    let t = para_text(body, pi);
338    if forward {
339        match words::next_stop(&t, ch) {
340            Some(s) => (pi, s),
341            None if pi + 1 < body.paragraphs.len() => (pi + 1, 0),
342            None => (pi, ch),
343        }
344    } else {
345        match words::prev_stop(&t, ch) {
346            Some(s) => (pi, s),
347            None if pi > 0 => (pi - 1, body.paragraphs.get(pi - 1).map(Paragraph::char_len).unwrap_or(0)),
348            None => (pi, 0),
349        }
350    }
351}

365/// Delete a word from `pos` as Ctrl+Backspace (`forward == false`) or Ctrl+Delete does in
366/// PowerPoint; returns the caret.
367///
368/// Ctrl+Backspace deletes back to the previous word stop (at a paragraph's start it joins the
369/// paragraph to the one before). Ctrl+Delete deletes the rest of the word, or the spaces after a
370/// word and the next word, or up to the next stop (see [`words::delete_forward_end`]), then
371/// tidies the spacing as PowerPoint's smart cut does: spaces left before closing punctuation or
372/// at the end of the paragraph after a word go, and sentence punctuation left against a word
373/// gets a space. At a paragraph's end it joins the next paragraph.
374pub fn delete_word(body: &mut TextBody, pos: Pos, forward: bool) -> Pos {
375    if body.paragraphs.is_empty() {
376        return (0, 0);
377    }
378    let (pi, ch) = clamp(body, pos);
379    let t = para_text(body, pi);
380    if !forward {
381        let to = match words::prev_stop(&t, ch) {
382            Some(s) => (pi, s),
383            None => char_move(body, (pi, ch), false),
384        };
385        return delete(body, to, (pi, ch));
386    }
387    let Some(e) = words::delete_forward_end(&t, ch) else {
388        let to = char_move(body, (pi, ch), true);
389        return delete(body, (pi, ch), to);
390    };
391    let at = delete(body, (pi, ch), (pi, e));
392    smart_space(body, at)
393}

395/// PowerPoint's smart-cut spacing after a word was deleted at `at`; returns the caret.
396fn smart_space(body: &mut TextBody, at: Pos) -> Pos {
397    let (pi, s) = at;
398    let chars: Vec<char> = para_text(body, pi).chars().collect();
399    let after = chars.get(s).copied();
400    let before = s.checked_sub(1).and_then(|i| chars.get(i)).copied();
401    if after.is_none_or(words::is_closing_punct) && before.is_some_and(|c| c == ' ') {
402        let mut w = s;
403        while w > 0 && chars.get(w - 1) == Some(&' ') {
404            w -= 1;
405        }
406        if w > 0 && chars.get(w - 1).is_some_and(|c| c.is_alphanumeric()) {
407            return delete(body, (pi, w), (pi, s));
408        }
409        return at;
410    }
411    if let (Some(b), Some(a)) = (before, after)
412        && words::is_sentence_punct(b)
413        && a.is_alphanumeric()
414    {
415        let digits = s >= 2 && chars.get(s - 2).is_some_and(|c| c.is_numeric()) && a.is_numeric();
416        if !digits {
417            return insert(body, at, " ", None);
418        }
419    }
420    at
421}

423/// The position after the grapheme cluster at `pos`, or the next paragraph's start at a paragraph end.
424pub fn cluster_end(body: &TextBody, pos: Pos) -> Pos {
425    let (pi, ch) = clamp(body, pos);
426    let text = body.paragraphs.get(pi).map(Paragraph::text).unwrap_or_default();
427    // Boundaries come from the whole paragraph, so a caret inside a cluster deletes only to its end.
428    let mut end = 0;
429    for g in text.graphemes(true) {
430        end += g.chars().count();
431        if end > ch {
432            return (pi, end);
433        }
434    }
435    char_move(body, pos, true)
436}

438/// One character left/right across paragraph boundaries.
439pub fn char_move(body: &TextBody, pos: Pos, forward: bool) -> Pos {
440    let (pi, ch) = clamp(body, pos);
441    let n = body.paragraphs.get(pi).map(Paragraph::char_len).unwrap_or(0);
442    if forward {
443        if ch < n {
444            (pi, ch + 1)
445        } else if pi + 1 < body.paragraphs.len() {
446            (pi + 1, 0)
447        } else {
448            (pi, ch)
449        }
450    } else if ch > 0 {
451        (pi, ch - 1)
452    } else if pi > 0 {
453        (pi - 1, body.paragraphs.get(pi - 1).map(Paragraph::char_len).unwrap_or(0))
454    } else {
455        (0, 0)
456    }
457}

459/// The position of the `n`th character of the body's plain text (`\n` between paragraphs).
460fn pos_of_char(body: &TextBody, mut n: usize) -> Pos {
461    for (pi, p) in body.paragraphs.iter().enumerate() {
462        let len = p.char_len();
463        if n <= len {
464            return (pi, n);
465        }
466        n -= len + 1;
467    }
468    end(body)
469}

471/// Make the body's plain text (paragraphs joined by `\n`) read `text`, as if it had been typed in
472/// place: only the span between the longest common prefix and suffix changes, so untouched text
473/// keeps its run and paragraph formatting and new text takes the formatting where it lands. Text
474/// that shares nothing with the old text replaces it as plain text.
475pub fn replace_text(body: &mut TextBody, text: &str) {
476    // Normalise line endings first so the common prefix/suffix cannot split a `\r\n` pair.
477    let text = &text.replace("\r\n", "\n").replace('\r', "\n");
478    let old: Vec<char> = body.text().chars().collect();
479    let new: Vec<char> = text.chars().collect();
480    if old == new {
481        return;
482    }
483    let pre = old.iter().zip(&new).take_while(|(a, b)| a == b).count();
484    let room = old.len().min(new.len()) - pre;
485    let suf = old.iter().rev().zip(new.iter().rev()).take(room).take_while(|(a, b)| a == b).count();
486    if pre == 0 && suf == 0 {
487        *body = TextBody { body: body.body.clone(), list_style: body.list_style.clone(), ..TextBody::from_text(text) };
488        return;
489    }
490    let a = pos_of_char(body, pre);
491    let b = pos_of_char(body, old.len() - suf);
492    let at = delete(body, a, b);
493    let mid: String = new.get(pre..new.len() - suf).unwrap_or(&[]).iter().collect();
494    if !mid.is_empty() {
495        insert(body, at, &mid, None);
496    }
497}

505/// The case Shift+F3 changes `s` to next, as PowerPoint cycles it: lowercase → Capitalize Each
506/// Word → UPPERCASE → lowercase; any other mix goes to UPPERCASE first.
507pub fn next_case(s: &str) -> &'static str {
508    if s == s.to_lowercase() && s.chars().any(char::is_alphabetic) {
509        "title"
510    } else if s == s.to_uppercase() && s.chars().any(char::is_alphabetic) {
511        "lower"
512    } else {
513        "upper"
514    }
515}

517/// Whether the `.` at `i` sits between two ASCII digits (a decimal point or version number).
518fn is_decimal_point(all: &[char], i: usize) -> bool {
519    i > 0 && all[i - 1].is_ascii_digit() && all.get(i + 1).is_some_and(|n| n.is_ascii_digit())

527/// Change the case of consecutive pieces of text (the runs of a paragraph) as one string, so
528/// `title` and `sentence` carry their word/sentence state across piece boundaries. Each output
529/// piece is the changed form of the matching input piece, even where a character changes length.
530pub fn change_case_parts(parts: &[&str], mode: &str) -> Vec<String> {
531    let mut start = true;
532    // All characters of all pieces, so word-final sigma sees neighbours across piece boundaries.
533    let all: Vec<char> = parts.iter().flat_map(|s| s.chars()).collect();
534    let mut base = 0;
535    parts
536        .iter()
537        .map(|s| match mode {
538            "upper" => s.to_uppercase(),
539            "lower" => s.to_lowercase(),
540            "toggle" => s.chars().map(|c| if c.is_uppercase() { c.to_lowercase().collect::<String>() } else { c.to_uppercase().collect() }).collect(),
541            _ => {
542                // `title` restarts after whitespace, sentence case after sentence punctuation.
543                let title = mode == "title";
544                let mut out = String::new();
545                for (j, c) in s.chars().enumerate() {
546                    let i = base + j;
547                    if start && c.is_alphabetic() {
548                        out.extend(c.to_uppercase());
549                        start = false;
550                    } else if c == 'Σ' && i > 0 && all[i - 1].is_alphabetic() && !all.get(i + 1).is_some_and(|n| n.is_alphabetic()) {
551                        // Word-final capital sigma lowercases to ς, as `str::to_lowercase` does.
552                        out.push('ς');
553                    } else {
554                        out.extend(c.to_lowercase());
555                    }
556                    if (title && c.is_whitespace()) || (!title && matches!(c, '.' | '!' | '?') && !(c == '.' && is_decimal_point(&all, i))) {
557                        start = true;
558                    }
559                }
560                base += s.chars().count();
561                out
562            }
563        })
564        .collect()
565}

638    /// The paste as it ran before the fast path: one `insert` (and normalize) per source run.
639    fn insert_body_by_runs(body: &mut TextBody, pos: Pos, src: &TextBody) -> Pos {
640        let mut caret = clamp(body, pos);
641        for (k, sp) in src.paragraphs.iter().enumerate() {
642            if k > 0 {
643                caret = insert(body, caret, "\n", None);
644                if let Some(p) = body.paragraphs.get_mut(caret.0) {
645                    p.level = sp.level;
646                    p.props = sp.props.clone();
647                }
648            }
649            for r in &sp.runs {
650                match r.kind {
651                    RunKind::Text => caret = insert(body, caret, &r.text, Some(r.props.clone())),
652                    RunKind::Break => caret = insert(body, caret, "\u{b}", Some(r.props.clone())),
653                    _ => {
654                        if let Some(p) = body.paragraphs.get_mut(caret.0) {
655                            let idx = split_at(p, caret.1);
656                            p.runs.insert(idx, r.clone());
657                            caret.1 += r.char_len();
658                        }
659                    }
660                }
661            }
662        }
663        caret
664    }

905        let mut b = before.clone();
906        replace_text(&mut b, "Overview\nFirst point\nSecond point");
907        assert_eq!(b, before);

913        assert_eq!((b.paragraphs[1].level, b.paragraphs[2].level), (1, 1));
914        // Typing at the end of the heading continues it in bold.

916        assert_eq!(b.paragraphs[0].runs.len(), 1);
917        assert_eq!(b.paragraphs[0].runs[0].props.bold, Some(true));
