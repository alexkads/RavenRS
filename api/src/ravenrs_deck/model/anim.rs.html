4//! the nested timing tree from it and the reader flattens files back into it.

10#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
11#[serde(default, rename_all = "camelCase")]
12pub struct Transition {
13    /// Effect id: `none`, `fade`, `push`, `wipe`, `split`, `reveal`, `cut`, `randomBar`, `shape`,
14    /// `uncover`, `cover`, `flash`, `morph`, `dissolve`, `checker`, `blinds`, `clock`, `zoom`… (see
15    /// [`TRANSITIONS`]).
16    pub kind: String,
17    /// Effect option, e.g. direction `l`, `r`, `u`, `d`, `horz`, `vert`, `in`, `out`.
18    pub option: String,
19    pub duration_ms: u32,
20    pub advance_on_click: bool,
21    pub advance_after_ms: Option<u32>,
22    /// Embedded sound media.
23    pub sound: Option<crate::model::MediaId>,
24    /// Loop Until Next Sound (`p:stSnd loop="1"`).
25    #[serde(skip_serializing_if = "std::ops::Not::not")]
26    pub sound_loop: bool,
27    /// [Stop Previous Sound] (`p:endSnd`): the slide stops a sound still playing.
28    #[serde(skip_serializing_if = "std::ops::Not::not")]
29    pub stop_sound: bool,
30    /// Unmodelled transition XML kept for round-trip.
31    #[serde(skip_serializing_if = "Option::is_none")]
32    pub raw: Option<String>,
33}

104/// The Effect Options label PowerPoint shows for a transition option. Options are stored as the
105/// file format's values: for moving transitions (Push, Wipe, Cover, Pan, Cube…) the direction
106/// things move, so "l" is the effect coming "From Right"; Glitter and the preset effects (Fall
107/// Over, Wind…) name the side itself; Ripple names where the wave heads.
108pub fn transition_option_label(kind: &str, option: &str) -> String {
109    let side = |c: char| match c {
110        'l' => "Left",
111        'r' => "Right",
112        'u' => "Top",
113        _ => "Bottom",
114    };
115    let opposite = |c: char| match c {
116        'l' => 'r',
117        'r' => 'l',
118        'u' => 'd',
119        _ => 'u',
120    };
121    let corner = |o: &str, flip: bool| -> Option<String> {
122        let mut cs = o.chars();
123        let (h, v) = (cs.next()?, cs.next()?);
124        let (h, v) = if flip { (opposite(h), opposite(v)) } else { (h, v) };
125        Some(format!("From {}-{}", side(v), side(h)))
126    };
127    let moving = matches!(
128        kind,
129        "push"
130            | "wipe"
131            | "cover"
132            | "uncover"
133            | "pan"
134            | "vortex"
135            | "cube"
136            | "box"
137            | "rotate"
138            | "orbit"
139            | "gallery"
140            | "conveyor"
141            | "ferris"
142            | "switch"
143            | "flip"
144    );
145    match (kind, option) {
146        (_, "horz") => "Horizontal".into(),
147        (_, "vert") => "Vertical".into(),
148        ("split", o) => format!("{} {}", if o.starts_with("horz") { "Horizontal" } else { "Vertical" }, if o.ends_with("In") { "In" } else { "Out" }),
149        ("fade" | "cut", "throughBlack") => "Through Black".into(),
150        ("fade", _) => "Smoothly".into(),
151        ("cut", _) => "Cut".into(),
152        ("clock", "counterClockwise") => "Counterclockwise".into(),
153        ("clock", "wedge") => "Wedge".into(),
154        ("clock", _) => "Clockwise".into(),
155        ("zoom", "rotate") => "Zoom and Rotate".into(),
156        ("ripple", "center") => "Center".into(),
157        ("ripple", o) => corner(o, true).unwrap_or_else(|| "Center".into()),
158        ("glitter", o) => {
159            let (pat, d) = match o.strip_prefix("hex") {
160                Some(d) => ("Hexagons", d.to_ascii_lowercase()),
161                None => ("Diamonds", o.to_string()),
162            };
163            format!("{pat} From {}", side(d.chars().next().unwrap_or('l')))
164        }
165        ("pageCurlDouble", o) => match o {
166            "singleL" => "Single Left".into(),
167            "singleR" => "Single Right".into(),
168            "r" => "Double Right".into(),
169            _ => "Double Left".into(),
170        },
171        ("shred", o) => {
172            format!("{} {}", if o.starts_with("particles") { "Particles" } else { "Strips" }, if o.ends_with("Out") { "Out" } else { "In" })
173        }
174        ("reveal", o) => format!(
175            "{} {}",
176            if o.starts_with("black") { "Through Black" } else { "Smoothly" },
177            if o.ends_with("Right") { "From Right" } else { "From Left" }
178        ),
179        ("flythrough", o) => {
180            format!("{}{}", if o.starts_with("out") { "Out" } else { "In" }, if o.ends_with("Bounce") { " with Bounce" } else { "" })
181        }
182        ("morph", o) => match o {
183            "words" => "Words".into(),
184            "characters" => "Characters".into(),
185            _ => "Objects".into(),
186        },
187        (_, o) if o.len() == 2 && moving => corner(o, true).unwrap_or_default(),
188        (_, o) if o.len() == 1 && moving => format!("From {}", side(opposite(o.chars().next().unwrap_or('l')))),
189        (_, "l") => "Left".into(),
190        (_, "r") => "Right".into(),
191        (_, "u") => "Top".into(),
192        (_, "d") => "Bottom".into(),
193        (_, "in") => "In".into(),
194        (_, "out") => "Out".into(),
195        (_, o) => o.to_string(),
196    }
197}

210impl AnimClass {
211    pub fn xml(self) -> &'static str {
212        match self {
213            AnimClass::Entrance => "entr",
214            AnimClass::Emphasis => "emph",
215            AnimClass::Exit => "exit",
216            AnimClass::Path => "path",
217            AnimClass::Media => "mediacall",
218        }
219    }
220    pub fn from_xml(s: &str) -> Self {
221        match s {
222            "emph" => AnimClass::Emphasis,
223            "exit" => AnimClass::Exit,
224            "path" => AnimClass::Path,
225            "mediacall" => AnimClass::Media,
226            _ => AnimClass::Entrance,
227        }
228    }
229}

240/// Text build: as one object, all at once by paragraph, or by paragraph level.
241#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
242#[serde(rename_all = "camelCase")]
243pub enum TextBuild {
244    #[default]
245    AsOne,
246    ByParagraph,
247    ByWord,
248    ByLetter,
249}
250
251/// Most paragraphs a by-paragraph build animates separately; later ones follow the last of them.
252pub const MAX_BUILD_PARAGRAPHS: usize = 1_000;
253
254/// What a paragraph does in a by-paragraph build.
255#[derive(Clone, Copy, Debug, PartialEq, Eq)]
256pub enum BuildRole {
257    /// Animated on its own: a filled paragraph at the shallowest filled level.
258    Top,
259    /// Follows the top paragraph with this index (deeper, empty or blank paragraphs after it).
260    Member(usize),
261    /// Before the first top paragraph: not animated.
262    Unowned,
263}
264
265/// Does the paragraph show any text? A by-paragraph build doesn't animate blank ones on their own.
266pub fn build_filled(p: &crate::model::text::Paragraph) -> bool {
267    p.runs.iter().any(|r| !r.text.trim().is_empty())
268}
269
270/// The role of every paragraph of `text` in a by-paragraph build, by paragraph index: what the
271/// timeline plays, and what the PPTX reader and writer map to one file effect per paragraph.
272pub fn paragraph_build_roles(text: &crate::model::text::TextBody) -> Vec<BuildRole> {
273    let top = text.paragraphs.iter().filter(|p| build_filled(p)).map(|p| p.level).min().unwrap_or(0);
274    let mut tops = 0;
275    let mut owner: Option<usize> = None;
276    text.paragraphs
277        .iter()
278        .enumerate()
279        .map(|(i, p)| {
280            if p.level <= top && build_filled(p) && tops < MAX_BUILD_PARAGRAPHS {
281                tops += 1;
282                owner = Some(i);
283                BuildRole::Top
284            } else {
285                owner.map_or(BuildRole::Unowned, BuildRole::Member)
286            }
287        })
288        .collect()
289}

323    pub smooth_start: f64,
324    pub smooth_end: f64,
325    pub bounce_end: f64,
326    pub auto_reverse: bool,
327}

481/// The path of a motion-path preset beyond Lines, Arcs, Turns, Shapes and Loops, in slide
482/// fractions from the shape's position (our path syntax: `M`, `L`, `Z`, ending in `E`).
483/// `None` for an effect that isn't one of these presets.
484pub fn preset_path(effect: &str) -> Option<String> {
485    use std::f64::consts::{PI, TAU};
486    use std::fmt::Write;
487    // Sample `f` over 0..=1 into a polyline starting at (0, 0).
488    fn poly(n: usize, closed: bool, f: impl Fn(f64) -> (f64, f64)) -> String {
489        let (x0, y0) = f(0.0);
490        let mut s = String::from("M 0 0");
491        for k in 1..=n {
492            let (x, y) = f(k as f64 / n as f64);
493            let _ = write!(s, " L {:.4} {:.4}", x - x0, y - y0);
494        }
495        if closed {
496            s.push_str(" Z");
497        }
498        s.push_str(" E");
499        s
500    }
501    fn pts(p: &[(f64, f64)], closed: bool) -> String {
502        let mut s = String::from("M 0 0");
503        for (x, y) in p {
504            let _ = write!(s, " L {x:.4} {y:.4}");
505        }
506        if closed {
507            s.push_str(" Z");
508        }
509        s.push_str(" E");
510        s
511    }
512    // A closed curve in polar form r(θ), starting at its top.
513    fn polar(scale: f64, r: impl Fn(f64) -> f64) -> String {
514        poly(96, true, |t| {
515            let a = -PI / 2.0 + TAU * t;
516            let rr = r(a + PI / 2.0) * scale;
517            (rr * a.cos(), rr * a.sin())
518        })
519    }
520    fn star(n: usize, outer: f64, inner: f64) -> String {
521        let mut p = Vec::new();
522        let start = (0.0, -outer);
523        for k in 1..=2 * n {
524            let a = -PI / 2.0 + PI * k as f64 / n as f64;
525            let r = if k % 2 == 1 { inner } else { outer };
526            p.push((r * a.cos() - start.0, r * a.sin() - start.1));
527        }
528        pts(&p, true)
529    }
530    let spiral = |dir: f64| {
531        poly(120, false, move |t| {
532            let r = 0.12 * (1.0 - t);
533            let a = dir * 3.0 * TAU * t;
534            (dir * (0.12 - r * a.cos()), -r * a.sin())
535        })
536    };
537    let bounce = |dir: f64| poly(90, false, move |t| (dir * 0.3 * t, 0.15 * t - 0.1 * (1.0 - t) * (3.0 * PI * t).sin().abs()));
538    Some(match effect {
539        "diagonalDownRight" => pts(&[(0.2, 0.2)], false),
540        "diagonalUpRight" => pts(&[(0.2, -0.2)], false),
541        "zigzag" => pts(&[(0.05, 0.04), (0.1, 0.0), (0.15, 0.04), (0.2, 0.0), (0.25, 0.04), (0.3, 0.0)], false),
542        "sineWave" => poly(64, false, |t| (0.3 * t, 0.04 * (2.0 * TAU * t).sin())),
543        "pathWave" => poly(48, false, |t| (0.3 * t, 0.06 * (TAU * t).sin())),
544        "decayingWave" => poly(96, false, |t| (0.3 * t, 0.07 * (-3.0 * t).exp() * (3.0 * TAU * t).sin())),
545        "heartbeat" => pts(&[(0.08, 0.0), (0.1, -0.08), (0.13, 0.06), (0.15, 0.0), (0.25, 0.0)], false),
546        "sCurve1" => poly(48, false, |t| (0.2 * t, 0.1 - 0.1 * (PI * t).cos())),
547        "sCurve2" => poly(48, false, |t| (0.1 - 0.1 * (PI * t).cos(), 0.2 * t)),
548        "stairsDown" => pts(&[(0.06, 0.0), (0.06, 0.06), (0.12, 0.06), (0.12, 0.12), (0.18, 0.12), (0.18, 0.18)], false),
549        "bounceRight" => bounce(1.0),
550        "bounceLeft" => bounce(-1.0),
551        "spiralRight" => spiral(1.0),
552        "spiralLeft" => spiral(-1.0),
553        "spring" => poly(160, false, |t| (0.3 * t + 0.03 * (5.0 * TAU * t).cos(), 0.03 * (5.0 * TAU * t).sin())),
554        "funnel" => poly(160, false, |t| {
555            let r = 0.12 * (1.0 - t);
556            (r * (3.0 * TAU * t).cos(), 0.25 * t + 0.03 * (3.0 * TAU * t).sin() * (1.0 - t))
557        }),
558        "swoosh" => poly(48, false, |t| (0.3 * t, -0.12 * (PI * t).sin() + 0.1 * t * t)),
559        "star4" => star(4, 0.1, 0.04),
560        "star5" => star(5, 0.1, 0.04),
561        "star6" => star(6, 0.1, 0.05),
562        "star8" => star(8, 0.1, 0.06),
563        "heart" => poly(96, true, |t| {
564            let a = TAU * t;
565            (0.006 * 16.0 * a.sin().powi(3), -0.006 * (13.0 * a.cos() - 5.0 * (2.0 * a).cos() - 2.0 * (3.0 * a).cos() - (4.0 * a).cos()))
566        }),
567        "crescentMoon" => {
568            // The outer arc of a circle, back along a smaller, offset one.
569            let mut p: Vec<(f64, f64)> = (1..=32)
570                .map(|k| {
571                    let a = -PI / 2.0 - PI * k as f64 / 32.0;
572                    (0.1 * a.cos(), 0.1 + 0.1 * a.sin())
573                })
574                .collect();
575            p.extend((1..=32).map(|k| {
576                let a = PI / 2.0 - PI * k as f64 / 32.0;
577                (0.06 * a.cos() - 0.03, 0.1 + 0.1 * a.sin())
578            }));
579            pts(&p, true)
580        }
581        "teardrop" => polar(0.1, |a| 0.6 + 0.4 * (a / 2.0).cos().abs().powf(3.0)),
582        "football" => poly(64, true, |t| {
583            let a = TAU * t;
584            (0.12 * a.sin(), 0.06 * (1.0 - a.cos()) * (1.0 - 0.3 * (2.0 * a).cos().abs()))
585        }),
586        "plusPath" => pts(
587            &[
588                (0.04, 0.0),
589                (0.04, 0.04),
590                (0.08, 0.04),
591                (0.08, 0.08),
592                (0.04, 0.08),
593                (0.04, 0.12),
594                (0.0, 0.12),
595                (0.0, 0.08),
596                (-0.04, 0.08),
597                (-0.04, 0.04),
598                (0.0, 0.04),
599            ],
600            true,
601        ),
602        "rightTriangle" => pts(&[(0.0, 0.15), (0.15, 0.15)], true),
603        "curvedSquare" => poly(96, true, |t| {
604            let a = -PI / 2.0 + TAU * t;
605            let sg = |v: f64| v.signum() * v.abs().sqrt();
606            (0.1 * sg(a.cos()), 0.1 + 0.1 * sg(a.sin()))
607        }),
608        "figure8Horizontal" => poly(96, true, |t| {
609            let a = TAU * t;
610            (0.15 * a.sin(), 0.06 * (2.0 * a).sin())
611        }),
612        "figure8Vertical" => poly(96, true, |t| {
613            let a = TAU * t;
614            (0.06 * (2.0 * a).sin(), 0.15 * (1.0 - a.cos()))
615        }),
616        "loopDeLoop" => poly(120, false, |t| (0.3 * t - 0.05 * (2.0 * TAU * t).sin(), -0.05 * (1.0 - (2.0 * TAU * t).cos()))),
617        "bean" => polar(0.1, |a| 0.75 + 0.25 * a.cos() + 0.15 * (2.0 * a).cos()),
618        "peanut" => polar(0.1, |a| 0.7 + 0.3 * (2.0 * a).cos().abs()),
619        "neutron" => polar(0.1, |a| 0.35 + 0.65 * (3.0 * a).cos().abs()),
620        _ => return None,
621    })
622}
