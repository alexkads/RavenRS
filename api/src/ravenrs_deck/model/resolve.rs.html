11/// Where a shape lives, for resolving inherited values.
12#[derive(Clone, Copy)]
13pub enum Owner {
14    Slide,
15    Layout,
16    Master,
17}

19pub struct Ctx<'a> {
20    pub pres: &'a Presentation,
21    pub master: &'a Master,
22    pub layout: Option<&'a Layout>,
23    pub owner: Owner,
24    pub scheme: ColorScheme,
25    /// The colour map in force: the slide's override, else its layout's, else the master's.
26    pub color_map: Vec<(String, String)>,
27}
28
29impl<'a> Ctx<'a> {
30    pub fn for_slide(pres: &'a Presentation, slide: &Slide) -> Option<Ctx<'a>> {
31        let (master, layout) = pres.master_for(slide)?;
32        let map = slide.color_map.as_ref().or(layout.and_then(|l| l.color_map.as_ref())).unwrap_or(&master.color_map).clone();
33        Some(Ctx { pres, master, layout, owner: Owner::Slide, scheme: master.scheme_with(&map), color_map: map })
34    }
35    pub fn for_layout(pres: &'a Presentation, master: &'a Master, layout: &'a Layout) -> Ctx<'a> {
36        let map = layout.color_map.as_ref().unwrap_or(&master.color_map).clone();
37        Ctx { pres, master, layout: Some(layout), owner: Owner::Layout, scheme: master.scheme_with(&map), color_map: map }
38    }
39    pub fn for_master(pres: &'a Presentation, master: &'a Master) -> Ctx<'a> {
40        Ctx { pres, master, layout: None, owner: Owner::Master, scheme: master.scheme(), color_map: master.color_map.clone() }
41    }
42    pub fn color(&self, c: &ColorRef, ph: Option<Rgba>) -> Rgba {
43        let c = match &c.base {
44            crate::model::ColorBase::Scheme { slot } if matches!(slot, SchemeSlot::Bg1 | SchemeSlot::Tx1 | SchemeSlot::Bg2 | SchemeSlot::Tx2) => {
45                ColorRef { base: crate::model::ColorBase::Scheme { slot: crate::model::map_slot(&self.color_map, *slot) }, mods: c.mods.clone() }
46            }
47            _ => c.clone(),
48        };
49        c.resolve(&self.master.theme.colors, ph)
50    }
51}

53fn compatible(slide: PhType, other: PhType) -> bool {
54    use PhType::*;
55    match slide {
56        Title | CtrTitle => matches!(other, Title | CtrTitle),
57        SubTitle | Body | Obj | Chart | Table | ClipArt | Diagram | Media | Picture => {
58            !matches!(other, Title | CtrTitle | Date | Footer | SlideNum | Header)
59        }
60        k => k == other,
61    }
62}

64fn find_ph<'s>(shapes: &'s [Shape], shape: &Shape, by_idx: bool) -> Option<&'s Shape> {
65    let ph = shape.ph.as_ref()?;
66    if by_idx
67        && ph.idx != 0
68        && let Some(s) = shapes.iter().find(|s| s.ph.as_ref().is_some_and(|p| p.idx == ph.idx && compatible(ph.kind, p.kind)))
69    {
70        return Some(s);
71    }
72    // Same type first, then a compatible one.
73    shapes
74        .iter()
75        .find(|s| s.ph.as_ref().is_some_and(|p| p.kind == ph.kind && (!by_idx || ph.idx == 0 || p.idx == ph.idx || ph.kind.is_title())))
76        .or_else(|| shapes.iter().find(|s| s.ph.as_ref().is_some_and(|p| p.kind == ph.kind)))
77        .or_else(|| shapes.iter().find(|s| s.ph.as_ref().is_some_and(|p| compatible(ph.kind, p.kind))))
78}

80fn master_ph(master: &Master, kind: PhType) -> Option<&Shape> {
81    let want = match kind {
82        PhType::Title | PhType::CtrTitle => PhType::Title,
83        PhType::Date | PhType::Footer | PhType::SlideNum | PhType::Header => kind,
84        _ => PhType::Body,
85    };
86    master.shapes.iter().find(|s| s.ph_type() == Some(want))
87}

89/// The layout and master placeholders a shape inherits from.
90pub fn parents<'a>(ctx: &Ctx<'a>, shape: &Shape) -> (Option<&'a Shape>, Option<&'a Shape>) {
91    let Some(ph) = shape.ph.as_ref() else { return (None, None) };
92    match ctx.owner {
93        Owner::Slide => {
94            let lp = ctx.layout.and_then(|l| find_ph(&l.shapes, shape, true));
95            let kind = lp.and_then(Shape::ph_type).unwrap_or(ph.kind);
96            (lp, master_ph(ctx.master, kind))
97        }
98        Owner::Layout => (None, master_ph(ctx.master, ph.kind)),
99        Owner::Master => (None, None),
100    }
101}

103/// The text style group (title / body / other) a shape draws from. Text in a table takes the
104/// master's other-text style, also when the table sits in a content placeholder, as in PowerPoint
105/// for Mac (a table cell is 14 pt under an `otherStyle` of 14 pt, a `defaultTextStyle` of 26 pt and
106/// a bulleted 36 pt `bodyStyle`, with or without `<p:ph idx="1"/>`, while a shape's or a text box's
107/// text is 26 pt); other shapes take the presentation's default text style.
108pub fn master_list_style<'a>(ctx: &Ctx<'a>, shape: &Shape) -> &'a ListStyle {
109    if matches!(shape.kind, crate::model::ShapeKind::Table(_)) {
110        return &ctx.master.other_style;
111    }
112    match shape.ph_type() {
113        Some(PhType::Title | PhType::CtrTitle) => &ctx.master.title_style,
114        Some(
115            PhType::Body
116            | PhType::Obj
117            | PhType::SubTitle
118            | PhType::Table
119            | PhType::Chart
120            | PhType::Diagram
121            | PhType::Media
122            | PhType::Picture
123            | PhType::ClipArt,
124        ) => &ctx.master.body_style,
125        Some(_) => &ctx.master.other_style,
126        None => &ctx.pres.default_text_style,
127    }
128}

130pub fn xfrm(ctx: &Ctx, shape: &Shape) -> Xfrm {
131    if let Some(x) = shape.xfrm {
132        return x;
133    }
134    let (lp, mp) = parents(ctx, shape);
135    lp.and_then(|s| s.xfrm).or_else(|| mp.and_then(|s| s.xfrm)).unwrap_or_else(|| {
136        let sz = ctx.pres.slide_size;
137        Xfrm::new(sz.width * 0.1, sz.height * 0.1, sz.width * 0.8, sz.height * 0.2)
138    })
139}

141/// Effective fill and the style colour (`phClr`) to resolve it with.
142pub fn fill(ctx: &Ctx, shape: &Shape) -> (Option<Fill>, Option<Rgba>) {
143    let (lp, mp) = parents(ctx, shape);
144    let style_fill = |s: &Shape| {
145        s.style.as_ref().map(|st| {
146            let (idx, col) = &st.fill_ref;
147            let ph = ctx.color(col, None);
148            let f = match *idx {
149                0 => Some(Fill::None),
150                i if i >= 1001 => ctx.master.theme.format.bg_fills.get((i - 1001) as usize).cloned(),
151                i => ctx.master.theme.format.fills.get((i - 1) as usize).cloned(),
152            };
153            (f, Some(ph))
154        })
155    };
156    for s in [Some(shape), lp, mp].into_iter().flatten() {
157        if let Some(f) = &s.fill {
158            // A gradient without stops (`a:gradFill` with no `a:gsLst`) takes them from the
159            // gradient its style refers to, keeping its own direction, as PowerPoint draws it.
160            if let Fill::Gradient(g) = f
161                && g.stops.is_empty()
162                && let Some((Some(Fill::Gradient(sg)), ph)) = style_fill(s)
163            {
164                let mut g = g.clone();
165                g.stops = sg.stops;
166                return (Some(Fill::Gradient(g)), ph);
167            }
168            return (Some(f.clone()), None);
169        }
170        if let Some(r) = style_fill(s) {
171            return r;
172        }
173    }
174    if shape.text_box {
175        return (Some(Fill::None), None);
176    }
177    (None, None)
178}

180pub fn line(ctx: &Ctx, shape: &Shape) -> (Line, Option<Rgba>) {
181    let (lp, mp) = parents(ctx, shape);
182    let mut out = Line::default();
183    let mut ph = None;
184    for s in [Some(shape), lp, mp].into_iter().flatten() {
185        if let Some(l) = &s.line {
186            out.inherit(l);
187        }
188        if let Some(st) = &s.style {
189            let (idx, col) = &st.line_ref;
190            if ph.is_none() {
191                ph = Some(ctx.color(col, None));
192            }
193            if *idx == 0 {
194                out.inherit(&Line::none());
195            } else if let Some(l) = ctx.master.theme.format.lines.get((*idx - 1) as usize) {
196                out.inherit(l);
197            }
198            break;
199        }
200    }
201    (out, ph)
202}

204pub fn effects(ctx: &Ctx, shape: &Shape) -> (Option<Effects>, Option<Rgba>) {
205    let (lp, mp) = parents(ctx, shape);
206    for s in [Some(shape), lp, mp].into_iter().flatten() {
207        if let Some(e) = &s.effects {
208            return (Some(e.clone()), None);
209        }
210        if let Some(st) = &s.style {
211            let (idx, col) = &st.effect_ref;
212            if *idx == 0 {
213                return (None, None);
214            }
215            return (ctx.master.theme.format.effects.get((*idx - 1) as usize).cloned(), Some(ctx.color(col, None)));
216        }
217    }
218    (None, None)
219}

221/// Turn a placeholder into a regular shape that looks the same: what it inherited from its layout
222/// and master placeholders (position, fill, line, effects, style, body properties and the text
223/// style of every outline level, the master's title or body style included) becomes its own, and
224/// `ph` is cleared. A shape that is not a placeholder is left as it is.
225///
226/// Theme references (`+mj-lt`, scheme colours) stay references, so the copy still follows the
227/// destination theme, as PowerPoint's default paste does.
228pub fn detach_placeholder(ctx: &Ctx, shape: &mut Shape) {
229    if shape.ph.is_none() {
230        return;
231    }
232    // The text first: `level_style` reads the shape's own style, which changes below.
233    let text = shape.text.as_ref().map(|_| (body(ctx, shape), (0..9).map(|l| Some(level_style(ctx, shape, l))).collect::<Vec<_>>()));
234    let own = shape.clone();
235    let (lp, mp) = parents(ctx, &own);
236    let chain = [Some(&own), lp, mp].into_iter().flatten();
237
238    // Each property comes from where `fill`, `line` and `effects` take it: the first shape in the
239    // chain with an explicit value or a style. Lines merge explicit values up to the first styled
240    // shape, whose style line fills the rest.
241    let fill_src = chain.clone().find(|s| s.fill.is_some() || s.style.is_some());
242    let effects_src = chain.clone().find(|s| s.effects.is_some() || s.style.is_some());
243    let mut line = None;
244    let mut line_src = None;
245    for s in chain.clone() {
246        if let Some(l) = &s.line {
247            line.get_or_insert_with(Line::default).inherit(l);
248        }
249        if s.style.is_some() {
250            line_src = Some(s);
251            break;
252        }
253    }
254    // Text takes its font reference from the shape's style or its layout placeholder's (a table's
255    // from its own only), never the master's.
256    let table = matches!(own.kind, crate::model::ShapeKind::Table(_));
257    let font_src = own.style.as_ref().or(lp.filter(|_| !table).and_then(|s| s.style.as_ref()));
258
259    // One style for the copy, each reference taken from the shape that supplied that property. A
260    // property with an explicit value ignores its reference, except a gradient without stops,
261    // which takes them from its own shape's style; a neutral reference (index 0) adds nothing.
262    let style_of = |s: Option<&Shape>| s.and_then(|s| s.style.clone());
263    let neutral = (0, ColorRef::scheme(SchemeSlot::Tx1));
264    let style = chain.clone().any(|s| s.style.is_some()).then(|| ShapeStyle {
265        fill_ref: style_of(fill_src).map_or_else(|| neutral.clone(), |st| st.fill_ref),
266        line_ref: style_of(line_src).map_or_else(|| neutral.clone(), |st| st.line_ref),
267        effect_ref: style_of(effects_src).map_or_else(|| neutral.clone(), |st| st.effect_ref),
268        font_ref: font_src.map_or_else(|| ("none".into(), None), |st| st.font_ref.clone()),
269    });
270
271    shape.xfrm = Some(xfrm(ctx, &own));
272    shape.fill = fill_src.and_then(|s| s.fill.clone());
273    shape.effects = effects_src.and_then(|s| s.effects.clone());
274    shape.line = line;
275    shape.style = style;
276    if let (Some(t), Some((body, levels))) = (shape.text.as_mut(), text) {
277        t.body = body;
278        t.list_style = ListStyle { levels };
279    }
280    shape.ph = None;
281}

283/// Body properties with defaults filled in.
284pub fn body(ctx: &Ctx, shape: &Shape) -> BodyProps {
285    let (lp, mp) = parents(ctx, shape);
286    let mut b = shape.text.as_ref().map(|t| t.body.clone()).unwrap_or_default();
287    for s in [lp, mp].into_iter().flatten() {
288        if let Some(t) = &s.text {
289            b.inherit(&t.body);
290        }
291    }
292    b.inherit(&BodyProps {
293        inset_l: Some(7.2),
294        inset_t: Some(3.6),
295        inset_r: Some(7.2),
296        inset_b: Some(3.6),
297        anchor: Some(if shape.ph.is_none() && !shape.text_box && !matches!(shape.kind, crate::model::ShapeKind::Table(_)) {
298            Anchor::Middle
299        } else {
300            Anchor::Top
301        }),
302        anchor_ctr: Some(false),
303        wrap: Some(true),
304        autofit: Some(if shape.text_box { AutoFit::Shape } else { AutoFit::None }),
305        vert: Some(crate::model::text::TextDir::Horizontal),
306        rot: Some(0.0),
307        columns: Some(1),
308        col_spacing: Some(0.0),
309        warp: None,
310        warp_adj: Vec::new(),
311        rtl_col: Some(false),
312        upright: Some(false),
313        space_first_last: Some(false),
314        compat_ln_spc: None,
315        vert_overflow: None,
316        horz_overflow: None,
317        raw3d: None,
318    });
319    b
320}

322/// The merged level style (paragraph + run defaults) for paragraphs at `level` in `shape`.
323pub fn level_style(ctx: &Ctx, shape: &Shape, level: u8) -> LevelStyle {
324    // A table's text inherits nothing from a placeholder the table sits in: PowerPoint draws it in
325    // the other-text style even under list styles on the layout's and master's placeholders.
326    let (lp, mp) = if matches!(shape.kind, crate::model::ShapeKind::Table(_)) { (None, None) } else { parents(ctx, shape) };
327    let mut out = LevelStyle::default();
328    let add = |out: &mut LevelStyle, ls: Option<&LevelStyle>| {
329        if let Some(l) = ls {
330            out.para.inherit(&l.para);
331            out.run.inherit(&l.run);
332        }
333    };
334    if let Some(t) = &shape.text {
335        add(&mut out, t.list_style.level(level));
336    }
337    for s in [lp, mp].into_iter().flatten() {
338        if let Some(t) = &s.text {
339            add(&mut out, t.list_style.level(level));
340        }
341    }
342    // Shape style font reference (theme font + colour) sits above the defaults.
343    if let Some(st) = shape.style.as_ref().or(lp.and_then(|s| s.style.as_ref())) {
344        let font = match st.font_ref.0.as_str() {
345            "major" => Some("+mj-lt".to_string()),
346            "minor" => Some("+mn-lt".to_string()),
347            _ => None,
348        };
349        let r = RunProps { font, fill: st.font_ref.1.clone().map(Fill::solid), ..Default::default() };
350        out.run.inherit(&r);
351    }
352    add(&mut out, master_list_style(ctx, shape).level(level));
353    add(&mut out, ctx.pres.default_text_style.level(level));
354    // Last resort.
355    out.run.inherit(&RunProps { size: Some(18.0), font: Some("+mn-lt".into()), bold: Some(false), italic: Some(false), ..Default::default() });
356    out.para.inherit(&ParaProps { margin_left: Some(0.0), indent: Some(0.0), align: Some(crate::model::text::Align::Left), ..Default::default() });
357    out
358}

360/// Fully resolved run properties for `run` in paragraph `para` (explicit → paragraph style).
361pub fn run(ctx: &Ctx, shape: &Shape, para: &crate::model::text::Paragraph, run: &RunProps) -> RunProps {
362    let mut r = run.clone();
363    // Paragraph `defRPr` fills gaps in the run, and the list style (including the layout
364    // placeholder this shape inherits from) fills whatever is still unset.
365    r.inherit(&para.def_props);
366    let lvl = level_style(ctx, shape, para.level);
367    r.inherit(&lvl.run);
368    // Hyperlinked text is drawn in the theme's hyperlink colour and underlined, as in PowerPoint,
369    // unless the run itself sets a colour (honoured since PowerPoint 2019) or an underline
370    // (`u="none"` removes it). Colours and underlines from list styles don't count.
371    if r.link.is_some() {
372        // A colour or underline on the run, or on the same `defRPr` that carries the link, is
373        // kept. A colour that arrived only from a list style is not.
374        let from_def = run.link.is_none();
375        if run.fill.is_none() && !(from_def && para.def_props.fill.is_some()) {
376            r.fill = Some(Fill::solid(ColorRef::scheme(SchemeSlot::Hlink)));
377        }
378        if run.underline.is_none() && !(from_def && para.def_props.underline.is_some()) {
379            r.underline = Some("sng".into());
380        }
381    }
382    r
383}

385pub fn para(ctx: &Ctx, shape: &Shape, para: &crate::model::text::Paragraph) -> ParaProps {
386    let mut p = para.props.clone();
387    let lvl = level_style(ctx, shape, para.level);
388    p.inherit(&lvl.para);
389    p
390}

392/// Concrete font family for a run (theme references resolved).
393pub fn font_family(ctx: &Ctx, r: &RunProps) -> String {
394    let f = r.font.as_deref().unwrap_or("+mn-lt");
395    let name = ctx.master.theme.font(f);
396    if name.is_empty() { ctx.master.theme.fonts.minor.latin.clone() } else { name }
397}

399/// Text colour of resolved run props.
400pub fn text_color(ctx: &Ctx, r: &RunProps) -> Rgba {
401    match &r.fill {
402        Some(Fill::Solid { color }) => ctx.color(color, None),
403        Some(Fill::Gradient(g)) => g.stops.first().map(|s| ctx.color(&s.color, None)).unwrap_or(Rgba::BLACK),
404        Some(Fill::None) => Rgba::TRANSPARENT,
405        _ => ctx.color(&ColorRef::scheme(SchemeSlot::Tx1), None),
406    }
407}

409/// The background fill (and its style colour) of a slide.
410pub fn background(ctx: &Ctx, slide: Option<&Slide>) -> (Fill, Option<Rgba>) {
411    let bg = slide.and_then(|s| s.background.as_ref()).or(ctx.layout.and_then(|l| l.background.as_ref())).or(ctx.master.background.as_ref());
412    match bg {
413        Some(Background::Fill { fill }) => (fill.clone(), None),
414        Some(Background::Ref { idx, color }) => {
415            let ph = ctx.color(color, None);
416            let f = if *idx >= 1001 {
417                ctx.master.theme.format.bg_fills.get((*idx - 1001) as usize).cloned()
418            } else {
419                ctx.master.theme.format.fills.get(idx.saturating_sub(1) as usize).cloned()
420            };
421            (f.unwrap_or(Fill::solid(ColorRef::rgb(ph))), Some(ph))
422        }
423        None => (Fill::solid(ColorRef::scheme(SchemeSlot::Bg1)), None),
424    }
425}
