1//! The DeckCraft document model.
2//!
3//! A [`Presentation`] has slide masters (each with a theme and layouts) and slides. Each slide
4//! uses a layout; placeholders on a slide inherit position, formatting and text styles from the
5//! matching placeholder on its layout and master ([`resolve`]). Slides and masters sit behind
6//! `Arc`s so an undo snapshot of the whole deck costs one clone of the slide list.
7#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
8#![forbid(unsafe_code)]

268/// True when every sample is the placeholder `0.5` written before a device reported pressure.
269/// Those strokes were drawn at [`InkStroke::width`] and still are.
270pub fn ink_pressure_is_legacy(points: &[(f64, f64, f32)]) -> bool {
271    !points.is_empty() && points.iter().all(|p| (p.2 - 0.5).abs() <= 1.0e-3)
272}
273
274/// Width of one ink sample. `pressure` of `1` is the pen width. A legacy flat stroke
275/// (`legacy`) draws at the pen width. A missing reading draws at the pen width too.
276/// A light touch stays visible (at least 8% of the pen).
277pub fn ink_sample_width(pen: f64, pressure: f32, legacy: bool) -> f64 {
278    let pen = if pen.is_finite() { pen.abs() } else { 0.0 };
279    if legacy {
280        return pen;
281    }
282    let p = if pressure.is_finite() { pressure.clamp(0.08, 1.0) } else { 1.0 };
283    pen * f64::from(p)
284}

493/// Walk shapes depth-first (groups included).
494pub fn walk<'a>(shapes: &'a [Shape], f: &mut dyn FnMut(&'a Shape, usize)) {
495    fn rec<'a>(shapes: &'a [Shape], depth: usize, f: &mut dyn FnMut(&'a Shape, usize)) {
496        if depth > 64 {
497            return;
498        }
499        for s in shapes {
500            f(s, depth);
501            rec(s.children(), depth + 1, f);
502        }
503    }
504    rec(shapes, 0, f);
505}

507pub fn find_shape(shapes: &[Shape], id: ShapeId) -> Option<&Shape> {
508    fn rec(shapes: &[Shape], id: ShapeId, depth: usize) -> Option<&Shape> {
509        if depth > 64 {
510            return None;
511        }
512        for s in shapes {
513            if s.id == id {
514                return Some(s);
515            }
516            if let Some(f) = rec(s.children(), id, depth + 1) {
517                return Some(f);
518            }
519        }
520        None
521    }
522    rec(shapes, id, 0)
523}

525pub fn find_shape_mut(shapes: &mut [Shape], id: ShapeId) -> Option<&mut Shape> {
526    fn rec(shapes: &mut [Shape], id: ShapeId, depth: usize) -> Option<&mut Shape> {
527        if depth > 64 {
528            return None;
529        }
530        for s in shapes.iter_mut() {
531            if s.id == id {
532                return Some(s);
533            }
534            if let ShapeKind::Group { children, .. } = &mut s.kind
535                && let Some(f) = rec(children, id, depth + 1)
536            {
537                return Some(f);
538            }
539        }
540        None
541    }
542    rec(shapes, id, 0)
543}

545/// The list holding `id` among its siblings: the tree's top level, or a group's children.
546pub fn sibling_list_mut(shapes: &mut Vec<Shape>, id: ShapeId) -> Option<&mut Vec<Shape>> {
547    fn rec(shapes: &mut Vec<Shape>, id: ShapeId, depth: usize) -> Option<&mut Vec<Shape>> {
548        if depth > 64 {
549            return None;
550        }
551        if shapes.iter().any(|s| s.id == id) {
552            return Some(shapes);
553        }
554        for s in shapes.iter_mut() {
555            if let ShapeKind::Group { children, .. } = &mut s.kind
556                && let Some(f) = rec(children, id, depth + 1)
557            {
558                return Some(f);
559            }
560        }
561        None
562    }
563    rec(shapes, id, 0)
564}

566/// The group directly holding `id`: `Some(None)` for a top-level shape, `None` when absent.
567pub fn parent_of(shapes: &[Shape], id: ShapeId) -> Option<Option<ShapeId>> {
568    fn rec(shapes: &[Shape], parent: Option<ShapeId>, id: ShapeId, depth: usize) -> Option<Option<ShapeId>> {
569        if depth > 64 {
570            return None;
571        }
572        for s in shapes {
573            if s.id == id {
574                return Some(parent);
575            }
576            if let Some(f) = rec(s.children(), Some(s.id), id, depth + 1) {
577                return Some(f);
578            }
579        }
580        None
581    }
582    rec(shapes, None, id, 0)
583}

798    /// The theme's colour scheme with the colour map applied (`bg1`… resolve to mapped colours).

802    /// The theme's colour scheme with `map` (a layout's or slide's override, or the master's own)
803    /// applied.
804    pub fn scheme_with(&self, map: &[(String, String)]) -> ColorScheme {
805        let mut s = self.theme.colors.clone();
806        if map.is_empty() {
807            return s;
808        }
809        let base = self.theme.colors.clone();
810        for (alias, slot) in [
811            (SchemeSlot::Bg1, SchemeSlot::Lt1),
812            (SchemeSlot::Tx1, SchemeSlot::Dk1),
813            (SchemeSlot::Bg2, SchemeSlot::Lt2),
814            (SchemeSlot::Tx2, SchemeSlot::Dk2),
815        ] {
816            s.set(slot, base.get(map_slot(map, alias)));
817        }
818        s
819    }
820}

822/// Resolve `bg1`/`tx1`/`bg2`/`tx2` through the colour map `map` (the default mapping when it has
823/// no entry).
824pub fn map_slot(map: &[(String, String)], slot: SchemeSlot) -> SchemeSlot {
825    let key = match slot {
826        SchemeSlot::Bg1 => "bg1",
827        SchemeSlot::Tx1 => "tx1",
828        SchemeSlot::Bg2 => "bg2",
829        SchemeSlot::Tx2 => "tx2",
830        other => return other,
831    };
832    map.iter().find(|(k, _)| k == key).and_then(|(_, v)| SchemeSlot::from_xml(v)).unwrap_or(match slot {
833        SchemeSlot::Bg1 => SchemeSlot::Lt1,
834        SchemeSlot::Tx1 => SchemeSlot::Dk1,
835        SchemeSlot::Bg2 => SchemeSlot::Lt2,
836        _ => SchemeSlot::Dk2,
837    })
838}

950/// A size whose sides may be missing, null or not numbers (NaN then); see `Presentation::slide_size`.
951fn lenient_size<'de, D: serde::Deserializer<'de>>(d: D) -> std::result::Result<Size, D::Error> {
952    let v = serde_json::Value::deserialize(d)?;
953    let side = |k: &str| v.get(k).and_then(serde_json::Value::as_f64).unwrap_or(f64::NAN);
954    Ok(Size::new(side("width"), side("height")))
955}

1032impl Presentation {
1033    /// A fresh id, unique across slides, shapes, layouts, masters and media in this deck. `None`
1034    /// once every id is in use (a hostile file can arrive that way): the deck cannot grow
1035    /// further, and callers must refuse the change rather than duplicate ids (#407).
1036    pub fn alloc_id(&mut self) -> Option<u32> {
1037        let next = self.next_id.max(1).checked_add(1)?;
1038        self.next_id = next;
1039        Some(next)
1040    }
1041    /// Make sure `next_id` is above every id in use (after loading a file).
1042    pub fn fix_next_id(&mut self) {
1043        let mut m = self.next_id;
1044        for s in &self.slides {
1045            m = m.max(s.id.0);
1046            walk(&s.shapes, &mut |sh, _| m = m.max(sh.id.0));
1047        }
1048        for ma in &self.masters {
1049            m = m.max(ma.id.0);
1050            walk(&ma.shapes, &mut |sh, _| m = m.max(sh.id.0));
1051            for l in &ma.layouts {
1052                m = m.max(l.id.0);
1053                walk(&l.shapes, &mut |sh, _| m = m.max(sh.id.0));
1054            }
1055        }
1056        for md in &self.media {
1057            m = m.max(md.id.0);
1058        }
1059        for extra in [self.notes_master.as_deref(), self.handout_master.as_deref()].into_iter().flatten() {
1060            m = m.max(extra.id.0);
1061            walk(&extra.shapes, &mut |sh, _| m = m.max(sh.id.0));
1062        }
1063        self.next_id = m;
1064    }
1065    pub fn slide_index(&self, id: SlideId) -> Option<usize> {
1066        self.slides.iter().position(|s| s.id == id)
1067    }
1068    pub fn slide(&self, id: SlideId) -> Option<&Slide> {
1069        self.slides.iter().find(|s| s.id == id).map(AsRef::as_ref)
1070    }
1071    pub fn slide_mut(&mut self, id: SlideId) -> Option<&mut Slide> {
1072        self.slides.iter_mut().find(|s| s.id == id).map(Arc::make_mut)
1073    }
1074    pub fn slide_at(&self, i: usize) -> Option<&Slide> {
1075        self.slides.get(i).map(AsRef::as_ref)
1076    }
1077    /// Master and layout for a layout id.
1078    pub fn layout(&self, id: LayoutId) -> Option<(&Master, &Layout)> {
1079        self.masters.iter().find_map(|m| m.layout(id).map(|l| (m.as_ref(), l)))
1080    }
1081    /// The master (and layout) a slide uses, falling back to the first.
1082    pub fn master_for(&self, slide: &Slide) -> Option<(&Master, Option<&Layout>)> {
1083        if let Some((m, l)) = self.layout(slide.layout) {
1084            return Some((m, Some(l)));
1085        }
1086        self.masters.first().map(|m| (m.as_ref(), m.layouts.first()))
1087    }
1088    pub fn first_master(&self) -> Option<&Master> {
1089        self.masters.first().map(AsRef::as_ref)
1090    }
1091    pub fn media(&self, id: MediaId) -> Option<&MediaItem> {
1092        self.media.iter().find(|m| m.id == id)
1093    }
1094    /// Add media bytes, reusing an identical existing item.
1095    pub fn add_media(&mut self, name: &str, content_type: &str, data: Vec<u8>) -> Option<MediaId> {
1096        if let Some(m) = self.media.iter().find(|m| m.data.len() == data.len() && *m.data == data) {
1097            return Some(m.id);
1098        }
1099        let id = MediaId(self.alloc_id()?);
1100        self.media.push(MediaItem { id, name: name.into(), content_type: content_type.into(), data: Arc::new(data), link: None });
1101        Some(id)
1102    }
1103    /// Section index containing a slide (by position), when sections exist.
1104    pub fn section_of(&self, slide: SlideId) -> Option<usize> {
1105        self.sections.iter().position(|s| s.slides.contains(&slide))
1106    }
1107    /// 1-based slide number as displayed (respects `first_slide_number`).
1108    pub fn slide_number(&self, index: usize) -> u32 {
1109        self.first_slide_number.saturating_add(u32::try_from(index).unwrap_or(u32::MAX))
1110    }
1111    /// Basic consistency check: unique slide/shape ids, layouts exist. Returns problems found.
1112    pub fn validate(&self) -> Vec<String> {
1113        let mut out = vec![];
1114        let mut seen = std::collections::HashSet::new();
1115        // Indexed once: a per-slide scan of every layout is slides × layouts work.
1116        let layouts: std::collections::HashSet<_> = self.masters.iter().flat_map(|m| m.layouts.iter().map(|l| l.id)).collect();
1117        for s in &self.slides {
1118            if !seen.insert(s.id) {
1119                out.push(format!("duplicate slide id {}", s.id));
1120            }
1121            if !layouts.contains(&s.layout) {
1122                out.push(format!("slide {} uses missing layout {}", s.id, s.layout));
1123            }
1124            let mut ids = std::collections::HashSet::new();
1125            walk(&s.shapes, &mut |sh, _| {
1126                if !ids.insert(sh.id) {
1127                    out.push(format!("slide {}: duplicate shape id {}", s.id, sh.id));
1128                }
1129            });
1130        }
1131        out
1132    }
