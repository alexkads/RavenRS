1//! Decoded picture buffer: reference marking (8.2.5), reference list construction (8.2.4) and output
2//! ordering / bumping (C.4).
3//!
4//! The buffer only does bookkeeping on picture headers; what a picture is (`F`: the software
5//! decoder's `FrameRef`, a hardware decoder's surface) is up to the decoder, so
6//! hardware decoders driven by the same headers (VA-API) share this logic and output order.
7
8use crate::craft::video::h264::decode::error::{Result, ensure};
9use crate::craft::video::h264::decode::slice::{Mmco, SliceHeader, SliceType};
10use std::sync::Arc;

12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub enum RefMark {
14    Unused,
15    Short,
16    Long,
17}

19/// Output-related metadata attached to a decoded frame.
20#[derive(Clone, Debug)]
21pub struct OutputMeta {
22    pub pts: i64,
23    pub key: bool,
24    pub crop: (u32, u32, u32, u32),
25    pub full_range: bool,
26    pub colour_primaries: u8,
27    pub transfer_characteristics: u8,
28    pub matrix_coefficients: u8,
29    pub sar: (u16, u16),
30    /// Decoded without deblocking (draft mode, non-reference picture).
31    pub draft: bool,
32}

34pub struct DpbEntry<F> {
35    pub frame: F,
36    pub poc: i32,
37    pub frame_num: u32,
38    pub long_term_frame_idx: u32,
39    pub mark: RefMark,
40    pub needed_for_output: bool,
41    pub non_existing: bool,
42    pub meta: Arc<OutputMeta>,
43}

45pub struct Dpb<F> {
46    pub entries: Vec<DpbEntry<F>>,
47    /// None = "no long-term frame indices".
48    pub max_long_term_frame_idx: Option<u32>,
49    pub capacity: usize,
50    pub max_reorder: usize,
51}
52
53/// A picture leaving the DPB for output.
54pub struct Output<F> {
55    pub frame: F,
56    pub poc: i32,
57    pub meta: Arc<OutputMeta>,
58}

60impl<F: Clone> Dpb<F> {
61    pub fn new() -> Self {
62        Dpb { entries: Vec::new(), max_long_term_frame_idx: None, capacity: 16, max_reorder: 16 }
63    }
64
65    fn frame_num_wrap(e: &DpbEntry<F>, cur_frame_num: u32, max_frame_num: u32) -> i32 {
66        if e.frame_num > cur_frame_num { e.frame_num as i32 - max_frame_num as i32 } else { e.frame_num as i32 }
67    }
68
69    fn num_refs(&self) -> usize {
70        self.entries.iter().filter(|e| e.mark != RefMark::Unused).count()
71    }
72
73    /// Remove entries not needed for output and not used for reference.
74    fn prune(&mut self) {
75        self.entries.retain(|e| e.needed_for_output || e.mark != RefMark::Unused);
76    }
77
78    /// Output the picture with the smallest POC (C.4.5.3 "bumping"). Returns false if none waits.
79    fn bump(&mut self, out: &mut Vec<Output<F>>) -> bool {
80        let Some(i) = self.entries.iter().enumerate().filter(|(_, e)| e.needed_for_output).min_by_key(|(_, e)| e.poc).map(|(i, _)| i) else {
81            return false;
82        };
83        let e = &mut self.entries[i];
84        e.needed_for_output = false;
85        out.push(Output { frame: e.frame.clone(), poc: e.poc, meta: e.meta.clone() });
86        self.prune();
87        true
88    }
89
90    /// Output everything in POC order and clear non-reference entries.
91    pub fn flush(&mut self, out: &mut Vec<Output<F>>) {
92        while self.bump(out) {}
93        self.prune();
94    }
95
96    /// IDR: mark all references unused; output (or drop) prior pictures.
97    pub fn idr(&mut self, no_output_of_prior_pics: bool, out: &mut Vec<Output<F>>) {
98        for e in &mut self.entries {
99            e.mark = RefMark::Unused;
100        }
101        if no_output_of_prior_pics {
102            self.entries.clear();
103        } else {
104            self.flush(out);
105        }
106        self.max_long_term_frame_idx = None;
107    }
108
109    /// Sliding window marking (8.2.5.3).
110    fn sliding_window(&mut self, cur_frame_num: u32, max_frame_num: u32, max_num_ref_frames: usize) {
111        let limit = max_num_ref_frames.max(1);
112        while self.num_refs() >= limit {
113            let victim = self
114                .entries
115                .iter()
116                .enumerate()
117                .filter(|(_, e)| e.mark == RefMark::Short)
118                .min_by_key(|(_, e)| Self::frame_num_wrap(e, cur_frame_num, max_frame_num))
119                .map(|(i, _)| i);
120            match victim {
121                Some(i) => self.entries[i].mark = RefMark::Unused,
122                None => break,
123            }
124        }
125        self.prune();
126    }

128    /// Insert "non-existing" frames for a frame_num gap (8.2.5.2). `make_frame` gives each one's
129    /// picture and POC.
130    pub fn fill_frame_num_gap(
131        &mut self,
132        prev_ref_frame_num: u32,
133        frame_num: u32,
134        max_frame_num: u32,
135        max_num_ref_frames: usize,
136        make_frame: &mut dyn FnMut(u32) -> (F, i32, Arc<OutputMeta>),
137        poc_state: &mut dyn FnMut(u32),
138    ) {
139        let mut unused = (prev_ref_frame_num + 1) % max_frame_num;
140        // Missing frames before the last `max_num_ref_frames` would be dropped by the sliding window
141        // (oldest FrameNumWrap first) before the gap ends, so only their POC / frame_num bookkeeping
142        // runs; building pictures for them would cost up to max_frame_num frames per access unit.
143        let gap = if frame_num >= unused { frame_num - unused } else { (frame_num % max_frame_num) + max_frame_num - unused };
144        let skip = gap.saturating_sub(max_num_ref_frames.max(1).try_into().unwrap_or(u32::MAX));
145        let mut guard = 0;
146        while unused != frame_num && guard < max_frame_num {
147            if guard < skip {
148                poc_state(unused);
149                unused = (unused + 1) % max_frame_num;
150                guard += 1;
151                continue;
152            }
153            self.sliding_window(unused, max_frame_num, max_num_ref_frames);
154            let (frame, poc, meta) = make_frame(unused);
155            poc_state(unused);
156            self.entries.push(DpbEntry {
157                poc,
158                frame,
159                frame_num: unused,
160                long_term_frame_idx: 0,
161                mark: RefMark::Short,
162                needed_for_output: false,
163                non_existing: true,
164                meta,
165            });
166            unused = (unused + 1) % max_frame_num;
167            guard += 1;
168        }
169    }

171    /// Initial reference picture lists (8.2.4.2) + modification (8.2.4.3) for a frame slice, each
172    /// entry made by `to_ref` from the DPB entry it refers to.
173    pub fn build_ref_lists<R>(&self, sh: &SliceHeader, cur_poc: i32, max_frame_num: u32, to_ref: impl Fn(&DpbEntry<F>) -> R) -> Result<[Vec<R>; 2]> {
174        let mut lists: [Vec<R>; 2] = [Vec::new(), Vec::new()];
175        if sh.slice_type.is_intra() {
176            return Ok(lists);
177        }
178        let cur_fn = sh.frame_num;
179        let pic_num = |e: &DpbEntry<F>| Self::frame_num_wrap(e, cur_fn, max_frame_num);
180        let short: Vec<&DpbEntry<F>> = self.entries.iter().filter(|e| e.mark == RefMark::Short).collect();
181        let mut long: Vec<&DpbEntry<F>> = self.entries.iter().filter(|e| e.mark == RefMark::Long).collect();
182        long.sort_by_key(|e| e.long_term_frame_idx);
183        let mut entry_lists: [Vec<&DpbEntry<F>>; 2] = [Vec::new(), Vec::new()];
184        if sh.slice_type != SliceType::B {
185            let mut s = short.clone();
186            s.sort_by_key(|e| std::cmp::Reverse(pic_num(e)));
187            entry_lists[0] = s.into_iter().chain(long.iter().copied()).collect();
188        } else {
189            let mut before: Vec<&DpbEntry<F>> = short.iter().copied().filter(|e| e.poc < cur_poc).collect();
190            let mut after: Vec<&DpbEntry<F>> = short.iter().copied().filter(|e| e.poc > cur_poc).collect();
191            before.sort_by_key(|e| std::cmp::Reverse(e.poc));
192            after.sort_by_key(|e| e.poc);
193            entry_lists[0] = before.iter().chain(after.iter()).chain(long.iter()).copied().collect();
194            entry_lists[1] = after.iter().chain(before.iter()).chain(long.iter()).copied().collect();
195            if entry_lists[1].len() > 1 && entry_lists[0].len() == entry_lists[1].len() {
196                let same = entry_lists[0].iter().zip(entry_lists[1].iter()).all(|(a, b)| std::ptr::eq(*a, *b));
197                if same {
198                    entry_lists[1].swap(0, 1);
199                }
200            }
201        }
202        let nlists = if sh.slice_type == SliceType::B { 2 } else { 1 };
203        for l in 0..nlists {
204            let n = sh.num_ref_idx_active[l] as usize;
205            let mut list: Vec<Option<&DpbEntry<F>>> = entry_lists[l].iter().copied().map(Some).collect();
206            list.truncate(n);
207            list.resize(n, None);
208            if !sh.ref_pic_list_mod[l].is_empty() {
209                let max_pic_num = max_frame_num as i32;
210                let curr_pic_num = cur_fn as i32;
211                let mut pred = curr_pic_num;
212                let mut ref_idx = 0usize;
213                for m in &sh.ref_pic_list_mod[l] {
214                    ensure!(ref_idx < n, "too many ref_pic_list_modification entries");
215                    list.push(None); // temporarily one longer
216                    let (target, is_long): (Option<&DpbEntry<F>>, bool) = match m.idc {
217                        0 | 1 => {
218                            let d = m.value as i32 + 1;
219                            let no_wrap = if m.idc == 0 {
220                                if pred - d < 0 { pred - d + max_pic_num } else { pred - d }
221                            } else if pred + d >= max_pic_num {
222                                pred + d - max_pic_num
223                            } else {
224                                pred + d
225                            };
226                            pred = no_wrap;
227                            let pn = if no_wrap > curr_pic_num { no_wrap - max_pic_num } else { no_wrap };
228                            (short.iter().copied().find(|e| pic_num(e) == pn), false)
229                        }
230                        2 => (long.iter().copied().find(|e| e.long_term_frame_idx == m.value), true),
231                        _ => continue,
232                    };
233                    let Some(target) = target else {
234                        // Missing picture: keep the list unchanged (error concealment).
235                        list.pop();
236                        ref_idx += 1;
237                        continue;
238                    };
239                    for c in (ref_idx + 1..=n).rev() {
240                        list[c] = list[c - 1];
241                    }
242                    list[ref_idx] = Some(target);
243                    ref_idx += 1;
244                    let mut nidx = ref_idx;
245                    for c in ref_idx..=n {
246                        let keep = match list[c] {
247                            None => true,
248                            Some(e) => {
249                                if is_long {
250                                    !(e.mark == RefMark::Long && e.long_term_frame_idx == target.long_term_frame_idx)
251                                } else {
252                                    !(e.mark == RefMark::Short && std::ptr::eq(e, target))
253                                }
254                            }
255                        };
256                        if keep {
257                            list[nidx] = list[c];
258                            nidx += 1;
259                        }
260                    }
261                    list.truncate(n);
262                }
263            }
264            // Fill missing entries (should not happen in conforming streams) with the first valid entry.
265            let fallback = list.iter().flatten().next().copied().or_else(|| self.entries.last());
266            let out: Vec<R> = list.iter().filter_map(|e| e.or(fallback)).map(&to_ref).collect();
267            lists[l] = out;
268        }
269        Ok(lists)
270    }

272    /// Adaptive memory control (8.2.5.4). Returns true if MMCO 5 was executed.
273    fn apply_mmcos(&mut self, mmcos: &[Mmco], cur_frame_num: u32, max_frame_num: u32, cur_long: &mut Option<u32>) -> Result<bool> {
274        let mut had5 = false;
275        let curr_pic_num = cur_frame_num as i32;
276        for m in mmcos {
277            match m.op {
278                1 => {
279                    let pn = curr_pic_num - (m.difference_of_pic_nums_minus1 as i32 + 1);
280                    for e in &mut self.entries {
281                        if e.mark == RefMark::Short && Self::frame_num_wrap(e, cur_frame_num, max_frame_num) == pn {
282                            e.mark = RefMark::Unused;
283                        }
284                    }
285                }
286                2 => {
287                    for e in &mut self.entries {
288                        if e.mark == RefMark::Long && e.long_term_frame_idx == m.long_term_pic_num {
289                            e.mark = RefMark::Unused;
290                        }
291                    }
292                }
293                3 => {
294                    ensure!(
295                        self.max_long_term_frame_idx.is_some_and(|max| m.long_term_frame_idx <= max),
296                        "MMCO long-term index exceeds the configured limit"
297                    );
298                    let pn = curr_pic_num - (m.difference_of_pic_nums_minus1 as i32 + 1);
299                    for e in &mut self.entries {
300                        if e.mark == RefMark::Long && e.long_term_frame_idx == m.long_term_frame_idx {
301                            e.mark = RefMark::Unused;
302                        }
303                    }
304                    for e in &mut self.entries {
305                        if e.mark == RefMark::Short && Self::frame_num_wrap(e, cur_frame_num, max_frame_num) == pn {
306                            e.mark = RefMark::Long;
307                            e.long_term_frame_idx = m.long_term_frame_idx;
308                        }
309                    }
310                }
311                4 => {
312                    let max = m.max_long_term_frame_idx_plus1;
313                    for e in &mut self.entries {
314                        if e.mark == RefMark::Long && (max == 0 || e.long_term_frame_idx > max - 1) {
315                            e.mark = RefMark::Unused;
316                        }
317                    }
318                    self.max_long_term_frame_idx = if max == 0 { None } else { Some(max - 1) };
319                }
320                5 => {
321                    for e in &mut self.entries {
322                        e.mark = RefMark::Unused;
323                    }
324                    self.max_long_term_frame_idx = None;
325                    had5 = true;
326                }
327                6 => {
328                    ensure!(
329                        self.max_long_term_frame_idx.is_some_and(|max| m.long_term_frame_idx <= max),
330                        "MMCO long-term index exceeds the configured limit"
331                    );
332                    for e in &mut self.entries {
333                        if e.mark == RefMark::Long && e.long_term_frame_idx == m.long_term_frame_idx {
334                            e.mark = RefMark::Unused;
335                        }
336                    }
337                    *cur_long = Some(m.long_term_frame_idx);
338                }
339                _ => {}
340            }
341        }
342        self.prune();
343        Ok(had5)
344    }

346    /// Reference marking for the just-decoded picture and insertion into the DPB with output bumping.
347    /// Returns pictures ready for output.
348    pub fn store_picture(
349        &mut self,
350        sh: &SliceHeader,
351        frame: F,
352        poc: i32,
353        (max_frame_num, max_num_ref_frames): (u32, usize),
354        meta: Arc<OutputMeta>,
355        out: &mut Vec<Output<F>>,
356    ) -> Result<()> {
357        let mut mark = RefMark::Unused;
358        let mut long_idx = 0;
359        let mut had5 = false;
360        if sh.nal_ref_idc != 0 {
361            if sh.idr {
362                if sh.long_term_reference {
363                    mark = RefMark::Long;
364                    self.max_long_term_frame_idx = Some(0);
365                } else {
366                    mark = RefMark::Short;
367                    self.max_long_term_frame_idx = None;
368                }
369            } else {
370                let mut cur_long = None;
371                if sh.adaptive_ref_pic_marking {
372                    had5 = self.apply_mmcos(&sh.mmcos, sh.frame_num, max_frame_num, &mut cur_long)?;
373                } else {
374                    self.sliding_window(sh.frame_num, max_frame_num, max_num_ref_frames);
375                }
376                if let Some(i) = cur_long {
377                    mark = RefMark::Long;
378                    long_idx = i;
379                } else {
380                    mark = RefMark::Short;
381                    // make room if MMCOs left the DPB over-full (non-conforming streams)
382                    if self.num_refs() >= max_num_ref_frames.max(1) && sh.adaptive_ref_pic_marking {
383                        self.sliding_window(sh.frame_num, max_frame_num, max_num_ref_frames);
384                    }
385                }
386            }
387        }
388        if mark == RefMark::Long {
389            ensure!(self.num_refs() < max_num_ref_frames.max(1), "long-term reference exceeds max_num_ref_frames");
390        }
391        if had5 {
392            // All earlier pictures are output before the current one (C.4.4).
393            self.flush(out);
394        }
395        let frame_num = if had5 { 0 } else { sh.frame_num };
396        // Bumping until there is space (C.4.5.1 / C.4.5.2).
397        if mark == RefMark::Unused {
398            loop {
399                let waiting = self.entries.len();
400                if waiting < self.capacity {
401                    break;
402                }
403                // non-reference picture with the smallest POC is output directly
404                let min_poc = self.entries.iter().filter(|e| e.needed_for_output).map(|e| e.poc).min();
405                if min_poc.is_none_or(|m| poc < m) {
406                    out.push(Output { frame, poc, meta });
407                    return Ok(());
408                }
409                if !self.bump(out) {
410                    break;
411                }
412            }
413        } else {
414            while self.entries.len() >= self.capacity {
415                if !self.bump(out) {
416                    break;
417                }
418            }
419        }
420        self.entries.push(DpbEntry {
421            frame,
422            poc,
423            frame_num,
424            long_term_frame_idx: long_idx,
425            mark,
426            needed_for_output: true,
427            non_existing: false,
428            meta,
429        });
430        // Low-delay output: honour max_num_reorder_frames.
431        while self.entries.iter().filter(|e| e.needed_for_output).count() > self.max_reorder {
432            if !self.bump(out) {
433                break;
434            }
435        }
436        Ok(())
437    }

464    /// A huge frame_num jump must only materialise the frames the sliding window keeps (#2779).
465    #[cfg(any())]
466    fn frame_num_gap_materialises_at_most_max_num_ref_frames() {
467        let mut dpb: Dpb<u32> = Dpb::new();
468        let m = meta();
469        let mut made = 0u32;
470        let mut make = |f: u32| {
471            made += 1;
472            (f, 0, m.clone())
473        };
474        let mut updated = Vec::new();
475        let mut upd = |f: u32| updated.push(f);
476        dpb.fill_frame_num_gap(0, 32768, 1 << 16, 4, &mut make, &mut upd);
477        assert!(made <= 4, "gap fill materialised {made} frames, expected <= 4");
478        assert_eq!(updated, (1..32768).collect::<Vec<u32>>());
479        let mut refs: Vec<u32> = dpb.entries.iter().filter(|e| e.mark == RefMark::Short).map(|e| e.frame_num).collect();
480        refs.sort_unstable();
481        assert_eq!(refs, vec![32764, 32765, 32766, 32767]);
482        assert!(dpb.entries.iter().all(|e| e.frame == e.frame_num));
483    }

485    /// Existing short-term references are still evicted by a gap no longer than the window.
486    #[cfg(any())]
487    fn short_frame_num_gap_keeps_sliding_window_semantics() {
488        let mut dpb: Dpb<u32> = Dpb::new();
489        let m = meta();
490        for f in 0..2 {
491            dpb.entries.push(DpbEntry {
492                frame: f,

494                frame_num: f,
495                long_term_frame_idx: 0,
496                mark: RefMark::Short,
497                needed_for_output: false,
498                non_existing: false,
499                meta: m.clone(),

502        let mut make = |f: u32| (f, 0, m.clone());
503        let mut upd = |_f: u32| {};
504        dpb.fill_frame_num_gap(1, 5, 16, 4, &mut make, &mut upd);
