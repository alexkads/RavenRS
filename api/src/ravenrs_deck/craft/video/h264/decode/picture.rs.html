5/// Planar 4:2:0 8-bit sample buffers (MB-aligned dimensions, stride == width).
6#[derive(Clone)]
7pub struct Planes {
8    pub y: Vec<u8>,
9    pub cb: Vec<u8>,
10    pub cr: Vec<u8>,
11    pub width: usize,
12    pub height: usize,
13    pub cwidth: usize,
14}
15
16impl Planes {
17    pub fn new(width: usize, height: usize) -> Self {
18        let (cw, ch) = (width / 2, height / 2);
19        Planes { y: vec![0; width * height], cb: vec![128; cw * ch], cr: vec![128; cw * ch], width, height, cwidth: cw }
20    }
21    pub fn gray(width: usize, height: usize) -> Self {
22        let mut p = Self::new(width, height);
23        p.y.fill(128);
24        p
25    }
26}

28/// Macroblock coding category.
29#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
30pub enum MbKind {
31    #[default]
32    None,
33    I4x4,
34    I8x8,
35    I16x16,
36    IPcm,
37    PSkip,
38    BSkip,
39    BDirect16x16,
40    /// Any other P or B inter macroblock.
41    Inter,
42}

44impl MbKind {
45    #[inline]
46    pub fn is_intra(self) -> bool {
47        matches!(self, MbKind::I4x4 | MbKind::I8x8 | MbKind::I16x16 | MbKind::IPcm)
48    }
49}

51/// Per-macroblock state kept for the whole picture (neighbour derivations, deblocking, co-located data).
52#[derive(Clone, Copy)]
53pub struct MbState {
54    /// Slice number within the picture; `u32::MAX` = not (yet) decoded.
55    pub slice_num: u32,
56    pub kind: MbKind,
57    pub transform_8x8: bool,
58    /// coded_block_pattern: bits 0..3 luma, bits 4..5 chroma.
59    pub cbp: u8,
60    pub qp: u8,
61    /// QPc for Cb / Cr (used by deblocking).
62    pub qpc: [u8; 2],
63    pub intra_chroma_mode: u8,
64    /// Intra4x4PredMode / Intra8x8PredMode per 4x4 block (raster order).
65    pub intra_modes: [u8; 16],
66    /// CAVLC: TotalCoeff per luma 4x4 block (raster). CABAC: coded_block_flag.
67    pub nnz: [u8; 16],
68    /// Same for chroma AC blocks [Cb, Cr][raster 2x2].
69    pub nnz_c: [[u8; 4]; 2],
70    /// CABAC coded_block_flag of DC blocks: bit0 luma (I16x16), bit1 Cb, bit2 Cr.
71    pub cbf_dc: u8,
72    /// Luma 4x4 blocks (raster bit) with non-zero coefficients, for deblocking bS = 2.
73    pub nz_mask: u16,
74    pub ref_idx: [[i8; 4]; 2],
75    /// Motion vectors per 4x4 block (raster order).
76    pub mv: [[[i16; 2]; 16]; 2],
77    /// |mvd| per 4x4 block (raster), saturated, for CABAC context selection.
78    pub mvd: [[[u8; 2]; 16]; 2],
79    /// Bit per 8x8 block predicted in direct mode.
80    pub direct8x8: u8,
81}
82
83impl MbState {
84    #[inline]
85    pub fn kind_is_skip(&self) -> bool {
86        matches!(self.kind, MbKind::PSkip | MbKind::BSkip)
87    }
88}

109        }

113/// One published macroblock row of a decoded frame: final (deblocked) samples plus the motion data
114/// needed as co-located information for direct prediction.
115pub struct FrameRow {
116    /// 16 luma lines.
117    pub y: Box<[u8]>,
118    /// 8 lines per chroma component.
119    pub cb: Box<[u8]>,
120    pub cr: Box<[u8]>,
121    /// Per 4x4 block: index mb_x * 16 + raster.
122    pub mv: [Box<[[i16; 2]]>; 2],
123    /// Per 8x8 block: index mb_x * 4 + b8.
124    pub ref_idx: [Box<[i8]>; 2],
125    /// Unique id of the referenced picture per 8x8 block (u32::MAX = none).
126    pub ref_id: [Box<[u32]>; 2],
127    pub intra: Box<[bool]>,
128}
129
130/// A decoded (or in-progress) frame as stored in the DPB and referenced by later pictures.
131///
132/// Rows are published in order by the thread decoding the picture; readers block (per row) until the
133/// data they need is available, which lets several pictures decode concurrently.
134pub struct Frame {
135    pub id: u32,
136    pub poc: i32,
137    pub width: usize,
138    pub height: usize,
139    pub cwidth: usize,
140    pub cheight: usize,
141    pub mb_w: usize,
142    rows: Box<[OnceLock<FrameRow>]>,
143}
144
145impl Frame {
146    pub fn new(id: u32, poc: i32, mb_w: usize, mb_h: usize) -> Self {
147        Frame {
148            id,
149            poc,
150            width: mb_w * 16,
151            height: mb_h * 16,
152            cwidth: mb_w * 8,
153            cheight: mb_h * 8,
154            mb_w,
155            rows: (0..mb_h).map(|_| OnceLock::new()).collect(),
156        }
157    }
158
159    pub fn mb_h(&self) -> usize {
160        self.rows.len()
161    }
162
163    /// Sample bytes this picture holds (luma and both chroma planes), as the decoder's memory
164    /// budget counts a picture.
165    pub fn bytes(&self) -> u64 {
166        (self.width as u64)
167            .saturating_mul(self.height as u64)
168            .saturating_add((self.cwidth as u64).saturating_mul(self.cheight as u64).saturating_mul(2))
169    }
170
171    /// Publish MB row `r` (ignored if already published).
172    pub fn publish(&self, r: usize, row: FrameRow) {
173        let _ = self.rows[r].set(row);
174    }
175
176    pub fn is_published(&self, r: usize) -> bool {
177        self.rows[r].get().is_some()
178    }
179
180    /// MB row `r`, waiting until it has been published.
181    #[inline]
182    pub fn row(&self, r: usize) -> &FrameRow {
183        {
184            self.rows[r].wait()
185        }
186        #[cfg(any())]
187        {
188            // Rows are decoded in order here, so a row is published before it is read; one that
189            // is not reads as gray with no motion instead of panicking.
190            self.rows[r].get_or_init(|| self.gray_row())
191        }
192    }
193
194    /// A gray intra row with no motion, what "non-existing" frames publish.
195    #[cfg(any())]
196    fn gray_row(&self) -> FrameRow {
197        let per_list = |n: usize| -> [Box<[_]>; 2] { [vec![u32::MAX; n].into(), vec![u32::MAX; n].into()] };
198        FrameRow {
199            y: vec![128; self.width * 16].into(),
200            cb: vec![128; self.cwidth * 8].into(),
201            cr: vec![128; self.cwidth * 8].into(),
202            mv: [vec![[0; 2]; self.mb_w * 16].into(), vec![[0; 2]; self.mb_w * 16].into()],
203            ref_idx: [vec![-1; self.mb_w * 4].into(), vec![-1; self.mb_w * 4].into()],
204            ref_id: per_list(self.mb_w * 4),
205            intra: vec![true; self.mb_w].into(),
206        }
207    }
208
209    /// Wait until the whole frame is available.
210    pub fn wait_complete(&self) {
211        for r in 0..self.rows.len() {
212            self.row(r);
213        }
214    }

216    /// Build a published row from a contiguous plane set (rows of MB row `r`) and motion data.
217    pub fn make_row(planes: &Planes, r: usize, mbs: &[MbState], ref_ids: &dyn Fn(&MbState, usize, i8) -> u32) -> FrameRow {
218        let w = planes.width;
219        let cw = planes.cwidth;
220        let mb_w = w / 16;
221        let y = planes.y[r * 16 * w..(r + 1) * 16 * w].into();
222        let cb = planes.cb[r * 8 * cw..(r + 1) * 8 * cw].into();
223        let cr = planes.cr[r * 8 * cw..(r + 1) * 8 * cw].into();
224        let row_mbs = &mbs[r * mb_w..(r + 1) * mb_w];
225        let mut mv = [vec![[0i16; 2]; mb_w * 16], vec![[0i16; 2]; mb_w * 16]];
226        let mut ref_idx = [vec![-1i8; mb_w * 4], vec![-1i8; mb_w * 4]];
227        let mut ref_id = [vec![u32::MAX; mb_w * 4], vec![u32::MAX; mb_w * 4]];
228        let mut intra = vec![true; mb_w];
229        for (x, st) in row_mbs.iter().enumerate() {
230            if st.slice_num == u32::MAX {
231                continue;
232            }
233            intra[x] = st.kind.is_intra();
234            for l in 0..2 {
235                mv[l][x * 16..x * 16 + 16].copy_from_slice(&st.mv[l]);
236                for b in 0..4 {
237                    let ri = st.ref_idx[l][b];
238                    ref_idx[l][x * 4 + b] = ri;
239                    if ri >= 0 {
240                        ref_id[l][x * 4 + b] = ref_ids(st, l, ri);
241                    }
242                }
243            }
244        }
245        let [mv0, mv1] = mv;
246        let [ri0, ri1] = ref_idx;
247        let [id0, id1] = ref_id;
248        FrameRow { y, cb, cr, mv: [mv0.into(), mv1.into()], ref_idx: [ri0.into(), ri1.into()], ref_id: [id0.into(), id1.into()], intra: intra.into() }
249    }
250
251    /// A frame with every row published from `planes` and no motion (intra), e.g. "non-existing"
252    /// frames inferred for frame_num gaps.
253    pub fn from_planes(id: u32, poc: i32, planes: &Planes) -> Self {
254        let (mb_w, mb_h) = (planes.width / 16, planes.height / 16);
255        let f = Frame::new(id, poc, mb_w, mb_h);
256        let blank = vec![MbState::default(); mb_w * mb_h];
257        for r in 0..mb_h {
258            f.publish(r, Frame::make_row(planes, r, &blank, &|_, _, _| u32::MAX));
259        }
260        f
261    }

263    /// Copy the w x h luma window at (x0, y0) (clamped to the picture) into `out` (stride `os`).
264    #[inline]
265    pub fn luma_window(&self, (x0, y0): (i32, i32), w: usize, h: usize, out: &mut [u8], os: usize) {
266        let (width, height) = (self.width, self.height);
267        for r in 0..h {
268            let yy = (y0 + r as i32).clamp(0, height as i32 - 1) as usize;
269            let row = self.row(yy >> 4);
270            let line = &row.y[(yy & 15) * width..(yy & 15) * width + width];
271            copy_line::<24>(line, x0, &mut out[r * os..], w);
272        }
273    }
274
275    /// Copy the w x h window of chroma component `c` at (x0, y0) (clamped) into `out`.
276    #[inline]
277    pub fn chroma_window(&self, c: usize, (x0, y0): (i32, i32), w: usize, h: usize, out: &mut [u8], os: usize) {
278        let (cw, ch) = (self.cwidth, self.cheight);
279        for r in 0..h {
280            let yy = (y0 + r as i32).clamp(0, ch as i32 - 1) as usize;
281            let row = self.row(yy >> 3);
282            let plane = if c == 0 { &row.cb } else { &row.cr };
283            let line = &plane[(yy & 7) * cw..(yy & 7) * cw + cw];
284            copy_line::<16>(line, x0, &mut out[r * os..], w);
285        }
286    }

288    /// Crop and copy the frame into planar buffers (waits for completion).
289    pub fn copy_cropped(&self, crop: (usize, usize, usize, usize)) -> (Vec<u8>, Vec<u8>, Vec<u8>) {
290        let (cx, cy, cw, ch) = crop;
291        let alloc = plane_allocator();
292        let mut y = alloc(cw * ch);
293        for r in cy..cy + ch {
294            let row = self.row(r >> 4);
295            y.extend_from_slice(&row.y[(r & 15) * self.width + cx..(r & 15) * self.width + cx + cw]);
296        }
297        let (ccx, ccy, ccw, cch) = (cx / 2, cy / 2, cw.div_ceil(2), ch.div_ceil(2));
298        let mut u = alloc(ccw * cch);
299        let mut v = alloc(ccw * cch);
300        for r in ccy..ccy + cch {
301            let row = self.row(r >> 3);
302            let o = (r & 7) * self.cwidth + ccx;
303            u.extend_from_slice(&row.cb[o..o + ccw]);
304            v.extend_from_slice(&row.cr[o..o + ccw]);
305        }
306        (y, u, v)
307    }
308}

310static PLANE_ALLOCATOR: std::sync::OnceLock<fn(usize) -> Vec<u8>> = std::sync::OnceLock::new();

312/// Where output planes come from: an empty buffer with room for the given number of samples.
313/// The host sets it once to recycle the planes of pictures it is done with; the first call wins.
314pub fn set_plane_allocator(alloc: fn(usize) -> Vec<u8>) {
315    let _ = PLANE_ALLOCATOR.set(alloc);
316}
317
318fn plane_allocator() -> fn(usize) -> Vec<u8> {
319    PLANE_ALLOCATOR.get().copied().unwrap_or(Vec::with_capacity)
320}

322/// Copy `w` samples of `line` starting at x0 into `out`; when possible a fixed-size block of `N`
323/// bytes is copied instead (cheaper than a variable-length copy; `out` must then have room for `N`).
324#[inline(always)]
325fn copy_line<const N: usize>(line: &[u8], x0: i32, out: &mut [u8], w: usize) {
326    if w <= N
327        && x0 >= 0
328        && let Some(src) = line.get(x0 as usize..).and_then(|l| l.first_chunk::<N>())
329        && let Some(dst) = out.first_chunk_mut::<N>()
330    {
331        *dst = *src;
332    } else {
333        copy_clamped(line, x0, &mut out[..w]);
334    }
335}
336
337/// Copy `out.len()` samples of `line` starting at x0, replicating edge samples outside the line.
338/// Kept out of line so the fixed-size path of `copy_line` stays an inline move rather than being
339/// merged with this variable-length copy into a libc memcpy call.
340#[cold]
341#[inline(never)]
342fn copy_clamped(line: &[u8], x0: i32, out: &mut [u8]) {
343    let w = out.len();
344    if x0 >= 0 && x0 as usize + w <= line.len() {
345        out.copy_from_slice(&line[x0 as usize..x0 as usize + w]);
346    } else {
347        let max = line.len() as i32 - 1;
348        for (c, o) in out.iter_mut().enumerate() {
349            *o = line[(x0 + c as i32).clamp(0, max) as usize];
350        }
351    }
352}
