6/// LevelScale4x4 / LevelScale8x8 for all six scaling lists and qP % 6, raster order.
7#[derive(Clone)]
8pub struct LevelScale {
9    /// [list 0..6 (IntraY, IntraCb, IntraCr, InterY, InterCb, InterCr)][qp % 6][pos]
10    pub ls4: [[[i32; 16]; 6]; 6],
11    /// [list 0..6 (IntraY, InterY, IntraCb, InterCb, IntraCr, InterCr)][qp % 6][pos]
12    pub ls8: [[[i32; 64]; 6]; 6],
13}
14
15impl LevelScale {
16    pub fn new(m: &ScalingMatrices) -> Box<Self> {
17        let mut s = Box::new(LevelScale { ls4: [[[0; 16]; 6]; 6], ls8: [[[0; 64]; 6]; 6] });
18        for l in 0..6 {
19            for q in 0..6 {
20                for p in 0..16 {
21                    s.ls4[l][q][p] = m.m4[l][p] as i32 * norm_adjust4(q, p) as i32;
22                }
23                for p in 0..64 {
24                    s.ls8[l][q][p] = m.m8[l][p] as i32 * norm_adjust8(q, p) as i32;
25                }
26            }
27        }
28        s
29    }
30}

32/// Coefficient scaling of one block (8.5.12.1 for 4x4 AC, 8.5.13.1 for 8x8) with the qP-dependent
33/// shift and rounding resolved once per block.
34#[derive(Clone, Copy)]
35pub struct Dequant {
36    shl: u32,
37    shr: u32,
38    round: i32,
39}
40
41impl Dequant {
42    /// 4x4 blocks (not the DC of Intra16x16 / chroma blocks).
43    pub fn new4(qp: i32) -> Self {
44        Self::new(qp / 6 - 4)
45    }
46
47    /// 8x8 blocks.
48    pub fn new8(qp: i32) -> Self {
49        Self::new(qp / 6 - 6)
50    }
51
52    fn new(shift: i32) -> Self {
53        if shift >= 0 { Dequant { shl: shift as u32, shr: 0, round: 0 } } else { Dequant { shl: 0, shr: -shift as u32, round: 1 << (-shift - 1) } }
54    }
55
56    #[inline(always)]
57    pub fn apply(self, c: i32, ls: i32) -> i32 {
58        (c.wrapping_mul(ls) << self.shl).wrapping_add(self.round) >> self.shr
59    }
60}

62/// Intra16x16 luma DC: inverse Hadamard + scaling (8.5.10). `c` is raster order (row*4+col);
63/// output replaces `c` with dcY (raster: row i -> 4x4 block row, col j -> block column).
64pub fn luma_dc_dequant(c: &mut [i32; 16], qp: i32, ls00: i32) {
65    let mut t = [0i32; 16];
66    // rows
67    for i in 0..4 {
68        let r = &c[i * 4..i * 4 + 4];
69        let s01 = r[0] + r[1];
70        let d01 = r[0] - r[1];
71        let s23 = r[2] + r[3];
72        let d23 = r[2] - r[3];
73        t[i * 4] = s01 + s23;
74        t[i * 4 + 1] = s01 - s23;
75        t[i * 4 + 2] = d01 - d23;
76        t[i * 4 + 3] = d01 + d23;
77    }
78    // columns
79    for j in 0..4 {
80        let s01 = t[j] + t[4 + j];
81        let d01 = t[j] - t[4 + j];
82        let s23 = t[8 + j] + t[12 + j];
83        let d23 = t[8 + j] - t[12 + j];
84        let f = [s01 + s23, s01 - s23, d01 - d23, d01 + d23];
85        for i in 0..4 {
86            c[i * 4 + j] = if qp >= 36 {
87                f[i].wrapping_mul(ls00) << (qp / 6 - 6)
88            } else {
89                f[i].wrapping_mul(ls00).wrapping_add(1 << (5 - qp / 6)) >> (6 - qp / 6)
90            };
91        }
92    }
93}

95/// 4:2:0 chroma DC: 2x2 transform + scaling (8.5.11). `c` = [c00, c01, c10, c11].
96pub fn chroma_dc_dequant_420(c: &mut [i32; 4], qp: i32, ls00: i32) {
97    let s0 = c[0].wrapping_add(c[1]);
98    let d0 = c[0].wrapping_sub(c[1]);
99    let s1 = c[2].wrapping_add(c[3]);
100    let d1 = c[2].wrapping_sub(c[3]);
101    let f = [s0.wrapping_add(s1), d0.wrapping_add(d1), s0.wrapping_sub(s1), d0.wrapping_sub(d1)];
102    for k in 0..4 {
103        c[k] = (f[k].wrapping_mul(ls00) << (qp / 6)) >> 5;
104    }
105}

107#[inline(always)]
108fn clip_u8(v: i32) -> u8 {
109    v.clamp(0, 255) as u8
110}
111
112/// A dequantised coefficient the inverse transforms are defined for: the dequantisation wraps
113/// by design (see `huge_levels_dequantise_without_overflow_panics`), so their butterflies
114/// clamp an r outside −2^15..2^15−1 to this range — where the spec leaves the output
115/// unspecified anyway (8.5.12.1, 8.5.13.1) — instead of overflowing the plain i32 additions.
116#[inline(always)]
117fn r_clamp(v: i32) -> i32 {
118    v.clamp(-(1 << 15), (1 << 15) - 1)
119}

121/// Inverse 4x4 transform of raster-order `d` and add to `dst` (8.5.12.2, 8.5.14).
122///
123/// Rows first, then columns (the spec's order: the `>> 1` steps make it matter). The column
124/// pass and the reconstruction run across all columns at once so they vectorise.
125pub fn idct4_add(d: &[i32; 16], dst: &mut [u8], stride: usize) {
126    let d: [i32; 16] = std::array::from_fn(|i| r_clamp(d[i]));
127    let mut t = [[0i32; 4]; 4];
128    for (i, row) in t.iter_mut().enumerate() {
129        let r = &d[i * 4..i * 4 + 4];
130        let e = r[0] + r[2];
131        let f = r[0] - r[2];
132        let g = (r[1] >> 1) - r[3];
133        let h = r[1] + (r[3] >> 1);
134        *row = [e + h, f + g, f - g, e - h];
135    }
136    let mut o = [[0i32; 4]; 4];
137    for j in 0..4 {
138        let e = t[0][j] + t[2][j];
139        let f = t[0][j] - t[2][j];
140        let g = (t[1][j] >> 1) - t[3][j];
141        let h = t[1][j] + (t[3][j] >> 1);
142        o[0][j] = e + h;
143        o[1][j] = f + g;
144        o[2][j] = f - g;
145        o[3][j] = e - h;
146    }
147    for (i, row) in o.iter().enumerate() {
148        let Some(line) = dst.get_mut(i * stride..).and_then(|d| d.first_chunk_mut::<4>()) else {
149            return;
150        };
151        for (p, &v) in line.iter_mut().zip(row) {
152            *p = clip_u8(*p as i32 + ((v + 32) >> 6));
153        }
154    }
155}

157/// `idct4_add` / `idct8_add` when all AC coefficients are zero: every output sample is
158/// `(dc + 32) >> 6`, so the transform reduces to adding that to the `N`x`N` block.
159pub fn idct_dc_add<const N: usize>(dc: i32, dst: &mut [u8], stride: usize) {
160    let v = (dc + 32) >> 6;
161    for i in 0..N {
162        let Some(line) = dst.get_mut(i * stride..).and_then(|d| d.first_chunk_mut::<N>()) else {
163            return;
164        };
165        for p in line {
166            *p = clip_u8(*p as i32 + v);
167        }
168    }
169}

171#[inline(always)]
172fn idct8_1d(d: [i32; 8]) -> [i32; 8] {
173    let a0 = d[0] + d[4];
174    let a4 = d[0] - d[4];
175    let a2 = (d[2] >> 1) - d[6];
176    let a6 = d[2] + (d[6] >> 1);
177    let b0 = a0 + a6;
178    let b2 = a4 + a2;
179    let b4 = a4 - a2;
180    let b6 = a0 - a6;
181    let a1 = -d[3] + d[5] - d[7] - (d[7] >> 1);
182    let a3 = d[1] + d[7] - d[3] - (d[3] >> 1);
183    let a5 = -d[1] + d[7] + d[5] + (d[5] >> 1);
184    let a7 = d[3] + d[5] + d[1] + (d[1] >> 1);
185    let b1 = a1 + (a7 >> 2);
186    let b7 = a7 - (a1 >> 2);
187    let b3 = a3 + (a5 >> 2);
188    let b5 = (a3 >> 2) - a5;
189    [b0 + b7, b2 + b5, b4 + b3, b6 + b1, b6 - b1, b4 - b3, b2 - b5, b0 - b7]
190}

192/// Inverse 8x8 transform of raster-order `d` and add to `dst` (8.5.13.2): rows, then columns,
193/// the column pass and reconstruction across all columns at once (they vectorise).
194pub fn idct8_add(d: &[i32; 64], dst: &mut [u8], stride: usize) {
195    let d: [i32; 64] = std::array::from_fn(|i| r_clamp(d[i]));
196    let mut t = [[0i32; 8]; 8];
197    for (row, src) in t.iter_mut().zip(d.as_chunks::<8>().0) {
198        *row = idct8_1d(*src);
199    }
200    let mut o = [[0i32; 8]; 8];
201    for j in 0..8 {
202        let col = idct8_1d(std::array::from_fn(|i| t[i][j]));
203        for i in 0..8 {
204            o[i][j] = col[i];
205        }
206    }
207    for (i, row) in o.iter().enumerate() {
208        let Some(line) = dst.get_mut(i * stride..).and_then(|d| d.first_chunk_mut::<8>()) else {
209            return;
210        };
211        for (p, &v) in line.iter_mut().zip(row) {
212            *p = clip_u8(*p as i32 + ((v + 32) >> 6));
213        }
214    }
215}

221    /// The straightforward per-column formulation (8.5.12.2 / 8.5.13.2) the vectorised
222    /// transforms must equal.
223    fn reference_add(d: &[i32], n: usize, dst: &mut [u8], stride: usize) {
224        let one = |v: &[i32]| -> Vec<i32> {
225            if n == 4 {
226                let (e, f, g, h) = (v[0] + v[2], v[0] - v[2], (v[1] >> 1) - v[3], v[1] + (v[3] >> 1));
227                vec![e + h, f + g, f - g, e - h]
228            } else {
229                idct8_1d(v.try_into().unwrap()).to_vec()
230            }
231        };
232        let rows: Vec<Vec<i32>> = (0..n).map(|i| one(&d[i * n..i * n + n])).collect();
233        for j in 0..n {
234            let col = one(&(0..n).map(|i| rows[i][j]).collect::<Vec<_>>());
235            for i in 0..n {
236                let p = &mut dst[i * stride + j];
237                *p = (*p as i32 + ((col[i] + 32) >> 6)).clamp(0, 255) as u8;
238            }
239        }
240    }

371    /// Corrupt streams can carry huge coefficient levels: dequantisation wraps (as release
372    /// builds always did) instead of panicking with "attempt to multiply with overflow" in
373    /// debug builds (found by mutation fuzzing an H.264 transport stream).
374    /// The same wrapping dequantisation hands the inverse transforms any i32 coefficient, and
375    /// their butterflies add in plain i32: `i32::MAX + i32::MAX` panicked a debug build. The
376    /// spec leaves the transforms' output unspecified when a dequantised value falls outside
377    /// −2^15..2^15−1 (8.5.12.1, 8.5.13.1), so those inputs are clamped to that range —
378    /// bit-exact for every stream inside it — instead of panicking.
379    #[cfg(any())]
380    fn huge_coefficients_are_clamped_to_the_spec_r_range() {
381        assert_eq!(r_clamp(-32769), -32768);
382        assert_eq!(r_clamp(-32768), -32768);
383        assert_eq!(r_clamp(32767), 32767);
384        assert_eq!(r_clamp(32768), 32767);
385        let clamp = |v: i32| v.clamp(-(1 << 15), (1 << 15) - 1);
386        for d in [[i32::MAX; 16], [i32::MIN; 16], std::array::from_fn(|i| if i % 2 == 0 { i32::MIN } else { i32::MAX })] {
387            let clamped = std::array::from_fn(|i| clamp(d[i]));
388            let (mut a, mut b) = ([0u8; 64], [0u8; 64]);
389            idct4_add(&d, &mut a, 8);
390            idct4_add(&clamped, &mut b, 8);
391            assert_eq!(a, b, "idct4 of extremes equals idct4 of the spec range");
392        }
393        for d in [[i32::MIN; 64], [i32::MAX; 64]] {
394            let clamped = std::array::from_fn(|i| clamp(d[i]));
395            let (mut a, mut b) = ([0u8; 64], [0u8; 64]);
396            idct8_add(&d, &mut a, 8);
397            idct8_add(&clamped, &mut b, 8);
398            assert_eq!(a, b, "idct8 of extremes equals idct8 of the spec range");
399        }
400    }
