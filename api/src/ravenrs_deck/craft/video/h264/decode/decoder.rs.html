1//! Top-level decoder: NAL dispatch, picture boundaries, POC, DPB and output.
2//!
3//! The calling thread parses headers and does all DPB / reference list bookkeeping (which only needs
4//! header information). Macroblock decoding of each picture runs as a job — on a thread pool with the
5//! `threads` feature — that publishes finished macroblock rows into the picture's shared [`Frame`];
6//! jobs of later pictures block per row on the reference data they need, so several pictures decode
7//! concurrently (frame-level parallelism).
8

22/// Most pictures one `decode` call may start: a chunk is an access unit, or a few in a transport
23/// stream packet. Each picture is a full frame, and small slices make them cheap to ask for.
24const MAX_PICTURES_PER_CALL: usize = 64;

26/// The decoder supports at most 16 reference pictures, so constructing more

201    /// Forget the stream (parameter sets, `avcC` configuration, reference pictures and pending
202    /// output) as if the decoder were new, e.g. to seek. Unlike making a new decoder, this keeps the
203    /// worker threads, the draft setting and the recycled picture buffers, and the pictures still
204    /// being decoded stop at their next macroblock row instead of running to the end.
205    pub fn reset(&mut self) {
206        self.shared.cancel.store(true, Ordering::Relaxed);
207        let pool = self.shared.pool.lock().map(|mut p| std::mem::take(&mut *p)).unwrap_or_default();
208        self.shared = Arc::new(Shared { pool: Mutex::new(pool), ..Shared::default() });
209        self.spss.fill(None);
210        self.ppss.fill(None);
211        self.nal_length_size = None;
212        self.dpb = Dpb::new();
213        self.poc_state = PocState::default();
214        self.prev_ref_frame_num = 0;
215        self.pending = None;
216        self.active_sps = None;
217        self.ls_cache.clear();
218        self.out_queue.clear();
219        self.queued_ids.clear();
220        self.queued_bytes = 0;
221    }

230    /// Parse an `avcC` record: SPS/PPS and the NAL length size used by [`Decoder::decode`].
231    pub fn configure_avcc(&mut self, avcc: &[u8]) -> Result<()> {
232        ensure!(avcc.len() >= 7, "avcC too short");
233        ensure!(avcc[0] == 1, "unsupported avcC version {}", avcc[0]);
234        let mut pos = 5;
235        let mut sets = Vec::new();
236        let read_sets = |pos: &mut usize, count: usize, sets: &mut Vec<(usize, usize)>| -> Result<()> {
237            for _ in 0..count {
238                ensure!(*pos + 2 <= avcc.len(), "avcC truncated");
239                let len = u16::from_be_bytes([avcc[*pos], avcc[*pos + 1]]) as usize;
240                *pos += 2;
241                ensure!(*pos + len <= avcc.len(), "avcC truncated");
242                sets.push((*pos, len));
243                *pos += len;
244            }
245            Ok(())
246        };
247        let nsps = (avcc[pos] & 0x1f) as usize;
248        pos += 1;
249        read_sets(&mut pos, nsps, &mut sets)?;
250        ensure!(pos < avcc.len(), "avcC truncated");
251        let npps = avcc[pos] as usize;
252        pos += 1;
253        read_sets(&mut pos, npps, &mut sets)?;
254        // Every entry is checked before any is handled: a record that carries anything but an
255        // SPS in its first list and a PPS in its second (a picture slice, say) starts no picture.
256        for (i, (start, len)) in sets.iter().enumerate() {
257            let nal = &avcc[*start..*start + *len];
258            let header = nal.first().copied().ok_or_else(|| Error::Invalid("empty avcC parameter set".into()))?;
259            let expected = if i < nsps { nal_type::SPS } else { nal_type::PPS };
260            ensure!(header & 0x1f == expected, "avcC parameter set has NAL type {}, expected {expected}", header & 0x1f);
261        }
262        for (start, len) in sets {
263            self.handle_nal(&avcc[start..start + len], 0)?;
264        }
265        self.nal_length_size = Some((avcc[4] & 3) as usize + 1);
266        Ok(())
267    }

274    /// Decode one access unit (all NAL units of one picture, Annex-B or length-prefixed per
275    /// configuration; a byte stream with several complete access units is accepted too). Returns
276    /// pictures that became ready for output, in output order; `pts` travels with the picture of
277    /// this access unit through reordering.
278    pub fn decode(&mut self, data: &[u8], pts: i64) -> Result<Vec<Picture>> {
279        let nals = match self.nal_length_size {
280            Some(n) => length_prefixed_nals(data, n)?,
281            None => annexb_nals(data),
282        };
283        self.call_pictures = 0;
284        let mut result = Ok(());
285        for nal in nals {
286            if let Err(e) = self.handle_nal(nal, pts) {
287                result = Err(e);
288                break;
289            }
290        }
291        // The access unit is complete: hand its picture to a decoding job.
292        self.submit_pending()?;
293        if let Some(e) = self.take_error() {
294            result = result.and(Err(e));
295        }
296        result?;
297        // Return finished pictures without blocking on frames still being decoded (bounded latency).
298        let max_queue = self.max_in_flight * 2;
299        let mut pics = Vec::new();
300        while let Some(o) = self.out_queue.front() {
301            if !o.frame.is_published(o.frame.mb_h() - 1) && self.out_queue.len() <= max_queue {
302                break;
303            }
304            let Some(o) = self.pop_out() else { break };
305            pics.push(make_picture(&o));
306        }
307        Ok(pics)
308    }

310    /// Output all remaining pictures (end of stream).
311    pub fn flush(&mut self) -> Vec<Picture> {
312        let _ = self.submit_pending();
313        let mut outs = Vec::new();
314        self.dpb.flush(&mut outs);
315        self.emit(outs);
316        let pics = std::iter::from_fn(|| self.pop_out()).map(|o| make_picture(&o)).collect();
317        while let Some(f) = self.in_flight.pop_front() {
318            f.wait_complete();
319        }
320        pics
321    }

340    fn handle_nal(&mut self, nal: &[u8], pts: i64) -> Result<()> {
341        if nal.is_empty() {
342            return Ok(());
343        }
344        let hdr = NalHeader::parse(nal[0])?;
345        match hdr.nal_unit_type {
346            nal_type::SPS => {
347                let sps = Sps::parse(&unescape_rbsp(&nal[1..]))?;
348                let id = sps.id as usize;
349                self.spss[id] = Some(Arc::new(sps));
350            }
351            nal_type::PPS => {
352                let rbsp = unescape_rbsp(&nal[1..]);
353                let spss: Vec<Option<Sps>> = self.spss.iter().map(|s| s.as_ref().map(|s| (**s).clone())).collect();
354                let pps = Pps::parse(&rbsp, &spss)?;
355                let id = pps.id as usize;
356                self.ppss[id] = Some(Arc::new(pps));
357            }
358            nal_type::SLICE | nal_type::IDR => self.handle_slice(nal, hdr, pts)?,
359            nal_type::SLICE_DPA | nal_type::SLICE_DPB | nal_type::SLICE_DPC => {
360                return unsupported("data partitioning (Extended profile)");
361            }
362            nal_type::END_SEQ | nal_type::END_STREAM => self.submit_pending()?,
363            // SEI, AUD, filler, SPS extension, prefix NAL, subset SPS, auxiliary and extension slices: skipped.
364            _ => {}
365        }
366        Ok(())
367    }

369    fn level_scale(&mut self, pps: &Arc<Pps>) -> Arc<LevelScale> {
370        if let Some((_, ls)) = self.ls_cache.iter().find(|(p, _)| Arc::ptr_eq(p, pps)) {
371            return ls.clone();
372        }
373        let ls: Arc<LevelScale> = Arc::from(LevelScale::new(&pps.scaling));
374        self.ls_cache.retain(|(p, _)| self.ppss.iter().flatten().any(|q| Arc::ptr_eq(p, q)));
375        self.ls_cache.push((pps.clone(), ls.clone()));
376        ls
377    }

379    fn handle_slice(&mut self, nal: &[u8], hdr: NalHeader, pts: i64) -> Result<()> {
380        let rbsp = unescape_rbsp(&nal[1..]);
381        let ppss = &self.ppss;
382        let spss = &self.spss;
383        let mut found: Option<(Arc<Pps>, Arc<Sps>)> = None;
384        let (sh, _, _) = SliceHeader::parse(&rbsp, hdr, |id| {
385            let pps = ppss[id as usize].as_ref().ok_or_else(|| Error::MissingParameterSet(format!("PPS {id}")))?;
386            let sps = spss[pps.sps_id as usize].as_ref().ok_or_else(|| Error::MissingParameterSet(format!("SPS {}", pps.sps_id)))?;
387            // Before the header fields that depend on the SPS: they are validated for the streams
388            // we decode (a 10-bit slice may have a QP below 0), so a stream we cannot decode must
389            // fail as unsupported here, not as invalid there (#626).
390            sps.check_supported()?;
391            found = Some((pps.clone(), sps.clone()));
392            Ok((&**pps, &**sps))
393        })?;
394        let Some((pps, sps)) = found else {
395            return Err(Error::MissingParameterSet("PPS".into()));
396        };
397        if pps.num_slice_groups > 1 {
398            return unsupported("slice groups (FMO)");
399        }
400        if sh.field_pic {
401            return unsupported("field pictures");
402        }
403        if matches!(sh.slice_type, SliceType::Sp | SliceType::Si) {
404            return unsupported("SP/SI slices");
405        }
406        if sh.redundant_pic_cnt > 0 {
407            return Ok(()); // redundant slices are ignored
408        }
409        let new_pic = match &self.pending {
410            None => true,
411            Some(p) => is_new_picture(&p.first, &sh, &sps) || sh.first_mb_in_slice == 0 || !Arc::ptr_eq(&p.sps, &sps),
412        };
413        if new_pic {
414            self.submit_pending()?;
415            if sh.first_mb_in_slice != 0 && self.active_sps.is_none() {
416                return invalid("stream does not start with the first slice of a picture");
417            }
418            self.start_picture(&sh, &sps, pts)?;
419        }
420        let ls = self.level_scale(&pps);
421        let Some(pending) = self.pending.as_mut() else {
422            return invalid("slice without a started picture");
423        };
424        let refs = self.dpb.build_ref_lists(&sh, pending.poc.frame(), sps.max_frame_num(), |e| RefPic {
425            frame: e.frame.clone(),
426            long_term: e.mark == RefMark::Long,
427        })?;
428        if !sh.slice_type.is_intra() {
429            ensure!(!refs[0].is_empty(), "no reference pictures available for inter slice");
430            if sh.slice_type.is_b() {
431                ensure!(!refs[1].is_empty(), "empty RefPicList1 in B slice");
432            }
433        }
434        let weighted = (pps.weighted_pred && sh.slice_type.is_p()) || (pps.weighted_bipred_idc != 0 && sh.slice_type.is_b());
435        let temporal = sh.slice_type.is_b() && !sh.direct_spatial_mv_pred;
436        let (cabac, st, mmcos, lt) = (
437            pps.entropy_coding_mode,
438            sh.slice_type,
439            sh.mmcos.len() as u64,
440            sh.long_term_reference as u64 + sh.mmcos.iter().filter(|m| m.op == 3 || m.op == 6).count() as u64,
441        );
442        pending.slices.push(SliceJob { sh, pps, sps, rbsp, refs, ls });
443        self.stat(|s| {
444            if cabac {
445                s.slices_cabac += 1;
446            } else {
447                s.slices_cavlc += 1;
448            }
449            match st {
450                SliceType::I | SliceType::Si => s.slices_i += 1,
451                SliceType::P | SliceType::Sp => s.slices_p += 1,
452                SliceType::B => s.slices_b += 1,
453            }
454            s.weighted_slices += weighted as u64;
455            s.temporal_direct_slices += temporal as u64;
456            s.mmco_ops += mmcos;
457            s.long_term_marks += lt;
458        });
459        Ok(())
460    }

467    /// Refuse an SPS whose pictures would not fit the memory budget, before anything is allocated.
468    fn check_budget(&self, sps: &Sps) -> Result<()> {
469        let chroma = sps.chroma_array_type();
470        let shape = crate::craft::video::budget::PictureShape {
471            width: u64::from(sps.width()),
472            height: u64::from(sps.height()),
473            shift_x: u32::from(chroma != 3),
474            shift_y: u32::from(chroma == 1),
475            chroma_planes: if chroma == 0 { 0 } else { 2 },
476            bytes_per_sample: 1,
477        };
478        // Pictures already held count at their own size, not the incoming SPS's: after a
479        // resolution change, larger pictures of the old size can still be in the DPB or waiting
480        // for output (one decode call may carry many access units). A buffer shared with a DPB
481        // entry is counted once.
482        let pic = shape.bytes().unwrap_or(u64::MAX);
483        // Queued bytes minus the queue entries that share a DPB frame (each distinct DPB frame once).
484        let mut ids: Vec<(u32, u64)> = self.dpb.entries.iter().map(|e| (e.frame.id, e.frame.bytes())).collect();
485        ids.sort_unstable();
486        ids.dedup();
487        let shared = ids.iter().fold(0u64, |n, (id, b)| n.saturating_add(self.queued_ids.get(id).copied().unwrap_or(0).saturating_mul(*b)));
488        let queued = self.queued_bytes.saturating_sub(shared);
489        let in_dpb = self.dpb.entries.iter().fold(0u64, |n, e| n.saturating_add(e.frame.bytes()));
490        let dpb = sps.max_dpb_frames().max(self.dpb.entries.len()) as u64;
491        // The DPB (the larger of its new slots and what it holds now), the picture being decoded,
492        // and what is in flight or queued.
493        let bytes = pic.saturating_mul(dpb).max(in_dpb).saturating_add(pic).saturating_add(pic.saturating_mul(self.max_in_flight as u64).max(queued));
494        ensure!(bytes <= self.memory_budget, "{}x{} pictures with {bytes} bytes in the decoder exceed the memory budget", sps.width(), sps.height());
495        Ok(())
496    }

498    fn start_picture(&mut self, sh: &SliceHeader, sps: &Arc<Sps>, pts: i64) -> Result<()> {
499        self.check_budget(sps)?;
500        self.call_pictures += 1;
501        if self.call_pictures > MAX_PICTURES_PER_CALL {
502            return Err(Error::Invalid("too many pictures in one chunk".into()));
503        }
504        // Activate SPS; a resolution change flushes the DPB.
505        let changed = match &self.active_sps {
506            None => true,
507            Some(a) => a.width() != sps.width() || a.height() != sps.height() || a.max_dpb_frames() != sps.max_dpb_frames(),
508        };
509        if changed && self.active_sps.is_some() {
510            let mut outs = Vec::new();
511            self.dpb.flush(&mut outs);
512            self.emit(outs);
513            self.dpb.entries.clear();
514        }
515        self.active_sps = Some(sps.clone());
516        self.dpb.capacity = sps.max_dpb_frames();
517        self.dpb.max_reorder = sps.max_num_reorder_frames().min(self.dpb.capacity);
518        let mb_w = sps.pic_width_in_mbs as usize;
519        let mb_h = sps.frame_height_in_mbs() as usize;
520        if sh.idr {
521            let mut outs = Vec::new();
522            self.dpb.idr(sh.no_output_of_prior_pics, &mut outs);
523            self.emit(outs);
524            self.prev_ref_frame_num = 0;
525        } else {
526            let max = sps.max_frame_num();
527            if sh.frame_num != self.prev_ref_frame_num && sh.frame_num != (self.prev_ref_frame_num + 1) % max {
528                let gap = (sh.frame_num + max - self.prev_ref_frame_num - 1) % max;
529                if gap > MAX_INFERRED_GAP_FRAMES {
530                    return Err(Error::Invalid("frame number gap exceeds the supported reference limit".into()));
531                }
532                // frame_num gap (8.2.5.2): insert "non-existing" frames. Their samples are never used by
533                // conforming streams; copy the latest decoded frame for concealment.
534                let meta = Arc::new(output_meta(sps, pts, false));
535                let next_id = &mut self.next_id;
536                let poc_state = &mut self.poc_state;
537                let last = self.dpb.entries.iter().filter(|e| !e.non_existing).max_by_key(|e| e.frame.id).map(|e| e.frame.clone());
538                let mut make = |_fnum: u32| {
539                    let id = *next_id;
540                    *next_id += 1;
541                    let planes = match &last {
542                        Some(f) if f.mb_w == mb_w && f.mb_h() == mb_h => {
543                            let mut p = Planes::new(mb_w * 16, mb_h * 16);
544                            let (y, u, v) = f.copy_cropped((0, 0, mb_w * 16, mb_h * 16));
545                            p.y = y;
546                            p.cb = u;
547                            p.cr = v;
548                            p
549                        }
550                        _ => Planes::gray(mb_w * 16, mb_h * 16),
551                    };
552                    (Arc::new(Frame::from_planes(id, 0, &planes)), 0, meta.clone())
553                };
554                let mut upd = |fnum: u32| poc_state.update_gap_frame(fnum, sps);
555                self.shared.stats.lock().map(|mut s| s.frame_num_gaps += 1).ok();
556                self.dpb.fill_frame_num_gap(self.prev_ref_frame_num, sh.frame_num, max, sps.max_num_ref_frames as usize, &mut make, &mut upd);
557                self.prev_ref_frame_num = (sh.frame_num + max - 1) % max;
558            }
559        }
560        let poc = self.poc_state.compute(sh, sps);
561        // MMCO5: the picture's POC becomes relative to itself (tempPicOrderCnt, 8.2.1).
562        let frame_poc = if sh.has_mmco5() { 0 } else { poc.frame() };
563        let id = self.next_id;
564        self.next_id += 1;
565        let frame = Arc::new(Frame::new(id, frame_poc, mb_w, mb_h));
566        self.pending = Some(PendingPic { frame, sps: sps.clone(), first: sh.clone(), poc, pts, key: sh.idr, slices: Vec::new() });
567        Ok(())
568    }
569
570    /// Start decoding the pending picture and do its reference marking / DPB insertion.
571    fn submit_pending(&mut self) -> Result<()> {
572        let Some(p) = self.pending.take() else { return Ok(()) };
573        let PendingPic { frame, sps, first, poc, pts, key, slices } = p;
574        let draft = self.draft && first.nal_ref_idc == 0;
575        self.dispatch(frame.clone(), slices, draft);
576        self.poc_state.update(&first, &poc);
577        if first.nal_ref_idc != 0 {
578            self.prev_ref_frame_num = if first.has_mmco5() { 0 } else { first.frame_num };
579        }
580        let meta = Arc::new(OutputMeta { draft, ..output_meta(&sps, pts, key) });
581        let mut outs = Vec::new();
582        let fpoc = frame.poc;
583        self.dpb.store_picture(&first, frame, fpoc, (sps.max_frame_num(), sps.max_num_ref_frames as usize), meta, &mut outs)?;
584        self.emit(outs);
585        Ok(())
586    }
587
588    fn dispatch(&mut self, frame: FrameRef, slices: Vec<SliceJob>, draft: bool) {
589        let shared = self.shared.clone();
590        if let Some(pool) = &self.pool {
591            self.in_flight.retain(|f| !f.is_published(f.mb_h() - 1));
592            while self.in_flight.len() >= self.max_in_flight {
593                if let Some(f) = self.in_flight.pop_front() {
594                    f.wait_complete();
595                }
596            }
597            self.in_flight.push_back(frame.clone());
598            pool.spawn_fifo(move || run_job(frame, slices, &shared, draft));
599            return;
600        }
601        run_job(frame, slices, &shared, draft);
602    }
603
604    fn emit(&mut self, outs: Vec<Output<FrameRef>>) {
605        for o in &outs {
606            *self.queued_ids.entry(o.frame.id).or_insert(0) += 1;
607            self.queued_bytes = self.queued_bytes.saturating_add(o.frame.bytes());
608        }
609        self.out_queue.extend(outs);
610    }
611
612    fn pop_out(&mut self) -> Option<Output<FrameRef>> {
613        let o = self.out_queue.pop_front()?;
614        if let Some(n) = self.queued_ids.get_mut(&o.frame.id) {
615            *n = n.saturating_sub(1);
616            if *n == 0 {
617                self.queued_ids.remove(&o.frame.id);
618            }
619        }
620        self.queued_bytes = self.queued_bytes.saturating_sub(o.frame.bytes());
621        Some(o)
622    }
623}
624
625/// Decode all slices of one picture and publish its rows.
626fn run_job(frame: FrameRef, slices: Vec<SliceJob>, shared: &Shared, draft: bool) {
627    // The picture state is set up inside the guard too: a panic there must not escape the job.
628    let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
629        let reuse = shared.pool.lock().ok().and_then(|mut p| p.pop());
630        let mut pic = match reuse {
631            Some(mut p) if p.mb_w == frame.mb_w && p.mb_h == frame.mb_h() => {
632                p.reset(frame.clone());
633                p
634            }
635            _ => PicState::new(frame.clone()),
636        };
637        pic.skip_deblock = draft;
638        pic.cancel = Some(shared.cancel.clone());
639        let mut first_err = None;
640        for s in &slices {
641            if pic.cancelled() {
642                break;
643            }
644            if let Err(e) = decode_slice(s, &mut pic) {
645                first_err.get_or_insert(e);
646            }
647        }
648        if pic.cancelled() {
649            // Nothing will show this picture: publish the remaining rows as they are.
650            pic.skip_deblock = true;
651        }
652        pic.flush_rows(true);
653        (pic, first_err)
654    }));
655    let (pic, err) = match result {
656        Ok((pic, err)) => (Some(pic), err),
657        Err(_) => {
658            // Make sure readers never block on this frame.
659            let fallback = Planes::gray(frame.width, frame.height);
660            let blank = vec![MbState::default(); frame.mb_w * frame.mb_h()];
661            for r in 0..frame.mb_h() {
662                if !frame.is_published(r) {
663                    frame.publish(r, Frame::make_row(&fallback, r, &blank, &|_, _, _| u32::MAX));
664                }
665            }
666            (None, Some(Error::Invalid("internal error while decoding a picture".into())))
667        }
668    };
669    if let Ok(mut s) = shared.stats.lock() {
670        s.pictures += 1;
671        s.draft_pictures += draft as u64;
672        for st in pic.iter().flat_map(|pic| &pic.mbs) {
673            match st.kind {
674                MbKind::I4x4 => s.mb_i4x4 += 1,
675                MbKind::I8x8 => s.mb_i8x8 += 1,
676                MbKind::I16x16 => s.mb_i16x16 += 1,
677                MbKind::IPcm => s.mb_pcm += 1,
678                MbKind::PSkip => s.mb_p_skip += 1,
679                MbKind::BSkip => s.mb_b_skip += 1,
680                MbKind::BDirect16x16 => s.mb_b_direct16x16 += 1,
681                MbKind::Inter => s.mb_inter += 1,
682                MbKind::None => {}
683            }
684            if st.transform_8x8 && !st.kind.is_intra() {
685                s.mb_inter_8x8_transform += 1;
686            }
687        }
688    }
689    if let (Some(e), Ok(mut slot)) = (err, shared.error.lock()) {
690        slot.get_or_insert(e);
691    }
692    if let (Some(pic), Ok(mut p)) = (pic, shared.pool.lock())
693        && p.len() < 4
694    {
695        p.push(pic);
696    }
697}
698
699fn decode_slice(s: &SliceJob, pic: &mut PicState) -> Result<()> {
700    let mut sd = SliceDecoder::new(&s.sh, &s.pps, &s.sps, pic, &s.refs, &s.ls)?;
701    if s.pps.entropy_coding_mode {
702        sd.decode_cabac(&s.rbsp)
703    } else {
704        let mut r = BitReader::new(&s.rbsp);
705        r.seek_bits(s.sh.header_bits);
706        sd.decode_cavlc(&mut r)
707    }
708}

710fn output_meta(sps: &Sps, pts: i64, key: bool) -> OutputMeta {
711    let vui = sps.vui.clone().unwrap_or_default();
712    OutputMeta {
713        pts,
714        key,
715        crop: sps.crop_rect(),
716        full_range: vui.full_range,
717        colour_primaries: vui.colour_primaries,
718        transfer_characteristics: vui.transfer_characteristics,
719        matrix_coefficients: vui.matrix_coefficients,
720        sar: vui.sar,
721        draft: false,
722    }
723}

725fn make_picture(o: &Output<FrameRef>) -> Picture {
726    let (cx, cy, cw, ch) = o.meta.crop;
727    let (cx, cy, cw, ch) = (cx as usize, cy as usize, cw as usize, ch as usize);
728    let (y, u, v) = o.frame.copy_cropped((cx, cy, cw, ch));
729    let (ccw, cch) = (cw.div_ceil(2), ch.div_ceil(2));
730    Picture {
731        width: cw as u32,
732        height: ch as u32,
733        chroma_width: ccw as u32,
734        chroma_height: cch as u32,
735        y,
736        u,
737        v,
738        y_stride: cw,
739        uv_stride: ccw,
740        pts: o.meta.pts,
741        poc: o.poc,
742        key: o.meta.key,
743        color: ColorInfo {
744            full_range: o.meta.full_range,
745            primaries: o.meta.colour_primaries,
746            transfer: o.meta.transfer_characteristics,
747            matrix: o.meta.matrix_coefficients,
748        },
749        sar: o.meta.sar,
750        draft: o.meta.draft,
751    }
