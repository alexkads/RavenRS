11/// Scaling matrices after applying fall-back rules, stored in raster order (weightScale).
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub struct ScalingMatrices {
14    /// [Intra Y, Intra Cb, Intra Cr, Inter Y, Inter Cb, Inter Cr], raster order.
15    pub m4: [[u8; 16]; 6],
16    /// [Intra Y, Inter Y, Intra Cb, Inter Cb, Intra Cr, Inter Cr], raster order.
17    pub m8: [[u8; 64]; 6],
18}
19
20impl ScalingMatrices {
21    pub fn flat() -> Self {
22        Self { m4: [[16; 16]; 6], m8: [[16; 64]; 6] }
23    }
24    pub fn is_flat(&self) -> bool {
25        self.m4.iter().all(|m| m.iter().all(|&v| v == 16)) && self.m8.iter().all(|m| m.iter().all(|&v| v == 16))
26    }
27}

29/// Lists as parsed, in zig-zag order, before inverse scanning.
30#[derive(Clone, Debug)]
31struct ZzLists {
32    l4: [[u8; 16]; 6],
33    l8: [[u8; 64]; 6],
34}
35
36fn zz4_to_raster(l: &[u8; 16]) -> [u8; 16] {
37    let mut out = [0; 16];
38    for (k, &v) in l.iter().enumerate() {
39        out[ZIGZAG4[k] as usize] = v;
40    }
41    out
42}
43fn zz8_to_raster(l: &[u8; 64]) -> [u8; 64] {
44    let mut out = [0; 64];
45    for (k, &v) in l.iter().enumerate() {
46        out[ZIGZAG8[k] as usize] = v;
47    }
48    out
49}
50
51impl ZzLists {
52    fn to_matrices(&self) -> ScalingMatrices {
53        let mut m = ScalingMatrices::flat();
54        for i in 0..6 {
55            m.m4[i] = zz4_to_raster(&self.l4[i]);
56            m.m8[i] = zz8_to_raster(&self.l8[i]);
57        }
58        m
59    }
60    fn flat() -> Self {
61        Self { l4: [[16; 16]; 6], l8: [[16; 64]; 6] }
62    }
63}

65/// Result of parsing one scaling_list(): Some(list) or None meaning "use default".
66fn parse_scaling_list<const N: usize>(r: &mut BitReader) -> Result<Option<[u8; N]>> {
67    let mut list = [0u8; N];
68    let mut last = 8i32;
69    let mut next = 8i32;
70    for (j, item) in list.iter_mut().enumerate() {
71        if next != 0 {
72            let delta = r.read_se()?;
73            ensure!((-128..=127).contains(&delta), "delta_scale out of range");
74            next = (last + delta + 256) % 256;
75            if j == 0 && next == 0 {
76                return Ok(None);
77            }
78        }
79        *item = if next == 0 { last as u8 } else { next as u8 };
80        last = *item as i32;
81    }
82    Ok(Some(list))
83}

85/// Parse scaling lists with a fall-back rule. `fallback` gives the lists used for rule B
86/// (sequence-level) or None for rule A.
87fn parse_scaling_lists(r: &mut BitReader, count: usize, fallback: Option<&ZzLists>) -> Result<ZzLists> {
88    let mut out = ZzLists::flat();
89    for i in 0..count.max(8) {
90        let present = if i < count { r.read_flag()? } else { false };
91        if i < 6 {
92            let parsed = if present { Some(parse_scaling_list::<16>(r)?) } else { None };
93            out.l4[i] = match parsed {
94                Some(Some(l)) => l,
95                Some(None) => {
96                    if i < 3 {
97                        DEFAULT_4X4_INTRA
98                    } else {
99                        DEFAULT_4X4_INTER
100                    }
101                }
102                None => match (i, fallback) {
103                    (0, None) => DEFAULT_4X4_INTRA,
104                    (3, None) => DEFAULT_4X4_INTER,
105                    (0 | 3, Some(f)) => f.l4[i],
106                    _ => out.l4[i - 1],
107                },
108            };
109        } else {
110            let k = i - 6;
111            if k >= 6 {
112                break;
113            }
114            let parsed = if present { Some(parse_scaling_list::<64>(r)?) } else { None };
115            out.l8[k] = match parsed {
116                Some(Some(l)) => l,
117                Some(None) => {
118                    if k % 2 == 0 {
119                        DEFAULT_8X8_INTRA
120                    } else {
121                        DEFAULT_8X8_INTER
122                    }
123                }
124                None => match (k, fallback) {
125                    (0, None) => DEFAULT_8X8_INTRA,
126                    (1, None) => DEFAULT_8X8_INTER,
127                    (0 | 1, Some(f)) => f.l8[k],
128                    _ => out.l8[k - 2],
129                },
130            };
131        }
132    }
133    // 8x8 chroma lists that were never signalled (count == 8) follow rule A/B chains as well.
134    if count <= 8 {
135        for k in 2..6 {
136            out.l8[k] = out.l8[k - 2];
137        }
138    }
139    Ok(out)
140}

143pub struct HrdParameters {
144    pub cpb_cnt: u32,
145    pub bit_rate_scale: u8,
146    pub cpb_size_scale: u8,
147    pub bit_rate_value_minus1: Vec<u32>,
148    pub cpb_size_value_minus1: Vec<u32>,
149    pub cbr_flag: Vec<bool>,
150    pub initial_cpb_removal_delay_length: u8,
151    pub cpb_removal_delay_length: u8,
152    pub dpb_output_delay_length: u8,
153    pub time_offset_length: u8,
154}
155
156fn parse_hrd(r: &mut BitReader) -> Result<HrdParameters> {
157    let cpb_cnt = r.read_ue_plus(1)?;
158    ensure!(cpb_cnt <= 32, "cpb_cnt_minus1 out of range");
159    let mut h = HrdParameters { cpb_cnt, bit_rate_scale: r.read_bits(4)? as u8, cpb_size_scale: r.read_bits(4)? as u8, ..Default::default() };
160    for _ in 0..cpb_cnt {
161        h.bit_rate_value_minus1.push(r.read_ue()?);
162        h.cpb_size_value_minus1.push(r.read_ue()?);
163        h.cbr_flag.push(r.read_flag()?);
164    }
165    h.initial_cpb_removal_delay_length = r.read_bits(5)? as u8 + 1;
166    h.cpb_removal_delay_length = r.read_bits(5)? as u8 + 1;
167    h.dpb_output_delay_length = r.read_bits(5)? as u8 + 1;
168    h.time_offset_length = r.read_bits(5)? as u8;
169    Ok(h)
170}

227    (0, 0),

243    (2, 1),

246fn parse_vui(r: &mut BitReader) -> Result<Vui> {
247    let mut v = Vui::default();
248    if r.read_flag()? {
249        v.aspect_ratio_idc = r.read_bits(8)? as u8;
250        if v.aspect_ratio_idc == 255 {
251            v.sar = (r.read_bits(16)? as u16, r.read_bits(16)? as u16);
252        } else if (v.aspect_ratio_idc as usize) < SAR_TABLE.len() {
253            v.sar = SAR_TABLE[v.aspect_ratio_idc as usize];
254        }
255    }
256    if r.read_flag()? {
257        v.overscan_appropriate = Some(r.read_flag()?);
258    }
259    if r.read_flag()? {
260        v.video_format = r.read_bits(3)? as u8;
261        v.full_range = r.read_flag()?;
262        if r.read_flag()? {
263            v.colour_primaries = r.read_bits(8)? as u8;
264            v.transfer_characteristics = r.read_bits(8)? as u8;
265            v.matrix_coefficients = r.read_bits(8)? as u8;
266        }
267    }
268    if r.read_flag()? {
269        v.chroma_loc = Some((r.read_ue()?, r.read_ue()?));
270    }
271    if r.read_flag()? {
272        let n = r.read_bits(32)?;
273        let t = r.read_bits(32)?;
274        let fixed = r.read_flag()?;
275        v.timing = Some((n, t, fixed));
276    }
277    if r.read_flag()? {
278        v.nal_hrd = Some(parse_hrd(r)?);
279    }
280    if r.read_flag()? {
281        v.vcl_hrd = Some(parse_hrd(r)?);
282    }
283    if v.nal_hrd.is_some() || v.vcl_hrd.is_some() {
284        v.low_delay_hrd = r.read_flag()?;
285    }
286    v.pic_struct_present = r.read_flag()?;
287    // Some streams truncate the VUI after this point; treat EOF as "absent".
288    if r.bits_left() > 0 && r.read_flag()? {
289        let br = (|| -> Result<BitstreamRestriction> {
290            Ok(BitstreamRestriction {
291                motion_vectors_over_pic_boundaries: r.read_flag()?,
292                max_bytes_per_pic_denom: r.read_ue()?,
293                max_bits_per_mb_denom: r.read_ue()?,
294                log2_max_mv_length_horizontal: r.read_ue()?,
295                log2_max_mv_length_vertical: r.read_ue()?,
296                max_num_reorder_frames: r.read_ue()?,
297                max_dec_frame_buffering: r.read_ue()?,
298            })
299        })();
300        v.bitstream_restriction = br.ok();
301    }
302    Ok(v)
303}

339impl Sps {
340    pub fn parse(rbsp: &[u8]) -> Result<Sps> {
341        let mut r = BitReader::new(rbsp);
342        let profile_idc = r.read_bits(8)? as u8;
343        let constraint_flags = r.read_bits(8)? as u8;
344        let level_idc = r.read_bits(8)? as u8;
345        let id = r.read_ue()?;
346        ensure!(id < 32, "seq_parameter_set_id {id} out of range");
347        let mut chroma_format_idc = 1;
348        let mut separate_colour_plane = false;
349        let mut bit_depth_luma = 8;
350        let mut bit_depth_chroma = 8;
351        let mut bypass = false;
352        let mut seq_scaling_matrix_present = false;
353        let mut scaling_lists = ZzLists::flat();
354        if matches!(profile_idc, 100 | 110 | 122 | 244 | 44 | 83 | 86 | 118 | 128 | 138 | 139 | 134 | 135) {
355            chroma_format_idc = r.read_ue()?;
356            ensure!(chroma_format_idc <= 3, "chroma_format_idc out of range");
357            if chroma_format_idc == 3 {
358                separate_colour_plane = r.read_flag()?;
359            }
360            bit_depth_luma = r.read_ue_plus(8)?;
361            bit_depth_chroma = r.read_ue_plus(8)?;
362            ensure!(bit_depth_luma <= 14 && bit_depth_chroma <= 14, "bit depth out of range");
363            bypass = r.read_flag()?;
364            seq_scaling_matrix_present = r.read_flag()?;
365            if seq_scaling_matrix_present {
366                let count = if chroma_format_idc != 3 { 8 } else { 12 };
367                scaling_lists = parse_scaling_lists(&mut r, count, None)?;
368            }
369        }
370        let log2_max_frame_num = r.read_ue_plus(4)?;
371        ensure!(log2_max_frame_num <= 16, "log2_max_frame_num out of range");
372        let pic_order_cnt_type = r.read_ue()?;
373        ensure!(pic_order_cnt_type <= 2, "pic_order_cnt_type out of range");
374        let mut log2_max_poc_lsb = 0;
375        let mut delta_pic_order_always_zero = false;
376        let mut offset_for_non_ref_pic = 0;
377        let mut offset_for_top_to_bottom_field = 0;
378        let mut offset_for_ref_frame = Vec::new();
379        if pic_order_cnt_type == 0 {
380            log2_max_poc_lsb = r.read_ue_plus(4)?;
381            ensure!(log2_max_poc_lsb <= 16, "log2_max_pic_order_cnt_lsb out of range");
382        } else if pic_order_cnt_type == 1 {
383            delta_pic_order_always_zero = r.read_flag()?;
384            offset_for_non_ref_pic = r.read_se()?;
385            offset_for_top_to_bottom_field = r.read_se()?;
386            let n = r.read_ue()?;
387            ensure!(n <= 255, "num_ref_frames_in_pic_order_cnt_cycle out of range");
388            for _ in 0..n {
389                offset_for_ref_frame.push(r.read_se()?);
390            }
391        }
392        let max_num_ref_frames = r.read_ue()?;
393        ensure!(max_num_ref_frames <= 16, "max_num_ref_frames out of range");
394        let gaps_in_frame_num_allowed = r.read_flag()?;
395        let pic_width_in_mbs = r.read_ue_plus(1)?;
396        let pic_height_in_map_units = r.read_ue_plus(1)?;
397        ensure!(pic_width_in_mbs <= 1024 && pic_height_in_map_units <= 1024, "picture too large");
398        let frame_mbs_only = r.read_flag()?;
399        let mb_adaptive_frame_field = if !frame_mbs_only { r.read_flag()? } else { false };
400        let direct_8x8_inference = r.read_flag()?;
401        let frame_crop = if r.read_flag()? { Some((r.read_ue()?, r.read_ue()?, r.read_ue()?, r.read_ue()?)) } else { None };
402        let vui = if r.read_flag()? { Some(parse_vui(&mut r)?) } else { None };
403        let scaling = if seq_scaling_matrix_present { scaling_lists.to_matrices() } else { ScalingMatrices::flat() };
404        let sps = Sps {
405            profile_idc,
406            constraint_flags,
407            level_idc,
408            id,
409            chroma_format_idc,
410            separate_colour_plane,
411            bit_depth_luma,
412            bit_depth_chroma,
413            qpprime_y_zero_transform_bypass: bypass,
414            seq_scaling_matrix_present,
415            scaling_lists,
416            scaling,
417            log2_max_frame_num,
418            pic_order_cnt_type,
419            log2_max_poc_lsb,
420            delta_pic_order_always_zero,
421            offset_for_non_ref_pic,
422            offset_for_top_to_bottom_field,
423            offset_for_ref_frame,
424            max_num_ref_frames,
425            gaps_in_frame_num_allowed,
426            pic_width_in_mbs,
427            pic_height_in_map_units,
428            frame_mbs_only,
429            mb_adaptive_frame_field,
430            direct_8x8_inference,
431            frame_crop,
432            vui,
433        };
434        if let Some((l, rr, t, b)) = sps.frame_crop {
435            let (cx, cy) = sps.crop_unit();
436            let ok = l.checked_add(rr).and_then(|h| h.checked_mul(cx)).is_some_and(|w| w < sps.width())
437                && t.checked_add(b).and_then(|v| v.checked_mul(cy)).is_some_and(|h| h < sps.height());
438            ensure!(ok, "frame cropping exceeds picture size");
439        }
440        Ok(sps)
441    }

443    /// ChromaArrayType.
444    pub fn chroma_array_type(&self) -> u32 {
445        if self.separate_colour_plane { 0 } else { self.chroma_format_idc }
446    }
447    pub fn frame_height_in_mbs(&self) -> u32 {
448        (2 - self.frame_mbs_only as u32) * self.pic_height_in_map_units
449    }
450    /// Decoded (uncropped) luma width.
451    pub fn width(&self) -> u32 {
452        self.pic_width_in_mbs * 16
453    }
454    pub fn height(&self) -> u32 {
455        self.frame_height_in_mbs() * 16
456    }
457    pub fn max_frame_num(&self) -> u32 {
458        1 << self.log2_max_frame_num
459    }
460    /// (CropUnitX, CropUnitY).
461    pub fn crop_unit(&self) -> (u32, u32) {
462        let (sub_w, sub_h) = match self.chroma_array_type() {
463            1 => (2, 2),
464            2 => (2, 1),
465            _ => (1, 1),
466        };
467        if self.chroma_array_type() == 0 { (1, 2 - self.frame_mbs_only as u32) } else { (sub_w, sub_h * (2 - self.frame_mbs_only as u32)) }
468    }
469    /// Cropping rectangle in luma samples: (x, y, width, height).
470    pub fn crop_rect(&self) -> (u32, u32, u32, u32) {
471        match self.frame_crop {
472            None => (0, 0, self.width(), self.height()),
473            Some((l, r, t, b)) => {
474                let (cx, cy) = self.crop_unit();
475                (
476                    l.saturating_mul(cx),
477                    t.saturating_mul(cy),
478                    self.width().saturating_sub(l.saturating_add(r).saturating_mul(cx)).max(1),
479                    self.height().saturating_sub(t.saturating_add(b).saturating_mul(cy)).max(1),
480                )
481            }
482        }
483    }

485    /// MaxDpbFrames from Table A-1 (and the VUI max_dec_frame_buffering when present).
486    pub fn max_dpb_frames(&self) -> usize {
487        let max_dpb_mbs: u32 = match self.level_idc {
488            9 => 396,
489            10 => 396,
490            11 => {
491                if self.constraint_flags & 0x10 != 0 && !matches!(self.profile_idc, 100 | 110 | 122 | 244 | 44) {
492                    396
493                } else {
494                    900
495                }
496            }
497            12 | 13 | 20 => 2376,
498            21 => 4752,
499            22 | 30 => 8100,
500            31 => 18000,
501            32 => 20480,
502            40 | 41 => 32768,
503            42 => 34816,
504            50 => 110400,
505            51 | 52 => 184320,
506            60..=62 => 696320,
507            _ => 184320,
508        };
509        let frame_mbs = self.pic_width_in_mbs * self.frame_height_in_mbs();
510        let mut n = (max_dpb_mbs / frame_mbs.max(1)).clamp(1, 16) as usize;
511        if let Some(br) = self.vui.as_ref().and_then(|v| v.bitstream_restriction.as_ref()) {
512            n = (br.max_dec_frame_buffering as usize).min(16);
513        }
514        n.max(self.max_num_ref_frames as usize).max(1)
515    }

517    /// Number of frames that may precede a frame in decoding order and follow it in output order.
518    pub fn max_num_reorder_frames(&self) -> usize {
519        if let Some(br) = self.vui.as_ref().and_then(|v| v.bitstream_restriction.as_ref()) {
520            return br.max_num_reorder_frames as usize;
521        }
522        // Profiles that cannot reorder (no B slices) with constraint_set3 (intra) etc.
523        if matches!(self.profile_idc, 44 | 86 | 100 | 110 | 122 | 244) && self.constraint_flags & 0x10 != 0 {
524            return 0;
525        }
526        self.max_dpb_frames()
527    }

529    pub fn check_supported(&self) -> Result<()> {
530        if u64::from(self.width()).saturating_mul(u64::from(self.height())) > MAX_LUMA_SAMPLES {
531            return unsupported("picture dimensions exceed the decoder memory limit");
532        }
533        if !self.frame_mbs_only {
534            return unsupported("interlaced (field / MBAFF) coding");
535        }
536        if self.chroma_array_type() != 1 {
537            return unsupported(format!("chroma_format_idc {} (only 4:2:0 is implemented)", self.chroma_format_idc));
538        }
539        if self.bit_depth_luma != 8 || self.bit_depth_chroma != 8 {
540            return unsupported(format!("bit depth luma {} chroma {} (only 8-bit is implemented)", self.bit_depth_luma, self.bit_depth_chroma));
541        }
542        if self.qpprime_y_zero_transform_bypass {
543            return unsupported("qpprime_y_zero_transform_bypass (lossless)");
544        }
545        Ok(())
546    }
547}

549/// Picture parameter set.
550#[derive(Clone, Debug)]
551pub struct Pps {
552    pub id: u32,
553    pub sps_id: u32,
554    pub entropy_coding_mode: bool,
555    pub bottom_field_pic_order_in_frame_present: bool,
556    pub num_slice_groups: u32,
557    pub slice_group_map_type: u32,
558    pub num_ref_idx_l0_default_active: u32,
559    pub num_ref_idx_l1_default_active: u32,
560    pub weighted_pred: bool,
561    pub weighted_bipred_idc: u32,
562    pub pic_init_qp: i32,
563    pub pic_init_qs: i32,
564    pub chroma_qp_index_offset: i32,
565    pub deblocking_filter_control_present: bool,
566    pub constrained_intra_pred: bool,
567    pub redundant_pic_cnt_present: bool,
568    pub transform_8x8_mode: bool,
569    pub pic_scaling_matrix_present: bool,
570    pub second_chroma_qp_index_offset: i32,
571    /// Effective scaling matrices (after PPS/SPS fall-back), raster order.
572    pub scaling: ScalingMatrices,
573}
574
575impl Pps {
576    /// Parse a PPS. Needs the SPS table because the tail depends on chroma_format_idc.
577    pub fn parse(rbsp: &[u8], spss: &[Option<Sps>]) -> Result<Pps> {
578        let mut r = BitReader::new(rbsp);
579        let id = r.read_ue()?;
580        ensure!(id < 256, "pic_parameter_set_id out of range");
581        let sps_id = r.read_ue()?;
582        ensure!(sps_id < 32, "seq_parameter_set_id out of range");
583        let sps = spss
584            .get(sps_id as usize)
585            .and_then(|s| s.as_ref())
586            .ok_or_else(|| crate::craft::video::h264::decode::Error::MissingParameterSet(format!("SPS {sps_id} for PPS {id}")))?;
587        let entropy_coding_mode = r.read_flag()?;
588        let bottom_field_pic_order_in_frame_present = r.read_flag()?;
589        let num_slice_groups_minus1 = r.read_ue()?;
590        ensure!(num_slice_groups_minus1 < 8, "num_slice_groups out of range");
591        let num_slice_groups = num_slice_groups_minus1 + 1;
592        let mut slice_group_map_type = 0;
593        if num_slice_groups > 1 {
594            // Parsed for completeness; FMO decoding is unsupported.
595            slice_group_map_type = r.read_ue()?;
596            match slice_group_map_type {
597                0 => {
598                    for _ in 0..num_slice_groups {
599                        r.read_ue()?;
600                    }
601                }
602                2 => {
603                    for _ in 0..num_slice_groups - 1 {
604                        r.read_ue()?;
605                        r.read_ue()?;
606                    }
607                }
608                3..=5 => {
609                    r.read_flag()?;
610                    r.read_ue()?;
611                }
612                6 => {
613                    let n_minus1 = r.read_ue()?;
614                    // pic_size_in_map_units_minus1: one slice_group_id per map unit of the picture.
615                    ensure!(n_minus1 < sps.pic_width_in_mbs.saturating_mul(sps.pic_height_in_map_units), "pic_size_in_map_units out of range");
616                    let n = n_minus1 + 1;
617                    let bits = 32 - (num_slice_groups - 1).leading_zeros();
618                    for _ in 0..n {
619                        r.read_bits(bits)?;
620                    }
621                }
622                _ => {}
623            }
624        }
625        let num_ref_idx_l0_minus1 = r.read_ue()?;
626        let num_ref_idx_l1_minus1 = r.read_ue()?;
627        ensure!(num_ref_idx_l0_minus1 < 32 && num_ref_idx_l1_minus1 < 32, "num_ref_idx_default_active out of range");
628        let num_ref_idx_l0_default_active = num_ref_idx_l0_minus1 + 1;
629        let num_ref_idx_l1_default_active = num_ref_idx_l1_minus1 + 1;
630        let weighted_pred = r.read_flag()?;
631        // 7.4.2.2.1 fall-back: a value of 3 shall be considered equal to 2.
632        let weighted_bipred_idc = (r.read_bits(2)?).min(2);
633        // 7.4.2.2: pic_init_qp_minus26 in -(26 + QpBdOffsetY)..=25, pic_init_qs_minus26 in -26..=25.
634        let qp_bd_offset_y = 6 * (sps.bit_depth_luma.saturating_sub(8)) as i32;
635        let qp_min = -(26 + qp_bd_offset_y);
636        let pic_init_qp_minus26 = r.read_se()?;
637        ensure!((qp_min..=25).contains(&pic_init_qp_minus26), "pic_init_qp_minus26 out of range");
638        let pic_init_qs_minus26 = r.read_se()?;
639        ensure!((-26..=25).contains(&pic_init_qs_minus26), "pic_init_qs_minus26 out of range");
640        let pic_init_qp = 26 + pic_init_qp_minus26;
641        let pic_init_qs = 26 + pic_init_qs_minus26;
642        let chroma_qp_index_offset = r.read_se()?;
643        ensure!((-12..=12).contains(&chroma_qp_index_offset), "chroma_qp_index_offset out of range");
644        let deblocking_filter_control_present = r.read_flag()?;
645        let constrained_intra_pred = r.read_flag()?;
646        let redundant_pic_cnt_present = r.read_flag()?;
647        let mut transform_8x8_mode = false;
648        let mut pic_scaling_matrix_present = false;
649        let mut second_chroma_qp_index_offset = chroma_qp_index_offset;
650        let mut scaling = sps.scaling.clone();
651        if r.more_rbsp_data() {
652            transform_8x8_mode = r.read_flag()?;
653            pic_scaling_matrix_present = r.read_flag()?;
654            if pic_scaling_matrix_present {
655                let count = 6 + if sps.chroma_format_idc != 3 { 2 } else { 6 } * transform_8x8_mode as usize;
656                // Fall-back rule B uses the sequence-level lists (or Flat when the SPS has none, but then
657                // rule A applies per 7.4.2.2).
658                let lists = if sps.seq_scaling_matrix_present {
659                    parse_scaling_lists(&mut r, count, Some(&sps.scaling_lists))?
660                } else {
661                    parse_scaling_lists(&mut r, count, None)?
662                };
663                scaling = lists.to_matrices();
664            }
665            second_chroma_qp_index_offset = r.read_se()?;
666            ensure!((-12..=12).contains(&second_chroma_qp_index_offset), "second_chroma_qp_index_offset out of range");
667        }
668        Ok(Pps {
669            id,
670            sps_id,
671            entropy_coding_mode,
672            bottom_field_pic_order_in_frame_present,
673            num_slice_groups,
674            slice_group_map_type,
675            num_ref_idx_l0_default_active,
676            num_ref_idx_l1_default_active,
677            weighted_pred,
678            weighted_bipred_idc,
679            pic_init_qp,
680            pic_init_qs,
681            chroma_qp_index_offset,
682            deblocking_filter_control_present,
683            constrained_intra_pred,
684            redundant_pic_cnt_present,
685            transform_8x8_mode,
686            pic_scaling_matrix_present,
687            second_chroma_qp_index_offset,
688            scaling,
689        })
690    }
691}

773            }

778    }
