145fn read_s(r: &mut BitReader, bits: u32) -> Result<i32> {
146    let v = r.read_bits(bits)? as i32;
147    Ok(if r.read_flag()? { -v } else { v })
148}
149
150fn read_delta_q(r: &mut BitReader) -> Result<i8> {
151    Ok(if r.read_flag()? { read_s(r, 4)? as i8 } else { 0 })
152}
153
154fn read_prob(r: &mut BitReader) -> Result<u8> {
155    Ok(if r.read_flag()? { r.read_bits(8)? as u8 } else { 255 })
156}

158fn frame_sync_code(r: &mut BitReader) -> Result<()> {
159    let (a, b, c) = (r.read_bits(8)?, r.read_bits(8)?, r.read_bits(8)?);
160    ensure!(a == 0x49 && b == 0x83 && c == 0x42, "invalid frame sync code");
161    Ok(())
162}

164fn color_config(r: &mut BitReader, profile: u8) -> Result<ColorConfig> {
165    let bit_depth = if profile >= 2 { if r.read_flag()? { 12 } else { 10 } } else { 8 };
166    let color_space = r.read_bits(3)? as u8;
167    let mut c = ColorConfig { bit_depth, color_space, color_range: true, subsampling_x: true, subsampling_y: true };
168    if color_space != 7 {
169        c.color_range = r.read_flag()?;
170        if profile == 1 || profile == 3 {
171            c.subsampling_x = r.read_flag()?;
172            c.subsampling_y = r.read_flag()?;
173            ensure!(!(c.subsampling_x && c.subsampling_y), "4:2:0 subsampling in profile {profile}");
174            ensure!(!r.read_flag()?, "reserved bit set in color config");
175        }
176    } else {
177        ensure!(profile == 1 || profile == 3, "RGB color space in profile {profile}");
178        c.subsampling_x = false;
179        c.subsampling_y = false;
180        ensure!(!r.read_flag()?, "reserved bit set in color config");
181    }
182    Ok(c)
183}

185fn frame_size(r: &mut BitReader, h: &mut FrameHeader) -> Result<()> {
186    h.width = r.read_bits(16)? + 1;
187    h.height = r.read_bits(16)? + 1;
188    Ok(())
189}

191fn render_size(r: &mut BitReader, h: &mut FrameHeader) -> Result<()> {
192    if r.read_flag()? {
193        h.render_width = r.read_bits(16)? + 1;
194        h.render_height = r.read_bits(16)? + 1;
195    } else {
196        h.render_width = h.width;
197        h.render_height = h.height;
198    }
199    Ok(())
200}

202fn compute_image_size(h: &mut FrameHeader) {
203    h.mi_cols = (h.width + 7) >> 3;
204    h.mi_rows = (h.height + 7) >> 3;
205    h.sb64_cols = (h.mi_cols + 7) >> 3;
206    h.sb64_rows = (h.mi_rows + 7) >> 3;

209/// setup_past_independence (7.2): resets segmentation features, loop filter deltas and the
210/// probabilities of `fc`.
211fn setup_past_independence(st: &mut HeaderState, fc: &mut FrameContext) {
212    st.seg.feature_data = [[0; 4]; 8];
213    st.seg.feature_enabled = [[false; 4]; 8];
214    st.seg.abs_or_delta_update = false;
215    st.lf.delta_enabled = true;
216    st.lf.ref_deltas = [1, 0, -1, -1];
217    st.lf.mode_deltas = [0, 0];
218    st.reset_segment_map = true;
219    *fc = FrameContext::default();
220}

222/// Intra-only (non-key) frame header tail (6.2): sync code, colour config, refresh flags, sizes.
223fn intra_only_header(r: &mut BitReader, h: &mut FrameHeader) -> Result<()> {
224    frame_sync_code(r)?;
225    h.color = if h.profile > 0 {
226        color_config(r, h.profile)?
227    } else {
228        ColorConfig { bit_depth: 8, color_space: 1, color_range: false, subsampling_x: true, subsampling_y: true }
229    };
230    h.refresh_frame_flags = r.read_bits(8)? as u8;
231    frame_size(r, h)?;
232    render_size(r, h)
233}

235/// frame_size_with_refs (6.2.5): the size is copied from the first flagged reference, or read explicitly.
236fn frame_size_with_refs(r: &mut BitReader, h: &mut FrameHeader, refs: &[Option<RefInfo>; 8]) -> Result<()> {
237    let mut found = false;
238    for i in 0..3 {
239        if r.read_flag()? {
240            let idx = h.ref_frame_idx[i] as usize;
241            let Some(info) = refs[idx] else {
242                return Err(Error::MissingReference(format!("frame size from empty slot {idx}")));
243            };
244            h.width = info.width;
245            h.height = info.height;
246            found = true;
247            break;
248        }
249    }
250    if !found {
251        frame_size(r, h)?;
252    }
253    h.size_from_ref = found;
254    Ok(())
255}

257/// Inter frame header tail (6.2): reference selection, sizes, motion vector precision and filter.
258fn inter_frame_header(r: &mut BitReader, h: &mut FrameHeader, st: &HeaderState, refs: &[Option<RefInfo>; 8]) -> Result<()> {
259    h.color = st.color;
260    h.refresh_frame_flags = r.read_bits(8)? as u8;
261    for i in 0..3 {
262        h.ref_frame_idx[i] = r.read_bits(3)? as u8;
263        h.ref_frame_sign_bias[LAST_FRAME as usize + i] = r.read_flag()?;
264    }
265    frame_size_with_refs(r, h, refs)?;
266    render_size(r, h)?;
267    h.allow_high_precision_mv = r.read_flag()?;
268    // read_interpolation_filter (6.2.7)
269    h.interp_filter = if r.read_flag()? { SWITCHABLE } else { LITERAL_TO_TYPE[r.read_bits(2)? as usize] };
270    Ok(())
271}

273/// Resets the probability contexts for intra / error resilient frames (7.2).
274fn reset_frame_contexts(st: &mut HeaderState, h: &mut FrameHeader) {
275    let mut fc = FrameContext::default();
276    setup_past_independence(st, &mut fc);
277    if h.frame_type == KEY_FRAME || h.error_resilient_mode || h.reset_frame_context == 3 {
278        for c in st.contexts.iter_mut() {
279            *c = fc.clone();
280        }
281    } else if h.reset_frame_context == 2 {
282        st.contexts[h.frame_context_idx as usize] = fc.clone();
283    }
284    h.frame_context_idx = 0;
285}

287/// loop_filter_params (6.2.8).
288fn read_loop_filter_params(r: &mut BitReader, lf: &mut LoopFilterParams) -> Result<()> {
289    lf.level = r.read_bits(6)? as u8;
290    lf.sharpness = r.read_bits(3)? as u8;
291    lf.delta_enabled = r.read_flag()?;
292    if lf.delta_enabled && r.read_flag()? {
293        for i in 0..4 {
294            if r.read_flag()? {
295                lf.ref_deltas[i] = read_s(r, 6)? as i8;
296            }
297        }
298        for i in 0..2 {
299            if r.read_flag()? {
300                lf.mode_deltas[i] = read_s(r, 6)? as i8;
301            }
302        }
303    }
304    Ok(())
305}

307/// segmentation_params (6.2.11).
308fn read_segmentation_params(r: &mut BitReader, seg: &mut Segmentation) -> Result<()> {
309    seg.enabled = r.read_flag()?;
310    seg.update_map = false;
311    seg.temporal_update = false;
312    if !seg.enabled {
313        return Ok(());
314    }
315    seg.update_map = r.read_flag()?;
316    if seg.update_map {
317        for i in 0..7 {
318            seg.tree_probs[i] = read_prob(r)?;
319        }
320        seg.temporal_update = r.read_flag()?;
321        for i in 0..3 {
322            seg.pred_probs[i] = if seg.temporal_update { read_prob(r)? } else { 255 };
323        }
324    }
325    if r.read_flag()? {
326        seg.abs_or_delta_update = r.read_flag()?;
327        for i in 0..MAX_SEGMENTS {
328            for j in 0..4 {
329                let mut value = 0i32;
330                let enabled = r.read_flag()?;
331                seg.feature_enabled[i][j] = enabled;
332                if enabled {
333                    value = r.read_bits(SEGMENTATION_FEATURE_BITS[j])? as i32;
334                    if SEGMENTATION_FEATURE_SIGNED[j] && r.read_flag()? {
335                        value = -value;
336                    }
337                }
338                seg.feature_data[i][j] = value as i16;
339            }
340        }
341    }
342    Ok(())
343}

345/// tile_info (6.2.13).
346fn read_tile_info(r: &mut BitReader, h: &mut FrameHeader) -> Result<()> {
347    let mut min_log2 = 0;
348    while (MAX_TILE_WIDTH_B64 << min_log2) < h.sb64_cols {
349        min_log2 += 1;
350    }
351    let mut max_log2 = 1;
352    while (h.sb64_cols >> max_log2) >= MIN_TILE_WIDTH_B64 {
353        max_log2 += 1;
354    }
355    max_log2 -= 1;
356    h.tile_cols_log2 = min_log2;
357    while h.tile_cols_log2 < max_log2 {
358        if r.read_flag()? {
359            h.tile_cols_log2 += 1;
360        } else {
361            break;
362        }
363    }
364    h.tile_rows_log2 = r.read_bits(1)?;
365    if h.tile_rows_log2 == 1 {
366        h.tile_rows_log2 += r.read_bits(1)?;
367    }
368    Ok(())
369}

371/// Parse the uncompressed header (6.2). `refs` gives the sizes of the eight reference slots.
372/// Updates the persistent state (probability contexts on reset, loop filter deltas,
373/// segmentation, colour configuration).
374pub fn parse_uncompressed(data: &[u8], st: &mut HeaderState, refs: &[Option<RefInfo>; 8]) -> Result<FrameHeader> {
375    let mut r = BitReader::new(data);
376    let mut h = FrameHeader::default();
377    ensure!(r.read_bits(2)? == 2, "invalid frame marker");
378    let lo = r.read_bits(1)? as u8;
379    let hi = r.read_bits(1)? as u8;
380    h.profile = (hi << 1) | lo;
381    if h.profile == 3 {
382        ensure!(r.read_bits(1)? == 0, "reserved bit set after profile");
383    }
384    h.show_existing_frame = r.read_flag()?;
385    if h.show_existing_frame {
386        h.frame_to_show_map_idx = r.read_bits(3)? as u8;
387        h.show_frame = true;
388        h.uncompressed_size = r.position().div_ceil(8);
389        return Ok(h);
390    }
391    st.last_frame_type = st.frame_type;
392    h.frame_type = r.read_bits(1)? as u8;
393    h.show_frame = r.read_flag()?;
394    h.error_resilient_mode = r.read_flag()?;
395    if h.frame_type == KEY_FRAME {
396        frame_sync_code(&mut r)?;
397        h.color = color_config(&mut r, h.profile)?;
398        frame_size(&mut r, &mut h)?;
399        render_size(&mut r, &mut h)?;
400        h.refresh_frame_flags = 0xff;
401        h.frame_is_intra = true;
402    } else {
403        h.intra_only = if h.show_frame { false } else { r.read_flag()? };
404        h.frame_is_intra = h.intra_only;
405        h.reset_frame_context = if h.error_resilient_mode { 0 } else { r.read_bits(2)? as u8 };
406        if h.intra_only {
407            intra_only_header(&mut r, &mut h)?;
408        } else {
409            inter_frame_header(&mut r, &mut h, st, refs)?;
410        }
411    }
412    compute_image_size(&mut h);
413    if !h.error_resilient_mode {
414        h.refresh_frame_context = r.read_flag()?;
415        h.frame_parallel_decoding_mode = r.read_flag()?;
416    } else {
417        h.refresh_frame_context = false;
418        h.frame_parallel_decoding_mode = true;
419    }
420    h.frame_context_idx = r.read_bits(2)? as u8;
421    if h.frame_is_intra || h.error_resilient_mode {
422        reset_frame_contexts(st, &mut h);
423    }
424    read_loop_filter_params(&mut r, &mut st.lf)?;
425    // quantization_params (6.2.9)
426    h.base_q_idx = r.read_bits(8)? as u8;
427    h.delta_q_y_dc = read_delta_q(&mut r)?;
428    h.delta_q_uv_dc = read_delta_q(&mut r)?;
429    h.delta_q_uv_ac = read_delta_q(&mut r)?;
430    h.lossless = h.base_q_idx == 0 && h.delta_q_y_dc == 0 && h.delta_q_uv_dc == 0 && h.delta_q_uv_ac == 0;
431    read_segmentation_params(&mut r, &mut st.seg)?;
432    read_tile_info(&mut r, &mut h)?;
433    h.header_size_in_bytes = r.read_bits(16)?;
434    // trailing_bits
435    h.uncompressed_size = r.position().div_ceil(8);
436    if h.frame_is_intra {
437        st.color = h.color;
438    }
439    st.frame_type = h.frame_type;
440    ensure!(h.header_size_in_bytes > 0, "empty compressed header");
441    Ok(h)
442}

444/// diff_update_prob (6.3.3).
445fn diff_update_prob(bd: &mut BoolDecoder, prob: &mut u8) {
446    if bd.read_bool(252) {
447        let delta = decode_term_subexp(bd);
448        *prob = inv_remap_prob(delta, *prob);
449    }
450}
451
452fn decode_term_subexp(bd: &mut BoolDecoder) -> usize {
453    if bd.read_literal(1) == 0 {
454        return bd.read_literal(4) as usize;
455    }
456    if bd.read_literal(1) == 0 {
457        return bd.read_literal(4) as usize + 16;
458    }
459    if bd.read_literal(1) == 0 {
460        return bd.read_literal(5) as usize + 32;
461    }
462    let v = bd.read_literal(7) as usize;
463    if v < 65 {
464        return v + 64;
465    }
466    let bit = bd.read_literal(1) as usize;
467    (v << 1) - 1 + bit
468}

470fn inv_recenter_nonneg(v: i32, m: i32) -> i32 {
471    if v > 2 * m {
472        v
473    } else if v & 1 != 0 {
474        m - ((v + 1) >> 1)
475    } else {
476        m + (v >> 1)
477    }
478}
479
480fn inv_remap_prob(delta: usize, prob: u8) -> u8 {
481    let v = INV_MAP_TABLE[delta.min(254)] as i32;
482    let m = prob as i32 - 1;
483    let r = if (m << 1) <= 255 { 1 + inv_recenter_nonneg(v, m) } else { 255 - inv_recenter_nonneg(v, 255 - 1 - m) };
484    r.clamp(1, 255) as u8
485}

487/// update_mv_prob (6.3.17).
488fn update_mv_prob(bd: &mut BoolDecoder, prob: &mut u8) {
489    if bd.read_bool(252) {
490        *prob = ((bd.read_literal(7) << 1) | 1) as u8;
491    }
492}

494/// tx_mode_probs (6.3.2): per-size transform probability updates for `TX_MODE_SELECT`.
495fn read_tx_mode_probs(bd: &mut BoolDecoder, fc: &mut FrameContext) {
496    for i in 0..2 {
497        diff_update_prob(bd, &mut fc.tx[1][i][0]);
498    }
499    for i in 0..2 {
500        for j in 0..2 {
501            diff_update_prob(bd, &mut fc.tx[2][i][j]);
502        }
503    }
504    for i in 0..2 {
505        for j in 0..3 {
506            diff_update_prob(bd, &mut fc.tx[3][i][j]);
507        }
508    }
509}

511/// read_coef_probs (6.3.7).
512fn read_coef_probs(bd: &mut BoolDecoder, h: &FrameHeader, fc: &mut FrameContext) {
513    let max_tx = TX_MODE_TO_BIGGEST_TX_SIZE[h.tx_mode as usize] as usize;
514    for tx in 0..=max_tx {
515        if bd.read_literal(1) == 1 {
516            for i in 0..2 {
517                for j in 0..2 {
518                    for k in 0..6 {
519                        let max_l = if k == 0 { 3 } else { 6 };
520                        for l in 0..max_l {
521                            for m in 0..3 {
522                                diff_update_prob(bd, &mut fc.coef[tx][i][j][k][l][m]);
523                            }
524                        }
525                    }
526                }
527            }
528        }
529    }
530}

532/// frame_reference_mode (6.3.12) including setup_compound_reference_mode (6.3.18).
533fn read_frame_reference_mode(bd: &mut BoolDecoder, h: &mut FrameHeader) {
534    let bias = h.ref_frame_sign_bias;
535    let compound_allowed = bias[2] != bias[1] || bias[3] != bias[1];
536    h.reference_mode = 0;
537    if !compound_allowed {
538        return;
539    }
540    if bd.read_literal(1) == 1 {
541        h.reference_mode = if bd.read_literal(1) == 1 { 2 } else { 1 };
542    }
543    if bias[LAST_FRAME as usize] == bias[GOLDEN_FRAME as usize] {
544        h.comp_fixed_ref = ALTREF_FRAME;
545        h.comp_var_ref = [LAST_FRAME, GOLDEN_FRAME];
546    } else if bias[LAST_FRAME as usize] == bias[ALTREF_FRAME as usize] {
547        h.comp_fixed_ref = GOLDEN_FRAME;
548        h.comp_var_ref = [LAST_FRAME, ALTREF_FRAME];
549    } else {
550        h.comp_fixed_ref = LAST_FRAME;
551        h.comp_var_ref = [GOLDEN_FRAME, ALTREF_FRAME];
552    }
553}

555/// frame_reference_mode_probs (6.3.13).
556fn read_frame_reference_mode_probs(bd: &mut BoolDecoder, reference_mode: u8, fc: &mut FrameContext) {
557    if reference_mode == 2 {
558        for i in 0..5 {
559            diff_update_prob(bd, &mut fc.comp_mode[i]);
560        }
561    }
562    if reference_mode != 1 {
563        for i in 0..5 {
564            diff_update_prob(bd, &mut fc.single_ref[i][0]);
565            diff_update_prob(bd, &mut fc.single_ref[i][1]);
566        }
567    }
568    if reference_mode != 0 {
569        for i in 0..5 {
570            diff_update_prob(bd, &mut fc.comp_ref[i]);
571        }
572    }
573}

575/// mv_probs (6.3.16).
576fn read_mv_probs(bd: &mut BoolDecoder, allow_high_precision_mv: bool, fc: &mut FrameContext) {
577    for j in 0..3 {
578        update_mv_prob(bd, &mut fc.mv_joint[j]);
579    }
580    for i in 0..2 {
581        update_mv_prob(bd, &mut fc.mv_sign[i]);
582        for j in 0..10 {
583            update_mv_prob(bd, &mut fc.mv_class[i][j]);
584        }
585        update_mv_prob(bd, &mut fc.mv_class0_bit[i]);
586        for j in 0..10 {
587            update_mv_prob(bd, &mut fc.mv_bits[i][j]);
588        }
589    }
590    for i in 0..2 {
591        for j in 0..2 {
592            for k in 0..3 {
593                update_mv_prob(bd, &mut fc.mv_class0_fr[i][j][k]);
594            }
595        }
596        for k in 0..3 {
597            update_mv_prob(bd, &mut fc.mv_fr[i][k]);
598        }
599    }
600    if allow_high_precision_mv {
601        for i in 0..2 {
602            update_mv_prob(bd, &mut fc.mv_class0_hp[i]);
603            update_mv_prob(bd, &mut fc.mv_hp[i]);
604        }
605    }
606}

608/// Probability updates that only inter frames carry (6.3.8 to 6.3.16).
609fn read_inter_frame_probs(bd: &mut BoolDecoder, h: &mut FrameHeader, fc: &mut FrameContext) {
610    for i in 0..7 {
611        for j in 0..3 {
612            diff_update_prob(bd, &mut fc.inter_mode[i][j]);
613        }
614    }
615    if h.interp_filter == SWITCHABLE {
616        for j in 0..4 {
617            for i in 0..2 {
618                diff_update_prob(bd, &mut fc.interp_filter[j][i]);
619            }
620        }
621    }
622    for i in 0..4 {
623        diff_update_prob(bd, &mut fc.is_inter[i]);
624    }
625    read_frame_reference_mode(bd, h);
626    read_frame_reference_mode_probs(bd, h.reference_mode, fc);
627    for i in 0..4 {
628        for j in 0..9 {
629            diff_update_prob(bd, &mut fc.y_mode[i][j]);
630        }
631    }
632    for i in 0..16 {
633        for j in 0..3 {
634            diff_update_prob(bd, &mut fc.partition[i][j]);
635        }
636    }
637    read_mv_probs(bd, h.allow_high_precision_mv, fc);
638}

640/// Parse the compressed header (6.3) into `h` and the probabilities `fc`.
641pub fn parse_compressed(data: &[u8], h: &mut FrameHeader, fc: &mut FrameContext) -> Result<()> {
642    let mut bd = BoolDecoder::new(data);
643    // read_tx_mode (6.3.1)
644    h.tx_mode = if h.lossless {
645        0
646    } else {
647        let mut m = bd.read_literal(2) as u8;
648        if m == 3 {
649            m += bd.read_literal(1) as u8;
650        }
651        m
652    };
653    if h.tx_mode == TX_MODE_SELECT {
654        read_tx_mode_probs(&mut bd, fc);
655    }
656    read_coef_probs(&mut bd, h, fc);
657    for i in 0..3 {
658        diff_update_prob(&mut bd, &mut fc.skip[i]);
659    }
660    if !h.frame_is_intra {
661        read_inter_frame_probs(&mut bd, h, fc);
662    }
663    Ok(())
664}

666/// Superframe index parsing (Annex B): the frames of a chunk.
667pub fn split_superframe(data: &[u8]) -> Vec<&[u8]> {
668    if let Some(&last) = data.last()
669        && last & 0xe0 == 0xc0
670    {
671        let sz_bytes = ((last >> 3) & 3) as usize + 1;
672        let num = (last & 7) as usize + 1;
673        let index_size = 2 + num * sz_bytes;
674        if data.len() >= index_size && data[data.len() - index_size] == last {
675            let mut frames = Vec::with_capacity(num);
676            let mut pos = 0usize;
677            let mut p = data.len() - index_size + 1;
678            let end = data.len() - index_size;
679            for _ in 0..num {
680                let mut sz = 0usize;
681                for b in 0..sz_bytes {
682                    sz |= (data[p + b] as usize) << (8 * b);
683                }
684                p += sz_bytes;
685                let Some(next) = pos.checked_add(sz).filter(|&next| next <= end) else {
686                    frames.push(&data[pos..end]);
687                    return frames;
688                };
689                frames.push(&data[pos..next]);
690                pos = next;
691            }
692            return frames;
693        }
694    }
695    vec![data]
696}
