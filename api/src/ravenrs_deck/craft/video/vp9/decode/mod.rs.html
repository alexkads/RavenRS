1//! Clean-room pure-Rust VP9 decoder, implemented from the public VP9 Bitstream & Decoding Process
2//! Specification (v0.6, WebM project).
3//!
4//! Supported: profiles 0-3 (8 / 10 / 12-bit; 4:2:0, 4:2:2, 4:4:0, 4:4:4), all coding tools:
5//! boolean decoder, forward and backward probability adaptation, segmentation, loop filter with
6//! deltas, all intra modes, inter prediction (8-tap regular / smooth / sharp and bilinear,
7//! scaled references), compound prediction, motion vector prediction, DCT / ADST 4..32 and the
8//! Walsh-Hadamard lossless transform, tiles (tile columns decode in parallel), superframes,
9//! hidden frames, show_existing_frame, intra-only frames, reference frame resizing and error
10//! resilient / frame parallel modes.
11//!
12//! ```ignore
13//! let mut dec = ravenrs_deck::craft::video::vp9::decode::Decoder::new();
14//! let chunk: Vec<u8> = Vec::new(); // one IVF / WebM frame
15//! for pic in dec.decode(&chunk, 0).unwrap() {
16//!     println!("{}x{} {}-bit", pic.width, pic.height, pic.bit_depth);
17//! }
18//! ```

43/// Stream parameters read from a key frame's uncompressed header (see [`keyframe_info`]).
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub struct KeyframeInfo {
46    pub profile: u8,
47    pub width: u32,
48    pub height: u32,
49    pub render_width: u32,
50    pub render_height: u32,
51    pub bit_depth: u32,
52    pub subsampling_x: bool,
53    pub subsampling_y: bool,
54    pub color: ColorInfo,
55}

57/// If the first frame of `chunk` (one container sample: a frame or a superframe) is a key frame,
58/// i.e. a random access point, its stream parameters. Intra-only frames and show_existing_frame
59/// are not random access points.
60pub fn keyframe_info(chunk: &[u8]) -> Option<KeyframeInfo> {
61    let first = header::split_superframe(chunk).into_iter().find(|f| !f.is_empty())?;
62    let mut st = header::HeaderState::default();
63    let h = header::parse_uncompressed(first, &mut st, &[None; 8]).ok()?;
64    if h.show_existing_frame || h.frame_type != header::KEY_FRAME {
65        return None;
66    }
67    Some(KeyframeInfo {
68        profile: h.profile,
69        width: h.width,
70        height: h.height,
71        render_width: h.render_width,
72        render_height: h.render_height,
73        bit_depth: h.color.bit_depth as u32,
74        subsampling_x: h.color.subsampling_x,
75        subsampling_y: h.color.subsampling_y,
76        color: ColorInfo { color_space: h.color.color_space, full_range: h.color.color_range },
77    })
78}

80/// Whether `chunk` starts with a key frame (decoding can start there).
81pub fn is_keyframe(chunk: &[u8]) -> bool {
82    keyframe_info(chunk).is_some()
83}

96/// Samples of one plane: 8-bit streams produce `U8`, higher bit depths `U16`.
97#[derive(Clone, Debug, PartialEq, Eq)]
98pub enum Plane {
99    U8(Vec<u8>),
100    U16(Vec<u16>),
101}
102
103impl Plane {
104    pub fn len(&self) -> usize {
105        match self {
106            Plane::U8(v) => v.len(),
107            Plane::U16(v) => v.len(),
108        }
109    }
110    pub fn is_empty(&self) -> bool {
111        self.len() == 0
112    }
113    pub fn as_u8(&self) -> Option<&[u8]> {
114        match self {
115            Plane::U8(v) => Some(v),
116            Plane::U16(_) => None,
117        }
118    }
119    pub fn as_u16(&self) -> Option<&[u16]> {
120        match self {
121            Plane::U8(_) => None,
122            Plane::U16(v) => Some(v),
123        }
124    }
125    /// Sample `i` widened to u16.
126    pub fn get(&self, i: usize) -> u16 {
127        match self {
128            Plane::U8(v) => v[i] as u16,
129            Plane::U16(v) => v[i],
130        }
131    }
132    /// Raw bytes as written by ffmpeg `rawvideo` (8-bit, or 16-bit little endian).
133    pub fn to_le_bytes(&self) -> Vec<u8> {
134        match self {
135            Plane::U8(v) => v.clone(),
136            Plane::U16(v) => v.iter().flat_map(|s| s.to_le_bytes()).collect(),
137        }
138    }
139}

162    pub color: ColorInfo,
163    /// Intended display size (render_size); has no effect on decoding.
