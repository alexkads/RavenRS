66/// Scratch rows above the samples of every band: the loop filter of superblock row r reads 8
67/// and modifies up to 7 rows of row r - 1 (the 16-wide filter of the horizontal edges on top of
68/// row r), so those rows are copied there while row r is filtered (see `decoder::post`).
69pub const BAND_PAD: usize = 8;

71/// One superblock row (64 luma rows) of a frame: per plane `BAND_PAD` scratch rows followed by

73pub struct Band {

93/// A decoded frame (also used as reference). Its samples are published one superblock row
94/// ("band") at a time, once loop filtered, so that later frames can predict from the finished
95/// top of a frame while its bottom is still being filtered (frame threading).
96pub struct Frame {
97    pub info: FrameInfo,
98    /// Allocated plane widths (whole superblocks) = row strides.
99    pub strides: [usize; 3],
100    /// Rows per band of each plane.
101    pub band_h: [usize; 3],
102    /// Visible (cropped) plane sizes.
103    pub vis: [(usize, usize); 3],
104    bands: Box<[std::sync::OnceLock<Band>]>,
105    pools: std::sync::Arc<Pools>,
106}
107
108impl Frame {
109    pub fn new(info: FrameInfo, pools: std::sync::Arc<Pools>) -> Frame {
110        let geo = plane_geometry(info.width, info.height, info.ss_x, info.ss_y);
111        let sb_rows = (info.height as usize).div_ceil(64);
112        Frame {
113            info,
114            strides: geo.map(|g| g.0),
115            band_h: [64, 64 >> info.ss_y as usize, 64 >> info.ss_y as usize],
116            vis: geo.map(|g| (g.2, g.3)),
117            bands: (0..sb_rows).map(|_| std::sync::OnceLock::new()).collect(),
118            pools,
119        }
120    }
121
122    pub fn sb_rows(&self) -> usize {
123        self.bands.len()
124    }
125
126    /// A band buffer for this frame's geometry (recycled memory, contents unspecified).
127    pub fn new_band(&self) -> Band {
128        Band { planes: std::array::from_fn(|p| self.pools.bands.take_any((self.band_h[p] + BAND_PAD) * self.strides[p], 0)) }
129    }
130
131    /// Publish band `r` (ignored if already published).
132    pub fn publish(&self, r: usize, band: Band) {
133        if let Err(b) = self.bands[r].set(band) {
134            for p in b.planes {
135                self.pools.bands.put(p);
136            }
137        }
138    }
139
140    pub fn is_published(&self, r: usize) -> bool {
141        self.bands[r].get().is_some()
142    }
143
144    pub fn is_complete(&self) -> bool {
145        self.bands.iter().all(|b| b.get().is_some())
146    }
147
148    /// Band `r`, waiting until it has been published.
149    #[inline]
150    pub fn band(&self, r: usize) -> &Band {
151        {
152            self.bands[r].wait()
153        }
154        #[cfg(any())]
155        {
156            // Bands are decoded in order before use; an unpublished one reads as empty.
157            static EMPTY: Band = Band { planes: [Vec::new(), Vec::new(), Vec::new()] };
158            self.bands[r].get().unwrap_or(&EMPTY)
159        }
160    }
161
162    /// Row `y` of plane `p` (all `strides[p]` samples), waiting until it is final.
163    #[inline]
164    pub fn row(&self, p: usize, y: usize) -> &[u16] {
165        let bh = self.band_h[p];
166        let s = self.strides[p];
167        let o = (y % bh + BAND_PAD) * s;
168        &self.band(y / bh).planes[p][o..o + s]
169    }
170
171    /// Rows `y0..y1` of plane `p` as one slice with stride `strides[p]` when they lie in one band:
172    /// (band samples, offset of row y0).
173    #[inline]
174    pub fn span(&self, p: usize, y0: usize, y1: usize) -> Option<(&[u16], usize)> {
175        let bh = self.band_h[p];
176        let r = y0 / bh;
177        if (y1 - 1) / bh != r {
178            return None;
179        }
180        Some((&self.band(r).planes[p], (y0 % bh + BAND_PAD) * self.strides[p]))
181    }
182
183    /// A complete frame from full planes (tests).
184    #[cfg(any())]
185    pub fn from_planes(info: FrameInfo, planes: [&[u16]; 3]) -> Frame {
186        let f = Frame::new(info, Default::default());
187        for r in 0..f.sb_rows() {
188            let mut b = f.new_band();
189            for p in 0..3 {
190                let (s, bh) = (f.strides[p], f.band_h[p]);
191                for k in 0..bh {
192                    let y = r * bh + k;
193                    if let Some(src) = planes[p].get(y * s..y * s + s) {
194                        b.planes[p][(k + BAND_PAD) * s..(k + BAND_PAD + 1) * s].copy_from_slice(src);
195                    }
196                }
197            }
198            f.publish(r, b);
199        }
200        f
201    }
202}
203
204impl Drop for Frame {
205    fn drop(&mut self) {
206        for b in std::mem::take(&mut self.bands).into_vec().into_iter().filter_map(|b| b.into_inner()) {
207            for p in b.planes {
208                self.pools.bands.put(p);
209            }
210        }
211    }
212}

214/// Geometry of the frame planes: allocation sizes for (MiCols, MiRows) rounded to superblocks.
215pub fn plane_geometry(width: u32, height: u32, ss_x: bool, ss_y: bool) -> [(usize, usize, usize, usize); 3] {
216    let sb_cols = (width as usize).div_ceil(64);
217    let sb_rows = (height as usize).div_ceil(64);
218    let (aw, ah) = (sb_cols * 64, sb_rows * 64);
219    let (sx, sy) = (ss_x as usize, ss_y as usize);
220    let cw = (width as usize + sx) >> sx;
221    let ch = (height as usize + sy) >> sy;
222    [(aw, ah, width as usize, height as usize), (aw >> sx, ah >> sy, cw, ch), (aw >> sx, ah >> sy, cw, ch)]
223}

225/// Recycled buffers (strip / frame planes, mode info), so that steady-state decoding does not
226/// allocate (and page-fault) megabytes per frame.
227pub struct Pool<T> {
228    bufs: std::sync::Mutex<Vec<Vec<T>>>,
229    max: usize,
230}
231
232impl<T> Default for Pool<T> {
233    fn default() -> Self {
234        Pool::with_max(32)
235    }
236}
237
238impl<T> Pool<T> {
239    /// A pool keeping at most `max` buffers.
240    pub fn with_max(max: usize) -> Self {
241        Pool { bufs: Default::default(), max }
242    }
243}
244
245impl<T: Clone> Pool<T> {
246    /// A buffer of `len` elements with unspecified contents (recycled memory when available;
247    /// `fill` only initialises fresh memory).
248    ///
249    /// Invariant for callers: every element that is read must have been written first. For the
250    /// sample planes and mode info grids this holds because every sample inside the mode info grid
251    /// is predicted before a later stage (loop filter, reference reads, intra edges) reads it, and
252    /// every mode info position is written by its tile before it is read. Reads are clamped to
253    /// that grid (intra edges to `max_x` / `max_y`, references to the visible size). A change that
254    /// reads outside the grid would see stale data from a recycled buffer.
255    pub fn take_any(&self, len: usize, fill: T) -> Vec<T> {
256        let found = {
257            let mut b = self.bufs.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
258            // Best fit by capacity: the first fit handed large buffers to small requests (band
259            // scratch rows, chroma bands), and once truncated they never served a large one
260            // again, so large requests kept allocating fresh (zeroed) memory.
261            let mut best: Option<(usize, usize)> = None;
262            for (i, v) in b.iter().enumerate() {
263                let c = v.capacity();
264                if c >= len && best.is_none_or(|(_, bc)| c < bc) {
265                    best = Some((i, c));
266                    if c == len {
267                        break;
268                    }
269                }
270            }
271            best.map(|(i, _)| b.swap_remove(i))
272        };
273        match found {
274            Some(mut v) => {
275                // Contents are unspecified: only memory beyond the current length is written.
276                if v.len() >= len {
277                    v.truncate(len);
278                } else {
279                    v.resize(len, fill);
280                }
281                v
282            }
283            None => vec![fill; len],
284        }
285    }
286
287    pub fn put(&self, v: Vec<T>) {
288        if v.capacity() == 0 {
289            return;
290        }
291        let mut b = self.bufs.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
292        if b.len() < self.max {
293            b.push(v);
294        }
295    }
296}
