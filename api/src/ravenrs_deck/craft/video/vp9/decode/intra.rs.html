5/// Edge availability and geometry of one intra-predicted transform block.
6pub struct IntraEdge {
7    pub have_left: bool,
8    pub have_above: bool,
9    pub not_on_right: bool,
10    pub tx_size: u8,
11    /// Last valid column / row of the plane (in plane coordinates).
12    pub max_x: usize,
13    pub max_y: usize,
14    pub bit_depth: u8,
15}
16
17/// Neighbouring samples of a transform block: `above[0]` is the top-left corner,
18/// `above[1..=2*size]` the above row and `left[..size]` the left column.
19struct Neighbours {
20    above: [i32; 65],
21    left: [i32; 32],
22}
23
24impl Neighbours {
25    /// aboveRow[i] for `i` in `-1..2*size`.
26    fn a(&self, i: isize) -> i32 {
27        self.above[(i + 1) as usize]
28    }
29}
30
31type Block = [[i32; 32]; 32];
32
33fn r1(v: i32) -> i32 {
34    (v + 1) >> 1
35}
36
37fn r2(v: i32) -> i32 {
38    (v + 2) >> 2
39}
40
41/// Gathers (and pads, per 8.5.1.1) the above row, top-left corner and left column.
42fn gather_neighbours(buf: &[u16], stride: usize, x_off: usize, x: usize, y: usize, e: &IntraEdge) -> Neighbours {
43    let size = 4usize << e.tx_size;
44    let base = 1i32 << (e.bit_depth - 1);
45    let at = |xx: usize, yy: usize| yy * stride + xx - x_off;
46    // aboveRow[-1..2*size) stored at above[0..2*size+1); leftCol[0..size).
47    let mut n = Neighbours { above: [0i32; 65], left: [0i32; 32] };
48    if !e.have_above {
49        n.above[1..=2 * size].fill(base - 1);
50    } else {
51        let row = (y - 1) * stride;
52        for i in 0..size {
53            n.above[1 + i] = buf[row + (x + i).min(e.max_x) - x_off] as i32;
54        }
55        if e.not_on_right && e.tx_size == 0 {
56            for i in size..2 * size {
57                n.above[1 + i] = buf[row + (x + i).min(e.max_x) - x_off] as i32;
58            }
59        } else {
60            let v = n.above[size];
61            n.above[1 + size..=2 * size].fill(v);
62        }
63    }
64    n.above[0] = if e.have_above && e.have_left {
65        buf[at((x - 1).min(e.max_x), y - 1)] as i32
66    } else if e.have_above {
67        base + 1
68    } else {
69        base - 1
70    };
71    if e.have_left {
72        for i in 0..size {
73            n.left[i] = buf[at(x - 1, (y + i).min(e.max_y))] as i32;
74        }
75    } else {
76        n.left[..size].fill(base + 1);
77    }
78    n
79}

81fn pred_d207(n: &Neighbours, size: usize) -> Block {
82    let left = &n.left;
83    let mut p = [[0i32; 32]; 32];
84    for j in 0..size {
85        p[size - 1][j] = left[size - 1];
86    }
87    for i in 0..size - 1 {
88        p[i][0] = r1(left[i] + left[i + 1]);
89    }
90    for i in 0..size.saturating_sub(2) {
91        p[i][1] = r2(left[i] + 2 * left[i + 1] + left[i + 2]);
92    }
93    p[size - 2][1] = r2(left[size - 2] + 3 * left[size - 1]);
94    for j in 2..size {
95        for i in (0..size - 1).rev() {
96            p[i][j] = p[i + 1][j - 2];
97        }
98    }
99    p
100}

102fn pred_d117(n: &Neighbours, size: usize) -> Block {
103    let (left, a) = (&n.left, |i: isize| n.a(i));
104    let mut p = [[0i32; 32]; 32];
105    for j in 0..size {
106        p[0][j] = r1(a(j as isize - 1) + a(j as isize));
107    }
108    p[1][0] = r2(left[0] + 2 * a(-1) + a(0));
109    for j in 1..size {
110        p[1][j] = r2(a(j as isize - 2) + 2 * a(j as isize - 1) + a(j as isize));
111    }
112    p[2][0] = r2(a(-1) + 2 * left[0] + left[1]);
113    for i in 3..size {
114        p[i][0] = r2(left[i - 3] + 2 * left[i - 2] + left[i - 1]);
115    }
116    for i in 2..size {
117        for j in 1..size {
118            p[i][j] = p[i - 2][j - 1];
119        }
120    }
121    p
122}

124fn pred_d135(n: &Neighbours, size: usize) -> Block {
125    let (left, a) = (&n.left, |i: isize| n.a(i));
126    let mut p = [[0i32; 32]; 32];
127    p[0][0] = r2(left[0] + 2 * a(-1) + a(0));
128    for j in 1..size {
129        p[0][j] = r2(a(j as isize - 2) + 2 * a(j as isize - 1) + a(j as isize));
130    }
131    p[1][0] = r2(a(-1) + 2 * left[0] + left[1]);
132    for i in 2..size {
133        p[i][0] = r2(left[i - 2] + 2 * left[i - 1] + left[i]);
134    }
135    for i in 1..size {
136        for j in 1..size {
137            p[i][j] = p[i - 1][j - 1];
138        }
139    }
140    p
141}

143fn pred_d153(n: &Neighbours, size: usize) -> Block {
144    let (left, a) = (&n.left, |i: isize| n.a(i));
145    let mut p = [[0i32; 32]; 32];
146    p[0][0] = r1(left[0] + a(-1));
147    for i in 1..size {
148        p[i][0] = r1(left[i - 1] + left[i]);
149    }
150    p[0][1] = r2(left[0] + 2 * a(-1) + a(0));
151    p[1][1] = r2(a(-1) + 2 * left[0] + left[1]);
152    for i in 2..size {
153        p[i][1] = r2(left[i - 2] + 2 * left[i - 1] + left[i]);
154    }
155    for j in 2..size {
156        p[0][j] = r2(a(j as isize - 3) + 2 * a(j as isize - 2) + a(j as isize - 1));
157    }
158    for i in 1..size {
159        for j in 2..size {
160            p[i][j] = p[i - 1][j - 2];
161        }
162    }
163    p
164}

166/// DC prediction value, depending on which edges are available.
167fn dc_value(n: &Neighbours, e: &IntraEdge) -> i32 {
168    let log2 = e.tx_size as usize + 2;
169    let size = 1usize << log2;
170    let left = n.left[..size].iter().sum::<i32>();
171    let above = n.above[1..=size].iter().sum::<i32>();
172    match (e.have_left, e.have_above) {
173        (true, true) => (left + above + size as i32) >> (log2 + 1),
174        (true, false) => (left + (1 << (log2 - 1))) >> log2,
175        (false, true) => (above + (1 << (log2 - 1))) >> log2,
176        (false, false) => 1i32 << (e.bit_depth - 1),
177    }
178}

180/// Predict a `4 << tx_size` square block at (`x`, `y`) of a plane buffer. `buf` is addressed as
181/// `buf[y * stride + x - x_off]` (the buffer may be a tile-column strip starting at column `x_off`).
182pub fn predict(buf: &mut [u16], stride: usize, x_off: usize, x: usize, y: usize, mode: u8, e: &IntraEdge) {
183    let size = 4usize << e.tx_size;
184    let n = gather_neighbours(buf, stride, x_off, x, y, e);
185    let a = |i: isize| n.a(i);
186    let max = (1i32 << e.bit_depth) - 1;
187    let o = y * stride + x - x_off;
188    let mut put = |i: usize, j: usize, v: i32| buf[o + i * stride + j] = v as u16;
189    let mut fill = |f: &dyn Fn(usize, usize) -> i32| {
190        for i in 0..size {
191            for j in 0..size {
192                put(i, j, f(i, j));
193            }
194        }
195    };
196    match mode {
197        V_PRED => fill(&|_, j| a(j as isize)),
198        H_PRED => fill(&|i, _| n.left[i]),
199        D207_PRED => {
200            let p = pred_d207(&n, size);
201            fill(&|i, j| p[i][j]);
202        }
203        D45_PRED => fill(&|i, j| {
204            let k = (i + j) as isize;
205            if i + j + 2 < 2 * size { r2(a(k) + 2 * a(k + 1) + a(k + 2)) } else { a(2 * size as isize - 1) }
206        }),
207        D63_PRED => fill(&|i, j| {
208            let k = (i / 2 + j) as isize;
209            if i & 1 == 1 { r2(a(k) + 2 * a(k + 1) + a(k + 2)) } else { r1(a(k) + a(k + 1)) }
210        }),
211        D117_PRED => {
212            let p = pred_d117(&n, size);
213            fill(&|i, j| p[i][j]);
214        }
215        D135_PRED => {
216            let p = pred_d135(&n, size);
217            fill(&|i, j| p[i][j]);
218        }
219        D153_PRED => {
220            let p = pred_d153(&n, size);
221            fill(&|i, j| p[i][j]);
222        }
223        TM_PRED => fill(&|i, j| (a(j as isize) + n.left[i] - a(-1)).clamp(0, max)),
224        _ => {
225            // DC_PRED
226            let v = dc_value(&n, e);
227            fill(&|_, _| v);
228        }
229    }
