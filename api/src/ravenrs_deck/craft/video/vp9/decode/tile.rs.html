29/// Read-only data shared by all tile columns of a frame.
30pub struct FrameShared<'a> {
31    pub h: &'a FrameHeader,
32    pub fc: &'a FrameContext,
33    pub seg: &'a Segmentation,
34    /// PrevSegmentIds (MiRows x MiCols).
35    pub prev_seg_ids: &'a [u8],
36    /// Mode info of the previous frame when UsePrevFrameMvs is 1.
37    pub prev_mi: Option<&'a MiGrid>,
38    /// References for LAST, GOLDEN, ALTREF.
39    pub refs: [Option<RefUse<'a>>; 3],
40    /// Quantizers per segment: [segment][plane > 0][dc = 0 / ac = 1].
41    pub seg_q: [[[i32; 2]; 2]; 8],
42    /// Whether syntax elements need to be counted (backward adaptation enabled).
43    pub counting: bool,
44    pub pools: &'a crate::craft::video::vp9::decode::frame::Pools,

128impl<'a> TileDecoder<'a> {
129    pub fn new(s: &'a FrameShared<'a>, mi_col_start: usize, mi_col_end: usize) -> Self {
130        let h = s.h;
131        let (mi_cols, mi_rows) = (h.mi_cols as usize, h.mi_rows as usize);
132        let (ss_x, ss_y) = (h.color.subsampling_x as usize, h.color.subsampling_y as usize);
133        let sb_rows = h.sb64_rows as usize;
134        let sb_cols = h.sb64_cols as usize;
135        let strip_w = (mi_col_end - mi_col_start).div_ceil(8) * 64;
136        let strip_h = sb_rows * 64;
137        let x0 = mi_col_start * 8;
138        let uv_len = (strip_w >> ss_x) * (strip_h >> ss_y);
139        // Every sample a later stage reads (inside the mode info grid) is predicted before use, so
140        // recycled planes need no clearing.
141        let planes = [s.pools.samples.take_any(strip_w * strip_h, 0), s.pools.samples.take_any(uv_len, 0), s.pools.samples.take_any(uv_len, 0)];
142        let mi_w = mi_col_end - mi_col_start;
143        let strip = Strip {
144            mi_col_start,
145            mi_col_end,
146            // Written for every position of the grid before it is read.
147            mi: s.pools.mi.take_any(mi_w * mi_rows, MiInfo::default()),
148            seg_ids: vec![0; mi_w * mi_rows],
149            planes,
150            strides: [strip_w, strip_w >> ss_x, strip_w >> ss_x],
151            x_off: [x0, x0 >> ss_x, x0 >> ss_x],
152            counts: Box::default(),
153            compound_blocks: 0,
154            scaled_blocks: 0,
155            intra_blocks: 0,
156            inter_blocks: 0,
157        };
158        let an = sb_cols * 16 + 16;
159        let ln = sb_rows * 16 + 16;
160        TileDecoder {
161            s,
162            h,
163            fc: s.fc,
164            strip,
165            mi_cols,
166            mi_rows,
167            ss_x,
168            ss_y,
169            bit_depth: h.color.bit_depth,
170            above_nonzero: [vec![0; an], vec![0; an], vec![0; an]],
171            left_nonzero: [vec![0; ln], vec![0; ln], vec![0; ln]],
172            above_partition: vec![0; sb_cols * 8 + 8],
173            left_partition: vec![0; sb_rows * 8 + 8],
174            above_seg_pred: vec![0; sb_cols * 8 + 8],
175            left_seg_pred: vec![0; sb_rows * 8 + 8],
176            ref_list_mv: [Mv::ZERO; 2],
177            ref_mv_count: 0,
178            mode_context: [0; 4],
179            nearest_mv: [Mv::ZERO; 2],
180            near_mv: [Mv::ZERO; 2],
181            best_mv: [Mv::ZERO; 2],
182            coefs: vec![0; 1024],
183            token_cache: vec![0; 1024],
184            pred: vec![0; 64 * 64],
185            pred2: vec![0; 64 * 64],
186            // Scaled prediction reads up to 5x the block height (8.5.2.3), unscaled h + 7 rows.
187            mc_tmp: vec![0; if s.refs.iter().flatten().any(|r| r.x_scale != 1 << 14 || r.y_scale != 1 << 14) { 64 * (64 * 5 + 16) } else { 64 * 71 }],
188            mc_win: vec![0; 71 * 71],
189            error: None,
190        }
191    }

198    /// decode_tile (6.4.2) for one tile of this column. Returns false if the data ran out badly.
199    pub fn decode_tile(&mut self, data: &[u8], mi_row_start: usize, mi_row_end: usize) -> bool {
200        let mut bd = BoolDecoder::new(data);
201        let (cs, ce) = (self.strip.mi_col_start, self.strip.mi_col_end);
202        for r in (mi_row_start..mi_row_end).step_by(8) {
203            // clear_left_context
204            for p in 0..3 {
205                let sy = if p > 0 { self.ss_y } else { 0 };
206                let s = (r * 2) >> sy;
207                let e = (s + (16 >> sy)).min(self.left_nonzero[p].len());
208                self.left_nonzero[p][s..e].fill(0);
209            }
210            self.left_partition[r..r + 8].fill(0);
211            self.left_seg_pred[r..r + 8].fill(0);
212            for c in (cs..ce).step_by(8) {
213                self.decode_partition(&mut bd, r, c, BLOCK_64X64);
214            }
215        }
216        true
217    }

219    fn decode_partition(&mut self, bd: &mut BoolDecoder, r: usize, c: usize, bsize: u8) {
220        if r >= self.mi_rows || c >= self.mi_cols {
221            return;
222        }
223        let num8x8 = NUM_8X8_WIDE[bsize as usize] as usize;
224        let half = num8x8 >> 1;
225        let has_rows = (r + half) < self.mi_rows;
226        let has_cols = (c + half) < self.mi_cols;
227        let bsl = MI_WIDTH_LOG2[bsize as usize] as usize;
228        let boffset = 3 - bsl;
229        let mut above = 0u8;
230        let mut left = 0u8;
231        for i in 0..num8x8 {
232            above |= self.above_partition[c + i];
233            left |= self.left_partition[r + i];
234        }
235        let above = (above >> boffset) & 1;
236        let left = (left >> boffset) & 1;
237        let ctx = bsl * 4 + left as usize * 2 + above as usize;
238        let probs: &[u8] = if self.h.frame_is_intra { &KF_PARTITION_PROBS[ctx * 3..ctx * 3 + 3] } else { &self.fc.partition[ctx] };
239        let partition = if has_rows && has_cols {
240            bd.read_tree(&PARTITION_TREE, probs)
241        } else if has_cols {
242            if bd.read_bool(probs[1]) { 3 } else { 1 }
243        } else if has_rows {
244            if bd.read_bool(probs[2]) { 3 } else { 2 }
245        } else {
246            3
247        };
248        if self.s.counting {
249            self.strip.counts.partition[ctx][partition as usize] += 1;
250        }
251        let subsize = SUBSIZE_LOOKUP[partition as usize][bsize as usize];
252        if subsize < BLOCK_8X8 || partition == 0 {
253            self.decode_block(bd, r, c, subsize);
254        } else if partition == 1 {
255            self.decode_block(bd, r, c, subsize);
256            if has_rows {
257                self.decode_block(bd, r + half, c, subsize);
258            }
259        } else if partition == 2 {
260            self.decode_block(bd, r, c, subsize);
261            if has_cols {
262                self.decode_block(bd, r, c + half, subsize);
263            }
264        } else {
265            self.decode_partition(bd, r, c, subsize);
266            self.decode_partition(bd, r, c + half, subsize);
267            self.decode_partition(bd, r + half, c, subsize);
268            self.decode_partition(bd, r + half, c + half, subsize);
269        }
270        if bsize == BLOCK_8X8 || partition != 3 {
271            let a = 15 >> B_WIDTH_LOG2[subsize as usize];
272            let l = 15 >> B_HEIGHT_LOG2[subsize as usize];
273            self.above_partition[c..c + num8x8].fill(a);
274            self.left_partition[r..r + num8x8].fill(l);
275        }
276    }

278    fn decode_block(&mut self, bd: &mut BoolDecoder, r: usize, c: usize, size: u8) {
279        let mut b = Blk { r, c, size, avail_u: r > 0, avail_l: c > self.strip.mi_col_start, ..Default::default() };
280        // Chroma block size must be valid (7.4.3); treat violations as a damaged stream.
281        if size >= BLOCK_8X8 && SS_SIZE_LOOKUP[size as usize][self.ss_x][self.ss_y] == BLOCK_INVALID {
282            b.size = BLOCK_8X8;
283        }
284        if self.h.frame_is_intra {
285            self.intra_frame_mode_info(bd, &mut b);
286        } else {
287            self.inter_frame_mode_info(bd, &mut b);
288        }
289        if b.is_inter {
290            self.strip.inter_blocks += 1;
291        } else {
292            self.strip.intra_blocks += 1;
293        }
294        self.residual(bd, &mut b);
295        if b.is_inter && b.size >= BLOCK_8X8 && b.eob_total == 0 {
296            b.skip = true;
297        }
298        let info = MiInfo {
299            sb_size: b.size,
300            skip: b.skip,
301            tx_size: b.tx_size,
302            y_mode: b.y_mode,
303            sub_modes: b.sub_modes,
304            seg_id: b.seg_id,
305            ref_frame: b.ref_frame,
306            interp_filter: b.interp_filter,
307            mv: b.block_mvs,
308        };
309        let bh = NUM_8X8_HIGH[b.size as usize] as usize;
310        let bw = NUM_8X8_WIDE[b.size as usize] as usize;
311        let mi_w = self.strip.mi_col_end - self.strip.mi_col_start;
312        let (y1, x1) = ((r + bh).min(self.mi_rows), (c + bw).min(self.strip.mi_col_end));
313        for y in r..y1 {
314            let row = y * mi_w;
315            for x in c..x1 {
316                self.strip.mi[row + x - self.strip.mi_col_start] = info;
317                self.strip.seg_ids[row + x - self.strip.mi_col_start] = b.seg_id;
318            }
319        }
320    }

324    fn intra_frame_mode_info(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
325        // intra_segment_id
326        b.seg_id = if self.s.seg.enabled && self.s.seg.update_map { bd.read_tree(&SEGMENT_TREE, &self.s.seg.tree_probs) } else { 0 };
327        self.read_skip(bd, b);
328        self.read_tx_size(bd, b, true);
329        b.ref_frame = [INTRA_FRAME, NONE];
330        b.is_inter = false;
331        let above_mi = if b.avail_u { Some(*self.mi_at(b.r - 1, b.c)) } else { None };
332        let left_mi = if b.avail_l { Some(*self.mi_at(b.r, b.c - 1)) } else { None };
333        if b.size >= BLOCK_8X8 {
334            let above = above_mi.map_or(DC_PRED, |m| m.sub_modes[2]);
335            let left = left_mi.map_or(DC_PRED, |m| m.sub_modes[1]);
336            let o = (above as usize * 10 + left as usize) * 9;
337            let mode = bd.read_tree(&INTRA_MODE_TREE, &KF_Y_MODE_PROBS[o..o + 9]);
338            b.y_mode = mode;
339            b.sub_modes = [mode; 4];
340        } else {
341            let n4w = NUM_4X4_WIDE[b.size as usize] as usize;
342            let n4h = NUM_4X4_HIGH[b.size as usize] as usize;
343            let mut mode = 0;
344            let mut idy = 0;
345            while idy < 2 {
346                let mut idx = 0;
347                while idx < 2 {
348                    let above = if idy > 0 { b.sub_modes[idx] } else { above_mi.map_or(DC_PRED, |m| m.sub_modes[2 + idx]) };
349                    let left = if idx > 0 { b.sub_modes[idy * 2] } else { left_mi.map_or(DC_PRED, |m| m.sub_modes[1 + idy * 2]) };
350                    let o = (above as usize * 10 + left as usize) * 9;
351                    mode = bd.read_tree(&INTRA_MODE_TREE, &KF_Y_MODE_PROBS[o..o + 9]);
352                    for y2 in 0..n4h {
353                        for x2 in 0..n4w {
354                            b.sub_modes[(idy + y2) * 2 + idx + x2] = mode;
355                        }
356                    }
357                    idx += n4w;
358                }
359                idy += n4h;
360            }
361            b.y_mode = mode;
362        }
363        let o = b.y_mode as usize * 9;
364        b.uv_mode = bd.read_tree(&INTRA_MODE_TREE, &KF_UV_MODE_PROBS[o..o + 9]);
365    }

367    fn read_skip(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
368        if self.s.seg.feature_active(b.seg_id, SEG_LVL_SKIP) {
369            b.skip = true;
370        } else {
371            let mut ctx = 0;
372            if b.avail_u {
373                ctx += self.mi_at(b.r - 1, b.c).skip as usize;
374            }
375            if b.avail_l {
376                ctx += self.mi_at(b.r, b.c - 1).skip as usize;
377            }
378            b.skip = bd.read_bool(self.fc.skip[ctx]);
379            if self.s.counting {
380                self.strip.counts.skip[ctx][b.skip as usize] += 1;
381            }
382        }
383    }

385    fn read_tx_size(&mut self, bd: &mut BoolDecoder, b: &mut Blk, allow_select: bool) {
386        let max_tx = MAX_TXSIZE[b.size as usize];
387        if allow_select && self.h.tx_mode == TX_MODE_SELECT && b.size >= BLOCK_8X8 {
388            let mut above = max_tx;
389            let mut left = max_tx;
390            if b.avail_u {
391                let m = self.mi_at(b.r - 1, b.c);
392                if !m.skip {
393                    above = m.tx_size;
394                }
395            }
396            if b.avail_l {
397                let m = self.mi_at(b.r, b.c - 1);
398                if !m.skip {
399                    left = m.tx_size;
400                }
401            }
402            if !b.avail_l {
403                left = above;
404            }
405            if !b.avail_u {
406                above = left;
407            }
408            let ctx = ((above + left) > max_tx) as usize;
409            let tree: &[i8] = match max_tx {
410                3 => &TX_SIZE_32_TREE,
411                2 => &TX_SIZE_16_TREE,
412                _ => &TX_SIZE_8_TREE,
413            };
414            b.tx_size = bd.read_tree(tree, &self.fc.tx[max_tx as usize][ctx]);
415            if self.s.counting {
416                self.strip.counts.tx[max_tx as usize][ctx][b.tx_size as usize] += 1;
417            }
418        } else {
419            b.tx_size = max_tx.min(TX_MODE_TO_BIGGEST_TX_SIZE[self.h.tx_mode as usize]);
420        }
421    }

423    fn inter_frame_mode_info(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
424        b.left_ref = if b.avail_l { self.mi_at(b.r, b.c - 1).ref_frame } else { [INTRA_FRAME, NONE] };
425        b.above_ref = if b.avail_u { self.mi_at(b.r - 1, b.c).ref_frame } else { [INTRA_FRAME, NONE] };
426        b.left_intra = b.left_ref[0] <= INTRA_FRAME;
427        b.above_intra = b.above_ref[0] <= INTRA_FRAME;
428        b.left_single = b.left_ref[1] <= NONE;
429        b.above_single = b.above_ref[1] <= NONE;
430        self.inter_segment_id(bd, b);
431        self.read_skip(bd, b);
432        self.read_is_inter(bd, b);
433        let allow = !b.skip || !b.is_inter;
434        self.read_tx_size(bd, b, allow);
435        if b.is_inter {
436            self.inter_block_mode_info(bd, b);
437        } else {
438            self.intra_block_mode_info(bd, b);
439        }
440    }

442    fn get_segment_id(&self, b: &Blk) -> u8 {
443        let bw = NUM_8X8_WIDE[b.size as usize] as usize;
444        let bh = NUM_8X8_HIGH[b.size as usize] as usize;
445        let xmis = (self.mi_cols - b.c).min(bw);
446        let ymis = (self.mi_rows - b.r).min(bh);
447        let mut seg = 7;
448        let prev = self.s.prev_seg_ids;
449        if prev.is_empty() {
450            return 0;
451        }
452        for y in 0..ymis {
453            for x in 0..xmis {
454                seg = seg.min(prev[(b.r + y) * self.mi_cols + b.c + x]);
455            }
456        }
457        seg
458    }

460    fn inter_segment_id(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
461        let seg = self.s.seg;
462        if !seg.enabled {
463            b.seg_id = 0;
464            return;
465        }
466        let predicted = self.get_segment_id(b);
467        if !seg.update_map {
468            b.seg_id = predicted;
469            return;
470        }
471        if seg.temporal_update {
472            let ctx = (self.left_seg_pred[b.r] + self.above_seg_pred[b.c]) as usize;
473            let pred_flag = bd.read_bool(seg.pred_probs[ctx]);
474            b.seg_id = if pred_flag { predicted } else { bd.read_tree(&SEGMENT_TREE, &seg.tree_probs) };
475            let bw = NUM_8X8_WIDE[b.size as usize] as usize;
476            let bh = NUM_8X8_HIGH[b.size as usize] as usize;
477            self.above_seg_pred[b.c..b.c + bw].fill(pred_flag as u8);
478            self.left_seg_pred[b.r..b.r + bh].fill(pred_flag as u8);
479        } else {
480            b.seg_id = bd.read_tree(&SEGMENT_TREE, &seg.tree_probs);
481        }
482    }

484    fn read_is_inter(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
485        if self.s.seg.feature_active(b.seg_id, SEG_LVL_REF_FRAME) {
486            b.is_inter = self.s.seg.feature_data[b.seg_id as usize][SEG_LVL_REF_FRAME] != INTRA_FRAME as i16;
487        } else {
488            let ctx = if b.avail_u && b.avail_l {

492            } else {

541    fn inter_block_mode_info(&mut self, bd: &mut BoolDecoder, b: &mut Blk) {
542        self.read_ref_frames(bd, b);
543        let is_compound = b.ref_frame[1] > INTRA_FRAME;
544        if is_compound {
545            self.strip.compound_blocks += 1;
546        }
547        for j in 0..1 + is_compound as usize {
548            let rf = b.ref_frame[j];
549            self.find_mv_refs(b, rf, -1);
550            self.find_best_ref_mvs(b, j);
551        }
552        b.sub_modes = [DC_PRED; 4];
553        if self.s.seg.feature_active(b.seg_id, SEG_LVL_SKIP) {
554            b.y_mode = ZEROMV;
555        } else if b.size >= BLOCK_8X8 {
556            let ctx = self.mode_context[b.ref_frame[0] as usize] as usize;
557            let inter_mode = bd.read_tree(&INTER_MODE_TREE, &self.fc.inter_mode[ctx.min(6)]);
558            if self.s.counting {
559                self.strip.counts.inter_mode[ctx.min(6)][inter_mode as usize] += 1;
560            }
561            b.y_mode = NEARESTMV + inter_mode;
562        }
563        b.interp_filter = if self.h.interp_filter == SWITCHABLE {
564            let left = if b.avail_l && b.left_ref[0] > INTRA_FRAME { self.mi_at(b.r, b.c - 1).interp_filter } else { 3 };
565            let above = if b.avail_u && b.above_ref[0] > INTRA_FRAME { self.mi_at(b.r - 1, b.c).interp_filter } else { 3 };
566            let ctx = if left == above {
567                left
568            } else if left == 3 && above != 3 {
569                above
570            } else if left != 3 && above == 3 {
571                left
572            } else {
573                3
574            } as usize;
575            let f = bd.read_tree(&INTERP_FILTER_TREE, &self.fc.interp_filter[ctx]);
576            if self.s.counting {
577                self.strip.counts.interp_filter[ctx][f as usize] += 1;
578            }
579            f
580        } else {
581            self.h.interp_filter
582        };
583        b.block_mvs = [[Mv::ZERO; 4]; 2];
584        if b.size < BLOCK_8X8 {
585            let n4w = NUM_4X4_WIDE[b.size as usize] as usize;
586            let n4h = NUM_4X4_HIGH[b.size as usize] as usize;
587            let mut idy = 0;
588            while idy < 2 {
589                let mut idx = 0;
590                while idx < 2 {
591                    let ctx = self.mode_context[b.ref_frame[0] as usize] as usize;
592                    let inter_mode = bd.read_tree(&INTER_MODE_TREE, &self.fc.inter_mode[ctx.min(6)]);
593                    if self.s.counting {
594                        self.strip.counts.inter_mode[ctx.min(6)][inter_mode as usize] += 1;
595                    }
596                    b.y_mode = NEARESTMV + inter_mode;
597                    if b.y_mode == NEARESTMV || b.y_mode == NEARMV {
598                        for j in 0..1 + is_compound as usize {
599                            self.append_sub8x8_mvs(b, (idy * 2 + idx) as i32, j);
600                        }
601                    }
602                    let mv = self.assign_mv(bd, b, is_compound);
603                    for y2 in 0..n4h {
604                        for x2 in 0..n4w {
605                            let blk = (idy + y2) * 2 + idx + x2;
606                            for l in 0..1 + is_compound as usize {
607                                b.block_mvs[l][blk] = mv[l];
608                            }
609                        }
610                    }
611                    idx += n4w;
612                }
613                idy += n4h;
614            }
615        } else {
616            let mv = self.assign_mv(bd, b, is_compound);
617            for l in 0..1 + is_compound as usize {
618                b.block_mvs[l] = [mv[l]; 4];
619            }
620        }
621    }

624        let seg = self.s.seg;
625        if seg.feature_active(b.seg_id, SEG_LVL_REF_FRAME) {
626            b.ref_frame = [(seg.feature_data[b.seg_id as usize][SEG_LVL_REF_FRAME] as i8).clamp(LAST_FRAME, ALTREF_FRAME), NONE];
627            return;
628        }
629        let h = self.h;
630        let comp_mode = if h.reference_mode == 2 {
631            let ctx = self.comp_mode_ctx(b);
632            let v = bd.read_bool(self.fc.comp_mode[ctx]);
633            if self.s.counting {
634                self.strip.counts.comp_mode[ctx][v as usize] += 1;
635            }
636            v as u8
637        } else {
638            h.reference_mode
639        };
640        if comp_mode == 1 {
641            let idx = h.ref_frame_sign_bias[h.comp_fixed_ref as usize] as usize;
642            let ctx = self.comp_ref_ctx(b);
643            let comp_ref = bd.read_bool(self.fc.comp_ref[ctx]);

649        } else {

663                LAST_FRAME

665            b.ref_frame[1] = NONE;

669    fn comp_mode_ctx(&self, b: &Blk) -> usize {
670        let fixed = self.h.comp_fixed_ref;
671        (if b.avail_u && b.avail_l {
672            if b.above_single && b.left_single {
673                ((b.above_ref[0] == fixed) ^ (b.left_ref[0] == fixed)) as u8
674            } else if b.above_single {
675                2 + (b.above_ref[0] == fixed || b.above_intra) as u8
676            } else if b.left_single {
677                2 + (b.left_ref[0] == fixed || b.left_intra) as u8
678            } else {
679                4
680            }
681        } else if b.avail_u {
682            if b.above_single { (b.above_ref[0] == fixed) as u8 } else { 3 }
683        } else if b.avail_l {
684            if b.left_single { (b.left_ref[0] == fixed) as u8 } else { 3 }
685        } else {
686            1
687        }) as usize
688    }

690    fn comp_ref_ctx(&self, b: &Blk) -> usize {
691        let h = self.h;
692        let fix_ref_idx = h.ref_frame_sign_bias[h.comp_fixed_ref as usize] as usize;
693        let var_ref_idx = 1 - fix_ref_idx;
694        let var0 = h.comp_var_ref[0];
695        let var1 = h.comp_var_ref[1];
696        let fixed = h.comp_fixed_ref;
697        (if b.avail_u && b.avail_l {
698            if b.above_intra && b.left_intra {
699                2
700            } else if b.left_intra {
701                if b.above_single { 1 + 2 * (b.above_ref[0] != var1) as u8 } else { 1 + 2 * (b.above_ref[var_ref_idx] != var1) as u8 }
702            } else if b.above_intra {
703                if b.left_single { 1 + 2 * (b.left_ref[0] != var1) as u8 } else { 1 + 2 * (b.left_ref[var_ref_idx] != var1) as u8 }
704            } else {
705                let vrfa = if b.above_single { b.above_ref[0] } else { b.above_ref[var_ref_idx] };
706                let vrfl = if b.left_single { b.left_ref[0] } else { b.left_ref[var_ref_idx] };
707                if vrfa == vrfl && var1 == vrfa {
708                    0
709                } else if b.left_single && b.above_single {
710                    if (vrfa == fixed && vrfl == var0) || (vrfl == fixed && vrfa == var0) {
711                        4
712                    } else if vrfa == vrfl {
713                        3
714                    } else {
715                        1
716                    }
717                } else if b.left_single || b.above_single {
718                    let vrfc = if b.left_single { vrfa } else { vrfl };
719                    let rfs = if b.above_single { vrfa } else { vrfl };
720                    if vrfc == var1 && rfs != var1 {
721                        1
722                    } else if rfs == var1 && vrfc != var1 {
723                        2
724                    } else {
725                        4
726                    }
727                } else if vrfa == vrfl {
728                    4
729                } else {
730                    2
731                }
732            }
733        } else if b.avail_u {
734            if b.above_intra {
735                2
736            } else if b.above_single {
737                3 * (b.above_ref[0] != var1) as u8
738            } else {
739                4 * (b.above_ref[var_ref_idx] != var1) as u8
740            }
741        } else if b.avail_l {
742            if b.left_intra {
743                2
744            } else if b.left_single {
745                3 * (b.left_ref[0] != var1) as u8
746            } else {
747                4 * (b.left_ref[var_ref_idx] != var1) as u8
748            }
749        } else {
750            2
751        }) as usize
752    }

868    fn read_mv(&mut self, bd: &mut BoolDecoder, r: usize) -> Mv {
869        let use_hp = self.h.allow_high_precision_mv && use_mv_hp(self.best_mv[r]);
870        let joint = bd.read_tree(&MV_JOINT_TREE, &self.fc.mv_joint);
871        if self.s.counting {
872            self.strip.counts.mv_joint[joint as usize] += 1;
873        }
874        let mut diff = [0i32; 2];
875        if joint == 2 || joint == 3 {
876            diff[0] = self.read_mv_component(bd, 0, use_hp);
877        }
878        if joint == 1 || joint == 3 {
879            diff[1] = self.read_mv_component(bd, 1, use_hp);
880        }
881        Mv::new(self.best_mv[r].row as i32 + diff[0], self.best_mv[r].col as i32 + diff[1])
882    }

884    fn read_mv_component(&mut self, bd: &mut BoolDecoder, comp: usize, use_hp: bool) -> i32 {
885        let fc = self.fc;
886        let counting = self.s.counting;
887        let counts = &mut self.strip.counts;
888        let sign = bd.read_bool(fc.mv_sign[comp]);
889        let class = bd.read_tree(&MV_CLASS_TREE, &fc.mv_class[comp]) as usize;
890        let mag = if class == 0 {
891            let c0 = bd.read_bool(fc.mv_class0_bit[comp]) as usize;
892            let fr = bd.read_tree(&MV_FR_TREE, &fc.mv_class0_fr[comp][c0]) as usize;
893            let hp = if use_hp { bd.read_bool(fc.mv_class0_hp[comp]) as usize } else { 1 };
894            if counting {
895                counts.mv_class0_bit[comp][c0] += 1;
896                counts.mv_class0_fr[comp][c0][fr] += 1;
897                counts.mv_class0_hp[comp][hp] += 1;
898            }
899            ((c0 << 3) | (fr << 1) | hp) + 1
900        } else {
901            let mut d = 0usize;
902            for i in 0..class {
903                let bit = bd.read_bool(fc.mv_bits[comp][i]) as usize;
904                if counting {
905                    counts.mv_bits[comp][i][bit] += 1;
906                }
907                d |= bit << i;
908            }
909            let base = 2usize << (class + 2);
910            let fr = bd.read_tree(&MV_FR_TREE, &fc.mv_fr[comp]) as usize;
911            let hp = if use_hp { bd.read_bool(fc.mv_hp[comp]) as usize } else { 1 };
912            if counting {
913                counts.mv_fr[comp][fr] += 1;
914                counts.mv_hp[comp][hp] += 1;
915            }
916            base + ((d << 3) | (fr << 1) | hp) + 1
917        };
918        if counting {
919            counts.mv_sign[comp][sign as usize] += 1;
920            counts.mv_class[comp][class] += 1;
921        }
922        if sign { -(mag as i32) } else { mag as i32 }
923    }

978        let to_left = -((b.c as i32 * 8) * 8);
979        let to_right = ((self.mi_cols as i32 - bw - b.c as i32) * 8) * 8;
980        v.clamp(to_left - border, (to_right + border).max(to_left - border))
981    }

983    /// find_mv_refs (6.5.1) for `ref_frame`; `block` is -1 or the sub-8x8 block index.
984    fn find_mv_refs(&mut self, b: &Blk, ref_frame: i8, block: i32) {
985        self.ref_mv_count = 0;
986        self.ref_list_mv = [Mv::ZERO; 2];
987        let mut different_ref_found = false;
988        let mut context_counter = 0usize;
989        let search = &MV_REF_BLOCKS[b.size as usize * 16..b.size as usize * 16 + 16];
990        let (r, c) = (b.r as isize, b.c as isize);
991        for i in 0..2 {
992            let (cr, cc) = (r + search[i * 2] as isize, c + search[i * 2 + 1] as isize);
993            if self.is_inside(cr, cc) {
994                let m = *self.mi_at(cr as usize, cc as usize);
995                different_ref_found = true;
996                context_counter += MODE_2_COUNTER[m.y_mode as usize] as usize;
997                for j in 0..2 {
998                    if m.ref_frame[j] == ref_frame {
999                        let idx = if block >= 0 { IDX_N_COLUMN_TO_SUBBLOCK[block as usize][(search[i * 2 + 1] == 0) as usize] as usize } else { 3 };
1000                        self.add_mv_ref_list(m.mv[j][idx]);
1001                        break;
1002                    }
1003                }
1004            }
1005        }
1006        for i in 2..8 {
1007            let (cr, cc) = (r + search[i * 2] as isize, c + search[i * 2 + 1] as isize);
1008            if self.is_inside(cr, cc) {
1009                different_ref_found = true;
1010                let m = *self.mi_at(cr as usize, cc as usize);
1011                self.if_same_ref_frame_add(&m, ref_frame);
1012            }
1013        }
1014        let prev = self.s.prev_mi.map(|g| *g.at(b.r, b.c));
1015        if let Some(p) = &prev {
1016            self.if_same_ref_frame_add(p, ref_frame);
1017        }
1018        if different_ref_found {
1019            for i in 0..8 {
1020                let (cr, cc) = (r + search[i * 2] as isize, c + search[i * 2 + 1] as isize);
1021                if self.is_inside(cr, cc) {
1022                    let m = *self.mi_at(cr as usize, cc as usize);
1023                    self.if_diff_ref_frame_add(&m, ref_frame);
1024                }
1025            }
1026        }
1027        if let Some(p) = &prev {
1028            self.if_diff_ref_frame_add(p, ref_frame);
1029        }
1030        self.mode_context[ref_frame as usize] = COUNTER_TO_CONTEXT[context_counter.min(18)];
1031        for i in 0..2 {
1032            let mv = self.ref_list_mv[i];
1033            self.ref_list_mv[i] = Mv::new(self.clamp_mv_row(b, mv.row as i32, MV_BORDER), self.clamp_mv_col(b, mv.col as i32, MV_BORDER));
1034        }
1035    }

1037    /// find_best_ref_mvs (6.5.12).
1038    fn find_best_ref_mvs(&mut self, b: &Blk, ref_list: usize) {
1039        for i in 0..2 {
1040            let mv = self.ref_list_mv[i];
1041            let (mut dr, mut dc) = (mv.row as i32, mv.col as i32);
1042            if !self.h.allow_high_precision_mv || !use_mv_hp(mv) {
1043                if dr & 1 != 0 {
1044                    dr += if dr > 0 { -1 } else { 1 };
1045                }
1046                if dc & 1 != 0 {
1047                    dc += if dc > 0 { -1 } else { 1 };
1048                }
1049            }
1050            let border = (BORDERINPIXELS - INTERP_EXTEND) << 3;
1051            self.ref_list_mv[i] = Mv::new(self.clamp_mv_row(b, dr, border), self.clamp_mv_col(b, dc, border));
1052        }
1053        self.nearest_mv[ref_list] = self.ref_list_mv[0];
1054        self.near_mv[ref_list] = self.ref_list_mv[1];
1055        self.best_mv[ref_list] = self.ref_list_mv[0];
1056    }

1058    /// append_sub8x8_mvs (6.5.14).
1059    fn append_sub8x8_mvs(&mut self, b: &Blk, block: i32, ref_list: usize) {
1060        self.find_mv_refs(b, b.ref_frame[ref_list], block);
1061        let mut sub = [Mv::ZERO; 2];
1062        let mut dst;
1063        let bm = &b.block_mvs[ref_list];
1064        if block == 0 {
1065            sub = self.ref_list_mv;
1066            dst = 2;
1067        } else if block <= 2 {
1068            sub[0] = bm[0];
1069            dst = 1;
1070        } else {
1071            sub[0] = bm[2];
1072            dst = 1;
1073            for idx in [1usize, 0] {
1074                if dst < 2 && bm[idx] != sub[0] {
1075                    sub[dst] = bm[idx];
1076                    dst += 1;
1077                }
1078            }
1079        }
1080        for n in 0..2 {
1081            if dst < 2 && self.ref_list_mv[n] != sub[0] {
1082                sub[dst] = self.ref_list_mv[n];
1083                dst += 1;
1084            }
1085        }
1086        if dst < 2 {
1087            sub[dst] = Mv::ZERO;
1088        }
1089        self.nearest_mv[ref_list] = sub[0];
1090        self.near_mv[ref_list] = sub[1];
1091    }
