10/// Frame-level loop filter setup (8.8.1) plus the per-level limits of 8.8.4.
11pub struct LfFrame {
12    /// LvlLookup[segment_id][ref][modeType].
13    pub lvl: [[[u8; 2]; 4]; 8],
14    /// (limit, blimit, thresh) per filter level.
15    pub limits: [(i32, i32, i32); 64],
16    pub mi_rows: usize,
17    pub mi_cols: usize,
18    pub ss_x: usize,
19    pub ss_y: usize,
20    pub bit_depth: u8,
21}
22
23impl LfFrame {
24    pub fn new(lf: &LoopFilterParams, seg: &Segmentation, mi_rows: usize, mi_cols: usize, ss_x: bool, ss_y: bool, bit_depth: u8) -> LfFrame {
25        let mut lvl = [[[0u8; 2]; 4]; 8];
26        let n_shift = lf.level >> 5;
27        for (seg_id, l) in lvl.iter_mut().enumerate() {
28            let mut lvl_seg = lf.level as i32;
29            if seg.feature_active(seg_id as u8, SEG_LVL_ALT_L) {
30                let d = seg.feature_data[seg_id][SEG_LVL_ALT_L] as i32;
31                lvl_seg = if seg.abs_or_delta_update { d } else { d + lf.level as i32 };
32                lvl_seg = lvl_seg.clamp(0, MAX_LOOP_FILTER);
33            }
34            if !lf.delta_enabled {
35                *l = [[lvl_seg as u8; 2]; 4];
36            } else {
37                let intra = lvl_seg + ((lf.ref_deltas[0] as i32) << n_shift);
38                l[0][0] = intra.clamp(0, MAX_LOOP_FILTER) as u8;
39                l[0][1] = l[0][0];
40                for rf in 1..4 {
41                    for mode in 0..2 {
42                        let v = lvl_seg + ((lf.ref_deltas[rf] as i32) << n_shift) + ((lf.mode_deltas[mode] as i32) << n_shift);
43                        l[rf][mode] = v.clamp(0, MAX_LOOP_FILTER) as u8;
44                    }
45                }
46            }
47        }
48        let mut limits = [(0, 0, 0); 64];
49        let sh = lf.sharpness as i32;
50        let shift = if sh > 4 {
51            2
52        } else if sh > 0 {
53            1
54        } else {
55            0
56        };
57        for (l, e) in limits.iter_mut().enumerate() {
58            let l = l as i32;
59            let limit = if sh > 0 { (l >> shift).clamp(1, 9 - sh) } else { (l >> shift).max(1) };
60            *e = (limit, 2 * (l + 2) + limit, l >> 4);
61        }
62        LfFrame { lvl, limits, mi_rows, mi_cols, ss_x: ss_x as usize, ss_y: ss_y as usize, bit_depth }
63    }
64
65    fn level(&self, mi: &MiInfo) -> u8 {
66        let mode_type = matches!(mi.y_mode, NEARESTMV | NEARMV | NEWMV) as usize;
67        let rf = mi.ref_frame[0].max(0) as usize;
68        self.lvl[mi.seg_id as usize & 7][rf][mode_type]
69    }
70}

72/// A mutable view of one plane: sample (x, y) of the plane is `data[(y - oy) * stride + x - ox]`.
73pub struct PlaneView<'a> {
74    pub data: &'a mut [u16],
75    pub stride: usize,
76    pub ox: usize,
77    pub oy: usize,
78}

80/// Loop filter one superblock (all planes, both passes) at (`row`, `col`) in MI units (8.8.2).
81pub fn filter_superblock(views: &mut [PlaneView; 3], mi: &MiGrid, f: &LfFrame, row: usize, col: usize) {
82    for (plane, view) in views.iter_mut().enumerate() {
83        for pass in 0..2 {
84            filter_sb_plane(view, mi, f, plane, pass, row, col);
85        }
86    }
87}

89fn filter_sb_plane(v: &mut PlaneView, mi: &MiGrid, f: &LfFrame, plane: usize, pass: usize, row: usize, col: usize) {
90    let (sub_x, sub_y) = if plane > 0 { (f.ss_x, f.ss_y) } else { (0, 0) };
91    let (sub, edge_len) = if pass == 0 { (sub_x, 64 >> sub_y) } else { (sub_y, 64 >> sub_x) };
92    let (mi_rows, mi_cols) = (f.mi_rows, f.mi_cols);
93    // Step between the samples across the edge.
94    let across = if pass == 0 { 1isize } else { v.stride as isize };
95    for edge in 0..(16 >> sub) {
96        let mut i = 0;
97        while i < edge_len {
98            // Group of 8 samples sharing loopRow / loopCol.
99            let (x, y) = if pass == 0 {
100                (col * 8 + edge * (4 << sub_x), row * 8 + (i << sub_y))
101            } else {
102                (col * 8 + (i << sub_x), row * 8 + edge * (4 << sub_y))
103            };
104            // Skip groups entirely off screen (the varying coordinate is re-checked per sample).
105            if x >= 8 * mi_cols || y >= 8 * mi_rows || (pass == 0 && x == 0) || (pass == 1 && y == 0) {
106                i += 8;
107                continue;
108            }
109            let loop_col = ((x >> 3) >> sub_x) << sub_x;
110            let loop_row = ((y >> 3) >> sub_y) << sub_y;
111            let m = mi.at(loop_row, loop_col);
112            let tx_sz = if plane > 0 {
113                if m.sb_size < BLOCK_8X8 { 0 } else { m.tx_size.min(MAX_TXSIZE[SS_SIZE_LOOKUP[m.sb_size as usize][f.ss_x][f.ss_y].min(12) as usize]) }
114            } else {
115                m.tx_size
116            };
117            let sb_size = if sub == 0 { m.sb_size } else { m.sb_size.max(BLOCK_16X16) } as usize;
118            let is_intra = m.ref_frame[0] <= INTRA_FRAME;
119            // Block sizes are powers of two, so the modulo is a mask.
120            let is_block_edge =
121                if pass == 0 { x & (8 * NUM_8X8_WIDE[sb_size] as usize - 1) == 0 } else { y & (8 * NUM_8X8_HIGH[sb_size] as usize - 1) == 0 };
122            let tx_ok = edge & ((1 << tx_sz) - 1) == 0 && (is_intra || !m.skip);
123            if !is_block_edge && !tx_ok {
124                i += 8;
125                continue;
126            }
127            let lvl = f.level(m);
128            if lvl == 0 {
129                i += 8;
130                continue;
131            }
132            let is32 = edge % 8 == 0;
133            let base_size = if tx_sz == TX_4X4 && is32 { TX_8X8 } else { tx_sz.min(TX_16X16) };
134            let filter_size = if base_size == TX_16X16
135                && ((pass == 0 && sub_x == 1 && (x >> 3) == mi_cols - 1) || (pass == 1 && sub_y == 1 && (y >> 3) == mi_rows - 1))
136            {
137                TX_8X8
138            } else {
139                base_size
140            };
141            let (limit, blimit, thresh) = f.limits[lvl as usize];
142            // Lanes of the group (the 8 samples along the edge) inside the MI grid.
143            let n =
144                if pass == 0 { (8 * mi_rows - y).div_ceil(1 << sub_y) } else { (8 * mi_cols - x).div_ceil(1 << sub_x) }.min(LANES).min(edge_len - i);
145            // A transform edge is not filtered in the last odd chroma column (8.8.2).
146            let odd_last = pass == 1 && sub_x == 1 && mi_cols & 1 == 1 && edge & 1 == 1;
147            let mut apply = [false; LANES];
148            for (l, a) in apply.iter_mut().enumerate().take(n) {
149                let is_tx_edge = tx_ok && !(odd_last && x + (l << sub_x) + 8 >= mi_cols * 8);
150                *a = is_block_edge || is_tx_edge;
151            }
152            if apply.iter().any(|&a| a) {
153                let px = x >> sub_x;
154                let py = y >> sub_y;
155                let pos = (py - v.oy) * v.stride + px - v.ox;
156                let along = if pass == 0 { v.stride } else { 1 };
157                let lim = Limits { apply, limit, blimit, thresh, filter_size, bit_depth: f.bit_depth };
158                let (d, st) = (&mut *v.data, across as usize);
159                match (f.bit_depth <= 10, filter_size == TX_16X16, pass == 1) {
160                    (true, false, false) => lanes16::filter_group::<4, false>(d, pos, along, st, &lim),
161                    (true, false, true) => lanes16::filter_group::<4, true>(d, pos, along, st, &lim),
162                    (true, true, false) => lanes16::filter_group::<8, false>(d, pos, along, st, &lim),
163                    (true, true, true) => lanes16::filter_group::<8, true>(d, pos, along, st, &lim),
164                    (false, false, false) => lanes32::filter_group::<4, false>(d, pos, along, st, &lim),
165                    (false, false, true) => lanes32::filter_group::<4, true>(d, pos, along, st, &lim),
166                    (false, true, false) => lanes32::filter_group::<8, false>(d, pos, along, st, &lim),
167                    (false, true, true) => lanes32::filter_group::<8, true>(d, pos, along, st, &lim),
168                }
169            }
170            i += 8;
171        }
172    }
173}

179struct Limits {
180    apply: [bool; LANES],
181    limit: i32,
182    blimit: i32,
183    thresh: i32,
184    filter_size: u8,

188/// The group filter, instantiated for 16-bit lanes (bit depths 8 and 10: every intermediate,
189/// including the 16-tap sums of at most 16 * 1023 + 8, fits) and 32-bit lanes (12-bit).
190macro_rules! group_filter {
191    ($m:ident, $t:ty) => {
192        mod $m {
193            use super::{LANES, Limits};
194            use crate::craft::video::vp9::decode::tables::*;
195            type T = $t;
196
197            /// Sample filtering process (8.8.5) for the up to 8 lanes of a group whose `apply` flag is set.
198            /// Lane `l` has its q0 sample at `pos + l * along`; samples across the edge are `step` apart
199            /// (`ROWS`: `along` is 1, i.e. a horizontal edge). `R` samples on each side are read (8 for
200            /// 16x16 filters, else 4). Every lane is computed with every filter and the result chosen per
201            /// lane (8.8.5.1 decides which), so the arithmetic is branch-free and vectorizes; unmodified
202            /// samples are written back unchanged.
203            #[inline(always)]
204            pub(super) fn filter_group<const R: usize, const ROWS: bool>(d: &mut [u16], pos: usize, along: usize, step: usize, g: &Limits) {
205                let Limits { apply, limit, blimit, thresh, filter_size, bit_depth } = *g;
206                let (limit, blimit, thresh) = (limit as T, blimit as T, thresh as T);
207                // v[8 + k][l] = sample at offset k across the edge in lane l.
208                let mut v = [[0 as T; LANES]; 16];
209                let start = pos - R * step;
210                if ROWS {
211                    for k in 0..2 * R {
212                        let row: &[u16; LANES] = d[start + k * step..start + k * step + LANES].try_into().expect("lanes");
213                        v[8 - R + k] = std::array::from_fn(|l| row[l] as T);
214                    }
215                } else {
216                    for l in 0..LANES {
217                        let s = start + l * along;
218                        let line: &[u16] = &d[s..s + 2 * R];
219                        for k in 0..2 * R {
220                            v[8 - R + k][l] = line[k] as T;
221                        }
222                    }
223                }
224                let shift = bit_depth as u32 - 8;
225                let limit_bd = limit << shift;
226                let blimit_bd = blimit << shift;
227                let thresh_bd = thresh << shift;
228                let one = 1 << shift;
229                let (p3, p2, p1, p0, q0, q1, q2, q3) = (&v[4], &v[5], &v[6], &v[7], &v[8], &v[9], &v[10], &v[11]);
230                // Filter mask process (8.8.5.1). Masks are 0 / -1 per lane, selects are bitwise
231                // and clipping uses min / max (`clamp` asserts its bounds), so every lane loop
232                // below compiles to vector instructions.
233                let le = |a: T, b: T| -((a <= b) as T);
234                let sel = |m: T, a: T, b: T| (a & m) | (b & !m);
235                let mut mask = [0 as T; LANES];
236                let mut hev = [0 as T; LANES];
237                let mut flat = [0 as T; LANES];
238                let mut flat2 = [0 as T; LANES];
239                let can_flat = -((filter_size >= TX_8X8) as T);
240                let can_flat2 = R == 8 && filter_size >= TX_16X16;
241                let apply: [T; LANES] = apply.map(|a| -(a as T));
242                for l in 0..LANES {
243                    let d_p1p0 = (p1[l] - p0[l]).abs();
244                    let d_q1q0 = (q1[l] - q0[l]).abs();
245                    mask[l] = apply[l]
246                        & le((p3[l] - p2[l]).abs(), limit_bd)
247                        & le((p2[l] - p1[l]).abs(), limit_bd)
248                        & le(d_p1p0, limit_bd)
249                        & le(d_q1q0, limit_bd)
250                        & le((q2[l] - q1[l]).abs(), limit_bd)
251                        & le((q3[l] - q2[l]).abs(), limit_bd)
252                        & le((p0[l] - q0[l]).abs() * 2 + ((p1[l] - q1[l]).abs() >> 1), blimit_bd);
253                    hev[l] = !(le(d_p1p0, thresh_bd) & le(d_q1q0, thresh_bd));
254                    flat[l] = can_flat
255                        & mask[l]
256                        & le(d_p1p0, one)
257                        & le(d_q1q0, one)
258                        & le((p2[l] - p0[l]).abs(), one)
259                        & le((q2[l] - q0[l]).abs(), one)
260                        & le((p3[l] - p0[l]).abs(), one)
261                        & le((q3[l] - q0[l]).abs(), one);
262                }
263                if mask.iter().fold(0, |a, &m| a | m) == 0 {
264                    return;
265                }
266                if can_flat2 {
267                    for l in 0..LANES {
268                        flat2[l] = flat[l]
269                            & le((v[0][l] - p0[l]).abs(), one)
270                            & le((v[1][l] - p0[l]).abs(), one)
271                            & le((v[2][l] - p0[l]).abs(), one)
272                            & le((v[3][l] - p0[l]).abs(), one)
273                            & le((v[12][l] - q0[l]).abs(), one)
274                            & le((v[13][l] - q0[l]).abs(), one)
275                            & le((v[14][l] - q0[l]).abs(), one)
276                            & le((v[15][l] - q0[l]).abs(), one);
277                    }
278                }
279                let any_flat = flat.iter().fold(0, |a, &f| a | f) != 0;
280                let any_flat2 = flat2.iter().fold(0, |a, &f| a | f) != 0;
281                // Results per position across the edge; start from the unfiltered samples.
282                let mut out = v;
283                // Narrow filter (8.8.5.2) for lanes that are masked but not flat.
284                {
285                    let lo = -(1 << (bit_depth - 1)) as T;
286                    let hi = ((1 << (bit_depth - 1)) - 1) as T;
287                    let c = |x: T| x.max(lo).min(hi);
288                    let off: T = 0x80 << shift;
289                    for l in 0..LANES {
290                        let (ps1, ps0, qs0, qs1) = (p1[l] - off, p0[l] - off, q0[l] - off, q1[l] - off);
291                        let filter = c((c(ps1 - qs1) & hev[l]) + 3 * (qs0 - ps0));
292                        let filter1 = c(filter + 4) >> 3;
293                        let filter2 = c(filter + 3) >> 3;
294                        let f = (filter1 + 1) >> 1;
295                        let narrow = mask[l] & !flat[l];
296                        let outer = narrow & !hev[l];
297                        out[8][l] = sel(narrow, c(qs0 - filter1) + off, out[8][l]);
298                        out[7][l] = sel(narrow, c(ps0 + filter2) + off, out[7][l]);
299                        out[9][l] = sel(outer, c(qs1 - f) + off, out[9][l]);
300                        out[6][l] = sel(outer, c(ps1 + f) + off, out[6][l]);
301                    }
302                }
303                // Wide filter (8.8.5.3) with 8 taps for flat lanes (p2..q2) ...
304                if any_flat {
305                    let w8 = wide_filter::<3>(&v);
306                    for k in 5..11 {
307                        for l in 0..LANES {
308                            out[k][l] = sel(flat[l] & !flat2[l], w8[k][l], out[k][l]);
309                        }
310                    }
311                }
312                // ... and 16 taps for flat2 lanes (p6..q6).
313                if any_flat2 {
314                    let w16 = wide_filter::<4>(&v);
315                    for k in 1..15 {
316                        for l in 0..LANES {
317                            out[k][l] = sel(flat2[l], w16[k][l], out[k][l]);
318                        }
319                    }
320                }
321                let (lo, hi) = if any_flat2 {
322                    (1, 15)
323                } else if any_flat {
324                    (5, 11)
325                } else {
326                    (6, 10)
327                };
328                if ROWS {
329                    for k in lo..hi {
330                        let o = start + (k + R - 8) * step; // k >= 8 - R
331                        let row: &mut [u16; LANES] = (&mut d[o..o + LANES]).try_into().expect("lanes");
332                        for l in 0..LANES {
333                            row[l] = out[k][l] as u16;
334                        }
335                    }
336                } else {
337                    // Whole lines: unmodified samples are rewritten with their own value.
338                    for l in 0..LANES {
339                        let s = start + l * along;
340                        let line: &mut [u16] = &mut d[s..s + 2 * R];
341                        for k in 0..2 * R {
342                            line[k] = out[8 - R + k][l] as u16;
343                        }
344                    }
345                }
346            }
347
348            /// Wide filter process (8.8.5.3) with 2^LOG2 taps on gathered lanes (positions 8 - 2^(LOG2-1)
349            /// .. 8 + 2^(LOG2-1) - 2 are produced); the sum over j of the specification is computed as a
350            /// sliding window over the clamped indices (identical results).
351            #[inline(always)]
352            fn wide_filter<const LOG2: u32>(v: &[[T; LANES]; 16]) -> [[T; LANES]; 16] {
353                let n = (1isize << (LOG2 - 1)) - 1;
354                let s = |k: isize| &v[(k.clamp(-(n + 1), n) + 8) as usize];
355                let mut out = [[0 as T; LANES]; 16];
356                let mut sum = [0 as T; LANES];
357                for j in -n..=n {
358                    let a = s(-n + j);
359                    for l in 0..LANES {
360                        sum[l] += a[l];
361                    }
362                }
363                let round: T = 1 << (LOG2 - 1);
364                for i in -n..n {
365                    let (cur, add, sub) = (s(i), s(i + 1 + n), s(i - n));
366                    let o = &mut out[(i + 8) as usize];
367                    for l in 0..LANES {
368                        o[l] = (sum[l] + cur[l] + round) >> LOG2;
369                        sum[l] += add[l] - sub[l];
370                    }
371                }
372                out
373            }
374        }
375    };
376}
