104/// get_tile_offset (6.4.1).
105fn tile_offset(i: usize, mis: usize, log2: u32) -> usize {
106    let sbs = (mis + 7) >> 3;
107    (((i * sbs) >> log2) << 3).min(mis)
108}

158    /// Draft mode for reduced-resolution playback (off by default): frames that no other frame
159    /// can reference (`refresh_frame_flags` 0) skip the loop filter and come out flagged
160    /// [`Picture::draft`]. Every other picture is unchanged: the loop filter changes neither the
161    /// mode info, segmentation map nor probabilities that later frames read.

163        self.draft = on;

166    /// Decode one chunk (a frame or a superframe, as stored in one IVF / WebM / MP4 sample).
167    /// Returns the pictures finished so far, in order; each carries the `pts` of the chunk that
168    /// showed it. With frame threads a picture may come out of a later call or [`Self::flush`].
169    pub fn decode(&mut self, data: &[u8], pts: i64) -> Result<Vec<Picture>> {
170        let frames = split_superframe(data);
171        if frames.len() > 1 {
172            self.stats.superframes += 1;
173        }
174        let queued = self.pending.len();
175        for f in frames {
176            if f.is_empty() {
177                continue;
178            }
179            if let Err(e) = self.decode_frame(f, pts) {
180                // A failed sample outputs nothing: drop what its earlier subframes queued so a
181                // later drain does not expand each repeat into its own picture.
182                self.pending.truncate(queued);
183                return Err(e);
184            }
185        }
186        Ok(self.drain(false))
187    }
188
189    /// End of stream: every picture still in flight.
190    pub fn flush(&mut self) -> Vec<Picture> {
191        let out = self.drain(true);
192        #[cfg(not(target_arch = "wasm32"))]
193        for j in self.jobs.drain(..) {
194            let _ = j.join();
195        }
196        out
197    }

223    /// Pictures at the front of the queue that are finished (`all`: wait for every one). The
224    /// queue is kept short: beyond a few pictures in flight the oldest is waited for.
225    fn drain(&mut self, all: bool) -> Vec<Picture> {
226        let max_lag = (self.threads / 2).clamp(2, 6);
227        let mut out = Vec::new();
228        loop {
229            let must = all || self.pending.len() > max_lag;
230            let Some(front) = self.pending.front_mut() else { break };
231            let pic = match front {
232                Pending::Ready(_) => match self.pending.pop_front() {
233                    Some(Pending::Ready(p)) => Some(p),
234                    _ => None,
235                },
236                #[cfg(not(target_arch = "wasm32"))]
237                Pending::Job(rx) => {
238                    let p = if must {
239                        rx.recv().ok()
240                    } else {
241                        match rx.try_recv() {
242                            Ok(p) => Some(p),
243                            Err(std::sync::mpsc::TryRecvError::Empty) => break,
244                            Err(std::sync::mpsc::TryRecvError::Disconnected) => None,
245                        }
246                    };
247                    self.pending.pop_front();
248                    p
249                }
250                Pending::Existing(f, pts, draft) => {
251                    if !must && !f.is_complete() {
252                        break;
253                    }
254                    let p = picture_of(f, *pts, *draft);
255                    self.pending.pop_front();
256                    Some(p)
257                }
258            };
259            out.extend(pic);
260        }
261        out
262    }

264    fn decode_frame(&mut self, data: &[u8], pts: i64) -> Result<()> {
265        let refs: [Option<RefInfo>; 8] =
266            std::array::from_fn(|i| self.slots[i].as_ref().map(|f| RefInfo { width: f.info.width, height: f.info.height }));
267        let mut h = parse_uncompressed(data, &mut self.st, &refs)?;
268        if h.show_existing_frame {
269            self.stats.show_existing += 1;
270            let f = self.slots[h.frame_to_show_map_idx as usize]
271                .clone()
272                .ok_or_else(|| Error::MissingReference(format!("show_existing_frame of empty slot {}", h.frame_to_show_map_idx)))?;
273            self.stats.shown += 1;
274            self.pending.push_back(Pending::Existing(f, pts, false));
275            return Ok(());
276        }
277        ensure!(h.width <= 16384 && h.height <= 16384, "frame size {}x{} too large", h.width, h.height);
278        ensure!(
279            u64::from(h.width).checked_mul(u64::from(h.height)).is_some_and(|n| n <= MAX_LUMA_SAMPLES),
280            "frame size {}x{} exceeds the decoder memory limit",
281            h.width,
282            h.height
283        );
284        // Eight reference slots, the new frame and those still decoding, on the 64-sample grid.
285        ensure!(
286            store_fits(h.width, h.height, &h.color, 8 + 1 + self.threads + 2, self.memory_budget),
287            "frame size {}x{} exceeds the decoder memory budget",
288            h.width,
289            h.height
290        );
291        let (mi_rows, mi_cols) = (h.mi_rows as usize, h.mi_cols as usize);
292        // compute_image_size semantics (7.2.6).
293        let size = (h.width, h.height);
294        let size_changed = self.last_size != Some(size);
295        if size_changed && self.last_size.is_some() {
296            self.stats.size_changes += 1;
297        }
298        if self.st.reset_segment_map || size_changed || self.prev_seg_ids.len() != mi_rows * mi_cols {
299            self.prev_seg_ids = vec![0; mi_rows * mi_cols];
300            self.st.reset_segment_map = false;
301        }
302        let use_prev = !size_changed
303            && self.last_show_frame
304            && !h.error_resilient_mode
305            && !h.frame_is_intra
306            && self.prev_mi.as_ref().is_some_and(|m| m.rows == mi_rows && m.cols == mi_cols);
307        self.last_size = Some(size);
308        self.last_show_frame = h.show_frame;
309
310        let off = h.uncompressed_size;
311        let hs = h.header_size_in_bytes as usize;
312        ensure!(off + hs <= data.len(), "compressed header exceeds frame data");
313        ensure!(off + hs < data.len(), "frame has no tile data");
314        let mut fc = self.st.contexts[h.frame_context_idx as usize].clone();
315        parse_compressed(&data[off..off + hs], &mut h, &mut fc)?;
316        self.count_frame(&h);
317
318        // References (8.5.2.3).
319        let mut refs: [Option<RefUse>; 3] = [None, None, None];
320        let ref_frames: Vec<Option<Arc<Frame>>> =
321            (0..3).map(|i| if h.frame_is_intra { None } else { self.slots[h.ref_frame_idx[i] as usize].clone() }).collect();
322        if !h.frame_is_intra {
323            for i in 0..3 {
324                let Some(rf) = &ref_frames[i] else {
325                    return Err(Error::MissingReference(format!("reference slot {} is empty", h.ref_frame_idx[i])));
326                };
327                let ri = &rf.info;
328                ensure!(
329                    ri.bit_depth == h.color.bit_depth && ri.ss_x == h.color.subsampling_x && ri.ss_y == h.color.subsampling_y,
330                    "reference frame format differs from the current frame"
331                );
332                let (rw, rh) = (ri.width as i64, ri.height as i64);
333                let (w, hh) = (h.width as i64, h.height as i64);
334                if 2 * w >= rw && 2 * hh >= rh && w <= 16 * rw && hh <= 16 * rh {
335                    let x_scale = ((rw << 14) / w) as i32;
336                    let y_scale = ((rh << 14) / hh) as i32;
337                    refs[i] = Some(RefUse { frame: rf, x_scale, y_scale, x_step: (16 * x_scale) >> 14, y_step: (16 * y_scale) >> 14 });
338                }
339            }
340        }
341        // Quantizers per segment (8.6.1).
342        let bd_idx = ((h.color.bit_depth - 8) >> 1) as usize;
343        let seg = self.st.seg.clone();
344        let mut seg_q = [[[0i32; 2]; 2]; 8];
345        for (s, q) in seg_q.iter_mut().enumerate() {
346            let qindex = if seg.feature_active(s as u8, SEG_LVL_ALT_Q) {
347                let d = seg.feature_data[s][SEG_LVL_ALT_Q] as i32;
348                (if seg.abs_or_delta_update { d } else { h.base_q_idx as i32 + d }).clamp(0, 255)
349            } else {
350                h.base_q_idx as i32
351            };
352            let dc = |b: i32| DC_QLOOKUP[bd_idx * 256 + b.clamp(0, 255) as usize];
353            let ac = |b: i32| AC_QLOOKUP[bd_idx * 256 + b.clamp(0, 255) as usize];
354            q[0] = [dc(qindex + h.delta_q_y_dc as i32), ac(qindex)];
355            q[1] = [dc(qindex + h.delta_q_uv_dc as i32), ac(qindex + h.delta_q_uv_ac as i32)];
356        }
357        // Tile layout and data (6.4).
358        let tile_cols = 1usize << h.tile_cols_log2;
359        let tile_rows = 1usize << h.tile_rows_log2;
360        let mut pos = off + hs;
361        let mut tile_data: Vec<Vec<&[u8]>> = vec![Vec::with_capacity(tile_rows); tile_cols];
362        for tr in 0..tile_rows {
363            for (tc, td) in tile_data.iter_mut().enumerate() {
364                let last = tr == tile_rows - 1 && tc == tile_cols - 1;
365                let sz = if last {
366                    data.len() - pos
367                } else {
368                    let Some(&b) = data.get(pos..).and_then(|d| d.first_chunk::<4>()) else {
369                        return Err(Error::Invalid("tile size beyond frame data".into()));
370                    };
371                    let s = u32::from_be_bytes(b) as usize;
372                    pos += 4;
373                    ensure!(s <= data.len() - pos, "tile size {s} beyond frame data");
374                    s
375                };
376                ensure!(sz > 0, "empty tile payload");
377                td.push(&data[pos..pos + sz]);
378                pos += sz;
379            }
380        }
381        let counting = !h.error_resilient_mode && !h.frame_parallel_decoding_mode;
382        let prev_mi = if use_prev { self.prev_mi.clone() } else { None };
383        let shared = FrameShared {
384            h: &h,
385            fc: &fc,
386            seg: &seg,
387            prev_seg_ids: &self.prev_seg_ids,
388            prev_mi: prev_mi.as_deref(),
389            refs,
390            seg_q,
391            counting,
392            pools: &self.bufs,
393        };
394        let col_bounds: Vec<(usize, usize)> =
395            (0..tile_cols).map(|i| (tile_offset(i, mi_cols, h.tile_cols_log2), tile_offset(i + 1, mi_cols, h.tile_cols_log2))).collect();
396        let row_bounds: Vec<(usize, usize)> =
397            (0..tile_rows).map(|i| (tile_offset(i, mi_rows, h.tile_rows_log2), tile_offset(i + 1, mi_rows, h.tile_rows_log2))).collect();
398        let decode_col = |ci: usize| -> (Strip, Option<Error>) {
399            let (cs, ce) = col_bounds[ci];
400            let mut td = TileDecoder::new(&shared, cs, ce);
401            let mut err = None;
402            if ce > cs {
403                for (ri, &(rs, re)) in row_bounds.iter().enumerate() {
404                    if !td.decode_tile(tile_data[ci][ri], rs, re) {
405                        err = Some(Error::Invalid("tile data exhausted".into()));
406                        break;
407                    }
408                }
409            }
410            let e = td.error.take().or(err);
411            (td.strip, e)
412        };
413        let results: Vec<(Strip, Option<Error>)> = self.run_parallel(tile_cols, &decode_col);
414        let mut counts = Counts::default();
415        let mut strips = Vec::with_capacity(results.len());
416        let mut error = None;
417        for (s, e) in results {
418            if let Some(e) = e {
419                error.get_or_insert(e);
420            }
421            if counting {
422                counts.add(&s.counts);
423            }
424            self.stats.compound_blocks += s.compound_blocks;
425            self.stats.scaled_ref_blocks += s.scaled_blocks;
426            self.stats.intra_blocks += s.intra_blocks;
427            self.stats.inter_blocks += s.inter_blocks;
428            strips.push(s);
429        }
430        if let Some(e) = error {
431            recycle_strips(strips, &self.bufs);
432            return Err(e);
433        }
434        // Release the references before the slots are refreshed, so that evicted frames can be
435        // recycled.
436        drop(ref_frames);
437        drop(prev_mi);
438        let (mi, seg_ids) = assemble_mi(&mut strips, mi_rows, mi_cols, &self.bufs);
439        let mi = Arc::new(mi);
440        let frame = Arc::new(Frame::new(
441            FrameInfo {
442                width: h.width,
443                height: h.height,
444                ss_x: h.color.subsampling_x,
445                ss_y: h.color.subsampling_y,
446                bit_depth: h.color.bit_depth,
447                color_space: h.color.color_space,
448                color_range: h.color.color_range,
449                render_width: h.render_width,
450                render_height: h.render_height,
451                key: h.frame_type == KEY_FRAME,
452                intra_only: h.intra_only,
453            },
454            self.bufs.clone(),
455        ));
456        // Loop filter (8.8), skipped in draft mode for frames nothing can reference.
457        let draft = self.draft && h.refresh_frame_flags == 0 && h.show_frame && self.st.lf.level > 0;
458        self.stats.draft_frames += draft as u64;
459        let lf = (self.st.lf.level > 0 && !draft)
460            .then(|| LfFrame::new(&self.st.lf, &seg, mi_rows, mi_cols, h.color.subsampling_x, h.color.subsampling_y, h.color.bit_depth));
461        let job = PostJob {
462            frame: frame.clone(),
463            strips,
464            mi: mi.clone(),
465            lf,
466            out: h.show_frame.then_some((pts, draft)),
467            pools: self.bufs.clone(),
468            pool: self.pool.clone(),
469        };
470        self.run_post(job);
471        // refresh_probs (6.1.2).
472        if counting {
473            let saved = &self.st.contexts[h.frame_context_idx as usize];
474            let mut a = saved.clone();
475            a.tx = fc.tx;
476            a.skip = fc.skip;
477            adapt_coef_probs(&mut a, &counts, h.frame_is_intra, self.st.last_frame_type == KEY_FRAME);
478            if !h.frame_is_intra {
479                a.tx = saved.tx;
480                a.skip = saved.skip;
481                adapt_noncoef_probs(&mut a, &counts, h.interp_filter == SWITCHABLE, h.tx_mode == TX_MODE_SELECT, h.allow_high_precision_mv);
482            }
483            fc = a;
484        }
485        if h.refresh_frame_context {
486            self.st.contexts[h.frame_context_idx as usize] = fc;
487        }
488        if seg.enabled && seg.update_map {
489            self.prev_seg_ids = seg_ids;
490        }
491        for i in 0..8 {
492            if (h.refresh_frame_flags >> i) & 1 == 1 {
493                self.slots[i] = Some(frame.clone());
494            }
495        }
496        if let Some(old) = self.prev_mi.replace(mi)
497            && let Ok(old) = Arc::try_unwrap(old)
498        {
499            self.bufs.mi.put(old.mi);
500        }
501        if h.show_frame {
502            self.stats.shown += 1;
503        } else {
504            self.stats.hidden += 1;
505        }
506        Ok(())
507    }

509    /// Run the post stage of a frame: on its own thread with frame threads (later frames wait
510    /// per band for the samples they read), else right here.
511    fn run_post(&mut self, job: PostJob) {
512        #[cfg(not(target_arch = "wasm32"))]
513        let job = if self.frame_threads() {
514            // Bound the post stages in flight (each is one thread).
515            while self.jobs.front().is_some_and(|j| j.is_finished()) {
516                let _ = self.jobs.pop_front().map(|j| j.join());
517            }
518            let max_jobs = (self.threads / 3).clamp(2, 6);
519            while self.jobs.len() >= max_jobs {
520                let _ = self.jobs.pop_front().map(|j| j.join());
521            }
522            let shown = job.out.is_some();
523            let (tx, rx) = std::sync::mpsc::channel();
524            // The job reaches the thread over a channel, so it is still here if no thread can
525            // be spawned and then runs inline below.
526            let (job_tx, job_rx) = std::sync::mpsc::channel::<PostJob>();
527            let spawned = std::thread::Builder::new().name("vp9-post".into()).spawn(move || {
528                if let Ok(job) = job_rx.recv()
529                    && let Some(p) = job.run()
530                {
531                    let _ = tx.send(p);
532                }
533            });
534            match spawned {
535                Ok(handle) => match job_tx.send(job) {
536                    Ok(()) => {
537                        self.jobs.push_back(handle);
538                        if shown {
539                            self.pending.push_back(Pending::Job(rx));
540                        }
541                        return;
542                    }
543                    // The thread is gone already: take the job back and run it here.
544                    Err(std::sync::mpsc::SendError(back)) => back,
545                },
546                // No thread available: run it here.
547                Err(_) => job,
548            }
549        } else {
550            job
551        };
552        if let Some(p) = job.run() {
553            self.pending.push_back(Pending::Ready(p));
554        }
555    }

557    fn count_frame(&mut self, h: &FrameHeader) {
558        let s = &mut self.stats;
559        s.frames += 1;
560        if h.frame_type == KEY_FRAME {
561            s.key_frames += 1;
562        } else if h.intra_only {
563            s.intra_only += 1;
564        } else {
565            s.inter_frames += 1;
566        }
567        s.error_resilient += h.error_resilient_mode as u64;
568        s.no_backward_adaptation += (h.error_resilient_mode || h.frame_parallel_decoding_mode) as u64;
569        s.lossless_frames += h.lossless as u64;
570        s.max_tile_cols = s.max_tile_cols.max(1 << h.tile_cols_log2);
571        s.max_tile_rows = s.max_tile_rows.max(1 << h.tile_rows_log2);
572        s.segmentation_frames += self.st.seg.enabled as u64;
573        s.tx_select_frames += (h.tx_mode == TX_MODE_SELECT) as u64;
574        s.switchable_interp_frames += (!h.frame_is_intra && h.interp_filter == SWITCHABLE) as u64;
575        s.high_precision_mv_frames += (!h.frame_is_intra && h.allow_high_precision_mv) as u64;
576        s.compound_frames += (!h.frame_is_intra && h.reference_mode != 0) as u64;
577        s.bit_depths[((h.color.bit_depth - 8) >> 1) as usize] += 1;
578        s.profiles[h.profile as usize] += 1;
579    }

582    fn run_parallel<T: Send>(&self, n: usize, f: &(dyn Fn(usize) -> T + Sync)) -> Vec<T> {
583        if let Some(pool) = &self.pool
584            && n > 1
585        {
586            use rayon::prelude::*;
587            return pool.install(|| (0..n).into_par_iter().map(f).collect());
588        }
589        (0..n).map(f).collect()
590    }

612/// The post stage of one frame: build each band from the tile column strips, loop filter it,
613/// publish it once no later filtering can change it, and copy the frame into the output picture.
614///
615/// The specification filters superblock by superblock in raster order (8.8.1). Superblock (r, c)
616/// changes samples of (r, c - 1) (the vertical edges on its left) and of (r - 1, c) (the
617/// horizontal edges on its top reach 7 rows up), and the last superblock before it to change the
618/// bottom rows of (r - 1, c) is (r - 1, c + 1). So row r may filter superblock c once row r - 1
619/// has filtered superblock c + 1: rows run as a wavefront, each two superblocks behind the row
620/// above, with the specification's result. Every row filters its own band: before superblock
621/// c it copies the bottom rows of band r - 1 in that superblock column (shared by row r - 1 once
622/// final for it) into its scratch rows, and once done it writes them back into band r - 1, which
623/// is then final and published. The post thread takes rows in order and helpers on the
624/// decoder's pool join in; with one thread the rows run one after the other, as before.
625struct PostJob {
626    frame: Arc<Frame>,
627    strips: Vec<Strip>,
628    mi: Arc<MiGrid>,
629    lf: Option<LfFrame>,
630    /// (pts, draft) when the frame is shown.
631    out: Option<(i64, bool)>,
632    pools: Arc<Pools>,
633    pool: Option<Arc<rayon::ThreadPool>>,
634}
635
636/// Publishes placeholder bands if a post job dies or fails, so that waiting frames cannot hang.
637struct PublishGuard<'a>(&'a Frame);
638
639impl Drop for PublishGuard<'_> {
640    fn drop(&mut self) {
641        for r in 0..self.0.sb_rows() {
642            if !self.0.is_published(r) {
643                self.0.publish(r, self.0.new_band());
644            }
645        }
646    }
647}
648
649impl PostJob {
650    fn run(self) -> Option<Picture> {
651        let f = &*self.frame;
652        let _guard = PublishGuard(f);
653        let mut pic = self.out.map(|(pts, draft)| PictureBuilder::new(f, pts, draft));
654        let Some(lf) = self.lf else {
655            // Without the loop filter every band is final as built.
656            for r in 0..f.sb_rows() {
657                let band = build_band(f, &self.strips, r);
658                if let Some(pic) = &mut pic {
659                    pic.add(f, r, &band);
660                }
661                f.publish(r, band);
662            }
663            recycle_strip_planes(self.strips, &self.pools);
664            return pic.map(PictureBuilder::finish);
665        };
666        let wave = Arc::new(Wave::new(self.frame.clone(), self.strips, self.mi, lf, self.pools, pic));
667        if let Some(pool) = &self.pool {
668            // Rows that can be filtered at the same time, the post thread being one of them.
669            let rows = wave.sb_rows.min(wave.sb_cols.div_ceil(2)).min(pool.current_num_threads() + 1);
670            for _ in 1..rows {
671                let w = wave.clone();
672                pool.spawn(move || w.work());
673            }
674        }
675        wave.work();
676        if !wave.wait_done() {
677            return None;
678        }
679        // Every row is done, so no helper reads the strips any more.
680        let strips = std::mem::take(&mut *wave.strips.write().unwrap_or_else(std::sync::PoisonError::into_inner));
681        recycle_strip_planes(strips, &wave.pools);
682        lock(&wave.pic).take().map(PictureBuilder::finish)
683    }
684}
685
686fn recycle_strip_planes(strips: Vec<Strip>, pools: &Pools) {
687    for s in strips {
688        for p in s.planes {
689            pools.samples.put(p);
690        }
691    }
692}

694/// Band `r` of a frame from the tile column strips, before the loop filter.
695fn build_band(f: &Frame, strips: &[Strip], r: usize) -> Band {
696    let mut band = f.new_band();
697    for p in 0..3 {
698        let (fs, bh) = (f.strides[p], f.band_h[p]);
699        let dst = &mut band.planes[p];
700        for s in strips {
701            let ss = s.strides[p];
702            let x0 = s.x_off[p];
703            let w = ss.min(fs.saturating_sub(x0));
704            for k in 0..bh {
705                let y = r * bh + k;
706                let d = (BAND_PAD + k) * fs + x0;
707                if let (Some(d), Some(s)) = (dst.get_mut(d..d + w), s.planes[p].get(y * ss..y * ss + w)) {
708                    d.copy_from_slice(s);
709                }
710            }
711        }
712    }
713    band
714}

716/// Progress of one superblock row of the wavefront.
717struct RowProgress {
718    /// Superblocks filtered; `sb_cols` once the row is done (and band r - 1 published).
719    done: AtomicUsize,
720    /// Threads waiting on `cv` (so that progress without a waiter costs no wake-up call).
721    waiters: AtomicUsize,
722    lock: Mutex<()>,
723    cv: Condvar,
724    /// The bottom `BAND_PAD` rows of the band of each plane, shared superblock column by column
725    /// once final for this row, for the row below.
726    bottom: Mutex<[Vec<u16>; 3]>,
727    /// The band once the row is done, until the row below has written its scratch rows back.
728    band: Mutex<Option<Band>>,
729}
730
731/// The shared state of a frame's loop filter wavefront (see [`PostJob`]). No lock is held while
732/// a superblock is filtered: each row filters its own band and shares only copies.
733struct Wave {
734    frame: Arc<Frame>,
735    strips: RwLock<Vec<Strip>>,
736    mi: Arc<MiGrid>,
737    lf: LfFrame,
738    pools: Arc<Pools>,
739    sb_rows: usize,
740    sb_cols: usize,
741    rows: Vec<RowProgress>,
742    /// The next row to take.
743    next: AtomicUsize,
744    failed: AtomicBool,
745    /// The output picture (shown frames), filled band by band as bands are published.
746    pic: Mutex<Option<PictureBuilder>>,
747}
748
749fn lock<T>(m: &Mutex<T>) -> MutexGuard<'_, T> {
750    m.lock().unwrap_or_else(std::sync::PoisonError::into_inner)
751}
752
753impl Wave {
754    fn new(frame: Arc<Frame>, strips: Vec<Strip>, mi: Arc<MiGrid>, lf: LfFrame, pools: Arc<Pools>, pic: Option<PictureBuilder>) -> Wave {
755        let sb_rows = frame.sb_rows();
756        let sb_cols = lf.mi_cols.div_ceil(8);
757        let rows = (0..sb_rows)
758            .map(|_| RowProgress {
759                done: AtomicUsize::new(0),
760                waiters: AtomicUsize::new(0),
761                lock: Mutex::new(()),
762                cv: Condvar::new(),
763                bottom: Mutex::new(std::array::from_fn(|p| pools.bands.take_any(BAND_PAD * frame.strides[p], 0))),
764                band: Mutex::new(None),
765            })
766            .collect();
767        Wave {
768            frame,
769            strips: RwLock::new(strips),
770            mi,
771            lf,
772            pools,
773            sb_rows,
774            sb_cols,
775            rows,
776            next: AtomicUsize::new(0),
777            failed: AtomicBool::new(false),
778            pic: Mutex::new(pic),
779        }
780    }
781
782    /// Take rows and filter them until none is left. A row is only taken by a running thread, so
783    /// a row waiting for the one above always waits for a thread that is making progress.
784    fn work(&self) {
785        while !self.failed.load(Ordering::Acquire) {
786            let r = self.next.fetch_add(1, Ordering::Relaxed);
787            if r >= self.sb_rows {
788                return;
789            }
790            let ok = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| self.row(r))).unwrap_or(false);
791            if !ok {
792                self.fail();
793                return;
794            }
795        }
796    }
797
798    /// Wait until every row is done (false if one failed).
799    fn wait_done(&self) -> bool {
800        self.sb_rows.checked_sub(1).is_none_or(|last| self.wait(last, self.sb_cols))
801    }
802
803    /// Wait until row `r` has filtered `need` superblocks (false if a row failed).
804    fn wait(&self, r: usize, need: usize) -> bool {
805        let Some(row) = self.rows.get(r) else { return false };
806        if row.done.load(Ordering::SeqCst) >= need {
807            return true;
808        }
809        let mut g = lock(&row.lock);
810        row.waiters.fetch_add(1, Ordering::SeqCst);
811        let ok = loop {
812            if row.done.load(Ordering::SeqCst) >= need {
813                break true;
814            }
815            if self.failed.load(Ordering::SeqCst) {
816                break false;
817            }
818            g = row.cv.wait(g).unwrap_or_else(std::sync::PoisonError::into_inner);
819        };
820        row.waiters.fetch_sub(1, Ordering::SeqCst);
821        ok
822    }
823
824    fn set_done(&self, r: usize, n: usize) {
825        if let Some(row) = self.rows.get(r) {
826            // A waiter registers under the lock before it checks `done`, so either it sees the
827            // new value or it is counted here and woken.
828            row.done.store(n, Ordering::SeqCst);
829            if row.waiters.load(Ordering::SeqCst) > 0 {
830                let _g = lock(&row.lock);
831                row.cv.notify_all();
832            }
833        }
834    }
835
836    /// Stop the wavefront: waiting rows give up, the post job publishes placeholders.
837    fn fail(&self) {
838        self.failed.store(true, Ordering::SeqCst);
839        for row in &self.rows {
840            let _g = lock(&row.lock);
841            row.cv.notify_all();
842        }
843    }
844
845    /// Publish final band `r` and copy it into the output picture.
846    fn publish(&self, r: usize, band: Band) {
847        if let Some(pic) = lock(&self.pic).as_mut() {
848            pic.add(&self.frame, r, &band);
849        }
850        self.frame.publish(r, band);
851    }
852
853    /// Share the bottom rows of superblock column `c` of row `r`'s band (final for this row).
854    fn share_bottom(&self, r: usize, band: &Band, c: usize) {
855        let Some(row) = self.rows.get(r) else { return };
856        let mut bottom = lock(&row.bottom);
857        for (p, dst) in bottom.iter_mut().enumerate() {
858            copy_column(&self.frame, Column { p, c, sb_cols: self.sb_cols }, (&band.planes[p], self.frame.band_h[p]), (dst, 0));
859        }
860    }
861
862    /// Build and filter row `r`; publish band r - 1 (and the last band) when done.
863    fn row(&self, r: usize) -> bool {
864        let f = &*self.frame;
865        let sb_cols = self.sb_cols;
866        let mut band = build_band(f, &self.strips.read().unwrap_or_else(std::sync::PoisonError::into_inner), r);
867        let above = r.checked_sub(1).and_then(|a| self.rows.get(a));
868        for c in 0..sb_cols {
869            if let Some(above) = above {
870                // Superblock c + 1 of the row above is filtered: the bottom rows of its column c
871                // are final for that row and go into our scratch rows.
872                if !self.wait(r - 1, (c + 2).min(sb_cols)) {
873                    return false;
874                }
875                let bottom = lock(&above.bottom);
876                for (p, src) in bottom.iter().enumerate() {
877                    copy_column(f, Column { p, c, sb_cols }, (src, 0), (&mut band.planes[p], 0));
878                }
879            }
880            let [a, b, d] = &mut band.planes;
881            let mut views = [band_view(f, a, 0, r), band_view(f, b, 1, r), band_view(f, d, 2, r)];
882            filter_superblock(&mut views, &self.mi, &self.lf, r * 8, c * 8);
883            // Superblock c filtered the left edge of column c, so column c - 1 is final here.
884            if c > 0 {
885                self.share_bottom(r, &band, c - 1);
886            }
887            if c + 1 < sb_cols {
888                self.set_done(r, c + 1);
889            }
890        }
891        self.share_bottom(r, &band, sb_cols.saturating_sub(1));
892        if let Some(above) = above {
893            // The scratch rows now hold the final bottom rows of band r - 1.
894            let Some(mut pb) = lock(&above.band).take() else { return false };
895            for p in 0..3 {
896                let (fs, bh) = (f.strides[p], f.band_h[p]);
897                if let (Some(d), Some(s)) = (pb.planes[p].get_mut(bh * fs..(bh + BAND_PAD) * fs), band.planes[p].get(..BAND_PAD * fs)) {
898                    d.copy_from_slice(s);
899                }
900            }
901            self.publish(r - 1, pb);
902        }
903        if r + 1 == self.sb_rows {
904            self.publish(r, band);
905        } else if let Some(row) = self.rows.get(r) {
906            *lock(&row.band) = Some(band);
907        }
908        self.set_done(r, sb_cols);
909        true
910    }
911}

913impl Drop for Wave {
914    fn drop(&mut self) {
915        recycle_strip_planes(std::mem::take(self.strips.get_mut().unwrap_or_else(std::sync::PoisonError::into_inner)), &self.pools);
916        for row in &self.rows {
917            for p in std::mem::take(&mut *lock(&row.bottom)) {
918                self.pools.bands.put(p);
919            }
920            if let Some(b) = lock(&row.band).take() {
921                for p in b.planes {
922                    self.pools.bands.put(p);
923                }
924            }
925        }
926    }
927}

929/// Superblock column `c` of plane `p` in a frame `sb_cols` superblocks wide.
930#[derive(Clone, Copy)]
931struct Column {
932    p: usize,
933    c: usize,
934    sb_cols: usize,
935}
936
937/// Copy `BAND_PAD` rows of superblock column `c` of plane `p` (the last column takes the rest of
938/// the stride) from row `src_row` of `src` to row `dst_row` of `dst` (each given as `(buffer, row)`), both `strides[p]` wide.
939fn copy_column(f: &Frame, col: Column, (src, src_row): (&[u16], usize), (dst, dst_row): (&mut [u16], usize)) {
940    let Column { p, c, sb_cols } = col;
941    let fs = f.strides[p];
942    let cw = fs / sb_cols.max(1);
943    let x0 = c * cw;
944    let x1 = if c + 1 >= sb_cols { fs } else { x0 + cw };
945    for k in 0..BAND_PAD {
946        let (s, d) = ((src_row + k) * fs, (dst_row + k) * fs);
947        if let (Some(d), Some(s)) = (dst.get_mut(d + x0..d + x1), src.get(s + x0..s + x1)) {
948            d.copy_from_slice(s);
949        }
950    }
951}
952
953/// The loop filter's view of band `r` of plane `p`: the band's rows plus (below row 0) the
954/// scratch rows holding the bottom of band r - 1.
955fn band_view<'a>(f: &Frame, d: &'a mut [u16], p: usize, r: usize) -> PlaneView<'a> {
956    let (fs, bh) = (f.strides[p], f.band_h[p]);
957    if r == 0 {
958        PlaneView { data: &mut d[BAND_PAD * fs..], stride: fs, ox: 0, oy: 0 }
959    } else {
960        PlaneView { data: d, stride: fs, ox: 0, oy: r * bh - BAND_PAD }
961    }
962}

964/// Mode info and segment ids of the whole frame from the tile column strips (whose mode info
965/// buffers go back to the pool).
966fn assemble_mi(strips: &mut [Strip], mi_rows: usize, mi_cols: usize, pools: &Pools) -> (MiGrid, Vec<u8>) {
967    if strips.len() == 1 {
968        let s = &mut strips[0];
969        return (MiGrid { cols: mi_cols, rows: mi_rows, mi: std::mem::take(&mut s.mi) }, std::mem::take(&mut s.seg_ids));
970    }
971    // Every position is copied from a strip below.
972    let mut mi = pools.mi.take_any(mi_rows * mi_cols, MiInfo::default());
973    let mut seg = vec![0u8; mi_rows * mi_cols];
974    for s in strips.iter_mut() {
975        let mw = s.mi_col_end - s.mi_col_start;
976        for r in 0..mi_rows {
977            mi[r * mi_cols + s.mi_col_start..r * mi_cols + s.mi_col_end].copy_from_slice(&s.mi[r * mw..r * mw + mw]);
978            seg[r * mi_cols + s.mi_col_start..r * mi_cols + s.mi_col_end].copy_from_slice(&s.seg_ids[r * mw..r * mw + mw]);
979        }
980        pools.mi.put(std::mem::take(&mut s.mi));
981    }
982    (MiGrid { cols: mi_cols, rows: mi_rows, mi }, seg)
983}

985/// Builds the cropped output picture band by band (narrowing 8-bit samples).
986struct PictureBuilder {
987    planes: [Plane; 3],
988    info: FrameInfo,
989    vis: [(usize, usize); 3],
990    pts: i64,
991    draft: bool,
992}
993
994impl PictureBuilder {
995    fn new(f: &Frame, pts: i64, draft: bool) -> Self {
996        let planes = f.vis.map(|(w, h)| if f.info.bit_depth == 8 { Plane::U8(vec![0; w * h]) } else { Plane::U16(vec![0; w * h]) });
997        PictureBuilder { planes, info: f.info, vis: f.vis, pts, draft }
998    }
999
1000    fn add(&mut self, f: &Frame, r: usize, band: &Band) {
1001        for p in 0..3 {
1002            let (w, h) = f.vis[p];
1003            let (fs, bh) = (f.strides[p], f.band_h[p]);
1004            let y0 = r * bh;
1005            for y in y0..(y0 + bh).min(h) {
1006                let src = &band.planes[p][(y - y0 + BAND_PAD) * fs..][..w];
1007                match &mut self.planes[p] {
1008                    Plane::U8(d) => {
1009                        for (d, &s) in d[y * w..y * w + w].iter_mut().zip(src) {
1010                            *d = s as u8;
1011                        }
1012                    }
1013                    Plane::U16(d) => d[y * w..y * w + w].copy_from_slice(src),
1014                }
1015            }
1016        }
1017    }
1018
1019    fn finish(self) -> Picture {
1020        let i = self.info;
1021        let [y, u, v] = self.planes;
1022        Picture {
1023            width: i.width,
1024            height: i.height,
1025            chroma_width: self.vis[1].0 as u32,
1026            chroma_height: self.vis[1].1 as u32,
1027            bit_depth: i.bit_depth as u32,
1028            subsampling_x: i.ss_x,
1029            subsampling_y: i.ss_y,
1030            y,
1031            u,
1032            v,
1033            y_stride: self.vis[0].0,
1034            uv_stride: self.vis[1].0,
1035            pts: self.pts,
1036            key: i.key,
1037            intra_only: i.intra_only,
1038            color: ColorInfo { color_space: i.color_space, full_range: i.color_range },
1039            render_width: i.render_width,
1040            render_height: i.render_height,
1041            draft: self.draft,
1042        }
1043    }
1044}

1046/// The output picture of a complete (or completing) frame (show_existing_frame).
1047fn picture_of(f: &Frame, pts: i64, draft: bool) -> Picture {
1048    let mut b = PictureBuilder::new(f, pts, draft);
1049    for r in 0..f.sb_rows() {
1050        b.add(f, r, f.band(r));
1051    }
1052    b.finish()
1053}
