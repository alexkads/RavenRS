36    pub temporal_id: u8,

39impl NalHeader {
40    pub fn parse(b: &[u8]) -> Result<NalHeader> {
41        ensure!(b.len() >= 2, "NAL unit too short");
42        ensure!(b[0] & 0x80 == 0, "forbidden_zero_bit set");
43        let tid = b[1] & 7;
44        ensure!(tid != 0, "nuh_temporal_id_plus1 is 0");
45        Ok(NalHeader { nal_type: (b[0] >> 1) & 0x3f, layer_id: ((b[0] & 1) << 5) | (b[1] >> 3), temporal_id: tid - 1 })
46    }
47    pub fn is_vcl(&self) -> bool {
48        self.nal_type < 32
49    }
50    pub fn is_irap(&self) -> bool {
51        (nal_type::BLA_W_LP..=nal_type::RSV_IRAP_23).contains(&self.nal_type)
52    }
53    pub fn is_idr(&self) -> bool {
54        self.nal_type == nal_type::IDR_W_RADL || self.nal_type == nal_type::IDR_N_LP
55    }
56    pub fn is_bla(&self) -> bool {
57        (nal_type::BLA_W_LP..=nal_type::BLA_N_LP).contains(&self.nal_type)
58    }
59    pub fn is_cra(&self) -> bool {
60        self.nal_type == nal_type::CRA
61    }
62    pub fn is_rasl(&self) -> bool {
63        self.nal_type == nal_type::RASL_N || self.nal_type == nal_type::RASL_R
64    }
65    pub fn is_radl(&self) -> bool {
66        self.nal_type == nal_type::RADL_N || self.nal_type == nal_type::RADL_R
67    }
68    /// Sub-layer non-reference picture (TRAIL_N, TSA_N, ..., RSV_VCL_N14).
69    pub fn is_sub_layer_non_ref(&self) -> bool {
70        self.nal_type <= nal_type::RSV_VCL_N14 && self.nal_type.is_multiple_of(2)
71    }
72}

138fn ceil_log2(n: u32) -> u32 {
139    if n <= 1 { 0 } else { 32 - (n - 1).leading_zeros() }
140}

152        let st = self.st_rps.s0.iter().chain(self.st_rps.s1.iter()).filter(|e| e.1).count();

157    /// Parse a slice segment header. `prev` is the header of the preceding independent slice segment
158    /// of the same picture (needed for dependent slice segments).
159    pub fn parse(
160        rbsp: &[u8],
161        nal: NalHeader,
162        lookup: impl FnOnce(u32) -> Result<(std::sync::Arc<Pps>, std::sync::Arc<Sps>)>,
163        prev: Option<&SliceHeader>,
164    ) -> Result<(SliceHeader, std::sync::Arc<Pps>, std::sync::Arc<Sps>)> {
165        let mut r = BitReader::new(rbsp);
166        let first = r.read_flag()?;
167        let no_output_of_prior_pics = if nal.is_irap() { r.read_flag()? } else { false };
168        let pps_id = r.read_ue()?;
169        ensure!(pps_id < 64, "slice_pic_parameter_set_id out of range");
170        let (pps, sps) = lookup(pps_id)?;
171        let mut dependent = false;
172        let mut segment_address = 0;
173        if !first {
174            if pps.dependent_slice_segments_enabled {
175                dependent = r.read_flag()?;
176            }
177            let pic_size = sps.pic_width_in_ctbs() * sps.pic_height_in_ctbs();
178            segment_address = r.read_bits(ceil_log2(pic_size))?;
179            ensure!(segment_address < pic_size, "slice_segment_address out of range");
180        }
181        let mut sh = if dependent {
182            let Some(p) = prev else { return invalid("dependent slice segment without a preceding slice") };
183            // entry points are the segment's own; sharing the clone's capacity let many short
184            // dependent segments each retain a copy of one large table
185            let mut s = SliceHeader { entry_points: Vec::new(), ..p.clone() };
186            s.nal = nal;
187            s
188        } else {
189            SliceHeader::with_pps_defaults(nal, first, no_output_of_prior_pics, pps_id, segment_address, &pps)
190        };
191        sh.first_slice_segment_in_pic = first;
192        sh.no_output_of_prior_pics = no_output_of_prior_pics;
193        sh.pps_id = pps_id;
194        sh.dependent = dependent;
195        sh.segment_address = segment_address;
196        if !dependent {
197            parse_independent_fields(&mut r, &mut sh, nal, &pps, &sps)?;
198        }
199        parse_entry_points_and_extension(&mut r, &mut sh, &pps, &sps)?;
200        // byte_alignment(): alignment_bit_equal_to_one then zeros
201        ensure!(r.read_flag()?, "missing alignment_bit_equal_to_one");
202        r.byte_align();
203        sh.data_offset = r.position() / 8;
204        ensure!(sh.data_offset <= rbsp.len(), "slice header runs past the NAL unit");
205        Ok((sh, pps, sps))
206    }
207}

250/// Fields present only in independent slice segments, up to the entry points.
251fn parse_independent_fields(r: &mut BitReader, sh: &mut SliceHeader, nal: NalHeader, pps: &Pps, sps: &Sps) -> Result<()> {
252    r.skip(pps.num_extra_slice_header_bits as usize)?;
253    let st = r.read_ue()?;
254    sh.slice_type = match st {
255        0 => SliceType::B,
256        1 => SliceType::P,
257        2 => SliceType::I,
258        _ => return invalid(format!("slice_type {st}")),
259    };
260    if nal.is_irap() {
261        ensure!(sh.slice_type == SliceType::I, "IRAP picture with inter slice");
262    }
263    if pps.output_flag_present {
264        sh.pic_output = r.read_flag()?;
265    }
266    if sps.separate_colour_plane {
267        r.skip(2)?;
268    }
269    if !nal.is_idr() {
270        parse_reference_pictures(r, sh, sps)?;
271    }
272    if sps.sao {
273        sh.sao_luma = r.read_flag()?;
274        if sps.chroma_array_type() != 0 {
275            sh.sao_chroma = r.read_flag()?;
276        }
277    }
278    if sh.slice_type != SliceType::I {
279        parse_inter_slice_params(r, sh, pps, sps)?;
280    }
281    parse_qp_and_loop_filter(r, sh, pps, sps)
282}

284/// POC LSB, short-term and long-term reference picture sets and the temporal MVP flag.
285fn parse_reference_pictures(r: &mut BitReader, sh: &mut SliceHeader, sps: &Sps) -> Result<()> {
286    sh.poc_lsb = r.read_bits(sps.log2_max_poc_lsb)?;
287    let from_sps = r.read_flag()?;
288    if !from_sps {
289        let start = r.position();
290        sh.st_rps = StRps::parse(r, sps.st_rps.len(), &sps.st_rps, sps.st_rps.len())?;
291        sh.st_rps_bits = r.position() - start;
292    } else {
293        ensure!(!sps.st_rps.is_empty(), "no short-term RPS in the SPS");
294        let idx = if sps.st_rps.len() > 1 { r.read_bits(ceil_log2(sps.st_rps.len() as u32))? as usize } else { 0 };
295        ensure!(idx < sps.st_rps.len(), "short_term_ref_pic_set_idx out of range");
296        sh.st_rps = sps.st_rps[idx].clone();
297    }
298    if sps.long_term_refs_present {
299        parse_long_term_refs(r, sh, sps)?;
300    }
301    if sps.temporal_mvp {
302        sh.temporal_mvp = r.read_flag()?;
303    }
304    Ok(())
305}

307/// Long-term reference pictures (num_long_term_sps .. delta_poc_msb_cycle_lt).
308fn parse_long_term_refs(r: &mut BitReader, sh: &mut SliceHeader, sps: &Sps) -> Result<()> {
309    let num_lt_sps = if !sps.lt_ref_pics.is_empty() { r.read_ue()? } else { 0 };
310    ensure!(num_lt_sps as usize <= sps.lt_ref_pics.len(), "num_long_term_sps out of range");
311    let num_lt_pics = r.read_ue()?;
312    ensure!(num_lt_sps.saturating_add(num_lt_pics) <= 32, "too many long-term pictures");
313    let mut msb_acc = 0u32;
314    for i in 0..num_lt_sps + num_lt_pics {
315        let (lsb, used) = if i < num_lt_sps {
316            let idx = if sps.lt_ref_pics.len() > 1 { r.read_bits(ceil_log2(sps.lt_ref_pics.len() as u32))? } else { 0 };
317            ensure!((idx as usize) < sps.lt_ref_pics.len(), "lt_idx_sps out of range");
318            sps.lt_ref_pics[idx as usize]
319        } else {
320            (r.read_bits(sps.log2_max_poc_lsb)?, r.read_flag()?)
321        };
322        let msb_present = r.read_flag()?;
323        let cycle = if msb_present { r.read_ue()? } else { 0 };
324        // DeltaPocMsbCycleLt accumulates within each of the two groups (7-52)
325        if i == 0 || i == num_lt_sps {
326            msb_acc = cycle;
327        } else {
328            msb_acc = msb_acc.wrapping_add(cycle);
329        }
330        sh.lt.push((lsb, used, msb_present, msb_acc));
331    }
332    Ok(())
333}

335/// ref_pic_lists_modification: list_entry_lX for each list.
336fn parse_list_modification(r: &mut BitReader, sh: &mut SliceHeader, total: u32) -> Result<()> {
337    let bits = ceil_log2(total);
338    for l in 0..if sh.is_b() { 2 } else { 1 } {
339        if r.read_flag()? {
340            let mut e = Vec::new();
341            for _ in 0..sh.num_ref_idx[l] {
342                let v = r.read_bits(bits)?;
343                ensure!(v < total, "list_entry out of range");
344                e.push(v);
345            }
346            sh.list_entry[l] = Some(e);
347        }
348    }
349    Ok(())
350}

352/// Inter-slice parameters: active reference counts, list modification, collocated picture,
353/// weighted prediction and merge candidates.
354fn parse_inter_slice_params(r: &mut BitReader, sh: &mut SliceHeader, pps: &Pps, sps: &Sps) -> Result<()> {
355    sh.num_ref_idx = [pps.num_ref_idx_l0_default, if sh.is_b() { pps.num_ref_idx_l1_default } else { 0 }];
356    if r.read_flag()? {
357        sh.num_ref_idx[0] = r.read_ue_plus(1)?;
358        if sh.is_b() {
359            sh.num_ref_idx[1] = r.read_ue_plus(1)?;
360        }
361    }
362    ensure!(sh.num_ref_idx[0] <= 15 && sh.num_ref_idx[1] <= 15, "num_ref_idx_active out of range");
363    let total = sh.num_pic_total_curr();
364    ensure!(total > 0, "inter slice without reference pictures in the RPS");
365    if pps.lists_modification_present && total > 1 {
366        parse_list_modification(r, sh, total)?;
367    }
368    if sh.is_b() {
369        sh.mvd_l1_zero = r.read_flag()?;
370    }
371    if pps.cabac_init_present {
372        sh.cabac_init_flag = r.read_flag()?;
373    }
374    if sh.temporal_mvp {
375        if sh.is_b() {
376            sh.collocated_from_l0 = r.read_flag()?;
377        }
378        let l = if sh.collocated_from_l0 { 0 } else { 1 };
379        if sh.num_ref_idx[l] > 1 {
380            sh.collocated_ref_idx = r.read_ue()?;
381            ensure!(sh.collocated_ref_idx < sh.num_ref_idx[l], "collocated_ref_idx out of range");
382        }
383    }
384    if (pps.weighted_pred && sh.slice_type == SliceType::P) || (pps.weighted_bipred && sh.is_b()) {
385        sh.pwt = Some(parse_pwt(r, sh, sps)?);
386    }
387    let five_minus = r.read_ue()?;
388    ensure!(five_minus <= 4, "five_minus_max_num_merge_cand out of range");
389    sh.max_num_merge_cand = 5 - five_minus;
390    Ok(())
391}

393/// Slice QP delta, chroma QP offsets and deblocking / in-loop filter controls.
394fn parse_qp_and_loop_filter(r: &mut BitReader, sh: &mut SliceHeader, pps: &Pps, sps: &Sps) -> Result<()> {
395    sh.qp_delta = r.read_se()?;
396    let Some(qp) = pps.init_qp.checked_add(sh.qp_delta) else { return invalid("SliceQpY out of range") };
397    let qp_bd = 6 * (sps.bit_depth_luma as i32 - 8);
398    ensure!((-qp_bd..=51).contains(&qp), "SliceQpY {qp} out of range");
399    if pps.slice_chroma_qp_offsets_present {
400        sh.cb_qp_offset = r.read_se()?;
401        sh.cr_qp_offset = r.read_se()?;
402        ensure!((-12..=12).contains(&sh.cb_qp_offset) && (-12..=12).contains(&sh.cr_qp_offset), "slice chroma QP offset out of range");
403    }
404    let override_flag = if pps.deblocking_override_enabled { r.read_flag()? } else { false };
405    if override_flag {
406        sh.deblocking_disabled = r.read_flag()?;
407        if !sh.deblocking_disabled {
408            sh.beta_offset_div2 = r.read_se()?;
409            sh.tc_offset_div2 = r.read_se()?;
410            ensure!((-6..=6).contains(&sh.beta_offset_div2) && (-6..=6).contains(&sh.tc_offset_div2), "deblocking offsets out of range");
411        }
412    }
413    if pps.loop_filter_across_slices && (sh.sao_luma || sh.sao_chroma || !sh.deblocking_disabled) {
414        sh.loop_filter_across_slices = r.read_flag()?;
415    }
416    Ok(())
417}

419/// `n` `entry_point_offset_minus1` values (`offset_len_minus1` + 1 bits each), as offsets.
420fn read_entry_point_offsets(r: &mut BitReader, n: u32) -> Result<Vec<u32>> {
421    let len = r.read_ue_plus(1)?;
422    ensure!(len <= 32, "offset_len_minus1 out of range");
423    let mut out = Vec::new();
424    for _ in 0..n {
425        // a 32-bit field can hold u32::MAX, whose offset does not fit in 32 bits
426        let Some(offset) = r.read_bits(len)?.checked_add(1) else { return invalid("entry_point_offset_minus1 out of range") };
427        out.push(offset);
428    }
429    Ok(out)
430}

432/// Entry point offsets and the slice header extension.
433fn parse_entry_points_and_extension(r: &mut BitReader, sh: &mut SliceHeader, pps: &Pps, sps: &Sps) -> Result<()> {
434    sh.entry_points.clear();
435    if pps.tiles_enabled || pps.entropy_coding_sync {
436        let n = r.read_ue()?;
437        ensure!(n <= sps.pic_width_in_ctbs() * sps.pic_height_in_ctbs(), "num_entry_point_offsets out of range");
438        if n > 0 {
439            sh.entry_points = read_entry_point_offsets(r, n)?;
440        }
441    }
442    if pps.slice_header_extension_present {
443        let len = r.read_ue()?;
444        ensure!(len <= 256, "slice_segment_header_extension_length out of range");
445        r.skip(8 * len as usize)?;
446    }
447    Ok(())
448}

450fn parse_pwt(r: &mut BitReader, sh: &SliceHeader, sps: &Sps) -> Result<PredWeightTable> {
451    let mut t = PredWeightTable { luma_log2_denom: r.read_ue()?, ..Default::default() };
452    ensure!(t.luma_log2_denom <= 7, "luma_log2_weight_denom out of range");
453    let chroma = sps.chroma_array_type() != 0;
454    if chroma {
455        let delta = r.read_se()?;
456        ensure!((-7..=7).contains(&delta), "delta_chroma_log2_weight_denom out of range");
457        let d = t.luma_log2_denom as i32 + delta;
458        ensure!((0..=7).contains(&d), "ChromaLog2WeightDenom out of range");
459        t.chroma_log2_denom = d as u32;
460    }
461    let lists = if sh.is_b() { 2 } else { 1 };
462    for l in 0..lists {
463        let n = sh.num_ref_idx[l] as usize;
464        let mut e = vec![WpEntry::default(); n];
465        for x in e.iter_mut() {
466            x.luma_flag = r.read_flag()?;
467        }
468        if chroma {
469            for x in e.iter_mut() {
470                x.chroma_flag = r.read_flag()?;
471            }
472        }
473        for x in e.iter_mut() {
474            x.luma = (1 << t.luma_log2_denom, 0);
475            if x.luma_flag {
476                let dw = r.read_se()?;
477                let off = r.read_se()?;
478                ensure!((-128..=127).contains(&dw) && (-128..=127).contains(&off), "luma weight out of range");
479                x.luma = ((1 << t.luma_log2_denom) + dw, off);
480            }
481            x.chroma = [(1 << t.chroma_log2_denom, 0); 2];
482            if x.chroma_flag {
483                let half = 128; // wpOffsetHalfRangeC (no high_precision_offsets)
484                for c in 0..2 {
485                    let dw = r.read_se()?;
486                    let doff = r.read_se()?;
487                    ensure!((-128..=127).contains(&dw) && (-4 * half..4 * half).contains(&doff), "chroma weight out of range");
488                    let w = (1 << t.chroma_log2_denom) + dw;
489                    let off = (half - ((half * w) >> t.chroma_log2_denom) + doff).clamp(-half, half - 1);
490                    x.chroma[c] = (w, off);
491                }
492            }
493        }
494        t.l[l] = e;
495    }
496    Ok(t)
497}

594    /// An IDR slice header with one entry point of `offset_minus1` (32-bit field).
595    fn idr_slice_qp(offset_minus1: u32, qp_delta: i32) -> Vec<u8> {
596        let mut w = BitWriter::new();
597        w.write_bit(true); // first_slice_segment_in_pic_flag
598        w.write_bit(false); // no_output_of_prior_pics_flag
599        w.write_ue(0); // pps id
600        w.write_ue(2); // I slice
601        w.write_se(qp_delta); // slice_qp_delta
602        w.write_ue(1); // num_entry_point_offsets
603        w.write_ue(31); // offset_len_minus1
604        w.write_bits(offset_minus1, 32);
605        w.rbsp_trailing();
606        w.finish()
607    }

618    }
