8    /// prediction_unit(): parse, derive motion, store it and predict. Returns merge_flag.
9    pub(super) fn prediction_unit(&mut self, pb: &PredBlock, skip: bool) -> Result<bool> {
10        let PredBlock { xc, yc, xp, yp, w, h, .. } = *pb;
11        let merge = if skip { true } else { self.dd(MERGE_FLAG) == 1 };
12        let mvf = if merge {
13            let idx = if self.sh.max_num_merge_cand > 1 {
14                let mut i = 0;
15                if self.dd(MERGE_IDX) == 1 {
16                    i = 1;
17                    while i < self.sh.max_num_merge_cand - 1 && self.bypass() == 1 {
18                        i += 1;
19                    }
20                }
21                i
22            } else {
23                0
24            };
25            self.derive_merge(pb, idx as usize)
26        } else {
27            let pred_idc = if self.sh.is_b() {
28                if w + h != 12 {
29                    let depth = self.pic.blk_at(xc, yc).depth as usize;
30                    if self.dd(INTER_PRED_IDC + depth) == 1 { 2 } else { self.dd(INTER_PRED_IDC + 4) }
31                } else {
32                    self.dd(INTER_PRED_IDC + 4)
33                }
34            } else {
35                0
36            };
37            let mut ref_idx = [-1i8; 2];
38            let mut mvd = [[0i32; 2]; 2];
39            let mut mvp_flag = [0u32; 2];
40            for l in 0..2 {
41                let used = if l == 0 { pred_idc != 1 } else { pred_idc != 0 };
42                if !used {
43                    continue;
44                }
45                let nref = self.sh.num_ref_idx[l];
46                let mut ri = 0;
47                if nref > 1 {
48                    while ri < nref - 1 {
49                        let bin = if ri < 2 { self.dd(REF_IDX + ri as usize) } else { self.bypass() };
50                        if bin == 0 {
51                            break;
52                        }
53                        ri += 1;
54                    }
55                }
56                ref_idx[l] = ri as i8;
57                if l == 1 && self.sh.mvd_l1_zero && pred_idc == 2 {
58                    mvd[1] = [0, 0];
59                } else {
60                    mvd[l] = self.mvd_coding()?;
61                }
62                mvp_flag[l] = self.dd(MVP_FLAG);
63            }
64            let mut f = MvField { mv: [[0; 2]; 2], ref_idx };
65            for l in 0..2 {
66                if ref_idx[l] < 0 {
67                    continue;
68                }
69                let mvp = self.derive_amvp(pb, l, ref_idx[l] as usize, mvp_flag[l] as usize);
70                // mvLX = (mvp + mvd) wrapped to 16 bits (8-94)
71                f.mv[l] = [(mvp[0] as i32 + mvd[l][0]) as i16, (mvp[1] as i32 + mvd[l][1]) as i16];
72            }
73            f
74        };
75        for l in 0..2 {
76            if mvf.ref_idx[l] >= 0 {
77                ensure!((mvf.ref_idx[l] as usize) < self.refs[l].len(), "reference index out of range");
78            }
79        }
80        // store motion
81        let w4 = self.pic.w4;
82        for by in (yp >> 2)..((yp + h) >> 2) {
83            let o = by as usize * w4;
84            self.pic.mvf[o + (xp >> 2) as usize..o + ((xp + w) >> 2) as usize].fill(mvf);
85        }
86        self.motion_compensate(xp, yp, w as usize, h as usize, &mvf);
87        Ok(merge)
88    }

90    fn mvd_coding(&mut self) -> Result<[i32; 2]> {
91        let g0 = [self.dd(ABS_MVD_GT0), self.dd(ABS_MVD_GT0)];
92        let mut g1 = [0, 0];
93        for k in 0..2 {
94            if g0[k] == 1 {
95                g1[k] = self.dd(ABS_MVD_GT1);
96            }
97        }
98        let mut out = [0i32; 2];
99        for k in 0..2 {
100            if g0[k] == 1 {
101                let mut v = 1i32;
102                if g1[k] == 1 {
103                    // EG1
104                    let mut kk = 1;
105                    let mut val = 0i32;
106                    while self.bypass() == 1 {
107                        val += 1 << kk;
108                        kk += 1;
109                        ensure!(kk < 31, "invalid abs_mvd_minus2");
110                    }
111                    val += self.bypass_bits(kk) as i32;
112                    v = val + 2;
113                }
114                if self.bypass() == 1 {
115                    v = -v;
116                }
117                out[k] = v;
118            }
119        }
120        Ok(out)
121    }

123    /// Motion compensation + weighted prediction of one PU into the picture.
124    fn motion_compensate(&mut self, xp: i32, yp: i32, w: usize, h: usize, f: &MvField) {
125        let weighted = match self.sh.slice_type {
126            SliceType::P => self.pps.weighted_pred,
127            SliceType::B => self.pps.weighted_bipred,
128            SliceType::I => false,
129        };
130        let bi = f.pred(0) && f.pred(1);
131        let pwt = self.sh.pwt.as_ref().filter(|_| weighted);
132        let (sx, sy) = self.pic.chroma.shifts();
133        for c in 0..3 {
134            let (x, y, bw, bh, bd) = if c == 0 { (xp, yp, w, h, self.pic.bd_y) } else { (xp >> sx, yp >> sy, w >> sx, h >> sy, self.pic.bd_c) };
135            // Explicit weights equal to the default ones (weight 2^denom and offset 0 for every
136            // list used, as signalled for most slices of a weighted-prediction stream) give exactly
137            // the default prediction: with log2WD = denom + shift1 the weighted sums of 8-265 /
138            // 8-266 are those of 8-262 / 8-263 multiplied by 2^denom.
139            let pwt = pwt.filter(|pwt| !default_weights(pwt, f, c));
140            // Uni-prediction at a full-sample position without weighting: the prediction
141            // ((s << shift3) + off) >> shift3 (8-228, 8-262) is the reference sample itself.
142            if !bi && pwt.is_none() {
143                let l = if f.pred(0) { 0 } else { 1 };
144                // the vector in this component's units: quarter luma samples, 1/8 chroma samples
145                let (mv, frac, sh) =
146                    if c == 0 { ([f.mv[l][0] as i32, f.mv[l][1] as i32], 3, 2) } else { (inter::chroma_mv(f.mv[l], (sx, sy)), 7, 3) };
147                if f.pred(l) && mv[0] & frac == 0 && mv[1] & frac == 0 {
148                    let stride = if c == 0 { self.pic.width } else { self.pic.cwidth };
149                    let frame = &self.refs[l][f.ref_idx[l] as usize].frame;
150                    let dst = &mut self.pic.planes[c][y as usize * stride + x as usize..];
151                    frame.copy_block(c, (x + (mv[0] >> sh), y + (mv[1] >> sh)), (bw, bh), dst, stride);
152                    continue;
153                }
154            }
155            let mut lists = [false; 2];
156            for l in 0..2 {
157                if !f.pred(l) {
158                    continue;
159                }
160                lists[l] = true;
161                let frame = &self.refs[l][f.ref_idx[l] as usize].frame;
162                let buf = if l == 0 || !bi { &mut self.pred0 } else { &mut self.pred1 };
163                if c == 0 {
164                    inter::mc_luma::<S>(frame, (x, y), f.mv[l], (bw, bh), buf, &mut self.mc);
165                } else {
166                    inter::mc_chroma::<S>(frame, c - 1, (x, y), inter::chroma_mv(f.mv[l], (sx, sy)), (bw, bh), buf, &mut self.mc);
167                }
168            }
169            let stride = if c == 0 { self.pic.width } else { self.pic.cwidth };
170            let dst = &mut self.pic.planes[c][y as usize * stride + x as usize..];
171            let Some(pwt) = pwt else {
172                if bi {
173                    inter::put_bi(&self.pred0, &self.pred1, bw, bh, bd, dst, stride);
174                } else {
175                    inter::put_uni(&self.pred0, bw, bh, bd, dst, stride);
176                }
177                continue;
178            };
179            let shift1 = 14 - bd;
180            let log2wd = if c == 0 { pwt.luma_log2_denom } else { pwt.chroma_log2_denom } + shift1;
181            let get = |l: usize| -> (i32, i32) {
182                let e = &pwt.l[l][f.ref_idx[l] as usize];
183                let (wt, o) = if c == 0 { e.luma } else { e.chroma[c - 1] };
184                (wt, o << (bd - 8))
185            };
186            if bi {
187                inter::put_weighted_bi([&self.pred0, &self.pred1], (bw, bh), bd, log2wd, [get(0), get(1)], dst, stride);
188            } else {
189                let l = if lists[0] { 0 } else { 1 };
190                inter::put_weighted_uni(&self.pred0, (bw, bh), bd, log2wd, get(l), dst, stride);
191            }
192        }
193    }
194}

196/// Whether the explicit weights of the lists `f` predicts from are the default ones for component
197/// `c` (weight 2^denom, offset 0).
198fn default_weights(pwt: &crate::craft::video::hevc::decode::slice::PredWeightTable, f: &MvField, c: usize) -> bool {
199    let denom = if c == 0 { pwt.luma_log2_denom } else { pwt.chroma_log2_denom };
200    (0..2).filter(|&l| f.pred(l)).all(|l| {
201        pwt.l[l].get(f.ref_idx[l] as usize).is_some_and(|e| {
202            let (wt, o) = if c == 0 { e.luma } else { e.chroma[c - 1] };
203            wt == 1 << denom && o == 0
204        })
205    })
