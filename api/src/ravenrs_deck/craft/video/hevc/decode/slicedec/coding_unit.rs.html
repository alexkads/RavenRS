9    pub(super) fn coding_quadtree(&mut self, x0: i32, y0: i32, log2: u32, depth: u32) -> Result<()> {
10        let size = 1i32 << log2;
11        let (w, h) = (self.pic.width as i32, self.pic.height as i32);
12        let split = if x0 + size <= w && y0 + size <= h && log2 > self.sps.log2_min_cb {
13            // ctxInc from neighbouring depths (9.3.4.2.2)
14            let mut inc = 0;
15            if self.pic.zavail(x0, y0, x0 - 1, y0, self.slice_addr) && self.pic.blk_at(x0 - 1, y0).depth as u32 > depth {
16                inc += 1;
17            }
18            if self.pic.zavail(x0, y0, x0, y0 - 1, self.slice_addr) && self.pic.blk_at(x0, y0 - 1).depth as u32 > depth {
19                inc += 1;
20            }
21            self.dd(SPLIT_CU + inc) == 1
22        } else {
23            log2 > self.sps.log2_min_cb
24        };
25        if self.pps.cu_qp_delta_enabled && log2 >= self.log2_min_cu_qp_delta {
26            self.is_cu_qp_delta_coded = false;
27            self.cu_qp_delta = 0;
28        }
29        if split {
30            let half = size >> 1;
31            let (x1, y1) = (x0 + half, y0 + half);
32            self.coding_quadtree(x0, y0, log2 - 1, depth + 1)?;
33            if x1 < w {
34                self.coding_quadtree(x1, y0, log2 - 1, depth + 1)?;
35            }
36            if y1 < h {
37                self.coding_quadtree(x0, y1, log2 - 1, depth + 1)?;
38            }
39            if x1 < w && y1 < h {
40                self.coding_quadtree(x1, y1, log2 - 1, depth + 1)?;
41            }
42            Ok(())
43        } else {
44            self.coding_unit(x0, y0, log2, depth)
45        }
46    }

48    /// Set per-4x4 state of a rectangle (luma coordinates).
49    pub(super) fn fill_blk(&mut self, x0: i32, y0: i32, w: i32, h: i32, f: impl Fn(&mut Blk)) {
50        let (bx0, by0) = ((x0 >> 2) as usize, (y0 >> 2) as usize);
51        let bx1 = (((x0 + w) as usize).min(self.pic.width).div_ceil(4)).max(bx0);
52        let by1 = (((y0 + h) as usize).min(self.pic.height).div_ceil(4)).max(by0);
53        for by in by0..by1 {
54            for b in &mut self.pic.blk[by * self.pic.w4 + bx0..by * self.pic.w4 + bx1] {
55                f(b);
56            }
57        }
58    }

60    /// qPY_PRED derivation (8.6.1) at the start of a quantization group.
61    fn derive_qp_pred(&mut self, xq: i32, yq: i32) {
62        let prev = self.last_qp;
63        let s = self.sps.log2_ctb;
64        let cur_ctb = ((yq >> s), (xq >> s));
65        let qa = if self.pic.zavail(xq, yq, xq - 1, yq, self.slice_addr) && ((yq >> s), ((xq - 1) >> s)) == cur_ctb {
66            self.pic.blk_at(xq - 1, yq).qp as i32
67        } else {
68            prev
69        };
70        let qb = if self.pic.zavail(xq, yq, xq, yq - 1, self.slice_addr) && (((yq - 1) >> s), (xq >> s)) == cur_ctb {
71            self.pic.blk_at(xq, yq - 1).qp as i32
72        } else {
73            prev
74        };
75        self.qp_y_pred = (qa + qb + 1) >> 1;
76    }

78    pub(super) fn update_qp(&mut self) {
79        let off = 6 * (self.pic.bd_y as i32 - 8);
80        self.qp_y = ((self.qp_y_pred + self.cu_qp_delta + 52 + 2 * off) % (52 + off)) - off;
81        let offc = 6 * (self.pic.bd_c as i32 - 8);
82        let qpi_cb = (self.qp_y + self.pps.cb_qp_offset + self.sh.cb_qp_offset).clamp(-offc, 57);
83        let qpi_cr = (self.qp_y + self.pps.cr_qp_offset + self.sh.cr_qp_offset).clamp(-offc, 57);
84        let is420 = self.pic.chroma == ChromaFormat::C420;
85        self.qp_cb = qpc(qpi_cb, is420) + offc;
86        self.qp_cr = qpc(qpi_cr, is420) + offc;
87    }

89    fn coding_unit(&mut self, x0: i32, y0: i32, log2: u32, depth: u32) -> Result<()> {
90        let n = 1i32 << log2;
91        // quantization group start
92        let qg_mask = (1 << self.log2_min_cu_qp_delta) - 1;
93        if (x0 & qg_mask) == 0 && (y0 & qg_mask) == 0 {
94            self.derive_qp_pred(x0, y0);
95        }
96        self.update_qp();
97        self.cu_transquant_bypass = if self.pps.transquant_bypass { self.dd(CU_TRANSQUANT_BYPASS) == 1 } else { false };
98        let mut skip = false;
99        if !self.sh.is_intra() {
100            let mut inc = 0;
101            if self.pic.zavail(x0, y0, x0 - 1, y0, self.slice_addr) && self.pic.blk_at(x0 - 1, y0).flags & F_SKIP != 0 {
102                inc += 1;
103            }
104            if self.pic.zavail(x0, y0, x0, y0 - 1, self.slice_addr) && self.pic.blk_at(x0, y0 - 1).flags & F_SKIP != 0 {
105                inc += 1;
106            }
107            skip = self.dd(CU_SKIP + inc) == 1;
108        }
109        // CU-level block state: depth, skip, coded; CU edges are transform and prediction edges
110        let bypass = self.cu_transquant_bypass;
111        let qp = self.qp_y as i8;
112        self.fill_blk(x0, y0, n, n, |b| {
113            b.depth = depth as u8;
114            b.flags = F_CODED | if skip { F_SKIP } else { 0 } | if bypass { F_NOFILTER } else { 0 };
115            b.qp = qp;
116            b.ipm = 1;
117            b.edges = 0;
118        });
119        self.mark_edges(x0, y0, n, n, E_TU_V | E_PU_V, E_TU_H | E_PU_H);
120        if skip {
121            self.cu_intra = false;
122            self.prediction_unit(&PredBlock { xc: x0, yc: y0, ncb: n, xp: x0, yp: y0, w: n, h: n, part_idx: 0, part: PartMode::P2Nx2N }, true)?;
123            self.finish_cu(x0, y0, n);
124            return Ok(());
125        }
126        let intra = if !self.sh.is_intra() { self.dd(PRED_MODE) == 1 } else { true };
127        self.cu_intra = intra;
128        let mut part = PartMode::P2Nx2N;
129        if !intra || log2 == self.sps.log2_min_cb {
130            part = self.parse_part_mode(intra, log2)?;
131        }
132        self.cu_part = part;
133        if intra {
134            self.fill_blk(x0, y0, n, n, |b| b.flags |= F_INTRA);
135            self.intra_split = part == PartMode::PNxN;
136            let pcm = part == PartMode::P2Nx2N
137                && self.sps.pcm
138                && log2 >= self.sps.log2_min_pcm
139                && log2 <= self.sps.log2_max_pcm
140                && self.c.decode_terminate() == 1;
141            if pcm {
142                self.pcm_sample(x0, y0, log2)?;
143                if self.sps.pcm_loop_filter_disabled {
144                    self.fill_blk(x0, y0, n, n, |b| b.flags |= F_NOFILTER);
145                }
146                self.finish_cu(x0, y0, n);
147                return Ok(());
148            }
149            self.parse_intra_modes(x0, y0, n, part)?;
150            self.max_trafo_depth = self.sps.max_th_depth_intra + self.intra_split as u32;
151            self.transform_tree(TtNode { x0, y0, xb: x0, yb: y0, log2, blk_idx: 0 }, 0, [[true; 2]; 2])?;
152        } else {
153            self.intra_split = false;
154            let mut merge2nx2n = false;
155            let h = n / 2;
156            let q = n / 4;
157            let parts: &[(i32, i32, i32, i32)] = match part {
158                PartMode::P2Nx2N => &[(0, 0, n, n)],
159                PartMode::P2NxN => &[(0, 0, n, h), (0, h, n, h)],
160                PartMode::PNx2N => &[(0, 0, h, n), (h, 0, h, n)],
161                PartMode::P2NxnU => &[(0, 0, n, q), (0, q, n, n - q)],
162                PartMode::P2NxnD => &[(0, 0, n, n - q), (0, n - q, n, q)],
163                PartMode::PnLx2N => &[(0, 0, q, n), (q, 0, n - q, n)],
164                PartMode::PnRx2N => &[(0, 0, n - q, n), (n - q, 0, q, n)],
165                PartMode::PNxN => &[(0, 0, h, h), (h, 0, h, h), (0, h, h, h), (h, h, h, h)],
166            };
167            for (i, &(px, py, pw, ph)) in parts.iter().enumerate() {
168                let merge =
169                    self.prediction_unit(&PredBlock { xc: x0, yc: y0, ncb: n, xp: x0 + px, yp: y0 + py, w: pw, h: ph, part_idx: i, part }, false)?;
170                if part == PartMode::P2Nx2N {
171                    merge2nx2n = merge;
172                }
173                if px != 0 {
174                    self.mark_edges(x0 + px, y0 + py, pw, ph, E_PU_V, 0);
175                }
176                if py != 0 {
177                    self.mark_edges(x0 + px, y0 + py, pw, ph, 0, E_PU_H);
178                }
179            }
180            let root_cbf = if !merge2nx2n { self.dd(RQT_ROOT_CBF) == 1 } else { true };
181            if root_cbf {
182                self.max_trafo_depth = self.sps.max_th_depth_inter;
183                self.transform_tree(TtNode { x0, y0, xb: x0, yb: y0, log2, blk_idx: 0 }, 0, [[true; 2]; 2])?;
184            }
185        }
186        self.finish_cu(x0, y0, n);
187        Ok(())
188    }

190    /// Store the final QpY of the CU (may have changed with cu_qp_delta).
191    fn finish_cu(&mut self, x0: i32, y0: i32, n: i32) {
192        let qp = self.qp_y as i8;
193        self.fill_blk(x0, y0, n, n, |b| b.qp = qp);
194        self.last_qp = self.qp_y;
195    }

197    /// Mark the left (vertical) and top (horizontal) edges of a rectangle.
198    pub(super) fn mark_edges(&mut self, x0: i32, y0: i32, w: i32, h: i32, v: u8, hz: u8) {
199        let w4 = self.pic.w4;
200        let (bx0, by0) = ((x0 >> 2) as usize, (y0 >> 2) as usize);
201        if v != 0 {
202            let by1 = ((y0 + h) as usize).min(self.pic.height).div_ceil(4);
203            for by in by0..by1 {
204                self.pic.blk[by * w4 + bx0].edges |= v;
205            }
206        }
207        if hz != 0 {
208            let bx1 = ((x0 + w) as usize).min(self.pic.width).div_ceil(4);
209            for b in &mut self.pic.blk[by0 * w4 + bx0..by0 * w4 + bx1] {
210                b.edges |= hz;
211            }
212        }
213    }

215    fn parse_part_mode(&mut self, intra: bool, log2: u32) -> Result<PartMode> {
216        if intra {
217            return Ok(if self.dd(PART_MODE) == 1 { PartMode::P2Nx2N } else { PartMode::PNxN });
218        }
219        if self.dd(PART_MODE) == 1 {
220            return Ok(PartMode::P2Nx2N);
221        }
222        let min = log2 == self.sps.log2_min_cb;
223        if min {
224            if self.dd(PART_MODE + 1) == 1 {
225                return Ok(PartMode::P2NxN);
226            }
227            if log2 == 3 {
228                return Ok(PartMode::PNx2N);
229            }
230            return Ok(if self.dd(PART_MODE + 2) == 1 { PartMode::PNx2N } else { PartMode::PNxN });
231        }
232        let amp = self.sps.amp;
233        let hor = self.dd(PART_MODE + 1) == 1;
234        if !amp {
235            return Ok(if hor { PartMode::P2NxN } else { PartMode::PNx2N });
236        }
237        let no_amp = self.dd(PART_MODE + 3) == 1;
238        if no_amp {
239            return Ok(if hor { PartMode::P2NxN } else { PartMode::PNx2N });
240        }
241        let b = self.bypass();
242        Ok(match (hor, b) {
243            (true, 0) => PartMode::P2NxnU,
244            (true, _) => PartMode::P2NxnD,
245            (false, 0) => PartMode::PnLx2N,
246            (false, _) => PartMode::PnRx2N,
247        })
248    }

250    fn pcm_sample(&mut self, x0: i32, y0: i32, log2: u32) -> Result<()> {
251        // the PCM samples start at the next byte boundary after the terminate bin
252        let start = self.c.bit_pos().div_ceil(8);
253        let data = self.c.data();
254        let n = 1usize << log2;
255        let (csx, csy) = self.pic.chroma.shifts();
256        let (ncw, nch) = (n >> csx, n >> csy);
257        let (bl, bc) = (self.sps.pcm_bit_depth_luma, self.sps.pcm_bit_depth_chroma);
258        let total_bits = n * n * bl as usize + 2 * ncw * nch * bc as usize;
259        ensure!(start + total_bits.div_ceil(8) <= data.len(), "PCM samples truncated");
260        let mut r = crate::craft::media::bitstream::BitReader::new(&data[start..]);
261        let (sy, sc) = (self.pic.bd_y - bl, self.pic.bd_c - bc);
262        let w = self.pic.width;
263        for y in 0..n {
264            for x in 0..n {
265                let v = r.read_bits(bl)? as u16;
266                self.pic.planes[0][(y0 as usize + y) * w + x0 as usize + x] = S::from_u16(v << sy);
267            }
268        }
269        let cw = self.pic.cwidth;
270        for c in 1..3 {
271            for y in 0..nch {
272                for x in 0..ncw {
273                    let v = r.read_bits(bc)? as u16;
274                    self.pic.planes[c][((y0 as usize >> csy) + y) * cw + (x0 as usize >> csx) + x] = S::from_u16(v << sc);
275                }
276            }
277        }
278        let end = start + r.position().div_ceil(8);
279        self.c.set_byte_pos(end);
280        self.c.init_engine()
281    }

283    fn parse_intra_modes(&mut self, x0: i32, y0: i32, n: i32, part: PartMode) -> Result<()> {
284        let nb = if part == PartMode::PNxN { 2 } else { 1 };
285        let pb = n / nb;
286        let mut prev = [false; 4];
287        for (i, p) in prev.iter_mut().enumerate().take((nb * nb) as usize) {
288            let _ = i;
289            *p = self.dd(PREV_INTRA_LUMA) == 1;
290        }
291        for j in 0..nb {
292            for i in 0..nb {
293                let k = (j * nb + i) as usize;
294                let (xp, yp) = (x0 + i * pb, y0 + j * pb);
295                // candidate modes (8.4.2)
296                let cand_a = if self.pic.zavail(xp, yp, xp - 1, yp, self.slice_addr) {
297                    let b = self.pic.blk_at(xp - 1, yp);
298                    if b.flags & F_INTRA != 0 { b.ipm as u32 } else { 1 }
299                } else {
300                    1
301                };
302                let cand_b = if self.pic.zavail(xp, yp, xp, yp - 1, self.slice_addr) && yp > ((yp >> self.sps.log2_ctb) << self.sps.log2_ctb) {
303                    let b = self.pic.blk_at(xp, yp - 1);
304                    if b.flags & F_INTRA != 0 { b.ipm as u32 } else { 1 }
305                } else {
306                    1
307                };
308                let mut list = if cand_a == cand_b {
309                    if cand_a < 2 { [0, 1, 26] } else { [cand_a, 2 + ((cand_a + 29) % 32), 2 + ((cand_a - 2 + 1) % 32)] }
310                } else {
311                    let c = if cand_a != 0 && cand_b != 0 {
312                        0
313                    } else if cand_a != 1 && cand_b != 1 {
314                        1
315                    } else {
316                        26
317                    };
318                    [cand_a, cand_b, c]
319                };
320                let mode = if prev[k] {
321                    let idx = if self.bypass() == 0 {
322                        0
323                    } else if self.bypass() == 0 {
324                        1
325                    } else {
326                        2
327                    };
328                    list[idx]
329                } else {
330                    let mut m = self.bypass_bits(5);
331                    list.sort_unstable();
332                    for &c in &list {
333                        if m >= c {
334                            m += 1;
335                        }
336                    }
337                    m
338                };
339                self.fill_blk(xp, yp, pb, pb, |b| b.ipm = mode as u8);
340            }
341        }
342        // chroma mode (one per CU in 4:2:0 and 4:2:2, derived from the first PU's luma mode:
343        // Table 8-2, then the 4:2:2 mapping of Table 8-3)
344        let icpm = if self.dd(INTRA_CHROMA) == 0 { 4 } else { self.bypass_bits(2) };
345        let luma = self.pic.blk_at(x0, y0).ipm as u32;
346        let mode = match icpm {
347            4 => luma,
348            _ => {
349                let m = [0, 26, 10, 1][icpm as usize];
350                if m == luma { 34 } else { m }
351            }
352        };
353        self.intra_chroma_mode = match self.pic.chroma {
354            ChromaFormat::C420 => mode,
355            ChromaFormat::C422 => INTRA_CHROMA_MODE_422.get(mode as usize).copied().unwrap_or(mode as u8) as u32,
356        };
357        Ok(())
358    }
