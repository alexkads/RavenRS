7/// Significant scan positions of one sub-block, in decoding (descending) order.
8#[derive(Default)]

11    n: usize,
12}

47/// sigCtx increment of sig_coeff_flag (9.3.4.2.5) for the position `(xp, yp)` inside the sub-block.
48fn sig_ctx_inc(p: &SigMapParams, (xp, yp): (usize, usize)) -> usize {
49    let (xc, yc) = (p.xs * 4 + xp, p.ys * 4 + yp);
50    if p.log2 == 2 {
51        return CTX_IDX_MAP[(yc << 2) + xc] as usize;
52    }
53    if xc + yc == 0 {
54        return 0;
55    }
56    let mut s = match p.prev_csbf {
57        0 => {
58            if xp + yp == 0 {
59                2
60            } else if xp + yp < 3 {
61                1
62            } else {
63                0
64            }
65        }
66        1 => {
67            if yp == 0 {
68                2
69            } else if yp == 1 {
70                1
71            } else {
72                0
73            }
74        }
75        2 => {
76            if xp == 0 {
77                2
78            } else if xp == 1 {
79                1
80            } else {
81                0
82            }
83        }
84        _ => 2,
85    };
86    if p.c == 0 {
87        if p.xs + p.ys > 0 {
88            s += 3;
89        }
90        s += if p.log2 == 3 { if p.scan_idx == 0 { 9 } else { 15 } } else { 21 };
91    } else {
92        s += if p.log2 == 3 { 9 } else { 12 };
93    }
94    s
95}

97/// A node of the transform tree: the block at (`x0`, `y0`) of size 2^`log2`, its parent's origin
98/// (`xb`, `yb`) and its index `blk_idx` among the parent's four children.
99#[derive(Clone, Copy)]
100pub(super) struct TtNode {
101    pub x0: i32,
102    pub y0: i32,
103    pub xb: i32,
104    pub yb: i32,
105    pub log2: u32,
106    pub blk_idx: u32,
107}

109impl<'a, S: Sample> SliceDecoder<'a, S> {
110    /// `parent_cbf_c[c][t]`: cbf_cb / cbf_cr (c) of the parent node, for the upper (t = 0) and, in
111    /// 4:2:2, the lower (t = 1) chroma block (7.3.8.8).
112    pub(super) fn transform_tree(&mut self, node: TtNode, depth: u32, parent_cbf_c: [[bool; 2]; 2]) -> Result<()> {
113        let TtNode { x0, y0, log2, .. } = node;
114        let split =
115            if log2 <= self.sps.log2_max_tb && log2 > self.sps.log2_min_tb && depth < self.max_trafo_depth && !(self.intra_split && depth == 0) {
116                self.dd(SPLIT_TRANSFORM + (5 - log2 as usize)) == 1
117            } else {
118                // interSplitFlag (7.4.9.8)
119                let inter_split = self.sps.max_th_depth_inter == 0 && !self.cu_intra && self.cu_part != PartMode::P2Nx2N && depth == 0;
120                log2 > self.sps.log2_max_tb || (self.intra_split && depth == 0) || inter_split
121            };
122        let mut cbf_c = [[false; 2]; 2];
123        if log2 > 2 {
124            // the second cbf of each chroma component (the lower block of the 4:2:2 chroma
125            // rectangle) is coded for a leaf, and at 8x8 (whose chroma is coded with the 4x4 luma
126            // blocks) also when split
127            let second = self.pic.chroma == ChromaFormat::C422 && (!split || log2 == 3);
128            for (c, cbf) in cbf_c.iter_mut().enumerate() {
129                if depth == 0 || parent_cbf_c[c][0] {
130                    cbf[0] = self.dd(CBF_CHROMA + depth as usize) == 1;
131                    if second {
132                        cbf[1] = self.dd(CBF_CHROMA + depth as usize) == 1;
133                    }
134                }
135            }
136        } else {
137            // 4x4 luma blocks: chroma cbfs are those of the parent (coded at blkIdx 3)
138            cbf_c = parent_cbf_c;
139            if depth == 0 {
140                cbf_c = [[false; 2]; 2];
141            }
142        }
143        if split {
144            let h = 1 << (log2 - 1);
145            for (blk_idx, (dx, dy)) in [(0, 0), (h, 0), (0, h), (h, h)].into_iter().enumerate() {
146                let child = TtNode { x0: x0 + dx, y0: y0 + dy, xb: x0, yb: y0, log2: log2 - 1, blk_idx: blk_idx as u32 };
147                self.transform_tree(child, depth + 1, cbf_c)?;
148            }
149            return Ok(());
150        }
151        let any_cbf_c = cbf_c.iter().flatten().any(|&b| b);
152        let cbf_luma = if self.cu_intra || depth != 0 || any_cbf_c { self.dd(CBF_LUMA + if depth == 0 { 1 } else { 0 }) == 1 } else { true };
153        let n = 1 << log2;
154        if depth > 0 {
155            self.mark_edges(x0, y0, n, n, E_TU_V, E_TU_H);
156        }
157        self.transform_unit(node, cbf_luma, cbf_c)
158    }

160    fn transform_unit(&mut self, node: TtNode, cbf_luma: bool, cbf_c: [[bool; 2]; 2]) -> Result<()> {
161        let TtNode { x0, y0, xb, yb, log2, blk_idx } = node;
162        let n = 1i32 << log2;
163        let chroma_here = log2 > 2 || blk_idx == 3;
164        let cbf_chroma = cbf_c.iter().flatten().any(|&b| b);
165        if (cbf_luma || cbf_chroma) && self.pps.cu_qp_delta_enabled && !self.is_cu_qp_delta_coded {
166            // cu_qp_delta_abs: prefix TU(5) with ctx 0,1,1,1,1, suffix EG0
167            let mut v = 0;
168            while v < 5 && self.dd(CU_QP_DELTA_ABS + if v == 0 { 0 } else { 1 }) == 1 {
169                v += 1;
170            }
171            if v == 5 {
172                let mut k = 0;
173                while self.bypass() == 1 {
174                    v += 1 << k;
175                    k += 1;
176                    ensure!(k < 32, "invalid cu_qp_delta_abs");
177                }
178                v += self.bypass_bits(k) as i32;
179            }
180            if v != 0 && self.bypass() == 1 {
181                v = -v;
182            }
183            let off = 6 * (self.pic.bd_y as i32 - 8);
184            ensure!(v >= -(26 + off / 2) && v <= 25 + off / 2, "CuQpDeltaVal out of range");
185            self.is_cu_qp_delta_coded = true;
186            self.cu_qp_delta = v;
187            self.update_qp();
188        }
189        // luma
190        if self.cu_intra {
191            let mode = self.pic.blk_at(x0, y0).ipm as u32;
192            self.intra_predict(x0, y0, log2, 0, mode);
193        }
194        if cbf_luma {
195            let qp_bd = 6 * (self.pic.bd_y as i32 - 8);
196            let qp = self.qp_y + qp_bd;
197            self.residual(x0, y0, log2, 0, qp)?;
198            self.fill_blk(x0, y0, n, n, |b| b.flags |= F_CBF);
199        }
200        if !chroma_here {
201            return Ok(());
202        }
203        // Chroma blocks of this transform unit (7.3.8.10): square blocks of (luma size) /
204        // SubWidthC at the chroma position of the unit (of the 8x8 parent for 4x4 luma blocks);
205        // in 4:2:2 the chroma rectangle is twice as tall and coded as two stacked square blocks,
206        // the lower one predicted from the reconstructed upper one (8.4.4.1).
207        let (sx, sy) = self.pic.chroma.shifts();
208        let (xc, yc, log2c) = if log2 > 2 { (x0 >> sx, y0 >> sy, log2 - 1) } else { (xb >> sx, yb >> sy, 2) };
209        let blocks = if self.pic.chroma == ChromaFormat::C422 { 2 } else { 1 };
210        for c in 1..3 {
211            for t in 0..blocks {
212                let yt = yc + ((t as i32) << log2c);
213                if self.cu_intra {
214                    let mode = self.intra_chroma_mode;
215                    self.intra_predict(xc, yt, log2c, c, mode);
216                }
217                if cbf_c[c - 1][t] {
218                    let qp = if c == 1 { self.qp_cb } else { self.qp_cr };
219                    self.residual(xc, yt, log2c, c, qp)?;
220                }
221            }
222        }
223        Ok(())
224    }

226    /// Intra prediction of an n x n block of component `c` at component position (x0, y0).
227    fn intra_predict(&mut self, x0: i32, y0: i32, log2: u32, c: usize, mode: u32) {
228        let n = 1usize << log2;
229        // shifts from component to luma positions (8.4.4.2.1: the availability of neighbouring
230        // samples is derived at the luma positions (xNbCmp * SubWidthC, yNbCmp * SubHeightC))
231        let (sx, sy) = if c == 0 { (0, 0) } else { self.pic.chroma.shifts() };
232        let (xl, yl) = (x0 << sx, y0 << sy);
233        // component samples per 4x4 luma block, down the left column and along the top row
234        // (4:2:2 chroma: 4 rows but 2 columns)
235        let (unit_y, unit_x) = (4usize >> sy, 4usize >> sx);
236        let stride = if c == 0 { self.pic.width } else { self.pic.cwidth };
237        let constrained = self.pps.constrained_intra_pred;
238        let mut al = [false; 129];
239        let mut at = [false; 129];
240        let addr = self.slice_addr;
241        let pic = &*self.pic;
242        let (units_l, units_t) = ((2 * n / unit_y) as u32, (2 * n / unit_x) as u32);
243        let mut ml = pic.zavail_run((xl, yl), (xl - 1, yl), true, units_l, addr);
244        let mut mt = pic.zavail_run((xl, yl), (xl, yl - 1), false, units_t, addr);
245        let mut corner = pic.zavail(xl, yl, xl - 1, yl - 1, addr);
246        if constrained {
247            let intra = |x: i32, y: i32| pic.blk_at(x, y).flags & F_INTRA != 0;
248            for k in 0..units_l {
249                if ml & (1 << k) != 0 && !intra(xl - 1, yl + 4 * k as i32) {
250                    ml &= !(1 << k);
251                }
252            }
253            for k in 0..units_t {
254                if mt & (1 << k) != 0 && !intra(xl + 4 * k as i32, yl - 1) {
255                    mt &= !(1 << k);
256                }
257            }
258            corner = corner && intra(xl - 1, yl - 1);
259        }
260        let plane = &pic.planes[c];
261        let r = &mut self.refs_buf;
262        if corner {
263            al[0] = true;
264            at[0] = true;
265            let v = plane[(y0 - 1) as usize * stride + (x0 - 1) as usize].to_u16();
266            r.left[0] = v;
267            r.top[0] = v;
268        }
269        for_runs(ml, unit_y, |a, b| {
270            let left = (y0 as usize + a) * stride + x0 as usize - 1;
271            al[1 + a..1 + b].fill(true);
272            for (k, v) in r.left[1 + a..1 + b].iter_mut().enumerate() {
273                *v = plane[left + k * stride].to_u16();
274            }
275        });
276        for_runs(mt, unit_x, |a, b| {
277            let top = (y0 as usize - 1) * stride + x0 as usize;
278            at[1 + a..1 + b].fill(true);
279            for (v, s) in r.top[1 + a..1 + b].iter_mut().zip(&plane[top + a..top + b]) {
280                *v = s.to_u16();
281            }
282        });
283        let bd = if c == 0 { self.pic.bd_y } else { self.pic.bd_c };
284        intra::substitute(r, &al, &at, n, bd);
285        if c == 0 {
286            intra::filter(r, mode, n, self.sps.strong_intra_smoothing, true, bd);
287        }
288        let off = y0 as usize * stride + x0 as usize;
289        intra::predict(r, mode, n, c == 0, bd, &mut self.pic.planes[c][off..], stride);
290    }

292    /// last_sig_coeff_{x,y}_{prefix,suffix}: the position of the last significant coefficient
293    /// (before the vertical-scan swap).
294    fn last_sig_coeff_pos(&mut self, log2: u32, c: usize) -> (u32, u32) {
295        let cmax = (log2 << 1) - 1;
296        let (off, shift) = if c == 0 { (3 * (log2 - 2) + ((log2 - 1) >> 2), (log2 + 1) >> 2) } else { (15, log2 - 2) };
297        let mut px = 0;
298        while px < cmax && self.dd(LAST_X_PREFIX + (off + (px >> shift)) as usize) == 1 {
299            px += 1;
300        }
301        let mut py = 0;
302        while py < cmax && self.dd(LAST_Y_PREFIX + (off + (py >> shift)) as usize) == 1 {
303            py += 1;
304        }
305        let mut lx = px;
306        if px > 3 {
307            let nb = (px >> 1) - 1;
308            lx = (1 << nb) * (2 + (px & 1)) + self.bypass_bits(nb);
309        }
310        let mut ly = py;
311        if py > 3 {
312            let nb = (py >> 1) - 1;
313            ly = (1 << nb) * (2 + (py & 1)) + self.bypass_bits(nb);
314        }
315        (lx, ly)
316    }

318    /// scanIdx (7.4.9.11): 0 diagonal, 1 horizontal, 2 vertical.
319    fn scan_idx(&self, x0: i32, y0: i32, log2: u32, c: usize) -> usize {
320        if !(self.cu_intra && (log2 == 2 || (log2 == 3 && c == 0))) {
321            return 0;
322        }
323        let m = if c == 0 { self.pic.blk_at(x0, y0).ipm as u32 } else { self.intra_chroma_mode };
324        if (6..=14).contains(&m) {
325            2
326        } else if (22..=30).contains(&m) {
327            1
328        } else {
329            0
330        }
331    }

333    /// sig_coeff_flag decoding of one coded sub-block, appending the significant scan positions
334    /// (descending) to `list`.
335    fn read_sig_map(&mut self, p: &SigMapParams, list: &mut SigList) {
336        let pos_scan = &SCAN_4[p.scan_idx];
337        let c_off_sig = if p.c == 0 { 0 } else { 27 };
338        let mut infer_dc = p.infer_dc;
339        let mut np = p.start;
340        while np >= 0 {
341            let (xp, yp) = (pos_scan[np as usize].0 as usize, pos_scan[np as usize].1 as usize);
342            if np > 0 || !infer_dc {
343                let sig_ctx = sig_ctx_inc(p, (xp, yp));
344                if self.c.decode_decision(SIG_COEFF + c_off_sig + sig_ctx) == 1 {
345                    list.push(np as u8);
346                    infer_dc = false;
347                }
348            } else {
349                // np == 0 with inferSbDcSigCoeffFlag
350                list.push(0);
351            }
352            np -= 1;
353        }
354    }

356    /// coeff_abs_level_greater1_flag / greater2_flag of one sub-block with `nsig` significant
357    /// coefficients; `ctx_set` as derived in 9.3.4.2.6.
358    fn read_greater_flags(&mut self, c: usize, ctx_set: usize, nsig: usize) -> GreaterFlags {
359        let mut g1ctx = 1u32;
360        let mut f = GreaterFlags::default();
361        let c_off_g1 = if c > 0 { 16 } else { 0 };
362        for k in 0..nsig.min(8) {
363            let inc = ctx_set * 4 + g1ctx.min(3) as usize + c_off_g1;
364            let flag = self.c.decode_decision(GT1 + inc);
365            f.gt1[k] = flag as u8;
366            if flag == 1 {
367                g1ctx = 0;
368                if f.first_g1.is_none() {
369                    f.first_g1 = Some(k);
370                }
371            } else if g1ctx > 0 {
372                g1ctx += 1;
373            }
374        }
375        f.g1ctx = g1ctx;
376        if f.first_g1.is_some() {
377            f.gt2 = self.c.decode_decision(GT2 + ctx_set + if c > 0 { 4 } else { 0 }) as u8;
378        }
379        f
380    }

382    /// Signs and remaining levels of one sub-block: the signed coefficient value of each
383    /// significant position, in the order of `list`.
384    fn read_levels(&mut self, list: &SigList, flags: &GreaterFlags, sdh: bool) -> Result<[i32; 16]> {
385        let nsig = list.n;
386        let last_sig_pos = list.pos[0] as i32;
387        let first_sig_pos = list.pos[nsig - 1] as i32;
388        let sign_hidden = sdh && last_sig_pos - first_sig_pos > 3;
389        let nsigns = if sign_hidden { nsig - 1 } else { nsig };
390        let mut signs = if nsigns > 0 { self.bypass_bits(nsigns as u32) << (32 - nsigns) } else { 0 };
391        let mut values = [0i32; 16];
392        let mut rice = 0u32;
393        let mut sum_abs = 0i32;
394        for k in 0..nsig {
395            let mut base = 1 + flags.gt1[k] as i32;
396            if Some(k) == flags.first_g1 {
397                base += flags.gt2 as i32;
398            }
399            let thresh = if k < 8 { if Some(k) == flags.first_g1 { 3 } else { 2 } } else { 1 };
400            let mut level = base;
401            if base == thresh {
402                let rem = self.coeff_abs_level_remaining(rice)?;
403                level = base + rem;
404                if level > 3 * (1 << rice) {
405                    rice = (rice + 1).min(4);
406                }
407            }
408            let mut v = level;
409            if !(sign_hidden && k == nsig - 1) {
410                if signs & 0x8000_0000 != 0 {
411                    v = -v;
412                }
413                signs <<= 1;
414            }
415            if sign_hidden {
416                sum_abs += level;
417                if k == nsig - 1 && sum_abs & 1 == 1 {
418                    v = -v;
419                }
420            }
421            values[k] = v;
422        }
423        Ok(values)
424    }

426    /// residual_coding() + scaling + transform + reconstruction for one transform block.
427    fn residual(&mut self, x0: i32, y0: i32, log2: u32, c: usize, qp: i32) -> Result<()> {
428        let n = 1usize << log2;
429        let bypass = self.cu_transquant_bypass;
430        let ts = if self.pps.transform_skip && !bypass && log2 == 2 { self.dd(TRANSFORM_SKIP + (c > 0) as usize) == 1 } else { false };
431        let (mut lx, mut ly) = self.last_sig_coeff_pos(log2, c);
432        let scan_idx = self.scan_idx(x0, y0, log2, c);
433        if scan_idx == 2 {
434            std::mem::swap(&mut lx, &mut ly);
435        }
436        ensure!((lx as usize) < n && (ly as usize) < n, "last significant coefficient outside the block");
437        let log2sb = log2 - 2;
438        let sb_scan = scan_order(log2sb, scan_idx);
439        let pos_scan = &SCAN_4[scan_idx];
440        let sb_w = 1usize << log2sb;
441        let last_sb = sb_scan.iter().position(|&(x, y)| x as u32 == lx >> 2 && y as u32 == ly >> 2).unwrap_or(0);
442        let last_pos = pos_scan.iter().position(|&(x, y)| x as u32 == lx & 3 && y as u32 == ly & 3).unwrap_or(0);
443        self.coeffs[..n * n].fill(0);
444        let mut csbf = [0u8; 64];
445        let mut prev_g1: Option<u32> = None;
446        let sdh = self.pps.sign_data_hiding && !bypass;
447        let (mut max_x, mut max_y) = (0usize, 0usize);
448        for i in (0..=last_sb).rev() {
449            let (xs, ys) = (sb_scan[i].0 as usize, sb_scan[i].1 as usize);
450            let mut infer_dc = false;
451            let right = if xs + 1 < sb_w { csbf[ys * 8 + xs + 1] } else { 0 };
452            let below = if ys + 1 < sb_w { csbf[(ys + 1) * 8 + xs] } else { 0 };
453            if i < last_sb && i > 0 {
454                let inc = (right | below).min(1) as usize + if c > 0 { 2 } else { 0 };
455                csbf[ys * 8 + xs] = self.c.decode_decision(CODED_SUB_BLOCK + inc) as u8;
456                infer_dc = true;
457            } else {
458                csbf[ys * 8 + xs] = 1;
459            }
460            // significance map
461            let mut list = SigList::default();
462            let start: i32 = if i == last_sb {
463                list.push(last_pos as u8);
464                last_pos as i32 - 1
465            } else {
466                15
467            };
468            if csbf[ys * 8 + xs] != 0 {
469                let params = SigMapParams { log2, c, scan_idx, xs, ys, prev_csbf: right + (below << 1), infer_dc, start };
470                self.read_sig_map(&params, &mut list);
471            }
472            if list.n == 0 {
473                continue;
474            }
475            // greater1 / greater2 flags
476            let mut ctx_set = if i == 0 || c > 0 { 0 } else { 2 };
477            if let Some(g) = prev_g1
478                && g == 0
479            {
480                ctx_set += 1;
481            }
482            let flags = self.read_greater_flags(c, ctx_set, list.n);
483            prev_g1 = Some(flags.g1ctx);
484            let values = self.read_levels(&list, &flags, sdh)?;
485            for k in 0..list.n {
486                let np = list.pos[k] as usize;
487                let (xc, yc) = (xs * 4 + pos_scan[np].0 as usize, ys * 4 + pos_scan[np].1 as usize);
488                self.coeffs[yc * n + xc] = values[k];
489                max_x = max_x.max(xc);
490                max_y = max_y.max(yc);
491            }
492        }
493        self.reconstruct((x0, y0), log2, c, qp, ts, (max_x, max_y))
494    }

496    fn coeff_abs_level_remaining(&mut self, rice: u32) -> Result<i32> {
497        let mut prefix = 0u32;
498        while prefix < 32 && self.bypass() == 1 {
499            prefix += 1;
500        }
501        ensure!(prefix < 32, "invalid coeff_abs_level_remaining");
502        if prefix <= 3 {
503            Ok(((prefix << rice) + self.bypass_bits(rice)) as i32)
504        } else {
505            let nb = prefix - 3 + rice;
506            ensure!(nb <= 24, "coeff_abs_level_remaining too large");
507            Ok(((((1u32 << (prefix - 3)) + 2) << rice) + self.bypass_bits(nb)) as i32)
508        }
509    }
510
511    /// Scaling, transform and addition to the prediction.
512    fn reconstruct(&mut self, (x0, y0): (i32, i32), log2: u32, c: usize, qp: i32, ts: bool, (max_x, max_y): (usize, usize)) -> Result<()> {
513        let n = 1usize << log2;
514        let bd = if c == 0 { self.pic.bd_y } else { self.pic.bd_c };
515        let coeffs = &mut self.coeffs[..n * n];
516        let mut dc_only = false;
517        if !self.cu_transquant_bypass {
518            // scaling (8.6.3)
519            let bd_shift = bd + log2 - 5;
520            let round = 1i64 << (bd_shift - 1);
521            let scale = (LEVEL_SCALE[(qp % 6) as usize] as i64) << (qp / 6);
522            let m: Option<&[u8]> = match &self.pic.scaling {
523                Some(s) if !(ts && n > 4) => {
524                    let size_id = (log2 - 2) as usize;
525                    let matrix_id = if self.cu_intra { c } else { 3 + c };
526                    Some(&s[size_id * 6 + matrix_id])
527                }
528                _ => None,
529            };
530            for y in 0..=max_y {
531                for x in 0..=max_x {
532                    let v = coeffs[y * n + x];
533                    if v != 0 {
534                        let f = m.map_or(16, |m| m[y * n + x] as i64);
535                        coeffs[y * n + x] = ((v as i64 * f * scale + round) >> bd_shift).clamp(-32768, 32767) as i32;
536                    }
537                }
538            }
539            let bd_shift = 20 - bd;
540            if ts {
541                transform::transform_skip(coeffs, n, bd_shift);
542            } else if max_x == 0 && max_y == 0 && !(self.cu_intra && c == 0 && n == 4) {
543                let v = transform::inverse_dc(coeffs[0], bd_shift);
544                coeffs.fill(v);
545                dc_only = true;
546            } else {
547                let dst = self.cu_intra && c == 0 && n == 4;
548                transform::inverse_transform(coeffs, n, dst, max_x, max_y, bd_shift);
549            }
550        }
551        let _ = dc_only;
552        let (stride, max) = (if c == 0 { self.pic.width } else { self.pic.cwidth }, (1i32 << bd) - 1);
553        let plane = &mut self.pic.planes[c];
554        for y in 0..n {
555            let row = &mut plane[(y0 as usize + y) * stride + x0 as usize..][..n];
556            for (x, s) in row.iter_mut().enumerate() {
557                *s = S::from_i32((s.to_i32() + coeffs[y * n + x]).clamp(0, max));
558            }
559        }
560        Ok(())
561    }
