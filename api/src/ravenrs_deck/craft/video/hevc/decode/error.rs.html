3/// Errors produced by the decoder.
4#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
5pub enum Error {
6    /// The bitstream ended early or contained an invalid Exp-Golomb code.
7    #[error("bitstream error: {0}")]
8    Bitstream(#[from] BitError),
9    /// A syntax element had a value outside its legal range.
10    #[error("invalid bitstream: {0}")]
11    Invalid(String),
12    /// A valid feature this decoder does not implement (yet).
13    #[error("unsupported: {0}")]
14    Unsupported(String),
15    /// A slice referenced a parameter set that has not been received.
16    #[error("missing parameter set: {0}")]
17    MissingParameterSet(String),
18}
19
20pub type Result<T> = std::result::Result<T, Error>;
21
22pub(crate) fn invalid<T>(msg: impl Into<String>) -> Result<T> {
23    Err(Error::Invalid(msg.into()))
24}
25
26pub(crate) fn unsupported<T>(msg: impl Into<String>) -> Result<T> {
27    Err(Error::Unsupported(msg.into()))
28}
