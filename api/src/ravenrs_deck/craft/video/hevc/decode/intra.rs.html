7/// Reference samples of an nTbS block: `left[0]` = `top[0]` = p[-1][-1], `left[1 + y]` = p[-1][y],
8/// `top[1 + x]` = p[x][-1] for x, y = 0..2*nTbS-1.
9#[derive(Clone)]
10pub struct Refs {
11    pub left: [u16; 129],
12    pub top: [u16; 129],
13}
14
15impl Default for Refs {
16    fn default() -> Self {
17        Refs { left: [0; 129], top: [0; 129] }
18    }
19}

21/// Substitution process (8.4.4.2.2). `avail_left[1 + y]` / `avail_top[1 + x]` (index 0 = corner) tell
22/// which samples are available; unavailable ones are replaced.
23pub fn substitute(r: &mut Refs, avail_left: &[bool; 129], avail_top: &[bool; 129], n: usize, bit_depth: u32) {
24    let n2 = 2 * n;
25    // Sequence order: p[-1][2n-1] .. p[-1][-1], p[0][-1] .. p[2n-1][-1]
26    let any = avail_left[..=n2].iter().any(|&a| a) || avail_top[1..=n2].iter().any(|&a| a);
27    if !any {
28        let v = 1u16 << (bit_depth - 1);
29        r.left[..=n2].fill(v);
30        r.top[..=n2].fill(v);
31        return;
32    }
33    if !avail_left[n2] {
34        // search upwards then along the top
35        let mut found = None;
36        for y in (0..n2).rev() {
37            if avail_left[y] {
38                found = Some(r.left[y]);
39                break;
40            }
41        }
42        if found.is_none() {
43            for x in 1..=n2 {
44                if avail_top[x] {
45                    found = Some(r.top[x]);
46                    break;
47                }
48            }
49        }
50        r.left[n2] = found.unwrap_or(0);
51    }
52    for y in (0..n2).rev() {
53        if !avail_left[y] {
54            r.left[y] = r.left[y + 1];
55        }
56    }
57    r.top[0] = r.left[0];
58    for x in 1..=n2 {
59        if !avail_top[x] {
60            r.top[x] = r.top[x - 1];
61        }
62    }
63}

65/// Filtering process of neighbouring samples (8.4.4.2.3) for luma (and 4:4:4 chroma).
66pub fn filter(r: &mut Refs, mode: u32, n: usize, strong_enabled: bool, is_luma: bool, bit_depth: u32) {
67    if mode == 1 || n == 4 {
68        return;
69    }
70    let min_dist = (mode as i32 - 26).abs().min((mode as i32 - 10).abs());
71    let thres = match n {
72        8 => 7,
73        16 => 1,
74        _ => 0,
75    };
76    if min_dist <= thres {
77        return;
78    }
79    let n2 = 2 * n;
80    let c = r.left[0] as i32;
81    if strong_enabled && is_luma && n == 32 {
82        let (bl, tr) = (r.left[n2] as i32, r.top[n2] as i32);
83        let thr = 1 << (bit_depth - 5);
84        if (c + tr - 2 * r.top[n] as i32).abs() < thr && (c + bl - 2 * r.left[n] as i32).abs() < thr {
85            for i in 0..63 {
86                r.left[1 + i] = (((63 - i as i32) * c + (i as i32 + 1) * bl + 32) >> 6) as u16;
87                r.top[1 + i] = (((63 - i as i32) * c + (i as i32 + 1) * tr + 32) >> 6) as u16;
88            }
89            return;
90        }
91    }
92    let (l, t) = (r.left, r.top);
93    let corner = ((l[1] as u32 + 2 * c as u32 + t[1] as u32 + 2) >> 2) as u16;
94    for i in 1..n2 {
95        r.left[i] = ((l[i - 1] as u32 + 2 * l[i] as u32 + l[i + 1] as u32 + 2) >> 2) as u16;
96        r.top[i] = ((t[i - 1] as u32 + 2 * t[i] as u32 + t[i + 1] as u32 + 2) >> 2) as u16;
97    }
98    r.left[0] = corner;
99    r.top[0] = corner;
100}

102/// Predict an n x n block with `mode` into `out` (stride `os`). `edge_filters`: DC / horizontal /
103/// vertical boundary smoothing (luma blocks smaller than 32x32).
104pub fn predict<S: Sample>(r: &Refs, mode: u32, n: usize, edge_filters: bool, bit_depth: u32, out: &mut [S], os: usize) {
105    match n {
106        4 => predict_n::<S, 4>(r, mode, edge_filters, bit_depth, out, os),
107        8 => predict_n::<S, 8>(r, mode, edge_filters, bit_depth, out, os),
108        16 => predict_n::<S, 16>(r, mode, edge_filters, bit_depth, out, os),
109        _ => predict_n::<S, 32>(r, mode, edge_filters, bit_depth, out, os),
110    }
111}

113#[inline(always)]
114fn predict_n<S: Sample, const N: usize>(r: &Refs, mode: u32, edge_filters: bool, bit_depth: u32, out: &mut [S], os: usize) {
115    match mode {
116        0 => planar::<S, N>(r, out, os),
117        1 => {
118            let log2 = N.trailing_zeros();
119            let sum: u32 = r.top[1..=N].iter().map(|&v| v as u32).sum::<u32>() + r.left[1..=N].iter().map(|&v| v as u32).sum::<u32>();
120            let dc = ((sum + N as u32) >> (log2 + 1)) as u16;
121            let dcs = S::from_u16(dc);
122            for y in 0..N {
123                out[y * os..y * os + N].fill(dcs);
124            }
125            if edge_filters && N < 32 {
126                let d = dc as u32;
127                out[0] = S::from_i32(((r.left[1] as u32 + 2 * d + r.top[1] as u32 + 2) >> 2) as i32);
128                for x in 1..N {
129                    out[x] = S::from_i32(((r.top[1 + x] as u32 + 3 * d + 2) >> 2) as i32);
130                }
131                for y in 1..N {
132                    out[y * os] = S::from_i32(((r.left[1 + y] as u32 + 3 * d + 2) >> 2) as i32);
133                }
134            }
135        }
136        _ => angular::<S, N>(r, mode, edge_filters, bit_depth, out, os),
137    }
138}

140fn planar<S: Sample, const N: usize>(r: &Refs, out: &mut [S], os: usize) {
141    let log2 = N.trailing_zeros();
142    let tr = r.top[1 + N] as i32;
143    let bl = r.left[1 + N] as i32;
144    let top = &r.top[1..1 + N];
145    for y in 0..N {
146        let l = r.left[1 + y] as i32;
147        let (wt, wb) = ((N - 1 - y) as i32, (y + 1) as i32);
148        for (x, (o, &t)) in out[y * os..y * os + N].iter_mut().zip(top).enumerate() {
149            let v = (N - 1 - x) as i32 * l + (x as i32 + 1) * tr + wt * t as i32 + wb * bl + N as i32;
150            *o = S::from_i32(v >> (log2 + 1));
151        }
152    }
153}

155/// Angular modes 2..34 (8.4.4.2.6). Horizontal modes (< 18) are predicted as the vertical mode
156/// on the transposed references into a tile, then transposed into `out`.
157/// Row `j` of a vertical angular prediction from `ref[x] = refs[base + x]`.
158#[inline(always)]
159fn angular_row<const N: usize>(refs: &[u16], base: i32, angle: i32, j: usize) -> [u16; N] {
160    let pos = (j as i32 + 1) * angle;
161    let fact = (pos & 31) as u32;
162    let p = &refs[(base + (pos >> 5) + 1) as usize..][..N + 1];
163    let mut v = [0u16; N];
164    if fact == 0 {
165        v.copy_from_slice(&p[..N]);
166    } else {
167        for (v, (&a, &b)) in v.iter_mut().zip(p.iter().zip(&p[1..])) {
168            *v = (((32 - fact) * a as u32 + fact * b as u32 + 16) >> 5) as u16;
169        }
170    }
171    v
172}

174fn angular<S: Sample, const N: usize>(r: &Refs, mode: u32, edge_filters: bool, bit_depth: u32, out: &mut [S], os: usize) {
175    let angle = INTRA_PRED_ANGLE[mode as usize] as i32;
176    let (main, side) = if mode >= 18 { (&r.top, &r.left) } else { (&r.left, &r.top) };
177    // ref[x] = refs[base + x] for x = -N..=2N; ref[x] = main[x] for x >= 0
178    let mut ext = [0u16; 3 * 32 + 1];
179    let (refs, base): (&[u16], i32) = if angle < 0 {
180        ext[N..=2 * N].copy_from_slice(&main[..=N]);
181        let lim = (N as i32 * angle) >> 5;
182        if lim < -1 {
183            let inv = INV_ANGLE[(mode - 11) as usize] as i32;
184            for x in lim..=-1 {
185                ext[(N as i32 + x) as usize] = side[((x * inv + 128) >> 8) as usize];
186            }
187        }
188        (&ext, N as i32)
189    } else {
190        (main, 0)
191    };
192    if mode >= 18 {
193        for j in 0..N {
194            for (o, v) in out[j * os..j * os + N].iter_mut().zip(angular_row::<N>(refs, base, angle, j)) {
195                *o = S::from_u16(v);
196            }
197        }
198    } else {
199        let mut tile = [[0u16; N]; N];
200        for (j, t) in tile.iter_mut().enumerate() {
201            *t = angular_row::<N>(refs, base, angle, j);
202        }
203        for y in 0..N {
204            for (o, t) in out[y * os..y * os + N].iter_mut().zip(&tile) {
205                *o = S::from_u16(t[y]);
206            }
207        }
208    }
209    if edge_filters && N < 32 {
210        let max = (1i32 << bit_depth) - 1;
211        if mode == 26 {
212            for y in 0..N {
213                let v = r.top[1] as i32 + ((r.left[1 + y] as i32 - r.left[0] as i32) >> 1);
214                out[y * os] = S::from_i32(v.clamp(0, max));
215            }
216        } else if mode == 10 {
217            for x in 0..N {
218                let v = r.left[1] as i32 + ((r.top[1 + x] as i32 - r.top[0] as i32) >> 1);
219                out[x] = S::from_i32(v.clamp(0, max));
220            }
221        }
222    }
223}

271    /// The previous per-sample implementation of planar and angular prediction (8.4.4.2.4, 8.4.4.2.6).
272    fn reference(r: &Refs, mode: u32, n: usize, out: &mut [u16]) {
273        if mode == 0 {
274            let log2 = n.trailing_zeros();
275            let (tr, bl) = (r.top[1 + n] as i32, r.left[1 + n] as i32);
276            for y in 0..n {
277                for x in 0..n {
278                    let (l, t) = (r.left[1 + y] as i32, r.top[1 + x] as i32);
279                    out[y * n + x] = (((n - 1 - x) as i32 * l + (x as i32 + 1) * tr + (n - 1 - y) as i32 * t + (y as i32 + 1) * bl + n as i32)
280                        >> (log2 + 1)) as u16;
281                }
282            }
283            return;
284        }
285        let angle = INTRA_PRED_ANGLE[mode as usize] as i32;
286        let mut refa = [0i32; 3 * 64 + 1];
287        let base = n as i32;
288        let (main, side) = if mode >= 18 { (&r.top, &r.left) } else { (&r.left, &r.top) };
289        for x in 0..=n {
290            refa[(base + x as i32) as usize] = main[x] as i32;
291        }
292        if angle < 0 {
293            let lim = (n as i32 * angle) >> 5;
294            if lim < -1 {
295                let inv = INV_ANGLE[(mode - 11) as usize] as i32;
296                for x in lim..=-1 {
297                    refa[(base + x) as usize] = side[((x * inv + 128) >> 8) as usize] as i32;
298                }
299            }
300        } else {
301            for x in n + 1..=2 * n {
302                refa[(base + x as i32) as usize] = main[x] as i32;
303            }
304        }
305        for j in 0..n {
306            let pos = (j as i32 + 1) * angle;
307            let (idx, fact) = (pos >> 5, pos & 31);
308            for i in 0..n {
309                let k = (base + i as i32 + idx + 1) as usize;
310                let v = if fact != 0 { ((32 - fact) * refa[k] + fact * refa[k + 1] + 16) >> 5 } else { refa[k] };
311                if mode < 18 {
312                    out[i * n + j] = v as u16;
313                } else {
314                    out[j * n + i] = v as u16;
315                }
316            }
317        }
318    }
