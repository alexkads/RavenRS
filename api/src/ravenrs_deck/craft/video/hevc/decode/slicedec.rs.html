197    /// z-scan order availability (6.4.1) of luma location (xn, yn) for the block at (xc, yc) in the slice
198    /// with SliceAddrRs `addr`.
199    #[inline]
200    pub fn zavail(&self, xc: i32, yc: i32, xn: i32, yn: i32, addr: u32) -> bool {
201        if xn < 0 || yn < 0 || xn >= self.width as i32 || yn >= self.height as i32 {
202            return false;
203        }
204        let s = self.log2_ctb;
205        if (xn >> s) == (xc >> s) && (yn >> s) == (yc >> s) {
206            return zorder(xn, yn) <= zorder(xc, yc);
207        }
208        let ctb_n = (yn >> s) as usize * self.wctb + (xn >> s) as usize;
209        let ctb_c = (yc >> s) as usize * self.wctb + (xc >> s) as usize;
210        self.ctb_avail(ctb_n, ctb_c, addr)
211    }
212
213    /// Availability of CTB `ctb_n` for a block in another CTB `ctb_c` (raster addresses).
214    #[inline]
215    fn ctb_avail(&self, ctb_n: usize, ctb_c: usize, addr: u32) -> bool {
216        let tsn = self.layout.rs_to_ts[ctb_n];
217        let tsc = self.layout.rs_to_ts[ctb_c];
218        tsn < tsc && self.ctb_addr[ctb_n] == addr && self.layout.tile_id[tsn as usize] == self.layout.tile_id[tsc as usize]
219    }
220
221    /// `zavail` of the `len` 4x4 units starting at luma location (xn, yn) and going down (`down`)
222    /// or right in steps of 4 samples, as a mask (bit i = unit i, len <= 32). Inside one CTB
223    /// z-order grows along both directions, so the available units of a CTB form a prefix.
224    pub fn zavail_run(&self, (xc, yc): (i32, i32), (xn, yn): (i32, i32), down: bool, len: u32, addr: u32) -> u32 {
225        if xn < 0 || yn < 0 || xn >= self.width as i32 || yn >= self.height as i32 {
226            return 0;
227        }
228        let s = self.log2_ctb;
229        let (start, lim) = if down { (yn, self.height as i32) } else { (xn, self.width as i32) };
230        let len = len.min(((lim - start + 3) >> 2) as u32);
231        let ctb_c = (yc >> s) as usize * self.wctb + (xc >> s) as usize;
232        let zc = zorder(xc, yc);
233        let mut mask = 0u32;
234        let mut i = 0;
235        while i < len {
236            let p = start + 4 * i as i32;
237            let end = (((((p >> s) + 1) << s) - start) >> 2).min(len as i32) as u32;
238            let (x, y) = if down { (xn, p) } else { (p, yn) };
239            let ctb_n = (y >> s) as usize * self.wctb + (x >> s) as usize;
240            if ctb_n == ctb_c {
241                while i < end {
242                    let q = start + 4 * i as i32;
243                    let z = if down { zorder(xn, q) } else { zorder(q, yn) };
244                    if z > zc {
245                        break;
246                    }
247                    mask |= 1 << i;
248                    i += 1;
249                }
250            } else if self.ctb_avail(ctb_n, ctb_c, addr) {
251                mask |= (u32::MAX >> (32 - end)) & !((1u32 << i) - 1);
252            }
253            i = end;
254        }
255        mask
256    }
257
258    #[inline]
259    pub fn blk_at(&self, x: i32, y: i32) -> &Blk {
260        &self.blk[(y as usize >> 2) * self.w4 + (x as usize >> 2)]
261    }
262}

264/// Z-scan order of the 4x4 block containing luma location (x, y) inside its CTB. Bits above the
265/// CTB size are equal for two locations in the same CTB, so they do not change comparisons.
266#[inline(always)]
267fn zorder(x: i32, y: i32) -> u8 {
268    ZORDER[((y as usize >> 2) & 15) << 4 | ((x as usize >> 2) & 15)]
269}
270
271/// Calls `f(a, b)` for each run of set bits of `m`, with the run as sample range [a, b) given
272/// `unit` samples per bit.
273#[inline]
274fn for_runs(mut m: u32, unit: usize, mut f: impl FnMut(usize, usize)) {
275    while m != 0 {
276        let a = m.trailing_zeros();
277        let len = (m >> a).trailing_ones();
278        f(a as usize * unit, (a + len) as usize * unit);
279        m &= !((u32::MAX >> (32 - len)) << a);
280    }
281}

283/// Morton index of 4x4 block (x, y) of a 64x64 CTB at `[y * 16 + x]`.
284static ZORDER: [u8; 256] = {
285    let mut t = [0u8; 256];
286    let mut i = 0;
287    while i < 256 {
288        let (x, y) = (i & 15, i >> 4);
289        let mut r = 0;
290        let mut b = 0;
291        while b < 4 {
292            r |= ((x >> b) & 1) << (2 * b) | ((y >> b) & 1) << (2 * b + 1);
293            b += 1;
294        }
295        t[i] = r as u8;
296        i += 1;
297    }
298    t
299};

364    #[inline]
365    fn bypass_bits(&mut self, n: u32) -> u32 {
366        let mut v = 0u32;
367        let mut left = n;
368        while left > 16 {
369            v = (v << 16) | self.c.decode_bypass_bits(16);
370            left -= 16;
371        }
372        // (bits beyond 32 shift out, as the bin-by-bin loop did)
373        v.checked_shl(left).unwrap_or(0) | self.c.decode_bypass_bits(left)
374    }

376    /// Decode the slice segment `job` into `pic`.
377    pub fn decode(pic: &'a mut PicState<S>, job: &'a SliceJob) -> Result<()> {
378        let sh = &job.sh;
379        let (sps, pps) = (&*job.sps, &*job.pps);
380        ensure!(sh.data_offset < job.rbsp.len(), "empty slice segment data");
381        let slice_qp =
382            pps.init_qp.checked_add(sh.qp_delta).ok_or_else(|| crate::craft::video::hevc::decode::Error::Invalid("SliceQpY out of range".into()))?;
383        let init_type = match sh.slice_type {
384            SliceType::I => 0,
385            SliceType::P => {
386                if sh.cabac_init_flag {
387                    2
388                } else {
389                    1
390                }
391            }
392            SliceType::B => {
393                if sh.cabac_init_flag {
394                    1
395                } else {
396                    2
397                }
398            }
399        };
400        // SliceAddrRs: address of the first CTB of the (independent) slice
401        let slice_addr = if sh.dependent {
402            match pic.slices.last() {
403                Some(s) => s.addr_rs,
404                None => return invalid("dependent slice segment without a preceding slice"),
405            }
406        } else {
407            sh.segment_address
408        };
409        if !sh.dependent {
410            let ids = |l: usize| job.refs[l].iter().map(|r| r.frame.id).collect::<Vec<_>>();
411            let pocs = |l: usize| job.refs[l].iter().map(|r| r.poc).collect::<Vec<_>>();
412            let lts = |l: usize| job.refs[l].iter().map(|r| r.long_term).collect::<Vec<_>>();
413            pic.slices.push(SliceInfo {
414                addr_rs: slice_addr,
415                deblock_disabled: sh.deblocking_disabled,
416                beta_offset: sh.beta_offset_div2 * 2,
417                tc_offset: sh.tc_offset_div2 * 2,
418                lf_across: sh.loop_filter_across_slices,
419                sao_luma: sh.sao_luma,
420                sao_chroma: sh.sao_chroma,
421                ref_ids: [ids(0), ids(1)],
422                ref_pocs: [pocs(0), pocs(1)],
423                ref_lt: [lts(0), lts(1)],
424            });
425            if !sh.deblocking_disabled {
426                pic.deblocking_enabled_anywhere = true;
427            }
428        }
429        let slice_idx = (pic.slices.len() - 1) as u32;
430        let col = if sh.temporal_mvp && !sh.is_intra() {
431            let l = if sh.is_b() && !sh.collocated_from_l0 { 1 } else { 0 };
432            job.refs[l].get(sh.collocated_ref_idx as usize).cloned()
433        } else {
434            None
435        };
436        let no_backward_pred = job.refs.iter().flatten().all(|r| r.poc <= pic.poc);
437        let log2_min_cu_qp_delta = sps.log2_ctb - pps.diff_cu_qp_delta_depth;
438        let c = Cabac::new(&job.rbsp, sh.data_offset, slice_qp, init_type)?;
439        let mut d = SliceDecoder {
440            pic,
441            sh,
442            sps,
443            pps,
444            refs: &job.refs,
445            c,
446            slice_idx,
447            slice_addr,
448            slice_qp,
449            init_type,
450            qp_y: slice_qp,
451            qp_y_pred: slice_qp,
452            last_qp: slice_qp,
453            is_cu_qp_delta_coded: false,
454            cu_qp_delta: 0,
455            log2_min_cu_qp_delta,
456            qp_cb: 0,
457            qp_cr: 0,
458            cu_transquant_bypass: false,
459            cu_intra: false,
460            intra_chroma_mode: 0,
461            max_trafo_depth: 0,
462            intra_split: false,
463            cu_part: PartMode::P2Nx2N,
464            coeffs: vec![0; 32 * 32],
465            pred0: vec![0; 64 * 64],
466            pred1: vec![0; 64 * 64],
467            mc: Default::default(),
468            refs_buf: Refs::default(),
469            col,
470            no_backward_pred,
471            ctb_x: 0,
472            ctb_y: 0,
473        };
474        d.decode_data()
475    }

477    fn decode_data(&mut self) -> Result<()> {
478        let layout = self.pic.layout.clone();
479        let wctb = self.pic.wctb as u32;
480        let npic = (self.pic.wctb * self.pic.hctb) as u32;
481        let mut ts = layout.rs_to_ts[self.sh.segment_address as usize];
482        let mut first = true;
483        loop {
484            let rs = layout.ts_to_rs[ts as usize];
485            ensure!(self.pic.ctb_slice[rs as usize] == u32::MAX, "CTB {rs} decoded twice");
486            let (rx, ry) = ((rs % wctb) as i32, (rs / wctb) as i32);
487            let ctb = 1i32 << self.sps.log2_ctb;
488            self.ctb_x = rx * ctb;
489            self.ctb_y = ry * ctb;
490            self.pic.ctb_slice[rs as usize] = self.slice_idx;
491            self.pic.ctb_addr[rs as usize] = self.slice_addr;
492            // context initialisation / synchronisation (9.3.1, 9.3.2.1)
493            let first_in_tile = ts == 0 || layout.tile_id[ts as usize] != layout.tile_id[ts as usize - 1];
494            let wpp_row_start = self.pps.entropy_coding_sync
495                && (rs.is_multiple_of(wctb) || layout.tile_id[ts as usize] != layout.tile_id[layout.rs_to_ts[rs as usize - 1] as usize]);
496            if first_in_tile {
497                if !first {
498                    init_contexts(&mut self.c.ctx, self.slice_qp, self.init_type);
499                }
500                self.last_qp = self.slice_qp;
501            } else if wpp_row_start {
502                let (x0, y0) = (self.ctb_x, self.ctb_y);
503                let avail = self.pic.zavail(x0, y0, x0 + ctb, y0 - ctb, self.slice_addr);
504                match (&self.pic.wpp_ctx, avail) {
505                    (Some(saved), true) => self.c.ctx = **saved,
506                    _ => init_contexts(&mut self.c.ctx, self.slice_qp, self.init_type),
507                }
508                self.last_qp = self.slice_qp;
509            } else if first && self.sh.dependent {
510                match &self.pic.ds_ctx {
511                    Some((saved, qp)) => {
512                        self.c.ctx = **saved;
513                        self.last_qp = *qp;
514                    }
515                    None => return invalid("dependent slice segment without stored contexts"),
516                }
517            }
518            first = false;
519            if self.sh.sao_luma || self.sh.sao_chroma {
520                self.parse_sao(rx, ry, rs)?;
521            }
522            self.coding_quadtree(self.ctb_x, self.ctb_y, self.sps.log2_ctb, 0)?;
523            let end = self.c.decode_terminate() == 1;
524            ensure!(!self.c.overrun(), "slice data overrun");
525            // WPP storage after the second CTB of a row in a tile (9.3.1)
526            if self.pps.entropy_coding_sync
527                && (rs % wctb == 1 || (rs > 1 && layout.tile_id[ts as usize] != layout.tile_id[layout.rs_to_ts[rs as usize - 2] as usize]))
528            {
529                self.pic.wpp_ctx = Some(Box::new(self.c.ctx));
530            }
531            crate::craft::video::hevc::decode::filter::ctb_decoded(self.pic, ry as usize);
532            ts += 1;
533            if end {
534                if self.pps.dependent_slice_segments_enabled {
535                    self.pic.ds_ctx = Some((Box::new(self.c.ctx), self.last_qp));
536                }
537                break;
538            }
539            ensure!(ts < npic, "slice data continues past the last CTB");
540            let nrs = layout.ts_to_rs[ts as usize];
541            let new_tile = self.pps.tiles_enabled && layout.tile_id[ts as usize] != layout.tile_id[ts as usize - 1];
542            let new_row = self.pps.entropy_coding_sync
543                && (nrs.is_multiple_of(wctb) || layout.tile_id[ts as usize] != layout.tile_id[layout.rs_to_ts[nrs as usize - 1] as usize]);
544            if new_tile || new_row {
545                ensure!(self.c.decode_terminate() == 1, "end_of_subset_one_bit is 0");
546                let pos = self.c.bit_pos().div_ceil(8);
547                self.c.set_byte_pos(pos);
548                self.c.init_engine()?;
549            }
550        }
551        Ok(())
552    }
553
554    fn parse_sao(&mut self, rx: i32, ry: i32, rs: u32) -> Result<()> {
555        let layout = self.pic.layout.clone();
556        let wctb = self.pic.wctb as u32;
557        let ts = layout.rs_to_ts[rs as usize];
558        let mut merge_left = false;
559        let mut merge_up = false;
560        if rx > 0 {
561            let left_in_slice = self.pic.ctb_addr[rs as usize - 1] == self.slice_addr;
562            let left_in_tile = layout.tile_id[ts as usize] == layout.tile_id[layout.rs_to_ts[rs as usize - 1] as usize];
563            if left_in_slice && left_in_tile {
564                merge_left = self.dd(SAO_MERGE) == 1;
565            }
566        }
567        if ry > 0 && !merge_left {
568            let up = rs - wctb;
569            let up_in_slice = self.pic.ctb_addr[up as usize] == self.slice_addr;
570            let up_in_tile = layout.tile_id[ts as usize] == layout.tile_id[layout.rs_to_ts[up as usize] as usize];
571            if up_in_slice && up_in_tile {
572                merge_up = self.dd(SAO_MERGE) == 1;
573            }
574        }
575        if merge_left {
576            self.pic.sao[rs as usize] = self.pic.sao[rs as usize - 1];
577            return Ok(());
578        }
579        if merge_up {
580            self.pic.sao[rs as usize] = self.pic.sao[(rs - wctb) as usize];
581            return Ok(());
582        }
583        let mut p = SaoParams::default();
584        for c in 0..3 {
585            let enabled = if c == 0 { self.sh.sao_luma } else { self.sh.sao_chroma };
586            if !enabled {
587                continue;
588            }
589            if c == 2 {
590                p.type_idx[2] = p.type_idx[1];
591                p.eo_class[2] = p.eo_class[1];
592            } else {
593                p.type_idx[c] = if self.dd(SAO_TYPE) == 0 {
594                    0
595                } else if self.bypass() == 0 {
596                    1
597                } else {
598                    2
599                };
600            }
601            if p.type_idx[c] == 0 {
602                continue;
603            }
604            let bd = if c == 0 { self.pic.bd_y } else { self.pic.bd_c };
605            let cmax = (1u32 << (bd.min(10) - 5)) - 1;
606            let mut abs = [0i16; 4];
607            for a in abs.iter_mut() {
608                let mut v = 0;
609                while v < cmax && self.bypass() == 1 {
610                    v += 1;
611                }
612                *a = v as i16;
613            }
614            if p.type_idx[c] == 1 {
615                for a in abs.iter_mut() {
616                    if *a != 0 && self.bypass() == 1 {
617                        *a = -*a;
618                    }
619                }
620                p.band_pos[c] = self.bypass_bits(5) as u8;
621                p.offsets[c] = abs;
622            } else {
623                p.offsets[c] = [abs[0], abs[1], -abs[2], -abs[3]];
624                if c == 0 {
625                    p.eo_class[0] = self.bypass_bits(2) as u8;
626                }
627                if c == 1 {
628                    p.eo_class[1] = self.bypass_bits(2) as u8;
629                }
630            }
631        }
632        self.pic.sao[rs as usize] = p;
633        Ok(())
634    }
