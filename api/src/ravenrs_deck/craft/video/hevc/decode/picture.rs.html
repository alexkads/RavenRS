8/// Motion data of one 16x16 block of a decoded picture (8.5.3.2.9 uses the motion of the top-left
9/// 4x4 block of each 16x16 block).
10#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
11pub struct ColMv {
12    pub mv: [[i16; 2]; 2],
13    /// POC of the referenced picture per list.
14    pub ref_poc: [i32; 2],
15    /// bit0/1: predFlagL0/L1, bit2/3: reference is long-term (L0/L1). 0 = intra / unavailable.
16    pub flags: u8,
17}

19/// Storage type of decoded samples: `u8` for 8-bit pictures (half the memory and cache traffic
20/// of 16-bit storage in prediction reads, filters and output), `u16` for higher bit depths.
21/// Values are always within `0..1 << bit_depth`.
22pub trait Sample: Copy + Default + PartialEq + std::fmt::Debug + Send + Sync + 'static {
23    fn to_i32(self) -> i32;
24    fn to_i16(self) -> i16;
25    fn to_u16(self) -> u16;
26    /// `v` (a sample value, already clipped to the bit depth) in the storage type.
27    fn from_i32(v: i32) -> Self;
28    /// `v` (a sample value, within the bit depth) in the storage type.
29    fn from_u16(v: u16) -> Self;
30    /// The rows of `f` if they are stored with this type.
31    fn rows(f: &Frame) -> Option<&[OnceLock<FrameRow<Self>>]>;
32    fn wrap_rows(rows: Box<[OnceLock<FrameRow<Self>>]>) -> Rows;
33}

101/// Whether pictures with these bit depths are stored as `u8` (both 8-bit), else as `u16`.
102pub fn stores_u8(bit_depth: u32, bit_depth_c: u32) -> bool {
103    bit_depth <= 8 && bit_depth_c <= 8
104}

106/// One published CTB row.
107pub struct FrameRow<S> {
108    /// Luma lines of the row (stride = picture width).
109    pub y: Box<[S]>,
110    pub cb: Box<[S]>,
111    pub cr: Box<[S]>,
112    /// Motion per 16x16 block: (width16 * rows16) entries, raster within the row.
113    pub col: Box<[ColMv]>,
114}

116/// The CTB rows of a frame, in its storage type.
117#[derive(Clone)]
118pub enum Rows {
119    U8(Arc<[OnceLock<FrameRow<u8>>]>),
120    U16(Arc<[OnceLock<FrameRow<u16>>]>),
121}

123/// A decoded (or in-progress) picture as stored in the DPB and referenced by later pictures.
124pub struct Frame {

126    pub poc: i32,

128    pub height: usize,
129    pub cwidth: usize,
130    pub cheight: usize,
131    pub chroma: ChromaFormat,
132    pub log2_ctb: u32,

138/// The value of a published row (waiting for it with the `threads` feature).
139#[inline(always)]
140fn wait<T>(cell: &OnceLock<T>) -> Option<&T> {
141    {
142        Some(cell.wait())
143    }
144    #[cfg(any())]
145    {
146        cell.get()
147    }
148}

150impl Frame {
151    #[allow(clippy::too_many_arguments)]
152    pub fn new(id: u32, poc: i32, width: usize, height: usize, chroma: ChromaFormat, log2_ctb: u32, bit_depth: u32, bit_depth_c: u32) -> Self {
153        let n = height.div_ceil(1 << log2_ctb);
154        fn cells<S: Sample>(n: usize) -> Rows {
155            S::wrap_rows((0..n).map(|_| OnceLock::new()).collect())
156        }
157        let rows = if stores_u8(bit_depth, bit_depth_c) { cells::<u8>(n) } else { cells::<u16>(n) };
158        let (sx, sy) = chroma.shifts();
159        Frame { id, poc, width, height, cwidth: width >> sx, cheight: height >> sy, chroma, log2_ctb, bit_depth, bit_depth_c, rows }
160    }

162    /// log2 of the chroma lines per CTB row (log2(CtbSizeY / SubHeightC)).
163    #[inline(always)]
164    pub fn log2_ctb_c(&self) -> u32 {
165        self.log2_ctb - self.chroma.shifts().1
166    }
167
168    pub fn num_rows(&self) -> usize {
169        match &self.rows {
170            Rows::U8(r) => r.len(),
171            Rows::U16(r) => r.len(),
172        }
173    }
174
175    /// Number of luma lines in CTB row `r`.
176    pub fn row_lines(&self, r: usize) -> usize {
177        let ctb = 1usize << self.log2_ctb;
178        ctb.min(self.height.saturating_sub(r * ctb))
179    }
180
181    pub fn width16(&self) -> usize {
182        self.width.div_ceil(16)
183    }

185    /// Publish CTB row `r` (ignored if already published, or not of the frame's sample type).
186    pub fn publish<S: Sample>(&self, r: usize, row: FrameRow<S>) {
187        if let Some(cell) = S::rows(self).and_then(|rows| rows.get(r)) {
188            let _ = cell.set(row);
189        }
190    }
191
192    pub fn is_published(&self, r: usize) -> bool {
193        match &self.rows {
194            Rows::U8(rows) => rows.get(r).is_some_and(|c| c.get().is_some()),
195            Rows::U16(rows) => rows.get(r).is_some_and(|c| c.get().is_some()),
196        }
197    }
198
199    pub fn is_complete(&self) -> bool {
200        let n = self.num_rows();
201        n == 0 || self.is_published(n - 1)
202    }

204    /// CTB row `r` in sample type `S`, waiting until it has been published (None: no such row,
205    /// or the frame stores another sample type).
206    #[inline]
207    pub fn row<S: Sample>(&self, r: usize) -> Option<&FrameRow<S>> {
208        wait(S::rows(self)?.get(r)?)
209    }

211    /// The motion of CTB row `r`, waiting until it has been published.
212    #[inline]
213    fn col_row(&self, r: usize) -> Option<&[ColMv]> {
214        match &self.rows {
215            Rows::U8(rows) => wait(rows.get(r)?).map(|row| &*row.col),
216            Rows::U16(rows) => wait(rows.get(r)?).map(|row| &*row.col),
217        }
218    }
219
220    pub fn wait_complete(&self) {
221        for r in 0..self.num_rows() {
222            let _ = self.col_row(r);
223        }
224    }

226    /// Motion of the 16x16 block containing luma sample (x, y) (inside the picture).
227    #[inline]
228    pub fn col_mv(&self, x: usize, y: usize) -> ColMv {
229        let r = y >> self.log2_ctb;
230        let y16 = (y & ((1 << self.log2_ctb) - 1)) >> 4;
231        self.col_row(r).and_then(|col| col.get(y16 * self.width16() + (x >> 4))).copied().unwrap_or_default()
232    }

234    /// Copy a w x h luma window at (x0, y0) with edge replication into `out` (stride `os`).
235    pub fn luma_window<S: Sample>(&self, (x0, y0): (i32, i32), (w, h): (usize, usize), out: &mut [i16], os: usize) {
236        let (width, height) = (self.width as i32, self.height as i32);
237        let sh = self.log2_ctb;
238        let mask = (1i32 << sh) - 1;
239        for r in 0..h {
240            let yy = (y0 + r as i32).clamp(0, height - 1);
241            let Some(row) = self.row::<S>((yy >> sh) as usize) else { return };
242            let off = ((yy & mask) * width) as usize;
243            copy_line(&row.y[off..off + width as usize], x0, &mut out[r * os..r * os + w]);
244        }
245    }

247    /// Copy a w x h window of chroma component `c` (0 = Cb, 1 = Cr) with edge replication.
248    pub fn chroma_window<S: Sample>(&self, c: usize, (x0, y0): (i32, i32), (w, h): (usize, usize), out: &mut [i16], os: usize) {
249        let (cw, ch) = (self.cwidth as i32, self.cheight as i32);
250        let sh = self.log2_ctb_c();
251        let mask = (1i32 << sh) - 1;
252        for r in 0..h {
253            let yy = (y0 + r as i32).clamp(0, ch - 1);
254            let Some(row) = self.row::<S>((yy >> sh) as usize) else { return };
255            let plane = if c == 0 { &row.cb } else { &row.cr };
256            let off = ((yy & mask) * cw) as usize;
257            copy_line(&plane[off..off + cw as usize], x0, &mut out[r * os..r * os + w]);
258        }
259    }

261    /// The w-sample runs at x0 of the h lines from y0 of plane `c` (0 = luma, 1 = Cb, 2 = Cr) into
262    /// `rows` (h = `rows.len()`), lines outside the picture replicated from its top / bottom edge
263    /// (8-228, 8-229): the reference window of a prediction block without copying it. False (and
264    /// `rows` partly filled) when the runs are not inside the picture horizontally.
265    pub fn window_rows<'f, S: Sample>(&'f self, c: usize, x0: i32, y0: i32, w: usize, rows: &mut [&'f [S]]) -> bool {
266        let (pw, ph, sh) = if c == 0 { (self.width, self.height, self.log2_ctb) } else { (self.cwidth, self.cheight, self.log2_ctb_c()) };
267        let Some(x) = usize::try_from(x0).ok().filter(|&x| x.saturating_add(w) <= pw) else { return false };
268        let mask = (1usize << sh) - 1;
269        let max_y = ph as i32 - 1;
270        let mut cur: Option<(usize, &'f [S])> = None;
271        for (r, slot) in rows.iter_mut().enumerate() {
272            let yy = y0.saturating_add(r as i32).min(max_y).max(0) as usize;
273            let plane = match cur {
274                Some((i, p)) if i == yy >> sh => p,
275                _ => {
276                    let Some(row) = self.row::<S>(yy >> sh) else { return false };
277                    let p: &'f [S] = match c {
278                        0 => &row.y,
279                        1 => &row.cb,
280                        _ => &row.cr,
281                    };
282                    cur = Some((yy >> sh, p));
283                    p
284                }
285            };
286            let off = (yy & mask) * pw + x;
287            let Some(run) = plane.get(off..off + w) else { return false };
288            *slot = run;
289        }
290        true
291    }

293    /// Copy the w x h block at (x0, y0) of plane `c` (0 = luma, 1 = Cb, 2 = Cr) into `dst`
294    /// (stride `ds`), samples outside the picture replicated from its edges (8-228, 8-229).
295    pub fn copy_block<S: Sample>(&self, c: usize, (x0, y0): (i32, i32), (w, h): (usize, usize), dst: &mut [S], ds: usize) {
296        let (pw, ph, sh) = if c == 0 { (self.width, self.height, self.log2_ctb) } else { (self.cwidth, self.cheight, self.log2_ctb_c()) };
297        let inside = usize::try_from(x0).ok().filter(|&x| x.saturating_add(w) <= pw);
298        for (r, d) in dst.chunks_mut(ds).take(h).enumerate() {
299            let yy = y0.saturating_add(r as i32).min(ph as i32 - 1).max(0) as usize;
300            let Some(row) = self.row::<S>(yy >> sh) else { return };
301            let plane = match c {
302                0 => &row.y,
303                1 => &row.cb,
304                _ => &row.cr,
305            };
306            let off = (yy & ((1 << sh) - 1)) * pw;
307            let Some(line) = plane.get(off..off + pw) else { return };
308            let Some(d) = d.get_mut(..w) else { return };
309            match inside.and_then(|x| line.get(x..x + w)) {
310                Some(s) => copy_run(s, d),
311                None => {
312                    let max = (pw as i32 - 1).max(0);
313                    for (i, d) in d.iter_mut().enumerate() {
314                        *d = line.get(x0.saturating_add(i as i32).clamp(0, max) as usize).copied().unwrap_or_default();
315                    }
316                }
317            }
318        }
319    }

321    /// Crop the frame into planar buffers (waits for completion): 8-bit planes when `eight_bit`,
322    /// else 16-bit ones. Rows stored in the output type are copied as they are.
323    pub fn cropped_planes(&self, crop: (usize, usize, usize, usize), eight_bit: bool) -> (Plane, Plane, Plane) {
324        fn planes<S: Sample, T>(
325            f: &Frame,
326            crop: (usize, usize, usize, usize),
327            conv: impl Fn(S) -> T,
328            wrap: impl Fn(Vec<T>) -> Plane,
329        ) -> (Plane, Plane, Plane) {
330            let (y, u, v) = f.copy_cropped::<S, T>(crop, conv);
331            (wrap(y), wrap(u), wrap(v))
332        }
333        match (&self.rows, eight_bit) {
334            (Rows::U8(_), true) => planes::<u8, u8>(self, crop, |v| v, Plane::U8),
335            (Rows::U8(_), false) => planes::<u8, u16>(self, crop, u16::from, Plane::U16),
336            // (8-bit luma with deeper chroma: the 8-bit output keeps the low bits, as before)
337            (Rows::U16(_), true) => planes::<u16, u8>(self, crop, |v| v as u8, Plane::U8),
338            (Rows::U16(_), false) => planes::<u16, u16>(self, crop, |v| v, Plane::U16),
339        }
340    }

342    /// The cropped planes of the frame converted sample by sample with `conv` (a plain copy for
343    /// the identity).
344    fn copy_cropped<S: Sample, T>(&self, crop: (usize, usize, usize, usize), conv: impl Fn(S) -> T) -> (Vec<T>, Vec<T>, Vec<T>) {
345        let (cx, cy, cw, ch) = crop;
346        let ctb = 1usize << self.log2_ctb;
347        let (mut y, mut u, mut v) = (Vec::new(), Vec::new(), Vec::new());
348        let put = |o: &mut Vec<T>, plane: &[S], off: usize, n: usize| {
349            if let Some(s) = plane.get(off..off.saturating_add(n)) {
350                o.extend(s.iter().map(|&v| conv(v)));
351            }
352        };
353        y.reserve_exact(cw.saturating_mul(ch));
354        for r in cy..cy.saturating_add(ch) {
355            let Some(row) = self.row::<S>(r / ctb) else { break };
356            put(&mut y, &row.y, (r % ctb) * self.width + cx, cw);
357        }
358        let (sw, sh) = (self.chroma.sub_width(), self.chroma.sub_height());
359        let (ccx, ccy, ccw, cch) = (cx / sw, cy / sh, cw.div_ceil(sw), ch.div_ceil(sh));
360        let cctb = ctb / sh;
361        u.reserve_exact(ccw.saturating_mul(cch));
362        v.reserve_exact(ccw.saturating_mul(cch));
363        for r in ccy..ccy.saturating_add(cch) {
364            let Some(row) = self.row::<S>(r / cctb) else { break };
365            let o = (r % cctb) * self.cwidth + ccx;
366            put(&mut u, &row.cb, o, ccw);
367            put(&mut v, &row.cr, o, ccw);
368        }
369        (y, u, v)
370    }

372    /// Publish every row not published yet as mid-grey without motion (a missing reference picture,
373    /// or a picture whose decoding failed: readers never block on it).
374    pub fn publish_gray(&self) {
375        fn fill<S: Sample>(f: &Frame) {
376            let (gy, gc) = (S::from_u16(1 << (f.bit_depth.clamp(1, 16) - 1)), S::from_u16(1 << (f.bit_depth_c.clamp(1, 16) - 1)));
377            let w16 = f.width16();
378            for r in 0..f.num_rows() {
379                if f.is_published(r) {
380                    continue;
381                }
382                let lines = f.row_lines(r);
383                let clines = lines.div_ceil(f.chroma.sub_height());
384                f.publish(
385                    r,
386                    FrameRow::<S> {
387                        y: vec![gy; lines * f.width].into(),
388                        cb: vec![gc; clines * f.cwidth].into(),
389                        cr: vec![gc; clines * f.cwidth].into(),
390                        col: vec![ColMv::default(); w16 * lines.div_ceil(16)].into(),
391                    },
392                );
393            }
394        }
395        match &self.rows {
396            Rows::U8(_) => fill::<u8>(self),
397            Rows::U16(_) => fill::<u16>(self),
398        }
399    }

426    /// A frame with every row published from full-picture planes and no motion.
427    #[cfg(any())]
428    pub fn from_planes(id: u32, poc: i32, like: &Frame, y: &[u16], cb: &[u16], cr: &[u16]) -> Frame {
429        fn fill<S: Sample>(f: &Frame, y: &[u16], cb: &[u16], cr: &[u16]) {
430            let ctb = 1usize << f.log2_ctb;
431            let sh = f.chroma.sub_height();
432            let w16 = f.width16();
433            let conv = |s: &[u16]| -> Box<[S]> { s.iter().map(|&v| S::from_u16(v)).collect() };
434            for r in 0..f.num_rows() {
435                let lines = f.row_lines(r);
436                let (y0, y1) = (r * ctb * f.width, (r * ctb + lines) * f.width);
437                let clines = lines.div_ceil(sh);
438                let (c0, c1) = (r * ctb / sh * f.cwidth, (r * ctb / sh + clines) * f.cwidth);
439                f.publish(
440                    r,
441                    FrameRow::<S> {
442                        y: conv(&y[y0..y1]),
443                        cb: conv(&cb[c0..c1]),
444                        cr: conv(&cr[c0..c1]),
445                        col: vec![ColMv::default(); w16 * lines.div_ceil(16)].into(),
446                    },
447                );
448            }
449        }
450        let f = Frame::new(id, poc, like.width, like.height, like.chroma, like.log2_ctb, like.bit_depth, like.bit_depth_c);
451        match &f.rows {
452            Rows::U8(_) => fill::<u8>(&f, y, cb, cr),
453            Rows::U16(_) => fill::<u16>(&f, y, cb, cr),
454        }
455        f
456    }

459/// `dst.copy_from_slice(src)` for the run lengths of prediction blocks as fixed-size copies: a
460/// libc `memcpy` call per short block line costs more than the copy.
461#[inline(always)]
462fn copy_run<S: Sample>(src: &[S], dst: &mut [S]) {
463    #[inline(always)]
464    fn fixed<S: Sample, const N: usize>(src: &[S], dst: &mut [S]) {
465        if let (Some(s), Some(d)) = (src.first_chunk::<N>(), dst.first_chunk_mut::<N>()) {
466            *d = *s;
467        }
468    }
469    match dst.len() {
470        2 => fixed::<S, 2>(src, dst),
471        4 => fixed::<S, 4>(src, dst),
472        6 => fixed::<S, 6>(src, dst),
473        8 => fixed::<S, 8>(src, dst),
474        12 => fixed::<S, 12>(src, dst),
475        16 => fixed::<S, 16>(src, dst),
476        24 => fixed::<S, 24>(src, dst),
477        32 => fixed::<S, 32>(src, dst),
478        48 => fixed::<S, 48>(src, dst),
479        64 => fixed::<S, 64>(src, dst),
480        n => {
481            if let Some(s) = src.get(..n) {
482                dst.copy_from_slice(s);
483            }
484        }
485    }
486}

488/// Copy `out.len()` samples of `line` starting at x0, replicating edge samples outside the line.
489#[inline(always)]
490fn copy_line<S: Sample>(line: &[S], x0: i32, out: &mut [i16]) {
491    let w = out.len();
492    if x0 >= 0 && x0 as usize + w <= line.len() {
493        let src = &line[x0 as usize..x0 as usize + w];
494        // fixed-size copies for the window widths (block width + 7 luma, + 3 chroma): see copy_run
495        #[inline(always)]
496        fn fixed<S: Sample, const N: usize>(src: &[S], out: &mut [i16]) {
497            if let (Some(s), Some(o)) = (src.first_chunk::<N>(), out.first_chunk_mut::<N>()) {
498                for j in 0..N {
499                    o[j] = s[j].to_i16();
500                }
501            }
502        }
503        match w {
504            5 => fixed::<S, 5>(src, out),
505            7 => fixed::<S, 7>(src, out),
506            9 => fixed::<S, 9>(src, out),
507            11 => fixed::<S, 11>(src, out),
508            15 => fixed::<S, 15>(src, out),
509            19 => fixed::<S, 19>(src, out),
510            23 => fixed::<S, 23>(src, out),
511            27 => fixed::<S, 27>(src, out),
512            31 => fixed::<S, 31>(src, out),
513            35 => fixed::<S, 35>(src, out),
514            39 => fixed::<S, 39>(src, out),
515            55 => fixed::<S, 55>(src, out),
516            71 => fixed::<S, 71>(src, out),
517            _ => {
518                for (o, &s) in out.iter_mut().zip(src) {
519                    *o = s.to_i16();
520                }
521            }
522        }
523    } else {
524        // the window crosses the left and / or right picture edge: the edge samples repeated
525        // outside, the samples inside copied as a run
526        let (Some(&first), Some(&last)) = (line.first(), line.last()) else { return };
527        let n = line.len() as i64;
528        let (x0, w) = (x0 as i64, w as i64);
529        // out[i] is inside the line for i in lo..hi
530        let lo = (-x0).clamp(0, w) as usize;
531        let hi = (n - x0).clamp(0, w) as usize;
532        let (left, rest) = out.split_at_mut(lo);
533        left.fill(first.to_i16());
534        let (mid, right) = rest.split_at_mut(hi.saturating_sub(lo));
535        let start = (x0 + lo as i64).max(0) as usize;
536        if let Some(src) = line.get(start..start + mid.len()) {
537            for (o, &s) in mid.iter_mut().zip(src) {
538                *o = s.to_i16();
539            }
540        }
541        right.fill(last.to_i16());
542    }
543}

554    fn planes(w: usize, h: usize, chroma: ChromaFormat, bd: u32, seed: u64) -> (Vec<u16>, Vec<u16>, Vec<u16>) {
555        let (cw, ch) = (w / chroma.sub_width(), h / chroma.sub_height());
556        let mut x = seed;
557        let mut next = move || {
558            x ^= x << 13;
559            x ^= x >> 7;
560            x ^= x << 17;
561            (x % (1 << bd)) as u16
562        };
563        let y = (0..w * h).map(|_| next()).collect();
564        let cb = (0..cw * ch).map(|_| next()).collect();
565        let cr = (0..cw * ch).map(|_| next()).collect();
566        (y, cb, cr)
567    }

569    /// 8-bit pictures are stored as `u8`, deeper ones as `u16`, and both crop back to exactly the
570    /// samples they were given, in 4:2:0 and 4:2:2.
571    #[cfg(any())]
572    fn storage_type_follows_the_bit_depth_and_crops_exactly() {
573        let (w, h) = (40usize, 36usize);
574        let crop = (2usize, 4usize, 33usize, 27usize);
575        for (bd, chroma) in [(8u32, ChromaFormat::C420), (10, ChromaFormat::C420), (8, ChromaFormat::C422), (10, ChromaFormat::C422)] {
576            let (sw, sh) = (chroma.sub_width(), chroma.sub_height());
577            let (y, cb, cr) = planes(w, h, chroma, bd, 0x9e37_79b9_7f4a_7c15 + bd as u64);
578            let like = Frame::new(0, 0, w, h, chroma, 4, bd, bd);
579            let f = Frame::from_planes(1, 0, &like, &y, &cb, &cr);
580            assert_eq!(matches!(f.rows, Rows::U8(_)), bd == 8);
581            assert!(f.is_complete());
582            let (py, pu, pv) = f.cropped_planes(crop, bd == 8);
583            let (cx, cy, cw, ch) = crop;
584            let want_y: Vec<u16> = (cy..cy + ch).flat_map(|r| y[r * w + cx..r * w + cx + cw].to_vec()).collect();
585            let (ccw, cch) = (cw.div_ceil(sw), ch.div_ceil(sh));
586            let pw = w / sw;
587            let want_c =
588                |p: &[u16]| -> Vec<u16> { (cy / sh..cy / sh + cch).flat_map(|r| p[r * pw + cx / sw..r * pw + cx / sw + ccw].to_vec()).collect() };
589            let got = |p: &Plane| -> Vec<u16> { (0..p.len()).map(|i| p.get(i)).collect() };
590            assert_eq!(got(&py), want_y, "luma bd {bd} {chroma:?}");
591            assert_eq!(got(&pu), want_c(&cb), "cb bd {bd} {chroma:?}");
592            assert_eq!(got(&pv), want_c(&cr), "cr bd {bd} {chroma:?}");
593            assert_eq!(py.as_u8().is_some(), bd == 8);
594        }
595    }
