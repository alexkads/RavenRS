2//! and temporal motion vector prediction (8.5.3.2.8, 8.5.3.2.9).

4use crate::craft::video::hevc::decode::picture::RefPic;
5use crate::craft::video::hevc::decode::slicedec::{F_INTRA, MvField, PartMode, SliceDecoder};

7/// l0CandIdx / l1CandIdx of Table 8-7.
8const COMB: [(usize, usize); 12] = [(0, 1), (1, 0), (0, 2), (2, 0), (1, 2), (2, 1), (0, 3), (3, 0), (1, 3), (3, 1), (2, 3), (3, 2)];
9
10/// Motion vector scaling (8-179 .. 8-183).
11pub fn scale_mv(mv: [i16; 2], td: i32, tb: i32) -> [i16; 2] {
12    let td = td.clamp(-128, 127);
13    let tb = tb.clamp(-128, 127);
14    if td == 0 {
15        return mv;
16    }
17    let tx = (16384 + (td.abs() >> 1)) / td;
18    let dsf = ((tb * tx + 32) >> 6).clamp(-4096, 4095);
19    let s = |c: i16| -> i16 {
20        let p = dsf * c as i32;
21        (p.signum() * ((p.abs() + 127) >> 8)).clamp(-32768, 32767) as i16
22    };
23    [s(mv[0]), s(mv[1])]
24}

26/// Same motion (used lists only).
27fn same_motion(a: &MvField, b: &MvField) -> bool {
28    for l in 0..2 {
29        if a.pred(l) != b.pred(l) {
30            return false;
31        }
32        if a.pred(l) && (a.ref_idx[l] != b.ref_idx[l] || a.mv[l] != b.mv[l]) {
33            return false;
34        }
35    }
36    true
37}

39/// A prediction block: the coding block (`xc`, `yc`, size `ncb`) it belongs to, its own position
40/// (`xp`, `yp`), size (`w` x `h`) and index `part_idx` within the coding block, and the coding
41/// block's partitioning mode.
42#[derive(Clone, Copy)]
43pub struct PredBlock {
44    pub xc: i32,
45    pub yc: i32,
46    pub ncb: i32,
47    pub xp: i32,
48    pub yp: i32,
49    pub w: i32,
50    pub h: i32,
51    pub part_idx: usize,
52    pub part: PartMode,
53}

55impl<'a, S: crate::craft::video::hevc::decode::picture::Sample> SliceDecoder<'a, S> {
56    /// Prediction block availability (6.4.2) of the neighbouring location (`xn`, `yn`).
57    fn pb_avail(&self, pb: &PredBlock, xn: i32, yn: i32) -> bool {
58        let PredBlock { xc, yc, ncb, xp, yp, w, h, part_idx, .. } = *pb;
59        let same_cb = xc <= xn && yc <= yn && xc + ncb > xn && yc + ncb > yn;
60        let avail = if !same_cb {
61            self.pic.zavail(xp, yp, xn, yn, self.slice_addr)
62        } else {
63            !((w << 1) == ncb && (h << 1) == ncb && part_idx == 1 && yc + h <= yn && xc + w > xn)
64        };
65        avail && self.pic.blk_at(xn, yn).flags & F_INTRA == 0
66    }
67
68    #[inline]
69    fn mv_at(&self, x: i32, y: i32) -> MvField {
70        self.pic.mvf[(y as usize >> 2) * self.pic.w4 + (x as usize >> 2)]
71    }
72
73    #[inline]
74    fn ref_pic(&self, l: usize, idx: i8) -> &RefPic {
75        &self.refs[l][idx as usize]
76    }

78    /// Merge mode motion (8.5.3.2.2).
79    pub fn derive_merge(&self, pb: &PredBlock, merge_idx: usize) -> MvField {
80        let PredBlock { xc, yc, ncb, w: w0, h: h0, part, .. } = *pb;
81        let pml = self.pps.log2_parallel_merge_level;
82        // with a parallel merge level above 4x4, the 8x8 coding block's 2Nx2N candidates are shared
83        let pb = if pml > 2 && ncb == 8 { PredBlock { xp: xc, yp: yc, w: ncb, h: ncb, part_idx: 0, ..*pb } } else { *pb };
84        let PredBlock { xp, yp, w, h, part_idx, .. } = pb;
85        let same_mer = |xn: i32, yn: i32| (xp >> pml) == (xn >> pml) && (yp >> pml) == (yn >> pml);
86        let avail = |xn: i32, yn: i32| self.pb_avail(&pb, xn, yn) && !same_mer(xn, yn);
87        let mut list: [MvField; 5] = [MvField::default(); 5];
88        let mut n = 0;
89        let max = self.sh.max_num_merge_cand as usize;
90        // spatial candidates (8.5.3.2.3)
91        let pa1 = (xp - 1, yp + h - 1);
92        let av_a1 = avail(pa1.0, pa1.1) && !(matches!(part, PartMode::PNx2N | PartMode::PnLx2N | PartMode::PnRx2N) && part_idx == 1);
93        let m_a1 = if av_a1 { self.mv_at(pa1.0, pa1.1) } else { MvField::default() };
94        if av_a1 {
95            list[n] = m_a1;
96            n += 1;
97        }
98        let pb1 = (xp + w - 1, yp - 1);
99        let av_b1 = avail(pb1.0, pb1.1) && !(matches!(part, PartMode::P2NxN | PartMode::P2NxnU | PartMode::P2NxnD) && part_idx == 1);
100        let m_b1 = if av_b1 { self.mv_at(pb1.0, pb1.1) } else { MvField::default() };
101        let fl_b1 = av_b1 && !(av_a1 && same_motion(&m_a1, &m_b1));
102        if fl_b1 {
103            list[n] = m_b1;
104            n += 1;
105        }
106        let pb0 = (xp + w, yp - 1);
107        let av_b0 = avail(pb0.0, pb0.1);
108        let fl_b0 = av_b0 && {
109            let m = self.mv_at(pb0.0, pb0.1);
110            !(av_b1 && same_motion(&m_b1, &m))
111        };
112        if fl_b0 {
113            list[n] = self.mv_at(pb0.0, pb0.1);
114            n += 1;
115        }
116        let pa0 = (xp - 1, yp + h);
117        let av_a0 = avail(pa0.0, pa0.1);
118        let fl_a0 = av_a0 && {
119            let m = self.mv_at(pa0.0, pa0.1);
120            !(av_a1 && same_motion(&m_a1, &m))
121        };
122        if fl_a0 {
123            list[n] = self.mv_at(pa0.0, pa0.1);
124            n += 1;
125        }
126        let pb2 = (xp - 1, yp - 1);
127        let av_b2 = avail(pb2.0, pb2.1);
128        let fl_b2 = av_b2 && {
129            let m = self.mv_at(pb2.0, pb2.1);
130            !((av_a1 && same_motion(&m_a1, &m)) || (av_b1 && same_motion(&m_b1, &m)))
131                && (fl_a0 as u32 + av_a1 as u32 + fl_b0 as u32 + fl_b1 as u32) != 4
132        };
133        if fl_b2 {
134            list[n] = self.mv_at(pb2.0, pb2.1);
135            n += 1;
136        }
137        if merge_idx < n {
138            return self.finish_merge(list[merge_idx], w0, h0);
139        }
140        // temporal candidate
141        let mut col = MvField::default();
142        let mut col_avail = false;
143        if let Some(mv) = self.temporal(xp, yp, w, h, 0, 0) {
144            col.mv[0] = mv;
145            col.ref_idx[0] = 0;
146            col_avail = true;
147        }
148        if self.sh.is_b()
149            && let Some(mv) = self.temporal(xp, yp, w, h, 1, 0)
150        {
151            col.mv[1] = mv;
152            col.ref_idx[1] = 0;
153            col_avail = true;
154        }
155        let mut cands: Vec<MvField> = list[..n].to_vec();
156        if col_avail {
157            cands.push(col);
158        }
159        if merge_idx < cands.len() {
160            return self.finish_merge(cands[merge_idx], w0, h0);
161        }
162        // combined bi-predictive candidates (8.5.3.2.4)
163        let num_orig = cands.len();
164        if self.sh.is_b() && num_orig > 1 && num_orig < max {
165            let mut comb = 0;
166            while comb < num_orig * (num_orig - 1) && cands.len() < max {
167                let (i0, i1) = COMB[comb];
168                let (c0, c1) = (cands[i0], cands[i1]);
169                if c0.pred(0) && c1.pred(1) && (self.ref_pic(0, c0.ref_idx[0]).poc != self.ref_pic(1, c1.ref_idx[1]).poc || c0.mv[0] != c1.mv[1]) {
170                    cands.push(MvField { mv: [c0.mv[0], c1.mv[1]], ref_idx: [c0.ref_idx[0], c1.ref_idx[1]] });
171                }
172                comb += 1;
173            }
174        }
175        if merge_idx < cands.len() {
176            return self.finish_merge(cands[merge_idx], w0, h0);
177        }
178        // zero candidates (8.5.3.2.5)
179        let num_ref = if self.sh.is_b() { self.sh.num_ref_idx[0].min(self.sh.num_ref_idx[1]) } else { self.sh.num_ref_idx[0] } as usize;
180        let mut zero_idx = 0;
181        while cands.len() <= merge_idx {
182            let r = if zero_idx < num_ref { zero_idx as i8 } else { 0 };
183            cands.push(MvField { mv: [[0; 2]; 2], ref_idx: [r, if self.sh.is_b() { r } else { -1 }] });
184            zero_idx += 1;
185        }
186        self.finish_merge(cands[merge_idx], w0, h0)
187    }
188
189    fn finish_merge(&self, mut f: MvField, w: i32, h: i32) -> MvField {
190        if f.pred(0) && f.pred(1) && w + h == 12 {
191            f.ref_idx[1] = -1;
192            f.mv[1] = [0, 0];
193        }
194        for l in 0..2 {
195            if !f.pred(l) {
196                f.mv[l] = [0, 0];
197            }
198        }
199        f
200    }

202    /// Temporal luma motion vector prediction (8.5.3.2.8).
203    pub fn temporal(&self, xp: i32, yp: i32, w: i32, h: i32, l: usize, ref_idx: usize) -> Option<[i16; 2]> {
204        let col = self.col.as_ref()?;
205        let s = self.sps.log2_ctb;
206        let (xbr, ybr) = (xp + w, yp + h);
207        if (yp >> s) == (ybr >> s) && ybr < self.pic.height as i32 && xbr < self.pic.width as i32 {
208            let r = self.col_mv(col, (xbr >> 4) << 4, (ybr >> 4) << 4, l, ref_idx);
209            if r.is_some() {
210                return r;
211            }
212        }
213        let (xc, yc) = (xp + (w >> 1), yp + (h >> 1));
214        self.col_mv(col, (xc >> 4) << 4, (yc >> 4) << 4, l, ref_idx)
215    }

217    /// Collocated motion vectors (8.5.3.2.9).
218    fn col_mv(&self, col: &RefPic, x: i32, y: i32, l: usize, ref_idx: usize) -> Option<[i16; 2]> {
219        let f = &col.frame;
220        if f.width != self.pic.width || f.height != self.pic.height {
221            return None;
222        }
223        let cm = f.col_mv(x as usize, y as usize);
224        if cm.flags & 3 == 0 {
225            return None;
226        }
227        let list_col = if cm.flags & 1 == 0 {
228            1
229        } else if cm.flags & 2 == 0 {
230            0
231        } else if self.no_backward_pred {
232            l
233        } else {
234            self.sh.collocated_from_l0 as usize
235        };
236        let mv = cm.mv[list_col];
237        let col_lt = (cm.flags >> (2 + list_col)) & 1 == 1;
238        let cur = self.refs[l].get(ref_idx)?;
239        if cur.long_term != col_lt {
240            return None;
241        }
242        let col_diff = col.poc - cm.ref_poc[list_col];
243        let cur_diff = self.pic.poc - cur.poc;
244        if cur.long_term || col_diff == cur_diff { Some(mv) } else { Some(scale_mv(mv, col_diff, cur_diff)) }
245    }

247    /// AMVP predictor (8.5.3.2.6, 8.5.3.2.7).
248    pub fn derive_amvp(&self, pb: &PredBlock, l: usize, ref_idx: usize, mvp_flag: usize) -> [i16; 2] {
249        let PredBlock { xp, yp, w, h, .. } = *pb;
250        let target = &self.refs[l][ref_idx];
251        let (tpoc, tlt) = (target.poc, target.long_term);
252        let cur_poc = self.pic.poc;
253        let y = 1 - l;
254        // exact reference match (no scaling)
255        let unscaled = |f: &MvField| -> Option<[i16; 2]> {
256            if f.pred(l) && self.ref_pic(l, f.ref_idx[l]).poc == tpoc {
257                Some(f.mv[l])
258            } else if f.pred(y) && self.ref_pic(y, f.ref_idx[y]).poc == tpoc {
259                Some(f.mv[y])
260            } else {
261                None
262            }
263        };
264        // long-term-compatible match, scaled when both references are short-term
265        let scaled = |f: &MvField| -> Option<[i16; 2]> {
266            let pick = if f.pred(l) && self.ref_pic(l, f.ref_idx[l]).long_term == tlt {
267                Some((f.mv[l], self.ref_pic(l, f.ref_idx[l])))
268            } else if f.pred(y) && self.ref_pic(y, f.ref_idx[y]).long_term == tlt {
269                Some((f.mv[y], self.ref_pic(y, f.ref_idx[y])))
270            } else {
271                None
272            };
273            pick.map(|(mv, r)| if r.poc != tpoc && !r.long_term && !tlt { scale_mv(mv, cur_poc - r.poc, cur_poc - tpoc) } else { mv })
274        };
275        let pa = [(xp - 1, yp + h), (xp - 1, yp + h - 1)];
276        let av_a = [self.pb_avail(pb, pa[0].0, pa[0].1), self.pb_avail(pb, pa[1].0, pa[1].1)];
277        let is_scaled = av_a[0] || av_a[1];
278        let mut a = None;
279        for k in 0..2 {
280            if av_a[k] && a.is_none() {
281                a = unscaled(&self.mv_at(pa[k].0, pa[k].1));
282            }
283        }
284        if a.is_none() {
285            for k in 0..2 {
286                if av_a[k] && a.is_none() {
287                    a = scaled(&self.mv_at(pa[k].0, pa[k].1));
288                }
289            }
290        }
291        let pbk = [(xp + w, yp - 1), (xp + w - 1, yp - 1), (xp - 1, yp - 1)];
292        let av_b = [0, 1, 2].map(|k| self.pb_avail(pb, pbk[k].0, pbk[k].1));
293        let mut b = None;
294        for k in 0..3 {
295            if av_b[k] && b.is_none() {
296                b = unscaled(&self.mv_at(pbk[k].0, pbk[k].1));
297            }
298        }
299        if !is_scaled && b.is_some() {
300            a = b;
301        }
302        if !is_scaled {
303            b = None;
304            for k in 0..3 {
305                if av_b[k] && b.is_none() {
306                    b = scaled(&self.mv_at(pbk[k].0, pbk[k].1));
307                }
308            }
309        }
310        let mut list: [[i16; 2]; 3] = [[0; 2]; 3];
311        let mut n = 0;
312        if let Some(a) = a {
313            list[n] = a;
314            n += 1;
315            if let Some(b) = b
316                && b != a
317            {
318                list[n] = b;
319                n += 1;
320            }
321        } else if let Some(b) = b {
322            list[n] = b;
323            n += 1;
324        }
325        if n < 2
326            && mvp_flag >= n
327            && let Some(c) = self.temporal(xp, yp, w, h, l, ref_idx)
328        {
329            list[n] = c;
330            n += 1;
331        }
332        let _ = n;
333        list[mvp_flag.min(1)]
334    }
