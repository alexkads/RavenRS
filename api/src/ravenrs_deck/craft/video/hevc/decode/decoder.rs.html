1//! Top-level decoder: NAL dispatch, picture boundaries, POC / RPS / DPB management and output.
2//!
3//! The calling thread parses parameter sets and slice headers and does all POC, RPS and DPB
4//! bookkeeping. The CTU layer of each picture runs as a job (on a thread pool with the `threads`
5//! feature) that publishes finished CTB rows into the picture's shared [`Frame`]; jobs of later pictures
6//! block per row on the reference data they read, so several pictures decode concurrently.
7

19/// Completed pictures a failed `decode` call leaves queued for the next call (older ones are dropped).
20const MAX_QUEUED_AFTER_ERROR: usize = 32;
21/// Most pictures one `decode` call may start: a chunk is an access unit, or a few in a transport
22/// stream packet. Each picture is a full frame, and small slices make them cheap to ask for.
23const MAX_PICTURES_PER_CALL: usize = 64;

168    /// Parse an `hvcC` record: parameter-set arrays and the NAL length size used by [`Decoder::decode`].
169    pub fn configure_hvcc(&mut self, hvcc: &[u8]) -> Result<()> {
170        ensure!(hvcc.len() >= 23, "hvcC too short");
171        ensure!(hvcc[0] == 1, "unsupported hvcC version {}", hvcc[0]);
172        let num_arrays = hvcc[22] as usize;
173        // Two passes: the first checks every array, so a record that carries anything but
174        // parameter sets and SEI (a picture slice, say) is refused before any NAL is handled.
175        for pass in 0..2 {
176            let mut pos = 23;
177            for _ in 0..num_arrays {
178                ensure!(pos <= hvcc.len() && hvcc.len() - pos >= 3, "hvcC truncated");
179                let array_type = hvcc[pos] & 0x3f;
180                let n = u16::from_be_bytes([hvcc[pos + 1], hvcc[pos + 2]]) as usize;
181                pos += 3;
182                for _ in 0..n {
183                    ensure!(pos <= hvcc.len() && hvcc.len() - pos >= 2, "hvcC truncated");
184                    let len = u16::from_be_bytes([hvcc[pos], hvcc[pos + 1]]) as usize;
185                    pos += 2;
186                    ensure!(len <= hvcc.len() - pos, "hvcC truncated");
187                    let nal = &hvcc[pos..pos + len];
188                    if pass == 0 {
189                        let header = NalHeader::parse(nal)?;
190                        ensure!(
191                            [nal_type::VPS, nal_type::SPS, nal_type::PPS, nal_type::PREFIX_SEI, nal_type::SUFFIX_SEI].contains(&array_type),
192                            "hvcC array has unsupported NAL type {array_type}"
193                        );
194                        ensure!(header.nal_type == array_type, "hvcC array type {array_type} disagrees with NAL header type {}", header.nal_type);
195                    } else {
196                        self.handle_nal(nal, 0)?;
197                    }
198                    pos += len;
199                }
200            }
201        }
202        self.nal_length_size = Some((hvcc[21] & 3) as usize + 1);
203        Ok(())
204    }

211    /// Decode one access unit (all NAL units of one picture, Annex-B or length-prefixed per
212    /// configuration; a byte stream with several complete access units is accepted too). Returns
213    /// pictures that became ready for output, in output order; `pts` travels with the picture of this
214    /// access unit through reordering.
215    pub fn decode(&mut self, data: &[u8], pts: i64) -> Result<Vec<Picture>> {
216        let nals = match self.nal_length_size {
217            Some(n) => match length_prefixed_nals(data, n) {
218                Ok(nals) => nals,
219                Err(e) => {
220                    self.trim_queue_after_error();
221                    return Err(e.into());
222                }
223            },
224            None => annexb_nals(data),
225        };
226        self.call_pictures = 0;
227        let mut result = Ok(());
228        for nal in nals {
229            if let Err(e) = self.handle_nal(nal, pts) {
230                result = Err(e);
231                break;
232            }
233        }
234        self.submit_pending();
235        if let Some(e) = self.take_error() {
236            result = result.and(Err(e));
237        }
238        if result.is_err() {
239            self.trim_queue_after_error();
240        }
241        result?;
242        let max_queue = self.max_in_flight * 2;
243        let mut pics = Vec::new();
244        while let Some(o) = self.out_queue.front() {
245            if !o.frame.is_complete() && self.out_queue.len() <= max_queue {
246                break;
247            }
248            let Some(o) = self.out_queue.pop_front() else { break };
249            pics.push(make_picture(&o));
250        }
251        Ok(pics)
252    }
253
254    /// A caller that keeps feeding data after errors never drains the queue: keep the newest few.
255    fn trim_queue_after_error(&mut self) {
256        let excess = self.out_queue.len().saturating_sub(MAX_QUEUED_AFTER_ERROR);
257        self.out_queue.drain(..excess);
258    }

260    /// Output all remaining pictures (end of stream).
261    pub fn flush(&mut self) -> Vec<Picture> {
262        self.submit_pending();
263        let mut outs = Vec::new();
264        self.dpb.flush(&mut outs);
265        self.out_queue.extend(outs);
266        let pics = self.out_queue.drain(..).map(|o| make_picture(&o)).collect();
267        while let Some(f) = self.in_flight.pop_front() {
268            f.wait_complete();
269        }
270        self.first_picture = true;
271        self.gray_reference = None;
272        pics
273    }

300    fn handle_nal(&mut self, nal: &[u8], pts: i64) -> Result<()> {
301        if nal.len() < 2 {
302            return Ok(());
303        }
304        let hdr = NalHeader::parse(nal)?;
305        if hdr.layer_id != 0 {
306            return Ok(());
307        }
308        match hdr.nal_type {
309            nal_type::VPS => {
310                let v = Vps::parse(&unescape_rbsp(&nal[2..]))?;
311                let id = v.id as usize;
312                self.vpss[id] = Some(Arc::new(v));
313            }
314            nal_type::SPS => {
315                let s = Sps::parse(&unescape_rbsp(&nal[2..]))?;
316                let id = s.id as usize;
317                self.spss[id] = Some(Arc::new(s));
318            }
319            nal_type::PPS => {
320                let p = Pps::parse(&unescape_rbsp(&nal[2..]))?;
321                let id = p.id as usize;
322                self.ppss[id] = Some(Arc::new(p));
323            }
324            nal_type::EOS | nal_type::EOB => {
325                self.submit_pending();
326                self.after_eos = true;
327            }
328            0..=9 | 16..=21 => self.handle_slice(nal, hdr, pts)?,
329            // AUD, SEI, filler, reserved and unspecified NAL units are skipped
330            _ => {}
331        }
332        Ok(())
333    }

335    fn layout_for(&mut self, pps: &Arc<Pps>, sps: &Arc<Sps>) -> Result<(Arc<Layout>, Option<Arc<Vec<Vec<u8>>>>)> {
336        if let Some((_, _, l, s)) = self.layout_cache.iter().find(|(p, s, _, _)| Arc::ptr_eq(p, pps) && Arc::ptr_eq(s, sps)) {
337            return Ok((l.clone(), s.clone()));
338        }
339        let layout = Arc::new(Layout::new(sps, pps)?);
340        let scaling = if sps.scaling_list_enabled {
341            let list = pps
342                .scaling_list
343                .as_ref()
344                .or(sps.scaling_list.as_ref())
345                .cloned()
346                .unwrap_or_else(crate::craft::video::hevc::decode::params::ScalingList::default_lists);
347            let mut v = Vec::with_capacity(24);
348            for size in 0..4 {
349                for m in 0..6 {
350                    v.push(list.factor(size, m));
351                }
352            }
353            Some(Arc::new(v))
354        } else {
355            None
356        };
357        if self.layout_cache.len() > 8 {
358            self.layout_cache.remove(0);
359        }
360        self.layout_cache.push((pps.clone(), sps.clone(), layout.clone(), scaling.clone()));
361        Ok((layout, scaling))
362    }

364    fn handle_slice(&mut self, nal: &[u8], hdr: NalHeader, pts: i64) -> Result<()> {
365        let rbsp = unescape_rbsp(&nal[2..]);
366        // a first_slice_segment_in_pic_flag of 1 starts a new picture
367        let first = rbsp.first().is_some_and(|b| b & 0x80 != 0);
368        if first {
369            self.submit_pending();
370        }
371        if hdr.is_rasl() && (self.skip_rasl || self.first_picture) {
372            if first {
373                self.stat(|s| s.skipped_rasl += 1);
374            }
375            return Ok(());
376        }
377        let ppss = &self.ppss;
378        let spss = &self.spss;
379        let prev = self.pending.as_ref().map(|p| &p.last_sh);
380        let (sh, pps, sps) = SliceHeader::parse(
381            &rbsp,
382            hdr,
383            |id| {
384                let pps = ppss[id as usize].as_ref().ok_or_else(|| Error::MissingParameterSet(format!("PPS {id}")))?;
385                let sps = spss[pps.sps_id as usize].as_ref().ok_or_else(|| Error::MissingParameterSet(format!("SPS {}", pps.sps_id)))?;
386                Ok((pps.clone(), sps.clone()))
387            },
388            prev,
389        )?;
390        sps.check_supported()?;
391        pps.check_supported()?;
392        // `start_picture` creates the frame and DPB state. Do not allocate those for a NAL that
393        // contains only a syntactically valid slice header and no CABAC payload.
394        ensure!(sh.data_offset < rbsp.len(), "empty slice segment data");
395        if first {
396            self.start_picture(&sh, &sps, &pps, pts)?;
397        } else {
398            let Some(p) = &self.pending else { return invalid("slice segment without the first slice of its picture") };
399            ensure!(Arc::ptr_eq(&p.pps, &pps), "PPS changed within a picture");
400        }
401        let Some(pending) = self.pending.as_mut() else {
402            return invalid("slice segment without the first slice of its picture");
403        };
404        let refs = build_ref_lists(&sh, &pending.rps)?;
405        if !sh.dependent {
406            pending.last_sh = sh.clone();
407        }
408        let (st, dependent, weighted, tmvp, sao, dbk_off, lt) =
409            (sh.slice_type, sh.dependent, sh.pwt.is_some(), sh.temporal_mvp, sh.sao_luma || sh.sao_chroma, sh.deblocking_disabled, !sh.lt.is_empty());
410        pending.slices.push(SliceJob { sh, pps, sps, rbsp, refs });
411        self.stat(|s| {
412            match st {
413                crate::craft::video::hevc::decode::slice::SliceType::I => s.slices_i += 1,
414                crate::craft::video::hevc::decode::slice::SliceType::P => s.slices_p += 1,
415                crate::craft::video::hevc::decode::slice::SliceType::B => s.slices_b += 1,
416            }
417            s.dependent_slices += dependent as u64;
418            s.weighted_slices += weighted as u64;
419            s.tmvp_slices += tmvp as u64;
420            s.sao_slices += sao as u64;
421            s.deblock_disabled_slices += dbk_off as u64;
422            s.long_term_slices += lt as u64;
423        });
424        Ok(())
425    }

432    /// Refuse an SPS whose pictures would not fit the memory budget, before anything is allocated.
433    fn check_budget(&self, sps: &Sps) -> Result<()> {
434        let chroma = sps.chroma_array_type();
435        let shape = crate::craft::video::budget::PictureShape {
436            width: u64::from(sps.width),
437            height: u64::from(sps.height),
438            shift_x: u32::from(sps.sub_width_c() == 2),
439            shift_y: u32::from(sps.sub_height_c() == 2),
440            chroma_planes: if chroma == 0 { 0 } else { 2 },
441            bytes_per_sample: 2,
442        };
443        let frame_bytes = |f: &Frame| {
444            (f.width as u64)
445                .saturating_mul(f.height as u64)
446                .saturating_add((f.cwidth as u64).saturating_mul(f.cheight as u64).saturating_mul(2))
447                .saturating_mul(2)
448        };
449        // Output frames waiting to be returned stay allocated after they leave the DPB, at their
450        // own size: a resolution change in this call can leave pictures of the previous SPS.
451        let queued: u64 = self
452            .out_queue
453            .iter()
454            .filter(|o| !self.dpb.entries.iter().any(|e| Arc::ptr_eq(&e.frame, &o.frame)) && !self.in_flight.iter().any(|f| Arc::ptr_eq(f, &o.frame)))
455            .map(|o| frame_bytes(&o.frame))
456            .fold(0, u64::saturating_add);
457        // Frames the DPB holds now count at their own size too (an SPS change flushes them into the
458        // queue before the new pictures are made): the DPB part is the larger of its new slots and that.
459        let held: u64 = self.dpb.entries.iter().map(|e| frame_bytes(&e.frame)).fold(0, u64::saturating_add);
460        // A queued frame still being decoded is one of the in-flight pictures already reserved, so the
461        // queue and the in-flight reservation overlap: charge the larger, not both.
462        let pictures = u64::from(sps.max_dec_pic_buffering) + 1 + self.max_in_flight as u64;
463        let slots = shape.store_bytes(u64::from(sps.max_dec_pic_buffering));
464        let flight = shape.store_bytes(self.max_in_flight as u64);
465        let total = shape
466            .store_bytes(1)
467            .zip(slots)
468            .zip(flight)
469            .map(|((one, slots), flight)| slots.max(held).saturating_add(one).saturating_add(flight).saturating_add(queued));
470        ensure!(
471            total.is_some_and(|b| b <= self.memory_budget),
472            "{}x{} pictures with {pictures} in the decoder and {queued} bytes queued exceed the memory budget",
473            sps.width,
474            sps.height
475        );
476        Ok(())
477    }

479    fn start_picture(&mut self, sh: &SliceHeader, sps: &Arc<Sps>, pps: &Arc<Pps>, pts: i64) -> Result<()> {
480        self.check_budget(sps)?;
481        ensure!(sh.first_slice_segment_in_pic && !sh.dependent, "invalid first slice segment");
482        self.call_pictures += 1;
483        ensure!(self.call_pictures <= MAX_PICTURES_PER_CALL, "too many pictures in one chunk");
484        let hdr = sh.nal;
485        let changed = match &self.active_sps {
486            None => true,
487            Some(a) => {
488                a.width != sps.width
489                    || a.height != sps.height
490                    || a.chroma_format_idc != sps.chroma_format_idc
491                    || a.bit_depth_luma != sps.bit_depth_luma
492                    || a.bit_depth_chroma != sps.bit_depth_chroma
493                    || a.log2_ctb != sps.log2_ctb
494            }
495        };
496        let mut outs = Vec::new();
497        if changed && self.active_sps.is_some() {
498            self.dpb.flush(&mut outs);
499        }
500        if changed {
501            self.gray_reference = None;
502        }
503        self.active_sps = Some(sps.clone());
504        self.dpb.max_dec_pic_buffering = sps.max_dec_pic_buffering as usize;
505        self.dpb.max_num_reorder = sps.max_num_reorder as usize;
506        self.dpb.max_latency = if sps.max_latency_increase_plus1 != 0 { sps.max_num_reorder + sps.max_latency_increase_plus1 - 1 } else { 0 };
507        let irap_no_rasl = hdr.is_irap() && (hdr.is_idr() || hdr.is_bla() || self.first_picture || self.after_eos);
508        if hdr.is_irap() {
509            self.skip_rasl = irap_no_rasl;
510            self.stat(|s| {
511                if hdr.is_idr() {
512                    s.idr += 1;
513                } else {
514                    s.cra += 1;
515                }
516            });
517        }
518        let max_lsb = sps.max_poc_lsb();
519        let poc = self.poc_state.compute(sh, max_lsb, irap_no_rasl);
520        let (w, h) = (sps.width as usize, sps.height as usize);
521        let Some(chroma) = ChromaFormat::from_idc(sps.chroma_array_type()) else {
522            return unsupported(format!("chroma_format_idc {}", sps.chroma_format_idc));
523        };
524        let template = Frame::new(0, 0, w, h, chroma, sps.log2_ctb, sps.bit_depth_luma, sps.bit_depth_chroma);
525        let next_id = &mut self.next_id;
526        let gray_reference = &mut self.gray_reference;
527        let mut missing = 0u64;
528        let mut make_missing = |p: i32| -> FrameRef {
529            missing += 1;
530            let id = *next_id;
531            *next_id += 1;
532            let gray = gray_reference.get_or_insert_with(|| Frame::gray(0, 0, &template));
533            Arc::new(gray.with_identity(id, p))
534        };
535        let rps = self.dpb.apply_rps(sh, poc, max_lsb, irap_no_rasl, &mut make_missing)?;
536        if missing > 0 {
537            self.stat(|s| s.missing_refs += missing);
538        }
539        if irap_no_rasl && !self.first_picture {
540            if sh.no_output_of_prior_pics && hdr.is_idr() {
541                self.dpb.clear();
542            } else {
543                self.dpb.flush(&mut outs);
544            }
545        } else {
546            self.dpb.bump_before_decode(&mut outs);
547        }
548        self.out_queue.extend(outs);
549        let (layout, scaling) = self.layout_for(pps, sps)?;
550        let _ = scaling;
551        let id = self.next_id;
552        self.next_id += 1;
553        let frame = Arc::new(Frame::new(id, poc, w, h, chroma, sps.log2_ctb, sps.bit_depth_luma, sps.bit_depth_chroma));
554        let output = sh.pic_output && !(hdr.is_rasl() && self.skip_rasl);
555        // Draft mode: a sub-layer non-reference picture of the highest sub-layer is never used
556        // for prediction, so its in-loop filters can be left out without changing anything else.
557        let draft = self.draft && hdr.is_sub_layer_non_ref() && hdr.temporal_id as u32 >= sps.max_sub_layers_minus1;
558        if draft {
559            self.stat(|s| s.draft_pictures += 1);
560        }
561        let meta = Arc::new(output_meta(sps, pts, hdr.is_irap(), draft));
562        self.pending = Some(PendingPic {
563            frame,
564            sps: sps.clone(),
565            pps: pps.clone(),
566            layout,
567            last_sh: sh.clone(),
568            first_sh: sh.clone(),
569            rps,
570            poc,
571            output,
572            meta,
573            slices: Vec::new(),
574        });
575        self.first_picture = false;
576        self.after_eos = false;
577        self.stat(|s| {
578            s.tiles_pictures += pps.tiles_enabled as u64;
579            s.wpp_pictures += pps.entropy_coding_sync as u64;
580            s.scaling_list_pictures += sps.scaling_list_enabled as u64;
581            s.transform_skip_pictures += pps.transform_skip as u64;
582        });
583        Ok(())
584    }
585
586    /// Start decoding the pending picture and insert it into the DPB.
587    fn submit_pending(&mut self) {
588        let Some(p) = self.pending.take() else { return };
589        let PendingPic { frame, sps, pps, layout, first_sh, poc, output, meta, slices, .. } = p;
590        let scaling =
591            self.layout_cache.iter().find(|(pp, ss, _, _)| Arc::ptr_eq(pp, &pps) && Arc::ptr_eq(ss, &sps)).and_then(|(_, _, _, s)| s.clone());
592        self.dispatch(PicJob { frame: frame.clone(), sps, pps, layout, scaling, slices, draft: meta.draft });
593        self.poc_state.update(&first_sh, poc);
594        let mut outs = Vec::new();
595        self.dpb.insert(frame, poc, output, meta, &mut outs);
596        self.out_queue.extend(outs);
597    }
598
599    fn dispatch(&mut self, job: PicJob) {
600        let shared = self.shared.clone();
601        if let Some(pool) = &self.pool {
602            self.in_flight.retain(|f| !f.is_complete());
603            while self.in_flight.len() >= self.max_in_flight {
604                if let Some(f) = self.in_flight.pop_front() {
605                    f.wait_complete();
606                }
607            }
608            self.in_flight.push_back(job.frame.clone());
609            pool.spawn_fifo(move || run_job(job, &shared));
610            return;
611        }
612        run_job(job, &shared);
613    }
614}

616/// Everything needed to decode one picture on a worker.
617struct PicJob {
618    frame: FrameRef,
619    sps: Arc<Sps>,
620    pps: Arc<Pps>,
621    layout: Arc<Layout>,
622    scaling: Option<Arc<Vec<Vec<u8>>>>,
623    slices: Vec<SliceJob>,
624    draft: bool,
625}
626
627/// Decode all slices of one picture and publish its rows, in the picture's sample type.
628fn run_job(job: PicJob, shared: &Shared) {
629    if stores_u8(job.frame.bit_depth, job.frame.bit_depth_c) { run_job_as::<u8>(job, shared) } else { run_job_as::<u16>(job, shared) }
630}
631
632fn run_job_as<S: Sample>(job: PicJob, shared: &Shared) {
633    let PicJob { frame, sps, pps, layout, scaling, slices, draft } = job;
634    // The picture state is set up inside the guard too: a panic there must not escape the job.
635    let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
636        let mut pic = PicState::<S>::new(frame.clone(), sps, pps, layout, scaling);
637        pic.draft = draft;
638        let mut first_err = None;
639        for s in &slices {
640            if let Err(e) = SliceDecoder::decode(&mut pic, s) {
641                first_err.get_or_insert(e);
642            }
643        }
644        crate::craft::video::hevc::decode::filter::finish(&mut pic);
645        (pic, first_err)
646    }));
647    let (pic, err) = match result {
648        Ok((pic, err)) => (Some(pic), err),
649        Err(_) => {
650            // make sure readers never block on this frame
651            frame.publish_gray();
652            (None, Some(Error::Invalid("internal error while decoding a picture".into())))
653        }
654    };
655    if let Ok(mut s) = shared.stats.lock() {
656        s.pictures += 1;
657        let [intra, skip, inter] = pic.as_ref().map_or([0; 3], |pic| block_counts(&pic.blk));
658        s.intra_4x4_blocks += intra;
659        s.skip_4x4_blocks += skip;
660        s.inter_4x4_blocks += inter;
661    }
662    if let (Some(e), Ok(mut slot)) = (err, shared.error.lock()) {
663        slot.get_or_insert(e);
664    }
665}

667/// Intra, skipped and other inter 4x4 blocks of a picture, branch-free (one pass over every
668/// block): intra blocks have F_INTRA, skipped ones F_SKIP without it, inter ones any other flags.
669fn block_counts(blk: &[Blk]) -> [u64; 3] {
670    let (mut intra, mut skip, mut coded) = (0u64, 0u64, 0u64);
671    for b in blk {
672        intra += (b.flags & F_INTRA != 0) as u64;
673        skip += (b.flags & (F_INTRA | F_SKIP) == F_SKIP) as u64;
674        coded += (b.flags != 0) as u64;
675    }
676    [intra, skip, coded.saturating_sub(intra + skip)]
677}

695fn make_picture(o: &Output<FrameRef>) -> Picture {
696    let (cx, cy, cw, ch) = o.meta.crop;
697    let (cx, cy, cw, ch) = (cx as usize, cy as usize, cw as usize, ch as usize);
698    let chroma = o.frame.chroma;
699    let (ccw, cch) = (cw.div_ceil(chroma.sub_width()), ch.div_ceil(chroma.sub_height()));
700    let bd = o.meta.bit_depth;
701    let (y, u, v) = o.frame.cropped_planes((cx, cy, cw, ch), bd <= 8);
702    Picture {
703        width: cw as u32,
704        height: ch as u32,
705        chroma_width: ccw as u32,
706        chroma_height: cch as u32,
707        chroma,
708        bit_depth: bd,
709        y,
710        u,
711        v,
712        y_stride: cw,
713        uv_stride: ccw,
714        pts: o.meta.pts,
715        poc: o.poc,
716        key: o.meta.key,
717        color: ColorInfo {
718            full_range: o.meta.full_range,
719            primaries: o.meta.colour_primaries,
720            transfer: o.meta.transfer_characteristics,
721            matrix: o.meta.matrix_coefficients,
722        },
723        sar: o.meta.sar,
724        draft: o.meta.draft,
725    }
726}

830            }
