1//! In-loop filters and row publication: deblocking (8.7.2) and sample adaptive offset (8.7.3), run
2//! row by row behind the CTU decoding so finished CTB rows can be published to later pictures early.
3//!
4//! Row r is deblocked once rows 0..=r+1 are decoded (row r+1 still needs the unfiltered bottom samples
5//! of row r for intra prediction); SAO of row r runs once row r+1 is deblocked; then row r is final.
6
7use crate::craft::video::hevc::decode::picture::{ColMv, FrameRow, Sample};
8use crate::craft::video::hevc::decode::slicedec::{E_PU_H, E_PU_V, E_TU_H, E_TU_V, F_CBF, F_CODED, F_INTRA, F_NOFILTER, MvField, PicState};
9use crate::craft::video::hevc::decode::spec_tables::{BETA_TABLE, TC_TABLE};
10use crate::craft::video::hevc::decode::tables::qpc;

12/// Called after each decoded CTB of CTB row `ry`.
13pub fn ctb_decoded<S: Sample>(pic: &mut PicState<S>, ry: usize) {
14    pic.row_done[ry] += 1;
15    while pic.rows_complete < pic.hctb && pic.row_done[pic.rows_complete] as usize == pic.wctb {
16        pic.rows_complete += 1;
17    }
18    advance(pic, false);
19}
20
21/// Run the filters / publication as far as the decoded rows allow (`all`: end of picture).
22pub fn advance<S: Sample>(pic: &mut PicState<S>, all: bool) {
23    let h = pic.hctb;
24    loop {
25        let r = pic.rows_deblocked;
26        if r < h && (all || pic.rows_complete >= (r + 2).min(h)) {
27            deblock_row(pic, r);
28            pic.rows_deblocked += 1;
29            continue;
30        }
31        let r = pic.rows_published;
32        if r < h && pic.rows_deblocked >= (r + 2).min(h) {
33            publish_row(pic, r);
34            pic.rows_published += 1;
35            continue;
36        }
37        break;
38    }
39}
40
41/// End of picture: filter and publish everything.
42pub fn finish<S: Sample>(pic: &mut PicState<S>) {
43    pic.rows_complete = pic.hctb;
44    advance(pic, true);
45}

47#[inline]
48fn tc_beta(q: i32, bs: i32, tc_off: i32, beta_off: i32, bd: u32) -> (i32, i32) {
49    let qb = (q + beta_off).clamp(0, 51);
50    let qt = (q + 2 * (bs - 1) + tc_off).clamp(0, 53);
51    ((TC_TABLE[qt as usize] as i32) << (bd - 8), (BETA_TABLE[qb as usize] as i32) << (bd - 8))
52}

54/// Boundary strength (8.7.2.4) between 4x4 blocks p and q, in slices `ps` and `qs` (indices into
55/// `pic.slices`, from [`edge_params`]).
56fn boundary_strength<S: Sample>(pic: &PicState<S>, pi: usize, qi: usize, tu_edge: bool, ps: u32, qs: u32) -> i32 {
57    let (p, q) = (&pic.blk[pi], &pic.blk[qi]);
58    if (p.flags | q.flags) & F_INTRA != 0 {
59        return 2;
60    }
61    if tu_edge && (p.flags | q.flags) & F_CBF != 0 {
62        return 1;
63    }
64    let (mp, mq) = (&pic.mvf[pi], &pic.mvf[qi]);
65    // Same motion in the same slice (same reference lists): same pictures, no motion difference.
66    if mp == mq && ps == qs {
67        return 0;
68    }
69    motion_bs(pic, ps, mp, qs, mq)
70}

72/// The picture referenced by list `l` of motion `f` in slice `s`.
73fn ref_id<S: Sample>(pic: &PicState<S>, s: u32, f: &MvField, l: usize) -> u32 {
74    if s == u32::MAX {
75        return u32::MAX;
76    }
77    pic.slices.get(s as usize).and_then(|sl| sl.ref_ids[l].get(f.ref_idx[l] as usize)).copied().unwrap_or(u32::MAX - 1)
78}

80fn motion_bs<S: Sample>(pic: &PicState<S>, ps: u32, p: &MvField, qs: u32, q: &MvField) -> i32 {
81    let np = p.pred(0) as u32 + p.pred(1) as u32;
82    let nq = q.pred(0) as u32 + q.pred(1) as u32;
83    if np != nq {
84        return 1;
85    }
86    let far = |a: [i16; 2], b: [i16; 2]| (a[0] as i32 - b[0] as i32).abs() >= 4 || (a[1] as i32 - b[1] as i32).abs() >= 4;
87    if np == 1 {
88        let lp = if p.pred(0) { 0 } else { 1 };
89        let lq = if q.pred(0) { 0 } else { 1 };
90        if ref_id(pic, ps, p, lp) != ref_id(pic, qs, q, lq) {
91            return 1;
92        }
93        return far(p.mv[lp], q.mv[lq]) as i32;
94    }
95    let (p0, p1) = (ref_id(pic, ps, p, 0), ref_id(pic, ps, p, 1));
96    let (q0, q1) = (ref_id(pic, qs, q, 0), ref_id(pic, qs, q, 1));
97    if !((p0 == q0 && p1 == q1) || (p0 == q1 && p1 == q0)) {
98        return 1;
99    }
100    if p0 != p1 {
101        if p0 == q0 { (far(p.mv[0], q.mv[0]) || far(p.mv[1], q.mv[1])) as i32 } else { (far(p.mv[0], q.mv[1]) || far(p.mv[1], q.mv[0])) as i32 }
102    } else {
103        ((far(p.mv[0], q.mv[0]) || far(p.mv[1], q.mv[1])) && (far(p.mv[0], q.mv[1]) || far(p.mv[1], q.mv[0]))) as i32
104    }
105}

107/// Whether the edge between blocks p and q (q = current block, owning the edge) is filtered, and the
108/// slice-level parameters (tc offset, beta offset) of q's slice, then the slices of p and q.
109fn edge_params<S: Sample>(pic: &PicState<S>, px: usize, py: usize, qx: usize, qy: usize) -> Option<(i32, i32, u32, u32)> {
110    let s = pic.log2_ctb;
111    let pc = (py >> s) * pic.wctb + (px >> s);
112    let qc = (qy >> s) * pic.wctb + (qx >> s);
113    let qs = pic.ctb_slice[qc];
114    let ps = pic.ctb_slice[pc];
115    if qs == u32::MAX || ps == u32::MAX {
116        return None;
117    }
118    let sq = &pic.slices[qs as usize];
119    if sq.deblock_disabled {
120        return None;
121    }
122    if pc != qc {
123        if pic.ctb_addr[pc] != pic.ctb_addr[qc] && !sq.lf_across {
124            return None;
125        }
126        if !pic.pps.loop_filter_across_tiles && pic.layout.tile_of_rs(pc as u32) != pic.layout.tile_of_rs(qc as u32) {
127            return None;
128        }
129    }
130    Some((sq.tc_offset, sq.beta_offset, ps, qs))
131}

133/// Deblock CTB row `r`: vertical edges of the row, then horizontal edges whose q side is in the row.
134fn deblock_row<S: Sample>(pic: &mut PicState<S>, r: usize) {
135    if !pic.deblocking_enabled_anywhere || pic.draft {
136        return;
137    }
138    let ctb = 1usize << pic.log2_ctb;
139    let y0 = r * ctb;
140    let y1 = (y0 + ctb).min(pic.height);
141    let (w, w4) = (pic.width, pic.w4);
142    let (bd, bdc) = (pic.bd_y, pic.bd_c);
143    let (cb_off, cr_off) = (pic.pps.cb_qp_offset, pic.pps.cr_qp_offset);
144    // chroma geometry: edges on the 8x8 chroma sample grid (8.7.2, edgeSpacing), each segment
145    // 4 chroma samples long, with bS taken from the luma segment at its first sample
146    let (sx, sy) = pic.chroma.shifts();
147    let is420 = sy == 1;
148    // vertical edges
149    let mut bs_v = vec![0i8; w4 * ((y1 - y0) / 4)];
150    for by in y0 / 4..y1 / 4 {
151        for bx in (2..w4).step_by(2) {
152            let qi = by * w4 + bx;
153            let e = pic.blk[qi].edges;
154            if e & (E_TU_V | E_PU_V) == 0 || pic.blk[qi].flags & F_CODED == 0 {
155                continue;
156            }
157            let (x, y) = (bx * 4, by * 4);
158            let Some((tc_off, beta_off, ps, qs)) = edge_params(pic, x - 1, y, x, y) else { continue };
159            let bs = boundary_strength(pic, qi - 1, qi, e & E_TU_V != 0, ps, qs);
160            bs_v[(by - y0 / 4) * w4 + bx] = bs as i8;
161            if bs == 0 {
162                continue;
163            }
164            let (p, q) = (&pic.blk[qi - 1], &pic.blk[qi]);
165            let qpl = (p.qp as i32 + q.qp as i32 + 1) >> 1;
166            let (tc, beta) = tc_beta(qpl, bs, tc_off, beta_off, bd);
167            let (nfp, nfq) = (p.flags & F_NOFILTER != 0, q.flags & F_NOFILTER != 0);
168            filter_luma(&mut pic.planes[0], EdgeSeg { o: y * w + x, step: 1, along: w }, beta, EdgeFilter { tc, nfp, nfq, bd });
169        }
170    }
171    // chroma vertical edges (8 chroma samples = 16 luma samples apart; a 4-line chroma segment
172    // spans 4 << sy luma lines), bS == 2 only
173    let cw = pic.cwidth;
174    for by in (y0 / 4..y1 / 4).step_by(1 << sy) {
175        for bx in (2 << sx..w4).step_by(2 << sx) {
176            if bs_v[(by - y0 / 4) * w4 + bx] != 2 {
177                continue;
178            }
179            let (x, y) = (bx * 4, by * 4);
180            let qi = by * w4 + bx;
181            let Some((tc_off, ..)) = edge_params(pic, x - 1, y, x, y) else { continue };
182            let (p, q) = (pic.blk[qi - 1], pic.blk[qi]);
183            let (nfp, nfq) = (p.flags & F_NOFILTER != 0, q.flags & F_NOFILTER != 0);
184            for (c, off) in [(1, cb_off), (2, cr_off)] {
185                let qpi = ((p.qp as i32 + q.qp as i32 + 1) >> 1) + off;
186                let qt = (qpc(qpi, is420) + 2 + tc_off).clamp(0, 53);
187                let tc = (TC_TABLE[qt as usize] as i32) << (bdc - 8);
188                filter_chroma(
189                    &mut pic.planes[c],
190                    EdgeSeg { o: (y >> sy) * cw + (x >> sx), step: 1, along: cw },
191                    EdgeFilter { tc, nfp, nfq, bd: bdc },
192                );
193            }
194        }
195    }
196    // horizontal edges with the q side in this row
197    let mut bs_h = vec![0i8; w4 * ((y1 - y0) / 4)];
198    for by in (y0 / 4..y1 / 4).filter(|b| b % 2 == 0 && *b > 0) {
199        for bx in 0..w4 {
200            let qi = by * w4 + bx;
201            let e = pic.blk[qi].edges;
202            if e & (E_TU_H | E_PU_H) == 0 || pic.blk[qi].flags & F_CODED == 0 {
203                continue;
204            }
205            let (x, y) = (bx * 4, by * 4);
206            let Some((tc_off, beta_off, ps, qs)) = edge_params(pic, x, y - 1, x, y) else { continue };
207            let pi = qi - w4;
208            let bs = boundary_strength(pic, pi, qi, e & E_TU_H != 0, ps, qs);
209            bs_h[(by - y0 / 4) * w4 + bx] = bs as i8;
210            if bs == 0 {
211                continue;
212            }
213            let (p, q) = (&pic.blk[pi], &pic.blk[qi]);
214            let qpl = (p.qp as i32 + q.qp as i32 + 1) >> 1;
215            let (tc, beta) = tc_beta(qpl, bs, tc_off, beta_off, bd);
216            let (nfp, nfq) = (p.flags & F_NOFILTER != 0, q.flags & F_NOFILTER != 0);
217            filter_luma(&mut pic.planes[0], EdgeSeg { o: y * w + x, step: w, along: 1 }, beta, EdgeFilter { tc, nfp, nfq, bd });
218        }
219    }
220    // chroma horizontal edges (8 chroma lines = 8 << sy luma lines apart; a 4-sample chroma
221    // segment spans 4 << sx luma samples), bS == 2 only
222    for by in (y0 / 4..y1 / 4).filter(|b| b % (2 << sy) == 0 && *b > 0) {
223        for bx in (0..w4).step_by(1 << sx) {
224            if bs_h[(by - y0 / 4) * w4 + bx] != 2 {
225                continue;
226            }
227            let (x, y) = (bx * 4, by * 4);
228            let qi = by * w4 + bx;
229            let Some((tc_off, ..)) = edge_params(pic, x, y - 1, x, y) else { continue };
230            let (p, q) = (pic.blk[qi - w4], pic.blk[qi]);
231            let (nfp, nfq) = (p.flags & F_NOFILTER != 0, q.flags & F_NOFILTER != 0);
232            for (c, off) in [(1, cb_off), (2, cr_off)] {
233                let qpi = ((p.qp as i32 + q.qp as i32 + 1) >> 1) + off;
234                let qt = (qpc(qpi, is420) + 2 + tc_off).clamp(0, 53);
235                let tc = (TC_TABLE[qt as usize] as i32) << (bdc - 8);
236                filter_chroma(
237                    &mut pic.planes[c],
238                    EdgeSeg { o: (y >> sy) * cw + (x >> sx), step: cw, along: 1 },
239                    EdgeFilter { tc, nfp, nfq, bd: bdc },
240                );
241            }
242        }
243    }
244}

246/// Where a 4-line edge segment lies in a plane: `o` = offset of q0 of the first line, `step` =
247/// distance between p/q samples across the edge, `along` = distance between lines.
248#[derive(Clone, Copy)]
249struct EdgeSeg {
250    o: usize,
251    step: usize,
252    along: usize,
253}
254
255/// How an edge segment is filtered: the clipping threshold `tc`, the no-filter flags of the p and
256/// q sides and the bit depth.
257#[derive(Clone, Copy)]
258struct EdgeFilter {
259    tc: i32,
260    nfp: bool,
261    nfq: bool,
262    bd: u32,
263}

265/// Filter one 4-sample luma edge segment.
266///
267/// The 4 x 8 samples (p3..p0, q0..q3 of each line) are read once into an array, filtered there
268/// and written back, a whole line at a time for vertical edges (`step` = 1) and a whole row of
269/// the 4 lines at a time for horizontal ones (`along` = 1): no per-sample index arithmetic or
270/// bounds checks in the filter itself.
271fn filter_luma<S: Sample>(s: &mut [S], seg: EdgeSeg, beta: i32, ef: EdgeFilter) {
272    let EdgeSeg { o, step, along } = seg;
273    let EdgeFilter { tc, nfp, nfq, bd } = ef;
274    // l[k][i]: line k, i = 0..4 -> p3, p2, p1, p0 and i = 4..8 -> q0, q1, q2, q3
275    let mut l = [[0i32; 8]; 4];
276    let Some(start) = o.checked_sub(4 * step) else { return };
277    if step == 1 {
278        for (k, line) in l.iter_mut().enumerate() {
279            let Some(src) = s.get(start + k * along..).and_then(|v| v.first_chunk::<8>()) else { return };
280            for (d, &v) in line.iter_mut().zip(src) {
281                *d = v.to_i32();
282            }
283        }
284    } else {
285        for i in 0..8 {
286            let Some(src) = s.get(start + i * step..).and_then(|v| v.first_chunk::<4>()) else { return };
287            for (k, &v) in src.iter().enumerate() {
288                l[k][i] = v.to_i32();
289            }
290        }
291    }
292    if !filter_luma_lines(&mut l, tc, beta, nfp, nfq, bd) {
293        return;
294    }
295    if step == 1 {
296        for (k, line) in l.iter().enumerate() {
297            let Some(dst) = s.get_mut(start + k * along..).and_then(|v| v.first_chunk_mut::<8>()) else { return };
298            for (d, &v) in dst.iter_mut().zip(line) {
299                *d = S::from_i32(v);
300            }
301        }
302    } else {
303        for i in 1..7 {
304            let Some(dst) = s.get_mut(start + i * step..).and_then(|v| v.first_chunk_mut::<4>()) else { return };
305            for (k, d) in dst.iter_mut().enumerate() {
306                *d = S::from_i32(l[k][i]);
307            }
308        }
309    }
310}

312/// The luma edge filter (8.7.2.5.3 decisions, 8.7.2.5.7 filtering) on the lines of
313/// [`filter_luma`]; false when the edge is left unfiltered (d >= beta).
314fn filter_luma_lines(l: &mut [[i32; 8]; 4], tc: i32, beta: i32, nfp: bool, nfq: bool, bd: u32) -> bool {
315    let px = |l: &[[i32; 8]; 4], k: usize, i: usize| l[k][3 - i];
316    let qx = |l: &[[i32; 8]; 4], k: usize, i: usize| l[k][4 + i];
317    let dp0 = (px(l, 0, 2) - 2 * px(l, 0, 1) + px(l, 0, 0)).abs();
318    let dp3 = (px(l, 3, 2) - 2 * px(l, 3, 1) + px(l, 3, 0)).abs();
319    let dq0 = (qx(l, 0, 2) - 2 * qx(l, 0, 1) + qx(l, 0, 0)).abs();
320    let dq3 = (qx(l, 3, 2) - 2 * qx(l, 3, 1) + qx(l, 3, 0)).abs();
321    let (dpq0, dpq3) = (dp0 + dq0, dp3 + dq3);
322    let (dp, dq) = (dp0 + dp3, dq0 + dq3);
323    let d = dpq0 + dpq3;
324    if d >= beta {
325        return false;
326    }
327    let dsam = |l: &[[i32; 8]; 4], k: usize, dpq: i32| -> bool {
328        dpq < (beta >> 2)
329            && (px(l, k, 3) - px(l, k, 0)).abs() + (qx(l, k, 0) - qx(l, k, 3)).abs() < (beta >> 3)
330            && (px(l, k, 0) - qx(l, k, 0)).abs() < ((5 * tc + 1) >> 1)
331    };
332    let strong = dsam(l, 0, 2 * dpq0) && dsam(l, 3, 2 * dpq3);
333    let dep = dp < ((beta + (beta >> 1)) >> 3);
334    let deq = dq < ((beta + (beta >> 1)) >> 3);
335    let max = (1i32 << bd) - 1;
336    for line in l.iter_mut() {
337        let [p3, p2, p1, p0, q0, q1, q2, q3] = *line;
338        if strong {
339            let t2 = 2 * tc;
340            if !nfp {
341                line[3] = ((p2 + 2 * p1 + 2 * p0 + 2 * q0 + q1 + 4) >> 3).clamp(p0 - t2, p0 + t2);
342                line[2] = ((p2 + p1 + p0 + q0 + 2) >> 2).clamp(p1 - t2, p1 + t2);
343                line[1] = ((2 * p3 + 3 * p2 + p1 + p0 + q0 + 4) >> 3).clamp(p2 - t2, p2 + t2);
344            }
345            if !nfq {
346                line[4] = ((p1 + 2 * p0 + 2 * q0 + 2 * q1 + q2 + 4) >> 3).clamp(q0 - t2, q0 + t2);
347                line[5] = ((p0 + q0 + q1 + q2 + 2) >> 2).clamp(q1 - t2, q1 + t2);
348                line[6] = ((p0 + q0 + q1 + 3 * q2 + 2 * q3 + 4) >> 3).clamp(q2 - t2, q2 + t2);
349            }
350        } else {
351            let mut delta = (9 * (q0 - p0) - 3 * (q1 - p1) + 8) >> 4;
352            if delta.abs() >= tc * 10 {
353                continue;
354            }
355            delta = delta.clamp(-tc, tc);
356            if !nfp {
357                line[3] = (p0 + delta).clamp(0, max);
358            }
359            if !nfq {
360                line[4] = (q0 - delta).clamp(0, max);
361            }
362            if dep && !nfp {
363                let dp = ((((p2 + p0 + 1) >> 1) - p1 + delta) >> 1).clamp(-(tc >> 1), tc >> 1);
364                line[2] = (p1 + dp).clamp(0, max);
365            }
366            if deq && !nfq {
367                let dq = ((((q2 + q0 + 1) >> 1) - q1 - delta) >> 1).clamp(-(tc >> 1), tc >> 1);
368                line[5] = (q1 + dq).clamp(0, max);
369            }
370        }
371    }
372    true
373}

375/// Filter one 4-sample chroma edge segment.
376fn filter_chroma<S: Sample>(s: &mut [S], seg: EdgeSeg, ef: EdgeFilter) {
377    let EdgeSeg { o, step, along } = seg;
378    let EdgeFilter { tc, nfp, nfq, bd } = ef;
379    let max = (1i32 << bd) - 1;
380    for k in 0..4 {
381        let base = o + k * along;
382        let (p0, p1) = (s[base - step].to_i32(), s[base - 2 * step].to_i32());
383        let (q0, q1) = (s[base].to_i32(), s[base + step].to_i32());
384        let delta = ((((q0 - p0) << 2) + p1 - q1 + 4) >> 3).clamp(-tc, tc);
385        if !nfp {
386            s[base - step] = S::from_i32((p0 + delta).clamp(0, max));
387        }
388        if !nfq {
389            s[base] = S::from_i32((q0 - delta).clamp(0, max));
390        }
391    }
392}

394/// SAO of CTB row `r` from the deblocked planes into new row buffers (luma, Cb, Cr).
395fn sao_row<S: Sample>(pic: &PicState<S>, r: usize) -> [Vec<S>; 3] {
396    let ctb = 1usize << pic.log2_ctb;
397    let (sx, sy) = pic.chroma.shifts();
398    let mut out: [Vec<S>; 3] = [Vec::new(), Vec::new(), Vec::new()];
399    for c in 0..3 {
400        // CTB size in this component's samples, horizontally and vertically
401        let (w, h, csx, csy) = if c == 0 { (pic.width, pic.height, ctb, ctb) } else { (pic.cwidth, pic.cheight, ctb >> sx, ctb >> sy) };
402        let y0 = r * csy;
403        let y1 = (y0 + csy).min(h);
404        out[c] = pic.planes[c][y0 * w..y1 * w].to_vec();
405        for rx in (0..pic.wctb).filter(|_| !pic.draft) {
406            let rs = r * pic.wctb + rx;
407            let si = pic.ctb_slice[rs];
408            if si == u32::MAX {
409                continue;
410            }
411            let sl = &pic.slices[si as usize];
412            if !(if c == 0 { sl.sao_luma } else { sl.sao_chroma }) {
413                continue;
414            }
415            let p = pic.sao[rs];
416            if p.type_idx[c] == 0 {
417                continue;
418            }
419            let x0 = rx * csx;
420            let x1 = (x0 + csx).min(w);
421            sao_ctb(pic, &SaoBlock { c, rs, x0, x1, y0, y1, w, h }, &p, &mut out[c]);
422        }
423    }
424    out
425}

427/// A CTB of plane `c` (CTB raster address `rs`) to apply SAO to: columns `x0..x1` and rows
428/// `y0..y1` of a plane `w` x `h` samples.
429#[derive(Clone, Copy)]
430struct SaoBlock {
431    c: usize,
432    rs: usize,
433    x0: usize,
434    x1: usize,
435    y0: usize,
436    y1: usize,
437    w: usize,
438    h: usize,
439}

441fn sao_ctb<S: Sample>(pic: &PicState<S>, blk: &SaoBlock, p: &crate::craft::video::hevc::decode::slicedec::SaoParams, dst: &mut [S]) {
442    let SaoBlock { c, rs, x0, x1, y0, y1, w, h } = *blk;
443    let bd = if c == 0 { pic.bd_y } else { pic.bd_c };
444    let max = (1i32 << bd) - 1;
445    // shifts from this component's sample positions to luma positions
446    let (scx, scy) = if c == 0 { (0, 0) } else { pic.chroma.shifts() };
447    let offs = p.offsets[c].map(|v| v as i32);
448    let plane = &pic.planes[c];
449    if p.type_idx[c] == 1 {
450        let band = (p.band_pos[c] as i32, offs);
451        for y in y0..y1 {
452            sao_band_run(&plane[y * w + x0..y * w + x1], &mut dst[(y - y0) * w + x0..(y - y0) * w + x1], bd - 5, band, max);
453        }
454    } else {
455        sao_edge_ctb(pic, c, rs, x0, x1, y0, y1, w, h, p, max, dst);
456    }
457    // Samples of CUs that must not be modified (transquant bypass, PCM without loop filter) get
458    // their deblocked values back.
459    if pic.pps.transquant_bypass || pic.sps.pcm_loop_filter_disabled {
460        // one 4x4 luma block covers (4 >> scx) x (4 >> scy) samples of this component
461        let (nx, ny) = (4 >> scx, 4 >> scy);
462        for by in ((y0 << scy) >> 2)..(((y1 << scy) + 3) >> 2).min(pic.h4) {
463            let ya = (by * 4) >> scy;
464            for bx in ((x0 << scx) >> 2)..(((x1 << scx) + 3) >> 2).min(pic.w4) {
465                if pic.blk[by * pic.w4 + bx].flags & F_NOFILTER == 0 {
466                    continue;
467                }
468                let xa = (bx * 4) >> scx;
469                let xb = (xa + nx).min(x1);
470                for y in ya..(ya + ny).min(y1) {
471                    dst[(y - y0) * w + xa..(y - y0) * w + xb].copy_from_slice(&plane[y * w + xa..y * w + xb]);
472                }
473            }
474        }
475    }
476}

478/// Edge offset of one CTB. Which of the 8 neighbouring CTBs may be used (inside the picture, not
479/// in another slice / tile without cross-boundary filtering) is worked out once; each line is then
480/// split into the left column, the inner run and the right column, and each part whose neighbours
481/// are usable goes through [`sao_edge_run`].
482#[allow(clippy::too_many_arguments)]
483fn sao_edge_ctb<S: Sample>(
484    pic: &PicState<S>,
485    c: usize,
486    rs: usize,
487    x0: usize,
488    x1: usize,
489    y0: usize,
490    y1: usize,
491    w: usize,
492    h: usize,
493    p: &crate::craft::video::hevc::decode::slicedec::SaoParams,
494    max: i32,
495    dst: &mut [S],
496) {
497    let offs = p.offsets[c].map(|v| v as i32);
498    let (hp, vp): ([isize; 2], [isize; 2]) = match p.eo_class[c] {
499        0 => ([-1, 1], [0, 0]),
500        1 => ([0, 0], [-1, 1]),
501        2 => ([-1, 1], [-1, 1]),
502        _ => ([1, -1], [-1, 1]),
503    };
504    let cur_slice = pic.ctb_slice[rs];
505    let lf_cur = pic.slices[cur_slice as usize].lf_across;
506    let cur_ts = pic.layout.rs_to_ts[rs];
507    let cross_ok = |nrs: usize| -> bool {
508        if nrs == rs {
509            return true;
510        }
511        let ns = pic.ctb_slice[nrs];
512        if ns == u32::MAX {
513            return false;
514        }
515        if pic.ctb_addr[nrs] != pic.ctb_addr[rs] {
516            // different slice: the later one (in decoding order) decides
517            let nts = pic.layout.rs_to_ts[nrs];
518            if nts < cur_ts && !lf_cur {
519                return false;
520            }
521            if cur_ts < nts && !pic.slices[ns as usize].lf_across {
522                return false;
523            }
524        }
525        pic.pps.loop_filter_across_tiles || pic.layout.tile_id[pic.layout.rs_to_ts[nrs] as usize] == pic.layout.tile_id[cur_ts as usize]
526    };
527    // avail[dy][dx]: the CTB at (rx + dx - 1, ry + dy - 1) may be used.
528    let (rx, ry) = (rs % pic.wctb, rs / pic.wctb);
529    let mut avail = [[false; 3]; 3];
530    for (dy, row) in avail.iter_mut().enumerate() {
531        for (dx, a) in row.iter_mut().enumerate() {
532            let inside = (dx != 0 || x0 > 0) && (dx != 2 || x1 < w) && (dy != 0 || y0 > 0) && (dy != 2 || y1 < h);
533            *a = inside && cross_ok((ry + dy - 1) * pic.wctb + rx + dx - 1);
534        }
535    }
536    let edge_map = [1usize, 2, 0, 3, 4];
537    // Offset per (2 + sum of neighbour signs); edgeIdx 0 adds nothing.
538    let by_sum = edge_map.map(|e| if e == 0 { 0 } else { offs[e - 1] });
539    let side = |v: isize, lo: usize, hi: usize| {
540        if v < lo as isize {
541            0
542        } else if v >= hi as isize {
543            2
544        } else {
545            1
546        }
547    };
548    let mx = (hp[0] != 0) as usize;
549    let plane = &pic.planes[c];
550    for y in y0..y1 {
551        let (ya, yb) = (y as isize + vp[0], y as isize + vp[1]);
552        let (dya, dyb) = (side(ya, y0, y1), side(yb, y0, y1));
553        for (xs, xe) in [(x0, x0 + mx), (x0 + mx, x1 - mx), (x1 - mx, x1)] {
554            if xs >= xe {
555                continue;
556            }
557            let (xa, xb) = (xs as isize + hp[0], xs as isize + hp[1]);
558            let n = xe - xs;
559            if !avail[dya][side(xa, x0, x1)] || !avail[dyb][side(xb, x0, x1)] {
560                continue;
561            }
562            let cur = &plane[y * w + xs..][..n];
563            let na = &plane[ya as usize * w + xa as usize..][..n];
564            let nb = &plane[yb as usize * w + xb as usize..][..n];
565            let out = &mut dst[(y - y0) * w + xs..][..n];
566            sao_edge_run(cur, na, nb, out, &by_sum, max);
567        }
568    }
569}

571/// Lanes of the SAO loops (16-bit arithmetic: samples are at most 12 bits, offsets 7 bits).
572const SAO_LANES: usize = 16;
573
574/// Edge offset of a run of samples whose two neighbours (`na`, `nb`) are all usable: per lane
575/// the sum of the two neighbour signs selects the offset (no table lookup), so the loop
576/// vectorises; bit-exact with the per-sample form.
577#[inline(always)]
578fn sao_edge_run<S: Sample>(cur: &[S], na: &[S], nb: &[S], out: &mut [S], by_sum: &[i32; 5], max: i32) {
579    let o = by_sum.map(|v| v as i16);
580    let max = max as i16;
581    let lane = |v: i16, a: i16, b: i16| -> S {
582        let s = ((v > a) as i16 - (v < a) as i16) + ((v > b) as i16 - (v < b) as i16);
583        let m = |k: i16| -((s == k) as i16);
584        let off = (o[0] & m(-2)) | (o[1] & m(-1)) | (o[3] & m(1)) | (o[4] & m(2));
585        S::from_i32((v + off).max(0).min(max) as i32)
586    };
587    let n = cur.len();
588    let full = n - n % SAO_LANES;
589    for i in (0..full).step_by(SAO_LANES) {
590        let (Some(c), Some(a), Some(b), Some(d)) = (lanes(cur, i), lanes(na, i), lanes(nb, i), lanes_mut(out, i)) else {
591            return;
592        };
593        for l in 0..SAO_LANES {
594            d[l] = lane(c[l].to_i16(), a[l].to_i16(), b[l].to_i16());
595        }
596    }
597    for i in full..n {
598        out[i] = lane(cur[i].to_i16(), na[i].to_i16(), nb[i].to_i16());
599    }
600}
601
602/// `SAO_LANES` samples from `i`, if all present.
603#[inline(always)]
604fn lanes<S>(s: &[S], i: usize) -> Option<&[S; SAO_LANES]> {
605    s.get(i..)?.first_chunk()
606}
607
608#[inline(always)]
609fn lanes_mut<S>(s: &mut [S], i: usize) -> Option<&mut [S; SAO_LANES]> {
610    s.get_mut(i..)?.first_chunk_mut()
611}

613/// Band offset of a run of samples: the band index relative to `band_pos` selects one of the
614/// four offsets per lane.
615#[inline(always)]
616fn sao_band_run<S: Sample>(src: &[S], out: &mut [S], shift: u32, (band_pos, offs): (i32, [i32; 4]), max: i32) {
617    let o = offs.map(|v| v as i16);
618    let (bp, max) = (band_pos as i16, max as i16);
619    let lane = |v: i16| -> S {
620        let k = ((v >> shift) - bp) & 31;
621        let m = |j: i16| -((k == j) as i16);
622        let off = (o[0] & m(0)) | (o[1] & m(1)) | (o[2] & m(2)) | (o[3] & m(3));
623        S::from_i32((v + off).max(0).min(max) as i32)
624    };
625    let n = src.len();
626    let full = n - n % SAO_LANES;
627    for i in (0..full).step_by(SAO_LANES) {
628        let (Some(c), Some(d)) = (lanes(src, i), lanes_mut(out, i)) else {
629            return;
630        };
631        for l in 0..SAO_LANES {
632            d[l] = lane(c[l].to_i16());
633        }
634    }
635    for i in full..n {
636        out[i] = lane(src[i].to_i16());
637    }
638}

640/// Publish CTB row `r` (final samples + compressed motion) into the frame.
641fn publish_row<S: Sample>(pic: &mut PicState<S>, r: usize) {
642    let ctb = 1usize << pic.log2_ctb;
643    let (y0, y1) = (r * ctb, ((r + 1) * ctb).min(pic.height));
644    let w16 = pic.width.div_ceil(16);
645    let rows16 = (y1 - y0).div_ceil(16);
646    let mut col = vec![ColMv::default(); w16 * rows16];
647    for j in 0..rows16 {
648        for i in 0..w16 {
649            let (x, y) = (i * 16, y0 + j * 16);
650            let bi = (y >> 2) * pic.w4 + (x >> 2);
651            let b = &pic.blk[bi];
652            if b.flags & F_CODED == 0 || b.flags & F_INTRA != 0 {
653                continue;
654            }
655            let f = pic.mvf[bi];
656            let ctbi = (y >> pic.log2_ctb) * pic.wctb + (x >> pic.log2_ctb);
657            let si = pic.ctb_slice[ctbi];
658            if si == u32::MAX {
659                continue;
660            }
661            let sl = &pic.slices[si as usize];
662            let mut cm = ColMv { mv: f.mv, ref_poc: [0; 2], flags: 0 };
663            for l in 0..2 {
664                if f.pred(l) {
665                    let ri = f.ref_idx[l] as usize;
666                    if ri < sl.ref_pocs[l].len() {
667                        cm.flags |= 1 << l;
668                        cm.ref_poc[l] = sl.ref_pocs[l][ri];
669                        if sl.ref_lt[l][ri] {
670                            cm.flags |= 4 << l;
671                        }
672                    }
673                }
674            }
675            col[j * w16 + i] = cm;
676        }
677    }
678    let [oy, ocb, ocr] = sao_row(pic, r);
679    let row = FrameRow { y: oy.into(), cb: ocb.into(), cr: ocr.into(), col: col.into() };
680    pic.frame.publish(r, row);
681}

685    use super::*;

687    /// The luma edge filter as it was before reading the samples into an array (indexing the
688    /// picture sample by sample), as the reference.
689    fn reference(s: &mut [u16], seg: EdgeSeg, beta: i32, ef: EdgeFilter) {
690        let EdgeSeg { o, step, along } = seg;
691        let EdgeFilter { tc, nfp, nfq, bd } = ef;
692        let px = |s: &[u16], k: usize, i: usize| s[o + k * along - (i + 1) * step] as i32;
693        let qx = |s: &[u16], k: usize, i: usize| s[o + k * along + i * step] as i32;
694        let dp0 = (px(s, 0, 2) - 2 * px(s, 0, 1) + px(s, 0, 0)).abs();
695        let dp3 = (px(s, 3, 2) - 2 * px(s, 3, 1) + px(s, 3, 0)).abs();
696        let dq0 = (qx(s, 0, 2) - 2 * qx(s, 0, 1) + qx(s, 0, 0)).abs();
697        let dq3 = (qx(s, 3, 2) - 2 * qx(s, 3, 1) + qx(s, 3, 0)).abs();
698        let (dpq0, dpq3) = (dp0 + dq0, dp3 + dq3);
699        let (dp, dq) = (dp0 + dp3, dq0 + dq3);
700        let d = dpq0 + dpq3;
701        if d >= beta {
702            return;
703        }
704        let dsam = |s: &[u16], k: usize, dpq: i32| -> bool {
705            dpq < (beta >> 2)
706                && (px(s, k, 3) - px(s, k, 0)).abs() + (qx(s, k, 0) - qx(s, k, 3)).abs() < (beta >> 3)
707                && (px(s, k, 0) - qx(s, k, 0)).abs() < ((5 * tc + 1) >> 1)
708        };
709        let strong = dsam(s, 0, 2 * dpq0) && dsam(s, 3, 2 * dpq3);
710        let dep = dp < ((beta + (beta >> 1)) >> 3);
711        let deq = dq < ((beta + (beta >> 1)) >> 3);
712        let max = (1i32 << bd) - 1;
713        for k in 0..4 {
714            let base = o + k * along;
715            let p = |i: usize| s[base - (i + 1) * step] as i32;
716            let q = |i: usize| s[base + i * step] as i32;
717            let (p0, p1, p2, p3) = (p(0), p(1), p(2), p(3));
718            let (q0, q1, q2, q3) = (q(0), q(1), q(2), q(3));
719            if strong {
720                let t2 = 2 * tc;
721                if !nfp {
722                    s[base - step] = ((p2 + 2 * p1 + 2 * p0 + 2 * q0 + q1 + 4) >> 3).clamp(p0 - t2, p0 + t2) as u16;
723                    s[base - 2 * step] = ((p2 + p1 + p0 + q0 + 2) >> 2).clamp(p1 - t2, p1 + t2) as u16;
724                    s[base - 3 * step] = ((2 * p3 + 3 * p2 + p1 + p0 + q0 + 4) >> 3).clamp(p2 - t2, p2 + t2) as u16;
725                }
726                if !nfq {
727                    s[base] = ((p1 + 2 * p0 + 2 * q0 + 2 * q1 + q2 + 4) >> 3).clamp(q0 - t2, q0 + t2) as u16;
728                    s[base + step] = ((p0 + q0 + q1 + q2 + 2) >> 2).clamp(q1 - t2, q1 + t2) as u16;
729                    s[base + 2 * step] = ((p0 + q0 + q1 + 3 * q2 + 2 * q3 + 4) >> 3).clamp(q2 - t2, q2 + t2) as u16;
730                }
731            } else {
732                let mut delta = (9 * (q0 - p0) - 3 * (q1 - p1) + 8) >> 4;
733                if delta.abs() >= tc * 10 {
734                    continue;
735                }
736                delta = delta.clamp(-tc, tc);
737                if !nfp {
738                    s[base - step] = (p0 + delta).clamp(0, max) as u16;
739                }
740                if !nfq {
741                    s[base] = (q0 - delta).clamp(0, max) as u16;
742                }
743                if dep && !nfp {
744                    let dp = ((((p2 + p0 + 1) >> 1) - p1 + delta) >> 1).clamp(-(tc >> 1), tc >> 1);
745                    s[base - 2 * step] = (p1 + dp).clamp(0, max) as u16;
746                }
747                if deq && !nfq {
748                    let dq = ((((q2 + q0 + 1) >> 1) - q1 - delta) >> 1).clamp(-(tc >> 1), tc >> 1);
749                    s[base + step] = (q1 + dq).clamp(0, max) as u16;
750                }
751            }
752        }
753    }

755    /// Same samples out of the gathered-block filter as out of the reference, for vertical and
756    /// horizontal edges, strong and weak filtering, both bit depths and the no-filter flags.
757    #[cfg(any())]
758    fn gathered_luma_filter_matches_the_per_sample_filter() {
759        let mut rng = 0x2545_f491_4f6c_dd1du64;
760        let mut next = move || {
761            rng ^= rng << 13;
762            rng ^= rng >> 7;
763            rng ^= rng << 17;
764            rng
765        };
766        let w = 16usize;
767        let mut filtered = 0;
768        for case in 0..4000 {
769            let bd = if case % 2 == 0 { 8 } else { 10 };
770            let max = (1u64 << bd) - 1;
771            // smooth blocks (strong filtering) to noisy ones (no filtering)
772            let base = next() % (max + 1);
773            let spread = [0u64, 2, 8, 64, max][case % 5];
774            let plane: Vec<u16> = (0..w * w).map(|_| (base + next() % (spread + 1)).min(max) as u16).collect();
775            let tc = (next() % 40) as i32;
776            let beta = (next() % 120) as i32;
777            let (nfp, nfq) = (next() % 4 == 0, next() % 4 == 0);
778            for vertical in [true, false] {
779                let (o, step, along) = if vertical { (4 * w + 8, 1, w) } else { (8 * w + 4, w, 1) };
780                let (mut a, mut b) = (plane.clone(), plane.clone());
781                let (seg, ef) = (EdgeSeg { o, step, along }, EdgeFilter { tc, nfp, nfq, bd });
782                filter_luma(&mut a, seg, beta, ef);
783                reference(&mut b, seg, beta, ef);
784                assert_eq!(a, b, "case {case} vertical {vertical}");
785                filtered += (a != plane) as usize;
786            }
787        }
788        assert!(filtered > 1000, "only {filtered} edges filtered");
789    }
