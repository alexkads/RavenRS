5/// Up-right diagonal scan (6.5.3) of a `N x N` block as (x, y) pairs.
6const fn diag<const L: usize>(n: usize) -> [(u8, u8); L] {
7    let mut out = [(0u8, 0u8); L];
8    let mut i = 0;
9    let mut x: i32 = 0;
10    let mut y: i32 = 0;
11    while i < L {
12        while y >= 0 {
13            if (x as usize) < n && (y as usize) < n {
14                out[i] = (x as u8, y as u8);
15                i += 1;
16            }
17            y -= 1;
18            x += 1;
19        }
20        y = x;
21        x = 0;
22    }
23    out
24}

26/// Horizontal (6.5.4, `vertical == false`) or vertical (6.5.5) scan.
27const fn hv<const L: usize>(n: usize, vertical: bool) -> [(u8, u8); L] {
28    let mut out = [(0u8, 0u8); L];
29    let mut i = 0;
30    while i < L {
31        let (a, b) = ((i % n) as u8, (i / n) as u8);
32        out[i] = if vertical { (b, a) } else { (a, b) };
33        i += 1;
34    }
35    out
36}

38pub static SCAN_1: [[(u8, u8); 1]; 3] = [[(0, 0)], [(0, 0)], [(0, 0)]];

43/// ScanOrder[log2BlockSize][scanIdx] (log2BlockSize 0..=3).
44#[inline]
45pub fn scan_order(log2: u32, scan_idx: usize) -> &'static [(u8, u8)] {
46    match log2 {
47        0 => &SCAN_1[scan_idx],
48        1 => &SCAN_2[scan_idx],
49        2 => &SCAN_4[scan_idx],
50        _ => &SCAN_8[scan_idx],
51    }
52}
53
54/// Up-right diagonal scan for log2 size 2 or 3.
55pub fn scan_diag(log2: u32) -> &'static [(u8, u8)] {
56    scan_order(log2, 0)
57}

59/// Table 8-10: QpC as a function of qPi (ChromaArrayType 1).
60#[inline]
61pub fn qpc_420(qpi: i32) -> i32 {
62    if qpi < 30 {
63        qpi
64    } else if qpi > 43 {
65        qpi - 6
66    } else {
67        QPC_30_43[(qpi - 30) as usize] as i32
68    }
69}

96    /// Table 8-3 (Rec. ITU-T H.265 v8 (08/2021), 8.4.3) is the 4:2:2 form of each luma direction:
97    /// chroma has half the horizontal resolution, so the direction of mode m (Table 8-5's
98    /// intraPredAngle, horizontal modes 2..=17 and vertical 18..=34) with its horizontal component
99    /// halved, mapped to the angular mode nearest in angle. Planar and DC stay. This rule
100    /// reproduces every one of the 35 entries, so a mistyped entry fails here without an oracle.
101    #[cfg(any())]
102    fn the_422_chroma_mode_table_is_the_luma_direction_with_half_the_width() {
103        let dir = |m: usize| -> (f64, f64) {
104            let a = f64::from(INTRA_PRED_ANGLE[m]);
105            if m < 18 { (-32.0, a) } else { (a, -32.0) }
106        };
107        let angle = |(x, y): (f64, f64)| y.atan2(x);
108        let gap = |a: f64, b: f64| {
109            let d = (a - b).rem_euclid(std::f64::consts::TAU);
110            d.min(std::f64::consts::TAU - d)
111        };
112        for m in 0..35usize {
113            let want = if m < 2 {
114                m
115            } else {
116                let (x, y) = dir(m);
117                let t = angle((x / 2.0, y));
118                let mut best = 2;
119                for k in 2..35 {
120                    if gap(angle(dir(k)), t) < gap(angle(dir(best)), t) - 1e-12 {
121                        best = k;
122                    }
123                }
124                best
125            };
126            assert_eq!(INTRA_CHROMA_MODE_422[m] as usize, want, "luma mode {m}");
127        }
128    }
