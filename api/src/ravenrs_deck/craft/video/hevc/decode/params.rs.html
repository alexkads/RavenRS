9/// HEVC picture state includes several per-pixel and per-block work buffers. This permits 8K
10/// pictures while rejecting dimensions that could exhaust memory during decode.
11pub const MAX_LUMA_SAMPLES: u64 = 1 << 26;
12
13fn check_picture_size(width: u32, height: u32) -> Result<()> {
14    ensure!(width > 0 && height > 0 && width <= 16888 && height <= 16888, "picture size {width}x{height} out of range");
15    ensure!(u64::from(width) * u64::from(height) <= MAX_LUMA_SAMPLES, "picture size {width}x{height} exceeds the decoded-frame budget");
16    Ok(())
17}

22    pub profile_space: u8,
23    pub tier: bool,
24    pub profile_idc: u8,
25    pub compatibility: u32,
26    pub progressive_source: bool,
27    pub interlaced_source: bool,
28    pub level_idc: u8,
29}

31pub fn parse_ptl(r: &mut BitReader, profile_present: bool, max_sub_layers_minus1: u32) -> Result<ProfileTierLevel> {
32    let mut p = ProfileTierLevel::default();
33    if profile_present {
34        p.profile_space = r.read_bits(2)? as u8;
35        p.tier = r.read_flag()?;
36        p.profile_idc = r.read_bits(5)? as u8;
37        p.compatibility = r.read_bits(32)?;
38        p.progressive_source = r.read_flag()?;
39        p.interlaced_source = r.read_flag()?;
40        r.skip(2)?; // non_packed, frame_only
41        r.skip(43)?; // constraint flags / reserved bits (always 43 bits)
42        r.skip(1)?; // inbld / reserved
43    }
44    p.level_idc = r.read_bits(8)? as u8;
45    let n = max_sub_layers_minus1 as usize;
46    let mut sub_profile = [false; 8];
47    let mut sub_level = [false; 8];
48    for i in 0..n {
49        sub_profile[i] = r.read_flag()?;
50        sub_level[i] = r.read_flag()?;
51    }
52    if n > 0 {
53        for _ in n..8 {
54            r.skip(2)?;
55        }
56    }
57    for i in 0..n {
58        if sub_profile[i] {
59            r.skip(2 + 1 + 5 + 32 + 4 + 43 + 1)?;
60        }
61        if sub_level[i] {
62            r.skip(8)?;
63        }
64    }
65    Ok(p)
66}

68fn skip_sub_layer_hrd(r: &mut BitReader, cpb_cnt: u32, sub_pic: bool) -> Result<()> {
69    for _ in 0..cpb_cnt {
70        r.read_ue()?;
71        r.read_ue()?;
72        if sub_pic {
73            r.read_ue()?;
74            r.read_ue()?;
75        }
76        r.skip(1)?;
77    }
78    Ok(())
79}

81/// hrd_parameters() (E.2.2), skipped.
82pub fn skip_hrd(r: &mut BitReader, common: bool, max_sub_layers_minus1: u32) -> Result<()> {
83    let (mut nal, mut vcl, mut sub_pic) = (false, false, false);
84    if common {
85        nal = r.read_flag()?;
86        vcl = r.read_flag()?;
87        if nal || vcl {
88            sub_pic = r.read_flag()?;
89            if sub_pic {
90                r.skip(8 + 5 + 1 + 5)?;
91            }
92            r.skip(8)?;
93            if sub_pic {
94                r.skip(4)?;
95            }
96            r.skip(15)?;
97        }
98    }
99    for _ in 0..=max_sub_layers_minus1 {
100        let fixed_general = r.read_flag()?;
101        let fixed_cvs = if !fixed_general { r.read_flag()? } else { true };
102        let mut low_delay = false;
103        if fixed_cvs {
104            r.read_ue()?;
105        } else {
106            low_delay = r.read_flag()?;
107        }
108        let mut cpb_cnt = 1;
109        if !low_delay {
110            cpb_cnt = r.read_ue_plus(1)?;
111            ensure!(cpb_cnt <= 32, "cpb_cnt_minus1 out of range");
112        }
113        if nal {
114            skip_sub_layer_hrd(r, cpb_cnt, sub_pic)?;
115        }
116        if vcl {
117            skip_sub_layer_hrd(r, cpb_cnt, sub_pic)?;
118        }
119    }
120    Ok(())
121}

131impl Vps {

169const SAR_TABLE: [(u16, u16); 17] = [
170    (0, 0),
171    (1, 1),
172    (12, 11),
173    (10, 11),
174    (16, 11),
175    (40, 33),
176    (24, 11),
177    (20, 11),
178    (32, 11),
179    (80, 33),
180    (18, 11),
181    (15, 11),
182    (64, 33),
183    (160, 99),
184    (4, 3),
185    (3, 2),
186    (2, 1),
187];
188
189fn parse_vui(r: &mut BitReader, max_sub_layers_minus1: u32) -> Result<Vui> {
190    let mut v = Vui::default();
191    if r.read_flag()? {
192        let idc = r.read_bits(8)?;
193        if idc == 255 {
194            v.sar = (r.read_bits(16)? as u16, r.read_bits(16)? as u16);
195        } else if (idc as usize) < SAR_TABLE.len() {
196            v.sar = SAR_TABLE[idc as usize];
197        }
198    }
199    if r.read_flag()? {
200        r.skip(1)?;
201    }
202    if r.read_flag()? {
203        r.skip(3)?;
204        v.full_range = r.read_flag()?;
205        if r.read_flag()? {
206            v.colour_primaries = r.read_bits(8)? as u8;
207            v.transfer_characteristics = r.read_bits(8)? as u8;
208            v.matrix_coefficients = r.read_bits(8)? as u8;
209        }
210    }
211    if r.read_flag()? {
212        r.read_ue()?;
213        r.read_ue()?;
214    }
215    r.skip(1)?; // neutral_chroma_indication_flag
216    v.field_seq = r.read_flag()?;
217    r.skip(1)?; // frame_field_info_present_flag
218    if r.read_flag()? {
219        for _ in 0..4 {
220            r.read_ue()?;
221        }
222    }
223    if r.read_flag()? {
224        let units = r.read_bits(32)?;
225        let scale = r.read_bits(32)?;
226        v.timing = Some((units, scale));
227        if r.read_flag()? {
228            r.read_ue()?;
229        }
230        if r.read_flag()? {
231            skip_hrd(r, true, max_sub_layers_minus1)?;
232        }
233    }
234    if r.read_flag()? {
235        r.skip(3)?;
236        for _ in 0..5 {
237            r.read_ue()?;
238        }
239    }
240    Ok(v)
241}

243/// Scaling list data (7.3.4) with prediction resolved: lists in up-right diagonal coefficient order.
244#[derive(Clone, Debug, PartialEq, Eq)]
245pub struct ScalingList {
246    /// `[sizeId][matrixId][i]`; sizeId 0 uses 16 entries.
247    pub lists: [[[u8; 64]; 6]; 4],
248    /// scaling_list_dc_coef for sizeId 2, 3.
249    pub dc: [[u8; 6]; 2],
250}
251
252impl ScalingList {
253    /// Default lists (Tables 7-5, 7-6).
254    pub fn default_lists() -> Self {
255        let mut lists = [[[16u8; 64]; 6]; 4];
256        for size in 1..4 {
257            for m in 0..6 {
258                lists[size][m] = if m < 3 { DEFAULT_SCALING_INTRA } else { DEFAULT_SCALING_INTER };
259            }
260        }
261        ScalingList { lists, dc: [[16; 6]; 2] }
262    }
263
264    pub fn parse(r: &mut BitReader) -> Result<Self> {
265        let mut sl = Self::default_lists();
266        for size in 0..4 {
267            let step = if size == 3 { 3 } else { 1 };
268            let mut m = 0;
269            while m < 6 {
270                let coef_num = if size == 0 { 16 } else { 64 };
271                if !r.read_flag()? {
272                    let delta = (r.read_ue()? as usize).saturating_mul(step);
273                    ensure!(delta <= m, "scaling_list_pred_matrix_id_delta out of range");
274                    if delta == 0 {
275                        // default list
276                        let d = Self::default_lists();
277                        sl.lists[size][m] = d.lists[size][m];
278                        if size > 1 {
279                            sl.dc[size - 2][m] = 16;
280                        }
281                    } else {
282                        let rm = m - delta;
283                        sl.lists[size][m] = sl.lists[size][rm];
284                        if size > 1 {
285                            sl.dc[size - 2][m] = sl.dc[size - 2][rm];
286                        }
287                    }
288                } else {
289                    let mut next: i32 = 8;
290                    if size > 1 {
291                        let dc = r.read_se()?;
292                        ensure!((-7..=247).contains(&dc), "scaling_list_dc_coef_minus8 out of range");
293                        next = dc + 8;
294                        sl.dc[size - 2][m] = next as u8;
295                    }
296                    for i in 0..coef_num {
297                        let d = r.read_se()?;
298                        ensure!((-128..=127).contains(&d), "scaling_list_delta_coef out of range");
299                        next = (next + d + 256).rem_euclid(256);
300                        sl.lists[size][m][i] = next as u8;
301                    }
302                }
303                m += step;
304            }
305        }
306        // 32x32 chroma lists (only used for ChromaArrayType 3) are copied from the 16x16 ones.
307        for m in [1, 2, 4, 5] {
308            sl.lists[3][m] = sl.lists[2][m];
309            sl.dc[1][m] = sl.dc[0][m];
310        }
311        Ok(sl)
312    }
313
314    /// ScalingFactor for sizeId (log2 size = sizeId + 2) and matrixId as a row-major `[y][x]` matrix
315    /// (7-44..7-48).
316    pub fn factor(&self, size_id: usize, matrix_id: usize) -> Vec<u8> {
317        let n = 4usize << size_id;
318        let mut out = vec![0u8; n * n];
319        if size_id == 0 {
320            let scan = scan_diag(2);
321            for (i, &(x, y)) in scan.iter().enumerate() {
322                out[y as usize * 4 + x as usize] = self.lists[0][matrix_id][i];
323            }
324            return out;
325        }
326        let rep = n / 8;
327        let scan = scan_diag(3);
328        for (i, &(x, y)) in scan.iter().enumerate() {
329            let v = self.lists[size_id][matrix_id][i];
330            for j in 0..rep {
331                for k in 0..rep {
332                    out[(y as usize * rep + j) * n + x as usize * rep + k] = v;
333                }
334            }
335        }
336        if size_id >= 2 {
337            out[0] = self.dc[size_id - 2][matrix_id];
338        }
339        out
340    }
341}

343/// A short-term reference picture set (7.4.8), resolved.
344#[derive(Clone, Debug, Default, PartialEq, Eq)]
345pub struct StRps {
346    /// DeltaPocS0 (negative, decreasing) with UsedByCurrPicS0.
347    pub s0: Vec<(i32, bool)>,
348    /// DeltaPocS1 (positive, increasing) with UsedByCurrPicS1.
349    pub s1: Vec<(i32, bool)>,
350}
351
352impl StRps {
353    pub fn num_delta_pocs(&self) -> usize {
354        self.s0.len() + self.s1.len()
355    }
356
357    /// st_ref_pic_set(idx) with the previously parsed candidate sets `sets` (idx == sets.len() for a
358    /// slice-header RPS).
359    pub fn parse(r: &mut BitReader, idx: usize, sets: &[StRps], num_sets: usize) -> Result<StRps> {
360        let inter = if idx != 0 { r.read_flag()? } else { false };
361        if inter {
362            let delta_idx = if idx == num_sets { (r.read_ue()? as usize).saturating_add(1) } else { 1 };
363            ensure!(delta_idx <= idx, "delta_idx_minus1 out of range");
364            let rf = &sets[idx - delta_idx];
365            let sign = r.read_flag()?;
366            let abs = r.read_ue_plus(1)?;
367            ensure!(abs <= 1 << 15, "abs_delta_rps_minus1 out of range");
368            let abs = abs as i32;
369            let delta_rps = if sign { -abs } else { abs };
370            let n = rf.num_delta_pocs();
371            let mut used = vec![false; n + 1];
372            let mut use_delta = vec![true; n + 1];
373            for j in 0..=n {
374                used[j] = r.read_flag()?;
375                if !used[j] {
376                    use_delta[j] = r.read_flag()?;
377                }
378            }
379            let nneg = rf.s0.len();
380            let mut out = StRps::default();
381            for j in (0..rf.s1.len()).rev() {
382                let d = rf.s1[j].0 + delta_rps;
383                if d < 0 && use_delta[nneg + j] {
384                    out.s0.push((d, used[nneg + j]));
385                }
386            }
387            if delta_rps < 0 && use_delta[n] {
388                out.s0.push((delta_rps, used[n]));
389            }
390            for j in 0..nneg {
391                let d = rf.s0[j].0 + delta_rps;
392                if d < 0 && use_delta[j] {
393                    out.s0.push((d, used[j]));
394                }
395            }
396            for j in (0..nneg).rev() {
397                let d = rf.s0[j].0 + delta_rps;
398                if d > 0 && use_delta[j] {
399                    out.s1.push((d, used[j]));
400                }
401            }
402            if delta_rps > 0 && use_delta[n] {
403                out.s1.push((delta_rps, used[n]));
404            }
405            for j in 0..rf.s1.len() {
406                let d = rf.s1[j].0 + delta_rps;
407                if d > 0 && use_delta[nneg + j] {
408                    out.s1.push((d, used[nneg + j]));
409                }
410            }
411            ensure!(out.num_delta_pocs() <= 16, "too many pictures in RPS");
412            Ok(out)
413        } else {
414            let nneg = r.read_ue()? as usize;
415            let npos = r.read_ue()? as usize;
416            ensure!(nneg <= 16 && npos <= 16 && nneg.saturating_add(npos) <= 16, "too many pictures in RPS");
417            let mut out = StRps::default();
418            let mut poc = 0i32;
419            for _ in 0..nneg {
420                let d = r.read_ue_plus(1)?;
421                ensure!(d <= 1 << 15, "delta_poc_s0_minus1 out of range");
422                let d = d as i32;
423                poc -= d;
424                out.s0.push((poc, r.read_flag()?));
425            }
426            poc = 0;
427            for _ in 0..npos {
428                let d = r.read_ue_plus(1)?;
429                ensure!(d <= 1 << 15, "delta_poc_s1_minus1 out of range");
430                let d = d as i32;
431                poc += d;
432                out.s1.push((poc, r.read_flag()?));
433            }
434            Ok(out)
435        }
436    }
437}

487impl Sps {
488    pub fn parse(rbsp: &[u8]) -> Result<Sps> {
489        let mut r = BitReader::new(rbsp);
490        let vps_id = r.read_bits(4)? as u8;
491        let max_sub_layers_minus1 = r.read_bits(3)?;
492        ensure!(max_sub_layers_minus1 <= 6, "sps_max_sub_layers_minus1 out of range");
493        r.skip(1)?;
494        let ptl = parse_ptl(&mut r, true, max_sub_layers_minus1)?;
495        let id = r.read_ue()?;
496        ensure!(id < 16, "sps_seq_parameter_set_id out of range");
497        let chroma_format_idc = r.read_ue()?;
498        ensure!(chroma_format_idc <= 3, "chroma_format_idc out of range");
499        let separate_colour_plane = if chroma_format_idc == 3 { r.read_flag()? } else { false };
500        let width = r.read_ue()?;
501        let height = r.read_ue()?;
502        check_picture_size(width, height)?;
503        let mut conf_win = (0, 0, 0, 0);
504        if r.read_flag()? {
505            conf_win = (r.read_ue()?, r.read_ue()?, r.read_ue()?, r.read_ue()?);
506        }
507        let bit_depth_luma = r.read_ue_plus(8)?;
508        let bit_depth_chroma = r.read_ue_plus(8)?;
509        ensure!(bit_depth_luma <= 16 && bit_depth_chroma <= 16, "bit depth out of range");
510        let log2_max_poc_lsb = r.read_ue_plus(4)?;
511        ensure!(log2_max_poc_lsb <= 16, "log2_max_pic_order_cnt_lsb_minus4 out of range");
512        let ordering_all = r.read_flag()?;
513        let (mut dpb, mut reorder, mut latency) = (1, 0, 0);
514        let start = if ordering_all { 0 } else { max_sub_layers_minus1 };
515        for _ in start..=max_sub_layers_minus1 {
516            // the values of the highest sub-layer are used
517            dpb = r.read_ue_plus(1)?;
518            reorder = r.read_ue()?;
519            latency = r.read_ue()?;
520        }
521        ensure!(dpb <= 16 && reorder <= 16, "sps_max_dec_pic_buffering out of range");
522        let log2_min_cb = r.read_ue_plus(3)?;
523        let log2_ctb = log2_min_cb.saturating_add(r.read_ue()?);
524        let log2_min_tb = r.read_ue_plus(2)?;
525        let log2_max_tb = log2_min_tb.saturating_add(r.read_ue()?);
526        ensure!((4..=6).contains(&log2_ctb) && log2_min_cb <= log2_ctb, "CTB size out of range");
527        ensure!(log2_min_tb < log2_min_cb && log2_max_tb <= log2_ctb.min(5), "transform block sizes out of range");
528        ensure!(width % (1 << log2_min_cb) == 0 && height % (1 << log2_min_cb) == 0, "picture size not a multiple of MinCbSizeY");
529        let max_th_depth_inter = r.read_ue()?;
530        let max_th_depth_intra = r.read_ue()?;
531        ensure!(
532            max_th_depth_inter <= log2_ctb - log2_min_tb && max_th_depth_intra <= log2_ctb - log2_min_tb,
533            "transform hierarchy depth out of range"
534        );
535        let scaling_list_enabled = r.read_flag()?;
536        let mut scaling_list = None;
537        if scaling_list_enabled {
538            scaling_list = Some(if r.read_flag()? { ScalingList::parse(&mut r)? } else { ScalingList::default_lists() });
539        }
540        let amp = r.read_flag()?;
541        let sao = r.read_flag()?;
542        let pcm = r.read_flag()?;
543        let (mut pcm_bit_depth_luma, mut pcm_bit_depth_chroma, mut log2_min_pcm, mut log2_max_pcm, mut pcm_loop_filter_disabled) =
544            (8, 8, 0, 0, false);
545        if pcm {
546            pcm_bit_depth_luma = r.read_bits(4)? + 1;
547            pcm_bit_depth_chroma = r.read_bits(4)? + 1;
548            log2_min_pcm = r.read_ue_plus(3)?;
549            log2_max_pcm = log2_min_pcm.saturating_add(r.read_ue()?);
550            pcm_loop_filter_disabled = r.read_flag()?;
551            ensure!(pcm_bit_depth_luma <= bit_depth_luma && pcm_bit_depth_chroma <= bit_depth_chroma, "PCM bit depth out of range");
552            ensure!(log2_max_pcm <= log2_ctb.min(5), "PCM size out of range");
553        }
554        let num_st = r.read_ue()? as usize;
555        ensure!(num_st <= 64, "num_short_term_ref_pic_sets out of range");
556        let mut st_rps = Vec::with_capacity(num_st);
557        for i in 0..num_st {
558            let s = StRps::parse(&mut r, i, &st_rps, num_st)?;
559            st_rps.push(s);
560        }
561        let long_term_refs_present = r.read_flag()?;
562        let mut lt_ref_pics = Vec::new();
563        if long_term_refs_present {
564            let n = r.read_ue()?;
565            ensure!(n <= 32, "num_long_term_ref_pics_sps out of range");
566            for _ in 0..n {
567                lt_ref_pics.push((r.read_bits(log2_max_poc_lsb)?, r.read_flag()?));
568            }
569        }
570        let temporal_mvp = r.read_flag()?;
571        let strong_intra_smoothing = r.read_flag()?;
572        let vui = if r.read_flag()? { Some(parse_vui(&mut r, max_sub_layers_minus1)?) } else { None };
573        let mut range_extension = false;
574        let mut extension_flags = 0;
575        if r.read_flag()? {
576            extension_flags = r.read_bits(8)? as u8;
577            if extension_flags & 0x80 != 0 {
578                // sps_range_extension(): any enabled tool is outside Main / Main 10
579                let bits = r.read_bits(9)?;
580                range_extension = bits != 0;
581            }
582        }
583        Ok(Sps {
584            vps_id,
585            max_sub_layers_minus1,
586            ptl,
587            id,
588            chroma_format_idc,
589            separate_colour_plane,

598            max_latency_increase_plus1: latency,

615            st_rps,

617            lt_ref_pics,
618            temporal_mvp,
619            strong_intra_smoothing,
620            vui,
621            range_extension,
622            extension_flags,
623        })
624    }

626    /// ChromaArrayType.
627    pub fn chroma_array_type(&self) -> u32 {
628        if self.separate_colour_plane { 0 } else { self.chroma_format_idc }
629    }
630    pub fn sub_width_c(&self) -> u32 {
631        if self.chroma_format_idc == 1 || self.chroma_format_idc == 2 { 2 } else { 1 }
632    }
633    pub fn sub_height_c(&self) -> u32 {
634        if self.chroma_format_idc == 1 { 2 } else { 1 }
635    }
636    pub fn ctb_size(&self) -> u32 {
637        1 << self.log2_ctb
638    }
639    pub fn pic_width_in_ctbs(&self) -> u32 {
640        self.width.div_ceil(self.ctb_size())
641    }
642    pub fn pic_height_in_ctbs(&self) -> u32 {
643        self.height.div_ceil(self.ctb_size())
644    }
645    pub fn max_poc_lsb(&self) -> i32 {
646        1 << self.log2_max_poc_lsb
647    }

649    /// Cropped output rectangle (x, y, width, height) in luma samples.
650    pub fn crop_rect(&self) -> (u32, u32, u32, u32) {
651        let (l, r, t, b) = self.conf_win;
652        let (sw, sh) = (self.sub_width_c(), self.sub_height_c());
653        let x = l.saturating_mul(sw).min(self.width.saturating_sub(1));
654        let y = t.saturating_mul(sh).min(self.height.saturating_sub(1));
655        let w = self.width.saturating_sub(x.saturating_add(r.saturating_mul(sw))).max(1);
656        let h = self.height.saturating_sub(y.saturating_add(b.saturating_mul(sh))).max(1);
657        (x, y, w, h)
658    }

660    /// Features this decoder does not handle yet.
661    pub fn check_supported(&self) -> Result<()> {
662        if self.chroma_array_type() != 1 && self.chroma_array_type() != 2 {
663            let what = match self.chroma_format_idc {
664                0 => "monochrome".to_string(),
665                3 if self.separate_colour_plane => "4:4:4 with separate colour planes".to_string(),
666                3 => "4:4:4".to_string(),
667                idc => format!("chroma_format_idc {idc}"),
668            };
669            return unsupported(format!("{what} (only 4:2:0 and 4:2:2)"));
670        }
671        if self.bit_depth_luma > 12 || self.bit_depth_chroma > 12 {
672            return unsupported(format!("bit depth {}/{}", self.bit_depth_luma, self.bit_depth_chroma));
673        }
674        if self.range_extension {
675            return unsupported("SPS range extension tools");
676        }
677        if self.extension_flags & 0x7f != 0 {
678            return unsupported("SPS multilayer / 3D / SCC extensions");
679        }
680        Ok(())
681    }

730impl Pps {
731    pub fn parse(rbsp: &[u8]) -> Result<Pps> {
732        let mut r = BitReader::new(rbsp);
733        let id = r.read_ue()?;
734        ensure!(id < 64, "pps_pic_parameter_set_id out of range");
735        let sps_id = r.read_ue()?;
736        ensure!(sps_id < 16, "pps_seq_parameter_set_id out of range");
737        let dependent_slice_segments_enabled = r.read_flag()?;
738        let output_flag_present = r.read_flag()?;
739        let num_extra_slice_header_bits = r.read_bits(3)?;
740        let sign_data_hiding = r.read_flag()?;
741        let cabac_init_present = r.read_flag()?;
742        let num_ref_idx_l0_default = r.read_ue_plus(1)?;
743        let num_ref_idx_l1_default = r.read_ue_plus(1)?;
744        ensure!(num_ref_idx_l0_default <= 15 && num_ref_idx_l1_default <= 15, "num_ref_idx_default_active out of range");
745        // init_qp_minus26 is −(26 + QpBdOffsetY) ..= 25. Bit depth is not known here (no SPS), so the
746        // widest legal window is 16-bit (QpBdOffsetY = 48). Reject before adding 26.
747        let init_qp_minus26 = r.read_se()?;
748        ensure!((-74..=25).contains(&init_qp_minus26), "init_qp_minus26 out of range");
749        let init_qp = 26 + init_qp_minus26;
750        let constrained_intra_pred = r.read_flag()?;
751        let transform_skip = r.read_flag()?;
752        let cu_qp_delta_enabled = r.read_flag()?;
753        let diff_cu_qp_delta_depth = if cu_qp_delta_enabled { r.read_ue()? } else { 0 };
754        ensure!(diff_cu_qp_delta_depth <= 3, "diff_cu_qp_delta_depth out of range");
755        let cb_qp_offset = r.read_se()?;
756        let cr_qp_offset = r.read_se()?;
757        ensure!((-12..=12).contains(&cb_qp_offset) && (-12..=12).contains(&cr_qp_offset), "chroma QP offset out of range");
758        let slice_chroma_qp_offsets_present = r.read_flag()?;
759        let weighted_pred = r.read_flag()?;
760        let weighted_bipred = r.read_flag()?;
761        let transquant_bypass = r.read_flag()?;
762        let tiles_enabled = r.read_flag()?;
763        let entropy_coding_sync = r.read_flag()?;
764        let (mut num_tile_columns, mut num_tile_rows, mut uniform_spacing) = (1, 1, true);
765        let (mut column_widths, mut row_heights) = (Vec::new(), Vec::new());
766        let mut loop_filter_across_tiles = true;
767        if tiles_enabled {
768            num_tile_columns = r.read_ue_plus(1)?;
769            num_tile_rows = r.read_ue_plus(1)?;
770            ensure!(num_tile_columns <= 64 && num_tile_rows <= 64, "too many tiles");
771            uniform_spacing = r.read_flag()?;
772            if !uniform_spacing {
773                for _ in 0..num_tile_columns - 1 {
774                    column_widths.push(r.read_ue()?.saturating_add(1));
775                }
776                for _ in 0..num_tile_rows - 1 {
777                    row_heights.push(r.read_ue()?.saturating_add(1));
778                }
779            }
780            loop_filter_across_tiles = r.read_flag()?;
781        }
782        let loop_filter_across_slices = r.read_flag()?;
783        let deblocking_control_present = r.read_flag()?;
784        let (mut deblocking_override_enabled, mut deblocking_disabled, mut beta_offset_div2, mut tc_offset_div2) = (false, false, 0, 0);
785        if deblocking_control_present {
786            deblocking_override_enabled = r.read_flag()?;
787            deblocking_disabled = r.read_flag()?;
788            if !deblocking_disabled {
789                beta_offset_div2 = r.read_se()?;
790                tc_offset_div2 = r.read_se()?;
791                ensure!((-6..=6).contains(&beta_offset_div2) && (-6..=6).contains(&tc_offset_div2), "deblocking offsets out of range");
792            }
793        }
794        let scaling_list = if r.read_flag()? { Some(ScalingList::parse(&mut r)?) } else { None };
795        let lists_modification_present = r.read_flag()?;
796        let log2_parallel_merge_level = r.read_ue_plus(2)?;
797        ensure!(log2_parallel_merge_level <= 6, "log2_parallel_merge_level out of range");
798        let slice_header_extension_present = r.read_flag()?;
799        let mut range_extension = false;
800        let mut extension_flags = 0;
801        if r.read_flag()? {
802            extension_flags = r.read_bits(8)? as u8;
803            if extension_flags & 0x80 != 0 {
804                // pps_range_extension(): treated as unsupported when it enables anything
805                let mut any = false;
806                if transform_skip {
807                    any |= r.read_ue()? != 0;
808                }
809                any |= r.read_flag()?; // cross_component_prediction
810                let chroma_qp_offset_list = r.read_flag()?;
811                any |= chroma_qp_offset_list;
812                if !chroma_qp_offset_list {
813                    // (with the list enabled the PPS is unsupported already and the list sits
814                    // before these fields)
815                    any |= r.read_ue()? != 0; // log2_sao_offset_scale_luma
816                    any |= r.read_ue()? != 0; // log2_sao_offset_scale_chroma
817                }
818                range_extension = any;
819            }
820        }
821        Ok(Pps {
822            id,
823            sps_id,
824            dependent_slice_segments_enabled,
825            output_flag_present,
826            num_extra_slice_header_bits,
827            sign_data_hiding,
828            cabac_init_present,
829            num_ref_idx_l0_default,
830            num_ref_idx_l1_default,
831            init_qp,
832            constrained_intra_pred,
833            transform_skip,
834            cu_qp_delta_enabled,
835            diff_cu_qp_delta_depth,
836            cb_qp_offset,
837            cr_qp_offset,
838            slice_chroma_qp_offsets_present,
839            weighted_pred,
840            weighted_bipred,
841            transquant_bypass,
842            tiles_enabled,
843            entropy_coding_sync,
844            num_tile_columns,
845            num_tile_rows,
846            uniform_spacing,
847            column_widths,
848            row_heights,
849            loop_filter_across_tiles,
850            loop_filter_across_slices,
851            deblocking_control_present,
852            deblocking_override_enabled,
853            deblocking_disabled,
854            beta_offset_div2,
855            tc_offset_div2,
856            scaling_list,
857            lists_modification_present,
858            log2_parallel_merge_level,
859            slice_header_extension_present,
860            range_extension,
861            extension_flags,
862        })
863    }

876/// CTB addressing for a PPS/SPS pair: tile boundaries and the raster <-> tile scan conversion
877/// (6.5.1).
878#[derive(Clone, Debug)]
879pub struct Layout {
880    pub width_ctbs: u32,
881    pub height_ctbs: u32,
882    /// Tile column boundaries in CTBs (num_tile_columns + 1 entries).
883    pub col_bd: Vec<u32>,
884    pub row_bd: Vec<u32>,
885    pub rs_to_ts: Vec<u32>,
886    pub ts_to_rs: Vec<u32>,
887    pub tile_id: Vec<u32>,
888}
889
890impl Layout {
891    pub fn new(sps: &Sps, pps: &Pps) -> Result<Layout> {
892        let (w, h) = (sps.pic_width_in_ctbs(), sps.pic_height_in_ctbs());
893        let (nc, nr) = (pps.num_tile_columns, pps.num_tile_rows);
894        ensure!(nc <= w && nr <= h, "more tiles than CTBs");
895        let mut colw = Vec::new();
896        let mut rowh = Vec::new();
897        if pps.uniform_spacing {
898            for i in 0..nc {
899                colw.push((i + 1) * w / nc - i * w / nc);
900            }
901            for j in 0..nr {
902                rowh.push((j + 1) * h / nr - j * h / nr);
903            }
904        } else {
905            let sw: u64 = pps.column_widths.iter().map(|&v| v as u64).sum();
906            let sh: u64 = pps.row_heights.iter().map(|&v| v as u64).sum();
907            ensure!(sw < w as u64 && sh < h as u64, "tile sizes exceed the picture");
908            colw.extend_from_slice(&pps.column_widths);
909            colw.push(w - sw as u32);
910            rowh.extend_from_slice(&pps.row_heights);
911            rowh.push(h - sh as u32);
912        }
913        let mut col_bd = vec![0];
914        let mut acc = 0;
915        for c in &colw {
916            acc += c;
917            col_bd.push(acc);
918        }
919        let mut row_bd = vec![0];
920        let mut acc = 0;
921        for r in &rowh {
922            acc += r;
923            row_bd.push(acc);
924        }
925        let n = (w * h) as usize;
926        let mut rs_to_ts = vec![0u32; n];
927        for rs in 0..w * h {
928            let (tbx, tby) = (rs % w, rs / w);
929            let tile_x = (0..nc as usize).rev().find(|&i| tbx >= col_bd[i]).unwrap_or(0);
930            let tile_y = (0..nr as usize).rev().find(|&j| tby >= row_bd[j]).unwrap_or(0);
931            let mut v = 0;
932            for i in 0..tile_x {
933                v += rowh[tile_y] * colw[i];
934            }
935            for j in 0..tile_y {
936                v += w * rowh[j];
937            }
938            v += (tby - row_bd[tile_y]) * colw[tile_x] + tbx - col_bd[tile_x];
939            rs_to_ts[rs as usize] = v;
940        }
941        let mut ts_to_rs = vec![0u32; n];
942        for rs in 0..n {
943            ts_to_rs[rs_to_ts[rs] as usize] = rs as u32;
944        }
945        let mut tile_id = vec![0u32; n];
946        let mut tid = 0;
947        for j in 0..nr as usize {
948            for i in 0..nc as usize {
949                for y in row_bd[j]..row_bd[j + 1] {
950                    for x in col_bd[i]..col_bd[i + 1] {
951                        tile_id[rs_to_ts[(y * w + x) as usize] as usize] = tid;
952                    }
953                }
954                tid += 1;
955            }
956        }
957        Ok(Layout { width_ctbs: w, height_ctbs: h, col_bd, row_bd, rs_to_ts, ts_to_rs, tile_id })
958    }
959
960    /// Tile id of the CTB at raster address `rs`.
961    #[inline]
962    pub fn tile_of_rs(&self, rs: u32) -> u32 {
963        self.tile_id[self.rs_to_ts[rs as usize] as usize]
964    }
