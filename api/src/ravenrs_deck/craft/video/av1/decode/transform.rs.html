8#[inline(always)]
9fn round2_64(x: i64, n: u32) -> i32 {
10    if n == 0 { x as i32 } else { ((x + (1i64 << (n - 1))) >> n) as i32 }
11}

13#[inline(always)]
14pub(crate) fn round2(x: i32, n: u32) -> i32 {
15    if n == 0 { x } else { (x + (1 << (n - 1))) >> n }
16}

18/// cos128( angle ) for every `angle & 255`.
19static COS128: [i32; 256] = {
20    let mut t = [0; 256];
21    let mut a = 0;
22    while a < 256 {
23        t[a] = match a {
24            0..=64 => COS128_LOOKUP[a] as i32,
25            65..=128 => -(COS128_LOOKUP[128 - a] as i32),
26            129..=192 => -(COS128_LOOKUP[a - 128] as i32),
27            _ => COS128_LOOKUP[256 - a] as i32,
28        };
29        a += 1;
30    }
31    t
32};
33
34#[inline(always)]
35fn cos128(angle: i32) -> i64 {
36    COS128[(angle & 255) as usize] as i64
37}
38
39#[inline(always)]
40fn sin128(angle: i32) -> i64 {
41    cos128(angle - 64)
42}

44/// One value of the 1D transforms, or `L` independent columns / rows of them processed
45/// together (lane-wise, so the steps vectorise).
46pub(crate) trait Lane: Copy {
47    const ZERO: Self;
48    /// (Round2( a * cos - b * sin, 12 ), Round2( a * sin + b * cos, 12 ))
49    fn rotate(a: Self, b: Self, cos: i32, sin: i32) -> (Self, Self);
50    /// (Clamp( x + y ), Clamp( x - y )) to [lo, hi]
51    fn hadamard(x: Self, y: Self, lo: i32, hi: i32) -> (Self, Self);
52    fn neg(self) -> Self;
53    /// Round2( v * m, 12 )
54    fn mul_round12(self, m: i32) -> Self;
55    fn mul(self, k: i32) -> Self;
56    fn adst4(t: &mut [Self]);
57}
58
59impl Lane for i32 {
60    const ZERO: i32 = 0;
61    #[inline(always)]
62    fn rotate(a: i32, b: i32, cos: i32, sin: i32) -> (i32, i32) {
63        let (a, b, c, s) = (a as i64, b as i64, cos as i64, sin as i64);
64        (round2_64(a * c - b * s, 12), round2_64(a * s + b * c, 12))
65    }
66    #[inline(always)]
67    fn hadamard(x: i32, y: i32, lo: i32, hi: i32) -> (i32, i32) {
68        ((x + y).clamp(lo, hi), (x - y).clamp(lo, hi))
69    }
70    #[inline(always)]
71    fn neg(self) -> i32 {
72        -self
73    }
74    #[inline(always)]
75    fn mul_round12(self, m: i32) -> i32 {
76        round2_64(self as i64 * m as i64, 12)
77    }
78    #[inline(always)]
79    fn mul(self, k: i32) -> i32 {
80        self * k
81    }
82    fn adst4(t: &mut [i32]) {
83        inverse_adst4_scalar(t);
84    }
85}

87/// Lanes in 32-bit arithmetic: exact for conforming streams up to 10 bits per sample (the
88/// spec requires every intermediate value to fit in 8 + BitDepth <= 18 bits, so products with
89/// the 12-bit cosines stay below 2^31). Wrapping operations keep corrupt streams from
90/// panicking.
91impl<const L: usize> Lane for [i32; L] {
92    const ZERO: Self = [0; L];
93    #[inline(always)]
94    fn rotate(a: Self, b: Self, cos: i32, sin: i32) -> (Self, Self) {
95        let mut x = [0; L];
96        let mut y = [0; L];
97        for l in 0..L {
98            let p = a[l].wrapping_mul(cos).wrapping_sub(b[l].wrapping_mul(sin));
99            let q = a[l].wrapping_mul(sin).wrapping_add(b[l].wrapping_mul(cos));
100            x[l] = p.wrapping_add(1 << 11) >> 12;
101            y[l] = q.wrapping_add(1 << 11) >> 12;
102        }
103        (x, y)
104    }
105    #[inline(always)]
106    fn hadamard(x: Self, y: Self, lo: i32, hi: i32) -> (Self, Self) {
107        let mut a = [0; L];
108        let mut b = [0; L];
109        for l in 0..L {
110            a[l] = x[l].wrapping_add(y[l]).max(lo).min(hi);
111            b[l] = x[l].wrapping_sub(y[l]).max(lo).min(hi);
112        }
113        (a, b)
114    }
115    #[inline(always)]
116    fn neg(self) -> Self {
117        self.map(|v| v.wrapping_neg())
118    }
119    #[inline(always)]
120    fn mul_round12(self, m: i32) -> Self {
121        self.map(|v| v.wrapping_mul(m).wrapping_add(1 << 11) >> 12)
122    }
123    #[inline(always)]
124    fn mul(self, k: i32) -> Self {
125        self.map(|v| v.wrapping_mul(k))
126    }
127    fn adst4(t: &mut [Self]) {
128        for l in 0..L {
129            let mut v = [t[0][l], t[1][l], t[2][l], t[3][l]];
130            inverse_adst4_scalar(&mut v);
131            for k in 0..4 {
132                t[k][l] = v[k];
133            }
134        }
135    }
136}

138const fn brev(num_bits: u32, x: usize) -> usize {
139    let mut t = 0;
140    let mut i = 0;
141    while i < num_bits {
142        let bit = (x >> i) & 1;
143        t += bit << (num_bits - 1 - i);
144        i += 1;
145    }
146    t
147}
148
149/// BREV[n][x] = brev( n, x ) for n <= 6.
150const BREV: [[u8; 64]; 7] = {
151    let mut t = [[0; 64]; 7];
152    let mut n = 0;
153    while n < 7 {
154        let mut x = 0;
155        while x < 1 << n {
156            t[n][x] = brev(n as u32, x) as u8;
157            x += 1;
158        }
159        n += 1;
160    }
161    t
162};

164/// Butterfly rotation B( a, b, angle, flip ).
165#[inline(always)]
166fn bf<T: Lane>(t: &mut [T], a: usize, b: usize, angle: i32, flip: bool) {
167    let (x, y) = T::rotate(t[a], t[b], cos128(angle) as i32, sin128(angle) as i32);
168    t[a] = x;
169    t[b] = y;
170    if flip {
171        t.swap(a, b);
172    }
173}

175/// Hadamard rotation H( a, b, flip, r ).
176#[inline(always)]
177fn hd<T: Lane>(t: &mut [T], a: usize, b: usize, flip: bool, r: u32) {
178    let (a, b) = if flip { (b, a) } else { (a, b) };
179    let lo = -(1i32 << (r - 1));
180    let hi = (1i32 << (r - 1)) - 1;
181    let (x, y) = T::hadamard(t[a], t[b], lo, hi);
182    t[a] = x;
183    t[b] = y;
184}

186/// Rotations and Hadamards that build the odd half of the larger transforms (first part of 7.13.2.3).
187#[inline(always)]
188fn dct_stage_outer_rotations<T: Lane, const N: u32>(t: &mut [T], r: u32) {
189    if N == 6 {
190        for i in 0..16 {
191            bf(t, 32 + i, 63 - i, 63 - 4 * BREV[4][i] as i32, false);
192        }
193    }
194    if N >= 5 {
195        for i in 0..8 {
196            bf(t, 16 + i, 31 - i, 6 + ((BREV[3][7 - i] as i32) << 3), false);
197        }
198    }
199    if N == 6 {
200        for i in 0..16 {
201            hd(t, 32 + i * 2, 33 + i * 2, i & 1 == 1, r);
202        }
203    }
204    if N >= 4 {
205        for i in 0..4 {
206            bf(t, 8 + i, 15 - i, 12 + ((BREV[2][3 - i] as i32) << 4), false);
207        }
208    }
209    if N >= 5 {
210        for i in 0..8 {
211            hd(t, 16 + 2 * i, 17 + 2 * i, i & 1 == 1, r);
212        }
213    }
214    if N == 6 {
215        for i in 0..4 {
216            for j in 0..2 {
217                bf(t, 62 - i * 4 - j, 33 + i * 4 + j, 60 - 16 * BREV[2][i] as i32 + 64 * j as i32, true);
218            }
219        }
220    }
221    if N >= 3 {
222        for i in 0..2 {
223            bf(t, 4 + i, 7 - i, 56 - 32 * i as i32, false);
224        }
225    }
226    if N >= 4 {
227        for i in 0..4 {
228            hd(t, 8 + 2 * i, 9 + 2 * i, i & 1 == 1, r);
229        }
230    }
231    if N >= 5 {
232        for i in 0..2 {
233            for j in 0..2 {
234                bf(t, 30 - 4 * i - j, 17 + 4 * i + j, 24 + ((j as i32) << 6) + (((1 - i) as i32) << 5), true);
235            }
236        }
237    }
238    if N == 6 {
239        for i in 0..8 {

378fn inverse_dct_n<T: Lane, const N: u32>(t: &mut [T], r: u32) {
379    // 7.13.2.2 permutation
380    let len = 1usize << N;
381    let mut copy = [T::ZERO; 64];
382    copy[..len].copy_from_slice(&t[..len]);
383    for i in 0..len {
384        t[i] = copy[BREV[N as usize][i] as usize];
385    }
386    dct_stage_outer_rotations::<T, N>(t, r);
387    dct_stage_even_half::<T, N>(t, r);
388    dct_stage_merge_quarters::<T, N>(t, r);
389    dct_stage_merge_halves::<T, N>(t, r);
390}

392const SINPI_1_9: i64 = 1321;
393const SINPI_2_9: i64 = 2482;
394const SINPI_3_9: i64 = 3344;
395const SINPI_4_9: i64 = 3803;

397fn inverse_adst4_scalar(t: &mut [i32]) {
398    let mut s = [0i64; 7];
399    let mut x = [0i64; 4];
400    let t0 = t[0] as i64;
401    let t1 = t[1] as i64;
402    let t2 = t[2] as i64;
403    let t3 = t[3] as i64;
404    s[0] = SINPI_1_9 * t0;
405    s[1] = SINPI_2_9 * t0;
406    s[2] = SINPI_3_9 * t1;
407    s[3] = SINPI_4_9 * t2;
408    s[4] = SINPI_1_9 * t2;
409    s[5] = SINPI_2_9 * t3;
410    s[6] = SINPI_4_9 * t3;
411    let a7 = t0 - t2;
412    let b7 = a7 + t3;
413    s[0] += s[3];
414    s[1] -= s[4];
415    s[3] = s[2];
416    s[2] = SINPI_3_9 * b7;
417    s[0] += s[5];
418    s[1] -= s[6];
419    x[0] = s[0] + s[3];
420    x[1] = s[1] + s[3];
421    x[2] = s[2];
422    x[3] = s[0] + s[1];
423    x[3] -= s[3];
424    for i in 0..4 {
425        t[i] = round2_64(x[i], 12);
426    }
427}

429fn adst_in_permute<T: Lane>(t: &mut [T], n: u32) {
430    let n0 = 1usize << n;
431    let mut copy = [T::ZERO; 16];
432    copy[..n0].copy_from_slice(&t[..n0]);
433    for i in 0..n0 {
434        let idx = if i & 1 == 1 { i - 1 } else { n0 - i - 1 };
435        t[i] = copy[idx];
436    }
437}

439fn adst_out_permute<T: Lane>(t: &mut [T], n: u32) {
440    let n0 = 1usize << n;
441    let mut copy = [T::ZERO; 16];
442    copy[..n0].copy_from_slice(&t[..n0]);
443    for i in 0..n0 {
444        let a = (i >> 3) & 1;
445        let b = ((i >> 2) & 1) ^ ((i >> 3) & 1);
446        let c = ((i >> 1) & 1) ^ ((i >> 2) & 1);
447        let d = (i & 1) ^ ((i >> 1) & 1);
448        let idx = ((d << 3) | (c << 2) | (b << 1) | a) >> (4 - n);
449        t[i] = if i & 1 == 1 { copy[idx].neg() } else { copy[idx] };
450    }
451}

541fn inverse_wht(t: &mut [i32], shift: u32) {
542    let mut a = t[0] >> shift;
543    let mut c = t[1] >> shift;
544    let mut d = t[2] >> shift;
545    let mut b = t[3] >> shift;
546    a += c;
547    d -= b;
548    let e = (a - d) >> 1;
549    b = e - b;
550    c = e - c;
551    a -= b;
552    d += c;
553    t[0] = a;
554    t[1] = b;
555    t[2] = c;
556    t[3] = d;
557}

582/// 2D inverse transform (7.13.3). `dequant` is 64x64 (row stride 64, only the top-left 32x32
583/// may be non-zero; with `dc_only` only the first value); the residual is written to
584/// `residual` with row stride `w`.
585pub(crate) fn inverse_transform_2d(
586    dequant: &[i32],
587    residual: &mut [i32],
588    tx_sz: usize,
589    tx_type: usize,
590    lossless: bool,
591    bit_depth: u32,
592    dc_only: bool,
593) {
594    let log2w = TX_WIDTH_LOG2[tx_sz] as u32;
595    let log2h = TX_HEIGHT_LOG2[tx_sz] as u32;
596    let w = 1usize << log2w;
597    let h = 1usize << log2h;
598    let row_shift = if lossless { 0 } else { TRANSFORM_ROW_SHIFT[tx_sz] as u32 };
599    let col_shift = if lossless { 0 } else { 4 };
600    let row_clamp = bit_depth + 8;
601    let col_clamp = (bit_depth + 6).max(16);
602    let rk = row_kind(tx_type);
603    let ck = col_kind(tx_type);
604    let mut t = [0i32; 64];
605    let rect2 = log2w.abs_diff(log2h) == 1;
606    if !lossless && bit_depth <= 10 {
607        let p = Tx2d { w, h, log2w, log2h, row_shift, col_shift, row_clamp, col_clamp, rk, ck, rect2 };
608        if dc_only && rk == Kind1d::Dct && ck == Kind1d::Dct && transform_dc_only(dequant[0], residual, &p) {
609            return;
610        }
611        if w >= 8 && h >= 8 {
612            transform_lanes::<8>(dequant, residual, &p);
613        } else {
614            transform_lanes::<4>(dequant, residual, &p);
615        }
616        return;
617    }
618    let in_w = w.min(32);
619    for i in 0..h {
620        // Every 1D transform maps an all-zero row to zeros (each step is linear with
621        // Round2( 0 ) = 0), so rows without coefficients (including rows 32..63) are zero.
622        if i >= 32 || dequant[i * 64..i * 64 + in_w].iter().all(|&v| v == 0) {
623            residual[i * w..i * w + w].fill(0);
624            continue;
625        }
626        t[..in_w].copy_from_slice(&dequant[i * 64..i * 64 + in_w]);
627        t[in_w..w].fill(0);
628        if rect2 {
629            for v in t[..w].iter_mut() {
630                *v = round2_64(*v as i64 * 2896, 12);
631            }
632        }
633        if lossless {
634            inverse_wht(&mut t, 2);
635        } else {
636            match rk {
637                Kind1d::Dct => inverse_dct(&mut t, log2w, row_clamp),
638                Kind1d::Adst => inverse_adst(&mut t, log2w, row_clamp),
639                Kind1d::Identity => inverse_identity(&mut t, log2w),
640            }
641        }
642        let lo = -(1i32 << (col_clamp - 1));
643        let hi = (1i32 << (col_clamp - 1)) - 1;
644        for j in 0..w {
645            residual[i * w + j] = round2(t[j], row_shift).clamp(lo, hi);
646        }
647    }
648    for j in 0..w {
649        for i in 0..h {
650            t[i] = residual[i * w + j];
651        }
652        if lossless {
653            inverse_wht(&mut t, 0);
654        } else {
655            match ck {
656                Kind1d::Dct => inverse_dct(&mut t, log2h, col_clamp),
657                Kind1d::Adst => inverse_adst(&mut t, log2h, col_clamp),
658                Kind1d::Identity => inverse_identity(&mut t, log2h),
659            }
660        }
661        for i in 0..h {
662            residual[i * w + j] = round2(t[i], col_shift);
663        }
664    }
665}
666
667struct Tx2d {
668    w: usize,
669    h: usize,
670    log2w: u32,
671    log2h: u32,
672    row_shift: u32,
673    col_shift: u32,
674    row_clamp: u32,
675    col_clamp: u32,
676    rk: Kind1d,
677    ck: Kind1d,
678    rect2: bool,
679}
680
681#[inline(always)]
682fn transform_1d<T: Lane>(t: &mut [T], kind: Kind1d, n: u32, r: u32) {
683    match kind {
684        Kind1d::Dct => inverse_dct(t, n, r),
685        Kind1d::Adst => inverse_adst(t, n, r),
686        Kind1d::Identity => inverse_identity(t, n),
687    }
688}
689
690/// [`transform_lanes`] for a DCT_DCT block whose only coefficient is DC: the other rows are
691/// zero and every column gets the same input, so one row and one column are transformed and
692/// the column is replicated. Returns false (nothing written) if the row output is not flat.
693fn transform_dc_only(dc: i32, residual: &mut [i32], p: &Tx2d) -> bool {
694    let (w, h) = (p.w, p.h);
695    let lo = -(1i32 << (p.col_clamp - 1));
696    let hi = (1i32 << (p.col_clamp - 1)) - 1;
697    let mut t = [[0i32; 1]; 64];
698    t[0] = [dc];
699    if p.rect2 {
700        t[0] = t[0].mul_round12(2896);
701    }
702    inverse_dct(&mut t[..], p.log2w, p.row_clamp);
703    let v = round2(t[0][0], p.row_shift).clamp(lo, hi);
704    if t[1..w].iter().any(|x| round2(x[0], p.row_shift).clamp(lo, hi) != v) {
705        return false;
706    }
707    let mut c = [[0i32; 1]; 64];
708    c[0] = [v];
709    inverse_dct(&mut c[..], p.log2h, p.col_clamp);
710    for i in 0..h {
711        residual[i * w..i * w + w].fill(round2(c[i][0], p.col_shift));
712    }
713    true
714}
715
716/// The 2D inverse transform with `L` rows (row pass) or columns (column pass) at a time;
717/// bit-identical to the one-at-a-time process for bit depths up to 10.
718fn transform_lanes<const L: usize>(dequant: &[i32], residual: &mut [i32], p: &Tx2d) {
719    let (w, h) = (p.w, p.h);
720    let in_w = w.min(32);
721    let lo = -(1i32 << (p.col_clamp - 1));
722    let hi = (1i32 << (p.col_clamp - 1)) - 1;
723    let mut t = [[0i32; L]; 64];
724    let mut i0 = 0;
725    while i0 < h {
726        // Rows without coefficients (and rows 32..63) transform to zeros.
727        if i0 >= 32 || (i0..i0 + L).all(|i| dequant[i * 64..i * 64 + in_w].iter().all(|&v| v == 0)) {
728            residual[i0 * w..(i0 + L) * w].fill(0);
729            i0 += L;
730            continue;
731        }
732        for (j, tj) in t[..w].iter_mut().enumerate() {
733            for l in 0..L {
734                tj[l] = if j < in_w { dequant[(i0 + l) * 64 + j] } else { 0 };
735            }
736        }
737        if p.rect2 {
738            for v in t[..w].iter_mut() {
739                *v = v.mul_round12(2896);
740            }
741        }
742        transform_1d(&mut t[..], p.rk, p.log2w, p.row_clamp);
743        for l in 0..L {
744            let out = &mut residual[(i0 + l) * w..(i0 + l + 1) * w];
745            for j in 0..w {
746                out[j] = round2(t[j][l], p.row_shift).clamp(lo, hi);
747            }
748        }
749        i0 += L;
750    }
751    let mut j0 = 0;
752    while j0 < w {
753        for i in 0..h {
754            t[i].copy_from_slice(&residual[i * w + j0..i * w + j0 + L]);
755        }
756        transform_1d(&mut t[..], p.ck, p.log2h, p.col_clamp);
757        for i in 0..h {
758            let out = &mut residual[i * w + j0..i * w + j0 + L];
759            for l in 0..L {
760                out[l] = round2(t[i][l], p.col_shift);
761            }
762        }
763        j0 += L;
764    }
765}
