1//! Symbol (arithmetic) decoder, spec 8.2.
2//!
3//! The state follows the specification literally: `SymbolValue` / `SymbolRange` are 15/16-bit
4//! quantities renormalised by reading `bits` new bits at a time, and `SymbolMaxBits` counts the
5//! bits still available (going negative while zero padding is consumed).
6
7const EC_PROB_SHIFT: u32 = 6;
8const EC_MIN_PROB: u32 = 4;
9
10pub(crate) struct SymbolDecoder<'a> {
11    data: &'a [u8],
12    /// Next byte of `data` to load into the window.
13    next: usize,
14    /// SymbolValue followed by `k` look-ahead bits: the bitstream with every bit inverted
15    /// (the spec's renormalisation XORs the new bits with ones), zero bits past the end of
16    /// the data reading as ones. SymbolValue = win >> k.
17    win: u64,
18    k: u32,
19    range: u32,
20    pub disable_update: bool,
21}
22
23impl<'a> SymbolDecoder<'a> {
24    /// `init_symbol( sz )` over `data` (the tile's bytes).
25    pub fn new(data: &'a [u8], disable_update: bool) -> Self {
26        let mut d = SymbolDecoder { data, next: 0, win: 0, k: 0, range: 1 << 15, disable_update };
27        // SymbolValue = ((1 << 15) - 1) ^ (first 15 bits): 15 inverted bits.
28        let mut filled = 0;
29        while filled < 15 + 32 {
30            d.win = (d.win << 8) | d.inverted_byte() as u64;
31            filled += 8;
32        }
33        d.k = filled - 15;
34        d
35    }
36
37    #[inline(always)]
38    fn inverted_byte(&mut self) -> u8 {
39        let b = self.data.get(self.next).copied().unwrap_or(0);
40        self.next += 1;
41        !b
42    }
43
44    /// `read_symbol( cdf )`: `cdf` has N + 1 entries (the last is the adaptation counter).
45    #[inline]
46    pub fn read_symbol(&mut self, cdf: &mut [u16]) -> usize {
47        let n = cdf.len() - 1;
48        let symbol = self.decode(cdf, n);
49        if !self.disable_update {
50            update_cdf(cdf, n, symbol);
51        }
52        symbol
53    }
54
55    #[inline(always)]
56    fn decode(&mut self, cdf: &[u16], n: usize) -> usize {
57        let value = (self.win >> self.k) as u32;
58        let mut cur = self.range;
59        let mut symbol = 0usize;
60        let mut prev;
61        loop {
62            prev = cur;
63            let f = (1u32 << 15) - cdf[symbol] as u32;
64            cur = ((self.range >> 8) * (f >> EC_PROB_SHIFT)) >> (7 - EC_PROB_SHIFT);
65            cur += EC_MIN_PROB * (n - symbol - 1) as u32;
66            if value >= cur {
67                break;
68            }
69            symbol += 1;
70        }
71        self.range = prev - cur;
72        self.win -= (cur as u64) << self.k;
73        self.renormalize();
74        symbol
75    }
76
77    /// Renormalisation (8.2.6): SymbolValue takes `bits` more (inverted) bits, which are
78    /// already in the window.
79    #[inline(always)]
80    fn renormalize(&mut self) {
81        let bits = 15 - (31 - self.range.leading_zeros());
82        self.range <<= bits;
83        self.k -= bits;
84        if self.k < 16 {
85            // SymbolValue < 2^16, so 16 + k <= 64 bits stay in the window.
86            while self.k <= 40 {
87                self.win = (self.win << 8) | self.inverted_byte() as u64;
88                self.k += 8;
89            }
90        }
91    }
92
93    /// `read_bool()`
94    #[inline]
95    pub fn read_bool(&mut self) -> u32 {
96        let cdf = [1u16 << 14, 1 << 15, 0];
97        self.decode(&cdf, 2) as u32
98    }
99
100    /// `read_literal( n )` (L(n))
101    pub fn read_literal(&mut self, n: u32) -> u32 {
102        let mut x = 0;
103        for _ in 0..n {
104            x = (x << 1) | self.read_bool();
105        }
106        x
107    }
108
109    /// NS(n)
110    pub fn read_ns(&mut self, n: u32) -> u32 {
111        if n <= 1 {
112            return 0;
113        }
114        let w = 32 - n.leading_zeros();
115        let m = (1u32 << w) - n;
116        let v = self.read_literal(w - 1);
117        if v < m {
118            return v;
119        }
120        let extra = self.read_literal(1);
121        (v << 1) - m + extra
122    }
123}
124
125/// CDF adaptation (8.2.6).
126#[inline(always)]
127pub(crate) fn update_cdf(cdf: &mut [u16], n: usize, symbol: usize) {
128    let count = cdf[n];
129    let rate = 3 + (count > 15) as u32 + (count > 31) as u32 + (31 - (n as u32).leading_zeros()).min(2);
130    // Entries below `symbol` move towards 0, the others towards 1 << 15 (the specification's
131    // comparison with tmp, resolved per entry: c >= 0 and c <= 1 << 15), without branching on
132    // the decoded symbol.
133    for (i, e) in cdf[..n - 1].iter_mut().enumerate() {
134        let c = *e as u32;
135        *e = if i < symbol { c - (c >> rate) } else { c + (((1 << 15) - c) >> rate) } as u16;
136    }
137    cdf[n] += (count < 32) as u16;
138}
