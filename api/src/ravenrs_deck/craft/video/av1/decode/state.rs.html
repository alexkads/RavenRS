1//! Per-frame decoding state shared by the tiles and the post-filters.
2//!
3//! [`FrameShared`] holds what tiles only read (headers, references, projected motion field);
4//! [`TileState`] what a tile writes (reconstruction, mode info, per-block filter parameters,
5//! entropy contexts). With one tile the frame's own buffers move into the tile state and back;
6//! with several, each tile decodes into private buffers covering its area (tiles never read
7//! outside themselves), which are then copied into the frame.
8
9use crate::craft::video::av1::decode::decoder::RefFrame;
10use crate::craft::video::av1::decode::frame::{FrameBuf, MiInfo};
11use crate::craft::video::av1::decode::header::{FrameHeader, RefInfo, SequenceHeader};
12use crate::craft::video::av1::decode::spec_tables::{DEFAULT_INTRA_FRAME_Y_MODE_CDF, NUM_REF_FRAMES, RESTORE_NONE, SUBSAMPLED_SIZE};
13use std::ops::{Deref, DerefMut};
14use std::sync::Arc;

16/// Read-only (during tile decoding) frame state.
17pub(crate) struct FrameShared {
18    pub seq: SequenceHeader,
19    pub fh: FrameHeader,
20    pub num_planes: usize,
21    pub ssx: usize,
22    pub ssy: usize,
23    pub bit_depth: u32,
24    pub lr_unit_rows: [usize; 3],
25    pub lr_unit_cols: [usize; 3],
26    pub cdef_cols: usize,
27    /// Reference slots (FrameStore and saved state) used by this frame.
28    pub refs: [Option<Arc<RefFrame>>; NUM_REF_FRAMES],
29    pub ref_info: [RefInfo; NUM_REF_FRAMES],
30    /// MotionFieldMvs[ ref ][ y8 * w8 + x8 ] (7.9); empty when unused.
31    pub motion_field: [Vec<[i32; 2]>; 8],
32    /// PrevSegmentIds (mi units).
33    pub prev_segment_ids: Vec<u8>,
34}

36impl FrameShared {
37    pub fn plane_residual_size(&self, subsize: usize, plane: usize) -> usize {
38        let sx = if plane > 0 { self.ssx } else { 0 };
39        let sy = if plane > 0 { self.ssy } else { 0 };
40        SUBSAMPLED_SIZE[subsize][sx][sy] as usize
41    }
42}

44/// A byte per 4x4 unit of a plane (LoopfilterTxSizes), possibly for a region only.
45#[derive(Clone, Default)]
46pub(crate) struct Grid8 {
47    pub data: Vec<u8>,
48    pub ox: usize,
49    pub oy: usize,
50    pub stride: usize,
51    pub rows: usize,
52}
53
54impl Grid8 {
55    fn new(ox: usize, oy: usize, stride: usize, rows: usize) -> Grid8 {
56        Grid8 { data: vec![0; stride * rows], ox, oy, stride, rows }
57    }
58    #[inline]
59    pub fn get(&self, row: usize, col: usize) -> u8 {
60        self.data[(row - self.oy) * self.stride + col - self.ox]
61    }
62    /// Stores outside the grid are dropped (blocks reaching past the allocated margin).
63    #[inline]
64    pub fn set(&mut self, row: usize, col: usize, v: u8) {
65        if col >= self.ox && col < self.ox + self.stride && row >= self.oy && row < self.oy + self.rows {
66            self.data[(row - self.oy) * self.stride + col - self.ox] = v;
67        }
68    }
69    fn copy_into(&self, dst: &mut Grid8) {
70        for y in 0..self.rows {
71            let d = (self.oy + y - dst.oy) * dst.stride + self.ox - dst.ox;
72            dst.data[d..d + self.stride].copy_from_slice(&self.data[y * self.stride..(y + 1) * self.stride]);
73        }
74    }
75}

77/// Loop restoration parameters per unit.
78#[derive(Clone, Default)]
79pub(crate) struct LrUnits {

81    pub lr_wiener: [Vec<[[i8; 3]; 2]>; 3],

83    pub lr_sgr_xqd: [Vec<[i8; 2]>; 3],
84}

86/// What tile decoding writes.
87pub(crate) struct TileState {
88    pub cur: FrameBuf,
89    pub mi: MiInfo,
90    /// cdef_idx per 64x64 block (-1: not coded).
91    pub cdef: Vec<i8>,
92    /// LoopfilterTxSizes[ plane ][ row ][ col ] in 4x4 units of each plane.
93    pub lf_tx_size: [Grid8; 3],
94    pub lr: LrUnits,
95    /// LR units this tile coded (per plane), for merging.
96    pub lr_written: [Vec<usize>; 3],
97    pub above_level: [Vec<u8>; 3],
98    pub above_dc: [Vec<u8>; 3],
99    pub above_seg_pred: Vec<u8>,
100    pub left_level: [Vec<u8>; 3],
101    pub left_dc: [Vec<u8>; 3],
102    pub left_seg_pred: Vec<u8>,
103    /// TileIntraFrameYModeCdf (reset per tile).
104    pub intra_frame_y_mode_cdf: [[[u16; 14]; 5]; 5],
105    cdef_cols: usize,
106}
107
108impl TileState {
109    fn with(cur: FrameBuf, mi: MiInfo, cdef: Vec<i8>, lf_tx_size: [Grid8; 3], lr: LrUnits, cdef_cols: usize) -> TileState {
110        let (mi_cols, mi_rows) = (mi.cols, mi.rows);
111        TileState {
112            cur,
113            mi,
114            cdef,
115            lf_tx_size,
116            lr,
117            lr_written: Default::default(),
118            above_level: std::array::from_fn(|_| vec![0; mi_cols + 64]),
119            above_dc: std::array::from_fn(|_| vec![0; mi_cols + 64]),
120            above_seg_pred: vec![0; mi_cols + 64],
121            left_level: std::array::from_fn(|_| vec![0; mi_rows + 64]),
122            left_dc: std::array::from_fn(|_| vec![0; mi_rows + 64]),
123            left_seg_pred: vec![0; mi_rows + 64],
124            intra_frame_y_mode_cdf: DEFAULT_INTRA_FRAME_Y_MODE_CDF,
125            cdef_cols,
126        }
127    }
128
129    #[inline]
130    pub fn cdef_idx(&self, r: usize, c: usize) -> i8 {
131        self.cdef[(r >> 4) * self.cdef_cols + (c >> 4)]
132    }
133
134    #[inline]
135    pub fn set_cdef_idx(&mut self, r: usize, c: usize, v: i8) {
136        let i = (r >> 4) * self.cdef_cols + (c >> 4);
137        if i < self.cdef.len() {
138            self.cdef[i] = v;
139        }
140    }
141
142    #[inline]
143    pub fn set_lf_tx_size(&mut self, plane: usize, row: usize, col: usize, v: u8) {
144        self.lf_tx_size[plane].set(row, col, v);
145    }
146}

170fn count_units_in_frame(unit_size: usize, frame_size: usize) -> usize {
171    ((frame_size + (unit_size >> 1)) / unit_size).max(1)
172}
173
174impl FrameState {
175    pub fn new(seq: &SequenceHeader, fh: &FrameHeader) -> FrameState {
176        let c = &seq.color;
177        let (ssx, ssy) = (c.subsampling_x as usize, c.subsampling_y as usize);
178        let mi_cols = fh.mi_cols as usize;
179        let mi_rows = fh.mi_rows as usize;
180        let cur = FrameBuf::new(fh.frame_width as usize, fh.frame_height as usize, c.num_planes, ssx, ssy, c.bit_depth);
181        let cdef_cols = mi_cols.div_ceil(16) + 2;
182        let cdef_rows = mi_rows.div_ceil(16) + 2;
183        let mut lr_unit_rows = [0; 3];
184        let mut lr_unit_cols = [0; 3];
185        let mut lr = LrUnits::default();
186        for plane in 0..c.num_planes {
187            if fh.lr.frame_restoration_type[plane] == RESTORE_NONE as u8 {
188                continue;
189            }
190            let sub_x = if plane == 0 { 0 } else { ssx };
191            let sub_y = if plane == 0 { 0 } else { ssy };
192            let unit = fh.lr.loop_restoration_size[plane] as usize;
193            lr_unit_rows[plane] = count_units_in_frame(unit, (fh.frame_height as usize + sub_y) >> sub_y);
194            lr_unit_cols[plane] = count_units_in_frame(unit, (fh.upscaled_width as usize + sub_x) >> sub_x);
195            let n = lr_unit_rows[plane] * lr_unit_cols[plane];
196            lr.lr_type[plane] = vec![RESTORE_NONE as u8; n];
197            lr.lr_wiener[plane] = vec![[[0; 3]; 2]; n];
198            lr.lr_sgr_set[plane] = vec![0; n];
199            lr.lr_sgr_xqd[plane] = vec![[0; 2]; n];
200        }
201        FrameState {
202            sh: FrameShared {
203                seq: seq.clone(),
204                fh: fh.clone(),
205                num_planes: c.num_planes,
206                ssx,
207                ssy,
208                bit_depth: c.bit_depth as u32,
209                lr_unit_rows,
210                lr_unit_cols,
211                cdef_cols,
212                refs: Default::default(),
213                ref_info: Default::default(),
214                motion_field: Default::default(),
215                prev_segment_ids: vec![0; mi_cols * mi_rows],
216            },
217            cur,
218            mi: MiInfo::new(mi_cols, mi_rows, fh.allow_screen_content_tools),
219            cdef: vec![-1; cdef_cols * cdef_rows],
220            lf_tx_size: std::array::from_fn(|_| Grid8::new(0, 0, mi_cols + 32, mi_rows + 32)),
221            lr,
222        }
223    }
224
225    #[inline]
226    pub fn cdef_idx(&self, r: usize, c: usize) -> i8 {
227        self.cdef[(r >> 4) * self.cdef_cols + (c >> 4)]
228    }
229
230    #[inline]
231    pub fn lf_tx_size(&self, plane: usize, row: usize, col: usize) -> usize {
232        self.lf_tx_size[plane].get(row, col) as usize
233    }

235    /// Move the frame's buffers into a tile state covering the whole frame.
236    pub fn take_tile_state(&mut self) -> TileState {
237        TileState::with(
238            std::mem::take(&mut self.cur),
239            std::mem::take(&mut self.mi),
240            std::mem::take(&mut self.cdef),
241            std::mem::take(&mut self.lf_tx_size),
242            std::mem::take(&mut self.lr),
243            self.cdef_cols,
244        )
245    }
246
247    /// Return the buffers taken by [`FrameState::take_tile_state`].
248    pub fn put_tile_state(&mut self, t: TileState) {
249        self.cur = t.cur;
250        self.mi = t.mi;
251        self.cdef = t.cdef;
252        self.lf_tx_size = t.lf_tx_size;
253        self.lr = t.lr;
254    }
255
256    /// Bytes of CDEF and loop-restoration metadata each parallel tile state copies whole.
257    pub fn tile_metadata_bytes(&self) -> usize {
258        let lr = &self.lr;
259        // 1 (type) + 6 (Wiener) + 1 (SGR set) + 2 (SGR weights) bytes per unit.
260        self.cdef.len() + lr.lr_type.iter().map(|v| v.len() * 10).sum::<usize>()
261    }
262
263    /// Private buffers for the tile covering mode info rows [r0, r1) x columns [c0, c1); tiles
264    /// on the right / bottom frame edge also cover the allocation margins.
265    pub fn tile_state(&self, r0: usize, r1: usize, c0: usize, c1: usize) -> TileState {
266        let (mi_cols, mi_rows) = (self.mi.cols, self.mi.rows);
267        let x1 = if c1 >= mi_cols { usize::MAX } else { c1 * 4 };
268        let y1 = if r1 >= mi_rows { usize::MAX } else { r1 * 4 };
269        let cur = self.cur.region_like(c0 * 4, r0 * 4, x1, y1);
270        let mi = MiInfo::region(mi_cols, mi_rows, c0, r0, c1 - c0, r1 - r0, !self.mi.palette_size[0].is_empty());
271        let lf = std::array::from_fn(|p| {
272            let (sx, sy) = if p == 0 { (0, 0) } else { (self.ssx, self.ssy) };
273            let full = &self.lf_tx_size[p];
274            let (gx0, gy0) = (c0 >> sx, r0 >> sy);
275            let gx1 = if c1 >= mi_cols { full.stride } else { c1 >> sx };
276            let gy1 = if r1 >= mi_rows { full.rows } else { r1 >> sy };
277            Grid8::new(gx0, gy0, gx1 - gx0, gy1 - gy0)
278        });
279        TileState::with(cur, mi, vec![-1; self.cdef.len()], lf, self.lr.clone(), self.cdef_cols)
280    }
281
282    /// Copy a tile's private buffers into the frame.
283    pub fn merge_tile_state(&mut self, t: &TileState) {
284        for p in 0..self.num_planes {
285            t.cur.planes[p].copy_into(&mut self.cur.planes[p]);
286            t.lf_tx_size[p].copy_into(&mut self.lf_tx_size[p]);
287            for &u in &t.lr_written[p] {
288                self.lr.lr_type[p][u] = t.lr.lr_type[p][u];
289                self.lr.lr_wiener[p][u] = t.lr.lr_wiener[p][u];
290                self.lr.lr_sgr_set[p][u] = t.lr.lr_sgr_set[p][u];
291                self.lr.lr_sgr_xqd[p][u] = t.lr.lr_sgr_xqd[p][u];
292            }
293        }
294        t.mi.copy_into(&mut self.mi);
295        for (d, &s) in self.cdef.iter_mut().zip(&t.cdef) {
296            if s != -1 {
297                *d = s;
298            }
299        }
300    }
