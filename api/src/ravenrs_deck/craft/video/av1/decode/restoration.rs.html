13/// Upscale a frame horizontally from FrameWidth to UpscaledWidth (7.16).
14pub(crate) fn upscale(fs: &FrameState, input: &FrameBuf) -> FrameBuf {
15    let fh = &fs.fh;
16    let up_w = fh.upscaled_width as usize;
17    let mut out = FrameBuf::new(up_w, fh.frame_height as usize, fs.num_planes, fs.ssx, fs.ssy, fs.bit_depth as u8);
18    let max = (1i32 << fs.bit_depth) - 1;
19    for plane in 0..fs.num_planes {
20        let (sub_x, sub_y) = if plane > 0 { (fs.ssx, fs.ssy) } else { (0, 0) };
21        let down_w = ((fh.frame_width as usize) + sub_x) >> sub_x;
22        let up_pw = (up_w + sub_x) >> sub_x;
23        let plane_h = ((fh.frame_height as usize) + sub_y) >> sub_y;
24        let sb = SUPERRES_SCALE_BITS as i64;
25        let step_x = (((down_w as i64) << sb) + (up_pw as i64 / 2)) / up_pw as i64;
26        let err = up_pw as i64 * step_x - ((down_w as i64) << sb);
27        let mut initial = (-(((up_pw - down_w) as i64) << (sb - 1)) + up_pw as i64 / 2) / up_pw as i64 + (1 << (SUPERRES_EXTRA_BITS - 1)) - err / 2;
28        initial &= SUPERRES_SCALE_MASK as i64;
29        let mi_w = fh.mi_cols as usize >> sub_x;
30        let max_x = (mi_w * 4 - 1) as i64;
31        let src = &input.planes[plane];
32        let dst = &mut out.planes[plane];
33        for y in 0..plane_h {
34            let row = src.row(y);
35            for x in 0..up_pw {
36                let src_x = -(1i64 << sb) + initial + x as i64 * step_x;
37                let src_px = src_x >> sb;
38                let subpel = ((src_x & SUPERRES_SCALE_MASK as i64) >> SUPERRES_EXTRA_BITS) as usize;
39                let mut sum = 0i32;
40                for k in 0..SUPERRES_FILTER_TAPS {
41                    let sx = (src_px + k as i64 - SUPERRES_FILTER_OFFSET as i64).clamp(0, max_x) as usize;
42                    sum += row[sx] as i32 * UPSCALE_FILTER[subpel][k] as i32;
43                }
44                dst.set(x, y, ((sum + 64) >> FILTER_BITS).clamp(0, max) as u16);
45            }
46        }
47    }
48    out
49}

51/// One plane's loop restoration inputs: UpscaledCurrFrame and UpscaledCdefFrame, the plane
52/// size and the restoration units.
53struct LrPlane<'a> {
54    cur: &'a Plane,
55    cdef: &'a Plane,
56    width: usize,
57    height: usize,
58    sub_y: usize,
59    bit_depth: u32,
60    unit_size: usize,
61    unit_rows: usize,
62    unit_cols: usize,
63    lr_type: &'a [u8],
64    wiener: &'a [[[i8; 3]; 2]],
65    sgr_set: &'a [u8],
66    sgr_xqd: &'a [[i8; 2]],

69/// Loop restoration (7.17): returns LrFrame given UpscaledCurrFrame and UpscaledCdefFrame.
70/// Bands of 64 luma rows are restored in parallel: every unit reads only the two input frames
71/// and writes only its own samples.
72pub(crate) fn loop_restoration(fs: &FrameState, cur: &FrameBuf, cdef: &FrameBuf, pool: &Pool) -> FrameBuf {
73    let fh = &fs.fh;
74    let mut lr = cdef.clone();
75    let ranges = bands(pool, fh.frame_height as usize);
76    let mut planes = lr.row_bands(&ranges);
77    pool.map_mut(&mut planes, |b, band| {
78        let (start, end) = ranges[b];
79        let mut scratch = Scratch::default();
80        for (plane, out) in band.iter_mut().enumerate().take(fs.num_planes) {
81            if fh.lr.frame_restoration_type[plane] == RESTORE_NONE as u8 {
82                continue;
83            }
84            let (sub_x, sub_y) = if plane > 0 { (fs.ssx, fs.ssy) } else { (0, 0) };
85            let p = LrPlane {
86                cur: &cur.planes[plane],
87                cdef: &cdef.planes[plane],
88                width: (fh.upscaled_width as usize + sub_x) >> sub_x,
89                height: (fh.frame_height as usize + sub_y) >> sub_y,
90                sub_y,
91                bit_depth: fs.bit_depth,
92                unit_size: fh.lr.loop_restoration_size[plane] as usize,
93                unit_rows: fs.lr_unit_rows[plane],
94                unit_cols: fs.lr_unit_cols[plane],
95                lr_type: &fs.lr.lr_type[plane],
96                wiener: &fs.lr.lr_wiener[plane],
97                sgr_set: &fs.lr.lr_sgr_set[plane],
98                sgr_xqd: &fs.lr.lr_sgr_xqd[plane],
99            };
100            p.restore(out, start >> sub_y, end >> sub_y, &mut scratch);
101        }
102    });
103    drop(planes);
104    lr
105}

107/// Row buffers reused across stripes.
108#[derive(Default)]
109struct Scratch {
110    /// Source rows of a stripe segment, 3 rows above and below and 3 columns left and right.
111    src: Vec<i32>,
112    /// Wiener horizontal pass output, or the self-guided A array.
113    inter: Vec<i32>,
114    /// Self-guided B array.
115    b: Vec<i32>,
116    /// Self-guided filter outputs of both passes.
117    flt: [Vec<i32>; 2],
118    /// Column sums of samples and squared samples over a box's rows.
119    col: Vec<i32>,
120    col_sq: Vec<i32>,
121}
122
123/// Padding of the source rows on every side.
124const PAD: usize = 3;
125
126impl LrPlane<'_> {
127    /// The row of the get source sample process (7.17.6) for row `y` of the stripe
128    /// [`stripe_start`, `stripe_end`]: pre-CDEF rows (at most 2 beyond the stripe) outside it.
129    fn source_row(&self, y: isize, stripe_start: isize, stripe_end: isize) -> &[u16] {
130        let y = y.min(self.height as isize - 1).max(0);
131        if y < stripe_start {
132            self.cur.row((stripe_start - 2).max(y) as usize)
133        } else if y > stripe_end {
134            self.cur.row((stripe_end + 2).min(y) as usize)
135        } else {
136            self.cdef.row(y as usize)
137        }
138    }
139
140    /// Restore rows [y0, y1) of the plane into `out`, in segments of rows that share a stripe
141    /// and a unit row.
142    fn restore(&self, out: &mut RowBand, y0: usize, y1: usize, s: &mut Scratch) {
143        let y1 = y1.min(self.height);
144        let off = 8 >> self.sub_y;
145        let stripe_h = 64 >> self.sub_y;
146        let mut y = y0;
147        while y < y1 {
148            // Stripes start 8 luma rows above multiples of 64 luma rows.
149            let stripe_num = (y + off) / stripe_h;
150            let next_stripe = (stripe_num + 1) * stripe_h - off;
151            let stripe = ((stripe_num * stripe_h) as isize - off as isize, next_stripe as isize - 1);
152            let unit_row = ((y + off) / self.unit_size).min(self.unit_rows - 1);
153            let mut end = y1.min(next_stripe);
154            if unit_row + 1 < self.unit_rows {
155                end = end.min((unit_row + 1) * self.unit_size - off);
156            }
157            self.segment(out, (y, end), stripe, unit_row, s);
158            y = end;
159        }
160    }
161
162    fn segment(&self, out: &mut RowBand, (ya, yb): (usize, usize), stripe: (isize, isize), unit_row: usize, s: &mut Scratch) {
163        let w = self.width;
164        let stride = w + 2 * PAD;
165        let rows = yb - ya + 2 * PAD;
166        s.src.clear();
167        s.src.resize(rows * stride, 0);
168        for (k, dst) in s.src.chunks_exact_mut(stride).enumerate() {
169            let src = &self.source_row(ya as isize - PAD as isize + k as isize, stripe.0, stripe.1)[..w];
170            dst[..PAD].fill(src[0] as i32);
171            for (d, &v) in dst[PAD..PAD + w].iter_mut().zip(src) {
172                *d = v as i32;
173            }
174            dst[PAD + w..].fill(src[w - 1] as i32);
175        }
176        for uc in 0..self.unit_cols {
177            let xa = uc * self.unit_size;
178            let xb = if uc + 1 == self.unit_cols { w } else { ((uc + 1) * self.unit_size).min(w) };
179            if xa >= xb {
180                continue;
181            }
182            let ui = unit_row * self.unit_cols + uc;
183            match self.lr_type[ui] {
184                t if t == RESTORE_WIENER as u8 => self.wiener(out, (ya, yb), (xa, xb), ui, s),
185                t if t == RESTORE_SGRPROJ as u8 => self.self_guided(out, (ya, yb), (xa, xb), ui, s),
186                _ => {}
187            }
188        }
189    }

191    fn wiener(&self, out: &mut RowBand, (ya, yb): (usize, usize), (xa, xb): (usize, usize), ui: usize, s: &mut Scratch) {
192        let bd = self.bit_depth;
193        let round0: u32 = if bd == 12 { 5 } else { 3 };
194        let round1: u32 = if bd == 12 { 9 } else { 11 };
195        let coef = &self.wiener[ui];
196        let mk = |c: &[i8; 3]| -> [i32; 7] {
197            let mut f = [0i32; 7];
198            f[3] = 128;
199            for i in 0..3 {
200                f[i] = c[i] as i32;
201                f[6 - i] = c[i] as i32;
202                f[3] -= 2 * c[i] as i32;
203            }
204            f
205        };
206        let vfilter = mk(&coef[0]);
207        let hfilter = mk(&coef[1]);
208        let fb = FILTER_BITS as u32;
209        let offset = 1i32 << (bd + fb - round0 - 1);
210        let limit = (1i32 << (bd + 1 + fb - round0)) - 1;
211        let stride = self.width + 2 * PAD;
212        let (w, h) = (xb - xa, yb - ya);
213        s.inter.clear();
214        s.inter.resize((h + 6) * w, 0);
215        for (src, inter) in s.src.chunks_exact(stride).zip(s.inter.chunks_exact_mut(w)) {
216            // Output column x reads source columns x - 3 ..= x + 3, padded index x ..= x + 6.
217            let src = &src[xa..xa + w + 6];
218            for (c, v) in inter.iter_mut().enumerate() {
219                let t = &src[c..c + 7];
220                let mut sum = 0;
221                for k in 0..7 {
222                    sum += hfilter[k] * t[k];
223                }
224                *v = ((sum + (1 << (round0 - 1))) >> round0).clamp(-offset, limit - offset);
225            }
226        }
227        let max = (1i32 << bd) - 1;
228        for r in 0..h {
229            let drow = &mut out.row_mut(ya + r)[xa..xb];
230            let col = &s.inter[r * w..(r + 7) * w];
231            for (c, d) in drow.iter_mut().enumerate() {
232                let mut sum = 0;
233                for k in 0..7 {
234                    sum += vfilter[k] * col[k * w + c];
235                }
236                *d = ((sum + (1 << (round1 - 1))) >> round1).clamp(0, max) as u16;
237            }
238        }
239    }

241    fn self_guided(&self, out: &mut RowBand, (ya, yb): (usize, usize), (xa, xb): (usize, usize), ui: usize, s: &mut Scratch) {
242        let set = self.sgr_set[ui] as usize;
243        let (w, h) = (xb - xa, yb - ya);
244        for pass in 0..2 {
245            if SGR_PARAMS[set][pass * 2] != 0 {
246                self.box_filter((ya, yb), (xa, xb), set, pass, s);
247            }
248        }
249        let xqd = self.sgr_xqd[ui];
250        let w0 = xqd[0] as i32;
251        let w1 = xqd[1] as i32;
252        let w2 = (1 << SGRPROJ_PRJ_BITS) - w0 - w1;
253        let r0 = SGR_PARAMS[set][0];
254        let r1 = SGR_PARAMS[set][2];
255        let max = (1i32 << self.bit_depth) - 1;
256        let sh = (SGRPROJ_RST_BITS + SGRPROJ_PRJ_BITS) as u32;
257        for i in 0..h {
258            let crow = &self.cdef.row(ya + i)[xa..xb];
259            let drow = &mut out.row_mut(ya + i)[xa..xb];
260            for j in 0..w {
261                let u = (crow[j] as i32) << SGRPROJ_RST_BITS;
262                let mut v = w1 * u;
263                v += if r0 != 0 { w0 * s.flt[0][i * w + j] } else { w0 * u };
264                v += if r1 != 0 { w2 * s.flt[1][i * w + j] } else { w2 * u };
265                let s = (v + (1 << (sh - 1))) >> sh;
266                drow[j] = s.clamp(0, max) as u16;
267            }
268        }
269    }

271    /// Box filter process (7.17.3) for one pass over rows [ya, yb) and columns [xa, xb), into
272    /// `s.flt[pass]`. A and B are computed for the rows and columns around the block once,
273    /// from running box sums over the source rows.
274    fn box_filter(&self, (ya, yb): (usize, usize), (xa, xb): (usize, usize), set: usize, pass: usize, s: &mut Scratch) {
275        let bd = self.bit_depth;
276        let r = SGR_PARAMS[set][pass * 2] as usize;
277        let eps = SGR_PARAMS[set][pass * 2 + 1] as i64;
278        let n = ((2 * r + 1) * (2 * r + 1)) as i64;
279        let n2e = n * n * eps;
280        let sc = ((1i64 << SGRPROJ_MTABLE_BITS) + n2e / 2) / n2e;
281        let one_over_n = ((1i64 << SGRPROJ_RECIP_BITS) + n / 2) / n;
282        let stride = self.width + 2 * PAD;
283        let (w, h) = (xb - xa, yb - ya);
284        // A and B at rows ya - 1 ..= yb and columns xa - 1 ..= xb.
285        let (aw, ah) = (w + 2, h + 2);
286        s.inter.clear();
287        s.inter.resize(aw * ah, 0);
288        s.b.clear();
289        s.b.resize(aw * ah, 0);
290        let taps = 2 * r + 1;
291        s.col.clear();
292        s.col.resize(aw + 2 * r, 0);
293        s.col_sq.clear();
294        s.col_sq.resize(aw + 2 * r, 0);
295        // Padded column of the first source sample of A column 0 (frame column xa - 1 - r).
296        let p0 = xa + PAD - 1 - r;
297        for i in 0..ah {
298            // Pass 0 only reads A and B on odd rows.
299            if pass == 0 && ((ya + i + 1) & 1) == 0 {
300                continue;
301            }
302            // Source row of frame row ya - 1 + i - r.
303            let k0 = i + PAD - 1 - r;
304            s.col.fill(0);
305            s.col_sq.fill(0);
306            for k in k0..k0 + taps {
307                let row = &s.src[k * stride + p0..k * stride + p0 + aw + 2 * r];
308                for ((c, q), &v) in s.col.iter_mut().zip(s.col_sq.iter_mut()).zip(row) {
309                    *c += v;
310                    *q += v * v;
311                }
312            }
313            let (mut b, mut a) = (0i32, 0i32);
314            for t in 0..taps - 1 {
315                b += s.col[t];
316                a += s.col_sq[t];
317            }
318            for j in 0..aw {
319                b += s.col[j + taps - 1];
320                a += s.col_sq[j + taps - 1];
321                let (a64, b64) = (a as i64, b as i64);
322                let aa = round2_64(a64, 2 * (bd - 8));
323                let d = round2_64(b64, bd - 8);
324                let p = (aa * n - d * d).max(0);
325                let z = round2_64(p * sc, SGRPROJ_MTABLE_BITS as u32);
326                let a2 = if z >= 255 {
327                    256
328                } else if z == 0 {
329                    1
330                } else {
331                    ((z << SGRPROJ_SGR_BITS) + z / 2) / (z + 1)
332                };
333                let b2 = ((1i64 << SGRPROJ_SGR_BITS) - a2) * b64 * one_over_n;
334                s.inter[i * aw + j] = a2 as i32;
335                s.b[i * aw + j] = round2_64(b2, SGRPROJ_RECIP_BITS as u32) as i32;
336                b -= s.col[j];
337                a -= s.col_sq[j];
338            }
339        }
340        let flt = &mut s.flt[pass];
341        flt.clear();
342        flt.resize(w * h, 0);
343        let (aa, bb) = (&s.inter, &s.b);
344        for i in 0..h {
345            let crow = &self.cdef.row(ya + i)[xa..xb];
346            let (up, mid, down) = (i * aw, (i + 1) * aw, (i + 2) * aw);
347            let odd = ((ya + i) & 1) == 1;
348            let shift = if pass == 0 && odd { 4 } else { 5 };
349            for j in 0..w {
350                let (a, b) = if pass == 0 {
351                    if odd {
352                        (6 * aa[mid + j + 1] + 5 * (aa[mid + j] + aa[mid + j + 2]), 6 * bb[mid + j + 1] + 5 * (bb[mid + j] + bb[mid + j + 2]))
353                    } else {
354                        (
355                            6 * (aa[up + j + 1] + aa[down + j + 1]) + 5 * (aa[up + j] + aa[up + j + 2] + aa[down + j] + aa[down + j + 2]),
356                            6 * (bb[up + j + 1] + bb[down + j + 1]) + 5 * (bb[up + j] + bb[up + j + 2] + bb[down + j] + bb[down + j + 2]),
357                        )
358                    }
359                } else {
360                    (
361                        4 * (aa[up + j + 1] + aa[mid + j] + aa[mid + j + 1] + aa[mid + j + 2] + aa[down + j + 1])
362                            + 3 * (aa[up + j] + aa[up + j + 2] + aa[down + j] + aa[down + j + 2]),
363                        4 * (bb[up + j + 1] + bb[mid + j] + bb[mid + j + 1] + bb[mid + j + 2] + bb[down + j + 1])
364                            + 3 * (bb[up + j] + bb[up + j + 2] + bb[down + j] + bb[down + j + 2]),
365                    )
366                };
367                let v = a as i64 * crow[j] as i64 + b as i64;
368                flt[i * w + j] = round2_64(v, (SGRPROJ_SGR_BITS + shift - SGRPROJ_RST_BITS) as u32) as i32;
369            }
370        }
371    }
372}

374#[inline(always)]
375fn round2_64(x: i64, n: u32) -> i64 {
376    if n == 0 { x } else { (x + (1i64 << (n - 1))) >> n }
377}

409    /// The former loop over 4x4 luma blocks for a frame of `up_w` x `fr_h` luma samples.
410    fn reference(p: &LrPlane, sub_x: i32, up_w: i32, fr_h: i32, out: &mut Plane) {
411        let sub_y = p.sub_y as i32;
412        let bd = p.bit_depth;
413        let unit_size = p.unit_size as i32;
414        let (unit_rows, unit_cols) = (p.unit_rows as i32, p.unit_cols as i32);
415        let plane_end_x = ((up_w + sub_x) >> sub_x) - 1;
416        let plane_end_y = ((fr_h + sub_y) >> sub_y) - 1;
417        let mut row = 0;
418        while row * 4 < fr_h {
419            let luma_y = row * 4;
420            let stripe_num = (luma_y + 8) / 64;
421            let stripe_start_y = (-8 + stripe_num * 64) >> sub_y;
422            let stripe_end_y = stripe_start_y + (64 >> sub_y) - 1;
423            let ctx = LrCtx { cur: p.cur, cdef: p.cdef, plane_end_x, plane_end_y, stripe_start_y, stripe_end_y };
424            let unit_row = (unit_rows - 1).min(((row * 4 + 8) >> sub_y) / unit_size);
425            let mut col = 0;
426            while col * 4 < up_w {
427                let unit_col = (unit_cols - 1).min(((col * 4) >> sub_x) / unit_size);
428                let x = (col * 4) >> sub_x;
429                let y = (row * 4) >> sub_y;
430                let w = (4 >> sub_x).min(plane_end_x - x + 1);
431                let h = (4 >> sub_y).min(plane_end_y - y + 1);
432                if w > 0 && h > 0 {
433                    let ui = (unit_row * unit_cols + unit_col) as usize;
434                    match p.lr_type[ui] {
435                        t if t == RESTORE_WIENER as u8 => wiener(p, &ctx, out, ui, x, y, w, h, bd),
436                        t if t == RESTORE_SGRPROJ as u8 => self_guided(p, &ctx, out, ui, x, y, w, h, bd),
437                        _ => {}
438                    }
439                }
440                col += 1;
441            }
442            row += 1;
443        }
444    }

446    #[allow(clippy::too_many_arguments)]
447    fn wiener(p: &LrPlane, ctx: &LrCtx, out: &mut Plane, ui: usize, x: i32, y: i32, w: i32, h: i32, bd: u32) {
448        let round0: u32 = if bd == 12 { 5 } else { 3 };
449        let round1: u32 = if bd == 12 { 9 } else { 11 };
450        let coef = &p.wiener[ui];
451        let mk = |c: &[i8; 3]| -> [i32; 7] {
452            let mut f = [0i32; 7];
453            f[3] = 128;
454            for i in 0..3 {
455                f[i] = c[i] as i32;
456                f[6 - i] = c[i] as i32;
457                f[3] -= 2 * c[i] as i32;
458            }
459            f
460        };
461        let vfilter = mk(&coef[0]);
462        let hfilter = mk(&coef[1]);
463        let fb = FILTER_BITS as u32;
464        let offset = 1i32 << (bd + fb - round0 - 1);
465        let limit = (1i32 << (bd + 1 + fb - round0)) - 1;
466        let mut inter = [[0i32; 4]; 10];
467        for r in 0..(h + 6) as usize {
468            for c in 0..w as usize {
469                let mut s = 0;
470                for t in 0..7 {
471                    s += hfilter[t] * ctx.sample(x + c as i32 + t as i32 - 3, y + r as i32 - 3);
472                }
473                let v = (s + (1 << (round0 - 1))) >> round0;
474                inter[r][c] = v.clamp(-offset, limit - offset);
475            }
476        }
477        let max = (1i32 << bd) - 1;
478        for r in 0..h as usize {
479            for c in 0..w as usize {
480                let mut s = 0;
481                for t in 0..7 {
482                    s += vfilter[t] * inter[r + t][c];
483                }
484                let v = (s + (1 << (round1 - 1))) >> round1;
485                out.set(x as usize + c, y as usize + r, v.clamp(0, max) as u16);
486            }
487        }
488    }

490    #[allow(clippy::too_many_arguments)]
491    fn self_guided(p: &LrPlane, ctx: &LrCtx, out: &mut Plane, ui: usize, x: i32, y: i32, w: i32, h: i32, bd: u32) {
492        let set = p.sgr_set[ui] as usize;
493        let mut flt0 = [[0i32; 4]; 4];
494        let mut flt1 = [[0i32; 4]; 4];
495        box_filter(ctx, x, y, w, h, set, 0, bd, &mut flt0);
496        box_filter(ctx, x, y, w, h, set, 1, bd, &mut flt1);
497        let xqd = p.sgr_xqd[ui];
498        let w0 = xqd[0] as i32;
499        let w1 = xqd[1] as i32;
500        let w2 = (1 << SGRPROJ_PRJ_BITS) - w0 - w1;
501        let r0 = SGR_PARAMS[set][0];
502        let r1 = SGR_PARAMS[set][2];
503        let max = (1i32 << bd) - 1;
504        let sh = (SGRPROJ_RST_BITS + SGRPROJ_PRJ_BITS) as u32;
505        for i in 0..h as usize {
506            for j in 0..w as usize {
507                let u = (ctx.cdef.at(x as usize + j, y as usize + i) as i32) << SGRPROJ_RST_BITS;
508                let mut v = w1 * u;
509                v += if r0 != 0 { w0 * flt0[i][j] } else { w0 * u };
510                v += if r1 != 0 { w2 * flt1[i][j] } else { w2 * u };
511                let s = (v + (1 << (sh - 1))) >> sh;
512                out.set(x as usize + j, y as usize + i, s.clamp(0, max) as u16);
513            }
514        }
515    }

517    #[allow(clippy::too_many_arguments)]
518    fn box_filter(ctx: &LrCtx, x: i32, y: i32, w: i32, h: i32, set: usize, pass: usize, bd: u32, f: &mut [[i32; 4]; 4]) {
519        let r = SGR_PARAMS[set][pass * 2] as i32;
520        if r == 0 {
521            return;
522        }
523        let eps = SGR_PARAMS[set][pass * 2 + 1] as i64;
524        let n = ((2 * r + 1) * (2 * r + 1)) as i64;
525        let n2e = n * n * eps;
526        let s = ((1i64 << SGRPROJ_MTABLE_BITS) + n2e / 2) / n2e;
527        let one_over_n = ((1i64 << SGRPROJ_RECIP_BITS) + n / 2) / n;
528        let mut a_arr = [[0i64; 6]; 6];
529        let mut b_arr = [[0i64; 6]; 6];
530        for i in -1..h + 1 {
531            for j in -1..w + 1 {
532                let mut a = 0i64;
533                let mut b = 0i64;
534                for dy in -r..=r {
535                    for dx in -r..=r {
536                        let c = ctx.sample(x + j + dx, y + i + dy) as i64;
537                        a += c * c;
538                        b += c;
539                    }
540                }
541                let a = round2_64(a, 2 * (bd - 8));
542                let d = round2_64(b, bd - 8);
543                let p = (a * n - d * d).max(0);
544                let z = round2_64(p * s, SGRPROJ_MTABLE_BITS as u32);
545                let a2 = if z >= 255 {
546                    256
547                } else if z == 0 {
548                    1
549                } else {
550                    ((z << SGRPROJ_SGR_BITS) + z / 2) / (z + 1)
551                };
552                let b2 = ((1i64 << SGRPROJ_SGR_BITS) - a2) * b * one_over_n;
553                a_arr[(i + 1) as usize][(j + 1) as usize] = a2;
554                b_arr[(i + 1) as usize][(j + 1) as usize] = round2_64(b2, SGRPROJ_RECIP_BITS as u32);
555            }
556        }
557        for i in 0..h {
558            let shift = if pass == 0 && (i & 1) == 1 { 4 } else { 5 };
559            for j in 0..w {
560                let mut a = 0i64;
561                let mut b = 0i64;
562                for dy in -1..=1i32 {
563                    for dx in -1..=1i32 {
564                        let weight = if pass == 0 {
565                            if ((i + dy) & 1) != 0 { if dx == 0 { 6 } else { 5 } } else { 0 }
566                        } else if dx == 0 || dy == 0 {
567                            4
568                        } else {
569                            3
570                        };
571                        a += weight * a_arr[(i + dy + 1) as usize][(j + dx + 1) as usize];
572                        b += weight * b_arr[(i + dy + 1) as usize][(j + dx + 1) as usize];
573                    }
574                }
575                let v = a * ctx.cdef.at((x + j) as usize, (y + i) as usize) as i64 + b;
576                f[i as usize][j as usize] = round2_64(v, (SGRPROJ_SGR_BITS + shift - SGRPROJ_RST_BITS) as u32) as i32;
577            }
578        }
579    }

583    impl Rng {
584        fn next(&mut self) -> u32 {
585            self.0 ^= self.0 << 13;
586            self.0 ^= self.0 >> 7;
587            self.0 ^= self.0 << 17;
588            (self.0 >> 32) as u32
589        }
590        fn below(&mut self, n: u32) -> u32 {
591            self.next() % n
592        }
593    }

595    /// Random samples: uniform noise, a smooth ramp with noise, or saturated values.
596    fn fill(rng: &mut Rng, p: &mut Plane, bd: u32) {
597        let max = (1u32 << bd) - 1;
598        let kind = rng.below(4);
599        let stride = p.stride;
600        for (i, v) in p.data.iter_mut().enumerate() {
601            let (x, y) = ((i % stride) as u32, (i / stride) as u32);
602            *v = match kind {
603                0 => rng.below(max + 1),
604                1 => ((x * 7 + y * 3) % (max + 1) + rng.below(16)).min(max),
605                2 => {
606                    if (x + y) % 2 == 0 {
607                        0
608                    } else {
609                        max
610                    }
611                }
612                _ => max - rng.below(4).min(max),
613            } as u16;
614        }
615    }

617    fn count_units(unit_size: usize, size: usize) -> usize {
618        ((size + (unit_size >> 1)) / unit_size).max(1)
619    }

621    #[cfg(any())]
622    fn stripes_match_per_block_reference() {
623        let mut rng = Rng(0x9e37_79b9_7f4a_7c15);
624        let subsampling = [(0, 0), (1, 1), (1, 0)];
625        let mut cases = 0;
626        for bd in [8u32, 10, 12] {
627            for luma_unit in [64usize, 128, 256] {
628                for (sub_x, sub_y) in subsampling {
629                    for _ in 0..6 {
630                        let up_w = 8 + rng.below(300) as i32;
631                        let fr_h = 8 + rng.below(300) as i32;
632                        // Chroma units of 4:2:0 may be half the luma size (lr_uv_shift).
633                        let unit_size = if sub_x == 1 && sub_y == 1 && rng.below(2) == 1 { luma_unit >> 1 } else { luma_unit };
634                        let (w, h) = (((up_w + sub_x) >> sub_x) as usize, ((fr_h + sub_y) >> sub_y) as usize);
635                        let (unit_rows, unit_cols) = (count_units(unit_size, h), count_units(unit_size, w));
636                        let n = unit_rows * unit_cols;
637                        let lr_type: Vec<u8> = (0..n).map(|_| rng.below(3) as u8).collect();
638                        let wiener: Vec<[[i8; 3]; 2]> = (0..n)
639                            .map(|_| {
640                                let mut c = || [rng.below(24) as i8 - 5, rng.below(48) as i8 - 23, rng.below(64) as i8 - 17];
641                                [c(), c()]
642                            })
643                            .collect();
644                        let sgr_set: Vec<u8> = (0..n).map(|_| rng.below(16) as u8).collect();
645                        let sgr_xqd: Vec<[i8; 2]> = (0..n).map(|_| [rng.below(128) as i8 - 96, rng.below(128) as i8 - 32]).collect();
646                        let (mut cur, mut cdef) = (Plane::new(w + 16, h + 16), Plane::new(w + 16, h + 16));
647                        fill(&mut rng, &mut cur, bd);
648                        fill(&mut rng, &mut cdef, bd);
649                        let p = LrPlane {
650                            cur: &cur,
651                            cdef: &cdef,
652                            width: w,
653                            height: h,
654                            sub_y: sub_y as usize,
655                            bit_depth: bd,
656                            unit_size,
657                            unit_rows,
658                            unit_cols,
659                            lr_type: &lr_type,
660                            wiener: &wiener,
661                            sgr_set: &sgr_set,
662                            sgr_xqd: &sgr_xqd,
663                        };
664                        let mut want = cdef.clone();
665                        reference(&p, sub_x, up_w, fr_h, &mut want);
666                        // One band, and bands of 64 luma rows.
667                        for band_rows in [usize::MAX, 64] {
668                            let ranges: Vec<(usize, usize)> = if band_rows == usize::MAX {
669                                vec![(0, usize::MAX)]
670                            } else {
671                                (0..(fr_h as usize).div_ceil(64)).map(|b| (b * 64, (b + 1) * 64)).collect()
672                            };
673                            let chroma: Vec<(usize, usize)> = ranges.iter().map(|&(s, e)| (s >> sub_y, e >> sub_y)).collect();
674                            let mut got = cdef.clone();
675                            let mut scratch = Scratch::default();
676                            for (band, &(s, e)) in got.row_bands(&chroma).iter_mut().zip(&chroma) {
677                                p.restore(band, s, e, &mut scratch);
678                            }
679                            for y in 0..h {
680                                assert_eq!(
681                                    &got.row(y)[..w],
682                                    &want.row(y)[..w],
683                                    "row {y}: {bd}-bit, unit {unit_size}, subsampling ({sub_x}, {sub_y}), {up_w}x{fr_h}, bands {band_rows}"
684                                );
685                            }
686                        }
687                        cases += 1;
688                    }
689                }
690            }
691        }
692        assert_eq!(cases, 162);
693    }
