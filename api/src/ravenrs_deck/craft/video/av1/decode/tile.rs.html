192    /// decode_tile( )
193    pub fn decode_tile(&mut self) -> Result<()> {
194        // clear_above_context( )
195        let t = &mut *self.t;
196        for p in 0..3 {
197            t.above_level[p].fill(0);
198            t.above_dc[p].fill(0);
199        }
200        t.above_seg_pred.fill(0);
201        self.delta_lf = [0; 4];
202        for plane in 0..self.fs.num_planes {
203            for pass in 0..2 {
204                self.ref_sgr_xqd[plane][pass] = SGRPROJ_XQD_MID[pass] as i32;
205                self.ref_lr_wiener[plane][pass] = WIENER_TAPS_MID.map(i32::from);
206            }
207        }
208        let sb128 = self.seq().use_128x128_superblock;
209        let sb_size = if sb128 { BLOCK_128X128 } else { BLOCK_64X64 };
210        let sb4 = NUM_4X4_BLOCKS_WIDE[sb_size] as usize;
211        let mut r = self.mi_row_start;
212        while r < self.mi_row_end {
213            // clear_left_context( )
214            let t = &mut *self.t;
215            for p in 0..3 {
216                t.left_level[p].fill(0);
217                t.left_dc[p].fill(0);
218            }
219            t.left_seg_pred.fill(0);
220            let mut c = self.mi_col_start;
221            while c < self.mi_col_end {
222                self.read_deltas = self.fh().delta_q_present;
223                self.clear_cdef(r, c);
224                self.clear_block_decoded_flags(r, c, sb4);
225                self.read_lr(r, c, sb_size);
226                self.decode_partition(r, c, sb_size)?;
227                c += sb4;
228            }
229            r += sb4;
230        }
231        Ok(())
232    }
233
234    fn clear_cdef(&mut self, r: usize, c: usize) {
235        let t = &mut *self.t;
236        t.set_cdef_idx(r, c, -1);
237        if self.fs.seq.use_128x128_superblock {
238            t.set_cdef_idx(r, c + 16, -1);
239            t.set_cdef_idx(r + 16, c, -1);
240            t.set_cdef_idx(r + 16, c + 16, -1);
241        }
242    }
243
244    fn clear_block_decoded_flags(&mut self, r: usize, c: usize, sb4: usize) {
245        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
246        for plane in 0..self.fs.num_planes {
247            let sub_x = if plane > 0 { ssx } else { 0 };
248            let sub_y = if plane > 0 { ssy } else { 0 };
249            let sb_w4 = (self.mi_col_end as isize - c as isize) >> sub_x;
250            let sb_h4 = (self.mi_row_end as isize - r as isize) >> sub_y;
251            let bd = &mut self.block_decoded[plane];
252            for y in -1..=((sb4 >> sub_y) as isize) {
253                for x in -1..=((sb4 >> sub_x) as isize) {
254                    let v = if y < 0 && x < sb_w4 { true } else { x < 0 && y < sb_h4 };
255                    bd[(y + 1) as usize][(x + 1) as usize] = v;
256                }
257            }
258            bd[(sb4 >> sub_y) + 1][0] = false;
259        }
260    }

262    #[inline]
263    pub(crate) fn block_decoded_at(&self, plane: usize, y: isize, x: isize) -> bool {
264        let (yy, xx) = (y + 1, x + 1);
265        if yy < 0 || xx < 0 || yy > SB_MAX as isize + 1 || xx > SB_MAX as isize + 1 {
266            return false;
267        }
268        self.block_decoded[plane][yy as usize][xx as usize]
269    }

271    fn decode_partition(&mut self, r: usize, c: usize, bsize: usize) -> Result<()> {
272        let (mi_rows, mi_cols) = (self.fs.fh.mi_rows as usize, self.fs.fh.mi_cols as usize);
273        if r >= mi_rows || c >= mi_cols {
274            return Ok(());
275        }
276        let avail_u = self.is_inside(r as isize - 1, c as isize);
277        let avail_l = self.is_inside(r as isize, c as isize - 1);
278        let num4x4 = NUM_4X4_BLOCKS_WIDE[bsize] as usize;
279        let half = num4x4 >> 1;
280        let quarter = half >> 1;
281        let has_rows = (r + half) < mi_rows;
282        let has_cols = (c + half) < mi_cols;
283        let partition = if bsize < BLOCK_8X8 {
284            PARTITION_NONE
285        } else {
286            let bsl = MI_WIDTH_LOG2[bsize] as usize;
287            let mi = &self.t.mi;
288            let above = avail_u && (MI_WIDTH_LOG2[mi.mi_size[mi.idx(r - 1, c)] as usize] as usize) < bsl;
289            let left = avail_l && (MI_HEIGHT_LOG2[mi.mi_size[mi.idx(r, c - 1)] as usize] as usize) < bsl;
290            let ctx = left as usize * 2 + above as usize;
291            if has_rows && has_cols {
292                let cdf: &mut [u16] = match bsl {
293                    1 => &mut self.cdf.partition_w8[ctx],
294                    2 => &mut self.cdf.partition_w16[ctx],
295                    3 => &mut self.cdf.partition_w32[ctx],
296                    4 => &mut self.cdf.partition_w64[ctx],
297                    _ => &mut self.cdf.partition_w128[ctx],
298                };
299                self.sd.read_symbol(cdf)
300            } else if has_cols || has_rows {
301                let pcdf: &[u16] = match bsl {
302                    2 => &self.cdf.partition_w16[ctx],
303                    3 => &self.cdf.partition_w32[ctx],
304                    4 => &self.cdf.partition_w64[ctx],
305                    _ => &self.cdf.partition_w128[ctx],
306                };
307                let psum = split_psum(pcdf, has_cols, bsize);
308                let bit = self.read_split_bool(psum);
309                if has_cols {
310                    if bit { PARTITION_SPLIT } else { PARTITION_HORZ }
311                } else if bit {
312                    PARTITION_SPLIT
313                } else {
314                    PARTITION_VERT
315                }
316            } else {
317                PARTITION_SPLIT
318            }
319        };
320        let sub_size = PARTITION_SUBSIZE[partition][bsize] as usize;
321        let split_size = PARTITION_SUBSIZE[PARTITION_SPLIT][bsize] as usize;
322        match partition {
323            PARTITION_NONE => self.decode_block(r, c, sub_size)?,
324            PARTITION_HORZ => {
325                self.decode_block(r, c, sub_size)?;
326                if has_rows {
327                    self.decode_block(r + half, c, sub_size)?;
328                }
329            }
330            PARTITION_VERT => {
331                self.decode_block(r, c, sub_size)?;
332                if has_cols {
333                    self.decode_block(r, c + half, sub_size)?;
334                }
335            }
336            PARTITION_SPLIT => {
337                self.decode_partition(r, c, sub_size)?;
338                self.decode_partition(r, c + half, sub_size)?;
339                self.decode_partition(r + half, c, sub_size)?;
340                self.decode_partition(r + half, c + half, sub_size)?;
341            }
342            PARTITION_HORZ_A => {
343                self.decode_block(r, c, split_size)?;
344                self.decode_block(r, c + half, split_size)?;
345                self.decode_block(r + half, c, sub_size)?;
346            }
347            PARTITION_HORZ_B => {
348                self.decode_block(r, c, sub_size)?;
349                self.decode_block(r + half, c, split_size)?;
350                self.decode_block(r + half, c + half, split_size)?;
351            }
352            PARTITION_VERT_A => {
353                self.decode_block(r, c, split_size)?;
354                self.decode_block(r + half, c, split_size)?;
355                self.decode_block(r, c + half, sub_size)?;
356            }
357            PARTITION_VERT_B => {
358                self.decode_block(r, c, sub_size)?;
359                self.decode_block(r, c + half, split_size)?;
360                self.decode_block(r + half, c + half, split_size)?;
361            }
362            PARTITION_HORZ_4 => {
363                for k in 0..4 {
364                    let rr = r + quarter * k;
365                    if k < 3 || rr < mi_rows {
366                        self.decode_block(rr, c, sub_size)?;
367                    }
368                }
369            }
370            _ => {
371                for k in 0..4 {
372                    let cc = c + quarter * k;
373                    if k < 3 || cc < mi_cols {
374                        self.decode_block(r, cc, sub_size)?;
375                    }
376                }
377            }
378        }
379        Ok(())
380    }

382    /// split_or_horz / split_or_vert: a binary symbol whose probability is derived from the
383    /// partition CDF (8.3.2). Returns true for PARTITION_SPLIT.
384    fn read_split_bool(&mut self, psum: i32) -> bool {
385        let mut cdf = [((1i32 << 15) - psum) as u16, 1 << 15, 0];
386        // no adaptation: the array is constructed for this read only
387        let save = self.sd.disable_update;
388        self.sd.disable_update = true;
389        let s = self.sd.read_symbol(&mut cdf);
390        self.sd.disable_update = save;
391        s == 1
392    }

394    fn decode_block(&mut self, r: usize, c: usize, sub_size: usize) -> Result<()> {
395        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
396        let num_planes = self.fs.num_planes;
397        // Conformance requirement (4:2:2 has no chroma size for 4x16 and similar blocks); indexing the size tables with BLOCK_INVALID would panic.
398        if num_planes > 1 && self.fs.plane_residual_size(sub_size, 1) == BLOCK_INVALID {
399            return Err(crate::craft::video::av1::decode::Error::Invalid("block size has no chroma size at this subsampling"));
400        }
401        let mut b = Block { mi_row: r, mi_col: c, mi_size: sub_size, ..Default::default() };
402        b.bw4 = NUM_4X4_BLOCKS_WIDE[sub_size] as usize;
403        b.bh4 = NUM_4X4_BLOCKS_HIGH[sub_size] as usize;
404        b.has_chroma = if b.bh4 == 1 && ssy == 1 && (r & 1) == 0 || b.bw4 == 1 && ssx == 1 && (c & 1) == 0 { false } else { num_planes > 1 };
405        b.avail_u = self.is_inside(r as isize - 1, c as isize);
406        b.avail_l = self.is_inside(r as isize, c as isize - 1);
407        b.avail_u_chroma = b.avail_u;
408        b.avail_l_chroma = b.avail_l;
409        if b.has_chroma {
410            if ssy == 1 && b.bh4 == 1 {
411                b.avail_u_chroma = self.is_inside(r as isize - 2, c as isize);
412            }
413            if ssx == 1 && b.bw4 == 1 {
414                b.avail_l_chroma = self.is_inside(r as isize, c as isize - 2);
415            }
416        } else {
417            b.avail_u_chroma = false;
418            b.avail_l_chroma = false;
419        }
420        b.ref_frame = [INTRA_FRAME as i8, NONE];
421        self.b = b;
422        if self.fs.fh.frame_is_intra {
423            self.intra_frame_mode_info()?;
424        } else {
425            self.inter_frame_mode_info()?;
426        }
427        self.palette_tokens();
428        self.read_block_tx_size();
429        if self.b.skip {
430            self.reset_block_context();
431        }
432        let b = &self.b;
433        let is_compound = b.ref_frame[1] > INTRA_FRAME as i8;
434        {
435            let mi = &mut self.t.mi;
436            let rows = b.bh4.min(mi.rows.saturating_sub(r));
437            let cols = b.bw4.min(mi.cols.saturating_sub(c));
438            let set_uv = b.ref_frame[0] == INTRA_FRAME as i8 && b.has_chroma;
439            let mvs = [Mv::new(b.mv_i[0][0], b.mv_i[0][1]), Mv::new(b.mv_i[1][0], b.mv_i[1][1])];
440            for y in 0..rows {
441                let i = mi.idx(r + y, c);
442                let span = i..i + cols;
443                mi.y_mode[span.clone()].fill(b.y_mode as u8);
444                if set_uv {
445                    mi.uv_mode[span.clone()].fill(b.uv_mode as u8);
446                }
447                mi.ref_frame[span.clone()].fill(b.ref_frame);
448                mi.written[span.clone()].fill(true);
449                if b.is_inter {
450                    if !b.use_intrabc {
451                        mi.comp_group_idx[span.clone()].fill(b.comp_group_idx);
452                        mi.compound_idx[span.clone()].fill(b.compound_idx);
453                    }
454                    mi.interp_filter[span.clone()].fill(b.interp_filter);
455                    for m in mi.mv[span].iter_mut() {
456                        m[0] = mvs[0];
457                        if is_compound {
458                            m[1] = mvs[1];
459                        }
460                    }
461                }
462            }
463        }
464        self.compute_prediction()?;
465        self.residual()?;
466        let b = &self.b;
467        let delta_lf = self.delta_lf;
468        let mi = &mut self.t.mi;
469        let rows = b.bh4.min(mi.rows.saturating_sub(r));
470        let cols = b.bw4.min(mi.cols.saturating_sub(c));
471        let dlf = [delta_lf[0] as i8, delta_lf[1] as i8, delta_lf[2] as i8, delta_lf[3] as i8];
472        for y in 0..rows {
473            let i = mi.idx(r + y, c);
474            let span = i..i + cols;
475            mi.is_inter[span.clone()].fill(b.is_inter);
476            mi.skip_mode[span.clone()].fill(b.skip_mode);
477            mi.skip[span.clone()].fill(b.skip);
478            mi.tx_size[span.clone()].fill(b.tx_size as u8);
479            mi.mi_size[span.clone()].fill(b.mi_size as u8);
480            mi.segment_id[span.clone()].fill(b.segment_id as u8);
481            if !mi.palette_size[0].is_empty() {
482                mi.palette_size[0][span.clone()].fill(b.palette_size_y as u8);
483                mi.palette_size[1][span.clone()].fill(b.palette_size_uv as u8);
484                mi.palette_colors[0][span.clone()].fill(b.palette_colors_y);
485                mi.palette_colors[1][span.clone()].fill(b.palette_colors_u);
486            }
487            mi.delta_lf[span.clone()].fill(dlf);
488            mi.motion_mode[span].fill(b.motion_mode);
489        }
490        Ok(())
491    }

493    fn reset_block_context(&mut self) {
494        let b = &self.b;
495        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
496        for plane in 0..(1 + 2 * b.has_chroma as usize) {
497            let sub_x = if plane > 0 { ssx } else { 0 };
498            let sub_y = if plane > 0 { ssy } else { 0 };
499            for i in (b.mi_col >> sub_x)..((b.mi_col + b.bw4) >> sub_x) {
500                self.t.above_level[plane][i] = 0;
501                self.t.above_dc[plane][i] = 0;
502            }
503            for i in (b.mi_row >> sub_y)..((b.mi_row + b.bh4) >> sub_y) {
504                self.t.left_level[plane][i] = 0;
505                self.t.left_dc[plane][i] = 0;
506            }
507        }
508    }

510    fn intra_frame_mode_info(&mut self) -> Result<()> {
511        self.b.skip = false;
512        let pre_skip = self.fs.fh.seg.seg_id_pre_skip;
513        if pre_skip {
514            self.intra_segment_id();
515        }
516        self.b.skip_mode = false;
517        self.read_skip();
518        if !pre_skip {
519            self.intra_segment_id();
520        }
521        self.read_cdef();
522        self.read_delta_qindex();
523        self.read_delta_lf();
524        self.read_deltas = false;
525        self.b.ref_frame = [INTRA_FRAME as i8, NONE];
526        self.b.use_intrabc = if self.fs.fh.allow_intrabc { self.sd.read_symbol(&mut self.cdf.intrabc) == 1 } else { false };
527        if self.b.use_intrabc {
528            self.b.is_inter = true;
529            self.b.y_mode = DC_PRED;
530            self.b.uv_mode = DC_PRED;
531            self.b.motion_mode = SIMPLE as u8;
532            self.b.compound_type = COMPOUND_AVERAGE;
533            self.b.palette_size_y = 0;
534            self.b.palette_size_uv = 0;
535            self.b.interp_filter = [crate::craft::video::av1::decode::header::BILINEAR; 2];
536            self.find_mv_stack(false);
537            self.assign_mv(false)?;
538            return Ok(());
539        }
540        self.b.is_inter = false;
541        let mi = &self.t.mi;
542        let (r, c) = (self.b.mi_row, self.b.mi_col);
543        let above = if self.b.avail_u { mi.y_mode[mi.idx(r - 1, c)] as usize } else { DC_PRED };
544        let left = if self.b.avail_l { mi.y_mode[mi.idx(r, c - 1)] as usize } else { DC_PRED };
545        let actx = INTRA_MODE_CONTEXT[above] as usize;
546        let lctx = INTRA_MODE_CONTEXT[left] as usize;
547        let mut cdf = self.t.intra_frame_y_mode_cdf[actx][lctx];
548        self.b.y_mode = self.sd.read_symbol(&mut cdf);
549        self.t.intra_frame_y_mode_cdf[actx][lctx] = cdf;
550        self.intra_angle_info_y();
551        if self.b.has_chroma {
552            self.read_uv_mode();
553            if self.b.uv_mode == UV_CFL_PRED {
554                self.read_cfl_alphas();
555            }
556            self.intra_angle_info_uv();
557        }
558        self.b.palette_size_y = 0;
559        self.b.palette_size_uv = 0;
560        let ms = self.b.mi_size;
561        if ms >= BLOCK_8X8 && block_width(ms) <= 64 && block_height(ms) <= 64 && self.fs.fh.allow_screen_content_tools {
562            self.palette_mode_info();
563        }
564        self.filter_intra_mode_info();
565        Ok(())
566    }

599    pub(crate) fn read_cfl_alphas(&mut self) {
600        let signs = self.sd.read_symbol(&mut self.cdf.cfl_sign);
601        let sign_u = (signs + 1) / 3;
602        let sign_v = (signs + 1) % 3;
603        self.b.cfl_alpha_u = if sign_u != CFL_SIGN_ZERO {
604            let ctx = (sign_u - 1) * 3 + sign_v;
605            let a = 1 + self.sd.read_symbol(&mut self.cdf.cfl_alpha[ctx]) as i32;
606            if sign_u == CFL_SIGN_NEG { -a } else { a }
607        } else {
608            0
609        };
610        self.b.cfl_alpha_v = if sign_v != CFL_SIGN_ZERO {
611            let ctx = (sign_v - 1) * 3 + sign_u;
612            let a = 1 + self.sd.read_symbol(&mut self.cdf.cfl_alpha[ctx]) as i32;
613            if sign_v == CFL_SIGN_NEG { -a } else { a }
614        } else {
615            0
616        };
617    }

628    pub(crate) fn read_segment_id(&mut self) {
629        let (r, c) = (self.b.mi_row, self.b.mi_col);
630        let mi = &self.t.mi;
631        let prev_ul = if self.b.avail_u && self.b.avail_l { mi.segment_id[mi.idx(r - 1, c - 1)] as i32 } else { -1 };
632        let prev_u = if self.b.avail_u { mi.segment_id[mi.idx(r - 1, c)] as i32 } else { -1 };
633        let prev_l = if self.b.avail_l { mi.segment_id[mi.idx(r, c - 1)] as i32 } else { -1 };
634        let pred = if prev_u == -1 {
635            if prev_l == -1 { 0 } else { prev_l }
636        } else if prev_l == -1 || prev_ul == prev_u {
637            prev_u
638        } else {
639            prev_l
640        };
641        if self.b.skip {
642            self.b.segment_id = pred as usize;
643        } else {
644            let ctx = if prev_ul < 0 {
645                0
646            } else if prev_ul == prev_u && prev_ul == prev_l {
647                2
648            } else if prev_ul == prev_u || prev_ul == prev_l || prev_u == prev_l {
649                1
650            } else {
651                0
652            };
653            let s = self.sd.read_symbol(&mut self.cdf.segment_id[ctx]) as i32;
654            let max = self.fs.fh.seg.last_active_seg_id as i32 + 1;
655            self.b.segment_id = neg_deinterleave(s, pred, max).clamp(0, 7) as usize;
656        }
657    }

663    pub(crate) fn read_skip(&mut self) {
664        if self.fs.fh.seg.seg_id_pre_skip && self.seg_feature_active(SEG_LVL_SKIP) {
665            self.b.skip = true;
666        } else {

668            let mi = &self.t.mi;
669            let mut ctx = 0;
670            if self.b.avail_u {

672            }
673            if self.b.avail_l {
674                ctx += mi.skip[mi.idx(r, c - 1)] as usize;
675            }
676            self.b.skip = self.sd.read_symbol(&mut self.cdf.skip[ctx]) == 1;
677        }

680    pub(crate) fn read_cdef(&mut self) {
681        let fh = &self.fs.fh;
682        if self.b.skip || fh.coded_lossless || !self.fs.seq.enable_cdef || fh.allow_intrabc {
683            return;
684        }
685        let cdef_size4 = NUM_4X4_BLOCKS_WIDE[BLOCK_64X64] as usize;
686        let mask = !(cdef_size4 - 1);
687        let r = self.b.mi_row & mask;
688        let c = self.b.mi_col & mask;
689        if self.t.cdef_idx(r, c) == -1 {
690            let bits = fh.cdef.bits;
691            let v = self.sd.read_literal(bits) as i8;
692            let w4 = self.b.bw4;
693            let h4 = self.b.bh4;
694            let mut i = r;
695            while i < r + h4 {
696                let mut j = c;
697                while j < c + w4 {
698                    self.t.set_cdef_idx(i, j, v);
699                    j += cdef_size4;
700                }
701                i += cdef_size4;
702            }
703        }
704    }

706    pub(crate) fn read_delta_qindex(&mut self) {
707        let sb_size = if self.fs.seq.use_128x128_superblock { BLOCK_128X128 } else { BLOCK_64X64 };
708        if self.b.mi_size == sb_size && self.b.skip {
709            return;
710        }
711        if self.read_deltas {
712            let mut abs = self.sd.read_symbol(&mut self.cdf.delta_q) as i32;
713            if abs == DELTA_Q_SMALL as i32 {
714                let rem_bits = self.sd.read_literal(3) + 1;
715                let abs_bits = self.sd.read_literal(rem_bits) as i32;
716                abs = abs_bits + (1 << rem_bits) + 1;
717            }
718            if abs != 0 {
719                let sign = self.sd.read_literal(1);
720                let reduced = if sign == 1 { -abs } else { abs };
721                self.current_q_index = (self.current_q_index + (reduced << self.fs.fh.delta_q_res)).clamp(1, 255);
722            }
723        }
724    }

726    pub(crate) fn read_delta_lf(&mut self) {
727        let sb_size = if self.fs.seq.use_128x128_superblock { BLOCK_128X128 } else { BLOCK_64X64 };
728        if self.b.mi_size == sb_size && self.b.skip {
729            return;
730        }
731        let fh = &self.fs.fh;
732        if self.read_deltas && fh.delta_lf_present {
733            let count = if fh.delta_lf_multi { if self.fs.num_planes > 1 { FRAME_LF_COUNT } else { FRAME_LF_COUNT - 2 } } else { 1 };
734            let multi = fh.delta_lf_multi;
735            let res = fh.delta_lf_res;
736            for i in 0..count {
737                let mut abs =
738                    if multi { self.sd.read_symbol(&mut self.cdf.delta_lf_multi[i]) } else { self.sd.read_symbol(&mut self.cdf.delta_lf) } as i32;
739                if abs == DELTA_LF_SMALL as i32 {
740                    let n = self.sd.read_literal(3) + 1;
741                    let bits = self.sd.read_literal(n) as i32;
742                    abs = bits + (1 << n) + 1;
743                }
744                if abs != 0 {
745                    let sign = self.sd.read_literal(1);
746                    let reduced = if sign == 1 { -abs } else { abs };
747                    self.delta_lf[i] = (self.delta_lf[i] + (reduced << res)).clamp(-(MAX_LOOP_FILTER as i32), MAX_LOOP_FILTER as i32);
748                }
749            }
750        }
751    }

769/// Probability of the split_or_horz / split_or_vert outcome (spec 8.3.2): the spec gives cdf
770/// values as P(X <= k) * 32768 (increasing); the probability of partition k is
771/// cdf[k] - cdf[k - 1].
772fn split_psum(pcdf: &[u16], split_or_horz: bool, bsize: usize) -> i32 {
773    let pr = |k: usize| pcdf[k] as i32 - if k == 0 { 0 } else { pcdf[k - 1] as i32 };
774    let mut psum = if split_or_horz {
775        pr(PARTITION_VERT) + pr(PARTITION_SPLIT) + pr(PARTITION_HORZ_A) + pr(PARTITION_VERT_A) + pr(PARTITION_VERT_B)
776    } else {
777        pr(PARTITION_HORZ) + pr(PARTITION_SPLIT) + pr(PARTITION_HORZ_A) + pr(PARTITION_HORZ_B) + pr(PARTITION_VERT_A)
778    };
779    if bsize != BLOCK_128X128 {
780        psum += if split_or_horz { pr(PARTITION_VERT_4) } else { pr(PARTITION_HORZ_4) };
781    }
782    psum
