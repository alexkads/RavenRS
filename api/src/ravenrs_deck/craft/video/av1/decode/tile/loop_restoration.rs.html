10    pub(super) fn read_lr(&mut self, r: usize, c: usize, bsize: usize) {
11        if self.fs.fh.allow_intrabc {
12            return;
13        }
14        let w = NUM_4X4_BLOCKS_WIDE[bsize] as usize;
15        let h = NUM_4X4_BLOCKS_HIGH[bsize] as usize;
16        for plane in 0..self.fs.num_planes {
17            if self.fs.fh.lr.frame_restoration_type[plane] == RESTORE_NONE as u8 {
18                continue;
19            }
20            let sub_x = if plane == 0 { 0 } else { self.fs.ssx };
21            let sub_y = if plane == 0 { 0 } else { self.fs.ssy };
22            let unit_size = self.fs.fh.lr.loop_restoration_size[plane] as usize;
23            let (unit_rows, unit_cols) = (self.fs.lr_unit_rows[plane], self.fs.lr_unit_cols[plane]);
24            let unit_row_start = (r * (4 >> sub_y)).div_ceil(unit_size);
25            let unit_row_end = unit_rows.min(((r + h) * (4 >> sub_y)).div_ceil(unit_size));
26            let (numerator, denominator) = if self.fs.fh.use_superres {
27                ((4 >> sub_x) * self.fs.fh.superres_denom as usize, unit_size * SUPERRES_NUM)
28            } else {
29                (4 >> sub_x, unit_size)
30            };
31            let unit_col_start = (c * numerator).div_ceil(denominator);
32            let unit_col_end = unit_cols.min(((c + w) * numerator).div_ceil(denominator));
33            for unit_row in unit_row_start..unit_row_end {
34                for unit_col in unit_col_start..unit_col_end {
35                    self.read_lr_unit(plane, unit_row, unit_col);
36                }
37            }
38        }
39    }

41    fn read_lr_unit(&mut self, plane: usize, unit_row: usize, unit_col: usize) {
42        let frt = self.fs.fh.lr.frame_restoration_type[plane];
43        let rtype = if frt == RESTORE_WIENER as u8 {
44            if self.sd.read_symbol(&mut self.cdf.use_wiener) == 1 { RESTORE_WIENER as u8 } else { RESTORE_NONE as u8 }
45        } else if frt == RESTORE_SGRPROJ as u8 {
46            if self.sd.read_symbol(&mut self.cdf.use_sgrproj) == 1 { RESTORE_SGRPROJ as u8 } else { RESTORE_NONE as u8 }
47        } else {
48            // restoration_type symbol: 0 = NONE, 1 = WIENER, 2 = SGRPROJ
49            match self.sd.read_symbol(&mut self.cdf.restoration_type) {
50                0 => RESTORE_NONE as u8,
51                1 => RESTORE_WIENER as u8,
52                _ => RESTORE_SGRPROJ as u8,
53            }
54        };
55        let idx = unit_row * self.fs.lr_unit_cols[plane] + unit_col;
56        self.t.lr.lr_type[plane][idx] = rtype;
57        self.t.lr_written[plane].push(idx);
58        if rtype == RESTORE_WIENER as u8 {
59            for pass in 0..2 {
60                let first = if plane > 0 {
61                    self.t.lr.lr_wiener[plane][idx][pass][0] = 0;
62                    1
63                } else {
64                    0
65                };
66                for j in first..3 {
67                    let v = self.decode_signed_subexp_with_ref_bool(
68                        WIENER_TAPS_MIN[j] as i32,
69                        WIENER_TAPS_MAX[j] as i32 + 1,
70                        WIENER_TAPS_K[j] as u32,
71                        self.ref_lr_wiener[plane][pass][j],
72                    );
73                    self.t.lr.lr_wiener[plane][idx][pass][j] = v as i8;
74                    self.ref_lr_wiener[plane][pass][j] = v;
75                }
76            }
77        } else if rtype == RESTORE_SGRPROJ as u8 {
78            let set = self.sd.read_literal(SGRPROJ_PARAMS_BITS as u32) as usize;
79            self.t.lr.lr_sgr_set[plane][idx] = set as u8;
80            for i in 0..2 {
81                let (min, max) = (SGRPROJ_XQD_MIN[i] as i32, SGRPROJ_XQD_MAX[i] as i32);
82                let radius = SGR_PARAMS[set][i * 2];
83                let v = if radius != 0 {
84                    self.decode_signed_subexp_with_ref_bool(min, max + 1, SGRPROJ_PRJ_SUBEXP_K as u32, self.ref_sgr_xqd[plane][i])
85                } else if i == 1 {
86                    ((1 << SGRPROJ_PRJ_BITS) - self.ref_sgr_xqd[plane][0]).clamp(min, max)
87                } else {
88                    0
89                };
90                self.t.lr.lr_sgr_xqd[plane][idx][i] = v as i8;
91                self.ref_sgr_xqd[plane][i] = v;
92            }
93        }
94    }

96    fn decode_signed_subexp_with_ref_bool(&mut self, low: i32, high: i32, k: u32, r: i32) -> i32 {
97        let x = self.decode_unsigned_subexp_with_ref_bool(high - low, k, r - low);
98        x + low
99    }
100
101    fn decode_unsigned_subexp_with_ref_bool(&mut self, mx: i32, k: u32, r: i32) -> i32 {
102        let v = self.decode_subexp_bool(mx, k);
103        if (r << 1) <= mx { inverse_recenter(r, v) } else { mx - 1 - inverse_recenter(mx - 1 - r, v) }
104    }
105
106    fn decode_subexp_bool(&mut self, num_syms: i32, k: u32) -> i32 {
107        let mut i = 0u32;
108        let mut mk = 0i32;
109        loop {
110            let b2 = if i > 0 { k + i - 1 } else { k };
111            let a = 1i32 << b2;
112            if num_syms <= mk + 3 * a {
113                return self.sd.read_ns((num_syms - mk) as u32) as i32 + mk;
114            } else if self.sd.read_literal(1) == 1 {
115                i += 1;
116                mk += a;
117            } else {
118                return self.sd.read_literal(b2) as i32 + mk;
119            }
120        }
121    }
