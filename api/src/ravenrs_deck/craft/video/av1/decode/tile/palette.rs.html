10    pub(crate) fn palette_mode_info(&mut self) {
11        let ms = self.b.mi_size;
12        let bsize_ctx = MI_WIDTH_LOG2[ms] as usize + MI_HEIGHT_LOG2[ms] as usize - 2;
13        let bd = self.fs.bit_depth;
14        let (r, c) = (self.b.mi_row, self.b.mi_col);
15        if self.b.y_mode == DC_PRED {
16            let mi = &self.t.mi;
17            let mut ctx = 0;
18            if self.b.avail_u && mi.palette_size[0][mi.idx(r - 1, c)] > 0 {
19                ctx += 1;
20            }
21            if self.b.avail_l && mi.palette_size[0][mi.idx(r, c - 1)] > 0 {
22                ctx += 1;
23            }
24            if self.sd.read_symbol(&mut self.cdf.palette_y_mode[bsize_ctx][ctx]) == 1 {
25                let n = self.sd.read_symbol(&mut self.cdf.palette_y_size[bsize_ctx]) + 2;
26                self.b.palette_size_y = n;
27                let cache = self.get_palette_cache(0);
28                let mut colors = [0u16; 8];
29                let mut idx = 0;
30                for &cv in &cache {
31                    if idx >= n {
32                        break;
33                    }
34                    if self.sd.read_literal(1) == 1 {
35                        colors[idx] = cv;
36                        idx += 1;
37                    }
38                }
39                if idx < n {
40                    colors[idx] = self.sd.read_literal(bd) as u16;
41                    idx += 1;
42                }
43                let mut palette_bits = 0;
44                if idx < n {
45                    let min_bits = bd - 3;
46                    palette_bits = min_bits + self.sd.read_literal(2);
47                }
48                while idx < n {
49                    let delta = self.sd.read_literal(palette_bits) as i32 + 1;
50                    colors[idx] = (colors[idx - 1] as i32 + delta).clamp(0, (1 << bd) - 1) as u16;
51                    let range = (1i32 << bd) - colors[idx] as i32 - 1;
52                    palette_bits = palette_bits.min(ceil_log2(range as u32));
53                    idx += 1;
54                }
55                colors[..n].sort_unstable();
56                self.b.palette_colors_y = colors;
57            }
58        }
59        if self.b.has_chroma && self.b.uv_mode == DC_PRED {
60            let ctx = (self.b.palette_size_y > 0) as usize;
61            if self.sd.read_symbol(&mut self.cdf.palette_uv_mode[ctx]) == 1 {
62                let n = self.sd.read_symbol(&mut self.cdf.palette_uv_size[bsize_ctx]) + 2;
63                self.b.palette_size_uv = n;
64                let cache = self.get_palette_cache(1);
65                let mut colors = [0u16; 8];
66                let mut idx = 0;
67                for &cv in &cache {
68                    if idx >= n {
69                        break;
70                    }
71                    if self.sd.read_literal(1) == 1 {
72                        colors[idx] = cv;
73                        idx += 1;
74                    }
75                }
76                if idx < n {
77                    colors[idx] = self.sd.read_literal(bd) as u16;
78                    idx += 1;
79                }
80                let mut palette_bits = 0;
81                if idx < n {
82                    palette_bits = bd - 3 + self.sd.read_literal(2);
83                }
84                while idx < n {
85                    let delta = self.sd.read_literal(palette_bits) as i32;
86                    colors[idx] = (colors[idx - 1] as i32 + delta).clamp(0, (1 << bd) - 1) as u16;
87                    let range = (1i32 << bd) - colors[idx] as i32;
88                    palette_bits = palette_bits.min(ceil_log2(range as u32));
89                    idx += 1;
90                }
91                colors[..n].sort_unstable();
92                self.b.palette_colors_u = colors;
93                let mut v = [0u16; 8];
94                if self.sd.read_literal(1) == 1 {
95                    let min_bits = bd - 4;
96                    let max_val = 1i32 << bd;
97                    let bits = min_bits + self.sd.read_literal(2);
98                    v[0] = self.sd.read_literal(bd) as u16;
99                    for idx in 1..n {
100                        let mut delta = self.sd.read_literal(bits) as i32;
101                        if delta != 0 && self.sd.read_literal(1) == 1 {
102                            delta = -delta;
103                        }
104                        let mut val = v[idx - 1] as i32 + delta;
105                        if val < 0 {
106                            val += max_val;
107                        }
108                        if val >= max_val {
109                            val -= max_val;
110                        }
111                        v[idx] = val.clamp(0, (1 << bd) - 1) as u16;
112                    }
113                } else {
114                    for item in v.iter_mut().take(n) {
115                        *item = self.sd.read_literal(bd) as u16;
116                    }
117                }
118                self.b.palette_colors_v = v;
119            }
120        }
121    }

123    fn get_palette_cache(&self, plane: usize) -> Vec<u16> {
124        let (r, c) = (self.b.mi_row, self.b.mi_col);
125        let mi = &self.t.mi;
126        let above_n = if (r * 4) % 64 != 0 { mi.palette_size[plane][mi.idx(r - 1, c)] as usize } else { 0 };
127        let left_n = if self.b.avail_l { mi.palette_size[plane][mi.idx(r, c - 1)] as usize } else { 0 };
128        let above: &[u16] = if above_n > 0 { &mi.palette_colors[plane][mi.idx(r - 1, c)][..above_n] } else { &[] };
129        let left: &[u16] = if left_n > 0 { &mi.palette_colors[plane][mi.idx(r, c - 1)][..left_n] } else { &[] };
130        let (mut ai, mut li) = (0, 0);
131        let mut out: Vec<u16> = Vec::with_capacity(16);
132        while ai < above_n && li < left_n {
133            let (ac, lc) = (above[ai], left[li]);
134            if lc < ac {
135                if out.last() != Some(&lc) {
136                    out.push(lc);
137                }
138                li += 1;
139            } else {
140                if out.last() != Some(&ac) {
141                    out.push(ac);
142                }
143                ai += 1;
144                if lc == ac {
145                    li += 1;
146                }
147            }
148        }
149        for &v in &above[ai..above_n] {
150            if out.last() != Some(&v) {
151                out.push(v);
152            }
153        }
154        for &v in &left[li..left_n] {
155            if out.last() != Some(&v) {
156                out.push(v);
157            }
158        }
159        out
160    }

162    pub(super) fn palette_tokens(&mut self) {
163        let ms = self.b.mi_size;
164        let bh = block_height(ms);
165        let bw = block_width(ms);
166        let (mi_rows, mi_cols) = (self.fs.fh.mi_rows as usize, self.fs.fh.mi_cols as usize);
167        let on_h = bh.min((mi_rows - self.b.mi_row) * 4);
168        let on_w = bw.min((mi_cols - self.b.mi_col) * 4);
169        if self.b.palette_size_y > 0 {
170            let n = self.b.palette_size_y;
171            let mut map = std::mem::take(&mut self.color_map_y);
172            self.read_color_map(&mut map, n, MapDims { bw, bh, on_w, on_h }, 0);
173            self.color_map_y = map;
174        }
175        if self.b.palette_size_uv > 0 {
176            let n = self.b.palette_size_uv;
177            let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
178            let (mut bw, mut bh, mut on_w, mut on_h) = (bw >> ssx, bh >> ssy, on_w >> ssx, on_h >> ssy);
179            if bw < 4 {
180                bw += 2;
181                on_w += 2;
182            }
183            if bh < 4 {
184                bh += 2;
185                on_h += 2;
186            }
187            let mut map = std::mem::take(&mut self.color_map_uv);
188            self.read_color_map(&mut map, n, MapDims { bw, bh, on_w, on_h }, 1);
189            self.color_map_uv = map;
190        }
191    }

193    /// Color index map (stride 64).
194    fn read_color_map(&mut self, map: &mut [u8], n: usize, dims: MapDims, plane: usize) {
195        let MapDims { bw, bh, on_w, on_h } = dims;
196        map[0] = self.sd.read_ns(n as u32) as u8;
197        for i in 1..(on_h + on_w - 1) {
198            let jmax = i.min(on_w - 1);
199            let jmin = (i as isize - on_h as isize + 1).max(0) as usize;
200            let mut j = jmax as isize;
201            while j >= jmin as isize {
202                let (rr, cc) = (i - j as usize, j as usize);
203                let (order, hash) = palette_color_context(map, rr, cc, n);
204                let ctx = PALETTE_COLOR_CONTEXT[hash] as usize;
205                let cdf: &mut [u16] = match (plane, n) {
206                    (0, 2) => &mut self.cdf.palette_2_y_color[ctx],
207                    (0, 3) => &mut self.cdf.palette_3_y_color[ctx],
208                    (0, 4) => &mut self.cdf.palette_4_y_color[ctx],
209                    (0, 5) => &mut self.cdf.palette_5_y_color[ctx],
210                    (0, 6) => &mut self.cdf.palette_6_y_color[ctx],
211                    (0, 7) => &mut self.cdf.palette_7_y_color[ctx],
212                    (0, _) => &mut self.cdf.palette_8_y_color[ctx],
213                    (_, 2) => &mut self.cdf.palette_2_uv_color[ctx],
214                    (_, 3) => &mut self.cdf.palette_3_uv_color[ctx],
215                    (_, 4) => &mut self.cdf.palette_4_uv_color[ctx],
216                    (_, 5) => &mut self.cdf.palette_5_uv_color[ctx],
217                    (_, 6) => &mut self.cdf.palette_6_uv_color[ctx],
218                    (_, 7) => &mut self.cdf.palette_7_uv_color[ctx],
219                    _ => &mut self.cdf.palette_8_uv_color[ctx],
220                };
221                let s = self.sd.read_symbol(cdf);
222                map[rr * 64 + cc] = order[s];
223                j -= 1;
224            }
225        }
226        for i in 0..on_h {
227            for j in on_w..bw {
228                map[i * 64 + j] = map[i * 64 + on_w - 1];
229            }
230        }
231        for i in on_h..bh {
232            for j in 0..bw {
233                map[i * 64 + j] = map[(on_h - 1) * 64 + j];
234            }
235        }
236    }

248/// get_palette_color_context( ): returns (ColorOrder, ColorContextHash).
249fn palette_color_context(map: &[u8], r: usize, c: usize, n: usize) -> ([u8; 8], usize) {
250    let mut scores = [0i32; PALETTE_COLORS];
251    let mut order = [0u8, 1, 2, 3, 4, 5, 6, 7];
252    if c > 0 {
253        scores[map[r * 64 + c - 1] as usize] += 2;
254    }
255    if r > 0 && c > 0 {
256        scores[map[(r - 1) * 64 + c - 1] as usize] += 1;
257    }
258    if r > 0 {
259        scores[map[(r - 1) * 64 + c] as usize] += 2;
260    }
261    for i in 0..PALETTE_NUM_NEIGHBORS {
262        let mut max_score = scores[i];
263        let mut max_idx = i;
264        for j in i + 1..n {
265            if scores[j] > max_score {
266                max_score = scores[j];
267                max_idx = j;
268            }
269        }
270        if max_idx != i {
271            let max_score = scores[max_idx];
272            let max_order = order[max_idx];
273            for k in (i + 1..=max_idx).rev() {
274                scores[k] = scores[k - 1];
275                order[k] = order[k - 1];
276            }
277            scores[i] = max_score;
278            order[i] = max_order;
279        }
280    }
281    let mut hash = 0usize;
282    for i in 0..PALETTE_NUM_NEIGHBORS {
283        hash += scores[i] as usize * PALETTE_COLOR_HASH_MULTIPLIERS[i] as usize;
284    }
285    (order, hash)
