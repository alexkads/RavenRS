13    /// compute_prediction( ) (5.11.33)
14    pub(super) fn compute_prediction(&mut self) -> Result<()> {
15        if !self.b.is_inter {
16            return Ok(());
17        }
18        let sb_mask = if self.fs.seq.use_128x128_superblock { 31 } else { 15 };
19        let sub_row = self.b.mi_row & sb_mask;
20        let sub_col = self.b.mi_col & sb_mask;
21        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
22        let is_inter_intra = self.b.ref_frame[1] == INTRA_FRAME as i8;
23        for plane in 0..(1 + 2 * self.b.has_chroma as usize) {
24            let plane_sz = self.fs.plane_residual_size(self.b.mi_size, plane);
25            let n4w = NUM_4X4_BLOCKS_WIDE[plane_sz] as usize;
26            let n4h = NUM_4X4_BLOCKS_HIGH[plane_sz] as usize;
27            let log2w = 2 + MI_WIDTH_LOG2[plane_sz] as u32;
28            let log2h = 2 + MI_HEIGHT_LOG2[plane_sz] as u32;
29            let sub_x = if plane > 0 { ssx } else { 0 };
30            let sub_y = if plane > 0 { ssy } else { 0 };
31            let base_x = (self.b.mi_col >> sub_x) * 4;
32            let base_y = (self.b.mi_row >> sub_y) * 4;
33            let mut cand_row = (self.b.mi_row >> sub_y) << sub_y;
34            let mut cand_col = (self.b.mi_col >> sub_x) << sub_x;
35            if is_inter_intra {
36                let mode = match self.b.interintra_mode {
37                    II_DC_PRED => DC_PRED,
38                    II_V_PRED => V_PRED,
39                    II_H_PRED => H_PRED,
40                    _ => SMOOTH_PRED,
41                };
42                let sbr = (sub_row >> sub_y) as isize;
43                let sbc = (sub_col >> sub_x) as isize;
44                let max_x = ((self.fs.fh.mi_cols as usize * 4) >> sub_x) as i32 - 1;
45                let max_y = ((self.fs.fh.mi_rows as usize * 4) >> sub_y) as i32 - 1;
46                let params = IntraParams {
47                    plane,
48                    x: base_x,
49                    y: base_y,
50                    have_left: if plane == 0 { self.b.avail_l } else { self.b.avail_l_chroma },
51                    have_above: if plane == 0 { self.b.avail_u } else { self.b.avail_u_chroma },
52                    have_above_right: self.block_decoded_at(plane, sbr - 1, sbc + n4w as isize),
53                    have_below_left: self.block_decoded_at(plane, sbr + n4h as isize, sbc - 1),
54                    mode,
55                    log2w,
56                    log2h,
57                    max_x,
58                    max_y,
59                    bit_depth: self.fs.bit_depth,
60                    angle_delta: 0,
61                    use_filter_intra: false,
62                    filter_intra_mode: 0,
63                    enable_intra_edge_filter: self.fs.seq.enable_intra_edge_filter,
64                    filter_type: if is_directional_mode(mode) && self.fs.seq.enable_intra_edge_filter { self.get_filter_type(plane) } else { false },
65                };
66                predict_intra(&mut self.t.cur.planes[plane], &params);
67            }
68            let mut pred_w = block_width(self.b.mi_size) >> sub_x;
69            let mut pred_h = block_height(self.b.mi_size) >> sub_y;
70            let mut some_use_intra = false;
71            for r in 0..(n4h << sub_y) {
72                for c in 0..(n4w << sub_x) {
73                    let (rr, cc) = (cand_row + r, cand_col + c);
74                    if rr < self.t.mi.rows && cc < self.t.mi.cols && self.t.mi.ref_frame[self.t.mi.idx(rr, cc)][0] == INTRA_FRAME as i8 {
75                        some_use_intra = true;
76                    }
77                }
78            }
79            if some_use_intra {
80                pred_w = n4w * 4;
81                pred_h = n4h * 4;
82                cand_row = self.b.mi_row;
83                cand_col = self.b.mi_col;
84            }
85            let mut r = 0;
86            let mut y = 0;
87            while y < n4h * 4 {
88                let mut c = 0;
89                let mut x = 0;
90                while x < n4w * 4 {
91                    self.predict_inter(PredBlock { plane, x: base_x + x, y: base_y + y, w: pred_w, h: pred_h }, cand_row + r, cand_col + c);
92                    x += pred_w;
93                    c += 1;
94                }
95                y += pred_h;
96                r += 1;
97            }
98        }
99        Ok(())
100    }

102    fn get_tx_size(&self, plane: usize, tx_sz: usize) -> usize {
103        if plane == 0 {
104            return tx_sz;
105        }
106        let uv_tx = MAX_TX_SIZE_RECT[self.fs.plane_residual_size(self.b.mi_size, plane)] as usize;
107        if TX_WIDTH[uv_tx] == 64 || TX_HEIGHT[uv_tx] == 64 {
108            if TX_WIDTH[uv_tx] == 16 {
109                return TX_16X32;
110            }
111            if TX_HEIGHT[uv_tx] == 16 {
112                return TX_32X16;
113            }
114            return TX_32X32;
115        }
116        uv_tx
117    }

119    pub(super) fn residual(&mut self) -> Result<()> {
120        let ms = self.b.mi_size;
121        let width_chunks = (block_width(ms) >> 6).max(1);
122        let height_chunks = (block_height(ms) >> 6).max(1);
123        let mi_size_chunk = if width_chunks > 1 || height_chunks > 1 { BLOCK_64X64 } else { ms };
124        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
125        for chunk_y in 0..height_chunks {
126            for chunk_x in 0..width_chunks {
127                let mi_row_chunk = self.b.mi_row + (chunk_y << 4);
128                let mi_col_chunk = self.b.mi_col + (chunk_x << 4);
129                for plane in 0..(1 + 2 * self.b.has_chroma as usize) {
130                    let tx_sz = if self.b.lossless { TX_4X4 } else { self.get_tx_size(plane, self.b.tx_size) };
131                    let step_x = TX_WIDTH[tx_sz] as usize >> 2;
132                    let step_y = TX_HEIGHT[tx_sz] as usize >> 2;
133                    let plane_sz = self.fs.plane_residual_size(mi_size_chunk, plane);
134                    let num4x4_w = NUM_4X4_BLOCKS_WIDE[plane_sz] as usize;
135                    let num4x4_h = NUM_4X4_BLOCKS_HIGH[plane_sz] as usize;
136                    let sub_x = if plane > 0 { ssx } else { 0 };
137                    let sub_y = if plane > 0 { ssy } else { 0 };
138                    let base_x = (mi_col_chunk >> sub_x) * 4;
139                    let base_y = (mi_row_chunk >> sub_y) * 4;
140                    if self.b.is_inter && !self.b.lossless && plane == 0 {
141                        self.transform_tree(base_x, base_y, num4x4_w * 4, num4x4_h * 4)?;
142                    } else {
143                        let base_x_block = (self.b.mi_col >> sub_x) * 4;
144                        let base_y_block = (self.b.mi_row >> sub_y) * 4;
145                        let mut y = 0;
146                        while y < num4x4_h {
147                            let mut x = 0;
148                            while x < num4x4_w {
149                                self.transform_block(
150                                    plane,
151                                    base_x_block,
152                                    base_y_block,
153                                    tx_sz,
154                                    x + ((chunk_x << 4) >> sub_x),
155                                    y + ((chunk_y << 4) >> sub_y),
156                                )?;
157                                x += step_x;
158                            }
159                            y += step_y;
160                        }
161                    }
162                }
163            }
164        }
165        Ok(())
166    }

168    fn transform_tree(&mut self, start_x: usize, start_y: usize, w: usize, h: usize) -> Result<()> {
169        let max_x = self.fs.fh.mi_cols as usize * 4;
170        let max_y = self.fs.fh.mi_rows as usize * 4;
171        if start_x >= max_x || start_y >= max_y {
172            return Ok(());
173        }
174        let row = start_y >> 2;
175        let col = start_x >> 2;
176        let mi = &self.t.mi;
177        let luma_tx = mi.inter_tx_size[mi.idx(row, col)] as usize;
178        let lw = TX_WIDTH[luma_tx] as usize;
179        let lh = TX_HEIGHT[luma_tx] as usize;
180        if w <= lw && h <= lh {
181            let tx_sz = find_tx_size(w, h);
182            self.transform_block(0, start_x, start_y, tx_sz, 0, 0)?;
183        } else if w > h {
184            self.transform_tree(start_x, start_y, w / 2, h)?;
185            self.transform_tree(start_x + w / 2, start_y, w / 2, h)?;
186        } else if w < h {
187            self.transform_tree(start_x, start_y, w, h / 2)?;
188            self.transform_tree(start_x, start_y + h / 2, w, h / 2)?;
189        } else {
190            self.transform_tree(start_x, start_y, w / 2, h / 2)?;
191            self.transform_tree(start_x + w / 2, start_y, w / 2, h / 2)?;
192            self.transform_tree(start_x, start_y + h / 2, w / 2, h / 2)?;
193            self.transform_tree(start_x + w / 2, start_y + h / 2, w / 2, h / 2)?;
194        }
195        Ok(())
196    }

198    fn transform_block(&mut self, plane: usize, base_x: usize, base_y: usize, tx_sz: usize, x: usize, y: usize) -> Result<()> {
199        let start_x = base_x + 4 * x;
200        let start_y = base_y + 4 * y;
201        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
202        let sub_x = if plane > 0 { ssx } else { 0 };
203        let sub_y = if plane > 0 { ssy } else { 0 };
204        let row = (start_y << sub_y) >> 2;
205        let col = (start_x << sub_x) >> 2;
206        let sb_mask = if self.fs.seq.use_128x128_superblock { 31 } else { 15 };
207        let sub_block_mi_row = row & sb_mask;
208        let sub_block_mi_col = col & sb_mask;
209        let step_x = TX_WIDTH[tx_sz] as usize >> 2;
210        let step_y = TX_HEIGHT[tx_sz] as usize >> 2;
211        let max_x = (self.fs.fh.mi_cols as usize * 4) >> sub_x;
212        let max_y = (self.fs.fh.mi_rows as usize * 4) >> sub_y;
213        if start_x >= max_x || start_y >= max_y {
214            return Ok(());
215        }
216        if !self.b.is_inter {
217            let palette = if plane == 0 { self.b.palette_size_y } else { self.b.palette_size_uv };
218            if palette > 0 {
219                self.predict_palette(plane, start_x, start_y, x, y, tx_sz);
220            } else {
221                let is_cfl = plane > 0 && self.b.uv_mode == UV_CFL_PRED;
222                let mode = if plane == 0 {
223                    self.b.y_mode
224                } else if is_cfl {
225                    DC_PRED
226                } else {
227                    self.b.uv_mode
228                };
229                let log2w = TX_WIDTH_LOG2[tx_sz] as u32;
230                let log2h = TX_HEIGHT_LOG2[tx_sz] as u32;
231                let have_left = (if plane == 0 { self.b.avail_l } else { self.b.avail_l_chroma }) || x > 0;
232                let have_above = (if plane == 0 { self.b.avail_u } else { self.b.avail_u_chroma }) || y > 0;
233                let sbr = (sub_block_mi_row >> sub_y) as isize;
234                let sbc = (sub_block_mi_col >> sub_x) as isize;
235                let have_above_right = self.block_decoded_at(plane, sbr - 1, sbc + step_x as isize);
236                let have_below_left = self.block_decoded_at(plane, sbr + step_y as isize, sbc - 1);
237                let filter_type = if is_directional_mode(mode) && self.fs.seq.enable_intra_edge_filter { self.get_filter_type(plane) } else { false };
238                let params = IntraParams {
239                    plane,
240                    x: start_x,
241                    y: start_y,
242                    have_left,
243                    have_above,
244                    have_above_right,
245                    have_below_left,
246                    mode,
247                    log2w,
248                    log2h,
249                    max_x: max_x as i32 - 1,
250                    max_y: max_y as i32 - 1,
251                    bit_depth: self.fs.bit_depth,
252                    angle_delta: if plane == 0 { self.b.angle_delta_y } else { self.b.angle_delta_uv },
253                    use_filter_intra: self.b.use_filter_intra,
254                    filter_intra_mode: self.b.filter_intra_mode,
255                    enable_intra_edge_filter: self.fs.seq.enable_intra_edge_filter,
256                    filter_type,
257                };
258                predict_intra(&mut self.t.cur.planes[plane], &params);
259                if is_cfl {
260                    self.predict_cfl(plane, start_x, start_y, tx_sz);
261                }
262            }
263            if plane == 0 {
264                self.b.max_luma_w = start_x + step_x * 4;
265                self.b.max_luma_h = start_y + step_y * 4;
266            }
267        }
268        if !self.b.skip {
269            let eob = self.coeffs(plane, start_x, start_y, tx_sz);
270            if eob > 0 {
271                self.reconstruct(plane, start_x, start_y, tx_sz, eob == 1);
272            }
273        }
274        for i in 0..step_y {
275            for j in 0..step_x {
276                let rr = (row >> sub_y) + i;
277                let cc = (col >> sub_x) + j;
278                self.t.set_lf_tx_size(plane, rr, cc, tx_sz as u8);
279                let by = (sub_block_mi_row >> sub_y) + i;
280                let bx = (sub_block_mi_col >> sub_x) + j;
281                if by + 1 < 35 && bx + 1 < 35 {
282                    self.block_decoded[plane][by + 1][bx + 1] = true;
283                }
284            }
285        }
286        Ok(())
287    }

289    pub(crate) fn get_filter_type(&self, plane: usize) -> bool {
290        let (ssx, ssy) = (self.fs.ssx, self.fs.ssy);
291        let b = &self.b;
292        let mi = &self.t.mi;
293        let is_smooth = |row: usize, col: usize| -> bool {
294            let i = mi.idx(row, col);
295            let mode = if plane == 0 {
296                mi.y_mode[i] as usize
297            } else {
298                if mi.ref_frame[i][0] > INTRA_FRAME as i8 {
299                    return false;
300                }
301                mi.uv_mode[i] as usize
302            };
303            mode == SMOOTH_PRED || mode == SMOOTH_V_PRED || mode == SMOOTH_H_PRED
304        };
305        let mut above_smooth = false;
306        let mut left_smooth = false;
307        if if plane == 0 { b.avail_u } else { b.avail_u_chroma } {
308            let mut r = b.mi_row as isize - 1;
309            let mut c = b.mi_col;
310            if plane > 0 {
311                if ssx == 1 && (b.mi_col & 1) == 0 {
312                    c += 1;
313                }
314                if ssy == 1 && (b.mi_row & 1) == 1 {
315                    r -= 1;
316                }
317            }
318            above_smooth = is_smooth(r as usize, c);
319        }
320        if if plane == 0 { b.avail_l } else { b.avail_l_chroma } {
321            let mut r = b.mi_row;
322            let mut c = b.mi_col as isize - 1;
323            if plane > 0 {
324                if ssx == 1 && (b.mi_col & 1) == 1 {
325                    c -= 1;
326                }
327                if ssy == 1 && (b.mi_row & 1) == 0 {
328                    r += 1;
329                }
330            }
331            left_smooth = is_smooth(r, c as usize);
332        }
333        above_smooth || left_smooth
334    }

336    fn predict_palette(&mut self, plane: usize, start_x: usize, start_y: usize, x: usize, y: usize, tx_sz: usize) {
337        let w = TX_WIDTH[tx_sz] as usize;
338        let h = TX_HEIGHT[tx_sz] as usize;
339        let palette = match plane {
340            0 => self.b.palette_colors_y,
341            1 => self.b.palette_colors_u,
342            _ => self.b.palette_colors_v,
343        };
344        let map = if plane == 0 { &self.color_map_y } else { &self.color_map_uv };
345        let pl = &mut self.t.cur.planes[plane];
346        for i in 0..h {
347            for j in 0..w {
348                let idx = map[(y * 4 + i) * 64 + x * 4 + j] as usize;
349                pl.set(start_x + j, start_y + i, palette[idx]);
350            }
351        }
352    }

354    fn predict_cfl(&mut self, plane: usize, start_x: usize, start_y: usize, tx_sz: usize) {
355        let w = TX_WIDTH[tx_sz] as usize;
356        let h = TX_HEIGHT[tx_sz] as usize;
357        let (sub_x, sub_y) = (self.fs.ssx, self.fs.ssy);
358        let alpha = if plane == 1 { self.b.cfl_alpha_u } else { self.b.cfl_alpha_v };
359        let bd = self.fs.bit_depth;
360        let mut l = [0i32; 32 * 32];
361        let mut avg: i64 = 0;
362        {
363            let luma = &self.t.cur.planes[0];
364            for i in 0..h {
365                let ly = ((start_y + i) << sub_y).min(self.b.max_luma_h - (1 << sub_y));
366                for j in 0..w {
367                    let lx = ((start_x + j) << sub_x).min(self.b.max_luma_w - (1 << sub_x));
368                    let mut t = 0i32;
369                    for dy in 0..=sub_y {
370                        for dx in 0..=sub_x {
371                            t += luma.at(lx + dx, ly + dy) as i32;
372                        }
373                    }
374                    let v = t << (3 - sub_x - sub_y);
375                    l[i * w + j] = v;
376                    avg += v as i64;
377                }
378            }
379        }
380        let shift = TX_WIDTH_LOG2[tx_sz] as u32 + TX_HEIGHT_LOG2[tx_sz] as u32;
381        let avg = ((avg + (1i64 << (shift - 1))) >> shift) as i32;
382        let pl = &mut self.t.cur.planes[plane];
383        for i in 0..h {
384            for j in 0..w {
385                let dc = pl.at(start_x + j, start_y + i) as i32;
386                let scaled = round2_signed(alpha * (l[i * w + j] - avg), 6);
387                pl.set(start_x + j, start_y + i, (dc + scaled).clamp(0, (1 << bd) - 1) as u16);
388            }
389        }
390    }
