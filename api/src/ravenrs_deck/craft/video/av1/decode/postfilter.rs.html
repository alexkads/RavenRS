12/// Luma rows per band of the row-parallel filters: whole 64x64 CDEF filter blocks, and no
13/// transform block crosses a band edge.
14const BAND_ROWS: usize = 64;
15
16/// The [start, end) luma rows of the bands covering `rows`: [`BAND_ROWS`] each (the last one
17/// runs to the end of the plane), or one band for everything when `pool` is sequential.
18pub(crate) fn bands(pool: &crate::craft::video::av1::decode::par::Pool, rows: usize) -> Vec<(usize, usize)> {
19    let n = if pool.is_parallel() { rows.div_ceil(BAND_ROWS).max(1) } else { 1 };
20    (0..n).map(|b| (b * BAND_ROWS, if b + 1 == n { usize::MAX } else { (b + 1) * BAND_ROWS })).collect()
21}

23/// The post-filters run on the frame worker, each in row bands on `pool` (the result does not
24/// depend on how the bands are scheduled). `draft`: only super-resolution (which sets the
25/// output size) runs.
26pub(crate) fn apply(fs: &mut FrameState, stats: &mut DecodeStats, pool: &crate::craft::video::av1::decode::par::Pool, draft: bool) {
27    let mut t = Timer::start();
28    let lvl = fs.fh.lf.level;
29    if (lvl[0] != 0 || lvl[1] != 0) && !draft {
30        loop_filter(fs, pool);
31    }
32    stats.add(Stage::LoopFilter, t.lap());
33    let cdef = if draft { None } else { crate::craft::video::av1::decode::cdef::apply(fs, pool) };
34    stats.add(Stage::Cdef, t.lap());
35    let (up_cur, up_cdef) = if fs.fh.use_superres {
36        let c = crate::craft::video::av1::decode::restoration::upscale(fs, &fs.cur);
37        let d = cdef.as_ref().map(|f| crate::craft::video::av1::decode::restoration::upscale(fs, f));
38        (c, d)
39    } else {
40        (std::mem::take(&mut fs.cur), cdef)
41    };
42    stats.add(Stage::Superres, t.lap());
43    fs.cur = if fs.fh.lr.uses_lr && !draft {
44        crate::craft::video::av1::decode::restoration::loop_restoration(fs, &up_cur, up_cdef.as_ref().unwrap_or(&up_cur), pool)
45    } else {
46        up_cdef.unwrap_or(up_cur)
47    };
48    stats.add(Stage::Restoration, t.lap());
49}

54/// Luma rows above a band that its horizontal edges read or write (p6 .. p0 of the widest
55/// filter, rounded up to whole 4x4 rows); chroma: `LF_MARGIN >> subsampling_y`.
56const LF_MARGIN: usize = 8;
57
58/// The loop filter in row bands, all of a frame's in one parallel pass (each join costs a
59/// worker wake-up, which is slow on a busy machine).
60///
61/// Vertical edges (pass 0) only touch the 4 rows of their own edge. A horizontal edge (pass 1)
62/// reads and writes at most `filter_size / 2` samples on each side, never more than the
63/// transform blocks on either side are high, so no two edges touch the same sample and
64/// any order gives the spec's raster-order result. Transform blocks never cross a band edge, so
65/// the only edges reaching outside a band are those on its top row, whose p side lies in the
66/// band above. Each band therefore also filters the vertical edges of a copy of the
67/// `LF_MARGIN` rows above it (the same work as the band above does on them), runs its top
68/// horizontal edges against that copy, and afterwards the samples those edges changed are
69/// written back; the band above never touches them.
70fn loop_filter(fs: &mut FrameState, pool: &crate::craft::video::av1::decode::par::Pool) {
71    let luma = bands(pool, fs.fh.mi_rows as usize * 4);
72    let planes: Vec<usize> = (0..fs.num_planes).filter(|&p| p == 0 || fs.fh.lf.level[1 + p] != 0).collect();
73    let mut cur = std::mem::take(&mut fs.cur);
74    let f: &FrameState = fs;
75    // Unfiltered copies of the rows above each band but the first.
76    let margins: Vec<[Vec<u16>; 3]> = luma
77        .iter()
78        .map(|&(start, _)| {
79            std::array::from_fn(|p| {
80                let sub_y = if p == 0 { 0 } else { f.ssy };
81                let pl = &cur.planes[p];
82                let (y0, y1) = ((start.saturating_sub(LF_MARGIN)) >> sub_y, start >> sub_y);
83                let rows = pl.data.get(y0 * pl.stride..y1 * pl.stride);
84                if start == 0 || !planes.contains(&p) { Vec::new() } else { rows.map(<[u16]>::to_vec).unwrap_or_default() }
85            })
86        })
87        .collect();
88    let mut bands: Vec<([RowBand; 3], [Vec<u16>; 3])> = cur.row_bands(&luma).into_iter().zip(margins).collect();
89    let changed = pool.map_mut(&mut bands, |b, (band, margin)| {
90        let mut changed: [Vec<(usize, u16)>; 3] = Default::default();
91        for &p in &planes {
92            changed[p] = filter_band(f, p, luma[b], &mut band[p], &mut margin[p]);
93        }
94        changed
95    });
96    drop(bands);
97    for (&(start, _), ch) in luma.iter().zip(&changed) {
98        for (p, list) in ch.iter().enumerate() {
99            let sub_y = if p == 0 { 0 } else { f.ssy };
100            let pl = &mut cur.planes[p];
101            let first = (start.saturating_sub(LF_MARGIN) >> sub_y) * pl.stride;
102            for &(i, v) in list {
103                if let Some(d) = pl.data.get_mut(first + i) {
104                    *d = v;
105                }
106            }
107        }
108    }
109    fs.cur = cur;
110}

112/// Both loop filter passes over the band of `plane` holding luma rows [start, end); `margin` is
113/// an unfiltered copy of the rows above the band (empty for the first band). Returns the samples
114/// of `margin` that the band's top horizontal edges changed, as (index, value).
115fn filter_band(fs: &FrameState, plane: usize, (start, end): (usize, usize), band: &mut RowBand, margin: &mut [u16]) -> Vec<(usize, u16)> {
116    let mi_rows = fs.fh.mi_rows as usize;
117    let (r0, r1) = (start / 4, mi_rows.min(end / 4));
118    let sub_y = if plane == 0 { 0 } else { fs.ssy };
119    let edges = |band: &mut RowBand, mut above: Option<&mut RowBand>, pass: usize, rows: std::ops::Range<usize>| {
120        let row_step = if plane == 0 { 1 } else { 1 << fs.ssy };
121        let col_step = if plane == 0 { 1 } else { 1 << fs.ssx };
122        let mi_cols = fs.fh.mi_cols as usize;
123        let mut row = rows.start;
124        while row < rows.end {
125            let mut col = 0;
126            while col < mi_cols {
127                edge(fs, band, above.as_deref_mut(), plane, pass, row, col);
128                col += col_step;
129            }
130            row += row_step;
131        }
132    };
133    edges(band, None, 0, r0..r1);
134    if margin.is_empty() {
135        edges(band, None, 1, r0..r1);
136        return Vec::new();
137    }
138    let stride = band.stride;
139    let y0 = start.saturating_sub(LF_MARGIN) >> sub_y;
140    let mut above = RowBand { data: margin, stride, y0 };
141    edges(&mut above, None, 0, r0.saturating_sub(LF_MARGIN / 4)..r0);
142    let before = above.data.to_vec();
143    edges(band, Some(&mut above), 1, r0..r1);
144    above.data.iter().zip(&before).enumerate().filter(|(_, (a, b))| a != b).map(|(i, (&a, _))| (i, a)).collect()
145}

147/// One loop filter edge (7.14.2) at mode info position (row, col): 4 lines of samples across
148/// it. `above` holds the rows above `band` for horizontal edges on the band's top row.
149fn edge(fs: &FrameState, band: &mut RowBand, above: Option<&mut RowBand>, plane: usize, pass: usize, row: usize, col: usize) {
150    let (sub_x, sub_y) = if plane == 0 { (0, 0) } else { (fs.ssx, fs.ssy) };
151    let (dx, dy) = if pass == 0 { (1i32, 0i32) } else { (0, 1) };
152    let x = col * 4;
153    let y = row * 4;
154    let row = row | sub_y;
155    let col = col | sub_x;
156    let on_screen = !(x >= fs.fh.frame_width as usize || y >= fs.fh.frame_height as usize || (pass == 0 && x == 0) || (pass == 1 && y == 0));
157    if !on_screen {
158        return;
159    }
160    let xp = x >> sub_x;
161    let yp = y >> sub_y;
162    let prev_row = row - ((dy as usize) << sub_y);
163    let prev_col = col - ((dx as usize) << sub_x);
164    let tx_sz = fs.lf_tx_size(plane, row >> sub_y, col >> sub_x);
165    let is_tx_edge = if pass == 0 { xp.is_multiple_of(TX_WIDTH[tx_sz] as usize) } else { yp.is_multiple_of(TX_HEIGHT[tx_sz] as usize) };
166    if !is_tx_edge {
167        return;
168    }
169    let mi = &fs.mi;
170    let i = mi.idx(row, col);
171    let mi_size = mi.mi_size[i] as usize;
172    let plane_size = fs.plane_residual_size(mi_size, plane);
173    let skip = mi.skip[i];
174    let is_intra = mi.ref_frame[i][0] <= INTRA_FRAME as i8;
175    let is_block_edge = if pass == 0 {
176        xp.is_multiple_of(4 * NUM_4X4_BLOCKS_WIDE[plane_size] as usize)
177    } else {
178        yp.is_multiple_of(4 * NUM_4X4_BLOCKS_HIGH[plane_size] as usize)
179    };
180    if !(is_block_edge || !skip || is_intra) {
181        return;
182    }
183    let prev_tx_sz = fs.lf_tx_size(plane, prev_row >> sub_y, prev_col >> sub_x);
184    // filter size (7.14.3)
185    let base_size = if pass == 0 { TX_WIDTH[prev_tx_sz].min(TX_WIDTH[tx_sz]) } else { TX_HEIGHT[prev_tx_sz].min(TX_HEIGHT[tx_sz]) } as usize;
186    let filter_size = if plane == 0 { 16.min(base_size) } else { 8.min(base_size) };
187    let (mut lvl, mut limit, mut blimit, mut thresh) = strength(fs, row, col, plane, pass);
188    if lvl == 0 {
189        (lvl, limit, blimit, thresh) = strength(fs, prev_row, prev_col, plane, pass);
190    }
191    if lvl == 0 {
192        return;
193    }
194    let t = LfThresh::new(limit, blimit, thresh, fs.bit_depth);
195    let chroma = plane > 0;
196    let stride = band.stride;
197    if let Some(above) = above.filter(|_| pass == 1 && yp == band.y0) {
198        // The p side is in the rows above the band: filter each line in a copy that spans both.
199        let (n_above, n_band) = (above.data.len() / stride.max(1), band.data.len() / stride.max(1));
200        for k in 0..4 {
201            let x = xp + k;
202            let mut line = [0u16; 2 * LF_LINE];
203            for j in 1..=n_above.min(LF_LINE) {
204                line[LF_LINE - j] = above.data[(n_above - j) * stride + x];
205            }
206            for j in 0..n_band.min(LF_LINE) {
207                line[LF_LINE + j] = band.data[j * stride + x];
208            }
209            sample_filter(&mut line, LF_LINE, 1, chroma, filter_size, t);
210            for j in 1..=n_above.min(LF_LINE) {
211                above.data[(n_above - j) * stride + x] = line[LF_LINE - j];
212            }
213            for j in 0..n_band.min(LF_LINE) {
214                band.data[j * stride + x] = line[LF_LINE + j];
215            }
216        }
217        return;
218    }
219    let (step, along) = if pass == 0 { (1, stride) } else { (stride, 1) };
220    let base = (yp - band.y0) * stride + xp;
221    for k in 0..4 {
222        sample_filter(band.data, base + k * along, step, chroma, filter_size, t);
223    }
224}

226/// Samples on each side of an edge that the sample filter can read (at most p6 .. q6).
227const LF_LINE: usize = 8;

229/// Adaptive filter strength (7.14.4): (lvl, limit, blimit, thresh).
230fn strength(fs: &FrameState, row: usize, col: usize, plane: usize, pass: usize) -> (i32, i32, i32, i32) {
231    let mi = &fs.mi;
232    let i = mi.idx(row, col);
233    let segment = mi.segment_id[i] as usize;
234    let rf = mi.ref_frame[i][0];
235    let mode = mi.y_mode[i] as usize;
236    let mode_type = (mode >= NEARESTMV && mode != GLOBALMV && mode != GLOBAL_GLOBALMV) as usize;
237    let delta_lf = if fs.fh.delta_lf_multi { mi.delta_lf[i][if plane == 0 { pass } else { plane + 1 }] } else { mi.delta_lf[i][0] } as i32;
238    // 7.14.5
239    let idx = if plane == 0 { pass } else { plane + 1 };
240    let base = (delta_lf + fs.fh.lf.level[idx] as i32).clamp(0, MAX_LOOP_FILTER as i32);
241    let mut lvl_seg = base;
242    let feature = SEG_LVL_ALT_LF_Y_V + idx;
243    if fs.fh.seg.enabled && fs.fh.seg.features.enabled[segment][feature] {
244        lvl_seg = (fs.fh.seg.features.data[segment][feature] + lvl_seg).clamp(0, MAX_LOOP_FILTER as i32);
245    }
246    if fs.fh.lf.delta_enabled {
247        let n_shift = lvl_seg >> 5;
248        let d = &fs.fh.lf.deltas;
249        if rf == INTRA_FRAME as i8 {
250            lvl_seg += d.ref_deltas[INTRA_FRAME] << n_shift;
251        } else if rf > INTRA_FRAME as i8 {
252            lvl_seg += (d.ref_deltas[rf as usize] << n_shift) + (d.mode_deltas[mode_type] << n_shift);
253        }
254        lvl_seg = lvl_seg.clamp(0, MAX_LOOP_FILTER as i32);
255    }
256    let lvl = lvl_seg;
257    let sharp = fs.fh.lf.sharpness as i32;
258    let shift = if sharp > 4 {
259        2
260    } else if sharp > 0 {
261        1
262    } else {
263        0
264    };
265    let limit = if sharp > 0 { (lvl >> shift).clamp(1, 9 - sharp) } else { (lvl >> shift).max(1) };
266    let blimit = 2 * (lvl + 2) + limit;
267    let thresh = lvl >> 4;
268    (lvl, limit, blimit, thresh)
269}

271/// Per-edge thresholds of the sample filter, scaled to the bit depth.
272#[derive(Clone, Copy)]
273struct LfThresh {
274    limit: i32,
275    blimit: i32,
276    thresh: i32,
277    /// Flatness threshold 1 << (BitDepth - 8).
278    flat: i32,
279    /// 0x80 << (BitDepth - 8) and 1 << (BitDepth - 1).
280    off: i32,
281    half: i32,
282}
283
284impl LfThresh {
285    fn new(limit: i32, blimit: i32, thresh: i32, bd: u32) -> LfThresh {
286        let sh = bd - 8;
287        LfThresh { limit: limit << sh, blimit: blimit << sh, thresh: thresh << sh, flat: 1 << sh, off: 0x80 << sh, half: 1 << (bd - 1) }
288    }
289}

291/// Sample filtering process (7.14.6) for one line across the edge: `base` indexes q0 and
292/// `step` is the distance between samples across the edge.
293#[inline(always)]
294fn sample_filter(d: &mut [u16], base: usize, step: usize, chroma: bool, filter_size: usize, t: LfThresh) {
295    let at = |d: &[u16], k: isize| -> i32 { d[(base as isize + k * step as isize) as usize] as i32 };
296    let (q0, q1, p0, p1) = (at(d, 0), at(d, 1), at(d, -1), at(d, -2));
297    let lim = t.limit;
298    // filter mask (7.14.6.2)
299    if (p1 - p0).abs() > lim || (q1 - q0).abs() > lim || (p0 - q0).abs() * 2 + (p1 - q1).abs() / 2 > t.blimit {
300        return;
301    }
302    let hev = (p1 - p0).abs() > t.thresh || (q1 - q0).abs() > t.thresh;
303    let mut flat = false;
304    if filter_size > 4 {
305        let (q2, p2) = (at(d, 2), at(d, -3));
306        if (p2 - p1).abs() > lim || (q2 - q1).abs() > lim {
307            return;
308        }
309        let tf = t.flat;
310        flat = (p1 - p0).abs() <= tf && (q1 - q0).abs() <= tf && (p2 - p0).abs() <= tf && (q2 - q0).abs() <= tf;
311        if !chroma {
312            // filterLen >= 8
313            let (q3, p3) = (at(d, 3), at(d, -4));
314            if (p3 - p2).abs() > lim || (q3 - q2).abs() > lim {
315                return;
316            }
317            flat = flat && (p3 - p0).abs() <= tf && (q3 - q0).abs() <= tf;
318        }
319    }
320    if !flat {
321        // narrow filter (7.14.6.3)
322        let c = |v: i32| v.clamp(-t.half, t.half - 1);
323        let off = t.off;
324        let (ps1, ps0, qs0, qs1) = (p1 - off, p0 - off, q0 - off, q1 - off);
325        let mut filter = if hev { c(ps1 - qs1) } else { 0 };
326        filter = c(filter + 3 * (qs0 - ps0));
327        let filter1 = c(filter + 4) >> 3;
328        let filter2 = c(filter + 3) >> 3;
329        d[base] = (c(qs0 - filter1) + off) as u16;
330        d[base - step] = (c(ps0 + filter2) + off) as u16;
331        if !hev {
332            let f = (filter1 + 1) >> 1;
333            d[base + step] = (c(qs1 - f) + off) as u16;
334            d[base - 2 * step] = (c(ps1 + f) + off) as u16;
335        }
336        return;
337    }
338    if chroma {
339        wide::<2, 1, 3>(d, base, step);
340    } else if filter_size == 8 {
341        wide::<3, 0, 3>(d, base, step);
342    } else {
343        let tf = t.flat;
344        let q0 = at(d, 0);
345        let p0 = at(d, -1);
346        let flat2 = (4..7).all(|k| (at(d, k) - q0).abs() <= tf && (at(d, -k - 1) - p0).abs() <= tf);
347        if flat2 { wide::<6, 1, 4>(d, base, step) } else { wide::<3, 0, 3>(d, base, step) }
348    }
349}

351/// Wide filter process (7.14.6.4) with n = N, the 2-weight taps |j| <= N2 and log2Size = LOG2.
352#[inline(always)]
353fn wide<const N: usize, const N2: usize, const LOG2: u32>(d: &mut [u16], base: usize, step: usize) {
354    // v[k] = sample at offset k - (N + 1) from q0, k in 0 .. 2N + 2
355    let mut v = [0i32; 14];
356    let first = base - (N + 1) * step;
357    for (k, x) in v[..2 * N + 2].iter_mut().enumerate() {
358        *x = d[first + k * step] as i32;
359    }
360    // e[m] = v[clamp(m - N, 0, 2N + 1)]: the clamped tap positions as one array, so
361    // F[i] = sum_{jj = 0}^{2N} e[i + 1 + jj] * (2 if |jj - N| <= N2 else 1), a sliding window.
362    let mut e = [0i32; 25];
363    for (m, x) in e[..4 * N + 1].iter_mut().enumerate() {
364        *x = v[m.saturating_sub(N).min(2 * N + 1)];
365    }
366    let mut f = [0i32; 12];
367    let mut win: i32 = e[1..2 * N + 2].iter().sum();
368    for i in 0..2 * N {
369        if i > 0 {
370            win += e[i + 1 + 2 * N] - e[i];
371        }
372        let centre: i32 = e[i + 1 + N - N2..=i + 1 + N + N2].iter().sum();
373        f[i] = (win + centre + (1 << (LOG2 - 1))) >> LOG2;
374    }
375    let out = base - N * step;
376    for i in 0..2 * N {
377        d[out + i * step] = f[i] as u16;
378    }
