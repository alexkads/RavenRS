38fn block_width(bs: usize) -> i32 {
39    4 * NUM_4X4_BLOCKS_WIDE[bs] as i32

45#[inline]
46pub(crate) fn round2_signed64(x: i64, n: u32) -> i64 {
47    if n == 0 {
48        return x;
49    }
50    if x >= 0 { (x + (1 << (n - 1))) >> n } else { -((-x + (1 << (n - 1))) >> n) }
51}

53pub(crate) fn has_newmv(mode: usize) -> bool {
54    matches!(mode, NEWMV | NEW_NEWMV | NEAR_NEWMV | NEW_NEARMV | NEAREST_NEWMV | NEW_NEARESTMV)
55}

58    /// lower_mv_precision( candMv ) (7.10.2.10)
59    pub(crate) fn lower_mv_precision(&self, mv: &mut MvP) {
60        let fh = &self.fs.fh;
61        if fh.allow_high_precision_mv {
62            return;
63        }
64        for c in mv.iter_mut() {
65            if fh.force_integer_mv {
66                let a = c.abs();
67                let a_int = (a + 3) >> 3;
68                *c = if *c > 0 { a_int << 3 } else { -(a_int << 3) };
69            } else if *c & 1 != 0 {
70                *c += if *c > 0 { -1 } else { 1 };
71            }
72        }
73    }

75    /// Setup global MV process (7.10.2.1).
76    pub(crate) fn setup_global_mv(&self, ref_list: usize) -> MvP {
77        let rf = self.b.ref_frame[ref_list];
78        let fh = &self.fs.fh;
79        let mut mv: MvP = [0, 0];
80        if rf > INTRA_FRAME as i8 {
81            let rf = rf as usize;
82            let typ = fh.gm_type[rf] as usize;
83            let gm = &fh.gm_params[rf];
84            if typ == TRANSLATION {
85                mv = [gm[0] >> (WARPEDMODEL_PREC_BITS - 3), gm[1] >> (WARPEDMODEL_PREC_BITS - 3)];
86            } else if typ != IDENTITY {
87                let bw = block_width(self.b.mi_size) as i64;
88                let bh = block_height(self.b.mi_size) as i64;
89                let x = self.b.mi_col as i64 * 4 + bw / 2 - 1;
90                let y = self.b.mi_row as i64 * 4 + bh / 2 - 1;
91                let p = 1i64 << WARPEDMODEL_PREC_BITS;
92                let xc = (gm[2] as i64 - p) * x + gm[3] as i64 * y + gm[0] as i64;
93                let yc = gm[4] as i64 * x + (gm[5] as i64 - p) * y + gm[1] as i64;
94                if fh.allow_high_precision_mv {
95                    mv = [round2_signed64(yc, WARPEDMODEL_PREC_BITS as u32 - 3) as i32, round2_signed64(xc, WARPEDMODEL_PREC_BITS as u32 - 3) as i32];
96                } else {
97                    mv = [
98                        round2_signed64(yc, WARPEDMODEL_PREC_BITS as u32 - 2) as i32 * 2,
99                        round2_signed64(xc, WARPEDMODEL_PREC_BITS as u32 - 2) as i32 * 2,
100                    ];
101                }
102            }
103        }
104        self.lower_mv_precision(&mut mv);
105        mv
106    }

108    /// Find MV stack process (7.10.2).
109    pub(crate) fn find_mv_stack(&mut self, is_compound: bool) {
110        let bw4 = self.b.bw4 as isize;
111        let bh4 = self.b.bh4 as isize;
112        let mut s = MvStack { global_mvs: [self.setup_global_mv(0), [0, 0]], ..Default::default() };
113        if is_compound {
114            s.global_mvs[1] = self.setup_global_mv(1);
115        }
116        self.mvs = s;
117        self.mvs.found_match = false;
118        self.scan_row(-1, is_compound);
119        let mut found_above = self.mvs.found_match;
120        self.mvs.found_match = false;
121        self.scan_col(-1, is_compound);
122        let mut found_left = self.mvs.found_match;
123        self.mvs.found_match = false;
124        if bw4.max(bh4) <= 16 {
125            self.scan_point(-1, bw4, is_compound);
126        }
127        if self.mvs.found_match {
128            found_above = true;
129        }
130        self.mvs.close_matches = found_above as usize + found_left as usize;
131        let num_nearest = self.mvs.num_mv_found;
132        let num_new = self.mvs.new_mv_count;
133        for idx in 0..num_nearest {
134            self.mvs.weight_stack[idx] += REF_CAT_LEVEL as u32;
135        }
136        self.mvs.zero_mv_context = 0;
137        if self.fs.fh.use_ref_frame_mvs {
138            self.temporal_scan(is_compound);
139        }
140        self.scan_point(-1, -1, is_compound);
141        if self.mvs.found_match {
142            found_above = true;
143        }
144        self.mvs.found_match = false;
145        self.scan_row(-3, is_compound);
146        if self.mvs.found_match {
147            found_above = true;
148        }
149        self.mvs.found_match = false;
150        self.scan_col(-3, is_compound);
151        if self.mvs.found_match {
152            found_left = true;
153        }
154        self.mvs.found_match = false;
155        if bh4 > 1 {
156            self.scan_row(-5, is_compound);
157        }
158        if self.mvs.found_match {
159            found_above = true;
160        }
161        self.mvs.found_match = false;
162        if bw4 > 1 {
163            self.scan_col(-5, is_compound);
164        }
165        if self.mvs.found_match {
166            found_left = true;
167        }
168        self.mvs.total_matches = found_above as usize + found_left as usize;
169        let n = self.mvs.num_mv_found;
170        self.sort_stack(0, num_nearest, is_compound);
171        self.sort_stack(num_nearest, n, is_compound);
172        if self.mvs.num_mv_found < 2 {
173            self.extra_search(is_compound);
174        }
175        self.context_and_clamping(is_compound, num_new);
176    }

178    fn scan_row(&mut self, delta_row: isize, is_compound: bool) {
179        let bw4 = self.b.bw4 as isize;
180        let (mi_row, mi_col) = (self.b.mi_row as isize, self.b.mi_col as isize);
181        let mi_cols = self.fs.fh.mi_cols as isize;
182        let end4 = bw4.min(mi_cols - mi_col).min(16);
183        let mut delta_col = 0;
184        let use_step16 = bw4 >= 16;
185        let mut delta_row = delta_row;
186        if delta_row.abs() > 1 {
187            delta_row += mi_row & 1;
188            delta_col = 1 - (mi_col & 1);
189        }
190        let mut i = 0;
191        while i < end4 {
192            let mv_row = mi_row + delta_row;
193            let mv_col = mi_col + delta_col + i;
194            if !self.inside(mv_row, mv_col) {
195                break;
196            }
197            let mi = &self.t.mi;
198            let mut len = bw4.min(NUM_4X4_BLOCKS_WIDE[mi.mi_size[mi.idx(mv_row as usize, mv_col as usize)] as usize] as isize);
199            if delta_row.abs() > 1 {
200                len = len.max(2);
201            }
202            if use_step16 {
203                len = len.max(4);
204            }
205            let weight = len as u32 * 2;
206            self.add_ref_mv_candidate(mv_row as usize, mv_col as usize, is_compound, weight);
207            i += len;
208        }
209    }

211    fn scan_col(&mut self, delta_col: isize, is_compound: bool) {
212        let bh4 = self.b.bh4 as isize;
213        let (mi_row, mi_col) = (self.b.mi_row as isize, self.b.mi_col as isize);
214        let mi_rows = self.fs.fh.mi_rows as isize;
215        let end4 = bh4.min(mi_rows - mi_row).min(16);
216        let mut delta_row = 0;
217        let use_step16 = bh4 >= 16;
218        let mut delta_col = delta_col;
219        if delta_col.abs() > 1 {
220            delta_row = 1 - (mi_row & 1);
221            delta_col += mi_col & 1;
222        }
223        let mut i = 0;
224        while i < end4 {
225            let mv_row = mi_row + delta_row + i;
226            let mv_col = mi_col + delta_col;
227            if !self.inside(mv_row, mv_col) {
228                break;
229            }
230            let mi = &self.t.mi;
231            let mut len = bh4.min(NUM_4X4_BLOCKS_HIGH[mi.mi_size[mi.idx(mv_row as usize, mv_col as usize)] as usize] as isize);
232            if delta_col.abs() > 1 {
233                len = len.max(2);
234            }
235            if use_step16 {
236                len = len.max(4);
237            }
238            let weight = len as u32 * 2;
239            self.add_ref_mv_candidate(mv_row as usize, mv_col as usize, is_compound, weight);
240            i += len;
241        }
242    }

244    fn scan_point(&mut self, delta_row: isize, delta_col: isize, is_compound: bool) {
245        let mv_row = self.b.mi_row as isize + delta_row;
246        let mv_col = self.b.mi_col as isize + delta_col;
247        if self.inside(mv_row, mv_col) && self.t.mi.written[self.t.mi.idx(mv_row as usize, mv_col as usize)] {
248            self.add_ref_mv_candidate(mv_row as usize, mv_col as usize, is_compound, 4);
249        }

252    fn temporal_scan(&mut self, is_compound: bool) {
253        let bw4 = self.b.bw4 as isize;
254        let bh4 = self.b.bh4 as isize;
255        let step_w4 = if bw4 >= 16 { 4 } else { 2 };
256        let step_h4 = if bh4 >= 16 { 4 } else { 2 };
257        let mut dr = 0;
258        while dr < bh4.min(16) {
259            let mut dc = 0;
260            while dc < bw4.min(16) {
261                self.add_tpl_ref_mv(dr, dc, is_compound);
262                dc += step_w4;
263            }
264            dr += step_h4;
265        }
266        let allow_extension = (2..16).contains(&bh4) && (2..16).contains(&bw4);
267        if allow_extension {
268            let pos = [(bh4, -2), (bh4, bw4), (bh4 - 2, bw4)];
269            for &(dr, dc) in &pos {
270                let row = (self.b.mi_row as isize & 15) + dr;
271                let col = (self.b.mi_col as isize & 15) + dc;
272                if (0..16).contains(&row) && (0..16).contains(&col) {
273                    self.add_tpl_ref_mv(dr, dc, is_compound);
274                }
275            }
276        }
277    }

279    fn add_tpl_ref_mv(&mut self, delta_row: isize, delta_col: isize, is_compound: bool) {
280        let mv_row = (self.b.mi_row as isize + delta_row) | 1;
281        let mv_col = (self.b.mi_col as isize + delta_col) | 1;
282        if !self.inside(mv_row, mv_col) {
283            return;
284        }
285        let x8 = (mv_col >> 1) as usize;
286        let y8 = (mv_row >> 1) as usize;
287        let w8 = self.fs.fh.mi_cols as usize >> 1;
288        if delta_row == 0 && delta_col == 0 {
289            self.mvs.zero_mv_context = 1;
290        }
291        let rf0 = self.b.ref_frame[0] as usize;
292        if !is_compound {
293            let mut cand = self.fs.motion_field[rf0][y8 * w8 + x8];
294            if cand[0] == INVALID_MV {
295                return;
296            }
297            self.lower_mv_precision(&mut cand);
298            if delta_row == 0 && delta_col == 0 {
299                let g = self.mvs.global_mvs[0];
300                self.mvs.zero_mv_context = ((cand[0] - g[0]).abs() >= 16 || (cand[1] - g[1]).abs() >= 16) as usize;
301            }
302            let n = self.mvs.num_mv_found;
303            let idx = (0..n).find(|&i| self.mvs.ref_stack_mv[i][0] == cand);
304            if let Some(i) = idx {
305                self.mvs.weight_stack[i] += 2;
306            } else if n < MAX_REF_MV_STACK_SIZE {
307                self.mvs.ref_stack_mv[n][0] = cand;
308                self.mvs.weight_stack[n] = 2;
309                self.mvs.num_mv_found += 1;
310            }
311        } else {
312            let rf1 = self.b.ref_frame[1] as usize;
313            let mut c0 = self.fs.motion_field[rf0][y8 * w8 + x8];
314            if c0[0] == INVALID_MV {
315                return;
316            }
317            let mut c1 = self.fs.motion_field[rf1][y8 * w8 + x8];
318            if c1[0] == INVALID_MV {
319                return;
320            }
321            self.lower_mv_precision(&mut c0);
322            self.lower_mv_precision(&mut c1);
323            if delta_row == 0 && delta_col == 0 {
324                let g = self.mvs.global_mvs;
325                self.mvs.zero_mv_context = ((c0[0] - g[0][0]).abs() >= 16
326                    || (c0[1] - g[0][1]).abs() >= 16
327                    || (c1[0] - g[1][0]).abs() >= 16
328                    || (c1[1] - g[1][1]).abs() >= 16) as usize;
329            }
330            let n = self.mvs.num_mv_found;
331            let idx = (0..n).find(|&i| self.mvs.ref_stack_mv[i][0] == c0 && self.mvs.ref_stack_mv[i][1] == c1);
332            if let Some(i) = idx {
333                self.mvs.weight_stack[i] += 2;
334            } else if n < MAX_REF_MV_STACK_SIZE {
335                self.mvs.ref_stack_mv[n] = [c0, c1];
336                self.mvs.weight_stack[n] = 2;
337                self.mvs.num_mv_found += 1;
338            }
339        }
340    }

342    fn add_ref_mv_candidate(&mut self, mv_row: usize, mv_col: usize, is_compound: bool, weight: u32) {
343        let mi = &self.t.mi;
344        let i = mi.idx(mv_row, mv_col);
345        if !mi.is_inter[i] {
346            return;
347        }
348        let rf = mi.ref_frame[i];
349        if !is_compound {
350            for cand_list in 0..2 {
351                if rf[cand_list] == self.b.ref_frame[0] {
352                    self.search_stack(mv_row, mv_col, cand_list, weight);
353                }
354            }
355        } else if rf[0] == self.b.ref_frame[0] && rf[1] == self.b.ref_frame[1] {
356            self.compound_search_stack(mv_row, mv_col, weight);
357        }
358    }

360    fn search_stack(&mut self, mv_row: usize, mv_col: usize, cand_list: usize, weight: u32) {
361        let mi = &self.t.mi;
362        let i = mi.idx(mv_row, mv_col);
363        let cand_mode = mi.y_mode[i] as usize;
364        let cand_size = mi.mi_size[i] as usize;
365        let large = block_width(cand_size).min(block_height(cand_size)) >= 8;
366        let gm0 = self.fs.fh.gm_type[self.b.ref_frame[0] as usize] as usize;
367        let mut cand = if (cand_mode == GLOBALMV || cand_mode == GLOBAL_GLOBALMV) && gm0 > TRANSLATION && large {
368            self.mvs.global_mvs[0]
369        } else {
370            let m = mi.mv[i][cand_list];
371            [m.row as i32, m.col as i32]
372        };
373        self.lower_mv_precision(&mut cand);
374        if has_newmv(cand_mode) {
375            self.mvs.new_mv_count += 1;
376        }
377        self.mvs.found_match = true;
378        let n = self.mvs.num_mv_found;
379        if let Some(idx) = (0..n).find(|&k| self.mvs.ref_stack_mv[k][0] == cand) {
380            self.mvs.weight_stack[idx] += weight;
381        } else if n < MAX_REF_MV_STACK_SIZE {
382            self.mvs.ref_stack_mv[n][0] = cand;
383            self.mvs.weight_stack[n] = weight;
384            self.mvs.num_mv_found += 1;
385        }
386    }

388    fn compound_search_stack(&mut self, mv_row: usize, mv_col: usize, weight: u32) {
389        let mi = &self.t.mi;
390        let i = mi.idx(mv_row, mv_col);
391        let mut cand = [[mi.mv[i][0].row as i32, mi.mv[i][0].col as i32], [mi.mv[i][1].row as i32, mi.mv[i][1].col as i32]];
392        let cand_mode = mi.y_mode[i] as usize;
393        if cand_mode == GLOBAL_GLOBALMV {
394            for (rl, c) in cand.iter_mut().enumerate() {
395                if self.fs.fh.gm_type[self.b.ref_frame[rl] as usize] as usize > TRANSLATION {
396                    *c = self.mvs.global_mvs[rl];
397                }
398            }
399        }
400        for c in cand.iter_mut() {
401            self.lower_mv_precision(c);
402        }
403        self.mvs.found_match = true;
404        let n = self.mvs.num_mv_found;
405        if let Some(idx) = (0..n).find(|&k| self.mvs.ref_stack_mv[k][0] == cand[0] && self.mvs.ref_stack_mv[k][1] == cand[1]) {
406            self.mvs.weight_stack[idx] += weight;
407        } else if n < MAX_REF_MV_STACK_SIZE {
408            self.mvs.ref_stack_mv[n] = cand;
409            self.mvs.weight_stack[n] = weight;
410            self.mvs.num_mv_found += 1;
411        }
412        if has_newmv(cand_mode) {
413            self.mvs.new_mv_count += 1;
414        }
415    }

417    fn sort_stack(&mut self, start: usize, end: usize, is_compound: bool) {
418        let mut end = end;
419        let s = &mut self.mvs;
420        while end > start {
421            let mut new_end = start;
422            for idx in start + 1..end {
423                if s.weight_stack[idx - 1] < s.weight_stack[idx] {
424                    s.weight_stack.swap(idx - 1, idx);
425                    for list in 0..1 + is_compound as usize {
426                        let t = s.ref_stack_mv[idx - 1][list];
427                        s.ref_stack_mv[idx - 1][list] = s.ref_stack_mv[idx][list];
428                        s.ref_stack_mv[idx][list] = t;
429                    }
430                    new_end = idx;
431                }
432            }
433            end = new_end;
434        }
435    }

437    fn extra_search(&mut self, is_compound: bool) {
438        self.mvs.ref_id_count = [0; 2];
439        self.mvs.ref_diff_count = [0; 2];
440        let (mi_row, mi_col) = (self.b.mi_row as isize, self.b.mi_col as isize);
441        let mut w4 = (self.b.bw4 as isize).min(16);
442        let mut h4 = (self.b.bh4 as isize).min(16);
443        w4 = w4.min(self.fs.fh.mi_cols as isize - mi_col);
444        h4 = h4.min(self.fs.fh.mi_rows as isize - mi_row);
445        let num4x4 = w4.min(h4);
446        for pass in 0..2 {
447            let mut idx = 0;
448            while idx < num4x4 && self.mvs.num_mv_found < 2 {
449                let (mv_row, mv_col) = if pass == 0 { (mi_row - 1, mi_col + idx) } else { (mi_row + idx, mi_col - 1) };
450                if !self.inside(mv_row, mv_col) {
451                    break;
452                }
453                self.add_extra_mv_candidate(mv_row as usize, mv_col as usize, is_compound);
454                let mi = &self.t.mi;
455                let sz = mi.mi_size[mi.idx(mv_row as usize, mv_col as usize)] as usize;
456                idx += if pass == 0 { NUM_4X4_BLOCKS_WIDE[sz] } else { NUM_4X4_BLOCKS_HIGH[sz] } as isize;
457            }
458        }
459        if is_compound {
460            let mut combined = [[[0i32; 2]; 2]; 2];
461            for list in 0..2 {
462                let mut comp_count = 0;
463                for idx in 0..self.mvs.ref_id_count[list] {
464                    combined[comp_count][list] = self.mvs.ref_id_mvs[list][idx];
465                    comp_count += 1;
466                }
467                let mut idx = 0;
468                while idx < self.mvs.ref_diff_count[list] && comp_count < 2 {
469                    combined[comp_count][list] = self.mvs.ref_diff_mvs[list][idx];
470                    comp_count += 1;
471                    idx += 1;
472                }
473                while comp_count < 2 {
474                    combined[comp_count][list] = self.mvs.global_mvs[list];
475                    comp_count += 1;
476                }
477            }
478            let n = self.mvs.num_mv_found;
479            if n == 1 {
480                if combined[0][0] == self.mvs.ref_stack_mv[0][0] && combined[0][1] == self.mvs.ref_stack_mv[0][1] {
481                    self.mvs.ref_stack_mv[n] = combined[1];
482                } else {
483                    self.mvs.ref_stack_mv[n] = combined[0];
484                }
485                self.mvs.weight_stack[n] = 2;
486                self.mvs.num_mv_found += 1;
487            } else {
488                for c in combined {
489                    let n = self.mvs.num_mv_found;
490                    self.mvs.ref_stack_mv[n] = c;
491                    self.mvs.weight_stack[n] = 2;
492                    self.mvs.num_mv_found += 1;
493                }
494            }
495        } else {
496            for idx in self.mvs.num_mv_found..2 {
497                self.mvs.ref_stack_mv[idx][0] = self.mvs.global_mvs[0];
498            }
499        }
500    }

502    fn add_extra_mv_candidate(&mut self, mv_row: usize, mv_col: usize, is_compound: bool) {
503        let mi = &self.t.mi;
504        let i = mi.idx(mv_row, mv_col);
505        let rfs = mi.ref_frame[i];
506        let mvs = mi.mv[i];
507        let bias = &self.fs.fh.ref_frame_sign_bias;
508        if is_compound {
509            for cand_list in 0..2 {
510                let cand_ref = rfs[cand_list];
511                if cand_ref > INTRA_FRAME as i8 {
512                    for list in 0..2 {
513                        let mut cand = [mvs[cand_list].row as i32, mvs[cand_list].col as i32];
514                        if cand_ref == self.b.ref_frame[list] && self.mvs.ref_id_count[list] < 2 {
515                            let k = self.mvs.ref_id_count[list];
516                            self.mvs.ref_id_mvs[list][k] = cand;
517                            self.mvs.ref_id_count[list] += 1;
518                        } else if self.mvs.ref_diff_count[list] < 2 {
519                            if bias[cand_ref as usize] != bias[self.b.ref_frame[list] as usize] {
520                                cand = [-cand[0], -cand[1]];
521                            }
522                            let k = self.mvs.ref_diff_count[list];
523                            self.mvs.ref_diff_mvs[list][k] = cand;
524                            self.mvs.ref_diff_count[list] += 1;
525                        }
526                    }
527                }
528            }
529        } else {
530            for cand_list in 0..2 {
531                let cand_ref = rfs[cand_list];
532                if cand_ref > INTRA_FRAME as i8 {
533                    let mut cand = [mvs[cand_list].row as i32, mvs[cand_list].col as i32];
534                    if bias[cand_ref as usize] != bias[self.b.ref_frame[0] as usize] {
535                        cand = [-cand[0], -cand[1]];
536                    }
537                    let n = self.mvs.num_mv_found;
538                    if !(0..n).any(|k| self.mvs.ref_stack_mv[k][0] == cand) {
539                        self.mvs.ref_stack_mv[n][0] = cand;
540                        self.mvs.weight_stack[n] = 2;
541                        self.mvs.num_mv_found += 1;
542                    }
543                }
544            }
545        }
546    }

548    fn context_and_clamping(&mut self, is_compound: bool, num_new: usize) {
549        let bw = block_width(self.b.mi_size);
550        let bh = block_height(self.b.mi_size);
551        let n = self.mvs.num_mv_found;
552        for idx in 0..n {
553            let mut z = 0;
554            if idx + 1 < n {
555                let w0 = self.mvs.weight_stack[idx];
556                let w1 = self.mvs.weight_stack[idx + 1];
557                if w0 >= REF_CAT_LEVEL as u32 {
558                    if w1 < REF_CAT_LEVEL as u32 {
559                        z = 1;
560                    }
561                } else {
562                    z = 2;
563                }
564            }
565            self.mvs.drl_ctx_stack[idx] = z;
566        }
567        for list in 0..1 + is_compound as usize {
568            for idx in 0..n {
569                let mut mv = self.mvs.ref_stack_mv[idx][list];
570                mv[0] = self.clamp_mv_row(mv[0], MV_BORDER as i32 + bh * 8);
571                mv[1] = self.clamp_mv_col(mv[1], MV_BORDER as i32 + bw * 8);
572                self.mvs.ref_stack_mv[idx][list] = mv;
573            }
574        }
575        let (close, total) = (self.mvs.close_matches, self.mvs.total_matches);
576        if close == 0 {
577            self.mvs.new_mv_context = total.min(1);
578            self.mvs.ref_mv_context = total;
579        } else if close == 1 {
580            self.mvs.new_mv_context = 3 - num_new.min(1);
581            self.mvs.ref_mv_context = 2 + total;
582        } else {
583            self.mvs.new_mv_context = 5 - num_new.min(1);
584            self.mvs.ref_mv_context = 5;
585        }
586    }

588    pub(crate) fn clamp_mv_row(&self, mvec: i32, border: i32) -> i32 {
589        let bh4 = self.b.bh4 as i32;
590        let to_top = -((self.b.mi_row as i32 * 4) * 8);
591        let to_bottom = ((self.fs.fh.mi_rows as i32 - bh4 - self.b.mi_row as i32) * 4) * 8;
592        mvec.clamp(to_top - border, to_bottom + border)
593    }

596        let bw4 = self.b.bw4 as i32;
597        let to_left = -((self.b.mi_col as i32 * 4) * 8);
598        let to_right = ((self.fs.fh.mi_cols as i32 - bw4 - self.b.mi_col as i32) * 4) * 8;
599        mvec.clamp(to_left - border, to_right + border)
600    }

602    /// has_overlappable_candidates( ) (7.10.3)
603    pub(crate) fn has_overlappable_candidates(&self) -> bool {
604        let mi = &self.t.mi;
605        let (mi_rows, mi_cols) = (self.fs.fh.mi_rows as usize, self.fs.fh.mi_cols as usize);
606        if self.b.avail_u {
607            let mut x4 = self.b.mi_col;
608            while x4 < mi_cols.min(self.b.mi_col + self.b.bw4) {
609                let c = (x4 | 1).min(mi_cols - 1);
610                if mi.ref_frame[mi.idx(self.b.mi_row - 1, c)][0] > INTRA_FRAME as i8 {
611                    return true;
612                }
613                x4 += 2;
614            }
615        }
616        if self.b.avail_l {
617            let mut y4 = self.b.mi_row;
618            while y4 < mi_rows.min(self.b.mi_row + self.b.bh4) {
619                let r = (y4 | 1).min(mi_rows - 1);
620                if mi.ref_frame[mi.idx(r, self.b.mi_col - 1)][0] > INTRA_FRAME as i8 {
621                    return true;
622                }
623                y4 += 2;
624            }
625        }
626        false
627    }

629    /// Find warp samples process (7.10.4).
630    pub(crate) fn find_warp_samples(&mut self) {
631        self.mvs.num_samples = 0;
632        self.mvs.num_samples_scanned = 0;
633        let w4 = self.b.bw4 as isize;
634        let h4 = self.b.bh4 as isize;
635        let (mi_row, mi_col) = (self.b.mi_row as isize, self.b.mi_col as isize);
636        let (mi_rows, mi_cols) = (self.fs.fh.mi_rows as isize, self.fs.fh.mi_cols as isize);
637        let mut do_top_left = true;
638        let mut do_top_right = true;
639        if self.b.avail_u {
640            let mi = &self.t.mi;
641            let src_w = NUM_4X4_BLOCKS_WIDE[mi.mi_size[mi.idx(mi_row as usize - 1, mi_col as usize)] as usize] as isize;
642            if w4 <= src_w {
643                let col_offset = -(mi_col & (src_w - 1));
644                if col_offset < 0 {
645                    do_top_left = false;
646                }
647                if col_offset + src_w > w4 {
648                    do_top_right = false;
649                }
650                self.add_sample(-1, 0);
651            } else {
652                let mut i = 0;
653                while i < w4.min(mi_cols - mi_col) {
654                    let mi = &self.t.mi;
655                    let src_w = NUM_4X4_BLOCKS_WIDE[mi.mi_size[mi.idx(mi_row as usize - 1, (mi_col + i) as usize)] as usize] as isize;
656                    let step = w4.min(src_w);
657                    self.add_sample(-1, i);
658                    i += step;
659                }
660            }
661        }
662        if self.b.avail_l {
663            let mi = &self.t.mi;
664            let src_h = NUM_4X4_BLOCKS_HIGH[mi.mi_size[mi.idx(mi_row as usize, mi_col as usize - 1)] as usize] as isize;
665            if h4 <= src_h {
666                let row_offset = -(mi_row & (src_h - 1));
667                if row_offset < 0 {
668                    do_top_left = false;
669                }
670                self.add_sample(0, -1);
671            } else {
672                let mut i = 0;
673                while i < h4.min(mi_rows - mi_row) {
674                    let mi = &self.t.mi;
675                    let src_h = NUM_4X4_BLOCKS_HIGH[mi.mi_size[mi.idx((mi_row + i) as usize, mi_col as usize - 1)] as usize] as isize;
676                    let step = h4.min(src_h);
677                    self.add_sample(i, -1);
678                    i += step;
679                }
680            }
681        }
682        if do_top_left {
683            self.add_sample(-1, -1);
684        }
685        if do_top_right && w4.max(h4) <= 16 {
686            self.add_sample(-1, w4);
687        }
688        if self.mvs.num_samples == 0 && self.mvs.num_samples_scanned > 0 {
689            self.mvs.num_samples = 1;
690        }
691    }

693    fn add_sample(&mut self, delta_row: isize, delta_col: isize) {
694        if self.mvs.num_samples_scanned >= LEAST_SQUARES_SAMPLES_MAX {
695            return;
696        }
697        let mv_row = self.b.mi_row as isize + delta_row;
698        let mv_col = self.b.mi_col as isize + delta_col;
699        if !self.inside(mv_row, mv_col) {
700            return;
701        }
702        let mi = &self.t.mi;
703        let i = mi.idx(mv_row as usize, mv_col as usize);
704        if !mi.written[i] {
705            return;
706        }
707        if mi.ref_frame[i][0] != self.b.ref_frame[0] || mi.ref_frame[i][1] != -1 {
708            return;
709        }
710        let cand_sz = mi.mi_size[i] as usize;
711        let cand_w4 = NUM_4X4_BLOCKS_WIDE[cand_sz] as isize;
712        let cand_h4 = NUM_4X4_BLOCKS_HIGH[cand_sz] as isize;
713        let cand_row = mv_row & !(cand_h4 - 1);
714        let cand_col = mv_col & !(cand_w4 - 1);
715        let mid_y = cand_row * 4 + cand_h4 * 2 - 1;
716        let mid_x = cand_col * 4 + cand_w4 * 2 - 1;
717        let threshold = block_width(self.b.mi_size).max(block_height(self.b.mi_size)).clamp(16, 112);
718        let cm = mi.mv[mi.idx(cand_row as usize, cand_col as usize)][0];
719        let mv_diff_row = (cm.row as i32 - self.b.mv_i[0][0]).abs();
720        let mv_diff_col = (cm.col as i32 - self.b.mv_i[0][1]).abs();
721        let valid = mv_diff_row + mv_diff_col <= threshold;
722        let cand = [mid_y as i32 * 8, mid_x as i32 * 8, mid_y as i32 * 8 + cm.row as i32, mid_x as i32 * 8 + cm.col as i32];
723        self.mvs.num_samples_scanned += 1;
724        if !valid && self.mvs.num_samples_scanned > 1 {
725            return;
726        }
727        self.mvs.cand_list[self.mvs.num_samples] = cand;
728        if valid {
729            self.mvs.num_samples += 1;
730        }
731    }
