362/// Zeroes the adaptation counter (last element) of every innermost CDF array.
363trait ClearCount {
364    fn clear_count(&mut self);
365}
366
367impl<const N: usize> ClearCount for [u16; N] {
368    fn clear_count(&mut self) {
369        self[N - 1] = 0;
370    }
371}
372
373impl<T: ClearCount, const M: usize> ClearCount for [T; M] {
374    fn clear_count(&mut self) {
375        for e in self.iter_mut() {
376            e.clear_count();
377        }
378    }
