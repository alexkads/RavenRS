7#[inline]
8fn round2(x: i32, n: u32) -> i32 {
9    if n == 0 { x } else { (x + (1 << (n - 1))) >> n }
10}
11
12struct Rng(u32);
13
14impl Rng {
15    /// get_random_number( bits ) (7.18.3.2)
16    fn next(&mut self, bits: u32) -> i32 {
17        let r = self.0;
18        let bit = (r ^ (r >> 1) ^ (r >> 3) ^ (r >> 12)) & 1;
19        let r = (r >> 1) | (bit << 15);
20        self.0 = r;
21        ((r >> (16 - bits)) & ((1 << bits) - 1)) as i32
22    }
23}

25const GRAIN_W: usize = 82;
26const GRAIN_H: usize = 73;
27/// Height of one luma noise stripe in the stripe buffers (32 rows plus 2 overlap rows).
28const STRIPE_H: usize = 34;
29
30/// A grain template: 73 rows of 82 samples.
31type GrainTemplate = Vec<[i32; GRAIN_W]>;
32
33fn new_template() -> GrainTemplate {
34    vec![[0i32; GRAIN_W]; GRAIN_H]
35}
36
37/// The allowed range of a grain sample at the picture's bit depth.
38#[derive(Clone, Copy)]
39struct GrainRange {
40    min: i32,
41    max: i32,
42}
43
44impl GrainRange {
45    fn new(bit_depth: u32) -> Self {
46        let center = 128i32 << (bit_depth - 8);
47        Self { min: -center, max: (256i32 << (bit_depth - 8)) - 1 - center }
48    }
49
50    fn clamp(self, v: i32) -> i32 {
51        v.clamp(self.min, self.max)
52    }
53
54    /// Overlap blend of a full-resolution grain sample with the one it overlaps (weights 27/17 then 17/27).
55    fn blend_full(self, old: i32, new: i32, first: bool) -> i32 {
56        let mixed = if first { old * 27 + new * 17 } else { old * 17 + new * 27 };
57        self.clamp(round2(mixed, 5))
58    }
59
60    /// Overlap blend of a subsampled grain sample with the one it overlaps.
61    fn blend_subsampled(self, old: i32, new: i32) -> i32 {
62        self.clamp(round2(old * 23 + new * 22, 5))
63    }
64}

69    x: usize,

73/// Fills `template` (limited to `cols` x `rows`) with scaled Gaussian noise, or zeros when `enabled` is false.
74fn fill_template(template: &mut GrainTemplate, rng: &mut Rng, enabled: bool, (cols, rows): (usize, usize), shift: u32) {
75    for row in template.iter_mut().take(rows) {
76        for v in row.iter_mut().take(cols) {
77            let gv = if enabled { GAUSSIAN_SEQUENCE[rng.next(11) as usize] as i32 } else { 0 };
78            *v = round2(gv, shift);
79        }
80    }
81}

83/// Weighted sum of the already-generated luma neighbours of `(y, x)` (auto-regressive filter, 7.18.3.3).
84fn luma_ar_sum(g: &FilmGrainParams, luma: &GrainTemplate, y: usize, x: usize, lag: i32) -> i32 {
85    let mut s = 0;
86    let mut pos = 0;
87    for dr in -lag..=0 {
88        for dc in -lag..=lag {
89            if dr == 0 && dc == 0 {
90                return s;
91            }
92            let c = g.ar_coeffs_y_plus_128[pos] as i32 - 128;
93            s += luma[(y as i32 + dr) as usize][(x as i32 + dc) as usize] * c;
94            pos += 1;
95        }
96    }
97    s
98}

100/// Luma grain template with the auto-regressive filter applied.
101fn generate_luma_grain(g: &FilmGrainParams, rng: &mut Rng, shift: u32, range: GrainRange) -> GrainTemplate {
102    let mut luma = new_template();
103    fill_template(&mut luma, rng, g.num_y_points > 0, (GRAIN_W, GRAIN_H), shift);
104    let ar_shift = g.ar_coeff_shift_minus_6 + 6;
105    let lag = g.ar_coeff_lag as i32;
106    for y in 3..GRAIN_H {
107        for x in 3..GRAIN_W - 3 {
108            let s = luma_ar_sum(g, &luma, y, x, lag);
109            luma[y][x] = range.clamp(luma[y][x] + round2(s, ar_shift));
110        }
111    }
112    luma
113}

115/// Chroma grain templates and the geometry they were generated with.

117    cb: GrainTemplate,
118    cr: GrainTemplate,
119    cols: usize,
120    rows: usize,

123/// Rounded average of the luma grain co-located with chroma sample `(y, x)`.
124fn collocated_luma_grain(luma: &GrainTemplate, y: usize, x: usize, sub: Subsampling) -> i32 {
125    let lx = ((x - 3) << sub.x) + 3;
126    let ly = ((y - 3) << sub.y) + 3;
127    let mut l = 0;
128    for i in 0..=sub.y {
129        for j in 0..=sub.x {
130            l += luma[ly + i][lx + j];
131        }
132    }
133    round2(l, (sub.x + sub.y) as u32)
134}

136/// Auto-regressive sums `(cb, cr)` for chroma sample `(y, x)`.
137fn chroma_ar_sums(g: &FilmGrainParams, luma: &GrainTemplate, chroma: &ChromaGrain, pos: (usize, usize), sub: Subsampling) -> (i32, i32) {
138    let (y, x) = pos;
139    let lag = g.ar_coeff_lag as i32;
140    let mut s0 = 0;
141    let mut s1 = 0;
142    let mut idx = 0;
143    for dr in -lag..=0 {
144        for dc in -lag..=lag {
145            let c0 = g.ar_coeffs_cb_plus_128[idx] as i32 - 128;
146            let c1 = g.ar_coeffs_cr_plus_128[idx] as i32 - 128;
147            if dr == 0 && dc == 0 {
148                if g.num_y_points > 0 {
149                    let l = collocated_luma_grain(luma, y, x, sub);
150                    s0 += l * c0;
151                    s1 += l * c1;
152                }
153                return (s0, s1);
154            }
155            let (yy, xx) = ((y as i32 + dr) as usize, (x as i32 + dc) as usize);
156            s0 += chroma.cb[yy][xx] * c0;
157            s1 += chroma.cr[yy][xx] * c1;
158            idx += 1;
159        }
160    }
161    (s0, s1)
162}

164/// Cb/Cr grain templates (all zero for monochrome pictures).
165fn generate_chroma_grain(g: &FilmGrainParams, luma: &GrainTemplate, rng: &mut Rng, sub: Subsampling, shift: u32, range: GrainRange) -> ChromaGrain {
166    let mut chroma = ChromaGrain {
167        cb: new_template(),
168        cr: new_template(),
169        cols: if sub.x != 0 { 44 } else { GRAIN_W },
170        rows: if sub.y != 0 { 38 } else { GRAIN_H },
171    };
172    let dims = (chroma.cols, chroma.rows);
173    rng.0 = g.grain_seed ^ 0xb524;
174    fill_template(&mut chroma.cb, rng, g.num_cb_points > 0 || g.chroma_scaling_from_luma, dims, shift);
175    rng.0 = g.grain_seed ^ 0x49d8;
176    fill_template(&mut chroma.cr, rng, g.num_cr_points > 0 || g.chroma_scaling_from_luma, dims, shift);
177    let ar_shift = g.ar_coeff_shift_minus_6 + 6;
178    for y in 3..chroma.rows {
179        for x in 3..chroma.cols - 3 {
180            let (s0, s1) = chroma_ar_sums(g, luma, &chroma, (y, x), sub);
181            chroma.cb[y][x] = range.clamp(chroma.cb[y][x] + round2(s0, ar_shift));
182            chroma.cr[y][x] = range.clamp(chroma.cr[y][x] + round2(s1, ar_shift));
183        }
184    }
185    chroma
186}

188/// Piecewise-linear scaling function of one plane, expanded to 256 entries (7.18.3.4).
189fn fill_plane_lut(l: &mut [i32; 256], n: usize, xs: &[u8; 16], ys: &[u8; 16]) {
190    for v in l.iter_mut().take(xs[0] as usize) {
191        *v = ys[0] as i32;
192    }
193    for i in 0..n - 1 {
194        let dy = ys[i + 1] as i32 - ys[i] as i32;
195        let dx = xs[i + 1] as i32 - xs[i] as i32;
196        // The header parser rejects non-increasing points; never divide by zero regardless.
197        if dx <= 0 {
198            continue;
199        }
200        let delta = dy * ((65536 + (dx >> 1)) / dx);
201        for x in 0..dx {
202            l[xs[i] as usize + x as usize] = ys[i] as i32 + ((x * delta + 32768) >> 16);
203        }
204    }
205    for v in l.iter_mut().skip(xs[n - 1] as usize) {
206        *v = ys[n - 1] as i32;
207    }
208}

210/// Scaling lookup tables for all planes, interpolated for high bit depths.
211struct ScalingLut {
212    lut: [[i32; 256]; 3],
213    bit_depth: u32,
214}
215
216impl ScalingLut {
217    fn new(g: &FilmGrainParams, num_planes: usize, bit_depth: u32) -> Self {
218        let mut lut = [[0i32; 256]; 3];
219        for (plane, l) in lut.iter_mut().enumerate().take(num_planes) {
220            let (n, xs, ys) = if plane == 0 || g.chroma_scaling_from_luma {
221                (g.num_y_points, &g.point_y_value, &g.point_y_scaling)
222            } else if plane == 1 {
223                (g.num_cb_points, &g.point_cb_value, &g.point_cb_scaling)
224            } else {
225                (g.num_cr_points, &g.point_cr_value, &g.point_cr_scaling)
226            };
227            if n != 0 {
228                fill_plane_lut(l, n, xs, ys);
229            }
230        }
231        Self { lut, bit_depth }
232    }
233
234    fn scale(&self, plane: usize, index: i32) -> i32 {
235        let sh = self.bit_depth - 8;
236        let x = index >> sh;
237        let rem = index - (x << sh);
238        if self.bit_depth == 8 || x == 255 {
239            self.lut[plane][x as usize]
240        } else {
241            let start = self.lut[plane][x as usize];
242            let end = self.lut[plane][x as usize + 1];
243            start + round2((end - start) * rem, sh)
244        }
245    }
246}

257/// Noise stripes of the whole picture (7.18.3.5).
258struct NoiseStripes {
259    stripes: Vec<[Vec<i32>; 3]>,
260    stripe_w: usize,
261    sub_y: usize,
262    overlap: bool,
263    range: GrainRange,

266/// Whether neighbouring noise blocks are blended, and the sample range of the result.
267#[derive(Clone, Copy)]
268struct OverlapBlend {
269    overlap: bool,
270    range: GrainRange,
271}
272
273/// Copies one random 34x34 block of grain into plane `s` of a stripe at horizontal position `x`.
274fn copy_stripe_block(
275    s: &mut [i32],
276    src: &GrainTemplate,
277    origin: (usize, usize),
278    (psx, psy): (usize, usize),
279    x: usize,
280    stripe_w: usize,
281    blend: OverlapBlend,
282) {
283    let (pox, poy) = origin;
284    let OverlapBlend { overlap, range } = blend;
285    for i in 0..(STRIPE_H >> psy) {
286        for j in 0..(STRIPE_H >> psx) {
287            let mut gv = src[poy + i][pox + j];
288            if psx == 0 {
289                let idx = i * stripe_w + x * 2 + j;
290                if j < 2 && overlap && x > 0 {
291                    gv = range.blend_full(s[idx], gv, j == 0);
292                }
293                s[idx] = gv;
294            } else {
295                let idx = i * stripe_w + x + j;
296                if j == 0 && overlap && x > 0 {
297                    gv = range.blend_subsampled(s[idx], gv);
298                }
299                s[idx] = gv;
300            }
301        }
302    }
303}
304
305impl NoiseStripes {
306    fn generate(g: &FilmGrainParams, rng: &mut Rng, luma: &GrainTemplate, chroma: &ChromaGrain, p: &StripeParams) -> Self {
307        let stripe_w = p.width + 64;
308        let n_stripes = (p.height.div_ceil(2)).div_ceil(16).max(1);
309        let mut stripes: Vec<[Vec<i32>; 3]> =
310            (0..n_stripes).map(|_| [vec![0; STRIPE_H * stripe_w], vec![0; STRIPE_H * stripe_w], vec![0; STRIPE_H * stripe_w]]).collect();
311        for (luma_num, stripe) in stripes.iter_mut().enumerate() {
312            let mut r = g.grain_seed;
313            r ^= (((luma_num * 37 + 178) & 255) << 8) as u32;
314            r ^= ((luma_num * 173 + 105) & 255) as u32;
315            rng.0 = r;
316            for x in (0..p.width.div_ceil(2)).step_by(16) {
317                let rand = rng.next(8) as usize;
318                let off_x = rand >> 4;
319                let off_y = rand & 15;
320                for (plane, s) in stripe.iter_mut().enumerate().take(p.num_planes) {
321                    let psx = if plane > 0 { p.sub.x } else { 0 };
322                    let psy = if plane > 0 { p.sub.y } else { 0 };
323                    let pox = if psx != 0 { 6 + off_x } else { 9 + off_x * 2 };
324                    let poy = if psy != 0 { 6 + off_y } else { 9 + off_y * 2 };
325                    let src = match plane {
326                        0 => luma,
327                        1 => &chroma.cb,
328                        _ => &chroma.cr,
329                    };
330                    copy_stripe_block(s, src, (pox, poy), (psx, psy), x, stripe_w, OverlapBlend { overlap: g.overlap_flag, range: p.range });
331                }
332            }
333        }
334        Self { stripes, stripe_w, sub_y: p.sub.y, overlap: g.overlap_flag, range: p.range }
335    }
336
337    /// Grain sample for `(y, x)` of `plane`, blended with the previous stripe where they overlap.
338    fn noise_at(&self, plane: usize, y: usize, x: usize) -> i32 {
339        let psy = if plane > 0 { self.sub_y } else { 0 };
340        let luma_num = y >> (5 - psy);
341        let i = y - (luma_num << (5 - psy));
342        let mut gv = self.stripes[luma_num][plane][i * self.stripe_w + x];
343        if psy == 0 {
344            if i < 2 && luma_num > 0 && self.overlap {
345                let old = self.stripes[luma_num - 1][plane][(i + 32) * self.stripe_w + x];
346                gv = self.range.blend_full(old, gv, i == 0);
347            }
348        } else if i < 1 && luma_num > 0 && self.overlap {
349            let old = self.stripes[luma_num - 1][plane][(i + 16) * self.stripe_w + x];
350            gv = self.range.blend_subsampled(old, gv);
351        }
352        gv
353    }
354}

356/// Sample limits and scaling shift used when adding grain to the picture.
357struct BlendParams {
358    bit_depth: u32,
359    min_value: i32,
360    max_luma: i32,
361    max_chroma: i32,
362    scaling_shift: u32,
363}
364
365impl BlendParams {
366    fn new(g: &FilmGrainParams, pic: &Picture) -> Self {
367        let bd = pic.bit_depth as u32;
368        let (min_value, max_luma, max_chroma) = if g.clip_to_restricted_range {
369            let ml = 235 << (bd - 8);
370            (16 << (bd - 8), ml, if pic.matrix_coefficients as usize == MC_IDENTITY { ml } else { 240 << (bd - 8) })
371        } else {
372            let m = (256 << (bd - 8)) - 1;
373            (0, m, m)
374        };
375        Self { bit_depth: bd, min_value, max_luma, max_chroma, scaling_shift: g.grain_scaling_minus_8 + 8 }
376    }
377}

379/// Per-chroma-plane film grain parameters used by the blend.

382    enabled: bool,
383    luma_mult: u32,
384    mult: u32,
385    offset: u32,
386}

388/// Adds scaled grain to the chroma planes (7.18.3.5).
389fn blend_chroma(pic: &mut Picture, g: &FilmGrainParams, lut: &ScalingLut, noise: &NoiseStripes, b: &BlendParams) {
390    let (w, h) = (pic.width as usize, pic.height as usize);
391    let (sub_x, sub_y) = (pic.subsampling_x as usize, pic.subsampling_y as usize);
392    let pix_max = (1i32 << b.bit_depth) - 1;
393    let cw = (w + sub_x) >> sub_x;
394    let ch = (h + sub_y) >> sub_y;
395    let [py, pu, pv] = &mut pic.planes;
396    let planes = [
397        ChromaBlendPlane { plane: 1, enabled: g.num_cb_points > 0, luma_mult: g.cb_luma_mult, mult: g.cb_mult, offset: g.cb_offset },
398        ChromaBlendPlane { plane: 2, enabled: g.num_cr_points > 0, luma_mult: g.cr_luma_mult, mult: g.cr_mult, offset: g.cr_offset },
399    ];
400    for y in 0..ch {
401        for x in 0..cw {
402            let lx = x << sub_x;
403            let ly = y << sub_y;
404            let lnx = (lx + 1).min(w - 1);
405            let avg = if sub_x != 0 { round2(py[ly * w + lx] as i32 + py[ly * w + lnx] as i32, 1) } else { py[ly * w + lx] as i32 };
406            for (cp, out) in planes.iter().zip([&mut *pu, &mut *pv]) {
407                if !(cp.enabled || g.chroma_scaling_from_luma) {
408                    continue;
409                }
410                let orig = out[y * cw + x] as i32;
411                let merged = if g.chroma_scaling_from_luma {
412                    avg
413                } else {
414                    let combined = avg * (cp.luma_mult as i32 - 128) + orig * (cp.mult as i32 - 128);
415                    ((combined >> 6) + ((cp.offset as i32 - 256) << (b.bit_depth - 8))).clamp(0, pix_max)
416                };
417                let n = round2(lut.scale(cp.plane, merged) * noise.noise_at(cp.plane, y, x), b.scaling_shift);
418                out[y * cw + x] = (orig + n).clamp(b.min_value, b.max_chroma) as u16;
419            }
420        }
421    }
422}

424/// Adds scaled grain to the luma plane (7.18.3.5).
425fn blend_luma(pic: &mut Picture, lut: &ScalingLut, noise: &NoiseStripes, b: &BlendParams) {
426    let (w, h) = (pic.width as usize, pic.height as usize);
427    let py = &mut pic.planes[0];
428    for y in 0..h {
429        for x in 0..w {
430            let orig = py[y * w + x] as i32;
431            let n = round2(lut.scale(0, orig) * noise.noise_at(0, y, x), b.scaling_shift);
432            py[y * w + x] = (orig + n).clamp(b.min_value, b.max_luma) as u16;
433        }
434    }
435}

437pub(crate) fn apply(pic: &mut Picture, g: &FilmGrainParams, seq: &SequenceHeader) {
438    let _ = seq;
439    let bd = pic.bit_depth as u32;
440    let mono = pic.mono_chrome;
441    let num_planes = if mono { 1 } else { 3 };
442    let sub = Subsampling { x: pic.subsampling_x as usize, y: pic.subsampling_y as usize };
443    let range = GrainRange::new(bd);
444
445    // Generate grain (7.18.3.3).
446    let mut rng = Rng(g.grain_seed);
447    let shift = 12 - bd + g.grain_scale_shift;
448    let luma = generate_luma_grain(g, &mut rng, shift, range);
449    let chroma = if mono {
450        ChromaGrain { cb: new_template(), cr: new_template(), cols: 0, rows: 0 }
451    } else {
452        generate_chroma_grain(g, &luma, &mut rng, sub, shift, range)
453    };
454
455    let lut = ScalingLut::new(g, num_planes, bd);
456    let stripe_params = StripeParams { width: pic.width as usize, height: pic.height as usize, num_planes, sub, range };
457    let noise = NoiseStripes::generate(g, &mut rng, &luma, &chroma, &stripe_params);
458
459    let blend = BlendParams::new(g, pic);
460    if !mono {
461        blend_chroma(pic, g, &lut, &noise, &blend);
462    }
463    if g.num_y_points > 0 {
464        blend_luma(pic, &lut, &noise, &blend);
465    }
466}
