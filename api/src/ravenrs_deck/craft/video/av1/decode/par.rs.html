1//! Optional data parallelism (rayon behind the `threads` feature; sequential otherwise and on
2//! wasm32).
3
4use std::sync::Arc;
5
6/// A handle to the decoder's worker pool (or none: run on the calling thread).
7#[derive(Clone, Default)]
8pub(crate) struct Pool {
9    pool: Option<Arc<rayon::ThreadPool>>,
10}
11
12impl Pool {
13    /// A pool of `threads` workers (sequential when `threads` <= 1, without the `threads`
14    /// feature, or on wasm32).
15    pub fn new(threads: usize) -> Pool {
16        {
17            let pool = if threads > 1 && cfg!(not(target_arch = "wasm32")) {
18                rayon::ThreadPoolBuilder::new().num_threads(threads).thread_name(|i| format!("av1-{i}")).build().ok().map(Arc::new)
19            } else {
20                None
21            };
22            Pool { pool }
23        }
24        #[cfg(any())]
25        {
26            let _ = threads;
27            Pool {}
28        }
29    }
30
31    pub fn is_parallel(&self) -> bool {
32        {
33            self.pool.is_some()
34        }
35        #[cfg(any())]
36        {
37            false
38        }
39    }
40
41    /// `f(i, &mut items[i])` for every item, in parallel when possible; results in order.
42    pub fn map_mut<T: Send, R: Send>(&self, items: &mut [T], f: impl Fn(usize, &mut T) -> R + Sync + Send) -> Vec<R> {
43        if let Some(pool) = &self.pool
44            && items.len() > 1
45        {
46            use rayon::prelude::*;
47            return pool.install(|| items.par_iter_mut().enumerate().map(|(i, t)| f(i, t)).collect());
48        }
49        items.iter_mut().enumerate().map(|(i, t)| f(i, t)).collect()
50    }
51}
