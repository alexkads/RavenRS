1//! Memory budget for decoded pictures, shared by the video decoders.
2//!
3//! A per-axis size cap does not bound memory: a decoder holds the current picture, every
4//! reference picture and the pictures still being decoded on worker threads. Each decoder checks
5//! the whole picture store against [`DECODE_MEMORY_BUDGET`] with checked arithmetic before it
6//! allocates for a new picture size, so a hostile header is an error, not an out-of-memory abort.
7
8/// Bytes the decoded pictures of one decoder may take: 4 GiB on 64-bit targets, 1 GiB on 32-bit
9/// ones (wasm32 has a 4 GiB address space in all).
10pub const DECODE_MEMORY_BUDGET: u64 = if cfg!(target_pointer_width = "64") { 4 << 30 } else { 1 << 30 };
11
12/// The shape of one decoded picture as a decoder allocates it.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub struct PictureShape {
15    /// Allocated luma samples per row and rows (padding included).
16    pub width: u64,
17    pub height: u64,
18    /// Chroma subsampling shifts (0 or 1); ignored when `chroma_planes` is 0.
19    pub shift_x: u32,
20    pub shift_y: u32,
21    /// Chroma planes (0 for monochrome, 2 otherwise).
22    pub chroma_planes: u64,
23    /// Bytes per sample (1 for 8-bit buffers, 2 for 16-bit ones).
24    pub bytes_per_sample: u64,
25}
26
27impl PictureShape {
28    /// Bytes of one picture, `None` if it does not fit in a `u64`.
29    pub fn bytes(&self) -> Option<u64> {
30        let luma = self.width.checked_mul(self.height)?;
31        let cw = self.width.div_ceil(1 << self.shift_x.min(1));
32        let ch = self.height.div_ceil(1 << self.shift_y.min(1));
33        let chroma = cw.checked_mul(ch)?.checked_mul(self.chroma_planes)?;
34        luma.checked_add(chroma)?.checked_mul(self.bytes_per_sample)
35    }
36
37    /// Bytes of `pictures` such pictures, `None` on overflow.
38    pub fn store_bytes(&self, pictures: u64) -> Option<u64> {
39        self.bytes()?.checked_mul(pictures)
40    }
41
42    /// Whether `pictures` such pictures fit in `budget` bytes.
43    pub fn fits(&self, pictures: u64, budget: u64) -> bool {
44        self.store_bytes(pictures).is_some_and(|b| b <= budget)
45    }
46}

48#[cfg(any())]
49mod tests {
50    use super::*;
51
52    fn shape(w: u64, h: u64) -> PictureShape {
53        PictureShape { width: w, height: h, shift_x: 1, shift_y: 1, chroma_planes: 2, bytes_per_sample: 2 }
54    }
55
56    #[cfg(any())]
57    fn bytes_count_luma_and_subsampled_chroma() {
58        assert_eq!(shape(4, 4).bytes(), Some((16 + 2 * 4) * 2));
59        // Odd sizes round chroma up.
60        assert_eq!(shape(3, 3).bytes(), Some((9 + 2 * 4) * 2));
61        let mono = PictureShape { chroma_planes: 0, bytes_per_sample: 1, ..shape(10, 10) };
62        assert_eq!(mono.bytes(), Some(100));
63        let full = PictureShape { shift_x: 0, shift_y: 0, ..shape(10, 10) };
64        assert_eq!(full.bytes(), Some(600));
65    }
66
67    #[cfg(any())]
68    fn exact_budget_boundary() {
69        let s = shape(4, 4); // 48 bytes
70        assert!(s.fits(10, 480));
71        assert!(!s.fits(10, 479));
72        assert!(!s.fits(11, 480));
73    }
74
75    #[cfg(any())]
76    fn overflow_is_never_within_budget() {
77        let huge = shape(u64::MAX / 2, 3);
78        assert_eq!(huge.bytes(), None);
79        assert!(!huge.fits(1, u64::MAX));
80        assert!(!shape(1 << 31, 1 << 31).fits(u64::MAX, u64::MAX));
81    }
82}
