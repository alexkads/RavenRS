6//! Audio file I/O and codecs for the Craft apps. Input and output are byte slices or a
7//! [`crate::craft::media::ByteSource`] (no filesystem access of its own); builds for
8//! `wasm32-unknown-unknown`.
9//!
10//! - [`decode`], [`probe`], [`Decoder`] (streaming, seekable), [`tracks`], [`detect_container`]
11//!   (feature `decode`): WAV and AIFF are read natively; AAC, ALAC, Opus, AC-3/E-AC-3,
12//!   MP1/MP2/MP3 and PCM in MP4/MOV, Matroska/WebM, Ogg or raw streams go through craft-media's
13//!   demuxers and [`TrackDecoder`]; everything else (FLAC, Vorbis, CAF, ADPCM, A-law, µ-law) through
14//!   `symphonia`, which is to be replaced by native decoders. Feature `decode-native` is the
15//!   same without symphonia.
16//! - [`TrackDecoder`] (any decoder feature): one track's packets from any demuxer to planar
17//!   `f32` in WAV channel order.
18//! - [`encode`], [`FileEncoder`]: WAV/BWF/RF64 and AIFF/AIFF-C (feature `pcm-encode`, which also
19//!   gives [`wav_float_header`]) and FLAC (feature `flac-encode`), whole-buffer or incremental into
20//!   any `Write + Seek`. Feature `lossless-encode` turns on both; feature `encode` turns on
21//!   `lossless-encode` and every codec's encoder.
22//! - [`wav_pcm`], [`aiff_pcm`] (feature `decode-native`) and [`pcm_wav_file`], [`pcm_wav_header`],
23//!   [`pcm_aiff_file`], [`pcm_aiff_header`] (feature `pcm-encode`): the integer PCM of a WAV or AIFF
24//!   file bit for bit, and minimal files around such PCM, for essence embedded in other containers.
25//! - [`peaks`] (feature `peaks`): min/max/RMS pyramid for waveform drawing.
26//! - Codec modules for packet-level use, each with its own features: [`aac`] (`aac-decode`,
27//!   `aac-encode`), [`alac`] (`alac-decode`, `alac-encode`), [`opus`] (`opus-decode`), [`ac3`]
28//!   (`ac3-decode`), [`mpa`] (`mpa-decode`), [`mp3`] (`mp3-decode`, `mp3-encode`),
29//!   [`flac`] (`flac-encode`: the packet encoder).
30//!
31//! The shared types below build with no features.
32

57/// The FLAC packet encoder (RFC 9639), for a caller that muxes FLAC itself (into MP4, Matroska or
58/// its own `.flac` writer): planar integer samples in, frames out, and the STREAMINFO to patch in
59/// at the end. [`FileEncoder`] and [`encode`] write whole `.flac` files from `f32` instead.
60///
61/// ```ignore
62/// use ravenrs_deck::craft::audio::flac::{Encoder, EncoderConfig, STREAMINFO_OFFSET};
63///
64/// let left: Vec<i32> = (0..48_000).map(|i| ((i as f64 * 0.05).sin() * 20_000.0) as i32).collect();
65/// let right = left.clone();
66/// let mut enc = Encoder::new(EncoderConfig::new(48_000, 2, 16))?;
67/// let mut file = enc.header(&[]);
68/// file.extend(enc.encode(&[&left, &right])?);
69/// file.extend(enc.finish());
70/// let at = STREAMINFO_OFFSET as usize;
71/// file[at..at + 34].copy_from_slice(&enc.streaminfo());
72/// assert_eq!(&file[..4], b"fLaC");
73/// assert!(file.len() < 48_000 * 2 * 2 / 2); // well under half of 16-bit PCM
74/// # Ok::<(), ravenrs_deck::craft::audio::flac::Error>(())
75/// ```
76#[cfg(any())]
77pub mod flac {
78    pub use crate::craft::audio::write::flac::encoder::{Encoder, EncoderConfig, Error, STREAMINFO_OFFSET};
79}

104/// Errors from probing, decoding and encoding.
105#[non_exhaustive]
106#[derive(Debug, thiserror::Error)]
107pub enum AudioError {
108    /// The format, codec or option is valid but not supported.
109    #[error("unsupported audio: {0}")]
110    Unsupported(String),
111    /// The input is corrupt or not what it claims to be.
112    #[error("malformed audio: {0}")]
113    Malformed(String),
114    /// The input (or requested output) exceeds a size limit.
115    #[error("audio too large: {0}")]
116    TooLarge(String),
117    /// Encoding failed (e.g. an empty buffer).
118    #[error("audio encode failed: {0}")]
119    Encode(String),
120    /// Reading the input or writing the output failed (a source that stopped serving reads, a
121    /// full disk). Bytes missing because the input is truncated are [`AudioError::Malformed`].
122    #[error("audio I/O failed: {0}")]
123    Io(#[from] std::io::Error),
124}
125
126/// Result alias for this module.
127pub type Result<T> = std::result::Result<T, AudioError>;

129// Codec errors convert into `AudioError`, so code mixing the file API with the codec modules can
130// use `?` throughout: a codec's "unsupported" stays `Unsupported`, anything else in the input is
131// `Malformed`, and an encoder's refusal is `Encode`.

141    }

205/// The file formats [`encode`] writes. Each needs its feature: WAV and AIFF `pcm-encode`, FLAC
206/// `flac-encode`; asking for one that is not built returns [`AudioError::Unsupported`].
207#[non_exhaustive]
208#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
209pub enum EncodeFormat {
210    /// WAV / BWF, RF64 above 4 GiB.
211    Wav,
212    /// AIFF (integer depths) or AIFF-C `fl32` (float).
213    Aiff,
214    /// FLAC (integer depths).
215    Flac,
216}
217
218impl EncodeFormat {
219    /// Every format, in the order menus list them. Whether [`encode`] writes one depends on its
220    /// feature.
221    pub const ALL: &[EncodeFormat] = &[EncodeFormat::Wav, EncodeFormat::Aiff, EncodeFormat::Flac];
222
223    /// The extension a new file of this format gets, lower case and without the dot: `"wav"`,
224    /// `"aif"`, `"flac"`. [`encode_format_for`] maps it back.
225    pub fn extension(self) -> &'static str {
226        match self {
227            EncodeFormat::Wav => "wav",
228            EncodeFormat::Aiff => "aif",
229            EncodeFormat::Flac => "flac",
230        }
231    }
232
233    /// The media type a file of this format is served or downloaded as.
234    pub fn mime_type(self) -> &'static str {
235        match self {
236            EncodeFormat::Wav => "audio/wav",
237            EncodeFormat::Aiff => "audio/aiff",
238            EncodeFormat::Flac => "audio/flac",
239        }
240    }
241
242    /// The container this format writes.
243    pub fn container(self) -> Container {
244        match self {
245            EncodeFormat::Wav => Container::Wav,
246            EncodeFormat::Aiff => Container::Aiff,
247            EncodeFormat::Flac => Container::Flac,
248        }
249    }
250}

324/// What `probe` learns about a file without decoding all of it.
325///
326/// `#[non_exhaustive]`: fields may be added, so it is only built by this crate.
327#[non_exhaustive]
328#[derive(Debug, Clone, PartialEq)]
329pub struct AudioInfo {
330    /// The file format around the audio.
331    pub container: Container,
332    /// The codec of the audio track ([`CodecId::Pcm`] for uncompressed audio).
333    pub codec: CodecId,
334    /// The stored sample representation.
335    pub sample_format: SampleFormat,
336    /// The decoded sample rate (for HE-AAC, the rate after spectral band replication).
337    pub sample_rate: u32,
338    /// Decoded channel count.
339    pub channels: u16,
340    /// Which speaker each decoded channel feeds. Decoded audio is in WAV channel order when the
341    /// layout is ordered.
342    pub layout: ChannelLayout,
343    /// Length in frames (samples per channel), after container trimming.
344    pub frames: u64,
345    /// Broadcast Wave metadata from a WAV `bext` chunk.
346    pub bwf: Option<BwfInfo>,
347    /// Packets that were damaged or failed to decode and were replaced by concealment (silence of
348    /// their length), and runs of FLAC frames lost to a failed checksum, counted as audio is decoded: by [`decode`] for the whole file, and so far by
349    /// a [`Decoder`]. 0 from [`probe`] when it read only headers.
350    pub damaged: u64,
351}
352
353impl AudioInfo {
354    /// Info for a stream with `layout`'s channels and no BWF metadata.
355    #[cfg(any())]
356    pub(crate) fn new(
357        container: Container,
358        codec: CodecId,
359        sample_format: SampleFormat,
360        sample_rate: u32,
361        layout: ChannelLayout,
362        frames: u64,
363    ) -> AudioInfo {
364        let channels = layout.channels();
365        AudioInfo { container, codec, sample_format, sample_rate, channels, layout, frames, bwf: None, damaged: 0 }
366    }
367
368    /// Duration in seconds.
369    pub fn duration_secs(&self) -> f64 {
370        if self.sample_rate == 0 { 0.0 } else { self.frames as f64 / f64::from(self.sample_rate) }
371    }
372}

388/// Options for `encode` and `FileEncoder`.
389///
390/// `#[non_exhaustive]`: build it with [`EncodeOptions::new`] and the `with_*` methods.
391///
392/// ```ignore
393/// use ravenrs_deck::craft::audio::{BitDepth, EncodeFormat, EncodeOptions};
394///
395/// let opts = EncodeOptions::new(EncodeFormat::Flac).with_bit_depth(BitDepth::Int16).with_dither(true);
396/// assert_eq!(opts.format, EncodeFormat::Flac);
397/// ```
398#[non_exhaustive]
399#[derive(Debug, Clone, PartialEq)]
400pub struct EncodeOptions {
401    /// Output file format.
402    pub format: EncodeFormat,
403    /// Sample depth to write.
404    pub bit_depth: BitDepth,
405    /// Add TPDF dither when quantizing to an integer depth.
406    pub dither: bool,
407    /// Write a BWF `bext` chunk (WAV only; ignored for other formats).
408    pub bwf: Option<BwfInfo>,
409    /// The speaker layout to record, for layouts a channel count alone cannot identify (5.1 with
410    /// side vs back surrounds, 7.0 vs 6.1, Ambisonics as [`ChannelLayout::unordered`]). `None` is
411    /// the usual layout for the channel count. WAV stores it as `dwChannelMask`, FLAC as a
412    /// `WAVEFORMATEXTENSIBLE_CHANNEL_MASK` comment when decoders would assume another layout for
413    /// the count (they read 5 and 6 channels with side surrounds and 7 as 6.1 with sides, so the
414    /// usual layouts for those counts get one; see the README's limits); AIFF has a fixed channel
415    /// order and ignores it. It must have as many channels as the audio.
416    pub layout: Option<ChannelLayout>,
417    /// Write WAV in the RF64 layout even below 4 GiB (WAV only; ignored for other formats).
418    pub force_rf64: bool,
419}
420
421impl EncodeOptions {
422    /// Options for `format`: 24-bit, no dither, no BWF chunk, the usual layout.

425    }

433    /// Sets `bit_depth`.
434    pub fn with_bit_depth(mut self, bit_depth: BitDepth) -> Self {
435        self.bit_depth = bit_depth;
436        self
437    }
438
439    /// Sets `dither`.
440    pub fn with_dither(mut self, dither: bool) -> Self {
441        self.dither = dither;
442        self
443    }
444
445    /// Sets `bwf`.
446    pub fn with_bwf(mut self, bwf: BwfInfo) -> Self {

448        self
449    }
450
451    /// Sets `layout`.

453        self.layout = Some(layout);

455    }
456
457    /// Sets `force_rf64`.

459        self.force_rf64 = force_rf64;

461    }
462}
