40fn packet_silence(dec: &mut RangeDecoder, total_bits: i32) -> (bool, i32) {
41    let mut tell = dec.tell();
42    let silence = if tell >= total_bits {
43        true
44    } else if tell == 1 {
45        dec.bit_logp(15)
46    } else {
47        false
48    };
49    if silence {
50        // Pretend we've read all the remaining bits.
51        dec.skip_to_end(total_bits);
52        tell = total_bits;
53    }
54    (silence, tell)
55}

57fn read_postfilter(dec: &mut RangeDecoder, start: usize, mut tell: i32, total_bits: i32) -> (f32, i32, usize, i32) {
58    let mut gain = 0.0;
59    let mut pitch = 0;
60    let mut tapset = 0;
61    if start == 0 && tell + 16 <= total_bits {
62        if dec.bit_logp(1) {
63            let octave = dec.uint(6);
64            pitch = ((16 << octave) + dec.bits(4 + octave)) as i32 - 1;
65            let qg = dec.bits(3);
66            if dec.tell() + 2 <= total_bits {
67                tapset = dec.icdf(&TAPSET_ICDF, 2);
68            }
69            gain = 0.09375 * (qg + 1) as f32;
70        }
71        tell = dec.tell();
72    }
73    (gain, pitch, tapset, tell)
74}

172    /// Decodes one CELT frame of `frame_size` samples (at the output rate) into interleaved `pcm`
173    /// (`channels` wide). With `accum`, the output is added to `pcm`. `data = None` (or fewer than
174    /// two bytes) runs the packet-loss concealment.
175    pub fn decode(
176        &mut self,
177        data: Option<&[u8]>,
178        frame_size: usize,
179        pcm: &mut [f32],
180        accum: bool,
181        dec: Option<&mut RangeDecoder>,
182    ) -> Result<usize, crate::craft::audio::opus::decode::Error> {
183        let n_full = frame_size * self.downsample;
184        let lm = match n_full {
185            120 => 0,
186            240 => 1,
187            480 => 2,
188            960 => 3,
189            _ => return Err(crate::craft::audio::opus::decode::Error::InvalidArgument("CELT frame size")),
190        };
191        let data = match data {
192            Some(d) if d.len() > 1 => d,
193            _ => {
194                self.decode_lost(n_full, lm);
195                self.deemphasis(n_full, pcm, accum);
196                return Ok(frame_size);
197            }
198        };
199        if data.len() > 1275 {
200            return Err(crate::craft::audio::opus::decode::Error::InvalidPacket("CELT frame too long"));
201        }
202        match dec {
203            Some(d) => self.decode_frame(d, data.len(), n_full, lm),
204            None => {
205                let mut own = RangeDecoder::new(data);
206                self.decode_frame(&mut own, data.len(), n_full, lm)
207            }
208        }
209        self.deemphasis(n_full, pcm, accum);
210        Ok(frame_size)
211    }

213    #[allow(clippy::cognitive_complexity, reason = "the RFC 6716 CELT frame decode, ported as-is from FilmCraft")]
214    fn decode_frame(&mut self, dec: &mut RangeDecoder, len: usize, n: usize, lm: usize) {
215        let cc = self.channels;
216        let c = self.stream_channels;
217        let m = self.m;
218        let mm = 1usize << lm;
219        let start = self.start;
220        let end = self.end;
221        let eff_end = end.min(NB_EBANDS);
222        if self.loss_count == 0 {
223            self.skip_plc = false;
224        }
225        if c == 1 {
226            for i in 0..NB_EBANDS {
227                self.old_band_e[i] = self.old_band_e[i].max(self.old_band_e[NB_EBANDS + i]);
228            }
229        }
230        let total_bits_raw = len as i32 * 8;
231        let (silence, tell) = packet_silence(dec, total_bits_raw);
232        let (postfilter_gain, postfilter_pitch, postfilter_tapset, mut tell) = read_postfilter(dec, start, tell, total_bits_raw);
233        let is_transient = if lm > 0 && tell + 3 <= total_bits_raw {
234            let t = dec.bit_logp(3);
235            tell = dec.tell();
236            t
237        } else {
238            false
239        };
240        let short_blocks = is_transient;
241        let intra = if tell + 3 <= total_bits_raw { dec.bit_logp(3) } else { false };
242        unquant_coarse_energy(start, end, &mut self.old_band_e, intra, dec, c, lm);
243        let mut tf_res = [0i32; NB_EBANDS];
244        tf_decode(start, end, is_transient, &mut tf_res, lm, dec);
245        tell = dec.tell();
246        let mut spread = 2usize;
247        if tell + 4 <= total_bits_raw {
248            spread = dec.icdf(&SPREAD_ICDF, 5);
249        }
250        let cap = m.init_caps(lm, c);
251        let mut offsets = [0i32; NB_EBANDS];
252        let mut dynalloc_logp = 6;
253        let mut total_bits = total_bits_raw << BITRES;
254        let mut tellf = dec.tell_frac() as i32;
255        for i in start..end {
256            let width = (c as i32 * (EBANDS[i + 1] - EBANDS[i])) << lm;
257            let quanta = (width << BITRES).min((6 << BITRES).max(width));
258            let mut loop_logp = dynalloc_logp;
259            let mut boost = 0;
260            while tellf + (loop_logp << BITRES) < total_bits && boost < cap[i] {
261                let flag = dec.bit_logp(loop_logp as u32);
262                tellf = dec.tell_frac() as i32;
263                if !flag {
264                    break;
265                }
266                boost += quanta;
267                total_bits -= quanta;
268                loop_logp = 1;
269            }
270            offsets[i] = boost;
271            if boost > 0 {
272                dynalloc_logp = 2.max(dynalloc_logp - 1);
273            }
274        }
275        let alloc_trim = if tellf + (6 << BITRES) <= total_bits { dec.icdf(&TRIM_ICDF, 7) as i32 } else { 5 };
276        let mut bits = ((len as i32 * 8) << BITRES) - dec.tell_frac() as i32 - 1;
277        let anti_collapse_rsv = if is_transient && lm >= 2 && bits >= ((lm as i32 + 2) << BITRES) { 1 << BITRES } else { 0 };
278        bits -= anti_collapse_rsv;
279        let alloc = compute_allocation(AllocationInput {
280            mode: m,
281            start,
282            end,
283            offsets: &offsets,
284            cap: &cap,
285            trim: alloc_trim,
286            total: bits,
287            channels: c,
288            lm,
289            decoder: dec,
290        });
291        unquant_fine_energy(start, end, &mut self.old_band_e, &alloc.fine_quant, dec, c);
292
293        // Shift the decode memory by one frame.
294        let ml = self.mem_len();
295        for ch in 0..cc {
296            let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
297            mem.copy_within(n..DECODE_BUFFER_SIZE + OVERLAP / 2, 0);
298        }
299
300        let mut collapse_masks = vec![0u32; c * NB_EBANDS];
301        let mut x = std::mem::take(&mut self.x);
302        x.clear();
303        x.resize(c * n, 0.0);
304        {
305            let (xa, xb) = x.split_at_mut(n);
306            let yb = if c == 2 { Some(xb) } else { None };
307            let mut seed = self.rng;
308            quant_all_bands(QuantAllBandsInput {
309                mode: m,
310                start,
311                end,
312                x_buf: xa,
313                y_buf: yb,
314                collapse_masks: &mut collapse_masks,
315                pulses: &alloc.pulses,
316                short_blocks,
317                spread,
318                dual_stereo: alloc.dual_stereo,
319                intensity: alloc.intensity,
320                tf_res: &tf_res,
321                total_bits: ((len as i32) * (8 << BITRES)) - anti_collapse_rsv,
322                balance: alloc.balance,
323                decoder: dec,
324                lm,
325                coded_bands: alloc.coded_bands,
326                seed: &mut seed,
327                disable_inv: self.disable_inv,
328            });
329            self.rng = seed;
330        }
331        let mut anti_collapse_on = false;
332        if anti_collapse_rsv > 0 {
333            anti_collapse_on = dec.bits(1) != 0;
334        }
335        let bits_left = len as i32 * 8 - dec.tell();
336        unquant_energy_finalise(EnergyFinalise {
337            start,
338            end,
339            old_e: &mut self.old_band_e,
340            fine_quant: &alloc.fine_quant,
341            fine_priority: &alloc.fine_priority,
342            bits_left,
343            decoder: dec,
344            channels: c,
345        });
346        if anti_collapse_on {
347            anti_collapse(AntiCollapseInput {
348                x_buf: &mut x,
349                collapse_masks: &collapse_masks,
350                lm,
351                channels: c,
352                size: n,
353                start,
354                end,
355                log_e: &self.old_band_e,
356                prev1: &self.old_log_e,
357                prev2: &self.old_log_e2,
358                pulses: &alloc.pulses,
359                seed: self.rng,
360            });
361        }
362        if silence {
363            for v in self.old_band_e[..c * NB_EBANDS].iter_mut() {
364                *v = -28.0;
365            }
366        }
367        self.synthesis(SynthesisFrame {
368            bands: &x,
369            samples: n,
370            start_band: start,
371            end_band: eff_end,
372            coded_channels: c,
373            transient: is_transient,
374            lm,
375            silence,
376        });
377        self.x = x;
378
379        // Post-filter.
380        for ch in 0..cc {
381            self.postfilter_period = self.postfilter_period.max(COMBFILTER_MINPERIOD);
382            self.postfilter_period_old = self.postfilter_period_old.max(COMBFILTER_MINPERIOD);
383            let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
384            let off = DECODE_BUFFER_SIZE - n;
385            comb_filter(
386                mem,
387                CombFilterParams {
388                    offset: off,
389                    old_period: self.postfilter_period_old,
390                    new_period: self.postfilter_period,
391                    samples: SHORT_MDCT,
392                    old_gain: self.postfilter_gain_old,
393                    new_gain: self.postfilter_gain,
394                    old_tapset: self.postfilter_tapset_old,
395                    new_tapset: self.postfilter_tapset,
396                    window: &m.window,
397                },
398            );
399            if lm != 0 {
400                comb_filter(
401                    mem,
402                    CombFilterParams {
403                        offset: off + SHORT_MDCT,
404                        old_period: self.postfilter_period,
405                        new_period: postfilter_pitch,
406                        samples: n - SHORT_MDCT,
407                        old_gain: self.postfilter_gain,
408                        new_gain: postfilter_gain,
409                        old_tapset: self.postfilter_tapset,
410                        new_tapset: postfilter_tapset,
411                        window: &m.window,
412                    },
413                );
414            }
415        }
416        self.postfilter_period_old = self.postfilter_period;
417        self.postfilter_gain_old = self.postfilter_gain;
418        self.postfilter_tapset_old = self.postfilter_tapset;
419        self.postfilter_period = postfilter_pitch;
420        self.postfilter_gain = postfilter_gain;
421        self.postfilter_tapset = postfilter_tapset;
422        if lm != 0 {
423            self.postfilter_period_old = self.postfilter_period;
424            self.postfilter_gain_old = self.postfilter_gain;
425            self.postfilter_tapset_old = self.postfilter_tapset;
426        }
427
428        if c == 1 {
429            let (a, b) = self.old_band_e.split_at_mut(NB_EBANDS);
430            b.copy_from_slice(a);
431        }
432        if !is_transient {
433            self.old_log_e2 = self.old_log_e;
434            self.old_log_e = self.old_band_e;
435            for i in 0..2 * NB_EBANDS {
436                self.background_log_e[i] = (self.background_log_e[i] + mm as f32 * 0.001).min(self.old_band_e[i]);
437            }
438        } else {
439            for i in 0..2 * NB_EBANDS {
440                self.old_log_e[i] = self.old_log_e[i].min(self.old_band_e[i]);
441            }
442        }
443        for ch in 0..2 {
444            for i in (0..start).chain(end..NB_EBANDS) {
445                self.old_band_e[ch * NB_EBANDS + i] = 0.0;
446                self.old_log_e[ch * NB_EBANDS + i] = -28.0;
447                self.old_log_e2[ch * NB_EBANDS + i] = -28.0;
448            }
449        }
450        self.rng = dec.rng();
451        self.loss_count = 0;
452    }

454    /// Denormalisation + IMDCT into the decode memory (`celt_synthesis`).
455    fn synthesis(&mut self, frame: SynthesisFrame<'_>) {
456        let SynthesisFrame { bands: x, samples: n, start_band: start, end_band: eff_end, coded_channels: c, transient: is_transient, lm, silence } =
457            frame;
458        let cc = self.channels;
459        let m = self.m;
460        let mm = 1usize << lm;
461        let (bcount, nb, shift) = if is_transient { (mm, SHORT_MDCT, MAX_LM) } else { (1, SHORT_MDCT << lm, MAX_LM - lm) };
462        let ml = self.mem_len();
463        let mut freq = std::mem::take(&mut self.freq);
464        freq.clear();
465        freq.resize(2 * n, 0.0);
466        let (f0, f1) = freq.split_at_mut(n);
467        let out_off = DECODE_BUFFER_SIZE - n;
468        if cc == 2 && c == 1 {
469            denormalise_bands(DenormaliseBandsInput {
470                x,
471                freq: f0,
472                band_log_e: &self.old_band_e,
473                start,
474                end: eff_end,
475                mm,
476                downsample: self.downsample,
477                silence,
478            });
479            f1.copy_from_slice(f0);
480            for (ch, f) in [(0usize, &*f1), (1usize, &*f0)] {
481                let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
482                for b in 0..bcount {
483                    m.mdct[shift].backward(&f[b..], bcount, &mut mem[out_off + nb * b..], &m.window, OVERLAP, &mut self.imdct);
484                }
485            }
486        } else if cc == 1 && c == 2 {
487            denormalise_bands(DenormaliseBandsInput {
488                x,
489                freq: f0,
490                band_log_e: &self.old_band_e,
491                start,
492                end: eff_end,
493                mm,
494                downsample: self.downsample,
495                silence,
496            });
497            denormalise_bands(DenormaliseBandsInput {
498                x: &x[n..],
499                freq: f1,
500                band_log_e: &self.old_band_e[NB_EBANDS..],
501                start,
502                end: eff_end,
503                mm,
504                downsample: self.downsample,
505                silence,
506            });
507            for i in 0..n {
508                f0[i] = 0.5 * (f0[i] + f1[i]);
509            }
510            let mem = &mut self.decode_mem[..ml];
511            for b in 0..bcount {
512                m.mdct[shift].backward(&f0[b..], bcount, &mut mem[out_off + nb * b..], &m.window, OVERLAP, &mut self.imdct);
513            }
514        } else {
515            for ch in 0..cc {
516                denormalise_bands(DenormaliseBandsInput {
517                    x: &x[ch * n..],
518                    freq: f0,
519                    band_log_e: &self.old_band_e[ch * NB_EBANDS..],
520                    start,
521                    end: eff_end,
522                    mm,
523                    downsample: self.downsample,
524                    silence,
525                });
526                let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
527                for b in 0..bcount {
528                    m.mdct[shift].backward(&f0[b..], bcount, &mut mem[out_off + nb * b..], &m.window, OVERLAP, &mut self.imdct);
529                }
530            }
531        }
532        self.freq = freq;
533    }

535    fn deemphasis(&mut self, n: usize, pcm: &mut [f32], accum: bool) {
536        let cc = self.channels;
537        let ml = self.mem_len();
538        let ds = self.downsample;
539        let out_off = DECODE_BUFFER_SIZE - n;
540        if ds == 1 && cc == 2 {
541            // Both channels in one loop: two independent IIR chains overlap their latency.
542            let (m0, m1) = self.decode_mem.split_at(ml);
543            let left = &m0[out_off..out_off + n];
544            let right = &m1[out_off..out_off + n];
545            let [mut s0, mut s1] = self.preemph_mem;
546            for ((o, &v0), &v1) in pcm[..2 * n].as_chunks_mut::<2>().0.iter_mut().zip(left).zip(right) {
547                let t0 = v0 + 1e-30 + s0;
548                let t1 = v1 + 1e-30 + s1;
549                s0 = PREEMPH * t0;
550                s1 = PREEMPH * t1;
551                if accum {
552                    o[0] += t0 * (1.0 / 32768.0);
553                    o[1] += t1 * (1.0 / 32768.0);
554                } else {
555                    o[0] = t0 * (1.0 / 32768.0);
556                    o[1] = t1 * (1.0 / 32768.0);
557                }
558            }
559            self.preemph_mem = [s0, s1];
560            return;
561        }
562        if ds == 1 && cc == 1 {
563            let mem = &self.decode_mem[out_off..out_off + n];
564            let mut state = self.preemph_mem[0];
565            for (o, &v) in pcm[..n].iter_mut().zip(mem) {
566                let t = v + 1e-30 + state;
567                state = PREEMPH * t;
568                if accum {
569                    *o += t * (1.0 / 32768.0);
570                } else {
571                    *o = t * (1.0 / 32768.0);
572                }
573            }
574            self.preemph_mem[0] = state;
575            return;
576        }
577        let mut scratch = std::mem::take(&mut self.pcm_scratch);
578        for ch in 0..cc {
579            let mem = &self.decode_mem[ch * ml + out_off..ch * ml + out_off + n];
580            let mut mstate = self.preemph_mem[ch];
581            scratch.clear();
582            for &v in mem {
583                let tmp = v + 1e-30 + mstate;
584                mstate = PREEMPH * tmp;
585                scratch.push(tmp);
586            }
587            self.preemph_mem[ch] = mstate;
588            for j in 0..n / ds {
589                let v = scratch[j * ds] * (1.0 / 32768.0);
590                let o = &mut pcm[j * cc + ch];
591                if accum {
592                    *o += v;
593                } else {
594                    *o = v;
595                }
596            }
597        }
598        self.pcm_scratch = scratch;
599    }

601    /// Packet-loss concealment: pitch-periodic extension for the first lost frames, then
602    /// decaying shaped noise.
603    fn decode_lost(&mut self, n: usize, lm: usize) {
604        let cc = self.channels;
605        let ml = self.mem_len();
606        let m = self.m;
607        let pitch_based = self.loss_count < 5 && self.start == 0 && !self.skip_plc;
608        if !pitch_based {
609            let decay = if self.loss_count == 0 { 1.5 } else { 0.5 };
610            for ch in 0..cc {
611                for i in self.start..self.end {
612                    let k = ch * NB_EBANDS + i;
613                    self.old_band_e[k] = self.background_log_e[k].max(self.old_band_e[k] - decay);
614                }
615            }
616            let mut seed = self.rng;
617            let mut x = std::mem::take(&mut self.x);
618            x.clear();
619            x.resize(cc * n, 0.0);
620            let eff_end = self.end.min(NB_EBANDS);
621            for ch in 0..cc {
622                for i in self.start..eff_end {
623                    let off = n * ch + ((EBANDS[i] as usize) << lm);
624                    let blen = ((EBANDS[i + 1] - EBANDS[i]) as usize) << lm;
625                    for v in x[off..off + blen].iter_mut() {
626                        seed = bands::lcg_rand(seed);
627                        *v = ((seed as i32) >> 20) as f32;
628                    }
629                    renormalise_vector(&mut x[off..off + blen], 1.0);
630                }
631            }
632            self.rng = seed;
633            for ch in 0..cc {
634                let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
635                mem.copy_within(n..DECODE_BUFFER_SIZE + OVERLAP / 2, 0);
636            }
637            let saved_sc = self.stream_channels;
638            self.stream_channels = cc;
639            self.synthesis(SynthesisFrame {
640                bands: &x,
641                samples: n,
642                start_band: self.start,
643                end_band: eff_end,
644                coded_channels: cc,
645                transient: false,
646                lm,
647                silence: false,
648            });
649            self.stream_channels = saved_sc;
650            self.x = x;
651            // As libopus: after noise, the history holds no pitch to extend until two good frames
652            // in a row have been decoded.
653            self.skip_plc = true;
654        } else {
655            // Pitch search on the (mono-mixed) history.
656            let hist_len = 1024usize;
657            let base = DECODE_BUFFER_SIZE - hist_len;
658            let mut mono = vec![0f32; hist_len];
659            for ch in 0..cc {
660                for (i, v) in mono.iter_mut().enumerate() {
661                    *v += self.decode_mem[ch * ml + base + i];
662                }
663            }
664            // The post-filter state is left alone: the next good frame cross-fades from it.
665            let period = if self.loss_count == 0 { find_pitch(&mono) } else { self.plc_pitch };
666            let period = period.clamp(PLC_PITCH_LAG_MIN, PLC_PITCH_LAG_MAX);
667            self.plc_pitch = period;
668            let fade = 0.8f32.powi(self.loss_count as i32 + 1);
669            let mut ext = vec![0f32; n + OVERLAP];
670            for ch in 0..cc {
671                let mem = &mut self.decode_mem[ch * ml..(ch + 1) * ml];
672                // Energy decay across the last two periods.
673                let e1: f32 = mem[DECODE_BUFFER_SIZE - period..DECODE_BUFFER_SIZE].iter().map(|v| v * v).sum();
674                let e0: f32 = mem[DECODE_BUFFER_SIZE - 2 * period..DECODE_BUFFER_SIZE - period].iter().map(|v| v * v).sum();
675                let decay = if e0 > 0.0 { (e1 / e0).sqrt().min(1.0) } else { 0.0 };
676                let mut g = 1.0f32;
677                for (i, e) in ext.iter_mut().enumerate() {
678                    if i > 0 && i % period == 0 {
679                        g *= decay;
680                    }
681                    let src = DECODE_BUFFER_SIZE - period + (i % period);
682                    *e = mem[src] * g * fade;
683                }
684                mem.copy_within(n..DECODE_BUFFER_SIZE + OVERLAP / 2, 0);
685                let out_off = DECODE_BUFFER_SIZE - n;
686                // Cross-fade the start with the previous overlap so the transition is smooth.
687                mem[out_off..out_off + n].copy_from_slice(&ext[..n]);
688                let tail = &ext[n..n + OVERLAP];
689                for i in 0..OVERLAP / 2 {
690                    mem[DECODE_BUFFER_SIZE + i] = m.window[i] * tail[OVERLAP - 1 - i] + m.window[OVERLAP - 1 - i] * tail[i];
691                }
692            }
693        }
694        self.loss_count += 1;
695    }
696}

698fn find_pitch(x: &[f32]) -> usize {
699    let n = x.len();
700    let win = 512.min(n - PLC_PITCH_LAG_MAX);
701    let tgt = &x[n - win..];
702    let mut best = PLC_PITCH_LAG_MIN;
703    let mut best_score = f32::MIN;
704    let e_t: f32 = tgt.iter().map(|v| v * v).sum::<f32>() + 1e-9;
705    for lag in PLC_PITCH_LAG_MIN..=PLC_PITCH_LAG_MAX {
706        let src = &x[n - win - lag..n - lag];
707        let mut xy = 0f32;
708        let mut yy = 1e-9f32;
709        for (a, b) in tgt.iter().zip(src) {
710            xy += a * b;
711            yy += b * b;
712        }
713        let score = xy / (e_t * yy).sqrt();
714        if score > best_score {
715            best_score = score;
716            best = lag;
717        }
718    }
719    best
720}

722fn tf_decode(start: usize, end: usize, is_transient: bool, tf_res: &mut [i32; NB_EBANDS], lm: usize, dec: &mut RangeDecoder) {
723    let mut budget = dec.storage() as i32 * 8;
724    let mut tell = dec.tell();
725    let mut logp: u32 = if is_transient { 2 } else { 4 };
726    let tf_select_rsv = lm > 0 && tell + (logp as i32) < budget;
727    budget -= tf_select_rsv as i32;
728    let mut tf_changed = 0;
729    let mut curr = 0;
730    for r in tf_res.iter_mut().take(end).skip(start) {
731        if tell + logp as i32 <= budget {
732            curr ^= dec.bit_logp(logp) as i32;
733            tell = dec.tell();
734            tf_changed |= curr;
735        }
736        *r = curr;
737        logp = if is_transient { 4 } else { 5 };
738    }
739    let mut tf_select = 0;
740    let ti = 4 * is_transient as usize;
741    if tf_select_rsv && TF_SELECT_TABLE[lm][ti + tf_changed as usize] != TF_SELECT_TABLE[lm][ti + 2 + tf_changed as usize] {
742        tf_select = dec.bit_logp(1) as usize;
743    }
744    for r in tf_res.iter_mut().take(end).skip(start) {
745        *r = TF_SELECT_TABLE[lm][ti + 2 * tf_select + *r as usize] as i32;
746    }
747}

749/// Pitch post-filter (in place on `buf[off .. off + n]`, reading history before `off`).
750struct CombFilterParams<'a> {
751    offset: usize,
752    old_period: i32,
753    new_period: i32,
754    samples: usize,
755    old_gain: f32,
756    new_gain: f32,
757    old_tapset: usize,
758    new_tapset: usize,
759    window: &'a [f32; OVERLAP],
760}
761
762fn comb_filter(buf: &mut [f32], params: CombFilterParams<'_>) {
763    let CombFilterParams {
764        offset: off,
765        old_period: t0,
766        new_period: t1,
767        samples: n,
768        old_gain: g0,
769        new_gain: g1,
770        old_tapset: tapset0,
771        new_tapset: tapset1,
772        window,
773    } = params;
774    if g0 == 0.0 && g1 == 0.0 {
775        return;
776    }
777    let t0 = t0.max(COMBFILTER_MINPERIOD) as usize;
778    let t1 = t1.max(COMBFILTER_MINPERIOD) as usize;
779    let g00 = g0 * COMB_GAINS[tapset0][0];
780    let g01 = g0 * COMB_GAINS[tapset0][1];
781    let g02 = g0 * COMB_GAINS[tapset0][2];
782    let g10 = g1 * COMB_GAINS[tapset1][0];
783    let g11 = g1 * COMB_GAINS[tapset1][1];
784    let g12 = g1 * COMB_GAINS[tapset1][2];
785    let mut x1 = buf[off + 1 - t1];
786    let mut x2 = buf[off - t1];
787    let mut x3 = buf[off - t1 - 1];
788    let mut x4 = buf[off - t1 - 2];
789    let overlap = if g0 == g1 && t0 == t1 && tapset0 == tapset1 { 0 } else { OVERLAP.min(n) };
790    let mut i = 0;
791    while i < overlap {
792        let p = off + i;
793        let x0 = buf[p + 2 - t1];
794        let f = window[i] * window[i];
795        let a = 1.0 - f;
796        buf[p] = buf[p]
797            + a * g00 * buf[p - t0]
798            + a * g01 * (buf[p + 1 - t0] + buf[p - t0 - 1])
799            + a * g02 * (buf[p + 2 - t0] + buf[p - t0 - 2])
800            + f * g10 * x2
801            + f * g11 * (x1 + x3)
802            + f * g12 * (x0 + x4);
803        x4 = x3;
804        x3 = x2;
805        x2 = x1;
806        x1 = x0;
807        i += 1;
808    }
809    if g1 == 0.0 {
810        return;
811    }
812    // Constant filter for the rest of the block.
813    let t = t1;
814    for i in overlap..n {
815        let p = off + i;
816        buf[p] = buf[p] + g10 * buf[p - t] + g11 * (buf[p + 1 - t] + buf[p - t - 1]) + g12 * (buf[p + 2 - t] + buf[p - t - 2]);
817    }
818}
