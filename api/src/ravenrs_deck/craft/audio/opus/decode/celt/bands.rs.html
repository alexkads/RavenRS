16/// Largest band partition (n) and pulse count (k) that can be coded: the last band of a 20 ms
17/// frame has 176 bins, and `get_pulses` yields at most 128 pulses.
18const PVQ_MAX_N: usize = 176;
19const PVQ_MAX_K: usize = 128;
20const PVQ_W: usize = PVQ_MAX_K + 1;
21
22/// V(n, k) for n <= PVQ_MAX_N, k <= PVQ_MAX_K, saturated at `u32::MAX`, built once.
23///
24/// Every (n, k) the bitstream can code has V(n, k) < 2^32 (see `rate::fits_in32`), and the
25/// decoder only reads entries V(n', k') <= V(n, k), so the saturation is never observable.
26fn pvq_table() -> &'static [u32] {
27    static TABLE: std::sync::OnceLock<Vec<u32>> = std::sync::OnceLock::new();
28    TABLE.get_or_init(|| {
29        let mut v = vec![0u32; (PVQ_MAX_N + 1) * PVQ_W];
30        v[0] = 1;
31        for n in 1..=PVQ_MAX_N {
32            v[n * PVQ_W] = 1;
33            for k in 1..PVQ_W {
34                let a = v[(n - 1) * PVQ_W + k];
35                let b = v[n * PVQ_W + k - 1];
36                let c = v[(n - 1) * PVQ_W + k - 1];
37                v[n * PVQ_W + k] = a.saturating_add(b).saturating_add(c);
38            }
39        }
40        v
41    })
42}

44/// Decodes a PVQ codeword of `k` pulses in `n` dimensions; returns the sum of squares.
45fn decode_pulses(y: &mut [i32], n: usize, k: usize, dec: &mut RangeDecoder) -> f32 {
46    debug_assert!(n <= PVQ_MAX_N && k <= PVQ_MAX_K);
47    let table = pvq_table();
48    let vv = |nn: usize, kx: usize| table[nn * PVQ_W + kx] as u64;
49    let total = vv(n, k);
50    let ft = total.min(u32::MAX as u64) as u32;
51    let mut i = if ft >= 2 { dec.uint(ft) as u64 } else { 0 };
52    let mut kk = k;
53    let mut ryy = 0f32;
54    for j in 0..n {
55        let rem = n - j;
56        if kk == 0 {
57            y[j] = 0;
58            continue;
59        }
60        let mut p = (vv(rem - 1, kk) + vv(rem, kk)) / 2;
61        let neg = i >= p;
62        if neg {
63            i -= p;
64        }
65        let k0 = kk;
66        p -= vv(rem - 1, kk);
67        while p > i && kk > 0 {
68            kk -= 1;
69            p -= vv(rem - 1, kk);
70        }
71        let mag = (k0 - kk) as i32;
72        y[j] = if neg { -mag } else { mag };
73        i -= p.min(i);
74        ryy += (mag * mag) as f32;
75    }
76    ryy
77}

79fn exp_rotation1(x: &mut [f32], len: usize, stride: usize, c: f32, s: f32) {
80    let ms = -s;
81    if len <= stride {
82        return;
83    }
84    for i in 0..len - stride {
85        let x1 = x[i];
86        let x2 = x[i + stride];
87        x[i + stride] = c * x2 + s * x1;
88        x[i] = c * x1 + ms * x2;
89    }
90    if len > 2 * stride {
91        for i in (0..=len - 2 * stride - 1).rev() {
92            let x1 = x[i];
93            let x2 = x[i + stride];
94            x[i + stride] = c * x2 + s * x1;
95            x[i] = c * x1 + ms * x2;
96        }
97    }
98}

100/// Inverse spreading rotation (decoder direction).
101fn exp_rotation(x: &mut [f32], len: usize, stride: usize, k: usize, spread: usize) {
102    const SPREAD_FACTOR: [usize; 3] = [15, 10, 5];
103    if 2 * k >= len || spread == SPREAD_NONE {
104        return;
105    }
106    let factor = SPREAD_FACTOR[spread - 1];
107    let gain = len as f32 / (len + factor * k) as f32;
108    let theta = 0.5 * gain * gain;
109    let c = (0.5 * std::f32::consts::PI * theta).cos();
110    let s = (0.5 * std::f32::consts::PI * (1.0 - theta)).cos();
111    let mut stride2 = 0;
112    if len >= 8 * stride {
113        stride2 = 1;
114        while (stride2 * stride2 + stride2) * stride + (stride >> 2) < len {
115            stride2 += 1;
116        }
117    }
118    let sub = len / stride;
119    for i in 0..stride {
120        let xs = &mut x[i * sub..(i + 1) * sub];
121        if stride2 != 0 {
122            exp_rotation1(xs, sub, stride2, s, c);
123        }
124        exp_rotation1(xs, sub, 1, c, s);
125    }
126}

128fn extract_collapse_mask(iy: &[i32], n: usize, b: usize) -> u32 {
129    if b <= 1 {
130        return 1;
131    }
132    let n0 = n / b;
133    let mut mask = 0u32;
134    for i in 0..b {
135        if iy[i * n0..(i + 1) * n0].iter().any(|&v| v != 0) {
136            mask |= 1 << i;
137        }
138    }
139    mask
140}

142pub fn renormalise_vector(x: &mut [f32], gain: f32) {
143    let e: f32 = 1e-15 + x.iter().map(|v| v * v).sum::<f32>();
144    let g = gain / e.sqrt();
145    for v in x.iter_mut() {
146        *v *= g;
147    }
148}

150fn haar1(x: &mut [f32], n0: usize, stride: usize) {
151    let n0 = n0 >> 1;
152    const S: f32 = std::f32::consts::FRAC_1_SQRT_2;
153    for i in 0..stride {
154        for j in 0..n0 {
155            let a = x[stride * 2 * j + i] * S;
156            let b = x[stride * (2 * j + 1) + i] * S;
157            x[stride * 2 * j + i] = a + b;
158            x[stride * (2 * j + 1) + i] = a - b;
159        }
160    }
161}

167fn deinterleave_hadamard(x: &mut [f32], n0: usize, stride: usize, hadamard: bool, tmp: &mut Vec<f32>) {
168    let n = n0 * stride;
169    tmp.clear();
170    tmp.resize(n, 0.0);
171    if hadamard {
172        let ord = ordery(stride);
173        for i in 0..stride {
174            for j in 0..n0 {
175                tmp[ord[i] * n0 + j] = x[j * stride + i];
176            }
177        }
178    } else {
179        for i in 0..stride {
180            for j in 0..n0 {
181                tmp[i * n0 + j] = x[j * stride + i];
182            }
183        }
184    }
185    x[..n].copy_from_slice(tmp);
186}

188fn interleave_hadamard(x: &mut [f32], n0: usize, stride: usize, hadamard: bool, tmp: &mut Vec<f32>) {
189    let n = n0 * stride;
190    tmp.clear();
191    tmp.resize(n, 0.0);
192    if hadamard {
193        let ord = ordery(stride);
194        for i in 0..stride {
195            for j in 0..n0 {
196                tmp[j * stride + i] = x[ord[i] * n0 + j];
197            }
198        }
199    } else {
200        for i in 0..stride {
201            for j in 0..n0 {
202                tmp[j * stride + i] = x[i * n0 + j];
203            }
204        }
205    }
206    x[..n].copy_from_slice(tmp);
207}

210fn frac_mul16(a: i32, b: i32) -> i32 {
211    (16384 + (a as i16 as i32) * (b as i16 as i32)) >> 15
212}

214fn bitexact_cos(x: i32) -> i32 {
215    let tmp = (4096 + x * x) >> 13;
216    let x2 = tmp;
217    let x2 = (32767 - x2) + frac_mul16(x2, -7651 + frac_mul16(x2, 8277 + frac_mul16(-626, x2)));
218    1 + x2
219}

221fn bitexact_log2tan(isin: i32, icos: i32) -> i32 {
222    let lc = ilog(icos as u32);
223    let ls = ilog(isin as u32);
224    let icos = icos << (15 - lc);
225    let isin = isin << (15 - ls);
226    (ls - lc) * (1 << 11) + frac_mul16(isin, frac_mul16(isin, -2597) + 7932) - frac_mul16(icos, frac_mul16(icos, -2597) + 7932)
227}

229fn isqrt32(v: u32) -> u32 {
230    let mut r = (v as f64).sqrt() as u32;
231    while (r as u64) * (r as u64) > v as u64 {
232        r -= 1;
233    }
234    while ((r + 1) as u64) * ((r + 1) as u64) <= v as u64 {
235        r += 1;
236    }
237    r
238}

240fn compute_qn(n: i32, b: i32, offset: i32, pulse_cap: i32, stereo: bool) -> i32 {
241    const EXP2_TABLE8: [i32; 8] = [16384, 17866, 19483, 21247, 23170, 25267, 27554, 30048];
242    let mut n2 = 2 * n - 1;
243    if stereo && n == 2 {
244        n2 -= 1;
245    }
246    let mut qb = sdiv(b + n2 * offset, n2);
247    qb = qb.min(b - pulse_cap - (4 << BITRES));
248    qb = qb.min(8 << BITRES);
249    if qb < (1 << BITRES >> 1) {
250        1
251    } else {
252        let qn = EXP2_TABLE8[(qb & 0x7) as usize] >> (14 - (qb >> BITRES));
253        (qn + 1) >> 1 << 1
254    }
255}

327    fn compute_theta(&mut self, n: i32, b: &mut i32, band: ThetaBand, fill: &mut u32) -> Split {
328        let ThetaBand { bb, b0, lm, stereo } = band;
329        let i = self.i;
330        let pulse_cap = self.m.log_n[i] + lm * (1 << BITRES);
331        let offset = (pulse_cap >> 1) - if stereo && n == 2 { QTHETA_OFFSET_TWOPHASE } else { QTHETA_OFFSET };
332        let mut qn = compute_qn(n, *b, offset, pulse_cap, stereo);
333        if stereo && i >= self.intensity {
334            qn = 1;
335        }
336        let tell = self.dec.tell_frac() as i32;
337        let mut itheta = 0i32;
338        let mut inv = false;
339        if qn != 1 {
340            if stereo && n > 2 {
341                let p0 = 3u32;
342                let x0 = (qn / 2) as u32;
343                let ft = p0 * (x0 + 1) + x0;
344                let fs = self.dec.decode(ft);
345                let x = if fs < (x0 + 1) * p0 { fs / p0 } else { x0 + 1 + (fs - (x0 + 1) * p0) };
346                let (fl, fh) = if x <= x0 { (p0 * x, p0 * (x + 1)) } else { ((x - 1 - x0) + (x0 + 1) * p0, (x - x0) + (x0 + 1) * p0) };
347                self.dec.update(fl, fh, ft);
348                itheta = x as i32;
349            } else if b0 > 1 || stereo {
350                itheta = self.dec.uint(qn as u32 + 1) as i32;
351            } else {
352                let half = (qn >> 1) as u32;
353                let ft = (half + 1) * (half + 1);
354                let fm = self.dec.decode(ft);
355                let (fl, fs);
356                if fm < (half * (half + 1)) >> 1 {
357                    let it = (isqrt32(8 * fm + 1) - 1) >> 1;
358                    fs = it + 1;
359                    fl = (it * (it + 1)) >> 1;
360                    itheta = it as i32;
361                } else {
362                    let q = qn as u32;
363                    let it = (2 * (q + 1) - isqrt32(8 * (ft - fm - 1) + 1)) >> 1;
364                    fs = q + 1 - it;
365                    fl = ft - (((q + 1 - it) * (q + 2 - it)) >> 1);
366                    itheta = it as i32;
367                }
368                self.dec.update(fl, fl + fs, ft);
369            }
370            itheta = ((itheta as u32 * 16384) / qn as u32) as i32;
371        } else if stereo {
372            if *b > 2 << BITRES && self.remaining_bits > 2 << BITRES {
373                inv = self.dec.bit_logp(2);
374            }
375            if self.disable_inv {
376                inv = false;
377            }
378            itheta = 0;
379        }
380        let qalloc = self.dec.tell_frac() as i32 - tell;
381        *b -= qalloc;
382        let (imid, iside, delta);
383        if itheta == 0 {
384            imid = 32767;
385            iside = 0;
386            *fill &= (1 << bb) - 1;
387            delta = -16384;
388        } else if itheta == 16384 {
389            imid = 0;
390            iside = 32767;
391            *fill &= ((1 << bb) - 1) << bb;
392            delta = 16384;
393        } else {
394            imid = bitexact_cos(itheta);
395            iside = bitexact_cos(16384 - itheta);
396            delta = frac_mul16((n - 1) << 7, bitexact_log2tan(iside, imid));
397        }
398        Split { inv, imid, iside, delta, itheta, qalloc }
399    }

401    fn quant_band_n1(&mut self, x: &mut [f32], y: Option<&mut [f32]>, lowband_out: Option<&mut [f32]>) -> u32 {
402        let dec_sign = |ctx: &mut Self| -> f32 {
403            let mut sign = 0;
404            if ctx.remaining_bits >= 1 << BITRES {
405                sign = ctx.dec.bits(1);
406                ctx.remaining_bits -= 1 << BITRES;
407            }
408            if sign != 0 { -1.0 } else { 1.0 }
409        };
410        x[0] = dec_sign(self);
411        if let Some(y) = y {
412            y[0] = dec_sign(self);
413        }
414        if let Some(lo) = lowband_out {
415            lo[0] = x[0];
416        }
417        1
418    }

420    fn quant_partition(&mut self, x: &mut [f32], input: QuantPartitionInput<'_>) -> u32 {
421        let QuantPartitionInput { samples: n, bits: mut b, blocks: mut bb, lowband, mut lm, gain, mut fill } = input;
422        let b0 = bb;
423        let i = self.i;
424        let cache = self.m.cache(lm, i);
425        let cache_max = cache[cache[0] as usize] as i32;
426        if lm != -1 && b > cache_max + 12 && n > 2 {
427            let n = n >> 1;
428            let (xa, ya) = x.split_at_mut(n);
429            lm -= 1;
430            if bb == 1 {
431                fill = (fill & 1) | (fill << 1);
432            }
433            bb = (bb + 1) >> 1;
434            let sp = self.compute_theta(n as i32, &mut b, ThetaBand { bb, b0, lm, stereo: false }, &mut fill);
435            let mid = sp.imid as f32 / 32768.0;
436            let side = sp.iside as f32 / 32768.0;
437            let mut delta = sp.delta;
438            if b0 > 1 && (sp.itheta & 0x3fff) != 0 {
439                if sp.itheta > 8192 {
440                    delta -= delta >> (4 - lm);
441                } else {
442                    delta = 0.min(delta + ((n as i32) << BITRES >> (5 - lm)));
443                }
444            }
445            let mut mbits = 0.max(b.min((b - delta) / 2));
446            let mut sbits = b - mbits;
447            self.remaining_bits -= sp.qalloc;
448            let next_lowband2 = lowband.map(|l| &l[n..]);
449            let mut rebalance = self.remaining_bits;
450            let mut cm;
451            if mbits >= sbits {
452                cm = self.quant_partition(xa, QuantPartitionInput { samples: n, bits: mbits, blocks: bb, lowband, lm, gain: gain * mid, fill });
453                rebalance = mbits - (rebalance - self.remaining_bits);
454                if rebalance > 3 << BITRES && sp.itheta != 0 {
455                    sbits += rebalance - (3 << BITRES);
456                }
457                cm |= self.quant_partition(
458                    ya,
459                    QuantPartitionInput { samples: n, bits: sbits, blocks: bb, lowband: next_lowband2, lm, gain: gain * side, fill: fill >> bb },
460                ) << (b0 >> 1);
461            } else {
462                cm = self.quant_partition(
463                    ya,
464                    QuantPartitionInput { samples: n, bits: sbits, blocks: bb, lowband: next_lowband2, lm, gain: gain * side, fill: fill >> bb },
465                ) << (b0 >> 1);
466                rebalance = sbits - (rebalance - self.remaining_bits);
467                if rebalance > 3 << BITRES && sp.itheta != 16384 {
468                    mbits += rebalance - (3 << BITRES);
469                }
470                cm |= self.quant_partition(xa, QuantPartitionInput { samples: n, bits: mbits, blocks: bb, lowband, lm, gain: gain * mid, fill });
471            }
472            return cm;
473        }
474        // No split.
475        let m = self.m;
476        let mut q = m.bits2pulses(i, lm, b);
477        let mut curr_bits = m.pulses2bits(i, lm, q);
478        self.remaining_bits -= curr_bits;
479        while self.remaining_bits < 0 && q > 0 {
480            self.remaining_bits += curr_bits;
481            q -= 1;
482            curr_bits = m.pulses2bits(i, lm, q);
483            self.remaining_bits -= curr_bits;
484        }
485        if q != 0 {
486            let k = get_pulses(q) as usize;
487            let mut iy = std::mem::take(&mut self.iy);
488            iy.clear();
489            iy.resize(n, 0);
490            let ryy = decode_pulses(&mut iy, n, k, self.dec);
491            let g = gain / ryy.sqrt();
492            for (xv, &yv) in x[..n].iter_mut().zip(iy.iter()) {
493                *xv = g * yv as f32;
494            }
495            exp_rotation(&mut x[..n], n, bb, k, self.spread);
496            let cm = extract_collapse_mask(&iy, n, bb);
497            self.iy = iy;
498            cm
499        } else {
500            let cm_mask = (1u32 << bb) - 1;
501            fill &= cm_mask;
502            if fill == 0 {
503                x[..n].fill(0.0);
504                0
505            } else {
506                let cm = match lowband {
507                    None => {
508                        for v in x[..n].iter_mut() {
509                            self.seed = lcg_rand(self.seed);
510                            *v = ((self.seed as i32) >> 20) as f32;
511                        }
512                        cm_mask
513                    }
514                    Some(lb) => {
515                        for j in 0..n {
516                            self.seed = lcg_rand(self.seed);
517                            let tmp = if self.seed & 0x8000 != 0 { 1.0 / 256.0 } else { -1.0 / 256.0 };
518                            x[j] = lb[j] + tmp;
519                        }
520                        fill
521                    }
522                };
523                renormalise_vector(&mut x[..n], gain);
524                cm
525            }
526        }
527    }

529    fn quant_band(&mut self, x: &mut [f32], input: QuantBandInput<'_>) -> u32 {
530        let QuantBandInput { samples: n, bits: b, blocks: mut bb, mut lowband, lm, lowband_out, gain, mut fill } = input;
531        let n0 = n;
532        let mut n_b = n;
533        let b0_orig = bb;
534        let mut time_divide = 0;
535        let mut recombine = 0;
536        let long_blocks = b0_orig == 1;
537        let mut tf_change = self.tf_change;
538        n_b /= bb;
539        if n == 1 {
540            return self.quant_band_n1(x, None, lowband_out);
541        }
542        if tf_change > 0 {
543            recombine = tf_change;
544        }
545        const BIT_INTERLEAVE: [u32; 16] = [0, 1, 1, 1, 2, 3, 3, 3, 2, 3, 3, 3, 2, 3, 3, 3];
546        for k in 0..recombine {
547            if let Some(lb) = lowband.as_deref_mut() {
548                haar1(lb, n >> k, 1 << k);
549            }
550            fill = BIT_INTERLEAVE[(fill & 0xF) as usize] | (BIT_INTERLEAVE[(fill >> 4) as usize] << 2);
551        }
552        bb >>= recombine;
553        n_b <<= recombine;
554        while (n_b & 1) == 0 && tf_change < 0 {
555            if let Some(lb) = lowband.as_deref_mut() {
556                haar1(lb, n_b, bb);
557            }
558            fill |= fill << bb;
559            bb <<= 1;
560            n_b >>= 1;
561            time_divide += 1;
562            tf_change += 1;
563        }
564        let b0 = bb;
565        let n_b0 = n_b;
566        let mut tmp = std::mem::take(&mut self.tmp);
567        if b0 > 1
568            && let Some(lb) = lowband.as_deref_mut()
569        {
570            deinterleave_hadamard(lb, n_b >> recombine, b0 << recombine, long_blocks, &mut tmp);
571        }
572        let mut cm = self.quant_partition(x, QuantPartitionInput { samples: n, bits: b, blocks: bb, lowband: lowband.as_deref(), lm, gain, fill });
573        if b0 > 1 {
574            interleave_hadamard(x, n_b >> recombine, b0 << recombine, long_blocks, &mut tmp);
575        }
576        self.tmp = tmp;
577        let mut n_b = n_b0;
578        bb = b0;
579        for _ in 0..time_divide {
580            bb >>= 1;
581            n_b <<= 1;
582            cm |= cm >> bb;
583            haar1(x, n_b, bb);
584        }
585        const BIT_DEINTERLEAVE: [u32; 16] = [0x00, 0x03, 0x0C, 0x0F, 0x30, 0x33, 0x3C, 0x3F, 0xC0, 0xC3, 0xCC, 0xCF, 0xF0, 0xF3, 0xFC, 0xFF];
586        for k in 0..recombine {
587            cm = BIT_DEINTERLEAVE[(cm & 0xF) as usize];
588            haar1(x, n0 >> k, 1 << k);
589        }
590        bb <<= recombine;
591        if let Some(lo) = lowband_out {
592            let s = (n0 as f32).sqrt();
593            for j in 0..n0 {
594                lo[j] = s * x[j];
595            }
596        }
597        cm & ((1u32 << bb) - 1)
598    }
599
600    fn quant_band_stereo(&mut self, x: &mut [f32], y: &mut [f32], input: QuantBandStereoInput<'_>) -> u32 {
601        let QuantBandStereoInput { samples: n, bits: mut b, blocks: bb, lowband, lm, lowband_out, mut fill } = input;
602        let orig_fill = fill;
603        if n == 1 {
604            return self.quant_band_n1(x, Some(y), lowband_out);
605        }
606        let sp = self.compute_theta(n as i32, &mut b, ThetaBand { bb, b0: bb, lm, stereo: true }, &mut fill);
607        let mid = sp.imid as f32 / 32768.0;
608        let side = sp.iside as f32 / 32768.0;
609        let mut cm;
610        if n == 2 {
611            let mut mbits = b;
612            let mut sbits = 0;
613            if sp.itheta != 0 && sp.itheta != 16384 {
614                sbits = 1 << BITRES;
615            }
616            mbits -= sbits;
617            let c = sp.itheta > 8192;
618            self.remaining_bits -= sp.qalloc + sbits;
619            let mut sign = 0;
620            if sbits != 0 {
621                sign = self.dec.bits(1) as i32;
622            }
623            let sign = (1 - 2 * sign) as f32;
624            if c {
625                cm = self.quant_band(y, QuantBandInput { samples: n, bits: mbits, blocks: bb, lowband, lm, lowband_out, gain: 1.0, fill: orig_fill });
626                x[0] = -sign * y[1];
627                x[1] = sign * y[0];
628            } else {
629                cm = self.quant_band(x, QuantBandInput { samples: n, bits: mbits, blocks: bb, lowband, lm, lowband_out, gain: 1.0, fill: orig_fill });
630                y[0] = -sign * x[1];
631                y[1] = sign * x[0];
632            }
633            x[0] *= mid;
634            x[1] *= mid;
635            y[0] *= side;
636            y[1] *= side;
637            let t = x[0];
638            x[0] = t - y[0];
639            y[0] += t;
640            let t = x[1];
641            x[1] = t - y[1];
642            y[1] += t;
643        } else {
644            let mut mbits = 0.max(b.min((b - sp.delta) / 2));
645            let mut sbits = b - mbits;
646            self.remaining_bits -= sp.qalloc;
647            let mut rebalance = self.remaining_bits;
648            if mbits >= sbits {
649                cm = self.quant_band(x, QuantBandInput { samples: n, bits: mbits, blocks: bb, lowband, lm, lowband_out, gain: 1.0, fill });
650                rebalance = mbits - (rebalance - self.remaining_bits);
651                if rebalance > 3 << BITRES && sp.itheta != 0 {
652                    sbits += rebalance - (3 << BITRES);
653                }
654                cm |= self.quant_band(
655                    y,
656                    QuantBandInput { samples: n, bits: sbits, blocks: bb, lowband: None, lm, lowband_out: None, gain: side, fill: fill >> bb },
657                );
658            } else {
659                cm = self.quant_band(
660                    y,
661                    QuantBandInput { samples: n, bits: sbits, blocks: bb, lowband: None, lm, lowband_out: None, gain: side, fill: fill >> bb },
662                );
663                rebalance = sbits - (rebalance - self.remaining_bits);
664                if rebalance > 3 << BITRES && sp.itheta != 16384 {
665                    mbits += rebalance - (3 << BITRES);
666                }
667                cm |= self.quant_band(x, QuantBandInput { samples: n, bits: mbits, blocks: bb, lowband, lm, lowband_out, gain: 1.0, fill });
668            }
669        }
670        if n != 2 {
671            stereo_merge(&mut x[..n], &mut y[..n], mid);
672        }
673        if sp.inv {
674            for v in y[..n].iter_mut() {
675                *v = -*v;
676            }
677        }
678        cm
679    }
680}
681
682fn stereo_merge(x: &mut [f32], y: &mut [f32], mid: f32) {
683    let mut xp = 0f32;
684    let mut side = 0f32;
685    for (a, b) in x.iter().zip(y.iter()) {
686        xp += a * b;
687        side += b * b;
688    }
689    xp *= mid;
690    let mid2 = mid;
691    let el = mid2 * mid2 + side - 2.0 * xp;
692    let er = mid2 * mid2 + side + 2.0 * xp;
693    if er < 6e-4 || el < 6e-4 {
694        y.copy_from_slice(x);
695        return;
696    }
697    let lgain = 1.0 / el.sqrt();
698    let rgain = 1.0 / er.sqrt();
699    for (a, b) in x.iter_mut().zip(y.iter_mut()) {
700        let l = mid * *a;
701        let r = *b;
702        *a = lgain * (l - r);
703        *b = rgain * (l + r);
704    }
705}

707/// Decodes all band shapes (`quant_all_bands`, decoder side).
708pub struct QuantAllBandsInput<'a, 'd> {
709    pub mode: &'a Mode,
710    pub start: usize,
711    pub end: usize,
712    pub x_buf: &'a mut [f32],
713    pub y_buf: Option<&'a mut [f32]>,
714    pub collapse_masks: &'a mut [u32],
715    pub pulses: &'a [i32; NB_EBANDS],
716    pub short_blocks: bool,
717    pub spread: usize,
718    pub dual_stereo: bool,
719    pub intensity: usize,
720    pub tf_res: &'a [i32; NB_EBANDS],
721    pub total_bits: i32,
722    pub balance: i32,
723    pub decoder: &'a mut RangeDecoder<'d>,
724    pub lm: usize,
725    pub coded_bands: usize,
726    pub seed: &'a mut u32,
727    pub disable_inv: bool,
728}
729
730pub fn quant_all_bands(input: QuantAllBandsInput<'_, '_>) {
731    let QuantAllBandsInput {
732        mode: m,
733        start,
734        end,
735        x_buf,
736        y_buf,
737        collapse_masks,
738        pulses,
739        short_blocks,
740        spread,
741        mut dual_stereo,
742        intensity,
743        tf_res,
744        total_bits,
745        mut balance,
746        decoder: dec,
747        lm,
748        coded_bands,
749        seed,
750        disable_inv,
751    } = input;
752    let mm = 1usize << lm;
753    let bb = if short_blocks { mm } else { 1 };
754    let norm_offset = mm * EBANDS[start] as usize;
755    let norm_len = mm * EBANDS[NB_EBANDS - 1] as usize - norm_offset;
756    let c = if y_buf.is_some() { 2 } else { 1 };
757    let mut norm = vec![0f32; norm_len];
758    let mut norm2 = vec![0f32; if c == 2 { norm_len } else { 0 }];
759    let mut scratch = vec![0f32; mm * (EBANDS[NB_EBANDS] - EBANDS[NB_EBANDS - 1]) as usize];
760    let mut scratch2 = scratch.clone();
761    let mut y_buf = y_buf;
762    let mut lowband_offset = 0usize;
763    let mut update_lowband = true;
764    let mut ctx = BandCtx {
765        m,
766        dec,
767        i: 0,
768        intensity,
769        spread,
770        tf_change: 0,
771        remaining_bits: 0,
772        seed: *seed,
773        disable_inv,
774        avoid_split_noise: bb > 1,
775        iy: Vec::new(),
776        tmp: Vec::new(),
777    };
778    for i in start..end {
779        ctx.i = i;
780        let last = i == end - 1;
781        let xo = mm * EBANDS[i] as usize;
782        let n = mm * EBANDS[i + 1] as usize - xo;
783        let tell = ctx.dec.tell_frac() as i32;
784        if i != start {
785            balance -= tell;
786        }
787        let remaining_bits = total_bits - tell - 1;
788        ctx.remaining_bits = remaining_bits;
789        let b = if i < coded_bands {
790            let curr_balance = sdiv(balance, 3.min(coded_bands as i32 - i as i32));
791            0.max(16383.min((remaining_bits + 1).min(pulses[i] + curr_balance)))
792        } else {
793            0
794        };
795        if (mm * EBANDS[i] as usize >= n + norm_offset || i == start + 1) && (update_lowband || lowband_offset == 0) {
796            lowband_offset = i;
797        }
798        if i == start + 1 {
799            let n1 = mm * (EBANDS[start + 1] - EBANDS[start]) as usize;
800            let n2 = mm * (EBANDS[start + 2] - EBANDS[start + 1]) as usize;
801            if n2 > n1 {
802                norm.copy_within(2 * n1 - n2..n1, n1);
803                if c == 2 {
804                    norm2.copy_within(2 * n1 - n2..n1, n1);
805                }
806            }
807        }
808        let tf_change = tf_res[i];
809        ctx.tf_change = tf_change;
810        let mut effective_lowband: Option<usize> = None;
811        let (mut x_cm, mut y_cm);
812        if lowband_offset != 0 && (spread != SPREAD_AGGRESSIVE || bb > 1 || tf_change < 0) {
813            let eff = (mm * EBANDS[lowband_offset] as usize).saturating_sub(norm_offset + n);
814            effective_lowband = Some(eff);
815            let mut fold_start = lowband_offset;
816            loop {
817                fold_start -= 1;
818                if mm * (EBANDS[fold_start] as usize) <= eff + norm_offset {
819                    break;
820                }
821            }
822            let mut fold_end = lowband_offset - 1;
823            loop {
824                fold_end += 1;
825                if !(fold_end < i && mm * (EBANDS[fold_end] as usize) < eff + norm_offset + n) {
826                    break;
827                }
828            }
829            x_cm = 0;
830            y_cm = 0;
831            let mut fi = fold_start;
832            loop {
833                x_cm |= collapse_masks[fi * c];
834                y_cm |= collapse_masks[fi * c + c - 1];
835                fi += 1;
836                if fi >= fold_end {
837                    break;
838                }
839            }
840        } else {
841            x_cm = (1u32 << bb) - 1;
842            y_cm = x_cm;
843        }
844        if dual_stereo && i == intensity {
845            dual_stereo = false;
846            for j in 0..(mm * EBANDS[i] as usize - norm_offset) {
847                norm[j] = 0.5 * (norm[j] + norm2[j]);
848            }
849        }
850        let out_off = xo - norm_offset;
851        let x = &mut x_buf[xo..xo + n];
852        if dual_stereo && let Some(y_buf) = y_buf.as_deref_mut() {
853            let y = &mut y_buf[xo..xo + n];
854            let lb = effective_lowband.map(|e| {
855                scratch[..n].copy_from_slice(&norm[e..e + n]);
856                &mut scratch[..n]
857            });
858            let lo = if last { None } else { Some(&mut norm[out_off..out_off + n]) };
859            x_cm = ctx.quant_band(
860                x,
861                QuantBandInput { samples: n, bits: b / 2, blocks: bb, lowband: lb, lm: lm as i32, lowband_out: lo, gain: 1.0, fill: x_cm },
862            );
863            let lb2 = effective_lowband.map(|e| {
864                scratch2[..n].copy_from_slice(&norm2[e..e + n]);
865                &mut scratch2[..n]
866            });
867            let lo2 = if last { None } else { Some(&mut norm2[out_off..out_off + n]) };
868            y_cm = ctx.quant_band(
869                y,
870                QuantBandInput { samples: n, bits: b / 2, blocks: bb, lowband: lb2, lm: lm as i32, lowband_out: lo2, gain: 1.0, fill: y_cm },
871            );
872        } else {
873            let lb = effective_lowband.map(|e| {
874                scratch[..n].copy_from_slice(&norm[e..e + n]);
875                &mut scratch[..n]
876            });
877            let lo = if last { None } else { Some(&mut norm[out_off..out_off + n]) };
878            if let Some(yb) = y_buf.as_deref_mut() {
879                let y = &mut yb[xo..xo + n];
880                x_cm = ctx.quant_band_stereo(
881                    x,
882                    y,
883                    QuantBandStereoInput { samples: n, bits: b, blocks: bb, lowband: lb, lm: lm as i32, lowband_out: lo, fill: x_cm | y_cm },
884                );
885            } else {
886                x_cm = ctx.quant_band(
887                    x,
888                    QuantBandInput { samples: n, bits: b, blocks: bb, lowband: lb, lm: lm as i32, lowband_out: lo, gain: 1.0, fill: x_cm | y_cm },
889                );
890            }
891            y_cm = x_cm;
892        }
893        collapse_masks[i * c] = x_cm;
894        collapse_masks[i * c + c - 1] = y_cm;
895        balance += pulses[i] + tell;
896        update_lowband = b > (n as i32) << BITRES;
897        ctx.avoid_split_noise = false;
898    }
899    *seed = ctx.seed;
900}

902/// Anti-collapse noise injection (RFC 6716 §4.3.5).
903pub struct AntiCollapseInput<'a> {
904    pub x_buf: &'a mut [f32],
905    pub collapse_masks: &'a [u32],
906    pub lm: usize,
907    pub channels: usize,
908    pub size: usize,
909    pub start: usize,
910    pub end: usize,
911    pub log_e: &'a [f32],
912    pub prev1: &'a [f32],
913    pub prev2: &'a [f32],
914    pub pulses: &'a [i32; NB_EBANDS],
915    pub seed: u32,
916}
917
918pub fn anti_collapse(input: AntiCollapseInput<'_>) {
919    let AntiCollapseInput { x_buf, collapse_masks, lm, channels: c, size, start, end, log_e, prev1, prev2, pulses, mut seed } = input;
920    for i in start..end {
921        let n0 = (EBANDS[i + 1] - EBANDS[i]) as usize;
922        let depth = ((1 + pulses[i]) as u32 / n0 as u32) >> lm;
923        let thresh = 0.5 * (-0.125 * depth as f32).exp2();
924        let sqrt_1 = 1.0 / ((n0 << lm) as f32).sqrt();
925        for ch in 0..c {
926            let mut p1 = prev1[ch * NB_EBANDS + i];
927            let mut p2 = prev2[ch * NB_EBANDS + i];
928            if c == 1 {
929                p1 = p1.max(prev1[NB_EBANDS + i]);
930                p2 = p2.max(prev2[NB_EBANDS + i]);
931            }
932            let ediff = (log_e[ch * NB_EBANDS + i] - p1.min(p2)).max(0.0);
933            let mut r = 2.0 * (-ediff).exp2();
934            if lm == 3 {
935                r *= std::f32::consts::SQRT_2;
936            }
937            r = r.min(thresh);
938            r *= sqrt_1;
939            let off = ch * size + ((EBANDS[i] as usize) << lm);
940            let xb = &mut x_buf[off..off + (n0 << lm)];
941            let mut renorm = false;
942            for k in 0..(1usize << lm) {
943                if collapse_masks[i * c + ch] & (1 << k) == 0 {
944                    for j in 0..n0 {
945                        seed = lcg_rand(seed);
946                        xb[(j << lm) + k] = if seed & 0x8000 != 0 { r } else { -r };
947                    }
948                    renorm = true;
949                }
950            }
951            if renorm {
952                renormalise_vector(xb, 1.0);
953            }
954        }
955    }
956}

958/// Multiplies the normalised shapes by the band energies (`denormalise_bands`).
959pub struct DenormaliseBandsInput<'a> {
960    pub x: &'a [f32],
961    pub freq: &'a mut [f32],
962    pub band_log_e: &'a [f32],
963    pub start: usize,
964    pub end: usize,
965    pub mm: usize,
966    pub downsample: usize,
967    pub silence: bool,
968}
969
970pub fn denormalise_bands(input: DenormaliseBandsInput<'_>) {
971    let DenormaliseBandsInput { x, freq, band_log_e, mut start, mut end, mm, downsample, silence } = input;
972    let n = mm * SHORT_MDCT;
973    let mut bound = mm * EBANDS[end] as usize;
974    if downsample != 1 {
975        bound = bound.min(n / downsample);
976    }
977    if silence {
978        bound = 0;
979        start = 0;
980        end = 0;
981    }
982    let s0 = mm * EBANDS[start] as usize;
983    freq[..s0].fill(0.0);
984    for i in start..end {
985        let lo = mm * EBANDS[i] as usize;
986        let hi = mm * EBANDS[i + 1] as usize;
987        let lg = band_log_e[i] + E_MEANS[i];
988        let g = lg.min(32.0).exp2();
989        for j in lo..hi {
990            freq[j] = x[j] * g;
991        }
992    }
993    let from = bound;
994    if from < n {
995        freq[from..n].fill(0.0);
996    }
997}
