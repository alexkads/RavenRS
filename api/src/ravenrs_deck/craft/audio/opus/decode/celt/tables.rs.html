41const fn transpose(t: [[u8; 11]; NB_EBANDS]) -> [[u8; NB_EBANDS]; 11] {
42    let mut out = [[0u8; NB_EBANDS]; 11];
43    let mut b = 0;
44    while b < NB_EBANDS {
45        let mut q = 0;
46        while q < 11 {
47            out[q][b] = t[b][q];
48            q += 1;
49        }
50        b += 1;
51    }
52    out
53}
