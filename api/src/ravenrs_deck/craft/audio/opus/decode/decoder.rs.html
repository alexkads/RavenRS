50fn apply_output_gain(pcm: &mut [f32], samples: usize, channels: usize, gain_q8: i16) {
51    if gain_q8 == 0 {
52        return;
53    }
54    let gain = (6.488_141e-4f32 * gain_q8 as f32).exp2();
55    for sample in pcm.iter_mut().take(samples * channels) {
56        *sample *= gain;
57    }
58}

61    /// Creates a decoder producing `channels` (1 or 2) at `rate` (8/12/16/24/48 kHz).
62    pub fn new(rate: u32, channels: usize) -> Result<StreamDecoder> {
63        if !SAMPLE_RATES.contains(&rate) {
64            return Err(Error::InvalidArgument("sample rate must be 8000, 12000, 16000, 24000 or 48000"));
65        }
66        if !(1..=2).contains(&channels) {
67            return Err(Error::InvalidArgument("channels must be 1 or 2"));
68        }
69        Ok(StreamDecoder {
70            rate,
71            channels,
72            celt: CeltDecoder::new(rate, channels),
73            silk: SilkDecoder::new(rate, channels),
74            stream_channels: channels,
75            bandwidth: Bandwidth::Fullband,
76            mode: None,
77            prev_mode: None,
78            frame_size: rate as usize / 400,
79            prev_redundancy: false,
80            last_packet_duration: 0,
81            range_final: 0,
82            gain_q8: 0,
83        })
84    }

130    /// Decodes a packet into interleaved samples appended to `out`. `None` conceals one lost
131    /// packet of the last packet's duration. Returns samples per channel.
132    pub fn decode_interleaved(&mut self, packet: Option<&[u8]>, out: &mut Vec<f32>) -> Result<usize> {
133        match packet {
134            Some(p) if !p.is_empty() => self.decode_packet(p, out),
135            _ => {
136                let n = if self.last_packet_duration > 0 { self.last_packet_duration } else { self.rate as usize / 50 };
137                self.conceal(n, out)
138            }
139        }
140    }

142    /// Longest concealment or FEC span accepted per call: 120 ms, the longest Opus packet.
143    fn max_span(&self) -> usize {
144        self.rate as usize * 3 / 25
145    }
146
147    /// Conceals `n` samples (a multiple of 2.5 ms, at most 120 ms).
148    pub fn conceal(&mut self, n: usize, out: &mut Vec<f32>) -> Result<usize> {
149        let f2_5 = self.rate as usize / 400;
150        if n == 0 || !n.is_multiple_of(f2_5) {
151            return Err(Error::InvalidArgument("concealment length must be a multiple of 2.5 ms"));
152        }
153        if n > self.max_span() {
154            return Err(Error::InvalidArgument("concealment length exceeds 120 ms"));
155        }
156        let start = out.len();
157        out.resize(start + n * self.channels, 0.0);
158        let mut done = 0;
159        while done < n {
160            let r = self.decode_frame(None, &mut out[start + done * self.channels..], n - done, false)?;
161            done += r;
162        }
163        self.last_packet_duration = n;
164        Ok(n)
165    }

167    /// Decodes the in-band FEC (LBRR) of `packet` for the previous (lost) packet of `n` samples.
168    /// `n` is at most 120 ms.
169    pub fn decode_fec(&mut self, packet: &[u8], n: usize, out: &mut Vec<f32>) -> Result<usize> {
170        if n > self.max_span() {
171            return Err(Error::InvalidArgument("FEC length exceeds 120 ms"));
172        }
173        let p = Packet::parse(packet)?;
174        let pfs = p.toc.frame_samples(self.rate);
175        if n < pfs || p.toc.mode == Mode::CeltOnly || self.mode == Some(Mode::CeltOnly) {
176            return self.conceal(n, out);
177        }
178        let start = out.len();
179        if n > pfs {
180            self.conceal(n - pfs, out)?;
181        }
182        self.mode = Some(p.toc.mode);
183        self.bandwidth = p.toc.bandwidth;
184        self.frame_size = pfs;
185        self.stream_channels = p.toc.channels();
186        let off = out.len();
187        out.resize(off + pfs * self.channels, 0.0);
188        self.decode_frame(Some(p.frames[0]), &mut out[off..], pfs, true)?;
189        self.last_packet_duration = n;
190        debug_assert_eq!(out.len() - start, n * self.channels);
191        Ok(n)
192    }

194    fn decode_packet(&mut self, data: &[u8], out: &mut Vec<f32>) -> Result<usize> {
195        let p = Packet::parse(data)?;
196        let fs = p.toc.frame_samples(self.rate);
197        self.mode = Some(p.toc.mode);
198        self.bandwidth = p.toc.bandwidth;
199        self.frame_size = fs;
200        self.stream_channels = p.toc.channels();
201        let start = out.len();
202        out.resize(start + fs * p.frames.len() * self.channels, 0.0);
203        let mut nb = 0;
204        for f in &p.frames {
205            let r = self.decode_frame(Some(f), &mut out[start + nb * self.channels..], fs, false)?;
206            nb += r;
207        }
208        out.truncate(start + nb * self.channels);
209        self.last_packet_duration = nb;
210        Ok(nb)
211    }

213    fn decode_silk_frame(&mut self, frame: SilkFrame, dec: &mut Option<&mut RangeDecoder>) -> Result<Vec<f32>> {
214        let SilkFrame { mode, bandwidth, has_packet, fec, frame_size, audio_size, rate, channels } = frame;
215        let mut pcm = Vec::new();
216        if mode == Mode::CeltOnly {
217            return Ok(pcm);
218        }
219        pcm.resize(frame_size.max(rate / 100) * channels, 0.0);
220        if self.prev_mode == Some(Mode::CeltOnly) {
221            self.silk.reset();
222        }
223        let internal_rate = if mode == Mode::SilkOnly {
224            match bandwidth {
225                Some(Bandwidth::Narrowband) => 8000,
226                Some(Bandwidth::Mediumband) => 12000,
227                _ => 16000,
228            }
229        } else {
230            16000
231        };
232        if has_packet {
233            self.silk.set_stream(self.stream_channels, internal_rate);
234        }
235        let lost = if !has_packet {
236            LostFlag::Lost
237        } else if fec {
238            LostFlag::Lbrr
239        } else {
240            LostFlag::Normal
241        };
242        let payload_ms = (1000 * audio_size / rate).max(10);
243        let mut decoded = 0;
244        while decoded < frame_size {
245            let first = decoded == 0;
246            let count = match self.silk.decode(dec.as_deref_mut(), lost, first, payload_ms, &mut pcm[decoded * channels..]) {
247                Ok(count) => count,
248                Err(_error) if lost != LostFlag::Normal => {
249                    let remaining = frame_size - decoded;
250                    pcm[decoded * channels..(decoded + remaining) * channels].fill(0.0);
251                    remaining
252                }
253                Err(error) => return Err(error),
254            };
255            if count == 0 {
256                break;
257            }
258            decoded += count;
259        }
260        Ok(pcm)
261    }

263    fn read_redundancy(&self, mode: Mode, fec: bool, packet_len: usize, dec: &mut Option<&mut RangeDecoder>) -> Redundancy {
264        let mut state = Redundancy { present: false, celt_to_silk: false, bytes: 0, packet_len };
265        let Some(decoder) = dec.as_deref_mut() else { return state };
266        let hybrid = self.mode == Some(Mode::Hybrid);
267        if fec || mode == Mode::CeltOnly || decoder.tell() + 17 + 20 * i32::from(hybrid) > 8 * state.packet_len as i32 {
268            return state;
269        }
270        state.present = if mode == Mode::Hybrid { decoder.bit_logp(12) } else { true };
271        if !state.present {
272            return state;
273        }
274        state.celt_to_silk = decoder.bit_logp(1);
275        state.bytes = if mode == Mode::Hybrid { decoder.uint(256) as usize + 2 } else { state.packet_len - ((decoder.tell() as usize + 7) >> 3) };
276        if state.bytes > state.packet_len || (state.packet_len - state.bytes) * 8 < decoder.tell() as usize {
277            state.packet_len = 0;
278            state.bytes = 0;
279            state.present = false;
280        } else {
281            state.packet_len -= state.bytes;
282        }
283        decoder.shrink_storage(state.bytes);
284        state
285    }

287    fn is_mode_transition(&self, has_packet: bool, mode: Mode) -> bool {
288        has_packet
289            && self.prev_mode.is_some()
290            && ((mode == Mode::CeltOnly && self.prev_mode != Some(Mode::CeltOnly) && !self.prev_redundancy)
291                || (mode != Mode::CeltOnly && self.prev_mode == Some(Mode::CeltOnly)))
292    }

294    /// `opus_decode_frame`: decodes one frame (or conceals when `data` is `None`/≤1 byte).
295    #[allow(clippy::cognitive_complexity, reason = "the RFC 6716 frame decoding sequence, ported as-is from FilmCraft")]
296    fn decode_frame(&mut self, data: Option<&[u8]>, pcm: &mut [f32], frame_size: usize, fec: bool) -> Result<usize> {
297        let ch = self.channels;
298        let fs = self.rate as usize;
299        let f20 = fs / 50;
300        let f10 = f20 >> 1;
301        let f5 = f10 >> 1;
302        let f2_5 = f5 >> 1;
303        let mut frame_size = frame_size.min(fs / 25 * 3);
304        let data = match data {
305            Some(d) if d.len() > 1 => Some(d),
306            _ => {
307                frame_size = frame_size.min(self.frame_size);
308                None
309            }
310        };
311        let (mut audiosize, mode, bandwidth);
312        match data {
313            Some(_) => {
314                audiosize = self.frame_size;
315                mode = self.mode.unwrap_or(Mode::CeltOnly);
316                bandwidth = Some(self.bandwidth);
317            }
318            None => {
319                audiosize = frame_size;
320                bandwidth = None;
321                mode = match self.prev_mode {
322                    None => {
323                        pcm[..audiosize * ch].fill(0.0);
324                        return Ok(audiosize);
325                    }
326                    Some(m) => m,
327                };
328                if audiosize > f20 {
329                    let mut done = 0;
330                    while done < audiosize {
331                        let r = self.decode_frame(None, &mut pcm[done * ch..], (audiosize - done).min(f20), false)?;
332                        done += r;
333                    }
334                    return Ok(frame_size);
335                } else if audiosize < f20 {
336                    if audiosize > f10 {
337                        audiosize = f10;
338                    } else if mode != Mode::SilkOnly && audiosize > f5 && audiosize < f10 {
339                        audiosize = f5;
340                    }
341                }
342            }
343        }
344        let data_len = data.map_or(0, |d| d.len());
345        let mut dec_storage;
346        let mut dec: Option<&mut RangeDecoder> = match data {
347            Some(d) => {
348                dec_storage = RangeDecoder::new(d);
349                Some(&mut dec_storage)
350            }
351            None => None,
352        };
353        let mut transition = self.is_mode_transition(data.is_some(), mode);
354        let mut pcm_transition = Vec::new();
355        if transition && mode == Mode::CeltOnly {
356            pcm_transition.resize(f5 * ch, 0.0);
357            let n = f5.min(audiosize);
358            self.decode_frame(None, &mut pcm_transition, n, false)?;
359        }
360        if audiosize > frame_size {
361            return Err(Error::InvalidArgument("frame larger than output buffer"));
362        }
363        frame_size = audiosize;
364
365        // SILK.
366        let pcm_silk = self.decode_silk_frame(
367            SilkFrame { mode, bandwidth, has_packet: data.is_some(), fec, frame_size, audio_size: audiosize, rate: fs, channels: ch },
368            &mut dec,
369        )?;
370
371        let Redundancy { present: redundancy, celt_to_silk, bytes: redundancy_bytes, packet_len: len } =
372            self.read_redundancy(mode, fec, data_len, &mut dec);
373        let mut start_band = 0;
374        if mode != Mode::CeltOnly {
375            start_band = 17;
376        }
377        if redundancy {
378            transition = false;
379        }
380        if transition && mode != Mode::CeltOnly {
381            pcm_transition.resize(f5 * ch, 0.0);
382            let n = f5.min(audiosize);
383            self.decode_frame(None, &mut pcm_transition, n, false)?;
384        }
385        if let Some(bw) = bandwidth {
386            self.celt.end = bw.celt_end_band();
387        }
388        self.celt.stream_channels = self.stream_channels;
389        let mut redundant_audio = vec![0f32; if redundancy { f5 * ch } else { 0 }];
390        let mut redundant_rng = 0u32;
391        let red_data = data.map(|d| &d[len..len + redundancy_bytes]);
392        if redundancy && celt_to_silk {
393            self.celt.start = 0;
394            self.celt.decode(red_data, f5, &mut redundant_audio, false, None)?;
395            redundant_rng = self.celt.rng;
396        }
397        self.celt.start = start_band;
398
399        if mode != Mode::SilkOnly {
400            let celt_frame = f20.min(frame_size);
401            if self.prev_mode.is_some() && Some(mode) != self.prev_mode && !self.prev_redundancy {
402                self.celt.reset();
403            }
404            let cdata = if fec { None } else { data.map(|d| &d[..len]) };
405            let cdec = if fec { None } else { dec.as_deref_mut() };
406            self.celt.decode(cdata, celt_frame, pcm, false, cdec)?;
407        } else {
408            pcm[..frame_size * ch].fill(0.0);
409            if self.prev_mode == Some(Mode::Hybrid) && !(redundancy && celt_to_silk && self.prev_redundancy) {
410                self.celt.start = 0;
411                let silence = [0xFFu8, 0xFF];
412                self.celt.decode(Some(&silence), f2_5, pcm, true, None)?;
413            }
414        }
415        if mode != Mode::CeltOnly {
416            for (o, s) in pcm[..frame_size * ch].iter_mut().zip(&pcm_silk) {
417                *o += *s;
418            }
419        }
420        let window = &crate::craft::audio::opus::decode::celt::mode().window;
421        if redundancy && !celt_to_silk {
422            self.celt.reset();
423            self.celt.start = 0;
424            self.celt.decode(red_data, f5, &mut redundant_audio, false, None)?;
425            redundant_rng = self.celt.rng;
426            let o = ch * (frame_size - f2_5);
427            let tail: Vec<f32> = pcm[o..o + f2_5 * ch].to_vec();
428            smooth_fade(&tail, &redundant_audio[ch * f2_5..], &mut pcm[o..], f2_5, ch, window, self.rate);
429        }
430        if redundancy && celt_to_silk {
431            pcm[..f2_5 * ch].copy_from_slice(&redundant_audio[..f2_5 * ch]);
432            let cur: Vec<f32> = pcm[ch * f2_5..ch * 2 * f2_5].to_vec();
433            smooth_fade(&redundant_audio[ch * f2_5..], &cur, &mut pcm[ch * f2_5..], f2_5, ch, window, self.rate);
434        }
435        if transition {
436            if audiosize >= f5 {
437                pcm[..f2_5 * ch].copy_from_slice(&pcm_transition[..f2_5 * ch]);
438                let cur: Vec<f32> = pcm[ch * f2_5..ch * 2 * f2_5].to_vec();
439                smooth_fade(&pcm_transition[ch * f2_5..], &cur, &mut pcm[ch * f2_5..], f2_5, ch, window, self.rate);
440            } else {
441                let cur: Vec<f32> = pcm[..ch * f2_5].to_vec();
442                smooth_fade(&pcm_transition, &cur, pcm, f2_5, ch, window, self.rate);
443            }
444        }
445        apply_output_gain(pcm, frame_size, ch, self.gain_q8);
446        self.range_final = match dec.as_deref() {
447            Some(d) if data_len > 1 => d.rng() ^ redundant_rng,
448            _ => 0,
449        };
450        self.prev_mode = Some(mode);
451        self.prev_redundancy = redundancy && !celt_to_silk;
452        Ok(audiosize)
453    }
454}

456fn smooth_fade(in1: &[f32], in2: &[f32], out: &mut [f32], overlap: usize, ch: usize, window: &[f32; OVERLAP], rate: u32) {
457    let inc = (48000 / rate) as usize;
458    for c in 0..ch {
459        for i in 0..overlap {
460            let w = window[i * inc] * window[i * inc];
461            out[i * ch + c] = w * in2[i * ch + c] + (1.0 - w) * in1[i * ch + c];
462        }
463    }
