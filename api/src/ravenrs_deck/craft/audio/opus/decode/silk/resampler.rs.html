1//! SILK output resampler (RFC 6716 §4.2.9: non-normative filter, normative delay budget).
2//!
3//! A windowed-sinc polyphase interpolator whose group delay matches the delay allocation of
4//! RFC 6716 Table 54 (and the integer delays used when no rate conversion is needed), so that the
5//! SILK output lines up with the CELT layer in Hybrid mode.
6
7/// Dot product with eight independent partial sums, so the additions pipeline and the loop
8/// vectorises instead of waiting on one 4-cycle add chain. Rounds differently from a
9/// sequential sum (the resampler is not normative).
10#[inline]
11fn dot8(a: &[f32], b: &[f32]) -> f32 {
12    let mut s = [0f32; 8];
13    let (ca, ra) = a.as_chunks::<8>();
14    let (cb, rb) = b.as_chunks::<8>();
15    for (x, y) in ca.iter().zip(cb) {
16        for l in 0..8 {
17            s[l] += x[l] * y[l];
18        }
19    }
20    let mut acc = ((s[0] + s[4]) + (s[1] + s[5])) + ((s[2] + s[6]) + (s[3] + s[7]));
21    for (x, y) in ra.iter().zip(rb) {
22        acc += x * y;
23    }
24    acc
25}

42    hist_start: i64,

44    n_out: u64,
45}

47fn gcd(a: u32, b: u32) -> u32 {
48    if b == 0 { a } else { gcd(b, a % b) }
49}

51/// Least-squares fractional-delay interpolator: taps at offsets `-past+1 ..= future` (in input
52/// samples) approximating a delay of `frac` samples over `[0, wp]`, with zero response weighted
53/// over `[ws, pi]` when `ws < pi` (decimation).
54fn design_fractional(past: i64, future: i64, frac: f64, wp: f64, ws: f64) -> Vec<f64> {
55    let offs: Vec<f64> = ((-past + 1)..=future).map(|m| m as f64).collect();
56    let n = offs.len();
57    // \int_a^b cos(w x) dw
58    let ic = |x: f64, a: f64, b: f64| if x.abs() < 1e-12 { b - a } else { ((b * x).sin() - (a * x).sin()) / x };
59    let stop_w = 30.0;
60    const LOW_W: f64 = 0.05;
61    let mut q = vec![vec![0f64; n + 1]; n];
62    for i in 0..n {
63        for k in 0..n {
64            let d = offs[i] - offs[k];
65            let mut v = ic(d, 0.0, 0.6 * wp) + LOW_W * ic(d, 0.6 * wp, wp);
66            if ws < std::f64::consts::PI {
67                v += stop_w * ic(d, ws, std::f64::consts::PI);
68            }
69            q[i][k] = v + if i == k { 1e-9 } else { 0.0 };
70        }
71        q[i][n] = ic(offs[i] - frac, 0.0, 0.6 * wp) + LOW_W * ic(offs[i] - frac, 0.6 * wp, wp);
72    }
73    // Gaussian elimination with partial pivoting.
74    for c in 0..n {
75        let p = (c..n).max_by(|&a, &b| q[a][c].abs().total_cmp(&q[b][c].abs())).unwrap_or(c);
76        q.swap(c, p);
77        let piv = q[c][c];
78        for r in 0..n {
79            if r != c {
80                let f = q[r][c] / piv;
81                if f != 0.0 {
82                    for k in c..=n {
83                        q[r][k] -= f * q[c][k];
84                    }
85                }
86            }
87        }
88    }
89    (0..n).map(|i| q[i][n] / q[i][i]).collect()
90}

92impl Resampler {
93    pub fn new(fin: u32, fout: u32) -> Resampler {
94        let mut r = Resampler {
95            fin,
96            fout,
97            copy_delay: 0,
98            phases: Vec::new(),
99            period: 1,
100            delay_q: 0,
101            half: 0,
102            future: 0,
103            hist: Vec::new(),
104            hist_start: 0,
105            n_in: 0,
106            n_out: 0,
107        };
108        if fin == fout {
109            r.copy_delay = match fin {
110                8000 => 4,
111                12000 => 9,
112                _ => 12,
113            };
114            r.hist = vec![0.0; r.copy_delay];
115            return r;
116        }
117        // Delay in milliseconds per internal bandwidth (RFC 6716 Table 54).
118        let delay_ms = match fin {
119            8000 => 0.538,
120            12000 => 0.692,
121            _ => 0.706,
122        };
123        let d_in = delay_ms * fin as f64 / 1000.0; // input samples
124        // Future taps are limited by the delay budget; past taps are free.
125        let future = (d_in.floor() as i64).max(1);
126        let past = 24i64;
127        r.half = past;
128        r.future = future;
129        r.delay_q = (d_in * fout as f64).round() as i64;
130        let g = gcd(fin, fout);
131        r.period = (fout / g) as u64;
132        let pi = std::f64::consts::PI;
133        // Passband edge and (for decimation) stopband edge in radians per input sample.
134        let (wp, ws) = if fout > fin { (0.94 * pi, pi) } else { (0.88 * pi * fout as f64 / fin as f64, pi * fout as f64 / fin as f64) };
135        for j in 0..r.period as i64 {
136            let num = j * fin as i64 - r.delay_q;
137            let base = num.div_euclid(fout as i64);
138            let frac = num.rem_euclid(fout as i64) as f64 / fout as f64;
139            let taps = design_fractional(past, future, frac, wp, ws);
140            r.phases.push((base, taps.iter().map(|&t| t as f32).collect()));
141        }
142        r.hist = vec![0.0; (past + future + 4) as usize];
143        r.hist_start = -(past + future + 4);
144        r
145    }
146
147    pub fn rates(&self) -> (u32, u32) {
148        (self.fin, self.fout)
149    }
150
151    /// Resamples `input` (16-bit scale) and appends the output samples.
152    pub fn process(&mut self, input: &[i16], out: &mut Vec<f32>) {
153        if self.fin == self.fout {
154            for &v in input {
155                self.hist.push(v as f32);
156            }
157            let n = self.hist.len() - self.copy_delay;
158            out.extend_from_slice(&self.hist[..n]);
159            self.hist.drain(..n);
160            return;
161        }
162        self.hist.extend(input.iter().map(|&v| v as f32));
163        self.n_in += input.len() as i64;
164        let total_out = (self.n_in as u64 * self.fout as u64) / self.fin as u64;
165        // phases[ph] repeats every `period` outputs, advancing the input position by `step`
166        // (= fin * period / fout, exact because period = fout / gcd).
167        let step = self.fin as i64 * self.period as i64 / self.fout as i64;
168        let mut cycle = (self.n_out / self.period) as i64;
169        let mut ph = (self.n_out % self.period) as usize;
170        while self.n_out < total_out {
171            let (base0, ref taps) = self.phases[ph];
172            let base = base0 + cycle * step;
173            ph += 1;
174            if ph as u64 == self.period {
175                ph = 0;
176                cycle += 1;
177            }
178            let first = base - self.half + 1;
179            let i0 = first - self.hist_start;
180            let mut acc = 0f32;
181            if i0 >= 0 && i0 as usize + taps.len() <= self.hist.len() {
182                // Window fully inside the history (always, after the first call).
183                let win = &self.hist[i0 as usize..i0 as usize + taps.len()];
184                acc = dot8(win, taps);
185            } else {
186                for (m, &t) in taps.iter().enumerate() {
187                    let idx = first + m as i64 - self.hist_start;
188                    if idx >= 0 && (idx as usize) < self.hist.len() {
189                        acc += self.hist[idx as usize] * t;
190                    }
191                }
192            }
193            out.push(acc);
194            self.n_out += 1;
195        }
196        // Trim history, keeping enough for the next taps.
197        let keep_from = self.n_in - (self.half + self.future + 4);
198        if keep_from > self.hist_start {
199            let drop = (keep_from - self.hist_start) as usize;
200            self.hist.drain(..drop.min(self.hist.len()));
201            self.hist_start = keep_from;
202        }
203    }
204}

210    #[cfg(any())]
211    fn upsample_sine_and_delay() {
212        for &(fin, fout) in &[(16000u32, 48000u32), (8000, 48000), (12000, 48000), (16000, 24000), (16000, 8000), (12000, 16000)] {
213            let mut r = Resampler::new(fin, fout);
214            let f = 440.0;
215            let n = fin as usize / 10;
216            let x: Vec<i16> = (0..n).map(|i| ((2.0 * std::f64::consts::PI * f * i as f64 / fin as f64).sin() * 10000.0) as i16).collect();
217            let mut y = Vec::new();
218            for c in x.chunks(fin as usize / 50) {
219                r.process(c, &mut y);
220            }
221            assert_eq!(y.len(), n * fout as usize / fin as usize);
222            // Compare against the ideal delayed sine.
223            let delay_ms = match fin {
224                8000 => 0.538,
225                12000 => 0.692,
226                _ => 0.706,
227            };
228            let mut err = 0f64;
229            let mut sig = 0f64;
230            for (j, &v) in y.iter().enumerate().skip(fout as usize / 100) {
231                let t = j as f64 / fout as f64 - delay_ms / 1000.0;
232                let ideal = (2.0 * std::f64::consts::PI * f * t).sin() * 10000.0;
233                err += (v as f64 - ideal).powi(2);
234                sig += ideal * ideal;
235            }
236            let snr = 10.0 * (sig / err).log10();
237            assert!(snr > 35.0, "{fin}->{fout}: {snr} dB");
238        }
239    }

241    #[cfg(any())]
242    fn dot8_matches_sequential_sum() {
243        for len in [1usize, 7, 8, 9, 35, 36, 71] {
244            let a: Vec<f32> = (0..len).map(|i| ((i * 37 % 19) as f32 - 9.0) * 1000.0).collect();
245            let b: Vec<f32> = (0..len).map(|i| ((i * 11 % 7) as f32 - 3.0) / 7.0).collect();
246            let seq: f64 = a.iter().zip(&b).map(|(&x, &y)| x as f64 * y as f64).sum();
247            assert!((dot8(&a, &b) as f64 - seq).abs() < 1e-2, "len {len}");
248        }
249    }
