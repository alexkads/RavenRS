11#[inline]
12pub fn smulwb(a: i32, b: i32) -> i32 {
13    ((a as i64 * (b as i16) as i64) >> 16) as i32
14}
15#[inline]
16pub fn smlawb(a: i32, b: i32, c: i32) -> i32 {
17    a.wrapping_add(smulwb(b, c))
18}
19#[inline]
20pub fn smulww(a: i32, b: i32) -> i32 {
21    ((a as i64 * b as i64) >> 16) as i32
22}
23#[inline]
24pub fn smmul(a: i32, b: i32) -> i32 {
25    ((a as i64 * b as i64) >> 32) as i32
26}
27#[inline]
28pub fn rshift_round(a: i32, s: u32) -> i32 {
29    if s == 1 { (a >> 1) + (a & 1) } else { ((a >> (s - 1)) + 1) >> 1 }
30}
31#[inline]
32pub fn rshift_round64(a: i64, s: u32) -> i64 {
33    if s == 1 { (a >> 1) + (a & 1) } else { ((a >> (s - 1)) + 1) >> 1 }
34}
35#[inline]
36pub fn sat16(a: i32) -> i32 {
37    a.clamp(-32768, 32767)
38}

40/// `silk_INVERSE32_varQ`: approximation of `(1 << q_res) / b32`.
41pub fn inverse32_varq(b32: i32, q_res: i32) -> i32 {
42    let b_headrm = b32.unsigned_abs().leading_zeros() as i32 - 1;
43    let b32_nrm = b32.wrapping_shl(b_headrm as u32);
44    let b32_inv = (i32::MAX >> 2) / (b32_nrm >> 16);
45    let mut result = b32_inv << 16;
46    let err_q32 = ((1i32 << 29).wrapping_sub(smulwb(b32_nrm, b32_inv))).wrapping_shl(3);
47    result = result.wrapping_add(((err_q32 as i64 * b32_inv as i64) >> 16) as i32);
48    let lshift = 61 - b_headrm - q_res;
49    if lshift <= 0 {
50        let s = (-lshift) as u32;
51        let lim_hi = i32::MAX >> s;
52        let lim_lo = i32::MIN >> s;
53        result.clamp(lim_lo, lim_hi) << s
54    } else if lshift < 32 {
55        result >> lshift
56    } else {
57        0
58    }
59}

61/// `silk_DIV32_varQ`: approximation of `(a32 << q_res) / b32`.
62pub fn div32_varq(a32: i32, b32: i32, q_res: i32) -> i32 {
63    let a_headrm = a32.unsigned_abs().leading_zeros() as i32 - 1;
64    let a32_nrm = a32.wrapping_shl(a_headrm as u32);
65    let b_headrm = b32.unsigned_abs().leading_zeros() as i32 - 1;
66    let b32_nrm = b32.wrapping_shl(b_headrm as u32);
67    let b32_inv = (i32::MAX >> 2) / (b32_nrm >> 16);
68    let mut result = smulwb(a32_nrm, b32_inv);
69    let a32_nrm = a32_nrm.wrapping_sub(((smmul(b32_nrm, result) as i64) << 3) as i32);
70    result = smlawb(result, a32_nrm, b32_inv);
71    let lshift = 29 + a_headrm - b_headrm - q_res;
72    if lshift < 0 {
73        let s = (-lshift) as u32;
74        let lim_hi = i32::MAX >> s;
75        let lim_lo = i32::MIN >> s;
76        result.clamp(lim_lo, lim_hi) << s
77    } else if lshift < 32 {
78        result >> lshift
79    } else {
80        0
81    }
82}

84fn sqrt_approx(x: i32) -> i32 {
85    if x <= 0 {
86        return 0;
87    }
88    let lz = x.leading_zeros() as i32;
89    let frac = (x.rotate_right((24 - lz).rem_euclid(32) as u32)) & 0x7f;
90    let mut y = if lz & 1 != 0 { 32768 } else { 46214 };
91    y >>= lz >> 1;
92    smlawb(y, y, 213 * frac)
93}

95/// Decodes the stage-1/stage-2 indices into stabilised NLSFs (Q15).
96pub fn nlsf_decode(indices: &[i32], wb: bool) -> [i32; MAX_LPC_ORDER] {
97    let order = if wb { 16 } else { 10 };
98    let i1 = indices[0] as usize;
99    let cb1: &[i32] = if wb { &NLSF_CB_WB[i1] } else { &NLSF_CB_NBMB[i1] };
100    let qstep = if wb { 9830 } else { 11796 };
101    // Residual dequantisation with backwards prediction.
102    let mut res_q10 = [0i32; MAX_LPC_ORDER];
103    let mut out_q10 = 0i32;
104    for i in (0..order).rev() {
105        let pred = if i + 1 < order {
106            let sel = if wb { NLSF_PSEL_WB[i1][i] } else { NLSF_PSEL_NBMB[i1][i] } as usize;
107            let w = if wb { NLSF_PRED_WB[sel][i] } else { NLSF_PRED_NBMB[sel][i] };
108            (out_q10 * w) >> 8
109        } else {
110            0
111        };
112        let mut v = indices[i + 1] << 10;
113        if v > 0 {
114            v -= 102;
115        } else if v < 0 {
116            v += 102;
117        }
118        out_q10 = smlawb(pred, v, qstep);
119        res_q10[i] = out_q10;
120    }
121    // Laroia weights.
122    let mut nlsf = [0i32; MAX_LPC_ORDER];
123    for k in 0..order {
124        let prev = if k == 0 { 0 } else { cb1[k - 1] };
125        let next = if k + 1 == order { 256 } else { cb1[k + 1] };
126        let w = (1024 / (cb1[k] - prev).max(1) + 1024 / (next - cb1[k]).max(1)).min(32767);
127        let w_q9 = sqrt_approx(w << 16);
128        let v = (cb1[k] << 7) + (res_q10[k] << 14) / w_q9.max(1);
129        nlsf[k] = v.clamp(0, 32767);
130    }
131    let dmin: &[i32] = if wb { &NLSF_MIN_SPACING_WB } else { &NLSF_MIN_SPACING_NBMB };
132    nlsf_stabilize(&mut nlsf[..order], dmin);
133    nlsf
134}

136pub fn nlsf_stabilize(nlsf: &mut [i32], dmin: &[i32]) {
137    let l = nlsf.len();
138    for _ in 0..20 {
139        let mut min_diff = nlsf[0] - dmin[0];
140        let mut idx = 0;
141        for i in 1..l {
142            let d = nlsf[i] - (nlsf[i - 1] + dmin[i]);
143            if d < min_diff {
144                min_diff = d;
145                idx = i;
146            }
147        }
148        let d = 32768 - (nlsf[l - 1] + dmin[l]);
149        if d < min_diff {
150            min_diff = d;
151            idx = l;
152        }
153        if min_diff >= 0 {
154            return;
155        }
156        if idx == 0 {
157            nlsf[0] = dmin[0];
158        } else if idx == l {
159            nlsf[l - 1] = 32768 - dmin[l];
160        } else {
161            let mut min_center = dmin[..idx].iter().sum::<i32>();
162            min_center += dmin[idx] >> 1;
163            let mut max_center = 32768 - dmin[idx + 1..=l].iter().sum::<i32>();
164            max_center -= dmin[idx] >> 1;
165            let c = rshift_round(nlsf[idx - 1] + nlsf[idx], 1);
166            let c = if min_center > max_center { c.clamp(max_center, min_center) } else { c.clamp(min_center, max_center) };
167            nlsf[idx - 1] = c - (dmin[idx] >> 1);
168            nlsf[idx] = nlsf[idx - 1] + dmin[idx];
169        }
170    }
171    // Fallback.
172    nlsf.sort_unstable();
173    nlsf[0] = nlsf[0].max(dmin[0]);
174    for i in 1..l {
175        nlsf[i] = nlsf[i].max(sat16(nlsf[i - 1] + dmin[i]));
176    }
177    nlsf[l - 1] = nlsf[l - 1].min(32768 - dmin[l]);
178    for i in (0..l - 1).rev() {
179        nlsf[i] = nlsf[i].min(nlsf[i + 1] - dmin[i + 1]);
180    }
181}

183fn bwexpander_32(ar: &mut [i32], mut chirp_q16: i32) {
184    let d = ar.len();
185    let chirp_minus_one = chirp_q16 - 65536;
186    for v in ar.iter_mut().take(d - 1) {
187        *v = smulww(chirp_q16, *v);
188        chirp_q16 += rshift_round(chirp_q16.wrapping_mul(chirp_minus_one), 16);
189    }
190    ar[d - 1] = smulww(chirp_q16, ar[d - 1]);
191}

193/// `silk_bwexpander` on Q12 coefficients (16-bit).
194pub fn bwexpander(ar: &mut [i32], mut chirp_q16: i32) {
195    let d = ar.len();
196    let chirp_minus_one = chirp_q16 - 65536;
197    for v in ar.iter_mut().take(d - 1) {
198        *v = rshift_round(chirp_q16.wrapping_mul(*v), 16);
199        chirp_q16 += rshift_round(chirp_q16.wrapping_mul(chirp_minus_one), 16);
200    }
201    ar[d - 1] = rshift_round(chirp_q16.wrapping_mul(ar[d - 1]), 16);
202}

204fn lpc_inverse_pred_gain(a_q12: &[i32]) -> i32 {
205    const QA: u32 = 24;
206    const A_LIMIT: i32 = 16_773_022;
207    let order = a_q12.len();
208    let mut a = [0i32; MAX_LPC_ORDER];
209    let mut dc = 0;
210    for k in 0..order {
211        dc += a_q12[k];
212        a[k] = a_q12[k] << (QA - 12);
213    }
214    if dc >= 4096 {
215        return 0;
216    }
217    let mut inv_gain: i32 = 1 << 30;
218    let mul32_frac_q31 = |x: i32, y: i32| rshift_round64(x as i64 * y as i64, 31) as i32;
219    for k in (1..order).rev() {
220        if a[k] > A_LIMIT || a[k] < -A_LIMIT {
221            return 0;
222        }
223        let rc_q31 = -(a[k] << (31 - QA));
224        let rc_mult1 = (1i32 << 30) - smmul(rc_q31, rc_q31);
225        inv_gain = smmul(inv_gain, rc_mult1) << 2;
226        if inv_gain < 107_374 {
227            return 0;
228        }
229        let mult2q = 32 - rc_mult1.unsigned_abs().leading_zeros() as i32;
230        let rc_mult2 = inverse32_varq(rc_mult1, mult2q + 30);
231        for n in 0..(k + 1) >> 1 {
232            let t1 = a[n];
233            let t2 = a[k - n - 1];
234            let v1 = rshift_round64(t1.saturating_sub(mul32_frac_q31(t2, rc_q31)) as i64 * rc_mult2 as i64, mult2q as u32);
235            if v1 > i32::MAX as i64 || v1 < i32::MIN as i64 {
236                return 0;
237            }
238            let v2 = rshift_round64(t2.saturating_sub(mul32_frac_q31(t1, rc_q31)) as i64 * rc_mult2 as i64, mult2q as u32);
239            if v2 > i32::MAX as i64 || v2 < i32::MIN as i64 {
240                return 0;
241            }
242            a[n] = v1 as i32;
243            a[k - n - 1] = v2 as i32;
244        }
245    }
246    if a[0] > A_LIMIT || a[0] < -A_LIMIT {
247        return 0;
248    }
249    let rc_q31 = -(a[0] << (31 - QA));
250    let rc_mult1 = (1i32 << 30) - smmul(rc_q31, rc_q31);
251    inv_gain = smmul(inv_gain, rc_mult1) << 2;
252    if inv_gain < 107_374 {
253        return 0;
254    }
255    inv_gain
256}

258/// Converts NLSFs (Q15) to stable Q12 LPC coefficients (`silk_NLSF2A`).
259pub fn nlsf2a(nlsf: &[i32]) -> [i32; MAX_LPC_ORDER] {
260    let d = nlsf.len();
261    let ordering: &[usize] = if d == 16 { &LSF_ORDER_WB } else { &LSF_ORDER_NBMB };
262    let mut c_q17 = [0i32; MAX_LPC_ORDER];
263    for k in 0..d {
264        let i = (nlsf[k] >> 8) as usize;
265        let f = nlsf[k] & 255;
266        let c0 = COS_Q12[i];
267        let c1 = COS_Q12[(i + 1).min(128)];
268        c_q17[ordering[k]] = (c0 * 256 + (c1 - c0) * f + 4) >> 3;
269    }
270    let d2 = d / 2;
271    let find_poly = |off: usize| -> [i64; 10] {
272        let mut out = [0i64; 10];
273        out[0] = 1 << 16;
274        out[1] = -(c_q17[off] as i64);
275        for k in 1..d2 {
276            let ftmp = c_q17[2 * k + off] as i64;
277            out[k + 1] = 2 * out[k - 1] - ((ftmp * out[k] + 32768) >> 16);
278            for n in (2..=k).rev() {
279                out[n] += out[n - 2] - ((ftmp * out[n - 1] + 32768) >> 16);
280            }
281            out[1] -= ftmp;
282        }
283        out
284    };
285    let p = find_poly(0);
286    let q = find_poly(1);
287    let mut a32 = [0i32; MAX_LPC_ORDER];
288    for k in 0..d2 {
289        let ptmp = p[k + 1] + p[k];
290        let qtmp = q[k + 1] - q[k];
291        a32[k] = (-qtmp - ptmp) as i32;
292        a32[d - k - 1] = (qtmp - ptmp) as i32;
293    }
294    let a32 = &mut a32[..d];
295    // Limit the dynamic range (silk_LPC_fit with QIN = 17, QOUT = 12).
296    let mut a_q12 = [0i32; MAX_LPC_ORDER];
297    let mut i = 0;
298    while i < 10 {
299        let mut maxabs = 0i32;
300        let mut idx = 0;
301        for (k, &v) in a32.iter().enumerate() {
302            let av = v.saturating_abs();
303            if av > maxabs {
304                maxabs = av;
305                idx = k;
306            }
307        }
308        let maxabs = rshift_round(maxabs, 5);
309        if maxabs > 32767 {
310            let maxabs = maxabs.min(163_838);
311            let chirp = 65470 - ((maxabs - 32767) << 14) / ((maxabs * (idx as i32 + 1)) >> 2);
312            bwexpander_32(a32, chirp);
313        } else {
314            break;
315        }
316        i += 1;
317    }
318    if i == 10 {
319        for k in 0..d {
320            a_q12[k] = sat16(rshift_round(a32[k], 5));
321            a32[k] = a_q12[k] << 5;
322        }
323    } else {
324        for k in 0..d {
325            a_q12[k] = rshift_round(a32[k], 5);
326        }
327    }
328    for i in 0..16 {
329        if lpc_inverse_pred_gain(&a_q12[..d]) == 0 {
330            bwexpander_32(a32, 65536 - (2 << i));
331            for k in 0..d {
332                a_q12[k] = rshift_round(a32[k], 5);
333            }
334        } else {
335            break;
336        }
337    }
338    a_q12
339}

341#[cfg(any())]
342mod tests {
343    use super::*;
344
345    #[cfg(any())]
346    fn flat_nlsf_gives_small_lpc() {
347        // Uniformly spaced NLSFs correspond to an (almost) flat spectrum.
348        let nlsf: Vec<i32> = (1..=16).map(|k| k * 32768 / 17).collect();
349        let a = nlsf2a(&nlsf);
350        assert!(a.iter().all(|v| v.abs() < 600), "{a:?}");
351        let nlsf: Vec<i32> = (1..=10).map(|k| k * 32768 / 11).collect();
352        let a = nlsf2a(&nlsf);
353        assert!(a[..10].iter().all(|v| v.abs() < 600), "{a:?}");
354    }
355
356    #[cfg(any())]
357    fn stabilize_enforces_spacing() {
358        let mut v = [100, 90, 80, 30000, 30001, 30002, 31000, 32000, 32700, 32760];
359        nlsf_stabilize(&mut v, &NLSF_MIN_SPACING_NBMB);
360        assert!(v[0] >= NLSF_MIN_SPACING_NBMB[0]);
361        for i in 1..10 {
362            assert!(v[i] - v[i - 1] >= NLSF_MIN_SPACING_NBMB[i], "{v:?}");
363        }
364        assert!(32768 - v[9] >= NLSF_MIN_SPACING_NBMB[10]);
365    }
366
367    #[cfg(any())]
368    fn inverse_approximations() {
369        for &b in &[65536, 100_000, 1 << 20, 81920, 1_686_110_208] {
370            let inv = inverse32_varq(b, 47) as f64;
371            let exact = 2f64.powi(47) / b as f64;
372            assert!((inv - exact).abs() / exact < 1e-3, "{b}");
373        }
374        let q = div32_varq(65536, 131072, 16) as f64;
375        assert!((q - 32768.0).abs() < 2.0);
376    }
