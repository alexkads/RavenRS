1//! SILK layer decoder (RFC 6716 §4.2).
2//!
3//! Parameter decoding and reconstruction follow the normative fixed-point arithmetic so that the
4//! output matches the reference decoder sample for sample at the internal rate. Resampling to the
5//! output rate uses our own linear-phase interpolator with the delays of RFC 6716 Table 54.
6

8mod resampler;

10pub(crate) mod tables;

125    fn set_fs(&mut self, fs_khz: usize, api_rate: u32) {
126        self.subfr_length = 5 * fs_khz;
127        let frame_length = self.nb_subfr * self.subfr_length;
128        if self.fs_khz != fs_khz || self.resampler.rates() != (fs_khz as u32 * 1000, api_rate) {
129            self.resampler = Resampler::new(fs_khz as u32 * 1000, api_rate);
130        }
131        if self.fs_khz != fs_khz || frame_length != self.frame_length {
132            if self.fs_khz != fs_khz {
133                self.ltp_mem_length = 20 * fs_khz;
134                self.lpc_order = if fs_khz == 16 { 16 } else { 10 };
135                self.first_frame_after_reset = true;
136                self.lag_prev = 100;
137                self.last_gain_index = 10;
138                self.prev_signal_type = 0;
139                self.out_buf = [0; OUT_BUF];
140                self.s_lpc_q14 = [0; MAX_LPC_ORDER];
141            }
142            self.fs_khz = fs_khz;
143            self.frame_length = frame_length;
144        }
145    }

147    fn decode_indices(&mut self, dec: &mut RangeDecoder, frame_index: usize, decode_lbrr: bool, cond: Cond) {
148        let ix = if decode_lbrr || self.vad_flags[frame_index] { dec.icdf(&TYPE_ACTIVE, 8) + 2 } else { dec.icdf(&TYPE_INACTIVE, 8) };
149        let ind = &mut self.indices;
150        ind.signal_type = ix >> 1;
151        ind.quant_offset_type = ix & 1;
152        if cond == Cond::Conditionally {
153            ind.gains[0] = dec.icdf(&DELTA_GAIN, 8) as i32;
154        } else {
155            ind.gains[0] = (dec.icdf(GAIN_MSB[ind.signal_type], 8) as i32) << 3;
156            ind.gains[0] += dec.icdf(&UNIFORM8, 8) as i32;
157        }
158        for i in 1..self.nb_subfr {
159            ind.gains[i] = dec.icdf(&DELTA_GAIN, 8) as i32;
160        }
161        let wb = self.lpc_order == 16;
162        let s1 = (ind.signal_type >> 1) + if wb { 2 } else { 0 };
163        let i1 = dec.icdf(NLSF_STAGE1[s1], 8);
164        ind.nlsf[0] = i1 as i32;
165        for i in 0..self.lpc_order {
166            let cb = if wb { 8 + NLSF_SEL_WB[i1][i] as usize } else { NLSF_SEL_NBMB[i1][i] as usize };
167            let mut v = dec.icdf(NLSF_STAGE2[cb], 8) as i32;
168            if v == 0 {
169                v -= dec.icdf(&NLSF_EXT, 8) as i32;
170            } else if v == 8 {
171                v += dec.icdf(&NLSF_EXT, 8) as i32;
172            }
173            ind.nlsf[i + 1] = v - 4;
174        }
175        ind.nlsf_interp_q2 = if self.nb_subfr == 4 { dec.icdf(&NLSF_INTERP, 8) as i32 } else { 4 };
176        if ind.signal_type == 2 {
177            let mut absolute = true;
178            if cond == Cond::Conditionally && self.ec_prev_signal_type == 2 {
179                let d = dec.icdf(&PITCH_DELTA, 8) as i32;
180                if d > 0 {
181                    ind.lag_index = self.ec_prev_lag_index + d - 9;
182                    absolute = false;
183                }
184            }
185            if absolute {
186                ind.lag_index = dec.icdf(&PITCH_HIGH, 8) as i32 * (self.fs_khz as i32 >> 1);
187                let low = match self.fs_khz {
188                    8 => PITCH_LOW[0],
189                    12 => PITCH_LOW[1],
190                    _ => PITCH_LOW[2],
191                };
192                ind.lag_index += dec.icdf(low, 8) as i32;
193            }
194            self.ec_prev_lag_index = ind.lag_index;
195            let contour = match (self.fs_khz, self.nb_subfr) {
196                (8, 2) => PITCH_CONTOUR[0],
197                (8, _) => PITCH_CONTOUR[1],
198                (_, 2) => PITCH_CONTOUR[2],
199                _ => PITCH_CONTOUR[3],
200            };
201            ind.contour_index = dec.icdf(contour, 8);
202            ind.per_index = dec.icdf(&LTP_PERIODICITY, 8);
203            for k in 0..self.nb_subfr {
204                ind.ltp_index[k] = dec.icdf(LTP_FILTER[ind.per_index], 8);
205            }
206            ind.ltp_scale_index = if cond == Cond::Independently { dec.icdf(&LTP_SCALE, 8) } else { 0 };
207        }
208        self.ec_prev_signal_type = ind.signal_type;
209        ind.seed = dec.icdf(&UNIFORM4, 8) as i32;
210    }

212    fn decode_frame(&mut self, dec: Option<&mut RangeDecoder>, out: &mut [i16], lost: LostFlag, cond: Cond) {
213        let l = self.frame_length;
214        let decode = match (lost, &dec) {
215            (LostFlag::Normal, Some(_)) => true,
216            (LostFlag::Lbrr, Some(_)) => self.lbrr_flags[self.n_frames_decoded],
217            _ => false,
218        };
219        if decode && let Some(dec) = dec {
220            self.decode_indices(dec, self.n_frames_decoded, lost == LostFlag::Lbrr, cond);
221            let mut pulses = [0i32; MAX_FRAME + 16];
222            decode_pulses(dec, &mut pulses, self.indices.signal_type, self.indices.quant_offset_type, l);
223            let ctrl = self.decode_parameters(cond);
224            self.decode_core(&ctrl, &pulses, out);
225            self.plc_lpc_q12 = ctrl.pred_coef_q12[1];
226            self.plc_gain_q16 = ctrl.gains_q16[self.nb_subfr - 1];
227            self.plc_pitch = if self.indices.signal_type == 2 { ctrl.pitch_l[self.nb_subfr - 1] } else { 0 };
228            self.loss_cnt = 0;
229            self.prev_signal_type = self.indices.signal_type;
230            self.first_frame_after_reset = false;
231            self.lag_prev = ctrl.pitch_l[self.nb_subfr - 1];
232        } else {
233            self.conceal(out);
234        }
235        // Update the output buffer.
236        let mv = self.ltp_mem_length - l;
237        self.out_buf.copy_within(l..l + mv, 0);
238        self.out_buf[mv..mv + l].copy_from_slice(&out[..l]);
239    }

241    fn decode_parameters(&mut self, cond: Cond) -> Ctrl {
242        let mut ctrl = Ctrl::default();
243        let nb = self.nb_subfr;
244        // Gains.
245        let mut prev = self.last_gain_index;
246        for k in 0..nb {
247            let ind = self.indices.gains[k];
248            if k == 0 && cond != Cond::Conditionally {
249                prev = ind.max(prev - 16);
250            } else {
251                let tmp = ind - 4;
252                let thr = 2 * 36 - 64 + prev;
253                if tmp > thr {
254                    prev += (tmp << 1) - thr;
255                } else {
256                    prev += tmp;
257                }
258            }
259            prev = prev.clamp(0, 63);
260            ctrl.gains_q16[k] = log2lin((smulwb(0x1D1C71, prev) + 2090).min(3967));
261        }
262        self.last_gain_index = prev;
263        // LPC.
264        let order = self.lpc_order;
265        let nlsf = nlsf_decode(&self.indices.nlsf, order == 16);
266        ctrl.pred_coef_q12[1] = nlsf2a(&nlsf[..order]);
267        if self.first_frame_after_reset {
268            self.indices.nlsf_interp_q2 = 4;
269        }
270        if self.indices.nlsf_interp_q2 < 4 {
271            let mut n0 = [0i32; MAX_LPC_ORDER];
272            for i in 0..order {
273                n0[i] = self.prev_nlsf_q15[i] + ((self.indices.nlsf_interp_q2 * (nlsf[i] - self.prev_nlsf_q15[i])) >> 2);
274            }
275            ctrl.pred_coef_q12[0] = nlsf2a(&n0[..order]);
276        } else {
277            ctrl.pred_coef_q12[0] = ctrl.pred_coef_q12[1];
278        }
279        self.prev_nlsf_q15 = nlsf;
280        if self.loss_cnt > 0 {
281            bwexpander(&mut ctrl.pred_coef_q12[0][..order], 63570);
282            bwexpander(&mut ctrl.pred_coef_q12[1][..order], 63570);
283        }
284        if self.indices.signal_type == 2 {
285            let fs = self.fs_khz as i32;
286            let min_lag = 2 * fs;
287            let max_lag = 18 * fs;
288            let lag = min_lag + self.indices.lag_index;
289            for k in 0..nb {
290                let off = match (self.fs_khz, nb) {
291                    (8, 2) => CB_NB_10[self.indices.contour_index][k],
292                    (8, _) => CB_NB_20[self.indices.contour_index][k],
293                    (_, 2) => CB_WB_10[self.indices.contour_index][k],
294                    _ => CB_WB_20[self.indices.contour_index][k],
295                };
296                ctrl.pitch_l[k] = (lag + off).clamp(min_lag, max_lag);
297            }
298            for k in 0..nb {
299                let ix = self.indices.ltp_index[k];
300                let taps = match self.indices.per_index {
301                    0 => LTP_TAPS_0[ix.min(7)],
302                    1 => LTP_TAPS_1[ix.min(15)],
303                    _ => LTP_TAPS_2[ix.min(31)],
304                };
305                for i in 0..LTP_ORDER {
306                    ctrl.ltp_coef_q14[k * LTP_ORDER + i] = taps[i] << 7;
307                }
308            }
309            ctrl.ltp_scale_q14 = [15565, 12288, 8192][self.indices.ltp_scale_index];
310        }
311        ctrl
312    }

314    fn decode_core(&mut self, ctrl: &Ctrl, pulses: &[i32], xq: &mut [i16]) {
315        let ind = &self.indices;
316        let l = self.frame_length;
317        let sub = self.subfr_length;
318        let order = self.lpc_order;
319        let ltp_mem = self.ltp_mem_length;
320        let mut s_ltp = [0i16; MAX_FRAME];
321        let mut s_ltp_q15 = [0i32; 2 * MAX_FRAME];
322        let mut res_q14 = [0i32; MAX_SUBFR];
323        let mut s_lpc = [0i32; MAX_SUBFR + MAX_LPC_ORDER];
324        let offset_q10 = [[100, 240], [32, 100]][ind.signal_type >> 1][ind.quant_offset_type];
325        let interp = ind.nlsf_interp_q2 < 4;
326        let mut exc = [0i32; MAX_FRAME];
327        let mut seed = ind.seed;
328        for i in 0..l {
329            seed = seed.wrapping_mul(196_314_165).wrapping_add(907_633_515);
330            let mut e = pulses[i] << 14;
331            if e > 0 {
332                e -= 80 << 4;
333            } else if e < 0 {
334                e += 80 << 4;
335            }
336            e += offset_q10 << 4;
337            if seed < 0 {
338                e = -e;
339            }
340            exc[i] = e;
341            seed = seed.wrapping_add(pulses[i]);
342        }
343        s_lpc[..MAX_LPC_ORDER].copy_from_slice(&self.s_lpc_q14);
344        let mut ltp_buf_idx = ltp_mem;
345        for k in 0..self.nb_subfr {
346            let a_q12 = &ctrl.pred_coef_q12[k >> 1];
347            let mut b_q14 = [0i32; LTP_ORDER];
348            b_q14.copy_from_slice(&ctrl.ltp_coef_q14[k * LTP_ORDER..(k + 1) * LTP_ORDER]);
349            let mut signal_type = ind.signal_type;
350            let gain_q10 = ctrl.gains_q16[k] >> 6;
351            let mut inv_gain_q31 = inverse32_varq(ctrl.gains_q16[k], 47);
352            let gain_adj_q16 = if ctrl.gains_q16[k] != self.prev_gain_q16 {
353                let g = div32_varq(self.prev_gain_q16, ctrl.gains_q16[k], 16);
354                for v in s_lpc[..MAX_LPC_ORDER].iter_mut() {
355                    *v = smulww(g, *v);
356                }
357                g
358            } else {
359                1 << 16
360            };
361            self.prev_gain_q16 = ctrl.gains_q16[k];
362            let mut lag = ctrl.pitch_l[k];
363            if self.loss_cnt > 0 && self.prev_signal_type == 2 && signal_type != 2 && k < 2 {
364                b_q14 = [0; LTP_ORDER];
365                b_q14[LTP_ORDER / 2] = 1 << 12;
366                signal_type = 2;
367                lag = self.lag_prev;
368            }
369            if signal_type == 2 {
370                let lag_u = lag as usize;
371                if k == 0 || (k == 2 && interp) {
372                    let start_idx = ltp_mem as isize - lag as isize - order as isize - (LTP_ORDER / 2) as isize;
373                    let start_idx = start_idx.max(0) as usize;
374                    if k == 2 {
375                        self.out_buf[ltp_mem..ltp_mem + 2 * sub].copy_from_slice(&xq[..2 * sub]);
376                    }
377                    let src_off = start_idx + k * sub;
378                    lpc_analysis_filter(&mut s_ltp[start_idx..ltp_mem], &self.out_buf[src_off..src_off + ltp_mem - start_idx], a_q12, order);
379                    if k == 0 {
380                        inv_gain_q31 = smulwb(inv_gain_q31, ctrl.ltp_scale_q14) << 2;
381                    }
382                    for i in 0..(lag_u + LTP_ORDER / 2).min(ltp_mem) {
383                        s_ltp_q15[ltp_buf_idx - i - 1] = smulwb(inv_gain_q31, s_ltp[ltp_mem - i - 1] as i32);
384                    }
385                } else if gain_adj_q16 != 1 << 16 {
386                    for i in 0..(lag_u + LTP_ORDER / 2).min(ltp_buf_idx) {
387                        let p = ltp_buf_idx - i - 1;
388                        s_ltp_q15[p] = smulww(gain_adj_q16, s_ltp_q15[p]);
389                    }
390                }
391                let base = ltp_buf_idx as isize - lag as isize + (LTP_ORDER / 2) as isize;
392                for i in 0..sub {
393                    let p = base + i as isize;
394                    let g = |o: isize| -> i32 { if p - o >= 0 { s_ltp_q15[(p - o) as usize] } else { 0 } };
395                    let mut pred = 2i32;
396                    pred = smlawb(pred, g(0), b_q14[0]);
397                    pred = smlawb(pred, g(1), b_q14[1]);
398                    pred = smlawb(pred, g(2), b_q14[2]);
399                    pred = smlawb(pred, g(3), b_q14[3]);
400                    pred = smlawb(pred, g(4), b_q14[4]);
401                    res_q14[i] = exc[k * sub + i].wrapping_add(pred << 1);
402                    s_ltp_q15[ltp_buf_idx] = res_q14[i] << 1;
403                    ltp_buf_idx += 1;
404                }
405            } else {
406                res_q14[..sub].copy_from_slice(&exc[k * sub..(k + 1) * sub]);
407            }
408            for i in 0..sub {
409                let mut pred = (order >> 1) as i32;
410                for j in 0..order {
411                    pred = smlawb(pred, s_lpc[MAX_LPC_ORDER + i - 1 - j], a_q12[j]);
412                }
413                let v = res_q14[i].saturating_add(pred.saturating_mul(16));
414                s_lpc[MAX_LPC_ORDER + i] = v;
415                xq[k * sub + i] = sat16(rshift_round(smulww(v, gain_q10), 8)) as i16;
416            }
417            s_lpc.copy_within(sub..sub + MAX_LPC_ORDER, 0);
418        }
419        self.s_lpc_q14.copy_from_slice(&s_lpc[..MAX_LPC_ORDER]);
420    }

422    /// Basic concealment: pitch-periodic repetition for voiced history, otherwise LPC-shaped
423    /// noise at the residual level of the history; both fade out over successive losses.
424    fn conceal(&mut self, out: &mut [i16]) {
425        let l = self.frame_length;
426        if l == 0 {
427            return;
428        }
429        self.loss_cnt += 1;
430        let order = self.lpc_order;
431        let mem = self.ltp_mem_length;
432        let fade = 0.85f32.powi(self.loss_cnt as i32);
433        let hist: Vec<f32> = self.out_buf[..mem].iter().map(|&v| v as f32).collect();
434        let pitch = self.plc_pitch as usize;
435        let mut y = hist.clone();
436        if pitch > 0 && pitch < mem {
437            for _ in 0..l {
438                let v = y[y.len() - pitch] * fade;
439                y.push(v);
440            }
441        } else {
442            let mut a12 = self.plc_lpc_q12;
443            bwexpander(&mut a12[..order], 64880);
444            self.plc_lpc_q12 = a12;
445            let a: Vec<f32> = a12[..order].iter().map(|&v| v as f32 / 4096.0).collect();
446            let mut e = 0f32;
447            for i in order..mem {
448                let mut r = hist[i];
449                for j in 0..order {
450                    r -= a[j] * hist[i - 1 - j];
451                }
452                e += r * r;
453            }
454            let rms = (e / (mem - order).max(1) as f32).sqrt() * fade;
455            for _ in 0..l {
456                self.plc_rand = self.plc_rand.wrapping_mul(196_314_165).wrapping_add(907_633_515);
457                let mut v = ((self.plc_rand as i32) >> 16) as f32 / 32768.0 * rms * 1.7;
458                for j in 0..order {
459                    v += a[j] * y[y.len() - 1 - j];
460                }
461                y.push(v.clamp(-32768.0, 32767.0));
462            }
463        }
464        for (o, v) in out.iter_mut().zip(&y[mem..mem + l]) {
465            *o = v.clamp(-32768.0, 32767.0) as i16;
466        }
467        // Keep the LPC synthesis state consistent for the next decoded frame.
468        let g = self.prev_gain_q16.max(1) as i64;
469        for j in 0..MAX_LPC_ORDER {
470            let v = out[l - MAX_LPC_ORDER + j] as i64;
471            self.s_lpc_q14[j] = ((v << 30) / g).clamp(i32::MIN as i64, i32::MAX as i64) as i32;
472        }
473        self.lag_prev = self.plc_pitch.max(1);
474        self.first_frame_after_reset = false;
475    }

487fn log2lin(in_log_q7: i32) -> i32 {
488    if in_log_q7 < 0 {
489        return 0;
490    }
491    if in_log_q7 >= 3967 {
492        return i32::MAX;
493    }
494    let out = 1i32 << (in_log_q7 >> 7);
495    let frac = in_log_q7 & 0x7F;
496    if in_log_q7 < 2048 {
497        out + ((out * smlawb(frac, frac * (128 - frac), -174)) >> 7)
498    } else {
499        out + (out >> 7) * smlawb(frac, frac * (128 - frac), -174)
500    }
501}
502
503fn lpc_analysis_filter(out: &mut [i16], input: &[i16], b: &[i32], d: usize) {
504    let len = out.len();
505    for ix in d..len {
506        let mut acc: i32 = 0;
507        for j in 0..d {
508            acc = acc.wrapping_add((input[ix - 1 - j] as i32).wrapping_mul(b[j]));
509        }
510        let v = ((input[ix] as i32) << 12).wrapping_sub(acc);
511        out[ix] = sat16(rshift_round(v, 12)) as i16;
512    }
513    for v in out.iter_mut().take(d.min(len)) {
514        *v = 0;
515    }
516}

518fn decode_pulses(dec: &mut RangeDecoder, pulses: &mut [i32], signal_type: usize, quant_offset_type: usize, frame_length: usize) {
519    let rate_level = dec.icdf(RATE_LEVEL[signal_type >> 1], 8);
520    let mut iter = frame_length >> 4;
521    if iter * 16 < frame_length {
522        iter += 1;
523    }
524    let mut sum_pulses = [0i32; 20];
525    let mut n_lshifts = [0i32; 20];
526    for i in 0..iter {
527        let mut s = dec.icdf(PULSE_COUNT[rate_level], 8) as i32;
528        while s == 17 {
529            n_lshifts[i] += 1;
530            let tab = if n_lshifts[i] == 10 { PULSE_COUNT[10] } else { PULSE_COUNT[9] };
531            s = dec.icdf(tab, 8) as i32;
532        }
533        sum_pulses[i] = s;
534    }
535    for i in 0..iter {
536        let blk = &mut pulses[i * 16..(i + 1) * 16];
537        if sum_pulses[i] > 0 {
538            shell_decode(dec, blk, sum_pulses[i]);
539        } else {
540            blk.fill(0);
541        }
542    }
543    for i in 0..iter {
544        let nls = n_lshifts[i];
545        if nls > 0 {
546            for p in pulses[i * 16..(i + 1) * 16].iter_mut() {
547                let mut a = *p;
548                for _ in 0..nls {
549                    a = (a << 1) + dec.icdf(&EXC_LSB, 8) as i32;
550                }
551                *p = a;
552            }
553            sum_pulses[i] |= nls << 5;
554        }
555    }
556    // Signs.
557    let base = 7 * (quant_offset_type + (signal_type << 1));
558    let len = (frame_length + 8) >> 4;
559    for i in 0..len {
560        let p = sum_pulses[i];
561        if p > 0 {
562            let tab = EXC_SIGN[base + ((p & 0x1F) as usize).min(6)];
563            for q in pulses[i * 16..(i + 1) * 16].iter_mut() {
564                if *q > 0 && dec.icdf(tab, 8) == 0 {
565                    *q = -*q;
566                }
567            }
568        }
569    }
570}
571
572fn shell_decode(dec: &mut RangeDecoder, out: &mut [i32], total: i32) {
573    fn split(dec: &mut RangeDecoder, out: &mut [i32], n: usize, count: i32) {
574        if n == 1 {
575            out[0] = count;
576            return;
577        }
578        if count == 0 {
579            out[..n].fill(0);
580            return;
581        }
582        let tab = match n {
583            16 => SHELL_16[(count - 1) as usize],
584            8 => SHELL_8[(count - 1) as usize],
585            4 => SHELL_4[(count - 1) as usize],
586            _ => SHELL_2[(count - 1) as usize],
587        };
588        let left = dec.icdf(tab, 8) as i32;
589        let h = n / 2;
590        let (a, b) = out.split_at_mut(h);
591        split(dec, a, h, left);
592        split(dec, b, h, count - left);
593    }
594    split(dec, out, 16, total.min(16));
595}

597#[derive(Default, Clone)]
598struct Stereo {
599    pred_prev_q13: [i32; 2],
600    s_mid: [i16; 2],
601    s_side: [i16; 2],
602}
603
604fn stereo_decode_pred(dec: &mut RangeDecoder) -> [i32; 2] {
605    let n = dec.icdf(&STEREO_PRED_JOINT, 8);
606    let mut ix = [[0usize; 3]; 2];
607    ix[0][2] = n / 5;
608    ix[1][2] = n - 5 * ix[0][2];
609    for v in ix.iter_mut() {
610        v[0] = dec.icdf(&UNIFORM3, 8);
611        v[1] = dec.icdf(&UNIFORM5, 8);
612    }
613    let mut pred = [0i32; 2];
614    for (p, v) in pred.iter_mut().zip(ix.iter_mut()) {
615        v[0] += 3 * v[2];
616        let low = STEREO_WEIGHTS_Q13[v[0]];
617        let step = smulwb(STEREO_WEIGHTS_Q13[v[0] + 1] - low, 6554);
618        *p = low + step * (2 * v[1] as i32 + 1);
619    }
620    pred[0] -= pred[1];
621    pred
622}
623
624fn stereo_ms_to_lr(st: &mut Stereo, x1: &mut [i16], x2: &mut [i16], pred_q13: [i32; 2], fs_khz: usize, len: usize) {
625    x1[..2].copy_from_slice(&st.s_mid);
626    x2[..2].copy_from_slice(&st.s_side);
627    st.s_mid.copy_from_slice(&x1[len..len + 2]);
628    st.s_side.copy_from_slice(&x2[len..len + 2]);
629    let mut p0 = st.pred_prev_q13[0];
630    let mut p1 = st.pred_prev_q13[1];
631    let denom_q16 = (1 << 16) / (8 * fs_khz as i32);
632    let d0 = rshift_round((pred_q13[0] - st.pred_prev_q13[0]) as i16 as i32 * denom_q16 as i16 as i32, 16);
633    let d1 = rshift_round((pred_q13[1] - st.pred_prev_q13[1]) as i16 as i32 * denom_q16 as i16 as i32, 16);
634    let interp = 8 * fs_khz;
635    for n in 0..len {
636        if n < interp {
637            p0 += d0;
638            p1 += d1;
639        } else if n == interp {
640            p0 = pred_q13[0];
641            p1 = pred_q13[1];
642        }
643        let sum = ((x1[n] as i32 + x1[n + 2] as i32) + ((x1[n + 1] as i32) << 1)) << 9;
644        let mut s = smlawb((x2[n + 1] as i32) << 8, sum, p0);
645        s = smlawb(s, (x1[n + 1] as i32) << 11, p1);
646        x2[n + 1] = sat16(rshift_round(s, 8)) as i16;
647    }
648    st.pred_prev_q13 = pred_q13;
649    for n in 0..len {
650        let a = x1[n + 1] as i32;
651        let b = x2[n + 1] as i32;
652        x1[n + 1] = sat16(a + b) as i16;
653        x2[n + 1] = sat16(a - b) as i16;
654    }
655}

657/// SILK decoder for up to two internal channels (`silk_decoder`).
658pub struct SilkDecoder {
659    api_rate: u32,
660    api_channels: usize,
661    ch: [Channel; 2],
662    stereo: Stereo,
663    channels_internal: usize,
664    channels_internal_prev: usize,
665    internal_rate: u32,
666    prev_decode_only_middle: bool,
667    ms_pred: [i32; 2],
668}
669
670impl SilkDecoder {
671    pub fn new(api_rate: u32, api_channels: usize) -> SilkDecoder {
672        SilkDecoder {
673            api_rate,
674            api_channels,
675            ch: [Channel::new(), Channel::new()],
676            stereo: Stereo::default(),
677            channels_internal: api_channels,
678            channels_internal_prev: 0,
679            internal_rate: 16000,
680            prev_decode_only_middle: false,
681            ms_pred: [0; 2],
682        }
683    }
684
685    pub fn reset(&mut self) {
686        self.ch = [Channel::new(), Channel::new()];
687        self.stereo = Stereo::default();
688        self.prev_decode_only_middle = false;
689        self.channels_internal_prev = 0;
690    }
691
692    pub fn set_stream(&mut self, channels: usize, internal_rate: u32) {
693        self.channels_internal = channels;
694        self.internal_rate = internal_rate;
695    }
696
697    fn prepare_frame(&mut self, new_packet: bool, payload_ms: usize) -> Result<bool> {
698        let nci = self.channels_internal;
699        if new_packet {
700            for channel in &mut self.ch {
701                channel.n_frames_decoded = 0;
702            }
703        }
704        if nci > self.channels_internal_prev && self.channels_internal_prev != 0 {
705            self.ch[1] = Channel::new();
706        }
707        let stereo_to_mono = nci == 1 && self.channels_internal_prev == 2 && self.internal_rate == 1000 * self.ch[0].fs_khz as u32;
708        if self.ch[0].n_frames_decoded == 0 {
709            let (frames, subframes) = match payload_ms {
710                10 => (1, 2),
711                20 => (1, 4),
712                40 => (2, 4),
713                60 => (3, 4),
714                _ => return Err(Error::InvalidPacket("bad SILK frame size")),
715            };
716            let fs_khz = ((self.internal_rate >> 10) + 1) as usize;
717            for channel in self.ch.iter_mut().take(nci) {
718                channel.n_frames_per_packet = frames;
719                channel.nb_subfr = subframes;
720                channel.set_fs(fs_khz, self.api_rate);
721            }
722        }
723        if self.api_channels == 2 && nci == 2 && self.channels_internal_prev == 1 {
724            self.stereo.pred_prev_q13 = [0; 2];
725            self.stereo.s_side = [0; 2];
726            self.ch[1].resampler = self.ch[0].resampler.clone();
727        }
728        self.channels_internal_prev = nci;
729        Ok(stereo_to_mono)
730    }
731
732    fn decode_packet_header(&mut self, dec: &mut Option<&mut RangeDecoder>, lost: LostFlag) -> Result<()> {
733        let nci = self.channels_internal;
734        if lost == LostFlag::Lost || self.ch[0].n_frames_decoded != 0 {
735            return Ok(());
736        }
737        let decoder = dec.as_deref_mut().ok_or(Error::InvalidPacket("missing SILK data"))?;
738        for channel in self.ch.iter_mut().take(nci) {
739            for index in 0..channel.n_frames_per_packet {
740                channel.vad_flags[index] = decoder.bit_logp(1);
741            }
742            channel.lbrr_flag = decoder.bit_logp(1);
743        }
744        for channel in self.ch.iter_mut().take(nci) {
745            channel.lbrr_flags = [false; 3];
746            if channel.lbrr_flag {
747                if channel.n_frames_per_packet == 1 {
748                    channel.lbrr_flags[0] = true;
749                } else {
750                    let sym = decoder.icdf(LBRR_FLAGS[channel.n_frames_per_packet - 2], 8) + 1;
751                    for index in 0..channel.n_frames_per_packet {
752                        channel.lbrr_flags[index] = (sym >> index) & 1 != 0;
753                    }
754                }
755            }
756        }
757        if lost == LostFlag::Normal {
758            self.skip_lbrr_data(decoder, nci);
759        }
760        Ok(())
761    }
762
763    fn skip_lbrr_data(&mut self, decoder: &mut RangeDecoder, nci: usize) {
764        for frame in 0..self.ch[0].n_frames_per_packet {
765            for channel in 0..nci {
766                if !self.ch[channel].lbrr_flags[frame] {
767                    continue;
768                }
769                if nci == 2 && channel == 0 {
770                    stereo_decode_pred(decoder);
771                    if !self.ch[1].lbrr_flags[frame] {
772                        decoder.icdf(&MID_ONLY, 8);
773                    }
774                }
775                let condition = if frame > 0 && self.ch[channel].lbrr_flags[frame - 1] { Cond::Conditionally } else { Cond::Independently };
776                let state = &mut self.ch[channel];
777                state.decode_indices(decoder, frame, true, condition);
778                let mut pulses = [0i32; MAX_FRAME + 16];
779                decode_pulses(decoder, &mut pulses, state.indices.signal_type, state.indices.quant_offset_type, state.frame_length);
780            }
781        }
782    }
783
784    /// Decodes one SILK frame (10 or 20 ms) into interleaved `out` at the API rate.
785    pub fn decode(
786        &mut self,
787        mut dec: Option<&mut RangeDecoder>,
788        lost: LostFlag,
789        new_packet: bool,
790        payload_ms: usize,
791        out: &mut [f32],
792    ) -> Result<usize> {
793        let nci = self.channels_internal;
794        let stereo_to_mono = self.prepare_frame(new_packet, payload_ms)?;
795        self.decode_packet_header(&mut dec, lost)?;
796        // Stereo prediction. Each frame starts out coding the side channel, as in libopus.
797        let mut decode_only_middle = false;
798        if nci == 2 {
799            let fi = self.ch[0].n_frames_decoded;
800            let coded = lost == LostFlag::Normal || (lost == LostFlag::Lbrr && self.ch[0].lbrr_flags[fi]);
801            match dec.as_deref_mut() {
802                Some(d) if coded => {
803                    self.ms_pred = stereo_decode_pred(d);
804                    let side_inactive =
805                        (lost == LostFlag::Normal && !self.ch[1].vad_flags[fi]) || (lost == LostFlag::Lbrr && !self.ch[1].lbrr_flags[fi]);
806                    decode_only_middle = if side_inactive { d.icdf(&MID_ONLY, 8) == 1 } else { false };
807                }
808                _ => {
809                    self.ms_pred = self.stereo.pred_prev_q13;
810                }
811            }
812        }
813        if nci == 2 && !decode_only_middle && self.prev_decode_only_middle {
814            let c = &mut self.ch[1];
815            c.out_buf = [0; OUT_BUF];
816            c.s_lpc_q14 = [0; MAX_LPC_ORDER];
817            c.lag_prev = 100;
818            c.last_gain_index = 10;
819            c.prev_signal_type = 0;
820            c.first_frame_after_reset = true;
821        }
822        let l = self.ch[0].frame_length;
823        let mut buf = [[0i16; MAX_FRAME + 2]; 2];
824        let has_side = if lost == LostFlag::Normal {
825            !decode_only_middle
826        } else {
827            !self.prev_decode_only_middle || (nci == 2 && lost == LostFlag::Lbrr && self.ch[1].lbrr_flags[self.ch[1].n_frames_decoded])
828        };
829        for n in 0..nci {
830            if n == 0 || has_side {
831                let frame_index = self.ch[0].n_frames_decoded as isize - n as isize;
832                let cond = if frame_index <= 0 {
833                    Cond::Independently
834                } else if lost == LostFlag::Lbrr {
835                    if self.ch[n].lbrr_flags[frame_index as usize - 1] { Cond::Conditionally } else { Cond::Independently }
836                } else if n > 0 && self.prev_decode_only_middle {
837                    Cond::IndependentlyNoLtpScaling
838                } else {
839                    Cond::Conditionally
840                };
841                let c = &mut self.ch[n];
842                c.decode_frame(dec.as_deref_mut(), &mut buf[n][2..], lost, cond);
843            } else {
844                buf[n][2..2 + l].fill(0);
845            }
846            self.ch[n].n_frames_decoded += 1;
847        }
848        if self.api_channels == 2 && nci == 2 {
849            let (a, b) = buf.split_at_mut(1);
850            stereo_ms_to_lr(&mut self.stereo, &mut a[0], &mut b[0], self.ms_pred, self.ch[0].fs_khz, l);
851        } else {
852            buf[0][..2].copy_from_slice(&self.stereo.s_mid);
853            self.stereo.s_mid.copy_from_slice(&buf[0][l..l + 2]);
854        }
855        let n_out = l * self.api_rate as usize / (self.ch[0].fs_khz * 1000);
856        let ach = self.api_channels;
857        let mut tmp = Vec::with_capacity(n_out);
858        for n in 0..ach.min(nci) {
859            tmp.clear();
860            self.ch[n].resampler.process(&buf[n][1..1 + l], &mut tmp);
861            for (i, &v) in tmp.iter().enumerate().take(n_out) {
862                out[i * ach + n] = v / 32768.0;
863            }
864        }
865        if ach == 2 && nci == 1 {
866            if stereo_to_mono {
867                tmp.clear();
868                self.ch[1].resampler.process(&buf[0][1..1 + l], &mut tmp);
869                for (i, &v) in tmp.iter().enumerate().take(n_out) {
870                    out[i * 2 + 1] = v / 32768.0;
871                }
872            } else {
873                for i in 0..n_out {
874                    out[i * 2 + 1] = out[i * 2];
875                }
876            }
877        }
878        if lost == LostFlag::Lost {
879            for c in self.ch.iter_mut().take(nci) {
880                c.last_gain_index = 10;
881            }
882        } else {
883            self.prev_decode_only_middle = decode_only_middle;
884        }
885        Ok(n_out)
886    }
887}
