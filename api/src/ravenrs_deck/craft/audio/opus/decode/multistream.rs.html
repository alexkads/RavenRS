1//! Public decoder: OpusHead-configured (multistream) decoding to planar f32.

3use crate::craft::audio::opus::decode::decoder::StreamDecoder;

8/// Opus decoder for an `OpusHead` configuration (mono, stereo, or multistream surround).
9///
10/// Produces planar f32 at the chosen output rate. Pre-skip is *not* removed unless
11/// [`Decoder::set_trim_pre_skip`] is enabled (containers usually handle it via edit lists /
12/// granule positions). The header's output gain is applied.
13pub struct Decoder {
14    head: OpusHead,
15    rate: u32,
16    streams: Vec<StreamDecoder>,
17    trim_pre_skip: bool,
18    skip_remaining: usize,
19    scratch: Vec<Vec<f32>>,
20}
21
22impl Decoder {
23    /// Creates a 48 kHz decoder from an `OpusHead` packet (Ogg / Matroska `CodecPrivate`).
24    pub fn new(opus_head: &[u8]) -> Result<Decoder> {
25        Self::from_head(OpusHead::parse(opus_head)?, 48000)
26    }
27
28    /// Creates a decoder for plain mono/stereo packets without a header.
29    pub fn simple(sample_rate: u32, channels: usize) -> Result<Decoder> {
30        Self::from_head(OpusHead::simple(channels)?, sample_rate)
31    }
32
33    /// Creates a decoder for a parsed header at `sample_rate` (8/12/16/24/48 kHz).
34    pub fn from_head(head: OpusHead, sample_rate: u32) -> Result<Decoder> {
35        head.validate()?;
36        let mut streams = Vec::with_capacity(head.streams);
37        for s in 0..head.streams {
38            let ch = if s < head.coupled_streams { 2 } else { 1 };
39            let mut d = StreamDecoder::new(sample_rate, ch)?;
40            d.set_gain_q8(head.output_gain);
41            streams.push(d);
42        }
43        Ok(Decoder { rate: sample_rate, streams, trim_pre_skip: false, skip_remaining: 0, scratch: vec![Vec::new(); head.streams], head })
44    }
45
46    /// The `OpusHead` the decoder was created from.
47    pub fn head(&self) -> &OpusHead {
48        &self.head
49    }
50
51    /// Output sampling rate in Hz.
52    pub fn sample_rate(&self) -> u32 {
53        self.rate
54    }
55
56    /// Output channels (the header's channel count).
57    pub fn channels(&self) -> usize {
58        self.head.channels
59    }
60
61    /// Pre-skip in samples at the output rate.
62    pub fn pre_skip(&self) -> usize {
63        self.head.pre_skip as usize * self.rate as usize / 48000
64    }
65
66    /// When enabled, the first [`Self::pre_skip`] output samples are dropped.
67    pub fn set_trim_pre_skip(&mut self, enable: bool) {
68        self.trim_pre_skip = enable;
69        self.skip_remaining = if enable { self.pre_skip() } else { 0 };
70    }
71
72    /// Disables intensity-stereo phase inversion (useful when downmixing; RFC 8251 section 10).
73    pub fn set_phase_inversion_disabled(&mut self, disabled: bool) {
74        for s in &mut self.streams {
75            s.set_phase_inversion_disabled(disabled);
76        }
77    }
78
79    /// Final range coder state of the first stream (conformance testing).
80    pub fn final_range(&self) -> u32 {
81        self.streams.first().map_or(0, |s| s.final_range())
82    }
83
84    /// Reset every stream's decoder state (e.g. after a seek). With [`Self::set_trim_pre_skip`]
85    /// on, trimming starts again too, which is right only when decoding restarts from the start of
86    /// the stream (RFC 7845 §4 applies pre-skip once); turn it off before a seek anywhere else.
87    pub fn reset(&mut self) {
88        for s in &mut self.streams {
89            s.reset();
90        }
91        self.skip_remaining = if self.trim_pre_skip { self.pre_skip() } else { 0 };
92    }
93
94    /// Decodes one packet (`None` = lost packet, concealed) into planar channels.
95    pub fn decode(&mut self, packet: Option<&[u8]>) -> Result<Vec<Vec<f32>>> {
96        self.decode_or_conceal(packet, None)
97    }
98
99    /// Conceals one lost or undecodable packet. With its bytes at hand and a TOC and framing that
100    /// parse, the concealment is as long as the packet; otherwise as long as the last packet
101    /// (20 ms before any).
102    pub fn conceal(&mut self, packet: Option<&[u8]>) -> Result<Vec<Vec<f32>>> {
103        let dur = packet.and_then(|d| self.packet_duration(d));
104        self.decode_or_conceal(None, dur)
105    }
106
107    /// Samples per channel, at the output rate, in a packet whose framing parses.
108    fn packet_duration(&self, data: &[u8]) -> Option<usize> {
109        let p = if self.streams.len() > 1 { Packet::parse_self_delimited(data).ok()?.0 } else { Packet::parse(data).ok()? };
110        Some(p.toc.frame_samples(self.rate) * p.frames.len()).filter(|&n| n > 0)
111    }
112
113    fn decode_or_conceal(&mut self, packet: Option<&[u8]>, conceal_len: Option<usize>) -> Result<Vec<Vec<f32>>> {
114        let n = self.streams.len();
115        for b in &mut self.scratch {
116            b.clear();
117        }
118        let mut samples = None;
119        match packet {
120            Some(data) if !data.is_empty() => {
121                let mut off = 0;
122                for s in 0..n {
123                    let rest = &data[off..];
124                    if rest.is_empty() {
125                        return Err(Error::InvalidPacket("truncated multistream packet"));
126                    }
127                    let used = if s + 1 < n { Packet::parse_self_delimited(rest)?.1 } else { rest.len() };
128                    let pkt = &rest[..used];
129                    let r = if s + 1 < n {
130                        // Re-frame the self-delimited packet as a regular one for the stream decoder.
131                        let (p, _) = Packet::parse_self_delimited(pkt)?;
132                        let regular = reframe(&p);
133                        self.streams[s].decode_interleaved(Some(&regular), &mut self.scratch[s])?
134                    } else {
135                        self.streams[s].decode_interleaved(Some(pkt), &mut self.scratch[s])?
136                    };
137                    if *samples.get_or_insert(r) != r {
138                        return Err(Error::InvalidPacket("multistream frame durations differ"));
139                    }
140                    off += used;
141                }
142            }
143            _ => {
144                let dur = conceal_len.unwrap_or(match self.streams[0].last_packet_duration() {
145                    0 => self.rate as usize / 50,
146                    d => d,
147                });
148                for s in 0..n {
149                    self.streams[s].conceal(dur, &mut self.scratch[s])?;
150                }
151                samples = Some(dur);
152            }
153        }
154        let samples = samples.unwrap_or(0);
155        let mut out = vec![vec![0f32; samples]; self.head.channels];
156        for (c, o) in out.iter_mut().enumerate() {
157            let m = self.head.mapping[c] as usize;
158            if m == 255 {
159                continue;
160            }
161            let (s, ch, nch) = if m < 2 * self.head.coupled_streams { (m / 2, m % 2, 2) } else { (m - self.head.coupled_streams, 0, 1) };
162            let src = &self.scratch[s];
163            for (i, v) in o.iter_mut().enumerate() {
164                *v = src[i * nch + ch];
165            }
166        }
167        if self.skip_remaining > 0 {
168            let k = self.skip_remaining.min(samples);
169            for o in &mut out {
170                o.drain(..k);
171            }
172            self.skip_remaining -= k;
173        }
174        Ok(out)
175    }
176}

178/// Rebuilds a regular (code 3 VBR) packet from a parsed self-delimited one.
179fn reframe(p: &Packet) -> Vec<u8> {
180    let mut v = vec![(p.toc.config << 3) | if p.toc.stereo { 4 } else { 0 } | 3];
181    v.push(0x80 | p.frames.len() as u8);
182    for f in &p.frames[..p.frames.len() - 1] {
183        let s = f.len();
184        if s < 252 {
185            v.push(s as u8);
186        } else {
187            let b0 = 252 + (s & 3);
188            v.push(b0 as u8);
189            v.push(((s - b0) >> 2) as u8);
190        }
191    }
192    for f in &p.frames {
193        v.extend_from_slice(f);
194    }
195    v
