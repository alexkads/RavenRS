19/// Errors from header parsing and decoding.
20#[non_exhaustive]
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub enum Error {
23    /// An argument is out of range (sampling rate, channel count, concealment length, output size).
24    InvalidArgument(&'static str),
25    /// A malformed packet (RFC 6716 §3.4).
26    InvalidPacket(&'static str),
27    /// A malformed `OpusHead` (RFC 7845 §5.1) or MP4 `dOps` box.
28    InvalidHeader(&'static str),
29    /// A valid stream uses a feature this decoder does not implement.
30    Unsupported(&'static str),
31}
32
33impl std::fmt::Display for Error {
34    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
35        match self {
36            Error::InvalidArgument(s) => write!(f, "invalid argument: {s}"),
37            Error::InvalidPacket(s) => write!(f, "invalid packet: {s}"),
38            Error::InvalidHeader(s) => write!(f, "invalid header: {s}"),
39            Error::Unsupported(s) => write!(f, "unsupported: {s}"),
40        }
41    }
42}
43
44impl std::error::Error for Error {}
45
46/// Result with this module's [`Error`].
47pub type Result<T> = std::result::Result<T, Error>;
