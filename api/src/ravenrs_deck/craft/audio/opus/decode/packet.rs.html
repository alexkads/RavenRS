5/// Coding mode of an Opus frame.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum Mode {
8    /// SILK only (TOC configs 0-11).
9    SilkOnly,
10    /// SILK and CELT (TOC configs 12-15).
11    Hybrid,
12    /// CELT only (TOC configs 16-31).
13    CeltOnly,
14}

16pub use crate::craft::audio::opus::Bandwidth;

18impl Bandwidth {
19    /// Number of CELT bands coded for this bandwidth.
20    pub fn celt_end_band(self) -> usize {
21        match self {
22            Bandwidth::Narrowband => 13,
23            Bandwidth::Mediumband | Bandwidth::Wideband => 17,
24            Bandwidth::SuperWideband => 19,
25            Bandwidth::Fullband => 21,
26        }
27    }
28}

30/// Decoded table-of-contents byte.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub struct Toc {
33    /// Configuration number (0..=31, RFC 6716 Table 2).
34    pub config: u8,
35    /// Coding mode, from `config`.
36    pub mode: Mode,
37    /// Audio bandwidth, from `config`.
38    pub bandwidth: Bandwidth,
39    /// Stereo flag (`s` bit).
40    pub stereo: bool,
41    /// Frame count code (0..=3).
42    pub code: u8,
43}
44
45impl Toc {
46    /// Decode a TOC byte (RFC 6716 §3.1).
47    pub fn parse(b: u8) -> Toc {
48        let config = b >> 3;
49        let (mode, bandwidth) = match config {
50            0..=3 => (Mode::SilkOnly, Bandwidth::Narrowband),
51            4..=7 => (Mode::SilkOnly, Bandwidth::Mediumband),
52            8..=11 => (Mode::SilkOnly, Bandwidth::Wideband),
53            12..=13 => (Mode::Hybrid, Bandwidth::SuperWideband),
54            14..=15 => (Mode::Hybrid, Bandwidth::Fullband),
55            16..=19 => (Mode::CeltOnly, Bandwidth::Narrowband),
56            20..=23 => (Mode::CeltOnly, Bandwidth::Wideband),
57            24..=27 => (Mode::CeltOnly, Bandwidth::SuperWideband),
58            _ => (Mode::CeltOnly, Bandwidth::Fullband),
59        };
60        Toc { config, mode, bandwidth, stereo: b & 4 != 0, code: b & 3 }
61    }
62
63    /// Samples per frame at 48 kHz.
64    pub fn frame_samples_48k(&self) -> usize {
65        self.frame_samples(48000)
66    }
67
68    /// Samples per frame at `rate`.
69    pub fn frame_samples(&self, rate: u32) -> usize {
70        let rate = rate as usize;
71        let c = self.config as usize;
72        match self.mode {
73            Mode::CeltOnly => (rate << (c & 3)) / 400,
74            Mode::Hybrid => {
75                if c & 1 != 0 {
76                    rate / 50
77                } else {
78                    rate / 100
79                }
80            }
81            Mode::SilkOnly => match c & 3 {
82                3 => rate * 60 / 1000,
83                s => (rate << s) / 100,
84            },
85        }
86    }
87
88    /// Channels the frames are coded with: 2 when `stereo`, else 1.
89    pub fn channels(&self) -> usize {
90        if self.stereo { 2 } else { 1 }
91    }
92}

94/// A parsed packet: TOC plus the byte ranges of each frame.
95#[derive(Clone, Debug)]
96pub struct Packet<'a> {
97    /// The packet's TOC byte.
98    pub toc: Toc,
99    /// Compressed data of each frame.
100    pub frames: Vec<&'a [u8]>,
101    /// Bytes of padding (code 3 only).
102    pub padding: usize,
103}
104
105fn parse_size(data: &[u8]) -> Option<(usize, usize)> {
106    match data.first() {
107        None => None,
108        Some(&b) if b < 252 => Some((b as usize, 1)),
109        Some(&b) => data.get(1).map(|&b1| (4 * b1 as usize + b as usize, 2)),
110    }
111}
112
113impl<'a> Packet<'a> {
114    /// Parses a packet (RFC 6716 section 3.2), enforcing requirements R1 to R7.
115    pub fn parse(data: &'a [u8]) -> Result<Packet<'a>, Error> {
116        Self::parse_impl(data, false).map(|(p, _)| p)
117    }
118
119    /// Parses a self-delimited packet (RFC 6716 Appendix B, used for all but the last stream of
120    /// a multistream packet); returns the packet and the number of bytes it occupies.
121    pub fn parse_self_delimited(data: &'a [u8]) -> Result<(Packet<'a>, usize), Error> {
122        Self::parse_impl(data, true)
123    }
124
125    fn parse_impl(data: &'a [u8], self_delimited: bool) -> Result<(Packet<'a>, usize), Error> {
126        const BAD: Error = Error::InvalidPacket("malformed packet framing");
127        if data.is_empty() {
128            return Err(Error::InvalidPacket("empty packet"));
129        }
130        let toc = Toc::parse(data[0]);
131        let fs = toc.frame_samples_48k();
132        let mut pos = 1usize;
133        let mut len = data.len() as isize - 1;
134        let mut sizes: Vec<isize> = Vec::new();
135        let mut padding = 0usize;
136        let mut cbr = false;
137        let count;
138        let mut last_size = len;
139        let read_size = |pos: &mut usize, len: &mut isize| -> Result<isize, Error> {
140            let avail = &data[*pos..(*pos + (*len).max(0) as usize).min(data.len())];
141            let (s, n) = parse_size(avail).ok_or(BAD)?;
142            *pos += n;
143            *len -= n as isize;
144            Ok(s as isize)
145        };
146        match toc.code {
147            0 => count = 1,
148            1 => {
149                count = 2;
150                cbr = true;
151                if !self_delimited {
152                    if len & 1 != 0 {
153                        return Err(BAD);
154                    }
155                    last_size = len / 2;
156                    sizes.push(last_size);
157                }
158            }
159            2 => {
160                count = 2;
161                let s0 = read_size(&mut pos, &mut len)?;
162                if s0 > len {
163                    return Err(BAD);
164                }
165                sizes.push(s0);
166                last_size = len - s0;
167            }
168            _ => {
169                if len < 1 {
170                    return Err(BAD);
171                }
172                let ch = data[pos];
173                pos += 1;
174                len -= 1;
175                count = (ch & 0x3F) as usize;
176                if count == 0 || fs * count > 5760 {
177                    return Err(BAD);
178                }
179                if ch & 0x40 != 0 {
180                    loop {
181                        if len <= 0 {
182                            return Err(BAD);
183                        }
184                        let b = data[pos];
185                        pos += 1;
186                        len -= 1;
187                        let tmp = if b == 255 { 254 } else { b as isize };
188                        len -= tmp;
189                        padding += tmp as usize;
190                        if b != 255 {
191                            break;
192                        }
193                    }
194                }
195                if len < 0 {
196                    return Err(BAD);
197                }
198                cbr = ch & 0x80 == 0;
199                if !cbr {
200                    last_size = len;
201                    for _ in 0..count - 1 {
202                        let before = len;
203                        let s = read_size(&mut pos, &mut len)?;
204                        if s > len {
205                            return Err(BAD);
206                        }
207                        sizes.push(s);
208                        last_size -= (before - len) + s;
209                    }
210                    if last_size < 0 {
211                        return Err(BAD);
212                    }
213                } else if !self_delimited {
214                    last_size = len / count as isize;
215                    if last_size * count as isize != len {
216                        return Err(BAD);
217                    }
218                    for _ in 0..count - 1 {
219                        sizes.push(last_size);
220                    }
221                }
222            }
223        }
224        if self_delimited {
225            let before = len;
226            let s = read_size(&mut pos, &mut len)?;
227            if s > len {
228                return Err(BAD);
229            }
230            if cbr {
231                if s * count as isize > len {
232                    return Err(BAD);
233                }
234                sizes = vec![s; count - 1];
235            } else if (before - len) + s > last_size {
236                return Err(BAD);
237            }
238            last_size = s;
239        }
240        if last_size > 1275 || sizes.iter().any(|&s| s > 1275) {
241            return Err(BAD);
242        }
243        sizes.push(last_size);
244        let mut frames = Vec::with_capacity(count);
245        for &s in &sizes {
246            let s = s as usize;
247            if pos + s > data.len() {
248                return Err(BAD);
249            }
250            frames.push(&data[pos..pos + s]);
251            pos += s;
252        }
253        let total = if self_delimited { pos + padding } else { data.len() };
254        if total > data.len() {
255            return Err(BAD);
256        }
257        Ok((Packet { toc, frames, padding }, total))
258    }
259}

261/// Number of samples (at 48 kHz) in a packet, or an error for malformed framing.
262pub fn packet_samples_48k(data: &[u8]) -> Result<usize, Error> {
263    let p = Packet::parse(data)?;
264    Ok(p.toc.frame_samples_48k() * p.frames.len())
265}

271    #[cfg(any())]
272    fn toc_configs() {
273        for cfg in 0u8..32 {
274            let t = Toc::parse(cfg << 3);
275            let ms10 = t.frame_samples(48000) * 10 / 48; // tenths of ms
276            let expect = match cfg {
277                0..=11 => [100, 200, 400, 600][cfg as usize & 3],
278                12..=15 => [100, 200][cfg as usize & 1],
279                _ => [25, 50, 100, 200][cfg as usize & 3],
280            };
281            assert_eq!(ms10, expect, "config {cfg}");
282        }
283        assert!(Toc::parse(0x04).stereo);
284    }
285
286    #[cfg(any())]
287    fn frame_codes() {
288        // code 0
289        let p = Packet::parse(&[0x00, 1, 2, 3]).unwrap();
290        assert_eq!(p.frames, vec![&[1u8, 2, 3][..]]);
291        // code 1 (odd payload is invalid)
292        assert!(Packet::parse(&[0x01, 1, 2, 3]).is_err());
293        let p = Packet::parse(&[0x01, 1, 2, 3, 4]).unwrap();
294        assert_eq!(p.frames.len(), 2);
295        // code 2
296        let p = Packet::parse(&[0x02, 1, 9, 8, 7]).unwrap();
297        assert_eq!(p.frames, vec![&[9u8][..], &[8u8, 7][..]]);
298        assert!(Packet::parse(&[0x02, 5, 1]).is_err());
299        // code 3 CBR with padding: 2 frames of 2 bytes, 3 bytes padding
300        let p = Packet::parse(&[0x03, 0x42, 3, 1, 2, 3, 4, 0, 0, 0]).unwrap();
301        assert_eq!(p.frames, vec![&[1u8, 2][..], &[3u8, 4][..]]);
302        assert_eq!(p.padding, 3);
303        // code 3 VBR
304        let p = Packet::parse(&[0x03, 0x83, 1, 2, 10, 20, 21, 30]).unwrap();
305        assert_eq!(p.frames, vec![&[10u8][..], &[20u8, 21][..], &[30u8][..]]);
306        // too many frames (> 120 ms)
307        assert!(Packet::parse(&[0x03 | (3 << 3), 0x03, 0]).is_err());
308        // zero frames
309        assert!(Packet::parse(&[0x03, 0x00]).is_err());
310        assert!(Packet::parse(&[]).is_err());
311    }
