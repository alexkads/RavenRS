70/// Decodes the packets of one audio track to planar `f32`, in WAV channel order when
71/// [`TrackDecoder::layout`] is ordered.
72///
73/// It takes the track's [`CodecParams`] (from `isobmff::Track::codec_params`,
74/// `matroska::Track::codec_params`, `ogg::Stream::codec_params`, or built by hand for another
75/// container) and decodes AAC (LC; HE-AAC as its upsampled LC core), ALAC, Opus, AC-3/E-AC-3,
76/// MP1/MP2, MP3 and PCM. Container timing (AAC priming, Opus pre-skip, end trimming) is the caller's:
77/// every decoded sample is returned.
78///
79/// ```ignore
80/// # fn demo(params: &ravenrs_deck::craft::audio::CodecParams, packets: &[Vec<u8>]) -> ravenrs_deck::craft::audio::Result<()> {
81/// use ravenrs_deck::craft::audio::TrackDecoder;
82///
83/// let lookahead: Vec<&[u8]> = packets.iter().take(8).map(Vec::as_slice).collect();
84/// let mut dec = TrackDecoder::new(params, &lookahead)?;
85/// println!("{} at {} Hz, {}", dec.codec().name(), dec.sample_rate(), dec.layout());
86/// for p in packets {
87///     let block = match dec.decode(p) {
88///         Ok(block) => block,
89///         Err(_) => dec.conceal(Some(p))?, // keep the timing across a damaged packet
90///     };
91///     // block.channels: one Vec<f32> per channel
92/// }
93/// # Ok(()) }
94/// ```
95pub struct TrackDecoder {
96    inner: Inner,
97    codec: CodecId,
98    sample_rate: u32,
99    layout: ChannelLayout,
100    sample_format: SampleFormat,
101    /// Frames of the last decoded packet, the length of a concealed one when the codec does not
102    /// say.
103    last_frames: usize,
104}

116impl TrackDecoder {
117    /// A decoder for the track `params` describes. `lookahead` holds the track's first packets
118    /// (up to [`LOOKAHEAD_PACKETS`]; empty when they are not at hand): HE-AAC signalled only
119    /// inside the stream is found in them, and AC-3 and MPEG audio take their rate and channel
120    /// layout (MPEG audio its layer too) from the first frame. Without them those are taken from
121    /// `params` and set from the first decoded packet.
122    ///
123    /// Streams without codec configuration are accepted where the packets carry it: AAC with
124    /// empty `extradata` takes its configuration from the ADTS headers of the lookahead (MPEG-TS,
125    /// raw `.aac`), the one most of them share, and ADTS headers are stripped from every packet;
126    /// Opus with empty `extradata` and 1 or 2 `channels` (RTP, MPEG-TS) needs no `OpusHead`. Other
127    /// missing configuration is [`AudioError::Malformed`].
128    ///
129    /// ALAC needs its magic cookie in `extradata`.
130    ///
131    /// Codecs without a native packet decoder yet (FLAC, Vorbis) are
132    /// [`AudioError::Unsupported`]; `crate::craft::audio::decode` decodes them through symphonia when the
133    /// `symphonia` feature is on.
134    pub fn new(params: &CodecParams, lookahead: &[&[u8]]) -> Result<TrackDecoder> {
135        let _ = lookahead;
136        let rate = params.sample_rate.unwrap_or(0);
137        let channels = usize::from(params.channels.unwrap_or(0));
138        // AC-3 and MPEG audio read their parameters from the first frame; AAC reads the lookahead;
139        // Opus needs none.
140        #[allow(unused_variables)]
141        let first = lookahead.first().copied().unwrap_or_default();
142        // MP4 and Matroska packets may hold several MPEG audio frames: then they are not all
143        // one frame long.
144        #[cfg(any())]
145        let several = lookahead.iter().any(|p| mpa_packet_frames(p).0 > 1);
146        let (inner, codec, sample_rate, layout, sample_format) = match params.codec {
147            #[cfg(any())]
148            CodecId::Aac | CodecId::HeAac => {
149                // ADTS-framed packets (MPEG-TS, raw `.aac`) carry their configuration in each
150                // header: without extradata, the headers of the lookahead give it.
151                let payloads: Vec<&[u8]> = lookahead
152                    .iter()
153                    .take(LOOKAHEAD_PACKETS)
154                    .flat_map(|p| adts_frames(p).map_or_else(|| vec![*p], |frames| frames.into_iter().map(|(_, unit)| unit).collect()))
155                    .collect();
156                let mut adts_format = None;
157                let asc = if params.extradata.is_empty() {
158                    let headers: Vec<aac::AdtsHeader> = lookahead.iter().filter_map(|p| aac::AdtsHeader::parse(p).ok()).collect();
159                    let h = adts_stream_header(&headers).ok_or_else(|| {
160                        AudioError::Malformed("AAC track without an AudioSpecificConfig (extradata), and its packets are not ADTS".into())
161                    })?;
162                    adts_format = Some(adts_format_of(&h));
163                    adts_config(&h, &payloads)?
164                } else {
165                    params.extradata.clone()
166                };
167                let asc = asc.as_slice();
168                let dec = aac::Decoder::new(asc)?;
169                let out = aac::output_rate(asc, payloads.iter().copied()).unwrap_or(dec.sample_rate());
170                let up = (dec.sample_rate().checked_mul(2) == Some(out)).then(|| aac::Upsample2x::new(dec.channels()));
171                let codec = if up.is_some() { CodecId::HeAac } else { CodecId::Aac };
172                // Only twice the core rate is upsampled to; any other signalled rate plays the core.
173                let out = if up.is_some() { out } else { dec.sample_rate() };
174                // a program config element that spells out a configuration has its speakers
175                let cfg = aac::speaker_config(dec.config());
176                let layout = aac_layout(cfg, dec.channels());
177                let order = layout.is_ordered().then(|| aac::wav_order(cfg)).flatten();
178                (Inner::Aac { dec: Box::new(dec), up, order, adts_format }, codec, out, layout, SampleFormat::Compressed)
179            }
180            #[cfg(any())]
181            CodecId::Alac => {
182                if params.extradata.is_empty() {
183                    return Err(AudioError::Malformed("ALAC track without a magic cookie (extradata)".into()));
184                }
185                let dec = alac::Decoder::new(&params.extradata)?;
186                let cfg = *dec.config();
187                // The cookie's rate is the codec's; the container's stands in when it is 0.
188                let rate = if cfg.sample_rate > 0 { cfg.sample_rate } else { rate };
189                if rate == 0 {
190                    return Err(AudioError::Malformed("ALAC track without a sample rate in its cookie or container".into()));
191                }
192                let (layout, order) = if cfg.has_standard_layout() {
193                    (ChannelLayout::from_mask(alac::channel_mask(cfg.channels)), alac::wav_order(cfg.channels))
194                } else {
195                    (ChannelLayout::unordered(u16::from(cfg.channels)), None)
196                };
197                let sample_format = match cfg.bit_depth {
198                    1..=8 => SampleFormat::Int8,
199                    9..=16 => SampleFormat::Int16,
200                    17..=24 => SampleFormat::Int24,
201                    _ => SampleFormat::Int32,
202                };
203                let scale = 2f64.powi(1 - i32::from(cfg.bit_depth));
204                (Inner::Alac { dec, order, scale }, CodecId::Alac, rate, layout, sample_format)
205            }
206            #[cfg(any())]
207            CodecId::Ac3 | CodecId::Eac3 => match ac3::parse_header(first) {
208                Ok(h) => {
209                    let codec = if h.is_eac3() { CodecId::Eac3 } else { CodecId::Ac3 };
210                    (Inner::Ac3(Box::default(), Ac3Packets::of(lookahead)), codec, h.sample_rate, ac3_layout(&h), SampleFormat::Compressed)
211                }
212                Err(_) => (
213                    Inner::Ac3(Box::default(), Ac3Packets::default()),
214                    params.codec,
215                    rate,
216                    ChannelLayout::unordered(crate::craft::audio::layout::count(channels)),
217                    SampleFormat::Compressed,
218                ),
219            },
220            CodecId::Opus => {
221                // Opus over RTP or in MPEG-TS has no OpusHead: mono and stereo need none (RFC 7845
222                // mapping family 0), so one is made from the channel count.
223                let head = match (params.extradata.is_empty(), channels) {
224                    (false, _) => opus::OpusHead::parse(&params.extradata)?,
225                    (true, 1 | 2) => opus::OpusHead::parse(&default_opus_head(channels as u8, rate))?,
226                    (true, _) => {
227                        return Err(AudioError::Malformed("Opus track without an OpusHead (extradata) needs 1 or 2 channels".into()));
228                    }
229                };
230                let layout = opus_layout(&head);
231                let order = if head.mapping_family == 1 { opus::wav_order(head.channels) } else { None };
232                let dec = opus::Decoder::from_head(head, 48_000)?;
233                (Inner::Opus { dec: Box::new(dec), order }, CodecId::Opus, 48_000, layout, SampleFormat::Compressed)
234            }
235            #[cfg(any())]
236            CodecId::Mp1 | CodecId::Mp2 | CodecId::Mp3 => match mpa::parse_header(first) {
237                #[cfg(any())]
238                Ok(h) if h.layer == 3 => {
239                    let mut dec = Box::<mp3::Decoder>::default();
240                    dec.prime(h);
241                    (Inner::Mp3 { dec, samples: h.samples(), several }, CodecId::Mp3, h.sample_rate, mpa_layout(&h), SampleFormat::Compressed)
242                }
243                Ok(h) if h.layer == 3 => return Err(unsupported(CodecId::Mp3)),
244                Ok(h) => {
245                    let codec = if h.layer == 1 { CodecId::Mp1 } else { CodecId::Mp2 };
246                    (Inner::Mpa { dec: Box::default(), several }, codec, h.sample_rate, mpa_layout(&h), SampleFormat::Compressed)
247                }
248                // Without a first frame, the container's rate and channels stand in until the
249                // first frame is decoded.
250                #[cfg(any())]
251                Err(_) if params.codec == CodecId::Mp3 => (
252                    Inner::Mp3 { dec: Box::default(), samples: if rate >= 32_000 { 1152 } else { 576 }, several },
253                    CodecId::Mp3,
254                    rate,
255                    ChannelLayout::default_for(crate::craft::audio::layout::count(channels)),
256                    SampleFormat::Compressed,
257                ),
258                Err(_) if params.codec == CodecId::Mp3 => return Err(unsupported(CodecId::Mp3)),
259                Err(_) => (
260                    Inner::Mpa { dec: Box::default(), several },
261                    params.codec,
262                    rate,
263                    ChannelLayout::default_for(crate::craft::audio::layout::count(channels)),
264                    SampleFormat::Compressed,
265                ),
266            },
267            CodecId::Pcm => {
268                let pcm = params.pcm.ok_or_else(|| AudioError::Malformed("PCM track without a sample format".into()))?;
269                let (kind, sample_format) = pcm_kind(pcm.bits, pcm.float, pcm.big_endian, !pcm.signed)
270                    .ok_or_else(|| AudioError::Unsupported(format!("{}-bit PCM", pcm.bits)))?;
271                if channels == 0 || rate == 0 {
272                    return Err(AudioError::Malformed("PCM track without channels or a sample rate".into()));
273                }
274                (
275                    Inner::Pcm { kind, channels },
276                    CodecId::Pcm,
277                    rate,
278                    ChannelLayout::default_for(crate::craft::audio::layout::count(channels)),
279                    sample_format,
280                )
281            }
282            other => return Err(unsupported(other)),
283        };
284        Ok(TrackDecoder { inner, codec, sample_rate, layout, sample_format, last_frames: 0 })
285    }

313    /// Decode one packet. An MP4 or Matroska packet may hold several AC-3 or MPEG audio frames;
314    /// an E-AC-3 dependent substream decodes to no frames. A packet that fails is an error for
315    /// every codec: call [`TrackDecoder::conceal`] for audio of its length, so what follows keeps
316    /// its timing.
317    pub fn decode(&mut self, packet: &[u8]) -> Result<AudioBuffer> {
318        // A packet decoded only in part says nothing of how long packets are.
319        #[allow(unused_mut)]
320        let mut partial = false;
321        let channels = match &mut self.inner {
322            #[cfg(any())]
323            Inner::Aac { dec, up, order, adts_format } => {
324                let mut out: Vec<Vec<f32>> = Vec::new();
325                let units = match adts_frames(packet) {
326                    Some(frames) => adts_units(frames, *adts_format)?,
327                    None => vec![packet],
328                };
329                // A failed frame leaves the decoder as it was, so the frames before it in the
330                // packet must too: their audio is replaced by the packet's concealment.
331                let before = (units.len() > 1).then(|| (dec.clone(), up.clone()));
332                let decoded = units.into_iter().try_for_each(|unit| {
333                    let mut block = dec.decode(unit)?;
334                    if let Some(up) = up.as_mut() {
335                        block = up.process(block);
336                    }
337                    append(&mut out, block)
338                });
339                if let Err(e) = decoded {
340                    if let Some(before) = before {
341                        (*dec, *up) = before;
342                    }
343                    return Err(e);
344                }
345                reorder(out, *order)
346            }
347            #[cfg(any())]
348            Inner::Alac { dec, order, scale } => {
349                let decoded = dec.decode(packet)?;
350                let floats = decoded.into_iter().map(|c| c.into_iter().map(|v| (f64::from(v) * *scale) as f32).collect()).collect();
351                reorder(floats, *order)
352            }
353            #[cfg(any())]
354            Inner::Mpa { dec, several } => {
355                let (out, first, count, cut) = mpa_frames(packet, |f| dec.decode(f))?;
356                *several |= count > 1;
357                partial = cut;
358                if let Some(h) = first.filter(|_| self.last_frames == 0) {
359                    self.sample_rate = h.sample_rate;
360                    self.layout = mpa_layout(&h);
361                }
362                out
363            }
364            #[cfg(any())]
365            Inner::Mp3 { dec, samples, several } => {
366                let (out, first, count, cut) = mpa_frames(packet, |f| dec.decode(f))?;
367                *several |= count > 1;
368                partial = cut;
369                if let Some(h) = first {
370                    *samples = h.samples();
371                    if self.last_frames == 0 {
372                        self.sample_rate = h.sample_rate;
373                        self.layout = mpa_layout(&h);
374                    }
375                }
376                out
377            }
378            #[cfg(any())]
379            Inner::Ac3(d, shape) => {
380                let mut out: Vec<Vec<f32>> = Vec::new();
381                let mut p = 0;
382                while let Some(rest) = packet.get(p..).filter(|r| !r.is_empty()) {
383                    let f = match d.decode(rest) {
384                        Ok(f) => f,
385                        Err(e) if out.is_empty() => return Err(e.into()),
386                        Err(_) => {
387                            partial = true;
388                            break;
389                        }
390                    };
391                    p += f.header.frame_bytes.max(1);
392                    shape.short |= f.header.is_primary() && f.header.samples() < AC3_FRAME_SAMPLES;
393                    if !f.channels.is_empty() {
394                        if self.last_frames == 0 {
395                            self.sample_rate = f.header.sample_rate;
396                            self.layout = ac3_layout(&f.header);
397                        }
398                        append(&mut out, f.channels)?;
399                    }
400                }
401                let frames = out.first().map_or(0, Vec::len);
402                shape.varies |= !partial && frames > 0 && shape.first > 0 && frames != shape.first;
403                out
404            }
405            Inner::Opus { dec, order } => {
406                let out = dec.decode(Some(packet))?;
407                reorder(out, *order)
408            }
409            Inner::Pcm { kind, channels } => {
410                let frames = (packet.len() / (kind.bytes() * *channels).max(1)) as u64;
411                deinterleave(packet, *kind, *channels, frames)?
412            }
413        };
414        if let Some(n) = channels.iter().map(Vec::len).min().filter(|&n| n > 0 && !partial) {
415            self.last_frames = n;
416            if channels.len() != usize::from(self.layout.channels()) {
417                self.layout = ChannelLayout::unordered(crate::craft::audio::layout::count(channels.len()));
418            }
419        }
420        Ok(AudioBuffer { sample_rate: self.sample_rate, channels })
421    }

423    /// What stands in for a lost or undecodable packet (`packet` when its bytes are at hand):
424    /// Opus's loss concealment, or silence, as long as the packet would have been (from its TOC or
425    /// frame header, the codec's frame length, or the last packet). Empty when the length is
426    /// unknown.
427    /// Fails with [`AudioError::TooLarge`](crate::craft::audio::AudioError::TooLarge) when that silence would pass
428    /// the decode limit (2^30 samples, as [`decode`](Self::decode) has) or cannot be allocated,
429    /// rather than return less than the packet's length.
430    pub fn conceal(&mut self, packet: Option<&[u8]>) -> Result<AudioBuffer> {
431        let data = packet.unwrap_or_default();
432        let frames = match &mut self.inner {
433            Inner::Opus { dec, order } => {
434                let order = *order;
435                if let Ok(out) = dec.conceal(packet) {
436                    return Ok(AudioBuffer { sample_rate: self.sample_rate, channels: reorder(out, order) });
437                }
438                None
439            }
440            #[cfg(any())]
441            Inner::Aac { up, .. } => {
442                let per_unit = aac::FRAME_LEN * if up.is_some() { 2 } else { 1 };
443                match adts_frames(data) {
444                    Some(f) => Some(f.iter().map(|(h, _)| usize::from(h.raw_data_blocks)).sum::<usize>().min(MAX_ADTS_FRAMES_PER_PACKET) * per_unit),
445                    // A packet lost whole: as long as the last one, in a stream of multi-frame
446                    // packets as in any other.
447                    None if data.is_empty() && self.last_frames > 0 => Some(self.last_frames),
448                    None => Some(per_unit),
449                }
450            }
451            // A damaged packet whose header survives says its length; otherwise a packet is as
452            // long as the last one (a full one at the start).
453            #[cfg(any())]
454            Inner::Alac { dec, .. } => {
455                Some(dec.packet_frames(data).unwrap_or(if self.last_frames > 0 { self.last_frames } else { dec.config().frame_length as usize }))
456            }
457            #[cfg(any())]
458            Inner::Ac3(..) => Some(ac3_packet_frames(data)),
459            #[cfg(any())]
460            // a packet lost before any has decoded is a frame
461            Inner::Mpa { .. } => Some(match mpa_packet_frames(data) {
462                (0, _) if self.last_frames == 0 && self.codec == CodecId::Mp1 => 384,
463                (0, _) if self.last_frames == 0 => 1152,
464                (_, n) => n,
465            }),
466            // a packet lost whole is as long as the last one, a frame at the start
467            #[cfg(any())]
468            Inner::Mp3 { samples, .. } => Some(match mpa_packet_frames(data) {
469                (0, _) if self.last_frames > 0 => self.last_frames,
470                (0, _) => *samples,
471                (_, n) => n,
472            }),
473            Inner::Pcm { kind, channels } => Some(data.len() / (kind.bytes() * *channels).max(1)),
474        };
475        let frames = frames.filter(|&n| n > 0).unwrap_or(self.last_frames);
476        Ok(AudioBuffer { sample_rate: self.sample_rate, channels: crate::craft::audio::pcm::silence(usize::from(self.layout.channels()), frames)? })
477    }

479    /// The frames every packet decodes to, for codecs whose packets all have one length: AAC
480    /// (1024, 2048 for HE-AAC; a packet of several ADTS frames decodes to that many times this) and
481    /// AC-3 / E-AC-3 (the frames of the last decoded packet, else of the first packet's primary
482    /// syncframes, else 1536) and MP1/MP2/MP3 (one frame per packet; 576 for MP3 at the lower
483    /// sampling frequencies); `None` for Opus, PCM and ALAC (whose last packet is shorter; the
484    /// container's timing says), for AC-3 / E-AC-3 whose packets read so far differ in length,
485    /// and for MPEG audio once a packet (in the lookahead or decoded) has held several frames.
486    pub fn frames_per_packet(&self) -> Option<usize> {
487        match &self.inner {
488            #[cfg(any())]
489            Inner::Aac { up, .. } => Some(aac::FRAME_LEN * if up.is_some() { 2 } else { 1 }),
490            #[cfg(any())]
491            Inner::Alac { .. } => None,
492            #[cfg(any())]
493            Inner::Ac3(_, shape) if shape.varies => None,
494            #[cfg(any())]
495            Inner::Ac3(_, shape) => Some(match (self.last_frames, shape.first) {
496                (0, 0) => AC3_FRAME_SAMPLES,
497                (0, n) | (n, _) => n,
498            }),
499            Inner::Opus { .. } => None,
500            #[cfg(any())]
501            Inner::Mpa { several: true, .. } => None,
502            #[cfg(any())]
503            Inner::Mpa { .. } => Some(if self.codec == CodecId::Mp1 { 384 } else { 1152 }),
504            #[cfg(any())]
505            Inner::Mp3 { several: true, .. } => None,
506            #[cfg(any())]
507            Inner::Mp3 { samples, .. } => Some(*samples),
508            Inner::Pcm { .. } => None,
509        }
510    }
511
512    /// The frames of a full packet, for codecs whose packets all have that length except ones
513    /// at the end of the stream: ALAC's frame length (only the last packet is shorter). Where
514    /// [`Self::frames_per_packet`] is `Some`, the same value.
515    #[cfg(any())]
516    pub(crate) fn full_packet_frames(&self) -> Option<usize> {
517        match &self.inner {
518            #[cfg(any())]
519            Inner::Alac { dec, .. } => Some(dec.config().frame_length as usize),
520            _ => self.frames_per_packet(),
521        }
522    }
523
524    /// How many frames a packet holds, read from the packet alone, for codecs whose packets
525    /// say: ALAC's headers (given the frame length).
526    #[cfg(any())]
527    pub(crate) fn packet_frames_reader(&self) -> Option<PacketFrames> {
528        match &self.inner {
529            #[cfg(any())]
530            Inner::Alac { .. } => Some(crate::craft::audio::alac::header_frames),
531            _ => None,
532        }
533    }
534
535    /// Frames to decode before a seek target and drop, so the decoder has settled when the target
536    /// is reached: two AAC frames, 80 ms of Opus (RFC 7845 §4.6), one AC-3 or MP1/MP2 frame
537    /// (their filter banks overlap one frame), for MP3 the frames its bit reservoir can reach back
538    /// over and two more, none for ALAC (its packets decode independently)
539    /// and PCM. E-AC-3 syncframes of fewer than six blocks may reuse exponents from any syncframe
540    /// before them, so once one has been read the stream decodes from its start (`usize::MAX`).
541    pub fn preroll(&self) -> usize {
542        match &self.inner {
543            #[cfg(any())]
544            Inner::Aac { up, .. } => 2 * aac::FRAME_LEN * if up.is_some() { 2 } else { 1 },
545            #[cfg(any())]
546            Inner::Alac { .. } => 0,
547            #[cfg(any())]
548            Inner::Ac3(_, shape) if shape.short => usize::MAX,
549            #[cfg(any())]
550            Inner::Ac3(..) => AC3_FRAME_SAMPLES,
551            Inner::Opus { .. } => 3840,
552            #[cfg(any())]
553            Inner::Mpa { .. } => self.last_frames.max(1152),
554            // the bit reservoir reaches up to 511 bytes back: as many frames as that takes
555            #[cfg(any())]
556            Inner::Mp3 { dec, samples, .. } => dec.preroll().unwrap_or(MP3_PREROLL_FRAMES * *samples),
557            Inner::Pcm { .. } => 0,
558        }
559    }
560
561    /// Forget the decoder state, for a seek: the next packet decodes as if it were the first
562    /// (codecs with overlap, such as AAC and Opus, need [`TrackDecoder::preroll`] frames to
563    /// settle).
564    pub fn reset(&mut self) {
565        // the packet before a seek says nothing about the length of a packet lost after it
566        self.last_frames = 0;
567        match &mut self.inner {
568            #[cfg(any())]
569            Inner::Aac { dec, up, .. } => {
570                dec.reset();
571                if let Some(up) = up {
572                    up.reset();
573                }
574            }
575            #[cfg(any())]
576            Inner::Alac { .. } => {}
577            #[cfg(any())]
578            Inner::Ac3(d, _) => d.reset(),
579            Inner::Opus { dec, .. } => dec.reset(),
580            #[cfg(any())]
581            Inner::Mpa { dec, .. } => dec.reset(),
582            #[cfg(any())]
583            Inner::Mp3 { dec, .. } => dec.reset(),
584            Inner::Pcm { .. } => {}
585        }
586    }
587}

618/// Frames of an MP3 stream decoded before a seek target when neither a lookahead nor a decoded
619/// frame has said what the stream is: the most any stream needs. The smallest frames (8 kbit/s at
620/// 24 kHz, stereo, with CRC) hold 1 byte of main data, so 255 bytes of reservoir span 255 of them,
621/// and two more settle the filter banks.
622#[cfg(any())]
623const MP3_PREROLL_FRAMES: usize = 257;

625/// The MPEG audio frames of a packet (an MP4 or Matroska packet may hold several), decoded by
626/// `decode`, the first frame's header, how many frames decoded, and whether a frame after the
627/// first failed, which ends the packet (then it decoded only in part).
628#[cfg(any())]
629#[allow(clippy::type_complexity)]
630fn mpa_frames(packet: &[u8], mut decode: impl FnMut(&[u8]) -> mpa::Result<mpa::Frame>) -> Result<(Vec<Vec<f32>>, Option<mpa::Header>, usize, bool)> {
631    let mut out: Vec<Vec<f32>> = Vec::new();
632    let mut first = None;
633    let mut count = 0;
634    let mut p = 0;
635    while let Some(rest) = packet.get(p..).filter(|r| !r.is_empty()) {
636        let f = match decode(rest) {
637            Ok(f) => f,
638            Err(e) if out.is_empty() => return Err(e.into()),
639            Err(_) => return Ok((out, first, count, true)),
640        };
641        p += f.header.frame_bytes.max(1);
642        first.get_or_insert(f.header);
643        count += 1;
644        append(&mut out, f.channels)?;
645    }
646    Ok((out, first, count, false))
647}

649/// The MPEG audio frames a packet holds by their headers, and the samples per channel they
650/// decode to: frames are counted while their headers parse, the last one even when the packet
651/// ends inside it.
652#[cfg(any())]
653fn mpa_packet_frames(packet: &[u8]) -> (usize, usize) {
654    let (mut frames, mut samples, mut p) = (0, 0, 0);
655    while let Some(h) = packet.get(p..).and_then(|rest| mpa::parse_header(rest).ok()) {
656        frames += 1;
657        samples += h.samples();
658        if h.frame_bytes == 0 {
659            break;
660        }
661        p += h.frame_bytes;
662    }
663    (frames, samples)
664}

666/// Append `more` to `out` channel by channel. A packet whose frames differ in channel count is
667/// malformed: one buffer cannot hold them.
668#[cfg(any())]
669fn append(out: &mut Vec<Vec<f32>>, more: Vec<Vec<f32>>) -> Result<()> {
670    if out.is_empty() {
671        *out = more;
672    } else if out.len() == more.len() {
673        for (o, c) in out.iter_mut().zip(more) {
674            o.extend(c);
675        }
676    } else {
677        return Err(AudioError::Malformed("the frames of a packet change the channel count".into()));
678    }
679    Ok(())
680}

694/// The frames of a six-block (AC-3 or E-AC-3) syncframe.
695#[cfg(any())]
696const AC3_FRAME_SAMPLES: usize = 1536;

698/// What the AC-3 / E-AC-3 packets read so far say of the track's packets.
699#[cfg(any())]
700#[derive(Clone, Copy, Debug, Default)]
701struct Ac3Packets {
702    /// The frames the first packet's primary syncframes hold (0 when unknown).
703    first: usize,
704    /// Packets have decoded to different lengths: a packet's index does not give its position.
705    varies: bool,
706    /// A primary syncframe of fewer than six blocks (E-AC-3), whose exponents may come from any
707    /// syncframe before it.
708    short: bool,
709}
710
711#[cfg(any())]
712impl Ac3Packets {
713    fn of(lookahead: &[&[u8]]) -> Ac3Packets {
714        let lengths: Vec<usize> = lookahead.iter().map(|p| ac3_packet_frames(p)).filter(|&n| n > 0).collect();
715        let short = lookahead.iter().any(|p| ac3_syncframes(p).any(|h| h.is_primary() && h.samples() < AC3_FRAME_SAMPLES));
716        Ac3Packets { first: lookahead.first().map_or(0, |p| ac3_packet_frames(p)), varies: lengths.windows(2).any(|w| w[0] != w[1]), short }
717    }
718}

720/// The headers of the syncframes `packet` holds, up to the first that does not parse.
721#[cfg(any())]
722fn ac3_syncframes(packet: &[u8]) -> impl Iterator<Item = ac3::Header> + '_ {
723    let mut p = 0;
724    std::iter::from_fn(move || {
725        let h = ac3::parse_header(packet.get(p..)?).ok()?;
726        p += h.frame_bytes.max(1);
727        Some(h)
728    })
729}

731/// The frames an AC-3 / E-AC-3 packet decodes to: those of every primary syncframe in it (an MP4
732/// or Matroska packet may hold several, as E-AC-3 frames of fewer than six blocks are grouped),
733/// up to the first that does not parse. 0 when none does.
734#[cfg(any())]
735fn ac3_packet_frames(packet: &[u8]) -> usize {
736    ac3_syncframes(packet).filter(ac3::Header::is_primary).map(|h| h.samples()).sum()
737}

745/// The configuration of an AAC stream from its ADTS `header`. Channel configuration 0 leaves
746/// the layout to a program config element in the stream, looked for in `first_units`.
747#[cfg(any())]
748fn adts_config(header: &aac::AdtsHeader, first_units: &[&[u8]]) -> Result<Vec<u8>> {
749    if header.channel_config != 0 {
750        return Ok(header.to_asc().to_bytes()?);
751    }
752    let mut dec = aac::Decoder::from_adts(header)?;
753    for unit in first_units {
754        let _ = dec.decode(unit);
755        if dec.config().pce.is_some() {
756            return Ok(dec.config().to_bytes()?);
757        }
758    }
759    Err(AudioError::Malformed("ADTS stream with channel configuration 0, and no program config element in its first packets".into()))
760}

762/// The whole frames at the start of an ADTS-framed packet, without a damaged or truncated end, or
763/// `None` when the packet does not start with one. A raw access unit never looks like one: its first
764/// three bits would be 7, ID_END, leaving nothing to decode.
765#[cfg(any())]
766fn adts_frames(packet: &[u8]) -> Option<Vec<(aac::AdtsHeader, &[u8])>> {
767    if packet.len() < 7 || packet[0] != 0xFF || packet[1] & 0xF6 != 0xF0 {
768        return None;
769    }
770    // Past the limit, what matters is only that there are too many: `adts_units` refuses them.
771    aac::split_adts_upto(packet, MAX_ADTS_FRAMES_PER_PACKET + 1).ok()
772}

774/// What an ADTS frame says the stream's format is: profile, sampling frequency index and channel
775/// configuration.
776#[cfg(any())]
777pub(crate) type AdtsFormat = (u8, u8, u8);
778
779#[cfg(any())]
780pub(crate) fn adts_format_of(h: &aac::AdtsHeader) -> AdtsFormat {
781    (h.object_type, h.sf_index, h.channel_config)
782}

784/// The raw access units of an ADTS-framed packet. In a stream whose headers gave its `format`,
785/// a frame that changes it would play wrongly, so it is malformed; with extradata, that decides the
786/// format and headers are not compared with it. Frames with several raw data blocks are
787/// unsupported.
788#[cfg(any())]
789fn adts_units(frames: Vec<(aac::AdtsHeader, &[u8])>, format: Option<AdtsFormat>) -> Result<Vec<&[u8]>> {
790    if frames.len() > MAX_ADTS_FRAMES_PER_PACKET {
791        return Err(AudioError::Malformed(format!("a packet of more than {MAX_ADTS_FRAMES_PER_PACKET} ADTS frames")));
792    }
793    if frames.iter().any(|(h, _)| h.raw_data_blocks > 1) {
794        return Err(AudioError::Unsupported("ADTS frames with several raw data blocks".into()));
795    }
796    if format.is_some_and(|f| frames.iter().any(|(h, _)| adts_format_of(h) != f)) {
797        return Err(AudioError::Malformed("an ADTS frame changes the stream's format".into()));
798    }
799    Ok(frames.into_iter().map(|(_, unit)| unit).collect())
800}

989    /// One ADTS frame (stereo, 48 kHz, or channel configuration 0 with a PCE) of silence.
990    #[cfg(any())]
991    fn adts_frame(channel_config: u8) -> Vec<u8> {
992        use crate::craft::media::bitstream::BitWriter;
993        let mut bw = BitWriter::new();
994        if channel_config == 0 {
995            bw.write_bits(5, 3); // ID_PCE
996            aac::ProgramConfig { front: vec![(false, 0), (true, 0)], ..Default::default() }.write(&mut bw).unwrap();
997            bw.write_bits(0, 3); // ID_SCE
998            bw.write_bits(0, 4);
999            bw.write_bits(0, 22); // global_gain, ics_info, no pulse, TNS or gain control data
1000        }
1001        bw.write_bits(1, 3); // ID_CPE
1002        bw.write_bits(0, 5); // tag, common_window
1003        bw.write_bits(0, 44); // two such channel streams
1004        bw.write_bits(7, 3); // ID_END
1005        let au = bw.finish();
1006        [&aac::AdtsHeader::write(2, 3, channel_config, au.len(), 0x7FF).unwrap()[..], &au].concat()
1007    }
