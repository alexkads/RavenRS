1//! Planar `f32` audio buffer.
2
3/// Planar f32 audio.
4///
5/// Every channel is expected to hold the same number of frames; [`AudioBuffer::frames`] reports the
6/// shortest channel so a ragged buffer never causes out-of-bounds access.
7#[derive(Debug, Clone, Default, PartialEq)]
8pub struct AudioBuffer {
9    /// Sample rate in Hz.
10    pub sample_rate: u32,
11    /// One `Vec` of samples per channel.
12    pub channels: Vec<Vec<f32>>,
13}
14
15impl AudioBuffer {
16    /// A silent buffer with `channels` channels of `frames` frames each.
17    pub fn new(sample_rate: u32, channels: u16, frames: usize) -> Self {
18        Self { sample_rate, channels: vec![vec![0.0; frames]; usize::from(channels)] }
19    }
20
21    /// Number of frames (the length of the shortest channel; 0 when there are no channels).
22    pub fn frames(&self) -> usize {
23        self.channels.iter().map(Vec::len).min().unwrap_or(0)
24    }
25
26    /// Number of channels.
27    pub fn num_channels(&self) -> u16 {
28        u16::try_from(self.channels.len()).unwrap_or(u16::MAX)
29    }
30
31    /// Duration in seconds (0 when the sample rate is 0).
32    pub fn duration_secs(&self) -> f64 {
33        if self.sample_rate == 0 {
34            return 0.0;
35        }
36        self.frames() as f64 / f64::from(self.sample_rate)
37    }
38
39    /// Interleaved copy of the samples (frame-major: `L0 R0 L1 R1 ...`).
40    pub fn interleaved(&self) -> Vec<f32> {
41        let frames = self.frames();
42        let mut out = Vec::with_capacity(frames.saturating_mul(self.channels.len()));
43        for f in 0..frames {
44            for ch in &self.channels {
45                out.push(ch.get(f).copied().unwrap_or(0.0));
46            }
47        }
48        out
49    }
50
51    /// Build a planar buffer from interleaved samples. A trailing partial frame is dropped;
52    /// `channels == 0` yields an empty buffer.
53    pub fn from_interleaved(sr: u32, channels: u16, data: &[f32]) -> Self {
54        let channels = usize::from(channels);
55        if channels == 0 {
56            return Self { sample_rate: sr, channels: Vec::new() };
57        }
58        let frames = data.len() / channels;
59        let mut out: Vec<Vec<f32>> = (0..channels).map(|_| Vec::with_capacity(frames)).collect();
60        for frame in data.chunks_exact(channels) {
61            for (ch, &s) in out.iter_mut().zip(frame) {
62                ch.push(s);
63            }
64        }
65        Self { sample_rate: sr, channels: out }
66    }
67
68    /// Absolute peak over all channels (NaN samples are ignored).
69    pub fn peak(&self) -> f32 {
70        self.channels.iter().flat_map(|c| c.iter()).fold(0.0f32, |acc, &s| acc.max(s.abs()))
71    }
72}
