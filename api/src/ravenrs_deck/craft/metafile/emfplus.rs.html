1//! EMF+ (MS-EMFPLUS): GDI+ drawing records carried in EMF comment records.
2//!
3//! Most EMF+ files are *dual*: the same picture is also drawn with plain EMF records as a
4//! fallback for GDI. Office draws metafiles with GDI+, which plays the EMF+ records and skips
5//! the plain ones (except after EmfPlusGetDC), and a fallback can be incomplete (a nested
6//! metafile it lacks), so this reader does the same. It plays the records:
7//! - brushes (solid, hatch, texture, linear and path gradients), pens, paths, images, fonts,
8//!   string formats and regions, including objects split across several records;
9//! - rectangles, polygons, lines, ellipses, paths and Béziers;
10//! - bitmaps (raw pixel formats, and PNG/JPEG/GIF/TIFF/BMP payloads) and strings (DrawString,
11//!   DrawDriverString with character codes);
12//! - world and page transforms, save/restore and containers;
13//! - arcs and pies;
14//! - clipping to rectangles and convex clip paths and regions (others: their bounding rectangle);
15//! - nested metafile images, and glyph-index driver strings in the Windows core fonts;
16//! - cardinal splines (DrawCurve, DrawClosedCurve, FillClosedCurve), FillRegion, and Clear
17//!   (over the picture frame).
18//!
19//! Every drawing record of MS-EMFPLUS §2.3.4 is played. Not drawn: image attributes, glyph
20//! indices of other fonts, and region combinations other than intersect and union (XOR, exclude
21//! and complement keep the left area). Deliberately ignored: rendering hints (antialias, text
22//! hint, interpolation, pixel offset, compositing; the renderer has its own), SetRenderingOrigin
23//! (hatch phase), SerializableObject image effects, the terminal-server records SetTSGraphics and
24//! SetTSClip, and StrokeFillPath (0x4037), which the specification lists but gives no layout for
25//! and GDI+ doesn't write.

62/// What a brush paints. Coordinates are in brush space, which `xf` maps to world space.
63#[derive(Clone)]
64enum Fill {
65    Solid([u8; 4]),
66    Hatch { style: u32, fg: [u8; 4], bg: [u8; 4] },
67    Linear { rect: [f64; 4], stops: Vec<(f64, [u8; 4])>, wrap: u32, xf: Xform },
68    Radial { center: P, radius: P, stops: Vec<(f64, [u8; 4])>, xf: Xform },
69    Texture { img: Arc<Rgba>, wrap: u32, xf: Xform },
70}

72impl Fill {
73    /// One colour standing for the brush (pens and text draw solid).
74    fn color(&self) -> [u8; 4] {
75        match self {
76            Fill::Solid(c) | Fill::Hatch { fg: c, .. } => *c,
77            Fill::Linear { stops, .. } | Fill::Radial { stops, .. } => ramp(stops, 0.5),
78            Fill::Texture { img, .. } => img.px.get(..4).and_then(|c| c.try_into().ok()).unwrap_or([0, 0, 0, 255]),
79        }
80    }
81
82    /// The colour at a brush-space point.
83    fn sample(&self, p: P) -> [u8; 4] {
84        match self {
85            Fill::Solid(c) | Fill::Hatch { fg: c, .. } => *c,
86            Fill::Linear { rect, stops, wrap, .. } => {
87                let t = if rect[2] == 0.0 { 0.0 } else { (p.0 - rect[0]) / rect[2] };
88                ramp(stops, wrap_t(t, *wrap))
89            }
90            Fill::Radial { center, radius, stops, .. } => {
91                let (dx, dy) = ((p.0 - center.0) / radius.0.max(1e-9), (p.1 - center.1) / radius.1.max(1e-9));
92                ramp(stops, dx.hypot(dy).min(1.0))
93            }
94            Fill::Texture { img, wrap, .. } => {
95                let (w, h) = (img.w as f64, img.h as f64);
96                if w == 0.0 || h == 0.0 {
97                    return [0, 0, 0, 0];
98                }
99                if *wrap == 4 && !((0.0..w).contains(&p.0) && (0.0..h).contains(&p.1)) {
100                    return [0, 0, 0, 0];
101                }
102                let x = (p.0.rem_euclid(w) as usize).min(img.w - 1);
103                let y = (p.1.rem_euclid(h) as usize).min(img.h - 1);
104                let i = (y * img.w + x) * 4;
105                img.px.get(i..i + 4).and_then(|c| c.try_into().ok()).unwrap_or([0, 0, 0, 0])
106            }
107        }
108    }
109
110    fn xf(&self) -> Xform {
111        match self {
112            Fill::Linear { xf, .. } | Fill::Radial { xf, .. } | Fill::Texture { xf, .. } => *xf,
113            _ => IDENTITY,
114        }
115    }
116}

118/// Where `t` lands under a wrap mode: tiled, mirrored (flip modes) or clamped.
119fn wrap_t(t: f64, wrap: u32) -> f64 {
120    if !t.is_finite() {
121        return 0.0;
122    }
123    match wrap {
124        1 | 3 => {
125            let m = t.rem_euclid(2.0);
126            if m > 1.0 { 2.0 - m } else { m }
127        }
128        4 => t.clamp(0.0, 1.0),
129        _ => t.rem_euclid(1.0),
130    }
131}

133/// The colour at `t` along sorted stops.
134fn ramp(stops: &[(f64, [u8; 4])], t: f64) -> [u8; 4] {
135    let (Some(first), Some(last)) = (stops.first(), stops.last()) else { return [0, 0, 0, 0] };
136    if t <= first.0 {
137        return first.1;
138    }
139    for w in stops.windows(2) {
140        let ((a, ca), (b, cb)) = (w[0], w[1]);
141        if t <= b {
142            let k = if b > a { (t - a) / (b - a) } else { 0.0 };
143            return lerp(ca, cb, k);
144        }
145    }
146    last.1
147}
148
149fn lerp(a: [u8; 4], b: [u8; 4], k: f64) -> [u8; 4] {
150    let mut c = [0u8; 4];
151    for i in 0..4 {
152        c[i] = (f64::from(a[i]) + (f64::from(b[i]) - f64::from(a[i])) * k).round().clamp(0.0, 255.0) as u8;
153    }
154    c
155}

260    /// Plays the EMF+ records of one EMR_COMMENT payload (after the identifier).
261    pub fn comment(&mut self, cv: &mut Canvas, data: Bytes) {
262        self.gdi = false;
263        let mut pos = 0usize;
264        while !cv.full() {
265            let Some(tail) = data.0.get(pos..) else { break };
266            let tail = Bytes(tail);
267            let (Some(typ), Some(flags), Some(size), Some(dsize)) = (tail.u16(0), tail.u16(2), tail.u32(4), tail.u32(8)) else {
268                break;
269            };
270            let (size, dsize) = (size as usize, dsize as usize);
271            if size < 12 || dsize > size - 12 {
272                break;
273            }
274            let Some(rec) = tail.slice(12, dsize) else { break };
275            // Any EMF+ record ends a GetDC window; EmfPlusGetDC opens it again for its own.
276            self.gdi = false;
277            let _ = self.record(cv, typ, flags, Bytes(rec));
278            let Some(next) = pos.checked_add(size) else { break };
279            pos = next;
280        }
281    }

292    fn record(&mut self, cv: &mut Canvas, typ: u16, flags: u16, r: Bytes) -> Option<()> {
293        if typ == 0x4001 {
294            self.dual = flags & 1 != 0;
295            if let Some(d) = r.u32(8).map(f64::from).filter(|d| *d > 0.0 && *d < 100_000.0) {
296                self.dpi = d;
297            }
298            return Some(());
299        }
300        if typ == 0x4004 {
301            self.gdi = true;
302            return Some(());
303        }
304        let done = match typ {
305            0x4008 => self.object(flags, r, cv.used().saturating_add(self.clip_bytes())),
306            0x4009 | 0x4013 | 0x401A..=0x401C | 0x4036 => self.record_image_text_region(cv, typ, flags, r),
307            0x4025..=0x4035 => self.record_state(cv, typ, flags, r),
308            0x4010..=0x4012 | 0x4016..=0x4019 => self.record_curve(cv, typ, flags, r),
309            _ => self.record_shape(cv, typ, flags, r),
310        };
311        // What this player retains (objects, a continued object's pieces, the current and saved
312        // clips) shares the canvas's byte budget.
313        cv.hold(self.object_bytes.saturating_add(self.partial.as_ref().map_or(0, |p| p.data.len())).saturating_add(self.clip_bytes()));
314        done
315    }

324    fn record_shape(&mut self, cv: &mut Canvas, typ: u16, flags: u16, r: Bytes) -> Option<()> {
325        let compressed = flags & 0x4000 != 0;
326        let id = usize::from(flags & 0xFF);
327        match typ {
328            0x400A => {
329                let fill = self.brush_of(flags, r.u32(0)?)?;
330                let p = self.painter();
331                for rc in rects(r, 8, r.u32(4)? as usize, compressed)? {
332                    if cv.full() {
333                        break;
334                    }
335                    p.fill(cv, &rect_ops(rc), &fill);
336                }
337                Some(())
338            }
339            0x400B => {
340                let pen = self.pen(id)?;
341                let p = self.painter();
342                for rc in rects(r, 4, r.u32(0)? as usize, compressed)? {
343                    if cv.full() {
344                        break;
345                    }
346                    p.stroke(cv, &rect_ops(rc), pen);
347                }
348                Some(())
349            }
350            0x400C => {
351                let fill = self.brush_of(flags, r.u32(0)?)?;
352                let pts = points(r, 8, r.u32(4)? as usize, compressed)?;
353                self.painter().fill(cv, &poly_ops(&pts, true), &fill);
354                Some(())
355            }
356            0x400D => {
357                let pen = self.pen(id)?;
358                let pts = points(r, 4, r.u32(0)? as usize, compressed)?;
359                self.painter().stroke(cv, &poly_ops(&pts, flags & 0x2000 != 0), pen);
360                Some(())
361            }
362            0x400E => {
363                let fill = self.brush_of(flags, r.u32(0)?)?;

366                Some(())
367            }
368            0x400F => {
369                let pen = self.pen(id)?;
370                let rect = *rects(r, 0, 1, compressed)?.first()?;
371                self.painter().stroke(cv, &ellipse_ops(rect), pen);
372                Some(())
373            }

375                let fill = self.brush_of(flags, r.u32(0)?)?;
376                let ops = self.path(id)?;

390    /// Curve, bezier and arc records.
391    fn record_curve(&mut self, cv: &mut Canvas, typ: u16, flags: u16, r: Bytes) -> Option<()> {
392        let compressed = flags & 0x4000 != 0;
393        let id = usize::from(flags & 0xFF);
394        match typ {
395            0x4019 => {
396                let pen = self.pen(id)?;
397                let pts = points(r, 4, r.u32(0)? as usize, compressed)?;
398                let (first, rest) = pts.split_first()?;
399                let mut ops = vec![Op::M(*first)];
400                ops.extend(rest.as_chunks::<3>().0.iter().map(|c| Op::C(c[0], c[1], c[2])));
401                self.painter().stroke(cv, &ops, pen);
402                Some(())
403            }
404            // Cardinal splines: FillClosedCurve, DrawClosedCurve, DrawCurve.
405            0x4016 => {
406                let fill = self.brush_of(flags, r.u32(0)?)?;
407                let tension = finite(r.f32(4)?)?;
408                let pts = points(r, 12, r.u32(8)? as usize, compressed)?;
409                self.painter().fill(cv, &curve_ops(&pts, tension, true, 0, usize::MAX), &fill);
410                Some(())
411            }
412            0x4017 => {
413                let pen = self.pen(id)?;
414                let tension = finite(r.f32(0)?)?;
415                let pts = points(r, 8, r.u32(4)? as usize, compressed)?;
416                self.painter().stroke(cv, &curve_ops(&pts, tension, true, 0, usize::MAX), pen);
417                Some(())
418            }
419            0x4018 => {
420                let pen = self.pen(id)?;
421                let tension = finite(r.f32(0)?)?;
422                let (offset, segments) = (r.u32(4)? as usize, r.u32(8)? as usize);
423                let pts = points(r, 16, r.u32(12)? as usize, compressed)?;
424                self.painter().stroke(cv, &curve_ops(&pts, tension, false, offset, segments), pen);
425                Some(())
426            }
427            0x4010 => {
428                let fill = self.brush_of(flags, r.u32(0)?)?;
429                let [start, sweep] = floats::<2>(r, 4)?;
430                let rect = *rects(r, 12, 1, compressed)?.first()?;
431                self.painter().fill(cv, &arc_ops(rect, start, sweep, true), &fill);
432                Some(())
433            }
434            0x4011 | 0x4012 => {
435                let pen = self.pen(id)?;
436                let [start, sweep] = floats::<2>(r, 0)?;
437                let rect = *rects(r, 8, 1, compressed)?.first()?;
438                self.painter().stroke(cv, &arc_ops(rect, start, sweep, typ == 0x4011), pen);
439                Some(())
440            }
441            _ => None,
442        }

445    /// Clear, region fill, image and text records.
446    fn record_image_text_region(&mut self, cv: &mut Canvas, typ: u16, flags: u16, r: Bytes) -> Option<()> {
447        let compressed = flags & 0x4000 != 0;
448        let id = usize::from(flags & 0xFF);
449        match typ {
450            // Clear: the whole picture in one colour, whatever the clip (a transparent clear
451            // leaves nothing to draw).
452            0x4009 => {
453                let c = argb(r.u32(0)?);
454                let [x, y, w, h] = self.frame?;
455                if c[3] == 0 || !(w > 0.0 && h > 0.0) {
456                    return Some(());
457                }
458                let to_device = invert(mul(cv.world(), cv.device_affine()))?;
459                let p = Painter { xf: to_device, clip: None };
460                p.fill(cv, &rect_ops([x, y, x + w, y + h]), &Fill::Solid(c));
461                Some(())
462            }
463            // FillRegion: the region's area (as the clip reads it).
464            0x4013 => {
465                let fill = self.brush_of(flags, r.u32(0)?)?;
466                let Some(Some(Obj::Region(Some(area)))) = self.objects.get(id).cloned() else { return None };
467                if area.len() < 3 {
468                    return None;
469                }
470                self.painter().fill(cv, &poly_ops(&area, true), &fill);
471                Some(())
472            }
473            0x401A => {
474                let img = self.image(id)?;
475                let dst = *rects(r, 24, 1, compressed)?.first()?;
476                self.draw_image(cv, &img, r, [dst[0], dst[1], dst[2] - dst[0], dst[3] - dst[1]])
477            }
478            0x401B => {
479                let img = self.image(id)?;
480                let pts = points(r, 28, 3, compressed)?;
481                let (p0, p1, p2) = (*pts.first()?, *pts.get(1)?, *pts.get(2)?);
482                // The destination parallelogram's top-left, top-right and bottom-left corners; a
483                // sheared or rotated one is drawn into its bounding rectangle's axes.
484                let dst = [p0.0, p0.1, p1.0 - p0.0 + (p2.0 - p0.0), p2.1 - p0.1 + (p1.1 - p0.1)];
485                self.draw_image(cv, &img, r, dst)
486            }
487            0x401C => self.draw_string(cv, flags, r),
488            0x4036 => self.draw_driver_string(cv, flags, r),
489            _ => None,
490        }
491    }

493    /// Save / restore, transform and clip records.
494    fn record_state(&mut self, cv: &mut Canvas, typ: u16, flags: u16, r: Bytes) -> Option<()> {
495        let id = usize::from(flags & 0xFF);
496        match typ {
497            0x4025 | 0x4028 => r.u32(0).map(|i| self.push(i)),
498            0x4027 => r.u32(24).map(|i| self.push(i)),
499            0x4026 | 0x4029 => r.u32(0).map(|i| self.pop(i)),
500            0x402A => matrix(r, 0).map(|m| self.world = m),
501            0x402B => {
502                self.world = IDENTITY;
503                Some(())
504            }
505            0x402C => matrix(r, 0).and_then(|m| self.apply(m, flags)),
506            0x402D => self.apply([1.0, 0.0, 0.0, 1.0, f64::from(r.f32(0)?), f64::from(r.f32(4)?)], flags),
507            0x402E => self.apply([f64::from(r.f32(0)?), 0.0, 0.0, f64::from(r.f32(4)?), 0.0, 0.0], flags),
508            0x402F => {
509                let a = f64::from(r.f32(0)?).to_radians();
510                self.apply([a.cos(), a.sin(), -a.sin(), a.cos(), 0.0, 0.0], flags)
511            }
512            0x4030 => {
513                let page = f64::from(r.f32(0)?) * unit_px(u32::from(flags & 0xFF), self.dpi);
514                if page.is_finite() && page > 0.0 && page < 1.0e6 {
515                    self.page = page;
516                }
517                Some(())
518            }
519            0x4031 => {
520                self.clip = None;
521                Some(())
522            }
523            0x4032 => {
524                let [x, y, w, h] = floats::<4>(r, 0)?;
525                let area = map_area(Some(rect_poly([x, y, x + w, y + h])), mul(self.xf(), cv.device_affine()));
526                self.combine(flags, area)
527            }
528            0x4033 => {
529                let area = map_area(Some(area_of_ops(&self.path(id)?)), mul(self.xf(), cv.device_affine()));
530                self.combine(flags, area)
531            }
532            0x4034 => {
533                let Some(Some(Obj::Region(reg))) = self.objects.get(id).cloned() else { return None };
534                let area = map_area(reg, mul(self.xf(), cv.device_affine()));
535                self.combine(flags, area)
536            }
537            0x4035 => {
538                let [dx, dy] = floats::<2>(r, 0)?;
539                let a = mul(self.xf(), cv.device_affine());
540                let (ox, oy) = (dx * a[0] + dy * a[2], dx * a[1] + dy * a[3]);
541                if let Some(c) = self.clip.as_mut() {
542                    for p in c.iter_mut() {
543                        *p = (p.0 + ox, p.1 + oy);
544                    }
545                }
546                Some(())
547            }
548            _ => None,
549        }
550    }

565        if let Some(i) = self.stack.iter().rposition(|s| s.id == id) {
566            let s = self.stack.swap_remove(i);

572    /// Combines a new clip area with the current one under the record's combine mode: replace,
573    /// intersect and union (as the rectangle around both). XOR, exclude and complement leave the
574    /// clip as it was.
575    fn combine(&mut self, flags: u16, new: Area) -> Option<()> {
576        self.clip = match (flags >> 8) & 0xF {
577            0 => new,
578            1 => area_intersect(self.clip.take(), new),
579            2 => area_union(self.clip.take(), new),
580            _ => self.clip.take(),
581        };
582        Some(())
583    }

585    fn object(&mut self, flags: u16, r: Bytes, canvas_used: usize) -> Option<()> {
586        let id = usize::from(flags & 0xFF);
587        let kind = (flags >> 8) & 0x7F;
588        if flags & 0x8000 != 0 {
589            // A continued object: the total size, then this record's share of the bytes.
590            let total = (r.u32(0)? as usize).min(MAX_OBJECT_BYTES);
591            let chunk = r.tail(4)?.0;
592            let p = match self.partial.take() {
593                Some(p) if p.id == id && p.kind == kind && p.total == total => p,
594                _ => Partial { id, kind, total, data: Vec::new() },
595            };
596            let mut p = p;
597            let room = p.total.saturating_sub(p.data.len());
598            p.data.extend_from_slice(&chunk[..chunk.len().min(room)]);
599            if p.data.len() >= p.total {
600                let obj = parse_object(p.kind, Bytes(&p.data));
601                self.store(p.id, obj, canvas_used);
602            } else {
603                self.partial = Some(p);
604            }
605            return Some(());
606        }
607        // The last piece of a continued object comes without the continue flag.
608        if let Some(mut p) = self.partial.take()
609            && p.id == id
610            && p.kind == kind
611        {
612            let room = p.total.saturating_sub(p.data.len());
613            p.data.extend_from_slice(&r.0[..r.0.len().min(room)]);
614            let obj = parse_object(kind, Bytes(&p.data));
615            self.store(id, obj, canvas_used);
616            return Some(());
617        }
618        let obj = parse_object(kind, r);
619        self.store(id, obj, canvas_used);
620        Some(())
621    }

623    /// `canvas_used` is what the canvas and the clips already retain: objects and items share one budget.
624    fn store(&mut self, id: usize, obj: Option<Obj>, canvas_used: usize) {
625        let Some(slot) = self.objects.get_mut(id) else { return };
626        if let Some(old) = slot.take() {
627            self.object_bytes = self.object_bytes.saturating_sub(Self::obj_cost(&old));
628        }
629        if let Some(new_obj) = obj {
630            let cost = Self::obj_cost(&new_obj);
631            if self.object_bytes.saturating_add(cost).saturating_add(canvas_used) <= crate::craft::metafile::MAX_BYTES {
632                self.object_bytes += cost;
633                *slot = Some(new_obj);
634            }
635        }
636    }

645            Obj::Path(ops) => ops.len().saturating_mul(std::mem::size_of::<Op>()),

721    /// DrawString: lines of text laid out in a rectangle by the string format's alignments.
722    fn draw_string(&self, cv: &mut Canvas, flags: u16, r: Bytes) -> Option<()> {
723        let font = self.font(usize::from(flags & 0xFF))?;
724        let color = self.brush_of(flags, r.u32(0)?)?.color();
725        let fmt = match self.objects.get(r.u32(4)? as usize) {
726            Some(Some(Obj::Format(f))) => *f,
727            _ => Format::default(),
728        };
729        let len = (r.u32(8)? as usize).min(MAX_CHARS);
730        let [x, y, w, h] = floats::<4>(r, 12)?;
731        let units: Vec<u16> = (0..len).map(|i| r.u16(28 + i * 2)).collect::<Option<_>>()?;
732        let lf = self.log_font(&font);
733        let em = f64::from(-lf.font.height);
734        let wrap = w > 0.0 && fmt.flags & 0x1000 == 0;
735        let lines: Vec<Vec<u16>> = units
736            .split(|u| *u == 0x0A)
737            .flat_map(|l| {
738                let l: Vec<u16> = l.iter().copied().filter(|u| *u != 0x0D).collect();
739                if wrap { wrap_line(&l, w, em) } else { vec![l] }
740            })
741            .collect();
742        let line_h = em * 1.2;
743        let (ax, ta_h) = match fmt.align {
744            1 => (x + w / 2.0, TA_CENTER),
745            2 => (x + w, TA_RIGHT),
746            _ => (x, 0),
747        };
748        let block = line_h * lines.len() as f64;
749        let top = match fmt.line_align {
750            1 => y + (h - block) / 2.0,
751            2 => y + h - block,
752            _ => y,
753        };
754        let clip = (w > 0.0 && h > 0.0 && fmt.flags & 0x4000 == 0).then_some([x, y, x + w, y + h]);
755        let p = self.painter();
756        for (i, line) in lines.iter().enumerate() {
757            if line.is_empty() {
758                continue;
759            }
760            let (text, dx) = crate::craft::metafile::text::decode_utf16(line, &[]);
761            let at = (ax, top + line_h * i as f64);
762            let (at, ta) = if fmt.line_align == 2 && lines.len() == 1 { ((ax, y + h), ta_h | TA_BOTTOM) } else { (at, ta_h) };
763            p.text(cv, at, text, dx, TextStyle { color, font: lf.clone(), align: ta, clip }, IDENTITY);
764        }
765        Some(())
766    }

768    /// DrawDriverString with character codes (CmapLookup): positioned glyphs on a baseline.
769    fn draw_driver_string(&self, cv: &mut Canvas, flags: u16, r: Bytes) -> Option<()> {
770        let font = self.font(usize::from(flags & 0xFF))?;
771        let color = self.brush_of(flags, r.u32(0)?)?.color();
772        let opts = r.u32(4)?;
773        let n = (r.u32(12)? as usize).min(MAX_CHARS);
774        let mut units: Vec<u16> = (0..n).map(|i| r.u16(16 + i * 2)).collect::<Option<_>>()?;
775        if opts & 1 == 0 {
776            // Glyph indices into the font. The crate knows no fonts, but the classic Windows core
777            // fonts keep the standard Macintosh glyph order for ASCII, so those map back.
778            units = glyphs_to_chars(&font.family, &units)?;
779        }
780        let pos = points(r, 16 + n * 2, n, false)?;
781        let m = if r.u32(8)? != 0 { matrix(r, 16 + n * 10)? } else { IDENTITY };
782        let lf = self.log_font(&font);
783        let first = *pos.first()?;
784        let dx: Vec<f32> = if opts & 4 != 0 {
785            Vec::new()
786        } else {
787            let mut d: Vec<f32> = pos.windows(2).map(|w| (w[1].0 - w[0].0) as f32).collect();
788            d.push(d.last().copied().unwrap_or(-lf.font.height * 0.5));
789            d
790        };
791        let (text, dx) = crate::craft::metafile::text::decode_utf16(&units, &dx);
792        self.painter().text(cv, first, text, dx, TextStyle { color, font: lf, align: TA_BASELINE, clip: None }, m);
793        Some(())
794    }

816/// Draws through the canvas with EMF+'s transform, clip and its own pen or brush, then puts the
817/// canvas's GDI state back.
818struct Painter {
819    xf: Xform,
820    clip: Area,
821}
822
823impl Painter {
824    fn begin(&self, cv: &mut Canvas, pre: Xform) {
825        let world = cv.world();
826        cv.save_dc();
827        cv.modify_world(mul(mul(pre, self.xf), world), 4);
828        cv.set_clip(self.clip.clone());
829    }
830
831    fn end(&self, cv: &mut Canvas) {
832        cv.restore_dc(-1);
833        cv.set_clip(None);
834    }
835
836    fn no_pen(cv: &mut Canvas) {
837        cv.set_pen(Pen::new(PS_NULL, 0.0, [0, 0, 0, 0]));
838    }
839
840    fn fill(&self, cv: &mut Canvas, ops: &[Op], fill: &Fill) {
841        self.begin(cv, IDENTITY);
842        Self::no_pen(cv);
843        match fill {
844            Fill::Solid(c) => {
845                cv.set_brush(Brush::new(BS_SOLID, *c));
846                cv.shape(ops.to_vec(), true, false);
847            }
848            Fill::Hatch { style, fg, bg } => match emfplus_hatch_tile(*style, *fg, Some(*bg)) {
849                Some(px) => cv.tile_shape(ops, Arc::new(RawTile { w: 8, h: 8, px, cell: cv.pattern_px })),
850                // Percentage styles are tiled above. The remaining GDI+ hatch styles have no tile
851                // mapping yet, so keep the existing blended-colour fallback instead of inventing
852                // a pattern that could disagree with the file.
853                None => {
854                    cv.set_brush(Brush::new(BS_SOLID, lerp(*fg, *bg, 0.5)));
855                    cv.shape(ops.to_vec(), true, false);
856                }
857            },
858            _ => {
859                // Device point to brush space: invert brush space → world → device.
860                if let Some(inv) = invert(mul(fill.xf(), cv.device_affine())) {
861                    cv.shade_shape(ops, &|p| fill.sample(apply(inv, p)));
862                }
863            }
864        }
865        self.end(cv);
866    }
867
868    fn stroke(&self, cv: &mut Canvas, ops: &[Op], (color, width): ([u8; 4], f64)) {
869        self.begin(cv, IDENTITY);
870        cv.set_pen(Pen::new(0, width, color));
871        cv.set_brush(Brush::new(BS_NULL, [0, 0, 0, 0]));
872        cv.shape(ops.to_vec(), false, true);
873        self.end(cv);
874    }
875
876    fn image(&self, cv: &mut Canvas, dst: [f64; 4], img: Rgba) {
877        self.begin(cv, IDENTITY);
878        cv.bitmap(dst, img);
879        self.end(cv);
880    }

882    /// A nested metafile's picture stretched into world rectangle `dst` = `[x, y, w, h]`.
883    fn picture(&self, cv: &mut Canvas, [x, y, w, h]: [f64; 4], pic: &Picture) {
884        self.begin(cv, IDENTITY);
885        let (sx, sy) = (w / f64::from(pic.width), h / f64::from(pic.height));
886        let m = mul([sx, 0.0, 0.0, sy, x, y], cv.device_affine());
887        let scale = (m[0] * m[3] - m[1] * m[2]).abs().sqrt();
888        let pt = |a: f32, b: f32| apply(m, (f64::from(a), f64::from(b)));
889        let ops = |segs: &[Seg]| -> Vec<Op> {
890            segs.iter()
891                .map(|s| match *s {
892                    Seg::Move(a, b) => Op::M(pt(a, b)),
893                    Seg::Line(a, b) => Op::L(pt(a, b)),
894                    Seg::Cubic(a, b, c, d, e, f) => Op::C(pt(a, b), pt(c, d), pt(e, f)),
895                    Seg::Close => Op::Z,
896                })
897                .collect()
898        };
899        // The nested picture's own clip regions are not carried over: each item takes the
900        // canvas's current clip.
901        for item in &pic.items {
902            if cv.full() {
903                break;
904            }
905            let raw = match item {
906                Item::Path { segs, fill, stroke, line, even_odd, .. } => Raw::Path {
907                    ops: ops(segs),
908                    fill: *fill,
909                    stroke: stroke.map(|(c, w)| (c, f64::from(w) * scale)),
910                    line: LineStyle { dash: line.dash.iter().map(|d| (f64::from(*d) * scale) as f32).collect(), ..line.clone() },
911                    even_odd: *even_odd,
912                    region: RawClip::default(),
913                },
914                Item::Pattern { segs, even_odd, tile, .. } => Raw::Pattern {
915                    ops: ops(segs),
916                    even_odd: *even_odd,
917                    tile: Arc::new(RawTile {
918                        w: tile.width as usize,
919                        h: tile.height as usize,
920                        px: tile.rgba.clone(),
921                        cell: f64::from(tile.cell[0]) * scale,
922                    }),
923                    region: RawClip::default(),
924                },
925                Item::Bitmap { rect, transform, width, height, rgba, .. } => {
926                    let (a, b) = (pt(rect[0], rect[1]), pt(rect[0] + rect[2], rect[1] + rect[3]));
927                    let mut img = Rgba { w: *width as usize, h: *height as usize, px: rgba.clone() };
928                    let rect = [a.0.min(b.0), a.1.min(b.1), (b.0 - a.0).abs(), (b.1 - a.1).abs()];
929                    let turned = mul(*transform, m);
930                    let xf = if turned[1] == 0.0 && turned[2] == 0.0 {
931                        crate::craft::metafile::dib::flip(&mut img, a.0 > b.0, a.1 > b.1);
932                        [rect[2], 0.0, 0.0, rect[3], rect[0], rect[1]]
933                    } else {
934                        turned
935                    };
936                    Raw::Bitmap { rect, xf, img, region: RawClip::default() }
937                }
938                Item::Text(t) => {
939                    let mut t = t.clone();
940                    t.transform = mul(t.transform, m);
941                    if let Some([x0, y0, x1, y1]) = t.clip {
942                        let (p, q) = (pt(x0, y0), pt(x1, y1));
943                        t.clip = Some([p.0.min(q.0) as f32, p.1.min(q.1) as f32, p.0.max(q.0) as f32, p.1.max(q.1) as f32]);
944                    }
945                    Raw::Text { text: t, region: RawClip::default() }
946                }
947            };
948            cv.push_device(raw);
949        }
950        self.end(cv);
951    }

953    fn text(&self, cv: &mut Canvas, at: P, text: String, dx: Vec<f32>, style: TextStyle, pre: Xform) {
954        self.begin(cv, pre);
955        cv.set_text_style(style.color, style.font, style.align);
956        let (opts, rect) = match style.clip {
957            Some(c) => (ETO_CLIPPED, Some(c)),
958            None => (0, None),
959        };
960        cv.text(at, text, dx, opts, rect);
961        self.end(cv);
962    }

965fn apply(m: Xform, p: P) -> P {
966    (p.0 * m[0] + p.1 * m[2] + m[4], p.0 * m[1] + p.1 * m[3] + m[5])
967}

969fn invert(m: Xform) -> Option<Xform> {
970    let det = m[0] * m[3] - m[1] * m[2];
971    if !det.is_finite() || det.abs() < 1e-12 {
972        return None;
973    }
974    let (a, b, c, d) = (m[3] / det, -m[1] / det, -m[2] / det, m[0] / det);
975    Some([a, b, c, d, -(m[4] * a + m[5] * c), -(m[4] * b + m[5] * d)])
976}

1008/// A brush object at `off`: version, type, then the type's data.
1009fn brush(r: Bytes, off: usize) -> Option<Fill> {
1010    let d = r.tail(off + 8)?;
1011    match r.u32(off + 4)? {
1012        0 => Some(Fill::Solid(argb(d.u32(0)?))),
1013        1 => Some(Fill::Hatch { style: d.u32(0)?, fg: argb(d.u32(4)?), bg: argb(d.u32(8)?) }),
1014        2 => {
1015            let bflags = d.u32(0)?;
1016            let wrap = d.u32(4)?;
1017            let (xf, off) = if bflags & 0x2 != 0 { (matrix(d, 8)?, 32) } else { (IDENTITY, 8) };
1018            let ImageData::Bitmap(img) = image(d.tail(off)?)? else { return None };
1019            Some(Fill::Texture { img: Arc::new(img), wrap, xf })
1020        }
1021        3 => path_gradient(d),
1022        4 => linear_gradient(d),
1023        _ => None,
1024    }
1025}

1027fn linear_gradient(d: Bytes) -> Option<Fill> {
1028    let bflags = d.u32(0)?;
1029    let wrap = d.u32(4)?;
1030    let rect = floats::<4>(d, 8)?;
1031    let (c0, c1) = (argb(d.u32(24)?), argb(d.u32(28)?));
1032    let mut off = 40;
1033    let mut xf = IDENTITY;
1034    if bflags & 0x2 != 0 {
1035        xf = matrix(d, off)?;
1036        off += 24;
1037    }
1038    let mut stops = vec![(0.0, c0), (1.0, c1)];
1039    if bflags & 0x4 != 0 {
1040        // Preset colours: positions, then one colour each.
1041        let n = (d.u32(off)? as usize).min(MAX_STOPS);
1042        let pos: Vec<f64> = (0..n).map(|i| d.f32(off + 4 + i * 4).map(f64::from)).collect::<Option<_>>()?;
1043        let cols: Vec<[u8; 4]> = (0..n).map(|i| d.u32(off + 4 + n * 4 + i * 4).map(argb)).collect::<Option<_>>()?;
1044        stops = pos.into_iter().zip(cols).collect();
1045    } else if bflags & 0x8 != 0 {
1046        // Blend factors: positions, then how far from the start colour to the end colour.
1047        let n = (d.u32(off)? as usize).min(MAX_STOPS);
1048        let pos: Vec<f64> = (0..n).map(|i| d.f32(off + 4 + i * 4).map(f64::from)).collect::<Option<_>>()?;
1049        let fac: Vec<f64> = (0..n).map(|i| d.f32(off + 4 + n * 4 + i * 4).map(f64::from)).collect::<Option<_>>()?;
1050        stops = pos.into_iter().zip(fac).map(|(p, k)| (p, lerp(c0, c1, k.clamp(0.0, 1.0)))).collect();
1051    }
1052    stops.retain(|s| s.0.is_finite());
1053    stops.sort_by(|a, b| a.0.total_cmp(&b.0));
1054    (!stops.is_empty()).then_some(Fill::Linear { rect, stops, wrap, xf })
1055}

1057/// A path gradient: from the centre colour to the (first) surrounding colour at the boundary,
1058/// taken as the ellipse in the boundary's bounding rectangle.
1059fn path_gradient(d: Bytes) -> Option<Fill> {
1060    let bflags = d.u32(0)?;
1061    let center_c = argb(d.u32(8)?);
1062    let [cx, cy] = floats::<2>(d, 12)?;
1063    let nc = (d.u32(20)? as usize).min(MAX_STOPS);
1064    let surround = if nc > 0 { argb(d.u32(24)?) } else { center_c };
1065    let mut off = 24 + nc * 4;
1066    let bbox = if bflags & 0x1 != 0 {
1067        let size = d.u32(off)? as usize;
1068        let b = ops_bbox(&path(d.tail(off + 4)?)?)?;
1069        off = off.checked_add(4 + size)?;
1070        b
1071    } else {
1072        let n = d.u32(off)? as usize;
1073        let pts = points(d, off + 4, n, false)?;
1074        off = off.checked_add(4 + 8 * n.min(MAX_POINTS))?;
1075        ops_bbox(&poly_ops(&pts, true))?
1076    };
1077    let xf = if bflags & 0x2 != 0 { matrix(d, off)? } else { IDENTITY };
1078    let radius = ((bbox[2] - bbox[0]) / 2.0, (bbox[3] - bbox[1]) / 2.0);
1079    Some(Fill::Radial { center: (cx, cy), radius, stops: vec![(0.0, center_c), (1.0, surround)], xf })
1080}

1106/// A path object: points, then one type byte per point (start, line, Bézier; 0x80 closes).
1107fn path(r: Bytes) -> Option<Vec<Op>> {
1108    let n = (r.u32(4)? as usize).min(MAX_POINTS);
1109    let pflags = r.u32(8)?;
1110    if pflags & 0x1800 != 0 {
1111        // Relative (0x0800) and run-length encoded (0x1000) points: not read yet.
1112        return None;
1113    }
1114    let compressed = pflags & 0x4000 != 0;
1115    let pts = points(r, 12, n, compressed)?;
1116    let toff = 12 + n * if compressed { 4 } else { 8 };
1117    let types = r.slice(toff, n)?;
1118    let mut ops = Vec::with_capacity(n + 4);
1119    let mut i = 0;
1120    while i < n {
1121        let (t, p) = (types[i], pts[i]);
1122        match t & 0x07 {
1123            0 => ops.push(Op::M(p)),
1124            3 if i + 2 < n => {
1125                ops.push(Op::C(p, pts[i + 1], pts[i + 2]));
1126                if types[i + 2] & 0x80 != 0 {
1127                    ops.push(Op::Z);
1128                }
1129                i += 3;
1130                continue;
1131            }
1132            _ => ops.push(Op::L(p)),
1133        }
1134        if t & 0x80 != 0 {
1135            ops.push(Op::Z);
1136        }
1137        i += 1;
1138    }
1139    Some(ops)
1140}

1149/// One region node at `off`: its area and its length in bytes.
1150fn region_node(r: Bytes, off: usize, depth: usize, work: &mut usize) -> Option<(Area, usize)> {
1151    *work = work.checked_add(1)?;
1152    if depth > MAX_REGION_DEPTH || *work > MAX_REGION_WORK {
1153        return None;
1154    }
1155    match r.u32(off)? {
1156        0x1000_0000 => {
1157            let [x, y, w, h] = floats::<4>(r, off + 4)?;
1158            Some((Some(rect_poly([x.min(x + w), y.min(y + h), x.max(x + w), y.max(y + h)])), 20))
1159        }
1160        0x1000_0001 => {
1161            let size = r.u32(off + 4)? as usize;
1162            if size < 12 {
1163                return None;
1164            }
1165            let data = Bytes(r.slice(off + 8, size)?);
1166            let points = (data.u32(4)? as usize).min(MAX_POINTS);
1167            *work = work.checked_add(points)?;
1168            if *work > MAX_REGION_WORK {
1169                return None;
1170            }
1171            let area = area_of_ops(&path(data)?);
1172            Some((Some(area), 8usize.checked_add(size)?))
1173        }
1174        0x1000_0002 => Some((Some(Vec::new()), 4)),
1175        0x1000_0003 => Some((None, 4)),
1176        mode @ 0..=5 => {
1177            let (a, la) = region_node(r, off + 4, depth + 1, work)?;
1178            let (b, lb) = region_node(r, off.checked_add(4 + la)?, depth + 1, work)?;
1179            let out = match mode {
1180                0 => b,
1181                1 => area_intersect(a, b),
1182                2 => area_union(a, b),
1183                // XOR, exclude, complement: the left area bounds the result closely enough.
1184                _ => a,
1185            };
1186            Some((out, (4 + la).checked_add(lb)?))
1187        }
1188        _ => None,
1189    }
1190}

1192/// The area a path encloses: its outline when that is one convex figure, else the rectangle
1193/// around it.
1194fn area_of_ops(ops: &[Op]) -> Vec<P> {
1195    let subs = flatten(ops);
1196    if let [(pts, _)] = subs.as_slice()
1197        && let Some(c) = convex(pts)
1198    {
1199        return c;
1200    }
1201    ops_bbox(ops).map(rect_poly).unwrap_or_default()
1202}

1204fn area_intersect(a: Area, b: Area) -> Area {
1205    match (a, b) {
1206        (None, x) | (x, None) => x,
1207        (Some(a), Some(b)) if a.len() >= 3 && b.len() >= 3 => {
1208            // Out of work: keep the area so far rather than call the region empty.
1209            match clip_polygon_within(&a, &b, &mut { MAX_AREA_CLIP_WORK }) {
1210                Some(p) => Some(convex(&p).unwrap_or_default()),
1211                None => Some(a),
1212            }
1213        }
1214        _ => Some(Vec::new()),
1215    }
1216}
1217
1218fn area_union(a: Area, b: Area) -> Area {
1219    match (a, b) {
1220        (Some(a), Some(b)) => {
1221            if a.len() < 3 {
1222                return Some(b);
1223            }
1224            if b.len() < 3 {
1225                return Some(a);
1226            }
1227            let all: Vec<P> = a.into_iter().chain(b).collect();
1228            Some(pts_bbox(&all).map(rect_poly).unwrap_or_default())
1229        }
1230        _ => None,
1231    }
1232}

1242/// An elliptical arc in `rect` from `start` sweeping `sweep` degrees (clockwise on screen), as
1243/// cubics; a pie adds the lines to and from the centre.
1244fn arc_ops([l, t, r, b]: [f64; 4], start: f64, sweep: f64, pie: bool) -> Vec<Op> {
1245    let (cx, cy, rx, ry) = ((l + r) / 2.0, (t + b) / 2.0, (r - l) / 2.0, (b - t) / 2.0);
1246    let sweep = sweep.clamp(-360.0, 360.0);
1247    // GDI+ angles are along rays from the centre; turn them into the ellipse's parameter.
1248    let param = |deg: f64| {
1249        let a = deg.to_radians();
1250        (rx * a.sin()).atan2(ry * a.cos())
1251    };
1252    let t0 = param(start);
1253    let mut dt = if sweep.abs() >= 360.0 { sweep.signum() * std::f64::consts::TAU } else { param(start + sweep) - t0 };
1254    if sweep.abs() < 360.0 {
1255        while sweep > 0.0 && dt < 0.0 {
1256            dt += std::f64::consts::TAU;
1257        }
1258        while sweep < 0.0 && dt > 0.0 {
1259            dt -= std::f64::consts::TAU;
1260        }
1261    }
1262    let at = |t: f64| (cx + rx * t.cos(), cy + ry * t.sin());
1263    let mut ops = Vec::with_capacity(8);
1264    if pie {
1265        ops.push(Op::M((cx, cy)));
1266        ops.push(Op::L(at(t0)));
1267    } else {
1268        ops.push(Op::M(at(t0)));
1269    }
1270    let n = ((dt.abs() / std::f64::consts::FRAC_PI_2).ceil() as usize).clamp(1, 4);
1271    let step = dt / n as f64;
1272    let k = 4.0 / 3.0 * (step / 4.0).tan();
1273    for i in 0..n {
1274        let (a, e) = (t0 + step * i as f64, t0 + step * (i + 1) as f64);
1275        let (p0, p3) = (at(a), at(e));
1276        let c1 = (p0.0 - k * rx * a.sin(), p0.1 + k * ry * a.cos());
1277        let c2 = (p3.0 + k * rx * e.sin(), p3.1 - k * ry * e.cos());
1278        ops.push(Op::C(c1, c2, p3));
1279    }
1280    if pie {
1281        ops.push(Op::Z);
1282    }
1283    ops
1284}

1319/// Bytes the items of a nested picture own: path segments, dashes, clip rectangles, pixels and
1320/// text, with each shared pattern tile and clip path counted once.
1321fn picture_cost(items: &[Item]) -> usize {
1322    let segs = |s: &Vec<Seg>| s.len().saturating_mul(std::mem::size_of::<Seg>());
1323    let region = |r: &Option<Vec<[f32; 4]>>| r.as_ref().map_or(0, |v| v.len().saturating_mul(16));
1324    let mut tiles = std::collections::HashSet::new();
1325    let mut clip_paths = std::collections::HashSet::new();
1326    let mut total = 0usize;
1327    for item in items {
1328        let (own, r, clips) = match item {
1329            Item::Path { segs: s, line, region: r, clips, .. } => (segs(s).saturating_add(line.dash.len().saturating_mul(4)), r, clips),
1330            Item::Bitmap { rgba, region: r, clips, .. } => (rgba.len(), r, clips),
1331            Item::Pattern { segs: s, tile, region: r, clips, .. } => {
1332                let tile_bytes = if tiles.insert(Arc::as_ptr(tile)) { tile.rgba.len() } else { 0 };
1333                (segs(s).saturating_add(tile_bytes), r, clips)
1334            }
1335            Item::Text(t) => {
1336                let own = t.text.len().saturating_add(t.dx.len().saturating_mul(std::mem::size_of::<f32>())).saturating_add(t.font.family.len());
1337                (own, &t.region, &t.clips)
1338            }
1339        };
1340        let clip_bytes = clips.iter().filter(|c| clip_paths.insert(Arc::as_ptr(c))).map(|c| segs(&c.segs)).fold(0usize, usize::saturating_add);
1341        total = total.saturating_add(128).saturating_add(own).saturating_add(region(r)).saturating_add(clip_bytes);
1342    }
1343    total
1344}

1346fn bitmap(r: Bytes) -> Option<Rgba> {
1347    let (w, h, stride) = (r.u32(8)? as i32, r.u32(12)? as i32, r.u32(16)? as i32);
1348    let format = r.u32(20)?;
1349    let data = r.tail(28)?;
1350    if r.u32(24)? == 1 {
1351        return crate::craft::metafile::dib::decode_compressed(data.0);
1352    }
1353    let (w, h) = (usize::try_from(w).ok()?, usize::try_from(h).ok()?);
1354    if w == 0 || h == 0 || w.checked_mul(h)? > crate::craft::metafile::MAX_PIXELS {
1355        return None;
1356    }
1357    let bpp = ((format >> 8) & 0xFF) as usize;
1358    let (pal, pixels) = if format & 0x0001_0000 != 0 {
1359        // Indexed formats carry a palette first: flags, count, ARGB entries.
1360        let n = (data.u32(4)? as usize).min(256);
1361        let pal: Vec<[u8; 4]> = (0..n).map(|i| data.u32(8 + i * 4).map(argb)).collect::<Option<_>>()?;
1362        (pal, data.tail(8 + n * 4)?)
1363    } else {
1364        (Vec::new(), data)
1365    };
1366    let row = (w.checked_mul(bpp)?).div_ceil(8);
1367    let stride_abs = (stride.unsigned_abs() as usize).max(row);
1368    let mut px = Vec::with_capacity(w * h * 4);
1369    for y in 0..h {
1370        let line = pixels.slice(y.checked_mul(stride_abs)?, row)?;
1371        for x in 0..w {
1372            let c = match (format, bpp) {
1373                (_, 1 | 4 | 8) if !pal.is_empty() => {
1374                    let bit = x * bpp;
1375                    let byte = *line.get(bit / 8)?;
1376                    let i = (byte >> (8 - bpp - bit % 8)) & ((1u16 << bpp) - 1) as u8;
1377                    pal.get(usize::from(i)).copied().unwrap_or([0, 0, 0, 255])
1378                }
1379                (_, 24) => {
1380                    let p = line.get(x * 3..x * 3 + 3)?;
1381                    [p[2], p[1], p[0], 255]
1382                }
1383                (0x0002_2009, 32) => {
1384                    let p = line.get(x * 4..x * 4 + 4)?;
1385                    [p[2], p[1], p[0], 255]
1386                }
1387                (0x000E_200B, 32) => {
1388                    let p = line.get(x * 4..x * 4 + 4)?;
1389                    let a = u32::from(p[3]);
1390                    let un = |v: u8| (u32::from(v) * 255 + a / 2).checked_div(a).map_or(v, |c| c.min(255) as u8);
1391                    [un(p[2]), un(p[1]), un(p[0]), p[3]]
1392                }
1393                (_, 32) => {
1394                    let p = line.get(x * 4..x * 4 + 4)?;
1395                    [p[2], p[1], p[0], p[3]]
1396                }
1397                (0x0002_1006, 16) => {
1398                    let v = u16::from_le_bytes([*line.get(x * 2)?, *line.get(x * 2 + 1)?]);
1399                    let s = |v: u16, bits: u32| ((u32::from(v) * 255) / ((1 << bits) - 1)) as u8;
1400                    [s(v >> 11, 5), s((v >> 5) & 0x3F, 6), s(v & 0x1F, 5), 255]
1401                }
1402                (_, 16) => {
1403                    let v = u16::from_le_bytes([*line.get(x * 2)?, *line.get(x * 2 + 1)?]);
1404                    let s = |v: u16| ((u32::from(v & 0x1F) * 255) / 31) as u8;
1405                    [s(v >> 10), s(v >> 5), s(v), 255]
1406                }
1407                _ => return None,
1408            };
1409            px.extend_from_slice(&c);
1410        }
1411    }
1412    // A negative stride stores the rows bottom-up.
1413    let mut img = Rgba { w, h, px };
1414    if stride < 0 {
1415        crate::craft::metafile::dib::flip(&mut img, false, true);
1416    }
1417    Some(img)
1418}

1452/// A GDI+ cardinal spline through `pts` as cubic Béziers: each segment's control points lie a
1453/// third of `tension` (GDI+'s default 0.5) along the neighbouring chords. `closed` joins the
1454/// last point back to the first; an open curve draws `segments` segments from point `offset`.
1455fn curve_ops(pts: &[P], tension: f64, closed: bool, offset: usize, segments: usize) -> Vec<Op> {
1456    let n = pts.len();
1457    if n < 2 {
1458        return Vec::new();
1459    }
1460    let k = tension.clamp(0.0, 10.0) / 3.0;
1461    let at = |i: isize| -> P {
1462        let i = if closed { i.rem_euclid(n as isize) as usize } else { i.clamp(0, n as isize - 1) as usize };
1463        pts.get(i).copied().unwrap_or((0.0, 0.0))
1464    };
1465    let (first, count) = if closed { (0, n) } else { (offset.min(n - 1), segments.min(n - 1 - offset.min(n - 1))) };
1466    let mut ops = vec![Op::M(at(first as isize))];
1467    for s in first..first + count {
1468        let i = s as isize;
1469        let (p0, p1, p2, p3) = (at(i - 1), at(i), at(i + 1), at(i + 2));
1470        let c1 = (p1.0 + (p2.0 - p0.0) * k, p1.1 + (p2.1 - p0.1) * k);
1471        let c2 = (p2.0 - (p3.0 - p1.0) * k, p2.1 - (p3.1 - p1.1) * k);
1472        ops.push(Op::C(c1, c2, p2));
1473    }

1488/// An ellipse in its bounding box, as four cubic arcs.
1489fn ellipse_ops([l, t, r, b]: [f64; 4]) -> Vec<Op> {
1490    let (cx, cy, rx, ry) = ((l + r) / 2.0, (t + b) / 2.0, (r - l) / 2.0, (b - t) / 2.0);
1491    let k = 0.552_284_75;
1492    let (ox, oy) = (rx * k, ry * k);

1494        Op::M((cx + rx, cy)),
1495        Op::C((cx + rx, cy + oy), (cx + ox, cy + ry), (cx, cy + ry)),
1496        Op::C((cx - ox, cy + ry), (cx - rx, cy + oy), (cx - rx, cy)),
1497        Op::C((cx - rx, cy - oy), (cx - ox, cy - ry), (cx, cy - ry)),
1498        Op::C((cx + ox, cy - ry), (cx + rx, cy - oy), (cx + rx, cy)),

1518/// A rough advance for a character in ems: the metafile crate has no fonts, so line breaks for
1519/// wrapped strings use typical proportional widths.
1520fn char_em(u: u16) -> f64 {
1521    match char::from_u32(u32::from(u)).unwrap_or('x') {
1522        ' ' | 'i' | 'l' | 'j' | '.' | ',' | ':' | ';' | '\'' | '!' | '|' | 'I' | 'f' | 't' | 'r' => 0.3,
1523        'm' | 'w' | 'M' | 'W' => 0.85,
1524        c if c.is_ascii_uppercase() || c.is_ascii_digit() => 0.62,
1525        c if (c as u32) >= 0x2E80 => 1.0,
1526        _ => 0.52,
1527    }
1528}

1530/// Greedy word wrap of one line to `width` world units at em size `em`; a word longer than the
1531/// line breaks between characters.
1532fn wrap_line(line: &[u16], width: f64, em: f64) -> Vec<Vec<u16>> {
1533    if em <= 0.0 || !width.is_finite() {
1534        return vec![line.to_vec()];
1535    }
1536    let trimmed = |v: &[u16]| v.iter().rposition(|u| *u != 0x20).map_or(0, |i| i + 1);
1537    let measure = |v: &[u16]| v.iter().map(|u| char_em(*u) * em).sum::<f64>();
1538
1539    let mut out: Vec<Vec<u16>> = Vec::new();
1540    let mut cur: Vec<u16> = Vec::new();
1541    let mut cur_total_w = 0.0;
1542
1543    for word in line.split_inclusive(|u| *u == 0x20) {
1544        let n_non_space = trimmed(word);
1545        let word_non_space_w = measure(&word[..n_non_space]);
1546        let word_total_w = measure(word);
1547
1548        if cur.is_empty() {
1549            if word_non_space_w <= width {
1550                cur.extend_from_slice(word);
1551                cur_total_w = word_total_w;
1552                continue;
1553            }
1554        } else if cur_total_w + word_non_space_w <= width {
1555            cur.extend_from_slice(word);
1556            cur_total_w += word_total_w;
1557            continue;
1558        }
1559
1560        if !cur.is_empty() {
1561            let n = trimmed(&cur);
1562            cur.truncate(n);
1563            out.push(std::mem::take(&mut cur));
1564        }
1565
1566        if word_non_space_w <= width {
1567            cur.extend_from_slice(word);
1568            cur_total_w = word_total_w;
1569            continue;
1570        }
1571
1572        // A word wider than the line on its own: break it between characters.
1573        let mut cur_w = 0.0;
1574        for &u in word {
1575            let u_w = char_em(u) * em;
1576            if cur.is_empty() {
1577                cur.push(u);
1578                cur_w = u_w;
1579            } else if cur_w + u_w <= width {
1580                cur.push(u);
1581                cur_w += u_w;
1582            } else {
1583                let n = trimmed(&cur);
1584                cur.truncate(n);
1585                out.push(std::mem::take(&mut cur));
1586                cur.push(u);
1587                cur_w = u_w;
1588            }
1589        }
1590        cur_total_w = cur_w;
1591    }
1592    let n = cur.iter().rposition(|u| *u != 0x20).map_or(0, |i| i + 1);
1593    cur.truncate(n);
1594    out.push(cur);
1595    out
1596}
