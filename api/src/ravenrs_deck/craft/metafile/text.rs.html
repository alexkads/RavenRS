1//! Text records: the font (LOGFONT), the text string decoded from its character set, and the
2//! neutral [`Text`] item. The metafile crate knows no fonts, so measuring, alignment against the
3//! font's metrics and glyph drawing are left to the renderer.
4
5/// A font chosen by a CREATEFONTINDIRECT record (MS-WMF 2.2.1.2 Font, MS-EMF 2.2.13 LogFont).
6#[derive(Debug, Clone, PartialEq, Default)]
7pub struct Font {
8    /// The face name the file asks for (`"Arial"`, `"Wingdings"`); may be empty.
9    pub family: String,
10    /// LOGFONT height in text units: negative is the character (em) height, positive the cell
11    /// height (ascent plus descent), 0 the default size.
12    pub height: f32,
13    /// Average character width in text units; 0 keeps the font's own proportions.
14    pub width: f32,
15    /// 100..=900 (400 regular, 700 bold); 0 is the default weight.
16    pub weight: u16,
17    pub italic: bool,
18    pub underline: bool,
19    pub strike: bool,
20    /// LOGFONT character set (0 ANSI, 2 SYMBOL, 238 EASTEUROPE…).
21    pub charset: u8,
22}

48/// One string drawn by EXTTEXTOUT, TEXTOUT or SMALLTEXTOUT.
49#[derive(Debug, Clone, PartialEq)]
50pub struct Text {
51    /// Text space to picture space as affine coefficients `[a, b, c, d, e, f]` (`x' = a·x + c·y + e`,
52    /// `y' = b·x + d·y + f`, kurbo's order). Text space has x along the baseline and y down, in the
53    /// metafile's logical units, with the reference point at the origin. Escapement and the
54    /// mapping are folded in; glyphs are never mirrored.
55    pub transform: [f64; 6],
56    pub text: String,
57    /// Advance after each character of `text`, in text units (the dx array). Empty when the file
58    /// gives none: the font's own advances apply.
59    pub dx: Vec<f32>,
60    pub font: Font,
61    pub color: [u8; 4],
62    /// Colour filled behind the character cells (OPAQUE background mode); None when transparent.
63    pub background: Option<[u8; 4]>,
64    pub h_align: HAlign,
65    pub v_align: VAlign,
66    /// Clip rectangle `[x0, y0, x1, y1]` in picture units (ETO_CLIPPED).
67    pub clip: Option<[f32; 4]>,
68    /// Clip region from the device context, in picture units. None is the default clip. Drawn
69    /// together with [`Text::clip`] when both are set.
70    pub region: Option<Vec<[f32; 4]>>,
71    /// Clip paths from the device context, as on [`crate::craft::metafile::Item::Path`].
72    pub clips: Vec<std::sync::Arc<crate::craft::metafile::ClipPath>>,
73}

87/// Horizontal and vertical alignment from a TA_* word (the vertical bits are 0 top, 8 bottom,
88/// 24 baseline; the horizontal ones 0 left, 2 right, 6 center).
89pub(crate) fn align(ta: u32) -> (HAlign, VAlign) {
90    let h = match ta & 0x06 {
91        6 => HAlign::Center,
92        2 => HAlign::Right,
93        _ => HAlign::Left,
94    };
95    let v = match ta & 0x18 {
96        0x18 => VAlign::Baseline,
97        0x08 => VAlign::Bottom,
98        _ => VAlign::Top,
99    };
100    (h, v)
101}

103/// Decodes 8-bit text in `charset` and merges `dx` to match, one entry per character. In a
104/// double-byte character set (Shift_JIS, UHC, GBK, Big5) a lead and trail byte are one character
105/// whose advance is the sum of both; a lead byte without a trail byte it pairs with is one
106/// undefined byte (U+FFFD). Johab is not decoded: its pairs become blanks. `dx` may be empty.
107pub(crate) fn decode_bytes(bytes: &[u8], charset: u8, dx: &[f32]) -> (String, Vec<f32>) {
108    let mut s = String::with_capacity(bytes.len());
109    let mut out_dx = Vec::with_capacity(dx.len().min(bytes.len()));
110    let mut i = 0;
111    while let Some(&b) = bytes.get(i) {
112        let pair = bytes.get(i + 1).and_then(|t| pair_char(b, *t, charset));
113        let n = if pair.is_some() { 2 } else { 1 };
114        s.push(pair.unwrap_or_else(|| byte_char(b, charset)));
115        if !dx.is_empty() {
116            out_dx.push((i..i + n).filter_map(|k| dx.get(k)).sum());
117        }
118        i += n;
119    }
120    (s, out_dx)
121}

123/// The character set 8-bit text in `font` is decoded in. DEFAULT_CHARSET leaves the choice to the
124/// font, and the Symbol font's bytes are its own codes.
125pub(crate) fn text_charset(font: &Font) -> u8 {
126    let symbol = ["Symbol", "Symbol (PCL6)"].iter().any(|n| font.family.eq_ignore_ascii_case(n));
127    if font.charset == DEFAULT_CHARSET && symbol { SYMBOL_CHARSET } else { font.charset }
128}

130/// The encoding of a character set this module has no table of its own for.
131fn encoding(charset: u8) -> Option<&'static encoding_rs::Encoding> {
132    Some(match charset {
133        // SHIFTJIS_CHARSET, HANGUL_CHARSET, GB2312_CHARSET, CHINESEBIG5_CHARSET.
134        128 => encoding_rs::SHIFT_JIS,
135        129 => encoding_rs::EUC_KR,
136        134 => encoding_rs::GBK,
137        136 => encoding_rs::BIG5,
138        // GREEK, TURKISH, VIETNAMESE, HEBREW, ARABIC, BALTIC and THAI_CHARSET.
139        161 => encoding_rs::WINDOWS_1253,
140        162 => encoding_rs::WINDOWS_1254,
141        163 => encoding_rs::WINDOWS_1258,
142        177 => encoding_rs::WINDOWS_1255,
143        178 => encoding_rs::WINDOWS_1256,
144        186 => encoding_rs::WINDOWS_1257,
145        222 => encoding_rs::WINDOWS_874,
146        _ => return None,
147    })
148}

150/// The one character that `enc` decodes `bytes` to; None when they are not exactly one.
151fn one_char(enc: &'static encoding_rs::Encoding, bytes: &[u8]) -> Option<char> {
152    let (text, bad) = enc.decode_without_bom_handling(bytes);
153    let mut chars = text.chars();
154    let c = chars.next()?;
155    (!bad && chars.next().is_none()).then_some(c)
156}

158/// The character of a lead and trail byte in a double-byte character set; None when the bytes
159/// are not a pair there. The ranges differ: Shift_JIS leaves its half-width katakana (0xA1..=0xDF)
160/// as single bytes, and GBK has no trail byte 0x7F.
161fn pair_char(lead: u8, trail: u8, charset: u8) -> Option<char> {
162    let (is_lead, is_trail) = match charset {
163        128 => (matches!(lead, 0x81..=0x9F | 0xE0..=0xFC), matches!(trail, 0x40..=0x7E | 0x80..=0xFC)),
164        129 => (matches!(lead, 0x81..=0xFE), matches!(trail, 0x41..=0xFE)),
165        134 => (matches!(lead, 0x81..=0xFE), matches!(trail, 0x40..=0x7E | 0x80..=0xFE)),
166        136 => (matches!(lead, 0x81..=0xFE), matches!(trail, 0x40..=0x7E | 0xA1..=0xFE)),
167        // JOHAB_CHARSET: not decoded.
168        130 => return (lead >= 0x81).then_some(' '),
169        _ => return None,
170    };
171    if !(is_lead && is_trail) {
172        return None;
173    }
174    one_char(encoding(charset)?, &[lead, trail])
175}

177/// UTF-16 text to a string with one dx entry per character: a surrogate pair takes the sum of
178/// its two advances. Unpaired surrogates become U+FFFD.
179pub(crate) fn decode_utf16(units: &[u16], dx: &[f32]) -> (String, Vec<f32>) {
180    let mut s = String::with_capacity(units.len());
181    let mut out_dx = Vec::with_capacity(dx.len().min(units.len()));
182    let mut i = 0;
183    for r in char::decode_utf16(units.iter().copied()) {
184        let c = r.unwrap_or(char::REPLACEMENT_CHARACTER);
185        let n = c.len_utf16();
186        s.push(c);
187        if !dx.is_empty() {
188            out_dx.push((i..i + n).filter_map(|k| dx.get(k)).sum());
189        }
190        i += n;
191    }
192    (s, out_dx)
193}

195/// One byte that is not part of a double-byte pair, in its Windows code page. Symbol fonts keep their codes in the private-use
196/// range (U+F000 + code), where symbol fonts map them.
197fn byte_char(b: u8, charset: u8) -> char {
198    if b < 0x80 && charset != SYMBOL_CHARSET {
199        return char::from(b);
200    }
201    let cp = match charset {
202        SYMBOL_CHARSET => 0xF000 | u32::from(b),
203        238 => high(b, &CP1250),
204        204 => high(b, &CP1251),
205        _ => match encoding(charset) {
206            // An undefined byte, and a lead byte with no trail byte, are U+FFFD.
207            Some(enc) => return one_char(enc, &[b]).unwrap_or(char::REPLACEMENT_CHARACTER),
208            None => high(b, &CP1252),
209        },
210    };
211    char::from_u32(cp).unwrap_or(char::REPLACEMENT_CHARACTER)
212}

214/// The code point of byte `b` (0x80..=0xFF) from a table of the upper half; 0 entries are
215/// undefined bytes, drawn as U+FFFD.
216fn high(b: u8, table: &[u16; 128]) -> u32 {
217    match table.get(usize::from(b.wrapping_sub(0x80))) {
218        Some(0) | None => 0xFFFD,
219        Some(v) => u32::from(*v),
220    }
221}
