1//! EMF and WMF writers (feature `write`).
2//!
3//! The writers take a list of drawing operations ([`Op`]) in paint order, in points with y down,
4//! and the region of that space the picture covers: its top-left corner is the picture's origin
5//! and its size the picture's size. What an application's document holds beyond these operations
6//! (gradients, patterns, type, effects, transparency of paths) is turned into them by the
7//! application.
8//!
9//! - **EMF** ([`emf`]): paths keep their Bézier curves; fills are solid brushes; strokes are
10//!   geometric pens with their width, cap, join, miter limit and dashes; clips are path clips
11//!   inside a saved device context; images keep their pixels and their transparency and may be
12//!   turned or skewed. The header's frame is the region in 0.01 mm.
13//! - **WMF** ([`wmf`]): the same operations with curves flattened into polygons and 16-bit
14//!   coordinates, behind a placeable header that gives the size. WMF has no clip paths, no
15//!   transparency and no turned images; [`Wmf`] says which of these the picture lost.
16//!
17//! The reader reads back what the writers write: the tests round-trip both formats through
18//! [`crate::craft::metafile::parse`].

30/// Most line segments one curve is flattened into, whatever its size and the tolerance.
31const MAX_FLATTEN_PER_CURVE: f64 = 4096.0;
32/// A curve whose control polygon is longer than this, or has a point further out, is written as
33/// a line: kurbo squares and cubes coordinates when it sizes its work, which overflows there.
34const MAX_FLATTEN_LENGTH: f64 = 1e60;
35/// Most dashes one stroke is cut into when its pen cannot draw them (the rest of the path is
36/// then left solid, so a tiny pattern on a long path cannot make an endless file).
37const MAX_DASHES: usize = 100_000;
38
39/// One drawing operation.
40#[derive(Clone, Debug)]
41pub enum Op {
42    /// A path filled with one colour.
43    Fill {
44        path: BezPath,
45        even_odd: bool,
46        rgb: [u8; 3],
47    },
48    /// A path stroked with a pen.
49    Stroke {
50        path: BezPath,
51        pen: Pen,
52    },
53    /// An image: `xf` maps its pixel space (`0..width`, `0..height`) into the picture space; its
54    /// alpha is multiplied by `opacity`.
55    Image {
56        image: Arc<Image>,
57        xf: Affine,
58        opacity: f32,
59    },
60    /// What follows, up to the matching [`Op::Unclip`], is clipped to `path` (inside the clips
61    /// around it). WMF has no clip paths and writes what follows whole.
62    Clip {
63        path: BezPath,
64        even_odd: bool,
65    },
66    Unclip,
67}

82/// A dash pattern.
83#[derive(Clone, Debug, PartialEq)]
84pub struct Dash {
85    /// Dash, gap, dash, gap… in points. An odd count repeats to an even one.
86    pub pattern: Vec<f64>,
87    /// How far into the pattern the path starts.
88    pub offset: f64,
89}
90
91impl Dash {
92    /// Whether the pattern draws dashes: every length finite and not negative, and not all zero.
93    fn is_dashed(&self) -> bool {
94        self.offset.is_finite() && self.pattern.iter().all(|v| v.is_finite() && *v >= 0.0) && self.pattern.iter().any(|v| *v > 0.0)
95    }
96
97    /// Whether an EMF pen draws this pattern itself (`PS_USERSTYLE`): it starts at the path's
98    /// start, has at most 16 entries once made even, and every entry is at least one device unit
99    /// (0.01 mm). Other patterns are written as one line per dash, cut with [`kurbo::dash`]; a
100    /// caller that cuts dashes differently (aligned to corners, say) asks here and passes the
101    /// dashes as the path with no pattern.
102    pub fn emf_pen_draws(&self) -> bool {
103        emf::dash_entries(self).is_some()
104    }
105}

107/// Pixels with straight (not premultiplied) alpha, RGBA8, row by row from the top.

136/// Writes `ops` over `region` as an EMF. The header's description names the application and the
137/// picture (`app`, `title`). None when the region is empty or not finite.
138pub fn emf(ops: &[Op], region: Rect, app: &str, title: &str) -> Option<Vec<u8>> {
139    Some(emf::write(ops, sane(region)?, app, title))
140}
141
142/// Writes `ops` over `region` as a placeable WMF. None when the region is empty or not finite.
143pub fn wmf(ops: &[Op], region: Rect) -> Option<Wmf> {
144    Some(wmf::write(ops, sane(region)?))
145}

147/// `region` with its corners in order, when it is finite and has an area.
148fn sane(region: Rect) -> Option<Rect> {
149    let r = region.abs();
150    ([r.x0, r.y0, r.x1, r.y1].iter().all(|v| v.is_finite()) && r.width() > 0.0 && r.height() > 0.0).then_some(r)
151}

153/// `bp` cut into its dashes (as it is when `dash` draws none), for pens that can't draw them.
154fn dashed(bp: &BezPath, dash: Option<&Dash>) -> BezPath {
155    let Some(d) = dash.filter(|d| d.is_dashed()) else { return bp.clone() };
156    let mut out = BezPath::new();
157    let mut dashes = 0usize;
158    for el in kurbo::dash(bp.iter(), d.offset, &d.pattern) {
159        dashes += usize::from(matches!(el, PathEl::MoveTo(_)));
160        if dashes > MAX_DASHES {
161            // Too many to write one by one: the stroke stays solid.
162            return bp.clone();
163        }
164        out.push(el);
165    }
166    out
167}

169/// `bp` flattened to within `tol`: each subpath's points and whether it closes.
170fn flatten(bp: &BezPath, tol: f64) -> Vec<(Vec<Point>, bool)> {
171    let mut out: Vec<(Vec<Point>, bool)> = vec![];
172    let mut cur: Vec<Point> = vec![];
173    flatten_capped(bp.iter(), tol, |el| match el {
174        PathEl::MoveTo(p) => {
175            if cur.len() > 1 {
176                out.push((std::mem::take(&mut cur), false));
177            }
178            cur = vec![p];
179        }
180        PathEl::LineTo(p) => cur.push(p),
181        PathEl::ClosePath => {
182            if cur.len() > 1 {
183                out.push((std::mem::take(&mut cur), true));
184            }
185            cur.clear();
186        }
187        _ => {}
188    });
189    if cur.len() > 1 {
190        out.push((cur, false));
191    }
192    out
193}

195/// [`kurbo::flatten`], each curve's tolerance raised so that it makes at most about
196/// [`MAX_FLATTEN_PER_CURVE`] segments. kurbo sizes its work and a buffer from the curve's
197/// coordinates before producing anything, so a small curve with far-out handles asks for
198/// gigabytes. The segments a curve needs grow with the square root of its control polygon's
199/// length over the tolerance; that ratio is held under the square of the cap. A curve whose
200/// points aren't finite is drawn as a line to its end.
201fn flatten_capped(path: impl IntoIterator<Item = PathEl>, tolerance: f64, mut callback: impl FnMut(PathEl)) {
202    let mut last = None;
203    let mut start = None;
204    for el in path {
205        let (from, to, length) = match el {
206            PathEl::MoveTo(p) => {
207                (last, start) = (Some(p), Some(p));
208                callback(el);
209                continue;
210            }
211            PathEl::LineTo(p) => {
212                last = Some(p);
213                callback(el);
214                continue;
215            }
216            PathEl::ClosePath => {
217                last = start;
218                callback(el);
219                continue;
220            }
221            PathEl::QuadTo(a, b) => (last, b, last.map(|p| p.distance(a) + a.distance(b))),
222            PathEl::CurveTo(a, b, c) => (last, c, last.map(|p| p.distance(a) + a.distance(b) + b.distance(c))),
223        };
224        last = Some(to);
225        let (Some(from), Some(length)) = (from, length) else { continue };
226        // A short polygon far from the origin still overflows kurbo's `p * (mt * 2.0)` products.
227        let far = |p: Point| !(p.x.abs() <= MAX_FLATTEN_LENGTH && p.y.abs() <= MAX_FLATTEN_LENGTH);
228        let controls_far = match el {
229            PathEl::QuadTo(a, _) => far(a),
230            PathEl::CurveTo(a, b, _) => far(a) || far(b),
231            _ => false,
232        };
233        if length.is_nan() || length > MAX_FLATTEN_LENGTH || far(from) || far(to) || controls_far {
234            callback(PathEl::LineTo(to));
235            continue;
236        }
237        let tol = tolerance.max(length / (MAX_FLATTEN_PER_CURVE * MAX_FLATTEN_PER_CURVE));
238        kurbo::flatten([PathEl::MoveTo(from), el], tol, |e| {
239            if !matches!(e, PathEl::MoveTo(_)) {
240                callback(e);
241            }
242        });
243    }
244}

246/// Little-endian bytes being written.
247#[derive(Default)]
248struct Out(Vec<u8>);
249
250impl Out {
251    fn len(&self) -> usize {
252        self.0.len()
253    }
254    fn u16(&mut self, v: u16) {
255        self.0.extend_from_slice(&v.to_le_bytes());
256    }
257    fn i16(&mut self, v: i16) {
258        self.0.extend_from_slice(&v.to_le_bytes());
259    }
260    fn u32(&mut self, v: u32) {
261        self.0.extend_from_slice(&v.to_le_bytes());
262    }
263    fn i32(&mut self, v: i32) {
264        self.0.extend_from_slice(&v.to_le_bytes());
265    }
266    fn f32(&mut self, v: f32) {
267        self.0.extend_from_slice(&v.to_le_bytes());
268    }
269    fn bytes(&mut self, b: &[u8]) {
270        self.0.extend_from_slice(b);
271    }
272    /// Zeros up to the next multiple of `n` bytes.
273    fn pad(&mut self, n: usize) {
274        while !self.0.len().is_multiple_of(n) {
275            self.0.push(0);
276        }
277    }
278    /// Overwrite the `u32` at `at` (a size or count known only later).
279    fn set_u32(&mut self, at: usize, v: u32) {
280        if let Some(b) = self.0.get_mut(at..at + 4) {
281            b.copy_from_slice(&v.to_le_bytes());
282        }
283    }
284    fn set_i32(&mut self, at: usize, v: i32) {
285        if let Some(b) = self.0.get_mut(at..at + 4) {
286            b.copy_from_slice(&v.to_le_bytes());
287        }
288    }
289}
