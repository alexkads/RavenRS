1//! Windows metafiles: reads Windows Metafiles (WMF), Enhanced Metafiles (EMF) and EMF+ into a
2//! neutral vector model, and (feature `write`) writes EMF and WMF.
3//!
4//! [`parse`] plays a file's records on a model of the device context and returns a [`Picture`]:
5//! filled and stroked paths with their line styles, pattern fills, RGBA bitmaps and text strings,
6//! each with the clip it was drawn under, in a picture frame with y down. Nothing in the model
7//! belongs to an application: mapping it onto a document (or drawing it) is the caller's.
8//! [`Picture::place`] fits a picture into a rectangle for callers that draw it.
9//!
10//! Text stays font-agnostic: a [`Text`] item names the font the file asks for and carries the
11//! string, its advances and alignment; the caller resolves the font and draws the glyphs.
12//!
13//! The reader is pure Rust over byte slices (no file system), so it builds for wasm. Every read is
14//! bounds-checked and every count is capped; hostile input yields a shorter picture or an [`Error`],
15//! never a panic. Records follow MS-WMF, MS-EMF and MS-EMFPLUS; the README lists the records read
16//! and the limits. Records of a type the reader does not know are counted in
17//! [`Picture::skipped_records`].
18#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
19#![forbid(unsafe_code)]

62/// A parsed metafile. Items use coordinates in `0..width` and `0..height`, y down.
63#[derive(Debug, Clone, PartialEq, Default)]
64pub struct Picture {
65    /// Size of the drawing in its own units: points for placeable WMF and EMF (from the frame), device
66    /// or logical units otherwise.
67    pub width: f32,
68    pub height: f32,
69    pub items: Vec<Item>,
70    /// Records of a type the reader does not know (neither drawn nor known to change nothing that
71    /// is kept). A caller can tell its user that part of the file was not read.
72    pub skipped_records: u32,
73}

93/// Caps, joins and dashes of a stroke. The default is GDI's: round caps and joins, a miter limit
94/// of 10, a solid line.
95#[derive(Debug, Clone, PartialEq)]
96pub struct LineStyle {
97    pub cap: Cap,
98    pub join: Join,
99    /// Longest miter as a multiple of the stroke width (for [`Join::Miter`]).
100    pub miter_limit: f32,
101    /// Dash, gap, dash, gap… lengths in the units of the stroke width, starting at the path's
102    /// start. Empty is a solid line.
103    pub dash: Vec<f32>,
104}
105
106impl Default for LineStyle {
107    fn default() -> Self {
108        LineStyle { cap: Cap::Round, join: Join::Round, miter_limit: 10.0, dash: Vec::new() }
109    }
110}

112impl LineStyle {
113    /// The same style with its dash lengths times `k`. None when a length stops being finite.
114    fn scaled(&self, k: f64) -> Option<LineStyle> {
115        let dash = self.dash.iter().map(|d| f(f64::from(*d) * k)).collect::<Option<Vec<f32>>>()?;
116        Some(LineStyle { dash, ..self.clone() })
117    }

128#[derive(Debug, Clone, PartialEq)]
129pub enum Item {
130    /// A filled and/or stroked path, in picture coordinates.
131    Path {
132        segs: Vec<Seg>,
133        fill: Option<[u8; 4]>,
134        /// Colour and width in picture units.
135        stroke: Option<([u8; 4], f32)>,
136        /// Caps, joins and dashes of the stroke (dash lengths in picture units).
137        line: LineStyle,
138        even_odd: bool,
139        /// Clip region in picture units, each rectangle `[x0, y0, x1, y1]`. None is the default clip.
140        region: Option<Vec<[f32; 4]>>,
141        /// Clip paths: the item shows inside all of them (and inside `region`). Outermost first.
142        clips: Vec<Arc<ClipPath>>,
143    },
144    /// A bitmap stretched into `rect` = `[x, y, w, h]` (picture units), RGBA8 rows top-down, straight
145    /// alpha.
146    Bitmap {
147        rect: [f32; 4],
148        /// The unit square (the whole bitmap, y down) to picture space as affine coefficients
149        /// `[a, b, c, d, e, f]` (`x' = a·x + c·y + e`, `y' = b·x + d·y + f`, kurbo's order). For an
150        /// upright bitmap this is `rect` again (`[w, 0, 0, h, x, y]`); under a world transform that
151        /// turns or skews the bitmap it carries the turn, and `rect` is the box around the turned bitmap.
152        transform: [f64; 6],
153        width: u32,
154        height: u32,
155        rgba: Vec<u8>,
156        /// Clip region in picture units. None is the default clip.
157        region: Option<Vec<[f32; 4]>>,
158        /// Clip paths, as on [`Item::Path`].
159        clips: Vec<Arc<ClipPath>>,
160    },
161    /// A path filled with a repeating tile (pattern and hatch brushes). Items filled with the same
162    /// brush share its tile.
163    Pattern {
164        segs: Vec<Seg>,
165        even_odd: bool,
166        tile: Arc<Tile>,
167        /// Clip region in picture units. None is the default clip.
168        region: Option<Vec<[f32; 4]>>,
169        /// Clip paths, as on [`Item::Path`].
170        clips: Vec<Arc<ClipPath>>,
171    },
172    /// A text string.
173    Text(Box<Text>),
174}

212/// Parses a WMF (placeable or plain) or EMF file.
213pub fn parse(bytes: &[u8]) -> Result<Picture, Error> {
214    dib::set_decode_budget(dib::MAX_DECODE_PIXELS);
215    parse_nested(bytes)
216}
217
218/// [`parse`] for a metafile inside the one being parsed: it draws on the same budgets.
219pub(crate) fn parse_nested(bytes: &[u8]) -> Result<Picture, Error> {
220    let b = Bytes(bytes);
221    let (cv, frame) = if emf::is_emf(b) {
222        emf::parse(b)?
223    } else if wmf::is_wmf(b) {
224        // A WMF converted from an EMF can carry the original in MFCOMMENT escapes; Office draws
225        // the WMF's own records (its dx rounding shows), so the copy is not used.
226        wmf::parse(b)?
227    } else {
228        return Err(Error::NotMetafile);
229    };
230    let skipped = cv.skipped;
231    let mut pic = build(cv.items, frame);
232    pic.skipped_records = skipped;
233    Ok(pic)
234}

236/// True when the bytes start like a WMF (placeable or plain header) or an EMF.

244    /// Enhanced Metafile (32-bit coordinates, Bézier curves, paths and clipping), with or without

246    Emf,

251/// The metafile format `bytes` start like, if any.
252pub fn sniff(bytes: &[u8]) -> Option<Kind> {
253    let b = Bytes(bytes);
254    if emf::is_emf(b) {
255        Some(Kind::Emf)
256    } else if wmf::is_wmf(b) {
257        Some(Kind::Wmf)
258    } else {
259        None
260    }
261}

263/// Maps device-space output into the picture frame. Without a usable frame the frame is the bounding
264/// box of the drawing, so the origin is still the top-left of what was drawn.
265fn build(raw: Vec<Raw>, frame: Option<Frame>) -> Picture {
266    let frame = frame.filter(|f| f.rect.iter().all(|v| v.is_finite()) && f.rect[2] > 0.0 && f.rect[3] > 0.0);
267    let (rect, size) = match frame {
268        Some(f) => (f.rect, f.size.unwrap_or((f.rect[2], f.rect[3]))),
269        None => match bbox(&raw) {
270            Some([x0, y0, x1, y1]) => ([x0, y0, x1 - x0, y1 - y0], (x1 - x0, y1 - y0)),
271            None => return Picture::default(),
272        },
273    };
274    let sx = if rect[2] > 0.0 { size.0 / rect[2] } else { 1.0 };
275    let sy = if rect[3] > 0.0 { size.1 / rect[3] } else { 1.0 };
276    let tf = move |p: P| ((p.0 - rect[0]) * sx, (p.1 - rect[1]) * sy);
277    let stroke_scale = (sx + sy) / 2.0;
278    // No stroke is wider than the picture's diagonal: anything wider is a mapping artefact.
279    let diag = size.0.hypot(size.1);
280    // Tiles shared by several fills stay shared.
281    let mut tiles: HashMap<*const canvas::RawTile, (Arc<canvas::RawTile>, Arc<Tile>)> = HashMap::new();
282    // So do clip paths. None is a path with a point that is not finite: its items are left out.
283    let mut clip_paths: ClipCache = HashMap::new();
284    let mut clip = |region: canvas::RawClip| -> Option<ItemClip> {
285        let mut paths = Vec::with_capacity(region.paths.len());
286        for raw in region.paths {
287            let key = Arc::as_ptr(&raw);
288            let mapped = match clip_paths.get(&key) {
289                Some((_, c)) => c.clone(),
290                None => {
291                    let c = raw
292                        .ops
293                        .iter()
294                        .map(|op| seg(op, tf))
295                        .collect::<Option<Vec<Seg>>>()
296                        .map(|segs| Arc::new(ClipPath { segs, even_odd: raw.even_odd }));
297                    // The raw path is kept with its key, so the address is never reused by another.
298                    clip_paths.insert(key, (raw, c.clone()));
299                    c
300                }
301            };
302            paths.push(mapped?);
303        }
304        Some((map_region(region.rects, &tf)?, paths))
305    };
306    let items = raw.into_iter().filter_map(|r| match r {
307        Raw::Path { ops, fill, stroke, line, even_odd, region } => {
308            let segs = ops.iter().map(|op| seg(op, tf)).collect::<Option<Vec<Seg>>>()?;
309            if segs.len() < 2 {
310                return None;
311            }
312            let stroke = stroke.and_then(|(c, w)| Some((c, f((w * stroke_scale).min(diag))?)));
313            let (region, clips) = clip(region)?;
314            Some(Item::Path { segs, fill, stroke, line: line.scaled(stroke_scale)?, even_odd, region, clips })
315        }
316        Raw::Bitmap { rect: r, xf, img, region } => {
317            let (x, y) = tf((r[0], r[1]));
318            let rect = [f(x)?, f(y)?, f(r[2] * sx)?, f(r[3] * sy)?];
319            let (e, g) = tf((xf[4], xf[5]));
320            let transform = [xf[0] * sx, xf[1] * sy, xf[2] * sx, xf[3] * sy, e, g];
321            if !transform.iter().all(|v| v.is_finite()) {
322                return None;
323            }
324            let (region, clips) = clip(region)?;
325            Some(Item::Bitmap { rect, transform, width: img.w as u32, height: img.h as u32, rgba: img.px, region, clips })
326        }
327        Raw::Pattern { ops, even_odd, tile, region } => {
328            let segs = ops.iter().map(|op| seg(op, tf)).collect::<Option<Vec<Seg>>>()?;
329            if segs.len() < 2 {
330                return None;
331            }
332            let key = Arc::as_ptr(&tile);
333            let tile = match tiles.get(&key) {
334                Some((_, t)) => t.clone(),
335                None => {
336                    let (ox, oy) = tf((0.0, 0.0));
337                    let t = Arc::new(Tile {
338                        width: u32::try_from(tile.w).ok()?,
339                        height: u32::try_from(tile.h).ok()?,
340                        rgba: tile.px.clone(),
341                        cell: [f(tile.cell * sx)?, f(tile.cell * sy)?],
342                        origin: [f(ox)?, f(oy)?],
343                    });
344                    // The raw tile is kept with its key, so the address is never reused by another.
345                    tiles.insert(key, (tile.clone(), t.clone()));
346                    t
347                }
348            };
349            let (region, clips) = clip(region)?;
350            Some(Item::Pattern { segs, even_odd, tile, region, clips })
351        }
352        Raw::Text { mut text, region } => {
353            // Device to picture: scale by (sx, sy) after moving the frame's corner to the origin.
354            let [a, b, c, d, e, g] = text.transform;
355            let (e, g) = tf((e, g));
356            text.transform = [a * sx, b * sy, c * sx, d * sy, e, g];
357            if !text.transform.iter().all(|v| v.is_finite()) {
358                return None;
359            }
360            if let Some([x0, y0, x1, y1]) = text.clip {
361                let (p, q) = (tf((f64::from(x0), f64::from(y0))), tf((f64::from(x1), f64::from(y1))));
362                text.clip = Some([f(p.0.min(q.0))?, f(p.1.min(q.1))?, f(p.0.max(q.0))?, f(p.1.max(q.1))?]);
363            }
364            (text.region, text.clips) = clip(region)?;
365            Some(Item::Text(text))
366        }
367    });
368    let items = items.collect();
369    Picture { width: f(size.0).unwrap_or(0.0), height: f(size.1).unwrap_or(0.0), items, skipped_records: 0 }
370}

378fn map_region(region: Option<Vec<[f64; 4]>>, tf: &impl Fn(P) -> P) -> Option<Option<Vec<[f32; 4]>>> {
379    let Some(rects) = region else { return Some(None) };
380    let mut out = Vec::with_capacity(rects.len());
381    for [x0, y0, x1, y1] in rects {
382        let (p, q) = (tf((x0, y0)), tf((x1, y1)));
383        out.push([f(p.0.min(q.0))?, f(p.1.min(q.1))?, f(p.0.max(q.0))?, f(p.1.max(q.1))?]);
384    }
385    Some(Some(out))
386}

388fn f(v: f64) -> Option<f32> {
389    let v = v as f32;
390    v.is_finite().then_some(v)
391}

393fn seg(op: &Op, tf: impl Fn(P) -> P) -> Option<Seg> {
394    let pt = |p: P| {
395        let (x, y) = tf(p);
396        Some((f(x)?, f(y)?))
397    };
398    Some(match *op {
399        Op::M(p) => {
400            let (x, y) = pt(p)?;
401            Seg::Move(x, y)
402        }
403        Op::L(p) => {
404            let (x, y) = pt(p)?;
405            Seg::Line(x, y)
406        }
407        Op::C(a, b, c) => {
408            let (x1, y1) = pt(a)?;
409            let (x2, y2) = pt(b)?;
410            let (x3, y3) = pt(c)?;
411            Seg::Cubic(x1, y1, x2, y2, x3, y3)
412        }
413        Op::Z => Seg::Close,
414    })
415}

417/// Bounding box `[x0, y0, x1, y1]` over the finite points of the output.
418fn bbox(raw: &[Raw]) -> Option<[f64; 4]> {
419    let mut b: Option<[f64; 4]> = None;
420    let mut grow = |p: P| {
421        if p.0.is_finite() && p.1.is_finite() {
422            b = Some(match b {
423                None => [p.0, p.1, p.0, p.1],
424                Some([x0, y0, x1, y1]) => [x0.min(p.0), y0.min(p.1), x1.max(p.0), y1.max(p.1)],
425            });
426        }
427    };
428    for r in raw {
429        match r {
430            Raw::Path { ops, .. } | Raw::Pattern { ops, .. } => {
431                for op in ops {
432                    match *op {
433                        Op::M(p) | Op::L(p) => grow(p),
434                        Op::C(a, c, d) => {
435                            grow(a);
436                            grow(c);
437                            grow(d);
438                        }
439                        Op::Z => {}
440                    }
441                }
442            }
443            Raw::Bitmap { rect, .. } => {
444                grow((rect[0], rect[1]));
445                grow((rect[0] + rect[2], rect[1] + rect[3]));
446            }
447            Raw::Text { text: t, .. } => grow((t.transform[4], t.transform[5])),
448        }
449    }
450    b
