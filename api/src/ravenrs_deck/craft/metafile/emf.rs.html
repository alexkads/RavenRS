13/// ExtTextOut options: no rectangle in the record (EMR_SMALLTEXTOUT, EmrText), and 8-bit
14/// characters (EMR_SMALLTEXTOUT).
15const ETO_NO_RECT: u32 = 0x0100;
16const ETO_SMALL_CHARS: u32 = 0x0200;
17
18const EMF_SIGNATURE: u32 = 0x464D_4520;
19/// Hundredths of a millimetre to points.
20const HMM_TO_PT: f64 = 72.0 / 2540.0;
21/// SRCCOPY, the plain copy of a bitmap.
22const SRCCOPY: u32 = 0x00CC_0020;
23/// LogPenEx pen type bits; PS_COSMETIC pens are one device pixel wide.
24const PS_TYPE_MASK: u32 = 0x000F_0000;
25const PS_COSMETIC: u32 = 0;
26
27pub(crate) fn is_emf(b: Bytes) -> bool {
28    b.u32(0) == Some(1) && b.u32(40) == Some(EMF_SIGNATURE)
29}

31/// The picture frame: `rclFrame` (0.01 mm) mapped to device units through `szlDevice` and
32/// `szlMillimeters`, sized in points. Without a usable frame, the device bounds stand in (1 unit = 1 point).
33pub(crate) fn parse(input: Bytes) -> Result<(Canvas, Option<Frame>), Error> {
34    if !is_emf(input) {
35        return Err(Error::BadHeader);
36    }
37    let hsize = input.u32(4).ok_or(Error::BadHeader)? as usize;
38    let bounds = rect32(input, 8).ok_or(Error::BadHeader)?;
39    let [l, t, r, b] = bounds;
40    let frame = frame_from_header(input).or_else(|| (r > l && b > t).then_some(Frame { rect: [l, t, r - l, b - t], size: None }));
41    let mut cv = Canvas::new();
42    if r > l && b > t {
43        cv.set_clip_surface([l, t, r, b]);
44    }
45    if let Some(px_mm) = device_px_per_mm(input) {
46        cv.set_device_mm(px_mm);
47    }
48    let mut plus = crate::craft::metafile::emfplus::Plus::new();
49    plus.frame = frame.as_ref().map(|f| f.rect);
50    let mut parsed = 0usize;
51    let mut pos = hsize.max(88);
52    while !cv.full() {
53        let (Some(typ), Some(size)) = (input.u32(pos), pos.checked_add(4).and_then(|o| input.u32(o))) else { break };
54        let size = size as usize;
55        if size < 8 || !size.is_multiple_of(4) {
56            break;
57        }
58        let Some(rec) = input.slice(pos, size) else { break };
59        if typ == 14 {
60            break;
61        }
62        if typ == 70 && Bytes(rec).u32(12) == Some(crate::craft::metafile::emfplus::EMF_PLUS) {
63            if let Some(data) = Bytes(rec).tail(16) {
64                plus.comment(&mut cv, data);
65            }
66        } else if plus.plays_gdi() {
67            record(&mut cv, typ, Bytes(rec));
68        }
69        parsed += 1;
70        let Some(next) = pos.checked_add(size) else { break };
71        pos = next;
72    }
73    if parsed == 0 {
74        return Err(Error::NoRecords);
75    }
76    Ok((cv, frame))
77}

79/// Device pixels per millimetre on each axis, from `szlDevice` (72) and the device's size:
80/// `szlMicrometers` (100) when the header has that extension (a header of 108 bytes or more) and
81/// it is set, which is a thousand times finer, else `szlMillimeters` (80).
82fn device_px_per_mm(input: Bytes) -> Option<(f64, f64)> {
83    let (dw, dh) = (input.i32(72)?, input.i32(76)?);
84    if dw <= 0 || dh <= 0 {
85        return None;
86    }
87    if input.u32(4)? >= 108
88        && let (Some(uw), Some(uh)) = (input.u32(100), input.u32(104))
89        && uw > 0
90        && uh > 0
91    {
92        return Some((f64::from(dw) * 1000.0 / f64::from(uw), f64::from(dh) * 1000.0 / f64::from(uh)));
93    }
94    let (mw, mh) = (input.i32(80)?, input.i32(84)?);
95    (mw > 0 && mh > 0).then(|| (f64::from(dw) / f64::from(mw), f64::from(dh) / f64::from(mh)))
96}

98/// The frame from `rclFrame` (offset 24), `szlDevice` (72) and `szlMillimeters` (80) in the header.
99pub(crate) fn frame_from_header(input: Bytes) -> Option<Frame> {
100    let [fl, ft, fr, fb] = rect32(input, 24)?;
101    if fr <= fl || fb <= ft {
102        return None;
103    }
104    // Device units per hundredth of a millimetre.
105    let (px_mm_x, px_mm_y) = device_px_per_mm(input)?;
106    let (sx, sy) = (px_mm_x / 100.0, px_mm_y / 100.0);
107    let rect = [fl * sx, ft * sy, (fr - fl) * sx, (fb - ft) * sy];
108    Some(Frame { rect, size: Some(((fr - fl) * HMM_TO_PT, (fb - ft) * HMM_TO_PT)) })
109}

111fn record(cv: &mut Canvas, typ: u32, r: Bytes) {
112    if cv.full() {
113        return;
114    }
115    let _ = match typ {
116        9 => xy32(r, 8).map(|p| cv.set_window_ext(p)),
117        10 => xy32(r, 8).map(|p| cv.set_window_org(p)),
118        11 => xy32(r, 8).map(|p| cv.set_viewport_ext(p)),
119        12 => xy32(r, 8).map(|p| cv.set_viewport_org(p)),
120        17 => r.u32(8).map(|m| cv.set_map_mode(m)),
121        19 => r.u32(8).map(|m| cv.set_polyfill(m)),
122        20 => r.u32(8).map(|m| cv.set_rop2(m)),
123        33 => {
124            cv.save_dc();
125            Some(())
126        }
127        34 => r.i32(8).map(|n| cv.restore_dc(n)),
128        // SCALEVIEWPORTEXTEX (31) and SCALEWINDOWEXTEX (32): xNum, xDenom, yNum, yDenom.
129        31 | 32 => (|| {
130            let v = [r.i32(8)?, r.i32(12)?, r.i32(16)?, r.i32(20)?].map(f64::from);
131            cv.scale_ext(typ == 32, (v[0], v[1]), (v[2], v[3]));
132            Some(())
133        })(),
134        57 => r.u32(8).map(|d| cv.set_arc_direction(d)),
135        58 => r.u32(8).map(|m| cv.set_miter_limit(f64::from(m))),
136        67 => r.u32(8).map(|m| cv.select_clip_path(m)),
137        68 => {
138            cv.abort_path();
139            Some(())
140        }
141        35 => xform(r, 8).map(|x| cv.modify_world(x, 4)),
142        36 => xform(r, 8).zip(r.u32(32)).map(|(x, m)| cv.modify_world(x, m)),
143        37 => r.u32(8).map(|i| cv.select(i)),
144        38 => create_pen(cv, r),
145        95 => create_ext_pen(cv, r),
146        39 => create_brush(cv, r),
147        93 | 94 => create_dib_brush(cv, r, typ == 93),
148        82 => create_font(cv, r),
149        18 => r.u32(8).map(|m| cv.set_bk_mode(m)),
150        22 => r.u32(8).map(|a| cv.set_text_align(a)),
151        24 => r.u32(8).map(|c| cv.set_text_color(c)),
152        25 => r.u32(8).map(|c| cv.set_bk_color(c)),
153        83 | 84 => ext_text_out(cv, r, typ == 84),
154        108 => small_text_out(cv, r),
155        40 => r.u32(8).map(|i| cv.delete_object(i)),
156        29 => rect32(r, 8).map(|rect| cv.exclude_clip_rect(rect)),
157        30 => rect32(r, 8).map(|rect| cv.intersect_clip_rect(rect)),
158        75 => ext_select_clip(cv, r),
159        27 => xy32(r, 8).map(|p| cv.move_to(p)),
160        54 => xy32(r, 8).map(|p| cv.line_to(p)),
161        42 => rect32(r, 8).map(|[l, t, rr, b]| cv.shape(geom::ellipse(l, t, rr, b), true, true)),
162        43 => rect32(r, 8).map(|[l, t, rr, b]| cv.shape(geom::rect(l, t, rr, b), true, true)),
163        44 => {
164            let (cw, ch) = (r.i32(24), r.i32(28));
165            let (Some(cw), Some(ch)) = (cw, ch) else { return };
166            rect32(r, 8).map(|[l, t, rr, b]| {
167                cv.shape(geom::round_rect(l, t, rr, b, f64::from(cw), f64::from(ch)), true, true);
168            })
169        }
170        45..=47 => arc32(r).map(|(bounds, s, e)| {
171            let kind = match typ {
172                45 => ArcKind::Arc,
173                46 => ArcKind::Chord,
174                _ => ArcKind::Pie,
175            };
176            let [l, t, rr, b] = bounds;
177            let (s, e) = cv.arc_ends(s, e);
178            cv.shape(geom::arc_shape(l, t, rr, b, s, e, kind), kind != ArcKind::Arc, true);
179        }),
180        4 | 87 => counted(r, typ == 4).map(|pts| cv.poly(&pts, false, false, true)),
181        3 | 86 => counted(r, typ == 3).map(|pts| cv.poly(&pts, true, true, true)),
182        6 | 89 => counted(r, typ == 6).map(|pts| cv.line_chain(&pts, false, true)),
183        // POLYBEZIER starts its own figure at the first point and leaves the current point alone;
184        // POLYBEZIERTO continues from the current point.
185        2 | 85 => counted(r, typ == 2).and_then(|pts| {
186            let from = *pts.first()?;
187            cv.bezier(from, pts.get(1..)?);
188            Some(())
189        }),
190        5 | 88 => counted(r, typ == 5).map(|pts| cv.bezier_to(&pts)),
191        7 | 90 => poly_groups(r, typ == 7).map(|(counts, pts)| cv.shape(geom::subpaths(&pts, &counts, false), false, true)),
192        8 | 91 => poly_groups(r, typ == 8).map(|(counts, pts)| cv.shape(geom::subpaths(&pts, &counts, true), true, true)),
193        59 => {
194            cv.begin_path();
195            Some(())
196        }
197        60 => {
198            cv.end_path();
199            Some(())
200        }
201        61 => {
202            cv.close_figure();
203            Some(())
204        }
205        62 => {
206            cv.paint_path(true, false);
207            Some(())
208        }
209        63 => {
210            cv.paint_path(true, true);
211            Some(())
212        }
213        64 => {
214            cv.paint_path(false, true);
215            Some(())
216        }
217        76 | 77 => bitblt(cv, r, typ == 77),
218        80 => set_dibits(cv, r),
219        81 => stretch_dibits(cv, r),
220        114 => alphablend(cv, r),
221        t if IGNORED.contains(&t) => Some(()),
222        _ => {
223            cv.skipped = cv.skipped.saturating_add(1);
224            Some(())
225        }
226    };
227}

229/// Records that change nothing the picture keeps: the header, the brush origin, mapper flags,
230/// stretch mode, colour adjustment, palettes, FLATTENPATH, comments (EMF+ ones are read before
231/// they get here), colour management and colour spaces, pixel formats, font linking and text
232/// justification. Every other record the reader does not play is counted as skipped.
233const IGNORED: &[u32] = &[1, 13, 16, 21, 23, 48, 49, 50, 51, 52, 65, 70, 98, 99, 100, 101, 104, 109, 110, 111, 112, 113, 119, 120, 121, 122];

235/// EMR_EXTSELECTCLIPRGN: `cbRgnData` bytes of `RGNDATA` (rectangles in logical units) and a region
236/// mode. No region data selects the default clip.
237fn ext_select_clip(cv: &mut Canvas, r: Bytes) -> Option<()> {
238    let cb = r.u32(8)? as usize;
239    let mode = r.u32(12)?;
240    if cb == 0 {
241        cv.reset_clip();
242        return Some(());
243    }
244    let header = 16;
245    if cb < 32 {
246        return Some(());
247    }
248    // RDH_RECTANGLES is the only region type GDI writes.
249    if r.u32(header + 4)? != 1 {
250        return Some(());
251    }
252    let n = (r.u32(header + 8)? as usize).min(crate::craft::metafile::clip::MAX_CLIP_RECTS);
253    let end = header.checked_add(cb)?;
254    let mut rects = Vec::new();
255    for i in 0..n {
256        let o = header + 32 + i * 16;
257        if o + 16 > end {
258            break;
259        }
260        rects.push(rect32(r, o)?);
261    }
262    cv.select_clip_rects(&rects, mode);
263    Some(())
264}

266/// Two little-endian i32 values at `off`, as a point.
267fn xy32(r: Bytes, off: usize) -> Option<P> {
268    Some((f64::from(r.i32(off)?), f64::from(r.i32(off + 4)?)))
269}

271/// RECTL (left, top, right, bottom) as i32 values widened to f64.
272fn rect32(r: Bytes, off: usize) -> Option<[f64; 4]> {
273    Some([f64::from(r.i32(off)?), f64::from(r.i32(off + 4)?), f64::from(r.i32(off + 8)?), f64::from(r.i32(off + 12)?)])
274}

281/// XFORM: six f32 values in M11, M12, M21, M22, Dx, Dy order.
282fn xform(r: Bytes, off: usize) -> Option<Xform> {
283    let mut x = [0.0f64; 6];
284    for (i, v) in x.iter_mut().enumerate() {
285        *v = f64::from(r.f32(off + i * 4)?);
286    }
287    Some(x)
288}

295/// A point count from a record, rejected when it exceeds the path budget (before any allocation).
296fn budget(n: u32) -> Option<usize> {
297    let n = n as usize;
298    (n <= MAX_PATH_OPS).then_some(n)
299}

301/// Point count at 24, then the points (32-bit or 16-bit) at 28.
302fn counted(r: Bytes, wide: bool) -> Option<Vec<P>> {
303    let n = budget(r.u32(24)?)?;
304    r.tail(28)?.points(0, n, wide)
305}

307/// Subpath count at 24, total point count at 28, per-subpath counts at 32, then the points.
308fn poly_groups(r: Bytes, wide: bool) -> Option<(Vec<usize>, Vec<P>)> {
309    let n = budget(r.u32(24)?)?;
310    // The record must hold the n counts before anything is allocated for them.
311    let raw = r.slice(32, n.checked_mul(4)?)?;
312    let counts: Vec<usize> = raw.as_chunks::<4>().0.iter().map(|c| u32::from_le_bytes(*c) as usize).collect();
313    let total = counts.iter().try_fold(0usize, |a, c| a.checked_add(*c))?;
314    let total = budget(u32::try_from(total).ok()?)?;
315    let pts = r.tail(n.checked_mul(4)?.checked_add(32)?)?.points(0, total, wide)?;
316    Some((counts, pts))
317}

319/// EMR_CREATEPEN: handle, LogPen (style, width x/y, colour).
320fn create_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
321    let slot = r.u32(8)? as usize;

323    let width = r.i32(16)?;
324    let color = r.u32(24)?;

329/// EMR_EXTCREATEPEN: handle, bitmap offsets (unused), LogPenEx (style, width, brush style, colour,
330/// hatch, the number of style entries at 48 and the entries from 52). A cosmetic pen's width is in
331/// device pixels (always 1), so it is a hairline; a BS_NULL brush draws no line at all. The style
332/// word keeps its end cap and join; the entries are the dashes and gaps of a PS_USERSTYLE pen.
333fn create_ext_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
334    let slot = r.u32(8)? as usize;
335    let (style, width, brush, color) = (r.u32(28)?, r.u32(32)?, r.u32(36)?, r.u32(40)?);
336    let width = if style & PS_TYPE_MASK == PS_COSMETIC { 0.0 } else { f64::from(width) };
337    let n = (r.u32(48).unwrap_or(0) as usize).min(MAX_DASH_ENTRIES);
338    let entries = (0..n).map_while(|i| r.u32(52 + i * 4)).map(f64::from).collect();
339    let style = if brush == BS_NULL { PS_NULL } else { style };
340    cv.add_object(Some(slot), Obj::Pen(Pen { style, width, color: colorref(color), entries }));
341    Some(())
342}

344/// EMR_CREATEBRUSHINDIRECT: handle, LogBrush32 (style, colour, hatch).
345fn create_brush(cv: &mut Canvas, r: Bytes) -> Option<()> {
346    let slot = r.u32(8)? as usize;
347    let style = r.u32(12)?;
348    let color = r.u32(16)?;
349    let mut b = Brush::new(style, colorref(color));
350    if style == BS_HATCHED {
351        b.hatch = r.u32(20).unwrap_or(0);
352    }
353    cv.add_object(Some(slot), Obj::Brush(b));
354    Some(())

357/// EMR_CREATEDIBPATTERNBRUSHPT (94) and EMR_CREATEMONOBRUSH (93): handle, colour usage, then the
358/// bitmap info and bits as offset/size pairs. A mono brush (or one whose palette holds indices,
359/// DIB_PAL_COLORS) takes the text and background colours. An undecodable bitmap leaves the slot
360/// empty.
361fn create_dib_brush(cv: &mut Canvas, r: Bytes, mono: bool) -> Option<()> {
362    let slot = r.u32(8)? as usize;
363    let usage = r.u32(12)?;
364    let (bo, bl) = (r.u32(16)? as usize, r.u32(20)? as usize);
365    let (po, pl) = (r.u32(24)? as usize, r.u32(28)? as usize);
366    let img = dib::decode(Bytes(r.slice(bo, bl)?), Bytes(r.slice(po, pl)?), false)?;
367    let b = Brush::pattern(img, mono || usage == 1)?;
368    cv.add_object(Some(slot), Obj::Brush(b));
369    Some(())
370}

372/// EMR_EXTCREATEFONTINDIRECTW: handle, then a LogFont (MS-EMF 2.2.13): height, width, escapement,
373/// orientation, weight (i32 each), italic, underline, strike-out, charset and four more bytes, then
374/// the face name in 32 UTF-16 units (NUL-terminated).
375fn create_font(cv: &mut Canvas, r: Bytes) -> Option<()> {
376    let slot = r.u32(8)? as usize;
377    let height = r.i32(12)?;
378    let width = r.i32(16).unwrap_or(0);
379    let escapement = r.i32(20).unwrap_or(0);
380    let weight = r.i32(28).unwrap_or(0);
381    let byte = |o: usize| r.0.get(o).copied().unwrap_or(0);
382    let units: Vec<u16> = (0..32).map_while(|i| r.u16(40 + 2 * i)).take_while(|u| *u != 0).collect();
383    let (family, _) = decode_utf16(&units, &[]);
384    let font = Font {
385        family: family.trim().to_string(),
386        height: height as f32,
387        width: width as f32,
388        weight: weight.clamp(0, 1000) as u16,
389        italic: byte(32) != 0,
390        underline: byte(33) != 0,
391        strike: byte(34) != 0,
392        charset: byte(35),
393    };
394    cv.add_object(Some(slot), Obj::Font(Arc::new(LogFont { font, escapement: f64::from(escapement) })));
395    Some(())
396}

398/// The dx array at `off` for `n` characters: i32 advances, or (dx, dy) pairs under ETO_PDY (only
399/// dx is used). Empty when absent or short.
400fn dx_array(r: Bytes, off: usize, n: usize, opts: u32) -> Vec<f32> {
401    let step = if opts & ETO_PDY != 0 { 8 } else { 4 };
402    if off == 0 {
403        return Vec::new();
404    }
405    let dx: Vec<f32> = (0..n).map_while(|i| r.i32(off.checked_add(i.checked_mul(step)?)?)).map(|v| v as f32).collect();
406    if dx.len() == n { dx } else { Vec::new() }
407}

409/// EMR_EXTTEXTOUTA (83) and EMR_EXTTEXTOUTW (84): bounds, graphics mode and scales, then an EmrText
410/// at 36: reference point, character count, string offset, options, the rectangle (absent under
411/// ETO_NO_RECT) and the dx offset. Offsets count from the record start. Glyph-index strings name
412/// glyphs of a font the reader cannot match, so they are left out.
413fn ext_text_out(cv: &mut Canvas, r: Bytes, wide: bool) -> Option<()> {
414    let at = xy32(r, 36)?;
415    let n = (r.u32(44)? as usize).min(MAX_CHARS);
416    let off = r.u32(48)? as usize;
417    let opts = r.u32(52)?;
418    let (rect, dx_off) = if opts & ETO_NO_RECT != 0 { (None, r.u32(56)?) } else { (rect32(r, 56), r.u32(72)?) };
419    if opts & ETO_GLYPH_INDEX != 0 {
420        return None;
421    }
422    let dx = dx_array(r, dx_off as usize, n, opts);
423    let (text, dx) = if wide {
424        let raw = r.slice(off, n.checked_mul(2)?)?;
425        let units: Vec<u16> = raw.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).collect();
426        decode_utf16(&units, &dx)
427    } else {
428        decode_bytes(r.slice(off, n)?, cv.font_charset(), &dx)
429    };
430    cv.text(at, text, dx, opts, rect);
431    Some(())
432}

434/// EMR_SMALLTEXTOUT (MS-EMF 2.3.5.37): x, y, character count, options, graphics mode, scales, the
435/// bounds (absent under ETO_NO_RECT), then the characters: 8-bit under ETO_SMALL_CHARS, else UTF-16.
436fn small_text_out(cv: &mut Canvas, r: Bytes) -> Option<()> {
437    let at = xy32(r, 8)?;
438    let n = (r.u32(16)? as usize).min(MAX_CHARS);
439    let opts = r.u32(20)?;
440    if opts & ETO_GLYPH_INDEX != 0 {
441        return None;
442    }
443    let (rect, off) = if opts & ETO_NO_RECT != 0 { (None, 36) } else { (rect32(r, 36), 52) };
444    let (text, _) = if opts & ETO_SMALL_CHARS != 0 {
445        decode_bytes(r.slice(off, n)?, cv.font_charset(), &[])
446    } else {
447        let raw = r.slice(off, n.checked_mul(2)?)?;
448        let units: Vec<u16> = raw.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).collect();
449        decode_utf16(&units, &[])
450    };
451    cv.text(at, text, Vec::new(), opts, rect);
452    Some(())
453}

455/// BITBLT (76) and STRETCHBLT (77): dest at 24..40, rop at 40, source x/y at 44, bmi at 84, bits at
456/// 92; STRETCHBLT adds the source w/h at 100. A source-less rop fills the destination instead.
457fn bitblt(cv: &mut Canvas, r: Bytes, stretch: bool) -> Option<()> {
458    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 32), i32s(r, 36)];
459    let rop = r.u32(40)?;
460    if Canvas::sourceless(rop) {
461        cv.fill_rop(dst, rop);
462        return Some(());
463    }
464    let (sw, sh) = if stretch { (i32s(r, 100), i32s(r, 104)) } else { (dst[2], dst[3]) };
465    let src = [i32s(r, 44), i32s(r, 48), sw, sh];
466    blit(cv, r, dst, src, (84, 92), Blend::Rop(rop))
467}

469/// ALPHABLEND: laid out like STRETCHBLT, with a BLENDFUNCTION at 40. Constant alpha at 42 applies to
470/// every pixel; AlphaFormat at 43 with AC_SRC_ALPHA (1) also uses the per-pixel alpha.
471fn alphablend(cv: &mut Canvas, r: Bytes) -> Option<()> {
472    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 32), i32s(r, 36)];
473    let constant = *r.0.get(42)?;
474    let per_pixel = (*r.0.get(43)? & 0x01) != 0;
475    let src = [i32s(r, 44), i32s(r, 48), i32s(r, 100), i32s(r, 104)];
476    blit(cv, r, dst, src, (84, 92), Blend::Alpha { constant, per_pixel })
477}

479/// STRETCHDIBITS: dest x/y at 24, src x/y/w/h at 32..48, bmi offset/size at 48, bits at 56, rop at 68,
480/// dest w/h at 72.
481fn stretch_dibits(cv: &mut Canvas, r: Bytes) -> Option<()> {
482    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 72), i32s(r, 76)];
483    let src = [i32s(r, 32), i32s(r, 36), i32s(r, 40), i32s(r, 44)];
484    let rop = r.u32(68)?;
485    blit(cv, r, dst, src, (48, 56), Blend::Rop(rop))
486}

488/// SETDIBITSTODEVICE (MS-EMF 2.3.1.5): dest x/y at 24, src x/y at 32, size (cxSrc, cySrc) at 40, bmi
489/// at 48, bits at 56, start scan at 68. The image keeps its own size; the bits begin at scan line
490/// `iStartScan`, so the source y counts from there.
491fn set_dibits(cv: &mut Canvas, r: Bytes) -> Option<()> {
492    let (cx, cy) = (i32s(r, 40), i32s(r, 44));
493    if cx <= 0.0 || cy <= 0.0 {
494        return None;
495    }
496    let dst = [i32s(r, 24), i32s(r, 28), cx, cy];
497    let src = [i32s(r, 32), i32s(r, 36) - f64::from(r.u32(68)?), cx, cy];
498    blit(cv, r, dst, src, (48, 56), Blend::Rop(SRCCOPY))
499}

501/// Decodes the bitmap at `at` = (offset of the bmi offset/length pair, offset of the bits pair), both
502/// inside the record, and stretches the source rectangle `src` into `dst`. Both are `[x, y, w, h]`; a
503/// non-positive source size means the whole bitmap. `blend` decides how it combines.
504fn blit(cv: &mut Canvas, r: Bytes, dst: [f64; 4], src: [f64; 4], at: (usize, usize), blend: Blend) -> Option<()> {
505    let (bo, bl) = (r.u32(at.0)? as usize, r.u32(at.0 + 4)? as usize);
506    let (po, pl) = (r.u32(at.1)? as usize, r.u32(at.1 + 4)? as usize);
507    if pl == 0 {
508        return None;
509    }
510    let info = Bytes(r.slice(bo, bl)?);
511    let pixels = Bytes(r.slice(po, pl)?);
512    let (mut img, dst) = dib::decode_clip(info, pixels, blend.per_pixel(), src.map(|v| v as i64), dst)?;
513    dib::prepare(&mut img, blend);
514    cv.bitmap(dst, img);
515    Some(())
516}
