1//! Placing a picture on a target (a page or a device): the fitted transform, paths as kurbo paths in
2//! target units and bitmaps as target rectangles. The rasteriser and the PDF writer only paint the result.

12/// Largest bitmap side the renderers draw: within the PDF image limit and the rasteriser's 16-bit sides.
13pub const MAX_BITMAP_SIDE: u32 = 30_000;
14/// Placed coordinates past this (target units) are far off any page; such items are left out.
15const MAX_COORD: f64 = 1e6;
16/// The most a crop side pads the source: a hundred times its size.
17const MIN_CROP: f32 = -100.0;
18/// Smallest placed bitmap side (target units), so the stretch onto it is never singular.
19const MIN_BITMAP_SIDE: f64 = 1e-6;
20/// Maximum aggregate path elements retained in placed output.
21const MAX_PLACED_ELEMENTS: usize = 500_000;
22/// The least of a source axis a crop keeps (0.5%). Files crop pictures to slivers (a header bar
23/// that is 2.7% of a page-sized metafile). A frame of at most 4000 units a side then spans at most
24/// 800,000 units for the whole source: what it covers stays inside `MAX_COORD` and is never left
25/// out as far off the page.
26const MIN_VISIBLE: f64 = 0.005;
27
28/// The part of a picture's source that `crop` (left, top, right, bottom fractions of the source)
29/// keeps, as fractions of the source: [left, top, width, height], bitmap or metafile alike. Each
30/// side is taken as 0–1 (junk as 0). A negative side pads the source instead of cutting it (down
31/// to -100), as a negative `a:srcRect` side does in Office files and a picture fitted whole into
32/// a frame of another shape needs: the window then starts before the source or ends past it.
33/// When the two sides of an axis leave less than 0.5% of it, the left (top) side stands and 0.5%
34/// shows.
35pub fn crop_window(crop: [f32; 4]) -> [f64; 4] {
36    let [l, t, r, b] = crop.map(|v| if v.is_finite() { f64::from(v.clamp(MIN_CROP, 1.0)) } else { 0.0 });
37    let axis = |lo: f64, hi: f64| {
38        let lo = lo.min(1.0 - MIN_VISIBLE);
39        (lo, (1.0 - lo - hi).max(MIN_VISIBLE))
40    };
41    let ((x, w), (y, h)) = (axis(l, r), axis(t, b));
42    [x, y, w, h]
43}

45/// One item of a placed picture, in target units. `region` is the device-context clip region and
46/// `clips` the clip paths: an item shows inside the region and inside every clip path.
47pub enum PlacedItem {
48    /// A filled and/or stroked path. The stroke width is in target units; 0 is a cosmetic hairline that
49    /// the renderer draws at its minimum width. `line` has its dash lengths in target units.
50    Path {
51        path: BezPath,
52        fill: Option<[u8; 4]>,
53        stroke: Option<([u8; 4], f64)>,
54        line: LineStyle,
55        even_odd: bool,
56        region: Option<Arc<BezPath>>,
57        clips: Vec<Arc<PlacedClip>>,
58    },
59    /// `Picture::items[index]`, a bitmap, stretched into `rect`. `transform` maps the unit square
60    /// (the whole bitmap) to target units: the same place when the bitmap is upright, and the
61    /// turned or skewed place when it is not.
62    Bitmap { index: usize, rect: Rect, transform: Affine, region: Option<Arc<BezPath>>, clips: Vec<Arc<PlacedClip>> },
63    /// `Picture::items[index]`, a pattern fill: `path` (target units) filled with the item's tile
64    /// repeated; `tile` maps tile pixels to target units.
65    Pattern { index: usize, path: BezPath, even_odd: bool, tile: Affine, region: Option<Arc<BezPath>>, clips: Vec<Arc<PlacedClip>> },
66    /// `Picture::items[index]`, a text string: `transform` maps its text space to target units;
67    /// `clip` (target units) bounds what it draws.
68    Text { index: usize, transform: Affine, clip: Option<Rect>, region: Option<Arc<BezPath>>, clips: Vec<Arc<PlacedClip>> },
69}

71/// A placed clip region: `Some(None)` is the default clip, `None` a region that does not meet the page.
72type PlacedRegion = Option<Option<Arc<BezPath>>>;
73
74/// A clip path in target units. Items under the same clip share it.
75pub struct PlacedClip {
76    pub path: BezPath,
77    pub even_odd: bool,
78}

80impl Picture {
81    /// Whether the picture has a finite, positive size, so it can be fitted into a frame.
82    pub fn has_size(&self) -> bool {
83        self.width.is_finite() && self.height.is_finite() && self.width > 0.0 && self.height > 0.0
84    }
85
86    /// The transform that fits the picture into `r` with the crop fractions `crop` (left, top, right,
87    /// bottom) removed first: the visible part fills `r`. None for a degenerate size.
88    pub fn fit(&self, r: Rect, crop: [f32; 4]) -> Option<Affine> {
89        let [cl, ct, vw, vh] = crop_window(crop);
90        let (w, h) = (f64::from(self.width), f64::from(self.height));
91        let sx = r.width() / (w * vw);
92        let sy = r.height() / (h * vh);
93        (sx.is_finite() && sy.is_finite() && sx > 0.0 && sy > 0.0)
94            .then(|| Affine::translate((r.x0 - cl * w * sx, r.y0 - ct * h * sy)) * Affine::scale_non_uniform(sx, sy))
95    }
96
97    /// The items placed into `r` with `crop` applied. Items that are not finite or reach past ±1e6
98    /// target units are left out (they are far off any page), as are bitmaps with no area. Stroke
99    /// widths are scaled to target units (along the mean axis scale, so a non-uniform fit keeps
100    /// strokes even).
101    pub fn place(&self, r: Rect, crop: [f32; 4]) -> Vec<PlacedItem> {
102        let Some(tf) = self.fit(r, crop) else { return Vec::new() };
103        let c = tf.as_coeffs();
104        let k = (c[0].abs() + c[3].abs()) / 2.0;
105        let mut out = Vec::with_capacity(self.items.len());
106        let mut total_elements = 0usize;
107        // The last region placed: consecutive items under the same clip share one placed path.
108        // Each region expanded counts against the element budget like the paths do.
109        let region_elements = std::cell::Cell::new(0usize);
110        let mut cached_region: Option<(Option<Vec<[f32; 4]>>, PlacedRegion)> = None;
111        let mut get_region = |reg: &Option<Vec<[f32; 4]>>| -> PlacedRegion {
112            if let Some((key, res)) = &cached_region
113                && key == reg
114            {
115                return res.clone();
116            }
117            let res = placed_region(&tf, reg);
118            let n = res.as_ref().and_then(Option::as_ref).map_or(0, |p| p.elements().len());
119            region_elements.set(region_elements.get().saturating_add(n));
120            cached_region = Some((reg.clone(), res.clone()));
121            res
122        };
123        // Whether the paths and regions placed so far are past the budget.
124        let over = |paths: usize| paths.saturating_add(region_elements.get()) > MAX_PLACED_ELEMENTS;
125
126        // Clip paths by address: the picture holds them for the whole call. None is a clip that is
127        // not finite or reaches past the coordinate limit; what it clips is left out.
128        let mut placed_clips: HashMap<*const ClipPath, Option<Arc<PlacedClip>>> = HashMap::new();
129        let mut clip_elements = 0usize;
130        let mut get_clips = |clips: &[Arc<ClipPath>]| -> Option<Vec<Arc<PlacedClip>>> {
131            clips
132                .iter()
133                .map(|c| {
134                    placed_clips
135                        .entry(Arc::as_ptr(c))
136                        .or_insert_with(|| {
137                            let path = tf * bez_path(&c.segs);
138                            clip_elements = clip_elements.saturating_add(path.elements().len());
139                            (clip_elements <= MAX_PLACED_ELEMENTS && in_range(path.bounding_box()))
140                                .then(|| Arc::new(PlacedClip { path, even_odd: c.even_odd }))
141                        })
142                        .clone()
143                })
144                .collect()
145        };
146
147        for (index, item) in self.items.iter().enumerate() {
148            if total_elements.saturating_add(region_elements.get()) >= MAX_PLACED_ELEMENTS {
149                break;
150            }
151            match item {
152                Item::Path { segs, fill, stroke, line, even_odd, region, clips } => {
153                    let path = tf * bez_path(segs);
154                    let Some(region) = get_region(region) else { continue };
155                    // This item's own region is expanded by now: it counts before the item is kept.
156                    if over(total_elements) {
157                        break;
158                    }
159                    let Some(clips) = get_clips(clips) else { continue };
160                    let line = LineStyle { dash: line.dash.iter().map(|d| (f64::from(*d) * k) as f32).collect(), ..line.clone() };
161                    let bounds = path.bounding_box();
162                    if in_range(bounds) && line.dash.iter().all(|d| d.is_finite()) {
163                        total_elements = total_elements.saturating_add(path.elements().len());
164                        // A pen wider than twice the distance across the frame and the path
165                        // already covers the whole frame: wider adds nothing the frame shows, only
166                        // work for the renderer flattening its round caps (the width is file data).
167                        let u = bounds.union(r);
168                        let widest = 2.0 * u.width().hypot(u.height());
169                        let stroke = stroke.map(|(c, w)| (c, (f64::from(w) * k).min(widest)));
170                        out.push(PlacedItem::Path { path, fill: *fill, stroke, line, even_odd: *even_odd, region, clips });
171                    }
172                }
173                Item::Pattern { segs, even_odd, tile: t, region, clips } => {
174                    let path = tf * bez_path(segs);
175                    let Some(region) = get_region(region) else { continue };
176                    // This item's own region is expanded by now: it counts before the item is kept.
177                    if over(total_elements) {
178                        break;
179                    }
180                    let Some(clips) = get_clips(clips) else { continue };
181                    let tile = tf
182                        * Affine::translate((f64::from(t.origin[0]), f64::from(t.origin[1])))
183                        * Affine::scale_non_uniform(f64::from(t.cell[0]), f64::from(t.cell[1]));
184                    let det = tile.determinant();
185                    if in_range(path.bounding_box()) && det.is_finite() && det.abs() > 1e-12 {
186                        total_elements = total_elements.saturating_add(path.elements().len());
187                        out.push(PlacedItem::Pattern { index, path, even_odd: *even_odd, tile, region, clips });
188                    }
189                }
190                Item::Text(t) => {
191                    let transform = tf * Affine::new(t.transform);
192                    let Some(region) = get_region(&t.region) else { continue };
193                    // This item's own region is expanded by now: it counts before the item is kept.
194                    if over(total_elements) {
195                        break;
196                    }
197                    let Some(clips) = get_clips(&t.clips) else { continue };
198                    let clip =
199                        t.clip.map(|[x0, y0, x1, y1]| tf.transform_rect_bbox(Rect::new(f64::from(x0), f64::from(y0), f64::from(x1), f64::from(y1))));
200                    let c = transform.as_coeffs();
201                    let origin = Rect::new(c[4], c[5], c[4], c[5]);
202                    if in_range(origin) && c.iter().all(|v| v.is_finite() && v.abs() <= MAX_COORD) && clip.is_none_or(in_range) {
203                        out.push(PlacedItem::Text { index, transform, clip, region, clips });
204                    }
205                }
206                Item::Bitmap { rect: [x, y, w, h], transform, region, clips, .. } => {
207                    let (x0, y0) = (f64::from(*x), f64::from(*y));
208                    let rect = tf.transform_rect_bbox(Rect::new(x0, y0, x0 + f64::from(*w), y0 + f64::from(*h)));
209                    let Some(region) = get_region(region) else { continue };
210                    // This item's own region is expanded by now: it counts before the item is kept.
211                    if over(total_elements) {
212                        break;
213                    }
214                    let Some(clips) = get_clips(clips) else { continue };
215                    let transform = tf * Affine::new(*transform);
216                    let finite = transform.as_coeffs().iter().all(|v| v.is_finite() && v.abs() <= MAX_COORD);
217                    if in_range(rect) && finite && rect.width() > MIN_BITMAP_SIDE && rect.height() > MIN_BITMAP_SIDE {
218                        out.push(PlacedItem::Bitmap { index, rect, transform, region, clips });
219                    }
220                }
221            }
222        }
223        out
224    }
225}

227impl Item {
228    /// Takes a bitmap's pixels out of the item (its geometry stays) and returns them as (width, height,
229    /// RGBA) when a renderer can draw them: both sides in 1..=[`MAX_BITMAP_SIDE`] and exactly that many
230    /// pixels. None for paths and for bitmaps that cannot be drawn.
231    pub fn take_pixels(&mut self) -> Option<(u32, u32, Vec<u8>)> {
232        let Item::Bitmap { width, height, rgba, .. } = self else { return None };
233        let (w, h, px) = (*width, *height, std::mem::take(rgba));
234        let sides = (1..=MAX_BITMAP_SIDE).contains(&w) && (1..=MAX_BITMAP_SIDE).contains(&h);
235        (sides && px.len() == (w as usize).saturating_mul(h as usize).saturating_mul(4)).then_some((w, h, px))
236    }
237}

239/// The device-context clip in target units. `Some(None)` is the default clip. `None` means the
240/// region does not meet the page, so the item draws nothing.
241fn placed_region(tf: &Affine, region: &Option<Vec<[f32; 4]>>) -> PlacedRegion {
242    let Some(rects) = region else { return Some(None) };
243    let mut path = BezPath::new();
244    for [x0, y0, x1, y1] in rects {
245        let r = tf.transform_rect_bbox(Rect::new(f64::from(*x0), f64::from(*y0), f64::from(*x1), f64::from(*y1)));
246        if in_range(r) && r.width() > 1e-9 && r.height() > 1e-9 {
247            path.extend(r.to_path(0.1));
248        }
249    }
250    (!path.is_empty()).then_some(Some(Arc::new(path)))
251}

253/// Segments as a kurbo path. A segment after a close opens a new subpath at the last start point.
254pub fn bez_path(segs: &[Seg]) -> BezPath {
255    let mut p = BezPath::new();
256    let mut start = (0.0, 0.0);
257    let mut open = false;
258    for s in segs {
259        match *s {
260            Seg::Move(x, y) => {
261                start = (f64::from(x), f64::from(y));
262                p.move_to(start);
263                open = true;
264            }
265            Seg::Line(x, y) => {
266                if !open {
267                    p.move_to(start);
268                }
269                p.line_to((f64::from(x), f64::from(y)));
270                open = true;
271            }
272            Seg::Cubic(a, b, c, d, e, f) => {
273                if !open {
274                    p.move_to(start);
275                }
276                p.curve_to((f64::from(a), f64::from(b)), (f64::from(c), f64::from(d)), (f64::from(e), f64::from(f)));
277                open = true;
278            }
279            Seg::Close => {
280                if open {
281                    p.close_path();
282                }
283                open = false;
284            }
285        }
286    }
287    p
288}

290/// Whether a placed box is finite and within ±[`MAX_COORD`].
291fn in_range(b: Rect) -> bool {
292    [b.x0, b.y0, b.x1, b.y1].iter().all(|v| v.is_finite() && v.abs() < MAX_COORD)
