1//! The size of a metafile picture without playing its records.
2
3use crate::craft::metafile::bytes::Bytes;
4use crate::craft::metafile::{Error, emf, f, parse, wmf};
5
6/// A placeable WMF header is 22 bytes, followed by the 9-word standard header.
7const PLACEABLE_KEY: u32 = 0x9AC6_CDD7;
8const PLACEABLE_LEN: usize = 22;
9const DEFAULT_INCH: f64 = 1440.0;
10
11/// The size of a WMF or EMF picture in points (device or logical units when the file states no
12/// size), the same as `parse(bytes)` would put in the picture's `width` and `height`, or the error
13/// `parse` would give.
14///
15/// An EMF and a placeable WMF carry their frame in the header, so the size is read from there
16/// without decoding a bitmap or building a path. Any other file (a plain WMF, an EMF without a
17/// usable frame) takes the full parse.
18pub fn size(bytes: &[u8]) -> Result<(f32, f32), Error> {
19    if let Some(size) = header_size(Bytes(bytes)) {
20        return Ok(size);
21    }
22    let pic = parse(bytes)?;
23    Ok((pic.width, pic.height))
24}
25
26/// The size when the header decides it and the file has a first record (otherwise `parse` fails
27/// with `NoRecords`, so the header alone is not enough). None sends the caller to the full parse.
28pub(crate) fn header_size(b: Bytes) -> Option<(f32, f32)> {
29    if emf::is_emf(b) {
30        return emf_size(b);
31    }
32    if wmf::is_wmf(b) && b.u32(0) == Some(PLACEABLE_KEY) {
33        return placeable_wmf_size(b);
34    }
35    None
36}
37
38/// `rclFrame` mapped to points. The frame must pass the checks `build` applies to it.
39fn emf_size(b: Bytes) -> Option<(f32, f32)> {
40    let hsize = b.u32(4)? as usize;
41    b.slice(8, 16)?;
42    let frame = emf::frame_from_header(b)?;
43    if !frame.rect.iter().all(|v| v.is_finite()) || frame.rect[2] <= 0.0 || frame.rect[3] <= 0.0 {
44        return None;
45    }
46    // The records start after the header; the parse needs one that is not the end record.
47    let pos = hsize.max(88);
48    let (typ, len) = (b.u32(pos)?, b.u32(pos.checked_add(4)?)? as usize);
49    if len < 8 || !len.is_multiple_of(4) || b.slice(pos, len).is_none() || typ == 14 {
50        return None;
51    }
52    let (w, h) = frame.size?;
53    Some((f(w).unwrap_or(0.0), f(h).unwrap_or(0.0)))
54}
55
56/// The placeable header's size in points, for a file whose records leave the target mapping alone
57/// (see [`sets_the_mapping`]); the full parse decides the others.
58fn placeable_wmf_size(b: Bytes) -> Option<(f32, f32)> {
59    let [l, t, r, bt] = [6, 8, 10, 12].map(|o| b.i16(o).map(f64::from));
60    let (l, t, r, bt) = (l?, t?, r?, bt?);
61    let inch = b.u16(14).filter(|v| *v > 0).map_or(DEFAULT_INCH, f64::from);
62    let (w, h) = (r - l, bt - t);
63    if w <= 0.0 || h <= 0.0 {
64        return None;
65    }
66    let hsize = b.u16(PLACEABLE_LEN + 2)? as usize;
67    if hsize < 9 {
68        return None;
69    }
70    let pos = PLACEABLE_LEN.checked_add(hsize.checked_mul(2)?)?;
71    let (words, func) = (b.u32(pos)? as usize, b.u16(pos.checked_add(4)?)?);
72    let len = words.checked_mul(2).filter(|l| *l >= 6)?;
73    if func == 0 || b.slice(pos, len).is_none() {
74        return None;
75    }
76    if sets_the_mapping(b, pos) {
77        return None;
78    }
79    Some((f(w * 72.0 / inch)?, f(h * 72.0 / inch)?))
80}
81
82/// Whether a record from `pos` on is SetMapMode (0x0103) or SetViewportExt (0x020E). Those can change
83/// the frame the full parse ends with (a zero-size viewport leaves none), so the header size is not
84/// enough. The records are only walked, not played, with the same stops as `wmf::parse`.
85fn sets_the_mapping(b: Bytes, mut pos: usize) -> bool {
86    while let (Some(words), Some(func)) = (b.u32(pos), pos.checked_add(4).and_then(|o| b.u16(o))) {
87        let Some(len) = (words as usize).checked_mul(2).filter(|l| *l >= 6) else { return false };
88        if func == 0 || b.slice(pos, len).is_none() {
89            return false;
90        }
91        if matches!(func, 0x0103 | 0x020E) {
92            return true;
93        }
94        let Some(next) = pos.checked_add(len) else { return false };
95        pos = next;
96    }
97    false
98}
