1//! WMF reader (MS-WMF). A file is an optional 22-byte placeable header, a 9-word standard header, then
2//! records of (size in 16-bit words, function, parameters). Parameters are read at byte 6 of each record.

13const PLACEABLE_KEY: u32 = 0x9AC6_CDD7;
14const DEFAULT_INCH: f64 = 1440.0;
15/// A placeable WMF plays into a target sized in points; brushes are laid out in 96-dpi pixels.
16const PT_PER_PX: f64 = 72.0 / 96.0;
17
18pub(crate) fn is_wmf(b: Bytes) -> bool {
19    if b.u32(0) == Some(PLACEABLE_KEY) {
20        return true;
21    }
22    matches!(b.u16(0), Some(1 | 2)) && b.u16(2) == Some(9) && matches!(b.u16(4), Some(0x100 | 0x300))
23}

25/// Reads all records. Returns the drawing state and the picture frame, or an error when no record
26/// could be read at all.
27pub(crate) fn parse(input: Bytes) -> Result<(Canvas, Option<Frame>), Error> {
28    let placeable = input.u32(0) == Some(PLACEABLE_KEY);
29    // Placeable bounds: logical corners (left, top, right, bottom) and their size in points.
30    let (hdr, bounds) = if placeable {
31        let [l, t, r, b] = [6, 8, 10, 12].map(|o| input.i16(o).map(f64::from));
32        let (Some(l), Some(t), Some(r), Some(b)) = (l, t, r, b) else { return Err(Error::BadHeader) };
33        let inch = input.u16(14).filter(|v| *v > 0).map_or(DEFAULT_INCH, f64::from);
34        let (w, h) = (r - l, b - t);
35        let bounds = (w > 0.0 && h > 0.0).then_some(([l, t, r, b], (w * 72.0 / inch, h * 72.0 / inch)));
36        (22, bounds)
37    } else {
38        (0, None)
39    };
40    let hsize = input.u16(hdr + 2).ok_or(Error::BadHeader)? as usize;
41    if hsize < 9 {
42        return Err(Error::BadHeader);
43    }
44    let mut pos = hdr.checked_add(hsize.checked_mul(2).ok_or(Error::BadHeader)?).ok_or(Error::BadHeader)?;
45    // Played into a target like Word does: the placeable bounds are the first window, the target (the
46    // picture size) the viewport. A plain WMF's target is its first window.
47    let mut cv = Canvas::new();
48    if let Some((c, size)) = bounds {
49        cv.play_into(Some([c[0], c[1], c[2] - c[0], c[3] - c[1]]), size);
50        cv.set_clip_surface([0.0, 0.0, size.0, size.1]);
51        cv.pattern_px = PT_PER_PX;
52    }
53    let mut player_viewport = bounds.is_some();
54    let mut parsed = 0usize;
55    while !cv.full() {
56        let (Some(words), Some(func)) = (input.u32(pos), pos.checked_add(4).and_then(|o| input.u16(o))) else { break };
57        let Some(len) = (words as usize).checked_mul(2).filter(|l| *l >= 6) else { break };
58        let Some(rec) = input.slice(pos, len) else { break };
59        if func == 0 {
60            break;
61        }
62        record(&mut cv, func, Bytes(rec));
63        if func == 0x020C && cv.window_set && !player_viewport && !cv.viewport_set {
64            let we = cv.map.window_ext();
65            cv.play_into(None, (we.0.abs(), we.1.abs()));
66            player_viewport = true;
67        }
68        parsed += 1;
69        let Some(next) = pos.checked_add(len) else { break };
70        pos = next;
71    }
72    if parsed == 0 {
73        return Err(Error::NoRecords);
74    }
75    let frame = match bounds {
76        // Played into its target: the frame is the target.
77        Some((_, size)) if !cv.viewport_set && cv.map.uses_extents() => Some(Frame { rect: [0.0, 0.0, size.0, size.1], size: Some(size) }),
78        // The file set its own viewport, so the drawing is in its device units and the logical bounds
79        // go through the final mapping.
80        Some((c, size)) => {
81            let (a, b) = (cv.map.device((c[0], c[1])), cv.map.device((c[2], c[3])));
82            let rect = [a.0.min(b.0), a.1.min(b.1), (b.0 - a.0).abs(), (b.1 - a.1).abs()];
83            (rect[2] > 0.0 && rect[3] > 0.0).then_some(Frame { rect, size: Some(size) })
84        }
85        None if player_viewport && !cv.viewport_set && cv.map.uses_extents() => {
86            let ve = cv.map.viewport_ext();
87            Some(Frame { rect: [0.0, 0.0, ve.0, ve.1], size: None })
88        }
89        None if cv.window_set => Some(Frame { rect: cv.map.window_rect(), size: None }),
90        None => None,
91    };
92    Ok((cv, frame))
93}

95fn record(cv: &mut Canvas, func: u16, r: Bytes) {
96    if cv.full() {
97        return;
98    }
99    let _ = match func {
100        0x0103 => r.u16(6).map(|m| cv.set_map_mode(u32::from(m))),
101        0x0106 => r.u16(6).map(|m| cv.set_polyfill(u32::from(m))),
102        0x0104 => r.u16(6).map(|m| cv.set_rop2(u32::from(m))),
103        // Y is stored before X in these records.
104        0x020B => xy16(r, 6).map(|p| cv.set_window_org(p)),
105        0x020C => xy16(r, 6).map(|p| cv.set_window_ext(p)),
106        0x020D => xy16(r, 6).map(|p| cv.set_viewport_org(p)),
107        0x020E => xy16(r, 6).map(|p| cv.set_viewport_ext(p)),
108        0x001E => {
109            cv.save_dc();
110            Some(())
111        }
112        0x0127 => r.i16(6).map(|n| cv.restore_dc(i32::from(n))),
113        0x0214 => xy16(r, 6).map(|p| cv.move_to(p)),
114        0x0213 => xy16(r, 6).map(|p| cv.line_to(p)),
115        0x0325 => counted_points(r).map(|pts| cv.poly(&pts, false, false, true)),
116        0x0324 => counted_points(r).map(|pts| cv.poly(&pts, true, true, true)),
117        0x0538 => polypolygon(r).map(|ops| cv.shape(ops, true, true)),
118        0x041B => rect16(r, 6).map(|[l, t, rr, b]| cv.shape(geom::rect(l, t, rr, b), true, true)),
119        0x061C => {
120            let [ew, eh] = [r.i16(8), r.i16(6)];
121            let (Some(ew), Some(eh)) = (ew, eh) else { return };
122            rect16(r, 10).map(|[l, t, rr, b]| {
123                cv.shape(geom::round_rect(l, t, rr, b, f64::from(ew), f64::from(eh)), true, true);
124            })
125        }
126        0x0415 => rect16(r, 6).map(|rect| cv.exclude_clip_rect(rect)),
127        0x0416 => rect16(r, 6).map(|rect| cv.intersect_clip_rect(rect)),
128        0x0418 => rect16(r, 6).map(|[l, t, rr, b]| cv.shape(geom::ellipse(l, t, rr, b), true, true)),
129        0x0817 | 0x081A | 0x0830 => arc16(r).map(|(bounds, s, e)| {
130            let kind = match func {
131                0x0817 => ArcKind::Arc,
132                0x081A => ArcKind::Pie,
133                _ => ArcKind::Chord,
134            };
135            let [l, t, rr, b] = bounds;
136            cv.shape(geom::arc_shape(l, t, rr, b, s, e, kind), kind != ArcKind::Arc, true);
137        }),
138        0x02FA => create_pen(cv, r),
139        0x02FC => create_brush(cv, r),
140        0x02FB => {
141            let f = font(r).unwrap_or_default();
142            cv.add_object(None, Obj::Font(Arc::new(f)));
143            Some(())
144        }
145        0x0142 => {
146            // A pattern the reader cannot decode still takes its handle slot.
147            let b = dib_pattern_brush(r).map_or(Obj::Other, Obj::Brush);
148            cv.add_object(None, b);
149            Some(())
150        }
151        // Palettes, old-style pattern brushes and regions take a handle slot too, so later
152        // SELECTOBJECT handles count them; selecting one draws nothing.
153        0x00F7 | 0x01F9 | 0x06FF => {
154            cv.add_object(None, Obj::Other);
155            Some(())
156        }
157        0x0209 => r.u32(6).map(|c| cv.set_text_color(c)),
158        0x0201 => r.u32(6).map(|c| cv.set_bk_color(c)),
159        0x0102 => r.u16(6).map(|m| cv.set_bk_mode(u32::from(m))),
160        0x012E => r.u16(6).map(|a| cv.set_text_align(u32::from(a))),
161        0x0A32 => ext_text_out(cv, r),
162        0x0521 => text_out(cv, r),
163        // PATBLT: rop(4), height, width, y, x.
164        0x061D => r.u32(6).map(|rop| cv.fill_rop([i16s(r, 16), i16s(r, 14), i16s(r, 12), i16s(r, 10)], rop)),
165        0x012D => r.u16(6).map(|h| cv.select(wmf_handle(h))),
166        0x01F0 => r.u16(6).map(|h| cv.delete_object(u32::from(h))),
167        // STRETCHDIB: rop(4) usage(2) srcH srcW ySrc xSrc destH destW yDest xDest, then the DIB.
168        0x0F43 => {
169            let rop = r.u32(6).unwrap_or(0);
170            dib_blit(cv, r, 28, rop, [i16s(r, 26), i16s(r, 24), i16s(r, 22), i16s(r, 20)], [i16s(r, 18), i16s(r, 16), i16s(r, 14), i16s(r, 12)])
171        }
172        0x0B41 => dib_stretchblt(cv, r),
173        0x0940 => dib_bitblt(cv, r),
174        f if IGNORED.contains(&f) => Some(()),
175        _ => {
176            cv.skipped = cv.skipped.saturating_add(1);
177            Some(())
178        }
179    };
180}

182/// Records that change nothing the picture keeps: relative-absolute and stretch modes, character
183/// spacing and justification, mapper flags, palettes, escapes and the layout. Every other record
184/// the reader does not play is counted as skipped.
185const IGNORED: &[u16] = &[0x0105, 0x0107, 0x0108, 0x020A, 0x0231, 0x0234, 0x0035, 0x0037, 0x0139, 0x0436, 0x0626, 0x0149];

187/// A 16-bit WMF handle as a canvas object index: stock objects are `0x8000 | index`, which the
188/// canvas reads with the 32-bit stock flag.
189fn wmf_handle(h: u16) -> u32 {
190    let h = u32::from(h);
191    if h & 0x8000 != 0 { (h & 0x7FFF) | 0x8000_0000 } else { h }
192}

194/// An i16 field as f64; 0 past the end.
195fn i16s(r: Bytes, off: usize) -> f64 {
196    r.i16(off).map_or(0.0, f64::from)

199/// CREATEPENINDIRECT: style, width (x, y), colour. Only the width's x component is used.
200fn create_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
201    let style = r.u16(6)?;
202    let width = r.i16(8)?;
203    let color = r.u32(12)?;
204    cv.add_object(None, Obj::Pen(Pen::new(u32::from(style), f64::from(width), colorref(color))));
205    Some(())
206}
207
208/// CREATEBRUSHINDIRECT: style, colour, hatch.
209fn create_brush(cv: &mut Canvas, r: Bytes) -> Option<()> {
210    let style = u32::from(r.u16(6)?);
211    let color = r.u32(8)?;
212    let mut b = Brush::new(style, colorref(color));
213    if style == BS_HATCHED {
214        b.hatch = u32::from(r.u16(12).unwrap_or(0));
215    }
216    cv.add_object(None, Obj::Brush(b));
217    Some(())
218}

220/// DIBCREATEPATTERNBRUSH: style, colour usage, then a DIB (MS-WMF 2.3.4.8). The DIB's own palette
221/// gives the colours; with DIB_PAL_COLORS (indices into a palette the reader does not keep) a
222/// monochrome DIB takes the text and background colours.
223fn dib_pattern_brush(r: Bytes) -> Option<Brush> {
224    let usage = r.u16(8)?;
225    let dib = r.tail(10)?;
226    let n = dib::info_len(dib)?;
227    let img = dib::decode(Bytes(dib.slice(0, n)?), dib.tail(n)?, false)?;
228    Brush::pattern(img, usage == 1)
229}

231/// Font object (MS-WMF 2.2.1.2): height, width, escapement, orientation, weight (i16 each), italic,
232/// underline, strike-out and charset bytes, three precision/quality bytes, pitch and family, then the
233/// face name (up to 32 bytes, NUL-terminated, in the font's character set).
234fn font(r: Bytes) -> Option<LogFont> {
235    let height = r.i16(6)?;
236    let width = r.i16(8).unwrap_or(0);
237    let escapement = r.i16(10).unwrap_or(0);
238    let weight = r.i16(14).unwrap_or(0);
239    let byte = |o: usize| r.0.get(o).copied().unwrap_or(0);
240    let charset = byte(19);
241    let name = r.0.get(24..).unwrap_or(&[]);
242    let name = name.get(..32).unwrap_or(name);
243    let name = name.split(|b| *b == 0).next().unwrap_or(&[]);
244    // A face name is in the font's own code page (East Asian fonts are often named in their
245    // script); a symbol font's name is not in symbol codes.
246    let (family, _) = decode_bytes(name, if charset == crate::craft::metafile::SYMBOL_CHARSET { 0 } else { charset }, &[]);
247    Some(LogFont {
248        font: Font {
249            family: family.trim().to_string(),
250            height: f32::from(height),
251            width: f32::from(width),
252            weight: weight.clamp(0, 1000) as u16,
253            italic: byte(16) != 0,
254            underline: byte(17) != 0,
255            strike: byte(18) != 0,
256            charset,
257        },
258        escapement: f64::from(escapement),
259    })
260}

267/// EXTTEXTOUT (MS-WMF 2.3.3.5): y, x, string length, options, the rectangle (left, top, right,
268/// bottom) when ETO_OPAQUE or ETO_CLIPPED is set, the string padded to a word, then an optional dx
269/// array of i16 advances.
270fn ext_text_out(cv: &mut Canvas, r: Bytes) -> Option<()> {
271    let (y, x) = (r.i16(6)?, r.i16(8)?);
272    let n = usize::from(r.u16(10)?).min(MAX_CHARS);
273    let opts = u32::from(r.u16(12)?);
274    let has_rect = opts & (ETO_OPAQUE | ETO_CLIPPED) != 0;
275    let rect = if has_rect { Some([i16s(r, 14), i16s(r, 16), i16s(r, 18), i16s(r, 20)]) } else { None };
276    let at = if has_rect { 22 } else { 14 };
277    // A string running past the record is cut to what the record holds.
278    let avail = r.0.len().saturating_sub(at);
279    let n = n.min(avail);
280    let bytes = r.slice(at, n)?;
281    let dx_at = at + n.next_multiple_of(2);
282    let dx: Vec<f32> = if r.0.len() >= dx_at + 2 * n { (0..n).filter_map(|i| r.i16(dx_at + 2 * i)).map(f32::from).collect() } else { Vec::new() };
283    let (text, dx) = decode_bytes(bytes, charset(cv), &dx);
284    cv.text((f64::from(x), f64::from(y)), text, dx, opts, rect);
285    Some(())
286}

288/// TEXTOUT (MS-WMF 2.3.3.20): string length, the string padded to a word, then y and x.
289fn text_out(cv: &mut Canvas, r: Bytes) -> Option<()> {
290    let n = usize::from(r.u16(6)?).min(MAX_CHARS);
291    let bytes = r.slice(8, n)?;
292    let at = 8 + n.next_multiple_of(2);
293    let (y, x) = (r.i16(at)?, r.i16(at + 2)?);
294    let (text, _) = decode_bytes(bytes, charset(cv), &[]);
295    cv.text((f64::from(x), f64::from(y)), text, Vec::new(), 0, None);
296    Some(())
297}

299/// (x, y) from a Y-then-X pair stored at `off`.
300fn xy16(r: Bytes, off: usize) -> Option<P> {
301    Some((f64::from(r.i16(off + 2)?), f64::from(r.i16(off)?)))
302}

304/// Rectangle stored as bottom, right, top, left (the order MS-WMF uses for RECTANGLE and ELLIPSE).
305fn rect16(r: Bytes, off: usize) -> Option<[f64; 4]> {
306    let (b, rr, t, l) = (r.i16(off)?, r.i16(off + 2)?, r.i16(off + 4)?, r.i16(off + 6)?);
307    Some([f64::from(l), f64::from(t), f64::from(rr), f64::from(b)])
308}

310/// ARC, PIE and CHORD: end Y, end X, start Y, start X, then the bounding rectangle (bottom, right, top, left).
311fn arc16(r: Bytes) -> Option<([f64; 4], P, P)> {
312    let end = xy16(r, 6)?;
313    let start = xy16(r, 10)?;
314    let bounds = rect16(r, 14)?;
315    Some((bounds, start, end))
316}

318/// A count of 16-bit points, then the points (POLYLINE, POLYGON).
319fn counted_points(r: Bytes) -> Option<Vec<P>> {
320    let n = r.u16(6)? as usize;
321    r.tail(8)?.points(0, n, false)
322}

324/// POLYPOLYGON: polygon count, one 16-bit point count per polygon, then all points.
325fn polypolygon(r: Bytes) -> Option<Vec<geom::Op>> {
326    let n = r.u16(6)? as usize;
327    let mut counts = Vec::with_capacity(n);
328    for i in 0..n {
329        counts.push(r.u16(i.checked_mul(2)?.checked_add(8)?)? as usize);
330    }
331    let total = counts.iter().try_fold(0usize, |a, c| a.checked_add(*c))?;
332    if total > MAX_PATH_OPS {
333        return None;
334    }
335    let pts = r.tail(n.checked_mul(2)?.checked_add(8)?)?.points(0, total, false)?;
336    Some(geom::subpaths(&pts, &counts, true))
337}

339/// Whether a DIBBITBLT or DIBSTRETCHBLT record carries a bitmap: without one it is exactly
340/// `(function >> 8) + 3` words long (MS-WMF 2.3.1.2, 2.3.1.3).
341fn has_bitmap(func: u16, r: Bytes) -> bool {
342    r.0.len() / 2 > usize::from(func >> 8) + 3
343}

345/// META_DIBBITBLT: rop(4), ySrc, xSrc, then height, width, yDest, xDest and the DIB at 22. The
346/// bitmap-less form has a reserved word before the height, so its fields sit 2 bytes later; it, or
347/// a source-less raster operation, fills the destination.
348fn dib_bitblt(cv: &mut Canvas, r: Bytes) -> Option<()> {
349    let rop = r.u32(6)?;
350    if !has_bitmap(0x0940, r) {
351        cv.fill_rop([i16s(r, 22), i16s(r, 20), i16s(r, 18), i16s(r, 16)], rop);
352        return Some(());
353    }
354    let dst = [i16s(r, 20), i16s(r, 18), i16s(r, 16), i16s(r, 14)];
355    if Canvas::sourceless(rop) {
356        cv.fill_rop(dst, rop);
357        return Some(());
358    }
359    let src = [i16s(r, 12), i16s(r, 10), dst[2], dst[3]];
360    dib_blit(cv, r, 22, rop, dst, src)
361}

363/// META_DIBSTRETCHBLT: rop(4), srcH, srcW, ySrc, xSrc, then destH, destW, yDest, xDest and the DIB at
364/// 26. The bitmap-less form has a reserved word after xSrc, so the destination sits 2 bytes later;
365/// it, or a source-less raster operation, fills the destination.
366fn dib_stretchblt(cv: &mut Canvas, r: Bytes) -> Option<()> {
367    let rop = r.u32(6)?;
368    if !has_bitmap(0x0B41, r) {
369        cv.fill_rop([i16s(r, 26), i16s(r, 24), i16s(r, 22), i16s(r, 20)], rop);
370        return Some(());
371    }
372    let dst = [i16s(r, 24), i16s(r, 22), i16s(r, 20), i16s(r, 18)];
373    if Canvas::sourceless(rop) {
374        cv.fill_rop(dst, rop);
375        return Some(());
376    }
377    let src = [i16s(r, 16), i16s(r, 14), i16s(r, 12), i16s(r, 10)];
378    dib_blit(cv, r, 26, rop, dst, src)
379}

381/// A DIB at `at`, stretched from source rectangle `src` into `dst` (both in logical units, pixels for `src`).
382/// The raster operation `rop` decides how the pixels combine (see [`dib::prepare`]).
383fn dib_blit(cv: &mut Canvas, r: Bytes, at: usize, rop: u32, dst: [f64; 4], src: [f64; 4]) -> Option<()> {
384    let dib = r.tail(at)?;
385    let n = dib::info_len(dib)?;
386    let info = dib.slice(0, n)?;
387    let bits = dib.tail(n)?;
388    let (mut img, dst) = dib::decode_clip(Bytes(info), bits, false, src.map(|v| v as i64), dst)?;
389    dib::prepare(&mut img, dib::Blend::Rop(rop));
390    cv.bitmap(dst, img);
391    Some(())
