1//! Matroska CodecID → codec mapping (Matroska codec specifications, RFC 9559 §5.1.4.1.21 and the
2//! codec mappings registry).
3
4/// Codec of a track, derived from `CodecID` and `CodecPrivate`.
5///
6/// Configuration records are passed through as the raw bytes found in `CodecPrivate`, which for
7/// the ISO codecs are exactly the ISOBMFF configuration boxes' payloads (avcC, hvcC, av1C, ASC…).
8#[derive(Clone, Debug, PartialEq)]
9pub enum Codec {
10    /// `V_MPEG4/ISO/AVC`; payload is an `AVCDecoderConfigurationRecord` (avcC).
11    Avc { avcc: Vec<u8> },
12    /// `V_MPEGH/ISO/HEVC`; payload is an `HEVCDecoderConfigurationRecord` (hvcC).
13    Hevc { hvcc: Vec<u8> },
14    /// `V_VP8`.
15    Vp8,
16    /// `V_VP9`; `CodecPrivate` (optional) holds VP9 codec feature metadata.
17    Vp9 { private: Vec<u8> },
18    /// `V_AV1`; payload is an `AV1CodecConfigurationRecord` (av1C).
19    Av1 { av1c: Vec<u8> },
20    /// `V_APV`; payload is an `APVDecoderConfigurationRecord` (apvC), if present.
21    Apv { apvc: Vec<u8> },
22    /// `V_PRORES`; `fourcc` from `CodecPrivate` (e.g. `apcn`), if present.
23    ProRes { fourcc: Option<[u8; 4]> },
24    /// `V_MJPEG`, or `V_MS/VFW/FOURCC` with an `MJPG` compression FourCC.
25    Mjpeg,
26    /// `V_MS/VFW/FOURCC` with any other FourCC; `bitmap_info_header` is the whole `CodecPrivate`.
27    VfwFourcc { fourcc: [u8; 4], bitmap_info_header: Vec<u8> },
28    /// `A_AAC` (and the legacy `A_AAC/MPEG{2,4}/<profile>` IDs); `asc` is the
29    /// `AudioSpecificConfig` (synthesized for legacy IDs without `CodecPrivate`).
30    Aac { asc: Vec<u8> },
31    /// `A_OPUS`; `head` is the `OpusHead` identification header.
32    Opus { head: Vec<u8> },
33    /// `A_VORBIS`; the three Vorbis headers (identification, comment, setup), split from the
34    /// Xiph-laced `CodecPrivate`.
35    Vorbis { headers: Vec<Vec<u8>> },
36    /// `A_FLAC`; `private` is `fLaC` + metadata blocks (STREAMINFO first).
37    Flac { private: Vec<u8> },
38    /// `A_PCM/INT/LIT`, `A_PCM/INT/BIG`, `A_PCM/FLOAT/IEEE`. Bit depth comes from the Audio element.
39    Pcm { float: bool, big_endian: bool, bits: u16 },
40    /// `A_AC3`.
41    Ac3,
42    /// `A_EAC3`.
43    Eac3,
44    /// `A_MPEG/L3`.
45    Mp3,
46    /// `A_MPEG/L2`.
47    Mp2,
48    /// `A_MPEG/L1`.
49    Mp1,
50    /// `S_TEXT/UTF8` (SubRip-style plain text).
51    SubRip,
52    /// `S_TEXT/WEBVTT`, or WebM's `D_WEBVTT/{SUBTITLES,CAPTIONS,DESCRIPTIONS,METADATA}`.
53    WebVtt,
54    /// `S_TEXT/ASS` / `S_TEXT/SSA`; `header` is the script header from `CodecPrivate`.
55    Ass { header: Vec<u8>, ssa: bool },
56    /// Any other CodecID (kept verbatim along with `Track::codec_private`).
57    Other(String),
58}
59
60impl Codec {
61    /// Short codec name, matching FFmpeg's `codec_name` where one exists (useful for logs/UI).
62    pub fn name(&self) -> String {
63        match self {
64            Codec::Avc { .. } => "h264".into(),
65            Codec::Hevc { .. } => "hevc".into(),
66            Codec::Vp8 => "vp8".into(),
67            Codec::Vp9 { .. } => "vp9".into(),
68            Codec::Av1 { .. } => "av1".into(),
69            Codec::Apv { .. } => "apv".into(),
70            Codec::ProRes { .. } => "prores".into(),
71            Codec::Mjpeg => "mjpeg".into(),
72            Codec::VfwFourcc { fourcc, .. } => String::from_utf8_lossy(fourcc).trim().to_ascii_lowercase(),
73            Codec::Aac { .. } => "aac".into(),
74            Codec::Opus { .. } => "opus".into(),
75            Codec::Vorbis { .. } => "vorbis".into(),
76            Codec::Flac { .. } => "flac".into(),
77            Codec::Pcm { float, big_endian, bits } => {
78                let k = if *float {
79                    'f'
80                } else if *bits == 8 {
81                    'u'
82                } else {
83                    's'
84                };
85                if *bits == 8 { format!("pcm_{k}8") } else { format!("pcm_{k}{bits}{}", if *big_endian { "be" } else { "le" }) }
86            }
87            Codec::Ac3 => "ac3".into(),
88            Codec::Eac3 => "eac3".into(),
89            Codec::Mp3 => "mp3".into(),
90            Codec::Mp2 => "mp2".into(),
91            Codec::Mp1 => "mp1".into(),
92            Codec::SubRip => "subrip".into(),
93            Codec::WebVtt => "webvtt".into(),
94            Codec::Ass { ssa, .. } => if *ssa { "ssa" } else { "ass" }.into(),
95            Codec::Other(id) => id.clone(),
96        }
97    }
98}

100const AAC_RATES: [u32; 13] = [96000, 88200, 64000, 48000, 44100, 32000, 24000, 22050, 16000, 12000, 11025, 8000, 7350];
101
102/// Build a 2-byte (or 5-byte with SBR extension) AudioSpecificConfig for legacy `A_AAC/...` IDs.
103fn synth_asc(object_type: u8, rate: f64, out_rate: Option<f64>, channels: u64, sbr: bool) -> Vec<u8> {
104    let rate_index = |r: f64| AAC_RATES.iter().position(|&x| x as f64 == r).unwrap_or(4) as u8;
105    let ri = rate_index(rate);
106    let ch = channels.min(7) as u8;
107    let mut v = vec![(object_type << 3) | (ri >> 1), ((ri & 1) << 7) | (ch << 3)];
108    if sbr {
109        // backward-compatible explicit signaling: sync 0x2b7, SBR object type 5, flag, ext rate index
110        let ext = rate_index(out_rate.unwrap_or(rate * 2.0));
111        v.push(0x56);
112        v.push(0xE5);
113        v.push(0x80 | (ext << 3));
114    }
115    v
116}

118/// Split a Xiph-laced `CodecPrivate` (Vorbis/Theora headers).
119fn split_xiph(p: &[u8]) -> Vec<Vec<u8>> {
120    let Some(&n) = p.first() else { return Vec::new() };
121    let mut pos = 1;
122    let mut sizes = Vec::new();
123    for _ in 0..n {
124        let mut s = 0usize;
125        loop {
126            let Some(&b) = p.get(pos) else { return Vec::new() };
127            pos += 1;
128            s += b as usize;
129            if b != 255 {
130                break;
131            }
132        }
133        sizes.push(s);
134    }
135    let mut out = Vec::new();
136    for s in sizes {
137        let Some(h) = p.get(pos..pos + s) else { return Vec::new() };
138        out.push(h.to_vec());
139        pos += s;
140    }
141    out.push(p[pos.min(p.len())..].to_vec());
142    out
143}

145/// Map a CodecID + CodecPrivate (+ audio params for legacy AAC / PCM) to a [`Codec`].
146pub(crate) fn map_codec(id: &str, private: &[u8], audio: Option<&crate::craft::media::matroska::AudioInfo>) -> Codec {
147    let p = private.to_vec();
148    match id {
149        "V_MPEG4/ISO/AVC" => Codec::Avc { avcc: p },
150        "V_MPEGH/ISO/HEVC" => Codec::Hevc { hvcc: p },
151        "V_VP8" => Codec::Vp8,
152        "V_VP9" => Codec::Vp9 { private: p },
153        "V_AV1" => Codec::Av1 { av1c: p },
154        "V_APV" => Codec::Apv { apvc: p },
155        "V_PRORES" => Codec::ProRes { fourcc: private.get(..4).and_then(|f| f.try_into().ok()) },
156        "V_MJPEG" => Codec::Mjpeg,
157        "V_MS/VFW/FOURCC" => {
158            let fourcc: [u8; 4] = private.get(16..20).and_then(|f| f.try_into().ok()).unwrap_or(*b"    ");
159            match &fourcc {
160                b"MJPG" | b"mjpg" | b"AVRn" | b"AVDJ" => Codec::Mjpeg,
161                b"H264" | b"h264" | b"avc1" | b"X264" | b"x264" => Codec::Avc { avcc: Vec::new() },
162                b"apch" | b"apcn" | b"apcs" | b"apco" | b"ap4h" | b"ap4x" => Codec::ProRes { fourcc: Some(fourcc) },
163                b"apv1" | b"APV1" => Codec::Apv { apvc: Vec::new() },
164                _ => Codec::VfwFourcc { fourcc, bitmap_info_header: p },
165            }
166        }
167        "A_AAC" => Codec::Aac { asc: p },
168        "A_OPUS" => Codec::Opus { head: p },
169        "A_VORBIS" => Codec::Vorbis { headers: split_xiph(private) },
170        "A_FLAC" => Codec::Flac { private: p },
171        "A_PCM/INT/LIT" | "A_PCM/INT/BIG" | "A_PCM/FLOAT/IEEE" => {
172            let bits = audio.and_then(|a| a.bit_depth).unwrap_or(if id == "A_PCM/FLOAT/IEEE" { 32 } else { 16 }) as u16;
173            Codec::Pcm { float: id == "A_PCM/FLOAT/IEEE", big_endian: id == "A_PCM/INT/BIG", bits }
174        }
175        "A_AC3" | "A_AC3/BSID9" | "A_AC3/BSID10" => Codec::Ac3,
176        "A_EAC3" => Codec::Eac3,
177        "A_MPEG/L3" => Codec::Mp3,
178        "A_MPEG/L2" => Codec::Mp2,
179        "A_MPEG/L1" => Codec::Mp1,
180        "S_TEXT/UTF8" => Codec::SubRip,
181        "S_TEXT/WEBVTT" | "D_WEBVTT/SUBTITLES" | "D_WEBVTT/CAPTIONS" | "D_WEBVTT/DESCRIPTIONS" | "D_WEBVTT/METADATA" => Codec::WebVtt,
182        "S_TEXT/ASS" | "S_ASS" => Codec::Ass { header: p, ssa: false },
183        "S_TEXT/SSA" | "S_SSA" => Codec::Ass { header: p, ssa: true },
184        _ if id.starts_with("A_AAC/") => {
185            if !private.is_empty() {
186                return Codec::Aac { asc: p };
187            }
188            let (rate, out_rate, ch) = audio.map_or((48000.0, None, 2), |a| (a.sampling_frequency, a.output_sampling_frequency, a.channels));
189            let object_type = if id.ends_with("/MAIN") {
190                1
191            } else if id.ends_with("/SSR") {
192                3
193            } else if id.ends_with("/LTP") {
194                4
195            } else {
196                2
197            };
198            let sbr = id.ends_with("/SBR");
199            Codec::Aac { asc: synth_asc(object_type, rate, out_rate, ch, sbr) }
200        }
201        _ => Codec::Other(id.to_string()),
202    }
203}

205#[cfg(any())]
206mod tests {
207    use super::*;
208
209    #[cfg(any())]
210    fn mapping() {
211        assert_eq!(map_codec("V_MPEG4/ISO/AVC", &[1, 2], None), Codec::Avc { avcc: vec![1, 2] });
212        assert_eq!(map_codec("V_APV", &[1, 2, 3], None), Codec::Apv { apvc: vec![1, 2, 3] });
213        assert_eq!(map_codec("V_APV", &[], None).name(), "apv");
214        assert_eq!(map_codec("V_PRORES", b"apcn", None), Codec::ProRes { fourcc: Some(*b"apcn") });
215        let mut bih = vec![0u8; 40];
216        bih[16..20].copy_from_slice(b"MJPG");
217        assert_eq!(map_codec("V_MS/VFW/FOURCC", &bih, None), Codec::Mjpeg);
218        bih[16..20].copy_from_slice(b"FFV1");
219        assert!(matches!(map_codec("V_MS/VFW/FOURCC", &bih, None), Codec::VfwFourcc { fourcc, .. } if &fourcc == b"FFV1"));
220        let a = crate::craft::media::matroska::AudioInfo {
221            sampling_frequency: 44100.0,
222            channels: 2,
223            bit_depth: Some(24),
224            output_sampling_frequency: None,
225        };
226        assert_eq!(map_codec("A_PCM/INT/LIT", &[], Some(&a)).name(), "pcm_s24le");
227        assert_eq!(map_codec("A_PCM/FLOAT/IEEE", &[], Some(&a)), Codec::Pcm { float: true, big_endian: false, bits: 24 });
228        // legacy AAC LC 44.1k stereo → 0x12 0x10
229        assert_eq!(map_codec("A_AAC/MPEG4/LC", &[], Some(&a)), Codec::Aac { asc: vec![0x12, 0x10] });
230        assert_eq!(map_codec("S_TEXT/UTF8", &[], None).name(), "subrip");
231        assert_eq!(map_codec("X_WHATEVER", &[], None), Codec::Other("X_WHATEVER".into()));
232    }
233
234    #[cfg(any())]
235    fn xiph_split() {
236        let mut p = vec![2u8, 3, 2];
237        p.extend_from_slice(&[1, 1, 1, 2, 2, 3, 3, 3, 3]);
238        let h = split_xiph(&p);
239        assert_eq!(h, vec![vec![1, 1, 1], vec![2, 2], vec![3, 3, 3, 3]]);
240    }
