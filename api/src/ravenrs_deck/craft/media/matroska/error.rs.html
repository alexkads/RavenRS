1use std::fmt;
2use std::io;
3
4/// Errors produced by the demuxer.
5#[derive(Debug)]
6pub enum Error {
7    /// Underlying I/O failure.
8    Io(io::Error),
9    /// Structurally invalid data that could not be recovered from.
10    Invalid(String),
11    /// Valid but unsupported feature (e.g. zlib-compressed or encrypted tracks when reading data).
12    Unsupported(String),
13    /// Not an EBML file, or an EBML file whose DocType is not `matroska`/`webm`.
14    NotMatroska(String),
15    /// A track index was out of range.
16    NoSuchTrack(usize),
17    /// A sample index was out of range.
18    NoSuchSample(usize),
19    /// The file describes more than the demuxer's limits allow (frames, tracks).
20    TooLarge(&'static str),
21}
22
23impl fmt::Display for Error {
24    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
25        match self {
26            Error::Io(e) => write!(f, "I/O error: {e}"),
27            Error::Invalid(s) => write!(f, "invalid data: {s}"),
28            Error::Unsupported(s) => write!(f, "unsupported: {s}"),
29            Error::NotMatroska(s) => write!(f, "not a Matroska/WebM file: {s}"),
30            Error::NoSuchTrack(i) => write!(f, "track index {i} out of range"),
31            Error::NoSuchSample(i) => write!(f, "sample index {i} out of range"),
32            Error::TooLarge(s) => write!(f, "size limit exceeded: {s}"),
33        }
34    }
35}
36
37impl std::error::Error for Error {
38    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
39        match self {
40            Error::Io(e) => Some(e),
41            _ => None,
42        }
43    }
44}
45
46impl From<io::Error> for Error {
47    fn from(e: io::Error) -> Self {
48        Error::Io(e)
49    }
50}
51
52pub type Result<T> = std::result::Result<T, Error>;
53
54pub(crate) fn invalid<T>(msg: impl Into<String>) -> Result<T> {
55    Err(Error::Invalid(msg.into()))
56}
