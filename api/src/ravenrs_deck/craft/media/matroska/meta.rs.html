24// A malformed Matroska file may contain arbitrarily deep ChapterAtom and SimpleTag chains.
25const MAX_META_DEPTH: usize = 64;
26/// Longest header-stripping prefix accepted. It is copied in front of every frame, laced
27/// zero-length frames included, so its length multiplies what a small file can make a reader
28/// produce. Real files strip a few bytes of codec header; a longer one is treated as an
29/// unsupported compression and its frames are not reconstructed.
30const MAX_STRIPPED_HEADER: usize = 256;
31/// Most `ContentEncoding` entries a track may list. A muxer writes one or two (compression,
32/// encryption); readers check the whole chain for every frame they return, so a longer chain
33/// would multiply that work by the frame count. A track with more is treated as having an
34/// unsupported compression, and its frames are returned as stored.
35const MAX_CONTENT_ENCODINGS: usize = 16;
36/// Largest accepted `TimestampScale` (about 4.3 s per tick). It keeps tick to nanosecond
37/// arithmetic on 64-bit timestamps within `i128`.
38const MAX_TIMESTAMP_SCALE: u64 = 1 << 32;
39
40/// `Info` element.
41#[derive(Clone, Debug, PartialEq)]
42pub struct SegmentInfo {
43    /// Nanoseconds per timestamp tick (default 1 000 000 = 1 ms). A file's value above 2^32 is
44    /// ignored.
45    pub timestamp_scale: u64,
46    /// `Duration` in timestamp ticks (float), if present.
47    pub duration: Option<f64>,
48    pub title: Option<String>,
49    pub muxing_app: String,
50    pub writing_app: String,
51    /// `DateUTC`: nanoseconds since 2001-01-01T00:00:00 UTC.
52    pub date_utc: Option<i64>,
53    pub segment_uuid: Option<[u8; 16]>,
54}

65            segment_uuid: None,

67    }

202pub(crate) fn parse_info(d: &[u8]) -> SegmentInfo {
203    let mut info = SegmentInfo::default();
204    for e in Children::new(d) {
205        match e.id {
206            TIMESTAMP_SCALE => {
207                let v = uint(e.data);
208                if v > 0 && v <= MAX_TIMESTAMP_SCALE {
209                    info.timestamp_scale = v;
210                }
211            }
212            DURATION => info.duration = Some(float(e.data)).filter(|v| v.is_finite() && *v >= 0.0),
213            TITLE => info.title = Some(string(e.data)),
214            MUXING_APP => info.muxing_app = string(e.data),
215            WRITING_APP => info.writing_app = string(e.data),
216            DATE_UTC => info.date_utc = Some(int(e.data)),
217            SEGMENT_UUID => info.segment_uuid = e.data.try_into().ok(),
218            _ => {}
219        }
220    }
221    info
222}

224pub(crate) fn parse_seekhead(d: &[u8]) -> Vec<SeekEntry> {
225    let mut out = Vec::new();
226    for s in Children::new(d).filter(|e| e.id == SEEK) {
227        let (mut id, mut pos) = (None, None);
228        for e in Children::new(s.data) {
229            match e.id {
230                SEEK_ID => id = Some(uint(e.data) as u32),
231                SEEK_POSITION => pos = Some(uint(e.data)),
232                _ => {}
233            }
234        }
235        if let (Some(id), Some(position)) = (id, pos) {
236            out.push(SeekEntry { id, position });
237        }
238    }
239    out
240}

242pub(crate) fn parse_cues(d: &[u8], out: &mut Vec<CuePoint>) {
243    for cp in Children::new(d).filter(|e| e.id == CUE_POINT) {
244        let mut time = None;
245        let mut positions = Vec::new();
246        for e in Children::new(cp.data) {
247            match e.id {
248                CUE_TIME => time = Some(uint(e.data)),
249                CUE_TRACK_POSITIONS => {
250                    let mut p = CuePosition { track: 0, cluster_position: u64::MAX, relative_position: None, block_number: None, duration: None };
251                    for c in Children::new(e.data) {
252                        match c.id {
253                            CUE_TRACK => p.track = uint(c.data),
254                            CUE_CLUSTER_POSITION => p.cluster_position = uint(c.data),
255                            CUE_RELATIVE_POSITION => p.relative_position = Some(uint(c.data)),
256                            CUE_BLOCK_NUMBER => p.block_number = Some(uint(c.data)),
257                            CUE_DURATION => p.duration = Some(uint(c.data)),
258                            _ => {}
259                        }
260                    }
261                    if p.track != 0 && p.cluster_position != u64::MAX {
262                        positions.push(p);
263                    }
264                }
265                _ => {}
266            }
267        }
268        if let Some(time) = time
269            && !positions.is_empty()
270        {
271            out.push(CuePoint { time, positions });
272        }
273    }
274}

276fn parse_chapter(d: &[u8], depth: usize) -> Chapter {
277    let mut c =
278        Chapter { uid: 0, string_uid: None, start_ns: 0, end_ns: None, hidden: false, enabled: true, displays: Vec::new(), children: Vec::new() };
279    for e in Children::new(d) {
280        match e.id {
281            CHAPTER_UID => c.uid = uint(e.data),
282            CHAPTER_STRING_UID => c.string_uid = Some(string(e.data)),
283            CHAPTER_TIME_START => c.start_ns = uint(e.data),
284            CHAPTER_TIME_END => c.end_ns = Some(uint(e.data)),
285            CHAPTER_FLAG_HIDDEN => c.hidden = uint(e.data) != 0,
286            CHAPTER_FLAG_ENABLED => c.enabled = uint(e.data) != 0,
287            CHAPTER_DISPLAY => {
288                let mut disp = ChapterDisplay::default();
289                for x in Children::new(e.data) {
290                    match x.id {
291                        CHAP_STRING => disp.string = string(x.data),
292                        CHAP_LANGUAGE => disp.languages.push(string(x.data)),
293                        CHAP_LANGUAGE_BCP47 => disp.languages_bcp47.push(string(x.data)),
294                        CHAP_COUNTRY => disp.countries.push(string(x.data)),
295                        _ => {}
296                    }
297                }
298                if disp.languages.is_empty() {
299                    disp.languages.push("eng".into());
300                }
301                c.displays.push(disp);
302            }
303            CHAPTER_ATOM if depth < MAX_META_DEPTH => c.children.push(parse_chapter(e.data, depth + 1)),
304            _ => {}
305        }
306    }
307    c
308}

310pub(crate) fn parse_chapters(d: &[u8], out: &mut Vec<Edition>) {
311    for ed in Children::new(d).filter(|e| e.id == EDITION_ENTRY) {
312        let mut edition = Edition::default();
313        for e in Children::new(ed.data) {
314            match e.id {
315                EDITION_UID => edition.uid = Some(uint(e.data)),
316                EDITION_FLAG_HIDDEN => edition.hidden = uint(e.data) != 0,
317                EDITION_FLAG_DEFAULT => edition.default = uint(e.data) != 0,
318                EDITION_FLAG_ORDERED => edition.ordered = uint(e.data) != 0,
319                CHAPTER_ATOM => edition.chapters.push(parse_chapter(e.data, 0)),
320                _ => {}
321            }
322        }
323        out.push(edition);
324    }

343pub(crate) fn parse_tags(d: &[u8], out: &mut Vec<Tag>) {
344    for tg in Children::new(d).filter(|e| e.id == TAG) {
345        let mut tag = Tag::default();

347            match e.id {
348                TARGETS => {

350                        match x.id {
351                            TARGET_TYPE_VALUE => tag.targets.type_value = uint(x.data),
352                            TARGET_TYPE => tag.targets.type_name = Some(string(x.data)),
353                            TAG_TRACK_UID => tag.targets.track_uids.push(uint(x.data)),
354                            TAG_EDITION_UID => tag.targets.edition_uids.push(uint(x.data)),
355                            TAG_CHAPTER_UID => tag.targets.chapter_uids.push(uint(x.data)),
356                            TAG_ATTACHMENT_UID => tag.targets.attachment_uids.push(uint(x.data)),
357                            _ => {}
358                        }
359                    }
360                }
361                SIMPLE_TAG => tag.simple.push(parse_simple_tag(e.data, 0)),
362                _ => {}
363            }
364        }
365        out.push(tag);
366    }

466fn parse_encodings(d: &[u8]) -> Vec<ContentEncoding> {
467    if Children::new(d).filter(|e| e.id == CONTENT_ENCODING).nth(MAX_CONTENT_ENCODINGS).is_some() {
468        return vec![ContentEncoding::Compression { algo: u64::MAX, settings: Vec::new(), scope: 1 }];
469    }
470    let mut list: Vec<(u64, ContentEncoding)> = Vec::new();
471    for ce in Children::new(d).filter(|e| e.id == CONTENT_ENCODING) {
472        let (mut order, mut scope, mut ty) = (0u64, 1u64, 0u64);
473        let mut comp: Option<(u64, Vec<u8>)> = None;
474        for e in Children::new(ce.data) {
475            match e.id {
476                CONTENT_ENCODING_ORDER => order = uint(e.data),
477                CONTENT_ENCODING_SCOPE => scope = uint(e.data),
478                CONTENT_ENCODING_TYPE => ty = uint(e.data),
479                CONTENT_COMPRESSION => {
480                    let (mut algo, mut settings) = (0u64, Vec::new());
481                    for x in Children::new(e.data) {
482                        match x.id {
483                            CONTENT_COMP_ALGO => algo = uint(x.data),
484                            CONTENT_COMP_SETTINGS => settings = x.data.to_vec(),
485                            _ => {}
486                        }
487                    }
488                    comp = Some((algo, settings));
489                }
490                _ => {}
491            }
492        }
493        let enc = if ty == 1 {
494            ContentEncoding::Encryption { scope }
495        } else {
496            let (algo, settings) = comp.unwrap_or((0, Vec::new()));
497            if algo == 3 && settings.len() <= MAX_STRIPPED_HEADER {
498                ContentEncoding::HeaderStripping { bytes: settings, scope }
499            } else {
500                ContentEncoding::Compression { algo, settings, scope }
501            }
502        };
503        list.push((order, enc));
504    }
505    // highest order is applied last on muxing → first to undo; keep decode order
506    list.sort_by_key(|e| std::cmp::Reverse(e.0));
507    list.into_iter().map(|e| e.1).collect()
508}

510pub(crate) fn parse_tracks(d: &[u8], timestamp_scale: u64) -> Vec<Track> {
511    let g = gcd(timestamp_scale, 1_000_000_000);
512    let timebase = (timestamp_scale / g, 1_000_000_000 / g);
513    let mut out = Vec::new();
514    for te in Children::new(d).filter(|e| e.id == TRACK_ENTRY) {
515        let mut t = Track {
516            number: 0,
517            uid: 0,
518            kind: TrackKind::Other(0),
519            enabled: true,
520            default: true,
521            forced: false,
522            hearing_impaired: false,
523            lacing: true,
524            default_duration_ns: None,
525            name: String::new(),
526            language: "eng".into(),
527            language_bcp47: None,
528            codec_id: String::new(),
529            codec_private: Vec::new(),
530            codec_name: String::new(),
531            codec: crate::craft::media::matroska::Codec::Other(String::new()),
532            codec_delay_ns: 0,
533            seek_pre_roll_ns: 0,
534            video: None,
535            audio: None,
536            encodings: Vec::new(),
537            timebase,
538            samples: Vec::new(),
539            sync: Vec::new(),
540        };
541        for e in Children::new(te.data) {
542            match e.id {
543                TRACK_NUMBER => t.number = uint(e.data),
544                TRACK_UID => t.uid = uint(e.data),
545                TRACK_TYPE => t.kind = TrackKind::from_type(uint(e.data)),
546                FLAG_ENABLED => t.enabled = uint(e.data) != 0,
547                FLAG_DEFAULT => t.default = uint(e.data) != 0,
548                FLAG_FORCED => t.forced = uint(e.data) != 0,
549                FLAG_HEARING_IMPAIRED => t.hearing_impaired = uint(e.data) != 0,
550                FLAG_LACING => t.lacing = uint(e.data) != 0,
551                DEFAULT_DURATION => t.default_duration_ns = Some(uint(e.data)).filter(|&v| v > 0),
552                NAME => t.name = string(e.data),
553                LANGUAGE => t.language = string(e.data),
554                LANGUAGE_BCP47 => t.language_bcp47 = Some(string(e.data)),
555                CODEC_ID => t.codec_id = string(e.data),
556                CODEC_PRIVATE => t.codec_private = e.data.to_vec(),
557                CODEC_NAME => t.codec_name = string(e.data),
558                CODEC_DELAY => t.codec_delay_ns = uint(e.data),
559                SEEK_PRE_ROLL => t.seek_pre_roll_ns = uint(e.data),
560                VIDEO => t.video = Some(parse_video(e.data)),
561                AUDIO => t.audio = Some(parse_audio(e.data)),
562                CONTENT_ENCODINGS => t.encodings = parse_encodings(e.data),
563                _ => {}
564            }
565        }
566        if t.number == 0 {
567            continue;
568        }
569        if t.kind == TrackKind::Audio && t.audio.is_none() {
570            t.audio = Some(AudioInfo::default());
571        }
572        // CodecPrivate may itself be header-stripped (scope bit 2 = private data)
573        // Prepends in list order, built in one pass: a hostile entry can list millions.
574        let mut prefix = Vec::new();
575        for enc in t.encodings.iter().rev() {
576            if let ContentEncoding::HeaderStripping { bytes, scope } = enc
577                && scope & 2 != 0
578            {
579                prefix.extend_from_slice(bytes);
580            }
581        }
582        if !prefix.is_empty() {
583            prefix.extend_from_slice(&t.codec_private);
584            t.codec_private = prefix;
585        }
586        t.codec = map_codec(&t.codec_id, &t.codec_private, t.audio.as_ref());
587        out.push(t);
588    }
589    out
590}

597    fn header_stripping(bytes: &[u8]) -> Vec<ContentEncoding> {
598        let mut comp = Vec::new();
599        el_uint(&mut comp, CONTENT_COMP_ALGO, 3);
600        el(&mut comp, CONTENT_COMP_SETTINGS, bytes);
601        let mut enc = Vec::new();
602        el(&mut enc, CONTENT_COMPRESSION, &comp);
603        let mut encs = Vec::new();
604        el(&mut encs, CONTENT_ENCODING, &enc);
605        parse_encodings(&encs)
606    }

608    #[cfg(any())]
609    fn a_long_stripped_header_is_not_put_in_front_of_frames() {
610        let kept = header_stripping(&[7; MAX_STRIPPED_HEADER]);
611        assert!(matches!(&kept[..], [ContentEncoding::HeaderStripping { bytes, .. }] if bytes.len() == MAX_STRIPPED_HEADER));
612        let refused = header_stripping(&[7; MAX_STRIPPED_HEADER + 1]);
613        assert!(matches!(&refused[..], [ContentEncoding::Compression { algo: 3, .. }]), "{refused:?}");
614    }

616    fn track_with_private(private: &[u8], strips: &[(&[u8], u64)]) -> Track {
617        let mut encs = Vec::new();
618        for &(bytes, scope) in strips {
619            let mut comp = Vec::new();
620            el_uint(&mut comp, CONTENT_COMP_ALGO, 3);
621            el(&mut comp, CONTENT_COMP_SETTINGS, bytes);
622            let mut enc = Vec::new();
623            el_uint(&mut enc, CONTENT_ENCODING_SCOPE, scope);
624            el(&mut enc, CONTENT_COMPRESSION, &comp);
625            el(&mut encs, CONTENT_ENCODING, &enc);
626        }
627        let mut te = Vec::new();
628        el_uint(&mut te, TRACK_NUMBER, 1);
629        el(&mut te, CODEC_ID, b"V_TEST");
630        el(&mut te, CODEC_PRIVATE, private);
631        el(&mut te, CONTENT_ENCODINGS, &encs);
632        let mut tracks = Vec::new();
633        el(&mut tracks, TRACK_ENTRY, &te);
634        let mut parsed = parse_tracks(&tracks, 1_000_000);
635        assert_eq!(parsed.len(), 1);
636        parsed.remove(0)
637    }
