216/// The layout of uncompressed PCM samples.
217#[non_exhaustive]
218#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
219pub struct PcmFormat {
220    /// Bits per sample (8, 16, 24, 32 for integers; 32 or 64 for floats).
221    pub bits: u16,
222    /// IEEE float samples.
223    pub float: bool,
224    /// Big-endian byte order.
225    pub big_endian: bool,
226    /// Signed integers (8-bit WAV and `raw ` QuickTime PCM are unsigned).
227    pub signed: bool,
228}
229
230impl PcmFormat {
231    /// Integer samples of `bits` bits.
232    pub fn int(bits: u16, signed: bool, big_endian: bool) -> PcmFormat {
233        PcmFormat { bits, float: false, big_endian, signed }
234    }
235
236    /// IEEE float samples of `bits` bits (32 or 64).
237    pub fn float(bits: u16, big_endian: bool) -> PcmFormat {
238        PcmFormat { bits, float: true, big_endian, signed: true }
239    }
240}

242/// What a decoder needs to know about a track, whatever container it came from.
243///
244/// `extradata` is the codec configuration in Matroska's `CodecPrivate` form, which ffmpeg's
245/// extradata also follows:
246///
247/// | Codec | `extradata` |
248/// |---|---|
249/// | AAC | `AudioSpecificConfig` |
250/// | Opus | the `OpusHead` identification header (RFC 7845 §5.1) |
251/// | Vorbis, Theora | the three headers, Xiph-laced |
252/// | Speex | the Speex header |
253/// | FLAC | `fLaC` followed by the metadata blocks (STREAMINFO first) |
254/// | ALAC | the magic cookie (`ALACSpecificConfig`) |
255/// | AC-3 / E-AC-3 | the MP4 `dac3` / `dec3` payload, or empty |
256/// | MP1/MP2/MP3, PCM | empty |
257/// | H.264 / HEVC / AV1 | the `avcC` / `hvcC` / `av1C` record |
258/// | VP9 | the MP4 `vpcC` payload, or Matroska's codec feature metadata |
259/// | APV | the `apvC` record |
260///
261/// `sample_rate` and `channels` are what the container declares (for HE-AAC with implicit
262/// signalling, usually the core rate), except that Opus is always 48 kHz, the rate it decodes at;
263/// `width` and `height` are the coded picture size of a video
264/// track. `tag` is the container's four-character code for the codec when it has one (the MP4
265/// sample entry format, Matroska's ProRes or VfW FourCC), which is how a ProRes profile (`apcn`,
266/// `ap4h`, …) or a DNx variant is told apart.
267///
268/// `#[non_exhaustive]`: build it with [`CodecParams::new`] and the `with_*` methods.
269#[non_exhaustive]
270#[derive(Debug, Clone, PartialEq, Eq)]
271pub struct CodecParams {
272    /// The codec.
273    pub codec: CodecId,
274    /// The codec configuration (see the table above).
275    pub extradata: Vec<u8>,
276    /// The container's four-character code for the codec, if it has one.
277    pub tag: Option<[u8; 4]>,
278    /// Audio: samples per second.
279    pub sample_rate: Option<u32>,
280    /// Audio: channel count.
281    pub channels: Option<u16>,
282    /// Audio: the sample layout of uncompressed PCM.
283    pub pcm: Option<PcmFormat>,
284    /// Video: coded picture width in pixels.
285    pub width: Option<u32>,
286    /// Video: coded picture height in pixels.
287    pub height: Option<u32>,
288}
289
290impl CodecParams {
291    /// Parameters for `codec`, with nothing else known yet.
292    pub fn new(codec: CodecId) -> CodecParams {
293        CodecParams { codec, extradata: Vec::new(), tag: None, sample_rate: None, channels: None, pcm: None, width: None, height: None }
294    }
295
296    /// Sets `tag`.
297    pub fn with_tag(mut self, tag: [u8; 4]) -> CodecParams {
298        self.tag = Some(tag);
299        self
300    }
301
302    /// The coded picture size; a zero dimension leaves both unknown.
303    pub fn with_size(mut self, width: u32, height: u32) -> CodecParams {
304        (self.width, self.height) = if width > 0 && height > 0 { (Some(width), Some(height)) } else { (None, None) };
305        self
306    }
307
308    /// Sets `extradata`.
309    pub fn with_extradata(mut self, extradata: impl Into<Vec<u8>>) -> CodecParams {
310        self.extradata = extradata.into();
311        self
312    }
313
314    /// Sets `sample_rate`; 0 leaves it unknown.
315    pub fn with_sample_rate(mut self, sample_rate: u32) -> CodecParams {
316        self.sample_rate = (sample_rate > 0).then_some(sample_rate);
317        self
318    }
319
320    /// Sets `channels`; 0 leaves it unknown.
321    pub fn with_channels(mut self, channels: u16) -> CodecParams {
322        self.channels = (channels > 0).then_some(channels);
323        self
324    }
325
326    /// Sets `pcm`.
327    pub fn with_pcm(mut self, pcm: PcmFormat) -> CodecParams {
328        self.pcm = Some(pcm);
329        self
330    }
331}

333/// Xiph lacing of `packets` (Matroska `CodecPrivate` for Vorbis): the count less one, each
334/// length but the last as a run of 255s and a remainder, then the packets. `None` for no packets
335/// or more than 256, which the one-byte count cannot hold.
336pub fn xiph_lace(packets: &[Vec<u8>]) -> Option<Vec<u8>> {
337    let mut out = vec![u8::try_from(packets.len().checked_sub(1)?).ok()?];
338    for p in packets.iter().take(packets.len().saturating_sub(1)) {
339        let mut n = p.len();
340        while n >= 255 {
341            out.push(255);
342            n -= 255;
343        }
344        out.push(n as u8);
345    }
346    for p in packets {
347        out.extend_from_slice(p);
348    }
349    Some(out)
350}

352#[cfg(any())]
353mod tests {
354    use super::*;
355
356    #[cfg(any())]
357    fn audio_and_video_are_disjoint() {
358        for c in [CodecId::Aac, CodecId::Opus, CodecId::Pcm, CodecId::Eac3, CodecId::Speex] {
359            assert!(c.is_audio() && !c.is_video(), "{c:?}");
360        }
361        for c in [CodecId::H264, CodecId::Av1, CodecId::ProRes, CodecId::Theora, CodecId::Dnxhd, CodecId::Apv] {
362            assert!(c.is_video() && !c.is_audio(), "{c:?}");
363        }
364        assert!(!CodecId::Unknown.is_audio() && !CodecId::Unknown.is_video());
365    }
366
367    #[cfg(any())]
368    fn params_builder() {
369        let p = CodecParams::new(CodecId::Aac).with_extradata([0x12, 0x10]).with_sample_rate(48_000).with_channels(2).with_sample_rate(0);
370        assert_eq!((p.codec, p.extradata.as_slice(), p.sample_rate, p.channels), (CodecId::Aac, &[0x12, 0x10][..], None, Some(2)));
371        let v = CodecParams::new(CodecId::ProRes).with_tag(*b"apcn").with_size(1920, 1080);
372        assert_eq!((v.tag, v.width, v.height), (Some(*b"apcn"), Some(1920), Some(1080)));
373        assert_eq!(v.with_size(0, 1080).width, None);
374    }
375
376    #[cfg(any())]
377    fn xiph_lacing() {
378        let laced = xiph_lace(&[vec![1; 30], vec![2; 300], vec![3; 5]]).unwrap();
379        assert_eq!(&laced[..4], &[2, 30, 255, 45]);
380        assert_eq!(laced.len(), 4 + 335);
381        assert_eq!(xiph_lace(&[]), None);
382        assert_eq!(xiph_lace(&vec![vec![0]; 257]), None);
383        assert!(xiph_lace(&vec![vec![0]; 256]).is_some());
384    }
