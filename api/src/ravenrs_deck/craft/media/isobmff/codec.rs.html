55/// SMPTE ST 2086 mastering display metadata, in the units of the `mdcv` box and the H.264/HEVC
56/// SEI: chromaticities in 0.00002 steps (order G, B, R as the SEI), luminance in 0.0001 cd/m².
57#[derive(Clone, Copy, Debug, PartialEq, Eq)]
58pub struct MasteringDisplay {
59    pub primaries: [(u16, u16); 3],
60    pub white_point: (u16, u16),
61    pub max_luminance: u32,
62    pub min_luminance: u32,
63}
64
65impl MasteringDisplay {
66    /// BT.2020 primaries, D65, with the given peak and black in cd/m².
67    pub fn bt2020(max_nits: f64, min_nits: f64) -> Self {
68        let q = |v: f64| (v / 0.00002).round() as u16;
69        MasteringDisplay {
70            primaries: [(q(0.170), q(0.797)), (q(0.131), q(0.046)), (q(0.708), q(0.292))],
71            white_point: (q(0.3127), q(0.3290)),
72            max_luminance: (max_nits * 10_000.0).round() as u32,
73            min_luminance: (min_nits * 10_000.0).round() as u32,
74        }
75    }
76    pub fn max_nits(&self) -> f64 {
77        self.max_luminance as f64 / 10_000.0
78    }
79    /// The 24-byte payload shared by `mdcv` and the SEI message.
80    pub fn to_bytes(&self) -> Vec<u8> {
81        let mut v = Vec::with_capacity(24);
82        for (x, y) in self.primaries {
83            v.extend_from_slice(&x.to_be_bytes());
84            v.extend_from_slice(&y.to_be_bytes());
85        }
86        v.extend_from_slice(&self.white_point.0.to_be_bytes());
87        v.extend_from_slice(&self.white_point.1.to_be_bytes());
88        v.extend_from_slice(&self.max_luminance.to_be_bytes());
89        v.extend_from_slice(&self.min_luminance.to_be_bytes());
90        v
91    }
92    pub fn parse(p: &[u8]) -> Option<Self> {
93        if p.len() < 24 {
94            return None;
95        }
96        let u16_at = |i: usize| u16::from_be_bytes([p[i], p[i + 1]]);
97        let u32_at = |i: usize| u32::from_be_bytes([p[i], p[i + 1], p[i + 2], p[i + 3]]);
98        Some(MasteringDisplay {
99            primaries: [(u16_at(0), u16_at(2)), (u16_at(4), u16_at(6)), (u16_at(8), u16_at(10))],
100            white_point: (u16_at(12), u16_at(14)),
101            max_luminance: u32_at(16),
102            min_luminance: u32_at(20),
103        })
104    }

287/// `avcC` — AVCDecoderConfigurationRecord (ISO/IEC 14496-15 §5.3.3).
288#[derive(Clone, Debug, PartialEq, Eq, Default)]
289pub struct AvcConfig {
290    pub profile: u8,
291    pub compatibility: u8,
292    pub level: u8,
293    /// NAL length field size in bytes (1, 2 or 4).
294    pub length_size: u8,
295    pub sps: Vec<Vec<u8>>,
296    pub pps: Vec<Vec<u8>>,
297    /// Bytes after the PPS list (high-profile extension: chroma format, bit depths, SPS-ext).
298    pub ext: Vec<u8>,
299}
300
301impl AvcConfig {
302    /// Build from parameter sets (NAL units without start codes); profile/level taken from the first SPS.
303    pub fn new(sps: Vec<Vec<u8>>, pps: Vec<Vec<u8>>, length_size: u8) -> Self {
304        let (profile, compatibility, level) = sps.first().filter(|s| s.len() >= 4).map(|s| (s[1], s[2], s[3])).unwrap_or((0, 0, 0));
305        AvcConfig { profile, compatibility, level, length_size, sps, pps, ext: Vec::new() }
306    }
307
308    pub fn parse(data: &[u8]) -> Result<Self> {
309        let mut c = Cur::labeled(data, "avcC");
310        let _version = c.u8()?;
311        let profile = c.u8()?;
312        let compatibility = c.u8()?;
313        let level = c.u8()?;
314        let length_size = (c.u8()? & 3) + 1;
315        let nsps = (c.u8()? & 0x1F) as usize;
316        let mut sps = Vec::with_capacity(nsps);
317        for _ in 0..nsps {
318            let n = c.u16()? as usize;
319            sps.push(c.bytes(n)?.to_vec());
320        }
321        let npps = c.u8()? as usize;
322        let mut pps = Vec::with_capacity(npps);
323        for _ in 0..npps {
324            let n = c.u16()? as usize;
325            pps.push(c.bytes(n)?.to_vec());
326        }
327        Ok(AvcConfig { profile, compatibility, level, length_size, sps, pps, ext: c.rest().to_vec() })
328    }
329
330    pub fn to_bytes(&self) -> Vec<u8> {
331        let mut b = BoxBuf::new();
332        b.u8(1);
333        b.u8(self.profile);
334        b.u8(self.compatibility);
335        b.u8(self.level);
336        b.u8(0xFC | (self.length_size.clamp(1, 4) - 1));
337        b.u8(0xE0 | (self.sps.len().min(31) as u8));
338        for s in self.sps.iter().take(31) {
339            b.u16(s.len() as u16);
340            b.bytes(s);
341        }
342        b.u8(self.pps.len().min(255) as u8);
343        for p in self.pps.iter().take(255) {
344            b.u16(p.len() as u16);
345            b.bytes(p);
346        }
347        b.bytes(&self.ext);
348        b.buf
349    }
350}

361#[derive(Clone, Debug, PartialEq, Eq, Default)]
362pub struct HevcConfig {
363    pub general_profile_space: u8,
364    pub general_tier_flag: bool,
365    pub general_profile_idc: u8,
366    pub general_profile_compatibility_flags: u32,
367    /// 48 bits of general constraint indicator flags.
368    pub general_constraint_indicator_flags: u64,
369    pub general_level_idc: u8,
370    pub min_spatial_segmentation_idc: u16,
371    pub parallelism_type: u8,
372    pub chroma_format_idc: u8,
373    pub bit_depth_luma: u8,
374    pub bit_depth_chroma: u8,
375    pub avg_frame_rate: u16,
376    pub constant_frame_rate: u8,
377    pub num_temporal_layers: u8,
378    pub temporal_id_nested: bool,
379    pub length_size: u8,
380    pub arrays: Vec<HevcNalArray>,
381}
382
383impl HevcConfig {
384    pub fn parse(data: &[u8]) -> Result<Self> {
385        let mut c = Cur::labeled(data, "hvcC");
386        let _version = c.u8()?;
387        let b = c.u8()?;
388        let compat = c.u32()?;
389        let hi = c.u16()? as u64;
390        let lo = c.u32()? as u64;
391        let level = c.u8()?;
392        let min_seg = c.u16()? & 0x0FFF;
393        let par = c.u8()? & 3;
394        let chroma = c.u8()? & 3;
395        let bdl = (c.u8()? & 7) + 8;
396        let bdc = (c.u8()? & 7) + 8;
397        let afr = c.u16()?;
398        let x = c.u8()?;
399        let narrays = c.u8()? as usize;
400        let mut arrays = Vec::new();
401        for _ in 0..narrays {
402            let t = c.u8()?;
403            let n = c.u16()? as usize;
404            c.check_count(n, 2)?;
405            let mut nalus = Vec::with_capacity(n);
406            for _ in 0..n {
407                let len = c.u16()? as usize;
408                nalus.push(c.bytes(len)?.to_vec());
409            }
410            arrays.push(HevcNalArray { completeness: t & 0x80 != 0, nal_type: t & 0x3F, nalus });
411        }
412        Ok(HevcConfig {
413            general_profile_space: b >> 6,
414            general_tier_flag: b & 0x20 != 0,
415            general_profile_idc: b & 0x1F,
416            general_profile_compatibility_flags: compat,
417            general_constraint_indicator_flags: (hi << 32) | lo,
418            general_level_idc: level,
419            min_spatial_segmentation_idc: min_seg,
420            parallelism_type: par,
421            chroma_format_idc: chroma,
422            bit_depth_luma: bdl,
423            bit_depth_chroma: bdc,
424            avg_frame_rate: afr,
425            constant_frame_rate: x >> 6,
426            num_temporal_layers: (x >> 3) & 7,
427            temporal_id_nested: x & 4 != 0,
428            length_size: (x & 3) + 1,
429            arrays,
430        })
431    }
432
433    pub fn to_bytes(&self) -> Vec<u8> {
434        let mut b = BoxBuf::new();
435        b.u8(1);
436        b.u8((self.general_profile_space << 6) | ((self.general_tier_flag as u8) << 5) | (self.general_profile_idc & 0x1F));
437        b.u32(self.general_profile_compatibility_flags);
438        b.u16((self.general_constraint_indicator_flags >> 32) as u16);
439        b.u32(self.general_constraint_indicator_flags as u32);
440        b.u8(self.general_level_idc);
441        b.u16(0xF000 | (self.min_spatial_segmentation_idc & 0x0FFF));
442        b.u8(0xFC | (self.parallelism_type & 3));
443        b.u8(0xFC | (self.chroma_format_idc & 3));
444        b.u8(0xF8 | (self.bit_depth_luma.saturating_sub(8) & 7));
445        b.u8(0xF8 | (self.bit_depth_chroma.saturating_sub(8) & 7));
446        b.u16(self.avg_frame_rate);
447        b.u8((self.constant_frame_rate << 6)
448            | ((self.num_temporal_layers & 7) << 3)
449            | ((self.temporal_id_nested as u8) << 2)
450            | (self.length_size.clamp(1, 4) - 1));
451        b.u8(self.arrays.len() as u8);
452        for a in &self.arrays {
453            b.u8(((a.completeness as u8) << 7) | (a.nal_type & 0x3F));
454            b.u16(a.nalus.len() as u16);
455            for n in &a.nalus {
456                b.u16(n.len() as u16);
457                b.bytes(n);
458            }
459        }
460        b.buf
461    }
462
463    fn nalus_of(&self, t: u8) -> Vec<&[u8]> {
464        self.arrays.iter().filter(|a| a.nal_type == t).flat_map(|a| a.nalus.iter().map(|n| n.as_slice())).collect()
465    }
466    pub fn vps(&self) -> Vec<&[u8]> {
467        self.nalus_of(32)
468    }
469    pub fn sps(&self) -> Vec<&[u8]> {
470        self.nalus_of(33)
471    }
472    pub fn pps(&self) -> Vec<&[u8]> {
473        self.nalus_of(34)
474    }
475}

477/// `av1C` — AV1CodecConfigurationRecord.
478#[derive(Clone, Debug, PartialEq, Eq, Default)]
479pub struct Av1Config {
480    pub seq_profile: u8,
481    pub seq_level_idx_0: u8,
482    pub seq_tier_0: bool,
483    pub high_bitdepth: bool,
484    pub twelve_bit: bool,
485    pub monochrome: bool,
486    pub chroma_subsampling_x: bool,
487    pub chroma_subsampling_y: bool,
488    pub chroma_sample_position: u8,
489    pub initial_presentation_delay_minus_one: Option<u8>,
490    pub config_obus: Vec<u8>,
491}
492
493impl Av1Config {
494    pub fn parse(data: &[u8]) -> Result<Self> {
495        let mut c = Cur::labeled(data, "av1C");
496        let m = c.u8()?;
497        if m & 0x80 == 0 {
498            return Err(Error::Invalid("av1C marker bit not set".into()));
499        }
500        let a = c.u8()?;
501        let b = c.u8()?;
502        let d = c.u8()?;
503        Ok(Av1Config {
504            seq_profile: a >> 5,
505            seq_level_idx_0: a & 0x1F,
506            seq_tier_0: b & 0x80 != 0,
507            high_bitdepth: b & 0x40 != 0,
508            twelve_bit: b & 0x20 != 0,
509            monochrome: b & 0x10 != 0,
510            chroma_subsampling_x: b & 0x08 != 0,
511            chroma_subsampling_y: b & 0x04 != 0,
512            chroma_sample_position: b & 3,
513            initial_presentation_delay_minus_one: if d & 0x10 != 0 { Some(d & 0x0F) } else { None },
514            config_obus: c.rest().to_vec(),
515        })
516    }
517    pub fn to_bytes(&self) -> Vec<u8> {
518        let mut b = BoxBuf::new();
519        b.u8(0x81);
520        b.u8((self.seq_profile << 5) | (self.seq_level_idx_0 & 0x1F));
521        b.u8(((self.seq_tier_0 as u8) << 7)
522            | ((self.high_bitdepth as u8) << 6)
523            | ((self.twelve_bit as u8) << 5)
524            | ((self.monochrome as u8) << 4)
525            | ((self.chroma_subsampling_x as u8) << 3)
526            | ((self.chroma_subsampling_y as u8) << 2)
527            | (self.chroma_sample_position & 3));
528        b.u8(match self.initial_presentation_delay_minus_one {
529            Some(d) => 0x10 | (d & 0x0F),
530            None => 0,
531        });
532        b.bytes(&self.config_obus);
533        b.buf
534    }
535}

544    pub full_range: bool,

573            level,
574            bit_depth: a >> 4,

576            full_range: a & 1 != 0,
577            colour_primaries,
578            transfer_characteristics,
579            matrix_coefficients,
580            codec_init,
581        })
582    }

585        b.u32(1 << 24);
586        b.u8(self.profile);
587        b.u8(self.level);

589        b.u8(self.colour_primaries);
590        b.u8(self.transfer_characteristics);
591        b.u8(self.matrix_coefficients);
592        b.u16(self.codec_init.len() as u16);
593        b.bytes(&self.codec_init);
594        b.buf
595    }

598/// AAC (MPEG-4 audio) configuration from `esds`.
599#[derive(Clone, Debug, PartialEq, Eq, Default)]
600pub struct AacConfig {
601    /// AudioSpecificConfig bytes (DecoderSpecificInfo).
602    pub asc: Vec<u8>,
603    /// Audio object type (2 = AAC-LC, 5 = SBR/HE-AAC, 29 = PS …) as signaled first in the ASC.
604    pub object_type: u8,
605    /// Sampling rate signaled by the ASC (core rate; 0 if the ASC is missing).
606    pub sample_rate: u32,
607    /// Channel configuration from the ASC (0 = defined elsewhere).
608    pub channel_config: u8,
609    /// DecoderConfigDescriptor fields.
610    pub max_bitrate: u32,
611    pub avg_bitrate: u32,
612    pub buffer_size: u32,
613}
614
615const AAC_RATES: [u32; 13] = [96000, 88200, 64000, 48000, 44100, 32000, 24000, 22050, 16000, 12000, 11025, 8000, 7350];
616
617impl AacConfig {
618    /// Build from an AudioSpecificConfig, decoding its header fields.
619    pub fn from_asc(asc: Vec<u8>) -> Self {
620        let (object_type, sample_rate, channel_config) = parse_asc(&asc).unwrap_or((0, 0, 0));
621        AacConfig { asc, object_type, sample_rate, channel_config, max_bitrate: 0, avg_bitrate: 0, buffer_size: 0 }
622    }
623    /// Channel count implied by the channel configuration (ISO/IEC 14496-3 Table 1.19 with Amd 4:
624    /// 7 and 12 are 7.1 front / back, 11 is 6.1, 13 is 22.2, 14 is 7.1 with front heights), 0 if
625    /// unspecified (a PCE) or reserved.
626    pub fn channels(&self) -> u32 {
627        match self.channel_config {
628            1..=6 => self.channel_config as u32,
629            11 => 7,
630            7 | 12 | 14 => 8,
631            13 => 24,
632            _ => 0,
633        }
634    }
635}

637/// Decode (audio object type, sampling frequency, channel configuration) from an AudioSpecificConfig.
638pub fn parse_asc(asc: &[u8]) -> Option<(u8, u32, u8)> {
639    let mut r = BitReader::new(asc);
640    let aot = |r: &mut BitReader| -> Option<u8> {
641        let t = r.read_bits(5).ok()? as u8;
642        if t == 31 { Some(32 + r.read_bits(6).ok()? as u8) } else { Some(t) }
643    };
644    let rate = |r: &mut BitReader| -> Option<u32> {
645        let i = r.read_bits(4).ok()? as usize;
646        if i == 15 { r.read_bits(24).ok() } else { AAC_RATES.get(i).copied() }
647    };
648    let t = aot(&mut r)?;
649    let sr = rate(&mut r)?;
650    let ch = r.read_bits(4).ok()? as u8;
651    Some((t, sr, ch))
652}

729/// `dfLa` — FLAC metadata blocks, with STREAMINFO decoded.
730#[derive(Clone, Debug, PartialEq, Eq, Default)]
731pub struct FlacConfig {
732    /// Raw metadata blocks (each with its 4-byte header) as stored in `dfLa`.
733    pub metadata_blocks: Vec<u8>,
734    pub sample_rate: u32,
735    pub channels: u8,
736    pub bits_per_sample: u8,
737    pub total_samples: u64,
738}
739
740impl FlacConfig {
741    pub fn parse(data: &[u8]) -> Result<Self> {
742        let mut c = Cur::labeled(data, "dfLa");
743        c.full_header()?;
744        let blocks = c.rest().to_vec();
745        let mut cfg = FlacConfig { metadata_blocks: blocks.clone(), ..Default::default() };
746        let mut b = Cur::labeled(&blocks, "dfLa");
747        let hdr = b.u32()?;
748        if (hdr >> 24) & 0x7F == 0 {
749            let si = b.bytes(((hdr & 0xFF_FFFF) as usize).min(34))?;
750            if si.len() >= 18 {
751                let mut r = BitReader::new(&si[10..18]);
752                cfg.sample_rate = r.read_bits(20).unwrap_or(0);
753                cfg.channels = r.read_bits(3).unwrap_or(0) as u8 + 1;
754                cfg.bits_per_sample = r.read_bits(5).unwrap_or(0) as u8 + 1;
755                cfg.total_samples = r.read_bits_u64(36).unwrap_or(0);
756            }
757        }
758        Ok(cfg)
759    }

762/// QuickTime timecode sample description (`tmcd`) plus the start frame read from its first sample.
763#[derive(Clone, Debug, PartialEq, Eq, Default)]
764pub struct TimecodeConfig {
765    /// Bit 0 drop frame, bit 1 24-hour max, bit 2 negative times OK, bit 3 counter.
766    pub flags: u32,
767    pub timescale: u32,
768    pub frame_duration: u32,
769    /// Nominal frames per second (`numberOfFrames`), e.g. 30 for 29.97.
770    pub frames_per_second: u8,
771    /// Frame number of the first timecode sample (filled in by the demuxer).
772    pub start_frame: Option<u32>,
773    /// Source reel name (`name` box) if present.
774    pub name: Option<String>,
775}
776
777impl TimecodeConfig {
778    pub fn drop_frame(&self) -> bool {
779        self.flags & 1 != 0
780    }
781    /// Format a frame number as `HH:MM:SS:FF` (`;` before the frames when drop-frame).
782    pub fn format_frame(&self, frame: u32) -> String {
783        let fps = self.frames_per_second.max(1) as u64;
784        let mut f = frame as u64;
785        if self.drop_frame() && fps >= 30 {
786            let drop = (fps / 30) * 2; // 2 frames per minute at 30, 4 at 60
787            let per_min = fps * 60 - drop;
788            let per_10 = per_min * 10 + drop;
789            let d = f / per_10;
790            let m = f % per_10;
791            f += drop * 9 * d + if m > drop { drop * ((m - drop) / per_min) } else { 0 };
792        }
793        let ff = f % fps;
794        let s = (f / fps) % 60;
795        let mi = (f / fps / 60) % 60;
796        let h = (f / fps / 3600) % 24;
797        let sep = if self.drop_frame() { ';' } else { ':' };
798        format!("{h:02}:{mi:02}:{s:02}{sep}{ff:02}")
799    }
800}

820    /// Raw `APVDecoderConfigurationRecord` bytes (after the 4-byte FullBox header).
821    pub raw_record: Vec<u8>,

824impl ApvConfig {
825    /// Build from a raw `APVDecoderConfigurationRecord` (starting with `configurationVersion = 1`).
826    pub fn from_record(record: Vec<u8>) -> Result<Self> {
827        let mut c = Cur::labeled(&record, "apvC");
828        let version = c.u8()?;
829        if version != 1 {
830            return Err(Error::Invalid(format!("unsupported APVDecoderConfigurationRecord version {version}")));
831        }
832        let num_entries = c.u8()?;
833        let mut cfg = ApvConfig { raw_record: record.clone(), ..Default::default() };
834        if num_entries >= 1 && c.remaining() >= 2 {
835            let _pbu_type = c.u8()?;
836            let num_fi = c.u8()?;
837            if num_fi >= 1 && c.remaining() >= 14 {
838                let flags = c.u8()?;
839                cfg.color_description_present = (flags & 0x02) != 0;
840                cfg.capture_time_distance_ignored = (flags & 0x01) != 0;
841                cfg.profile_idc = c.u8()?;
842                cfg.level_idc = c.u8()?;
843                cfg.band_idc = c.u8()?;
844                cfg.frame_width = c.u32()?;
845                cfg.frame_height = c.u32()?;
846                let cb = c.u8()?;
847                cfg.chroma_format_idc = cb >> 4;
848                cfg.bit_depth_minus8 = cb & 0x0F;
849                cfg.capture_time_distance = c.u8()?;
850                if cfg.color_description_present && c.remaining() >= 4 {
851                    cfg.color_primaries = c.u8()?;
852                    cfg.transfer_characteristics = c.u8()?;
853                    cfg.matrix_coefficients = c.u8()?;
854                    cfg.full_range = (c.u8()? & 0x80) != 0;
855                }
856            }
857        }
858        Ok(cfg)
859    }
860
861    /// Parse the payload of an `apvC` box (with or without the 4-byte `FullBox` header).
862    pub fn parse(data: &[u8]) -> Result<Self> {
863        if data.len() >= 5 && data[..4] == [0, 0, 0, 0] && data[4] == 1 {
864            Self::from_record(data[4..].to_vec())
865        } else if !data.is_empty() && data[0] == 1 {
866            Self::from_record(data.to_vec())
867        } else {
868            let mut c = Cur::labeled(data, "apvC");
869            let (v, _) = c.full_header()?;
870            if v != 0 {
871                return Err(Error::Invalid(format!("unsupported apvC FullBox version {v}")));
872            }
873            Self::from_record(c.rest().to_vec())
874        }
875    }
876
877    /// Serialize the full `apvC` box payload (4-byte `FullBox` header + `APVDecoderConfigurationRecord`).
878    pub fn to_bytes(&self) -> Vec<u8> {
879        [&[0u8; 4][..], &self.record_bytes()].concat() // FullBox version = 0, flags = 0
880    }
881
882    /// Serialize the bare `APVDecoderConfigurationRecord` (Matroska's `CodecPrivate` for `V_APV`).
883    pub fn record_bytes(&self) -> Vec<u8> {
884        if !self.raw_record.is_empty() {
885            return self.raw_record.clone();
886        }
887        let mut b = BoxBuf::new();
888        b.u8(1); // configurationVersion = 1
889        b.u8(1); // number_of_configuration_entry = 1
890        b.u8(1); // pbu_type = 1 (primary frame)
891        b.u8(1); // number_of_frame_info = 1
892        let flags = (if self.color_description_present { 0x02 } else { 0 }) | (if self.capture_time_distance_ignored { 0x01 } else { 0 });
893        b.u8(flags);
894        b.u8(self.profile_idc);
895        b.u8(self.level_idc);
896        b.u8(self.band_idc);
897        b.u32(self.frame_width);
898        b.u32(self.frame_height);
899        b.u8((self.chroma_format_idc << 4) | (self.bit_depth_minus8 & 0x0F));
900        b.u8(self.capture_time_distance);
901        if self.color_description_present {
902            b.u8(self.color_primaries);
903            b.u8(self.transfer_characteristics);
904            b.u8(self.matrix_coefficients);
905            b.u8(if self.full_range { 0x80 } else { 0 });
906        }
907        b.buf
908    }

971/// Parse one `stsd` entry. `handler` is the track handler type; `is_qt` selects QuickTime
972/// interpretation of the sound description version field.
973pub(crate) fn parse_sample_entry(format: FourCc, payload: &[u8], handler: FourCc, is_qt: bool) -> Result<SampleEntry> {
974    let mut c = Cur::labeled(payload, "sample entry");
975    c.skip(6)?;
976    let data_reference_index = c.u16()?;
977    let f = &format.0;
978    let unknown = || CodecConfig::Unknown { fourcc: format, raw: payload.to_vec() };
979    if f == b"tmcd" || (handler == b"tmcd" && !is_video_fourcc(f)) {
980        let mut tc = TimecodeConfig::default();
981        c.skip(4)?;
982        tc.flags = c.u32()?;
983        tc.timescale = c.u32()?;
984        tc.frame_duration = c.u32()?;
985        tc.frames_per_second = c.u8()?;
986        c.skip(1)?;
987        for b in boxes(c.rest()).map_while(|b| b.ok()) {
988            if b.kind == b"name" && b.payload.len() >= 4 {
989                let n = u16::from_be_bytes([b.payload[0], b.payload[1]]) as usize;
990                let s = b.payload.get(4..4 + n).unwrap_or(&b.payload[4..]);
991                tc.name = Some(String::from_utf8_lossy(s).into_owned());
992            }
993        }
994        return Ok(SampleEntry { format, data_reference_index, codec: CodecConfig::Timecode(tc), video: None, audio: None, bitrate: None });
995    }
996    // `raw ` names both uncompressed video and 8-bit unsigned PCM; the handler decides.
997    let raw_audio = f == b"raw " && handler == b"soun";
998    let video_like = (is_video_fourcc(f) && !raw_audio) || (handler == b"vide" && !is_audio_fourcc(f));
999    let audio_like = is_audio_fourcc(f) || (handler == b"soun" && !video_like);
1000    if video_like {
1001        let mut v = VideoParams::default();
1002        let _version = c.u16()?;
1003        let _revision = c.u16()?;
1004        v.vendor = c.fourcc()?;
1005        v.temporal_quality = c.u32()?;
1006        v.spatial_quality = c.u32()?;
1007        v.width = c.u16()?;
1008        v.height = c.u16()?;
1009        v.horiz_resolution = c.u32()?;
1010        v.vert_resolution = c.u32()?;
1011        c.skip(4)?;
1012        v.frame_count = c.u16()?;
1013        let name = c.bytes(32)?;
1014        let n = (name[0] as usize).min(31);
1015        v.compressor_name = String::from_utf8_lossy(&name[1..1 + n]).into_owned();
1016        v.depth = c.u16()?;
1017        let _ctab = c.i16()?;
1018        let mut codec = None;
1019        let mut bitrate = None;
1020        let mut log_seen = false;
1021        for b in boxes(c.rest()).map_while(|b| b.ok()) {
1022            let p = b.payload;
1023            match &b.kind.0 {
1024                b"avcC" if matches!(f, b"avc1" | b"avc3") => codec = Some(CodecConfig::Avc(AvcConfig::parse(p)?)),
1025                b"hvcC" if matches!(f, b"hvc1" | b"hev1") => codec = Some(CodecConfig::Hevc(HevcConfig::parse(p)?)),
1026                b"av1C" if f == b"av01" => codec = Some(CodecConfig::Av1(Av1Config::parse(p)?)),
1027                b"vpcC" if f == b"vp09" => codec = Some(CodecConfig::Vp9(VpcConfig::parse(p)?)),
1028                b"apvC" if f == b"apv1" => codec = Some(CodecConfig::Apv(ApvConfig::parse(p)?)),
1029                b"colr" => v.color = parse_colr(p).ok(),
1030                b"logs" => {
1031                    // Two identifiers are ambiguous: neither is believed.
1032                    v.log_transfer_function =
1033                        if log_seen { None } else { std::str::from_utf8(p).ok().filter(|s| valid_log_identifier(s)).map(str::to_owned) };
1034                    log_seen = true;
1035                }
1036                b"mdcv" => v.mastering_display = MasteringDisplay::parse(p),
1037                b"clli" if p.len() >= 4 => v.content_light = Some((u16::from_be_bytes([p[0], p[1]]), u16::from_be_bytes([p[2], p[3]]))),
1038                b"pasp" => {
1039                    let mut c = Cur::labeled(p, "pasp");
1040                    v.pixel_aspect = Some((c.u32()?, c.u32()?));
1041                }
1042                b"clap" => {
1043                    let mut c = Cur::labeled(p, "clap");
1044                    v.clean_aperture = Some(CleanAperture {
1045                        width: (c.u32()?, c.u32()?),
1046                        height: (c.u32()?, c.u32()?),
1047                        horiz_offset: (c.i32()?, c.u32()?),
1048                        vert_offset: (c.i32()?, c.u32()?),
1049                    });
1050                }
1051                b"fiel" => {
1052                    let mut c = Cur::labeled(p, "fiel");
1053                    v.field_info = Some(FieldInfo { fields: c.u8()?, detail: c.u8()? });
1054                }
1055                b"gama" => v.gamma = Some(Cur::labeled(p, "gama").u32()?),
1056                b"btrt" => bitrate = Some(parse_btrt(p)?),
1057                _ => {}
1058            }
1059        }
1060        let codec = match codec {
1061            Some(c) => c,
1062            None => match f {
1063                b"apch" | b"apcn" | b"apcs" | b"apco" | b"ap4h" | b"ap4x" => CodecConfig::ProRes { fourcc: format },
1064                b"jpeg" | b"mjpa" | b"mjpb" => CodecConfig::Jpeg { fourcc: format },
1065                b"AVdn" | b"AVdh" => CodecConfig::Dnx { fourcc: format },
1066                b"apv1" => CodecConfig::Apv(ApvConfig::default()),
1067                _ => unknown(),
1068            },
1069        };
1070        return Ok(SampleEntry { format, data_reference_index, codec, video: Some(v), audio: None, bitrate });
1071    }
1072    if audio_like {
1073        let mut a = AudioParams { qt_version: c.u16()?, ..Default::default() };
1074        let _revision = c.u16()?;
1075        let _vendor = c.u32()?;
1076        a.channels = c.u16()? as u32;
1077        a.sample_size = c.u16()?;
1078        a.compression_id = c.i16()?;
1079        let _packet_size = c.u16()?;
1080        a.sample_rate = c.u32()? as f64 / 65536.0;
1081        if is_qt && a.qt_version == 1 {
1082            a.samples_per_packet = c.u32()?;
1083            a.bytes_per_packet = c.u32()?;
1084            a.bytes_per_frame = c.u32()?;
1085            a.bytes_per_sample = c.u32()?;
1086        } else if is_qt && a.qt_version == 2 {
1087            let _size_of_struct = c.u32()?;
1088            a.sample_rate = c.f64()?;
1089            a.channels = c.u32()?;
1090            if a.channels > u32::from(u16::MAX) {
1091                return Err(Error::Invalid(format!("QuickTime v2 sound entry with {} channels", a.channels)));
1092            }
1093            let _always = c.u32()?;
1094            a.const_bits_per_channel = c.u32()?;
1095            a.lpcm_flags = c.u32()?;
1096            a.const_bytes_per_packet = c.u32()?;
1097            a.const_frames_per_packet = c.u32()?;
1098        }
1099        // Flatten QuickTime `wave` children into the list of extension boxes.
1100        let mut ext: Vec<(FourCc, &[u8])> = Vec::new();
1101        for b in boxes(c.rest()).map_while(|b| b.ok()) {
1102            if b.kind == b"wave" {
1103                for w in boxes(b.payload).map_while(|b| b.ok()) {
1104                    ext.push((w.kind, w.payload));
1105                }
1106            } else {
1107                ext.push((b.kind, b.payload));
1108            }
1109        }
1110        let get = |k: &[u8; 4]| ext.iter().find(|(kk, _)| kk == k).map(|(_, p)| *p);
1111        if let Some(p) = get(b"enda") {
1112            let v = if p.len() >= 2 { u16::from_be_bytes([p[0], p[1]]) } else { p.first().copied().unwrap_or(0) as u16 };
1113            a.little_endian = Some(v != 0);
1114        }
1115        let bitrate = get(b"btrt").and_then(|p| parse_btrt(p).ok());
1116        let codec = match f {
1117            b"mp4a" => match get(b"esds") {
1118                Some(p) => parse_esds(p)?,
1119                None => unknown(),
1120            },
1121            b".mp3" | b".mp2" => CodecConfig::Mp3,
1122            b"alac" => match get(b"alac") {
1123                Some(p) if p.len() >= 4 => CodecConfig::Alac { cookie: p[4..].to_vec() },
1124                _ => unknown(),
1125            },
1126            b"Opus" => match get(b"dOps") {
1127                Some(p) => CodecConfig::Opus(OpusConfig::parse(p)?),
1128                None => unknown(),
1129            },
1130            b"fLaC" => match get(b"dfLa") {
1131                Some(p) => CodecConfig::Flac(FlacConfig::parse(p)?),
1132                None => unknown(),
1133            },
1134            b"ac-3" => CodecConfig::Ac3 { dac3: get(b"dac3").unwrap_or(&[]).to_vec() },
1135            b"ec-3" => CodecConfig::Eac3 { dec3: get(b"dec3").unwrap_or(&[]).to_vec() },
1136            b"lpcm" | b"sowt" | b"twos" | b"in24" | b"in32" | b"fl32" | b"fl64" | b"raw " | b"NONE" => CodecConfig::Pcm(qt_pcm(f, &a)?),
1137            b"ipcm" | b"fpcm" => {
1138                let mut pcm = PcmConfig {
1139                    bits: a.sample_size,
1140                    float: f == b"fpcm",
1141                    big_endian: true,
1142                    signed: true,
1143                    channels: a.channels,
1144                    sample_rate: a.sample_rate,
1145                };
1146                if let Some(p) = get(b"pcmC") {
1147                    let mut c = Cur::labeled(p, "pcmC");
1148                    c.full_header()?;
1149                    let flags = c.u8()?;
1150                    pcm.big_endian = flags & 1 == 0;
1151                    pcm.bits = c.u8()? as u16;
1152                }
1153                CodecConfig::Pcm(pcm)
1154            }
1155            _ => unknown(),
1156        };
1157        return Ok(SampleEntry { format, data_reference_index, codec, video: None, audio: Some(a), bitrate });
1158    }
1159    Ok(SampleEntry { format, data_reference_index, codec: unknown(), video: None, audio: None, bitrate: None })
1160}

1162fn qt_pcm(f: &[u8; 4], a: &AudioParams) -> Result<PcmConfig> {
1163    let mut p = PcmConfig { channels: a.channels, sample_rate: a.sample_rate, signed: true, ..Default::default() };
1164    match f {
1165        b"lpcm" => {
1166            p.bits = u16::try_from(a.const_bits_per_channel)
1167                .map_err(|_| Error::Invalid(format!("lpcm with {} bits per channel", a.const_bits_per_channel)))?;
1168            p.float = a.lpcm_flags & 1 != 0;
1169            p.big_endian = a.lpcm_flags & 2 != 0;
1170            p.signed = a.lpcm_flags & 4 != 0 || p.float;
1171            if p.bits == 0 {
1172                p.bits = a.sample_size;
1173            }
1174        }
1175        b"sowt" => {
1176            p.bits = if a.sample_size == 0 { 16 } else { a.sample_size };
1177        }
1178        b"twos" => {
1179            p.bits = if a.sample_size == 0 { 16 } else { a.sample_size };
1180            p.big_endian = true;
1181        }
1182        b"raw " | b"NONE" => {
1183            p.bits = if a.sample_size == 0 { 8 } else { a.sample_size };
1184            p.signed = p.bits != 8;
1185            p.big_endian = true;
1186        }
1187        b"in24" | b"in32" | b"fl32" | b"fl64" => {
1188            p.bits = match f {
1189                b"in24" => 24,
1190                b"in32" | b"fl32" => 32,
1191                _ => 64,
1192            };
1193            p.float = f[0] == b'f';
1194            p.big_endian = !a.little_endian.unwrap_or(false);
1195        }
1196        _ => {}
1197    }
1198    Ok(p)
1199}

1218fn descriptor(c: &mut Cur) -> Result<(u8, usize)> {
1219    let tag = c.u8()?;
1220    let mut len = 0usize;
1221    for _ in 0..4 {
1222        let b = c.u8()?;
1223        len = (len << 7) | (b & 0x7F) as usize;
1224        if b & 0x80 == 0 {
1225            break;
1226        }
1227    }
1228    Ok((tag, len))
1229}

1231fn parse_esds(p: &[u8]) -> Result<CodecConfig> {
1232    let mut c = Cur::labeled(p, "esds");
1233    c.full_header()?;
1234    let (tag, len) = descriptor(&mut c)?;
1235    if tag != 3 {
1236        return Err(Error::Invalid(format!("esds: expected ES_Descriptor, got tag {tag}")));
1237    }
1238    let es = c.bytes(len.min(c.remaining()))?;
1239    let mut c = Cur::labeled(es, "ES_Descriptor");
1240    c.u16()?;
1241    let fl = c.u8()?;
1242    if fl & 0x80 != 0 {
1243        c.u16()?;
1244    }
1245    if fl & 0x40 != 0 {
1246        let n = c.u8()? as usize;
1247        c.skip(n)?;
1248    }
1249    if fl & 0x20 != 0 {
1250        c.u16()?;
1251    }
1252    while c.remaining() > 0 {
1253        let (tag, len) = descriptor(&mut c)?;
1254        let body = c.bytes(len.min(c.remaining()))?;
1255        if tag != 4 {
1256            continue;
1257        }
1258        let mut d = Cur::labeled(body, "DecoderConfigDescriptor");
1259        let oti = d.u8()?;
1260        let _stream_type = d.u8()?;
1261        let buffer_size = d.u24()?;
1262        let max_bitrate = d.u32()?;
1263        let avg_bitrate = d.u32()?;
1264        let mut asc = Vec::new();
1265        while d.remaining() > 0 {
1266            let (t, l) = descriptor(&mut d)?;
1267            let b = d.bytes(l.min(d.remaining()))?;
1268            if t == 5 {
1269                asc = b.to_vec();
1270                break;
1271            }
1272        }
1273        return Ok(match oti {
1274            0x69 | 0x6B => CodecConfig::Mp3,
1275            _ => {
1276                let mut a = AacConfig::from_asc(asc);
1277                a.max_bitrate = max_bitrate;
1278                a.avg_bitrate = avg_bitrate;
1279                a.buffer_size = buffer_size;
1280                CodecConfig::Aac(a)
1281            }
1282        });
1283    }
1284    Err(Error::Invalid("esds without DecoderConfigDescriptor".into()))
1285}

1290fn put_descriptor(b: &mut BoxBuf, tag: u8, body: &[u8]) {
1291    b.u8(tag);
1292    let n = body.len() as u32;
1293    // Always use the 4-byte length form for simplicity (as many encoders do).
1294    b.u8(0x80 | ((n >> 21) & 0x7F) as u8);
1295    b.u8(0x80 | ((n >> 14) & 0x7F) as u8);
1296    b.u8(0x80 | ((n >> 7) & 0x7F) as u8);
1297    b.u8((n & 0x7F) as u8);
1298    b.bytes(body);
1299}

1301pub(crate) fn esds_payload(oti: u8, asc: &[u8], max_bitrate: u32, avg_bitrate: u32, buffer_size: u32) -> Vec<u8> {
1302    let mut dcd = BoxBuf::new();
1303    dcd.u8(oti);
1304    dcd.u8(0x15); // streamType audio (5) << 2 | upstream 0 | reserved 1
1305    dcd.u8((buffer_size >> 16) as u8);
1306    dcd.u16(buffer_size as u16);
1307    dcd.u32(max_bitrate);
1308    dcd.u32(avg_bitrate);
1309    if !asc.is_empty() {
1310        put_descriptor(&mut dcd, 5, asc);
1311    }
1312    let mut es = BoxBuf::new();
1313    es.u16(1); // ES_ID
1314    es.u8(0);
1315    put_descriptor(&mut es, 4, &dcd.buf);
1316    put_descriptor(&mut es, 6, &[2]); // SLConfigDescriptor, predefined = 2 (MP4)
1317    let mut b = BoxBuf::new();
1318    b.u32(0);
1319    put_descriptor(&mut b, 3, &es.buf);
1320    b.buf
1321}

1431    /// Serialize as a complete sample entry box. For PCM the format is chosen from the brand.
1432    pub(crate) fn write(&self, b: &mut BoxBuf, qt: bool) -> Result<()> {
1433        // An identifier the reader would not accept is refused, not truncated.
1434        if self.video.as_ref().and_then(|v| v.log_transfer_function.as_deref()).is_some_and(|s| !valid_log_identifier(s)) {
1435            return Err(Error::Invalid("invalid log transfer-function identifier".into()));
1436        }
1437        if let CodecConfig::Unknown { fourcc, raw } = &self.codec {
1438            let m = b.start(&fourcc.0);
1439            b.bytes(raw);
1440            b.end(m);
1441            return Ok(());
1442        }
1443        if let CodecConfig::Timecode(tc) = &self.codec {
1444            let m = b.start(b"tmcd");
1445            b.zeros(6);
1446            b.u16(self.data_reference_index);
1447            b.u32(0);
1448            b.u32(tc.flags);
1449            b.u32(tc.timescale);
1450            b.u32(tc.frame_duration);
1451            b.u8(tc.frames_per_second);
1452            b.u8(0);
1453            if let Some(name) = &tc.name {
1454                let n = b.start(b"name");
1455                b.u16(name.len() as u16);
1456                b.u16(0);
1457                b.bytes(name.as_bytes());
1458                b.end(n);
1459            }
1460            b.end(m);
1461            return Ok(());
1462        }
1463        if let Some(v) = &self.video {
1464            self.write_video(b, v);
1465            return Ok(());
1466        }
1467        let a = self.audio.clone().ok_or_else(|| Error::Mux("sample entry has neither video nor audio params".into()))?;
1468        // Choose the on-disk format and sound description version.
1469        let (format, version, pcm) = match &self.codec {
1470            CodecConfig::Pcm(p) => {
1471                if qt {
1472                    // Version 0 stores the rate as 16.16 fixed point; `raw ` is unsigned and
1473                    // `twos`/`sowt` are signed. Everything else needs `lpcm` v2.
1474                    let v0_rate = p.sample_rate < 65536.0;
1475                    if v0_rate && p.bits == 16 && !p.float && p.signed {
1476                        (if p.big_endian { *b"twos" } else { *b"sowt" }, 0u16, Some(*p))
1477                    } else if v0_rate && p.bits == 8 && !p.float {
1478                        (if p.signed { *b"twos" } else { *b"raw " }, 0, Some(*p))
1479                    } else {
1480                        (*b"lpcm", 2, Some(*p))
1481                    }
1482                } else {
1483                    (if p.float { *b"fpcm" } else { *b"ipcm" }, 0, Some(*p))
1484                }
1485            }
1486            _ => (self.format.0, if qt { a.qt_version.min(2) } else { 0 }, None),
1487        };
1488        let m = b.start(&format);
1489        b.zeros(6);
1490        b.u16(self.data_reference_index);
1491        if version == 2 {
1492            let p = pcm.unwrap_or_default();
1493            b.u16(2);
1494            b.u16(0);
1495            b.u32(0);
1496            b.u16(3);
1497            b.u16(16);
1498            b.i16(-2);
1499            b.u16(0);
1500            b.u32(65536);
1501            b.u32(72);
1502            b.f64(p.sample_rate);
1503            b.u32(p.channels);
1504            b.u32(0x7F00_0000);
1505            b.u32(p.bits as u32);
1506            let flags = (p.float as u32) | ((p.big_endian as u32) << 1) | (((p.signed && !p.float) as u32) << 2) | 8;
1507            b.u32(flags);
1508            b.u32(p.bytes_per_frame());
1509            b.u32(1);
1510        } else {
1511            b.u16(version);
1512            b.u16(0);
1513            b.u32(0);
1514            b.u16(a.channels.min(u16::MAX as u32) as u16);
1515            b.u16(match pcm {
1516                Some(p) => p.bits,
1517                None => {
1518                    if a.sample_size == 0 {
1519                        16
1520                    } else {
1521                        a.sample_size
1522                    }
1523                }
1524            });
1525            b.i16(if version == 1 { a.compression_id } else { 0 });
1526            b.u16(0);
1527            let sr = if a.sample_rate < 65536.0 { (a.sample_rate * 65536.0).round() as u32 } else { 0 };
1528            b.u32(sr);
1529            if version == 1 {
1530                b.u32(a.samples_per_packet);
1531                b.u32(a.bytes_per_packet);
1532                b.u32(a.bytes_per_frame);
1533                b.u32(a.bytes_per_sample);
1534            }
1535        }
1536        match &self.codec {
1537            CodecConfig::Aac(c) => {
1538                b.leaf(b"esds", &esds_payload(0x40, &c.asc, c.max_bitrate, c.avg_bitrate, c.buffer_size));
1539            }
1540            CodecConfig::Mp3 if format == *b"mp4a" => {
1541                b.leaf(b"esds", &esds_payload(0x6B, &[], 0, 0, 0));
1542            }

1544                let m = b.start_full(b"alac", 0, 0);
1545                b.bytes(cookie);
1546                b.end(m);
1547            }
1548            CodecConfig::Opus(o) => b.leaf(b"dOps", &o.to_bytes()),
1549            CodecConfig::Flac(fl) => {
1550                let m = b.start_full(b"dfLa", 0, 0);
1551                b.bytes(&fl.metadata_blocks);
1552                b.end(m);
1553            }
1554            CodecConfig::Ac3 { dac3 } => b.leaf(b"dac3", dac3),
1555            CodecConfig::Eac3 { dec3 } => b.leaf(b"dec3", dec3),
1556            CodecConfig::Pcm(p) if !qt => {
1557                let m = b.start_full(b"pcmC", 0, 0);
1558                b.u8(if p.big_endian { 0 } else { 1 });
1559                b.u8(p.bits as u8);
1560                b.end(m);
1561            }
1562            CodecConfig::Pcm(p) if p.channels == 6 => {
1563                // QuickTime channel layout: kAudioChannelLayoutTag_MPEG_5_1_A (L R C LFE Ls Rs)
1564                let m = b.start_full(b"chan", 0, 0);
1565                b.u32((121 << 16) | 6);
1566                b.u32(0);
1567                b.u32(0);
1568                b.end(m);
1569            }
1570            _ => {}
1571        }
1572        if let Some(br) = self.bitrate {
1573            write_btrt(b, br);
1574        }
1575        b.end(m);
1576        Ok(())
1577    }

1583        b.u16(0);
1584        b.u16(0);
