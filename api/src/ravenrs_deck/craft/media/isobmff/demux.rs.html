9/// Hard cap on the sample count of the whole file, all tracks and fragments (after PCM chunk merging).
10const MAX_SAMPLES: usize = 1 << 26;
11/// Bytes of file each sample needs beyond the first `MIN_SAMPLE_ALLOWANCE` samples. A [`Sample`]
12/// takes 40 bytes of memory and a constant-size `stsz` or a `trun` without per-sample fields
13/// describes one in a byte or none, so without this a file could make the sample tables about
14/// 40 times its own size. A table-backed sample has a 4-byte `stsz` or `trun` entry.
15const FILE_BYTES_PER_SAMPLE: u64 = 4;
16/// Samples any file may hold, however short: 40 MiB of sample tables.
17const MIN_SAMPLE_ALLOWANCE: usize = 1 << 20;
18/// Hard cap on the size of an in-memory `moov`/`moof` box.
19const MAX_HEADER_BOX: u64 = 1 << 30;
20/// Most text all metadata entries may hold together; entries past it are dropped. One `keys`
21/// name can be named by many `ilst` items, each a copy, so the file's size does not bound this.
22const MAX_METADATA_BYTES: usize = 4 << 20;
23/// What an entry costs beyond its text, so that empty entries are bounded too.
24const METADATA_ENTRY_COST: usize = 64;
25
26/// Largest sample built from a PCM chunk; larger chunks are split. Writers that put a whole
27/// recording in one chunk would otherwise give samples past `u32::MAX` bytes.
28const PCM_MAX_SAMPLE_BYTES: u64 = 1 << 24;
29
30/// Frames in one piece of a split PCM chunk: whole units, at most `PCM_MAX_SAMPLE_BYTES`.
31fn pcm_piece_frames(unit: u64, unit_bytes: u64) -> u64 {
32    (PCM_MAX_SAMPLE_BYTES / unit_bytes.max(1)).max(1) * unit
33}

35/// What is left of the file-wide limits while tracks and fragments are parsed.
36struct SampleBudget {
37    /// Samples (what the file's length allows, at most `MAX_SAMPLES`, minus those already built).
38    samples: usize,
39    /// Bytes of file for samples whose count no table entry backs (constant-size `stsz`, `trun`
40    /// without per-sample fields). Each such sample claims at least one byte, so a file can't
41    /// describe more of them than it has bytes, however many tracks or fragments repeat the claim.
42    implied_bytes: u64,
43}
44
45impl SampleBudget {
46    fn new(file_len: u64) -> Self {
47        let by_len = usize::try_from(file_len / FILE_BYTES_PER_SAMPLE).unwrap_or(usize::MAX);
48        Self { samples: by_len.clamp(MIN_SAMPLE_ALLOWANCE, MAX_SAMPLES), implied_bytes: file_len }
49    }
50
51    fn take(&mut self, n: usize) -> Result<()> {
52        self.samples = self.samples.checked_sub(n).ok_or(Error::TooLarge("sample count"))?;
53        Ok(())
54    }
55}

148impl Track {
149    /// The first sample entry's codec configuration.
150    pub fn codec(&self) -> Option<&CodecConfig> {
151        self.entries.first().map(|e| &e.codec)
152    }
153
154    /// What a decoder needs for this track, in the container-neutral form the codec crates take:
155    /// [`Track::codec_params_for`] the first sample's description (the first entry for a track
156    /// without samples, or when that index names no entry). A track whose samples use more than
157    /// one description ([`Track::has_mixed_descriptions`]) needs `codec_params_for` per sample.
158    pub fn codec_params(&self) -> crate::craft::media::CodecParams {
159        let first = self.samples.first().map(|s| s.description_index);
160        self.codec_params_for(first.filter(|&i| i >= 1 && i as usize <= self.entries.len()).unwrap_or(1))
161    }
162
163    /// What a decoder needs for the samples whose [`Sample::description_index`] is
164    /// `description_index` (1-based, into [`Track::entries`]); [`crate::craft::media::CodecId::Unknown`] when
165    /// there is no such entry.
166    pub fn codec_params_for(&self, description_index: u32) -> crate::craft::media::CodecParams {
167        let Some(entry) = description_index.checked_sub(1).and_then(|i| self.entries.get(i as usize)) else {
168            return crate::craft::media::CodecParams::new(crate::craft::media::CodecId::Unknown);
169        };
170        let codec = &entry.codec;
171        let id = self.mpeg_audio.get(description_index as usize - 1).copied().flatten().unwrap_or(codec.codec_id());
172        let mut p = crate::craft::media::CodecParams::new(id).with_extradata(codec.extradata()).with_tag(entry.format.0);
173        if let Some(v) = &entry.video {
174            p = p.with_size(u32::from(v.width), u32::from(v.height));
175        }
176        if let Some(a) = &entry.audio {
177            p = p.with_sample_rate(a.sample_rate.round() as u32).with_channels(u16::try_from(a.channels).unwrap_or(0));
178        }
179        match codec {
180            CodecConfig::Pcm(c) => {
181                p = p.with_pcm(if c.float {
182                    crate::craft::media::PcmFormat::float(c.bits, c.big_endian)
183                } else {
184                    crate::craft::media::PcmFormat::int(c.bits, c.signed, c.big_endian)
185                });
186                p = p.with_sample_rate(c.sample_rate.round() as u32).with_channels(u16::try_from(c.channels).unwrap_or(0));
187            }
188            CodecConfig::Opus(o) => p = p.with_sample_rate(48_000).with_channels(u16::from(o.output_channels)),
189            _ => {}
190        }
191        p
192    }

194    /// True when the samples use more than one sample description, so no single
195    /// [`Track::codec_params`] describes them all.
196    pub fn has_mixed_descriptions(&self) -> bool {
197        self.samples.windows(2).any(|w| w[0].description_index != w[1].description_index)
198    }
199    pub fn video(&self) -> Option<&crate::craft::media::isobmff::codec::VideoParams> {
200        self.entries.first().and_then(|e| e.video.as_ref())
201    }
202    /// Clockwise quarter turns (0–3) the `tkhd` matrix applies for display, or None when it is not
203    /// a pure 0/90/180/270° rotation (mirrored, scaled or sheared — left to the caller).
204    ///
205    /// ISO/IEC 14496-12 §8.3.2: a sample point (p, q) is displayed at p′ = a·p + c·q + x,
206    /// q′ = b·p + d·q + y for the matrix {a, b, u, c, d, v, x, y, w}, with y pointing down. So
207    /// {0, 1, −1, 0} sends the top row to the right-hand column (90° clockwise — iPhone portrait
208    /// video), {−1, 0, 0, −1} is 180° and {0, −1, 1, 0} is 270°. Only the signs matter: a·d − b·c
209    /// carries the scale, which the presentation size already reflects.
210    pub fn display_rotation(&self) -> Option<u8> {
211        let [a, b, _, c, d, _, _, _, _] = self.matrix;
212        match (a.signum(), b.signum(), c.signum(), d.signum()) {
213            (1, 0, 0, 1) => Some(0),
214            (0, 1, -1, 0) => Some(1),
215            (-1, 0, 0, -1) => Some(2),
216            (0, -1, 1, 0) => Some(3),
217            _ => None,
218        }
219    }
220    pub fn audio(&self) -> Option<&crate::craft::media::isobmff::codec::AudioParams> {
221        self.entries.first().and_then(|e| e.audio.as_ref())
222    }
223    /// Presentation (edit-adjusted) pts of sample `i`.
224    pub fn presentation_pts(&self, i: usize) -> Option<i64> {
225        self.samples.get(i).map(|s| s.pts.saturating_add(self.edit_offset))
226    }
227    /// Index of the sample being presented at raw media time `pts` (the last sample, in
228    /// presentation order, whose pts is ≤ `pts`; the first sample if `pts` precedes all).
229    pub fn sample_at_pts(&self, pts: i64) -> Option<usize> {
230        if self.pts_order.is_empty() {
231            return None;
232        }
233        let k = self.pts_order.partition_point(|&i| self.samples[i as usize].pts <= pts);
234        Some(self.pts_order[k.saturating_sub(1)] as usize)
235    }
236    /// Like [`sample_at_pts`](Self::sample_at_pts) but for edit-adjusted presentation time.
237    pub fn sample_at_presentation_time(&self, t: i64) -> Option<usize> {
238        self.sample_at_pts(t.saturating_sub(self.edit_offset))
239    }
240    /// The closest sync sample at or before `index` (decode order); 0 if there is none.
241    pub fn sync_sample_before(&self, index: usize) -> usize {
242        let k = self.sync_index.partition_point(|&i| i as usize <= index);
243        if k == 0 { 0 } else { self.sync_index[k - 1] as usize }
244    }
245    /// Indices of all sync samples.
246    pub fn sync_samples(&self) -> &[u32] {
247        &self.sync_index
248    }
249
250    fn finish_indexes(&mut self) {
251        self.sync_index = self.samples.iter().enumerate().filter(|(_, s)| s.is_sync).map(|(i, _)| i as u32).collect();
252        let mut order: Vec<u32> = (0..self.samples.len() as u32).collect();
253        order.sort_by_key(|&i| (self.samples[i as usize].pts, i));
254        self.pts_order = order;
255    }
256}

299impl Mp4File {
300    /// Read the bytes of sample `index` of track `track` (index into [`tracks`](Self::tracks)).
301    pub fn read_sample<S: ByteSource + ?Sized>(&self, src: &S, track: usize, index: usize) -> Result<Vec<u8>> {
302        let t = self.tracks.get(track).ok_or(Error::NoSuchTrack(track))?;
303        let s = t.samples.get(index).ok_or(Error::NoSuchSample(index))?;
304        read_range(src, s.offset, s.size as u64)
305    }
306    pub fn sample_at_pts(&self, track: usize, pts: i64) -> Option<usize> {
307        self.tracks.get(track)?.sample_at_pts(pts)
308    }
309    pub fn sync_sample_before(&self, track: usize, index: usize) -> Option<usize> {
310        Some(self.tracks.get(track)?.sync_sample_before(index))
311    }
312    /// The chapter list: the QuickTime chapter track (the text track an enabled track's `tref/chap`
313    /// points at; titles read from `src`, starts in the movie timescale, edit-adjusted) when there is
314    /// one, else the Nero `chpl` list. Empty when the file has neither.
315    ///
316    /// Chapters outside the track's first non-empty edit are left out, and one spanning that edit's
317    /// start begins where the edit does; later edits are ignored. Titles are cut to
318    /// [`MAX_CHAPTER_TITLE_BYTES`](crate::craft::media::isobmff::MAX_CHAPTER_TITLE_BYTES).
319    pub fn read_chapters<S: ByteSource + ?Sized>(&self, src: &S) -> Result<Vec<crate::craft::media::isobmff::Chapter>> {
320        let chap_id = self
321            .tracks
322            .iter()
323            .filter(|t| t.enabled)
324            .flat_map(|t| &t.references)
325            .find(|(k, _)| k == b"chap")
326            .and_then(|(_, ids)| ids.first().copied());
327        let Some(t) = chap_id.and_then(|id| self.tracks.iter().find(|t| t.id == id)) else { return Ok(self.chpl_chapters.clone()) };
328        let edit = t.edits.iter().find(|e| e.media_time != -1);
329        let media_start = edit.map_or(i64::MIN, |e| e.media_time);
330        let media_end = edit
331            .filter(|e| e.segment_duration > 0)
332            .map_or(i64::MAX, |e| media_start.saturating_add(rescale(i128::from(e.segment_duration), self.timescale, t.timescale)));
333        let mut out = Vec::new();
334        for s in t.samples.iter().take(crate::craft::media::isobmff::MAX_CHAPTERS.saturating_mul(16)) {
335            if s.pts >= media_end || (s.pts.saturating_add(i64::from(s.duration)) <= media_start && s.pts < media_start) {
336                continue;
337            }
338            let data = read_range(src, s.offset, u64::from(s.size).min(MAX_CHAPTER_SAMPLE_BYTES))?;
339            let title = if u64::from(s.size) > MAX_CHAPTER_SAMPLE_BYTES {
340                crate::craft::media::isobmff::chapters::parse_text_sample_prefix(&data)
341            } else {
342                crate::craft::media::isobmff::chapters::parse_text_sample(&data)
343            };
344            let title = title.unwrap_or_default();
345            let title =
346                crate::craft::media::isobmff::chapters::truncate_utf8(&title, crate::craft::media::isobmff::MAX_CHAPTER_TITLE_BYTES).to_string();
347            let pts = s.pts.max(media_start).saturating_add(t.edit_offset).max(0) as u64;
348            let start = if t.timescale == 0 {
349                0
350            } else {
351                (u128::from(pts) * u128::from(self.timescale) / u128::from(t.timescale)).min(u128::from(u64::MAX)) as u64
352            };
353            out.push(crate::craft::media::isobmff::Chapter { start, title });
354        }
355        Ok(out)
356    }
357    /// First track of the given kind.
358    pub fn track_of_kind(&self, kind: TrackKind) -> Option<usize> {
359        self.tracks.iter().position(|t| t.kind == kind)
360    }
361}

363/// Most of a chapter sample read for its title: its 16-bit length and far more title than is kept.
364/// A longer title is decoded from what this holds of it.
365const MAX_CHAPTER_SAMPLE_BYTES: u64 = 4098;
366
367fn read_range<S: ByteSource + ?Sized>(src: &S, offset: u64, size: u64) -> Result<Vec<u8>> {
368    let end = offset.checked_add(size).ok_or(Error::Truncated("sample"))?;
369    if end > src.len() {
370        return Err(Error::Truncated("sample data"));
371    }
372    let mut v = vec![0u8; size as usize];
373    src.read_at(offset, &mut v)?;
374    Ok(v)
375}

377/// Parse an MP4/MOV file. Only box headers, `ftyp`, `moov` and `moof` boxes are read.
378pub fn open<S: ByteSource>(src: S) -> Result<Mp4File> {
379    let len = src.len();
380    let mut offset = 0u64;
381    let mut ftyp: Option<Vec<u8>> = None;
382    let mut moov: Option<(u64, Vec<u8>)> = None;
383    let mut moofs: Vec<(u64, Vec<u8>)> = Vec::new();
384    while offset + 8 <= len {
385        let n = (len - offset).min(32) as usize;
386        let mut hb = [0u8; 32];
387        src.read_at(offset, &mut hb[..n])?;
388        let h = BoxHeader::parse(&hb[..n])?;
389        let size = h.size.unwrap_or(len - offset);
390        let end = offset.saturating_add(size).min(len);
391        let body_len = end.saturating_sub(offset + h.header_len);
392        let read_body = |max: u64| -> Result<Vec<u8>> {
393            if body_len > max {
394                return Err(Error::TooLarge("header box"));
395            }
396            read_range(&src, offset + h.header_len, body_len)
397        };
398        match &h.kind.0 {
399            b"ftyp" if ftyp.is_none() => ftyp = Some(read_body(1 << 16)?),
400            b"moov" if moov.is_none() => moov = Some((offset, read_body(MAX_HEADER_BOX)?)),
401            b"moof" => moofs.push((offset, read_body(MAX_HEADER_BOX)?)),
402            _ => {}
403        }
404        if h.size.is_none() {
405            break;
406        }
407        // A box that claims to run past the end of the file ends the scan.
408        offset = end;
409    }
410    let (moov_offset, moov) = moov.ok_or(Error::NoMoov)?;
411    let mut file = Mp4File {
412        major_brand: FourCc::default(),
413        minor_version: 0,
414        compatible_brands: Vec::new(),
415        is_quicktime: true,
416        timescale: 0,
417        duration: 0,
418        tracks: Vec::new(),
419        metadata: Metadata::default(),
420        fragmented: false,
421        moov_offset,
422        fragment_count: moofs.len(),
423        chpl_chapters: Vec::new(),
424    };
425    if let Some(f) = &ftyp {
426        let mut c = Cur::labeled(f, "ftyp");
427        file.major_brand = c.fourcc()?;
428        file.minor_version = c.u32().unwrap_or(0);
429        while c.remaining() >= 4 {
430            file.compatible_brands.push(c.fourcc()?);
431        }
432        file.is_quicktime = file.major_brand == b"qt  ";
433    }
434    let mut trex: HashMap<u32, Trex> = HashMap::new();
435    let mut trak_payloads = Vec::new();
436    let mut meta_left = MAX_METADATA_BYTES;
437    let mut chpl = None;
438    for b in boxes(&moov) {
439        let b = b?;
440        match &b.kind.0 {
441            b"cmov" => return Err(Error::Unsupported("compressed movie header (cmov)".into())),
442            b"mvhd" => {
443                let mut c = Cur::labeled(b.payload, "mvhd");
444                let (v, _) = c.full_header()?;
445                if v == 1 {
446                    c.skip(16)?;
447                    file.timescale = c.u32()?;
448                    file.duration = c.u64()?;
449                } else {
450                    c.skip(8)?;
451                    file.timescale = c.u32()?;
452                    file.duration = c.u32()? as u64;
453                }
454            }
455            b"trak" => trak_payloads.push(b.payload),
456            b"mvex" => {
457                file.fragmented = true;
458                for m in boxes(b.payload).map_while(|b| b.ok()) {
459                    if m.kind == b"trex" {
460                        let mut c = Cur::labeled(m.payload, "trex");
461                        c.full_header()?;
462                        let id = c.u32()?;
463                        trex.insert(id, Trex { description_index: c.u32()?, duration: c.u32()?, size: c.u32()?, flags: c.u32()? });
464                    }
465                }
466            }
467            b"udta" => {
468                parse_udta(b.payload, &mut file.metadata, &mut meta_left);
469                chpl = find(b.payload, b"chpl");
470            }
471            b"meta" => parse_meta(b.payload, &mut file.metadata, &mut meta_left),
472            _ => {}
473        }
474    }
475    if let Some(p) = chpl {
476        file.chpl_chapters = crate::craft::media::isobmff::chapters::parse_chpl(p, file.timescale);
477    }
478    let mut budget = SampleBudget::new(len);
479    for p in trak_payloads {
480        let t = parse_trak(p, &file, len, &mut budget)?;
481        file.tracks.push(t);
482    }
483    // Movie fragments.
484    let mut next_dts: Vec<i64> =
485        file.tracks.iter().map(|t| t.samples.last().map(|s| s.dts.saturating_add(i64::from(s.duration))).unwrap_or(0)).collect();
486    // A duplicate track ID resolves to its first track.
487    let mut track_index = HashMap::new();
488    for (i, t) in file.tracks.iter().enumerate() {
489        track_index.entry(t.id).or_insert(i);
490    }
491    for (moof_off, moof) in &moofs {
492        parse_moof(*moof_off, moof, &mut file.tracks, &track_index, &trex, &mut next_dts, &mut budget)?;
493    }
494    for t in &mut file.tracks {
495        if file.fragmented && !moofs.is_empty() {
496            let d: u64 = t.samples.iter().map(|s| s.duration as u64).sum();
497            t.duration = t.duration.max(d);
498            if t.timescale > 0 && file.timescale > 0 {
499                let md = (t.duration as u128 * file.timescale as u128 / t.timescale as u128) as u64;
500                file.duration = file.duration.max(md);
501            }
502        }
503        t.finish_indexes();
504        // Timecode start frame from the first sample.
505        if let Some(CodecConfig::Timecode(tc)) = t.entries.first_mut().map(|e| &mut e.codec)
506            && let Some(s) = t.samples.first()
507            && s.size >= 4
508        {
509            let mut b = [0u8; 4];
510            if src.read_at(s.offset, &mut b).is_ok() {
511                tc.start_frame = Some(u32::from_be_bytes(b));
512            }
513        }
514        t.mpeg_audio = mpeg_audio_layers(t, &src);
515    }
516    Ok(file)
517}

519/// For each [`CodecConfig::Mp3`] entry, the layer its first sample's frame header names (these
520/// entries name none: ffmpeg stores MP2 as `mp4a` + 0x6B), else what the sample entry says
521/// (QuickTime `.mp2`, `.mp3`).
522fn mpeg_audio_layers<S: ByteSource>(t: &Track, src: &S) -> Vec<Option<crate::craft::media::CodecId>> {
523    use crate::craft::media::CodecId;
524    let from_header = |index: u32| {
525        let s = t.samples.iter().find(|s| s.description_index == index && s.size >= 4)?;
526        let mut b = [0u8; 4];
527        src.read_at(s.offset, &mut b).ok()?;
528        if b[0] != 0xFF || b[1] & 0xE0 != 0xE0 {
529            return None;
530        }
531        match (b[1] >> 1) & 3 {
532            3 => Some(CodecId::Mp1),
533            2 => Some(CodecId::Mp2),
534            1 => Some(CodecId::Mp3),
535            _ => None,
536        }
537    };
538    (1..)
539        .zip(&t.entries)
540        .map(|(index, e)| match e.codec {
541            CodecConfig::Mp3 => from_header(index).or(Some(if e.format == b".mp2" { CodecId::Mp2 } else { CodecId::Mp3 })),
542            _ => None,
543        })
544        .collect()
545}

555fn parse_language(code: u16) -> String {
556    if code < 0x400 {
557        // QuickTime Macintosh language code.
558        return if code == 0 { "eng".into() } else { "und".into() };
559    }
560    let c = |s: u16| (((code >> s) & 0x1F) as u8 + 0x60) as char;
561    let s: String = [c(10), c(5), c(0)].into_iter().collect();
562    if s.chars().all(|ch| ch.is_ascii_lowercase()) { s } else { "und".into() }
563}

565fn rescale(v: i128, from: u32, to: u32) -> i64 {
566    if from == 0 {
567        return 0;
568    }
569    (v * to as i128 / from as i128).clamp(i64::MIN as i128, i64::MAX as i128) as i64
570}

572fn parse_trak(p: &[u8], file: &Mp4File, file_len: u64, budget: &mut SampleBudget) -> Result<Track> {
573    let mut t = Track {
574        id: 0,
575        kind: TrackKind::Other,
576        handler: FourCc::default(),
577        handler_name: String::new(),
578        timescale: 0,
579        duration: 0,
580        tkhd_duration: 0,
581        language: "und".into(),
582        width: 0,
583        height: 0,
584        enabled: true,
585        alternate_group: 0,
586        volume: 0,
587        matrix: [0x10000, 0, 0, 0, 0x10000, 0, 0, 0, 0x4000_0000],
588        entries: Vec::new(),
589        samples: Vec::new(),
590        edits: Vec::new(),
591        edit_offset: 0,
592        references: Vec::new(),
593        sdtp: Vec::new(),
594        sample_groups: Vec::new(),
595        pcm_chunked: false,
596        sync_index: Vec::new(),
597        pts_order: Vec::new(),
598        mpeg_audio: Vec::new(),
599    };
600    let mut mdia = None;
601    for b in boxes(p) {
602        let b = b?;
603        match &b.kind.0 {
604            b"tkhd" => {
605                let mut c = Cur::labeled(b.payload, "tkhd");
606                let (v, flags) = c.full_header()?;
607                t.enabled = flags & 1 != 0;
608                if v == 1 {
609                    c.skip(16)?;
610                    t.id = c.u32()?;
611                    c.skip(4)?;
612                    t.tkhd_duration = c.u64()?;
613                } else {
614                    c.skip(8)?;
615                    t.id = c.u32()?;
616                    c.skip(4)?;
617                    t.tkhd_duration = c.u32()? as u64;
618                }
619                c.skip(8)?;
620                let _layer = c.i16()?;
621                t.alternate_group = c.u16()?;
622                t.volume = c.i16()?;
623                c.skip(2)?;
624                for m in t.matrix.iter_mut() {
625                    *m = c.i32()?;
626                }
627                t.width = c.u32()? >> 16;
628                t.height = c.u32()? >> 16;
629            }
630            b"edts" => {
631                if let Some(e) = find(b.payload, b"elst") {
632                    let mut c = Cur::labeled(e, "elst");
633                    let (v, _) = c.full_header()?;
634                    let n = c.count(if v == 1 { 20 } else { 12 })?;
635                    for _ in 0..n {
636                        let (segment_duration, media_time) = if v == 1 { (c.u64()?, c.i64()?) } else { (c.u32()? as u64, c.i32()? as i64) };
637                        t.edits.push(Edit { segment_duration, media_time, media_rate: c.i32()? });
638                    }
639                }
640            }
641            b"tref" => {
642                for r in boxes(b.payload).map_while(|b| b.ok()) {
643                    let ids = r.payload.as_chunks::<4>().0.iter().map(|c| u32::from_be_bytes([c[0], c[1], c[2], c[3]])).collect();
644                    t.references.push((r.kind, ids));
645                }
646            }
647            b"mdia" => mdia = Some(b.payload),
648            _ => {}
649        }
650    }
651    let mdia = mdia.ok_or_else(|| Error::Invalid("trak without mdia".into()))?;
652    let mut minf = None;
653    for b in boxes(mdia) {
654        let b = b?;
655        match &b.kind.0 {
656            b"mdhd" => {
657                let mut c = Cur::labeled(b.payload, "mdhd");
658                let (v, _) = c.full_header()?;
659                if v == 1 {
660                    c.skip(16)?;
661                    t.timescale = c.u32()?;
662                    t.duration = c.u64()?;
663                } else {
664                    c.skip(8)?;
665                    t.timescale = c.u32()?;
666                    t.duration = c.u32()? as u64;
667                }
668                t.language = parse_language(c.u16()?);
669            }
670            b"hdlr" => {
671                let mut c = Cur::labeled(b.payload, "hdlr");
672                c.full_header()?;
673                let _pre = c.u32()?;
674                t.handler = c.fourcc()?;
675                c.skip(12)?;
676                let name = c.rest();
677                // ISO: null-terminated UTF-8; QuickTime: Pascal string.
678                let s = if !name.is_empty() && name[0] as usize == name.len() - 1 && name.len() > 1 {
679                    &name[1..]
680                } else {
681                    name.split(|&b| b == 0).next().unwrap_or(&[])
682                };
683                t.handler_name = String::from_utf8_lossy(s).into_owned();
684            }
685            b"minf" => minf = Some(b.payload),
686            _ => {}
687        }
688    }
689    t.kind = match &t.handler.0 {
690        b"vide" => TrackKind::Video,
691        b"soun" => TrackKind::Audio,
692        b"tmcd" => TrackKind::Timecode,
693        b"text" | b"sbtl" | b"subt" | b"clcp" => TrackKind::Subtitle,
694        _ => TrackKind::Other,
695    };
696    // Edit offset in media timescale.
697    let mut empty = 0i128;
698    for e in &t.edits {
699        if e.media_time == -1 {
700            empty += e.segment_duration as i128;
701        } else {
702            t.edit_offset = rescale(empty, file.timescale, t.timescale).saturating_sub(e.media_time);
703            break;
704        }
705    }
706    if let Some(stbl) = minf.and_then(|m| find(m, b"stbl")) {
707        parse_stbl(stbl, &mut t, file.is_quicktime, file_len, budget)?;
708    }
709    Ok(t)
710}

712struct SttsCursor<'a> {
713    runs: &'a [(u32, u32)],
714    run: usize,
715    used: u32,
716}
717
718impl SttsCursor<'_> {
719    /// Advance by `n` samples, returning the summed duration.
720    fn take(&mut self, mut n: u64) -> u64 {
721        let mut total = 0u64;
722        while n > 0 && self.run < self.runs.len() {
723            let (count, delta) = self.runs[self.run];
724            let avail = (count - self.used) as u64;
725            let k = avail.min(n);
726            total = total.saturating_add(k.saturating_mul(delta as u64));
727            n -= k;
728            self.used += k as u32;
729            if self.used >= count {
730                self.run += 1;
731                self.used = 0;
732            }
733        }
734        total
735    }
736    fn next_value(&mut self) -> u32 {
737        while self.run < self.runs.len() && self.runs[self.run].0 == 0 {
738            self.run += 1;
739        }
740        let Some(&(_, d)) = self.runs.get(self.run) else { return 0 };
741        self.take(1);
742        d
743    }
744}

758fn parse_stbl_tables(stbl: &[u8], t: &mut Track, is_qt: bool, file_len: u64) -> Result<StblTables> {
759    let mut stts: Vec<(u32, u32)> = Vec::new();
760    let mut ctts: Vec<(u32, u32)> = Vec::new();
761    let mut stss: Option<Vec<u32>> = None;
762    let mut stsc: Vec<(u32, u32, u32)> = Vec::new();
763    let mut const_size = 0u32;
764    let mut sample_count = 0usize;
765    let mut sizes: Vec<u32> = Vec::new();
766    let mut chunks: Vec<u64> = Vec::new();
767    let mut sbgp: Vec<(FourCc, Vec<(u32, u32)>)> = Vec::new();
768    let mut sgpd: HashMap<FourCc, Vec<Vec<u8>>> = HashMap::new();
769    for b in boxes(stbl) {
770        let b = b?;
771        let mut c = Cur::labeled(b.payload, "stbl child");
772        match &b.kind.0 {
773            b"stsd" => {
774                c.full_header()?;
775                let n = c.count(8)?;
776                for e in boxes(c.rest()).take(n) {
777                    let e = e?;
778                    t.entries.push(parse_sample_entry(e.kind, e.payload, t.handler, is_qt)?);
779                }
780            }
781            b"stts" => {
782                c.full_header()?;
783                let n = c.count(8)?;
784                stts = (0..n).map(|_| Ok((c.u32()?, c.u32()?))).collect::<Result<_>>()?;
785            }
786            b"ctts" => {
787                c.full_header()?;
788                let n = c.count(8)?;
789                ctts = (0..n).map(|_| Ok((c.u32()?, c.u32()?))).collect::<Result<_>>()?;
790            }
791            b"stss" => {
792                c.full_header()?;
793                let n = c.count(4)?;
794                let mut v: Vec<u32> = (0..n).map(|_| c.u32()).collect::<Result<_>>()?;
795                v.sort_unstable();
796                stss = Some(v);
797            }
798            b"stsc" => {
799                c.full_header()?;
800                let n = c.count(12)?;
801                stsc = (0..n).map(|_| Ok((c.u32()?, c.u32()?, c.u32()?))).collect::<Result<_>>()?;
802            }
803            b"stsz" => {
804                c.full_header()?;
805                const_size = c.u32()?;
806                let n = c.u32()? as usize;
807                if const_size == 0 {
808                    c.check_count(n, 4)?;
809                    sizes = (0..n).map(|_| c.u32()).collect::<Result<_>>()?;
810                    sample_count = n;
811                } else {
812                    // Samples of constant size must fit in the file: clamp absurd counts.
813                    sample_count = (n as u64).min(file_len / const_size as u64) as usize;
814                }
815            }
816            b"stz2" => {
817                c.full_header()?;
818                let field = c.u32()? & 0xFF;
819                let n = c.u32()? as usize;
820                let bits = match field {
821                    4 | 8 | 16 => field as usize,
822                    _ => return invalid(format!("stz2 field size {field}")),
823                };
824                if n.checked_mul(bits).is_none_or(|b| b.div_ceil(8) > c.remaining()) {
825                    return Err(Error::Truncated("stz2"));
826                }
827                let d = c.rest();
828                sizes = (0..n)
829                    .map(|i| match bits {
830                        4 => ((d[i / 2] >> if i % 2 == 0 { 4 } else { 0 }) & 0xF) as u32,
831                        8 => d[i] as u32,
832                        _ => u16::from_be_bytes([d[2 * i], d[2 * i + 1]]) as u32,
833                    })
834                    .collect();
835                sample_count = n;
836            }
837            b"stco" => {
838                c.full_header()?;
839                let n = c.count(4)?;
840                chunks = (0..n).map(|_| c.u32().map(|v| v as u64)).collect::<Result<_>>()?;
841            }
842            b"co64" => {
843                c.full_header()?;
844                let n = c.count(8)?;
845                chunks = (0..n).map(|_| c.u64()).collect::<Result<_>>()?;
846            }
847            b"sdtp" => {
848                c.full_header()?;
849                t.sdtp = c.rest().to_vec();
850            }
851            b"sbgp" => {
852                let (v, _) = c.full_header()?;
853                let gt = c.fourcc()?;
854                if v == 1 {
855                    c.u32()?;
856                }
857                let n = c.count(8)?;
858                let runs = (0..n).map(|_| Ok((c.u32()?, c.u32()?))).collect::<Result<_>>()?;
859                sbgp.push((gt, runs));
860            }
861            b"sgpd" => {
862                let (v, _) = c.full_header()?;
863                let gt = c.fourcc()?;
864                let default_len = if v >= 1 { c.u32()? } else { 0 };
865                if v >= 2 {
866                    c.u32()?;
867                }
868                let n = c.count(0)?;
869                let mut descs = Vec::new();
870                for _ in 0..n {
871                    let l = if v >= 1 && default_len == 0 { c.u32()? } else { default_len };
872                    if l == 0 {
873                        // Version 0 without known length: keep the remainder as a single entry.
874                        descs.push(c.rest().to_vec());
875                        break;
876                    }
877                    descs.push(c.bytes(l as usize)?.to_vec());
878                }
879                sgpd.insert(gt, descs);
880            }
881            _ => {}
882        }
883    }
884    let sample_groups =
885        sbgp.into_iter().map(|(gt, runs)| SampleGroup { grouping_type: gt, runs, descriptions: sgpd.remove(&gt).unwrap_or_default() }).collect();
886    Ok(StblTables { stts, ctts, stss, stsc, const_size, sample_count, sizes, chunks, sample_groups })
887}

889fn parse_stbl(stbl: &[u8], t: &mut Track, is_qt: bool, file_len: u64, budget: &mut SampleBudget) -> Result<()> {
890    let StblTables { stts, ctts, stss, stsc, const_size, mut sample_count, sizes, chunks, sample_groups } =
891        parse_stbl_tables(stbl, t, is_qt, file_len)?;
892    t.sample_groups = sample_groups;
893    if sample_count == 0 || chunks.is_empty() || stsc.is_empty() {
894        return Ok(());
895    }
896    if stsc[0].0 == 0 || stsc.windows(2).any(|w| w[1].0 <= w[0].0) {
897        return invalid("stsc first_chunk values are not increasing from 1");
898    }
899    let pcm = match t.entries.first().map(|e| &e.codec) {
900        Some(CodecConfig::Pcm(p)) if const_size != 0 => Some(*p),
901        _ => None,
902    };
903    let mut stts_c = SttsCursor { runs: &stts, run: 0, used: 0 };
904    let mut ctts_c = SttsCursor { runs: &ctts, run: 0, used: 0 };
905    let mut samples = Vec::new();
906    let mut sidx = 0usize;
907    let mut dts = 0i64;
908    let mut stsc_i = 0usize;
909    if let Some(p) = pcm {
910        t.pcm_chunked = true;
911        let a = t.audio().cloned().unwrap_or_default();
912        // Frames come in units of `unit` frames and `unit_bytes` bytes (QuickTime v1 packets).
913        let (unit, unit_bytes) = if a.qt_version == 1 && a.bytes_per_frame > 0 && a.samples_per_packet > 0 {
914            (u64::from(a.samples_per_packet), u64::from(a.bytes_per_frame))
915        } else if const_size > 1 {
916            (1, u64::from(const_size))
917        } else {
918            (1, u64::from(p.bytes_per_frame()))
919        };
920        let frame_bytes = |frames: u64| -> u64 { (frames / unit).saturating_mul(unit_bytes) };
921        let piece_frames = pcm_piece_frames(unit, unit_bytes);
922        if chunks.len() > budget.samples {
923            return Err(Error::TooLarge("chunk count"));
924        }
925        samples.reserve(chunks.len());
926        for (ci, &off) in chunks.iter().enumerate() {
927            while stsc_i + 1 < stsc.len() && stsc[stsc_i + 1].0 as usize <= ci + 1 {
928                stsc_i += 1;
929            }
930            let spc = (stsc[stsc_i].1 as usize).min(sample_count - sidx);
931            if spc == 0 {
932                continue;
933            }
934            // A chunk past PCM_MAX_SAMPLE_BYTES becomes several samples.
935            let mut left = spc as u64;
936            let mut pos = off;
937            while left > 0 {
938                if samples.len() >= budget.samples {
939                    return Err(Error::TooLarge("sample count"));
940                }
941                let split = frame_bytes(left) > PCM_MAX_SAMPLE_BYTES;
942                let frames = if split { piece_frames } else { left };
943                let dur = stts_c.take(frames);
944                let size = frame_bytes(frames);
945                if split {
946                    // Each piece is at least half of PCM_MAX_SAMPLE_BYTES, so the file's size
947                    // bounds how many a hostile chunk size can make.
948                    budget.implied_bytes = budget.implied_bytes.checked_sub(size).ok_or(Error::TooLarge("PCM chunk size"))?;
949                }
950                samples.push(Sample {
951                    offset: pos,
952                    size: size.min(u32::MAX as u64) as u32,
953                    dts,
954                    pts: dts,
955                    duration: dur.min(u32::MAX as u64) as u32,
956                    is_sync: true,
957                    description_index: stsc[stsc_i].2,
958                });
959                dts = dts.saturating_add(dur.min(i64::MAX as u64) as i64);
960                pos = pos.saturating_add(size);
961                left -= frames;
962            }
963            sidx += spc;
964            if sidx >= sample_count {
965                break;
966            }
967        }
968    } else {
969        if const_size != 0 {
970            sample_count = sample_count.min((budget.implied_bytes / const_size as u64) as usize);
971        }
972        if sample_count > budget.samples {
973            return Err(Error::TooLarge("sample count"));
974        }
975        // Reserve only what the chunk map can place. A declared count far past it would keep
976        // capacity the budget below never charges, track after track.
977        let mut mapped = 0usize;
978        let mut map_i = 0usize;
979        for ci in 0..chunks.len() {
980            while map_i + 1 < stsc.len() && stsc[map_i + 1].0 as usize <= ci + 1 {
981                map_i += 1;
982            }
983            mapped = mapped.saturating_add(stsc.get(map_i).map_or(0, |e| e.1 as usize));
984            if mapped >= sample_count {
985                break;
986            }
987        }
988        samples.try_reserve(sample_count.min(mapped)).map_err(|_| Error::TooLarge("sample count"))?;
989        let mut stss_i = 0usize;
990        'chunks: for (ci, &off) in chunks.iter().enumerate() {
991            while stsc_i + 1 < stsc.len() && stsc[stsc_i + 1].0 as usize <= ci + 1 {
992                stsc_i += 1;
993            }
994            let (_, spc, sdi) = stsc[stsc_i];
995            let mut pos = off;
996            for _ in 0..spc {
997                if sidx >= sample_count {
998                    break 'chunks;
999                }
1000                let size = if const_size != 0 { const_size } else { sizes[sidx] };
1001                let duration = stts_c.next_value();
1002                let cts = ctts_c.next_value() as i32 as i64;
1003                let is_sync = match &stss {
1004                    None => true,
1005                    Some(v) => {
1006                        while stss_i < v.len() && (v[stss_i] as usize) < sidx + 1 {
1007                            stss_i += 1;
1008                        }
1009                        stss_i < v.len() && v[stss_i] as usize == sidx + 1
1010                    }
1011                };
1012                samples.push(Sample { offset: pos, size, dts, pts: dts.saturating_add(cts), duration, is_sync, description_index: sdi });
1013                pos = pos.saturating_add(size as u64);
1014                dts = dts.saturating_add(duration as i64);
1015                sidx += 1;
1016            }
1017        }
1018    }
1019    if pcm.is_none() && const_size != 0 {
1020        budget.implied_bytes -= samples.len() as u64 * const_size as u64;
1021    }
1022    budget.take(samples.len())?;
1023    t.samples = samples;
1024    Ok(())
1025}

1027fn parse_moof(
1028    moof_off: u64,
1029    moof: &[u8],
1030    tracks: &mut [Track],
1031    track_index: &HashMap<u32, usize>,
1032    trex: &HashMap<u32, Trex>,
1033    next_dts: &mut [i64],
1034    budget: &mut SampleBudget,
1035) -> Result<()> {
1036    // End of data of the previous traf, for the base-offset default of subsequent trafs.
1037    let mut prev_end: Option<u64> = None;
1038    for b in boxes(moof) {
1039        let b = b?;
1040        if b.kind != b"traf" {
1041            continue;
1042        }
1043        let mut tfhd = None;
1044        let mut tfdt = None;
1045        let mut truns = Vec::new();
1046        for c in boxes(b.payload) {
1047            let c = c?;
1048            match &c.kind.0 {
1049                b"tfhd" => tfhd = Some(c.payload),
1050                b"tfdt" => tfdt = Some(c.payload),
1051                b"trun" => truns.push(c.payload),
1052                _ => {}
1053            }
1054        }
1055        let tfhd = tfhd.ok_or_else(|| Error::Invalid("traf without tfhd".into()))?;
1056        let mut c = Cur::labeled(tfhd, "tfhd");
1057        let (_, fl) = c.full_header()?;
1058        let id = c.u32()?;
1059        let Some(&ti) = track_index.get(&id) else { continue };
1060        let d = trex.get(&id).copied().unwrap_or_default();
1061        let base = if fl & 1 != 0 {
1062            c.u64()?
1063        } else if fl & 0x20000 != 0 {
1064            moof_off
1065        } else {
1066            prev_end.unwrap_or(moof_off)
1067        };
1068        let sdi = if fl & 2 != 0 { c.u32()? } else { d.description_index };
1069        let def_dur = if fl & 8 != 0 { c.u32()? } else { d.duration };
1070        let def_size = if fl & 0x10 != 0 { c.u32()? } else { d.size };
1071        let def_flags = if fl & 0x20 != 0 { c.u32()? } else { d.flags };
1072        if let Some(p) = tfdt {
1073            let mut c = Cur::labeled(p, "tfdt");
1074            let (v, _) = c.full_header()?;
1075            let t = if v == 1 { c.u64()? } else { c.u32()? as u64 };
1076            next_dts[ti] = t.min(i64::MAX as u64) as i64;
1077        }
1078        let mut data_pos = base;
1079        for tr in truns {
1080            let mut c = Cur::labeled(tr, "trun");
1081            let (v, tf) = c.full_header()?;
1082            let per = [0x100u32, 0x200, 0x400, 0x800].iter().filter(|&&f| tf & f != 0).count() * 4;
1083            let n = c.u32()? as usize;
1084            if per > 0 {
1085                c.check_count(n, per)?;
1086            } else {
1087                let implied = (n as u64).saturating_mul(def_size.max(1) as u64);
1088                budget.implied_bytes = budget.implied_bytes.checked_sub(implied).ok_or(Error::TooLarge("trun sample count"))?;
1089            }
1090            budget.take(n)?;
1091            if tf & 1 != 0 {
1092                data_pos = base.wrapping_add_signed(c.i32()? as i64);
1093            }
1094            let first_flags = if tf & 4 != 0 { Some(c.u32()?) } else { None };
1095            let t = &mut tracks[ti];
1096            t.samples.reserve(n);
1097            for i in 0..n {
1098                let dur = if tf & 0x100 != 0 { c.u32()? } else { def_dur };
1099                let size = if tf & 0x200 != 0 { c.u32()? } else { def_size };
1100                let flags = if tf & 0x400 != 0 { c.u32()? } else { def_flags };
1101                // first_sample_flags overrides the flags of the first sample only.
1102                let flags = if i == 0 { first_flags.unwrap_or(flags) } else { flags };
1103                let cto = if tf & 0x800 != 0 {
1104                    let raw = c.u32()?;
1105                    if v == 0 { raw as i64 } else { raw as i32 as i64 }
1106                } else {
1107                    0
1108                };
1109                let dts = next_dts[ti];
1110                t.samples.push(Sample {
1111                    offset: data_pos,
1112                    size,
1113                    dts,
1114                    pts: dts.saturating_add(cto),
1115                    duration: dur,
1116                    is_sync: flags & 0x1_0000 == 0,
1117                    description_index: sdi,
1118                });
1119                data_pos = data_pos.saturating_add(size as u64);
1120                next_dts[ti] = dts.saturating_add(dur as i64);
1121            }
1122        }
1123        prev_end = Some(data_pos);
1124    }
1125    Ok(())
1126}

1128/// Takes an entry of `key_len` and `value_len` bytes from `left`; false once it would not fit.
1129fn charge_metadata(left: &mut usize, key_len: usize, value_len: usize) -> bool {
1130    let cost = key_len.saturating_add(value_len).saturating_add(METADATA_ENTRY_COST);
1131    match left.checked_sub(cost) {
1132        Some(rest) => {
1133            *left = rest;
1134            true
1135        }
1136        None => false,
1137    }
1138}

1140/// Length of `String::from_utf8_lossy(bytes)` without building it: each invalid sequence
1141/// becomes one three-byte U+FFFD.
1142fn lossy_len(bytes: &[u8]) -> usize {
1143    bytes.utf8_chunks().map(|c| c.valid().len() + if c.invalid().is_empty() { 0 } else { 3 }).sum()
1144}

1146fn parse_udta(p: &[u8], md: &mut Metadata, left: &mut usize) {
1147    for b in boxes(p).map_while(|b| b.ok()) {
1148        if b.kind == b"meta" {
1149            parse_meta(b.payload, md, left);
1150        } else if b.kind.0[0] == 0xA9 && b.payload.len() >= 4 {
1151            let n = u16::from_be_bytes([b.payload[0], b.payload[1]]) as usize;
1152            // Charged at the stored text's length, before the value is converted.
1153            let key = b.kind.to_string_lossy();
1154            if let Some(s) = b.payload.get(4..4 + n)
1155                && charge_metadata(left, key.len(), lossy_len(s))
1156            {
1157                md.entries.push((key, String::from_utf8_lossy(s).into_owned()));
1158            }
1159        }
1160    }
1161}
1162
1163fn parse_meta(p: &[u8], md: &mut Metadata, left: &mut usize) {
1164    // QuickTime `meta` is a plain box; ISO `meta` is a full box.
1165    let body = if p.len() >= 8 && &p[4..8] == b"hdlr" { p } else { p.get(4..).unwrap_or(&[]) };
1166    let mut keys: Vec<String> = Vec::new();
1167    if let Some(k) = find(body, b"keys") {
1168        let mut c = Cur::labeled(k, "keys");
1169        if c.full_header().is_ok()
1170            && let Ok(n) = c.count(8)
1171        {
1172            for _ in 0..n {
1173                let Ok(sz) = c.u32() else { break };
1174                let Ok(_ns) = c.u32() else { break };
1175                let Ok(name) = c.bytes((sz as usize).saturating_sub(8)) else { break };
1176                keys.push(String::from_utf8_lossy(name).into_owned());
1177            }
1178        }
1179    }
1180    let Some(ilst) = find(body, b"ilst") else { return };
1181    for item in boxes(ilst).map_while(|b| b.ok()) {
1182        let idx = item.kind.as_u32() as usize;
1183        let named = if idx >= 1 { keys.get(idx - 1) } else { None };
1184        if let Some(d) = find(item.payload, b"data")
1185            && d.len() >= 8
1186        {
1187            let ty = u32::from_be_bytes([d[0], d[1], d[2], d[3]]) & 0xFF_FFFF;
1188            // Charged before the key is copied (every item naming one long key is another copy)
1189            // and before the value is converted, at the lengths they will be stored at.
1190            if ty == 1 || ty == 0 {
1191                let fallback = if named.is_none() { Some(item.kind.to_string_lossy()) } else { None };
1192                let key_len = named.or(fallback.as_ref()).map_or(0, String::len);
1193                if charge_metadata(left, key_len, lossy_len(&d[8..])) {
1194                    let key = named.cloned().or(fallback).unwrap_or_default();
1195                    md.entries.push((key, String::from_utf8_lossy(&d[8..]).into_owned()));
1196                }
1197            }
1198        }
1199    }
