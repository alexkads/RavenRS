1//! Chapter lists, in the two forms players read.
2//!
3//! - **QuickTime chapter track** (QuickTime File Format, "Chapter lists" and "Text media"): a
4//!   disabled text track whose samples are the chapter titles, referenced from a video or audio track
5//!   by a `tref/chap` track reference. QuickTime Player, Apple TV, iOS and Final Cut read this form.
6//!   Each sample is a 16-bit big-endian length, the title, and an `encd` atom with value `0x100`
7//!   marking the text as UTF-8 (the convention Apple's own chapter tracks in `.m4a`/`.m4v` files use;
8//!   plain QuickTime text is Mac Roman).
9//! - **Nero chapter list** `moov/udta/chpl`, version 1: `u32` reserved, `u8` chapter count, then per
10//!   chapter a `u64` start in 100 ns units, a `u8` title length and the UTF-8 title. It has no
11//!   official specification; this is the layout publicly documented for Nero Digital files and
12//!   written by many tools. VLC and ffprobe (`-show_chapters`) read it.
13//!
14//! Both forms carry at most [`MAX_CHAPTERS`] chapters with titles of at most
15//! [`MAX_CHAPTER_TITLE_BYTES`] bytes (the `chpl` count and length fields are one byte each).
16
17use crate::craft::media::isobmff::bytes::{BoxBuf, Cur};
18
19/// One chapter mark: `start` in the movie timescale ([`crate::craft::media::isobmff::WriterOptions::movie_timescale`]
20/// when writing, [`crate::craft::media::isobmff::Mp4File::timescale`] when reading), `title` UTF-8.
21#[derive(Clone, Debug, PartialEq, Eq)]
22pub struct Chapter {
23    pub start: u64,
24    pub title: String,
25}
26
27/// Most chapters written (the `chpl` count is one byte); later ones are dropped.
28pub const MAX_CHAPTERS: usize = 255;
29/// Longest chapter title in bytes (the `chpl` length is one byte); longer titles are cut at a
30/// character boundary.
31pub const MAX_CHAPTER_TITLE_BYTES: usize = 255;
32
33/// `chpl` times are in 100 ns units.
34const CHPL_UNITS_PER_SECOND: u64 = 10_000_000;
35
36/// Cut `s` to at most `max` bytes without splitting a character.
37pub(crate) fn truncate_utf8(s: &str, max: usize) -> &str {
38    if s.len() <= max {
39        return s;
40    }
41    let mut end = max;
42    while end > 0 && !s.is_char_boundary(end) {
43        end -= 1;
44    }
45    s.get(..end).unwrap_or("")
46}

48/// The chapters a file can carry: sorted by start, one per start time (the first title given for
49/// a time wins), starts before `duration` only (when it is known), titles cut to
50/// [`MAX_CHAPTER_TITLE_BYTES`] with control characters replaced by spaces, at most
51/// [`MAX_CHAPTERS`].
52pub fn normalize_chapters(chapters: &[Chapter], duration: Option<u64>) -> Vec<Chapter> {
53    let mut v: Vec<&Chapter> = chapters.iter().filter(|c| duration.is_none_or(|d| c.start < d)).collect();
54    v.sort_by_key(|c| c.start);
55    v.dedup_by_key(|c| c.start);
56    v.into_iter()
57        .take(MAX_CHAPTERS)
58        .map(|c| {
59            let clean: String = c.title.chars().map(|ch| if ch.is_control() { ' ' } else { ch }).collect();
60            Chapter { start: c.start, title: truncate_utf8(clean.trim(), MAX_CHAPTER_TITLE_BYTES).to_string() }
61        })
62        .collect()
63}
64
65fn rescale(v: u64, from: u64, to: u64) -> u64 {
66    if from == 0 {
67        return 0;
68    }
69    (u128::from(v) * u128::from(to) / u128::from(from)).min(u128::from(u64::MAX)) as u64
70}

72/// Write `udta/chpl` content (the box itself) for already normalized chapters.
73pub(crate) fn write_chpl(b: &mut BoxBuf, chapters: &[Chapter], movie_timescale: u32) {
74    let m = b.start_full(b"chpl", 1, 0);
75    b.u32(0);
76    let n = chapters.len().min(MAX_CHAPTERS);
77    b.u8(n as u8);
78    for c in chapters.iter().take(n) {
79        b.u64(rescale(c.start, u64::from(movie_timescale), CHPL_UNITS_PER_SECOND));
80        let t = truncate_utf8(&c.title, MAX_CHAPTER_TITLE_BYTES);
81        b.u8(t.len() as u8);
82        b.bytes(t.as_bytes());
83    }
84    b.end(m);
85}

87/// Parse a `chpl` payload (after the box header) into chapters in `movie_timescale` units.
88/// Malformed or truncated lists yield the chapters read so far.
89pub(crate) fn parse_chpl(p: &[u8], movie_timescale: u32) -> Vec<Chapter> {
90    let mut out = Vec::new();
91    let mut c = Cur::labeled(p, "chpl");
92    let Ok((version, _)) = c.full_header() else { return out };
93    if version >= 1 && c.skip(4).is_err() {
94        return out;
95    }
96    let Ok(n) = c.u8() else { return out };
97    for _ in 0..n {
98        let Ok(start) = c.u64() else { break };
99        let Ok(len) = c.u8() else { break };
100        let Ok(title) = c.bytes(len as usize) else { break };
101        out.push(Chapter {
102            start: rescale(start, CHPL_UNITS_PER_SECOND, u64::from(movie_timescale)),
103            title: String::from_utf8_lossy(title).into_owned(),
104        });
105    }
106    out
107}

109/// Payload of the QuickTime `text` sample description (after its 8-byte box header), QTFF "Text
110/// sample description": reserved + data reference index, display flags, justification, background
111/// colour, default text box, reserved, font number and face, reserved, foreground colour, an empty
112/// Pascal font name, then an `ftab` font table naming one font.
113pub(crate) fn text_sample_entry_payload() -> Vec<u8> {
114    let mut b = BoxBuf::new();
115    b.zeros(6);
116    b.u16(1); // data reference index
117    b.u32(0); // display flags
118    b.i32(0); // justification: left
119    b.zeros(6); // background colour (RGB)
120    b.zeros(8); // default text box (top, left, bottom, right)
121    b.zeros(8); // reserved
122    b.u16(1); // font number (matches the ftab entry)
123    b.u16(0); // font face: plain
124    b.u8(0); // reserved
125    b.u16(0); // reserved
126    b.zeros(6); // foreground colour (RGB)
127    b.u8(0); // text name: empty Pascal string
128    let f = b.start(b"ftab");
129    b.u16(1);
130    b.u16(1);
131    let name = b"Sans-Serif";
132    b.u8(name.len() as u8);
133    b.bytes(name);
134    b.end(f);
135    b.buf
136}

138/// One chapter-track sample: length-prefixed UTF-8 title plus an `encd` atom (UTF-8).
139pub(crate) fn text_sample(title: &str) -> Vec<u8> {
140    let t = truncate_utf8(title, MAX_CHAPTER_TITLE_BYTES);
141    let mut b = BoxBuf::new();
142    b.u16(t.len() as u16);
143    b.bytes(t.as_bytes());
144    let e = b.start(b"encd");
145    b.u32(0x100);
146    b.end(e);
147    b.buf
148}

150/// The title in a QuickTime text sample (UTF-8, or UTF-16 when it starts with a byte order mark).
151pub fn parse_text_sample(data: &[u8]) -> Option<String> {
152    let mut c = Cur::labeled(data, "text sample");
153    let n = c.u16().ok()? as usize;
154    decode_title(c.bytes(n).ok()?)
155}
156
157/// [`parse_text_sample`] of the start of a longer sample: as much of the title as `data` holds.
158pub(crate) fn parse_text_sample_prefix(data: &[u8]) -> Option<String> {
159    let (declared, rest) = data.split_first_chunk::<2>()?;
160    decode_title(rest.get(..usize::from(u16::from_be_bytes(*declared))).unwrap_or(rest))
161}
162
163fn decode_title(t: &[u8]) -> Option<String> {
164    if let Some(rest) = t.strip_prefix(&[0xFE, 0xFF]) {
165        let units: Vec<u16> = rest.as_chunks::<2>().0.iter().map(|u| u16::from_be_bytes(*u)).collect();
166        return Some(String::from_utf16_lossy(&units));
167    }
168    if let Some(rest) = t.strip_prefix(&[0xFF, 0xFE]) {
169        let units: Vec<u16> = rest.as_chunks::<2>().0.iter().map(|u| u16::from_le_bytes(*u)).collect();
170        return Some(String::from_utf16_lossy(&units));
171    }
172    // trailing atoms (`encd`, `styl` …) are not part of the text
173    Some(String::from_utf8_lossy(t).into_owned())
174}

176#[cfg(any())]
177mod tests {
178    use super::*;

180    fn ch(start: u64, title: &str) -> Chapter {
181        Chapter { start, title: title.into() }
182    }
183
184    #[cfg(any())]
185    fn normalize_sorts_dedups_drops_and_caps() {
186        let v = normalize_chapters(&[ch(500, "b"), ch(0, "a"), ch(500, "dup"), ch(2000, "past end"), ch(1999, "\tend\n")], Some(2000));
187        assert_eq!(v, vec![ch(0, "a"), ch(500, "b"), ch(1999, "end")]);
188        let many: Vec<Chapter> = (0..1000).map(|i| ch(i, "x")).collect();
189        assert_eq!(normalize_chapters(&many, None).len(), MAX_CHAPTERS);
190        let long = "é".repeat(300);
191        let t = &normalize_chapters(&[ch(0, &long)], None)[0].title;
192        assert!(t.len() <= MAX_CHAPTER_TITLE_BYTES && t.chars().all(|c| c == 'é'));
193        assert!(normalize_chapters(&[ch(0, "a")], Some(0)).is_empty());
194    }
195
196    #[cfg(any())]
197    fn chpl_roundtrip() {
198        let chapters = vec![ch(0, "Intro"), ch(1500, "Teil 2 — ü"), ch(u64::MAX, "far")];
199        let mut b = BoxBuf::new();
200        write_chpl(&mut b, &chapters, 1000);
201        // box header (8) + version/flags (4) + reserved (4) + count (1)
202        assert_eq!(&b.buf[4..8], b"chpl");
203        assert_eq!(b.buf[8], 1);
204        assert_eq!(b.buf[16], 3);
205        assert_eq!(u64::from_be_bytes(b.buf[17..25].try_into().unwrap()), 0);
206        assert_eq!(u64::from_be_bytes(b.buf[31..39].try_into().unwrap()), 15_000_000);
207        let back = parse_chpl(&b.buf[8..], 1000);
208        assert_eq!(back[..2], chapters[..2]);
209        assert_eq!(back.len(), 3);
210    }
211
212    #[cfg(any())]
213    fn chpl_parse_survives_garbage() {
214        let mut b = BoxBuf::new();
215        write_chpl(&mut b, &[ch(0, "a"), ch(10, "bb")], 1000);
216        let p = b.buf[8..].to_vec();
217        for cut in 0..p.len() {
218            let _ = parse_chpl(&p[..cut], 1000);
219        }
220        let mut lying = p.clone();
221        lying[8] = 255; // count says 255
222        assert_eq!(parse_chpl(&lying, 1000).len(), 2);
223        assert!(parse_chpl(&p, 0).iter().all(|c| c.start == 0));
224    }
225
226    #[cfg(any())]
227    fn text_sample_roundtrip() {
228        let s = text_sample("Kapitel “1”");
229        assert_eq!(parse_text_sample(&s).as_deref(), Some("Kapitel “1”"));
230        assert_eq!(&s[s.len() - 12..s.len() - 4], &[0, 0, 0, 12, b'e', b'n', b'c', b'd']);
231        assert_eq!(parse_text_sample(&[0, 4, 0xFE, 0xFF, 0, b'A']).as_deref(), Some("A"));
232        assert_eq!(parse_text_sample(&[0, 9, b'a']), None);
233        assert_eq!(parse_text_sample(&[]), None);
234        // The start of a longer sample: what it holds of the title, and no further than the title.
235        assert_eq!(parse_text_sample_prefix(&[0, 9, b'a']).as_deref(), Some("a"));
236        assert_eq!(parse_text_sample_prefix(&[0, 1, b'a', b'b']).as_deref(), Some("a"));
237        assert_eq!(parse_text_sample_prefix(&[0]), None);
238    }
