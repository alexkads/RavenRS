1//! Clean-room Ogg demuxer, written from RFC 3533 (the Ogg encapsulation format), RFC 7845 (Ogg
2//! Opus) and the Vorbis I specification's Ogg mapping (section A).
3//!
4//! [`open`] reads every page (capture pattern, header, lacing values, CRC-32 checked; damaged
5//! regions are skipped by searching for the next `OggS`), groups them by logical bitstream and
6//! assembles packets — including packets that continue across pages — as lists of byte ranges,
7//! with the granule position of the page each packet completes on. Codec headers (Opus:
8//! `OpusHead` + `OpusTags`; Vorbis: identification, comment, setup) are collected separately from
9//! the audio packets.
10//!
11//! [`OpusTiming`] implements RFC 7845 §4: packet start positions from granule positions and
12//! TOC-derived durations, the pre-skip, and end trimming of the final page, so a player can seek
13//! to any 48 kHz sample.
14//!
15//! Layer L0: no dependencies beyond `std`, no `unsafe`, builds for `wasm32-unknown-unknown`.
16//!
17//! From FilmCraft's `filmcraft-ogg` (storytold/filmcraft@8fda2f2). Public items keep their names, so
18//! FilmCraft can re-export this module. `ByteSource` is the crate's one source trait (see the
19//! README).

67/// CRC-32 of RFC 3533 §6: polynomial 0x04C11DB7, initial value 0, no reflection, no final XOR.
68pub fn crc32(data: &[u8]) -> u32 {
69    crc32_update(0, data)
70}
71
72/// `CRC_TABLES[k][b]`: the CRC register after feeding byte `b` followed by `k` zero bytes
73/// (slice-by-16: sixteen independent lookups per 16 input bytes instead of a serial chain of 16).
74static CRC_TABLES: [[u32; 256]; 16] = crc_tables();
75
76const fn crc_tables() -> [[u32; 256]; 16] {
77    let mut t = [[0u32; 256]; 16];
78    let mut i = 0;
79    while i < 256 {
80        let mut r = (i as u32) << 24;
81        let mut b = 0;
82        while b < 8 {
83            r = if r & 0x8000_0000 != 0 { (r << 1) ^ 0x04C1_1DB7 } else { r << 1 };
84            b += 1;
85        }
86        t[0][i] = r;
87        i += 1;
88    }
89    let mut k = 1;
90    while k < 16 {
91        let mut i = 0;
92        while i < 256 {
93            let p = t[k - 1][i];
94            t[k][i] = (p << 8) ^ t[0][(p >> 24) as usize];
95            i += 1;
96        }
97        k += 1;
98    }
99    t
100}

102/// Continue an RFC 3533 CRC-32 over `data` from register `crc`.
103pub fn crc32_update(mut crc: u32, data: &[u8]) -> u32 {
104    let t = &CRC_TABLES;
105    let (chunks, rest) = data.as_chunks::<16>();
106    for c in chunks {
107        let w = crc ^ u32::from_be_bytes([c[0], c[1], c[2], c[3]]);
108        crc = t[15][(w >> 24) as usize]
109            ^ t[14][((w >> 16) & 0xFF) as usize]
110            ^ t[13][((w >> 8) & 0xFF) as usize]
111            ^ t[12][(w & 0xFF) as usize]
112            ^ t[11][c[4] as usize]
113            ^ t[10][c[5] as usize]
114            ^ t[9][c[6] as usize]
115            ^ t[8][c[7] as usize]
116            ^ t[7][c[8] as usize]
117            ^ t[6][c[9] as usize]
118            ^ t[5][c[10] as usize]
119            ^ t[4][c[11] as usize]
120            ^ t[3][c[12] as usize]
121            ^ t[2][c[13] as usize]
122            ^ t[1][c[14] as usize]
123            ^ t[0][c[15] as usize];
124    }
125    rest.iter().fold(crc, |c, &b| (c << 8) ^ t[0][((c >> 24) as u8 ^ b) as usize])
126}

128/// The codec of a logical bitstream (from its first packet).
129#[non_exhaustive]
130#[derive(Clone, Copy, Debug, PartialEq, Eq)]
131pub enum Codec {
132    /// Opus (RFC 7845).
133    Opus,
134    /// Vorbis.
135    Vorbis,
136    /// FLAC (the Ogg FLAC mapping).
137    Flac,
138    /// Theora video.
139    Theora,
140    /// Speex.
141    Speex,
142    /// A codec this module does not recognize.
143    Unknown,
144}
145
146impl Codec {
147    fn of(first: &[u8]) -> Codec {
148        if first.starts_with(b"OpusHead") {
149            Codec::Opus
150        } else if first.starts_with(b"\x01vorbis") {
151            Codec::Vorbis
152        } else if first.starts_with(b"\x7fFLAC") {
153            Codec::Flac
154        } else if first.starts_with(b"\x80theora") {
155            Codec::Theora
156        } else if first.starts_with(b"Speex   ") {
157            Codec::Speex
158        } else {
159            Codec::Unknown
160        }
161    }
162    /// Header packets before the audio packets (for FLAC see [`flac_header_count`]).
163    fn header_count(self) -> usize {
164        match self {
165            Codec::Opus => 2,
166            Codec::Vorbis | Codec::Theora => 3,
167            Codec::Speex => 2, // see `speex_header_count`
168            Codec::Flac | Codec::Unknown => 1,
169        }
170    }
171    /// The shared identifier for this codec.
172    pub fn codec_id(self) -> crate::craft::media::CodecId {
173        match self {
174            Codec::Opus => crate::craft::media::CodecId::Opus,
175            Codec::Vorbis => crate::craft::media::CodecId::Vorbis,
176            Codec::Flac => crate::craft::media::CodecId::Flac,
177            Codec::Theora => crate::craft::media::CodecId::Theora,
178            Codec::Speex => crate::craft::media::CodecId::Speex,
179            Codec::Unknown => crate::craft::media::CodecId::Unknown,
180        }
181    }
182    /// A short display name: `"Opus"`, `"Vorbis"`.
183    pub fn name(self) -> &'static str {
184        match self {
185            Codec::Opus => "Opus",
186            Codec::Vorbis => "Vorbis",
187            Codec::Flac => "FLAC",
188            Codec::Theora => "Theora",
189            Codec::Speex => "Speex",
190            Codec::Unknown => "unknown",
191        }
192    }
193}

195/// The most header bytes kept for one stream: room for two of the largest FLAC metadata blocks (a
196/// 4-byte block header and up to 2^24 - 1 bytes, such as a picture) and the rest of the headers.
197const MAX_HEADER_BYTES: usize = 64 << 20;
198/// Bytes of a stream's first packet that identify its codec and header count (the Speex header's
199/// `extra_headers` field ends at 72).
200const FIRST_PACKET_PREFIX: usize = 80;
201
202/// Header packets of an Ogg FLAC stream, from its first packet (0x7F "FLAC", version, a 16-bit
203/// big-endian count of the header packets after it, then `fLaC` and STREAMINFO): that count plus
204/// the first. A count of 0 means unknown: the metadata packets then run to the one whose block
205/// header has the last-block flag, or end with STREAMINFO when its own flag is set.
206fn flac_header_count(first: &[u8]) -> usize {
207    match first.get(7..9).map(|b| u16::from_be_bytes([b[0], b[1]])) {
208        Some(n) if n > 0 => 1 + usize::from(n),
209        _ if first.get(13).is_some_and(|b| b & 0x80 != 0) => 1,
210        _ => usize::MAX,
211    }
212}
213
214/// Header packets of a Speex stream: the Speex header, the comment header and the
215/// `extra_headers` the Speex header states (a 32-bit little-endian field at offset 68).
216fn speex_header_count(first: &[u8]) -> usize {
217    let extra = first.get(68..72).map_or(0, |b| u32::from_le_bytes([b[0], b[1], b[2], b[3]]));
218    2 + usize::try_from(extra).unwrap_or(usize::MAX).min(usize::from(u16::MAX))
219}

266impl Stream {
267    /// What a decoder needs for this stream, in the container-neutral form the codec crates take:
268    /// the `OpusHead`, the Vorbis or Theora headers Xiph-laced, `fLaC` + the FLAC metadata blocks
269    /// (the last one flagged last), or the Speex header.
270    pub fn codec_params(&self) -> crate::craft::media::CodecParams {
271        use crate::craft::media::CodecId as Id;
272        let first = self.headers.first().map(Vec::as_slice).unwrap_or_default();
273        let le_u32 = |at: usize| first.get(at..at + 4).map_or(0, |b| u32::from_le_bytes([b[0], b[1], b[2], b[3]]));
274        match self.codec {
275            Codec::Opus => crate::craft::media::CodecParams::new(Id::Opus)
276                .with_extradata(first)
277                .with_sample_rate(48_000)
278                .with_channels(first.get(9).copied().map_or(0, u16::from)),
279            Codec::Vorbis => crate::craft::media::CodecParams::new(Id::Vorbis)
280                .with_extradata(crate::craft::media::xiph_lace(&self.headers).unwrap_or_default())
281                .with_sample_rate(le_u32(12))
282                .with_channels(first.get(11).copied().map_or(0, u16::from)),
283            // The first packet is 0x7F "FLAC", version, header count, then `fLaC` + STREAMINFO;
284            // the other header packets are metadata blocks.
285            Codec::Flac => {
286                let mut extradata = first.get(9..).unwrap_or_default().to_vec();
287                // block headers: STREAMINFO's after `fLaC`, then one per metadata packet
288                let mut blocks = vec![4];
289                for h in self.headers.iter().skip(1) {
290                    blocks.push(extradata.len());
291                    extradata.extend_from_slice(h);
292                }
293                // exactly the last block is flagged last, also when an audio frame ended the
294                // metadata of a stream that does not state its header count
295                let blocks: Vec<usize> = blocks.into_iter().filter(|&at| at < extradata.len()).collect();
296                for (k, &at) in blocks.iter().enumerate() {
297                    if k + 1 == blocks.len() {
298                        extradata[at] |= 0x80;
299                    } else {
300                        extradata[at] &= 0x7F;
301                    }
302                }
303                // STREAMINFO follows `fLaC` and its 4-byte block header: 10 bytes of block and frame
304                // sizes, then a 20-bit sample rate and 3 bits of channels less one.
305                let info = first.get(27..30).map_or(0, |b| (u32::from(b[0]) << 16) | (u32::from(b[1]) << 8) | u32::from(b[2]));
306                let channels = if info == 0 { 0 } else { ((info >> 1) & 7) as u16 + 1 };
307                crate::craft::media::CodecParams::new(Id::Flac).with_extradata(extradata).with_sample_rate(info >> 4).with_channels(channels)
308            }
309            // the identification header: version, the frame size in macroblocks, then the 24-bit
310            // big-endian picture width and height at offsets 14 and 17
311            Codec::Theora => {
312                let be_u24 = |at: usize| first.get(at..at + 3).map_or(0, |b| (u32::from(b[0]) << 16) | (u32::from(b[1]) << 8) | u32::from(b[2]));
313                crate::craft::media::CodecParams::new(Id::Theora)
314                    .with_extradata(crate::craft::media::xiph_lace(&self.headers).unwrap_or_default())
315                    .with_size(be_u24(14), be_u24(17))
316            }
317            // the Speex header: rate at offset 36 and channels at 48, 32-bit little-endian
318            Codec::Speex => crate::craft::media::CodecParams::new(Id::Speex)
319                .with_extradata(first)
320                .with_sample_rate(le_u32(36))
321                .with_channels(u16::try_from(le_u32(48)).unwrap_or(0)),
322            Codec::Unknown => crate::craft::media::CodecParams::new(Id::Unknown),
323        }
324    }
325}

327impl Stream {
328    /// File a completed packet: the first one sets the codec and how many headers follow it, then
329    /// header packets are read in, and the rest are audio.
330    fn add_packet(&mut self, src: &(impl ByteSource + ?Sized), pk: Packet, warnings: &mut Warnings) -> Result<()> {
331        let size = pk.size as usize;
332        // the first packet is read once: for its codec, and then as a header (a packet too large
333        // to be one is only read as far as the codec needs)
334        let mut first = None;
335        if self.headers.is_empty() && self.packets.is_empty() {
336            let data = read_parts(src, &pk.parts, if size <= MAX_HEADER_BYTES { size } else { FIRST_PACKET_PREFIX })?;
337            self.codec = Codec::of(&data);
338            self.header_count = match self.codec {
339                Codec::Flac => flac_header_count(&data),
340                Codec::Speex => speex_header_count(&data),
341                c => c.header_count(),
342            };
343            first = Some(data).filter(|d| d.len() == size);
344        }
345        if self.headers.len() >= self.header_count || !self.packets.is_empty() {
346            self.packets.push(pk);
347            return Ok(());
348        }
349        // a count the file states (FLAC, Speex) or leaves open is capped, so a hostile first
350        // packet cannot turn the audio into headers; a packet past the cap is not read
351        if self.headers.len() >= MAX_HEADERS || self.header_bytes.saturating_add(size) > MAX_HEADER_BYTES {
352            warnings.push(format!("stream {}: header packets beyond {MAX_HEADERS} or {MAX_HEADER_BYTES} bytes read as data", self.serial));
353            self.header_count = self.headers.len();
354            self.packets.push(pk);
355            return Ok(());
356        }
357        let data = match first {
358            Some(data) => data,
359            None => read_parts(src, &pk.parts, size)?,
360        };
361        if self.header_count == usize::MAX && !self.headers.is_empty() {
362            // FLAC with an unstated count: metadata packets run to the one flagged last. A packet
363            // that is no metadata block (an audio frame starts with 0xFF) ends them, so audio is
364            // never held as headers.
365            if !is_flac_metadata_block(&data) {
366                self.header_count = self.headers.len();
367                self.packets.push(pk);
368                return Ok(());
369            }
370            if data[0] & 0x80 != 0 {
371                self.header_count = self.headers.len() + 1;
372            }
373        }
374        self.header_bytes = self.header_bytes.saturating_add(data.len());
375        self.headers.push(data);
376        Ok(())
377    }
378}

396/// Bytes the damage-recovery scan may read in one call.
397const SCAN_CHUNK: usize = 64 << 10;
398/// Bytes damage recovery may read again beyond twice the file's length (see [`Recovery`]).
399const RECOVERY_ALLOWANCE: u64 = 1 << 20;
400/// Cap on the packets kept from one file, all streams and headers together.
401const MAX_PACKETS: usize = 1 << 26;
402/// Bytes of file each kept packet needs beyond the first `MIN_PACKET_ALLOWANCE`. A kept packet
403/// takes about 60 bytes of memory and a zero lacing value describes one in a single byte, so
404/// without this a file could make the packet lists about 60 times its own size.
405const FILE_BYTES_PER_PACKET: u64 = 8;
406/// Packets any file may keep, however short.
407const MIN_PACKET_ALLOWANCE: usize = 1 << 20;
408/// The most header packets kept for one stream, whatever count its first packet states.
409const MAX_HEADERS: usize = 1024;
410/// Warnings kept about damaged pages; the rest are counted in one closing warning.
411const MAX_DAMAGE_WARNINGS: usize = 1024;
412
413/// The warnings of [`open`]: those about damaged pages are capped at `MAX_DAMAGE_WARNINGS`, so a
414/// file of damage cannot make a warning per byte.
415#[derive(Default)]
416struct Warnings {
417    list: Vec<String>,
418    damage: usize,
419}
420
421impl Warnings {
422    fn push(&mut self, warning: String) {
423        self.list.push(warning);
424    }
425    fn damage(&mut self, warning: impl FnOnce() -> String) {
426        if self.damage < MAX_DAMAGE_WARNINGS {
427            self.list.push(warning());
428        }
429        self.damage += 1;
430    }
431    fn finish(mut self) -> Vec<String> {
432        if let Some(more) = self.damage.checked_sub(MAX_DAMAGE_WARNINGS).filter(|&n| n > 0) {
433            self.list.push(format!("{more} more damaged regions not listed"));
434        }
435        self.list
436    }
437}

439/// The cost of damage recovery. Every read of a page or a scan in [`open`] is charged here: the
440/// first read of a byte is free, and bytes read again (scanning through the body of a page that
441/// failed its CRC, a scan's read past the pattern it found, the overlapping bodies of fake page
442/// headers) are charged to a budget of twice the file's length plus `RECOVERY_ALLOWANCE`. Pages
443/// are then read at most about three times over, and damage costs a large file no more, in
444/// proportion, than a small one. Header packets are read once more when they are filed; packets do
445/// not overlap, so that adds at most the file's length.
446struct Recovery {
447    budget: u64,
448    /// Bytes before this offset have been read.
449    seen: u64,
450}
451
452impl Recovery {
453    fn new(len: u64) -> Recovery {
454        Recovery { budget: len.saturating_mul(2).saturating_add(RECOVERY_ALLOWANCE), seen: 0 }
455    }
456    /// Charge reading `start..end`; false when the budget does not cover it.
457    fn charge(&mut self, start: u64, end: u64) -> bool {
458        let again = self.seen.min(end).saturating_sub(start);
459        self.seen = self.seen.max(end);
460        match self.budget.checked_sub(again) {
461            Some(left) => {
462                self.budget = left;
463                true
464            }
465            None => {
466                self.budget = 0;
467                false
468            }
469        }
470    }
471}

473/// Where a damage-recovery scan stopped.
474#[derive(Debug, PartialEq, Eq)]
475enum Capture {
476    /// The next capture pattern starts here.
477    At(u64),
478    /// The source ended first.
479    End,
480    /// The file's recovery budget ran out first.
481    BudgetSpent,
482}

484/// Read every page and assemble the packets of every logical bitstream.
485pub fn open(src: &(impl ByteSource + ?Sized)) -> Result<OggFile> {
486    let len = src.len();
487    let mut streams: Vec<Stream> = Vec::new();
488    let mut by_serial: HashMap<u32, usize> = HashMap::new();
489    let mut partial: HashMap<u32, Partial> = HashMap::new();
490    let mut warnings = Warnings::default();
491    let mut pos = 0u64;
492    let mut found_any = false;
493    let mut recovery = Recovery::new(len);
494    let mut packet_budget = usize::try_from(len / FILE_BYTES_PER_PACKET).unwrap_or(usize::MAX).clamp(MIN_PACKET_ALLOWANCE, MAX_PACKETS);
495    let mut hdr = [0u8; 27 + 255];
496    while pos + 27 <= len {
497        src.read_at(pos, &mut hdr[..27])?;
498        if !recovery.charge(pos, pos + 27) {
499            warnings.push(past_budget(pos));
500            break;
501        }
502        if &hdr[..4] != b"OggS" || hdr[4] != 0 {
503            match find_capture(src, pos + 1, len, &mut recovery)? {
504                Capture::At(p) => {
505                    if found_any {
506                        warnings.damage(|| format!("lost sync at byte {pos}; resumed at {p}"));
507                    }
508                    pos = p;
509                    continue;
510                }
511                Capture::End => break,
512                Capture::BudgetSpent => {
513                    warnings.push(past_budget(pos));
514                    break;
515                }
516            }
517        }
518        let nseg = hdr[26] as usize;
519        let head = 27 + nseg;
520        let page_len = if pos + head as u64 <= len {
521            src.read_at(pos + 27, &mut hdr[27..head])?;
522            head as u64 + hdr[27..head].iter().map(|&b| b as u64).sum::<u64>()
523        } else {
524            u64::MAX
525        };
526        let fits = pos.saturating_add(page_len) <= len;
527        // the lacing values and the body: fake page headers a few bytes apart would otherwise
528        // each have their overlapping bodies read and hashed for free
529        if !recovery.charge(pos + 27, if fits { pos + page_len } else { (pos + head as u64).min(len) }) {
530            warnings.push(past_budget(pos));
531            break;
532        }
533        if !fits {
534            // truncated, or a damaged lacing value: look for a page further on
535            warnings.damage(|| format!("page at byte {pos} runs past the end of the file"));
536            pos += 1;
537            continue;
538        }
539        let mut page = vec![0u8; page_len as usize];
540        page[..head].copy_from_slice(&hdr[..head]);
541        src.read_at(pos + head as u64, &mut page[head..])?;
542        let stored_crc = u32::from_le_bytes([page[22], page[23], page[24], page[25]]);
543        page[22..26].fill(0);
544        if crc32(&page) != stored_crc {
545            warnings.damage(|| format!("bad CRC on page at byte {pos}"));
546            pos += 1;
547            continue;
548        }
549        found_any = true;
550        let flags = page[5];
551        let granule = i64::from_le_bytes(page[6..14].try_into().unwrap_or([0; 8]));
552        let serial = u32::from_le_bytes([page[14], page[15], page[16], page[17]]);
553        let seq = u32::from_le_bytes([page[18], page[19], page[20], page[21]]);
554        let lacing = page[27..27 + nseg].to_vec();
555        let body_off = pos + 27 + nseg as u64;
556        // a beginning-of-stream page for a serial already seen starts the next link of a chained
557        // file (RFC 3533 §4), which may reuse the serial; one that carries on the page sequence of
558        // a stream with no end-of-stream page yet is a stray flag in that stream
559        if flags & 0x02 != 0
560            && let Some(&si) = by_serial.get(&serial)
561        {
562            let st = &streams[si];
563            if !st.saw_eos && st.next_seq == Some(seq) {
564                warnings.damage(|| format!("stray beginning-of-stream flag on page at byte {pos}, read as part of stream {serial}"));
565            } else {
566                if !st.saw_eos {
567                    warnings.damage(|| format!("stream {serial} starts again at byte {pos} before its end-of-stream page"));
568                }
569                by_serial.remove(&serial);
570                if partial.remove(&serial).is_some() {
571                    warnings.damage(|| format!("incomplete packet before page at byte {pos}"));
572                }
573            }
574        }
575        let si = *by_serial.entry(serial).or_insert_with(|| {
576            streams.push(Stream {
577                serial,
578                codec: Codec::Unknown,
579                headers: Vec::new(),
580                packets: Vec::new(),
581                pages: 0,
582                last_granule: None,
583                saw_eos: false,
584                header_count: 1,
585                header_bytes: 0,
586                next_seq: None,
587            });
588            streams.len() - 1
589        });
590        let st = &mut streams[si];
591        // a gap in the page sequence numbers is a lost page: a packet it continued is incomplete
592        if st.next_seq.is_some_and(|n| n != seq) {
593            warnings.damage(|| format!("pages missing before page at byte {pos}"));
594            partial.remove(&serial);
595        }
596        st.next_seq = Some(seq.wrapping_add(1));
597        let page_index = st.pages;
598        st.pages += 1;
599        if flags & 0x04 != 0 {
600            st.saw_eos = true;
601        }
602        // a page that does not continue a pending packet (a page was lost): drop the packet
603        let continued = flags & 0x01 != 0;
604        if !continued && partial.remove(&serial).is_some() {
605            warnings.damage(|| format!("incomplete packet before page at byte {pos}"));
606        }
607        let mut completed: Vec<Packet> = Vec::new();
608        let mut off = 0u64;
609        let mut seg_start = 0u64;
610        // a continued page without a pending packet (a page was lost): drop the fragment
611        let mut skip_fragment = continued && !partial.contains_key(&serial);
612        for (k, &l) in lacing.iter().enumerate() {
613            off += l as u64;
614            if l < 255 || k + 1 == nseg {
615                let (a, n) = (seg_start, off - seg_start);
616                seg_start = off;
617                if skip_fragment {
618                    if l < 255 {
619                        skip_fragment = false;
620                    }
621                    continue;
622                }
623                let p = partial.entry(serial).or_insert_with(|| Partial { parts: Vec::new(), size: 0, prefix: Vec::new() });
624                if n > 0 {
625                    let Some(size) = p.size.checked_add(n as u32) else {
626                        warnings.push(format!("packet over 4 GiB dropped at byte {pos}"));
627                        partial.remove(&serial);
628                        skip_fragment = l == 255;
629                        continue;
630                    };
631                    p.parts.push((body_off + a, n as u32));
632                    p.size = size;
633                    let s = (body_off - pos + a) as usize;
634                    let need = 2usize.saturating_sub(p.prefix.len()).min(n as usize);
635                    p.prefix.extend_from_slice(&page[s..s + need]);
636                }
637                if l < 255
638                    && let Some(p) = partial.remove(&serial)
639                {
640                    let mut prefix = [0u8; 2];
641                    prefix[..p.prefix.len().min(2)].copy_from_slice(&p.prefix[..p.prefix.len().min(2)]);
642                    completed.push(Packet { parts: p.parts, size: p.size, granule: None, page: page_index, eos: flags & 0x04 != 0, prefix });
643                }
644            }
645        }
646        let Some(left) = packet_budget.checked_sub(completed.len()) else {
647            warnings.push(format!("stopped at byte {pos}: more packets than a file of this size can hold"));
648            break;
649        };
650        packet_budget = left;
651        if granule != -1 {
652            st.last_granule = Some(granule);
653            if let Some(last) = completed.last_mut() {
654                last.granule = Some(granule);
655            }
656        }
657        for pk in completed {
658            st.add_packet(src, pk, &mut warnings)?;
659        }
660        pos += page_len;
661    }
662    if !found_any {
663        return Err(Error::NotOgg);
664    }
665    if !partial.is_empty() {
666        warnings.push("file ends inside a packet".into());
667    }
668    Ok(OggFile { streams, warnings: warnings.finish() })
669}

671fn past_budget(pos: u64) -> String {
672    format!("damage at byte {pos}; the rest of the file is past the recovery budget")
673}

675/// A FLAC metadata block: a block type other than the invalid 127 (which an audio frame's 0xFF sync
676/// byte would read as; types 7–126 are reserved but allowed) and as many bytes as its 4-byte
677/// header says.
678fn is_flac_metadata_block(data: &[u8]) -> bool {
679    match data {
680        [kind, a, b, c, body @ ..] => kind & 0x7F != 0x7F && body.len() == (usize::from(*a) << 16 | usize::from(*b) << 8 | usize::from(*c)),
681        _ => false,
682    }
683}

685/// Scan for the next capture pattern from `from`, charging the bytes read to `recovery`. Reads start small and double up to `SCAN_CHUNK`, so a nearby pattern costs a
686/// small read.
687fn find_capture(src: &(impl ByteSource + ?Sized), from: u64, len: u64, recovery: &mut Recovery) -> Result<Capture> {
688    let mut at = from;
689    let mut chunk = 256;
690    let mut buf = Vec::new();
691    while at + 4 <= len {
692        // clamped before the cast, so a span of 2^32 bytes cannot truncate to 0 on 32-bit targets
693        let positions = (len - at - 3).min(chunk as u64) as usize;
694        let n = positions + 3;
695        buf.resize(n, 0);
696        src.read_at(at, &mut buf[..n])?;
697        if !recovery.charge(at, at + n as u64) {
698            return Ok(Capture::BudgetSpent);
699        }
700        if let Some(i) = buf[..n].windows(4).position(|w| w == b"OggS") {
701            return Ok(Capture::At(at + i as u64));
702        }
703        at += positions as u64;
704        chunk = (chunk * 2).min(SCAN_CHUNK);
705    }
706    Ok(Capture::End)
707}

709/// The first `max` bytes of a packet's parts.
710fn read_parts(src: &(impl ByteSource + ?Sized), parts: &[(u64, u32)], max: usize) -> Result<Vec<u8>> {
711    // Ranges come from a scan of some source; check them against this one before sizing a buffer.
712    let len = src.len();
713    if parts.iter().any(|&(o, n)| o.checked_add(u64::from(n)).is_none_or(|end| end > len)) {
714        return Err(Error::Invalid("packet range lies outside the source".into()));
715    }
716    let mut v = Vec::with_capacity(parts.iter().map(|p| p.1 as usize).sum::<usize>().min(max));
717    for &(o, n) in parts {
718        let a = v.len();
719        if a == max {
720            break;
721        }
722        v.resize(a + (n as usize).min(max - a), 0);
723        src.read_at(o, &mut v[a..])?;
724    }
725    Ok(v)
726}

728impl OggFile {
729    /// The first stream of `codec`.
730    pub fn stream_of(&self, codec: Codec) -> Option<usize> {
731        self.streams.iter().position(|s| s.codec == codec)
732    }
733    /// The bytes of packet `i` of stream `s`.
734    pub fn read_packet(&self, src: &(impl ByteSource + ?Sized), s: usize, i: usize) -> Result<Vec<u8>> {
735        let p = self.streams.get(s).and_then(|st| st.packets.get(i)).ok_or_else(|| Error::Invalid(format!("no packet {i} in stream {s}")))?;
736        read_parts(src, &p.parts, usize::MAX)
737    }
738}

740/// Duration of an Opus packet in 48 kHz samples from its TOC (RFC 6716 §3.1); `None` if invalid.
741pub fn opus_packet_samples(prefix: [u8; 2], size: u32) -> Option<u32> {
742    if size == 0 {
743        return None;
744    }
745    let toc = prefix[0];
746    let config = toc >> 3;
747    let frame = match config {
748        0..=11 => [480, 960, 1920, 2880][(config & 3) as usize],
749        12..=15 => [480, 960][(config & 1) as usize],
750        _ => [120, 240, 480, 960][(config & 3) as usize],
751    };
752    let frames = match toc & 3 {
753        0 => 1,
754        1 | 2 => 2,
755        _ => {
756            if size < 2 {
757                return None;
758            }
759            (prefix[1] & 0x3F) as u32
760        }
761    };
762    let n = frame * frames;
763    (n > 0 && n <= 5760).then_some(n)
764}

766/// RFC 7845 §4 timing of an Opus stream.
767#[non_exhaustive]
768#[derive(Clone, Debug, PartialEq, Eq)]
769pub struct OpusTiming {
770    /// Samples to drop at the start (the `OpusHead` pre-skip).
771    pub pre_skip: u32,
772    /// Start of each packet's decoded output in output samples (48 kHz): 0 is the first sample
773    /// after the pre-skip; the pre-skip samples have negative positions.
774    pub starts: Vec<i64>,
775    /// Decoded samples of each packet.
776    pub durations: Vec<u32>,
777    /// Playable samples (end trimming applied).
778    pub total: i64,
779}
780
781/// Largest granule position [`OpusTiming::of`] trusts: 2^62 samples at 48 kHz is about 3 million years.
782pub const MAX_OPUS_GRANULE: i64 = 1 << 62;
783
784impl OpusTiming {
785    /// Timing of the Opus stream `s` (`pre_skip` from its `OpusHead`).
786    ///
787    /// A granule position outside `0..=MAX_OPUS_GRANULE` (a negative value other than -1, or an
788    /// absurdly large one) is treated like a missing granule: the page's packets continue from the
789    /// previous position instead of jumping to a nonsense one.
790    pub fn of(stream: &Stream, pre_skip: u32) -> OpusTiming {
791        // Granule positions come straight from the file, so the arithmetic saturates: a hostile
792        // value gives wrong positions, never an overflow.
793        let durations: Vec<u32> = stream.packets.iter().map(|p| opus_packet_samples(p.prefix, p.size).unwrap_or(0)).collect();
794        let n = durations.len();
795        let mut ends: Vec<Option<i64>> = vec![None; n];
796        let mut i = 0;
797        let mut cursor: Option<i64> = None;
798        let mut trimmed_end: Option<i64> = None;
799        let granule_of = |k: usize| stream.packets.get(k).and_then(|p| p.granule).filter(|g| (0..=MAX_OPUS_GRANULE).contains(g));
800        while i < n {
801            // packets completing on the same page: i..j, the last carries the page granule
802            let mut j = i;
803            while j < n && granule_of(j).is_none() {
804                j += 1;
805            }
806            if j == n {
807                // trailing packets without a granule (truncated file): continue from the cursor
808                let mut c = cursor.unwrap_or(0);
809                for k in i..n {
810                    c = c.saturating_add(i64::from(durations[k]));
811                    ends[k] = Some(c);
812                }
813                // the total counts these packets too
814                cursor = Some(c);
815                break;
816            }
817            let Some(g) = granule_of(j) else { break };
818            let sum = durations[i..=j].iter().fold(0i64, |a, &d| a.saturating_add(i64::from(d)));
819            let eos = stream.packets[j].eos;
820            let end = match cursor.map(|c| c.saturating_add(sum)) {
821                // end trimming on the last page (RFC 7845 §4.5): keep the forward positions
822                Some(f) if eos && g < f => {
823                    trimmed_end = Some(g);
824                    f
825                }
826                Some(f) if f == g => f,
827                // the only page with audio is also the last: its packets start at 0 and the
828                // shortfall is trimmed at the end (RFC 7845 §4.5)
829                None if eos && g < sum => {
830                    trimmed_end = Some(g);
831                    sum
832                }
833                // first page, or a discontinuity: trust the granule
834                _ => g,
835            };
836            let mut e = end;
837            for k in (i..=j).rev() {
838                ends[k] = Some(e);
839                e = e.saturating_sub(i64::from(durations[k]));
840            }
841            cursor = Some(end);
842            i = j + 1;
843        }
844        let raw_starts: Vec<i64> = ends.iter().zip(&durations).map(|(e, &d)| e.unwrap_or(0).saturating_sub(i64::from(d))).collect();
845        let base = raw_starts.first().copied().unwrap_or(0).max(0);
846        let shift = base.saturating_add(i64::from(pre_skip));
847        let starts: Vec<i64> = raw_starts.iter().map(|s| s.saturating_sub(shift)).collect();
848        let end_all = trimmed_end.or(cursor).unwrap_or(0);
849        let total = end_all.saturating_sub(shift).max(0);
850        OpusTiming { pre_skip, starts, durations, total }
851    }
852}

855#[non_exhaustive]

860    /// Samples per second.
861    pub sample_rate: u32,
862    /// Nominal bit rate in bits per second (0 or negative when not set).
863    pub bitrate_nominal: i32,
864}

866impl VorbisInfo {
867    /// Reads the Vorbis identification header; `None` when it is not one.
868    pub fn parse(h: &[u8]) -> Option<VorbisInfo> {
869        if h.len() < 30 || !h.starts_with(b"\x01vorbis") {

871        }

874            sample_rate: u32::from_le_bytes([h[12], h[13], h[14], h[15]]),
875            bitrate_nominal: i32::from_le_bytes([h[20], h[21], h[22], h[23]]),
876        })
877    }
