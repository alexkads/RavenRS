1//! Which link targets from a document may become clickable links in a written PDF.
2
3/// Whether a link target from a (possibly untrusted) document may be opened by whoever opens
4/// the file: `http`/`https` URLs with a host, and `mailto` with an address.
5///
6/// Everything else is refused: `file:`, `javascript:`, `data:`, relative paths, network
7/// shares (`\\host\share`, `//host/share`, `smb:`) and other schemes, which an operating system
8/// may hand to a protocol handler. Leading and trailing whitespace is ignored; schemes compare
9/// case-insensitively.
10pub fn safe_url(url: &str) -> bool {
11    let u = url.trim();
12    let Some((scheme, rest)) = u.split_once(':') else { return false };
13    match scheme.to_ascii_lowercase().as_str() {
14        "http" | "https" => rest.strip_prefix("//").is_some_and(|r| !host(r).is_empty()),
15        "mailto" => has_recipient(rest),
16        _ => false,
17    }
18}
19
20/// The host of an authority-first URL remainder: before the path, query or fragment, after any
21/// userinfo, without the port.
22fn host(rest: &str) -> &str {
23    let authority = rest.split(['/', '?', '#', '\\']).next().unwrap_or("");
24    let host_port = authority.rsplit_once('@').map_or(authority, |(_, h)| h);
25    match host_port.strip_prefix('[') {
26        Some(v6) => v6.split_once(']').map_or("", |(h, _)| h),
27        None => host_port.split(':').next().unwrap_or(""),
28    }
29}
30
31/// Whether a `mailto:` remainder names an address, before its headers or in a `to` header.
32fn has_recipient(rest: &str) -> bool {
33    let (to, headers) = rest.split_once('?').unwrap_or((rest, ""));
34    !to.trim().is_empty() || headers.split('&').any(|h| h.split_once('=').is_some_and(|(k, v)| k.eq_ignore_ascii_case("to") && !v.trim().is_empty()))
35}

37#[cfg(any())]
38mod tests {
39    use super::safe_url;
40
41    #[cfg_attr(target_arch = "wasm32", wasm_bindgen_test::wasm_bindgen_test)]
42    #[cfg_attr(not(target_arch = "wasm32"), test)]
43    fn only_web_and_mail_links_are_safe_to_open() {
44        for url in [
45            "https://example.com",
46            " HTTP://example.com/x",
47            "https://user@example.com:8443/x?q#f",
48            "https://[::1]/",
49            "mailto:someone@example.com",
50            "mailto:?to=someone@example.com&subject=hi",
51        ] {
52            assert!(safe_url(url), "{url}");
53        }
54        let refused = [
55            "https://",
56            "https://?q=x",
57            "https:///x",
58            "https://user@/x",
59            "https://:80/",
60            "mailto:",
61            "mailto:?subject=hello",
62            "file:///C:/notes.pdf",
63            "notes.pdf",
64            "javascript:alert(1)",
65            "\t JavaScript:alert(1)",
66            "java\nscript:alert(1)",
67            "vbscript:MsgBox(1)",
68            "data:text/html,<script>alert(1)</script>",
69            "ms-msdt:/id PCWDiagnostic",
70            "ms-word:ofe|u|\\\\attacker.example\\share\\x.docx",
71            "ldap://attacker.example/x",
72            "//attacker.example/share",
73            "\\\\attacker.example\\share",
74            "file://attacker.example/share/x",
75            "file:////attacker.example/share/x",
76            "smb://attacker.example/share",
77            "search-ms:query=x&crumb=location:\\\\attacker.example\\share",
78            "search:query=x&crumb=location:\\\\attacker.example\\share",
79        ];
80        for url in refused {
81            assert!(!safe_url(url), "{url}");
82        }
83    }
