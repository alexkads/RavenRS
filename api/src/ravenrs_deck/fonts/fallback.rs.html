1//! Font fallback per grapheme cluster, and ligature caret positions.
2//!
3//! The cascade for a cluster (a base character with its marks, never split across faces) is:
4//! 1. the faces the document asks for, in order (the run's slot typeface, then the theme's script
5//!    font, `a:font[@script="Arab"]`);
6//! 2. for scripts with a platform cascade (Arabic, Hebrew, Thai, Devanagari): the craft-fonts face
7//!    bundled for the script (Noto Naskh Arabic for Arabic), so every machine draws the same; then
8//!    openly licensed or OS-native families found by name among the installed fonts (no platform
9//!    API is called); then any installed face covering the cluster;
10//! 3. any loaded face (bundled first: Japanese keeps its craft-fonts faces), then the first
11//!    requested face (drawn as .notdef).
12

20/// Variation selector 15: text presentation.
21const TEXT_SELECTOR: char = '\u{FE0E}';
22/// Combining enclosing keycap.
23const KEYCAP: char = '\u{20E3}';

25/// Characters that never decide a face: they're drawn by whatever face the cluster gets.
26fn ignorable(c: char) -> bool {
27    c.is_whitespace()
28        || c.is_control()
29        || matches!(c as u32, 0x200B..=0x200F | 0x202A..=0x202E | 0x2060..=0x2069 | 0xFE00..=0xFE0F | 0xFEFF | 0xE0100..=0xE01EF)
30}

32/// Is the cluster an emoji drawn in colour: a pictograph that defaults to emoji presentation
33/// (Unicode's Emoji_Presentation, including BMP ones such as ⭐ ✅ ☔ ⌚ ⏩), or a sequence with an
34/// emoji presentation selector (U+FE0F), a zero-width joiner between pictographs, a skin tone
35/// modifier or regional indicators (flags)? Keycaps ("1️⃣") are not: PowerPoint draws them as text
36/// (an outlined key), as it draws ♥ ✔ ☎ ⬆ ▶ without a selector.
37pub fn is_emoji_cluster(cluster: &[char]) -> bool {
38    if cluster.contains(&'\u{FE0E}') || cluster.contains(&'\u{20E3}') {
39        return false;
40    }
41    let pictographic = |c: char| matches!(c as u32, 0x1F000..=0x1FAFF | 0x2600..=0x27BF | 0x2B00..=0x2BFF | 0x2190..=0x21FF | 0x2300..=0x23FF);
42    let default_emoji = |c: char| {
43        matches!(
44            c as u32,
45            0x1F300..=0x1F64F
46                | 0x1F680..=0x1F6FF
47                | 0x1F900..=0x1F9FF
48                | 0x1FA70..=0x1FAFF
49                | 0x1F1E6..=0x1F1FF
50                | 0x1F004
51                | 0x1F0CF
52                | 0x1F18E
53                | 0x1F191..=0x1F19A
54                | 0x1F201
55                | 0x1F21A
56                | 0x1F22F
57                | 0x1F232..=0x1F236
58                | 0x1F238..=0x1F23A
59                | 0x1F250..=0x1F251
60                | 0x231A..=0x231B
61                | 0x23E9..=0x23EC
62                | 0x23F0
63                | 0x23F3
64                | 0x25FD..=0x25FE
65                | 0x2614..=0x2615
66                | 0x2648..=0x2653
67                | 0x267F
68                | 0x2693
69                | 0x26A1
70                | 0x26AA..=0x26AB
71                | 0x26BD..=0x26BE
72                | 0x26C4..=0x26C5
73                | 0x26CE
74                | 0x26D4
75                | 0x26EA
76                | 0x26F2..=0x26F3
77                | 0x26F5
78                | 0x26FA
79                | 0x26FD
80                | 0x2705
81                | 0x270A..=0x270B
82                | 0x2728
83                | 0x274C
84                | 0x274E
85                | 0x2753..=0x2755
86                | 0x2757
87                | 0x2795..=0x2797
88                | 0x27B0
89                | 0x27BF
90                | 0x2B1B..=0x2B1C
91                | 0x2B50
92                | 0x2B55
93        )
94    };
95    cluster.iter().any(|&c| default_emoji(c))
96        || (cluster.contains(&'\u{FE0F}') && cluster.iter().any(|&c| pictographic(c)))
97        || (cluster.contains(&'\u{200D}') && cluster.iter().any(|&c| pictographic(c)))
98}

100/// Does `face` draw every character of the cluster?
101pub fn covers_cluster(face: &FontFace, cluster: &[char]) -> bool {
102    cluster.iter().all(|c| ignorable(*c) || face.covers(*c))
103}

105/// Installed families tried by name for a script after the bundled face, before scanning every
106/// installed font. Microsoft font names are not listed (AGENTS.md §1.1); an installed one is
107/// still used by the scan when nothing earlier covers the cluster.
108fn os_families(script: ScriptTag) -> &'static [&'static str] {
109    match &script {
110        b"Arab" => &["Noto Naskh Arabic", "Noto Sans Arabic", "Geeza Pro", "SF Arabic", "DejaVu Sans"],
111        b"Hebr" => &["Noto Sans Hebrew", "Arial Hebrew", "SF Hebrew", "DejaVu Sans"],
112        b"Thai" => &["Noto Sans Thai", "Thonburi"],
113        b"Deva" => &["Noto Sans Devanagari", "Kohinoor Devanagari", "Devanagari Sangam MN"],
114        _ => &[],
115    }
116}

118/// The craft-fonts families for a script (manifest order), e.g. Noto Naskh Arabic for `Arab`.
119pub fn bundled_families(script: ScriptTag) -> Vec<&'static str> {
120    let tag = std::str::from_utf8(&script).unwrap_or("");
121    let mut v: Vec<&'static str> = CRAFT_FONTS.iter().filter(|f| f.scripts.contains(&tag)).map(|f| f.family).collect();
122    v.dedup();
123    v
124}

126impl FontDb {
127    /// The face for one grapheme cluster: the first of `chain` that draws all of it, else the
128    /// bundled face for its script, else an installed face for the script, else any loaded
129    /// face that draws it, else `chain[0]` (or the default face if `chain` is empty).
130    pub fn cascade(&self, cluster: &[char], chain: &[Arc<FontFace>], style: &str) -> Arc<FontFace> {
131        // Emoji take a colour face (with its ZWJ sequences, skin tones and flags) before the
132        // document's fonts, as PowerPoint draws them, even when a text font has a plain glyph.
133        if is_emoji_cluster(cluster) {
134            if let Some(f) = chain.iter().find(|f| f.is_color() && covers_cluster(f, cluster)) {
135                return f.clone();
136            }
137            if let Some(key) = cluster.iter().copied().find(|c| !ignorable(*c))
138                && let Some(f) = self.color_face_for(key, cluster)
139            {
140                return f;
141            }
142        }
143        if let Some(f) = chain.iter().find(|f| covers_cluster(f, cluster)) {
144            return f.clone();
145        }
146        // Text presentation (U+FE0E) and keycaps: a plain face, not a colour emoji one.
147        let text_style = cluster.contains(&TEXT_SELECTOR) || cluster.contains(&KEYCAP);
148        if text_style
149            && let Some(key) = cluster.iter().copied().find(|c| !ignorable(*c))
150            && let Some(f) = self.plain_face_for(key, cluster)
151        {
152            return f;
153        }
154        let first = || chain.first().cloned().unwrap_or_else(|| self.face(crate::fonts::DEFAULT_FAMILY, style));
155        let Some(key) = cluster.iter().copied().find(|c| !ignorable(*c)) else { return first() };
156        let script = cluster.iter().find_map(|c| script_of(*c));
157        // Scripts with a platform cascade (Arabic, Hebrew, Thai, Devanagari): the bundled face,
158        // then installed ones; the others keep the loaded-faces-first order (Japanese documents
159        // prefer the bundled craft-fonts faces).
160        if let Some(s) = script.filter(|s| !os_families(*s).is_empty()) {
161            let bundled = bundled_families(s);
162            for fam in &bundled {
163                let f = self.face(fam, style);
164                if f.family.eq_ignore_ascii_case(fam) && covers_cluster(&f, cluster) {
165                    return f;
166                }
167            }
168            for fam in os_families(s) {
169                if bundled.contains(fam) {
170                    continue;
171                }
172                if self.is_loaded(fam) || (self.system_fallback_enabled() && self.has_family(fam)) {
173                    // With system fallback off, only a face that is already loaded: `face` could load more catalog files.
174                    let f = if self.system_fallback_enabled() {
175                        self.face(fam, style)
176                    } else {
177                        let Some(f) = self.find(fam, style) else { continue };
178                        f
179                    };
180                    if f.family.eq_ignore_ascii_case(fam) && covers_cluster(&f, cluster) {
181                        return f;
182                    }
183                }
184            }
185            #[cfg(not(target_arch = "wasm32"))]
186            if let Some(f) = self.system_face_for(key, cluster, &bundled) {
187                return f;
188            }
189        }
190        let exclude = chain.first().map(|f| f.id()).unwrap_or(u32::MAX);
191        let f = match self.fallback_for(key, exclude) {
192            Some(f) if covers_cluster(&f, cluster) => f,
193            Some(f) if chain.iter().all(|c| !c.covers(key)) => f,
194            // A letter with marks the run's font lacks (stacked combining marks): a loaded plain
195            // face that draws the whole cluster, as PowerPoint draws the marks, before the run's
196            // font with missing-glyph boxes.
197            _ => {
198                // Searched before `first` runs: it may load the default family, which needs the
199                // faces' write lock, so the read lock must not be held across it.
200                let found = self.read_faces().iter().find(|f| !f.is_color() && covers_cluster(f, cluster)).cloned();
201                found.unwrap_or_else(first)
202            }
203        };
204        // A pictograph that defaults to text presentation (⚠ ♥ ⬆ ▶) whose first fallback is the
205        // colour emoji font: PowerPoint draws it from a plain symbol font.
206        if f.is_color()
207            && !is_emoji_cluster(cluster)
208            && let Some(plain) = self.plain_face_for(key, cluster)
209        {
210            return plain;
211        }
212        f
213    }

215    /// An installed (non-bundled) face covering the cluster, loading one from the system catalog
216    /// if no loaded face does.
217    #[cfg(not(target_arch = "wasm32"))]
218    fn system_face_for(&self, key: char, cluster: &[char], bundled: &[&str]) -> Option<Arc<FontFace>> {
219        let pick = |db: &FontDb| {
220            let faces = db.read_faces();
221            let mut v: Vec<&Arc<FontFace>> = faces
222                .iter()
223                .filter(|f| !bundled.iter().any(|b| f.family.eq_ignore_ascii_case(b)) && !CRAFT_FONTS.iter().any(|c| c.family == f.family))
224                .filter(|f| covers_cluster(f, cluster))
225                .collect();
226            v.sort_by_key(|f| (f.italic, (f.weight - 400.0).abs() as i32));
227            v.first().map(|f| (*f).clone())
228        };
229        if let Some(f) = pick(self) {
230            return Some(f);
231        }
232        if self.system_fallback(key) { pick(self) } else { None }
233    }
234}

236/// Most caret records read for one ligature glyph.
237const MAX_LIGATURE_CARETS: usize = 64;
238
239/// Caret positions inside a ligature glyph from GDEF `LigCaretList`, in font units from the
240/// glyph's origin (sorted ascending). Empty when the font has none for `gid`.
241pub fn ligature_carets(face: &FontFace, gid: u32) -> Vec<i32> {
242    let Some(font) = face.skrifa() else { return Vec::new() };
243    let Ok(gdef) = font.gdef() else { return Vec::new() };
244    let Some(Ok(list)) = gdef.lig_caret_list() else { return Vec::new() };
245    let Ok(cov) = list.coverage() else { return Vec::new() };
246    let Some(idx) = cov.get(skrifa::GlyphId::new(gid)) else { return Vec::new() };
247    let Ok(lig) = list.lig_glyphs().get(idx as usize) else { return Vec::new() };
248    // The limit applies to the records read, not to the carets kept: a font can list thousands of
249    // records it then has to be read through for every ligature laid out.
250    let mut out: Vec<i32> = lig
251        .caret_values()
252        .iter()
253        .take(MAX_LIGATURE_CARETS)
254        .filter_map(|cv| cv.ok())
255        .filter_map(|cv| match cv {
256            skrifa::raw::tables::gdef::CaretValue::Format1(f) => Some(f.coordinate() as i32),
257            skrifa::raw::tables::gdef::CaretValue::Format3(f) => Some(f.coordinate() as i32),
258            // Format 2 is a contour point index; outlines aren't consulted here.
259            skrifa::raw::tables::gdef::CaretValue::Format2(_) => None,
260        })
261        .collect();
262    out.sort_unstable();
263    out
264}

270    #[test]
271    fn requested_face_wins_when_it_covers() {
272        let db = FontDb::global();
273        let inter = db.face(crate::fonts::DEFAULT_FAMILY, "Regular");
274        let f = db.cascade(&['a'], std::slice::from_ref(&inter), "Regular");
275        assert!(Arc::ptr_eq(&f, &inter));
276        // Spaces and joiners never force a fallback.
277        let f = db.cascade(&[' ', '\u{200D}'], std::slice::from_ref(&inter), "Regular");
278        assert!(Arc::ptr_eq(&f, &inter));
279        // Nothing to go on: the default face.
280        assert!(!db.cascade(&[], &[], "Regular").family.is_empty());
281    }

283    #[test]
284    fn arabic_cluster_falls_back_to_an_arabic_face() {
285        let db = FontDb::global();
286        let inter = db.face(crate::fonts::DEFAULT_FAMILY, "Regular");
287        let cluster = ['\u{0628}', '\u{064E}']; // beh + fatha: one cluster, one face.
288        let f = db.cascade(&cluster, std::slice::from_ref(&inter), "Regular");
289        if covers_cluster(&f, &cluster) {
290            assert!(f.covers('\u{0628}'));
291        } else {
292            eprintln!("skipped: no installed or bundled face covers Arabic");
293        }
294    }

297    #[test]
298    fn falling_back_to_the_default_face_does_not_deadlock() {
299        // An empty chain and a character no loaded face draws: the loaded faces are searched,
300        // then the default family is looked up, which loads it (taking the faces' write lock).
301        // Holding the search's read lock across that lookup hung the thread.
302        let (tx, rx) = std::sync::mpsc::channel();
303        std::thread::spawn(move || {
304            let db = FontDb::with_font_dirs(Vec::new());
305            db.set_system_fallback(false);
306            let _ = tx.send(db.cascade(&['\u{10FFFD}'], &[], "Regular").family.clone());
307        });
308        let family = rx.recv_timeout(std::time::Duration::from_secs(30)).expect("cascade did not return");
309        assert!(!family.is_empty());
310    }
