44/// The TrueType or OpenType font inside a PresentationML embedded-font part
45/// (`application/x-fontdata`).
46///
47/// The part is the font itself, or an Embedded OpenType wrapper around it (the `0x504C` magic
48/// at byte 34, from the W3C Embedded OpenType submission). A MicroType Express subset is neither,
49/// so this returns none; the original part stays in the file for the round trip.
50pub fn embedded_sfnt(part: &[u8]) -> Option<&[u8]> {
51    eot_payload(part).or_else(|| sfnt_or_collection(part).then_some(part))
52}
53
54fn sfnt_or_collection(b: &[u8]) -> bool {
55    b.len() >= 12
56        && (b.starts_with(&[0x00, 0x01, 0x00, 0x00])
57            || b.starts_with(b"OTTO")
58            || b.starts_with(b"true")
59            || b.starts_with(b"typ1")
60            || b.starts_with(b"ttcf"))
61}
62
63/// Font data at the end of an Embedded OpenType blob, when the header is one.
64fn eot_payload(b: &[u8]) -> Option<&[u8]> {
65    if b.len() < 82 || u16::from_le_bytes([b[34], b[35]]) != 0x504C {
66        return None;
67    }
68    let version = u32::from_le_bytes(b.get(8..12)?.try_into().ok()?);
69    if !matches!(version, 0x0001_0000 | 0x0002_0001 | 0x0002_0002) {
70        return None;
71    }
72    let eot_size = u32::from_le_bytes(b.get(0..4)?.try_into().ok()?) as usize;
73    let font_size = u32::from_le_bytes(b.get(4..8)?.try_into().ok()?) as usize;
74    if font_size < 12 || eot_size < font_size.saturating_add(82) || eot_size > b.len() {
75        return None;
76    }
77    let font = b.get(eot_size - font_size..eot_size)?;
78    sfnt_or_collection(font).then_some(font)
79}

81/// Japanese faces in document-fallback order: Mincho (serif) families first, matching the
82/// serif default text font, then the others; Regular before other styles.
83pub fn japanese_document_fonts() -> Vec<&'static CraftFont> {
84    let mut v: Vec<_> = japanese_fonts().collect();
85    v.sort_by_key(|f| (!f.family.contains("Mincho"), f.style != "Regular"));
86    v
87}

89/// Japanese faces in UI order: BIZ UDPGothic first (the UI face), then the others; `bold` puts
90/// bold styles before regular ones.
91pub fn japanese_ui_fonts(bold: bool) -> Vec<&'static CraftFont> {
92    let mut v: Vec<_> = japanese_fonts().collect();
93    v.sort_by_key(|f| (f.family != "BIZ UDPGothic", (f.style == "Bold") != bold));
94    v
95}

100/// An OpenType feature setting: `"liga"`, `"-kern"`, `"ss01"`.
101pub fn feature(tag: &str) -> Option<Feature> {
102    let (on, t) = match tag.strip_prefix('-') {
103        Some(r) => (false, r),
104        None => (true, tag.strip_prefix('+').unwrap_or(tag)),
105    };
106    let b = t.as_bytes();
107    if b.len() != 4 {
108        return None;
109    }
110    Some(Feature::new(Tag::new(&[b[0], b[1], b[2], b[3]]), on as u32, ..))
111}

113/// A strong right-to-left character (Hebrew, Arabic, …)?
114pub fn is_rtl(c: char) -> bool {
115    use unicode_bidi::BidiClass::{AL, R};
116    matches!(unicode_bidi::bidi_class(c), R | AL)
117}

119/// Shape `text` with `face`. `chars` lets callers substitute characters (e.g. uppercase for All
120/// Caps) while keeping clusters pointing into the original string. Glyphs come in logical
121/// order: each direction run is shaped with [`shape_range`] (right-to-left runs right to left,
122/// with the whole string as context), then its clusters put back in text order (each cluster's
123/// glyphs keep the shaper's order). Paragraph layout uses [`shape_range`] directly and keeps the
124/// visual order.
125pub fn shape(face: &FontFace, text: &str, features: &[Feature], map: impl Fn(char) -> char) -> Vec<ShapedGlyph> {
126    let mut out = Vec::with_capacity(text.len());
127    for (r, rtl) in direction_runs(text) {
128        let mut g = shape_range(face, text, r, &ShapeParams { features, rtl, ..Default::default() }, &map);
129        if rtl {
130            // Visual (clusters descending) → logical; the sort is stable, so each cluster's
131            // glyphs keep the shaper's order.
132            g.sort_by_key(|x| x.cluster);
133        }
134        out.extend(g);
135    }
136    out
137}

139/// Maximal runs of one direction (neutrals join the run they're in): (byte range, right-to-left).
140fn direction_runs(text: &str) -> Vec<(std::ops::Range<usize>, bool)> {
141    let mut runs: Vec<(std::ops::Range<usize>, bool)> = Vec::new();
142    let mut cur: Option<bool> = None;
143    let mut start = 0;
144    for (i, c) in text.char_indices() {
145        let strong = if is_rtl(c) {
146            Some(true)
147        } else if unicode_bidi::bidi_class(c) == unicode_bidi::BidiClass::L {
148            Some(false)
149        } else {
150            None
151        };
152        match (cur, strong) {
153            (None, Some(d)) => cur = Some(d),
154            (Some(a), Some(d)) if a != d => {
155                runs.push((start..i, a));
156                start = i;
157                cur = Some(d);
158            }
159            _ => {}
160        }
161    }
162    runs.push((start..text.len(), cur.unwrap_or(false)));
163    runs
164}

166/// Glyph id of the first of `chars` the face has (0 = .notdef).
167pub fn first_glyph(face: &FontFace, chars: &[char]) -> u32 {
168    chars.iter().map(|c| face.glyph_for(*c)).find(|g| *g != 0).unwrap_or(0)
169}
