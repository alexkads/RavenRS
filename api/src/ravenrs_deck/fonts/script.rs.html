1//! Script and font-slot classification.
2//!
3//! DrawingML gives every run three typefaces: `a:latin`, `a:ea` (East Asian) and `a:cs` (complex
4//! script: Arabic, Hebrew, Thai, Indic…). Which one a character uses follows its Unicode script
5//! (UAX #24). Characters with no script of their own (Common and Inherited: spaces, digits,
6//! punctuation, combining marks) take the script and slot of the text around them, so a space or
7//! a parenthesis between two Arabic words is shaped with them.
8
9/// An ISO 15924 script tag, e.g. `*b"Arab"`.
10pub type ScriptTag = [u8; 4];
11
12pub const ARABIC: ScriptTag = *b"Arab";
13pub const HEBREW: ScriptTag = *b"Hebr";
14pub const LATIN: ScriptTag = *b"Latn";
15
16/// The DrawingML typeface slot a character is drawn with.
17#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
18pub enum FontSlot {
19    /// `a:latin`.
20    #[default]
21    Latin,
22    /// `a:ea`.
23    EastAsian,
24    /// `a:cs`.
25    ComplexScript,
26}
27

93/// Code points inside the Arabic blocks whose script is Common or Inherited: the Arabic comma,
94/// semicolon and question mark, tatweel, the harakat (combining vowel marks) and the BOM-like
95/// U+FEFF. They take the script of the text around them.
96fn arabic_common(cp: u32) -> bool {
97    matches!(cp, 0x060C | 0x061B | 0x061F | 0x0640 | 0x064B..=0x0655 | 0x0670 | 0xFEFF | 0xFD3E | 0xFD3F)
98}
99
100/// The script of `c`, or `None` for Common and Inherited characters (spaces, digits,
101/// punctuation, combining marks), which take the script of their neighbours.
102pub fn script_of(c: char) -> Option<ScriptTag> {
103    let cp = c as u32;
104    if arabic_common(cp) {
105        return None;
106    }
107    // Ranges are sorted and disjoint: binary search on the start.
108    let i = RANGES.partition_point(|(lo, _, _)| *lo <= cp);
109    let (lo, hi, tag) = RANGES.get(i.checked_sub(1)?)?;
110    (cp >= *lo && cp <= *hi).then_some(*tag)
111}

113/// The slot a script is drawn with.
114pub fn slot_of_script(s: ScriptTag) -> FontSlot {
115    match &s {
116        b"Hani" | b"Hira" | b"Kana" | b"Hang" | b"Bopo" => FontSlot::EastAsian,
117        b"Arab" | b"Hebr" | b"Syrc" | b"Thaa" | b"Nkoo" | b"Deva" | b"Beng" | b"Guru" | b"Gujr" | b"Orya" | b"Taml" | b"Telu" | b"Knda" | b"Mlym"
118        | b"Sinh" | b"Thai" | b"Laoo" | b"Tibt" | b"Mymr" | b"Khmr" => FontSlot::ComplexScript,
119        _ => FontSlot::Latin,
120    }
121}
122
123/// The slot of `c` on its own: `None` for characters that take their neighbours' slot. CJK
124/// punctuation and full-width forms are East Asian although their script is Common.
125pub fn slot_of(c: char) -> Option<FontSlot> {
126    if let Some(s) = script_of(c) {
127        return Some(slot_of_script(s));
128    }
129    let cp = c as u32;
130    if (0x3000..=0x303F).contains(&cp) || (0xFF00..=0xFFEF).contains(&cp) || (0x3200..=0x33FF).contains(&cp) {
131        return Some(FontSlot::EastAsian);
132    }
133    None
134}

136/// A default BCP 47 language for a script (shaping picks `locl` forms by language).
137pub fn default_language(s: ScriptTag) -> Option<&'static str> {
138    Some(match &s {
139        b"Arab" => "ar",
140        b"Hebr" => "he",
141        b"Syrc" => "syr",
142        b"Thaa" => "dv",
143        b"Thai" => "th",
144        b"Deva" => "hi",
145        _ => return None,
146    })
147}

149/// Resolve every character's script and slot, giving Common/Inherited characters the value of the
150/// preceding character with one (or, at the start, the following one; Latin when there is none).
151pub fn resolve(chars: &[char]) -> Vec<(Option<ScriptTag>, FontSlot)> {
152    let own: Vec<(Option<ScriptTag>, Option<FontSlot>)> = chars.iter().map(|c| (script_of(*c), slot_of(*c))).collect();
153    let first = own.iter().find_map(|(s, sl)| sl.map(|sl| (*s, sl))).unwrap_or((None, FontSlot::Latin));
154    let mut prev = first;
155    own.into_iter()
156        .map(|(s, sl)| {
157            if let Some(sl) = sl {
158                prev = (s, sl);
159            }
160            prev
161        })
162        .collect()
163}

165/// Arabic letters that join to the following letter (joining type D, dual-joining): a kashida
166/// (tatweel) may be inserted after them. Right-joining (alef, dal, reh, waw…) and non-joining
167/// (hamza) letters are excluded.
168pub fn joins_following(c: char) -> bool {
169    let cp = c as u32;
170    let letter = matches!(cp, 0x0620..=0x064A | 0x066E..=0x06D3 | 0x06FA..=0x06FC | 0x06FF | 0x0750..=0x077F | 0x08A0..=0x08C9);
171    letter
172        && !matches!(
173            cp,
174            0x0621..=0x0625 | 0x0627 | 0x0629 | 0x062F..=0x0632 | 0x0648 | 0x0671..=0x0673 | 0x0675..=0x0677 | 0x0688..=0x0699 | 0x06C0
175                | 0x06C3..=0x06CB | 0x06CD | 0x06CF | 0x06D2 | 0x06D3 | 0x0759..=0x075B | 0x076B | 0x076C | 0x0771 | 0x0773 | 0x0774
176                | 0x0778 | 0x0779 | 0x08AA..=0x08AC | 0x08AE | 0x08B1 | 0x08B2 | 0x08B9
177        )
178}

180/// An Arabic letter that joins to the preceding letter (joining type D or R).
181pub fn joins_preceding(c: char) -> bool {
182    let cp = c as u32;
183    matches!(cp, 0x0622..=0x064A | 0x066E..=0x06D3 | 0x06FA..=0x06FC | 0x06FF | 0x0750..=0x077F | 0x08A0..=0x08C9) && cp != 0x0621
184}

186#[cfg(test)]
187mod tests {
188    use super::*;
189
190    #[test]
191    fn scripts_and_slots() {
192        assert_eq!(script_of('س'), Some(ARABIC));
193        assert_eq!(script_of('ש'), Some(HEBREW));
194        assert_eq!(script_of('R'), Some(LATIN));
195        assert_eq!(script_of('2'), None);
196        assert_eq!(script_of('('), None);
197        // Harakat and tatweel are Inherited/Common.
198        assert_eq!(script_of('\u{064E}'), None);
199        assert_eq!(script_of('\u{0640}'), None);
200        assert_eq!(slot_of('س'), Some(FontSlot::ComplexScript));
201        assert_eq!(slot_of('日'), Some(FontSlot::EastAsian));
202        assert_eq!(slot_of('、'), Some(FontSlot::EastAsian));
203        assert_eq!(slot_of('a'), Some(FontSlot::Latin));
204        assert_eq!(slot_of(' '), None);
205        assert_eq!(script_of('\u{10FFFF}'), None);
206        assert_eq!(script_of('\0'), None);
207    }
208
209    #[test]
210    fn neutrals_take_the_surrounding_slot() {
211        let chars: Vec<char> = "سنة 2024 (Retrait) فقط".chars().collect();
212        let r = resolve(&chars);
213        // "سنة 2024 (" follows Arabic: complex script.
214        for (i, c) in chars.iter().enumerate().take(10) {
215            assert_eq!(r[i].1, FontSlot::ComplexScript, "{c:?}");
216        }
217        // "Retrait)" is Latin; the space after ")" stays Latin; "فقط" is complex script.
218        assert_eq!(r[10].1, FontSlot::Latin);
219        assert_eq!(r[17].1, FontSlot::Latin);
220        assert_eq!(r[19], (Some(ARABIC), FontSlot::ComplexScript));
221        // Leading neutrals take the first strong character's slot.
222        let r = resolve(&['(', 'س']);
223        assert_eq!(r[0].1, FontSlot::ComplexScript);
224        assert!(resolve(&[]).is_empty());
225        assert_eq!(resolve(&['1'])[0], (None, FontSlot::Latin));
226    }
227
228    #[test]
229    fn joining_classes() {
230        assert!(joins_following('س'));
231        assert!(joins_following('ن'));
232        assert!(!joins_following('ا'));
233        assert!(!joins_following('د'));
234        assert!(!joins_following('ء'));
235        assert!(joins_preceding('ا'));
236        assert!(!joins_preceding('ء'));
237        assert!(!joins_following('a'));
238    }

240    #[test]
241    fn ranges_are_sorted_and_disjoint() {
242        for w in RANGES.windows(2) {
243            assert!(w[0].1 < w[1].0, "{:x?} {:x?}", w[0], w[1]);
244        }
245    }
