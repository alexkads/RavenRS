1//! OpenType shaping with HarfRust.
2//!
3//! [`shape_range`] shapes one item of a paragraph (one face, one direction, one script) with the
4//! rest of the paragraph as pre/post context, so Arabic letters at an item boundary (a colour
5//! change inside a word, a bold letter) still take their joining forms. Glyphs come back in
6//! visual order, exactly as the shaper produced them: right-to-left items are not reversed.
7
8use std::ops::Range;
9
10use harfrust::{Direction, Feature, Language, Script, ShapeOptions, Tag, UnicodeBuffer};
11use skrifa::MetadataProvider;
12use skrifa::instance::Size;
13
14use crate::fonts::FontFace;
15use crate::fonts::script::ScriptTag;
16
17/// One shaped glyph, in font units.
18#[derive(Clone, Copy, Debug, PartialEq)]
19pub struct ShapedGlyph {
20    pub gid: u32,
21    /// Byte offset (in the shaped string) of the cluster this glyph belongs to.
22    pub cluster: usize,
23    pub x_advance: i32,
24    pub x_offset: i32,
25    pub y_offset: i32,
26}
27
28/// How to shape an item.
29#[derive(Clone, Copy, Debug, Default)]
30pub struct ShapeParams<'a> {
31    pub features: &'a [Feature],
32    /// Right-to-left (odd bidi level). Mirrored characters (brackets) take their mirrored glyphs.
33    pub rtl: bool,
34    /// ISO 15924 script (`*b"Arab"`); `None` lets the shaper guess from the text.
35    pub script: Option<ScriptTag>,
36    /// BCP 47 language (`"ar"`, `"ar-MA"`); `None` = the script's default.
37    pub language: Option<&'a str>,
38}
39
40/// Most bytes of text given to the shaper at once. The shaper lets one call grow its buffers to
41/// 256 glyphs per character (a font's substitutions decide how many), so a long item is shaped in
42/// pieces of this size, each with the rest of the paragraph as context.
43const MAX_SHAPE_CHUNK: usize = 1024;
44/// Most glyphs kept per character of a piece: a piece the font expands further is laid out one
45/// glyph per character from the cmap instead.
46const MAX_GLYPHS_PER_CHAR: usize = 32;
47
48/// Shape `para[range]` with `face`, the rest of `para` as context. Glyphs are in visual order
49/// (left to right) and clusters are byte offsets into `para`. `map` substitutes characters (All
50/// Caps) without moving clusters. An empty or out-of-bounds range shapes nothing.
51pub fn shape_range(face: &FontFace, para: &str, range: Range<usize>, params: &ShapeParams, map: impl Fn(char) -> char) -> Vec<ShapedGlyph> {
52    let (Some(_), Some(text), Some(_)) = (para.get(..range.start), para.get(range.clone()), para.get(range.end..)) else {
53        return Vec::new();
54    };
55    let chunks = chunks(text, MAX_SHAPE_CHUNK);
56    let mut pieces: Vec<Vec<ShapedGlyph>> = Vec::with_capacity(chunks.len());
57    let shaped = face.hb().map(|hb| {
58        let shaper = face.shaper.shaper(&hb).instance(face.instance.as_ref()).build();
59        // Pieces of one item share the script the whole item would be guessed to have.
60        let item_script = (chunks.len() > 1 && params.script.is_none()).then(|| {
61            let mut probe = UnicodeBuffer::new();
62            for c in text.chars() {
63                probe.add(map(c), 0);
64            }
65            probe.guess_segment_properties();
66            probe.script()
67        });
68        for c in &chunks {
69            let r = (range.start + c.start)..(range.start + c.end);
70            let (Some(pre), Some(piece), Some(post)) = (para.get(..r.start), para.get(r.clone()), para.get(r.end..)) else { continue };
71            let mut buf = UnicodeBuffer::new();
72            buf.set_pre_context(pre);
73            buf.set_post_context(post);
74            for (i, c) in piece.char_indices() {
75                buf.add(map(c), (r.start + i) as u32);
76            }
77            buf.guess_segment_properties();
78            if let Some(s) = params.script.and_then(|t| Script::from_iso15924_tag(Tag::new(&t))) {
79                buf.set_script(s);
80            } else if let Some(s) = item_script {
81                buf.set_script(s);
82            }
83            if let Some(l) =
84                params.language.or_else(|| params.script.and_then(crate::fonts::script::default_language)).and_then(|l| l.parse::<Language>().ok())
85            {
86                buf.set_language(l);
87            }
88            buf.set_direction(if params.rtl { Direction::RightToLeft } else { Direction::LeftToRight });
89            let gb = shaper.shape(buf, ShapeOptions::new().features(params.features));
90            let budget = piece.chars().count().saturating_mul(MAX_GLYPHS_PER_CHAR);
91            if gb.len() > budget {
92                // Font-controlled expansion past anything real text needs: keep it bounded.
93                pieces.push(cmap_glyphs(face, para, r, params.rtl, &map));
94                continue;
95            }
96            pieces.push(
97                gb.glyph_infos()
98                    .iter()
99                    .zip(gb.glyph_positions())
100                    .map(|(info, pos)| ShapedGlyph {
101                        gid: info.glyph_id,
102                        cluster: info.cluster as usize,
103                        x_advance: pos.x_advance,
104                        x_offset: pos.x_offset,
105                        y_offset: pos.y_offset,
106                    })
107                    .collect(),
108            );
109        }
110    });
111    if shaped.is_none() {
112        // No shaper for this face: one glyph per character from the cmap, in visual order.
113        return cmap_glyphs(face, para, range, params.rtl, &map);
114    }
115    // Each piece is in visual order; right to left, the later pieces come first.
116    if params.rtl {
117        pieces.reverse();
118    }
119    let mut out = Vec::with_capacity(pieces.iter().map(Vec::len).sum());
120    for p in pieces {
121        out.extend(p);
122    }
123    out
124}

126/// One glyph per character of `para[range]` from the cmap, in visual order.
127fn cmap_glyphs(face: &FontFace, para: &str, range: Range<usize>, rtl: bool, map: &impl Fn(char) -> char) -> Vec<ShapedGlyph> {
128    let (Some(text), Some(f)) = (para.get(range.clone()), face.skrifa()) else { return Vec::new() };
129    let cmap = f.charmap();
130    let gm = f.glyph_metrics(Size::unscaled(), face.location());
131    let mut out = Vec::with_capacity(text.len());
132    for (i, c) in text.char_indices() {
133        let g = cmap.map(map(c)).unwrap_or_default();
134        let adv = gm.advance_width(g).unwrap_or(face.upem as f32 * 0.5);
135        out.push(ShapedGlyph { gid: g.to_u32(), cluster: range.start + i, x_advance: adv.round() as i32, x_offset: 0, y_offset: 0 });
136    }
137    if rtl {
138        out.reverse();
139    }
140    out
141}

143/// `text` cut into byte ranges of at most `max` bytes (or one character, if longer), at the last
144/// space of a piece where there is one, else before a character that starts a cluster (not a
145/// combining mark or joiner, not after a joiner). Text that fits is one piece.
146fn chunks(text: &str, max: usize) -> Vec<Range<usize>> {
147    use unicode_bidi::BidiClass::{BN, NSM};
148    let mut out = Vec::new();
149    let mut start = 0;
150    while text.len() - start > max {
151        let window = text.get(start..).unwrap_or_default();
152        let mut space = None;
153        let mut boundary = None;
154        let mut any = None;
155        let mut prev: Option<char> = None;
156        for (i, c) in window.char_indices() {
157            if i > max {
158                break;
159            }
160            if i > 0 {
161                any = Some(i);
162                let joined = prev.is_some_and(|p| matches!(unicode_bidi::bidi_class(p), BN));
163                if !joined && !matches!(unicode_bidi::bidi_class(c), NSM | BN) {
164                    boundary = Some(i);
165                    if prev.is_some_and(char::is_whitespace) {
166                        space = Some(i);
167                    }
168                }
169            }
170            prev = Some(c);
171        }
172        // A piece always moves forward by at least one character.
173        let first = window.chars().next().map_or(window.len(), char::len_utf8);
174        let cut = space.filter(|&i| i >= max / 2).or(boundary).or(any).unwrap_or(first).max(first);
175        out.push(start..start + cut);
176        start += cut;
177    }
178    if start < text.len() || out.is_empty() {
179        out.push(start..text.len());
180    }
181    out
182}

184#[cfg(test)]
185mod tests {
186    use super::*;
187    use crate::fonts::{DEFAULT_FAMILY, FontDb};

189    #[test]
190    fn rtl_items_come_in_visual_order_and_mirror() {
191        let face = FontDb::global().face(DEFAULT_FAMILY, "Regular");
192        let text = "(ab)";
193        let p = ShapeParams { rtl: true, ..Default::default() };
194        let g = shape_range(&face, text, 0..text.len(), &p, |c| c);
195        // Visual order: last cluster first.
196        assert_eq!(g.iter().map(|g| g.cluster).collect::<Vec<_>>(), vec![3, 2, 1, 0]);
197        // L4: in a right-to-left run "(" is drawn with the ")" glyph and vice versa.
198        if face.covers('(') && face.covers(')') {
199            assert_eq!(g[3].gid, face.glyph_for(')'));
200            assert_eq!(g[0].gid, face.glyph_for('('));
201        }
202    }
203
204    #[test]
205    fn clusters_point_into_the_whole_paragraph() {
206        let face = FontDb::global().face(DEFAULT_FAMILY, "Regular");
207        let para = "Hello world";
208        let g = shape_range(&face, para, 6..11, &ShapeParams::default(), |c| c);
209        assert_eq!(g.first().map(|g| g.cluster), Some(6));
210        assert!(shape_range(&face, para, 6..99, &ShapeParams::default(), |c| c).is_empty());
211        // Not a char boundary.
212        assert!(shape_range(&face, "سنة", 1..2, &ShapeParams::default(), |c| c).is_empty());
213    }
214
215    #[test]
216    fn long_items_are_shaped_in_bounded_pieces() {
217        for text in ["word ".repeat(2_000), "e\u{301}".repeat(2_000), "x".repeat(5_000), "\u{1f469}\u{200d}\u{1f4bb} ".repeat(500)] {
218            let pieces = chunks(&text, MAX_SHAPE_CHUNK);
219            assert_eq!(pieces.first().map(|r| r.start), Some(0));
220            assert_eq!(pieces.last().map(|r| r.end), Some(text.len()));
221            for w in pieces.windows(2) {
222                assert_eq!(w[0].end, w[1].start);
223            }
224            for r in &pieces {
225                assert!(!r.is_empty() && r.len() <= MAX_SHAPE_CHUNK);
226                // Never before a combining mark, never after a joiner.
227                let rest = &text[r.end..];
228                assert!(!rest.starts_with('\u{301}'));
229                assert!(!text[..r.end].ends_with('\u{200d}') || rest.is_empty());
230            }
231        }
232        assert_eq!(chunks("short", MAX_SHAPE_CHUNK), vec![0..5]);
233        assert_eq!(chunks("", MAX_SHAPE_CHUNK), vec![0..0]);
234    }
235
236    #[test]
237    fn long_items_keep_every_cluster_in_visual_order() {
238        let face = FontDb::global().face(DEFAULT_FAMILY, "Regular");
239        let text = "ab cd ".repeat(1_000);
240        let g = shape_range(&face, &text, 0..text.len(), &ShapeParams::default(), |c| c);
241        assert_eq!(g.len(), text.len());
242        assert!(g.windows(2).all(|w| w[0].cluster < w[1].cluster));
243        let rtl = shape_range(&face, &text, 0..text.len(), &ShapeParams { rtl: true, ..Default::default() }, |c| c);
244        assert_eq!(rtl.len(), text.len());
245        assert!(rtl.windows(2).all(|w| w[0].cluster > w[1].cluster));
246    }
