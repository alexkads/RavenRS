1//! Draw › Ink to Shape: recognise a hand-drawn stroke as a line, rectangle, ellipse or triangle.
2//!
3//! Pure geometry so it is unit-tested with synthetic strokes. A stroke is closed when its ends
4//! meet; a closed stroke is classified by how much of its bounding box it covers (rectangle ≈ 1,
5//! ellipse ≈ π/4, triangle ≈ ½) and, for ellipses, by how evenly it keeps its distance from the
6//! centre.
7
8/// What a stroke was recognised as, in its bounding box `(x, y, w, h)`.
9#[derive(Clone, Copy, Debug, PartialEq)]
10pub enum Recognized {
11    /// From `(x0, y0)` to `(x1, y1)`.
12    Line {
13        x0: f64,
14        y0: f64,
15        x1: f64,
16        y1: f64,
17    },
18    Rect {
19        x: f64,
20        y: f64,
21        w: f64,
22        h: f64,
23    },
24    Ellipse {
25        x: f64,
26        y: f64,
27        w: f64,
28        h: f64,
29    },
30    /// Apex at the top (`flip_v` false) or at the bottom.
31    Triangle {
32        x: f64,
33        y: f64,
34        w: f64,
35        h: f64,
36        flip_v: bool,
37    },
38}
39
40/// Strokes smaller than this (points) stay ink.
41const MIN_SIZE: f64 = 6.0;
42const MAX_POINTS: usize = 100_000;
43
44pub fn recognize(points: &[(f64, f64)]) -> Option<Recognized> {
45    let pts: Vec<(f64, f64)> = points.iter().copied().filter(|(x, y)| x.is_finite() && y.is_finite()).take(MAX_POINTS).collect();
46    if pts.len() < 2 {
47        return None;
48    }
49    let (mut x0, mut y0, mut x1, mut y1) = (f64::MAX, f64::MAX, f64::MIN, f64::MIN);
50    for (x, y) in &pts {
51        x0 = x0.min(*x);
52        y0 = y0.min(*y);
53        x1 = x1.max(*x);
54        y1 = y1.max(*y);
55    }
56    let (w, h) = (x1 - x0, y1 - y0);
57    let size = w.max(h);
58    if !size.is_finite() || size < MIN_SIZE {
59        return None;
60    }
61    let length: f64 = pts.windows(2).map(|p| (p[1].0 - p[0].0).hypot(p[1].1 - p[0].1)).sum();
62    let (first, last) = (pts[0], pts[pts.len() - 1]);
63    let gap = (last.0 - first.0).hypot(last.1 - first.1);
64    if length <= 0.0 {
65        return None;
66    }
67    // Open stroke: a line if nearly straight, otherwise not a shape.
68    if gap > 0.25 * size {
69        return (gap / length > 0.92).then_some(Recognized::Line { x0: first.0, y0: first.1, x1: last.0, y1: last.1 });
70    }
71    // Closed but flat: a line drawn back and forth isn't a shape.
72    if w.min(h) < 0.15 * size {
73        return None;
74    }
75    // Covered area (shoelace, the gap closed) over the bounding box.
76    let mut a2 = 0.0;
77    for k in 0..pts.len() {
78        let (p, q) = (pts[k], pts[(k + 1) % pts.len()]);
79        a2 += p.0 * q.1 - q.0 * p.1;
80    }
81    let cover = (a2.abs() / 2.0) / (w * h);
82    // Ellipse fit: distance from the centre in box-normalised units should stay near 1.
83    let (cx, cy) = (x0 + w / 2.0, y0 + h / 2.0);
84    let radial: Vec<f64> = pts.iter().map(|(x, y)| ((x - cx) / (w / 2.0)).hypot((y - cy) / (h / 2.0))).collect();
85    let spread = radial.iter().map(|r| (r - 1.0).abs()).sum::<f64>() / radial.len() as f64;
86    if cover > 0.86 {
87        Some(Recognized::Rect { x: x0, y: y0, w, h })
88    } else if cover > 0.66 && spread < 0.12 {
89        Some(Recognized::Ellipse { x: x0, y: y0, w, h })
90    } else if (0.35..0.66).contains(&cover) {
91        // Apex: the side of the box with fewer points near it is where the triangle narrows.
92        let near_top = pts.iter().filter(|(_, y)| *y < y0 + h * 0.15).count();
93        let near_bottom = pts.iter().filter(|(_, y)| *y > y1 - h * 0.15).count();
94        Some(Recognized::Triangle { x: x0, y: y0, w, h, flip_v: near_bottom < near_top })
95    } else {
96        None
97    }
98}

104    fn poly(corners: &[(f64, f64)], per_side: usize) -> Vec<(f64, f64)> {
105        let mut v = vec![];
106        for k in 0..corners.len() {
107            let (a, b) = (corners[k], corners[(k + 1) % corners.len()]);
108            for i in 0..per_side {
109                let t = i as f64 / per_side as f64;
110                // A little hand wobble.
111                let wob = ((i * 7 + k * 3) % 5) as f64 * 0.4 - 0.8;
112                v.push((a.0 + (b.0 - a.0) * t + wob, a.1 + (b.1 - a.1) * t - wob));
113            }
114        }
115        v.push((corners[0].0 + 2.0, corners[0].1 + 1.5));
116        v
117    }

119    #[test]
120    fn rectangles_ellipses_triangles_and_lines() {
121        let r = recognize(&poly(&[(100.0, 100.0), (300.0, 100.0), (300.0, 200.0), (100.0, 200.0)], 30)).unwrap();
122        assert!(matches!(r, Recognized::Rect { w, h, .. } if (w - 200.0).abs() < 4.0 && (h - 100.0).abs() < 4.0), "{r:?}");
123        let ell: Vec<(f64, f64)> = (0..=80)
124            .map(|i| {
125                let a = i as f64 / 80.0 * std::f64::consts::TAU;
126                (200.0 + 120.0 * a.cos() + (i % 3) as f64 * 0.5, 150.0 + 60.0 * a.sin())
127            })
128            .collect();
129        assert!(matches!(recognize(&ell), Some(Recognized::Ellipse { .. })), "{:?}", recognize(&ell));
130        let up = recognize(&poly(&[(200.0, 50.0), (300.0, 250.0), (100.0, 250.0)], 30)).unwrap();
131        assert!(matches!(up, Recognized::Triangle { flip_v: false, .. }), "{up:?}");
132        let down = recognize(&poly(&[(100.0, 50.0), (300.0, 50.0), (200.0, 250.0)], 30)).unwrap();
133        assert!(matches!(down, Recognized::Triangle { flip_v: true, .. }), "{down:?}");
134        let line: Vec<(f64, f64)> = (0..=40).map(|i| (10.0 + i as f64 * 5.0, 20.0 + i as f64 * 2.0 + (i % 2) as f64 * 0.6)).collect();
135        assert!(matches!(recognize(&line), Some(Recognized::Line { x0, x1, .. }) if x0 < x1), "{:?}", recognize(&line));
136    }

138    #[test]
139    fn scribbles_dots_and_garbage_stay_ink() {
140        assert_eq!(recognize(&[]), None);
141        assert_eq!(recognize(&[(1.0, 1.0)]), None);
142        assert_eq!(recognize(&[(0.0, 0.0), (2.0, 2.0), (1.0, 3.0)]), None, "too small");
143        let zig: Vec<(f64, f64)> = (0..40).map(|i| (i as f64 * 5.0, if i % 2 == 0 { 0.0 } else { 60.0 })).collect();
144        assert_eq!(recognize(&zig), None, "an open zigzag is no line");
145        assert_eq!(recognize(&[(f64::NAN, 0.0), (f64::INFINITY, 1.0)]), None);
146        let there_and_back: Vec<(f64, f64)> = (0..=20).chain((0..=20).rev()).map(|i| (i as f64 * 10.0, 5.0)).collect();
147        assert_eq!(recognize(&there_and_back), None);
148    }

150    #[test]
151    fn ink_to_shape_command_converts_and_undoes() {
152        use crate::model::{InkStroke, Shape, ShapeId, ShapeKind};
153        use serde_json::json;
154        let mut s = crate::engine::Session::with_new();
155        s.execute("slide.new", &json!({"layout": "blank"})).unwrap();
156        let rect = poly(&[(100.0, 100.0), (300.0, 100.0), (300.0, 200.0), (100.0, 200.0)], 30);
157        let zig: Vec<(f64, f64)> = (0..40).map(|i| (i as f64 * 5.0, if i % 2 == 0 { 300.0 } else { 360.0 })).collect();
158        let mk = |v: &[(f64, f64)]| InkStroke {
159            points: v.iter().map(|(x, y)| (*x, *y, 0.5)).collect(),
160            color: crate::color::Rgba::rgb(200, 0, 0),
161            width: 3.0,
162            highlighter: false,
163        };
164        let strokes = vec![mk(&rect), mk(&zig)];
165        s.edit(|doc, sel| {
166            let id = ShapeId(doc.alloc_id().expect("id"));
167            crate::engine::shapes_mut(doc, sel).unwrap().push(Shape {
168                id,
169                name: "Ink".into(),
170                xfrm: Some(crate::geom::Xfrm::new(0.0, 0.0, 960.0, 540.0)),
171                kind: ShapeKind::Ink { strokes },
172                ..Default::default()
173            });
174            Ok(serde_json::Value::Null)
175        })
176        .unwrap();
177        let n0 = s.doc().unwrap().shapes().len();
178        let r = s.execute("draw.inkToShape", &json!({})).unwrap();
179        assert_eq!(r["kept"], 1);
180        let sh = s.doc().unwrap().shapes();
181        assert_eq!(sh.len(), n0 + 1, "ink keeps the zigzag, one rectangle added");
182        let made = sh.last().unwrap();
183        assert!(matches!(&made.geom, crate::model::Geom::Preset { name, .. } if name == "rect"));
184        assert_eq!(made.line.as_ref().and_then(|l| l.width), Some(3.0));
185        s.execute("edit.undo", &json!({})).unwrap();
186        assert_eq!(s.doc().unwrap().shapes().len(), n0);
187        // Nothing recognisable: a no-op, not an error. Empty slide too.
188        s.execute("draw.inkToShape", &json!({})).unwrap();
189        s.execute("slide.new", &json!({"layout": "blank"})).unwrap();
190        assert_eq!(s.execute("draw.inkToShape", &json!({})).unwrap()["kept"], 0);
191    }
