1//! A sample deck built entirely in code (our own text, shapes and data), for demos, screenshots
2//! and tests: `deckcraft --sample`.
3
4use serde_json::{Value, json};
5
6use crate::engine::{Result, Session};
7
8fn run(s: &mut Session, id: &str, p: Value) -> Result<Value> {
9    s.execute(id, &p)
10}
11
12fn last_id(v: &Value) -> u64 {
13    v.get("id").and_then(Value::as_u64).unwrap_or(0)
14}
15
16/// Open the sample presentation in a new document.
17pub fn open_sample(s: &mut Session) -> Result<()> {
18    run(s, "file.new", json!({"theme": "DeckCraft"}))?;
19    let st = s.doc()?;
20    let title = st.current_slide().and_then(|x| x.shapes.first()).map(|x| x.id.0).unwrap_or(0);
21    let sub = st.current_slide().and_then(|x| x.shapes.get(1)).map(|x| x.id.0).unwrap_or(0);
22    run(s, "text.set", json!({"id": title, "text": "Northern Lights"}))?;
23    run(s, "text.set", json!({"id": sub, "text": "A short tour of the aurora — where it comes from, when to look, and how to photograph it"}))?;
24    run(s, "slide.notes", json!({"text": "Welcome everyone. This deck was made in DeckCraft."}))?;
25    // Decorative bands on the title slide.
26    let b = run(s, "shape.insert", json!({"preset": "rect", "rect": [0, 470, 960, 70]}))?;
27    run(s, "shape.fill", json!({"id": last_id(&b), "gradient": {"stops": [[0, "accent1"], [0.5, "accent4"], [1, "accent2"]], "angle": 0}}))?;
28    run(s, "shape.line", json!({"id": last_id(&b), "none": true}))?;
29    run(s, "shape.rename", json!({"id": last_id(&b), "name": "Aurora Band"}))?;
30    run(s, "transition.set", json!({"kind": "fade"}))?;
31
32    // Agenda.
33    run(s, "slide.new", json!({"layout": "titleAndContent", "title": "What we'll cover"}))?;
34    let body = s.doc()?.current_slide().and_then(|x| x.shapes.get(1)).map(|x| x.id.0).unwrap_or(0);
35    run(
36        s,
37        "text.set",
38        json!({"id": body, "text": "The science: solar wind meets the magnetosphere\nWhere and when to see it\n\tHigh latitudes, dark skies, long winter nights\n\tKp index and space-weather forecasts\nPhotographing the aurora\nSafety and etiquette in the field"}),
39    )?;
40    run(s, "transition.set", json!({"kind": "push", "option": "u"}))?;
41    run(s, "edit.select", json!({"ids": [body]}))?;
42    run(s, "animation.set", json!({"effect": "fade", "class": "entrance"}))?;
43    run(s, "animation.options", json!({"textBuild": "byParagraph"}))?;
44
45    // Section header.
46    run(s, "slide.new", json!({"layout": "sectionHeader", "title": "The Science", "body": "Charged particles, magnetic fields and glowing gases"}))?;
47    run(s, "design.background", json!({"color": "dk2"}))?;
48    let st = s.doc()?;
49    let ids: Vec<u64> = st.current_slide().map(|x| x.shapes.iter().map(|s| s.id.0 as u64).collect()).unwrap_or_default();
50    run(s, "edit.select", json!({"ids": ids}))?;
51    run(s, "format.color", json!({"color": "#FFFFFF"}))?;
52    run(s, "transition.set", json!({"kind": "wipe", "option": "r"}))?;
53
54    // Process diagram from shapes.
55    run(s, "slide.new", json!({"layout": "titleOnly", "title": "From the Sun to your sky"}))?;
56    let steps = [("Solar wind", "accent2"), ("Magnetosphere", "accent1"), ("Particles collide", "accent4"), ("Light!", "accent3")];
57    let mut prev: Option<u64> = None;
58    for (i, (label, color)) in steps.iter().enumerate() {
59        let x = 60.0 + i as f64 * 220.0;
60        let v = run(s, "shape.insert", json!({"preset": if i == 3 { "star5" } else { "roundRect" }, "rect": [x, 210, 190, 120], "text": label}))?;
61        let id = last_id(&v);
62        run(s, "shape.fill", json!({"id": id, "color": color}))?;
63        run(s, "shape.line", json!({"id": id, "none": true}))?;
64        run(s, "shape.effects", json!({"id": id, "shadow": "bottom"}))?;
65        run(s, "edit.select", json!({"ids": [id]}))?;
66        run(s, "format.size", json!({"size": 17}))?;
67        run(s, "format.bold", json!({"on": true}))?;
68        run(s, "animation.add", json!({"effect": "zoom", "class": "entrance", "start": if i == 0 { "onClick" } else { "afterPrevious" }}))?;
69        if let Some(p) = prev {
70            let _ = p;
71            let a = run(s, "shape.insert", json!({"preset": "rightArrow", "rect": [x - 36.0, 255, 32, 30]}))?;
72            run(s, "shape.fill", json!({"id": last_id(&a), "color": "tx2"}))?;
73            run(s, "shape.line", json!({"id": last_id(&a), "none": true}))?;
74        }
75        prev = Some(id);
76    }
77    let cap =
78        run(s, "insert.textBox", json!({"rect": [60, 380, 840, 60], "text": "Oxygen glows green and red; nitrogen adds blue and purple fringes."}))?;
79    run(s, "edit.select", json!({"ids": [last_id(&cap)]}))?;
80    run(s, "format.alignCenter", json!({}))?;
81    run(s, "format.size", json!({"size": 20}))?;
82    run(s, "format.italic", json!({"on": true}))?;
83    run(s, "transition.set", json!({"kind": "morph"}))?;
84
85    // Chart.
86    run(s, "slide.new", json!({"layout": "titleOnly", "title": "Best months to look (clear nights)"}))?;
87    run(
88        s,
89        "insert.chart",
90        json!({"type": "column", "rect": [80, 140, 800, 360], "title": "Average clear, dark nights per month", "categories": ["Sep", "Oct", "Nov", "Dec", "Jan", "Feb", "Mar"], "series": [{"name": "Tromsø", "values": [6, 8, 7, 6, 7, 9, 11]}, {"name": "Fairbanks", "values": [9, 10, 8, 9, 11, 12, 13]}, {"name": "Yellowknife", "values": [8, 11, 9, 10, 12, 13, 14]}]}),
91    )?;
92    run(s, "transition.set", json!({"kind": "fade"}))?;
93
94    // Table.
95    run(s, "slide.new", json!({"layout": "titleOnly", "title": "Camera settings that work"}))?;
96    run(
97        s,
98        "insert.table",
99        json!({"rows": 5, "cols": 3, "rect": [100, 150, 760, 260], "data": [["Setting", "Faint aurora", "Bright, active aurora"], ["Aperture", "f/1.4 – f/2.8", "f/2.8"], ["Shutter", "8 – 15 s", "1 – 4 s"], ["ISO", "1600 – 3200", "800 – 1600"], ["Focus", "Manual, on a bright star", "Manual, on a bright star"]]}),
100    )?;
101    run(s, "transition.set", json!({"kind": "cover", "option": "l"}))?;
102
103    // Two content: comparison with shapes.
104    run(s, "slide.new", json!({"layout": "twoContent", "title": "Field checklist"}))?;
105    let st = s.doc()?;
106    let (l, r) =
107        st.current_slide().map(|x| (x.shapes.get(1).map(|s| s.id.0).unwrap_or(0), x.shapes.get(2).map(|s| s.id.0).unwrap_or(0))).unwrap_or((0, 0));
108    run(s, "text.set", json!({"id": l, "text": "Bring\n\tTripod and spare batteries\n\tHeadlamp with a red mode\n\tWarm layers and a thermos"}))?;
109    run(
110        s,
111        "text.set",
112        json!({"id": r, "text": "Remember\n\tCheck the forecast and cloud cover\n\tLet your eyes adapt for 20 minutes\n\tLeave no trace"}),
113    )?;
114    run(s, "transition.set", json!({"kind": "split"}))?;
115
116    // Closing.
117    run(s, "slide.new", json!({"layout": "title", "title": "Clear skies!", "body": "Made with DeckCraft — free and open source"}))?;
118    let moon = run(s, "shape.insert", json!({"preset": "moon", "rect": [820, 40, 80, 120]}))?;
119    run(s, "shape.fill", json!({"id": last_id(&moon), "color": "accent6"}))?;
120    run(s, "shape.line", json!({"id": last_id(&moon), "none": true}))?;
121    run(s, "shape.effects", json!({"id": last_id(&moon), "glow": {"color": "accent6", "radius": 10}}))?;
122    run(s, "transition.set", json!({"kind": "fade"}))?;
123
124    run(s, "section.add", json!({"name": "Introduction", "at": 0}))?;
125    run(s, "section.add", json!({"name": "Science", "at": 2}))?;
126    run(s, "section.add", json!({"name": "In the Field", "at": 5}))?;
127    run(s, "slide.go", json!({"index": 0}))?;
128    run(s, "edit.deselect", json!({}))?;
129    run(s, "edit.deselect", json!({}))?;
130    // The sample opens clean: no undo history, not dirty.
131    if let Some(st) = s.active_mut() {
132        st.history.undo.clear();
133        st.history.redo.clear();
134        st.saved_doc = st.doc.clone();
135        st.title = "Northern Lights".into();
136    }
137    Ok(())
138}
