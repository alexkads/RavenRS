307    /// The shape in Edit Points, while it is still the one selected shape.
308    pub fn editing_points(&self, st: &crate::engine::DocState) -> Option<ShapeId> {
309        // Only a freeform has points: after Undo takes the conversion back, Edit Points is over.
310        self.edit_points.filter(|id| {
311            st.selection.shapes.as_slice() == [*id]
312                && st.selection.text.is_none()
313                && st.shape(*id).is_some_and(|sh| matches!(sh.geom, crate::model::Geom::Custom { .. }))
314        })
315    }

318    pub fn drawing_freeform(&self) -> bool {

320    }

323/// Handle positions in slide coordinates for a box (8 resize + rotate).
324pub fn box_handles(x: &Xfrm) -> ([Point; 8], Point) {
325    let a = x.affine();
326    let (w, h) = (x.w, x.h);
327    let local = [(0.0, 0.0), (w / 2.0, 0.0), (w, 0.0), (w, h / 2.0), (w, h), (w / 2.0, h), (0.0, h), (0.0, h / 2.0)];
328    let pts = local.map(|(px, py)| a * Point::new(px, py));
329    // The rotation handle sits above the top edge (in the shape's rotated frame).
330    let up = if x.flip_v { 1.0 } else { -1.0 };
331    let rot =
332        Affine::translate(Vec2::new(x.x + w / 2.0, x.y + h / 2.0)) * Affine::rotate(x.rot.to_radians()) * Point::new(0.0, up * (h / 2.0 + 20.0));
333    (pts, rot)
334}

336/// WordArt adjust handles for `sh`, in shape-local points (the text rectangle). Empty when the
337/// resolved body has no warp DeckCraft draws, or the shape has no text to store one on.
338pub fn wordart_handles(
339    doc: &crate::model::Presentation,
340    sel: &crate::engine::Selection,
341    sh: &Shape,
342    w: f64,
343    h: f64,
344) -> Vec<crate::geom::warp::WarpHandle> {
345    if sh.text.is_none() {
346        return Vec::new();
347    }
348    let Some(ctx) = crate::engine::cmd::text::ctx_for(doc, sel) else { return Vec::new() };
349    let bp = crate::model::resolve::body(&ctx, sh);
350    let Some(name) = bp.warp.as_deref().filter(|n| crate::geom::warp::is_supported(n)) else { return Vec::new() };
351    let tr = crate::render::shape_geometry(sh, w, h).text_rect;
352    crate::geom::warp::handles(name, &bp.warp_adj, tr)
353}

355/// Write one warp adjust, keeping the others from `base` (empty = the preset defaults).
356fn set_warp_adj(sh: &mut Shape, name: &str, base: &[f64], index: usize, value: f64) {
357    let Some(text) = sh.text.as_mut() else { return };
358    if !value.is_finite() || index > 7 {
359        return;
360    }
361    let defaults = crate::geom::warp::adjust_defaults(name);
362    let n = defaults.len().max(index + 1).min(8);
363    if n == 0 {
364        return;
365    }
366    let mut adj = Vec::with_capacity(n);
367    for i in 0..n {
368        let v = if i == index { value } else { base.get(i).copied().filter(|v| v.is_finite()).or_else(|| defaults.get(i).copied()).unwrap_or(0.0) };
369        adj.push(v);
370    }
371    text.body.warp = Some(name.to_string());
372    text.body.warp_adj = adj;
373}

379    #[test]
380    fn resize_corner_and_flip() {
381        let o = Xfrm::new(100.0, 100.0, 100.0, 50.0);
382        let n = resize_box(&o, 4, Vec2::new(20.0, 10.0), false, false);
383        assert!((n.x - 100.0).abs() < 1e-9 && (n.w - 120.0).abs() < 1e-9 && (n.h - 60.0).abs() < 1e-9);
384        let n = resize_box(&o, 0, Vec2::new(10.0, 10.0), false, false);
385        assert!((n.x - 110.0).abs() < 1e-9 && (n.w - 90.0).abs() < 1e-9);
386        let n = resize_box(&o, 3, Vec2::new(-150.0, 0.0), false, false);
387        assert!(n.flip_h && (n.w - 50.0).abs() < 1e-9 && (n.x - 50.0).abs() < 1e-9);
388        let n = resize_box(&o, 4, Vec2::new(100.0, 0.0), true, false);
389        assert!((n.w / n.h - 2.0).abs() < 1e-9);
390        let n = resize_box(&o, 3, Vec2::new(10.0, 0.0), false, true);
391        assert!((n.w - 120.0).abs() < 1e-9 && (n.x - 90.0).abs() < 1e-9);
392    }

397        let anchor = o.affine() * Point::new(0.0, 0.0);
