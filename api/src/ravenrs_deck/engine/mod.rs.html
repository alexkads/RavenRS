1//! The DeckCraft engine façade.
2//!
3//! Every user-visible action is a command with a stable id (`slide.new`, `shape.insert`,
4//! `text.insert`, `format.bold`…) and JSON parameters. The egui UI, the CLI, the control channel
5//! and MCP all go through [`Session::execute`]. Pointer tools reduce to commands too, so every
6//! gesture is journaled, undoable and replayable.

46#[derive(Debug, thiserror::Error)]
47pub enum EngineError {
48    #[error("unknown command `{0}`")]
49    UnknownCommand(String),
50    #[error("command `{0}` is not available right now: {1}")]
51    Disabled(String, String),
52    #[error("invalid parameters for `{cmd}`: {msg}")]
53    BadParams { cmd: String, msg: String },
54    #[error("no active presentation")]
55    NoDocument,
56    #[error("{0}")]
57    Other(String),
58    #[error("internal error in `{0}` (the presentation was kept as it was): {1}")]
59    Internal(String, String),
60}
61
62impl From<crate::model::ModelError> for EngineError {
63    fn from(e: crate::model::ModelError) -> Self {
64        EngineError::Other(e.to_string())
65    }
66}

70/// Text being edited in place: a shape's body (or a table cell, or the slide's notes).
71#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
72#[serde(rename_all = "camelCase")]
73pub struct TextSel {
74    pub shape: ShapeId,
75    /// Selection anchor and caret: (paragraph, char).
76    pub anchor: (usize, usize),
77    pub caret: (usize, usize),
78    /// Editing a table cell (row, col).
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub cell: Option<(usize, usize)>,
81    /// Editing the slide's speaker notes (shape is ignored).
82    #[serde(default)]
83    pub notes: bool,
84    /// The caret position that sits at the end of a wrapped line (after End) rather than at the
85    /// start of the next line. Ignored once the caret moves anywhere else.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub line_end: Option<(usize, usize)>,
88}
89
90impl TextSel {
91    pub fn is_range(&self) -> bool {
92        self.anchor != self.caret
93    }
94    pub fn ordered(&self) -> ((usize, usize), (usize, usize)) {
95        if self.anchor <= self.caret { (self.anchor, self.caret) } else { (self.caret, self.anchor) }
96    }
97    /// Is the caret drawn at the end of the line before (upstream), not the start of the next?
98    pub fn at_line_end(&self) -> bool {
99        self.line_end == Some(self.caret)
100    }
101}

242static NEXT_UID: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);

244/// A document id unique across processes, not just within this one: recovery entries are named
245/// by it in a folder every running instance shares. Ids are this process's random base (see
246/// [`recovery::instance_id`]) plus a counter.
247fn next_uid() -> u64 {
248    recovery::instance_id().wrapping_add(NEXT_UID.fetch_add(1, std::sync::atomic::Ordering::Relaxed)) & recovery::UID_MASK
249}

251impl DocState {
252    pub fn new(doc: Presentation, path: Option<String>, title: String) -> Self {
253        let doc = Arc::new(doc);
254        DocState {
255            saved_doc: doc.clone(),
256            doc,
257            selection: Selection::default(),
258            history: History { limit: 500, ..Default::default() },
259            path,
260            revision: 1,
261            interaction: None,
262            uid: next_uid(),
263            title,
264            missing_fonts: Vec::new(),
265            composition: None,
266            compare: None,
267            last_duplicate: None,
268            auto_list: None,
269        }
270    }
271    pub fn is_dirty(&self) -> bool {
272        !Arc::ptr_eq(&self.doc, &self.saved_doc)
273    }
274    pub fn title(&self) -> String {
275        self.path
276            .as_deref()
277            .and_then(|p| std::path::Path::new(p).file_stem())
278            .map(|n| n.to_string_lossy().to_string())
279            .unwrap_or_else(|| self.title.clone())
280    }
281    pub fn current_slide(&self) -> Option<&Slide> {
282        self.doc.slides.get(self.selection.slide).map(|s| s.as_ref())
283    }
284    /// Shapes of the edited tree (current slide, or the master/layout in Slide Master view).
285    pub fn shapes(&self) -> &[Shape] {
286        match self.selection.target {
287            Target::Slides => self.current_slide().map(|s| s.shapes.as_slice()).unwrap_or(&[]),
288            Target::Master { master } => self.doc.masters.get(master).map(|m| m.shapes.as_slice()).unwrap_or(&[]),
289            Target::Handout | Target::NotesMaster => page_master(&self.doc, self.selection.target).map(|m| m.shapes.as_slice()).unwrap_or(&[]),
290            Target::Layout { master, layout } => {
291                self.doc.masters.get(master).and_then(|m| m.layouts.get(layout)).map(|l| l.shapes.as_slice()).unwrap_or(&[])
292            }
293        }
294    }
295    pub fn shape(&self, id: ShapeId) -> Option<&Shape> {
296        crate::model::find_shape(self.shapes(), id)
297    }
298    pub fn selected_shapes(&self) -> Vec<&Shape> {
299        self.selection.shapes.iter().filter_map(|id| self.shape(*id)).collect()
300    }
301}

308/// The handout or notes master as a one-master presentation at the notes page size, for drawing
309/// it with the layout renderer.
310pub fn master_page(doc: &Presentation, t: Target) -> Option<Presentation> {
311    let m = page_master(doc, t)?.clone();
312    let mut p = doc.clone();
313    p.slide_size = doc.notes_size;
314    p.masters = vec![m];
315    p.slides.clear();
316    Some(p)
317}

319/// Mutable access to the edited shape tree of a presentation.
320pub fn shapes_mut<'a>(doc: &'a mut Presentation, sel: &Selection) -> Option<&'a mut Vec<Shape>> {
321    match sel.target {
322        Target::Slides => doc.slides.get_mut(sel.slide).map(|s| &mut Arc::make_mut(s).shapes),
323        Target::Master { master } => doc.masters.get_mut(master).map(|m| &mut Arc::make_mut(m).shapes),
324        t @ (Target::Handout | Target::NotesMaster) => page_master_mut(doc, t).map(|m| &mut Arc::make_mut(m).shapes),
325        Target::Layout { master, layout } => {
326            doc.masters.get_mut(master).and_then(|m| Arc::make_mut(m).layouts.get_mut(layout)).map(|l| &mut l.shapes)
327        }
328    }

392impl Prefs {
393    /// The preferences with every value in a range the editor can use: `prefs.json` is a file
394    /// on disk, and a hand-edited or damaged one (a zero grid, a huge paste offset or undo
395    /// limit) must not break editing or exhaust memory. Out-of-range numbers take their
396    /// defaults or the nearest usable value.
397    pub fn sanitized(mut self) -> Prefs {
398        let d = Prefs::default();
399        let within = |v: f64, lo: f64, hi: f64, default: f64| if v.is_finite() { v.clamp(lo, hi) } else { default };
400        self.grid_spacing =
401            if self.grid_spacing.is_finite() && self.grid_spacing > 0.0 { self.grid_spacing.clamp(0.5, 1000.0) } else { d.grid_spacing };
402        self.duplicate_offset = within(self.duplicate_offset, -1000.0, 1000.0, d.duplicate_offset);
403        self.undo_limit = self.undo_limit.min(MAX_UNDO_LIMIT);
404        self.recovery_minutes = within(self.recovery_minutes, 0.1, 24.0 * 60.0, d.recovery_minutes);
405        if self.author.chars().count() > 256 {
406            self.author = self.author.chars().take(256).collect();
407        }
408        self
409    }
410}

528impl Session {
529    /// Install the process panic hook and construct an empty session with default preferences,
530    /// clipboard, tool state, and no active document.
531    pub fn new() -> Self {
532        guard::install_panic_hook();
533        Session {
534            docs: vec![],
535            active: None,
536            prefs: Prefs::default(),
537            journal: vec![],
538            clipboard: Clipboard::default(),
539            ui_requests: vec![],
540            tool: tools::ToolState::default(),
541            untitled: 0,
542            default_look: None,
543            painter: None,
544            anim_painter: None,
545            depth: 0,
546            recovery_dir: None,
547            media_status: Default::default(),
548            gif_paused: Default::default(),
549            mac_keys: cfg!(target_os = "macos"),
550            panics_caught: 0,
551            loaded: vec![],
552            next_loaded: 0,
553            composition_settles: 0,
554        }
555    }
556    /// Hand over a presentation read on another thread ([`cmd::file::read_presentation_file`]);
557    /// `file.adopt` with the returned token adds it as a document, through the command path like
558    /// `file.open` (journaled, panic-guarded). Only the latest few wait; older ones are dropped.
559    pub fn stage_loaded(&mut self, f: cmd::file::LoadedFile) -> u64 {
560        const KEEP: usize = 4;
561        self.next_loaded = self.next_loaded.wrapping_add(1);
562        if self.loaded.len() >= KEEP {
563            self.loaded.remove(0);
564        }
565        self.loaded.push((self.next_loaded, f));
566        self.next_loaded
567    }
568    /// The staged presentation for `token`, removed from the session.
569    pub(crate) fn take_loaded(&mut self, token: u64) -> Option<cmd::file::LoadedFile> {
570        let i = self.loaded.iter().position(|(t, _)| *t == token)?;
571        Some(self.loaded.remove(i).1)
572    }
573    /// A session with one new presentation open.
574    pub fn with_new() -> Self {
575        let mut s = Self::new();
576        let _ = s.execute("file.new", &Value::Null);
577        s
578    }

584    }

640    /// Resolve and run a command, settling composition first, checking eligibility, rolling back a
641    /// failed outer command, and committing the resulting undo entry or journal record.
642    fn execute_unguarded(&mut self, id: &str, params: &Value) -> Result<Value> {
643        let spec = find_command(id).ok_or_else(|| EngineError::UnknownCommand(id.into()))?;
644        // Any other edit (or a click, a key, undo) settles an input method's composition first.
645        if self.depth == 0 && (spec.journal || spec.undoable) && !matches!(id, "text.compose" | "text.commit") {
646            self.settle_composition();
647        }
648        if let Err(e) = (spec.enabled)(self) {
649            // A command told which shapes to act on doesn't depend on what is selected.
650            let named = (e == cmd::NOTHING_SELECTED || e == cmd::SELECT_TWO || e == cmd::merge::IN_GROUP)
651                && (params.get("ids").is_some() || params.get("id").is_some());
652            if !named {
653                return Err(EngineError::Disabled(id.into(), e));
654            }
655        }
656        let before = self.active().map(|d| (d.uid, d.doc.clone(), d.selection.clone()));
657        let vertex_kind = self.tool.vertex_kind.clone();
658        let outer = self.depth == 0;
659        self.depth += 1;
660        let r = (spec.run)(self, params);
661        self.depth -= 1;
662        let r = match r {
663            Ok(r) => r,
664            Err(e) => {
665                // A command made of several edits can fail after some of them went through; the
666                // outermost one puts the presentation back, so a failed command changes nothing.
667                if outer
668                    && let (Some((uid, old, sel)), Some(st)) = (before, self.active_mut())
669                    && st.uid == uid
670                    && !Arc::ptr_eq(&old, &st.doc)
671                {
672                    st.doc = old;
673                    st.selection = sel;
674                    st.revision += 1;
675                }
676                if outer {
677                    self.tool.vertex_kind = vertex_kind;
678                }
679                return Err(e);
680            }
681        };
682        if let Some(st) = self.active_mut() {
683            clamp_selection(st);
684        }
685        if let (Some((uid, old, sel)), Some(st)) = (before, self.active_mut())
686            && st.uid == uid
687            && !Arc::ptr_eq(&old, &st.doc)
688            && st.interaction.is_none()
689            && spec.undoable
690            && outer
691        {
692            push_undo(st, HistoryEntry { label: spec.label.to_string(), doc: old, selection: sel, vertex_kind });
693        }
694        if spec.journal && outer {
695            self.journal.push((id.to_string(), params.clone()));
696            if self.journal.len() > 10_000 {
697                self.journal.drain(..1000);
698            }
699        }
700        Ok(if outer { self.with_editing(id, r) } else { r })
701    }

703    /// In Slide Master / layout (or handout / notes master) view the insert commands edit the
704    /// master or layout, not a slide; their answer says so with `editing: {kind, name}`, as
705    /// `slide.inspect` does. Nothing is added on slides, so those answers stay as they were.
706    fn with_editing(&self, id: &str, mut r: Value) -> Value {
707        let inserts = id.starts_with("insert.") || matches!(id, "shape.insert" | "shape.freeform" | "shape.connect" | "slide.new");
708        let (true, Some(st), Some(o)) = (inserts, self.active(), r.as_object_mut()) else { return r };
709        let doc = &st.doc;
710        let edited = match st.selection.target {
711            Target::Master { master } => doc.masters.get(master).map(|m| ("master", m.name.clone())),
712            Target::Layout { master, layout } => doc.masters.get(master).and_then(|m| m.layouts.get(layout)).map(|l| ("layout", l.name.clone())),
713            t @ (Target::Handout | Target::NotesMaster) => {
714                page_master(doc, t).map(|m| (if matches!(t, Target::Handout) { "handout" } else { "notesMaster" }, m.name.clone()))
715            }
716            Target::Slides => None,
717        };
718        if let Some((kind, name)) = edited {
719            o.entry("editing").or_insert_with(|| serde_json::json!({"kind": kind, "name": name}));
720        }
721        r
722    }

728    /// Mutate the active presentation (copy-on-write) and bump the revision.
729    pub fn edit<T>(&mut self, f: impl FnOnce(&mut Presentation, &mut Selection) -> Result<T>) -> Result<T> {
730        let st = self.doc_mut()?;
731        let mut doc = (*st.doc).clone();
732        let mut sel = st.selection.clone();
733        let r = f(&mut doc, &mut sel)?;
734        cmd::arrange::refit_groups(&st.doc, &mut doc);
735        connect::reroute(&mut doc, &sel);
736        st.doc = Arc::new(doc);
737        st.selection = sel;
738        st.revision += 1;
739        Ok(r)
740    }

742    /// Keep an input method's composition as it stands (#101): its text stays and the whole
743    /// composition becomes one undo step. Nothing happens without one.
744    ///
745    /// This is also how the input method's own commit ends a composition: the platform has
746    /// already finished that one, so it must not ask for an interrupt.
747    pub fn end_composition(&mut self) {
748        let vertex_kind = self.tool.vertex_kind.clone();
749        let Some(st) = self.active_mut() else { return };
750        let Some(c) = st.composition.take() else { return };
751        if !Arc::ptr_eq(&c.before, &st.doc) {
752            push_undo(st, HistoryEntry { label: "Type Text".into(), doc: c.before, selection: c.before_selection, vertex_kind });
753        }
754    }
755
756    /// Settle a composition because the application did something else (a click, a key, a
757    /// command, a target change) rather than because the input method committed it.
758    ///
759    /// The platform still holds its own copy of that composition, so a host must publish one
760    /// `should_interrupt_composition` for each increment of [`Self::composition_settles`]: the
761    /// backend then cancels it instead of committing its text into whatever is selected next.
762    pub fn settle_composition(&mut self) {
763        if self.active().is_some_and(|d| d.composition.is_some()) {
764            self.composition_settles = self.composition_settles.wrapping_add(1);
765        }
766        self.end_composition();
767    }
768
769    /// How many compositions the application settled itself (clicks, keys, other commands).
770    ///
771    /// A host compares this with the value it last published; every increment is one platform
772    /// composition that must be interrupted. The input method's own commits do not count.
773    pub fn composition_settles(&self) -> u64 {
774        self.composition_settles
775    }

777    /// Change only the selection (no undo step).
778    pub fn select(&mut self, f: impl FnOnce(&Presentation, &mut Selection)) -> Result<()> {
779        let st = self.doc_mut()?;
780        let doc = st.doc.clone();
781        // A caret placed anywhere starts downstream; only End puts it back at a line's end.
782        if let Some(t) = st.selection.text.as_mut() {
783            t.line_end = None;
784        }
785        let owner = |sel: &Selection| sel.text.as_ref().map(|t| (sel.slide, sel.target, t.shape, t.cell, t.notes));
786        let before = owner(&st.selection);
787        f(&doc, &mut st.selection);
788        clamp_selection(st);
789        st.revision += 1;
790        // A composition belongs to the text body it began in: leaving that body settles it.
791        if st.composition.is_some() && owner(&st.selection) != before {
792            self.settle_composition();
793        }
794        Ok(())
795    }
796}

798/// Keep the selection pointing at things that exist.
799fn clamp_selection(st: &mut DocState) {
800    let n = st.doc.slides.len();
801    if n == 0 {
802        st.selection.slide = 0;
803    } else if st.selection.slide >= n {
804        st.selection.slide = n - 1;
805    }
806    let ids: Vec<ShapeId> = st.selection.shapes.iter().copied().filter(|id| st.shape(*id).is_some()).collect();
807    st.selection.shapes = ids;
808    if let Some(t) = &st.selection.text
809        && !t.notes
810        && st.shape(t.shape).is_none()
811    {
812        st.selection.text = None;
813    }
814    let slides: Vec<SlideId> = st.selection.slides.iter().copied().filter(|id| st.doc.slide(*id).is_some()).collect();
815    st.selection.slides = slides;
816    match st.selection.target {
817        Target::Master { master } if master >= st.doc.masters.len() => st.selection.target = Target::Slides,
818        t @ (Target::Handout | Target::NotesMaster) if page_master(&st.doc, t).is_none() => st.selection.target = Target::Slides,
819        Target::Layout { master, layout } if st.doc.masters.get(master).is_none_or(|m| layout >= m.layouts.len()) => {
820            st.selection.target = Target::Slides
821        }
822        _ => {}
823    }
824}

826/// Default for [`History::byte_limit`].
827pub const UNDO_BYTE_BUDGET: usize = 256 << 20;

829/// A rough size of a text body in memory: its paragraphs and runs, at their inline struct sizes.
830fn text_cost(t: &crate::model::text::TextBody) -> usize {
831    use crate::model::text::{Paragraph, Run};
832    let (para, run) = (std::mem::size_of::<Paragraph>(), std::mem::size_of::<Run>());
833    t.paragraphs.iter().map(|p| para + p.runs.iter().map(|r| run + r.text.len()).sum::<usize>()).sum()
834}
835
836/// A rough size of comments in memory, replies included.
837fn comments_cost(cs: &[crate::model::Comment], depth: usize) -> usize {
838    let kids = |c: &crate::model::Comment| if depth < 64 { comments_cost(&c.replies, depth + 1) } else { 0 };
839    cs.iter().map(|c| 256 + c.author.len() + c.initials.len() + c.text.len() + c.date.len() + kids(c)).sum()
840}
841
842/// A rough size of a slide's own data besides its shapes: notes, comments, tags and kept XML,
843/// all copied with the slide when an edit changes it.
844fn slide_extra_cost(s: &crate::model::Slide) -> usize {
845    text_cost(&s.notes)
846        + comments_cost(&s.comments, 0)
847        + s.name.len()
848        + s.tags.iter().map(|(k, v)| 48 + k.len() + v.len()).sum::<usize>()
849        + s.raw_ext.as_ref().map_or(0, String::len)
850}
851
852/// A rough size of a shape in memory: the struct, its text and kept XML, its children.
853fn shape_cost(sh: &crate::model::Shape, depth: usize) -> usize {
854    let text: usize = sh.text.as_ref().map_or(0, text_cost);
855    let kids: usize = if depth < 64 { sh.children().iter().map(|c| shape_cost(c, depth + 1)).sum() } else { 0 };
856    768 + text + sh.raw_ext.as_ref().map_or(0, String::len) + kids
857}
858
859fn shapes_cost(shapes: &[crate::model::Shape]) -> usize {
860    shapes.iter().map(|s| shape_cost(s, 0)).sum()
861}

863/// Estimated bytes `doc` keeps alive that `newer` doesn't share: its slides, masters and media
864/// that aren't the same allocations (undo steps share everything an edit didn't touch).
865pub fn undo_cost(doc: &Presentation, newer: &Presentation) -> usize {
866    use std::collections::HashSet;
867    let slides: HashSet<*const crate::model::Slide> = newer.slides.iter().map(Arc::as_ptr).collect();
868    let masters: HashSet<*const crate::model::Master> = newer.masters.iter().map(Arc::as_ptr).collect();
869    let media: HashSet<*const Vec<u8>> = newer.media.iter().map(|m| Arc::as_ptr(&m.data)).collect();
870    let mut n = 64 * doc.slides.len();
871    for s in doc.slides.iter().filter(|s| !slides.contains(&Arc::as_ptr(s))) {
872        n += 512 + shapes_cost(&s.shapes) + slide_extra_cost(s);
873    }
874    for m in doc.masters.iter().filter(|m| !masters.contains(&Arc::as_ptr(m))) {
875        n += 4096 + shapes_cost(&m.shapes) + m.layouts.iter().map(|l| 512 + shapes_cost(&l.shapes)).sum::<usize>();
876    }
877    for m in doc.media.iter().filter(|m| !media.contains(&Arc::as_ptr(&m.data))) {
878        n += m.data.len();
879    }
880    n
881}

883pub(crate) fn push_undo(st: &mut DocState, e: HistoryEntry) {
884    let h = &mut st.history;
885    h.costs.insert(Arc::as_ptr(&e.doc) as usize, undo_cost(&e.doc, &st.doc));
886    h.undo.push(e);
887    h.redo.clear();
888    if h.undo.len() > h.limit.max(1) {
889        h.undo.remove(0);
890    }
891    // Memory too: drop the oldest steps while the kept ones pass the budget (the newest always
892    // stays, however big). A deck-wide edit keeps a copy of every slide, so 500 of them on a big
893    // deck could hold hundreds of copies.
894    let budget = if h.byte_limit == 0 { UNDO_BYTE_BUDGET } else { h.byte_limit };
895    let cost = |h: &History, e: &HistoryEntry| h.costs.get(&(Arc::as_ptr(&e.doc) as usize)).copied().unwrap_or(0);
896    let mut total: usize = h.undo.iter().map(|e| cost(h, e)).sum();
897    let mut drop = 0;
898    while total > budget && drop + 1 < h.undo.len() {
899        total -= cost(h, &h.undo[drop]);
900        drop += 1;
901    }
902    if drop > 0 {
903        h.undo.drain(..drop);
904    }
905    // Costs of steps that are gone (dropped, or redo steps cleared above).
906    if h.costs.len() > h.undo.len() + 16 {
907        let live: std::collections::HashSet<usize> = h.undo.iter().chain(h.redo.iter()).map(|e| Arc::as_ptr(&e.doc) as usize).collect();
908        h.costs.retain(|k, _| live.contains(k));
909    }
910}
