1//! Last-resort guard: a panic that escapes a command —
2//! any edit, import or export, from the UI, the CLI, MCP or the control channel — becomes an
3//! error, and the document is kept as it was before the command (and saved for crash recovery
4//! when a recovery directory is set). It is a safety net: commands still must not panic.
5
6use std::panic::{AssertUnwindSafe, catch_unwind};
7
8use serde_json::Value;
9
10use crate::engine::{EngineError, Result, Session};
11
12/// Text of a panic payload (`panic!` with a literal or a formatted message).
13pub fn panic_message(payload: &(dyn std::any::Any + Send)) -> String {
14    payload
15        .downcast_ref::<&str>()
16        .map(ToString::to_string)
17        .or_else(|| payload.downcast_ref::<String>().cloned())
18        .unwrap_or_else(|| "unknown panic".into())
19}
20
21/// Log every panic (message and location) before it unwinds, then run the previous hook.
22/// Installed once per process, when the first [`Session`] is made.
23pub fn install_panic_hook() {
24    static ONCE: std::sync::Once = std::sync::Once::new();
25    ONCE.call_once(|| {
26        let previous = std::panic::take_hook();
27        std::panic::set_hook(Box::new(move |info| {
28            let at = info.location().map(|l| format!(" at {}:{}", l.file(), l.line())).unwrap_or_default();
29            log::error!("panic{at}: {}", panic_message(info.payload()));
30            previous(info);
31        }));
32    });
33}

35impl Session {
36    /// Run command `id` (`f`) inside the guard.
37    pub(crate) fn guarded(&mut self, id: &str, f: impl FnOnce(&mut Session) -> Result<Value>) -> Result<Value> {
38        let before = self.active().map(|d| (d.uid, d.doc.clone(), d.selection.clone(), d.revision));
39        let vertex_kind = self.tool.vertex_kind.clone();
40        let depth = self.depth;
41        match catch_unwind(AssertUnwindSafe(|| f(self))) {
42            Ok(r) => r,
43            Err(payload) => {
44                self.depth = depth;
45                // The remembered point type names a shape of the document it was chosen in.
46                let same = before.as_ref().is_some_and(|b| self.active().is_some_and(|st| st.uid == b.0));
47                self.tool.vertex_kind = if same { vertex_kind } else { None };
48                let msg = panic_message(payload.as_ref());
49                // Keep the document as it was before the command.
50                if let (Some((uid, doc, selection, revision)), Some(st)) = (before, self.active_mut())
51                    && st.uid == uid
52                {
53                    st.doc = doc;
54                    st.selection = selection;
55                    st.interaction = None;
56                    st.revision = revision.max(st.revision).saturating_add(1);
57                }
58                log::error!("command `{id}` panicked: {msg}");
59                self.panics_caught = self.panics_caught.saturating_add(1);
60                if let Some(dir) = self.recovery_dir.clone()
61                    && let Err(e) = crate::engine::recovery::save(self, &dir)
62                {
63                    log::error!("recovery save after `{id}` failed: {e}");
64                }
65                Err(EngineError::Internal(id.to_string(), msg))
66            }
67        }
68    }
69}

77    #[test]
78    fn a_panicking_command_is_an_error_and_keeps_the_document() {
79        let mut s = Session::new();
80        s.execute("file.new", &json!({})).unwrap();
81        s.execute("shape.insert", &json!({"preset": "rect", "rect": [36, 36, 200, 200]})).unwrap();
82        let before = s.active().unwrap().doc.clone();
83        // A command that edits the document and then panics half way.
84        let err = s
85            .guarded("test.panic", |s| {
86                s.execute("shape.insert", &json!({"preset": "ellipse", "rect": [36, 236, 200, 400]}))?;
87                panic!("boom");
88            })
89            .unwrap_err();
90        assert!(err.to_string().contains("test.panic") && err.to_string().contains("boom"), "{err}");
91        let st = s.active().unwrap();
92        assert!(std::sync::Arc::ptr_eq(&st.doc, &before), "the document is the one from before the command");
93        assert_eq!(s.panics_caught, 1, "the guard counts what it caught");
94        // The session keeps working.
95        s.execute("shape.insert", &json!({"preset": "ellipse", "rect": [36, 236, 200, 400]})).unwrap();
96    }

98    #[test]
99    fn a_panicking_command_keeps_the_remembered_point_type() {
100        let mut s = Session::new();
101        s.execute("file.new", &json!({})).unwrap();
102        let kept = Some((crate::model::ShapeId(1), 0, 0, "straight".to_string()));
103        s.tool.vertex_kind = kept.clone();
104        let r = s.guarded("test.panic", |s| {
105            s.tool.vertex_kind = Some((crate::model::ShapeId(1), 0, 0, "corner".to_string()));
106            panic!("boom");
107        });
108        assert!(r.is_err());
109        assert_eq!(s.tool.vertex_kind, kept);
110        // A command that opens another document and then panics leaves no point type behind: the
111        // one from before names a shape of the other document.
112        let r = s.guarded("test.panic", |s| {
113            s.execute("file.new", &json!({}))?;
114            s.tool.vertex_kind = Some((crate::model::ShapeId(1), 0, 0, "corner".to_string()));
115            panic!("boom");
116        });
117        assert!(r.is_err());
118        assert_eq!(s.tool.vertex_kind, None);
119    }

121    #[test]
122    fn a_command_that_fails_after_editing_leaves_the_document_alone() {
123        let mut s = Session::new();
124        s.execute("file.new", &json!({})).unwrap();
125        let a = s.execute("shape.insert", &json!({"preset": "rect", "rect": [36, 36, 100, 100]})).unwrap()["id"].as_u64().unwrap();
126        let b = s.execute("shape.insert", &json!({"preset": "rect", "rect": [236, 36, 100, 100]})).unwrap()["id"].as_u64().unwrap();
127        let before = s.active().unwrap().doc.clone();
128        let undo_len = s.active().unwrap().history.undo.len();
129        // `shape.connect` inserts the requested preset first, then rejects it because it is not a line.
130        let r = s.execute("shape.connect", &json!({"from": a, "to": b, "preset": "rect"}));
131        assert!(r.is_err());
132        let st = s.active().unwrap();
133        assert!(std::sync::Arc::ptr_eq(&st.doc, &before), "the stray shape is gone");
134        assert_eq!(st.history.undo.len(), undo_len);
135    }
