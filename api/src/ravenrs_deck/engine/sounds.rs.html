1//! Built-in transition sounds, synthesised on demand so DeckCraft ships no sound files: short
2//! mono 16-bit WAVs named like PowerPoint's (Chime, Click, Camera, Whoosh, Drum Roll, Laser).
3
4use std::f64::consts::TAU;
5
6/// (id, label) of the built-in sounds, in menu order.
7pub const BUILTIN: [(&str, &str); 6] =
8    [("chime", "Chime"), ("click", "Click"), ("camera", "Camera"), ("whoosh", "Whoosh"), ("drumRoll", "Drum Roll"), ("laser", "Laser")];
9
10const RATE: u32 = 22_050;
11
12/// The WAV bytes of built-in sound `id`, or `None` for an unknown id.
13pub fn builtin(id: &str) -> Option<Vec<u8>> {
14    let mut noise = Lcg(0x9E37_79B9);
15    let samples: Vec<f64> = match id {
16        "chime" => tone(1.2, |t| {
17            let env = (-t * 3.0).exp() * attack(t, 0.005);
18            env * (0.6 * (TAU * 880.0 * t).sin() + 0.3 * (TAU * 1318.5 * t).sin() + 0.15 * (TAU * 1760.0 * t).sin())
19        }),
20        "click" => tone(0.03, |t| (-t * 300.0).exp() * noise.next()),
21        "camera" => tone(0.35, |t| {
22            let a = (-t * 250.0).exp();
23            let b = if t > 0.18 { (-(t - 0.18) * 180.0).exp() } else { 0.0 };
24            (a + 0.8 * b) * noise.next()
25        }),
26        "whoosh" => {
27            let mut lp = 0.0;
28            tone(0.7, |t| {
29                // Noise through a low-pass filter that opens and closes.
30                let k = 0.02 + 0.25 * (std::f64::consts::PI * t / 0.7).sin();
31                lp += k * (noise.next() - lp);
32                3.0 * lp * (std::f64::consts::PI * t / 0.7).sin()
33            })
34        }
35        "drumRoll" => tone(1.4, |t| {
36            let beat = (t * 24.0).fract() / 24.0;
37            let swell = 0.4 + 0.6 * (t / 1.4);
38            swell * (-beat * 90.0).exp() * (0.7 * noise.next() + 0.5 * (TAU * 110.0 * t).sin())
39        }),
40        "laser" => {
41            let mut phase = 0.0;
42            tone(0.5, |t| {
43                phase += TAU * (2400.0 * (-t * 5.0).exp() + 200.0) / RATE as f64;
44                (-t * 4.0).exp() * attack(t, 0.003) * phase.sin().signum() * 0.35
45            })
46        }
47        _ => return None,
48    };
49    Some(wav(&samples))
50}
51
52fn attack(t: f64, len: f64) -> f64 {
53    (t / len).min(1.0)
54}
55
56fn tone(secs: f64, mut f: impl FnMut(f64) -> f64) -> Vec<f64> {
57    let n = (secs * RATE as f64) as usize;
58    (0..n).map(|i| f(i as f64 / RATE as f64)).collect()
59}
60
61/// A small deterministic noise source (-1..1).
62struct Lcg(u32);
63impl Lcg {
64    fn next(&mut self) -> f64 {
65        self.0 = self.0.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
66        (self.0 >> 8) as f64 / (1u32 << 23) as f64 - 1.0
67    }
68}
69
70fn wav(samples: &[f64]) -> Vec<u8> {
71    let data_len = (samples.len() * 2) as u32;
72    let mut v = Vec::with_capacity(44 + data_len as usize);
73    v.extend_from_slice(b"RIFF");
74    v.extend_from_slice(&(36 + data_len).to_le_bytes());
75    v.extend_from_slice(b"WAVEfmt ");
76    v.extend_from_slice(&16u32.to_le_bytes());
77    v.extend_from_slice(&1u16.to_le_bytes()); // PCM
78    v.extend_from_slice(&1u16.to_le_bytes()); // mono
79    v.extend_from_slice(&RATE.to_le_bytes());
80    v.extend_from_slice(&(RATE * 2).to_le_bytes());
81    v.extend_from_slice(&2u16.to_le_bytes());
82    v.extend_from_slice(&16u16.to_le_bytes());
83    v.extend_from_slice(b"data");
84    v.extend_from_slice(&data_len.to_le_bytes());
85    for s in samples {
86        let x = (s.clamp(-1.0, 1.0) * 0.8 * i16::MAX as f64) as i16;
87        v.extend_from_slice(&x.to_le_bytes());
88    }
89    v
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95
96    #[test]
97    fn every_builtin_sound_is_a_playable_wav() {
98        for (id, _) in BUILTIN {
99            let bytes = builtin(id).unwrap_or_default();
100            assert!(bytes.len() > 44, "{id}");
101            let info = crate::media::probe(&std::sync::Arc::new(bytes)).expect(id);
102            assert!(info.audio.is_some() && info.playable(), "{id}");
103            assert!(info.duration() > 0.01 && info.duration() < 2.0, "{id}: {}", info.duration());
104        }
105        assert!(builtin("nope").is_none());
106    }
107}
