49fn restore(s: &mut Session, redo: bool) -> Result<Value> {
50    let vertex_kind = s.tool.vertex_kind.clone();
51    let st = s.doc_mut()?;
52    let (from, to) = if redo { (&mut st.history.redo, &mut st.history.undo) } else { (&mut st.history.undo, &mut st.history.redo) };
53    let Some(e) = from.pop() else { return ok() };
54    to.push(HistoryEntry { label: e.label.clone(), doc: st.doc.clone(), selection: st.selection.clone(), vertex_kind });
55    st.doc = e.doc;
56    st.selection = e.selection;
57    st.interaction = None;
58    // A later shape can get an undone copy's id: Duplicate starts over with the usual offset.
59    st.last_duplicate = None;
60    st.revision += 1;
61    // The point type Edit Points remembered with the restored document: handle drags follow it.
62    s.tool.vertex_kind = e.vertex_kind;
63    Ok(json!({"label": e.label}))
64}

78fn copy_inner(s: &mut Session, p: &Value) -> Result<String> {
79    let st = s.doc()?;
80    // Text selection.
81    if let Some(t) = &st.selection.text {
82        if t.is_range() {
83            let body = super::text::body_of(st, t).cloned().unwrap_or_default();
84            let (a, b) = t.ordered();
85            let plain = crate::model::edit::text_range(&body, a, b);
86            let sliced = crate::model::edit::slice(&body, a, b);
87            s.clipboard = crate::engine::Clipboard { text: Some(sliced), plain: plain.clone(), ..Default::default() };
88            return Ok(plain);
89        }
90        return Ok(String::new());
91    }
92    if slides_scope(s, p) {
93        let ids: Vec<SlideId> =
94            if st.selection.slides.is_empty() { st.current_slide().map(|x| vec![x.id]).unwrap_or_default() } else { st.selection.slides.clone() };
95        let slides: Vec<Slide> = ids.iter().filter_map(|id| st.doc.slide(*id).cloned()).collect();
96        let mut mids = vec![];
97        for sl in &slides {
98            mids.extend(crate::engine::media_refs::slide_media(sl));
99        }
100        let media = st.doc.media.iter().filter(|m| mids.contains(&m.id)).cloned().collect();
101        let plain = slides.iter().map(|x| x.title()).collect::<Vec<_>>().join("\n");
102        s.clipboard = crate::engine::Clipboard { slides, media, plain: plain.clone(), ..Default::default() };
103        return Ok(plain);
104    }
105    // Selected table cells copy as their text: a tab between cells and a paragraph per row.
106    if let Some(plain) = super::table::selected_cells_text(st).filter(|_| p.get("id").is_none() && p.get("ids").is_none()) {
107        s.clipboard = crate::engine::Clipboard { text: Some(TextBody::from_text(&plain)), plain: plain.clone(), ..Default::default() };
108        return Ok(plain);
109    }
110    let mut shapes: Vec<Shape> = st
111        .selection
112        .shapes
113        .iter()
114        .filter_map(|id| {
115            st.shape(*id).cloned().map(|mut sh| {
116                if sh.xfrm.is_none() {
117                    sh.xfrm = Some(xfrm_of(&st.doc, &st.selection, &sh));
118                }
119                sh
120            })
121        })
122        .collect();
123    // A shape in a group is copied with the box it is drawn in, so it pastes where it was seen.
124    for sh in &mut shapes {
125        if let Some(x) = sh.xfrm.and_then(|x| box_on_slide(&st.doc, &st.selection, st.shapes(), sh.id, &x)) {
126            sh.xfrm = Some(x);
127        }
128    }
129    // Copying is the last chance to read what a placeholder inherits: a cut removes it, and the
130    // paste can be in another deck. Slide Master view pastes the originals instead.
131    let placeholders: Vec<Shape> = shapes.iter().filter(|x| x.ph.is_some()).cloned().collect();
132    detach_on_slide(&st.doc, &st.selection, &mut shapes);
133    let mids = crate::engine::media_refs::shapes_media(&shapes);
134    let media = st.doc.media.iter().filter(|m| mids.contains(&m.id)).cloned().collect();
135    let plain = shapes.iter().filter_map(|x| x.text.as_ref().map(|t| t.text())).collect::<Vec<_>>().join("\n");
136    // The theme of what is being edited: a slide's or layout's colour map, else the master's.
137    let source_theme = super::text::ctx_for(&st.doc, &st.selection).map(|ctx| crate::engine::SourceTheme {
138        colors: ctx.master.theme.colors.clone(),
139        color_map: ctx.color_map.clone(),
140        fonts: ctx.master.theme.fonts.clone(),
141    });
142    s.clipboard = crate::engine::Clipboard { shapes, placeholders, media, plain: plain.clone(), source_theme, ..Default::default() };
143    Ok(plain)
144}

146/// On a slide, a copy of a placeholder is a regular shape: give `shapes` the look they inherit
147/// there. In Slide Master view a copy stays a placeholder, so this changes nothing.
148pub(crate) fn detach_on_slide(doc: &crate::model::Presentation, sel: &crate::engine::Selection, shapes: &mut [Shape]) {
149    if sel.target == crate::engine::Target::Slides
150        && let Some(ctx) = super::text::ctx_for(doc, sel)
151    {
152        shapes.iter_mut().for_each(|sh| crate::model::resolve::detach_placeholder(&ctx, sh));
153    }
154}

156/// Where shape `id` is drawn on the slide when it is inside a group: `x`, its box in its group's
157/// space, mapped out through each group around it, as Ungroup places a freed shape. `None` for a
158/// shape on the top level or not found. A turned shape in a group stretched more one way than the
159/// other is drawn skewed, which a box cannot hold, so it gets the unskewed box Ungroup gives it.
160fn box_on_slide(
161    doc: &crate::model::Presentation,
162    sel: &crate::engine::Selection,
163    shapes: &[Shape],
164    id: ShapeId,
165    x: &crate::geom::Xfrm,
166) -> Option<crate::geom::Xfrm> {
167    fn rec(
168        doc: &crate::model::Presentation,
169        sel: &crate::engine::Selection,
170        shapes: &[Shape],
171        id: ShapeId,
172        x: &crate::geom::Xfrm,
173        depth: usize,
174    ) -> Option<crate::geom::Xfrm> {
175        if depth > 64 {
176            return None;
177        }
178        shapes.iter().find_map(|sh| {
179            let crate::model::ShapeKind::Group { children, child } = &sh.kind else {
180                return None;
181            };
182            let inner = if children.iter().any(|c| c.id == id) { *x } else { rec(doc, sel, children, id, x, depth + 1)? };
183            Some(super::arrange::child_to_parent(&xfrm_of(doc, sel, sh), child, &inner))
184        })
185    }
186    rec(doc, sel, shapes, id, x, 0)
187}

193fn copy(s: &mut Session, p: &Value) -> Result<Value> {
194    // A caret copies nothing; the previous clipboard must not leak into this answer.
195    let caret = copying_caret(s);
196    let t = copy_inner(s, p)?;
197    let html = if caret { String::new() } else { copied_html(s) };
198    Ok(json!({"text": t, "html": html}))
199}

201/// HTML for what was just copied, when it is text (a selection, or the text on the copied shapes).
202fn copied_html(s: &Session) -> String {
203    if let Some(body) = &s.clipboard.text {
204        return richclip::body_to_html(body);
205    }
206    let mut paragraphs = Vec::new();
207    for sh in &s.clipboard.shapes {
208        if let Some(t) = &sh.text {
209            paragraphs.extend(t.paragraphs.clone());
210        }
211    }
212    if paragraphs.is_empty() {
213        return String::new();
214    }
215    richclip::body_to_html(&TextBody { paragraphs, ..Default::default() })
216}

218fn cut(s: &mut Session, p: &Value) -> Result<Value> {
219    let caret = copying_caret(s);
220    let t = copy_inner(s, p)?;
221    if caret {
222        // Nothing is selected, so there is nothing to cut; `delete` would remove the next character.
223        return Ok(json!({"text": t, "html": ""}));
224    }
225    let html = copied_html(s);
226    delete(s, p)?;
227    Ok(json!({"text": t, "html": html}))
228}

230/// Give pasted shapes fresh ids (groups included) and return the old-to-new map. Connector ends
231/// attached to a shape that was copied too follow its copy; ends attached to a shape outside the
232/// copied set are detached (the connector keeps its geometry), so a copy never stays glued to
233/// the original or to an unrelated shape that happens to share an id in another deck.
234pub(crate) fn reid(cmd: &str, doc: &mut crate::model::Presentation, shapes: &mut [Shape]) -> Result<HashMap<ShapeId, ShapeId>> {
235    fn rec(cmd: &str, doc: &mut crate::model::Presentation, shapes: &mut [Shape], map: &mut HashMap<ShapeId, ShapeId>, depth: usize) -> Result<()> {
236        if depth > 64 {
237            return Ok(());
238        }
239        for s in shapes {
240            let id = ShapeId(doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE))?);
241            map.insert(s.id, id);
242            s.id = id;
243            if let Some(ch) = s.children_mut() {
244                rec(cmd, doc, ch, map, depth + 1)?;
245            }
246        }
247        Ok(())
248    }
249    let mut map = HashMap::new();
250    rec(cmd, doc, shapes, &mut map, 0)?;
251    walk_mut(shapes, &mut |s| {
252        if let crate::model::ShapeKind::Connector { start, end } = &mut s.kind {
253            for e in [start, end] {
254                *e = e.and_then(|(id, site)| map.get(&id).map(|n| (*n, site)));
255            }
256        }
257    });
258    Ok(map)
259}

261/// Point copied animations at the copies: targets and triggers in `map` follow it; an animation
262/// whose target wasn't copied is dropped, and a trigger that wasn't copied is cleared.
263pub(crate) fn remap_animations(anims: &mut Vec<crate::model::Animation>, map: &HashMap<ShapeId, ShapeId>) {
264    anims.retain_mut(|a| {
265        let Some(n) = map.get(&a.shape) else { return false };
266        a.shape = *n;
267        a.trigger = a.trigger.and_then(|t| map.get(&t).copied());
268        true
269    });
270}

272/// Visit every shape mutably, depth-first (groups included, depth-bounded like `walk`).
273fn walk_mut(shapes: &mut [Shape], f: &mut dyn FnMut(&mut Shape)) {
274    fn rec(shapes: &mut [Shape], f: &mut dyn FnMut(&mut Shape), depth: usize) {
275        if depth > 64 {
276            return;
277        }
278        for s in shapes {
279            f(s);
280            if let Some(ch) = s.children_mut() {
281                rec(ch, f, depth + 1);
282            }
283        }
284    }
285    rec(shapes, f, 0);
286}

288/// Rewrite a media id the clipboard used (`map`: clipboard id to destination id).
289fn remap_media(map: &HashMap<MediaId, MediaId>) -> impl FnMut(&mut MediaId) + '_ {
290    move |id: &mut MediaId| {
291        if let Some(n) = map.get(id) {
292            *id = *n;
293        }
294    }
295}

297/// Add the clipboard's media to `doc` and return the ids that changed: an id already used in
298/// `doc` by other bytes (another deck) gets a fresh id, and the pasted shapes must use it.
299fn bring_media(cmd: &str, doc: &mut crate::model::Presentation, media: &[crate::model::MediaItem]) -> Result<HashMap<MediaId, MediaId>> {
300    let mut map = HashMap::new();
301    for m in media {
302        match doc.media(m.id) {
303            None => {
304                // Keeps the foreign id, so later ids must start past it.
305                doc.media.push(m.clone());
306                doc.next_id = doc.next_id.max(m.id.0);
307            }
308            Some(x) if x.data == m.data => {}
309            Some(_) => {
310                let id = doc.add_media(&m.name, &m.content_type, m.data.as_ref().clone()).ok_or_else(|| bad(cmd, NO_ID_SPACE))?;
311                map.insert(m.id, id);
312            }
313        }
314    }
315    Ok(map)
316}

318fn paste(s: &mut Session, p: &Value) -> Result<Value> {
319    // External plain text (from the system clipboard) wins when given and differs from ours.
320    // HTML from Pages or Keynote keeps bold, italic and underline when it says the same words.
321    if let Some(t) = str_param(p, "text")
322        && t != s.clipboard.plain
323    {
324        if let Some(html) = str_param(p, "html")
325            && let Some(body) = richclip::html_to_body(html)
326            && richclip::same_words(&body.text(), t)
327        {
328            return paste_external_body(s, &body);
329        }
330        return paste_plain(s, t);
331    }
332    let editing = s.doc()?.selection.text.is_some();
333    if editing {
334        if let Some(body) = s.clipboard.text.clone() {
335            return super::text::paste_body(s, &body);
336        }
337        let plain = s.clipboard.plain.clone();
338        return paste_plain(s, &plain);
339    }
340    let clip = s.clipboard.clone();
341    if !clip.slides.is_empty() {
342        let at = s.doc()?.selection.slide + 1;
343        return s.edit(|doc, sel| {
344            let media = bring_media("edit.paste", doc, &clip.media)?;
345            let mut new_ids = vec![];
346            for (k, sl) in clip.slides.iter().enumerate() {
347                let mut sl = sl.clone();
348                sl.id = SlideId(doc.alloc_id().ok_or_else(|| bad("edit.paste", NO_ID_SPACE))?);
349                let map = reid("edit.paste", doc, &mut sl.shapes)?;
350                remap_animations(&mut sl.animations, &map);
351                crate::engine::media_refs::visit_mut::slide(&mut sl, &mut remap_media(&media));
352                if doc.layout(sl.layout).is_none() {
353                    sl.layout = doc.masters.first().and_then(|m| m.layouts.first()).map(|l| l.id).unwrap_or_default();
354                }
355                new_ids.push(sl.id);
356                let i = (at + k).min(doc.slides.len());
357                doc.slides.insert(i, Arc::new(sl));
358            }
359            sel.slide = (at + new_ids.len()).saturating_sub(1).min(doc.slides.len().saturating_sub(1));
360            sel.slides = new_ids.clone();
361            sel.shapes.clear();
362            Ok(json!({"slides": new_ids.len()}))
363        });
364    }
365    if !clip.shapes.is_empty() {
366        s.clipboard.pastes += 1;
367        let off = s.prefs.duplicate_offset * s.clipboard.pastes as f64;
368        // Pasting on another slide keeps the position; on the same slide it offsets.
369        let same_slide = s.doc().ok().is_some_and(|d| clip.shapes.iter().any(|c| d.shape(c.id).is_some()));
370        let off = if same_slide { off } else { 0.0 };
371        return s.edit(|doc, sel| {
372            let media = bring_media("edit.paste", doc, &clip.media)?;
373            let mut shapes = clip.shapes.clone();
374            if sel.target != crate::engine::Target::Slides {
375                for sh in &mut shapes {
376                    if let Some(original) = clip.placeholders.iter().find(|p| p.id == sh.id) {
377                        sh.clone_from(original);
378                    }
379                }
380            }
381            reid("edit.paste", doc, &mut shapes)?;
382            crate::engine::media_refs::visit_mut::shapes(&mut shapes, &mut remap_media(&media));
383            let (mut taken, mut count) = super::current_names(doc, sel);
384            for sh in &mut shapes {
385                // Only a name already on the slide is changed (two shapes with one name are what the
386                // issue is about); otherwise it stays, as Morph pairs shapes across slides by name.
387                if taken.contains(&sh.name)
388                    && let Some((base, n)) = sh.name.rsplit_once(' ')
389                    && !n.is_empty()
390                    && n.bytes().all(|b| b.is_ascii_digit())
391                {
392                    sh.name = super::insert::first_free_name(base, &taken, count);
393                }
394                crate::model::walk(std::slice::from_ref(sh), &mut |m, _| {
395                    taken.insert(m.name.clone());
396                    count += 1;
397                });
398                if let Some(x) = sh.xfrm.as_mut() {
399                    x.x += off;
400                    x.y += off;
401                }
402                // A placeholder pasted on a slide becomes a regular shape. One copied from a slide
403                // already has its look; see `copy_inner`.
404                if sh.ph.is_some() && sel.target == crate::engine::Target::Slides {
405                    sh.ph = None;
406                }
407            }
408            let ids: Vec<ShapeId> = shapes.iter().map(|x| x.id).collect();
409            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("edit.paste", "no slide"))?;
410            list.extend(shapes);
411            sel.shapes = ids.clone();
412            sel.text = None;
413            Ok(json!({"ids": ids}))
414        });
415    }
416    if !clip.plain.is_empty() {
417        let t = clip.plain.clone();
418        return paste_plain(s, &t);
419    }
420    ok()
421}

423fn paste_external_body(s: &mut Session, body: &TextBody) -> Result<Value> {
424    if s.doc()?.selection.text.is_some() {
425        return super::text::paste_body(s, body);
426    }
427    let plain = body.text();
428    let st = s.doc()?;
429    let size = st.doc.slide_size;
430    let rect = [size.width * 0.25, size.height * 0.4, size.width * 0.5, 40.0];
431    let made = s.execute("insert.textBox", &json!({"rect": rect, "text": plain}))?;
432    let raw = made.get("id").and_then(Value::as_u64).ok_or_else(|| bad("edit.paste", "no text box"))?;
433    let id = u32::try_from(raw).map_err(|_| bad("edit.paste", "no text box"))?;
434    let body = body.clone();
435    s.edit(|doc, sel| {
436        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("edit.paste", "no slide"))?;
437        let sh = crate::model::find_shape_mut(list, ShapeId(id)).ok_or_else(|| bad("edit.paste", "no text box"))?;
438        if let Some(t) = sh.text.as_mut() {
439            let (wrap, autofit) = (t.body.wrap, t.body.autofit);
440            *t = body.clone();
441            t.body.wrap = wrap;
442            t.body.autofit = autofit;
443        }
444        sel.shapes = vec![ShapeId(id)];
445        sel.text = None;
446        Ok(json!({"id": id}))
447    })
448}

450fn paste_plain(s: &mut Session, text: &str) -> Result<Value> {
451    if s.doc()?.selection.text.is_some() {
452        return super::text::insert_text(s, text, None);
453    }
454    // No text box being edited: a new text box with the text.
455    let st = s.doc()?;
456    let size = st.doc.slide_size;
457    let rect = [size.width * 0.25, size.height * 0.4, size.width * 0.5, 40.0];
458    s.execute("insert.textBox", &json!({"rect": rect, "text": text}))
459}

461fn paste_special(s: &mut Session, p: &Value) -> Result<Value> {
462    match str_param(p, "as").unwrap_or("destinationTheme") {
463        "destinationTheme" | "useDestinationTheme" => paste(s, &json!({})),
464        "text" | "keepTextOnly" => paste_text(s, &json!({})),
465        "keepSource" | "keepSourceFormatting" => {
466            let saved = s.clipboard.clone();
467            if let Some(theme) = &saved.source_theme {
468                bake_theme(&mut s.clipboard.shapes, theme);
469                bake_theme(&mut s.clipboard.placeholders, theme);
470            }
471            let r = paste(s, &json!({}));
472            let pastes = s.clipboard.pastes;
473            s.clipboard = saved;
474            s.clipboard.pastes = pastes;
475            r
476        }
477        "picture" => paste_picture(s),
478        other => Err(bad("edit.pasteSpecial", format!("unknown `as`: {other} (destinationTheme, keepSource, picture, text)"))),
479    }
480}

482/// Replace theme colours and theme fonts with the values they have in `scheme` / `fonts`, so the
483/// shapes keep their look under another theme.
484pub(crate) fn bake_theme(shapes: &mut [Shape], theme: &crate::engine::SourceTheme) {
485    fn rec(v: &mut Value, theme: &crate::engine::SourceTheme, depth: usize) {
486        if depth > 256 {
487            return;
488        }
489        match v {
490            Value::Object(o) => {
491                if o.get("kind").and_then(Value::as_str) == Some("scheme")
492                    && let Some(slot) = o.get("slot").and_then(|x| serde_json::from_value::<crate::color::SchemeSlot>(x.clone()).ok())
493                    && slot != crate::color::SchemeSlot::PhClr
494                    && let Ok(rgb) = serde_json::to_value(theme.colors.get(crate::model::map_slot(&theme.color_map, slot)))
495                {
496                    o.clear();
497                    o.insert("kind".into(), json!("rgb"));
498                    o.insert("rgb".into(), rgb);
499                    return;
500                }
501                for x in o.values_mut() {
502                    rec(x, theme, depth + 1);
503                }
504            }
505            Value::Array(a) => a.iter_mut().for_each(|x| rec(x, theme, depth + 1)),
506            Value::String(t) if t.starts_with('+') => {
507                let set = match t.get(1..3) {
508                    Some("mj") => &theme.fonts.major,
509                    Some("mn") => &theme.fonts.minor,
510                    _ => return,
511                };
512                let face = match t.get(3..) {
513                    Some("-lt") => &set.latin,
514                    Some("-ea") => &set.ea,
515                    Some("-cs") => &set.cs,
516                    _ => return,
517                };
518                if !face.is_empty() {
519                    *t = face.clone();
520                }
521            }
522            _ => {}
523        }
524    }
525    for sh in shapes.iter_mut() {
526        let Ok(mut v) = serde_json::to_value(&*sh) else { continue };
527        rec(&mut v, theme, 0);
528        if let Ok(baked) = serde_json::from_value::<Shape>(v) {
529            *sh = baked;
530        }
531    }
532}

534/// Paste Special › Picture: the copied shapes drawn into one PNG picture.
535fn paste_picture(s: &mut Session) -> Result<Value> {
536    let mut clip = s.clipboard.clone();
537    if clip.shapes.is_empty() {
538        return Err(bad("edit.pasteSpecial", "copy shapes first to paste them as a picture"));
539    }
540    if let Some(theme) = &clip.source_theme {
541        bake_theme(&mut clip.shapes, theme);
542    }
543    let st = s.doc()?;
544    let mut doc = (*st.doc).clone();
545    let media = bring_media("edit.pasteSpecial", &mut doc, &clip.media)?;
546    crate::engine::media_refs::visit_mut::shapes(&mut clip.shapes, &mut remap_media(&media));
547    let mut slide = st.current_slide().cloned().ok_or_else(|| bad("edit.pasteSpecial", "no slide"))?;
548    let xs: Vec<crate::geom::Xfrm> = clip.shapes.iter().filter_map(|x| x.xfrm).collect();
549    let b = super::arrange::union_bounds(&xs).ok_or_else(|| bad("edit.pasteSpecial", "nothing to draw"))?;
550    let gx = crate::geom::Xfrm::new(b.x0, b.y0, b.width(), b.height());
551    let gid = ShapeId(u32::MAX - 1);
552    let one = if clip.shapes.len() == 1 {
553        let mut sh = clip.shapes[0].clone();
554        sh.id = gid;
555        sh
556    } else {
557        Shape { id: gid, xfrm: Some(gx), kind: crate::model::ShapeKind::Group { children: clip.shapes.clone(), child: gx }, ..Default::default() }
558    };
559    slide.shapes = vec![one];
560    const SCALE: f64 = 2.0;
561    let img = crate::render::render_shape(&doc, &slide, gid, SCALE).ok_or_else(|| bad("edit.pasteSpecial", "the shapes could not be drawn"))?;
562    // render_shape draws the bounds plus an 8 pt margin (room for shadows and outlines).
563    let (w, h) = (img.width as f64 / SCALE, img.height as f64 / SCALE);
564    let same_slide = clip.shapes.iter().any(|c| st.shape(c.id).is_some());
565    let off = if same_slide { s.prefs.duplicate_offset } else { 0.0 };
566    let png = img.to_png();
567    s.execute("insert.picture", &json!({"data": base64_encode(&png), "name": "Picture.png", "rect": [b.x0 - 8.0 + off, b.y0 - 8.0 + off, w, h]}))
568}

575/// Duplicate gives a copy the next automatic name ("Rectangle 4" after "Rectangle 1"), as
576/// PowerPoint does, and keeps a name the user chose. PowerPoint tells the two apart by comparing
577/// the number with the shape's id; ids here are numbered across the deck, so any "{kind} {n}"
578/// counts as automatic.
579fn rename_automatic(sh: &mut Shape, taken: &mut std::collections::HashSet<String>, count: usize) {
580    let base = if sh.is_group() { "Group".to_string() } else { super::insert::auto_name_base(sh) };
581    let n = sh.name.strip_prefix(&base).and_then(|r| r.strip_prefix(' '));
582    if n.is_some_and(|n| !n.is_empty() && n.bytes().all(|b| b.is_ascii_digit())) {
583        sh.name = super::insert::first_free_name(&base, taken, count);
584        taken.insert(sh.name.clone());
585    }
586}

588fn duplicate(s: &mut Session, p: &Value) -> Result<Value> {
589    let ids = targets(s, p)?;
590    let off = s.prefs.duplicate_offset;
591    let last = s.doc()?.last_duplicate.clone();
592    let (new_ids, made) = s.edit(|doc, sel| {
593        let list = super::current_shapes(doc, sel).cloned().unwrap_or_default();
594        let mut copies: Vec<Shape> = list.iter().filter(|x| ids.contains(&x.id)).cloned().collect();
595        // A shape inside a group is copied into that group, with the map from the group's space onto
596        // the slide so that its copy is offset on the slide as a top-level copy is.
597        let top = copies.len();
598        let mut to_slide = vec![None; top];
599        for id in grouped_targets(&list, &ids) {
600            if let Some((m, a)) = crate::engine::connect::locate(doc, sel, &list, id) {
601                copies.push(m.clone());
602                to_slide.push(Some(a));
603            }
604        }
605        let sources: Vec<ShapeId> = copies.iter().map(|c| c.id).collect();
606        let boxes: Vec<_> = copies.iter().map(|c| xfrm_of(doc, sel, c)).collect();
607        detach_on_slide(doc, sel, &mut copies);
608        // Duplicating the copies of the last Duplicate again repeats how far they were moved
609        // since (#966); anything else gets the usual offset.
610        let (dx, dy) = last
611            .filter(|(slide, target, pairs)| {
612                *slide == sel.slide && *target == sel.target && pairs.len() == sources.len() && pairs.iter().all(|(_, c)| sources.contains(c))
613            })
614            .and_then(|(_, _, pairs)| {
615                let (src, copy) = pairs.first()?;
616                // Measured on the slide, so the copy of a shape in a group repeats its drawn move.
617                let (a, to_a) = crate::engine::connect::locate(doc, sel, &list, *src)?;
618                let (b, to_b) = crate::engine::connect::locate(doc, sel, &list, *copy)?;
619                let (a, b) = (xfrm_of(doc, sel, a), xfrm_of(doc, sel, b));
620                let d = to_b * crate::geom::Point::new(b.x, b.y) - to_a * crate::geom::Point::new(a.x, a.y);
621                (d.x.is_finite() && d.y.is_finite()).then_some((d.x, d.y))
622            })
623            .unwrap_or((off, off));
624        reid("edit.duplicate", doc, &mut copies)?;
625        // Every name on the slide, group members included, so a copy never repeats one.
626        let mut taken = std::collections::HashSet::new();
627        let mut count = 0;
628        crate::model::walk(&list, &mut |sh, _| {
629            taken.insert(sh.name.clone());
630            count += 1;
631        });
632        walk_mut(&mut copies, &mut |c| {
633            rename_automatic(c, &mut taken, count);
634            count += 1;
635        });
636        for ((c, b), to) in copies.iter_mut().zip(boxes).zip(&to_slide) {
637            let (dx, dy) = to.map_or((dx, dy), |a| offset_in(a, dx, dy));
638            let mut x = b;
639            x.x += dx;
640            x.y += dy;
641            c.xfrm = Some(x);
642            if sel.target == crate::engine::Target::Slides {
643                c.ph = None;
644            }
645        }
646        let new_ids: Vec<ShapeId> = copies.iter().map(|c| c.id).collect();
647        let shapes = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("edit.duplicate", "no slide"))?;
648        let mut copies = copies.into_iter();
649        shapes.extend(copies.by_ref().take(top));
650        // A member's copy goes last among its group's children, as a top-level copy goes last on the slide.
651        for (c, src) in copies.zip(sources.iter().skip(top)) {
652            crate::model::sibling_list_mut(shapes, *src).ok_or_else(|| bad("edit.duplicate", "no such shape"))?.push(c);
653        }
654        sel.shapes = new_ids.clone();
655        let made = (sel.slide, sel.target, sources.into_iter().zip(new_ids.iter().copied()).collect());
656        Ok((new_ids, made))
657    })?;
658    s.doc_mut()?.last_duplicate = Some(made);
659    Ok(json!({"ids": new_ids}))
660}

662/// The shapes in `ids` that are inside groups, in drawing order, without those in a group that is
663/// in `ids` too: they are copied along with it.
664fn grouped_targets(list: &[Shape], ids: &[ShapeId]) -> Vec<ShapeId> {
665    fn rec(shapes: &[Shape], ids: &[ShapeId], nested: bool, out: &mut Vec<ShapeId>, depth: usize) {
666        if depth > 64 {
667            return;
668        }
669        for sh in shapes {
670            if !ids.contains(&sh.id) {
671                rec(sh.children(), ids, true, out, depth + 1);
672            } else if nested {
673                out.push(sh.id);
674            }
675        }
676    }
677    let mut out = vec![];
678    rec(list, ids, false, &mut out, 0);
679    out
680}

682/// The offset in the space `to_slide` maps onto the slide that moves a shape by (`dx`, `dy`) on
683/// the slide. A group with no width or height draws nothing, and the offset stays as it is.
684fn offset_in(to_slide: crate::geom::Affine, dx: f64, dy: f64) -> (f64, f64) {
685    let back = to_slide.inverse();
686    let v = back * crate::geom::Point::new(dx, dy) - back * crate::geom::Point::ZERO;
687    if v.x.is_finite() && v.y.is_finite() { (v.x, v.y) } else { (dx, dy) }
688}

690fn delete(s: &mut Session, p: &Value) -> Result<Value> {
691    if s.doc()?.selection.text.is_some() && ids_param(p, "ids").is_none() {
692        return super::text::delete_dir(s, "forward");
693    }
694    if slides_scope(s, p) && ids_param(p, "ids").is_none() {
695        return s.execute("slide.delete", &json!({}));
696    }
697    // With table cells selected, Delete clears their text and keeps the table, as in PowerPoint.
698    let sel = &s.doc()?.selection;
699    if ids_param(p, "ids").is_none() && p.get("id").is_none() && sel.cells.is_some_and(|c| sel.shapes == [c.0]) {
700        return super::table::clear_selected_cells(s);
701    }
702    let ids = targets(s, p)?;
703    if ids.is_empty() {
704        return ok();
705    }
706    s.edit(|doc, sel| {
707        let shapes = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("edit.delete", "no slide"))?;
708        let mut before = Vec::new();
709        crate::model::walk(shapes, &mut |s, _| before.push(s.id));
710        let n = remove_shapes(shapes, &ids);
711        // Everything that is gone: the given ids plus group descendants and emptied groups.
712        let mut left = Vec::new();
713        crate::model::walk(shapes, &mut |s, _| left.push(s.id));
714        let mut gone = ids.clone();
715        gone.extend(before.into_iter().filter(|i| !left.contains(i)));
716        // Animations of deleted shapes go too.
717        if sel.target == crate::engine::Target::Slides
718            && let Some(sl) = doc.slides.get_mut(sel.slide)
719        {
720            let sl = Arc::make_mut(sl);
721            sl.animations.retain(|a| !gone.contains(&a.shape));
722            // Effects triggered by a deleted shape fall back to the main sequence.
723            for a in sl.animations.iter_mut().filter(|a| a.trigger.is_some_and(|t| gone.contains(&t))) {
724                a.trigger = None;
725            }
726        }
727        sel.shapes.retain(|i| !ids.contains(i));
728        sel.text = None;
729        Ok(json!({"deleted": n}))
730    })
731}

733/// Remove shapes by id anywhere in the tree; empty groups go too.
734pub(crate) fn remove_shapes(shapes: &mut Vec<Shape>, ids: &[ShapeId]) -> usize {
735    let before = count(shapes);
736    fn rec(v: &mut Vec<Shape>, ids: &[ShapeId], depth: usize) {
737        v.retain(|s| !ids.contains(&s.id));
738        if depth > 64 {
739            return;
740        }
741        for s in v.iter_mut() {
742            if let Some(ch) = s.children_mut() {
743                rec(ch, ids, depth + 1);
744            }
745        }
746        v.retain(|s| !s.is_group() || !s.children().is_empty());
747    }
748    rec(shapes, ids, 0);
749    before.saturating_sub(count(shapes))
750}
751
752fn count(v: &[Shape]) -> usize {
753    let mut n = 0;
754    crate::model::walk(v, &mut |_, _| n += 1);
755    n
756}

758fn select_all(s: &mut Session, p: &Value) -> Result<Value> {
759    match p.get("scope") {
760        None | Some(Value::Null) => {}
761        Some(Value::String(v)) if v == "slides" => return select_all_slides(s),
762        Some(_) => return Err(bad("edit.selectAll", "scope must be \"slides\"")),
763    }
764    if s.doc()?.selection.text.is_some() {
765        return s.execute("text.selectAll", &json!({}));
766    }
767    s.select(|_, sel| {
768        sel.text = None;
769    })?;
770    let ids: Vec<ShapeId> = s.doc()?.shapes().iter().filter(|x| !x.hidden && !x.locked).map(|x| x.id).collect();
771    s.select(|_, sel| sel.shapes = ids.clone())?;
772    Ok(json!({"ids": ids}))
773}

775/// Thumbnail-pane / sorter select all: every slide, hidden ones included. The current
776/// slide stays where it is (the canvas must not jump), so this is not `slide.selectSlides`.
777fn select_all_slides(s: &mut Session) -> Result<Value> {
778    if s.doc()?.selection.target != crate::engine::Target::Slides {
779        return Err(bad("edit.selectAll", "the slide list is not shown while editing a master"));
780    }
781    let ids: Vec<SlideId> = s.doc()?.doc.slides.iter().map(|x| x.id).collect();
782    s.select(|_, sel| {
783        sel.slides = ids.clone();
784        sel.shapes.clear();
785        sel.text = None;
786        sel.cells = None;
787    })?;
788    Ok(json!({"selected": ids}))
789}

791fn deselect(s: &mut Session, _p: &Value) -> Result<Value> {
792    s.painter = None;
793    s.anim_painter = None;
794    s.select(|_, sel| {
795        if let Some(t) = sel.text.take() {
796            // Esc while editing text selects the shape.
797            if !t.notes {
798                sel.shapes = vec![t.shape];
799            }
800        } else {
801            sel.shapes.clear();
802            sel.cells = None;
803        }
804    })?;
805    ok()
806}

808fn select(s: &mut Session, p: &Value) -> Result<Value> {
809    let ids = match ids_param(p, "ids") {
810        Some(v) => v,
811        None => id_param(p, "id", "edit.select")?.map(|i| vec![i]).unwrap_or_default(),
812    };
813    let add = bool_or(p, "add", false);
814    let toggle = bool_or(p, "toggle", false);
815    let valid: Vec<ShapeId> = ids.into_iter().filter(|i| s.active().is_some_and(|d| d.shape(*i).is_some())).collect();
816    s.select(|_, sel| {
817        sel.text = None;
818        sel.cells = None;
819        if toggle {
820            for i in &valid {
821                if let Some(k) = sel.shapes.iter().position(|x| x == i) {
822                    sel.shapes.remove(k);
823                } else {
824                    sel.shapes.push(*i);
825                }
826            }
827        } else if add {
828            for i in &valid {
829                if !sel.shapes.contains(i) {
830                    sel.shapes.push(*i);
831                }
832            }
833        } else {
834            sel.shapes = valid.clone();
835        }
836    })?;
837    Ok(json!({"selected": s.doc()?.selection.shapes}))
