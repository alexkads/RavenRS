154/// A table as drawn on the slide: where it is, its drawn row heights and its visible cells.
155#[derive(Clone, Debug)]
156pub struct TableFrame {
157    /// Table-local (points from the table's top-left corner) → slide space, as the table is drawn.
158    pub affine: Affine,
159    pub cols: Vec<f64>,
160    /// Row heights as drawn: a stored height is a minimum, rows grow to fit their text.
161    pub heights: Vec<f64>,
162    /// Visible cells (merges honoured): (row, col, table-local rect).
163    pub cells: Vec<(usize, usize, Rect)>,
164}
165
166/// A border between table rows or columns that can be dragged.
167#[derive(Clone, Copy, Debug, PartialEq, Eq)]
168pub enum TableBorder {
169    /// The right edge of this column (the last one is the table's right edge).
170    Column(usize),
171    /// The bottom edge of this row (the last one is the table's bottom edge).
172    Row(usize),
173}
174
175/// The drawn geometry of table shape `sh` on the edited slide (`None` if it isn't a table).
176pub fn table_frame(st: &DocState, sh: &Shape) -> Option<TableFrame> {
177    let ShapeKind::Table(t) = &sh.kind else { return None };
178    // The table is drawn in its own box; rows that grow past it don't move it.
179    let x = sh.xfrm.unwrap_or_else(|| xfrm_of(&st.doc, &st.selection, sh));
180    let heights = table_drawn_heights(&st.doc, &st.selection, t);
181    let cells = crate::render::table_cell_rects(t, &heights);
182    Some(TableFrame { affine: x.affine(), cols: t.cols.clone(), heights, cells })
183}
184
185impl TableFrame {
186    pub fn width(&self) -> f64 {
187        self.cols.iter().map(|w| w.max(0.0)).sum()
188    }
189
190    pub fn height(&self) -> f64 {
191        self.heights.iter().map(|h| h.max(0.0)).sum()
192    }
193
194    /// Table-local point for slide point `p`.
195    pub fn local(&self, p: Point) -> Point {
196        self.affine.inverse() * p
197    }
198
199    /// Table-local rect of visible cell (`r`, `c`).
200    pub fn cell_rect(&self, r: usize, c: usize) -> Option<Rect> {
201        self.cells.iter().find(|(cr, cc, _)| (*cr, *cc) == (r, c)).map(|x| x.2)
202    }
203
204    /// The visible cell at table-local point `p` (points outside the table go to the nearest cell).
205    pub fn cell_at(&self, p: Point) -> Option<(usize, usize)> {
206        let q = Point::new(p.x.clamp(0.0, self.width().max(0.0)), p.y.clamp(0.0, self.height().max(0.0)));
207        let inside = |r: &Rect| q.x >= r.x0 && q.x <= r.x1 && q.y >= r.y0 && q.y <= r.y1;
208        // Edges are shared: prefer the cell the point is strictly inside.
209        self.cells
210            .iter()
211            .find(|(_, _, r)| q.x >= r.x0 && q.x < r.x1 && q.y >= r.y0 && q.y < r.y1)
212            .or_else(|| self.cells.iter().rev().find(|(_, _, r)| inside(r)))
213            .map(|(r, c, _)| (*r, *c))
214    }
215
216    /// The draggable border within `tol` of table-local point `p`: the inner borders and the right
217    /// and bottom edges (the left and top edges move the table). Borders hidden inside merged
218    /// cells don't count.
219    pub fn border_at(&self, p: Point, tol: f64) -> Option<TableBorder> {
220        let (w, h) = (self.width(), self.height());
221        if !(p.x >= -tol && p.y >= -tol && p.x <= w + tol && p.y <= h + tol) {
222            return None;
223        }
224        let q = Point::new(p.x.clamp(0.0, w), p.y.clamp(0.0, h));
225        let mut best: Option<(f64, TableBorder)> = None;
226        let mut x = 0.0;
227        for (i, cw) in self.cols.iter().enumerate() {
228            x += cw.max(0.0);
229            let d = (p.x - x).abs();
230            // A cell ending at this edge at this height shows the border.
231            let shown = self.cells.iter().any(|(_, _, r)| (r.x1 - x).abs() < 1e-6 && q.y >= r.y0 && q.y <= r.y1);
232            if d <= tol && shown && best.is_none_or(|b| d < b.0) {
233                best = Some((d, TableBorder::Column(i)));
234            }
235        }
236        let mut y = 0.0;
237        for (i, rh) in self.heights.iter().enumerate() {
238            y += rh.max(0.0);
239            let d = (p.y - y).abs();
240            let shown = self.cells.iter().any(|(_, _, r)| (r.y1 - y).abs() < 1e-6 && q.x >= r.x0 && q.x <= r.x1);
241            if d <= tol && shown && best.is_none_or(|b| d < b.0) {
242                best = Some((d, TableBorder::Row(i)));
243            }
244        }
245        best.map(|b| b.1)
246    }
247}

249/// A `[r, c]` cell param: `Ok(None)` when absent (callers may fall back to the selection), `Err` when
250/// present but not a valid cell, so a bad cell never acts on the selection.
251fn cell_param(p: &Value, key: &str, cmd: &str) -> Result<Option<(usize, usize)>> {
252    let v = match p.get(key) {
253        None | Some(Value::Null) => return Ok(None),
254        Some(v) => v,
255    };
256    // Exactly [row, col]: an entry with more is as malformed as one with less.
257    let idx = |x: &Value| x.as_u64().and_then(|x| usize::try_from(x).ok());
258    match v.as_array().map(Vec::as_slice) {
259        Some([r, c]) if let (Some(r), Some(c)) = (idx(r), idx(c)) => Ok(Some((r, c))),
260        _ => Err(bad(cmd, format!("`{key}` is not a valid cell: {v}"))),
261    }
262}

264/// The table's row/col to act on: params, the edited cell, or the selected cells.
265/// A present but invalid `row`/`col` is an error; only an absent one falls back.
266fn at(s: &Session, p: &Value, cmd: &str) -> Result<(usize, usize)> {
267    let st = s.active();
268    let cell = st.and_then(|d| d.selection.text.as_ref().and_then(|t| t.cell)).or_else(|| st.and_then(|d| d.selection.cells.map(|c| c.1)));
269    Ok((opt_usize_param(p, "row", cmd)?.or(cell.map(|c| c.0)).unwrap_or(0), opt_usize_param(p, "col", cmd)?.or(cell.map(|c| c.1)).unwrap_or(0)))
270}

272fn with_table(s: &mut Session, p: &Value, cmd: &str, f: impl Fn(&mut Table) -> Result<()>) -> Result<Value> {
273    let id = match id_param(p, "id", cmd)? {
274        Some(i) => i,
275        None => {
276            let st = s.doc()?;
277            st.selection
278                .text
279                .as_ref()
280                .filter(|t| t.cell.is_some())
281                .map(|t| t.shape)
282                .or(st.selection.cells.map(|c| c.0))
283                .or_else(|| st.selection.shapes.first().copied())
284                .ok_or_else(|| bad(cmd, "select a table"))?
285        }
286    };
287    let r = edit_shapes(s, &json!({"id": id}), cmd, |sh| match &mut sh.kind {
288        ShapeKind::Table(t) => {
289            f(t)?;
290            t.normalize();
291            let (w, h) = (t.width(), t.height());
292            if let Some(x) = sh.xfrm.as_mut() {
293                x.w = w;
294                x.h = h;
295            }
296            Ok(())
297        }
298        _ => Err(bad(cmd, "not a table")),
299    })?;
300    // The edited cell may have moved out of range.
301    s.select(|doc, sel| {
302        let dims = super::current_shapes(doc, sel)
303            .and_then(|v| crate::model::find_shape(v, id))
304            .and_then(|x| if let ShapeKind::Table(t) = &x.kind { Some((t.n_rows(), t.n_cols())) } else { None });
305        if let (Some(t), Some((nr, nc))) = (sel.text.as_mut(), dims)
306            && let Some((r, c)) = t.cell
307            && (r >= nr || c >= nc)
308        {
309            t.cell = Some((r.min(nr.saturating_sub(1)), c.min(nc.saturating_sub(1))));
310            t.anchor = (0, 0);
311            t.caret = (0, 0);
312        }
313    })?;
314    Ok(r)
315}

317fn rows_cols(s: &mut Session, p: &Value, op: &str, name: &'static str) -> Result<Value> {
318    let (r, c) = at(s, p, name)?;
319    let op = op.to_string();
320    with_table(s, p, name, move |t| {
321        match op.as_str() {
322            "rowAbove" => t.insert_row(r),
323            "rowBelow" => t.insert_row(r.saturating_add(1)),
324            "colLeft" => t.insert_col(c),
325            "colRight" => t.insert_col(c.saturating_add(1)),
326            "delRow" => {
327                if !t.delete_row(r) {
328                    return Err(bad(name, "can't delete the last row"));
329                }
330            }
331            _ => {
332                if !t.delete_col(c) {
333                    return Err(bad(name, "can't delete the last column"));
334                }
335            }
336        }
337        Ok(())
338    })
339}

341fn merge(s: &mut Session, p: &Value) -> Result<Value> {
342    let sel = s.doc()?.selection.cells;
343    let from = cell_param(p, "from", "table.merge")?.or(sel.map(|c| c.1)).ok_or_else(|| bad("table.merge", "missing `from`"))?;
344    let to = cell_param(p, "to", "table.merge")?.or(sel.map(|c| c.2)).ok_or_else(|| bad("table.merge", "missing `to`"))?;
345    with_table(
346        s,
347        p,
348        "table.merge",
349        move |t| if t.merge(from.0, from.1, to.0, to.1) { Ok(()) } else { Err(bad("table.merge", "can't merge those cells")) },
350    )
351}

353fn split(s: &mut Session, p: &Value) -> Result<Value> {
354    let cell = match cell_param(p, "cell", "table.split")? {
355        Some(c) => c,
356        None => at(s, p, "table.split")?,
357    };
358    with_table(s, p, "table.split", move |t| if t.split(cell.0, cell.1) { Ok(()) } else { Err(bad("table.split", "that cell isn't merged")) })
359}

393/// At most this many `cells` in one table cell command.
394const MAX_CELLS: usize = 100_000;
395
396/// The cells a command acts on: `cells`, else the selected cell range, else the cell holding the
397/// caret, else every cell.
398fn target_cells(s: &Session, p: &Value, cmd: &str) -> Result<impl Fn(&Table) -> Result<Vec<(usize, usize)>> + use<>> {
399    // `cells` is a list of [row, col]: anything else, or an entry that is not one, is refused, never
400    // dropped or read as absent (which would act on the selection or the whole table).
401    let cells: Option<Vec<(usize, usize)>> = match p.get("cells") {
402        None | Some(Value::Null) => None,
403        Some(Value::Array(a)) if a.len() > MAX_CELLS => {
404            return Err(bad(cmd, format!("`cells` lists {} cells; at most {MAX_CELLS}", a.len())));
405        }
406        Some(Value::Array(a)) => Some(
407            a.iter()
408                .map(|c| {
409                    cell_param(&json!({ "cell": c }), "cell", cmd)?.ok_or_else(|| bad(cmd, format!("`cells` entry {c} is not a [row, col] cell")))
410                })
411                .collect::<Result<_>>()?,
412        ),
413        Some(v) => return Err(bad(cmd, format!("`cells` is a list of [row, col] cells, not {v}"))),
414    };
415    let st = s.doc()?;
416    // A selected range grows over the merged cells it touches; with only a caret, the caret's cell is used.
417    let sel = st.selection.cells.map(|(_, a, b)| (a, b, true)).or_else(|| st.selection.text.as_ref().and_then(|t| t.cell).map(|c| (c, c, false)));
418    let cmd = cmd.to_string();
419    Ok(move |t: &Table| match (&cells, sel) {
420        // Each listed cell must be in the table, so a command never reports success for a cell
421        // it could not edit.
422        (Some(c), _) => match c.iter().find(|(r, c)| *r >= t.n_rows() || *c >= t.n_cols()) {
423            Some((r, c)) => Err(bad(&cmd, format!("no cell [{r}, {c}]: the table has {} rows and {} columns", t.n_rows(), t.n_cols()))),
424            None => Ok(c.clone()),
425        },
426        (None, Some((a, b, range))) => {
427            let (a, b) = if range { cell_block(t, a, b) } else { (a, b) };
428            // The selection can name cells outside the table; only walk the ones that exist.
429            let (r1, c1) = (a.0.max(b.0).min(t.n_rows().saturating_sub(1)), a.1.max(b.1).min(t.n_cols().saturating_sub(1)));
430            Ok((a.0.min(b.0)..=r1).flat_map(|r| (a.1.min(b.1)..=c1).map(move |c| (r, c))).collect())
431        }
432        _ => Ok((0..t.n_rows()).flat_map(|r| (0..t.n_cols()).map(move |c| (r, c))).collect()),
433    })
434}

436fn cell_fill(s: &mut Session, p: &Value) -> Result<Value> {
437    let fill = if bool_or(p, "none", false) { Some(crate::model::Fill::None) } else { color_param(p, "color").map(crate::model::Fill::solid) };
438    let targets = target_cells(s, p, "table.cellFill")?;
439    with_table(s, p, "table.cellFill", move |t| {
440        for (r, c) in targets(t)? {
441            if let Some(cell) = t.cell_mut(r, c) {
442                cell.fill.clone_from(&fill);
443            }
444        }
445        Ok(())
446    })
447}

454fn borders(s: &mut Session, p: &Value) -> Result<Value> {
455    use crate::model::Line;
456    let which = str_param(p, "which").unwrap_or("all").to_string();
457    const WHICH: [&str; 12] = ["all", "outside", "inside", "insideH", "insideV", "top", "bottom", "left", "right", "diagDown", "diagUp", "none"];
458    if !WHICH.contains(&which.as_str()) {
459        return Err(bad("table.borders", format!("unknown `which` {which:?}; one of {}", WHICH.join(", "))));
460    }
461    let color = color_param(p, "color").unwrap_or_else(|| crate::model::ColorRef::rgb(crate::color::Rgba::rgb(0, 0, 0)));
462    let width = f64_param(p, "width").filter(|w| w.is_finite()).unwrap_or(1.0).clamp(0.0, 100.0);
463    let mut line = Line::solid(color, width);
464    if let Some(d) = str_param(p, "dash").filter(|d| *d != "solid") {
465        if !crate::model::style::Dash::PRESETS.contains(&d) {
466            return Err(bad("table.borders", format!("unknown dash {d:?}; one of {}", crate::model::style::Dash::PRESETS.join(", "))));
467        }
468        line.dash = Some(crate::model::style::Dash::Preset { name: d.to_string() });
469    }
470    // "none" clears to no line, so the table style's own lines don't show through either.
471    let line = if which == "none" { Line::none() } else { line };
472    let targets = target_cells(s, p, "table.borders")?;
473    with_table(s, p, "table.borders", move |t| {
474        let list = targets(t)?;
475        let (Some(r0), Some(r1), Some(c0), Some(c1)) =
476            (list.iter().map(|x| x.0).min(), list.iter().map(|x| x.0).max(), list.iter().map(|x| x.1).min(), list.iter().map(|x| x.1).max())
477        else {
478            return Ok(());
479        };
480        let (rows, cols) = (t.n_rows(), t.n_cols());
481        // Set one side of a cell and the facing side of its neighbour, as PowerPoint does.
482        let set = |t: &mut Table, r: usize, c: usize, side: usize| {
483            if let Some(cell) = t.cell_mut(r, c) {
484                cell.borders[side] = Some(line.clone());
485            }
486            let n = match side {
487                LEFT => c.checked_sub(1).map(|c| (r, c, RIGHT)),
488                RIGHT => c.checked_add(1).filter(|c| *c < cols).map(|c| (r, c, LEFT)),
489                TOP => r.checked_sub(1).map(|r| (r, c, BOTTOM)),
490                _ => r.checked_add(1).filter(|r| *r < rows).map(|r| (r, c, TOP)),
491            };
492            if let Some((r, c, side)) = n
493                && let Some(cell) = t.cell_mut(r, c)
494            {
495                cell.borders[side] = Some(line.clone());
496            }
497        };
498        for (r, c) in list {
499            if r < r0 || r > r1 || c < c0 || c > c1 {
500                continue;
501            }
502            let outer = [c == c0, c == c1, r == r0, r == r1];
503            for side in [LEFT, RIGHT, TOP, BOTTOM] {
504                let horizontal = side == TOP || side == BOTTOM;
505                let on = match which.as_str() {
506                    "all" | "none" => true,
507                    "outside" => outer[side],
508                    "inside" => !outer[side],
509                    "insideH" => horizontal && !outer[side],
510                    "insideV" => !horizontal && !outer[side],
511                    "top" => side == TOP && outer[side],
512                    "bottom" => side == BOTTOM && outer[side],
513                    "left" => side == LEFT && outer[side],
514                    "right" => side == RIGHT && outer[side],
515                    _ => false,
516                };
517                if on {
518                    set(t, r, c, side);
519                }
520            }
521            if let Some(cell) = t.cell_mut(r, c) {
522                match which.as_str() {
523                    "diagDown" => cell.diag_down = Some(line.clone()),
524                    "diagUp" => cell.diag_up = Some(line.clone()),
525                    "none" => {
526                        cell.diag_down = None;
527                        cell.diag_up = None;
528                    }
529                    _ => {}
530                }
531            }
532        }
533        Ok(())
534    })
535}

537fn cell_margins(s: &mut Session, p: &Value) -> Result<Value> {
538    // PowerPoint's presets (left/right, top/bottom).
539    let base = match str_param(p, "preset") {
540        Some("none") => [0.0; 4],
541        Some("narrow") => [3.6; 4],
542        Some("wide") => [14.4, 14.4, 14.4, 14.4],
543        Some("normal") | None => [7.2, 7.2, 3.6, 3.6],
544        Some(o) => return Err(bad("table.cellMargins", format!("unknown preset {o:?}; one of normal, none, narrow, wide"))),
545    };
546    let side = |k: &str, i: usize| f64_param(p, k).filter(|v| v.is_finite()).map_or(base[i], |v| v.clamp(0.0, 400.0));
547    let m = [side("left", 0), side("right", 1), side("top", 2), side("bottom", 3)];
548    let targets = target_cells(s, p, "table.cellMargins")?;
549    with_table(s, p, "table.cellMargins", move |t| {
550        for (r, c) in targets(t)? {
551            if let Some(cell) = t.cell_mut(r, c) {
552                cell.margins = Some(m);
553            }
554        }
555        Ok(())
556    })
557}

559fn cell_align(s: &mut Session, p: &Value) -> Result<Value> {
560    use crate::model::text::{Align, Anchor};
561    // A value that is not a string is refused, not taken as absent.
562    for k in ["align", "anchor"] {
563        if p.get(k).is_some_and(|v| !v.is_null() && !v.is_string()) {
564            return Err(bad("table.cellAlign", format!("`{k}` is a name such as {}", if k == "align" { "left or center" } else { "top or middle" })));
565        }
566    }
567    // Without `align`, the vertical anchor is the whole command (top when not given).
568    let align = match str_param(p, "align") {
569        None => None,
570        Some("left" | "l") => Some(Align::Left),
571        Some("center" | "centre" | "ctr") => Some(Align::Center),
572        Some("right" | "r") => Some(Align::Right),
573        Some("justify" | "just") => Some(Align::Justify),
574        Some(o) => return Err(bad("table.cellAlign", format!("unknown align {o:?}; one of left, center, right, justify"))),
575    };
576    let anchor = match (str_param(p, "anchor"), align) {
577        (None, Some(_)) => None,
578        (a, _) => Some(match a.unwrap_or("top") {
579            "middle" | "center" | "ctr" => Anchor::Middle,
580            "bottom" | "b" => Anchor::Bottom,
581            "top" | "t" => Anchor::Top,
582            o => return Err(bad("table.cellAlign", format!("unknown anchor {o:?}; one of top, middle, bottom"))),
583        }),
584    };
585    let targets = target_cells(s, p, "table.cellAlign")?;
586    with_table(s, p, "table.cellAlign", move |t| {
587        for (r, c) in targets(t)? {
588            if let Some(cell) = t.cell_mut(r, c) {
589                if anchor.is_some() {
590                    cell.anchor = anchor;
591                }
592                if let Some(a) = align {
593                    for para in &mut cell.text.paragraphs {
594                        para.props.align = Some(a);
595                    }
596                }
597            }
598        }
599        Ok(())
600    })
601}

603fn distribute(s: &mut Session, p: &Value, rows: bool) -> Result<Value> {
604    with_table(s, p, "table.distribute", move |t| {
605        if rows {
606            let h = t.height() / t.n_rows().max(1) as f64;
607            t.rows.iter_mut().for_each(|r| r.height = h);
608        } else {
609            let w = t.width() / t.n_cols().max(1) as f64;
610            t.cols.iter_mut().for_each(|c| *c = w);
611        }
612        Ok(())
613    })
614}

616fn column_width(s: &mut Session, p: &Value) -> Result<Value> {
617    let (_, c) = at(s, p, "table.columnWidth")?;
618    let w = f64_param(p, "width").ok_or_else(|| bad("table.columnWidth", "missing `width`"))?.clamp(4.0, 4000.0);
619    let keep_total = bool_or(p, "keepTotal", false);
620    with_table(s, p, "table.columnWidth", move |t| {
621        // Keeping the table's width, the column to the right gives or takes the difference (dragging
622        // an inner border). The last column has none: the table gets wider or narrower.
623        let pair = t.cols.get(c).zip(t.cols.get(c.saturating_add(1))).map(|(a, b)| a.max(0.0) + b.max(0.0)).filter(|x| x.is_finite());
624        match pair {
625            Some(pair) if keep_total => {
626                let min = MIN_COLUMN_WIDTH.min(pair / 2.0);
627                let w = w.max(min).min(pair - min);
628                if let Some(x) = t.cols.get_mut(c) {
629                    *x = w;
630                }
631                if let Some(x) = t.cols.get_mut(c.saturating_add(1)) {
632                    *x = pair - w;
633                }
634            }
635            _ => {
636                if let Some(x) = t.cols.get_mut(c) {
637                    *x = w;
638                }
639            }
640        }
641        Ok(())
642    })
643}

671/// The cell range from `a` to `b` in `t`, grown until every merged cell it touches lies inside
672/// it, as (the corner on `a`'s side, the corner on `b`'s side). Cells past the table's edge are
673/// clamped to it.
674pub fn cell_block(t: &Table, a: (usize, usize), b: (usize, usize)) -> ((usize, usize), (usize, usize)) {
675    let (nr, nc) = (t.n_rows().max(1), t.n_cols().max(1));
676    let (a, b) = ((a.0.min(nr - 1), a.1.min(nc - 1)), (b.0.min(nr - 1), b.1.min(nc - 1)));
677    let (mut r0, mut r1, mut c0, mut c1) = (a.0.min(b.0), a.0.max(b.0), a.1.min(b.1), a.1.max(b.1));
678    // Each pass takes in the merged cells that overlap the block; stop when one adds nothing.
679    for _ in 0..nr + nc {
680        let mut grown = false;
681        for (r, row) in t.rows.iter().enumerate() {
682            for (c, cell) in row.cells.iter().enumerate() {
683                if cell.is_covered() {
684                    continue;
685                }
686                // Spans come from the file; the additions saturate so that they cannot overflow on 32-bit targets.
687                let end = |at: usize, span: u32, n: usize| at.saturating_add((span.max(1) - 1) as usize).min(n - 1);
688                let (er, ec) = (end(r, cell.row_span, nr), end(c, cell.grid_span, nc));
689                let overlaps = r <= r1 && er >= r0 && c <= c1 && ec >= c0;
690                if overlaps && (r < r0 || er > r1 || c < c0 || ec > c1) {
691                    (r0, r1, c0, c1) = (r0.min(r), r1.max(er), c0.min(c), c1.max(ec));
692                    grown = true;
693                }
694            }
695        }
696        if !grown {
697            break;
698        }
699    }
700    let pick = |lo: usize, hi: usize, first: bool| if first { lo } else { hi };
701    let from = (pick(r0, r1, a.0 <= b.0), pick(c0, c1, a.1 <= b.1));
702    let to = (pick(r0, r1, a.0 > b.0), pick(c0, c1, a.1 > b.1));
703    (from, to)
704}

706/// The selected cell range of table `id` (`t`), grown over merged cells, as its top-left and
707/// bottom-right cells. `None` when no range of that table is selected.
708pub fn selected_range(
709    cells: Option<(crate::model::ShapeId, (usize, usize), (usize, usize))>,
710    id: crate::model::ShapeId,
711    t: &Table,
712) -> Option<((usize, usize), (usize, usize))> {
713    let (_, a, b) = cells.filter(|c| c.0 == id)?;
714    let (x, y) = cell_block(t, a, b);
715    Some(((x.0.min(y.0), x.1.min(y.1)), (x.0.max(y.0), x.1.max(y.1))))
716}

718/// Whether cell (`r`, `c`) lies in `range` (from [`selected_range`]); without a range every cell does.
719pub fn in_range(range: Option<((usize, usize), (usize, usize))>, r: usize, c: usize) -> bool {
720    range.is_none_or(|((r0, c0), (r1, c1))| (r0..=r1).contains(&r) && (c0..=c1).contains(&c))
721}

723/// Shift+arrow with a cell range selected: the cell the range was extended to moves by (`dr`,
724/// `dc`), on past a merged cell until the range changes. The cell it started from stays. Returns
725/// false when no cell range of the selected table is selected.
726pub(crate) fn extend_selected_cells(s: &mut Session, dr: isize, dc: isize) -> Result<bool> {
727    let st = s.doc()?;
728    let Some((id, from, to)) = st.selection.cells.filter(|c| st.selection.shapes == [c.0]) else { return Ok(false) };
729    let Some(ShapeKind::Table(t)) = st.shape(id).map(|sh| &sh.kind) else { return Ok(false) };
730    let (nr, nc) = (t.n_rows(), t.n_cols());
731    let old = cell_block(t, from, to);
732    let mut end = to;
733    while cell_block(t, from, end) == old {
734        let r = end.0.checked_add_signed(dr).filter(|r| *r < nr);
735        let c = end.1.checked_add_signed(dc).filter(|c| *c < nc);
736        let (Some(r), Some(c)) = (r, c) else { break };
737        end = (r, c);
738    }
739    s.select(|_, sel| sel.cells = Some((id, from, end)))?;
740    Ok(true)
741}

743/// The text of the selected range of cells, for Copy and Cut: a tab between the cells of a row
744/// and a line per row (a merged cell once, where it starts). `None` without a cell range of the
745/// selected table.
746pub(crate) fn selected_cells_text(st: &DocState) -> Option<String> {
747    let id = st.selection.cells.filter(|c| st.selection.shapes == [c.0])?.0;
748    let ShapeKind::Table(t) = &st.shape(id)?.kind else { return None };
749    let ((r0, c0), (r1, c1)) = selected_range(st.selection.cells, id, t)?;
750    let rows = (r0..=r1).map(|r| {
751        let cells = (c0..=c1).filter_map(|c| t.cell(r, c).filter(|x| !x.is_covered()).map(|x| x.text.text()));
752        cells.collect::<Vec<_>>().join("\t")
753    });
754    Some(rows.collect::<Vec<_>>().join("\n"))
755}

757/// Delete with a cell range selected: the text in those cells goes and the cells stay, each
758/// keeping its first paragraph's formatting.
759pub(crate) fn clear_selected_cells(s: &mut Session) -> Result<Value> {
760    let Some((id, ..)) = s.doc()?.selection.cells else { return ok() };
761    let cells = s.doc()?.selection.cells;
762    with_table(s, &json!({"id": id}), "edit.delete", move |t| {
763        let range = selected_range(cells, id, t);
764        for (r, row) in t.rows.iter_mut().enumerate() {
765            for (c, cell) in row.cells.iter_mut().enumerate() {
766                if in_range(range, r, c) {
767                    crate::model::edit::delete(&mut cell.text, (0, 0), (usize::MAX, usize::MAX));
768                }
769            }
770        }
771        Ok(())
772    })
773}

775fn point_param(p: &Value, key: &str) -> Option<(f64, f64)> {
776    let a = p.get(key)?;
777    let (x, y) = (a.get(0)?.as_f64()?, a.get(1)?.as_f64()?);
778    (x.is_finite() && y.is_finite()).then_some((x, y))
779}

781/// The table under slide point `at` (frontmost), or the one named by `id`, with its origin.
782pub(crate) fn table_at(s: &Session, p: &Value, at: (f64, f64)) -> Option<(crate::model::ShapeId, (f64, f64))> {
783    let st = s.doc().ok()?;
784    // A malformed id names no table (never the one under the pointer instead).
785    let want = id_param(p, "id", "table.draw").ok()?;
786    st.shapes().iter().rev().find_map(|sh| {
787        let x = sh.xfrm?;
788        let ShapeKind::Table(t) = &sh.kind else { return None };
789        let inside = at.0 >= x.x && at.1 >= x.y && at.0 <= x.x + t.width() && at.1 <= x.y + t.height();
790        (want.map_or(inside, |w| w == sh.id)).then_some((sh.id, (x.x, x.y)))
791    })
792}

794fn draw_line(s: &mut Session, p: &Value) -> Result<Value> {
795    let from = point_param(p, "from").ok_or_else(|| bad("table.drawLine", "missing `from`"))?;
796    let to = point_param(p, "to").ok_or_else(|| bad("table.drawLine", "missing `to`"))?;
797    let (id, o) = table_at(s, p, from).ok_or_else(|| bad("table.drawLine", "no table there"))?;
798    let (a, b) = ((from.0 - o.0, from.1 - o.1), (to.0 - o.0, to.1 - o.1));
799    with_table(s, &json!({"id": id.0}), "table.drawLine", move |t| {
800        if crate::engine::drawtable::draw_line(t, a, b) { Ok(()) } else { Err(bad("table.drawLine", "the line doesn't divide any cell")) }
801    })
802}
