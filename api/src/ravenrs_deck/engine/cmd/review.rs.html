108/// Seconds since the Unix epoch: the system clock, or the browser's on wasm (which has no `SystemTime`).
109fn now_secs() -> u64 {
110    #[cfg(not(target_arch = "wasm32"))]
111    {
112        std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map(|d| d.as_secs()).unwrap_or(0)
113    }
114    #[cfg(target_arch = "wasm32")]
115    {
116        (js_sys::Date::now() / 1000.0).max(0.0) as u64
117    }
118}

120/// A UTC timestamp as saved on comments: `2026-10-10T13:10:35Z`.
121fn iso_utc(secs: u64) -> String {
122    let (y, m, d) = super::design::civil((secs / 86_400) as i64);
123    let t = secs % 86_400;
124    format!("{y:04}-{m:02}-{d:02}T{:02}:{:02}:{:02}Z", t / 3600, t / 60 % 60, t % 60)
125}
126
127fn now() -> String {
128    iso_utc(now_secs())
129}

131fn add(s: &mut Session, p: &Value) -> Result<Value> {
132    let text = str_param(p, "text").ok_or_else(|| bad("comment.add", "missing `text`"))?.to_string();
133    let author = s.prefs.author.clone();
134    let shape = id_param(p, "id", "comment.add")?.or_else(|| s.active().and_then(|d| d.selection.shapes.first().copied()));
135    let (x, y) = (f64_or(p, "x", 10.0), f64_or(p, "y", 10.0));
136    s.edit(|doc, sel| {
137        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("comment.add", "no slide"))?);
138        let initials: String = author.split_whitespace().filter_map(|w| w.chars().next()).collect();
139        sl.comments.push(Comment { author: author.clone(), initials, text, date: now(), x, y, resolved: false, replies: vec![], shape });
140        Ok(json!({"index": sl.comments.len() - 1}))
141    })
142}

144fn reply(s: &mut Session, p: &Value) -> Result<Value> {
145    let i = usize_param(p, "index").ok_or_else(|| bad("comment.reply", "missing `index`"))?;
146    let text = str_param(p, "text").unwrap_or("").to_string();
147    let author = s.prefs.author.clone();
148    s.edit(|doc, sel| {
149        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("comment.reply", "no slide"))?);
150        let c = sl.comments.get_mut(i).ok_or_else(|| bad("comment.reply", "no such comment"))?;
151        c.replies.push(Comment { author, text, date: now(), ..Default::default() });
152        Ok(())
153    })?;
154    ok()
155}

157fn delete(s: &mut Session, p: &Value) -> Result<Value> {
158    let all = bool_or(p, "all", false);
159    let i = match p.get("index") {
160        None | Some(Value::Null) => None,
161        Some(v) => Some(
162            v.as_u64()
163                .and_then(|v| usize::try_from(v).ok())
164                .ok_or_else(|| bad("comment.delete", format!("`index` is not a valid comment index: {v}")))?,
165        ),
166    };
167    s.edit(|doc, sel| {
168        if all {
169            for sl in &mut doc.slides {
170                Arc::make_mut(sl).comments.clear();
171            }
172            return Ok(());
173        }
174        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("comment.delete", "no slide"))?);
175        match i {
176            Some(i) if i < sl.comments.len() => {
177                sl.comments.remove(i);
178            }
179            None => sl.comments.clear(),
180            _ => return Err(bad("comment.delete", "no such comment")),
181        }
182        Ok(())
183    })?;
184    ok()
185}

204/// Flat `(slide, comment index)` list in deck order.
205fn comment_threads(doc: &crate::model::Presentation) -> Vec<(usize, usize)> {
206    let mut out = Vec::new();
207    for (si, sl) in doc.slides.iter().enumerate() {
208        for ci in 0..sl.comments.len() {
209            out.push((si, ci));
210        }
211    }
212    out
213}

215/// Step to the next or previous comment. With `{slide, index}` matching a thread, move one step
216/// and stop at the ends. Without `index`, next is the first thread on or after `slide` (the
217/// current slide when omitted) and previous is the last thread strictly before it.
218fn neighbour(s: &mut Session, p: &Value, forward: bool) -> Result<Value> {
219    let cmd = if forward { "comment.next" } else { "comment.prev" };
220    let (threads, slide, index) = {
221        let st = s.doc()?;
222        (comment_threads(&st.doc), usize_param(p, "slide").unwrap_or(st.selection.slide), super::opt_usize_param(p, "index", cmd)?)
223    };
224    if threads.is_empty() {
225        return Err(bad(cmd, "no comments"));
226    }
227    let pos = index.and_then(|i| threads.iter().position(|&(sl, c)| sl == slide && c == i));
228    let target = match (pos, forward) {
229        (Some(i), true) => threads.get(i + 1).copied(),
230        (Some(i), false) => i.checked_sub(1).and_then(|j| threads.get(j).copied()),
231        (None, true) => threads.iter().copied().find(|(sl, _)| *sl >= slide),
232        (None, false) => threads.iter().copied().rev().find(|(sl, _)| *sl < slide),
233    };
234    let Some((slide, index)) = target else {
235        return Err(bad(cmd, "no more comments"));
236    };
237    s.select(|doc, sel| {
238        sel.slide = slide;
239        sel.slides = doc.slides.get(slide).map(|x| vec![x.id]).unwrap_or_default();
240        sel.shapes.clear();
241        sel.text = None;
242        sel.cells = None;
243        sel.target = Target::Slides;
244    })?;
245    let text = s.doc()?.doc.slides.get(slide).and_then(|sl| sl.comments.get(index)).map(|c| c.text.clone()).unwrap_or_default();
246    Ok(json!({"slide": slide, "index": index, "text": text}))
247}

249fn accessibility(s: &mut Session, _p: &Value) -> Result<Value> {
250    let st = s.doc()?;
251    let mut out = vec![];
252    for (i, sl) in st.doc.slides.iter().enumerate() {
253        if sl.title().trim().is_empty() {
254            out.push(json!({"slide": i, "issue": "Missing slide title"}));
255        }
256        crate::model::walk(&sl.shapes, &mut |sh, _| {
257            let needs_alt = matches!(
258                sh.kind,
259                crate::model::ShapeKind::Picture { .. }
260                    | crate::model::ShapeKind::Chart(_)
261                    | crate::model::ShapeKind::Media(_)
262                    | crate::model::ShapeKind::Group { .. }
263            );
264            if needs_alt && sh.descr.trim().is_empty() && !sh.decorative {
265                out.push(json!({"slide": i, "shape": sh.id, "name": sh.name, "issue": "Missing alternative text"}));
266            }
267        });
268        // Low-contrast text against the slide background.
269        if let Some(ctx) = crate::model::resolve::Ctx::for_slide(&st.doc, sl) {
270            let bg = match crate::model::resolve::background(&ctx, Some(sl)) {
271                (crate::model::Fill::Solid { color }, ph) => Some(ctx.color(&color, ph)),
272                _ => None,
273            };
274            if let Some(bg) = bg {
275                crate::model::walk(&sl.shapes, &mut |sh, _| {
276                    let Some(t) = &sh.text else { return };
277                    let (fill, _) = crate::model::resolve::fill(&ctx, sh);
278                    if fill.as_ref().is_some_and(|f| !f.is_none()) {
279                        return;
280                    }
281                    let low = t
282                        .paragraphs
283                        .iter()
284                        .flat_map(|para| {
285                            // A line break or blank run shows no glyphs, so its colour can't be hard to read.
286                            para.runs.iter().filter(|run| run.kind != crate::model::text::RunKind::Break && !run.text.trim().is_empty()).map(|run| {
287                                let rp = crate::model::resolve::run(&ctx, sh, para, &run.props);
288                                crate::model::resolve::text_color(&ctx, &rp).contrast(bg)
289                            })
290                        })
291                        .find(|contrast| *contrast < 3.0);
292                    if let Some(contrast) = low {
293                        out.push(
294                            json!({"slide": i, "shape": sh.id, "name": sh.name, "issue": format!("Hard-to-read text contrast ({contrast:.1}:1)")}),
295                        );
296                    }
297                });
298            }
299        }
300    }
301    Ok(Value::Array(out))
302}

334/// The language a check uses: the `lang` parameter, else the first run that says, else US English.
335fn check_lang(doc: &crate::model::Presentation, p: &Value) -> String {
336    if let Some(l) = str_param(p, "lang") {
337        return l.to_string();
338    }
339    let mut found = None;
340    for sl in &doc.slides {
341        crate::model::walk(&sl.shapes, &mut |sh, _| {
342            if found.is_none()
343                && let Some(t) = &sh.text
344            {
345                found = t.paragraphs.iter().flat_map(|p| &p.runs).find_map(|r| r.props.lang.clone());
346            }
347        });
348    }
349    found.unwrap_or_else(|| "en-US".into())
350}

352fn spelling(s: &mut Session, p: &Value) -> Result<Value> {
353    let st = s.doc()?;
354    let default = check_lang(&st.doc, p);
355    let ignore: Vec<String> =
356        p.get("ignore").and_then(Value::as_array).map(|a| a.iter().filter_map(Value::as_str).map(str::to_lowercase).collect()).unwrap_or_default();
357    let mut dicts: std::collections::HashMap<String, Option<Arc<crate::engine::spell::Dictionary>>> = Default::default();
358    let mut out = vec![];
359    // Each misspelt word of `para`, reported with `at` (where the paragraph is) merged in.
360    let mut check = |para: &crate::model::Paragraph, at: Value, out: &mut Vec<Value>| {
361        for w in crate::engine::spell::repeated(para, &default) {
362            let mut v = json!({"start": w.start, "end": w.end, "word": w.text, "lang": w.lang, "repeated": true, "deleteFrom": w.repeated, "suggestions": []});
363            if let (Some(o), Some(a)) = (v.as_object_mut(), at.as_object()) {
364                o.extend(a.clone());
365            }
366            out.push(v);
367        }
368        let rep: std::collections::HashSet<usize> = crate::engine::spell::repeated(para, &default).iter().map(|w| w.start).collect();
369        for w in crate::engine::spell::paragraph_words(para, &default) {
370            if ignore.contains(&w.text.to_lowercase()) || rep.contains(&w.start) {
371                continue;
372            }
373            let dict = dicts.entry(w.lang.clone()).or_insert_with(|| crate::engine::spell::dictionary(&w.lang)).clone();
374            let bad = match &dict {
375                Some(d) => !crate::engine::spell::is_correct(d, &w.text),
376                None => COMMON.contains(&w.text.to_lowercase().as_str()),
377            };
378            if bad {
379                let sug = dict.as_ref().map(|d| d.suggest(&w.text, 5)).unwrap_or_default();
380                let mut v = json!({"start": w.start, "end": w.end, "word": w.text, "lang": w.lang, "suggestions": sug});
381                if let (Some(o), Some(a)) = (v.as_object_mut(), at.as_object()) {
382                    o.extend(a.clone());
383                }
384                out.push(v);
385            }
386        }
387    };
388    for (i, sl) in st.doc.slides.iter().enumerate() {
389        crate::model::walk(&sl.shapes, &mut |sh, _| {
390            if let Some(t) = &sh.text {
391                for (pi, para) in t.paragraphs.iter().enumerate() {
392                    check(para, json!({"slide": i, "shape": sh.id, "para": pi}), &mut out);
393                }
394            }
395            if let crate::model::ShapeKind::Table(tb) = &sh.kind {
396                for (r, row) in tb.rows.iter().enumerate() {
397                    for (c, cell) in row.cells.iter().enumerate().filter(|(_, c)| !c.is_covered()) {
398                        for (pi, para) in cell.text.paragraphs.iter().enumerate() {
399                            check(para, json!({"slide": i, "shape": sh.id, "row": r, "col": c, "para": pi}), &mut out);
400                        }
401                    }
402                }
403            }
404        });
405        // Speaker notes, a paragraph per line.
406        for (pi, para) in sl.notes.paragraphs.iter().enumerate() {
407            check(para, json!({"slide": i, "notes": true, "para": pi}), &mut out);
408        }
409    }
410    Ok(Value::Array(out))
411}

419/// Replace whole-word `from` with `to` in `text`; how many were replaced.
420fn replace_word(text: &mut String, from: &str, to: &str, all: bool) -> usize {
421    let is_word = |c: char| c.is_alphanumeric() || c == '\'' || c == '’';
422    // The result is built in one pass; the text is not edited in place for every match.
423    let mut out = String::new();
424    let (mut n, mut at, mut copied) = (0, 0, 0);
425    while let Some(i) = text.get(at..).and_then(|t| t.find(from)).map(|i| i + at) {
426        let end = i + from.len();
427        // The character before the match as the edited text would have it.
428        let before = text.get(copied..i).and_then(|t| t.chars().next_back()).or_else(|| out.chars().next_back()).is_some_and(is_word);
429        let after = text.get(end..).and_then(|t| t.chars().next()).is_some_and(is_word);
430        if before || after {
431            at = end;
432            continue;
433        }
434        out.push_str(text.get(copied..i).unwrap_or_default());
435        out.push_str(to);
436        n += 1;
437        (at, copied) = (end, end);
438        if !all {
439            break;
440        }
441    }
442    if n > 0 {
443        out.push_str(text.get(copied..).unwrap_or_default());
444        *text = out;
445    }
446    n
447}

449fn spelling_change(s: &mut Session, p: &Value) -> Result<Value> {
450    let word = str_param(p, "word").filter(|w| !w.is_empty()).ok_or_else(|| bad("review.spellingChange", "missing `word`"))?.to_string();
451    let to = str_param(p, "to").ok_or_else(|| bad("review.spellingChange", "missing `to`"))?.to_string();
452    let all = bool_or(p, "all", false);
453    let shape = p.get("shape").and_then(Value::as_u64);
454    if shape.is_none() && !all {
455        return Err(bad("review.spellingChange", "give `shape`, or `all: true`"));
456    }
457    // One exact occurrence (a squiggled word): paragraph `para`, from character `start`.
458    if let (Some(id), Some(para), Some(start)) = (shape, usize_param(p, "para"), usize_param(p, "start")) {
459        let id = crate::model::ShapeId(u32::try_from(id).map_err(|_| bad("review.spellingChange", "bad `shape`"))?);
460        let end = start.checked_add(word.chars().count()).ok_or_else(|| bad("review.spellingChange", "bad `start`"))?;
461        let cell = usize_param(p, "row").zip(usize_param(p, "col"));
462        let slide = {
463            let st = s.doc()?;
464            let found = st.doc.slides.iter().position(|sl| {
465                crate::model::find_shape(&sl.shapes, id)
466                    .and_then(|sh| match (cell, &sh.kind) {
467                        (Some((r, c)), crate::model::ShapeKind::Table(tb)) => tb.cell(r, c)?.text.paragraphs.get(para),
468                        (Some(_), _) => None,
469                        (None, _) => sh.text.as_ref()?.paragraphs.get(para),
470                    })
471                    .is_some_and(|pp| pp.text().chars().skip(start).take(end - start).collect::<String>() == word)
472            });
473            found.ok_or_else(|| bad("review.spellingChange", "the word isn't there"))?
474        };
475        s.edit(|doc, _| {
476            let sl = doc.slides.get_mut(slide).ok_or_else(|| bad("review.spellingChange", "no slide"))?;
477            let sh = crate::model::find_shape_mut(&mut Arc::make_mut(sl).shapes, id).ok_or_else(|| bad("review.spellingChange", "no shape"))?;
478            let pp = match (cell, &mut sh.kind) {
479                (Some((r, c)), crate::model::ShapeKind::Table(tb)) => tb.cell_mut(r, c).and_then(|cl| cl.text.paragraphs.get_mut(para)),
480                (Some(_), _) => None,
481                (None, _) => sh.text.as_mut().and_then(|t| t.paragraphs.get_mut(para)),
482            };
483            if let Some(pp) = pp {
484                replace_chars(pp, start, end, &to);
485            }
486            Ok(())
487        })?;
488        return Ok(json!({"changed": 1}));
489    }
490    // Nothing to change makes no undo step.
491    let mut any = false;
492    let st = s.doc()?;
493    for sl in &st.doc.slides {
494        crate::model::walk(&sl.shapes, &mut |sh, _| {
495            if shape.is_none_or(|id| serde_json::to_value(sh.id).ok().and_then(|v| v.as_u64()) == Some(id))
496                && let Some(t) = &sh.text
497            {
498                any |= t.paragraphs.iter().flat_map(|p| &p.runs).any(|r| replace_word(&mut r.text.clone(), &word, &to, false) > 0);
499            }
500        });
501    }
502    if !any {
503        return Ok(json!({"changed": 0}));
504    }
505    let changed = s.edit(|doc, _| {
506        let mut n = 0;
507        let id_of = |sh: &crate::model::Shape| serde_json::to_value(sh.id).ok().and_then(|v| v.as_u64());
508        for sl in &mut doc.slides {
509            let mut hit = false;
510            crate::model::walk(&sl.shapes, &mut |sh, _| {
511                hit |= shape.is_none_or(|id| id_of(sh) == Some(id)) && sh.text.as_ref().is_some_and(|t| t.text().contains(word.as_str()));
512            });
513            if !hit || (!all && n > 0) {
514                continue;
515            }
516            walk_text_mut(&mut Arc::make_mut(sl).shapes, 0, &mut |sh| {
517                if shape.is_some_and(|id| id_of(sh) != Some(id)) || (!all && n > 0) {
518                    return;
519                }
520                let Some(t) = sh.text.as_mut() else { return };
521                for r in t.paragraphs.iter_mut().flat_map(|p| p.runs.iter_mut()) {
522                    if !all && n > 0 {
523                        break;
524                    }
525                    n += replace_word(&mut r.text, &word, &to, all);
526                }
527            });
528        }
529        Ok(n)
530    })?;
531    Ok(json!({"changed": changed}))
532}

534/// Replace characters `start..end` of `p` with `to`, in the run where they start (later runs lose
535/// the rest of the range).
536fn replace_chars(p: &mut crate::model::Paragraph, start: usize, end: usize, to: &str) {
537    let mut at = 0;
538    let mut placed = false;
539    for r in &mut p.runs {
540        let n = r.char_len();
541        let (a, b) = (start.max(at), end.min(at + n));
542        if a < b && r.kind == crate::model::text::RunKind::Text {
543            let byte = |c: usize| r.text.char_indices().nth(c).map_or(r.text.len(), |(i, _)| i);
544            let (x, y) = (byte(a - at), byte(b - at));
545            r.text.replace_range(x..y, if placed { "" } else { to });
546            placed = true;
547        }
548        at += n;
549    }
550    p.runs.retain(|r| !(r.kind == crate::model::text::RunKind::Text && r.text.is_empty()));
551}
552
553fn walk_text_mut(shapes: &mut [crate::model::Shape], depth: usize, f: &mut dyn FnMut(&mut crate::model::Shape)) {
554    if depth > 64 {
555        return;
556    }
557    for sh in shapes.iter_mut() {
558        f(sh);
559        if let crate::model::ShapeKind::Group { children, .. } = &mut sh.kind {
560            walk_text_mut(children, depth + 1, f);
561        }
562    }
563}

617            if sh.custom_show.is_some() {
618                sh.range = None;
619            }
620        }
621        Ok(serde_json::to_value(&*sh).unwrap_or_default())
622    })

641/// The selected text when it lies in one paragraph, else the word at the caret.
642fn word_under_caret(s: &Session) -> Option<String> {
643    let st = s.doc().ok()?;
644    let t = st.selection.text.clone()?;
645    let body = super::text::body_of(st, &t)?;
646    let (a, b) = if t.anchor != t.caret { t.ordered() } else { crate::model::edit::word_at(body, t.caret) };
647    if a.0 != b.0 {
648        return None;
649    }
650    let text = body.paragraphs.get(a.0)?.text();
651    let w: String = text.chars().skip(a.1).take(b.1.saturating_sub(a.1)).collect();
652    let w = w.trim().to_string();
653    (!w.is_empty()).then_some(w)
654}

678fn revised_bytes(p: &Value) -> Result<(String, Vec<u8>)> {
679    if let Some(data) = str_param(p, "data") {
680        let bytes = base64_decode(data).ok_or_else(|| bad("review.compare", "invalid base64 `data`"))?;
681        return Ok((str_param(p, "name").unwrap_or("Revised").to_string(), bytes));
682    }
683    #[cfg(not(target_arch = "wasm32"))]
684    if let Some(path) = str_param(p, "path") {

686        return Ok((path.to_string(), bytes));
687    }
688    Err(bad("review.compare", "give `path` or `data`"))
689}

697fn compare(s: &mut Session, p: &Value) -> Result<Value> {
698    let (name, bytes) = revised_bytes(p)?;
699    let revised = super::file::open_presentation(&name, &bytes)?;
700    let st = s.doc_mut()?;
701    let by_position = crate::engine::compare::unrelated(&st.doc, &revised);
702    st.compare = Some(crate::engine::compare::Compare { name, revised: Arc::new(revised), by_position, rejected: Default::default() });
703    s.ui_requests.push(UiRequest::Pane { id: "revisions".into() });
704    revisions_json(s.doc()?)
705}

711fn accept(s: &mut Session, p: &Value) -> Result<Value> {
712    let cmp = s.doc()?.compare.clone().ok_or_else(|| bad("review.accept", "not comparing"))?;
713    let rev = cmp.revised.clone();
714    let all = bool_param(p, "all").unwrap_or(false);
715    let index = usize_param(p, "index");
716    if !all && index.is_none() {
717        return Err(bad("review.accept", "give `index` or `all`"));
718    }
719    s.edit(|doc, sel| {
720        let mut accepted = 0usize;
721        // Each accept shifts indices, so the list is recomputed; bounded by the slide counts.
722        let limit = doc.slides.len().saturating_add(rev.slides.len()).saturating_add(1);
723        for _ in 0..limit {
724            let changes = crate::engine::compare::pending(doc, &cmp);
725            let c = if all {
726                changes.into_iter().find(|c| !(c.kind == crate::engine::compare::ChangeKind::Deleted && doc.slides.len() <= 1))
727            } else {
728                changes.into_iter().nth(index.unwrap_or(0))
729            };
730            let Some(c) = c else { break };
731            let Some(at) = crate::engine::compare::accept(doc, &rev, &c)? else { break };
732            accepted += 1;
733            sel.slide = at;
734            sel.shapes.clear();
735            sel.text = None;
736            if !all {
737                break;
738            }
739        }
740        if accepted == 0 {
741            return Err(bad("review.accept", "no such revision"));
742        }
743        Ok(json!({"accepted": accepted}))
744    })
745}
746
747fn reject(s: &mut Session, p: &Value) -> Result<Value> {
748    let all = bool_param(p, "all").unwrap_or(false);
749    let index = usize_param(p, "index");
750    if !all && index.is_none() {
751        return Err(bad("review.reject", "give `index` or `all`"));
752    }
753    let st = s.doc_mut()?;
754    let doc = st.doc.clone();
755    let c = st.compare.as_mut().ok_or_else(|| bad("review.reject", "not comparing"))?;
756    let pending = crate::engine::compare::pending(&doc, c);
757    let picked: Vec<_> = if all { pending } else { pending.into_iter().nth(index.unwrap_or(0)).into_iter().collect() };
758    if picked.is_empty() {
759        return Err(bad("review.reject", "no such revision"));
760    }
761    c.rejected.extend(picked.iter().map(|ch| crate::engine::compare::ChangeKey::of(&doc, ch)));
762    Ok(json!({"rejected": picked.len()}))
763}

773    fn replace_word_in_place(text: &mut String, from: &str, to: &str, all: bool) -> usize {
774        let is_word = |c: char| c.is_alphanumeric() || c == '\'' || c == '’';
775        let mut n = 0;
776        let mut at = 0;
777        while let Some(i) = text.get(at..).and_then(|t| t.find(from)).map(|i| i + at) {
778            let end = i + from.len();
779            let before = text.get(..i).and_then(|t| t.chars().next_back()).is_some_and(is_word);
780            let after = text.get(end..).and_then(|t| t.chars().next()).is_some_and(is_word);
781            if before || after {
782                at = end;
783                continue;
784            }
785            text.replace_range(i..end, to);
786            n += 1;
787            at = i + to.len();
788            if !all {
789                break;
790            }
791        }
792        n
793    }

967mod accessibility_contrast_tests {

969    use crate::model::text::{Paragraph, Run, RunKind, RunProps};

971    use serde_json::json;
972
973    fn run(text: &str, color: Rgba) -> Run {
974        Run::with(text, RunProps { fill: Some(Fill::solid(ColorRef::rgb(color))), ..Default::default() })
975    }
976
977    fn text_shape(id: u32, name: &str, paragraphs: Vec<Paragraph>) -> Shape {
978        Shape { id: ShapeId(id), name: name.into(), text: Some(TextBody { paragraphs, ..Default::default() }), ..Default::default() }
979    }
980
981    #[test]
982    fn contrast_checks_mixed_runs_later_paragraphs_and_group_descendants_once() {
983        let mut s = Session::with_new();
984        s.edit(|doc, _| {
985            let black = Rgba::BLACK;
986            let white = Rgba::WHITE;
987            let mixed = Paragraph { runs: vec![run("visible", black), run("hidden", white)], ..Default::default() };
988            let later = vec![
989                Paragraph { runs: vec![run("visible", black)], ..Default::default() },
990                Paragraph { runs: vec![run("hidden", white)], ..Default::default() },
991            ];
992            let child = text_shape(5, "nested text", vec![Paragraph { runs: vec![run("hidden", white)], ..Default::default() }]);
993            let group = Shape {
994                id: ShapeId(4),
995                name: "group".into(),
996                descr: "group of text".into(),
997                kind: ShapeKind::Group {
998                    children: vec![Shape {
999                        id: ShapeId(7),
1000                        name: "nested group".into(),
1001                        descr: "nested group of text".into(),
1002                        kind: ShapeKind::Group { children: vec![child], child: Default::default() },
1003                        ..Default::default()
1004                    }],
1005                    child: Default::default(),
1006                },
1007                ..Default::default()
1008            };
1009            let sl = Arc::make_mut(doc.slides.get_mut(0).ok_or_else(|| bad("test", "no slide"))?);
1010            sl.shapes = vec![
1011                text_shape(2, "mixed", vec![mixed]),
1012                text_shape(3, "later paragraph", later),
1013                group,
1014                text_shape(6, "black control", vec![Paragraph { runs: vec![run("visible", black)], ..Default::default() }]),
1015                text_shape(
1016                    8,
1017                    "invisible white runs",
1018                    vec![Paragraph {
1019                        runs: vec![run("visible", black), Run { kind: RunKind::Break, ..run("\n", white) }, run("  ", white), run("visible", black)],
1020                        ..Default::default()
1021                    }],
1022                ),
1023            ];
1024            Ok(())
1025        })
1026        .unwrap();
1027
1028        let issues = s.execute("review.accessibility", &json!({})).unwrap();
1029        let contrasts: Vec<_> = issues
1030            .as_array()
1031            .unwrap()
1032            .iter()
1033            .filter(|issue| issue["issue"].as_str().unwrap_or("").starts_with("Hard-to-read text contrast"))
1034            .collect();
1035        assert_eq!(contrasts.len(), 3, "{issues}");
1036        for id in [2, 3, 5] {
1037            assert!(contrasts.iter().any(|issue| issue["shape"] == id), "missing shape {id}: {issues}");
1038        }
1039        assert!(!contrasts.iter().any(|issue| issue["shape"] == 6), "black control warned: {issues}");
1040        assert!(!contrasts.iter().any(|issue| issue["shape"] == 8), "white break or spaces warned: {issues}");
1041    }
