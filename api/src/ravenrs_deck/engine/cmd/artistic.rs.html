1//! Picture Format ▸ Artistic Effects: a picture redrawn as a pencil sketch, blurred, grainy…
2//!
3//! The effect is applied to the pixels and the result becomes the picture, as PowerPoint does;
4//! the original is kept (`PictureFill::artistic`) so the effect can be changed or removed, and
5//! both are saved (`a14:imgLayer`). The effects approximate PowerPoint's look.

38/// Largest side processed; bigger pictures are scaled down first.
39const MAX_SIDE: u32 = 3000;

41fn luma(p: &Rgba<u8>) -> f32 {
42    0.299 * p[0] as f32 + 0.587 * p[1] as f32 + 0.114 * p[2] as f32
43}

45fn gray(img: &RgbaImage) -> GrayImage {
46    GrayImage::from_fn(img.width(), img.height(), |x, y| image::Luma([luma(img.get_pixel(x, y)).round().clamp(0.0, 255.0) as u8]))
47}

49/// Sobel edge strength, 0–255.
50fn edges(g: &GrayImage) -> GrayImage {
51    let (w, h) = g.dimensions();
52    let at = |x: i64, y: i64| g.get_pixel(x.clamp(0, w as i64 - 1) as u32, y.clamp(0, h as i64 - 1) as u32)[0] as f32;
53    GrayImage::from_fn(w, h, |x, y| {
54        let (x, y) = (x as i64, y as i64);
55        let gx = at(x + 1, y - 1) + 2.0 * at(x + 1, y) + at(x + 1, y + 1) - at(x - 1, y - 1) - 2.0 * at(x - 1, y) - at(x - 1, y + 1);
56        let gy = at(x - 1, y + 1) + 2.0 * at(x, y + 1) + at(x + 1, y + 1) - at(x - 1, y - 1) - 2.0 * at(x, y - 1) - at(x + 1, y - 1);
57        image::Luma([((gx * gx + gy * gy).sqrt() / 4.0).clamp(0.0, 255.0) as u8])
58    })
59}

61/// Applies `effect` to `img` (alpha kept).
62pub fn apply(effect: &str, img: &RgbaImage) -> Option<RgbaImage> {
63    let (w, h) = img.dimensions();
64    let scale = ((w * w + h * h) as f32).sqrt() / 1000.0;
65    let alpha = |x: u32, y: u32| img.get_pixel(x, y)[3];
66    let out = match effect {
67        "artisticBlur" => image::imageops::blur(img, (6.0 * scale).max(1.0)),
68        "artisticPencilGrayscale" | "artisticLineDrawing" | "artisticPhotocopy" => {
69            let soft = image::imageops::blur(&gray(img), (0.8 * scale).max(0.5));
70            let e = edges(&soft);
71            RgbaImage::from_fn(w, h, |x, y| {
72                let ev = e.get_pixel(x, y)[0] as f32;
73                let gv = soft.get_pixel(x, y)[0] as f32;
74                let v = match effect {
75                    "artisticPencilGrayscale" => 255.0 - ev * 2.0 - (255.0 - gv) * 0.25,
76                    "artisticLineDrawing" => {
77                        if ev > 28.0 {
78                            255.0 - (ev - 28.0) * 3.0
79                        } else {
80                            255.0
81                        }
82                    }
83                    _ => {
84                        if ev > 40.0 || gv < 70.0 {
85                            0.0
86                        } else {
87                            255.0
88                        }
89                    }
90                };
91                let v = v.clamp(0.0, 255.0) as u8;
92                Rgba([v, v, v, alpha(x, y)])
93            })
94        }
95        "artisticFilmGrain" => {
96            // Deterministic noise, so the same picture gives the same grain.
97            let mut seed: u32 = 0x9E37_79B9;
98            RgbaImage::from_fn(w, h, |x, y| {
99                seed ^= seed << 13;
100                seed ^= seed >> 17;
101                seed ^= seed << 5;
102                let n = (seed % 61) as f32 - 30.0;
103                let p = img.get_pixel(x, y);
104                let c = |v: u8| ((v as f32 - 128.0) * 1.1 + 128.0 + n).clamp(0.0, 255.0) as u8;
105                Rgba([c(p[0]), c(p[1]), c(p[2]), p[3]])
106            })
107        }
108        "artisticCutout" => {
109            let soft = image::imageops::blur(img, (1.5 * scale).max(0.5));
110            RgbaImage::from_fn(w, h, |x, y| {
111                let p = soft.get_pixel(x, y);
112                let q = |v: u8| ((v as f32 / 255.0 * 3.0).round() / 3.0 * 255.0) as u8;
113                Rgba([q(p[0]), q(p[1]), q(p[2]), alpha(x, y)])
114            })
115        }
116        "artisticGlowEdges" => {
117            let e = edges(&gray(img));
118            RgbaImage::from_fn(w, h, |x, y| {
119                let k = (e.get_pixel(x, y)[0] as f32 / 255.0 * 2.5).min(1.0);
120                let p = img.get_pixel(x, y);
121                let c = |v: u8| (v as f32 * k + 255.0 * k * 0.3).clamp(0.0, 255.0) as u8;
122                Rgba([c(p[0]), c(p[1]), c(p[2]), p[3]])
123            })
124        }
125        _ => return None,
126    };
127    Some(out)
128}

130/// Decodes `data`, applies `effect` and encodes a PNG.
131fn process(effect: &str, data: &[u8]) -> Result<Vec<u8>> {
132    let img = image::load_from_memory(data).map_err(|_| bad("picture.artisticEffect", "this picture's format can't be processed"))?;
133    let img = if img.width().max(img.height()) > MAX_SIDE { img.thumbnail(MAX_SIDE, MAX_SIDE) } else { img };
134    let out = apply(effect, &img.to_rgba8()).ok_or_else(|| bad("picture.artisticEffect", "unknown effect"))?;
135    let mut png = Vec::new();
136    image::DynamicImage::ImageRgba8(out)
137        .write_to(&mut std::io::Cursor::new(&mut png), image::ImageFormat::Png)
138        .map_err(|_| bad("picture.artisticEffect", "the processed picture couldn't be stored"))?;
139    Ok(png)
140}

142fn artistic_effect(s: &mut Session, p: &Value) -> Result<Value> {
143    let effect = match str_param(p, "effect") {
144        Some("none") => None,
145        Some(e) if EFFECTS.iter().any(|(n, _)| *n == e) => Some(e.to_string()),
146        _ => return Err(bad("picture.artisticEffect", "`effect` is none or one of artisticBlur, artisticPencilGrayscale, …")),
147    };
148    let ids = targets(s, p)?;
149    let st = s.doc()?;
150    // (shape, original picture, its bytes and name)
151    let mut work: Vec<(ShapeId, MediaId, Option<(Vec<u8>, String)>)> = Vec::new();
152    for id in ids {
153        let Some(ShapeKind::Picture { fill }) = st.shape(id).map(|sh| &sh.kind) else { continue };
154        let original = fill.artistic.as_ref().map_or(fill.media, |a| a.original);
155        let item = st.doc.media(original).map(|m| (m.data.to_vec(), m.name.clone()));
156        work.push((id, original, item));
157    }
158    if work.is_empty() {
159        return Err(bad("picture.artisticEffect", "select a picture"));
160    }
161    let mut results: Vec<(ShapeId, MediaId, Option<(String, Vec<u8>)>)> = Vec::new();
162    for (id, original, item) in work {
163        let processed = match (&effect, item) {
164            (None, _) => None,
165            (Some(e), Some((data, name))) => {
166                Some((format!("{}-{}.png", name.rsplit_once('.').map_or(name.as_str(), |x| x.0), e), process(e, &data)?))
167            }
168            (Some(_), None) => return Err(bad("picture.artisticEffect", "the picture's data is missing")),
169        };
170        results.push((id, original, processed));
171    }
172    let effect2 = effect.clone();
173    s.edit(move |doc, sel| {
174        let mut set: Vec<(ShapeId, MediaId, Option<MediaId>)> = Vec::new();
175        for (id, original, processed) in &results {
176            let m = match processed {
177                Some((name, png)) => Some(doc.add_media(name, "image/png", png.clone()).ok_or_else(|| bad("picture.artisticEffect", NO_ID_SPACE))?),
178                None => None,
179            };
180            set.push((*id, *original, m));
181        }
182        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("picture.artisticEffect", "no slide"))?;
183        for (id, original, m) in set {
184            if let Some(sh) = crate::model::find_shape_mut(list, id)
185                && let ShapeKind::Picture { fill } = &mut sh.kind
186            {
187                match (m, &effect2) {
188                    (Some(m), Some(e)) => {
189                        fill.media = m;
190                        fill.artistic = Some(Artistic { effect: e.clone(), original });
191                    }
192                    _ => {
193                        fill.media = original;
194                        fill.artistic = None;
195                    }
196                }
197            }
198        }
199        Ok(())
200    })?;
201    Ok(json!({"effect": effect}))
