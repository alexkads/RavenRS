11pub fn specs() -> Vec<CommandSpec> {
12    vec![
13        cmd!(query "document.inspect", "Inspect Presentation", [], None, "{} → slides with titles, layouts, shape counts; selection; size; theme", has_doc, document),
14        cmd!(query "slide.inspect", "Inspect Slide", [], None, "{index?} → the slide's shapes with ids, kinds, boxes, text, fills, animations, transition, notes (text, and notesBody with its formatting)", has_doc, slide),
15        cmd!(query "slide.actions", "Inspect Slide Actions", [], None, "{index?} → {index, slideId, actions: [{owner, event, type, target, hyperlink}]}; lists shape click/hover actions and text-run links/hover actions (including table cells) without executing them", has_doc, actions),

17        cmd!(query "commands.list", "List Commands", [], None, "{filter?: substring} → [{id, label, params, enabled}]", always, commands),

21fn document(s: &mut Session, _p: &Value) -> Result<Value> {
22    let st = s.doc()?;
23    let d = &st.doc;
24    let slides: Vec<Value> = d
25        .slides
26        .iter()
27        .enumerate()
28        .map(|(i, sl)| {
29            let layout = d.layout(sl.layout).map(|(_, l)| l.name.clone()).unwrap_or_default();
30            json!({"index": i, "id": sl.id, "title": sl.title(), "layout": layout, "shapes": sl.shapes.len(), "hidden": sl.hidden, "transition": sl.transition.as_ref().map(|t| t.kind.clone()), "animations": sl.animations.len(), "notes": sl.notes_text()})
31        })
32        .collect();
33    let m = d.masters.first();
34    Ok(json!({
35        "title": st.title(),
36        "path": st.path,
37        "dirty": st.is_dirty(),
38        "slideSize": [d.slide_size.width, d.slide_size.height],
39        "slides": slides,
40        "sections": d.sections.iter().map(|x| json!({"name": x.name, "slides": x.slides.len()})).collect::<Vec<_>>(),
41        "theme": m.map(|m| m.theme.name.clone()),
42        "fonts": m.map(|m| json!({"major": m.theme.fonts.major.latin, "minor": m.theme.fonts.minor.latin})),
43        "layouts": m.map(|m| m.layouts.iter().map(|l| json!({"id": l.id, "name": l.name})).collect::<Vec<_>>()),
44        "selection": serde_json::to_value(&st.selection).unwrap_or_default(),
45        "media": d.media.iter().map(|x| json!({"id": x.id, "name": x.name, "type": x.content_type, "bytes": x.data.len()})).collect::<Vec<_>>(),
46        "undo": st.history.undo.last().map(|e| e.label.clone()),
47        "redo": st.history.redo.last().map(|e| e.label.clone()),
48    }))
49}

51pub fn shape_summary(ctx: Option<&Ctx>, sh: &Shape, depth: usize) -> Value {
52    let x = ctx.map(|c| resolve::xfrm(c, sh)).or(sh.xfrm).unwrap_or_default();
53    let fill = ctx.map(|c| {
54        let (f, ph) = resolve::fill(c, sh);
55        match f {
56            Some(crate::model::Fill::Solid { color }) => json!(format!("#{}", c.color(&color, ph).hex())),
57            Some(crate::model::Fill::None) | None => json!("none"),
58            Some(other) => json!(serde_json::to_value(&other).ok().and_then(|v| v.get("kind").cloned()).unwrap_or_default()),
59        }
60    });
61    let mut v = json!({
62        "id": sh.id,
63        "name": sh.name,
64        "kind": sh.kind_name(),
65        "box": [round(x.x), round(x.y), round(x.w), round(x.h)],
66        "rotation": x.rot,
67    });
68    if let Some(o) = v.as_object_mut() {
69        if let Some(p) = &sh.ph {
70            o.insert("placeholder".into(), json!(p.kind.xml()));
71        }
72        if let Some(n) = sh.geom.preset_name()
73            && matches!(sh.kind, ShapeKind::Shape | ShapeKind::Connector { .. })
74        {
75            o.insert("preset".into(), json!(n));
76        }
77        if let Some(t) = &sh.text
78            && !t.is_empty()
79        {
80            o.insert("text".into(), json!(t.text()));
81        }
82        if let Some(f) = fill {
83            o.insert("fill".into(), f);
84        }
85        if sh.hidden {
86            o.insert("hidden".into(), json!(true));
87        }
88        if !sh.descr.is_empty() {
89            o.insert("altText".into(), json!(sh.descr));
90        }
91        match &sh.kind {
92            ShapeKind::Table(t) => {
93                o.insert("table".into(), json!({"rows": t.n_rows(), "cols": t.n_cols(), "style": t.style, "cells": t.rows.iter().map(|r| r.cells.iter().map(|c| c.text.text()).collect::<Vec<_>>()).collect::<Vec<_>>()}));
94            }
95            ShapeKind::Chart(c) => {
96                o.insert("chart".into(), json!({"type": c.kind.label(), "title": c.title, "categories": c.categories, "series": c.series.iter().map(|s| json!({"name": s.name, "values": s.values})).collect::<Vec<_>>()}));
97            }
98            ShapeKind::Group { children, .. } if depth < 8 => {
99                o.insert("children".into(), Value::Array(children.iter().map(|c| shape_summary(None, c, depth + 1)).collect()));
100            }
101            ShapeKind::Media(m) => {
102                o.insert("media".into(), json!({"id": m.media, "video": m.video, "autoplay": m.autoplay, "loop": m.loop_play}));
103            }
104            _ => {}
105        }
106    }
107    v
108}

114fn slide(s: &mut Session, p: &Value) -> Result<Value> {
115    let st = s.doc()?;
116    let i = usize_param(p, "index").unwrap_or(st.selection.slide);
117    let sl = st.doc.slides.get(i).ok_or_else(|| bad("slide.inspect", format!("no slide {i}")))?;
118    let ctx = Ctx::for_slide(&st.doc, sl);
119    let shapes: Vec<Value> = sl.shapes.iter().map(|sh| shape_summary(ctx.as_ref(), sh, 0)).collect();
120    let bg = ctx.as_ref().map(|c| match resolve::background(c, Some(sl)) {
121        (crate::model::Fill::Solid { color }, ph) => json!(format!("#{}", c.color(&color, ph).hex())),
122        (f, _) => serde_json::to_value(&f).ok().and_then(|v| v.get("kind").cloned()).unwrap_or_default(),
123    });
124    Ok(json!({
125        "index": i,
126        "id": sl.id,
127        "layout": st.doc.layout(sl.layout).map(|(_, l)| l.name.clone()),
128        "title": sl.title(),
129        "background": bg,
130        "shapes": shapes,
131        "transition": serde_json::to_value(&sl.transition).unwrap_or_default(),
132        "animations": serde_json::to_value(&sl.animations).unwrap_or_default(),
133        "notes": sl.notes_text(),
134        "notesBody": sl.notes,
135        "comments": sl.comments.len(),
136        "hidden": sl.hidden,
137    }))
138}

140fn actions(s: &mut Session, p: &Value) -> Result<Value> {
141    let st = s.doc()?;
142    let i = opt_usize_param(p, "index", "slide.actions")?.unwrap_or(st.selection.slide);
143    let sl = st.doc.slides.get(i).ok_or_else(|| bad("slide.actions", format!("no slide {i}")))?;
144    let mut found = Vec::new();
145    for sh in &sl.shapes {
146        collect_actions(sh, &mut Vec::new(), &mut found);
147    }
148    Ok(json!({"index": i, "slideId": sl.id, "actions": found}))
149}
150
151fn collect_actions(sh: &Shape, path: &mut Vec<Value>, found: &mut Vec<Value>) {
152    path.push(json!({"id": sh.id, "name": sh.name}));
153    let owner = json!({"shapeId": sh.id, "shapeName": sh.name, "path": path});
154    if let Some(link) = &sh.click {
155        found.push(action_entry(owner.clone(), "click", link));
156    }
157    if let Some(link) = &sh.hover {
158        found.push(action_entry(owner.clone(), "hover", link));
159    }
160    if let Some(text) = &sh.text {
161        collect_text_actions(text, &owner, found);
162    }
163    if let ShapeKind::Table(table) = &sh.kind {
164        for (row_index, row) in table.rows.iter().enumerate() {
165            for (cell_index, cell) in row.cells.iter().enumerate() {
166                let mut cell_owner = owner.clone();
167                cell_owner["tableRowIndex"] = json!(row_index);
168                cell_owner["tableCellIndex"] = json!(cell_index);
169                collect_text_actions(&cell.text, &cell_owner, found);
170            }
171        }
172    }
173    if let ShapeKind::Group { children, .. } = &sh.kind {
174        for child in children {
175            collect_actions(child, path, found);
176        }
177    }
178    path.pop();
179}
180
181fn collect_text_actions(text: &crate::model::text::TextBody, owner: &Value, found: &mut Vec<Value>) {
182    for (paragraph_index, paragraph) in text.paragraphs.iter().enumerate() {
183        for (run_index, run) in paragraph.runs.iter().enumerate() {
184            for (event, link) in [("textLink", run.props.link.as_ref()), ("textHover", run.props.hover.as_ref())] {
185                if let Some(link) = link {
186                    let mut text_owner = owner.clone();
187                    text_owner["paragraphIndex"] = json!(paragraph_index);
188                    text_owner["runIndex"] = json!(run_index);
189                    text_owner["text"] = json!(run.content());
190                    found.push(action_entry(text_owner, event, link));
191                }
192            }
193        }
194    }
195}

197fn action_entry(owner: Value, event: &str, link: &Hyperlink) -> Value {
198    let action = serde_json::to_value(&link.action).unwrap_or_default();
199    let kind = action.get("kind").cloned().unwrap_or(Value::Null);
200    let target = ["url", "slide", "name", "path"].iter().find_map(|key| action.get(key).cloned()).unwrap_or(Value::Null);
201    json!({"owner": owner, "event": event, "type": kind, "target": target, "hyperlink": link})
202}

204fn shape(s: &mut Session, p: &Value) -> Result<Value> {
205    let st = s.doc()?;
206    let id =
207        id_param(p, "id", "shape.inspect")?.or_else(|| st.selection.shapes.first().copied()).ok_or_else(|| bad("shape.inspect", "missing `id`"))?;
208    let sh = st.shape(id).ok_or_else(|| bad("shape.inspect", format!("no shape {id}")))?;
209    let x = xfrm_of(&st.doc, &st.selection, sh);
210    Ok(json!({"shape": serde_json::to_value(sh).unwrap_or_default(), "box": x}))
211}

213fn commands(s: &mut Session, p: &Value) -> Result<Value> {
214    let f = str_param(p, "filter").unwrap_or("").to_ascii_lowercase();
215    Ok(Value::Array(
216        s.commands()
217            .into_iter()
218            .filter(|c| f.is_empty() || c.id.to_ascii_lowercase().contains(&f) || c.label.to_ascii_lowercase().contains(&f))
219            .map(|c| serde_json::to_value(c).unwrap_or_default())
220            .collect(),
221    ))
222}

236    #[test]
237    fn slide_actions_lists_group_shape_and_text_actions() {
238        let mut s = Session::with_new();
239        let mut child = Shape { id: ShapeId(12), name: "Child".into(), ..Default::default() };
240        child.click = Some(link(Action::NextSlide));
241        child.hover = Some(link(Action::Url { url: "https://example.com/hover".into() }));
242        let run_props = RunProps { link: Some(link(Action::Slide { slide: crate::model::SlideId(7) })), ..Default::default() };
243        let mut paragraph = Paragraph::new("linked");
244        paragraph.runs = vec![Run::with("linked", run_props)];
245        child.text = Some(TextBody { paragraphs: vec![paragraph], ..Default::default() });
246        let group = Shape {
247            id: ShapeId(11),
248            name: "Group".into(),
249            kind: ShapeKind::Group { children: vec![child], child: Xfrm::default() },
250            ..Default::default()
251        };
252        let st = s.doc_mut().unwrap();
253        let slide_index = st.selection.slide;
254        let presentation = std::sync::Arc::make_mut(&mut st.doc);
255        std::sync::Arc::make_mut(&mut presentation.slides[slide_index]).shapes = vec![group];
256
257        let v = s.execute("slide.actions", &json!({})).unwrap();
258        let actions = v["actions"].as_array().unwrap();
259        assert_eq!(actions.len(), 3, "{v}");
260        assert_eq!(actions[0]["event"], "click");
261        assert_eq!(actions[0]["type"], "nextSlide");
262        assert_eq!(actions[0]["owner"]["path"][0]["name"], "Group");
263        assert_eq!(actions[0]["owner"]["path"][1]["id"], 12);
264        assert_eq!(actions[1]["event"], "hover");
265        assert_eq!(actions[1]["target"], "https://example.com/hover");
266        assert_eq!(actions[2]["event"], "textLink");
267        assert_eq!(actions[2]["owner"]["runIndex"], 0);
268        assert_eq!(actions[2]["target"], 7);
269
270        let cell_props = RunProps { hover: Some(link(Action::Url { url: "https://example.com/cell".into() })), ..Default::default() };
271        let mut cell_text = TextBody::from_text("cell link");
272        cell_text.paragraphs[0].runs = vec![Run::with("cell link", cell_props)];
273        let table =
274            Table { rows: vec![Row { cells: vec![Cell { text: cell_text, ..Default::default() }], ..Default::default() }], ..Default::default() };
275        let table_shape = Shape { id: ShapeId(13), name: "Table".into(), kind: ShapeKind::Table(table), ..Default::default() };
276        let st = s.doc_mut().unwrap();
277        let presentation = std::sync::Arc::make_mut(&mut st.doc);
278        std::sync::Arc::make_mut(&mut presentation.slides[slide_index]).shapes.push(table_shape);
279        let v = s.execute("slide.actions", &json!({})).unwrap();
280        let table_action = v["actions"].as_array().unwrap().last().cloned().unwrap();
281        assert_eq!(table_action["event"], "textHover");
282        assert_eq!(table_action["owner"]["tableRowIndex"], 0);
283        assert_eq!(table_action["owner"]["tableCellIndex"], 0);
284        assert_eq!(table_action["target"], "https://example.com/cell");
285    }
