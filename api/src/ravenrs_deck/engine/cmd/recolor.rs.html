48/// (id, label), in PowerPoint's gallery order: colour modes, then dark and light variations.
49pub fn presets() -> Vec<(String, String)> {
50    let mut v: Vec<(String, String)> = [
51        ("none", "No Recolor"),
52        ("grayscale", "Grayscale"),
53        ("sepia", "Sepia"),
54        ("washout", "Washout"),
55        ("bw25", "Black and White: 25%"),
56        ("bw50", "Black and White: 50%"),
57        ("bw75", "Black and White: 75%"),
58    ]
59    .iter()
60    .map(|(a, b)| (a.to_string(), b.to_string()))
61    .collect();
62    for (id, label, _) in ACCENTS {
63        v.push((format!("{id}Dark"), format!("{label} Dark")));
64    }
65    for (id, label, _) in ACCENTS {
66        v.push((format!("{id}Light"), format!("{label} Light")));
67    }
68    v
69}

71/// Set `a`'s recolour to `preset`, clearing any earlier one. `false` for an unknown preset.
72pub fn apply(a: &mut PictureAdjust, preset: &str) -> bool {
73    let washed = a.brightness == 0.7 && a.contrast == -0.7;
74    let mut n = PictureAdjust { grayscale: false, duotone: None, bilevel: None, ..a.clone() };
75    if washed {
76        n.brightness = 0.0;
77        n.contrast = 0.0;
78    }
79    let rgb = |r, g, b| ColorRef::rgb(Rgba::rgb(r, g, b));
80    match preset {
81        "none" => {}
82        "grayscale" => n.grayscale = true,
83        // PowerPoint's sepia is a duotone from dark brown to light cream.
84        "sepia" => n.duotone = Some((rgb(0x40, 0x2A, 0x10), rgb(0xF2, 0xE3, 0xC6))),
85        // Washout is PowerPoint's +70% brightness, −70% contrast.
86        "washout" => {
87            n.brightness = 0.7;
88            n.contrast = -0.7;
89        }
90        "bw25" => n.bilevel = Some(0.25),
91        "bw50" => n.bilevel = Some(0.5),
92        "bw75" => n.bilevel = Some(0.75),
93        other => {
94            let Some((slot, dark)) = ACCENTS.iter().find_map(|(id, _, slot)| {
95                other.strip_prefix(id).and_then(|rest| match rest {
96                    "Dark" => Some((*slot, true)),
97                    "Light" => Some((*slot, false)),
98                    _ => None,
99                })
100            }) else {
101                return false;
102            };
103            let c = ColorRef::scheme(slot);
104            n.duotone = Some(if dark { (rgb(0, 0, 0), c) } else { (c, rgb(0xFF, 0xFF, 0xFF)) });
105        }
106    }
107    *a = n;
108    true
109}

111fn recolor(s: &mut Session, p: &Value) -> Result<Value> {
112    let preset = str_param(p, "preset").ok_or_else(|| bad("picture.recolor", "missing `preset`"))?.to_string();
113    if !presets().iter().any(|(id, _)| *id == preset) {
114        return Err(bad("picture.recolor", format!("unknown preset `{preset}`; see picture.recolors")));
115    }
116    edit_shapes(s, p, "picture.recolor", |sh| {
117        if let ShapeKind::Picture { fill } = &mut sh.kind {
118            apply(&mut fill.adjust, &preset);
119        }
120        Ok(())
121    })
122}

124fn transparent_color(s: &mut Session, p: &Value) -> Result<Value> {
125    if let Some(at) = p.get("at") {
126        let (id, c) = picture_pixel_at(s, p, at)?;
127        return edit_shapes(s, &json!({"ids": [id]}), "picture.transparentColor", |sh| {
128            if let ShapeKind::Picture { fill } = &mut sh.kind {
129                fill.adjust.clear_color = Some(c);
130            }
131            Ok(())
132        });
133    }
134    let c = if bool_or(p, "none", false) {
135        None
136    } else {
137        let hex = str_param(p, "color").ok_or_else(|| bad("picture.transparentColor", "give `color` (#RRGGBB) or `none: true`"))?;
138        Some(Rgba::from_hex(hex.trim_start_matches('#')).ok_or_else(|| bad("picture.transparentColor", format!("bad colour `{hex}`")))?)
139    };
140    edit_shapes(s, p, "picture.transparentColor", |sh| {
141        if let ShapeKind::Picture { fill } = &mut sh.kind {
142            fill.adjust.clear_color = c;
143        }
144        Ok(())
145    })
146}

148/// The topmost target picture under slide point `at` and the colour of its own pixel there: the
149/// picture as stored, before corrections and recolouring, so the colour matches the pixels that
150/// `clear_color` is compared with.
151fn picture_pixel_at(s: &Session, p: &Value, at: &Value) -> Result<(ShapeId, Rgba)> {
152    const CMD: &str = "picture.transparentColor";
153    let xy = |i: usize| at.get(i).and_then(Value::as_f64).filter(|v| v.is_finite());
154    let (Some(x), Some(y)) = (xy(0), xy(1)) else {
155        return Err(bad(CMD, "`at` is [x, y] in slide units"));
156    };
157    let mut ids = targets(s, p)?;
158    let st = s.doc()?;
159    let shapes = current_shapes(&st.doc, &st.selection).ok_or_else(|| bad(CMD, "no slide"))?;
160    // Same order as the hit-test: later in the shape list is in front, and a group's children sit
161    // where the group is. The selection order says nothing about what is on top.
162    ids.sort_by_cached_key(|id| std::cmp::Reverse(stack_path(shapes, *id)));
163    for id in ids {
164        let Some((sh, xf, to_slide)) = placed(&st.doc, &st.selection, shapes, id, Affine::IDENTITY, 0) else { continue };
165        let ShapeKind::Picture { fill } = &sh.kind else { continue };
166        // The camera turns the picture where it is drawn, so the click is mapped back through it too.
167        let turn = text::ctx_for(&st.doc, &st.selection).and_then(|c| crate::render::camera_turn(&c, sh, xf.w, xf.h)).unwrap_or(Affine::IDENTITY);
168        let local = (to_slide * turn).inverse() * Point::new(x, y);
169        if !(local.is_finite() && (0.0..=xf.w).contains(&local.x) && (0.0..=xf.h).contains(&local.y)) {
170            continue;
171        }
172        let PictureMode::Stretch { fill_rect: [l, t, r, b] } = fill.mode else {
173            return Err(bad(CMD, "a tiled picture has no single pixel under a point; give `color`"));
174        };
175        let inner = Rect::new(l * xf.w, t * xf.h, xf.w - r * xf.w, xf.h - b * xf.h);
176        let dest = crate::render::crop_dest(inner, fill.crop);
177        let img = st.doc.media(fill.media).and_then(|m| crate::render::decode_image(&m.data)).ok_or_else(|| bad(CMD, "the picture can't be read"))?;
178        let (u, v) = ((local.x - dest.x0) / dest.width(), (local.y - dest.y0) / dest.height());
179        if !((0.0..=1.0).contains(&u) && (0.0..=1.0).contains(&v)) {
180            continue;
181        }
182        let (w, h) = (img.width(), img.height());
183        let px = ((u * f64::from(w)) as u32).min(w.saturating_sub(1));
184        let py = ((v * f64::from(h)) as u32).min(h.saturating_sub(1));
185        let Some(c) = img.get_pixel_checked(px, py) else { continue };
186        // A transparent pixel keeps a colour nobody sees (often black): clearing it would clear
187        // that colour where it is drawn.
188        if c[3] == 0 {
189            return Err(bad(CMD, "that spot of the picture is already transparent"));
190        }
191        return Ok((id, Rgba::rgb(c[0], c[1], c[2])));
192    }
193    Err(bad(CMD, "the point is not on a selected picture"))
194}

196/// Where `id` sits in the paint order: its index in each list from the slide down to it. A larger
197/// path is drawn later, so in front.
198fn stack_path(shapes: &[Shape], id: ShapeId) -> Option<Vec<usize>> {
199    for (i, sh) in shapes.iter().enumerate() {
200        if sh.id == id {
201            return Some(vec![i]);
202        }
203        if let ShapeKind::Group { children, .. } = &sh.kind
204            && let Some(mut rest) = stack_path(children, id)
205        {
206            rest.insert(0, i);
207            return Some(rest);
208        }
209    }
210    None
211}

213/// Shape `id` among `shapes` (searched into groups), its box, and the map from that box to slide
214/// space.
215fn placed<'a>(
216    doc: &crate::model::Presentation,
217    sel: &crate::engine::Selection,
218    shapes: &'a [Shape],
219    id: ShapeId,
220    to_slide: Affine,
221    depth: usize,
222) -> Option<(&'a Shape, Xfrm, Affine)> {
223    if depth > 64 {
224        return None;
225    }
226    for sh in shapes {
227        let x = xfrm_of(doc, sel, sh);
228        if sh.id == id {
229            return Some((sh, x, to_slide * x.affine()));
230        }
231        if let ShapeKind::Group { children, child } = &sh.kind {
232            let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
233            if let Some(found) = placed(doc, sel, children, id, to_slide * crate::geom::group_child_affine(&x, ch), depth + 1) {
234                return Some(found);
235            }
236        }
237    }
238    None
