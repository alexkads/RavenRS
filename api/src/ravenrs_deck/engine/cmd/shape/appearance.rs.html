7pub(super) fn adjust(s: &mut Session, p: &Value) -> Result<Value> {
8    let i = opt_usize_param(p, "index", "shape.adjust")?.unwrap_or(0).min(16);
9    let v = f64_param(p, "value").ok_or_else(|| bad("shape.adjust", "missing `value`"))?;
10    edit_geometry(s, p, "shape.adjust", |sh| {
11        if let Geom::Preset { name, adj } = &mut sh.geom {
12            let defaults = crate::geom::preset::info(name).map(|x| x.defaults).unwrap_or(&[]);
13            while adj.len() <= i {
14                let k = adj.len();
15                adj.push(defaults.get(k).copied().unwrap_or(0.0));
16            }
17            if let Some(slot) = adj.get_mut(i) {
18                *slot = v;
19            }
20        }
21        Ok(())
22    })
23}

25/// Rectangles and rounded rectangles become custom geometry whose corners are smoothed
26/// ([`crate::geom::squircle`]); the radius is the rounded rectangle's own unless `radius` says.
27pub(super) fn smooth_corners(s: &mut Session, p: &Value) -> Result<Value> {
28    const CMD: &str = "shape.smoothCorners";
29    let smoothing =
30        f64_param(p, "smoothing").filter(|v| v.is_finite()).ok_or_else(|| bad(CMD, "missing `smoothing` (0–100)"))?.clamp(0.0, 100.0) / 100.0;
31    let radius = f64_param(p, "radius").filter(|v| v.is_finite()).map(|v| v.max(0.0));
32    edit_geometry(s, p, CMD, |sh| {
33        let Some(x) = sh.xfrm else { return Err(bad(CMD, "the shape has no size of its own")) };
34        let ss = x.w.min(x.h).max(0.0);
35        let own = match &sh.geom {
36            Geom::Preset { name, adj } if name == "roundRect" => Some(ss * adj.first().copied().unwrap_or(16667.0).clamp(0.0, 50000.0) / 100000.0),
37            Geom::Preset { name, .. } if name == "rect" => Some(ss * 16667.0 / 100000.0),
38            _ => None,
39        };
40        // A custom outline qualifies only when it is a squircle this command drew with `radius`
41        // (re-smoothing as the slider moves), never an arbitrary freeform.
42        let custom = match (&sh.geom, radius) {
43            (Geom::Custom { paths }, Some(r)) => match paths.as_slice() {
44                [only] => (0..=100).any(|k| crate::geom::squircle::path_d(only.w, only.h, r, f64::from(k) / 100.0) == only.d),
45                _ => false,
46            },
47            _ => false,
48        };
49        let r = match (radius, own) {
50            (Some(r), _) if own.is_some() || custom => r,
51            (None, Some(r)) => r,
52            _ => return Err(bad(CMD, "only rectangles, rounded rectangles and shapes already smoothed with this `radius` can be smoothed")),
53        };
54        sh.geom = Geom::Custom {
55            paths: vec![crate::model::CustomPath { w: x.w, h: x.h, d: crate::geom::squircle::path_d(x.w, x.h, r, smoothing), ..Default::default() }],
56        };
57        Ok(())
58    })
59}

70/// `{stops: [[pos 0–1, color], …], kind?: linear|radial|rectangular|path, angle?: deg (linear),
71/// focus?: [l, t, r, b] fractions (radial/rectangular/path: where the first stop sits),
72/// rotateWithShape?}`. The renderer keeps only linear and radial gradients upright when
73/// `rotateWithShape` is false; rectangular and path gradients turn with the shape.
74pub(crate) fn gradient_param(v: &Value) -> Option<Gradient> {
75    let stops: Vec<GradientStop> = v
76        .get("stops")?
77        .as_array()?
78        .iter()
79        .filter_map(|s| {
80            let a = s.as_array()?;
81            Some(GradientStop { pos: a.first()?.as_f64()?.clamp(0.0, 1.0), color: color_value(a.get(1)?)? })
82        })
83        .collect();
84    if stops.len() < 2 {
85        return None;
86    }
87    let focus = v
88        .get("focus")
89        .and_then(Value::as_array)
90        .and_then(|a| {
91            let f = |i: usize| a.get(i).and_then(Value::as_f64).filter(|x| x.is_finite()).map(|x| x.clamp(0.0, 1.0));
92            Some([f(0)?, f(1)?, f(2)?, f(3)?])
93        })
94        .unwrap_or([0.5, 0.5, 0.5, 0.5]);
95    let mut stops = stops;
96    stops.sort_by(|a, b| a.pos.total_cmp(&b.pos));
97    let path = |p: &str| GradientShape::Path { path: p.into(), focus };
98    let shape = match v.get("kind").and_then(Value::as_str) {
99        Some("radial" | "circle") => path("circle"),
100        Some("rectangular" | "rect") => path("rect"),
101        Some("path" | "shape") => path("shape"),
102        _ => GradientShape::Linear {
103            angle: v.get("angle").and_then(Value::as_f64).filter(|a| a.is_finite()).unwrap_or(90.0).rem_euclid(360.0),
104            scaled: false,
105        },
106    };
107    Some(Gradient {
108        tile: [0.0; 4],
109        stops,
110        shape,
111        rotate_with_shape: v.get("rotateWithShape").and_then(Value::as_bool).unwrap_or(true),
112        flip: v.get("flip").and_then(Value::as_str).and_then(crate::model::style::GradientFlip::from_xml),
113    })
114}

116/// Fill and outline apply to a group's visible members as well as its container. Other edits
117/// (notably geometry) still target only the selected shape. The whole traversal stays in one edit.
118fn edit_group_appearance(s: &mut Session, p: &Value, cmd: &str, f: impl Fn(&mut crate::model::Shape) -> Result<()>) -> Result<Value> {
119    fn apply(sh: &mut crate::model::Shape, depth: usize, f: &dyn Fn(&mut crate::model::Shape) -> Result<()>) -> Result<()> {
120        if depth > 64 {
121            return Ok(());
122        }
123        f(sh)?;
124        if let Some(children) = sh.children_mut() {
125            for child in children {
126                apply(child, depth + 1, f)?;
127            }
128        }
129        Ok(())
130    }
131    edit_shapes(s, p, cmd, |sh| apply(sh, 0, &f))
132}

134pub(super) fn fill(s: &mut Session, p: &Value) -> Result<Value> {
135    let new_fill = requested_fill(s, p)?;
136    let transparency = f64_param(p, "transparency");
137    let pic_props = p.get("pictureProps").filter(|v| v.is_object());
138    if new_fill.is_none() && transparency.is_none() && pic_props.is_none() {
139        return Err(bad(
140            "shape.fill",
141            "give `color`, `none`, `background`, `gradient`, `pattern`, `picture`, `pictureProps`, `transparency` or `reset`",
142        ));
143    }
144    // Transparency on an inherited (style) fill needs the effective colour.
145    let inherited = inherited_fills(s, p)?;
146    let background = matches!(new_fill, Some(Some(Fill::Background)));
147    if background {
148        let st = s.doc()?;
149        let mut any = false;
150        for id in targets(s, p)? {
151            if let Some(sh) = st.shape(id) {
152                crate::model::walk(std::slice::from_ref(sh), &mut |sh, _| any |= matches!(sh.kind, crate::model::ShapeKind::Shape));
153            }
154        }
155        if !any {
156            return Err(bad("shape.fill", "slide background fill applies to shapes and text boxes"));
157        }
158    }
159    edit_group_appearance(s, p, "shape.fill", |sh| {
160        if background && !matches!(sh.kind, crate::model::ShapeKind::Shape) {
161            if matches!(sh.kind, crate::model::ShapeKind::Group { .. }) && !group_fill_used(sh, 0) {
162                sh.fill = None;
163            }
164            return Ok(());
165        }
166        apply_fill(sh, new_fill.as_ref(), pic_props, transparency, &inherited)
167    })
168}

170/// Only a shape or text box (`p:sp`) saves a slide background fill (`useBgFill`). Pictures,
171/// connectors, tables and charts keep their fill, and a group's own fill is cleared unless one
172/// of them is filled with it (`grpFill`).
173fn group_fill_used(sh: &crate::model::Shape, depth: usize) -> bool {
174    depth <= 64
175        && sh.children().iter().any(|c| match c.kind {
176            crate::model::ShapeKind::Shape => false,
177            crate::model::ShapeKind::Group { .. } => matches!(c.fill, None | Some(Fill::Group)) && group_fill_used(c, depth + 1),
178            _ => c.fill == Some(Fill::Group),
179        })
180}

182/// Applies Format Shape ▸ Picture or texture fill options to a picture fill.
183pub(super) fn picture_props(pf: &mut PictureFill, pp: &Value) {
184    let f = |k: &str| pp.get(k).and_then(Value::as_f64).filter(|v| v.is_finite());
185    if let Some(up) = pp.get("rotateWithShape").and_then(Value::as_bool) {
186        pf.upright = !up;
187    }
188    match pp.get("tile").and_then(Value::as_bool) {
189        Some(true) if !matches!(pf.mode, PictureMode::Tile { .. }) => {
190            pf.mode = PictureMode::Tile { tx: 0.0, ty: 0.0, sx: 1.0, sy: 1.0, flip: "none".into(), align: "tl".into() };
191        }
192        Some(false) if !matches!(pf.mode, PictureMode::Stretch { .. }) => pf.mode = PictureMode::default(),
193        _ => {}
194    }
195    match &mut pf.mode {
196        PictureMode::Tile { tx, ty, sx, sy, flip, align } => {
197            if let Some(v) = f("offsetX") {
198                *tx = v.clamp(-4000.0, 4000.0);
199            }
200            if let Some(v) = f("offsetY") {
201                *ty = v.clamp(-4000.0, 4000.0);
202            }
203            if let Some(v) = f("scaleX") {
204                *sx = v.clamp(0.01, 10.0);
205            }
206            if let Some(v) = f("scaleY") {
207                *sy = v.clamp(0.01, 10.0);
208            }
209            if let Some(m) = pp.get("mirror").and_then(Value::as_str).filter(|m| matches!(*m, "none" | "x" | "y" | "xy")) {
210                *flip = m.to_string();
211            }
212            if let Some(a) =
213                pp.get("align").and_then(Value::as_str).filter(|a| matches!(*a, "tl" | "t" | "tr" | "l" | "ctr" | "r" | "bl" | "b" | "br"))
214            {
215                *align = a.to_string();
216            }
217        }
218        PictureMode::Stretch { fill_rect } => {
219            if let Some(arr) = pp.get("stretch").and_then(Value::as_array) {
220                for (slot, v) in fill_rect.iter_mut().zip(arr) {
221                    if let Some(v) = v.as_f64().filter(|v| v.is_finite()) {
222                        *slot = v.clamp(-10.0, 0.99);
223                    }
224                }
225            }
226        }
227    }
228}

234pub(super) fn line(s: &mut Session, p: &Value) -> Result<Value> {
235    let reset = bool_or(p, "reset", false);
236    let none = bool_or(p, "none", false);
237    let color = color_param(p, "color");
238    let width = f64_param(p, "width").map(|w| w.clamp(0.0, 1584.0));
239    let dash = str_param(p, "dash").map(|d| Dash::Preset { name: d.to_string() });
240    let head = str_param(p, "head").map(line_end);
241    let tail = str_param(p, "tail").map(line_end);
242    let cap = str_param(p, "cap").map(|c| match c {
243        "round" => crate::model::style::LineCap::Round,
244        "square" => crate::model::style::LineCap::Square,
245        _ => crate::model::style::LineCap::Flat,
246    });
247    let compound = str_param(p, "compound").map(|c| match c {
248        "double" => crate::model::style::Compound::Double,
249        "thickThin" => crate::model::style::Compound::ThickThin,
250        "thinThick" => crate::model::style::Compound::ThinThick,
251        "triple" => crate::model::style::Compound::Triple,
252        _ => crate::model::style::Compound::Single,
253    });
254    let end_size = |k: &str| str_param(p, k).filter(|v| matches!(*v, "sm" | "med" | "lg")).map(str::to_string);
255    let (head_size, tail_size) = (end_size("headSize"), end_size("tailSize"));
256    let join = str_param(p, "join").map(|c| match c {
257        "bevel" => crate::model::style::LineJoin::Bevel,
258        "miter" => crate::model::style::LineJoin::Miter,
259        _ => crate::model::style::LineJoin::Round,
260    });
261    edit_group_appearance(s, p, "shape.line", |sh| {
262        if reset {
263            sh.line = None;
264            return Ok(());
265        }
266        let l = sh.line.get_or_insert_with(Line::default);
267        if none {
268            l.fill = Some(Fill::None);
269        }
270        if let Some(c) = &color {
271            l.fill = Some(Fill::solid(c.clone()));
272        }
273        if width.is_some() {
274            l.width = width;
275            if l.fill.is_none() || matches!(l.fill, Some(Fill::None)) && !none {
276                l.fill = Some(Fill::solid(ColorRef::scheme(SchemeSlot::Tx1)));
277            }
278        }
279        if dash.is_some() {
280            l.dash.clone_from(&dash);
281        }
282        let keep_size = |old: &Option<LineEnd>, new: &Option<LineEnd>| {
283            new.clone().map(|mut n| {
284                if let Some(o) = old {
285                    n.w.clone_from(&o.w);
286                    n.len.clone_from(&o.len);
287                }
288                n
289            })
290        };
291        if let Some(h) = &head {
292            l.head = keep_size(&l.head, h);
293        }
294        if let Some(t) = &tail {
295            l.tail = keep_size(&l.tail, t);
296        }
297        if cap.is_some() {
298            l.cap = cap;
299        }
300        if join.is_some() {
301            l.join = join;
302        }
303        if compound.is_some() {
304            l.compound = compound;
305        }
306        // Arrowhead size (width and length together), kept when the kind changes.
307        for (end, size) in [(&mut l.head, &head_size), (&mut l.tail, &tail_size)] {
308            if let (Some(e), Some(sz)) = (end.as_mut(), size) {
309                e.w.clone_from(sz);
310                e.len.clone_from(sz);
311            }
312        }
313        Ok(())
314    })
315}

317fn shadow_preset(v: &Value) -> Option<Option<Shadow>> {
318    let black = |a: i32| ColorRef::rgb(Rgba::BLACK).with(ColorTransform::Alpha(a));
319    let base = |inner: bool| Shadow {
320        color: black(40000),
321        blur: 4.0,
322        dist: 3.0,
323        dir: 45.0,
324        inner,
325        sx: 1.0,
326        sy: 1.0,
327        kx: 0.0,
328        ky: 0.0,
329        align: "tl".into(),
330        rotate_with_shape: false,
331    };
332    match v {
333        Value::String(s) => Some(match s.as_str() {
334            "none" => None,
335            "inner" => Some(base(true)),
336            "perspective" => Some(Shadow { sy: 0.23, kx: -1200000.0 / 60000.0, dir: 90.0, dist: 0.0, blur: 6.0, align: "bl".into(), ..base(false) }),
337            "bottom" => Some(Shadow { dir: 90.0, ..base(false) }),
338            "center" => Some(Shadow { dist: 0.0, blur: 8.0, ..base(false) }),
339            _ => Some(base(false)),
340        }),
341        Value::Object(o) => Some(Some(Shadow {
342            color: o.get("color").and_then(color_value).unwrap_or(black(40000)),
343            blur: o.get("blur").and_then(Value::as_f64).unwrap_or(4.0).clamp(0.0, 200.0),
344            dist: o.get("dist").and_then(Value::as_f64).unwrap_or(3.0).clamp(0.0, 200.0),
345            dir: o.get("dir").and_then(Value::as_f64).unwrap_or(45.0),
346            inner: o.get("inner").and_then(Value::as_bool).unwrap_or(false),
347            ..base(false)
348        })),
349        _ => None,
350    }
351}

353pub(super) fn effects(s: &mut Session, p: &Value) -> Result<Value> {
354    let reset = bool_or(p, "reset", false);
355    let shadow = p.get("shadow").and_then(shadow_preset);
356    let glow: Option<Option<Glow>> = p.get("glow").map(|g| match g {
357        Value::Object(o) => Some(Glow {
358            color: o.get("color").and_then(color_value).unwrap_or(ColorRef::scheme(SchemeSlot::Accent1).with(ColorTransform::Alpha(40000))),
359            radius: o.get("radius").and_then(Value::as_f64).unwrap_or(8.0).clamp(0.0, 150.0),
360        }),
361        Value::Number(n) => Some(Glow {
362            color: ColorRef::scheme(SchemeSlot::Accent1).with(ColorTransform::Alpha(40000)),
363            radius: n.as_f64().unwrap_or(8.0).clamp(0.0, 150.0),
364        }),
365        _ => None,
366    });
367    let soft = p.get("softEdges").map(|v| v.as_f64().filter(|x| *x > 0.0).map(|x| x.min(100.0)));
368    let refl = str_param(p, "reflection").map(|r| match r {
369        "none" => None,
370        "half" => Some(Reflection { blur: 0.5, start_alpha: 0.5, end_alpha: 0.0, end_pos: 0.55, dist: 4.0, ..Default::default() }),
371        "full" => Some(Reflection { blur: 0.5, start_alpha: 0.5, end_alpha: 0.0, end_pos: 0.9, dist: 4.0, ..Default::default() }),
372        _ => Some(Reflection { blur: 0.5, start_alpha: 0.5, end_alpha: 0.0, end_pos: 0.35, dist: 0.0, ..Default::default() }),
373    });
374    let three_d = p.get("threeD").filter(|v| v.is_object());
375    let bevel = str_param(p, "bevel").map(|b| if b == "none" { None } else { Some(b.to_string()) });
376    let shadow_props = p.get("shadowProps").filter(|v| v.is_object()).cloned();
377    let glow_props = p.get("glowProps").filter(|v| v.is_object()).cloned();
378    let refl_props = p.get("reflectionProps").filter(|v| v.is_object()).cloned();
379    edit_shapes(s, p, "shape.effects", |sh| {
380        if reset {
381            sh.effects = None;
382            return Ok(());
383        }
384        let e = sh.effects.get_or_insert_with(Effects::default);
385        if let Some(sw) = &shadow {
386            match sw {
387                Some(x) if x.inner => {
388                    e.inner_shadow = Some(x.clone());
389                    e.outer_shadow = None;
390                }
391                Some(x) => {
392                    e.outer_shadow = Some(x.clone());
393                    e.inner_shadow = None;
394                }
395                None => {
396                    e.outer_shadow = None;
397                    e.inner_shadow = None;
398                }
399            }
400        }
401        if let Some(g) = &glow {
402            e.glow.clone_from(g);
403        }
404        if let Some(sf) = soft {
405            e.soft_edge = sf;
406        }
407        if let Some(r) = &refl {
408            e.reflection.clone_from(r);
409        }
410        if bevel.is_some() || three_d.is_some() {
411            let mut s3 = e.raw3d.as_deref().map(crate::engine::cmd::three_d::Shape3d::parse).unwrap_or_default();
412            if let Some(b) = &bevel {
413                s3.apply(&json!({"bevelTop": b.as_deref().unwrap_or("none")}));
414            }
415            if let Some(t) = three_d {
416                if t.get("reset").and_then(Value::as_bool) == Some(true) {
417                    s3 = crate::engine::cmd::three_d::Shape3d::default();
418                }
419                s3.apply(t);
420            }
421            crate::engine::cmd::three_d::store(e, &s3);
422        }
423        if let Some(v) = &shadow_props {
424            let slot = if e.inner_shadow.is_some() { &mut e.inner_shadow } else { &mut e.outer_shadow };
425            let mut x = slot.take().or_else(|| shadow_preset(&json!("outer")).flatten()).ok_or_else(|| bad("shape.effects", "no shadow preset"))?;
426            if let Some(c) = v.get("color").and_then(color_value) {
427                let a = x.color.alpha();
428                x.color = with_alpha(c, a);
429            }
430            if let Some(t) = num(v, "transparency") {
431                x.color = with_alpha(x.color.clone(), 1.0 - t.clamp(0.0, 1.0));
432            }
433            if let Some(b) = num(v, "blur") {
434                x.blur = b.clamp(0.0, 100.0);
435            }
436            if let Some(d) = num(v, "dist") {
437                x.dist = d.clamp(0.0, 200.0);
438            }
439            if let Some(d) = num(v, "dir") {
440                x.dir = d.rem_euclid(360.0);
441            }
442            // PowerPoint's Size: 0–200%, both axes; a zero scale would collapse the shadow's matrix.
443            if let Some(z) = num(v, "size") {
444                let z = z.clamp(0.01, 2.0);
445                x.sx = z;
446                x.sy = z;
447            }
448            *slot = Some(x);
449        }
450        if let Some(v) = &glow_props {
451            let mut g =
452                e.glow.take().unwrap_or(Glow { color: ColorRef::scheme(SchemeSlot::Accent1).with(ColorTransform::Alpha(40000)), radius: 8.0 });
453            if let Some(c) = v.get("color").and_then(color_value) {
454                let a = g.color.alpha();
455                g.color = with_alpha(c, a);
456            }
457            if let Some(t) = num(v, "transparency") {
458                g.color = with_alpha(g.color.clone(), 1.0 - t.clamp(0.0, 1.0));
459            }
460            if let Some(r) = num(v, "radius") {
461                g.radius = r.clamp(0.0, 150.0);
462            }
463            e.glow = (g.radius > 0.0).then_some(g);
464        }
465        if let Some(v) = &refl_props {
466            let mut r = e.reflection.take().unwrap_or(Reflection {
467                blur: 0.5,
468                start_alpha: 0.5,
469                end_alpha: 0.0,
470                end_pos: 0.35,
471                dist: 0.0,
472                ..Default::default()
473            });
474            if let Some(t) = num(v, "transparency") {
475                r.start_alpha = 1.0 - t.clamp(0.0, 1.0);
476            }
477            if let Some(sz) = num(v, "size") {
478                r.end_pos = sz.clamp(0.0, 1.0);
479            }
480            if let Some(d) = num(v, "dist") {
481                r.dist = d.clamp(0.0, 100.0);
482            }
483            if let Some(b) = num(v, "blur") {
484                r.blur = b.clamp(0.0, 100.0);
485            }
486            e.reflection = Some(r);
487        }
488        Ok(())
489    })
490}

492/// A finite number at `key` of an effect's properties.
493fn num(v: &Value, key: &str) -> Option<f64> {
494    v.get(key).and_then(Value::as_f64).filter(|x| x.is_finite())
495}
496
497/// `c` with its alpha set to `a` (0–1); fully opaque drops the transform.
498fn with_alpha(mut c: ColorRef, a: f64) -> ColorRef {
499    c.mods.retain(|m| !matches!(m, ColorTransform::Alpha(_)));
500    if a < 1.0 {
501        c.mods.push(ColorTransform::Alpha((a.clamp(0.0, 1.0) * 100_000.0).round() as i32));
502    }
503    c
504}

506/// Theme quick styles: 7 rows (outline only, solid fill variants, intense…) × 6 accents.
507pub fn quick_style_for(index: usize) -> (ShapeStyle, Option<Fill>, Option<Line>) {
508    let slots = [
509        SchemeSlot::Dk1,
510        SchemeSlot::Accent1,
511        SchemeSlot::Accent2,
512        SchemeSlot::Accent3,
513        SchemeSlot::Accent4,
514        SchemeSlot::Accent5,
515        SchemeSlot::Accent6,
516    ];
517    let slot = slots.get(index % 7).copied().unwrap_or(SchemeSlot::Accent1);
518    let row = index / 7;
519    let mut st = ShapeStyle::accent(slot);
520    let c = ColorRef::scheme(slot);
521    match row {
522        0 => {
523            // Colored outline, light fill.
524            st.fill_ref = (1, ColorRef::scheme(SchemeSlot::Lt1));
525            st.line_ref = (2, c);
526            st.font_ref.1 = Some(ColorRef::scheme(SchemeSlot::Dk1));
527            (st, None, None)
528        }
529        1 => (st, None, None),
530        2 => {
531            st.fill_ref = (1, c.clone().with(ColorTransform::LumMod(20000)).with(ColorTransform::LumOff(80000)));
532            st.line_ref = (0, c.clone());
533            st.font_ref.1 = Some(c.with(ColorTransform::LumMod(50000)));
534            (st, None, Some(Line::none()))
535        }
536        3 => {
537            st.fill_ref = (2, c);
538            (st, None, Some(Line::none()))
539        }
540        4 => {
541            st.fill_ref = (3, c.clone());
542            st.effect_ref = (3, c);
543            (st, None, Some(Line::none()))
544        }
545        _ => {
546            st.fill_ref = (1, c.clone().with(ColorTransform::LumMod(75000)));
547            st.line_ref = (1, ColorRef::scheme(SchemeSlot::Lt1));
548            (st, None, None)
549        }
550    }
551}

553pub(super) fn quick_style(s: &mut Session, p: &Value) -> Result<Value> {
554    let i = opt_usize_param(p, "index", "shape.quickStyle")?.unwrap_or(1).min(41);
555    let (st, f, l) = quick_style_for(i);
556    edit_shapes(s, p, "shape.quickStyle", |sh| {
557        sh.style = Some(st.clone());
558        sh.fill.clone_from(&f);
559        sh.line.clone_from(&l);
560        sh.effects = None;
561        if let Some(t) = sh.text.as_mut() {
562            crate::model::edit::format_all(t, &|r| r.fill = None);
563        }
564        Ok(())
565    })
566}
