3pub(super) fn sites(s: &mut Session, p: &Value) -> Result<Value> {
4    let id = id_param(p, "id", "shape.sites")?.ok_or_else(|| bad("shape.sites", "missing `id`"))?;
5    let st = s.doc()?;
6    // In slide space, as shape.connect uses them, for a shape inside a group too.
7    let v = crate::engine::connect::sites_of(&st.doc, &st.selection, st.shapes(), id).ok_or_else(|| bad("shape.sites", "no such shape"))?;
8    Ok(json!(v.iter().map(|q| [q.x, q.y]).collect::<Vec<_>>()))
9}

11pub(super) fn connect(s: &mut Session, p: &Value) -> Result<Value> {
12    let (Some(from), Some(to)) = (id_param(p, "from", "shape.connect")?, id_param(p, "to", "shape.connect")?) else {
13        return Err(bad("shape.connect", "missing `from` / `to`"));
14    };
15    let st = s.doc()?;
16    // Anywhere in the slide's tree: a shape inside a group takes a connector too (PowerPoint
17    // glues to grouped shapes), its sites where the group draws them.
18    let site_list = |id| {
19        crate::engine::connect::sites_of(&st.doc, &st.selection, st.shapes(), id)
20            .ok_or_else(|| bad("shape.connect", format!("no shape {} on the current slide", id.0)))
21    };
22    let (a, b) = (site_list(from)?, site_list(to)?);
23    if a.is_empty() || b.is_empty() {
24        return Err(bad("shape.connect", "both shapes need connection sites (lines and groups have none)"));
25    }
26    let pick = |v: &[crate::geom::Point], key: &str| usize_param(p, key).filter(|i| *i < v.len());
27    let (ia, ib) = match (pick(&a, "fromSite"), pick(&b, "toSite")) {
28        (Some(i), Some(j)) => (i, j),
29        (fi, fj) => {
30            let mut best = (f64::MAX, 0, 0);
31            for (i, pa) in a.iter().enumerate().filter(|(i, _)| fi.is_none_or(|f| f == *i)) {
32                for (j, pb) in b.iter().enumerate().filter(|(j, _)| fj.is_none_or(|f| f == *j)) {
33                    let d = (*pa - *pb).hypot();
34                    if d < best.0 {
35                        best = (d, i, j);
36                    }
37                }
38            }
39            (best.1, best.2)
40        }
41    };
42    let (p0, p1) = (a[ia], b[ib]);
43    let id = match id_param(p, "id", "shape.connect")? {
44        // The ends are in slide space and rerouting follows the slide's own connectors, so a line
45        // inside a group would be drawn in the group's space and never follow its shapes.
46        Some(id) if !st.shapes().iter().any(|sh| sh.id == id) && st.shape(id).is_some_and(|sh| sh.is_line()) => {
47            return Err(bad("shape.connect", "`id` names a line inside a group; ungroup it first, or leave `id` out for a new connector"));
48        }
49        Some(id) => id,
50        None => {
51            let preset = str_param(p, "preset").unwrap_or("straightConnector1");
52            let r = s.execute("shape.insert", &json!({"preset": preset, "rect": [p0.x, p0.y, 1.0, 1.0]}))?;
53            crate::model::ShapeId(r.get("id").and_then(Value::as_u64).ok_or_else(|| bad("shape.connect", "insert failed"))? as u32)
54        }
55    };
56    s.edit(|doc, sel| {
57        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("shape.connect", "no slide"))?;
58        let sh = crate::model::find_shape_mut(list, id).ok_or_else(|| bad("shape.connect", "no such line"))?;
59        if !sh.is_line() {
60            return Err(bad("shape.connect", "`id` is not a line or connector"));
61        }
62        sh.xfrm = Some(crate::engine::tools::line_xfrm(p0, p1, 0.0));
63        sh.kind = crate::model::ShapeKind::Connector { start: Some((from, ia as u32)), end: Some((to, ib as u32)) };
64        Ok(())
65    })?;
66    Ok(json!({"id": id.0, "fromSite": ia, "toSite": ib}))
67}

69/// Path data through `pts` (shape-local): straight segments, or a Catmull-Rom curve as cubics.
70pub fn freeform_path(pts: &[crate::geom::Point], closed: bool, smooth: bool) -> String {
71    let f = |v: f64| {
72        let s = format!("{:.2}", v);
73        s.trim_end_matches('0').trim_end_matches('.').to_string()
74    };
75    let mut d = format!("M {} {}", f(pts[0].x), f(pts[0].y));
76    let n = pts.len();
77    if smooth && n > 2 {
78        let at = |i: isize| -> crate::geom::Point {
79            if closed { pts[i.rem_euclid(n as isize) as usize] } else { pts[i.clamp(0, n as isize - 1) as usize] }
80        };
81        let segs = if closed { n } else { n - 1 };
82        for i in 0..segs as isize {
83            let (p0, p1, p2, p3) = (at(i - 1), at(i), at(i + 1), at(i + 2));
84            let c1 = p1 + (p2 - p0) / 6.0;
85            let c2 = p2 - (p3 - p1) / 6.0;
86            d += &format!(" C {} {} {} {} {} {}", f(c1.x), f(c1.y), f(c2.x), f(c2.y), f(p2.x), f(p2.y));
87        }
88    } else {
89        for q in &pts[1..] {
90            d += &format!(" L {} {}", f(q.x), f(q.y));
91        }
92    }
93    if closed {
94        d += " Z";
95    }
96    d
97}

99pub(super) fn freeform(s: &mut Session, p: &Value) -> Result<Value> {
100    let pts: Vec<crate::geom::Point> = p
101        .get("points")
102        .and_then(Value::as_array)
103        .map(|a| {
104            a.iter()
105                .filter_map(|v| Some(crate::geom::Point::new(v.get(0)?.as_f64()?, v.get(1)?.as_f64()?)))
106                .filter(|q| q.x.is_finite() && q.y.is_finite())
107                .take(crate::engine::tools::MAX_GESTURE_POINTS)
108                .collect()
109        })
110        .unwrap_or_default();
111    if pts.len() < 2 {
112        return Err(bad("shape.freeform", "need at least two points"));
113    }
114    let closed = bool_or(p, "closed", false) && pts.len() > 2;
115    let smooth = bool_or(p, "smooth", false);
116    let mut bb = crate::geom::Rect::from_points(pts[0], pts[0]);
117    for q in &pts {
118        bb = bb.union_pt(*q);
119    }
120    let (w, h) = (bb.width().max(1.0), bb.height().max(1.0));
121    let local: Vec<crate::geom::Point> = pts.iter().map(|q| crate::geom::Point::new(q.x - bb.x0, q.y - bb.y0)).collect();
122    let d = freeform_path(&local, closed, smooth);
123    // Closed shapes take the default shape style, open paths the line style.
124    let r = s.execute("shape.insert", &json!({"preset": if closed { "rect" } else { "line" }, "rect": [bb.x0, bb.y0, w, h]}))?;
125    let id = crate::model::ShapeId(r.get("id").and_then(Value::as_u64).ok_or_else(|| bad("shape.freeform", "insert failed"))? as u32);
126    s.edit(|doc, sel| {
127        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("shape.freeform", "no slide"))?;
128        let sh = crate::model::find_shape_mut(list, id).ok_or_else(|| bad("shape.freeform", "insert failed"))?;
129        sh.geom = Geom::Custom {
130            paths: vec![crate::model::CustomPath {
131                w,
132                h,
133                d: d.clone(),
134                fill: if closed { crate::geom::preset::FillMode::Norm } else { crate::geom::preset::FillMode::None },
135                stroke: true,
136            }],
137        };
138        sh.xfrm = Some(crate::geom::Xfrm::new(bb.x0, bb.y0, w, h));
139        sh.kind = crate::model::ShapeKind::Shape;
140        sh.name = format!("Freeform {}", id.0);
141        Ok(())
142    })?;
143    Ok(json!({"id": id.0}))
144}

146/// Draw › Ink to Shape: each recognisable stroke of the slide's ink becomes a preset shape
147/// outlined in the stroke's colour and width; the rest stays ink. One undo step.
148pub(super) fn ink_to_shape(s: &mut Session, _p: &Value) -> Result<Value> {
149    use crate::engine::inkshape::{Recognized, recognize};
150    use crate::model::{Shape, ShapeId, ShapeKind};
151    let shapes = s.doc()?.shapes();
152    let Some(ink) = shapes.iter().find(|sh| matches!(sh.kind, ShapeKind::Ink { .. })) else {
153        return Ok(json!({"shapes": [], "kept": 0}));
154    };
155    let (ink_id, off) = (ink.id, ink.xfrm.map(|x| (x.x, x.y)).unwrap_or((0.0, 0.0)));
156    let ShapeKind::Ink { strokes } = &ink.kind else { return Ok(json!({"shapes": [], "kept": 0})) };
157    let found: Vec<Option<Recognized>> = strokes
158        .iter()
159        .map(|st| if st.highlighter { None } else { recognize(&st.points.iter().map(|(x, y, _)| (x + off.0, y + off.1)).collect::<Vec<_>>()) })
160        .collect();
161    if found.iter().all(Option::is_none) {
162        return Ok(json!({"shapes": [], "kept": strokes.len()}));
163    }
164    s.edit(|doc, sel| {
165        let mut new = vec![];
166        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("draw.inkToShape", "no slide"))?;
167        let Some(pos) = list.iter().position(|sh| sh.id == ink_id) else { return Err(bad("draw.inkToShape", "ink gone")) };
168        let ShapeKind::Ink { strokes } = &mut list[pos].kind else { return Err(bad("draw.inkToShape", "not ink")) };
169        let all = std::mem::take(strokes);
170        let mut kept = vec![];
171        let mut made = vec![];
172        for (st, r) in all.into_iter().zip(found.iter()) {
173            let Some(r) = r else {
174                kept.push(st);
175                continue;
176            };
177            let (preset, xfrm) = match *r {
178                Recognized::Line { x0, y0, x1, y1 } => {
179                    ("line", crate::engine::tools::line_xfrm(crate::geom::Point::new(x0, y0), crate::geom::Point::new(x1, y1), 0.0))
180                }
181                Recognized::Rect { x, y, w, h } => ("rect", crate::geom::Xfrm::new(x, y, w, h)),
182                Recognized::Ellipse { x, y, w, h } => ("ellipse", crate::geom::Xfrm::new(x, y, w, h)),
183                Recognized::Triangle { x, y, w, h, flip_v } => ("triangle", crate::geom::Xfrm { flip_v, ..crate::geom::Xfrm::new(x, y, w, h) }),
184            };
185            made.push((preset, xfrm, st.color, st.width));
186        }
187        let remaining = kept.len();
188        *strokes = kept;
189        if remaining == 0 {
190            list.remove(pos);
191        }
192        for (preset, xfrm, color, width) in made {
193            let id = ShapeId(doc.alloc_id().ok_or_else(|| bad("draw.inkToShape", NO_ID_SPACE))?);
194            let name = format!(
195                "{} {}",
196                match preset {
197                    "line" => "Straight Connector",
198                    "rect" => "Rectangle",
199                    "ellipse" => "Oval",
200                    _ => "Isosceles Triangle",
201                },
202                id.0
203            );
204            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("draw.inkToShape", "no slide"))?;
205            list.push(Shape {
206                id,
207                name,
208                xfrm: Some(xfrm),
209                geom: Geom::preset(preset),
210                fill: Some(Fill::None),
211                line: Some(Line::solid(ColorRef::rgb(color), width.clamp(0.25, 100.0))),
212                kind: ShapeKind::Shape,
213                ..Default::default()
214            });
215            new.push(id);
216        }
217        sel.shapes = new.clone();
218        Ok(json!({"shapes": new.iter().map(|i| i.0).collect::<Vec<_>>(), "kept": remaining}))
219    })
