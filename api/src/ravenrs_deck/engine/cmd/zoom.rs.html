1//! Insert ▸ Zoom: Summary Zoom, Section Zoom and Slide Zoom.
2//!
3//! A zoom is a picture of the target slide that jumps to it in the slide show. With Return to
4//! Zoom (on for summary and section zooms, as in PowerPoint) the show comes back to the zoom
5//! slide after the target slide, or the target section's last slide. In .pptx a zoom is saved as
6//! the linked picture PowerPoint itself writes as the fallback for older versions.
7

31/// One zoom tile: the slide it shows and jumps to, and the slide after which the show returns.
32struct Target {
33    slide: SlideId,
34    index: usize,
35    last: SlideId,
36}

38/// `n` tiles of `aspect` (w / h) laid out in a centred grid inside `area`.
39fn grid(n: usize, area: Xfrm, aspect: f64) -> Vec<Xfrm> {
40    let cols = (n as f64).sqrt().ceil().max(1.0) as usize;
41    let rows = n.div_ceil(cols).max(1);
42    let gap = 12.0;
43    let cw = (area.w - gap * (cols as f64 - 1.0)) / cols as f64;
44    let ch = (area.h - gap * (rows as f64 - 1.0)) / rows as f64;
45    let (w, h) = if cw / ch > aspect { (ch * aspect, ch) } else { (cw, cw / aspect) };
46    // A single slide zoom is a quarter of the slide wide, as PowerPoint inserts it.
47    let (w, h) = if n == 1 { (w.min(area.w / 2.0), h.min(area.w / 2.0 / aspect)) } else { (w, h) };
48    let total_w = cols.min(n) as f64 * w + gap * (cols.min(n) as f64 - 1.0);
49    let total_h = rows as f64 * h + gap * (rows as f64 - 1.0);
50    let (x0, y0) = (area.x + (area.w - total_w) / 2.0, area.y + (area.h - total_h) / 2.0);
51    (0..n).map(|k| Xfrm::new(x0 + (k % cols) as f64 * (w + gap), y0 + (k / cols) as f64 * (h + gap), w, h)).collect()
52}

54fn indices(p: &Value, key: &str, n: usize, what: &str) -> Result<Vec<usize>> {
55    let Some(Value::Array(a)) = p.get(key) else { return Ok(vec![]) };
56    let mut v = vec![];
57    for x in a {
58        let i = x
59            .as_u64()
60            .map(|i| i as usize)
61            .filter(|i| *i < n)
62            .ok_or_else(|| bad("insert.zoom", format!("there is no {what} {x} (indices count from 0)")))?;
63        if !v.contains(&i) {
64            v.push(i);
65        }
66    }
67    Ok(v)
68}

70fn zoom(s: &mut Session, p: &Value) -> Result<Value> {
71    let kind = str_param(p, "kind").unwrap_or("slide");
72    let st = s.doc()?;
73    let doc = &st.doc;
74    let n = doc.slides.len();
75    let targets: Vec<Target> = match kind {
76        "slide" => indices(p, "slides", n, "slide")?
77            .into_iter()
78            .filter_map(|i| doc.slides.get(i).map(|sl| Target { slide: sl.id, index: i, last: sl.id }))
79            .collect(),
80        "section" => {
81            let secs = indices(p, "sections", doc.sections.len(), "section")?;
82            secs.into_iter()
83                .filter_map(|k| {
84                    let sec = doc.sections.get(k)?;
85                    let first = *sec.slides.first()?;
86                    let last = *sec.slides.last()?;
87                    Some(Target { slide: first, index: doc.slide_index(first)?, last })
88                })
89                .collect()
90        }
91        "summary" => {
92            let mut starts = indices(p, "slides", n, "slide")?;
93            if starts.is_empty() {
94                // Default: the first slide of every section, or every slide's start when there are none.
95                starts = doc.sections.iter().filter_map(|sec| sec.slides.first().and_then(|id| doc.slide_index(*id))).collect();
96            }
97            starts.sort_unstable();
98            starts
99                .iter()
100                .enumerate()
101                .filter_map(|(k, &i)| {
102                    // Each part runs to the slide before the next part (or the end).
103                    let end = starts.get(k + 1).map(|j| j.saturating_sub(1)).unwrap_or(n.saturating_sub(1)).max(i);
104                    Some(Target { slide: doc.slides.get(i)?.id, index: i, last: doc.slides.get(end)?.id })
105                })
106                .collect()
107        }
108        other => return Err(bad("insert.zoom", format!("unknown zoom kind `{other}` (slide, section or summary)"))),
109    };
110    if targets.is_empty() {
111        return Err(bad("insert.zoom", "choose the slides (or sections) to zoom to"));
112    }
113    let back = bool_or(p, "returnToZoom", kind != "slide");
114    let size = doc.slide_size;
115    // Pictures of the targets, about 480 px wide.
116    let scale = (480.0 / size.width.max(1.0)).clamp(0.05, 4.0);
117    let pictures: Vec<(Vec<u8>, String)> = targets
118        .iter()
119        .map(|t| {
120            let (png, _, _) = super::file::render_png(doc, t.index, scale, false);
121            let title = doc.slides.get(t.index).map(|sl| sl.title()).unwrap_or_default();
122            let label =
123                if title.trim().is_empty() { format!("Slide {}", t.index + 1) } else { title.lines().next().unwrap_or("").trim().to_string() };
124            (png, label)
125        })
126        .collect();
127    let summary = kind == "summary";
128    let first_target = targets.iter().map(|t| t.index).min().unwrap_or(0);
129    s.edit(|doc, sel| {
130        let aspect = size.width / size.height.max(1.0);
131        let area = if summary {
132            Xfrm::new(size.width * 0.06, size.height * 0.26, size.width * 0.88, size.height * 0.66)
133        } else if targets.len() == 1 {
134            Xfrm::new(size.width * 0.5, size.height * 0.5, size.width * 0.45, size.height * 0.45)
135        } else {
136            Xfrm::new(size.width * 0.08, size.height * 0.12, size.width * 0.84, size.height * 0.8)
137        };
138        let boxes = grid(targets.len(), area, aspect);
139        let mut shapes = vec![];
140        for ((t, (png, label)), b) in targets.iter().zip(pictures).zip(boxes) {
141            let media = doc.add_media(&format!("zoom-{}.png", t.index + 1), "image/png", png).ok_or_else(|| bad("insert.zoom", NO_ID_SPACE))?;
142            let id = ShapeId(doc.alloc_id().ok_or_else(|| bad("insert.zoom", NO_ID_SPACE))?);
143            shapes.push(Shape {
144                id,
145                name: format!(
146                    "{} Zoom {}",
147                    if summary {
148                        "Summary"
149                    } else if kind == "section" {
150                        "Section"
151                    } else {
152                        "Slide"
153                    },
154                    t.index + 1
155                ),
156                descr: label.clone(),
157                xfrm: Some(b),
158                kind: ShapeKind::Picture { fill: PictureFill { media, ..Default::default() } },
159                click: Some(Hyperlink {
160                    action: Action::Slide { slide: t.slide },
161                    tooltip: label,
162                    highlight_click: false,
163                    zoom_return: back.then_some(t.last),
164                    sound: None,
165                    stop_sound: false,
166                }),
167                ..Default::default()
168            });
169        }
170        let ids: Vec<ShapeId> = shapes.iter().map(|x| x.id).collect();
171        if summary {
172            // A new Title Only slide before the first part, as PowerPoint makes it.
173            let layout =
174                defaults::layout_of_kind(doc, LayoutType::TitleOnly).or_else(|| doc.slides.get(first_target).map(|x| x.layout)).unwrap_or_default();
175            let mut slide = defaults::new_slide(doc, layout).ok_or_else(|| bad("insert.zoom", NO_ID_SPACE))?;
176            if let Some(sh) = slide.shapes.iter_mut().find(|x| x.ph_type().is_some_and(|k| k.is_title())) {
177                sh.text = Some(TextBody::from_text("Summary"));
178            }
179            slide.shapes.extend(shapes);
180            let at = first_target.min(doc.slides.len());
181            doc.slides.insert(at, Arc::new(slide));
182            sel.slide = at;
183            sel.slides.clear();
184        } else {
185            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("insert.zoom", "no slide"))?;
186            list.extend(shapes);
187        }
188        sel.shapes = ids.clone();
189        sel.text = None;
190        Ok(json!({"kind": kind, "ids": ids, "returnToZoom": back, "slide": sel.slide}))
191    })
192}
