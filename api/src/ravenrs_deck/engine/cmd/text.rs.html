137/// The text body a text selection edits.
138pub fn body_of<'a>(st: &'a DocState, t: &TextSel) -> Option<&'a TextBody> {
139    if t.notes {
140        return st.current_slide().map(|s| &s.notes);
141    }
142    let sh = st.shape(t.shape)?;
143    if let (Some((r, c)), ShapeKind::Table(tb)) = (t.cell, &sh.kind) {
144        return tb.cell(r, c).map(|x| &x.text);
145    }
146    sh.text.as_ref()
147}
148
149fn body_mut<'a>(doc: &'a mut Presentation, sel: &Selection, t: &TextSel) -> Option<&'a mut TextBody> {
150    if t.notes {
151        return doc.slides.get_mut(sel.slide).map(|s| &mut Arc::make_mut(s).notes);
152    }
153    let shapes = crate::engine::shapes_mut(doc, sel)?;
154    let sh = crate::model::find_shape_mut(shapes, t.shape)?;
155    if let Some((r, c)) = t.cell {
156        if let ShapeKind::Table(tb) = &mut sh.kind {
157            return tb.cell_mut(r, c).map(|x| &mut x.text);
158        }
159        return None;
160    }
161    Some(sh.text.get_or_insert_with(TextBody::default))
162}

164/// Edit the active text body and selection; refits auto-sizing text boxes afterwards.
165pub(crate) fn edit_text<T>(s: &mut Session, f: impl FnOnce(&mut TextBody, &mut TextSel) -> T) -> Result<T> {
166    let t0 = s.doc()?.selection.text.clone().ok_or_else(|| bad("text", NO_TEXT_POINT))?;
167    // Materialise an inherited placeholder box before editing (so autofit can change it).
168    let r = s.edit(|doc, sel| {
169        let mut t = t0.clone();
170        let body = body_mut(doc, sel, &t).ok_or_else(|| bad("text", "the text being edited is gone"))?;
171        let r = f(body, &mut t);
172        // The saved shrink-on-overflow scale was for the old text: fit the new text afresh. A
173        // body whose file carried no `fontScale` keeps none (100 %) here; `fit_shrink_scale`
174        // below fills in the computed scale when the new text overflows (#724).
175        if let Some(AutoFit::Shrink { font_scale, line_reduction }) = &mut body.body.autofit {
176            if font_scale.is_some() {
177                *font_scale = Some(1.0);
178            }
179            *line_reduction = 0.0;
180        }
181        sel.text = Some(t);
182        Ok(r)
183    })?;
184    if !t0.notes && t0.cell.is_none() {
185        fit_text_box(s, t0.shape)?;
186    }
187    Ok(r)
188}
