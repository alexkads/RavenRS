69fn class_of(s: Option<&str>) -> Option<AnimClass> {
70    s.map(|c| match c {
71        "emphasis" | "emph" => AnimClass::Emphasis,
72        "exit" => AnimClass::Exit,
73        "path" | "motion" | "motionPath" => AnimClass::Path,
74        "media" => AnimClass::Media,
75        _ => AnimClass::Entrance,
76    })

87/// The effect option `given` as one of `allowed`. Besides the stored codes ("b", "lt", "u", …)
88/// it takes the direction words people type: "fromBottom", "bottom", "fromTopLeft",
89/// "upRight". `None` when it matches none of them: the renderer reads direction codes letter
90/// by letter, so an unchecked "fromBottom" would fly in from the right and the bottom.
91fn resolve_option(allowed: &[&str], given: &str) -> Option<String> {
92    if allowed.is_empty() || allowed.contains(&given) {
93        return Some(given.to_string());
94    }
95    let low = given.to_ascii_lowercase();
96    if let Some(o) = allowed.iter().find(|o| o.to_ascii_lowercase() == low) {
97        return Some(o.to_string());
98    }
99    let mut rest = low.strip_prefix("from").unwrap_or(&low).trim_start_matches(['-', '_', ' ']);
100    // Direction words in order, each as both its edge letter and its arrow letter.
101    let mut dirs: Vec<(char, char)> = Vec::new();
102    while !rest.is_empty() {
103        let words: [(&str, (char, char)); 8] = [
104            ("bottom", ('b', 'd')),
105            ("down", ('b', 'd')),
106            ("top", ('t', 'u')),
107            ("up", ('t', 'u')),
108            ("left", ('l', 'l')),
109            ("right", ('r', 'r')),
110            ("-", ('-', '-')),
111            ("_", ('-', '-')),
112        ];
113        let (w, d) = words.iter().find(|(w, _)| rest.starts_with(w))?;
114        if d.0 != '-' {
115            dirs.push(*d);
116        }
117        rest = &rest[w.len()..];
118    }
119    if dirs.is_empty() || dirs.len() > 2 {
120        return None;
121    }
122    let mut cands: Vec<String> = Vec::new();
123    for pick in [|d: &(char, char)| d.0, |d: &(char, char)| d.1] {
124        let code: String = dirs.iter().map(pick).collect();
125        let rev: String = code.chars().rev().collect();
126        cands.push(code);
127        cands.push(rev);
128    }
129    cands.into_iter().find(|c| allowed.contains(&c.as_str()))
130}

132fn build(p: &Value, slide: (f64, f64)) -> Result<Animation> {
133    let effect = str_param(p, "effect").ok_or_else(|| bad("animation", "missing `effect`"))?;
134    let class = class_of(str_param(p, "class"));
135    let info = match class {
136        Some(c) => animation_info(effect, c),
137        None => ANIMATIONS.iter().find(|a| a.0 == effect),
138    }
139    .ok_or_else(|| bad("animation", format!("unknown effect `{effect}`")))?;
140    let mut a = Animation {
141        class: info.2,
142        effect: info.0.to_string(),
143        option: match str_param(p, "option") {
144            Some(o) => {
145                resolve_option(info.5, o).ok_or_else(|| bad("animation", format!("option `{o}` is not one of {:?} for `{}`", info.5, info.0)))?
146            }
147            None => info.5.first().map(|o| o.to_string()).unwrap_or_default(),
148        },
149        duration_ms: p.get("duration").and_then(Value::as_u64).map(|v| v.min(600_000) as u32).unwrap_or(info.4),
150        delay_ms: p.get("delay").and_then(Value::as_u64).map(|v| v.min(600_000) as u32).unwrap_or(0),
151        preset_id: Some(info.3),
152        ..Default::default()
153    };
154    if let Some(st) = str_param(p, "start") {
155        a.start = start_of(st);
156    }
157    if a.class == AnimClass::Path {
158        a.path = Some(path_param(p, slide)?.unwrap_or_else(|| default_path(&a.effect, &a.option)));
159    } else if p.get("path").is_some() {
160        return Err(bad("animation", "`path` only fits a motion path (class `path`)"));
161    }
162    if a.effect == "spin" {
163        a.amount = Some(360.0);
164    }
165    if a.effect == "growShrink" {
166        a.amount = Some(1.5);
167    }
168    Ok(a)
169}

171/// Most points of a motion path given as an array, and longest path string given as text.
172const MAX_PATH_POINTS: usize = 1_000;
173const MAX_PATH_TEXT: usize = 16 * 1024;
174/// Largest offset of a point, in slide points; anything beyond is not a place on any slide.
175const MAX_PATH_OFFSET: f64 = 1.0e6;
176
177/// A number of a path string: slide fractions, rounded to 6 places.
178fn path_num(v: f64) -> String {
179    let t = format!("{:.6}", v);
180    let t = t.trim_end_matches('0').trim_end_matches('.');
181    if t == "-0" || t.is_empty() { "0".into() } else { t.into() }
182}

184/// Does a raw motion path string draw: only commands and numbers, every number finite, and at
185/// least two points? (A lone `e`, or `1e999`, would play as an empty or partial path.)
186fn raw_path_draws(text: &str) -> bool {
187    let cs: Vec<char> = text.chars().collect();
188    let (mut i, mut nums) = (0, 0usize);
189    while let Some(&c) = cs.get(i) {
190        if c.is_ascii_digit() || c == '.' || ((c == '-' || c == '+') && cs.get(i + 1).is_some_and(|d| d.is_ascii_digit() || *d == '.')) {
191            let start = i;
192            i += 1;
193            while cs.get(i).is_some_and(|d| d.is_ascii_digit() || *d == '.') {
194                i += 1;
195            }
196            if cs.get(i).is_some_and(|d| *d == 'e' || *d == 'E') {
197                let n1 = cs.get(i + 1).copied().unwrap_or(' ');
198                let n2 = cs.get(i + 2).copied().unwrap_or(' ');
199                if n1.is_ascii_digit() || ((n1 == '-' || n1 == '+') && n2.is_ascii_digit()) {
200                    i += 2;
201                    while cs.get(i).is_some_and(char::is_ascii_digit) {
202                        i += 1;
203                    }
204                }
205            }
206            let tok: String = cs.get(start..i).map(|t| t.iter().collect()).unwrap_or_default();
207            match tok.parse::<f64>() {
208                Ok(v) if v.is_finite() => nums += 1,
209                _ => return false,
210            }
211        } else if " ,MmLlCcQqHhVvZzE".contains(c) {
212            i += 1;
213        } else {
214            return false;
215        }
216    }
217    nums >= 4
218}

220/// The motion path `path` asks for, as the OOXML path string animations store: either points
221/// (`[[x, y], …]`, slide points as offsets from where the shape sits, joined by lines, `closed?`
222/// to return to the first) or that string itself.
223fn path_param(p: &Value, slide: (f64, f64)) -> Result<Option<String>> {
224    let Some(v) = p.get("path") else { return Ok(None) };
225    let refuse = |why: &str| bad("animation", format!("`path` {why}"));
226    match v {
227        Value::String(text) => {
228            let ok = text.len() <= MAX_PATH_TEXT
229                && text.contains(|c: char| c.is_ascii_digit())
230                && text.chars().all(|c| c.is_ascii_digit() || " .+-eE,MmLlCcQqHhVvZz".contains(c))
231                && raw_path_draws(text);
232            if !ok {
233                return Err(refuse("string must be an OOXML motion path such as `M 0 0 L 0.25 0 E` (at most 16 KiB)"));
234            }
235            Ok(Some(text.clone()))
236        }
237        Value::Array(points) => {
238            if !(2..=MAX_PATH_POINTS).contains(&points.len()) {
239                return Err(refuse(&format!("needs 2 to {MAX_PATH_POINTS} points")));
240            }
241            if !(slide.0.is_finite() && slide.1.is_finite() && slide.0 > 0.0 && slide.1 > 0.0) {
242                return Err(refuse("can't be placed: the slide has no size"));
243            }
244            let mut out = String::new();
245            for (k, pt) in points.iter().enumerate() {
246                let xy = pt.as_array().filter(|a| a.len() == 2).and_then(|a| Some((a.first()?.as_f64()?, a.get(1)?.as_f64()?)));
247                let Some((x, y)) = xy.filter(|(x, y)| x.abs() <= MAX_PATH_OFFSET && y.abs() <= MAX_PATH_OFFSET) else {
248                    return Err(refuse(&format!("point {k} must be [x, y], two finite numbers within ±{MAX_PATH_OFFSET} points")));
249                };
250                out.push_str(&format!("{} {} {} ", if k == 0 { "M" } else { "L" }, path_num(x / slide.0), path_num(y / slide.1)));
251            }
252            if p.get("closed").and_then(Value::as_bool) == Some(true) {
253                out.push_str("Z ");
254            }
255            out.push('E');
256            Ok(Some(out))
257        }
258        _ => Err(refuse("must be [[x, y], …] or an OOXML path string")),
259    }
260}

262/// Default motion paths, in slide-fraction units relative to the shape's position.
263pub fn default_path(effect: &str, option: &str) -> String {
264    let (dx, dy) = match option {
265        "up" => (0.0, -0.25),
266        "right" => (0.25, 0.0),
267        "left" => (-0.25, 0.0),
268        _ => (0.0, 0.25),
269    };
270    match effect {
271        "arcs" => format!("M 0 0 C {} {} {} {} {dx} {dy} E", dx * 0.1 + dy * 0.5, dy * 0.1 - dx * 0.5, dx * 0.9 + dy * 0.5, dy * 0.9 - dx * 0.5),
272        "turns" => format!("M 0 0 L {} {} C {} {} {dx} {} {dx} {dy} E", dx * 0.5, dy * 0.5, dx, dy * 0.5, dy * 0.75),
273        "shapes" => "M 0 0 C 0.1 0 0.1 0.15 0 0.15 C -0.1 0.15 -0.1 0 0 0 Z E".into(),
274        "loops" => "M 0 0 C 0.1 -0.1 0.2 0.1 0.1 0.1 C 0 0.1 0.1 -0.1 0.25 0 E".into(),
275        other => crate::model::anim::preset_path(other).unwrap_or_else(|| format!("M 0 0 L {dx} {dy} E")),
276    }
277}

279fn set(s: &mut Session, p: &Value) -> Result<Value> {
280    let ids = targets(s, p)?;
281    let a = build(p, slide_extent(s)?)?;
282    s.edit(|doc, sel| {
283        let sl = doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation.set", "no slide"))?;
284        let sl = Arc::make_mut(sl);
285        // Replace each shape's animations, keeping their place in the sequence.
286        let mut out = vec![];
287        let mut placed = vec![];
288        for x in sl.animations.drain(..) {
289            if ids.contains(&x.shape) {
290                if !placed.contains(&x.shape) {
291                    placed.push(x.shape);
292                    out.push(Animation { shape: x.shape, start: x.start, ..a.clone() });
293                }
294            } else {
295                out.push(x);
296            }
297        }
298        for id in &ids {
299            if !placed.contains(id) {
300                out.push(Animation { shape: *id, ..a.clone() });
301            }
302        }
303        if a.effect == "none" {
304            out.retain(|x| !ids.contains(&x.shape));
305        }
306        sl.animations = out;
307        Ok(json!({"count": sl.animations.len()}))
308    })
309}

317fn add(s: &mut Session, p: &Value) -> Result<Value> {
318    let ids = targets(s, p)?;
319    let a = build(p, slide_extent(s)?)?;
320    let path = a.path.clone();
321    s.edit(|doc, sel| {
322        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation.add", "no slide"))?);
323        for (k, id) in ids.iter().enumerate() {
324            let mut x = Animation { shape: *id, ..a.clone() };
325            if k > 0 {
326                x.start = AnimStart::WithPrevious;
327            }
328            sl.animations.push(x);
329        }
330        Ok(match &path {
331            Some(path) => json!({"count": sl.animations.len(), "path": path}),
332            None => json!({"count": sl.animations.len()}),
333        })
334    })
335}

337fn remove(s: &mut Session, p: &Value) -> Result<Value> {
338    let index = opt_usize_param(p, "index", "animation.remove")?;
339    let ids = if index.is_none() { targets(s, p)? } else { vec![] };
340    s.edit(|doc, sel| {
341        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation.remove", "no slide"))?);
342        match index {
343            Some(i) if i < sl.animations.len() => {
344                sl.animations.remove(i);
345            }
346            Some(_) => return Err(bad("animation.remove", "no such animation")),
347            None => sl.animations.retain(|a| !ids.contains(&a.shape)),
348        }
349        Ok(())
350    })?;
351    ok()
352}

354fn move_anim(s: &mut Session, p: &Value) -> Result<Value> {
355    let i = opt_usize_param(p, "index", "animation.move")?.ok_or_else(|| bad("animation.move", "missing `index`"))?;
356    let to = usize_param(p, "to").ok_or_else(|| bad("animation.move", "missing `to`"))?;
357    s.edit(|doc, sel| {
358        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation.move", "no slide"))?);
359        if i >= sl.animations.len() {
360            return Err(bad("animation.move", "no such animation"));
361        }
362        let a = sl.animations.remove(i);
363        let to = to.min(sl.animations.len());
364        sl.animations.insert(to, a);
365        Ok(())
366    })?;
367    ok()
368}

370fn each_target(s: &mut Session, p: &Value, f: impl Fn(&mut Animation)) -> Result<Value> {
371    let index = opt_usize_param(p, "index", "animation")?;
372    let ids = if index.is_none() { targets(s, p)? } else { vec![] };
373    s.edit(|doc, sel| {
374        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation", "no slide"))?);
375        let mut n = 0;
376        for (k, a) in sl.animations.iter_mut().enumerate() {
377            if index == Some(k) || (index.is_none() && ids.contains(&a.shape)) {
378                f(a);
379                n += 1;
380            }
381        }
382        if n == 0 {
383            return Err(bad("animation", "no animation to change (select an animated object or give `index`)"));
384        }
385        Ok(json!({"changed": n}))
386    })
387}

389fn timing(s: &mut Session, p: &Value) -> Result<Value> {
390    let start = str_param(p, "start").map(start_of);
391    let dur = p.get("duration").and_then(Value::as_u64).map(|v| v.min(600_000) as u32);
392    let delay = p.get("delay").and_then(Value::as_u64).map(|v| v.min(600_000) as u32);
393    let repeat = p.get("repeat").and_then(Value::as_u64).map(|v| v.min(u32::MAX as u64) as u32);
394    let rewind = bool_param(p, "rewind");
395    let trigger = p.get("trigger").map(|v| v.as_u64().and_then(|x| u32::try_from(x).ok()).map(crate::model::ShapeId));
396    each_target(s, p, |a| {
397        if let Some(x) = start {
398            a.start = x;
399        }
400        if let Some(x) = dur {
401            a.duration_ms = x;
402        }
403        if let Some(x) = delay {
404            a.delay_ms = x;
405        }
406        if let Some(x) = repeat {
407            a.repeat = x;
408        }
409        if let Some(x) = rewind {
410            a.rewind = x;
411        }
412        if let Some(t) = trigger {
413            a.trigger = t;
414        }
415    })
416}

418fn options(s: &mut Session, p: &Value) -> Result<Value> {
419    let option = str_param(p, "option").map(String::from);
420    let build_mode = str_param(p, "textBuild").map(|b| match b {
421        "byParagraph" => TextBuild::ByParagraph,
422        "byWord" => TextBuild::ByWord,
423        "byLetter" => TextBuild::ByLetter,
424        _ => TextBuild::AsOne,
425    });
426    let path = path_param(p, slide_extent(s)?)?;
427    each_target(s, p, |a| {
428        if let (Some(path), AnimClass::Path) = (&path, a.class) {
429            a.path = Some(path.clone());
430        }
431        if let Some(o) = &option {
432            // Each target takes the option in its own effect's terms; one it has no such
433            // option for keeps its own.
434            let allowed = animation_info(&a.effect, a.class).map(|i| i.5).unwrap_or(&[]);
435            if let Some(o) = resolve_option(allowed, o) {
436                if a.class == AnimClass::Path && path.is_none() {
437                    a.path = Some(default_path(&a.effect, &o));
438                }
439                a.option = o;
440            }
441        }
442        if let Some(b) = build_mode {
443            a.text_build = b;
444        }
445    })
446}

473fn painter_pick(s: &mut Session, p: &Value) -> Result<Value> {
474    let st = s.doc()?;
475    let id = id_param(p, "id", "animation.painter")?
476        .or_else(|| st.selection.shapes.first().copied())
477        .ok_or_else(|| bad("animation.painter", "select an animated shape first"))?;
478    let sl = st.current_slide().ok_or_else(|| bad("animation.painter", "no slide"))?;
479    let anims: Vec<Animation> = sl.animations.iter().filter(|a| a.shape == id).cloned().collect();
480    if anims.is_empty() {
481        return Err(bad("animation.painter", format!("shape {id} has no animations to copy")));
482    }
483    let sticky = bool_or(p, "sticky", false);
484    let count = anims.len();
485    s.anim_painter = Some((anims, sticky));
486    Ok(json!({"armed": true, "count": count}))
487}

489fn painter_apply(s: &mut Session, p: &Value) -> Result<Value> {
490    let Some((painted, sticky)) = s.anim_painter.clone() else {
491        return Err(bad("animation.painterApply", "pick animations with the Animation Painter first"));
492    };
493    if !sticky {
494        s.anim_painter = None;
495    }
496    let ids = targets(s, p)?;
497    if ids.is_empty() {
498        return Err(bad("animation.painterApply", "no target shapes selected"));
499    }
500    s.edit(|doc, sel| {
501        let sl = Arc::make_mut(doc.slides.get_mut(sel.slide).ok_or_else(|| bad("animation.painterApply", "no slide"))?);
502        // As in PowerPoint, painting replaces the target's own animations (painting a shape onto
503        // itself doesn't double them). A paragraph effect keeps its paragraph only if the target
504        // has that paragraph; otherwise it animates the whole shape.
505        sl.animations.retain(|a| !ids.contains(&a.shape));
506        let mut added = 0;
507        for id in &ids {
508            let paras = crate::model::find_shape(&sl.shapes, *id).and_then(|sh| sh.text.as_ref()).map_or(0, |t| t.paragraphs.len());
509            for a in &painted {
510                let mut cloned = a.clone();
511                cloned.shape = *id;
512                if cloned.paragraph.is_some_and(|i| i as usize >= paras) {
513                    cloned.paragraph = None;
514                }
515                sl.animations.push(cloned);
516                added += 1;
517            }
518        }
519        Ok(json!({"applied": added}))
520    })

563        // Insert two shapes

565        let id1 = s1.get("id").unwrap().as_u64().unwrap();

569        // Add 2 animations to shape 1: fade (entrance) and spin (emphasis)
570        s.execute("animation.add", &json!({"effect": "fade", "class": "entrance", "ids": [id1]})).unwrap();
571        s.execute("animation.add", &json!({"effect": "spin", "class": "emphasis", "ids": [id1]})).unwrap();
572
573        // Shape 2 initially has no animations
574        let anims_before = s.execute("animation.get", &json!({})).unwrap();
575        let arr_before = anims_before.as_array().unwrap();
576        assert_eq!(arr_before.len(), 2);
577        assert!(arr_before.iter().all(|a| a.get("shape").unwrap().as_u64().unwrap() == id1));
578
579        // Pick animations from shape 1
580        s.select(|_, sel| {
581            sel.shapes = vec![crate::model::ShapeId(id1 as u32)];
582        })
583        .unwrap();
584        let res = s.execute("animation.painter", &json!({})).unwrap();
585        assert_eq!(res.get("armed").unwrap().as_bool(), Some(true));
586        assert_eq!(res.get("count").unwrap().as_u64(), Some(2));
587        assert!(s.anim_painter.is_some());
588
589        // Apply animations to shape 2
590        s.select(|_, sel| {
591            sel.shapes = vec![crate::model::ShapeId(id2 as u32)];
592        })
593        .unwrap();
594        let applied = s.execute("animation.painterApply", &json!({})).unwrap();
595        assert_eq!(applied.get("applied").unwrap().as_u64(), Some(2));
596
597        // Single-click should disarm the painter
598        assert!(s.anim_painter.is_none());
599
600        // Painting again replaces shape 2's animations instead of adding to them; undo restores.
601        s.execute("animation.painter", &json!({"id": id1})).unwrap();
602        s.execute("animation.painterApply", &json!({"ids": [id2]})).unwrap();
603        let count_on = |s: &mut Session, id: u64| {
604            let all = s.execute("animation.get", &json!({})).unwrap();
605            all.as_array().unwrap().iter().filter(|a| a.get("shape").unwrap().as_u64().unwrap() == id).count()
606        };
607        assert_eq!(count_on(&mut s, id2), 2);
608        s.execute("edit.undo", &json!({})).unwrap();
609        assert_eq!(count_on(&mut s, id2), 2, "undo restores the first painting");
610        s.execute("edit.undo", &json!({})).unwrap();
611        assert_eq!(count_on(&mut s, id2), 0, "undo removes the painted animations");
612        s.execute("edit.redo", &json!({})).unwrap();
613
614        // Verify shape 2 now has the 2 animations cloned
615        let anims_after = s.execute("animation.get", &json!({})).unwrap();
616        let arr_after = anims_after.as_array().unwrap();
617        assert_eq!(arr_after.len(), 4);
618        let s2_anims: Vec<_> = arr_after.iter().filter(|a| a.get("shape").unwrap().as_u64().unwrap() == id2).collect();
619        assert_eq!(s2_anims.len(), 2);
620        assert_eq!(s2_anims[0].get("effect").unwrap().as_str(), Some("fade"));
621        assert_eq!(s2_anims[1].get("effect").unwrap().as_str(), Some("spin"));
622
623        // Picking from shape with no animations returns an error
624        s.execute("animation.clear", &json!({})).unwrap();
625        let err = s.execute("animation.painter", &json!({"id": id1}));
626        assert!(err.is_err());
627    }
