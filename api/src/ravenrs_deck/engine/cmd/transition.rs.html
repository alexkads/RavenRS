47fn slide_indices(s: &Session, p: &Value, cmd: &str) -> Result<Vec<usize>> {
48    let st = s.doc()?;
49    if let Some(i) = opt_usize_param(p, "index", cmd)? {
50        return Ok(vec![i]);
51    }
52    let v: Vec<usize> = st.selection.slides.iter().filter_map(|id| st.doc.slide_index(*id)).collect();
53    Ok(if v.is_empty() { vec![st.selection.slide] } else { v })
54}

56fn set(s: &mut Session, p: &Value) -> Result<Value> {
57    let kind = str_param(p, "kind").ok_or_else(|| bad("transition.set", "missing `kind`"))?;
58    let info =
59        TRANSITIONS.iter().find(|t| t.0.eq_ignore_ascii_case(kind)).ok_or_else(|| bad("transition.set", format!("unknown transition `{kind}`")))?;
60    let option = str_param(p, "option").map(String::from).or_else(|| info.4.first().map(|o| o.to_string())).unwrap_or_default();
61    let duration = p.get("duration").and_then(Value::as_u64).map(|v| v.min(59_000) as u32);
62    let list = slide_indices(s, p, "transition.set")?;
63    s.edit(|doc, _| {
64        for i in list {
65            if let Some(sl) = doc.slides.get_mut(i) {
66                let sl = Arc::make_mut(sl);
67                let mut t = sl.transition.clone().unwrap_or_default();
68                t.kind = info.0.to_string();
69                t.option = option.clone();
70                t.duration_ms = duration.unwrap_or(info.3);
71                t.raw = None;
72                sl.transition = if info.0 == "none" && t.advance_after_ms.is_none() && t.advance_on_click && t.sound.is_none() && !t.stop_sound {
73                    None
74                } else {
75                    Some(t)
76                };
77            }
78        }
79        Ok(())
80    })?;
81    Ok(json!({"kind": info.0}))
82}

84fn options(s: &mut Session, p: &Value) -> Result<Value> {
85    let option = str_param(p, "option").ok_or_else(|| bad("transition.options", "missing `option`"))?.to_string();
86    let list = slide_indices(s, p, "transition.options")?;
87    s.edit(|doc, _| {
88        for i in list {
89            if let Some(sl) = doc.slides.get_mut(i)
90                && let Some(t) = Arc::make_mut(sl).transition.as_mut()
91            {
92                t.option = option.clone();
93            }
94        }
95        Ok(())
96    })?;
97    ok()
98}

100fn timing(s: &mut Session, p: &Value) -> Result<Value> {
101    let list = slide_indices(s, p, "transition.timing")?;
102    s.edit(|doc, _| {
103        for i in list {
104            if let Some(sl) = doc.slides.get_mut(i) {
105                let t = Arc::make_mut(sl).transition.get_or_insert_with(Transition::default);
106                if let Some(d) = p.get("duration").and_then(Value::as_u64) {
107                    t.duration_ms = d.min(59_000) as u32;
108                }
109                if let Some(c) = bool_param(p, "onClick") {
110                    t.advance_on_click = c;
111                }
112                match p.get("after") {
113                    Some(Value::Null) => t.advance_after_ms = None,
114                    Some(v) => {
115                        if let Some(x) = v.as_u64() {
116                            t.advance_after_ms = Some(x.min(86_400_000) as u32);
117                        }
118                    }
119                    None => {}
120                }
121            }
122        }
123        Ok(())
124    })?;
125    ok()
126}

128/// Transitions › Sound: embed an audio file as the slides' transition sound, or `none` to remove it.
129fn sound(s: &mut Session, p: &Value) -> Result<Value> {
130    let list = slide_indices(s, p, "transition.sound")?;
131    let looping = bool_param(p, "loop");
132    let edit_all = |s: &mut Session, f: &dyn Fn(&mut Transition)| {
133        s.edit(|doc, _| {
134            for &i in &list {
135                if let Some(sl) = doc.slides.get_mut(i) {
136                    f(Arc::make_mut(sl).transition.get_or_insert_with(Transition::default));
137                }
138            }
139            Ok(())
140        })
141    };
142    if bool_param(p, "none") == Some(true) {
143        edit_all(s, &|t| {
144            t.sound = None;
145            t.stop_sound = false;
146            t.sound_loop = false;
147        })?;
148        return Ok(json!({"sound": null}));
149    }
150    if bool_param(p, "stopPrevious") == Some(true) {
151        edit_all(s, &|t| {
152            t.sound = None;
153            t.sound_loop = false;
154            t.stop_sound = true;
155        })?;
156        return Ok(json!({"sound": null, "stopPrevious": true}));
157    }
158    let wants_file = p.get("path").is_some() || p.get("data").is_some() || p.get("builtin").is_some();
159    if !wants_file {
160        // Only Loop Until Next Sound changes.
161        let Some(l) = looping else { return Err(bad("transition.sound", "give `path`, `data`, `builtin`, `none`, `stopPrevious` or `loop`")) };
162        edit_all(s, &|t| t.sound_loop = l && t.sound.is_some())?;
163        return Ok(json!({"loop": l}));
164    }
165    let (name, bytes) = match str_param(p, "builtin") {
166        Some(id) => {
167            let (id, label) = crate::engine::sounds::BUILTIN.iter().find(|b| b.0.eq_ignore_ascii_case(id)).ok_or_else(|| {
168                let ids: Vec<&str> = crate::engine::sounds::BUILTIN.iter().map(|b| b.0).collect();
169                bad("transition.sound", format!("unknown built-in sound `{id}` (one of {})", ids.join(", ")))
170            })?;
171            let bytes = crate::engine::sounds::builtin(id).ok_or_else(|| bad("transition.sound", "built-in sound failed"))?;
172            (format!("{label}.wav"), bytes)
173        }
174        None => super::insert::media_bytes(p, "transition.sound")?,
175    };
176    let ct = super::insert::content_type(&name, &bytes);
177    let bytes = Arc::new(bytes);
178    let audible = crate::media::probe(&bytes).is_ok_and(|i| i.audio.is_some());
179    let bytes = Arc::try_unwrap(bytes).unwrap_or_else(|a| (*a).clone());
180    if !audible && !ct.starts_with("audio/") {
181        return Err(bad("transition.sound", format!("{name} is not a sound file ({ct})")));
182    }
183    let lp = looping.unwrap_or(false);
184    let media = s.edit(|doc, _| {
185        // The same built-in sound is embedded once.
186        let existing = doc.media.iter().find(|m| m.name == name && m.data.as_slice() == bytes.as_slice()).map(|m| m.id);
187        let media = match existing {
188            Some(m) => m,
189            None => doc.add_media(&name, ct, bytes).ok_or_else(|| bad("transition.sound", NO_ID_SPACE))?,
190        };
191        for &i in &list {
192            if let Some(sl) = doc.slides.get_mut(i) {
193                let t = Arc::make_mut(sl).transition.get_or_insert_with(Transition::default);
194                t.sound = Some(media);
195                t.stop_sound = false;
196                t.sound_loop = lp;
197            }
198        }
199        Ok(media)
200    })?;
201    Ok(json!({"sound": name, "media": media.0, "playable": audible, "loop": lp}))
202}

204fn apply_all(s: &mut Session, _p: &Value) -> Result<Value> {
205    s.edit(|doc, sel| {
206        let t = doc.slides.get(sel.slide).and_then(|x| x.transition.clone());
207        for sl in &mut doc.slides {
208            Arc::make_mut(sl).transition.clone_from(&t);
209        }
210        Ok(())
211    })?;
212    ok()
213}

215fn list(_s: &mut Session, _p: &Value) -> Result<Value> {
216    Ok(Value::Array(TRANSITIONS.iter().map(|t| json!({"id": t.0, "label": t.1, "category": t.2, "duration": t.3, "options": t.4})).collect()))
217}

226    fn invalid_index_params_are_rejected_not_defaulted_to_selection() {
227        let mut s = Session::with_new();
228        while s.doc().expect("doc").doc.slides.len() < 3 {
229            s.execute("slide.new", &json!({})).expect("new slide");
230        }
231        for bad in [json!(-1), json!(1.5)] {
232            for (cmd, p) in [
233                ("transition.set", json!({"kind": "fade", "index": bad})),
234                ("transition.options", json!({"option": "x", "index": bad})),
235                ("transition.timing", json!({"duration": 500, "index": bad})),
236                ("transition.sound", json!({"none": true, "index": bad})),
237            ] {
238                assert!(s.execute(cmd, &p).is_err(), "{cmd} must reject {p}");
239                assert!(s.doc().expect("doc").doc.slides.iter().all(|sl| sl.transition.is_none()), "{cmd} {p}");
240            }
241        }
242    }
