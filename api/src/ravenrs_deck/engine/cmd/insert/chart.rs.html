6pub fn chart_type(name: &str) -> ChartType {
7    match name.to_ascii_lowercase().replace(['-', '_', ' '], "").as_str() {
8        "bar" | "clusteredbar" => ChartType::Bar,
9        "stackedbar" => ChartType::StackedBar,
10        "percentbar" => ChartType::PercentBar,
11        "line" => ChartType::Line,
12        "linemarkers" | "linewithmarkers" => ChartType::LineMarkers,
13        "stackedline" => ChartType::StackedLine,
14        "area" => ChartType::Area,
15        "stackedarea" => ChartType::StackedArea,
16        "pie" => ChartType::Pie,
17        "doughnut" | "donut" => ChartType::Doughnut,
18        "scatter" | "xy" => ChartType::Scatter,
19        "bubble" => ChartType::Bubble,
20        "radar" => ChartType::Radar,
21        "filledradar" => ChartType::FilledRadar,
22        "stackedcolumn" => ChartType::StackedColumn,
23        "percentcolumn" => ChartType::PercentColumn,
24        "waterfall" => ChartType::Waterfall,
25        "funnel" => ChartType::Funnel,
26        "histogram" => ChartType::Histogram,
27        "pareto" => ChartType::Pareto,
28        "treemap" => ChartType::Treemap,
29        "sunburst" => ChartType::Sunburst,
30        "boxwhisker" | "box" => ChartType::BoxWhisker,
31        "stock" => ChartType::Stock,
32        "surface" => ChartType::Surface,
33        "combo" => ChartType::Combo,
34        _ => ChartType::Column,
35    }
36}

38pub(super) fn chart(s: &mut Session, p: &Value) -> Result<Value> {
39    let kind = chart_type(str_param(p, "type").unwrap_or("column"));
40    let mut c = Chart::sample(kind);
41    if let Some(cats) = p.get("categories").and_then(Value::as_array) {
42        c.categories = cats.iter().map(|v| v.as_str().map(String::from).unwrap_or_else(|| v.to_string())).collect();
43    }
44    if let Some(series) = p.get("series").and_then(Value::as_array) {
45        c.series = series
46            .iter()
47            .map(|sv| crate::model::chart::Series {
48                name: sv.get("name").and_then(Value::as_str).unwrap_or("Series").to_string(),
49                values: sv.get("values").and_then(Value::as_array).map(|a| a.iter().map(Value::as_f64).collect()).unwrap_or_default(),
50                ..Default::default()
51            })
52            .collect();
53    }
54    if let Some(t) = p.get("title") {
55        c.title = t.as_str().map(String::from);
56    }
57    let size = s.doc()?.doc.slide_size;
58    let rect = rect_param(p, "rect").unwrap_or(Xfrm::new(size.width * 0.15, size.height * 0.15, size.width * 0.7, size.height * 0.7));
59    let id = add_shape(s, "insert.chart", Shape { xfrm: Some(rect), kind: ShapeKind::Chart(Box::new(c)), ..Default::default() }, true)?;
60    Ok(json!({"id": id}))
61}

68fn with_chart(s: &mut Session, p: &Value, cmd: &str, f: impl Fn(&mut Chart) -> Result<()>) -> Result<Value> {
69    edit_shapes(s, p, cmd, |sh| match &mut sh.kind {
70        ShapeKind::Chart(c) => f(c),
71        _ => Err(bad(cmd, "not a chart")),
72    })
73}

75pub(super) fn chart_type_cmd(s: &mut Session, p: &Value) -> Result<Value> {
76    let k = chart_type(str_param(p, "type").unwrap_or("column"));
77    with_chart(s, p, "chart.type", |c| {
78        c.kind = k;
79        c.vary_colors =
80            matches!(k, ChartType::Pie | ChartType::Doughnut | ChartType::Funnel | ChartType::Treemap | ChartType::Sunburst | ChartType::Waterfall)
81                || c.series.len() == 1 && c.vary_colors;
82        if matches!(k, ChartType::Waterfall | ChartType::Histogram | ChartType::Pareto | ChartType::BoxWhisker) {
83            c.gridlines = true;
84        }
85        c.raw = None;
86        Ok(())
87    })
88}

90pub(super) fn chart_data(s: &mut Session, p: &Value) -> Result<Value> {
91    let cats: Option<Vec<String>> = p
92        .get("categories")
93        .and_then(Value::as_array)
94        .map(|a| a.iter().map(|v| v.as_str().map(String::from).unwrap_or_else(|| v.to_string())).collect());
95    let series: Option<Vec<crate::model::chart::Series>> = p.get("series").and_then(Value::as_array).map(|a| {
96        a.iter()
97            .map(|sv| crate::model::chart::Series {
98                name: sv.get("name").and_then(Value::as_str).unwrap_or("Series").to_string(),
99                values: sv.get("values").and_then(Value::as_array).map(|v| v.iter().map(Value::as_f64).collect()).unwrap_or_default(),
100                ..Default::default()
101            })
102            .collect()
103    });
104    let levels: Option<Vec<Vec<String>>> = p.get("levels").and_then(Value::as_array).map(|a| {
105        a.iter()
106            .take(crate::model::chart::MAX_LEVELS)
107            .map(|l| {
108                l.as_array().map(|v| v.iter().map(|x| x.as_str().map(String::from).unwrap_or_else(|| x.to_string())).collect()).unwrap_or_default()
109            })
110            .collect()
111    });
112    with_chart(s, p, "chart.data", |c| {
113        if let Some(cs) = &cats {
114            c.categories = cs.clone();
115        }
116        if let Some(l) = &levels {
117            c.category_levels = l.clone();
118        }
119        if let Some(se) = &series {
120            // Keep series formatting by position.
121            let old = std::mem::take(&mut c.series);
122            c.series = se
123                .iter()
124                .enumerate()
125                .map(|(i, n)| {
126                    let mut x = old.get(i).cloned().unwrap_or_default();
127                    x.name = n.name.clone();
128                    x.values = n.values.clone();
129                    x
130                })
131                .collect();
132        }
133        c.raw = None;
134        Ok(())
135    })
136}

138fn apply_bin_options(c: &mut crate::model::chart::Chart, bin_count: Option<Option<u32>>, bin_size: Option<Option<f64>>) {
139    if let Some(n) = bin_count {
140        c.bin_count = n;
141        if n.is_some() {
142            c.bin_size = None;
143        }
144    }
145    if let Some(w) = bin_size {
146        c.bin_size = w;
147        if w.is_some() {
148            c.bin_count = None;
149        }
150    }
151}

153fn apply_subtotals_and_bars(c: &mut crate::model::chart::Chart, p: &Value, subtotals: Option<&[usize]>) {
154    if let Some(t) = subtotals {
155        let n = c.series.first().map_or(0, |s| s.values.len());
156        let mut t: Vec<usize> = t.iter().copied().filter(|i| *i < n).collect();
157        t.sort_unstable();
158        t.dedup();
159        c.subtotals = t;
160    }
161    if let Some(b) = p.get("upDownBars").and_then(Value::as_bool) {
162        c.up_down_bars = b;
163        c.raw = None;
164    }
165    if let Some(q) = p.get("quartiles").and_then(Value::as_str) {
166        c.inclusive_quartiles = q.eq_ignore_ascii_case("inclusive");
167        c.raw = None;
168    }
169}

171pub(super) fn chart_options(s: &mut Session, p: &Value) -> Result<Value> {
172    // Histogram bins: a count or a width (setting one clears the other; null for automatic).
173    let bin_count = p.get("binCount").map(|v| v.as_u64().map(|n| n.clamp(1, crate::model::chart::MAX_BINS as u64) as u32));
174    let bin_size = p.get("binSize").map(|v| v.as_f64().filter(|w| w.is_finite() && *w > 0.0));
175    if p.get("binSize").is_some_and(|v| !v.is_null()) && bin_size.flatten().is_none() {
176        return Err(bad("chart.options", "binSize must be a positive number"));
177    }
178    // Waterfall total points.
179    let subtotals: Option<Vec<usize>> =
180        p.get("subtotals").and_then(Value::as_array).map(|a| a.iter().filter_map(Value::as_u64).map(|i| i as usize).collect());
181    // Colour of the chart's text (title, axes, legend, data labels); null goes back to PowerPoint's
182    // default (tx1 at lumMod 65 % / lumOff 35 %).
183    let text_color = match p.get("textColor") {
184        None => None,
185        Some(Value::Null) => Some(None),
186        Some(v) => {
187            Some(Some(color_value(v).ok_or_else(|| bad("chart.options", "`textColor` is a colour: #RRGGBB, accent1…accent6, tx1…, or null"))?))
188        }
189    };
190    with_chart(s, p, "chart.options", |c| {
191        if let Some(t) = &text_color {
192            c.text_color = t.clone();
193            c.legend_entries.iter_mut().for_each(|e| e.color = None);
194        }
195        apply_bin_options(c, bin_count, bin_size);
196        apply_subtotals_and_bars(c, p, subtotals.as_deref());
197        if bin_count.is_some() || bin_size.is_some() || subtotals.is_some() {
198            c.raw = None;
199        }
200        if let Some(t) = p.get("title") {
201            c.title = t.as_str().map(String::from);
202        }
203        if let Some(l) = p.get("legend") {
204            c.legend = l.as_str().map(String::from);
205        }
206        if let Some(v) = bool_param(p, "dataLabels") {
207            c.data_labels = v;
208            // A series' own setting would otherwise keep overriding the chart-wide switch.
209            for series in &mut c.series {
210                series.data_labels = None;
211            }
212        }
213        if let Some(v) = bool_param(p, "gridlines") {
214            c.gridlines = v;
215        }
216        if let Some(v) = str_param(p, "palette") {
217            c.palette = v.to_string();
218        }
219        c.raw = None;
220        Ok(())
221    })
