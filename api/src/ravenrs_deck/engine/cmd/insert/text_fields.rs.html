6pub(super) fn word_art(s: &mut Session, p: &Value) -> Result<Value> {
7    let text = str_param(p, "text").unwrap_or("Your text here");
8    let size = s.doc()?.doc.slide_size;
9    let style = usize_param(p, "style").unwrap_or(0);
10    let mut shape = new_text_box(Xfrm::new(size.width * 0.2, size.height * 0.4, size.width * 0.6, 80.0), text);
11    if let Some(t) = shape.text.as_mut() {
12        for para in &mut t.paragraphs {
13            para.props.align = Some(crate::model::text::Align::Center);
14            for r in &mut para.runs {
15                r.props.size = Some(54.0);
16                r.props.bold = Some(style % 2 == 1);
17                r.props.fill = Some(Fill::solid(ColorRef::scheme(match style % 4 {
18                    0 => SchemeSlot::Tx1,
19                    1 => SchemeSlot::Accent1,
20                    2 => SchemeSlot::Accent2,
21                    _ => SchemeSlot::Accent4,
22                })));
23                if style % 3 == 2 {
24                    r.props.outline = Some(crate::model::Line::solid(ColorRef::scheme(SchemeSlot::Bg1), 1.0));
25                }
26            }
27        }
28    }
29    let id = add_shape(s, "insert.wordArt", shape, true)?;
30    crate::engine::cmd::text::fit_text_box(s, id)?;
31    Ok(json!({"id": id}))
32}

34fn field_run(s: &mut Session, field: &str, text: &str) -> Result<Value> {
35    if s.doc()?.selection.text.is_some() {
36        return crate::engine::cmd::text::insert_field(s, field, text);
37    }
38    // Not editing: turn on the footer field for the slide (Header & Footer).
39    let key = if field == "slidenum" { "slideNumber" } else { "date" };
40    s.execute("design.headerFooter", &json!({key: true}))
41}

63struct DateOpts {
64    format: String,
65    update: bool,
66    /// BCP 47 tag when the caller set `language`; absent keeps the surrounding run's language.
67    lang: Option<String>,
68    sample: String,

71/// `true` when the call carries a format, update flag or language (apply it). Empty params open
72/// the dialog instead.
73fn date_time_apply(p: &Value) -> bool {
74    p.as_object().is_some_and(|o| o.contains_key("format") || o.contains_key("update") || o.contains_key("language"))
75}

77fn canon_lang(raw: &str) -> Option<&'static str> {
78    match raw.trim().to_ascii_lowercase().replace('_', "-").as_str() {
79        "en" | "en-us" => Some("en-US"),
80        "en-gb" => Some("en-GB"),
81        "fr" | "fr-fr" => Some("fr-FR"),
82        "ja" | "ja-jp" => Some("ja-JP"),
83        "sv" | "sv-se" => Some("sv-SE"),
84        _ => None,
85    }
86}

88fn date_opts(p: &Value) -> Result<DateOpts> {
89    if p.get("format").is_some() {
90        let f = str_param(p, "format").filter(|f| DATE_FORMATS.contains(f));
91        if f.is_none() {
92            return Err(bad("insert.dateTime", "`format` must be datetime1..datetime13"));
93        }
94    }
95    if p.get("update").is_some() && bool_param(p, "update").is_none() {
96        return Err(bad("insert.dateTime", "`update` must be true or false"));
97    }
98    let lang = if p.get("language").is_some() {
99        let raw = str_param(p, "language").filter(|l| !l.trim().is_empty());
100        let Some(raw) = raw else {
101            return Err(bad("insert.dateTime", "`language` must be en-US, en-GB, fr-FR, ja-JP or sv-SE"));
102        };
103        Some(canon_lang(raw).ok_or_else(|| bad("insert.dateTime", "`language` must be en-US, en-GB, fr-FR, ja-JP or sv-SE"))?.to_string())
104    } else {
105        None
106    };
107    let format = str_param(p, "format").unwrap_or("datetime1").to_string();
108    let sample = date_sample(&format, lang.as_deref())?;
109    Ok(DateOpts { format, update: bool_or(p, "update", true), lang, sample })
110}

112/// The text `format` shows now in `lang`. Errors when that language has no names for the format
113/// (month names and AM/PM are English only). An empty string where there is no clock (wasm).
114fn date_sample(format: &str, lang: Option<&str>) -> Result<String> {
115    use crate::text::datetime::{DateTime, field_text};
116    let probe = DateTime::new(2026, 10, 9, 20, 5, 7);
117    if field_text(format, lang, probe).is_none() {
118        return Err(bad("insert.dateTime", "this format isn't available in that language"));
119    }
120    Ok(DateTime::now_local().and_then(|now| field_text(format, lang, now)).unwrap_or_default())
121}

123pub(super) fn date_time(s: &mut Session, p: &Value) -> Result<Value> {
124    if !date_time_apply(p) {
125        s.ui_requests.push(crate::engine::UiRequest::Dialog { id: "dateTime".into(), params: json!({}) });
126        return ok();
127    }
128    let opts = date_opts(p)?;
129    let (doc, sel) = {
130        let st = s.doc()?;
131        (st.doc.clone(), st.selection.clone())
132    };
133    let r = apply_date(s, &opts);
134    if r.is_err()
135        && let Some(st) = s.active_mut()
136    {
137        st.doc = doc;
138        st.selection = sel;
139    }
140    r
141}

143fn apply_date(s: &mut Session, o: &DateOpts) -> Result<Value> {
144    if s.doc()?.selection.text.is_some() {
145        if o.update {
146            crate::engine::cmd::text::insert_field(s, &o.format, &o.sample)?;
147            if let Some(lang) = &o.lang {
148                stamp_caret_lang(s, lang)?;
149            }
150        } else {
151            insert_plain(s, &o.sample, o.lang.as_deref())?;
152        }
153        return Ok(json!({"where": "caret", "format": o.format, "update": o.update, "text": o.sample}));
154    }
155    // Handout or Notes Master view: the master's date placeholder (the selected one, else the
156    // master's own), which every handout or notes page then shows.
157    if s.doc()?.selection.target.is_page_master() {
158        return write_master_date(s, o);
159    }
160    // Not editing: the slide's date placeholder (this slide only). A layout with no date
161    // placeholder gets a text box instead, so the chosen format is still inserted.
162    s.execute("design.headerFooter", &json!({"date": true, "all": false}))?;
163    if !write_date_placeholder(s, o)? {
164        return insert_date_box(s, o);
165    }
166    Ok(json!({"where": "footer", "format": o.format, "update": o.update, "text": o.sample}))
167}

169/// Put the date in the date placeholder of the handout or notes master being edited.
170fn write_master_date(s: &mut Session, o: &DateOpts) -> Result<Value> {
171    let (format, sample, lang, update) = (o.format.clone(), o.sample.clone(), o.lang.clone(), o.update);
172    let has_date = {
173        let st = s.doc()?;
174        crate::engine::page_master(&st.doc, st.selection.target)
175            .is_some_and(|m| m.shapes.iter().any(|x| x.ph_type() == Some(crate::model::PhType::Date)))
176    };
177    if !has_date {
178        return Err(bad(
179            "insert.dateTime",
180            "this master has no date placeholder; show it with handout.placeholders or notes.placeholders {date: true}",
181        ));
182    }
183    // The text the master holds afterwards, which the answer reports (it can differ from the
184    // caller's sample when the placeholder's own language formats a fixed date).
185    let mut written = o.sample.clone();
186    s.edit(|doc, sel| {
187        let Some(m) = crate::engine::page_master_mut(doc, sel.target) else { return Ok(false) };
188        let m = std::sync::Arc::make_mut(m);
189        let date = |x: &crate::model::Shape| x.ph_type() == Some(crate::model::PhType::Date);
190        let chosen = sel.shapes.iter().find_map(|id| m.shapes.iter().position(|x| x.id == *id && date(x)));
191        let Some(i) = chosen.or_else(|| m.shapes.iter().position(date)) else { return Ok(false) };
192        let Some(sh) = m.shapes.get_mut(i) else { return Ok(false) };
193        let mut props = sh.text.as_ref().and_then(|t| t.paragraphs.first()).and_then(|p| p.runs.first()).map(|r| r.props.clone()).unwrap_or_default();
194        // A fixed date reads in the placeholder's own language when the caller named none.
195        let text = match (&lang, &props.lang) {
196            (None, Some(l)) if !update => date_sample(&format, Some(l)).unwrap_or_else(|_| sample.clone()),
197            _ => sample.clone(),
198        };
199        if let Some(lang) = lang {
200            props.lang = Some(lang);
201        }
202        let kind = if update { RunKind::Field { field: format } } else { RunKind::Text };
203        let body = sh.text.clone().unwrap_or_default();
204        let mut para = body.paragraphs.first().cloned().unwrap_or_default();
205        written.clone_from(&text);
206        para.runs = vec![crate::model::Run { text, props, kind }];
207        sh.text = Some(TextBody { paragraphs: vec![para], ..body });
208        Ok(true)
209    })?;
210    Ok(json!({"where": "master", "format": o.format, "update": o.update, "text": written}))
211}

213/// Point the current slide's date placeholder at this format (a field, or the sample as plain text).
214fn write_date_placeholder(s: &mut Session, o: &DateOpts) -> Result<bool> {
215    let format = o.format.clone();
216    let sample = o.sample.clone();
217    let lang = o.lang.clone();
218    let update = o.update;
219    s.edit(|doc, sel| {
220        let Some(sl) = doc.slides.get_mut(sel.slide) else { return Ok(false) };
221        let sl = std::sync::Arc::make_mut(sl);
222        let Some(sh) = sl.shapes.iter_mut().find(|x| x.ph_type() == Some(crate::model::PhType::Date)) else {
223            return Ok(false);
224        };
225        let mut props = sh.text.as_ref().and_then(|t| t.paragraphs.first()).and_then(|p| p.runs.first()).map(|r| r.props.clone()).unwrap_or_default();
226        if let Some(lang) = lang {
227            props.lang = Some(lang);
228        }
229        let kind = if update { RunKind::Field { field: format } } else { RunKind::Text };
230        let para = crate::model::Paragraph { runs: vec![crate::model::Run { text: sample, props, kind }], ..Default::default() };
231        let body = sh.text.clone().unwrap_or_default();
232        sh.text = Some(TextBody { paragraphs: vec![para], ..body });
233        Ok(true)
234    })
235}

237fn insert_date_box(s: &mut Session, o: &DateOpts) -> Result<Value> {
238    let size = s.doc()?.doc.slide_size;
239    let y = (size.height - 48.0).max(0.0);
240    s.execute("insert.textBox", &json!({"rect": [36.0, y, 360.0, 28.0], "text": ""}))?;
241    if o.update {
242        crate::engine::cmd::text::insert_field(s, &o.format, &o.sample)?;
243    } else {
244        crate::engine::cmd::text::insert_text(s, &o.sample, None)?;
245    }
246    if let Some(lang) = &o.lang {
247        stamp_caret_lang(s, lang)?;
248    }
249    Ok(json!({"where": "textBox", "format": o.format, "update": o.update, "text": o.sample}))
250}

252/// Insert `text` at the caret. `lang`, when set, is on that run before runs with equal props merge,
253/// so the date stays its own run next to the surrounding text.
254fn insert_plain(s: &mut Session, text: &str, lang: Option<&str>) -> Result<()> {
255    let text = text.to_string();
256    let lang = lang.map(str::to_string);
257    crate::engine::cmd::text::edit_text(s, move |body, t| {
258        let (a, b) = t.ordered();
259        let at = crate::model::edit::delete(body, a, b);
260        let mut props = crate::model::edit::props_at(body, at);
261        if let Some(lang) = lang.clone() {
262            props.lang = Some(lang);
263        }
264        let c = crate::model::edit::insert(body, at, &text, Some(props));
265        t.anchor = c;
266        t.caret = c;
267    })?;
268    Ok(())
269}

271/// Tag the run that ends at the caret with `lang`, so a date field keeps that language when it updates.
272fn stamp_caret_lang(s: &mut Session, lang: &str) -> Result<()> {
273    let lang = lang.to_string();
274    crate::engine::cmd::text::edit_text(s, move |body, t| {
275        let Some(p) = body.paragraphs.get_mut(t.caret.0) else { return };
276        let mut at = 0usize;
277        for r in &mut p.runs {
278            let n = r.char_len();
279            if n > 0 && at < t.caret.1 && t.caret.1 <= at + n {
280                r.props.lang = Some(lang);
281                return;
282            }
283            at += n;
284        }
285    })?;
286    Ok(())
287}
