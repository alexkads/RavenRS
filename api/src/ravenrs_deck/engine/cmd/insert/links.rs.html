6pub(super) fn hyperlink(s: &mut Session, p: &Value) -> Result<Value> {
7    use crate::model::text::{Action, Hyperlink};
8    // `display`: what a caret with nothing to link inserts, the address or the slide's title.
9    let (action, display) = if let Some(u) = str_param(p, "url") {
10        (Action::Url { url: u.to_string() }, u.strip_prefix("mailto:").unwrap_or(u).to_string())
11    } else if let Some(i) = usize_param(p, "slide") {
12        let slide = s.doc()?.doc.slides.get(i).ok_or_else(|| bad("insert.hyperlink", "no such slide"))?;
13        let title = slide.title();
14        (Action::Slide { slide: slide.id }, if title.is_empty() { format!("Slide {}", i + 1) } else { title })
15    } else {
16        return Err(bad("insert.hyperlink", "missing `url` or `slide`"));
17    };
18    let display = str_param(p, "text").map_or(display, str::to_string);
19    let link = Hyperlink {
20        action,
21        tooltip: str_param(p, "tooltip").unwrap_or("").to_string(),
22        highlight_click: bool_or(p, "highlightClick", false),
23        sound: None,
24        stop_sound: false,
25        zoom_return: None,
26    };
27    if s.doc()?.selection.text.is_some() {
28        return crate::engine::cmd::text::link_text(s, link, &display);
29    }
30    edit_shapes(s, p, "insert.hyperlink", |sh| {
31        sh.click = Some(link.clone());
32        Ok(())
33    })
34}

36/// Insert ▸ Action (Action Settings): what clicking the selected shapes, or pointing at them, does
37/// in a show.
38pub(super) fn shape_action(s: &mut Session, p: &Value) -> Result<Value> {
39    use crate::model::text::{Action, Hyperlink};
40    const CMD: &str = "shape.action";
41    let hover = match str_param(p, "on").unwrap_or("click") {
42        "click" => false,
43        "hover" | "over" => true,
44        o => return Err(bad(CMD, format!("`on` is click or hover, not `{o}`"))),
45    };
46    let need = |k: &str| str_param(p, k).map(str::to_string).filter(|v| !v.trim().is_empty()).ok_or_else(|| bad(CMD, format!("missing `{k}`")));
47    let action = match str_param(p, "action").unwrap_or("none") {
48        "none" => None,
49        "next" => Some(Action::NextSlide),
50        "previous" => Some(Action::PreviousSlide),
51        "first" => Some(Action::FirstSlide),
52        "last" => Some(Action::LastSlide),
53        "lastViewed" => Some(Action::LastViewed),
54        "endShow" => Some(Action::EndShow),
55        "playMedia" => Some(Action::PlayMedia),
56        "url" => Some(Action::Url { url: need("url")? }),
57        "customShow" => {
58            let name = need("show")?;
59            if !s.doc()?.doc.custom_shows.iter().any(|c| c.name == name) {
60                return Err(bad(CMD, format!("no custom show `{name}`")));
61            }
62            Some(Action::CustomShow { name, returns: true })
63        }
64        "program" => Some(Action::Program { path: need("path")? }),
65        "slide" => {
66            let i = usize_param(p, "slide").ok_or_else(|| bad(CMD, "missing `slide`"))?;
67            let slide = s.doc()?.doc.slides.get(i).ok_or_else(|| bad(CMD, "no such slide"))?;
68            Some(Action::Slide { slide: slide.id })
69        }
70        a => return Err(bad(CMD, format!("unknown action `{a}`"))),
71    };
72    let highlight = bool_param(p, "highlight").unwrap_or(true);
73    let tooltip = str_param(p, "tooltip").unwrap_or("").to_string();
74    let link = action.map(|action| Hyperlink { action, tooltip, highlight_click: highlight, sound: None, stop_sound: false, zoom_return: None });
75    edit_shapes(s, p, CMD, |sh| {
76        if hover {
77            sh.hover = link.clone();
78        } else {
79            sh.click = link.clone();
80        }
81        Ok(())
82    })
83}

85pub(super) fn action_button(s: &mut Session, p: &Value) -> Result<Value> {
86    use crate::model::text::{Action, Hyperlink};
87    let kind = str_param(p, "kind").unwrap_or("custom");
88    let (preset, action) = match kind {
89        "back" | "previous" => ("actionButtonBackPrevious", Some(Action::PreviousSlide)),
90        "forward" | "next" => ("actionButtonForwardNext", Some(Action::NextSlide)),
91        "beginning" => ("actionButtonBeginning", Some(Action::FirstSlide)),
92        "end" => ("actionButtonEnd", Some(Action::LastSlide)),
93        "home" => ("actionButtonHome", Some(Action::FirstSlide)),
94        "information" => ("actionButtonInformation", None),
95        "return" => ("actionButtonReturn", Some(Action::LastViewed)),
96        "movie" => ("actionButtonMovie", None),
97        "document" => ("actionButtonDocument", None),
98        "sound" => ("actionButtonSound", None),
99        "help" => ("actionButtonHelp", None),
100        _ => ("actionButtonBlank", None),
101    };
102    let rect = rect_param(p, "rect").unwrap_or_else(|| default_rect(s, 48.0, 48.0));
103    let shape = Shape {
104        xfrm: Some(rect),
105        geom: Geom::preset(preset),
106        style: Some(ShapeStyle::accent(SchemeSlot::Accent1)),
107        click: action.map(|a| Hyperlink {
108            action: a,
109            tooltip: String::new(),
110            highlight_click: true,
111            sound: None,
112            stop_sound: false,
113            zoom_return: None,
114        }),
115        ..Default::default()
116    };
117    let id = add_shape(s, "insert.actionButton", shape, true)?;
118    Ok(json!({"id": id}))
119}
