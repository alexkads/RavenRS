59/// Insert a photo album after the current slide.
60///
61/// PowerPoint starts a new presentation. These slides go into the open one, so the album is one
62/// undo step. Omitting `pictures` asks the UI to open the dialog and does not edit.
63pub(super) fn photo_album(s: &mut Session, p: &Value) -> Result<Value> {
64    if p.get("pictures").is_none_or(Value::is_null) {
65        s.ui_requests.push(crate::engine::UiRequest::Dialog { id: "photoAlbum".into(), params: json!({}) });
66        return Ok(json!({"dialog": "photoAlbum"}));
67    }
68    let opts = album_opts(p)?;
69    let items = album_items(p)?;
70    commit_album(s, items, opts, None)
71}

73pub(super) fn photo_album_edit(s: &mut Session, p: &Value) -> Result<Value> {
74    if p.get("pictures").is_none_or(Value::is_null) {
75        let seed = album_seed(s)?;
76        let album = seed.get("album").cloned().unwrap_or(Value::Null);
77        s.ui_requests.push(crate::engine::UiRequest::Dialog { id: "photoAlbum".into(), params: seed });
78        return Ok(json!({"dialog": "photoAlbum", "album": album}));
79    }
80    let opts = album_opts(p)?;
81    let items = album_items(p)?;
82    let (id, ids) = album_target(s, p)?;
83    // The editor is seeded within the album limits; replacing the album from a trimmed seed would
84    // delete what it left out, so refuse rather than lose it.
85    let st = s.doc()?;
86    if items_from_slides(&st.doc, &album_slides(st, &ids)).1 {
87        return Err(bad("insert.photoAlbumEdit", "this album holds more than the editor can carry; editing it would delete the rest"));
88    }
89    commit_album(s, items, opts, Some((id, ids)))
90}

92fn commit_album(s: &mut Session, items: Vec<AlbumItem>, opts: AlbumOpts, replace: Option<(u32, Vec<u64>)>) -> Result<Value> {
93    let (snapshot, sel) = {
94        let st = s.doc()?;
95        (st.doc.clone(), st.selection.clone())
96    };
97    let result = (|| {
98        let album_id = match replace.as_ref().map(|(id, _)| *id) {
99            Some(id) if id != 0 => id,
100            _ => {
101                let cmd = if replace.is_some() { "insert.photoAlbumEdit" } else { "insert.photoAlbum" };
102                s.edit(|doc, _| doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE)))?
103            }
104        };
105        let at = if let Some((_, ids)) = &replace {
106            let st = s.doc()?;
107            let indices: Vec<usize> = ids.iter().filter_map(|id| st.doc.slide_index(crate::model::SlideId(*id as u32))).collect();
108            let Some(at) = indices.iter().copied().min() else {
109                return Err(bad("insert.photoAlbumEdit", "that photo album is no longer in this presentation"));
110            };
111            s.execute("slide.delete", &json!({"ids": ids}))?;
112            Some(at)
113        } else {
114            None
115        };
116        let mut opts = opts;
117        if opts.theme.as_deref().unwrap_or("").is_empty() {
118            opts.theme = s.doc().ok().and_then(|st| st.doc.masters.first().map(|m| m.theme.name.clone())).filter(|n| !n.is_empty());
119        }
120        apply_theme_if_needed(s, opts.theme.as_deref())?;
121        insert_album(s, items, &opts, at, album_id)
122    })();
123    if result.is_err()
124        && let Some(st) = s.active_mut()
125    {
126        st.doc = snapshot;
127        st.selection = sel;
128    }
129    result
130}

136fn album_layout(p: &Value) -> Result<AlbumLayout> {
137    let key = match p.get("layout") {
138        None => return Ok(AlbumLayout { cols: 1, rows: 1, per: 1, title: false, fit: false }),
139        Some(Value::String(s)) => s.as_str(),
140        Some(_) => return Err(bad("insert.photoAlbum", "`layout` must be text")),
141    };
142    match album_norm(key).as_str() {
143        "fit" => Ok(AlbumLayout { cols: 1, rows: 1, per: 1, title: false, fit: true }),
144        "1" | "one" => Ok(AlbumLayout { cols: 1, rows: 1, per: 1, title: false, fit: false }),
145        "1title" | "onetitle" => Ok(AlbumLayout { cols: 1, rows: 1, per: 1, title: true, fit: false }),
146        "2" | "two" => Ok(AlbumLayout { cols: 2, rows: 1, per: 2, title: false, fit: false }),
147        "2title" | "twotitle" => Ok(AlbumLayout { cols: 2, rows: 1, per: 2, title: true, fit: false }),
148        "4" | "four" => Ok(AlbumLayout { cols: 2, rows: 2, per: 4, title: false, fit: false }),
149        "4title" | "fourtitle" => Ok(AlbumLayout { cols: 2, rows: 2, per: 4, title: true, fit: false }),
150        _ => Err(bad("insert.photoAlbum", format!("unknown layout `{key}` (use fit, 1, 2, 4, 1title, 2title or 4title)"))),
151    }
152}

206fn album_opts(p: &Value) -> Result<AlbumOpts> {
207    let layout = album_layout(p)?;
208    let frame = album_frame(p)?;
209    let theme = match p.get("theme") {
210        None | Some(Value::Null) => None,
211        Some(Value::String(s)) if s.trim().is_empty() => None,
212        Some(Value::String(s)) => {
213            let name = s.trim();
214            let found = crate::model::theme::builtin_themes().into_iter().find(|t| t.name.eq_ignore_ascii_case(name));
215            let Some(t) = found else {
216                return Err(bad("insert.photoAlbum", format!("unknown theme `{name}`")));
217            };
218            Some(t.name)
219        }
220        Some(_) => return Err(bad("insert.photoAlbum", "`theme` must be a theme name")),
221    };
222    Ok(AlbumOpts {
223        layout_key: layout_key_of(layout),
224        frame_key: frame_key_of(frame).into(),
225        layout,
226        frame,
227        captions: album_captions(p)?,
228        grayscale: album_bool(p, "grayscale")?,
229        theme,
230    })
231}

233/// A finite number, or 0 when the key is absent. A present non-number is an error.
234fn album_finite(item: &Value, key: &str) -> Result<f64> {
235    match item.get(key) {
236        None | Some(Value::Null) => Ok(0.0),
237        Some(v) => v.as_f64().filter(|n| n.is_finite()).ok_or_else(|| bad("insert.photoAlbum", format!("`{key}` must be a number"))),
238    }
239}

241fn album_rot(item: &Value) -> Result<f64> {
242    let n = album_finite(item, "rot")?;
243    Ok(((n % 360.0) + 360.0) % 360.0)

246fn album_unit(item: &Value, key: &str) -> Result<f64> {
247    let n = album_finite(item, key)?;
248    if !(-1.0..=1.0).contains(&n) {
249        return Err(bad("insert.photoAlbum", format!("`{key}` must be between -1 and 1")));
250    }
251    Ok(n)

254fn album_items(p: &Value) -> Result<Vec<AlbumItem>> {
255    let Some(arr) = p.get("pictures").and_then(Value::as_array) else {
256        return Err(bad("insert.photoAlbum", "`pictures` must be a list"));
257    };
258    if arr.is_empty() {
259        return Err(bad("insert.photoAlbum", "add at least one picture or text box"));
260    }
261    if arr.len() > ALBUM_MAX_ITEMS {
262        return Err(bad("insert.photoAlbum", "a photo album holds at most 48 pictures or text boxes"));
263    }
264    let mut items = Vec::with_capacity(arr.len());
265    let mut total = 0usize;
266    for item in arr {
267        let Some(obj) = item.as_object() else {
268            return Err(bad("insert.photoAlbum", "each album item must be a picture or a text box"));
269        };
270        let has_media = obj.contains_key("data") || obj.contains_key("path");
271        if !has_media {
272            let Some(text) = obj.get("text").and_then(Value::as_str) else {
273                return Err(bad("insert.photoAlbum", "each album item needs `path`, `data` or `text`"));
274            };
275            let text = text.trim();
276            if text.is_empty() {
277                return Err(bad("insert.photoAlbum", "a text box needs some text"));
278            }
279            if text.chars().count() > ALBUM_MAX_TEXT {
280                return Err(bad("insert.photoAlbum", "text is longer than 400 characters"));
281            }
282            items.push(AlbumItem::Text(text.to_string()));
283            continue;
284        }
285        let (name, bytes) = media_bytes(item, "insert.photoAlbum")?;
286        if bytes.len() > ALBUM_MAX_BYTES {
287            return Err(bad("insert.photoAlbum", format!("{name} is larger than 32 MB")));
288        }
289        let next = total.saturating_add(bytes.len());
290        if next > ALBUM_MAX_TOTAL {
291            return Err(bad("insert.photoAlbum", "the album is larger than 64 MB"));
292        }
293        total = next;
294        let ct = content_type(&name, &bytes);
295        if !ct.starts_with("image/") {
296            return Err(bad("insert.photoAlbum", format!("{name} is not a picture")));
297        }
298        if ct != "image/svg+xml" && crate::render::decode_image(&bytes).is_none() {
299            return Err(bad("insert.photoAlbum", format!("{name}: the picture can't be read")));
300        }
301        let (px_w, px_h) = image_px(&bytes);
302        items.push(AlbumItem::Picture(AlbumPic {
303            name,
304            bytes,
305            px_w,
306            px_h,
307            rot: album_rot(item)?,
308            brightness: album_unit(item, "brightness")?,
309            contrast: album_unit(item, "contrast")?,
310        }));
311    }
312    Ok(items)
313}

315fn image_px(bytes: &[u8]) -> (f64, f64) {
316    image::ImageReader::new(std::io::Cursor::new(bytes))
317        .with_guessed_format()
318        .ok()
319        .and_then(|r| r.into_dimensions().ok())
320        .map(|(w, h)| (w as f64, h as f64))
321        .filter(|(w, h)| *w > 0.0 && *h > 0.0)
322        .unwrap_or((4.0, 3.0))
323}

325fn file_stem(name: &str) -> String {
326    let base = std::path::Path::new(name).file_name().and_then(std::ffi::OsStr::to_str).unwrap_or(name);
327    let stem = match base.rsplit_once('.') {
328        Some((s, ext)) if !s.is_empty() && !ext.is_empty() => s,
329        _ => base,
330    };
331    if stem.is_empty() { "Picture".to_string() } else { stem.to_string() }
332}

334fn album_plan(items: Vec<AlbumItem>, per: usize) -> Vec<AlbumSlide> {
335    let per = per.max(1);
336    let mut out = Vec::new();
337    let mut batch = Vec::new();
338    for item in items {
339        match item {
340            AlbumItem::Text(text) => {
341                if !batch.is_empty() {
342                    out.push(AlbumSlide::Pictures(std::mem::take(&mut batch)));
343                }
344                out.push(AlbumSlide::Text(text));
345            }
346            AlbumItem::Picture(pic) => {
347                batch.push(pic);
348                if batch.len() >= per {
349                    out.push(AlbumSlide::Pictures(std::mem::take(&mut batch)));
350                }
351            }
352        }
353    }
354    if !batch.is_empty() {
355        out.push(AlbumSlide::Pictures(batch));
356    }
357    out
358}

365fn fit_in(box_w: f64, box_h: f64, px_w: f64, px_h: f64) -> (f64, f64) {
366    let aspect = if px_h > 0.0 { px_w / px_h } else { 4.0 / 3.0 };
367    let box_w = box_w.max(1.0);
368    let box_h = box_h.max(1.0);
369    if !aspect.is_finite() || aspect <= 0.0 {
370        return (box_w, box_h);
371    }
372    if aspect >= box_w / box_h { (box_w, (box_w / aspect).max(1.0)) } else { ((box_h * aspect).max(1.0), box_h) }
373}

375fn place_album(width: f64, height: f64, layout: AlbumLayout, n: usize, captions: bool) -> (Option<[f64; 4]>, Vec<Placed>) {
376    let margin = if layout.fit { 18.0 } else { 36.0 };
377    let title_band = if layout.title { 48.0 } else { 0.0 };
378    let gap = 12.0;
379    let cap_h = 22.0;
380    let cap_gap = 4.0;
381    let left = margin;
382    let top = margin + title_band;
383    let grid_w = (width - margin - left).max(1.0);
384    let grid_h = (height - margin - top).max(1.0);
385    let cols = layout.cols.max(1);
386    let rows = layout.rows.max(1);
387    let cell_w = ((grid_w - gap * (cols.saturating_sub(1) as f64)) / cols as f64).max(1.0);
388    let cell_h = ((grid_h - gap * (rows.saturating_sub(1) as f64)) / rows as f64).max(1.0);
389    let pic_h = if captions { (cell_h - cap_h - cap_gap).max(1.0) } else { cell_h };
390    let mut placed = Vec::with_capacity(n);
391    for i in 0..n {
392        let col = i % cols;
393        let row = i / cols;
394        if row >= rows {
395            break;
396        }
397        let cell_x = left + col as f64 * (cell_w + gap);
398        let cell_y = top + row as f64 * (cell_h + gap);
399        placed.push(Placed { pic: [cell_x, cell_y, cell_w, pic_h], caption: None });
400    }
401    let title = if layout.title { Some([margin, 8.0, (width - 2.0 * margin).max(1.0), 32.0]) } else { None };
402    (title, placed)
403}

405fn place_pictures(width: f64, height: f64, layout: AlbumLayout, pics: &[AlbumPic], captions: bool) -> (Option<[f64; 4]>, Vec<Placed>) {
406    let (title, slots) = place_album(width, height, layout, pics.len(), captions);
407    let placed = pics
408        .iter()
409        .zip(slots)
410        .map(|(pic, slot)| {
411            let (dw, dh) = fit_in(slot.pic[2], slot.pic[3], pic.px_w, pic.px_h);
412            let x = slot.pic[0] + (slot.pic[2] - dw) / 2.0;
413            let y = slot.pic[1] + (slot.pic[3] - dh) / 2.0;
414            let caption = if captions { Some([x, y + dh + 4.0, dw.max(36.0), 22.0]) } else { None };
415            Placed { pic: [x, y, dw, dh], caption }
416        })
417        .collect();
418    (title, placed)
419}

441fn parse_album_mark(raw: &str) -> Option<AlbumMark> {
442    let rest = raw.strip_prefix("deckcraft-album:")?;
443    let (id_str, rest) = rest.split_once(';').unwrap_or((rest, ""));
444    let id = id_str.parse::<u32>().ok().filter(|id| *id != 0)?;
445    let mut mark = AlbumMark { id, layout: String::new(), frame: String::new(), captions: false, grayscale: false, theme: String::new() };
446    if !rest.is_empty() {
447        for part in rest.split(';') {
448            let Some((k, v)) = part.split_once('=') else { continue };
449            match k {
450                "layout" => mark.layout = v.to_string(),
451                "frame" => mark.frame = v.to_string(),
452                "captions" => mark.captions = v == "1",
453                "gray" => mark.grayscale = v == "1",
454                "theme" => mark.theme = v.to_string(),
455                _ => {}
456            }
457        }
458    }
459    Some(mark)
460}

510fn album_name_run(doc: &crate::model::Presentation, index: usize) -> Vec<u64> {
511    if doc.slides.get(index).is_none_or(|s| s.name != "Photo Album") {
512        return Vec::new();
513    }
514    let mut start = index;
515    while start > 0 && doc.slides.get(start - 1).is_some_and(|s| s.name == "Photo Album") {
516        start -= 1;
517    }
518    let mut end = index;
519    while doc.slides.get(end + 1).is_some_and(|s| s.name == "Photo Album") {
520        end += 1;
521    }
522    doc.slides[start..=end].iter().map(|s| u64::from(s.id.0)).collect()
523}

525/// The album under the selection, or the one named by `album`. Id 0 is a name-only run (no marker).
526fn album_target(s: &Session, p: &Value) -> Result<(u32, Vec<u64>)> {
527    let st = s.doc()?;
528    if let Some(id) = album_id_param(p) {
529        let ids = slides_with_album(&st.doc, id);
530        if ids.is_empty() {
531            return Err(bad("insert.photoAlbumEdit", "that photo album is no longer in this presentation"));
532        }
533        return Ok((id, ids));
534    }
535    let Some(slide) = st.doc.slides.get(st.selection.slide) else {
536        return Err(bad("insert.photoAlbumEdit", "this slide is not part of a photo album"));
537    };
538    if let Some(mark) = slide.raw_ext.as_deref().and_then(parse_album_mark) {
539        let ids = slides_with_album(&st.doc, mark.id);
540        if ids.is_empty() {
541            return Err(bad("insert.photoAlbumEdit", "this slide is not part of a photo album"));
542        }
543        return Ok((mark.id, ids));
544    }
545    if slide.name == "Photo Album" {
546        let ids = album_name_run(&st.doc, st.selection.slide);
547        if !ids.is_empty() {
548            return Ok((0, ids));
549        }
550    }
551    Err(bad("insert.photoAlbumEdit", "this slide is not part of a photo album"))
552}

554/// Whether a slide holds anything besides album items: shapes, notes, comments, a background, a
555/// transition, animations, a colour map or tags.
556fn has_slide_content(s: &crate::model::Slide) -> bool {
557    !s.shapes.is_empty()
558        || !s.notes.text().trim().is_empty()
559        || !s.comments.is_empty()
560        || s.background.is_some()
561        || s.transition.is_some()
562        || !s.animations.is_empty()
563        || s.color_map.is_some()
564        || !s.tags.is_empty()
565}

567fn pic_count(slide: &crate::model::Slide) -> usize {
568    slide.shapes.iter().filter(|sh| matches!(sh.kind, ShapeKind::Picture { .. })).count()
569}

571fn infer_album(slides: &[&crate::model::Slide], theme_now: &str) -> (String, String, bool, bool, String) {
572    let max_pics = slides.iter().map(|s| pic_count(s)).max().unwrap_or(1);
573    let per = if max_pics >= 3 {
574        "4"
575    } else if max_pics == 2 {
576        "2"
577    } else {
578        "1"
579    };
580    let title =
581        slides.iter().any(|s| pic_count(s) > 0 && s.shapes.iter().any(|sh| sh.text_box && sh.xfrm.is_some_and(|x| x.y < 36.0 && x.h <= 48.0)));
582    let layout = if title { format!("{per}title") } else { per.into() };
583    let width = slides.iter().find_map(|s| {
584        s.shapes.iter().find_map(|sh| match &sh.kind {
585            ShapeKind::Picture { .. } => sh.line.as_ref().and_then(|l| l.width),
586            _ => None,
587        })
588    });
589    let frame = match width {
590        None => "none",
591        Some(w) if (w - 6.0).abs() < 0.2 => "simple",
592        Some(w) if (w - 2.25).abs() < 0.2 => "rounded",
593        Some(w) if w < 0.2 => "none",
594        Some(_) => "rect",
595    };
596    let captions = slides.iter().any(|s| {
597        let bottoms: Vec<f64> =
598            s.shapes.iter().filter(|sh| matches!(sh.kind, ShapeKind::Picture { .. })).filter_map(|sh| sh.xfrm.map(|x| x.y + x.h)).collect();
599        !bottoms.is_empty() && s.shapes.iter().any(|sh| sh.text_box && sh.xfrm.is_some_and(|x| bottoms.iter().any(|b| x.y + 1.0 >= *b)))
600    });
601    let mut any = false;
602    let mut gray = true;
603    for s in slides {
604        for sh in &s.shapes {
605            if let ShapeKind::Picture { fill } = &sh.kind {
606                any = true;
607                gray &= fill.adjust.grayscale;
608            }
609        }
610    }
611    (layout, frame.into(), captions, any && gray, theme_now.to_string())
612}

614/// The album's items for the dialog, within the limits `insert.photoAlbum` itself takes (items,
615/// bytes per picture and in all), checked before a picture is encoded: an album slide set can
616/// name one large picture many times.
617///
618/// The flag is true when something was left out for a limit.
619fn items_from_slides(doc: &crate::model::Presentation, slides: &[&crate::model::Slide]) -> (Vec<Value>, bool) {
620    let mut items = Vec::new();
621    let mut total = 0usize;
622    let mut omitted = false;
623    'slides: for slide in slides {
624        let pics: Vec<&crate::model::Shape> = slide.shapes.iter().filter(|sh| matches!(sh.kind, ShapeKind::Picture { .. })).collect();
625        if pics.is_empty() {
626            let text = slide
627                .shapes
628                .iter()
629                .filter(|sh| sh.text_box)
630                .filter_map(|sh| sh.text.as_ref().map(|t| t.text()))
631                .find(|t| !t.trim().is_empty())
632                .unwrap_or_default();
633            if !text.trim().is_empty() {
634                if items.len() >= ALBUM_MAX_ITEMS {
635                    omitted = true;
636                    break 'slides;
637                }
638                items.push(json!({"text": text.trim()}));
639            } else if items.len() >= ALBUM_MAX_ITEMS && has_slide_content(slide) {
640                // Past the item limit, a slide with other content would be deleted by a replace.
641                omitted = true;
642                break 'slides;
643            }
644            continue;
645        }
646        for sh in pics {
647            let ShapeKind::Picture { fill } = &sh.kind else { continue };
648            let media = doc.media(fill.media);
649            let size = media.map_or(0, |m| m.data.len());
650            if items.len() >= ALBUM_MAX_ITEMS || size > ALBUM_MAX_BYTES || total.saturating_add(size) > ALBUM_MAX_TOTAL {
651                omitted = true;
652                break 'slides;
653            }
654            total += size;
655            let name = if !sh.descr.is_empty() {
656                sh.descr.clone()
657            } else {
658                media.map(|m| m.name.clone()).filter(|n| !n.is_empty()).unwrap_or_else(|| "picture".into())
659            };
660            items.push(json!({
661                "name": name,
662                "data": media.map(|m| base64_encode(&m.data)).unwrap_or_default(),
663                "rot": sh.xfrm.map(|x| x.rot).unwrap_or(0.0),
664                "brightness": fill.adjust.brightness,
665                "contrast": fill.adjust.contrast,
666            }));
667        }
668    }
669    (items, omitted)
670}

672/// The slides of album `ids` as the editor would seed them.
673fn album_slides<'a>(st: &'a crate::engine::DocState, ids: &[u64]) -> Vec<&'a crate::model::Slide> {
674    ids.iter().filter_map(|sid| u32::try_from(*sid).ok()).filter_map(|n| st.doc.slide(crate::model::SlideId(n))).collect()
675}
676
677fn album_seed(s: &Session) -> Result<Value> {
678    let (id, ids) = album_target(s, &Value::Null)?;
679    let st = s.doc()?;
680    let slides = album_slides(st, &ids);
681    let (items, _) = items_from_slides(&st.doc, &slides);
682    let theme_now = st.doc.masters.first().map(|m| m.theme.name.clone()).filter(|n| !n.is_empty()).unwrap_or_else(|| "DeckCraft".into());
683    let inferred = infer_album(&slides, &theme_now);
684    let mark = slides.first().and_then(|sl| sl.raw_ext.as_deref().and_then(parse_album_mark));
685    let (layout, frame, captions, grayscale, theme) = if let Some(mark) = mark {
686        (
687            if mark.layout.is_empty() { inferred.0 } else { mark.layout },
688            if mark.frame.is_empty() { inferred.1 } else { mark.frame },
689            mark.captions,
690            mark.grayscale,
691            if mark.theme.is_empty() { theme_now } else { mark.theme },
692        )
693    } else {
694        inferred
695    };
696    Ok(json!({
697        "album": if id == 0 { Value::Null } else { json!(id) },
698        "layout": layout,
699        "frame": frame,
700        "captions": captions,
701        "grayscale": grayscale,
702        "theme": theme,
703        "items": items,
704    }))
705}

707fn insert_album(s: &mut Session, items: Vec<AlbumItem>, opts: &AlbumOpts, at: Option<usize>, album_id: u32) -> Result<Value> {
708    let (width, height, first) = {
709        let st = s.doc()?;
710        let first = at.unwrap_or_else(|| if st.doc.slides.is_empty() { 0 } else { st.selection.slide.saturating_add(1) });
711        (st.doc.slide_size.width, st.doc.slide_size.height, first)
712    };
713    let layout = opts.layout;
714    let frame = opts.frame;
715    let captions = opts.captions;
716    let plan = album_plan(items, layout.per);
717    let mut slides = 0usize;
718    let mut pictures = 0usize;
719    let mut pin = at;
720    for slide in plan {
721        if let Some(index) = pin.take() {
722            s.execute("slide.new", &json!({"layout": "blank", "at": index}))?;
723        } else {
724            s.execute("slide.new", &json!({"layout": "blank"}))?;
725        }
726        stamp_album_slide(s, album_id, opts)?;
727        slides += 1;
728        match slide {
729            AlbumSlide::Text(text) => {
730                let rect = if layout.title {
731                    [36.0, 18.0, (width - 72.0).max(36.0), 48.0]
732                } else {
733                    [36.0, 36.0, (width - 72.0).max(36.0), (height - 72.0).max(36.0)]
734                };
735                s.execute("insert.textBox", &json!({"rect": rect, "text": text}))?;
736            }
737            AlbumSlide::Pictures(pics) => {
738                let stem = pics.first().map(|p| file_stem(&p.name)).unwrap_or_default();
739                let (title, placed) = place_pictures(width, height, layout, &pics, captions);
740                for (pic, cell) in pics.iter().zip(placed) {
741                    let data = base64_encode(&pic.bytes);
742                    let inserted = s.execute("insert.picture", &json!({"name": pic.name, "data": data, "rect": cell.pic}))?;
743                    let Some(id) = inserted.get("id").and_then(Value::as_u64) else {
744                        return Err(bad("insert.photoAlbum", "the picture was not inserted"));
745                    };
746                    apply_frame(s, frame, id)?;
747                    if opts.grayscale || pic.brightness.abs() > 1e-6 || pic.contrast.abs() > 1e-6 {
748                        s.execute(
749                            "picture.adjust",
750                            &json!({"id": id, "grayscale": opts.grayscale, "brightness": pic.brightness, "contrast": pic.contrast}),
751                        )?;
752                    }
753                    if pic.rot > 0.01 {
754                        s.execute("shape.rotate", &json!({"id": id, "deg": pic.rot}))?;
755                    }
756                    pictures += 1;
757                    if let Some(cap) = cell.caption {
758                        s.execute("insert.textBox", &json!({"rect": cap, "text": file_stem(&pic.name)}))?;
759                    }
760                }
761                if let Some(rect) = title {
762                    s.execute("insert.textBox", &json!({"rect": rect, "text": stem}))?;
763                }
764            }
765        }
766    }
767    s.execute("slide.go", &json!({"index": first}))?;
768    Ok(json!({"slides": slides, "first": first, "pictures": pictures, "album": album_id}))
769}
