6pub(super) fn scale_shapes(shapes: &mut [Shape], sx: f64, sy: f64, k: f64, ox: f64, oy: f64) {
7    for sh in shapes {
8        if let Some(x) = sh.xfrm.as_mut() {
9            x.x = x.x * sx + ox;
10            x.y = x.y * sy + oy;
11            x.w *= sx;
12            x.h *= sy;
13        }
14        if let Some(t) = sh.text.as_mut() {
15            crate::model::edit::format_all(t, &|r| {
16                if let Some(sz) = r.size.as_mut() {
17                    *sz = (*sz * k).max(1.0);
18                }
19            });
20        }
21    }
22}

24pub(super) fn slide_size(s: &mut Session, p: &Value) -> Result<Value> {
25    let size = if let Some(pre) = str_param(p, "preset") {
26        let k = pre.to_ascii_lowercase();
27        let found = defaults::SLIDE_SIZES.iter().find(|(l, _, _)| l.to_ascii_lowercase().starts_with(&k) || l.to_ascii_lowercase().contains(&k));
28        let (_, w, h) = found.ok_or_else(|| bad("design.slideSize", format!("unknown preset `{pre}`")))?;
29        Size::new(*w, *h)
30    } else {
31        Size::new(
32            f64_param(p, "w").ok_or_else(|| bad("design.slideSize", "missing `w`"))?.clamp(72.0, 4032.0),
33            f64_param(p, "h").ok_or_else(|| bad("design.slideSize", "missing `h`"))?.clamp(72.0, 4032.0),
34        )
35    };
36    let size = match str_param(p, "orientation") {
37        Some("portrait") if size.width > size.height => Size::new(size.height, size.width),
38        Some("landscape") if size.height > size.width => Size::new(size.height, size.width),
39        Some("portrait" | "landscape") | None => size,
40        Some(o) => return Err(bad("design.slideSize", format!("unknown orientation `{o}`"))),
41    };
42    let mode = str_param(p, "scale").unwrap_or("ensureFit").to_string();
43    s.edit(|doc, _| {
44        let old = doc.slide_size;
45        let (sx, sy) = (size.width / old.width.max(1.0), size.height / old.height.max(1.0));
46        let (sx, sy, k, ox, oy) = match mode.as_str() {
47            "none" => (1.0, 1.0, 1.0, 0.0, 0.0),
48            "maximize" => {
49                let k = sx.max(sy);
50                (k, k, k, (size.width - old.width * k) / 2.0, (size.height - old.height * k) / 2.0)
51            }
52            _ => {
53                let k = sx.min(sy);
54                (k, k, k, (size.width - old.width * k) / 2.0, (size.height - old.height * k) / 2.0)
55            }
56        };
57        // Masters and layouts stretch to the new size; slides scale by mode.
58        let (msx, msy) = (size.width / old.width.max(1.0), size.height / old.height.max(1.0));
59        for m in &mut doc.masters {
60            let m = Arc::make_mut(m);
61            scale_shapes(&mut m.shapes, msx, msy, 1.0, 0.0, 0.0);
62            for l in &mut m.layouts {
63                scale_shapes(&mut l.shapes, msx, msy, 1.0, 0.0, 0.0);
64            }
65        }
66        for sl in &mut doc.slides {
67            scale_shapes(&mut Arc::make_mut(sl).shapes, sx, sy, k, ox, oy);
68        }
69        doc.slide_size = size;
70        Ok(())
71    })?;
72    Ok(json!({"w": size.width, "h": size.height}))
73}

75pub(super) fn header_footer(s: &mut Session, p: &Value) -> Result<Value> {
76    s.edit(|doc, sel| {
77        let hf = &mut doc.header_footer;
78        if let Some(v) = bool_param(p, "date") {
79            hf.date = v;
80        }
81        if let Some(v) = str_param(p, "dateText") {
82            hf.date_text = v.into();
83        }
84        if let Some(v) = str_param(p, "dateFormat") {
85            // Refused rather than ignored: a format that renders nothing would leave the
86            // placeholder showing its saved text for ever.
87            if !crate::text::datetime::DATE_KINDS.contains(&v) {
88                return Err(bad("design.headerFooter", "dateFormat: one of datetime1…datetime13"));
89            }
90            hf.date_format = v.into();
91        }
92        if let Some(v) = bool_param(p, "slideNumber") {
93            hf.slide_number = v;
94        }
95        if let Some(v) = bool_param(p, "footer") {
96            hf.footer = v;
97        }
98        if let Some(v) = str_param(p, "footerText") {
99            hf.footer_text = v.into();
100            hf.footer = hf.footer || !v.is_empty();
101        }
102        if let Some(v) = bool_param(p, "hideOnTitle") {
103            hf.hide_on_title = v;
104        }
105        let hf = hf.clone();
106        let date_given = str_param(p, "dateText").is_some();
107        let format_given = str_param(p, "dateFormat").is_some();
108        let all = bool_or(p, "all", true);
109        let list: Vec<usize> = if all { (0..doc.slides.len()).collect() } else { vec![sel.slide] };
110        for i in list {
111            let Some(layout) = doc.slides.get(i).map(|x| x.layout) else { continue };
112            let lay = doc.layout(layout).map(|(_, l)| l.clone());
113            let is_title = lay.as_ref().is_some_and(|l| l.kind == crate::model::LayoutType::Title);
114            let want = |k: PhType| {
115                !(hf.hide_on_title && is_title)
116                    && match k {
117                        PhType::Date => hf.date,
118                        PhType::SlideNum => hf.slide_number,
119                        PhType::Footer => hf.footer,
120                        _ => false,
121                    }
122            };
123            let mut new_ids = vec![];
124            for _ in 0..3 {
125                new_ids.push(ShapeId(doc.alloc_id().ok_or_else(|| bad("design.headerFooter", NO_ID_SPACE))?));
126            }
127            let Some(sl) = doc.slides.get_mut(i) else { continue };
128            let sl = Arc::make_mut(sl);
129            for (n, k) in [PhType::Date, PhType::Footer, PhType::SlideNum].into_iter().enumerate() {
130                let has = sl.shapes.iter().position(|x| x.ph_type() == Some(k));
131                match (has, want(k)) {
132                    (Some(pos), false) => {
133                        sl.shapes.remove(pos);
134                    }
135                    (None, true) => {
136                        let lp = lay.as_ref().and_then(|l| l.shapes.iter().find(|x| x.ph_type() == Some(k)).cloned());
137                        let Some(lp) = lp else { continue };
138                        let body = match k {
139                            PhType::SlideNum => TextBody {
140                                paragraphs: vec![crate::model::Paragraph {
141                                    runs: vec![Run {
142                                        text: "‹#›".into(),
143                                        props: Default::default(),
144                                        kind: RunKind::Field { field: "slidenum".into() },
145                                    }],
146                                    ..Default::default()
147                                }],
148                                ..Default::default()
149                            },
150                            PhType::Date if hf.date_text.is_empty() => TextBody {
151                                paragraphs: vec![crate::model::Paragraph {
152                                    runs: vec![Run {
153                                        text: date_now(&hf.date_format),
154                                        props: Default::default(),
155                                        kind: RunKind::Field { field: date_kind(&hf.date_format).into() },
156                                    }],
157                                    ..Default::default()
158                                }],
159                                ..Default::default()
160                            },
161                            PhType::Date => TextBody::from_text(&hf.date_text),
162                            _ => TextBody::from_text(&hf.footer_text),
163                        };
164                        sl.shapes.push(Shape {
165                            id: new_ids.get(n).copied().unwrap_or_default(),
166                            name: lp.name.clone(),
167                            ph: lp.ph.clone(),
168                            text: Some(body),
169                            ..Default::default()
170                        });
171                    }
172                    (Some(pos), true) if k == PhType::Footer => {
173                        if let Some(sh) = sl.shapes.get_mut(pos) {
174                            sh.text = Some(TextBody::from_text(&hf.footer_text));
175                        }
176                    }
177                    (Some(pos), true) if k == PhType::Date && (date_given || format_given) => {
178                        if let Some(sh) = sl.shapes.get_mut(pos) {
179                            set_date(sh, &hf.date_text, &hf.date_format);
180                        }
181                    }
182                    _ => {}
183                }
184            }
185        }
186        Ok(())
187    })?;
188    ok()
189}

191/// The date field type `format` names, `datetime1` when it is empty or unknown.
192fn date_kind(format: &str) -> &str {
193    if crate::text::datetime::DATE_KINDS.contains(&format) { format } else { "datetime1" }
194}
195
196/// The saved text of a date field of `format`, as it shows now.
197fn date_now(format: &str) -> String {
198    let kind = date_kind(format);
199    crate::text::datetime::DateTime::now_local().and_then(|now| crate::text::datetime::field_text(kind, None, now)).unwrap_or_else(today)
200}

202/// Point a slide's date placeholder at a fixed date (its own plain text) or, for an empty
203/// `fixed`, at an automatic date field (one it already has keeps its format).
204fn set_date(sh: &mut Shape, fixed: &str, format: &str) {
205    let runs = || sh.text.iter().flat_map(|t| &t.paragraphs).flat_map(|p| &p.runs);
206    let field_now = runs().find_map(|r| match &r.kind {
207        RunKind::Field { field } if crate::text::datetime::is_date_field(field) => Some(field.clone()),
208        _ => None,
209    });
210    let has_field = field_now.is_some();
211    let props = runs().next().map(|r| r.props.clone()).unwrap_or_default();
212    let want = date_kind(format);
213    let (text, kind) = match fixed {
214        // Automatic: keep the field, but follow a format that has changed.
215        "" if field_now.as_deref() == Some(want) => return,
216        "" if has_field && format.is_empty() => return,
217        "" => (date_now(format), RunKind::Field { field: want.into() }),
218        _ if !has_field && sh.text.as_ref().is_some_and(|t| t.text() == fixed) => return,
219        _ => (fixed.to_string(), RunKind::Text),
220    };
221    let para = crate::model::Paragraph { runs: vec![Run { text, props, kind }], ..Default::default() };
222    sh.text = Some(TextBody { paragraphs: vec![para], ..sh.text.clone().unwrap_or_default() });
223}

225/// Today's local date as a `datetime1` field shows it (M/D/YYYY): the saved text of a new date
226/// field. Empty where there is no clock (wasm).
227pub fn today() -> String {
228    date_field_text("datetime1")
229}

231/// What date field `kind` shows now (empty if unknown or there is no clock).
232pub fn date_field_text(kind: &str) -> String {
233    use crate::text::datetime::{DateTime, field_text};
234    DateTime::now_local().and_then(|now| field_text(kind, None, now)).unwrap_or_default()
235}

237/// Days since 1970-01-01 → (year, month, day), exact for every `i64`.
238pub fn civil(z: i64) -> (i64, u32, u32) {
239    use chrono::Datelike;
240    // `NaiveDate` covers only about ±262143 years, and the Gregorian calendar repeats every 400
241    // years (146_097 days). chrono converts the day within its cycle and whole cycles go back
242    // into the year (an i128 for i64::MAX/MIN). 1970-01-01 is `num_days_from_ce` 719_163, and
243    // 0..146_097 days from it always fits.
244    let (cycles, rest) = (z.div_euclid(146_097), z.rem_euclid(146_097));
245    match i32::try_from(rest + 719_163).ok().and_then(chrono::NaiveDate::from_num_days_from_ce_opt) {
246        Some(d) => {
247            let y = i128::from(d.year()) + i128::from(cycles) * 400;
248            (i64::try_from(y).unwrap_or(if y < 0 { i64::MIN } else { i64::MAX }), d.month(), d.day())
249        }
250        None => (1970, 1, 1),
251    }
252}

254#[cfg(test)]
255mod header_footer_date_tests {
256    use crate::model::PhType;
257    use crate::model::text::RunKind;
258    use serde_json::json;

260    use crate::engine::Session;
261
262    /// The date field of the Date placeholders of every slide: (saved text, field type).
263    fn dates(s: &Session) -> Vec<(String, Option<String>)> {
264        let d = &s.doc().unwrap().doc;
265        d.slides
266            .iter()
267            .flat_map(|sl| sl.shapes.iter())
268            .filter(|sh| sh.ph_type() == Some(PhType::Date))
269            .flat_map(|sh| sh.text.iter())
270            .flat_map(|t| t.paragraphs.iter())
271            .flat_map(|p| p.runs.iter())
272            .map(|r| {
273                let kind = match &r.kind {
274                    RunKind::Field { field } => Some(field.clone()),
275                    _ => None,
276                };
277                (r.text.clone(), kind)
278            })
279            .collect()
280    }
281
282    fn sample(kind: &str) -> String {
283        let now = crate::text::datetime::DateTime::now_local().expect("a local date");
284        crate::text::datetime::field_text(kind, None, now).expect("a sample")
285    }
286
287    /// Runs `design.headerFooter` and checks the slide's date run is an automatic `kind` field.
288    /// The clock is read before and after the command, so a second ticking over between the
289    /// command and the read cannot fail the comparison. Returns the run text.
290    fn run_automatic(s: &mut Session, params: serde_json::Value, kind: &str) -> String {
291        let before = sample(kind);
292        s.execute("design.headerFooter", &params).unwrap();
293        let after = sample(kind);
294        let got = dates(s);
295        assert_eq!(got.len(), 1, "{got:?}");
296        assert_eq!(got[0].1.as_deref(), Some(kind));
297        assert!(got[0].0 == before || got[0].0 == after, "{got:?} is neither {before:?} nor {after:?}");
298        got[0].0.clone()
299    }
300
301    /// Header & Footer's automatic date takes the format it is given. `HeaderFooter::date_format`
302    /// existed for this and was never read: the placeholder was always `datetime1`, and a
303    /// `dateFormat` parameter was accepted and silently dropped.
304    #[test]
305    fn the_automatic_date_takes_the_format_it_is_given() {
306        let mut s = Session::with_new();
307
308        // Default: the short date, as before.
309        run_automatic(&mut s, json!({"date": true, "dateText": "", "all": true}), "datetime1");
310
311        // A format: the field type changes and the saved text is that format's.
312        run_automatic(&mut s, json!({"date": true, "dateFormat": "datetime5", "all": true}), "datetime5");
313        assert_eq!(s.doc().unwrap().doc.header_footer.date_format, "datetime5");
314
315        // A time format works the same way, and these two differ, so the format is really read.
316        let time = run_automatic(&mut s, json!({"date": true, "dateFormat": "datetime11", "all": true}), "datetime11");
317        assert_ne!(time, sample("datetime5"));
318
319        // Asking again changes nothing (and does not rewrite the run).
320        let again = run_automatic(&mut s, json!({"date": true, "dateFormat": "datetime11", "all": true}), "datetime11");
321        assert_eq!(again, time);
322
323        // A fixed date still wins: plain text, no field.
324        s.execute("design.headerFooter", &json!({"date": true, "dateText": "4 July 1776", "all": true})).unwrap();
325        assert_eq!(dates(&s), vec![("4 July 1776".to_string(), None)]);
326
327        // Back to automatic: the remembered format comes back, not datetime1.
328        run_automatic(&mut s, json!({"date": true, "dateText": "", "dateFormat": "datetime11", "all": true}), "datetime11");
329
330        // An unknown format is refused rather than ignored, and changes nothing.
331        for bad in ["datetime0", "datetime14", "nonsense", ""] {
332            assert!(s.execute("design.headerFooter", &json!({"date": true, "dateFormat": bad})).is_err(), "{bad}");
333        }
334        assert_eq!(s.doc().unwrap().doc.header_footer.date_format, "datetime11");
335        let got = dates(&s);
336        assert_eq!(got.len(), 1, "{got:?}");
337        assert_eq!(got[0].1.as_deref(), Some("datetime11"));
338    }
