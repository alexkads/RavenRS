1//! Design Ideas: arrangements of what is on the current slide (picture full-bleed with the title
2//! on a band, picture on one half, a framed picture, text-only layouts). An idea moves and
3//! resizes the slide's own shapes and crops pictures to their new box; it changes only that
4//! slide. Everything is computed here, offline.

24const PICTURE_IDEAS: [Idea; 4] = [
25    ("fullBleed", "Full-bleed picture, title on a band"),
26    ("pictureLeft", "Picture left, text right"),
27    ("pictureRight", "Picture right, text left"),
28    ("framed", "Framed picture beside the title"),
29];
30const TEXT_IDEAS: [Idea; 3] =
31    [("titleColumn", "Title column beside the text"), ("centred", "Centred title over the text"), ("titleBottom", "Text above, title at the foot")];

33/// The roles of a slide's top-level shapes, by index.
34struct Roles {
35    title: Option<usize>,
36    picture: Option<usize>,
37    text: Vec<usize>,
38}
39
40fn roles(shapes: &[Shape]) -> Roles {
41    let mut r = Roles { title: None, picture: None, text: vec![] };
42    for (i, s) in shapes.iter().enumerate() {
43        if s.hidden {
44            continue;
45        }
46        if s.ph.as_ref().is_some_and(|p| p.kind.is_title()) && r.title.is_none() {
47            r.title = Some(i);
48        } else if matches!(s.kind, ShapeKind::Picture { .. }) {
49            // The largest picture leads.
50            let area = |j: usize| shapes.get(j).and_then(|s| s.xfrm).map(|x| x.w * x.h).unwrap_or(0.0);
51            if r.picture.is_none_or(|p| area(i) > area(p)) {
52                r.picture = Some(i);
53            }
54        } else if s.text.as_ref().is_some_and(|t| !t.text().trim().is_empty()) {
55            r.text.push(i);
56        }
57    }
58    r
59}

61/// Whether `s` paints a fill or an outline once inherited and style-referenced values are
62/// resolved, as the renderer does (`fill: None` alone means inherited, not invisible).
63fn shows(ctx: Option<&Ctx>, s: &Shape) -> bool {
64    let visible = |f: &Option<Fill>| f.as_ref().is_some_and(|f| !f.is_none());
65    match ctx {
66        Some(ctx) => visible(&resolve::fill(ctx, s).0) || visible(&resolve::line(ctx, s).0.fill),
67        None => visible(&s.fill) || visible(&s.line.as_ref().and_then(|l| l.fill.clone())),
68    }
69}

71/// Whether an idea may move this shape: a picture, a placeholder or a plain text box. Diagram
72/// boxes, connectors, groups, tables, charts and media are laid out by their own geometry, which
73/// the ideas do not know, so a slide holding one gets no ideas.
74fn arrangeable(ctx: Option<&Ctx>, s: &Shape) -> bool {
75    match s.kind {
76        ShapeKind::Picture { .. } => true,
77        ShapeKind::Shape => {
78            let is_title = s.ph.as_ref().is_some_and(|p| p.kind.is_title());
79            let has_text = s.text.as_ref().is_some_and(|t| !t.text().trim().is_empty());
80            // Only titles and shapes with text are moved (see `roles`): an empty one that shows
81            // (filled or outlined) would stay where it is, over the rearranged slide.
82            (s.ph.is_some() || s.text_box) && (is_title || has_text || !shows(ctx, s))
83        }
84        _ => false,
85    }
86}

88/// The ideas for slide `index` (none when it has nothing to arrange, or other content in the way).
89pub fn ideas_for(doc: &Presentation, index: usize) -> Vec<Idea> {
90    let Some(slide) = doc.slides.get(index) else { return vec![] };
91    let ctx = Ctx::for_slide(doc, slide);
92    if slide.shapes.iter().any(|s| !s.hidden && !arrangeable(ctx.as_ref(), s)) {
93        return vec![];
94    }
95    // The ideas move one picture; a second would stay put, over the rearranged text.
96    if slide.shapes.iter().filter(|s| !s.hidden && matches!(s.kind, ShapeKind::Picture { .. })).count() > 1 {
97        return vec![];
98    }
99    let r = roles(&slide.shapes);
100    if r.picture.is_some() {
101        PICTURE_IDEAS.to_vec()
102    } else if r.title.is_some() && !r.text.is_empty() {
103        TEXT_IDEAS.to_vec()
104    } else {
105        vec![]
106    }
107}

109/// Crop a picture whose source has aspect `src` (w/h) to fill a box of aspect `dst`, centred.
110fn cover_crop(src: f64, dst: f64) -> [f64; 4] {
111    if !(src.is_finite() && dst.is_finite() && src > 0.0 && dst > 0.0) {
112        return [0.0; 4];
113    }
114    if src > dst {
115        let cut = (1.0 - dst / src) / 2.0;
116        [cut, 0.0, cut, 0.0]
117    } else {
118        let cut = (1.0 - src / dst) / 2.0;
119        [0.0, cut, 0.0, cut]
120    }
121}

123/// The light fill behind the title in the full-bleed idea. Its opacity is an `alpha` transform
124/// (about 220/255), as PPTX stores it, so the fill compares equal after a save and reopen.
125fn band_fill() -> Fill {
126    Fill::solid(crate::model::ColorRef::rgb(Rgba::rgb(255, 255, 255)).with(crate::model::ColorTransform::Alpha(86274)))
127}
128
129/// The band as released versions up to 0.10 stored it (opacity in the colour's own alpha).
130fn legacy_band_fill() -> Fill {
131    Fill::solid(crate::model::ColorRef::rgb(Rgba::rgba(255, 255, 255, 220)))
132}
133
134/// Whether `f` is a Full-bleed band, in the current or the legacy form.
135fn is_band(f: &Fill) -> bool {
136    *f == band_fill() || *f == legacy_band_fill()
137}

139/// Apply idea `id` to slide `index` of `doc` (`sel` resolves inherited placeholder boxes).
140pub fn apply_idea(doc: &mut Presentation, sel: &Selection, index: usize, id: &str) -> Result<()> {
141    if !ideas_for(doc, index).iter().any(|i| i.0 == id) {
142        return Err(bad("design.applyIdea", format!("no idea `{id}` for this slide (see design.ideas)")));
143    }
144    let (sw, sh) = (doc.slide_size.width.max(1.0), doc.slide_size.height.max(1.0));
145    let sel = Selection { slide: index, ..sel.clone() };
146    let shapes = doc.slides.get(index).map(|s| s.shapes.clone()).ok_or_else(|| bad("design.applyIdea", "no slide"))?;
147    let r = roles(&shapes);
148    let boxes: Vec<Xfrm> = shapes.iter().map(|s| xfrm_of(doc, &sel, s)).collect();
149    let m = sw.min(sh) * 0.06;
150    let mut place: Vec<(usize, Xfrm)> = vec![];
151    // Text shapes stacked in a box.
152    let stack = |b: Xfrm, out: &mut Vec<(usize, Xfrm)>| {
153        let n = r.text.len().max(1) as f64;
154        let gap = m * 0.4;
155        let h = ((b.h - gap * (n - 1.0)) / n).max(1.0);
156        for (k, i) in r.text.iter().enumerate() {
157            out.push((*i, Xfrm::new(b.x, b.y + k as f64 * (h + gap), b.w, h)));
158        }
159    };
160    let mut band = false;
161    match id {
162        "fullBleed" => {
163            if let Some(p) = r.picture {
164                place.push((p, Xfrm::new(0.0, 0.0, sw, sh)));
165            }
166            if let Some(t) = r.title {
167                place.push((t, Xfrm::new(m, sh * 0.68, sw - 2.0 * m, sh * 0.2)));
168                band = true;
169            }
170            stack(Xfrm::new(m, sh * 0.06, sw * 0.45, sh * 0.5), &mut place);
171        }
172        "pictureLeft" | "pictureRight" => {
173            let left = id == "pictureLeft";
174            let half = sw * 0.5;
175            let (px, tx) = if left { (0.0, half + m) } else { (half, m) };
176            if let Some(p) = r.picture {
177                place.push((p, Xfrm::new(px, 0.0, half, sh)));
178            }
179            let tw = half - 2.0 * m;
180            if let Some(t) = r.title {
181                place.push((t, Xfrm::new(tx, m, tw, sh * 0.24)));
182            }
183            stack(Xfrm::new(tx, m + sh * 0.28, tw, sh - 2.0 * m - sh * 0.28), &mut place);
184        }
185        "framed" => {
186            if let Some(p) = r.picture {
187                place.push((p, Xfrm::new(sw * 0.42, m * 1.5, sw * 0.58 - m * 1.5, sh - 3.0 * m)));
188            }
189            let tw = sw * 0.42 - 2.0 * m;
190            if let Some(t) = r.title {
191                place.push((t, Xfrm::new(m, m * 1.5, tw, sh * 0.3)));
192            }
193            stack(Xfrm::new(m, m * 1.5 + sh * 0.34, tw, sh - 3.0 * m - sh * 0.34), &mut place);
194        }
195        "titleColumn" => {
196            if let Some(t) = r.title {
197                place.push((t, Xfrm::new(m, m, sw * 0.32, sh - 2.0 * m)));
198            }
199            stack(Xfrm::new(sw * 0.36, m, sw * 0.64 - m, sh - 2.0 * m), &mut place);
200        }
201        "centred" => {
202            if let Some(t) = r.title {
203                place.push((t, Xfrm::new(sw * 0.1, m, sw * 0.8, sh * 0.26)));
204            }
205            stack(Xfrm::new(sw * 0.18, m + sh * 0.3, sw * 0.64, sh - 2.0 * m - sh * 0.3), &mut place);
206        }
207        "titleBottom" => {
208            stack(Xfrm::new(m, m, sw - 2.0 * m, sh * 0.62), &mut place);
209            if let Some(t) = r.title {
210                place.push((t, Xfrm::new(m, sh * 0.72, sw - 2.0 * m, sh * 0.2)));
211            }
212        }
213        _ => return Err(bad("design.applyIdea", format!("no idea `{id}`"))),
214    }
215    let slide = doc.slides.get_mut(index).ok_or_else(|| bad("design.applyIdea", "no slide"))?;
216    let slide = std::sync::Arc::make_mut(slide);
217    for (i, x) in place {
218        let (Some(s), Some(old)) = (slide.shapes.get_mut(i), boxes.get(i)) else { continue };
219        if let ShapeKind::Picture { fill } = &mut s.kind {
220            // The source's aspect, from the current box and crop.
221            let c = fill.crop;
222            let (fw, fh) = ((1.0 - c[0] - c[2]).max(0.01), (1.0 - c[1] - c[3]).max(0.01));
223            let src = (old.w / fw) / (old.h / fh).max(1e-6);
224            fill.crop = cover_crop(src, x.w / x.h.max(1e-6));
225        }
226        s.xfrm = Some(Xfrm { rot: 0.0, ..x });
227    }
228    // The picture goes to the back, so text over it stays readable.
229    if let Some(p) = r.picture
230        && p < slide.shapes.len()
231    {
232        let pic = slide.shapes.remove(p);
233        slide.shapes.insert(0, pic);
234    }
235    if let Some(t) = slide.shapes.iter_mut().find(|s| !s.hidden && s.ph.as_ref().is_some_and(|p| p.kind.is_title())) {
236        if band {
237            t.fill = Some(band_fill());
238        } else if t.fill.as_ref().is_some_and(is_band) {
239            // The band belongs to full-bleed only: another idea takes it away again.
240            t.fill = None;
241        }
242    }
243    Ok(())
244}

246fn ideas_cmd(s: &mut Session, p: &Value) -> Result<Value> {
247    let st = s.doc()?;
248    let i = opt_usize_param(p, "slide", "design.ideas")?.unwrap_or(st.selection.slide);
249    Ok(json!(ideas_for(&st.doc, i).iter().map(|(id, label)| json!({"id": id, "label": label})).collect::<Vec<_>>()))
250}
251
252fn apply_cmd(s: &mut Session, p: &Value) -> Result<Value> {
253    let id = str_param(p, "id").ok_or_else(|| bad("design.applyIdea", "missing `id`"))?.to_string();
254    let slide = opt_usize_param(p, "slide", "design.applyIdea")?;
255    s.edit(|doc, sel| {
256        let i = slide.unwrap_or(sel.slide);
257        apply_idea(doc, sel, i, &id)?;
258        Ok(json!({"slide": i, "id": id}))
259    })
260}

312    fn overlaps(a: Xfrm, b: Xfrm) -> bool {
313        a.x < b.x + b.w - 0.5 && b.x < a.x + a.w - 0.5 && a.y < b.y + b.h - 0.5 && b.y < a.y + a.h - 0.5
314    }
