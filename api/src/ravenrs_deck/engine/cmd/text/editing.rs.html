8pub(super) fn edit(s: &mut Session, p: &Value) -> Result<Value> {
9    if bool_or(p, "notes", false) {
10        let end = s.doc()?.current_slide().map(|x| ed::end(&x.notes)).unwrap_or((0, 0));
11        s.select(|_, sel| {
12            sel.text = Some(TextSel { notes: true, anchor: end, caret: end, ..Default::default() });
13            sel.shapes.clear();
14            sel.cells = None;
15        })?;
16        return ok();
17    }
18    let id = match id_param(p, "id", "text.edit")? {
19        Some(i) => i,
20        None => s.doc()?.selection.shapes.first().copied().ok_or_else(|| bad("text.edit", "select a shape with text"))?,
21    };
22    let sh = shape_of(s, id)?;
23    let cell = p.get("cell").and_then(Value::as_array).and_then(|a| Some((a.first()?.as_u64()? as usize, a.get(1)?.as_u64()? as usize)));
24    let body = match (&sh.kind, cell) {
25        (ShapeKind::Table(t), Some((r, c))) => t.cell(r, c).map(|x| x.text.clone()).ok_or_else(|| bad("text.edit", "no such cell"))?,
26        (ShapeKind::Table(_), None) => return Err(bad("text.edit", "say which `cell` of the table")),
27        (ShapeKind::Shape | ShapeKind::Connector { .. }, _) => sh.text.clone().unwrap_or_default(),
28        _ => return Err(bad("text.edit", "this object doesn't hold text")),
29    };
30    let at = pos_param(p, "at").unwrap_or_else(|| ed::end(&body));
31    // A shape without a text body gets one on first edit.
32    if sh.text.is_none() && cell.is_none() {
33        s.edit(|doc, sel| {
34            if let Some(list) = crate::engine::shapes_mut(doc, sel)
35                && let Some(x) = crate::model::find_shape_mut(list, id)
36            {
37                x.text = Some(TextBody::default());
38            }
39            Ok(())
40        })?;
41    }
42    s.select(|_, sel| {
43        sel.shapes = vec![id];
44        sel.text = Some(TextSel { shape: id, anchor: at, caret: at, cell, notes: false, line_end: None });
45        sel.cells = None;
46    })?;
47    Ok(json!({"id": id, "at": [at.0, at.1]}))
48}

50pub(super) fn exit(s: &mut Session, _p: &Value) -> Result<Value> {
51    s.select(|_, sel| {
52        if let Some(t) = sel.text.take()
53            && !t.notes
54        {
55            sel.shapes = vec![t.shape];
56        }
57    })?;
58    ok()
59}

61/// Smart quotes and simple autocorrect applied to typed text.
62fn typed(s: &Session, body: &TextBody, at: Pos, text: &str) -> String {
63    if !s.prefs.smart_quotes || text.chars().count() != 1 {
64        return text.to_string();
65    }
66    let prev = body.paragraphs.get(at.0).and_then(|p| p.text().chars().nth(at.1.wrapping_sub(1)));
67    let opening = prev.is_none_or(|c| c.is_whitespace() || "([{".contains(c));
68    match text {
69        "\"" => (if opening { "“" } else { "”" }).to_string(),
70        "'" => (if opening { "‘" } else { "’" }).to_string(),
71        _ => text.to_string(),
72    }
73}

75pub(crate) fn insert_text(s: &mut Session, text: &str, props: Option<RunProps>) -> Result<Value> {
76    let inherited = inherited_aligns_for_edit(s);
77    let text = {
78        let st = s.doc()?;
79        match &st.selection.text {
80            Some(t) => {
81                let body = body_of(st, t).cloned().unwrap_or_default();
82                typed(s, &body, t.ordered().0, text)
83            }
84            None => text.to_string(),
85        }
86    };
87    let pos = edit_text(s, |body, t| {
88        let (a, b) = t.ordered();
89        let at = ed::delete(body, a, b);
90        let props = props.clone().or_else(|| if a != b { Some(ed::props_at(body, at)) } else { unlinked_at_link_end(body, at) });
91        let c = ed::insert(body, at, &text, props);
92        t.anchor = c;
93        t.caret = c;
94        let last = body.paragraphs.len().saturating_sub(1);
95        for pi in at.0..=c.0.min(last) {
96            if let Some(para) = body.paragraphs.get_mut(pi) {
97                promote_rtl_if_needed(para, &inherited);
98            }
99        }
100        c
101    })?;
102    autocorrect(s, text.chars().count() == 1)?;
103    // AutoCorrect can move the caret: a replaced word, or a list lead it removed.
104    let pos = s.doc()?.selection.text.as_ref().map_or(pos, |t| t.caret);
105    Ok(json!({"caret": [pos.0, pos.1]}))
106}

108/// Per outline level (0..=8), whether shape `id` on the current slide already resolves an
109/// alignment for its paragraphs (its own list style, the layout's or the master's). All `false`
110/// when the shape can't be resolved (notes, a shape that is gone).
111pub(super) fn inherited_aligns(s: &Session, id: ShapeId) -> [bool; 9] {
112    let mut out = [false; 9];
113    let Ok(st) = s.doc() else { return out };
114    let Some(sh) = st.shape(id) else { return out };
115    let Some(ctx) = ctx_for(&st.doc, &st.selection) else { return out };
116    for (lvl, set) in out.iter_mut().enumerate() {
117        let para = crate::model::text::Paragraph { level: lvl as u8, ..Default::default() };
118        *set = resolve::para(&ctx, sh, &para).align.is_some();
119    }
120    out
121}

123/// The aligns [`inherited_aligns`] gives for the text being edited (none for notes).
124fn inherited_aligns_for_edit(s: &Session) -> [bool; 9] {
125    match s.doc().ok().and_then(|st| st.selection.text.clone()) {
126        Some(t) if !t.notes => inherited_aligns(s, t.shape),
127        _ => [false; 9],
128    }
129}

131/// A paragraph with no direction of its own whose first strong character is right-to-left
132/// (Arabic, Hebrew, Kurdish…) becomes right-to-left, and right-aligned when nothing (the
133/// paragraph or what it inherits, `inherited`) sets its alignment.
134pub(super) fn promote_rtl_if_needed(para: &mut crate::model::text::Paragraph, inherited: &[bool; 9]) {
135    if para.props.rtl.is_none() && crate::model::bidi::first_strong_is_rtl(&para.text()) {
136        para.props.rtl = Some(true);
137        let inherits_align = inherited.get(usize::from(para.level)).copied().unwrap_or(false);
138        if para.props.align.is_none() && !inherits_align {
139            para.props.align = Some(Align::Right);
140        }
141    }
142}

144/// Typing right after a hyperlink doesn't extend it (as in PowerPoint): at the end of a linked
145/// run, the props to type with are the run's without the link. `None` elsewhere.
146pub(super) fn unlinked_at_link_end(body: &TextBody, at: Pos) -> Option<RunProps> {
147    let mut props = ed::props_at(body, at);
148    props.link.as_ref()?;
149    let len = body.paragraphs.get(at.0).map_or(0, |p| p.text().chars().count());
150    // props_at(c + 1) is the props of the character at c.
151    if at.1 < len && ed::props_at(body, (at.0, at.1 + 1)).link == props.link {
152        return None;
153    }
154    props.link = None;
155    Some(props)
156}

158pub(crate) fn paste_body(s: &mut Session, src: &TextBody) -> Result<Value> {
159    let inherited = inherited_aligns_for_edit(s);
160    let pos = edit_text(s, |body, t| {
161        let (a, b) = t.ordered();
162        let at = ed::delete(body, a, b);
163        let c = ed::insert_body(body, at, src);
164        t.anchor = c;
165        t.caret = c;
166        let last = body.paragraphs.len().saturating_sub(1);
167        for pi in at.0..=c.0.min(last) {
168            if let Some(para) = body.paragraphs.get_mut(pi) {
169                promote_rtl_if_needed(para, &inherited);
170            }
171        }
172        c
173    })?;
174    Ok(json!({"caret": [pos.0, pos.1]}))
175}

177pub(crate) fn insert_field(s: &mut Session, field: &str, text: &str) -> Result<Value> {
178    let field = field.to_string();
179    let text = if text.is_empty() { "#".to_string() } else { text.to_string() };
180    edit_text(s, move |body, t| {
181        let (a, b) = t.ordered();
182        let at = ed::delete(body, a, b);
183        let props = ed::props_at(body, at);
184        let src = TextBody {
185            paragraphs: vec![crate::model::text::Paragraph {
186                runs: vec![Run { text: text.clone(), props, kind: RunKind::Field { field: field.clone() } }],
187                ..Default::default()
188            }],
189            ..Default::default()
190        };
191        let c = ed::insert_body(body, at, &src);
192        t.anchor = c;
193        t.caret = c;
194    })?;
195    ok()
196}

198#[cfg(test)]
199mod rtl_promote_tests {
200    use crate::model::text::{Align, Paragraph, Run};
201
202    use super::promote_rtl_if_needed;
203
204    fn para(text: &str) -> Paragraph {
205        Paragraph { runs: vec![Run::new(text)], ..Default::default() }
206    }
207
208    #[test]
209    fn right_aligns_only_when_no_alignment_is_set_or_inherited() {
210        let mut p = para("مرحبا");
211        promote_rtl_if_needed(&mut p, &[false; 9]);
212        assert_eq!((p.props.rtl, p.props.align), (Some(true), Some(Align::Right)));
213        let mut p = para("مرحبا");
214        promote_rtl_if_needed(&mut p, &[true; 9]);
215        assert_eq!((p.props.rtl, p.props.align), (Some(true), None));
216        let mut p = para("Hello مرحبا");
217        promote_rtl_if_needed(&mut p, &[false; 9]);
218        assert_eq!((p.props.rtl, p.props.align), (None, None));
219    }
