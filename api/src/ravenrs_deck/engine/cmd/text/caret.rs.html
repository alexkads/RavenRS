10/// A table cell's text laid out the way it is drawn, mapped into slide coordinates.
11pub(super) fn cell_on_slide(st: &DocState, id: ShapeId, row: usize, col: usize) -> Option<CellText> {
12    let sh = st.shape(id)?;
13    let ShapeKind::Table(tb) = &sh.kind else { return None };
14    let cell = tb.cell(row, col)?;
15    let ctx = ctx_for(&st.doc, &st.selection)?;
16    let heights = crate::engine::cmd::table_drawn_heights(&st.doc, &st.selection, tb);
17    let rect = crate::render::table_cell_rects(tb, &heights).into_iter().find(|(r, c, _)| *r == row && *c == col)?.2;
18    let mut body = crate::render::table_cell_body(&ctx, tb, row, col, cell);
19    if body.paragraphs.is_empty() {
20        body.paragraphs.push(crate::model::text::Paragraph::default());
21    }
22    let tmp = crate::model::Shape::table_cell(body.clone());
23    let layout =
24        crate::text::layout(&ctx, &tmp, &body, &crate::text::Opts { rect, fields: &crate::text::NoFields, prompt_color: None, no_shrink: false });
25    let x = xfrm_of(&st.doc, &st.selection, sh);
26    let to_slide = crate::render::text_transform(x.affine(), rect, layout.rotation);
27    Some(CellText { layout, to_slide })
28}

40pub(crate) fn delete_dir(s: &mut Session, dir: &str) -> Result<Value> {
41    if dir == "backward" && undo_auto_list(s)? {
42        return ok();
43    }
44    let dir = dir.to_string();
45    edit_text(s, move |body, t| {
46        let (a, b) = t.ordered();
47        let at = if dir == "wordBackward" || dir == "wordForward" {
48            // Ctrl+Backspace/Delete work from the start of a selection and leave the rest of it
49            // (PowerPoint for Windows).
50            ed::delete_word(body, a, dir == "wordForward")
51        } else if a != b {
52            ed::delete(body, a, b)
53        } else {
54            let other = match dir.as_str() {
55                "forward" => ed::cluster_end(body, a),
56                _ => {
57                    // Backspace at the start of an indented/bulleted paragraph outdents first.
58                    if a.1 == 0
59                        && let Some(para) = body.paragraphs.get_mut(a.0)
60                        && para.level > 0
61                    {
62                        para.level -= 1;
63                        t.anchor = a;
64                        t.caret = a;
65                        return;
66                    }
67                    ed::char_move(body, a, false)
68                }
69            };
70            ed::delete(body, a, other)
71        };
72        t.anchor = at;
73        t.caret = at;
74    })?;
75    ok()
76}

78/// A `before`/`after` count from a control or input-method request.
79///
80/// Input-derived numbers are hostile: a value that does not fit `usize` (on 32-bit wasm, say) is
81/// clamped to `usize::MAX` and then bounded by the walk, never truncated to a smaller count that
82/// would delete different text than asked.
83pub(super) fn count_param(p: &Value, name: &str) -> usize {
84    p.get(name).and_then(Value::as_u64).map_or(0, |v| usize::try_from(v).unwrap_or(usize::MAX))
85}

87/// Delete at most the requested character or byte counts around the selection or pre-edit,
88/// preserving selection direction and composition boundaries and recording one undo step
89/// outside composition.
90pub(super) fn delete_surrounding(s: &mut Session, p: &Value) -> Result<Value> {
91    let before = count_param(p, "before");
92    let after = count_param(p, "after");
93    let bytes = bool_or(p, "bytes", false);
94    let st = s.doc()?;
95    let t = st.selection.text.clone().ok_or_else(|| bad("text.deleteSurrounding", "not editing text"))?;
96    let body = body_of(st, &t).ok_or_else(|| bad("text.deleteSurrounding", "the text being edited is gone"))?;
97    let comp = composition_range(body, st.composition.as_ref());
98    // The ranges are in the same logical coordinates `ime_surrounding` publishes: one byte for a
99    // paragraph separator and for a soft line break. A range may therefore cross paragraphs.
100    let ordered = {
101        let a = model_pos(body, t.anchor).ok_or_else(|| bad("text.deleteSurrounding", "the selection is not in the text"))?;
102        let b = model_pos(body, t.caret).ok_or_else(|| bad("text.deleteSurrounding", "the selection is not in the text"))?;
103        if a <= b { (a, b) } else { (b, a) }
104    };
105    // The selection (or the input method's pre-edit) stays exactly where it is. Only whole
106    // scalars fully inside the requested before/after ranges around it are removed.
107    let (start, end) = match comp {
108        Some((pi, cs, ce)) => ((pi, cs), (pi, ce)),
109        None => ordered,
110    };
111    let (from, before_chars, _) = walk_backward(body, comp, start, before, bytes);
112    let (to, after_chars, _) = walk_forward(body, comp, end, after, bytes);
113    let deleted = before_chars + after_chars;
114    if deleted == 0 {
115        return Ok(json!({"deleted": 0}));
116    }
117    let before_doc = st.doc.clone();
118    let before_sel = st.selection.clone();
119    let composing = st.composition.is_some();
120    let vertex_kind = s.tool.vertex_kind.clone();
121    edit_text(s, |body, t| {
122        // Delete the after-range first, so the positions of the earlier range stay valid; then
123        // the text before the selection.
124        if after_chars > 0 {
125            ed::delete(body, end, to);
126        }
127        if before_chars > 0 {
128            ed::delete(body, from, start);
129        }
130        t.anchor = remap_after_delete(t.anchor, start, from, before_chars);
131        t.caret = remap_after_delete(t.caret, start, from, before_chars);
132        t.line_end = None;
133    })?;
134    if composing {
135        if let Some(c) = s.doc_mut()?.composition.as_mut() {
136            c.start = if before_chars == 0 {
137                start
138            } else if from.0 == start.0 {
139                (start.0, start.1 - before_chars)
140            } else {
141                from
142            };
143        }
144    } else {
145        push_undo(s.doc_mut()?, HistoryEntry { label: "Delete Text".into(), doc: before_doc, selection: before_sel, vertex_kind });
146    }
147    Ok(json!({"deleted": deleted}))
148}

150/// Map a text position across the deletion of `before_chars` characters that begins at `start`
151/// and reaches back to `from`. Within the start paragraph its characters shift; from an earlier
152/// paragraph those paragraphs merge, so later paragraph numbers shift too.
153fn remap_after_delete(pos: (usize, usize), start: (usize, usize), from: (usize, usize), before_chars: usize) -> (usize, usize) {
154    if before_chars == 0 || pos < start {
155        return pos;
156    }
157    if from.0 == start.0 {
158        // Only endpoints in the start paragraph move; later paragraphs keep their text.
159        if pos.0 == start.0 { (pos.0, pos.1 - before_chars) } else { pos }
160    } else {
161        (from.0 + (pos.0 - start.0), if pos.0 == start.0 { from.1 + (pos.1 - start.1) } else { pos.1 })
162    }
163}

169/// Is paragraph `para` right-to-left (its lines run from the right)?
170fn para_rtl(layout: Option<&crate::text::TextLayout>, body: &TextBody, para: usize) -> bool {
171    match layout.and_then(|l| l.lines.iter().find(|x| x.para == para)) {
172        Some(line) => line.rtl,
173        None => body.paragraphs.get(para).and_then(|x| x.props.rtl).unwrap_or(false),
174    }
175}

177pub(super) fn move_caret(s: &mut Session, p: &Value) -> Result<Value> {
178    use crate::text::Affinity;
179    let to = str_param(p, "to").unwrap_or("right").to_string();
180    let extend = bool_or(p, "extend", false);
181    let st = s.doc()?;
182    let t = st.selection.text.clone().ok_or_else(|| bad("text.move", "not editing text"))?;
183    let body = body_of(st, &t).cloned().unwrap_or_default();
184    let layout = layout_for(st, &t);
185    let lp = |c: Pos| crate::text::Pos { para: c.0, ch: c.1 };
186    let caret = t.caret;
187    let aff = if t.at_line_end() { Affinity::Upstream } else { Affinity::Downstream };
188    // Left and right are on screen: in a right-to-left paragraph left goes forward in the text.
189    let rtl = para_rtl(layout.as_ref(), &body, caret.0);
190    let to = match (to.as_str(), rtl) {
191        ("left", true) => "right",
192        ("right", true) => "left",
193        ("wordLeft", true) => "wordRight",
194        ("wordRight", true) => "wordLeft",
195        (x, _) => x,
196    }
197    .to_string();
198    let collapse_to = |left: bool| if left { t.ordered().0 } else { t.ordered().1 };
199    let mut upstream = false;
200    let new = match to.as_str() {
201        "left" if t.is_range() && !extend => collapse_to(true),
202        "right" if t.is_range() && !extend => collapse_to(false),
203        "left" => ed::char_move(&body, caret, false),
204        "right" => ed::char_move(&body, caret, true),
205        // Ctrl+arrows with a selection go on from its start (left) or end (right).
206        "wordLeft" => ed::word_move(&body, if extend { caret } else { collapse_to(true) }, false),
207        "wordRight" => ed::word_move(&body, if extend { caret } else { collapse_to(false) }, true),
208        "paraStart" => (caret.0, 0),
209        "paraEnd" => (caret.0, body.paragraphs.get(caret.0).map(|x| x.char_len()).unwrap_or(0)),
210        "paraUp" => ed::para_up(&body, caret),
211        "paraDown" => ed::para_down(&body, caret),
212        "start" => (0, 0),
213        "end" => ed::end(&body),
214        "up" | "down" | "lineStart" | "lineEnd" => match &layout {
215            Some(l) => {
216                let x = l.caret_drawn(lp(caret), aff).map(|c| c.0).unwrap_or(0.0);
217                let (r, a) = match to.as_str() {
218                    "up" => l.vertical_affine(lp(caret), aff, false, x),
219                    "down" => l.vertical_affine(lp(caret), aff, true, x),
220                    "lineStart" => (l.line_bounds_affine(lp(caret), aff).0, Affinity::Downstream),
221                    // End stays on its line: after a soft wrap the caret sits at the line's end.
222                    _ => l.line_end_caret(lp(caret), aff),
223                };
224                upstream = a == Affinity::Upstream;
225                (r.para, r.ch)
226            }
227            None => match to.as_str() {
228                "up" => (caret.0.saturating_sub(1), caret.1),
229                "down" => (caret.0.saturating_add(1).min(body.paragraphs.len().saturating_sub(1)), caret.1),
230                "lineStart" => (caret.0, 0),
231                _ => (caret.0, body.paragraphs.get(caret.0).map(|x| x.char_len()).unwrap_or(0)),
232            },
233        },
234        other => return Err(bad("text.move", format!("unknown direction `{other}`"))),
235    };
236    let new = (new.0.min(body.paragraphs.len().saturating_sub(1)), new.1.min(body.paragraphs.get(new.0).map(|x| x.char_len()).unwrap_or(0)));
237    s.select(|_, sel| {
238        if let Some(tt) = sel.text.as_mut() {
239            tt.caret = new;
240            tt.line_end = if upstream { Some(new) } else { None };
241            if !extend {
242                tt.anchor = new;
243            }
244        }
245    })?;
246    Ok(json!({"caret": [new.0, new.1], "lineEnd": upstream}))
247}

289pub(super) fn select_para(s: &mut Session, p: &Value) -> Result<Value> {
290    let st = s.doc()?;
291    let t = st.selection.text.clone().ok_or_else(|| bad("text.selectParagraph", "not editing text"))?;
292    let at = pos_param(p, "at").unwrap_or(t.caret);
293    let body = body_of(st, &t);
294    // Clamp to the last paragraph so a huge index can neither overflow nor select past the body.
295    let at = (at.0.min(body.map_or(0, |b| b.paragraphs.len().saturating_sub(1))), at.1);
296    let n = body.and_then(|b| b.paragraphs.get(at.0)).map(|x| x.char_len()).unwrap_or(0);
297    // The paragraph mark is part of the paragraph (PowerPoint's triple-click selects it too).
298    let end = if body.is_some_and(|b| at.0.saturating_add(1) < b.paragraphs.len()) { (at.0.saturating_add(1), 0) } else { (at.0, n) };
299    s.select(|_, sel| {
300        if let Some(t) = sel.text.as_mut() {
301            t.anchor = (at.0, 0);
302            t.caret = end;
303        }
304    })?;
305    ok()
306}

308/// Turn leading tabs of a paragraph into its outline level (capped at 8) and strip them.
309/// Returns whether any tab was found.
310pub(crate) fn apply_leading_tabs(para: &mut crate::model::text::Paragraph) -> bool {
311    let tabs = para.runs.first().map(|r| r.text.chars().take_while(|c| *c == '\t').count()).unwrap_or(0);
312    if tabs == 0 {
313        return false;
314    }
315    if let Some(r) = para.runs.first_mut() {
316        r.text = r.text.trim_start_matches('\t').to_string();
317    }
318    para.level = tabs.min(8) as u8;
319    true
320}

322pub(super) fn set(s: &mut Session, p: &Value) -> Result<Value> {
323    let text = str_param(p, "text").ok_or_else(|| bad("text.set", "missing `text`"))?.to_string();
324    let id = match id_param(p, "id", "text.set")? {
325        Some(i) => i,
326        None => s.doc()?.selection.shapes.first().copied().ok_or_else(|| bad("text.set", "missing `id` (or select a shape)"))?,
327    };
328    let cell = p.get("cell").and_then(Value::as_array).and_then(|a| Some((a.first()?.as_u64()? as usize, a.get(1)?.as_u64()? as usize)));
329    let inherited = inherited_aligns(s, id);
330    edit_shapes(s, &json!({"id": id}), "text.set", |sh| {
331        let target = match (&mut sh.kind, cell) {
332            (ShapeKind::Table(t), Some((r, c))) => t.cell_mut(r, c).map(|x| &mut x.text),
333            (ShapeKind::Table(_), None) => None,
334            _ => Some(sh.text.get_or_insert_with(TextBody::default)),
335        };
336        let Some(body) = target else { return Err(bad("text.set", "no text here (tables need `cell`)")) };
337        let keep = body.paragraphs.first().and_then(|p| p.runs.first()).map(|r| r.props.clone()).unwrap_or_default();
338        let levels: Vec<(u8, crate::model::text::ParaProps)> = body.paragraphs.iter().map(|p| (p.level, p.props.clone())).collect();
339        let mut nb = TextBody::from_text(&text);
340        for (i, para) in nb.paragraphs.iter_mut().enumerate() {
341            // Leading tabs set the outline level.
342            if !apply_leading_tabs(para)
343                && let Some((lv, pp)) = levels.get(i)
344            {
345                para.level = *lv;
346                para.props = pp.clone();
347            }
348            promote_rtl_if_needed(para, &inherited);
349            for r in &mut para.runs {
350                r.props = keep.clone();
351            }
352        }
353        body.paragraphs = nb.paragraphs;
354        Ok(())
355    })?;
356    fit_text_box(s, id)?;
357    ok()
358}

360pub(super) fn get(s: &mut Session, p: &Value) -> Result<Value> {
361    let st = s.doc()?;
362    let id = match id_param(p, "id", "text.get")? {
363        Some(i) => i,
364        None => match &st.selection.text {
365            Some(t) => t.shape,
366            None => st.selection.shapes.first().copied().ok_or_else(|| bad("text.get", "missing `id`"))?,
367        },
368    };
369    let sh = st.shape(id).ok_or_else(|| bad("text.get", "no such shape"))?;
370    let t = sh.text.clone().unwrap_or_default();
371    Ok(json!({"text": t.text(), "paragraphs": t.paragraphs.iter().map(|p| json!({"level": p.level, "text": p.text()})).collect::<Vec<_>>()}))
372}
