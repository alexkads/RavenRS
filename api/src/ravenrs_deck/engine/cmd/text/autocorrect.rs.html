74/// The replacement for `word` from your entries, then the built-in list.
75fn replacement(prefs: &crate::engine::Prefs, word: &str) -> Option<String> {
76    if let Some((_, to)) = prefs.ac_entries.iter().find(|(k, _)| k == word) {
77        return Some(to.clone());
78    }
79    if let Some((_, to)) = prefs.ac_entries.iter().find(|(k, _)| k.eq_ignore_ascii_case(word)) {
80        return Some(to.clone());
81    }
82    let (k, to) = AUTOCORRECT.iter().find(|(k, _)| k.eq_ignore_ascii_case(word))?;
83    if prefs.ac_removed.iter().any(|r| r.eq_ignore_ascii_case(k)) {
84        return None;
85    }
86    // "i" only on its own, lower case.
87    if *k == "i" && word != "i" {
88        return None;
89    }
90    Some(if word.chars().next().is_some_and(char::is_uppercase) { ed::change_case(to, "sentence") } else { to.to_string() })
91}

93/// What a just-typed `word` becomes (`None`: unchanged). `sentence_start`: nothing but spaces
94/// or a sentence's end (`.`, `!`, `?`) before it in the paragraph. `typing`: one character was
95/// typed (the case rules apply only then).
96pub fn correct_word(prefs: &crate::engine::Prefs, word: &str, sentence_start: bool, typing: bool) -> Option<String> {
97    let mut out = replacement(prefs, word);
98    if typing && out.is_none() {
99        let chars: Vec<char> = word.chars().collect();
100        let letters = chars.iter().all(|c| c.is_alphabetic());
101        if prefs.ac_fractions {
102            out = match word {
103                "1/2" => Some("½".into()),
104                "1/4" => Some("¼".into()),
105                "3/4" => Some("¾".into()),
106                _ => None,
107            };
108        }
109        if out.is_none() && letters && chars.len() >= 3 {
110            let upper = |c: &char| c.is_uppercase();
111            let rest_lower = chars.get(2..).is_some_and(|r| r.iter().all(|c| c.is_lowercase()));
112            let rest_upper = chars.get(1..).is_some_and(|r| r.iter().all(upper));
113            if prefs.ac_two_initials && chars.first().is_some_and(upper) && chars.get(1).is_some_and(upper) && rest_lower {
114                out =
115                    Some(chars.iter().enumerate().map(|(i, c)| if i == 1 { c.to_lowercase().collect::<String>() } else { c.to_string() }).collect());
116            } else if prefs.ac_caps_lock && chars.first().is_some_and(|c| c.is_lowercase()) && rest_upper {
117                out = Some(
118                    chars
119                        .iter()
120                        .enumerate()
121                        .map(|(i, c)| if i == 0 { c.to_uppercase().collect::<String>() } else { c.to_lowercase().collect() })
122                        .collect(),
123                );
124            }
125        }
126        if out.is_none() && prefs.ac_days && DAYS.contains(&word) {
127            out = Some(ed::change_case(word, "sentence"));
128        }
129    }
130    if typing && prefs.ac_sentence_caps && sentence_start {
131        let w = out.clone().unwrap_or_else(|| word.to_string());
132        let mut cs = w.chars();
133        if let Some(f) = cs.next()
134            && f.is_lowercase()
135            && w.chars().all(char::is_alphabetic)
136        {
137            out = Some(f.to_uppercase().chain(cs).collect());
138        }
139    }
140    out.filter(|o| o != word)
141}

143/// After a space or punctuation, correct the word just typed.
144pub(super) fn autocorrect(s: &mut Session, typing: bool) -> Result<()> {
145    if !s.prefs.autocorrect && !(typing && s.prefs.ac_lists) {
146        return Ok(());
147    }
148    let st = s.doc()?;
149    let Some(t) = st.selection.text.clone() else { return Ok(()) };
150    let Some(body) = body_of(st, &t) else { return Ok(()) };
151    let line: Vec<char> = body.paragraphs.get(t.caret.0).map(|p| p.text().chars().collect()).unwrap_or_default();
152    let end = t.caret.1;
153    if end == 0 || !line.get(end - 1).is_some_and(|c| c.is_whitespace() || ".,;:!?".contains(*c)) {
154        return Ok(());
155    }
156    if typing && s.prefs.ac_lists {
157        // A lead such as `i.` or `a.` is kept as typed, so the space after it starts its list.
158        let lead: String = line.get(..end).map(|l| l.iter().collect()).unwrap_or_default();
159        if auto_list_style(&lead).is_some() || start_auto_list(s, &t, &line, end)? {
160            return Ok(());
161        }
162    }
163    if !s.prefs.autocorrect {
164        return Ok(());
165    }
166    let mut word_end = end - 1;
167    let mut start = word_end;
168    while start > 0 && line.get(start - 1).is_some_and(|c| !c.is_whitespace()) {
169        start -= 1;
170    }
171    // "word," and "word." also count: the punctuation was typed before the space.
172    let mut word: String = line.get(start..word_end).map(|w| w.iter().collect()).unwrap_or_default();
173    if replacement(&s.prefs, &word).is_none() {
174        let trimmed = word.trim_end_matches(|c: char| ".,;:!?".contains(c));
175        if !trimmed.is_empty() && trimmed.len() < word.len() && !word.ends_with("...") {
176            word_end -= word.chars().count() - trimmed.chars().count();
177            word = trimmed.to_string();
178        }
179    }
180    let before = line.get(..start).map(|b| b.iter().rev().find(|c| !c.is_whitespace()).copied()).unwrap_or(None);
181    let sentence_start = before.is_none_or(|c| ".!?".contains(c));
182    // "word..." becomes "word…", the word corrected too.
183    let ellipsis = word.chars().count() > 3 && word.ends_with("...") && replacement(&s.prefs, "...").is_some();
184    let rep = if ellipsis {
185        let stem = word.trim_end_matches('.');
186        Some(format!("{}…", correct_word(&s.prefs, stem, sentence_start, typing).unwrap_or_else(|| stem.to_string())))
187    } else {
188        correct_word(&s.prefs, &word, sentence_start, typing)
189    };
190    let Some(rep) = rep else {
191        if typing && s.prefs.ac_ordinals && is_ordinal(&word) {
192            return superscript_suffix(s, &t, word_end);
193        }
194        return Ok(());
195    };
196    let delta = rep.chars().count() as isize - word.chars().count() as isize;
197    let st = s.doc_mut()?;
198    let mut d = (*st.doc).clone();
199    let sel = st.selection.clone();
200    if let Some(b) = body_mut(&mut d, &sel, &t) {
201        ed::delete(b, (t.caret.0, start), (t.caret.0, word_end));
202        ed::insert(b, (t.caret.0, start), &rep, None);
203    }
204    st.doc = Arc::new(d);
205    let c = (t.caret.0, (end as isize + delta).max(0) as usize);
206    if let Some(tt) = st.selection.text.as_mut() {
207        tt.caret = c;
208        tt.anchor = c;
209    }
210    st.revision += 1;
211    Ok(())
212}

214/// A list AutoFormat started: the text being edited right after it, the typed lead with its
215/// space, the paragraph's own formatting before, and the revision the change left.
216#[derive(Clone, Debug)]
217pub struct AutoList {
218    sel: TextSel,
219    typed: String,
220    props: crate::model::text::ParaProps,
221    revision: u64,
222}

224/// The list a paragraph that starts with `lead` and a space or tab becomes, as the command and
225/// parameters that make it: `*` •, `-` or `–` –, `--` or `—` ■, `>` ➢, `<>` ◆, and the numbering
226/// styles `1.`, `1)`, `I.`, `i.`, `A.`, `a.` and `a)`.
227fn auto_list_style(lead: &str) -> Option<(&'static str, Value)> {
228    let bullet = |c: &str| Some(("format.bullets", json!({"on": true, "char": c})));
229    let number = |scheme: &str| Some(("format.numbering", json!({"on": true, "scheme": scheme})));
230    match lead {
231        "*" => bullet("•"),
232        "-" | "–" => bullet("–"),
233        "--" | "—" => bullet("■"),
234        ">" => bullet("➢"),
235        "<>" => bullet("◆"),
236        "1." => number("arabicPeriod"),
237        "1)" => number("arabicParenR"),
238        "I." => number("romanUcPeriod"),
239        "i." => number("romanLcPeriod"),
240        "A." => number("alphaUcPeriod"),
241        "a." => number("alphaLcPeriod"),
242        "a)" => number("alphaLcParenR"),
243        _ => None,
244    }
245}

247/// AutoFormat's automatic bulleted and numbered lists: when the caret's paragraph holds only a
248/// list lead (`*`, `1.`…) before the space or tab just typed, the lead and the space are removed
249/// and the paragraph becomes a list paragraph in that style. `false` when nothing changed.
250fn start_auto_list(s: &mut Session, t: &TextSel, line: &[char], end: usize) -> Result<bool> {
251    if t.is_range() || !line.get(end - 1).is_some_and(|c| *c == ' ' || *c == '\t') {
252        return Ok(false);
253    }
254    let lead: String = line.get(..end - 1).map(|l| l.iter().collect()).unwrap_or_default();
255    let Some((id, params)) = auto_list_style(&lead) else { return Ok(false) };
256    let typed: String = line.get(..end).map(|l| l.iter().collect()).unwrap_or_default();
257    let para = t.caret.0;
258    let props = body_of(s.doc()?, t).and_then(|b| b.paragraphs.get(para)).map(|p| p.props.clone()).unwrap_or_default();
259    edit_text(s, |body, tt| {
260        ed::delete(body, (para, 0), (para, end));
261        tt.anchor = (para, 0);
262        tt.caret = (para, 0);
263    })?;
264    s.execute(id, &params)?;
265    let st = s.doc_mut()?;
266    if let Some(sel) = st.selection.text.clone() {
267        st.auto_list = Some(AutoList { sel, typed, props, revision: st.revision });
268    }
269    Ok(true)
270}

272/// Backspace right after AutoFormat started a list: the paragraph gets back the typed lead and
273/// space and the formatting it had. `false` when the last change was anything else.
274pub(super) fn undo_auto_list(s: &mut Session) -> Result<bool> {
275    let st = s.doc_mut()?;
276    let Some(a) = st.auto_list.take() else { return Ok(false) };
277    if a.revision != st.revision || st.selection.text.as_ref() != Some(&a.sel) {
278        return Ok(false);
279    }
280    edit_text(s, |body, t| {
281        if let Some(p) = body.paragraphs.get_mut(a.sel.caret.0) {
282            p.props = a.props.clone();
283        }
284        let c = ed::insert(body, a.sel.caret, &a.typed, None);
285        t.anchor = c;
286        t.caret = c;
287    })?;
288    Ok(true)
289}

291/// `word` is a number with the ordinal suffix it takes (1st, 22nd, 103rd, 11th), as AutoFormat
292/// superscripts it.
293fn is_ordinal(word: &str) -> bool {
294    let digits = word.trim_end_matches(|c: char| c.is_ascii_lowercase());
295    let suffix = word.get(digits.len()..).unwrap_or_default();
296    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) || suffix.len() != 2 {
297        return false;
298    }
299    let d = digits.as_bytes();
300    let teen = d.len() >= 2 && d.get(d.len() - 2) == Some(&b'1');
301    let want = match d.last() {
302        _ if teen => "th",
303        Some(b'1') => "st",
304        Some(b'2') => "nd",
305        Some(b'3') => "rd",
306        _ => "th",
307    };
308    suffix == want
309}

311/// Superscript the two letters before `end` in the caret's paragraph (an ordinal's suffix).
312fn superscript_suffix(s: &mut Session, t: &crate::engine::TextSel, end: usize) -> Result<()> {
313    let (a, b) = ((t.caret.0, end.saturating_sub(2)), (t.caret.0, end));
314    let st = s.doc_mut()?;
315    if body_of(st, t).is_none_or(|body| ed::props_at(body, b).baseline.unwrap_or(0.0) > 0.0) {
316        return Ok(());
317    }
318    let mut d = (*st.doc).clone();
319    let sel = st.selection.clone();
320    if let Some(body) = body_mut(&mut d, &sel, t) {
321        ed::format(body, a, b, &|r| r.baseline = Some(0.3));
322    }
323    st.doc = Arc::new(d);
324    st.revision += 1;
325    Ok(())
326}

328pub(super) fn autocorrect_options(s: &mut Session, p: &Value) -> Result<Value> {
329    let pr = &mut s.prefs;
330    for (key, slot) in [
331        ("on", &mut pr.autocorrect),
332        ("twoInitials", &mut pr.ac_two_initials),
333        ("sentenceCaps", &mut pr.ac_sentence_caps),
334        ("days", &mut pr.ac_days),
335        ("capsLock", &mut pr.ac_caps_lock),
336        ("fractions", &mut pr.ac_fractions),
337        ("ordinals", &mut pr.ac_ordinals),
338        ("lists", &mut pr.ac_lists),
339    ] {
340        if let Some(v) = p.get(key).and_then(Value::as_bool) {
341            *slot = v;
342        }
343    }
344    if let Some(Value::Array(a)) = p.get("add") {
345        let (Some(from), Some(to)) = (a.first().and_then(Value::as_str), a.get(1).and_then(Value::as_str)) else {
346            return Err(bad("text.autocorrect", "`add` is [from, to]"));
347        };
348        let from = from.trim();
349        if from.is_empty() || from.chars().any(char::is_whitespace) || from.chars().count() > 64 || to.chars().count() > 255 {
350            return Err(bad("text.autocorrect", "`from` is one word (up to 64 characters), `to` up to 255"));
351        }
352        pr.ac_entries.retain(|(k, _)| k != from);
353        pr.ac_removed.retain(|k| !k.eq_ignore_ascii_case(from));
354        if pr.ac_entries.len() >= 2000 {
355            return Err(bad("text.autocorrect", "the list is full"));
356        }
357        pr.ac_entries.push((from.to_string(), to.to_string()));
358    }
359    if let Some(from) = str_param(p, "remove") {
360        let had = pr.ac_entries.len();
361        pr.ac_entries.retain(|(k, _)| k != from);
362        if had == pr.ac_entries.len()
363            && AUTOCORRECT.iter().any(|(k, _)| k.eq_ignore_ascii_case(from))
364            && !pr.ac_removed.iter().any(|r| r.eq_ignore_ascii_case(from))
365        {
366            pr.ac_removed.push(from.to_string());
367        }
368    }
369    let list: Vec<(String, String)> = pr
370        .ac_entries
371        .iter()
372        .cloned()
373        .chain(
374            AUTOCORRECT
375                .iter()
376                .filter(|(k, _)| {
377                    !pr.ac_removed.iter().any(|r| r.eq_ignore_ascii_case(k)) && !pr.ac_entries.iter().any(|(e, _)| e.eq_ignore_ascii_case(k))
378                })
379                .map(|(k, v)| (k.to_string(), v.to_string())),
380        )
381        .collect();
382    Ok(json!({
383        "on": pr.autocorrect, "twoInitials": pr.ac_two_initials, "sentenceCaps": pr.ac_sentence_caps,
384        "days": pr.ac_days, "capsLock": pr.ac_caps_lock, "fractions": pr.ac_fractions,
385        "ordinals": pr.ac_ordinals, "lists": pr.ac_lists, "entries": list,
386    }))
