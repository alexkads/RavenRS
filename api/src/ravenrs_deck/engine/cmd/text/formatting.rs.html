7/// Apply run formatting to the text selection (or, when not editing, to all text of the
8/// selected shapes). A caret without a range formats the word it's in (PowerPoint behaviour),
9/// or sets the formatting for what is typed next at a word boundary.
10pub(crate) fn format_run(s: &mut Session, f: &(dyn Fn(&mut RunProps) + Send + Sync)) -> Result<Value> {
11    format_run_levels(s, f, &|r, _| f(r))
12}

14/// Like [`format_run`], with a separate change for the list-style levels of a layout or master
15/// placeholder: `lf` gets the level's own run props and the level's effective ones (what a run at
16/// that level resolves to), so a relative change (grow/shrink) steps each level from its own value.
17pub(crate) fn format_run_levels(
18    s: &mut Session,
19    f: &(dyn Fn(&mut RunProps) + Send + Sync),
20    lf: &(dyn Fn(&mut RunProps, &RunProps) + Send + Sync),
21) -> Result<Value> {
22    let st = s.doc()?;
23    if let Some(t) = st.selection.text.clone() {
24        let (a, b) = format_range(st, &t);
25        // Only the levels of the paragraphs the selected range touches.
26        let levels = if a == b { Vec::new() } else { master_levels(st, &t, t.ordered()) };
27        edit_text(s, |body, _| {
28            if levels.is_empty() {
29                ed::format(body, a, b, f);
30            } else {
31                format_by_level(body, a, b, lf, &levels);
32            }
33            for (l, eff) in &levels {
34                let mut lvl = body.list_style.level(*l).cloned().unwrap_or_default();
35                lf(&mut lvl.run, eff);
36                body.list_style.set(*l, lvl);
37            }
38        })?;
39        return ok();
40    }
41    let ids = st.selection.shapes.clone();
42    if ids.is_empty() {
43        return Err(bad("format", SELECT_TEXT_OR_SHAPE));
44    }
45    let cells = st.selection.cells;
46    // Effective run props of every level of the selected placeholders, resolved before editing.
47    let effective: Vec<(ShapeId, Vec<RunProps>)> = if editing_master(st) {
48        let ctx = crate::engine::cmd::text::ctx_for(&st.doc, &st.selection);
49        ids.iter()
50            .filter_map(|id| {
51                let sh = st.shape(*id).filter(|sh| sh.ph.is_some())?;
52                let ctx = ctx.as_ref()?;
53                Some((*id, (0..LIST_LEVELS).map(|l| crate::model::resolve::level_style(ctx, sh, l).run).collect()))
54            })
55            .collect()
56    } else {
57        Vec::new()
58    };
59    edit_shapes(s, &json!({"ids": ids}), "format", |sh| {
60        let id = sh.id;
61        match &mut sh.kind {
62            ShapeKind::Table(t) => {
63                // Only the selected cells, when a range of them is selected.
64                let range = crate::engine::cmd::table::selected_range(cells, id, t);
65                for (r, row) in t.rows.iter_mut().enumerate() {
66                    for (c, cell) in row.cells.iter_mut().enumerate() {
67                        if crate::engine::cmd::table::in_range(range, r, c) {
68                            ed::format_all(&mut cell.text, f);
69                        }
70                    }
71                }
72            }
73            _ => {
74                let body = sh.text.get_or_insert_with(TextBody::default);
75                let eff = effective.iter().find(|(i, _)| *i == id).map(|(_, e)| e);
76                match eff {
77                    Some(eff) => {
78                        let levels: Vec<(u8, RunProps)> = (0..LIST_LEVELS).zip(eff.iter().cloned()).collect();
79                        format_by_level(body, (0, 0), (usize::MAX, usize::MAX), lf, &levels);
80                    }
81                    None => ed::format_all(body, f),
82                }
83                // The whole shape is selected: every level of its list style.
84                if let Some(eff) = eff {
85                    for (l, eff) in (0..LIST_LEVELS).zip(eff) {
86                        let mut lvl = body.list_style.level(l).cloned().unwrap_or_default();
87                        lf(&mut lvl.run, eff);
88                        body.list_style.set(l, lvl);
89                    }
90                }
91            }
92        }
93        Ok(())
94    })?;
95    for id in ids {
96        fit_text_box(s, id)?;
97    }
98    ok()
99}

101/// Format the runs of [a, b) paragraph by paragraph, each with its own level's effective props, so
102/// a relative change (grow/shrink) steps every paragraph from its own size instead of one shared
103/// value. `a`/`b` are clamped to the body.
104fn format_by_level(body: &mut TextBody, a: Pos, b: Pos, lf: &(dyn Fn(&mut RunProps, &RunProps) + Send + Sync), levels: &[(u8, RunProps)]) {
105    let Some(last) = body.paragraphs.len().checked_sub(1) else { return };
106    let (first, end) = (a.0.min(last), b.0.min(last));
107    for pi in first..=end {
108        let p = &body.paragraphs[pi];
109        let from = if pi == first { a.1.min(p.char_len()) } else { 0 };
110        let to = if pi == end { b.1.min(p.char_len()) } else { p.char_len() };
111        if from == to && p.char_len() > 0 {
112            continue;
113        }
114        let lvl = p.level.min(LIST_LEVELS - 1);
115        let Some((_, eff)) = levels.iter().find(|(l, _)| *l == lvl) else { continue };
116        ed::format(body, (pi, from), (pi, to), &|r| lf(r, eff));
117    }
118}

120/// List-style levels (`lvl1pPr`–`lvl9pPr`) a placeholder passes on to the slides that use it.
121const LIST_LEVELS: u8 = 9;

123/// Formatting a placeholder on a layout or master (Slide Master view) also sets its list style,
124/// which is what the placeholders on slides inherit; its own runs may be few or none.
125fn editing_master(st: &DocState) -> bool {
126    matches!(st.selection.target, crate::engine::Target::Layout { .. } | crate::engine::Target::Master { .. })
127}

129/// The distinct list-style levels of the paragraphs `range` touches in a layout or master
130/// placeholder, with each level's effective run props. Empty anywhere else.
131fn master_levels(st: &DocState, t: &TextSel, range: (Pos, Pos)) -> Vec<(u8, RunProps)> {
132    if !editing_master(st) || t.notes || t.cell.is_some() {
133        return Vec::new();
134    }
135    let (Some(sh), Some(body), Some(ctx)) =
136        (st.shape(t.shape).filter(|sh| sh.ph.is_some()), body_of(st, t), crate::engine::cmd::text::ctx_for(&st.doc, &st.selection))
137    else {
138        return Vec::new();
139    };
140    let mut levels: Vec<u8> =
141        body.paragraphs.iter().enumerate().filter(|(i, _)| (range.0.0..=range.1.0).contains(i)).map(|(_, p)| p.level.min(LIST_LEVELS - 1)).collect();
142    if body.paragraphs.is_empty() {
143        levels.push(0);
144    }
145    levels.sort_unstable();
146    levels.dedup();
147    levels.into_iter().map(|l| (l, crate::model::resolve::level_style(&ctx, sh, l).run)).collect()
148}

150/// What `format_run` formats while editing: the selection, or for a bare caret the word it's in
151/// (without its trailing space). Empty when the caret is at a word boundary. Insert Hyperlink
152/// edits this same span, so the dialog reads it too.
153pub fn format_range(st: &DocState, t: &TextSel) -> (Pos, Pos) {
154    let body = body_of(st, t).cloned().unwrap_or_default();
155    let (mut a, mut b) = t.ordered();
156    if a == b {
157        let (wa, wb) = ed::word_at(&body, a);
158        let inside = wa.1 < a.1 && a.1 < wb.1;
159        if inside {
160            a = wa;
161            b = wb;
162            // Don't include the trailing space.
163            let line: Vec<char> = body.paragraphs.get(b.0).map(|p| p.text().chars().collect()).unwrap_or_default();
164            while b.1 > a.1 && line.get(b.1 - 1) == Some(&' ') {
165                b.1 -= 1;
166            }
167        }
168    }
169    (a, b)
170}

172/// Link the text being edited: the selection or the word at the caret, as `format_run` would.
173/// With nothing to link (caret at a word boundary, empty paragraph) `display` is inserted at the
174/// caret as linked text in the surrounding formatting, as PowerPoint's Insert Hyperlink does.
175pub(crate) fn link_text(s: &mut Session, link: Hyperlink, display: &str) -> Result<Value> {
176    let st = s.doc()?;
177    let Some(t) = st.selection.text.clone() else { return Err(bad("insert.hyperlink", "not editing text")) };
178    let (a, b) = format_range(st, &t);
179    if a != b {
180        return format_run(s, &move |rp| rp.link = Some(link.clone()));
181    }
182    if display.is_empty() {
183        return Err(bad("insert.hyperlink", "nothing to link: select text or give `text`"));
184    }
185    let pos = edit_text(s, |body, t| {
186        let mut props = ed::props_at(body, a);
187        props.link = Some(link);
188        let c = ed::insert(body, a, display, Some(props));
189        t.anchor = c;
190        t.caret = c;
191        c
192    })?;
193    Ok(json!({"caret": [pos.0, pos.1]}))
194}

196/// Apply paragraph changes to the paragraphs touched by the selection (or all paragraphs of the
197/// selected shapes).
198pub(crate) fn format_para(s: &mut Session, f: &(dyn Fn(&mut crate::model::text::Paragraph) + Send + Sync)) -> Result<Value> {
199    let st = s.doc()?;
200    if let Some(t) = st.selection.text.clone() {
201        let (a, b) = t.ordered();
202        // Only the list-style levels of the paragraphs touched, on a layout or master placeholder.
203        let levels: Vec<u8> = master_levels(st, &t, (a, b)).into_iter().map(|(l, _)| l).collect();
204        edit_text(s, |body, _| {
205            if body.paragraphs.is_empty() {
206                body.paragraphs.push(Default::default());
207            }
208            let last = body.paragraphs.len().saturating_sub(1);
209            for pi in a.0..=b.0.min(last) {
210                if let Some(p) = body.paragraphs.get_mut(pi) {
211                    f(p);
212                }
213            }
214            for l in levels {
215                format_list_level(body, l, f);
216            }
217        })?;
218        return ok();
219    }
220    let ids = st.selection.shapes.clone();
221    if ids.is_empty() {
222        return Err(bad("format", SELECT_TEXT_OR_SHAPE));
223    }
224    let cells = st.selection.cells;
225    let on_master = editing_master(st);
226    edit_shapes(s, &json!({"ids": ids}), "format", |sh| {
227        let id = sh.id;
228        let inherited = on_master && sh.ph.is_some();
229        match &mut sh.kind {
230            ShapeKind::Table(t) => {
231                let range = crate::engine::cmd::table::selected_range(cells, id, t);
232                for (r, row) in t.rows.iter_mut().enumerate() {
233                    for (c, cell) in row.cells.iter_mut().enumerate() {
234                        if crate::engine::cmd::table::in_range(range, r, c) {
235                            cell.text.paragraphs.iter_mut().for_each(f);
236                        }
237                    }
238                }
239            }
240            _ => {
241                let body = sh.text.get_or_insert_with(TextBody::default);
242                if body.paragraphs.is_empty() {
243                    body.paragraphs.push(Default::default());
244                }
245                body.paragraphs.iter_mut().for_each(f);
246                // The whole shape is selected: every level of its list style.
247                if inherited {
248                    for l in 0..LIST_LEVELS {
249                        format_list_level(body, l, f);
250                    }
251                }
252            }
253        }
254        Ok(())
255    })?;
256    for id in ids {
257        fit_text_box(s, id)?;
258    }
259    ok()
260}

262/// Apply a paragraph change to one level of a body's list style.
263fn format_list_level(body: &mut TextBody, l: u8, f: &(dyn Fn(&mut crate::model::text::Paragraph) + Send + Sync)) {
264    let mut lvl = body.list_style.level(l).cloned().unwrap_or_default();
265    let mut p = crate::model::text::Paragraph { level: l, props: lvl.para, ..Default::default() };
266    f(&mut p);
267    // A level change (indent/outdent) means nothing for a list-style level.
268    if p.level == l {
269        lvl.para = p.props;
270        body.list_style.set(l, lvl);
271    }
272}

274/// Apply body (text box) changes to the edited or selected shapes.
275pub(crate) fn format_body(s: &mut Session, f: &(dyn Fn(&mut crate::model::text::BodyProps) + Send + Sync)) -> Result<Value> {
276    let st = s.doc()?;
277    let ids: Vec<ShapeId> = match &st.selection.text {
278        Some(t) if !t.notes => vec![t.shape],
279        _ => st.selection.shapes.clone(),
280    };
281    if ids.is_empty() {
282        return Err(bad("format", "select a shape"));
283    }
284    edit_shapes(s, &json!({"ids": ids}), "format", |sh| {
285        f(&mut sh.text.get_or_insert_with(TextBody::default).body);
286        Ok(())
287    })?;
288    for id in ids {
289        fit_text_box(s, id)?;
290    }
291    ok()
292}

300pub(super) fn replace_font(s: &mut Session, p: &Value) -> Result<Value> {
301    let from = str_param(p, "from").map(str::trim).filter(|x| !x.is_empty()).ok_or_else(|| bad("font.replace", "missing `from`"))?.to_string();
302    let to = str_param(p, "to").map(str::trim).filter(|x| !x.is_empty()).ok_or_else(|| bad("font.replace", "missing `to`"))?.to_string();
303    if to.len() > 256 || to.chars().any(char::is_control) {
304        return Err(bad("font.replace", "`to` is not a font name"));
305    }
306    // Count on a copy first so a replace that finds nothing leaves no undo step.
307    let mut probe = (*s.doc()?.doc).clone();
308    if crate::engine::fonts::replace_font(&mut probe, &from, &to) == 0 {
309        return Ok(json!({"replaced": 0}));
310    }
311    let n = s.edit(|doc, _| Ok(crate::engine::fonts::replace_font(doc, &from, &to)))?;
312    let st = s.doc_mut()?;
313    st.missing_fonts = crate::engine::fonts::missing_fonts(&st.doc);
314    Ok(json!({"replaced": n}))
315}

327    #[test]
328    fn replaces_runs_tables_notes_and_theme_and_undoes() {
329        let mut s = Session::with_new();
330        s.execute("slide.new", &json!({"layout": "titleAndContent", "title": "T", "body": "B"})).unwrap();
331        let theme = s.doc().unwrap().doc.masters[0].theme.fonts.minor.latin.clone();
332        s.execute("insert.textBox", &json!({"rect": [10, 10, 200, 50], "text": "Hello"})).unwrap();
333        s.execute("text.exit", &json!({})).unwrap();
334        s.execute("format.font", &json!({"family": "Comic Neue"})).unwrap();
335        assert!(fonts(&mut s).iter().any(|f| f == "Comic Neue"), "{:?}", fonts(&mut s));
336        let r = s.execute("font.replace", &json!({"from": "comic neue", "to": "Lato"})).unwrap();
337        assert!(r["replaced"].as_u64().unwrap() >= 1, "{r}");
338        let used = fonts(&mut s);
339        assert!(!used.iter().any(|f| f == "Comic Neue") && used.iter().any(|f| f == "Lato"), "{used:?}");
340        // The theme's body font is replaced too, so placeholders follow.
341        s.execute("font.replace", &json!({"from": theme, "to": "Lato"})).unwrap();
342        assert_eq!(s.doc().unwrap().doc.masters[0].theme.fonts.minor.latin, "Lato");
343        assert!(s.doc().unwrap().doc.validate().is_empty());
344        s.execute("edit.undo", &json!({})).unwrap();
345        s.execute("edit.undo", &json!({})).unwrap();
346        assert!(fonts(&mut s).iter().any(|f| f == "Comic Neue"));
347    }
