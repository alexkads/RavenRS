6/// Formatting picked up by the Format Painter.
7#[derive(Clone, Debug, Default)]
8pub struct Painted {
9    pub run: Option<RunProps>,
10    pub para: Option<ParaProps>,
11    pub fill: Option<Fill>,
12    pub line: Option<Line>,
13    pub effects: Option<Effects>,
14    pub style: Option<crate::model::ShapeStyle>,
15}

17pub(super) fn painter_pick(s: &mut Session, p: &Value) -> Result<Value> {
18    let st = s.doc()?;
19    let mut painted = Painted::default();
20    if let Some(t) = &st.selection.text {
21        if let Some(b) = crate::engine::cmd::text::body_of(st, t) {
22            let a = t.ordered().0;
23            painted.run = Some(ed::props_at(b, (a.0, a.1 + 1)));
24            painted.para = b.paragraphs.get(a.0).map(|x| x.props.clone());
25        }
26    } else if let Some(sh) = st.selected_shapes().first() {
27        painted.fill = sh.fill.clone();
28        painted.line = sh.line.clone();
29        painted.effects = sh.effects.clone();
30        painted.style = sh.style.clone();
31        painted.run = sh.text.as_ref().and_then(|t| t.paragraphs.first()).and_then(|p| p.runs.first()).map(|r| r.props.clone());
32    }
33    s.painter = Some((painted, bool_or(p, "sticky", false)));
34    Ok(json!({"armed": true}))
35}

37pub(crate) fn apply_painted(sh: &mut Shape, p: &Painted) {
38    if p.fill.is_some() || p.style.is_some() {
39        sh.fill.clone_from(&p.fill);
40        sh.line.clone_from(&p.line);
41        sh.effects.clone_from(&p.effects);
42        sh.style.clone_from(&p.style);
43    }
44    if let (Some(r), Some(t)) = (&p.run, sh.text.as_mut()) {
45        ed::format_all(t, &|rp| *rp = r.clone());
46    }
47}

49/// The paragraphs the text selection covers whole: it starts at or before their start and
50/// reaches their end. A paragraph the selection only touches the start of (it ends at offset
51/// 0, as after a triple-click) is not covered.
52fn covered_paragraphs(s: &Session) -> Vec<usize> {
53    let Ok(st) = s.doc() else { return vec![] };
54    let Some(t) = &st.selection.text else { return vec![] };
55    let Some(body) = crate::engine::cmd::text::body_of(st, t) else { return vec![] };
56    let (a, b) = t.ordered();
57    // A hostile selection can point past the body: walk only the paragraphs it has.
58    (a.0..=b.0.min(body.paragraphs.len().saturating_sub(1)))
59        .filter(|&pi| {
60            let Some(para) = body.paragraphs.get(pi) else { return false };
61            let starts = pi > a.0 || a.1 == 0;
62            let ends = if pi < b.0 {
63                true
64            } else if b.0 > a.0 && b.1 == 0 {
65                false
66            } else {
67                b.1 >= para.char_len()
68            };
69            starts && ends
70        })
71        .collect()
72}

74pub(super) fn painter_apply(s: &mut Session, p: &Value) -> Result<Value> {
75    let Some((painted, sticky)) = s.painter.clone() else { return Err(bad("format.painterApply", "pick formatting with the Format Painter first")) };
76    if !sticky {
77        s.painter = None;
78    }
79    if s.doc()?.selection.text.as_ref().is_some_and(|t| t.is_range()) && ids_param(p, "ids").is_none() {
80        if let Some(r) = painted.run.clone() {
81            format_run(s, &move |rp| *rp = r.clone())?;
82        }
83        // Paragraph formatting (bullets, alignment, spacing) goes only onto paragraphs the
84        // selection covers whole; painting part of a bulleted line changes just its text.
85        let covered = covered_paragraphs(s);
86        if let (Some(pp), false) = (painted.para.clone(), covered.is_empty()) {
87            crate::engine::cmd::text::edit_text(s, |body, _| {
88                for &pi in &covered {
89                    if let Some(para) = body.paragraphs.get_mut(pi) {
90                        para.props = pp.clone();
91                    }
92                }
93            })?;
94        }
95        return ok();
96    }
97    edit_shapes(s, p, "format.painterApply", |sh| {
98        apply_painted(sh, &painted);
99        Ok(())
100    })
101}

103pub(super) fn state(s: &mut Session, _p: &Value) -> Result<Value> {
104    let r = current_run(s);
105    let pp = current_para(s);
106    let st = s.doc()?;
107    let family = crate::engine::cmd::text::ctx_for(&st.doc, &st.selection).map(|c| resolve::font_family(&c, &r)).unwrap_or_default();
108    Ok(json!({
109        "font": family,
110        "fontRef": r.font,
111        "size": r.size,
112        "bold": r.bold.unwrap_or(false),
113        "italic": r.italic.unwrap_or(false),
114        "underline": r.underline.as_deref().is_some_and(|u| u != "none"),
115        "strike": r.strike.is_some_and(|x| x != Strike::None),
116        "superscript": r.baseline.unwrap_or(0.0) > 0.0,
117        "subscript": r.baseline.unwrap_or(0.0) < 0.0,
118        "align": pp.align.map(|a| a.xml()),
119        "bullets": matches!(pp.bullet, Some(Bullet::Char { .. })),
120        "numbering": matches!(pp.bullet, Some(Bullet::AutoNum { .. })),
121        "lang": r.lang,
122        "lineSpacing": serde_json::to_value(pp.line_spacing).unwrap_or_default(),
123        "painter": s.painter.is_some(),
124    }))
125}

127/// Add a font from its bytes for this session (it then shows in the font menu and draws).
128pub(super) fn font_add(_s: &mut Session, p: &Value) -> Result<Value> {
129    let data = str_param(p, "data").and_then(base64_decode).ok_or_else(|| bad("font.add", "missing or invalid base64 `data`"))?;
130    if data.len() > 64 << 20 {
131        return Err(bad("font.add", "font file larger than 64 MB"));
132    }
133    let faces = crate::fonts::FontDb::global().add_font(data);
134    if faces == 0 {
135        return Err(bad("font.add", "not a font DeckCraft can read (or already added)"));
136    }
137    Ok(json!({"faces": faces}))
138}

140/// Catalog the installed fonts again, so fonts installed while DeckCraft runs appear.
141pub(super) fn font_rescan(_s: &mut Session, _p: &Value) -> Result<Value> {
142    let db = crate::fonts::FontDb::global();
143    #[cfg(not(target_arch = "wasm32"))]
144    let faces = db.load_system_fonts();
145    #[cfg(target_arch = "wasm32")]
146    let faces = 0usize;
147    Ok(json!({"faces": faces, "families": db.families().len()}))
148}
