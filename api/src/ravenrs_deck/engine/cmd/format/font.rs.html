58/// PowerPoint's font size list, as the Font Size box lists it (grow/shrink steps through it).
59pub const SIZES: [f64; 24] =
60    [8.0, 9.0, 10.0, 10.5, 11.0, 12.0, 14.0, 16.0, 18.0, 20.0, 24.0, 28.0, 32.0, 36.0, 40.0, 44.0, 48.0, 54.0, 60.0, 66.0, 72.0, 80.0, 88.0, 96.0];
61
62/// Grow steps up the list, then 8 points at a time past its top; shrink steps back down the same
63/// way. A size past the top used to shrink straight to 66 (#2059).
64fn step(cur: f64, up: bool) -> f64 {
65    let top = SIZES[SIZES.len() - 1];
66    if up {
67        SIZES.iter().copied().find(|v| *v > cur + 0.01).unwrap_or((cur + 8.0).min(4000.0))
68    } else if cur - 0.01 > top {
69        (cur - 8.0).max(top)
70    } else {
71        SIZES.iter().rev().copied().find(|v| *v < cur - 0.01).unwrap_or((cur - 1.0).max(1.0))
72    }
73}

75fn grow_shrink(s: &mut Session, up: bool) -> Result<Value> {
76    // Each run steps from its own size, like PowerPoint.
77    let st = s.doc()?;
78    let ctx_sizes = current_run(s).size.unwrap_or(18.0);
79    let _ = st;
80    let base = ctx_sizes;
81    // A layout/master level steps from its own effective size, not the first run's.
82    format_run_levels(
83        s,
84        &move |r| {
85            let cur = r.size.unwrap_or(base);
86            r.size = Some(step(cur, up));
87        },
88        &move |r, eff| {
89            let cur = r.size.or(eff.size).unwrap_or(18.0);
90            r.size = Some(step(cur, up));
91        },
92    )
93}

103pub(super) fn color(s: &mut Session, p: &Value) -> Result<Value> {
104    let c = color_param(p, "color").ok_or_else(|| bad("format.color", "missing or invalid `color`"))?;
105    // A selected chart has no runs: Font Color colours its text (title, axes, legend, labels),
106    // which used to answer and change nothing.
107    let st = s.doc()?;
108    if st.selection.text.is_none() {
109        let charts: Vec<ShapeId> =
110            st.selection.shapes.iter().copied().filter(|id| matches!(st.shape(*id).map(|sh| &sh.kind), Some(ShapeKind::Chart(_)))).collect();
111        let only_charts = charts.len() == st.selection.shapes.len();
112        if !charts.is_empty() {
113            let tc = c.clone();
114            edit_shapes(s, &json!({"ids": charts}), "format.color", |sh| {
115                if let ShapeKind::Chart(ch) = &mut sh.kind {
116                    ch.text_color = Some(tc.clone());
117                    ch.legend_entries.iter_mut().for_each(|e| e.color = None);
118                    ch.raw = None;
119                }
120                Ok(())
121            })?;
122            if only_charts {
123                return ok();
124            }
125        }
126    }
127    format_run(s, &move |r| r.fill = Some(Fill::solid(c.clone())))
128}

130/// `a:alpha` for `transparency` (0–1) on `c`, replacing any alpha it had; fully opaque drops it.
131fn with_transparency(c: &mut ColorRef, transparency: f64) {
132    c.mods.retain(|m| !matches!(m, ColorTransform::Alpha(_)));
133    if transparency > 0.0 {
134        c.mods.push(ColorTransform::Alpha(((1.0 - transparency) * 100_000.0).round() as i32));
135    }
136}

138/// Text Fill › Transparency: the text keeps its colour and gets an alpha. A run without its own
139/// fill takes the colour it shows now (from the placeholder, master or theme) so only its
140/// opacity changes.
141pub(super) fn text_transparency(s: &mut Session, p: &Value) -> Result<Value> {
142    let v = f64_param(p, "value").ok_or_else(|| bad("format.textTransparency", "missing `value` (0–1)"))?;
143    if !v.is_finite() {
144        return Err(bad("format.textTransparency", "`value` must be a number from 0 to 1"));
145    }
146    let v = v.clamp(0.0, 1.0);
147    let shown = match current_run(s).fill {
148        Some(f @ (Fill::Solid { .. } | Fill::Gradient(_))) => f,
149        _ => Fill::solid(ColorRef::scheme(SchemeSlot::Tx1)),
150    };
151    format_run(s, &move |r| {
152        let mut f = match &r.fill {
153            Some(f @ (Fill::Solid { .. } | Fill::Gradient(_))) => f.clone(),
154            Some(Fill::None) => return,
155            _ => shown.clone(),
156        };
157        match &mut f {
158            Fill::Solid { color } => with_transparency(color, v),
159            Fill::Gradient(g) => g.stops.iter_mut().for_each(|st| with_transparency(&mut st.color, v)),
160            _ => {}
161        }
162        r.fill = Some(f);
163    })
164}

166/// The `lang` and `noProof` params of the language commands, checked: a run-props setter.
167fn language_edit(id: &str, p: &Value) -> Result<impl Fn(&mut RunProps) + Send + Sync + 'static> {
168    let lang = str_param(p, "lang").map(str::to_string);
169    let no_proof = bool_param(p, "noProof");
170    if lang.is_none() && no_proof.is_none() {
171        return Err(bad(id, "give `lang` or `noProof`"));
172    }
173    if lang.as_deref().is_some_and(|l| l.is_empty() || l.len() > 35 || !l.chars().all(|c| c.is_ascii_alphanumeric() || c == '-')) {
174        return Err(bad(id, "`lang` is a language tag such as en-US"));
175    }
176    Ok(move |r: &mut RunProps| {
177        if let Some(l) = &lang {
178            r.lang = Some(l.clone());
179        }
180        if let Some(n) = no_proof {
181            r.no_proof = n.then_some(true);
182        }
183    })
184}

192/// Every run in the deck, and the text styles new text inherits from, in one language.
193pub(super) fn language_all(s: &mut Session, p: &Value) -> Result<Value> {
194    let f = language_edit("format.languageAll", p)?;
195    /// Runs, end marks and the body's own list style levels (never creating one).
196    fn body(t: &mut crate::model::text::TextBody, f: &dyn Fn(&mut RunProps)) {
197        ed::format_all(t, f);
198        t.list_style.levels.iter_mut().flatten().for_each(|lv| f(&mut lv.run));
199    }
200    /// Every text body in `list`: shapes, table cells and groups (nested at most 64 deep).
201    fn shapes(list: &mut [Shape], f: &dyn Fn(&mut RunProps), depth: usize) {
202        if depth > 64 {
203            return;
204        }
205        for sh in list {
206            if let Some(t) = &mut sh.text {
207                body(t, f);
208            }
209            match &mut sh.kind {
210                crate::model::ShapeKind::Table(t) => t.rows.iter_mut().flat_map(|r| &mut r.cells).for_each(|c| body(&mut c.text, f)),
211                crate::model::ShapeKind::Group { children, .. } => shapes(children, f, depth + 1),
212                _ => {}
213            }
214        }
215    }
216    /// A master or presentation text style's levels: what new text inherits.
217    fn styles(l: &mut crate::model::text::ListStyle, f: &dyn Fn(&mut RunProps)) {
218        // Level 1 always, so a deck without a default text style gets one.
219        if l.level(0).is_none() {
220            l.set(0, Default::default());
221        }
222        l.levels.iter_mut().flatten().for_each(|lv| f(&mut lv.run));
223    }
224    s.edit(|doc, _| {
225        for sl in &mut doc.slides {
226            let sl = Arc::make_mut(sl);
227            shapes(&mut sl.shapes, &f, 0);
228            body(&mut sl.notes, &f);
229        }
230        let extra = [doc.notes_master.as_mut(), doc.handout_master.as_mut()];
231        for m in doc.masters.iter_mut().chain(extra.into_iter().flatten()) {
232            let m = Arc::make_mut(m);
233            shapes(&mut m.shapes, &f, 0);
234            m.layouts.iter_mut().for_each(|l| shapes(&mut l.shapes, &f, 0));
235            [&mut m.title_style, &mut m.body_style, &mut m.other_style].into_iter().for_each(|l| styles(l, &f));
236        }
237        styles(&mut doc.default_text_style, &f);
238        Ok(())
239    })?;
240    ok()
241}

265/// Character formatting applied together, so the Font dialog is one undo step.
266struct FontEdits {
267    family: Option<String>,
268    size: Option<f64>,
269    bold: Option<bool>,
270    italic: Option<bool>,
271    underline: Option<String>,
272    color: Option<crate::model::ColorRef>,
273    strike: Option<Strike>,
274    baseline: Option<f64>,
275    caps: Option<Caps>,
276    spacing: Option<f64>,
277}
278
279impl FontEdits {
280    fn is_empty(&self) -> bool {
281        self.family.is_none()
282            && self.size.is_none()
283            && self.bold.is_none()
284            && self.italic.is_none()
285            && self.underline.is_none()
286            && self.color.is_none()
287            && self.strike.is_none()
288            && self.baseline.is_none()
289            && self.caps.is_none()
290            && self.spacing.is_none()
291    }
292
293    fn write(&self, r: &mut RunProps) {
294        if let Some(f) = &self.family {
295            r.font = Some(f.clone());
296        }
297        if let Some(v) = self.size {
298            r.size = Some(v);
299        }
300        if let Some(v) = self.bold {
301            r.bold = Some(v);
302        }
303        if let Some(v) = self.italic {
304            r.italic = Some(v);
305        }
306        if let Some(u) = &self.underline {
307            r.underline = Some(u.clone());
308        }
309        if let Some(c) = &self.color {
310            r.fill = Some(Fill::solid(c.clone()));
311        }
312        if let Some(v) = self.strike {
313            r.strike = Some(v);
314        }
315        if let Some(v) = self.baseline {
316            r.baseline = Some(v);
317        }
318        if let Some(v) = self.caps {
319            r.caps = Some(v);
320        }
321        if let Some(v) = self.spacing {
322            r.spacing = Some(v);
323        }
324    }
325}

343fn font_edits(p: &Value) -> Result<FontEdits> {
344    let family = match font_str(p, "family")? {
345        None => None,
346        Some(s) => {
347            let t = s.trim();
348            if t.is_empty() || t.chars().count() > 200 {
349                return Err(bad("format.fontDialog", "`family` must be 1–200 characters"));
350            }
351            Some(t.to_string())
352        }
353    };
354    let size = match p.get("size") {
355        None => None,
356        Some(v) => {
357            let n = v.as_f64().filter(|n| n.is_finite()).ok_or_else(|| bad("format.fontDialog", "`size` must be a finite number"))?;
358            Some(n.clamp(1.0, 4000.0))
359        }
360    };
361    let underline = match font_str(p, "underline")? {
362        None => None,
363        Some(s) => {
364            if !matches!(s, "none" | "sng" | "dbl" | "heavy" | "dotted" | "dash" | "wavy") {
365                return Err(bad("format.fontDialog", "`underline` must be none, sng, dbl, heavy, dotted, dash or wavy"));
366            }
367            Some(s.to_string())
368        }
369    };
370    let color =
371        if p.get("color").is_some() { Some(color_param(p, "color").ok_or_else(|| bad("format.fontDialog", "invalid `color`"))?) } else { None };
372    let strike = match font_str(p, "strike")? {
373        None => None,
374        Some("none") => Some(Strike::None),
375        Some("sng" | "single") => Some(Strike::Single),
376        Some("dbl" | "double") => Some(Strike::Double),
377        Some(_) => return Err(bad("format.fontDialog", "`strike` must be none, sng or dbl")),
378    };
379    let baseline = match p.get("baseline") {
380        None => None,
381        Some(Value::String(s)) => match s.as_str() {
382            "none" => Some(0.0),
383            "super" | "superscript" => Some(0.3),
384            "sub" | "subscript" => Some(-0.25),
385            _ => return Err(bad("format.fontDialog", "`baseline` must be none, super, sub or a fraction")),
386        },
387        Some(Value::Number(n)) => {
388            let f = n.as_f64().filter(|v| v.is_finite()).ok_or_else(|| bad("format.fontDialog", "`baseline` must be finite"))?;
389            Some(f.clamp(-1.0, 1.0))
390        }
391        Some(_) => return Err(bad("format.fontDialog", "`baseline` must be none, super, sub or a fraction")),
392    };
393    let caps = match font_str(p, "caps")? {
394        None => None,
395        Some("none") => Some(Caps::None),
396        Some("small") => Some(Caps::Small),
397        Some("all") => Some(Caps::All),
398        Some(_) => return Err(bad("format.fontDialog", "`caps` must be none, small or all")),
399    };
400    let spacing = match p.get("spacing") {
401        None => None,
402        Some(v) => {
403            let n = v.as_f64().filter(|n| n.is_finite()).ok_or_else(|| bad("format.fontDialog", "`spacing` must be a finite number"))?;
404            Some(n.clamp(-100.0, 400.0))
405        }
406    };
407    Ok(FontEdits { family, size, bold: font_bool(p, "bold")?, italic: font_bool(p, "italic")?, underline, color, strike, baseline, caps, spacing })
408}

410/// A colour as a `format.fontDialog` / `format.color` parameter, including transforms.
411fn color_command_value(c: &crate::model::ColorRef) -> Value {
412    use crate::model::ColorBase;
413    let mut o = serde_json::Map::new();
414    for m in &c.mods {
415        if let Some(v) = m.value() {
416            o.insert(m.xml_name().to_string(), json!(v));
417        }
418    }
419    match &c.base {
420        ColorBase::Rgb { rgb } if o.is_empty() => json!(format!("#{}", rgb.hex())),
421        ColorBase::Rgb { rgb } => {
422            o.insert("rgb".into(), json!(format!("#{}", rgb.hex())));
423            Value::Object(o)
424        }
425        ColorBase::Scheme { slot } => {
426            o.insert("scheme".into(), json!(slot.xml_name()));
427            Value::Object(o)
428        }
429        ColorBase::Preset { name } => json!(name),
430        ColorBase::System { last, .. } => json!(format!("#{}", last.hex())),
431    }
432}

434fn font_dialog_params(s: &Session) -> Value {
435    let r = current_run(s);
436    let family = s
437        .doc()
438        .ok()
439        .and_then(|st| crate::engine::cmd::text::ctx_for(&st.doc, &st.selection).map(|c| resolve::font_family(&c, &r)))
440        .unwrap_or_default();
441    let baseline = r.baseline.unwrap_or(0.0);
442    let mut v = json!({
443        "family": family,
444        "size": r.size.unwrap_or(18.0),
445        "bold": r.bold.unwrap_or(false),
446        "italic": r.italic.unwrap_or(false),
447        "underline": r.underline.as_deref().filter(|u| *u != "none").unwrap_or("none"),
448        "strike": match r.strike {
449            Some(Strike::Double) => "dbl",
450            Some(Strike::Single) => "sng",
451            _ => "none",
452        },
453        "baseline": if baseline > 0.0 { "super" } else if baseline < 0.0 { "sub" } else { "none" },
454        "caps": match r.caps {
455            Some(Caps::Small) => "small",
456            Some(Caps::All) => "all",
457            _ => "none",
458        },
459        "spacing": r.spacing.unwrap_or(0.0),
460    });
461    if let Some(Fill::Solid { color }) = &r.fill {
462        v["color"] = color_command_value(color);
463    }
464    v
465}

467pub(super) fn font_dialog(s: &mut Session, p: &Value) -> Result<Value> {
468    let edits = font_edits(p)?;
469    if edits.is_empty() {
470        s.ui_requests.push(crate::engine::UiRequest::Dialog { id: "font".into(), params: font_dialog_params(s) });
471        return ok();
472    }
473    format_run(s, &|r| edits.write(r))
474}
