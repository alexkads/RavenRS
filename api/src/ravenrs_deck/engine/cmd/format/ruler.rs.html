4/// What the horizontal ruler shows while editing text: the paragraph's indents and tab stops,
5/// and where its text area sits on the slide.
6pub(super) fn ruler(s: &mut Session, _p: &Value) -> Result<Value> {
7    let props = current_para(s);
8    let st = s.doc()?;
9    let t = st.selection.text.clone().ok_or_else(|| bad("format.ruler", "not editing text"))?;
10    let (x, y, w, rotated) = match (t.notes, st.shape(t.shape)) {
11        (false, Some(sh)) => {
12            let ctx = crate::engine::cmd::text::ctx_for(&st.doc, &st.selection);
13            let x = match &ctx {
14                Some(c) => resolve::xfrm(c, sh),
15                None => sh.xfrm.unwrap_or_default(),
16            };
17            let body = match &ctx {
18                Some(c) => resolve::body(c, sh),
19                None => sh.text.as_ref().map(|b| b.body.clone()).unwrap_or_default(),
20            };
21            let (l, r, tp) = (body.inset_l.unwrap_or(7.2), body.inset_r.unwrap_or(7.2), body.inset_t.unwrap_or(3.6));
22            (x.x + l, x.y + tp, (x.w - l - r).max(0.0), x.rot != 0.0 || body.vert.is_some_and(|v| v != TextDir::Horizontal))
23        }
24        _ => (0.0, 0.0, 0.0, true),
25    };
26    let tabs: Vec<Value> = props.tabs.clone().unwrap_or_default().iter().map(|t| json!({"pos": t.pos, "align": t.align})).collect();
27    Ok(json!({
28        "x": x, "y": y, "width": w, "rotated": rotated,
29        "marginLeft": props.margin_left.unwrap_or(0.0),
30        "indent": props.indent.unwrap_or(0.0),
31        "tabs": tabs,
32        "defaultTab": props.default_tab.unwrap_or(72.0),
33    }))
34}

36const MAX_TABS: usize = 32;
37
38pub(super) fn tabs(s: &mut Session, p: &Value) -> Result<Value> {
39    const CMD: &str = "format.tabs";
40    let fin = |v: Option<f64>| v.filter(|x| x.is_finite()).map(|x| x.clamp(0.0, 4000.0));
41    let align_of = |v: &Value| match v.get("align").and_then(Value::as_str).unwrap_or("l") {
42        "ctr" | "center" => "ctr",
43        "r" | "right" => "r",
44        "dec" | "decimal" => "dec",
45        _ => "l",
46    };
47    let add = p.get("add").and_then(|a| fin(a.get("pos").and_then(Value::as_f64)).map(|pos| (pos, align_of(a).to_string())));
48    let mv = p.get("move").and_then(|m| Some((fin(m.get("from").and_then(Value::as_f64))?, fin(m.get("to").and_then(Value::as_f64))?)));
49    let remove = fin(p.get("remove").and_then(Value::as_f64));
50    // `set` replaces every stop: entries without a usable position are dropped, an unknown
51    // alignment or a non-object entry is an error; stops within 0.05 pt collapse (the last wins).
52    let set = match p.get("set") {
53        None => None,
54        Some(Value::Array(items)) => {
55            if items.len() > 64 {
56                return Err(bad(CMD, "`set` has more than 64 tab stops"));
57            }
58            let mut out: Vec<TabStop> = Vec::new();
59            for (i, item) in items.iter().enumerate() {
60                let Some(obj) = item.as_object() else {
61                    return Err(bad(CMD, format!("tab stop {i} must be an object")));
62                };
63                let Some(pos) = obj.get("pos").and_then(Value::as_f64).filter(|n| n.is_finite()).map(clamp_pt) else {
64                    continue;
65                };
66                let align = match obj.get("align").and_then(Value::as_str) {
67                    None => "l",
68                    Some(a) => tab_align(a).ok_or_else(|| bad(CMD, format!("tab stop {i} align must be l, ctr, r or dec")))?,
69                };
70                if let Some(prev) = out.iter_mut().find(|t| (t.pos - pos).abs() < 0.05) {
71                    *prev = TabStop { pos, align: align.to_string() };
72                } else {
73                    out.push(TabStop { pos, align: align.to_string() });
74                }
75            }
76            Some(out)
77        }
78        Some(_) => return Err(bad(CMD, "`set` must be an array")),
79    };
80    let clear = bool_or(p, "clear", false);
81    let default_tab = finite_key(p, CMD, "defaultTab")?.map(|v| v.clamp(1.0, 1584.0));
82    let edits_stops = set.is_some() || add.is_some() || mv.is_some() || remove.is_some() || clear;
83    if !edits_stops && default_tab.is_none() {
84        // No fields: open the Tabs dialog on the caret's paragraph.
85        s.ui_requests.push(crate::engine::UiRequest::Dialog { id: "tabs".into(), params: tabs_params(s) });
86        return ok();
87    }
88    // The stops as the caret's paragraph shows them (inherited ones included), so editing a
89    // placeholder's tab that came from its layout works like editing one set on the paragraph.
90    let base = current_para(s).tabs.unwrap_or_default();
91    let near = |a: f64, b: f64| (a - b).abs() < 0.5;
92    let mut stops = if clear { vec![] } else { set.unwrap_or(base) };
93    if let Some(at) = remove {
94        stops.retain(|t| !near(t.pos, at));
95    }
96    if let Some((from, to)) = mv
97        && let Some(t) = stops.iter_mut().find(|t| near(t.pos, from))
98    {
99        t.pos = to;
100    }
101    if let Some((pos, align)) = add {
102        stops.retain(|t| !near(t.pos, pos));
103        stops.push(TabStop { pos, align });
104    }
105    stops.sort_by(|a, b| a.pos.total_cmp(&b.pos));
106    stops.dedup_by(|a, b| near(a.pos, b.pos));
107    stops.truncate(MAX_TABS);
108    format_para(s, &move |para| {
109        if edits_stops {
110            para.props.tabs = Some(stops.clone());
111        }
112        if let Some(d) = default_tab {
113            para.props.default_tab = Some(d);
114        }
115    })
116}
