5pub(super) fn text_shadow(s: &mut Session, p: &Value) -> Result<Value> {
6    let v = on_param(p, Some(current_run(s).shadow.is_some()));
7    let sh = crate::model::style::Shadow {
8        color: crate::model::ColorRef::rgb(crate::color::Rgba::BLACK).with(crate::color::ColorTransform::Alpha(43000)),
9        blur: 3.0,
10        dist: 2.0,
11        dir: 45.0,
12        inner: false,
13        sx: 1.0,
14        sy: 1.0,
15        kx: 0.0,
16        ky: 0.0,
17        align: String::new(),
18        rotate_with_shape: false,
19    };
20    format_run(s, &move |r| r.shadow = if v { Some(sh.clone()) } else { None })
21}

29pub(super) fn text_warp(s: &mut Session, p: &Value) -> Result<Value> {
30    let preset = str_param(p, "preset").ok_or_else(|| bad("format.textWarp", "missing `preset`"))?;
31    let clear = preset == "none" || preset == "textNoShape";
32    if !clear && !crate::geom::warp::is_supported(preset) {
33        return Err(bad("format.textWarp", format!("unknown text transform `{preset}`")));
34    }
35    let adj = match p.get("adj") {
36        None => Vec::new(),
37        Some(Value::Array(arr)) => {
38            if arr.len() > 8 {
39                return Err(bad("format.textWarp", "at most 8 adjust values"));
40            }
41            let mut v = Vec::with_capacity(arr.len());
42            for a in arr {
43                let Some(n) = a.as_f64().filter(|n| n.is_finite()) else {
44                    return Err(bad("format.textWarp", "adjust values must be finite numbers"));
45                };
46                v.push(n.clamp(-2.0e9, 2.0e9));
47            }
48            v
49        }
50        Some(_) => return Err(bad("format.textWarp", "`adj` must be an array of numbers")),
51    };
52    let stored = if clear { None } else { Some(preset.to_string()) };
53    format_body(s, &|b| {
54        b.warp.clone_from(&stored);
55        b.warp_adj.clone_from(&adj);
56    })
57}

59pub(super) fn case(s: &mut Session, p: &Value) -> Result<Value> {
60    let mode = str_param(p, "mode").unwrap_or("cycle").to_string();
61    let st = s.doc()?;
62    if let Some(t) = st.selection.text.clone() {
63        let (mut a, mut b) = t.ordered();
64        let body = crate::engine::cmd::text::body_of(st, &t).cloned().unwrap_or_default();
65        let collapsed = a == b;
66        if collapsed {
67            // Shift+F3 with a bare caret changes the word it is in and puts the caret at its start.
68            let (wa, wb) = ed::word_at(&body, a);
69            let is_word = ed::text_range(&body, wa, wb).chars().any(char::is_alphanumeric);
70            if !(is_word && wa.1 <= a.1 && a.1 < wb.1) {
71                return ok();
72            }
73            (a, b) = (wa, wb);
74        }
75        let m = if mode == "cycle" { ed::next_case(&ed::text_range(&body, a, b)).to_string() } else { mode.clone() };
76        crate::engine::cmd::text::edit_text(s, move |body, t| {
77            let last = body.paragraphs.len().saturating_sub(1);
78            for pi in a.0..=b.0.min(last) {
79                let Some(para) = body.paragraphs.get_mut(pi) else { continue };
80                let n = para.char_len();
81                let from = if pi == a.0 { a.1 } else { 0 };
82                let to = if pi == b.0 { b.1 } else { n };
83                let mut pos = 0;
84                let mut pieces = Vec::new();
85                for (ri, r) in para.runs.iter().enumerate() {
86                    // Breaks and fields are read for context (a break ends a word) but not rewritten.
87                    let len = r.char_len();
88                    let rs = from.max(pos).saturating_sub(pos);
89                    let re = to.min(pos + len).saturating_sub(pos);
90                    if re > rs {
91                        pieces.push((ri, rs, re));
92                    }
93                    pos += len;
94                }
95                let mids: Vec<String> = pieces
96                    .iter()
97                    .map(|&(ri, rs, re)| para.runs.get(ri).map(|r| r.content().chars().skip(rs).take(re - rs).collect()).unwrap_or_default())
98                    .collect();
99                let changed = ed::change_case_parts(&mids.iter().map(String::as_str).collect::<Vec<_>>(), &m);
100                for (&(ri, rs, re), new) in pieces.iter().zip(changed) {
101                    let Some(r) = para.runs.get_mut(ri).filter(|r| r.kind == crate::model::text::RunKind::Text) else { continue };
102                    let pre: String = r.text.chars().take(rs).collect();
103                    let post: String = r.text.chars().skip(re).collect();
104                    r.text = format!("{pre}{new}{post}");
105                }
106            }
107            if collapsed {
108                t.anchor = a;
109                t.caret = a;
110            }
111        })?;
112        return ok();
113    }
114    let mode = if mode == "cycle" {
115        let all: String =
116            st.selection.shapes.iter().filter_map(|id| st.shape(*id)).filter_map(|sh| sh.text.as_ref()).map(|t| t.text()).collect::<Vec<_>>().join(
117                "
118",
119            );
120        ed::next_case(&all).to_string()
121    } else {
122        mode
123    };
124    edit_shapes(s, &json!({}), "format.case", |sh| {
125        if let Some(t) = sh.text.as_mut() {
126            for para in &mut t.paragraphs {
127                // Breaks and fields are read for context (a break ends a word) but not rewritten.
128                let texts: Vec<&str> = para.runs.iter().map(|r| r.content()).collect();
129                let changed = ed::change_case_parts(&texts, &mode);
130                for (r, new) in para.runs.iter_mut().zip(changed) {
131                    if r.kind == crate::model::text::RunKind::Text {
132                        r.text = new;
133                    }
134                }
135            }
136        }
137        Ok(())
138    })
139}
