1//! Home ▸ New Slide ▸ Reuse Slides: insert slides from another presentation.
2//!
3//! The slides take the destination's look by default (each one moves to the destination layout
4//! with the same name, else the same kind); with `keepFormatting` the source master and its
5//! layouts come along. Pictures, media and sounds are copied under fresh ids, and shape ids are
6//! renumbered with animations, triggers and connector ends following them. Links to another
7//! reused slide follow its copy.
8

20/// Scale physical measurements in imported content to the destination slide's coordinate space.
21/// Text has one size, so it uses the geometric mean of the horizontal and vertical scale.
22fn scale_text(t: &mut TextBody, sx: f64, sy: f64) {
23    let unit = (sx * sy).sqrt();
24    for v in [&mut t.body.inset_l, &mut t.body.inset_r, &mut t.body.col_spacing].into_iter().flatten() {
25        *v *= sx;
26    }
27    for v in [&mut t.body.inset_t, &mut t.body.inset_b].into_iter().flatten() {
28        *v *= sy;
29    }
30    scale_list(&mut t.list_style, sx, unit);
31    for p in &mut t.paragraphs {
32        for v in [&mut p.props.margin_left, &mut p.props.margin_right, &mut p.props.indent, &mut p.props.default_tab].into_iter().flatten() {
33            *v *= sx;
34        }
35        if let Some(stops) = &mut p.props.tabs {
36            for stop in stops {
37                stop.pos *= sx;
38            }
39        }
40        for spacing in [&mut p.props.line_spacing, &mut p.props.space_before, &mut p.props.space_after] {
41            if let Some(Spacing::Pts(v)) = spacing {
42                *v *= unit;
43            }
44        }
45        if let Some(v) = &mut p.props.bullet_size_points {
46            *v *= unit;
47        }
48        scale_run(&mut p.def_props, unit);
49        scale_run(&mut p.end_props, unit);
50        for r in &mut p.runs {
51            scale_run(&mut r.props, unit);
52        }
53    }
54}

56fn scale_list(list: &mut ListStyle, sx: f64, unit: f64) {
57    for level in list.levels.iter_mut().flatten() {
58        for v in
59            [&mut level.para.margin_left, &mut level.para.margin_right, &mut level.para.indent, &mut level.para.default_tab].into_iter().flatten()
60        {
61            *v *= sx;
62        }
63        for spacing in [&mut level.para.line_spacing, &mut level.para.space_before, &mut level.para.space_after] {
64            if let Some(Spacing::Pts(v)) = spacing {
65                *v *= unit;
66            }
67        }
68        if let Some(v) = &mut level.para.bullet_size_points {
69            *v *= unit;
70        }
71        scale_run(&mut level.run, unit);
72    }
73}

75fn scale_run(run: &mut RunProps, factor: f64) {
76    for v in [&mut run.size, &mut run.spacing, &mut run.kern, &mut run.soft_edge].into_iter().flatten() {
77        *v *= factor;
78    }

81fn scale_shapes(shapes: &mut [crate::model::Shape], sx: f64, sy: f64, depth: usize) {
82    if depth > 64 {
83        return;
84    }
85    let unit = (sx * sy).sqrt();
86    for shape in shapes {
87        if let Some(x) = &mut shape.xfrm {
88            x.x *= sx;
89            x.w *= sx;
90            x.y *= sy;
91            x.h *= sy;
92        }
93        if let Some(text) = &mut shape.text {
94            scale_text(text, sx, sy);
95        }
96        match &mut shape.kind {
97            ShapeKind::Table(table) => {
98                for col in &mut table.cols {
99                    *col *= sx;
100                }
101                for row in &mut table.rows {
102                    row.height *= sy;
103                    for cell in &mut row.cells {
104                        if let Some(margins) = &mut cell.margins {
105                            if let Some(v) = margins.get_mut(0) {
106                                *v *= sx;
107                            }
108                            if let Some(v) = margins.get_mut(1) {
109                                *v *= sx;
110                            }
111                            if let Some(v) = margins.get_mut(2) {
112                                *v *= sy;
113                            }
114                            if let Some(v) = margins.get_mut(3) {
115                                *v *= sy;
116                            }
117                        }
118                        scale_text(&mut cell.text, sx, sy);
119                    }
120                }
121            }
122            ShapeKind::Group { children, child } => {
123                child.x *= sx;
124                child.w *= sx;
125                child.y *= sy;
126                child.h *= sy;
127                scale_shapes(children, sx, sy, depth + 1);
128            }
129            _ => {}
130        }
131        if let Some(effects) = &mut shape.effects {
132            for shadow in [&mut effects.outer_shadow, &mut effects.inner_shadow].into_iter().flatten() {
133                shadow.blur *= unit;
134                shadow.dist *= unit;
135            }
136            if let Some(glow) = &mut effects.glow {
137                glow.radius *= unit;
138            }
139            if let Some(reflection) = &mut effects.reflection {
140                reflection.blur *= unit;
141                reflection.dist *= unit;
142            }
143        }
144    }
145}

147fn scale_master(master: &mut Master, sx: f64, sy: f64) {
148    scale_shapes(&mut master.shapes, sx, sy, 0);
149    let unit = (sx * sy).sqrt();
150    scale_list(&mut master.title_style, sx, unit);
151    scale_list(&mut master.body_style, sx, unit);
152    scale_list(&mut master.other_style, sx, unit);
153    for layout in &mut master.layouts {
154        scale_shapes(&mut layout.shapes, sx, sy, 0);
155    }
156}

189/// Point every media id under `x` at its copy in the destination (`map`: source → destination).
190fn remap(map: &HashMap<MediaId, MediaId>) -> impl FnMut(&mut MediaId) + '_ {
191    move |id: &mut MediaId| {
192        if let Some(to) = map.get(id) {
193            *id = *to;
194        }
195    }

204/// Point connector ends at the renumbered shapes.
205fn remap_connectors(shapes: &mut [crate::model::Shape], map: &HashMap<ShapeId, ShapeId>, depth: usize) {
206    if depth > 64 {
207        return;
208    }
209    for sh in shapes {
210        if let ShapeKind::Connector { start, end } = &mut sh.kind {
211            for c in [start, end] {
212                if let Some((id, _)) = c.as_mut() {
213                    *id = map.get(id).copied().unwrap_or(*id);
214                }
215            }
216        }
217        if let Some(ch) = sh.children_mut() {
218            remap_connectors(ch, map, depth + 1);
219        }
220    }
221}

223/// Point a link to a reused slide at its copy (`map`: source → destination).
224fn relink(h: &mut Option<Hyperlink>, map: &HashMap<SlideId, SlideId>) {
225    let Some(h) = h else { return };
226    if let Action::Slide { slide } = &mut h.action {
227        *slide = map.get(slide).copied().unwrap_or(*slide);
228    }
229    if let Some(z) = &mut h.zoom_return {
230        *z = map.get(z).copied().unwrap_or(*z);
231    }
232}

239fn relink_text(t: &mut TextBody, map: &HashMap<SlideId, SlideId>) {
240    for lv in t.list_style.levels.iter_mut().flatten() {
241        relink_run(&mut lv.run, map);
242    }
243    for p in &mut t.paragraphs {
244        relink_run(&mut p.def_props, map);
245        relink_run(&mut p.end_props, map);
246        for r in &mut p.runs {
247            relink_run(&mut r.props, map);
248        }
249    }

252/// Point the slide links under `shapes` (click and hover actions, text links) at the copies.
253fn relink_shapes(shapes: &mut [crate::model::Shape], map: &HashMap<SlideId, SlideId>, depth: usize) {
254    if depth > 64 {
255        return;
256    }
257    for sh in shapes {
258        relink(&mut sh.click, map);
259        relink(&mut sh.hover, map);
260        if let Some(t) = &mut sh.text {
261            relink_text(t, map);
262        }
263        if let ShapeKind::Table(t) = &mut sh.kind {
264            for c in t.rows.iter_mut().flat_map(|r| r.cells.iter_mut()) {
265                relink_text(&mut c.text, map);
266            }
267        }
268        if let Some(ch) = sh.children_mut() {
269            relink_shapes(ch, map, depth + 1);
270        }
271    }
272}

274/// Fresh shape ids for `shapes`, and the old → new map.
275fn renumber(cmd: &str, doc: &mut Presentation, shapes: &mut [crate::model::Shape]) -> Result<HashMap<ShapeId, ShapeId>> {
276    let old = shape_ids(shapes);
277    super::edit::reid(cmd, doc, shapes)?;
278    let map: HashMap<ShapeId, ShapeId> = old.into_iter().zip(shape_ids(shapes)).collect();
279    remap_connectors(shapes, &map, 0);
280    Ok(map)
281}

283/// The destination layout a reused slide moves to: same name, else same kind, else the first.
284fn matching_layout(doc: &Presentation, name: &str, kind: crate::model::LayoutType) -> LayoutId {
285    let all = || doc.masters.iter().flat_map(|m| m.layouts.iter());
286    all()
287        .find(|l| !name.is_empty() && l.name.eq_ignore_ascii_case(name))
288        .or_else(|| all().find(|l| l.kind == kind))
289        .or_else(|| all().next())
290        .map(|l| l.id)
291        .unwrap_or_default()
292}

294fn collect_table_styles(shapes: &[crate::model::Shape], styles: &mut Vec<String>, seen: &mut HashSet<String>, depth: usize) {
295    if depth > 64 {
296        return;
297    }
298    for shape in shapes {
299        if let ShapeKind::Table(table) = &shape.kind
300            && seen.insert(table.style.to_ascii_lowercase())
301        {
302            styles.push(table.style.clone());
303        }
304        collect_table_styles(shape.children(), styles, seen, depth + 1);
305    }
306}

308fn remap_table_styles(shapes: &mut [crate::model::Shape], styles: &HashMap<String, String>, depth: usize) {
309    if depth > 64 {
310        return;
311    }
312    for shape in shapes {
313        if let ShapeKind::Table(table) = &mut shape.kind
314            && let Some(style) = styles.get(&table.style.to_ascii_lowercase())
315        {
316            table.style = style.clone();
317        }
318        if let Some(children) = shape.children_mut() {
319            remap_table_styles(children, styles, depth + 1);
320        }
321    }
322}

324fn same_table_style(a: &TableStyleDef, b: &TableStyleDef) -> bool {
325    a.name == b.name && a.parts == b.parts && a.background == b.background && a.raw == b.raw

328/// Copy only definitions used on selected slides. Conflicting GUIDs receive a fresh GUID so
329/// existing destination tables continue to resolve to their original definitions.
330fn import_table_styles(cmd: &str, doc: &mut Presentation, src: &Presentation, used: &[String]) -> Result<HashMap<String, String>> {
331    let mut map = HashMap::new();
332    let mut source_defs = HashMap::new();
333    let mut source_ids = HashSet::new();
334    for (index, style) in src.table_styles.iter().enumerate() {
335        let key = style.id.to_ascii_lowercase();
336        source_defs.entry(key.clone()).or_insert(index);
337        source_ids.insert(key);
338    }
339    let mut destination_defs = HashMap::new();
340    for (index, style) in doc.table_styles.iter().enumerate() {
341        destination_defs.entry(style.id.to_ascii_lowercase()).or_insert(index);
342    }
343    for id in used {
344        let key = id.to_ascii_lowercase();
345        if map.contains_key(&key) {
346            continue;
347        }
348        let Some(source_index) = source_defs.get(&key).copied() else { continue };
349        let Some(source) = src.table_styles.get(source_index) else { continue };
350        let destination = destination_defs.get(&key).and_then(|index| doc.table_styles.get(*index));
351        let mut imported = source.clone();
352        if let Some(existing) = destination {
353            if same_table_style(existing, source) {
354                map.insert(key, existing.id.clone());
355                continue;
356            }
357            let candidate = loop {
358                let n = doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE))?;
359                let candidate = format!("{{00000000-0000-4000-8000-{n:012X}}}");
360                let key = candidate.to_ascii_lowercase();
361                if !destination_defs.contains_key(&key) && !source_ids.contains(&key) {
362                    break candidate;
363                }
364            };
365            // The definition's own XML names it too, and PPTX export writes that XML as is.
366            imported.raw = imported.raw.replacen(&format!("styleId=\"{}\"", source.id), &format!("styleId=\"{candidate}\""), 1);
367            imported.id.clone_from(&candidate);
368            map.insert(key, candidate);
369        } else {
370            map.insert(key, imported.id.clone());
371        }
372        destination_defs.insert(imported.id.to_ascii_lowercase(), doc.table_styles.len());
373        doc.table_styles.push(imported);
374    }
375    Ok(map)
376}

378fn reuse(s: &mut Session, p: &Value) -> Result<Value> {
379    let (name, bytes) = super::insert::media_bytes(p, "slide.reuse")?;
380    let src = super::file::open_presentation(&name, &bytes)?;
381    let n = src.slides.len();
382    let picked: Vec<usize> = match p.get("slides") {
383        Some(Value::Array(a)) => {
384            let mut v = vec![];
385            for x in a {
386                let i = x.as_u64().map(|i| i as usize).filter(|i| *i < n);
387                let i = i.ok_or_else(|| bad("slide.reuse", format!("{name} has {n} slides; {x} isn't one of them (indices count from 0)")))?;
388                if !v.contains(&i) {
389                    v.push(i);
390                }
391            }
392            v
393        }
394        Some(Value::String(t)) if !t.trim().is_empty() => super::file::parse_slide_range(t, n)?,
395        _ => (0..n).collect(),
396    };
397    if picked.is_empty() {
398        return Err(bad("slide.reuse", format!("{name} has no slides to reuse")));
399    }
400    let keep = bool_or(p, "keepFormatting", false);
401    let at = usize_param(p, "at");
402    let source_size = src.slide_size;
403    s.edit(|doc, sel| {
404        let at = at.unwrap_or(sel.slide + 1).min(doc.slides.len());
405        let mut used_table_styles = Vec::new();
406        let mut seen_table_styles = HashSet::new();
407        for &i in &picked {
408            if let Some(slide) = src.slides.get(i) {
409                collect_table_styles(&slide.shapes, &mut used_table_styles, &mut seen_table_styles, 0);
410            }
411        }
412        let table_style_map = import_table_styles("slide.reuse", doc, &src, &used_table_styles)?;
413        let scale = |source: f64, destination: f64| {
414            let ratio = destination / source;
415            if source > 0.0 && destination > 0.0 && ratio.is_finite() && ratio > 0.0 { ratio } else { 1.0 }
416        };
417        let sx = scale(source_size.width, doc.slide_size.width);
418        let sy = scale(source_size.height, doc.slide_size.height);
419        // Media: what the slides use is copied under fresh ids.
420        let mut media_map: HashMap<MediaId, MediaId> = HashMap::new();
421        // Keep Source Formatting: bring each source master used, once, with fresh ids.
422        let mut layout_map: HashMap<LayoutId, LayoutId> = HashMap::new();
423        if keep {
424            let mut masters_done: Vec<MasterId> = vec![];
425            for &i in &picked {
426                let Some((m, _)) = src.slides.get(i).and_then(|sl| src.master_for(sl)) else { continue };
427                if masters_done.contains(&m.id) {
428                    continue;
429                }
430                masters_done.push(m.id);
431                let mut m = import_master("slide.reuse", doc, &src, m, &mut media_map, &mut layout_map)?;
432                scale_master(&mut m, sx, sy);
433                doc.masters.push(Arc::new(m));
434            }
435        }
436        let mut copies: Vec<Slide> = vec![];
437        // Source slide → its copy, so links between reused slides follow (forward ones too).
438        let mut slide_map: HashMap<SlideId, SlideId> = HashMap::new();
439        for &i in &picked {
440            let Some(from) = src.slides.get(i) else { continue };
441            copy_media("slide.reuse", doc, &src, &media_refs::slide_media(from), &mut media_map)?;
442            let mut sl: Slide = from.as_ref().clone();
443            remap_table_styles(&mut sl.shapes, &table_style_map, 0);
444            scale_shapes(&mut sl.shapes, sx, sy, 0);
445            media_refs::visit_mut::slide(&mut sl, &mut remap(&media_map));
446            sl.id = SlideId(doc.alloc_id().ok_or_else(|| bad("slide.reuse", NO_ID_SPACE))?);
447            let map = renumber("slide.reuse", doc, &mut sl.shapes)?;
448            for a in &mut sl.animations {
449                a.shape = map.get(&a.shape).copied().unwrap_or(a.shape);
450                a.trigger = a.trigger.map(|t| map.get(&t).copied().unwrap_or(t));
451            }
452            sl.layout = match layout_map.get(&sl.layout) {
453                Some(l) => *l,
454                None => {
455                    let (lname, kind) =
456                        src.layout(from.layout).map(|(_, l)| (l.name.clone(), l.kind)).unwrap_or_else(|| (String::new(), Default::default()));
457                    matching_layout(doc, &lname, kind)
458                }
459            };
460            slide_map.insert(from.id, sl.id);
461            copies.push(sl);
462        }
463        let mut new_ids = vec![];
464        for (k, mut sl) in copies.into_iter().enumerate() {
465            relink_shapes(&mut sl.shapes, &slide_map, 0);
466            relink_text(&mut sl.notes, &slide_map);
467            new_ids.push(sl.id);
468            doc.slides.insert((at + k).min(doc.slides.len()), Arc::new(sl));
469        }
470        sel.slide = (at + new_ids.len()).saturating_sub(1).min(doc.slides.len().saturating_sub(1));
471        sel.slides = new_ids.clone();
472        sel.shapes.clear();
473        sel.text = None;
474        Ok(json!({"slides": new_ids.len(), "ids": new_ids, "from": name, "keepFormatting": keep}))
475    })
476}

478/// A copy of `src`'s master `m` ready to join `doc`: its media copied in, fresh master, layout and
479/// shape ids. `layouts` gains each source layout id → its copy's.
480pub(crate) fn import_master(
481    cmd: &str,
482    doc: &mut Presentation,
483    src: &Presentation,
484    m: &Master,
485    media_map: &mut HashMap<MediaId, MediaId>,
486    layouts: &mut HashMap<LayoutId, LayoutId>,
487) -> Result<Master> {
488    let mut used = vec![];
489    media_refs::visit::master(m, &mut |id: &MediaId| used.push(*id));
490    copy_media(cmd, doc, src, &used, media_map)?;
491    let mut m = m.clone();
492    media_refs::visit_mut::master(&mut m, &mut remap(media_map));
493    m.id = MasterId(doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE))?);
494    renumber(cmd, doc, &mut m.shapes)?;
495    for l in &mut m.layouts {
496        let new = LayoutId(doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE))?);
497        layouts.insert(l.id, new);
498        l.id = new;
499        renumber(cmd, doc, &mut l.shapes)?;
500    }
501    Ok(m)
502}

504/// Copy the source media `used` into `doc` under fresh ids, once each (identical bytes already in
505/// `doc` are shared; a linked item keeps its link).
506fn copy_media(cmd: &str, doc: &mut Presentation, src: &Presentation, used: &[MediaId], map: &mut HashMap<MediaId, MediaId>) -> Result<()> {
507    for &id in used {
508        if map.contains_key(&id) {
509            continue;
510        }
511        let Some(m) = src.media(id) else { continue };
512        let new = if m.link.is_some() {
513            let mut item = m.clone();
514            item.id = MediaId(doc.alloc_id().ok_or_else(|| bad(cmd, NO_ID_SPACE))?);
515            let new = item.id;
516            doc.media.push(item);
517            new
518        } else {
519            doc.add_media(&m.name, &m.content_type, m.data.as_ref().clone()).ok_or_else(|| bad(cmd, NO_ID_SPACE))?
520        };
521        map.insert(id, new);
522    }
523    Ok(())
524}
