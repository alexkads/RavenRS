26/// A picture compressed for one shape: new bytes, content type and the crop it now needs.
27struct Packed {
28    bytes: Vec<u8>,
29    content_type: &'static str,
30    crop: [f64; 4],
31}
32
33/// Whether `crop` cuts something away (and only cuts: negative values pad, and are kept).
34fn real_crop(c: &[f64; 4]) -> bool {
35    c.iter().all(|v| (0.0..1.0).contains(v)) && c.iter().any(|v| *v > 0.0) && c[0] + c[2] < 1.0 && c[1] + c[3] < 1.0
36}

38/// Largest picture decoded for compression, in pixels (a 61 MP camera frame fits); bigger ones
39/// are left as they are.
40const MAX_PIXELS: u64 = 64_000_000;
41/// Most memory the decoder may allocate for one picture.
42const MAX_DECODE_BYTES: u64 = 512 << 20;
43
44/// Decode `data` within [`MAX_PIXELS`] and [`MAX_DECODE_BYTES`]: the size is read from the header
45/// first, so a picture claiming to be huge is never decoded.
46fn decode(data: &[u8], format: image::ImageFormat) -> Option<image::DynamicImage> {
47    let (w, h) = image::ImageReader::with_format(std::io::Cursor::new(data), format).into_dimensions().ok()?;
48    if w == 0 || h == 0 || u64::from(w) * u64::from(h) > MAX_PIXELS {
49        return None;
50    }
51    let mut reader = image::ImageReader::with_format(std::io::Cursor::new(data), format);
52    let mut limits = image::Limits::default();
53    limits.max_image_width = Some(w);
54    limits.max_image_height = Some(h);
55    limits.max_alloc = Some(MAX_DECODE_BYTES);
56    reader.limits(limits);
57    reader.decode().ok()
58}

60/// Compress `data` for a picture shown `w_pt` × `h_pt` with `crop`, at `ppi`. `None` when the
61/// format isn't a raster we re-encode (SVG, EMF, GIF…) or nothing would get smaller.
62fn pack(data: &[u8], crop: [f64; 4], w_pt: f64, h_pt: f64, ppi: f64, delete_cropped: bool) -> Option<Packed> {
63    use image::ImageFormat as F;
64    let format = image::guess_format(data).ok()?;
65    if !matches!(format, F::Png | F::Jpeg | F::Bmp | F::Tiff | F::WebP) {
66        return None;
67    }
68    let mut img = decode(data, format)?;
69    let (iw, ih) = (img.width() as f64, img.height() as f64);
70    if iw < 1.0 || ih < 1.0 {
71        return None;
72    }
73    let cut = delete_cropped && real_crop(&crop);
74    let mut new_crop = crop;
75    // The share of the image that is visible across and down.
76    let (mut fx, mut fy) = ((1.0 - crop[0] - crop[2]).max(0.01), (1.0 - crop[1] - crop[3]).max(0.01));
77    if cut {
78        let x = (crop[0] * iw).floor() as u32;
79        let y = (crop[1] * ih).floor() as u32;
80        let w = ((fx * iw).round() as u32).clamp(1, img.width() - x.min(img.width() - 1));
81        let h = ((fy * ih).round() as u32).clamp(1, img.height() - y.min(img.height() - 1));
82        img = img.crop_imm(x, y, w, h);
83        new_crop = [0.0; 4];
84        (fx, fy) = (1.0, 1.0);
85    } else if !crop.iter().all(|v| v.is_finite()) {
86        return None;
87    }
88    // Pixels needed for the whole (possibly still cropped) image at `ppi`.
89    let need_w = (w_pt.abs() / 72.0 * ppi / fx).ceil().max(1.0);
90    let need_h = (h_pt.abs() / 72.0 * ppi / fy).ceil().max(1.0);
91    let k = (need_w / img.width() as f64).max(need_h / img.height() as f64);
92    let resized = k < 1.0;
93    if resized {
94        let w = ((img.width() as f64 * k).round() as u32).max(1);
95        let h = ((img.height() as f64 * k).round() as u32).max(1);
96        img = img.resize_exact(w, h, image::imageops::FilterType::Lanczos3);
97    }
98    if !resized && !cut {
99        return None;
100    }
101    let alpha = img.color().has_alpha() && img.to_rgba8().pixels().any(|p| p.0[3] < 255);
102    let mut out = std::io::Cursor::new(Vec::new());
103    let content_type = if format == F::Jpeg && !alpha {
104        let rgb = img.to_rgb8();
105        image::codecs::jpeg::JpegEncoder::new_with_quality(&mut out, 85).encode_image(&rgb).ok()?;
106        "image/jpeg"
107    } else {
108        img.write_to(&mut out, F::Png).ok()?;
109        "image/png"
110    };
111    let bytes = out.into_inner();
112    // Never make a picture bigger just to resample it; deleting a crop is worth it regardless.
113    if bytes.len() >= data.len() && !cut {
114        return None;
115    }
116    Some(Packed { bytes, content_type, crop: new_crop })
117}

119/// Pictures accepted by `want`, with width and height measured in slide space through group transforms.
120fn pictures_mut<'a>(
121    shapes: &'a mut [Shape],
122    want: &dyn Fn(ShapeId) -> bool,
123    out: &mut Vec<(&'a mut Shape, f64, f64)>,
124    to_slide: crate::geom::Affine,
125    depth: usize,
126) {
127    if depth > 64 {
128        return;
129    }
130    for sh in shapes {
131        let Some(xfrm) = sh.xfrm else { continue };
132        let local_to_slide = to_slide * xfrm.affine();
133        if matches!(&sh.kind, ShapeKind::Picture { .. }) && want(sh.id) {
134            let origin = local_to_slide * crate::geom::Point::ZERO;
135            let across = local_to_slide * crate::geom::Point::new(xfrm.w, 0.0);
136            let down = local_to_slide * crate::geom::Point::new(0.0, xfrm.h);
137            out.push((sh, (across - origin).hypot(), (down - origin).hypot()));
138            continue;
139        }
140        if let ShapeKind::Group { children, child } = &mut sh.kind {
141            let child_rect = crate::geom::Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
142            let child_to_slide = to_slide * crate::geom::group_child_affine(&xfrm, child_rect);
143            pictures_mut(children, want, out, child_to_slide, depth + 1);
144        }
145    }
146}

148fn compress(s: &mut Session, p: &Value) -> Result<Value> {
149    let ppi = f64_param(p, "ppi").unwrap_or(220.0);
150    if !ppi.is_finite() || !(36.0..=1200.0).contains(&ppi) {
151        return Err(bad("picture.compress", "`ppi` must be between 36 and 1200 (PowerPoint offers 330, 220, 150 and 96)"));
152    }
153    let delete_cropped = bool_or(p, "deleteCropped", true);
154    let all = bool_or(p, "all", false);
155    let ids: HashSet<ShapeId> = if all { HashSet::new() } else { targets(s, p)?.into_iter().collect() };
156    if !all && ids.is_empty() {
157        return Err(bad("picture.compress", "select a picture, or pass all: true"));
158    }
159    let current = s.doc()?.selection.slide;
160    s.edit(|doc, _| {
161        let want = |id: ShapeId| all || ids.contains(&id);
162        let mut before = 0usize;
163        let mut after = 0usize;
164        let mut count = 0usize;
165        let mut changed: Vec<MediaId> = vec![];
166        // The same picture at the same size and crop compresses once.
167        let mut cache: HashMap<(u32, [u64; 4], u64, u64), Option<(MediaId, [f64; 4], usize)>> = HashMap::new();
168        let range: Vec<usize> = if all { (0..doc.slides.len()).collect() } else { vec![current] };
169        for i in range {
170            let Some(slide) = doc.slides.get(i).cloned() else { continue };
171            let mut slide = slide.as_ref().clone();
172            let mut pics = vec![];
173            pictures_mut(&mut slide.shapes, &want, &mut pics, crate::geom::Affine::IDENTITY, 0);
174            let mut touched = false;
175            for (sh, w, h) in pics {
176                let ShapeKind::Picture { fill } = &mut sh.kind else { continue };
177                let PictureFill { media, crop, .. } = fill.clone();
178                let key = (media.0, crop.map(f64::to_bits), w.to_bits(), h.to_bits());
179                let hit = match cache.get(&key) {
180                    Some(v) => *v,
181                    None => {
182                        let packed = doc.media(media).filter(|m| !m.data.is_empty()).and_then(|m| {
183                            let old = m.data.len();
184                            pack(&m.data, crop, w, h, ppi, delete_cropped).map(|pk| (pk, old, m.name.clone()))
185                        });
186                        let v = match packed {
187                            Some((pk, old, name)) => {
188                                before += old;
189                                after += pk.bytes.len();
190                                let stem = std::path::Path::new(&name).file_stem().map(|x| x.to_string_lossy().to_string()).unwrap_or(name);
191                                let ext = if pk.content_type == "image/jpeg" { "jpg" } else { "png" };
192                                let id = doc
193                                    .add_media(&format!("{stem}.{ext}"), pk.content_type, pk.bytes)
194                                    .ok_or_else(|| bad("picture.compress", NO_ID_SPACE))?;
195                                Some((id, pk.crop, old))
196                            }
197                            None => None,
198                        };
199                        cache.insert(key, v);
200                        v
201                    }
202                };
203                if let Some((id, new_crop, _)) = hit {
204                    changed.push(media);
205                    fill.media = id;
206                    fill.crop = new_crop;
207                    count += 1;
208                    touched = true;
209                }
210            }
211            if touched && let Some(sl) = doc.slides.get_mut(i) {
212                *sl = Arc::new(slide);
213            }
214        }
215        // Drop the originals nothing uses any more, so the saved file actually shrinks.
216        if !changed.is_empty() {
217            // Typed walk of every reference (artistic originals, fills, posters, sounds, fonts…).
218            let used: HashSet<MediaId> = crate::engine::media_refs::presentation_media(doc).into_iter().collect();
219            doc.media.retain(|m| !changed.contains(&m.id) || used.contains(&m.id));
220        }
221        Ok(json!({"pictures": count, "before": before, "after": after, "ppi": ppi}))
222    })
223}

372    /// A PNG whose header claims `w` × `h` (no pixel data): enough for the size check.
373    fn png_header(w: u32, h: u32) -> Vec<u8> {
374        fn crc(bytes: &[u8]) -> u32 {
375            let mut c = 0xFFFF_FFFFu32;
376            for b in bytes {
377                c ^= u32::from(*b);
378                for _ in 0..8 {
379                    c = if c & 1 == 1 { 0xEDB8_8320 ^ (c >> 1) } else { c >> 1 };
380                }
381            }
382            !c
383        }
384        fn chunk(out: &mut Vec<u8>, kind: &[u8], data: &[u8]) {
385            let body: Vec<u8> = kind.iter().chain(data).copied().collect();
386            out.extend((data.len() as u32).to_be_bytes());
387            out.extend(&body);
388            out.extend(crc(&body).to_be_bytes());
389        }
390        let mut ihdr = w.to_be_bytes().to_vec();
391        ihdr.extend(h.to_be_bytes());
392        ihdr.extend([8, 2, 0, 0, 0]);
393        let mut out = b"\x89PNG\r\n\x1a\n".to_vec();
394        chunk(&mut out, b"IHDR", &ihdr);
395        chunk(&mut out, b"IDAT", &[]);
396        chunk(&mut out, b"IEND", &[]);
397        out
398    }
