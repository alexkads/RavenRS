102fn new(s: &mut Session, p: &Value) -> Result<Value> {
103    let template = str_param(p, "template").filter(|t| !t.is_empty());
104    let theme_name = str_param(p, "theme").filter(|t| !t.is_empty());
105    if template.is_some() && theme_name.is_some() {
106        return Err(bad("file.new", "pass `theme` or `template`, not both"));
107    }
108    if template.is_some() && bool_or(p, "blank", false) {
109        return Err(bad("file.new", "a template is not a blank presentation"));
110    }
111    let size = slide_size(p);
112    let (doc, template_name) = if let Some(name) = template {
113        let doc = crate::engine::templates::build(name, size)?;
114        let chosen = crate::engine::templates::find(name).map(|t| t.name);
115        (doc, chosen)
116    } else {
117        let theme = theme_name.and_then(theme_named).unwrap_or_default();
118        (defaults::blank_presentation(size, theme, !bool_or(p, "blank", false)), None)
119    };
120    let slides = doc.slides.len();
121    let title = s.next_untitled();
122    let i = s.add_document(DocState::new(doc, None, title));
123    let mut out = json!({"document": i, "slides": slides});
124    if let Some(name) = template_name {
125        out["template"] = json!(name);
126    }
127    Ok(out)
128}

135/// What a file lost on opening.
136#[derive(Debug, Default, Clone)]
137pub struct Damage {
138    /// Pictures and media a `.deckcraft` file had lost or damaged.
139    pub media: Vec<String>,
140    /// Slides a `.pptx` lists but that couldn't be read, so they are not in the presentation.
141    pub slides: Vec<String>,
142}
143
144impl Damage {
145    pub fn is_empty(&self) -> bool {
146        self.media.is_empty() && self.slides.is_empty()
147    }
148}

150/// [`open_presentation`], plus what the file had lost or damaged.
151pub fn open_checked(name: &str, bytes: &[u8]) -> Result<(Presentation, Damage)> {
152    if crate::format::sniff(bytes) {
153        let (p, media) = crate::format::load_checked(bytes).map_err(|e| EngineError::Other(e.to_string()))?;
154        return Ok((p, Damage { media, slides: vec![] }));
155    }
156    open_other(name, bytes).map(|(p, slides)| (p, Damage { media: vec![], slides }))
157}
158
159fn open_other(name: &str, bytes: &[u8]) -> Result<(Presentation, Vec<String>)> {
160    if crate::pptx::sniff(bytes) {
161        let (mut p, lost_slides) = crate::pptx::import_checked(bytes).map_err(|e| EngineError::Other(e.to_string()))?;
162        crate::format::repair(&mut p);
163        probe_clips(&mut p);
164        return Ok((p, lost_slides));
165    }
166    if crate::pptx::is_encrypted(bytes) {
167        return Err(EngineError::Other(format!("{name} is password-protected. DeckCraft can't open password-protected presentations yet.")));
168    }
169    let lower = name.to_ascii_lowercase();
170    let rtf = crate::format::rtf::sniff_rtf(bytes);
171    if rtf || lower.ends_with(".txt") || lower.ends_with(".md") {
172        let text = String::from_utf8_lossy(bytes);
173        let text = if rtf { std::borrow::Cow::Owned(crate::format::rtf::rtf_to_outline(&text)) } else { text };
174        let mut p = defaults::blank_presentation(defaults::WIDE, Default::default(), false);
175        if lower.ends_with(".md") {
176            crate::format::markdown_to_slides(&mut p, &text);
177        } else {
178            crate::format::outline_to_slides(&mut p, &text);
179        }
180        if p.slides.is_empty() {
181            p = defaults::new_presentation(None);
182        }
183        return Ok((p, vec![]));
184    }
185    Err(EngineError::Other(format!("{name}: not a presentation DeckCraft can read")))
186}

188/// Fill in each audio and video clip's length and picture size from its bytes: a `.pptx` doesn't
189/// store them. Headers only, so opening never decodes a whole audio file.
190fn probe_clips(p: &mut Presentation) {
191    fn rec(shapes: &mut [Shape], media: &[MediaItem], probed: &mut HashMap<MediaId, Option<MediaInfo>>, depth: usize) {
192        if depth > 64 {
193            return;
194        }
195        for s in shapes {
196            if let ShapeKind::Media(m) = &mut s.kind
197                && (m.duration_ms == 0 || m.width == 0 && m.height == 0)
198                && let Some(info) = probed
199                    .entry(m.media)
200                    .or_insert_with(|| media.iter().find(|i| i.id == m.media).and_then(|i| crate::media::probe_headers(&i.data).ok()))
201            {
202                if m.duration_ms == 0 {
203                    m.duration_ms = info.duration_ms.min(u32::MAX as u64) as u32;
204                }
205                if m.width == 0
206                    && m.height == 0
207                    && let Some(v) = &info.video
208                {
209                    (m.width, m.height) = (v.width, v.height);
210                }
211            }
212            if let Some(children) = s.children_mut() {
213                rec(children, media, probed, depth + 1);
214            }
215        }
216    }
217    let mut probed = HashMap::new();
218    // Just read, so no slide is shared yet and `make_mut` doesn't copy.
219    for slide in &mut p.slides {
220        rec(&mut Arc::make_mut(slide).shapes, &p.media, &mut probed, 0);
221    }
222}

224/// An outline file's text: RTF is read into the tab-indented outline, anything else as UTF-8.

230pub fn add_opened(s: &mut Session, name: &str, path: Option<String>, p: Presentation) -> usize {
231    let title = std::path::Path::new(name).file_stem().map(|n| n.to_string_lossy().to_string()).unwrap_or_else(|| name.to_string());
232    crate::engine::fonts::install_embedded_fonts(&p);
233    let missing_fonts = crate::engine::fonts::missing_fonts(&p);
234    let mut st = DocState::new(p, path, title);
235    st.missing_fonts = missing_fonts;
236    s.add_document(st)
237}

239/// A presentation read from disk, not yet added to a session: [`read_presentation_file`] needs no
240/// session, so a host can run it on a background thread, hand the result to
241/// [`Session::stage_loaded`] and add it with `file.adopt`.
242#[derive(Debug)]
243pub struct LoadedFile {
244    pub path: String,
245    pub doc: Presentation,
246    /// A native or PPTX file (saving goes back to it); an outline import has no file to save to.
247    pub native: bool,
248    /// What the file had lost or damaged.
249    pub damaged: Damage,
250}

255    let bytes = crate::engine::io::read_file(path).map_err(EngineError::Other)?;
256    loaded_file(path, &bytes)

259/// Parse `bytes` as the presentation file `path`, as [`read_presentation_file`] does after
260/// reading it. The web host bridge uses it for a file its page sent, which then saves back to
261/// that name like a file opened on the desktop.
262pub fn loaded_file(path: &str, bytes: &[u8]) -> Result<LoadedFile> {
263    let (doc, damaged) = open_checked(path, bytes)?;
264    let native = crate::format::sniff(bytes) || crate::pptx::sniff(bytes);
265    Ok(LoadedFile { path: path.to_string(), doc, native, damaged })
266}

268/// Add a loaded presentation as the active document: the shared end of `file.open` and
269/// `file.adopt`. A damaged file opens as an untitled repaired copy, so Save can't overwrite the
270/// original with the empty media.
271fn adopt_loaded(s: &mut Session, f: LoadedFile) -> Result<Value> {
272    let LoadedFile { path, doc, native, damaged } = f;
273    let keep_path = native && damaged.is_empty();
274    let i = add_opened(s, &path, keep_path.then(|| path.clone()), doc);
275    report_damage(s, &damaged)?;
276    let st = s.doc()?;
277    Ok(json!({
278        "document": i, "slides": st.doc.slides.len(), "missingFonts": st.missing_fonts,
279        "damagedMedia": damaged.media, "lostSlides": damaged.slides,
280    }))
281}

289/// `file.adopt {path, token?}`: add the presentation a host read in the background and staged
290/// with [`Session::stage_loaded`]. Without that staged file (a replayed journal, a stale token)
291/// it reads `path` like `file.open`, so the journal entry stays replayable.
292fn adopt(s: &mut Session, p: &Value) -> Result<Value> {
293    let path = str_param(p, "path").ok_or_else(|| bad("file.adopt", "missing `path`"))?;
294    let staged = p.get("token").and_then(Value::as_u64).and_then(|t| s.take_loaded(t)).filter(|f| f.path == path);
295    match staged {
296        Some(f) => adopt_loaded(s, f),
297        #[cfg(not(target_arch = "wasm32"))]
298        None => adopt_loaded(s, read_presentation_file(path)?),
299        #[cfg(target_arch = "wasm32")]
300        None => Err(bad("file.adopt", format!("{path} was not loaded"))),
301    }
302}

308fn open_bytes(s: &mut Session, p: &Value) -> Result<Value> {
309    let name = str_param(p, "name").unwrap_or("Presentation");
310    let data = str_param(p, "data").and_then(base64_decode).ok_or_else(|| bad("file.openBytes", "missing or invalid base64 `data`"))?;
311    let (doc, damaged) = open_checked(name, &data)?;
312    let i = add_opened(s, name, None, doc);
313    report_damage(s, &damaged)?;
314    Ok(json!({"document": i, "missingFonts": s.doc()?.missing_fonts, "damagedMedia": damaged.media, "lostSlides": damaged.slides}))
315}

317/// Tell the user which pictures, media and slides a damaged file lost, and mark the copy repaired.
318fn report_damage(s: &mut Session, damaged: &Damage) -> Result<()> {
319    if damaged.is_empty() {
320        return Ok(());
321    }
322    let st = s.doc_mut()?;
323    st.title = format!("{} (Repaired)", st.title);
324    let mut text = String::new();
325    let n = damaged.media.len();
326    if n > 0 {
327        text = format!(
328            "{n} picture{} or media file{} in this presentation couldn't be read and {} empty. ",
329            if n == 1 { "" } else { "s" },
330            if n == 1 { "" } else { "s" },
331            if n == 1 { "is" } else { "are" },
332        );
333    }
334    let n = damaged.slides.len();
335    if n > 0 {
336        text.push_str(&format!(
337            "{n} slide{} in this file couldn't be read and {} left out. ",
338            if n == 1 { "" } else { "s" },
339            if n == 1 { "was" } else { "were" },
340        ));
341    }
342    text.push_str("It was opened as a repaired copy; save it under a new name to keep the original file.");
343    s.ui_requests.push(crate::engine::UiRequest::Message { text });
344    Ok(())
345}

347/// Encode the presentation in `format` (`deckcraft`, `pptx`, `potx`, `ppsx`, `outline`, `pdf`, `gif`).
348pub fn save_bytes(doc: &Presentation, format: &str) -> Result<Vec<u8>> {
349    match format {
350        f @ ("pptx" | "potx" | "ppsx") => crate::pptx::export_as(doc, f).map_err(|e| EngineError::Other(e.to_string())),
351        "outline" | "txt" => Ok(crate::format::slides_to_outline(doc).into_bytes()),
352        "rtf" => Ok(crate::format::rtf::slides_to_rtf(doc).into_bytes()),
353        "pdf" => crate::pdf::export(doc, &Default::default()).map_err(|e| EngineError::Other(e.to_string())),
354        "gif" => gif_bytes(doc, &Value::Null),
355        "png" | "jpeg" | "jpg" => Err(EngineError::Other("saving a presentation as a single image isn't supported; use file.export".into())),
356        _ => crate::format::save(doc).map_err(|e| EngineError::Other(e.to_string())),
357    }
358}

363/// PDF options from command params: `{layout: slides|notes|handouts, perPage, dpi, slides,
364/// includeHidden, textLayer, frame, paper: letter|a4|legal|a3, orientation: portrait|landscape}` for a presentation of `count` slides. `slides` is a list of
365/// 0-based indices or a range as typed in PowerPoint's Print page (`"1-3, 5"`, 1-based).
366pub fn pdf_options(p: &Value, count: usize) -> Result<crate::pdf::PdfOptions> {
367    let mut o = crate::pdf::PdfOptions { layout: crate::pdf::PageLayout::Slides, ..Default::default() };
368    o.layout = match str_param(p, "layout").unwrap_or("slides") {
369        "notes" => crate::pdf::PageLayout::Notes,
370        "handouts" => {
371            let per_page = opt_usize_param(p, "perPage", "file.export")?.unwrap_or(6);
372            if ![1, 2, 3, 4, 6, 9].contains(&per_page) {
373                return Err(bad("file.export", format!("perPage {per_page} isn't one of 1, 2, 3, 4, 6, 9")));
374            }
375            crate::pdf::PageLayout::Handouts { per_page: per_page as u8 }
376        }
377        _ => crate::pdf::PageLayout::Slides,
378    };
379    o.dpi = f64_or(p, "dpi", o.dpi).clamp(36.0, 600.0);
380    o.slides = match p.get("slides") {
381        Some(Value::Array(a)) => Some(a.iter().filter_map(|v| v.as_u64().map(|x| x as usize)).filter(|i| *i < count).collect()),
382        Some(Value::String(text)) if !text.trim().is_empty() => Some(parse_slide_range(text, count)?),
383        _ => None,
384    };
385    o.include_hidden = bool_or(p, "includeHidden", false);
386    o.text_layer = bool_or(p, "textLayer", true);
387    o.frame_slides = bool_or(p, "frame", true);
388    let landscape = match str_param(p, "orientation").unwrap_or("portrait") {
389        "portrait" => false,
390        "landscape" => true,
391        other => return Err(bad("file.export", format!("orientation “{other}” isn't portrait or landscape"))),
392    };
393    let paper = str_param(p, "paper").unwrap_or("letter");
394    o.paper = crate::pdf::paper_size(paper, landscape).ok_or_else(|| {
395        let ids: Vec<&str> = crate::pdf::PAPERS.iter().map(|p| p.0).collect();
396        bad("file.export", format!("paper “{paper}” isn't one of {}", ids.join(", ")))
397    })?;
398    Ok(o)
399}

401/// A slide range as typed in the Print page's Slides box (`"1-3, 5"`; 1-based, either separator
402/// `,` or `;`, ranges either way round) → 0-based indices in order, without repeats. Numbers past
403/// the last slide or a range that isn't numbers is an error, as in PowerPoint.
404pub fn parse_slide_range(text: &str, count: usize) -> Result<Vec<usize>> {
405    let invalid = || bad("file.export", format!("“{text}” isn't a valid slide range: use numbers and ranges such as 1-3, 5"));
406    let mut out: Vec<usize> = vec![];
407    let mut seen: std::collections::HashSet<usize> = std::collections::HashSet::new();
408    for part in text.split([',', ';']).map(str::trim).filter(|p| !p.is_empty()) {
409        let (a, b) = match part.split_once('-') {
410            Some((a, b)) => (a.trim(), b.trim()),
411            None => (part, part),
412        };
413        let digits = |s: &str| !s.is_empty() && s.bytes().all(|c| c.is_ascii_digit());
414        if !digits(a) || !digits(b) {
415            return Err(invalid());
416        }
417        let (a, b) = (a.parse::<usize>().map_err(|_| invalid())?, b.parse::<usize>().map_err(|_| invalid())?);
418        let (lo, hi) = (a.min(b), a.max(b));
419        if lo == 0 || hi > count {
420            return Err(bad("file.export", format!("the presentation has {count} slides; “{part}” is outside them")));
421        }
422        for i in lo - 1..hi {
423            if seen.insert(i) {
424                out.push(i);
425            }
426        }
427    }
428    if out.is_empty() {
429        return Err(invalid());
430    }
431    Ok(out)
432}

439/// The output extensions [`format_for_path`] maps to a writer of that format.
440pub const WRITABLE_EXTENSIONS: &[&str] = &["deckcraft", "pptx", "potx", "ppsx", "pdf", "png", "jpg", "jpeg", "gif", "txt", "md", "rtf"];
441
442/// The format a file named `path` is written in, or `None` when its extension names no format
443/// DeckCraft writes (`.odp`, `.svg`, no extension …): [`format_for_path`] would save
444/// those as native `.deckcraft` bytes under the wrong name.
445pub fn writable_format_for_path(path: &str) -> Option<&'static str> {
446    let ext = std::path::Path::new(path).extension()?.to_str()?.to_ascii_lowercase();
447    WRITABLE_EXTENSIONS.contains(&ext.as_str()).then(|| format_for_path(path))
448}

455pub fn format_for_path(path: &str) -> &'static str {
456    let l = path.to_ascii_lowercase();
457    if l.ends_with(".pptx") {
458        "pptx"
459    } else if l.ends_with(".potx") {
460        "potx"
461    } else if l.ends_with(".ppsx") {
462        "ppsx"
463    } else if l.ends_with(".png") {
464        "png"
465    } else if l.ends_with(".jpg") || l.ends_with(".jpeg") {
466        "jpeg"
467    } else if l.ends_with(".pdf") {
468        "pdf"
469    } else if l.ends_with(".gif") {
470        "gif"
471    } else if l.ends_with(".txt") || l.ends_with(".md") {
472        "outline"
473    } else if l.ends_with(".rtf") {
474        "rtf"
475    } else {
476        "deckcraft"
477    }
478}

480/// The active presentation is saved as it is now: no unsaved changes, and `path` (when given)
481/// becomes its file. `file.save` calls it after writing; the web host bridge calls it once the
482/// page has the bytes.
483pub fn mark_saved(s: &mut Session, path: Option<String>) -> Result<()> {
484    let st = s.doc_mut()?;
485    st.saved_doc = st.doc.clone();
486    if path.is_some() {
487        st.path = path;
488    }
489    st.revision += 1;
490    let uid = st.uid;
491    if let Some(dir) = &s.recovery_dir {
492        crate::engine::recovery::discard(dir, uid);
493    }
494    Ok(())
495}

528/// `scale` for rendering `doc`'s slides to an image: within 0.01–16, and small enough that the
529/// slide fits the renderer's largest picture ([`crate::render::MAX_SIDE`]). Past that the
530/// renderer clips, so a 4× export of a 4032 pt slide came out cropped to its first 16,000 pixels.
531pub fn image_scale(doc: &Presentation, scale: f64) -> f64 {
532    let scale = if scale.is_finite() { scale.clamp(0.01, 16.0) } else { 1.0 };
533    let longest = doc.slide_size.width.max(doc.slide_size.height).max(1.0);
534    scale.min(f64::from(crate::render::MAX_SIDE) / longest)
535}

537/// Render slide `index` (current when absent) to PNG bytes.
538pub fn render_png(doc: &Presentation, index: usize, scale: f64, edit: bool) -> (Vec<u8>, u32, u32) {
539    render_png_with(doc, index, scale, edit, false)
540}
541
542/// [`render_png`]; `transparent` leaves the slide background out (#784).
543pub fn render_png_with(doc: &Presentation, index: usize, scale: f64, edit: bool, transparent: bool) -> (Vec<u8>, u32, u32) {
544    let scale = image_scale(doc, scale);
545    let opts = crate::render::RenderOpts { scale, edit, transparent_background: transparent, ..Default::default() };
546    let img = crate::render::render_slide(doc, index, &opts);
547    (img.to_png_at(crate::render::dpi_for_scale(scale)), img.width, img.height)
548}
549
550fn finite_scale(scale: f64) -> f64 {
551    if scale.is_finite() { scale.clamp(0.01, 16.0) } else { 1.0 }
552}
553
554fn picture_bytes(img: &crate::render::Image, format: &str, scale: f64) -> Vec<u8> {
555    let dpi = crate::render::dpi_for_scale(scale);
556    if format == "png" { img.to_png_at(dpi) } else { img.to_jpeg_at(92, Some(dpi)) }
557}

559fn render(s: &mut Session, p: &Value) -> Result<Value> {
560    let st = s.doc()?;
561    let i = opt_usize_param(p, "slide", "file.render")?.unwrap_or(st.selection.slide);
562    if i >= st.doc.slides.len() {
563        return Err(bad("file.render", format!("no slide {i}")));
564    }
565    let (png, w, h) = render_png_with(&st.doc, i, f64_or(p, "scale", 1.0), bool_or(p, "edit", false), bool_or(p, "transparentBackground", false));
566    Ok(json!({"png": base64_encode(&png), "width": w, "height": h}))
567}

569/// The colour the slide is drawn with at one point: what the Eyedropper takes.
570fn color_at(s: &mut Session, p: &Value) -> Result<Value> {
571    let st = s.doc()?;
572    let i = opt_usize_param(p, "slide", "slide.colorAt")?.unwrap_or(st.selection.slide);
573    if i >= st.doc.slides.len() {
574        return Err(bad("slide.colorAt", format!("no slide {i}")));
575    }
576    let coord = |key: &str| {
577        p.get(key)
578            .and_then(Value::as_f64)
579            .filter(|v| v.is_finite())
580            .ok_or_else(|| bad("slide.colorAt", format!("`{key}` (a number, in slide units) is required")))
581    };
582    let (x, y) = (coord("x")?, coord("y")?);
583    let scale = f64_or(p, "scale", 2.0).clamp(0.01, 16.0);
584    let img = crate::render::render_slide(&st.doc, i, &crate::render::RenderOpts { scale, ..Default::default() });
585    let (px, py) = (x * scale, y * scale);
586    let inside = px >= 0.0 && py >= 0.0 && px < f64::from(img.width) && py < f64::from(img.height);
587    let [r, g, b, a] =
588        img.straight_at(px as u32, py as u32).filter(|_| inside).ok_or_else(|| bad("slide.colorAt", "the point is outside the slide"))?;
589    Ok(json!({"color": format!("{r:02X}{g:02X}{b:02X}"), "alpha": a, "x": x, "y": y}))
590}

592fn export(s: &mut Session, p: &Value) -> Result<Value> {
593    let path = str_param(p, "path").ok_or_else(|| bad("file.export", "missing `path`"))?.to_string();
594    let format = str_param(p, "format").unwrap_or(format_for_path(&path)).to_string();
595    let st = s.doc()?;
596    let scale = finite_scale(f64_or(p, "scale", 2.0));
597    let transparent = bool_or(p, "transparentBackground", false);
598    if transparent && format != "png" {
599        return Err(bad("file.export", format!("a transparent background is for PNG export, not {format}")));
600    }
601    match format.as_str() {
602        "png" | "jpeg" | "jpg" => {
603            let slides: Vec<usize> = match opt_usize_param(p, "slide", "file.export")? {
604                Some(i) => vec![i],
605                None if bool_or(p, "all", false) => (0..st.doc.slides.len()).collect(),
606                None => vec![st.selection.slide],
607            };
608            // Every slide checked before any file is written, so a bad index leaves nothing behind.
609            if let Some(i) = slides.iter().find(|i| **i >= st.doc.slides.len()) {
610                return Err(bad("file.export", format!("no slide {i}")));
611            }
612            let scale = image_scale(&st.doc, scale);
613            let mut written = vec![];
614            for (k, i) in slides.iter().enumerate() {
615                let img = crate::render::render_slide(
616                    &st.doc,
617                    *i,
618                    &crate::render::RenderOpts { scale, transparent_background: transparent, ..Default::default() },
619                );
620                let bytes = picture_bytes(&img, &format, scale);
621                let out = if slides.len() == 1 {
622                    path.clone()
623                } else {
624                    let pth = std::path::Path::new(&path);
625                    let stem = pth.file_stem().map(|x| x.to_string_lossy().to_string()).unwrap_or_else(|| "Slide".into());
626                    let ext = pth.extension().map(|x| x.to_string_lossy().to_string()).unwrap_or_else(|| format.clone());
627                    pth.with_file_name(format!("{stem}{}.{ext}", k + 1)).to_string_lossy().to_string()
628                };
629                // Say which slides made it when one can't be written (disk full part-way).
630                write_file(&out, &bytes)
631                    .map_err(|e| if written.is_empty() { e } else { EngineError::Other(format!("{e} (slides 1–{k} were written)")) })?;
632                written.push(out);
633            }
634            Ok(json!({"files": written}))
635        }
636        "pdf" => {
637            let bytes = pdf_bytes(&st.doc, p)?;
638            write_file(&path, &bytes)?;
639            Ok(json!({"path": path, "bytes": bytes.len()}))
640        }
641        "gif" => {
642            // `slides` may be a range, as for PDF ("1-3,5").
643            let mut gp = p.clone();
644            if let Some(text) = p.get("slides").and_then(Value::as_str) {
645                gp["slides"] = json!(parse_slide_range(text, st.doc.slides.len())?);
646            }
647            let out = crate::engine::gif_export::encode(&st.doc, &crate::engine::gif_export::GifOptions::from_params(&gp)?)?;
648            write_file(&path, &out.bytes)?;
649            Ok(json!({
650                "path": path,
651                "bytes": out.bytes.len(),
652                "frames": out.frames,
653                "width": out.width,
654                "height": out.height,
655                "secondsPerSlide": f64::from(out.delay_cs) / 100.0,
656            }))
657        }
658        "pptx" | "potx" | "ppsx" | "deckcraft" | "outline" | "txt" | "rtf" => {
659            let bytes = save_bytes(&st.doc, &format)?;
660            write_file(&path, &bytes)?;
661            Ok(json!({"path": path, "bytes": bytes.len()}))
662        }
663        other => Err(bad("file.export", format!("unknown format `{other}`"))),
664    }
665}

696    let fields = ["title", "subject", "author", "keywords", "comments", "category", "company"];
697    if fields.iter().any(|f| p.get(*f).is_some()) {
698        s.edit(|doc, _| {
699            let pr = &mut doc.props;
700            for f in fields {
701                if let Some(v) = str_param(p, f) {
702                    let slot = match f {
703                        "title" => &mut pr.title,
704                        "subject" => &mut pr.subject,
705                        "author" => &mut pr.author,
706                        "keywords" => &mut pr.keywords,
707                        "comments" => &mut pr.comments,
708                        "category" => &mut pr.category,
709                        _ => &mut pr.company,
710                    };
711                    *slot = v.to_string();
712                }
713            }
714            Ok(())
715        })?;
716    }
717    Ok(serde_json::to_value(&s.doc()?.doc.props).unwrap_or_default())
