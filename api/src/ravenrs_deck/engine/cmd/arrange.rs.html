74fn order(s: &mut Session, p: &Value, how: i32) -> Result<Value> {
75    let ids = targets(s, p)?;
76    // Already in place (front of its group, say): no change and no undo step.
77    let mut trial = s.doc()?.shapes().to_vec();
78    order_tree(&mut trial, &ids, how, 0);
79    if stacking(&trial) == stacking(s.doc()?.shapes()) {
80        return ok();
81    }
82    s.edit(|doc, sel| {
83        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("arrange", "no slide"))?;
84        order_tree(list, &ids, how, 0);
85        Ok(())
86    })?;
87    ok()
88}

90/// Every shape id with its depth, in tree order.
91fn stacking(v: &[Shape]) -> Vec<(ShapeId, usize)> {
92    let mut out = vec![];
93    crate::model::walk(v, &mut |sh, d| out.push((sh.id, d)));
94    out
95}

97/// Restack `ids` among their siblings, in every list of the tree: a shape inside a group moves
98/// within its group and never leaves it (PowerPoint's Selection pane arrows work the same way).
99fn order_tree(list: &mut Vec<Shape>, ids: &[ShapeId], how: i32, depth: usize) {
100    if depth > 64 {
101        return;
102    }
103    if list.iter().any(|x| ids.contains(&x.id)) {
104        order_list(list, ids, how);
105    }
106    for sh in list.iter_mut() {
107        if let Some(children) = sh.children_mut() {
108            order_tree(children, ids, how, depth + 1);
109        }
110    }
111}
112
113fn order_list(list: &mut Vec<Shape>, ids: &[ShapeId], how: i32) {
114    match how {
115        2 => {
116            let (mut moved, rest): (Vec<Shape>, Vec<Shape>) = list.drain(..).partition(|x| ids.contains(&x.id));
117            list.extend(rest);
118            list.append(&mut moved);
119        }
120        -2 => {
121            let (mut moved, rest): (Vec<Shape>, Vec<Shape>) = list.drain(..).partition(|x| ids.contains(&x.id));
122            moved.extend(rest);
123            *list = moved;
124        }
125        1 => {
126            for i in (0..list.len().saturating_sub(1)).rev() {
127                let (a, b) = (list.get(i).map(|x| ids.contains(&x.id)), list.get(i + 1).map(|x| ids.contains(&x.id)));
128                if a == Some(true) && b == Some(false) {
129                    list.swap(i, i + 1);
130                }
131            }
132        }
133        _ => {
134            for i in 1..list.len() {
135                let (a, b) = (list.get(i - 1).map(|x| ids.contains(&x.id)), list.get(i).map(|x| ids.contains(&x.id)));
136                if a == Some(false) && b == Some(true) {
137                    list.swap(i - 1, i);
138                }
139            }
140        }
141    }
142}

144/// Move one shape to `index` among its siblings (0 = back): the Selection pane's drag and drop.
145/// A shape stays in the list that holds it, so a group's child stays in its group.
146fn reorder(s: &mut Session, p: &Value) -> Result<Value> {
147    let id = id_param(p, "id", "arrange.reorder")?.ok_or_else(|| bad("arrange.reorder", "missing `id`"))?;
148    let to = usize_param(p, "index").ok_or_else(|| bad("arrange.reorder", "missing `index`"))?;
149    let st = s.doc()?;
150    let siblings: &[Shape] = match crate::model::parent_of(st.shapes(), id) {
151        None => return Err(bad("arrange.reorder", "no such shape")),
152        Some(None) => st.shapes(),
153        Some(Some(g)) => st.shape(g).map(Shape::children).unwrap_or(&[]),
154    };
155    let from = siblings.iter().position(|x| x.id == id).ok_or_else(|| bad("arrange.reorder", "no such shape"))?;
156    if from == to.min(siblings.len().saturating_sub(1)) {
157        return Ok(json!({"index": from}));
158    }
159    s.edit(|doc, sel| {
160        let tree = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("arrange.reorder", "no slide"))?;
161        let list = crate::model::sibling_list_mut(tree, id).ok_or_else(|| bad("arrange.reorder", "no such shape"))?;
162        let i = list.iter().position(|x| x.id == id).ok_or_else(|| bad("arrange.reorder", "no such shape"))?;
163        let sh = list.remove(i);
164        let at = to.min(list.len());
165        list.insert(at, sh);
166        Ok(json!({"index": at}))
167    })
168}

170/// Bounds of shapes (rotation included).
171pub(crate) fn union_bounds(xs: &[Xfrm]) -> Option<Rect> {
172    let mut it = xs.iter().map(Xfrm::bounds);
173    let first = it.next()?;
174    Some(it.fold(first, |a, b| a.union(b)))
175}

177/// How deep groups may nest: as deep as the PPTX reader reads and the renderer draws. A saved
178/// .deckcraft file stops opening at about 39 (the JSON reader's recursion limit), and walks over
179/// the shape tree stop at 64.
180pub const MAX_GROUP_DEPTH: usize = 32;
181
182/// How many groups deep `shapes` nest (0 without groups).
183fn group_depth(shapes: &[Shape]) -> usize {
184    let mut d = 0;
185    crate::model::walk(shapes, &mut |sh, depth| {
186        if sh.is_group() {
187            d = d.max(depth + 1);
188        }
189    });
190    d
191}

193fn group(s: &mut Session, p: &Value) -> Result<Value> {
194    let ids = targets(s, p)?;
195    s.edit(|doc, sel| {
196        let current = super::current_shapes(doc, sel).cloned().unwrap_or_default();
197        let mut members: Vec<Shape> = current.iter().filter(|x| ids.contains(&x.id)).cloned().collect();
198        if members.len() < 2 {
199            return Err(bad("arrange.group", "select two or more objects on the slide"));
200        }
201        if group_depth(&members) + 1 > MAX_GROUP_DEPTH {
202            return Err(bad("arrange.group", format!("groups can be nested at most {MAX_GROUP_DEPTH} deep")));
203        }
204        for m in &mut members {
205            if m.xfrm.is_none() {
206                m.xfrm = Some(xfrm_of(doc, sel, m));
207            }
208            // Placeholders lose their link when grouped (PowerPoint keeps them out of groups).
209            m.ph = None;
210        }
211        let xs: Vec<Xfrm> = members.iter().filter_map(|m| m.xfrm).collect();
212        let b = union_bounds(&xs).ok_or_else(|| bad("arrange.group", "nothing to group"))?;
213        let gx = Xfrm::new(b.x0, b.y0, b.width(), b.height());
214        let id = ShapeId(doc.alloc_id().ok_or_else(|| bad("arrange.group", NO_ID_SPACE))?);
215        let n = super::current_shapes(doc, sel).map(|v| v.len()).unwrap_or(0);
216        let g = Shape {
217            id,
218            name: format!("Group {}", n + 1),
219            xfrm: Some(gx),
220            kind: ShapeKind::Group { children: members, child: gx },
221            ..Default::default()
222        };
223        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("arrange.group", "no slide"))?;
224        let at = list.iter().rposition(|x| ids.contains(&x.id)).unwrap_or(list.len());
225        list.insert(at + 1, g);
226        list.retain(|x| !ids.contains(&x.id));
227        sel.shapes = vec![id];
228        Ok(json!({"id": id}))
229    })
230}

232/// After an edit: refit every group whose children's geometry changed to the union of its
233/// children (PowerPoint keeps a group's box on its children, so moving one child grows or
234/// shrinks the group). The children keep their place on the slide: the group's box and its child
235/// space change together. `old` is the presentation before the edit; only slides the edit
236/// touched are looked at, and a group whose own box changed (it was moved or resized) is left to
237/// that edit.
238pub(crate) fn refit_groups(old: &crate::model::Presentation, new: &mut crate::model::Presentation) {
239    // Each slide, master and layout is compared with its own earlier state, found by id: a move,
240    // insertion or deletion shifts the indices, and shape ids repeat across slides.
241    let old_slides: std::collections::HashMap<_, _> = old.slides.iter().map(|s| (s.id, s)).collect();
242    for slide in new.slides.iter_mut() {
243        let Some(before) = old_slides.get(&slide.id) else { continue };
244        if std::sync::Arc::ptr_eq(before, slide) {
245            continue;
246        }
247        let shapes = &mut std::sync::Arc::make_mut(slide).shapes;
248        refit_list(Some(&before.shapes), shapes, 0);
249    }
250    // A child edited in Slide Master, layout, notes or handout view refits its group there too.
251    for master in new.masters.iter_mut() {
252        if let Some(before) = old.masters.iter().find(|m| m.id == master.id)
253            && !std::sync::Arc::ptr_eq(before, master)
254        {
255            refit_master(before, std::sync::Arc::make_mut(master));
256        }
257    }
258    for (before, after) in [(old.notes_master.as_ref(), new.notes_master.as_mut()), (old.handout_master.as_ref(), new.handout_master.as_mut())] {
259        if let (Some(before), Some(after)) = (before, after)
260            && !std::sync::Arc::ptr_eq(before, after)
261        {
262            refit_master(before, std::sync::Arc::make_mut(after));
263        }
264    }
265}

267fn refit_master(old: &crate::model::Master, new: &mut crate::model::Master) {
268    refit_list(Some(&old.shapes), &mut new.shapes, 0);
269    for layout in new.layouts.iter_mut() {
270        refit_list(old.layouts.iter().find(|l| l.id == layout.id).map(|l| &l.shapes), &mut layout.shapes, 0);
271    }
272}

274fn refit_list(old: Option<&Vec<Shape>>, list: &mut [Shape], depth: usize) {
275    // Most edits touch a slide without groups: nothing to refit, and no per-shape lookup. Nesting
276    // is bounded like the model's other walks.
277    if depth > 64 || !list.iter().any(|sh| matches!(sh.kind, ShapeKind::Group { .. })) {
278        return;
279    }
280    let by_id: std::collections::HashMap<ShapeId, &Shape> = old.map(|o| o.iter().map(|x| (x.id, x)).collect()).unwrap_or_default();
281    for sh in list.iter_mut() {
282        let before = by_id.get(&sh.id).copied();
283        let old_children = before.and_then(|b| match &b.kind {
284            ShapeKind::Group { children, .. } => Some(children),
285            _ => None,
286        });
287        let own_before = before.map(|b| {
288            (
289                b.xfrm,
290                match &b.kind {
291                    ShapeKind::Group { child, .. } => Some(*child),
292                    _ => None,
293                },
294            )
295        });
296        let smart = sh.diagram.is_some();
297        let ShapeKind::Group { children, child } = &mut sh.kind else { continue };
298        refit_list(old_children, children, depth + 1);
299        if smart {
300            // A SmartArt graphic's box is its own; its shapes are regenerated inside it.
301            continue;
302        }
303        let (Some(old_children), Some((old_x, old_child))) = (old_children, own_before) else { continue };
304        if old_x != sh.xfrm || old_child != Some(*child) || children.is_empty() {
305            continue;
306        }
307        let same = old_children.len() == children.len() && old_children.iter().zip(children.iter()).all(|(a, b)| a.id == b.id && a.xfrm == b.xfrm);
308        if same {
309            continue;
310        }
311        let Some(group) = sh.xfrm else { continue };
312        let xs: Vec<Xfrm> = children.iter().filter_map(|c| c.xfrm).collect();
313        let Some(u) = union_bounds(&xs) else { continue };
314        if ![u.x0, u.y0, u.x1, u.y1].iter().all(|v| v.is_finite()) {
315            continue;
316        }
317        let union = Xfrm::new(u.x0, u.y0, u.width(), u.height());
318        let placed = child_to_parent(&group, child, &union);
319        sh.xfrm = Some(Xfrm { x: placed.x, y: placed.y, w: placed.w, h: placed.h, rot: group.rot, flip_h: group.flip_h, flip_v: group.flip_v });
320        *child = union;
321    }
322}

324/// A child's xfrm in slide space, given its group.
325pub fn child_to_parent(group: &Xfrm, ch: &Xfrm, c: &Xfrm) -> Xfrm {
326    let sx = if ch.w.abs() > 1e-9 { group.w / ch.w } else { 1.0 };
327    let sy = if ch.h.abs() > 1e-9 { group.h / ch.h } else { 1.0 };
328    let a = crate::geom::group_child_affine(group, Rect::new(ch.x, ch.y, ch.x + ch.w, ch.y + ch.h));
329    let center = a * Point::new(c.x + c.w / 2.0, c.y + c.h / 2.0);
330    let (w, h) = (c.w * sx.abs(), c.h * sy.abs());
331    let mut rot = c.rot + group.rot;
332    let (mut fh, mut fv) = (c.flip_h, c.flip_v);
333    if group.flip_h {
334        fh = !fh;
335        rot = -c.rot + group.rot;
336    }
337    if group.flip_v {
338        fv = !fv;
339        rot = -rot + 2.0 * group.rot;
340    }
341    Xfrm { x: center.x - w / 2.0, y: center.y - h / 2.0, w, h, rot: ((rot % 360.0) + 360.0) % 360.0, flip_h: fh, flip_v: fv }
342}

344fn ungroup(s: &mut Session, p: &Value) -> Result<Value> {
345    let ids = targets(s, p)?;
346    s.edit(|doc, sel| {
347        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("arrange.ungroup", "no slide"))?;
348        let mut out = Vec::with_capacity(list.len());
349        let mut new_sel = vec![];
350        for sh in list.drain(..) {
351            if ids.contains(&sh.id)
352                && let (ShapeKind::Group { children, child }, Some(gx)) = (&sh.kind, sh.xfrm)
353            {
354                for c in children {
355                    let mut c = c.clone();
356                    if let Some(cx) = c.xfrm {
357                        c.xfrm = Some(child_to_parent(&gx, child, &cx));
358                    }
359                    // Nested groups scale too: their child space stays, their box is mapped.
360                    new_sel.push(c.id);
361                    out.push(c);
362                }
363                continue;
364            }
365            out.push(sh);
366        }
367        *list = out;
368        // A group inside a group: its children take its place in the outer group, mapped into the
369        // outer group's child space, where the inner group's own box is.
370        for id in &ids {
371            let Some(siblings) = crate::model::sibling_list_mut(list, *id) else { continue };
372            let Some(at) = siblings.iter().position(|x| x.id == *id) else { continue };
373            let freed: Vec<Shape> = match siblings.get(at) {
374                Some(Shape { kind: ShapeKind::Group { children, child }, xfrm: Some(gx), .. }) => children
375                    .iter()
376                    .map(|c| {
377                        let mut c = c.clone();
378                        if let Some(cx) = c.xfrm {
379                            c.xfrm = Some(child_to_parent(gx, child, &cx));
380                        }
381                        c
382                    })
383                    .collect(),
384                _ => continue,
385            };
386            let freed_ids: Vec<ShapeId> = freed.iter().map(|c| c.id).collect();
387            siblings.splice(at..=at, freed);
388            // Ungrouped along with an outer group, it was listed among that group's children (once per expanded ancestor).
389            match new_sel.iter().position(|x| x == id) {
390                Some(i) => {
391                    new_sel.splice(i..=i, freed_ids);
392                    new_sel.retain(|x| x != id);
393                }
394                None => new_sel.extend(freed_ids),
395            }
396        }
397        // A shape freed from a named group and then from a named group around it is listed once.
398        let mut listed = std::collections::HashSet::new();
399        new_sel.retain(|x| listed.insert(*x));
400        sel.shapes = new_sel.clone();
401        Ok(json!({"ids": new_sel}))
402    })
403}

405fn regroup(s: &mut Session, _p: &Value) -> Result<Value> {
406    // Regroup the most recently ungrouped objects (from the undo history).
407    let st = s.doc()?;
408    let last = st.history.undo.iter().rev().find(|e| e.label == "Ungroup").map(|e| e.selection.shapes.clone());
409    match last {
410        Some(prev_group) => {
411            // The ungroup's selection held the group id; its children are the current selection when nothing else changed.
412            let _ = prev_group;
413            let ids = s.doc()?.selection.shapes.clone();
414            if ids.len() >= 2 { s.execute("arrange.group", &json!({"ids": ids})) } else { Err(bad("arrange.regroup", "nothing to regroup")) }
415        }
416        None => Err(bad("arrange.regroup", "nothing to regroup")),
417    }
418}

420fn align(s: &mut Session, p: &Value) -> Result<Value> {
421    let edge = str_param(p, "edge").unwrap_or("left").to_string();
422    let ids = targets(s, p)?;
423    let st = s.doc()?;
424    let size = st.doc.slide_size;
425    let boxes: Vec<(ShapeId, Xfrm)> = ids.iter().filter_map(|id| st.shape(*id).map(|x| (*id, xfrm_of(&st.doc, &st.selection, x)))).collect();
426    let to_slide = match str_param(p, "to") {
427        Some("slide") => true,
428        Some(_) => false,
429        None => boxes.len() < 2,
430    };
431    let reference = if to_slide {
432        Rect::new(0.0, 0.0, size.width, size.height)
433    } else {
434        union_bounds(&boxes.iter().map(|b| b.1).collect::<Vec<_>>()).unwrap_or_default()
435    };
436    edit_geometry(s, p, "arrange.align", |sh| {
437        if let Some(x) = sh.xfrm.as_mut() {
438            let b = x.bounds();
439            let (dx, dy) = match edge.as_str() {
440                "left" => (reference.x0 - b.x0, 0.0),
441                "center" => (reference.center().x - b.center().x, 0.0),
442                "right" => (reference.x1 - b.x1, 0.0),
443                "top" => (0.0, reference.y0 - b.y0),
444                "middle" => (0.0, reference.center().y - b.center().y),
445                "bottom" => (0.0, reference.y1 - b.y1),
446                _ => (0.0, 0.0),
447            };
448            x.x += dx;
449            x.y += dy;
450        }
451        Ok(())
452    })
453}

455fn distribute(s: &mut Session, p: &Value) -> Result<Value> {
456    let horiz = !str_param(p, "dir").unwrap_or("horizontal").starts_with('v');
457    let ids = targets(s, p)?;
458    let st = s.doc()?;
459    let size = st.doc.slide_size;
460    let mut boxes: Vec<(ShapeId, Rect)> =
461        ids.iter().filter_map(|id| st.shape(*id).map(|x| (*id, xfrm_of(&st.doc, &st.selection, x).bounds()))).collect();
462    let to_slide = str_param(p, "to") == Some("slide") || boxes.len() < 3;
463    if boxes.is_empty() {
464        return ok();
465    }
466    boxes.sort_by(|a, b| if horiz { a.1.x0.total_cmp(&b.1.x0) } else { a.1.y0.total_cmp(&b.1.y0) });
467    let total: f64 = boxes.iter().map(|b| if horiz { b.1.width() } else { b.1.height() }).sum();
468    let (start, end) = if to_slide {
469        if horiz { (0.0, size.width) } else { (0.0, size.height) }
470    } else {
471        let lo = |f: fn(&Rect) -> f64| boxes.iter().map(|b| f(&b.1)).fold(f64::INFINITY, f64::min);
472        let hi = |f: fn(&Rect) -> f64| boxes.iter().map(|b| f(&b.1)).fold(f64::NEG_INFINITY, f64::max);
473        if horiz { (lo(|r| r.x0), hi(|r| r.x1)) } else { (lo(|r| r.y0), hi(|r| r.y1)) }
474    };
475    let n = boxes.len();
476    let gap = if n > 1 { ((end - start) - total) / (n - 1) as f64 } else { 0.0 };
477    let mut targets_pos: Vec<(ShapeId, f64)> = vec![];
478    let mut cur = start;
479    if n == 1 {
480        let b = boxes.first().map(|b| b.1).unwrap_or_default();
481        cur = start + ((end - start) - if horiz { b.width() } else { b.height() }) / 2.0;
482    }
483    for (id, b) in &boxes {
484        let d = cur - if horiz { b.x0 } else { b.y0 };
485        targets_pos.push((*id, d));
486        cur += (if horiz { b.width() } else { b.height() }) + gap;
487    }
488    edit_geometry(s, p, "arrange.distribute", |sh| {
489        if let (Some(x), Some((_, d))) = (sh.xfrm.as_mut(), targets_pos.iter().find(|(i, _)| *i == sh.id)) {
490            if horiz {
491                x.x += d;
492            } else {
493                x.y += d;
494            }
495        }
496        Ok(())
497    })
498}

500/// Guides are kept on the slide; a file can hold at most this many.
501const MAX_GUIDES: usize = 200;
502
503fn guide_limit(s: &Session, vertical: bool) -> Result<f64> {
504    let size = s.doc()?.doc.slide_size;
505    Ok(if vertical { size.width } else { size.height })
506}
507
508fn guide_add(s: &mut Session, p: &Value) -> Result<Value> {
509    if bool_or(p, "defaults", false) {
510        if !s.doc()?.doc.guides.is_empty() {
511            return Ok(json!({"added": 0}));
512        }
513        let size = s.doc()?.doc.slide_size;
514        return s.edit(|doc, _| {
515            doc.guides = vec![
516                crate::model::DrawGuide { vertical: true, pos: size.width / 2.0 },
517                crate::model::DrawGuide { vertical: false, pos: size.height / 2.0 },
518            ];
519            Ok(json!({"added": 2}))
520        });
521    }
522    let vertical = bool_or(p, "vertical", true);
523    let limit = guide_limit(s, vertical)?;
524    let pos = p.get("pos").and_then(Value::as_f64).filter(|v| v.is_finite()).unwrap_or(limit / 2.0).clamp(0.0, limit);
525    if s.doc()?.doc.guides.len() >= MAX_GUIDES {
526        return Err(bad("guide.add", "too many guides"));
527    }
528    s.edit(|doc, _| {
529        doc.guides.push(crate::model::DrawGuide { vertical, pos });
530        Ok(json!({"index": doc.guides.len() - 1}))
531    })
532}
533
534fn guide_index(s: &Session, p: &Value, cmd: &str) -> Result<usize> {
535    let i = usize_param(p, "index").ok_or_else(|| bad(cmd, "missing `index`"))?;
536    if i >= s.doc()?.doc.guides.len() {
537        return Err(bad(cmd, "no such guide"));
538    }
539    Ok(i)
540}

542fn guide_move(s: &mut Session, p: &Value) -> Result<Value> {
543    let i = guide_index(s, p, "guide.move")?;
544    let vertical = s.doc()?.doc.guides.get(i).is_some_and(|g| g.vertical);
545    let limit = guide_limit(s, vertical)?;
546    let pos = p.get("pos").and_then(Value::as_f64).filter(|v| v.is_finite()).ok_or_else(|| bad("guide.move", "missing `pos`"))?.clamp(0.0, limit);
547    s.edit(|doc, _| {
548        if let Some(g) = doc.guides.get_mut(i) {
549            g.pos = pos;
550        }
551        Ok(json!({"index": i, "pos": pos}))
552    })
553}
554
555fn guide_delete(s: &mut Session, p: &Value) -> Result<Value> {
556    let i = guide_index(s, p, "guide.delete")?;
557    s.edit(|doc, _| {
558        if i < doc.guides.len() {
559            doc.guides.remove(i);
560        }
561        ok()
562    })
563}

567}
