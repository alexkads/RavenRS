40/// The media shape a command acts on: `id` / `ids` / the selection, else the only media shape on
41/// the current slide.
42fn target(s: &Session, p: &Value, cmd: &str) -> Result<(ShapeId, MediaClip)> {
43    let st = s.doc()?;
44    let media_of = |id: ShapeId| st.shape(id).and_then(|sh| if let ShapeKind::Media(m) = &sh.kind { Some((id, m.clone())) } else { None });
45    let ids = targets(s, p)?;
46    if let Some(found) = ids.iter().find_map(|id| media_of(*id)) {
47        return Ok(found);
48    }
49    if !ids.is_empty() && (p.get("id").is_some() || p.get("ids").is_some()) {
50        return Err(bad(cmd, "not a media shape"));
51    }
52    let all: Vec<(ShapeId, MediaClip)> = st.shapes().iter().filter_map(|sh| media_of(sh.id)).collect();
53    match all.len() {
54        1 => Ok(all.into_iter().next().ok_or_else(|| bad(cmd, "no media"))?),
55        0 => Err(bad(cmd, "no audio or video on this slide")),
56        _ => Err(bad(cmd, "select an audio or video, or pass `id`")),
57    }
58}

60fn request(s: &mut Session, p: &Value, cmd: &str, action: &str, ms: Option<u64>) -> Result<Value> {
61    let (shape, _) = target(s, p, cmd)?;
62    s.ui_requests.push(UiRequest::Media { action: action.into(), shape, ms });
63    Ok(json!({"id": shape}))
64}
65
66fn ms_param(p: &Value, key: &str) -> Option<u64> {
67    p.get(key).and_then(Value::as_f64).filter(|v| v.is_finite() && *v >= 0.0).map(|v| v as u64)
68}
69
70fn play(s: &mut Session, p: &Value) -> Result<Value> {
71    request(s, p, "media.play", "play", ms_param(p, "from"))
72}
73fn pause(s: &mut Session, p: &Value) -> Result<Value> {
74    request(s, p, "media.pause", "pause", None)
75}
76fn toggle(s: &mut Session, p: &Value) -> Result<Value> {
77    request(s, p, "media.toggle", "toggle", None)
78}
79fn stop(s: &mut Session, p: &Value) -> Result<Value> {
80    request(s, p, "media.stop", "stop", None)
81}
82fn seek(s: &mut Session, p: &Value) -> Result<Value> {
83    let ms = ms_param(p, "ms").ok_or_else(|| bad("media.seek", "missing `ms`"))?;
84    request(s, p, "media.seek", "seek", Some(ms))
85}

87pub(crate) fn probe_json(info: &crate::media::MediaInfo) -> Value {
88    json!({
89        "container": info.container,
90        "durationMs": info.duration_ms,
91        "playable": info.playable(),
92        "audio": info.audio.as_ref().map(|a| json!({"codec": a.codec, "sampleRate": a.sample_rate, "channels": a.channels, "decodable": a.decodable})),
93        "video": info.video.as_ref().map(|v| json!({"codec": v.codec, "width": v.width, "height": v.height, "fps": (v.fps * 1000.0).round() / 1000.0, "frames": v.frames, "decodable": v.decodable})),
94    })
95}

97fn info(s: &mut Session, p: &Value) -> Result<Value> {
98    let (id, m) = target(s, p, "media.info")?;
99    let st = s.doc()?;
100    let item = st.doc.media(m.media).ok_or_else(|| EngineError::Other("the media data is missing".into()))?;
101    let probe = crate::media::probe(&item.data).map(|i| probe_json(&i)).unwrap_or_else(|e| json!({"error": e.to_string()}));
102    Ok(json!({
103        "id": id,
104        "name": item.name,
105        "contentType": item.content_type,
106        "bytes": item.data.len(),
107        "video": m.video,
108        "durationMs": m.duration_ms,
109        "width": m.width,
110        "height": m.height,
111        "poster": m.poster.is_some(),
112        "probe": probe,
113        "options": {
114            "autoplay": m.autoplay, "loop": m.loop_play, "rewind": m.rewind, "acrossSlides": m.play_across_slides,
115            "hide": m.hide_while_not_playing, "fullScreen": m.full_screen, "volume": m.volume,
116            "trimStart": m.trim_start_ms, "trimEnd": m.trim_end_ms, "fadeIn": m.fade_in_ms, "fadeOut": m.fade_out_ms,
117        },
118        "status": s.media_status.get(&id),
119    }))
120}

122/// A poster frame PNG for video bytes at `ms`.
123pub(crate) fn poster_png(bytes: &crate::media::Bytes, ms: u64) -> Option<Vec<u8>> {
124    match crate::media::poster_png(bytes, ms as f64 / 1000.0) {
125        Ok(png) => Some(png),
126        Err(e) => {
127            log::info!("no poster frame: {e}");
128            None
129        }
130    }
131}
132
133fn poster_frame(s: &mut Session, p: &Value) -> Result<Value> {
134    let (id, m) = target(s, p, "media.posterFrame")?;
135    if !m.video {
136        return Err(bad("media.posterFrame", "only videos have poster frames"));
137    }
138    let ms = ms_param(p, "ms").or_else(|| s.media_status.get(&id).map(|st: &MediaStatus| st.position_ms)).unwrap_or(0);
139    let data = s.doc()?.doc.media(m.media).map(|i| i.data.clone()).ok_or_else(|| EngineError::Other("the media data is missing".into()))?;
140    let png = poster_png(&data, ms).ok_or_else(|| EngineError::Other("no frame could be decoded there".into()))?;
141    let name = format!("poster{id}.png");
142    let poster = s.edit(|doc, _| doc.add_media(&name, "image/png", png).ok_or_else(|| bad("media.posterFrame", NO_ID_SPACE)))?;
143    edit_shapes(s, &json!({"id": id}), "media.posterFrame", |sh| {
144        if let ShapeKind::Media(m) = &mut sh.kind {
145            m.poster = Some(poster);
146        }
147        Ok(())
148    })?;
149    Ok(json!({"id": id, "ms": ms}))
150}

152/// Pause or play an animated GIF picture (`id` / `ids` / the selection, else the only one on the
153/// slide). The UI host keeps its clock: a paused GIF holds its frame and resumes from it.
154fn gif_play(s: &mut Session, p: &Value) -> Result<Value> {
155    let cmd = "media.gifPlay";
156    let st = s.doc()?;
157    let is_gif = |id: ShapeId| {
158        let Some(ShapeKind::Picture { fill }) = st.shape(id).map(|sh| &sh.kind) else { return false };
159        st.doc.media(fill.media).and_then(|m| crate::engine::render::gif_frame(&m.data, &fill.adjust, 0.0)).is_some()
160    };
161    let ids = targets(s, p)?;
162    let id = match ids.iter().copied().find(|id| is_gif(*id)) {
163        Some(id) => id,
164        None if !ids.is_empty() && (p.get("id").is_some() || p.get("ids").is_some()) => return Err(bad(cmd, "not an animated GIF")),
165        None => match st.shapes().iter().map(|sh| sh.id).filter(|id| is_gif(*id)).collect::<Vec<_>>().as_slice() {
166            [id] => *id,
167            [] => return Err(bad(cmd, "no animated GIF on this slide")),
168            _ => return Err(bad(cmd, "select an animated GIF, or pass `id`")),
169        },
170    };
171    let play = bool_param(p, "play").unwrap_or_else(|| s.gif_paused.contains(&id));
172    if play {
173        s.gif_paused.remove(&id);
174    } else {
175        s.gif_paused.insert(id);
176    }
177    Ok(json!({"id": id, "playing": play}))
178}

188    fn wav() -> Vec<u8> {
189        let rate = 8000u32;
190        let samples: Vec<i16> = (0..rate / 2).map(|i| ((i as f32 * 0.3).sin() * 3000.0) as i16).collect();
191        let data: Vec<u8> = samples.iter().flat_map(|s| s.to_le_bytes()).collect();
192        let mut b = b"RIFF".to_vec();
193        b.extend_from_slice(&(36 + data.len() as u32).to_le_bytes());
194        b.extend_from_slice(b"WAVEfmt ");
195        for v in [
196            16u32.to_le_bytes().to_vec(),
197            1u16.to_le_bytes().to_vec(),
198            1u16.to_le_bytes().to_vec(),
199            rate.to_le_bytes().to_vec(),
200            (rate * 2).to_le_bytes().to_vec(),
201            2u16.to_le_bytes().to_vec(),
202            16u16.to_le_bytes().to_vec(),
203        ] {
204            b.extend_from_slice(&v);
205        }
206        b.extend_from_slice(b"data");
207        b.extend_from_slice(&(data.len() as u32).to_le_bytes());
208        b.extend_from_slice(&data);
209        b
210    }

212    #[test]
213    fn insert_audio_probes_and_play_requests_reach_the_host() {
214        let mut s = Session::with_new();
215        let _ = s.execute("slide.new", &json!({"layout": "blank"}));
216        let r = s.execute("insert.audio", &json!({"name": "tone.wav", "data": base64_encode(&wav())})).expect("insert");
217        assert_eq!(r["durationMs"], 500);
218        assert_eq!(r["playable"], true);
219        let id = r["id"].as_u64().expect("id");
220        s.ui_requests.clear();
221        s.execute("media.play", &json!({"from": 100})).expect("play");
222        assert_eq!(s.ui_requests, vec![UiRequest::Media { action: "play".into(), shape: ShapeId(id as u32), ms: Some(100) }]);
223        s.execute("media.seek", &json!({"id": id, "ms": 250})).expect("seek");
224        assert!(s.execute("media.seek", &json!({"id": id})).is_err());
225        // The host reports state; media.info returns it with the probe.
226        s.media_status.insert(ShapeId(id as u32), MediaStatus { state: "playing".into(), position_ms: 250, duration_ms: 500, error: None });
227        let i = s.execute("media.info", &json!({})).expect("info");
228        assert_eq!(i["probe"]["audio"]["sampleRate"], 8000);
229        assert_eq!(i["probe"]["container"], "wav");
230        assert_eq!(i["status"]["positionMs"], 250);
231        assert_eq!(i["status"]["state"], "playing");
232        assert!(s.execute("media.posterFrame", &json!({})).is_err(), "audio has no poster frame");
233    }
