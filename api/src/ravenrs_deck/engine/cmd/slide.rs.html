80/// Resolve a layout by id, name or kind keyword.
81pub fn find_layout(doc: &Presentation, key: &Value, master: usize) -> Option<LayoutId> {
82    if let Some(n) = key.as_u64() {
83        let id = LayoutId(u32::try_from(n).ok()?);
84        if doc.layout(id).is_some() {
85            return Some(id);
86        }
87    }
88    let s = key.as_str()?;
89    let m = doc.masters.get(master).or(doc.masters.first())?;
90    let norm = |x: &str| x.chars().filter(|c| c.is_alphanumeric()).collect::<String>().to_ascii_lowercase();
91    let k = norm(s);
92    let kind = match k.as_str() {
93        "title" | "titleslide" => Some(LayoutType::Title),
94        "titleandcontent" | "content" | "obj" => Some(LayoutType::TitleAndContent),
95        "sectionheader" | "section" => Some(LayoutType::SectionHeader),
96        "twocontent" | "two" => Some(LayoutType::TwoContent),
97        "comparison" => Some(LayoutType::Comparison),
98        "titleonly" => Some(LayoutType::TitleOnly),
99        "blank" => Some(LayoutType::Blank),
100        "contentwithcaption" => Some(LayoutType::ContentWithCaption),
101        "picturewithcaption" | "picture" => Some(LayoutType::PictureWithCaption),
102        "titleandverticaltext" => Some(LayoutType::TitleAndVerticalText),
103        "verticaltitleandtext" => Some(LayoutType::VerticalTitleAndText),
104        _ => None,
105    };
106    m.layouts.iter().find(|l| norm(&l.name) == k).or_else(|| kind.and_then(|kd| m.layouts.iter().find(|l| l.kind == kd))).map(|l| l.id)
107}

113fn new(s: &mut Session, p: &Value) -> Result<Value> {
114    s.edit(|doc, sel| {
115        let master = current_master(doc, sel);
116        let layout = match p.get("layout") {
117            Some(k) => find_layout(doc, k, master).ok_or_else(|| bad("slide.new", format!("no layout {k}")))?,
118            None => {
119                // PowerPoint picks Title and Content after a title slide, else the current layout.
120                let cur = doc.slides.get(sel.slide).map(|x| x.layout);
121                let is_title = cur.and_then(|l| doc.layout(l)).is_some_and(|(_, l)| l.kind == LayoutType::Title);
122                match cur {
123                    Some(l) if !is_title => l,
124                    _ => defaults::layout_of_kind(doc, LayoutType::TitleAndContent).or(cur).unwrap_or_default(),
125                }
126            }
127        };
128        let mut slide = defaults::new_slide(doc, layout).ok_or_else(|| bad("slide.new", NO_ID_SPACE))?;
129        if let Some(t) = str_param(p, "title")
130            && let Some(sh) = slide.shapes.iter_mut().find(|x| x.ph_type().is_some_and(|k| k.is_title()))
131        {
132            sh.text = Some(TextBody::from_text(t));
133        }
134        if let Some(b) = str_param(p, "body")
135            && let Some(sh) = body_placeholder(&mut slide.shapes)
136        {
137            let mut body = TextBody::from_text(b);
138            // Leading tabs set the outline level, as in text.set.
139            for para in &mut body.paragraphs {
140                super::text::apply_leading_tabs(para);
141            }
142            sh.text = Some(body);
143        }
144        let at = usize_param(p, "at").unwrap_or(if doc.slides.is_empty() { 0 } else { sel.slide + 1 }).min(doc.slides.len());
145        let id = slide.id;
146        // Inherit the section of the slide before.
147        let section = at.checked_sub(1).and_then(|i| doc.slides.get(i)).and_then(|prev| doc.section_of(prev.id));
148        doc.slides.insert(at, Arc::new(slide));
149        if let Some(si) = section
150            && let Some(sec) = doc.sections.get_mut(si)
151        {
152            let pos = at
153                .checked_sub(1)
154                .and_then(|i| doc.slides.get(i))
155                .and_then(|prev| sec.slides.iter().position(|x| *x == prev.id))
156                .map(|k| k + 1)
157                .unwrap_or(sec.slides.len());
158            sec.slides.insert(pos.min(sec.slides.len()), id);
159        } else if let Some(first) = doc.sections.first_mut() {
160            first.slides.insert(0, id);
161        }
162        sel.slide = at;
163        sel.slides = vec![id];
164        sel.shapes.clear();
165        sel.text = None;
166        sel.target = Target::Slides;
167        Ok(json!({"index": at, "id": id}))
168    })
169}

171/// The placeholder a new slide's `body` goes into: the layout's main content placeholder.
172/// Prefers `idx="1"` (the content slot PowerPoint fills), then a full-size text placeholder over
173/// a `half`/`quarter` one, then a text placeholder over a picture/chart/table one; ties keep
174/// shape order, so a layout with a single body placeholder behaves as before.
175fn body_placeholder(shapes: &mut [crate::model::Shape]) -> Option<&mut crate::model::Shape> {
176    use crate::model::PhType;
177    let rank = |sh: &crate::model::Shape| {
178        let ph = sh.ph.as_ref()?;
179        if ph.kind.is_title() {
180            return None;
181        }
182        let text = matches!(ph.kind, PhType::Body | PhType::Obj | PhType::SubTitle);
183        let small = matches!(ph.size.as_deref(), Some("half" | "quarter"));
184        Some((ph.idx != 1, small, !text))
185    };
186    let best = shapes.iter().enumerate().filter_map(|(i, sh)| rank(sh).map(|r| (r, i))).min()?.1;
187    shapes.get_mut(best)
188}

199fn duplicate(s: &mut Session, p: &Value) -> Result<Value> {
200    let indices: Vec<usize> = match opt_usize_param(p, "index", "slide.duplicate")? {
201        Some(i) => vec![i],
202        None => {
203            let st = s.doc()?;
204            let mut v: Vec<usize> = st.selection.slides.iter().filter_map(|id| st.doc.slide_index(*id)).collect();
205            if v.is_empty() {
206                v.push(st.selection.slide);
207            }
208            v.sort_unstable();
209            v
210        }
211    };
212    s.edit(|doc, sel| {
213        let last = *indices.last().unwrap_or(&0);
214        let mut new_ids = vec![];
215        for (k, i) in indices.iter().enumerate() {
216            let Some(src) = doc.slides.get(*i).map(|x| x.as_ref().clone()) else { continue };
217            let mut copy = src;
218            copy.id = SlideId(doc.alloc_id().ok_or_else(|| bad("slide.duplicate", NO_ID_SPACE))?);
219            // Animations follow the new shape ids, triggers included.
220            let map = super::edit::reid("slide.duplicate", doc, &mut copy.shapes)?;
221            super::edit::remap_animations(&mut copy.animations, &map);
222            let section = doc.section_of(src_id(doc, *i));
223            let at = (last + 1 + k).min(doc.slides.len());
224            new_ids.push(copy.id);
225            let nid = copy.id;
226            doc.slides.insert(at, Arc::new(copy));
227            if let Some(si) = section
228                && let Some(sec) = doc.sections.get_mut(si)
229            {
230                sec.slides.push(nid);
231                // Keep the section list in document order so the copy sits next to its source.
232                let order: Vec<SlideId> = doc.slides.iter().map(|x| x.id).collect();
233                sec.slides.sort_by_key(|x| order.iter().position(|o| o == x).unwrap_or(usize::MAX));
234            }
235        }
236        sel.slide = (last + new_ids.len()).min(doc.slides.len().saturating_sub(1));
237        sel.slides = new_ids.clone();
238        sel.shapes.clear();
239        sel.text = None;
240        Ok(json!({"ids": new_ids}))
241    })
242}

248fn delete(s: &mut Session, p: &Value) -> Result<Value> {
249    let ids: Vec<SlideId> = {
250        let st = s.doc()?;
251        if let Some(a) = p.get("ids").and_then(Value::as_array) {
252            a.iter().filter_map(Value::as_u64).filter_map(|v| u32::try_from(v).ok()).map(SlideId).collect()
253        } else if let Some(i) = opt_usize_param(p, "index", "slide.delete")? {
254            st.doc.slides.get(i).map(|x| vec![x.id]).unwrap_or_default()
255        } else if !st.selection.slides.is_empty() {
256            st.selection.slides.clone()
257        } else {
258            st.current_slide().map(|x| vec![x.id]).unwrap_or_default()
259        }
260    };
261    if ids.is_empty() {
262        return Err(bad("slide.delete", "no such slide"));
263    }
264    s.edit(|doc, sel| {
265        let first = ids.iter().filter_map(|id| doc.slide_index(*id)).min().unwrap_or(0);
266        doc.slides.retain(|x| !ids.contains(&x.id));
267        for sec in &mut doc.sections {
268            sec.slides.retain(|x| !ids.contains(x));
269        }
270        for show in &mut doc.custom_shows {
271            show.slides.retain(|x| !ids.contains(x));
272        }
273        sel.slide = first.min(doc.slides.len().saturating_sub(1));
274        sel.slides = doc.slides.get(sel.slide).map(|x| vec![x.id]).unwrap_or_default();
275        sel.shapes.clear();
276        sel.text = None;
277        Ok(json!({"deleted": ids.len(), "remaining": doc.slides.len()}))
278    })
279}

281/// Moves one slide (`from`, else the current slide) or several (`ids`) to `to`, in one undo step.
282/// `to` counts in the deck without the moved slides; they land there as one run, in the order
283/// they had, join the section of the slide before them, and stay selected.
284fn move_slide(s: &mut Session, p: &Value) -> Result<Value> {
285    let ids = p.get("ids").filter(|v| !v.is_null());
286    let from = match ids {
287        None => opt_usize_param(p, "from", "slide.move")?.unwrap_or(s.doc()?.selection.slide),
288        Some(_) => 0,
289    };
290    let to = usize_param(p, "to").ok_or_else(|| bad("slide.move", "missing `to`"))?;
291    s.edit(|doc, sel| {
292        let moving: HashSet<SlideId> = match ids {
293            Some(v) => slide_ids_param(v, doc, "slide.move")?,
294            None => {
295                let id = doc.slides.get(from).map(|x| x.id).ok_or_else(|| bad("slide.move", format!("no slide {from}")))?;
296                HashSet::from([id])
297            }
298        };
299        let (block, mut rest): (Vec<_>, Vec<_>) = std::mem::take(&mut doc.slides).into_iter().partition(|x| moving.contains(&x.id));
300        let to = to.min(rest.len());
301        let tail = rest.split_off(to);
302        let moved: Vec<SlideId> = block.iter().map(|x| x.id).collect();
303        rest.extend(block);
304        rest.extend(tail);
305        doc.slides = rest;
306        reassign_sections(doc, &moved, to);
307        sel.slide = to;
308        sel.slides = moved;
309        sel.shapes.clear();
310        sel.text = None;
311        Ok(json!({"index": to, "moved": moving.len()}))
312    })
313}

315/// The slides an `ids` param names: a non-empty list of ids of slides in `doc` (repeats fold).
316/// The first id that is not a slide of `doc` is an error, so the set never outgrows the deck.
317fn slide_ids_param(v: &Value, doc: &Presentation, cmd: &str) -> Result<HashSet<SlideId>> {
318    let a = v.as_array().ok_or_else(|| bad(cmd, "`ids` must be a list of slide ids"))?;
319    let known: HashSet<SlideId> = doc.slides.iter().map(|x| x.id).collect();
320    let mut out = HashSet::new();
321    for x in a {
322        let id = x.as_u64().and_then(|n| u32::try_from(n).ok()).map(SlideId).filter(|id| known.contains(id));
323        out.insert(id.ok_or_else(|| bad(cmd, format!("no slide with id {x}")))?);
324    }
325    if out.is_empty() {
326        return Err(bad(cmd, "`ids` is empty"));
327    }
328    Ok(out)
329}

331/// Keeps sections in step after `moved` landed as a run at `at`: the run joins the section of
332/// the slide just before it (the first section when it opens the deck), in document order.
333fn reassign_sections(doc: &mut Presentation, moved: &[SlideId], at: usize) {
334    if doc.sections.is_empty() {
335        return;
336    }
337    let moving: HashSet<SlideId> = moved.iter().copied().collect();
338    for sec in &mut doc.sections {
339        sec.slides.retain(|x| !moving.contains(x));
340    }
341    let neighbour = at.checked_sub(1).and_then(|i| doc.slides.get(i)).map(|x| x.id);
342    let si = neighbour.and_then(|n| doc.section_of(n)).unwrap_or(0);
343    let order: HashMap<SlideId, usize> = doc.slides.iter().enumerate().map(|(i, x)| (x.id, i)).collect();
344    if let Some(sec) = doc.sections.get_mut(si) {
345        sec.slides.extend_from_slice(moved);
346        sec.slides.sort_by_key(|x| order.get(x).copied().unwrap_or(usize::MAX));
347    }
348}

350fn go_to(s: &mut Session, i: usize) -> Result<Value> {
351    s.select(|doc, sel| {
352        let n = doc.slides.len();
353        sel.slide = i.min(n.saturating_sub(1));
354        sel.slides = doc.slides.get(sel.slide).map(|x| vec![x.id]).unwrap_or_default();
355        sel.shapes.clear();
356        sel.text = None;
357        sel.cells = None;
358        sel.range_anchor = None;
359        sel.target = Target::Slides;
360    })?;
361    Ok(json!({"index": s.doc()?.selection.slide}))
362}

364fn go(s: &mut Session, p: &Value) -> Result<Value> {
365    let i = match (usize_param(p, "index"), p.get("id").and_then(Value::as_u64)) {
366        (Some(i), _) => i,
367        (None, Some(id)) => s
368            .doc()?
369            .doc
370            .slide_index(SlideId(u32::try_from(id).map_err(|_| bad("slide.go", "no such slide"))?))
371            .ok_or_else(|| bad("slide.go", "no such slide"))?,
372        _ => return Err(bad("slide.go", "missing `index`")),
373    };
374    if i >= s.doc()?.doc.slides.len() {
375        return Err(bad("slide.go", format!("no slide {i}")));
376    }
377    go_to(s, i)
378}
379fn next(s: &mut Session, _p: &Value) -> Result<Value> {
380    let i = s.doc()?.selection.slide + 1;
381    go_to(s, i)
382}
383fn prev(s: &mut Session, _p: &Value) -> Result<Value> {
384    let i = s.doc()?.selection.slide.saturating_sub(1);
385    go_to(s, i)
386}
387fn first(s: &mut Session, _p: &Value) -> Result<Value> {
388    go_to(s, 0)
389}
390fn last(s: &mut Session, _p: &Value) -> Result<Value> {
391    let n = s.doc()?.doc.slides.len();
392    go_to(s, n.saturating_sub(1))
393}

395fn select_slides(s: &mut Session, p: &Value) -> Result<Value> {
396    let add = bool_or(p, "add", false);
397    let ids: Vec<SlideId> = {
398        let st = s.doc()?;
399        let mut v: Vec<SlideId> = p
400            .get("ids")
401            .and_then(Value::as_array)
402            .map(|a| a.iter().filter_map(Value::as_u64).filter_map(|x| u32::try_from(x).ok()).map(SlideId).collect())
403            .unwrap_or_default();
404        if let Some(a) = p.get("indices").and_then(Value::as_array) {
405            v.extend(a.iter().filter_map(Value::as_u64).filter_map(|i| st.doc.slides.get(i as usize)).map(|x| x.id));
406        }
407        v
408    };
409    s.select(|doc, sel| {
410        if add {
411            let mut have: std::collections::HashSet<_> = sel.slides.iter().copied().collect();
412            for i in &ids {
413                if have.insert(*i) {
414                    sel.slides.push(*i);
415                }
416            }
417        } else {
418            sel.slides = ids.clone();
419        }
420        if let Some(last) = ids.last().and_then(|i| doc.slide_index(*i)) {
421            sel.slide = last;
422        }
423        sel.shapes.clear();
424        sel.text = None;
425        sel.range_anchor = None;
426    })?;
427    Ok(json!({"selected": s.doc()?.selection.slides}))
428}

430/// Shift-click in the slide list: select every slide from `from` to `to`. Without `from` the range
431/// starts at the anchor of the previous Shift-click (so a second Shift-click extends from the same
432/// slide, as in PowerPoint), else at the current slide.
433/// With `add` the slides already selected stay selected (Ctrl+Shift-click).
434fn select_range(s: &mut Session, p: &Value) -> Result<Value> {
435    let to = usize_param(p, "to").ok_or_else(|| bad("slide.selectRange", "missing `to`"))?;
436    let given = opt_usize_param(p, "from", "slide.selectRange")?;
437    let add = bool_or(p, "add", false);
438    let (from, count) = {
439        let st = s.doc()?;
440        let sel = &st.selection;
441        let current = st.doc.slides.get(sel.slide).map(|x| x.id);
442        let anchor = sel.range_anchor.filter(|(_, end)| Some(*end) == current).and_then(|(a, _)| st.doc.slide_index(a));
443        (given.or(anchor).unwrap_or(sel.slide), st.doc.slides.len())
444    };
445    if to >= count || from >= count {
446        return Err(bad("slide.selectRange", "no such slide"));
447    }
448    let (a, b) = (from.min(to), from.max(to));
449    s.select(|doc, sel| {
450        let range: Vec<SlideId> = doc.slides.iter().skip(a).take(b - a + 1).map(|x| x.id).collect();
451        if add {
452            for id in range {
453                if !sel.slides.contains(&id) {
454                    sel.slides.push(id);
455                }
456            }
457        } else {
458            sel.slides = range;
459        }
460        sel.slide = to;
461        sel.range_anchor = doc.slides.get(from).zip(doc.slides.get(to)).map(|(a, b)| (a.id, b.id));
462        sel.shapes.clear();
463        sel.text = None;
464        sel.cells = None;
465        sel.target = Target::Slides;
466    })?;
467    Ok(json!({"selected": s.doc()?.selection.slides}))
468}

470fn hide(s: &mut Session, p: &Value) -> Result<Value> {
471    let targets: Vec<usize> = match opt_usize_param(p, "index", "slide.hide")? {
472        Some(i) => vec![i],
473        None => {
474            let st = s.doc()?;
475            let v: Vec<usize> = st.selection.slides.iter().filter_map(|id| st.doc.slide_index(*id)).collect();
476            if v.is_empty() { vec![st.selection.slide] } else { v }
477        }
478    };
479    s.edit(|doc, _| {

481        let value = bool_param(p, "hidden").unwrap_or(!all_hidden);
482        for i in &targets {
483            if let Some(sl) = doc.slides.get_mut(*i) {
484                Arc::make_mut(sl).hidden = value;
485            }
486        }
487        Ok(json!({"hidden": value}))
488    })

491/// Re-map a slide's placeholders onto a new layout (content kept, matched by type/index).
492pub fn apply_layout(doc: &mut Presentation, index: usize, layout: LayoutId) -> Result<()> {
493    let lay_shapes: Vec<crate::model::Shape> =
494        doc.layout(layout).map(|(_, l)| l.shapes.clone()).ok_or_else(|| bad("slide.layout", "no such layout"))?;
495    let mut fresh = vec![];
496    for ls in lay_shapes.iter().filter(|x| x.ph.as_ref().is_some_and(|p| !p.kind.is_footer_kind())) {
497        let id = crate::model::ShapeId(doc.alloc_id().ok_or_else(|| bad("slide.layout", NO_ID_SPACE))?);
498        fresh.push((id, ls.clone()));
499    }
500    let slide = doc.slides.get_mut(index).ok_or_else(|| bad("slide.layout", "no slide"))?;
501    let slide = Arc::make_mut(slide);
502    let mut old: Vec<crate::model::Shape> = std::mem::take(&mut slide.shapes);
503    let mut out = vec![];
504    for (id, ls) in fresh {
505        let Some(lph) = ls.ph.clone() else { continue };
506        // Find matching old placeholder: same idx, else same title-ness/compatible type.
507        let k = old
508            .iter()
509            .position(|o| o.ph.as_ref().is_some_and(|p| p.kind == lph.kind && (p.idx == lph.idx || lph.kind.is_title())))
510            .or_else(|| old.iter().position(|o| o.ph.as_ref().is_some_and(|p| p.kind.is_title() == lph.kind.is_title() && !p.kind.is_footer_kind())));
511        match k {
512            Some(k) => {
513                let mut o = old.remove(k);
514                if let Some(p) = o.ph.as_mut() {
515                    p.kind = lph.kind;
516                    p.idx = lph.idx;
517                }
518                o.xfrm = None;
519                out.push(o);
520            }
521            None => out.push(crate::model::Shape { id, name: ls.name.clone(), ph: Some(lph), text: Some(TextBody::default()), ..Default::default() }),
522        }
523    }
524    // Leftover placeholders with content stay as they were (positioned); empty ones go.
525    for o in old {
526        let empty_ph = o.ph.is_some() && o.text.as_ref().is_none_or(|t| t.is_empty()) && matches!(o.kind, crate::model::ShapeKind::Shape);
527        if !empty_ph {
528            out.push(o);
529        }
530    }
531    slide.shapes = out;
532    slide.layout = layout;
533    Ok(())
534}

536fn set_layout(s: &mut Session, p: &Value) -> Result<Value> {
537    let key = p.get("layout").cloned().ok_or_else(|| bad("slide.layout", "missing `layout`"))?;
538    let targets: Vec<usize> = match opt_usize_param(p, "index", "slide.layout")? {
539        Some(i) => vec![i],
540        None => {
541            let st = s.doc()?;
542            let v: Vec<usize> = st.selection.slides.iter().filter_map(|id| st.doc.slide_index(*id)).collect();
543            if v.is_empty() { vec![st.selection.slide] } else { v }
544        }
545    };
546    s.edit(|doc, sel| {
547        let master = current_master(doc, sel);
548        let layout = find_layout(doc, &key, master).ok_or_else(|| bad("slide.layout", format!("no layout {key}")))?;
549        for i in &targets {
550            apply_layout(doc, *i, layout)?;
551        }
552        sel.shapes.clear();
553        sel.text = None;
554        Ok(json!({"layout": layout}))
555    })
556}

558fn reset(s: &mut Session, p: &Value) -> Result<Value> {
559    let i = index_of(s, p)?;
560    s.edit(|doc, sel| {
561        let layout = doc.slides.get(i).map(|x| x.layout).unwrap_or_default();
562        apply_layout(doc, i, layout)?;
563        if let Some(sl) = doc.slides.get_mut(i) {
564            let sl = Arc::make_mut(sl);
565            sl.background = None;
566            for sh in &mut sl.shapes {
567                if sh.ph.is_some() {
568                    sh.xfrm = None;
569                    sh.fill = None;
570                    sh.line = None;
571                    sh.effects = None;
572                    sh.geom = Default::default();
573                    if let Some(t) = sh.text.as_mut() {
574                        t.body = Default::default();
575                        t.list_style = Default::default();
576                        for para in &mut t.paragraphs {
577                            para.props = Default::default();
578                            for r in &mut para.runs {
579                                r.props = Default::default();
580                            }
581                        }
582                    }
583                }
584            }
585        }
586        sel.shapes.clear();
587        Ok(())
588    })?;
589    ok()
590}

626fn section_add(s: &mut Session, p: &Value) -> Result<Value> {
627    let at = opt_usize_param(p, "at", "section.add")?.unwrap_or(s.doc()?.selection.slide);
628    let name = str_param(p, "name").unwrap_or("Untitled Section").to_string();
629    s.edit(|doc, _| {
630        let at = at.min(doc.slides.len().saturating_sub(1));
631        let ids: Vec<SlideId> = doc.slides.iter().map(|x| x.id).collect();
632        if doc.sections.is_empty() {
633            // The slides before `at` go into a "Default Section".
634            if at > 0 {
635                doc.sections.push(Section { name: "Default Section".into(), slides: ids.get(..at).map(<[_]>::to_vec).unwrap_or_default() });
636            }
637            doc.sections.push(Section { name, slides: ids.get(at..).map(<[_]>::to_vec).unwrap_or_default() });
638        } else {
639            let Some(start) = ids.get(at).copied() else { return Err(bad("section.add", "no slide")) };
640            let si = doc.section_of(start).unwrap_or(0);
641            let sec = doc.sections.get_mut(si).ok_or_else(|| bad("section.add", "no section"))?;
642            let k = sec.slides.iter().position(|x| *x == start).unwrap_or(sec.slides.len());
643            let tail = sec.slides.split_off(k);
644            doc.sections.insert(si + 1, Section { name, slides: tail });
645            doc.sections.retain(|x| !x.slides.is_empty() || x.name != "Default Section");
646        }
647        Ok(json!({"sections": doc.sections.len()}))
648    })
649}

662fn section_remove(s: &mut Session, p: &Value) -> Result<Value> {
663    let i = usize_param(p, "index").ok_or_else(|| bad("section.remove", "missing `index`"))?;
664    let with_slides = bool_or(p, "slides", false);
665    s.edit(|doc, sel| {
666        if i >= doc.sections.len() {
667            return Err(bad("section.remove", "no such section"));
668        }
669        let sec = doc.sections.remove(i);
670        if with_slides {
671            doc.slides.retain(|x| !sec.slides.contains(&x.id));
672            sel.slide = sel.slide.min(doc.slides.len().saturating_sub(1));
673        } else if let Some(prev) = i.checked_sub(1).and_then(|k| doc.sections.get_mut(k)) {
674            prev.slides.extend(sec.slides);
675        } else if let Some(next) = doc.sections.get_mut(0) {
676            let mut v = sec.slides;
677            v.append(&mut next.slides);
678            next.slides = v;
679        }
680        Ok(())
681    })?;
682    ok()
683}

693fn section_move(s: &mut Session, p: &Value) -> Result<Value> {
694    let i = usize_param(p, "index").ok_or_else(|| bad("section.move", "missing `index`"))?;
695    let to = usize_param(p, "to").ok_or_else(|| bad("section.move", "missing `to`"))?;
696    s.edit(|doc, _| {
697        if i >= doc.sections.len() {
698            return Err(bad("section.move", "no such section"));
699        }
700        let sec = doc.sections.remove(i);
701        let to = to.min(doc.sections.len());
702        doc.sections.insert(to, sec);
703        // Reorder slides to follow sections.
704        let order: Vec<SlideId> = doc.sections.iter().flat_map(|x| x.slides.iter().copied()).collect();
705        doc.slides.sort_by_key(|x| order.iter().position(|o| *o == x.id).unwrap_or(usize::MAX));
706        Ok(())
707    })?;
708    ok()
709}

711fn from_outline(s: &mut Session, p: &Value) -> Result<Value> {
712    let text = str_param(p, "text").unwrap_or("").to_string();
713    let markdown = bool_param(p, "markdown").unwrap_or(false);
714    s.edit(|doc, sel| {
715        let before = doc.slides.len();
716        let n = if markdown { crate::format::markdown_to_slides(doc, &text) } else { crate::format::outline_to_slides(doc, &text) };
717        // both append; move them after the current slide.
718        let at = (sel.slide + 1).min(before);
719        let new: Vec<_> = doc.slides.drain(before..).collect();
720        for (k, sl) in new.into_iter().enumerate() {
721            doc.slides.insert(at + k, sl);
722        }
723        // Join the section of the slide before, as slide.new does.
724        let ids: Vec<SlideId> = doc.slides.iter().skip(at).take(n).map(|x| x.id).collect();
725        let prev = at.checked_sub(1).and_then(|i| doc.slides.get(i)).map(|x| x.id);
726        let section = prev.and_then(|p| doc.section_of(p));
727        if let Some(sec) = section.and_then(|si| doc.sections.get_mut(si)) {
728            let pos = prev.and_then(|p| sec.slides.iter().position(|x| *x == p)).map(|k| k + 1).unwrap_or(sec.slides.len());
729            let pos = pos.min(sec.slides.len());
730            sec.slides.splice(pos..pos, ids);
731        } else if let Some(first) = doc.sections.first_mut() {
732            first.slides.splice(0..0, ids);
733        }
734        Ok(json!({"slides": n}))
735    })
736}

947        let st = s.doc().expect("doc");

995    #[test]
996    fn a_second_range_extends_from_the_original_anchor() {
997        let mut s = deck_of(8);
998        s.execute("slide.go", &json!({"index": 2})).expect("go");
999        s.execute("slide.selectRange", &json!({"to": 5})).expect("range");
1000        assert_eq!(selected_indices(&s), vec![2, 3, 4, 5]);
1001        // Shift-click further on: still anchored at 2, not at 5.
1002        s.execute("slide.selectRange", &json!({"to": 7})).expect("range");
1003        assert_eq!(selected_indices(&s), vec![2, 3, 4, 5, 6, 7]);
1004        // Shift-click back across the anchor.
1005        s.execute("slide.selectRange", &json!({"to": 0})).expect("range");
1006        assert_eq!(selected_indices(&s), vec![0, 1, 2]);
1007        // A plain click on another slide starts a new anchor.
1008        s.execute("slide.go", &json!({"index": 4})).expect("go");
1009        s.execute("slide.selectRange", &json!({"to": 6})).expect("range");
1010        assert_eq!(selected_indices(&s), vec![4, 5, 6]);
1011    }
