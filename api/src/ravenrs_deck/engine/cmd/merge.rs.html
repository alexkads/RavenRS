33/// Why Merge Shapes is unavailable while a shape inside a group is selected.
34pub(crate) const IN_GROUP: &str = "a shape in a group can't be merged; ungroup it first";

36/// Two or more shapes selected, none of them inside a group: the result replaces its sources in
37/// the slide's top-level list, so a member would be drawn outside its group's scale and offset.
38fn can_merge(s: &Session) -> std::result::Result<(), String> {
39    has_two(s)?;
40    let in_group =
41        s.active().is_some_and(|st| st.selection.shapes.iter().any(|id| st.shape(*id).is_some() && !st.shapes().iter().any(|t| t.id == *id)));
42    if in_group { Err(IN_GROUP.into()) } else { Ok(()) }
43}

45/// The filled outline of `s` in slide points (curves flattened). A text box with no fill and no
46/// outline is its letters, as in PowerPoint (merging text with a picture fills the letters).
47fn area_of(doc: &crate::model::Presentation, sel: &crate::engine::Selection, s: &Shape) -> Area {
48    if let Some((letters, _)) = letters_of(doc, sel, s) {
49        return letters;
50    }
51    let x = xfrm_of(doc, sel, s);
52    let geo = crate::render::shape_geometry(s, x.w, x.h);
53    let a = x.affine();
54    let mut out: Area = vec![];
55    for sp in geo.paths.iter().filter(|p| p.fill != crate::geom::preset::FillMode::None) {
56        flatten(&sp.path, a, &mut out);
57    }
58    out
59}

61/// The letters of a text box with no fill and no outline, in slide points, and their colour.
62fn letters_of(doc: &crate::model::Presentation, sel: &crate::engine::Selection, s: &Shape) -> Option<(Area, Option<crate::color::Rgba>)> {
63    use crate::model::resolve;
64    use crate::model::style::Fill;
65    if !matches!(s.kind, ShapeKind::Shape) || s.text.as_ref().is_none_or(|t| t.is_empty()) {
66        return None;
67    }
68    let ctx = super::text::ctx_for(doc, sel)?;
69    let none = |f: Option<&Fill>| f.is_none_or(Fill::is_none);
70    if !none(resolve::fill(&ctx, s).0.as_ref()) || !none(resolve::line(&ctx, s).0.fill.as_ref()) {
71        return None;
72    }
73    let (path, colour) = crate::render::text_outline(&ctx, s, xfrm_of(doc, sel, s).affine());
74    let mut out: Area = vec![];
75    flatten(&path, crate::geom::Affine::IDENTITY, &mut out);
76    (!out.is_empty()).then_some((out, colour))
77}

79/// Append the closed polygons of `path` (mapped by `a`) to `out`.
80fn flatten(path: &crate::geom::BezPath, a: crate::geom::Affine, out: &mut Area) {
81    let mut cur: Vec<[f64; 2]> = vec![];
82    let flush = |cur: &mut Vec<[f64; 2]>, out: &mut Area| {
83        if cur.len() > 2 {
84            out.push(std::mem::take(cur));
85        } else {
86            cur.clear();
87        }
88    };
89    crate::geom::preset_flatten(path, &mut |el| match el {
90        PathEl::MoveTo(p) => {
91            flush(&mut cur, out);
92            let q = a * p;
93            cur.push([q.x, q.y]);
94        }
95        PathEl::LineTo(p) => {
96            let q = a * p;
97            cur.push([q.x, q.y]);
98        }
99        PathEl::ClosePath => flush(&mut cur, out),
100        _ => {}
101    });
102    flush(&mut cur, out);
103}

105/// Keep a picture where it was on the slide when its box becomes `to`: crop it to the new box.
106/// Only for upright, unflipped pictures stretched over their box.
107fn recrop(fill: &mut crate::model::style::PictureFill, from: &Xfrm, to: &Xfrm) {
108    use crate::model::style::PictureMode;
109    if from.rot != 0.0 || from.flip_h || from.flip_v || !matches!(fill.mode, PictureMode::Stretch { fill_rect } if fill_rect == [0.0; 4]) {
110        return;
111    }
112    let [l, t, r, b] = fill.crop;
113    let (fw, fh) = (1.0 - l - r, 1.0 - t - b);
114    if !(fw > 1e-9 && fh > 1e-9 && from.w > 0.0 && from.h > 0.0) {
115        return;
116    }
117    // The whole image in slide points.
118    let (iw, ih) = (from.w / fw, from.h / fh);
119    let (ix, iy) = (from.x - l * iw, from.y - t * ih);
120    let c = [(to.x - ix) / iw, (to.y - iy) / ih, (ix + iw - to.x - to.w) / iw, (iy + ih - to.y - to.h) / ih];
121    if c.iter().all(|v| v.is_finite()) {
122        fill.crop = c;
123    }
124}

126fn overlay(a: &Area, b: &Area, rule: OverlayRule) -> Area {
127    if a.is_empty() && matches!(rule, OverlayRule::Union | OverlayRule::Xor) {
128        return b.clone();
129    }
130    if b.is_empty() {
131        return if matches!(rule, OverlayRule::Intersect) { vec![] } else { a.clone() };
132    }
133    a.overlay(b, rule, FillRule::NonZero).into_iter().flatten().filter(|c| c.len() > 2).collect()
134}

136fn area_size(a: &Area) -> f64 {
137    a.iter()
138        .map(|c| {
139            let n = c.len();
140            (0..n).map(|i| c[i][0] * c[(i + 1) % n][1] - c[(i + 1) % n][0] * c[i][1]).sum::<f64>() / 2.0
141        })
142        .sum::<f64>()
143        .abs()
144}

146/// The merged areas for `op` (one per resulting shape).
147pub fn merge_areas(op: &str, areas: &[Area]) -> Vec<Area> {
148    let Some((first, rest)) = areas.split_first() else { return vec![] };
149    let fold = |rule| rest.iter().fold(first.clone(), |acc, b| overlay(&acc, b, rule));
150    let out = match op {
151        "union" => vec![fold(OverlayRule::Union)],
152        "combine" => vec![fold(OverlayRule::Xor)],
153        "intersect" => vec![fold(OverlayRule::Intersect)],
154        "subtract" => {
155            let others = rest.iter().fold(Area::new(), |acc, b| overlay(&acc, b, OverlayRule::Union));
156            vec![overlay(first, &others, OverlayRule::Difference)]
157        }
158        _ => {
159            // Fragment: every region of the arrangement becomes its own shape.
160            let mut pieces: Vec<Area> = vec![first.clone()];
161            let mut covered = first.clone();
162            for b in rest {
163                let mut next = vec![];
164                for p in &pieces {
165                    next.push(overlay(p, b, OverlayRule::Intersect));
166                    next.push(overlay(p, b, OverlayRule::Difference));
167                }
168                next.push(overlay(b, &covered, OverlayRule::Difference));
169                covered = overlay(&covered, b, OverlayRule::Union);
170                pieces = next;
171            }
172            // Split pieces into connected outer contours with their holes is not needed for
173            // rendering; keep each piece as one shape.
174            pieces
175        }
176    };
177    out.into_iter().filter(|a| area_size(a) > 0.01).collect()
178}

180/// A custom-geometry box for `area`: its bounds and path data relative to them.
181fn area_geom(area: &Area) -> Option<(Xfrm, Geom)> {
182    let mut b: Option<Rect> = None;
183    for p in area.iter().flatten() {
184        let r = Rect::new(p[0], p[1], p[0], p[1]);
185        b = Some(b.map_or(r, |x| x.union(r)));
186    }
187    let b = b?;
188    let (w, h) = (b.width().max(0.01), b.height().max(0.01));
189    let f = |v: f64| {
190        let s = format!("{v:.2}");
191        s.trim_end_matches('0').trim_end_matches('.').to_string()
192    };
193    let mut d = String::new();
194    for c in area {
195        for (i, p) in c.iter().enumerate() {
196            d += &format!("{}{} {} ", if i == 0 { "M " } else { "L " }, f(p[0] - b.x0), f(p[1] - b.y0));
197        }
198        d += "Z ";
199    }
200    let geom = Geom::Custom { paths: vec![CustomPath { w, h, d: d.trim_end().to_string(), ..Default::default() }] };
201    Some((Xfrm::new(b.x0, b.y0, w, h), geom))
202}

204fn merge(s: &mut Session, p: &Value) -> Result<Value> {
205    let op = str_param(p, "op").unwrap_or("union").to_ascii_lowercase();
206    if !OPS.iter().any(|o| o.0 == op) {
207        return Err(bad("shape.merge", format!("unknown op `{op}` (union, combine, fragment, intersect, subtract)")));
208    }
209    let ids = targets(s, p)?;
210    let st = s.doc()?;
211    let shapes: Vec<Shape> = ids.iter().filter_map(|id| st.shape(*id).cloned()).collect();
212    if let Some(member) = shapes.iter().find(|x| !st.shapes().iter().any(|t| t.id == x.id)) {
213        let name = if member.name.is_empty() { format!("shape {}", member.id.0) } else { format!("`{}`", member.name) };
214        return Err(bad("shape.merge", format!("{name} is in a group; ungroup it first to merge it")));
215    }
216    let mergeable = |x: &Shape| matches!(x.kind, ShapeKind::Shape | ShapeKind::Picture { .. }) && !x.is_line();
217    if shapes.len() < 2 || !shapes.iter().all(mergeable) {
218        return Err(bad("shape.merge", "select two or more shapes (not lines, groups, tables or charts)"));
219    }
220    let areas: Vec<Area> = shapes.iter().map(|x| area_of(&st.doc, &st.selection, x)).collect();
221    let results = merge_areas(&op, &areas);
222    let mut template = shapes[0].clone();
223    let template_box = xfrm_of(&st.doc, &st.selection, &template);
224    // Letters merged become shapes filled with the text's colour.
225    let letters = letters_of(&st.doc, &st.selection, &template);
226    if let Some((_, colour)) = &letters {
227        use crate::model::style::{ColorRef, Fill, Line};
228        template.fill = Some(Fill::solid(ColorRef::rgb(colour.unwrap_or(crate::color::Rgba::BLACK))));
229        template.line = Some(Line { fill: Some(Fill::None), ..Default::default() });
230        template.text_box = false;
231    }
232    let drop_text = letters.is_some() || shapes.iter().skip(1).any(|x| letters_of(&st.doc, &st.selection, x).is_some());
233    let mut new_ids = vec![];
234    s.edit(|doc, sel| {
235        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("shape.merge", "no slide"))?;
236        let at = list.iter().position(|x| x.id == template.id).unwrap_or(list.len());
237        let mut made = vec![];
238        for (k, area) in results.iter().enumerate() {
239            let Some((x, geom)) = area_geom(area) else { continue };
240            let mut sh = template.clone();
241            sh.id = ShapeId(0);
242            sh.ph = None;
243            sh.xfrm = Some(x);
244            sh.geom = geom;
245            sh.name = format!("Freeform {}", k + 1);
246            if let ShapeKind::Picture { fill } = &mut sh.kind {
247                recrop(fill, &template_box, &x);
248            }
249            // Text stays with the first piece only, and not at all once it became outlines.
250            if k > 0 || drop_text {
251                sh.text = None;
252            }
253            made.push(sh);
254        }
255        for sh in &mut made {
256            sh.id = ShapeId(doc.alloc_id().ok_or_else(|| bad("shape.merge", NO_ID_SPACE))?);
257            new_ids.push(sh.id);
258        }
259        let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| bad("shape.merge", "no slide"))?;
260        let at = at.min(list.len());
261        for (k, sh) in made.into_iter().enumerate() {
262            list.insert(at + k, sh);
263        }
264        list.retain(|x| !ids.contains(&x.id));
265        let _ = list;
266        if sel.target == crate::engine::Target::Slides
267            && let Some(sl) = doc.slides.get_mut(sel.slide)
268        {
269            let sl = std::sync::Arc::make_mut(sl);
270            sl.animations.retain(|a| !ids.contains(&a.shape));
271            for a in sl.animations.iter_mut().filter(|a| a.trigger.is_some_and(|t| ids.contains(&t))) {
272                a.trigger = None;
273            }
274        }
275        sel.shapes = new_ids.clone();
276        Ok(())
277    })?;
278    Ok(json!({"ids": new_ids.iter().map(|i| i.0).collect::<Vec<_>>()}))
279}

400    /// selected. Before, the result went to the slide's top level with the member's box from the

403    fn a_shape_in_a_group_is_refused() {
404        fn rect(s: &mut Session, x: f64) -> u64 {
405            s.execute("shape.insert", &json!({"preset": "rect", "rect": [x, 100, 100, 100]})).unwrap()["id"].as_u64().unwrap()
406        }

410        let g = s.execute("arrange.group", &json!({"ids": [a, b]})).unwrap()["id"].as_u64().unwrap();
411        s.execute("shape.move", &json!({"ids": [g], "dx": 300})).unwrap();
412        let before = s.doc().unwrap().current_slide().unwrap().shapes.clone();
413        let name = s.doc().unwrap().shape(ShapeId(a as u32)).unwrap().name.clone();
414        for ids in [json!([a, b]), json!([c, a])] {
415            let err = s.execute("shape.merge", &json!({"op": "union", "ids": ids})).unwrap_err().to_string();
416            assert!(err.contains(&format!("`{name}` is in a group; ungroup it first")), "{ids}: {err}");
417        }
418        assert_eq!(s.doc().unwrap().current_slide().unwrap().shapes, before, "a refused merge leaves the slide as it was");

420        let spec = crate::engine::cmd::find_command("shape.merge").unwrap();

422        assert_eq!((spec.enabled)(&s), Err("a shape in a group can't be merged; ungroup it first".to_string()));
423        assert!(s.execute("shape.merge", &json!({"op": "union"})).is_err());
424        // Named top-level shapes don't depend on the selection.
425        let d = rect(&mut s, 550.0);
426        s.execute("edit.select", &json!({"ids": [a, b]})).unwrap();
427        assert_eq!(s.execute("shape.merge", &json!({"op": "union", "ids": [c, d]})).unwrap()["ids"].as_array().map(Vec::len), Some(1));
