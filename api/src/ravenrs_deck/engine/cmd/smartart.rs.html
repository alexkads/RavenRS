1//! SmartArt: a diagram's outline (the Text Pane) and the shapes generated from it.
2//!
3//! A SmartArt graphic is a group whose `diagram` holds its layout and its nodes (text + outline
4//! level). Every edit changes the nodes and regenerates the group's children inside the group's
5//! child rectangle, so the diagram keeps its size and place. Text typed straight into a diagram
6//! shape is read back into the nodes before each regeneration.

119/// The gallery category a layout is shown under in Convert to SmartArt and Change Layout.
120pub fn category(kind: &str) -> &'static str {
121    match kind {
122        "pictureList" | "pictureCaption" | "pictureProcess" | "pictureHList" => "picture",
123        "process"
124        | "chevron"
125        | "verticalProcess"
126        | "stepUp"
127        | "timeline"
128        | "gear"
129        | "funnel"
130        | "accentProcess"
131        | "continuousBlockProcess"
132        | "bendingProcess" => "process",
133        "cycle" | "blockCycle" | "textCycle" | "continuousCycle" | "segmentedCycle" | "multidirectionalCycle" | "pie" => "cycle",
134        "hierarchy" | "horizontalHierarchy" | "orgChart" | "halfCircleOrgChart" | "nameTitleOrgChart" => "hierarchy",
135        "radial" | "venn" | "target" | "linearVenn" | "stackedVenn" | "nestedTarget" => "relationship",
136        "matrix" | "titledMatrix" | "gridMatrix" | "cycleMatrix" => "matrix",
137        "pyramid" | "invertedPyramid" | "segmentedPyramid" => "pyramid",
138        _ => "list",
139    }
140}

142/// A layout's gallery picture: the graphic drawn with sample nodes (or `nodes`) in a 320 × 200 pt
143/// group at the origin, in the style and colours given, with no text in the samples.
144pub fn preview(kind: &str, style: &str, colors: &str, nodes: Option<&[DiagramNode]>) -> Shape {
145    let sample = |levels: &[u8]| -> Vec<DiagramNode> { levels.iter().map(|l| DiagramNode { level: *l, ..Default::default() }).collect() };
146    let nodes = match nodes {
147        Some(n) if !n.is_empty() => n.to_vec(),
148        _ => match category(kind) {
149            "hierarchy" => sample(&[0, 1, 2, 2, 1]),
150            "matrix" => sample(&[0, 0, 0, 0, 0]),
151            "list" if kind != "blockList" && kind != "tableList" => sample(&[0, 1, 0, 1, 0, 1]),
152            _ => sample(&[0, 0, 0, 0]),
153        },
154    };
155    let d = Diagram { kind: kind.into(), nodes, style: style.into(), colors: colors.into(), ..Default::default() };
156    let area = Xfrm::new(0.0, 0.0, 320.0, 200.0);
157    let mut children = layout(&d, area);
158    for (k, c) in children.iter_mut().enumerate() {
159        c.id = ShapeId(10_000 + k as u32);
160    }
161    Shape {
162        id: ShapeId(9_999),
163        name: "Diagram preview".into(),
164        xfrm: Some(area),
165        kind: ShapeKind::Group { children, child: area },
166        ..Default::default()
167    }
168}

284    fn lv(n: &[DiagramNode]) -> Vec<u8> {
285        n.iter().map(|x| x.level).collect()

287    fn mk(levels: &[u8]) -> Vec<DiagramNode> {
288        levels
289            .iter()
290            .enumerate()
291            .map(|(i, l)| DiagramNode { text: format!("n{i}"), level: *l, picture: None, assistant: false, title: String::new() })
292            .collect()
293    }

295    #[test]
296    fn outline_edits() {
297        let mut n = mk(&[0, 1, 1, 0]);
298        assert_eq!(add(&mut n, Some(0), "after", "x"), 3, "after the node's sub-points");
299        assert_eq!(lv(&n), [0, 1, 1, 0, 0]);
300        assert_eq!(add(&mut n, Some(0), "below", "y"), 3);
301        assert_eq!(lv(&n), [0, 1, 1, 1, 0, 0]);
302        assert_eq!(add(&mut n, Some(0), "above", "z"), 0);
303        assert_eq!(lv(&n), [0, 1, 2, 2, 2, 0, 0]);
304        assert!(remove(&mut n, 1));
305        assert_eq!(lv(&n), [0, 1, 1, 1, 0, 0]);
306        assert!(!promote_node(&mut n, 0));
307        assert!(!demote_node(&mut n, 0), "nothing above the first node");
308        assert!(!demote_node(&mut n, 2) || lv(&n)[2] == 2);
309        let mut m = mk(&[0, 0, 1]);
310        assert!(demote_node(&mut m, 1));
311        assert_eq!(lv(&m), [0, 1, 2], "sub-points move with their node");
312        assert!(promote_node(&mut m, 1));
313        assert_eq!(lv(&m), [0, 0, 1]);
314        let mut bad = mk(&[2, 4, 0]);
315        normalize(&mut bad);
316        assert_eq!(lv(&bad), [0, 1, 0]);
317        assert_eq!(lv(&nodes_from_lines(["A", "\tB", "\t\tC", "", "D"])), [0, 1, 2, 0]);
318    }

320    #[test]
321    fn layouts_map_nodes_to_shapes() {
322        let area = Xfrm::new(0.0, 0.0, 800.0, 400.0);
323        let d = Diagram { kind: "blockList".into(), nodes: mk(&[0, 1, 1, 0]), ..Default::default() };
324        let v = layout(&d, area);
325        assert_eq!(v.len(), 2);
326        let t = v[0].text.as_ref().map(|b| b.paragraphs.len());
327        assert_eq!(t, Some(3), "sub-points are bullets in their parent's shape");
328        let h = Diagram { kind: "hierarchy".into(), nodes: mk(&[0, 1, 2, 1]), ..Default::default() };
329        let v = layout(&h, area);
330        assert_eq!(v.len(), 4 + 3, "a box per node and a line per parent link");
331        for sh in &v {
332            let x = sh.xfrm.unwrap_or_default();
333            assert!(x.x >= -0.01 && x.x + x.w <= 800.01 && x.y >= -0.01 && x.y + x.h <= 400.01, "{x:?}");
334        }
335        assert!(layout(&Diagram { kind: "cycle".into(), ..Default::default() }, area).is_empty());
336        // Vertical Bullet List: a bar per top node, its sub-points in a box below.
337        let v = layout_items(&Diagram { kind: "list".into(), nodes: mk(&[0, 1, 1, 0]), ..Default::default() }, area);
338        let maps: Vec<Vec<usize>> = v.into_iter().map(|(_, m)| m).collect();
339        assert_eq!(maps, [vec![0], vec![1, 2], vec![3]]);
340    }

342    #[test]
343    fn every_layout_shows_every_node_inside_its_area() {
344        let area = Xfrm::new(50.0, 40.0, 800.0, 400.0);
345        for (kind, _) in LAYOUTS {
346            for style in std::iter::once("").chain(STYLES.iter().map(|(k, _)| *k)) {
347                let d = Diagram {
348                    kind: kind.into(),
349                    nodes: mk(&[0, 1, 0, 0, 1, 0]),
350                    style: style.into(),
351                    colors: "accent2".into(),
352                    ..Default::default()
353                };
354                let items = layout_items(&d, area);
355                let mut shown: Vec<usize> = items.iter().flat_map(|(_, m)| m.clone()).collect();
356                shown.sort_unstable();
357                // Gear draws three gears, as PowerPoint does: the fourth group isn't shown.
358                let want: Vec<usize> = (0..if kind == "gear" { 5 } else { 6 }).collect();
359                assert_eq!(shown, want, "{kind}");
360                for (sh, _) in &items {
361                    let x = sh.xfrm.unwrap_or_default();
362                    let (cx, cy) = (x.x + x.w / 2.0, x.y + x.h / 2.0);
363                    assert!(x.w.is_finite() && x.h.is_finite() && x.w >= 0.0 && x.h >= 0.0, "{kind} {x:?}");
364                    assert!((49.0..=851.0).contains(&cx) && (39.0..=441.0).contains(&cy), "{kind} {x:?}");
365                }
366            }
367        }
368    }

390    #[test]
391    fn assistants_sit_beside_their_parent() {
392        let mut v = mk(&[0, 1, 1]);
393        let a = add(&mut v, Some(0), "assistant", "Aide");
394        assert_eq!((a, v[1].level, v[1].assistant), (1, 1, true));
395        let d = Diagram { kind: "orgChart".into(), nodes: v, ..Default::default() };
396        let items = layout_items(&d, Xfrm::new(0.0, 0.0, 600.0, 400.0));
397        let boxed = |k: usize| items.iter().find(|(_, m)| m == &vec![k]).and_then(|(s, _)| s.xfrm).unwrap_or_default();
398        let (ceo, aide, kid) = (boxed(0), boxed(1), boxed(2));
399        assert!(aide.y > ceo.y && aide.y < kid.y, "the assistant's row is between its parent and the children");
400        assert!(aide.x > ceo.x, "beside the parent's trunk");
401    }
