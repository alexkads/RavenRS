151fn theme(s: &mut Session, p: &Value) -> Result<Value> {
152    let name = str_param(p, "name").ok_or_else(|| bad("design.theme", "missing `name`"))?;
153    let t =
154        builtin_themes().into_iter().find(|t| t.name.eq_ignore_ascii_case(name)).ok_or_else(|| bad("design.theme", format!("no theme `{name}`")))?;
155    s.edit(|doc, sel| {
156        let mi = master_index(doc, sel);
157        let m = doc.masters.get_mut(mi).ok_or_else(|| bad("design.theme", "no master"))?;
158        let m = Arc::make_mut(m);
159        m.theme = t.clone();
160        m.name = t.name.clone();
161        m.color_map = vec![];
162        Ok(())
163    })?;
164    Ok(json!({"theme": name}))
165}

167fn browse_theme(s: &mut Session, p: &Value) -> Result<Value> {
168    let bytes = if let Some(data) = str_param(p, "data") {
169        super::base64_decode(data).ok_or_else(|| bad("design.browseTheme", "invalid base64 `data`"))?
170    } else {
171        #[cfg(not(target_arch = "wasm32"))]
172        {
173            let path = str_param(p, "path").ok_or_else(|| bad("design.browseTheme", "give `path` or `data`"))?;
174            crate::engine::io::read_file(path).map_err(crate::engine::EngineError::Other)?
175        }
176        #[cfg(target_arch = "wasm32")]
177        return Err(bad("design.browseTheme", "give `data`"));
178    };
179    if !crate::pptx::sniff(&bytes) {
180        return Err(bad("design.browseTheme", "not a theme or PowerPoint file"));
181    }
182    // A presentation or template brings its slide master and layouts, as in PowerPoint; a .thmx
183    // (its main part is a theme, not a presentation) brings the theme alone.
184    if let Ok(src) = super::file::open_presentation("theme.pptx", &bytes)
185        && src.masters.first().is_some_and(|m| !m.layouts.is_empty())
186    {
187        return apply_master(s, &src);
188    }
189    let (theme, media) = crate::pptx::import_theme(&bytes).map_err(|e| bad("design.browseTheme", e.to_string()))?;
190    let name = theme.name.clone();
191    s.edit(|doc, sel| {
192        // The theme's pictures (format-scheme blip fills) get ids of this deck.
193        let mut map = std::collections::HashMap::new();
194        for m in &media {
195            let id = doc.add_media(&m.name, &m.content_type, m.data.as_ref().clone()).ok_or_else(|| bad("design.browseTheme", NO_ID_SPACE))?;
196            map.insert(m.id.0, id.0);
197        }
198        let theme = if map.is_empty() { theme.clone() } else { remap_media_ids(&theme, &map)? };
199        let mi = master_index(doc, sel);
200        let m = Arc::make_mut(doc.masters.get_mut(mi).ok_or_else(|| bad("design.browseTheme", "no master"))?);
201        m.theme = theme;
202        m.color_map = vec![];
203        Ok(())
204    })?;
205    Ok(json!({"theme": name}))
206}

208/// Design › Browse for Themes… with a `.pptx` / `.potx`: its first master and layouts replace the
209/// master the current slide uses (a preserved master stays beside it), stretched to this deck's
210/// slide size. Each slide on it moves to the new layout with the same name, else the same kind,
211/// keeping its content as Layout does.
212fn apply_master(s: &mut Session, src: &Presentation) -> Result<Value> {
213    let from = src.masters.first().ok_or_else(|| bad("design.browseTheme", "no master"))?;
214    let theme = from.theme.name.clone();
215    s.edit(|doc, sel| {
216        let mi = master_index(doc, sel);
217        let old = doc.masters.get(mi).cloned().ok_or_else(|| bad("design.browseTheme", "no master"))?;
218        let (mut media, mut layouts) = Default::default();
219        let mut m = super::reuse::import_master("design.browseTheme", doc, src, from, &mut media, &mut layouts)?;
220        let (sx, sy) = (doc.slide_size.width / src.slide_size.width.max(1.0), doc.slide_size.height / src.slide_size.height.max(1.0));
221        if (sx - 1.0).abs() > 1e-9 || (sy - 1.0).abs() > 1e-9 {
222            scale_shapes(&mut m.shapes, sx, sy, 1.0, 0.0, 0.0);
223            for l in &mut m.layouts {
224                scale_shapes(&mut l.shapes, sx, sy, 1.0, 0.0, 0.0);
225            }
226        }
227        let moves: Vec<(usize, crate::model::LayoutId)> =
228            doc.slides.iter().enumerate().filter_map(|(i, sl)| old.layout(sl.layout).and_then(|l| matching_layout(&m, l)).map(|l| (i, l))).collect();
229        let layouts = m.layouts.len();
230        if old.preserve {
231            doc.masters.insert(mi, Arc::new(m));
232        } else if let Some(slot) = doc.masters.get_mut(mi) {
233            *slot = Arc::new(m);
234        }
235        for (i, l) in &moves {
236            super::slide::apply_layout(doc, *i, *l)?;
237        }
238        if let Target::Layout { master, .. } = sel.target {
239            sel.target = Target::Master { master };
240        }
241        sel.shapes.clear();
242        sel.text = None;
243        Ok(json!({"theme": theme, "layouts": layouts, "slides": moves.len()}))
244    })
245}

247/// The layout of `m` a slide on `from` moves to: same name, else same kind (not for custom
248/// layouts), else Title and Content, else the first.
249fn matching_layout(m: &crate::model::Master, from: &crate::model::Layout) -> Option<crate::model::LayoutId> {
250    use crate::model::LayoutType;
251    let ls = &m.layouts;
252    ls.iter()
253        .find(|l| !from.name.is_empty() && l.name.trim().eq_ignore_ascii_case(from.name.trim()))
254        .or_else(|| ls.iter().find(|l| from.kind != LayoutType::Custom && l.kind == from.kind))
255        .or_else(|| ls.iter().find(|l| l.kind == LayoutType::TitleAndContent))
256        .or_else(|| ls.first())
257        .map(|l| l.id)
258}

260/// `theme` with every `media` id field rewritten through `map` (format-scheme picture fills).
261fn remap_media_ids(theme: &crate::model::theme::Theme, map: &std::collections::HashMap<u32, u32>) -> Result<crate::model::theme::Theme> {
262    fn rec(v: &mut Value, map: &std::collections::HashMap<u32, u32>, depth: usize) {
263        if depth > 64 {
264            return;
265        }
266        match v {
267            Value::Object(o) => {
268                for (k, x) in o.iter_mut() {
269                    if k == "media"
270                        && let Some(n) = x.as_u64().and_then(|n| u32::try_from(n).ok())
271                        && let Some(new) = map.get(&n)
272                    {
273                        *x = json!(new);
274                    } else {
275                        rec(x, map, depth + 1);
276                    }
277                }
278            }
279            Value::Array(a) => a.iter_mut().for_each(|x| rec(x, map, depth + 1)),
280            _ => {}
281        }
282    }
283    let mut v = serde_json::to_value(theme).map_err(|e| bad("design.browseTheme", e.to_string()))?;
284    rec(&mut v, map, 0);
285    serde_json::from_value(v).map_err(|e| bad("design.browseTheme", e.to_string()))
286}

288fn master_index(doc: &Presentation, sel: &crate::engine::Selection) -> usize {
289    match sel.target {
290        Target::Master { master } | Target::Layout { master, .. } => master,
291        Target::Slides | Target::Handout | Target::NotesMaster => {
292            doc.slides.get(sel.slide).and_then(|s| doc.masters.iter().position(|m| m.layout(s.layout).is_some())).unwrap_or(0)
293        }
294    }
295}

297fn themes(_s: &mut Session, _p: &Value) -> Result<Value> {
298    Ok(json!({
299        "themes": builtin_themes().iter().map(|t| json!({"name": t.name, "major": t.fonts.major.latin, "minor": t.fonts.minor.latin, "colors": t.colors.colors.iter().map(|c| format!("#{}", c.hex())).collect::<Vec<_>>()})).collect::<Vec<_>>(),
300        "colorSchemes": builtin_color_schemes().iter().map(|c| c.name.clone()).collect::<Vec<_>>(),
301        "fontSchemes": builtin_font_schemes().iter().map(|f| json!({"name": f.name, "major": f.major.latin, "minor": f.minor.latin})).collect::<Vec<_>>(),
302        "effectSchemes": builtin_format_schemes().iter().map(|f| f.name.clone()).collect::<Vec<_>>(),
303    }))
