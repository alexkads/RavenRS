8fn slot(i: usize) -> SchemeSlot {
9    [SchemeSlot::Accent1, SchemeSlot::Accent2, SchemeSlot::Accent3, SchemeSlot::Accent4, SchemeSlot::Accent5, SchemeSlot::Accent6][i % 6]
10}

12/// The accent shape `i` is drawn in under Change Colors: `colorful` cycles the accents, `accentN`
13/// keeps one.
14pub(super) fn accent(d: &Diagram, i: usize) -> SchemeSlot {
15    let digit = |s: &str, k: usize| s.chars().nth(k).and_then(|c| c.to_digit(10)).map(|n| n as usize);
16    if let Some(r) = d.colors.strip_prefix("range")
17        && let Some(a @ 1..=6) = digit(r, 0)
18    {
19        // Colorful Range: alternate between two neighbouring accents.
20        return slot(a - 1 + i % 2);
21    }
22    match d.colors.strip_prefix("accent").and_then(|r| digit(r, 0)) {
23        Some(n @ 1..=6) => slot(n - 1),
24        _ => slot(i),
25    }
26}

28/// Gradient Range: each shape a step darker than the one before (the steps cycle).
29pub(super) fn shade(d: &Diagram, i: usize) -> Option<ColorTransform> {
30    d.colors.ends_with("Grad").then(|| ColorTransform::LumMod(100_000 - (i % 4) as i32 * 15_000))
31}

33/// The 3-D XML (`scene3d` + `sp3d`) for a 3-D SmartArt Style: a top bevel (`w`, `h` in points)
34/// lit by the three-point rig.
35fn three_d(bevel: &str, w: f64, h: f64, camera: &str) -> String {
36    format!(
37        "<a:scene3d><a:camera prst=\"{camera}\"/><a:lightRig rig=\"threePt\" dir=\"t\"/></a:scene3d><a:sp3d prstMaterial=\"matte\"><a:bevelT w=\"{}\" h=\"{}\" prst=\"{bevel}\"/></a:sp3d>",
38        (w * 12700.0).round() as i64,
39        (h * 12700.0).round() as i64
40    )
41}

51/// Fill, outline, text colour and effects for the SmartArt style. `light` shapes (the bullet
52/// boxes of the bullet lists) are a pale tint of the accent with dark text.
53pub(super) fn paint(d: &Diagram, sh: &mut Shape, i: usize, light: bool) {
54    let a = accent(d, i);
55    let base = match shade(d, i) {
56        Some(t) => ColorRef::scheme(a).with(t),
57        None => ColorRef::scheme(a),
58    };
59    let tint = |pct: i32| ColorRef::scheme(a).with(ColorTransform::LumMod(100_000 - pct * 1000)).with(ColorTransform::LumOff(pct * 1000));
60    let dark_text = light || matches!(d.style.as_str(), "outline" | "powder");
61    sh.style = Some(ShapeStyle::accent(a));
62    match d.style.as_str() {
63        _ if light => {
64            sh.fill = Some(Fill::solid(tint(80)));
65            sh.line = Some(Line::solid(tint(60), 1.0));
66        }
67        "outline" => {
68            sh.fill = Some(Fill::solid(ColorRef::scheme(SchemeSlot::Lt1)));
69            sh.line = Some(Line::solid(base.clone(), 2.25));
70        }
71        "subtle" => {
72            sh.fill = Some(Fill::solid(base.clone()));
73            sh.line = Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 1.5));
74            sh.effects = Some(Effects { outer_shadow: Some(shadow(4.0, 3.0)), ..Default::default() });
75        }
76        "polished" | "inset" | "cartoon" | "powder" | "brick" => {
77            let (bevel, bw, line, fill) = match d.style.as_str() {
78                "polished" => ("circle", 12.0, Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 0.75)), base.clone()),
79                "inset" => ("relaxedInset", 10.0, None, base.clone()),
80                "cartoon" => ("softRound", 8.0, Some(Line::solid(ColorRef::scheme(SchemeSlot::Tx1), 2.0)), base.clone()),
81                "powder" => ("convex", 14.0, None, tint(40)),
82                _ => ("slope", 6.0, Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 1.0)), base.clone()),
83            };
84            sh.fill = Some(Fill::solid(fill));
85            sh.line = Some(line.unwrap_or_else(|| Line { fill: Some(Fill::None), ..Default::default() }));
86            sh.effects = Some(Effects {
87                outer_shadow: (d.style != "inset").then(|| shadow(6.0, 4.0)),
88                bevel: Some(bevel.into()),
89                raw3d: Some(three_d(bevel, bw, bw * 0.6, if d.style == "brick" { BRICK_CAMERA } else { "orthographicFront" })),
90                ..Default::default()
91            });
92        }
93        "intense" => {
94            sh.fill = Some(Fill::Gradient(Gradient {
95                stops: vec![
96                    GradientStop { pos: 0.0, color: tint(30) },
97                    GradientStop { pos: 1.0, color: base.clone().with(ColorTransform::LumMod(75_000)) },
98                ],
99                shape: GradientShape::Linear { angle: 90.0, scaled: false },
100                rotate_with_shape: true,
101                tile: [0.0; 4],
102                flip: None,
103            }));
104            sh.line = Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 1.0));
105            sh.effects = Some(Effects { outer_shadow: Some(shadow(8.0, 5.0)), ..Default::default() });
106        }
107        _ => {
108            sh.fill = Some(Fill::solid(base.clone()));
109            sh.line = Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 1.5));
110        }
111    }
112    if let Some(body) = &mut sh.text {
113        let c = if dark_text { ColorRef::scheme(SchemeSlot::Tx1) } else { ColorRef::scheme(SchemeSlot::Lt1) };
114        for p in &mut body.paragraphs {
115            for r in &mut p.runs {
116                r.props.fill = Some(Fill::solid(c.clone()));
117            }
118        }
119    }
120}

138/// A shape with text: the first line as the title (centred, larger), the rest as bullets. With
139/// `title` false every line is a bullet (the bullet boxes of the bullet lists).
140pub(super) fn text_shape(preset: &str, x: Xfrm, lines: &[&DiagramNode], title: bool) -> Shape {
141    let mut body = TextBody::default();
142    body.body.autofit = Some(AutoFit::Shrink { font_scale: Some(1.0), line_reduction: 0.0 });
143    let top = lines.first().map(|n| if title { n.level } else { n.level.saturating_sub(1) }).unwrap_or(0);
144    for (k, n) in lines.iter().enumerate() {
145        let mut para = Paragraph::new(&n.text);
146        let head = title && k == 0;
147        if head {
148            para.props.align = Some(Align::Center);
149        } else {
150            para.level = n.level.saturating_sub(top).saturating_sub(1).min(MAX_LEVEL);
151            para.props.align = Some(Align::Left);
152            para.props.bullet = Some(crate::model::text::Bullet::Char { char: "•".into() });
153            para.props.margin_left = Some(12.0 * (para.level as f64 + 1.0));
154            para.props.indent = Some(-10.0);
155        }
156        let size = if head { 20.0 } else { 14.0 };
157        for r in &mut para.runs {
158            r.props.size = Some(size);
159        }
160        para.end_props.size = Some(size);
161        body.paragraphs.push(para);
162    }
163    if body.paragraphs.is_empty() {
164        body.paragraphs.push(Paragraph::new(""));
165    }
166    if !title {
167        body.body.anchor = Some(crate::model::text::Anchor::Top);
168    }
169    Shape { name: "Diagram shape".into(), xfrm: Some(x), geom: Geom::preset(preset), text: Some(body), ..Default::default() }
170}

179pub(super) fn line_shape(a: (f64, f64), b: (f64, f64)) -> Shape {
180    let mut x = Xfrm::new(a.0.min(b.0), a.1.min(b.1), (b.0 - a.0).abs(), (b.1 - a.1).abs());
181    x.flip_h = (b.0 < a.0) != (b.1 < a.1);
182    Shape {
183        name: "Diagram connector".into(),
184        xfrm: Some(x),
185        geom: Geom::preset("line"),
186        fill: Some(Fill::None),
187        line: Some(Line::solid(ColorRef::scheme(SchemeSlot::Accent1).with(ColorTransform::LumMod(60_000)), 1.25)),
188        ..Default::default()
189    }
190}

197/// Lay out `d`, painted with its style and colours, each shape with the nodes it shows; text is
198/// shrunk per shape so its longest word fits on a line.
199pub(super) fn layout_items(d: &Diagram, area: Xfrm) -> Vec<Item> {
200    let mut v = layout_raw(d, area);
201    for (sh, _) in &mut v {
202        fit_words(sh);
203    }
204    v
205}

207/// Rough advance of a character as a fraction of the font size (a sans text face).
208const CHAR_W: f64 = 0.6;
209
210/// Smallest font scale shrinking to fit goes to.
211pub(super) const MIN_SCALE: f64 = 0.25;
212
213/// Shrink the shape's text (all paragraphs together, as PowerPoint's per-shape font size does) so
214/// no word is wider than its line and would break mid-word. Overflow in height is left to the
215/// body's shrink-on-overflow autofit.
216fn fit_words(sh: &mut Shape) {
217    let Some(w) = sh.xfrm.map(|x| x.w) else { return };
218    // Round shapes keep their text in the inscribed rectangle (about 71% of the width).
219    let text_w = match sh.geom.preset_name() {
220        Some("ellipse" | "gear6" | "gear9") => w * 0.71,
221        _ => w,
222    };
223    let Some(body) = &mut sh.text else { return };
224    let avail = text_w - body.body.inset_l.unwrap_or(7.2) - body.body.inset_r.unwrap_or(7.2);
225    let mut scale: f64 = 1.0;
226    for p in &body.paragraphs {
227        let line = avail - p.props.margin_left.unwrap_or(0.0).max(0.0);
228        for r in &p.runs {
229            let size = r.props.size.unwrap_or(18.0);
230            let word = r.text.split_whitespace().map(|t| t.chars().count()).max().unwrap_or(0) as f64;
231            let need = word * size * CHAR_W;
232            if need > 0.0 {
233                scale = scale.min(line.max(0.0) / need);
234            }
235        }
236    }
237    let scale = scale.max(MIN_SCALE);
238    if scale >= 0.999 {
239        return;
240    }
241    let shrink = |v: &mut Option<f64>| {
242        if let Some(x) = v {
243            *x = (*x * scale * 2.0).floor() / 2.0;
244        }
245    };
246    for p in &mut body.paragraphs {
247        for r in &mut p.runs {
248            shrink(&mut r.props.size);
249        }
250        shrink(&mut p.end_props.size);
251        shrink(&mut p.props.margin_left);
252        if let Some(i) = &mut p.props.indent {
253            *i *= scale;
254        }
255    }
256}

258/// A picture placeholder for node `node`: its picture when it has one, else a pale box.
259pub(super) fn picture_shape(d: &Diagram, preset: &str, x: Xfrm, i: usize, node: usize) -> Shape {
260    let mut sh = Shape { name: "Diagram picture".into(), xfrm: Some(x), geom: Geom::preset(preset), ..Default::default() };
261    paint(d, &mut sh, i, true);
262    if let Some(m) = d.nodes.get(node).and_then(|n| n.picture) {
263        sh.fill = Some(Fill::Picture(crate::model::style::PictureFill { media: m, ..Default::default() }));
264        sh.line = Some(Line::solid(ColorRef::scheme(SchemeSlot::Lt1), 1.5));
265    }
266    sh
