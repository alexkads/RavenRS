5/// Organization Chart: [`hierarchy`] with elbow connectors, and assistants beside their parent on
6/// a row of their own between the parent and its other children. `half_circle` draws each node as
7/// its text in an open half ring (Half Circle Organization Chart).
8pub(super) fn org_chart(d: &Diagram, area: Xfrm) -> Vec<Item> {
9    let half_circle = d.kind == "halfCircleOrgChart";
10    let name_title = d.kind == "nameTitleOrgChart";
11    let nodes = &d.nodes[..d.nodes.len().min(MAX_SHAPES)];
12    if nodes.is_empty() {
13        return vec![];
14    }
15    let mut parent: Vec<Option<usize>> = vec![None; nodes.len()];
16    let mut stack: Vec<usize> = vec![];
17    for (i, n) in nodes.iter().enumerate() {
18        while stack.last().is_some_and(|&p| nodes[p].level >= n.level) {
19            stack.pop();
20        }
21        parent[i] = stack.last().copied();
22        stack.push(i);
23    }
24    let asst = |i: usize| nodes[i].assistant && parent[i].is_some();
25    let kids = |i: Option<usize>| -> Vec<usize> { (0..nodes.len()).filter(|&k| parent[k] == i && !asst(k)).collect() };
26    let mut leaves = vec![0usize; nodes.len()];
27    for i in (0..nodes.len()).rev() {
28        let k = kids(Some(i));
29        leaves[i] = if k.is_empty() { 1 } else { k.iter().map(|&c| leaves[c]).sum() };
30    }
31    // Row slots: a node's level is 2 per depth; assistants take the odd slot under their parent.
32    let mut slot = vec![0usize; nodes.len()];
33    for i in 0..nodes.len() {
34        slot[i] = match parent[i] {
35            None => 0,
36            Some(p) if asst(i) => slot[p] + 1,
37            Some(p) => slot[p] + 2,
38        };
39    }
40    let mut used: Vec<usize> = slot.clone();
41    used.sort_unstable();
42    used.dedup();
43    let row = |i: usize| used.iter().position(|&s| s == slot[i]).unwrap_or(0);
44    let rows = used.len().max(1);
45    let total: usize = kids(None).iter().map(|&r| leaves[r]).sum::<usize>().max(1);
46    let cell_w = area.w / total as f64;
47    let row_h = area.h / rows as f64;
48    let (bw, bh) = ((cell_w * 0.85).min(area.w * 0.3), row_h * 0.62);
49    let mut left = vec![0usize; nodes.len()];
50    let mut next_root = 0;
51    for i in 0..nodes.len() {
52        if asst(i) {
53            continue;
54        }
55        let mut start = match parent[i] {
56            None => {
57                let s = next_root;
58                next_root += leaves[i];
59                s
60            }
61            Some(p) => left[p],
62        };
63        if let Some(p) = parent[i] {
64            for s in kids(Some(p)) {
65                if s == i {
66                    break;
67                }
68                start += leaves[s];
69            }
70        }
71        left[i] = start;
72    }
73    let mut side = vec![0usize; nodes.len()];
74    let boxes: Vec<Xfrm> = (0..nodes.len())
75        .map(|i| {
76            let y = area.y + row(i) as f64 * row_h + (row_h - bh) / 2.0;
77            match parent[i] {
78                Some(p) if asst(i) => {
79                    // Assistants alternate right and left of the parent's trunk.
80                    let k = (0..i).filter(|&j| parent[j] == Some(p) && asst(j)).count();
81                    side[i] = k;
82                    let pc = area.x + (left[p] as f64 + leaves[p] as f64 / 2.0) * cell_w;
83                    let off = bw * 0.15 + (k / 2) as f64 * bw * 1.1;
84                    let x = if k % 2 == 0 { pc + off } else { pc - off - bw };
85                    Xfrm::new(x.clamp(area.x, area.x + area.w - bw), y, bw, bh)
86                }
87                _ => {
88                    let cx = area.x + (left[i] as f64 + leaves[i] as f64 / 2.0) * cell_w;
89                    Xfrm::new(cx - bw / 2.0, y, bw, bh)
90                }
91            }
92        })
93        .collect();
94    let mut out: Vec<Item> = vec![];
95    for i in 0..nodes.len() {
96        let Some(p) = parent[i] else { continue };
97        let (a, b) = (boxes[p], boxes[i]);
98        let pc = a.x + a.w / 2.0;
99        if asst(i) {
100            let my = b.y + b.h / 2.0;
101            out.push((line_shape((pc, a.y + a.h), (pc, my)), vec![]));
102            let ex = if b.x > pc { b.x } else { b.x + b.w };
103            out.push((line_shape((pc, my), (ex, my)), vec![]));
104        } else {
105            let mid = b.y - (row_h - bh) / 2.0;
106            let cc = b.x + b.w / 2.0;
107            out.push((line_shape((pc, a.y + a.h), (pc, mid)), vec![]));
108            out.push((line_shape((pc.min(cc), mid), (pc.max(cc), mid)), vec![]));
109            out.push((line_shape((cc, mid), (cc, b.y)), vec![]));
110        }
111    }
112    let _ = side;
113    for (i, n) in nodes.iter().enumerate() {
114        let depth = slot[i] / 2;
115        if half_circle {
116            // Two half rings round the text: the upper left one in the accent, the lower right one
117            // in its tint.
118            let x = boxes[i];
119            for (k, (a0, a1)) in [(140.0, 310.0), (320.0, 130.0)].into_iter().enumerate() {
120                let mut arc = deco("blockArc", x, 0.0);
121                arc.geom = Geom::Preset { name: "blockArc".into(), adj: vec![a0 * 60_000.0, a1 * 60_000.0, 4_000.0] };
122                paint(d, &mut arc, depth, false);
123                if k == 1 {
124                    let c = ColorRef::scheme(accent(d, depth)).with(ColorTransform::LumMod(60_000)).with(ColorTransform::LumOff(40_000));
125                    arc.fill = Some(Fill::solid(c));
126                }
127                arc.line = Some(Line { fill: Some(Fill::None), ..Default::default() });
128                arc.effects = None;
129                out.push((arc, vec![]));
130            }
131            let mut t = text_shape("rect", x, &[n], true);
132            paint(d, &mut t, depth, true);
133            t.fill = Some(Fill::None);
134            t.line = Some(Line { fill: Some(Fill::None), ..Default::default() });
135            out.push((t, vec![i]));
136        } else {
137            let mut x = boxes[i];
138            if name_title {
139                // The name in the upper part; the title box overlaps the lower edge.
140                x.h *= 0.85;
141            }
142            let mut sh = text_shape("rect", x, &[n], true);
143            paint(d, &mut sh, depth, false);
144            out.push((sh, vec![i]));
145            if name_title {
146                let tb = Xfrm::new(x.x + x.w * 0.12, x.y + x.h * 0.72, x.w * 0.8, x.h * 0.42);
147                let title = DiagramNode { text: n.title.clone(), ..Default::default() };
148                let mut t = text_shape("rect", tb, &[&title], true);
149                paint(d, &mut t, depth, true);
150                t.fill = Some(Fill::solid(ColorRef::scheme(SchemeSlot::Lt1)));
151                if let Some(b) = &mut t.text {
152                    for p in &mut b.paragraphs {
153                        for r in &mut p.runs {
154                            r.props.size = Some(11.0);
155                        }
156                        p.end_props.size = Some(11.0);
157                    }
158                }
159                out.push((t, vec![]));
160            }
161        }
162    }
163    out
164}

166/// An organisation chart: one box per node, rows by level, each subtree as wide as its leaves.
167pub(super) fn hierarchy(d: &Diagram, area: Xfrm, horizontal: bool) -> Vec<Item> {
168    let nodes = &d.nodes[..d.nodes.len().min(MAX_SHAPES)];
169    if nodes.is_empty() {
170        return vec![];
171    }
172    let mut parent: Vec<Option<usize>> = vec![None; nodes.len()];
173    let mut stack: Vec<usize> = vec![];
174    for (i, n) in nodes.iter().enumerate() {
175        while stack.last().is_some_and(|&p| nodes[p].level >= n.level) {
176            stack.pop();
177        }
178        parent[i] = stack.last().copied();
179        stack.push(i);
180    }
181    let kids = |i: Option<usize>| -> Vec<usize> { (0..nodes.len()).filter(|&k| parent[k] == i).collect() };
182    let mut leaves = vec![0usize; nodes.len()];
183    for i in (0..nodes.len()).rev() {
184        let k = kids(Some(i));
185        leaves[i] = if k.is_empty() { 1 } else { k.iter().map(|&c| leaves[c]).sum() };
186    }
187    let depth: Vec<usize> = (0..nodes.len())
188        .map(|i| {
189            let (mut d, mut p) = (0, parent[i]);
190            while let Some(q) = p {
191                d += 1;
192                p = parent[q];
193            }
194            d
195        })
196        .collect();
197    let rows = depth.iter().max().copied().unwrap_or(0) + 1;
198    let total: usize = kids(None).iter().map(|&r| leaves[r]).sum::<usize>().max(1);
199    // Horizontal Hierarchy: the same tree turned a quarter, levels from left to right.
200    let (span, across) = if horizontal { (area.h, area.w) } else { (area.w, area.h) };
201    let cell_w = span / total as f64;
202    let row_h = across / rows as f64;
203    let (bw, bh) = if horizontal { (cell_w * 0.8, row_h * 0.7) } else { (cell_w * 0.85, row_h * 0.6) };
204    // Left edge (in leaves) of each node's span.
205    let mut left = vec![0usize; nodes.len()];
206    let mut next_root = 0;
207    for i in 0..nodes.len() {
208        let mut start = match parent[i] {
209            None => {
210                let s = next_root;
211                next_root += leaves[i];
212                s
213            }
214            Some(p) => left[p],
215        };
216        if let Some(p) = parent[i] {
217            for s in kids(Some(p)) {
218                if s == i {
219                    break;
220                }
221                start += leaves[s];
222            }
223        }
224        left[i] = start;
225    }
226    let boxes: Vec<Xfrm> = (0..nodes.len())
227        .map(|i| {
228            let c = (left[i] as f64 + leaves[i] as f64 / 2.0) * cell_w;
229            let r = depth[i] as f64 * row_h + (row_h - bh) / 2.0;
230            if horizontal { Xfrm::new(area.x + r, area.y + c - bw / 2.0, bh, bw) } else { Xfrm::new(area.x + c - bw / 2.0, area.y + r, bw, bh) }
231        })
232        .collect();
233    let mut out: Vec<Item> = nodes
234        .iter()
235        .enumerate()
236        .map(|(i, n)| {
237            let mut sh = text_shape("roundRect", boxes[i], &[n], true);
238            paint(d, &mut sh, depth[i], false);
239            (sh, vec![i])
240        })
241        .collect();
242    for i in 0..nodes.len() {
243        if let Some(p) = parent[i] {
244            let (a, b) = (boxes[p], boxes[i]);
245            let (p, q) = if horizontal {
246                ((a.x + a.w, a.y + a.h / 2.0), (b.x, b.y + b.h / 2.0))
247            } else {
248                ((a.x + a.w / 2.0, a.y + a.h), (b.x + b.w / 2.0, b.y))
249            };
250            out.push((line_shape(p, q), vec![]));
251        }
252    }
253    out
