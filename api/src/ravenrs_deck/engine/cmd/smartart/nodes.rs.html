4/// Nodes from text lines: each leading tab is one outline level.
5pub fn nodes_from_lines<'a>(lines: impl IntoIterator<Item = &'a str>) -> Vec<DiagramNode> {
6    lines
7        .into_iter()
8        .filter(|l| !l.trim().is_empty())
9        .map(|l| {
10            let tabs = l.chars().take_while(|c| *c == '\t').count();
11            DiagramNode {
12                text: l.trim().to_string(),
13                level: u8::try_from(tabs).unwrap_or(MAX_LEVEL).min(MAX_LEVEL),
14                picture: None,
15                assistant: false,
16                title: String::new(),
17            }
18        })
19        .collect()
20}

22/// Levels made consistent: the first node is top level and no node is more than one level
23/// deeper than the node before it.
24pub fn normalize(nodes: &mut [DiagramNode]) {
25    let mut prev: Option<u8> = None;
26    for n in nodes.iter_mut() {
27        n.level = match prev {
28            None => 0,
29            Some(p) => n.level.min(p.saturating_add(1)).min(MAX_LEVEL),
30        };
31        prev = Some(n.level);
32    }
33}

35/// Index one past the last descendant of node `i`.
36fn subtree_end(nodes: &[DiagramNode], i: usize) -> usize {
37    let lv = nodes.get(i).map(|n| n.level).unwrap_or(0);
38    let mut j = i + 1;
39    while nodes.get(j).is_some_and(|n| n.level > lv) {
40        j += 1;
41    }
42    j
43}

45/// Add a node next to node `index` (PowerPoint's Add Shape After/Before/Below/Above). Returns the
46/// new node's index.
47pub fn add(nodes: &mut Vec<DiagramNode>, index: Option<usize>, position: &str, text: &str) -> usize {
48    let Some(i) = index.filter(|i| *i < nodes.len()) else {
49        nodes.push(DiagramNode { text: text.into(), level: 0, picture: None, assistant: false, title: String::new() });
50        return nodes.len() - 1;
51    };
52    let lv = nodes[i].level;
53    let (at, level) = match position {
54        "before" => (i, lv),
55        "below" => (subtree_end(nodes, i), (lv + 1).min(MAX_LEVEL)),
56        "above" => {
57            let end = subtree_end(nodes, i);
58            for n in &mut nodes[i..end] {
59                n.level = (n.level + 1).min(MAX_LEVEL);
60            }
61            (i, lv)
62        }
63        "assistant" => (i + 1, (lv + 1).min(MAX_LEVEL)),
64        _ => (subtree_end(nodes, i), lv),
65    };
66    nodes.insert(at, DiagramNode { text: text.into(), level, picture: None, assistant: position == "assistant", title: String::new() });
67    at
68}

70/// Remove node `i`; its sub-points move up one level.
71pub fn remove(nodes: &mut Vec<DiagramNode>, i: usize) -> bool {
72    if i >= nodes.len() {
73        return false;
74    }
75    let end = subtree_end(nodes, i);
76    for n in &mut nodes[i + 1..end] {
77        n.level = n.level.saturating_sub(1);
78    }
79    nodes.remove(i);
80    normalize(nodes);
81    true
82}

84/// Move node `i` and its sub-points one level up. False when it is already top level.
85pub fn promote_node(nodes: &mut [DiagramNode], i: usize) -> bool {
86    if nodes.get(i).is_none_or(|n| n.level == 0) {
87        return false;
88    }
89    let end = subtree_end(nodes, i);
90    for n in &mut nodes[i..end] {
91        n.level -= 1;
92    }
93    true
94}

96/// Move node `i` and its sub-points one level down, under the node before it. False when there is
97/// nothing to go under.
98pub fn demote_node(nodes: &mut [DiagramNode], i: usize) -> bool {
99    let (Some(prev), Some(cur)) = (i.checked_sub(1).and_then(|p| nodes.get(p)), nodes.get(i)) else { return false };
100    if cur.level > prev.level || cur.level >= MAX_LEVEL {
101        return false;
102    }
103    let end = subtree_end(nodes, i);
104    if nodes[i..end].iter().any(|n| n.level >= MAX_LEVEL) {
105        return false;
106    }
107    for n in &mut nodes[i..end] {
108        n.level += 1;
109    }
110    true
111}

116/// Top-level nodes, each with its sub-points.
117pub(super) fn groups(d: &Diagram) -> Vec<Vec<usize>> {
118    let mut out: Vec<Vec<usize>> = vec![];
119    for (i, n) in d.nodes.iter().enumerate() {
120        match out.last_mut() {
121            Some(g) if n.level > 0 => g.push(i),
122            _ => out.push(vec![i]),
123        }
124    }
125    out.truncate(MAX_SHAPES);
126    out
127}

129/// Read text typed into the diagram's shapes back into its nodes.
130pub(super) fn sync_text(d: &mut Diagram, children: &[Shape]) {
131    if d.imported {
132        return;
133    }
134    let map: Vec<Vec<usize>> = layout_items(d, Xfrm::new(0.0, 0.0, 100.0, 100.0)).into_iter().map(|(_, m)| m).collect();
135    for (shape, idx) in children.iter().zip(map) {
136        let Some(body) = &shape.text else { continue };
137        if idx.is_empty() || body.paragraphs.len() != idx.len() {
138            continue;
139        }
140        for (p, i) in body.paragraphs.iter().zip(idx) {
141            if let Some(n) = d.nodes.get_mut(i) {
142                n.text = p.text();
143            }
144        }
145    }
