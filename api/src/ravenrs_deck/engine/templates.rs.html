175/// Starters whose name, summary, theme or a slide title contains `query` (empty matches all).
176pub fn matching(query: &str) -> Vec<&'static Starter> {
177    let q = query.trim().to_lowercase();
178    let hit = |text: &str| text.to_lowercase().contains(&q);
179    starters()
180        .iter()
181        .filter(|t| q.is_empty() || hit(t.name) || hit(t.summary) || hit(t.theme) || t.slides.iter().any(|slide| hit(slide.title)))
182        .collect()
183}

185pub fn find(name: &str) -> Option<&'static Starter> {
186    let name = name.trim();
187    starters().iter().find(|t| t.name.eq_ignore_ascii_case(name))
188}

190fn text_with_levels(text: &str) -> TextBody {
191    let mut body = TextBody::from_text(text);
192    for para in &mut body.paragraphs {
193        let tabs = para.runs.first().map(|r| r.text.chars().take_while(|c| *c == '\t').count()).unwrap_or(0);
194        if tabs > 0
195            && let Some(run) = para.runs.first_mut()
196        {
197            run.text = run.text.trim_start_matches('\t').to_string();
198            para.level = tabs.min(8) as u8;
199        }
200    }
201    body
202}

204/// A new presentation filled with `name`'s slides. `size` is the slide size in points.
205pub fn build(name: &str, size: Size) -> Result<Presentation> {
206    let starter = find(name).ok_or_else(|| {
207        let names = starters().iter().map(|t| t.name).collect::<Vec<_>>().join(", ");
208        bad("file.new", format!("no template `{name}`; use {names}"))
209    })?;
210    let theme = crate::model::theme::builtin_themes()
211        .into_iter()
212        .find(|t| t.name.eq_ignore_ascii_case(starter.theme))
213        .ok_or_else(|| bad("file.new", format!("template `{}` names a missing theme `{}`", starter.name, starter.theme)))?;
214    let mut doc = defaults::blank_presentation(size, theme, false);
215    for seed in starter.slides {
216        let key = Value::String(seed.layout.to_string());
217        let layout = find_layout(&doc, &key, 0)
218            .ok_or_else(|| bad("file.new", format!("template `{}` uses an unknown layout `{}`", starter.name, seed.layout)))?;
219        let mut slide = defaults::new_slide(&mut doc, layout).ok_or_else(|| bad("file.new", crate::engine::cmd::NO_ID_SPACE))?;
220        if !seed.title.is_empty()
221            && let Some(sh) = slide.shapes.iter_mut().find(|sh| sh.ph_type().is_some_and(|k| k.is_title()))
222        {
223            sh.text = Some(text_with_levels(seed.title));
224        }
225        let mut bodies = seed.bodies.iter();
226        for sh in slide.shapes.iter_mut().filter(|sh| sh.ph_type().is_some_and(|k| !k.is_title())) {
227            let Some(text) = bodies.next() else { break };
228            if !text.is_empty() {
229                sh.text = Some(text_with_levels(text));
230            }
231        }
232        if bodies.next().is_some() {
233            return Err(bad("file.new", format!("template `{}` has more text than the `{}` layout holds", starter.name, seed.layout)));
234        }
235        if !seed.notes.is_empty() {
236            slide.notes = text_with_levels(seed.notes);
237        }
238        doc.slides.push(Arc::new(slide));
239    }
240    Ok(doc)
241}

250    fn body(doc: &Presentation, slide: usize, n: usize) -> (String, u8) {
251        let para_slide = &doc.slides[slide];
252        let text = para_slide
253            .shapes
254            .iter()
255            .filter(|sh| sh.ph_type().is_some_and(|k| !k.is_title()))
256            .nth(n)
257            .and_then(|sh| sh.text.as_ref())
258            .cloned()
259            .unwrap_or_default();
260        let level = text.paragraphs.get(1).map(|p| p.level).unwrap_or(0);
261        (text.text(), level)
262    }

264    #[test]
265    fn every_starter_opens_on_its_theme_and_is_a_clean_document() {
266        let mut s = Session::new();
267        for starter in starters() {
268            s.execute("file.new", &json!({"template": starter.name.to_lowercase()})).unwrap();
269            let st = s.active().unwrap();
270            assert_eq!(st.doc.masters[0].theme.name, starter.theme, "{}", starter.name);
271            assert_eq!(st.doc.slides.len(), starter.slides.len(), "{}", starter.name);
272            assert_eq!(st.doc.slides[0].title(), starter.slides[0].title, "{}", starter.name);
273            assert!(!st.is_dirty(), "{}", starter.name);
274            assert!(st.history.undo.is_empty(), "{}", starter.name);
275            assert_eq!(st.selection.slide, 0);
276        }
277        let listed = s.execute("file.templates", &json!({})).unwrap();
278        assert_eq!(listed["templates"].as_array().map(Vec::len), Some(starters().len()));
279    }

283        let mut s = Session::new();

287        assert_eq!(doc.slides[0].notes_text(), "Open with what changed since last week.");
288        let (agenda, level) = body(doc, 1, 0);
289        assert!(agenda.contains("One concrete change"), "{agenda}");
290        assert_eq!(level, 1);
291        let (done, _) = body(doc, 3, 0);
292        let (next, _) = body(doc, 3, 1);
293        assert!(done.starts_with("Done"), "{done}");
294        assert!(next.starts_with("Next"), "{next}");
295    }
