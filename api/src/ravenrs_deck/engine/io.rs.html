1//! Reading and writing the files a command names by path.
2//!
3//! Every such read and write goes through [`read_file`] and [`write_file`], so the automation
4//! roots ([`crate::engine::file_access`]) are checked in one place. A command module that called
5//! `std::fs` itself would bypass them; a test in `file_access` keeps that from happening.
6
7/// The bytes of the file at `path`, inside the automation read root when there is one.
8#[cfg(not(target_arch = "wasm32"))]
9pub fn read_file(path: &str) -> Result<Vec<u8>, String> {
10    crate::engine::file_access::check_read(path)?;
11    let meta = std::fs::metadata(path).map_err(|e| format!("{path}: {e}"))?;
12    if !meta.is_file() {
13        return Err(format!("{path}: not a regular file"));
14    }
15    const MAX_FILE_SIZE: u64 = 2 << 30; // 2 GiB
16    if meta.len() > MAX_FILE_SIZE {
17        return Err(format!("{path}: file size {} exceeds 2 GiB limit", meta.len()));
18    }
19    std::fs::read(path).map_err(|e| format!("{path}: {e}"))
20}

27/// Write `bytes` to the file at `path`, inside the automation write root when there is one.
28///
29/// The bytes go to a temporary sibling first and are renamed over `path`, so a failed save never
30/// truncates the old file. The sibling is in the same folder, so it is inside the root whenever
31/// `path` is.
32#[cfg(not(target_arch = "wasm32"))]
33pub fn write_file(path: &str, bytes: &[u8]) -> Result<(), String> {
34    use std::io::Write;
35    use std::sync::atomic::{AtomicU64, Ordering};
36    static COUNTER: AtomicU64 = AtomicU64::new(0);
37    crate::engine::file_access::check_write(path)?;
38    let err = |e: std::io::Error| format!("{path}: {e}");
39
40    let target_path = match std::fs::canonicalize(path) {
41        Ok(canonical) => canonical,
42        Err(_) => std::path::PathBuf::from(path),
43    };
44    let existing_perms = std::fs::metadata(&target_path).ok().map(|m| m.permissions());
45
46    let mut attempts = 0;
47    let (tmp, mut file) = loop {
48        // A short fixed-size name: appending to the target's name could exceed the file name limit.
49        let tmp_name = format!(".deckcraft.{}.{}.saving", std::process::id(), COUNTER.fetch_add(1, Ordering::Relaxed));
50        let tmp = target_path.with_file_name(tmp_name);
51        // The sibling must pass the same check: when `path` is the write root itself it lands outside.
52        crate::engine::file_access::check_write(&tmp.to_string_lossy())?;
53        match std::fs::OpenOptions::new().write(true).create_new(true).open(&tmp) {
54            Ok(f) => break (tmp, f),
55            Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists && attempts < 16 => attempts += 1,
56            Err(e) => return Err(err(e)),
57        }
58    };
59    if let Some(perms) = existing_perms {
60        let _ = file.set_permissions(perms);
61    }
62    let written = file.write_all(bytes).and_then(|()| file.sync_all());
63    drop(file);
64    let result = written.and_then(|()| std::fs::rename(&tmp, &target_path));
65    if let Err(e) = result {
66        let _ = std::fs::remove_file(&tmp);
67        return Err(err(e));
68    }
69    Ok(())
70}

77/// The size in bytes of the file at `path`; `None` when it can't be read (or is outside the
78/// automation read root, so File › Info doesn't reveal files the agent may not read).
79#[cfg(not(target_arch = "wasm32"))]
80pub fn file_len(path: &str) -> Option<u64> {
81    crate::engine::file_access::check_read(path).ok()?;
82    std::fs::metadata(path).ok().map(|m| m.len())
83}

90#[cfg(test)]
91mod tests {
92    use super::*;
93
94    #[test]
95    fn read_file_rejects_directory() {
96        let dir = std::env::temp_dir();
97        let res = read_file(dir.to_str().unwrap());
98        assert!(res.is_err());
99    }
100
101    #[test]
102    fn write_file_preserves_permissions_and_writes_content() {
103        let tmp_dir = std::env::temp_dir().join("deckcraft_io_test");
104        let _ = std::fs::create_dir_all(&tmp_dir);
105        let test_file = tmp_dir.join("test_write.txt");
106        let path_str = test_file.to_str().unwrap();
107
108        assert!(write_file(path_str, b"hello world").is_ok());
109        let read = read_file(path_str).unwrap();
110        assert_eq!(read, b"hello world");
111
112        #[cfg(unix)]
113        {
114            use std::os::unix::fs::PermissionsExt;
115            let perms = std::fs::Permissions::from_mode(0o640);
116            std::fs::set_permissions(&test_file, perms).unwrap();
117
118            assert!(write_file(path_str, b"updated world").is_ok());
119            let new_meta = std::fs::metadata(&test_file).unwrap();
120            assert_eq!(new_meta.permissions().mode() & 0o777, 0o640);
121        }
122
123        let _ = std::fs::remove_file(&test_file);
124        let _ = std::fs::remove_dir(&tmp_dir);
125    }
126
127    #[test]
128    fn write_file_overwrites_file_with_name_near_the_length_limit() {
129        let dir = std::env::temp_dir().join("deckcraft_io_long_name_test");
130        let _ = std::fs::create_dir_all(&dir);
131        let file = dir.join(format!("{}.pptx", "a".repeat(250)));
132        std::fs::write(&file, b"old").unwrap();
133        let path_str = file.to_str().unwrap();
134
135        write_file(path_str, b"new").unwrap();
136        assert_eq!(read_file(path_str).unwrap(), b"new");
137
138        let _ = std::fs::remove_file(&file);
139    }
140
141    #[test]
142    fn write_file_to_the_write_root_itself_creates_no_sibling_outside_it() {
143        let base = std::env::temp_dir().join(format!("deckcraft_io_root_test_{}", std::process::id()));
144        let root = base.join("root");
145        std::fs::create_dir_all(&root).unwrap();
146        let roots = crate::engine::file_access::AutomationRoots::new(&[], std::slice::from_ref(&root)).unwrap().unwrap();
147        let res = crate::engine::file_access::confine(Some(&roots), || write_file(root.to_str().unwrap(), b"x"));
148        assert!(res.unwrap_err().contains("outside the write root"));
149        assert_eq!(std::fs::read_dir(&base).unwrap().count(), 1, "only the root is there");
150        let _ = std::fs::remove_dir_all(&base);
151    }
