10/// Connection sites of a top-level shape `s` in slide coordinates (index = site index stored on
11/// connectors). For a shape inside a group use [`sites_of`] or [`sites_in`].
12pub fn sites(doc: &Presentation, sel: &Selection, s: &Shape) -> Vec<Point> {
13    sites_in(doc, sel, s, Affine::IDENTITY)
14}
15
16/// Connection sites of `s` in slide coordinates, `to_abs` mapping the space its box lives in (the
17/// slide, or the child space of the groups around it) onto the slide.
18pub fn sites_in(doc: &Presentation, sel: &Selection, s: &Shape, to_abs: Affine) -> Vec<Point> {
19    if s.is_line() || matches!(s.kind, ShapeKind::Group { .. } | ShapeKind::Ink { .. }) {
20        return vec![];
21    }
22    let x = xfrm_of(doc, sel, s);
23    let mut local = crate::render::shape_geometry(s, x.w, x.h).sites;
24    if local.is_empty() {
25        local = vec![Point::new(x.w / 2.0, 0.0), Point::new(0.0, x.h / 2.0), Point::new(x.w / 2.0, x.h), Point::new(x.w, x.h / 2.0)];
26    }
27    let a = to_abs * x.affine();
28    local.into_iter().map(|p| a * p).collect()
29}

31/// The shape `id` anywhere in `shapes` (a group's member included) with the affine that maps the
32/// space its box lives in onto the slide: a connector glues to a grouped shape, as in PowerPoint,
33/// and its sites are where the group draws them.
34pub fn locate<'a>(doc: &Presentation, sel: &Selection, shapes: &'a [Shape], id: ShapeId) -> Option<(&'a Shape, Affine)> {
35    fn rec<'a>(doc: &Presentation, sel: &Selection, shapes: &'a [Shape], id: ShapeId, to_abs: Affine, depth: usize) -> Option<(&'a Shape, Affine)> {
36        if depth > 64 {
37            return None;
38        }
39        for s in shapes {
40            if s.id == id {
41                return Some((s, to_abs));
42            }
43            if let ShapeKind::Group { children, child } = &s.kind {
44                let x = xfrm_of(doc, sel, s);
45                let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
46                let nested = to_abs * crate::geom::group_child_affine(&x, ch);
47                if let Some(f) = rec(doc, sel, children, id, nested, depth + 1) {
48                    return Some(f);
49                }
50            }
51        }
52        None
53    }
54    rec(doc, sel, shapes, id, Affine::IDENTITY, 0)
55}

57/// Connection sites of shape `id` anywhere in `shapes`, in slide coordinates; `None` when there is
58/// no such shape.
59pub fn sites_of(doc: &Presentation, sel: &Selection, shapes: &[Shape], id: ShapeId) -> Option<Vec<Point>> {
60    locate(doc, sel, shapes, id).map(|(s, a)| sites_in(doc, sel, s, a))
61}

63/// The nearest connection site within `radius` of `p` on any visible shape except `exclude`.
64pub fn nearest_site(
65    doc: &Presentation,
66    sel: &Selection,
67    shapes: &[Shape],
68    p: Point,
69    radius: f64,
70    exclude: Option<ShapeId>,
71) -> Option<(ShapeId, u32, Point)> {
72    let mut best: Option<(f64, ShapeId, u32, Point)> = None;
73    for s in shapes.iter().filter(|s| Some(s.id) != exclude && !s.hidden) {
74        for (i, q) in sites(doc, sel, s).into_iter().enumerate() {
75            let d = (q - p).hypot();
76            if d <= radius && best.is_none_or(|b| d < b.0) {
77                best = Some((d, s.id, i as u32, q));
78            }
79        }
80    }
81    best.map(|b| (b.1, b.2, b.3))
82}

84/// The shape whose sites should show while drawing or dragging a connector end near `p`.
85pub fn site_shape_at(doc: &Presentation, sel: &Selection, shapes: &[Shape], p: Point, margin: f64, exclude: Option<ShapeId>) -> Option<ShapeId> {
86    shapes
87        .iter()
88        .rev()
89        .filter(|s| Some(s.id) != exclude && !s.hidden && !s.is_line() && !matches!(s.kind, ShapeKind::Ink { .. }))
90        .find(|s| xfrm_of(doc, sel, s).bounds().inflate(margin, margin).contains(p))
91        .map(|s| s.id)
92}

94/// Line endpoints (start, end) of a line-like shape's box.
95pub fn endpoints(x: &Xfrm) -> (Point, Point) {
96    let a = x.affine();
97    (a * Point::new(0.0, 0.0), a * Point::new(x.w, x.h))
98}

100/// Move the ends of connectors glued to shapes on the edited slide to the current site positions.
101/// Connections to shapes that no longer exist are dropped.
102pub fn reroute(doc: &mut Presentation, sel: &Selection) {
103    let Some(list) = crate::engine::shapes_mut(doc, sel) else { return };
104    if !list.iter().any(|s| matches!(s.kind, ShapeKind::Connector { start: Some(_), .. } | ShapeKind::Connector { end: Some(_), .. })) {
105        return;
106    }
107    let snapshot: Vec<Shape> = list.clone();
108    let view = doc.clone();
109    let site = |id: ShapeId, idx: u32| -> Option<Point> {
110        // Anywhere in the tree: a connector glued to a shape that is then grouped keeps following it.
111        let v = sites_of(&view, sel, &snapshot, id)?;
112        v.get(idx as usize).or(v.first()).copied()
113    };
114    let mut updates: Vec<(usize, Xfrm, bool, bool)> = vec![];
115    for (i, s) in snapshot.iter().enumerate() {
116        let ShapeKind::Connector { start, end } = &s.kind else { continue };
117        if start.is_none() && end.is_none() {
118            continue;
119        }
120        let x = xfrm_of(&view, sel, s);
121        let (mut p0, mut p1) = endpoints(&x);
122        let a = start.and_then(|(id, k)| site(id, k));
123        let b = end.and_then(|(id, k)| site(id, k));
124        if let Some(q) = a {
125            p0 = q;
126        }
127        if let Some(q) = b {
128            p1 = q;
129        }
130        let nx = crate::engine::tools::line_xfrm(p0, p1, x.rot);
131        let changed = (nx.x - x.x).abs() > 1e-6
132            || (nx.y - x.y).abs() > 1e-6
133            || (nx.w - x.w).abs() > 1e-6
134            || (nx.h - x.h).abs() > 1e-6
135            || nx.flip_h != x.flip_h
136            || nx.flip_v != x.flip_v;
137        let drop_a = start.is_some() && a.is_none();
138        let drop_b = end.is_some() && b.is_none();
139        if changed || drop_a || drop_b {
140            updates.push((i, nx, drop_a, drop_b));
141        }
142    }
143    let Some(list) = crate::engine::shapes_mut(doc, sel) else { return };
144    for (i, nx, drop_a, drop_b) in updates {
145        if let Some(s) = list.get_mut(i) {
146            s.xfrm = Some(nx);
147            if let ShapeKind::Connector { start, end } = &mut s.kind {
148                if drop_a {
149                    *start = None;
150                }
151                if drop_b {
152                    *end = None;
153                }
154            }
155        }
156    }
157}

159/// Glue end `end` (0 = start, 1 = end) of line `id` to `to` (or unglue with `None`). Plain lines
160/// become connectors when glued.
161pub fn set_glue(list: &mut [Shape], id: ShapeId, end: u8, to: Option<(ShapeId, u32)>) {
162    let Some(s) = list.iter_mut().find(|s| s.id == id) else { return };
163    if !matches!(s.kind, ShapeKind::Connector { .. }) {
164        if to.is_none() || !matches!(s.kind, ShapeKind::Shape) {
165            return;
166        }
167        s.kind = ShapeKind::Connector { start: None, end: None };
168    }
169    if let ShapeKind::Connector { start, end: e } = &mut s.kind {
170        if end == 0 {
171            *start = to;
172        } else {
173            *e = to;
174        }
175    }
