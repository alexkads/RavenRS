5    /// End the freeform being drawn: create the shape (if it has at least two points).
6    pub fn finish_freeform(&mut self, closed: bool) -> Result<Value> {
7        let Some(Drag::Freeform { mode, points, .. }) = self.tool.drag.take() else { return Ok(Value::Null) };
8        self.tool.path_preview.clear();
9        self.tool.path_closing = false;
10        if !self.tool.sticky {
11            self.tool.kind = ToolKind::Select;
12        }
13        if points.len() < 2 {
14            return Ok(Value::Null);
15        }
16        let pts: Vec<[f64; 2]> = points.iter().map(|q| [q.x, q.y]).collect();
17        self.execute("shape.freeform", &json!({"points": pts, "closed": closed, "smooth": mode == "curve"}))
18    }

26    /// Show the sites of the shape near `p` and return the site an end at `p` would glue to.
27    pub(super) fn update_sites(&mut self, p: Point, tol: f64, exclude: Option<ShapeId>) -> Option<(ShapeId, u32, Point)> {
28        let (sites, glue) = match self.active() {
29            Some(st) => {
30                let shapes = st.shapes();
31                let near = crate::engine::connect::site_shape_at(&st.doc, &st.selection, shapes, p, tol * 3.0, exclude);
32                let sites = near
33                    .and_then(|id| shapes.iter().find(|s| s.id == id))
34                    .map(|s| crate::engine::connect::sites(&st.doc, &st.selection, s))
35                    .unwrap_or_default();
36                (sites, crate::engine::connect::nearest_site(&st.doc, &st.selection, shapes, p, (tol * 2.0).max(6.0), exclude))
37            }
38            None => (vec![], None),
39        };
40        self.tool.sites = sites;
41        self.tool.glue = glue.map(|g| g.2);
42        glue
43    }

45    pub(super) fn set_xfrm(&mut self, id: ShapeId, x: Xfrm) -> Result<()> {
46        // Rows scale from their drawn heights: a stored height is only a minimum (often 0).
47        let drawn = self.active().and_then(|st| match &st.shape(id)?.kind {
48            ShapeKind::Table(t) => Some(table_rows(st, t)),
49            _ => None,
50        });
51        self.edit(|doc, sel| {
52            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("tool", "no slide"))?;
53            if let Some(sh) = crate::model::find_shape_mut(list, id) {
54                sh.xfrm = Some(x);
55                if let ShapeKind::Table(t) = &mut sh.kind {
56                    // Tables resize their columns and rows proportionally.
57                    let heights: Vec<f64> = match &drawn {
58                        Some(d) if d.len() == t.rows.len() => d.clone(),
59                        _ => t.rows.iter().map(|r| r.height).collect(),
60                    };
61                    let (tw, th) = (t.width().max(1e-6), heights.iter().map(|h| h.max(0.0)).sum::<f64>());
62                    let kx = x.w / tw;
63                    t.cols.iter_mut().for_each(|c| *c *= kx);
64                    if th > 1e-6 {
65                        let ky = x.h / th;
66                        t.rows.iter_mut().zip(&heights).for_each(|(r, h)| r.height = h.max(0.0) * ky);
67                    }
68                }
69            }
70            Ok(())
71        })
72    }

74    pub(super) fn set_xfrm_cmd(&mut self, id: ShapeId, x: Xfrm) -> Result<Value> {
75        self.execute("shape.setBounds", &json!({"id": id, "x": x.x, "y": x.y, "w": x.w, "h": x.h}))?;
76        if x.flip_h || x.flip_v {
77            self.edit(|doc, sel| {
78                if let Some(sh) = crate::engine::shapes_mut(doc, sel).and_then(|l| crate::model::find_shape_mut(l, id))
79                    && let Some(xx) = sh.xfrm.as_mut()
80                {
81                    xx.flip_h = x.flip_h;
82                    xx.flip_v = x.flip_v;
83                }
84                Ok(())
85            })?;
86        }
87        Ok(Value::Null)
88    }

99    pub(super) fn execute_dup_in_place(&mut self, ids: &[ShapeId]) -> Result<()> {
100        self.edit(|doc, sel| {
101            let list = cmd::current_shapes(doc, sel).cloned().unwrap_or_default();
102            let mut copies: Vec<Shape> = list.iter().filter(|x| ids.contains(&x.id)).cloned().collect();
103            let boxes: Vec<Xfrm> = copies.iter().map(|c| xfrm_of(doc, sel, c)).collect();
104            cmd::edit::detach_on_slide(doc, sel, &mut copies);
105            cmd::edit::reid("edit.duplicate", doc, &mut copies)?;
106            for (c, b) in copies.iter_mut().zip(boxes) {
107                c.xfrm = Some(b);
108                c.ph = None;
109            }
110            let new_ids: Vec<ShapeId> = copies.iter().map(|c| c.id).collect();
111            let shapes = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("duplicate", "no slide"))?;
112            shapes.extend(copies);
113            sel.shapes = new_ids;
114            Ok(())
115        })
116    }

118    /// The drawing guide within `tol` of `p` (only while guides are shown).
119    pub fn guide_at(&self, p: Point, tol: f64) -> Option<usize> {
120        if !self.tool.guides_visible {
121            return None;
122        }
123        let st = self.active()?;
124        st.doc
125            .guides
126            .iter()
127            .enumerate()
128            .map(|(i, g)| (i, if g.vertical { (p.x - g.pos).abs() } else { (p.y - g.pos).abs() }))
129            .filter(|(_, d)| *d <= tol.max(2.0))
130            .min_by(|a, b| a.1.total_cmp(&b.1))
131            .map(|(i, _)| i)
132    }

134    /// Smart guides: snap `r` edges/centres to the slide and other shapes. Returns the offset to
135    /// apply and the guides to draw.
136    pub(super) fn snap_rect(&self, r: Rect, exclude: &[ShapeId], tol: f64, enabled: bool) -> (Vec2, Vec<Guide>) {
137        self.snap_edges(r, exclude, tol, enabled, [true; 3], [true; 3])
138    }

140    /// `snap_rect` looking only at some of `r`'s edges: `xs_on` is left, centre, right and `ys_on`
141    /// top, centre, bottom. Resizing passes just the edges being dragged, so a fixed edge that
142    /// already sits on a line (distance 0) doesn't win and leave the dragged edge unsnapped.
143    fn snap_edges(&self, r: Rect, exclude: &[ShapeId], tol: f64, enabled: bool, xs_on: [bool; 3], ys_on: [bool; 3]) -> (Vec2, Vec<Guide>) {
144        let Some(st) = self.active() else { return (Vec2::ZERO, vec![]) };
145        if !enabled || !self.prefs.smart_guides {
146            return (Vec2::ZERO, vec![]);
147        }
148        let size = st.doc.slide_size;
149        let mut xs: Vec<(f64, Rect)> = vec![
150            (0.0, Rect::new(0.0, 0.0, 0.0, size.height)),
151            (size.width / 2.0, Rect::new(size.width / 2.0, 0.0, size.width / 2.0, size.height)),
152            (size.width, Rect::new(size.width, 0.0, size.width, size.height)),
153        ];
154        let mut ys: Vec<(f64, Rect)> = vec![
155            (0.0, Rect::new(0.0, 0.0, size.width, 0.0)),
156            (size.height / 2.0, Rect::new(0.0, size.height / 2.0, size.width, size.height / 2.0)),
157            (size.height, Rect::new(0.0, size.height, size.width, size.height)),
158        ];
159        if self.tool.guides_visible {
160            for g in &st.doc.guides {
161                if g.vertical {
162                    xs.push((g.pos, Rect::new(g.pos, 0.0, g.pos, size.height)));
163                } else {
164                    ys.push((g.pos, Rect::new(0.0, g.pos, size.width, g.pos)));
165                }
166            }
167        }
168        for sh in st.shapes() {
169            if exclude.contains(&sh.id) || sh.hidden {
170                continue;
171            }
172            let b = xfrm_of(&st.doc, &st.selection, sh).bounds();
173            for x in [b.x0, b.center().x, b.x1] {
174                xs.push((x, b));
175            }
176            for y in [b.y0, b.center().y, b.y1] {
177                ys.push((y, b));
178            }
179        }
180        let mut best_x: Option<(f64, f64, Rect)> = None;
181        for edge in [r.x0, r.center().x, r.x1].into_iter().zip(xs_on).filter_map(|(e, on)| on.then_some(e)) {
182            for (x, b) in &xs {
183                let d = x - edge;
184                if d.abs() <= tol && best_x.is_none_or(|(bd, _, _)| d.abs() < bd.abs()) {
185                    best_x = Some((d, *x, *b));
186                }
187            }
188        }
189        let mut best_y: Option<(f64, f64, Rect)> = None;
190        for edge in [r.y0, r.center().y, r.y1].into_iter().zip(ys_on).filter_map(|(e, on)| on.then_some(e)) {
191            for (y, b) in &ys {
192                let d = y - edge;
193                if d.abs() <= tol && best_y.is_none_or(|(bd, _, _)| d.abs() < bd.abs()) {
194                    best_y = Some((d, *y, *b));
195                }
196            }
197        }
198        let mut off = Vec2::ZERO;
199        let mut guides = vec![];
200        if let Some((d, x, b)) = best_x {
201            off.x = d;
202            let rr = r + Vec2::new(d, 0.0);
203            guides.push(Guide { vertical: true, pos: x, from: b.y0.min(rr.y0), to: b.y1.max(rr.y1) });
204        } else if self.prefs.snap_to_grid && xs_on.contains(&true) {
205            let at = if xs_on[0] {
206                r.x0
207            } else if xs_on[2] {
208                r.x1
209            } else {
210                r.center().x
211            };
212            off.x = crate::geom::snap(at, self.prefs.grid_spacing) - at;
213        }
214        if let Some((d, y, b)) = best_y {
215            off.y = d;
216            let rr = r + Vec2::new(0.0, d);
217            guides.push(Guide { vertical: false, pos: y, from: b.x0.min(rr.x0), to: b.x1.max(rr.x1) });
218        } else if self.prefs.snap_to_grid && ys_on.contains(&true) {
219            let at = if ys_on[0] {
220                r.y0
221            } else if ys_on[2] {
222                r.y1
223            } else {
224                r.center().y
225            };
226            off.y = crate::geom::snap(at, self.prefs.grid_spacing) - at;
227        }
228        (off, guides)
229    }

231    /// Snap the edges that resize handle `handle` drags in the resized box `x` to the smart guides.
232    pub(super) fn snap_resize(&mut self, x: Xfrm, handle: u8, exclude: &[ShapeId], tol: f64) -> Xfrm {
233        if x.rot != 0.0 {
234            self.tool.guides.clear();
235            return x;
236        }
237        let r = x.rect();
238        // Only the dragged edges move, so only they look for a line to snap to. The handles sit in
239        // the box's unflipped frame: in a flipped box, also one that a drag past the opposite edge
240        // just flipped, the dragged handle is on the other side.
241        let (left, right) = (matches!(handle, 0 | 6 | 7), matches!(handle, 2..=4));
242        let (top, bottom) = (matches!(handle, 0..=2), matches!(handle, 4..=6));
243        let (moves_left, moves_right) = if x.flip_h { (right, left) } else { (left, right) };
244        let (moves_top, moves_bottom) = if x.flip_v { (bottom, top) } else { (top, bottom) };
245        let (off, guides) = self.snap_edges(r, exclude, tol * 1.5, true, [moves_left, false, moves_right], [moves_top, false, moves_bottom]);
246        let mut nx = x;
247        if moves_left {
248            nx.x += off.x;
249            nx.w -= off.x;
250        } else if moves_right {
251            nx.w += off.x;
252        }
253        if moves_top {
254            nx.y += off.y;
255            nx.h -= off.y;
256        } else if moves_bottom {
257            nx.h += off.y;
258        }
259        nx.w = nx.w.max(0.0);
260        nx.h = nx.h.max(0.0);
261        self.tool.guides = guides;
262        nx
263    }

274    pub(super) fn add_ink(&mut self, points: Vec<(f64, f64, f32)>) -> Result<Value> {
275        let ToolKind::Ink { mode, color, width } = self.tool.kind.clone() else { return Ok(Value::Null) };
276        let stroke = crate::model::InkStroke { points, color, width, highlighter: mode == "highlighter" };
277        // Strokes join the slide's ink layer (one ink shape per slide).
278        let existing = self.doc()?.shapes().iter().rev().find(|s| matches!(s.kind, ShapeKind::Ink { .. })).map(|s| s.id);
279        self.edit(|doc, sel| {
280            let id = match existing {
281                Some(id) => id,
282                None => {
283                    let id = ShapeId(doc.alloc_id().ok_or_else(|| cmd::bad("ink.draw", cmd::NO_ID_SPACE))?);
284                    let size = doc.slide_size;
285                    let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("ink", "no slide"))?;
286                    list.push(Shape {
287                        id,
288                        name: "Ink".into(),
289                        xfrm: Some(Xfrm::new(0.0, 0.0, size.width, size.height)),
290                        kind: ShapeKind::Ink { strokes: vec![] },
291                        ..Default::default()
292                    });
293                    id
294                }
295            };
296            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("ink", "no slide"))?;
297            if let Some(sh) = crate::model::find_shape_mut(list, id)
298                && let ShapeKind::Ink { strokes } = &mut sh.kind
299            {
300                strokes.push(stroke);
301            }
302            Ok(json!({"ink": id}))
303        })
304    }

306    pub(super) fn erase_at(&mut self, p: Point, tol: f64) -> Result<Value> {
307        let st = self.doc()?;
308        let Some(ink) = st.shapes().iter().rev().find(|s| matches!(s.kind, ShapeKind::Ink { .. })).map(|s| s.id) else { return Ok(Value::Null) };
309        let before = st.doc.clone();
310        let sel_before = st.selection.clone();
311        self.edit(|doc, sel| {
312            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("ink", "no slide"))?;
313            if let Some(sh) = crate::model::find_shape_mut(list, ink)
314                && let ShapeKind::Ink { strokes } = &mut sh.kind
315            {
316                strokes.retain(|s| !s.points.iter().any(|(x, y, _)| (Point::new(*x, *y) - p).hypot() <= tol * 2.0 + s.width));
317            }
318            Ok(())
319        })?;
320        let vertex_kind = self.tool.vertex_kind.clone();
321        if let Some(st) = self.active_mut() {
322            crate::engine::push_undo(st, HistoryEntry { label: "Erase".into(), doc: before, selection: sel_before, vertex_kind });
323        }
324        Ok(Value::Null)
325    }
