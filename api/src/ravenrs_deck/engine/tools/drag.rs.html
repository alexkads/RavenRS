18impl Session {
19    /// Update the active drag interaction using the pointer position and modifiers.
20    pub(super) fn drag(&mut self, p: Point, mods: Mods, tol: f64) -> Result<Value> {
21        if matches!(self.tool.drag, Some(Drag::Ink { .. })) {
22            self.extend_ink(p);
23            return Ok(Value::Null);
24        }
25        let Some(drag) = self.tool.drag.clone() else { return Ok(Value::Null) };
26        match drag {
27            Drag::Guide { index } => {
28                let vertical = self.active().and_then(|d| d.doc.guides.get(index)).is_some_and(|g| g.vertical);
29                let pos = if vertical { p.x } else { p.y };
30                // Guides move in whole points unless snapping is off, as in PowerPoint.
31                let pos = if self.prefs.snap_to_grid && !mods.alt { pos.round() } else { pos };
32                self.tool.guide_drag = Some((index, pos));
33            }
34            Drag::Move { ids, start, origs, moved, duplicate, enter_text } => {
35                return self.drag_move(MoveDrag { ids, start, origs, moved, duplicate, enter_text }, p, mods, tol);
36            }
37            Drag::Resize { id, orig, ids, handle, origs, start } => {
38                let keep_aspect = mods.shift || self.is_picture(id);
39                let nx = resize_box(&orig, handle, p - start, keep_aspect, mods.alt);
40                let nx = self.snap_resize(nx, handle, &ids, tol);
41                let sx = if orig.w > 1e-9 { (nx.w / orig.w) * if nx.flip_h != orig.flip_h { -1.0 } else { 1.0 } - 1.0 } else { 0.0 };
42                let sy = if orig.h > 1e-9 { (nx.h / orig.h) * if nx.flip_v != orig.flip_v { -1.0 } else { 1.0 } - 1.0 } else { 0.0 };
43                let div = if mods.alt { 2.0 } else { 1.0 };
44                let hx = match handle {
45                    0 | 6 | 7 => -1.0,
46                    2..=4 => 1.0,
47                    _ => 0.0,
48                };
49                let hy = match handle {
50                    0..=2 => -1.0,
51                    4..=6 => 1.0,
52                    _ => 0.0,
53                };
54                for (tid, o) in &origs {
55                    let target_nx = if *tid == id {
56                        nx
57                    } else {
58                        let picture = self.is_picture(*tid);
59                        let (sx, sy) = if picture && matches!(handle, 0 | 2 | 4 | 6) {
60                            // Use the axis with the larger relative change so an unchanged
61                            // dimension cannot cancel shrinking. Preserve each axis's flip.
62                            let scale_x = 1.0 + sx;
63                            let scale_y = 1.0 + sy;
64                            let scale = if sx.abs() >= sy.abs() { scale_x.abs() } else { scale_y.abs() };
65                            (scale.copysign(scale_x) - 1.0, scale.copysign(scale_y) - 1.0)
66                        } else {
67                            (sx, sy)
68                        };
69
70                        let lp = Affine::rotate(o.rot.to_radians())
71                            * Point::new(
72                                hx * sx * o.w / div * if o.flip_h { -1.0 } else { 1.0 },
73                                hy * sy * o.h / div * if o.flip_v { -1.0 } else { 1.0 },
74                            );
75                        resize_box(o, handle, Vec2::new(lp.x, lp.y), picture, mods.alt)
76                    };
77                    self.set_xfrm(*tid, target_nx)?;
78                }
79            }
80            Drag::LineEnd { id, end, orig } => {
81                let a = orig.affine();
82                let (p0, p1) = (a * Point::new(0.0, 0.0), a * Point::new(orig.w, orig.h));
83                let mut q = p;
84                let other = if end == 0 { p1 } else { p0 };
85                if mods.shift {
86                    q = constrain_45(other, q);
87                }
88                let glue = self.update_sites(q, tol, Some(id));
89                if let Some((_, _, at)) = glue {
90                    q = at;
91                }
92                let (s0, s1) = if end == 0 { (q, p1) } else { (p0, q) };
93                self.set_xfrm(id, line_xfrm(s0, s1, orig.rot))?;
94                self.edit(|doc, sel| {
95                    if let Some(list) = crate::engine::shapes_mut(doc, sel) {
96                        crate::engine::connect::set_glue(list, id, end, glue.map(|g| (g.0, g.1)));
97                    }
98                    Ok(())
99                })?;
100            }
101            Drag::Rotate { ids, center, start, origs } => {
102                let ang = (p.y - center.y).atan2(p.x - center.x);
103                let mut delta = (ang - start).to_degrees();
104                for (id, o) in &origs {
105                    let mut r = o.rot + delta;
106                    if mods.shift {
107                        r = (r / 15.0).round() * 15.0;
108                        delta = r - o.rot;
109                    }
110                    let mut nx = *o;
111                    nx.rot = ((r % 360.0) + 360.0) % 360.0;
112                    if ids.len() > 1 {
113                        // Rotate positions around the common centre too.
114                        let c = o.center();
115                        let rc = Affine::translate(center.to_vec2()) * Affine::rotate(delta.to_radians()) * Affine::translate(-center.to_vec2()) * c;
116                        nx.x = rc.x - o.w / 2.0;
117                        nx.y = rc.y - o.h / 2.0;
118                    }
119                    self.set_xfrm(*id, nx)?;
120                }
121            }
122            Drag::Adjust { id, index, orig } => return self.drag_adjust(id, index, orig, p),
123            Drag::Vertex { id, path, index } => {
124                self.tool.drag = Some(Drag::Vertex { id, path, index });
125                self.edit_vertex(id, path, index, p)?;
126                return Ok(Value::Null);
127            }
128            Drag::Handle { id, path, index, out } => {
129                self.tool.drag = Some(Drag::Handle { id, path, index, out });
130                let side = if out { "out" } else { "in" };
131                // Shift, Ctrl or Alt held while dragging a handle makes the point smooth, straight or
132                // corner, as in PowerPoint; Cmd and Option count as Ctrl and Alt.
133                let held = if mods.shift {
134                    Some("smooth")
135                } else if mods.cmd {
136                    Some("straight")
137                } else if mods.alt {
138                    Some("corner")
139                } else {
140                    None
141                };
142                if let Err(e) = self.execute_inner_move_handle(id, path, index, side, p, held) {
143                    log::debug!("edit handle: {e}");
144                }
145                return Ok(Value::Null);
146            }
147            Drag::Warp { id, index, name, base, orig } => {
148                let v = {
149                    let st = self.doc()?;
150                    let Some(sh) = st.shape(id) else { return Ok(Value::Null) };
151                    let tr = crate::render::shape_geometry(sh, orig.w, orig.h).text_rect;
152                    let hs = crate::geom::warp::handles(&name, &base, tr);
153                    let Some(h) = hs.iter().find(|h| h.index == index) else { return Ok(Value::Null) };
154                    h.value_at(orig.affine().inverse() * p)
155                };
156                self.edit(|doc, sel| {
157                    let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("adjust", "no slide"))?;
158                    if let Some(sh) = crate::model::find_shape_mut(list, id) {
159                        set_warp_adj(sh, &name, &base, index, v);
160                    }
161                    Ok(())
162                })?;
163            }
164            Drag::Marquee { start, .. } => {
165                self.tool.marquee = Some(Rect::from_points(start, p));
166            }
167            Drag::Create { start } => self.drag_create(start, p, mods, tol),
168            Drag::TextSelect { shape, press, .. } => return self.drag_text_select(shape, press, p),
169            Drag::TableBorder { id, column, index, start, size, frame, moved } => {
170                let local = frame.inverse() * p;
171                let d = if column { local.x - start.x } else { local.y - start.y };
172                if !moved && d.abs() < tol.max(1.0) {
173                    return Ok(Value::Null);
174                }
175                if !moved {
176                    self.begin(if column { "Column Width" } else { "Row Height" });
177                }
178                // An inner column border moves between its two columns (the table keeps its
179                // width); the right edge widens the last column and the table. A row's height is
180                // a minimum: rows still grow to fit their text.
181                if column {
182                    let width = (size + d).max(cmd::table::MIN_COLUMN_WIDTH);
183                    self.execute("table.columnWidth", &json!({"id": id, "col": index, "width": width, "keepTotal": true}))?;
184                } else {
185                    self.execute("table.rowHeight", &json!({"id": id, "row": index, "height": (size + d).max(1.0)}))?;
186                }
187                self.tool.drag = Some(Drag::TableBorder { id, column, index, start, size, frame, moved: true });
188            }
189            Drag::Freeform { mode, mut points, pressed } => {
190                // Dragging scribbles (except for curves, which only take clicked points).
191                if pressed && mode != "curve" && points.len() < MAX_GESTURE_POINTS && points.last().is_none_or(|l| (*l - p).hypot() >= 1.5) {
192                    points.push(p);
193                }
194                let mut v = points.clone();
195                if mode == "curve" {
196                    v.push(p);
197                }
198                self.tool.path_preview = v;
199                self.tool.drag = Some(Drag::Freeform { mode, points, pressed });
200            }
201            // Handled above.
202            Drag::Ink { .. } => {}
203        }
204        Ok(Value::Null)
205    }

207    /// Ink grows by one sample per move: extend it (and its preview) in place rather than
208    /// copying the whole stroke twice per event, up to [`MAX_GESTURE_POINTS`].
209    fn extend_ink(&mut self, p: Point) {
210        let sample = self.ink_sample(p);
211        if let Some(Drag::Ink { points }) = self.tool.drag.as_mut()
212            && points.len() < MAX_GESTURE_POINTS
213        {
214            points.push(sample);
215            self.tool.ink_preview.push(sample);
216        }
217    }

219    fn drag_move(&mut self, drag: MoveDrag, p: Point, mods: Mods, tol: f64) -> Result<Value> {
220        let MoveDrag { ids, start, origs, moved, duplicate, enter_text } = drag;
221        let mut d = p - start;
222        if !moved && (d.hypot() < tol.max(1.0) || origs.is_empty()) {
223            return Ok(Value::Null);
224        }
225        if !moved {
226            self.begin(if duplicate { "Duplicate" } else { "Move" });
227            if duplicate {
228                // Alt-drag: drag a copy, leave the original.
229                self.execute_dup_in_place(&ids)?;
230            }
231        }
232        let ids_now = if duplicate { self.doc()?.selection.shapes.clone() } else { ids.clone() };
233        if mods.shift {
234            if d.x.abs() > d.y.abs() {
235                d.y = 0.0;
236            } else {
237                d.x = 0.0;
238            }
239        }
240        // Smart guides on the moved group's bounds.
241        let bounds = origs.iter().map(|(_, x)| x.bounds()).reduce(|a, b| a.union(b)).unwrap_or_default();
242        let moved_b = bounds + d;
243        let (snap, guides) = self.snap_rect(moved_b, &ids_now, tol * 1.5, !mods.cmd);
244        d += snap;
245        self.tool.guides = guides;
246        let origs2: Vec<(ShapeId, Xfrm)> = if duplicate { ids_now.iter().copied().zip(origs.iter().map(|o| o.1)).collect() } else { origs.clone() };
247        self.edit(|doc, sel| {
248            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("move", "no slide"))?;
249            for (id, o) in &origs2 {
250                if let Some(sh) = crate::model::find_shape_mut(list, *id) {
251                    sh.xfrm = Some(Xfrm { x: o.x + d.x, y: o.y + d.y, ..*o });
252                }
253            }
254            // A connector dragged away from shapes that stay put comes unglued at those ends.
255            let mut moved: Vec<ShapeId> = vec![];
256            for (id, _) in &origs2 {
257                if let Some(sh) = crate::model::find_shape(list, *id) {
258                    crate::model::walk(std::slice::from_ref(sh), &mut |s, _| moved.push(s.id));
259                }
260            }
261            for sh in list.iter_mut().filter(|s| moved.contains(&s.id)) {
262                if let ShapeKind::Connector { start, end } = &mut sh.kind {
263                    for c in [start, end] {
264                        if c.is_some_and(|(to, _)| !moved.contains(&to)) {
265                            *c = None;
266                        }
267                    }
268                }
269            }
270            Ok(())
271        })?;
272        self.tool.drag =
273            Some(Drag::Move { ids: ids_now, start, origs: origs2, moved: true, duplicate: false, enter_text: enter_text.filter(|_| false) });
274        Ok(Value::Null)
275    }

277    fn drag_adjust(&mut self, id: ShapeId, index: usize, orig: Xfrm, p: Point) -> Result<Value> {
278        let st = self.doc()?;
279        let Some(sh) = st.shape(id).cloned() else { return Ok(Value::Null) };
280        let local = orig.affine().inverse() * p;
281        let geo = crate::render::shape_geometry(&sh, orig.w, orig.h);
282        let Some(h) = geo.handles.iter().find(|h| h.index == index) else { return Ok(Value::Null) };
283        let free2d = matches!(sh.geom.preset_name(), Some(n) if n.contains("Callout") && !n.contains("Arrow"));
284        let vals: Vec<(usize, f64)> = if free2d {
285            let dx = (local.x - orig.w / 2.0) / orig.w.max(1e-6) * 1e5;
286            let dy = (local.y - orig.h / 2.0) / orig.h.max(1e-6) * 1e5;
287            vec![(0, dx.round()), (1, dy.round())]
288        } else {
289            vec![(index, h.value_at(local))]
290        };
291        self.edit(|doc, sel| {
292            let list = crate::engine::shapes_mut(doc, sel).ok_or_else(|| cmd::bad("adjust", "no slide"))?;
293            if let Some(sh) = crate::model::find_shape_mut(list, id)
294                && let crate::model::Geom::Preset { name, adj } = &mut sh.geom
295            {
296                let defaults = crate::geom::preset::info(name).map(|x| x.defaults).unwrap_or(&[]);
297                for (i, v) in &vals {
298                    while adj.len() <= *i {
299                        let k = adj.len();
300                        adj.push(defaults.get(k).copied().unwrap_or(0.0));
301                    }
302                    if let Some(slot) = adj.get_mut(*i) {
303                        *slot = *v;
304                    }
305                }
306            }
307            Ok(())
308        })?;
309        Ok(Value::Null)
310    }

312    fn drag_create(&mut self, start: Point, p: Point, mods: Mods, tol: f64) {
313        let line = matches!(&self.tool.kind, ToolKind::Shape { preset } if crate::geom::preset::is_line_like(preset));
314        let x = if line {
315            let mut q = if mods.shift { constrain_45(start, p) } else { p };
316            if let Some((_, _, at)) = self.update_sites(q, tol, None) {
317                q = at;
318            }
319            let s0 = self.glue_at(start, tol).map(|g| g.2).unwrap_or(start);
320            line_xfrm(s0, q, 0.0)
321        } else if matches!(&self.tool.kind, ToolKind::Shape { preset } if preset == DRAW_TABLE)
322            && crate::engine::cmd::table::table_at(self, &Value::Null, (start.x, start.y)).is_some()
323        {
324            // The pencil inside a table draws a straight border, across or down.
325            let q = if (p.x - start.x).abs() >= (p.y - start.y).abs() { Point::new(p.x, start.y) } else { Point::new(start.x, p.y) };
326            let r = Rect::from_points(start, q);
327            Xfrm::new(r.x0, r.y0, r.width(), r.height())
328        } else {
329            let mut r = Rect::from_points(start, p);
330            if mods.shift {
331                let s = r.width().max(r.height());
332                let sx = if p.x < start.x { -s } else { s };
333                let sy = if p.y < start.y { -s } else { s };
334                r = Rect::from_points(start, start + Vec2::new(sx, sy));
335            }
336            if mods.alt {
337                let d = p - start;
338                r = Rect::new(start.x - d.x.abs(), start.y - d.y.abs(), start.x + d.x.abs(), start.y + d.y.abs());
339            }
340            Xfrm::new(r.x0, r.y0, r.width(), r.height())
341        };
342        self.tool.preview = Some(x);
343    }

345    fn drag_text_select(&mut self, shape: ShapeId, press: Option<TextSel>, p: Point) -> Result<Value> {
346        // In a table, leaving the cell the drag started in selects whole cells, as in
347        // PowerPoint; coming back to that cell selects text in it again.
348        let st = self.doc()?;
349        if let Some(start) = press.as_ref().and_then(|t| t.cell)
350            && let Some(f) = st.shape(shape).and_then(|sh| cmd::table::table_frame(st, sh))
351            && let Some(cell) = f.cell_at(f.local(p))
352        {
353            if cell != start {
354                self.select(|_, sel| {
355                    sel.text = None;
356                    sel.shapes = vec![shape];
357                    sel.cells = Some((shape, start, cell));
358                })?;
359                return Ok(Value::Null);
360            }
361            if st.selection.text.is_none() {
362                self.select(|_, sel| {
363                    sel.cells = None;
364                    sel.text = press;
365                })?;
366            }
367        }
368        let st = self.doc()?;
369        if let Some(t) = st.selection.text.as_ref().filter(|t| t.shape == shape) {
370            let pos = cmd::text::text_pos_at(st, t, p);
371            self.select(|_, sel| {
372                if let Some(t) = sel.text.as_mut() {
373                    t.caret = pos;
374                }
375            })?;
376        }
377        Ok(Value::Null)
378    }

380    /// Moves a vertex during a drag (one undo step for the whole drag).
381    fn edit_vertex(&mut self, id: ShapeId, path: usize, index: usize, p: Point) -> Result<()> {
382        let r = cmd::points::move_point_raw(self, id, path, index, p);
383        if let Err(e) = r {
384            log::debug!("edit point: {e}");
385        }
386        Ok(())
387    }

389    /// Moves a control handle during a drag. `held` is the point type set by the modifier held
390    /// during the drag; the point keeps that type for later drags, as after `shape.pointType`.
391    fn execute_inner_move_handle(&mut self, id: ShapeId, path: usize, index: usize, side: &str, p: Point, held: Option<&str>) -> Result<Value> {
392        let kind = match held {
393            Some(k) => Some(k.to_string()),
394            None => self.tool.vertex_kind.clone().filter(|(s, pp, i, _)| (*s, *pp, *i) == (id, path, index)).map(|k| k.3),
395        };
396        let r =
397            cmd::points::move_handle_raw(self, &json!({"id": id.0, "path": path, "index": index, "side": side, "x": p.x, "y": p.y, "type": kind}))?;
398        if let Some(k) = held {
399            self.tool.vertex_kind = Some((id, path, index, k.to_string()));
400        }
401        Ok(r)
402    }
