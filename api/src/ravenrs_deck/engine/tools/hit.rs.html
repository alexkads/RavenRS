7/// The topmost text-holding shape under `p` inside group `sh`, recursing into nested groups.
8/// Group children live in the group's child space (`ShapeKind::Group::child`), so the point is
9/// mapped through `group_child_affine`, exactly as `hit_shape` does for group hit-testing.
10pub(super) fn text_member_at(st: &crate::engine::DocState, sh: &Shape, p: Point, tol: f64) -> Option<ShapeId> {
11    let ShapeKind::Group { children, child } = &sh.kind else { return None };
12    let x = xfrm_of(&st.doc, &st.selection, sh);
13    let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
14    member_at(children, crate::geom::group_child_affine(&x, ch), p, tol)
15}

17/// Depth-first (topmost-first) search of `children` for a text-holding shape under `p`.
18/// `to_abs` maps this level's child space to slide space.
19fn member_at(children: &[Shape], to_abs: Affine, p: Point, tol: f64) -> Option<ShapeId> {
20    let q = to_abs.inverse() * p;
21    for c in children.iter().rev() {
22        let Some(cx) = c.xfrm else { continue };
23        let local = cx.affine().inverse() * q;
24        if local.x < -tol || local.y < -tol || local.x > cx.w + tol || local.y > cx.h + tol {
25            continue;
26        }
27        if let ShapeKind::Group { children: gc, child: gch } = &c.kind {
28            let gch = Rect::new(gch.x, gch.y, gch.x + gch.w, gch.y + gch.h);
29            let nested = to_abs * crate::geom::group_child_affine(&cx, gch);
30            if let Some(id) = member_at(gc, nested, p, tol) {
31                return Some(id);
32            }
33        } else if matches!(c.kind, ShapeKind::Shape | ShapeKind::Connector { .. }) && !c.is_line() {
34            return Some(c.id);
35        }
36    }
37    None
38}

48/// Whether a visible shape drawn in front of `target` is under `p`: the shapes above it in its
49/// list, topmost first, and inside a group that holds it, the members above it.
50pub(super) fn shape_in_front(st: &crate::engine::DocState, shapes: &[Shape], target: ShapeId, p: Point, tol: f64, depth: usize) -> bool {
51    if depth > 64 {
52        return false;
53    }
54    for s in shapes.iter().rev() {
55        if s.id == target {
56            return false;
57        }
58        if crate::model::find_shape(std::slice::from_ref(s), target).is_some() {
59            // The group holding the text: look among its members.
60            return shape_in_front(st, s.children(), target, p, tol, depth + 1);
61        }
62        if !s.hidden && hit_shape(st, s, p, tol).is_some() {
63            return true;
64        }
65    }
66    false
67}

80pub(super) enum EditZone {
81    /// Well inside: the caret goes there.
82    Inside,
83    /// On the frame: a band `tol` wide on either side of the box edge.
84    Frame,
85    Outside,
86}

88/// Classify slide point `p` against box `x`, in the box's own (rotated, flipped) frame. Inside
89/// small boxes the band narrows to a quarter of the box so their text stays reachable.
90pub(super) fn edit_zone(x: &Xfrm, p: Point, tol: f64) -> EditZone {
91    let l = x.affine().inverse() * p;
92    let (w, h) = (x.w.max(0.0), x.h.max(0.0));
93    let tol = if tol.is_finite() { tol.max(0.0) } else { 0.0 };
94    if !(l.x.is_finite() && l.y.is_finite()) || l.x < -tol || l.y < -tol || l.x > w + tol || l.y > h + tol {
95        return EditZone::Outside;
96    }
97    let inner = tol.min(w / 4.0).min(h / 4.0);
98    if l.x > inner && l.y > inner && l.x < w - inner && l.y < h - inner { EditZone::Inside } else { EditZone::Frame }
99}

105pub(super) fn constrain_45(a: Point, b: Point) -> Point {
106    let d = b - a;
107    let ang = d.y.atan2(d.x);
108    let snapped = (ang / std::f64::consts::FRAC_PI_4).round() * std::f64::consts::FRAC_PI_4;
109    let len = d.hypot();
110    a + Vec2::new(snapped.cos(), snapped.sin()) * len
111}

119/// New box when dragging resize handle `h` by `d` (slide space), keeping the shape's rotation.
120pub fn resize_box(o: &Xfrm, h: u8, d: Vec2, keep_aspect: bool, from_center: bool) -> Xfrm {
121    // Work in the shape's rotated frame.
122    let rot = Affine::rotate(-o.rot.to_radians());
123    let ld = rot * Point::new(d.x, d.y);
124    let (mut l, mut t, mut r, mut b) = (0.0, 0.0, o.w, o.h);
125    let (sx, sy) = (if o.flip_h { -1.0 } else { 1.0 }, if o.flip_v { -1.0 } else { 1.0 });
126    let (dx, dy) = (ld.x * sx, ld.y * sy);
127    match h {
128        0 => {
129            l += dx;
130            t += dy;
131        }
132        1 => t += dy,
133        2 => {
134            r += dx;
135            t += dy;
136        }
137        3 => r += dx,
138        4 => {
139            r += dx;
140            b += dy;
141        }
142        5 => b += dy,
143        6 => {
144            l += dx;
145            b += dy;
146        }
147        _ => l += dx,
148    }
149    if from_center {
150        // Symmetric growth about the centre.
151        let gw = match h {
152            0 | 6 | 7 => -dx,
153            2..=4 => dx,
154            _ => 0.0,
155        };
156        let gh = match h {
157            0..=2 => -dy,
158            4..=6 => dy,
159            _ => 0.0,
160        };
161        l = -gw;
162        r = o.w + gw;
163        t = -gh;
164        b = o.h + gh;
165    }
166    let mut w = r - l;
167    let mut hh = b - t;
168    if keep_aspect && o.w > 1e-9 && o.h > 1e-9 && !matches!(h, 1 | 3 | 5 | 7) {
169        let ar = o.w / o.h;
170        if (w / ar).abs() > hh.abs() {
171            hh = w.abs() / ar * hh.signum().max(if hh == 0.0 { 1.0 } else { hh.signum() });
172        } else {
173            w = hh.abs() * ar * w.signum().max(if w == 0.0 { 1.0 } else { w.signum() });
174        }
175        if matches!(h, 0 | 6) {
176            l = r - w;
177        } else {
178            r = l + w;
179        }
180        if matches!(h, 0 | 2) {
181            t = b - hh;
182        } else {
183            b = t + hh;
184        }
185        if from_center {
186            let (cx, cy) = (o.w / 2.0, o.h / 2.0);
187            l = cx - w / 2.0;
188            r = cx + w / 2.0;
189            t = cy - hh / 2.0;
190            b = cy + hh / 2.0;
191        }
192    }
193    // Dragging past the opposite edge flips.
194    let (mut fh, mut fv) = (o.flip_h, o.flip_v);
195    if r < l {
196        std::mem::swap(&mut l, &mut r);
197        fh = !fh;
198    }
199    if b < t {
200        std::mem::swap(&mut t, &mut b);
201        fv = !fv;
202    }
203    let (nw, nh) = (r - l, b - t);
204    // New centre in slide space: old centre + rotated offset of the local centre change.
205    let local_c = Point::new((l + r) / 2.0 * sx + if sx < 0.0 { o.w } else { 0.0 }, (t + b) / 2.0 * sy + if sy < 0.0 { o.h } else { 0.0 });
206    let c = o.affine() * Point::new(if o.flip_h { o.w - local_c.x } else { local_c.x }, if o.flip_v { o.h - local_c.y } else { local_c.y });
207    Xfrm { x: c.x - nw / 2.0, y: c.y - nh / 2.0, w: nw, h: nh, rot: o.rot, flip_h: fh, flip_v: fv }
208}

210/// Whether `p` lies in a cell of table `id`, not in the grab band around it (`cell_at` would take
211/// the nearest cell).
212pub(super) fn in_table_cell(st: &crate::engine::DocState, id: ShapeId, p: Point) -> bool {
213    st.shape(id).and_then(|sh| cmd::table::table_frame(st, sh)).is_some_and(|f| {
214        let q = f.local(p);
215        (0.0..=f.width()).contains(&q.x) && (0.0..=f.height()).contains(&q.y) && f.cell_at(q).is_some()
216    })
217}

219/// Resolve a shape's target at slide point `p`, using its effective fill for interior hits.
220pub(super) fn hit_shape(st: &crate::engine::DocState, sh: &Shape, p: Point, tol: f64) -> Option<Hit> {
221    let x = xfrm_of(&st.doc, &st.selection, sh);
222    let inv = x.affine().inverse();
223    let local = inv * p;
224    // Table borders resize rows and columns.
225    let near = local.x >= -tol && local.y >= -tol && local.x <= x.w + tol && local.y <= x.h + tol;
226    if near && let Some(f) = cmd::table::table_frame(st, sh) {
227        match f.border_at(f.local(p), tol) {
228            Some(cmd::table::TableBorder::Column(col)) => return Some(Hit::TableColumn { id: sh.id, col }),
229            Some(cmd::table::TableBorder::Row(row)) => return Some(Hit::TableRow { id: sh.id, row }),
230            None => {}
231        }
232    }
233    match &sh.kind {
234        ShapeKind::Group { children, child } => {
235            // A group is hit when any child is hit; the group is what gets selected.
236            let ch = Rect::new(child.x, child.y, child.x + child.w, child.y + child.h);
237            let m = crate::geom::group_child_affine(&x, ch);
238            let q = m.inverse() * p;
239            for c in children.iter().rev() {
240                if let Some(cx) = c.xfrm {
241                    let cl = cx.affine().inverse() * q;
242                    if cl.x >= -tol && cl.y >= -tol && cl.x <= cx.w + tol && cl.y <= cx.h + tol {
243                        return Some(Hit::Shape { id: sh.id, text: false });
244                    }
245                }
246            }
247            None
248        }
249        ShapeKind::Ink { strokes } => {
250            for s in strokes {
251                if s.points.iter().any(|(px, py, _)| (Point::new(*px, *py) - p).hypot() <= tol + s.width) {
252                    return Some(Hit::Shape { id: sh.id, text: false });
253                }
254            }
255            None
256        }
257        _ => {
258            if sh.is_line() {
259                let geo = crate::render::shape_geometry(sh, x.w, x.h);
260                let lw = sh.line.as_ref().and_then(|l| l.width).unwrap_or(1.0);
261                if crate::geom::near_path(&geo.outline(), local, tol.max(lw)) {
262                    return Some(Hit::Shape { id: sh.id, text: false });
263                }
264                return None;
265            }
266            let inside = local.x >= -tol && local.y >= -tol && local.x <= x.w + tol && local.y <= x.h + tol;
267            if !inside {
268                return None;
269            }
270            let has_text = matches!(sh.kind, ShapeKind::Shape) && (sh.text.as_ref().is_some_and(|t| !t.is_empty()) || sh.ph.is_some() || sh.text_box);
271            // Match rendering: an explicit No Fill overrides the style, and style/theme fills
272            // can themselves be No Fill. Empty, text-less interiors then let clicks through.
273            let fill =
274                cmd::text::ctx_for(&st.doc, &st.selection).map(|ctx| crate::model::resolve::fill(&ctx, sh).0).unwrap_or_else(|| sh.fill.clone());
275            let filled = !matches!(fill, None | Some(crate::model::Fill::None));
276            let geo = crate::render::shape_geometry(sh, x.w, x.h);
277            let in_geo = {
278                use crate::geom::KurboShape;
279                geo.outline().winding(local) != 0 || crate::geom::near_path(&geo.outline(), local, tol)
280            };
281            if has_text
282                || sh.ph.is_some()
283                || matches!(
284                    sh.kind,
285                    ShapeKind::Picture { .. }
286                        | ShapeKind::Table(_)
287                        | ShapeKind::Chart(_)
288                        | ShapeKind::Media(_)
289                        | ShapeKind::Model3d { .. }
290                        | ShapeKind::Opaque { .. }
291                )
292                || (filled && in_geo)
293                || crate::geom::near_path(&geo.outline(), local, tol)
294            {
295                return Some(Hit::Shape { id: sh.id, text: has_text || matches!(sh.kind, ShapeKind::Shape) });
296            }
297            None
298        }
299    }
300}
