15    /// What's under `p` on the current slide.
16    pub fn hit_test(&self, p: Point, tol: f64) -> Option<Hit> {
17        let st = self.active()?;
18        // In Edit Points, the edited shape's vertices come first.
19        if let Some(id) = self.tool.editing_points(st)
20            && let Some(sh) = st.shape(id)
21        {
22            if let Some((path, index)) = self.tool.edit_vertex
23                && let Some((hin, hout, _)) = cmd::points::slide_handles(&st.doc, &st.selection, sh, path, index)
24            {
25                for (h, out) in [(hout, true), (hin, false)] {
26                    if h.is_some_and(|q| (q - p).hypot() <= tol * 1.6) {
27                        return Some(Hit::Handle { id, path, index, out });
28                    }
29                }
30            }
31            if let Some((path, index)) = cmd::points::vertex_at(&st.doc, &st.selection, sh, p, tol * 1.6) {
32                return Some(Hit::Vertex { id, path, index });
33            }
34        }
35        // Handles of selected shapes first.
36        if st.selection.text.is_none() || st.selection.shapes.len() == 1 {
37            for id in st.selection.shapes.iter().rev() {
38                let Some(sh) = st.shape(*id) else { continue };
39                // A locked object shows no live handles: it can't be sized, rotated or reshaped.
40                if sh.locked {
41                    continue;
42                }
43                let x = xfrm_of(&st.doc, &st.selection, sh);
44                if sh.is_line() {
45                    let a = x.affine();
46                    for (end, q) in [(0u8, a * Point::new(0.0, 0.0)), (1u8, a * Point::new(x.w, x.h))] {
47                        if (q - p).hypot() <= tol * 1.6 {
48                            return Some(Hit::LineEnd { id: *id, end });
49                        }
50                    }
51                    continue;
52                }
53                let geo = crate::render::shape_geometry(sh, x.w, x.h);
54                for hd in &geo.handles {
55                    if (x.affine() * hd.pos - p).hypot() <= tol * 1.6 {
56                        return Some(Hit::Adjust { id: *id, index: hd.index });
57                    }
58                }
59                for hd in wordart_handles(&st.doc, &st.selection, sh, x.w, x.h) {
60                    if (x.affine() * hd.pos - p).hypot() <= tol * 1.6 {
61                        return Some(Hit::Warp { id: *id, index: hd.index });
62                    }
63                }
64                let (pts, rot) = box_handles(&x);
65                if (rot - p).hypot() <= tol * 1.8 && st.selection.text.is_none() {
66                    return Some(Hit::Rotate { id: *id });
67                }
68                for (i, q) in pts.iter().enumerate() {
69                    if (*q - p).hypot() <= tol * 1.6 {
70                        return Some(Hit::Resize { id: *id, handle: i as u8 });
71                    }
72                }
73            }
74        }
75        // The frame of the text being edited: grabbing it moves the box, unless a shape in front
76        // of the text covers that spot (#931).
77        if let Some(e) = edited_box(st)
78            && e.movable
79            && edit_zone(&e.xfrm, p, tol) == EditZone::Frame
80            && !shape_in_front(st, st.shapes(), e.id, p, tol, 0)
81        {
82            return Some(Hit::Frame { id: e.id });
83        }
84        // Shapes, topmost first.
85        // Locked objects are still picked (to format them or type in them); they just don't move.
86        for sh in st.shapes().iter().rev() {
87            if sh.hidden {
88                continue;
89            }
90            if let Some(h) = hit_shape(st, sh, p, tol) {
91                return Some(h);
92            }
93        }
94        None
95    }

97    /// Feed a pointer event to the active tool. Returns what happened (for agents).
98    pub fn pointer(&mut self, ev: PointerEvent) -> Result<Value> {
99        let tol = if ev.tol > 0.0 { ev.tol } else { 4.0 };
100        let p = Point::new(ev.x, ev.y);
101        self.tool.pressure = ev.pressure.filter(|v| v.is_finite()).map(|v| v.clamp(0.0, 1.0));
102        if !(p.x.is_finite() && p.y.is_finite()) {
103            return Ok(Value::Null);
104        }
105        // A press (not a hover) settles an input method's composition, as clicking does in PowerPoint.
106        if !matches!(ev.kind, PointerKind::Move) {
107            self.settle_composition();
108        }
109        match ev.kind {
110            PointerKind::Move => {
111                if let Some(Drag::Freeform { points, mode, .. }) = &self.tool.drag {
112                    let closing = points.len() > 2 && points.first().is_some_and(|s| (*s - p).hypot() <= tol * 2.0);
113                    let mut v = points.clone();
114                    v.push(p);
115                    self.tool.path_preview = v;
116                    self.tool.path_smooth = mode == "curve";
117                    self.tool.path_closing = closing;
118                    return Ok(Value::Null);
119                }
120                self.tool.hover = self.hit_test(p, tol);
121                self.tool.guide_hover =
122                    if self.tool.hover.is_none() && matches!(self.tool.kind, ToolKind::Select) { self.guide_at(p, tol) } else { None };
123                let line_tool = matches!(&self.tool.kind, ToolKind::Shape { preset } if crate::geom::preset::is_line_like(preset));
124                if line_tool {
125                    self.update_sites(p, tol, None);
126                } else {
127                    self.tool.sites.clear();
128                    self.tool.glue = None;
129                }
130                Ok(Value::Null)
131            }
132            PointerKind::Down => self.down(p, ev.mods, tol),
133            PointerKind::Drag => self.drag(p, ev.mods, tol),
134            PointerKind::Up => self.up(p, ev.mods),
135            PointerKind::DoubleClick => {
136                let before = self.text_selection();
137                let r = self.double_click(p, tol)?;
138                self.paint_if_selection_changed(before, r)
139            }
140            PointerKind::ContextClick => self.context_click(p, tol),
141            PointerKind::TripleClick => {
142                if self.active().is_some_and(|d| d.selection.text.is_some()) {
143                    let before = self.text_selection();
144                    let r = self.execute("text.selectParagraph", &json!({}))?;
145                    self.paint_if_selection_changed(before, r)
146                } else {
147                    Ok(Value::Null)
148                }
149            }
150        }
151    }

174    pub(crate) fn begin(&mut self, label: &str) {
175        let vertex_kind = self.tool.vertex_kind.clone();
176        if let Some(st) = self.active_mut()
177            && st.interaction.is_none()
178        {
179            st.interaction =
180                Some(crate::engine::Interaction { label: label.into(), doc: st.doc.clone(), selection: st.selection.clone(), vertex_kind });

184    fn commit(&mut self) {
185        if let Some(st) = self.active_mut()
186            && let Some(i) = st.interaction.take()
187            && !Arc::ptr_eq(&i.doc, &st.doc)
188        {
189            crate::engine::push_undo(st, HistoryEntry { label: i.label, doc: i.doc, selection: i.selection, vertex_kind: i.vertex_kind });
190        }

193    /// Cancels a point or handle drag in Edit Points when Esc is pressed: ends the drag, puts back the
194    /// document and selection saved at the press, and adds no undo step. Returns whether a drag was cancelled.
195    pub(crate) fn cancel_point_drag(&mut self) -> bool {
196        if !matches!(self.tool.drag, Some(Drag::Vertex { .. } | Drag::Handle { .. })) {
197            return false;
198        }
199        self.tool.drag = None;
200        if let Some(st) = self.active_mut()
201            && st.interaction.as_ref().is_some_and(|i| i.label == "Edit Points")
202            && let Some(i) = st.interaction.take()
203        {
204            // A modifier held during the drag may have set a point type; the drag is undone, so is that.
205            let kind = i.vertex_kind;
206            if !Arc::ptr_eq(&i.doc, &st.doc) {
207                st.doc = i.doc;
208                st.selection = i.selection;
209                st.revision += 1;
210            }
211            self.tool.vertex_kind = kind;
212        }
213        true
214    }

224    /// Handle a pointer press and start the applicable tool interaction.
225    fn down(&mut self, p: Point, mods: Mods, tol: f64) -> Result<Value> {
226        if let Some(Drag::Freeform { mode, mut points, .. }) = self.tool.drag.clone() {
227            // Next click of a freeform: close on the start point, else add a vertex.
228            if points.len() > 2 && points.first().is_some_and(|s| (*s - p).hypot() <= tol * 2.0) {
229                self.tool.drag = Some(Drag::Freeform { mode, points, pressed: false });
230                return self.finish_freeform(true);
231            }
232            self.tool.press_added_vertex = points.len() < super::MAX_GESTURE_POINTS;
233            if self.tool.press_added_vertex {
234                points.push(p);
235            }
236            self.tool.path_preview = points.clone();
237            self.tool.drag = Some(Drag::Freeform { mode, points, pressed: true });
238            return Ok(Value::Null);
239        }
240        match self.tool.kind.clone() {
241            ToolKind::Shape { preset } if is_freeform_tool(&preset) => {
242                self.tool.drag = Some(Drag::Freeform { mode: preset.clone(), points: vec![p], pressed: true });
243                self.tool.path_preview = vec![p];
244                self.tool.path_smooth = preset == "curve";
245                self.tool.path_closing = false;
246                return Ok(Value::Null);
247            }
248            ToolKind::Shape { preset } if preset == TABLE_ERASER => {
249                // Click a border to merge the cells on either side; elsewhere does nothing.
250                return Ok(self.execute("table.eraseLine", &json!({"at": [p.x, p.y], "tol": tol.max(4.0)})).unwrap_or(Value::Null));
251            }
252            ToolKind::Shape { .. } | ToolKind::TextBox => {
253                self.tool.drag = Some(Drag::Create { start: p });
254                self.tool.preview = Some(Xfrm::new(p.x, p.y, 0.0, 0.0));
255                return Ok(Value::Null);
256            }
257            ToolKind::Ink { mode, .. } => {
258                if mode == "eraser" {
259                    return self.erase_at(p, tol);
260                }
261                let sample = self.ink_sample(p);
262                self.tool.drag = Some(Drag::Ink { points: vec![sample] });
263                self.tool.ink_preview = vec![sample];
264                return Ok(Value::Null);
265            }
266            ToolKind::Select => {}
267        }
268        // Clicking inside the text being edited moves the caret / starts a text selection.
269        let before = self.text_selection();
270        if let Some((shape, pos)) = self.place_caret(p, tol, mods.shift)? {
271            let press = self.doc()?.selection.text.clone();
272            self.tool.drag = Some(Drag::TextSelect { shape, before, press });
273            return Ok(json!({"caret": [pos.0, pos.1]}));
274        }
275        let hit = self.hit_test(p, tol);
276        // Edit Points: drag a vertex, Ctrl-click one to delete it or the outline to add one;
277        // a click anywhere else leaves Edit Points.
278        if let Some(eid) = self.doc().ok().and_then(|st| self.tool.editing_points(st)) {
279            if let Some(Hit::Vertex { id, path, index }) = hit {
280                if mods.cmd {
281                    return self.execute("shape.deletePoint", &json!({"id": id, "path": path, "index": index}));
282                }
283                self.tool.edit_vertex = Some((path, index));
284                self.begin("Edit Points");
285                self.tool.drag = Some(Drag::Vertex { id, path, index });
286                return Ok(json!({"vertex": [path, index]}));
287            }
288            if let Some(Hit::Handle { id, path, index, out }) = hit {
289                self.begin("Edit Points");
290                self.tool.drag = Some(Drag::Handle { id, path, index, out });
291                return Ok(json!({"handle": [path, index]}));
292            }
293            let st = self.doc()?;
294            if let Some(sh) = st.shape(eid)
295                && let Some((path, seg, t)) = cmd::points::segment_at(&st.doc, &st.selection, sh, p, tol * 1.6)
296            {
297                if mods.cmd {
298                    return self.execute("shape.addPoint", &json!({"id": eid, "path": path, "segment": seg, "t": t}));
299                }
300                return Ok(Value::Null);
301            }
302            self.tool.edit_points = None;
303            self.tool.edit_vertex = None;
304        }
305        let st = self.doc()?;
306        match hit {
307            Some(Hit::TableColumn { id, col }) => self.begin_table_border(id, true, col, p)?,
308            Some(Hit::TableRow { id, row }) => self.begin_table_border(id, false, row, p)?,
309            Some(Hit::Resize { id, handle }) => {
310                let orig = st.shape(id).map(|s| xfrm_of(&st.doc, &st.selection, s)).unwrap_or_default();
311                let ids: Vec<ShapeId> = if st.selection.shapes.contains(&id) { st.selection.shapes.clone() } else { vec![id] };
312                let origs: Vec<(ShapeId, Xfrm)> =
313                    ids.iter().filter_map(|i| st.shape(*i).filter(|s| !s.locked).map(|s| (*i, xfrm_of(&st.doc, &st.selection, s)))).collect();
314                let ids: Vec<ShapeId> = origs.iter().map(|(id, _)| *id).collect();
315                self.begin("Size");
316                self.tool.drag = Some(Drag::Resize { id, orig, ids, handle, origs, start: p });
317            }
318            Some(Hit::LineEnd { id, end }) => {
319                let orig = st.shape(id).map(|s| xfrm_of(&st.doc, &st.selection, s)).unwrap_or_default();
320                self.begin("Size");
321                self.tool.drag = Some(Drag::LineEnd { id, end, orig });
322            }
323            Some(Hit::Rotate { id }) => {
324                let ids: Vec<ShapeId> = if st.selection.shapes.contains(&id) { st.selection.shapes.clone() } else { vec![id] };
325                let origs: Vec<(ShapeId, Xfrm)> = ids.iter().filter_map(|i| st.shape(*i).map(|s| (*i, xfrm_of(&st.doc, &st.selection, s)))).collect();
326                let center = st.shape(id).map(|s| xfrm_of(&st.doc, &st.selection, s).center()).unwrap_or(p);
327                let start = (p.y - center.y).atan2(p.x - center.x);
328                self.begin("Rotate");
329                self.tool.drag = Some(Drag::Rotate { ids, center, start, origs });
330            }
331            // Only reached outside Edit Points (handled above).
332            Some(Hit::Vertex { .. } | Hit::Handle { .. }) => {}
333            Some(Hit::Adjust { id, index }) => {
334                let orig = st.shape(id).map(|s| xfrm_of(&st.doc, &st.selection, s)).unwrap_or_default();
335                self.begin("Adjust Shape");
336                self.tool.drag = Some(Drag::Adjust { id, index, orig });
337            }
338            Some(Hit::Frame { id }) => {
339                // Like PowerPoint: the box leaves text editing and is selected whole (a click
340                // without a drag stops there), and dragging moves it as one undo step.
341                let origs: Vec<(ShapeId, Xfrm)> = st.shape(id).map(|s| (id, xfrm_of(&st.doc, &st.selection, s))).into_iter().collect();
342                self.select(|_, sel| {
343                    sel.shapes = vec![id];
344                    sel.text = None;
345                    sel.cells = None;
346                })?;
347                self.tool.drag = Some(Drag::Move { ids: vec![id], start: p, origs, moved: false, duplicate: mods.alt, enter_text: None });
348            }
349            Some(Hit::Warp { id, index }) => {
350                let orig = st.shape(id).map(|s| xfrm_of(&st.doc, &st.selection, s)).unwrap_or_default();
351                let found = st.shape(id).and_then(|sh| {
352                    let ctx = crate::engine::cmd::text::ctx_for(&st.doc, &st.selection)?;
353                    let bp = crate::model::resolve::body(&ctx, sh);
354                    bp.warp.map(|n| (n, bp.warp_adj))
355                });
356                if let Some((name, base)) = found {
357                    self.begin("Adjust Decorative Text");
358                    self.tool.drag = Some(Drag::Warp { id, index, name, base, orig });
359                }
360            }
361            Some(Hit::Shape { id, .. })
362                if !(mods.shift || mods.cmd)
363                    && st.selection.cells.is_some_and(|c| c.0 == id && st.selection.shapes == [id])
364                    && in_table_cell(st, id, p) =>
365            {
366                // A press in a table whose cells are selected starts again in the cell under the
367                // pointer: a click puts the caret there and a drag selects another range. A press
368                // on the table outside its cells selects and moves the table like any shape.
369                let before = self.text_selection();
370                if self.edit_text_at(id, p)? {
371                    self.select(|_, sel| sel.cells = None)?;
372                    let press = self.doc()?.selection.text.clone();
373                    self.tool.drag = Some(Drag::TextSelect { shape: id, before, press });
374                }
375            }
376            Some(Hit::Shape { id, text }) => {
377                let selected = st.selection.shapes.contains(&id);
378                let toggle = mods.shift || mods.cmd;
379                let ids: Vec<ShapeId> = if toggle {
380                    let mut v = st.selection.shapes.clone();
381                    if selected {
382                        v.retain(|x| *x != id);
383                    } else {
384                        v.push(id);
385                    }
386                    v
387                } else if selected {
388                    st.selection.shapes.clone()
389                } else {
390                    vec![id]
391                };
392                // Locked objects stay put when the selection is dragged.
393                let origs: Vec<(ShapeId, Xfrm)> =
394                    ids.iter().filter_map(|i| st.shape(*i).filter(|s| !s.locked).map(|s| (*i, xfrm_of(&st.doc, &st.selection, s)))).collect();
395                // A click (no drag) on text puts the caret there; unselected text boxes and
396                // placeholders take the caret straight away like PowerPoint. A click in a cell of a
397                // selected table puts the caret there; a click on its grab band keeps the table selected.
398                let table = st.shape(id).is_some_and(|s| matches!(s.kind, ShapeKind::Table(_)));
399                // A click on the frame band of a text box selects the box whole, as PowerPoint
400                // does, so it can be moved, duplicated or deleted; only the inside takes the caret.
401                let on_frame = text
402                    && is_text_first(st, id)
403                    && st.shape(id).is_some_and(|s| edit_zone(&xfrm_of(&st.doc, &st.selection, s), p, tol) == EditZone::Frame);
404                let enter_text =
405                    (!toggle && !on_frame && ((text && (selected || is_text_first(st, id))) || (table && selected && in_table_cell(st, id, p))))
406                        .then_some((id, p));
407                let ids2 = ids.clone();
408                self.select(|_, sel| {
409                    sel.shapes = ids2;
410                    sel.text = None;
411                    sel.cells = None;
412                })?;
413                let duplicate = mods.alt && origs.len() == ids.len();
414                self.tool.drag = Some(Drag::Move { ids, start: p, origs, moved: false, duplicate, enter_text });
415                // Format Painter applies on click.
416                if self.painter.is_some() && !toggle {
417                    let _ = self.execute("format.painterApply", &json!({"ids": [id]}));
418                }
419                // Animation Painter applies on click.
420                if self.anim_painter.is_some() && !toggle {
421                    let _ = self.execute("animation.painterApply", &json!({"ids": [id]}));
422                }
423            }
424            None if self.guide_at(p, tol).is_some() => {
425                // Away from every shape a press on a guide picks the guide up.
426                if let Some(index) = self.guide_at(p, tol) {
427                    let pos = st.doc.guides.get(index).map(|g| g.pos).unwrap_or(0.0);
428                    self.tool.drag = Some(Drag::Guide { index });
429                    self.tool.guide_drag = Some((index, pos));
430                }
431            }
432            None => {
433                let add = mods.shift || mods.cmd;
434                self.select(|_, sel| {
435                    sel.text = None;
436                    sel.cells = None;
437                    if !add {
438                        sel.shapes.clear();
439                    }
440                })?;
441                self.tool.drag = Some(Drag::Marquee { start: p, add });
442                self.tool.marquee = Some(Rect::from_points(p, p));
443            }
444        }
445        Ok(json!({"hit": serde_json::to_value(&hit).unwrap_or_default()}))
446    }

448    /// Handle a pointer release and finalize the active tool interaction.
449    fn up(&mut self, p: Point, mods: Mods) -> Result<Value> {
450        let drag = self.tool.drag.take();
451        self.tool.guides.clear();
452        self.tool.sites.clear();
453        self.tool.glue = None;
454        let mut out = Value::Null;
455        match drag {
456            Some(Drag::Guide { index }) => {
457                // Dropped off the slide: the guide is deleted.
458                let (_, pos) = self.tool.guide_drag.take().unwrap_or((index, 0.0));
459                let size = self.doc()?.doc.slide_size;
460                let vertical = self.doc()?.doc.guides.get(index).is_some_and(|g| g.vertical);
461                let limit = if vertical { size.width } else { size.height };
462                out = if pos < 0.0 || pos > limit {
463                    self.execute("guide.delete", &json!({"index": index}))?
464                } else {
465                    self.execute("guide.move", &json!({"index": index, "pos": pos}))?
466                };
467            }
468            Some(Drag::Move { moved: false, enter_text: Some((id, at)), .. })
469                // Plain click on text: caret at the click (in a table, in the clicked cell).
470                if self.edit_text_at(id, at)? => {
471                    out = json!({"editing": id});
472                }
473            Some(Drag::Marquee { start, add }) => {
474                let r = Rect::from_points(start, p);
475                self.tool.marquee = None;
476                if r.width() > 1.0 || r.height() > 1.0 {
477                    let st = self.doc()?;
478                    let ids: Vec<ShapeId> = st
479                        .shapes()
480                        .iter()
481                        .filter(|s| !s.hidden && !s.locked)
482                        .filter(|s| contains_rect(r, xfrm_of(&st.doc, &st.selection, s).bounds()))
483                        .map(|s| s.id)
484                        .collect();
485                    self.select(|_, sel| {
486                        if add {
487                            for i in &ids {
488                                if !sel.shapes.contains(i) {
489                                    sel.shapes.push(*i);
490                                }
491                            }
492                        } else {
493                            sel.shapes = ids.clone();
494                        }
495                    })?;
496                    out = json!({"selected": ids});
497                }
498            }
499            Some(Drag::Create { start }) => {
500                let kind = self.tool.kind.clone();
501                let preview = self.tool.preview.take();
502                let dragged = (p - start).hypot() > 2.0;
503                let res = match &kind {
504                    ToolKind::Shape { preset } if preset == DRAW_TABLE => {
505                        // Inside a table: split cells along the line. Elsewhere: a one-cell table.
506                        if crate::engine::cmd::table::table_at(self, &Value::Null, (start.x, start.y)).is_some() {
507                            self.execute("table.drawLine", &json!({"from": [start.x, start.y], "to": [p.x, p.y]})).unwrap_or(Value::Null)
508                        } else if let Some(x) = preview.filter(|x| dragged && x.w >= 8.0 && x.h >= 8.0) {
509                            self.execute("insert.table", &json!({"rows": 1, "cols": 1, "rect": [x.x, x.y, x.w, x.h]}))?
510                        } else {
511                            Value::Null
512                        }
513                    }
514                    ToolKind::Shape { preset } => {
515                        let line = crate::geom::preset::is_line_like(preset);
516                        let rect = match preview.filter(|_| dragged) {
517                            Some(x) => x,
518                            None if line => Xfrm::new(start.x, start.y, 72.0, 0.0),
519                            None => Xfrm::new(start.x, start.y, 72.0, 72.0),
520                        };
521                        let r = self.execute("shape.insert", &json!({"preset": preset, "rect": [rect.x, rect.y, rect.w, rect.h], "flip": format!("{}{}", if rect.flip_h { "h" } else { "" }, if rect.flip_v { "v" } else { "" })}))?;
522                        if let (Some(id), Some(x)) = (r.get("id").and_then(Value::as_u64), preview.filter(|_| dragged && line)) {
523                            let id = ShapeId(id as u32);
524                            let _ = self.set_xfrm_cmd(id, x);
525                            // Glue the ends that landed on connection sites.
526                            let (p0, p1) = crate::engine::connect::endpoints(&x);
527                            let (g0, g1) = (self.glue_at(p0, 4.0), self.glue_at(p1, 4.0));
528                            if g0.is_some() || g1.is_some() {
529                                let _ = self.edit(|doc, sel| {
530                                    if let Some(list) = crate::engine::shapes_mut(doc, sel) {
531                                        crate::engine::connect::set_glue(list, id, 0, g0.map(|g| (g.0, g.1)));
532                                        crate::engine::connect::set_glue(list, id, 1, g1.map(|g| (g.0, g.1)));
533                                    }
534                                    Ok(())
535                                });
536                            }
537                        }
538                        r
539                    }
540                    ToolKind::TextBox => {
541                        let w = preview.filter(|_| dragged).map(|x| x.w).unwrap_or(0.0);
542                        let at = preview.filter(|_| dragged).map(|x| Point::new(x.x, x.y)).unwrap_or(start);
543                        self.execute("insert.textBox", &json!({"rect": [at.x, at.y, w, 30.0]}))?
544                    }
545                    _ => Value::Null,
546                };
547                // The table pencil stays on until you pick another tool, as in PowerPoint.
548                if !self.tool.sticky && !matches!(&kind, ToolKind::Shape { preset } if preset == DRAW_TABLE) {
549                    self.tool.kind = ToolKind::Select;
550                }
551                out = res;
552            }
553            Some(Drag::Freeform { mode, points, .. }) => {
554                if mode == "scribble" {
555                    // A scribble ending back at its start closes into a filled shape.
556                    let closed = points.len() > 3 && points.first().zip(points.last()).is_some_and(|(a, b)| (*a - *b).hypot() <= 6.0);
557                    self.tool.drag = Some(Drag::Freeform { mode, points, pressed: false });
558                    return self.finish_freeform(closed);
559                }
560                // Polygon / curve: keep going until double-click, Enter or a click on the start.
561                self.tool.drag = Some(Drag::Freeform { mode, points, pressed: false });
562                return Ok(Value::Null);
563            }
564            Some(Drag::Ink { points }) => {
565                self.tool.ink_preview.clear();
566                if points.len() > 1 {
567                    self.begin("Ink");
568                    out = self.add_ink(points)?;
569                }
570            }
571            Some(Drag::Resize { origs, .. }) => {
572                let changed: Vec<ShapeId> = origs
573                    .iter()
574                    .filter_map(|(id, orig)| {
575                        let current = self.active().and_then(|st| st.shape(*id).map(|s| xfrm_of(&st.doc, &st.selection, s)))?;
576
577                        (current != *orig).then_some(*id)
578                    })
579                    .collect();
580                if !changed.is_empty() {
581                    for id in changed {
582                        let _ = cmd::text::fit_text_box(self, id);
583                        self.refit_after_resize(id)?;
584                    }
585                }
586                self.commit();
587            }
588            Some(Drag::TextSelect { before, .. }) => {
589                out = self.paint_if_selection_changed(before, out)?;
590                self.commit();
591            }
592            Some(_) => self.commit(),
593            None => {}
594        }
595        // Any open interaction (move etc.) ends here.
596        self.commit();
597        let _ = mods;
598        Ok(out)
599    }
600
601    pub(super) fn is_picture(&self, id: ShapeId) -> bool {
602        self.active().and_then(|d| d.shape(id)).is_some_and(|s| matches!(s.kind, ShapeKind::Picture { .. } | ShapeKind::Media(_)))
603    }
