1//! Review › Compare: the differences between the open presentation and a revised copy of it, slide
2//! by slide, and accepting a revised slide into the open one or rejecting it.
3//!
4//! Slides match by id (a revised copy of the same deck keeps them); when the two decks share no
5//! slide id at all, by position. Shapes match by id, then by name.
6
7use std::collections::{HashMap, HashSet};
8use std::sync::Arc;

14/// The revised presentation a document is being compared with.
15#[derive(Clone, Debug)]
16pub struct Compare {
17    pub name: String,
18    pub revised: Arc<Presentation>,
19    /// Slides match by position (the decks shared no slide id when the compare started); fixed
20    /// for the session so accepting slides doesn't switch the matching half way.
21    pub by_position: bool,
22    /// Revisions rejected: no longer listed and never accepted, the open deck left as it is.
23    pub rejected: HashSet<ChangeKey>,
24}
25
26/// What a revision is about, stable while slides move: the open slide (for changed and deleted)
27/// and the revised slide (for changed and inserted).
28#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
29pub struct ChangeKey {
30    kind: ChangeKind,
31    open: Option<SlideId>,
32    revised: Option<usize>,
33}
34
35impl ChangeKey {
36    pub fn of(doc: &Presentation, change: &Change) -> ChangeKey {
37        let open = match change.kind {
38            ChangeKind::Inserted => None,
39            ChangeKind::Changed | ChangeKind::Deleted => doc.slides.get(change.slide).map(|s| s.id),
40        };
41        ChangeKey { kind: change.kind, open, revised: change.revised }
42    }
43}
44
45/// The revisions still to review: [`diff`] without the rejected ones.
46pub fn pending(doc: &Presentation, c: &Compare) -> Vec<Change> {
47    let mut out = diff(doc, &c.revised, c.by_position);
48    out.retain(|ch| !is_rejected(doc, c, ch));
49    out
50}
51
52/// Whether `ch` was rejected. Positional matching re-pairs slides by index when the open deck gains
53/// or loses a slide, which changes the open half of the key; the revised index stays put, so any
54/// rejected key for the same revised slide counts. An open slide pushed past the end of the revised
55/// deck turns into a deletion; it stays kept when a rejected key names it.
56fn is_rejected(doc: &Presentation, c: &Compare, ch: &Change) -> bool {
57    if c.rejected.contains(&ChangeKey::of(doc, ch)) {
58        return true;
59    }
60    if !c.by_position {
61        return false;
62    }
63    match ch.kind {
64        ChangeKind::Deleted => {
65            let open = doc.slides.get(ch.slide).map(|s| s.id);
66            open.is_some() && c.rejected.iter().any(|k| k.open == open)
67        }
68        _ => ch.revised.is_some() && c.rejected.iter().any(|k| k.revised == ch.revised),
69    }
70}

72/// Whether `cur` and `rev` share no slide id (then slides match by position).
73pub fn unrelated(cur: &Presentation, rev: &Presentation) -> bool {
74    let ids: HashSet<SlideId> = rev.slides.iter().map(|s| s.id).collect();
75    !cur.slides.iter().any(|s| ids.contains(&s.id))
76}

78#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize)]
79#[serde(rename_all = "camelCase")]
80pub enum ChangeKind {
81    /// Only in the revised deck.
82    Inserted,
83    /// Only in the open deck.
84    Deleted,
85    /// In both, different.
86    Changed,
87}

89/// One slide-level revision.
90#[derive(Clone, Debug, PartialEq, Serialize)]
91#[serde(rename_all = "camelCase")]
92pub struct Change {
93    pub kind: ChangeKind,
94    /// Index in the open deck (for inserted: where it would go).
95    pub slide: usize,
96    /// The editor's number of that slide (`slide` + 1).
97    pub number: usize,
98    /// Which side has it alone: `deck` (the open presentation; accepting deletes it) or `file`
99    /// (the compared file; accepting adds it). Absent when both have it and differ.
100    #[serde(skip_serializing_if = "Option::is_none")]
101    pub only_in: Option<&'static str>,
102    /// Index in the revised deck (none for deleted).
103    pub revised: Option<usize>,
104    /// What differs, readable: `Title 1: text`, `Picture 3 inserted`, `notes`…
105    pub details: Vec<String>,
106}

110/// Slide pairs (open index, revised index).
111fn pairs(cur: &Presentation, rev: &Presentation, by_position: bool) -> Vec<(usize, usize)> {
112    if by_position {
113        return (0..cur.slides.len().min(rev.slides.len())).map(|i| (i, i)).collect();
114    }
115    let rev_ids: HashMap<SlideId, usize> = rev.slides.iter().enumerate().map(|(i, s)| (s.id, i)).collect();
116    cur.slides.iter().enumerate().filter_map(|(i, s)| rev_ids.get(&s.id).map(|j| (i, *j))).collect()
117}

119pub fn diff(cur: &Presentation, rev: &Presentation, by_position: bool) -> Vec<Change> {
120    let pairs = pairs(cur, rev, by_position);
121    let cur_matched: HashSet<usize> = pairs.iter().map(|p| p.0).collect();
122    let rev_matched: HashSet<usize> = pairs.iter().map(|p| p.1).collect();
123    let mut out = vec![];
124    let mut media = Media::new(cur, rev);
125    for (i, j) in &pairs {
126        let (Some(a), Some(b)) = (cur.slides.get(*i), rev.slides.get(*j)) else { continue };
127        let details = slide_details(cur, a, rev, b, &mut media);
128        if !details.is_empty() {
129            out.push(Change { kind: ChangeKind::Changed, slide: *i, number: i.saturating_add(1), only_in: None, revised: Some(*j), details });
130        }
131    }
132    for (i, s) in cur.slides.iter().enumerate() {
133        if !cur_matched.contains(&i) {
134            out.push(Change {
135                kind: ChangeKind::Deleted,
136                slide: i,
137                number: i.saturating_add(1),
138                only_in: Some("deck"),
139                revised: None,
140                details: vec![slide_title(s)],
141            });
142        }
143    }
144    // Each revised slide's open match (the first pair naming it).
145    let mut rev_to_cur: HashMap<usize, usize> = HashMap::new();
146    for p in &pairs {
147        rev_to_cur.entry(p.1).or_insert(p.0);
148    }
149    // After the open slide that matches the nearest matched revised slide before it.
150    let mut at = 0;
151    for (j, s) in rev.slides.iter().enumerate() {
152        if let Some(i) = rev_to_cur.get(&j) {
153            at = i + 1;
154        }
155        if !rev_matched.contains(&j) {
156            out.push(Change {
157                kind: ChangeKind::Inserted,
158                slide: at,
159                number: at.saturating_add(1),
160                only_in: Some("file"),
161                revised: Some(j),
162                details: vec![slide_title(s)],
163            });
164        }
165    }
166    out.sort_by_key(|c| (c.slide, c.revised.unwrap_or(0)));
167    out
168}

170fn slide_title(s: &Slide) -> String {
171    let t = s
172        .shapes
173        .iter()
174        .find(|x| x.ph.as_ref().is_some_and(|p| matches!(p.kind, crate::model::PhType::Title | crate::model::PhType::CtrTitle)))
175        .and_then(|x| x.text.as_ref())
176        .map(|b| b.text())
177        .unwrap_or_default();
178    let t = t.trim();
179    if t.is_empty() { "Untitled slide".into() } else { t.chars().take(60).collect() }
180}

186/// Both decks' media bytes by id, and which picture pairs were already compared: a deck can
187/// show one picture many times, and its bytes are compared once per pair.
188struct Media<'a> {
189    cur: HashMap<MediaId, &'a [u8]>,
190    rev: HashMap<MediaId, &'a [u8]>,
191    same: HashMap<(MediaId, MediaId), bool>,
192}
193
194impl<'a> Media<'a> {
195    fn new(cur: &'a Presentation, rev: &'a Presentation) -> Self {
196        // The first item with an id wins, as in `Presentation::media`.
197        let index = |p: &'a Presentation| {
198            let mut m = HashMap::new();
199            for x in &p.media {
200                m.entry(x.id).or_insert(x.data.as_slice());
201            }
202            m
203        };
204        Media { cur: index(cur), rev: index(rev), same: HashMap::new() }
205    }
206    /// Whether the pictures (by media id, `None` for a shape that is not a picture) differ.
207    fn differ(&mut self, a: Option<MediaId>, b: Option<MediaId>) -> bool {
208        let (ba, bb) = (a.and_then(|x| self.cur.get(&x)), b.and_then(|y| self.rev.get(&y)));
209        match (a, b) {
210            (Some(x), Some(y)) if ba.is_some() && bb.is_some() => !*self.same.entry((x, y)).or_insert_with(|| ba == bb),
211            _ => ba.is_some() || bb.is_some(),
212        }
213    }
214}

216/// A shape without its ids (shape and media), for comparing across decks.
217fn plain_shape(s: &Shape) -> Shape {
218    let mut c = s.clone();
219    fn rec(shapes: &mut [Shape]) {
220        for s in shapes {
221            s.id = ShapeId(0);
222            if let Some(ch) = s.children_mut() {
223                rec(ch);
224            }
225        }
226    }
227    rec(std::slice::from_mut(&mut c));
228    c
229}

231/// The box a shape is drawn with: its own, else the one it inherits from its layout or master
232/// placeholder (`ctx`).
233fn drawn_box(ctx: Option<&Ctx>, s: &Shape) -> Option<crate::model::Xfrm> {
234    s.xfrm.or_else(|| ctx.map(|c| crate::model::resolve::xfrm(c, s)))
235}

237/// The rotation and flips a shape is drawn with (none without a box).
238fn turn(ctx: Option<&Ctx>, s: &Shape) -> (f64, bool, bool) {
239    drawn_box(ctx, s).map_or((0.0, false, false), |x| (x.rot, x.flip_h, x.flip_v))
240}

242fn shape_details(ca: Option<&Ctx>, a: &Shape, cb: Option<&Ctx>, b: &Shape, media: &mut Media, out: &mut Vec<String>) {
243    let label = shape_label(b);
244    let text = |s: &Shape| s.text.as_ref().map(|t| t.text()).unwrap_or_default();
245    let mut what = vec![];
246    if text(a) != text(b) {
247        what.push("text");
248    } else if a.text != b.text {
249        what.push("text formatting");
250    }
251    // Boxes as drawn: a stored box equal to the inherited one is no change.
252    match (&drawn_box(ca, a), &drawn_box(cb, b)) {
253        (Some(x), Some(y)) if (x.x, x.y) != (y.x, y.y) && (x.w, x.h) == (y.w, y.h) => what.push("moved"),
254        (Some(x), Some(y)) if (x.w, x.h) != (y.w, y.h) => what.push("resized"),
255        _ => {}
256    }
257    // Rotation and flips are reported alongside a move or resize, not hidden behind it. No stored
258    // box means the inherited one (the layout's or master's), as the shape is drawn.
259    let (ta, tb) = (turn(ca, a), turn(cb, b));
260    if ta.0 != tb.0 {
261        what.push("rotated");
262    }
263    if (ta.1, ta.2) != (tb.1, tb.2) {
264        what.push("flipped");
265    }
266    if a.geom != b.geom {
267        what.push("shape");
268    }
269    if a.fill != b.fill || a.line != b.line || a.style != b.style || a.effects != b.effects {
270        what.push("formatting");
271    }
272    let pic = |s: &Shape| match &s.kind {
273        ShapeKind::Picture { fill } => Some(fill.media),
274        _ => None,
275    };
276    if media.differ(pic(a), pic(b)) {
277        what.push("picture");
278    }
279    // Anything the checks above do not cover (alt text, hidden, a link…) is reported too, also
280    // beside them: the fields they compared are left out of this comparison.
281    let (mut x, mut y) = (plain_shape(a), plain_shape(b));
282    for z in [&mut x, &mut y] {
283        z.text = None;
284        z.xfrm = None;
285        z.geom = Default::default();
286        z.fill = None;
287        z.line = None;
288        z.style = None;
289        z.effects = Default::default();
290    }
291    if let (ShapeKind::Picture { .. }, ShapeKind::Picture { .. }) = (&x.kind, &y.kind) {
292        x.kind = ShapeKind::Shape;
293        y.kind = ShapeKind::Shape;
294    }
295    if x != y {
296        what.push("properties");
297    }
298    if !what.is_empty() {
299        out.push(format!("{label}: {}", what.join(", ")));
300    }
301}

303fn slide_details(cp: &Presentation, a: &Slide, rp: &Presentation, b: &Slide, media: &mut Media) -> Vec<String> {
304    let mut out = vec![];
305    let mut used: HashSet<usize> = HashSet::new();
306    let (ca, cb) = (Ctx::for_slide(cp, a), Ctx::for_slide(rp, b));
307    // The revised shapes by id (first wins) and by name (in order, with a cursor past used ones).
308    let mut by_id: HashMap<ShapeId, usize> = HashMap::new();
309    let mut by_name: HashMap<&str, (Vec<usize>, usize)> = HashMap::new();
310    for (k, x) in b.shapes.iter().enumerate() {
311        by_id.entry(x.id).or_insert(k);
312        if !x.name.is_empty() {
313            by_name.entry(x.name.as_str()).or_default().0.push(k);
314        }
315    }
316    for sa in &a.shapes {
317        let k = by_id.get(&sa.id).copied().or_else(|| {
318            let (ks, next) = by_name.get_mut(sa.name.as_str())?;
319            while ks.get(*next).is_some_and(|k| used.contains(k)) {
320                *next += 1;
321            }
322            ks.get(*next).copied()
323        });
324        match k {
325            Some(k) if !used.contains(&k) => {
326                used.insert(k);
327                if let Some(sb) = b.shapes.get(k) {
328                    shape_details(ca.as_ref(), sa, cb.as_ref(), sb, media, &mut out);
329                }
330            }
331            _ => out.push(format!("{} deleted (only in the deck)", shape_label(sa))),
332        }
333    }
334    for (k, sb) in b.shapes.iter().enumerate() {
335        if !used.contains(&k) {
336            out.push(format!("{} inserted (only in the file)", shape_label(sb)));
337        }
338    }
339    let order = |s: &Slide| s.shapes.iter().map(|x| x.name.clone()).collect::<Vec<_>>();
340    if out.is_empty() && order(a) != order(b) {
341        out.push("order of objects".into());
342    }
343    let layout_name = |p: &Presentation, s: &Slide| p.layout(s.layout).map(|(_, l)| l.name.clone());
344    if layout_name(cp, a) != layout_name(rp, b) {
345        out.push("layout".into());
346    }
347    if a.notes.text() != b.notes.text() {
348        out.push("notes".into());
349    }
350    if a.transition != b.transition {
351        out.push("transition".into());
352    }
353    if a.animations.len() != b.animations.len()
354        || a.animations
355            .iter()
356            .zip(&b.animations)
357            .any(|(x, y)| (&x.effect, x.class, x.start, x.duration_ms, x.delay_ms) != (&y.effect, y.class, y.start, y.duration_ms, y.delay_ms))
358    {
359        out.push("animations".into());
360    }
361    if a.background != b.background {
362        out.push("background".into());
363    }
364    if a.hidden != b.hidden {
365        out.push(if b.hidden { "hidden" } else { "shown" }.into());
366    }
367    out.truncate(MAX_DETAILS);
368    out
369}

375/// The revised slide made to live in `doc`: its media copied over (remapped on an id clash), shape
376/// ids kept unless another slide of `doc` uses them (then fresh, animations following), and its
377/// layout matched by id, then by name, then the first layout. An error when `doc` has used every
378/// object id and cannot take the slide without duplicating ids (#407).
379pub fn adopt(doc: &mut Presentation, rev: &Presentation, slide: &Slide, replacing: Option<SlideId>) -> crate::engine::Result<Slide> {
380    let mut sl = slide.clone();
381    // Media.
382    let mut media_map: HashMap<MediaId, MediaId> = HashMap::new();
383    // Both decks' media by id (the first item with an id wins, as in `Presentation::media`),
384    // looked up once per reference instead of searched.
385    let mut rev_at: HashMap<MediaId, usize> = HashMap::new();
386    for (i, m) in rev.media.iter().enumerate() {
387        rev_at.entry(m.id).or_insert(i);
388    }
389    let mut doc_at: HashMap<MediaId, usize> = HashMap::new();
390    for (i, m) in doc.media.iter().enumerate() {
391        doc_at.entry(m.id).or_insert(i);
392    }
393    for id in crate::engine::media_refs::slide_media(&sl) {
394        let Some(m) = rev_at.get(&id).and_then(|&i| rev.media.get(i)) else { continue };
395        match doc_at.get(&id).and_then(|&i| doc.media.get(i)) {
396            Some(x) if x.data == m.data => {}
397            None => {
398                // Keeps the revision's id, so later ids must start past it.
399                doc_at.insert(id, doc.media.len());
400                doc.media.push(m.clone());
401                doc.next_id = doc.next_id.max(id.0);
402            }
403            Some(_) => {
404                let before = doc.media.len();
405                let new = doc
406                    .add_media(&m.name, &m.content_type, m.data.as_ref().clone())
407                    .ok_or_else(|| crate::engine::cmd::bad("review.accept", crate::engine::cmd::NO_ID_SPACE))?;
408                if doc.media.len() > before {
409                    doc_at.entry(new).or_insert(before);
410                }
411                media_map.insert(id, new);
412            }
413        }
414    }
415    // Shape ids used elsewhere in the open deck.
416    let mut taken: HashSet<u32> = HashSet::new();
417    for s in doc.slides.iter().filter(|s| Some(s.id) != replacing) {
418        crate::model::walk(&s.shapes, &mut |x, _| {
419            taken.insert(x.id.0);
420        });
421    }
422    let mut id_map: HashMap<ShapeId, ShapeId> = HashMap::new();
423    fn rec(
424        doc: &mut Presentation,
425        shapes: &mut [Shape],
426        taken: &HashSet<u32>,
427        ids: &mut HashMap<ShapeId, ShapeId>,
428        depth: usize,
429    ) -> crate::engine::Result<()> {
430        if depth > 64 {
431            return Ok(());
432        }
433        for s in shapes {
434            if taken.contains(&s.id.0) {
435                let new = ShapeId(doc.alloc_id().ok_or_else(|| crate::engine::cmd::bad("review.accept", crate::engine::cmd::NO_ID_SPACE))?);
436                ids.insert(s.id, new);
437                s.id = new;
438            }
439            if let Some(ch) = s.children_mut() {
440                rec(doc, ch, taken, ids, depth + 1)?;
441            }
442        }
443        Ok(())
444    }
445    rec(doc, &mut sl.shapes, &taken, &mut id_map, 0)?;
446    crate::engine::media_refs::visit_mut::slide(&mut sl, &mut |m: &mut MediaId| {
447        if let Some(n) = media_map.get(m) {
448            *m = *n;
449        }
450    });
451    for a in &mut sl.animations {
452        if let Some(n) = id_map.get(&a.shape) {
453            a.shape = *n;
454        }
455        if let Some(t) = a.trigger.as_mut()
456            && let Some(n) = id_map.get(t)
457        {
458            *t = *n;
459        }
460    }
461    let mut max = 0;
462    max_shape_id(&sl.shapes, &mut max);
463    doc.next_id = doc.next_id.max(max);
464    // Layout.
465    if doc.layout(sl.layout).is_none() {
466        let name = rev.layout(slide.layout).map(|(_, l)| l.name.clone());
467        let by_name = name.and_then(|n| doc.masters.iter().flat_map(|m| &m.layouts).find(|l| l.name == n).map(|l| l.id));
468        sl.layout = by_name.or_else(|| doc.masters.first().and_then(|m| m.layouts.first()).map(|l| l.id)).unwrap_or_default();
469    }
470    // A placeholder without a box of its own is drawn with its layout's; where this deck's layout
471    // draws it otherwise (placed or turned differently), the file's box is stored on it, so the
472    // accepted slide looks as the file does and compares equal to it.
473    if let (Some(file), Some(here)) = (Ctx::for_slide(rev, slide), Ctx::for_slide(doc, &sl)) {
474        let boxes: Vec<Option<crate::model::Xfrm>> = slide
475            .shapes
476            .iter()
477            .zip(&sl.shapes)
478            .map(|(f, h)| {
479                let want = crate::model::resolve::xfrm(&file, f);
480                (h.xfrm.is_none() && crate::model::resolve::xfrm(&here, h) != want).then_some(want)
481            })
482            .collect();
483        for (h, b) in sl.shapes.iter_mut().zip(boxes) {
484            if b.is_some() {
485                h.xfrm = b;
486            }
487        }
488    }
489    match replacing {
490        Some(id) => sl.id = id,
491        None => {
492            if doc.slides.iter().any(|s| s.id == sl.id) || sl.id.0 == 0 {
493                sl.id = SlideId(doc.alloc_id().ok_or_else(|| crate::engine::cmd::bad("review.accept", crate::engine::cmd::NO_ID_SPACE))?);
494            }
495            doc.next_id = doc.next_id.max(sl.id.0);
496        }
497    }
498    Ok(sl)
499}

501/// Apply `change` (from [`diff`] of `doc` against `rev`) to `doc`. Returns the slide index to show
502/// (`None` when the change no longer applies), or an error when `doc` has used every object id.
503pub fn accept(doc: &mut Presentation, rev: &Presentation, change: &Change) -> crate::engine::Result<Option<usize>> {
504    match change.kind {
505        ChangeKind::Changed => {
506            let Some(b) = change.revised.and_then(|r| rev.slides.get(r)).cloned() else { return Ok(None) };
507            let Some(id) = doc.slides.get(change.slide).map(|s| s.id) else { return Ok(None) };
508            let sl = adopt(doc, rev, &b, Some(id))?;
509            let Some(slot) = doc.slides.get_mut(change.slide) else { return Ok(None) };
510            *slot = Arc::new(sl);
511            Ok(Some(change.slide))
512        }
513        ChangeKind::Inserted => {
514            let Some(b) = change.revised.and_then(|r| rev.slides.get(r)).cloned() else { return Ok(None) };
515            let sl = adopt(doc, rev, &b, None)?;
516            let at = change.slide.min(doc.slides.len());
517            doc.slides.insert(at, Arc::new(sl));
518            Ok(Some(at))
519        }
520        ChangeKind::Deleted => {
521            if change.slide >= doc.slides.len() || doc.slides.len() <= 1 {
522                return Ok(None);
523            }
524            doc.slides.remove(change.slide);
525            Ok(Some(change.slide.min(doc.slides.len().saturating_sub(1))))
526        }
527    }
528}

536    /// The sample deck, and a revised copy of it as base64 `.deckcraft` bytes: slide 1's title
537    /// retyped, slide 2 deleted, a new slide at the end, a shape moved on slide 3.
538    fn decks() -> (Session, String, usize) {
539        let mut s = Session::new();
540        crate::engine::sample::open_sample(&mut s).unwrap();
541        let n = s.doc().unwrap().doc.slides.len();
542        let orig = s.execute("file.saveBytes", &json!({})).unwrap()["data"].as_str().unwrap().to_string();
543        // Make the revision in a second document.
544        s.execute("file.openBytes", &json!({"name": "rev.deckcraft", "data": orig})).unwrap();
545        let first = s.doc().unwrap().doc.slides[0].shapes.iter().find(|x| x.text.is_some()).unwrap().id.0;
546        s.execute("edit.select", &json!({"ids": [first]})).unwrap();
547        s.execute("text.edit", &json!({"id": first})).unwrap();
548        s.execute("edit.selectAll", &json!({})).unwrap();
549        s.execute("text.insert", &json!({"text": "Revised title"})).unwrap();
550        s.execute("text.exit", &json!({})).unwrap();
551        s.execute("slide.delete", &json!({"index": 1})).unwrap();
552        s.execute("slide.go", &json!({"index": 1})).ok();
553        let moved = s.doc().unwrap().doc.slides[1].shapes[0].id.0;
554        s.execute("shape.move", &json!({"ids": [moved], "dx": 30, "dy": 0})).ok();
555        s.execute("slide.go", &json!({"index": n - 2})).ok();
556        s.execute("slide.new", &json!({"layout": "titleAndContent", "title": "Brand new", "body": "x"})).unwrap();
557        let rev = s.execute("file.saveBytes", &json!({})).unwrap()["data"].as_str().unwrap().to_string();
558        s.close_document(1);
559        s.set_active(0);
560        (s, rev, n)
561    }

616    /// A deck with one slide: a table whose cell has a picture fill, a picture background and a
617    /// transition sound. `tag` starts the bytes of each.
618    fn media_deck(tag: &str) -> crate::model::Presentation {
619        use std::sync::Arc;
620
621        use crate::model::style::PictureFill;
622        use crate::model::{Background, Fill, Shape, ShapeId, ShapeKind, Table, Transition, defaults};
623
624        let mut p = defaults::new_presentation(None);
625        let cell = p.add_media("cell.png", "image/png", format!("{tag} cell").into_bytes()).expect("media");
626        let back = p.add_media("back.png", "image/png", format!("{tag} background").into_bytes()).expect("media");
627        let sound = p.add_media("sound.wav", "audio/wav", format!("{tag} sound").into_bytes()).expect("media");
628        let mut t = Table::new(1, 1, 100.0, 40.0);
629        t.rows[0].cells[0].fill = Some(Fill::Picture(PictureFill { media: cell, ..Default::default() }));
630        let id = ShapeId(p.alloc_id().expect("id"));
631        let mut s = (*p.slides.first().cloned().unwrap_or_default()).clone();
632        s.shapes.push(Shape { id, kind: ShapeKind::Table(t), ..Default::default() });
633        s.background = Some(Background::Fill { fill: Fill::Picture(PictureFill { media: back, ..Default::default() }) });
634        s.transition = Some(Transition { kind: "fade".into(), sound: Some(sound), ..Default::default() });
635        p.slides = vec![Arc::new(s)];
636        p
637    }

814    #[test]
815    fn reject_keeps_the_open_deck_and_accept_all_skips_rejected_revisions() {
816        let (mut s, rev, n) = decks();
817        s.execute("review.compare", &json!({"name": "rev.deckcraft", "data": rev})).unwrap();
818        let before = s.execute("review.revisions", &json!({})).unwrap()["changes"].as_array().unwrap().clone();
819        let ins = before.iter().position(|c| c["kind"] == "inserted").unwrap();
820        let doc = s.doc().unwrap().doc.clone();
821        let undo = s.doc().unwrap().history.undo.len();
822        assert_eq!(s.execute("review.reject", &json!({"index": ins})).unwrap()["rejected"], 1);
823        assert!(std::sync::Arc::ptr_eq(&doc, &s.doc().unwrap().doc), "rejecting leaves the open deck as it is");
824        let left = s.execute("review.revisions", &json!({})).unwrap();
825        assert_eq!(kinds(&left).len(), before.len() - 1);
826        assert!(!kinds(&left).contains(&"inserted".into()), "{left}");
827        // Accept All takes the rest, never the rejected slide.
828        s.execute("review.accept", &json!({"all": true})).unwrap();
829        let st = s.doc().unwrap();
830        assert_eq!(st.doc.slides.len(), n - 1, "the deleted slide goes, the rejected insertion never comes");
831        assert!(!st.doc.slides.iter().any(|x| super::slide_title(x) == "Brand new"));
832        assert!(s.execute("review.revisions", &json!({})).unwrap()["changes"].as_array().unwrap().is_empty());
833        s.execute("edit.undo", &json!({})).unwrap();
834        assert!(std::sync::Arc::ptr_eq(&doc, &s.doc().unwrap().doc));
835        assert_eq!(s.doc().unwrap().history.undo.len(), undo, "rejecting is not an undo step");
836        // Reject All: nothing left to review, the deck as it was; Accept All then has nothing to take.
837        assert_eq!(s.execute("review.reject", &json!({"all": true})).unwrap()["rejected"], before.len() - 1);
838        assert!(s.execute("review.revisions", &json!({})).unwrap()["changes"].as_array().unwrap().is_empty());
839        assert!(s.execute("review.accept", &json!({"all": true})).is_err());
840        assert!(std::sync::Arc::ptr_eq(&doc, &s.doc().unwrap().doc));
841        assert!(s.execute("review.reject", &json!({"all": true})).is_err(), "nothing left to reject");
842        assert!(s.execute("review.reject", &json!({})).is_err());
843        s.execute("review.endCompare", &json!({})).unwrap();
844        assert!(s.execute("review.reject", &json!({"index": 0})).is_err());
845    }

888        other.execute("slide.new", &json!({"layout": "title", "title": "B"})).unwrap();
889        let data = other.execute("file.saveBytes", &json!({"format": "pptx"})).unwrap()["data"].as_str().unwrap().to_string();
890        let v = s.execute("review.compare", &json!({"name": "o.pptx", "data": data})).unwrap();
891        assert_eq!(v["byPosition"], true, "{v}");
892        let changed = |v: &Value| -> Vec<u64> {
893            v["changes"].as_array().unwrap().iter().filter(|c| c["kind"] == "changed").map(|c| c["revised"].as_u64().unwrap()).collect()
894        };
895        let revised = changed(&v);
896        let last = *revised.last().expect("a changed slide");
897        let at = v["changes"].as_array().unwrap().iter().position(|c| c["revised"] == last).unwrap();
898        s.execute("review.reject", &json!({"index": at})).unwrap();
899        // A new slide at the front shifts every open slide down by one.
900        s.execute("slide.new", &json!({"layout": "title", "title": "N"})).unwrap();
901        let n_at = slide_texts(&s).iter().position(|t| t.contains('N')).unwrap();
902        s.execute("slide.move", &json!({"from": n_at, "to": 0})).unwrap();
903        assert_eq!(slide_texts(&s).iter().position(|t| t.contains('N')), Some(0));
904        let now = s.execute("review.revisions", &json!({})).unwrap();
905        assert!(!changed(&now).contains(&last), "{now}");
906        s.execute("review.accept", &json!({"all": true})).unwrap();
907        let texts = slide_texts(&s);
908        assert!(!texts.iter().any(|t| t.contains('B')), "{texts:?}");
909    }

911    /// By position, inserting a slide before a rejected pair pushes the pair's open slide past the
912    /// end of the revised deck; it must not turn into a deletion that Accept All carries out.
913    #[test]
914    fn a_rejected_positional_revision_does_not_become_a_deletion_when_a_slide_is_inserted() {
915        let mut s = Session::with_new();
916        let mut other = Session::with_new();
917        for t in ["A", "X"] {
918            s.execute("slide.new", &json!({"layout": "title", "title": t})).unwrap();
919        }
920        for t in ["B", "Y"] {
921            other.execute("slide.new", &json!({"layout": "title", "title": t})).unwrap();
922        }
923        let data = other.execute("file.saveBytes", &json!({"format": "pptx"})).unwrap()["data"].as_str().unwrap().to_string();
924        let v = s.execute("review.compare", &json!({"name": "o.pptx", "data": data})).unwrap();
925        assert_eq!(v["byPosition"], true, "{v}");
926        let changes = v["changes"].as_array().unwrap();
927        let last = changes.iter().filter(|c| c["kind"] == "changed").map(|c| c["revised"].as_u64().unwrap()).max().expect("a changed slide");
928        let at = changes.iter().position(|c| c["kind"] == "changed" && c["revised"] == last).unwrap();
929        let open_id = s.doc().unwrap().doc.slides[last as usize].id;
930        s.execute("review.reject", &json!({"index": at})).unwrap();

934        s.execute("review.accept", &json!({"all": true})).unwrap();
935        assert!(s.doc().unwrap().doc.slides.iter().any(|sl| sl.id == open_id), "the open slide whose revision was rejected was deleted");
