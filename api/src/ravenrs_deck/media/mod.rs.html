1//! DeckCraft media: everything between embedded media bytes and the speakers / the screen.
2//!
3//! - [`probe`]: container, duration, audio channels / sample rate, video size / frame rate.
4//! - [`audio::decode`]: every common audio format to interleaved f32 [`audio::Pcm`] — MP3, AAC
5//!   (M4A, ADTS), ALAC, WAV, AIFF, CAF, FLAC, Ogg Vorbis, Opus (Ogg / WebM / Matroska) and the audio
6//!   track of MP4/MOV/WebM videos. WMA (ASF) is recognised and probed but not decoded.
7//! - [`video::VideoDecoder`]: H.264, HEVC, VP9 and AV1 in MP4/MOV and WebM/Matroska to RGBA
8//!   [`video::Frame`]s (our own decoders, ported from FilmCraft); [`video::poster_png`].
9//! - [`clip::ClipParams`]: trim, fade in/out and volume.
10//! - [`player::Player`]: the playback mixer and clock the UI host drives, over an
11//!   [`player::AudioOut`] it injects (cpal on desktop; none on the web or in tests, where the clock
12//!   runs on wall time); [`feed::VideoFeed`] decodes video ahead of that clock.
13//!
14//! Layer L2: depends only on the codec crates, never on the model or the UI. Builds for
15//! `wasm32-unknown-unknown` (no threads there: decoding runs inline).
16#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
17#![forbid(unsafe_code)]

52/// Container formats recognised from the first bytes.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum Container {
55    /// ISO BMFF: MP4, M4A, MOV, 3GP.
56    Mp4,
57    /// Matroska / WebM.
58    Matroska,
59    Ogg,
60    Wav,
61    Aiff,
62    Caf,
63    Flac,
64    Mp3,
65    /// AAC in ADTS frames (`.aac`).
66    Adts,
67    /// Advanced Systems Format (WMA / WMV).
68    Asf,
69    Unknown,
70}

72/// The ASF header object GUID (30 26 B2 75 8E 66 CF 11 A6 D9 00 AA 00 62 CE 6C).
73pub(crate) const ASF_HEADER: [u8; 16] = [0x30, 0x26, 0xB2, 0x75, 0x8E, 0x66, 0xCF, 0x11, 0xA6, 0xD9, 0x00, 0xAA, 0x00, 0x62, 0xCE, 0x6C];
74
75impl Container {
76    pub fn sniff(b: &[u8]) -> Container {
77        let at = |o: usize, m: &[u8]| b.get(o..o + m.len()) == Some(m);
78        if at(4, b"ftyp") || at(4, b"moov") || at(4, b"mdat") || at(4, b"wide") || at(4, b"free") {
79            Container::Mp4
80        } else if at(0, &[0x1A, 0x45, 0xDF, 0xA3]) {
81            Container::Matroska
82        } else if at(0, b"OggS") {
83            Container::Ogg
84        } else if (at(0, b"RIFF") || at(0, b"RF64")) && at(8, b"WAVE") {
85            Container::Wav
86        } else if at(0, b"FORM") && (at(8, b"AIFF") || at(8, b"AIFC")) {
87            Container::Aiff
88        } else if at(0, b"caff") {
89            Container::Caf
90        } else if at(0, b"fLaC") {
91            Container::Flac
92        } else if at(0, &ASF_HEADER) {
93            Container::Asf
94        } else if at(0, b"ID3") {
95            // ID3v2 is followed by MP3 frames (or, rarely, ADTS). The chain of tags is walked
96            // iteratively — a crafted file can carry thousands of empty tags — and only the
97            // bytes after the last one are sniffed.
98            let mut rest = b;
99            while rest.starts_with(b"ID3") {
100                let size = rest.get(6..10).map(|s| s.iter().fold(0usize, |a, &x| (a << 7) | (x & 0x7f) as usize)).unwrap_or(0);
101                rest = rest.get(10 + size..).unwrap_or(&[]);
102            }
103            match Container::sniff(rest) {
104                Container::Adts => Container::Adts,
105                _ => Container::Mp3,
106            }
107        } else if b.len() > 1 && b[0] == 0xFF && b[1] & 0xF0 == 0xF0 && b[1] & 0x06 == 0 {
108            Container::Adts
109        } else if b.len() > 1 && b[0] == 0xFF && b[1] & 0xE0 == 0xE0 && b[1] & 0x06 != 0 {
110            Container::Mp3
111        } else {
112            Container::Unknown
113        }
114    }
115
116    pub fn name(self) -> &'static str {
117        match self {
118            Container::Mp4 => "mp4",
119            Container::Matroska => "matroska",
120            Container::Ogg => "ogg",
121            Container::Wav => "wav",
122            Container::Aiff => "aiff",
123            Container::Caf => "caf",
124            Container::Flac => "flac",
125            Container::Mp3 => "mp3",
126            Container::Adts => "adts",
127            Container::Asf => "asf",
128            Container::Unknown => "unknown",
129        }
130    }
131}

136#[cfg(test)]
137mod tests {
138    use super::*;
139
140    #[test]
141    fn id3_tag_chains_do_not_recurse_the_stack() {
142        // 2 MB of empty ID3v2 tags (10 bytes each) chained ahead of the MP3 frames: walking
143        // the chain must not cost one stack frame per tag.
144        let mut b = Vec::with_capacity(200_000 * 10 + 2);
145        for _ in 0..200_000 {
146            b.extend_from_slice(&[b'I', b'D', b'3', 4, 0, 0, 0, 0, 0, 0]);
147        }
148        b.extend_from_slice(&[0xFF, 0xFB]); // an MP3 frame sync
149        assert_eq!(Container::sniff(&b), Container::Mp3);
150    }
