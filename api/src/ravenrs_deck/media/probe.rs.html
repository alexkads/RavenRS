1//! Media probing: container, duration, audio and video stream parameters — without decoding
2//! (except for audio files whose container doesn't state a length).

26#[derive(Clone, Debug, Default, PartialEq)]
27pub struct MediaInfo {
28    pub container: &'static str,
29    /// Duration in milliseconds (0 when unknown).
30    pub duration_ms: u64,
31    pub audio: Option<AudioInfo>,
32    pub video: Option<VideoInfo>,
33}
34
35impl MediaInfo {
36    pub fn duration(&self) -> f64 {
37        self.duration_ms as f64 / 1000.0
38    }
39    /// Something in it can be played.
40    pub fn playable(&self) -> bool {
41        self.audio.as_ref().is_some_and(|a| a.decodable) || self.video.as_ref().is_some_and(|v| v.decodable)
42    }
43}
44
45fn ms(secs: f64) -> u64 {
46    if secs.is_finite() && secs > 0.0 { (secs * 1000.0).round() as u64 } else { 0 }
47}

49/// Probe `bytes`.
50pub fn probe(bytes: &Bytes) -> Result<MediaInfo> {
51    let mut info = probe_headers(bytes)?;
52    if info.duration_ms == 0 && info.audio.as_ref().is_some_and(|a| a.decodable) && info.video.is_none() {
53        // No length in the header (raw MP3 / ADTS without an index): decode to measure.
54        if let Ok(pcm) = crate::media::audio::decode(bytes) {
55            info.duration_ms = ms(pcm.duration());
56        }
57    }
58    Ok(info)
59}

61/// [`probe`] from the headers alone, without decoding: a raw MP3 or ADTS stream without an
62/// index reports a `duration_ms` of 0.
63pub fn probe_headers(bytes: &Bytes) -> Result<MediaInfo> {
64    let c = Container::sniff(bytes);
65    let mut info = match c {
66        Container::Mp4 => probe_mp4(bytes)?,
67        Container::Matroska => probe_mkv(bytes)?,
68        Container::Asf => probe_asf(bytes)?,
69        Container::Ogg => probe_ogg(bytes).unwrap_or_else(|| MediaInfo { container: "ogg", ..Default::default() }),
70        _ => MediaInfo { container: c.name(), ..Default::default() },
71    };
72    if info.audio.is_none() && info.video.is_none() || info.audio.as_ref().is_some_and(|a| a.sample_rate == 0) {
73        // Plain audio files (and anything our demuxers didn't describe): symphonia's probe.
74        // Symphonia opens an MP4 wherever its probe finds an `ftyp` (after leading ID3v2 tags too); other known containers (a WAV's audio bytes can spell an `ftyp`) skip the check.
75        if matches!(c, Container::Mp4 | Container::Unknown | Container::Mp3) && !crate::media::mp4guard::probe_fits(bytes) {
76            return Err(MediaError::Corrupt("the MP4's sample tables do not fit the file".into()));
77        }
78        let (codec, rate, channels, frames) = std::panic::catch_unwind(|| crate::media::audio::probe_symphonia(bytes, c.name()))
79            .ok()
80            .flatten()
81            .ok_or_else(|| MediaError::Unsupported("this kind of media file".into()))?;
82        info.audio = Some(AudioInfo { codec: codec_label(&codec), sample_rate: rate, channels, decodable: true });
83        if info.duration_ms == 0
84            && let Some(n) = frames
85            && rate > 0
86        {
87            info.duration_ms = ms(n as f64 / rate as f64);
88        }
89    }
90    Ok(info)
91}

93fn codec_label(short: &str) -> String {
94    match short {
95        "mp3" => "MP3".into(),
96        "mp2" => "MP2".into(),
97        "mp1" => "MP1".into(),
98        "aac" => "AAC".into(),
99        "alac" => "ALAC".into(),
100        "flac" => "FLAC".into(),
101        "vorbis" => "Vorbis".into(),
102        s if s.starts_with("pcm") => "PCM".into(),
103        s if s.starts_with("adpcm") => "ADPCM".into(),
104        s => s.to_string(),
105    }
106}

108fn probe_mp4(bytes: &Bytes) -> Result<MediaInfo> {
109    use crate::craft::media::isobmff::{CodecConfig, TrackKind};
110    let src: &[u8] = bytes;
111    let f = crate::craft::media::isobmff::open(src).map_err(|e| MediaError::Corrupt(e.to_string()))?;
112    let mut info = MediaInfo { container: if f.is_quicktime { "mov" } else { "mp4" }, ..Default::default() };
113    let mut dur = if f.timescale > 0 { f.duration as f64 / f.timescale as f64 } else { 0.0 };
114    if let Some(ti) = f.track_of_kind(TrackKind::Video) {
115        let t = &f.tracks[ti];
116        let secs = if t.timescale > 0 { t.duration as f64 / t.timescale as f64 } else { 0.0 };
117        if let Some(e) = t.entries.first() {
118            let (w, h) = e.video.as_ref().map(|v| (v.width as u32, v.height as u32)).unwrap_or((t.width, t.height));
119            let decodable = matches!(e.codec, CodecConfig::Avc(_) | CodecConfig::Hevc(_) | CodecConfig::Vp9(_) | CodecConfig::Av1(_));
120            let frames = t.samples.len() as u64;
121            info.video = Some(VideoInfo {
122                codec: codec_name(&e.codec),
123                width: w,
124                height: h,
125                fps: if secs > 0.0 { frames as f64 / secs } else { 0.0 },
126                frames,
127                decodable,
128            });
129        }
130        dur = dur.max(secs);
131    }
132    if let Some(ti) = f.track_of_kind(TrackKind::Audio) {
133        let t = &f.tracks[ti];
134        if let Some(e) = t.entries.first() {
135            let (rate, channels) = e.audio.as_ref().map(|a| (a.sample_rate.round() as u32, a.channels as u16)).unwrap_or((0, 0));
136            let decodable =
137                matches!(e.codec, CodecConfig::Aac(_) | CodecConfig::Mp3 | CodecConfig::Pcm(_) | CodecConfig::Alac { .. } | CodecConfig::Flac(_));
138            info.audio = Some(AudioInfo { codec: codec_name(&e.codec), sample_rate: rate, channels, decodable });
139        }
140        if t.timescale > 0 {
141            dur = dur.max(t.duration as f64 / t.timescale as f64);
142        }
143    }
144    info.duration_ms = ms(dur);
145    Ok(info)
146}

148fn probe_mkv(bytes: &Bytes) -> Result<MediaInfo> {
149    use crate::craft::media::matroska::{Codec, TrackKind};
150    let src: &[u8] = bytes;
151    let f = crate::craft::media::matroska::open(src).map_err(|e| MediaError::Corrupt(e.to_string()))?;
152    let mut info = MediaInfo { container: if f.is_webm() { "webm" } else { "matroska" }, ..Default::default() };
153    let dur = f.duration_ns().map(|n| n as f64 / 1e9).unwrap_or(0.0);
154    info.duration_ms = ms(dur);
155    if let Some(t) = f.tracks.iter().find(|t| t.kind == TrackKind::Video) {
156        let (w, h) = t.video.as_ref().map(|v| (v.pixel_width, v.pixel_height)).unwrap_or((0, 0));
157        let frames = t.samples.len() as u64;
158        let fps = match t.default_duration_ns {
159            Some(d) if d > 0 => 1e9 / d as f64,
160            _ if dur > 0.0 => frames as f64 / dur,
161            _ => 0.0,
162        };
163        let decodable = matches!(t.codec, Codec::Avc { .. } | Codec::Hevc { .. } | Codec::Vp9 { .. } | Codec::Av1 { .. });
164        info.video = Some(VideoInfo { codec: mkv_codec_name(&t.codec), width: w, height: h, fps, frames, decodable });
165    }
166    if let Some(t) = f.tracks.iter().find(|t| t.kind == TrackKind::Audio) {
167        let (rate, channels) = t
168            .audio
169            .as_ref()
170            .map(|a| (a.output_sampling_frequency.unwrap_or(a.sampling_frequency).round() as u32, a.channels as u16))
171            .unwrap_or((0, 0));
172        let rate = if matches!(t.codec, Codec::Opus { .. }) { 48_000 } else { rate };
173        let decodable =
174            matches!(t.codec, Codec::Opus { .. } | Codec::Vorbis { .. } | Codec::Aac { .. } | Codec::Flac { .. } | Codec::Mp3 | Codec::Pcm { .. });
175        info.audio = Some(AudioInfo { codec: mkv_codec_name(&t.codec), sample_rate: rate, channels, decodable });
176    }
177    Ok(info)
178}

180fn probe_ogg(bytes: &Bytes) -> Option<MediaInfo> {
181    let src: &[u8] = bytes;
182    let f = crate::craft::media::ogg::open(src).ok()?;
183    let s = f.stream_of(crate::craft::media::ogg::Codec::Opus)?;
184    let st = f.streams.get(s)?;
185    let head = crate::craft::audio::opus::OpusHead::parse(st.headers.first()?).ok()?;
186    let timing = crate::craft::media::ogg::OpusTiming::of(st, head.pre_skip as u32);
187    Some(MediaInfo {
188        container: "ogg",
189        duration_ms: ms(timing.total as f64 / 48_000.0),
190        audio: Some(AudioInfo { codec: "Opus".into(), sample_rate: 48_000, channels: head.channels as u16, decodable: true }),
191        video: None,
192    })
193}

195/// Windows Media (ASF): enough of the header to describe the file. We have no WMA/WMV decoder.
196fn probe_asf(b: &[u8]) -> Result<MediaInfo> {
197    const FILE_PROPS: [u8; 16] = [0xA1, 0xDC, 0xAB, 0x8C, 0x47, 0xA9, 0xCF, 0x11, 0x8E, 0xE4, 0x00, 0xC0, 0x0C, 0x20, 0x53, 0x65];
198    const STREAM_PROPS: [u8; 16] = [0x91, 0x07, 0xDC, 0xB7, 0xB7, 0xA9, 0xCF, 0x11, 0x8E, 0xE6, 0x00, 0xC0, 0x0C, 0x20, 0x53, 0x65];
199    const AUDIO_MEDIA: [u8; 16] = [0x40, 0x9E, 0x69, 0xF8, 0x4D, 0x5B, 0xCF, 0x11, 0xA8, 0xFD, 0x00, 0x80, 0x5F, 0x5C, 0x44, 0x2B];
200    const VIDEO_MEDIA: [u8; 16] = [0xC0, 0xEF, 0x19, 0xBC, 0x4D, 0x5B, 0xCF, 0x11, 0xA8, 0xFD, 0x00, 0x80, 0x5F, 0x5C, 0x44, 0x2B];
201    let u16le = |o: usize| b.get(o..o + 2).map(|s| u16::from_le_bytes([s[0], s[1]])).unwrap_or(0);
202    let u32le = |o: usize| b.get(o..o + 4).map(|s| u32::from_le_bytes([s[0], s[1], s[2], s[3]])).unwrap_or(0);
203    let u64le = |o: usize| b.get(o..o + 8).and_then(|s| s.try_into().ok()).map(u64::from_le_bytes).unwrap_or(0);
204    if b.get(..16) != Some(&ASF_HEADER[..]) {
205        return Err(MediaError::Corrupt("not an ASF file".into()));
206    }
207    let header_end = (u64le(16) as usize).min(b.len());
208    let mut info = MediaInfo { container: "asf", ..Default::default() };
209    let mut pos: usize = 30;
210    while pos.checked_add(24).is_some_and(|end| end <= header_end) {
211        let guid = &b[pos..pos + 16];
212        let size = u64le(pos + 16) as usize;
213        if size < 24 {
214            break;
215        }
216        let d = pos + 24;
217        if guid == FILE_PROPS {
218            // file id 16, file size 8, creation 8, packets 8, play duration 8 (100 ns), send 8, preroll 8 (ms)
219            let play = u64le(d + 40) as f64 / 1e7;
220            let preroll = u64le(d + 56) as f64 / 1e3;
221            info.duration_ms = ms(play - preroll);
222        } else if guid == STREAM_PROPS {
223            let kind = b.get(d..d + 16);
224            // type 16, error correction type 16, time offset 8, type data len 4, ec len 4, flags 2, reserved 4
225            let td = d + 54;
226            if kind == Some(&AUDIO_MEDIA[..]) {
227                let tag = u16le(td);
228                let codec = match tag {
229                    0x0160 => "WMA 1",
230                    0x0161 => "WMA 2",
231                    0x0162 => "WMA Pro",
232                    0x0163 => "WMA Lossless",
233                    0x000A => "WMA Voice",
234                    _ => "WMA",
235                };
236                info.audio = Some(AudioInfo { codec: codec.into(), sample_rate: u32le(td + 4), channels: u16le(td + 2), decodable: false });
237            } else if kind == Some(&VIDEO_MEDIA[..]) {
238                info.video = Some(VideoInfo { codec: "WMV".into(), width: u32le(td), height: u32le(td + 4), decodable: false, ..Default::default() });
239            }
240        }
241        // A hostile size must not wrap the walk (or walk off the end): an object that
242        // overruns the header ends it, which is also the correct ASF semantics.
243        let Some(next) = pos.checked_add(size).filter(|&p| p <= header_end) else { break };
244        pos = next;
245    }
246    Ok(info)
247}

249#[cfg(test)]
250mod tests {
251    use super::*;
252
253    #[test]
254    fn hostile_asf_object_sizes_end_the_walk() {
255        // The header object GUID followed by 48 × 0xFF: the walk reads usize::MAX back as
256        // every object size and must end instead of wrapping `pos` (and panicking).
257        let mut b = crate::media::ASF_HEADER.to_vec();
258        b.extend_from_slice(&[0xFF; 48]);
259        assert!(matches!(probe_asf(&b), Ok(info) if info.container == "asf"));
260    }
