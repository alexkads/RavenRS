1//! Checks an MP4's box tree before symphonia reads it. Symphonia's MP4 reader reserves memory
2//! from the entry counts in the sample tables (and reads each sample whole at its `stsz` size)
3//! without checking them against the file, so a damaged count asks for gigabytes and the process
4//! aborts — an allocation failure cannot be caught. A file passes when every box fits inside its
5//! parent, every table's entries fit inside its box and every sample fits inside the file.
6
7/// Boxes whose payload is a sequence of boxes.
8const CONTAINERS: [&[u8; 4]; 11] = [b"moov", b"trak", b"mdia", b"minf", b"stbl", b"edts", b"dinf", b"udta", b"mvex", b"moof", b"traf"];
9
10/// Whether symphonia can be given `b` (an MP4) without a damaged size reaching an allocation.
11pub(crate) fn tables_fit(b: &[u8]) -> bool {
12    boxes_fit(b, b.len() as u64, 0)
13}
14
15/// Bytes symphonia's probe scans for a format marker before giving up.
16const PROBE_SEARCH_LIMIT: usize = 1024 * 1024;
17
18/// Whether symphonia can be given `b`, whatever the sniffer made of it: its probe takes the first
19/// `ftyp` within its search window as the start of an MP4 (the size field 4 bytes before it), so
20/// the table check must start where that parser will, not at offset 0.
21pub(crate) fn probe_fits(b: &[u8]) -> bool {
22    if crate::media::Container::sniff(b) == crate::media::Container::Mp4 && !tables_fit(b) {
23        return false;
24    }
25    // The probe reads leading ID3v2 tags as metadata and searches again after the last one.
26    let mut rest = b;
27    while rest.starts_with(b"ID3") {
28        let size = rest.get(6..10).map(|s| s.iter().fold(0usize, |a, &x| (a << 7) | (x & 0x7f) as usize)).unwrap_or(0);
29        rest = rest.get(10 + size..).unwrap_or(&[]);
30    }
31    let head = &rest[..rest.len().min(PROBE_SEARCH_LIMIT + 16)];
32    match head.windows(4).position(|w| w == b"ftyp") {
33        Some(p) => tables_fit(&rest[p.saturating_sub(4)..]),
34        None => true,
35    }
36}

38fn be32(b: &[u8], at: usize) -> Option<u64> {
39    b.get(at..at.checked_add(4)?).map(|s| u32::from_be_bytes([s[0], s[1], s[2], s[3]]) as u64)
40}

42fn boxes_fit(b: &[u8], file_len: u64, depth: u32) -> bool {
43    if depth > 16 {
44        return false;
45    }
46    let mut at = 0usize;
47    while b.len() - at >= 8 {
48        let Some(size32) = be32(b, at) else { return false };
49        let kind = &b[at + 4..at + 8];
50        let (header, size) = match size32 {
51            0 => (8, (b.len() - at) as u64),
52            1 => match b.get(at + 8..at + 16) {
53                Some(s) => (16, u64::from_be_bytes(s.try_into().unwrap_or_default())),
54                None => return false,
55            },
56            n => (8, n),
57        };
58        if size < header as u64 || size > (b.len() - at) as u64 {
59            // A box that runs past the end of the file is fine at the top level (the file was cut
60            // short; symphonia stops there), but never inside a parent.
61            return depth == 0 && !matches!(kind, b"moov" | b"moof");
62        }
63        let body = &b[at + header..at + size as usize];
64        let ok = if CONTAINERS.iter().any(|c| c.as_slice() == kind) {
65            boxes_fit(body, file_len, depth + 1)
66        } else if kind == b"meta" {
67            // ISO `meta` is a full box (4 bytes of version and flags first); QuickTime's is not.
68            let skip = if be32(body, 0) == Some(0) { 4 } else { 0 };
69            boxes_fit(&body[skip.min(body.len())..], file_len, depth + 1)
70        } else {
71            table_fits(kind, body, file_len)
72        };
73        if !ok {
74            return false;
75        }
76        at += size as usize;
77    }
78    true
79}

81/// A sample table (a full box: version and flags, then a count and that many entries).
82fn table_fits(kind: &[u8], body: &[u8], file_len: u64) -> bool {
83    let entry = match kind {
84        b"stts" | b"ctts" | b"co64" => 8,
85        b"stsc" => 12,
86        b"stco" | b"stss" => 4,
87        b"stsz" => {
88            let (Some(size), Some(count)) = (be32(body, 4), be32(body, 8)) else { return false };
89            if size != 0 {
90                return size <= file_len;
91            }
92            let Some(entries) = body.get(12..) else { return false };
93            return count <= entries.len() as u64 / 4
94                && entries.as_chunks::<4>().0.iter().take(count as usize).all(|s| u32::from_be_bytes(*s) as u64 <= file_len);
95        }
96        _ => return true,
97    };
98    be32(body, 4).is_some_and(|count| count <= body.len().saturating_sub(8) as u64 / entry)
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    /// An MP4 whose `stsc` count asks for far more entries than the file holds.
106    fn mp4_with_huge_stsc() -> Vec<u8> {
107        let stsc = [&24u32.to_be_bytes()[..], b"stsc", &[0, 0, 0, 0], &u32::MAX.to_be_bytes(), &[0; 8]].concat();
108        let stbl = [&(stsc.len() as u32 + 8).to_be_bytes()[..], b"stbl", &stsc].concat();
109        let moov = [&(stbl.len() as u32 + 8).to_be_bytes()[..], b"moov", &stbl].concat();
110        [&16u32.to_be_bytes()[..], b"ftyp", b"isom", &[0; 4], &moov].concat()
111    }
112
113    #[test]
114    fn a_prefixed_mp4_is_checked_where_symphonia_starts_it() {
115        let mp4 = mp4_with_huge_stsc();
116        assert!(!tables_fit(&mp4));
117        assert!(!probe_fits(&mp4));
118        let prefixed = [&b"junk before the file "[..], &mp4].concat();
119        assert_eq!(crate::media::Container::sniff(&prefixed), crate::media::Container::Unknown);
120        assert!(tables_fit(&prefixed), "offset 0 sees no MP4 boxes");
121        assert!(!probe_fits(&prefixed));
122        let far = [&vec![0u8; PROBE_SEARCH_LIMIT + 16][..], &mp4].concat();
123        assert!(probe_fits(&far), "past the probe window symphonia never reads it as an MP4");
124    }
125}
