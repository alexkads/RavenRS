1//! Y'CbCr → RGBA8 conversion for decoded pictures.
2
3/// Y'CbCr matrix (ISO/IEC 23091-2 MatrixCoefficients).
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5pub enum Matrix {
6    Bt601,
7    Bt709,
8    Bt2020,
9}
10
11impl Matrix {
12    /// From a `matrix_coefficients` code; unspecified codes pick by picture height (SD → 601).
13    pub fn from_code(code: u8, height: u32) -> Matrix {
14        match code {
15            1 => Matrix::Bt709,
16            5 | 6 => Matrix::Bt601,
17            9 | 10 => Matrix::Bt2020,
18            _ if height <= 576 => Matrix::Bt601,
19            _ => Matrix::Bt709,
20        }
21    }
22    /// (Kr, Kb).
23    fn k(self) -> (f32, f32) {
24        match self {
25            Matrix::Bt601 => (0.299, 0.114),
26            Matrix::Bt709 => (0.2126, 0.0722),
27            Matrix::Bt2020 => (0.2627, 0.0593),
28        }
29    }
30}
31
32/// Borrowed planes of one picture. Chroma planes are `width >> sx` (rounded up) wide.
33pub struct Planes<'a, T> {
34    pub y: &'a [T],
35    pub u: &'a [T],
36    pub v: &'a [T],
37    pub y_stride: usize,
38    pub c_stride: usize,
39    pub width: usize,
40    pub height: usize,
41    /// Chroma subsampling shifts (1 = halved).
42    pub sx: u32,
43    pub sy: u32,
44    pub bits: u32,
45    /// No chroma: grey.
46    pub mono: bool,
47}
48
49/// Convert to tightly packed RGBA8.
50pub fn to_rgba<T: Copy + Into<u32>>(p: &Planes<T>, m: Matrix, full_range: bool) -> Vec<u8> {
51    let (kr, kb) = m.k();
52    let kg = 1.0 - kr - kb;
53    let max = ((1u32 << p.bits) - 1) as f32;
54    let scale = (1u32 << p.bits.saturating_sub(8)) as f32;
55    // Normalised luma / chroma: Y in 0..1, Cb/Cr in -0.5..0.5.
56    let (y_off, y_mul, c_mul) = if full_range { (0.0, 1.0 / max, 1.0 / max) } else { (16.0 * scale, 1.0 / (219.0 * scale), 1.0 / (224.0 * scale)) };
57    let mid = (1u32 << (p.bits - 1)) as f32;
58    let cr_r = 2.0 * (1.0 - kr);
59    let cb_b = 2.0 * (1.0 - kb);
60    let cb_g = -2.0 * kb * (1.0 - kb) / kg;
61    let cr_g = -2.0 * kr * (1.0 - kr) / kg;
62    let mut out = vec![0u8; p.width * p.height * 4];
63    let px = |v: f32| (v * 255.0 + 0.5).clamp(0.0, 255.0) as u8;
64    for row in 0..p.height {
65        let yrow = row * p.y_stride;
66        let crow = (row >> p.sy) * p.c_stride;
67        let o = &mut out[row * p.width * 4..(row + 1) * p.width * 4];
68        for col in 0..p.width {
69            let yv: u32 = p.y.get(yrow + col).copied().map(Into::into).unwrap_or(0);
70            let yn = (yv as f32 - y_off) * y_mul;
71            let (cb, cr) = if p.mono {
72                (0.0, 0.0)
73            } else {
74                let ci = crow + (col >> p.sx);
75                let u: u32 = p.u.get(ci).copied().map(Into::into).unwrap_or(mid as u32);
76                let v: u32 = p.v.get(ci).copied().map(Into::into).unwrap_or(mid as u32);
77                ((u as f32 - mid) * c_mul, (v as f32 - mid) * c_mul)
78            };
79            let k = col * 4;
80            o[k] = px(yn + cr_r * cr);
81            o[k + 1] = px(yn + cb_g * cb + cr_g * cr);
82            o[k + 2] = px(yn + cb_b * cb);
83            o[k + 3] = 255;
84        }
85    }
86    out
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    fn one(y: u8, u: u8, v: u8, full: bool) -> [u8; 4] {
94        let p = Planes { y: &[y], u: &[u], v: &[v], y_stride: 1, c_stride: 1, width: 1, height: 1, sx: 0, sy: 0, bits: 8, mono: false };
95        let o = to_rgba(&p, Matrix::Bt709, full);
96        [o[0], o[1], o[2], o[3]]
97    }
98
99    #[test]
100    fn limited_range_black_white_and_red() {
101        assert_eq!(one(16, 128, 128, false), [0, 0, 0, 255]);
102        assert_eq!(one(235, 128, 128, false), [255, 255, 255, 255]);
103        // BT.709 red: Y 63, Cb 102, Cr 240.
104        let r = one(63, 102, 240, false);
105        assert!(r[0] > 250 && r[1] < 5 && r[2] < 5, "{r:?}");
106    }
107
108    #[test]
109    fn full_range_grey() {
110        assert_eq!(one(128, 128, 128, true), [128, 128, 128, 255]);
111    }
112
113    #[test]
114    fn ten_bit_white() {
115        let y = [940u16];
116        let c = [512u16];
117        let p = Planes { y: &y, u: &c, v: &c, y_stride: 1, c_stride: 1, width: 1, height: 1, sx: 0, sy: 0, bits: 10, mono: false };
118        assert_eq!(to_rgba(&p, Matrix::Bt709, false), vec![255, 255, 255, 255]);
119    }
