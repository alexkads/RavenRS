1//! Text layout for slide text bodies.
2//!
3//! [`layout`] resolves every run's formatting through the placeholder/style chain and lays each
4//! paragraph out in the order bidirectional text needs:
5//! 1. embedding levels for the whole logical paragraph (UAX #9, base level from `a:pPr/@rtl`);
6//! 2. itemization by level, script, face (per-cluster fallback through the `a:latin`/`a:ea`/`a:cs`
7//!    slots and the theme's script fonts) and shaping style, each item shaped with the whole
8//!    paragraph as context; paint-only attributes (colour, highlight, underline) never split a
9//!    word, so Arabic joining and ligatures survive them;
10//! 3. line breaking in logical order (UAX #14 opportunities at cluster boundaries, greedy fill);
11//! 4. per line, rules L1/L2 reorder the characters visually (L4 mirroring is done by the shaper).
12//!
13//! It then places bullets and numbers from the paragraph's start edge (the right edge for RTL
14//! paragraphs), applies indents, spacing and alignment (kashida for `justLow`), anchors the block
15//! in the shape's text box, and shrinks text on overflow when the body asks for it. The result is
16//! positioned glyph runs in shape-local coordinates plus per-line caret geometry for editing:
17//! bidirectional carets with affinity, logical and visual movement, and selection as the union of
18//! visual boxes.
19#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
20#![forbid(unsafe_code)]

122/// Which side of a boundary a caret belongs to: `Downstream` sticks to the character after it
123/// (the default), `Upstream` to the character before it, which differs at direction changes and
124/// at the end of a wrapped line.
125#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
126pub enum Affinity {
127    Upstream,
128    #[default]
129    Downstream,
130}

132/// Caret movement: through the text order, or across the screen.
133#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
134pub enum MoveMode {
135    /// Next/previous grapheme in logical order (`forward` = towards the end of the text).
136    #[default]
137    Logical,
138    /// Next grapheme boundary to the right/left on screen (`forward` = right).
139    Visual,
140}

142impl LineInfo {
143    fn len(&self) -> usize {
144        self.end.saturating_sub(self.start)
145    }
146    fn odd(&self, k: usize) -> bool {
147        self.levels.get(k).is_some_and(|l| l % 2 == 1)
148    }
149    /// Leading edge (where its direction begins) of character `k` of the line.
150    fn leading(&self, k: usize) -> Option<f64> {
151        self.edges.get(k).map(|e| if self.odd(k) { e.1 } else { e.0 })
152    }
153    /// Trailing edge (where its direction ends) of character `k` of the line.
154    fn trailing(&self, k: usize) -> Option<f64> {
155        self.edges.get(k).map(|e| if self.odd(k) { e.0 } else { e.1 })
156    }
157    fn shift_x(&mut self, dx: f64) {
158        for x in &mut self.caret_x {
159            *x += dx;
160        }
161        for e in &mut self.edges {
162            e.0 += dx;
163            e.1 += dx;
164        }
165        self.extent.0 += dx;
166        self.extent.1 += dx;
167    }
168    /// Graphemes on the line as (first, end) character indices from the line start.
169    fn graphemes(&self) -> Vec<(usize, usize)> {
170        let n = self.len();
171        let mut out = Vec::new();
172        let mut k = 0;
173        while k < n {
174            let mut e = k + 1;
175            while e < n && !self.boundary.get(e).copied().unwrap_or(true) {
176                e += 1;
177            }
178            out.push((k, e));
179            k = e;
180        }
181        out
182    }
183    /// Visual extent of characters `from..to` (k from the line start).
184    fn grapheme_box(&self, from: usize, to: usize) -> Option<(f64, f64)> {
185        let es = self.edges.get(from..to)?;
186        let l = es.iter().map(|e| e.0).fold(f64::INFINITY, f64::min);
187        let r = es.iter().map(|e| e.1).fold(f64::NEG_INFINITY, f64::max);
188        (l <= r).then_some((l, r))
189    }
190    /// Visual x intervals covering paragraph characters `from..to` on this line, left to right
191    /// and merged where they touch. Bidi text can make a logical range several boxes.
192    pub fn spans(&self, from: usize, to: usize) -> Vec<(f64, f64)> {
193        let a = from.max(self.start) - self.start;
194        let b = to.min(self.end).saturating_sub(self.start);
195        merge_spans(self.edges.get(a..b.max(a)).unwrap_or(&[]).to_vec())
196    }
197    /// Horizontal extent of paragraph characters `from..to` on this line.
198    pub fn span(&self, from: usize, to: usize) -> Option<(f64, f64)> {
199        let s = self.spans(from, to);
200        Some((s.first()?.0, s.last()?.1))
201    }
202    /// Caret stops left to right: (x, character offset in the paragraph, affinity).
203    fn visual_stops(&self) -> Vec<(f64, usize, Affinity)> {
204        let mut gs: Vec<(f64, f64, usize, usize, bool)> =
205            self.graphemes().into_iter().filter_map(|(a, b)| self.grapheme_box(a, b).map(|(l, r)| (l, r, a, b, self.odd(a)))).collect();
206        gs.sort_by(|x, y| x.0.total_cmp(&y.0));
207        let mut out: Vec<(f64, usize, Affinity)> = Vec::with_capacity(gs.len() * 2 + 1);
208        let mut push = |s: (f64, usize, Affinity)| {
209            if !out.last().is_some_and(|l: &(f64, usize, Affinity)| (l.0 - s.0).abs() < 0.01 && l.1 == s.1) {
210                out.push(s);
211            }
212        };
213        for (l, r, a, b, rtl) in gs {
214            let (a, b) = (self.start + a, self.start + b);
215            if rtl {
216                push((l, b, Affinity::Upstream));
217                push((r, a, Affinity::Downstream));
218            } else {
219                push((l, a, Affinity::Downstream));
220                push((r, b, Affinity::Upstream));
221            }
222        }
223        if out.is_empty() {
224            out.push((self.caret_x.first().copied().unwrap_or(0.0), self.start, Affinity::Downstream));
225        }
226        out
227    }
228}

230/// Non-empty x intervals sorted left to right, merged where they touch.
231fn merge_spans(mut v: Vec<(f64, f64)>) -> Vec<(f64, f64)> {
232    v.retain(|e| e.1 > e.0);
233    v.sort_by(|x, y| x.0.total_cmp(&y.0));
234    let mut out: Vec<(f64, f64)> = Vec::with_capacity(v.len());
235    for e in v {
236        match out.last_mut() {
237            Some(last) if e.0 <= last.1 + 0.01 => last.1 = last.1.max(e.1),
238            _ => out.push(e),
239        }
240    }
241    out
242}

252    /// The line containing a caret position (the later line at a soft wrap boundary).
253    pub fn line_of(&self, pos: Pos) -> Option<usize> {
254        self.line_of_affine(pos, Affinity::Downstream)
255    }
256    /// The line containing a caret position: at a soft wrap boundary, the earlier line for an
257    /// upstream caret and the later one for a downstream caret.
258    pub fn line_of_affine(&self, pos: Pos, affinity: Affinity) -> Option<usize> {
259        let mut found = None;
260        for (i, l) in self.lines.iter().enumerate() {
261            if l.para == pos.para && pos.ch >= l.start && pos.ch <= l.end {
262                if affinity == Affinity::Upstream && pos.ch == l.end && pos.ch > l.start {
263                    return Some(i);
264                }
265                found = Some(i);
266                if pos.ch < l.end {
267                    break;
268                }
269            }
270        }
271        found
272            .or_else(|| self.lines.iter().rposition(|l| l.para == pos.para && l.start <= pos.ch))
273            .or_else(|| self.lines.iter().rposition(|l| l.para == pos.para))
274    }
275    /// Caret line segment (x, top, bottom) in shape-local, unrotated coordinates (downstream).
276    pub fn caret(&self, pos: Pos) -> Option<(f64, f64, f64)> {
277        self.caret_at(pos, Affinity::Downstream)
278    }
279    /// Caret line segment (x, top, bottom) for a position and affinity: downstream at the leading
280    /// edge of the character after the boundary, upstream at the trailing edge of the one before.
281    pub fn caret_at(&self, pos: Pos, affinity: Affinity) -> Option<(f64, f64, f64)> {
282        let li = self.line_of_affine(pos, affinity)?;
283        let l = self.lines.get(li)?;
284        let k = pos.ch.clamp(l.start, l.end) - l.start;
285        let x = match affinity {
286            Affinity::Downstream if k < l.len() => l.leading(k),
287            Affinity::Upstream if k > 0 => l.trailing(k - 1),
288            _ => None,
289        };
290        let x = x.or_else(|| l.caret_x.get(k).copied()).or(l.caret_x.last().copied())?;
291        Some((x, l.top, l.bottom))
292    }
293    /// Nearest caret position for a point.
294    pub fn hit(&self, p: Point) -> Pos {
295        self.hit_affinity(p).0
296    }
297    /// Nearest caret position for a point, with the affinity that keeps the caret on the side of
298    /// the character that was hit (bidi text has two carets at a direction change).
299    pub fn hit_affinity(&self, p: Point) -> (Pos, Affinity) {
300        let mut best: Option<&LineInfo> = None;
301        for l in &self.lines {
302            if p.y >= l.top && p.y < l.bottom {
303                // In multi-column bodies pick the line whose column contains x.
304                if best.is_none() || p.x >= l.extent.0.min(l.caret_x.first().copied().unwrap_or(l.extent.0)) - 4.0 {
305                    best = Some(l);
306                }
307            }
308        }
309        let l = match best {
310            Some(l) => l,
311            None => {
312                let first = self.lines.first();
313                let last = self.lines.last();
314                match (first, last) {
315                    (Some(f), _) if p.y < f.top => f,
316                    (_, Some(l)) => l,
317                    _ => return (Pos::default(), Affinity::Downstream),
318                }
319            }
320        };
321        let mut gs: Vec<(f64, f64, usize, usize, bool)> =
322            l.graphemes().into_iter().filter_map(|(a, b)| l.grapheme_box(a, b).map(|(x0, x1)| (x0, x1, a, b, l.odd(a)))).collect();
323        if gs.is_empty() {
324            return (Pos { para: l.para, ch: l.start }, Affinity::Downstream);
325        }
326        gs.sort_by(|x, y| x.0.total_cmp(&y.0));
327        let hit = gs.iter().find(|g| p.x >= g.0 && p.x <= g.1 && g.1 > g.0).copied();
328        let (g, left_half) = match hit {
329            Some(g) => (g, p.x < (g.0 + g.1) / 2.0),
330            None => match (gs.first().copied(), gs.last().copied()) {
331                (Some(f), _) if p.x < f.0 => (f, true),
332                (_, Some(z)) if p.x > z.1 => (z, false),
333                // Between zero-width graphemes: the nearest one.
334                _ => match gs.iter().min_by(|a, b| ((a.0 + a.1) / 2.0 - p.x).abs().total_cmp(&((b.0 + b.1) / 2.0 - p.x).abs())).copied() {
335                    Some(g) => (g, p.x < (g.0 + g.1) / 2.0),
336                    None => return (Pos { para: l.para, ch: l.start }, Affinity::Downstream),
337                },
338            },
339        };
340        let (_, _, a, b, rtl) = g;
341        let (before, after) = (Pos { para: l.para, ch: l.start + a }, Pos { para: l.para, ch: l.start + b });
342        // The leading half of a character puts the caret before it, the trailing half after it.
343        if left_half != rtl { (before, Affinity::Downstream) } else { (after, Affinity::Upstream) }
344    }
345    /// Move a caret one grapheme: logically (`forward` = towards the end of the text, crossing
346    /// paragraphs) or visually (`forward` = to the right on screen, continuing on the next or
347    /// previous line in the paragraph's direction at the line's edge).
348    pub fn move_caret(&self, pos: Pos, affinity: Affinity, forward: bool, mode: MoveMode) -> (Pos, Affinity) {
349        if mode == MoveMode::Visual
350            && let Some(li) = self.line_of_affine(pos, affinity)
351            && let Some(l) = self.lines.get(li)
352        {
353            let stops = l.visual_stops();
354            let cur_x = self.caret_at(pos, affinity).map(|c| c.0).unwrap_or(0.0);
355            let idx = stops
356                .iter()
357                .position(|s| s.1 == pos.ch && s.2 == affinity)
358                .or_else(|| stops.iter().position(|s| s.1 == pos.ch))
359                .or_else(|| stops.iter().enumerate().min_by(|a, b| (a.1.0 - cur_x).abs().total_cmp(&(b.1.0 - cur_x).abs())).map(|(i, _)| i));
360            if let Some(idx) = idx {
361                let mut j = idx;
362                loop {
363                    let next = if forward { j.checked_add(1) } else { j.checked_sub(1) };
364                    let Some(nj) = next else { break };
365                    let Some(s) = stops.get(nj) else { break };
366                    // Skip stops that wouldn't move the caret on screen or in the text.
367                    if s.1 != pos.ch && (s.0 - cur_x).abs() > 0.01 {
368                        return (Pos { para: l.para, ch: s.1 }, s.2);
369                    }
370                    j = nj;
371                }
372            }
373            // Off the line's edge: on to the neighbouring line in reading order.
374            let logical_forward = forward != l.rtl;
375            let target = if logical_forward { li.checked_add(1) } else { li.checked_sub(1) };
376            return match target.and_then(|t| self.lines.get(t)) {
377                Some(t) if logical_forward => (Pos { para: t.para, ch: t.start }, Affinity::Downstream),
378                Some(t) => (Pos { para: t.para, ch: t.end }, if t.para == l.para { Affinity::Upstream } else { Affinity::Downstream }),
379                None => (pos, affinity),
380            };
381        }
382        let Some(b) = self.boundaries.get(pos.para) else { return (pos, affinity) };
383        let len = b.len().saturating_sub(1);
384        let ch = pos.ch.min(len);
385        let next = if forward {
386            (ch + 1..=len).find(|&k| b.get(k).copied().unwrap_or(true))
387        } else {
388            (0..ch).rev().find(|&k| b.get(k).copied().unwrap_or(true))
389        };
390        match next {
391            Some(k) => (Pos { para: pos.para, ch: k }, Affinity::Downstream),
392            None if forward && pos.para + 1 < self.boundaries.len() => (Pos { para: pos.para + 1, ch: 0 }, Affinity::Downstream),
393            None if !forward && pos.para > 0 => {
394                let prev = pos.para - 1;
395                (Pos { para: prev, ch: self.boundaries.get(prev).map(|b| b.len().saturating_sub(1)).unwrap_or(0) }, Affinity::Downstream)
396            }
397            None => (Pos { para: pos.para, ch }, Affinity::Downstream),
398        }
399    }
400    /// Selection highlight rectangles between two positions: per line, the union of the visual
401    /// boxes of the selected characters (one logical range can be several boxes in bidi text).
402    pub fn selection_rects(&self, a: Pos, b: Pos) -> Vec<Rect> {
403        let (s, e) = if a <= b { (a, b) } else { (b, a) };
404        let mut out = vec![];
405        for l in &self.lines {
406            let lp = Pos { para: l.para, ch: l.start };
407            let le = Pos { para: l.para, ch: l.end };
408            if le < s || lp > e {
409                continue;
410            }
411            let from = if s > lp { s.ch } else { l.start };
412            let to = if e < le { e.ch } else { l.end };
413            let mut spans = l.spans(from, to);
414            // Selecting across a paragraph end shows a small extra box for the newline, at the
415            // paragraph's end side.
416            if e.para > l.para && to == l.end {
417                let end_x = l.caret_x.last().copied().unwrap_or(l.extent.1);
418                spans.push(if l.rtl { (end_x - 6.0, end_x) } else { (end_x, end_x + 6.0) });
419                spans = merge_spans(spans);
420            }
421            for (x0, x1) in spans {
422                if x1 > x0 {
423                    out.push(Rect::new(x0, l.top, x1, l.bottom));
424                }
425            }
426        }
427        out
428    }
429    /// Underlines for text an input method is composing, from `a` to `b` (#101): `(x0, x1, y)` per
430    /// visual span, a little below each line's baseline (PowerPoint underlines the composition
431    /// in place). A composition that wraps gets one underline per line.
432    pub fn composition_underlines(&self, a: Pos, b: Pos) -> Vec<(f64, f64, f64)> {
433        let (s, e) = if a <= b { (a, b) } else { (b, a) };
434        let mut out = vec![];
435        if s == e {
436            return out;
437        }
438        for l in &self.lines {
439            let lp = Pos { para: l.para, ch: l.start };
440            let le = Pos { para: l.para, ch: l.end };
441            if le <= s || lp >= e {
442                continue;
443            }
444            let from = if s > lp { s.ch } else { l.start };
445            let to = if e < le { e.ch } else { l.end };
446            // Under the baseline, in the descent, never below the line.
447            let y = (l.baseline + (l.bottom - l.baseline) * 0.35).min(l.bottom);
448            for (x0, x1) in l.spans(from, to) {
449                if x1 > x0 {
450                    out.push((x0, x1, y));
451                }
452            }
453        }
454        out
455    }

460    /// The line a caret is on. An upstream caret at the start of a soft-wrapped line (where End
461    /// leaves it, after the previous line's trailing spaces) belongs to the line before.
462    pub fn line_index(&self, pos: Pos, affinity: Affinity) -> Option<usize> {
463        if affinity == Affinity::Upstream && pos.ch > 0 {
464            for (i, w) in self.lines.windows(2).enumerate() {
465                if let [a, b] = w
466                    && b.para == pos.para
467                    && a.para == pos.para
468                    && b.start == pos.ch
469                {
470                    return Some(i);
471                }
472            }
473        }
474        self.line_of_affine(pos, affinity)
475    }
476    /// Up/Down from a caret with its affinity: the position on the line above/below nearest `x`.
477    pub fn vertical_affine(&self, pos: Pos, affinity: Affinity, down: bool, x: f64) -> (Pos, Affinity) {
478        let Some(li) = self.line_index(pos, affinity) else { return (pos, affinity) };
479        let target = if down { li.checked_add(1) } else { li.checked_sub(1) };
480        let Some(l) = target.and_then(|t| self.lines.get(t)) else { return (pos, affinity) };
481        let p = Point::new(x, (l.top + l.bottom) / 2.0);
482        let mut h = self.hit(p);
483        if h.para != l.para {
484            h.para = l.para;
485        }
486        h.ch = h.ch.clamp(l.start, l.end);
487        (h, Affinity::Downstream)
488    }
489    /// Start/end of the visual line containing `pos` (logical: Home/End).
490    pub fn line_bounds(&self, pos: Pos) -> (Pos, Pos) {
491        self.line_bounds_affine(pos, Affinity::Downstream)
492    }
493    /// Start/end of the visual line a caret with this affinity is on.
494    pub fn line_bounds_affine(&self, pos: Pos, affinity: Affinity) -> (Pos, Pos) {
495        match self.line_index(pos, affinity).and_then(|i| self.lines.get(i)) {
496            Some(l) => (Pos { para: l.para, ch: l.start }, Pos { para: l.para, ch: l.end }),
497            None => (pos, pos),
498        }
499    }
500    /// Where End puts the caret: after a soft wrap, past the line's trailing spaces at the start
501    /// of the next line with upstream affinity (so it is still drawn on this line), else the end
502    /// of the paragraph.
503    pub fn line_end_caret(&self, pos: Pos, affinity: Affinity) -> (Pos, Affinity) {
504        let Some(li) = self.line_index(pos, affinity) else { return (pos, Affinity::Downstream) };
505        let Some(l) = self.lines.get(li) else { return (pos, Affinity::Downstream) };
506        match self.lines.get(li + 1) {
507            Some(n) if n.para == l.para && n.start > l.start => (Pos { para: l.para, ch: n.start }, Affinity::Upstream),
508            _ => (Pos { para: l.para, ch: l.end }, Affinity::Downstream),
509        }
510    }
511    /// The caret to draw (x, top, bottom): an upstream caret after a soft wrap is drawn at the end
512    /// of the line it belongs to.
513    pub fn caret_drawn(&self, pos: Pos, affinity: Affinity) -> Option<(f64, f64, f64)> {
514        let li = self.line_index(pos, affinity)?;
515        let l = self.lines.get(li)?;
516        if pos.ch > l.end {
517            let x = l.caret_x.last().copied()?;
518            return Some((x, l.top, l.bottom));
519        }
520        self.caret_at(pos, affinity)
521    }
522    /// Bounding box of all glyphs/lines (shape-local).
523    pub fn bounds(&self) -> Rect {
524        let mut r: Option<Rect> = None;
525        for l in &self.lines {
526            let lr = Rect::new(l.extent.0, l.top, l.extent.1.max(l.extent.0), l.bottom);
527            r = Some(r.map(|a| a.union(lr)).unwrap_or(lr));
528        }
529        r.unwrap_or(self.inner)
530    }
531}

533/// Height the shape needs so its text fits (Resize shape to fit text): text height + insets.
534pub fn fit_height(ctx: &Ctx, shape: &Shape, body: &TextBody, rect: Rect, fields: &dyn Fields) -> f64 {
535    let bp = resolve::body(ctx, shape);
536    let l = layout(ctx, shape, body, &Opts { rect, fields, prompt_color: None, no_shrink: true });
537    l.content_height + bp.inset_t.unwrap_or(3.6) + bp.inset_b.unwrap_or(3.6)
538}
539
540/// Width a non-wrapping text box needs.
541pub fn fit_width(ctx: &Ctx, shape: &Shape, body: &TextBody, rect: Rect, fields: &dyn Fields) -> f64 {
542    let bp = resolve::body(ctx, shape);
543    let wide = Rect::new(rect.x0, rect.y0, rect.x0 + 100_000.0, rect.y1);
544    let mut s2 = shape.clone();
545    if let Some(t) = s2.text.as_mut() {
546        t.body.wrap = Some(false);
547    }
548    let l = layout(ctx, &s2, body, &Opts { rect: wide, fields, prompt_color: None, no_shrink: true });
549    let w = l.lines.iter().map(|li| li.start_offset + (li.extent.1 - li.extent.0)).fold(0.0, f64::max);
550    w + bp.inset_l.unwrap_or(7.2) + bp.inset_r.unwrap_or(7.2) + 1.0
551}
