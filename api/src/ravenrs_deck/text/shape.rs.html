11pub(crate) fn shape_para(
12    ctx: &Ctx,
13    shape: &Shape,
14    body: &TextBody,
15    pi: usize,
16    scale: f64,
17    opts: &Opts,
18    number: Option<u32>,
19    vertical: Upright,
20) -> ParaShaped {
21    let db = FontDb::global();
22    let Some(para) = body.paragraphs.get(pi) else {
23        let face = db.face("Inter", "Regular");
24        return ParaShaped {
25            cells: vec![],
26            styles: vec![],
27            faces: vec![],
28            props: ParaProps::default(),
29            lead_size: 18.0,
30            lead_face: face.clone(),
31            end_size: 18.0,
32            end_face: face,
33            bullet: None,
34            level: 0,
35            rtl: false,
36        };
37    };
38    let props = resolve::para(ctx, shape, para);
39    let rtl = props.rtl.unwrap_or(false);
40    let mut styles: Vec<Style> = vec![];
41    let mut faces: Vec<Arc<FontFace>> = vec![];
42    // The logical paragraph: its text and each character's style.
43    let mut text = String::new();
44    let mut char_style: Vec<usize> = vec![];
45    for run in &para.runs {
46        let rp = resolve::run(ctx, shape, para, &run.props);
47        let st = make_style(ctx, &rp, scale, opts.prompt_color);
48        let t: String = match &run.kind {
49            RunKind::Text => run.text.clone(),
50            RunKind::Break => "\u{b}".into(),
51            RunKind::Field { field } => opts.fields.field_in(field, rp.lang.as_deref()).unwrap_or_else(|| run.text.clone()),
52            RunKind::Math { .. } => run.text.clone(),
53        };
54        let si = styles.len();
55        styles.push(st);
56        char_style.extend(std::iter::repeat_n(si, t.chars().count()));
57        text.push_str(&t);
58    }
59    let cells = shape_cells(ctx, &text, &char_style, &styles, &mut faces, rtl, vertical);
60    let lead = para.runs.first().map(|r| resolve::run(ctx, shape, para, &r.props)).unwrap_or_else(|| resolve::run(ctx, shape, para, &para.end_props));
61    let lead_style = make_style(ctx, &lead, scale, opts.prompt_color);
62    let lead_size = lead_style.size;
63    let lead_face = lead_style.face.clone();
64    // The paragraph mark, which ends the last line when no break does.
65    let end_style = make_style(ctx, &resolve::run(ctx, shape, para, &para.end_props), scale, opts.prompt_color);
66    let (end_size, end_face) = (end_style.size, end_style.face.clone());
67    // Bullet.
68    let bullet = match &props.bullet {
69        Some(Bullet::Char { char }) => {
70            let mut rp = lead.clone();
71            if let Some(f) = &props.bullet_font {
72                rp.font = Some(f.clone());
73            }
74            rp.bold = Some(false);
75            rp.italic = Some(false);
76            if let Some(c) = &props.bullet_color {
77                rp.fill = Some(Fill::solid(c.clone()));
78            }
79            rp.size = Some(props.bullet_size_points.unwrap_or(lead.size.unwrap_or(18.0) * props.bullet_size.unwrap_or(1.0)));
80            rp.underline = None;
81            rp.strike = None;
82            rp.highlight = None;
83            let mut st = make_style(ctx, &rp, scale, opts.prompt_color);
84            st.baseline = 0.0;
85            // Symbol fonts (Wingdings) aren't available: map common private-use bullets.
86            let ch = map_symbol_bullet(char, props.bullet_font.as_deref());
87            bullet_face(db, &mut st, &ch);
88
89            // A symbol font (Wingdings, Symbol…) that is installed draws its own glyph, found
90            // at U+F000 + the character code as such fonts map them; otherwise common symbol
91            // bullets are mapped to Unicode look-alikes.
92            let ch = match symbol_font_char(&st.face, char, props.bullet_font.as_deref()) {
93                Some(c) => c.to_string(),
94                None => map_symbol_bullet(char, props.bullet_font.as_deref()),
95            };
96            bullet_face(db, &mut st, &ch);
97            // A face the cascade still picks may lack the bullet: draw the dot rather than .notdef.
98            let chars: Vec<char> = ch.chars().collect();
99            let ch = if crate::fonts::covers_cluster(&st.face, &chars) { ch } else { "•".into() };
100            bullet_face(db, &mut st, &ch);
101            Some((ch, st))
102        }
103        Some(Bullet::AutoNum { scheme, start_at }) => {
104            let mut rp = lead.clone();
105            if let Some(c) = &props.bullet_color {
106                rp.fill = Some(Fill::solid(c.clone()));
107            }
108            rp.size = Some(props.bullet_size_points.unwrap_or(lead.size.unwrap_or(18.0) * props.bullet_size.unwrap_or(1.0)));
109            rp.underline = None;
110            let mut st = make_style(ctx, &rp, scale, opts.prompt_color);
111            let label = autonum(scheme, start_at.saturating_sub(1).saturating_add(number.unwrap_or(1)));
112            bullet_face(db, &mut st, &label);
113            Some((label, st))
114        }
115        _ => None,
116    };
117    ParaShaped { cells, styles, faces, props, lead_size, lead_face, end_size, end_face, bullet, level: para.level, rtl }
118}

120/// A face that draws the bullet `label` when the bullet's own can't (circled numbers ①, arrows
121/// and other symbols missing from the text font), by the usual fallback.
122fn bullet_face(db: &FontDb, st: &mut Style, label: &str) {
123    let chars: Vec<char> = label.chars().collect();
124    if !chars.is_empty() && !crate::fonts::covers_cluster(&st.face, &chars) {
125        st.face = db.cascade(&chars, std::slice::from_ref(&st.face), st.style_name);
126    }
127}

129/// Shape a logical paragraph into cells: bidi levels, per-grapheme faces, items of one level,
130/// script, face and shaping style, each shaped with the whole paragraph as context.
131/// A run's outer shadow and glow (inner shadows aren't drawn on text), at the layout's `scale`.
132pub(crate) fn run_effects(ctx: &Ctx, r: &RunProps, scale: f64) -> Option<RunEffects> {
133    let shadow = r.shadow.as_ref().filter(|s| !s.inner).map(|s| {
134        let a = s.dir.to_radians();
135        TextShadow { color: ctx.color(&s.color, None), blur: s.blur.max(0.0) * scale, dx: a.cos() * s.dist * scale, dy: a.sin() * s.dist * scale }
136    });
137    let glow = r.glow.as_ref().filter(|g| g.radius > 0.0).map(|g| (ctx.color(&g.color, None), g.radius * scale));
138    (shadow.is_some() || glow.is_some()).then_some(RunEffects { shadow, glow })
139}

141#[allow(clippy::too_many_arguments)]
142fn shape_cells(
143    ctx: &Ctx,
144    text: &str,
145    char_style: &[usize],
146    styles: &[Style],
147    faces: &mut Vec<Arc<FontFace>>,
148    rtl: bool,
149    vertical: Upright,
150) -> Vec<Cell> {
151    let db = FontDb::global();
152    let chars: Vec<char> = text.chars().collect();
153    let n = chars.len();
154    // Byte offset of each character (and of the end).
155    let mut byte_of: Vec<usize> = text.char_indices().map(|(b, _)| b).collect();
156    byte_of.push(text.len());
157    let char_at_byte: HashMap<usize, usize> = byte_of.iter().enumerate().map(|(i, b)| (*b, i)).collect();
158    let style_rtl: Vec<bool> = styles.iter().map(|st| st.lang.as_deref().is_some_and(bidi::is_rtl_lang)).collect();
159    let rtl_lang: Vec<bool> = char_style.iter().map(|&si| style_rtl.get(si).copied().unwrap_or(false)).collect();
160    let levels = bidi::levels_lang(text, &chars, rtl, &rtl_lang);
161    let scripts = script::resolve(&chars);
162    let mut boundary = vec![false; n];
163    let mut graphemes: Vec<(usize, usize)> = Vec::new();
164    for (b, g) in text.grapheme_indices(true) {
165        let Some(&i) = char_at_byte.get(&b) else { continue };
166        if let Some(x) = boundary.get_mut(i) {
167            *x = true;
168        }
169        graphemes.push((i, i + g.chars().count()));
170    }
171    let face_index = |faces: &mut Vec<Arc<FontFace>>, f: &Arc<FontFace>| -> usize {
172        if let Some(i) = faces.iter().position(|x| Arc::ptr_eq(x, f)) {
173            return i;
174        }
175        faces.push(f.clone());
176        faces.len() - 1
177    };
178    // A face per grapheme cluster (never split between faces).
179    let mut char_face = vec![0usize; n];
180    let mut chains: HashMap<(usize, FontSlot, Option<ScriptTag>), Vec<Arc<FontFace>>> = HashMap::new();
181    let mut prev: Option<Arc<FontFace>> = None;
182    for &(a, b) in &graphemes {
183        let Some(cluster) = chars.get(a..b) else { continue };
184        let si = char_style.get(a).copied().unwrap_or(0);
185        let Some(st) = styles.get(si) else { continue };
186        let (sc, slot) = scripts.get(a).copied().unwrap_or((None, FontSlot::Latin));
187        // Spaces, digits and punctuation stay in the face of the text before them when it can
188        // draw them, so a phrase isn't cut into items at every space; not in a symbol font
189        // (`a:sym`), which PowerPoint uses for the symbol codes alone (Wingdings' space is about
190        // four times a text font's).
191        // Emoji, and pictographs asked to look like text (U+FE0E), pick their own face.
192        let neutral =
193            cluster.iter().all(|c| script::script_of(*c).is_none()) && !crate::fonts::is_emoji_cluster(cluster) && !cluster.contains(&'\u{FE0E}');
194        // A symbol-font character (U+F0xx) with an `a:sym` font draws in that font when it's there.
195        // Some installed Symbol faces (Apple's on macOS) have no private-use mapping: they map the
196        // glyphs to real Unicode, so the PUA code is remapped to that character when shaping.
197        let sym_face = st
198            .sym
199            .as_deref()
200            .filter(|_| cluster.first().is_some_and(|c| ('\u{F000}'..='\u{F0FF}').contains(c)))
201            .filter(|f| db.has_family(f))
202            .map(|f| db.face(f, st.style_name))
203            .filter(|f| covers_symbol(f, cluster.first().copied().unwrap_or(' ')));
204        let face = match (sym_face, prev.as_ref()) {
205            (Some(f), _) => f,
206            (None, Some(p))
207                if neutral
208                    && !p.is_color()
209                    && !is_symbol_font(&p.family)
210                    && crate::fonts::covers_cluster(p, cluster)
211                    && char_style.get(a.wrapping_sub(1)) == Some(&si) =>
212            {
213                p.clone()
214            }
215            (None, _) => {
216                let chain = chains.entry((si, slot, sc)).or_insert_with(|| face_chain(ctx, st, slot, sc));
217                db.cascade(cluster, chain, st.style_name)
218            }
219        };
220        let fi = face_index(faces, &face);
221        for f in char_face.get_mut(a..b).into_iter().flatten() {
222            *f = fi;
223        }
224        prev = Some(face);
225    }
226    let mut cells: Vec<Cell> = (0..n)
227        .map(|i| Cell {
228            ch: chars.get(i).copied().unwrap_or(' '),
229            style: char_style.get(i).copied().unwrap_or(0),
230            face: char_face.get(i).copied().unwrap_or(0),
231            level: levels.get(i).copied().unwrap_or(0),
232            glyphs: vec![],
233            adv: 0.0,
234            head: i,
235            boundary: boundary.get(i).copied().unwrap_or(true),
236            upright: false,
237        })
238        .collect();
239    // In East Asian vertical text, ideographs, kana, Hangul, full-width forms and the like stand
240    // upright (a whole grapheme at a time); other text lies on its side.
241    if vertical != Upright::None {
242        for &(a, b) in &graphemes {
243            let up = chars.get(a).is_some_and(|c| if vertical == Upright::All { !c.is_whitespace() } else { is_upright(*c) });
244            for c in cells.get_mut(a..b).into_iter().flatten() {
245                c.upright = up;
246            }
247        }
248    }
249    // Items: maximal ranges of one level, face, script and shaping style (case mapping, language).
250    let upper = |i: usize| styles.get(char_style.get(i).copied().unwrap_or(0)).is_some_and(|st| st.caps != Caps::None);
251    let lang = |i: usize| styles.get(char_style.get(i).copied().unwrap_or(0)).and_then(|st| st.lang.as_deref());
252    let kern = |i: usize| styles.get(char_style.get(i).copied().unwrap_or(0)).is_some_and(|st| st.kern);
253    let up = |i: usize| cells.get(i).is_some_and(|c| c.upright);
254    let key = |i: usize| (levels.get(i).copied(), char_face.get(i).copied(), scripts.get(i).map(|s| s.0), upper(i), lang(i), kern(i), up(i));
255    let mut items: Vec<(usize, usize)> = Vec::new();
256    let mut a = 0;
257    while a < n {
258        let mut b = a + 1;
259        while b < n && key(b) == key(a) {
260            b += 1;
261        }
262        items.push((a, b));
263        a = b;
264    }
265    // Upright glyphs keep their horizontal forms: PowerPoint turns "、", "。" and "「" as they are
266    // (no `vert` alternates).
267    for (a, b) in items {
268        shape_item(&mut cells, styles, faces, text, &byte_of, &char_at_byte, a..b, scripts.get(a).and_then(|s| s.0), lang(a), upper(a));
269    }
270    // Stacked text: each character takes one em down the line, centred across it (PowerPoint
271    // stacks 18 pt Latin letters about 18 pt apart, a space only its own width).
272    if vertical == Upright::All {
273        for i in 0..n {
274            let Some(c) = cells.get(i) else { continue };
275            if !c.upright || c.head != i || c.glyphs.is_empty() {
276                continue;
277            }
278            let size = styles.get(c.style).map(|st| st.size).unwrap_or(18.0);
279            let width: f64 = (i..n).take_while(|&j| cells.get(j).is_some_and(|d| d.head == i)).filter_map(|j| cells.get(j)).map(|d| d.adv).sum();
280            let pad = (size - width) / 2.0;
281            if let Some(c) = cells.get_mut(i) {
282                for g in &mut c.glyphs {
283                    g.1 += pad;
284                }
285            }
286            for j in i..n {
287                match cells.get_mut(j) {
288                    Some(d) if d.head == i => d.adv = if j == i { size } else { 0.0 },
289                    _ => break,
290                }
291            }
292        }
293    }
294    cells
295}

297/// A character in capitals (`cap="all"`/`"small"`): its uppercase when that is one character.
298/// PowerPoint leaves characters whose capital is a sequence as they are: "ß" stays "ß" (not "SS"),
299/// "ﬁ" and "ŉ" stay.
300pub(crate) fn caps_char(ch: char) -> char {
301    let mut up = ch.to_uppercase();
302    match (up.next(), up.next()) {
303        (Some(u), None) => u,
304        _ => ch,
305    }
306}

308/// Does `c` stand upright in East Asian vertical text? Ideographs, kana, Hangul, Bopomofo, CJK
309/// symbols and punctuation, full-width forms, enclosed and compatibility CJK, circled numbers
310/// and emoji (Unicode's Vertical_Orientation U and T, roughly); Latin, digits and the rest of the
311/// half-width text lie on their side as in `vert`.
312pub(crate) fn is_upright(c: char) -> bool {
313    matches!(c as u32,
314        0x1100..=0x11FF
315            | 0x2460..=0x24FF
316            | 0x2E80..=0x2FFF
317            | 0x3000..=0x303F
318            | 0x3040..=0x30FF
319            | 0x3100..=0x312F
320            | 0x3130..=0x318F
321            | 0x3190..=0x31FF
322            | 0x3200..=0x33FF
323            | 0x3400..=0x4DBF
324            | 0x4E00..=0x9FFF
325            | 0xA960..=0xA97F
326            | 0xAC00..=0xD7FF
327            | 0xF900..=0xFAFF
328            | 0xFE10..=0xFE1F
329            | 0xFE30..=0xFE6F
330            | 0xFF01..=0xFF60
331            | 0xFFE0..=0xFFE6
332            | 0x1F000..=0x1FAFF
333            | 0x20000..=0x3FFFF
334    )
335}

351/// The OpenType features to change from the shaper's defaults for an item, as PowerPoint draws
352/// it: kerning only where the run asks for it (`kern`), and no standard or contextual ligatures in
353/// Latin, Greek and Cyrillic ("fi", "ffi" and Calibri's "ti" stay apart). Other scripts need
354/// their ligatures (Arabic lam-alef), and emoji sequences are joined by them.
355pub(crate) fn shaping_features(cells: &[Cell], styles: &[Style], script: Option<ScriptTag>) -> Vec<crate::fonts::Feature> {
356    let mut f = vec![];
357    if !cells.first().and_then(|c| styles.get(c.style)).is_some_and(|st| st.kern) {
358        f.extend(crate::fonts::feature("-kern"));
359    }
360    let alphabetic = matches!(script.as_ref(), Some(b"Latn" | b"Grek" | b"Cyrl"));
361    let emoji = cells.iter().any(|c| matches!(c.ch as u32, 0x200D | 0x20E3 | 0xFE0F | 0x2600..=0x27BF | 0x1F000..=0x1FAFF | 0xE0020..=0xE007F));
362    if alphabetic && !emoji {
363        f.extend(crate::fonts::feature("-liga"));
364        f.extend(crate::fonts::feature("-clig"));
365    }
366    f
367}

369/// Shape cells `range` (one item) and hang each cluster's glyphs on its first cell, sharing the
370/// cluster's advance among its graphemes (by the ligature's GDEF carets when it has them).
371#[allow(clippy::too_many_arguments)]
372fn shape_item(
373    cells: &mut [Cell],
374    styles: &[Style],
375    faces: &[Arc<FontFace>],
376    text: &str,
377    byte_of: &[usize],
378    char_at_byte: &HashMap<usize, usize>,
379    range: std::ops::Range<usize>,
380    script: Option<ScriptTag>,
381    lang: Option<&str>,
382    upper: bool,
383) {
384    let Some(first) = cells.get(range.start) else { return };
385    let (fi, level) = (first.face, first.level);
386    let Some(face) = faces.get(fi) else { return };
387    let rtl = level % 2 == 1;
388    let (Some(&b0), Some(&b1)) = (byte_of.get(range.start), byte_of.get(range.end)) else { return };
389    let features = shaping_features(cells.get(range.clone()).unwrap_or(&[]), styles, script);
390    let params = ShapeParams { features: &features, rtl, script, language: lang };
391    let sym = styles.get(first.style).and_then(|st| st.sym.as_deref());
392    let glyphs = crate::fonts::shape_range(face, text, b0..b1, &params, |ch| {
393        let ch = if upper { caps_char(ch) } else { ch };
394        symbol_map_char(face, ch, sym).unwrap_or(ch)
395    });
396    let upem = face.upem.max(1.0);
397    let scale_of = |cell: &Cell| {
398        let st = styles.get(cell.style);
399        let size = st.map(|s| s.size).unwrap_or(18.0);
400        let small = st.is_some_and(|s| s.caps == Caps::Small) && cell.ch.is_lowercase();
401        let size = if small { size * 0.8 } else { size };
402        // Superscript and subscript glyphs are drawn at two thirds of the size, so they advance by
403        // two thirds too.
404        let size = if st.is_some_and(|s| s.baseline != 0.0) { size * SUB_SUPER } else { size };
405        size / upem
406    };
407    // Cluster heads in this item, ascending; each cluster spans to the next head.
408    let mut heads: Vec<usize> = glyphs.iter().filter_map(|g| char_at_byte.get(&g.cluster).copied()).filter(|h| range.contains(h)).collect();
409    heads.sort_unstable();
410    heads.dedup();
411    let mut width: HashMap<usize, f64> = HashMap::new();
412    let mut lig: HashMap<usize, (u32, usize)> = HashMap::new();
413    for g in &glyphs {
414        let Some(&h) = char_at_byte.get(&g.cluster) else { continue };
415        let Some(cell) = cells.get_mut(h) else { continue };
416        if matches!(cell.ch, '\t'..='\u{b}') {
417            continue;
418        }
419        let s = scale_of(cell);
420        let w = width.entry(h).or_insert(0.0);
421        cell.glyphs.push((g.gid, *w + g.x_offset as f64 * s, -(g.y_offset as f64) * s));
422        *w += g.x_advance as f64 * s;
423        let e = lig.entry(h).or_insert((g.gid, 0));
424        e.1 += 1;
425    }
426    for (hi, &h) in heads.iter().enumerate() {
427        let end = heads.get(hi + 1).copied().unwrap_or(range.end).min(range.end);
428        let w = width.get(&h).copied().unwrap_or(0.0);
429        for c in cells.get_mut(h..end).into_iter().flatten() {
430            c.head = h;
431        }
432        let stops: Vec<usize> = (h..end).filter(|&i| i == h || cells.get(i).is_some_and(|c| c.boundary)).collect();
433        let m = stops.len().max(1);
434        // Visual widths of the cluster's graphemes, left to right.
435        let mut bounds = vec![0.0];
436        if m > 1
437            && let Some(&(gid, 1)) = lig.get(&h)
438        {
439            let s = cells.get(h).map(scale_of).unwrap_or(0.0);
440            let carets = crate::fonts::ligature_carets(face, gid);
441            if carets.len() + 1 >= m {
442                bounds.extend(carets.iter().take(m - 1).map(|c| (*c as f64 * s).clamp(0.0, w)));
443            }
444        }
445        if bounds.len() != m {
446            bounds = (0..m).map(|j| w * j as f64 / m as f64).collect();
447        }
448        bounds.push(w);
449        for (j, &i) in stops.iter().enumerate() {
450            let vj = if rtl { m - 1 - j } else { j };
451            let share = bounds.get(vj + 1).copied().unwrap_or(w) - bounds.get(vj).copied().unwrap_or(0.0);
452            if let Some(c) = cells.get_mut(i) {
453                c.adv = share.max(0.0);
454            }
455        }
456    }
457    // Tracking (character spacing) after every drawn grapheme and space.
458    for i in range {
459        let Some(c) = cells.get(i) else { continue };
460        let drawn = cells.get(c.head).is_some_and(|hd| !hd.glyphs.is_empty());
461        if c.boundary
462            && (drawn || c.ch == ' ')
463            && let Some(sp) = styles.get(c.style).map(|s| s.spacing)
464            && let Some(c) = cells.get_mut(i)
465        {
466            c.adv += sp;
467        }
468    }
