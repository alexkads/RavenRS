1fn roman(mut n: u32, upper: bool) -> String {
2    let table = [
3        (1000, "m"),
4        (900, "cm"),
5        (500, "d"),
6        (400, "cd"),
7        (100, "c"),
8        (90, "xc"),
9        (50, "l"),
10        (40, "xl"),
11        (10, "x"),
12        (9, "ix"),
13        (5, "v"),
14        (4, "iv"),
15        (1, "i"),
16    ];
17    let mut s = String::new();
18    if n == 0 || n > 3999 {
19        return n.to_string();
20    }
21    for (v, r) in table {
22        while n >= v {
23            s.push_str(r);
24            n -= v;
25        }
26    }
27    if upper { s.to_uppercase() } else { s }
28}

30fn alpha_num(n: u32, upper: bool) -> String {
31    if n == 0 {
32        return String::new();
33    }
34    let n = n - 1;
35    let c = (b'a' + (n % 26) as u8) as char;
36    let reps = (n / 26 + 1).min(50);
37    let s: String = std::iter::repeat_n(c, reps as usize).collect();
38    if upper { s.to_uppercase() } else { s }
39}

41/// Letters of a numbering alphabet: 1 → the first, past the end the letter doubles ("aa"), as
42/// the Latin alphabetic schemes do.
43fn letters(list: &[char], n: u32) -> String {
44    let len = list.len().max(1) as u32;
45    let Some(i) = n.checked_sub(1) else { return String::new() };
46    let c = list.get((i % len) as usize).copied().unwrap_or('?');
47    std::iter::repeat_n(c, (i / len + 1).min(50) as usize).collect()
48}

50/// `n` written with the digits `zero..zero+9` (Thai, Devanagari, full-width digits).
51fn digits_in(n: u32, zero: char) -> String {
52    n.to_string().chars().map(|d| d.to_digit(10).and_then(|v| char::from_u32(zero as u32 + v)).unwrap_or(d)).collect()
53}

57/// Chinese numerals: 10 → 十, 11 → 十一, 25 → 二十五, 105 → 一百零五.
58pub(crate) fn chinese_num(n: u32) -> String {
59    if n == 0 || n > 99_999 {
60        return n.to_string();
61    }
62    let units = ['\0', '十', '百', '千', '万'];
63    let ds: Vec<u32> = n.to_string().chars().filter_map(|c| c.to_digit(10)).collect();
64    let mut out = String::new();
65    let mut zero = false;
66    for (i, d) in ds.iter().enumerate() {
67        let place = ds.len() - 1 - i;
68        if *d == 0 {
69            zero = true;
70            continue;
71        }
72        if zero {
73            out.push('零');
74            zero = false;
75        }
76        // "十" alone for 10–19, not "一十".
77        if !(*d == 1 && place == 1 && i == 0) {
78            out.push(CJK_DIGITS.get(*d as usize).copied().unwrap_or('?'));
79        }
80        if let Some(u) = units.get(place).filter(|u| **u != '\0') {
81            out.push(*u);
82        }
83    }
84    out
85}

87/// Text of an auto-number bullet: `arabicPeriod` 3 → "3.". The East Asian, Thai, Hindi, Arabic
88/// and Hebrew schemes are written as PowerPoint draws them: `ea1ChsPeriod` 10 → "十．",
89/// `ea1JpnKorPeriod` 10 → "一〇．" (digit by digit), `arabicDbPeriod` 13 → "１３．",
90/// `thaiAlphaParenR` 4 → "ง)", `hindiAlpha1Period` 1 → "क.", `arabic2Minus` 10 → "ي-"
91/// (abjad order), `circleNumWdWhitePlain` 13 → "③" (those circles go to ten).
92pub fn autonum(scheme: &str, n: u32) -> String {
93    const THAI: [char; 41] = [
94        'ก', 'ข', 'ค', 'ง', 'จ', 'ฉ', 'ช', 'ซ', 'ฌ', 'ญ', 'ฎ', 'ฏ', 'ฐ', 'ฑ', 'ฒ', 'ณ', 'ด', 'ต', 'ถ', 'ท', 'ธ', 'น', 'บ', 'ป', 'ผ', 'ฝ', 'พ', 'ฟ',
95        'ภ', 'ม', 'ย', 'ร', 'ล', 'ว', 'ศ', 'ษ', 'ส', 'ห', 'ฬ', 'อ', 'ฮ',
96    ];
97    const HINDI_VOWELS: [char; 11] = ['अ', 'आ', 'इ', 'ई', 'उ', 'ऊ', 'ऋ', 'ए', 'ऐ', 'ओ', 'औ'];
98    const HINDI_CONSONANTS: [char; 33] = [
99        'क', 'ख', 'ग', 'घ', 'ङ', 'च', 'छ', 'ज', 'झ', 'ञ', 'ट', 'ठ', 'ड', 'ढ', 'ण', 'त', 'थ', 'द', 'ध', 'न', 'प', 'फ', 'ब', 'भ', 'म', 'य', 'र', 'ल',
100        'व', 'श', 'ष', 'स', 'ह',
101    ];
102    const ARABIC_ALPHABET: [char; 28] =
103        ['ا', 'ب', 'ت', 'ث', 'ج', 'ح', 'خ', 'د', 'ذ', 'ر', 'ز', 'س', 'ش', 'ص', 'ض', 'ط', 'ظ', 'ع', 'غ', 'ف', 'ق', 'ك', 'ل', 'م', 'ن', 'ه', 'و', 'ي'];
104    const ARABIC_ABJAD: [char; 28] =
105        ['ا', 'ب', 'ج', 'د', 'ه', 'و', 'ز', 'ح', 'ط', 'ي', 'ك', 'ل', 'م', 'ن', 'س', 'ع', 'ف', 'ص', 'ق', 'ر', 'ش', 'ت', 'ث', 'خ', 'ذ', 'ض', 'ظ', 'غ'];
106    const HEBREW: [char; 22] = ['א', 'ב', 'ג', 'ד', 'ה', 'ו', 'ז', 'ח', 'ט', 'י', 'כ', 'ל', 'מ', 'נ', 'ס', 'ע', 'פ', 'צ', 'ק', 'ר', 'ש', 'ת'];
107    let circled = |first: u32, count: u32| char::from_u32(first + n.saturating_sub(1) % count).map(String::from).unwrap_or_else(|| n.to_string());
108    let core = if scheme.starts_with("romanUc") {
109        roman(n, true)
110    } else if scheme.starts_with("romanLc") {
111        roman(n, false)
112    } else if scheme.starts_with("alphaUc") {
113        alpha_num(n, true)
114    } else if scheme.starts_with("alphaLc") {
115        alpha_num(n, false)
116    } else if scheme.starts_with("circleNumWdWhite") {
117        return circled(0x2460, 10);
118    } else if scheme.starts_with("circleNumWdBlack") {
119        return circled(0x2776, 10);
120    } else if scheme.starts_with("circleNum") {
121        return char::from_u32(0x2460 + n.saturating_sub(1).min(19)).map(String::from).unwrap_or_else(|| n.to_string());
122    } else if scheme.starts_with("ea1Chs") || scheme.starts_with("ea1Cht") {
123        chinese_num(n)
124    } else if scheme.starts_with("ea1Jpn") {
125        n.to_string().chars().map(|d| d.to_digit(10).and_then(|v| CJK_DIGITS.get(v as usize).copied()).unwrap_or(d)).collect()
126    } else if scheme.starts_with("arabicDb") {
127        digits_in(n, '０')
128    } else if scheme.starts_with("thaiAlpha") {
129        letters(&THAI, n)
130    } else if scheme.starts_with("thaiNum") {
131        digits_in(n, '๐')
132    } else if scheme.starts_with("hindiAlpha1") {
133        letters(&HINDI_CONSONANTS, n)
134    } else if scheme.starts_with("hindiAlpha") {
135        letters(&HINDI_VOWELS, n)
136    } else if scheme.starts_with("hindiNum") {
137        digits_in(n, '०')
138    } else if scheme.starts_with("arabic1") {
139        letters(&ARABIC_ALPHABET, n)
140    } else if scheme.starts_with("arabic2") {
141        letters(&ARABIC_ABJAD, n)
142    } else if scheme.starts_with("hebrew2") {
143        letters(&HEBREW, n)
144    } else {
145        n.to_string()
146    };
147    // East Asian schemes end with a full-width full stop.
148    let wide = scheme.starts_with("ea1") || scheme.starts_with("arabicDb");
149    if scheme.ends_with("ParenBoth") {
150        format!("({core})")
151    } else if scheme.ends_with("ParenR") {
152        format!("{core})")
153    } else if scheme.ends_with("Period") {
154        format!("{core}{}", if wide { '．' } else { '.' })
155    } else if scheme.ends_with("Minus") {
156        format!("{core}-")
157    } else {
158        core
159    }
160}
