34/// Share of the size the ideographic em box's centre sits above the baseline (CJK fonts put the
35/// em box from 0.88 em above to 0.12 em below it).
36pub const UPRIGHT_CENTRE: f64 = 0.38;
37
38impl GlyphRun {
39    /// The point an upright glyph at pen position (`x`, `y`) turns about: the centre of its em box,
40    /// `advance` wide (`None` when the run is not upright).
41    pub fn upright_centre(&self, gid: u32, x: f64, y: f64) -> Option<(f64, f64)> {
42        if !self.upright {
43            return None;
44        }
45        let adv = self.face.advance(gid) * self.size / self.face.upem.max(1.0);
46        Some((x + adv / 2.0, y - self.size * UPRIGHT_CENTRE))
47    }
48}

86/// How an underline fills its box.
87#[derive(Clone, Copy, Debug, Default, PartialEq)]
88pub enum DecoLine {
89    /// The whole box.
90    #[default]
91    Solid,
92    /// Dashes or dots across the box's height, from its left end.
93    Dashed(Dash),
94    /// A zigzag between the box's top and bottom: `stroke` wide, one wave every `period`.
95    Wave { stroke: f64, period: f64 },
96}
97
98/// Dash patterns of the dotted and dashed underlines, in units of the line's thickness. Measured
99/// on PowerPoint's export: dots and their gaps as long as the line is thick, dashes three
100/// thicknesses long with gaps of two.
101#[derive(Clone, Copy, Debug, PartialEq, Eq)]
102pub enum Dash {
103    Dot,
104    Dash,
105    Long,
106    DotDash,
107    DotDotDash,
108}
109
110impl Dash {
111    /// On, off, on, off… lengths in thicknesses.
112    fn pattern(self) -> &'static [f64] {
113        match self {
114            Dash::Dot => &[1.0, 1.0],
115            Dash::Dash => &[3.0, 2.0],
116            Dash::Long => &[6.0, 2.0],
117            Dash::DotDash => &[3.0, 2.0, 1.0, 2.0],
118            Dash::DotDotDash => &[3.0, 2.0, 1.0, 2.0, 1.0, 2.0],
119        }
120    }
121}
122
123impl Deco {
124    /// The filled boxes of a solid or dashed line (dashes start at the box's left end). A wavy
125    /// line gives a thin box along its middle, for drawing that can't stroke a path.
126    pub fn pieces(&self) -> Vec<Rect> {
127        let r = self.rect;
128        match self.line {
129            DecoLine::Solid => vec![r],
130            DecoLine::Wave { stroke, .. } => {
131                let c = (r.y0 + r.y1) / 2.0;
132                vec![Rect::new(r.x0, c - stroke / 2.0, r.x1, c + stroke / 2.0)]
133            }
134            DecoLine::Dashed(d) => {
135                let unit = r.height().max(0.1);
136                let pat = d.pattern();
137                let mut out = Vec::new();
138                let mut x = r.x0;
139                let mut k = 0usize;
140                while x < r.x1 && out.len() < 10_000 {
141                    let len = pat.get(k % pat.len()).copied().unwrap_or(1.0) * unit;
142                    if k.is_multiple_of(2) {
143                        out.push(Rect::new(x, r.y0, (x + len).min(r.x1), r.y1));
144                    }
145                    x += len;
146                    k += 1;
147                }
148                out
149            }
150        }
151    }
152
153    /// The zigzag of a wavy underline, left to right from the box's top, and its stroke width;
154    /// None for other lines.
155    pub fn wave(&self) -> Option<(Vec<Point>, f64)> {
156        let DecoLine::Wave { stroke, period } = self.line else { return None };
157        let r = self.rect;
158        let half = (period / 2.0).max(0.1);
159        let n = ((r.width().max(0.0) / half).ceil() as usize).min(20_000);
160        let pts = (0..=n)
161            .map(|i| {
162                let x = (r.x0 + i as f64 * half).min(r.x1);
163                Point::new(x, if i % 2 == 0 { r.y0 } else { r.y1 })
164            })
165            .collect();
166        Some((pts, stroke))
167    }
168}

170/// An underline style (`a:rPr/@u`).
171#[derive(Clone, Copy, Debug, PartialEq)]
172pub(crate) enum Underline {
173    Single,
174    Double,
175    Heavy,
176    /// Dotted or dashed, heavy or not.
177    Dashed(Dash, bool),
178    /// Wavy, heavy or not.
179    Wavy(bool),
180}
181
182impl Underline {
183    pub(crate) fn from_xml(u: &str) -> Option<Self> {
184        Some(match u {
185            "none" => return None,
186            // PowerPoint underlines the spaces of a "words" underline too.
187            "words" => Underline::Single,
188            // At text sizes PowerPoint's double wave reads as two straight lines.
189            "dbl" | "wavyDbl" => Underline::Double,
190            "heavy" => Underline::Heavy,
191            "dotted" => Underline::Dashed(Dash::Dot, false),
192            "dottedHeavy" => Underline::Dashed(Dash::Dot, true),
193            "dash" => Underline::Dashed(Dash::Dash, false),
194            "dashHeavy" => Underline::Dashed(Dash::Dash, true),
195            "dashLong" => Underline::Dashed(Dash::Long, false),
196            "dashLongHeavy" => Underline::Dashed(Dash::Long, true),
197            "dotDash" => Underline::Dashed(Dash::DotDash, false),
198            "dotDashHeavy" => Underline::Dashed(Dash::DotDash, true),
199            "dotDotDash" => Underline::Dashed(Dash::DotDotDash, false),
200            "dotDotDashHeavy" => Underline::Dashed(Dash::DotDotDash, true),
201            "wavy" => Underline::Wavy(false),
202            "wavyHeavy" => Underline::Wavy(true),
203            _ => Underline::Single,
204        })
205    }
206
207    /// The boxes of this underline below the baseline (top, bottom, fill), from the font's
208    /// underline position `p` and thickness `t` at the run's size. Measured on PowerPoint's export
209    /// (Arial 96 pt: p = 10.2, t = 7.0): single at p..p+t; heavy from p-t to p+0.55t; double as two
210    /// lines 0.55t thick spanning p-t..p+t; wavy a thin zigzag from p-0.6t to p+1.1t, a wave every
211    /// 3.9t.
212    pub(crate) fn boxes(self, p: f64, t: f64) -> Vec<(f64, f64, DecoLine)> {
213        let heavy = (p - t, p + 0.55 * t);
214        match self {
215            Underline::Single => vec![(p, p + t, DecoLine::Solid)],
216            Underline::Heavy => vec![(heavy.0, heavy.1, DecoLine::Solid)],
217            Underline::Double => vec![(p - t, p - 0.45 * t, DecoLine::Solid), (p + 0.45 * t, p + t, DecoLine::Solid)],
218            Underline::Dashed(d, false) => vec![(p, p + t, DecoLine::Dashed(d))],
219            Underline::Dashed(d, true) => vec![(heavy.0, heavy.1, DecoLine::Dashed(d))],
220            Underline::Wavy(h) => {
221                let stroke = (t * 0.15).max(0.4) * if h { 2.0 } else { 1.0 };
222                vec![(p - 0.6 * t, p + 1.1 * t, DecoLine::Wave { stroke, period: t * 3.9 })]
223            }
224        }
225    }
