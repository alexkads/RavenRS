4pub(crate) fn measure_str(face: &FontFace, size: f64, s: &str) -> (f64, Vec<(u32, f64)>) {
5    let g = crate::fonts::shape(face, s, &[], |c| c);
6    let k = size / face.upem.max(1.0);
7    let mut x = 0.0;
8    let mut out = vec![];
9    for gl in g {
10        out.push((gl.gid, x + gl.x_offset as f64 * k));
11        x += gl.x_advance as f64 * k;
12    }
13    (x, out)
14}

16/// Single line spacing as a multiple of the font size. PowerPoint spaces single-spaced lines 1.2
17/// times the size apart in every font (observed on twenty Latin, CJK, Thai and Indic fonts at 18
18/// and 40 pt), whatever the font's own ascent and descent.
19///
20/// Caveat: this was measured on one Windows install of desktop PowerPoint. It may be a Windows or
21/// machine setting; PowerPoint on macOS or the web may space lines by the font's own metrics.
22/// Keep it a single tunable value, and re-measure on the other platform before changing it.
23pub(crate) const LINE_PITCH: f64 = 1.2;
24
25/// A line's height above and below the baseline at `size`: [`LINE_PITCH`] × size, split where
26/// the font's Windows ascent and descent put the baseline.
27pub(crate) fn line_metrics(face: &FontFace, size: f64) -> (f64, f64) {
28    let h = size * LINE_PITCH;
29    (h * face.line_split, h * (1.0 - face.line_split))
30}

32/// Hangul syllables and jamo: PowerPoint wraps Korean at word boundaries (spaces), not between
33/// syllables.
34fn is_hangul(c: char) -> bool {
35    matches!(c, '\u{1100}'..='\u{11FF}' | '\u{3130}'..='\u{318F}' | '\u{A960}'..='\u{A97F}' | '\u{AC00}'..='\u{D7FF}')

38/// East Asian commas and full stops, the punctuation PowerPoint lets hang past the end of a line
39/// (`hangingPunct`). Closing brackets don't hang: "「。」" overflowing by its "」" moves to the next
40/// line whole.
41fn is_hanging_punct(c: char) -> bool {
42    matches!(c, '、' | '。' | '，' | '．' | '､' | '｡')
43}

85/// A character that continues a Latin-style word: a letter or digit that Unicode does not break
86/// between, as opposed to CJK ideographs, kana, spaces, punctuation and symbols such as `&` or `×`
87/// (which share the AL and AI line-break classes with letters).
88fn is_word_char(c: char) -> bool {
89    use unicode_linebreak::BreakClass::{Alphabetic, Ambiguous, HebrewLetter, Numeric};
90    matches!(unicode_linebreak::break_property(c as u32), Alphabetic | Ambiguous | HebrewLetter | Numeric) && c.is_alphanumeric()
91}

93/// Break opportunities: allowed[i] = a line may start at char i. Unicode line breaking, then
94/// PowerPoint's differences: without `latinLnBrk` a word doesn't break after `/`, `\`, `?`, `!`, `}`
95/// or `+` inside a word (measured in PowerPoint 2016 for Windows; before an ideograph or a space
96/// the Unicode break stays), and Korean breaks only between words.
97pub(crate) fn break_opportunities(text: &str, latin_break: bool) -> (Vec<bool>, Vec<bool>) {
98    let n = text.chars().count();
99    let mut allowed = vec![false; n + 1];
100    let mut mandatory = vec![false; n + 1];
101    let mut byte_to_char = std::collections::HashMap::new();
102    for (ci, (b, _)) in text.char_indices().enumerate() {
103        byte_to_char.insert(b, ci);
104    }
105    byte_to_char.insert(text.len(), n);
106    for (b, op) in unicode_linebreak::linebreaks(text) {
107        if let Some(&ci) = byte_to_char.get(&b) {
108            if let Some(a) = allowed.get_mut(ci) {
109                *a = true;
110            }
111            if op == unicode_linebreak::BreakOpportunity::Mandatory
112                && let Some(m) = mandatory.get_mut(ci)
113            {
114                *m = true;
115            }
116        }
117    }
118    let chars: Vec<char> = text.chars().collect();
119    for (i, w) in chars.windows(2).enumerate() {
120        let (prev, next) = (w[0], w[1]);
121        // PowerPoint never breaks after a zero-width space, nor before an em dash (only after it).
122        let keep = if (is_hangul(prev) && is_hangul(next)) || prev == '\u{200B}' || next == '\u{2014}' || is_cj_nonstarter(next) {
123            false
124        } else {
125            latin_break || !matches!(prev, '/' | '\\' | '?' | '!' | '}' | '+') || !is_word_char(next)
126        };
127        if !keep
128            && !mandatory.get(i + 1).copied().unwrap_or(false)
129            && let Some(a) = allowed.get_mut(i + 1)
130        {
131            *a = false;
132        }
133    }
134    (allowed, mandatory)
135}

155pub(crate) fn break_lines(p: &ParaShaped, params: &BreakParams) -> Vec<RawLine> {
156    let BreakParams { first_w, rest_w, wrap, tab_stops, default_tab, x0_first, x0_rest } = *params;
157    let text: String = p.cells.iter().map(|c| c.ch).collect();
158    let (allowed, _) = break_opportunities(&text, p.props.latin_line_break.unwrap_or(false));
159    let hang = p.props.hanging_punct.unwrap_or(false);
160    let n = p.cells.len();
161    let mut lines = vec![];
162    let mut start = 0;
163    if n == 0 {
164        return vec![RawLine { start: 0, end: 0, next: 0 }];
165    }
166    while start < n {
167        let avail = if lines.is_empty() { first_w } else { rest_w };
168        let x0 = if lines.is_empty() { x0_first } else { x0_rest };
169        let mut x = 0.0;
170        let mut last_break: Option<usize> = None;
171        let mut i = start;
172        let mut end = n;
173        let mut next = n;
174        while i < n {
175            let c = p.cells.get(i).map(|c| c.ch).unwrap_or(' ');
176            if c == '\u{b}' || c == '\n' {
177                end = i;
178                next = i + 1;
179                break;
180            }
181            let adv = if c == '\t' {
182                tab_advance(x0 + x, tab_stops, default_tab, tab_segment(&p.cells, i))
183            } else {
184                p.cells.get(i).map(|c| c.adv).unwrap_or(0.0)
185            };
186            let breakable = |j: usize| p.cells.get(j).is_none_or(|c| c.breakable(j));
187            if i > start && allowed.get(i).copied().unwrap_or(false) && breakable(i) {
188                last_break = Some(i);
189            }
190            // Closing East Asian punctuation hangs past the end instead of wrapping
191            // (`hangingPunct`); the line breaks after it.
192            if wrap && hang && x + adv > avail + 0.01 && x <= avail + 0.01 && is_hanging_punct(c) && i > start {
193                end = i + 1;
194                next = i + 1;
195                break;
196            }
197            // A tab that would run past the end moves to the next line (and to its first stop there).
198            if wrap && c == '\t' && x + adv > avail + 0.01 && i > start {
199                end = i;
200                next = i;
201                while end > start && p.cells.get(end - 1).is_some_and(|c| c.ch == ' ') {
202                    end -= 1;
203                }
204                break;
205            }
206            if wrap && x + adv > avail + 0.01 && !c.is_whitespace() && i > start {
207                // Break at the last opportunity, or force mid-word (between clusters).
208                let b = last_break
209                    .filter(|b| *b > start)
210                    .unwrap_or_else(|| (start + 1..=i).rev().find(|&j| breakable(j)).or_else(|| (i + 1..n).find(|&j| breakable(j))).unwrap_or(n));
211                end = b;
212                next = b;
213                // Trailing spaces belong to the line but don't count.
214                while end > start && p.cells.get(end - 1).is_some_and(|c| c.ch == ' ') {
215                    end -= 1;
216                }
217                break;
218            }
219            x += adv;
220            i += 1;
221        }
222        if i >= n {
223            end = n;
224            next = n;
225        }
226        lines.push(RawLine { start, end, next });
227        if next <= start {
228            // No progress: force one char.
229            let last = lines.len() - 1;
230            let to = (start + 1..n).find(|&j| p.cells.get(j).is_none_or(|c| c.breakable(j))).unwrap_or(n);
231            if let Some(l) = lines.get_mut(last) {
232                l.end = to;
233                l.next = to;
234            }
235            start = to;
236        } else {
237            start = next;
238        }
239        if next == n && p.cells.last().is_some_and(|c| c.ch == '\u{b}') && start >= n {
240            // A trailing line break opens an empty last line.
241            lines.push(RawLine { start: n, end: n, next: n });
242        }
243    }
244    lines
245}

247/// How a tab stop aligns the text after it (`a:tab/@algn`).
248#[derive(Clone, Copy, Debug, PartialEq)]
249enum TabAlign {
250    Left,
251    Center,
252    Right,
253    /// The first "." of the text sits at the stop (text without one ends there).
254    Decimal,
255}

258pub(crate) struct Tab {
259    pos: f64,
260    align: TabAlign,
261}

263/// The paragraph's tab stops in position order (a stop at no finite position is left out), so
264/// [`tab_advance`] finds the next one by binary search instead of a scan per tab.
265pub(crate) fn tabs_of(props: &ParaProps) -> Vec<Tab> {
266    let mut tabs: Vec<Tab> = props
267        .tabs
268        .as_ref()
269        .map(|t| {
270            t.iter()
271                .filter(|s| s.pos.is_finite())
272                .map(|s| Tab {
273                    pos: s.pos,
274                    align: match s.align.as_str() {
275                        "ctr" => TabAlign::Center,
276                        "r" => TabAlign::Right,
277                        "dec" => TabAlign::Decimal,
278                        _ => TabAlign::Left,
279                    },
280                })
281                .collect()
282        })
283        .unwrap_or_default();
284    tabs.sort_by(|a, b| a.pos.total_cmp(&b.pos));
285    tabs
286}

288/// Width of the text after the tab at cell `i` up to the next tab or line break, and the width of
289/// that text before its first ".".
290pub(crate) fn tab_segment(cells: &[Cell], i: usize) -> (f64, f64) {
291    let mut w = 0.0;
292    let mut dec = None;
293    for c in cells.iter().skip(i + 1) {
294        if matches!(c.ch, '\t'..='\u{b}') {
295            break;
296        }
297        if c.ch == '.' && dec.is_none() {
298            dec = Some(w);
299        }
300        w += c.adv;
301    }
302    (w, dec.unwrap_or(w))
303}

305/// Advance of a tab at `x` (from the start edge) to the next stop, placing the text after it
306/// (`seg`: its width, and its width before the decimal point) as the stop aligns it but never
307/// moving back, as PowerPoint does; past the last stop, to the next default stop.
308pub(crate) fn tab_advance(x: f64, stops: &[Tab], default_tab: f64, seg: (f64, f64)) -> f64 {
309    // `stops` is in position order (`tabs_of`): the next stop is the first past x.
310    // A NaN `x` (a NaN spacing upstream) skips the lookup and takes the finite default-stop path.
311    if let Some(s) = stops.get(stops.partition_point(|s| s.pos <= x + 0.5)).filter(|_| !x.is_nan()) {
312        let to = s.pos - x;
313        return match s.align {
314            TabAlign::Left => to,
315            TabAlign::Right => (to - seg.0).max(0.0),
316            TabAlign::Center => (to - seg.0 / 2.0).max(0.0),
317            TabAlign::Decimal => (to - seg.1).max(0.0),
318        };
319    }
320    let d = if default_tab > 1.0 { default_tab } else { 72.0 };
321    let next = ((x / d).floor() + 1.0) * d;
322    (next - x).max(1.0)
323}
