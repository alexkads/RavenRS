15/// Pairs of (shape on `a`, shape on `b`) that Morph animates from one to the other: first by
16/// `!!name` (forced match), then by exact name, then by the same kind and text, then by the same
17/// kind and preset geometry if neither carries text. A text box with neither a name nor its text on
18/// the other slide stays unpaired and fades in place, as in PowerPoint (#119). Each shape is used
19/// once; top-level shapes only (groups move whole).
20///
21/// Each pass compares a key of the two shapes, so the shapes of `b` are indexed by key once per
22/// pass: matching costs the sum of the two shape lists, not their product.
23pub fn morph_pairs(a: &Slide, b: &Slide) -> Vec<(ShapeId, ShapeId)> {
24    let mut pairs: Vec<(ShapeId, ShapeId)> = Vec::new();
25    let mut used_a = vec![false; a.shapes.len()];
26    let mut used_b = vec![false; b.shapes.len()];
27    // Each shape's text, worked out once rather than inside every comparison.
28    let text_a: Vec<String> = a.shapes.iter().map(text_of).collect();
29    let text_b: Vec<String> = b.shapes.iter().map(text_of).collect();
30    let forced = |s: &Shape, _: &str| (s.name.starts_with("!!")).then(|| MorphKey::Name(s.name.clone()));
31    let named = |s: &Shape, _: &str| (!s.name.is_empty()).then(|| MorphKey::Name(s.name.clone()));
32    let same_text = |s: &Shape, t: &str| (!t.trim().is_empty()).then(|| MorphKey::Text(s.kind_name(), t.to_string()));
33    let same_geom = |s: &Shape, t: &str| match s.geom.preset_name() {
34        Some(g) if t.trim().is_empty() => Some(MorphKey::Geom(s.kind_name(), g.to_string())),
35        _ => None,
36    };
37    let passes: [&dyn Fn(&Shape, &str) -> Option<MorphKey>; 4] = [&forced, &named, &same_text, &same_geom];
38    for key in passes {
39        // Unused shapes of `b` by key, in slide order: the front of each queue is the first unused
40        // match, which is the one a search through `b` would find.
41        let mut index: HashMap<MorphKey, VecDeque<usize>> = HashMap::new();
42        for (j, sb) in b.shapes.iter().enumerate() {
43            if used_b.get(j).copied().unwrap_or(true) {
44                continue;
45            }
46            if let Some(k) = key(sb, text_b.get(j).map_or("", String::as_str)) {
47                index.entry(k).or_default().push_back(j);
48            }
49        }
50        if index.is_empty() {
51            continue;
52        }
53        for (i, sa) in a.shapes.iter().enumerate() {
54            if used_a.get(i).copied().unwrap_or(true) {
55                continue;
56            }
57            let Some(k) = key(sa, text_a.get(i).map_or("", String::as_str)) else { continue };
58            let Some(j) = index.get_mut(&k).and_then(VecDeque::pop_front) else { continue };
59            let Some(sb) = b.shapes.get(j) else { continue };
60            if let Some(u) = used_a.get_mut(i) {
61                *u = true;
62            }
63            if let Some(u) = used_b.get_mut(j) {
64                *u = true;
65            }
66            pairs.push((sa.id, sb.id));
67        }
68    }
69    pairs
70}

72/// What one [`morph_pairs`] pass matches shapes on.
73#[derive(Clone, Debug, PartialEq, Eq, Hash)]
74enum MorphKey {
75    Name(String),
76    Text(&'static str, String),
77    Geom(&'static str, String),
78}

80fn lerp(a: f64, b: f64, t: f64) -> f64 {
81    let v = a + (b - a) * t;
82    if v.is_finite() {
83        v
84    } else if t < 0.5 {
85        a
86    } else {
87        b
88    }
89}

91/// Interpolate a box from `a` to `b` at `t` (0..1): position and size linearly, rotation along the
92/// shortest way round; flips switch at the midpoint.
93pub fn morph_xfrm(a: Xfrm, b: Xfrm, t: f64) -> Xfrm {
94    let t = clampf(t, 0.0, 1.0, 0.0);
95    if t >= 1.0 {
96        return b;
97    }
98    if t <= 0.0 {
99        return a;
100    }
101    let ra = if a.rot.is_finite() { a.rot } else { 0.0 };
102    let rb = if b.rot.is_finite() { b.rot } else { 0.0 };
103    let mut d = (rb - ra).rem_euclid(360.0);
104    if d > 180.0 {
105        d -= 360.0;
106    }
107    let rot = (ra + d * t).rem_euclid(360.0);
108    let late = t >= 0.5;
109    Xfrm {
110        x: lerp(a.x, b.x, t),
111        y: lerp(a.y, b.y, t),
112        w: lerp(a.w, b.w, t),
113        h: lerp(a.h, b.h, t),
114        rot: if rot.is_finite() { rot } else { 0.0 },
115        flip_h: if late { b.flip_h } else { a.flip_h },
116        flip_v: if late { b.flip_v } else { a.flip_v },
117    }
118}

120/// One frame of a Morph transition, ready for the renderer's `render_blend`: the backdrops of the
121/// two slides cross-fade by [`MorphFrame::mix`], the shapes are drawn in order on top.
122#[derive(Clone, Debug, Default)]
123pub struct MorphFrame {
124    /// Shapes with fresh ids, each flagged `true` when it comes from the old slide (it resolves
125    /// its placeholder, theme and style against that slide).
126    pub shapes: Vec<(Shape, bool)>,
127    /// Opacity of the frame's shapes by id; shapes not listed are opaque.
128    pub opacity: Vec<(ShapeId, f64)>,
129    pub mix: f64,
130    /// With the Words or Characters option: pairs of frame shapes whose texts morph by words or
131    /// characters instead of cross-fading (the renderer's `TextMorph`).
132    pub text: Vec<MorphText>,
133    /// Frame shapes whose outline turns from another geometry into their own (the renderer's
134    /// `PathMorph`).
135    pub paths: Vec<MorphPath>,
136}

138/// A frame shape whose outline morphs from `from` (the old shape's geometry) into its own.
139#[derive(Clone, Debug, PartialEq)]
140pub struct MorphPath {
141    pub id: ShapeId,
142    pub from: Geom,
143    /// The boxes of the old and the new shape on their own slides.
144    pub from_box: Xfrm,
145    pub to_box: Xfrm,
146}

148/// Two frame shapes whose texts morph by words or characters.
149#[derive(Clone, Copy, Debug, PartialEq)]
150pub struct MorphText {
151    /// Frame shape ids: from the old slide, from the new slide.
152    pub old: ShapeId,
153    pub new: ShapeId,
154    /// The boxes of the two shapes on their own slides.
155    pub from: Xfrm,
156    pub to: Xfrm,
157    pub chars: bool,
158}

160impl MorphFrame {
161    /// Opacity of shape `id` of the frame.
162    pub fn opacity_of(&self, id: ShapeId) -> f64 {
163        self.opacity.iter().find(|(s, _)| *s == id).map(|(_, o)| *o).unwrap_or(1.0)
164    }
165}

167fn mix_color(a: Rgba, b: Rgba, t: f64) -> Rgba {
168    let m = |x: u8, y: u8| clampf(lerp(x as f64, y as f64, t), 0.0, 255.0, 0.0).round() as u8;
169    Rgba { r: m(a.r, b.r), g: m(a.g, b.g), b: m(a.b, b.b), a: m(a.a, b.a) }
170}

187/// The content of a shape apart from what Morph interpolates (identity, box and fill).
188fn content(s: &Shape) -> Shape {
189    Shape { id: ShapeId(0), name: String::new(), xfrm: None, fill: None, ..s.clone() }
190}

192/// Whether two shapes differ in their geometry only, and it is drawn as an outline (shapes,
193/// pictures, connectors), so the outline can morph.
194fn reshaped(a: &Shape, b: &Shape) -> bool {
195    let outlined = |s: &Shape| matches!(s.kind, ShapeKind::Shape | ShapeKind::Picture { .. } | ShapeKind::Connector { .. });
196    a.geom != b.geom && outlined(a) && outlined(b) && Shape { geom: Geom::default(), ..content(a) } == Shape { geom: Geom::default(), ..content(b) }
197}

199/// The content of a shape apart from what Morph interpolates, with a picture's crop left out.
200fn uncropped(s: &Shape) -> Shape {
201    let mut c = content(s);
202    if let ShapeKind::Picture { fill } = &mut c.kind {
203        fill.crop = [0.0; 4];
204    }
205    c
206}

208fn crop_of(s: &Shape) -> [f64; 4] {
209    match &s.kind {
210        ShapeKind::Picture { fill } => fill.crop,
211        _ => [0.0; 4],
212    }
213}

215/// Whether two pictures differ in their crop only (apart from box and fill), so the crop can morph.
216fn recropped(a: &Shape, b: &Shape) -> bool {
217    matches!(a.kind, ShapeKind::Picture { .. }) && crop_of(a) != crop_of(b) && uncropped(a) == uncropped(b)
218}

220/// The crop (left, top, right, bottom) of a picture morphing from box `xa` cropped by `ca` to box
221/// `xb` cropped by `cb`, when its box is `x` at progress `t`: the whole image moves and scales
222/// linearly between where it lies on either slide, and the crop is what of it `x` shows. Falls back
223/// to blending the crops when a crop leaves nothing visible.
224pub fn morph_crop(ca: [f64; 4], xa: Xfrm, cb: [f64; 4], xb: Xfrm, x: Xfrm, t: f64) -> [f64; 4] {
225    // Along one axis: (box start, box size, crop before, crop after) → the crops at `t`.
226    let axis = |(pa, sa, la, ha): (f64, f64, f64, f64), (pb, sb, lb, hb): (f64, f64, f64, f64), (p, s): (f64, f64)| {
227        let (va, vb) = (1.0 - la - ha, 1.0 - lb - hb);
228        if va <= 1e-6 || vb <= 1e-6 {
229            return (lerp(la, lb, t), lerp(ha, hb, t));
230        }
231        // The whole image: its extent and start on either slide.
232        let (ea, eb) = (sa / va, sb / vb);
233        let e = lerp(ea, eb, t);
234        if !e.is_finite() || e.abs() <= 1e-9 {
235            return (lerp(la, lb, t), lerp(ha, hb, t));
236        }
237        let o = lerp(pa - la * ea, pb - lb * eb, t);
238        let lo = (p - o) / e;
239        (lo, 1.0 - lo - s / e)
240    };
241    let (l, r) = axis((xa.x, xa.w, ca[0], ca[2]), (xb.x, xb.w, cb[0], cb[2]), (x.x, x.w));
242    let (tp, bt) = axis((xa.y, xa.h, ca[1], ca[3]), (xb.y, xb.h, cb[1], cb[3]), (x.y, x.h));
243    let c = [l, tp, r, bt];
244    if c.iter().all(|v| v.is_finite()) { c } else { cb }
245}

247/// Morph frame from slide `a` to slide `b` at progress `t` (0..1, eased here). Paired shapes
248/// ([`morph_pairs`]) move along [`morph_xfrm`]; when they differ only in solid fill colour the
249/// colour blends, when pictures differ only in crop the crop morphs ([`morph_crop`]), when only
250/// in geometry (and fill colour) the outline morphs ([`MorphFrame::paths`]),
251/// otherwise the new shape fades in under the fading old one on the same moving
252/// box; with the transition option `words` or `characters` their texts are listed in
253/// [`MorphFrame::text`] to morph word by word or character by character. Unpaired shapes of `a` fade out, unpaired shapes of `b` fade in, and shapes of `b` for
254/// which `hidden_b` is true (entrance animations, hidden media) stay out of the frame.
255pub fn morph_frame(pres: &Presentation, a: &Slide, b: &Slide, t: f64, hidden_b: &dyn Fn(ShapeId) -> bool) -> MorphFrame {
256    let s = smooth(clampf(t, 0.0, 1.0, 0.0));
257    let ca = Ctx::for_slide(pres, a);
258    let cb = Ctx::for_slide(pres, b);
259    let pairs: Vec<(ShapeId, ShapeId)> = morph_pairs(a, b).into_iter().filter(|(_, ib)| !hidden_b(*ib)).collect();
260    let mut next = 1u32;
261    walk(&a.shapes, &mut |x, _| next = next.max(x.id.0.saturating_add(1)));
262    walk(&b.shapes, &mut |x, _| next = next.max(x.id.0.saturating_add(1)));
263    let mut f = MorphFrame { mix: s, ..Default::default() };
264    let by = b.transition.as_ref().map(|tr| tr.option.as_str()).unwrap_or("");
265    let (by_text, chars) = (by == "words" || by == "characters", by == "characters");
266    let has_text = |x: &Shape| x.text.as_ref().is_some_and(|t| !t.is_empty()) && !matches!(x.kind, ShapeKind::Group { .. } | ShapeKind::Table(_));
267    let mut text = Vec::new();
268    let mut paths = Vec::new();
269    let mut push = |mut sh: Shape, old: bool, op: f64| {
270        let id = ShapeId(next);
271        sh.id = id;
272        next = next.saturating_add(1);
273        if op < 1.0 {
274            f.opacity.push((id, op));
275        }
276        f.shapes.push((sh, old));
277        id
278    };
279    // Lookups by id, built once (the first shape or pair with an id wins, as a search would).
280    let paired_a: HashSet<ShapeId> = pairs.iter().map(|(ia, _)| *ia).collect();
281    let mut partner: HashMap<ShapeId, ShapeId> = HashMap::with_capacity(pairs.len());
282    for (ia, ib) in &pairs {
283        partner.entry(*ib).or_insert(*ia);
284    }
285    let mut a_by_id: HashMap<ShapeId, &Shape> = HashMap::with_capacity(a.shapes.len());
286    for x in &a.shapes {
287        a_by_id.entry(x.id).or_insert(x);
288    }
289    for sa in &a.shapes {
290        if !paired_a.contains(&sa.id) {
291            push(sa.clone(), true, 1.0 - s);
292        }
293    }
294    for sb in &b.shapes {
295        let pair = partner.get(&sb.id).and_then(|ia| a_by_id.get(ia).copied());
296        let Some(sa) = pair else {
297            if !hidden_b(sb.id) {
298                push(sb.clone(), false, s);
299            }
300            continue;
301        };
302        let (xa, xb) = (box_of(ca.as_ref(), sa), box_of(cb.as_ref(), sb));
303        let x = match (xa, xb) {
304            (Some(xa), Some(xb)) => Some(morph_xfrm(xa, xb, s)),
305            (_, xb) => xb,
306        };
307        let reshape = reshaped(sa, sb);
308        let recrop = recropped(sa, sb);
309        if reshape || recrop || content(sa) == content(sb) {
310            let mut m = Shape { xfrm: x, ..sb.clone() };
311            if recrop && let ShapeKind::Picture { fill } = &mut m.kind {
312                fill.crop = match (xa, xb, x) {
313                    (Some(xa), Some(xb), Some(x)) => morph_crop(crop_of(sa), xa, crop_of(sb), xb, x, s),
314                    _ => {
315                        let (ca, cb) = (crop_of(sa), crop_of(sb));
316                        [lerp(ca[0], cb[0], s), lerp(ca[1], cb[1], s), lerp(ca[2], cb[2], s), lerp(ca[3], cb[3], s)]
317                    }
318                };
319            }
320            if let (Some(fa), Some(fb)) = (solid_color(ca.as_ref(), sa), solid_color(cb.as_ref(), sb))
321                && fa != fb
322            {
323                m.fill = Some(Fill::solid(ColorRef::rgb(mix_color(fa, fb, s))));
324            }
325            let id = push(m, false, 1.0);
326            if reshape && let (Some(from_box), Some(to_box)) = (xa, xb) {
327                paths.push(MorphPath { id, from: sa.geom.clone(), from_box, to_box });
328            }
329        } else {
330            let new = push(Shape { xfrm: x, ..sb.clone() }, false, s);
331            let old = push(Shape { xfrm: x, ..sa.clone() }, true, 1.0 - s);
332            // The same text in shapes that differ otherwise (a title slide's centred title and a
333            // content slide's title: other placeholder, size, alignment) moves and resizes word
334            // by word even with the Objects option, as PowerPoint does, instead of cross-fading.
335            let same_text = text_of(sa).trim() == text_of(sb).trim();
336            if (by_text || same_text)
337                && has_text(sa)
338                && has_text(sb)
339                && let (Some(from), Some(to)) = (box_of(ca.as_ref(), sa), box_of(cb.as_ref(), sb))
340            {
341                text.push(MorphText { old, new, from, to, chars });
342            }
343        }
344    }
345    f.text = text;
346    f.paths = paths;
347    f
