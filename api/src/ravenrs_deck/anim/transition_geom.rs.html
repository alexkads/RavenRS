10/// Smoothstep ease in-out (PowerPoint's fades).
11pub(crate) fn ss(x: f64) -> f64 {
12    smooth(x)
13}
14
15/// Quadratic ease in-out: velocity ramps up for the first half and down for the second
16/// (PowerPoint's moves: Push, Cover, Uncover, Pan…).
17pub(crate) fn q2(x: f64) -> f64 {
18    ease(x, 0.5, 0.5, 0.0)
19}
20
21/// Power ease in-out of degree `n` (2: quadratic, 3: cubic, 4: quartic).
22pub(crate) fn pow_inout(x: f64, n: f64) -> f64 {
23    let x = x.clamp(0.0, 1.0);
24    if x < 0.5 { 0.5 * (2.0 * x).powf(n) } else { 1.0 - 0.5 * (2.0 - 2.0 * x).powf(n) }
25}
26
27/// Quadratic ease-in.
28pub(crate) fn ein(x: f64) -> f64 {
29    let x = x.clamp(0.0, 1.0);
30    x * x
31}
32
33/// Quadratic ease-out.
34pub(crate) fn eout(x: f64) -> f64 {
35    let x = 1.0 - x.clamp(0.0, 1.0);
36    1.0 - x * x
37}

39/// Progress through the window [a, b] of t, clamped to 0..1.
40pub(crate) fn win(t: f64, a: f64, b: f64) -> f64 {
41    if b <= a {
42        return if t >= b { 1.0 } else { 0.0 };
43    }
44    ((t - a) / (b - a)).clamp(0.0, 1.0)
45}

49pub(crate) fn hash64(mut x: u64) -> u64 {
50    x = x.wrapping_add(0x9E37_79B9_7F4A_7C15);
51    x = (x ^ (x >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
52    x = (x ^ (x >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
53    x ^ (x >> 31)
54}

56/// Deterministic pseudo-random value in [0, 1).
57pub(crate) fn rnd(i: u64) -> f64 {
58    (hash64(i) >> 11) as f64 / (1u64 << 53) as f64
59}

89/// Rotate a point about a pivot by `deg` (clockwise on screen), in a slide `a` wide per unit of
90/// height, so angles and shapes stay true.
91pub(crate) fn rot(p: P, c: P, deg: f64, a: f64) -> P {
92    let (s, co) = deg.to_radians().sin_cos();
93    let (x, y) = ((p[0] - c[0]) * a, p[1] - c[1]);
94    [c[0] + (x * co - y * s) / a, c[1] + x * s + y * co]
95}

97pub(crate) fn rot_quad(q: Q, c: P, deg: f64, a: f64) -> Q {
98    [rot(q[0], c, deg, a), rot(q[1], c, deg, a), rot(q[2], c, deg, a), rot(q[3], c, deg, a)]
99}

101/// Motion direction of an option: l, r, u, d and diagonals.
102pub(crate) fn dir(opt: &str, default: (f64, f64)) -> (f64, f64) {
103    let mut d = (0.0, 0.0);
104    if opt.contains('l') {
105        d.0 = -1.0;
106    }
107    if opt.contains('r') {
108        d.0 = 1.0;
109    }
110    if opt.contains('u') {
111        d.1 = -1.0;
112    }
113    if opt.contains('d') {
114        d.1 = 1.0;
115    }
116    if d == (0.0, 0.0) { default } else { d }
117}

119/// A rectangle with a soft border `f` wide (slide units) fading out to nothing: the rectangle at
120/// full opacity, four edge ramps and four corner ramps. `src` maps onto `dst`; the border samples
121/// the texture's edge.
122pub(crate) fn soft_rect(source: Source, src: R, dst: R, f: f64, alpha: f64) -> Vec<Layer> {
123    let mut v = vec![Layer::rect(source, src, dst, alpha)];
124    if f <= 1e-6 {
125        return v;
126    }
127    let (l, t, r, b) = (dst[0], dst[1], dst[2], dst[3]);
128    let (fx, fy) = (f, f * 16.0 / 9.0);
129    let uv_at = |x: f64, y: f64| -> P {
130        let u = if r > l { src[0] + (x.clamp(l, r) - l) / (r - l) * (src[2] - src[0]) } else { src[0] };
131        let w = if b > t { src[1] + (y.clamp(t, b) - t) / (b - t) * (src[3] - src[1]) } else { src[1] };
132        [u, w]
133    };
134    let mut q = |pts: Q, fade: [f64; 4]| {
135        let uv = pts.map(|p| uv_at(p[0], p[1]));
136        let xs = uv.map(|p| p[0]);
137        let ys = uv.map(|p| p[1]);
138        let mn = |a: [f64; 4]| a.iter().copied().fold(f64::INFINITY, f64::min);
139        let mx = |a: [f64; 4]| a.iter().copied().fold(f64::NEG_INFINITY, f64::max);
140        v.push(Layer { source, src: [mn(xs), mn(ys), mx(xs), mx(ys)], quad: pts, uv, alpha, fade });
141    };
142    q([[l, t - fy], [r, t - fy], [r, t], [l, t]], [0.0, 0.0, 1.0, 1.0]);
143    q([[l, b], [r, b], [r, b + fy], [l, b + fy]], [1.0, 1.0, 0.0, 0.0]);
144    q([[l - fx, t], [l, t], [l, b], [l - fx, b]], [0.0, 1.0, 1.0, 0.0]);
145    q([[r, t], [r + fx, t], [r + fx, b], [r, b]], [1.0, 0.0, 0.0, 1.0]);
146    q([[l - fx, t - fy], [l, t - fy], [l, t], [l - fx, t]], [0.0, 0.0, 1.0, 0.0]);
147    q([[r, t - fy], [r + fx, t - fy], [r + fx, t], [r, t]], [0.0, 0.0, 0.0, 1.0]);
148    q([[r, b], [r + fx, b], [r + fx, b + fy], [r, b + fy]], [1.0, 0.0, 0.0, 0.0]);
149    q([[l - fx, b], [l, b], [l, b + fy], [l - fx, b + fy]], [0.0, 1.0, 0.0, 0.0]);
150    v
151}

155/// The source shown in place through an opacity field `f(x, y)` (0..1) sampled on an `n` × `m`
156/// grid: each cell is a quad with its corners' opacities. Cells with nothing to show are left out
157/// and runs of fully opaque cells merge, so most frames stay small.
158pub(crate) fn field(source: Source, n: usize, m: usize, f: &dyn Fn(f64, f64) -> f64) -> Vec<Layer> {
159    let (n, m) = (n.clamp(1, 256), m.clamp(1, 256));
160    let mut vals = vec![0.0; (n + 1) * (m + 1)];
161    for j in 0..=m {
162        for i in 0..=n {
163            let v = f(i as f64 / n as f64, j as f64 / m as f64);
164            if let Some(x) = vals.get_mut(j * (n + 1) + i) {
165                *x = if v.is_finite() { v.clamp(0.0, 1.0) } else { 0.0 };
166            }
167        }
168    }
169    let at = |i: usize, j: usize| vals.get(j * (n + 1) + i).copied().unwrap_or(0.0);
170    let mut out = Vec::new();
171    for j in 0..m {
172        let (y0, y1) = (j as f64 / m as f64, (j + 1) as f64 / m as f64);
173        let mut run: Option<usize> = None;
174        for i in 0..=n {
175            let full = i < n && at(i, j) >= 1.0 && at(i + 1, j) >= 1.0 && at(i, j + 1) >= 1.0 && at(i + 1, j + 1) >= 1.0;
176            if full {
177                run.get_or_insert(i);
178                continue;
179            }
180            if let Some(s) = run.take() {
181                out.push(Layer::inplace(source, [s as f64 / n as f64, y0, i as f64 / n as f64, y1], 1.0));
182            }
183            if i == n {
184                break;
185            }
186            let c = [at(i, j), at(i + 1, j), at(i + 1, j + 1), at(i, j + 1)];
187            if c.iter().all(|v| *v <= 0.0) {
188                continue;
189            }
190            out.push(Layer::inplace(source, [i as f64 / n as f64, y0, (i + 1) as f64 / n as f64, y1], 1.0).faded(c));
191        }
192    }
193    out
194}

196/// Opacity of a soft edge at position `p` when the edge's middle is at `c` and it ramps over `w`.
197pub(crate) fn edge(c: f64, p: f64, w: f64) -> f64 {
198    if w <= 1e-9 {
199        return if p <= c { 1.0 } else { 0.0 };
200    }
201    ((c - p) / w + 0.5).clamp(0.0, 1.0)
202}

204/// Point on the unit square's boundary seen from its centre at angle `ang` (radians, clockwise
205/// from 12 o'clock, measured on a slide `a` wide per unit of height).
206pub(crate) fn boundary(ang: f64, a: f64) -> P {
207    let (dx, dy) = (ang.sin() / a, -ang.cos());
208    let m = (dx.abs() / 0.5).max(dy.abs() / 0.5).max(1e-9);
209    [0.5 + dx / m, 0.5 + dy / m]
210}

212/// The source shown in place through an angular opacity field `f(angle)` (radians clockwise from
213/// 12 o'clock, 0..2π): a fan of thin wedges from the centre.
214pub(crate) fn fan(source: Source, a: f64, steps: usize, f: &dyn Fn(f64) -> f64) -> Vec<Layer> {
215    let n = steps.clamp(8, 720);
216    let mut angles: Vec<f64> = (0..=n).map(|k| 2.0 * PI * k as f64 / n as f64).collect();
217    // The square's corners, so wedges reach them.
218    for c in [[1.0, 0.0], [1.0, 1.0], [0.0, 1.0], [0.0, 0.0]] {
219        let ang = ((c[0] - 0.5) * a).atan2(-(c[1] - 0.5));
220        angles.push(ang.rem_euclid(2.0 * PI));
221    }
222    angles.sort_by(f64::total_cmp);
223    angles.dedup_by(|x, y| (*x - *y).abs() < 1e-9);
224    let mut out = Vec::new();
225    let c = [0.5, 0.5];
226    for w in angles.windows(2) {
227        let [x, y] = w else { continue };
228        let (fx, fy) = (f(*x).clamp(0.0, 1.0), f(*y).clamp(0.0, 1.0));
229        if fx <= 0.0 && fy <= 0.0 {
230            continue;
231        }
232        let (bx, by) = (boundary(*x, a), boundary(*y, a));
233        out.push(Layer::tri_uv(source, [c, bx, by], [c, bx, by], [(fx + fy) / 2.0, fx, fy], 1.0));
234    }
235    out
236}

240/// Keep the part of a convex polygon where `n · p <= d`.
241pub(crate) fn clip(poly: &[P], n: P, d: f64) -> Vec<P> {
242    let mut out = Vec::with_capacity(poly.len() + 2);
243    let side = |p: P| n[0] * p[0] + n[1] * p[1] - d;
244    for (i, p) in poly.iter().enumerate() {
245        let q = poly.get((i + 1) % poly.len()).copied().unwrap_or(*p);
246        let (sp, sq) = (side(*p), side(q));
247        if sp <= 0.0 {
248            out.push(*p);
249        }
250        if (sp < 0.0 && sq > 0.0) || (sp > 0.0 && sq < 0.0) {
251            let k = sp / (sp - sq);
252            out.push([p[0] + (q[0] - p[0]) * k, p[1] + (q[1] - p[1]) * k]);
253        }
254    }
255    out
256}

258/// A convex polygon as triangles: vertices drawn at `map(p)` with texture coordinate `p`.
259pub(crate) fn poly_layers(source: Source, poly: &[P], alpha: f64, map: &dyn Fn(P) -> P) -> Vec<Layer> {
260    let Some(first) = poly.first().copied() else { return vec![] };
261    poly.windows(2)
262        .skip(1)
263        .filter_map(|w| match w {
264            [b, c] => Some(Layer::tri_uv(source, [map(first), map(*b), map(*c)], [first, *b, *c], [1.0; 3], alpha)),
265            _ => None,
266        })
267        .collect()
268}

276/// A point in the 3-D scene: x right and y down in slide heights (the slide spans x ∈ [−a/2, a/2],
277/// y ∈ [−½, ½] at z = 0), z away from the viewer.
278#[derive(Clone, Copy, Debug, PartialEq)]
279pub(crate) struct V3 {
280    pub x: f64,
281    pub y: f64,
282    pub z: f64,
283}
284
285pub(crate) fn v3(x: f64, y: f64, z: f64) -> V3 {
286    V3 { x, y, z }
287}
288
289impl V3 {
290    pub fn add(self, o: V3) -> V3 {
291        v3(self.x + o.x, self.y + o.y, self.z + o.z)
292    }
293    pub fn sub(self, o: V3) -> V3 {
294        v3(self.x - o.x, self.y - o.y, self.z - o.z)
295    }
296    /// Rotate about the vertical axis through `c` (degrees; positive turns the right side away).
297    pub fn yaw(self, c: V3, deg: f64) -> V3 {
298        let (s, co) = deg.to_radians().sin_cos();
299        let p = self.sub(c);
300        v3(p.x * co - p.z * s, p.y, p.x * s + p.z * co).add(c)
301    }
302    /// Rotate about the horizontal axis through `c` (degrees; positive turns the top away).
303    pub fn pitch(self, c: V3, deg: f64) -> V3 {
304        let (s, co) = deg.to_radians().sin_cos();
305        let p = self.sub(c);
306        v3(p.x, p.y * co + p.z * s, -p.y * s + p.z * co).add(c)
307    }
308    /// Rotate in the screen plane about `c` (degrees, clockwise).
309    pub fn roll(self, c: V3, deg: f64) -> V3 {
310        let (s, co) = deg.to_radians().sin_cos();
311        let p = self.sub(c);
312        v3(p.x * co - p.y * s, p.x * s + p.y * co, p.z).add(c)
313    }
314}

316/// A perspective camera looking at the slide plane from `d` slide heights away.
317#[derive(Clone, Copy, Debug)]
318pub(crate) struct Cam {
319    pub a: f64,
320    pub d: f64,
321}
322
323impl Cam {
324    pub fn p(&self, v: V3) -> P {
325        let k = self.d / (self.d + v.z).max(self.d * 0.02);
326        [v.x * k / self.a + 0.5, v.y * k + 0.5]
327    }
328    /// The 3-D corners (tl, tr, br, bl) of the unit-space rectangle `r` lying on the slide plane.
329    pub fn rect(&self, r: R) -> [V3; 4] {
330        corners(r).map(|p| v3((p[0] - 0.5) * self.a, p[1] - 0.5, 0.0))
331    }
332    /// The unit-space point of a 3-D point on the slide plane.
333    pub fn unit(&self, x: f64, y: f64) -> V3 {
334        v3((x - 0.5) * self.a, y - 0.5, 0.0)
335    }
336    pub fn quad(&self, c: [V3; 4]) -> Q {
337        c.map(|v| self.p(v))
338    }
339}

341/// Signed area of a quad (positive: front side facing the viewer).
342pub(crate) fn area(q: &Q) -> f64 {
343    let mut s = 0.0;
344    for i in 0..4 {
345        let (a, b) = (q[i], q[(i + 1) % 4]);
346        s += a[0] * b[1] - b[0] * a[1];
347    }
348    s / 2.0
349}

351/// A card: `src` of the source mapped onto 3-D corners, if it faces the viewer; `shade` (0..1)
352/// darkens it. Seen from behind it is left out unless `back` is set (then drawn mirrored).
353pub(crate) fn card(cam: &Cam, source: Source, src: R, c: [V3; 4], alpha: f64, shade: f64, back: bool) -> Vec<Layer> {
354    let q = cam.quad(c);
355    if !q.iter().all(|p| p[0].is_finite() && p[1].is_finite()) {
356        return vec![];
357    }
358    let front = area(&q) > 0.0;
359    if !front && !back {
360        return vec![];
361    }
362    let l = Layer::quad(source, src, q, alpha);
363    let mut v = vec![l];
364    if shade > 0.001 {
365        v.push(l.shade(shade));
366    }
367    v
368}

370/// How much a card facing along `n` (unit normal toward the viewer is (0, 0, −1)) is darkened by
371/// the light behind the viewer: 0 facing straight on.
372pub(crate) fn facing_shade(c: [V3; 4], k: f64) -> f64 {
373    let u = c[1].sub(c[0]);
374    let w = c[3].sub(c[0]);
375    let n = v3(u.y * w.z - u.z * w.y, u.z * w.x - u.x * w.z, u.x * w.y - u.y * w.x);
376    let len = (n.x * n.x + n.y * n.y + n.z * n.z).sqrt();
377    if len < 1e-12 {
378        return 0.0;
379    }
380    let cos = (n.z / len).abs();
381    (k * (1.0 - cos)).clamp(0.0, 1.0)
