1//! Triangle meshes for transition layers.
2//!
3//! A [`Layer`] is a textured quad. Drawn as two triangles, a trapezoid (a slide seen in
4//! perspective) bends its texture along the diagonal, because GPUs and simple rasterisers
5//! interpolate texture coordinates linearly in screen space. [`layer_mesh`] splits such quads into a
6//! grid whose vertices follow the projective map from the source rectangle to the quad, so the
7//! picture looks like a flat card turned in 3-D. Rectangles and parallelograms stay two triangles,
8//! wedges one: the per-frame cost stays a handful of vertices per layer.
9
10use crate::anim::transition::Layer;
11
12/// One mesh vertex: position in slide-unit space (0..1, y down), texture coordinate (0..1) and
13/// opacity (the layer's alpha times its corner fade, interpolated).
14#[derive(Clone, Copy, Debug, PartialEq)]
15pub struct MeshVertex {
16    pub pos: [f64; 2],
17    pub uv: [f64; 2],
18    pub a: f64,
19}
20
21/// Grid cells per side for quads that need a projective subdivision.
22const GRID: usize = 8;
23
24/// The triangles of a layer: vertices and index triples into them.
25pub fn layer_mesh(l: &Layer) -> (Vec<MeshVertex>, Vec<[u32; 3]>) {
26    let q = l.quad;
27    let uv = l.uv;
28    let fa = l.fade.map(|f| f * l.alpha);
29    // A wedge (triangle): the 4th corner repeats the 3rd.
30    if q[3] == q[2] && uv[3] == uv[2] {
31        let v = (0..3).map(|k| MeshVertex { pos: q[k], uv: uv[k], a: fa[k] }).collect();
32        return (v, vec![[0, 1, 2]]);
33    }
34    let affine = {
35        // Parallelogram: opposite edges equal.
36        let e1 = [q[1][0] - q[0][0], q[1][1] - q[0][1]];
37        let e2 = [q[2][0] - q[3][0], q[2][1] - q[3][1]];
38        (e1[0] - e2[0]).abs() < 1e-9 && (e1[1] - e2[1]).abs() < 1e-9
39    };
40    let h = if affine { None } else { square_to_quad(q) };
41    let Some(h) = h else {
42        let v = (0..4).map(|k| MeshVertex { pos: q[k], uv: uv[k], a: fa[k] }).collect();
43        return (v, vec![[0, 1, 2], [0, 2, 3]]);
44    };
45    // Small quads (tiles, shards) bend their texture too little to matter: fewer cells.
46    let span = q.iter().flat_map(|p| q.iter().map(move |r| (p[0] - r[0]).abs().max((p[1] - r[1]).abs()))).fold(0.0, f64::max);
47    let n = ((span * 2.0 * GRID as f64).ceil() as usize).clamp(1, GRID);
48    let mut v = Vec::with_capacity((n + 1) * (n + 1));
49    for j in 0..=n {
50        let b = j as f64 / n as f64;
51        for i in 0..=n {
52            let a = i as f64 / n as f64;
53            // Texture: bilinear in (a, b) over the four uv corners (exact for rectangles).
54            let top = lerp2(uv[0], uv[1], a);
55            let bot = lerp2(uv[3], uv[2], a);
56            let t = lerp2(top, bot, b);
57            let at = fa[0] + (fa[1] - fa[0]) * a;
58            let ab = fa[3] + (fa[2] - fa[3]) * a;
59            v.push(MeshVertex { pos: apply(&h, a, b), uv: t, a: at + (ab - at) * b });
60        }
61    }
62    let mut tris = Vec::with_capacity(n * n * 2);
63    let w = (n + 1) as u32;
64    for j in 0..n as u32 {
65        for i in 0..n as u32 {
66            let p = j * w + i;
67            tris.push([p, p + 1, p + w + 1]);
68            tris.push([p, p + w + 1, p + w]);
69        }
70    }
71    (v, tris)
72}

74fn lerp2(a: [f64; 2], b: [f64; 2], t: f64) -> [f64; 2] {
75    [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t]
76}

78/// Projective map taking the unit square's corners (0,0), (1,0), (1,1), (0,1) to `q`'s tl, tr, br,
79/// bl (Heckbert's square-to-quad). `None` when the quad is degenerate.
80fn square_to_quad(q: [[f64; 2]; 4]) -> Option<[f64; 8]> {
81    let [[x0, y0], [x1, y1], [x2, y2], [x3, y3]] = q;
82    let sx = x0 - x1 + x2 - x3;
83    let sy = y0 - y1 + y2 - y3;
84    let (dx1, dx2, dy1, dy2) = (x1 - x2, x3 - x2, y1 - y2, y3 - y2);
85    let den = dx1 * dy2 - dx2 * dy1;
86    if !den.is_finite() || den.abs() < 1e-12 {
87        return None;
88    }
89    let g = (sx * dy2 - dx2 * sy) / den;
90    let h = (dx1 * sy - sx * dy1) / den;
91    let m = [x1 - x0 + g * x1, x3 - x0 + h * x3, x0, y1 - y0 + g * y1, y3 - y0 + h * y3, y0, g, h];
92    m.iter().all(|v| v.is_finite()).then_some(m)
93}
94
95fn apply(m: &[f64; 8], u: f64, v: f64) -> [f64; 2] {
96    let w = m[6] * u + m[7] * v + 1.0;
97    let w = if w.abs() < 1e-9 { 1e-9_f64.copysign(w) } else { w };
98    [(m[0] * u + m[1] * v + m[2]) / w, (m[3] * u + m[4] * v + m[5]) / w]
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use crate::anim::transition::{Layer, Source};
105
106    #[test]
107    fn rect_is_two_triangles() {
108        let (v, t) = layer_mesh(&Layer::full(Source::New, 1.0));
109        assert_eq!((v.len(), t.len()), (4, 2));
110    }
111
112    #[test]
113    fn trapezoid_is_subdivided_projectively() {
114        // A card turned about its left edge: the right edge is shorter.
115        let q = [[0.0, 0.0], [0.6, 0.2], [0.6, 0.8], [0.0, 1.0]];
116        let l = Layer::quad(Source::Old, [0.0, 0.0, 1.0, 1.0], q, 1.0);
117        let (v, t) = layer_mesh(&l);
118        assert_eq!(v.len(), (GRID + 1) * (GRID + 1));
119        assert_eq!(t.len(), GRID * GRID * 2);
120        // A small tile in perspective needs no more than a couple of triangles.
121        let small = Layer::quad(Source::Old, [0.0, 0.0, 1.0, 1.0], [[0.5, 0.5], [0.52, 0.505], [0.52, 0.515], [0.5, 0.52]], 1.0);
122        assert!(layer_mesh(&small).1.len() <= 2);
123        // Corners land on the quad's corners.
124        assert_eq!(v[0].pos, q[0]);
125        let last = v[v.len() - 1].pos;
126        assert!((last[0] - 0.6).abs() < 1e-9 && (last[1] - 0.8).abs() < 1e-9);
127        // The texture's middle column sits nearer the far (short) edge than halfway: foreshortening.
128        let mid = v[GRID / 2].pos[0];
129        assert!(mid > 0.3, "{mid}");
130    }
131
132    #[test]
133    fn degenerate_quads_do_not_blow_up() {
134        let q = [[0.5, 0.5]; 4];
135        let l = Layer::quad(Source::Old, [0.0, 0.0, 1.0, 1.0], q, 1.0);
136        let (v, _) = layer_mesh(&l);
137        assert!(v.iter().all(|p| p.pos[0].is_finite() && p.pos[1].is_finite()));
138    }
