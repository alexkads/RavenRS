1//! Slide transitions as layers of textured quads.
2//!
3//! A frame is drawn over black: each [`Layer`] maps part of a source image (old slide, new slide,
4//! their backgrounds or contents alone, or a solid colour) onto a quad in slide-unit space (0..1, y
5//! down), with an opacity per corner. Perspective comes from projecting 3-D cards
6//! ([`crate::anim::layer_mesh`] keeps their textures straight); soft edges come from per-corner opacity
7//! ramps; wedges are triangles (4th corner repeats the 3rd).
8//!
9//! The motions, curves, directions and option meanings were measured frame by frame on
10//! PowerPoint's video export of synthetic decks (`plan/powerpoint/tools/synth_transitions.ps1`).
11//! The effects themselves live in [`crate::anim::transition_fx`].
12
13use serde::{Deserialize, Serialize};
14
15use crate::anim::clampf;

17/// What a layer shows.
18#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
19pub enum Source {
20    /// The outgoing slide.
21    Old,
22    /// The incoming slide.
23    New,
24    Black,
25    White,
26    /// The outgoing slide's background (background fill plus master and layout graphics) alone.
27    OldBack,
28    /// The incoming slide's background alone.
29    NewBack,
30    /// The outgoing slide's own shapes alone, over transparency.
31    OldContent,
32    /// The incoming slide's own shapes alone, over transparency.
33    NewContent,
34}
35
36impl Source {
37    /// The whole-slide source to draw instead when the parts aren't available: a background
38    /// becomes its slide, content becomes nothing.
39    pub fn whole(self) -> Option<Source> {
40        match self {
41            Source::OldBack => Some(Source::Old),
42            Source::NewBack => Some(Source::New),
43            Source::OldContent | Source::NewContent => None,
44            s => Some(s),
45        }
46    }
47    /// Does this source need the slide rendered in parts (background / content)?
48    pub fn is_part(self) -> bool {
49        matches!(self, Source::OldBack | Source::NewBack | Source::OldContent | Source::NewContent)
50    }
51}
52
53fn ones() -> [f64; 4] {
54    [1.0; 4]
55}
56
57/// One textured quad of a transition frame.
58#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
59pub struct Layer {
60    pub source: Source,
61    /// Source sub-rectangle in uv space (0..1): l, t, r, b (bounds of `uv`).
62    pub src: [f64; 4],
63    /// Destination corners in slide-unit space: tl, tr, br, bl.
64    pub quad: [[f64; 2]; 4],
65    /// Texture coordinate of each destination corner (tl, tr, br, bl). For plain rectangles these
66    /// are the corners of `src`; triangles and mirrored flaps carry their own. Draw with these.
67    pub uv: [[f64; 2]; 4],
68    pub alpha: f64,
69    /// Opacity of each corner (tl, tr, br, bl), multiplied by `alpha` and interpolated across the
70    /// quad: soft edges and feathered masks.
71    #[serde(default = "ones")]
72    pub fade: [f64; 4],
73}
74
75pub(crate) type R = [f64; 4];
76pub(crate) type P = [f64; 2];
77pub(crate) type Q = [[f64; 2]; 4];
78pub(crate) const FULL: R = [0.0, 0.0, 1.0, 1.0];
79
80pub(crate) fn corners(r: R) -> Q {
81    [[r[0], r[1]], [r[2], r[1]], [r[2], r[3]], [r[0], r[3]]]
82}
83
84impl Layer {
85    /// The whole source over the whole slide.
86    pub fn full(source: Source, alpha: f64) -> Layer {
87        Layer::rect(source, FULL, FULL, alpha)
88    }
89    /// `src` rectangle of the source drawn into the `dst` rectangle.
90    pub fn rect(source: Source, src: R, dst: R, alpha: f64) -> Layer {
91        Layer { source, src, quad: corners(dst), uv: corners(src), alpha, fade: ones() }
92    }
93    /// `src` rectangle of the source drawn onto a quad.
94    pub fn quad(source: Source, src: R, quad: Q, alpha: f64) -> Layer {
95        Layer { source, src, quad, uv: corners(src), alpha, fade: ones() }
96    }
97    /// The part of the source under `dst` drawn in place.
98    pub(crate) fn inplace(source: Source, dst: R, alpha: f64) -> Layer {
99        Layer::rect(source, dst, dst, alpha)
100    }
101    /// A triangle with its own texture coordinates and corner opacities.
102    pub(crate) fn tri_uv(source: Source, p: [P; 3], uv: [P; 3], fade: [f64; 3], alpha: f64) -> Layer {
103        let xs = [uv[0][0], uv[1][0], uv[2][0]];
104        let ys = [uv[0][1], uv[1][1], uv[2][1]];
105        let mn = |v: [f64; 3]| v.iter().copied().fold(f64::INFINITY, f64::min);
106        let mx = |v: [f64; 3]| v.iter().copied().fold(f64::NEG_INFINITY, f64::max);
107        Layer {
108            source,
109            src: [mn(xs), mn(ys), mx(xs), mx(ys)],
110            quad: [p[0], p[1], p[2], p[2]],
111            uv: [uv[0], uv[1], uv[2], uv[2]],
112            alpha,
113            fade: [fade[0], fade[1], fade[2], fade[2]],
114        }
115    }
116    /// With corner opacities.
117    pub(crate) fn faded(mut self, fade: [f64; 4]) -> Layer {
118        self.fade = fade;
119        self
120    }
121    /// A black copy of this layer's shape at `k` of its opacity: shading a card.
122    pub(crate) fn shade(&self, k: f64) -> Layer {
123        Layer { source: Source::Black, alpha: self.alpha * k.clamp(0.0, 1.0), ..*self }
124    }
125    fn sanitized(mut self) -> Layer {
126        let f = |v: f64| if v.is_finite() { v.clamp(-1.0e6, 1.0e6) } else { 0.0 };
127        for v in self.src.iter_mut() {
128            *v = f(*v).clamp(0.0, 1.0);
129        }
130        for c in self.quad.iter_mut().chain(self.uv.iter_mut()) {
131            c[0] = f(c[0]);
132            c[1] = f(c[1]);
133        }
134        for c in self.uv.iter_mut() {
135            c[0] = c[0].clamp(0.0, 1.0);
136            c[1] = c[1].clamp(0.0, 1.0);
137        }
138        for a in self.fade.iter_mut() {
139            *a = clampf(*a, 0.0, 1.0, 0.0);
140        }
141        self.alpha = clampf(self.alpha, 0.0, 1.0, 0.0);
142        self
143    }
144}

146/// Slide aspect assumed by [`transition_layers`] (16:9).
147pub const DEFAULT_ASPECT: f64 = 16.0 / 9.0;
148
149/// Layers of the transition `kind` (a [`crate::model::anim::TRANSITIONS`] id) with `option` at
150/// progress `t` (0..1) on a 16:9 slide. See [`transition_frame`].
151pub fn transition_layers(kind: &str, option: &str, t: f64) -> Vec<Layer> {
152    transition_frame(kind, option, t, DEFAULT_ASPECT)
153}
154
155/// Layers of the transition `kind` with `option` at linear progress `t` (0..1; the effect applies
156/// its own easing) on a slide `aspect` wide per unit of height. Draw in order over black. At
157/// `t >= 1` the result is the new slide alone; at `t <= 0` (or NaN) the old slide alone. Morph
158/// returns a plain crossfade (the UI renders morph frames with [`crate::anim::morph_frame`]). Layers
159/// with [`Source::is_part`] sources need the slides rendered as background and content
160/// ([`uses_slide_parts`]).
161pub fn transition_frame(kind: &str, option: &str, t: f64, aspect: f64) -> Vec<Layer> {
162    let t = clampf(t, 0.0, 1.0, 0.0);
163    let aspect = if aspect.is_finite() && aspect > 0.01 { aspect.min(100.0) } else { DEFAULT_ASPECT };
164    let v = if t >= 1.0 {
165        vec![Layer::full(Source::New, 1.0)]
166    } else if t <= 0.0 {
167        vec![Layer::full(Source::Old, 1.0)]
168    } else {
169        crate::anim::transition_fx::layers(kind, option, t, aspect, 0)
170    };
171    v.into_iter().map(Layer::sanitized).filter(|l| l.alpha > 0.0 && l.fade.iter().any(|a| *a > 0.0)).collect()
172}

174/// Does `kind` animate the slides' backgrounds and contents separately (PowerPoint's Dynamic
175/// Content transitions and Reveal)? Then the frame uses [`Source::OldBack`] & co.
176pub fn uses_slide_parts(kind: &str, option: &str) -> bool {
177    match kind {
178        "pan" | "ferris" | "conveyor" | "rotate" | "window" | "orbit" | "flythrough" => true,
179        "reveal" => !option.starts_with("black"),
180        _ => false,
181    }
