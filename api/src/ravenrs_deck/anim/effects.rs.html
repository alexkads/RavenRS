11/// Everything an effect needs besides its progress.
12#[derive(Clone, Debug)]
13pub(crate) struct EffectSpec {
14    pub class: AnimClass,
15    /// Base effect id (exit ids mapped to their entrance counterpart).
16    pub effect: String,
17    pub option: String,
18    pub accel: f64,
19    pub decel: f64,
20    pub bounce: f64,
21    pub amount: Option<f64>,
22    pub color: Option<Rgba>,
23    pub path: Option<MotionPath>,
24    /// Shape box in points (sanitized).
25    pub xf: Xfrm,
26    pub slide_w: f64,
27    pub slide_h: f64,
28}

30/// Map an exit effect id to the entrance effect it mirrors.
31pub(crate) fn exit_base(effect: &str) -> &str {
32    match effect {
33        "disappear" => "appear",
34        "fadeOut" => "fade",
35        "flyOut" => "fly",
36        "floatOut" => "float",
37        "splitOut" => "split",
38        "wipeOut" => "wipe",
39        "shapeOut" => "shape",
40        "wheelOut" => "wheel",
41        "randomBarsOut" => "randomBars",
42        "shrinkTurn" => "grow",
43        "zoomOut" => "zoom",
44        "swivelOut" => "swivel",
45        "bounceOut" => "bounce",
46        "dissolveOut" => "dissolve",
47        "blindsOut" => "blinds",
48        "boxOut" => "box",
49        "checkerboardOut" => "checkerboard",
50        "diamondOut" => "diamond",
51        "peekOut" => "peek",
52        "stripsOut" => "strips",
53        "sinkDown" => "rise",
54        "spinnerOut" => "spinner",
55        "contract" => "expand",
56        other => other,
57    }
58}

60/// Emphasis effects whose end state persists after they finish (PowerPoint keeps the spun angle,
61/// the new size, the new colour and transparency).
62pub(crate) fn emphasis_persists(effect: &str) -> bool {
63    matches!(
64        effect,
65        "spin"
66            | "growShrink"
67            | "desaturate"
68            | "darken"
69            | "lighten"
70            | "transparency"
71            | "objectColor"
72            | "complementaryColor"
73            | "lineColor"
74            | "fillColor"
75            | "fontColor"
76    )
77}

79fn uses_default_smoothing(effect: &str) -> bool {
80    matches!(effect, "fly" | "float" | "rise" | "peek" | "lines" | "arcs" | "turns" | "shapes" | "loops" | "customPath")
81        || crate::model::anim::preset_path(effect).is_some()
82}

84impl EffectSpec {
85    /// Eased progress.
86    pub fn eased(&self, p: f64) -> f64 {
87        if self.accel <= 0.0 && self.decel <= 0.0 && self.bounce <= 0.0 && uses_default_smoothing(&self.effect) {
88            smooth(p)
89        } else {
90            ease(p, self.accel, self.decel, self.bounce)
91        }
92    }
93
94    /// The state delta at linear progress `p` (0..1) while the effect plays. `cur` is the shape's
95    /// current fill colour (for colour emphasis).
96    pub fn eval(&self, p: f64, cur: Rgba) -> AnimState {
97        let p = crate::anim::clampf(p, 0.0, 1.0, 0.0);
98        let s = match self.class {
99            AnimClass::Entrance => self.entrance(p, false),
100            AnimClass::Exit => self.entrance(1.0 - p, true),
101            AnimClass::Emphasis => self.emphasis(p, cur),
102            AnimClass::Path => self.motion(p),
103            AnimClass::Media => AnimState::default(),
104        };
105        s.sanitized()
106    }
107
108    /// The state the effect leaves behind once done.
109    pub fn end_state(&self, cur: Rgba) -> AnimState {
110        match self.class {
111            AnimClass::Entrance | AnimClass::Media => AnimState::default(),
112            AnimClass::Exit => AnimState::hidden(),
113            AnimClass::Emphasis if emphasis_persists(&self.effect) => self.emphasis(1.0, cur).sanitized(),
114            AnimClass::Emphasis => AnimState::default(),
115            AnimClass::Path => self.motion(1.0).sanitized(),
116        }
117    }

119    fn entrance(&self, p: f64, exit: bool) -> AnimState {
120        let e = self.eased(p);
121        let q = 1.0 - e;
122        let o = self.option.as_str();
123        let Xfrm { x, y, w, h, .. } = self.xf;
124        let (sw, sh) = (self.slide_w, self.slide_h);
125        let mut st = AnimState::default();
126        let center_clip = |f: f64, horz: bool, vert: bool| {
127            let f = f.clamp(0.0, 1.0) / 2.0;
128            let (l, r) = if horz { (0.5 - f, 0.5 + f) } else { (0.0, 1.0) };
129            let (t, b) = if vert { (0.5 - f, 0.5 + f) } else { (0.0, 1.0) };
130            Some([l, t, r, b])
131        };
132        match self.effect.as_str() {
133            "appear" => st.visible = p > 0.0,
134            "fade" | "dissolve" | "randomBars" | "blinds" | "checkerboard" | "wheel" => st.opacity = e,
135            "fly" => {
136                let o = if o.is_empty() { "b" } else { o };
137                if o.contains('l') {
138                    st.offset_x = -(x + w) * q;
139                }
140                if o.contains('r') {
141                    st.offset_x = (sw - x) * q;
142                }
143                if o.contains('t') {
144                    st.offset_y = -(y + h) * q;
145                }
146                if o.contains('b') {
147                    st.offset_y = (sh - y) * q;
148                }
149            }
150            "float" | "rise" => {
151                // Float Up enters from below and leaves upward; Float Down the reverse.
152                let down = o == "d";
153                let dir = if down { -1.0 } else { 1.0 };
154                let dir = if exit { -dir } else { dir };
155                st.offset_y = dir * 0.1 * sh * q;
156                st.opacity = e;
157            }
158            "split" => {
159                let horz = o.starts_with("horz");
160                // "In" closes from the edges: approximated with the centre reveal plus a fade.
161                st.clip = if horz { center_clip(e, false, true) } else { center_clip(e, true, false) };
162                if o.ends_with("In") {
163                    st.opacity = e;
164                }
165            }
166            "wipe" => {
167                st.clip = Some(match o {
168                    "l" => [0.0, 0.0, e, 1.0],
169                    "t" => [0.0, 0.0, 1.0, e],
170                    "r" => [1.0 - e, 0.0, 1.0, 1.0],
171                    _ => [0.0, 1.0 - e, 1.0, 1.0],
172                });
173            }
174            "shape" | "circle" | "box" | "diamond" | "expandClip" => {
175                let round = matches!(self.effect.as_str(), "circle" | "diamond")
176                    || o.starts_with("circle")
177                    || o.starts_with("diamond")
178                    || o.starts_with("plus");
179                let f = if round { (e * std::f64::consts::SQRT_2).min(1.0) } else { e };
180                st.clip = center_clip(f, true, true);
181                if o.ends_with("In") || o == "in" {
182                    st.opacity = e;
183                }
184            }
185            "grow" => {
186                st.scale_x = e;
187                st.scale_y = e;
188                st.rotate = -90.0 * q;
189                st.opacity = e;
190            }
191            "zoom" => {
192                let s = if o == "out" { 1.0 + 3.0 * q } else { e };
193                st.scale_x = s;
194                st.scale_y = s;
195                st.opacity = e;
196            }
197            "swivel" => {
198                let s = e * (q * 2.5 * PI).cos();
199                if o == "vert" {
200                    st.scale_y = s;
201                } else {
202                    st.scale_x = s;
203                }
204                st.opacity = e.min(1.0);
205            }
206            "bounce" => {
207                let b = bounce_out(p);
208                st.offset_y = -(0.25 * sh) * (1.0 - b);
209                st.offset_x = -0.1 * sw * (1.0 - p);
210                st.opacity = (p * 4.0).min(1.0);
211            }
212            "peek" => match o {
213                "l" => {
214                    st.offset_x = -w * q;
215                    st.clip = Some([q, 0.0, 1.0, 1.0]);
216                }
217                "r" => {
218                    st.offset_x = w * q;
219                    st.clip = Some([0.0, 0.0, e, 1.0]);
220                }
221                "t" => {
222                    st.offset_y = -h * q;
223                    st.clip = Some([0.0, q, 1.0, 1.0]);
224                }
225                _ => {
226                    st.offset_y = h * q;
227                    st.clip = Some([0.0, 0.0, 1.0, e]);
228                }
229            },
230            "strips" => {
231                st.clip = Some(match o {
232                    "ru" => [0.0, q, e, 1.0],
233                    "ld" => [q, 0.0, 1.0, e],
234                    "rd" => [0.0, 0.0, e, e],
235                    _ => [q, q, 1.0, 1.0],
236                });
237            }
238            "expand" => {
239                st.scale_x = 0.3 + 0.7 * e;
240                st.opacity = e;
241            }
242            "spinner" => {
243                st.rotate = 360.0 * q;
244                st.scale_x = 0.5 + 0.5 * e;
245                st.scale_y = 0.5 + 0.5 * e;
246                st.opacity = e;
247            }
248            _ => st.opacity = e,
249        }
250        st
251    }

253    fn emphasis(&self, p: f64, cur: Rgba) -> AnimState {
254        let e = self.eased(p);
255        let bump = (p * PI).sin();
256        let mut st = AnimState::default();
257        let target = self.color.unwrap_or(Rgba::rgb(0xC0, 0x00, 0x00));
258        match self.effect.as_str() {
259            "pulse" => {
260                st.scale_x = 1.0 + 0.1 * bump;
261                st.scale_y = st.scale_x;
262            }
263            "colorPulse" => st.tint = Some(cur.lerp(target, bump)),
264            "teeter" => st.rotate = 8.0 * (p * 4.0 * PI).sin() * (1.0 - p * 0.5),
265            "spin" => {
266                let amt = self.amount.filter(|a| a.is_finite()).unwrap_or(360.0).clamp(-36_000.0, 36_000.0);
267                let sign = if self.option == "counterClockwise" { -1.0 } else { 1.0 };
268                st.rotate = sign * amt * e;
269            }
270            "growShrink" => {
271                let amt = self.amount.filter(|a| a.is_finite() && *a > 0.0).unwrap_or(1.5).clamp(0.0, 100.0);
272                let s = 1.0 + (amt - 1.0) * e;
273                match self.option.as_str() {
274                    "horz" => st.scale_x = s,
275                    "vert" => st.scale_y = s,
276                    _ => {
277                        st.scale_x = s;
278                        st.scale_y = s;
279                    }
280                }
281            }
282            "desaturate" => {
283                let l = (0.299 * cur.r as f64 + 0.587 * cur.g as f64 + 0.114 * cur.b as f64).round().clamp(0.0, 255.0) as u8;
284                st.tint = Some(cur.lerp(Rgba::rgba(l, l, l, cur.a), e));
285            }
286            "darken" => st.tint = Some(cur.lerp(Rgba::rgba(0, 0, 0, cur.a), 0.4 * e)),
287            "lighten" => st.tint = Some(cur.lerp(Rgba::rgba(255, 255, 255, cur.a), 0.4 * e)),
288            "transparency" => {
289                // `amount` is the transparency (0.5 = 50 %).
290                let a = self.amount.filter(|a| a.is_finite()).unwrap_or(0.5).clamp(0.0, 1.0);
291                st.opacity = 1.0 - a * e;
292            }
293            "objectColor" | "fillColor" | "lineColor" | "fontColor" => st.tint = Some(cur.lerp(target, e)),
294            "complementaryColor" => st.tint = Some(cur.lerp(Rgba::rgba(255 - cur.r, 255 - cur.g, 255 - cur.b, cur.a), e)),
295            "underline" | "boldFlash" => {
296                st.scale_x = 1.0 + 0.04 * bump;
297                st.scale_y = st.scale_x;
298                if self.effect == "boldFlash" {
299                    st.tint = Some(cur.lerp(Rgba::rgba(0, 0, 0, cur.a), 0.3 * bump));
300                }
301            }
302            "wave" => {
303                st.offset_y = -0.02 * self.slide_h * bump;
304                st.scale_y = 1.0 + 0.05 * bump;
305            }
306            _ => {}
307        }
308        st
309    }
310
311    fn motion(&self, p: f64) -> AnimState {
312        let e = self.eased(p);
313        let (fx, fy) = match &self.path {
314            Some(mp) => mp.at(e),
315            None => (0.0, 0.0),
316        };
317        AnimState { offset_x: fx * self.slide_w, offset_y: fy * self.slide_h, ..Default::default() }
318    }
319}

321/// The path for a motion-path animation: its own, or the preset default.
322pub(crate) fn motion_path(effect: &str, option: &str, own: Option<&str>) -> MotionPath {
323    match own {
324        Some(s) if !s.trim().is_empty() => MotionPath::parse(s),
325        _ => MotionPath::parse(&default_path(effect, option)),
326    }
