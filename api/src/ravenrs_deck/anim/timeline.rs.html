13/// Delay between paragraphs of a by-paragraph build that isn't on click (seconds).
14pub const PARAGRAPH_STAGGER: f64 = 0.5;
15const MAX_EFFECTS: usize = 100_000;
16const MAX_ITERATIONS: f64 = 1.0e6;

18/// Which sequence an effect belongs to.
19#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
20pub enum Sequence {
21    /// The slide's main click sequence.
22    Main,
23    /// An interactive sequence started by clicking this shape.
24    Trigger(ShapeId),
25}

27/// Timing of one (sub-)effect, for the Animation Pane timeline and the UI.
28#[derive(Clone, Debug, PartialEq)]
29pub struct EffectTiming {
30    /// Index into `slide.animations`.
31    pub anim: usize,
32    /// Animated shape and paragraph (`None` = the whole shape).
33    pub shape: ShapeId,
34    pub paragraph: Option<usize>,
35    pub sequence: Sequence,
36    /// Click step within its sequence.
37    pub step: usize,
38    /// Start within the step (seconds, delay included).
39    pub start: f64,
40    /// One iteration (seconds).
41    pub duration: f64,
42    /// Repeat count (fractional allowed); 1 = once.
43    pub iterations: f64,
44    /// Repeats until the next click.
45    pub until_next_click: bool,
46    pub auto_reverse: bool,
47    pub rewind: bool,
48    /// Time from start to end (all iterations, reverse included; one iteration if until-click).
49    pub span: f64,
50    pub class: AnimClass,
51}

95/// A slide's animations laid out in time.
96#[derive(Clone, Debug, Default)]
97pub struct Timeline {
98    effects: Vec<Effect>,
99    main: Vec<StepInfo>,
100    triggers: Vec<(ShapeId, Vec<StepInfo>)>,
101    /// Base fill colour per shape (for colour emphasis).
102    colors: Vec<(ShapeId, Rgba)>,
103    /// By-paragraph builds animate first-level paragraphs; deeper paragraphs follow their parent:
104    /// (shape, member paragraph, owner paragraph).
105    groups: Vec<(ShapeId, usize, usize)>,
106    /// Lookups built with the timeline, so evaluating a shape or paragraph costs its own effects
107    /// rather than a scan of every effect, group and colour on the slide.
108    index: TimelineIndex,
109}
110
111#[derive(Clone, Debug, Default)]
112struct TimelineIndex {
113    /// Effects (indices into `effects`, in order) by target shape and paragraph.
114    by_target: HashMap<(ShapeId, Option<usize>), Vec<usize>>,
115    /// Effects (indices into `effects`, in order) by target shape.
116    by_shape: HashMap<ShapeId, Vec<usize>>,
117    /// Owner paragraph of each member paragraph (the first group recorded wins).
118    owner: HashMap<(ShapeId, usize), usize>,
119    /// Groups (member, owner) by shape, in order.
120    groups_by_shape: HashMap<ShapeId, Vec<(usize, usize)>>,
121    /// Base colour by shape (the first recorded wins).
122    color: HashMap<ShapeId, Rgba>,
123}
124
125impl TimelineIndex {
126    fn build(effects: &[Effect], groups: &[(ShapeId, usize, usize)], colors: &[(ShapeId, Rgba)]) -> TimelineIndex {
127        let mut ix = TimelineIndex::default();
128        for (i, e) in effects.iter().enumerate() {
129            ix.by_target.entry((e.timing.shape, e.timing.paragraph)).or_default().push(i);
130            ix.by_shape.entry(e.timing.shape).or_default().push(i);
131        }
132        for &(shape, member, owner) in groups {
133            ix.owner.entry((shape, member)).or_insert(owner);
134            ix.groups_by_shape.entry(shape).or_default().push((member, owner));
135        }
136        for &(shape, c) in colors {
137            ix.color.entry(shape).or_insert(c);
138        }
139        ix
140    }
141}

153fn finite_or(v: f64, d: f64) -> f64 {
154    if v.is_finite() { v } else { d }

157fn sanitize_xfrm(x: Xfrm, sw: f64, sh: f64) -> Xfrm {
158    Xfrm {
159        x: finite_or(x.x, 0.0).clamp(-1.0e7, 1.0e7),
160        y: finite_or(x.y, 0.0).clamp(-1.0e7, 1.0e7),
161        w: finite_or(x.w, sw / 2.0).clamp(0.0, 1.0e7),
162        h: finite_or(x.h, sh / 2.0).clamp(0.0, 1.0e7),
163        ..x
164    }
165}

178    /// Build with a colour scheme for theme colours and a lookup for shape boxes (placeholders
179    /// inherit their box from the layout; the lookup wins over `shape.xfrm`).
180    pub fn with(slide: &Slide, slide_w: f64, slide_h: f64, scheme: Option<&ColorScheme>, xfrm_of: &dyn Fn(ShapeId) -> Option<Xfrm>) -> Timeline {
181        let sw = finite_or(slide_w, 960.0).clamp(1.0, 1.0e7);
182        let sh = finite_or(slide_h, 540.0).clamp(1.0, 1.0e7);
183        let owned;
184        let scheme = match scheme {
185            Some(s) => Some(s),
186            None => {
187                owned = default_scheme();
188                owned.as_ref()
189            }
190        };
191        let resolve = |c: &crate::model::ColorRef| -> Rgba {
192            match scheme {
193                Some(s) => c.resolve(s, None),
194                None => match &c.base {
195                    crate::model::ColorBase::Rgb { rgb } => *rgb,
196                    _ => Rgba::rgb(0x15, 0x60, 0x82),
197                },
198            }
199        };
200        let accent = scheme.map(|s| s.get(SchemeSlot::Accent1)).unwrap_or(Rgba::rgb(0x15, 0x60, 0x82));
201
202        let mut tl = Timeline::default();
203        if slide.animations.is_empty() {
204            return tl;
205        }
206        // Every shape by id, built once: resolving each animation's target by tree search would
207        // cost shapes x animations.
208        let shapes = shape_index(&slide.shapes);
209        // Paragraph targets of each shape, worked out for its first by-paragraph animation only.
210        let mut para_cache: HashMap<ShapeId, Arc<Vec<Option<usize>>>> = HashMap::new();
211        // Group the animations by sequence in one pass, sequences in order of first appearance
212        // (main first), so many distinct triggers cost linear work, not a scan per trigger.
213        let mut seqs: Vec<(Sequence, Vec<usize>)> = vec![(Sequence::Main, Vec::new())];
214        let mut seq_index: HashMap<Sequence, usize> = HashMap::from([(Sequence::Main, 0)]);
215        for (ai, a) in slide.animations.iter().enumerate() {
216            let s = a.trigger.map(Sequence::Trigger).unwrap_or(Sequence::Main);
217            let i = *seq_index.entry(s).or_insert_with(|| {
218                seqs.push((s, Vec::new()));
219                seqs.len() - 1
220            });
221            if let Some((_, members)) = seqs.get_mut(i) {
222                members.push(ai);
223            }
224        }
225        // Paragraph groups already recorded (the list itself keeps their order).
226        let mut grouped: HashSet<(ShapeId, usize, usize)> = HashSet::new();
227        for (seq, members) in seqs {
228            if tl.effects.len() >= MAX_EFFECTS {
229                break;
230            }
231            let mut steps: Vec<StepInfo> = Vec::new();
232            let mut group_begin = 0.0f64;
233            let mut max_end = 0.0f64;
234            for ai in members {
235                let Some(a) = slide.animations.get(ai) else { continue };
236                if tl.effects.len() >= MAX_EFFECTS {
237                    break;
238                }
239                let shape = shapes.get(&a.shape).copied();
240                // Paragraph targets of this animation.
241                let paras: Arc<Vec<Option<usize>>> = match (a.paragraph, a.text_build) {
242                    (Some(p), _) => Arc::new(vec![Some(p as usize)]),
243                    (None, TextBuild::ByParagraph) => {
244                        let cached = para_cache.entry(a.shape).or_insert_with(|| {
245                            let mut v: Vec<Option<usize>> = vec![];
246                            if let Some(t) = shape.and_then(|s| s.text.as_ref()) {
247                                for (i, role) in paragraph_build_roles(t).into_iter().enumerate() {
248                                    match role {
249                                        BuildRole::Top => v.push(Some(i)),
250                                        BuildRole::Member(o) => {
251                                            if grouped.insert((a.shape, i, o)) {
252                                                tl.groups.push((a.shape, i, o));
253                                            }
254                                        }
255                                        BuildRole::Unowned => {}
256                                    }
257                                }
258                            }
259                            if v.is_empty() {
260                                v.push(None)
261                            }
262                            Arc::new(v)
263                        });
264                        Arc::clone(cached)
265                    }
266                    _ => Arc::new(vec![None]),
267                };
268                let xf = xfrm_of(a.shape).or_else(|| shape.and_then(|s| s.xfrm)).unwrap_or(Xfrm::new(sw / 4.0, sh / 4.0, sw / 2.0, sh / 2.0));
269                let xf = sanitize_xfrm(xf, sw, sh);
270                let spec = Arc::new(build_spec(a, xf, sw, sh, &resolve));
271                let dur = secs(a.duration_ms);
272                let delay = secs(a.delay_ms);
273                let until_next_click = a.repeat == u32::MAX;
274                let iterations = if until_next_click || a.repeat <= 1 { 1.0 } else { (a.repeat as f64).min(MAX_ITERATIONS) };
275                let span = dur * if a.auto_reverse { 2.0 } else { 1.0 } * iterations;
276                let after = match a.after.as_deref() {
277                    Some("hide") => After::Hide,
278                    Some("hideOnNextClick") => After::HideOnNextClick,
279                    Some(c) => Rgba::from_hex(c).map(After::Dim).unwrap_or(After::None),
280                    None => After::None,
281                };
282                let mut first_start = 0.0;
283                for (k, para) in paras.iter().enumerate() {
284                    let mode = if k == 0 {
285                        a.start
286                    } else if a.start == AnimStart::OnClick {
287                        AnimStart::OnClick
288                    } else {
289                        AnimStart::WithPrevious
290                    };
291                    let first_of_seq = steps.is_empty();
292                    let mode = if first_of_seq && matches!(seq, Sequence::Trigger(_)) { AnimStart::OnClick } else { mode };
293                    match mode {
294                        AnimStart::OnClick => {
295                            steps.push(StepInfo { auto: false, duration: 0.0 });
296                            group_begin = 0.0;
297                            max_end = 0.0;
298                        }
299                        AnimStart::WithPrevious => {
300                            if first_of_seq {
301                                steps.push(StepInfo { auto: true, duration: 0.0 });
302                            }
303                        }
304                        AnimStart::AfterPrevious => {
305                            if first_of_seq {
306                                steps.push(StepInfo { auto: true, duration: 0.0 });
307                            }
308                            group_begin = max_end;
309                        }
310                    }
311                    let start = if k > 0 && a.start != AnimStart::OnClick { first_start + k as f64 * PARAGRAPH_STAGGER } else { group_begin + delay };
312                    if k == 0 {
313                        first_start = start;
314                    }
315                    let end = start + span;
316                    max_end = max_end.max(end);
317                    let step = steps.len().saturating_sub(1);
318                    if let Some(s) = steps.last_mut() {
319                        // Media plays on by itself once started: it holds back an After Previous
320                        // effect (`max_end`), but not the next click (#116).
321                        s.duration = s.duration.max(if a.class == AnimClass::Media { start } else { end });
322                    }
323                    tl.effects.push(Effect {
324                        timing: EffectTiming {
325                            anim: ai,
326                            shape: a.shape,
327                            paragraph: *para,
328                            sequence: seq,
329                            step,
330                            start,
331                            duration: dur,
332                            iterations,
333                            until_next_click,
334                            auto_reverse: a.auto_reverse,
335                            rewind: a.rewind,
336                            span,
337                            class: a.class,
338                        },
339                        spec: Arc::clone(&spec),
340                        after: after.clone(),
341                    });
342                }
343            }
344            match seq {
345                Sequence::Main => tl.main = steps,
346                Sequence::Trigger(s) => tl.triggers.push((s, steps)),
347            }
348        }
349        // Base colours of animated shapes.
350        let mut colored: HashSet<ShapeId> = HashSet::new();
351        for e in &tl.effects {
352            let id = e.timing.shape;
353            if !colored.insert(id) {
354                continue;
355            }
356            let c = match shapes.get(&id).copied().and_then(|s| s.fill.as_ref()) {
357                Some(Fill::Solid { color }) => resolve(color),
358                _ => accent,
359            };
360            tl.colors.push((id, c));
361        }
362        tl.index = TimelineIndex::build(&tl.effects, &tl.groups, &tl.colors);
363        tl
364    }

382    /// Media effects (Play, Pause, Stop) of main step `step` that start after `from` and by `to`
383    /// seconds into it, in order: what a show running the step from `from` to `to` sets off.
384    pub fn media_cues(&self, step: usize, from: f64, to: f64) -> Vec<EffectTiming> {
385        let mut v: Vec<EffectTiming> = self
386            .effects
387            .iter()
388            .map(|e| &e.timing)
389            .filter(|e| e.class == AnimClass::Media && e.sequence == Sequence::Main && e.step == step && e.start > from && e.start <= to)
390            .cloned()
391            .collect();
392        v.sort_by(|a, b| a.start.total_cmp(&b.start));
393        v
394    }

401    /// Map a time on the "play everything" clock to `(step, t)` for [`Timeline::state`].
402    pub fn locate(&self, time: f64) -> (usize, f64) {
403        let mut t = finite_or(time, 0.0).max(0.0);
404        for (i, s) in self.main.iter().enumerate() {
405            if t < s.duration || (t == 0.0 && s.duration == 0.0 && i == 0) {
406                return (i, t);
407            }
408            t -= s.duration;
409        }
410        (self.main.len(), 0.0)
411    }

413    /// Where effect `anim` (index into `slide.animations`) starts on the play-everything clock:
414    /// Animation Pane ▸ Play From. `None` for a triggered effect or one that isn't on the slide.
415    pub fn start_of(&self, anim: usize) -> Option<f64> {
416        let e = self
417            .effects
418            .iter()
419            .map(|e| &e.timing)
420            .filter(|e| e.anim == anim && e.sequence == Sequence::Main)
421            .min_by(|a, b| a.step.cmp(&b.step).then(a.start.total_cmp(&b.start)))?;
422        let before: f64 = self.main.iter().take(e.step).map(|s| s.duration).sum();
423        Some(before + e.start)
424    }

450    /// Paragraph indices of `shape` animated separately (by-paragraph builds).
451    pub fn paragraph_targets(&self, shape: ShapeId) -> Vec<usize> {
452        let mine = self.index.by_shape.get(&shape).map_or(&[][..], Vec::as_slice);
453        let mut v: Vec<usize> = mine.iter().filter_map(|&i| self.effects.get(i)).filter_map(|e| e.timing.paragraph).collect();
454        let owners: HashSet<usize> = v.iter().copied().collect();
455        if let Some(groups) = self.index.groups_by_shape.get(&shape) {
456            v.extend(groups.iter().filter(|(_, owner)| owners.contains(owner)).map(|(member, _)| *member));
457        }
458        v.sort_unstable();
459        v.dedup();
460        v
461    }

469    /// State of `shape` (or of one of its paragraphs, `para`) when `step` main steps have been

482    /// [`Timeline::state_with`] applying the effects of triggered sequences in the order their steps
483    /// were started (`stamps`), after those of the main sequence. Without stamps they apply in
484    /// Animation Pane order.
485    pub fn state_ordered(
486        &self,
487        shape: ShapeId,
488        para: Option<usize>,
489        step: usize,
490        t: f64,
491        triggers: &[TriggerPlay],
492        stamps: &[StepStamp],
493    ) -> AnimState {
494        let para = para.map(|p| self.index.owner.get(&(shape, p)).copied().unwrap_or(p));
495        let mine = self.index.by_target.get(&(shape, para)).map_or(&[][..], Vec::as_slice).iter().filter_map(|&i| self.effects.get(i));
496        let base_color = self.index.color.get(&shape).copied().unwrap_or(Rgba::rgb(0x15, 0x60, 0x82));
497        let mut st = AnimState::default();
498        // Initial visibility comes from the first effect in the Animation Pane order.
499        let mut ordered: Vec<&Effect> = mine.collect();
500        ordered.sort_by_key(|e| e.timing.anim);
501        if let Some(e0) = ordered.first()
502            && e0.timing.class == AnimClass::Entrance
503        {
504            st.visible = false;
505        }
506        // Then play in time order: main sequence first, then triggered sequences as they were clicked.
507        if !stamps.is_empty() {
508            let when = |e: &Effect| match e.timing.sequence {
509                Sequence::Main => 0,
510                Sequence::Trigger(tr) => {
511                    stamps.iter().find(|s| s.trigger == tr && s.step == e.timing.step).map_or(u64::MAX, |s| s.at.saturating_add(1))
512                }
513            };
514            ordered.sort_by_key(|e| (when(e), e.timing.anim));
515        }
516        for e in &ordered {
517            let (pstep, pt) = match e.timing.sequence {
518                Sequence::Main => (step, t),
519                Sequence::Trigger(tr) => triggers.iter().find(|p| p.trigger == tr).map(|p| (p.step, p.t)).unwrap_or((0, 0.0)),
520            };
521            let cur = st.tint.unwrap_or(base_color);
522            let revert = e.timing.rewind || e.timing.auto_reverse;
523            match status(&e.timing, pstep, pt) {
524                Status::NotStarted => {}
525                Status::Active(p) => {
526                    if e.timing.class == AnimClass::Entrance || e.timing.class == AnimClass::Exit {
527                        st.visible = true;
528                    }
529                    st = st.compose(&e.spec.eval(p, cur));
530                }
531                Status::Done => {
532                    match e.timing.class {
533                        AnimClass::Entrance => st.visible = !revert,
534                        AnimClass::Exit => st.visible = revert,
535                        AnimClass::Emphasis | AnimClass::Path if !revert => {
536                            let end = e.spec.end_state(cur);
537                            st = st.compose(&end);
538                        }
539                        _ => {}
540                    }
541                    match e.after {
542                        After::Hide => st.visible = false,
543                        After::HideOnNextClick if pstep > e.timing.step => st.visible = false,
544                        After::Dim(c) if pstep > e.timing.step => st.tint = Some(c),
545                        _ => {}
546                    }
547                }
548            }
549        }
550        st.sanitized()
551    }

553    /// State on the "play everything" clock (Animation Pane ▸ Play All, preview).
554    pub fn state_at(&self, shape: ShapeId, para: Option<usize>, time: f64) -> AnimState {
555        let (s, t) = self.locate(time);
556        self.state(shape, para, s, t)
557    }

562        return Status::Done;
563    }
564    if e.step > step {
565        return Status::NotStarted;
566    }
567    let t = finite_or(t, 0.0);
568    if t <= 0.0 {
569        return Status::NotStarted;
570    }
571    let local = t - e.start;
572    if local < 0.0 {
573        return Status::NotStarted;
574    }
575    let dur = e.duration;
576    if dur.is_nan() || dur <= 0.0 || !dur.is_finite() {
577        return Status::Done;
578    }
579    if !e.until_next_click && local >= e.span {
580        return Status::Done;
581    }
582    let period = if e.auto_reverse { 2.0 * dur } else { dur };
583    let c = local % period;
584    let c = if c.is_finite() { c } else { 0.0 };
585    let p = if c > dur { 2.0 - c / dur } else { c / dur };
586    Status::Active(p.clamp(0.0, 1.0))
587}

589/// Shapes by id in the order `Slide::shape` searches (a shape before its children, groups at most
590/// 64 deep); the first shape with an id wins a duplicate.
591fn shape_index(shapes: &[Shape]) -> HashMap<ShapeId, &Shape> {
592    fn walk<'a>(shapes: &'a [Shape], depth: usize, out: &mut HashMap<ShapeId, &'a Shape>) {
593        if depth > 64 {
594            return;
595        }
596        for s in shapes {
597            out.entry(s.id).or_insert(s);
598            walk(s.children(), depth + 1, out);
599        }
600    }
601    let mut out = HashMap::new();
602    walk(shapes, 0, &mut out);
603    out
604}

606fn build_spec(a: &Animation, xf: Xfrm, sw: f64, sh: f64, resolve: &dyn Fn(&crate::model::ColorRef) -> Rgba) -> EffectSpec {
607    let effect = if a.class == AnimClass::Exit { exit_base(&a.effect).to_string() } else { a.effect.clone() };
608    let path = if a.class == AnimClass::Path { Some(motion_path(&a.effect, &a.option, a.path.as_deref())) } else { None };
609    EffectSpec {
610        class: a.class,
611        effect,
612        // Sink Down is Rise Up played backwards, leaving downward.
613        option: if a.effect == "sinkDown" { "d".into() } else { a.option.clone() },
614        accel: a.smooth_start,
615        decel: a.smooth_end,
616        bounce: a.bounce_end,
617        amount: a.amount,
618        color: a.color.as_ref().map(resolve),
619        path,
620        xf,
621        slide_w: sw,
622        slide_h: sh,
623    }
