1//! Clicks in a slide show: which shape a click lands on, what it starts (a triggered animation, a
2//! link, the next step) and what the show remembers about each slide it has visited.
3//!
4//! The rules follow PowerPoint, as observed on its slide show:
5//! - the topmost visible shape under the pointer decides; a shape without an action on top of one
6//!   with an action blocks it (the click then counts as a click on the background);
7//! - a shape with a triggered animation plays the next step of it per click, and starts over once all
8//!   its steps have played; clicking it while a step plays completes the step at once;
9//! - a shape with a link and a triggered animation follows the link at once. The animation's clock
10//!   runs only while its slide is shown, so it goes on where it stopped when the show returns;
11//! - with "advance on click" off a click never leaves the slide, but it still plays animation steps;
12//! - a slide the show returns to (link or arrow keys) looks as it was left.
13

19/// What was played on one slide.
20#[derive(Clone, Debug, Default, PartialEq)]
21pub(crate) struct SlideRun {
22    /// The show has been on this slide.
23    pub visited: bool,
24    /// Main-sequence steps played when the show left the slide.
25    pub step: usize,
26    /// Triggered sequences started by clicks.
27    pub triggers: Vec<TriggerRun>,
28    /// Steps of triggered sequences started so far (the stamps of [`StepStamp`]).
29    pub clock: u64,
30}
31
32/// Progress of one triggered sequence.
33#[derive(Clone, Debug, PartialEq)]
34pub(crate) struct TriggerRun {
35    pub shape: ShapeId,
36    /// Clicks whose step has played to its end.
37    pub done: usize,
38    /// Seconds the current step has played; `None` when no step is playing.
39    pub elapsed: Option<f64>,
40    /// When each step of this run was started, by step.
41    pub stamps: Vec<u64>,
42}

44/// One shape of the slide for hit testing, in drawing order (back to front; a group before its
45/// children), with whether the pointer is inside its box.
46#[derive(Clone, Debug, PartialEq)]
47pub struct HitShape {
48    pub id: ShapeId,
49    /// Index of the enclosing group in the same list.
50    pub parent: Option<usize>,
51    pub is_group: bool,
52    /// The shape's own action.
53    pub link: Option<Action>,
54    pub under_pointer: bool,
55}

57/// What the pointer is over: a shape that does something when clicked.
58#[derive(Clone, Debug, PartialEq)]
59pub struct Target {
60    pub shape: ShapeId,
61    pub link: Option<Action>,
62}

64/// What a click did.
65#[derive(Clone, Debug, PartialEq)]
66pub enum Click {
67    /// Nothing happened.
68    None,
69    /// A triggered animation started or was completed: redraw.
70    Animated,
71    /// Navigation inside the show (already applied to the state).
72    Show(ShowAction),
73    /// Open this URL outside the show.
74    Open(String),
75    /// End the show.
76    Exit,
77}

79impl From<LinkJump> for Click {
80    fn from(j: LinkJump) -> Click {
81        match j {
82            LinkJump::Show(a) => Click::Show(a),
83            LinkJump::Open(u) => Click::Open(u),
84            LinkJump::Exit => Click::Exit,
85            LinkJump::None => Click::None,
86        }
87    }
88}

90fn finite_or_zero(v: f64) -> f64 {
91    if v.is_finite() { v } else { 0.0 }
92}

94impl ShowState {
95    fn animated(&self, pres: &Presentation) -> bool {
96        !pres.show.without_animation
97    }
98
99    /// Does clicking `shape` start a triggered sequence? A sequence can be started again once it has
100    /// played to its end.
101    fn trigger_live(&self, pres: &Presentation, tl: &Timeline, shape: ShapeId) -> bool {
102        self.animated(pres) && tl.trigger_steps(shape) > 0
103    }
104
105    /// The playheads of the current slide's triggered sequences at the last tick, for
106    /// [`Timeline::state_with`].
107    pub fn trigger_plays(&self) -> Vec<TriggerPlay> {
108        match self.runs.get(&self.slide) {
109            Some(r) => r.triggers.iter().map(|t| TriggerPlay { trigger: t.shape, step: t.done, t: t.elapsed.unwrap_or(0.0) }).collect(),
110            None => vec![],
111        }
112    }
113
114    /// When the steps of the current slide's triggered sequences were started.
115    pub fn trigger_stamps(&self) -> Vec<StepStamp> {
116        let Some(r) = self.runs.get(&self.slide) else { return vec![] };
117        r.triggers.iter().flat_map(|t| t.stamps.iter().enumerate().map(|(step, &at)| StepStamp { trigger: t.shape, step, at })).collect()
118    }
119
120    /// Is a triggered animation of the current slide playing (so the picture changes every frame)?
121    pub fn trigger_playing(&self) -> bool {
122        self.runs.get(&self.slide).is_some_and(|r| r.triggers.iter().any(|t| t.elapsed.is_some()))
123    }
124
125    /// Animation state of `shape` on the current slide: main step `step` with `t_main` seconds
126    /// played, plus the triggered sequences.
127    pub fn anim_state(&self, tl: &Timeline, shape: ShapeId, t_main: f64) -> AnimState {
128        tl.state_ordered(shape, None, self.step, t_main, &self.trigger_plays(), &self.trigger_stamps())
129    }
130
131    /// Advance the clock of the playing triggered animations to `now` (seconds, any steady clock)
132    /// and complete the steps that reached their end. [`Click::Animated`] when the picture changed.
133    pub fn tick(&mut self, tl: &Timeline, now: f64) -> Click {
134        let dt = match self.last_now {
135            Some(last) if now.is_finite() => (now - last).clamp(0.0, 3600.0),
136            _ => 0.0,
137        };
138        if now.is_finite() {
139            self.last_now = Some(now);
140        }
141        let Some(run) = self.runs.get_mut(&self.slide) else { return Click::None };
142        let mut changed = false;
143        for t in run.triggers.iter_mut() {
144            let Some(e) = t.elapsed else { continue };
145            let e = finite_or_zero(e) + dt;
146            if e >= tl.trigger_step_duration(t.shape, t.done) {
147                t.done += 1;
148                t.elapsed = None;
149                changed = true;
150            } else {
151                t.elapsed = Some(e);
152            }
153        }
154        if changed { Click::Animated } else { Click::None }
155    }
156
157    /// Index of the topmost shape a click lands on: the front-most one under the pointer that is
158    /// shown. A group without an action does not count; its children and the space inside its box
159    /// are not it.
160    pub fn top(&self, pres: &Presentation, tl: &Timeline, shapes: &[HitShape], visible: &dyn Fn(ShapeId) -> bool) -> Option<usize> {
161        let shown = |mut i: usize| -> bool {
162            // A shape is shown when it and the groups around it are.
163            for _ in 0..=shapes.len() {
164                let Some(h) = shapes.get(i) else { return false };
165                if !visible(h.id) {
166                    return false;
167                }
168                match h.parent {
169                    Some(p) => i = p,
170                    None => return true,
171                }
172            }
173            false
174        };
175        (0..shapes.len())
176            .rev()
177            .find(|&i| shapes.get(i).is_some_and(|h| h.under_pointer && !(h.is_group && !self.interactive(pres, tl, h)) && shown(i)))
178    }
179
180    fn interactive(&self, pres: &Presentation, tl: &Timeline, h: &HitShape) -> bool {
181        h.link.is_some() || self.trigger_live(pres, tl, h.id)
182    }
183
184    /// The shape under the pointer that a click would act on, `None` when a click would be a click
185    /// on the background. `visible` tells whether a shape is currently shown (not yet entered or
186    /// already exited shapes cannot be clicked).
187    pub fn target(&self, pres: &Presentation, tl: &Timeline, shapes: &[HitShape], visible: &dyn Fn(ShapeId) -> bool) -> Option<Target> {
188        // The topmost shape, or the group it is in, that has an action. A shape without one blocks
189        // what lies beneath it.
190        let mut at = self.top(pres, tl, shapes, visible);
191        for _ in 0..=shapes.len() {
192            let h = shapes.get(at?)?;
193            if self.interactive(pres, tl, h) {
194                return Some(Target { shape: h.id, link: h.link.clone() });
195            }
196            at = h.parent;
197        }
198        None
199    }

201    /// A mouse click. `shapes` and `visible` describe what is under the pointer (see
202    /// [`ShowState::target`]); `last_viewed` is the slide shown before this one.
203    pub fn click(
204        &mut self,
205        pres: &Presentation,
206        tl: &Timeline,
207        shapes: &[HitShape],
208        visible: &dyn Fn(ShapeId) -> bool,
209        now: f64,
210        last_viewed: Option<usize>,
211    ) -> Click {
212        self.tick(tl, now);
213        if self.ended {
214            return Click::None;
215        }
216        let Some(t) = self.target(pres, tl, shapes, visible) else { return self.background_click(pres) };
217        let mut animated = false;
218        if self.trigger_live(pres, tl, t.shape) {
219            let slide = self.slide;
220            let run = self.runs.entry(slide).or_default();
221            let at = match run.triggers.iter().position(|r| r.shape == t.shape) {
222                Some(p) => p,
223                None => {
224                    run.triggers.push(TriggerRun { shape: t.shape, done: 0, elapsed: None, stamps: vec![] });
225                    run.triggers.len() - 1
226                }
227            };
228            let stamp = run.clock.saturating_add(1);
229            if let Some(r) = run.triggers.get_mut(at) {
230                if r.elapsed.is_some() {
231                    // A click on a shape whose step is playing completes the step.
232                    r.done += 1;
233                    r.elapsed = None;
234                } else {
235                    if r.done >= tl.trigger_steps(t.shape) {
236                        // All steps have played: the sequence starts over.
237                        r.done = 0;
238                        r.stamps.clear();
239                    }
240                    r.stamps.push(stamp);
241                    run.clock = stamp;
242                    if tl.trigger_step_duration(t.shape, r.done) > 0.0 {
243                        r.elapsed = Some(0.0);
244                    } else {
245                        r.done += 1;
246                    }
247                }
248                animated = true;
249            }
250        }
251        match t.link {
252            // The link is followed at once, whatever the animation still has to play.
253            Some(a) => self.follow(pres, &a, last_viewed).into(),
254            None if animated => Click::Animated,
255            None => Click::None,
256        }
257    }

259    /// A click that is on no shape with an action: next animation step, then next slide, unless the
260    /// slide has "advance on click" off, which leaves only the animation steps.
261    fn background_click(&mut self, pres: &Presentation) -> Click {
262        let advance = pres.slides.get(self.slide).and_then(|s| s.transition.as_ref()).is_none_or(|t| t.advance_on_click);
263        let steps_left = self.playing || self.step < ShowState::steps(pres, self.slide);
264        if advance || steps_left { Click::Show(self.next(pres)) } else { Click::None }
265    }
