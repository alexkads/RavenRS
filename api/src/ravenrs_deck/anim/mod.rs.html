1//! Slide show timing for DeckCraft.
2//!
3//! - [`Timeline`]: a slide's animation list grouped into click steps, with per-shape (and
4//!   per-paragraph) [`AnimState`] at any point of the show.
5//! - [`transition_layers`]: a slide transition frame as textured quads of the old/new slide images.
6//! - [`morph_frame`]: a Morph transition frame (shapes matched by [`morph_pairs`], moved by
7//!   [`morph_xfrm`]).
8//! - [`ShowState`]: slide show navigation (hidden slides, custom shows, loop, timings), what was
9//!   played on each visited slide, and what a click does ([`ShowState::click`]).
10//!
11//! Pure logic: no rendering, no UI toolkit. Every function accepts hostile numbers (NaN, huge or
12//! negative times) and never panics.

47/// Animation state of one shape (or one paragraph of a shape's text) at one moment of the show.
48///
49/// Same meaning as the renderer's `ShapeState`: translation in points, scale and rotation (degrees)
50/// around the shape's centre, a reveal clip in shape-local unit space `[l, t, r, b]`, and a fill
51/// colour override.
52#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
53pub struct AnimState {
54    pub visible: bool,
55    pub opacity: f64,
56    pub offset_x: f64,
57    pub offset_y: f64,
58    pub scale_x: f64,
59    pub scale_y: f64,
60    pub rotate: f64,
61    pub clip: Option<[f64; 4]>,
62    pub tint: Option<Rgba>,
63}
64
65impl Default for AnimState {
66    fn default() -> Self {
67        AnimState { visible: true, opacity: 1.0, offset_x: 0.0, offset_y: 0.0, scale_x: 1.0, scale_y: 1.0, rotate: 0.0, clip: None, tint: None }
68    }
69}
70
71impl AnimState {
72    /// A hidden state.
73    pub fn hidden() -> Self {
74        AnimState { visible: false, ..Default::default() }
75    }
76    /// Is this the identity (nothing to apply)?
77    pub fn is_identity(&self) -> bool {
78        *self == AnimState::default()
79    }
80    /// Apply `d` on top of `self`: opacities and scales multiply, offsets and rotations add, clips
81    /// intersect, a tint replaces.
82    pub fn compose(&self, d: &AnimState) -> AnimState {
83        let clip = match (self.clip, d.clip) {
84            (Some(a), Some(b)) => Some([a[0].max(b[0]), a[1].max(b[1]), a[2].min(b[2]), a[3].min(b[3])]),
85            (a, None) => a,
86            (None, b) => b,
87        };
88        AnimState {
89            visible: self.visible && d.visible,
90            opacity: self.opacity * d.opacity,
91            offset_x: self.offset_x + d.offset_x,
92            offset_y: self.offset_y + d.offset_y,
93            scale_x: self.scale_x * d.scale_x,
94            scale_y: self.scale_y * d.scale_y,
95            rotate: self.rotate + d.rotate,
96            clip,
97            tint: d.tint.or(self.tint),
98        }
99    }
100    /// Replace any non-finite number with its neutral value and clamp opacity and clip.
101    pub fn sanitized(mut self) -> AnimState {
102        let f = |v: f64, d: f64| if v.is_finite() { v } else { d };
103        self.opacity = f(self.opacity, 1.0).clamp(0.0, 1.0);
104        self.offset_x = f(self.offset_x, 0.0);
105        self.offset_y = f(self.offset_y, 0.0);
106        self.scale_x = f(self.scale_x, 1.0);
107        self.scale_y = f(self.scale_y, 1.0);
108        self.rotate = f(self.rotate, 0.0);
109        self.clip = self.clip.map(|c| {
110            let l = f(c[0], 0.0).clamp(0.0, 1.0);
111            let t = f(c[1], 0.0).clamp(0.0, 1.0);
112            let r = f(c[2], 1.0).clamp(0.0, 1.0);
113            let b = f(c[3], 1.0).clamp(0.0, 1.0);
114            [l, t, r.max(l), b.max(t)]
115        });
116        self
117    }
118}

120/// Clamp a hostile number into `[lo, hi]`, mapping NaN to `nan`.
121pub(crate) fn clampf(v: f64, lo: f64, hi: f64, nan: f64) -> f64 {
122    if v.is_nan() { nan } else { v.clamp(lo, hi) }
