1//! Time easing: acceleration / deceleration fractions and bounce.
2
3use crate::anim::clampf;
4
5/// Map linear progress `p` (0..1) through PowerPoint-style easing: `accel` and `decel` are the
6/// fractions of the duration spent speeding up and slowing down (velocity ramps linearly);
7/// `bounce` (0..1) makes the last part of the motion bounce against the end.
8pub fn ease(p: f64, accel: f64, decel: f64, bounce: f64) -> f64 {
9    let p = clampf(p, 0.0, 1.0, 0.0);
10    let mut a = clampf(accel, 0.0, 1.0, 0.0);
11    let mut d = clampf(decel, 0.0, 1.0, 0.0);
12    if a + d > 1.0 {
13        let s = a + d;
14        a /= s;
15        d /= s;
16    }
17    let v = 2.0 / (2.0 - a - d);
18    let mut y = if a > 0.0 && p < a {
19        v * p * p / (2.0 * a)
20    } else if d > 0.0 && p > 1.0 - d {
21        let q = 1.0 - p;
22        1.0 - v * q * q / (2.0 * d)
23    } else {
24        v * a / 2.0 + v * (p - a)
25    };
26    let b = clampf(bounce, 0.0, 1.0, 0.0);
27    if b > 0.0 {
28        // Reach the end early, then bounce back with decaying hops over the last `b` fraction.
29        let hit = 1.0 - b;
30        if p < hit {
31            let q = if hit > 0.0 { p / hit } else { 1.0 };
32            y = q * q;
33        } else {
34            let s = (p - hit) / b;
35            y = 1.0 - bounce_hops(s) * 0.25;
36        }
37    }
38    clampf(y, -1.0, 2.0, 0.0)
39}

41/// Decaying hops for s in 0..1: 0 at the ends of each hop, peaks shrinking.
42fn bounce_hops(s: f64) -> f64 {
43    let s = clampf(s, 0.0, 1.0, 1.0);
44    let hops = 3.0;
45    let k = (s * hops).floor().min(hops - 1.0);
46    let local = s * hops - k;
47    let amp = 0.5f64.powf(k);
48    amp * (local * std::f64::consts::PI).sin()
49}

51/// Classic bounce-out: falls to 1 and bounces with decaying height.
52pub(crate) fn bounce_out(p: f64) -> f64 {
53    let p = clampf(p, 0.0, 1.0, 0.0);
54    let n = 7.5625;
55    let d = 2.75;
56    let y = if p < 1.0 / d {
57        n * p * p
58    } else if p < 2.0 / d {
59        let q = p - 1.5 / d;
60        n * q * q + 0.75
61    } else if p < 2.5 / d {
62        let q = p - 2.25 / d;
63        n * q * q + 0.9375
64    } else {
65        let q = p - 2.625 / d;
66        n * q * q + 0.984375
67    };
68    y.min(1.0)
69}

71/// Smooth ease-in-out (cubic smoothstep).
72pub(crate) fn smooth(p: f64) -> f64 {
73    let p = clampf(p, 0.0, 1.0, 0.0);
74    p * p * (3.0 - 2.0 * p)
75}
