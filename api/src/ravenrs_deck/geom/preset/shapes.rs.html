5pub(super) fn wedge_rect(w: f64, h: f64, a0: f64, a1: f64, r: f64) -> Geometry {
6    let tip = Point::new(w / 2.0 + w * a0 / 1e5, h / 2.0 + h * a1 / 1e5);
7    let inside = tip.x > 0.0 && tip.x < w && tip.y > 0.0 && tip.y < h;
8    let (dx, dy) = ((tip.x - w / 2.0) / w.max(1e-9), (tip.y - h / 2.0) / h.max(1e-9));
9    let mut b = Pb::new();
10    let r = r.min(w / 2.0).min(h / 2.0);
11    // Walk clockwise from the top-left; insert the wedge on the edge facing the tip.
12    let horiz = dx.abs() > dy.abs();
13    // The wedge's base: 2..5 twelfths of the edge on the tip's side of the centre when it is
14    // before it, 7..10 twelfths when after (presetShapeDefinitions' xg1/xg2, yg1/yg2).
15    let (bx0, bx1) = if tip.x < w / 2.0 { (w * 2.0 / 12.0, w * 5.0 / 12.0) } else { (w * 7.0 / 12.0, w * 10.0 / 12.0) };
16    let (by0, by1) = if tip.y < h / 2.0 { (h * 2.0 / 12.0, h * 5.0 / 12.0) } else { (h * 7.0 / 12.0, h * 10.0 / 12.0) };
17    b.m(r, 0.0);
18    if !inside && !horiz && dy < 0.0 {
19        b.l(bx0, 0.0).l(tip.x, tip.y).l(bx1, 0.0);
20    }
21    b.l(w - r, 0.0);
22    if r > 0.0 {
23        b.arc(r, r, 270.0, 90.0);
24    }
25    if !inside && horiz && dx > 0.0 {
26        b.l(w, by0).l(tip.x, tip.y).l(w, by1);
27    }
28    b.l(w, h - r);
29    if r > 0.0 {
30        b.arc(r, r, 0.0, 90.0);
31    }
32    if !inside && !horiz && dy > 0.0 {
33        b.l(bx1, h).l(tip.x, tip.y).l(bx0, h);
34    }
35    b.l(r, h);
36    if r > 0.0 {
37        b.arc(r, r, 90.0, 90.0);
38    }
39    if !inside && horiz && dx < 0.0 {
40        b.l(0.0, by1).l(tip.x, tip.y).l(0.0, by0);
41    }
42    b.l(0.0, r);
43    if r > 0.0 {
44        b.arc(r, r, 180.0, 90.0);
45    }
46    b.z();
47    let mut g = simple(b.take(), Rect::new(0.0, 0.0, w, h).inset(-r * 0.29289), w, h);
48    g.handles.push(tip_handle(w, h, tip));
49    g
50}

83pub(super) fn cloud(name: &str, w: f64, h: f64, a0: f64, a1: f64) -> Geometry {
84    // Drawn in the path's own 43200-unit square, then stretched to the box (as DrawingML lays out
85    // a path with its own w and h), so the arcs keep their shape's proportions at any aspect.
86    let to_box = kurbo::Affine::scale_non_uniform(w / 43200.0, h / 43200.0);
87    let deg = |v: f64| v / 60000.0;
88    let mut b = Pb::new();
89    b.m(3900.0, 14370.0);
90    for (wr, hr, st, sw) in CLOUD_ARCS {
91        b.arc_spec(wr, hr, deg(st), deg(sw));
92    }
93    b.z();
94    let mut creases = Pb::new();
95    for (x, y, wr, hr, st, sw) in CLOUD_CREASES {
96        creases.m(x, y).arc_spec(wr, hr, deg(st), deg(sw));
97    }
98    let mut outline = b.take();
99    outline.apply_affine(to_box);
100    let mut paths = vec![fill(outline)];
101    if name == "cloud" {
102        let mut c = creases.take();
103        c.apply_affine(to_box);
104        paths.push(stroke_only(c));
105    }
106    let text_rect = Rect::new(w * 2977.0 / 21600.0, h * 3262.0 / 21600.0, w * 17087.0 / 21600.0, h * 17337.0 / 21600.0);
107    let mut handles = vec![];
108    if name == "cloudCallout" {
109        let tip = Point::new(w / 2.0 + w * a0 / 1e5, h / 2.0 + h * a1 / 1e5);
110        let c = Point::new(w / 2.0, h / 2.0);
111        let mut bubbles = Pb::new();
112        for (k, r) in [(0.55, 0.06), (0.75, 0.045), (0.95, 0.03)] {
113            let p = c + (tip - c) * k;
114            if !(p.x > w * 0.1 && p.x < w * 0.9 && p.y > h * 0.1 && p.y < h * 0.9) || k > 0.9 {
115                bubbles.ellipse(p.x, p.y, w * r, h * r);
116            }
117        }
118        paths.push(fill(bubbles.take()));
119        handles.push(tip_handle(w, h, tip));
120    }
121    Geometry { paths, text_rect, handles, sites: mid_sites(w, h) }
122}

124/// The equation shapes, by ECMA-376's formulas (checked against PowerPoint's outlines at several
125/// sizes and adjust values). Their bars span 73.49% of the width; the plus and multiply bars are
126/// `adj1` of the shorter side thick, the others' `adj1` of the height.
127pub(super) fn math(name: &str, w: f64, h: f64, a0: f64, a1: f64, a2: f64) -> Geometry {
128    let pin = |lo: f64, v: f64, hi: f64| v.max(lo).min(hi.max(lo));
129    let ss = w.min(h);
130    let (hc, vc) = (w / 2.0, h / 2.0);
131    let dx1 = w * 73490.0 / 200000.0;
132    let (x1, x2) = (hc - dx1, hc + dx1);
133    let mut b = Pb::new();
134    let text = match name {
135        "mathPlus" => {
136            let a = pin(0.0, a0, 73490.0);
137            let dy1 = h * 73490.0 / 200000.0;
138            let dx2 = ss * a / 200000.0;
139            let (y1, y2, y3, y4) = (vc - dy1, vc - dx2, vc + dx2, vc + dy1);
140            let (xa, xb) = (hc - dx2, hc + dx2);
141            b.poly(&[(x1, y2), (xa, y2), (xa, y1), (xb, y1), (xb, y2), (x2, y2), (x2, y3), (xb, y3), (xb, y4), (xa, y4), (xa, y3), (x1, y3)]);
142            Rect::new(x1, y2, x2, y3)
143        }
144        "mathMinus" => {
145            let dy1 = h * pin(0.0, a0, 100000.0) / 200000.0;
146            b.poly(&[(x1, vc - dy1), (x2, vc - dy1), (x2, vc + dy1), (x1, vc + dy1)]);
147            Rect::new(x1, vc - dy1, x2, vc + dy1)
148        }
149        "mathEqual" => {
150            let a = pin(0.0, a0, 36745.0);
151            let g = pin(0.0, a1, 100000.0 - 2.0 * a);
152            let (dy1, dy2) = (h * a / 100000.0, h * g / 200000.0);
153            let (y2, y3) = (vc - dy2, vc + dy2);
154            let (y1, y4) = (y2 - dy1, y3 + dy1);
155            b.poly(&[(x1, y1), (x2, y1), (x2, y2), (x1, y2)]);
156            b.poly(&[(x1, y3), (x2, y3), (x2, y4), (x1, y4)]);
157            Rect::new(x1, y1, x2, y4)
158        }
159        "mathNotEqual" => {
160            let a = pin(0.0, a0, 50000.0);
161            let cr_ang = pin(4200000.0, a1, 6600000.0);
162            let g = pin(0.0, a2, 100000.0 - 2.0 * a);
163            let (dy1, dy2) = (h * a / 100000.0, h * g / 200000.0);
164            let hd2 = (h / 2.0).max(1e-9);
165            let (y2, y3) = (vc - dy2, vc + dy2);
166            let (y1, y4) = (y2 - dy1, y3 + dy1);
167            let cadj2 = cr_ang - 5400000.0;
168            let xadj2 = hd2 * (cadj2 / 60000.0).to_radians().tan();
169            let len = xadj2.hypot(hd2);
170            let bhw = len * dy1 / hd2;
171            let x7 = hc + xadj2 - bhw / 2.0;
172            let at = |y: f64| x7 - xadj2 * y / hd2;
173            let (x6, x5, x4, x3) = (at(y1), at(y2), at(y3), at(y4));
174            let rx7 = x7 + bhw;
175            let dx7 = dy1 * hd2 / len;
176            let dy3 = dy1 * xadj2 / len;
177            let (rx, lx, ry, ly) = if cadj2 > 0.0 { (x7 + dx7, x7, dy3, 0.0) } else { (rx7, rx7 - dx7, 0.0, -dy3) };
178            b.poly(&[
179                (x1, y1),
180                (x6, y1),
181                (lx, ly),
182                (rx, ry),
183                (x6 + bhw, y1),
184                (x2, y1),
185                (x2, y2),
186                (x5 + bhw, y2),
187                (x4 + bhw, y3),
188                (x2, y3),
189                (x2, y4),
190                (x3 + bhw, y4),
191                (w - lx, h - ly),
192                (w - rx, h - ry),
193                (x3, y4),
194                (x1, y4),
195                (x1, y3),
196                (x4, y3),
197                (x5, y2),
198                (x1, y2),
199            ]);
200            Rect::new(x1, y1, x2, y4)
201        }
202        "mathDivide" => {
203            let a = pin(1000.0, a0, 36745.0);
204            let max3 = ((73490.0 - a) / 4.0).min(36745.0 * w / h.max(1e-9));
205            let r3 = pin(1000.0, a2, max3);
206            let g = pin(0.0, a1, 73490.0 - 4.0 * r3 - a);
207            let dy1 = h * a / 200000.0;
208            let yg = h * g / 100000.0;
209            let rad = h * r3 / 100000.0;
210            let (y3, y4) = (vc - dy1, vc + dy1);
211            let y2 = y3 - (yg + rad);
212            let y1 = y2 - rad;
213            b.ellipse(hc, y1 + rad, rad, rad);
214            b.ellipse(hc, h - y1 - rad, rad, rad);
215            b.poly(&[(x1, y3), (x2, y3), (x2, y4), (x1, y4)]);
216            Rect::new(x1, y3, x2, y4)
217        }
218        _ => {
219            // mathMultiply: an X whose arms run along the box's diagonals.
220            let th = ss * pin(0.0, a0, 51965.0) / 100000.0;
221            let ang = h.atan2(w);
222            let (sa, ca, ta) = (ang.sin(), ang.cos(), ang.tan().max(1e-9));
223            let lm = w.hypot(h) * (1.0 - 51965.0 / 100000.0);
224            let (xm, ym) = (ca * lm / 2.0, sa * lm / 2.0);
225            let (dxam, dyam) = (sa * th / 2.0, ca * th / 2.0);
226            let (xa, ya, xb, yb) = (xm - dxam, ym + dyam, xm + dxam, ym - dyam);
227            let yc = (hc - xb) * ta + yb;
228            let (xd, xe) = (w - xb, w - xa);
229            let xfe = (vc - ya) / ta;
230            let (xf, xl) = (xe - xfe, xa + xfe);
231            let (yg, yh, yi) = (h - ya, h - yb, h - yc);
232            b.poly(&[(xa, ya), (xb, yb), (hc, yc), (xd, yb), (xe, ya), (xf, vc), (xe, yg), (xd, yh), (hc, yi), (xb, yh), (xa, yg), (xl, vc)]);
233            Rect::new(xa, yb, xe, yh)
234        }
235    };
236    Geometry { paths: vec![fill(b.take())], text_rect: text, handles: vec![], sites: mid_sites(w, h) }
237}

239pub(super) fn ribbon(up: bool, w: f64, h: f64, a0: f64, a1: f64) -> Geometry {
240    let fold = h * a0.clamp(0.0, 33333.0) / 1e5;
241    let cw = w * a1.clamp(25000.0, 75000.0) / 1e5; // centre band width
242    let x0 = (w - cw) / 2.0;
243    let x1 = w - x0;
244    let ear = w / 8.0;
245    let (top, bot) = (0.0, h);
246    let mut b = Pb::new();
247    let mut shade_p = Pb::new();
248    if up {
249        // centre band sits high; ends hang low.
250        b.poly(&[(x0, top), (x1, top), (x1, bot - fold), (x0, bot - fold)]);
251        let mut ends = Pb::new();
252        ends.poly(&[(0.0, fold), (x0 + ear * 0.5, fold), (x0 + ear * 0.5, bot), (0.0, bot), (ear, (fold + bot) / 2.0)]);
253        ends.poly(&[(w, fold), (x1 - ear * 0.5, fold), (x1 - ear * 0.5, bot), (w, bot), (w - ear, (fold + bot) / 2.0)]);
254        shade_p.poly(&[(x0, bot - fold), (x0 + ear * 0.5, bot - fold), (x0 + ear * 0.5, bot)]);
255        shade_p.poly(&[(x1, bot - fold), (x1 - ear * 0.5, bot - fold), (x1 - ear * 0.5, bot)]);
256        let text = Rect::new(x0, top, x1, bot - fold);
257        return Geometry {
258            paths: vec![fill(ends.take()), shade(shade_p.take(), FillMode::DarkenLess), fill(b.take())],
259            text_rect: text,
260            handles: vec![],
261            sites: mid_sites(w, h),
262        };
263    }
264    b.poly(&[(x0, top + fold), (x1, top + fold), (x1, bot), (x0, bot)]);
265    let mut ends = Pb::new();
266    ends.poly(&[(0.0, top), (x0 + ear * 0.5, top), (x0 + ear * 0.5, bot - fold), (0.0, bot - fold), (ear, (bot - fold) / 2.0)]);
267    ends.poly(&[(w, top), (x1 - ear * 0.5, top), (x1 - ear * 0.5, bot - fold), (w, bot - fold), (w - ear, (bot - fold) / 2.0)]);
268    shade_p.poly(&[(x0, top + fold), (x0 + ear * 0.5, top + fold), (x0 + ear * 0.5, top)]);
269    shade_p.poly(&[(x1, top + fold), (x1 - ear * 0.5, top + fold), (x1 - ear * 0.5, top)]);
270    let text = Rect::new(x0, top + fold, x1, bot);
271    Geometry {
272        paths: vec![fill(ends.take()), shade(shade_p.take(), FillMode::DarkenLess), fill(b.take())],
273        text_rect: text,
274        handles: vec![],
275        sites: mid_sites(w, h),
276    }
277}

279pub(super) fn scroll(vertical: bool, w: f64, h: f64, a0: f64) -> Geometry {
280    let ss = w.min(h);
281    let r = ss * a0.clamp(0.0, 25000.0) / 1e5 / 2.0;
282    let mut b = Pb::new();
283    let mut curl = Pb::new();
284    if vertical {
285        b.poly(&[(r, r * 2.0), (w - r, r * 2.0), (w - r, h - r), (r, h - r)]);
286        b.ellipse(w / 2.0, r, w / 2.0 - r * 0.0, r);
287        curl.ellipse(r, h - r, r, r);
288        curl.ellipse(w - r * 2.0, r, r * 0.5, r * 0.5);
289        let text = Rect::new(r, r * 2.0, w - r, h - r);
290        return Geometry {
291            paths: vec![fill(b.take()), shade(curl.take(), FillMode::DarkenLess)],
292            text_rect: text,
293            handles: vec![],
294            sites: mid_sites(w, h),
295        };
296    }
297    b.poly(&[(r * 2.0, r), (w - r, r), (w - r, h - r), (r * 2.0, h - r)]);
298    b.ellipse(r, h / 2.0, r, h / 2.0);
299    curl.ellipse(w - r, r, r, r);
300    curl.ellipse(r, h - r * 2.0, r * 0.5, r * 0.5);
301    let text = Rect::new(r * 2.0, r, w - r, h - r);
302    Geometry { paths: vec![fill(b.take()), shade(curl.take(), FillMode::DarkenLess)], text_rect: text, handles: vec![], sites: mid_sites(w, h) }
303}

305pub(super) fn action_button(name: &str, w: f64, h: f64) -> Geometry {
306    let body = corner_rect(w, h, [Corner::Square; 4]);
307    let ss = w.min(h);
308    let (cx, cy) = (w / 2.0, h / 2.0);
309    let s = ss * 0.3;
310    let mut g = Pb::new();
311    match name {
312        "actionButtonBackPrevious" => {
313            g.poly(&[(cx + s, cy - s), (cx + s, cy + s), (cx - s, cy)]);
314        }
315        "actionButtonForwardNext" => {
316            g.poly(&[(cx - s, cy - s), (cx - s, cy + s), (cx + s, cy)]);
317        }
318        "actionButtonBeginning" => {
319            g.poly(&[(cx + s, cy - s), (cx + s, cy + s), (cx - s * 0.6, cy)]);
320            g.poly(&[(cx - s, cy - s), (cx - s * 0.7, cy - s), (cx - s * 0.7, cy + s), (cx - s, cy + s)]);
321        }
322        "actionButtonEnd" => {
323            g.poly(&[(cx - s, cy - s), (cx - s, cy + s), (cx + s * 0.6, cy)]);
324            g.poly(&[(cx + s * 0.7, cy - s), (cx + s, cy - s), (cx + s, cy + s), (cx + s * 0.7, cy + s)]);

328                (cx, cy - s),
329                (cx + s, cy),
330                (cx + s * 0.7, cy),
331                (cx + s * 0.7, cy + s),
332                (cx - s * 0.7, cy + s),
333                (cx - s * 0.7, cy),
334                (cx - s, cy),

336        }

356        }
357        "actionButtonSound" => {
358            g.poly(&[
359                (cx - s, cy - s * 0.35),
360                (cx - s * 0.5, cy - s * 0.35),
361                (cx, cy - s),
362                (cx, cy + s),
363                (cx - s * 0.5, cy + s * 0.35),
364                (cx - s, cy + s * 0.35),
365            ]);
366        }

370        paths: vec![fill(body), shade(g.take(), FillMode::DarkenLess)],

372        handles: vec![],
