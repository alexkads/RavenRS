11impl Pb {
12    pub(super) fn new() -> Self {
13        Pb { p: BezPath::new(), cur: Point::ZERO }
14    }
15    pub(super) fn m(&mut self, x: f64, y: f64) -> &mut Self {
16        self.cur = Point::new(x, y);
17        self.p.move_to(self.cur);
18        self
19    }
20    pub(super) fn l(&mut self, x: f64, y: f64) -> &mut Self {
21        self.cur = Point::new(x, y);
22        self.p.line_to(self.cur);
23        self
24    }
25    pub(super) fn c(&mut self, x1: f64, y1: f64, x2: f64, y2: f64, x: f64, y: f64) -> &mut Self {
26        self.cur = Point::new(x, y);
27        self.p.curve_to(Point::new(x1, y1), Point::new(x2, y2), self.cur);
28        self
29    }
30    pub(super) fn q(&mut self, x1: f64, y1: f64, x: f64, y: f64) -> &mut Self {
31        self.cur = Point::new(x, y);
32        self.p.quad_to(Point::new(x1, y1), self.cur);
33        self
34    }
35    /// Elliptic arc continuing from the current point, which lies on the ellipse at parametric
36    /// angle `st` (degrees); sweeps `sw` degrees (positive = clockwise on screen).
37    pub(super) fn arc(&mut self, wr: f64, hr: f64, st: f64, sw: f64) -> &mut Self {
38        let (s, e) = (st.to_radians(), (st + sw).to_radians());
39        let c = Point::new(self.cur.x - wr * s.cos(), self.cur.y - hr * s.sin());
40        let a = Arc::new(c, Vec2::new(wr.abs(), hr.abs()), s, e - s, 0.0);
41        a.to_cubic_beziers(0.05, |p1, p2, p| {
42            self.p.curve_to(p1, p2, p);
43        });
44        self.cur = Point::new(c.x + wr * e.cos(), c.y + hr * e.sin());
45        self
46    }
47    /// A DrawingML `arcTo`: angles are where the ray from the ellipse's centre at that angle
48    /// meets it (not the ellipse's parameter), so elliptical arcs start and end where the
49    /// standard's shapes expect.
50    pub(super) fn arc_spec(&mut self, wr: f64, hr: f64, st: f64, sw: f64) -> &mut Self {
51        let par = |a: f64| {
52            let a = a.to_radians();
53            (wr * a.sin()).atan2(hr * a.cos())
54        };
55        let t0 = par(st);
56        let mut d = par(st + sw) - t0;
57        let tau = std::f64::consts::TAU;
58        if sw.abs() >= 360.0 {
59            d = tau.copysign(sw);
60        } else {
61            while sw > 0.0 && d < 0.0 {
62                d += tau;
63            }
64            while sw < 0.0 && d > 0.0 {
65                d -= tau;
66            }
67        }
68        self.arc(wr, hr, t0.to_degrees(), d.to_degrees())
69    }
70    pub(super) fn z(&mut self) -> &mut Self {
71        self.p.close_path();
72        self
73    }
74    pub(super) fn poly(&mut self, pts: &[(f64, f64)]) -> &mut Self {
75        for (i, (x, y)) in pts.iter().enumerate() {
76            if i == 0 {
77                self.m(*x, *y);
78            } else {
79                self.l(*x, *y);
80            }
81        }
82        self.z()
83    }
84    pub(super) fn ellipse(&mut self, cx: f64, cy: f64, rx: f64, ry: f64) -> &mut Self {
85        self.m(cx + rx, cy);
86        self.arc(rx, ry, 0.0, 360.0);
87        self.z()
88    }
89    pub(super) fn take(&mut self) -> BezPath {
90        std::mem::take(&mut self.p)
91    }
92}

114/// Parametric angle on an rx×ry ellipse for a visual angle `deg` (file angles are visual).
115pub(super) fn vis(deg: f64, rx: f64, ry: f64) -> f64 {

120pub(super) fn polygon_regular(n: usize, w: f64, h: f64, phase: f64) -> Vec<(f64, f64)> {
121    let pts: Vec<(f64, f64)> = (0..n)
122        .map(|i| {
123            let a = (phase + 360.0 * i as f64 / n as f64).to_radians();
124            (a.cos(), a.sin())
125        })
126        .collect();
127    normalize(&pts, w, h)
128}

130/// Scale `pts` so their bounding box fills 0..w × 0..h.
131fn normalize(pts: &[(f64, f64)], w: f64, h: f64) -> Vec<(f64, f64)> {
132    let (mut x0, mut y0, mut x1, mut y1) = (f64::MAX, f64::MAX, f64::MIN, f64::MIN);
133    for (x, y) in pts {
134        x0 = x0.min(*x);
135        y0 = y0.min(*y);
136        x1 = x1.max(*x);
137        y1 = y1.max(*y);
138    }
139    let sx = if x1 - x0 > 1e-12 { w / (x1 - x0) } else { 0.0 };
140    let sy = if y1 - y0 > 1e-12 { h / (y1 - y0) } else { 0.0 };
141    pts.iter().map(|(x, y)| ((x - x0) * sx, (y - y0) * sy)).collect()
142}

144/// An n-pointed star as ECMA-376 draws it: points on an ellipse around (hc, vc · vf) with radii
145/// half the box stretched by the preset's `hf` / `vf` (so a five-pointed star fills its box and
146/// looks regular in a square), the inner corners `inner` of the way out.
147pub(super) fn star(n: usize, w: f64, h: f64, inner: f64) -> Vec<(f64, f64)> {
148    let (hf, vf) = match n {
149        5 => (1.05146, 1.10557),
150        6 => (1.15470, 1.0),
151        7 => (1.02572, 1.05210),
152        10 => (1.05146, 1.0),
153        _ => (1.0, 1.0),
154    };
155    let (rx, ry, cy) = (w / 2.0 * hf, h / 2.0 * vf, h / 2.0 * vf);
156    let mut pts = Vec::with_capacity(n * 2);
157    for i in 0..n * 2 {
158        let a = (-90.0 + 180.0 * i as f64 / n as f64).to_radians();
159        let r = if i % 2 == 0 { 1.0 } else { inner };
160        pts.push((w / 2.0 + r * rx * a.cos(), cy + r * ry * a.sin()));
161    }
162    pts
163}

165/// The text rectangle ECMA-376's presetShapeDefinitions give a star (`inner` is the adjust value
166/// over 50000), where known: star4's is the square inside its inner points, star5's spans the
167/// inner pentagon from its upper side points to its bottom inner point.
168pub(super) fn star_text_rect(n: usize, w: f64, h: f64, inner: f64) -> Option<Rect> {
169    let (hc, vc) = (w / 2.0, h / 2.0);
170    match n {
171        4 => {
172            let (dx, dy) = (hc * inner * 45f64.to_radians().cos(), vc * inner * 45f64.to_radians().sin());
173            Some(Rect::new(hc - dx, vc - dy, hc + dx, vc + dy))
174        }
175        5 => {
176            let (swd2, shd2, svc) = (hc * 1.05146, vc * 1.10557, vc * 1.10557);
177            let (iwd2, ihd2) = (swd2 * inner, shd2 * inner);
178            let sdx1 = iwd2 * 18f64.to_radians().cos();
179            let sdy1 = ihd2 * 54f64.to_radians().sin();
180            Some(Rect::new(hc - sdx1, svc - sdy1, hc + sdx1, svc + ihd2))
181        }
182        _ => None,
183    }
184}

189    Round(f64),
190    Snip(f64),

193/// A rectangle with each corner (TL, TR, BR, BL) square, rounded or snipped.
194pub(super) fn corner_rect(w: f64, h: f64, c: [Corner; 4]) -> BezPath {
195    let mut b = Pb::new();
196    let size = |k: Corner| match k {
197        Corner::Square => 0.0,
198        Corner::Round(r) | Corner::Snip(r) => r.max(0.0),
199    };
200    let s0 = size(c[0]);
201    b.m(s0, 0.0);
202    // top edge → TR
203    let s1 = size(c[1]);
204    b.l(w - s1, 0.0);
205    match c[1] {
206        Corner::Round(r) if r > 0.0 => {
207            b.arc(r, r, 270.0, 90.0);
208        }
209        Corner::Snip(_) => {
210            b.l(w, s1);
211        }
212        _ => {}
213    }
214    let s2 = size(c[2]);
215    b.l(w, h - s2);
216    match c[2] {
217        Corner::Round(r) if r > 0.0 => {
218            b.arc(r, r, 0.0, 90.0);
219        }
220        Corner::Snip(_) => {
221            b.l(w - s2, h);
222        }
223        _ => {}
224    }
225    let s3 = size(c[3]);
226    b.l(s3, h);
227    match c[3] {
228        Corner::Round(r) if r > 0.0 => {
229            b.arc(r, r, 90.0, 90.0);
230        }
231        Corner::Snip(_) => {
232            b.l(0.0, h - s3);
233        }
234        _ => {}
235    }
236    b.l(0.0, s0);
237    match c[0] {
238        Corner::Round(r) if r > 0.0 => {
239            b.arc(r, r, 180.0, 90.0);
240        }
241        Corner::Snip(_) => {
242            b.l(s0, 0.0);
243        }
244        _ => {}
245    }
246    b.z();
247    b.take()
248}

254/// Orientation for single-axis arrows built along +x.
255#[derive(Clone, Copy)]
256pub(super) enum Dir {
257    Right,
258    Left,
259    Up,
260    Down,
261}

263/// Map canonical (u along the arrow's length L, v across thickness T) to shape space.
264pub(super) fn orient(d: Dir, w: f64, h: f64, pts: &[(f64, f64)]) -> Vec<(f64, f64)> {
265    pts.iter()
266        .map(|&(u, v)| match d {
267            Dir::Right => (u, v),
268            Dir::Left => (w - u, v),
269            Dir::Up => (v, h - u),
270            Dir::Down => (v, u),
271        })
272        .collect()
273}

275pub(super) fn len_thick(d: Dir, w: f64, h: f64) -> (f64, f64) {
276    match d {
277        Dir::Right | Dir::Left => (w, h),
278        Dir::Up | Dir::Down => (h, w),
279    }

282pub(super) fn arrow(d: Dir, w: f64, h: f64, a1: f64, a2: f64) -> Geometry {
283    let (l, t) = len_thick(d, w, h);
284    let ss = l.min(t);
285    let a1 = a1.clamp(0.0, 100000.0) / 100000.0;
286    let dx = (ss * a2.clamp(0.0, 100000.0 * l / ss.max(1e-9)) / 100000.0).min(l);
287    let y1 = t / 2.0 - t * a1 / 2.0;
288    let y2 = t / 2.0 + t * a1 / 2.0;
289    let pts = [(0.0, y1), (l - dx, y1), (l - dx, 0.0), (l, t / 2.0), (l - dx, t), (l - dx, y2), (0.0, y2)];
290    let pts = orient(d, w, h, &pts);
291    let tr = orient(d, w, h, &[(0.0, y1), (l - dx + dx * (y1 / (t / 2.0).max(1e-9)), y2)]);
292    let mut text = Rect::from_points(tr[0], tr[1]);
293    text = text.abs();
294    let mut g = Geometry { paths: vec![fill(Pb::new().poly(&pts).take())], text_rect: text, handles: vec![], sites: mid_sites(w, h) };
295    // handle 0: shaft thickness on the head base; handle 1: head length.
296    let o = orient(d, w, h, &[(l - dx, t / 2.0), (l, t / 2.0)]);
297    let (o0, o1) = (Point::new(o[0].0, o[0].1), Point::new(o[1].0, o[1].1));
298    let across = match d {
299        Dir::Right | Dir::Left => Vec2::new(0.0, -t / 2.0 / 100000.0),
300        Dir::Up | Dir::Down => Vec2::new(-t / 2.0 / 100000.0, 0.0),
301    };
302    let along = (o0 - o1) * (ss / 100000.0 / dx.max(1e-9));
303    g.handles.push(Handle::new(0, o0, across, a1 * 100000.0, 0.0, 100000.0));
304    if dx > 1e-9 {
305        g.handles.push(Handle::new(1, o1, along, a2, 0.0, 100000.0 * l / ss.max(1e-9)));
306    }
307    g
308}

310pub(super) fn double_arrow(vertical: bool, w: f64, h: f64, a1: f64, a2: f64) -> Geometry {
311    let (l, t) = if vertical { (h, w) } else { (w, h) };
312    let ss = l.min(t);
313    let a1 = a1.clamp(0.0, 100000.0) / 100000.0;
314    let dx = (ss * a2.max(0.0) / 100000.0).min(l / 2.0);
315    let y1 = t / 2.0 - t * a1 / 2.0;
316    let y2 = t / 2.0 + t * a1 / 2.0;
317    let pts = [(0.0, t / 2.0), (dx, 0.0), (dx, y1), (l - dx, y1), (l - dx, 0.0), (l, t / 2.0), (l - dx, t), (l - dx, y2), (dx, y2), (dx, t)];
318    let pts = orient(if vertical { Dir::Down } else { Dir::Right }, w, h, &pts);
319    let r = if vertical { Rect::new(y1, dx, y2, h - dx) } else { Rect::new(dx, y1, w - dx, y2) };
320    Geometry { paths: vec![fill(Pb::new().poly(&pts).take())], text_rect: r, handles: vec![], sites: mid_sites(w, h) }
321}
