1//! Morph between two outlines: one shape turns into another (a circle into a star).
2//!
3//! The approach is manim's: both outlines get the same number of contours and each contour pair the
4//! same number of cubic curves, then every control point moves in a straight line. On top of that,
5//! closed contours are turned and reversed so that their points travel as short a way as possible
6//! (a triangle does not twist on its way into a circle), and extra points go to the longest curves.
7
8use kurbo::{BezPath, PathEl, Point, Rect, Vec2};
9
10use crate::geom::preset::{FillMode, Geometry, SubPath};
11
12/// A contour as cubic curves: the start anchor, then three points per curve (handle, handle,
13/// anchor).
14#[derive(Clone, Debug, PartialEq)]
15pub struct Contour {
16    pub closed: bool,
17    pub pts: Vec<Point>,
18}
19
20impl Contour {
21    pub fn curves(&self) -> usize {
22        self.pts.len().saturating_sub(1) / 3
23    }
24}
25
26fn line_to(p: &mut Vec<Point>, to: Point) {
27    let Some(&from) = p.last() else { return };
28    p.extend([from.lerp(to, 1.0 / 3.0), from.lerp(to, 2.0 / 3.0), to]);
29}
30
31fn close(c: &mut Contour) {
32    if let (Some(&first), Some(&last)) = (c.pts.first(), c.pts.last())
33        && first != last
34    {
35        line_to(&mut c.pts, first);
36    }
37    c.closed = true;
38}
39
40fn pen(cur: &mut Option<Contour>, start: Point) -> &mut Contour {
41    cur.get_or_insert_with(|| Contour { closed: false, pts: vec![start] })
42}
43
44fn keep(c: Contour, out: &mut Vec<Contour>) {
45    if c.curves() > 0 && c.pts.iter().all(|p| p.is_finite()) {
46        out.push(c);
47    }
48}

50/// The contours of a path, every segment as a cubic curve. Contours without a curve or with
51/// non-finite points are left out.
52pub fn contours(path: &BezPath) -> Vec<Contour> {
53    let mut out = Vec::new();
54    let mut cur: Option<Contour> = None;
55    let mut start = Point::ZERO;
56    for el in path.elements() {
57        match *el {
58            PathEl::MoveTo(p) => {
59                if let Some(c) = cur.take() {
60                    keep(c, &mut out);
61                }
62                start = p;
63                cur = Some(Contour { closed: false, pts: vec![p] });
64            }
65            PathEl::LineTo(p) => line_to(&mut pen(&mut cur, start).pts, p),
66            PathEl::QuadTo(q, p) => {
67                let c = pen(&mut cur, start);
68                if let Some(&a) = c.pts.last() {
69                    c.pts.extend([a.lerp(q, 2.0 / 3.0), p.lerp(q, 2.0 / 3.0), p]);
70                }
71            }
72            PathEl::CurveTo(a, b, p) => pen(&mut cur, start).pts.extend([a, b, p]),
73            PathEl::ClosePath => {
74                if let Some(mut c) = cur.take()
75                    && c.curves() > 0
76                {
77                    close(&mut c);
78                    keep(c, &mut out);
79                }
80            }
81        }
82    }
83    if let Some(c) = cur {
84        keep(c, &mut out);
85    }
86    out
87}

89/// Contours back to a path.
90pub fn to_path(cs: &[Contour]) -> BezPath {
91    let mut bp = BezPath::new();
92    for c in cs {
93        let Some(&s) = c.pts.first() else { continue };
94        bp.move_to(s);
95        for [a, b, p] in c.pts.get(1..).unwrap_or(&[]).as_chunks::<3>().0 {
96            bp.curve_to(*a, *b, *p);
97        }
98        if c.closed {
99            bp.close_path();
100        }
101    }
102    bp
103}

107/// Rough curve length: mean of chord and control polygon.
108fn curve_length(c: &[Point]) -> f64 {
109    let [p0, p1, p2, p3] = c else { return 0.0 };
110    (p0.distance(*p3) + p0.distance(*p1) + p1.distance(*p2) + p2.distance(*p3)) / 2.0
111}
112
113/// Appends curve `c` (four points) split into `n` pieces of equal parameter length (de Casteljau),
114/// without its start anchor.
115fn split_curve(c: &[Point], n: usize, out: &mut Vec<Point>) {
116    let [a, b, d, e] = c else { return };
117    let (mut p0, mut p1, mut p2, p3) = (*a, *b, *d, *e);
118    for j in (2..=n).rev() {
119        let t = 1.0 / j as f64;
120        let (ab, bc, cd) = (p0.lerp(p1, t), p1.lerp(p2, t), p2.lerp(p3, t));
121        let (abc, bcd) = (ab.lerp(bc, t), bc.lerp(cd, t));
122        let m = abc.lerp(bcd, t);
123        out.extend([ab, abc, m]);
124        (p0, p1, p2) = (m, bcd, cd);
125    }
126    out.extend([p1, p2, p3]);
127}

129/// A piece length ordered totally, so it can sit in a heap.
130#[derive(PartialEq)]
131struct Share(f64);
132impl Eq for Share {}
133impl PartialOrd for Share {
134    fn partial_cmp(&self, o: &Self) -> Option<std::cmp::Ordering> {
135        Some(self.cmp(o))
136    }
137}
138impl Ord for Share {
139    fn cmp(&self, o: &Self) -> std::cmp::Ordering {
140        self.0.total_cmp(&o.0)
141    }
142}

144/// Brings a contour to `n` curves. Unlike manim (which splits every curve equally often), the extra
145/// splits go to the longest pieces, so a long side gets more points than a short one.
146fn refine(c: &Contour, n: usize) -> Contour {
147    let have = c.curves();
148    if have >= n || have == 0 {
149        return c.clone();
150    }
151    let curves: Vec<&[Point]> = (0..have).filter_map(|k| c.pts.get(3 * k..3 * k + 4)).collect();
152    let len: Vec<f64> = curves.iter().map(|q| curve_length(q) + 1e-9).collect();
153    let mut parts = vec![1usize; curves.len()];
154    // The curve with the longest piece gets the next split (the lowest index on a tie); a heap keeps
155    // this O(n log n) instead of rescanning every curve for each split.
156    let mut heap: std::collections::BinaryHeap<(Share, std::cmp::Reverse<usize>)> =
157        len.iter().enumerate().map(|(k, l)| (Share(*l), std::cmp::Reverse(k))).collect();
158    for _ in 0..n - have {
159        let Some((_, std::cmp::Reverse(best))) = heap.pop() else { break };
160        if let Some(p) = parts.get_mut(best) {
161            *p += 1;
162            heap.push((Share(len.get(best).copied().unwrap_or(0.0) / *p as f64), std::cmp::Reverse(best)));
163        }
164    }
165    let mut out = Vec::with_capacity(1 + 3 * n);
166    out.extend(c.pts.first().copied());
167    for (q, p) in curves.iter().zip(&parts) {
168        split_curve(q, *p, &mut out);
169    }
170    Contour { closed: c.closed, pts: out }
171}

173const MIN_CURVES: usize = 12;
174const MAX_CURVES: usize = 60;
175/// Start points tried per closed contour; longer contours are sampled evenly.
176const MAX_TRIES: usize = 128;
177
178/// How many curves a contour pair gets: a common multiple of both counts while that stays small
179/// (then a triangle meets a circle at matching places and the morph stays symmetric), and at least
180/// [`MIN_CURVES`] for a smooth middle.
181fn curves_for(na: usize, nb: usize) -> usize {
182    fn gcd(x: usize, y: usize) -> usize {
183        if y == 0 { x } else { gcd(y, x % y) }
184    }
185    let hi = na.max(nb).max(1);
186    let mut n = if na > 0 && nb > 0 { (na / gcd(na, nb)).saturating_mul(nb) } else { hi };
187    if n > MAX_CURVES {
188        n = hi;
189    }
190    n.saturating_mul(MIN_CURVES.div_ceil(n).max(1))
191}

193/// Box around all points, handles included.
194fn bounds<'a>(pts: impl IntoIterator<Item = &'a Point>) -> Option<Rect> {
195    pts.into_iter().fold(None, |r: Option<Rect>, p| Some(r.map_or(Rect::from_points(*p, *p), |r| r.union_pt(*p))))
196}
197
198fn center(pts: &[Point]) -> Point {
199    bounds(pts).map(|r| r.center()).unwrap_or(Point::ZERO)
200}

202/// The same contour, starting at anchor `s`, optionally run backwards.
203fn turn(pts: &[Point], s: usize, back: bool) -> Vec<Point> {
204    let q: Vec<Point> = if back { pts.iter().rev().copied().collect() } else { pts.to_vec() };
205    if s == 0 {
206        return q;
207    }
208    let body = q.get(1..).unwrap_or(&[]);
209    let cut = (3 * s).min(body.len());
210    let mut out = Vec::with_capacity(q.len());
211    out.extend(cut.checked_sub(1).and_then(|i| body.get(i)).copied());
212    out.extend_from_slice(body.get(cut..).unwrap_or(&[]));
213    out.extend_from_slice(body.get(..cut).unwrap_or(&[]));
214    out
215}

217/// One contour pair of equal curve count: `b` runs backwards if `back` is set, then the closed side
218/// (`b` if it is closed, else `a`) gets the start that lets the points travel least. Both sides are
219/// compared around their own centre, so where the shapes are does not matter.
220fn match_pair(a: &Contour, b: &Contour, back: bool) -> (Contour, Contour, f64) {
221    let bp = if back { turn(&b.pts, 0, true) } else { b.pts.clone() };
222    let turn_a = !b.closed && a.closed;
223    let (fixed, free) = if turn_a { (&bp, &a.pts) } else { (&a.pts, &bp) };
224    let (fc, gc) = (center(fixed), center(free));
225    let tries = if turn_a || b.closed { free.len().saturating_sub(1) / 3 } else { 1 };
226    let mut best = free.clone();
227    let mut best_cost = f64::INFINITY;
228    for s in (0..tries.max(1)).step_by(tries.div_ceil(MAX_TRIES).max(1)) {
229        let q = turn(free, s, false);
230        let mut cost = 0.0;
231        for (p, f) in q.iter().zip(fixed.iter()) {
232            cost += ((*p - gc) - (*f - fc)).hypot2();
233            if cost >= best_cost {
234                break;
235            }
236        }
237        if cost < best_cost {
238            best_cost = cost;
239            best = q;
240        }
241    }
242    let (pa, pb) = if turn_a { (best, bp) } else { (a.pts.clone(), best) };
243    (Contour { closed: a.closed, pts: pa }, Contour { closed: b.closed, pts: pb }, best_cost)
244}

246/// Where a contour without partner shrinks to (or grows from): the spot at the same place relative
247/// to the other outline's box, or its own centre without one.
248fn place(c: &Contour, from: Option<Rect>, to: Option<Rect>) -> Point {
249    let p = center(&c.pts);
250    let (Some(from), Some(to)) = (from, to) else { return p };
251    let fx = if from.width() > 0.0 { (p.x - from.x0) / from.width() } else { 0.5 };
252    let fy = if from.height() > 0.0 { (p.y - from.y0) / from.height() } else { 0.5 };
253    Point::new(to.x0 + to.width() * fx, to.y0 + to.height() * fy)
254}

256/// Two contour lists brought to the same number of contours and the same number of curves per
257/// contour pair. Contours pair up in order; one without partner meets a point (see [`place`];
258/// `ba`/`bb` are the boxes of the whole outlines). Whether `b` runs backwards is decided once for all
259/// its contours: turning single ones around would change which areas a nonzero fill covers.
260pub fn align(a: &[Contour], b: &[Contour], ba: Option<Rect>, bb: Option<Rect>) -> (Vec<Contour>, Vec<Contour>) {
261    // (a, b, either side a point).
262    let mut pairs: Vec<(Contour, Contour, bool)> = Vec::new();
263    for k in 0..a.len().max(b.len()) {
264        let (ca, cb) = (a.get(k), b.get(k));
265        let n = curves_for(ca.map_or(0, Contour::curves), cb.map_or(0, Contour::curves));
266        let point = |c: &Contour, from, to, closed| Contour { closed, pts: vec![place(c, from, to); 1 + 3 * n] };
267        match (ca, cb) {
268            (Some(x), Some(y)) => pairs.push((refine(x, n), refine(y, n), false)),
269            (Some(x), None) => pairs.push((refine(x, n), point(x, ba, bb, x.closed), true)),
270            (None, Some(y)) => pairs.push((point(y, bb, ba, y.closed), refine(y, n), true)),
271            (None, None) => {}
272        }
273    }
274    let mut best: Option<(Vec<Contour>, Vec<Contour>, f64)> = None;
275    for back in [false, true] {
276        let (mut xs, mut ys, mut cost) = (Vec::new(), Vec::new(), 0.0);
277        for (x, y, point) in &pairs {
278            if *point {
279                xs.push(x.clone());
280                ys.push(if back { Contour { closed: y.closed, pts: turn(&y.pts, 0, true) } } else { y.clone() });
281            } else {
282                let (mx, my, c) = match_pair(x, y, back);
283                xs.push(mx);
284                ys.push(my);
285                cost += c;
286            }
287        }
288        if best.as_ref().is_none_or(|b| cost < b.2) {
289            best = Some((xs, ys, cost));
290        }
291    }
292    best.map(|(x, y, _)| (x, y)).unwrap_or_default()
293}

312/// Two shape geometries brought to the same structure once, then blended per frame with
313/// [`GeometryMorph::at`]. Sub-paths pair up in order; one without partner grows out of (or shrinks
314/// into) a point. When a pair differs in fill mode, outline or fill rule, the two looks cross-fade
315/// (see [`Fade`]).
316#[derive(Clone, Debug)]
317pub struct GeometryMorph {
318    parts: Vec<Part>,
319    text: (Rect, Rect),
320}
321
322fn look(s: &SubPath) -> Look {
323    Look { fill: s.fill, stroke: s.stroke, even_odd: s.even_odd }
324}

326/// How strongly a sub-path of a morphed geometry shows: its fill and its outline, 0..1.
327#[derive(Clone, Copy, Debug, PartialEq)]
328pub struct Fade {
329    pub fill: f64,
330    pub stroke: f64,
331}
332
333/// The layers a pair of looks is drawn with at `t`, bottom first. A fill or an outline only one side
334/// has fades out (or in). When both fill, the side with the even-odd rule lies opaque underneath and
335/// the other fades over it, so a donut's hole fills smoothly; with the same rule both cover the same
336/// area, so this is a plain cross-fade (shading of 3-D faces).
337fn layers(a: Option<Look>, b: Option<Look>, t: f64) -> Vec<(Look, Fade)> {
338    let (a, b) = match (a, b) {
339        (Some(a), Some(b)) if a != b => (a, b),
340        (Some(x), _) | (None, Some(x)) => return vec![(x, Fade { fill: 1.0, stroke: 1.0 })],
341        (None, None) => return vec![],
342    };
343    let side = |on_a: bool, on_b: bool| match (on_a, on_b) {
344        (true, true) => 1.0,
345        (true, false) => 1.0 - t,
346        (false, true) => t,
347        (false, false) => 0.0,
348    };
349    let (stroke, line) = (a.stroke || b.stroke, side(a.stroke, b.stroke));
350    let (fa, fb) = (a.fill != FillMode::None, b.fill != FillMode::None);
351    if fa && fb {
352        let (under, over, alpha) = if a.even_odd || !b.even_odd { (a, b, t) } else { (b, a, 1.0 - t) };
353        vec![(Look { stroke, ..under }, Fade { fill: 1.0, stroke: line }), (Look { stroke: false, ..over }, Fade { fill: alpha, stroke: 0.0 })]
354    } else {
355        let filled = if fa { a } else { b };
356        vec![(Look { stroke, ..filled }, Fade { fill: side(fa, fb), stroke: line })]
357    }
358}

360impl GeometryMorph {
361    /// `a` and `b` each in their own shape-local space.
362    pub fn new(a: &Geometry, b: &Geometry) -> Self {
363        let ca: Vec<Vec<Contour>> = a.paths.iter().map(|s| contours(&s.path)).collect();
364        let cb: Vec<Vec<Contour>> = b.paths.iter().map(|s| contours(&s.path)).collect();
365        let ba = bounds(ca.iter().flatten().flat_map(|c| &c.pts));
366        let bb = bounds(cb.iter().flatten().flat_map(|c| &c.pts));
367        let mut parts = Vec::new();
368        for k in 0..a.paths.len().max(b.paths.len()) {
369            let (xa, xb) = (ca.get(k).map(Vec::as_slice).unwrap_or(&[]), cb.get(k).map(Vec::as_slice).unwrap_or(&[]));
370            let (pa, pb) = align(xa, xb, ba, bb);
371            parts.push(Part { a: a.paths.get(k).map(look), b: b.paths.get(k).map(look), ca: pa, cb: pb });
372        }
373        GeometryMorph { parts, text: (a.text_rect, b.text_rect) }
374    }
375
376    /// The geometry at `t` (0..1), with a [`Fade`] for each of its sub-paths. The points of `a` are
377    /// first scaled by `sa` and those of `b` by `sb`, to bring both into the current box.
378    pub fn at(&self, t: f64, sa: Vec2, sb: Vec2) -> (Geometry, Vec<Fade>) {
379        let t = if t.is_finite() { t.clamp(0.0, 1.0) } else { 0.0 };
380        let scale = |p: Point, s: Vec2| Point::new(p.x * s.x, p.y * s.y);
381        let (mut paths, mut fades) = (Vec::new(), Vec::new());
382        for part in &self.parts {
383            let cs: Vec<Contour> = part
384                .ca
385                .iter()
386                .zip(&part.cb)
387                .map(|(x, y)| Contour {
388                    closed: x.closed && y.closed,
389                    pts: x.pts.iter().zip(&y.pts).map(|(p, q)| scale(*p, sa).lerp(scale(*q, sb), t)).collect(),
390                })
391                .collect();
392            let path = to_path(&cs);
393            for (lk, fade) in layers(part.a, part.b, t) {
394                paths.push(SubPath { path: path.clone(), fill: lk.fill, stroke: lk.stroke, even_odd: lk.even_odd });
395                fades.push(fade);
396            }
397        }
398        let r = |r: Rect, s: Vec2| Rect::new(r.x0 * s.x, r.y0 * s.y, r.x1 * s.x, r.y1 * s.y);
399        let (ta, tb) = (r(self.text.0, sa), r(self.text.1, sb));
400        let text_rect = Rect::from_points(ta.origin().lerp(tb.origin(), t), Point::new(ta.x1, ta.y1).lerp(Point::new(tb.x1, tb.y1), t));
401        (Geometry { paths, text_rect, handles: vec![], sites: vec![] }, fades)
402    }
403}

497    fn extra_sub_paths_grow_out_of_a_point() {
498        let (a, b) = (build("rect", 100.0, 100.0, &[]).unwrap(), build("cube", 100.0, 100.0, &[]).unwrap());
499        assert!(b.paths.len() > 1);

503        assert_eq!(start.paths.len(), b.paths.len());
504        for sp in start.paths.iter().skip(1) {

507        }

511        let back = GeometryMorph::new(&b, &a).at(1.0, one, one).0;
512        assert!(back.paths.iter().skip(1).all(|sp| sp.path.bounding_box().area() < 1e-9));

515    #[test]
516    fn differing_looks_cross_fade() {
517        let one = Vec2::new(1.0, 1.0);
518        let fades = |a: &Geometry, b: &Geometry, t: f64| {
519            let (g, f) = GeometryMorph::new(a, b).at(t, one, one);
520            g.paths.iter().zip(f).map(|(s, f)| (s.fill, s.even_odd, s.stroke, f.fill, f.stroke)).collect::<Vec<_>>()
521        };
522        // A filled rectangle into a line: the fill fades out, the outline stays.
523        let (r, line) = (build("rect", 100.0, 100.0, &[]).unwrap(), build("line", 100.0, 0.0, &[]).unwrap());
524        assert_eq!(fades(&r, &line, 0.25), [(FillMode::Norm, false, true, 0.75, 1.0)]);
525        assert_eq!(fades(&line, &r, 0.25), [(FillMode::Norm, false, true, 0.25, 1.0)]);
526        // A donut into a circle: the donut (even-odd) lies underneath, the circle fades in over its hole.
527        let (d, e) = (build("donut", 100.0, 100.0, &[]).unwrap(), build("ellipse", 100.0, 100.0, &[]).unwrap());
528        assert!(d.paths[0].even_odd && !e.paths[0].even_odd);
529        assert_eq!(fades(&d, &e, 0.25), [(FillMode::Norm, true, true, 1.0, 1.0), (FillMode::Norm, false, false, 0.25, 0.0)]);
530        assert_eq!(fades(&e, &d, 0.25), [(FillMode::Norm, true, true, 1.0, 1.0), (FillMode::Norm, false, false, 0.75, 0.0)]);
531        // Same looks: one layer, fully there.
532        assert_eq!(fades(&e, &r, 0.5), [(FillMode::Norm, false, true, 1.0, 1.0)]);
533    }

547    #[test]
548    fn empty_and_degenerate_geometries_do_not_crash() {
549        let empty = Geometry { paths: vec![], text_rect: Rect::ZERO, handles: vec![], sites: vec![] };
550        let r = build("rect", 10.0, 10.0, &[]).unwrap();
551        let line = build("line", 10.0, 0.0, &[]).unwrap();
552        let mut nan = BezPath::new();
553        nan.move_to((f64::NAN, 0.0));
554        nan.line_to((1.0, 1.0));
555        let bad = Geometry { paths: vec![SubPath { path: nan, fill: FillMode::Norm, stroke: true, even_odd: false }], ..empty.clone() };
556        for (x, y) in [(&empty, &empty), (&empty, &r), (&r, &empty), (&r, &line), (&line, &r), (&bad, &r), (&r, &bad)] {
557            for t in [0.0, 0.5, 1.0, f64::INFINITY] {
558                let g = GeometryMorph::new(x, y).at(t, Vec2::new(1.0, 1.0), Vec2::new(0.0, 1e300)).0;
559                // At most two layers per sub-path pair.
560                assert!(g.paths.len() <= 2 * x.paths.len().max(y.paths.len()));
561            }
562        }
563        // A rectangle into a line: the line is open, so the morph ends open.
564        let g = GeometryMorph::new(&r, &line).at(1.0, Vec2::new(1.0, 1.0), Vec2::new(1.0, 1.0)).0;
565        assert!(!g.paths[0].path.elements().iter().any(|e| matches!(e, PathEl::ClosePath)));
566    }

573            let a = k as f64 / 2000.0 * std::f64::consts::TAU;

576        }
577        p.close_path();
578        let many = contours(&p);

580        let (x, y) = align(&many, &e, None, None);
581        assert_eq!(x[0].curves(), y[0].curves());
582        assert_eq!(x[0].curves(), 2000);
