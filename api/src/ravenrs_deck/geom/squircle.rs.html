1//! Rectangles with smoothed ("squircle") corners, as design tools offer next to the corner radius
2//! (#286). A plain rounded corner joins the straight edge with a sudden change of curvature; a
3//! smoothed corner starts curving earlier, along a cubic Bézier, and keeps a shorter circular arc
4//! in the middle, so the outline flows into the edge.
5//!
6//! For a corner of radius `r` and smoothing `s` (0 = circular, 1 = no arc left), the corner takes
7//! `p = (1 + s) · r` of each edge. The arc keeps `90° · (1 − s)` around the corner's diagonal and
8//! each side of it is a cubic whose handles lie on the edge, so the curve is tangent to the edge
9//! where it starts and to the arc where it meets it. When `p` would exceed half the shorter side,
10//! the smoothing is reduced first, then the radius. The construction is our own derivation from
11//! these constraints.
12
13/// The outline of a `w` x `h` rectangle with corners of radius `radius` smoothed by `smoothing`
14/// (0..=1), as path data in the rectangle's own coordinates (`M`, `L`, `C`, `Z`; y down).
15pub fn path_d(w: f64, h: f64, radius: f64, smoothing: f64) -> String {
16    let w = if w.is_finite() { w.max(0.0) } else { 0.0 };
17    let h = if h.is_finite() { h.max(0.0) } else { 0.0 };
18    let half = w.min(h) / 2.0;
19    let r = if radius.is_finite() { radius.clamp(0.0, half) } else { 0.0 };
20    let mut s = if smoothing.is_finite() { smoothing.clamp(0.0, 1.0) } else { 0.0 };
21    if r <= 0.0 {
22        return format!("M 0 0 L {w} 0 L {w} {h} L 0 {h} Z");
23    }
24    s = s.min(half / r - 1.0).max(0.0);
25    let c = corner(r, s);
26    // The top-right corner in local terms: `u` runs back along the top edge from the corner,
27    // `v` down the right edge. Each corner maps (u, v) to the rectangle by symmetry.
28    let corners: [&dyn Fn(f64, f64) -> (f64, f64); 4] = [
29        &|u, v| (w - u, v),     // top right: from the top edge to the right edge
30        &|u, v| (w - v, h - u), // bottom right: from the right edge to the bottom edge
31        &|u, v| (u, h - v),     // bottom left: from the bottom edge to the left edge
32        &|u, v| (v, u),         // top left: from the left edge to the top edge
33    ];
34    let mut d = String::new();
35    let pt = |f: &dyn Fn(f64, f64) -> (f64, f64), (u, v): (f64, f64)| {
36        let (x, y) = f(u, v);
37        format!("{} {}", round(x), round(y))
38    };
39    for (i, f) in corners.iter().enumerate() {
40        let segs = c.segments();
41        if i == 0 {
42            d.push_str(&format!("M {}", pt(*f, segs[0])));
43        } else {
44            d.push_str(&format!(" L {}", pt(*f, segs[0])));
45        }
46        for k in 0..3 {
47            let b = 1 + k * 3;
48            d.push_str(&format!(" C {} {} {}", pt(*f, segs[b]), pt(*f, segs[b + 1]), pt(*f, segs[b + 2])));
49        }
50    }
51    d.push_str(" Z");
52    d
53}
54
55fn round(v: f64) -> f64 {
56    (v * 1000.0).round() / 1000.0 + 0.0
57}
58
59/// One corner in (u, v) terms: distance back along the incoming edge, distance along the outgoing.
60struct Corner {
61    p: f64,
62    a: f64,
63    b: f64,
64    c: f64,
65    d: f64,
66    r: f64,
67    arc: f64,
68}
69
70fn corner(r: f64, s: f64) -> Corner {
71    use std::f64::consts::{FRAC_PI_2, FRAC_PI_4};
72    let p = (1.0 + s) * r;
73    let arc = FRAC_PI_2 * (1.0 - s);
74    // The arc's chord, and where the edge curve meets the arc.
75    let arc_chord = (arc / 2.0).sin() * r * std::f64::consts::SQRT_2;
76    let alpha = (FRAC_PI_2 - arc) / 2.0;
77    let p3_p4 = r * (alpha / 2.0).tan();
78    let beta = FRAC_PI_4 * s;
79    let c = p3_p4 * beta.cos();
80    let d = c * beta.tan();
81    let b = ((p - arc_chord - c - d) / 3.0).max(0.0);
82    let a = 2.0 * b;
83    Corner { p, a, b, c, d, r, arc }
84}
85
86impl Corner {
87    /// Ten points: the start on the incoming edge, then three cubics (control, control, end):
88    /// edge → arc start, the arc, arc end → outgoing edge.
89    fn segments(&self) -> [(f64, f64); 10] {
90        let Corner { p, a, b, c, d, r, arc } = *self;
91        // Edge curve into the arc: handles on the edge, ending where the arc starts.
92        let e0 = (p, 0.0);
93        let e1 = (p - a, 0.0);
94        let e2 = (p - a - b, 0.0);
95        let e3 = (p - a - b - c, d);
96        // The arc around the centre (r, r) in (u, v), symmetric about the corner's diagonal.
97        // Angles from +u towards +v: the corner itself is at 5π/4 from the centre; the arc runs
98        // from the incoming edge's side (larger angle) to the outgoing edge's side.
99        let start = std::f64::consts::FRAC_PI_4 * 5.0 + arc / 2.0;
100        let end = start - arc;
101        let at = |t: f64| (r + r * t.cos(), r + r * t.sin());
102        let k = 4.0 / 3.0 * (arc / 4.0).tan() * r;
103        let a0 = at(start);
104        let a3 = at(end);
105        // Tangent of the arc travelled towards smaller angles: (sin t, −cos t).
106        let a1 = (a0.0 + k * start.sin(), a0.1 - k * start.cos());
107        let a2 = (a3.0 - k * end.sin(), a3.1 + k * end.cos());
108        // Mirror of the edge curve onto the outgoing edge (swap u and v), reversed.
109        let o1 = (e2.1, e2.0);
110        let o2 = (e1.1, e1.0);
111        let o3 = (e0.1, e0.0);
112        let _ = (e3, a0);
113        [e0, e1, e2, a0, a1, a2, a3, o1, o2, o3]
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    fn numbers(d: &str) -> Vec<f64> {
122        d.split_whitespace().filter_map(|t| t.parse().ok()).collect()
123    }
124
125    #[test]
126    fn no_radius_is_a_rectangle() {
127        assert_eq!(path_d(100.0, 50.0, 0.0, 0.6), "M 0 0 L 100 0 L 100 50 L 0 50 Z");
128    }
129
130    #[test]
131    fn edge_curve_meets_the_arc() {
132        for s in [0.0, 0.3, 0.6, 1.0] {
133            let c = corner(10.0, s);
134            let seg = c.segments();
135            let e3 = (c.p - c.a - c.b - c.c, c.d);
136            // The arc starts on the circle and where the edge curve ends.
137            assert!((seg[3].0 - e3.0).abs() < 1e-6 && (seg[3].1 - e3.1).abs() < 1e-6, "s={s}: {:?} vs {:?}", seg[3], e3);
138            // The arc is symmetric about the diagonal u = v.
139            assert!((seg[3].0 - seg[6].1).abs() < 1e-9 && (seg[3].1 - seg[6].0).abs() < 1e-9);
140        }
141    }
142
143    #[test]
144    fn zero_smoothing_is_a_circular_corner() {
145        let c = corner(10.0, 0.0);
146        let seg = c.segments();
147        assert!((c.p - 10.0).abs() < 1e-9);
148        assert!((seg[3].0 - 10.0).abs() < 1e-9 && seg[3].1.abs() < 1e-9);
149        assert!((seg[6].0).abs() < 1e-9 && (seg[6].1 - 10.0).abs() < 1e-9);
150    }
151
152    #[test]
153    fn smoothing_uses_more_of_the_edge() {
154        let d0 = numbers(&path_d(200.0, 100.0, 20.0, 0.0));
155        let d6 = numbers(&path_d(200.0, 100.0, 20.0, 0.6));
156        // The first point is where the top-right corner starts on the top edge.
157        assert_eq!((d0[0], d0[1]), (180.0, 0.0));
158        assert_eq!((d6[0], d6[1]), (168.0, 0.0));
159    }
160
161    #[test]
162    fn stays_inside_and_reduces_smoothing_when_the_rectangle_is_small() {
163        for (w, h, r, s) in [(100.0, 40.0, 20.0, 1.0), (40.0, 40.0, 50.0, 0.8), (10.0, 300.0, 4.0, 0.6), (300.0, 120.0, 30.0, 0.6)] {
164            let n = numbers(&path_d(w, h, r, s));
165            assert!(n.chunks(2).all(|p| p[0] >= -1e-6 && p[0] <= w + 1e-6 && p[1] >= -1e-6 && p[1] <= h + 1e-6), "{w}x{h} r{r} s{s}");
166            // Corners never cross the middle of an edge.
167            assert!(n[0] >= w / 2.0 - 1e-6);
168        }
169        assert!(numbers(&path_d(f64::NAN, 10.0, f64::INFINITY, f64::NAN)).iter().all(|v| v.is_finite()));
170    }
171
172    #[test]
173    fn four_symmetric_corners() {
174        let n = numbers(&path_d(120.0, 80.0, 16.0, 0.6));
175        // 4 corners x 10 points, 2 numbers each.
176        assert_eq!(n.len(), 80);
177        let pts: Vec<(f64, f64)> = n.chunks(2).map(|p| (p[0], p[1])).collect();
178        for &(x, y) in &pts {
179            let mirrored = (120.0 - x, 80.0 - y);
180            assert!(pts.iter().any(|q| (q.0 - mirrored.0).abs() < 0.01 && (q.1 - mirrored.1).abs() < 0.01), "{x},{y}");
181        }
182    }
