1//! WordArt text warps (DrawingML `a:bodyPr/a:prstTxWarp`).
2//!
3//! A warp preset is a list of bands laid over the shape's text rectangle (insets are ignored).
4//! Text lines are shared out over the bands in order (the first ⌈n/k⌉ lines to the first band
5//! and so on). A band either
6//! - stretches its lines between a top and a bottom guide ([`Band::Fill`]): the ink box of the
7//!   band's lines fills the band vertically (a band whose ink is shorter than the tallest band's
8//!   keeps the same scale, centred), and horizontally the text is scaled so that the
9//!   widest band is filled along the mean length of its guides (one scale for the whole text), or
10//! - sets its lines at their natural size along one guide ([`Band::Follow`], arches and circles):
11//!   glyphs keep their size, are placed by arc length (aligned as in the paragraph, the whole
12//!   guide being the line's width: insets don't count) and stand on
13//!   the guide's left normal; lines further from the guide stack outwards. Lines that are longer
14//!   than the guide are shrunk to fit.
15//!
16//! Positions along a guide are by arc length; a fill band maps the same fraction of its top and
17//! bottom guide's length and interpolates between them. The guide curves were derived from
18//! observation: synthetic decks of `I` bars were drawn with PowerPoint and every curve fitted to
19//! the measured bar ends (most fit within a point on a 420 × 200 pt box).
20use kurbo::{Arc, BezPath, PathEl, Point, Rect, Vec2};
21
22/// Flattening tolerance for guides, in points.
23const TOL: f64 = 0.02;
24
25/// A guide curve flattened to a polyline with cumulative arc lengths.
26#[derive(Clone, Debug, Default)]
27pub struct Guide {
28    pts: Vec<Point>,
29    cum: Vec<f64>,
30    closed: bool,
31}
32
33impl Guide {
34    fn from_points(raw: impl IntoIterator<Item = Point>, closed: bool) -> Guide {
35        let mut pts: Vec<Point> = Vec::new();
36        let mut cum: Vec<f64> = Vec::new();
37        for p in raw {
38            if !(p.x.is_finite() && p.y.is_finite()) {
39                continue;
40            }
41            match pts.last() {
42                Some(q) => {
43                    let d = (p - *q).hypot();
44                    if d > 1e-9 {
45                        cum.push(cum.last().copied().unwrap_or(0.0) + d);
46                        pts.push(p);
47                    }
48                }
49                None => {
50                    pts.push(p);
51                    cum.push(0.0);
52                }
53            }
54            if pts.len() > 100_000 {
55                break;
56            }
57        }
58        Guide { pts, cum, closed }
59    }
60
61    /// A guide through `path` (any curves), flattened.
62    pub fn from_path(path: &BezPath, closed: bool) -> Guide {
63        let mut pts = Vec::new();
64        kurbo::flatten(path.iter(), TOL, |el| match el {
65            PathEl::MoveTo(p) | PathEl::LineTo(p) => pts.push(p),
66            _ => {}
67        });
68        Guide::from_points(pts, closed)
69    }
70
71    /// Total arc length.
72    pub fn length(&self) -> f64 {
73        self.cum.last().copied().unwrap_or(0.0)
74    }
75
76    /// Whether the guide is a closed loop (positions past its end wrap around).
77    pub fn is_closed(&self) -> bool {
78        self.closed
79    }
80
81    /// Point and unit tangent at arc length `s`. Open guides continue along their end tangents
82    /// past either end; closed guides wrap.
83    pub fn at(&self, s: f64) -> (Point, Vec2) {
84        let len = self.length();
85        let (Some(&first), Some(&last)) = (self.pts.first(), self.pts.last()) else {
86            return (Point::ZERO, Vec2::new(1.0, 0.0));
87        };
88        if self.pts.len() < 2 || len <= 1e-9 {
89            return (first, Vec2::new(1.0, 0.0));
90        }
91        let s = if s.is_finite() { s } else { 0.0 };
92        let s = if self.closed { s.rem_euclid(len) } else { s };
93        let seg = |i: usize| -> (Point, Point, f64) {
94            let a = self.pts.get(i).copied().unwrap_or(first);
95            let b = self.pts.get(i + 1).copied().unwrap_or(last);
96            (a, b, self.cum.get(i).copied().unwrap_or(0.0))
97        };
98        let n = self.pts.len();
99        let i = if s <= 0.0 {
100            0
101        } else if s >= len {
102            n - 2
103        } else {
104            self.cum.partition_point(|c| *c <= s).saturating_sub(1).min(n - 2)
105        };
106        let (a, b, c0) = seg(i);
107        let d = b - a;
108        let l = d.hypot();
109        let t = if l > 1e-12 { d / l } else { Vec2::new(1.0, 0.0) };
110        (a + t * (s - c0), t)
111    }
112
113    /// The parallel curve `d` points away on the left of the direction of travel (above a
114    /// guide running left to right).
115    pub fn offset(&self, d: f64) -> Guide {
116        let n = self.pts.len();
117        if n < 2 || !d.is_finite() || d == 0.0 {
118            return self.clone();
119        }
120        let unit = |i: usize| -> Vec2 {
121            let (a, b) = (self.pts.get(i).copied().unwrap_or(Point::ZERO), self.pts.get(i + 1).copied().unwrap_or(Point::ZERO));
122            let v = b - a;
123            let l = v.hypot();
124            if l > 1e-12 { v / l } else { Vec2::ZERO }
125        };
126        let pts = (0..n).map(|i| {
127            let before = if i > 0 {
128                unit(i - 1)
129            } else if self.closed {
130                unit(n - 2)
131            } else {
132                Vec2::ZERO
133            };
134            let after = if i + 1 < n {
135                unit(i)
136            } else if self.closed {
137                unit(0)
138            } else {
139                Vec2::ZERO
140            };
141            let t = before + after;
142            let l = t.hypot();
143            let t = if l > 1e-12 { t / l } else { Vec2::new(1.0, 0.0) };
144            self.pts.get(i).copied().unwrap_or(Point::ZERO) + Vec2::new(t.y, -t.x) * d
145        });
146        Guide::from_points(pts.collect::<Vec<_>>(), self.closed)
147    }
148
149    /// The point at fraction `t` of the length.
150    pub fn at_frac(&self, t: f64) -> Point {
151        self.at(t * self.length()).0
152    }
153}

157pub enum LineAnchor {
158    /// The last line; earlier lines stack outwards (arch up, circle).

166#[derive(Clone, Debug)]
167pub enum Band {
168    /// Text stretched between two guides.
169    Fill { top: Guide, bottom: Guide },
170    /// Text at its natural size along a guide.
171    Follow { path: Guide, anchor: LineAnchor },
172}
173
174impl Band {
175    /// The length text is measured against: the mean of a fill band's guides.
176    pub fn length(&self) -> f64 {
177        match self {
178            Band::Fill { top, bottom } => (top.length() + bottom.length()) / 2.0,
179            Band::Follow { path, .. } => path.length(),
180        }
181    }
182}

289/// A yellow adjust handle for one warp guide value. `pos` is in the same space as `rect`
290/// (shape-local points). Dragging projects the pointer onto the handle's axis, or around the
291/// text rectangle's centre for an angle.
292#[derive(Clone, Copy, Debug, PartialEq)]
293pub struct WarpHandle {
294    pub index: usize,
295    pub pos: Point,
296    kind: HandleKind,
297}
298
299#[derive(Clone, Copy, Debug, PartialEq)]
300enum HandleKind {
301    /// `pos = origin + dir * raw`. `value_at` projects onto that line and clamps.
302    Line { origin: Point, dir: Vec2, min: f64, max: f64 },
303    /// Angle about `center` (y down, 0° at +x). `invert` is arch-down, whose guide angle is `180° − adj`.
304    Angle { center: Point, invert: bool },
305}
306
307impl WarpHandle {
308    /// The raw guide value (`adj` / `adjN`) for a pointer at `p`.
309    pub fn value_at(&self, p: Point) -> f64 {
310        match self.kind {
311            HandleKind::Line { origin, dir, min, max } => {
312                let l2 = dir.hypot2();
313                if l2 < 1e-18 || !(p.x.is_finite() && p.y.is_finite()) {
314                    return min;
315                }
316                let v = (p - origin).dot(dir) / l2;
317                if !v.is_finite() {
318                    return min;
319                }
320                let (lo, hi) = (min.min(max), max.max(min));
321                v.clamp(lo, hi).round()
322            }
323            HandleKind::Angle { center, invert } => {
324                let d = p - center;
325                if !(d.x.is_finite() && d.y.is_finite()) || d.hypot() < 1e-9 {
326                    return 0.0;
327                }
328                let mut deg = d.y.atan2(d.x).to_degrees();
329                if invert {
330                    deg = 180.0 - deg;
331                }
332                (deg.rem_euclid(360.0) * 60_000.0).round()
333            }
334        }
335    }
336}

338fn stored_raw(adj: &[f64], defaults: &[f64], i: usize) -> f64 {
339    adj.get(i).copied().filter(|v| v.is_finite()).or_else(|| defaults.get(i).copied()).unwrap_or(0.0)
340}

342fn unit(rect: Rect, u: f64, v: f64) -> Point {
343    Point::new(rect.x0 + u * rect.width(), rect.y0 + v * rect.height())
344}

346/// A fraction handle: unit position `(x, sign * frac)`, raw = frac × 100 000, clamped to `lo..=hi`.
347fn frac_handle(rect: Rect, index: usize, x: f64, frac: f64, lo: f64, hi: f64, sign: f64) -> WarpHandle {
348    let frac = frac.clamp(lo, hi);
349    WarpHandle {
350        index,
351        pos: unit(rect, x, sign * frac),
352        kind: HandleKind::Line {
353            origin: unit(rect, x, 0.0),
354            dir: Vec2::new(0.0, sign * rect.height() / 100_000.0),
355            min: lo * 100_000.0,
356            max: hi * 100_000.0,
357        },
358    }
359}

376/// Angle handle, inset from the ellipse so it doesn't sit on a resize handle.
377fn angle_handle(rect: Rect, index: usize, deg: f64, invert: bool) -> WarpHandle {
378    let geom = if invert { 180.0 - deg } else { deg };
379    let rad = geom.to_radians();
380    let c = rect.center();
381    WarpHandle {
382        index,
383        pos: Point::new(c.x + rect.width() * 0.36 * rad.cos(), c.y + rect.height() * 0.36 * rad.sin()),
384        kind: HandleKind::Angle { center: c, invert },
385    }
386}

388/// Inner-ellipse radius (pour presets): the handle sits on the downward point of that ellipse.
389fn radius_handle(rect: Rect, r: f64) -> WarpHandle {
390    let r = r.clamp(0.0, 0.99);
391    WarpHandle {
392        index: 1,
393        pos: unit(rect, 0.5, 0.5 + 0.5 * r),
394        kind: HandleKind::Line { origin: unit(rect, 0.5, 0.5), dir: Vec2::new(0.0, 0.5 * rect.height() / 100_000.0), min: 0.0, max: 99_000.0 },
395    }
396}

408/// Yellow handles for warp `name` over text rectangle `rect`. Empty when `name` isn't a drawn
409/// preset or `rect` isn't a usable box. Missing adjust values take the preset defaults.
410pub fn handles(name: &str, adj: &[f64], rect: Rect) -> Vec<WarpHandle> {
411    if ![rect.x0, rect.y0, rect.x1, rect.y1].iter().all(|v| v.is_finite()) {
412        return Vec::new();
413    }
414    let rect = Rect::new(rect.x0.min(rect.x1), rect.y0.min(rect.y1), rect.x0.max(rect.x1), rect.y0.max(rect.y1));
415    if rect.width() < 1e-3 || rect.height() < 1e-3 {
416        return Vec::new();
417    }
418    let defaults = adjust_defaults(name);
419    if defaults.is_empty() {
420        return Vec::new();
421    }
422    let raw = |i| stored_raw(adj, defaults, i);
423    let frac = |i| raw(i) / 100_000.0;
424    let deg = |i| (raw(i) / 60_000.0).rem_euclid(360.0);
425    let y = |out: &mut Vec<WarpHandle>, x, lo, hi, sign| out.push(frac_handle(rect, 0, x, frac(0), lo, hi, sign));
426    let xh = |out: &mut Vec<WarpHandle>, yy, lo, hi| out.push(frac_x_handle(rect, 0, yy, frac(0), lo, hi));
427    let mut out = Vec::with_capacity(defaults.len());
428    match name {

436        "textFadeUp" => xh(&mut out, 0.22, 0.0, 0.49999),
437        "textFadeDown" => xh(&mut out, 0.78, 0.0, 0.49999),
438        "textSlantUp" => y(&mut out, 0.18, 0.44444, 1.0, 1.0),
439        "textSlantDown" => y(&mut out, 0.18, 0.0, 0.55556, 1.0),
440        "textCascadeUp" => y(&mut out, 0.82, 0.0, 1.0, 1.0),
441        "textCascadeDown" => y(&mut out, 0.18, 0.0, 1.0, 1.0),
442        "textInflate" => y(&mut out, 0.62, 0.0, 0.2, -1.0),
443        "textDeflate" => y(&mut out, 0.62, 0.0, 0.375, 1.0),
444        "textInflateBottom" => y(&mut out, 0.62, 0.6, 1.0, 1.0),

498impl Adj<'_> {
499    fn raw(&self, i: usize) -> f64 {
500        self.raw.get(i).copied().filter(|v| v.is_finite()).or_else(|| self.defaults.get(i).copied()).unwrap_or(0.0)
501    }
502    /// Adjust `i` as a fraction (1/100 000), pinned to `lo..=hi` (fractions).
503    fn frac(&self, i: usize, lo: f64, hi: f64) -> f64 {
504        (self.raw(i) / 100_000.0).clamp(lo, hi)
505    }
506    /// Adjust `i` as an angle in degrees (1/60 000), in 0..360.
507    fn angle(&self, i: usize) -> f64 {
508        (self.raw(i) / 60_000.0).rem_euclid(360.0)
509    }
510}

512fn line(pts: &[(f64, f64)]) -> Vec<Seg> {
513    pts.iter().enumerate().map(|(i, &(x, y))| if i == 0 { Seg::M(x, y) } else { Seg::L(x, y) }).collect()
514}
515
516/// Quadratic from (0, y0) through control (0.5, c) to (1, y0).
517fn quad(y0: f64, c: f64) -> Vec<Seg> {
518    vec![Seg::M(0.0, y0), Seg::Q(0.5, c, 1.0, y0)]
519}
520
521/// A sweep in (0, 360]: zero means a full turn.
522fn turn(deg: f64) -> f64 {
523    let d = deg.rem_euclid(360.0);
524    if d < 1e-9 { 360.0 } else { d }
525}
526
527/// Arc of the ellipse inscribed in the rectangle, scaled by `s` about its centre.
528fn ellipse(s: f64, start: f64, sweep: f64) -> Vec<Seg> {
529    vec![Seg::A(0.5, 0.5, 0.5 * s, 0.5 * s, start, sweep)]
530}
531
532/// The arch-up arc for adjust angle `a`: symmetric about the top (270°), clockwise.
533fn arch_up(a: f64) -> (f64, f64) {
534    (a, turn(2.0 * (270.0 - a)))
535}
536
537/// The arch-down arc for adjust angle `a`: symmetric about the bottom (90°), counter-clockwise.
538fn arch_down(a: f64) -> (f64, f64) {
539    (180.0 - a, -turn(180.0 - 2.0 * a))
540}
541
542/// The circle arc for adjust angle `a`: from `a` clockwise, symmetric about 0°.
543fn circle(a: f64) -> (f64, f64) {
544    (a, turn(2.0 * (360.0 - a)))
545}

547/// One cubic wave period from x0 to x1 at height y: up first when `k` > 0 (y decreases).
548fn wave(out: &mut Vec<Seg>, x0: f64, x1: f64, y: f64, k: f64) {
549    let l = x1 - x0;
550    out.push(Seg::C(x0 + l / 3.0, y - k, x0 + 2.0 * l / 3.0, y + k, x1, y));
551}
552
553/// Wave guides: `periods` cubic waves over the width, shifted apart by `shift` (fraction).
554fn waves(a: f64, shift: f64, periods: usize, up_first: bool) -> Spec {
555    let d = 2.0 * shift;
556    let (top_x, bot_x) = if d >= 0.0 { ((0.0, 1.0 - d), (d, 1.0)) } else { ((-d, 1.0), (0.0, 1.0 + d)) };
557    let k = if up_first { a * 10.0 / 3.0 } else { -a * 10.0 / 3.0 };
558    let guide = |(x0, x1): (f64, f64), y: f64| {
559        let mut v = vec![Seg::M(x0, y)];
560        let n = periods.max(1) as f64;
561        for i in 0..periods.max(1) {
562            let f = i as f64;
563            wave(&mut v, x0 + (x1 - x0) * f / n, x0 + (x1 - x0) * (f + 1.0) / n, y, k);
564        }
565        v
566    };
567    Spec::Fill(guide(top_x, a), guide(bot_x, 1.0 - a))
568}

759/// Build a guide from unit-space segments mapped onto `r`.
760fn guide(segs: &[Seg], r: Rect) -> Guide {
761    let mut p = BezPath::new();
762    let mut started = false;
763    let mut closed = false;
764    for s in segs {
765        match *s {
766            Seg::M(x, y) => {
767                p.move_to((x, y));
768                started = true;
769            }
770            Seg::L(x, y) => {
771                if started {
772                    p.line_to((x, y))
773                } else {
774                    p.move_to((x, y))
775                }
776                started = true;
777            }
778            Seg::Q(cx, cy, x, y) => {
779                if started {
780                    p.quad_to((cx, cy), (x, y));
781                }
782            }
783            Seg::C(x1, y1, x2, y2, x, y) => {
784                if started {
785                    p.curve_to((x1, y1), (x2, y2), (x, y));
786                }
787            }
788            Seg::A(cx, cy, rx, ry, start, sweep) => {
789                let arc = Arc::new((cx, cy), (rx.max(0.0), ry.max(0.0)), start.to_radians(), sweep.to_radians(), 0.0);
790                let s0 = Point::new(cx + rx * start.to_radians().cos(), cy + ry * start.to_radians().sin());
791                if started {
792                    p.line_to(s0);
793                } else {
794                    p.move_to(s0);
795                }
796                started = true;
797                for el in arc.append_iter(1e-5) {
798                    p.push(el);
799                }
800                closed = sweep.abs() >= 359.999;
801            }
802        }
803    }
804    let m = kurbo::Affine::new([r.width(), 0.0, 0.0, r.height(), r.x0, r.y0]);
805    p.apply_affine(m);
806    Guide::from_path(&p, closed)
807}

809/// The warp `name` with adjust values `adj` (raw guide values; missing ones take the defaults)
810/// over text rectangle `rect`. `None` for unknown presets, `textNoShape` or a non-finite rect.
811pub fn warp(name: &str, adj: &[f64], rect: Rect) -> Option<Warp> {
812    if ![rect.x0, rect.y0, rect.x1, rect.y1].iter().all(|v| v.is_finite()) {
813        return None;
814    }
815    let r = Rect::new(rect.x0.min(rect.x1), rect.y0.min(rect.y1), rect.x0.max(rect.x1), rect.y0.max(rect.y1));
816    let a = Adj { raw: adj, defaults: adjust_defaults(name) };
817    let specs = spec(name, &a)?;
818    let bands = specs
819        .into_iter()
820        .map(|s| match s {
821            Spec::Fill(t, b) => Band::Fill { top: guide(&t, r), bottom: guide(&b, r) },
822            Spec::Follow(p, anchor) => Band::Follow { path: guide(&p, r), anchor },
823        })
824        .collect();
825    Some(Warp { bands })
826}

828/// The band line `line` of `lines` goes to when there are `bands` bands: the first ⌈n/k⌉ lines
829/// share the first band, and so on.
830pub fn band_of_line(line: usize, lines: usize, bands: usize) -> usize {
831    if bands <= 1 || lines == 0 {
832        return 0;
833    }
834    // Band b holds lines ceil(b·n/k) .. ceil((b+1)·n/k).
835    (0..bands).rev().find(|&b| (b * lines).div_ceil(bands) <= line).unwrap_or(0)
836}

870/// Maps points of laid-out lines onto a warp.
871#[derive(Clone, Debug)]
872pub struct Mapper<'a> {

875    offsets: Vec<Guide>,
876    zoom: f64,

878
879impl Warp {
880    /// Plan the mapping of `lines`, laid out in the inner text box `inner` (the line's place in
881    /// it gives its alignment along follow guides; the insets themselves don't shorten a guide).
882    pub fn mapper(&self, lines: &[WarpLine], inner: Rect) -> Mapper<'_> {
883        let nb = self.bands.len().max(1);
884        let n = lines.len();
885        let band_of: Vec<usize> = (0..n).map(|i| band_of_line(i, n, nb).min(nb - 1)).collect();
886        // Ink of the whole text, for the shared horizontal centre and scale of fill bands.
887        let ink_all = lines.iter().filter_map(|l| l.ink).reduce(|a, b| a.union(b));
888        let cx = ink_all.map(|r| r.center().x).unwrap_or(0.0);
889        let mut k_fill = f64::INFINITY;
890        let mut zoom: f64 = 1.0;
891        let mut band_y: Vec<Option<(f64, f64)>> = vec![None; nb];
892        for (i, l) in lines.iter().enumerate() {
893            let Some(r) = l.ink else { continue };
894            let b = band_of.get(i).copied().unwrap_or(0);
895            if let Some(slot) = band_y.get_mut(b) {
896                *slot = Some(match *slot {
897                    Some((y0, y1)) => (y0.min(r.y0), y1.max(r.y1)),
898                    None => (r.y0, r.y1),
899                });
900            }
901        }
902        // Every fill band maps a box as tall as the tallest band's ink, centred on its own ink, so
903        // a band with fewer lines keeps them at the same size, centred between its guides.
904        let tallest = band_y.iter().flatten().map(|(y0, y1)| y1 - y0).fold(0.0, f64::max);
905        let band_y: Vec<Option<(f64, f64)>> =
906            band_y.iter().map(|r| r.map(|(y0, y1)| ((y0 + y1 - tallest) / 2.0, (y0 + y1 + tallest) / 2.0))).collect();
907        for (b, band) in self.bands.iter().enumerate() {
908            if let Band::Fill { top, bottom } = band {
909                let half = lines
910                    .iter()
911                    .enumerate()
912                    .filter(|(i, _)| band_of.get(*i) == Some(&b))
913                    .filter_map(|(_, l)| l.ink)
914                    .map(|r| (r.x0 - cx).abs().max((r.x1 - cx).abs()))
915                    .fold(0.0, f64::max);
916                if half > 1e-9 {
917                    k_fill = k_fill.min(band.length() / (2.0 * half));
918                }
919                if let Some(Some((y0, y1))) = band_y.get(b) {
920                    let thick = [0.0, 0.25, 0.5, 0.75, 1.0].iter().map(|&u| (top.at_frac(u) - bottom.at_frac(u)).hypot()).fold(0.0, f64::max);
921                    if y1 - y0 > 1e-9 {
922                        zoom = zoom.max(thick / (y1 - y0));
923                    }
924                }
925            }
926        }
927        if !k_fill.is_finite() {
928            k_fill = 1.0;
929        }
930        zoom = zoom.max(k_fill);
931        // Per band, measured once: its widest line and its first and last lines' baselines.
932        let mut stats: Vec<Option<(f64, f64, f64)>> = vec![None; nb];
933        for (l, &b) in lines.iter().zip(&band_of) {
934            let w = l.extent.1 - l.extent.0;
935            if let Some(slot) = stats.get_mut(b) {
936                match slot {
937                    Some((widest, _, last)) => {
938                        *widest = widest.max(w);
939                        *last = l.ref_y;
940                    }
941                    None => *slot = Some((0.0f64.max(w), l.ref_y, l.ref_y)),
942                }
943            }
944        }
945        let mut maps = Vec::with_capacity(n);
946        let mut offsets = Vec::new();
947        for (i, l) in lines.iter().enumerate() {
948            let b = band_of.get(i).copied().unwrap_or(0);
949            let band = stats.get(b).copied().flatten();
950            match self.bands.get(b) {
951                Some(Band::Fill { .. }) | None => {
952                    let (y0, y1) = band_y.get(b).copied().flatten().unwrap_or((0.0, 1.0));
953                    let len = self.bands.get(b).map(Band::length).unwrap_or(0.0);
954                    maps.push(LineMap::Fill { band: b, cx, k: k_fill, len, y0, y1 });
955                }
956                Some(Band::Follow { path, anchor }) => {
957                    let len = path.length();
958                    let widest = band.map_or(0.0, |s| s.0);
959                    let k = if widest > len && widest > 1e-9 { len / widest } else { 1.0 };
960                    let ref_y = match anchor {
961                        LineAnchor::Last => band.map(|s| s.2),
962                        LineAnchor::First => band.map(|s| s.1),
963                        LineAnchor::Middle => band.map(|s| (s.1 + s.2) / 2.0),
964                    }
965                    .unwrap_or(l.ref_y);
966                    // Lines off the guide run on a parallel curve at their natural length.
967                    let d = (ref_y - l.ref_y) * k;
968                    let (guide, len) = if d.abs() > 1e-6 && d.is_finite() && offsets.len() < 4096 {
969                        let g = path.offset(d);
970                        let len = g.length();
971                        offsets.push(g);
972                        (Some(offsets.len() - 1), len)
973                    } else {
974                        (None, len)
975                    };
976                    let (e0, e1) = l.extent;
977                    let w = (e1 - e0).max(0.0) * k;
978                    let (gl, gr) = ((e0 - inner.x0).max(0.0), (inner.x1 - e1).max(0.0));
979                    let f = if gl + gr > 1e-9 { gl / (gl + gr) } else { 0.5 };
980                    let s0 = f * (len - w);
981                    maps.push(LineMap::Follow { band: b, guide, e0, s0, k, ref_y: l.ref_y });
982                }
983            }
984        }
985        Mapper { warp: self, lines: maps, offsets, zoom: if zoom.is_finite() { zoom.clamp(1.0, 1.0e4) } else { 1.0 } }
986    }
987}
988
989impl Mapper<'_> {
990    /// How much the warp magnifies text at most (for flattening tolerances), ≥ 1.
991    pub fn zoom(&self) -> f64 {
992        self.zoom
993    }
994
995    /// Map point `p` of line `line` (layout coordinates) onto the warp.
996    pub fn map(&self, line: usize, p: Point) -> Point {
997        let out = match self.lines.get(line) {
998            Some(&LineMap::Fill { band, cx, k, len, y0, y1 }) => {
999                let Some(Band::Fill { top, bottom }) = self.warp.bands.get(band) else { return p };
1000                let u = if len > 1e-9 { 0.5 + (p.x - cx) * k / len } else { 0.5 };
1001                let v = if y1 - y0 > 1e-9 { (p.y - y0) / (y1 - y0) } else { 0.5 };
1002                let (t, b) = (top.at_frac(u), bottom.at_frac(u));
1003                t.lerp(b, v)
1004            }
1005            Some(&LineMap::Follow { band, guide, e0, s0, k, ref_y }) => {
1006                let path = match (guide.and_then(|g| self.offsets.get(g)), self.warp.bands.get(band)) {
1007                    (Some(g), _) => g,
1008                    (None, Some(Band::Follow { path, .. })) => path,
1009                    _ => return p,
1010                };
1011                let s = s0 + (p.x - e0) * k;
1012                let d = (ref_y - p.y) * k;
1013                let (q, t) = path.at(s);
1014                q + Vec2::new(t.y, -t.x) * d
1015            }
1016            None => p,
1017        };
1018        if out.x.is_finite() && out.y.is_finite() { out } else { p }
1019    }
