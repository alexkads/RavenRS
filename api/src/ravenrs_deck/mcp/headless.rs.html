27fn s<'a>(p: &'a Value, k: &str) -> Option<&'a str> {
28    p.get(k).and_then(Value::as_str)
29}

31pub fn pointer_kind(k: &str) -> Option<PointerKind> {
32    Some(match k {
33        "down" => PointerKind::Down,
34        "drag" => PointerKind::Drag,
35        "up" => PointerKind::Up,
36        "move" => PointerKind::Move,
37        "doubleclick" | "dblclick" => PointerKind::DoubleClick,
38        "tripleclick" => PointerKind::TripleClick,
39        "contextclick" | "rightclick" => PointerKind::ContextClick,
40        _ => return None,
41    })
42}
43
44pub fn mods_from(p: &Value) -> Mods {
45    let b = |n: &str| p.get(n).and_then(Value::as_bool).unwrap_or(false);
46    Mods { shift: b("shift"), alt: b("alt"), cmd: b("cmd") || b("ctrl") }
47}

49/// A tool name (`select`, `textBox`, `shape`, `pen`, `highlighter`, `eraser`) to a [`ToolKind`].
50pub fn tool_kind(name: &str, preset: Option<&str>) -> Option<ToolKind> {
51    Some(match name {
52        "select" | "selection" | "arrow" => ToolKind::Select,
53        "textBox" | "text" | "textbox" => ToolKind::TextBox,
54        "shape" => ToolKind::Shape { preset: preset.unwrap_or("rect").to_string() },
55        "pen" => ToolKind::Ink { mode: "pen".into(), color: crate::engine::model::Rgba::BLACK, width: 2.0 },
56        "highlighter" => ToolKind::Ink { mode: "highlighter".into(), color: crate::engine::model::Rgba::rgb(255, 230, 0), width: 10.0 },
57        "eraser" => ToolKind::Ink { mode: "eraser".into(), color: crate::engine::model::Rgba::BLACK, width: 2.0 },
58        other if crate::engine::geom::preset::info(other).is_some() => ToolKind::Shape { preset: other.to_string() },
59        other if crate::engine::tools::is_freeform_tool(other) => ToolKind::Shape { preset: other.to_string() },
60        _ => return None,
61    })
62}

81    fn exec(&mut self, id: &str, params: &Value) -> Result<Value, String> {
82        let r = self.session.execute(id, params).map_err(|e| {
83            let ui_only = ["app.", "view.", "window.", "ui."].iter().any(|p| id.starts_with(p));
84            if ui_only && crate::engine::find_command(id).is_none() { format!("`{id}` is a UI command; {NEEDS_APP}") } else { e.to_string() }
85        });
86        self.session.ui_requests.clear();
87        r
88    }
89
90    fn pointer(&mut self, p: &Value) -> Result<Value, String> {
91        let events = p.get("events").and_then(Value::as_array).ok_or("missing `events`")?;
92        let base = p.get("mods").map(mods_from).unwrap_or_default();
93        if self.session.active().is_none() {
94            return Err("no presentation open (use new_presentation first)".into());
95        }
96        let mut last = Value::Null;
97        for e in events {
98            let kind = s(e, "kind").unwrap_or("");
99            let kind = pointer_kind(kind)
100                .ok_or_else(|| format!("unknown pointer kind `{kind}` (down|drag|up|move|doubleclick|tripleclick|contextclick)"))?;
101            let x = e.get("x").and_then(Value::as_f64).ok_or("pointer event needs `x`")?;
102            let y = e.get("y").and_then(Value::as_f64).ok_or("pointer event needs `y`")?;
103            let mods = e.get("mods").map(mods_from).unwrap_or(base);
104            let pressure = e.get("pressure").and_then(Value::as_f64).map(|v| v as f32);
105            last = self.session.pointer(PointerEvent { kind, x, y, mods, tol: 3.0, pressure }).map_err(|e| e.to_string())?;
106        }
107        self.session.ui_requests.clear();
108        Ok(json!({"result": last, "selection": self.session.active().map(|d| serde_json::to_value(&d.selection).unwrap_or_default())}))
109    }
110
111    fn render(&mut self, p: &Value) -> Result<Value, String> {
112        let st = self.session.active().ok_or("no presentation open")?;
113        let i = p.get("slide").and_then(Value::as_u64).map(|v| v as usize).unwrap_or(st.selection.slide);
114        if i >= st.doc.slides.len() {
115            return Err(format!("no slide {i}"));
116        }
117        let scale = p.get("scale").and_then(Value::as_f64).unwrap_or(1.0).clamp(0.05, 8.0);
118        let img = crate::engine::render::render_slide(
119            &st.doc,
120            i,
121            &crate::engine::render::RenderOpts { scale, edit: p.get("edit").and_then(Value::as_bool).unwrap_or(false), ..Default::default() },
122        );
123        let png = img.to_png_at(crate::engine::render::dpi_for_scale(scale));
124        if let Some(path) = s(p, "path") {
125            crate::engine::io::write_file(path, &png)?;
126            return Ok(json!({"path": path, "width": img.width, "height": img.height}));
127        }
128        Ok(json!({"png": crate::engine::cmd::base64_encode(&png), "width": img.width, "height": img.height, "slide": i}))
129    }
130}

132impl Backend for Headless {
133    fn call(&mut self, method: &str, params: Value) -> Result<Value, String> {
134        let roots = self.roots.clone();
135        file_access::confine(roots.as_ref(), || self.answer(method, params))
136    }
137    fn has_ui(&self) -> bool {
138        false
139    }
140    fn describe(&self) -> String {
141        "headless (in-process engine; no window)".into()
142    }
143}
144
145impl Headless {
146    fn answer(&mut self, method: &str, params: Value) -> Result<Value, String> {
147        let p = &params;
148        match method {
149            "engine.execute" | "command" | "ui.menu.invoke" => {
150                let id = s(p, "command").or(s(p, "id")).ok_or("missing `command`")?.to_string();
151                let cp = p.get("params").cloned().filter(|v| !v.is_null()).unwrap_or(json!({}));
152                self.exec(&id, &cp)
153            }
154            "engine.commands" => Ok(serde_json::to_value(self.session.commands()).unwrap_or_default()),
155            "document.inspect" => self.exec("document.inspect", &json!({})),
156            "ui.pointer" => self.pointer(p),
157            "ui.key" => {
158                let key = s(p, "key").ok_or("missing `key`")?;
159                self.session.key(key, mods_from(p)).map_err(|e| e.to_string())
160            }
161            "ui.text" => {
162                let t = s(p, "text").ok_or("missing `text`")?;
163                self.session.type_text(t).map_err(|e| e.to_string())
164            }
165            "ui.tool.select" => {
166                let name = s(p, "tool").ok_or("missing `tool`")?;
167                let kind = tool_kind(name, s(p, "preset")).ok_or_else(|| format!("unknown tool `{name}`"))?;
168                self.session.set_tool(kind);
169                Ok(json!({"tool": name}))
170            }
171            "ui.render" | "app.renderSlide" => self.render(p),
172            "app.open" => self.exec("file.open", p),
173            "app.save" => self.exec("file.save", p),
174            "app.export" => self.exec("file.export", p),
175            "ui.inspect" => Ok(json!({"headless": true, "tool": serde_json::to_value(&self.session.tool.kind).unwrap_or_default()})),
176            m if m.starts_with("ui.") || m.starts_with("app.") => Err(format!("`{m}`: {NEEDS_APP}")),
177            other => Err(format!("unknown method `{other}`")),
178        }
179    }
180}
