9/// The result of `tools/call`: MCP content blocks plus the `isError` flag.
10#[derive(Clone, Debug, PartialEq)]
11pub struct ToolResult {
12    pub content: Vec<Value>,
13    pub is_error: bool,
14}
15
16impl ToolResult {
17    pub fn text(t: impl Into<String>) -> Self {
18        Self { content: vec![json!({"type": "text", "text": t.into()})], is_error: false }
19    }
20    pub fn json(v: &Value) -> Self {
21        Self::text(serde_json::to_string_pretty(v).unwrap_or_default())
22    }
23    pub fn error(t: impl Into<String>) -> Self {
24        Self { content: vec![json!({"type": "text", "text": t.into()})], is_error: true }
25    }
26    pub fn image(png_base64: String, info: &Value) -> Self {
27        Self {
28            content: vec![json!({"type": "image", "data": png_base64, "mimeType": "image/png"}), json!({"type": "text", "text": info.to_string()})],
29            is_error: false,
30        }
31    }
32    pub fn to_value(&self) -> Value {
33        json!({"content": self.content, "isError": self.is_error})
34    }
35}
36
37fn num(desc: &str) -> Value {
38    json!({"type": "number", "description": desc})
39}
40fn int(desc: &str) -> Value {
41    json!({"type": "integer", "minimum": 0, "description": desc})
42}
43fn string(desc: &str) -> Value {
44    json!({"type": "string", "description": desc})
45}
46fn boolean(desc: &str) -> Value {
47    json!({"type": "boolean", "description": desc})
48}
49fn mods_schema() -> Value {
50    json!({"type": "object", "additionalProperties": false, "description": "Modifier keys (cmd = Command on macOS / Ctrl elsewhere; alt = Option)", "properties": {"shift": {"type": "boolean"}, "alt": {"type": "boolean"}, "cmd": {"type": "boolean"}}})
51}
52fn obj(props: Value, required: &[&str]) -> Value {
53    let mut o = json!({"type": "object", "properties": props, "additionalProperties": false});
54    if !required.is_empty() {
55        o["required"] = json!(required);
56    }
57    o
58}
59fn tool(name: &str, title: &str, desc: &str, schema: Value, read_only: bool) -> Value {
60    json!({"name": name, "title": title, "description": desc, "inputSchema": schema, "annotations": {"title": title, "readOnlyHint": read_only, "openWorldHint": false}})
61}

301fn wrap(r: Result<Value, String>) -> ToolResult {
302    match r {
303        Ok(v) => ToolResult::json(&v),

308/// Keys other than the tool's schema properties are refused, so a misspelled argument does not
309/// succeed with the tool's defaults. Keys inside a free object such as `run_command`'s `params`
310/// belong to the command and are left for the command to read.
311fn unknown_argument_error(name: &str, args: &Value) -> Option<String> {
312    let defs = tool_definitions();
313    let def = defs.iter().find(|t| t.get("name").and_then(Value::as_str) == Some(name))?;
314    let mut allowed: Vec<&str> = def
315        .get("inputSchema")
316        .and_then(|s| s.get("properties"))
317        .and_then(Value::as_object)
318        .map(|p| p.keys().map(String::as_str).collect())
319        .unwrap_or_default();
320    allowed.sort_unstable();
321    let obj = args.as_object()?;
322    let mut unknown: Vec<&str> = obj.keys().map(String::as_str).filter(|k| !allowed.contains(k)).collect();
323    unknown.sort_unstable();
324    let first = unknown.first()?;
325    let expected = if allowed.is_empty() { "none".to_string() } else { allowed.join(", ") };
326    let (what, list) = if unknown.len() == 1 {
327        ("unknown argument", format!("{first:?}"))
328    } else {
329        ("unknown arguments", unknown.iter().map(|k| format!("{k:?}")).collect::<Vec<_>>().join(", "))
330    };
331    Some(format!("{name}: {what} {list} (expected: {expected})"))
332}

334pub fn call_tool(b: &mut dyn Backend, name: &str, a: &Value) -> ToolResult {
335    let a = if a.is_object() { a.clone() } else { json!({}) };
336    if let Some(msg) = unknown_argument_error(name, &a) {
337        return ToolResult::error(msg);
338    }
339    match name {
340        "list_commands" => wrap(b.call("engine.commands", json!({})).map(|v| {
341            let f = a.get("filter").and_then(Value::as_str).unwrap_or("").to_lowercase();
342            Value::Array(
343                v.as_array()
344                    .cloned()
345                    .unwrap_or_default()
346                    .into_iter()
347                    .filter(|c| {
348                        f.is_empty()
349                            || c.get("id").and_then(Value::as_str).is_some_and(|i| i.to_lowercase().contains(&f))
350                            || c.get("label").and_then(Value::as_str).is_some_and(|i| i.to_lowercase().contains(&f))
351                    })
352                    .collect(),
353            )
354        })),
355        "run_command" => {
356            let Some(c) = a.get("command").and_then(Value::as_str) else { return ToolResult::error("missing `command`") };
357            wrap(exec(b, c, a.get("params").cloned().unwrap_or(json!({}))))
358        }
359        "batch" => {
360            use crate::mcp::server::{MAX_BATCH_BYTES, MAX_BATCH_ITEMS};
361            let keep = a.get("keepGoing").and_then(Value::as_bool).unwrap_or(false);
362            let commands = a.get("commands").and_then(Value::as_array).map(Vec::as_slice).unwrap_or_default();
363            if commands.len() > MAX_BATCH_ITEMS {
364                return ToolResult::error(format!("a batch holds at most {MAX_BATCH_ITEMS} commands"));
365            }
366            let mut out = vec![];
367            let mut used = 0usize;
368            for c in commands {
369                let id = c.get("command").and_then(Value::as_str).unwrap_or("");
370                let (entry, failed) = match exec(b, id, c.get("params").cloned().unwrap_or(json!({}))) {
371                    Ok(v) => (json!({"command": id, "ok": true, "result": v}), false),
372                    Err(e) => (json!({"command": id, "ok": false, "error": e}), true),
373                };
374                // What the reply carries: the entry as text, escaped again inside the JSON-RPC reply.
375                // Failures count too: with keepGoing, a remote backend's large errors add up.
376                used = used.saturating_add(json!(entry.to_string()).to_string().len());
377                if used > MAX_BATCH_BYTES {
378                    let shown: String = id.chars().take(128).collect();
379                    out.push(json!({"command": shown, "ok": false, "error": "the batch's results passed 64 MiB; stopped here"}));
380                    return ToolResult { content: vec![json!({"type": "text", "text": Value::Array(out).to_string()})], is_error: true };
381                }
382                out.push(entry);
383                if failed && !keep {
384                    return ToolResult { content: vec![json!({"type": "text", "text": Value::Array(out).to_string()})], is_error: true };
385                }
386            }
387            // The reply is pretty-printed, which is longer than the compact entries counted above.
388            let reply = ToolResult::json(&Value::Array(out));
389            let text = reply.content[0]["text"].as_str().unwrap_or_default();
390            if json!(text).to_string().len() > MAX_BATCH_BYTES {
391                return ToolResult::error("the batch's results passed 64 MiB; nothing is returned");
392            }
393            reply
394        }
395        "new_presentation" => wrap(exec(b, "file.new", a)),
396        "open_presentation" => wrap(exec(b, "file.open", a)),
397        "save_presentation" => wrap(exec(b, "file.save", a)),
398        "export" => wrap(exec(b, "file.export", a)),
399        "inspect_document" => wrap(b.call("document.inspect", json!({}))),
400        "inspect_slide" => wrap(exec(b, "slide.inspect", a)),
401        "render_slide" => match b.call("ui.render", a.clone()) {
402            Ok(v) => match v.get("png").and_then(Value::as_str) {
403                Some(png) => {
404                    ToolResult::image(png.to_string(), &json!({"width": v.get("width"), "height": v.get("height"), "slide": v.get("slide")}))
405                }
406                None => ToolResult::json(&v),
407            },
408            Err(e) => ToolResult::error(e),
409        },
410        "add_slide" => wrap(exec(b, "slide.new", a)),
411        "go_to_slide" => wrap(exec(b, "slide.go", a)),
412        "add_shape" => {
413            let r = exec(
414                b,
415                "shape.insert",
416                json!({"preset": a.get("preset"), "rect": [f(&a, "x"), f(&a, "y"), f(&a, "w"), f(&a, "h")], "text": a.get("text")}),
417            );
418            let Ok(v) = r else { return wrap(r) };
419            let id = v.get("id").cloned().unwrap_or_default();
420            if let Some(c) = a.get("fill").filter(|c| c.is_string())
421                && let Err(e) = exec(b, "shape.fill", json!({"id": id, "color": c}))
422            {
423                return ToolResult::error(e);
424            }
425            if let Some(c) = a.get("line").and_then(Value::as_str) {
426                let p = if c == "none" { json!({"id": id, "none": true}) } else { json!({"id": id, "color": c}) };
427                if let Err(e) = exec(b, "shape.line", p) {
428                    return ToolResult::error(e);
429                }
430            }
431            ToolResult::json(&v)
432        }
433        "add_text_box" => {
434            let r =
435                exec(b, "insert.textBox", json!({"rect": [f(&a, "x"), f(&a, "y"), f(&a, "w"), f(&a, "h").unwrap_or(30.0)], "text": a.get("text")}));
436            let Ok(v) = r else { return wrap(r) };
437            let id = v.get("id").cloned().unwrap_or_default();
438            let _ = exec(b, "text.exit", json!({}));
439            let _ = exec(b, "edit.select", json!({"ids": [id]}));
440            if let Some(s) = f(&a, "size") {
441                let _ = exec(b, "format.size", json!({"size": s}));
442            }
443            if let Some(c) = a.get("color").filter(|c| c.is_string()) {
444                let _ = exec(b, "format.color", json!({"color": c}));
445            }
446            if let Some(al) = a.get("align").and_then(Value::as_str) {
447                let _ = exec(b, "format.align", json!({"align": al}));
448            }
449            ToolResult::json(&v)
450        }
451        "set_text" => wrap(exec(b, "text.set", a)),
452        "insert_picture" => {
453            let mut p = a.clone();
454            if let (Some(x), Some(y), Some(w), Some(h)) = (f(&a, "x"), f(&a, "y"), f(&a, "w"), f(&a, "h")) {
455                p["rect"] = json!([x, y, w, h]);
456            }
457            wrap(exec(b, "insert.picture", p))
458        }
459        "insert_table" => {
460            let mut p = a.clone();
461            if let (Some(x), Some(y), Some(w), Some(h)) = (f(&a, "x"), f(&a, "y"), f(&a, "w"), f(&a, "h")) {
462                p["rect"] = json!([x, y, w, h]);
463            }
464            wrap(exec(b, "insert.table", p))
465        }
466        "insert_chart" => {
467            let mut p = a.clone();
468            if let (Some(x), Some(y), Some(w), Some(h)) = (f(&a, "x"), f(&a, "y"), f(&a, "w"), f(&a, "h")) {
469                p["rect"] = json!([x, y, w, h]);
470            }
471            wrap(exec(b, "insert.chart", p))
472        }
473        "select" => wrap(exec(b, "edit.select", a)),
474        "pointer" => wrap(b.call("ui.pointer", a)),
475        "select_tool" => wrap(b.call("ui.tool.select", a)),
476        "key" => {
477            let mut p = a.get("mods").cloned().filter(Value::is_object).unwrap_or(json!({}));
478            p["key"] = a.get("key").cloned().unwrap_or_default();
479            wrap(b.call("ui.key", p))
480        }
481        "type_text" => wrap(b.call("ui.text", a)),
482        "screenshot" => {
483            if !b.has_ui() {
484                return ToolResult::error(format!("screenshot: {NEEDS_APP}. Use render_slide to see a slide headlessly."));
485            }
486            match b.call("ui.screenshot", json!({"base64": true})) {
487                Ok(v) => match v.get("png").and_then(Value::as_str) {
488                    Some(png) => ToolResult::image(png.to_string(), &json!({"width": v.get("width"), "height": v.get("height")})),
489                    None => ToolResult::json(&v),
490                },
491                Err(e) => ToolResult::error(e),
492            }
493        }
494        "call_app" => {
495            let Some(m) = a.get("method").and_then(Value::as_str) else { return ToolResult::error("missing `method`") };
496            wrap(b.call(m, a.get("params").cloned().unwrap_or(json!({}))))
497        }
498        other => ToolResult::error(format!("unknown tool `{other}`")),
499    }
