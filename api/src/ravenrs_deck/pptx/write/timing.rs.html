11pub fn transition(w: &mut W, x: &mut Exp, o: &mut Out, t: &Transition) {
12    let adv_click = (!t.advance_on_click).then_some("0");
13    let adv_tm = t.advance_after_ms.map(|v| v.min(86_400_000));
14    let spd = match t.duration_ms {
15        0..=600 => "fast",
16        601..=900 => "med",
17        _ => "slow",
18    };
19    let sound = t.sound.and_then(|m| x.media_rel(o, m, "audio").map(|r| (r, x.media(m).map(|i| i.name.clone()).unwrap_or_default())));
20    let snd = |w: &mut W| {
21        if let Some((r, name)) = &sound {
22            w.open0("p:sndAc");
23            w.open("p:stSnd", A::new().t("loop", t.sound_loop));
24            let name = if name.is_empty() { "sound.wav" } else { name.as_str() };
25            w.empty("p:snd", A::new().a("r:embed", r).a("name", name));
26            w.close("p:stSnd");
27            w.close("p:sndAc");
28        } else if t.stop_sound {
29            w.open0("p:sndAc");
30            w.empty0("p:endSnd");
31            w.close("p:sndAc");
32        }
33    };
34    let mut effect = W::frag();
35    let ns = if t.kind == "none" { Some("p") } else { super::super::maps::transition_to_xml(&mut effect, &t.kind, &t.option) };
36    let ns = match ns {
37        Some(ns) => ns,
38        None => {
39            // Unknown kind: write the kept element if there is one, else a fade.
40            if let Some(raw) = &t.raw
41                && let Ok(d) = crate::pptx::xml::parse(raw.as_bytes())
42                && let Some(e) = d.root.elements().find(|e| !matches!(e.local(), "sndAc" | "extLst"))
43                && !crate::pptx::read::has_rel_refs(e)
44            {
45                let prefix = e.prefix().to_string();
46                let mut ns_decl = d.ns_decls();
47                ns_decl.retain(|(k, _)| k == &format!("xmlns:{prefix}"));
48                let mut e2 = e.clone();
49                for (k, v) in ns_decl {
50                    e2.attrs.push((k, v));
51                }
52                effect.raw(&e2.to_xml());
53                match prefix.as_str() {
54                    "p14" => "p14",
55                    "p15" => "p15",
56                    "p159" => "p159",
57                    "p" => "p",
58                    _ => {
59                        effect = W::frag();
60                        effect.empty0("p:fade");
61                        "p"
62                    }
63                }
64            } else {
65                effect.empty0("p:fade");
66                "p"
67            }
68        }
69    };
70    let is_none = t.kind == "none";
71    let dur = t.duration_ms.min(60_000);
72    w.open("mc:AlternateContent", A::new().a("xmlns:mc", NS_MC));
73    let mut ch = A::new().a("xmlns:p14", NS_P14);
74    let req = match ns {
75        "p15" => {
76            ch = ch.a("xmlns:p15", NS_P15);
77            "p15"
78        }
79        "p159" => {
80            ch = ch.a("xmlns:p159", NS_P159);
81            "p159"
82        }
83        _ => "p14",
84    };
85    w.open("mc:Choice", ch.a("Requires", req));
86    w.open("p:transition", A::new().a("spd", spd).o("p14:dur", (!is_none).then_some(dur)).o("advClick", adv_click).o("advTm", adv_tm));
87    if !is_none {
88        w.raw(&effect.s);
89    }
90    snd(w);
91    w.close("p:transition");
92    w.close("mc:Choice");
93    w.open0("mc:Fallback");
94    w.open("p:transition", A::new().a("spd", spd).o("advClick", adv_click).o("advTm", adv_tm));
95    if !is_none {
96        if ns == "p" {
97            w.raw(&effect.s);
98        } else {
99            w.empty0("p:fade");
100        }
101    }
102    snd(w);
103    w.close("p:transition");
104    w.close("mc:Fallback");
105    w.close("mc:AlternateContent");
106}

126fn tgt(w: &mut W, spid: u32, para: Option<u32>) {
127    w.open0("p:tgtEl");
128    match para {
129        Some(p) => {
130            w.open("p:spTgt", A::new().a("spid", spid));
131            w.open0("p:txEl");
132            w.empty("p:pRg", A::new().a("st", p).a("end", p));
133            w.close("p:txEl");
134            w.close("p:spTgt");
135        }
136        None => w.empty("p:spTgt", A::new().a("spid", spid)),
137    }
138    w.close("p:tgtEl");
139}

141fn cbhvr(w: &mut W, ids: &mut Ids, e: &Eff, dur: u32, delay: Option<u32>, attrs: &[&str], additive: Option<&str>) {
142    w.open("p:cBhvr", A::new().o("additive", additive));
143    let a = A::new().a("id", ids.next()).a("dur", dur.max(1)).a("fill", "hold");
144    match delay {
145        Some(d) => {
146            w.open("p:cTn", a);
147            w.open0("p:stCondLst");
148            w.empty("p:cond", A::new().a("delay", d));
149            w.close("p:stCondLst");
150            w.close("p:cTn");
151        }
152        None => w.empty("p:cTn", a),
153    }
154    tgt(w, e.spid, e.paragraph);
155    if !attrs.is_empty() {
156        w.open0("p:attrNameLst");
157        for n in attrs {
158            w.elt("p:attrName", n);
159        }
160        w.close("p:attrNameLst");
161    }
162    w.close("p:cBhvr");
163}

180fn anim_prop(w: &mut W, ids: &mut Ids, e: &Eff, dur: u32, attr: &str, from: &str, to: &str) {
181    w.open("p:anim", A::new().a("calcmode", "lin").a("valueType", "num"));
182    cbhvr(w, ids, e, dur, None, &[attr], Some("base"));
183    w.open0("p:tavLst");
184    for (tm, v) in [(0, from), (100_000, to)] {
185        w.open("p:tav", A::new().a("tm", tm));
186        w.open0("p:val");
187        w.val("p:strVal", v);
188        w.close("p:val");
189        w.close("p:tav");
190    }
191    w.close("p:tavLst");
192    w.close("p:anim");
193}

195fn filter_for(effect: &str, option: &str) -> &'static str {
196    match effect {
197        "wipe" | "wipeOut" => match option {
198            "t" => "wipe(down)",
199            "l" => "wipe(right)",
200            "r" => "wipe(left)",
201            _ => "wipe(up)",
202        },
203        "split" | "splitOut" => match option {
204            "horzOut" => "barn(outHorizontal)",
205            "vertIn" => "barn(inVertical)",
206            "vertOut" => "barn(outVertical)",
207            _ => "barn(inHorizontal)",
208        },
209        "randomBars" | "randomBarsOut" => {
210            if option == "vert" {
211                "randombar(vertical)"
212            } else {
213                "randombar(horizontal)"
214            }
215        }
216        "blinds" | "blindsOut" => {
217            if option == "vert" {
218                "blinds(vertical)"
219            } else {
220                "blinds(horizontal)"
221            }
222        }
223        "box" | "boxOut" => {
224            if option == "out" {
225                "box(out)"
226            } else {
227                "box(in)"
228            }
229        }
230        "circle" => {
231            if option == "out" {
232                "circle(out)"
233            } else {
234                "circle(in)"
235            }
236        }
237        "diamond" | "diamondOut" => {
238            if option == "out" {
239                "diamond(out)"
240            } else {
241                "diamond(in)"
242            }
243        }
244        "shape" | "shapeOut" => match option {
245            "circleOut" => "circle(out)",
246            "boxIn" => "box(in)",
247            "boxOut" => "box(out)",
248            "diamondIn" => "diamond(in)",
249            "diamondOut" => "diamond(out)",
250            "plusIn" => "plus(in)",
251            "plusOut" => "plus(out)",
252            _ => "circle(in)",
253        },
254        "checkerboard" | "checkerboardOut" => {
255            if option == "down" {
256                "checkerboard(down)"
257            } else {
258                "checkerboard(across)"
259            }
260        }
261        "dissolve" | "dissolveOut" => "dissolve",
262        "wheel" | "wheelOut" => match option {
263            "2" => "wheel(2)",
264            "3" => "wheel(3)",
265            "4" => "wheel(4)",
266            "8" => "wheel(8)",
267            _ => "wheel(1)",
268        },
269        "strips" | "stripsOut" => match option {
270            "ru" => "strips(upRight)",
271            "ld" => "strips(downLeft)",
272            "rd" => "strips(downRight)",
273            _ => "strips(upLeft)",
274        },
275        "peek" | "peekOut" => match option {
276            "t" => "slide(fromTop)",
277            "l" => "slide(fromLeft)",
278            "r" => "slide(fromRight)",
279            _ => "slide(fromBottom)",
280        },
281        _ => "fade",
282    }
283}

285fn fly_from(option: &str) -> (&'static str, &'static str) {
286    let x = if option.contains('l') {
287        "0-#ppt_w/2"
288    } else if option.contains('r') {
289        "1+#ppt_w/2"
290    } else {
291        "#ppt_x"
292    };
293    let y = if option.contains('t') {
294        "0-#ppt_h/2"
295    } else if option.contains('b') || option.is_empty() {
296        "1+#ppt_h/2"
297    } else {
298        "#ppt_y"
299    };
300    (x, y)
301}

303fn behaviors(w: &mut W, x: &mut Exp, ids: &mut Ids, e: &Eff) {
304    let a = e.a;
305    let dur = a.duration_ms.clamp(1, 3_600_000);
306    match a.class {
307        AnimClass::Entrance => {
308            set_vis(w, ids, e, 0, true);
309            match a.effect.as_str() {
310                "appear" => {}
311                "fly" => {
312                    let (fx, fy) = fly_from(&a.option);
313                    anim_prop(w, ids, e, dur, "ppt_x", fx, "#ppt_x");
314                    anim_prop(w, ids, e, dur, "ppt_y", fy, "#ppt_y");
315                }
316                "zoom" | "grow" | "expand" | "spinner" => {
317                    anim_prop(w, ids, e, dur, "ppt_w", "0", "#ppt_w");
318                    anim_prop(w, ids, e, dur, "ppt_h", "0", "#ppt_h");
319                    anim_effect(w, ids, e, dur, "fade", false);
320                }
321                "float" | "rise" => {
322                    anim_prop(w, ids, e, dur, "ppt_y", if a.option == "d" { "#ppt_y-.1" } else { "#ppt_y+.1" }, "#ppt_y");
323                    anim_effect(w, ids, e, dur, "fade", false);
324                }
325                other => anim_effect(w, ids, e, dur, filter_for(other, &a.option), false),
326            }
327        }
328        AnimClass::Exit => {
329            match a.effect.as_str() {
330                "disappear" => {}
331                "flyOut" => {
332                    let (fx, fy) = fly_from(&a.option);
333                    anim_prop(w, ids, e, dur, "ppt_x", "#ppt_x", fx);
334                    anim_prop(w, ids, e, dur, "ppt_y", "#ppt_y", fy);
335                }
336                "floatOut" | "sinkDown" => {
337                    let to = if a.effect == "sinkDown" || a.option == "d" { "#ppt_y+.1" } else { "#ppt_y-.1" };
338                    anim_prop(w, ids, e, dur, "ppt_y", "#ppt_y", to);
339                    anim_effect(w, ids, e, dur, "fade", true);
340                }
341                "zoomOut" | "shrinkTurn" | "spinnerOut" | "contract" => {
342                    anim_prop(w, ids, e, dur, "ppt_w", "#ppt_w", "0");
343                    anim_prop(w, ids, e, dur, "ppt_h", "#ppt_h", "0");
344                    anim_effect(w, ids, e, dur, "fade", true);
345                }
346                other => anim_effect(w, ids, e, dur, filter_for(other, &a.option), true),
347            }
348            set_vis(w, ids, e, if a.effect == "disappear" { 0 } else { dur.saturating_sub(1) }, false);
349        }
350        AnimClass::Emphasis => match a.effect.as_str() {
351            "spin" | "teeter" => {
352                let deg = a.amount.unwrap_or(if a.effect == "teeter" { 4.0 } else { 360.0 });
353                let deg = if a.option == "counterClockwise" { -deg.abs() } else { deg };
354                let by = (deg.clamp(-36_000.0, 36_000.0) * 60_000.0).round() as i64;
355                w.open("p:animRot", A::new().a("by", by));
356                cbhvr(w, ids, e, dur, None, &["r"], None);
357                w.close("p:animRot");
358            }
359            "transparency" => {
360                w.open0("p:set");
361                cbhvr(w, ids, e, dur, None, &["style.opacity"], None);
362                w.open0("p:to");
363                w.val("p:strVal", format!("{:.2}", (1.0 - a.amount.unwrap_or(0.5)).clamp(0.0, 1.0)));
364                w.close("p:to");
365                w.close("p:set");
366            }
367            "colorPulse" => {
368                // As PowerPoint writes it: text and fill go to the colour and back (autoRev). The behaviour
369                // duration is the one import reads as the effect duration.
370                let c = a.color.clone().unwrap_or(crate::model::ColorRef::rgb(crate::color::Rgba::rgb(0xC0, 0x00, 0x00)));
371                for (attr, child) in [("style.color", true), ("fillcolor", false)] {
372                    w.open("p:animClr", A::new().a("clrSpc", "rgb").a("dir", "cw"));
373                    w.open("p:cBhvr", A::new().o("override", child.then_some("childStyle")));
374                    w.empty("p:cTn", A::new().a("id", ids.next()).a("dur", dur.max(1)).a("autoRev", "1").a("fill", "remove"));
375                    tgt(w, e.spid, e.paragraph);
376                    w.open0("p:attrNameLst");
377                    w.elt("p:attrName", attr);
378                    w.close("p:attrNameLst");
379                    w.close("p:cBhvr");
380                    w.open0("p:to");
381                    super::dml::color(w, &c);
382                    w.close("p:to");
383                    w.close("p:animClr");
384                }
385            }
386            // As PowerPoint writes them: a relative HSL shift of text, fill and line colour.
387            "darken" | "desaturate" => {
388                let (s, l) = if a.effect == "darken" { (0, -25_098) } else { (-70_588, 0) };
389                for attr in ["style.color", "fillcolor", "stroke.color"] {
390                    w.open("p:animClr", A::new().a("clrSpc", "hsl").a("dir", "cw"));
391                    cbhvr(w, ids, e, dur, None, &[attr], None);
392                    w.open0("p:by");
393                    w.empty("p:hsl", A::new().a("h", 0).a("s", s).a("l", l));
394                    w.close("p:by");
395                    w.close("p:animClr");
396                }
397            }
398            "fillColor" | "objectColor" | "fontColor" | "lineColor" | "complementaryColor" | "lighten" => {
399                let attr = match a.effect.as_str() {
400                    "fontColor" => "style.color",
401                    "lineColor" => "stroke.color",
402                    _ => "fillcolor",
403                };
404                let c = a.color.clone().unwrap_or(crate::model::ColorRef::scheme(crate::color::SchemeSlot::Accent2));
405                w.open("p:animClr", A::new().a("clrSpc", "rgb").a("dir", "cw"));
406                cbhvr(w, ids, e, dur, None, &[attr], None);
407                w.open0("p:to");
408                super::dml::color(w, &c);
409                w.close("p:to");
410                w.close("p:animClr");
411            }
412            _ => {
413                let s = a.amount.filter(|v| v.is_finite() && *v > 0.0).unwrap_or(if a.effect == "growShrink" { 1.5 } else { 1.1 });
414                let (sx, sy) = match a.option.as_str() {
415                    "horz" => (s, 1.0),
416                    "vert" => (1.0, s),
417                    _ => (s, s),
418                };
419                w.open0("p:animScale");
420                cbhvr(w, ids, e, dur, None, &[], None);
421                w.empty(
422                    "p:by",
423                    A::new().a("x", (sx.clamp(0.01, 100.0) * 100_000.0).round() as i64).a("y", (sy.clamp(0.01, 100.0) * 100_000.0).round() as i64),
424                );
425                w.close("p:animScale");
426            }
427        },
428        AnimClass::Path => {
429            let path = a
430                .path
431                .clone()
432                .filter(|p| !p.trim().is_empty() && p.chars().all(|c| c.is_ascii() && c != '"' && c != '<' && c != '&'))
433                .unwrap_or_else(|| {
434                    match a.option.as_str() {
435                        "up" => "M 0 0 L 0 -0.25 E",
436                        "right" => "M 0 0 L 0.25 0 E",
437                        "left" => "M 0 0 L -0.25 0 E",
438                        _ => "M 0 0 L 0 0.25 E",
439                    }
440                    .to_string()
441                });
442            w.open("p:animMotion", A::new().a("origin", "layout").a("path", path).a("pathEditMode", "relative").a("ptsTypes", ""));
443            cbhvr(w, ids, e, dur, None, &["ppt_x", "ppt_y"], None);
444            w.close("p:animMotion");
445        }
446        AnimClass::Media => {
447            let cmd = match a.effect.as_str() {
448                "pause" => "togglePause",
449                "stop" => "stop",
450                _ => "playFrom(0.0)",
451            };
452            w.open("p:cmd", A::new().a("type", "call").a("cmd", cmd));
453            cbhvr(w, ids, e, 1, None, &[], None);
454            w.close("p:cmd");
455        }
456    }
457    let _ = x;
458}

460fn effect_par(w: &mut W, x: &mut Exp, ids: &mut Ids, e: &Eff, first_in_click: bool) {
461    let a = e.a;
462    let node = match e.start {
463        AnimStart::OnClick => "clickEffect",
464        AnimStart::WithPrevious => "withEffect",
465        AnimStart::AfterPrevious => "afterEffect",
466    };
467    let _ = first_in_click;
468    let preset = crate::pptx::maps::preset_for(a);
469    let sub = if preset == 47 {
470        0
471    } else {
472        crate::pptx::maps::subtype_from_option(&a.effect, &a.option).or(a.preset_subtype.filter(|_| a.preset_id == Some(preset))).unwrap_or(0)
473    };
474    let repeat = match a.repeat {
475        u32::MAX => Some("indefinite".to_string()),
476        0 | 1 => None,
477        n => Some((n.min(10_000) * 1000).to_string()),
478    };
479    let accel = (a.smooth_start > 0.0).then(|| (a.smooth_start.clamp(0.0, 1.0) * 100_000.0).round() as i64);
480    let decel = (a.smooth_end > 0.0).then(|| (a.smooth_end.clamp(0.0, 1.0) * 100_000.0).round() as i64);
481    // `bounceEnd` is a positive percentage (100000 = 100%), same scale as accel/decel. The model stores 0–1.
482    let bounce =
483        (a.bounce_end.is_finite() && a.bounce_end > 0.0).then(|| (a.bounce_end.clamp(0.0, 1.0) * 100_000.0).round() as i64).filter(|n| *n > 0);
484    w.open0("p:par");
485    w.open(
486        "p:cTn",
487        A::new()
488            .a("id", ids.next())
489            .a("presetID", preset)
490            .a("presetClass", a.class.xml())
491            .a("presetSubtype", sub)
492            .o("repeatCount", repeat)
493            .o("accel", accel)
494            .o("decel", decel)
495            .o("bounceEnd", bounce)
496            .t("autoRev", a.auto_reverse)
497            .a("fill", if a.rewind { "remove" } else { "hold" })
498            .a("grpId", e.grp)
499            .a("nodeType", node),
500    );
501    w.open0("p:stCondLst");
502    w.empty("p:cond", A::new().a("delay", a.delay_ms.min(3_600_000)));
503    w.close("p:stCondLst");
504    w.open0("p:childTnLst");
505    behaviors(w, x, ids, e);
506    w.close("p:childTnLst");
507    w.close("p:cTn");
508    w.close("p:par");
509}

511fn end_time(e: &Eff) -> u32 {
512    let rep = match e.a.repeat {
513        0 | 1 => 1,
514        u32::MAX => 1,
515        n => n.min(100),
516    };
517    e.a.delay_ms.saturating_add(e.a.duration_ms.saturating_mul(rep)).min(3_600_000)
518}

520/// click groups → after-groups → effects
521type Groups<'a> = Vec<(bool, Vec<Vec<Eff<'a>>>)>;
522
523fn group<'a>(effects: Vec<Eff<'a>>) -> Groups<'a> {
524    let mut clicks: Groups = vec![];
525    for e in effects {
526        match e.start {
527            AnimStart::OnClick => clicks.push((false, vec![vec![e]])),
528            AnimStart::WithPrevious => match clicks.last_mut().and_then(|c| c.1.last_mut()) {
529                Some(g) => g.push(e),
530                None => clicks.push((true, vec![vec![e]])),
531            },
532            AnimStart::AfterPrevious => match clicks.last_mut() {
533                Some(c) => c.1.push(vec![e]),
534                None => clicks.push((true, vec![vec![e]])),
535            },
536        }
537    }
538    clicks
539}
540
541fn click_pars(w: &mut W, x: &mut Exp, ids: &mut Ids, clicks: &Groups, interactive: bool) {
542    for (ci, (auto, afters)) in clicks.iter().enumerate() {
543        w.open0("p:par");
544        w.open("p:cTn", A::new().a("id", ids.next()).a("fill", "hold"));
545        w.open0("p:stCondLst");
546        if interactive && ci == 0 {
547            w.empty("p:cond", A::new().a("delay", 0));
548        } else {
549            w.empty("p:cond", A::new().a("delay", "indefinite"));
550            if *auto {
551                w.open("p:cond", A::new().a("evt", "onBegin").a("delay", 0));
552                w.val("p:tn", 2);
553                w.close("p:cond");
554            }
555        }
556        w.close("p:stCondLst");
557        w.open0("p:childTnLst");
558        let mut t = 0u32;
559        for g in afters {
560            w.open0("p:par");
561            w.open("p:cTn", A::new().a("id", ids.next()).a("fill", "hold"));
562            w.open0("p:stCondLst");
563            w.empty("p:cond", A::new().a("delay", t));
564            w.close("p:stCondLst");
565            w.open0("p:childTnLst");
566            for (i, e) in g.iter().enumerate() {
567                effect_par(w, x, ids, e, i == 0);
568            }
569            w.close("p:childTnLst");
570            w.close("p:cTn");
571            w.close("p:par");
572            t = t.saturating_add(g.iter().map(end_time).max().unwrap_or(0));
573        }
574        w.close("p:childTnLst");
575        w.close("p:cTn");
576        w.close("p:par");
577    }
578}

580/// Most effects one slide's timing writes; animations past it are left out.
581const MAX_EFFECTS: usize = 100_000;
582
583/// The shapes of the tree by id, the first in document order for a repeated id (as `find_shape`
584/// finds it), so each animation resolves its shape without a tree walk.
585fn shape_index<'a>(shapes: &'a [Shape], depth: usize, out: &mut std::collections::HashMap<ShapeId, &'a Shape>) {
586    if depth > 64 {
587        return;
588    }
589    for s in shapes {
590        out.entry(s.id).or_insert(s);
591        shape_index(s.children(), depth + 1, out);
592    }
593}
594
595/// Write `p:timing` for a slide's animations.
596pub fn timing(w: &mut W, x: &mut Exp, anims: &[Animation], shapes: &[Shape], idmap: &IdMap) {
597    let mut grp_count: std::collections::HashMap<u32, u32> = Default::default();
598    let mut main: Vec<Eff> = vec![];
599    let mut triggered: Vec<(u32, Vec<Eff>)> = vec![];
600    let mut bld: Vec<(u32, u32, bool)> = vec![];
601    // The last paragraph effect written: (shape, class, trigger, grpId).
602    let mut prev_para: Option<(ShapeId, AnimClass, Option<ShapeId>, u32)> = None;
603    let mut by_id = std::collections::HashMap::new();
604    shape_index(shapes, 0, &mut by_id);
605    let mut roles_of: std::collections::HashMap<ShapeId, Vec<BuildRole>> = Default::default();
606    let mut effect_count = 0usize;
607    let mut warned = false;
608    for a in anims.iter().take(10_000) {
609        if effect_count >= MAX_EFFECTS {
610            if !warned {
611                log::warn!("pptx: more than {MAX_EFFECTS} animation effects; the rest are left out");
612            }
613            break;
614        }
615        let Some(spid) = idmap.get(a.shape) else {
616            log::warn!("pptx: animation on missing shape {} skipped", a.shape);
617            continue;
618        };
619        let Some(&shape) = by_id.get(&a.shape) else { continue };
620        let text_sp = matches!(shape.kind, ShapeKind::Shape) && shape.text.is_some();
621        let is_text_sp = matches!(shape.kind, ShapeKind::Shape) && shape.text.as_ref().is_some_and(|t| !t.is_empty());
622        let text = shape.text.as_ref().filter(|_| a.text_build == TextBuild::ByParagraph && is_text_sp);
623        let no_roles = Vec::new();
624        let roles = match text {
625            Some(t) => roles_of.entry(a.shape).or_insert_with(|| paragraph_build_roles(t)),
626            None => &no_roles,
627        };
628        let by_para = roles.contains(&BuildRole::Top);
629        // PowerPoint keeps a shape's consecutive paragraph effects in one build (`grpId`) marked
630        // `build="p"`; the reader turns them back into a by-paragraph build only if that plays them
631        // the same way.
632        let para_effect = !by_para && text_sp && a.paragraph.is_some();
633        let joins = para_effect && prev_para.is_some_and(|(s, c, t, _)| s == a.shape && c == a.class && t == a.trigger);
634        let grp = match prev_para {
635            Some((.., g)) if joins => g,
636            _ => {
637                let c = grp_count.entry(spid).or_insert(0);
638                let g = *c;
639                *c += 1;
640                g
641            }
642        };
643        prev_para = para_effect.then_some((a.shape, a.class, a.trigger, grp));
644        let mut effs = vec![];
645        if by_para {
646            // One effect per paragraph the build animates on its own, deeper ones with their
647            // parent; indices in the file count empty paragraphs too.
648            let mut tops = 0;
649            for (i, role) in roles.iter().enumerate() {
650                let start = match role {
651                    BuildRole::Top => {
652                        tops += 1;
653                        if tops == 1 {
654                            a.start
655                        } else if a.start == AnimStart::OnClick {
656                            AnimStart::OnClick
657                        } else {
658                            AnimStart::AfterPrevious
659                        }
660                    }
661                    BuildRole::Member(_) if text.and_then(|t| t.paragraphs.get(i)).is_some_and(build_filled) => AnimStart::WithPrevious,
662                    _ => continue,
663                };
664                let Ok(pi) = u32::try_from(i) else { break };
665                if effect_count + effs.len() >= MAX_EFFECTS {
666                    if !warned {
667                        log::warn!("pptx: more than {MAX_EFFECTS} animation effects; the rest are left out");
668                        warned = true;
669                    }
670                    break;
671                }
672                effs.push(Eff { a, spid, start, paragraph: Some(pi), grp });
673            }
674        } else {
675            effs.push(Eff { a, spid, start: a.start, paragraph: a.paragraph, grp });
676        }
677        effect_count += effs.len();
678        if text_sp && !joins {
679            bld.push((spid, grp, by_para || para_effect));
680        }
681        match a.trigger.and_then(|t| idmap.get(t)) {
682            Some(t) => match triggered.iter_mut().find(|(s, _)| *s == t) {
683                Some((_, v)) => v.extend(effs),
684                None => triggered.push((t, effs)),
685            },
686            None => main.extend(effs),
687        }
688    }
689    if main.is_empty() && triggered.is_empty() {
690        return;
691    }
692    let mut ids = Ids(0);
693    w.open0("p:timing");
694    w.open0("p:tnLst");
695    w.open0("p:par");
696    w.open("p:cTn", A::new().a("id", ids.next()).a("dur", "indefinite").a("restart", "never").a("nodeType", "tmRoot"));
697    w.open0("p:childTnLst");
698    if !main.is_empty() {
699        let clicks = group(main);
700        w.open("p:seq", A::new().a("concurrent", 1).a("nextAc", "seek"));
701        w.open("p:cTn", A::new().a("id", ids.next()).a("dur", "indefinite").a("nodeType", "mainSeq"));
702        w.open0("p:childTnLst");
703        click_pars(w, x, &mut ids, &clicks, false);
704        w.close("p:childTnLst");
705        w.close("p:cTn");
706        w.open0("p:prevCondLst");
707        w.open("p:cond", A::new().a("evt", "onPrev").a("delay", 0));
708        w.open0("p:tgtEl");
709        w.empty0("p:sldTgt");
710        w.close("p:tgtEl");
711        w.close("p:cond");
712        w.close("p:prevCondLst");
713        w.open0("p:nextCondLst");
714        w.open("p:cond", A::new().a("evt", "onNext").a("delay", 0));
715        w.open0("p:tgtEl");
716        w.empty0("p:sldTgt");
717        w.close("p:tgtEl");
718        w.close("p:cond");
719        w.close("p:nextCondLst");
720        w.close("p:seq");
721    }
722    for (trig, effs) in triggered {
723        // The first step starts with the button's click. Later On Click effects must remain
724        // separate steps; With/After Previous effects still belong to their preceding step.
725        let effs: Vec<Eff> =
726            effs.into_iter().enumerate().map(|(i, e)| Eff { start: if i == 0 { AnimStart::OnClick } else { e.start }, ..e }).collect();
727        let clicks = group(effs);
728        let trig_cond = |w: &mut W| {
729            w.open("p:cond", A::new().a("evt", "onClick").a("delay", 0));
730            w.open0("p:tgtEl");
731            w.empty("p:spTgt", A::new().a("spid", trig));
732            w.close("p:tgtEl");
733            w.close("p:cond");
734        };
735        w.open("p:seq", A::new().a("concurrent", 1).a("nextAc", "seek"));
736        w.open(
737            "p:cTn",
738            A::new()
739                .a("id", ids.next())
740                .a("restart", "whenNotActive")
741                .a("fill", "hold")
742                .a("evtFilter", "cancelBubble")
743                .a("nodeType", "interactiveSeq"),
744        );
745        w.open0("p:stCondLst");
746        trig_cond(w);
747        w.close("p:stCondLst");
748        w.open("p:endSync", A::new().a("evt", "end").a("delay", 0));
749        w.empty("p:rtn", A::new().a("val", "all"));
750        w.close("p:endSync");
751        w.open0("p:childTnLst");
752        click_pars(w, x, &mut ids, &clicks, true);
753        w.close("p:childTnLst");
754        w.close("p:cTn");
755        w.open0("p:nextCondLst");
756        trig_cond(w);
757        w.close("p:nextCondLst");
758        w.close("p:seq");
759    }
760    w.close("p:childTnLst");
761    w.close("p:cTn");
762    w.close("p:par");
763    w.close("p:tnLst");
764    if !bld.is_empty() {
765        w.open0("p:bldLst");
766        for (spid, grp, build_p) in bld {
767            w.empty("p:bldP", A::new().a("spid", spid).a("grpId", grp).o("build", build_p.then_some("p")).t("animBg", !build_p));
768        }
769        w.close("p:bldLst");
770    }
771    w.close("p:timing");
772}
