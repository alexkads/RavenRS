13/// Column letter for a 0-based index (A, B, … Z, AA…).
14fn col(i: usize) -> String {
15    let mut n = i + 1;
16    let mut s = String::new();
17    while n > 0 && s.len() < 4 {
18        let r = (n - 1) % 26;
19        s.insert(0, (b'A' + r as u8) as char);
20        n = (n - 1) / 26;
21    }
22    s
23}

29fn simple_fill(w: &mut W, f: &Fill) {
30    match f {
31        Fill::None => w.empty0("a:noFill"),
32        Fill::Solid { color } => super::dml::solid(w, color),
33        Fill::Pattern(p) => {
34            w.open("a:pattFill", A::new().a("prst", if p.preset.is_empty() { "pct5" } else { &p.preset }));
35            w.open0("a:fgClr");
36            super::dml::color(w, &p.fg);
37            w.close("a:fgClr");
38            w.open0("a:bgClr");
39            super::dml::color(w, &p.bg);
40            w.close("a:bgClr");
41            w.close("a:pattFill");
42        }
43        Fill::Gradient(_) | Fill::Picture(_) | Fill::Group | Fill::Background => {
44            if let Fill::Gradient(g) = f
45                && let Some(s) = g.stops.first()
46            {
47                super::dml::solid(w, &s.color);
48            }
49        }
50    }
51}

53fn sp_pr(w: &mut W, fill: Option<&Fill>, line: Option<&crate::model::Line>) {
54    if fill.is_none() && line.is_none() {
55        return;
56    }
57    w.open0("c:spPr");
58    if let Some(f) = fill {
59        simple_fill(w, f);
60    }
61    if let Some(l) = line {
62        w.open("a:ln", A::new().o("w", l.width.map(super::dml::emu_pos)));
63        if let Some(f) = &l.fill {
64            simple_fill(w, f);
65        }
66        w.close("a:ln");
67    }
68    w.close("c:spPr");
69}

84/// `c:txPr` with the chart's text colour (`Chart::text_color`) as the default run properties, on
85/// the axes, the legend, the data labels and the chart itself, as PowerPoint writes it. Nothing
86/// without a colour: the theme's default then applies on reading.
87fn tx_pr(w: &mut W, c: &Chart) {
88    let Some(color) = &c.text_color else { return };
89    w.open0("c:txPr");
90    w.empty0("a:bodyPr");
91    w.empty0("a:lstStyle");
92    w.open0("a:p");
93    w.open0("a:pPr");
94    def_rpr(w, Some(color));
95    w.close("a:pPr");
96    w.empty("a:endParaRPr", A::new().a("lang", "en-US"));
97    w.close("a:p");
98    w.close("c:txPr");
99}

101/// `c:legendEntry` with the entry's own text size, weight and colour.
102fn legend_entry(w: &mut W, e: &crate::model::chart::LegendEntry) {
103    w.open0("c:legendEntry");
104    w.val("c:idx", e.idx);
105    w.open0("c:txPr");
106    w.empty0("a:bodyPr");
107    w.empty0("a:lstStyle");
108    w.open0("a:p");
109    w.open0("a:pPr");
110    let mut a = A::new();
111    if let Some(sz) = e.size.filter(|v| v.is_finite() && *v > 0.0) {
112        // 1 to 400 pt, as the reader takes it, so a size reads back as written.
113        a = a.a("sz", ((sz * 100.0).round().clamp(100.0, 40_000.0) as u32).to_string());
114    }
115    if let Some(b) = e.bold {
116        a = a.a("b", if b { "1" } else { "0" });
117    }
118    match &e.color {
119        Some(c) => {
120            w.open("a:defRPr", a);
121            super::dml::solid(w, c);
122            w.close("a:defRPr");
123        }
124        None => w.empty("a:defRPr", a),
125    }
126    w.close("a:pPr");
127    w.empty("a:endParaRPr", A::new().a("lang", "en-US"));
128    w.close("a:p");
129    w.close("c:txPr");
130    w.close("c:legendEntry");
131}

133fn rich(w: &mut W, text: &str, color: Option<&crate::model::ColorRef>) {
134    w.open0("c:tx");
135    w.open0("c:rich");
136    w.empty0("a:bodyPr");
137    w.empty0("a:lstStyle");
138    for line in text.split('\n') {
139        w.open0("a:p");
140        if color.is_some() {
141            w.open0("a:pPr");
142            def_rpr(w, color);
143            w.close("a:pPr");
144        }
145        w.open0("a:r");
146        if let Some(c) = color {
147            w.open("a:rPr", A::new().a("lang", "en-US"));
148            super::dml::solid(w, c);
149            w.close("a:rPr");
150        }
151        w.elt("a:t", line);
152        w.close("a:r");
153        w.close("a:p");
154    }
155    w.close("c:rich");
156    w.close("c:tx");
157}

167/// Points one cache writes. Excel and PowerPoint hold 32,000 points per series, so this is far
168/// above any chart they make; a longer vector is cut here with a warning, not refused.
169const MAX_CACHE_POINTS: usize = 1_000_000;

171/// `vals` cut to [`MAX_CACHE_POINTS`].
172fn capped<T>(vals: &[T]) -> &[T] {
173    if vals.len() > MAX_CACHE_POINTS {
174        log::warn!("pptx: chart data cache cut from {} to {MAX_CACHE_POINTS} points", vals.len());
175    }
176    &vals[..vals.len().min(MAX_CACHE_POINTS)]
177}

179fn str_cache(w: &mut W, f: &str, vals: &[String]) {
180    let vals = capped(vals);
181    w.open0("c:strRef");
182    w.elt("c:f", f);
183    w.open0("c:strCache");
184    w.val("c:ptCount", vals.len());
185    for (i, v) in vals.iter().enumerate() {
186        w.open("c:pt", A::new().a("idx", i));
187        w.elt("c:v", v);
188        w.close("c:pt");
189    }
190    w.close("c:strCache");
191    w.close("c:strRef");
192}

194fn num_cache(w: &mut W, f: &str, vals: &[Option<f64>]) {
195    let vals = capped(vals);
196    w.open0("c:numRef");
197    w.elt("c:f", f);
198    w.open0("c:numCache");
199    w.elt("c:formatCode", "General");
200    w.val("c:ptCount", vals.len());
201    for (i, v) in vals.iter().enumerate() {
202        if let Some(v) = v.filter(|v| v.is_finite()) {
203            w.open("c:pt", A::new().a("idx", i));
204            w.elt("c:v", &fmt_num(v));
205            w.close("c:pt");
206        }
207    }
208    w.close("c:numCache");
209    w.close("c:numRef");
210}

254fn error_bars(w: &mut W, e: &ErrorBars, g: Group) {
255    let word = |v: &str, ok: &[&'static str], d: &'static str| ok.iter().copied().find(|o| *o == v).unwrap_or(d);
256    w.open0("c:errBars");
257    if matches!(g, Group::Scatter | Group::Bubble) {
258        w.val("c:errDir", "y");
259    }
260    w.val("c:errBarType", word(&e.bar_type, &["both", "plus", "minus"], "both"));
261    let vt = word(&e.val_type, &["fixedVal", "percentage", "stdDev", "stdErr", "cust"], "fixedVal");
262    w.val("c:errValType", vt);
263    w.val("c:noEndCap", if e.end_cap { 0 } else { 1 });
264    if vt == "cust" {
265        w.open0("c:plus");
266        num_lit(w, &e.plus);
267        w.close("c:plus");
268        w.open0("c:minus");
269        num_lit(w, &e.minus);
270        w.close("c:minus");
271    } else if e.value.is_finite() {
272        w.val("c:val", fmt_num(e.value));
273    }
274    w.close("c:errBars");
275}

277/// Per-point fills of pie, doughnut and bar series.
278fn point_fills(w: &mut W, s: &Series, g: Group) {
279    if !matches!(g, Group::Pie | Group::Doughnut | Group::Bar) {
280        return;
281    }
282    for (idx, f) in s.point_fills.iter().take(1000) {
283        w.open0("c:dPt");
284        w.val("c:idx", idx);
285        if g == Group::Bar {
286            w.val("c:invertIfNegative", 0);
287        }
288        if g != Group::Bar {
289            w.val("c:bubble3D", 0);
290        }
291        sp_pr(w, Some(f), None);
292        w.close("c:dPt");
293    }
294}

299fn series(w: &mut W, c: &Chart, s: &Series, i: usize, kind: ChartType, g: Group) {
300    w.open0("c:ser");
301    w.val("c:idx", i);
302    w.val("c:order", i);
303    w.open0("c:tx");
304    str_cache(w, &format!("Sheet1!${}$1", col(i + 1)), std::slice::from_ref(&s.name));
305    w.close("c:tx");
306    if g == Group::Stock && s.line.is_none() {
307        // A stock chart's series are points joined by the high-low lines, not lines of their own.
308        w.open0("c:spPr");
309        w.open("a:ln", A::new().a("w", 19050));
310        w.empty0("a:noFill");
311        w.close("a:ln");
312        w.close("c:spPr");
313    } else {
314        sp_pr(w, s.fill.as_ref(), s.line.as_ref());
315    }
316    if g == Group::Stock {
317        // High-low-close: the close is a dash on the high-low line; with up/down bars (or a set
318        // marker) the series keep their own.
319        let close = !c.up_down_bars && i + 1 == c.series.len() && c.series.len() == 3;
320        let sym = s.marker.as_deref().filter(|m| !m.is_empty() && m.chars().all(|ch| ch.is_ascii_alphanumeric())).unwrap_or(if close {
321            "dash"
322        } else {
323            "none"
324        });
325        w.open0("c:marker");
326        w.val("c:symbol", sym);
327        if sym != "none" {
328            w.val("c:size", 7);
329        }
330        w.close("c:marker");
331    }
332    if g == Group::Bar {
333        w.val("c:invertIfNegative", 0);
334    }
335    if matches!(g, Group::Line | Group::Scatter | Group::Radar) {
336        let sym = match (&s.marker, kind) {
337            (Some(m), _) if !m.is_empty() => Some(m.as_str()),
338            (_, ChartType::LineMarkers) => None,
339            (_, ChartType::Scatter) => None,
340            _ => Some("none"),
341        };
342        if let Some(sym) = sym.filter(|s| s.chars().all(|c| c.is_ascii_alphanumeric())) {
343            w.open0("c:marker");
344            w.val("c:symbol", sym);
345            w.close("c:marker");
346        }
347    }
348    point_fills(w, s, g);
349    if let Some(d) = s.data_labels {
350        data_labels(w, c, d);
351    }
352    // After labels, before the category and value caches (ECMA-376 series content order).
353    if matches!(g, Group::Bar | Group::Line | Group::Area | Group::Scatter) {
354        for t in s.trendlines.iter().take(16) {
355            trendline(w, t);
356        }
357    }
358    let n = c.categories.len().max(s.values.len()).min(MAX_CACHE_POINTS);
359    let last = n + 1;
360    if let Some(e) = &s.error_bars {
361        error_bars(w, e, g);
362    }
363    if matches!(g, Group::Scatter | Group::Bubble) {
364        let xs: Vec<Option<f64>> = if s.x_values.is_empty() { (1..=s.values.len()).map(|v| Some(v as f64)).collect() } else { s.x_values.clone() };
365        w.open0("c:xVal");
366        num_cache(w, &format!("Sheet1!$A$2:$A${last}"), &xs);
367        w.close("c:xVal");
368        w.open0("c:yVal");
369        num_cache(w, &format!("Sheet1!${0}$2:${0}${last}", col(i + 1)), &s.values);
370        w.close("c:yVal");
371        if g == Group::Bubble {
372            let sizes: Vec<Option<f64>> = if s.sizes.is_empty() { s.values.iter().map(|_| Some(1.0)).collect() } else { s.sizes.clone() };
373            w.open0("c:bubbleSize");
374            num_cache(w, &format!("Sheet1!${0}$2:${0}${last}", col(i + 2)), &sizes);
375            w.close("c:bubbleSize");
376            w.val("c:bubble3D", 0);
377        } else {
378            w.val("c:smooth", if s.smooth { 1 } else { 0 });
379        }
380    } else {
381        w.open0("c:cat");
382        str_cache(w, &format!("Sheet1!$A$2:$A${last}"), &c.categories);
383        w.close("c:cat");
384        w.open0("c:val");
385        num_cache(w, &format!("Sheet1!${0}$2:${0}${last}", col(i + 1)), &s.values);
386        w.close("c:val");
387        if g == Group::Line {
388            w.val("c:smooth", if s.smooth { 1 } else { 0 });
389        }
390    }
391    w.close("c:ser");
392}

398fn trend_name(s: &str) -> Option<String> {
399    let cleaned: String =
400        s.chars().filter(|c| matches!(c, '\t' | '\n' | '\r') || (*c >= ' ' && *c != '\u{FFFE}' && *c != '\u{FFFF}')).take(255).collect();
401    let cleaned = cleaned.trim();
402    if cleaned.is_empty() { None } else { Some(cleaned.to_string()) }

405fn trendline(w: &mut W, t: &Trendline) {
406    w.open0("c:trendline");
407    if let Some(name) = t.name.as_deref().and_then(trend_name) {
408        w.elt("c:name", &name);
409    }
410    w.val("c:trendlineType", t.kind.xml());
411    if t.kind == TrendlineType::Poly
412        && let Some(o) = t.order
413    {
414        w.val("c:order", o.clamp(2, 6));
415    }
416    if t.kind == TrendlineType::MovingAvg
417        && let Some(p) = t.period
418    {
419        w.val("c:period", p.clamp(2, 255));
420    }
421    if let Some(v) = trend_num(t.forward) {
422        w.val("c:forward", fmt_num(v));
423    }
424    if let Some(v) = trend_num(t.backward) {
425        w.val("c:backward", fmt_num(v));
426    }
427    if let Some(v) = trend_num(t.intercept) {
428        w.val("c:intercept", fmt_num(v));
429    }
430    if t.disp_r_sqr {
431        w.val("c:dispRSqr", 1);
432    }
433    if t.disp_eq {
434        w.val("c:dispEq", 1);
435    }
436    w.close("c:trendline");
437}

458fn axes(w: &mut W, c: &Chart, g: Group) {
459    let horizontal = c.kind.horizontal() && g == Group::Bar;
460    let (cat_pos, val_pos) = if horizontal { ("l", "b") } else { ("b", "l") };
461    if g == Group::Scatter || g == Group::Bubble {
462        w.open0("c:valAx");
463        w.val("c:axId", AX_CAT);
464        w.open0("c:scaling");
465        w.val("c:orientation", "minMax");
466        w.close("c:scaling");
467        w.val("c:delete", 0);
468        w.val("c:axPos", "b");
469        if let Some(t) = &c.axis_titles.0 {
470            title(w, t, c);
471        }
472        w.empty("c:numFmt", A::new().a("formatCode", "General").a("sourceLinked", 1));
473        w.val("c:majorTickMark", "out");
474        w.val("c:minorTickMark", "none");
475        w.val("c:tickLblPos", "nextTo");
476        tx_pr(w, c);
477        w.val("c:crossAx", AX_VAL);
478        w.val("c:crosses", "autoZero");
479        w.val("c:crossBetween", "midCat");
480        w.close("c:valAx");
481    } else {
482        w.open0("c:catAx");
483        w.val("c:axId", AX_CAT);
484        w.open0("c:scaling");
485        w.val("c:orientation", if c.cat_reversed { "maxMin" } else { "minMax" });
486        w.close("c:scaling");
487        w.val("c:delete", 0);
488        w.val("c:axPos", cat_pos);
489        if let Some(t) = &c.axis_titles.0 {
490            title(w, t, c);
491        }
492        w.empty("c:numFmt", A::new().a("formatCode", "General").a("sourceLinked", 1));
493        w.val("c:majorTickMark", "none");
494        w.val("c:minorTickMark", "none");
495        w.val("c:tickLblPos", "nextTo");
496        tx_pr(w, c);
497        w.val("c:crossAx", AX_VAL);
498        w.val("c:crosses", "autoZero");
499        w.val("c:auto", 1);
500        w.val("c:lblAlgn", "ctr");
501        w.val("c:lblOffset", 100);
502        w.val("c:noMultiLvlLbl", 0);
503        w.close("c:catAx");
504    }
505    w.open0("c:valAx");
506    w.val("c:axId", AX_VAL);
507    w.open0("c:scaling");
508    if let Some(b) = c.y_log_base {
509        w.val("c:logBase", fmt_num(b));
510    }
511    w.val("c:orientation", "minMax");
512    if let Some(m) = c.y_max.filter(|v| v.is_finite()) {
513        w.val("c:max", fmt_num(m));
514    }
515    if let Some(m) = c.y_min.filter(|v| v.is_finite()) {
516        w.val("c:min", fmt_num(m));
517    }
518    w.close("c:scaling");
519    w.val("c:delete", 0);
520    w.val("c:axPos", val_pos);
521    if c.gridlines {
522        w.empty0("c:majorGridlines");
523    }
524    if c.minor_gridlines {
525        w.empty0("c:minorGridlines");
526    }
527    if let Some(t) = &c.axis_titles.1 {
528        title(w, t, c);
529    }
530    match c.y_format.as_deref().filter(|f| !f.is_empty()) {
531        Some(f) => w.empty("c:numFmt", A::new().a("formatCode", f).a("sourceLinked", 0)),
532        None => w.empty("c:numFmt", A::new().a("formatCode", if c.kind.percent() { "0%" } else { "General" }).a("sourceLinked", 1)),
533    }
534    w.val("c:majorTickMark", "none");
535    w.val("c:minorTickMark", "none");
536    w.val("c:tickLblPos", "nextTo");
537    tx_pr(w, c);
538    w.val("c:crossAx", AX_CAT);
539    w.val("c:crosses", "autoZero");
540    w.val("c:crossBetween", if g == Group::Area || g == Group::Scatter { "midCat" } else { "between" });
541    w.close("c:valAx");
542}

544/// Points `series` writes into its caches for group `g`, synthesized vectors included.
545fn series_points(c: &Chart, s: &Series, g: Group) -> usize {
546    let v = s.values.len();
547    if matches!(g, Group::Scatter | Group::Bubble) {
548        let xs = if s.x_values.is_empty() { v } else { s.x_values.len() };
549        let sizes = if g != Group::Bubble {
550            0
551        } else if s.sizes.is_empty() {
552            v
553        } else {
554            s.sizes.len()
555        };
556        xs.saturating_add(v).saturating_add(sizes)
557    } else {
558        c.categories.len().saturating_add(v)
559    }
560}

562fn plot_group(w: &mut W, c: &Chart, kind: ChartType, list: &[(usize, &Series)]) {
563    let g = group_of(kind);
564    let grouping = if kind.percent() {
565        "percentStacked"
566    } else if kind.stacked() {
567        "stacked"
568    } else if matches!(g, Group::Bar) {
569        "clustered"
570    } else {
571        "standard"
572    };
573    let vary = if c.vary_colors { 1 } else { 0 };
574    let tag = match g {
575        Group::Bar => "c:barChart",
576        Group::Line => "c:lineChart",
577        Group::Area => "c:areaChart",
578        Group::Pie => "c:pieChart",
579        Group::Doughnut => "c:doughnutChart",
580        Group::Scatter => "c:scatterChart",
581        Group::Radar => "c:radarChart",
582        Group::Bubble => "c:bubbleChart",
583        Group::Stock => "c:stockChart",
584    };
585    w.open0(tag);
586    match g {
587        Group::Bar => {
588            w.val("c:barDir", if kind.horizontal() { "bar" } else { "col" });
589            w.val("c:grouping", grouping);
590        }
591        Group::Line | Group::Area => w.val("c:grouping", grouping),
592        Group::Scatter => w.val("c:scatterStyle", "lineMarker"),
593        Group::Radar => w.val("c:radarStyle", if kind == ChartType::FilledRadar { "filled" } else { "marker" }),
594        _ => {}
595    }
596    // `c:stockChart` has no `c:varyColors`.
597    if g != Group::Stock {
598        w.val("c:varyColors", vary);
599    }
600    // Each series repeats the categories in its own cache, so the output grows with series times
601    // categories: stop adding series once a group's caches pass the budget (the first is always written).
602    // The first series is always written; each of its caches is cut at MAX_CACHE_POINTS.
603    let mut left = MAX_EXPORT_POINTS;
604    for (n, (i, s)) in list.iter().enumerate() {
605        let cost = series_points(c, s, g);
606        if n > 0 && cost > left {
607            log::warn!("pptx: chart export stopped after {n} series; the data caches passed {MAX_EXPORT_POINTS} points");
608            break;
609        }
610        left = left.saturating_sub(cost);
611        series(w, c, s, *i, kind, g);
612    }
613    if c.data_labels {
614        data_labels(w, c, true);
615    }
616    match g {
617        Group::Bar => {
618            w.val("c:gapWidth", (c.gap_width.clamp(0.0, 5.0) * 100.0).round() as i64);
619            let ov = if kind.stacked() { 100 } else { (c.overlap.clamp(-1.0, 1.0) * 100.0).round() as i64 };
620            if ov != 0 {
621                w.val("c:overlap", ov);
622            }
623        }
624        Group::Line => w.val("c:marker", 1),
625        Group::Stock => {
626            w.open0("c:hiLowLines");
627            w.open0("c:spPr");
628            w.open("a:ln", A::new().a("w", 9525));
629            w.open0("a:solidFill");
630            w.empty("a:schemeClr", A::new().a("val", "tx1"));
631            w.close("a:solidFill");
632            w.close("a:ln");
633            w.close("c:spPr");
634            w.close("c:hiLowLines");
635            if c.up_down_bars {
636                w.open0("c:upDownBars");
637                w.val("c:gapWidth", 150);
638                w.empty0("c:upBars");
639                w.empty0("c:downBars");
640                w.close("c:upDownBars");
641            }
642        }
643        Group::Pie => w.val("c:firstSliceAng", 0),
644        Group::Doughnut => {
645            w.val("c:firstSliceAng", 0);
646            w.val("c:holeSize", (c.hole_size.clamp(0.1, 0.9) * 100.0).round() as i64);
647        }
648        _ => {}
649    }
650    if !matches!(g, Group::Pie | Group::Doughnut) {
651        w.val("c:axId", AX_CAT);
652        w.val("c:axId", AX_VAL);
653    }
654    w.close(tag);
655}

669fn kept_chartex(c: &Chart) -> Option<Vec<u8>> {
670    let raw = c.raw.as_ref()?;
671    let doc = crate::pptx::xml::parse(raw.as_bytes()).ok()?;
672    let mut root = doc.root.clone();
673    crate::pptx::read::resolve_mc(&mut root, 0);
674    let mut back = crate::pptx::read::chartex_from(&root)?;
675    back.raw = c.raw.clone();
676    // The sidecar is not modelled chart state.
677    back.kept = c.kept.clone();
678    if &back != c {
679        return None;
680    }
681    let mut keep = doc.root;
682    // The workbook link stays only when its part is written back.
683    drop_unkept_refs(&mut keep, c);
684    let mut w = W::new();
685    w.raw(&keep.to_xml());
686    Some(w.finish())
687}

689/// The chart part XML, and whether it is the original part (so its relationships still apply).
690pub fn chart_xml(c: &Chart) -> (Vec<u8>, bool) {
691    // A chart too large to model is written back as it was read.
692    if c.verbatim
693        && let Some(raw) = &c.raw
694        && let Ok(doc) = crate::pptx::xml::parse(raw.as_bytes())
695        && doc.root.is("chartSpace")
696    {
697        let mut keep = doc.root;
698        drop_unkept_refs(&mut keep, c);
699        let mut w = W::new();
700        w.raw(&keep.to_xml());
701        return (w.finish(), true);
702    }
703    // Reuse the kept part when the chart is unchanged since it was read.
704    if let Some(raw) = &c.raw
705        && !raw.contains(NS_CX)
706        && let Ok(doc) = crate::pptx::xml::parse(raw.as_bytes())
707        && doc.root.is("chartSpace")
708    {
709        let mut root = doc.root.clone();
710        crate::pptx::read::resolve_mc(&mut root, 0);
711        if let Some(back) = crate::pptx::read::chart_from_el(&root) {
712            let mut me = c.clone();
713            me.raw = None;
714            // The sidecar is not modelled chart state. Comparing it would throw away an unchanged chart.
715            me.kept = Default::default();
716            if back == me {
717                let mut keep = doc.root;
718                // Without the parts those elements point at, PowerPoint repairs the file.
719                drop_unkept_refs(&mut keep, c);
720                let mut w = W::new();
721                w.raw(&keep.to_xml());
722                return (w.finish(), true);
723            }
724        }
725    }
726    let mut w = W::new();
727    w.open("c:chartSpace", A::new().a("xmlns:c", NS_C).a("xmlns:a", NS_A).a("xmlns:r", NS_R));
728    w.val("c:date1904", 0);
729    w.val("c:roundedCorners", 0);
730    w.val("c:style", c.style.clamp(1, 48));
731    w.open0("c:chart");
732    match &c.title {
733        Some(t) => {
734            title(&mut w, t, c);
735            w.val("c:autoTitleDeleted", 0);
736        }
737        None => w.val("c:autoTitleDeleted", 1),
738    }
739    w.open0("c:plotArea");
740    w.empty0("c:layout");
741    let indexed: Vec<(usize, &Series)> = c.series.iter().enumerate().collect();
742    let main = match c.kind {
743        ChartType::Combo => c.series.first().and_then(|s| s.kind).unwrap_or(ChartType::Column),
744        k => k,
745    };
746    if c.kind == ChartType::Combo {
747        // One plot group per series type, sharing the axes.
748        let mut kinds: Vec<ChartType> = vec![];
749        for s in &c.series {
750            let k = s.kind.unwrap_or(main);
751            let k = if matches!(group_of(k), Group::Pie | Group::Doughnut | Group::Scatter | Group::Bubble) { ChartType::Column } else { k };
752            if !kinds.contains(&k) {
753                kinds.push(k);
754            }
755        }
756        for k in kinds {
757            let list: Vec<(usize, &Series)> = indexed
758                .iter()
759                .filter(|(_, s)| {
760                    let sk = s.kind.unwrap_or(main);
761                    let sk =
762                        if matches!(group_of(sk), Group::Pie | Group::Doughnut | Group::Scatter | Group::Bubble) { ChartType::Column } else { sk };
763                    sk == k
764                })
765                .copied()
766                .collect();
767            plot_group(&mut w, c, k, &list);
768        }
769        axes(&mut w, c, Group::Bar);
770    } else {
771        if !matches!(
772            c.kind,
773            ChartType::Column
774                | ChartType::StackedColumn
775                | ChartType::PercentColumn
776                | ChartType::Bar
777                | ChartType::StackedBar
778                | ChartType::PercentBar
779                | ChartType::Line
780                | ChartType::LineMarkers
781                | ChartType::StackedLine
782                | ChartType::Area
783                | ChartType::StackedArea
784                | ChartType::Pie
785                | ChartType::Doughnut
786                | ChartType::Scatter
787                | ChartType::Bubble
788                | ChartType::Radar
789                | ChartType::FilledRadar
790                | ChartType::Stock
791        ) {
792            log::warn!("pptx: {} chart written as a column chart", c.kind.label());
793        }
794        plot_group(&mut w, c, main, &indexed);
795        let g = group_of(main);
796        if !matches!(g, Group::Pie | Group::Doughnut) {
797            axes(&mut w, c, g);
798        }
799    }
800    if let Some(f) = &c.plot_fill {
801        sp_pr(&mut w, Some(f), None);
802    }
803    w.close("c:plotArea");
804    if let Some(pos) = &c.legend {
805        let p = match pos.as_str() {
806            "b" | "t" | "l" | "r" | "tr" => pos.as_str(),
807            _ => "r",
808        };
809        w.open0("c:legend");
810        w.val("c:legendPos", p);
811        for e in &c.legend_entries {
812            legend_entry(&mut w, e);
813        }
814        w.val("c:overlay", u8::from(c.legend_overlay));
815        tx_pr(&mut w, c);
816        w.close("c:legend");
817    }
818    w.val("c:plotVisOnly", 1);
819    w.val("c:dispBlanksAs", "gap");
820    w.close("c:chart");
821    if let Some(f) = &c.chart_fill {
822        sp_pr(&mut w, Some(f), None);
823    }
824    tx_pr(&mut w, c);
825    w.close("c:chartSpace");
826    (w.finish(), false)
827}

829/// The relationship ids of an unchanged chart that [`write_kept`] writes back: external ones, and
830/// internal ones whose part was kept.
831fn kept_rel_ids(c: &Chart) -> std::collections::HashSet<&str> {
832    let parts: std::collections::HashSet<String> =
833        c.kept.parts.iter().filter(|p| kept_payload(&p.name)).map(|p| crate::pptx::opc::norm(&p.name)).collect();
834    c.kept.rels.iter().filter(|r| r.external || parts.contains(&crate::pptx::opc::norm(&r.part))).map(|r| r.id.as_str()).collect()
835}

837/// A kept part that is written as a part of its own (rels parts are rewritten instead).
838fn kept_payload(name: &str) -> bool {
839    !name.to_ascii_lowercase().ends_with(".rels") && crate::pptx::opc::sidecar_part(name)
840}

842/// Drop `c:externalData` / `c:userShapes` (and the `cx:` ones) that point at a relationship that
843/// is not written back: without the part PowerPoint repairs the file.
844fn drop_unkept_refs(root: &mut crate::pptx::xml::El, c: &Chart) {
845    let ids = kept_rel_ids(c);
846    fn rec(e: &mut crate::pptx::xml::El, ids: &std::collections::HashSet<&str>, depth: usize) {
847        if depth > crate::pptx::xml::MAX_DEPTH {
848            return;
849        }
850        e.children.retain(|n| match n {
851            crate::pptx::xml::Node::El(x) if x.is("externalData") || x.is("userShapes") => x.attr("r:id").is_some_and(|id| ids.contains(id)),
852            _ => true,
853        });
854        for ch in e.elements_mut() {
855            rec(ch, ids, depth + 1);
856        }
857    }
858    rec(root, &ids, 0);
859}

861/// A fresh name for a kept part: `ppt/<folder>/keptN.<ext>`. The writer generates no `kept…`
862/// names and `n` is unique within the package, so a kept part never lands on (or duplicates) a
863/// part the writer makes, whatever the source file called it.
864fn kept_name(original: &str, n: u32) -> String {
865    let norm = crate::pptx::opc::norm(original);
866    let area = norm.split('/').nth(1).unwrap_or("embeddings");
867    let area = if matches!(area, "embeddings" | "charts" | "drawings" | "media") { area } else { "embeddings" };
868    let file = crate::pptx::opc::file_of(&norm);
869    let ext = file
870        .rsplit_once('.')
871        .map(|(_, e)| e)
872        .filter(|e| !e.is_empty() && e.len() <= 8 && e.chars().all(|c| c.is_ascii_alphanumeric()))
873        .unwrap_or("bin");
874    format!("ppt/{area}/kept{n}.{ext}")
875}

877/// Write an unchanged chart's own relationships and the parts they reach. Each kept part is
878/// written under a fresh name ([`kept_name`]) and the relationships between kept parts are
879/// rewritten to those names; a relationship to anything that was not kept is dropped.
880/// `n_kept` numbers kept parts across the package.
881pub fn write_kept(pkg: &mut crate::pptx::opc::PackageOut, chart_name: &str, c: &Chart, n_kept: &mut u32) {
882    use std::collections::HashMap;
883
884    use crate::pptx::opc::{RelsOut, norm, relative, rels_path};
885
886    if c.kept.rels.is_empty() {
887        return;
888    }
889    // Original part name (normalized) → the name written.
890    let mut names: HashMap<String, String> = HashMap::new();
891    for p in c.kept.parts.iter().filter(|p| kept_payload(&p.name)) {
892        let key = norm(&p.name);
893        if names.contains_key(&key) {
894            continue;
895        }
896        *n_kept += 1;
897        let new = kept_name(&p.name, *n_kept);
898        let ct = if !p.content_type.is_empty() { p.content_type.clone() } else { crate::pptx::mime_for(&p.name).to_string() };
899        // An override per part: a kept part never sets the package-wide default for its extension.
900        pkg.xml_part(&new, &ct, p.bytes.clone());
901        names.insert(key, new);
902    }
903    let mut rels = RelsOut::default();
904    for r in &c.kept.rels {
905        if r.external {
906            rels.items.push((r.id.clone(), r.rel_type.clone(), r.target.clone(), true));
907        } else if let Some(new) = names.get(&norm(&r.part)) {
908            rels.items.push((r.id.clone(), r.rel_type.clone(), relative(chart_name, new), false));
909        } else {
910            log::warn!("pptx: chart relationship {} has no kept part; dropped", r.id);
911        }
912    }
913    pkg.rels(chart_name, &rels);
914    let mut parts_by_norm = std::collections::HashMap::new();
915    for p in &c.kept.parts {
916        parts_by_norm.entry(norm(&p.name)).or_insert(p);
917    }
918    let mut seen_payloads = std::collections::HashSet::new();
919    // The kept parts' own relationships, pointed at the new names.
920    for p in c.kept.parts.iter().filter(|p| kept_payload(&p.name)) {
921        let norm_name = norm(&p.name);
922        if !seen_payloads.insert(norm_name.clone()) {
923            continue;
924        }
925        let Some(new_source) = names.get(&norm_name) else { continue };
926        let rp = norm(&rels_path(&p.name));
927        let Some(rels_part) = parts_by_norm.get(&rp) else { continue };
928        let mut out = RelsOut::default();
929        for r in crate::pptx::opc::rels_from_bytes(&rels_part.bytes, &p.name).items {
930            if r.external {
931                out.items.push((r.id, r.ty, r.target_raw, true));
932            } else if let Some(new) = names.get(&norm(&r.target)) {
933                out.items.push((r.id, r.ty, relative(new_source, new), false));
934            } else {
935                log::warn!("pptx: relationship {} of kept chart part {} has no kept target; dropped", r.id, p.name);
936            }
937        }
938        pkg.rels(new_source, &out);
939    }
940}

981/// Points one `chartEx` part may carry in all, categories repeated for every series: Excel's 255
982/// series x 32,000 points. Series past it are left out with a warning (the first is always written).
983const MAX_CHARTEX_POINTS: usize = 255 * 32_000;
984
985/// How many of `c`'s series fit [`MAX_CHARTEX_POINTS`] (at least one).
986fn chartex_series_within_budget(c: &Chart) -> usize {
987    let level = |v: &Vec<String>| v.len().min(MAX_CACHE_POINTS);
988    let cats = if c.categories.is_empty() {
989        0
990    } else {
991        level(&c.categories).saturating_add(c.category_levels.iter().map(level).fold(0usize, usize::saturating_add))
992    };
993    let mut left = MAX_CHARTEX_POINTS;
994    for (n, s) in c.series.iter().enumerate() {
995        let cost = cats.saturating_add(s.values.len().min(MAX_CACHE_POINTS));
996        if n > 0 && cost > left {
997            log::warn!("pptx: chart part stopped after {n} series; its data passed {MAX_CHARTEX_POINTS} points");
998            return n;
999        }
1000        left = left.saturating_sub(cost);
1001    }
1002    c.series.len()
1003}

1005/// Writes an Office 2016 chart from the model, as PowerPoint lays the part out: one `cx:data`
1006/// per series (categories and values), then the plot area with each series' layout, the axes
1007/// the layout uses, the title and the legend.
1008pub fn build_chartex(c: &Chart) -> Vec<u8> {
1009    let n_series = chartex_series_within_budget(c);
1010    let mut w = W::new();
1011    w.open("cx:chartSpace", A::new().a("xmlns:a", NS_A).a("xmlns:r", NS_R).a("xmlns:cx", NS_CX));
1012    w.open0("cx:chartData");
1013    for (i, s) in c.series.iter().enumerate().take(n_series) {
1014        w.open("cx:data", A::new().a("id", i));
1015        if !c.categories.is_empty() {
1016            w.open("cx:strDim", A::new().a("type", "cat"));
1017            cx_level(&mut w, &c.categories, None);
1018            for lv in &c.category_levels {
1019                cx_level(&mut w, lv, None);
1020            }
1021            w.close("cx:strDim");
1022        }
1023        let ty = match c.kind {
1024            ChartType::Treemap | ChartType::Sunburst => "size",
1025            ChartType::RegionMap => "colorVal",
1026            _ => "val",
1027        };
1028        w.open("cx:numDim", A::new().a("type", ty));
1029        let vals: Vec<String> = s.values.iter().map(|v| v.map(fmt_num).unwrap_or_default()).collect();
1030        cx_level(&mut w, &vals, Some(s.number_format.as_deref().unwrap_or("General")));
1031        w.close("cx:numDim");
1032        w.close("cx:data");
1033    }
1034    w.close("cx:chartData");
1035    w.open0("cx:chart");
1036    if let Some(t) = &c.title {
1037        w.open("cx:title", A::new().a("pos", "t").a("align", "ctr").a("overlay", "0"));
1038        cx_text(&mut w, t);
1039        w.close("cx:title");
1040    }
1041    w.open0("cx:plotArea");
1042    w.open0("cx:plotAreaRegion");
1043    let layout = match c.kind {
1044        ChartType::Waterfall => "waterfall",
1045        ChartType::Funnel => "funnel",
1046        ChartType::Treemap => "treemap",
1047        ChartType::Sunburst => "sunburst",
1048        ChartType::BoxWhisker => "boxWhisker",
1049        ChartType::RegionMap => "regionMap",
1050        _ => "clusteredColumn",
1051    };
1052    let one = !matches!(c.kind, ChartType::BoxWhisker);
1053    for (i, s) in c.series.iter().enumerate().take(if one { 1 } else { n_series }) {
1054        w.open("cx:series", A::new().a("layoutId", layout).a("uniqueId", format!("{{DC00000{i}-0000-4000-8000-000000000000}}")));
1055        cx_text(&mut w, &s.name);
1056        if c.series_data_labels(s) {
1057            w.open("cx:dataLabels", A::new().a("pos", if c.kind == ChartType::Funnel { "ctr" } else { "outEnd" }));
1058            w.empty("cx:visibility", A::new().a("seriesName", "0").a("categoryName", "0").a("value", "1"));
1059            w.close("cx:dataLabels");
