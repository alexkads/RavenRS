15/// cNvPr ids for one part.
16pub struct IdMap {
17    pub map: HashMap<ShapeId, u32>,
18    next: u32,
19}
20
21impl IdMap {
22    pub fn build(shapes: &[Shape]) -> Self {
23        let mut map = HashMap::new();
24        let mut used = HashSet::new();
25        let mut max = 1u32;
26        walk(shapes, &mut |s, _| {
27            if s.id.0 >= 2 && s.id.0 < i32::MAX as u32 {
28                max = max.max(s.id.0);
29            }
30        });
31        let mut next = max;
32        walk(shapes, &mut |s, _| {
33            let id = if s.id.0 >= 2 && s.id.0 < i32::MAX as u32 && !used.contains(&s.id.0) {
34                s.id.0
35            } else {
36                next = next.saturating_add(1);
37                next
38            };
39            used.insert(id);
40            map.entry(s.id).or_insert(id);
41        });
42        IdMap { map, next }
43    }
44    pub fn get(&self, id: ShapeId) -> Option<u32> {
45        self.map.get(&id).copied()
46    }
47    pub fn fresh(&mut self) -> u32 {
48        self.next = self.next.saturating_add(1);
49        self.next
50    }
51}

53pub struct Ctx<'a> {
54    pub ids: &'a mut IdMap,
55    /// The shape id written for the shape currently being written (duplicate model ids get fresh ones).
56    pub written: HashSet<u32>,
57}
58
59impl Ctx<'_> {
60    fn id_for(&mut self, s: &Shape) -> u32 {
61        let id = self.ids.get(s.id).unwrap_or(0);
62        if id >= 2 && self.written.insert(id) {
63            return id;
64        }
65        let f = self.ids.fresh();
66        self.written.insert(f);
67        f
68    }
69}

75/// Non-adjust shape guides some presets carry in their `avLst` (ECMA-376 default values).
76fn extra_guides(name: &str) -> &'static [(&'static str, i64)] {
77    match name {
78        "star5" | "pentagon" => &[("hf", 105_146), ("vf", 110_557)],
79        "star7" | "heptagon" => &[("hf", 102_572), ("vf", 105_210)],
80        "star6" => &[("hf", 115_470)],
81        "star10" => &[("hf", 105_146)],
82        "hexagon" => &[("vf", 115_470)],
83        "decagon" => &[("vf", 105_146)],
84        _ => &[],
85    }
86}

88pub fn geometry(w: &mut W, g: &Geom) {
89    match g {
90        Geom::Preset { name, adj } => {
91            let name = preset_name(name);
92            w.open("a:prstGeom", A::new().a("prst", name));
93            if adj.is_empty() {
94                w.empty0("a:avLst");
95            } else {
96                // PowerPoint wants the complete guide list once any is given.
97                let defaults = crate::geom::preset::info(name).map(|i| i.defaults).unwrap_or(&[]);
98                let mut vals: Vec<f64> = adj.iter().take(16).copied().collect();
99                while vals.len() < defaults.len() {
100                    vals.push(defaults.get(vals.len()).copied().unwrap_or(0.0));
101                }
102                w.open0("a:avLst");
103                // A lone guide is `adj`, except in these presets, which name it `adj1` (as PowerPoint
104                // writes them); PowerPoint won't open a file that renames it (#214).
105                let adj1 = matches!(name, "bentConnector3" | "curvedConnector3" | "frame" | "mathPlus" | "mathMinus" | "mathMultiply");
106                let single = !adj1 && if defaults.is_empty() { vals.len() == 1 } else { defaults.len() == 1 };
107                for (i, v) in vals.iter().enumerate() {
108                    let n = if single && i == 0 { "adj".to_string() } else { format!("adj{}", i + 1) };
109                    // A near-zero-width connector can need a large ratio for an ordinary-size bend.
110                    // Keep guide literals bounded to the signed integer serialization domain.
111                    let v = if v.is_finite() { v.round().clamp(i64::MIN as f64, i64::MAX as f64) as i64 } else { 0 };
112                    w.empty("a:gd", A::new().a("name", n).a("fmla", format!("val {v}")));
113                }
114                for (n, v) in extra_guides(name) {
115                    w.empty("a:gd", A::new().a("name", *n).a("fmla", format!("val {v}")));
116                }
117                w.close("a:avLst");
118            }
119            w.close("a:prstGeom");
120        }
121        Geom::Custom { paths } => {
122            if paths.is_empty() {
123                w.open("a:prstGeom", A::new().a("prst", "rect"));
124                w.empty0("a:avLst");
125                w.close("a:prstGeom");
126            } else {
127                crate::pptx::custgeom::write_cust_geom(w, paths);
128            }
129        }
130    }
131}

133fn cnvpr(w: &mut W, x: &mut Exp, o: &mut Out, id: u32, s: &Shape) {
134    let a = A::new()
135        .a("id", id)
136        .a("name", if s.name.is_empty() { format!("Shape {id}") } else { s.name.clone() })
137        .o("descr", (!s.descr.is_empty()).then_some(s.descr.as_str()))
138        .t("hidden", s.hidden)
139        .o("title", (!s.title.is_empty()).then_some(s.title.as_str()));
140    if s.click.is_none() && s.hover.is_none() && !s.decorative {
141        w.empty("p:cNvPr", a);
142        return;
143    }
144    w.open("p:cNvPr", a);
145    if let Some(h) = &s.click {
146        dml::hyperlink(w, x, o, "a:hlinkClick", h);
147    }
148    if let Some(h) = &s.hover {
149        dml::hyperlink(w, x, o, "a:hlinkHover", h);
150    }
151    if s.decorative {
152        w.open0("a:extLst");
153        w.open("a:ext", A::new().a("uri", "{C183D7F6-B498-43B3-948B-1728B52AA6E4}"));
154        w.empty("adec:decorative", A::new().a("xmlns:adec", "http://schemas.microsoft.com/office/drawing/2017/decorative").a("val", "1"));
155        w.close("a:ext");
156        w.close("a:extLst");
157    }
158    w.close("p:cNvPr");
159}

161fn nv_pr(w: &mut W, x: &mut Exp, o: &mut Out, s: &Shape) {
162    if s.ph.is_none() && s.tags.is_empty() {
163        w.empty0("p:nvPr");
164        return;
165    }
166    w.open0("p:nvPr");
167    if let Some(ph) = &s.ph {
168        let ty = match ph.kind {
169            PhType::Obj => None,
170            k => Some(k.xml()),
171        };
172        let sz = ph.size.as_deref().filter(|z| matches!(*z, "full" | "half" | "quarter"));
173        w.empty(
174            "p:ph",
175            A::new()
176                .o("type", ty)
177                .o("orient", ph.vertical.then_some("vert"))
178                .o("sz", sz)
179                .o("idx", (ph.idx != 0).then_some(ph.idx))
180                .t("hasCustomPrompt", ph.has_custom_prompt),
181        );
182    }
183    super::cust_data(w, x, o, &s.tags);
184    w.close("p:nvPr");
185}

187/// Locking attributes. Lock writes the full set PowerPoint writes for the element's kind (and
188/// reads back as locked): see `read::shapes::is_locked`.
189fn locks(w: &mut W, tag: &str, s: &Shape, extra: A) {
190    let a = extra.t("noGrp", s.ph.is_some() || s.locked);
191    let a = match tag {
192        "a:grpSpLocks" => a.t("noUngrp", s.locked).t("noRot", s.locked).t("noMove", s.locked).t("noResize", s.locked),
193        _ => {
194            let a = a
195                .t("noRot", s.locked)
196                .t("noMove", s.locked)
197                .t("noResize", s.locked)
198                .t("noEditPoints", s.locked)
199                .t("noAdjustHandles", s.locked)
200                .t("noChangeArrowheads", s.locked)
201                .t("noChangeShapeType", s.locked);
202            if tag == "a:picLocks" { a.t("noCrop", s.locked) } else { a }
203        }
204    };
205    if a.v.is_empty() {
206        return;
207    }
208    w.empty(tag, a);
209}

215fn style_tag(w: &mut W, st: &ShapeStyle, tag: &str) {
216    w.open0(tag);
217    for (tag, (idx, c)) in [("a:lnRef", &st.line_ref), ("a:fillRef", &st.fill_ref), ("a:effectRef", &st.effect_ref)] {
218        w.open(tag, A::new().a("idx", idx));
219        dml::color(w, c);
220        w.close(tag);
221    }
222    let idx = match st.font_ref.0.as_str() {
223        "major" | "minor" | "none" => st.font_ref.0.as_str(),
224        _ => "minor",
225    };
226    match &st.font_ref.1 {
227        Some(c) => {
228            w.open("a:fontRef", A::new().a("idx", idx));
229            dml::color(w, c);
230            w.close("a:fontRef");
231        }
232        None => w.empty("a:fontRef", A::new().a("idx", idx)),
233    }
234    w.close(tag);
235}

242fn sp_pr_tag(w: &mut W, x: &mut Exp, o: &mut Out, s: &Shape, geom: bool, tag: &str) {
243    w.open0(tag);
244    if let Some(xf) = &s.xfrm {
245        dml::xfrm(w, "a:xfrm", xf, None);
246    }
247    if geom && (s.xfrm.is_some() || s.ph.is_none() || !matches!(&s.geom, Geom::Preset { name, adj } if name == "rect" && adj.is_empty())) {
248        geometry(w, &s.geom);
249    }
250    if let Some(f) = &s.fill {
251        dml::fill(w, x, o, f);
252    }
253    if let Some(l) = &s.line {
254        dml::line(w, x, o, "a:ln", l);
255    }
256    if let Some(fx) = &s.effects {
257        dml::effects(w, fx);
258        if let Some(raw) = &fx.raw3d {
259            w.raw(raw);
260        } else if let Some(xml) = fx.bevel.as_deref().and_then(dml::bevel_sp3d) {
261            w.raw(&xml);
262        }
263    }
264    w.close(tag);
265}

277pub fn shape(w: &mut W, x: &mut Exp, o: &mut Out, cx: &mut Ctx, s: &Shape, depth: usize) {
278    match &s.kind {
279        ShapeKind::Shape => {
280            if s.text.as_ref().is_some_and(dml::has_math) {
281                // Equations: OMML for readers that know it, plain text otherwise.
282                w.open("mc:AlternateContent", A::new().a("xmlns:mc", NS_MC));
283                w.open("mc:Choice", A::new().a("xmlns:a14", crate::pptx::opc::NS_A14).a("Requires", "a14"));
284                let id = cx.id_for(s);
285                sp(w, x, o, id, s, true);
286                w.close("mc:Choice");
287                w.open0("mc:Fallback");
288                sp(w, x, o, id, s, false);
289                w.close("mc:Fallback");
290                w.close("mc:AlternateContent");
291            } else {
292                let id = cx.id_for(s);
293                sp(w, x, o, id, s, false);
294            }
295        }
296        ShapeKind::Picture { fill } => {
297            let id = cx.id_for(s);
298            w.open0("p:pic");
299            w.open0("p:nvPicPr");
300            cnvpr(w, x, o, id, s);
301            w.open0("p:cNvPicPr");
302            locks(w, "a:picLocks", s, A::new().a("noChangeAspect", "1"));
303            w.close("p:cNvPicPr");
304            nv_pr(w, x, o, s);
305            w.close("p:nvPicPr");
306            dml::blip_fill(w, x, o, "p:blipFill", fill);
307            let mut s2 = s.clone();
308            // Group fill is a separate backing behind transparent image pixels.
309            if matches!(s2.fill, Some(Fill::Picture(_) | Fill::Background)) {
310                s2.fill = None;
311            }
312            sp_pr(w, x, o, &s2, true);
313            if let Some(st) = &s.style {
314                style(w, st);
315            }
316            w.close("p:pic");
317        }
318        ShapeKind::Group { children, child } if s.diagram.is_some() && depth < 32 => {
319            if let Some(d) = &s.diagram {
320                diagram(w, x, o, cx, s, d, children, child);
321            }
322        }
323        ShapeKind::Group { children, child } => {
324            let id = cx.id_for(s);
325            w.open0("p:grpSp");
326            w.open0("p:nvGrpSpPr");
327            cnvpr(w, x, o, id, s);
328            w.open0("p:cNvGrpSpPr");
329            locks(w, "a:grpSpLocks", s, A::new());
330            w.close("p:cNvGrpSpPr");
331            nv_pr(w, x, o, s);
332            w.close("p:nvGrpSpPr");
333            w.open0("p:grpSpPr");
334            let xf = s.xfrm.unwrap_or(*child);
335            dml::xfrm(w, "a:xfrm", &xf, Some(child));
336            if let Some(f) = &s.fill {
337                dml::fill(w, x, o, f);
338            }
339            if let Some(fx) = &s.effects {
340                if fx.fill_overlay.is_some()
341                    || fx.glow.is_some()
342                    || fx.inner_shadow.is_some()
343                    || fx.outer_shadow.is_some()
344                    || fx.reflection.is_some()
345                    || fx.soft_edge.is_some()
346                {
347                    dml::effects(w, fx);
348                }
349                if let Some(raw) =
350                    fx.raw3d.as_deref().map(|s| s.find("<a:sp3d").map_or(s, |i| &s[..i])).and_then(|s| dml::valid_raw3d(s, &["scene3d"]))
351                {
352                    w.raw(raw);
353                }
354            }
355            w.close("p:grpSpPr");
356            shapes(w, x, o, cx, children, depth + 1);
357            w.close("p:grpSp");
358        }
359        ShapeKind::Connector { start, end } => {
360            let id = cx.id_for(s);
361            w.open0("p:cxnSp");
362            w.open0("p:nvCxnSpPr");
363            cnvpr(w, x, o, id, s);
364            let st = start.and_then(|(sid, site)| cx.ids.get(sid).map(|i| (i, site)));
365            let en = end.and_then(|(sid, site)| cx.ids.get(sid).map(|i| (i, site)));
366            if st.is_none() && en.is_none() && !s.locked {
367                w.empty0("p:cNvCxnSpPr");
368            } else {
369                w.open0("p:cNvCxnSpPr");
370                locks(w, "a:cxnSpLocks", s, A::new());
371                if let Some((i, site)) = st {
372                    w.empty("a:stCxn", A::new().a("id", i).a("idx", site));
373                }
374                if let Some((i, site)) = en {
375                    w.empty("a:endCxn", A::new().a("id", i).a("idx", site));
376                }
377                w.close("p:cNvCxnSpPr");
378            }
379            nv_pr(w, x, o, s);
380            w.close("p:nvCxnSpPr");
381            let mut s2 = s.clone();
382            if matches!(s2.fill, Some(Fill::Background)) {
383                s2.fill = None;
384            }
385            if s2.geom.preset_name() == Some("rect") {
386                s2.geom = Geom::preset("line");
387            }
388            sp_pr(w, x, o, &s2, true);
389            if let Some(st) = &s.style {
390                style(w, st);
391            }
392            w.close("p:cxnSp");
393        }
394        ShapeKind::Table(t) => {
395            let id = cx.id_for(s);
396            frame_open(w, x, o, id, s);
397            w.open("a:graphicData", A::new().a("uri", "http://schemas.openxmlformats.org/drawingml/2006/table"));
398            table(w, x, o, t);
399            w.close("a:graphicData");
400            frame_close(w);
401        }
402        ShapeKind::Chart(c) => {
403            let id = cx.id_for(s);
404            // An unchanged Office 2016 chart goes back as itself, with a plain chart of the same
405            // data for readers without `cx1`, as PowerPoint writes it.
406            let ex = x.chartex_rel(o, c);
407            if let Some(exid) = &ex {
408                w.open("mc:AlternateContent", A::new().a("xmlns:mc", "http://schemas.openxmlformats.org/markup-compatibility/2006"));
409                w.open("mc:Choice", A::new().a("xmlns:cx1", "http://schemas.microsoft.com/office/drawing/2015/9/8/chartex").a("Requires", "cx1"));
410                frame_open(w, x, o, id, s);
411                w.open("a:graphicData", A::new().a("uri", super::chart::NS_CX));
412                w.empty("cx:chart", A::new().a("xmlns:cx", super::chart::NS_CX).a("r:id", exid));
413                w.close("a:graphicData");
414                frame_close(w);
415                w.close("mc:Choice");
416                w.open0("mc:Fallback");
417            }
418            let mut plain = (**c).clone();
419            plain.raw = plain.raw.filter(|r| !r.contains(super::chart::NS_CX));
420            if ex.is_some() {
421                // The kept workbook belongs to the `chartEx` part; the fallback has none.
422                plain.kept = Default::default();
423            }
424            let rid = x.chart_rel(o, &plain);
425            frame_open(w, x, o, id, s);
426            w.open("a:graphicData", A::new().a("uri", "http://schemas.openxmlformats.org/drawingml/2006/chart"));
427            w.empty("c:chart", A::new().a("xmlns:c", crate::pptx::opc::NS_C).a("r:id", rid));
428            w.close("a:graphicData");
429            frame_close(w);
430            if ex.is_some() {
431                w.close("mc:Fallback");
432                w.close("mc:AlternateContent");
433            }
434        }
435        ShapeKind::Media(m) => {
436            let id = cx.id_for(s);
437            media(w, x, o, id, s, m);
438        }
439        ShapeKind::Model3d { media, poster } => write_model(w, x, o, cx, s, *media, *poster),
440        ShapeKind::Ink { strokes } => {
441            // Ink becomes a group of freeform lines.
442            let id = cx.id_for(s);
443            let xf = s.xfrm.unwrap_or_default();
444            w.open0("p:grpSp");
445            w.open0("p:nvGrpSpPr");
446            cnvpr(w, x, o, id, s);
447            w.empty0("p:cNvGrpSpPr");
448            w.empty0("p:nvPr");
449            w.close("p:nvGrpSpPr");
450            w.open0("p:grpSpPr");
451            dml::xfrm(w, "a:xfrm", &xf, Some(&xf));
452            w.close("p:grpSpPr");
453            for (i, st) in strokes.iter().enumerate().take(10_000) {
454                let Some(stroke) = ink_shape(st, i) else { continue };
455                let sid = cx.ids.fresh();
456                sp(w, x, o, sid, &stroke, false);
457            }
458            w.close("p:grpSp");
459        }
460        ShapeKind::Opaque { xml, preview, label, parts } => opaque(w, x, o, cx, s, xml, *preview, label, parts),
461    }
462}

484fn write_model(w: &mut W, x: &mut Exp, o: &mut Out, cx: &mut Ctx, s: &Shape, media: crate::model::MediaId, poster: Option<crate::model::MediaId>) {
485    let Some(rid) = model_rel(x, o, media) else {
486        log::warn!("pptx: 3-D model '{}' has no embedded part", s.name);
487        return;
488    };
489    let id = cx.id_for(s);
490    let poster = poster.filter(|p| x.media(*p).is_some_and(|m| !m.data.is_empty()));
491    let Some(poster) = poster else {
492        write_model_part(w, x, o, id, s, &rid);
493        return;
494    };
495    // Older readers draw the fallback picture; readers that know `am3d` use the model.
496    w.open("mc:AlternateContent", A::new().a("xmlns:mc", NS_MC));
497    w.open("mc:Choice", A::new().a("xmlns:am3d", "http://schemas.microsoft.com/office/drawing/2017/model3d").a("Requires", "am3d"));
498    write_model_part(w, x, o, id, s, &rid);
499    w.close("mc:Choice");
500    w.open0("mc:Fallback");
501    let pic = Shape {
502        id: ShapeId(0),
503        name: s.name.clone(),
504        xfrm: s.xfrm,
505        kind: ShapeKind::Picture { fill: crate::model::style::PictureFill { media: poster, ..Default::default() } },
506        ..Default::default()
507    };
508    let pid = cx.ids.fresh();
509    shape_with_id(w, x, o, cx, &pic, pid);
510    w.close("mc:Fallback");
511    w.close("mc:AlternateContent");
512}

514/// A SmartArt graphic: a graphic frame naming its data, layout, quick style and colour parts,
515/// with the group's shapes as the cached drawing PowerPoint shows until it lays the diagram out.
516#[allow(clippy::too_many_arguments)]
517fn diagram(w: &mut W, x: &mut Exp, o: &mut Out, cx: &mut Ctx, s: &Shape, d: &crate::model::Diagram, children: &[Shape], child: &Xfrm) {
518    use crate::pptx::dgm;
519    let id = cx.id_for(s);
520    let n = x.next_diagram();
521    let frame = s.xfrm.unwrap_or(*child);
522    let name = |kind: &str| format!("ppt/diagrams/{kind}{n}.xml");
523    // The drawing: the group's shapes in the frame's own coordinates.
524    let dname = name("drawing");
525    let mut dout = Out::new(&dname);
526    let mut dw = W::new();
527    dw.open(
528        "dsp:drawing",
529        A::new().a("xmlns:dgm", dgm::NS_DGM).a("xmlns:dsp", dgm::NS_DSP).a("xmlns:a", crate::pptx::opc::NS_A).a("xmlns:r", crate::pptx::opc::NS_R),
530    );
531    dw.open0("dsp:spTree");
532    dw.open0("dsp:nvGrpSpPr");
533    dw.empty("dsp:cNvPr", A::new().a("id", 0).a("name", ""));
534    dw.empty0("dsp:cNvGrpSpPr");
535    dw.close("dsp:nvGrpSpPr");
536    dw.empty0("dsp:grpSpPr");
537    let (sx, sy) = (frame.w / child.w.max(1e-6), frame.h / child.h.max(1e-6));
538    let mut k = 0;
539    let mut flat = vec![];
540    walk(children, &mut |c, _| match &c.kind {
541        ShapeKind::Shape => flat.push(c.clone()),
542        ShapeKind::Connector { .. } => {
543            let mut sc = c.clone();
544            if sc.geom.preset_name() == Some("rect") {
545                sc.geom = Geom::preset("line");
546            }
547            flat.push(sc);
548        }
549        ShapeKind::Picture { fill } => {
550            let mut sc = c.clone();
551            sc.kind = ShapeKind::Shape;
552            sc.fill = Some(crate::model::style::Fill::Picture(fill.clone()));
553            flat.push(sc);
554        }
555        _ => {}
556    });
557    for mut c in flat.into_iter().take(10_000) {
558        if let Some(xf) = &mut c.xfrm {
559            *xf = Xfrm { x: (xf.x - child.x) * sx, y: (xf.y - child.y) * sy, w: xf.w * sx, h: xf.h * sy, ..*xf };
560        }
561        k += 1;
562        dw.open("dsp:sp", A::new().a("modelId", format!("{{{:08X}-5D1A-4C0A-9E5B-{k:012X}}}", s.id.0)));
563        dw.open0("dsp:nvSpPr");
564        dw.empty("dsp:cNvPr", A::new().a("id", 0).a("name", ""));
565        dw.empty0("dsp:cNvSpPr");
566        dw.close("dsp:nvSpPr");
567        sp_pr_tag(&mut dw, x, &mut dout, &c, true, "dsp:spPr");
568        if let Some(st) = &c.style {
569            style_tag(&mut dw, st, "dsp:style");
570        }
571        if let Some(tb) = &c.text {
572            dml::text_body(&mut dw, x, &mut dout, "dsp:txBody", tb, false);
573        }
574        dw.close("dsp:sp");
575    }
576    dw.close("dsp:spTree");
577    dw.close("dsp:drawing");
578    x.pkg.xml_part(&dname, dgm::CT_DRAWING, dw.finish());
579    x.pkg.rels(&dname, &dout.rels);
580    let rel = |o: &mut Out, kind: &str, part: &str| o.rels.add_new(kind, &crate::pptx::opc::relative(&o.name, part), false);
581    let drawing_rel = rel(o, dgm::RT_DRAWING, &dname);
582    let data = name("data");
583    // Node pictures: on the data model's presentation points, through the data part's rels.
584    let mut mout = Out::new(&data);
585    let pics: Vec<(usize, String)> =
586        d.nodes.iter().enumerate().filter_map(|(k, n)| n.picture.and_then(|m| x.media_rel(&mut mout, m, "image")).map(|r| (k, r))).collect();
587    x.pkg.xml_part(&data, dgm::CT_DATA, dgm::data_xml(d, s.id.0, &drawing_rel, &pics));
588    x.pkg.rels(&data, &mout.rels);
589    let dm = rel(o, "diagramData", &data);
590    let layout = name("layout");
591    let lo_xml = d.layout_xml.clone().map(String::into_bytes).unwrap_or_else(|| dgm::layout_xml(&d.kind));
592    x.pkg.xml_part(&layout, dgm::CT_LAYOUT, lo_xml);
593    let lo = rel(o, "diagramLayout", &layout);
594    let style = name("quickStyle");
595    x.pkg.xml_part(&style, dgm::CT_STYLE, d.style_xml.clone().map(String::into_bytes).unwrap_or_else(|| dgm::style_xml(&d.style)));
596    let qs = rel(o, "diagramQuickStyle", &style);
597    let colors = name("colors");
598    x.pkg.xml_part(&colors, dgm::CT_COLORS, d.colors_xml.clone().map(String::into_bytes).unwrap_or_else(|| dgm::colors_xml(&d.colors)));
599    let cs = rel(o, "diagramColors", &colors);
600    let mut fs = s.clone();
601    fs.xfrm = Some(frame);
602    frame_open(w, x, o, id, &fs);
603    w.open("a:graphicData", A::new().a("uri", "http://schemas.openxmlformats.org/drawingml/2006/diagram"));
604    w.empty("dgm:relIds", A::new().a("xmlns:dgm", dgm::NS_DGM).a("r:dm", dm).a("r:lo", lo).a("r:qs", qs).a("r:cs", cs));
605    w.close("a:graphicData");
606    frame_close(w);
607}

609fn ink_shape(st: &crate::model::InkStroke, i: usize) -> Option<Shape> {
610    let pts: Vec<(f64, f64)> = st.points.iter().filter(|p| p.0.is_finite() && p.1.is_finite()).map(|p| (p.0, p.1)).collect();
611    let (x0, y0) = pts.iter().fold((f64::MAX, f64::MAX), |a, p| (a.0.min(p.0), a.1.min(p.1)));
612    let (x1, y1) = pts.iter().fold((f64::MIN, f64::MIN), |a, p| (a.0.max(p.0), a.1.max(p.1)));
613    if pts.len() < 2 {
614        return None;
615    }
616    let (w, h) = ((x1 - x0).max(1.0), (y1 - y0).max(1.0));
617    // PPTX keeps one line width. A pressured stroke exports at its mean sample width;
618    // a legacy flat `0.5` stroke stays at the pen width.
619    let legacy = crate::model::ink_pressure_is_legacy(&st.points);
620    let width = if legacy {
621        st.width
622    } else {
623        let n = st.points.len().max(1) as f64;
624        st.points.iter().map(|p| crate::model::ink_sample_width(st.width, p.2, false)).sum::<f64>() / n
625    };
626    let mut d = String::new();
627    for (k, (px, py)) in pts.iter().enumerate() {
628        d.push_str(if k == 0 { "M " } else { "L " });
629        d.push_str(&format!("{:.2} {:.2} ", px - x0, py - y0));
630    }
631    let mut c = crate::model::style::ColorRef::rgb(st.color.with_alpha(255));
632    if st.highlighter {
633        c = c.with(crate::color::ColorTransform::Alpha(50_000));
634    }
635    Some(Shape {
636        id: ShapeId(0),
637        name: format!("Ink {}", i + 1),
638        xfrm: Some(Xfrm::new(x0, y0, w, h)),
639        geom: Geom::Custom { paths: vec![crate::model::CustomPath { w, h, d, fill: crate::geom::preset::FillMode::None, stroke: true }] },
640        fill: Some(Fill::None),
641        line: Some(crate::model::Line {
642            cap: Some(crate::model::style::LineCap::Round),
643            join: Some(crate::model::style::LineJoin::Round),
644            ..crate::model::Line::solid(c, width.max(0.25))
645        }),
646        ..Default::default()
647    })
648}

650fn sp(w: &mut W, x: &mut Exp, o: &mut Out, id: u32, s: &Shape, math: bool) {
651    w.open("p:sp", A::new().t("useBgFill", matches!(s.fill, Some(Fill::Background))));
652    w.open0("p:nvSpPr");
653    cnvpr(w, x, o, id, s);
654    let mut cnv = A::new();
655    if s.text_box {
656        cnv = cnv.a("txBox", "1");
657    }
658    let lock = s.ph.is_some() || s.locked;
659    if lock {
660        w.open("p:cNvSpPr", cnv);
661        locks(w, "a:spLocks", s, A::new());
662        w.close("p:cNvSpPr");
663    } else {
664        w.empty("p:cNvSpPr", cnv);
665    }
666    nv_pr(w, x, o, s);
667    w.close("p:nvSpPr");
668    let tmp;
669    let sref = if matches!(s.fill, Some(Fill::Background)) {
670        let mut c = s.clone();
671        c.fill = None;
672        tmp = c;
673        &tmp
674    } else {
675        s
676    };
677    sp_pr(w, x, o, sref, true);
678    if let Some(st) = &s.style {
679        style(w, st);
680    }
681    if let Some(tb) = &s.text {
682        // A shape's text without its own anchor sits in the middle in the model; a file without
683        // one would put it at the top, so say so.
684        if s.ph.is_none() && !s.text_box && tb.body.anchor.is_none() {
685            let mut tb = tb.clone();
686            tb.body.anchor = Some(crate::model::text::Anchor::Middle);
687            dml::text_body(w, x, o, "p:txBody", &tb, math);
688        } else {
689            dml::text_body(w, x, o, "p:txBody", tb, math);
690        }
691    }
692    w.close("p:sp");
693}

718fn table(w: &mut W, x: &mut Exp, o: &mut Out, t: &Table) {
719    let mut t = t.clone();
720    t.normalize();
721    w.open0("a:tbl");
722    let (guid, _) = crate::pptx::tables::guid_for_style(&t.style);
723    if x.table_style_set.insert(t.style.clone()) {
724        x.table_styles.push(t.style.clone());
725    }
726    w.open(
727        "a:tblPr",
728        A::new()
729            .t("firstRow", t.first_row)
730            .t("firstCol", t.first_col)
731            .t("lastRow", t.last_row)
732            .t("lastCol", t.last_col)
733            .t("bandRow", t.band_row)
734            .t("bandCol", t.band_col)
735            .b("rtl", t.rtl),
736    );
737    if let Some(f) = &t.background {
738        dml::fill(w, x, o, f);
739    }
740    w.elt("a:tableStyleId", &guid);
741    w.close("a:tblPr");
742    w.open0("a:tblGrid");
743    for c in &t.cols {
744        w.empty("a:gridCol", A::new().a("w", emu_pos(*c)));
745    }
746    w.close("a:tblGrid");
747    for row in &t.rows {
748        w.open("a:tr", A::new().a("h", emu_pos(row.height)));
749        for c in &row.cells {
750            w.open(
751                "a:tc",
752                A::new()
753                    .o("gridSpan", (c.grid_span > 1).then_some(c.grid_span))
754                    .o("rowSpan", (c.row_span > 1).then_some(c.row_span))
755                    .t("hMerge", c.h_merge)
756                    .t("vMerge", c.v_merge),
757            );
758            dml::text_body(w, x, o, "a:txBody", &c.text, false);
759            let mut a = A::new();
760            if let Some(m) = c.margins {
761                a = a.a("marL", emu(m[0])).a("marR", emu(m[1])).a("marT", emu(m[2])).a("marB", emu(m[3]));
762            }
763            a = a.o("vert", c.vertical.map(|v| v.xml())).o(
764                "anchor",
765                c.anchor.map(|a| match a {
766                    crate::model::text::Anchor::Justified | crate::model::text::Anchor::Distributed => "t",
767                    other => other.xml(),
768                }),
769            );
770            a = a.b("anchorCtr", c.anchor_ctr);
771            a = a.o("horzOverflow", crate::model::text::TextOverflow::horz_xml(c.horz_overflow));
772            w.open("a:tcPr", a);
773            for (tag, l) in [
774                ("a:lnL", &c.borders[0]),
775                ("a:lnR", &c.borders[1]),
776                ("a:lnT", &c.borders[2]),
777                ("a:lnB", &c.borders[3]),
778                ("a:lnTlToBr", &c.diag_down),
779                ("a:lnBlToTr", &c.diag_up),
780            ] {
781                if let Some(l) = l {
782                    dml::line(w, x, o, tag, l);
783                }
784            }
785            if let Some(f) = &c.fill {
786                match f {
787                    Fill::Background => {}
788                    f => dml::fill(w, x, o, f),
789                }
790            }
791            w.close("a:tcPr");
792            w.close("a:tc");
793        }
794        w.close("a:tr");
795    }
796    w.close("a:tbl");
797}

799fn media(w: &mut W, x: &mut Exp, o: &mut Out, id: u32, s: &Shape, m: &crate::model::MediaClip) {
800    let kind = if m.video { "video" } else { "audio" };
801    let link = x.media_rel(o, m.media, kind);
802    let embed = x.media_rel_new(o, m.media, crate::pptx::opc::RT_MEDIA);
803    w.open0("p:pic");
804    w.open0("p:nvPicPr");
805    let mut s2 = s.clone();
806    if s2.click.is_none() {
807        s2.click = Some(crate::model::text::Hyperlink {
808            action: crate::model::text::Action::PlayMedia,
809            tooltip: String::new(),
810            highlight_click: false,
811            sound: None,
812            stop_sound: false,
813            zoom_return: None,
814        });
815    }
816    cnvpr(w, x, o, id, &s2);
817    w.open0("p:cNvPicPr");
818    w.empty("a:picLocks", A::new().a("noChangeAspect", "1"));
819    w.close("p:cNvPicPr");
820    w.open0("p:nvPr");
821    if let Some(ph) = &s.ph {
822        let _ = ph;
823    }
824    let tag = if m.video { "a:videoFile" } else { "a:audioFile" };
825    if let Some(l) = &link {
826        w.empty(tag, A::new().a("r:link", l));
827    }
828    if let Some(e) = &embed {
829        w.open0("p:extLst");
830        w.open("p:ext", A::new().a("uri", "{DAA4B4D4-6D71-4841-9C94-3DE7FCFB9230}"));
831        w.open("p14:media", A::new().a("xmlns:p14", crate::pptx::opc::NS_P14).a("r:embed", e));
832        if m.trim_start_ms > 0 || m.trim_end_ms > 0 {
833            w.empty(
834                "p14:trim",
835                A::new().o("st", (m.trim_start_ms > 0).then_some(m.trim_start_ms)).o("end", (m.trim_end_ms > 0).then_some(m.trim_end_ms)),
836            );
837        }
838        if m.fade_in_ms > 0 || m.fade_out_ms > 0 {
839            w.empty("p14:fade", A::new().o("in", (m.fade_in_ms > 0).then_some(m.fade_in_ms)).o("out", (m.fade_out_ms > 0).then_some(m.fade_out_ms)));
840        }
841        if !m.bookmarks.is_empty() {
842            w.open0("p14:bmkLst");
843            for (n, t) in m.bookmarks.iter().take(1000) {
844                w.empty("p14:bmk", A::new().a("name", n).a("time", t));
845            }
846            w.close("p14:bmkLst");
847        }
848        w.close("p14:media");
849        w.close("p:ext");
850        w.close("p:extLst");
851    }
852    w.close("p:nvPr");
853    w.close("p:nvPicPr");
854    let poster = crate::model::style::PictureFill { media: m.poster.unwrap_or_default(), crop: m.crop, ..Default::default() };
855    dml::blip_fill(w, x, o, "p:blipFill", &poster);
856    let mut s3 = s.clone();
857    s3.fill = None;
858    sp_pr(w, x, o, &s3, true);
859    w.close("p:pic");
860}

862#[allow(clippy::too_many_arguments)]
863fn opaque(
864    w: &mut W,
865    x: &mut Exp,
866    o: &mut Out,
867    cx: &mut Ctx,
868    s: &Shape,
869    xml: &str,
870    preview: Option<crate::model::MediaId>,
871    label: &str,
872    parts: &[(String, String, crate::model::MediaId)],
873) {
874    // Kept XML is written back as it was when every part it references was kept too, with those
875    // parts re-added under this slide's relationship ids.
876    if let Ok(doc) = crate::pptx::xml::parse(xml.as_bytes())
877        && matches!(doc.root.local(), "graphicFrame" | "sp" | "grpSp" | "cxnSp")
878    {
879        let mut part_map: std::collections::HashMap<&str, (&str, crate::model::MediaId)> = std::collections::HashMap::new();
880        for (id, kind, m) in parts {
881            part_map.entry(id.as_str()).or_insert((kind.as_str(), *m));
882        }
883        if let Some(ids) = crate::pptx::read::rel_refs(&doc.root)
884            .iter()
885            .map(|r| {
886                let (kind, m) = part_map.get(r.as_str())?;
887                Some((r.clone(), x.media_rel(o, *m, kind)?))
888            })
889            .collect::<Option<std::collections::HashMap<_, _>>>()
890        {
891            let mut root = doc.root;
892            set_rel_ids(&mut root, &ids, 0);
893            ole_fallback(&mut root, x, o, preview);
894            let id = cx.id_for(s);
895            set_cnvpr_id(&mut root, id);
896            w.raw(&root.to_xml());
897            return;
898        }
899    }
900    log::warn!("pptx: object '{label}' written as its preview picture");
901    let id = cx.id_for(s);
902    let mut p = s.clone();
903    if p.name.is_empty() {
904        p.name = label.to_string();
905    }
906    match preview {
907        Some(m) if x.media(m).is_some() => {
908            p.kind = ShapeKind::Picture { fill: crate::model::style::PictureFill { media: m, ..Default::default() } };
909            p.ph = None;
910            shape_with_id(w, x, o, cx, &p, id);
911        }
912        _ => {
913            p.kind = ShapeKind::Shape;
914            p.ph = None;
915            p.geom = Geom::preset("rect");
916            p.fill = Some(Fill::None);
917            p.text = None;
918            sp(w, x, o, id, &p, false);
919        }
920    }
921}

923fn shape_with_id(w: &mut W, x: &mut Exp, o: &mut Out, cx: &mut Ctx, s: &Shape, id: u32) {
924    // Pictures only: reuse the given id.
925    if let ShapeKind::Picture { fill } = &s.kind {
926        w.open0("p:pic");
927        w.open0("p:nvPicPr");
928        cnvpr(w, x, o, id, s);
929        w.open0("p:cNvPicPr");
930        w.empty("a:picLocks", A::new().a("noChangeAspect", "1"));
931        w.close("p:cNvPicPr");
932        w.empty0("p:nvPr");
933        w.close("p:nvPicPr");
934        dml::blip_fill(w, x, o, "p:blipFill", fill);
935        // As for any picture, a colour or pattern fill shows behind the image (an object's preview
936        // keeps its fallback picture's fill).
937        let mut s2 = s.clone();
938        if matches!(s2.fill, Some(Fill::Picture(_) | Fill::Background)) {
939            s2.fill = None;
940        }
941        sp_pr(w, x, o, &s2, true);
942        w.close("p:pic");
943    } else {
944        shape(w, x, o, cx, s, 0);
945    }
946}

948/// A legacy OLE object finds its preview in the slide's VML drawing through `spid`, and PowerPoint
949/// won't open the slide without it. We don't write that drawing, so write PowerPoint's own
950/// fallback form instead: no `spid`, and the preview as a `p:pic`.
951fn ole_fallback(root: &mut crate::pptx::xml::El, x: &mut Exp, o: &mut Out, preview: Option<crate::model::MediaId>) {
952    let xfrm = root.child("xfrm").map(|t| {
953        let at = |e: &str, a: &str| t.child(e).and_then(|c| c.attr(a)).unwrap_or("0").to_string();
954        (at("off", "x"), at("off", "y"), at("ext", "cx"), at("ext", "cy"))
955    });
956    let Some(ole) = root
957        .elements_mut()
958        .find(|e| e.is("graphic"))
959        .and_then(|g| g.elements_mut().find(|e| e.is("graphicData")))
960        .and_then(|d| d.elements_mut().find(|e| e.is("oleObj")))
961    else {
962        return;
963    };
964    if ole.child("pic").is_some() || ole.attr("spid").is_none() {
965        return;
966    }
967    ole.attrs.retain(|(k, _)| k != "spid");
968    let rel = preview.and_then(|m| x.media_rel(o, m, "image")).unwrap_or_else(|| x.placeholder_rel(o));
969    let (ox, oy, cx, cy) = xfrm.unwrap_or_default();
970    let pic = format!(
971        r#"<p:pic><p:nvPicPr><p:cNvPr id="0" name=""/><p:cNvPicPr/><p:nvPr/></p:nvPicPr><p:blipFill><a:blip r:embed="{rel}"/><a:stretch><a:fillRect/></a:stretch></p:blipFill><p:spPr><a:xfrm><a:off x="{ox}" y="{oy}"/><a:ext cx="{cx}" cy="{cy}"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom></p:spPr></p:pic>"#
972    );
973    if let Ok(d) = crate::pptx::xml::parse(pic.as_bytes()) {
974        ole.children.push(crate::pptx::xml::Node::El(d.root));
975    }
976}

978fn set_rel_ids(e: &mut crate::pptx::xml::El, ids: &std::collections::HashMap<String, String>, depth: usize) {
979    if depth > crate::pptx::xml::MAX_DEPTH {
980        return;
981    }
982    for (k, v) in &mut e.attrs {
983        if crate::pptx::read::is_rel_attr(k, v)
984            && let Some(new) = ids.get(v.as_str())
985        {
986            v.clone_from(new);
987        }
988    }
989    for c in e.elements_mut() {
990        set_rel_ids(c, ids, depth + 1);
991    }
992}
993
994fn set_cnvpr_id(el: &mut crate::pptx::xml::El, id: u32) {
995    fn rec(e: &mut crate::pptx::xml::El, id: u32, depth: usize) -> bool {
996        if depth > 8 {
997            return false;
998        }
999        if e.is("cNvPr") {
1000            e.set_attr("id", &id.to_string());
1001            return true;
1002        }
1003        e.elements_mut().any(|c| rec(c, id, depth + 1))
1004    }
1005    rec(el, id, 0);
1006}

1008/// The fixed group header every shape tree starts with.
1009pub fn tree_header(w: &mut W) {
1010    w.open0("p:nvGrpSpPr");
1011    w.empty("p:cNvPr", A::new().a("id", 1).a("name", ""));
1012    w.empty0("p:cNvGrpSpPr");
1013    w.empty0("p:nvPr");
1014    w.close("p:nvGrpSpPr");
1015    w.open0("p:grpSpPr");
1016    w.open0("a:xfrm");
1017    w.empty("a:off", A::new().a("x", 0).a("y", 0));
1018    w.empty("a:ext", A::new().a("cx", 0).a("cy", 0));
1019    w.empty("a:chOff", A::new().a("x", 0).a("y", 0));
1020    w.empty("a:chExt", A::new().a("cx", 0).a("cy", 0));
1021    w.close("a:xfrm");
1022    w.close("p:grpSpPr");
1023}
