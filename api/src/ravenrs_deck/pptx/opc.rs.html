9/// Largest single part we decompress (media included).
10pub const MAX_PART: u64 = 512 * 1024 * 1024;
11/// Largest XML part we parse.
12pub const MAX_XML_PART: usize = 96 * 1024 * 1024;
13/// Total decompressed bytes accepted for one package.
14pub const MAX_TOTAL: u64 = 2 * 1024 * 1024 * 1024;
15/// Total bytes we are willing to decompress for one package, counting parts we then discard
16/// (too large, or failing part-way): bounds inflate work, not just what is kept.
17pub const MAX_INFLATE_WORK: u64 = 2 * MAX_TOTAL;
18/// Maximum number of zip entries.
19pub const MAX_ENTRIES: usize = 50_000;

39/// The files of a package, keyed by normalised part name (no leading slash, lowercase).
40pub struct Package {
41    parts: HashMap<String, (String, Vec<u8>)>,
42    ct_cache: std::sync::OnceLock<(HashMap<String, String>, HashMap<String, String>)>,
43}
44
45impl Package {
46    pub fn open(bytes: &[u8]) -> Result<Self, PptxError> {
47        Self::open_within(bytes, MAX_PART, MAX_TOTAL, MAX_INFLATE_WORK)
48    }
49    /// [`Package::open`] with explicit limits: per part, kept in total, and inflated in total.
50    fn open_within(bytes: &[u8], max_part: u64, max_total: u64, max_work: u64) -> Result<Self, PptxError> {
51        let mut zip = zip::ZipArchive::new(Cursor::new(bytes)).map_err(|e| PptxError::NotPptx(format!("{e}")))?;
52        if zip.len() > MAX_ENTRIES {
53            return Err(PptxError::NotPptx(format!("too many entries ({})", zip.len())));
54        }
55        let mut parts = HashMap::new();
56        let mut total: u64 = 0;
57        let mut work: u64 = 0;
58        for i in 0..zip.len() {
59            if work >= max_work {
60                log::warn!("pptx: decompression budget spent; skipping the remaining {} entries", zip.len() - i);
61                break;
62            }
63            let mut f = match zip.by_index(i) {
64                Ok(f) => f,
65                Err(e) => {
66                    log::warn!("pptx: skipping unreadable zip entry {i}: {e}");
67                    continue;
68                }
69            };
70            if f.is_dir() {
71                continue;
72            }
73            // RavenRS: zip 9 says when an entry name can't be read; such an entry is skipped.
74            let Ok(name) = f.name().map(|n| n.trim_start_matches('/').to_string()) else {
75                log::warn!("pptx: skipping zip entry {i} with an unreadable name");
76                continue;
77            };
78            let declared = f.size();
79            if declared > max_part {
80                log::warn!("pptx: skipping oversized part {name} ({declared} bytes)");
81                continue;
82            }
83            let mut data = Vec::with_capacity(declared.min(16 * 1024 * 1024) as usize);
84            let cap = max_part.min(max_total.saturating_sub(total)).min(max_work.saturating_sub(work));
85            let read = (&mut f).take(cap.saturating_add(1)).read_to_end(&mut data);
86            // Charge what was inflated whether or not the part is kept.
87            work = work.saturating_add(data.len() as u64);
88            match read {
89                Ok(_) if data.len() as u64 > cap => {
90                    log::warn!("pptx: part {name} exceeds size limits; skipped");
91                    continue;
92                }
93                Ok(_) => {}
94                Err(e) => {
95                    log::warn!("pptx: could not read {name}: {e}");
96                    continue;
97                }
98            }
99            total = total.saturating_add(data.len() as u64);
100            // A name kept with `\` would reach code that splits paths on `/`.
101            parts.insert(norm(&name), (name.replace('\\', "/"), data));
102        }
103        Ok(Package { parts, ct_cache: std::sync::OnceLock::new() })
104    }
105    pub fn get(&self, name: &str) -> Option<&[u8]> {
106        self.parts.get(&norm(name)).map(|(_, d)| d.as_slice())
107    }
108    /// An XML part's bytes, when it is within the XML size limit.
109    pub fn xml_bytes(&self, name: &str) -> Option<&[u8]> {
110        let b = self.get(name)?;
111        if b.len() > MAX_XML_PART {
112            log::warn!("pptx: XML part {name} too large");
113            return None;
114        }
115        Some(b)
116    }
117    /// Parse an XML part.
118    pub fn xml(&self, name: &str) -> Option<xml::Doc> {
119        let b = self.get(name)?;
120        if b.len() > MAX_XML_PART {
121            log::warn!("pptx: XML part {name} too large");
122            return None;
123        }
124        match xml::parse(b) {
125            Ok(d) => Some(d),
126            Err(e) => {
127                log::warn!("pptx: {name}: {e}");
128                None
129            }
130        }
131    }
132    /// Relationships of a part (`ppt/slides/slide1.xml` → `ppt/slides/_rels/slide1.xml.rels`).
133    pub fn rels(&self, part: &str) -> Rels {
134        let rp = rels_path(part);
135        let Some(doc) = self.xml(&rp) else { return Rels::default() };
136        rels_of(&doc, part)
137    }
138    /// Content types: (extension defaults, part overrides).
139    pub fn content_type(&self, part: &str) -> Option<String> {
140        let (overrides, defaults) = self.ct_cache.get_or_init(|| {
141            let mut ov = HashMap::new();
142            let mut def = HashMap::new();
143            if let Some(doc) = self.xml("[Content_Types].xml") {
144                for o in doc.root.children_named("Override") {
145                    if let (Some(n), Some(ct)) = (o.attr("PartName"), o.attr("ContentType")) {
146                        ov.insert(n.to_ascii_lowercase(), ct.to_string());
147                    }
148                }
149                for d in doc.root.children_named("Default") {
150                    if let (Some(e), Some(ct)) = (d.attr("Extension"), d.attr("ContentType")) {
151                        def.insert(e.to_ascii_lowercase(), ct.to_string());
152                    }
153                }
154            }
155            (ov, def)
156        });
157        let p = format!("/{}", norm(part));
158        if let Some(ct) = overrides.get(&p) {
159            return Some(ct.clone());
160        }
161        let ext = part.rsplit('.').next().unwrap_or("").to_ascii_lowercase();
162        defaults.get(&ext).cloned()
163    }
164    pub fn original_name(&self, name: &str) -> Option<&str> {
165        self.parts.get(&norm(name)).map(|(n, _)| n.as_str())
166    }
167}

169/// The relationships in a parsed rels part, with internal targets resolved against `part`'s folder.
170fn rels_of(doc: &xml::Doc, part: &str) -> Rels {
171    let base = dir_of(part);
172    let mut out = Rels::default();
173    for r in doc.root.children_named("Relationship") {
174        let (Some(id), Some(ty), Some(target)) = (r.attr("Id"), r.attr("Type"), r.attr("Target")) else { continue };
175        let external = r.attr("TargetMode").is_some_and(|m| m.eq_ignore_ascii_case("external"));
176        let target_raw = target.to_string();
177        let target = if external { target_raw.clone() } else { resolve(&base, target) };
178        out.by_id.entry(id.to_string()).or_insert(out.items.len());
179        out.items.push(Rel {
180            id: id.to_string(),
181            kind: ty.rsplit('/').next().unwrap_or(ty).to_string(),
182            ty: ty.to_string(),
183            target_raw,
184            target,
185            external,
186        });
187    }
188    out
189}

191/// The relationships of `part` from the bytes of its rels part (kept parts carried by a chart).
192pub fn rels_from_bytes(bytes: &[u8], part: &str) -> Rels {
193    match xml::parse(bytes) {
194        Ok(doc) => rels_of(&doc, part),
195        Err(e) => {
196            log::warn!("pptx: relationships of {part}: {e}");

198        }

242/// A part a chart may carry along beside its own `chartN.xml` (workbook, style, drawing, picture).
243/// Chart documents themselves are rewritten, and nothing outside these folders is copied: a relationship
244/// must not be able to replace `ppt/slides/slide1.xml` on save.
245pub fn sidecar_part(name: &str) -> bool {
246    let n = norm(name);
247    let segs: Vec<&str> = n.split('/').filter(|s| !s.is_empty()).collect();
248    if segs.iter().any(|s| *s == "." || *s == "..") {
249        return false;
250    }
251    let (Some(root), Some(area)) = (segs.first().copied(), segs.get(1).copied()) else { return false };
252    if root != "ppt" || !matches!(area, "embeddings" | "charts" | "drawings" | "media") {
253        return false;
254    }
255    let file = segs.last().copied().unwrap_or("");
256    let stem = file.strip_prefix("chart").unwrap_or("");
257    let num = stem.strip_suffix(".xml.rels").or_else(|| stem.strip_suffix(".xml")).unwrap_or("");
258    !(stem != file && !num.is_empty() && num.chars().all(|c| c.is_ascii_digit()))
259}

280fn percent_decode(s: &str) -> String {
281    let b = s.as_bytes();
282    let mut out = Vec::with_capacity(b.len());
283    let mut i = 0;
284    while i < b.len() {
285        let c = b.get(i).copied().unwrap_or(0);
286        if c == b'%'
287            && let Some(h) = b
288                .get(i + 1..i + 3)
289                .filter(|h| h.iter().all(u8::is_ascii_hexdigit))
290                .and_then(|h| std::str::from_utf8(h).ok())
291                .and_then(|h| u8::from_str_radix(h, 16).ok())
292        {
293            out.push(h);
294            i += 3;
295            continue;
296        }
297        out.push(c);
298        i += 1;
299    }
300    String::from_utf8_lossy(&out).into_owned()
301}

303/// Resolve a relative target against a base directory into a part name.
304pub fn resolve(base: &str, target: &str) -> String {
305    let target = percent_decode(target.split('#').next().unwrap_or("")).replace('\\', "/");
306    let target = target.as_str();
307    let mut segs: Vec<&str> = if target.starts_with('/') { vec![] } else { base.split('/').filter(|s| !s.is_empty()).collect() };
308    for s in target.split('/') {
309        match s {
310            "" | "." => {}
311            ".." => {
312                segs.pop();
313            }
314            s => segs.push(s),
315        }
316    }
317    segs.join("/")
318}

320/// Relative target from the directory of `from` to part `to` (both absolute part names).
321pub fn relative(from: &str, to: &str) -> String {
322    let fd_s = dir_of(from);
323    let fd: Vec<&str> = fd_s.split('/').filter(|s| !s.is_empty()).collect();
324    let td_s = dir_of(to);
325    let td: Vec<&str> = td_s.split('/').filter(|s| !s.is_empty()).collect();
326    let common = fd.iter().zip(td.iter()).take_while(|(a, b)| a == b).count();
327    let mut out = String::new();
328    for _ in common..fd.len() {
329        out.push_str("../");
330    }
331    for s in td.iter().skip(common) {
332        out.push_str(s);
333        out.push('/');
334    }
335    out.push_str(file_of(to));
336    out
337}

342/// Relationships being built for one part.
343#[derive(Default)]
344pub struct RelsOut {
345    pub items: Vec<(String, String, String, bool)>,
346    /// (type, target, external) → id of the relationship [`RelsOut::add`] made for it.
347    added: HashMap<(String, String, bool), String>,
348}
349
350impl RelsOut {
351    /// Add (or reuse) a relationship; `kind` is a full type URI or a suffix of the standard base.
352    pub fn add(&mut self, kind: &str, target: &str, external: bool) -> String {
353        let ty = if kind.starts_with("http") { kind.to_string() } else { format!("{RT_BASE}{kind}") };
354        let key = (ty, target.to_string(), external);
355        if let Some(id) = self.added.get(&key) {
356            return id.clone();
357        }
358        let id = format!("rId{}", self.items.len() + 1);
359        self.items.push((id.clone(), key.0.clone(), key.1.clone(), external));
360        self.added.insert(key, id.clone());
361        id
362    }
363    /// Always add a new relationship even if an identical one exists.
364    pub fn add_new(&mut self, kind: &str, target: &str, external: bool) -> String {
365        let ty = if kind.starts_with("http") { kind.to_string() } else { format!("{RT_BASE}{kind}") };
366        let id = format!("rId{}", self.items.len() + 1);
367        // `add` reuses the first relationship for a target, whichever call made it.
368        self.added.entry((ty.clone(), target.to_string(), external)).or_insert_with(|| id.clone());
369        self.items.push((id.clone(), ty, target.to_string(), external));
370        id
371    }
372    pub fn to_xml(&self) -> Vec<u8> {
373        let mut w = W::new();
374        w.open("Relationships", A::new().a("xmlns", "http://schemas.openxmlformats.org/package/2006/relationships"));
375        for (id, ty, target, ext) in &self.items {
376            let mut a = A::new().a("Id", id).a("Type", ty).a("Target", target);
377            if *ext {
378                a = a.a("TargetMode", "External");
379            }
380            w.empty("Relationship", a);
381        }
382        w.close("Relationships");
383        w.finish()
384    }
385}

387/// A package being written.
388pub struct PackageOut {
389    files: Vec<(String, Vec<u8>, bool)>,
390    defaults: Vec<(String, String)>,
391    overrides: Vec<(String, String)>,
392}
393
394impl Default for PackageOut {
395    fn default() -> Self {
396        PackageOut {
397            files: vec![],
398            defaults: vec![
399                ("rels".into(), "application/vnd.openxmlformats-package.relationships+xml".into()),
400                ("xml".into(), "application/xml".into()),
401            ],
402            overrides: vec![],
403        }
404    }
405}
406
407impl PackageOut {
408    /// Add an XML part with its content type override.
409    pub fn xml_part(&mut self, name: &str, content_type: &str, data: Vec<u8>) {
410        self.overrides.push((format!("/{name}"), content_type.to_string()));
411        self.files.push((name.to_string(), data, true));
412    }
413    pub fn rels(&mut self, part: &str, rels: &RelsOut) {
414        if rels.items.is_empty() {
415            return;
416        }
417        let path = rels_path(part);
418        if let Some(existing) = self.files.iter_mut().find(|(name, ..)| *name == path) {
419            existing.1 = rels.to_xml();
420        } else {
421            self.files.push((path, rels.to_xml(), true));
422        }
423    }
424    /// Add a part with no content type of its own (a relationships part).
425    pub fn raw(&mut self, name: &str, data: Vec<u8>) {
426        self.files.push((name.to_string(), data, true));
427    }
428    /// Add a binary part; its extension gets a Default content type.
429    pub fn binary(&mut self, name: &str, content_type: &str, data: Vec<u8>) {
430        let ext = name.rsplit('.').next().unwrap_or("bin").to_ascii_lowercase();
431        if !self.defaults.iter().any(|(e, _)| *e == ext) {
432            self.defaults.push((ext, content_type.to_string()));
433        } else if !self.defaults.iter().any(|(e, c)| *e == ext && c == content_type) {
434            self.overrides.push((format!("/{name}"), content_type.to_string()));
435        }
436        self.files.push((name.to_string(), data, false));
437    }
438    pub fn finish(self) -> Result<Vec<u8>, PptxError> {
439        let mut ct = W::new();
440        ct.open("Types", A::new().a("xmlns", "http://schemas.openxmlformats.org/package/2006/content-types"));
441        for (e, c) in &self.defaults {
442            ct.empty("Default", A::new().a("Extension", e).a("ContentType", c));
443        }
444        for (p, c) in &self.overrides {
445            ct.empty("Override", A::new().a("PartName", p).a("ContentType", c));
446        }
447        ct.close("Types");
448        let mut zw = zip::ZipWriter::new(Cursor::new(Vec::new()));
449        let deflate = zip::write::SimpleFileOptions::default().compression_method(zip::CompressionMethod::Deflated);
450        let stored = zip::write::SimpleFileOptions::default().compression_method(zip::CompressionMethod::Stored);
451        let err = |e: &dyn std::fmt::Display| PptxError::Write(e.to_string());
452        zw.start_file("[Content_Types].xml", deflate).map_err(|e| err(&e))?;
453        zw.write_all(&ct.finish()).map_err(|e| err(&e))?;
454        let mut written = std::collections::HashSet::new();
455        for (name, data, compress) in &self.files {
456            // Zip entry names must be unique: a second part under a name already written is
457            // dropped rather than producing a package readers reject or resolve differently.
458            if !written.insert(norm(name)) {
459                log::warn!("pptx: part {name} written twice; keeping the first");
460                continue;
461            }
462            let large = data.len() as u64 >= u32::MAX as u64 / 2;
463            let opts = if *compress { deflate } else { stored }.large_file(large);
464            zw.start_file(name.as_str(), opts).map_err(|e| err(&e))?;
465            zw.write_all(data).map_err(|e| err(&e))?;
466        }
467        let cur = zw.finish().map_err(|e| err(&e))?;
468        Ok(cur.into_inner())
469    }
470}

520        x.push_str(&format!(r#"<Relationship Id="rId0" Type="{RT_BASE}image" Target="../media/dup.png"/></Relationships>"#));
521        let rels = rels_from_bytes(x.as_bytes(), "ppt/slides/slide1.xml");
522        assert_eq!(rels.target("rId0"), Some("ppt/media/image0.png"), "the first of a repeated id wins");
523        assert!((0..n).all(|i| rels.target(&format!("rId{i}")) == Some(&*format!("ppt/media/image{i}.png"))));
524
525        let mut out = RelsOut::default();
526        let ids: Vec<String> = (0..n).map(|i| out.add("hyperlink", &format!("https://example.com/{i}"), true)).collect();
527        assert_eq!(out.add("hyperlink", "https://example.com/7", true), ids[7], "an identical target reuses its id");
528        assert_eq!(out.add("hyperlink", "https://example.com/7", false), format!("rId{}", n + 1), "internal is distinct");
529        let fresh = out.add_new("image", "../media/a.png", false);
530        assert_ne!(out.add_new("image", "../media/a.png", false), fresh);
531        assert_eq!(out.add("image", "../media/a.png", false), fresh, "add reuses the first add_new made");
532    }
