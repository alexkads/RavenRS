1//! SmartArt (DrawingML diagrams): the outline ⇄ the data model, and the layout, quick style and
2//! colour definitions DeckCraft writes for its own layouts.
3//!
4//! A diagram's data part holds its points (`dgm:pt`: the document, one per node, and the
5//! transitions PowerPoint pairs with each link) and its links (`dgm:cxn` of type `parOf`, parent →
6//! child, ordered by `srcOrd`). The outline is that tree flattened depth first.

25/// PowerPoint's quick style for a SmartArt Style.
26pub fn qs_id(style: &str) -> String {
27    let s = match style {
28        "subtle" => "simple3",
29        "intense" => "simple5",
30        "outline" => "simple4",
31        "polished" => "3d1",
32        "inset" => "3d2",
33        "cartoon" => "3d3",
34        "powder" => "3d4",
35        "brick" => "3d5",
36        _ => "simple1",
37    };
38    format!("{QS_BASE}{s}")
39}

41/// The SmartArt Style closest to a quick style id.
42pub fn style_for(qs: &str) -> &'static str {
43    match qs.rsplit('/').next().unwrap_or("") {
44        "simple1" | "" => "",
45        "simple4" => "outline",
46        "3d1" => "polished",
47        "3d2" => "inset",
48        "3d3" => "cartoon",
49        "3d4" => "powder",
50        "3d5" => "brick",
51        "simple2" | "simple3" => "subtle",
52        _ => "intense",
53    }
54}

56/// PowerPoint's colour definition for Change Colors.
57pub fn cs_id(colors: &str) -> String {
58    if let Some(a @ 2..=5) = colors.strip_prefix("range").and_then(|r| r.get(..1)).and_then(|n| n.parse::<u8>().ok()) {
59        return format!("{CS_BASE}colorful{a}");
60    }
61    let grad = colors.ends_with("Grad");
62    match colors.strip_prefix("accent").and_then(|r| r.get(..1)).and_then(|n| n.parse::<u8>().ok()) {
63        Some(n @ 1..=6) => format!("{CS_BASE}accent{n}_{}", if grad { 3 } else { 2 }),
64        _ => format!("{CS_BASE}colorful1"),
65    }
66}

68/// The accents a Change Colors choice fills shapes with, in turn.
69fn fill_accents(colors: &str) -> Vec<String> {
70    if let Some(a @ 2..=5) = colors.strip_prefix("range").and_then(|r| r.get(..1)).and_then(|n| n.parse::<u8>().ok()) {
71        return vec![format!("accent{a}"), format!("accent{}", a + 1)];
72    }
73    match colors.strip_prefix("accent").and_then(|r| r.get(..1)).and_then(|n| n.parse::<u8>().ok()) {
74        Some(n @ 1..=6) => vec![format!("accent{n}")],
75        _ => (1..=6).map(|k| format!("accent{k}")).collect(),
76    }
77}

79/// Change Colors from a colour definition id: `accentN_…` keeps accent N, anything else is colourful.
80pub fn colors_for(cs: &str) -> String {
81    let last = cs.rsplit('/').next().unwrap_or("");
82    if let Some(a @ 2..=5) = last.strip_prefix("colorful").and_then(|n| n.parse::<u8>().ok()) {
83        return format!("range{a}{}", a + 1);
84    }
85    let mut parts = last.strip_prefix("accent").unwrap_or("").split('_');
86    match (parts.next().and_then(|n| n.parse::<u8>().ok()), parts.next()) {
87        (Some(n @ 1..=6), Some("3" | "4" | "5")) => format!("accent{n}Grad"),
88        (Some(n @ 1..=6), _) => format!("accent{n}"),
89        _ => String::new(),
90    }
91}

146/// The DeckCraft layout closest to a PowerPoint layout (by its id and category).
147pub fn kind_for(layout_id: &str, category: &str) -> &'static str {
148    let last = layout_id.rsplit('/').next().unwrap_or("").to_ascii_lowercase();
149    // DeckCraft before Picture Accent Process got its own id wrote it as pList2 (Horizontal
150    // Picture List) with its own layout part in the process category; PowerPoint's pList2 is a list.
151    if last == "plist2" && category.eq_ignore_ascii_case("process") {
152        return "pictureProcess";
153    }
154    // Exact ids first: the layouts DeckCraft writes, and their closest siblings.
155    for k in [
156        "blockList",
157        "hBulletList",
158        "process",
159        "chevron",
160        "verticalProcess",
161        "stepUp",
162        "cycle",
163        "blockCycle",
164        "radial",
165        "hierarchy",
166        "pyramid",
167        "invertedPyramid",
168        "matrix",
169        "titledMatrix",
170        "pictureList",
171        "pictureCaption",
172        "pictureProcess",
173        "pictureHList",
174        "timeline",
175        "venn",
176        "target",
177        "textCycle",
178        "continuousCycle",
179        "horizontalHierarchy",
180        "gridMatrix",
181        "gear",
182        "funnel",
183        "segmentedCycle",
184        "multidirectionalCycle",
185        "orgChart",
186        "halfCircleOrgChart",
187        "nameTitleOrgChart",
188        "verticalBoxList",
189        "trapezoidList",
190        "tableList",
191        "accentProcess",
192        "continuousBlockProcess",
193        "bendingProcess",
194        "linearVenn",
195        "stackedVenn",
196        "nestedTarget",
197        "cycleMatrix",
198        "segmentedPyramid",
199        "pie",
200        "list",
201    ] {
202        if layout_id == layout_id_of(k) {
203            return k;
204        }
205    }
206    let s = format!("{last} {}", category.to_ascii_lowercase());
207    if s.contains("gear") {
208        "gear"
209    } else if s.contains("funnel") {
210        "funnel"
211    } else if s.contains("venn") {
212        "venn"
213    } else if s.contains("target") {
214        "target"
215    } else if s.contains("timeline") {
216        "timeline"
217    } else if s.contains("nameandtitle") {
218        "nameTitleOrgChart"
219    } else if s.contains("orgchart") || s.contains("organizationchart") || s.contains("organizationalchart") {
220        "orgChart"
221    } else if s.contains("hierarchy") {
222        "hierarchy"
223    } else if s.contains("radial") {
224        "radial"
225    } else if s.contains("cycle") {
226        "cycle"
227    } else if s.contains("pyramid") {
228        "pyramid"
229    } else if s.contains("matrix") {
230        "matrix"
231    } else if s.contains("chevron") {
232        "chevron"
233    } else if s.contains("process") || s.contains("arrow") {
234        "process"
235    } else if last == "default" {
236        "blockList"
237    } else if last.starts_with("hlist") {
238        "hBulletList"
239    } else {
240        "list"
241    }
242}

248fn category(kind: &str) -> &'static str {
249    match kind {
250        "process"
251        | "chevron"
252        | "verticalProcess"
253        | "stepUp"
254        | "pictureProcess"
255        | "timeline"
256        | "gear"
257        | "funnel"
258        | "accentProcess"
259        | "continuousBlockProcess"
260        | "bendingProcess" => "process",
261        "venn" | "target" | "linearVenn" | "stackedVenn" | "nestedTarget" => "relationship",
262        "cycle" | "blockCycle" | "radial" | "textCycle" | "continuousCycle" | "segmentedCycle" | "multidirectionalCycle" | "cycleMatrix" | "pie" => {
263            "cycle"
264        }
265        "hierarchy" | "horizontalHierarchy" | "orgChart" | "halfCircleOrgChart" | "nameTitleOrgChart" => "hierarchy",
266        "pyramid" | "invertedPyramid" | "segmentedPyramid" => "pyramid",
267        "matrix" | "titledMatrix" | "gridMatrix" => "matrix",
268        _ => "list",
269    }
270}

277fn text(w: &mut W, t: &str) {
278    w.open0("dgm:t");
279    w.empty0("a:bodyPr");
280    w.empty0("a:lstStyle");
281    w.open0("a:p");
282    if t.is_empty() {
283        w.empty("a:endParaRPr", A::new().a("lang", "en-US"));
284    } else {
285        w.open0("a:r");
286        w.empty("a:rPr", A::new().a("lang", "en-US"));
287        w.elt("a:t", t);
288        w.close("a:r");
289    }
290    w.close("a:p");
291    w.close("dgm:t");
292}

294/// `ppt/diagrams/dataN.xml` for the outline. `drawing_rel` is the slide's relationship id of the
295/// cached drawing; `pics` the data part's relationship ids of node pictures, by node index.
296pub fn data_xml(d: &Diagram, seed: u32, drawing_rel: &str, pics: &[(usize, String)]) -> Vec<u8> {
297    let lo = if d.layout_id.is_empty() { layout_id(&d.kind).to_string() } else { d.layout_id.clone() };
298    let mut w = W::new();
299    w.open("dgm:dataModel", A::new().a("xmlns:dgm", NS_DGM).a("xmlns:a", crate::pptx::opc::NS_A).a("xmlns:r", crate::pptx::opc::NS_R));
300    w.open0("dgm:ptLst");
301    let doc = guid(seed, 0);
302    w.open("dgm:pt", A::new().a("modelId", &doc).a("type", "doc"));
303    w.empty(
304        "dgm:prSet",
305        A::new()
306            .a("loTypeId", &lo)
307            .a("loCatId", category(&d.kind))
308            .a("qsTypeId", qs_id(&d.style))
309            .a("qsCatId", "simple")
310            .a("csTypeId", cs_id(&d.colors))
311            .a("csCatId", if d.colors.starts_with("accent") { d.colors.as_str() } else { "colorful" })
312            .a("phldr", "1"),
313    );
314    w.empty0("dgm:spPr");
315    text(&mut w, "");
316    w.close("dgm:pt");
317    // Node k: model id 1 + 3k; its link's parent and sibling transitions 2 + 3k and 3 + 3k.
318    let node = |k: usize| guid(seed, 1 + 3 * k);
319    for (k, n) in d.nodes.iter().enumerate() {
320        let mut a = A::new().a("modelId", node(k));
321        if n.assistant {
322            a = a.a("type", "asst");
323        }
324        w.open("dgm:pt", a);
325        w.empty("dgm:prSet", A::new().a("phldrT", "[Text]"));
326        w.empty0("dgm:spPr");
327        text(&mut w, &n.text);
328        w.close("dgm:pt");
329        let cxn = guid(seed, 0x10_0000 + k);
330        for (i, ty) in [(2, "parTrans"), (3, "sibTrans")] {
331            w.open("dgm:pt", A::new().a("modelId", guid(seed, i + 3 * k)).a("type", ty).a("cxnId", &cxn));
332            w.empty0("dgm:prSet");
333            w.empty0("dgm:spPr");
334            text(&mut w, "");
335            w.close("dgm:pt");
336        }
337    }
338    // A node's picture sits on the presentation point of its picture placeholder, as PowerPoint
339    // stores it (`presAssocID` names the node, the blip is a relationship of the data part).
340    // Name and Title: a node's title is the text of its title box's presentation point.
341    for (k, n) in d.nodes.iter().enumerate() {
342        if n.title.is_empty() {
343            continue;
344        }
345        w.open("dgm:pt", A::new().a("modelId", guid(seed, 0x30_0000 + k)).a("type", "pres"));
346        w.empty("dgm:prSet", A::new().a("presAssocID", node(k)).a("presName", "titleText1").a("presStyleLbl", "fgAcc1").a("custT", "1"));
347        w.empty0("dgm:spPr");
348        text(&mut w, &n.title);
349        w.close("dgm:pt");
350    }
351    for (k, rid) in pics {
352        if *k >= d.nodes.len() {
353            continue;
354        }
355        w.open("dgm:pt", A::new().a("modelId", guid(seed, 0x20_0000 + k)).a("type", "pres"));
356        w.empty("dgm:prSet", A::new().a("presAssocID", node(*k)).a("presName", "Image").a("presStyleLbl", "fgImgPlace1"));
357        w.open0("dgm:spPr");
358        w.open0("a:blipFill");
359        w.empty("a:blip", A::new().a("r:embed", rid.as_str()));
360        w.open0("a:stretch");
361        w.empty0("a:fillRect");
362        w.close("a:stretch");
363        w.close("a:blipFill");
364        w.close("dgm:spPr");
365        w.close("dgm:pt");
366    }
367    w.close("dgm:ptLst");
368    w.open0("dgm:cxnLst");
369    let mut stack: Vec<(u8, String)> = vec![];
370    let mut ords: HashMap<String, usize> = HashMap::new();
371    for (k, n) in d.nodes.iter().enumerate() {
372        while stack.last().is_some_and(|(l, _)| *l >= n.level) {
373            stack.pop();
374        }
375        let parent = stack.last().map(|(_, id)| id.clone()).unwrap_or_else(|| doc.clone());
376        let ord = ords.entry(parent.clone()).or_insert(0);
377        w.empty(
378            "dgm:cxn",
379            A::new()
380                .a("modelId", guid(seed, 0x10_0000 + k))
381                .a("srcId", &parent)
382                .a("destId", node(k))
383                .a("srcOrd", *ord)
384                .a("destOrd", 0)
385                .a("parTransId", guid(seed, 2 + 3 * k))
386                .a("sibTransId", guid(seed, 3 + 3 * k)),
387        );
388        *ord += 1;
389        stack.push((n.level, node(k)));
390    }
391    w.close("dgm:cxnLst");
392    w.empty0("dgm:bg");
393    w.empty0("dgm:whole");
394    w.open0("dgm:extLst");
395    w.open("a:ext", A::new().a("uri", "http://schemas.microsoft.com/office/drawing/2008/diagram"));
396    w.empty(
397        "dsp:dataModelExt",
398        A::new().a("xmlns:dsp", NS_DSP).a("relId", drawing_rel).a("minVer", "http://schemas.openxmlformats.org/drawingml/2006/diagram"),
399    );
400    w.close("a:ext");
401    w.close("dgm:extLst");
402    w.close("dgm:dataModel");
403    w.finish()
404}

406/// The outline of a data part: the `parOf` tree under the document point, depth first, or the data points in `ptLst` order when no `parOf` connections exist.
407pub fn outline(data: &El) -> Vec<DiagramNode> {
408    outline_ids(data).into_iter().map(|(_, n)| n).collect()
409}

411/// The relationship ids (of the data part) of node pictures, by node index in [`outline`]: the
412/// blip fill of a presentation point associated with the node.
413pub fn node_pictures(data: &El) -> Vec<(usize, String)> {
414    let ids: HashMap<String, usize> = outline_ids(data).into_iter().enumerate().map(|(k, (id, _))| (id, k)).collect();
415    let Some(pts) = data.child("ptLst") else { return vec![] };
416    let mut out: Vec<(usize, String)> = vec![];
417    for pt in pts.children_named("pt").take(MAX_NODES * 8) {
418        if pt.attr("type") != Some("pres") {
419            continue;
420        }
421        let Some(k) = pt.child("prSet").and_then(|p| p.attr("presAssocID")).and_then(|a| ids.get(a)) else { continue };
422        let Some(rid) = pt.child("spPr").and_then(|s| s.child("blipFill")).and_then(|b| b.child("blip")).and_then(|b| b.attr("r:embed")) else {
423            continue;
424        };
425        if !out.iter().any(|(j, _)| j == k) {
426            out.push((*k, rid.to_string()));
427        }
428    }
429    out
430}

432/// Extracts diagram outline `(modelId, DiagramNode)` pairs in tree order from a `dgm:dataModel` element.
433fn outline_ids(data: &El) -> Vec<(String, DiagramNode)> {
434    let Some(pts) = data.child("ptLst") else { return vec![] };
435    let mut texts: HashMap<&str, String> = HashMap::new();
436    let mut assistants: std::collections::HashSet<&str> = std::collections::HashSet::new();
437    let mut titles: HashMap<&str, String> = HashMap::new();
438    let mut pt_order: Vec<&str> = Vec::new();
439    let mut doc: Option<&str> = None;
440    for pt in pts.children_named("pt").take(MAX_NODES * 4) {
441        let Some(id) = pt.attr("modelId") else { continue };
442        match pt.attr("type").unwrap_or("node") {
443            "doc" => doc = doc.or(Some(id)),
444            ty @ ("node" | "asst") => {
445                let t = pt.child("t").map(paragraphs).unwrap_or_default();
446                texts.insert(id, t);
447                pt_order.push(id);
448                if ty == "asst" {
449                    assistants.insert(id);
450                }
451            }
452            "pres" => {
453                // Name and Title: the title box's own text, on its presentation point.
454                let pr = pt.child("prSet");
455                if pr.and_then(|p| p.attr("presName")).is_some_and(|n| n.starts_with("titleText"))
456                    && let Some(node) = pr.and_then(|p| p.attr("presAssocID"))
457                    && let Some(t) = pt.child("t").map(paragraphs).filter(|t| !t.is_empty())
458                {
459                    titles.insert(node, t);
460                }
461            }
462            _ => {}
463        }
464    }
465    let mut kids: HashMap<&str, Vec<(i64, &str)>> = HashMap::new();
466    if let Some(cx) = data.child("cxnLst") {
467        for c in cx.children_named("cxn").take(MAX_NODES * 4) {
468            if c.attr("type").unwrap_or("parOf") != "parOf" {
469                continue;
470            }
471            let (Some(src), Some(dst)) = (c.attr("srcId"), c.attr("destId")) else { continue };
472            if texts.contains_key(dst) {
473                kids.entry(src).or_default().push((c.i64("srcOrd").unwrap_or(0), dst));
474            }
475        }
476    }
477    for v in kids.values_mut() {
478        v.sort_by_key(|(o, _)| *o);
479    }
480    let mut out = vec![];
481    let mut seen = std::collections::HashSet::new();
482    // Depth first, iteratively; a node reached twice (a cycle in a hostile file) is skipped.
483    let mut stack: Vec<(&str, u8)> = doc.and_then(|d| kids.get(d)).map(|v| v.iter().rev().map(|(_, id)| (*id, 0u8)).collect()).unwrap_or_default();
484    while let Some((id, level)) = stack.pop() {
485        if out.len() >= MAX_NODES || !seen.insert(id) {
486            continue;
487        }
488        out.push((
489            id.to_string(),
490            DiagramNode {
491                text: texts.get(id).cloned().unwrap_or_default(),
492                level: level.min(8),
493                picture: None,
494                assistant: assistants.contains(id),
495                title: titles.get(id).cloned().unwrap_or_default(),
496            },
497        ));
498        if let Some(v) = kids.get(id) {
499            stack.extend(v.iter().rev().map(|(_, c)| (*c, level.saturating_add(1))));
500        }
501    }
502    if out.is_empty() && kids.is_empty() {
503        for id in pt_order {
504            if out.len() >= MAX_NODES || !seen.insert(id) {
505                continue;
506            }
507            out.push((
508                id.to_string(),
509                DiagramNode {
510                    text: texts.get(id).cloned().unwrap_or_default(),
511                    level: 0,
512                    picture: None,
513                    assistant: assistants.contains(id),
514                    title: titles.get(id).cloned().unwrap_or_default(),
515                },
516            ));
517        }
518    }
519    out
520}

522/// A text body's paragraphs joined by spaces.
523fn paragraphs(t: &El) -> String {
524    t.children_named("p")
525        .map(|p| {
526            let mut v = vec![];
527            p.find_all("t", &mut v);
528            v.iter().map(|e| e.text()).collect::<String>()
529        })
530        .filter(|s| !s.trim().is_empty())
531        .collect::<Vec<_>>()
532        .join(" ")
533}

535/// The layout algorithm and node shape of a DeckCraft layout.
536fn alg(kind: &str) -> (&'static str, Vec<(&'static str, &'static str)>, &'static str) {
537    match kind {
538        "process" | "stepUp" | "pictureProcess" | "pictureCaption" | "pictureHList" | "timeline" => {
539            ("lin", vec![("linDir", "fromL"), ("nodeVertAlign", "mid")], "roundRect")
540        }
541        "chevron" => ("lin", vec![("linDir", "fromL"), ("nodeVertAlign", "mid")], "chevron"),
542        "verticalProcess" => ("lin", vec![("linDir", "fromT"), ("nodeHorzAlign", "ctr")], "roundRect"),
543        "cycle"
544        | "radial"
545        | "venn"
546        | "target"
547        | "textCycle"
548        | "continuousCycle"
549        | "gear"
550        | "segmentedCycle"
551        | "multidirectionalCycle"
552        | "cycleMatrix"
553        | "pie" => ("cycle", vec![("stAng", "0"), ("spanAng", "360")], "ellipse"),
554        "blockCycle" => ("cycle", vec![("stAng", "0"), ("spanAng", "360")], "roundRect"),
555        "pyramid" | "invertedPyramid" | "segmentedPyramid" => {
556            ("pyra", vec![("linDir", if kind == "pyramid" { "fromT" } else { "fromB" }), ("txDir", "fromT"), ("pyraAcctPos", "bef")], "trapezoid")
557        }
558        "matrix" | "titledMatrix" | "gridMatrix" | "blockList" => (
559            "snake",
560            vec![("grDir", "tL"), ("flowDir", "row"), ("contDir", "sameDir"), ("off", "ctr")],
561            if kind == "blockList" { "rect" } else { "roundRect" },
562        ),
563        "hBulletList" => ("lin", vec![("linDir", "fromL"), ("nodeVertAlign", "t")], "roundRect"),
564        _ => ("lin", vec![("linDir", "fromT"), ("vertAlign", "mid")], "roundRect"),
565    }
566}

568fn node_layout(w: &mut W, shape: &str, bullets: bool) {
569    w.open("dgm:layoutNode", A::new().a("name", "node"));
570    w.open0("dgm:varLst");
571    w.empty("dgm:bulletEnabled", A::new().a("val", if bullets { "1" } else { "0" }));
572    w.close("dgm:varLst");
573    w.empty("dgm:alg", A::new().a("type", "tx"));
574    w.open("dgm:shape", A::new().a("type", shape).a("r:blip", ""));
575    w.empty0("dgm:adjLst");
576    w.close("dgm:shape");
577    w.empty("dgm:presOf", A::new().a("axis", if bullets { "desOrSelf" } else { "self" }).a("ptType", "node"));
578    w.open0("dgm:constrLst");
579    for side in ["tMarg", "bMarg", "lMarg", "rMarg"] {
580        w.empty("dgm:constr", A::new().a("type", side).a("refType", "primFontSz").a("fact", "0.3"));
581    }
582    w.close("dgm:constrLst");
583    w.open0("dgm:ruleLst");
584    w.empty("dgm:rule", A::new().a("type", "primFontSz").a("val", "5").a("fact", "NaN").a("max", "NaN"));
585    w.close("dgm:ruleLst");
586    w.close("dgm:layoutNode");
587}

603/// `ppt/diagrams/layoutN.xml`: a layout definition for DeckCraft's layout `kind`, under the id of
604/// the closest built-in PowerPoint layout.
605pub fn layout_xml(kind: &str) -> Vec<u8> {
606    let mut w = W::new();
607    w.open(
608        "dgm:layoutDef",
609        A::new().a("xmlns:dgm", NS_DGM).a("xmlns:a", crate::pptx::opc::NS_A).a("xmlns:r", crate::pptx::opc::NS_R).a("uniqueId", layout_id(kind)),
610    );
611    w.empty("dgm:title", A::new().a("val", ""));
612    w.empty("dgm:desc", A::new().a("val", ""));
613    w.open0("dgm:catLst");
614    w.empty("dgm:cat", A::new().a("type", category(kind)).a("pri", "1000"));
615    w.close("dgm:catLst");
616    w.open("dgm:layoutNode", A::new().a("name", "diagram"));
617    w.open0("dgm:varLst");
618    w.empty0("dgm:dir");
619    w.empty("dgm:resizeHandles", A::new().a("val", "exact"));
620    w.close("dgm:varLst");
621    if matches!(kind, "hierarchy" | "horizontalHierarchy" | "orgChart" | "halfCircleOrgChart" | "nameTitleOrgChart") {
622        // Each node over the row of its children, recursively.
623        w.empty("dgm:alg", A::new().a("type", "hierChild"));
624        w.open("dgm:shape", A::new().a("r:blip", ""));
625        w.empty0("dgm:adjLst");
626        w.close("dgm:shape");
627        w.empty0("dgm:presOf");
628        w.empty0("dgm:constrLst");
629        w.empty0("dgm:ruleLst");
630        w.open("dgm:forEach", A::new().a("name", "level").a("axis", "ch").a("ptType", "node"));
631        w.open("dgm:layoutNode", A::new().a("name", "hierRoot"));
632        w.empty("dgm:alg", A::new().a("type", "hierRoot"));
633        w.open("dgm:shape", A::new().a("r:blip", ""));
634        w.empty0("dgm:adjLst");
635        w.close("dgm:shape");
636        w.empty0("dgm:presOf");
637        w.empty0("dgm:constrLst");
638        w.empty0("dgm:ruleLst");
639        node_layout(&mut w, "roundRect", false);
640        w.open("dgm:layoutNode", A::new().a("name", "hierChild"));
641        w.empty("dgm:alg", A::new().a("type", "hierChild"));
642        w.open("dgm:shape", A::new().a("r:blip", ""));
643        w.empty0("dgm:adjLst");
644        w.close("dgm:shape");
645        w.empty0("dgm:presOf");
646        w.empty0("dgm:constrLst");
647        w.empty0("dgm:ruleLst");
648        w.empty("dgm:forEach", A::new().a("name", "repeat").a("ref", "level"));
649        w.close("dgm:layoutNode");
650        w.close("dgm:layoutNode");
651        w.close("dgm:forEach");
652    } else {
653        let (ty, params, shape) = alg(kind);
654        w.open("dgm:alg", A::new().a("type", ty));
655        for (k, v) in params {
656            w.empty("dgm:param", A::new().a("type", k).a("val", v));
657        }
658        w.close("dgm:alg");
659        w.open("dgm:shape", A::new().a("r:blip", ""));
660        w.empty0("dgm:adjLst");
661        w.close("dgm:shape");
662        w.empty0("dgm:presOf");
663        w.open0("dgm:constrLst");
664        w.empty("dgm:constr", A::new().a("type", "w").a("for", "ch").a("forName", "node").a("refType", "w"));
665        w.empty(
666            "dgm:constr",
667            A::new().a("type", "h").a("for", "ch").a("forName", "node").a("refType", "w").a("refFor", "ch").a("refForName", "node").a("fact", "0.6"),
668        );
669        w.empty(
670            "dgm:constr",
671            A::new().a("type", "w").a("for", "ch").a("forName", "space").a("refType", "w").a("refFor", "ch").a("refForName", "node").a("fact", "0.1"),
672        );
673        w.empty("dgm:constr", A::new().a("type", "primFontSz").a("for", "ch").a("forName", "node").a("op", "equ").a("val", "65"));
674        w.close("dgm:constrLst");
675        w.empty0("dgm:ruleLst");
676        w.open("dgm:forEach", A::new().a("name", "nodes").a("axis", "ch").a("ptType", "node"));
677        node_layout(&mut w, shape, true);
678        if !matches!(kind, "cycle" | "blockCycle" | "radial") {
679            space(&mut w);
680        }
681        w.close("dgm:forEach");
682    }
683    w.close("dgm:layoutNode");
684    w.close("dgm:layoutDef");
685    w.finish()
686}

700/// `ppt/diagrams/quickStyleN.xml`: flat shapes drawn from the theme's first styles.
701pub fn style_xml(style: &str) -> Vec<u8> {
702    // Brick Scene looks at the diagram from above and to the side.
703    let camera = if style == "brick" { "isometricOffAxis2Top" } else { "orthographicFront" };
704    let mut w = W::new();

710    w.close("dgm:catLst");
711    w.open0("dgm:scene3d");
712    w.empty("a:camera", A::new().a("prst", camera));
713    w.empty("a:lightRig", A::new().a("rig", "threePt").a("dir", "t"));
714    w.close("dgm:scene3d");
715    w.open("dgm:styleLbl", A::new().a("name", "node1"));
716    w.open0("dgm:scene3d");
717    w.empty("a:camera", A::new().a("prst", camera));
718    w.empty("a:lightRig", A::new().a("rig", "threePt").a("dir", "t"));
719    w.close("dgm:scene3d");
720    match bevel(style) {
721        Some((prst, w_emu)) => {
722            w.open0("dgm:sp3d");
723            w.empty("a:bevelT", A::new().a("w", w_emu.to_string()).a("h", (w_emu * 3 / 5).to_string()).a("prst", prst));
724            w.close("dgm:sp3d");
725        }
726        None => w.empty0("dgm:sp3d"),
727    }
728    w.empty("dgm:txPr", A::new());
729    w.open0("dgm:style");
730    for (tag, idx) in [("a:lnRef", "2"), ("a:fillRef", "1"), ("a:effectRef", "0")] {

732        w.empty("a:scrgbClr", A::new().a("r", "0").a("g", "0").a("b", "0"));
733        w.close(tag);
734    }
735    w.open("a:fontRef", A::new().a("idx", "minor"));
736    w.empty("a:schemeClr", A::new().a("val", "lt1"));
737    w.close("a:fontRef");
738    w.close("dgm:style");
739    w.close("dgm:styleLbl");
740    w.close("dgm:styleDef");
741    w.finish()
742}

744/// `ppt/diagrams/colorsN.xml`: accent 1 fills with light text.
745pub fn colors_xml(colors: &str) -> Vec<u8> {
746    let mut w = W::new();
747    w.open("dgm:colorsDef", A::new().a("xmlns:dgm", NS_DGM).a("xmlns:a", crate::pptx::opc::NS_A).a("uniqueId", cs_id(colors)));
748    w.empty("dgm:title", A::new().a("val", ""));
749    w.empty("dgm:desc", A::new().a("val", ""));
750    let accents = fill_accents(colors);
751    let single = accents.len() == 1;
752    let cat = if single { accents.first().map(String::as_str).unwrap_or("accent1") } else { "colorful" };
753    w.open0("dgm:catLst");
754    w.empty("dgm:cat", A::new().a("type", cat).a("pri", "11200"));
755    w.close("dgm:catLst");
756    w.open("dgm:styleLbl", A::new().a("name", "node1"));
757    if colors.ends_with("Grad") {
758        // Gradient Range: the accent shaded darker across the shapes.
759        w.open("dgm:fillClrLst", A::new().a("meth", "repeat"));
760        for k in 0..4 {
761            w.open("a:schemeClr", A::new().a("val", cat));
762            w.empty("a:lumMod", A::new().a("val", (100_000 - k * 15_000).to_string()));
763            w.close("a:schemeClr");
764        }
765        w.close("dgm:fillClrLst");
766    } else {
767        w.open("dgm:fillClrLst", A::new().a("meth", "repeat"));
768        for a in &accents {
769            w.empty("a:schemeClr", A::new().a("val", a.as_str()));
770        }
771        w.close("dgm:fillClrLst");
772    }
773    w.open("dgm:linClrLst", A::new().a("meth", "repeat"));
774    w.empty("a:schemeClr", A::new().a("val", "lt1"));
775    w.close("dgm:linClrLst");
776    w.empty0("dgm:effectClrLst");
777    w.empty0("dgm:txLinClrLst");
778    w.empty0("dgm:txFillClrLst");
779    w.empty0("dgm:txEffectClrLst");
780    w.close("dgm:styleLbl");
781    w.close("dgm:colorsDef");
782    w.finish()
783}
