1//! Just enough of the Compound File Binary (CFB/OLE2) container to tell a password-protected
2//! presentation apart: PowerPoint wraps an encrypted `.pptx` in a CFB holding `EncryptionInfo`
3//! and `EncryptedPackage` streams (ECMA-376 / [MS-OFFCRYPTO]) instead of a zip. Every read is
4//! bounds-checked and every loop bounded. Copied from GridCraft's `crates/xlsx/src/cfb.rs`.
5
6/// The CFB signature at the start of the file.
7const SIGNATURE: [u8; 8] = [0xD0, 0xCF, 0x11, 0xE0, 0xA1, 0xB1, 0x1A, 0xE1];
8
9const END_OF_CHAIN: u32 = 0xFFFF_FFFE;
10/// Sector ids at or above this are markers (free, end of chain, FAT/DIFAT sector), not sectors.
11const MAX_REG_SECT: u32 = 0xFFFF_FFFA;
12/// Directory sectors followed at most (4096 entries with 512-byte sectors).
13const MAX_DIR_SECTORS: usize = 1024;
14/// DIFAT sectors followed at most.
15const MAX_DIFAT_SECTORS: usize = 1024;
16
17/// Stream names that mark an encrypted (password-protected) OOXML package.
18const ENCRYPTED_STREAMS: [&str; 2] = ["EncryptedPackage", "EncryptionInfo"];
19
20fn u16_at(b: &[u8], at: usize) -> Option<u16> {
21    let s = b.get(at..at.checked_add(2)?)?;
22    Some(u16::from_le_bytes([*s.first()?, *s.get(1)?]))
23}
24
25fn u32_at(b: &[u8], at: usize) -> Option<u32> {
26    let s = b.get(at..at.checked_add(4)?)?;
27    Some(u32::from_le_bytes([*s.first()?, *s.get(1)?, *s.get(2)?, *s.get(3)?]))
28}
29
30/// Whether `bytes` is a compound file holding an encryption stream. Walks the directory through
31/// the FAT; when the structure can't be followed (truncated or malformed), falls back to scanning
32/// the bytes for the UTF-16LE stream names.
33pub(crate) fn is_encrypted(bytes: &[u8]) -> bool {
34    if !bytes.starts_with(&SIGNATURE) {
35        return false;
36    }
37    match directory_names(bytes) {
38        Some(names) => names.iter().any(|n| ENCRYPTED_STREAMS.iter().any(|e| n.eq_ignore_ascii_case(e))),
39        None => ENCRYPTED_STREAMS.iter().any(|e| contains_utf16(bytes, e)),
40    }
41}
42
43fn contains_utf16(hay: &[u8], needle: &str) -> bool {
44    let n: Vec<u8> = needle.encode_utf16().flat_map(u16::to_le_bytes).collect();
45    !n.is_empty() && hay.windows(n.len()).any(|w| w == n.as_slice())
46}

48/// Names of every directory entry, or `None` when the directory can't be read.
49fn directory_names(b: &[u8]) -> Option<Vec<String>> {
50    if !b.starts_with(&SIGNATURE) {
51        return None;
52    }
53    let shift = u16_at(b, 30)?;
54    if shift != 9 && shift != 12 {
55        return None;
56    }
57    let size = 1usize << shift;
58    let per = size / 4;
59    let sector = |id: u32| -> Option<&[u8]> {
60        let start = (id as usize).checked_add(1)?.checked_mul(size)?;
61        b.get(start..start.checked_add(size)?)
62    };
63    // The DIFAT: the FAT's sector ids, 109 in the header then chained DIFAT sectors.
64    let mut difat: Vec<u32> = (0..109).filter_map(|i| u32_at(b, 76 + i * 4)).filter(|&s| s < MAX_REG_SECT).collect();
65    let mut next = u32_at(b, 68)?;
66    for _ in 0..MAX_DIFAT_SECTORS {
67        if next >= MAX_REG_SECT {
68            break;
69        }
70        let s = sector(next)?;
71        difat.extend((0..per - 1).filter_map(|i| u32_at(s, i * 4)).filter(|&x| x < MAX_REG_SECT));
72        next = u32_at(s, (per - 1) * 4)?;
73    }
74    let fat_next = |id: u32| -> Option<u32> {
75        let idx = id as usize;
76        let fat_sector = *difat.get(idx / per)?;
77        u32_at(sector(fat_sector)?, (idx % per) * 4)
78    };
79    let mut names = Vec::new();
80    let mut id = u32_at(b, 48)?;
81    for _ in 0..MAX_DIR_SECTORS {
82        if id == END_OF_CHAIN {
83            return Some(names);
84        }
85        if id >= MAX_REG_SECT {
86            return None;
87        }
88        let s = sector(id)?;
89        for e in s.as_chunks::<128>().0 {
90            let len = usize::from(u16_at(e, 64)?);
91            if !(2..=64).contains(&len) {
92                continue;
93            }
94            let units: Vec<u16> = (0..(len - 2) / 2).filter_map(|i| u16_at(e, i * 2)).collect();
95            names.push(String::from_utf16_lossy(&units));
96        }
97        id = fat_next(id)?;
98    }
99    // A directory longer than the bound (or a cyclic chain): report what was read.
100    Some(names)
101}

109    /// A minimal valid compound file (512-byte sectors: FAT at sector 0, directory at sector 1)
110    /// whose directory holds a root entry plus the given stream names.
111    fn build(streams: &[&str]) -> Vec<u8> {
112        let mut b = vec![0u8; 512 * 3];
113        b[..8].copy_from_slice(&SIGNATURE);
114        b[24..26].copy_from_slice(&0x3Eu16.to_le_bytes()); // minor version
115        b[26..28].copy_from_slice(&3u16.to_le_bytes()); // major version
116        b[28..30].copy_from_slice(&0xFFFEu16.to_le_bytes()); // byte order
117        b[30..32].copy_from_slice(&9u16.to_le_bytes()); // sector shift
118        b[32..34].copy_from_slice(&6u16.to_le_bytes()); // mini sector shift
119        b[44..48].copy_from_slice(&1u32.to_le_bytes()); // FAT sectors
120        b[48..52].copy_from_slice(&1u32.to_le_bytes()); // first directory sector
121        b[60..64].copy_from_slice(&END_OF_CHAIN.to_le_bytes()); // no mini FAT
122        b[68..72].copy_from_slice(&END_OF_CHAIN.to_le_bytes()); // no DIFAT sectors
123        for i in 0..109 {
124            b[76 + i * 4..80 + i * 4].copy_from_slice(&0xFFFF_FFFFu32.to_le_bytes());
125        }
126        b[76..80].copy_from_slice(&0u32.to_le_bytes()); // FAT lives in sector 0
127        // FAT (sector 0, at byte 512): sector 0 is a FAT sector, sector 1 ends its chain.
128        for i in 0..128 {
129            b[512 + i * 4..516 + i * 4].copy_from_slice(&0xFFFF_FFFFu32.to_le_bytes());
130        }
131        b[512..516].copy_from_slice(&0xFFFF_FFFDu32.to_le_bytes());
132        b[516..520].copy_from_slice(&END_OF_CHAIN.to_le_bytes());
133        // Directory (sector 1, at byte 1024).
134        for (k, name) in std::iter::once("Root Entry").chain(streams.iter().copied()).take(4).enumerate() {
135            let e = 1024 + k * 128;
136            let units: Vec<u16> = name.encode_utf16().take(31).collect();
137            for (i, u) in units.iter().enumerate() {
138                b[e + i * 2..e + i * 2 + 2].copy_from_slice(&u.to_le_bytes());
139            }
140            b[e + 64..e + 66].copy_from_slice(&((units.len() as u16 + 1) * 2).to_le_bytes());
141            b[e + 66] = if k == 0 { 5 } else { 2 };
142        }
143        b
144    }

146    #[test]
147    fn reads_directory_names() {
148        let b = build(&["EncryptionInfo", "EncryptedPackage"]);
149        assert_eq!(directory_names(&b), Some(vec!["Root Entry".into(), "EncryptionInfo".into(), "EncryptedPackage".into()]));
150        assert!(is_encrypted(&b));
151        // A PowerPoint 97-2003 .ppt is a compound file too, without the encryption streams.
152        assert!(!is_encrypted(&build(&["PowerPoint Document", "Current User"])));
153        assert!(!is_encrypted(b"PK\x03\x04"));
154    }

156    #[test]
157    fn hostile_headers_never_panic() {
158        let mut b = build(&["PowerPoint Document"]);
159        for at in [30usize, 48, 68, 76, 512, 516, 1024 + 64] {
160            for v in [0u32, 1, 7, 0xFFFF_FFFA, 0xFFFF_FFFF, 0x7FFF_FFFF] {
161                let mut c = b.clone();
162                c[at..at + 4].copy_from_slice(&v.to_le_bytes());
163                let _ = is_encrypted(&c);
164                for cut in [0usize, 8, 100, 600, 1100] {
165                    let _ = is_encrypted(&c[..cut.min(c.len())]);
166                }
167            }
168        }
169        // A directory chain that loops back on itself stops at the bound.
170        b[516..520].copy_from_slice(&1u32.to_le_bytes());
171        assert!(!is_encrypted(&b));
172    }
