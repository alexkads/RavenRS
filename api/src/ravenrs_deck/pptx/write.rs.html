85fn ext_for(content_type: &str, name: &str, data: &[u8]) -> String {
86    let by_ct = match content_type {
87        "image/png" => "png",
88        "image/jpeg" | "image/jpg" => "jpeg",
89        "image/gif" => "gif",
90        "image/bmp" => "bmp",
91        "image/tiff" => "tiff",
92        "image/svg+xml" => "svg",
93        "image/x-emf" | "image/emf" => "emf",
94        "image/x-wmf" | "image/wmf" => "wmf",
95        "image/webp" => "webp",
96        "video/mp4" => "mp4",
97        "video/quicktime" => "mov",
98        "video/x-ms-wmv" => "wmv",
99        "video/x-msvideo" => "avi",
100        "video/webm" => "webm",
101        "audio/mpeg" => "mp3",
102        "audio/wav" | "audio/x-wav" | "audio/wave" => "wav",
103        "audio/mp4" | "audio/x-m4a" => "m4a",
104        "audio/x-ms-wma" => "wma",
105        "audio/ogg" => "ogg",
106        "audio/flac" => "flac",
107        "model/gltf-binary" => "glb",
108        _ => "",
109    };
110    if !by_ct.is_empty() {
111        return by_ct.to_string();
112    }
113    if data.starts_with(b"\x89PNG") {
114        return "png".into();
115    }
116    if data.starts_with(&[0xFF, 0xD8]) {
117        return "jpeg".into();
118    }
119    if data.starts_with(b"GIF8") {
120        return "gif".into();
121    }
122    let e = name.rsplit_once('.').map(|(_, e)| e.to_ascii_lowercase()).unwrap_or_default();
123    if !e.is_empty() && e.len() <= 5 && e.chars().all(|c| c.is_ascii_alphanumeric()) { e } else { "bin".into() }
124}

127    /// The media item `id`, found through the index rather than a scan of every item.
128    pub fn media(&self, id: MediaId) -> Option<&'a crate::model::MediaItem> {
129        let p: &'a Presentation = self.p;
130        self.media_at.get(&id).and_then(|&i| p.media.get(i))
131    }
132    fn media_part(&mut self, id: MediaId) -> Option<String> {
133        if let Some(p) = self.media_parts.get(&id) {
134            return Some(p.clone());
135        }
136        let m = self.media(id)?;
137        if m.data.is_empty() {
138            return None;
139        }
140        let ext = ext_for(&m.content_type, &m.name, &m.data);
141        let ct = if m.content_type.is_empty() || !m.content_type.contains('/') {
142            crate::pptx::mime_for(&format!("x.{ext}")).to_string()
143        } else {
144            m.content_type.clone()
145        };
146        let name = if ct.starts_with("image/") {
147            self.n_image += 1;
148            format!("ppt/media/image{}.{ext}", self.n_image)
149        } else {
150            self.n_media += 1;
151            format!("ppt/media/media{}.{ext}", self.n_media)
152        };
153        self.pkg.binary(&name, &ct, m.data.as_ref().clone());
154        self.media_parts.insert(id, name.clone());
155        Some(name)
156    }
157    /// A relationship from `o` to a media item (embedded part, or external link).
158    pub fn media_rel(&mut self, o: &mut Out, id: MediaId, kind: &str) -> Option<String> {
159        let m = self.media(id)?;
160        if m.data.is_empty() {
161            let link = m.link.clone()?;
162            return Some(o.rels.add(kind, &link, true));
163        }
164        let part = self.media_part(id)?;
165        Some(o.rels.add(kind, &opc::relative(&o.name, &part), false))
166    }
167    pub fn media_rel_new(&mut self, o: &mut Out, id: MediaId, kind: &str) -> Option<String> {
168        let m = self.media(id)?;
169        if m.data.is_empty() {
170            return None;
171        }
172        let part = self.media_part(id)?;
173        Some(o.rels.add(kind, &opc::relative(&o.name, &part), false))
174    }

252fn background(w: &mut W, x: &mut Exp, o: &mut Out, bg: &Background) {
253    w.open0("p:bg");
254    match bg {
255        Background::Fill { fill } => {
256            w.open0("p:bgPr");
257            match fill {
258                crate::model::Fill::Background | crate::model::Fill::Group => w.empty0("a:noFill"),
259                f => dml::fill(w, x, o, f),
260            }
261            w.empty0("a:effectLst");
262            w.close("p:bgPr");
263        }
264        Background::Ref { idx, color } => {
265            let idx = if (1..=3).contains(idx) || (1001..=1003).contains(idx) { *idx } else { 1001 };
266            w.open("p:bgRef", A::new().a("idx", idx));
267            dml::color(w, color);
268            w.close("p:bgRef");
269        }
270    }
271    w.close("p:bg");
272}

323/// The twelve colour map attributes, each a theme slot (the default for anything else).
324fn clr_map_attrs(map: &[(String, String)]) -> A {
325    let mut a = A::new();
326    for (k, d) in CLR_KEYS {
327        let v = map
328            .iter()
329            .find(|(mk, _)| mk == k)
330            .map(|(_, v)| v.as_str())
331            .filter(|v| crate::color::SchemeSlot::from_xml(v).is_some_and(|s| crate::color::SchemeSlot::THEME.contains(&s)))
332            .unwrap_or(d);
333        a = a.a(k, v);
334    }
335    a
336}

338fn ext_list(w: &mut W, raw: Option<&str>) {
339    let Some(raw) = raw.filter(|r| !r.trim().is_empty()) else { return };
340    // Only keep well-formed extension elements.
341    let Ok(d) = crate::pptx::xml::parse(format!("<x>{raw}</x>").as_bytes()) else { return };
342    let exts: Vec<String> = d.root.elements().filter(|e| e.is("ext") && e.has_attr("uri")).map(|e| e.to_xml()).collect();
343    if exts.is_empty() {
344        return;
345    }
346    w.open0("p:extLst");
347    for e in exts {
348        w.raw(&e);
349    }
350    w.close("p:extLst");
351}

353/// Strip an `<p:extLst>` wrapper (masters/layouts keep the whole element).
354fn ext_children(raw: Option<&str>) -> Option<String> {
355    let raw = raw?;
356    let d = crate::pptx::xml::parse(raw.as_bytes()).ok()?;
357    if d.root.is("extLst") {

359        return Some(d.root.elements().map(|e| crate::pptx::read::keep_xml(e, &ns)).collect());
360    }
361    Some(raw.to_string())
362}

364/// The slide's extension list, plus the modern-comments anchor when the slide has a threaded
365/// comment part: the rel reference is how PowerPoint finds it.
366fn slide_ext_list(w: &mut W, raw: Option<&str>, comment_rel: Option<&str>) {
367    let mut exts: Vec<String> = raw
368        .filter(|r| !r.trim().is_empty())
369        .and_then(|raw| crate::pptx::xml::parse(format!("<x>{raw}</x>").as_bytes()).ok())
370        .map(|d| d.root.elements().filter(|e| e.is("ext") && e.has_attr("uri")).map(|e| e.to_xml()).collect())
371        .unwrap_or_default();
372    if let Some(rid) = comment_rel {
373        let mut e = W::frag();
374        e.open("p:ext", A::new().a("uri", crate::pptx::COMMENTS_EXT_URI));
375        e.empty("p188:commentRel", A::new().a("xmlns:p188", NS_P188).a("r:id", rid));
376        e.close("p:ext");
377        exts.push(e.s);
378    }
379    if exts.is_empty() {
380        return;
381    }
382    w.open0("p:extLst");
383    for e in exts {
384        w.raw(&e);
385    }
386    w.close("p:extLst");
387}

427fn notes_master_write(p: &Presentation) -> Option<NotesMasterWrite> {
428    let nm = p.notes_master.as_deref();
429    let kept = nm.filter(|m| !m.shapes.is_empty());
430    let has_notes = p.slides.iter().any(|s| !s.notes.is_empty());
431    if !has_notes && nm.is_none() {
432        return None;
433    }
434    let default_bg = || Background::Ref { idx: 1001, color: crate::model::ColorRef::scheme(crate::color::SchemeSlot::Bg1) };
435    Some(match kept {
436        Some(m) => NotesMasterWrite {
437            shapes: m.shapes.clone(),
438            background: Some(m.background.clone().unwrap_or_else(default_bg)),
439            color_map: m.color_map.clone(),
440            style: if m.body_style.is_empty() { notes_style() } else { m.body_style.clone() },
441            theme: Some(m.theme.clone()),
442        },
443        // A master without placeholders of its own gets PowerPoint's defaults, keeping its
444        // background and colour map.
445        None => NotesMasterWrite {
446            shapes: notes_master_shapes(p.notes_size),
447            background: Some(nm.and_then(|m| m.background.clone()).unwrap_or_else(default_bg)),
448            color_map: nm.map(|m| m.color_map.clone()).unwrap_or_default(),
449            style: notes_style(),
450            theme: None,
451        },
452    })
453}

478/// A plausible xsd:dateTime (`YYYY-MM-DD[THH:MM:SS(.frac)?(Z|±HH:MM)?]`): the ranges are
479/// checked, so a model value like `2025-13-45T99:99` falls back to the default instead of
480/// reaching a required `dt`/`created` attribute.
481fn valid_date(s: &str) -> bool {
482    let b = s.as_bytes();
483    let n = |r: std::ops::Range<usize>| s.get(r).and_then(|v| v.parse::<u32>().ok());
484    let (Some(_), Some(mo), Some(d)) = (n(0..4), n(5..7), n(8..10)) else { return false };
485    if b.get(4) != Some(&b'-') || b.get(7) != Some(&b'-') || !(1..=12).contains(&mo) || !(1..=31).contains(&d) {
486        return false;
487    }
488    match b.get(10) {
489        None => true,
490        Some(b'T') => match (n(11..13), n(14..16), n(17..19)) {
491            (Some(h), Some(mi), Some(sec)) => h <= 23 && mi <= 59 && sec <= 60 && b.get(13) == Some(&b':') && b.get(16) == Some(&b':'),
492            _ => false,
493        },
494        Some(_) => false,
495    }
496}

498/// A copy of the presentation with every slide pointing at a layout that exists.
499fn sanitized(p: &Presentation) -> std::borrow::Cow<'_, Presentation> {
500    let mut need = p.masters.is_empty() || p.masters.iter().any(|m| m.layouts.is_empty());
501    let layouts: std::collections::HashSet<_> = p.masters.iter().flat_map(|m| m.layouts.iter().map(|l| l.id)).collect();
502    need |= p.slides.iter().any(|s| !layouts.contains(&s.layout));
503    if !need {
504        return std::borrow::Cow::Borrowed(p);
505    }
506    let mut q = p.clone();
507    if q.masters.is_empty() {
508        let (m, next) = defaults::build_master(q.slide_size, Default::default(), q.next_id.max(1));
509        q.next_id = next;
510        q.masters.push(Arc::new(m));
511    }
512    for m in q.masters.iter_mut() {
513        if m.layouts.is_empty() {
514            // An exhausted id space exports without the synthesized layout rather than with a
515            // duplicated one (#407).
516            let Some(raw) = q.next_id.max(1).checked_add(1) else { continue };
517            let id = LayoutId(raw);
518            q.next_id = raw;
519            Arc::make_mut(m).layouts.push(Layout {
520                id,
521                name: "Blank".into(),
522                kind: crate::model::LayoutType::Blank,
523                show_master_shapes: true,
524                ..Default::default()
525            });
526        }
527    }
528    let first = q.masters.first().and_then(|m| m.layouts.first()).map(|l| l.id).unwrap_or_default();
529    let ok: Vec<LayoutId> = q.masters.iter().flat_map(|m| m.layouts.iter().map(|l| l.id)).collect();
530    for s in q.slides.iter_mut() {
531        if !ok.contains(&s.layout) {
532            Arc::make_mut(s).layout = first;
533        }
534    }
535    std::borrow::Cow::Owned(q)
536}

594        x.pkg.xml_part(&tname, CT_THEME, txml);
595        x.pkg.rels(&tname, &to.rels);

734        .iter()
735        .filter(|k| {

748    }
749    x.pkg.rels("ppt/presentation.xml", &pres.rels);
750    for k in &kept {
751        if k.content_type.is_empty() {
752            x.pkg.raw(&k.name, k.xml.clone().into_bytes());
753        } else {
754            x.pkg.xml_part(&k.name, &k.content_type, k.xml.clone().into_bytes());
755        }
756    }
757
758    // Document properties and package relationships.
759    x.pkg.xml_part("docProps/core.xml", CT_CORE, core_xml(p));
760    x.pkg.xml_part("docProps/app.xml", CT_APP, app_xml(p));
761    let mut root = RelsOut::default();
762    root.add("officeDocument", "ppt/presentation.xml", false);
763    root.add(opc::RT_CORE, "docProps/core.xml", false);
764    root.add("extended-properties", "docProps/app.xml", false);
765    for k in kept.iter().filter(|k| !k.rel_type.is_empty() && k.from_root) {
766        root.add_new(&k.rel_type, &k.name, false);
767    }
768    x.pkg.rels("", &root);
769    x.pkg.finish()
770}

850fn comments_xml(list: &[Comment], authors: &mut Vec<(String, String, u32)>, author_map: &mut std::collections::HashMap<String, usize>) -> Vec<u8> {
851    let mut w = W::new();
852    root_open(&mut w, "p:cmLst", A::new());
853    let mut flat: Vec<&Comment> = vec![];
854    for c in list.iter().take(10_000) {
855        flat.push(c);
856        for r in c.replies.iter().take(1000) {
857            flat.push(r);
858        }
859    }
860    for c in flat {
861        let ai = author_slot(c, authors, author_map);
862        let idx = match authors.get_mut(ai) {
863            Some(a) => {
864                a.2 += 1;
865                a.2
866            }
867            None => 1,
868        };
869        let dt = if valid_date(&c.date) { c.date.clone() } else { "2024-01-01T00:00:00.000".into() };
870        w.open("p:cm", A::new().a("authorId", ai).a("dt", dt).a("idx", idx));
871        let pos = |v: f64| if v.is_finite() { (v * 8.0).round().clamp(0.0, 1.0e9) as i64 } else { 0 };
872        w.empty("p:pos", A::new().a("x", pos(c.x)).a("y", pos(c.y)));
873        w.elt("p:text", &c.text);
874        w.close("p:cm");
875    }
876    w.close("p:cmLst");
877    w.finish()
878}

880/// The author's slot in the shared bucket list, adding them on first use. Legacy parts index the
881/// list by number, the modern authors part gives every slot a GUID.
882fn author_slot(c: &Comment, authors: &mut Vec<(String, String, u32)>, map: &mut std::collections::HashMap<String, usize>) -> usize {
883    let author = if c.author.is_empty() { "Author" } else { &c.author };
884    if let Some(&i) = map.get(author) {
885        return i;
886    }
887    let ini = if c.initials.is_empty() { author.chars().filter(|ch| ch.is_alphabetic()).take(2).collect() } else { c.initials.clone() };
888    let idx = authors.len();
889    authors.push((author.to_string(), ini, 0));
890    map.insert(author.to_string(), idx);
891    idx
892}

911fn modern_comment(
912    w: &mut W,
913    c: &Comment,
914    part: &str,
915    ti: usize,
916    authors: &mut Vec<(String, String, u32)>,
917    author_map: &mut std::collections::HashMap<String, usize>,
918    depth: usize,
919) {
920    let ai = author_slot(c, authors, author_map);
921    let author = authors.get(ai).map(|(n, _, _)| n.clone()).unwrap_or_default();
922    let dt = if valid_date(&c.date) { c.date.clone() } else { "2024-01-01T00:00:00.000".into() };
923    let mut a =
924        A::new().a("id", crate::pptx::tables::guid(&format!("deckcraft-comment:{part}:{ti}"))).a("authorId", author_guid(&author)).a("created", dt);
925    if c.resolved {
926        a = a.a("status", "resolved");
927    }
928    w.open("p188:cm", a);
929    w.empty0("p188:unknownAnchor");
930    let emu = |v: f64| if v.is_finite() { (v * 12_700.0).round().clamp(0.0, 1.0e9) as i64 } else { 0 };
931    w.empty("p188:pos", A::new().a("x", emu(c.x)).a("y", emu(c.y)));
932    if depth < 4 && !c.replies.is_empty() {
933        w.open0("p188:replyLst");
934        for (ri, r) in c.replies.iter().take(1000).enumerate() {
935            let ri_ai = author_slot(r, authors, author_map);
936            let r_author = authors.get(ri_ai).map(|(n, _, _)| n.clone()).unwrap_or_default();
937            let rdt = if valid_date(&r.date) { r.date.clone() } else { "2024-01-01T00:00:00.000".into() };
938            let mut ra = A::new()
939                .a("id", crate::pptx::tables::guid(&format!("deckcraft-comment:{part}:{ti}:{ri}")))
940                .a("authorId", author_guid(&r_author))
941                .a("created", rdt);
942            if r.resolved {
943                ra = ra.a("status", "resolved");
944            }
945            w.open("p188:reply", ra);
946            tx_body(w, &r.text);
947            w.close("p188:reply");
948        }
949        w.close("p188:replyLst");
950    }
951    tx_body(w, &c.text);
952    w.close("p188:cm");
953}

975/// Sections with every slide in exactly one section, in slide order.
976fn section_slides(p: &Presentation) -> Vec<(String, Vec<usize>)> {
977    let mut out: Vec<(String, Vec<usize>)> = p.sections.iter().map(|s| (s.name.clone(), vec![])).collect();
978    if out.is_empty() {
979        return out;
980    }
981    // Each slide's first section, found once: a section may name a slide many times.
982    let mut first: HashMap<SlideId, usize> = HashMap::new();
983    for (k, sec) in p.sections.iter().enumerate() {
984        for id in &sec.slides {
985            first.entry(*id).or_insert(k);
986        }
987    }
988    let mut cur = 0usize;
989    for (i, s) in p.slides.iter().enumerate() {
990        if let Some(&k) = first.get(&s.id)
991            && k >= cur
992        {
993            cur = k;
994        }
995        if let Some(sec) = out.get_mut(cur) {
996            sec.1.push(i);
997        }
998    }
999    out
1000}

1020/// Font parts for `p:embeddedFontLst`. One part per distinct media item, named `ppt/fonts/fontN.fntdata`.
1021fn write_embedded_fonts(x: &mut Exp, o: &mut Out) -> Vec<WrittenFont> {
1022    struct Plan {
1023        typeface: String,
1024        panose: Option<String>,
1025        pitch_family: Option<u8>,
1026        charset: Option<u8>,
1027        regular: Option<MediaId>,
1028        bold: Option<MediaId>,
1029        italic: Option<MediaId>,
1030        bold_italic: Option<MediaId>,
1031    }
1032    let mut plans = Vec::new();
1033    for font in x.p.embedded_fonts.iter().take(64) {
1034        let typeface = font.typeface.trim();
1035        if typeface.is_empty() {
1036            continue;
1037        }
1038        plans.push(Plan {
1039            typeface: typeface.chars().take(255).collect(),
1040            panose: font.panose.clone().filter(|s| s.len() == 20 && s.chars().all(|c| c.is_ascii_hexdigit())),
1041            pitch_family: font.pitch_family,
1042            charset: font.charset,
1043            regular: font.regular,
1044            bold: font.bold,
1045            italic: font.italic,
1046            bold_italic: font.bold_italic,
1047        });
1048    }
1049    let mut seen_part: HashMap<MediaId, String> = HashMap::new();
1050    let mut n = 0u32;
1051    let mut rel = |x: &mut Exp, o: &mut Out, id: Option<MediaId>| -> Option<String> {
1052        let id = id?;
1053        if let Some(rid) = seen_part.get(&id) {
1054            return Some(rid.clone());
1055        }
1056        let m = x.media(id)?;
1057        if m.data.is_empty() || m.data.len() > 32 * 1024 * 1024 {
1058            return None;
1059        }
1060        let bytes = m.data.as_ref().clone();
1061        n = n.saturating_add(1);
1062        let name = format!("ppt/fonts/font{n}.fntdata");
1063        x.pkg.binary(&name, "application/x-fontdata", bytes);
1064        let rid = o.rels.add("font", &opc::relative("ppt/presentation.xml", &name), false);
1065        seen_part.insert(id, rid.clone());
1066        Some(rid)
1067    };
1068    let mut seen_face = Vec::new();
1069    let mut out = Vec::new();
1070    for font in plans {
1071        let written = WrittenFont {
1072            typeface: font.typeface,
1073            panose: font.panose,
1074            pitch_family: font.pitch_family,
1075            charset: font.charset,
1076            regular: rel(x, o, font.regular),
1077            bold: rel(x, o, font.bold),
1078            italic: rel(x, o, font.italic),
1079            bold_italic: rel(x, o, font.bold_italic),
1080        };
1081        if written.any() && !seen_face.iter().any(|s| s == &written.typeface) {
1082            seen_face.push(written.typeface.clone());
1083            out.push(written);
1084        }
1085    }
1086    out
1087}

1089fn presentation_xml(x: &mut Exp, o: &mut Out, masters: &[(u64, String)], notes: Option<&str>, handout: Option<&str>, slides: &[String]) -> Vec<u8> {
1090    let fonts = write_embedded_fonts(x, o);

1095    w.open0("p:sldMasterIdLst");
1096    for (id, rid) in masters {

1129                if let Some(rid) = rid {

1144                    let rid = o.rels.add("slide", &opc::relative("ppt/presentation.xml", &part), false);

1165                inner.open0("p14:sldIdLst");

1167                    inner.empty("p14:sldId", A::new().a("id", 256 + si));
1168                }
1169                inner.close("p14:sldIdLst");
1170                inner.close("p14:section");
1171            }
1172            inner.close("p14:sectionLst");
1173            inner.close("p:ext");
1174        }
1175        if let Some(r) = raw {
1176            inner.raw(r);
1177        }
1178        ext_list(&mut w, Some(&inner.s));
