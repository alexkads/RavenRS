1//! A small, bounded XML DOM over quick-xml for reading parts, plus an escaping string builder for
2//! writing them.
3//!
4//! Reading is lenient about namespaces: elements and attributes are matched by local name, which is
5//! how real-world files are best understood (prefixes are conventional and stable in practice).
6
7use std::fmt::Write as _;

13/// Maximum element nesting accepted when parsing.
14pub const MAX_DEPTH: usize = 256;
15/// Maximum number of elements in one part.
16pub const MAX_NODES: usize = 4_000_000;
17
18#[derive(Clone, Debug, Default, PartialEq)]
19pub struct El {
20    /// Qualified name as written (`p:sp`).
21    pub name: String,
22    pub attrs: Vec<(String, String)>,
23    pub children: Vec<Node>,
24}
25
26#[derive(Clone, Debug, PartialEq)]
27pub enum Node {
28    El(El),
29    Text(String),
30}

32fn local_of(name: &str) -> &str {
33    name.rsplit_once(':').map(|(_, l)| l).unwrap_or(name)
34}
35
36impl El {
37    pub fn local(&self) -> &str {
38        local_of(&self.name)
39    }
40    pub fn prefix(&self) -> &str {
41        self.name.split_once(':').map(|(p, _)| p).unwrap_or("")
42    }
43    pub fn is(&self, local: &str) -> bool {
44        self.local() == local
45    }
46    /// Attribute by local name (`id` matches `id`; `r:id` matches `id` only when no plain `id`
47    /// exists). Pass a qualified name (`r:id`) to require that prefix's local name with any prefix.
48    pub fn attr(&self, name: &str) -> Option<&str> {
49        if let Some((_, l)) = name.split_once(':') {
50            return self.attrs.iter().find(|(k, _)| k.contains(':') && !k.starts_with("xmlns") && local_of(k) == l).map(|(_, v)| v.as_str());
51        }
52        self.attrs
53            .iter()
54            .find(|(k, _)| k == name)
55            .or_else(|| self.attrs.iter().find(|(k, _)| !k.starts_with("xmlns") && local_of(k) == name))
56            .map(|(_, v)| v.as_str())
57    }
58    pub fn has_attr(&self, name: &str) -> bool {
59        self.attr(name).is_some()
60    }
61    pub fn set_attr(&mut self, name: &str, value: &str) {
62        if let Some(a) = self.attrs.iter_mut().find(|(k, _)| k == name) {
63            a.1 = value.to_string();
64        } else {
65            self.attrs.push((name.to_string(), value.to_string()));
66        }
67    }
68    pub fn i64(&self, name: &str) -> Option<i64> {
69        let v = self.attr(name)?.trim();
70        v.parse::<i64>().ok().or_else(|| v.parse::<f64>().ok().filter(|f| f.is_finite()).map(|f| f.clamp(-9.0e18, 9.0e18) as i64))
71    }
72    pub fn i32(&self, name: &str) -> Option<i32> {
73        self.i64(name).map(|v| v.clamp(i32::MIN as i64, i32::MAX as i64) as i32)
74    }
75    pub fn u32(&self, name: &str) -> Option<u32> {
76        self.i64(name).map(|v| v.clamp(0, u32::MAX as i64) as u32)
77    }
78    pub fn f64(&self, name: &str) -> Option<f64> {
79        self.attr(name)?.trim().parse::<f64>().ok().filter(|f| f.is_finite())
80    }
81    pub fn bool(&self, name: &str) -> Option<bool> {
82        match self.attr(name)?.trim() {
83            "1" | "true" | "on" => Some(true),
84            "0" | "false" | "off" => Some(false),
85            _ => None,
86        }
87    }
88    pub fn elements(&self) -> impl Iterator<Item = &El> {
89        self.children.iter().filter_map(|n| match n {
90            Node::El(e) => Some(e),
91            Node::Text(_) => None,
92        })
93    }
94    pub fn elements_mut(&mut self) -> impl Iterator<Item = &mut El> {
95        self.children.iter_mut().filter_map(|n| match n {
96            Node::El(e) => Some(e),
97            Node::Text(_) => None,
98        })
99    }
100    pub fn child(&self, local: &str) -> Option<&El> {
101        self.elements().find(|e| e.is(local))
102    }
103    pub fn children_named<'a>(&'a self, local: &'a str) -> impl Iterator<Item = &'a El> + 'a {
104        self.elements().filter(move |e| e.is(local))
105    }
106    /// Follow a chain of child local names.
107    pub fn path(&self, names: &[&str]) -> Option<&El> {
108        let mut cur = self;
109        for n in names {
110            cur = cur.child(n)?;
111        }
112        Some(cur)
113    }
114    /// First descendant (depth-first, bounded) with this local name.
115    pub fn find(&self, local: &str) -> Option<&El> {
116        fn rec<'a>(e: &'a El, local: &str, depth: usize) -> Option<&'a El> {
117            if depth > MAX_DEPTH {
118                return None;
119            }
120            for c in e.elements() {
121                if c.is(local) {
122                    return Some(c);
123                }
124                if let Some(f) = rec(c, local, depth + 1) {
125                    return Some(f);
126                }
127            }
128            None
129        }
130        rec(self, local, 0)
131    }
132    /// All descendants with this local name (bounded depth).
133    pub fn find_all<'a>(&'a self, local: &str, out: &mut Vec<&'a El>) {
134        fn rec<'a>(e: &'a El, local: &str, depth: usize, out: &mut Vec<&'a El>) {
135            if depth > MAX_DEPTH {
136                return;
137            }
138            for c in e.elements() {
139                if c.is(local) {
140                    out.push(c);
141                }
142                rec(c, local, depth + 1, out);
143            }
144        }
145        rec(self, local, 0, out)
146    }
147    /// Concatenated text content of this element and its descendants.
148    pub fn text(&self) -> String {
149        let mut s = String::new();
150        fn rec(e: &El, s: &mut String, depth: usize) {
151            if depth > MAX_DEPTH {
152                return;
153            }
154            for c in &e.children {
155                match c {
156                    Node::Text(t) => s.push_str(t),
157                    Node::El(e) => rec(e, s, depth + 1),
158                }
159            }
160        }
161        rec(self, &mut s, 0);
162        s
163    }
164    /// Value of `<x val="…"/>` child.
165    pub fn child_val(&self, local: &str) -> Option<&str> {
166        self.child(local).and_then(|c| c.attr("val"))
167    }
168    /// Serialize to a string (no XML declaration).
169    pub fn to_xml(&self) -> String {
170        let mut s = String::new();
171        write_el(self, &mut s, 0);
172        s
173    }
174}
175
176fn write_el(e: &El, s: &mut String, depth: usize) {
177    if depth > MAX_DEPTH {
178        return;
179    }
180    s.push('<');
181    s.push_str(&e.name);
182    for (k, v) in &e.attrs {
183        let _ = write!(s, " {}=\"{}\"", k, esc_attr(v));
184    }
185    if e.children.is_empty() {
186        s.push_str("/>");
187        return;
188    }
189    s.push('>');
190    for c in &e.children {
191        match c {
192            Node::Text(t) => s.push_str(&esc(t)),
193            Node::El(c) => write_el(c, s, depth + 1),
194        }
195    }
196    s.push_str("</");
197    s.push_str(&e.name);
198    s.push('>');
199}

214/// Parse XML bytes into a DOM. Bounded depth and node count; entity references are resolved.
215pub fn parse(bytes: &[u8]) -> Result<Doc, PptxError> {
216    // Skip a UTF-8 BOM; UTF-16 parts (rare) are decoded first.
217    let owned;
218    let bytes = if bytes.starts_with(&[0xFF, 0xFE]) || bytes.starts_with(&[0xFE, 0xFF]) {
219        owned = decode_utf16(bytes);
220        owned.as_bytes()
221    } else {
222        bytes.strip_prefix(&[0xEF, 0xBB, 0xBF]).unwrap_or(bytes)
223    };
224    let mut reader = quick_xml::Reader::from_reader(bytes);
225    {
226        let c = reader.config_mut();
227        c.trim_text(false);
228        c.check_end_names = false;
229        c.allow_unmatched_ends = true;
230        c.expand_empty_elements = false;
231    }
232    let mut stack: Vec<El> = vec![];
233    let mut root: Option<El> = None;
234    let mut nodes = 0usize;
235    let mut buf = Vec::new();
236    loop {
237        let ev = reader.read_event_into(&mut buf).map_err(|e| PptxError::Xml(format!("{e}")))?;
238        match ev {
239            Event::Start(b) | Event::Empty(b) if root.is_some() => {
240                // Content after the root element: ignore.
241                let _ = b;
242            }
243            Event::Start(b) => {
244                nodes += 1;
245                if nodes > MAX_NODES || stack.len() >= MAX_DEPTH {
246                    return Err(PptxError::Xml("document too large or too deeply nested".into()));
247                }
248                stack.push(start_el(&b, &reader));
249            }
250            Event::Empty(b) => {
251                nodes += 1;
252                if nodes > MAX_NODES {
253                    return Err(PptxError::Xml("document too large".into()));
254                }
255                let el = start_el(&b, &reader);
256                match stack.last_mut() {
257                    Some(p) => p.children.push(Node::El(el)),
258                    None => root = Some(el),
259                }
260            }
261            Event::End(_) => {
262                if let Some(el) = stack.pop() {
263                    match stack.last_mut() {
264                        Some(p) => p.children.push(Node::El(el)),
265                        None => root = Some(el),
266                    }
267                }
268            }
269            Event::Text(t) => {
270                if let Some(p) = stack.last_mut() {
271                    let s = t.decode().map(|c| c.into_owned()).unwrap_or_default();
272                    push_text(p, &line_ends(s));
273                }
274            }
275            Event::CData(t) => {
276                if let Some(p) = stack.last_mut() {
277                    let s = String::from_utf8_lossy(&t).into_owned();
278                    push_text(p, &line_ends(s));
279                }
280            }
281            Event::GeneralRef(r) => {
282                if let Some(p) = stack.last_mut() {
283                    let s = if r.is_char_ref() {
284                        r.resolve_char_ref().ok().flatten().map(String::from).unwrap_or_default()
285                    } else {
286                        match r.decode().map(|c| c.into_owned()).unwrap_or_default().as_str() {
287                            "amp" => "&".into(),
288                            "lt" => "<".into(),
289                            "gt" => ">".into(),
290                            "quot" => "\"".into(),
291                            "apos" => "'".into(),
292                            _ => String::new(),
293                        }
294                    };
295                    push_text(p, &s);
296                }
297            }
298            Event::Eof => break,
299            _ => {}
300        }
301        buf.clear();
302    }
303    // Unclosed elements (truncated document): fold them up so partial content is still usable.
304    while let Some(el) = stack.pop() {
305        match stack.last_mut() {
306            Some(p) => p.children.push(Node::El(el)),
307            None => root = Some(el),
308        }
309    }
310    let root = root.ok_or_else(|| PptxError::Xml("no root element".into()))?;
311    Ok(Doc { root })
312}

314/// XML 1.0 §2.11: a literal CR LF or lone CR in the document reads as LF. A `&#13;` reference is
315/// not literal and reaches `push_text` through `GeneralRef`, so it stays a CR.
316fn line_ends(s: String) -> String {
317    if s.contains('\r') { s.replace("\r\n", "\n").replace('\r', "\n") } else { s }
318}
319
320fn push_text(p: &mut El, s: &str) {
321    if s.is_empty() {
322        return;
323    }
324    if let Some(Node::Text(t)) = p.children.last_mut() {
325        t.push_str(s);
326    } else {
327        p.children.push(Node::Text(s.to_string()));
328    }
329}

331fn start_el(b: &quick_xml::events::BytesStart, reader: &quick_xml::Reader<&[u8]>) -> El {
332    let name = String::from_utf8_lossy(b.name().as_ref()).into_owned();
333    let mut attrs = vec![];
334    for a in b.attributes().with_checks(false).flatten() {
335        let k = String::from_utf8_lossy(a.key.as_ref()).into_owned();
336        let v = a
337            .decoded_and_normalized_value(quick_xml::XmlVersion::Implicit1_0, reader.decoder())
338            .map(|c| c.into_owned())
339            .unwrap_or_else(|_| String::from_utf8_lossy(&a.value).into_owned());
340        attrs.push((k, v));
341    }
342    El { name, attrs, children: vec![] }
343}

345fn decode_utf16(bytes: &[u8]) -> String {
346    let le = bytes.starts_with(&[0xFF, 0xFE]);
347    let units: Vec<u16> =
348        bytes.get(2..).unwrap_or(&[]).as_chunks::<2>().0.iter().map(|c| if le { u16::from_le_bytes(*c) } else { u16::from_be_bytes(*c) }).collect();
349    let s = String::from_utf16_lossy(&units);
350    // Drop the encoding declaration so quick-xml reads it as UTF-8.
351    match s.strip_prefix("<?xml").and_then(|r| r.find("?>").map(|i| i + 7)) {
352        Some(end) => s.get(end..).unwrap_or("").to_string(),
353        None => s,
354    }
355}

357/// Is `c` allowed in XML 1.0 text?
358fn xml_char(c: char) -> bool {
359    matches!(c, '\t' | '\n' | '\r') || (c >= ' ' && c != '\u{FFFE}' && c != '\u{FFFF}')
360}
361
362/// Escape text content (and drop characters XML 1.0 forbids).
363pub fn esc(s: &str) -> String {
364    let mut o = String::with_capacity(s.len());
365    for c in s.chars() {
366        match c {
367            '&' => o.push_str("&amp;"),
368            '<' => o.push_str("&lt;"),
369            '>' => o.push_str("&gt;"),
370            '\r' => o.push_str("&#13;"),
371            c if xml_char(c) => o.push(c),
372            _ => {}
373        }
374    }
375    o
376}
377
378/// Escape an attribute value.
379pub fn esc_attr(s: &str) -> String {
380    let mut o = String::with_capacity(s.len());
381    for c in s.chars() {
382        match c {
383            '&' => o.push_str("&amp;"),
384            '<' => o.push_str("&lt;"),
385            '>' => o.push_str("&gt;"),
386            '"' => o.push_str("&quot;"),
387            '\n' => o.push_str("&#10;"),
388            '\r' => o.push_str("&#13;"),
389            '\t' => o.push_str("&#9;"),
390            c if xml_char(c) => o.push(c),
391            _ => {}
392        }
393    }
394    o
395}

403/// Attribute list builder.
404#[derive(Default)]
405pub struct A {
406    pub v: Vec<(&'static str, String)>,
407}
408
409impl A {
410    pub fn new() -> Self {
411        A { v: vec![] }
412    }
413    pub fn a(mut self, k: &'static str, v: impl ToString) -> Self {
414        self.v.push((k, v.to_string()));
415        self
416    }
417    pub fn o<T: ToString>(mut self, k: &'static str, v: Option<T>) -> Self {
418        if let Some(v) = v {
419            self.v.push((k, v.to_string()));
420        }
421        self
422    }
423    /// Boolean attribute written as `1`/`0` when `Some`.
424    pub fn b(mut self, k: &'static str, v: Option<bool>) -> Self {
425        if let Some(v) = v {
426            self.v.push((k, if v { "1" } else { "0" }.to_string()));
427        }
428        self
429    }
430    /// Boolean attribute written only when true.
431    pub fn t(mut self, k: &'static str, v: bool) -> Self {
432        if v {
433            self.v.push((k, "1".to_string()));
434        }
435        self
436    }
437}

439impl W {
440    pub fn new() -> Self {
441        W { s: String::from("<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n") }
442    }
443    pub fn frag() -> Self {
444        W { s: String::new() }
445    }
446    fn attrs(&mut self, a: &A) {
447        for (k, v) in &a.v {
448            let _ = write!(self.s, " {}=\"{}\"", k, esc_attr(v));
449        }
450    }
451    pub fn open(&mut self, name: &str, a: A) {
452        self.s.push('<');
453        self.s.push_str(name);
454        self.attrs(&a);
455        self.s.push('>');
456    }
457    pub fn open0(&mut self, name: &str) {
458        self.s.push('<');
459        self.s.push_str(name);
460        self.s.push('>');
461    }
462    pub fn close(&mut self, name: &str) {
463        self.s.push_str("</");
464        self.s.push_str(name);
465        self.s.push('>');
466    }
467    pub fn empty(&mut self, name: &str, a: A) {
468        self.s.push('<');
469        self.s.push_str(name);
470        self.attrs(&a);
471        self.s.push_str("/>");
472    }
473    pub fn empty0(&mut self, name: &str) {
474        self.s.push('<');
475        self.s.push_str(name);
476        self.s.push_str("/>");
477    }
478    /// `<name val="v"/>`
479    pub fn val(&mut self, name: &str, v: impl ToString) {
480        self.empty(name, A::new().a("val", v));
481    }
482    pub fn text(&mut self, t: &str) {
483        self.s.push_str(&esc(t));
484    }
485    /// `<name>text</name>`
486    pub fn elt(&mut self, name: &str, t: &str) {
487        self.open0(name);
488        self.text(t);
489        self.close(name);
490    }
491    pub fn raw(&mut self, xml: &str) {
492        self.s.push_str(xml);
493    }
494    pub fn finish(self) -> Vec<u8> {
495        self.s.into_bytes()
496    }
497}

505        let d = parse(b"<a:p><a:t>one\r\ntwo\rthree&#13;four</a:t><a:t><![CDATA[x\r\ny]]></a:t></a:p>").expect("parse");
