8/// (our kind, namespace prefix, element) for transitions that map one-to-one.
9const SIMPLE: &[(&str, &str, &str)] =
10    &[("dissolve", "p", "dissolve"), ("random", "p", "random"), ("honeycomb", "p14", "honeycomb"), ("flash", "p14", "flash")];

12/// Read a transition effect element → (kind, option). `None` for effects we don't model.
13pub fn transition_from_xml(e: &El) -> Option<(String, String)> {
14    let (k, o) = transition_from_xml_raw(e)?;
15    // Kinds without options keep none.
16    let has_opts = crate::model::anim::TRANSITIONS.iter().find(|t| t.0 == k).is_none_or(|t| !t.4.is_empty());
17    Some((k, if has_opts { o } else { String::new() }))
18}

20/// Preset transitions (`p15:prstTrans`) whose "Left" option PowerPoint writes with `invX="1"`
21/// (the others write it for "Right"), as observed on PowerPoint's own files.
22const INVX_LEFT: &[&str] = &["wind", "airplane", "origami"];
23
24fn transition_from_xml_raw(e: &El) -> Option<(String, String)> {
25    // Defaults are what PowerPoint leaves out: e.g. it writes Push Left as `<p:push/>`.
26    let dir = |d: &str| e.attr("dir").unwrap_or(d).to_string();
27    let k = |s: &str| s.to_string();
28    Some(match e.local() {
29        "fade" => (k("fade"), if e.bool("thruBlk") == Some(true) { k("throughBlack") } else { k("smoothly") }),
30        "cut" => (k("cut"), if e.bool("thruBlk") == Some(true) { k("throughBlack") } else { String::new() }),
31        "push" => (k("push"), dir("l")),
32        "wipe" => (k("wipe"), dir("l")),
33        // Diagonal wipes.
34        "strips" => (k("wipe"), dir("lu")),
35        "split" => {
36            let o = e.attr("orient").unwrap_or("horz");
37            let d = e.attr("dir").unwrap_or("out");
38            (k("split"), format!("{o}{}", if d == "in" { "In" } else { "Out" }))
39        }
40        "randomBar" => (k("randomBar"), dir("horz")),
41        "circle" => (k("shape"), k("circle")),
42        "diamond" => (k("shape"), k("diamond")),
43        "plus" => (k("shape"), k("plus")),
44        // Shape In / Out (the slide zooming in a box).
45        "zoom" => (k("shape"), if e.attr("dir") == Some("in") { k("in") } else { k("out") }),
46        // The Zoom transition, and its Zoom and Rotate option.
47        "warp" => (k("zoom"), if e.attr("dir") == Some("in") { k("in") } else { k("out") }),
48        "newsflash" => (k("zoom"), k("rotate")),
49        "pull" => (k("uncover"), dir("l")),
50        "cover" => (k("cover"), dir("l")),
51        "checker" => (k("checker"), dir("horz")),
52        "blinds" => (k("blinds"), dir("horz")),
53        "comb" => (k("comb"), dir("horz")),
54        "wheel" => (k("clock"), k("clockwise")),
55        "wheelReverse" => (k("clock"), k("counterClockwise")),
56        "wedge" => (k("clock"), k("wedge")),
57        "vortex" => (k("vortex"), dir("l")),
58        "switch" => (k("switch"), dir("r")),
59        "flip" => (k("flip"), dir("r")),
60        "ripple" => (k("ripple"), dir("center")),
61        "doors" => (k("doors"), dir("horz")),
62        "window" => (k("window"), dir("horz")),
63        "ferris" => (k("ferris"), dir("l")),
64        "gallery" => (k("gallery"), dir("l")),
65        "conveyor" => (k("conveyor"), dir("l")),
66        "pan" => (k("pan"), dir("l")),
67        "glitter" => {
68            let d = dir("l");
69            if e.attr("pattern") == Some("hexagon") { (k("glitter"), format!("hex{}", d.to_ascii_uppercase())) } else { (k("glitter"), d) }
70        }
71        "shred" => {
72            // `crush` is what DeckCraft wrote before it knew PowerPoint's `rectangle`.
73            let particles = matches!(e.attr("pattern"), Some("rectangle" | "crush"));
74            let out = e.attr("dir") == Some("out");
75            (k("shred"), format!("{}{}", if particles { "particles" } else { "strips" }, if out { "Out" } else { "In" }))
76        }
77        "reveal" => {
78            let black = e.bool("thruBlk") == Some(true);
79            let right = e.attr("dir") == Some("r");
80            (k("reveal"), format!("{}{}", if black { "black" } else { "smoothly" }, if right { "Right" } else { "Left" }))
81        }
82        "flythrough" => {
83            let out = e.attr("dir") == Some("out");
84            let b = e.bool("hasBounce") == Some(true);
85            (k("flythrough"), format!("{}{}", if out { "out" } else { "in" }, if b { "Bounce" } else { "" }))
86        }
87        "prism" => {
88            let content = e.bool("isContent") == Some(true);
89            let inv = e.bool("isInverted") == Some(true);
90            let kind = match (content, inv) {
91                (false, false) => "cube",
92                (false, true) => "box",
93                (true, false) => "rotate",
94                (true, true) => "orbit",
95            };
96            (k(kind), dir("l"))
97        }
98        "prstTrans" => {
99            let prst = e.attr("prst").unwrap_or("fallOver");
100            let inv = e.bool("invX") == Some(true);
101            if prst == "pageCurlSingle" {
102                return Some((k("pageCurlDouble"), if inv { k("singleR") } else { k("singleL") }));
103            }
104            let known = crate::model::anim::TRANSITIONS.iter().any(|t| t.0 == prst);
105            if !known {
106                return None;
107            }
108            let left = if INVX_LEFT.contains(&prst) { inv } else { !inv };
109            (k(prst), if left { k("l") } else { k("r") })
110        }
111        "morph" => (
112            k("morph"),
113            match e.attr("option") {
114                Some("byWord") => k("words"),
115                Some("byChar") => k("characters"),
116                _ => k("objects"),
117            },
118        ),
119        other => (k(SIMPLE.iter().find(|s| s.2 == other)?.0), String::new()),
120    })
121}

123/// Write the effect element for a transition. Returns the namespace prefix it needs
124/// (`p`, `p14`, `p15`, `p159`), or `None` when the kind is unknown.
125pub fn transition_to_xml(w: &mut W, kind: &str, option: &str) -> Option<&'static str> {
126    let lr = |d: &'static str| match option {
127        "l" | "r" => option.to_string(),
128        _ => d.to_string(),
129    };
130    let lrud = |d: &'static str| match option {
131        "l" | "r" | "u" | "d" => option.to_string(),
132        _ => d.to_string(),
133    };
134    let hv = |d: &'static str| match option {
135        "horz" | "vert" => option.to_string(),
136        _ => d.to_string(),
137    };
138    let eight = |d: &'static str| match option {
139        "l" | "r" | "u" | "d" | "lu" | "ru" | "ld" | "rd" => option.to_string(),
140        _ => d.to_string(),
141    };
142    match kind {
143        "fade" => w.empty("p:fade", A::new().t("thruBlk", option == "throughBlack")),
144        "cut" => w.empty("p:cut", A::new().t("thruBlk", option == "throughBlack")),
145        "push" => w.empty("p:push", A::new().a("dir", lrud("u"))),
146        "wipe" => match option {
147            "lu" | "ru" | "ld" | "rd" => w.empty("p:strips", A::new().a("dir", option)),
148            _ => w.empty("p:wipe", A::new().a("dir", lrud("r"))),
149        },
150        "split" => {
151            let (o, d) = match option {
152                "vertIn" => ("vert", "in"),
153                "vertOut" => ("vert", "out"),
154                "horzIn" => ("horz", "in"),
155                _ => ("horz", "out"),
156            };
157            w.empty("p:split", A::new().a("orient", o).a("dir", d));
158        }
159        "randomBar" => w.empty("p:randomBar", A::new().a("dir", hv("vert"))),
160        "shape" => match option {
161            "diamond" => w.empty0("p:diamond"),
162            "plus" => w.empty0("p:plus"),
163            "in" | "out" => w.empty("p:zoom", A::new().a("dir", option)),
164            _ => w.empty0("p:circle"),
165        },
166        "zoom" => match option {
167            "rotate" => w.empty0("p:newsflash"),
168            _ => {
169                w.empty("p14:warp", A::new().a("dir", if option == "out" { "out" } else { "in" }));
170                return Some("p14");
171            }
172        },
173        "uncover" => w.empty("p:pull", A::new().a("dir", eight("l"))),
174        "cover" => w.empty("p:cover", A::new().a("dir", eight("l"))),
175        "checker" => w.empty("p:checker", A::new().a("dir", hv("horz"))),
176        "blinds" => w.empty("p:blinds", A::new().a("dir", hv("horz"))),
177        "comb" => w.empty("p:comb", A::new().a("dir", hv("horz"))),
178        "clock" => match option {
179            "counterClockwise" => {
180                w.empty("p14:wheelReverse", A::new().a("spokes", 1));
181                return Some("p14");
182            }
183            "wedge" => w.empty0("p:wedge"),
184            _ => w.empty("p:wheel", A::new().a("spokes", 1)),
185        },
186        "dissolve" => w.empty0("p:dissolve"),
187        "random" => w.empty0("p:random"),
188        "honeycomb" => {
189            w.empty0("p14:honeycomb");
190            return Some("p14");
191        }
192        "flash" => {
193            w.empty0("p14:flash");
194            return Some("p14");
195        }
196        "vortex" => {
197            w.empty("p14:vortex", A::new().a("dir", lrud("l")));
198            return Some("p14");
199        }
200        "glitter" => {
201            let hex = option.starts_with("hex");
202            let d = option.trim_start_matches("hex").to_ascii_lowercase();
203            let d = if matches!(d.as_str(), "l" | "r" | "u" | "d") { d } else { "l".to_string() };
204            w.empty("p14:glitter", A::new().a("dir", d).o("pattern", hex.then_some("hexagon")));
205            return Some("p14");
206        }
207        "switch" | "flip" | "gallery" | "conveyor" | "ferris" => {
208            let name = match kind {
209                "switch" => "p14:switch",
210                "flip" => "p14:flip",
211                "gallery" => "p14:gallery",
212                "conveyor" => "p14:conveyor",
213                _ => "p14:ferris",
214            };
215            w.empty(name, A::new().a("dir", lr("l")));
216            return Some("p14");
217        }
218        "ripple" => {
219            let d = match option {
220                "lu" | "ru" | "ld" | "rd" => option,
221                _ => "center",
222            };
223            w.empty("p14:ripple", A::new().a("dir", d));
224            return Some("p14");
225        }
226        "doors" | "window" => {
227            let name = if kind == "doors" { "p14:doors" } else { "p14:window" };
228            w.empty(name, A::new().a("dir", hv("vert")));
229            return Some("p14");
230        }
231        "pan" => {
232            w.empty("p14:pan", A::new().a("dir", lrud("d")));
233            return Some("p14");
234        }
235        "shred" => {
236            let particles = option.starts_with("particles");
237            w.empty(
238                "p14:shred",
239                A::new().a("pattern", if particles { "rectangle" } else { "strip" }).a("dir", if option.ends_with("Out") { "out" } else { "in" }),
240            );
241            return Some("p14");
242        }
243        "reveal" => {
244            w.empty("p14:reveal", A::new().t("thruBlk", option.starts_with("black")).a("dir", if option.ends_with("Right") { "r" } else { "l" }));
245            return Some("p14");
246        }
247        "flythrough" => {
248            w.empty(
249                "p14:flythrough",
250                A::new().a("dir", if option.starts_with("out") { "out" } else { "in" }).t("hasBounce", option.ends_with("Bounce")),
251            );
252            return Some("p14");
253        }
254        "cube" | "box" | "rotate" | "orbit" => {
255            w.empty(
256                "p14:prism",
257                A::new().a("dir", lrud("l")).t("isContent", matches!(kind, "rotate" | "orbit")).t("isInverted", matches!(kind, "box" | "orbit")),
258            );
259            return Some("p14");
260        }
261        "pageCurlDouble" if option.starts_with("single") => {
262            w.empty("p15:prstTrans", A::new().a("prst", "pageCurlSingle").t("invX", option == "singleR"));
263            return Some("p15");
264        }
265        "fallOver" | "drape" | "curtains" | "wind" | "prestige" | "fracture" | "crush" | "peelOff" | "pageCurlDouble" | "airplane" | "origami" => {
266            let inv = if INVX_LEFT.contains(&kind) { option == "l" } else { option == "r" };
267            w.empty("p15:prstTrans", A::new().a("prst", kind).t("invX", inv));
268            return Some("p15");
269        }
270        "morph" => {
271            let o = match option {
272                "words" => "byWord",
273                "characters" => "byChar",
274                _ => "byObject",
275            };
276            w.empty("p159:morph", A::new().a("option", o));
277            return Some("p159");
278        }
279        _ => return None,
280    }
281    Some("p")
282}

335pub fn option_from_subtype(effect: &str, sub: u32) -> String {
336    SUBTYPES.iter().find(|(e, _)| *e == effect).and_then(|(_, m)| m.iter().find(|(_, s)| *s == sub)).map(|(o, _)| o.to_string()).unwrap_or_default()
337}

339pub fn subtype_from_option(effect: &str, option: &str) -> Option<u32> {
340    SUBTYPES.iter().find(|(e, _)| *e == effect).and_then(|(_, m)| m.iter().find(|(o, _)| *o == option)).map(|(_, s)| *s)
341}

343/// Our effect id for a preset (first match in [`ANIMATIONS`]).
344pub fn effect_for_preset(class: AnimClass, preset: u32) -> Option<&'static str> {
345    ANIMATIONS.iter().find(|a| a.2 == class && a.3 == preset && a.0 != "customPath").map(|a| a.0)
346}

348/// Effect used for presets we don't know.
349pub fn fallback_effect(class: AnimClass) -> &'static str {
350    match class {
351        AnimClass::Entrance => "fade",
352        AnimClass::Exit => "fadeOut",
353        AnimClass::Emphasis => "pulse",
354        AnimClass::Path => "customPath",
355        AnimClass::Media => "play",
356    }
357}

359/// The preset id to write for an animation.
360pub fn preset_for(a: &crate::model::Animation) -> u32 {
361    // Float Down is PowerPoint's preset 47 (Float Up is 42).
362    if matches!(a.effect.as_str(), "float" | "floatOut") && a.option == "d" {
363        return 47;
364    }
365    if let Some(p) = a.preset_id {
366        let consistent = match effect_for_preset(a.class, p) {
367            Some(e) => e == a.effect,
368            None => a.effect == fallback_effect(a.class),
369        };
370        if consistent {
371            return p;
372        }
373    }
374    match crate::model::anim::animation_info(&a.effect, a.class) {
375        Some(info) if info.2 == a.class => info.3,
376        _ => match a.class {
377            AnimClass::Entrance | AnimClass::Exit => 10,
378            AnimClass::Emphasis => 26,
379            AnimClass::Path => 0,
380            AnimClass::Media => 1,
381        },
382    }
383}

390    /// #147: these exits, with their directions, fell back to a plain fade (PowerPoint's ids).
391    #[test]
392    fn exit_presets_import_as_their_own_effects() {
393        for (preset, sub, effect, option) in [
394            (3, 10, "blindsOut", "horz"),
395            (4, 32, "boxOut", "out"),
396            (5, 5, "checkerboardOut", "down"),
397            (8, 16, "diamondOut", "in"),
398            (12, 4, "peekOut", "b"),
399            (18, 6, "stripsOut", "rd"),
400            (37, 0, "sinkDown", ""),
401            (49, 0, "spinnerOut", ""),
402            (55, 0, "contract", ""),
403        ] {
404            let e = effect_for_preset(AnimClass::Exit, preset).expect("known exit");
405            assert_eq!((e, option_from_subtype(e, sub).as_str()), (effect, option), "preset {preset}");
406        }
407    }

409    #[test]
410    fn every_transition_kind_round_trips() {
411        for (kind, _, _, _, opts) in crate::model::anim::TRANSITIONS {
412            if *kind == "none" {
413                continue;
414            }
415            let opts: Vec<&str> = if opts.is_empty() { vec![""] } else { opts.to_vec() };
416            for o in opts {
417                let mut w = W::frag();
418                let ns = transition_to_xml(&mut w, kind, o).unwrap_or_else(|| panic!("no xml for {kind}"));
419                assert!(!ns.is_empty());
420                let d = parse(w.s.as_bytes()).unwrap();
421                let (k, opt) = transition_from_xml(&d.root).unwrap_or_else(|| panic!("no read for {kind}"));
422                assert_eq!(k, *kind, "kind for {kind}/{o}");
423                if !o.is_empty() {
424                    assert_eq!(opt, o, "option for {kind}");
425                }
426            }
427        }
428    }
