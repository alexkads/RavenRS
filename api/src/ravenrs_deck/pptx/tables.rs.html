20/// GUIDs we write as built-in references (no definition needed).
21const BUILTIN_WRITE: [&str; 3] = ["none", "grid", "medium2-accent1"];

23fn fnv(s: &str, seed: u64) -> u64 {
24    let mut h = 0xcbf2_9ce4_8422_2325u64 ^ seed;
25    for b in s.bytes() {
26        h ^= b as u64;
27        h = h.wrapping_mul(0x0100_0000_01b3);
28    }
29    h
30}

32/// A stable GUID derived from text (version-4 layout).
33pub fn guid(text: &str) -> String {
34    let a = fnv(text, 0x5163_7261_6674);
35    let b = fnv(text, 0x7461_626c_6573);
36    format!(
37        "{{{:08X}-{:04X}-4{:03X}-8{:03X}-{:012X}}}",
38        (a >> 32) as u32,
39        (a >> 16) as u16,
40        (a & 0x0fff) as u16,
41        (b >> 48) as u16 & 0x0fff,
42        b & 0xffff_ffff_ffff
43    )
44}

46fn own_guid(style: &str) -> String {
47    guid(&format!("deckcraft-table-style:{style}"))
48}

50/// Is `g` the GUID DeckCraft writes for one of its own styles?
51pub fn is_own_guid(g: &str) -> bool {
52    let up = g.to_ascii_uppercase();
53    crate::model::table::table_styles().iter().any(|(id, _)| own_guid(id) == up)
54}

56/// Table style GUID → our style id (or the GUID itself when unknown).
57pub fn style_from_guid(g: &str) -> String {
58    let up = g.to_ascii_uppercase();
59    if let Some((_, s)) = KNOWN.iter().find(|(k, _)| *k == up) {
60        return s.to_string();
61    }
62    for (id, _) in crate::model::table::table_styles() {
63        if own_guid(&id) == up {
64            return id;
65        }
66    }
67    if g.is_empty() { "none".into() } else { g.to_string() }
68}

70/// GUID to write for our style id, and whether a definition must be written.
71pub fn guid_for_style(style: &str) -> (String, bool) {
72    if style.starts_with('{') {
73        return (style.to_string(), false);
74    }
75    if BUILTIN_WRITE.contains(&style)
76        && let Some((g, _)) = KNOWN.iter().find(|(_, s)| *s == style)
77    {
78        return (g.to_string(), false);
79    }
80    let known = crate::model::table::table_styles().iter().any(|(id, _)| id == style);
81    if known { (own_guid(style), true) } else { ("{5C22544A-7EE6-4342-B048-85BDC9FD1C3A}".into(), false) }
82}

84fn clr(w: &mut W, slot: &str, mods: &[(&str, i32)]) {
85    if mods.is_empty() {
86        w.val("a:schemeClr", slot);
87        return;
88    }
89    w.open("a:schemeClr", A::new().a("val", slot));
90    for (n, v) in mods {
91        w.val(n, v);
92    }
93    w.close("a:schemeClr");
94}

96fn solid(w: &mut W, slot: &str, mods: &[(&str, i32)]) {
97    w.open0("a:fill");
98    w.open0("a:solidFill");
99    clr(w, slot, mods);
100    w.close("a:solidFill");
101    w.close("a:fill");
102}

104fn borders(w: &mut W, b: Option<(&str, &[(&str, i32)], i32)>) {
105    w.open0("a:tcBdr");
106    if let Some((slot, mods, width)) = b {
107        for side in ["a:left", "a:right", "a:top", "a:bottom", "a:insideH", "a:insideV"] {
108            w.open0(side);
109            w.open("a:ln", A::new().a("w", width).a("cmpd", "sng"));
110            w.open0("a:solidFill");
111            clr(w, slot, mods);
112            w.close("a:solidFill");
113            w.close("a:ln");
114            w.close(side);
115        }
116    }
117    w.close("a:tcBdr");
118}

120fn txt(w: &mut W, bold: bool, color: Option<&str>) {
121    w.open("a:tcTxStyle", A::new().o("b", bold.then_some("on")));
122    w.open("a:fontRef", A::new().a("idx", "minor"));
123    w.val("a:prstClr", "black");
124    w.close("a:fontRef");
125    match color {
126        Some(c) => w.val("a:schemeClr", c),
127        None => w.val("a:schemeClr", "dk1"),
128    }
129    w.close("a:tcTxStyle");
130}

132/// Write one `a:tblStyle` for our style id.
133fn style_def(w: &mut W, id: &str, name: &str) {
134    let kind = id.split('-').next().unwrap_or("medium2");
135    let acc = match id.rsplit('-').next().unwrap_or("accent1") {
136        "tx1" => "dk1",
137        a if a.starts_with("accent") => a,
138        _ => "accent1",
139    };
140    let acc: &str = acc;
141    w.open("a:tblStyle", A::new().a("styleId", own_guid(id)).a("styleName", name));
142    let part =
143        |w: &mut W, tag: &str, bold: bool, text: Option<&str>, fill: Option<(&str, &[(&str, i32)])>, bdr: Option<(&str, &[(&str, i32)], i32)>| {
144            w.open0(tag);
145            if bold || text.is_some() {
146                txt(w, bold, text);
147            }
148            w.open0("a:tcStyle");
149            borders(w, bdr);
150            if let Some((s, m)) = fill {
151                solid(w, s, m);
152            }
153            w.close("a:tcStyle");
154            w.close(tag);
155        };
156    match kind {
157        "grid" => {
158            part(w, "a:wholeTbl", false, Some("dk1"), None, Some(("dk1", &[], 9525)));
159        }
160        "light1" => {
161            part(w, "a:wholeTbl", false, Some("dk1"), None, None);
162            part(w, "a:band1H", false, None, Some((acc, &[("a:tint", 20000)])), None);
163            part(w, "a:band1V", false, None, Some((acc, &[("a:tint", 20000)])), None);
164            part(w, "a:lastCol", true, None, None, None);
165            part(w, "a:firstCol", true, None, None, None);
166            part(w, "a:lastRow", true, None, None, None);
167            part(w, "a:firstRow", true, None, None, Some((acc, &[], 12700)));
168        }
169        "light2" => {
170            part(w, "a:wholeTbl", false, Some("dk1"), None, Some((acc, &[], 12700)));
171            // Schema order: … lastCol, firstCol, lastRow, …, firstRow.
172            part(w, "a:firstCol", true, None, None, None);
173            part(w, "a:lastRow", true, None, None, None);
174            part(w, "a:firstRow", true, Some("lt1"), Some((acc, &[])), Some((acc, &[], 12700)));
175        }
176        "dark1" => {
177            part(w, "a:wholeTbl", false, Some("lt1"), Some((acc, &[("a:shade", 75000)])), Some(("lt1", &[], 12700)));
178            part(w, "a:band1H", false, None, Some((acc, &[("a:shade", 60000)])), None);
179            part(w, "a:band1V", false, None, Some((acc, &[("a:shade", 60000)])), None);
180            part(w, "a:firstCol", true, None, None, None);
181            part(w, "a:lastRow", true, None, None, None);
182            part(w, "a:firstRow", true, Some("lt1"), Some(("dk1", &[])), None);
183        }
184        "medium4" => {
185            part(w, "a:wholeTbl", false, Some("dk1"), Some((acc, &[("a:tint", 20000)])), Some((acc, &[], 9525)));
186            part(w, "a:band1H", false, None, Some((acc, &[("a:tint", 40000)])), None);
187            part(w, "a:band1V", false, None, Some((acc, &[("a:tint", 40000)])), None);
188            part(w, "a:lastCol", true, None, None, None);
189            part(w, "a:firstCol", true, None, None, None);
190            part(w, "a:lastRow", true, None, None, None);
191            part(w, "a:firstRow", true, None, Some((acc, &[("a:tint", 40000)])), None);
192        }
193        _ => {
194            // medium2
195            part(w, "a:wholeTbl", false, Some("dk1"), Some((acc, &[("a:tint", 20000)])), Some(("lt1", &[], 12700)));
196            part(w, "a:band1H", false, None, Some((acc, &[("a:tint", 40000)])), None);
197            part(w, "a:band1V", false, None, Some((acc, &[("a:tint", 40000)])), None);
198            part(w, "a:lastCol", true, Some("lt1"), Some((acc, &[])), None);
199            part(w, "a:firstCol", true, Some("lt1"), Some((acc, &[])), None);
200            part(w, "a:lastRow", true, Some("lt1"), Some((acc, &[])), None);
201            part(w, "a:firstRow", true, Some("lt1"), Some((acc, &[])), None);
202        }
203    }
204    w.close("a:tblStyle");
205}

207/// `ppt/tableStyles.xml` with definitions for the styles in `used`: ours, generated, and those the
208/// opened file defined (`defs`), written back as they were read.
209pub fn table_styles_xml(used: &[String], defs: &[crate::model::table::TableStyleDef]) -> Vec<u8> {
210    let mut w = W::new();
211    w.open("a:tblStyleLst", A::new().a("xmlns:a", NS_A).a("def", "{5C22544A-7EE6-4342-B048-85BDC9FD1C3A}"));
212    let names = crate::model::table::table_styles();
213    // Indexes instead of scans: written ids (as given and ASCII lower-case), and the file's
214    // definitions by lower-case id (the first wins).
215    let mut done: std::collections::HashSet<&str> = std::collections::HashSet::new();
216    let mut done_lower: std::collections::HashSet<String> = std::collections::HashSet::new();
217    let mut def_of: std::collections::HashMap<String, &crate::model::table::TableStyleDef> = std::collections::HashMap::new();
218    for d in defs {
219        def_of.entry(d.id.to_ascii_lowercase()).or_insert(d);
220    }
221    for s in used {
222        if !done.insert(s.as_str()) {
223            continue;
224        }
225        done_lower.insert(s.to_ascii_lowercase());
226        if let Some(d) = def_of.get(&s.to_ascii_lowercase())
227            && crate::pptx::xml::parse(d.raw.as_bytes()).is_ok_and(|x| x.root.is("tblStyle"))
228        {
229            w.raw(&d.raw);
230            continue;
231        }
232        let (_, define) = guid_for_style(s);
233        if !define {
234            continue;
235        }
236        let name = names.iter().find(|(id, _)| id == s).map(|(_, n)| format!("DeckCraft {n}")).unwrap_or_else(|| s.clone());
237        style_def(&mut w, s, &name);
238    }
239    // Styles the file defined but no table uses still belong to the deck.
240    for d in defs {
241        if done_lower.contains(&d.id.to_ascii_lowercase()) {
242            continue;
243        }
244        if crate::pptx::xml::parse(d.raw.as_bytes()).is_ok_and(|x| x.root.is("tblStyle")) {
245            done_lower.insert(d.id.to_ascii_lowercase());
246            w.raw(&d.raw);
247        }
248    }
249    w.close("a:tblStyleLst");
250    w.finish()
251}

257    #[test]
258    fn style_parts_follow_schema_order() {
259        const ORDER: [&str; 14] = [
260            "tblBg", "wholeTbl", "band1H", "band2H", "band1V", "band2V", "lastCol", "firstCol", "lastRow", "seCell", "swCell", "firstRow", "neCell",
261            "nwCell",
262        ];
263        let used: Vec<String> = crate::model::table::table_styles().into_iter().map(|(id, _)| id).collect();
264        let d = crate::pptx::xml::parse(&table_styles_xml(&used, &[])).unwrap();
265        let mut n = 0;
266        for st in d.root.children_named("tblStyle") {
267            n += 1;
268            let pos: Vec<usize> = st.elements().map(|e| ORDER.iter().position(|o| *o == e.local()).unwrap()).collect();
269            assert!(pos.windows(2).all(|w| w[0] < w[1]), "{:?}: {pos:?}", st.attr("styleName"));
270        }
271        assert!(n > 30);
272    }
273
274    #[test]
275    fn guids_round_trip() {
276        for (id, _) in crate::model::table::table_styles() {
277            let (g, _) = guid_for_style(&id);
278            assert_eq!(style_from_guid(&g), id, "{id} via {g}");
279            assert_eq!(g.len(), 38);
280        }
281        assert_eq!(style_from_guid("{5C22544A-7EE6-4342-B048-85BDC9FD1C3A}"), "medium2-accent1");
282        assert_eq!(style_from_guid("{ABCDEF00-0000-0000-0000-000000000000}"), "{ABCDEF00-0000-0000-0000-000000000000}");
283    }
