1//! A minimal PNG encoder for the placeholder image written when a picture's media is missing.

3fn crc32(data: &[u8]) -> u32 {
4    let mut c = 0xFFFF_FFFFu32;
5    for b in data {
6        c ^= *b as u32;
7        for _ in 0..8 {
8            c = if c & 1 != 0 { 0xEDB8_8320 ^ (c >> 1) } else { c >> 1 };
9        }
10    }
11    !c
12}
13
14fn adler32(data: &[u8]) -> u32 {
15    let (mut a, mut b) = (1u32, 0u32);
16    for x in data {
17        a = (a + *x as u32) % 65521;
18        b = (b + a) % 65521;
19    }
20    (b << 16) | a
21}

23fn chunk(out: &mut Vec<u8>, kind: &[u8; 4], data: &[u8]) {
24    out.extend_from_slice(&(data.len() as u32).to_be_bytes());
25    let mut c = Vec::with_capacity(data.len() + 4);
26    c.extend_from_slice(kind);
27    c.extend_from_slice(data);
28    out.extend_from_slice(&c);
29    out.extend_from_slice(&crc32(&c).to_be_bytes());
30}

32/// A 1×1 fully transparent RGBA PNG.
33pub fn transparent_1x1() -> Vec<u8> {
34    let mut out = b"\x89PNG\r\n\x1a\n".to_vec();
35    let mut ihdr = vec![];
36    ihdr.extend_from_slice(&1u32.to_be_bytes());
37    ihdr.extend_from_slice(&1u32.to_be_bytes());
38    ihdr.extend_from_slice(&[8, 6, 0, 0, 0]);
39    chunk(&mut out, b"IHDR", &ihdr);
40    // One scanline: filter byte + RGBA, stored (uncompressed) deflate block.
41    let raw = [0u8, 0, 0, 0, 0];
42    let mut z = vec![0x78, 0x01, 0x01];
43    z.extend_from_slice(&(raw.len() as u16).to_le_bytes());
44    z.extend_from_slice(&(!(raw.len() as u16)).to_le_bytes());
45    z.extend_from_slice(&raw);
46    z.extend_from_slice(&adler32(&raw).to_be_bytes());
47    chunk(&mut out, b"IDAT", &z);
48    chunk(&mut out, b"IEND", &[]);
49    out
50}
