16/// File shape ids at or above this are reassigned.
17pub const MAX_KEPT_ID: u32 = 1 << 30;
18const MAX_TREE_DEPTH: usize = 32;
19const MAX_SHAPES: usize = 20_000;
20
21/// Shape id bookkeeping for one part.
22#[derive(Default)]
23pub struct IdCtx {
24    pub used: HashSet<u32>,
25    /// File id → model id.
26    pub map: HashMap<u32, ShapeId>,
27    /// Always allocate new ids (shapes pulled in from other parts).
28    pub fresh: bool,
29    pub count: usize,
30}
31
32impl IdCtx {
33    fn assign(&mut self, imp: &mut Imp, file_id: Option<u32>) -> ShapeId {
34        if !self.fresh
35            && let Some(i) = file_id
36            && (2..MAX_KEPT_ID).contains(&i)
37            && !self.used.contains(&i)
38        {
39            self.used.insert(i);
40            self.map.insert(i, ShapeId(i));
41            return ShapeId(i);
42        }
43        let n = imp.alloc();
44        self.used.insert(n);
45        if let Some(i) = file_id {
46            self.map.entry(i).or_insert(ShapeId(n));
47        }
48        ShapeId(n)
49    }
50    pub fn lookup(&self, file_id: u32) -> Option<ShapeId> {
51        self.map.get(&file_id).copied()
52    }
53}

55/// Read the children of an `spTree` / `grpSp`.
56pub fn sp_tree(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, tree: &El, depth: usize) -> Vec<Shape> {
57    let mut out = vec![];
58    if depth > MAX_TREE_DEPTH {
59        log::warn!("pptx: group nesting too deep in {}", part.name);
60        return out;
61    }
62    for e in tree.elements() {
63        if ctx.count >= MAX_SHAPES {
64            log::warn!("pptx: too many shapes in {}", part.name);
65            break;
66        }
67        let s = match e.local() {
68            "sp" => Some(sp(imp, part, ctx, e)),
69            "pic" => Some(pic(imp, part, ctx, e)),
70            "grpSp" => Some(grp(imp, part, ctx, e, depth)),
71            "cxnSp" => Some(cxn(imp, part, ctx, e)),
72            "graphicFrame" => Some(frame(imp, part, ctx, e, depth)),
73            "contentPart" => content_part(imp, part, ctx, e).or_else(|| {
74                log::warn!("pptx: content part in {} not imported", part.name);
75                None
76            }),
77            "AlternateContent" => read_model_alternate(imp, part, ctx, e),
78            _ => None,
79        };
80        if let Some(s) = s {
81            ctx.count += 1;
82            out.push(s);
83        }
84    }
85    if depth == 0 {
86        remap_connectors(&mut out, ctx, 0);
87    }
88    out
89}

91/// ActiveX controls (`p:controls`): each control's picture (`p:pic`), the image PowerPoint shows
92/// for it, as a picture above the slide's shapes. Controls without one are left out.
93pub fn controls(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, controls: &El) -> Vec<Shape> {
94    let mut out = vec![];
95    for c in controls.children_named("control") {
96        if ctx.count >= MAX_SHAPES {
97            break;
98        }
99        if let Some(p) = c.child("pic") {
100            ctx.count += 1;
101            out.push(pic(imp, part, ctx, p));
102        }
103    }
104    out
105}

107fn remap_connectors(shapes: &mut [Shape], ctx: &IdCtx, depth: usize) {
108    if depth > MAX_TREE_DEPTH {
109        return;
110    }
111    for s in shapes.iter_mut() {
112        match &mut s.kind {
113            ShapeKind::Connector { start, end } => {
114                for e in [start, end] {
115                    *e = e.and_then(|(id, site)| ctx.lookup(id.0).map(|m| (m, site)));
116                }
117            }
118            ShapeKind::Group { children, .. } => remap_connectors(children, ctx, depth + 1),
119            _ => {}
120        }
121    }
122}

124/// PowerPoint's Lock (right-click > Lock, the Selection pane padlock) writes a full set of locking
125/// attributes; it reads an object as locked only when the whole set for its kind is there, so a
126/// lone `noMove` or `noResize` (other tools' partial locks) isn't Lock.
127fn is_locked(l: &El) -> bool {
128    let set: &[&str] = match l.local() {
129        "grpSpLocks" => &["noGrp", "noUngrp", "noRot", "noMove", "noResize"],
130        "graphicFrameLocks" => &["noGrp", "noDrilldown", "noMove", "noResize"],
131        _ => &["noGrp", "noRot", "noMove", "noResize", "noEditPoints", "noAdjustHandles", "noChangeArrowheads", "noChangeShapeType"],
132    };
133    set.iter().all(|a| l.bool(a) == Some(true))
134}

136/// Common non-visual properties (`p:nvSpPr`, `p:nvPicPr`…).
137fn base(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, nv: Option<&El>) -> Shape {
138    let c = nv.and_then(|n| n.child("cNvPr"));
139    let mut s = Shape { id: ctx.assign(imp, c.and_then(|c| c.u32("id"))), ..Default::default() };
140    if let Some(c) = c {
141        s.name = c.attr("name").unwrap_or("").to_string();
142        s.descr = c.attr("descr").unwrap_or("").to_string();
143        s.title = c.attr("title").unwrap_or("").to_string();
144        s.hidden = c.bool("hidden").unwrap_or(false);
145        s.click = c.child("hlinkClick").and_then(|h| dml::hyperlink(imp, part, h));
146        s.hover = c.child("hlinkHover").and_then(|h| dml::hyperlink(imp, part, h));
147        s.decorative = c.find("decorative").and_then(|d| d.bool("val")).unwrap_or(false);
148    }
149    if let Some(nv) = nv {
150        for e in nv.elements() {
151            match e.local() {
152                "cNvSpPr" => {
153                    s.text_box = e.bool("txBox").unwrap_or(false);
154                    s.locked = e.child("spLocks").is_some_and(is_locked);
155                }
156                "cNvPicPr" | "cNvGrpSpPr" | "cNvCxnSpPr" | "cNvGraphicFramePr" => {
157                    s.locked = e.elements().find(|l| l.local().ends_with("Locks")).is_some_and(is_locked);
158                }
159                "nvPr" => {
160                    if let Some(ph) = e.child("ph") {
161                        s.ph = Some(Placeholder {
162                            kind: ph.attr("type").map(PhType::from_xml).unwrap_or(PhType::Obj),
163                            idx: ph.u32("idx").unwrap_or(0),
164                            vertical: ph.attr("orient") == Some("vert"),
165                            size: ph.attr("sz").map(String::from),
166                            has_custom_prompt: ph.bool("hasCustomPrompt").unwrap_or(false),
167                        });
168                    }
169                    s.tags = super::tags(imp, part, e);
170                }
171                _ => {}
172            }
173        }
174    }
175    s
176}

178fn preset_geom(pg: &El) -> Geom {
179    let name = pg.attr("prst").unwrap_or("rect").to_string();
180    let defaults: Vec<f64> = preset::info(&name).map(|i| i.defaults.to_vec()).unwrap_or_default();
181    let mut adj = defaults;
182    let mut any = false;
183    if let Some(av) = pg.child("avLst") {
184        for gd in av.children_named("gd").take(16) {
185            let n = gd.attr("name").unwrap_or("");
186            let idx = if n == "adj" {
187                0
188            } else if let Some(k) = n.strip_prefix("adj").and_then(|k| k.parse::<usize>().ok()).filter(|k| (1..=16).contains(k)) {
189                k - 1
190            } else {
191                continue;
192            };
193            let Some(v) =
194                gd.attr("fmla").and_then(|f| f.trim().strip_prefix("val")).and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite())
195            else {
196                continue;
197            };
198            if adj.len() <= idx {
199                adj.resize(idx + 1, 0.0);
200            }
201            if let Some(slot) = adj.get_mut(idx) {
202                *slot = v;
203                any = true;
204            }
205        }
206    }
207    if !any {
208        adj.clear();
209    }
210    Geom::Preset { name, adj }
211}

213/// A placeholder whose `spPr` names no geometry takes its layout's (or master's) placeholder's:
214/// marked here with an empty custom geometry, filled in once the layouts are read
215/// (`inherit_placeholder_geometry`).
216fn mark_inherited_geometry(s: &mut Shape, pr: Option<&El>) {
217    if s.ph.is_some() && pr.is_none_or(|pr| pr.child("prstGeom").is_none() && pr.child("custGeom").is_none()) {
218        s.geom = Geom::Custom { paths: vec![] };
219    }
220}

222/// `spPr` → transform, geometry, fill, line, effects.
223fn sp_pr(imp: &mut Imp, part: &Part, s: &mut Shape, pr: &El) {
224    if let Some(x) = pr.child("xfrm") {
225        s.xfrm = Some(dml::xfrm(x).0);
226    }
227    if let Some(pg) = pr.child("prstGeom") {
228        s.geom = preset_geom(pg);
229    } else if let Some(cg) = pr.child("custGeom") {
230        let (w, h) = s.xfrm.map(|x| (crate::geom::pt_to_emu(x.w) as f64, crate::geom::pt_to_emu(x.h) as f64)).unwrap_or((21600.0, 21600.0));
231        let paths = crate::pptx::custgeom::read_cust_geom(cg, w, h);
232        if !paths.is_empty() {
233            s.geom = Geom::Custom { paths };
234        }
235    }
236    s.fill = dml::fill_of(imp, part, pr);
237    if let Some(ln) = pr.child("ln") {
238        s.line = Some(dml::line(imp, part, ln));
239    }
240    if let Some(fx) = pr.child("effectLst") {
241        s.effects = Some(dml::effects(fx));
242    }
243    let raw: String = pr.elements().filter(|c| c.is("scene3d") || c.is("sp3d")).map(|c| part.keep(c)).collect();
244    if !raw.is_empty() {
245        let fx = s.effects.get_or_insert_with(Effects::default);
246        fx.bevel = pr.path(&["sp3d", "bevelT"]).map(|b| b.attr("prst").unwrap_or("circle").to_string());
247        fx.raw3d = Some(raw);
248    }
249}

251fn style(st: &El) -> ShapeStyle {
252    let r = |n: &str| {
253        st.child(n)
254            .map(|e| (e.u32("idx").unwrap_or(0).min(1003), dml::color(e).unwrap_or(ColorRef::scheme(SchemeSlot::Accent1))))
255            .unwrap_or((0, ColorRef::scheme(SchemeSlot::Accent1)))
256    };
257    let font = st.child("fontRef");
258    ShapeStyle {
259        line_ref: r("lnRef"),
260        fill_ref: r("fillRef"),
261        effect_ref: r("effectRef"),
262        font_ref: (font.and_then(|f| f.attr("idx")).unwrap_or("minor").to_string(), font.and_then(dml::color)),
263    }
264}

266fn sp(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El) -> Shape {
267    let mut s = base(imp, part, ctx, e.child("nvSpPr"));
268    if let Some(pr) = e.child("spPr") {
269        sp_pr(imp, part, &mut s, pr);
270    }
271    mark_inherited_geometry(&mut s, e.child("spPr"));
272    if e.bool("useBgFill") == Some(true) {
273        s.fill = Some(Fill::Background);
274    }
275    s.style = e.child("style").map(style);
276    s.text = e.child("txBody").map(|t| dml::text_body(imp, part, t));
277    s
278}

281    v.and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite()).map(|v| v.clamp(0.0, u32::MAX as f64) as u32).unwrap_or(0)

284fn find_content_part(el: &El) -> Option<&El> {
285    fn rec(el: &El, depth: usize) -> Option<&El> {
286        if depth > 16 {
287            return None;
288        }
289        if el.is("contentPart") {
290            return Some(el);
291        }
292        el.elements().find_map(|c| rec(c, depth + 1))
293    }
294    rec(el, 0)
295}

297/// `p:contentPart` whose relationship is a 3-D model. Other content parts (ink) are left out.
298fn content_part(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El) -> Option<Shape> {
299    let id = e.attr("r:id").filter(|s| !s.is_empty())?;
300    let rel = part.rels.get(id)?;
301    if !super::is_model3d(rel) {
302        return None;
303    }
304    let target = rel.target.clone();
305    let media = imp.media_for(&target).or_else(|| {
306        log::warn!("pptx: 3-D model part missing in {}", part.name);
307        None
308    })?;
309    let mut s = base(imp, part, ctx, e.child("nvContentPartPr"));
310    if let Some(x) = e.child("xfrm") {
311        s.xfrm = Some(dml::xfrm(x).0);
312    }
313    s.kind = ShapeKind::Model3d { media, poster: None };
314    Some(s)
315}

317/// A 3-D model wrapped in `mc:AlternateContent`: the model choice, plus the fallback picture.
318/// If the model part is missing, the fallback picture is the shape.
319fn read_model_alternate(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El) -> Option<Shape> {
320    if let Some(content) = e.children_named("Choice").find_map(find_content_part)
321        && let Some(mut s) = content_part(imp, part, ctx, content)
322    {
323        let poster = e.child("Fallback").and_then(|fb| {
324            let mut pics = vec![];
325            fb.find_all("blip", &mut pics);
326            pics.into_iter().find_map(|b| dml::media_ref(imp, part, b))
327        });
328        if let ShapeKind::Model3d { media, .. } = s.kind {
329            s.kind = ShapeKind::Model3d { media, poster };
330        }
331        return Some(s);
332    }
333    e.child("Fallback").and_then(|fb| fb.elements().find(|c| c.is("pic")).map(|p| pic(imp, part, ctx, p)))
334}

336fn pic(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El) -> Shape {
337    let mut s = base(imp, part, ctx, e.child("nvPicPr"));
338    if let Some(pr) = e.child("spPr") {
339        sp_pr(imp, part, &mut s, pr);
340    }
341    // A picture in a picture placeholder is cut into the layout's shape (a circle, say).
342    mark_inherited_geometry(&mut s, e.child("spPr"));
343    s.style = e.child("style").map(style);
344    let pf = e.child("blipFill").and_then(|b| dml::blip_fill(imp, part, b));
345    let nvpr = e.path(&["nvPicPr", "nvPr"]);
346    let vf = nvpr.and_then(|n| n.child("videoFile"));
347    let af = nvpr.and_then(|n| n.child("audioFile"));
348    let p14 = nvpr.and_then(|n| n.child("extLst")).and_then(|x| x.find("media"));
349    if vf.is_some() || af.is_some() || p14.is_some() {
350        let media = p14.and_then(|m| dml::media_ref(imp, part, m)).or_else(|| vf.or(af).and_then(|f| dml::media_ref(imp, part, f)));
351        if let Some(media) = media {
352            let mut clip = MediaClip {
353                media,
354                video: vf.is_some() || af.is_none(),
355                poster: pf.as_ref().map(|p| p.media),
356                crop: pf.as_ref().map(|p| p.crop).unwrap_or_default(),
357                volume: 1.0,
358                ..Default::default()
359            };
360            if let Some(m) = p14 {
361                if let Some(t) = m.child("trim") {
362                    clip.trim_start_ms = ms(t.attr("st"));
363                    clip.trim_end_ms = ms(t.attr("end"));
364                }
365                if let Some(f) = m.child("fade") {
366                    clip.fade_in_ms = ms(f.attr("in"));
367                    clip.fade_out_ms = ms(f.attr("out"));
368                }
369                if let Some(l) = m.child("bmkLst") {
370                    clip.bookmarks =
371                        l.children_named("bmk").take(1000).map(|b| (b.attr("name").unwrap_or("").to_string(), ms(b.attr("time")))).collect();
372                }
373            }
374            s.kind = ShapeKind::Media(clip);
375            return s;
376        }
377    }
378    match pf {
379        Some(fill) => s.kind = ShapeKind::Picture { fill },
380        None => log::warn!("pptx: picture {} in {} has no readable image", s.name, part.name),
381    }
382    s
383}

385fn grp(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El, depth: usize) -> Shape {
386    let mut s = base(imp, part, ctx, e.child("nvGrpSpPr"));
387    let mut child = None;
388    if let Some(pr) = e.child("grpSpPr") {
389        if let Some(x) = pr.child("xfrm") {
390            let (xf, ch) = dml::xfrm(x);
391            s.xfrm = Some(xf);
392            child = ch;
393        }
394        s.fill = dml::fill_of(imp, part, pr);
395        if let Some(fx) = pr.child("effectLst") {
396            s.effects = Some(dml::effects(fx));
397        }
398        if let Some(sc) = pr.child("scene3d") {
399            let raw = part.keep_inner(sc);
400            if !raw.is_empty() && raw.len() <= 64 * 1024 {
401                s.effects.get_or_insert_with(Effects::default).raw3d = Some(raw);
402            }
403        }
404    }
405    let children = sp_tree(imp, part, ctx, e, depth + 1);
406    let child = child.or_else(|| s.xfrm.map(|x| Xfrm::new(x.x, x.y, x.w, x.h))).unwrap_or_default();
407    s.kind = ShapeKind::Group { children, child };
408    s
409}

411fn cxn(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El) -> Shape {
412    let nv = e.child("nvCxnSpPr");
413    let mut s = base(imp, part, ctx, nv);
414    if let Some(pr) = e.child("spPr") {

420    s.kind = ShapeKind::Connector { start: end("stCxn"), end: end("endCxn") };
421    s
422}

424fn frame(imp: &mut Imp, part: &Part, ctx: &mut IdCtx, e: &El, depth: usize) -> Shape {
425    let mut s = base(imp, part, ctx, e.child("nvGraphicFramePr"));
426    if let Some(x) = e.child("xfrm") {
427        s.xfrm = Some(dml::xfrm(x).0);
428    }
429    let gd = e.path(&["graphic", "graphicData"]);
430    let uri = gd.and_then(|g| g.attr("uri")).unwrap_or("");
431    if let Some(gd) = gd {
432        if let Some(tbl) = gd.child("tbl") {
433            s.kind = ShapeKind::Table(table(imp, part, tbl));
434            return s;
435        }
436        if let Some(c) = gd.child("chart")
437            && let Some(chart) = super::chart::frame_chart(imp, part, c.attr("r:id").unwrap_or(""), uri)
438        {
439            s.kind = ShapeKind::Chart(Box::new(chart));
440            return s;
441        }
442        if uri.ends_with("/diagram")
443            && let Some(g) = smartart(imp, part, gd, &s, depth)
444        {
445            return g;
446        }
447    }
448    // Something we don't model: keep its XML and show a preview picture when it has one.
449    let preview = {
450        let mut pics = vec![];
451        e.find_all("blip", &mut pics);
452        pics.into_iter().find_map(|b| dml::media_ref(imp, part, b))
453    }
454    .or_else(|| gd.and_then(|g| g.find("oleObj")).and_then(|o| vml_preview(imp, part, o.attr("spid")?)));
455    // An embedded object's fallback picture (`p:oleObj/p:pic`) can carry a fill and an outline,
456    // which PowerPoint draws behind and around the preview (old equation objects on coloured
457    // boxes). The frame's own position stays.
458    if let Some(pr) = gd.and_then(|g| g.find("oleObj")).and_then(|o| o.find("pic")).and_then(|p| p.child("spPr")) {
459        if s.fill.is_none() {
460            s.fill = dml::fill_of(imp, part, pr);
461        }
462        if s.line.is_none() {
463            s.line = pr.child("ln").map(|ln| dml::line(imp, part, ln));
464        }
465    }
466    let label = if let Some(o) = gd.and_then(|g| g.find("oleObj")) {
467        o.attr("progId").or(o.attr("name")).unwrap_or("Object").to_string()
468    } else if uri.ends_with("/diagram") {
469        "SmartArt".to_string()
470    } else if uri.contains("chart") {
471        "Chart".to_string()
472    } else {
473        uri.rsplit('/').next().unwrap_or("Object").to_string()
474    };
475    // Keep the parts it points to so a save writes it back whole (#203). A part with relationships
476    // of its own, or an external link, can't be carried: the save then falls back to the preview.
477    let parts = super::rel_refs(e)
478        .into_iter()
479        .map(|id| {
480            let rel = part.rels.get(&id).filter(|r| !r.external)?;
481            let pkg = imp.pkg;
482            if *imp.has_rels.entry(crate::pptx::opc::norm(&rel.target)).or_insert_with(|| !pkg.rels(&rel.target).items.is_empty()) {
483                return None;
484            }
485            let (kind, target) = (rel.ty.clone(), rel.target.clone());
486            let media = if rel.kind == "diagramData" { diagram_data(imp, &target) } else { imp.media_for(&target) };
487            Some((id, kind, media?))
488        })
489        .collect::<Option<Vec<_>>>()
490        .unwrap_or_default();
491    s.kind = ShapeKind::Opaque { xml: part.keep(e), preview, label, parts };
492    s
493}

495/// The preview picture of an old-style embedded object (`p:oleObj spid` with no `p:pic`): the
496/// `v:imagedata` of the legacy VML drawing shape with that id, as PowerPoint 2007 wrote them.
497fn vml_preview(imp: &mut Imp, part: &Part, spid: &str) -> Option<crate::model::MediaId> {
498    for rel in part.rels.items.iter().filter(|r| r.kind == "vmlDrawing" && !r.external) {
499        let key = crate::pptx::opc::norm(&rel.target);
500        if !imp.vml_previews.contains_key(&key) {
501            let index = vml_index(imp.pkg, &rel.target);
502            imp.vml_previews.insert(key.clone(), index);
503        }
504        let Some(target) = imp.vml_previews.get(&key).and_then(|m| m.get(spid)).cloned() else { continue };
505        return imp.media_for(&target);
506    }
507    None
508}

510/// Every shape id in a VML drawing with a picture, mapped to that picture's part: parsed once per
511/// drawing however many objects look it up.
512fn vml_index(pkg: &crate::pptx::opc::Package, vml: &str) -> std::collections::HashMap<String, String> {
513    let mut out = std::collections::HashMap::new();
514    let Some(doc) = pkg.xml(vml) else { return out };
515    let rels = pkg.rels(vml);
516    let mut shapes = vec![];
517    doc.root.find_all("shape", &mut shapes);
518    for s in shapes {
519        let Some(id) = s.attr("id") else { continue };
520        if out.contains_key(id) {
521            continue; // the first shape with an id wins, as the old lookup did
522        }
523        let Some(img) = s.find("imagedata") else { continue };
524        let Some(rid) = img.attr("o:relid").or(img.attr("r:id")) else { continue };
525        if let Some(r) = rels.get(rid).filter(|r| !r.external) {
526            out.insert(id.to_string(), r.target.clone());
527        }
528    }
529    out
530}

532/// A diagram's data part, without the link to its cached drawing: that link is a slide
533/// relationship id, which a save renumbers. Only diagrams whose drawing is missing or empty get
534/// here, and those are drawn from their data.
535fn diagram_data(imp: &mut Imp, target: &str) -> Option<crate::model::MediaId> {
536    let norm_target = crate::pptx::opc::norm(target);
537    if let Some(&id) = imp.media_by_part.get(&norm_target) {
538        return Some(id);
539    }
540    fn strip(e: &mut El, depth: usize) {
541        if depth > crate::pptx::xml::MAX_DEPTH {
542            return;
543        }
544        e.children.retain(|n| !matches!(n, crate::pptx::xml::Node::El(c) if c.is("ext") && c.find("dataModelExt").is_some()));
545        for c in e.elements_mut() {
546            strip(c, depth + 1);
547        }
548    }
549    let mut doc = imp.pkg.xml(target)?;
550    strip(&mut doc.root, 0);
551    let xml = format!("<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n{}", doc.root.to_xml());
552    let id = crate::model::MediaId(imp.alloc());
553    let content_type = imp.pkg.content_type(target)?;
554    imp.media.push(crate::model::MediaItem {
555        id,
556        name: crate::pptx::opc::file_of(target).to_string(),
557        content_type,
558        data: std::sync::Arc::new(xml.into_bytes()),
559        link: None,
560    });
561    imp.media_by_part.insert(norm_target, id);
562    Some(id)
563}

565/// SmartArt: read the pre-drawn shapes of its drawing part as a group.
566fn smartart(imp: &mut Imp, part: &Part, gd: &El, frame: &Shape, depth: usize) -> Option<Shape> {
567    if depth > 8 {
568        return None;
569    }
570    let rel_ids = gd.child("relIds")?;
571    let dm = part.rels.get(rel_ids.attr("r:dm").unwrap_or(""))?;
572    let data = imp.pkg.xml(&dm.target)?;
573    let drawing_rel = data.root.find("dataModelExt").and_then(|x| x.attr("relId")).and_then(|id| part.rels.get(id)).map(|r| r.target.clone())?;
574    let mut d = imp.pkg.xml(&drawing_rel)?;
575    super::resolve_mc(&mut d.root, 0);
576    let dpart = Part::new(imp.pkg, &drawing_rel, &d);
577    let tree = d.root.child("spTree")?;
578    let mut sub = IdCtx { fresh: true, ..Default::default() };
579    let children = sp_tree(imp, &dpart, &mut sub, tree, depth + 1);
580    if children.is_empty() {
581        return None;
582    }
583    let children = smartart_text_rects(imp, &mut sub, tree, children, depth + 1);
584    let mut g = frame.clone();
585    let x = frame.xfrm.unwrap_or_default();
586    g.kind = ShapeKind::Group { children, child: Xfrm::new(0.0, 0.0, x.w, x.h) };
587    g.diagram = diagram_outline(imp, part, rel_ids, &data.root);
588    // Node pictures, from the blips of the data model's presentation points.
589    if let Some(d) = &mut g.diagram {
590        let rels = imp.pkg.rels(&dm.target);
591        for (k, rid) in crate::pptx::dgm::node_pictures(&data.root) {
592            let Some(target) = rels.get(&rid).filter(|r| !r.external).map(|r| r.target.clone()) else { continue };
593            let media = imp.media_for(&target);
594            if let Some(n) = d.nodes.get_mut(k) {
595                n.picture = media;
596            }
597        }
598    }
599    Some(g)
600}

602/// A SmartArt drawing shape (`dsp:sp`) keeps its text rectangle in `dsp:txXfrm`, apart from the
603/// shape's own box: the text of a narrow shape turned by 90° is often laid out across it. Put each
604/// shape's text on that rectangle. `shapes` are what [`sp_tree`] read from `tree`, in order.
605fn smartart_text_rects(imp: &mut Imp, ctx: &mut IdCtx, tree: &El, shapes: Vec<Shape>, depth: usize) -> Vec<Shape> {
606    if depth > MAX_TREE_DEPTH {
607        return shapes;
608    }
609    let mut out = Vec::with_capacity(shapes.len());
610    let mut shapes = shapes.into_iter();
611    // The same elements `sp_tree` turns into shapes.
612    for e in tree.elements().filter(|e| matches!(e.local(), "sp" | "pic" | "grpSp" | "cxnSp" | "graphicFrame")) {
613        let Some(mut s) = shapes.next() else { break };
614        let mut text_box = None;
615        match (e.local(), &mut s.kind) {
616            ("grpSp", ShapeKind::Group { children, .. }) => {
617                let kids = std::mem::take(children);
618                *children = smartart_text_rects(imp, ctx, e, kids, depth + 1);
619            }
620            ("sp", ShapeKind::Shape) => {
621                if let Some(t) = e.child("txXfrm") {
622                    text_box = text_on_rect(imp, ctx, &mut s, dml::xfrm(t).0);
623                }
624            }
625            _ => {}
626        }
627        out.push(s);
628        out.extend(text_box);
629    }
630    out.extend(shapes);
631    out
632}

634/// Lay out the text of `s` on `t`, a text rectangle in the same space as the shape's box, turned by
635/// the shape's rotation plus its own. When that is the shape's text area shrunk by some margin, the
636/// shape keeps its text with wider insets. Otherwise (a rectangle reaching outside the shape, or
637/// turned against it) the text moves to a new frameless text shape over `s`, which is returned:
638/// both export as plain DrawingML that any reader lays out the same way.
639fn text_on_rect(imp: &mut Imp, ctx: &mut IdCtx, s: &mut Shape, t: Xfrm) -> Option<Shape> {
640    const EPS: f64 = 0.01;
641    let x = s.xfrm?;
642    if !x.is_finite() || !t.is_finite() || t.w <= EPS || t.h <= EPS {
643        return None;
644    }
645    let body = s.text.as_mut().filter(|b| !b.is_empty())?;
646    let turn = |a: f64| a.rem_euclid(360.0);
647    let upright = |a: f64| turn(a + EPS) < 2.0 * EPS;
648    let bp = &mut body.body;
649    if !x.flip_h
650        && !x.flip_v
651        && upright(t.rot)
652        && bp.rot.is_none_or(upright)
653        && bp.vert.is_none_or(|v| v == TextDir::Horizontal)
654        && bp.upright != Some(true)
655    {
656        // Same orientation as the shape: express the rectangle as insets from the shape's text area.
657        let c = x.affine().inverse() * t.center();
658        let (l, top) = (c.x - t.w / 2.0, c.y - t.h / 2.0);
659        let (r, b) = (l + t.w, top + t.h);
660        let area = text_area(&s.geom, x.w, x.h);
661        let extra = [l - area.x0, top - area.y0, area.x1 - r, area.y1 - b];
662        if extra.iter().all(|v| v.is_finite() && *v >= -EPS) {
663            let [el, et, er, eb] = extra.map(|v| v.max(0.0));
664            bp.inset_l = Some(bp.inset_l.unwrap_or(7.2) + el);
665            bp.inset_t = Some(bp.inset_t.unwrap_or(3.6) + et);
666            bp.inset_r = Some(bp.inset_r.unwrap_or(7.2) + er);
667            bp.inset_b = Some(bp.inset_b.unwrap_or(3.6) + eb);
668            return None;
669        }
670    }
671    if ctx.count >= MAX_SHAPES {
672        return None;
673    }
674    ctx.count += 1;
675    let text = s.text.take();
676    // No fill, outline or effects; the style keeps only the text colour and font.
677    let style = s.style.clone().map(|mut st| {
678        st.fill_ref.0 = 0;
679        st.line_ref.0 = 0;
680        st.effect_ref.0 = 0;
681        st
682    });
683    Some(Shape {
684        id: ctx.assign(imp, None),
685        name: s.name.clone(),
686        xfrm: Some(Xfrm { rot: turn(x.rot + t.rot), flip_h: false, flip_v: false, ..t }),
687        fill: Some(Fill::None),
688        line: Some(crate::model::style::Line { fill: Some(Fill::None), ..Default::default() }),
689        style,
690        text,
691        click: s.click.clone(),
692        hover: s.hover.clone(),
693        ..Default::default()
694    })
695}

697/// Where a shape's geometry puts its text, in shape-local points (as the renderer computes it).
698fn text_area(geom: &Geom, w: f64, h: f64) -> crate::geom::Rect {
699    let full = crate::geom::Rect::new(0.0, 0.0, w, h);
700    match geom {
701        Geom::Preset { name, adj } => {
702            preset::build(name, w, h, adj).or_else(|| preset::build("rect", w, h, &[])).map(|g| g.text_rect).unwrap_or(full)
703        }
704        Geom::Custom { .. } => full,
705    }
706}

708/// The outline of a SmartArt graphic (for the Text Pane), with its layout, quick style and colour
709/// parts kept to be written back.
710fn diagram_outline(imp: &Imp, part: &Part, rel_ids: &El, data: &El) -> Option<crate::model::Diagram> {
711    let nodes = crate::pptx::dgm::outline(data);
712    if nodes.is_empty() {
713        return None;
714    }
715    let raw = |attr: &str| -> Option<String> {
716        let rel = part.rels.get(rel_ids.attr(attr)?)?;
717        let bytes = imp.pkg.get(&rel.target)?;
718        (bytes.len() <= 4 << 20).then(|| String::from_utf8(bytes.to_vec()).ok()).flatten()
719    };
720    let layout_xml = raw("r:lo");
721    let (layout_id, cat) = layout_xml
722        .as_deref()
723        .and_then(|x| crate::pptx::xml::parse(x.as_bytes()).ok())
724        .map(|d| {
725            let cat = d.root.child("catLst").and_then(|c| c.child("cat")).and_then(|c| c.attr("type")).unwrap_or("").to_string();
726            (d.root.attr("uniqueId").unwrap_or("").to_string(), cat)
727        })
728        .unwrap_or_default();
729    let lo_type = data
730        .path(&["ptLst"])
731        .and_then(|l| l.children_named("pt").find(|p| p.attr("type") == Some("doc")))
732        .and_then(|p| p.child("prSet"))
733        .and_then(|p| p.attr("loTypeId"))
734        .unwrap_or("");
735    let id = if layout_id.is_empty() { lo_type.to_string() } else { layout_id };
736    let pr = data.path(&["ptLst"]).and_then(|l| l.children_named("pt").find(|p| p.attr("type") == Some("doc"))).and_then(|p| p.child("prSet"));
737    let style = crate::pptx::dgm::style_for(pr.and_then(|p| p.attr("qsTypeId")).unwrap_or("")).to_string();
738    let colors = crate::pptx::dgm::colors_for(pr.and_then(|p| p.attr("csTypeId")).unwrap_or(""));
739    Some(crate::model::Diagram {
740        kind: crate::pptx::dgm::kind_for(&id, &cat).to_string(),
741        nodes,
742        style,
743        colors,
744        imported: true,
745        layout_id: id,
746        layout_xml,
747        style_xml: raw("r:qs"),
748        colors_xml: raw("r:cs"),
749    })
750}

752fn table(imp: &mut Imp, part: &Part, tbl: &El) -> Table {
753    // A table naming no style is drawn by PowerPoint with thin dark grid lines ("No Style, Table
754    // Grid"), not without lines.
755    let mut t = Table { style: "grid".into(), first_row: false, band_row: false, ..Default::default() };
756    if let Some(pr) = tbl.child("tblPr") {
757        t.background = pr.elements().find(|e| e.local().ends_with("Fill") && e.local() != "noFill").and_then(|f| dml::fill_el(imp, part, f));
758        t.first_row = pr.bool("firstRow").unwrap_or(false);
759        t.first_col = pr.bool("firstCol").unwrap_or(false);
760        t.last_row = pr.bool("lastRow").unwrap_or(false);
761        t.last_col = pr.bool("lastCol").unwrap_or(false);
762        t.band_row = pr.bool("bandRow").unwrap_or(false);
763        t.band_col = pr.bool("bandCol").unwrap_or(false);
764        t.rtl = pr.bool("rtl");
765        if let Some(id) = pr.child("tableStyleId").map(|e| e.text()) {
766            // A style the file defines is drawn from its definition (by its GUID).
767            // DeckCraft's own styles (written with GUIDs derived from their ids) map back to them.
768            let up = id.trim().to_ascii_uppercase();
769            t.style = if imp.table_style_ids.contains(&up) && !crate::pptx::tables::is_own_guid(&up) {
770                up
771            } else {
772                crate::pptx::tables::style_from_guid(id.trim())
773            };
774        }
775    }
776    t.cols = tbl
777        .child("tblGrid")
778        .map(|g| g.children_named("gridCol").take(1000).map(|c| dml::pt(c, "w").unwrap_or(72.0).clamp(0.0, 20_000.0)).collect())
779        .unwrap_or_default();
780    for tr in tbl.children_named("tr").take(1000) {
781        let mut row = Row { height: dml::pt(tr, "h").unwrap_or(0.0).clamp(0.0, 20_000.0), cells: vec![] };
782        for tc in tr.children_named("tc").take(1000) {
783            row.cells.push(cell(imp, part, tc));
784        }
785        t.rows.push(row);
786    }
787    if t.cols.is_empty() {
788        let n = t.rows.iter().map(|r| r.cells.len()).max().unwrap_or(0);
789        t.cols = vec![72.0; n];
790    }
791    t.normalize();
792    t
793}

795fn cell(imp: &mut Imp, part: &Part, tc: &El) -> Cell {
796    let mut c = Cell {
797        grid_span: tc.u32("gridSpan").unwrap_or(1).clamp(1, 1000),
798        row_span: tc.u32("rowSpan").unwrap_or(1).clamp(1, 1000),
799        h_merge: tc.bool("hMerge").unwrap_or(false),
800        v_merge: tc.bool("vMerge").unwrap_or(false),
801        ..Default::default()
802    };
803    if let Some(tx) = tc.child("txBody") {
804        c.text = dml::text_body(imp, part, tx);
805    }
806    if let Some(pr) = tc.child("tcPr") {
807        if ["marL", "marR", "marT", "marB"].iter().any(|n| pr.has_attr(n)) {
808            c.margins = Some([
809                dml::pt(pr, "marL").unwrap_or(7.2),
810                dml::pt(pr, "marR").unwrap_or(7.2),
811                dml::pt(pr, "marT").unwrap_or(3.6),
812                dml::pt(pr, "marB").unwrap_or(3.6),
813            ]);
814        }
815        c.anchor = pr.attr("anchor").and_then(Anchor::from_xml);
816        c.anchor_ctr = pr.bool("anchorCtr");
817        c.horz_overflow = pr.attr("horzOverflow").and_then(crate::model::text::TextOverflow::from_horz_xml);
818        c.vertical = pr.attr("vert").and_then(TextDir::from_xml);
819        for (i, n) in ["lnL", "lnR", "lnT", "lnB"].iter().enumerate() {
820            if let Some(ln) = pr.child(n)
821                && let Some(b) = c.borders.get_mut(i)
822            {
823                *b = Some(dml::line(imp, part, ln));
824            }
825        }
826        c.diag_down = pr.child("lnTlToBr").map(|l| dml::line(imp, part, l));
827        c.diag_up = pr.child("lnBlToTr").map(|l| dml::line(imp, part, l));
828        c.fill = dml::fill_of(imp, part, pr);
829    }
830    c
