8/// Fills of chart parts never reference images we keep; read colour and pattern fills only.
9fn simple_fill(sp: Option<&El>) -> Option<Fill> {
10    let sp = sp?;
11    for e in sp.elements() {
12        match e.local() {
13            "noFill" => return Some(Fill::None),
14            "solidFill" => return super::dml::color(e).map(|color| Fill::Solid { color }),
15            "gradFill" => return Some(Fill::Gradient(super::dml::gradient(e))),
16            "pattFill" => return Some(Fill::Pattern(super::dml::pattern(e))),
17            _ => {}
18        }
19    }
20    None
21}

51/// Slots a decoded chart holds, counting the chart itself as at least one.
52pub(crate) fn point_count(c: &Chart) -> usize {
53    let series = |s: &crate::model::chart::Series| {
54        let bars = s.error_bars.as_ref().map_or(0, |e| e.plus.len().saturating_add(e.minus.len()));
55        s.values.len().saturating_add(s.x_values.len()).saturating_add(s.sizes.len()).saturating_add(bars)
56    };
57    let levels: usize = c.category_levels.iter().map(Vec::len).sum();
58    1usize.saturating_add(c.categories.len()).saturating_add(levels).saturating_add(c.series.iter().map(series).fold(0, usize::saturating_add))
59}

71const MAX_CACHE_POINTS: usize = 100_000;

73/// String values from `c:cat`/`c:tx` (str or num refs, caches or literals). Every slot returned
74/// is taken from `budget`; once it runs out the values are cut short.
75fn strings(e: &El, budget: &mut usize) -> Vec<String> {
76    let cap = MAX_CACHE_POINTS.min(*budget);
77    let out = strings_capped(e, cap);
78    *budget = budget.saturating_sub(out.len());
79    out
80}

82fn strings_capped(e: &El, cap: usize) -> Vec<String> {
83    let cache = e.find("strCache").or_else(|| e.find("numCache")).or_else(|| e.find("strLit")).or_else(|| e.find("numLit"));
84    if let Some(c) = cache {
85        let n = (c.child("ptCount").and_then(|p| p.u32("val")).unwrap_or(0) as usize).min(cap);
86        let mut out = vec![String::new(); n];
87        for pt in c.children_named("pt").take(MAX_CACHE_POINTS) {
88            let i = pt.u32("idx").unwrap_or(0) as usize;
89            if i >= cap {
90                continue;
91            }
92            let v = pt.child("v").map(|v| v.text()).unwrap_or_default();
93            if i >= out.len() {
94                out.resize(i + 1, String::new());
95            }
96            if let Some(s) = out.get_mut(i) {
97                *s = v;
98            }
99        }
100        return out;
101    }
102    if let Some(ml) = e.find("multiLvlStrCache") {
103        // Use the innermost level.
104        if let Some(l) = ml.child("lvl") {
105            return l.children_named("pt").take(cap).map(|p| p.child("v").map(|v| v.text()).unwrap_or_default()).collect();
106        }
107    }
108    if cap > 0
109        && let Some(v) = e.child("v")
110    {
111        return vec![v.text()];
112    }
113    vec![]
114}

120/// Parse a chart part. Returns `None` for something that isn't a chart we can read.
121pub fn read_chart_xml(bytes: &[u8]) -> Option<Chart> {
122    let doc = crate::pptx::xml::parse(bytes).ok()?;
123    let mut root = doc.root;
124    super::resolve_mc(&mut root, 0);
125    if !root.is("chartSpace") {
126        return None;
127    }
128    let mut c = chart_from(&root)?;
129    c.raw = Some(String::from_utf8_lossy(bytes).into_owned());
130    Some(c)
131}

133/// The chart a graphic frame points at. A chart that would cross the presentation's point or
134/// decode limits is not modelled: it is kept as its original part with the relationships it
135/// reaches (`verbatim`), so a save writes it back whole instead of dropping it and its workbook.
136pub(crate) fn frame_chart(imp: &mut super::Imp, part: &super::Part, rid: &str, uri: &str) -> Option<Chart> {
137    let rel = part.rels.get(rid)?;
138    let chartex = uri.ends_with("/chartex");
139    let bytes = imp.pkg.xml_bytes(&rel.target)?;
140    if imp.chart_decode_bytes.saturating_add(bytes.len()) > MAX_CHART_DECODE_BYTES {
141        log::warn!("pptx: chart {} is past the presentation's chart decode limit and is not kept", rel.target);
142        return None;
143    }
144    imp.chart_decode_bytes += bytes.len();
145    let parsed = if imp.chart_points < imp.chart_points_cap {
146        let c = if chartex { super::read_chartex_xml(bytes) } else { super::read_chart_xml(bytes) }?;
147        let n = point_count(&c);
148        if imp.chart_points.saturating_add(n) <= imp.chart_points_cap {
149            imp.chart_points = imp.chart_points.saturating_add(n);
150            let mut c = c;
151            c.kept = capture_kept(imp, &rel.target);
152            return Some(c);
153        }
154        Some(c)
155    } else {
156        None
157    };
158    // Past the limit: keep the part itself (charged to the sidecar budget) and show an empty chart.
159    if chartex || imp.kept_sidecar_bytes.saturating_add(bytes.len()) > MAX_KEPT_TOTAL_BYTES {
160        log::warn!("pptx: chart {} is past the presentation's chart limits and is not kept", rel.target);
161        return None;
162    }
163    imp.kept_sidecar_bytes += bytes.len();
164    log::warn!("pptx: chart {} is past the presentation's chart limits; shown empty and written back unchanged", rel.target);
165    let mut c = parsed.unwrap_or_default();
166    c.series.clear();
167    c.categories.clear();
168    c.category_levels.clear();
169    c.raw = Some(String::from_utf8_lossy(bytes).into_owned());
170    c.verbatim = true;
171    c.kept = capture_kept(imp, &rel.target);
172    Some(c)
173}

179/// [`chart_from`] decoding at most `budget` data slots in all.
180fn chart_within(root: &El, mut budget: usize) -> Option<Chart> {
181    let ch = root.child("chart")?;
182    let pa = ch.child("plotArea")?;
183    let mut c = Chart { categories: vec![], series: vec![], title: None, legend: None, gridlines: false, ..Default::default() };
184    c.style = root.child_val("style").and_then(|v| v.parse().ok()).unwrap_or(2);
185    if let Some(t) = ch.child("title") {
186        c.title = Some(rich_text(t).unwrap_or_else(|| "Chart Title".into()));
187    }
188    if ch.child("autoTitleDeleted").and_then(|a| a.bool("val")) == Some(true) {
189        c.title = None;
190    }
191    c.legend = ch.child("legend").map(|l| l.child_val("legendPos").unwrap_or("r").to_string());
192    c.legend_overlay = ch.child("legend").and_then(|l| l.child("overlay")).and_then(|o| o.bool("val")).unwrap_or(false);
193    // An entry's own text (`c:legendEntry/c:txPr`) over the legend's, as for any text.
194    c.legend_entries = ch
195        .child("legend")
196        .into_iter()
197        .flat_map(|l| l.children_named("legendEntry"))
198        .take(MAX_LEGEND_ENTRIES)
199        .filter_map(|e| {
200            // The paragraph's own run properties first, then the list style's defaults
201            // (`a:lstStyle` comes before `a:p`, and is the less specific of the two).
202            let tx = e.child("txPr");
203            let para = tx.and_then(|t| t.child("p"));
204            let entry = crate::model::chart::LegendEntry {
205                idx: e.child("idx").and_then(|i| i.u32("val"))?,
206                size: text_size(para).or_else(|| text_size(tx)),
207                bold: text_bold(para).or_else(|| text_bold(tx)),
208                color: text_color(para).or_else(|| text_color(tx)),
209            };
210            (entry.size.is_some() || entry.bold.is_some() || entry.color.is_some()).then_some(entry)
211        })
212        .collect();
213    c.chart_fill = simple_fill(root.child("spPr"));
214    c.plot_fill = simple_fill(pa.child("spPr"));
215
216    let mut kinds: Vec<ChartType> = vec![];
217    // The value axis the first plot group uses is the primary one; a group on another value axis
218    // (a combo chart's line on the right) is on the secondary axis.
219    let val_map: std::collections::HashMap<&str, bool> =
220        pa.children_named("valAx").filter_map(|a| Some((a.child_val("axId")?, matches!(a.child_val("axPos"), Some("l" | "r"))))).collect();
221    let primary_val = primary_value_axis(pa, &val_map);
222    for g in pa.elements() {
223        let name = g.local();
224        if !name.ends_with("Chart") {
225            continue;
226        }
227        let Some(kind) = group_chart_type(name, g) else { continue };
228        if name == "stockChart" {
229            c.up_down_bars = g.child("upDownBars").is_some();
230        }
231        let group_labels = read_group_settings(g, &mut c);
232        kinds.push(kind);
233        let val_axis = group_value_axis(&val_map, g);
234        let secondary = val_axis.is_some() && val_axis != primary_val;
235        for s in g.children_named("ser").take(1000) {
236            let ser = read_series(s, g, (kind, secondary, group_labels), &mut c.categories, &mut budget);
237            c.series.push(ser);
238        }
239    }
240    // `autoTitleDeleted val="0"` without `c:title`: PowerPoint and LibreOffice draw the series name as the title of a single-series chart.
241    if ch.child("title").is_none() && ch.child("autoTitleDeleted").and_then(|a| a.bool("val")) == Some(false) && c.series.len() == 1 {
242        c.title = c.series.first().map(|s| s.name.clone()).filter(|n| !n.is_empty());
243    }
244    let first = *kinds.first()?;
245    c.kind = if kinds.iter().any(|k| *k != first) { ChartType::Combo } else { first };
246    if c.kind != ChartType::Combo {
247        for s in &mut c.series {
248            s.kind = None;
249            s.secondary_axis = false;
250        }
251    }
252    c.lang = root.child_val("lang").map(str::to_string);
253    let (val_ax, cat_ax) = read_axes(&mut c, pa, &primary_val);
254    read_text_styles(&mut c, root, ch, val_ax, cat_ax);
255    Some(c)
256}

258/// A group's value axis: its vertical one when it has two (scatter), else its one.
259fn group_value_axis(val_map: &std::collections::HashMap<&str, bool>, g: &El) -> Option<String> {
260    let ids: Vec<(&str, bool)> =
261        g.children_named("axId").filter_map(|a| a.attr("val")).filter_map(|id| val_map.get(id).copied().map(|v| (id, v))).collect();
262    ids.iter().find(|(_, vertical)| *vertical).or(ids.first()).map(|(id, _)| id.to_string())
263}

265/// The primary value axis is the one on the left (bottom for bar charts); the secondary one
266/// sits on the right or top, or crosses the category axis at its maximum.
267fn primary_value_axis(pa: &El, val_map: &std::collections::HashMap<&str, bool>) -> Option<String> {
268    let secondary_placed = |a: &El| a.child_val("crosses") == Some("max") || matches!(a.child_val("axPos"), Some("r" | "t"));
269    let val_axes_by_id: std::collections::HashMap<&str, &El> = pa.children_named("valAx").filter_map(|a| Some((a.child_val("axId")?, a))).collect();
270    let used: Vec<String> = pa.elements().filter(|g| g.local().ends_with("Chart")).filter_map(|g| group_value_axis(val_map, g)).collect();
271    used.iter().find(|id| val_axes_by_id.get(id.as_str()).is_some_and(|a| !secondary_placed(a))).or(used.first()).cloned()
272}

274/// The chart type of a plot group (`name` is its element name), or `None` for an unknown group.
275fn group_chart_type(name: &str, g: &El) -> Option<ChartType> {
276    let grouping = g.child_val("grouping").unwrap_or("clustered");
277    Some(match name {
278        "barChart" | "bar3DChart" => {
279            let bar = g.child_val("barDir") == Some("bar");
280            match (bar, grouping) {
281                (false, "stacked") => ChartType::StackedColumn,
282                (false, "percentStacked") => ChartType::PercentColumn,
283                (false, _) => ChartType::Column,
284                (true, "stacked") => ChartType::StackedBar,
285                (true, "percentStacked") => ChartType::PercentBar,
286                (true, _) => ChartType::Bar,
287            }
288        }
289        "lineChart" | "line3DChart" => {
290            let markers = g.child_val("marker").is_some_and(|v| v == "1" || v == "true")
291                && g.children_named("ser").all(|s| s.path(&["marker", "symbol"]).and_then(|m| m.attr("val")) != Some("none"));
292            match grouping {
293                "stacked" | "percentStacked" => ChartType::StackedLine,
294                _ if markers => ChartType::LineMarkers,
295                _ => ChartType::Line,
296            }
297        }
298        "areaChart" | "area3DChart" => {
299            if matches!(grouping, "stacked" | "percentStacked") {
300                ChartType::StackedArea
301            } else {
302                ChartType::Area
303            }
304        }
305        "pieChart" | "pie3DChart" | "ofPieChart" => ChartType::Pie,
306        "doughnutChart" => ChartType::Doughnut,
307        "scatterChart" => ChartType::Scatter,
308        "bubbleChart" => ChartType::Bubble,
309        "radarChart" => {
310            if g.child_val("radarStyle") == Some("filled") {
311                ChartType::FilledRadar
312            } else {
313                ChartType::Radar
314            }
315        }
316        "stockChart" => ChartType::Stock,
317        "surfaceChart" | "surface3DChart" => ChartType::Surface,
318        _ => return None,
319    })
320}

322/// Chart-wide settings a plot group carries (vary colours, gap, overlap, hole, data labels).
323/// Returns the group's own data-label setting.
324fn read_group_settings(g: &El, c: &mut Chart) -> Option<bool> {
325    // A missing <c:varyColors> means "vary by point" (the schema default is true), as PowerPoint draws it.
326    c.vary_colors = g.child_val("varyColors").is_none_or(|v| v == "1" || v == "true");
327    // A bar group without c:gapWidth / c:overlap takes the schema's defaults (150 %, 0 %),
328    // not the 219 % / -27 % PowerPoint writes into its new charts.
329    let bar_group = g.is("barChart") || g.is("bar3DChart");
330    if let Some(v) = g.child_val("gapWidth").and_then(|v| v.parse::<f64>().ok()) {
331        c.gap_width = (v / 100.0).clamp(0.0, 5.0);
332    } else if bar_group {
333        c.gap_width = 1.5;
334    }
335    if let Some(v) = g.child_val("overlap").and_then(|v| v.parse::<f64>().ok()) {
336        c.overlap = (v / 100.0).clamp(-1.0, 1.0);
337    } else if bar_group {
338        c.overlap = 0.0;
339    }
340    if let Some(v) = g.child_val("holeSize").and_then(|v| v.parse::<f64>().ok()) {
341        c.hole_size = (v / 100.0).clamp(0.0, 0.9);
342    }
343    // `<c:showVal/>` without a val attribute means true, as in the schema's default.
344    // The setting belongs to this plot group: a group that asks for no labels keeps them
345    // off, even when another group of a combo chart asks for them (#868).
346    let group_labels = g.path(&["dLbls", "showVal"]).map(|v| v.bool("val").unwrap_or(true));
347    if group_labels == Some(true) {
348        c.data_labels = true;
349    }
350    group_labels
351}

353/// Data labels' number format: the series' labels', the group's, else the source data's.
354fn label_number_format(s: &El, g: &El) -> Option<String> {
355    let label_fmt = |d: Option<&El>| {
356        d.and_then(|d| d.child("numFmt")).filter(|n| n.attr("sourceLinked") != Some("1")).and_then(|n| n.attr("formatCode")).map(str::to_string)
357    };
358    let source = s.child("val").or_else(|| s.child("yVal")).and_then(|v| v.find("numCache")).and_then(|c| c.child("formatCode")).map(|f| f.text());
359    label_fmt(s.child("dLbls")).or_else(|| label_fmt(g.child("dLbls"))).or(source).filter(|f| !f.is_empty())
360}

362/// Error bars on the values: a scatter chart's y bars, any other chart's only ones.
363fn read_error_bars(s: &El, kind: ChartType, budget: &mut usize) -> Option<ErrorBars> {
364    let e = s
365        .children_named("errBars")
366        .find(|e| e.child_val("errDir").is_none_or(|d| d == "y" || !matches!(kind, ChartType::Scatter | ChartType::Bubble)))?;
367    let d = ErrorBars::default();
368    Some(ErrorBars {
369        bar_type: e.child_val("errBarType").unwrap_or("both").to_string(),
370        val_type: e.child_val("errValType").unwrap_or("fixedVal").to_string(),
371        value: e.child_val("val").and_then(|v| v.parse().ok()).filter(|v: &f64| v.is_finite()).unwrap_or(d.value),
372        plus: e.child("plus").map(|p| numbers(p, budget)).unwrap_or_default(),
373        minus: e.child("minus").map(|p| numbers(p, budget)).unwrap_or_default(),
374        end_cap: e.child("noEndCap").and_then(|v| v.bool("val")) != Some(true),
375    })
376}

378/// One `c:ser` of group `g`. `group` is (chart type, on the secondary axis, the group's data-label
379/// setting); the first series with categories sets `categories`.
380fn read_series(s: &El, g: &El, group: (ChartType, bool, Option<bool>), categories: &mut Vec<String>, budget: &mut usize) -> Series {
381    let (kind, secondary, group_labels) = group;
382    let mut ser = Series {
383        name: s.child("tx").map(|t| strings(t, budget)).and_then(|v| v.into_iter().next()).unwrap_or_default(),
384        fill: simple_fill(s.child("spPr")),
385        line: simple_line(s.child("spPr")),
386        smooth: s.child("smooth").and_then(|v| v.bool("val")).unwrap_or(false),
387        marker: s.path(&["marker", "symbol"]).and_then(|m| m.attr("val")).filter(|m| *m != "none").map(String::from),
388        kind: Some(kind),
389        secondary_axis: secondary,
390        ..Default::default()
391    };
392    // `<c:dLbls><c:delete val="1"/>` hides every label of the series, whatever the group says.
393    if s.path(&["dLbls", "delete"]).map(|v| v.bool("val").unwrap_or(true)) == Some(true) {
394        ser.data_labels = Some(false);
395    // `<c:showVal/>` without a val attribute means true, as in the schema's default.
396    } else if let Some(d) = s.child("dLbls").and_then(|d| d.child("showVal")).map(|v| v.bool("val").unwrap_or(true)) {
397        ser.data_labels = Some(d);
398    } else if let Some(false) = group_labels {
399        // The group asks for no labels and the series does not say otherwise.
400        ser.data_labels = Some(false);
401    }
402    for t in s.children_named("trendline").take(16) {
403        if let Some(t) = trendline(t) {
404            ser.trendlines.push(t);
405        }
406    }
407    ser.number_format = label_number_format(s, g);
408    for dp in s.children_named("dPt") {
409        if let (Some(i), Some(f)) = (dp.child("idx").and_then(|i| i.u32("val")), simple_fill(dp.child("spPr"))) {
410            ser.point_fills.push((i as usize, f));
411        }
412    }
413    if let Some(cat) = s.child("cat").or_else(|| s.child("xVal")) {
414        let cats = strings(cat, budget);
415        if matches!(kind, ChartType::Scatter | ChartType::Bubble) {
416            ser.x_values = cats.iter().map(|v| v.trim().parse::<f64>().ok()).collect();
417        }
418        if categories.is_empty() {
419            *categories = cats;
420        }
421    }
422    if let Some(v) = s.child("val").or_else(|| s.child("yVal")) {
423        ser.values = numbers(v, budget);
424    }
425    ser.error_bars = read_error_bars(s, kind, budget);
426    if let Some(b) = s.child("bubbleSize") {
427        ser.sizes = numbers(b, budget);
428    }
429    ser
430}

432/// Axis scaling, titles, number formats and visibility. Returns the primary value axis and the
433/// category axis.
434fn read_axes<'a>(c: &mut Chart, pa: &'a El, primary_val: &Option<String>) -> (Option<&'a El>, Option<&'a El>) {
435    let vertical = |a: &&El| matches!(a.child_val("axPos"), Some("l" | "r"));
436    let by_id = |id: &Option<String>| pa.children_named("valAx").find(|a| id.is_some() && a.child_val("axId") == id.as_deref());
437    let val_ax = by_id(primary_val).or_else(|| pa.children_named("valAx").find(vertical)).or_else(|| pa.children_named("valAx").last());
438    let num = |a: &El, k: &str| a.child_val(k).and_then(|v| v.parse::<f64>().ok()).filter(|v| v.is_finite());
439    // An axis' number format: its own unless linked to the source (`sourceLinked="1"`, or no
440    // c:numFmt at all), in which case the data's format of the first series plotted on it.
441    let source_format = |ax: &El| {
442        let id = ax.child_val("axId");
443        pa.elements()

448            })

460    if c.kind == ChartType::Combo && c.series.iter().any(|s| s.secondary_axis) {

469            c.y2_hidden = deleted(a);
470        }
471    }
472    let cat_ax = pa.child("catAx").or_else(|| pa.child("dateAx")).or_else(|| pa.children_named("valAx").find(|a| !vertical(a)));

551const MAX_KEPT_PARTS: usize = 256;
552const MAX_KEPT_DEPTH: usize = 8;
553pub const MAX_KEPT_TOTAL_BYTES: usize = 32 * 1024 * 1024;
554
555/// Relationships and parts a chart points at, so a later save can write them back.
556pub(crate) fn capture_kept(imp: &mut super::Imp, chart_part: &str) -> crate::model::chart::ChartKept {
557    use crate::model::chart::{ChartKept, ChartRel};
558
559    let key = crate::pptx::opc::norm(chart_part);
560    if let Some(cached) = imp.kept_charts.get(&key) {
561        // Every frame that shows this chart gets its own copy, so each copy is charged like the first.
562        let bytes: usize = cached.parts.iter().map(part_bytes).sum::<usize>() + cached.rels.iter().map(rel_bytes).sum::<usize>();
563        if imp.kept_sidecar_bytes.saturating_add(bytes) > MAX_KEPT_TOTAL_BYTES {
564            log::warn!("pptx: chart sidecar retained byte budget exceeded for {chart_part}");
565            return ChartKept::default();
566        }
567        imp.kept_sidecar_bytes += bytes;
568        return cached.clone();
569    }
570
571    let charged_before = imp.kept_sidecar_bytes;
572    if imp.rejected_kept.get(&key) == Some(&charged_before) {
573        return ChartKept::default();
574    }
575    let mut kept = ChartKept::default();
576    // The chart document itself is rewritten; don't store a second copy of it.
577    let mut seen = std::collections::HashMap::from([(key.clone(), true)]);
578    for r in imp.pkg.rels(chart_part).items {
579        if !safe_rel(&r.id, &r.ty, &r.target_raw) {
580            log::warn!("pptx: skipping chart relationship {}", r.id);
581            continue;
582        }
583        if r.external {
584            kept.rels.push(ChartRel { id: r.id, rel_type: r.ty, target: r.target_raw, part: String::new(), external: true });
585            continue;
586        }
587        if copy_part(imp.pkg, &r.target, &mut kept.parts, &mut seen, &mut imp.kept_sidecar_bytes, 0) {
588            kept.rels.push(ChartRel { id: r.id, rel_type: r.ty, target: r.target_raw, part: r.target, external: false });
589        }
590    }
591    // The first capture's relationship strings and the parts' names and content types are retained
592    // too (the part bytes were charged as copied).
593    let rel_total: usize =
594        kept.rels.iter().map(rel_bytes).sum::<usize>() + kept.parts.iter().map(|p| p.name.len() + p.content_type.len()).sum::<usize>();
595    if imp.kept_sidecar_bytes.saturating_add(rel_total) > MAX_KEPT_TOTAL_BYTES {
596        log::warn!("pptx: chart sidecar retained byte budget exceeded for {chart_part}");
597        // The dropped parts must not stay charged against later charts.
598        imp.kept_sidecar_bytes = charged_before;
599        imp.rejected_kept.insert(key, charged_before);
600        return ChartKept::default();
601    }
602    imp.kept_sidecar_bytes += rel_total;
603    imp.kept_charts.insert(key, kept.clone());
604    kept
605}

607fn safe_rel(id: &str, ty: &str, target: &str) -> bool {
608    [id, ty, target].iter().all(|s| !s.is_empty() && s.len() <= 4096 && !s.chars().any(|c| c.is_control() || matches!(c, '<' | '>' | '"' | '\'')))
609}

611/// `true` when `part` is stored (now or from an earlier relationship).
612fn copy_part(
613    pkg: &crate::pptx::opc::Package,
614    part: &str,
615    out: &mut Vec<crate::model::chart::ChartPart>,
616    seen: &mut std::collections::HashMap<String, bool>,
617    retained_bytes: &mut usize,
618    depth: usize,
619) -> bool {
620    use crate::model::chart::ChartPart;
621
622    if depth > MAX_KEPT_DEPTH || out.len() >= MAX_KEPT_PARTS {
623        log::warn!("pptx: chart part closure stopped at {part}");
624        return false;
625    }
626    let key = crate::pptx::opc::norm(part);
627    if let Some(ok) = seen.get(&key) {
628        return *ok;
629    }
630    if !crate::pptx::opc::sidecar_part(part) {
631        log::warn!("pptx: chart relationship target {part} is not kept");
632        seen.insert(key, false);
633        return false;
634    }
635    let Some(bytes) = pkg.get(part) else {
636        log::warn!("pptx: chart relationship target {part} is missing");
637        seen.insert(key, false);
638        return false;
639    };
640    let name = pkg.original_name(part).unwrap_or(part).trim_start_matches('/').to_string();
641    if !crate::pptx::opc::sidecar_part(&name) {
642        log::warn!("pptx: chart part name {name} is not kept");
643        seen.insert(key, false);
644        return false;
645    }
646    let content_type = pkg.content_type(part).unwrap_or_default();
647    let stored_bytes = if *retained_bytes + bytes.len() <= MAX_KEPT_TOTAL_BYTES {
648        *retained_bytes += bytes.len();
649        bytes.to_vec()
650    } else {
651        log::warn!("pptx: chart sidecar retained byte budget exceeded for {part}");
652        Vec::new()
653    };
654    out.push(ChartPart { name, content_type, bytes: stored_bytes, native_entry: None });
655    seen.insert(key, true);
656    let rels_name = crate::pptx::opc::rels_path(part);
657    if crate::pptx::opc::sidecar_part(&rels_name)
658        && seen.get(&crate::pptx::opc::norm(&rels_name)).is_none()
659        && let Some(rels_bytes) = pkg.get(&rels_name)
660    {
661        seen.insert(crate::pptx::opc::norm(&rels_name), true);
662        let name = pkg.original_name(&rels_name).unwrap_or(&rels_name).trim_start_matches('/').to_string();
663        let stored_rels = if *retained_bytes + rels_bytes.len() <= MAX_KEPT_TOTAL_BYTES {
664            *retained_bytes += rels_bytes.len();
665            rels_bytes.to_vec()
666        } else {
667            Vec::new()
668        };
669        out.push(ChartPart {
670            name,
671            content_type: "application/vnd.openxmlformats-package.relationships+xml".into(),
672            bytes: stored_rels,
673            native_entry: None,
674        });
675        let targets: Vec<String> = pkg.rels(part).items.into_iter().filter(|r| !r.external).map(|r| r.target).collect();
676        for target in targets {
677            copy_part(pkg, &target, out, seen, retained_bytes, depth + 1);
678        }
679    }
680    true
681}
