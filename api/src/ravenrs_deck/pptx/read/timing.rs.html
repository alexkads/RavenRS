11fn ms_attr(e: &El, name: &str) -> Option<u32> {
12    let v = e.attr(name)?.trim();
13    if v == "indefinite" {
14        return None;
15    }
16    v.parse::<f64>().ok().filter(|v| v.is_finite()).map(|v| v.clamp(0.0, 86_400_000.0) as u32)
17}

19pub fn transition(imp: &mut Imp, part: &Part, t: &El) -> Option<Transition> {
20    let effect = t.elements().find(|e| !matches!(e.local(), "sndAc" | "extLst"));
21    let mut tr = Transition { advance_on_click: t.bool("advClick").unwrap_or(true), advance_after_ms: ms_attr(t, "advTm"), ..Default::default() };
22    match effect.and_then(maps::transition_from_xml) {
23        Some((k, o)) => {
24            tr.kind = k;
25            tr.option = o;
26        }
27        None => match effect {
28            Some(e) => {
29                tr.kind = e.local().to_string();
30                tr.raw = Some(part.keep(t));
31            }
32            None => tr.kind = "none".into(),
33        },
34    }
35    let default_dur = crate::model::anim::TRANSITIONS.iter().find(|x| x.0 == tr.kind).map(|x| x.3).unwrap_or(1000);
36    // `spd` defaults to "fast" in the schema; PowerPoint omits both attributes for 0.5 s.
37    tr.duration_ms = ms_attr(t, "dur").unwrap_or(match t.attr("spd") {
38        Some("med") => 750,
39        Some("slow") => default_dur.max(1000),
40        _ => 500,
41    });
42    if let Some(snd) = t.path(&["sndAc", "stSnd", "snd"]) {
43        tr.sound = dml::media_ref(imp, part, snd);
44        tr.sound_loop = t.path(&["sndAc", "stSnd"]).and_then(|s| s.attr("loop")).is_some_and(|v| v == "1" || v == "true");
45    }
46    tr.stop_sound = t.path(&["sndAc", "endSnd"]).is_some();
47    if tr.kind == "none" && tr.advance_on_click && tr.advance_after_ms.is_none() && tr.sound.is_none() && !tr.stop_sound {
48        return None;
49    }
50    Some(tr)
51}

53struct Found<'a> {
54    ctn: &'a El,
55    trigger: Option<u32>,
56    /// Click group the effect sits in: a new one starts at every time node whose start is `indefinite`.
57    group: usize,
58    /// Sum of the start delays of the time nodes between the click group and the effect, in ms.
59    offset: u32,
60}
61
62/// Where `collect` is in the time tree: the click group and the delay accumulated inside it.
63#[derive(Clone, Copy)]
64struct At {
65    trigger: Option<u32>,
66    group: usize,
67    offset: u32,
68}
69
70fn collect<'a>(e: &'a El, at: At, groups: &mut usize, out: &mut Vec<Found<'a>>, depth: usize) {
71    if depth > 64 || out.len() > 10_000 {
72        return;
73    }
74    for c in e.elements() {
75        if c.is("cTn") && c.has_attr("presetClass") {
76            out.push(Found { ctn: c, trigger: at.trigger, group: at.group, offset: at.offset });
77            continue;
78        }
79        let mut at = at;
80        if c.is("seq")
81            && let Some(ctn) = c.child("cTn")
82            && ctn.attr("nodeType") == Some("interactiveSeq")
83        {
84            at.trigger = ctn.path(&["stCondLst", "cond", "tgtEl", "spTgt"]).and_then(|s| s.u32("spid"));
85        }
86        if c.is("cTn")
87            && let Some(cond) = c.path(&["stCondLst", "cond"])
88        {
89            if cond.attr("delay").map(str::trim) == Some("indefinite") {
90                *groups += 1;
91                at.group = *groups;
92                at.offset = 0;
93            } else {
94                at.offset = at.offset.saturating_add(ms_attr(cond, "delay").unwrap_or(0));
95            }
96        }
97        collect(c, at, groups, out, depth + 1);
98    }
99}

101/// How long an effect holds back an After Previous one, as the show timeline counts it.
102fn span_ms(a: &Animation) -> u32 {
103    let iterations = if a.repeat == u32::MAX || a.repeat <= 1 { 1 } else { a.repeat.min(1_000) };
104    a.duration_ms.saturating_mul(if a.auto_reverse { 2 } else { 1 }).saturating_mul(iterations)
105}

107/// Start times (ms after the click) that the flat effect list plays at, effect after effect.
108fn flat_starts(anims: &[&Animation]) -> Vec<u32> {
109    let (mut begin, mut end) = (0u32, 0u32);
110    anims
111        .iter()
112        .map(|a| {
113            if a.start == AnimStart::AfterPrevious {
114                begin = end;
115            }
116            let start = begin.saturating_add(a.delay_ms);
117            end = end.max(start.saturating_add(span_ms(a)));
118            start
119        })
120        .collect()
121}

123/// PowerPoint places each effect at the delays of all its enclosing time nodes, so a file may list
124/// effects out of time order (a countdown written first, then effects at 5 s placed only by their
125/// parent's delay). The flat list only knows "after/with previous" plus the effect's own delay, so
126/// a click group listed out of time order, whose flat reading disagrees with the file, is sorted by
127/// start time and its starts and delays rewritten to land every effect where PowerPoint plays it.
128/// Every other group is left untouched.
129fn place_by_parent_delays(out: &mut [(Animation, u32, String, bool, usize, u32)], at: &[(Option<u32>, usize, u32)]) {
130    let mut groups: Vec<Vec<usize>> = vec![];
131    let mut slot: std::collections::HashMap<(Option<u32>, usize), usize> = std::collections::HashMap::new();
132    for (i, &(t, g, _)) in at.iter().enumerate() {
133        let n = *slot.entry((t, g)).or_insert_with(|| {
134            groups.push(vec![]);
135            groups.len() - 1
136        });
137        groups[n].push(i);
138    }
139    for idx in groups {
140        // More than one click in a group means the tree isn't one click per group: leave it.
141        if idx.iter().skip(1).any(|&i| out[i].0.start == AnimStart::OnClick) {
142            continue;
143        }
144        let abs: Vec<u32> = idx.iter().map(|&i| at[i].2.saturating_add(out[i].0.delay_ms)).collect();
145        // Only a file that lists effects out of time order needs this: files PowerPoint or DeckCraft
146        // save are in order, and their reading stays exactly as before.
147        if abs.is_sorted() {
148            continue;
149        }
150        let flat = flat_starts(&idx.iter().map(|&i| &out[i].0).collect::<Vec<_>>());
151        if flat == abs {
152            continue;
153        }
154        // The group's click stays on whichever effect now plays first.
155        let had_click = out[idx[0]].0.start == AnimStart::OnClick;
156        let mut order: Vec<usize> = (0..idx.len()).collect();
157        order.sort_by_key(|&k| abs[k]);
158        let mut sorted: Vec<(Animation, u32, String, bool, usize, u32)> = order.iter().map(|&k| out[idx[k]].clone()).collect();
159        let (mut begin, mut end) = (0u32, 0u32);
160        for (n, (&k, item)) in order.iter().zip(sorted.iter_mut()).enumerate() {
161            let a = &mut item.0;
162            let t = abs[k];
163            if n == 0 && had_click {
164                a.start = AnimStart::OnClick;
165                a.delay_ms = t;
166            } else if a.start == AnimStart::AfterPrevious && t >= end {
167                begin = end;
168                a.delay_ms = t - end;
169            } else {
170                a.start = AnimStart::WithPrevious;
171                a.delay_ms = t.saturating_sub(begin);
172            }
173            item.5 = 0;
174            end = end.max(t.saturating_add(span_ms(a)));
175        }
176        for (&i, item) in idx.iter().zip(sorted) {
177            out[i] = item;
178        }
179    }
180}

182fn max_dur(e: &El, depth: usize) -> Option<u32> {
183    if depth > 64 {
184        return None;
185    }
186    let mut best: Option<u32> = None;
187    for c in e.elements() {
188        if c.is("cTn")
189            && let Some(d) = ms_attr(c, "dur")
190            && d > 1
191        {
192            best = Some(best.map_or(d, |b: u32| b.max(d)));
193        }
194        if let Some(d) = max_dur(c, depth + 1) {
195            best = Some(best.map_or(d, |b| b.max(d)));
196        }
197    }
198    best
199}

201/// Does every behaviour time node (`p:cBhvr/p:cTn`) under the effect node `c` carry `autoRev`?
202fn behaviour_auto_reverses(c: &El) -> bool {
203    let mut behaviours = vec![];
204    c.find_all("cBhvr", &mut behaviours);
205    !behaviours.is_empty() && behaviours.iter().all(|b| b.child("cTn").and_then(|t| t.bool("autoRev")).unwrap_or(false))
206}

208/// Volume, mute, Loop until Stopped, Hide While Not Playing and Play Full Screen live in the timing tree
209/// (`p:video`/`p:audio` > `p:cMediaNode`), not on the picture: copy them onto the target's `MediaClip`.
210pub fn media_settings(timing: &El, ctx: &IdCtx, shapes: &mut [Shape]) {
211    let mut nodes = vec![];
212    for tag in ["video", "audio"] {
213        timing.find_all(tag, &mut nodes);
214    }
215    for node in nodes.into_iter().take(10_000) {
216        let Some(cm) = node.child("cMediaNode") else { continue };
217        let Some(id) = cm.path(&["tgtEl", "spTgt"]).and_then(|t| t.u32("spid")).and_then(|f| ctx.lookup(f)) else { continue };
218        let Some(ShapeKind::Media(clip)) = find_shape_mut(shapes, id).map(|s| &mut s.kind) else { continue };
219        if let Some(v) = cm.attr("vol").and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite()) {
220            clip.volume = (v / 100_000.0).clamp(0.0, 1.0);
221        }
222        if cm.bool("mute") == Some(true) {
223            clip.volume = 0.0;
224        }
225        if cm.bool("showWhenStopped") == Some(false) {
226            clip.hide_while_not_playing = true;
227        }
228        if cm.child("cTn").and_then(|c| c.attr("repeatCount")) == Some("indefinite") {
229            clip.loop_play = true;
230        }
231        if node.bool("fullScrn") == Some(true) {
232            clip.full_screen = true;
233        }
234    }
235}

237/// Flatten the timing tree into the Animation Pane list.
238pub fn animations(imp: &mut Imp, part: &Part, ctx: &IdCtx, timing: &El, shapes: &[Shape]) -> Vec<Animation> {
239    let _ = (imp, part);
240    let Some(tn) = timing.child("tnLst") else { return vec![] };
241    let mut found = vec![];
242    collect(tn, At { trigger: None, group: 0, offset: 0 }, &mut 0, &mut found, 0);
243    // Paragraph builds (`bldP`) by (spid, grpId) -> whether `build="p"`.
244    let bld_p: std::collections::HashMap<(u32, String), bool> = timing
245        .child("bldLst")
246        .map(|b| {
247            b.children_named("bldP")
248                .filter_map(|p| Some(((p.u32("spid")?, p.attr("grpId").unwrap_or("0").to_string()), p.attr("build") == Some("p"))))
249                .fold(std::collections::HashMap::new(), |mut m, (k, v)| {
250                    m.entry(k).and_modify(|e| *e |= v).or_insert(v);
251                    m
252                })
253        })
254        .unwrap_or_default();
255    let mut out: Vec<(Animation, u32, String, bool, usize, u32)> = vec![];
256    let mut at: Vec<(Option<u32>, usize, u32)> = vec![];
257    for f in found {
258        let c = f.ctn;
259        let class = AnimClass::from_xml(c.attr("presetClass").unwrap_or("entr"));
260        let preset = c.u32("presetID").unwrap_or(0);
261        let sub = c.u32("presetSubtype").unwrap_or(0);
262        let Some(tgt) = c.find("spTgt") else { continue };
263        let Some(file_id) = tgt.u32("spid") else { continue };
264        let Some(shape) = ctx.lookup(file_id) else { continue };
265        let paragraph = tgt.path(&["txEl", "pRg"]).and_then(|p| p.u32("st"));
266        let effect = maps::effect_for_preset(class, preset).unwrap_or_else(|| maps::fallback_effect(class));
267        // PowerPoint writes Float Down as its own preset (47), not as Float with a direction.
268        let (effect, option) = match (class, preset) {
269            (AnimClass::Entrance, 47) => ("float", "d".to_string()),
270            (AnimClass::Exit, 47) => ("floatOut", "d".to_string()),
271            _ => (effect, maps::option_from_subtype(effect, sub)),
272        };
273        let start = match c.attr("nodeType") {
274            Some("clickEffect") => AnimStart::OnClick,
275            Some("afterEffect") => AnimStart::AfterPrevious,
276            _ => AnimStart::WithPrevious,
277        };
278        let mut a = Animation {
279            shape,
280            class,
281            effect: effect.to_string(),
282            option,
283            start,
284            duration_ms: max_dur(c, 0).unwrap_or(0),
285            delay_ms: c.path(&["stCondLst", "cond"]).and_then(|d| ms_attr(d, "delay")).unwrap_or(0),
286            paragraph,
287            trigger: f.trigger.and_then(|t| ctx.lookup(t)),
288            preset_id: Some(preset),
289            preset_subtype: Some(sub),
290            // PowerPoint's "Auto-reverse" is `autoRev` on the behaviour nodes (`p:cBhvr/p:cTn`); Color Pulse
291            // always writes it there and plays back and forth on its own, so it isn't read as the option.
292            auto_reverse: c.bool("autoRev").unwrap_or(false) || (effect != "colorPulse" && behaviour_auto_reverses(c)),
293            // PowerPoint's "Rewind when done playing" is `fill="remove"` on the effect node.
294            rewind: c.attr("fill") == Some("remove"),
295            smooth_start: c.f64("accel").map(|v| v / 100_000.0).unwrap_or(0.0),
296            smooth_end: c.f64("decel").map(|v| v / 100_000.0).unwrap_or(0.0),
297            bounce_end: c.f64("bounceEnd").map(|v| (v / 100_000.0).clamp(0.0, 1.0)).unwrap_or(0.0),
298            ..Default::default()
299        };
300        a.repeat = match c.attr("repeatCount") {
301            Some("indefinite") => u32::MAX,
302            Some(v) => v.parse::<f64>().ok().filter(|v| v.is_finite()).map(|v| (v / 1000.0).round().clamp(1.0, 10_000.0) as u32).unwrap_or(1),
303            None => 1,
304        };
305        if let Some(m) = c.find("animMotion") {
306            a.path = m.attr("path").map(String::from);
307        }
308        if let Some(r) = c.find("animRot") {
309            a.amount = r.f64("by").map(|v| v / 60_000.0);
310            if a.effect == "spin" && a.amount.is_some_and(|v| v < 0.0) {
311                a.option = "counterClockwise".into();
312            }
313        } else if let Some(s) = c.find("animScale").and_then(|s| s.child("by")) {
314            a.amount = s.f64("x").map(|v| v / 100_000.0);
315        }
316        if let Some(clr) = c.find("animClr").and_then(|x| x.child("to")) {
317            a.color = dml::color(clr);
318        }
319        let grp = c.attr("grpId").unwrap_or("0").to_string();
320        let build = paragraph.is_some() && bld_p.get(&(file_id, grp.clone())) != Some(&false);
321        out.push((a, file_id, grp, build, f.group, f.offset));
322        at.push((f.trigger, f.group, f.offset));
323    }
324    place_by_parent_delays(&mut out, &at);
325    // A paragraph build's effects, one per paragraph, become one by-paragraph animation only when
326    // the timeline plays that back exactly: PowerPoint lets each paragraph keep its own start and
327    // effect, and other shapes' effects can sit between them (#844).
328    let same_build = |x: &(Animation, u32, String, bool, usize, u32), y: &(Animation, u32, String, bool, usize, u32)| {
329        x.3 && y.3 && x.1 == y.1 && x.2 == y.2 && x.0.trigger == y.0.trigger
330    };
331    let mut anims = Vec::with_capacity(out.len());
332    for run in out.chunk_by(same_build) {
333        let effects: Vec<&Animation> = run.iter().map(|(a, ..)| a).collect();
334        let offsets: Vec<u32> = run.iter().map(|r| r.5).collect();
335        let build = run.first().is_some_and(|r| r.3);
336        let explicit_bld = run.first().is_some_and(|r| bld_p.get(&(r.1, r.2.clone())) == Some(&true));
337        let same_click_group = run.first().is_some_and(|first| first.0.start == AnimStart::OnClick || run.iter().all(|r| r.4 == first.4));
338        let text = effects.first().and_then(|a| find_shape(shapes, a.shape)).and_then(|s| s.text.as_ref());
339        match (effects.first(), text) {
340            (Some(first), Some(text)) if build && (explicit_bld || same_click_group) && plays_as_build(&effects, &offsets, text, explicit_bld) => {
341                anims.push(Animation { text_build: TextBuild::ByParagraph, paragraph: None, ..(*first).clone() });
342            }
343            _ => anims.extend(effects.into_iter().cloned()),
344        }
345    }
346    anims
347}

349/// Are `effects` (paragraph effects of one shape, in file order) what a by-paragraph build of the
350/// first one plays: one effect per paragraph the build animates, in paragraph order, later top
351/// paragraphs on click (after previous when the build doesn't start on click), deeper ones with
352/// their parent, all with the same settings?
353fn plays_as_build(effects: &[&Animation], offsets: &[u32], text: &TextBody, explicit_bld: bool) -> bool {
354    let roles = paragraph_build_roles(text);
355    let expected = roles.iter().enumerate().filter(|(i, r)| match r {
356        BuildRole::Top => true,
357        BuildRole::Member(_) => text.paragraphs.get(*i).is_some_and(build_filled),
358        BuildRole::Unowned => false,
359    });
360    let Some(first) = effects.first() else { return false };
361    let off0 = offsets.first().copied().unwrap_or(0);
362    let flat = flat_starts(effects);
363    let parent_starts_match = offsets
364        .iter()
365        .zip(effects.iter())
366        .zip(flat.iter())
367        .all(|((&off, a), &ft)| off == off0 || off.saturating_add(a.delay_ms) == off0.saturating_add(ft));
368    if !explicit_bld && (!parent_starts_match || (first.start != AnimStart::OnClick && (span_ms(first) != 500 || first.delay_ms != 0 || off0 != 0))) {
369        return false;
370    }
371    let settings = |a: &Animation| Animation { start: AnimStart::OnClick, paragraph: None, ..a.clone() };
372    let base = settings(first);
373    let mut matched = 0;
374    for (k, (i, role)) in expected.enumerate() {
375        let Some(a) = effects.get(k) else { return false };
376        let start_ok = match role {
377            BuildRole::Top if k == 0 => true,
378            BuildRole::Top if first.start == AnimStart::OnClick => a.start == AnimStart::OnClick,
379            BuildRole::Top => a.start == AnimStart::AfterPrevious,
380            _ => a.start == AnimStart::WithPrevious,
381        };
382        if a.paragraph != u32::try_from(i).ok() || !start_ok || settings(a) != base {
383            return false;
384        }
385        matched = k + 1;
386    }
387    matched == effects.len()
