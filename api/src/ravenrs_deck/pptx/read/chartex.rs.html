1//! Office 2016 charts (`cx:chartSpace`, the `chartEx` part): waterfall, funnel, treemap,
2//! sunburst, histogram, Pareto and box & whisker. Their data sits in `cx:chartData` as levels of
3//! points, and each series names its layout (`layoutId`).
4
5use super::chart::rich_text;
6use crate::model::chart::{Chart, ChartType, Series};
7use crate::pptx::xml::El;
8use std::collections::HashMap;
9
10/// Points per level and levels per dimension we keep (guards against hostile parts).
11const MAX_POINTS: usize = 100_000;
12const MAX_CHARTEX_POINTS: usize = 1_000_000;
13
14/// One `cx:data`: its categories (the innermost level) and named numeric dimensions.
15#[derive(Default)]
16struct Data {
17    categories: Vec<String>,
18    /// Outer category levels, innermost first.
19    levels: Vec<Vec<String>>,
20    numbers: HashMap<String, (Vec<Option<f64>>, Option<String>)>,
21}
22
23/// A level's points, by `idx`.
24fn points(lvl: &El, budget: &mut usize) -> Vec<String> {
25    if *budget == 0 {
26        return Vec::new();
27    }
28    let cap = MAX_POINTS.min(*budget);
29    let mut pts = Vec::new();
30    let mut max_idx = 0;
31    for pt in lvl.children_named("pt").take(cap) {
32        let i = pt.u32("idx").unwrap_or(0) as usize;
33        if i >= cap {
34            continue;
35        }
36        max_idx = max_idx.max(i + 1);
37        pts.push((i, pt.text()));
38    }
39    if pts.is_empty() {
40        return Vec::new();
41    }
42    let n = max_idx.min(cap);
43    let mut out = vec![String::new(); n];
44    for (i, s) in pts {
45        if let Some(target) = out.get_mut(i) {
46            *target = s;
47        }
48    }
49    *budget = budget.saturating_sub(out.len());
50    out
51}

53fn data(d: &El, budget: &mut usize) -> Data {
54    let mut out = Data::default();
55    for dim in d.elements().take(64) {
56        if *budget == 0 {
57            break;
58        }
59        // The first level is the innermost: a treemap's leaves, a waterfall's categories.
60        let Some(lvl) = dim.child("lvl") else { continue };
61        let ty = dim.attr("type").unwrap_or("");
62        if dim.is("strDim") || (dim.is("numDim") && ty == "cat") {
63            if ty == "cat" || out.categories.is_empty() {
64                out.categories = points(lvl, budget);
65                // Further levels are the hierarchy's parents, innermost first.
66                out.levels = dim.children_named("lvl").skip(1).take(crate::model::chart::MAX_LEVELS).map(|l| points(l, budget)).collect();
67            }
68        } else if dim.is("numDim") {
69            let vals = points(lvl, budget).into_iter().map(|s| s.trim().parse::<f64>().ok().filter(|v| v.is_finite())).collect();
70            out.numbers.insert(ty.to_string(), (vals, lvl.attr("formatCode").map(str::to_string)));
71        }
72    }
73    out
74}

76fn kind_of(layout: &str) -> Option<ChartType> {
77    Some(match layout {
78        "waterfall" => ChartType::Waterfall,
79        "funnel" => ChartType::Funnel,
80        "treemap" => ChartType::Treemap,
81        "sunburst" => ChartType::Sunburst,
82        "clusteredColumn" => ChartType::Histogram,
83        "boxWhisker" => ChartType::BoxWhisker,
84        "regionMap" => ChartType::RegionMap,
85        _ => return None,
86    })
87}

89/// Parse a `chartEx` part. `None` for anything else, and for layouts DeckCraft has no type for.
90pub fn read_chartex_xml(bytes: &[u8]) -> Option<Chart> {
91    let doc = crate::pptx::xml::parse(bytes).ok()?;
92    let mut root = doc.root;
93    super::resolve_mc(&mut root, 0);
94    let mut c = chartex_from(&root)?;
95    c.raw = Some(String::from_utf8_lossy(bytes).into_owned());
96    Some(c)
97}

99pub(crate) fn chartex_from(root: &El) -> Option<Chart> {
100    if !root.is("chartSpace") {
101        return None;
102    }
103    let mut decode_budget = MAX_CHARTEX_POINTS;
104    let datas: HashMap<String, Data> = root
105        .child("chartData")?
106        .children_named("data")
107        .take(64)
108        .map(|d| (d.attr("id").unwrap_or("0").to_string(), data(d, &mut decode_budget)))
109        .collect();
110    let ch = root.child("chart")?;
111    let region = ch.child("plotArea")?.child("plotAreaRegion")?;
112    let mut c = Chart { categories: vec![], series: vec![], title: None, legend: None, gridlines: false, ..Default::default() };
113    c.title = ch.child("title").map(|t| rich_text(t).filter(|s| !s.is_empty()).unwrap_or_else(|| "Chart Title".into()));
114    c.legend = ch.child("legend").map(|l| l.attr("pos").unwrap_or("t").to_string());
115    let mut kind = None;
116    let mut output_budget = MAX_CHARTEX_POINTS;
117    let mut categories_set = false;
118    for s in region.children_named("series").take(1000) {
119        let layout = s.attr("layoutId").unwrap_or("");
120        if layout == "paretoLine" {
121            // The cumulative line PowerPoint adds to a Pareto histogram.
122            kind = Some(ChartType::Pareto);
123            continue;
124        }
125        let k = kind_of(layout)?;
126        if s.bool("hidden") == Some(true) {
127            continue;
128        }
129        kind = Some(match kind {
130            Some(ChartType::Pareto) => ChartType::Pareto,
131            _ => k,
132        });
133        let d = datas.get(s.child("dataId").and_then(|i| i.attr("val")).unwrap_or("0"))?;
134        let (values_src, format) = ["val", "size", "colorVal"]
135            .iter()
136            .find_map(|t| d.numbers.get(*t))
137            .or_else(|| d.numbers.values().next())
138            .map(|(v, f)| (v.as_slice(), f.clone()))
139            .unwrap_or((&[], None));
140        let take_len = values_src.len().min(output_budget);
141        let values = values_src[..take_len].to_vec();
142        output_budget = output_budget.saturating_sub(values.len());
143        if !categories_set && (!d.categories.is_empty() || !d.levels.is_empty()) {
144            categories_set = true;
145            c.categories = d.categories.clone();
146            c.category_levels = d.levels.clone();
147            output_budget = output_budget.saturating_sub(c.categories.len());
148            for lvl in &c.category_levels {
149                output_budget = output_budget.saturating_sub(lvl.len());
150            }
151        }
152        if let Some(st) = s.child("layoutPr").and_then(|l| l.child("statistics")) {
153            c.inclusive_quartiles = st.attr("quartileMethod") == Some("inclusive");
154        }
155        if let Some(b) = s.child("layoutPr").and_then(|l| l.child("binning")) {
156            c.bin_count = b.child("binCount").and_then(|n| n.u32("val")).filter(|n| *n > 0);
157            c.bin_size = b.child("binSize").and_then(|n| n.attr("val")).and_then(|v| v.parse::<f64>().ok()).filter(|v| v.is_finite() && *v > 0.0);
158        }
159        if let Some(st) = s.child("layoutPr").and_then(|l| l.child("subtotals")) {
160            c.subtotals = st.children_named("idx").filter_map(|i| i.u32("val")).map(|i| i as usize).take(MAX_POINTS).collect();
161        }
162        let name = s.child("tx").and_then(|t| t.find("v")).map(|v| v.text()).unwrap_or_default();
163        c.series.push(Series {
164            name,
165            values,
166            data_labels: s.child("dataLabels").map(|l| l.bool("hidden") != Some(true)),
167            number_format: format,
168            ..Default::default()
169        });
170    }
171    c.kind = kind?;
172    c.data_labels = c.series.iter().any(|s| s.data_labels == Some(true));
173    c.vary_colors = matches!(c.kind, ChartType::Funnel | ChartType::Treemap | ChartType::Sunburst | ChartType::Waterfall);
174    // Waterfall, histogram and box charts have a value axis with gridlines by default.
175    c.gridlines = matches!(c.kind, ChartType::Waterfall | ChartType::Histogram | ChartType::Pareto | ChartType::BoxWhisker);
176    Some(c)
177}
