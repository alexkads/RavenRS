1//! Raven Test Studio, first step (#634, part of #627).
2//!
3//! A scenario is a flow of blocks: open the app, type, click, check. The Studio runs it with
4//! `ravenrs-e2e` against an app, marks each block as it goes (done, failed with the error, or
5//! not run), and keeps a log of the run. Everything on screen is a RavenRS component:
6//! `FlowLibrary` keeps the scenarios, `FlowToolbox` has the blocks, `FlowCanvas` shows the
7//! scenario and `FlowInspector` edits the selected block.
8//!
9//! For now the app under test is a small form that comes with the Studio
10//! ([`sample::expense_app`]) and runs in the same process. Opening the person's own app in
11//! another process, scenario files and the live preview come in the next steps of #627.
12//!
13//! The Studio is tested by the tool it is made of: its tests drive it with `ravenrs-e2e`, as
14//! `tests/suite/self_test.rs` shows.

64fn icon(kind: &str) -> IconName {
65    match kind {
66        "open" => IconName::Play,
67        "click" | "select" | "expand" | "collapse" | "scroll" | "drag" => IconName::MousePointer,
68        "type" | "fill" => IconName::Keyboard,
69        "wait" => IconName::Clock,
70        _ => IconName::Check,
71    }
72}

74/// Opens the app an Open block names: the sample form, another Studio with the sample
75/// scenario, or another Studio whose scenario fails on purpose, compiled in; or an app opened
76/// apart in a child process, such as Raven Finanças ([`apps`]). A Studio opened here runs its
77/// own scenario the same way, so the Studio can test the Studio.
78pub fn open(name: &str) -> Option<Result<runner::Session, Box<runner::Failure>>> {
79    let studio = |scenario: FlowGraph| {
80        Ok(runner::Session::Here(Box::new(ravenrs_e2e::E2e::new(
81            app_with(scenario, Rc::new(MemoryStorage::default())),
82            Theme::raven_dark(),
83        ))))
84    };
85    // An app of a project the Studio was pointed at comes first: pointed at its own folder, the
86    // Studio tests another Studio in a process of its own, not the one compiled in.
87    if apps::learned(name) {
88        return apps::open(name);
89    }
90    match name {
91        sample::STUDIO_APP => Some(studio(sample::scenario())),
92        sample::BROKEN_STUDIO_APP => Some(studio(sample::broken_scenario())),
93        other => runner::open_sample(other).or_else(|| apps::open(other)),
94    }
95}

122    /// The bar at the bottom: the version, how the Studio's own test went, and the updater's
123    /// notice when there is one.
124    fn status_bar(&self) -> StatusBar {
125        let version = StatusBar::text(format!("Raven Test Studio {VERSION}"));
126        let bar = match &self.health {
127            Some(health) => {
128                let (said, tone) = if health.passed() {
129                    (
130                        format!("Autoteste: {}", runner::summary(&health.outcomes, health.total)),
131                        Semantic::Success,
132                    )
133                } else {
134                    (
135                        format!("Autoteste falhou: {}", runner::summary(&health.outcomes, health.total)),
136                        Semantic::Danger,
137                    )
138                };
139                StatusBar::new().left((version, StatusBar::text(said).foreground(tone).test_id("health")))
140            }
141            None => StatusBar::new().left((version,)),
142        };
143        let Some(notice) = self.updater.notice() else { return bar };
144        let buttons: Vec<Button> = notice
145            .actions
146            .iter()
147            .filter(|action| {
148                !matches!(action, NoticeAction::News | NoticeAction::HowToUpdate | NoticeAction::Details)
149            })
150            .map(|&action| self.notice_button(action))
151            .collect();
152        let bar = bar.center((
153            StatusBar::text(notice.text).foreground(Semantic::Ink).truncate().grow(1.0),
154            HStack::new(buttons).spacing(4.0),
155        ));
156        if notice.important { bar.center_tone(Tone::Warning) } else { bar }
157    }
158
159    /// A button of the updater's notice, doing what Raven Finanças does with it.
160    fn notice_button(&self, action: NoticeAction) -> Button {
161        let updater = self.updater.clone();
162        let button = StatusBar::action(action.label());
163        let button = if action.is_primary() { button.style(ButtonStyle::Primary) } else { button };
164        button.on_click(move || match action {
165            NoticeAction::Update | NoticeAction::Retry => {
166                if let Some(news) = updater.status().news().cloned() {
167                    updater.install(news);
168                }
169            }
170            NoticeAction::Reopen => updater.reopen(),
171            NoticeAction::CheckAgain => updater.ask(),
172            _ => updater.dismiss(),
173        })
174    }
175}

183impl View for Studio {
184    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
185        // The updater's threads wake the screen when they have something to say.
186        let waker = cx.ui_waker();
187        self.updater.connect(move || waker.wake());
188        self.updater.poll();
189        // What was chosen on the icon's menu, then the menu as the window now stands.
190        if let Some(link) = &self.tray {
191            for id in link.take() {
192                if self.updater.tray_select(&id).is_some() {
193                    continue;
194                }
195                #[cfg(not(target_arch = "wasm32"))]
196                if id == tray::RUN_SELF_TEST {
197                    let waker = cx.ui_waker();
198                    link.run_self_test(move || waker.wake());
199                }
200            }
201            link.menu().set(link.entries(self.health.as_ref(), self.updater.tray_entries()));
202        }
203        let start = self.scenario.clone();
204        let graph = cx.use_state(move || start);
205        let selection = cx.use_state(FlowSelection::default);
206        let history = cx.use_state(FlowHistory::new);
207        let arrivals = cx.use_state(|| None::<FlowNode>);
208        let view = cx.use_state(FlowViewport::fit);
209        let run = cx.use_state(|| Rc::new(RefCell::new(Run::new(open))));
210        let log = cx.use_state(Vec::<Outcome>::new);
211
212        let mut canvas = FlowCanvas::new(graph.binding())
213            .view(view.binding())
214            .label("Cenário")
215            .selection(selection.binding())
216            .history(history.binding())
217            .arrivals(arrivals.binding());
218        let mut inspector =
219            FlowInspector::new(graph.binding(), selection.binding()).history(history.binding());
220        for (kind, caption) in KINDS {
221            let tone = if kind == "check" { Semantic::Success } else { Semantic::Accent };
222            canvas = canvas.kind(kind, FlowKind::new(icon(kind), caption).tone(tone));
223            inspector = inspector.kind(kind, caption);
224        }
225
226        let memo = cx.use_state(|| None::<(String, Vec<Outcome>)>);
227        let from_cache = cx.use_state(|| false);
228        let runs = Runs {
229            graph: graph.clone(),
230            run: run.clone(),
231            log: log.clone(),
232            memo,
233            from_cache: from_cache.clone(),
234        };
235        let finished = run.get().borrow().finished(&graph.get());
236        let ran = !log.get().is_empty();
237        #[cfg(not(target_arch = "wasm32"))]
238        let handing = cx.use_state(Handing::shared);
239        #[cfg(not(target_arch = "wasm32"))]
240        let turn = cx.use_state(|| 0_u32);
241        // What the agent said comes from another thread; the Studio's own report of the same
242        // scenario marks the blocks, never the agent's words.
243        #[cfg(not(target_arch = "wasm32"))]
244        let delegation = {
245            let shared = handing.get();
246            let mut h = lock(&shared);
247            let fresh = match &h.done {
248                Some(Ok(done)) if !h.shown => Some(done.checked.outcomes.clone()),
249                _ => None,
250            };
251            if let Some(outcomes) = fresh {
252                h.shown = true;
253                graph.set(runner::applied(&graph.get(), &outcomes));
254                log.set(outcomes);
255                from_cache.set(false);
256            }
257            DelegationView {
258                running: h.running.clone(),
259                card: h.done.as_ref().map(|r| match r {
260                    Ok(d) => (d.card(), d.confirmed()),
261                    Err(why) => (format!("A delegação não terminou: {why}"), false),
262                }),
263                hint: (!self.agent.installed()).then(|| {
264                    format!(
265                        "O {} não está instalado nesta máquina: instale o CLI e entre nele para delegar",
266                        self.agent.agent().program()
267                    )
268                }),
269            }
270        };
271        #[cfg(target_arch = "wasm32")]
272        let delegation = DelegationView::default();
273        let busy = delegation.running.is_some();
274        let bar = HStack::new((
275            Text::new("Raven Test Studio").style(TextStyle::Headline),
276            Spacer::new(),
277            run_button(&runs, true),
278            run_button(&runs, false),
279            next_button(&graph, &run, &log, finished),
280            #[cfg(not(target_arch = "wasm32"))]
281            delegate_button(&graph, &handing, &turn, &self.agent, busy, cx.ui_waker()),
282            stop_button(&runs, ran, busy, {
283                #[cfg(not(target_arch = "wasm32"))]
284                {
285                    Some(handing.get())
286                }
287                #[cfg(target_arch = "wasm32")]
288                {
289                    None::<()>
290                }
291            }),
292        ))
293        .spacing(8.0)
294        .padding(8.0);
295
296        cx.child(
297            &VStack::new((
298                bar,
299                HStack::new((
300                    VStack::new((
301                        FlowLibrary::new(graph.binding())
302                            .title("Cenários")
303                            .key("scenarios")
304                            .history(history.binding())
305                            .selection(selection.binding())
306                            .view(view.binding())
307                            .storage(self.storage.clone()),
308                        sample::toolbox().title("Blocos").arrivals(arrivals.binding()),
309                    ))
310                    .spacing(0.0)
311                    .align(Align::Stretch),
312                    canvas,
313                    VStack::new((
314                        inspector,
315                        run_log(&log.get(), runner::path(&graph.get()).len(), from_cache.get(), &delegation),
316                    ))
317                    .spacing(0.0)
318                    .align(Align::Stretch),
319                ))
320                .align(Align::Stretch)
321                .grow(1.0),
322                self.status_bar(),
323            ))
324            .spacing(0.0)
325            .align(Align::Stretch)
326            .grow(1.0),
327        )
328    }
329}

333/// The last full run of this window, by the key of the scenario it ran.
334type Memo = State<Option<(String, Vec<Outcome>)>>;

336/// What the run buttons share.
337#[derive(Clone)]
338struct Runs {
339    graph: State<FlowGraph>,
340    run: RunState,
341    log: State<Vec<Outcome>>,
342    memo: Memo,
343    from_cache: State<bool>,
344}

346/// Runs the whole scenario from the start, stopping at the first block that fails. With
347/// `use_cache`, a scenario that did not change since the last full run comes back from it.
348fn run_button(runs: &Runs, use_cache: bool) -> impl View {
349    let runs = runs.clone();
350    let (label, id, style) = if use_cache {
351        ("Rodar", "run", ButtonStyle::Primary)
352    } else {
353        ("Rodar sem cache", "run-fresh", ButtonStyle::Secondary)
354    };
355    Button::new(label)
356        .icon(IconName::Play)
357        .style(style)
358        .on_click(move || {
359            let graph = runs.graph.get();
360            let key = cache::scenario_key(&graph);
361            if use_cache && let Some((_, outcomes)) = runs.memo.get().filter(|(k, _)| *k == key) {
362                runs.graph.set(runner::applied(&graph, &outcomes));
363                runs.run.set(Rc::new(RefCell::new(Run::finished_with(open))));
364                runs.log.set(outcomes);
365                runs.from_cache.set(true);
366                return;
367            }
368            let mut g = runner::cleared(&graph);
369            let mut fresh = Run::new(open);
370            let outcomes = fresh.to_end(&mut g);
371            runs.run.set(Rc::new(RefCell::new(fresh)));
372            runs.graph.set(g);
373            runs.memo.set(Some((key, outcomes.clone())));
374            runs.log.set(outcomes);
375            runs.from_cache.set(false);
376        })
377        .test_id(id)
378}

380/// Runs one more block, starting a run when none is going.
381fn next_button(
382    graph: &State<FlowGraph>,
383    run: &RunState,
384    log: &State<Vec<Outcome>>,
385    finished: bool,
386) -> impl View {
387    let (graph, run, log) = (graph.clone(), run.clone(), log.clone());
388    Button::new("Próximo passo")
389        .style(ButtonStyle::Secondary)
390        .disabled(finished)
391        .on_click(move || {
392            let mut g = graph.get();
393            let outcome = run.get().borrow_mut().step(&mut g);
394            if let Some(outcome) = outcome {
395                graph.set(g);
396                let mut all = log.get();
397                all.push(outcome);
398                log.set(all);
399            }
400        })
401        .test_id("next")
402}

404/// Puts every block back and forgets the run; the cache keeps the last full run. While an
405/// agent has the scenario, it ends the agent's CLI instead.
406#[cfg(not(target_arch = "wasm32"))]
407fn stop_button(runs: &Runs, ran: bool, busy: bool, handing: Option<Shared>) -> impl View {
408    let runs = runs.clone();
409    Button::new("Parar")
410        .style(ButtonStyle::Ghost)
411        .disabled(!ran && !busy)
412        .on_click(move || {
413            if let Some(handing) = handing.as_ref().filter(|_| busy) {
414                lock(handing).stop.store(true, std::sync::atomic::Ordering::SeqCst);
415                return;
416            }
417            runs.graph.set(runner::cleared(&runs.graph.get()));
418            runs.run.set(Rc::new(RefCell::new(Run::new(open))));
419            runs.log.set(Vec::new());
420            runs.from_cache.set(false);
421        })
422        .test_id("stop")
423}

425/// What the log shows of a delegation: that an agent has the scenario, its card once it
426/// answered, and why the button is off when the agent is not installed.
427#[derive(Clone, Debug, Default)]
428struct DelegationView {
429    running: Option<String>,
430    card: Option<(String, bool)>,
431    hint: Option<String>,
432}

434/// A delegation of this window, shared with the thread that runs it.
435#[cfg(not(target_arch = "wasm32"))]
436#[derive(Default)]
437struct Handing {
438    /// "Delegado ao Codex (modelo)…" while the agent works.
439    running: Option<String>,
440    /// Set by "Parar"; the agent's CLI is ended at its next event.
441    stop: std::sync::Arc<std::sync::atomic::AtomicBool>,
442    /// What came back.
443    done: Option<Result<delegate::Delegation, String>>,
444    /// Whether the window already marked its blocks with the report that came back.
445    shown: bool,
446}
447
448#[cfg(not(target_arch = "wasm32"))]
449type Shared = std::sync::Arc<std::sync::Mutex<Handing>>;
450
451#[cfg(not(target_arch = "wasm32"))]
452impl Handing {
453    fn shared() -> Shared {
454        Shared::default()
455    }
456}
457
458#[cfg(not(target_arch = "wasm32"))]
459fn lock(shared: &Shared) -> std::sync::MutexGuard<'_, Handing> {
460    shared.lock().unwrap_or_else(std::sync::PoisonError::into_inner)
461}

463/// Hands the scenario on the canvas to the agent, in a thread of its own, so the window keeps
464/// answering while the agent works; the thread wakes the window when the answer is in.
465#[cfg(not(target_arch = "wasm32"))]
466fn delegate_button(
467    graph: &State<FlowGraph>,
468    handing: &State<Shared>,
469    turn: &State<u32>,
470    agent: &ravenrs_agent_cli::Cli,
471    busy: bool,
472    waker: UiWaker,
473) -> impl View {
474    let name = match agent.agent() {
475        ravenrs_agent_cli::Agent::Codex => "Codex",
476        ravenrs_agent_cli::Agent::Claude => "Claude",
477    };
478    let has_open = graph.get().nodes.iter().any(|n| n.kind == "open");
479    let (graph, handing, turn, agent) = (graph.clone(), handing.get(), turn.clone(), agent.clone());
480    let installed = agent.installed();
481    Button::new(format!("Delegar ao {name}"))
482        .icon(IconName::Sparkles)
483        .style(ButtonStyle::Secondary)
484        .disabled(busy || !installed || !has_open)
485        .on_click(move || {
486            let mut h = lock(&handing);
487            h.shown = false;
488            let written = sample::scenario_of(&graph.get(), "Cenário da janela").and_then(|scenario| {
489                let file = std::env::temp_dir()
490                    .join(format!("raven-test-studio-delegado-{}.raventest", std::process::id()));
491                std::fs::write(&file, scenario.write()).map(|()| file).map_err(|e| e.to_string())
492            });
493            let file = match written {
494                Ok(file) => file,
495                Err(why) => {
496                    h.done = Some(Err(why));
497                    turn.update(|t| *t += 1);
498                    return;
499                }
500            };
501            let model = ravenrs_agent_cli::cheapest_model(agent.agent());
502            h.running =
503                Some(format!("Delegado ao {name} ({})…", model.as_deref().unwrap_or("modelo padrão")));
504            h.done = None;
505            let stop = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
506            h.stop = stop.clone();
507            drop(h);
508            let (shared, agent, waker) = (handing.clone(), agent.clone(), waker.clone());
509            std::thread::spawn(move || {
510                let asked = file.to_string_lossy().into_owned();
511                let result = delegate::delegate_until(&agent, &asked, model, &|| {
512                    stop.load(std::sync::atomic::Ordering::SeqCst)
513                });
514                let _ = std::fs::remove_file(&file);
515                let mut h = lock(&shared);
516                h.running = None;
517                h.done = Some(result);
518                drop(h);
519                waker.wake();
520            });
521            turn.update(|t| *t += 1);
522        })
523        .test_id("delegate")
524}

526/// The log of the run: a line that sums it up, whether it came from the cache, the agent's
527/// card when the scenario was delegated, then one line per block that ran.
528fn run_log(outcomes: &[Outcome], total: usize, from_cache: bool, delegation: &DelegationView) -> impl View {
529    let note = if from_cache {
530        "Do cache: o cenário não mudou desde a última vez que rodou nesta janela"
531    } else {
532        ""
533    };
534    let head =
535        if outcomes.is_empty() { "Ainda não rodou".to_owned() } else { runner::summary(outcomes, total) };
536    let lines: Vec<_> = outcomes
537        .iter()
538        .enumerate()
539        .map(|(i, o)| {
540            let line = match &o.error {
541                None => format!("{}. {}: passou", i + 1, o.title),
542                Some(e) => format!("{}. {}: {e}", i + 1, o.title),
543            };
544            Text::new(line).style(TextStyle::Callout).test_id(format!("log-{}", i + 1))
545        })
546        .collect();
547    VStack::new((
548        Text::new("Registro").style(TextStyle::Caption),
549        Text::new(head).style(TextStyle::Headline).test_id("summary"),
550        Text::new(note).style(TextStyle::Caption).test_id("cache-note"),
551        Text::new(delegation.hint.clone().unwrap_or_default())
552            .style(TextStyle::Caption)
553            .test_id("delegate-hint"),
554        Text::new(match (&delegation.running, &delegation.card) {
555            (Some(running), _) => running.clone(),
556            (None, Some((card, _))) => card.clone(),
557            (None, None) => String::new(),
558        })
559        .style(TextStyle::Callout)
560        .foreground(match &delegation.card {
561            Some((_, false)) if delegation.running.is_none() => Semantic::Danger,
562            _ => Semantic::Ink,
563        })
564        .test_id("delegation"),
565        VStack::new(lines).spacing(4.0),
566    ))
567    .spacing(8.0)
568    .padding(16.0)
569}

588/// What `--run` takes by name. A Studio installed from scratch brings no test scenarios: only
589/// its own health check, "O Studio testa o Studio", and `financas`, which runs the scenarios
590/// Raven Finanças brings in its own `tests/e2e/`. Any other scenario is a `.raventest` file,
591/// kept in the app or in a QA team's folder, and run by path ([`run_named`]).
592pub const SCENARIOS: [(&str, &str); 2] =
593    [("autoteste", "O Studio testa o Studio"), ("financas", "Os cenários que o Raven Finanças traz")];

595/// Runs a scenario that comes with the Studio, with no window, and reports it; `None` when
596/// there is no scenario by that name. With a `cache`, a scenario that did not change on a
597/// build that did not change comes back from it instead of running again.
598pub fn run_bundled(name: &str, cache: Option<&mut cache::RunCache>) -> Option<report::Report> {
599    run_bundled_with(name, cache, &RunOptions::default())
600}
601
602/// [`run_bundled`] as `options` say: filming, and where to keep the films. A run that films
603/// runs again, since the cache keeps no film.
604pub fn run_bundled_with(
605    name: &str,
606    cache: Option<&mut cache::RunCache>,
607    options: &RunOptions,
608) -> Option<report::Report> {
609    run_bundled_kept(name, cache, options).map(|(report, _)| report)
610}
611
612/// [`run_bundled_with`], also giving the window left open at the end, when the run keeps it. A
613/// run with the window runs again, since the person wants to see it.
614pub fn run_bundled_kept(
615    name: &str,
616    cache: Option<&mut cache::RunCache>,
617    options: &RunOptions,
618) -> Option<(report::Report, Option<runner::Inspection>)> {
619    let cache =
620        if options.film == runner::FilmMode::Never && !options.window.headed() { cache } else { None };
621    let (key, title) = SCENARIOS.iter().find(|(key, _)| *key == name)?;
622    let graph = match *key {
623        "autoteste" => sample::self_test(),
624        "financas" => match apps::scenarios(apps::FINANCE_APP) {
625            Some(Ok(scenarios)) => sample::from_scenarios(&scenarios),
626            Some(Err(why)) => {
627                return Some((report::Report::refused_with(title, apps::FINANCE_APP, *why), None));
628            }
629            None => {
630                return Some((
631                    report::Report::refused(
632                        title,
633                        apps::FINANCE_APP,
634                        "o Studio não conhece o Raven Finanças",
635                    ),
636                    None,
637                ));
638            }
639        },
640        _ => return None,
641    };
642    // An app opened apart is built apart: the Studio's build says nothing about it, so its
643    // scenario always runs.
644    let kept = if sample::opens_apart(&graph) {
645        None
646    } else {
647        cache::RunCache::key(&graph, cache::build_key().as_deref())
648    };
649    if let (Some(c), Some(k)) = (cache.as_deref(), kept.as_deref())
650        && let Some(report) = c.get(k)
651    {
652        return Some((report, None));
653    }
654    let (report, inspection) = report::kept(title, graph, run_for(title, options));
655    if let (Some(c), Some(k)) = (cache, kept.as_deref()) {
656        // A cache that cannot be written only means the next run runs again.
657        let _ = c.put(k, &report);
658    }
659    Some((report, inspection))
660}
661
662/// A run of the scenario called `title` as `options` say: filming, the window and its pace.
663fn run_for(title: &str, options: &RunOptions) -> Run {
664    Run::new(open)
665        .filming(options.film, options.folder_of(title))
666        .presenting(options.window, options.control.clone())
667}

669/// Runs what `name` says: a name of [`SCENARIOS`], or the path of a `.raventest` file or of a
670/// folder of them. `Err` says why nothing ran.
671pub fn run_named(name: &str, cache: Option<&mut cache::RunCache>) -> Result<report::Report, String> {
672    run_named_with(name, cache, &RunOptions::default())
673}
674
675/// [`run_named`] as `options` say.
676pub fn run_named_with(
677    name: &str,
678    cache: Option<&mut cache::RunCache>,
679    options: &RunOptions,
680) -> Result<report::Report, String> {
681    run_named_kept(name, cache, options).map(|(report, _)| report)
682}
683
684/// [`run_named_with`], also giving the window left open at the end, when the run keeps it.
685pub fn run_named_kept(
686    name: &str,
687    cache: Option<&mut cache::RunCache>,
688    options: &RunOptions,
689) -> Result<(report::Report, Option<runner::Inspection>), String> {
690    if let Some(kept) = run_bundled_kept(name, cache, options) {
691        return Ok(kept);
692    }
693    let path = std::path::Path::new(name);
694    if path.exists() {
695        return run_files_kept(path, options);
696    }
697    let known: Vec<&str> = SCENARIOS.iter().map(|(key, _)| *key).collect();
698    Err(format!(
699        "não há cenário {name:?}; por nome: {}, ou o caminho de um arquivo .raventest ou de uma pasta deles",
700        known.join(", ")
701    ))
702}

704/// Runs `.raventest` files with no window and reports them: one file, or every file of a folder
705/// (the QA team's own, or an app's `tests/e2e/`), one after the other.

707    run_files_with(path, &RunOptions::default())
708}

710/// [`run_files`] as `options` say. The files are read as `cargo ravenrs e2e` reads them, and run
711/// on the Studio's diagram, which opens the apps compiled into the Studio too.
712pub fn run_files_with(path: &std::path::Path, options: &RunOptions) -> Result<report::Report, String> {
713    run_files_kept(path, options).map(|(report, _)| report)
714}
715
716/// [`run_files_with`], also giving the window left open at the end, when the run keeps it.
717pub fn run_files_kept(
718    path: &std::path::Path,
719    options: &RunOptions,
720) -> Result<(report::Report, Option<runner::Inspection>), String> {
721    use ravenrs_e2e_runner::run::{files_at, read, title_of};
722    let scenarios = read(&files_at(path)?)?;
723    let title = title_of(path, &scenarios);
724    Ok(report::kept(&title, sample::from_scenarios(&scenarios), run_for(&title, options)))
725}

778/// [`installed_app`], keeping `link`'s menu beside the clock as the window's state changes and
779/// acting on what is chosen there.
780pub fn installed_app_in_tray(link: Option<tray::TrayLink>) -> Localized<Studio> {
781    let updater = versao::idle();
782    updater.watch();
783    let mut kept = cache::default_file().map(|file| cache::RunCache::at(&file));
784    let health = run_bundled("autoteste", kept.as_mut());
785    let storage: Rc<dyn Storage> = Rc::new(MemoryStorage::default());
786    let mut studio = studio(storage, FlowGraph::new(), updater, health);
787    studio.tray = link;
788    Localized::new(Labels::pt_br(), studio)
789}
