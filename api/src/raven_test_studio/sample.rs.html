42/// A block of a scenario: `kind` says the action, `line` the target and the text.
43pub fn block(id: &str, kind: &str, title: &str, line: &str) -> FlowNode {
44    let node = FlowNode::new(id, kind, title, 0.0, 0.0).subtitle(line);
45    let node = if kind == "open" { node } else { node.port(FlowPort::input("in")) };
46    node.port(FlowPort::output("out"))
47}
48
49fn at(node: FlowNode, x: f32, y: f32) -> FlowNode {
50    FlowNode { x, y, ..node }
51}
52
53/// Joins block `from` to block `to`.
54pub fn link(id: &str, from: &str, to: &str) -> FlowEdge {
55    FlowEdge::new(id, FlowEndpoint::new(from, "out"), FlowEndpoint::new(to, "in"))
56}

58/// The blocks joined one after the other, four to a row.
59fn chain(blocks: Vec<FlowNode>) -> FlowGraph {
60    let ids: Vec<String> = blocks.iter().map(|b| b.id.clone()).collect();
61    let mut graph = FlowGraph::new();
62    for (i, node) in blocks.into_iter().enumerate() {
63        graph = graph.node(at(node, (i % 4) as f32 * 288.0, (i / 4) as f32 * 144.0));
64    }
65    for (i, pair) in ids.windows(2).enumerate() {
66        graph = graph.edge(link(&format!("e{}", i + 1), &pair[0], &pair[1]));
67    }
68    graph
69}

94/// Scenarios read from `.raventest` files, one after the other in a single flow: each starts
95/// with its Open block, so a run opens each app anew.
96pub fn from_scenarios(scenarios: &[ravenrs_e2e_wire::Scenario]) -> FlowGraph {
97    let blocks = scenarios
98        .iter()
99        .enumerate()
100        .flat_map(|(s, scenario)| {
101            scenario.blocks.iter().enumerate().map(move |(b, block)| {
102                let id = if scenarios.len() == 1 {
103                    format!("b{}", b + 1)
104                } else {
105                    format!("s{}b{}", s + 1, b + 1)
106                };
107                self::block(&id, block.kind.word(), &block.title, &block.line())
108            })
109        })
110        .collect();
111    chain(blocks)
112}

114/// The scenario a flow stands for, as a `.raventest` file writes it: its blocks in the order
115/// they run, read the way the runner reads them. `Err` says which block cannot be written.
116pub fn scenario_of(graph: &FlowGraph, name: &str) -> Result<ravenrs_e2e_wire::Scenario, String> {
117    use ravenrs_e2e_wire::{Block, Kind};
118    let mut blocks = Vec::new();
119    for id in crate::runner::path(graph) {
120        let Some(node) = graph.nodes.iter().find(|n| n.id == id) else { continue };
121        let kind = Kind::read(&node.kind)
122            .ok_or_else(|| format!("{}: bloco desconhecido {}", node.title, node.kind))?;
123        let line = node.subtitle.as_deref().unwrap_or_default().trim();
124        let (target, text) = match line.split_once('=') {
125            Some((aim, text)) if kind != Kind::Open => (aim.trim().to_owned(), Some(text.trim().to_owned())),
126            _ => (line.to_owned(), None),
127        };
128        blocks.push(Block { kind, target, text, title: node.title.clone() });
129    }
130    if blocks.first().map(|b| b.kind) != Some(Kind::Open) {
131        return Err("o cenário começa por um bloco Início, que abre o app".to_owned());
132    }
133    Ok(ravenrs_e2e_wire::Scenario { name: name.to_owned(), blocks })
134}

136/// Whether a scenario opens an app apart, one not compiled into the Studio, whose build the
137/// Studio's cache knows nothing about.
138pub fn opens_apart(graph: &FlowGraph) -> bool {
139    graph.nodes.iter().any(|n| {
140        let name = n.subtitle.as_deref().unwrap_or_default();
141        n.kind == "open"
142            && (crate::apps::learned(name)
143                || !matches!(
144                    n.subtitle.as_deref(),
145                    Some(EXPENSE_APP | STUDIO_APP | BROKEN_STUDIO_APP) | None
146                ))
147    })
148}
