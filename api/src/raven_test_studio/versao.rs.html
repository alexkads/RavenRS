1//! Who the Studio is to the installer and the updater of `ravenrs-app-update`, the same that

4use ravenrs_app_update::{InstalledApp, Updater};
5
6/// Raven Test Studio, as installed by compiling the main branch.
7pub static APP: InstalledApp = InstalledApp {
8    name: "Raven Test Studio",
9    package: "raven-test-studio",
10    home_dir: ".raven-test-studio",
11    bundle_id: "dev.ravenrs.raven-test-studio",
12    icons: "tools/raven-test-studio/assets",
13    mac_category: "public.app-category.developer-tools",
14    linux_categories: "Development;",
15    source: "https://github.com/alexkads/RavenRS/archive/refs/heads/master.tar.gz",
16    env_prefix: "RTS",
17    installed_env: "RAVEN_TEST_STUDIO_INSTALLED",
18    news_urls: &[
19        "https://raw.githubusercontent.com/alexkads/RavenRS/master/tools/raven-test-studio/novidades.json",
20    ],
21    history_urls: &[
22        "https://raw.githubusercontent.com/alexkads/RavenRS/master/tools/raven-test-studio/historico-de-novidades.json",
23    ],
24    installer_url: "https://raw.githubusercontent.com/alexkads/RavenRS/master/tools/raven-test-studio/install.cmd",
25    issues_url: "https://github.com/alexkads/RavenRS/issues",
26    version: env!("CARGO_PKG_VERSION"),
27    // The installer sets it when it compiles.
28    installed: option_env!("RAVEN_TEST_STUDIO_INSTALLED").is_some(),
29    history: include_str!("../historico-de-novidades.json"),
30};
31
32/// The Studio's version, the one its news speak of.
33pub const VERSION: &str = APP.version;
34
35/// An updater that does nothing until asked: the window calls `watch` on it once it opens.
36pub fn idle() -> Updater {
37    Updater::idle(&APP)
38}
