1//! The apps a scenario opens apart, from `ravenrs-e2e-runner`, and Raven Finanças, which the
2//! Studio knows by the declaration in its own `Cargo.toml`, read when the Studio is built, so
3//! the installed Studio opens the installed app with no checkout.
4
5use std::path::Path;
6
7pub use ravenrs_e2e_runner::app::*;
8
9use ravenrs_e2e_runner::session::{Failure, Session};
10
11/// The name Raven Finanças declares, which `--run financas` opens.
12pub const FINANCE_APP: &str = "Raven Finanças";
13
14/// Raven Finanças' own declaration, read when the Studio is built.
15const FINANCE_MANIFEST: &str = include_str!("../../../examples/family-finance/Cargo.toml");
16
17/// Makes Raven Finanças known, after any app learned from a project, with its code when the
18/// Studio runs inside a RavenRS checkout.
19fn know_finance() {
20    if let Some(app) = App::declared_in(FINANCE_MANIFEST) {
21        know(App { source: checkout_with(&app.package), ..app });
22    }
23}
24
25/// The app named `name`: one learned from a project, or Raven Finanças.
26pub fn known(name: &str) -> Option<App> {
27    know_finance();
28    ravenrs_e2e_runner::app::known(name)
29}
30
31/// The scenarios the app named `name` brings, read from its hello.
32pub fn scenarios(name: &str) -> Option<Result<Vec<ravenrs_e2e_wire::Scenario>, Box<Failure>>> {
33    know_finance();
34    ravenrs_e2e_runner::app::scenarios(name)
35}
36
37/// Opens the app named `name` apart, if it is one the Studio knows.
38pub fn open(name: &str) -> Option<Result<Session, Box<Failure>>> {
39    know_finance();
40    ravenrs_e2e_runner::app::open(name)
41}
42
43/// Opens Raven Finanças from `binary`, as its variable does.
44pub fn open_finance_from(binary: &Path) -> Result<Session, Box<Failure>> {
45    open_from(binary)
46}
