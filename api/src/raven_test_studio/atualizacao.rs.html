1//! Whether there is a newer Studio, for whoever asks with no window: a person at the command
2//! line, or the pane beside a Claude conversation. The answer is the one the window's status bar
3//! gives, from the same rule of `ravenrs-app-update`.
4
5use ravenrs_app_update::{Decision, InstallKind, News, check_now};
6use serde_json::{Value, json};
7
8use crate::versao::APP;
9
10/// What a check found: the version running, the latest one, and what to do about it.
11#[derive(Clone, Debug)]
12pub struct Answer {
13    /// The version running now.
14    pub installed: &'static str,
15    /// The latest version the repository announces, when the network answered.
16    pub latest: Option<News>,
17    /// The latest one is newer than the one running.
18    pub newer: bool,
19    /// This copy was installed by the installer, which can update it; a checkout updates with
20    /// `git pull`.
21    pub updates_itself: bool,
22    /// Why the network gave no answer.
23    pub error: Option<String>,
24}
25
26impl Answer {
27    /// What `check_now` said, as an answer.
28    pub fn of(checked: Result<(News, Decision), String>, kind: InstallKind) -> Self {
29        let updates_itself = kind == InstallKind::Installed;
30        match checked {
31            Ok((news, decision)) => Self {
32                installed: APP.version,
33                newer: matches!(decision, Decision::Announce { .. }),
34                latest: Some(news),
35                updates_itself,
36                error: None,
37            },
38            Err(reason) => Self {
39                installed: APP.version,
40                latest: None,
41                newer: false,
42                updates_itself,
43                error: Some(reason),
44            },
45        }
46    }
47
48    /// For a tool, such as the pane.
49    pub fn json(&self) -> Value {
50        json!({
51            "installed": self.installed,
52            "latest": self.latest.as_ref().map(|n| n.versao.clone()),
53            "newer": self.newer,
54            "updates_itself": self.updates_itself,
55            "news": self.latest.as_ref().filter(|_| self.newer).map(|n| json!({
56                "title": n.titulo,
57                "items": n.novidades,
58                "why": n.por_que_atualizar,
59                "important": n.importante,
60                "issues": n.issues,
61            })),
62            "error": self.error,
63        })
64    }
65
66    /// For a person.
67    pub fn text(&self) -> String {
68        if let Some(reason) = &self.error {
69            return format!(
70                "Raven Test Studio {}: não deu para saber se há versão nova: {reason}",
71                self.installed
72            );
73        }
74        let Some(news) = self.latest.as_ref().filter(|_| self.newer) else {
75            return format!("Raven Test Studio {}: é a versão mais nova", self.installed);
76        };
77        let mut out = format!("Raven Test Studio {} → {}: {}\n", self.installed, news.versao, news.titulo);
78        for item in &news.novidades {
79            out.push_str(&format!("  · {item}\n"));
80        }
81        out.push_str(if self.updates_itself {
82            "Atualize com: raven-test-studio --atualizar"
83        } else {
84            "Esta cópia é de um checkout do RavenRS: atualize com git pull"
85        });
86        out
87    }
88}
89
90/// Asks the network now.
91pub fn ask() -> Answer {
92    Answer::of(check_now(&APP), APP.install_kind())
93}
