7/// An HTTP method used by the adapters.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum Metodo {
10    /// GET.
11    Get,
12    /// POST.
13    Post,
14    /// PATCH.
15    Patch,
16}
17
18/// A request, ready to go.
19#[derive(Clone, PartialEq, Eq)]
20pub struct RequisicaoHttp {
21    /// The method.
22    pub metodo: Metodo,
23    /// The full address.
24    pub url: String,
25    /// The headers, in order.
26    pub cabecalhos: Vec<(String, String)>,
27    /// The body, when there is one.
28    pub corpo: Option<Vec<u8>>,
29}
30
31impl RequisicaoHttp {
32    /// The value of a header, ignoring case in its name.
33    #[must_use]
34    pub fn cabecalho(&self, nome: &str) -> Option<&str> {
35        self.cabecalhos.iter().find(|(k, _)| k.eq_ignore_ascii_case(nome)).map(|(_, v)| v.as_str())
36    }
37
38    /// The body as JSON, for tests and handlers.
39    #[must_use]
40    pub fn json(&self) -> Option<serde_json::Value> {
41        self.corpo.as_deref().and_then(|corpo| serde_json::from_slice(corpo).ok())
42    }
43}
44
45/// `Debug` hides `Authorization`, which holds a secret.
46impl fmt::Debug for RequisicaoHttp {
47    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
48        let nomes: Vec<&str> = self.cabecalhos.iter().map(|(k, _)| k.as_str()).collect();
49        f.debug_struct("RequisicaoHttp")
50            .field("metodo", &self.metodo)
51            .field("url", &self.url)
52            .field("cabecalhos", &nomes)
53            .field("corpo_bytes", &self.corpo.as_ref().map(Vec::len))
54            .finish()
55    }
56}

58/// A response.
59#[derive(Clone, Debug, PartialEq, Eq)]
60pub struct RespostaHttp {
61    /// The status code.
62    pub status: u16,
63    /// The body.
64    pub corpo: Vec<u8>,
65}
66
67/// Why a request got no HTTP response. The caller cannot know if it arrived.
68#[derive(Clone, Debug, PartialEq, Eq)]
69#[non_exhaustive]
70pub enum ErroTransporte {
71    /// The time limit ran out.
72    Tempo,
73    /// The network failed.
74    Rede(String),
75    /// This transport refuses this address.
76    Recusado(String),
77}
78
79impl fmt::Display for ErroTransporte {
80    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
81        match self {
82            Self::Tempo => f.write_str("o tempo acabou"),
83            Self::Rede(motivo) => write!(f, "falha de rede: {motivo}"),
84            Self::Recusado(motivo) => write!(f, "endereço recusado: {motivo}"),
85        }
86    }
87}
88
89impl std::error::Error for ErroTransporte {}
90
91impl From<ErroTransporte> for ErroProvedor {
92    fn from(erro: ErroTransporte) -> Self {
93        match erro {
94            ErroTransporte::Tempo => Self::Tempo,
95            outra => Self::Indisponivel(outra.to_string()),
96        }
97    }
98}

100/// Sends a request and returns any HTTP response, also an error status.
101pub trait TransporteHttp {
102    /// Sends it.
103    fn enviar(&self, requisicao: &RequisicaoHttp) -> Result<RespostaHttp, ErroTransporte>;
104}
105
106impl<T: TransporteHttp + ?Sized> TransporteHttp for &T {
107    fn enviar(&self, requisicao: &RequisicaoHttp) -> Result<RespostaHttp, ErroTransporte> {
108        (**self).enviar(requisicao)
109    }
110}

115#[cfg(all(feature = "http", not(target_arch = "wasm32")))]
116mod nativo {
117    use std::io::Read;
118    use std::time::Duration;
119
120    use ureq::Agent;
121    use ureq::config::Config;
122
123    use super::{ErroTransporte, Metodo, RequisicaoHttp, RespostaHttp, TransporteHttp};
124
125    /// The largest response read, far above any documented answer.
126    const LIMITE_RESPOSTA: u64 = 4 * 1024 * 1024;
127
128    /// The real network through `ureq`. It follows no redirect, and only local addresses may use
129    /// plain HTTP. It has never reached a real provider.
130    #[derive(Debug)]
131    pub struct TransporteUreq {
132        agente: Agent,
133    }
134
135    impl TransporteUreq {
136        /// A transport with this time limit per request.
137        #[must_use]
138        pub fn new(limite: Duration) -> Self {
139            let config = Config::builder()
140                .timeout_global(Some(limite))
141                .max_redirects(0)
142                .http_status_as_error(false)
143                .build();
144            Self { agente: Agent::new_with_config(config) }
145        }
146    }
147
148    fn permitido(url: &str) -> bool {
149        url.starts_with("https://")
150            || url.starts_with("http://localhost:")
151            || url.starts_with("http://127.0.0.1:")
152    }
153
154    fn erro_ureq(erro: &ureq::Error) -> ErroTransporte {
155        if matches!(erro, ureq::Error::Timeout(_)) {
156            ErroTransporte::Tempo
157        } else {
158            ErroTransporte::Rede(erro.to_string())
159        }
160    }
161
162    impl TransporteHttp for TransporteUreq {
163        fn enviar(&self, requisicao: &RequisicaoHttp) -> Result<RespostaHttp, ErroTransporte> {
164            if !permitido(&requisicao.url) {
165                return Err(ErroTransporte::Recusado("só https, ou http local".into()));
166            }
167            let corpo = requisicao.corpo.as_deref().unwrap_or_default();
168            let cabecalhos = requisicao.cabecalhos.iter();
169            let resposta = match requisicao.metodo {
170                Metodo::Get => {
171                    let mut pedido = self.agente.get(&requisicao.url);
172                    for (nome, valor) in cabecalhos {
173                        pedido = pedido.header(nome, valor);
174                    }
175                    pedido.call()
176                }
177                Metodo::Post => {
178                    let mut pedido = self.agente.post(&requisicao.url);
179                    for (nome, valor) in cabecalhos {
180                        pedido = pedido.header(nome, valor);
181                    }
182                    pedido.send(corpo)
183                }
184                Metodo::Patch => {
185                    let mut pedido = self.agente.patch(&requisicao.url);
186                    for (nome, valor) in cabecalhos {
187                        pedido = pedido.header(nome, valor);
188                    }
189                    pedido.send(corpo)
190                }
191            }
192            .map_err(|erro| erro_ureq(&erro))?;
193            let status = resposta.status().as_u16();
194            let mut bytes = Vec::new();
195            resposta
196                .into_body()
197                .into_reader()
198                .take(LIMITE_RESPOSTA)
199                .read_to_end(&mut bytes)
200                .map_err(|erro| ErroTransporte::Rede(erro.to_string()))?;
201            Ok(RespostaHttp { status, corpo: bytes })
202        }
203    }

216                corpo: None,
217            };
