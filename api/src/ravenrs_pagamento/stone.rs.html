1//! Stone: the fiscal data of Connect 2.0, and AutoTEF over local HTTP.
2//!
3//! Stone has more than one product and they work differently:
4//!
5//! - **Connect 2.0**: an order in the Pagar.me API goes to an Android machine, not a pinpad. It has
6//!   no sandbox, a webhook is mandatory, and there is no refund route. Only [`dados_de_charge_paid`]
7//!   exists here, for the server that receives the webhook.
8//! - **TEF Stone**: a native library, reached through [`crate::auxiliar`].
9//! - **AutoTEF**: [`AutoTef`]. **NOT CONFIRMED:** its documentation answered 403. Everything here
10//!   comes from third-party code that integrates with it, so even the existence of the product
11//!   today is unknown. Field names and units may differ. Nothing has run against the real one.
12

23/// What a Connect 2.0 `charge.paid` webhook gives the fiscal document.
24///
25/// The webhook has no CNPJ of the acquirer: it comes from the table, with an alert. The
26/// documentation's table says `metadata.schemeName` and its example says `metadata.scheme_name`;
27/// the example is what is read, and the other is accepted too.
28#[must_use]
29pub fn dados_de_charge_paid(webhook: &Value) -> DadosFiscaisPagamento {
30    let metadata = &webhook["data"]["metadata"];
31    let mut dados = DadosFiscaisPagamento::integrado();
32    dados.com_c_aut(
33        metadata["authorization_code"].as_str().or_else(|| metadata["authorizationCode"].as_str()),
34        "o webhook charge.paid veio sem authorization_code",
35    );
36    dados.com_cnpj_da_rede(Some("STONE"));
37    dados.com_bandeira(metadata["scheme_name"].as_str().or_else(|| metadata["schemeName"].as_str()));
38    dados.com_terminal(
39        metadata["terminal_serial_number"].as_str().or_else(|| metadata["terminalSerialNumber"].as_str()),
40    );
41    dados
42}

47/// Stone AutoTEF over local HTTP. **Not confirmed with Stone**; see the module text.
48///
49/// A payment call blocks until the customer finishes, up to the limit of the transport.
50#[derive(Debug)]
51pub struct AutoTef<T> {
52    transporte: T,
53    base: String,
54    pix: Option<(IdPagamento, String)>,
55}

57impl<T: TransporteHttp> AutoTef<T> {
58    /// An adapter at the default local address.
59    #[must_use]
60    pub fn new(transporte: T) -> Self {
61        Self { transporte, base: URL_AUTOTEF.to_owned(), pix: None }
62    }

64    /// `POST api/Activate`: ties the service to a Stone code and a pinpad port.
65    pub fn ativar(&self, codigo_stone: &str, porta_pinpad: &str, parceiro: &str) -> Result<(), ErroProvedor> {
66        self.pedir(
67            "api/Activate",
68            &json!({ "StoneCode": codigo_stone, "ConnectionName": porta_pinpad, "PartnerName": parceiro }),
69        )
70        .map(|_| ())
71    }

73    fn pedir(&self, caminho: &str, corpo: &Value) -> Result<Value, ErroProvedor> {
74        let requisicao = RequisicaoHttp {
75            metodo: Metodo::Post,
76            url: format!("{}{caminho}", self.base),
77            cabecalhos: vec![("Content-Type".to_owned(), "application/json".to_owned())],
78            corpo: Some(corpo.to_string().into_bytes()),
79        };
80        let resposta = self.transporte.enviar(&requisicao)?;
81        let json: Value = serde_json::from_slice(&resposta.corpo).unwrap_or(Value::Null);
82        match resposta.status {
83            200..=299 => Ok(json),
84            408 | 429 | 500..=599 => Err(ErroProvedor::Indisponivel(format!("HTTP {}", resposta.status))),
85            status => Err(ErroProvedor::Recusado {
86                codigo: status.to_string(),
87                motivo: json["message"].as_str().map_or_else(|| format!("HTTP {status}"), str::to_owned),
88            }),
89        }
90    }
91
92    fn aprovacao(&self, id: &IdPagamento, valor_centavos: u64, resposta: &Value) -> Aprovacao {
93        let recibo = &resposta["receipt"];
94        let nsu = recibo["acquirerTransactionKey"].as_str().or_else(|| resposta["transactionId"].as_str());
95        let mut fiscais = DadosFiscaisPagamento::integrado();
96        fiscais.com_c_aut(recibo["authorisationCode"].as_str(), "a resposta veio sem authorisationCode");
97        fiscais.com_cnpj_da_rede(Some("STONE"));
98        fiscais.com_bandeira(recibo["brandName"].as_str());
99        let linhas = |via: &Value| -> Vec<String> {
100            via.as_str().map_or_else(Vec::new, |texto| {
101                texto.replace("\\r\\n", "\n").lines().map(str::to_owned).collect()
102            })
103        };
104        Aprovacao {
105            referencia: ReferenciaTransacao {
106                id: Some(id.clone()),
107                rede: Some("STONE".to_owned()),
108                nsu: nsu.map(str::to_owned),
109                controle: recibo["initiatorTransactionKey"].as_str().map(str::to_owned),
110                valor_centavos,
111                extras: std::collections::BTreeMap::new(),
112            },
113            dados_fiscais: fiscais,
114            vias: Vias {
115                cliente: linhas(&recibo["clientVia"]),
116                estabelecimento: linhas(&recibo["merchantVia"]),
117                unica: Vec::new(),
118                impressas_no_terminal: false,
119            },
120            mensagem_operador: None,
121            tipo_efetivo: None,
122            avisos: vec![AvisoAprovacao::Outro("AutoTEF não confirmado com a Stone".to_owned())],
123        }
124    }
125}

127fn valor_decimal(centavos: u64) -> Value {
128    // The third-party code sends the amount in reais as a number, so this is the one place money is
129    // a float. Two decimals survive this at any amount a checkout sees.
130    json!(centavos as f64 / 100.0)
131}

133impl<T: TransporteHttp> ProvedorPagamento for AutoTef<T> {
134    fn id(&self) -> ProvedorId {
135        ProvedorId::StoneAutoTef
136    }
137
138    fn capacidades(&self) -> Capacidades {
139        Capacidades::STONE_AUTOTEF
140    }
141
142    fn iniciar(&mut self, pedido: &PedidoPagamento) -> Result<Passo, ErroProvedor> {
143        if pedido.tipo == TipoPagamento::Pix {
144            let resposta = self.pedir(
145                "api/Pix/Pay",
146                &json!({ "amount": valor_decimal(pedido.valor_centavos), "expiresIn": 600 }),
147            )?;
148            let transacao = resposta["transactionId"]
149                .as_str()
150                .ok_or_else(|| ErroProvedor::Protocolo("Pix sem transactionId".into()))?;
151            self.pix = Some((pedido.id.clone(), transacao.to_owned()));
152            return Ok(Passo::Perguntar(Pergunta::QrCode {
153                conteudo: resposta["qrCodeContent"].as_str().unwrap_or_default().to_owned(),
154                mensagem: Some("Pague com Pix".to_owned()),
155            }));
156        }
157        let (conta, parcelas, tipo_parcela) = match pedido.tipo {
158            TipoPagamento::Debito => ("Debit", 1, 0),
159            TipoPagamento::Credito { parcelas } if parcelas > 1 => ("Credit", parcelas, 3),
160            TipoPagamento::Credito { .. } => ("Credit", 1, 1),
161            _ => ("Undefined", 1, 1),
162        };
163        let resposta = self.pedir(
164            "api/Pay",
165            &json!({
166                "amount": valor_decimal(pedido.valor_centavos),
167                "hasAlcoholicDrink": false,
168                "accountType": conta,
169                "installment": { "number": parcelas, "type": tipo_parcela },
170            }),
171        )?;
172        let aprovado = resposta["receipt"]["acquirerTransactionKey"].as_str().is_some_and(|k| !k.is_empty())
173            || resposta["wasSuccessfull"].as_bool() == Some(true);
174        if aprovado {
175            Ok(Passo::Aprovado(Box::new(self.aprovacao(&pedido.id, pedido.valor_centavos, &resposta))))
176        } else {
177            Ok(Passo::Negado {
178                codigo: "AUTOTEF".to_owned(),
179                motivo: resposta["message"].as_str().unwrap_or("Transação não aprovada").to_owned(),
180            })
181        }
182    }
183
184    fn continuar(&mut self, id: &IdPagamento, entrada: EntradaOperador) -> Result<Passo, ErroProvedor> {
185        let Some((_, transacao)) = self.pix.clone().filter(|(pix, _)| pix == id) else {
186            return Err(ErroProvedor::Protocolo("nada em andamento neste adaptador".into()));
187        };
188        if entrada == EntradaOperador::Cancelar {
189            self.pedir("api/Pix/Cancel", &json!({ "transactionId": transacao }))?;
190            self.pix = None;
191            return Ok(Passo::Cancelado { motivo: "Pix cancelado".to_owned() });
192        }
193        let resposta = self.pedir("api/Pix/Status", &json!({ "transactionId": transacao }))?;
194        if resposta["isExpired"].as_bool() == Some(true) {
195            self.pix = None;
196            return Ok(Passo::Cancelado { motivo: "Pix expirou".to_owned() });
197        }
198        if resposta["statusPaymentPix"].as_i64() == Some(1) {
199            self.pix = None;
200            let mut com_id = resposta;
201            com_id["transactionId"] = json!(transacao);
202            return Ok(Passo::Aprovado(Box::new(self.aprovacao(id, 0, &com_id))));
203        }
204        Ok(Passo::Aguardar { mensagem: Some("Aguardando o Pix".to_owned()), tentar_em_ms: 2_000 })
205    }
206
207    /// The third-party code sends no third leg, so this is never called.
208    fn confirmar(&mut self, _: &ReferenciaTransacao) -> Result<(), ErroProvedor> {
209        Ok(())
210    }
211
212    fn desfazer(&mut self, _: &ReferenciaTransacao) -> Result<(), ErroProvedor> {
213        Err(ErroProvedor::SemSuporte("o AutoTEF, como lido, só tem cancelamento".into()))
214    }
215
216    fn cancelar(&mut self, _: &IdPagamento) -> Result<Cancelamento, ErroProvedor> {
217        Err(ErroProvedor::SemSuporte("o operador não interrompe um pagamento com cartão no AutoTEF".into()))
218    }
219
220    fn estornar(&mut self, referencia: &ReferenciaTransacao) -> Result<(), ErroProvedor> {
221        let nsu =
222            referencia.nsu.as_deref().ok_or_else(|| ErroProvedor::Protocolo("referência sem NSU".into()))?;
223        // The third-party code tells Pix by the length of the key.
224        let corpo = if nsu.len() >= 27 {
225            self.pedir("api/Pix/Cancel", &json!({ "transactionId": nsu }))
226        } else {
227            self.pedir(
228                "api/cancel",
229                &json!({
230                    "acquirerTransactionKey": nsu,
231                    "amount": valor_decimal(referencia.valor_centavos),
232                    "transactionType": "Undefined",
233                }),
234            )
235        };
236        corpo.map(|_| ())
237    }
238
239    /// There is no query in what was read, so a lost answer cannot be recovered here.
240    fn consultar(&mut self, _: &IdPagamento) -> Result<SituacaoProvedor, ErroProvedor> {
241        Ok(SituacaoProvedor::Indeterminada)
242    }
243
244    fn pendencias(&mut self) -> Result<Vec<ReferenciaTransacao>, ErroProvedor> {
245        Ok(Vec::new())
246    }
247}

303    #[test]
304    fn pagamento_com_recibo_aprova_e_mapeia() {
305        let local = Local::default();
306        *local.resposta.borrow_mut() = json!({
307            "receipt": {
308                "acquirerTransactionKey": "NSU1",
309                "initiatorTransactionKey": "INI1",
310                "authorisationCode": "AUT1",
311                "brandName": "VISA",
312                "clientVia": "LINHA 1\\r\\nLINHA 2",
313                "merchantVia": "LOJA"
314            }
315        });
316        let mut autotef = AutoTef::new(&local);
317        let Passo::Aprovado(aprovacao) = autotef.iniciar(&pedido(TipoPagamento::Debito)).expect("paga")
318        else {
319            panic!("devia aprovar");
320        };
321        assert_eq!(aprovacao.dados_fiscais.c_aut.as_deref(), Some("AUT1"));
322        assert_eq!(aprovacao.dados_fiscais.t_band.as_deref(), Some("01"));
323        assert_eq!(aprovacao.vias.via_cliente(), ["LINHA 1", "LINHA 2"]);
324        assert!(
325            aprovacao
326                .avisos
327                .iter()
328                .any(|a| matches!(a, AvisoAprovacao::Outro(t) if t.contains("não confirmado")))
329        );
330        let enviado = &local.pedidos.borrow()[0];
331        assert_eq!(enviado.url, "http://localhost:8000/api/Pay");
332        assert_eq!(enviado.json().expect("json")["accountType"], "Debit");
333    }

344    fn pix_mostra_qr_e_acompanha() {
345        let local = Local::default();
346        *local.resposta.borrow_mut() = json!({ "transactionId": "T1", "qrCodeContent": "PIX" });
347        let mut autotef = AutoTef::new(&local);

349        assert!(matches!(

353        *local.resposta.borrow_mut() = json!({ "statusPaymentPix": 0 });
354        assert!(matches!(autotef.continuar(&id, EntradaOperador::Nada), Ok(Passo::Aguardar { .. })));
355        *local.resposta.borrow_mut() = json!({ "statusPaymentPix": 1 });
356        let Passo::Aprovado(aprovacao) = autotef.continuar(&id, EntradaOperador::Nada).expect("pago") else {
357            panic!("devia aprovar");
358        };
359        assert_eq!(aprovacao.referencia.nsu.as_deref(), Some("T1"));
