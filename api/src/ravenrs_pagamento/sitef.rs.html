1//! The iterative loop of CliSiTef as a pure translation, with no library in sight.
2//!
3//! The real helper process calls `IniciaFuncaoSiTefInterativo` and then
4//! `ContinuaFuncaoSiTefInterativo` while it returns 10000. Each round gives a command, a field type,
5//! limits and a buffer. [`CicloSiTef`] turns one round into what the screen must do, and collects
6//! the fields of the answer. The helper only has to move bytes between this and the library.
7//!
8//! Field and command numbers were checked against the behavior of the LGPL unit `ACBrTEFCliSiTefComum`
9//! and the public Fiserv pages; see `docs/PAGAMENTOS.md`. **Never run against a real CliSiTef or
10//! against SitDemo.** What field 951 holds in practice is not confirmed.
11
12use crate::dominio::{
13    Aprovacao, EntradaOperador, IdPagamento, OpcaoMenu, Passo, Pergunta, ReferenciaTransacao, TipoCampo, Vias,
14};
15use crate::fiscal::{DadosFiscaisPagamento, cnpj_valido};
16use crate::provedor::ErroProvedor;

18/// What `IniciaFuncaoSiTefInterativo` and `ContinuaFuncaoSiTefInterativo` return to go on.
19pub const RETORNO_CONTINUA: i32 = 10000;
20/// Field: the fiscal CNPJ of the acquirer.
21pub const CAMPO_CNPJ_CREDENCIADORA: i32 = 950;
22/// Field: the fiscal brand.
23pub const CAMPO_BANDEIRA_FISCAL: i32 = 951;
24/// Field: the fiscal authorization number.
25pub const CAMPO_AUTORIZACAO_FISCAL: i32 = 952;
26/// Field: the transaction number at SiTef.
27pub const CAMPO_NSU_SITEF: i32 = 133;
28/// Field: the transaction number at the host.
29pub const CAMPO_NSU_HOST: i32 = 134;
30/// Field: the first copy of the receipt.
31pub const CAMPO_VIA_1: i32 = 121;
32/// Field: the second copy of the receipt.
33pub const CAMPO_VIA_2: i32 = 122;
34
35/// One round of the loop, as the library gave it.
36#[derive(Clone, Debug, PartialEq, Eq)]
37pub struct RodadaSiTef {
38    /// `ProximoComando`.
39    pub comando: i32,
40    /// `TipoCampo`.
41    pub tipo_campo: i32,
42    /// `TamanhoMinimo`.
43    pub minimo: u16,
44    /// `TamanhoMaximo`.
45    pub maximo: u16,
46    /// The text in the buffer.
47    pub buffer: String,
48}
49
50/// What to do with a round.
51#[derive(Clone, Debug, PartialEq, Eq)]
52pub enum AcaoSiTef {
53    /// Call the library again at once, with an empty buffer.
54    Prosseguir,
55    /// Show this on the screen and bring the answer to [`CicloSiTef::responder`].
56    Passo(Passo),
57}
58
59/// How an answer goes back to the library.
60#[derive(Clone, Debug, PartialEq, Eq)]
61pub struct RetornoSiTef {
62    /// `Continua`: 0 goes on, 1 goes back, 2 interrupts, -1 ends the collection.
63    pub continua: i32,
64    /// The buffer.
65    pub buffer: String,
66}

68/// Translates rounds and collects the answer of one transaction.
69#[derive(Clone, Debug)]
70pub struct CicloSiTef {
71    id: IdPagamento,
72    valor_centavos: u64,
73    cnpj: Option<String>,
74    bandeira: Option<String>,
75    autorizacao: Option<String>,
76    nsu_sitef: Option<String>,
77    nsu_host: Option<String>,
78    via1: Vec<String>,
79    via2: Vec<String>,
80}
81
82impl CicloSiTef {
83    /// A new transaction.
84    #[must_use]
85    pub fn new(id: IdPagamento, valor_centavos: u64) -> Self {
86        Self {
87            id,
88            valor_centavos,
89            cnpj: None,
90            bandeira: None,
91            autorizacao: None,
92            nsu_sitef: None,
93            nsu_host: None,
94            via1: Vec::new(),
95            via2: Vec::new(),
96        }
97    }
98
99    /// Reads a round. Command 0 only delivers a field, and the messages need no answer.
100    pub fn rodada(&mut self, rodada: &RodadaSiTef) -> AcaoSiTef {
101        let texto = rodada.buffer.trim().to_owned();
102        match rodada.comando {
103            0 => {
104                self.guardar(rodada.tipo_campo, &rodada.buffer);
105                AcaoSiTef::Prosseguir
106            }
107            1..=3 | 5 | 13..=15 => {
108                AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Mensagem { texto, exige_ok: false }))
109            }
110            4 | 11 | 12 | 16 => AcaoSiTef::Prosseguir,
111            20 => AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Confirmacao { texto })),
112            21 => AcaoSiTef::Passo(Passo::Perguntar(ler_menu(&rodada.buffer))),
113            22 => AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Mensagem { texto, exige_ok: true })),
114            23 => AcaoSiTef::Passo(Passo::Aguardar { mensagem: None, tentar_em_ms: 200 }),
115            30 | 31 | 34 | 35 | 41 => AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Campo {
116                rotulo: texto,
117                tipo: match rodada.comando {
118                    34 => TipoCampo::Valor,
119                    35 => TipoCampo::CodigoBarras,
120                    41 => TipoCampo::Mascarado,
121                    _ => TipoCampo::Texto,
122                },
123                minimo: rodada.minimo,
124                maximo: rodada.maximo,
125            })),
126            50 => AcaoSiTef::Passo(Passo::Perguntar(Pergunta::QrCode { conteudo: texto, mensagem: None })),
127            51 | 52 => AcaoSiTef::Prosseguir,
128            // A command with no known meaning goes on rather than freezing the checkout on a guess.
129            _ => AcaoSiTef::Prosseguir,
130        }
131    }
132
133    /// Turns what the operator did into the next call of the loop.
134    #[must_use]
135    pub fn responder(entrada: &EntradaOperador) -> RetornoSiTef {
136        let (continua, buffer) = match entrada {
137            EntradaOperador::Nada | EntradaOperador::Ok => (0, String::new()),
138            EntradaOperador::Sim => (0, "0".to_owned()),
139            EntradaOperador::Nao => (0, "1".to_owned()),
140            EntradaOperador::Opcao(indice) => (0, (indice + 1).to_string()),
141            EntradaOperador::Texto(texto) => (0, texto.clone()),
142            EntradaOperador::Voltar => (1, String::new()),
143            EntradaOperador::Cancelar => (-1, String::new()),
144        };
145        RetornoSiTef { continua, buffer }
146    }
147
148    /// The end of the loop, by the code the library returned.
149    pub fn fim(&self, retorno: i32) -> Result<Passo, ErroProvedor> {
150        match retorno {
151            0 => Ok(Passo::Aprovado(Box::new(self.aprovacao()))),
152            -2 => Ok(Passo::Cancelado { motivo: "Cancelado pelo operador".to_owned() }),
153            // Positive codes come from the network or the issuer, negative ones from the library.
154            n if n > 0 => {
155                Ok(Passo::Negado { codigo: n.to_string(), motivo: format!("Não autorizada ({n})") })
156            }
157            n => Err(ErroProvedor::Indisponivel(format!("CliSiTef retornou {n}"))),
158        }
159    }
160
161    fn guardar(&mut self, campo: i32, valor: &str) {
162        match campo {
163            CAMPO_CNPJ_CREDENCIADORA => self.cnpj = Some(valor.trim().to_owned()),
164            CAMPO_BANDEIRA_FISCAL => self.bandeira = Some(valor.trim().to_owned()),
165            CAMPO_AUTORIZACAO_FISCAL => self.autorizacao = Some(valor.trim().to_owned()),
166            CAMPO_NSU_SITEF => self.nsu_sitef = Some(valor.trim().to_owned()),
167            CAMPO_NSU_HOST => self.nsu_host = Some(valor.trim().to_owned()),
168            CAMPO_VIA_1 => self.via1 = linhas(valor),
169            CAMPO_VIA_2 => self.via2 = linhas(valor),
170            _ => {}
171        }
172    }
173
174    fn aprovacao(&self) -> Aprovacao {
175        let mut fiscais = DadosFiscaisPagamento::integrado();
176        fiscais.com_c_aut(self.autorizacao.as_deref(), "o CliSiTef não entregou o campo 952");
177        match self.cnpj.as_deref() {
178            Some(cnpj) if cnpj_valido(cnpj) => fiscais.cnpj = Some(cnpj.to_owned()),
179            Some(cnpj) => fiscais.com_cnpj_do_provedor(cnpj),
180            None => fiscais.com_cnpj_da_rede(None),
181        }
182        // Field 951 may be a code or a name; which one it is in practice was not confirmed.
183        match self.bandeira.as_deref() {
184            Some(codigo) if codigo.len() == 2 && codigo.bytes().all(|b| b.is_ascii_digit()) => {
185                fiscais.t_band = Some(codigo.to_owned());
186            }
187            outra => fiscais.com_bandeira(outra),
188        }
189        Aprovacao {
190            referencia: ReferenciaTransacao {
191                id: Some(self.id.clone()),
192                rede: None,
193                nsu: self.nsu_host.clone().or_else(|| self.nsu_sitef.clone()),
194                controle: self.nsu_sitef.clone(),
195                valor_centavos: self.valor_centavos,
196                extras: std::collections::BTreeMap::new(),
197            },
198            dados_fiscais: fiscais,
199            // The source only says first and second copy; which one is the customer's is not confirmed.
200            vias: Vias {
201                cliente: self.via1.clone(),
202                estabelecimento: self.via2.clone(),
203                unica: Vec::new(),
204                impressas_no_terminal: false,
205            },
206            mensagem_operador: None,
207            tipo_efetivo: None,
208            avisos: Vec::new(),
209        }
210    }
211}

213/// The options come as `1:Debito;2:Credito;` after an optional title line.
214fn ler_menu(buffer: &str) -> Pergunta {
215    let opcoes = buffer
216        .split(';')
217        .filter_map(|parte| {
218            let (valor, texto) = parte.split_once(':')?;
219            Some(OpcaoMenu { valor: valor.trim().to_owned(), texto: texto.trim().to_owned() })
220        })
221        .collect();
222    Pergunta::Menu { titulo: String::new(), opcoes }
223}
224
225fn linhas(texto: &str) -> Vec<String> {
226    texto.lines().map(str::to_owned).collect()
227}

242    #[test]
243    fn cada_familia_de_comando_vira_o_passo_certo() {
244        let mut ciclo = novo();
245        assert_eq!(ciclo.rodada(&rodada(0, 133, "123")), AcaoSiTef::Prosseguir);
246        assert!(matches!(
247            ciclo.rodada(&rodada(1, 0, "INSIRA O CARTAO")),
248            AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Mensagem { exige_ok: false, .. }))
249        ));
250        assert!(matches!(
251            ciclo.rodada(&rodada(20, 0, "CONFIRMA?")),
252            AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Confirmacao { .. }))
253        ));
254        assert!(matches!(
255            ciclo.rodada(&rodada(22, 0, "TECLE ENTER")),
256            AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Mensagem { exige_ok: true, .. }))
257        ));
258        assert!(matches!(ciclo.rodada(&rodada(23, 0, "")), AcaoSiTef::Passo(Passo::Aguardar { .. })));
259        assert!(matches!(
260            ciclo.rodada(&rodada(34, 0, "VALOR")),
261            AcaoSiTef::Passo(Passo::Perguntar(Pergunta::Campo { tipo: TipoCampo::Valor, .. }))
262        ));
263        assert!(matches!(
264            ciclo.rodada(&rodada(50, 0, "QR")),
265            AcaoSiTef::Passo(Passo::Perguntar(Pergunta::QrCode { .. }))
266        ));
267        assert_eq!(ciclo.rodada(&rodada(9999, 0, "")), AcaoSiTef::Prosseguir);
268    }

289    #[test]
290    fn campos_viram_aprovacao_com_dados_fiscais() {
291        let mut ciclo = novo();
292        for (campo, valor) in [
293            (950, "16501555000157"),
294            (951, "02"),
295            (952, "AUT9"),
296            (133, "000123"),
297            (134, "HOST9"),
298            (121, "VIA1\nL2"),
299            (122, "VIA2"),
300        ] {
301            ciclo.rodada(&rodada(0, campo, valor));
302        }
303        let Ok(Passo::Aprovado(aprovacao)) = ciclo.fim(0) else { panic!("devia aprovar") };
304        assert_eq!(aprovacao.dados_fiscais.c_aut.as_deref(), Some("AUT9"));
305        assert_eq!(aprovacao.dados_fiscais.t_band.as_deref(), Some("02"));
306        assert_eq!(aprovacao.dados_fiscais.cnpj.as_deref(), Some("16501555000157"));
307        assert!(aprovacao.dados_fiscais.alertas.is_empty());
308        assert_eq!(aprovacao.referencia.nsu.as_deref(), Some("HOST9"));
309        assert_eq!(aprovacao.vias.cliente, ["VIA1", "L2"]);
310    }
