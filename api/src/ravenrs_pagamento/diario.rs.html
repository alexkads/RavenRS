1//! The journal: what was about to happen is written before it happens.
2//!
3//! It keeps ids, states, references, fiscal data and receipts. It never keeps what the operator
4//! typed, since that may be a tax id, and no provider adapter puts a card number in a reference.
5

14/// An action on the provider that is about to be attempted.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
16#[serde(rename_all = "snake_case")]
17pub enum Acao {
18    /// Start the payment.
19    Iniciar,
20    /// Confirm it.
21    Confirmar,
22    /// Undo it.
23    Desfazer,
24    /// Cancel it while in progress.
25    Cancelar,
26    /// Refund it.
27    Estornar,
28}
29
30/// What each line of the journal says.
31#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
32#[serde(rename_all = "snake_case")]
33pub enum Conteudo {
34    /// A new payment.
35    Aberto {
36        /// The request.
37        pedido: PedidoPagamento,
38        /// Who will process it.
39        provedor: ProvedorId,
40    },
41    /// The provider is about to be called for this.
42    Intencao {
43        /// The action.
44        acao: Acao,
45    },
46    /// What the adapter needs to find this payment after a restart.
47    Rastro {
48        /// Opaque to all but the adapter.
49        valor: String,
50    },
51    /// The state changes now.
52    Transicao {
53        /// What moved it.
54        evento: Evento,
55        /// The approval, when this is one.
56        aprovacao: Option<Box<Aprovacao>>,
57        /// The reason, when there is one for the operator.
58        motivo: Option<String>,
59    },
60}

62/// A line of the journal.
63#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
64pub struct Lancamento {
65    /// The position in the journal, from 1.
66    pub seq: u64,
67    /// The payment.
68    pub id: IdPagamento,
69    /// What it says.
70    pub conteudo: Conteudo,
71}

73/// Why the journal failed. Without a journal no provider is called.
74#[derive(Clone, Debug, PartialEq, Eq)]
75#[non_exhaustive]
76pub enum ErroDiario {
77    /// It could not be written or read.
78    Io(String),
79    /// A line in the middle does not parse: this is damage, not a power loss.
80    Corrompido {
81        /// The line number, from 1.
82        linha: usize,
83    },
84    /// The lines do not form a legal history.
85    Incoerente(String),
86}
87
88impl fmt::Display for ErroDiario {
89    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
90        match self {
91            Self::Io(motivo) => write!(f, "falha no diário: {motivo}"),
92            Self::Corrompido { linha } => write!(f, "diário corrompido na linha {linha}"),
93            Self::Incoerente(motivo) => write!(f, "diário incoerente: {motivo}"),
94        }
95    }
96}
97
98impl std::error::Error for ErroDiario {}

100/// Where the journal lives.
101pub trait Diario {
102    /// Writes one line durably, and returns only after that.
103    fn gravar(&mut self, lancamento: &Lancamento) -> Result<(), ErroDiario>;
104
105    /// Reads all lines, in order.
106    fn ler(&self) -> Result<Vec<Lancamento>, ErroDiario>;
107}
108
109/// What the journal knows about one payment.
110#[derive(Clone, Debug, PartialEq, Eq)]
111pub struct Registro {
112    /// The request.
113    pub pedido: PedidoPagamento,
114    /// Who processes it.
115    pub provedor: ProvedorId,
116    /// The current state.
117    pub estado: Estado,
118    /// The approval, once there is one.
119    pub aprovacao: Option<Box<Aprovacao>>,
120    /// The reason of a refusal or a cancellation.
121    pub motivo: Option<String>,
122    /// The action last attempted that has no outcome yet.
123    pub intencao: Option<Acao>,
124    /// What the adapter needs to find this payment after a restart.
125    pub rastro: Option<String>,
126}
127
128/// Applies one line to the map of payments.
129pub(crate) fn aplicar(
130    registros: &mut BTreeMap<IdPagamento, Registro>,
131    lancamento: &Lancamento,
132) -> Result<(), ErroDiario> {
133    let id = &lancamento.id;
134    match &lancamento.conteudo {
135        Conteudo::Aberto { pedido, provedor } => {
136            if registros.contains_key(id) {
137                return Err(ErroDiario::Incoerente(format!("{id} aberto duas vezes")));
138            }
139            registros.insert(
140                id.clone(),
141                Registro {
142                    pedido: pedido.clone(),
143                    provedor: *provedor,
144                    estado: Estado::Criado,
145                    aprovacao: None,
146                    motivo: None,
147                    intencao: None,
148                    rastro: None,
149                },
150            );
151        }
152        Conteudo::Intencao { acao } => {
153            let registro =
154                registros.get_mut(id).ok_or_else(|| ErroDiario::Incoerente(format!("{id} sem abertura")))?;
155            registro.intencao = Some(*acao);
156        }
157        Conteudo::Rastro { valor } => {
158            let registro =
159                registros.get_mut(id).ok_or_else(|| ErroDiario::Incoerente(format!("{id} sem abertura")))?;
160            registro.rastro = Some(valor.clone());
161        }
162        Conteudo::Transicao { evento, aprovacao, motivo } => {
163            let registro =
164                registros.get_mut(id).ok_or_else(|| ErroDiario::Incoerente(format!("{id} sem abertura")))?;
165            registro.estado = transitar(registro.estado, *evento)
166                .map_err(|erro| ErroDiario::Incoerente(format!("{id}: {erro}")))?;
167            if aprovacao.is_some() {
168                registro.aprovacao.clone_from(aprovacao);
169            }
170            if motivo.is_some() {
171                registro.motivo.clone_from(motivo);
172            }
173            // Sending is what the intention of starting was about, and it stays open until the
174            // provider answers. Every other move settles the intention.
175            if *evento != Evento::Enviar {
176                registro.intencao = None;
177            }
178        }
179    }
180    Ok(())
181}

183#[derive(Debug, Default)]
184struct DiarioMemoriaInterno {
185    linhas: Vec<Lancamento>,
186    resta: Option<usize>,
187}
188
189/// A journal in memory, shared by its clones, for tests and for targets with no files.
190#[derive(Clone, Debug, Default)]
191pub struct DiarioMemoria(Arc<Mutex<DiarioMemoriaInterno>>);
192
193impl DiarioMemoria {
194    /// An empty journal.
195    #[must_use]
196    pub fn new() -> Self {
197        Self::default()
198    }
199
200    /// Simulates the power going off: after this many more lines, every write fails.
201    pub fn cair_depois_de(&self, linhas: usize) {
202        self.0.lock().unwrap_or_else(PoisonError::into_inner).resta = Some(linhas);
203    }
204
205    /// The power came back: writes work again.
206    pub fn religar(&self) {
207        self.0.lock().unwrap_or_else(PoisonError::into_inner).resta = None;
208    }
209
210    /// How many lines it holds.
211    #[must_use]
212    pub fn tamanho(&self) -> usize {
213        self.0.lock().unwrap_or_else(PoisonError::into_inner).linhas.len()
214    }
215}
216
217impl Diario for DiarioMemoria {
218    fn gravar(&mut self, lancamento: &Lancamento) -> Result<(), ErroDiario> {
219        let mut interno = self.0.lock().unwrap_or_else(PoisonError::into_inner);
220        if let Some(resta) = interno.resta.as_mut() {
221            if *resta == 0 {
222                return Err(ErroDiario::Io("energia caiu (simulada)".into()));
223            }
224            *resta -= 1;
225        }
226        interno.linhas.push(lancamento.clone());
227        Ok(())
228    }
229
230    fn ler(&self) -> Result<Vec<Lancamento>, ErroDiario> {
231        Ok(self.0.lock().unwrap_or_else(PoisonError::into_inner).linhas.clone())
232    }
233}

239mod arquivo {
240    use std::fs::{self, File, OpenOptions};
241    use std::io::{Seek, SeekFrom, Write};
242    use std::path::{Path, PathBuf};
243
244    use super::{Diario, ErroDiario, Lancamento};
245
246    /// The native default: one JSON per line, flushed to the disk before each return.
247    ///
248    /// A partial last line is what a power loss leaves, so opening cuts it off. Damage anywhere
249    /// else is refused with [`ErroDiario::Corrompido`]. The file holds receipts and fiscal data:
250    /// keep it where only the point of sale reads.
251    #[derive(Debug)]
252    pub struct DiarioArquivo {
253        caminho: PathBuf,
254        arquivo: File,
255    }
256
257    fn io(erro: &std::io::Error) -> ErroDiario {
258        ErroDiario::Io(erro.to_string())
259    }
260
261    impl DiarioArquivo {
262        /// Opens or creates the journal, creating its folder if needed.
263        pub fn abrir(caminho: impl AsRef<Path>) -> Result<Self, ErroDiario> {
264            let caminho = caminho.as_ref().to_path_buf();
265            if let Some(pasta) = caminho.parent().filter(|p| !p.as_os_str().is_empty()) {
266                fs::create_dir_all(pasta).map_err(|e| io(&e))?;
267            }
268            let mut arquivo =
269                OpenOptions::new().create(true).read(true).append(true).open(&caminho).map_err(|e| io(&e))?;
270            let bytes = fs::read(&caminho).map_err(|e| io(&e))?;
271            let bom = bytes.iter().rposition(|b| *b == b'\n').map_or(0, |fim| fim + 1);
272            if bom != bytes.len() {
273                arquivo.set_len(bom as u64).map_err(|e| io(&e))?;
274                arquivo.seek(SeekFrom::End(0)).map_err(|e| io(&e))?;
275                arquivo.sync_all().map_err(|e| io(&e))?;
276            }
277            Ok(Self { caminho, arquivo })
278        }
279    }
280
281    impl Diario for DiarioArquivo {
282        fn gravar(&mut self, lancamento: &Lancamento) -> Result<(), ErroDiario> {
283            let mut linha = serde_json::to_vec(lancamento).map_err(|e| ErroDiario::Io(e.to_string()))?;
284            linha.push(b'\n');
285            self.arquivo.write_all(&linha).map_err(|e| io(&e))?;
286            self.arquivo.sync_data().map_err(|e| io(&e))
287        }
288
289        fn ler(&self) -> Result<Vec<Lancamento>, ErroDiario> {
290            let texto = fs::read(&self.caminho).map_err(|e| io(&e))?;
291            let mut linhas = Vec::new();
292            for (indice, linha) in texto.split(|b| *b == b'\n').enumerate() {
293                if linha.is_empty() {
294                    continue;
295                }
296                let lancamento = serde_json::from_slice(linha)
297                    .map_err(|_| ErroDiario::Corrompido { linha: indice + 1 })?;
298                linhas.push(lancamento);
299            }
300            Ok(linhas)
301        }
302    }
303}

318        let mut diario = DiarioMemoria::new();

320        assert!(diario.gravar(&abertura(1)).is_ok());
321        assert!(diario.gravar(&abertura(2)).is_err());

332            Lancamento { seq: 1, id: id.clone(), conteudo: Conteudo::Intencao { acao: Acao::Iniciar } };
