47/// `RSP_STAT`, the result of a command.
48#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
49pub struct StatusAbecs(pub u16);
50
51impl StatusAbecs {
52    /// Success.
53    pub const OK: Self = Self(0);
54    /// Secure communication was asked for and does not exist.
55    pub const NOSEC: Self = Self(3);
56    /// The `CLEAR` key.
57    pub const BACKSP: Self = Self(8);
58    /// A call out of order, or a command it does not know.
59    pub const INVCALL: Self = Self(10);
60    /// A bad parameter.
61    pub const INVPARM: Self = Self(11);
62    /// The time given to the customer ran out.
63    pub const TIMEOUT: Self = Self(12);
64    /// The customer pressed `CANCEL`.
65    pub const CANCEL: Self = Self(13);
66    /// A mandatory parameter is missing.
67    pub const MANDAT: Self = Self(19);
68    /// An internal error of the pinpad.
69    pub const INTERR: Self = Self(40);
70    /// The media file does not exist.
71    pub const MFNFOUND: Self = Self(100);
72    /// The media format or size is not accepted.
73    pub const MFERRFMT: Self = Self(101);
74    /// The loaded media does not match what `MLI` announced.
75    pub const MFERR: Self = Self(102);
76
77    /// True for [`Self::OK`].
78    #[must_use]
79    pub const fn ok(self) -> bool {
80        self.0 == 0
81    }
82}

84/// A message of the fixed table `GCD` shows, by its index. For PCI reasons, free text is not possible.
85///
86/// Only what a checkout uses has a name. Indexes for a card number, for the card security code and
87/// for the last digits of a card are refused, also through [`Self::Outra`].
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub enum MensagemGcd {
90    /// Index 0003h: phone number.
91    Telefone,
92    /// Index 0005h: area code and phone number.
93    DddTelefone,
94    /// Index 0007h: CPF.
95    Cpf,
96    /// Index 000Dh: CNPJ.
97    Cnpj,
98    /// Index 001Eh: loyalty code.
99    Fidelidade,
100    /// Index 0030h: rate the service from 0 to 10.
101    AvaliacaoAtendimento,
102    /// Index 0031h: a token.
103    Token,
104    /// Any other index of the table.
105    Outra(u16),
106}
107
108impl MensagemGcd {
109    /// The index in the table.
110    #[must_use]
111    pub const fn indice(self) -> u16 {
112        match self {
113            Self::Telefone => 0x0003,
114            Self::DddTelefone => 0x0005,
115            Self::Cpf => 0x0007,
116            Self::Cnpj => 0x000D,
117            Self::Fidelidade => 0x001E,
118            Self::AvaliacaoAtendimento => 0x0030,
119            Self::Token => 0x0031,
120            Self::Outra(indice) => indice,
121        }
122    }
123
124    /// Whether this crate refuses it: reading card data in clear is not something a POS software does.
125    #[must_use]
126    pub const fn proibida(self) -> bool {
127        matches!(self.indice(), 0x000B | 0x000C | 0x0032)
128    }
129}

152/// Why a command could not be built or a response could not be read.
153#[derive(Clone, Debug, PartialEq, Eq)]
154#[non_exhaustive]
155pub enum ErroComando {
156    /// A text or list does not fit where it goes.
157    Tamanho(&'static str),
158    /// A media name has a forbidden character or the wrong length.
159    NomeMidia(String),
160    /// This index would read card data in clear.
161    MensagemProibida(u16),
162    /// The bytes do not form a response.
163    RespostaInvalida(&'static str),
164}
165
166impl fmt::Display for ErroComando {
167    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
168        match self {
169            Self::Tamanho(campo) => write!(f, "tamanho inválido em {campo}"),
170            Self::NomeMidia(nome) => write!(f, "nome de mídia inválido: {nome:?}"),
171            Self::MensagemProibida(indice) => {
172                write!(f, "mensagem {indice:04X}h leria dados de cartão em aberto")
173            }
174            Self::RespostaInvalida(motivo) => write!(f, "resposta inválida: {motivo}"),
175        }
176    }
177}

181/// A command to the pinpad.
182#[derive(Clone, Debug, PartialEq, Eq)]
183#[non_exhaustive]
184pub enum Comando {
185    /// `OPN` in its classic format. The secure format needs RSA and is not here.
186    Abrir,
187    /// `GIN`, general information. Index zero is the pinpad itself.
188    InformacaoGeral {
189        /// `GIN_ACQIDX`.
190        rede: u8,
191    },
192    /// `GIX`. An empty list asks for what the pinpad returns by default.
193    InformacaoEstendida {
194        /// The `PP_*` identifiers wanted, at most 64.
195        ids: Vec<u16>,
196    },
197    /// `DSP`: two rows of sixteen columns.
198    Exibir {
199        /// The first row.
200        linha1: String,
201        /// The second row.
202        linha2: String,
203    },
204    /// `DEX`: free text, with line breaks, up to 160 characters.
205    ExibirLivre {
206        /// The text.
207        mensagem: String,
208    },
209    /// `GKY`: waits for a key that is not a digit.
210    LerTecla,
211    /// `GCD`: asks the customer to type a value, in clear, under a message of the fixed table.
212    Capturar {
213        /// The message.
214        mensagem: MensagemGcd,
215        /// The least digits.
216        minimo: Option<u8>,
217        /// The most digits, up to 32.
218        maximo: Option<u8>,
219        /// How long to wait for the customer, in seconds.
220        tempo_s: Option<u8>,
221        /// Accepts letters, which only exists from the version 2.20.
222        alfanumerico: bool,
223    },
224    /// `MNU`: a menu for the customer, up to 20 options of 24 characters.
225    Menu {
226        /// The title.
227        titulo: Option<String>,
228        /// The options.
229        opcoes: Vec<String>,
230        /// How long to wait for the customer, in seconds.
231        tempo_s: Option<u8>,
232    },
233    /// `CLO`: releases the pinpad and leaves two rows on its display.
234    Fechar {
235        /// The first row.
236        linha1: String,
237        /// The second row.
238        linha2: String,
239    },
240    /// `MLI`: starts loading a media file.
241    CarregarMidiaInicio {
242        /// Up to eight letters and digits.
243        nome: String,
244        /// The whole size.
245        tamanho: u32,
246        /// The CRC of the whole file.
247        crc: u16,
248        /// The format.
249        tipo: TipoMidia,
250    },
251    /// `MLR`: a piece of the file, up to [`TAMANHO_MAXIMO_MLR`] bytes.
252    CarregarMidiaDados {
253        /// The bytes.
254        dados: Vec<u8>,
255    },
256    /// `MLE`: ends the load and makes the pinpad check it.
257    CarregarMidiaFim,
258    /// `DSI`: shows a media file already loaded.
259    ExibirMidia {
260        /// Its name.
261        nome: String,
262    },
263}

265/// The command to announce a file: its size and CRC as the pinpad will check them.
266pub fn comando_inicio_midia(nome: &str, tipo: TipoMidia, arquivo: &[u8]) -> Result<Comando, ErroComando> {
267    let tamanho = u32::try_from(arquivo.len()).map_err(|_| ErroComando::Tamanho("arquivo de mídia"))?;
268    Ok(Comando::CarregarMidiaInicio { nome: nome.to_owned(), tamanho, crc: crc16(arquivo), tipo })
269}

271fn latin1(texto: &str) -> Vec<u8> {
272    texto.chars().map(|c| u8::try_from(u32::from(c)).unwrap_or(b'?')).collect()
273}
274
275fn linha16(texto: &str) -> Vec<u8> {
276    let mut bytes = latin1(texto);
277    bytes.truncate(16);
278    bytes.resize(16, b' ');
279    bytes
280}
281
282fn nome_midia(nome: &str) -> Result<Vec<u8>, ErroComando> {
283    if nome.is_empty() || nome.len() > 8 || !nome.bytes().all(|b| b.is_ascii_alphanumeric()) {
284        return Err(ErroComando::NomeMidia(nome.to_owned()));
285    }
286    Ok(nome.as_bytes().to_vec())
287}

289/// Lays parameters out as "Abecs Command" blocks of at most 999 bytes each.
290fn blocos_abecs(id: &str, parametros: &[(u16, Vec<u8>)]) -> Result<Vec<u8>, ErroComando> {
291    let mut blocos: Vec<Vec<u8>> = Vec::new();
292    let mut atual: Vec<u8> = Vec::new();
293    for (parid, valor) in parametros {
294        if valor.len() > TAMANHO_MAXIMO_PARAMETRO {
295            return Err(ErroComando::Tamanho("parâmetro"));
296        }
297        if atual.len() + 4 + valor.len() > 999 {
298            blocos.push(std::mem::take(&mut atual));
299        }
300        atual.extend_from_slice(&parid.to_be_bytes());
301        // The check above keeps this length inside two bytes.
302        atual.extend_from_slice(&(valor.len() as u16).to_be_bytes());
303        atual.extend_from_slice(valor);
304    }
305    if !atual.is_empty() {
306        blocos.push(atual);
307    }
308    let mut saida = id.as_bytes().to_vec();
309    for bloco in blocos {
310        saida.extend_from_slice(format!("{:03}", bloco.len()).as_bytes());
311        saida.extend_from_slice(&bloco);
312    }
313    Ok(saida)
314}
315
316fn bloco_fixo(id: &str, conteudo: &[u8]) -> Vec<u8> {
317    let mut saida = id.as_bytes().to_vec();
318    saida.extend_from_slice(format!("{:03}", conteudo.len()).as_bytes());
319    saida.extend_from_slice(conteudo);
320    saida
321}

323impl Comando {
324    /// True for commands that wait for the customer, which have no time limit on the link.
325    #[must_use]
326    pub const fn bloqueante(&self) -> bool {
327        matches!(self, Self::LerTecla | Self::Capturar { .. } | Self::Menu { .. })
328    }
329
330    /// The bytes of `PKTDATA`.
331    pub fn codificar(&self) -> Result<Vec<u8>, ErroComando> {
332        match self {
333            Self::Abrir => Ok(b"OPN".to_vec()),
334            Self::InformacaoGeral { rede } => {
335                if *rede > 99 {
336                    return Err(ErroComando::Tamanho("GIN_ACQIDX"));
337                }
338                Ok(bloco_fixo("GIN", format!("{rede:02}").as_bytes()))
339            }
340            Self::InformacaoEstendida { ids } => {
341                if ids.len() > 64 {
342                    return Err(ErroComando::Tamanho("SPE_IDLIST"));
343                }
344                if ids.is_empty() {
345                    return Ok(b"GIX".to_vec());
346                }
347                let lista = ids.iter().flat_map(|id| id.to_be_bytes()).collect();
348                blocos_abecs("GIX", &[(SPE_IDLIST, lista)])
349            }
350            Self::Exibir { linha1, linha2 } => {
351                let mut texto = linha16(linha1);
352                texto.extend(linha16(linha2));
353                Ok(bloco_fixo("DSP", &texto))
354            }
355            Self::ExibirLivre { mensagem } => {
356                let texto: Vec<u8> =
357                    latin1(mensagem).into_iter().map(|b| if b == b'\n' { 0x0D } else { b }).collect();
358                if texto.len() > 160 {
359                    return Err(ErroComando::Tamanho("DEX_MSG"));
360                }
361                let mut conteudo = format!("{:03}", texto.len()).into_bytes();
362                conteudo.extend(texto);
363                Ok(bloco_fixo("DEX", &conteudo))
364            }
365            Self::LerTecla => Ok(b"GKY".to_vec()),
366            Self::Capturar { mensagem, minimo, maximo, tempo_s, alfanumerico } => {
367                if mensagem.proibida() {
368                    return Err(ErroComando::MensagemProibida(mensagem.indice()));
369                }
370                if maximo.is_some_and(|m| m > 32 || minimo.is_some_and(|n| n > m)) {
371                    return Err(ErroComando::Tamanho("SPE_MAXDIG"));
372                }
373                let mut parametros = vec![(SPE_MSGIDX, mensagem.indice().to_be_bytes().to_vec())];
374                if let Some(minimo) = minimo {
375                    parametros.push((SPE_MINDIG, vec![*minimo]));
376                }
377                if let Some(maximo) = maximo {
378                    parametros.push((SPE_MAXDIG, vec![*maximo]));
379                }
380                if let Some(tempo) = tempo_s {
381                    parametros.push((SPE_TIMEOUT, vec![*tempo]));
382                }
383                if *alfanumerico {
384                    parametros.push((SPE_GCDOPT, b"1000".to_vec()));
385                }
386                blocos_abecs("GCD", &parametros)
387            }
388            Self::Menu { titulo, opcoes, tempo_s } => {
389                if opcoes.is_empty() || opcoes.len() > 20 {
390                    return Err(ErroComando::Tamanho("SPE_MNUOPT"));
391                }
392                let mut parametros = Vec::new();
393                if let Some(tempo) = tempo_s {
394                    parametros.push((SPE_TIMEOUT, vec![*tempo]));
395                }
396                for opcao in opcoes {
397                    let texto = latin1(opcao);
398                    if texto.len() > 24 {
399                        return Err(ErroComando::Tamanho("SPE_MNUOPT"));
400                    }
401                    parametros.push((SPE_MNUOPT, texto));
402                }
403                if let Some(titulo) = titulo {
404                    let texto = latin1(titulo);
405                    if texto.len() > 128 {
406                        return Err(ErroComando::Tamanho("SPE_DSPMSG"));
407                    }
408                    parametros.push((SPE_DSPMSG, texto));
409                }
410                blocos_abecs("MNU", &parametros)
411            }
412            Self::Fechar { linha1, linha2 } => {
413                let mut texto = linha16(linha1);
414                texto.extend(linha16(linha2));
415                Ok(bloco_fixo("CLO", &texto))
416            }
417            Self::CarregarMidiaInicio { nome, tamanho, crc, tipo } => {
418                let mut info = tamanho.to_be_bytes().to_vec();
419                info.extend_from_slice(&crc.to_be_bytes());
420                info.push(tipo.codigo());
421                info.extend_from_slice(&[0, 0, 0]);
422                blocos_abecs("MLI", &[(SPE_MFNAME, nome_midia(nome)?), (SPE_MFINFO, info)])
423            }
424            Self::CarregarMidiaDados { dados } => {
425                if dados.is_empty() || dados.len() > TAMANHO_MAXIMO_MLR {
426                    return Err(ErroComando::Tamanho("SPE_DATAIN"));
427                }
428                let parametros: Vec<(u16, Vec<u8>)> = dados
429                    .chunks(TAMANHO_MAXIMO_PARAMETRO)
430                    .map(|parte| (SPE_DATAIN, parte.to_vec()))
431                    .collect();
432                blocos_abecs("MLR", &parametros)
433            }
434            Self::CarregarMidiaFim => Ok(b"MLE".to_vec()),
435            Self::ExibirMidia { nome } => blocos_abecs("DSI", &[(SPE_MFNAME, nome_midia(nome)?)]),
436        }
437    }
438}

440/// A response of the pinpad.
441#[derive(Clone, Debug, PartialEq, Eq)]
442pub struct RespostaAbecs {
443    /// `RSP_ID`.
444    pub id: [u8; 3],
445    /// `RSP_STAT`.
446    pub status: StatusAbecs,
447    /// The data blocks, without their length prefixes.
448    pub blocos: Vec<Vec<u8>>,
449}
450
451impl RespostaAbecs {
452    /// Reads the content of a packet.
453    pub fn ler(pacote: &[u8]) -> Result<Self, ErroComando> {
454        if pacote.len() < 6 {
455            return Err(ErroComando::RespostaInvalida("menos de seis bytes"));
456        }
457        let id = [pacote[0], pacote[1], pacote[2]];
458        let status = numero(&pacote[3..6]).ok_or(ErroComando::RespostaInvalida("RSP_STAT"))?;
459        let mut blocos = Vec::new();
460        let mut resto = &pacote[6..];
461        while !resto.is_empty() {
462            if resto.len() < 3 {
463                return Err(ErroComando::RespostaInvalida("RSP_LEN incompleto"));
464            }
465            let tamanho = usize::from(numero(&resto[..3]).ok_or(ErroComando::RespostaInvalida("RSP_LEN"))?);
466            let fim = 3 + tamanho;
467            if resto.len() < fim {
468                return Err(ErroComando::RespostaInvalida("bloco incompleto"));
469            }
470            blocos.push(resto[3..fim].to_vec());
471            resto = &resto[fim..];
472        }
473        Ok(Self { id, status: StatusAbecs(status), blocos })
474    }
475
476    /// True for a notification, which comes in the middle of a blocking command.
477    #[must_use]
478    pub fn notificacao(&self) -> bool {
479        &self.id == b"NTM"
480    }
481
482    /// The fields of an "Abecs Command" response, by `PP_*` identifier.
483    pub fn campos(&self) -> Result<BTreeMap<u16, Vec<u8>>, ErroComando> {
484        let mut campos = BTreeMap::new();
485        for bloco in &self.blocos {
486            let mut resto = bloco.as_slice();
487            while !resto.is_empty() {
488                if resto.len() < 4 {
489                    return Err(ErroComando::RespostaInvalida("RSP_DATID incompleto"));
490                }
491                let id = u16::from_be_bytes([resto[0], resto[1]]);
492                let tamanho = usize::from(u16::from_be_bytes([resto[2], resto[3]]));
493                if resto.len() < 4 + tamanho {
494                    return Err(ErroComando::RespostaInvalida("RSP_DAT incompleto"));
495                }
496                campos.insert(id, resto[4..4 + tamanho].to_vec());
497                resto = &resto[4 + tamanho..];
498            }
499        }
500        Ok(campos)
501    }
502
503    /// The value of a field as text.
504    pub fn texto(&self, id: u16) -> Result<Option<String>, ErroComando> {
505        Ok(self.campos()?.get(&id).map(|valor| de_latin1(valor)))
506    }
507}
508
509/// Turns bytes the pinpad sent into text.
510#[must_use]
511pub fn de_latin1(bytes: &[u8]) -> String {
512    bytes.iter().map(|b| char::from(*b)).collect()
513}
514
515fn numero(bytes: &[u8]) -> Option<u16> {
516    let mut valor = 0u16;
517    for byte in bytes {
518        if !byte.is_ascii_digit() {
519            return None;
520        }
521        valor = valor * 10 + u16::from(byte - b'0');
522    }
523    Some(valor)
524}

526/// What `GIN` with index zero says.
527#[derive(Clone, Debug, PartialEq, Eq)]
528pub struct InformacaoGeral {
529    /// The maker.
530    pub fabricante: String,
531    /// The model.
532    pub modelo: String,
533    /// Whether it reads contactless cards.
534    pub aproximacao: bool,
535    /// The system version.
536    pub sistema: String,
537    /// The version of the specification it follows, such as `2.20`.
538    pub especificacao: String,
539    /// The version of its manager application.
540    pub gerenciador: String,
541    /// Its serial number.
542    pub serie: String,
543}
544
545impl InformacaoGeral {
546    /// Reads the fixed format of the first block.
547    pub fn ler(resposta: &RespostaAbecs) -> Result<Self, ErroComando> {
548        let bloco = resposta.blocos.first().ok_or(ErroComando::RespostaInvalida("GIN sem dados"))?;
549        if bloco.len() != 100 {
550            return Err(ErroComando::RespostaInvalida("GIN com tamanho diferente de 100"));
551        }
552        let campo =
553            |inicio: usize, tamanho: usize| de_latin1(&bloco[inicio..inicio + tamanho]).trim().to_owned();
554        Ok(Self {
555            fabricante: campo(0, 20),
556            modelo: campo(20, 19),
557            aproximacao: bloco[39] == b'C',
558            sistema: campo(40, 20),
559            especificacao: campo(60, 4),
560            gerenciador: campo(64, 16),
561            serie: campo(80, 20),
562        })
563    }
564}
