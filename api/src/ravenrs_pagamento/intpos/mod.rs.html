1//! PayGo Windows and the standard manager, through files. Only files: no `unsafe`, no library.
2//!
3//! The steps are always these: write `req/intpos.tmp`, rename it to `req/intpos.001`, wait for
4//! `resp/intpos.sts`, which says the manager took it, then wait for `resp/intpos.001`. The manager
5//! talks to the operator and the pinpad on its own window, which is why no question reaches ours.
6//!
7//! **Never run against a real manager, and those are Windows programs.** Only
8//! [`GerenciadorSimulado`] was run. Field numbers come from the public PayGo pages and from reading
9//! the behavior of an LGPL implementation; see `docs/PAGAMENTOS.md`.

32/// How long the manager has to write the status file, as others use it.
33pub const TEMPO_STS_MS: u64 = 7_000;
34/// How long to wait for a response before reporting the time out. The manager may still answer.
35pub const TEMPO_RESPOSTA_MS: u64 = 300_000;
36const INTERVALO_MS: u32 = 250;
37/// PayGo: the interface version these fields follow.
38const VERSAO_INTERFACE_PAYGO: &str = "225";
39/// PayGo field 706: fixed value 4, plus 8 (separate copies) and 128 (long NSU).
40const CAPACIDADES_PAYGO: &str = "140";

42/// The files of the exchange. The last one is ours, and keeps an approval until its third leg.
43#[derive(Clone, Debug, PartialEq, Eq)]
44pub struct Caminhos {
45    /// `req/intpos.tmp`.
46    pub req_tmp: PathBuf,
47    /// `req/intpos.001`.
48    pub req: PathBuf,
49    /// `resp/intpos.001`.
50    pub resp: PathBuf,
51    /// `resp/intpos.sts`.
52    pub sts: PathBuf,
53    /// Our note of the approval in the air, kept out of the manager's folders.
54    pub pendente: PathBuf,
55}

112/// A provider that speaks through files. The clock's wait runs in third legs and refunds.
113#[derive(Debug)]
114pub struct ProvedorArquivos<R> {
115    caminhos: Caminhos,
116    perfil: Perfil,
117    relogio: R,
118    em_curso: Option<EmCurso>,
119    tempo_sts_ms: u64,
120    tempo_resposta_ms: u64,
121}

123fn io_erro(erro: &io::Error) -> ErroProvedor {
124    ErroProvedor::Indisponivel(format!("arquivo: {erro}"))
125}
126
127/// A number for field 001-000 that is the same for the same payment and step, also after a restart.
128fn identificacao(id: &str, passo: &str) -> u64 {
129    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
130    for byte in id.bytes().chain(std::iter::once(b':')).chain(passo.bytes()) {
131        hash ^= u64::from(byte);
132        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
133    }
134    // Ten digits, with no zero, which an empty field would look like.
135    hash % 9_999_999_999 + 1
136}

138fn gravar_sincronizado(caminho: &Path, bytes: &[u8]) -> io::Result<()> {
139    let mut arquivo = fs::File::create(caminho)?;
140    arquivo.write_all(bytes)?;
141    arquivo.sync_all()
142}
143
144fn apagar_se_existe(caminho: &Path) -> io::Result<()> {
145    match fs::remove_file(caminho) {
146        Err(erro) if erro.kind() != io::ErrorKind::NotFound => Err(erro),
147        _ => Ok(()),
148    }
149}
150
151fn ler_se_existe(caminho: &Path) -> io::Result<Option<ArquivoIntPos>> {
152    match fs::read(caminho) {
153        Ok(bytes) => Ok(Some(ArquivoIntPos::ler(&bytes))),
154        Err(erro) if erro.kind() == io::ErrorKind::NotFound => Ok(None),
155        Err(erro) => Err(erro),
156    }
157}

159impl<R: Relogio> ProvedorArquivos<R> {
160    /// A provider over these files. The folders are created if missing.
161    pub fn new(caminhos: Caminhos, perfil: Perfil, relogio: R) -> Result<Self, ErroProvedor> {
162        for caminho in [&caminhos.req, &caminhos.resp, &caminhos.pendente] {
163            if let Some(pasta) = caminho.parent() {
164                fs::create_dir_all(pasta).map_err(|e| io_erro(&e))?;
165            }
166        }
167        Ok(Self {
168            caminhos,
169            perfil,
170            relogio,
171            em_curso: None,
172            tempo_sts_ms: TEMPO_STS_MS,
173            tempo_resposta_ms: TEMPO_RESPOSTA_MS,
174        })
175    }
176
177    /// Other time limits, in milliseconds.
178    #[must_use]
179    pub fn com_tempos(mut self, sts_ms: u64, resposta_ms: u64) -> Self {
180        self.tempo_sts_ms = sts_ms;
181        self.tempo_resposta_ms = resposta_ms;
182        self
183    }

185    /// `ATV`: whether the manager is alive. It waits at most the status limit.
186    pub fn ativo(&mut self) -> Result<bool, ErroProvedor> {
187        let numero = identificacao("atv", &self.relogio.agora_ms().to_string());
188        let pedido = self.pedido("ATV", numero);
189        match self.trocar_so_status(&pedido, numero) {
190            Ok(()) => Ok(true),
191            Err(ErroProvedor::Tempo) => Ok(false),
192            Err(erro) => Err(erro),
193        }
194    }
195
196    fn pedido(&self, comando: &str, numero: u64) -> ArquivoIntPos {
197        let mut arquivo = ArquivoIntPos::novo(comando, numero);
198        if let Perfil::PayGo { empresa, aplicacao, versao, registro_certificacao } = &self.perfil {
199            arquivo.definir(733, 0, VERSAO_INTERFACE_PAYGO);
200            arquivo.definir(738, 0, registro_certificacao.as_str());
201            if !matches!(comando, "ATV" | "CNF" | "NCN") {
202                arquivo.definir(706, 0, CAPACIDADES_PAYGO);
203                arquivo.definir(716, 0, empresa.as_str());
204                arquivo.definir(735, 0, aplicacao.as_str());
205                arquivo.definir(736, 0, versao.as_str());
206            }
207        }
208        arquivo
209    }
210
211    /// Clears what the last exchange left, writes the temporary file and renames it. The rename is
212    /// what the manager sees, so it never reads half a file.
213    fn publicar(&self, pedido: &ArquivoIntPos) -> Result<(), ErroProvedor> {
214        if self.caminhos.req.exists() {
215            return Err(ErroProvedor::Indisponivel("o gerenciador não retirou a requisição anterior".into()));
216        }
217        apagar_se_existe(&self.caminhos.sts).map_err(|e| io_erro(&e))?;
218        apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
219        gravar_sincronizado(&self.caminhos.req_tmp, &pedido.codificar()).map_err(|e| io_erro(&e))?;
220        fs::rename(&self.caminhos.req_tmp, &self.caminhos.req).map_err(|e| io_erro(&e))
221    }
222
223    /// True if the status file for this number arrived; it is removed then.
224    fn status_chegou(&self, numero: u64) -> Result<bool, ErroProvedor> {
225        match ler_se_existe(&self.caminhos.sts).map_err(|e| io_erro(&e))? {
226            Some(sts) if sts.identificacao() == Some(numero.to_string().as_str()) => {
227                apagar_se_existe(&self.caminhos.sts).map_err(|e| io_erro(&e))?;
228                Ok(true)
229            }
230            _ => Ok(false),
231        }
232    }
233
234    fn resposta_chegou(&self, numero: u64) -> Result<Option<ArquivoIntPos>, ErroProvedor> {
235        match ler_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))? {
236            Some(resposta)
237                if resposta.completo() && resposta.identificacao() == Some(numero.to_string().as_str()) =>
238            {
239                Ok(Some(resposta))
240            }
241            _ => Ok(None),
242        }
243    }
244
245    /// For commands with no response file: waits for the status, blocking up to the limit.
246    fn trocar_so_status(&mut self, pedido: &ArquivoIntPos, numero: u64) -> Result<(), ErroProvedor> {
247        self.publicar(pedido)?;
248        let limite = self.relogio.agora_ms() + self.tempo_sts_ms;
249        loop {
250            if self.status_chegou(numero)? {
251                return Ok(());
252            }
253            if self.relogio.agora_ms() >= limite {
254                // Taking it back means the manager never saw it; it can simply be sent again.
255                let _ = apagar_se_existe(&self.caminhos.req);
256                return Err(ErroProvedor::Tempo);
257            }
258            self.relogio.esperar_ms(u64::from(INTERVALO_MS));
259        }
260    }
261
262    /// Keeps the answer in our note before anything else, so that a crash here loses nothing.
263    fn anotar(
264        &self,
265        id: Option<&IdPagamento>,
266        pix: bool,
267        resposta: &ArquivoIntPos,
268    ) -> Result<(), ErroProvedor> {
269        let nota = NotaPendente {
270            id: id.cloned(),
271            pix,
272            resposta: resposta.codificar().iter().map(|b| char::from(*b)).collect(),
273        };
274        let bytes = serde_json::to_vec(&nota).map_err(|e| ErroProvedor::Protocolo(e.to_string()))?;
275        let temporario = self.caminhos.pendente.with_extension("tmp");
276        gravar_sincronizado(&temporario, &bytes).map_err(|e| io_erro(&e))?;
277        fs::rename(&temporario, &self.caminhos.pendente).map_err(|e| io_erro(&e))
278    }
279
280    fn nota(&self) -> Result<Option<NotaPendente>, ErroProvedor> {
281        match fs::read(&self.caminhos.pendente) {
282            Ok(bytes) => serde_json::from_slice(&bytes)
283                .map(Some)
284                .map_err(|_| ErroProvedor::Protocolo("nota de pendência ilegível".into())),
285            Err(erro) if erro.kind() == io::ErrorKind::NotFound => Ok(None),
286            Err(erro) => Err(io_erro(&erro)),
287        }
288    }
289
290    fn resposta_da_nota(nota: &NotaPendente) -> ArquivoIntPos {
291        let bytes: Vec<u8> =
292            nota.resposta.chars().map(|c| u8::try_from(u32::from(c)).unwrap_or(b'?')).collect();
293        ArquivoIntPos::ler(&bytes)
294    }
295
296    fn terceira_perna(
297        &mut self,
298        comando: &str,
299        referencia: &ReferenciaTransacao,
300    ) -> Result<(), ErroProvedor> {
301        let chave = referencia.controle.as_deref().or(referencia.nsu.as_deref()).unwrap_or("sem-controle");
302        let numero = identificacao(chave, comando);
303        let mut pedido = self.pedido(comando, numero);
304        if let Some(rede) = &referencia.rede {
305            pedido.definir(10, 0, rede.as_str());
306        }
307        if let Some(nsu) = &referencia.nsu {
308            pedido.definir(12, 0, nsu.as_str());
309        }
310        if let Some(controle) = &referencia.controle {
311            pedido.definir(27, 0, controle.as_str());
312        }
313        self.trocar_so_status(&pedido, numero)?;
314        // The note only goes away when it is about this same transaction.
315        let e_esta = self
316            .nota()?
317            .is_some_and(|nota| Self::resposta_da_nota(&nota).valor(27) == referencia.controle.as_deref());
318        if e_esta {
319            apagar_se_existe(&self.caminhos.pendente).map_err(|e| io_erro(&e))?;
320        }
321        Ok(())
322    }
323}

325/// True when field 009-000 says it worked.
326fn aprovada(resposta: &ArquivoIntPos) -> bool {
327    resposta.valor(9) == Some("0")
328}

330fn negado(resposta: &ArquivoIntPos) -> Passo {
331    Passo::Negado {
332        codigo: resposta.valor(9).unwrap_or("?").to_owned(),
333        motivo: resposta.valor(30).unwrap_or("Transação não aprovada").to_owned(),

337fn referencia(id: Option<&IdPagamento>, resposta: &ArquivoIntPos) -> ReferenciaTransacao {
338    let mut extras = BTreeMap::new();
339    for (nome, campo) in [("data", 22), ("hora", 23)] {
340        if let Some(valor) = resposta.valor(campo) {
341            extras.insert(nome.to_owned(), valor.to_owned());
342        }
343    }
344    ReferenciaTransacao {
345        id: id.cloned(),
346        rede: resposta.valor(10).map(str::to_owned),
347        nsu: resposta.valor(12).map(str::to_owned),
348        controle: resposta.valor(27).map(str::to_owned),
349        valor_centavos: resposta.valor(3).and_then(|valor| valor.parse().ok()).unwrap_or(0),
350        extras,
351    }
352}

354fn aprovacao(id: Option<&IdPagamento>, pix: bool, resposta: &ArquivoIntPos) -> Aprovacao {
355    let mut fiscais = DadosFiscaisPagamento::integrado();
356    // PayGo: 013-000 does not apply to Pix, and for Pix 012-000 returns the end-to-end id.
357    let carteira = resposta.valor(749) == Some("8");
358    if pix || carteira {
359        fiscais.com_c_aut(resposta.valor(12), "a resposta do Pix veio sem o campo 012-000");
360        if fiscais.c_aut.is_some() {
361            fiscais.alertas.push(AlertaFiscal::CAutPixPeloEndToEnd);
362        }
363    } else {
364        fiscais.com_c_aut(resposta.valor(13), "a resposta veio sem o campo 013-000");
365        fiscais.com_bandeira(resposta.valor(748).or_else(|| resposta.valor(40)));
366    }
367    fiscais.com_cnpj_da_rede(resposta.valor(10));
368    let tipo_efetivo = match resposta.valor(731) {
369        _ if pix || carteira => Some(TipoPagamento::Pix),
370        Some("1") => Some(TipoPagamento::Credito {
371            parcelas: resposta.valor(18).and_then(|valor| valor.parse().ok()).unwrap_or(1),
372        }),
373        Some("2") => Some(TipoPagamento::Debito),
374        Some("3") => Some(TipoPagamento::Voucher),
375        _ => None,
376    };
377    Aprovacao {
378        referencia: referencia(id, resposta),
379        dados_fiscais: fiscais,
380        vias: Vias {
381            cliente: resposta.linhas(712, 713),
382            estabelecimento: resposta.linhas(714, 715),
383            unica: resposta.linhas(28, 29),
384            impressas_no_terminal: false,
385        },
386        mensagem_operador: resposta.valor(30).map(str::to_owned),
387        tipo_efetivo,
388        avisos: Vec::new(),
389    }
390}

392impl<R: Relogio> ProvedorPagamento for ProvedorArquivos<R> {
393    fn id(&self) -> ProvedorId {
394        match self.perfil {
395            Perfil::GerenciadorPadrao => ProvedorId::GerenciadorPadrao,
396            Perfil::PayGo { .. } => ProvedorId::PayGoArquivos,
397        }
398    }
399
400    fn capacidades(&self) -> Capacidades {
401        Capacidades::ARQUIVOS
402    }
403
404    /// Refuses to start while an approval is in the air: a new request clears the manager's files.
405    fn iniciar(&mut self, pedido: &PedidoPagamento) -> Result<Passo, ErroProvedor> {
406        if self.nota()?.is_some() {
407            return Err(ErroProvedor::Recusado {
408                codigo: "PENDENCIA".into(),
409                motivo: "há uma transação sem confirmação; recupere antes".into(),
410            });
411        }
412        let numero = identificacao(pedido.id.as_str(), "CRT");
413        let mut arquivo = self.pedido("CRT", numero);
414        if let Some(documento) = &pedido.documento {
415            arquivo.definir(2, 0, documento.as_str());
416        }
417        arquivo.definir(3, 0, pedido.valor_centavos.to_string());
418        arquivo.definir(4, 0, "0");
419        match pedido.tipo {
420            TipoPagamento::Escolher => {}
421            TipoPagamento::Credito { .. } => arquivo.definir(731, 0, "1"),
422            TipoPagamento::Debito => arquivo.definir(731, 0, "2"),
423            TipoPagamento::Voucher => arquivo.definir(731, 0, "3"),
424            TipoPagamento::Pix => arquivo.definir(749, 0, "8"),
425        }
426        self.publicar(&arquivo)?;
427        self.em_curso = Some(EmCurso {
428            id: pedido.id.clone(),
429            identificacao: numero,
430            tipo: pedido.tipo,
431            fase: Fase::Status { limite: self.relogio.agora_ms() + self.tempo_sts_ms },
432        });
433        Ok(Passo::Aguardar { mensagem: Some("Aguardando o gerenciador".into()), tentar_em_ms: INTERVALO_MS })
434    }
435
436    fn continuar(&mut self, id: &IdPagamento, entrada: EntradaOperador) -> Result<Passo, ErroProvedor> {
437        let Some(curso) = self.em_curso.clone().filter(|curso| &curso.id == id) else {
438            return Err(ErroProvedor::Protocolo(format!("{id} não está em curso neste adaptador")));
439        };
440        let agora = self.relogio.agora_ms();
441        match curso.fase {
442            Fase::Status { limite } => {
443                if self.status_chegou(curso.identificacao)? {
444                    let limite = agora + self.tempo_resposta_ms;
445                    if let Some(atual) = self.em_curso.as_mut() {
446                        atual.fase = Fase::Resposta { limite };
447                    }
448                } else if agora >= limite {
449                    self.em_curso = None;
450                    if self.caminhos.req.exists() {
451                        // Taken back before the manager saw it: nothing happened, for certain.
452                        apagar_se_existe(&self.caminhos.req).map_err(|e| io_erro(&e))?;
453                        return Err(ErroProvedor::Recusado {
454                            codigo: "GP-INATIVO".into(),
455                            motivo: "o gerenciador não respondeu".into(),
456                        });
457                    }
458                    return Err(ErroProvedor::Tempo);
459                }
460                Ok(Passo::Aguardar {
461                    mensagem: Some("Aguardando o gerenciador".into()),
462                    tentar_em_ms: INTERVALO_MS,
463                })
464            }
465            Fase::Resposta { limite } => {
466                let Some(resposta) = self.resposta_chegou(curso.identificacao)? else {
467                    if agora >= limite {
468                        self.em_curso = None;
469                        return Err(ErroProvedor::Tempo);
470                    }
471                    let mensagem = if entrada == EntradaOperador::Cancelar {
472                        "Cancele na janela do gerenciador"
473                    } else {
474                        "Conclua na janela do gerenciador"
475                    };
476                    return Ok(Passo::Aguardar {
477                        mensagem: Some(mensagem.into()),
478                        tentar_em_ms: INTERVALO_MS,
479                    });
480                };
481                self.em_curso = None;
482                let pix = curso.tipo == TipoPagamento::Pix;
483                if aprovada(&resposta) {
484                    self.anotar(Some(id), pix, &resposta)?;
485                    apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
486                    Ok(Passo::Aprovado(Box::new(aprovacao(Some(id), pix, &resposta))))
487                } else {
488                    apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
489                    Ok(negado(&resposta))
490                }
491            }
492        }
493    }

508    /// `CNC`, then its own `CNF`. It blocks until the manager answers, up to the limit, since the
509    /// operator works on the manager's window in between.
510    fn estornar(&mut self, referencia: &ReferenciaTransacao) -> Result<(), ErroProvedor> {
511        let nsu =
512            referencia.nsu.as_deref().ok_or_else(|| ErroProvedor::Protocolo("referência sem NSU".into()))?;
513        let numero = identificacao(nsu, "CNC");
514        let mut pedido = self.pedido("CNC", numero);
515        pedido.definir(3, 0, referencia.valor_centavos.to_string());
516        pedido.definir(4, 0, "0");
517        pedido.definir(12, 0, nsu);
518        if let Some(rede) = &referencia.rede {
519            pedido.definir(10, 0, rede.as_str());
520        }
521        for (nome, campo) in [("data", 22), ("hora", 23)] {
522            if let Some(valor) = referencia.extras.get(nome) {
523                pedido.definir(campo, 0, valor.as_str());
524            }
525        }
526        self.publicar(&pedido)?;
527        let limite = self.relogio.agora_ms() + self.tempo_resposta_ms;
528        let resposta = loop {
529            if let Some(resposta) = self.resposta_chegou(numero)? {
530                break resposta;
531            }
532            if self.relogio.agora_ms() >= limite {
533                return Err(ErroProvedor::Tempo);
534            }
535            self.relogio.esperar_ms(u64::from(INTERVALO_MS));
536        };
537        let _ = self.status_chegou(numero);
538        apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
539        if !aprovada(&resposta) {
540            return Err(ErroProvedor::Recusado {
541                codigo: resposta.valor(9).unwrap_or("?").to_owned(),
542                motivo: resposta.valor(30).unwrap_or("Cancelamento negado").to_owned(),
543            });
544        }
545        let do_estorno = self::referencia(referencia.id.as_ref(), &resposta);
546        self.terceira_perna("CNF", &do_estorno)
547    }
548
549    fn consultar(&mut self, id: &IdPagamento) -> Result<SituacaoProvedor, ErroProvedor> {
550        if self.em_curso.as_ref().is_some_and(|curso| &curso.id == id) {
551            return Ok(SituacaoProvedor::EmAndamento);
552        }
553        if let Some(nota) = self.nota()?.filter(|nota| nota.id.as_ref() == Some(id)) {
554            let resposta = Self::resposta_da_nota(&nota);
555            return Ok(SituacaoProvedor::Aprovada(Box::new(aprovacao(Some(id), nota.pix, &resposta))));
556        }
557        let numero = identificacao(id.as_str(), "CRT");
558        if let Some(resposta) = self.resposta_chegou(numero)? {
559            if aprovada(&resposta) {
560                self.anotar(Some(id), false, &resposta)?;
561                apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
562                return Ok(SituacaoProvedor::Aprovada(Box::new(aprovacao(Some(id), false, &resposta))));
563            }
564            apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
565            return Ok(SituacaoProvedor::Negada {
566                codigo: resposta.valor(9).unwrap_or("?").to_owned(),
567                motivo: resposta.valor(30).unwrap_or("Transação não aprovada").to_owned(),
568            });
569        }
570        if let Some(pedido) = ler_se_existe(&self.caminhos.req).map_err(|e| io_erro(&e))?
571            && pedido.identificacao() == Some(numero.to_string().as_str())
572        {
573            // The manager never took it. Taking it back settles the matter.
574            apagar_se_existe(&self.caminhos.req).map_err(|e| io_erro(&e))?;
575            return Ok(SituacaoProvedor::Desconhecida);
576        }
577        // It may be with the manager, on its window, and no file tells that.
578        Ok(SituacaoProvedor::Indeterminada)
579    }
580
581    fn pendencias(&mut self) -> Result<Vec<ReferenciaTransacao>, ErroProvedor> {
582        if let Some(nota) = self.nota()? {
583            let resposta = Self::resposta_da_nota(&nota);
584            return Ok(vec![referencia(nota.id.as_ref(), &resposta)]);
585        }
586        // An approved response nobody took, from an earlier session.
587        if let Some(resposta) = ler_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?
588            && resposta.completo()
589            && aprovada(&resposta)
590            && resposta.comando() == Some("CRT")
591        {
592            self.anotar(None, false, &resposta)?;
593            apagar_se_existe(&self.caminhos.resp).map_err(|e| io_erro(&e))?;
594            return Ok(vec![referencia(None, &resposta)]);
595        }
596        Ok(Vec::new())
597    }
