1//! A chat app: the conversations in a side bar that collapses to its strip, the conversation in the
2//! middle, and what the assistant made in a panel at the right.
3
4use ravenrs::prelude::*;
5
6const DRAFT: &str = r#"# Monthly sales report
7
8| Region | Sales | vs. last month |
9|--------|------:|---------------:|
10| North  | 128   | +12%           |
11| South  | 94    | -3%            |
12| East   | 151   | +21%           |
13"#;
14
15fn start() -> DockLayout {
16    DockLayout::new()
17        .center(DockNode::tabs(["conversation"]))
18        .left(DockNode::tabs(["chats"]), 260.0)
19        .right(DockNode::tabs(["artifact"]), 420.0)
20}
21
22fn panels() -> Vec<DockPanel> {
23    vec![
24        DockPanel::new("chats", "Chats").icon(IconName::MessageSquare).closable(false),
25        DockPanel::new("conversation", "Sales report").closable(false),
26        DockPanel::new("artifact", "Report draft").icon(IconName::FileText),
27    ]
28}
29
30/// The chat app, as a whole window.
31pub fn chat() -> impl View {
32    let layout = State::new(start());
33    let messages = State::new(vec![
34        ChatMessage::person("Can you turn last month's sales into a short report?"),
35        ChatMessage::assistant(
36            "Here is a draft with the three regions and how each did against the month before. \
37             It is open beside the conversation, so you can edit it there.",
38        ),
39        ChatMessage::person("Great. Can you say which region grew the most?"),
40        ChatMessage::assistant("East grew the most, 21%, followed by North with 12%. South fell 3%."),
41    ]);
42    let draft = State::new(DRAFT.to_owned());
43    let content = move |id: &str| -> AnyView {
44        match id {
45            "chats" => Box::new(
46                VStack::new((
47                    Button::new("New chat").icon(IconName::Plus).style(ButtonStyle::Primary),
48                    Text::new("Today").style(TextStyle::Caption),
49                    Text::new("Sales report"),
50                    Text::new("Trip to the coast"),
51                    Text::new("Yesterday").style(TextStyle::Caption),
52                    Text::new("Rust lifetimes"),
53                    Text::new("Birthday party ideas"),
54                ))
55                .spacing(10.0)
56                .padding(12.0),
57            ),
58            "conversation" => Box::new(Chat::new(messages.binding()).placeholder("Reply…")),
59            "artifact" => Box::new(
60                CodeEditor::new(draft.binding()).language(CodeLanguage::Plain).line_numbers(false).fill(),
61            ),
62            other => Box::new(Text::new(other.to_owned())),
63        }
64    };
65    let area = DockArea::new(layout.binding(), panels(), content)
66        .tab_bar(DockTabBar::WhenSeveral)
67        .label("Workbench");
68    VStack::new((TitleBar::new("Raven Chat"), area)).spacing(0.0).grow(1.0).align(Align::Stretch)
69}
