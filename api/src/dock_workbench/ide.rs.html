1//! An IDE: a menu bar and a toolbar over the work, the toolbox and the servers hidden at the left
2//! edge, the project tree over the properties at the right, the list of errors, the output and the
3//! call stack below, and a search window floating over the code.
4
5use std::collections::HashMap;
6
7use ravenrs::prelude::*;
8
9use crate::console;

11const ORDER: &str = r#"use crate::customer::Customer;
12
13/// An order, with its customer and its lines.
14pub struct Order {
15    pub customer: Customer,
16    pub lines: Vec<Line>,
17}
18
19impl Order {
20    pub fn total(&self) -> f64 {
21        self.lines.iter().map(Line::amount).sum()
22    }
23}
24"#;
25
26const CUSTOMER: &str = r#"/// Who buys.
27pub struct Customer {
28    pub name: String,
29    pub tax_id: String,
30}
31"#;
32
33const STARTUP: &str = r#"fn main() {
34    let app = shop::App::new();
35    app.run();
36}
37"#;
38
39const OUTPUT: &str = "Build started...\n1>------ Build started: Project: Shop, Configuration: Debug ------\n1>   Compiling shop v0.1.0\n1>error[E0425]: cannot find type `Line` in this scope\n========== Build: 0 succeeded, 1 failed ==========";

41fn start() -> DockLayout {
42    let mut layout = DockLayout::new()
43        .center(DockNode::tabs(["order.rs", "customer.rs", "main.rs"]))
44        .left(DockNode::tabs(["toolbox", "servers"]), 240.0)
45        .right(DockNode::vertical([DockNode::tabs(["solution"]), DockNode::tabs(["properties"])]), 300.0)
46        .bottom(DockNode::tabs(["errors", "output", "stack"]), 200.0)
47        .top(DockNode::tabs(["find"]), 120.0);
48    layout.auto_hide("toolbox");
49    layout.auto_hide("servers");
50    layout.float("find", Rect::new(560.0, 70.0, 360.0, 190.0));
51    layout
52}
53
54fn panels() -> Vec<DockPanel> {
55    let document = |id: &str| DockPanel::new(id, id).kind(DockPanelKind::Document).icon(IconName::Code);
56    vec![
57        DockPanel::new("toolbox", "Toolbox").icon(IconName::Package),
58        DockPanel::new("servers", "Server Explorer"),
59        document("order.rs").unsaved(true),
60        document("customer.rs"),
61        document("main.rs"),
62        DockPanel::new("solution", "Solution Explorer").icon(IconName::Layers),
63        DockPanel::new("properties", "Properties").icon(IconName::Settings),
64        DockPanel::new("errors", "Error List").icon(IconName::CircleAlert),
65        DockPanel::new("output", "Output"),
66        DockPanel::new("stack", "Call Stack"),
67        DockPanel::new("find", "Find and Replace").icon(IconName::Search),
68    ]
69}

71/// The IDE, as a whole window.
72pub fn ide() -> impl View {
73    let layout = State::new(start());
74    let sources: HashMap<&'static str, State<String>> =
75        [("order.rs", ORDER), ("customer.rs", CUSTOMER), ("main.rs", STARTUP)]
76            .map(|(id, text)| (id, State::new(text.to_owned())))
77            .into();
78    let (find, replace) = (State::new("total".to_owned()), State::new(String::new()));
79    let content = move |id: &str| -> AnyView {
80        match id {
81            "toolbox" => Box::new(
82                VStack::new((Text::new("Button"), Text::new("Text field"), Text::new("Data grid")))
83                    .spacing(8.0)
84                    .padding(12.0),
85            ),
86            "servers" => Box::new(Text::new("Data connections").padding(12.0)),
87            "solution" => Box::new(
88                TreeView::new(vec![
89                    TreeNode::new("shop", "Solution 'Shop' (1 project)").child(
90                        TreeNode::new("project", "Shop")
91                            .icon(IconName::Package)
92                            .child(TreeNode::new("order.rs", "order.rs").icon(IconName::Code))
93                            .child(TreeNode::new("customer.rs", "customer.rs").icon(IconName::Code))
94                            .child(TreeNode::new("main.rs", "main.rs").icon(IconName::Code)),
95                    ),
96                ])
97                .padding(8.0),
98            ),
99            "properties" => Box::new(
100                VStack::new((
101                    PropertyRow::new("File name", "order.rs"),
102                    PropertyRow::new("Build action", "Compile"),
103                    PropertyRow::new("Encoding", "UTF-8"),
104                    PropertyRow::new("Full path", "src/order.rs"),
105                ))
106                .spacing(4.0)
107                .padding(12.0),
108            ),
109            "errors" => Box::new(
110                VStack::new((
111                    Text::new("E0425 · cannot find type `Line` in this scope · order.rs · line 7"),
112                    Text::new("W0001 · unused import · customer.rs · line 1"),
113                ))
114                .spacing(6.0)
115                .padding(12.0),
116            ),
117            "output" => Box::new(console(OUTPUT)),
118            "stack" => Box::new(Text::new("Not debugging").padding(12.0)),
119            "find" => Box::new(
120                VStack::new((
121                    TextField::new(find.binding()).placeholder("Find"),
122                    TextField::new(replace.binding()).placeholder("Replace"),
123                    HStack::new((Button::new("Find next"), Button::new("Replace all"))).spacing(8.0),
124                ))
125                .spacing(8.0)
126                .padding(12.0),
127            ),
128            document => match sources.get(document) {
129                Some(source) => {
130                    Box::new(CodeEditor::new(source.binding()).language(CodeLanguage::Rust).fill())
131                }
132                None => Box::new(Text::new(document.to_owned())),
133            },
134        }
135    };
136    let menus = Menubar::new()
137        .menu("File")
138        .item(MenuItem::new("New project"))
139        .item(MenuItem::new("Save all").shortcut("Ctrl+Shift+S"))
140        .menu("Edit")
141        .item(MenuItem::new("Undo").shortcut("Ctrl+Z"))
142        .menu("View")
143        .menu("Project")
144        .menu("Build")
145        .item(MenuItem::new("Build solution").shortcut("Ctrl+Shift+B"))
146        .menu("Debug")
147        .menu("Tools")
148        .menu("Window")
149        .menu("Help");
150    let tools = Toolbar::new()
151        .action(ToolbarAction::new("New").icon(IconName::Plus))
152        .action(ToolbarAction::new("Save").icon(IconName::Save))
153        .separator()
154        .action(ToolbarAction::new("Undo").icon(IconName::Undo2))
155        .action(ToolbarAction::new("Redo").icon(IconName::Redo2))
156        .separator()
157        .action(ToolbarAction::new("Start").icon(IconName::Play));
158    let area = DockArea::new(layout.binding(), panels(), content).label("Workbench");
159    let status = StatusBar::new()
160        .left((StatusBar::text("Ready"),))
161        .right((StatusBar::text("Ln 10, Col 5"), StatusBar::text("INS")));
162    VStack::new((menus, tools, area, status)).spacing(0.0).grow(1.0).align(Align::Stretch)
163}
