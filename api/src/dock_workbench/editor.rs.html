1//! A code editor: a rail of views at the left switches the side bar (files, search, source
2//! control), the code is in the middle split in two, the terminal, the problems and the output are
3//! below, and the outline is hidden at the right edge, a tab on its rail.
4

52/// Where everything starts: the side bar at the left, the code split in two, the panels below, and
53/// the outline hidden at the right edge.
54fn start() -> DockLayout {
55    let mut layout = DockLayout::new()
56        .center(DockNode::horizontal([DockNode::tabs(["main.rs", "lib.rs"]), DockNode::tabs(["sales.rs"])]))
57        .left(DockNode::tabs(SIDE), 260.0)
58        .bottom(DockNode::tabs(["terminal", "problems", "output"]), 210.0)
59        .right(DockNode::tabs(["outline"]), 240.0);
60    layout.auto_hide("outline");
61    layout
62}
63
64fn panels() -> Vec<DockPanel> {
65    let document = |id: &str| DockPanel::new(id, id).kind(DockPanelKind::Document).icon(IconName::FileText);
66    vec![
67        DockPanel::new("files", "Explorer").icon(IconName::Folder).closable(false),
68        DockPanel::new("search", "Search").icon(IconName::Search).closable(false),
69        DockPanel::new("git", "Source control").icon(IconName::GitBranch).closable(false),
70        document("main.rs"),
71        document("lib.rs"),
72        document("sales.rs").unsaved(true),
73        DockPanel::new("terminal", "Terminal").icon(IconName::Terminal),
74        DockPanel::new("problems", "Problems").icon(IconName::TriangleAlert),
75        DockPanel::new("output", "Output"),
76        DockPanel::new("outline", "Outline").icon(IconName::Layers),
77    ]
78}

80/// The files of the project.

82    TreeView::new(vec![
83        TreeNode::new("src", "src")
84            .icon(IconName::Folder)
85            .child(TreeNode::new("main.rs", "main.rs").icon(IconName::FileText))

93/// The code editor, as a whole window.
94pub fn editor() -> impl View {
95    let layout = State::new(start());
96    let sources: HashMap<&'static str, State<String>> =
97        [("main.rs", MAIN), ("sales.rs", SALES), ("lib.rs", LIB)]
98            .map(|(id, text)| (id, State::new(text.to_owned())))
99            .into();
100    let query = State::new("total".to_owned());
101    let content = move |id: &str| -> AnyView {
102        match id {
103            "files" => Box::new(files().padding(8.0)),
104            "search" => Box::new(
105                VStack::new((
106                    TextField::new(query.binding()).placeholder("Search"),
107                    Text::new("2 results in 1 file").style(TextStyle::Caption),
108                    Text::new("sales.rs · pub fn total(&self) -> f64"),
109                    Text::new("main.rs · println!(\"{}\", order.total());"),
110                ))
111                .spacing(8.0)
112                .padding(12.0),
113            ),
114            "git" => Box::new(
115                VStack::new((
116                    Text::new("Changes").style(TextStyle::Caption),
117                    Text::new("M  sales.rs"),
118                    Button::new("Commit").style(ButtonStyle::Primary),
119                ))
120                .spacing(8.0)
121                .padding(12.0),
122            ),
123            "terminal" => Box::new(console(TERMINAL)),
124            "output" => Box::new(console(OUTPUT)),
125            "problems" => Box::new(
126                VStack::new((
127                    Text::new("sales.rs:14 · unused variable `customer`"),
128                    Text::new("main.rs:6 · this value is never read"),
129                ))
130                .spacing(6.0)
131                .padding(12.0),
132            ),
133            "outline" => Box::new(
134                TreeView::new(vec![
135                    TreeNode::new("order", "struct Order")
136                        .child(TreeNode::new("new", "fn new"))
137                        .child(TreeNode::new("total", "fn total")),
138                ])
139                .padding(8.0),
140            ),
141            document => match sources.get(document) {
142                Some(source) => {
143                    Box::new(CodeEditor::new(source.binding()).language(CodeLanguage::Rust).fill())
144                }
145                None => Box::new(Text::new(document.to_owned())),
146            },
147        }
148    };
149    // The rail shows which view of the side bar is in front, and brings another.
150    let (reads, writes) = (layout.clone(), layout.clone());
151    let side = Binding::new(
152        move || {
153            let now = reads.get();
154            SIDE.iter().position(|id| now.is_active(id)).unwrap_or(0)
155        },
156        move |index: usize| {
157            if let Some(id) = SIDE.get(index) {
158                writes.update(|now| {
159                    now.activate(id);
160                });
161            }
162        },
163    );
164    let rail = ToolRail::new(
165        side,
166        [
167            (IconName::Folder, "Explorer"),
168            (IconName::Search, "Search"),
169            (IconName::GitBranch, "Source control"),
170        ],
171    )
172    .icons_only(true);
173    let area = DockArea::new(layout.binding(), panels(), content).label("Workbench");
174    let status = StatusBar::new()
175        .left((StatusBar::action("main"), StatusBar::text("0 errors, 2 warnings")))
176        .right((StatusBar::text("Ln 14, Col 9"), StatusBar::text("UTF-8"), StatusBar::text("Rust")));
177    VStack::new((
178        TitleBar::new("sales.rs — shop"),
179        HStack::new((rail, area)).spacing(0.0).grow(1.0).align(Align::Stretch),
180        status,
181    ))
182    .spacing(0.0)
183    .grow(1.0)
184    .align(Align::Stretch)
185}
