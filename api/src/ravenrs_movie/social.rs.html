13const GAP: f32 = 24.0;

15/// One moment of a film laid out for a tall or square screen, such as the videos of
16/// social networks: the chapter on top, the film in the middle and, under it, the
17/// title and words of what it explains, so it reads with the sound off.
18///
19/// The film takes the height left and is cut to a window that follows the spot it
20/// explains, or the pointer, so it stays big enough to read on a phone.
21#[derive(Debug)]
22#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
23pub struct SocialFrame {
24    movie: Movie,
25    at_ms: u64,
26    size: Size,
27    brand: String,
28}
29
30#[cfg_attr(feature = "live", ravenrs_live::expose)]
31impl SocialFrame {
32    /// The moment `at_ms` of `movie` on a screen of `size` logical pixels.
33    pub fn new(movie: Movie, at_ms: u64, size: Size) -> Self {
34        Self { movie, at_ms, size, brand: String::new() }
35    }
36
37    /// A name shown at the foot of every frame, such as the product's.
38    pub fn brand(mut self, brand: impl Into<String>) -> Self {
39        self.brand = brand.into();
40        self
41    }
42}

44/// What the window over the film follows at `at`: the spot it explains, or else the
45/// pointer, or else the middle, in logical pixels of the film.
46fn focus(movie: &Movie, at: u64) -> (f32, f32) {
47    let logical = movie.logical_size();
48    if let Some(spot) = shown_at(movie, at).and_then(|i| movie.hotspots().get(i))
49        && !spot.is_general()
50    {
51        return (spot.x + spot.width / 2.0, spot.y + spot.height / 2.0);
52    }
53    let pointer = match movie.project().place_at(at) {
54        Some(Place::Film(ms)) => movie.pointer_at(ms),
55        _ => None,
56    };
57    pointer.map_or((logical.width / 2.0, logical.height / 2.0), |p| (p.x, p.y))
58}

60/// The words under the film take about this much, and the headline and the name at
61/// the foot about this.
62const WORDS_HEIGHT: f32 = 190.0;
63const HEADLINE_HEIGHT: f32 = 44.0;
64const BRAND_HEIGHT: f32 = 24.0;
65
66impl View for SocialFrame {
67    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
68        let theme = *cx.theme();
69        let at = self.at_ms;
70        let padding = (self.size.width * 0.06).round();
71        let width = self.size.width - padding * 2.0;
72        let logical = self.movie.logical_size();
73        let chapter = self.movie.chapter_at(at).and_then(|i| self.movie.chapters().get(i));
74        let spot = shown_at(&self.movie, at).and_then(|i| self.movie.hotspots().get(i));
75        let headline = chapter.map(|chapter| {
76            Text::new(chapter.title.clone()).style(TextStyle::Title).font_weight(Weight::BOLD)
77        });
78        // The film fills the height that is left, cut to a window that follows what
79        // it explains: a whole wide screen would be too small to read on a phone.
80        let natural = width * logical.height / logical.width.max(1.0);
81        let room =
82            self.size.height - padding * 2.0 - HEADLINE_HEIGHT - WORDS_HEIGHT - BRAND_HEIGHT - GAP * 4.0;
83        let window = room.clamp(natural, width * 1.25);
84        let film_width = window * logical.width / logical.height.max(1.0);
85        let scale = film_width / logical.width.max(1.0);
86        let (fx, fy) = focus(&self.movie, at);
87        let x = (fx * scale - width / 2.0).clamp(0.0, (film_width - width).max(0.0));
88        let y = (fy * scale - window / 2.0)
89            .clamp(0.0, (film_width * logical.height / logical.width.max(1.0) - window).max(0.0));
90        let film = MovieFrame::new(self.movie.clone(), at).width(film_width).without_boxes();
91        let film = cx.child(&film);
92        let shifted = cx.container(
93            LayoutStyle { absolute: Some((-x, -y)), ..LayoutStyle::stack(Axis::Vertical) },
94            &[film],
95            Paint::None,
96        );
97        let windowed = cx.container(
98            LayoutStyle {
99                width: Some(width),
100                height: Some(window),
101                overflow_hidden: true,
102                ..LayoutStyle::stack(Axis::Vertical)
103            },
104            &[shifted],
105            Paint::Quad(QuadPaint {
106                corner_radius: theme.radii.lg,
107                border: Some(Border { width: 1.0, color: theme.color(Semantic::Border) }),
108                ..QuadPaint::new(self.movie.background())
109            }),
110        );
111        cx.clip(windowed);
112        // A step the film would wait at says so, since nobody clicks it here.
113        let words = spot.map(|spot| {
114            let eyebrow = (spot.kind == HotspotKind::Stop).then(|| {
115                Text::new(chapter.map_or_else(String::new, |c| c.title.clone()))
116                    .style(TextStyle::Caption)
117                    .foreground(Semantic::Accent)
118            });
119            VStack::new((
120                eyebrow,
121                Text::new(spot.title.clone()).style(TextStyle::Headline).font_weight(Weight::SEMIBOLD),
122                Text::new(spot.text.clone()).style(TextStyle::Body).foreground(Semantic::InkMuted),
123            ))
124            .spacing(8.0)
125            .align(Align::Start)
126            .padding(20.0)
127            .background(theme.color(Semantic::SurfaceElevated))
128            .corner_radius(theme.radii.lg)
129            .border(1.0, theme.color(Semantic::Border))
130        });
131        let brand = (!self.brand.is_empty())
132            .then(|| Text::new(self.brand.clone()).style(TextStyle::Callout).foreground(Semantic::Accent));
133        // What is left over goes above the headline and above the name, so the
134        // middle stays in the middle and the name at the foot.
135        let grow_to_end = |cx: &mut BuildCx<'_>, child: Option<NodeId>| {
136            cx.container(
137                LayoutStyle {
138                    grow: 1.0,
139                    justify: Justify::End,
140                    align: Align::Start,
141                    ..LayoutStyle::stack(Axis::Vertical)
142                },
143                &child.into_iter().collect::<Vec<_>>(),
144                Paint::None,
145            )
146        };
147        let headline = headline.map(|headline| cx.child(&headline));
148        let top = grow_to_end(cx, headline);
149        let words = words.map(|words| cx.child(&words));
150        let brand = brand.map(|brand| cx.child(&brand));
151        let bottom = grow_to_end(cx, brand);
152        let column = cx.container(
153            LayoutStyle {
154                spacing: GAP,
155                grow: 1.0,
156                align: Align::Stretch,
157                ..LayoutStyle::stack(Axis::Vertical)
158            },
159            &[Some(top), Some(windowed), words, Some(bottom)].into_iter().flatten().collect::<Vec<_>>(),
160            Paint::None,
161        );
162        let node = cx.container(
163            LayoutStyle {
164                width: Some(self.size.width),
165                height: Some(self.size.height),
166                padding: padding.into(),
167                align: Align::Stretch,
168                overflow_hidden: true,
169                ..LayoutStyle::stack(Axis::Vertical)
170            },
171            &[column],
172            Paint::Quad(QuadPaint::new(theme.color(Semantic::Canvas))),
173        );
174        cx.semantics(node, Semantics::new(Role::Image));
175        cx.clip(node);
176        node
177    }
178}

180#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part1"))]
181impl Movie {
182    /// The same film with the camera brought `zoom` times closer to each spot it
183    /// explains, for as long as it explains it. Spots about the whole screen, and
184    /// moments the edits already zoom, stay as they are.
185    pub fn following(&self, zoom: f32) -> Movie {
186        let logical = self.logical_size();
187        let zoom = zoom.max(1.0);
188        let (shown_width, shown_height) = (logical.width / zoom, logical.height / zoom);
189        let mut project = self.project().clone();
190        let zoomed = |project: &crate::project::Project, from: u64, until: u64| {
191            project.overlays.iter().any(|overlay| {
192                matches!(overlay.kind, OverlayKind::Zoom { .. })
193                    && overlay.start_ms < until
194                    && from < overlay.end_ms()
195            })
196        };
197        for spot in self.hotspots().iter().filter(|spot| !spot.is_general()) {
198            if spot.until_ms <= spot.from_ms || zoomed(self.project(), spot.from_ms, spot.until_ms) {
199                continue;
200            }
201            let (width, height) = (shown_width.max(spot.width), shown_height.max(spot.height));
202            let x = (spot.x + spot.width / 2.0 - width / 2.0).clamp(0.0, (logical.width - width).max(0.0));
203            let y =
204                (spot.y + spot.height / 2.0 - height / 2.0).clamp(0.0, (logical.height - height).max(0.0));
205            project.add_overlay(
206                OverlayKind::Zoom { region: Region::new(x, y, width, height) },
207                spot.from_ms,
208                spot.until_ms - spot.from_ms,
209            );
210        }
211        self.with_project(project)
212    }
213}
