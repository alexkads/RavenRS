2//! already hold. No model and no network: it works the same in the browser.

10/// How much a word counts in each part of a moment.
11const TITLE_WEIGHT: f32 = 2.0;
12const TEXT_WEIGHT: f32 = 1.0;
13const CHAPTER_WEIGHT: f32 = 0.5;
14/// The usual BM25 constants: how fast repeating a word stops helping, and how
15/// much a long text is discounted.
16const SATURATION: f32 = 1.2;
17const LENGTH_DISCOUNT: f32 = 0.75;
18/// A chapter begins the lesson, so it goes ahead of a step that says the same.
19const CHAPTER_BONUS: f32 = 1.25;
20/// How much of the shorter word a shared start has to cover: "compra" is not
21/// "computador".
22const MOST_OF_THE_WORD: f32 = 0.7;
23/// An answer far weaker than the best one is noise.
24const FAR_BELOW_THE_BEST: f32 = 0.25;
25/// Words that say nothing about the subject, folded, in Portuguese and English.
26const STOP_WORDS: &[&str] = &[
27    "a", "ao", "aos", "as", "com", "como", "da", "das", "de", "do", "dos", "e", "em", "eu", "faco", "fazer",
28    "foi", "ha", "isso", "mais", "meu", "meus", "minha", "minhas", "na", "nas", "no", "nos", "o", "onde",
29    "os", "ou", "outra", "outras", "outro", "outros", "para", "pelo", "pela", "por", "posso", "pra", "qual",
30    "quando", "que", "quero", "se", "sem", "seu", "sua", "um", "uma", "uns", "umas", "an", "and", "are",
31    "can", "does", "for", "how", "i", "in", "is", "it", "my", "of", "on", "or", "the", "to", "what", "where",
32    "with",
33];

35/// What a moment of a film is.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37#[non_exhaustive]
38pub enum MomentKind {
39    /// The start of a chapter.
40    Chapter,
41    /// A step that waits for a click.
42    Step,
43    /// A note on the screen.
44    Note,
45}
46
47/// A place in a film that can be asked for.
48#[derive(Clone, Debug, PartialEq, Eq)]
49pub struct Moment {
50    /// The film, by its position in the library.
51    pub film: usize,
52    /// When, in milliseconds of the film's timeline.
53    pub at_ms: u64,
54    /// What it is.
55    pub kind: MomentKind,
56    /// Its title.
57    pub title: String,
58    /// What it says.
59    pub text: String,
60}
61
62/// A moment that answers a question.
63#[derive(Clone, Debug, PartialEq)]
64pub struct Answer {
65    /// The moment.
66    pub moment: Moment,
67    /// How well it answers, comparable only within one search.
68    pub score: f32,
69}
70
71/// A moment with its text ready to compare.
72#[derive(Debug)]
73struct Indexed {
74    moment: Moment,
75    title: Vec<String>,
76    text: Vec<String>,
77    chapter: Vec<String>,
78}
79
80impl Indexed {
81    fn length(&self) -> usize {
82        self.title.len() + self.text.len()
83    }
84
85    /// How much `word` appears here, counting near words for less.
86    fn presence(&self, word: &str) -> f32 {
87        let in_field = |words: &[String]| words.iter().map(|other| likeness(word, other)).sum::<f32>();
88        in_field(&self.title) * TITLE_WEIGHT
89            + in_field(&self.text) * TEXT_WEIGHT
90            + in_field(&self.chapter) * CHAPTER_WEIGHT
91    }
92}

94#[derive(Debug)]
95struct Library {
96    films: Vec<(String, Movie)>,
97    moments: Vec<Indexed>,
98    average_length: f32,
99}

101/// The films of an app, ready to be asked about. Cloning is cheap.
102///
103/// ```
104/// use ravenrs_core::{Color, Size};
105/// use ravenrs_movie::{FilmBuilder, MovieLibrary};
106///
107/// let mut film = FilmBuilder::new(Size::new(320.0, 200.0), Color::WHITE);
108/// film.frame(0, &[]);
109/// film.chapter("Pagar uma conta", "Como dar baixa no que venceu.", 0);
110/// let library = MovieLibrary::new().film("Manual", film.finish(1000)?);
111/// assert_eq!(library.ask("como pago as contas?", 3)[0].moment.title, "Pagar uma conta");
112/// # Ok::<(), ravenrs_movie::MovieError>(())
113/// ```
114#[derive(Clone, Debug)]
115pub struct MovieLibrary {
116    inner: Rc<Library>,
117}
118
119impl MovieLibrary {
120    /// A library with no films.
121    pub fn new() -> Self {
122        Self { inner: Rc::new(Library { films: Vec::new(), moments: Vec::new(), average_length: 0.0 }) }
123    }
124
125    /// The library with one more film. Build it once: this reads all the text.
126    #[must_use]
127    pub fn film(self, title: impl Into<String>, movie: Movie) -> Self {
128        let mut films: Vec<(String, Movie)> = self.inner.films.clone();
129        films.push((title.into(), movie));
130        let moments: Vec<Indexed> =
131            films.iter().enumerate().flat_map(|(i, (_, movie))| moments_of(i, movie)).collect();
132        let total: usize = moments.iter().map(Indexed::length).sum();
133        let average_length = total as f32 / moments.len().max(1) as f32;
134        Self { inner: Rc::new(Library { films, moments, average_length }) }
135    }
136
137    /// The film at `index`, with its title.
138    pub fn get(&self, index: usize) -> Option<(&str, &Movie)> {
139        self.inner.films.get(index).map(|(title, movie)| (title.as_str(), movie))
140    }
141
142    /// How many films it has.
143    pub fn len(&self) -> usize {
144        self.inner.films.len()
145    }
146
147    /// Whether it has no films.
148    pub fn is_empty(&self) -> bool {
149        self.inner.films.is_empty()
150    }
151
152    /// The moments that best answer `question`, the best first, at most `limit`.
153    /// Accents and case do not matter, and near forms of a word count for less:
154    /// "pago" finds "pagar". With no word in common, there is no answer.
155    pub fn ask(&self, question: &str, limit: usize) -> Vec<Answer> {
156        let library = &self.inner;
157        let words = question_words(question);
158        if words.is_empty() {
159            return Vec::new();
160        }
161        let total = library.moments.len() as f32;
162        let mut scores = vec![(0.0_f32, 0_usize); library.moments.len()];
163        for word in &words {
164            let presences: Vec<f32> = library.moments.iter().map(|m| m.presence(word)).collect();
165            let having = presences.iter().filter(|p| **p > 0.0).count() as f32;
166            let rarity = (1.0 + (total - having + 0.5) / (having + 0.5)).ln();
167            for ((score, matched), (presence, moment)) in
168                scores.iter_mut().zip(presences.iter().zip(&library.moments))
169            {
170                if *presence > 0.0 {
171                    let length = moment.length() as f32 / library.average_length.max(1.0);
172                    let discount = 1.0 - LENGTH_DISCOUNT + LENGTH_DISCOUNT * length;
173                    *score += rarity * presence * (SATURATION + 1.0) / (presence + SATURATION * discount);
174                    *matched += 1;
175                }
176            }
177        }
178        let mut answers: Vec<Answer> = library
179            .moments
180            .iter()
181            .zip(scores)
182            .filter(|(_, (score, _))| *score > 0.0)
183            .map(|(indexed, (score, matched))| Answer {
184                moment: indexed.moment.clone(),
185                // Answering every word beats answering one word very well.
186                score: score * matched as f32 / words.len() as f32
187                    * if indexed.moment.kind == MomentKind::Chapter { CHAPTER_BONUS } else { 1.0 },
188            })
189            .collect();
190        // On a tie the chapter wins, and then the earlier.
191        answers.sort_by(|a, b| {
192            b.score
193                .total_cmp(&a.score)
194                .then_with(|| rank(a.moment.kind).cmp(&rank(b.moment.kind)))
195                .then_with(|| (a.moment.film, a.moment.at_ms).cmp(&(b.moment.film, b.moment.at_ms)))
196        });
197        let best = answers.first().map_or(0.0, |answer| answer.score);
198        answers.retain(|answer| answer.score >= best * FAR_BELOW_THE_BEST);
199        answers.truncate(limit);
200        answers
201    }
202}

210fn rank(kind: MomentKind) -> u8 {
211    match kind {
212        MomentKind::Chapter => 0,
213        MomentKind::Step => 1,
214        MomentKind::Note => 2,
215    }
216}

218fn moments_of(film: usize, movie: &Movie) -> Vec<Indexed> {
219    let chapters = movie.chapters().iter().map(|chapter| {
220        let moment = Moment {
221            film,
222            at_ms: chapter.at_ms,
223            kind: MomentKind::Chapter,
224            title: chapter.title.clone(),
225            text: chapter.text.clone(),
226        };
227        indexed(moment, Vec::new())
228    });
229    // A box about the whole screen says what its chapter says.
230    let spots = movie.hotspots().iter().filter(|spot| !spot.is_general()).map(|spot| {
231        let moment = Moment {
232            film,
233            at_ms: spot.from_ms,
234            kind: if spot.kind == HotspotKind::Stop { MomentKind::Step } else { MomentKind::Note },
235            title: spot.title.clone(),
236            text: spot.text.clone(),
237        };
238        let chapter = movie.chapter_at(spot.from_ms).and_then(|i| movie.chapters().get(i));
239        indexed(moment, chapter.map_or_else(Vec::new, |chapter| words_of(&chapter.title)))
240    });
241    chapters.chain(spots).collect()
242}
243
244fn indexed(moment: Moment, chapter: Vec<String>) -> Indexed {
245    let (title, text) = (words_of(&moment.title), words_of(&moment.text));
246    Indexed { moment, title, text, chapter }
247}

249/// The folded words of a text, leaving out the ones that say nothing.
250fn words_of(text: &str) -> Vec<String> {
251    all_words(text).into_iter().filter(|word| !STOP_WORDS.contains(&word.as_str())).collect()
252}
253
254/// The words of a question. A question made only of small words still asks.
255fn question_words(question: &str) -> Vec<String> {
256    let words = words_of(question);
257    if words.is_empty() { all_words(question) } else { words }
258}
259
260fn all_words(text: &str) -> Vec<String> {
261    let folded: String = text.chars().flat_map(fold).collect();
262    folded.split(|c: char| !c.is_alphanumeric()).filter(|w| !w.is_empty()).map(str::to_owned).collect()
263}
264
265/// How close two folded words are, from 0 to 1.
266fn likeness(a: &str, b: &str) -> f32 {
267    if a == b {
268        return 1.0;
269    }
270    let (a, b): (Vec<char>, Vec<char>) = (a.chars().collect(), b.chars().collect());
271    let (shorter, longer) = (a.len().min(b.len()), a.len().max(b.len()));
272    let singular = |word: &[char]| -> usize { word.len() - usize::from(word.last() == Some(&'s')) };
273    if a[..singular(&a)] == b[..singular(&b)] {
274        return 0.95;
275    }
276    if shorter >= 5 && one_edit_apart(&a, &b) {
277        return 0.7;
278    }
279    if root(&a) == root(&b) {
280        return 0.6;
281    }
282    // A long shared start: "pagar" and "pagamento", or a word still being typed.
283    let common = a.iter().zip(&b).take_while(|(x, y)| x == y).count();
284    if common >= 4 && common as f32 >= shorter as f32 * MOST_OF_THE_WORD {
285        0.3 + 0.5 * common as f32 / longer as f32
286    } else {
287        0.0
288    }
289}

291/// A word without the plural and the usual endings of verbs, so "pago", "pague"
292/// and "pagar" meet. This is not a real stemmer: it only has to bring near forms
293/// together, and a root under three letters is left alone.
294fn root(word: &[char]) -> &[char] {
295    let word = word.strip_suffix(&['s']).unwrap_or(word);
296    let ending = [
297        &['i', 'n', 'g'][..],
298        &['a', 'r'],
299        &['e', 'r'],
300        &['i', 'r'],
301        &['o', 'u'],
302        &['e', 'i'],
303        &['u', 'e'],
304        &['e', 'd'],
305        &['a'],
306        &['e'],
307        &['i'],
308        &['o'],
309    ]
310    .into_iter()
311    .find(|ending| word.ends_with(ending) && word.len() >= ending.len() + 3);
312    ending.map_or(word, |ending| &word[..word.len() - ending.len()])
313}

315/// One letter wrong, missing, extra, or two swapped: the usual slips of typing.
316fn one_edit_apart(a: &[char], b: &[char]) -> bool {
317    if a.len().abs_diff(b.len()) > 1 {
318        return false;
319    }
320    let start = a.iter().zip(b).take_while(|(x, y)| x == y).count();
321    let (rest_a, rest_b) = (&a[start..], &b[start..]);
322    let tail = |skip_a: usize, skip_b: usize| rest_a.get(skip_a..) == rest_b.get(skip_b..);
323    let swapped = rest_a.len() >= 2
324        && rest_b.len() >= 2
325        && rest_a[0] == rest_b[1]
326        && rest_a[1] == rest_b[0]
327        && tail(2, 2);
328    tail(1, 1) || tail(1, 0) || tail(0, 1) || swapped
329}

331#[cfg(test)]
332mod tests {
333    use super::{all_words, likeness, words_of};
334
335    #[test]
336    fn words_are_folded_and_small_words_go_away() {
337        assert_eq!(all_words("Lançar: uma DESPESA!"), ["lancar", "uma", "despesa"]);
338        assert_eq!(words_of("Como lançar uma despesa?"), ["lancar", "despesa"]);
339    }
340
341    #[test]
342    fn near_words_count_less_than_the_same_word() {
343        assert_eq!(likeness("conta", "conta"), 1.0);
344        assert!(likeness("conta", "contas") > 0.9);
345        assert!(likeness("despeza", "despesa") > 0.5, "one wrong letter");
346        assert!(likeness("despsea", "despesa") > 0.5, "two swapped letters");
347        assert!(likeness("pagar", "pague") > 0.0, "the same root");
348        assert!(likeness("pagar", "pagamento") > 0.0, "the same root, longer");
349        assert_eq!(likeness("carro", "conta"), 0.0);
350        assert!(likeness("salvo", "salvar") > 0.0, "the same root");
351        assert!(likeness("saving", "save") > 0.0, "the same root");
352        assert!(likeness("lanc", "lancamentos") > 0.0, "still being typed");
353        assert_eq!(likeness("saldo", "salvar"), 0.0, "sharing a start is not sharing a meaning");
354        assert_eq!(likeness("saldo", "salve"), 0.0);
355        assert_eq!(likeness("compra", "computador"), 0.0);
356    }
