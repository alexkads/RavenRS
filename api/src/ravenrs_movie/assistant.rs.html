17/// How many answers it offers: more than this and it is a list to read, not an answer.
18const ANSWERS: usize = 4;
19const SPACING: f32 = 12.0;
20/// Below this width the names of the films do not fit side by side.
21const TABS_FROM: f32 = 860.0;
22const PICKER_WIDTH: f32 = 280.0;
23/// What goes around the film: the question, the choice of film and the controls.
24const AROUND_THE_FILM: f32 = 184.0;

26/// The words a [`MovieAssistant`] shows. English by default.
27#[derive(Clone, Debug)]
28#[non_exhaustive]
29pub struct AssistantLabels {
30    /// Inside the empty question field.
31    pub placeholder: String,
32    /// When no film answers the question.
33    pub nothing_found: String,
34    /// The name of the choice of film, for screen readers.
35    pub film: String,
36    /// The player's own words.
37    pub player: MovieLabels,
38}

40impl Default for AssistantLabels {
41    fn default() -> Self {
42        Self {
43            placeholder: "Ask how to do something".into(),
44            nothing_found: "No film talks about that. Try other words.".into(),
45            film: "Film".into(),
46            player: MovieLabels::default(),

48    }
49}

63/// A question field over the films of a [`MovieLibrary`]. Each answer is a moment
64/// of a film, and choosing one opens the film right there.
65///
66/// It looks only at the words the films hold, on this machine, with no model.
67///
68/// ```
69/// use ravenrs_movie::{MovieAssistant, MovieLibrary};
70///
71/// let assistant = MovieAssistant::new(MovieLibrary::new()).width(960.0);
72/// ```
73#[derive(Debug)]
74pub struct MovieAssistant {
75    library: MovieLibrary,
76    width: f32,
77    height: Option<f32>,
78    labels: AssistantLabels,
79    cue: Option<MovieCue>,
80    narration: Option<Narration>,
81}
82
83impl MovieAssistant {
84    /// An assistant over `library`, with an empty question.
85    pub fn new(library: MovieLibrary) -> Self {
86        Self {
87            library,
88            width: 960.0,
89            height: None,
90            labels: AssistantLabels::default(),
91            cue: None,
92            narration: None,
93        }
94    }
95
96    /// The width of the whole assistant, in logical pixels (960 by default).
97    pub fn width(mut self, width: f32) -> Self {
98        self.width = width;
99        self
100    }
101
102    /// The height it should fit in: the film gets smaller rather than pushing its
103    /// controls out of view. With none, the film takes the whole width.
104    pub fn height(mut self, height: f32) -> Self {
105        self.height = Some(height);
106        self
107    }
108
109    /// Recordings of what the films say, in place of the system's voice.
110    #[must_use]
111    pub fn narration(mut self, narration: Narration) -> Self {
112        self.narration = Some(narration);
113        self
114    }
115
116    /// The words it shows.
117    pub fn labels(mut self, labels: AssistantLabels) -> Self {
118        self.labels = labels;
119        self
120    }
121
122    /// Lets something outside it, such as a [`MovieChat`](crate::MovieChat), open a
123    /// moment of the film it shows.
124    #[must_use]
125    pub fn cue(mut self, cue: MovieCue) -> Self {
126        self.cue = Some(cue);
127        self
128    }
129}

131/// Without the steps inside what it also found: "Pagar uma conta" and "Pague uma
132/// conta" read as the same thing twice. Only steps are kept when nothing else came up.
133pub(crate) fn whole_answers(answers: Vec<Answer>) -> Vec<Answer> {
134    let whole: Vec<Answer> =
135        answers.iter().filter(|answer| answer.moment.kind != MomentKind::Step).cloned().collect();
136    if whole.is_empty() { answers } else { whole }
137}

139/// A row that says which moment an answer is and opens it.
140pub(crate) fn moment_row(
141    library: &MovieLibrary,
142    answer: &Answer,
143    open: impl Fn(&Moment) + 'static,
144) -> ListItem {
145    let moment = answer.moment.clone();
146    let film = library.get(moment.film).map_or("", |(title, _)| title);
147    let when = clock_label(moment.at_ms);
148    let trailing = if library.len() > 1 { format!("{film} · {when}") } else { when };
149    let chosen = moment.clone();
150    ListItem::new(moment.title.clone())
151        .subtitle(moment.text.clone())
152        .icon(IconName::Play)
153        .trailing(trailing)
154        .on_click(move || open(&chosen))
155}

157/// The same in a narrow place, such as a chat: the name and where it is, one under the other.
158pub(crate) fn short_moment_row(
159    library: &MovieLibrary,
160    answer: &Answer,
161    open: impl Fn(&Moment) + 'static,
162) -> ListItem {
163    let moment = answer.moment.clone();
164    let film = library.get(moment.film).map_or("", |(title, _)| title);
165    let when = clock_label(moment.at_ms);
166    let chosen = moment.clone();
167    ListItem::new(moment.title.clone())
168        .subtitle(if library.len() > 1 { format!("{film} · {when}") } else { when })
169        .icon(IconName::Play)
170        .on_click(move || open(&chosen))
171}

173/// What choosing an answer does. The answers then step aside, so the film is in
174/// view, until the question changes.
175#[derive(Clone)]
176struct Open {
177    film: State<usize>,
178    question: State<String>,
179    answered: State<Option<String>>,
180    cue: MovieCue,
181}
182
183impl Open {
184    fn moment(&self, moment: &Moment) {
185        self.film.set(moment.film);
186        self.answered.set(Some(self.question.get()));
187        self.cue.go_to(moment.at_ms);
188    }
189}

191impl View for MovieAssistant {
192    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
193        let question: State<String> = cx.use_state(String::new);
194        let film: State<usize> = cx.use_state(|| 0);
195        let answered: State<Option<String>> = cx.use_state(|| None);
196        let finished: State<BTreeSet<usize>> = cx.use_state(BTreeSet::new);
197        let own_cue = cx.use_state(MovieCue::new).get();
198        let cue = self.cue.clone().unwrap_or(own_cue);
199        if let Some(asked) = cue.take_film() {
200            film.set(asked);
201        }
202        let open = Open {
203            film: film.clone(),
204            question: question.clone(),
205            answered: answered.clone(),
206            cue: cue.clone(),
207        };
208
209        let asked = question.get();
210        let waiting = answered.get().as_ref() != Some(&asked);
211        let answers = if waiting { whole_answers(self.library.ask(&asked, ANSWERS)) } else { Vec::new() };
212        let (library, first) = (self.library.clone(), open.clone());
213        let field = TextField::new(question.binding())
214            .placeholder(self.labels.placeholder.clone())
215            .icon(IconName::Search)
216            .on_submit(move |text| {
217                if let Some(best) = library.ask(&text, 1).first() {
218                    first.moment(&best.moment);
219                }
220            });
221        let rows: Vec<ListItem> = answers
222            .iter()
223            .map(|answer| {
224                let open = open.clone();
225                moment_row(&self.library, answer, move |moment| open.moment(moment))
226            })
227            .collect();
228        let nothing = (waiting && !asked.trim().is_empty() && rows.is_empty()).then(|| {
229            Text::new(self.labels.nothing_found.clone())
230                .style(TextStyle::Callout)
231                .foreground(Semantic::InkMuted)
232        });
233        let field = cx.child(&field.width(self.width));
234        // The answers open over the film, which stays where it is.
235        if !rows.is_empty() || nothing.is_some() {
236            let answers = Card::new(
237                VStack::new((VStack::new(rows).spacing(2.0).align(Align::Stretch), nothing))
238                    .align(Align::Stretch),
239            )
240            .padding(8.0)
241            .width(self.width);
242            let (close, asked) = (answered.clone(), asked.clone());
243            let dismiss = std::rc::Rc::new(move || close.set(Some(asked.clone())));
244            cx.overlay(field, Placement::Below, &answers, Some(dismiss));
245        }
246        let mut children = vec![field];
247
248        if self.library.len() > 1 {
249            // What was already watched is marked, to show how far someone has come.
250            let seen = finished.get();
251            let titles: Vec<String> = (0..self.library.len())
252                .filter_map(|i| self.library.get(i).map(|f| (i, f.0)))
253                .map(|(i, title)| if seen.contains(&i) { format!("✓ {title}") } else { title.to_owned() })
254                .collect();
255            children.push(if self.width >= TABS_FROM {
256                cx.child(&SegmentedControl::new(film.binding(), titles).label(self.labels.film.clone()))
257            } else {
258                cx.child(
259                    &Select::new(film.binding(), titles)
260                        .label(self.labels.film.clone())
261                        .width(PICKER_WIDTH.min(self.width)),
262                )
263            });
264        }
265        let shown = film.get();
266        if let Some((_, movie)) = self.library.get(shown) {
267            let mut player =
268                MoviePlayer::new(movie.clone()).width(self.width).labels(self.labels.player.clone()).cue(cue);
269            if let Some(height) = self.height {
270                player = player.max_film_height(height - AROUND_THE_FILM);
271            }
272            if let Some(narration) = &self.narration {
273                player = player.narration(narration.clone());
274            }
275            let mark = finished.clone();
276            player = player.on_finished(move || {
277                mark.update(|seen| {
278                    seen.insert(shown);
279                });
280            });
281            if let Some((title, _)) = self.library.get(shown + 1) {
282                let go = film.clone();
283                player = player.next(title, move || go.set(shown + 1));
284            }
285            // Each film keeps its own place.
286            children.push(cx.keyed_child(shown as u64, &player));
287        }
288        cx.container(
289            LayoutStyle { spacing: SPACING, ..LayoutStyle::stack(Axis::Vertical) },
290            &children,
291            Paint::None,
292        )
293    }
