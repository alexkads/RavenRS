14/// How many answers it offers: more than this and it is a list to read, not an answer.
15const ANSWERS: usize = 4;
16const SPACING: f32 = 12.0;

18/// The words a [`MovieAssistant`] shows. English by default.
19#[derive(Clone, Debug)]
20#[non_exhaustive]
21pub struct AssistantLabels {
22    /// Inside the empty question field.
23    pub placeholder: String,
24    /// When no film answers the question.
25    pub nothing_found: String,
26    /// The player's own words.
27    pub player: MovieLabels,
28}
29
30impl Default for AssistantLabels {
31    fn default() -> Self {
32        Self {
33            placeholder: "Ask how to do something".into(),
34            nothing_found: "No film talks about that. Try other words.".into(),
35            player: MovieLabels::default(),
36        }
37    }
38}

51/// A question field over the films of a [`MovieLibrary`]. Each answer is a moment
52/// of a film, and choosing one opens the film right there.
53///
54/// It looks only at the words the films hold, on this machine, with no model.
55///
56/// ```
57/// use ravenrs_movie::{MovieAssistant, MovieLibrary};
58///
59/// let assistant = MovieAssistant::new(MovieLibrary::new()).width(960.0);
60/// ```
61#[derive(Debug)]
62pub struct MovieAssistant {
63    library: MovieLibrary,
64    width: f32,
65    labels: AssistantLabels,
66    cue: Option<MovieCue>,
67}
68
69impl MovieAssistant {
70    /// An assistant over `library`, with an empty question.
71    pub fn new(library: MovieLibrary) -> Self {
72        Self { library, width: 960.0, labels: AssistantLabels::default(), cue: None }
73    }
74
75    /// The width of the whole assistant, in logical pixels (960 by default).
76    pub fn width(mut self, width: f32) -> Self {
77        self.width = width;
78        self
79    }
80
81    /// The words it shows.
82    pub fn labels(mut self, labels: AssistantLabels) -> Self {
83        self.labels = labels;
84        self
85    }
86
87    /// Lets something outside it, such as a [`MovieChat`](crate::MovieChat), open a
88    /// moment of the film it shows.
89    #[must_use]
90    pub fn cue(mut self, cue: MovieCue) -> Self {
91        self.cue = Some(cue);
92        self
93    }
94}

96/// A row that says which moment an answer is and opens it.
97pub(crate) fn moment_row(
98    library: &MovieLibrary,
99    answer: &Answer,
100    open: impl Fn(&Moment) + 'static,
101) -> ListItem {
102    let moment = answer.moment.clone();
103    let film = library.get(moment.film).map_or("", |(title, _)| title);
104    let when = clock_label(moment.at_ms);
105    let trailing = if library.len() > 1 { format!("{film} · {when}") } else { when };
106    let chosen = moment.clone();
107    ListItem::new(moment.title.clone())
108        .subtitle(moment.text.clone())
109        .icon(match moment.kind {
110            MomentKind::Step => IconName::MousePointer,
111            MomentKind::Note => IconName::Info,
112            _ => IconName::Play,
113        })
114        .trailing(trailing)
115        .on_click(move || open(&chosen))
116}

118/// What choosing an answer does. The answers then step aside, so the film is in
119/// view, until the question changes.
120#[derive(Clone)]
121struct Open {
122    film: State<usize>,
123    question: State<String>,
124    answered: State<Option<String>>,
125    cue: MovieCue,
126}
127
128impl Open {
129    fn moment(&self, moment: &Moment) {
130        self.film.set(moment.film);
131        self.answered.set(Some(self.question.get()));
132        self.cue.go_to(moment.at_ms);
133    }
134}

136impl View for MovieAssistant {
137    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
138        let question: State<String> = cx.use_state(String::new);
139        let film: State<usize> = cx.use_state(|| 0);
140        let answered: State<Option<String>> = cx.use_state(|| None);
141        let own_cue = cx.use_state(MovieCue::new).get();
142        let cue = self.cue.clone().unwrap_or(own_cue);
143        if let Some(asked) = cue.take_film() {
144            film.set(asked);
145        }
146        let open = Open {
147            film: film.clone(),
148            question: question.clone(),
149            answered: answered.clone(),
150            cue: cue.clone(),
151        };
152
153        let asked = question.get();
154        let waiting = answered.get().as_ref() != Some(&asked);
155        let answers = if waiting { self.library.ask(&asked, ANSWERS) } else { Vec::new() };
156        let (library, first) = (self.library.clone(), open.clone());
157        let field = TextField::new(question.binding())
158            .placeholder(self.labels.placeholder.clone())
159            .icon(IconName::Search)
160            .on_submit(move |text| {
161                if let Some(best) = library.ask(&text, 1).first() {
162                    first.moment(&best.moment);
163                }
164            });
165        let rows: Vec<ListItem> = answers
166            .iter()
167            .map(|answer| {
168                let open = open.clone();
169                moment_row(&self.library, answer, move |moment| open.moment(moment))
170            })
171            .collect();
172        let nothing = (waiting && !asked.trim().is_empty() && rows.is_empty()).then(|| {
173            Text::new(self.labels.nothing_found.clone())
174                .style(TextStyle::Callout)
175                .foreground(Semantic::InkMuted)
176        });
177        let answered = (!rows.is_empty()).then(|| List::new(rows));
178        let ask = Card::new(VStack::new((field, answered, nothing)).spacing(8.0).align(Align::Stretch))
179            .width(self.width);
180        let mut children = vec![cx.child(&ask)];
181
182        if self.library.len() > 1 {
183            let titles: Vec<String> =
184                (0..self.library.len()).filter_map(|i| self.library.get(i)).map(|f| f.0.to_owned()).collect();
185            children.push(cx.child(&SegmentedControl::new(film.binding(), titles)));
186        }
187        let shown = film.get();
188        if let Some((_, movie)) = self.library.get(shown) {
189            let player =
190                MoviePlayer::new(movie.clone()).width(self.width).labels(self.labels.player.clone()).cue(cue);
191            // Each film keeps its own place.
192            children.push(cx.keyed_child(shown as u64, &player));
193        }
194        cx.container(
195            LayoutStyle { spacing: SPACING, ..LayoutStyle::stack(Axis::Vertical) },
196            &children,
197            Paint::None,
198        )
199    }
