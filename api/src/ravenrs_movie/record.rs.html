12const FRAME: f64 = 1.0 / 30.0;
13/// How fast the pointer travels, in logical pixels a second, and its limits.
14const POINTER_SPEED: f32 = 900.0;
15const POINTER_TRAVEL: (f64, f64) = (0.25, 0.8);
16const SECONDS_PER_LETTER: f64 = 0.06;
17/// How long reading or hearing a text takes, at an easy pace.
18const READING_PER_LETTER: f64 = 0.075;
19const READING_LEAST: f64 = 2.0;
20const READING_START: f64 = 0.8;
21/// The breath after a recorded box, before the next thing happens.
22const AFTER_SPEAKING: f64 = 0.6;
23
24/// What can stop a recording.
25#[derive(Debug)]
26#[non_exhaustive]
27#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
28pub enum RecordError {
29    /// The script names something that is not on the screen.
30    NotOnScreen {
31        /// What it looked for.
32        name: String,
33        /// What there was to find.
34        available: Vec<String>,
35    },
36    /// The film could not be put together.
37    Movie(MovieError),
38}
39
40impl std::fmt::Display for RecordError {
41    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
42        match self {
43            Self::NotOnScreen { name, available } => {
44                write!(f, "'{name}' is not on the screen; there is {}", available.join(", "))
45            }
46            Self::Movie(e) => e.fmt(f),
47        }
48    }
49}
50
51impl std::error::Error for RecordError {}
52
53impl From<MovieError> for RecordError {
54    fn from(e: MovieError) -> Self {
55        Self::Movie(e)
56    }
57}

59/// How long it takes to read or hear a box: the voice says its title too.
60fn reading_seconds(title: &str, text: &str) -> f64 {
61    let letters = title.chars().count() + text.chars().count();
62    (READING_START + letters as f64 * READING_PER_LETTER).max(READING_LEAST)
63}

65/// Runs an app from a script and films it: the steps are the ones a person would
66/// take, found by the names screen readers see. See the [crate documentation](crate).
67#[derive(Debug)]
68#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
69pub struct Recorder {
70    /// The app, driven as a test drives it: the same session, finding and input as `ravenrs-e2e`.
71    e2e: E2e,
72    clock: f64,
73    pointer: Point,
74    logical: Size,
75    /// The film, at a person's pace: the same writer a test films with.
76    film: SessionFilm,
77    narration: Option<Narration>,
78}
79
80#[cfg_attr(feature = "live", ravenrs_live::expose)]
81impl Recorder {
82    /// Films `root` in a window of `size`.
83    pub fn new(root: impl View + 'static, theme: Theme, size: Size) -> Self {
84        let mut e2e = E2e::new(root, theme).viewport(size, 1.0);
85        e2e.frame_at(0.0);
86        let film = SessionFilm::start(&mut e2e, Pace::Person);
87        let mut recorder = Self {
88            e2e,
89            clock: 0.0,
90            pointer: Point::new(size.width / 2.0, size.height / 2.0),
91            logical: size,
92            film,
93            narration: None,
94        };
95        recorder.tick();
96        recorder
97    }
98
99    /// The recordings the player will say the boxes with. Each box then lasts as
100    /// long as its recording, and a little more, so that the voice and the box
101    /// end together and the film never ends while the voice is still speaking.
102    /// A box with no recording lasts as long as reading it takes.
103    #[must_use]
104    pub fn narration(mut self, narration: Narration) -> Self {
105        self.narration = Some(narration);
106        self
107    }
108
109    /// How long a box stays: as long as its recording, or as reading it takes.
110    /// The player says the title and then the text.
111    fn speaking_seconds(&self, title: &str, text: &str) -> f64 {
112        let said = format!("{title}. {text}");
113        match self.narration.as_ref().and_then(|narration| narration.seconds(&said)) {
114            Some(seconds) => (seconds + AFTER_SPEAKING).max(READING_LEAST),
115            None => reading_seconds(title, text),
116        }
117    }
118
119    fn milliseconds(&self) -> u64 {
120        (self.clock * 1000.0) as u64
121    }
122
123    /// Advances the clock by one frame and films what the app draws.
124    fn tick(&mut self) {
125        self.clock += FRAME;
126        self.film.at(self.milliseconds());
127        self.e2e.frame_at(self.clock);
128    }
129
130    /// What is called `name`, found as a test finds it: `#id` by its test id, a text by the
131    /// exact name, and, when that is not one thing, what a person would reach for: the one
132    /// that takes input first, then a name that starts with it.
133    fn find(&mut self, name: &str) -> Result<AccessNode, RecordError> {
134        let not_found = |e2e: &E2e, ids: bool| RecordError::NotOnScreen {
135            name: name.to_owned(),
136            available: e2e
137                .tree()
138                .nodes
139                .iter()
140                .filter_map(|n| {
141                    if ids {
142                        n.test_id.as_ref().map(|t| format!("#{t}"))
143                    } else {
144                        n.name.clone().filter(|_| n.focusable)
145                    }
146                })
147                .collect(),
148        };
149        if let Some(id) = name.strip_prefix('#') {
150            return self.e2e.find(&Query::test_id(id.trim())).map_err(|_| not_found(&self.e2e, true));
151        }
152        let interactive = |n: &AccessNode| n.focusable || n.role == Role::Button || n.role == Role::TextInput;
153        let exact = self.e2e.find_all(&Query::text(name)).unwrap_or_default();
154        if let Some(found) = exact.iter().find(|n| interactive(n)).or_else(|| exact.first()) {
155            return Ok(found.clone());
156        }
157        let starts = |n: &&AccessNode| n.name.as_deref().is_some_and(|text| text.starts_with(name));
158        let nodes = &self.e2e.tree().nodes;
159        nodes
160            .iter()
161            .filter(starts)
162            .find(|n| interactive(n))
163            .or_else(|| nodes.iter().find(starts))
164            .cloned()
165            .ok_or_else(|| not_found(&self.e2e, false))
166    }

168    /// Lets the film run with nothing happening.
169    pub fn wait(&mut self, seconds: f64) -> &mut Self {
170        let until = self.clock + seconds;
171        while self.clock < until {
172            self.tick();
173        }
174        self
175    }

177    /// Starts a chapter here.
178    pub fn chapter(&mut self, title: &str, text: &str) -> &mut Self {
179        self.film.chapter(title, text);
180        self
181    }

183    /// Starts a chapter here and explains it in a box in the middle of the film,
184    /// for as long as it takes to read or hear.
185    pub fn intro(&mut self, title: &str, text: &str) -> &mut Self {
186        self.chapter(title, text);
187        let seconds = self.speaking_seconds(title, text);
188        self.film.hotspot(Hotspot {
189            kind: HotspotKind::Note,
190            x: 0.0,
191            y: 0.0,
192            width: 0.0,
193            height: 0.0,
194            title: title.to_owned(),
195            text: text.to_owned(),
196            from_ms: self.milliseconds(),
197            until_ms: self.milliseconds() + (seconds * 1000.0) as u64,
198        });
199        self.wait(seconds)
200    }

202    /// A [`Self::note`] that lasts as long as `text` takes to hear, or to read
203    /// when it has no recording.
204    pub fn explain(&mut self, name: &str, title: &str, text: &str) -> Result<&mut Self, RecordError> {
205        let seconds = self.speaking_seconds(title, text);
206        self.note(name, title, text, seconds)
207    }

209    /// Slides the pointer to the middle of what is called `name`, or of what has the test id
210    /// after `#` (`#salvar`), as a test of `ravenrs-e2e` finds it.
211    pub fn move_to(&mut self, name: &str) -> Result<&mut Self, RecordError> {
212        let target = self.find(name)?.bounds.center();
213        let start = self.pointer;
214        let distance = (target.x - start.x).hypot(target.y - start.y);
215        let travel = f64::from(distance / POINTER_SPEED).clamp(POINTER_TRAVEL.0, POINTER_TRAVEL.1);
216        let begin = self.clock;
217        self.e2e.pointer_to(start);
218        while self.clock - begin < travel {
219            let t = ((self.clock - begin) / travel) as f32;
220            let progress = t * t * (3.0 - 2.0 * t);
221            self.pointer = Point::new(
222                start.x + (target.x - start.x) * progress,
223                start.y + (target.y - start.y) * progress,
224            );
225            self.e2e.pointer_to(self.pointer);
226            self.tick();
227        }
228        self.pointer = target;
229        self.e2e.pointer_to(target);
230        self.tick();
231        Ok(self)
232    }

234    /// Slides the pointer to what is called `name` and clicks it.
235    pub fn click(&mut self, name: &str) -> Result<&mut Self, RecordError> {
236        self.move_to(name)?;
237        let position = self.pointer;
238        self.e2e.mouse_down(position);
239        self.tick();
240        self.e2e.mouse_up(position);
241        Ok(self.wait(0.5))
242    }
243
244    /// Clicks the field called `name` and types `text` into it, letter by letter.
245    pub fn type_into(&mut self, name: &str, text: &str) -> Result<&mut Self, RecordError> {
246        self.click(name)?;
247        // A click on a field in a dialog can land on its label: the keyboard has to get there.
248        let id = self.find(name)?.id;
249        self.e2e.focus(id);
250        for character in text.chars() {
251            self.press(Key::Character(character.to_string()));
252            self.wait(SECONDS_PER_LETTER);
253        }
254        Ok(self.wait(0.3))
255    }
256
257    /// Presses a key.
258    pub fn press(&mut self, key: Key) -> &mut Self {
259        self.e2e.key_down(key, Modifiers::default());
260        self.tick();
261        self
262    }

264    /// Rolls the wheel over what is under the pointer, `pixels` in all; positive
265    /// goes down the page.
266    pub fn scroll(&mut self, pixels: f32) -> &mut Self {
267        const STEPS: u16 = 15;
268        for _ in 0..STEPS {
269            let delta = Point::new(0.0, -pixels / f32::from(STEPS));
270            self.e2e.scroll_at(self.pointer, delta);
271            self.tick();
272        }
273        self.wait(0.3)
274    }

276    /// Marks what is called `name` and says `text` about it for `seconds`, in
277    /// which the film goes on. Clicking the mark in the player brings the text back.
278    pub fn note(
279        &mut self,
280        name: &str,
281        title: &str,
282        text: &str,
283        seconds: f64,
284    ) -> Result<&mut Self, RecordError> {
285        let hotspot = self.hotspot(name, HotspotKind::Note, title, text, seconds)?;
286        self.film.hotspot(hotspot);
287        Ok(self.wait(seconds))
288    }

290    /// The film stops here, with what is called `name` marked and `text` saying
291    /// what to do, until someone clicks the mark. Follow it with the step itself,
292    /// such as [`Self::click`], which is what the film shows next. When the
293    /// player is not stopping at steps, the box stays for as long as it takes to
294    /// read or hear; when it is, that time is skipped.
295    pub fn guide(&mut self, name: &str, title: &str, text: &str) -> Result<&mut Self, RecordError> {
296        // The pointer gets there first: the step asks for what it is already on.
297        self.move_to(name)?;
298        let seconds = self.speaking_seconds(title, text);
299        let hotspot = self.hotspot(name, HotspotKind::Stop, title, text, seconds)?;
300        self.film.hotspot(hotspot);
301        Ok(self.wait(seconds))
302    }
303
304    fn hotspot(
305        &mut self,
306        name: &str,
307        kind: HotspotKind,
308        title: &str,
309        text: &str,
310        seconds: f64,
311    ) -> Result<Hotspot, RecordError> {
312        let bounds = self.find(name)?.bounds;
313        // A hair larger than what it marks, so the border does not cover it.
314        const MARGIN: f32 = 3.0;
315        Ok(Hotspot {
316            kind,
317            x: (bounds.x() - MARGIN).max(0.0),
318            y: (bounds.y() - MARGIN).max(0.0),
319            width: (bounds.width() + MARGIN * 2.0).min(self.logical.width),
320            height: (bounds.height() + MARGIN * 2.0).min(self.logical.height),
321            title: title.to_owned(),
322            text: text.to_owned(),
323            from_ms: self.milliseconds(),
324            until_ms: self.milliseconds() + (seconds * 1000.0) as u64,
325        })
326    }
