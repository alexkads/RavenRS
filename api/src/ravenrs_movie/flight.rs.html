24const REPORT_MAGIC: &[u8] = b"RAVENBUG1\n";
25/// The fewest seconds between two frames it keeps: a report does not need more,
26/// and the app it watches should not pay for more.
27const SHORTEST_GAP: f64 = 1.0 / 15.0;
28const PREVIEW_WIDTH: f32 = 640.0;
29
30#[derive(Clone, Debug)]
31struct Stretch {
32    film: FilmBuilder,
33    started: f64,
34}
35
36#[derive(Debug)]
37struct Flight {
38    keep_seconds: f64,
39    size: Size,
40    newer: Option<Stretch>,
41    older: Option<Stretch>,
42    last_frame: Vec<Primitive>,
43    last_at: f64,
44    now: f64,
45    paused: bool,
46}
47
48impl Flight {
49    /// The stretch being written, starting another when this one is full. Two
50    /// stretches of half the time each always hold at least that half, and old
51    /// memory goes all at once instead of frame by frame.
52    fn stretch(&mut self, now: f64, background: ravenrs_core::Color) -> &mut Stretch {
53        let full = self.newer.as_ref().is_none_or(|s| now - s.started >= self.keep_seconds / 2.0);
54        if full {
55            self.older = self.newer.take();
56            self.last_frame.clear();
57            self.newer = Some(Stretch { film: FilmBuilder::new(self.size, background), started: now });
58        }
59        self.newer
60            .get_or_insert_with(|| Stretch { film: FilmBuilder::new(self.size, background), started: now })
61    }
62
63    fn painted(&mut self, scene: &ravenrs_core::Scene, now: f64) {
64        self.now = now;
65        let primitives = scene.primitives();
66        if self.paused || (now - self.last_at < SHORTEST_GAP && !self.last_frame.is_empty()) {
67            return;
68        }
69        if self.last_frame == primitives
70            && self.newer.as_ref().is_some_and(|s| now - s.started < self.keep_seconds / 2.0)
71        {
72            return;
73        }
74        let stretch = self.stretch(now, scene.clear_color());
75        let at = ((now - stretch.started).max(0.0) * 1000.0) as u64;
76        stretch.film.frame(at, primitives);
77        self.last_frame = primitives.to_vec();
78        self.last_at = now;
79    }
80
81    fn pointer(&mut self, position: Point, clicked: bool, now: f64) {
82        self.now = now;
83        if self.paused {
84            return;
85        }
86        if let Some(stretch) = &mut self.newer {
87            let at = ((now - stretch.started).max(0.0) * 1000.0) as u64;
88            stretch.film.pointer(at, position);
89            if clicked {
90                stretch.film.click(at, position);
91            }
92        }
93    }
94
95    /// The stretches as one film, without disturbing the recording.
96    fn film(&self) -> Option<Movie> {
97        let newer = self.newer.as_ref()?;
98        let stretches: Vec<&Stretch> = self.older.iter().chain([newer]).collect();
99        let origin = stretches[0].started;
100        let mut whole: Option<FilmBuilder> = None;
101        let mut last_ms = 0;
102        for stretch in stretches {
103            let ms = ((self.now - stretch.started).max(0.0) * 1000.0) as u64;
104            let Ok(part) = stretch.film.clone().finish(ms) else {
105                continue;
106            };
107            let shift = ((stretch.started - origin).max(0.0) * 1000.0) as u64;
108            let logical = part.logical_size();
109            let background = part.background();
110            let film = whole.get_or_insert_with(|| FilmBuilder::new(logical, background));
111            for (at, list) in part.frames() {
112                let drawn: Vec<Primitive> = list.iter().filter_map(|i| part.primitive(*i).cloned()).collect();
113                film.frame(at + shift, &drawn);
114            }
115            for key in part.pointer_keys() {
116                film.pointer(key.at_ms + shift, Point::new(key.x, key.y));
117            }
118            for key in part.click_keys() {
119                film.click(key.at_ms + shift, Point::new(key.x, key.y));
120            }
121            last_ms = last_ms.max(ms + shift);
122        }
123        whole?.finish(last_ms).ok()
124    }
125}

127/// Keeps the last minutes on screen in memory, the way a flight recorder does, so
128/// that a bug can show what led to it. Put it anywhere in the app's views: it
129/// draws nothing. A film costs little because each thing drawn is stored once and
130/// a screen that does not change costs nothing, and it runs only when the app paints.
131///
132/// It sees all that the app shows, so it writes nothing anywhere: only
133/// [`Self::film`] gets anything out of it.
134///
135/// ```
136/// use ravenrs_movie::FlightRecorder;
137///
138/// // The last three minutes.
139/// let recorder = FlightRecorder::new(180.0);
140/// assert!(recorder.film().is_none(), "nothing on screen yet");
141/// ```
142#[derive(Clone, Debug)]
143pub struct FlightRecorder {
144    flight: Rc<RefCell<Flight>>,
145}
146
147impl FlightRecorder {
148    /// A recorder that keeps at least half of `seconds`, and up to all of them.
149    pub fn new(seconds: f64) -> Self {
150        Self {
151            flight: Rc::new(RefCell::new(Flight {
152                keep_seconds: seconds.max(1.0),
153                size: Size::new(1.0, 1.0),
154                newer: None,
155                older: None,
156                last_frame: Vec::new(),
157                last_at: f64::NEG_INFINITY,
158                now: 0.0,
159                paused: false,
160            })),
161        }
162    }
163
164    /// The film so far, to look at, edit or send. Recording goes on.
165    pub fn film(&self) -> Option<Movie> {
166        self.flight.borrow().film()
167    }
168
169    /// Stops or resumes noting anything, for screens that should stay out of it.
170    pub fn set_paused(&self, paused: bool) {
171        self.flight.borrow_mut().paused = paused;
172    }
173}
174
175impl View for FlightRecorder {
176    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
177        let size = cx.viewport();
178        {
179            let mut flight = self.flight.borrow_mut();
180            if flight.size != size {
181                // A film has one size, so a resized window starts over.
182                flight.size = size;
183                flight.newer = None;
184                flight.older = None;
185                flight.last_frame.clear();
186            }
187        }
188        let flight = self.flight.clone();
189        cx.observe_activity(Rc::new(move |activity: UiActivity<'_>, now: f64| {
190            // Its own dialog paints inside a borrow only if it reads the film while
191            // painting, which it does not; if that changes, a frame is skipped, no more.
192            let Ok(mut flight) = flight.try_borrow_mut() else {
193                return;
194            };
195            match activity {
196                UiActivity::Painted(scene) => flight.painted(scene, now),
197                UiActivity::PointerMoved(position) => flight.pointer(position, false, now),
198                UiActivity::Clicked(position) => flight.pointer(position, true, now),
199                _ => {}
200            }
201        }));
202        cx.empty()
203    }
204}

206/// What someone says about a bug, for the development team.
207#[derive(Clone, Debug)]
208pub struct BugReport {
209    /// What happened, in their words.
210    pub description: String,
211    /// The film of what led to it, if they chose to include it.
212    pub film: Option<Movie>,
213    /// What the app adds, such as its version. Shown before it is sent.
214    pub details: Vec<(String, String)>,
215}
216
217#[derive(Serialize, Deserialize)]
218struct ReportHeader {
219    description: String,
220    details: Vec<(String, String)>,
221}
222
223impl BugReport {
224    /// The report as the bytes of one file.
225    pub fn to_bytes(&self) -> Vec<u8> {
226        let header = ReportHeader { description: self.description.clone(), details: self.details.clone() };
227        let header = serde_json::to_vec(&header).unwrap_or_default();
228        let mut bytes = Vec::new();
229        bytes.extend_from_slice(REPORT_MAGIC);
230        bytes.extend_from_slice(&(header.len() as u64).to_le_bytes());
231        bytes.extend_from_slice(&header);
232        if let Some(film) = &self.film {
233            bytes.extend_from_slice(&film.to_bytes());
234        }
235        bytes
236    }
237
238    /// Reads a report back from the bytes of a file.
239    pub fn from_bytes(bytes: &[u8]) -> Result<Self, MovieError> {
240        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
241        let rest =
242            bytes.strip_prefix(REPORT_MAGIC).ok_or_else(|| damaged("it does not start like a report"))?;
243        let (length, rest) = rest.split_first_chunk::<8>().ok_or_else(|| damaged("no header"))?;
244        let length = usize::try_from(u64::from_le_bytes(*length)).map_err(|_| damaged("header too long"))?;
245        let header = rest.get(..length).ok_or_else(|| damaged("the header was cut"))?;
246        let header: ReportHeader =
247            serde_json::from_slice(header).map_err(|e| MovieError::Damaged(e.to_string()))?;
248        let rest = &rest[length..];
249        let film = if rest.is_empty() { None } else { Some(Movie::from_bytes(rest)?) };
250        Ok(Self { description: header.description, film, details: header.details })
251    }
252}

321/// A report being written.
322#[derive(Clone)]
323struct Draft {
324    editor: Option<EditorState>,
325}
326
327/// The dialog that turns what the recorder has into a [`BugReport`].
328///
329/// The film is taken when the dialog opens, so it does not show the report being
330/// written. The person watches it, can leave it out, and can cover or cut parts
331/// of it in the editor before sending. Where the report goes is up to the app.
332#[derive(Clone)]
333pub struct BugReporter {
334    recorder: FlightRecorder,
335    open: State<bool>,
336    editing: State<bool>,
337    include: State<bool>,
338    description: State<String>,
339    draft: Rc<RefCell<Draft>>,
340    labels: Rc<BugLabels>,
341    details: Rc<dyn Fn() -> Vec<(String, String)>>,
342    on_send: Rc<dyn Fn(BugReport)>,
343}
344
345impl std::fmt::Debug for BugReporter {
346    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
347        f.debug_struct("BugReporter").field("recorder", &self.recorder).finish_non_exhaustive()
348    }
349}
350
351impl BugReporter {
352    /// A reporter over what `recorder` keeps; `on_send` gets the finished report.
353    pub fn new(recorder: FlightRecorder, on_send: impl Fn(BugReport) + 'static) -> Self {
354        Self {
355            recorder,
356            open: State::new(false),
357            editing: State::new(false),
358            include: State::new(true),
359            description: State::new(String::new()),
360            draft: Rc::new(RefCell::new(Draft { editor: None })),
361            labels: Rc::new(BugLabels::default()),
362            details: Rc::new(Vec::new),
363            on_send: Rc::new(on_send),
364        }
365    }
366
367    /// The words it shows.
368    pub fn labels(mut self, labels: BugLabels) -> Self {
369        self.labels = Rc::new(labels);
370        self
371    }
372
373    /// What the app adds to every report, such as its version.
374    pub fn details(mut self, details: impl Fn() -> Vec<(String, String)> + 'static) -> Self {
375        self.details = Rc::new(details);
376        self
377    }
378
379    /// Opens the dialog, with the film up to now.
380    pub fn open(&self) {
381        self.draft.borrow_mut().editor = self.recorder.film().map(EditorState::new);
382        self.recorder.set_paused(true);
383        self.description.set(String::new());
384        self.include.set(true);
385        self.editing.set(false);
386        self.open.set(true);
387    }
388
389    fn close(&self) {
390        self.open.set(false);
391        self.editing.set(false);
392        self.draft.borrow_mut().editor = None;
393        self.recorder.set_paused(false);
394    }
395
396    fn send(&self) {
397        let film = if self.include.get() {
398            self.draft.borrow().editor.as_ref().map(EditorState::movie)
399        } else {
400            None
401        };
402        (self.on_send)(BugReport { description: self.description.get(), film, details: (self.details)() });
403        self.close();
404    }
405}

407impl View for BugReporter {
408    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
409        if !self.open.get() {
410            return cx.empty();
411        }
412        let labels = &self.labels;
413        let viewport = cx.viewport();
414        let (closing, read_open, cancel) = (self.clone(), self.open.clone(), self.clone());
415        let open = Binding::new(
416            move || read_open.get(),
417            move |open: bool| {
418                if !open {
419                    closing.close();
420                }
421            },
422        );
423        let editor = self.draft.borrow().editor.clone();
424        if self.editing.get()
425            && let Some(state) = editor
426        {
427            let area = Size::new(viewport.width - 120.0, viewport.height - 200.0);
428            let back = self.editing.clone();
429            let dialog = Dialog::new(open, labels.edit_film.clone())
430                .width(area.width + 48.0)
431                .content(
432                    MovieEditor::new(state)
433                        .labels(labels.editor.clone())
434                        .area(area)
435                        .frame(Some(area.width), Some(area.height)),
436                )
437                .action(Button::new(labels.done_editing.clone()).on_click(move || back.set(false)));
438            return cx.child(&dialog);
439        }
440        let include = self.include.get();
441        let has_film = editor.is_some();
442        let film = editor.filter(|_| include).map(|state| {
443            let edit = self.editing.clone();
444            VStack::new((
445                Text::new(labels.film_warning.clone())
446                    .style(TextStyle::Callout)
447                    .foreground(Semantic::InkMuted),
448                MoviePlayer::new(state.movie())
449                    .labels(labels.player.clone())
450                    .width(PREVIEW_WIDTH.min(viewport.width - 120.0)),
451                Button::new(labels.edit_film.clone())
452                    .icon(IconName::EyeOff)
453                    .style(ButtonStyle::Secondary)
454                    .on_click(move || edit.set(true)),
455            ))
456            .spacing(10.0)
457            .align(Align::Start)
458        });
459        let send = self.clone();
460        let written = !self.description.get().trim().is_empty();
461        let dialog = Dialog::new(open, labels.title.clone())
462            .description(labels.intro.clone())
463            .width(PREVIEW_WIDTH + 64.0)
464            .content(
465                ravenrs_widgets::ScrollView::new((VStack::new((
466                    Text::new(labels.description.clone()).style(TextStyle::Callout),
467                    TextField::new(self.description.binding()),
468                    has_film.then(|| Toggle::new(self.include.binding()).label(labels.include_film.clone())),
469                    film,
470                ))
471                .spacing(10.0)
472                .align(Align::Stretch),))
473                .frame(None, Some((viewport.height - 260.0).max(200.0))),
474            )
475            .action(
476                Button::new(labels.cancel.clone())
477                    .style(ButtonStyle::Secondary)
478                    .on_click(move || cancel.close()),
479            )
480            .action(Button::new(labels.send.clone()).disabled(!written).on_click(move || send.send()));
481        cx.child(&dialog)
482    }
