285    /// The film with the edits so far.
286    pub fn movie(&self) -> Movie {
287        self.inner.movie.borrow().clone()
288    }

290    fn touch(&self) {
291        self.revision.update(|n| *n += 1);
292    }
293
294    /// Makes an edit. Edits with the same `key` in a row count as one undo.
295    fn edit(&self, key: Option<&str>, work: impl FnOnce(&mut Project)) {
296        let inner = &self.inner;
297        let mut project = inner.movie.borrow().project().clone();
298        let same = key.is_some() && inner.last_edit.borrow().as_deref() == key;
299        if !same {
300            let mut undo = inner.undo.borrow_mut();
301            undo.push(project.clone());
302            if undo.len() > UNDO_LIMIT {
303                undo.remove(0);
304            }
305            inner.redo.borrow_mut().clear();
306        }
307        *inner.last_edit.borrow_mut() = key.map(str::to_owned);
308        work(&mut project);
309        let movie = inner.movie.borrow().with_project(project);
310        *inner.movie.borrow_mut() = movie;
311        self.keep_position_inside();
312        self.touch();
313    }
314
315    fn keep_position_inside(&self) {
316        let duration = self.movie().duration_ms();
317        if self.inner.position.get() > duration {
318            self.inner.position.set(duration);
319        }
320    }
321
322    fn swap(&self, from: &RefCell<Vec<Project>>, to: &RefCell<Vec<Project>>) {
323        let Some(project) = from.borrow_mut().pop() else {
324            return;
325        };
326        to.borrow_mut().push(self.movie().project().clone());
327        let movie = self.movie().with_project(project);
328        *self.inner.movie.borrow_mut() = movie;
329        *self.inner.last_edit.borrow_mut() = None;
330        self.inner.selection.set(Selection::Nothing);
331        self.keep_position_inside();
332        self.touch();
333    }
334
335    /// Takes the last edit back.
336    pub fn undo(&self) {
337        self.swap(&self.inner.undo, &self.inner.redo);
338    }
339
340    /// Puts back what [`Self::undo`] took.
341    pub fn redo(&self) {
342        self.swap(&self.inner.redo, &self.inner.undo);
343    }

359    /// Plays from the playhead, or pauses.
360    pub fn toggle_playing(&self) {
361        if self.inner.running.get().is_some() {
362            self.inner.running.set(None);
363        } else {
364            if self.position_ms() >= self.movie().duration_ms() {
365                self.inner.position.set(0);

370    }

405    /// Adds an effect at the playhead and selects it.
406    pub fn add_overlay(&self, kind: OverlayKind) -> u64 {
407        let at = self.position_ms();
408        let mut id = 0;
409        self.edit(None, |project| {
410            let length = NEW_EFFECT_MS.min(project.duration_ms().saturating_sub(at).max(500));
411            id = project.add_overlay(kind, at, length);
412        });
413        self.inner.selection.set(Selection::Overlay(id));
414        self.touch();
415        id

418    /// Lights `region` for the last seconds of the film, replacing what `previous`
419    /// lit, and says how to replace this one. It marks where a problem is.
420    pub fn light_at_end(&self, region: Region, previous: Option<u64>) -> u64 {
421        let mut id = 0;
422        self.edit(None, |project| {
423            if let Some(previous) = previous {
424                project.remove_overlay(previous);
425            }
426            let length = NEW_EFFECT_MS.min(project.duration_ms()).max(1);
427            let at = project.duration_ms().saturating_sub(length);
428            id = project.add_overlay(OverlayKind::Spotlight { region }, at, length);
429        });
430        self.touch();
431        id
432    }

478    /// Runs the clock while playing. Called while building: only plain cells change.
479    fn advance(&self, now: f64) {
480        let Some((since, from)) = self.inner.running.get() else {
481            return;
482        };
483        if since.is_nan() {
484            self.inner.running.set(Some((now, from)));
485            return;
486        }
487        let at = from + ((now - since).max(0.0) * 1000.0) as u64;
488        let duration = self.movie().duration_ms();
489        if at >= duration {
490            self.inner.position.set(duration);
491            self.inner.running.set(None);
492        } else {
493            self.inner.position.set(at);
494        }
495    }

534/// What an overlay goes under on the timeline.
535#[derive(Clone, Copy, PartialEq, Eq)]
536enum Row {
537    Interaction,
538    Annotations,
539    Screen,
540    Pointer,
541}

543fn row_of(kind: &OverlayKind) -> Row {
544    match kind {
545        OverlayKind::Hotspot { .. } => Row::Interaction,
546        OverlayKind::Callout { .. } | OverlayKind::Caption { .. } => Row::Annotations,
547        OverlayKind::PointerHighlight => Row::Pointer,
548        _ => Row::Screen,
549    }

552/// The ids blocks have on the timeline: clips below, overlays above.
553const OVERLAY_IDS: u64 = 1 << 32;
554/// The steps that came with the film, which only say where they are.
555const FILM_STEP_IDS: u64 = 2 << 32;
556/// A step lasts no time, but it has to be seen and clicked.
557const STEP_MS: u64 = 700;
558
559fn on_timeline_edit(state: &EditorState, edit: TimelineEdit) {
560    match edit {
561        TimelineEdit::Began { .. } => *state.inner.last_edit.borrow_mut() = None,
562        TimelineEdit::Moved { id, start_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
563            let id = id - OVERLAY_IDS;
564            state.inner.selection.set(Selection::Overlay(id));
565            state.change_overlay(id, "move", |overlay| overlay.start_ms = start_ms);
566        }
567        TimelineEdit::TrimmedStart { id, by_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
568            state.change_overlay(id - OVERLAY_IDS, "resize", |overlay| {
569                let end = overlay.end_ms();
570                let start = (overlay.start_ms as i64 + by_ms).clamp(0, end as i64 - 100) as u64;
571                overlay.start_ms = start;
572                overlay.duration_ms = end - start;
573            });
574        }
575        TimelineEdit::TrimmedEnd { id, by_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
576            state.change_overlay(id - OVERLAY_IDS, "resize", |overlay| {
577                overlay.duration_ms = (overlay.duration_ms as i64 + by_ms).max(100) as u64;
578            });
579        }
580        TimelineEdit::TrimmedStart { id, by_ms } => {
581            let index = id as usize;
582            state.edit(Some(&format!("trim:{index}")), |project| project.trim_start(index, by_ms));
583        }
584        TimelineEdit::TrimmedEnd { id, by_ms } => {
585            let index = id as usize;
586            state.edit(Some(&format!("trim:{index}")), |project| project.trim_end(index, by_ms));
587        }
588        _ => {}
589    }
590}

592fn timeline(state: &EditorState, labels: &EditorLabels) -> Timeline {
593    let movie = state.movie();
594    let project = movie.project();
595    let selection = state.inner.selection.get();
596    let overlays_in = |row: Row| -> Vec<TimelineItem> {
597        project
598            .overlays
599            .iter()
600            .filter(|overlay| row_of(&overlay.kind) == row)
601            .map(|overlay| {
602                let tone = match row {
603                    Row::Interaction => Tone::Warning,
604                    Row::Annotations => Tone::Accent,
605                    Row::Screen => Tone::Success,
606                    Row::Pointer => Tone::Neutral,
607                };
608                TimelineItem::new(
609                    overlay.id + OVERLAY_IDS,
610                    overlay.start_ms,
611                    overlay.duration_ms,
612                    label_of(&overlay.kind, labels),
613                )
614                .tone(tone)
615                .selected(selection == Selection::Overlay(overlay.id))
616                .movable()
617                .trimmable()
618            })
619            .collect()
620    };
621    let clips: Vec<TimelineItem> = project
622        .clips
623        .iter()
624        .enumerate()
625        .map(|(index, clip)| {
626            let start = project.clip_start_ms(index);
627            let label = match &clip.kind {
628                ClipKind::Film { .. } if (clip.speed - 1.0).abs() > 0.01 => {
629                    format!("{} {} · {}×", labels.film_clip, index + 1, clip.speed)
630                }
631                ClipKind::Film { .. } => format!("{} {}", labels.film_clip, index + 1),
632                ClipKind::Title { title, .. } => title.clone(),
633            };
634            let moment = if start == 0 { movie.poster_ms() } else { start + 1 };
635            let item = TimelineItem::new(index as u64, start, clip.duration_ms(), label)
636                .tone(if matches!(clip.kind, ClipKind::Title { .. }) { Tone::Success } else { Tone::Accent })
637                .selected(selection == Selection::Clip(index));
638            // What a stretch of film starts with, in vectors like the rest. A card
639            // has its words on the block already, and they would not fit in a thumbnail.
640            if matches!(clip.kind, ClipKind::Film { .. }) {
641                item.leading(Composition::new(movie.clone(), moment).width(THUMBNAIL)).trimmable()
642            } else {
643                item
644            }
645        })
646        .collect();
647    // The steps recorded with the film, beside those added here.
648    let added: Vec<&str> = project
649        .overlays
650        .iter()
651        .filter_map(|overlay| match &overlay.kind {
652            OverlayKind::Hotspot { title, .. } => Some(title.as_str()),
653            _ => None,
654        })
655        .collect();
656    let mut steps = overlays_in(Row::Interaction);
657    for (index, spot) in movie.hotspots().iter().enumerate() {
658        if !added.contains(&spot.title.as_str()) {
659            let length = spot.until_ms.saturating_sub(spot.from_ms).max(STEP_MS);
660            steps.push(
661                TimelineItem::new(FILM_STEP_IDS + index as u64, spot.from_ms, length, spot.title.clone())
662                    .tone(Tone::Warning),
663            );
664        }
665    }
666    let starts: Vec<u64> = movie.hotspots().iter().map(|spot| spot.from_ms).collect();
667    let (select, seek, edit) = (state.clone(), state.clone(), state.clone());
668    Timeline::new(vec![
669        TimelineTrack::new(labels.tab_interaction.clone(), OVERLAY_ROW, steps),
670        TimelineTrack::new(labels.tab_annotations.clone(), OVERLAY_ROW, overlays_in(Row::Annotations)),
671        TimelineTrack::new(labels.tab_screen.clone(), OVERLAY_ROW, overlays_in(Row::Screen)),
672        TimelineTrack::new(labels.tab_pointer.clone(), OVERLAY_ROW, overlays_in(Row::Pointer)),
673        TimelineTrack::new(labels.track_film.clone(), CLIP_ROW, clips),
674    ])
675    .markers(
676        movie.chapters().iter().map(|c| TimelineMarker { at_ms: c.at_ms, title: c.title.clone() }).collect(),
677    )
678    .position_ms(state.position_ms())
679    .pixels_per_second(state.inner.density.get())
680    .heads_width(TRACK_HEADS)
681    .on_seek(move |ms| seek.seek(ms))
682    .on_select(move |id| {
683        if id >= FILM_STEP_IDS {
684            if let Some(at) = usize::try_from(id - FILM_STEP_IDS).ok().and_then(|i| starts.get(i)) {
685                select.seek(*at);
686            }
687        } else if id >= OVERLAY_IDS {
688            select.select_overlay(id - OVERLAY_IDS);
689        } else {
690            select.select_clip(id as usize);
691        }
692    })
693    .on_edit(move |change| on_timeline_edit(&edit, change))
694}

696/// The selected region on the canvas, to drag and resize.
697struct RegionHandle {
698    state: EditorState,
699    id: u64,
700    frame: Rect,
701    scale: f32,
702}
703
704impl View for RegionHandle {
705    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
706        let theme = *cx.theme();
707        let color = theme.color(Semantic::FocusRing);
708        let (state, id, scale) = (self.state.clone(), self.id, self.scale);
709        let size = HANDLE * 1.5;
710        let corner = cx.container(
711            LayoutStyle {
712                absolute: Some((self.frame.width() - size, self.frame.height() - size)),
713                width: Some(size),
714                height: Some(size),
715                ..LayoutStyle::stack(Axis::Vertical)
716            },
717            &[],
718            Paint::Quad(QuadPaint { corner_radius: 2.0, ..QuadPaint::new(color) }),
719        );
720        cx.semantics(corner, Semantics::hidden());
721        let start = cx.use_state(|| Rc::new(Cell::new(None::<(Point, Region)>))).get();
722        let resize = state.clone();
723        cx.on_drag(
724            corner,
725            Rc::new(move |drag: Drag| {
726                if let Some(mut region) = resize.selected_overlay().and_then(|o| region_of(&o.kind)) {
727                    if drag.phase == DragPhase::Start {
728                        *resize.inner.last_edit.borrow_mut() = None;
729                    }
730                    region.width = ((drag.position.x - drag.container.x()) / scale).max(16.0);
731                    region.height = ((drag.position.y - drag.container.y()) / scale).max(16.0);
732                    resize.change_overlay(id, "region", |overlay| set_region(&mut overlay.kind, region));
733                }
734            }),
735        );
736        cx.cursor(corner, CursorIcon::ColResize);
737        let node = cx.container(
738            LayoutStyle {
739                absolute: Some((self.frame.x(), self.frame.y())),
740                width: Some(self.frame.width()),
741                height: Some(self.frame.height()),
742                ..LayoutStyle::stack(Axis::Vertical)
743            },
744            &[corner],
745            Paint::Quad(QuadPaint {
746                fill: Color::TRANSPARENT.into(),
747                corner_radius: 0.0,
748                border: Some(Border { width: 2.0, color }),
749                shadow: None,
750            }),
751        );
752        cx.on_drag(
753            node,
754            Rc::new(move |drag: Drag| {
755                let Some(region) = state.selected_overlay().and_then(|o| region_of(&o.kind)) else {
756                    return;
757                };
758                if drag.phase == DragPhase::Start {
759                    start.set(Some((drag.position, region)));
760                    *state.inner.last_edit.borrow_mut() = None;
761                    return;
762                }
763                if let Some((from, was)) = start.get() {
764                    let moved = Region {
765                        x: was.x + (drag.position.x - from.x) / scale,
766                        y: was.y + (drag.position.y - from.y) / scale,
767                        ..was
768                    };
769                    state.change_overlay(id, "region", |overlay| set_region(&mut overlay.kind, moved));
770                }
771            }),
772        );
773        cx.cursor(node, CursorIcon::Pointer);
774        cx.semantics(node, Semantics::hidden());
775        node
776    }
777}

779/// The canvas: the moment under the playhead, and the region being placed.
780struct Canvas {
781    state: EditorState,
782    width: f32,
783}
784
785impl View for Canvas {
786    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
787        let state = &self.state;
788        let movie = state.movie();
789        let logical = movie.logical_size();
790        let height = self.width * logical.height / logical.width.max(1.0);
791        let selected = state.selected_overlay();
792        let placing_zoom = matches!(selected.as_ref().map(|o| &o.kind), Some(OverlayKind::Zoom { .. }));
793        let at = state.position_ms();
794        let mut film = Composition::new(movie.clone(), at).width(self.width);
795        if placing_zoom {
796            film = film.ignoring_zoom();
797        }
798        let mut children = vec![cx.child(&film)];
799        if let Some(overlay) = &selected
800            && let Some(region) = region_of(&overlay.kind)
801            && overlay.covers(at)
802        {
803            let camera = camera_at(&movie, at, self.width, placing_zoom);
804            children.push(cx.child(&RegionHandle {
805                state: state.clone(),
806                id: overlay.id,
807                frame: camera.region(region),
808                scale: camera.scale(),
809            }));
810        }
811        let node = cx.container(
812            LayoutStyle {
813                width: Some(self.width),
814                height: Some(height),
815                overflow_hidden: true,
816                ..LayoutStyle::stack(Axis::Vertical)
817            },
818            &children,
819            Paint::None,
820        );
821        cx.clip(node);
822        node
823    }
824}

829/// An editor for a film, in the style of screen-recording editors.
830///
831/// Tools are on the left, with what each offers; the canvas in the middle; what
832/// is selected on the right; and the tracks along the bottom. Clips are cut at
833/// the playhead, trimmed at their edges and deleted, closing the gap. Effects sit
834/// on tracks over them, where they slide and stretch, and regions are dragged on
835/// the canvas. Nothing changes the film, every edit can be undone, and the film
836/// can be tried at any time with the same player people will watch it in.
837///
838/// ```no_run
839/// use ravenrs_movie::{EditorState, Movie, MovieEditor};
840///
841/// let film = Movie::from_bytes(&std::fs::read("manual.ravenmovie").unwrap_or_default())?;
842/// let state = EditorState::new(film);
843/// let editor = MovieEditor::new(state).title("Getting started").on_save(|film: &Movie| {
844///     let _ = std::fs::write("manual.ravenmovie", film.to_bytes());
845/// });
846/// # Ok::<(), ravenrs_movie::MovieError>(())
847/// ```
848#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
849pub struct MovieEditor {
850    state: EditorState,
851    labels: EditorLabels,
852    player_labels: MovieLabels,
853    title: String,
854    on_save: Option<SaveHandler>,
855    area: Option<Size>,
856}
857
858impl std::fmt::Debug for MovieEditor {
859    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
860        f.debug_struct("MovieEditor").field("state", &self.state).finish_non_exhaustive()
861    }
862}

1204            if region_of(&overlay.kind).is_some() {
1205                let axis = |state: &EditorState, vertical: bool, name: &str| {
1206                    let (read, write) = (state.clone(), state.clone());
1207                    let value = Binding::new(
1208                        move || {
1209                            let region = read.selected_overlay().and_then(|o| region_of(&o.kind));
1210                            region.map_or(0, |r| if vertical { r.y as i32 } else { r.x as i32 })
1211                        },
1212                        move |new: i32| {
1213                            write.change_overlay(id, "region", |overlay| {
1214                                if let Some(mut region) = region_of(&overlay.kind) {
1215                                    if vertical {
1216                                        region.y = new as f32;
1217                                    } else {
1218                                        region.x = new as f32;
1219                                    }
1220                                    set_region(&mut overlay.kind, region);
1221                                }
1222                            });
1223                        },
1224                    );
1225                    Stepper::new(value).range(0, 8000).label(name.to_owned())
1226                };
1227                items.push(Box::new(axis(state, false, "X")));
1228                items.push(Box::new(axis(state, true, "Y")));
1229                items.push(Box::new(muted(&labels.drag_hint)));
1230            }
1231        } else if let Some((index, clip)) = state.selected_clip() {
1232            match &clip.kind {
1233                ClipKind::Film { .. } => {
1234                    items.push(Box::new(panel_title(&labels.film_clip)));
1235                    items.push(Box::new(muted(&clock(clip.duration_ms()))));
1236                }
1237                ClipKind::Title { .. } => {
1238                    items.push(Box::new(panel_title(&labels.title_card)));
1239                    let words = |second: bool| {
1240                        let (read, write) = (state.clone(), state.clone());
1241                        Binding::new(
1242                            move || match read.selected_clip().map(|(_, clip)| clip.kind) {
1243                                Some(ClipKind::Title { title, subtitle, .. }) => {
1244                                    if second {
1245                                        subtitle
1246                                    } else {
1247                                        title
1248                                    }
1249                                }
1250                                _ => String::new(),
1251                            },
1252                            move |new: String| {
1253                                write.edit(Some(&format!("card:{index}:{second}")), |project| {
1254                                    if let Some(ClipKind::Title { title, subtitle, .. }) =
1255                                        project.clips.get_mut(index).map(|clip| &mut clip.kind)
1256                                    {
1257                                        *(if second { subtitle } else { title }) = new;
1258                                    }
1259                                });
1260                            },
1261                        )
1262                    };
1263                    items.push(Box::new(muted(&labels.text)));
1264                    items.push(Box::new(TextField::new(words(false))));
1265                    items.push(Box::new(muted(&labels.subtitle)));
1266                    items.push(Box::new(TextField::new(words(true))));
1267                }
1268            }
1269        } else {
1270            items.push(Box::new(panel_title(&labels.properties)));
1271            items.push(Box::new(muted(&labels.nothing_selected)));
1272        }
1273        ScrollView::new((VStack::new(items).spacing(10.0).align(Align::Stretch),))
1274            .padding(12.0)
1275            .background(Semantic::Surface)
1276            .frame(Some(PROPERTIES_WIDTH), None)
1277    }

1279    fn transport(&self) -> impl View + use<> {
1280        let labels = &self.labels;
1281        let state = &self.state;
1282        let running = state.inner.running.get().is_some();
1283        let movie = state.movie();
1284        let at = state.position_ms();
1285        let chapter_to = |target: Option<u64>| {
1286            let seek = state.clone();
1287            move || {
1288                if let Some(ms) = target {
1289                    seek.seek(ms);
1290                }
1291            }
1292        };
1293        let previous = movie.chapters().iter().rev().find(|c| c.at_ms < at).map(|c| c.at_ms);
1294        let next = movie.chapters().iter().find(|c| c.at_ms > at).map(|c| c.at_ms);
1295        let toggle = state.clone();
1296        HStack::new((
1297            IconButton::new(IconName::ChevronLeft)
1298                .label(labels.previous_chapter.clone())
1299                .disabled(previous.is_none())
1300                .on_click(chapter_to(previous)),
1301            Button::new(if running { labels.pause.clone() } else { labels.play.clone() })
1302                .icon(if running { IconName::Pause } else { IconName::Play })
1303                .size(ButtonSize::Small)
1304                .on_click(move || toggle.toggle_playing()),
1305            IconButton::new(IconName::ChevronRight)
1306                .label(labels.next_chapter.clone())
1307                .disabled(next.is_none())
1308                .on_click(chapter_to(next)),
1309            Text::new(format!("{} / {}", clock(at), clock(movie.duration_ms()))).style(TextStyle::Callout),
1310        ))
1311        .spacing(6.0)
1312        .align(Align::Center)
1313        .frame(None, Some(TRANSPORT))
1314    }

1316    fn timeline_bar(&self) -> impl View + use<> {
1317        let labels = &self.labels;
1318        let state = &self.state;
1319        let (split, delete, chapter) = (state.clone(), state.clone(), state.clone());
1320        let new_chapter = labels.new_chapter.clone();
1321        let selected = state.inner.selection.get() != Selection::Nothing;
1322        let speed_clip = state.selected_clip().filter(|(_, clip)| matches!(clip.kind, ClipKind::Film { .. }));
1323        let (read, write) = (state.clone(), state.clone());
1324        let speed = Binding::new(
1325            move || {
1326                let speed = read.selected_clip().map_or(1.0, |(_, clip)| clip.speed);
1327                SPEEDS.iter().position(|s| (*s - speed).abs() < 0.01).unwrap_or(1)
1328            },
1329            move |choice: usize| write.set_speed(SPEEDS.get(choice).copied().unwrap_or(1.0)),
1330        );
1331        let (read, write) = (state.clone(), state.clone());
1332        let density = Binding::new(
1333            move || read.inner.density.get(),
1334            move |value: f32| {
1335                write.inner.density.set(value);
1336                write.touch();
1337            },
1338        );
1339        let tool = |label: &str, icon: IconName| {
1340            Button::new(label.to_owned()).icon(icon).style(ButtonStyle::Ghost).size(ButtonSize::Small)
1341        };
1342        HStack::new((
1343            tool(&labels.split, IconName::Scissors).on_click(move || split.split()),
1344            tool(&labels.delete, IconName::Trash2)
1345                .disabled(!selected)
1346                .on_click(move || delete.delete_selected()),
1347            tool(&labels.chapter, IconName::Flag).on_click(move || chapter.add_chapter(&new_chapter)),
1348            speed_clip.map(|_| {
1349                HStack::new((
1350                    muted(&labels.speed),
1351                    Select::new(speed, SPEEDS.iter().map(|s| format!("{s}×"))).width(90.0),
1352                ))
1353                .spacing(6.0)
1354                .align(Align::Center)
1355            }),
1356            Spacer::new(),
1357            Slider::new(density).range(5.0, 200.0).step(1.0).label(labels.timeline_zoom.clone()).width(130.0),
1358        ))
1359        .spacing(4.0)
1360        .align(Align::Center)
1361        .padding((0.0, 12.0))
1362        .frame(None, Some(TIMELINE_BAR))
1363    }

1366impl View for MovieEditor {
1367    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1368        let _ = self.state.revision.get();
1369        let now = cx.now();
1370        self.state.advance(now);
1371        if self.state.inner.running.get().is_some() {
1372            cx.wake_at(now + FRAME);
1373        }
1374        let (undo, redo, delete) = (self.state.clone(), self.state.clone(), self.state.clone());
1375        cx.shortcut(Key::Character("z".into()), true, Rc::new(move || undo.undo()));
1376        cx.shortcut(Key::Character("y".into()), true, Rc::new(move || redo.redo()));
1377        cx.shortcut(Key::Delete, false, Rc::new(move || delete.delete_selected()));
1378
1379        let viewport = self.area.unwrap_or_else(|| cx.viewport());
1380        let movie = self.state.movie();
1381        let logical = movie.logical_size();
1382        let timeline = timeline(&self.state, &self.labels);
1383        let timeline_height = timeline.total_height() + TIMELINE_BAR;
1384        let middle_height = (viewport.height - TOP_BAR - timeline_height).max(200.0);
1385        let by_width = (viewport.width - RAIL_WIDTH - LIBRARY_WIDTH - PROPERTIES_WIDTH - 48.0).max(240.0);
1386        let by_height =
1387            (middle_height - TRANSPORT - 16.0).max(120.0) * logical.width / logical.height.max(1.0);
1388        let canvas_width = by_width.min(by_height);
1389
1390        let center: AnyView = if self.state.inner.trying.get() {
1391            Box::new(
1392                ScrollView::new((MoviePlayer::new(movie.clone())
1393                    .labels(self.player_labels.clone())
1394                    .width(by_width.max(480.0)),))
1395                .padding(8.0)
1396                .grow(1.0),
1397            )
1398        } else {
1399            Box::new(
1400                VStack::new((
1401                    Spacer::new(),
1402                    Canvas { state: self.state.clone(), width: canvas_width },
1403                    Spacer::new(),
1404                    self.transport(),
1405                ))
1406                .spacing(0.0)
1407                .align(Align::Center)
1408                .grow(1.0),
1409            )
1410        };
1411        let tree = VStack::new((
1412            self.top_bar(),
1413            Divider::horizontal(),
1414            HStack::new((self.rail(), self.library(), center, self.properties()))
1415                .spacing(0.0)
1416                .align(Align::Stretch)
1417                .grow(1.0),
1418            Divider::horizontal(),
1419            VStack::new((self.timeline_bar(), timeline))
1420                .spacing(0.0)
1421                .align(Align::Stretch)
1422                .background(Semantic::Surface),
1423        ))
1424        .spacing(0.0)
1425        .align(Align::Stretch)
1426        .grow(1.0);
1427        cx.child(&tree)
1428    }
