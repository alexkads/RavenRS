299/// What is being edited, kept by the app so that it can be read and saved. All
300/// the edits the editor can make are methods here too, for scripts and tests.

302pub struct EditorState {
303    inner: Rc<Inner>,
304    revision: State<u64>,
305}

307impl EditorState {
308    /// Starts editing `movie`, with whatever edits it already has.
309    pub fn new(movie: Movie) -> Self {
310        Self {
311            inner: Rc::new(Inner {
312                movie: RefCell::new(movie),
313                position: Cell::new(0),
314                running: Cell::new(None),
315                selection: Cell::new(Selection::Nothing),
316                undo: RefCell::new(Vec::new()),
317                redo: RefCell::new(Vec::new()),
318                last_edit: RefCell::new(None),
319                density: Cell::new(28.0),
320                tab: Cell::new(Tab::Annotations),
321                trying: Cell::new(false),
322            }),
323            revision: State::new(0),
324        }
325    }
326
327    /// The film with the edits so far.
328    pub fn movie(&self) -> Movie {
329        self.inner.movie.borrow().clone()
330    }
331
332    fn touch(&self) {
333        self.revision.update(|n| *n += 1);
334    }
335
336    /// Makes an edit. Edits with the same `key` in a row count as one undo.
337    fn edit(&self, key: Option<&str>, work: impl FnOnce(&mut Project)) {
338        let inner = &self.inner;
339        let mut project = inner.movie.borrow().project().clone();
340        let same = key.is_some() && inner.last_edit.borrow().as_deref() == key;
341        if !same {
342            let mut undo = inner.undo.borrow_mut();
343            undo.push(project.clone());
344            if undo.len() > UNDO_LIMIT {
345                undo.remove(0);
346            }
347            inner.redo.borrow_mut().clear();
348        }
349        *inner.last_edit.borrow_mut() = key.map(str::to_owned);
350        work(&mut project);
351        let movie = inner.movie.borrow().with_project(project);
352        *inner.movie.borrow_mut() = movie;
353        self.keep_position_inside();
354        self.touch();
355    }
356
357    fn keep_position_inside(&self) {
358        let duration = self.movie().duration_ms();
359        if self.inner.position.get() > duration {
360            self.inner.position.set(duration);
361        }
362    }
363
364    fn swap(&self, from: &RefCell<Vec<Project>>, to: &RefCell<Vec<Project>>) {
365        let Some(project) = from.borrow_mut().pop() else {
366            return;
367        };
368        to.borrow_mut().push(self.movie().project().clone());
369        let movie = self.movie().with_project(project);
370        *self.inner.movie.borrow_mut() = movie;
371        *self.inner.last_edit.borrow_mut() = None;
372        self.inner.selection.set(Selection::Nothing);
373        self.keep_position_inside();
374        self.touch();
375    }
376
377    /// Takes the last edit back.
378    pub fn undo(&self) {
379        self.swap(&self.inner.undo, &self.inner.redo);
380    }
381
382    /// Puts back what [`Self::undo`] took.
383    pub fn redo(&self) {
384        self.swap(&self.inner.redo, &self.inner.undo);
385    }
386
387    /// Where the playhead is, in milliseconds of the timeline.
388    pub fn position_ms(&self) -> u64 {
389        self.inner.position.get()
390    }
391
392    /// Moves the playhead.
393    pub fn seek(&self, ms: u64) {
394        self.inner.position.set(ms.min(self.movie().duration_ms()));
395        if self.inner.running.get().is_some() {
396            self.inner.running.set(Some((f64::NAN, self.inner.position.get())));
397        }
398        self.touch();
399    }
400
401    /// Plays from the playhead, or pauses.
402    pub fn toggle_playing(&self) {
403        if self.inner.running.get().is_some() {
404            self.inner.running.set(None);
405        } else {
406            if self.position_ms() >= self.movie().duration_ms() {
407                self.inner.position.set(0);
408            }
409            self.inner.running.set(Some((f64::NAN, self.position_ms())));
410        }
411        self.touch();
412    }
413
414    /// Cuts the clip at the playhead in two.
415    pub fn split(&self) {
416        let at = self.position_ms();
417        self.edit(None, |project| {
418            project.split(at);
419        });
420    }
421
422    /// Deletes what is selected: a clip takes its time with it.
423    pub fn delete_selected(&self) {
424        match self.inner.selection.get() {
425            Selection::Clip(index) => self.edit(None, |project| project.delete_clip(index)),
426            Selection::Overlay(id) => self.edit(None, |project| project.remove_overlay(id)),
427            Selection::Nothing => return,
428        }
429        self.inner.selection.set(Selection::Nothing);
430        self.touch();
431    }
432
433    /// Selects the clip at this place on the main track.
434    pub fn select_clip(&self, index: usize) {
435        self.inner.selection.set(Selection::Clip(index));
436        *self.inner.last_edit.borrow_mut() = None;
437        self.touch();
438    }
439
440    /// Selects an overlay.
441    pub fn select_overlay(&self, id: u64) {
442        self.inner.selection.set(Selection::Overlay(id));
443        *self.inner.last_edit.borrow_mut() = None;
444        self.touch();
445    }
446
447    /// Adds an effect at the playhead and selects it.
448    pub fn add_overlay(&self, kind: OverlayKind) -> u64 {
449        let at = self.position_ms();
450        let mut id = 0;
451        self.edit(None, |project| {
452            let length = NEW_EFFECT_MS.min(project.duration_ms().saturating_sub(at).max(500));
453            id = project.add_overlay(kind, at, length);
454        });
455        self.inner.selection.set(Selection::Overlay(id));
456        self.touch();
457        id
458    }
459
460    /// Puts a title card at the playhead and selects it.
461    pub fn add_title(&self, title: &str) {
462        let at = self.position_ms();
463        let mut index = 0;
464        self.edit(None, |project| index = project.insert_title(at, title, "", NEW_EFFECT_MS));
465        self.inner.selection.set(Selection::Clip(index));
466        self.touch();
467    }
468
469    /// Starts a chapter at the playhead.
470    pub fn add_chapter(&self, title: &str) {
471        let at_ms = self.position_ms();
472        self.edit(None, |project| project.markers.push(Marker { title: title.to_owned(), at_ms }));
473    }
474
475    /// Changes how fast the selected clip plays.
476    pub fn set_speed(&self, speed: f32) {
477        if let Selection::Clip(index) = self.inner.selection.get() {
478            self.edit(None, |project| project.set_speed(index, speed));
479        }
480    }
481
482    fn change_overlay(&self, id: u64, key: &str, work: impl FnOnce(&mut Overlay)) {
483        self.edit(Some(&format!("{key}:{id}")), |project| {
484            if let Some(overlay) = project.overlay_mut(id) {
485                work(overlay);
486            }
487        });
488    }
489
490    fn selected_overlay(&self) -> Option<Overlay> {
491        let Selection::Overlay(id) = self.inner.selection.get() else {
492            return None;
493        };
494        self.movie().project().overlays.iter().find(|o| o.id == id).cloned()
495    }
496
497    fn selected_clip(&self) -> Option<(usize, Clip)> {
498        let Selection::Clip(index) = self.inner.selection.get() else {
499            return None;
500        };
501        self.movie().project().clips.get(index).cloned().map(|clip| (index, clip))
502    }
503
504    /// Runs the clock while playing. Called while building: only plain cells change.
505    fn advance(&self, now: f64) {
506        let Some((since, from)) = self.inner.running.get() else {
507            return;
508        };
509        if since.is_nan() {
510            self.inner.running.set(Some((now, from)));
511            return;
512        }
513        let at = from + ((now - since).max(0.0) * 1000.0) as u64;
514        let duration = self.movie().duration_ms();
515        if at >= duration {
516            self.inner.position.set(duration);
517            self.inner.running.set(None);
518        } else {
519            self.inner.position.set(at);
520        }
521    }
522}

560/// What an overlay goes under on the timeline.
561#[derive(Clone, Copy, PartialEq, Eq)]
562enum Row {
563    Interaction,
564    Annotations,
565    Screen,
566    Pointer,
567}
568
569fn row_of(kind: &OverlayKind) -> Row {
570    match kind {
571        OverlayKind::Hotspot { .. } => Row::Interaction,
572        OverlayKind::Callout { .. } | OverlayKind::Caption { .. } => Row::Annotations,
573        OverlayKind::PointerHighlight => Row::Pointer,
574        _ => Row::Screen,
575    }
576}

578/// The ids blocks have on the timeline: clips below, overlays above.
579const OVERLAY_IDS: u64 = 1 << 32;
580/// The steps that came with the film, which only say where they are.
581const FILM_STEP_IDS: u64 = 2 << 32;
582/// A step lasts no time, but it has to be seen and clicked.
583const STEP_MS: u64 = 700;
584
585fn on_timeline_edit(state: &EditorState, edit: TimelineEdit) {
586    match edit {
587        TimelineEdit::Began { .. } => *state.inner.last_edit.borrow_mut() = None,
588        TimelineEdit::Moved { id, start_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
589            let id = id - OVERLAY_IDS;
590            state.inner.selection.set(Selection::Overlay(id));
591            state.change_overlay(id, "move", |overlay| overlay.start_ms = start_ms);
592        }
593        TimelineEdit::TrimmedStart { id, by_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
594            state.change_overlay(id - OVERLAY_IDS, "resize", |overlay| {
595                let end = overlay.end_ms();
596                let start = (overlay.start_ms as i64 + by_ms).clamp(0, end as i64 - 100) as u64;
597                overlay.start_ms = start;
598                overlay.duration_ms = end - start;
599            });
600        }
601        TimelineEdit::TrimmedEnd { id, by_ms } if (OVERLAY_IDS..FILM_STEP_IDS).contains(&id) => {
602            state.change_overlay(id - OVERLAY_IDS, "resize", |overlay| {
603                overlay.duration_ms = (overlay.duration_ms as i64 + by_ms).max(100) as u64;
604            });
605        }
606        TimelineEdit::TrimmedStart { id, by_ms } => {
607            let index = id as usize;
608            state.edit(Some(&format!("trim:{index}")), |project| project.trim_start(index, by_ms));
609        }
610        TimelineEdit::TrimmedEnd { id, by_ms } => {
611            let index = id as usize;
612            state.edit(Some(&format!("trim:{index}")), |project| project.trim_end(index, by_ms));
613        }
614        _ => {}
615    }
616}

618fn timeline(state: &EditorState, labels: &EditorLabels) -> Timeline {
619    let movie = state.movie();
620    let project = movie.project();
621    let selection = state.inner.selection.get();
622    let overlays_in = |row: Row| -> Vec<TimelineItem> {
623        project
624            .overlays
625            .iter()
626            .filter(|overlay| row_of(&overlay.kind) == row)
627            .map(|overlay| {
628                let tone = match row {
629                    Row::Interaction => Tone::Warning,
630                    Row::Annotations => Tone::Accent,
631                    Row::Screen => Tone::Success,
632                    Row::Pointer => Tone::Neutral,
633                };
634                TimelineItem::new(
635                    overlay.id + OVERLAY_IDS,
636                    overlay.start_ms,
637                    overlay.duration_ms,
638                    label_of(&overlay.kind, labels),
639                )
640                .tone(tone)
641                .selected(selection == Selection::Overlay(overlay.id))
642                .movable()
643                .trimmable()
644            })
645            .collect()
646    };
647    let clips: Vec<TimelineItem> = project
648        .clips
649        .iter()
650        .enumerate()
651        .map(|(index, clip)| {
652            let start = project.clip_start_ms(index);
653            let label = match &clip.kind {
654                ClipKind::Film { .. } if (clip.speed - 1.0).abs() > 0.01 => {
655                    format!("{} {} · {}×", labels.film_clip, index + 1, clip.speed)
656                }
657                ClipKind::Film { .. } => format!("{} {}", labels.film_clip, index + 1),
658                ClipKind::Title { title, .. } => title.clone(),
659            };
660            let moment = if start == 0 { movie.poster_ms() } else { start + 1 };
661            let item = TimelineItem::new(index as u64, start, clip.duration_ms(), label)
662                .tone(if matches!(clip.kind, ClipKind::Title { .. }) { Tone::Success } else { Tone::Accent })
663                .selected(selection == Selection::Clip(index));
664            // What a stretch of film starts with, in vectors like the rest. A card
665            // has its words on the block already, and they would not fit in a thumbnail.
666            if matches!(clip.kind, ClipKind::Film { .. }) {
667                item.leading(Composition::new(movie.clone(), moment).width(THUMBNAIL)).trimmable()
668            } else {
669                item
670            }
671        })
672        .collect();
673    // The steps recorded with the film, beside those added here.
674    let added: Vec<&str> = project
675        .overlays
676        .iter()
677        .filter_map(|overlay| match &overlay.kind {
678            OverlayKind::Hotspot { title, .. } => Some(title.as_str()),
679            _ => None,
680        })
681        .collect();
682    let mut steps = overlays_in(Row::Interaction);
683    for (index, spot) in movie.hotspots().iter().enumerate() {
684        if !added.contains(&spot.title.as_str()) {
685            let length = spot.until_ms.saturating_sub(spot.from_ms).max(STEP_MS);
686            steps.push(
687                TimelineItem::new(FILM_STEP_IDS + index as u64, spot.from_ms, length, spot.title.clone())
688                    .tone(Tone::Warning),
689            );
690        }
691    }
692    let starts: Vec<u64> = movie.hotspots().iter().map(|spot| spot.from_ms).collect();
693    let (select, seek, edit) = (state.clone(), state.clone(), state.clone());
694    Timeline::new(vec![
695        TimelineTrack::new(labels.tab_interaction.clone(), OVERLAY_ROW, steps),
696        TimelineTrack::new(labels.tab_annotations.clone(), OVERLAY_ROW, overlays_in(Row::Annotations)),
697        TimelineTrack::new(labels.tab_screen.clone(), OVERLAY_ROW, overlays_in(Row::Screen)),
698        TimelineTrack::new(labels.tab_pointer.clone(), OVERLAY_ROW, overlays_in(Row::Pointer)),
699        TimelineTrack::new(labels.track_film.clone(), CLIP_ROW, clips),
700    ])
701    .markers(
702        movie.chapters().iter().map(|c| TimelineMarker { at_ms: c.at_ms, title: c.title.clone() }).collect(),
703    )
704    .position_ms(state.position_ms())
705    .pixels_per_second(state.inner.density.get())
706    .heads_width(TRACK_HEADS)
707    .on_seek(move |ms| seek.seek(ms))
708    .on_select(move |id| {
709        if id >= FILM_STEP_IDS {
710            if let Some(at) = usize::try_from(id - FILM_STEP_IDS).ok().and_then(|i| starts.get(i)) {
711                select.seek(*at);
712            }
713        } else if id >= OVERLAY_IDS {
714            select.select_overlay(id - OVERLAY_IDS);
715        } else {
716            select.select_clip(id as usize);
717        }
718    })
719    .on_edit(move |change| on_timeline_edit(&edit, change))
720}

722/// The selected region on the canvas, to drag and resize.
723struct RegionHandle {
724    state: EditorState,
725    id: u64,
726    frame: Rect,
727    scale: f32,
728}
729
730impl View for RegionHandle {
731    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
732        let theme = *cx.theme();
733        let color = theme.color(Semantic::FocusRing);
734        let (state, id, scale) = (self.state.clone(), self.id, self.scale);
735        let size = HANDLE * 1.5;
736        let corner = cx.container(
737            LayoutStyle {
738                absolute: Some((self.frame.width() - size, self.frame.height() - size)),
739                width: Some(size),
740                height: Some(size),
741                ..LayoutStyle::stack(Axis::Vertical)
742            },
743            &[],
744            Paint::Quad(QuadPaint { corner_radius: 2.0, ..QuadPaint::new(color) }),
745        );
746        cx.semantics(corner, Semantics::hidden());
747        let start = cx.use_state(|| Rc::new(Cell::new(None::<(Point, Region)>))).get();
748        let resize = state.clone();
749        cx.on_drag(
750            corner,
751            Rc::new(move |drag: Drag| {
752                if let Some(mut region) = resize.selected_overlay().and_then(|o| region_of(&o.kind)) {
753                    if drag.phase == DragPhase::Start {
754                        *resize.inner.last_edit.borrow_mut() = None;
755                    }
756                    region.width = ((drag.position.x - drag.container.x()) / scale).max(16.0);
757                    region.height = ((drag.position.y - drag.container.y()) / scale).max(16.0);
758                    resize.change_overlay(id, "region", |overlay| set_region(&mut overlay.kind, region));
759                }
760            }),
761        );
762        cx.cursor(corner, CursorIcon::ColResize);
763        let node = cx.container(
764            LayoutStyle {
765                absolute: Some((self.frame.x(), self.frame.y())),
766                width: Some(self.frame.width()),
767                height: Some(self.frame.height()),
768                ..LayoutStyle::stack(Axis::Vertical)
769            },
770            &[corner],
771            Paint::Quad(QuadPaint {
772                fill: Color::TRANSPARENT.into(),
773                corner_radius: 0.0,
774                border: Some(Border { width: 2.0, color }),
775                shadow: None,
776            }),
777        );
778        cx.on_drag(
779            node,
780            Rc::new(move |drag: Drag| {
781                let Some(region) = state.selected_overlay().and_then(|o| region_of(&o.kind)) else {
782                    return;
783                };
784                if drag.phase == DragPhase::Start {
785                    start.set(Some((drag.position, region)));
786                    *state.inner.last_edit.borrow_mut() = None;
787                    return;
788                }
789                if let Some((from, was)) = start.get() {
790                    let moved = Region {
791                        x: was.x + (drag.position.x - from.x) / scale,
792                        y: was.y + (drag.position.y - from.y) / scale,
793                        ..was
794                    };
795                    state.change_overlay(id, "region", |overlay| set_region(&mut overlay.kind, moved));
796                }
797            }),
798        );
799        cx.cursor(node, CursorIcon::Pointer);
800        cx.semantics(node, Semantics::hidden());
801        node
802    }
803}

805/// The canvas: the moment under the playhead, and the region being placed.
806struct Canvas {
807    state: EditorState,
808    width: f32,
809}
810
811impl View for Canvas {
812    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
813        let state = &self.state;
814        let movie = state.movie();
815        let logical = movie.logical_size();
816        let height = self.width * logical.height / logical.width.max(1.0);
817        let selected = state.selected_overlay();
818        let placing_zoom = matches!(selected.as_ref().map(|o| &o.kind), Some(OverlayKind::Zoom { .. }));
819        let at = state.position_ms();
820        let mut film = Composition::new(movie.clone(), at).width(self.width);
821        if placing_zoom {
822            film = film.ignoring_zoom();
823        }
824        let mut children = vec![cx.child(&film)];
825        if let Some(overlay) = &selected
826            && let Some(region) = region_of(&overlay.kind)
827            && overlay.covers(at)
828        {
829            let camera = camera_at(&movie, at, self.width, placing_zoom);
830            children.push(cx.child(&RegionHandle {
831                state: state.clone(),
832                id: overlay.id,
833                frame: camera.region(region),
834                scale: camera.scale(),
835            }));
836        }
837        let node = cx.container(
838            LayoutStyle {
839                width: Some(self.width),
840                height: Some(height),
841                overflow_hidden: true,
842                ..LayoutStyle::stack(Axis::Vertical)
843            },
844            &children,
845            Paint::None,
846        );
847        cx.clip(node);
848        node
849    }

855/// An editor for a film, in the style of screen-recording editors.
856///
857/// Tools are on the left, with what each offers; the canvas in the middle; what
858/// is selected on the right; and the tracks along the bottom. Clips are cut at
859/// the playhead, trimmed at their edges and deleted, closing the gap. Effects sit
860/// on tracks over them, where they slide and stretch, and regions are dragged on
861/// the canvas. Nothing changes the film, every edit can be undone, and the film
862/// can be tried at any time with the same player people will watch it in.
863///
864/// ```no_run
865/// use ravenrs_movie::{EditorState, Movie, MovieEditor};
866///
867/// let film = Movie::from_bytes(&std::fs::read("manual.ravenmovie").unwrap_or_default())?;
868/// let state = EditorState::new(film);
869/// let editor = MovieEditor::new(state).title("Getting started").on_save(|film: &Movie| {
870///     let _ = std::fs::write("manual.ravenmovie", film.to_bytes());
871/// });
872/// # Ok::<(), ravenrs_movie::MovieError>(())
873/// ```
874pub struct MovieEditor {
875    state: EditorState,
876    labels: EditorLabels,
877    player_labels: MovieLabels,
878    title: String,
879    on_save: Option<SaveHandler>,
880    area: Option<Size>,
881}
882
883impl std::fmt::Debug for MovieEditor {
884    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
885        f.debug_struct("MovieEditor").field("state", &self.state).finish_non_exhaustive()
886    }
887}

1218                },
1219            );
1220            items.push(Box::new(muted(&format!(
1221                "{}: {:.1} s",
1222                labels.duration,
1223                overlay.duration_ms as f32 / 1000.0
1224            ))));
1225            items.push(Box::new(
1226                Slider::new(duration).range(0.5, total.max(1.0)).step(0.1).label(labels.duration.clone()),
1227            ));
1228            if region_of(&overlay.kind).is_some() {
1229                let axis = |state: &EditorState, vertical: bool, name: &str| {
1230                    let (read, write) = (state.clone(), state.clone());
1231                    let value = Binding::new(
1232                        move || {
1233                            let region = read.selected_overlay().and_then(|o| region_of(&o.kind));
1234                            region.map_or(0, |r| if vertical { r.y as i32 } else { r.x as i32 })
1235                        },
1236                        move |new: i32| {
1237                            write.change_overlay(id, "region", |overlay| {
1238                                if let Some(mut region) = region_of(&overlay.kind) {
1239                                    if vertical {
1240                                        region.y = new as f32;
1241                                    } else {
1242                                        region.x = new as f32;
1243                                    }
1244                                    set_region(&mut overlay.kind, region);
1245                                }
1246                            });
1247                        },
1248                    );
1249                    Stepper::new(value).range(0, 8000).label(name.to_owned())
1250                };
1251                items.push(Box::new(axis(state, false, "X")));
1252                items.push(Box::new(axis(state, true, "Y")));
1253                items.push(Box::new(muted(&labels.drag_hint)));
1254            }
1255        } else if let Some((index, clip)) = state.selected_clip() {
1256            match &clip.kind {
1257                ClipKind::Film { .. } => {
1258                    items.push(Box::new(panel_title(&labels.film_clip)));

1260                }
1261                ClipKind::Title { .. } => {

1263                    let words = |second: bool| {

1266                            move || match read.selected_clip().map(|(_, clip)| clip.kind) {
1267                                Some(ClipKind::Title { title, subtitle, .. }) => {
1268                                    if second {
1269                                        subtitle
1270                                    } else {
1271                                        title
1272                                    }
1273                                }
1274                                _ => String::new(),
1275                            },
1276                            move |new: String| {
1277                                write.edit(Some(&format!("card:{index}:{second}")), |project| {
1278                                    if let Some(ClipKind::Title { title, subtitle, .. }) =
1279                                        project.clips.get_mut(index).map(|clip| &mut clip.kind)
1280                                    {
1281                                        *(if second { subtitle } else { title }) = new;
1282                                    }
1283                                });
1284                            },
1285                        )
1286                    };
1287                    items.push(Box::new(muted(&labels.text)));
1288                    items.push(Box::new(TextField::new(words(false))));
1289                    items.push(Box::new(muted(&labels.subtitle)));
1290                    items.push(Box::new(TextField::new(words(true))));
1291                }
1292            }
1293        } else {
1294            items.push(Box::new(panel_title(&labels.properties)));
1295            items.push(Box::new(muted(&labels.nothing_selected)));
1296        }
1297        ScrollView::new((VStack::new(items).spacing(10.0).align(Align::Stretch),))
1298            .padding(12.0)
1299            .background(Semantic::Surface)
1300            .frame(Some(PROPERTIES_WIDTH), None)
1301    }

1303    fn transport(&self) -> impl View + use<> {
1304        let labels = &self.labels;
1305        let state = &self.state;
1306        let running = state.inner.running.get().is_some();
1307        let movie = state.movie();
1308        let at = state.position_ms();
1309        let chapter_to = |target: Option<u64>| {
1310            let seek = state.clone();
1311            move || {
1312                if let Some(ms) = target {
1313                    seek.seek(ms);
1314                }
1315            }
1316        };
1317        let previous = movie.chapters().iter().rev().find(|c| c.at_ms < at).map(|c| c.at_ms);
1318        let next = movie.chapters().iter().find(|c| c.at_ms > at).map(|c| c.at_ms);
1319        let toggle = state.clone();
1320        HStack::new((
1321            IconButton::new(IconName::ChevronLeft)
1322                .label(labels.previous_chapter.clone())
1323                .disabled(previous.is_none())
1324                .on_click(chapter_to(previous)),
1325            Button::new(if running { labels.pause.clone() } else { labels.play.clone() })
1326                .icon(if running { IconName::Pause } else { IconName::Play })
1327                .size(ButtonSize::Small)
1328                .on_click(move || toggle.toggle_playing()),
1329            IconButton::new(IconName::ChevronRight)
1330                .label(labels.next_chapter.clone())
1331                .disabled(next.is_none())
1332                .on_click(chapter_to(next)),
1333            Text::new(format!("{} / {}", clock(at), clock(movie.duration_ms()))).style(TextStyle::Callout),
1334        ))
1335        .spacing(6.0)
1336        .align(Align::Center)
1337        .frame(None, Some(TRANSPORT))
1338    }

1340    fn timeline_bar(&self) -> impl View + use<> {
1341        let labels = &self.labels;
1342        let state = &self.state;
1343        let (split, delete, chapter) = (state.clone(), state.clone(), state.clone());
1344        let new_chapter = labels.new_chapter.clone();
1345        let selected = state.inner.selection.get() != Selection::Nothing;
1346        let speed_clip = state.selected_clip().filter(|(_, clip)| matches!(clip.kind, ClipKind::Film { .. }));
1347        let (read, write) = (state.clone(), state.clone());
1348        let speed = Binding::new(
1349            move || {
1350                let speed = read.selected_clip().map_or(1.0, |(_, clip)| clip.speed);
1351                SPEEDS.iter().position(|s| (*s - speed).abs() < 0.01).unwrap_or(1)
1352            },
1353            move |choice: usize| write.set_speed(SPEEDS.get(choice).copied().unwrap_or(1.0)),
1354        );
1355        let (read, write) = (state.clone(), state.clone());
1356        let density = Binding::new(
1357            move || read.inner.density.get(),
1358            move |value: f32| {
1359                write.inner.density.set(value);
1360                write.touch();
1361            },
1362        );
1363        let tool = |label: &str, icon: IconName| {
1364            Button::new(label.to_owned()).icon(icon).style(ButtonStyle::Ghost).size(ButtonSize::Small)
1365        };
1366        HStack::new((
1367            tool(&labels.split, IconName::Scissors).on_click(move || split.split()),
1368            tool(&labels.delete, IconName::Trash2)
1369                .disabled(!selected)
1370                .on_click(move || delete.delete_selected()),
1371            tool(&labels.chapter, IconName::Flag).on_click(move || chapter.add_chapter(&new_chapter)),
1372            speed_clip.map(|_| {
1373                HStack::new((
1374                    muted(&labels.speed),
1375                    Select::new(speed, SPEEDS.iter().map(|s| format!("{s}×"))).width(90.0),
1376                ))
1377                .spacing(6.0)
1378                .align(Align::Center)
1379            }),
1380            Spacer::new(),
1381            Slider::new(density).range(5.0, 200.0).step(1.0).label(labels.timeline_zoom.clone()).width(130.0),
1382        ))
1383        .spacing(4.0)
1384        .align(Align::Center)
1385        .padding((0.0, 12.0))
1386        .frame(None, Some(TIMELINE_BAR))
1387    }

1390impl View for MovieEditor {
1391    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
1392        let _ = self.state.revision.get();
1393        let now = cx.now();
1394        self.state.advance(now);
1395        if self.state.inner.running.get().is_some() {
1396            cx.wake_at(now + FRAME);
1397        }
1398        let (undo, redo, delete) = (self.state.clone(), self.state.clone(), self.state.clone());
1399        cx.shortcut(Key::Character("z".into()), true, Rc::new(move || undo.undo()));
1400        cx.shortcut(Key::Character("y".into()), true, Rc::new(move || redo.redo()));
1401        cx.shortcut(Key::Delete, false, Rc::new(move || delete.delete_selected()));
1402
1403        let viewport = self.area.unwrap_or_else(|| cx.viewport());
1404        let movie = self.state.movie();
1405        let logical = movie.logical_size();
1406        let timeline = timeline(&self.state, &self.labels);
1407        let timeline_height = timeline.total_height() + TIMELINE_BAR;
1408        let middle_height = (viewport.height - TOP_BAR - timeline_height).max(200.0);
1409        let by_width = (viewport.width - RAIL_WIDTH - LIBRARY_WIDTH - PROPERTIES_WIDTH - 48.0).max(240.0);
1410        let by_height =
1411            (middle_height - TRANSPORT - 16.0).max(120.0) * logical.width / logical.height.max(1.0);
1412        let canvas_width = by_width.min(by_height);
1413
1414        let center: AnyView = if self.state.inner.trying.get() {
1415            Box::new(
1416                ScrollView::new((MoviePlayer::new(movie.clone())
1417                    .labels(self.player_labels.clone())
1418                    .width(by_width.max(480.0)),))
1419                .padding(8.0)
1420                .grow(1.0),
1421            )
1422        } else {
1423            Box::new(
1424                VStack::new((
1425                    Spacer::new(),
1426                    Canvas { state: self.state.clone(), width: canvas_width },
1427                    Spacer::new(),
1428                    self.transport(),
1429                ))
1430                .spacing(0.0)
1431                .align(Align::Center)
1432                .grow(1.0),
1433            )
1434        };
1435        let tree = VStack::new((
1436            self.top_bar(),
1437            Divider::horizontal(),
1438            HStack::new((self.rail(), self.library(), center, self.properties()))
1439                .spacing(0.0)
1440                .align(Align::Stretch)
1441                .grow(1.0),
1442            Divider::horizontal(),
1443            VStack::new((self.timeline_bar(), timeline))
1444                .spacing(0.0)
1445                .align(Align::Stretch)
1446                .background(Semantic::Surface),
1447        ))
1448        .spacing(0.0)
1449        .align(Align::Stretch)
1450        .grow(1.0);
1451        cx.child(&tree)
1452    }
