24/// What can go wrong with a film.

35impl std::fmt::Display for MovieError {
36    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
37        match self {
38            Self::Damaged(why) => write!(f, "the film is damaged: {why}"),
39            Self::Empty => f.write_str("the film has no frames"),
40        }
41    }

287    /// A moment to show before it plays: past any fade from black at the start, and

349    /// Where the pointer was at this moment of the film.
350    pub fn pointer_at(&self, film_ms: u64) -> Option<Point> {
351        let keys = &self.film.header.pointer;
352        let after = keys.partition_point(|key| key.at_ms <= film_ms);
353        match (after.checked_sub(1).and_then(|i| keys.get(i)), keys.get(after)) {
354            (Some(a), Some(b)) if b.at_ms > a.at_ms => {
355                let t = (film_ms - a.at_ms) as f32 / (b.at_ms - a.at_ms) as f32;
356                Some(Point::new(a.x + (b.x - a.x) * t, a.y + (b.y - a.y) * t))
357            }
358            (Some(a), _) => Some(Point::new(a.x, a.y)),
359            (None, Some(b)) => Some(Point::new(b.x, b.y)),
360            (None, None) => None,
361        }
362    }

371    /// The whole film as the bytes of a file, with its edits.
372    pub fn to_bytes(&self) -> Vec<u8> {
373        let header = serde_json::to_vec(&self.film.header).unwrap_or_default();
374        let project = serde_json::to_vec(&self.project).unwrap_or_default();
375        let body = deflate(&header);
376        let edits = deflate(&project);
377        let mut bytes = Vec::new();
378        bytes.extend_from_slice(MAGIC);
379        for part in [&body, &edits] {
380            bytes.extend_from_slice(&(part.len() as u64).to_le_bytes());
381            bytes.extend_from_slice(part);
382        }
383        for blob in &self.film.blobs {
384            bytes.extend_from_slice(blob);
385        }
386        bytes
387    }

389    /// Reads a film back from the bytes of a file.
390    pub fn from_bytes(bytes: &[u8]) -> Result<Self, MovieError> {
391        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
392        let mut rest = bytes.strip_prefix(MAGIC).ok_or_else(|| damaged("it does not start like a film"))?;
393        let mut part = || -> Result<Vec<u8>, MovieError> {
394            let (length, after) =
395                rest.split_first_chunk::<8>().ok_or_else(|| damaged("a part is missing"))?;
396            let length =
397                usize::try_from(u64::from_le_bytes(*length)).map_err(|_| damaged("a part is too long"))?;
398            let data = after.get(..length).ok_or_else(|| damaged("a part was cut"))?;
399            rest = &after[length..];
400            inflate(data)
401        };
402        let header = part()?;
403        let project = part()?;
404        let header: Header =
405            serde_json::from_slice(&header).map_err(|e| MovieError::Damaged(e.to_string()))?;
406        let project: Project =
407            serde_json::from_slice(&project).map_err(|e| MovieError::Damaged(e.to_string()))?;
408        let mut blobs = Vec::with_capacity(header.blob_lengths.len());
409        for length in &header.blob_lengths {
410            let length = usize::try_from(*length).map_err(|_| damaged("a blob is too long"))?;
411            let blob = rest.get(..length).ok_or_else(|| damaged("a blob was cut"))?;
412            blobs.push(blob.to_vec());
413            rest = &rest[length..];
414        }
415        if header.frames.is_empty() {
416            return Err(MovieError::Empty);
417        }
418        let film = Film::open(header, blobs)?;
419        Ok(Self::with(Rc::new(film), project))
420    }

423/// Chapters and hotspots, moved to where the edits put them, with the hotspots
424/// added on the timeline.
425fn edited_marks(header: &Header, project: &Project) -> (Vec<Chapter>, Vec<Hotspot>) {
426    let mut chapters: Vec<Chapter> = header
427        .chapters
428        .iter()
429        .filter_map(|chapter| {
430            // If its first moments were cut, it starts where the film picks up again.
431            let next = project.clips.iter().filter_map(|clip| match clip.kind {
432                crate::project::ClipKind::Film { from_ms, .. } if from_ms >= chapter.at_ms => Some(from_ms),
433                _ => None,
434            });
435            let at = project.film_to_timeline(chapter.at_ms).or_else(|| {
436                let end = header
437                    .chapters
438                    .iter()
439                    .map(|c| c.at_ms)
440                    .find(|at| *at > chapter.at_ms)
441                    .unwrap_or(u64::MAX);
442                next.filter(|from| *from < end).min().and_then(|from| project.film_to_timeline(from))
443            })?;
444            Some(Chapter { at_ms: at, ..chapter.clone() })
445        })
446        .collect();
447    chapters.extend(project.markers.iter().map(|marker| Chapter {
448        title: marker.title.clone(),
449        text: String::new(),
450        at_ms: marker.at_ms,
451    }));
452    chapters.sort_by_key(|chapter| chapter.at_ms);
453    let mut hotspots: Vec<Hotspot> = header
454        .hotspots
455        .iter()
456        .filter_map(|spot| {
457            let from = project.film_to_timeline(spot.from_ms)?;
458            let length = spot.until_ms.saturating_sub(spot.from_ms);
459            Some(Hotspot { from_ms: from, until_ms: from + length, ..spot.clone() })
460        })
461        .collect();
462    for overlay in &project.overlays {
463        if let OverlayKind::Hotspot { region, title, text, stop } = &overlay.kind {
464            hotspots.push(Hotspot {
465                kind: if *stop { HotspotKind::Stop } else { HotspotKind::Note },
466                x: region.x,
467                y: region.y,
468                width: region.width,
469                height: region.height,
470                title: title.clone(),
471                text: text.clone(),
472                from_ms: overlay.start_ms,
473                until_ms: overlay.end_ms(),
474            });
475        }
476    }
477    hotspots.sort_by_key(|spot| spot.from_ms);
478    (chapters, hotspots)
479}

481/// Clicks and keys on the edited timeline: a part that was cut makes no sound.
482fn edited_sounds(header: &Header, project: &Project) -> Vec<(u64, FilmSound)> {
483    let clicks = header.clicks.iter().map(|click| (click.at_ms, FilmSound::Click));
484    let keys = header.keys.iter().map(|at| (*at, FilmSound::Key));
485    let mut sounds: Vec<(u64, FilmSound)> =
486        clicks.chain(keys).filter_map(|(at, sound)| Some((project.film_to_timeline(at)?, sound))).collect();
487    sounds.sort_by_key(|(at, _)| *at);
488    sounds
489}

491#[cfg_attr(feature = "live", ravenrs_live::expose)]
492impl Film {
493    fn open(header: Header, blobs: Vec<Vec<u8>>) -> Result<Self, MovieError> {
494        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
495        let mut fonts = Vec::with_capacity(header.fonts.len());
496        for font in &header.fonts {
497            fonts.push(match font {
498                StoredFont::Builtin(number) => ravenrs_text::builtin_font(*number)
499                    .ok_or_else(|| damaged("it asks for a font that is not here"))?,
500                StoredFont::Embedded { blob, index } => {
501                    let bytes =
502                        inflate(blobs.get(*blob as usize).ok_or_else(|| damaged("a font is missing"))?)?;
503                    FontRef { id: font_id(&bytes), data: Arc::new(bytes), index: *index }
504                }
505            });
506        }
507        let mut images = Vec::with_capacity(header.images.len());
508        for image in &header.images {
509            let pixels =
510                inflate(blobs.get(image.blob as usize).ok_or_else(|| damaged("a picture is missing"))?)?;
511            images.push(
512                ImageData::from_rgba8(image.width, image.height, pixels)
513                    .ok_or_else(|| damaged("a picture is the wrong size"))?,
514            );
515        }
516        let primitives = header
517            .shapes
518            .iter()
519            .map(|shape| shape.primitive(&fonts, &images))
520            .collect::<Result<Vec<_>, _>>()?;
521        let mut frames: Vec<(u64, Rc<[u32]>)> = Vec::with_capacity(header.frames.len());
522        let mut list: Vec<u32> = Vec::new();
523        for frame in &header.frames {
524            let (head, tail) = (frame.head as usize, frame.tail as usize);
525            if head + tail > list.len() {
526                return Err(damaged("a frame does not fit the one before"));
527            }
528            let mut next = Vec::with_capacity(head + frame.mid.len() + tail);
529            next.extend_from_slice(&list[..head]);
530            next.extend_from_slice(&frame.mid);
531            next.extend_from_slice(&list[list.len() - tail..]);
532            if next.iter().any(|shape| *shape as usize >= primitives.len()) {
533                return Err(damaged("a frame draws something that is not there"));
534            }
535            frames.push((frame.at, next.as_slice().into()));
536            list = next;
537        }
538        Ok(Self { header, blobs, primitives, frames })
539    }
540}

542fn font_id(bytes: &[u8]) -> u64 {
543    // FNV-1a, with a high bit so that it does not meet the ids fonts get at run time.
544    let hash = bytes.iter().fold(0xcbf2_9ce4_8422_2325_u64, |hash, byte| {
545        (hash ^ u64::from(*byte)).wrapping_mul(0x0100_0000_01b3)
546    });
547    hash | (1 << 63)
548}

550#[cfg_attr(feature = "live", ravenrs_live::expose)]
551impl Stored {
552    fn primitive(&self, fonts: &[FontRef], images: &[ImageData]) -> Result<Primitive, MovieError> {
553        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
554        Ok(match self {
555            Self::Q { r, c, f, b, s } => Primitive::Quad(Quad {
556                rect: rect_of(*r),
557                corner_radii: *c,
558                fill: match f {
559                    StoredBrush::S(color) => Brush::Solid(color_of(*color)),
560                    StoredBrush::G(start, end, from, to) => Brush::LinearGradient(LinearGradient {
561                        start: Point::new(start[0], start[1]),
562                        end: Point::new(end[0], end[1]),
563                        from: color_of(*from),
564                        to: color_of(*to),
565                    }),
566                },
567                border: b.map(|(width, color)| Border { width, color: color_of(color) }),
568                shadow: s.map(|(offset, blur, color)| Shadow {
569                    offset: Point::new(offset[0], offset[1]),
570                    blur,
571                    color: color_of(color),
572                }),
573            }),
574            Self::T { f, s, c, y, g } => Primitive::Glyphs(GlyphRun {
575                font: fonts.get(usize::from(*f)).ok_or_else(|| damaged("a font is missing"))?.clone(),
576                size: *s,
577                color: color_of(*c),
578                glyphs: g.iter().map(|(id, x)| PositionedGlyph { id: *id, x: *x, y: *y }).collect(),
579            }),
580            Self::I { i, r, u, c, o } => {
581                let image =
582                    images.get(usize::from(*i)).ok_or_else(|| damaged("a picture is missing"))?.clone();
583                Primitive::Image(ImageQuad {
584                    rect: rect_of(*r),
585                    image,
586                    uv: *u,
587                    corner_radii: *c,
588                    opacity: *o,
589                })
590            }
591            Self::C(r) => Primitive::PushClip(rect_of(*r)),
592            Self::P => Primitive::PopClip,
593        })
594    }
595}

597fn deflate(bytes: &[u8]) -> Vec<u8> {
598    let mut encoder = ZlibEncoder::new(Vec::new(), Compression::best());
599    // Writing to a `Vec` does not fail.
600    let _ = encoder.write_all(bytes);
601    encoder.finish().unwrap_or_default()
602}
603
604fn inflate(data: &[u8]) -> Result<Vec<u8>, MovieError> {
605    let mut bytes = Vec::new();
606    ZlibDecoder::new(data).read_to_end(&mut bytes).map_err(|e| MovieError::Damaged(e.to_string()))?;
607    Ok(bytes)
608}

610fn shrink(image: &ImageData, rect: Rect) -> (u32, u32, Vec<u8>) {
611    let (width, height) = (image.width(), image.height());
612    let wanted = |size: f32, have: u32| ((size * IMAGE_DETAIL).ceil() as u32).clamp(1, have);
613    let (new_width, new_height) = (wanted(rect.width(), width), wanted(rect.height(), height));
614    if (new_width, new_height) == (width, height) {
615        return (width, height, image.pixels().to_vec());
616    }
617    let mut pixels = Vec::with_capacity(new_width as usize * new_height as usize * 4);
618    for y in 0..new_height {
619        let (top, bottom) =
620            (y * height / new_height, ((y + 1) * height / new_height).max(y * height / new_height + 1));
621        for x in 0..new_width {
622            let (left, right) =
623                (x * width / new_width, ((x + 1) * width / new_width).max(x * width / new_width + 1));
624            let mut sum = [0_u32; 4];
625            for sy in top..bottom {
626                for sx in left..right {
627                    let at = (sy as usize * width as usize + sx as usize) * 4;
628                    for (total, value) in sum.iter_mut().zip(&image.pixels()[at..at + 4]) {
629                        *total += u32::from(*value);
630                    }
631                }
632            }
633            let count = (bottom - top) * (right - left);
634            pixels.extend(sum.map(|total| (total / count) as u8));
635        }
636    }
637    (new_width, new_height, pixels)
638}

640/// Puts a film together from what the app draws, keeping each different thing once.
641#[derive(Clone, Debug)]
642#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
643pub struct FilmBuilder {
644    header: Header,
645    blobs: Vec<Vec<u8>>,
646    known: HashMap<String, u32>,
647    known_fonts: HashMap<(u64, u32), u16>,
648    known_images: HashMap<u64, u16>,
649    list: Vec<u32>,
650}
651
652#[cfg_attr(feature = "live", ravenrs_live::expose)]
653impl FilmBuilder {
654    /// A film of an app of this logical size, over this color.
655    pub fn new(logical: Size, background: Color) -> Self {
656        Self {
657            header: Header {
658                logical: [logical.width, logical.height],
659                background: tidy_color(background),
660                duration_ms: 0,
661                fonts: Vec::new(),
662                images: Vec::new(),
663                shapes: Vec::new(),
664                frames: Vec::new(),
665                pointer: Vec::new(),
666                clicks: Vec::new(),
667                keys: Vec::new(),
668                chapters: Vec::new(),
669                hotspots: Vec::new(),
670                blob_lengths: Vec::new(),
671            },
672            blobs: Vec::new(),
673            known: HashMap::new(),
674            known_fonts: HashMap::new(),
675            known_images: HashMap::new(),
676            list: Vec::new(),
677        }
678    }
679
680    fn add_blob(&mut self, bytes: &[u8]) -> u32 {
681        let blob = deflate(bytes);
682        self.header.blob_lengths.push(blob.len() as u64);
683        self.blobs.push(blob);
684        (self.blobs.len() - 1) as u32
685    }
686
687    fn font(&mut self, font: &FontRef) -> u16 {
688        let key = (font.id, font.index);
689        if let Some(known) = self.known_fonts.get(&key) {
690            return *known;
691        }
692        let stored = match ravenrs_text::builtin_font_number(font.bytes()) {
693            Some(number) => StoredFont::Builtin(number),
694            None => StoredFont::Embedded { blob: self.add_blob(font.bytes()), index: font.index },
695        };
696        self.header.fonts.push(stored);
697        let number = (self.header.fonts.len() - 1) as u16;
698        self.known_fonts.insert(key, number);
699        number
700    }
701
702    fn image(&mut self, quad: &ImageQuad) -> u16 {
703        if let Some(known) = self.known_images.get(&quad.image.id()) {
704            return *known;
705        }
706        let (width, height, pixels) = shrink(&quad.image, quad.rect);
707        let blob = self.add_blob(&pixels);
708        self.header.images.push(StoredImage { width, height, blob });
709        let number = (self.header.images.len() - 1) as u16;
710        self.known_images.insert(quad.image.id(), number);
711        number
712    }
713
714    fn stored(&mut self, primitive: &Primitive) -> Option<Stored> {
715        Some(match primitive {
716            Primitive::Quad(quad) => Stored::Q {
717                r: tidy_rect(quad.rect),
718                c: quad.corner_radii.map(tidy),
719                f: match quad.fill {
720                    Brush::Solid(color) => StoredBrush::S(tidy_color(color)),
721                    Brush::LinearGradient(g) => StoredBrush::G(
722                        [g.start.x, g.start.y],
723                        [g.end.x, g.end.y],
724                        tidy_color(g.from),
725                        tidy_color(g.to),
726                    ),
727                    _ => return None,
728                },
729                b: quad.border.map(|border| (tidy(border.width), tidy_color(border.color))),
730                s: quad.shadow.map(|shadow| {
731                    (
732                        [tidy(shadow.offset.x), tidy(shadow.offset.y)],
733                        tidy(shadow.blur),
734                        tidy_color(shadow.color),
735                    )
736                }),
737            },
738            Primitive::Glyphs(run) => {
739                let y = run.glyphs.first().map_or(0.0, |glyph| tidy(glyph.y));
740                Stored::T {
741                    f: self.font(&run.font),
742                    s: tidy(run.size),
743                    c: tidy_color(run.color),
744                    y,
745                    g: run.glyphs.iter().map(|glyph| (glyph.id, tidy(glyph.x))).collect(),
746                }
747            }
748            Primitive::Image(quad) => Stored::I {
749                i: self.image(quad),
750                r: tidy_rect(quad.rect),
751                u: quad.uv,
752                c: quad.corner_radii.map(tidy),
753                o: quad.opacity,
754            },
755            Primitive::PushClip(rect) => Stored::C(tidy_rect(*rect)),
756            Primitive::PopClip => Stored::P,
757            _ => return None,
758        })
759    }
760
761    /// Adds what the app draws at `at_ms`. Nothing is added when it is the same
762    /// as the last time: that picture just stays longer on screen.
763    pub fn frame(&mut self, at_ms: u64, primitives: &[Primitive]) {
764        self.header.duration_ms = self.header.duration_ms.max(at_ms);
765        let mut list = Vec::with_capacity(primitives.len());
766        for primitive in primitives {
767            let Some(stored) = self.stored(primitive) else {
768                continue;
769            };
770            // The text of a thing is its identity, floats included.
771            let key = serde_json::to_string(&stored).unwrap_or_default();
772            let next = self.header.shapes.len() as u32;
773            let index = *self.known.entry(key).or_insert(next);
774            if index == next {
775                self.header.shapes.push(stored);
776            }
777            list.push(index);
778        }
779        if !self.header.frames.is_empty() && list == self.list {
780            return;
781        }
782        let head = list.iter().zip(&self.list).take_while(|(a, b)| a == b).count();
783        let tail =
784            list[head..].iter().rev().zip(self.list[head..].iter().rev()).take_while(|(a, b)| a == b).count();
785        let mid = list[head..list.len() - tail].to_vec();
786        self.header.frames.push(StoredFrame { at: at_ms, head: head as u32, tail: tail as u32, mid });
787        self.list = list;
788    }

790    /// Notes where the pointer is.
791    pub fn pointer(&mut self, at_ms: u64, position: Point) {
792        self.header.pointer.push(PointerKey { at_ms, x: tidy(position.x), y: tidy(position.y) });
793    }

795    /// Notes a click.
796    pub fn click(&mut self, at_ms: u64, position: Point) {
797        self.header.clicks.push(PointerKey { at_ms, x: tidy(position.x), y: tidy(position.y) });
798    }
799
800    /// Notes a key being pressed, for its sound.
801    pub fn key(&mut self, at_ms: u64) {
802        self.header.keys.push(at_ms);
803    }
804
805    /// Starts a chapter at `at_ms`.
806    pub fn chapter(&mut self, title: impl Into<String>, text: impl Into<String>, at_ms: u64) {
807        self.header.chapters.push(Chapter { title: title.into(), text: text.into(), at_ms });
808    }

810    /// Adds a spot on the screen.
811    pub fn hotspot(&mut self, hotspot: Hotspot) {

813    }
814
815    /// The film, lasting until `duration_ms` even if the picture stopped changing before.
816    pub fn finish(mut self, duration_ms: u64) -> Result<Movie, MovieError> {
817        self.header.duration_ms = self.header.duration_ms.max(duration_ms);
818        if self.header.frames.is_empty() {
819            return Err(MovieError::Empty);
820        }
821        let project = Project::whole(self.header.duration_ms);
822        let film = Film::open(self.header, self.blobs)?;
823        Ok(Movie::with(Rc::new(film), project))
824    }
