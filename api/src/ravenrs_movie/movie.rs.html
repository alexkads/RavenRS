24/// What can go wrong with a film.
25#[derive(Debug)]
26#[non_exhaustive]
27pub enum MovieError {
28    /// The file is not a film, or it was cut short.
29    Damaged(String),
30    /// The film has no frames.
31    Empty,
32}
33
34impl std::fmt::Display for MovieError {
35    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
36        match self {
37            Self::Damaged(why) => write!(f, "the film is damaged: {why}"),
38            Self::Empty => f.write_str("the film has no frames"),
39        }
40    }
41}

335    /// Where the pointer was at this moment of the film.
336    pub fn pointer_at(&self, film_ms: u64) -> Option<Point> {
337        let keys = &self.film.header.pointer;
338        let after = keys.partition_point(|key| key.at_ms <= film_ms);
339        match (after.checked_sub(1).and_then(|i| keys.get(i)), keys.get(after)) {
340            (Some(a), Some(b)) if b.at_ms > a.at_ms => {
341                let t = (film_ms - a.at_ms) as f32 / (b.at_ms - a.at_ms) as f32;
342                Some(Point::new(a.x + (b.x - a.x) * t, a.y + (b.y - a.y) * t))
343            }
344            (Some(a), _) => Some(Point::new(a.x, a.y)),
345            (None, Some(b)) => Some(Point::new(b.x, b.y)),
346            (None, None) => None,
347        }
348    }

357    /// The whole film as the bytes of a file, with its edits.
358    pub fn to_bytes(&self) -> Vec<u8> {
359        let header = serde_json::to_vec(&self.film.header).unwrap_or_default();
360        let project = serde_json::to_vec(&self.project).unwrap_or_default();
361        let body = deflate(&header);
362        let edits = deflate(&project);
363        let mut bytes = Vec::new();
364        bytes.extend_from_slice(MAGIC);
365        for part in [&body, &edits] {
366            bytes.extend_from_slice(&(part.len() as u64).to_le_bytes());
367            bytes.extend_from_slice(part);
368        }
369        for blob in &self.film.blobs {
370            bytes.extend_from_slice(blob);
371        }
372        bytes
373    }

375    /// Reads a film back from the bytes of a file.
376    pub fn from_bytes(bytes: &[u8]) -> Result<Self, MovieError> {
377        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
378        let mut rest = bytes.strip_prefix(MAGIC).ok_or_else(|| damaged("it does not start like a film"))?;
379        let mut part = || -> Result<Vec<u8>, MovieError> {
380            let (length, after) =
381                rest.split_first_chunk::<8>().ok_or_else(|| damaged("a part is missing"))?;
382            let length =
383                usize::try_from(u64::from_le_bytes(*length)).map_err(|_| damaged("a part is too long"))?;
384            let data = after.get(..length).ok_or_else(|| damaged("a part was cut"))?;
385            rest = &after[length..];
386            inflate(data)
387        };
388        let header = part()?;
389        let project = part()?;
390        let header: Header =
391            serde_json::from_slice(&header).map_err(|e| MovieError::Damaged(e.to_string()))?;
392        let project: Project =
393            serde_json::from_slice(&project).map_err(|e| MovieError::Damaged(e.to_string()))?;
394        let mut blobs = Vec::with_capacity(header.blob_lengths.len());
395        for length in &header.blob_lengths {
396            let length = usize::try_from(*length).map_err(|_| damaged("a blob is too long"))?;
397            let blob = rest.get(..length).ok_or_else(|| damaged("a blob was cut"))?;
398            blobs.push(blob.to_vec());
399            rest = &rest[length..];
400        }
401        if header.frames.is_empty() {
402            return Err(MovieError::Empty);
403        }
404        let film = Film::open(header, blobs)?;
405        Ok(Self::with(Rc::new(film), project))
406    }
407}

409/// Chapters and hotspots, moved to where the edits put them, with the hotspots
410/// added on the timeline.
411fn edited_marks(header: &Header, project: &Project) -> (Vec<Chapter>, Vec<Hotspot>) {
412    let mut chapters: Vec<Chapter> = header
413        .chapters
414        .iter()
415        .filter_map(|chapter| {
416            // If its first moments were cut, it starts where the film picks up again.
417            let next = project.clips.iter().filter_map(|clip| match clip.kind {
418                crate::project::ClipKind::Film { from_ms, .. } if from_ms >= chapter.at_ms => Some(from_ms),
419                _ => None,
420            });
421            let at = project.film_to_timeline(chapter.at_ms).or_else(|| {
422                let end = header
423                    .chapters
424                    .iter()
425                    .map(|c| c.at_ms)
426                    .find(|at| *at > chapter.at_ms)
427                    .unwrap_or(u64::MAX);
428                next.filter(|from| *from < end).min().and_then(|from| project.film_to_timeline(from))
429            })?;
430            Some(Chapter { at_ms: at, ..chapter.clone() })
431        })
432        .collect();
433    chapters.extend(project.markers.iter().map(|marker| Chapter {
434        title: marker.title.clone(),
435        text: String::new(),
436        at_ms: marker.at_ms,
437    }));
438    chapters.sort_by_key(|chapter| chapter.at_ms);
439    let mut hotspots: Vec<Hotspot> = header
440        .hotspots
441        .iter()
442        .filter_map(|spot| {
443            let from = project.film_to_timeline(spot.from_ms)?;
444            let length = spot.until_ms.saturating_sub(spot.from_ms);
445            Some(Hotspot { from_ms: from, until_ms: from + length, ..spot.clone() })
446        })
447        .collect();
448    for overlay in &project.overlays {
449        if let OverlayKind::Hotspot { region, title, text, stop } = &overlay.kind {
450            hotspots.push(Hotspot {
451                kind: if *stop { HotspotKind::Stop } else { HotspotKind::Note },
452                x: region.x,
453                y: region.y,
454                width: region.width,
455                height: region.height,
456                title: title.clone(),
457                text: text.clone(),
458                from_ms: overlay.start_ms,
459                until_ms: overlay.end_ms(),
460            });
461        }
462    }
463    hotspots.sort_by_key(|spot| spot.from_ms);
464    (chapters, hotspots)
465}

467/// Clicks and keys on the edited timeline: a part that was cut makes no sound.
468fn edited_sounds(header: &Header, project: &Project) -> Vec<(u64, FilmSound)> {
469    let clicks = header.clicks.iter().map(|click| (click.at_ms, FilmSound::Click));
470    let keys = header.keys.iter().map(|at| (*at, FilmSound::Key));
471    let mut sounds: Vec<(u64, FilmSound)> =
472        clicks.chain(keys).filter_map(|(at, sound)| Some((project.film_to_timeline(at)?, sound))).collect();
473    sounds.sort_by_key(|(at, _)| *at);
474    sounds
475}

477impl Film {
478    fn open(header: Header, blobs: Vec<Vec<u8>>) -> Result<Self, MovieError> {
479        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
480        let mut fonts = Vec::with_capacity(header.fonts.len());
481        for font in &header.fonts {
482            fonts.push(match font {
483                StoredFont::Builtin(number) => ravenrs_text::builtin_font(*number)
484                    .ok_or_else(|| damaged("it asks for a font that is not here"))?,
485                StoredFont::Embedded { blob, index } => {
486                    let bytes =
487                        inflate(blobs.get(*blob as usize).ok_or_else(|| damaged("a font is missing"))?)?;
488                    FontRef { id: font_id(&bytes), data: Arc::new(bytes), index: *index }
489                }
490            });
491        }
492        let mut images = Vec::with_capacity(header.images.len());
493        for image in &header.images {
494            let pixels =
495                inflate(blobs.get(image.blob as usize).ok_or_else(|| damaged("a picture is missing"))?)?;
496            images.push(
497                ImageData::from_rgba8(image.width, image.height, pixels)
498                    .ok_or_else(|| damaged("a picture is the wrong size"))?,
499            );
500        }
501        let primitives = header
502            .shapes
503            .iter()
504            .map(|shape| shape.primitive(&fonts, &images))
505            .collect::<Result<Vec<_>, _>>()?;
506        let mut frames: Vec<(u64, Rc<[u32]>)> = Vec::with_capacity(header.frames.len());
507        let mut list: Vec<u32> = Vec::new();
508        for frame in &header.frames {
509            let (head, tail) = (frame.head as usize, frame.tail as usize);
510            if head + tail > list.len() {
511                return Err(damaged("a frame does not fit the one before"));
512            }
513            let mut next = Vec::with_capacity(head + frame.mid.len() + tail);
514            next.extend_from_slice(&list[..head]);
515            next.extend_from_slice(&frame.mid);
516            next.extend_from_slice(&list[list.len() - tail..]);
517            if next.iter().any(|shape| *shape as usize >= primitives.len()) {
518                return Err(damaged("a frame draws something that is not there"));
519            }
520            frames.push((frame.at, next.as_slice().into()));
521            list = next;
522        }
523        Ok(Self { header, blobs, primitives, frames })
524    }
525}

527fn font_id(bytes: &[u8]) -> u64 {
528    // FNV-1a, with a high bit so that it does not meet the ids fonts get at run time.
529    let hash = bytes.iter().fold(0xcbf2_9ce4_8422_2325_u64, |hash, byte| {
530        (hash ^ u64::from(*byte)).wrapping_mul(0x0100_0000_01b3)
531    });
532    hash | (1 << 63)
533}

535impl Stored {
536    fn primitive(&self, fonts: &[FontRef], images: &[ImageData]) -> Result<Primitive, MovieError> {
537        let damaged = |why: &str| MovieError::Damaged(why.to_owned());
538        Ok(match self {
539            Self::Q { r, c, f, b, s } => Primitive::Quad(Quad {
540                rect: rect_of(*r),
541                corner_radii: *c,
542                fill: match f {
543                    StoredBrush::S(color) => Brush::Solid(color_of(*color)),
544                    StoredBrush::G(start, end, from, to) => Brush::LinearGradient(LinearGradient {
545                        start: Point::new(start[0], start[1]),
546                        end: Point::new(end[0], end[1]),
547                        from: color_of(*from),
548                        to: color_of(*to),
549                    }),
550                },
551                border: b.map(|(width, color)| Border { width, color: color_of(color) }),
552                shadow: s.map(|(offset, blur, color)| Shadow {
553                    offset: Point::new(offset[0], offset[1]),
554                    blur,
555                    color: color_of(color),
556                }),
557            }),
558            Self::T { f, s, c, y, g } => Primitive::Glyphs(GlyphRun {
559                font: fonts.get(usize::from(*f)).ok_or_else(|| damaged("a font is missing"))?.clone(),
560                size: *s,
561                color: color_of(*c),
562                glyphs: g.iter().map(|(id, x)| PositionedGlyph { id: *id, x: *x, y: *y }).collect(),
563            }),
564            Self::I { i, r, u, c, o } => {
565                let image =
566                    images.get(usize::from(*i)).ok_or_else(|| damaged("a picture is missing"))?.clone();
567                Primitive::Image(ImageQuad {
568                    rect: rect_of(*r),
569                    image,
570                    uv: *u,
571                    corner_radii: *c,
572                    opacity: *o,
573                })
574            }
575            Self::C(r) => Primitive::PushClip(rect_of(*r)),
576            Self::P => Primitive::PopClip,
577        })
578    }
579}

581fn deflate(bytes: &[u8]) -> Vec<u8> {
582    let mut encoder = ZlibEncoder::new(Vec::new(), Compression::best());
583    // Writing to a `Vec` does not fail.
584    let _ = encoder.write_all(bytes);
585    encoder.finish().unwrap_or_default()
586}

588fn inflate(data: &[u8]) -> Result<Vec<u8>, MovieError> {
589    let mut bytes = Vec::new();
590    ZlibDecoder::new(data).read_to_end(&mut bytes).map_err(|e| MovieError::Damaged(e.to_string()))?;
591    Ok(bytes)
592}

594fn shrink(image: &ImageData, rect: Rect) -> (u32, u32, Vec<u8>) {
595    let (width, height) = (image.width(), image.height());
596    let wanted = |size: f32, have: u32| ((size * IMAGE_DETAIL).ceil() as u32).clamp(1, have);
597    let (new_width, new_height) = (wanted(rect.width(), width), wanted(rect.height(), height));
598    if (new_width, new_height) == (width, height) {
599        return (width, height, image.pixels().to_vec());
600    }
601    let mut pixels = Vec::with_capacity(new_width as usize * new_height as usize * 4);
602    for y in 0..new_height {
603        let (top, bottom) =
604            (y * height / new_height, ((y + 1) * height / new_height).max(y * height / new_height + 1));
605        for x in 0..new_width {
606            let (left, right) =
607                (x * width / new_width, ((x + 1) * width / new_width).max(x * width / new_width + 1));
608            let mut sum = [0_u32; 4];
609            for sy in top..bottom {
610                for sx in left..right {
611                    let at = (sy as usize * width as usize + sx as usize) * 4;
612                    for (total, value) in sum.iter_mut().zip(&image.pixels()[at..at + 4]) {
613                        *total += u32::from(*value);
614                    }
615                }
616            }
617            let count = (bottom - top) * (right - left);
618            pixels.extend(sum.map(|total| (total / count) as u8));
619        }
620    }
621    (new_width, new_height, pixels)
622}

624/// Puts a film together from what the app draws, keeping each different thing once.
625#[derive(Clone, Debug)]
626pub struct FilmBuilder {
627    header: Header,
628    blobs: Vec<Vec<u8>>,
629    known: HashMap<String, u32>,
630    known_fonts: HashMap<(u64, u32), u16>,
631    known_images: HashMap<u64, u16>,
632    list: Vec<u32>,
633}
634
635impl FilmBuilder {
636    /// A film of an app of this logical size, over this color.
637    pub fn new(logical: Size, background: Color) -> Self {
638        Self {
639            header: Header {
640                logical: [logical.width, logical.height],
641                background: tidy_color(background),
642                duration_ms: 0,
643                fonts: Vec::new(),
644                images: Vec::new(),
645                shapes: Vec::new(),
646                frames: Vec::new(),
647                pointer: Vec::new(),
648                clicks: Vec::new(),
649                keys: Vec::new(),
650                chapters: Vec::new(),
651                hotspots: Vec::new(),
652                blob_lengths: Vec::new(),
653            },
654            blobs: Vec::new(),
655            known: HashMap::new(),
656            known_fonts: HashMap::new(),
657            known_images: HashMap::new(),
658            list: Vec::new(),
659        }
660    }
661
662    fn add_blob(&mut self, bytes: &[u8]) -> u32 {
663        let blob = deflate(bytes);
664        self.header.blob_lengths.push(blob.len() as u64);
665        self.blobs.push(blob);
666        (self.blobs.len() - 1) as u32
667    }
668
669    fn font(&mut self, font: &FontRef) -> u16 {
670        let key = (font.id, font.index);
671        if let Some(known) = self.known_fonts.get(&key) {
672            return *known;
673        }
674        let stored = match ravenrs_text::builtin_font_number(font.bytes()) {
675            Some(number) => StoredFont::Builtin(number),
676            None => StoredFont::Embedded { blob: self.add_blob(font.bytes()), index: font.index },
677        };
678        self.header.fonts.push(stored);
679        let number = (self.header.fonts.len() - 1) as u16;
680        self.known_fonts.insert(key, number);
681        number
682    }
683
684    fn image(&mut self, quad: &ImageQuad) -> u16 {
685        if let Some(known) = self.known_images.get(&quad.image.id()) {
686            return *known;
687        }
688        let (width, height, pixels) = shrink(&quad.image, quad.rect);
689        let blob = self.add_blob(&pixels);
690        self.header.images.push(StoredImage { width, height, blob });
691        let number = (self.header.images.len() - 1) as u16;
692        self.known_images.insert(quad.image.id(), number);
693        number
694    }
695
696    fn stored(&mut self, primitive: &Primitive) -> Option<Stored> {
697        Some(match primitive {
698            Primitive::Quad(quad) => Stored::Q {
699                r: tidy_rect(quad.rect),
700                c: quad.corner_radii.map(tidy),
701                f: match quad.fill {
702                    Brush::Solid(color) => StoredBrush::S(tidy_color(color)),
703                    Brush::LinearGradient(g) => StoredBrush::G(
704                        [g.start.x, g.start.y],
705                        [g.end.x, g.end.y],
706                        tidy_color(g.from),
707                        tidy_color(g.to),
708                    ),
709                    _ => return None,
710                },
711                b: quad.border.map(|border| (tidy(border.width), tidy_color(border.color))),
712                s: quad.shadow.map(|shadow| {
713                    (
714                        [tidy(shadow.offset.x), tidy(shadow.offset.y)],
715                        tidy(shadow.blur),
716                        tidy_color(shadow.color),
717                    )
718                }),
719            },
720            Primitive::Glyphs(run) => {
721                let y = run.glyphs.first().map_or(0.0, |glyph| tidy(glyph.y));
722                Stored::T {
723                    f: self.font(&run.font),
724                    s: tidy(run.size),
725                    c: tidy_color(run.color),
726                    y,
727                    g: run.glyphs.iter().map(|glyph| (glyph.id, tidy(glyph.x))).collect(),
728                }
729            }
730            Primitive::Image(quad) => Stored::I {
731                i: self.image(quad),
732                r: tidy_rect(quad.rect),
733                u: quad.uv,
734                c: quad.corner_radii.map(tidy),
735                o: quad.opacity,
736            },
737            Primitive::PushClip(rect) => Stored::C(tidy_rect(*rect)),
738            Primitive::PopClip => Stored::P,
739            _ => return None,
740        })
741    }
742
743    /// Adds what the app draws at `at_ms`. Nothing is added when it is the same
744    /// as the last time: that picture just stays longer on screen.
745    pub fn frame(&mut self, at_ms: u64, primitives: &[Primitive]) {
746        self.header.duration_ms = self.header.duration_ms.max(at_ms);
747        let mut list = Vec::with_capacity(primitives.len());
748        for primitive in primitives {
749            let Some(stored) = self.stored(primitive) else {
750                continue;
751            };
752            // The text of a thing is its identity, floats included.
753            let key = serde_json::to_string(&stored).unwrap_or_default();
754            let next = self.header.shapes.len() as u32;
755            let index = *self.known.entry(key).or_insert(next);
756            if index == next {
757                self.header.shapes.push(stored);
758            }
759            list.push(index);
760        }
761        if !self.header.frames.is_empty() && list == self.list {
762            return;
763        }
764        let head = list.iter().zip(&self.list).take_while(|(a, b)| a == b).count();
765        let tail =
766            list[head..].iter().rev().zip(self.list[head..].iter().rev()).take_while(|(a, b)| a == b).count();
767        let mid = list[head..list.len() - tail].to_vec();
768        self.header.frames.push(StoredFrame { at: at_ms, head: head as u32, tail: tail as u32, mid });
769        self.list = list;
770    }
771
772    /// Notes where the pointer is.
773    pub fn pointer(&mut self, at_ms: u64, position: Point) {
774        self.header.pointer.push(PointerKey { at_ms, x: tidy(position.x), y: tidy(position.y) });
775    }
776
777    /// Notes a click.
778    pub fn click(&mut self, at_ms: u64, position: Point) {
779        self.header.clicks.push(PointerKey { at_ms, x: tidy(position.x), y: tidy(position.y) });
780    }
781
782    /// Notes a key being pressed, for its sound.
783    pub fn key(&mut self, at_ms: u64) {
784        self.header.keys.push(at_ms);
785    }
786
787    /// Starts a chapter at `at_ms`.
788    pub fn chapter(&mut self, title: impl Into<String>, text: impl Into<String>, at_ms: u64) {
789        self.header.chapters.push(Chapter { title: title.into(), text: text.into(), at_ms });
790    }
791
792    /// Adds a spot on the screen.
793    pub fn hotspot(&mut self, hotspot: Hotspot) {
794        self.header.hotspots.push(hotspot);
795    }
796
797    /// The film, lasting until `duration_ms` even if the picture stopped changing before.
798    pub fn finish(mut self, duration_ms: u64) -> Result<Movie, MovieError> {
799        self.header.duration_ms = self.header.duration_ms.max(duration_ms);
800        if self.header.frames.is_empty() {
801            return Err(MovieError::Empty);
802        }
803        let project = Project::whole(self.header.duration_ms);
804        let film = Film::open(self.header, self.blobs)?;
805        Ok(Movie::with(Rc::new(film), project))
806    }
807}
