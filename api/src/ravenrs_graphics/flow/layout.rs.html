1//! Arranging a diagram by itself, in layers: every connection runs one way, from one layer to a
2//! later one, and the nodes of a layer are ordered so that lines cross as little as they can.
3//!
4//! It is the usual layered method, kept small: cycles are broken by turning back the
5//! connections that close them, each node goes to the layer after the longest path that reaches
6//! it, the nodes of each layer are ordered by the average place of their neighbours over a few
7//! sweeps, and each node then moves towards its neighbours as far as the room in its layer lets
8//! it, so a chain comes out straight.
9
10use super::{FlowGraph, FlowNode};
11
12/// The way the layers follow one another.
13#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
14#[non_exhaustive]
15#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
16pub enum FlowDirection {
17    /// From left to right: the first layer is the leftmost column.
18    #[default]
19    Right,
20    /// From top to bottom: the first layer is the top row.
21    Down,
22}
23
24/// The arrangement [`FlowLayout`] makes.
25#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
26#[non_exhaustive]
27#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
28pub enum FlowLayoutKind {
29    /// Layers in which every connection runs one way and few lines cross: for flows.
30    #[default]
31    Layered,
32    /// A tree, as an org chart: each node centred over the nodes it leads to, which sit side by
33    /// side. A node reached from two others hangs under the first; a forest is trees side by
34    /// side.
35    Tree,
36    /// A mind map: the first node nothing leads to in the middle, half its branches growing to
37    /// the right and half to the left, each as a tree.
38    MindMap,
39}
40
41/// How [`FlowLayout::apply`] arranges a diagram: the kind of arrangement, the direction and
42/// the room between layers and between the nodes of a layer, in units of the diagram.
43#[derive(Clone, Copy, Debug, PartialEq)]
44#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
45pub struct FlowLayout {
46    /// Layers or a tree.
47    pub kind: FlowLayoutKind,
48    /// The way the layers follow one another.
49    pub direction: FlowDirection,
50    /// The room between one layer and the next.
51    pub layer_gap: f32,
52    /// The room between two nodes of a layer.
53    pub node_gap: f32,
54}
55
56impl Default for FlowLayout {
57    fn default() -> Self {
58        Self::new(FlowDirection::Right)
59    }
60}

62/// How many times the order of the layers is swept, down and back up.
63const SWEEPS: usize = 4;

67    /// Layers in this direction, 96 units apart, with 48 between the nodes of a layer.
68    pub fn new(direction: FlowDirection) -> Self {
69        Self { kind: FlowLayoutKind::Layered, direction, layer_gap: 96.0, node_gap: 48.0 }
70    }

72    /// A tree growing in this direction, as an org chart (down) or a mind map (right).
73    pub fn tree(direction: FlowDirection) -> Self {
74        Self { kind: FlowLayoutKind::Tree, ..Self::new(direction) }

83    pub fn gaps(mut self, layer_gap: f32, node_gap: f32) -> Self {
84        self.layer_gap = layer_gap.max(0.0);
85        self.node_gap = node_gap.max(0.0);

89    /// Where each node goes, its left and top edges, in the order of `graph.nodes`. The
90    /// arranged diagram starts where the diagram started: its box keeps its top left corner.
91    ///
92    /// Only the nodes outside any group are arranged, with a group standing for everything in
93    /// it and its connections; what a group holds moves with it and keeps its own arrangement.
94    pub fn positions(&self, graph: &FlowGraph) -> Vec<(f32, f32)> {
95        let outer = FlowGraph {
96            // The app's data of the whole diagram stays with the diagram.
97            data: serde_json::Value::Null,
98            nodes: graph.nodes.iter().filter(|n| n.parent.is_none()).cloned().collect(),
99            edges: graph
100                .edges
101                .iter()
102                .map(|e| {
103                    let mut e = e.clone();
104                    e.from.node = graph.top(&e.from.node);
105                    e.to.node = graph.top(&e.to.node);
106                    e
107                })
108                .collect(),
109        };
110        let placed = self.arrange(&outer);
111        graph
112            .nodes
113            .iter()
114            .map(|n| {
115                let top = graph.top(&n.id);
116                let moved = outer.nodes.iter().zip(&placed).find(|(o, _)| o.id == top);
117                match moved {
118                    Some((o, (x, y))) => (n.x + x - o.x, n.y + y - o.y),
119                    None => (n.x, n.y),
120                }
121            })
122            .collect()
123    }

125    /// Where each node of a diagram with no groups goes.
126    fn arrange(&self, graph: &FlowGraph) -> Vec<(f32, f32)> {
127        let n = graph.nodes.len();
128        if n == 0 {
129            return Vec::new();
130        }
131        let links = links(graph);
132        let places = match self.kind {
133            FlowLayoutKind::Tree => self.tree_places(&graph.nodes, &links),
134            FlowLayoutKind::MindMap => self.mind_places(&graph.nodes, &links),
135            _ => {
136                let layer = layers(n, &links);
137                let mut order = order(n, &links, &layer);
138                self.place(&graph.nodes, &links, &layer, &mut order)
139            }
140        };
141        let origin = graph.bounds().map_or((0.0, 0.0), |b| (b.x(), b.y()));
142        let (least_x, least_y) =
143            places.iter().fold((f32::INFINITY, f32::INFINITY), |(x, y), p| (x.min(p.0), y.min(p.1)));
144        places.into_iter().map(|(x, y)| (x - least_x + origin.0, y - least_y + origin.1)).collect()
145    }
146
147    /// Moves every node of `graph` where [`Self::positions`] puts it. Says whether any moved.
148    /// A mind map also turns the ports of the nodes left of the middle to face it: their way in
149    /// on the right, their way out on the left.
150    pub fn apply(&self, graph: &mut FlowGraph) -> bool {
151        let places = self.positions(graph);
152        let mut moved = false;
153        for (node, (x, y)) in graph.nodes.iter_mut().zip(places) {
154            if node.x != x || node.y != y {
155                node.x = x;
156                node.y = y;
157                moved = true;
158            }
159        }
160        if self.kind == FlowLayoutKind::MindMap
161            && let Some(middle) = middle_of(graph)
162        {
163            let centre = graph.nodes[middle].rect().center().x;
164            for (i, node) in graph.nodes.iter_mut().enumerate() {
165                let left = i != middle && node.rect().center().x < centre;
166                for port in &mut node.ports {
167                    if !port.side.is_horizontal() {
168                        continue;
169                    }
170                    let wanted = match (port.direction == super::PortDirection::Input, left) {
171                        (true, false) | (false, true) => super::PortSide::Left,
172                        _ => super::PortSide::Right,
173                    };
174                    if port.side != wanted {
175                        port.side = wanted;
176                        moved = true;
177                    }
178                }
179            }
180        }
181        moved
182    }

184    /// Where each node of a mind map goes: the branches of the middle split between the right
185    /// and the left, each half a tree, the left one mirrored.
186    fn mind_places(&self, nodes: &[FlowNode], links: &[(usize, usize)]) -> Vec<(f32, f32)> {
187        let n = nodes.len();
188        let reached: Vec<bool> = (0..n).map(|i| links.iter().any(|(_, b)| *b == i)).collect();
189        let middle = (0..n).find(|i| !reached[*i]).unwrap_or(0);
190        let branches: Vec<usize> = links.iter().filter(|(a, _)| *a == middle).map(|(_, b)| *b).collect();
191        let half = branches.len().div_ceil(2);
192        let under = |from: usize| -> Vec<usize> {
193            let mut found = vec![from];
194            let mut i = 0;
195            while i < found.len() {
196                for (a, b) in links {
197                    if *a == found[i] && !found.contains(b) && *b != middle {
198                        found.push(*b);
199                    }
200                }
201                i += 1;
202            }
203            found
204        };
205        let side = |chosen: &[usize]| -> Vec<usize> {
206            let mut all = vec![middle];
207            for b in chosen {
208                for i in under(*b) {
209                    if !all.contains(&i) {
210                        all.push(i);
211                    }
212                }
213            }
214            all
215        };
216        let (right, left) = (side(&branches[..half]), side(&branches[half..]));
217        let tree = |members: &[usize]| -> Vec<(f32, f32)> {
218            let sub: Vec<FlowNode> = members.iter().map(|i| nodes[*i].clone()).collect();
219            let sub_links: Vec<(usize, usize)> = links
220                .iter()
221                .filter_map(|(a, b)| {
222                    Some((members.iter().position(|m| m == a)?, members.iter().position(|m| m == b)?))
223                })
224                .collect();
225            Self { kind: FlowLayoutKind::Tree, direction: FlowDirection::Right, ..*self }
226                .tree_places(&sub, &sub_links)
227        };
228        let mut places: Vec<(f32, f32)> = nodes.iter().map(|n| (n.x, n.y)).collect();
229        let r = tree(&right);
230        let (rx, ry) = r[0];
231        for (k, i) in right.iter().enumerate() {
232            places[*i] = (r[k].0 - rx, r[k].1 - ry);
233        }
234        if left.len() > 1 {
235            let l = tree(&left);
236            let (lx, ly) = l[0];
237            let w = nodes[middle].width;
238            for (k, i) in left.iter().enumerate().skip(1) {
239                // Mirrored: as far left of the middle's left edge as the tree put it right.
240                let dx = l[k].0 - lx;
241                places[*i] = (w - dx - nodes[*i].width, l[k].1 - ly);
242            }
243        }
244        // What no branch reaches stays below, in a row.
245        let placed: Vec<usize> = right.iter().chain(&left).copied().collect();
246        let bottom = placed.iter().map(|i| places[*i].1 + nodes[*i].height).fold(0.0_f32, f32::max);
247        let mut x = 0.0;
248        for i in (0..n).filter(|i| !placed.contains(i)) {
249            places[i] = (x, bottom + self.layer_gap);
250            x += nodes[i].width + self.node_gap;
251        }
252        places
253    }

255    /// Where each node goes in a tree: the left and top edges.
256    fn tree_places(&self, nodes: &[FlowNode], links: &[(usize, usize)]) -> Vec<(f32, f32)> {
257        let n = nodes.len();
258        // Who hangs under whom: walked breadth first from the nodes nothing leads to, then from
259        // the first node not reached yet, so that a cycle still becomes a tree.
260        let mut children: Vec<Vec<usize>> = vec![Vec::new(); n];
261        let mut depth = vec![0usize; n];
262        let mut seen = vec![false; n];
263        let mut walked: Vec<usize> = Vec::with_capacity(n);
264        let mut roots: Vec<usize> = Vec::new();
265        let starts: Vec<usize> =
266            (0..n).filter(|&i| !links.iter().any(|&(_, b)| b == i)).chain(0..n).collect();
267        for start in starts {
268            if seen[start] {
269                continue;
270            }
271            seen[start] = true;
272            roots.push(start);
273            let mut next = std::collections::VecDeque::from([start]);
274            while let Some(u) = next.pop_front() {
275                walked.push(u);
276                for &(a, b) in links {
277                    if a == u && !seen[b] {
278                        seen[b] = true;
279                        depth[b] = depth[u] + 1;
280                        children[u].push(b);
281                        next.push_back(b);
282                    }
283                }
284            }
285        }
286        // How wide each subtree is across the layers, the deepest first.
287        let mut span = vec![0.0_f32; n];
288        for &u in walked.iter().rev() {
289            let own = self.extents(&nodes[u]).1;
290            let below: f32 = children[u].iter().map(|&c| span[c]).sum::<f32>()
291                + self.node_gap * children[u].len().saturating_sub(1) as f32;
292            span[u] = own.max(below);
293        }
294        // Along the layers: each depth as deep as its deepest node.
295        let levels = depth.iter().copied().max().map_or(0, |d| d + 1);
296        let mut along = vec![0.0_f32; levels];
297        let mut at = 0.0;
298        for (d, slot) in along.iter_mut().enumerate() {
299            *slot = at;
300            let deepest =
301                (0..n).filter(|&i| depth[i] == d).map(|i| self.extents(&nodes[i]).0).fold(0.0, f32::max);
302            at += deepest + self.layer_gap;
303        }
304        // Across: each subtree in its own band, its root centred over it.
305        let mut centre = vec![0.0_f32; n];
306        let mut bands: Vec<(usize, f32)> = Vec::new();
307        let mut edge = 0.0;
308        for &r in &roots {
309            bands.push((r, edge));
310            edge += span[r] + self.node_gap;
311        }
312        while let Some((u, start)) = bands.pop() {
313            centre[u] = start + span[u] / 2.0;
314            let below: f32 = children[u].iter().map(|&c| span[c]).sum::<f32>()
315                + self.node_gap * children[u].len().saturating_sub(1) as f32;
316            let mut child = start + (span[u] - below) / 2.0;
317            for &c in &children[u] {
318                bands.push((c, child));
319                child += span[c] + self.node_gap;
320            }
321        }
322        (0..n)
323            .map(|i| {
324                let (_, across) = self.extents(&nodes[i]);
325                let (a, c) = (along[depth[i]], centre[i] - across / 2.0);
326                match self.direction {
327                    FlowDirection::Right => (a, c),
328                    FlowDirection::Down => (c, a),
329                }
330            })
331            .collect()
332    }

334    /// The size of a node along the layers (its extent in the direction they follow) and across.

336        match self.direction {
337            FlowDirection::Right => (node.width, node.height),
338            FlowDirection::Down => (node.height, node.width),
339        }

342    /// The left and top edge of each node, with layers and orders already chosen.
343    fn place(
344        &self,
345        nodes: &[FlowNode],
346        links: &[(usize, usize)],
347        layer: &[usize],
348        order: &mut [Vec<usize>],
349    ) -> Vec<(f32, f32)> {
350        // Along the layers: each layer as deep as its deepest node.
351        let mut along = Vec::with_capacity(order.len());
352        let mut at = 0.0;
353        for members in order.iter() {
354            along.push(at);
355            let depth = members.iter().map(|&i| self.extents(&nodes[i]).0).fold(0.0, f32::max);
356            at += depth + self.layer_gap;
357        }
358        // Across: the centre of each node, the first layer packed, each later one pulled
359        // towards the centres of the neighbours before it.
360        let mut centre = vec![0.0_f32; nodes.len()];
361        for (l, members) in order.iter().enumerate() {
362            let wanted: Vec<Option<f32>> = members
363                .iter()
364                .map(|&i| {
365                    let near: Vec<f32> = links
366                        .iter()
367                        .filter_map(|&(a, b)| match () {
368                            _ if b == i && layer[a] < l => Some(centre[a]),
369                            _ if a == i && layer[b] < l => Some(centre[b]),
370                            _ => None,
371                        })
372                        .collect();
373                    (!near.is_empty()).then(|| near.iter().sum::<f32>() / near.len() as f32)
374                })
375                .collect();
376            self.spread(nodes, members, &wanted, &mut centre);
377        }
378        order.iter().enumerate().flat_map(|(l, members)| members.iter().map(move |&i| (l, i))).fold(
379            vec![(0.0, 0.0); nodes.len()],
380            |mut out, (l, i)| {
381                let (_, across) = self.extents(&nodes[i]);
382                let (a, c) = (along[l], centre[i] - across / 2.0);
383                out[i] = match self.direction {
384                    FlowDirection::Right => (a, c),
385                    FlowDirection::Down => (c, a),
386                };
387                out
388            },
389        )
390    }

392    /// Places the centres of one layer, in its order: as near as each wants to be, at least a
393    /// gap apart, and shifted as a whole so that, on average, each is where it wanted.
394    fn spread(&self, nodes: &[FlowNode], members: &[usize], wanted: &[Option<f32>], centre: &mut [f32]) {
395        let half = |i: usize| self.extents(&nodes[i]).1 / 2.0;
396        // Packed from zero, for the nodes that want nowhere in particular.
397        let mut packed = Vec::with_capacity(members.len());
398        let mut edge = 0.0;
399        for &i in members {
400            packed.push(edge + half(i));
401            edge += half(i) * 2.0 + self.node_gap;
402        }
403        let mut placed: Vec<f32> = Vec::with_capacity(members.len());
404        for (k, &i) in members.iter().enumerate() {
405            let want = wanted[k].unwrap_or(packed[k]);
406            let least = match k {
407                0 => f32::NEG_INFINITY,
408                _ => placed[k - 1] + half(members[k - 1]) + self.node_gap + half(i),
409            };
410            placed.push(want.max(least));
411        }
412        let pulled: Vec<(f32, f32)> =
413            placed.iter().zip(wanted).filter_map(|(p, w)| w.map(|w| (*p, w))).collect();
414        let shift = match pulled.len() {
415            0 => 0.0,
416            count => pulled.iter().map(|(p, w)| w - p).sum::<f32>() / count as f32,
417        };
418        for (&i, p) in members.iter().zip(placed) {
419            centre[i] = p + shift;
420        }
421    }
422}

424/// The connections between two different nodes, by index, once each, ignoring the ports.
425/// The middle of a mind map: the first node nothing leads to.
426fn middle_of(graph: &FlowGraph) -> Option<usize> {
427    (0..graph.nodes.len()).find(|i| !graph.edges.iter().any(|e| e.to.node == graph.nodes[*i].id))
428}

430fn links(graph: &FlowGraph) -> Vec<(usize, usize)> {
431    let index = |id: &str| graph.nodes.iter().position(|n| n.id == id);
432    let mut links: Vec<(usize, usize)> = Vec::new();
433    for edge in &graph.edges {
434        if let (Some(a), Some(b)) = (index(&edge.from.node), index(&edge.to.node))
435            && a != b
436            && !links.contains(&(a, b))
437        {
438            links.push((a, b));
439        }
440    }
441    links
442}

444/// The layer of each node: one after the longest path that reaches it, once the connections
445/// that close a cycle are turned back.
446fn layers(n: usize, links: &[(usize, usize)]) -> Vec<usize> {
447    let forward = acyclic(n, links);
448    let mut incoming = vec![0usize; n];
449    for &(_, b) in &forward {
450        incoming[b] += 1;
451    }
452    let mut ready: Vec<usize> = (0..n).filter(|&i| incoming[i] == 0).rev().collect();
453    let mut layer = vec![0usize; n];
454    while let Some(a) = ready.pop() {
455        for &(_, b) in forward.iter().filter(|(from, _)| *from == a) {
456            layer[b] = layer[b].max(layer[a] + 1);
457            incoming[b] -= 1;
458            if incoming[b] == 0 {
459                ready.push(b);
460            }
461        }
462    }
463    layer
464}

466/// The connections with those that close a cycle turned back, found by walking the nodes in
467/// their order, depth first.
468fn acyclic(n: usize, links: &[(usize, usize)]) -> Vec<(usize, usize)> {
469    // 0 not seen, 1 on the path being walked, 2 done.
470    let mut state = vec![0u8; n];
471    let mut back: Vec<(usize, usize)> = Vec::new();
472    for start in 0..n {
473        if state[start] != 0 {
474            continue;
475        }
476        let mut stack: Vec<(usize, usize)> = vec![(start, 0)];
477        state[start] = 1;
478        while let Some(&mut (node, ref mut next)) = stack.last_mut() {
479            let out: Vec<usize> = links.iter().filter(|(a, _)| *a == node).map(|(_, b)| *b).collect();
480            if let Some(&target) = out.get(*next) {
481                *next += 1;
482                match state[target] {
483                    0 => {
484                        state[target] = 1;
485                        stack.push((target, 0));
486                    }
487                    1 => back.push((node, target)),
488                    _ => {}
489                }
490            } else {
491                state[node] = 2;
492                stack.pop();
493            }
494        }
495    }
496    links.iter().map(|&(a, b)| if back.contains(&(a, b)) { (b, a) } else { (a, b) }).collect()
497}

499/// The nodes of each layer, ordered so that lines cross little: first as the diagram has them,
500/// then by the average place of their neighbours in the layer before (sweeping down) and after
501/// (sweeping up).
502fn order(n: usize, links: &[(usize, usize)], layer: &[usize]) -> Vec<Vec<usize>> {
503    let count = layer.iter().copied().max().map_or(0, |m| m + 1);
504    let mut order: Vec<Vec<usize>> = vec![Vec::new(); count];
505    for i in 0..n {
506        order[layer[i]].push(i);
507    }
508    let neighbours = |i: usize, of: usize| -> Vec<usize> {
509        links
510            .iter()
511            .filter_map(|&(a, b)| match () {
512                _ if a == i && layer[b] == of => Some(b),
513                _ if b == i && layer[a] == of => Some(a),
514                _ => None,
515            })
516            .collect()
517    };
518    for sweep in 0..SWEEPS * 2 {
519        let down = sweep % 2 == 0;
520        let layers: Vec<usize> =
521            if down { (1..count).collect() } else { (0..count.saturating_sub(1)).rev().collect() };
522        for l in layers {
523            let other = if down { l - 1 } else { l + 1 };
524            let place = |j: usize| order[other].iter().position(|&k| k == j).unwrap_or(0) as f32;
525            let mut keyed: Vec<(f32, usize)> = order[l]
526                .iter()
527                .enumerate()
528                .map(|(k, &i)| {
529                    let near = neighbours(i, other);
530                    let key = match near.len() {
531                        0 => k as f32,
532                        count => near.iter().map(|&j| place(j)).sum::<f32>() / count as f32,
533                    };
534                    (key, i)
535                })
536                .collect();
537            keyed.sort_by(|a, b| a.0.total_cmp(&b.0));
538            order[l] = keyed.into_iter().map(|(_, i)| i).collect();
539        }
540    }
541    order
542}

562            [("m", "a"), ("m", "b"), ("m", "c"), ("m", "d"), ("a", "a1")].iter().enumerate()
563        {
564            g = g.edge(FlowEdge::new(
565                format!("e{i}"),
566                FlowEndpoint::new(*from, "out"),
567                FlowEndpoint::new(*to, "in"),
568            ));
569        }
570        FlowLayout::mind_map().apply(&mut g);
571        let x = |id: &str| g.find_node(id).map_or(0.0, |n| n.rect().center().x);
572        let m = x("m");
573        assert!(x("a") > m && x("b") > m && x("a1") > x("a"), "half to the right");
574        assert!(x("c") < m && x("d") < m, "half to the left");
575        let side = |id: &str, port: &str| g.find_node(id).and_then(|n| n.find_port(port)).map(|p| p.side);
576        assert_eq!(side("c", "in"), Some(crate::flow::PortSide::Right), "facing the middle");
577        assert_eq!(side("c", "out"), Some(crate::flow::PortSide::Left));
578        assert_eq!(side("a", "in"), Some(crate::flow::PortSide::Left));
579        let before = g.clone();
580        FlowLayout::mind_map().apply(&mut g);
581        assert_eq!(g, before, "arranging again changes nothing");
582    }
