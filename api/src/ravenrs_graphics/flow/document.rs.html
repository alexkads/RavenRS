1//! Diagrams kept by name: what a library of diagrams writes to storage and reads back.
2
3use serde::{Deserialize, Serialize};
4
5use super::FlowGraph;
6
7/// A diagram with a name, as a library keeps it.
8#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
9#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
10pub struct FlowDocument {
11    /// Unique among the documents of a shelf.
12    pub id: String,
13    /// What the person called it.
14    pub name: String,
15    /// The diagram.
16    pub graph: FlowGraph,
17}
18
19/// Every document a library keeps, in the order they were made.
20#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
21#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
22pub struct FlowShelf {
23    /// The documents.
24    pub documents: Vec<FlowDocument>,
25}
26
27#[cfg_attr(feature = "live", ravenrs_live::expose)]
28impl FlowShelf {
29    /// The shelf written as JSON, which [`Self::from_json`] reads back.
30    pub fn to_json(&self) -> String {
31        serde_json::to_string(self).unwrap_or_default()
32    }
33
34    /// A shelf read from JSON; nothing readable gives an empty shelf, and says why.
35    pub fn from_json(text: &str) -> Result<Self, String> {
36        serde_json::from_str(text).map_err(|e| e.to_string())
37    }
38
39    /// The document with this id.
40    pub fn find(&self, id: &str) -> Option<&FlowDocument> {
41        self.documents.iter().find(|d| d.id == id)
42    }
43
44    /// An id no document has yet.
45    pub fn fresh_id(&self) -> String {
46        (1..).map(|n| format!("d{n}")).find(|id| self.find(id).is_none()).unwrap_or_default()
47    }
48
49    /// Keeps `document`, in place of the one with its id, or after the others.
50    pub fn put(&mut self, document: FlowDocument) {
51        match self.documents.iter_mut().find(|d| d.id == document.id) {
52            Some(kept) => *kept = document,
53            None => self.documents.push(document),
54        }
55    }
56
57    /// Forgets the document with this id. Says whether there was one.
58    pub fn remove(&mut self, id: &str) -> bool {
59        let before = self.documents.len();
60        self.documents.retain(|d| d.id != id);
61        before != self.documents.len()
62    }
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68    use crate::flow::FlowNode;
69
70    #[test]
71    fn a_shelf_keeps_replaces_and_forgets_documents() {
72        let mut shelf = FlowShelf::default();
73        let id = shelf.fresh_id();
74        let graph = FlowGraph::new().node(FlowNode::new("a", "", "A", 0.0, 0.0));
75        shelf.put(FlowDocument { id: id.clone(), name: "One".into(), graph: graph.clone() });
76        assert_ne!(shelf.fresh_id(), id);
77        shelf.put(FlowDocument { id: id.clone(), name: "Renamed".into(), graph });
78        assert_eq!(shelf.documents.len(), 1);
79        assert_eq!(shelf.find(&id).map(|d| d.name.as_str()), Some("Renamed"));
80        let back = FlowShelf::from_json(&shelf.to_json());
81        assert_eq!(back.as_ref(), Ok(&shelf));
82        assert!(shelf.remove(&id) && shelf.documents.is_empty());
83        assert!(FlowShelf::from_json("not json").is_err());
84    }
85}
