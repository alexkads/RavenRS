1//! How one node or one line looks apart from the theme: colors written as `#rrggbb`, so they
2//! go to a file and come back as they are.
3
4use ravenrs_core::Color;
5use serde::{Deserialize, Serialize};

7/// What a node shows apart from the theme. Every part left out is the theme's.
8#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
9#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
10pub struct FlowStyle {
11    /// Inside the outline, as `#rrggbb` or `#rrggbbaa`.
12    #[serde(default, skip_serializing_if = "Option::is_none")]
13    pub fill: Option<String>,
14    /// The outline's color.
15    #[serde(default, skip_serializing_if = "Option::is_none")]
16    pub stroke: Option<String>,
17    /// The outline's width, in units of the diagram.
18    #[serde(default, skip_serializing_if = "Option::is_none")]
19    pub stroke_width: Option<f32>,
20    /// The words' color.
21    #[serde(default, skip_serializing_if = "Option::is_none")]
22    pub text: Option<String>,
23    /// The title under the outline instead of inside it, as small round or diamond shapes
24    /// carry it.
25    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
26    pub title_below: bool,
27    /// A sign drawn large in the middle of the outline, such as `✕` or `+`.
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub marker: Option<String>,
30}
31
32#[cfg_attr(feature = "live", ravenrs_live::expose)]
33impl FlowStyle {
34    /// The theme's look.
35    pub fn new() -> Self {
36        Self::default()
37    }

39    /// Filled with `color` (`#rrggbb`).
40    pub fn fill(mut self, color: impl Into<String>) -> Self {
41        self.fill = Some(color.into());
42        self
43    }
44
45    /// Outlined in `color`, `width` units wide.
46    pub fn stroke(mut self, color: impl Into<String>, width: f32) -> Self {
47        self.stroke = Some(color.into());
48        self.stroke_width = Some(width);
49        self
50    }
51
52    /// Only the outline's width, in the theme's color.
53    pub fn stroke_width(mut self, width: f32) -> Self {
54        self.stroke_width = Some(width);
55        self
56    }
57
58    /// Words in `color`.
59    pub fn text(mut self, color: impl Into<String>) -> Self {
60        self.text = Some(color.into());
61        self
62    }
63
64    /// The title under the outline.
65    pub fn title_below(mut self) -> Self {
66        self.title_below = true;
67        self
68    }
69
70    /// A sign drawn large in the middle.
71    pub fn marker(mut self, marker: impl Into<String>) -> Self {
72        self.marker = Some(marker.into());
73        self
74    }
75
76    /// Whether it is all the theme's.
77    pub fn is_plain(&self) -> bool {
78        *self == Self::default()
79    }
80
81    /// The fill as a color, when it is one.
82    pub fn fill_color(&self) -> Option<Color> {
83        self.fill.as_deref().and_then(parse_color)
84    }
85
86    /// The outline's color, when it is one.
87    pub fn stroke_color(&self) -> Option<Color> {
88        self.stroke.as_deref().and_then(parse_color)
89    }
90
91    /// The words' color, when it is one.
92    pub fn text_color(&self) -> Option<Color> {
93        self.text.as_deref().and_then(parse_color)
94    }
95}

97/// A color written `#rgb`, `#rrggbb` or `#rrggbbaa`; anything else is none.
98pub fn parse_color(text: &str) -> Option<Color> {
99    let hex = text.trim().strip_prefix('#')?;
100    let digit = |i: usize, n: usize| u8::from_str_radix(hex.get(i..i + n)?, 16).ok();
101    let (r, g, b, a) = match hex.len() {
102        3 => {
103            let one = |i| digit(i, 1).map(|v| v * 17);
104            (one(0)?, one(1)?, one(2)?, 255)
105        }
106        6 => (digit(0, 2)?, digit(2, 2)?, digit(4, 2)?, 255),
107        8 => (digit(0, 2)?, digit(2, 2)?, digit(4, 2)?, digit(6, 2)?),
108        _ => return None,
109    };
110    let f = |v: u8| f32::from(v) / 255.0;
111    Some(Color::rgba(f(r), f(g), f(b), f(a)))
112}

114/// A color written `#rrggbb` (with `aa` when it is not opaque).
115pub fn color_hex(color: Color) -> String {
116    let [r, g, b, a] = color.to_array();
117    let byte = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
118    if a >= 0.999 {
119        format!("#{:02x}{:02x}{:02x}", byte(r), byte(g), byte(b))
120    } else {
121        format!("#{:02x}{:02x}{:02x}{:02x}", byte(r), byte(g), byte(b), byte(a))
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128
129    #[test]
130    fn colors_are_read_and_written_as_hex() {
131        assert_eq!(parse_color("#fff"), Some(Color::rgba(1.0, 1.0, 1.0, 1.0)));
132        assert_eq!(parse_color("#ff000080").map(|c| c.to_array()[3] > 0.49), Some(true));
133        assert_eq!(parse_color("red"), None);
134        assert_eq!(parse_color("#12345"), None);
135        let c = parse_color("#3a7bd5").expect("a color");
136        assert_eq!(color_hex(c), "#3a7bd5");
137        let style = FlowStyle::new().fill("#3a7bd5").stroke("#000", 3.0).title_below().marker("✕");
138        assert!(!style.is_plain() && FlowStyle::new().is_plain());
139        assert_eq!(style.fill_color(), Some(c));
140    }
