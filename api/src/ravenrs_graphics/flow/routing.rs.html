1//! Lines at right angles that go around the nodes in their way, and a cache so that a line is
2//! worked out again only when something near it changes.
3//!
4//! A [`EdgeShape::Step`](super::EdgeShape::Step) line first tries the usual three runs; when
5//! they cross a node, it searches a coarse grid made of the edges of the nodes near it (each
6//! with some room around) for the shortest way at right angles that crosses none of them, each
7//! turn costing as much as a stretch of line, so that clean routes win.
8
9use std::collections::HashMap;
10use std::hash::{DefaultHasher, Hash, Hasher};
11
12use ravenrs_core::{Point, Rect};
13
14use super::{PORT_RUN, PortSide};

16/// The room a line keeps from a node it goes around, in units of the diagram.
17pub const ROUTE_MARGIN: f32 = 12.0;
18/// How far around the ends of a line a node can be and still be in its way.
19const NEAR: f32 = 240.0;
20/// What a turn costs, in units of length.
21const TURN_COST: f32 = 40.0;
22
23/// The corners of a line at right angles from `from`, leaving a port on `from_side`, to `to`,
24/// arriving at a port on `to_side`, going around `obstacles` (with [`ROUTE_MARGIN`] of room).
25/// The first and last points are the ports; the runs out of them come next.
26pub fn route_around(
27    from: Point,
28    from_side: PortSide,
29    to: Point,
30    to_side: PortSide,
31    obstacles: &[Rect],
32) -> Vec<Point> {
33    let plain = super::step_points(from, from_side, to, to_side);
34    let walls: Vec<Rect> = obstacles.iter().map(|r| r.inflate(ROUTE_MARGIN)).collect();
35    let clear = |points: &[Point]| {
36        points.windows(2).skip(1).take(points.len().saturating_sub(3)).all(|w| !blocked(w[0], w[1], &walls))
37    };
38    if clear(&plain) {
39        return plain;
40    }
41    let (a, b) = (from_side.outward(), to_side.outward());
42    let start = Point::new(from.x + a.x * PORT_RUN, from.y + a.y * PORT_RUN);
43    let end = Point::new(to.x + b.x * PORT_RUN, to.y + b.y * PORT_RUN);
44    match search(start, from_side.is_horizontal(), end, &walls) {
45        Some(mut middle) => {
46            let mut points = vec![from];
47            points.append(&mut middle);
48            points.push(to);
49            tidy(points)
50        }
51        // Boxed in on every side: the plain line, through whatever is there.
52        None => plain,
53    }
54}

56/// Whether the straight run from `p` to `q` (across or down) goes through the inside of a wall.
57fn blocked(p: Point, q: Point, walls: &[Rect]) -> bool {
58    walls.iter().any(|r| {
59        if (p.y - q.y).abs() < 1e-3 {
60            let (x0, x1) = (p.x.min(q.x), p.x.max(q.x));
61            p.y > r.y() && p.y < r.max_y() && x1 > r.x() && x0 < r.max_x()
62        } else {
63            let (y0, y1) = (p.y.min(q.y), p.y.max(q.y));
64            p.x > r.x() && p.x < r.max_x() && y1 > r.y() && y0 < r.max_y()
65        }
66    })
67}

69/// The shortest way at right angles from `start` to `end` on the grid of the walls' edges,
70/// leaving `start` across when `across`, each turn costing [`TURN_COST`]; its corners, ends
71/// included.
72fn search(start: Point, across: bool, end: Point, walls: &[Rect]) -> Option<Vec<Point>> {
73    let mut xs = vec![start.x, end.x, (start.x + end.x) / 2.0];
74    let mut ys = vec![start.y, end.y, (start.y + end.y) / 2.0];
75    for r in walls {
76        xs.extend([r.x(), r.max_x()]);
77        ys.extend([r.y(), r.max_y()]);
78    }
79    for v in [&mut xs, &mut ys] {
80        v.sort_by(f32::total_cmp);
81        v.dedup_by(|a, b| (*a - *b).abs() < 1e-3);
82    }
83    let (w, h) = (xs.len(), ys.len());
84    let index = |x: f32, y: f32| {
85        let i = xs.iter().position(|v| (v - x).abs() < 1e-3)?;
86        let j = ys.iter().position(|v| (v - y).abs() < 1e-3)?;
87        Some(j * w + i)
88    };
89    let (from, to) = (index(start.x, start.y)?, index(end.x, end.y)?);
90    let point = |k: usize| Point::new(xs[k % w], ys[k / w]);
91    let inside =
92        |p: Point| walls.iter().any(|r| p.x > r.x() && p.x < r.max_x() && p.y > r.y() && p.y < r.max_y());
93    // States are (grid point, heading across?); a plain scan for the cheapest open state is
94    // enough for the few dozen points near one line.
95    let states = w * h * 2;
96    let mut cost = vec![f32::INFINITY; states];
97    let mut back = vec![usize::MAX; states];
98    let mut done = vec![false; states];
99    let first = from * 2 + usize::from(across);
100    cost[first] = 0.0;
101    loop {
102        let open = (0..states)
103            .filter(|&s| !done[s] && cost[s].is_finite())
104            .min_by(|&a, &b| cost[a].total_cmp(&cost[b]))?;
105        done[open] = true;
106        let (k, heading_across) = (open / 2, open % 2 == 1);
107        if k == to {
108            let mut corners = vec![point(k)];
109            let mut s = open;
110            while back[s] != usize::MAX {
111                s = back[s];
112                corners.push(point(s / 2));
113            }
114            corners.reverse();
115            return Some(corners);
116        }
117        let (i, j) = (k % w, k / w);
118        let steps = [
119            (i.checked_sub(1), Some(j), true),
120            ((i + 1 < w).then_some(i + 1), Some(j), true),
121            (Some(i), j.checked_sub(1), false),
122            (Some(i), (j + 1 < h).then_some(j + 1), false),
123        ];
124        for (ni, nj, goes_across) in steps {
125            let (Some(ni), Some(nj)) = (ni, nj) else { continue };
126            let next = nj * w + ni;
127            let (p, q) = (point(k), point(next));
128            if (inside(q) && next != to) || blocked(p, q, walls) {
129                continue;
130            }
131            let turn = if goes_across == heading_across { 0.0 } else { TURN_COST };
132            let s = next * 2 + usize::from(goes_across);
133            let c = cost[open] + (q.x - p.x).abs() + (q.y - p.y).abs() + turn;
134            if c < cost[s] {
135                cost[s] = c;
136                back[s] = open;
137            }
138        }
139    }
140}

142/// The corners only: points that do not turn the line are dropped.
143fn tidy(points: Vec<Point>) -> Vec<Point> {
144    let mut kept: Vec<Point> = Vec::with_capacity(points.len());
145    for p in points {
146        if kept.last().is_some_and(|q: &Point| (q.x - p.x).abs() < 1e-3 && (q.y - p.y).abs() < 1e-3) {
147            continue;
148        }
149        if kept.len() >= 2 {
150            let (u, v) = (kept[kept.len() - 2], kept[kept.len() - 1]);
151            let cross = (v.x - u.x) * (p.y - v.y) - (v.y - u.y) * (p.x - v.x);
152            if cross.abs() < 1e-3 {
153                kept.pop();
154            }
155        }
156        kept.push(p);
157    }
158    kept
159}

161/// The nodes near a line from `from` to `to`, whose boxes it must go around.
162pub(super) fn near(from: Point, to: Point, boxes: impl Iterator<Item = Rect>) -> Vec<Rect> {
163    let (x0, y0) = (from.x.min(to.x) - NEAR, from.y.min(to.y) - NEAR);
164    let (x1, y1) = (from.x.max(to.x) + NEAR, from.y.max(to.y) + NEAR);
165    boxes.filter(|r| r.max_x() > x0 && r.x() < x1 && r.max_y() > y0 && r.y() < y1).collect()
166}

168/// Lines already worked out, each kept with what it was worked out from, so that drawing a
169/// diagram again works out only the lines whose ends or nearby nodes changed.
170#[derive(Clone, Debug, Default)]
171pub struct RouteCache {
172    routes: HashMap<String, (u64, Vec<Point>)>,
173    computed: usize,
174}
175
176impl RouteCache {
177    /// Nothing worked out yet.
178    pub fn new() -> Self {
179        Self::default()
180    }
181
182    /// How many lines were worked out (not found already worked out) since it was made.
183    pub fn computed(&self) -> usize {
184        self.computed
185    }
186
187    /// The line `id` from these ends around these boxes, worked out unless it was already.
188    pub(super) fn get(
189        &mut self,
190        id: &str,
191        ends: (Point, PortSide, Point, PortSide),
192        boxes: &[Rect],
193    ) -> Vec<Point> {
194        let mut hasher = DefaultHasher::new();
195        for v in [ends.0.x, ends.0.y, ends.2.x, ends.2.y] {
196            v.to_bits().hash(&mut hasher);
197        }
198        (ends.1 as u8, ends.3 as u8).hash(&mut hasher);
199        for r in boxes {
200            for v in [r.x(), r.y(), r.width(), r.height()] {
201                v.to_bits().hash(&mut hasher);
202            }
203        }
204        let key = hasher.finish();
205        if let Some((known, points)) = self.routes.get(id)
206            && *known == key
207        {
208            return points.clone();
209        }
210        self.computed += 1;
211        let points = route_around(ends.0, ends.1, ends.2, ends.3, boxes);
212        self.routes.insert(id.to_owned(), (key, points.clone()));
213        points
214    }
