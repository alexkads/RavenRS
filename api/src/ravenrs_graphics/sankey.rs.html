10/// Which column a node goes to, among those its flows allow.
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
12#[non_exhaustive]
13pub enum SankeyAlign {
14    /// As far left as its flows allow.
15    Left,
16    /// As far right as its flows allow.
17    Right,
18    /// Like [`Self::Left`], except that a node with nothing coming in sits right
19    /// before the first thing it feeds.
20    Center,
21    /// Like [`Self::Left`], except that a node with nothing going out goes to the last
22    /// column.
23    #[default]
24    Justify,
25}
26
27/// A flow from one node to another, given by their places in the list of nodes.
28#[derive(Clone, Copy, Debug, PartialEq)]
29pub struct SankeyLink {
30    /// Where it comes from.
31    pub source: usize,
32    /// Where it goes.
33    pub target: usize,
34    /// How much. One that is not above zero has no ribbon, and does not count.
35    pub value: f64,
36}

38/// How a graph is laid out.
39#[derive(Clone, Copy, Debug, PartialEq)]
40#[non_exhaustive]
41pub struct SankeyStyle {
42    /// How wide a node is.
43    pub node_width: f32,
44    /// The room between two nodes of a column.
45    pub node_gap: f32,
46    /// Which column a node goes to.
47    pub align: SankeyAlign,
48    /// How many passes put nodes closer to what they connect to.
49    pub passes: usize,
50}
51
52impl Default for SankeyStyle {
53    fn default() -> Self {
54        Self { node_width: 12.0, node_gap: 14.0, align: SankeyAlign::default(), passes: 16 }
55    }
56}

58/// Why a graph has no layout.
59#[derive(Clone, Debug, PartialEq, Eq)]
60#[non_exhaustive]
61pub enum SankeyError {
62    /// The flows go in a circle, through these nodes, the first one again at the end.
63    Cycle(Vec<usize>),
64    /// A flow names a node past the end of the list.
65    NoSuchNode(usize),
66}
67
68impl fmt::Display for SankeyError {
69    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
70        match self {
71            Self::Cycle(nodes) => {
72                let names: Vec<String> = nodes.iter().map(ToString::to_string).collect();
73                write!(f, "the flows go in a circle: {}", names.join(" -> "))
74            }
75            Self::NoSuchNode(node) => write!(f, "a flow names node {node}, which does not exist"),
76        }
77    }
78}

98/// A flow as it landed: a ribbon from the side of a node to the side of another.
99#[derive(Clone, Copy, Debug, PartialEq)]
100#[non_exhaustive]
101pub struct SankeyFlow {
102    /// Where it comes from.
103    pub source: usize,
104    /// Where it goes.
105    pub target: usize,
106    /// How much.
107    pub value: f64,
108    /// Where it starts: on this side, at this height, its middle.
109    pub from: Point,
110    /// Where it ends.
111    pub to: Point,
112    /// How thick it is.
113    pub width: f32,
114}
115
116impl SankeyFlow {
117    /// The two control points' place along the way, halfway across.
118    fn middle(&self) -> f32 {
119        (self.from.x + self.to.x) / 2.0
120    }
121
122    /// The outline of the ribbon.
123    pub fn path(&self) -> Vec<PathSegment> {
124        let (half, middle) = (self.width / 2.0, self.middle());
125        let at = Point::new;
126        let (x0, x1, y0, y1) = (self.from.x, self.to.x, self.from.y, self.to.y);
127        vec![
128            PathSegment::Move(at(x0, y0 - half)),
129            PathSegment::Cubic(at(middle, y0 - half), at(middle, y1 - half), at(x1, y1 - half)),
130            PathSegment::Line(at(x1, y1 + half)),
131            PathSegment::Cubic(at(middle, y1 + half), at(middle, y0 + half), at(x0, y0 + half)),
132            PathSegment::Close,
133        ]
134    }
135
136    /// Whether the ribbon covers this point.
137    pub fn contains(&self, point: Point) -> bool {
138        if point.x < self.from.x || point.x > self.to.x {
139            return false;
140        }
141        let (x0, x1, middle) = (self.from.x, self.to.x, self.middle());
142        let x_at = |t: f32| {
143            let u = 1.0 - t;
144            u * u * u * x0 + 3.0 * u * t * middle * (u + t) + t * t * t * x1
145        };
146        // Across, a curve only ever goes forward, so halving finds the place on it.
147        let (mut low, mut high) = (0.0_f32, 1.0_f32);
148        for _ in 0..16 {
149            let t = (low + high) / 2.0;
150            if x_at(t) < point.x {
151                low = t;
152            } else {
153                high = t;
154            }
155        }
156        let t = (low + high) / 2.0;
157        let share = t * t * (3.0 - 2.0 * t);
158        let y = self.from.y + (self.to.y - self.from.y) * share;
159        (point.y - y).abs() <= self.width / 2.0
160    }
161}

172/// A diagram laid out.
173#[derive(Clone, Debug, Default, PartialEq)]
174#[non_exhaustive]
175pub struct SankeyLayout {
176    /// Every node, in the order given.
177    pub nodes: Vec<SankeyNode>,
178    /// Every flow with a value, in the order given.
179    pub flows: Vec<SankeyFlow>,
180    /// How many columns there are.
181    pub columns: usize,
182}
183
184impl SankeyLayout {
185    /// What is under `point`: a node, over the flows that reach it.
186    pub fn hit(&self, point: Point) -> Option<SankeyHit> {
187        if let Some(node) = self.nodes.iter().position(|node| node.rect.contains(point)) {
188            return Some(SankeyHit::Node(node));
189        }
190        // The last one drawn is the one on top.
191        self.flows.iter().rposition(|flow| flow.contains(point)).map(SankeyHit::Flow)
192    }
193}

195/// Each node's order so that flows only go forward, or the circle that prevents it.
196fn in_order(count: usize, links: &[SankeyLink]) -> Result<Vec<usize>, SankeyError> {
197    let mut waiting = vec![0_usize; count];
198    for link in links {
199        waiting[link.target] += 1;
200    }
201    let mut ready: Vec<usize> = (0..count).filter(|node| waiting[*node] == 0).collect();
202    let mut order = Vec::with_capacity(count);
203    while let Some(node) = ready.pop() {
204        order.push(node);
205        for link in links.iter().filter(|link| link.source == node) {
206            waiting[link.target] -= 1;
207            if waiting[link.target] == 0 {
208                ready.push(link.target);
209            }
210        }
211    }
212    if order.len() == count {
213        return Ok(order);
214    }
215    // What is left all has something coming in from what is left: going backwards
216    // from any of them comes to a node seen before.
217    let Some(start) = (0..count).find(|node| waiting[*node] > 0) else { return Ok(order) };
218    let mut path = vec![start];
219    loop {
220        let last = path[path.len() - 1];
221        let previous = links.iter().find(|l| l.target == last && waiting[l.source] > 0).map(|l| l.source);
222        let Some(previous) = previous else { return Err(SankeyError::Cycle(path)) };
223        if let Some(seen) = path.iter().position(|node| *node == previous) {
224            let mut circle: Vec<usize> = path[seen..].to_vec();
225            circle.push(previous);
226            circle.reverse();
227            return Err(SankeyError::Cycle(circle));
228        }
229        path.push(previous);
230    }
231}

233fn usable(links: &[SankeyLink]) -> Vec<SankeyLink> {
234    links.iter().copied().filter(|link| link.value.is_finite() && link.value > 0.0).collect()
235}

237/// Whether a graph can be laid out, without laying it out.
238pub fn check(count: usize, links: &[SankeyLink]) -> Result<(), SankeyError> {
239    let links = usable(links);
240    if let Some(link) = links.iter().find(|link| link.source >= count || link.target >= count) {
241        return Err(SankeyError::NoSuchNode(link.source.max(link.target)));
242    }
243    in_order(count, &links).map(|_| ())
244}

246/// The column of each node, and how many there are.
247fn columns_of(
248    count: usize,
249    links: &[SankeyLink],
250    order: &[usize],
251    align: SankeyAlign,
252) -> (Vec<usize>, usize) {
253    let mut depth = vec![0_usize; count];
254    for node in order {
255        for link in links.iter().filter(|link| link.source == *node) {
256            depth[link.target] = depth[link.target].max(depth[*node] + 1);
257        }
258    }
259    let mut height = vec![0_usize; count];
260    for node in order.iter().rev() {
261        for link in links.iter().filter(|link| link.target == *node) {
262            height[link.source] = height[link.source].max(height[*node] + 1);
263        }
264    }
265    let columns = depth.iter().copied().max().map_or(0, |deepest| deepest + 1);
266    let has_in = |node: usize| links.iter().any(|link| link.target == node);
267    let has_out = |node: usize| links.iter().any(|link| link.source == node);
268    let column = (0..count)
269        .map(|node| match align {
270            SankeyAlign::Left => depth[node],
271            SankeyAlign::Right => columns - 1 - height[node],
272            SankeyAlign::Center if !has_in(node) && has_out(node) => {
273                let first = links.iter().filter(|l| l.source == node).map(|l| depth[l.target]).min();
274                first.map_or(0, |at| at.saturating_sub(1))
275            }
276            SankeyAlign::Center => depth[node],
277            SankeyAlign::Justify if !has_out(node) && has_in(node) => columns - 1,
278            _ => depth[node],
279        })
280        .collect();
281    (column, columns)
282}

284/// Keeps a column's nodes from covering each other and inside the area.
285fn resolve(ys: &mut [f32], heights: &[f32], column: &mut [usize], area: Rect, gap: f32) {
286    column.sort_by(|a, b| ys[*a].total_cmp(&ys[*b]));
287    let mut next = area.y();
288    for node in column.iter() {
289        ys[*node] = ys[*node].max(next);
290        next = ys[*node] + heights[*node] + gap;
291    }
292    let mut limit = area.max_y();
293    for node in column.iter().rev() {
294        ys[*node] = ys[*node].min(limit - heights[*node]);
295        limit = ys[*node] - gap;
296    }
297}

299/// Lays a graph of `count` nodes out inside `area`. Flows with no value are left out.
300pub fn layout(
301    count: usize,
302    links: &[SankeyLink],
303    area: Rect,
304    style: &SankeyStyle,
305) -> Result<SankeyLayout, SankeyError> {
306    check(count, links)?;
307    let links = usable(links);
308    let order = in_order(count, &links)?;
309    let (column_of, columns) = columns_of(count, &links, &order, style.align);
310    if columns == 0 {
311        return Ok(SankeyLayout::default());
312    }
313    let mut value = vec![0.0_f64; count];
314    for (node, slot) in value.iter_mut().enumerate() {
315        let coming: f64 = links.iter().filter(|l| l.target == node).map(|l| l.value).sum();
316        let going: f64 = links.iter().filter(|l| l.source == node).map(|l| l.value).sum();
317        *slot = coming.max(going);
318    }
319    let mut by_column: Vec<Vec<usize>> = vec![Vec::new(); columns];
320    for node in 0..count {
321        by_column[column_of[node]].push(node);
322    }
323    // One scale for the whole diagram, the one the fullest column allows.
324    let scale = by_column
325        .iter()
326        .filter(|column| !column.is_empty())
327        .map(|column| {
328            let total: f64 = column.iter().map(|node| value[*node]).sum();
329            let room = area.height() - style.node_gap * (column.len() - 1) as f32;
330            if total > 0.0 { room.max(1.0) / total as f32 } else { f32::INFINITY }
331        })
332        .fold(f32::INFINITY, f32::min);
333    let scale = if scale.is_finite() { scale } else { 1.0 };
334    let heights: Vec<f32> = value.iter().map(|v| (*v as f32 * scale).max(1.0)).collect();
335
336    // Each column starts in the middle, and each pass brings each node closer to what
337    // it connects to, more boldly at first.
338    let mut ys = vec![area.y(); count];
339    for column in &by_column {
340        let total: f32 = column.iter().map(|node| heights[*node]).sum::<f32>()
341            + style.node_gap * column.len().saturating_sub(1) as f32;
342        let mut y = area.y() + (area.height() - total).max(0.0) / 2.0;
343        for node in column {
344            ys[*node] = y;
345            y += heights[*node] + style.node_gap;
346        }
347    }
348    for pass in 0..style.passes {
349        let strength = 0.99_f32.powi(pass as i32) * 0.8;
350        for column in (1..columns).chain((0..columns.saturating_sub(1)).rev()) {
351            let forward = pass % 2 == 0;
352            let nodes: Vec<usize> = by_column[column].clone();
353            for node in nodes {
354                let toward = |other: usize| ys[other] + heights[other] / 2.0;
355                let (mut sum, mut weight) = (0.0_f32, 0.0_f32);
356                for link in &links {
357                    let other = if forward && link.target == node {
358                        Some(link.source)
359                    } else if !forward && link.source == node {
360                        Some(link.target)
361                    } else {
362                        None
363                    };
364                    if let Some(other) = other {
365                        sum += toward(other) * link.value as f32;
366                        weight += link.value as f32;
367                    }
368                }
369                if weight > 0.0 {
370                    let wanted = sum / weight - heights[node] / 2.0;
371                    ys[node] += (wanted - ys[node]) * strength;
372                }
373            }
374            resolve(&mut ys, &heights, &mut by_column[column], area, style.node_gap);
375        }
376    }
377
378    // Many columns in little room: nodes get thinner, so flows still go forward.
379    let node_width = style.node_width.min(area.width() / (columns * 2) as f32);
380    let span = (area.width() - node_width).max(0.0);
381    let x_of = |column: usize| {
382        if columns > 1 { area.x() + span * column as f32 / (columns - 1) as f32 } else { area.x() }
383    };
384    let mut nodes: Vec<SankeyNode> = (0..count)
385        .map(|node| SankeyNode {
386            rect: Rect::new(x_of(column_of[node]), ys[node], node_width, heights[node]),
387            value: value[node],
388            column: column_of[node],
389            incoming: Vec::new(),
390            outgoing: Vec::new(),
391        })
392        .collect();
393    for (place, link) in links.iter().enumerate() {
394        nodes[link.source].outgoing.push(place);
395        nodes[link.target].incoming.push(place);
396    }
397    let middle = |node: usize| ys[node] + heights[node] / 2.0;
398    let mut starts = vec![0.0_f32; links.len()];
399    let mut ends = vec![0.0_f32; links.len()];
400    for node in nodes.iter_mut() {
401        // Ribbons leave and arrive in the order of what is at their other ends, so
402        // that they cross as little as they can.
403        node.outgoing.sort_by(|a, b| middle(links[*a].target).total_cmp(&middle(links[*b].target)));
404        node.incoming.sort_by(|a, b| middle(links[*a].source).total_cmp(&middle(links[*b].source)));
405        let mut y = node.rect.y();
406        for link in &node.outgoing {
407            starts[*link] = y + links[*link].value as f32 * scale / 2.0;
408            y += links[*link].value as f32 * scale;
409        }
410        let mut y = node.rect.y();
411        for link in &node.incoming {
412            ends[*link] = y + links[*link].value as f32 * scale / 2.0;
413            y += links[*link].value as f32 * scale;
414        }
415    }
416    let flows = links
417        .iter()
418        .enumerate()
419        .map(|(place, link)| SankeyFlow {
420            source: link.source,
421            target: link.target,
422            value: link.value,
423            from: Point::new(nodes[link.source].rect.max_x(), starts[place]),
424            to: Point::new(nodes[link.target].rect.x(), ends[place]),
425            width: link.value as f32 * scale,
426        })
427        .collect();
428    Ok(SankeyLayout { nodes, flows, columns })
429}

549        let inside = (0..300).map(|y| Point::new(quarter, y as f32)).filter(|p| flow.contains(*p)).count();
550        assert!((inside as f32 - flow.width).abs() <= 2.0, "{inside} {}", flow.width);
551    }
