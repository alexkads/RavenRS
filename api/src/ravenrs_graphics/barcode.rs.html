1//! Barcodes as shapes. What a code is comes from `ravenrs-barcode`, and it is all here too, so
2//! everything that draws finds both in one place.
3

10/// The shape of a pattern inside `rect`, centered and with its quiet zone.
11pub fn shape(pattern: &Pattern, rect: Rect, color: Color) -> Shape {
12    let mut path = Vec::new();
13    match pattern {
14        Pattern::Linear { modules, .. } => {
15            let module = rect.width() / (modules.len() as f32 + 2.0 * QUIET_LINEAR as f32);
16            let left = rect.x() + QUIET_LINEAR as f32 * module;
17            for (start, length) in runs(modules) {
18                let x = left + start as f32 * module;
19                append_rectangle(&mut path, x, rect.y(), x + length as f32 * module, rect.max_y(), 0.0);
20            }
21        }
22        Pattern::Matrix { size, cells } => {
23            let module = rect.width().min(rect.height()) / (*size as f32 + 2.0 * QUIET_MATRIX as f32);
24            let side = *size as f32 * module;
25            let (left, top) =
26                (rect.x() + (rect.width() - side) / 2.0, rect.y() + (rect.height() - side) / 2.0);
27            for (row, cells) in cells.chunks((*size).max(1)).enumerate() {
28                let y = top + row as f32 * module;
29                for (start, length) in runs(cells) {
30                    let x = left + start as f32 * module;
31                    append_rectangle(&mut path, x, y, x + length as f32 * module, y + module, 0.0);
32                }
33            }
34        }
35    }
36    Shape::filled(path, color)
37}

39#[cfg(test)]
40mod tests {
41    use super::*;
42    use crate::vector::PathSegment;
43
44    #[test]
45    fn shapes_stay_inside_their_box_and_keep_the_quiet_zone() {
46        let rect = Rect::new(10.0, 20.0, 200.0, 50.0);
47        for (symbology, data) in
48            [(Symbology::Code128, "ABC123"), (Symbology::Qr, "x"), (Symbology::Ean13, "400638133393")]
49        {
50            let pattern = encode(symbology, data).map_err(|e| e.to_string());
51            let drawn = shape(&pattern.unwrap_or_else(|e| panic!("{e}")), rect, Color::BLACK);
52            assert!(!drawn.path.is_empty());
53            for segment in &drawn.path {
54                if let PathSegment::Move(p) | PathSegment::Line(p) = segment {
55                    assert!(p.x > rect.x() && p.x < rect.max_x() + 0.001, "{p:?}");
56                    assert!(p.y >= rect.y() - 0.001 && p.y <= rect.max_y() + 0.001, "{p:?}");
57                }
58            }
59        }
60    }
