154/// The outline of `shape` filling `r`.
155pub fn shape_outline(shape: FlowShape, r: Rect) -> Vec<PathSegment> {
156    let (x, y, w, h) = (r.x(), r.y(), r.width(), r.height());
157    let p = Point::new;
158    let polygon = |points: &[Point]| {
159        let mut path: Vec<PathSegment> = points
160            .iter()
161            .enumerate()
162            .map(|(i, &q)| if i == 0 { PathSegment::Move(q) } else { PathSegment::Line(q) })
163            .collect();
164        path.push(PathSegment::Close);
165        path
166    };
167    // A quarter of an ellipse as a cubic: control points 0.552 of the way.
168    let k = 0.552_284_8;
169    match shape {
170        FlowShape::Card | FlowShape::Rectangle | FlowShape::Text => {
171            polygon(&[p(x, y), p(x + w, y), p(x + w, y + h), p(x, y + h)])
172        }
173        FlowShape::RoundedRectangle => {
174            let mut path = rounded(
175                &[p(x + w / 2.0, y), p(x + w, y), p(x + w, y + h), p(x, y + h), p(x, y), p(x + w / 2.0, y)],
176                w.min(h) / 2.0,
177            );
178            path.push(PathSegment::Close);
179            path
180        }
181        FlowShape::Ellipse => {
182            let (cx, cy, rx, ry) = (x + w / 2.0, y + h / 2.0, w / 2.0, h / 2.0);
183            vec![
184                PathSegment::Move(p(cx + rx, cy)),
185                PathSegment::Cubic(p(cx + rx, cy + ry * k), p(cx + rx * k, cy + ry), p(cx, cy + ry)),
186                PathSegment::Cubic(p(cx - rx * k, cy + ry), p(cx - rx, cy + ry * k), p(cx - rx, cy)),
187                PathSegment::Cubic(p(cx - rx, cy - ry * k), p(cx - rx * k, cy - ry), p(cx, cy - ry)),
188                PathSegment::Cubic(p(cx + rx * k, cy - ry), p(cx + rx, cy - ry * k), p(cx + rx, cy)),
189                PathSegment::Close,
190            ]
191        }
192        FlowShape::Diamond => {
193            polygon(&[p(x + w / 2.0, y), p(x + w, y + h / 2.0), p(x + w / 2.0, y + h), p(x, y + h / 2.0)])
194        }
195        FlowShape::Parallelogram => {
196            let skew = (w * 0.15).min(h * 0.5);
197            polygon(&[p(x + skew, y), p(x + w, y), p(x + w - skew, y + h), p(x, y + h)])
198        }
199        FlowShape::Document => {
200            let foot = y + h * 0.85;
201            let dip = h * 0.12;
202            vec![
203                PathSegment::Move(p(x, y)),
204                PathSegment::Line(p(x + w, y)),
205                PathSegment::Line(p(x + w, foot)),
206                PathSegment::Cubic(
207                    p(x + w * 0.75, foot - dip),
208                    p(x + w * 0.5, foot - dip),
209                    p(x + w * 0.5, foot),
210                ),
211                PathSegment::Cubic(p(x + w * 0.5, foot + dip), p(x + w * 0.25, foot + dip), p(x, foot)),
212                PathSegment::Close,
213            ]
214        }
215        FlowShape::Cylinder => {
216            let ry = (h * 0.12).min(w * 0.25);
217            let (cx, rx) = (x + w / 2.0, w / 2.0);
218            let (top, foot) = (y + ry, y + h - ry);
219            vec![
220                PathSegment::Move(p(x, top)),
221                PathSegment::Cubic(p(x, top - ry * k), p(cx - rx * k, y), p(cx, y)),
222                PathSegment::Cubic(p(cx + rx * k, y), p(x + w, top - ry * k), p(x + w, top)),
223                PathSegment::Line(p(x + w, foot)),
224                PathSegment::Cubic(p(x + w, foot + ry * k), p(cx + rx * k, y + h), p(cx, y + h)),
225                PathSegment::Cubic(p(cx - rx * k, y + h), p(x, foot + ry * k), p(x, foot)),
226                PathSegment::Close,
227            ]
228        }
229        FlowShape::Note => {
230            let fold = w.min(h) * 0.15;
231            polygon(&[p(x, y), p(x + w, y), p(x + w, y + h - fold), p(x + w - fold, y + h), p(x, y + h)])
232        }
233    }
234}

236/// Whether `p` is inside the closed polyline `outline`.
237pub fn contains(outline: &[Point], p: Point) -> bool {
238    let mut inside = false;
239    let n = outline.len();
240    for i in 0..n {
241        let (a, b) = (outline[i], outline[(i + 1) % n]);
242        if (a.y > p.y) != (b.y > p.y) && p.x < a.x + (p.y - a.y) / (b.y - a.y) * (b.x - a.x) {
243            inside = !inside;
244        }
245    }
246    inside
247}

480    /// Where a point of the unturned node goes once the node is turned around its middle.
481    pub fn turned(&self, p: Point) -> Point {
482        if self.rotation == 0.0 {
483            return p;
484        }
485        let c = self.rect().center();
486        let (sin, cos) = self.rotation.to_radians().sin_cos();
487        let (x, y) = (p.x - c.x, p.y - c.y);
488        Point::new(c.x + x * cos - y * sin, c.y + x * sin + y * cos)
489    }

491    /// The box around the node as it shows, turned.
492    pub fn extent(&self) -> Rect {
493        let r = self.rect();
494        if self.rotation == 0.0 {
495            return r;
496        }
497        let corners = [
498            r.origin,
499            Point::new(r.max_x(), r.y()),
500            Point::new(r.max_x(), r.max_y()),
501            Point::new(r.x(), r.max_y()),
502        ]
503        .map(|p| self.turned(p));
504        let (x0, y0) =
505            corners.iter().fold((f32::INFINITY, f32::INFINITY), |(x, y), p| (x.min(p.x), y.min(p.y)));
506        let (x1, y1) =
507            corners.iter().fold((f32::NEG_INFINITY, f32::NEG_INFINITY), |(x, y), p| (x.max(p.x), y.max(p.y)));
508        Rect::new(x0, y0, x1 - x0, y1 - y0)
509    }

511    /// The side a port on `side` faces once the node is turned: the nearest of the four.
512    pub fn facing(&self, side: PortSide) -> PortSide {
513        if self.rotation == 0.0 {
514            return side;
515        }
516        let out = side.outward();
517        let (sin, cos) = self.rotation.to_radians().sin_cos();
518        let (x, y) = (out.x * cos - out.y * sin, out.x * sin + out.y * cos);
519        if x.abs() >= y.abs() {
520            if x >= 0.0 { PortSide::Right } else { PortSide::Left }
521        } else if y >= 0.0 {
522            PortSide::Bottom
523        } else {
524            PortSide::Top
525        }
526    }

565    /// Where a port sits on the unturned node.
566    fn flat_port_point(&self, id: &str) -> Option<Point> {
567        let port = self.find_port(id)?;
568        let on_side: Vec<&FlowPort> = self.ports.iter().filter(|p| p.side == port.side).collect();
569        let index = on_side.iter().position(|p| p.id == id)? as f32;
570        let share = (index + 1.0) / (on_side.len() as f32 + 1.0);
571        let on_box = match port.side {
572            PortSide::Left => Point::new(self.x, self.y + self.height * share),
573            PortSide::Right => Point::new(self.x + self.width, self.y + self.height * share),
574            PortSide::Top => Point::new(self.x + self.width * share, self.y),
575            PortSide::Bottom => Point::new(self.x + self.width * share, self.y + self.height),
576        };
577        Some(self.onto_outline(on_box, port.side))
578    }

580    /// A point on the box's edge brought in, square to `side`, to where it meets the outline.
581    fn onto_outline(&self, p: Point, side: PortSide) -> Point {
582        if matches!(self.shape, FlowShape::Card | FlowShape::Rectangle | FlowShape::Note | FlowShape::Text) {
583            return p;
584        }
585        let outline = self.outline();
586        let n = outline.len();
587        let mut found: Option<f32> = None;
588        for i in 0..n {
589            let (a, b) = (outline[i], outline[(i + 1) % n]);
590            let at = if side.is_horizontal() {
591                ((a.y - p.y) * (b.y - p.y) <= 0.0 && a.y != b.y)
592                    .then(|| a.x + (p.y - a.y) / (b.y - a.y) * (b.x - a.x))
593            } else {
594                ((a.x - p.x) * (b.x - p.x) <= 0.0 && a.x != b.x)
595                    .then(|| a.y + (p.x - a.x) / (b.x - a.x) * (b.y - a.y))
596            };
597            let Some(at) = at else { continue };
598            let outer = matches!(side, PortSide::Right | PortSide::Bottom);
599            found = Some(match found {
600                Some(best) if outer => best.max(at),
601                Some(best) => best.min(at),
602                None => at,
603            });
604        }
605        match (found, side.is_horizontal()) {
606            (Some(x), true) => Point::new(x, p.y),
607            (Some(y), false) => Point::new(p.x, y),
608            (None, _) => p,
609        }
610    }
611}

831/// The version of the JSON [`FlowGraph::to_json`] writes. Version 1 had no `format` field;
832/// [`FlowGraph::from_json`] reads it and every version up to this one.
833pub const FLOW_FORMAT: u32 = 2;
834
835/// The field of a node's [`data`](FlowNode::data) that lists, as JSON pointers (`/target`,
836/// `/routes/0/to`), where its data names other nodes of the diagram by id. Pasting the node
837/// renames those ids as the pasted nodes are renamed.
838pub const NODE_REFS: &str = "$node_refs";
839
840/// What a pasted node's data names: the same text, the id it was given if it was pasted too.
841fn rename_refs(data: &mut serde_json::Value, renamed: &dyn Fn(&str) -> Option<String>) {
842    let pointers: Vec<String> = match data.get(NODE_REFS) {
843        Some(serde_json::Value::Array(list)) => {
844            list.iter().filter_map(|p| p.as_str().map(str::to_owned)).collect()
845        }
846        _ => return,
847    };
848    for pointer in pointers {
849        if let Some(serde_json::Value::String(id)) = data.pointer_mut(&pointer)
850            && let Some(new) = renamed(id)
851        {
852            *id = new;
853        }
854    }
855}

857/// Moves a diagram's JSON from version `from` to the next one. A change of the format that
858/// needs old JSON rewritten adds its step here, by the version it starts from.
859fn migrate(json: &mut serde_json::Value, from: u32) {
860    // 1 → 2: version 2 adds `data` to nodes and to the diagram, which version 1 has none of, so
861    // nothing is rewritten.
862    if let Some(object) = json.as_object_mut() {
863        object.insert("format".to_owned(), serde_json::Value::from(from + 1));
864    }
865}

1084    /// Puts `id` in the group `parent` (or in none). Refused, saying false, when `parent` is
1085    /// not a group or is `id` or inside it. The node and what it holds move after the group in
1086    /// the diagram's order, so the group is drawn behind them.
1087    pub fn set_parent(&mut self, id: &str, parent: Option<&str>) -> bool {
1088        if let Some(group) = parent {
1089            let ok = self.find_node(group).is_some_and(|g| g.group)
1090                && group != id
1091                && !self.descendants(id).iter().any(|d| d == group);
1092            if !ok {
1093                return false;
1094            }
1095        }
1096        let Some(node) = self.find_node_mut(id) else { return false };
1097        if node.parent.as_deref() == parent {
1098            return true;
1099        }
1100        node.parent = parent.map(str::to_owned);
1101        if parent.is_some() {
1102            let moving = self.with_descendants(&[id.to_owned()]);
1103            let (mut kept, taken): (Vec<FlowNode>, Vec<FlowNode>) =
1104                std::mem::take(&mut self.nodes).into_iter().partition(|n| !moving.contains(&n.id));
1105            kept.extend(taken);
1106            self.nodes = kept;
1107        }
1108        true
1109    }

1111    /// Puts these nodes in a new group `group`, sized to hold them with
1112    /// [`FlowNode::GROUP_PADDING`] around, and in the group they shared, if they all shared
1113    /// one. The group goes behind the first of them. Returns false, changing nothing, when
1114    /// there are no such nodes or the id is taken.
1115    pub fn group(&mut self, members: &[String], group: FlowNode) -> bool {
1116        if self.find_node(&group.id).is_some() {
1117            return false;
1118        }
1119        let inside: Vec<&FlowNode> = self.nodes.iter().filter(|n| members.contains(&n.id)).collect();
1120        let Some(first) = inside.first() else { return false };
1121        let bounds = inside.iter().skip(1).fold(first.rect(), |all, n| union(all, n.rect()));
1122        let shared = first.parent.clone().filter(|p| inside.iter().all(|n| n.parent.as_ref() == Some(p)));
1123        let (side, top) = FlowNode::GROUP_PADDING;
1124        let group = FlowNode {
1125            x: bounds.x() - side,
1126            y: bounds.y() - top,
1127            width: bounds.width() + side * 2.0,
1128            height: bounds.height() + top + side,
1129            parent: shared,
1130            group: true,
1131            ..group
1132        };
1133        let id = group.id.clone();
1134        let at = self.nodes.iter().position(|n| members.contains(&n.id)).unwrap_or(0);
1135        self.nodes.insert(at, group);
1136        for node in self.nodes.iter_mut().filter(|n| members.contains(&n.id)) {
1137            node.parent = Some(id.clone());
1138        }
1139        true
1140    }

1142    /// Brings these nodes, with what the groups among them hold, in front of every other node,
1143    /// keeping their order among themselves. A node in a group stays after its group, so the
1144    /// group is still drawn behind it. Says whether the order changed.
1145    pub fn bring_to_front(&mut self, ids: &[String]) -> bool {
1146        let moving = self.with_descendants(ids);
1147        let before: Vec<String> = self.nodes.iter().map(|n| n.id.clone()).collect();
1148        let (mut kept, taken): (Vec<FlowNode>, Vec<FlowNode>) =
1149            std::mem::take(&mut self.nodes).into_iter().partition(|n| !moving.contains(&n.id));
1150        kept.extend(taken);
1151        self.nodes = kept;
1152        self.nodes.iter().map(|n| &n.id).ne(before.iter())
1153    }

1155    /// Sends these nodes, with what the groups among them hold, behind every other node: to the
1156    /// very back, or, for a node in a group, to just after its group. Says whether the order
1157    /// changed.
1158    pub fn send_to_back(&mut self, ids: &[String]) -> bool {
1159        let before: Vec<String> = self.nodes.iter().map(|n| n.id.clone()).collect();
1160        let moving = self.with_descendants(ids);
1161        // Each node that does not go along with its group, last first, so that the ones sent
1162        // back keep their order among themselves.
1163        let heads: Vec<String> = self
1164            .nodes
1165            .iter()
1166            .filter(|n| moving.contains(&n.id) && n.parent.as_ref().is_none_or(|p| !moving.contains(p)))
1167            .map(|n| n.id.clone())
1168            .rev()
1169            .collect();
1170        for head in heads {
1171            let unit = self.with_descendants(std::slice::from_ref(&head));
1172            let (taken, kept): (Vec<FlowNode>, Vec<FlowNode>) =
1173                std::mem::take(&mut self.nodes).into_iter().partition(|n| unit.contains(&n.id));
1174            self.nodes = kept;
1175            let parent = taken.iter().find(|n| n.id == head).and_then(|n| n.parent.clone());
1176            let at = parent.and_then(|p| self.nodes.iter().position(|n| n.id == p)).map_or(0, |i| i + 1);
1177            self.nodes.splice(at..at, taken);
1178        }
1179        self.nodes.iter().map(|n| &n.id).ne(before.iter())
1180    }

1182    /// Takes out the group `id`, leaving what it held in the group it was in. Says whether it
1183    /// was a group.
1184    pub fn ungroup(&mut self, id: &str) -> bool {
1185        let Some(group) = self.find_node(id).filter(|g| g.group).cloned() else { return false };
1186        for node in self.nodes.iter_mut().filter(|n| n.parent.as_deref() == Some(id)) {
1187            node.parent = group.parent.clone();
1188        }
1189        self.nodes.retain(|n| n.id != id);
1190        self.edges.retain(|e| e.from.node != id && e.to.node != id);
1191        true
1192    }

1194    /// Adds a copy of `piece` moved by `(dx, dy)`, on top of the other nodes. Nodes and
1195    /// connections whose ids are taken get new ones, and the connections follow their nodes;
1196    /// a connection whose ends are not both in `piece` stays out. Returns the ids the copies
1197    /// got, nodes first.
1198    pub fn insert(&mut self, piece: &FlowGraph, dx: f32, dy: f32) -> (Vec<String>, Vec<String>) {
1199        self.insert_with(piece, dx, dy, |_, _| {})
1200    }

1202    /// As [`insert`](Self::insert), and then `remap` is called on each node put in, with a
1203    /// function from an id of `piece` to the id its copy was given: the app renames there what
1204    /// the node's data says of other nodes that [`NODE_REFS`] does not list.
1205    pub fn insert_with(
1206        &mut self,
1207        piece: &FlowGraph,
1208        dx: f32,
1209        dy: f32,
1210        mut remap: impl FnMut(&mut FlowNode, &dyn Fn(&str) -> Option<String>),
1211    ) -> (Vec<String>, Vec<String>) {
1212        let mut renamed: Vec<(&str, String)> = Vec::with_capacity(piece.nodes.len());
1213        let first = self.nodes.len();
1214        for node in &piece.nodes {
1215            let id = self.fresh_node_id(&node.id);
1216            self.nodes.push(FlowNode { id: id.clone(), x: node.x + dx, y: node.y + dy, ..node.clone() });
1217            renamed.push((&node.id, id));
1218        }
1219        let new_id = |old: &str| renamed.iter().find(|(o, _)| *o == old).map(|(_, id)| id.clone());
1220        // The copies sit in the copies of their groups, or in none, and name the copies of
1221        // the nodes their data names.
1222        for node in &mut self.nodes[first..] {
1223            node.parent = node.parent.as_deref().and_then(new_id);
1224            rename_refs(&mut node.data, &new_id);
1225            remap(node, &new_id);
1226        }
1227        let mut edges = Vec::new();
1228        for edge in &piece.edges {
1229            let (Some(from), Some(to)) = (new_id(&edge.from.node), new_id(&edge.to.node)) else { continue };
1230            let id =
1231                if self.find_edge(&edge.id).is_none() { edge.id.clone() } else { self.fresh_edge_id("e") };
1232            self.edges.push(FlowEdge {
1233                id: id.clone(),
1234                from: FlowEndpoint::new(from, edge.from.port.clone()),
1235                to: FlowEndpoint::new(to, edge.to.port.clone()),
1236                waypoints: edge.waypoints.iter().map(|(x, y)| (x + dx, y + dy)).collect(),
1237                ..edge.clone()
1238            });
1239            edges.push(id);
1240        }
1241        (renamed.into_iter().map(|(_, id)| id).collect(), edges)
1242    }

1256    /// Reads a diagram written by [`Self::to_json`] (or by hand), moving JSON of an earlier
1257    /// version to this one, and checks that it holds together: ids are not repeated and every
1258    /// connection joins ports that exist. JSON of a newer version is refused, so an older build
1259    /// never drops what it does not know; the app's [`data`](FlowNode::data) is kept as it came.
1260    pub fn from_json(text: &str) -> Result<Self, FlowJsonError> {
1261        let mut json: serde_json::Value =
1262            serde_json::from_str(text).map_err(|e| FlowJsonError::Syntax(e.to_string()))?;
1263        let mut format = json.get("format").and_then(serde_json::Value::as_u64).unwrap_or(1) as u32;
1264        if format > FLOW_FORMAT {
1265            return Err(FlowJsonError::Newer(format));
1266        }
1267        while format < FLOW_FORMAT {
1268            migrate(&mut json, format);
1269            format += 1;
1270        }
1271        if let Some(object) = json.as_object_mut() {
1272            object.remove("format");
1273        }
1274        let graph: Self = serde_json::from_value(json).map_err(|e| FlowJsonError::Syntax(e.to_string()))?;
1275        graph.check()?;
1276        Ok(graph)
1277    }

1279    /// The first thing that keeps the diagram from holding together, if any: a repeated node,
1280    /// port or connection id, a connection to a node or port that is not there, or a node in a
1281    /// group that is not there, is not a group or holds it.
1282    pub fn check(&self) -> Result<(), FlowJsonError> {
1283        for (i, node) in self.nodes.iter().enumerate() {
1284            if self.nodes[..i].iter().any(|n| n.id == node.id) {
1285                return Err(FlowJsonError::RepeatedNode(node.id.clone()));
1286            }
1287            if let Some(parent) = &node.parent {
1288                if !self.find_node(parent).is_some_and(|g| g.group) {
1289                    return Err(FlowJsonError::MissingGroup { node: node.id.clone(), group: parent.clone() });
1290                }
1291                if parent == &node.id || self.descendants(&node.id).contains(parent) {
1292                    return Err(FlowJsonError::GroupCycle(node.id.clone()));
1293                }
1294            }
1295            for (j, port) in node.ports.iter().enumerate() {
1296                if node.ports[..j].iter().any(|p| p.id == port.id) {
1297                    return Err(FlowJsonError::RepeatedPort { node: node.id.clone(), port: port.id.clone() });
1298                }
1299            }
1300        }
1301        for (i, edge) in self.edges.iter().enumerate() {
1302            if self.edges[..i].iter().any(|e| e.id == edge.id) {
1303                return Err(FlowJsonError::RepeatedEdge(edge.id.clone()));
1304            }
1305            for end in [&edge.from, &edge.to] {
1306                if self.port(end).is_none() {
1307                    return Err(FlowJsonError::MissingEnd { edge: edge.id.clone(), end: end.clone() });
1308                }
1309            }
1310        }
1311        Ok(())
1312    }
1313}

1339    },

1343        node: String,

1345        group: String,
1346    },

1380/// The diagrams before and after each change, for undo and redo. It keeps whole diagrams,
1381/// which is simple and right for diagrams of a few thousand nodes; the oldest go beyond
1382/// [`FlowHistory::LIMIT`].
1383#[derive(Clone, Debug, Default, PartialEq)]
1384#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
1385pub struct FlowHistory {
1386    past: Vec<FlowGraph>,
1387    future: Vec<FlowGraph>,
1388}
1389
1390#[cfg_attr(feature = "live", ravenrs_live::expose)]
1391impl FlowHistory {
1392    /// How many changes it remembers.
1393    pub const LIMIT: usize = 100;
1394
1395    /// Nothing to undo or redo.
1396    pub fn new() -> Self {
1397        Self::default()
1398    }
1399
1400    /// Remembers the diagram as it was before a change, and forgets what could be redone.
1401    pub fn record(&mut self, before: FlowGraph) {
1402        self.future.clear();
1403        if self.past.last() == Some(&before) {
1404            return;
1405        }
1406        self.past.push(before);
1407        if self.past.len() > Self::LIMIT {
1408            self.past.remove(0);
1409        }
1410    }

1413    /// brings back.

1415        let before = self.past.pop()?;
1416        self.future.push(now);
1417        Some(before)
1418    }
1419
1420    /// The diagram the last undo went back from, given the diagram as it is `now`.
1421    pub fn redo(&mut self, now: FlowGraph) -> Option<FlowGraph> {
1422        let after = self.future.pop()?;
1423        self.past.push(now);
1424        Some(after)
1425    }
1426
1427    /// Whether there is a change to undo.
1428    pub fn can_undo(&self) -> bool {
1429        !self.past.is_empty()
1430    }
1431
1432    /// Whether there is an undo to redo.
1433    pub fn can_redo(&self) -> bool {
1434        !self.future.is_empty()
1435    }
1436
1437    /// Forgets everything, as when another diagram is opened.
1438    pub fn clear(&mut self) {
1439        self.past.clear();
1440        self.future.clear();
1441    }
1442}

1450/// How far a line runs straight out of a port before it turns.
1451pub const PORT_RUN: f32 = 20.0;
1452/// The radius of the corners of a [`EdgeShape::Step`] line.
1453pub const STEP_RADIUS: f32 = 8.0;

1455/// The outline of a line from `from`, leaving a port on `from_side`, to `to`, arriving at a
1456/// port on `to_side`.
1457pub fn route(
1458    from: Point,
1459    from_side: PortSide,
1460    to: Point,
1461    to_side: PortSide,
1462    shape: EdgeShape,
1463) -> Vec<PathSegment> {
1464    match shape {
1465        EdgeShape::Straight => vec![PathSegment::Move(from), PathSegment::Line(to)],
1466        EdgeShape::Curve => {
1467            let reach = |side: PortSide| {
1468                let gap = if side.is_horizontal() { (to.x - from.x).abs() } else { (to.y - from.y).abs() };
1469                (gap * 0.5).max(PORT_RUN * 2.0)
1470            };
1471            let (a, b) = (from_side.outward(), to_side.outward());
1472            let (ra, rb) = (reach(from_side), reach(to_side));
1473            vec![
1474                PathSegment::Move(from),
1475                PathSegment::Cubic(
1476                    Point::new(from.x + a.x * ra, from.y + a.y * ra),
1477                    Point::new(to.x + b.x * rb, to.y + b.y * rb),
1478                    to,
1479                ),
1480            ]
1481        }
1482        EdgeShape::Step => rounded(&step_points(from, from_side, to, to_side), STEP_RADIUS),
1483    }
1484}

1486/// The outline of a line from `from` to `to` passing through `through`, in order. A straight
1487/// line goes from point to point; a curve passes smoothly through them, leaving and reaching
1488/// its ports square to their sides; a step line runs at right angles, turning at each point.
1489pub fn route_through(
1490    from: Point,
1491    from_side: PortSide,
1492    through: &[Point],
1493    to: Point,
1494    to_side: PortSide,
1495    shape: EdgeShape,
1496) -> Vec<PathSegment> {
1497    let mut points = Vec::with_capacity(through.len() + 2);
1498    points.push(from);
1499    points.extend_from_slice(through);
1500    points.push(to);
1501    match shape {
1502        EdgeShape::Straight => {
1503            let mut path = vec![PathSegment::Move(from)];
1504            path.extend(points[1..].iter().map(|&p| PathSegment::Line(p)));
1505            path
1506        }
1507        EdgeShape::Curve => {
1508            // A Catmull-Rom curve: the way through each point is along the line between its
1509            // neighbours; at the ends, straight out of the port.
1510            let n = points.len();
1511            let tangent = |i: usize| -> Point {
1512                if i == 0 {
1513                    let (d, out) = (distance(points[0], points[1]), from_side.outward());
1514                    Point::new(out.x * d, out.y * d)
1515                } else if i == n - 1 {
1516                    let (d, out) = (distance(points[n - 2], points[n - 1]), to_side.outward());
1517                    Point::new(-out.x * d, -out.y * d)
1518                } else {
1519                    Point::new(
1520                        (points[i + 1].x - points[i - 1].x) / 2.0,
1521                        (points[i + 1].y - points[i - 1].y) / 2.0,
1522                    )
1523                }
1524            };
1525            let mut path = vec![PathSegment::Move(from)];
1526            for i in 0..n - 1 {
1527                let (a, b) = (points[i], points[i + 1]);
1528                let (ta, tb) = (tangent(i), tangent(i + 1));
1529                path.push(PathSegment::Cubic(
1530                    Point::new(a.x + ta.x / 3.0, a.y + ta.y / 3.0),
1531                    Point::new(b.x - tb.x / 3.0, b.y - tb.y / 3.0),
1532                    b,
1533                ));
1534            }
1535            path
1536        }
1537        EdgeShape::Step => {
1538            let (a, b) = (from_side.outward(), to_side.outward());
1539            let out = Point::new(from.x + a.x * PORT_RUN, from.y + a.y * PORT_RUN);
1540            let into = Point::new(to.x + b.x * PORT_RUN, to.y + b.y * PORT_RUN);
1541            let mut corners = vec![from, out];
1542            // Leaving a side port goes on along it first; then each turn alternates.
1543            let mut level = from_side.is_horizontal();
1544            for &q in through.iter().chain(std::iter::once(&into)) {
1545                let Some(&c) = corners.last() else { break };
1546                if (c.x - q.x).abs() > 0.01 && (c.y - q.y).abs() > 0.01 {
1547                    corners.push(if level { Point::new(q.x, c.y) } else { Point::new(c.x, q.y) });
1548                    level = !level;
1549                }
1550                corners.push(q);
1551            }
1552            corners.push(to);
1553            rounded(&simplified(corners), STEP_RADIUS)
1554        }
1555    }
1556}

1558/// Where a bend point made at `at` goes among `controls` (a line's start, its points and its
1559/// end): just before the corner that ends the stretch passing nearest `at`.
1560pub fn bend_index(controls: &[Point], at: Point) -> usize {
1561    (1..controls.len())
1562        .min_by(|&i, &j| {
1563            let d = |k: usize| distance_to(&[controls[k - 1], controls[k]], at);
1564            d(i).total_cmp(&d(j))
1565        })
1566        .unwrap_or(1)
1567}

1569/// The corners without repeats and without those that do not turn.
1570fn simplified(mut points: Vec<Point>) -> Vec<Point> {
1571    points.dedup_by(|p, q| (p.x - q.x).abs() < 0.01 && (p.y - q.y).abs() < 0.01);
1572    let mut kept: Vec<Point> = Vec::with_capacity(points.len());
1573    for p in points {
1574        if kept.len() >= 2 {
1575            let (u, v) = (kept[kept.len() - 2], kept[kept.len() - 1]);
1576            let cross = (v.x - u.x) * (p.y - v.y) - (v.y - u.y) * (p.x - v.x);
1577            if cross.abs() < 0.01 {
1578                kept.pop();
1579            }
1580        }
1581        kept.push(p);
1582    }
1583    kept
1584}

1586/// The corners of a line at right angles between two ports.
1587pub(crate) fn step_points(from: Point, from_side: PortSide, to: Point, to_side: PortSide) -> Vec<Point> {
1588    let (a, b) = (from_side.outward(), to_side.outward());
1589    let out = Point::new(from.x + a.x * PORT_RUN, from.y + a.y * PORT_RUN);
1590    let into = Point::new(to.x + b.x * PORT_RUN, to.y + b.y * PORT_RUN);
1591    let mut points = vec![from, out];
1592    if from_side.is_horizontal() {
1593        let middle = (out.x + into.x) / 2.0;
1594        points.push(Point::new(middle, out.y));
1595        points.push(Point::new(middle, into.y));
1596    } else {
1597        let middle = (out.y + into.y) / 2.0;
1598        points.push(Point::new(out.x, middle));
1599        points.push(Point::new(into.x, middle));
1600    }
1601    points.extend([into, to]);
1602    simplified(points)
1603}

1605/// A polyline with each corner rounded by up to `radius`.
1606fn rounded(points: &[Point], radius: f32) -> Vec<PathSegment> {
1607    let Some(&first) = points.first() else { return Vec::new() };
1608    let mut path = vec![PathSegment::Move(first)];
1609    for i in 1..points.len() {
1610        let p = points[i];
1611        let Some(&next) = points.get(i + 1) else {
1612            path.push(PathSegment::Line(p));
1613            break;
1614        };
1615        let prev = points[i - 1];
1616        let (d1, d2) = (distance(prev, p), distance(p, next));
1617        let r = radius.min(d1 / 2.0).min(d2 / 2.0);
1618        if r < 0.5 {
1619            path.push(PathSegment::Line(p));
1620            continue;
1621        }
1622        let toward = |a: Point, b: Point, d: f32, by: f32| {
1623            Point::new(a.x + (b.x - a.x) / d * by, a.y + (b.y - a.y) / d * by)
1624        };
1625        let start = toward(p, prev, d1, r);
1626        let end = toward(p, next, d2, r);
1627        // A quarter circle as a cubic: control points 0.552 of the way to the corner.
1628        let k = 0.552_284_8;
1629        let c1 = Point::new(start.x + (p.x - start.x) * k, start.y + (p.y - start.y) * k);
1630        let c2 = Point::new(end.x + (p.x - end.x) * k, end.y + (p.y - end.y) * k);
1631        path.push(PathSegment::Line(start));
1632        path.push(PathSegment::Cubic(c1, c2, end));
1633    }
1634    path
1635}

1637fn distance(a: Point, b: Point) -> f32 {
1638    (b.x - a.x).hypot(b.y - a.y)
1639}

1641/// The points along an outline, with curves cut into short straight pieces.
1642pub fn flatten(path: &[PathSegment]) -> Vec<Point> {
1643    let mut points: Vec<Point> = Vec::new();
1644    let mut at = Point::new(0.0, 0.0);
1645    for segment in path {
1646        match *segment {
1647            PathSegment::Move(p) | PathSegment::Line(p) => {
1648                points.push(p);
1649                at = p;
1650            }
1651            PathSegment::Cubic(c1, c2, end) => {
1652                let length = distance(at, c1) + distance(c1, c2) + distance(c2, end);
1653                let steps = ((length / 6.0).ceil() as usize).clamp(2, 64);
1654                for step in 1..=steps {
1655                    let t = step as f32 / steps as f32;
1656                    let u = 1.0 - t;
1657                    let (a, b, c, d) = (u * u * u, 3.0 * u * u * t, 3.0 * u * t * t, t * t * t);
1658                    points.push(Point::new(
1659                        a * at.x + b * c1.x + c * c2.x + d * end.x,
1660                        a * at.y + b * c1.y + c * c2.y + d * end.y,
1661                    ));
1662                }
1663                at = end;
1664            }
1665            PathSegment::Close => {}
1666        }
1667    }
1668    points
1669}

1671/// How far `p` is from the nearest point of the polyline.
1672pub fn distance_to(points: &[Point], p: Point) -> f32 {
1673    if let [only] = points {
1674        return distance(*only, p);
1675    }
1676    points
1677        .windows(2)
1678        .map(|w| {
1679            let (a, b) = (w[0], w[1]);
1680            let (dx, dy) = (b.x - a.x, b.y - a.y);
1681            let length = dx * dx + dy * dy;
1682            let t = if length <= f32::EPSILON {
1683                0.0
1684            } else {
1685                (((p.x - a.x) * dx + (p.y - a.y) * dy) / length).clamp(0.0, 1.0)
1686            };
1687            distance(Point::new(a.x + dx * t, a.y + dy * t), p)
1688        })
1689        .fold(f32::INFINITY, f32::min)
1690}

1692/// The length of the polyline.
1693pub fn length(points: &[Point]) -> f32 {
1694    points.windows(2).map(|w| distance(w[0], w[1])).sum()
1695}

1697/// The point halfway along the polyline, where a label goes.
1698pub fn midpoint(points: &[Point]) -> Option<Point> {
1699    let half = length(points) / 2.0;
1700    let mut walked = 0.0;
1701    for w in points.windows(2) {
1702        let piece = distance(w[0], w[1]);
1703        if walked + piece >= half && piece > 0.0 {
1704            let t = (half - walked) / piece;
1705            return Some(Point::new(w[0].x + (w[1].x - w[0].x) * t, w[0].y + (w[1].y - w[0].y) * t));
1706        }
1707        walked += piece;
1708    }
1709    points.first().copied()
1710}

1712/// The polyline without its last `by` units of length.
1713pub fn trim_end(points: &[Point], by: f32) -> Vec<Point> {
1714    let mut kept = points.to_vec();
1715    let mut left = by;
1716    while left > 0.0 && kept.len() >= 2 {
1717        let (last, before) = (kept[kept.len() - 1], kept[kept.len() - 2]);
1718        let piece = distance(before, last);
1719        if piece > left {
1720            let t = (piece - left) / piece;
1721            let end = kept.len() - 1;
1722            kept[end] = Point::new(before.x + (last.x - before.x) * t, before.y + (last.y - before.y) * t);
1723            break;
1724        }
1725        left -= piece;
1726        kept.pop();
1727    }
1728    kept
1729}

1731/// The polyline without its first `by` units of length.
1732pub fn trim_start(points: &[Point], by: f32) -> Vec<Point> {
1733    let reversed: Vec<Point> = points.iter().rev().copied().collect();
1734    let mut kept = trim_end(&reversed, by);
1735    kept.reverse();
1736    kept
1737}

1739/// The direction the polyline is going at its end, as a unit vector.
1740pub fn end_direction(points: &[Point]) -> Option<Point> {
1741    let (&last, rest) = points.split_last()?;
1742    let before = rest.iter().rev().find(|p| distance(**p, last) > 0.5)?;
1743    let d = distance(*before, last);
1744    Some(Point::new((last.x - before.x) / d, (last.y - before.y) / d))
1745}

1747/// The pieces of the polyline drawn when it is dashed: `dash` long, `gap` apart, the pattern
1748/// shifted along by `phase`, so a phase that grows over time makes the dashes travel.
1749pub fn dashes(points: &[Point], dash: f32, gap: f32, phase: f32) -> Vec<Vec<Point>> {
1750    let period = dash + gap;
1751    if period <= 0.0 || points.len() < 2 {
1752        return Vec::new();
1753    }
1754    let mut pieces = Vec::new();
1755    let mut current: Vec<Point> = Vec::new();
1756    // Where in the pattern the start of the line falls.
1757    let mut position = (-phase).rem_euclid(period);
1758    for w in points.windows(2) {
1759        let (a, b) = (w[0], w[1]);
1760        let piece = distance(a, b);
1761        if piece <= 0.0 {
1762            continue;
1763        }
1764        let at = |t: f32| Point::new(a.x + (b.x - a.x) * t / piece, a.y + (b.y - a.y) * t / piece);
1765        let mut walked = 0.0;
1766        while walked < piece - 1e-4 {
1767            if position < dash {
1768                let step = (dash - position).min(piece - walked);
1769                if current.is_empty() {
1770                    current.push(at(walked));
1771                }
1772                walked += step;
1773                position += step;
1774                current.push(at(walked));
1775                if position >= dash - 1e-4 {
1776                    pieces.push(std::mem::take(&mut current));
1777                }
1778            } else {
1779                let step = (period - position).min(piece - walked);
1780                walked += step;
1781                position += step;
1782                if position >= period - 1e-4 {
1783                    position = 0.0;
1784                }
1785            }
1786        }
1787    }
1788    if current.len() >= 2 {
1789        pieces.push(current);
1790    }
1791    pieces
1792}

1794/// A small triangle with its tip at `tip`, pointing along `direction` (a unit vector).
1795pub fn arrow_head(tip: Point, direction: Point, size: f32) -> Vec<PathSegment> {
1796    let back = Point::new(tip.x - direction.x * size, tip.y - direction.y * size);
1797    let side = Point::new(-direction.y * size * 0.5, direction.x * size * 0.5);
1798    vec![
1799        PathSegment::Move(tip),
1800        PathSegment::Line(Point::new(back.x + side.x, back.y + side.y)),
1801        PathSegment::Line(Point::new(back.x - side.x, back.y - side.y)),
1802        PathSegment::Close,
1803    ]
1804}

1806/// The outline of a cap `size` long with its tip at `tip`, pointing along `direction` (a unit
1807/// vector): filled or drawn as a line as [`EdgeCap::filled`] says. Nothing for [`EdgeCap::None`].
1808pub fn cap_outline(cap: EdgeCap, tip: Point, direction: Point, size: f32) -> Vec<PathSegment> {
1809    let along = |by: f32| Point::new(tip.x - direction.x * by, tip.y - direction.y * by);
1810    let across = |p: Point, by: f32| Point::new(p.x - direction.y * by, p.y + direction.x * by);
1811    match cap {
1812        EdgeCap::Arrow => arrow_head(tip, direction, size),
1813        EdgeCap::OpenArrow => {
1814            let back = along(size);
1815            vec![
1816                PathSegment::Move(across(back, size * 0.55)),
1817                PathSegment::Line(tip),
1818                PathSegment::Line(across(back, -size * 0.55)),
1819            ]
1820        }
1821        EdgeCap::Diamond => {
1822            let middle = along(size * 0.75);
1823            vec![
1824                PathSegment::Move(tip),
1825                PathSegment::Line(across(middle, size * 0.5)),
1826                PathSegment::Line(along(size * 1.5)),
1827                PathSegment::Line(across(middle, -size * 0.5)),
1828                PathSegment::Close,
1829            ]
1830        }
1831        EdgeCap::Circle => {
1832            // Four quarter circles as cubics, around a centre half the size back from the tip.
1833            let r = size / 2.0;
1834            let c = along(r);
1835            let k = 0.552_284_8 * r;
1836            let at = |dx: f32, dy: f32| Point::new(c.x + dx, c.y + dy);
1837            vec![
1838                PathSegment::Move(at(r, 0.0)),
1839                PathSegment::Cubic(at(r, k), at(k, r), at(0.0, r)),
1840                PathSegment::Cubic(at(-k, r), at(-r, k), at(-r, 0.0)),
1841                PathSegment::Cubic(at(-r, -k), at(-k, -r), at(0.0, -r)),
1842                PathSegment::Cubic(at(k, -r), at(r, -k), at(r, 0.0)),
1843                PathSegment::Close,
1844            ]
1845        }
1846        EdgeCap::None => Vec::new(),
1847    }
1848}

1850/// `value` on the nearest line of a grid `step` apart (none when `step` is not positive).
1851pub fn snap(value: f32, step: f32) -> f32 {
1852    if step > 0.0 { (value / step).round() * step } else { value }
