9/// The kind of a chart.
10#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
11#[serde(rename_all = "camelCase")]
12#[non_exhaustive]
13pub enum ChartKind {
14    /// Vertical bars, side by side per series.
15    #[default]
16    Bar,
17    /// Lines with a dot at each point.
18    Line,
19    /// Lines with the area under them filled.
20    Area,
21    /// Slices of a circle; uses the first series.
22    Pie,
23    /// Slices of a ring; uses the first series.
24    Doughnut,
25    /// Dots at numeric positions.
26    Scatter,
27    /// A small line without axes or text, to sit in a row of a table.
28    Sparkline,
29    /// A smoothed line over a filled area, without axes or text.
30    SparkArea,
31}

152/// Roughly how wide a character is, as a share of the font size; enough to
153/// decide how many labels fit.
154const CHAR_WIDTH: f32 = 0.56;
155/// A slice smaller than this share of the circle has no percentage on it.
156const SLICE_LABEL_MIN: f64 = 0.05;
157const TICKS: usize = 5;
158/// The radius of a dot in a scatter chart.
159const SCATTER_DOT: f32 = 2.5;
160const GAP: f32 = 6.0;

162/// Round numbers for an axis: lower bound, step and how many ticks.
163pub fn nice_ticks(min: f64, max: f64, wanted: usize) -> (f64, f64, usize) {
164    let (min, max) = if (max - min).abs() < f64::EPSILON {
165        if min == 0.0 { (0.0, 1.0) } else { (min.min(0.0), max.max(0.0)) }
166    } else {
167        (min, max)
168    };
169    let rough = (max - min) / wanted.max(1) as f64;
170    let magnitude = 10f64.powf(rough.log10().floor());
171    let residual = rough / magnitude;
172    let step = magnitude
173        * if residual <= 1.0 {
174            1.0
175        } else if residual <= 2.0 {
176            2.0
177        } else if residual <= 2.5 {
178            2.5
179        } else if residual <= 5.0 {
180            5.0
181        } else {
182            10.0
183        };
184    let low = (min / step).floor() * step;
185    let high = (max / step).ceil() * step;
186    (low, step, ((high - low) / step).round() as usize + 1)
187}

206/// Appends an arc around `center`, from one angle to another (radians, clockwise on the page).
207fn arc(path: &mut Vec<PathSegment>, center: Point, radius: f32, from: f32, to: f32) {
208    let segments = ((to - from).abs() / std::f32::consts::FRAC_PI_2).ceil().max(1.0) as usize;
209    let step = (to - from) / segments as f32;
210    let handle = 4.0 / 3.0 * (step / 4.0).tan() * radius;
211    let at = |angle: f32| Point::new(center.x + radius * angle.cos(), center.y + radius * angle.sin());
212    for index in 0..segments {
213        let a = from + step * index as f32;
214        let b = a + step;
215        let (start, end) = (at(a), at(b));
216        path.push(PathSegment::Cubic(
217            Point::new(start.x - handle * a.sin(), start.y + handle * a.cos()),
218            Point::new(end.x + handle * b.sin(), end.y - handle * b.cos()),
219            end,
220        ));
221    }
222}

229/// Lays a chart out inside `rect`, in this style.
230pub fn draw_styled(
231    data: &ChartData,
232    rect: Rect,
233    options: &ChartOptions<'_>,
234    style: &ChartStyle<'_>,
235) -> ChartDrawing {
236    let mut drawing = ChartDrawing::default();
237    if matches!(options.kind, ChartKind::Sparkline | ChartKind::SparkArea) {
238        sparkline(data, rect, options, style, &mut drawing);
239        return drawing;
240    }
241    let mut area = rect;
242    if !options.title.trim().is_empty() {
243        let height = options.size * 2.0;
244        let mut title = label(
245            options.title,
246            Rect::new(area.x(), area.y(), area.width(), height),
247            LabelAlign::Left,
248            options.size * 1.25,
249            options.ink,
250        );
251        title.bold = true;
252        drawing.labels.push(title);
253        area = Rect::new(area.x(), area.y() + height, area.width(), (area.height() - height).max(0.0));
254    }
255    let round = matches!(options.kind, ChartKind::Pie | ChartKind::Doughnut);
256    let entries: Vec<(String, Color)> = if round {
257        // A category with nothing to show has no slice, so it has no place in the legend.
258        let values = data.series.first().map(|s| s.values.as_slice());
259        data.categories
260            .iter()
261            .enumerate()
262            .filter(|(i, _)| values.is_none_or(|v| v.get(*i).is_some_and(|value| *value > 0.0)))
263            .map(|(i, name)| (name.clone(), slice_color(style, i)))
264            .collect()
265    } else if data.series.len() > 1 {
266        data.series.iter().map(|s| (s.name.clone(), s.color)).collect()
267    } else {
268        Vec::new()
269    };
270    if options.legend && !entries.is_empty() {
271        area = legend(&entries, area, options, &mut drawing);
272    }
273    if round {
274        circle(data, area, options, style, &mut drawing);
275    } else {
276        cartesian(data, area, options, style, &mut drawing);
277    }
278    drawing
279}

281fn slice_color(style: &ChartStyle<'_>, index: usize) -> Color {
282    let palette = if style.palette.is_empty() { &PALETTE[..] } else { style.palette };
283    palette[index % palette.len()]

286/// Draws the legend at the bottom of `area`; returns what is left above it.
287fn legend(
288    entries: &[(String, Color)],
289    area: Rect,
290    options: &ChartOptions<'_>,
291    drawing: &mut ChartDrawing,
292) -> Rect {
293    let size = options.size * 0.85;
294    let line = size * 1.6;
295    let swatch = size * 0.75;
296    let widths: Vec<f32> = entries
297        .iter()
298        .map(|(name, _)| swatch + 4.0 + name.chars().count() as f32 * size * CHAR_WIDTH + 10.0)
299        .collect();
300    // How many lines the entries wrap to.
301    let (mut lines, mut used) = (1, 0.0);
302    for width in &widths {
303        if used + width > area.width() && used > 0.0 {
304            lines += 1;
305            used = 0.0;
306        }
307        used += width;
308    }
309    // A legend can take up to half the room; past that, the last entries are left out.
310    let room = ((area.height() / 2.0 / line).floor() as usize).max(1);
311    let lines = lines.min(room);
312    let top = area.max_y() - line * lines as f32;
313    let (mut x, mut row) = (area.x(), 0);
314    for ((name, color), width) in entries.iter().zip(&widths) {
315        if x + width > area.max_x() && x > area.x() {
316            x = area.x();
317            row += 1;
318            if row >= lines {
319                break;
320            }
321        }
322        let y = top + row as f32 * line;
323        drawing.shapes.push(rectangle(Rect::new(x, y + (line - swatch) / 2.0, swatch, swatch), *color));
324        drawing.labels.push(label(
325            name.clone(),
326            Rect::new(x + swatch + 4.0, y, (width - swatch - 4.0).max(1.0), line),
327            LabelAlign::Left,
328            size,
329            options.muted,
330        ));
331        x += width;
332    }
333    Rect::new(area.x(), area.y(), area.width(), (top - area.y() - GAP).max(0.0))
334}

336fn circle(
337    data: &ChartData,
338    area: Rect,
339    options: &ChartOptions<'_>,
340    style: &ChartStyle<'_>,
341    drawing: &mut ChartDrawing,
342) {
343    let Some(series) = data.series.first() else { return };
344    let total: f64 = series.values.iter().filter(|v| **v > 0.0).sum();
345    if total <= 0.0 || area.width() <= 0.0 || area.height() <= 0.0 {
346        return;
347    }
348    let center = area.center();
349    let radius = area.width().min(area.height()) / 2.0 - 1.0;
350    let hole = if options.kind == ChartKind::Doughnut { radius * 0.58 } else { 0.0 };
351    let mut angle = -std::f32::consts::FRAC_PI_2;
352    for (index, value) in series.values.iter().enumerate() {
353        if *value <= 0.0 {
354            continue;
355        }
356        let share = *value / total;
357        let end = angle + share as f32 * std::f32::consts::TAU;
358        let at = |r: f32, a: f32| Point::new(center.x + r * a.cos(), center.y + r * a.sin());
359        let mut path = vec![PathSegment::Move(at(radius, angle))];
360        arc(&mut path, center, radius, angle, end);
361        if hole > 0.0 {
362            path.push(PathSegment::Line(at(hole, end)));
363            arc(&mut path, center, hole, end, angle);
364        } else {
365            path.push(PathSegment::Line(center));
366        }
367        path.push(PathSegment::Close);
368        drawing.shapes.push(Shape {
369            path,
370            fill: Some(slice_color(style, index)),
371            stroke: Some((1.0, style.separator)),
372        });
373        if share >= SLICE_LABEL_MIN {
374            let middle = (angle + end) / 2.0;
375            let position = at(if hole > 0.0 { (radius + hole) / 2.0 } else { radius * 0.66 }, middle);
376            let size = options.size * 0.85;
377            drawing.labels.push(label(
378                format!("{:.0}%", share * 100.0),
379                Rect::new(position.x - 30.0, position.y - size, 60.0, size * 2.0),
380                LabelAlign::Center,
381                size,
382                Color::WHITE,
383            ));
384        }
385        angle = end;
386    }
387}

389fn cartesian(
390    data: &ChartData,
391    area: Rect,
392    options: &ChartOptions<'_>,
393    style: &ChartStyle<'_>,
394    drawing: &mut ChartDrawing,
395) {
396    let values = data.series.iter().flat_map(|s| s.values.iter().copied()).filter(|v| v.is_finite());
397    let (mut min, mut max) =
398        values.fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), v| (a.min(v), b.max(v)));
399    if !min.is_finite() || data.categories.is_empty() {
400        return;
401    }
402    // Bars and areas start from zero, or their sizes would mislead.
403    if matches!(options.kind, ChartKind::Bar | ChartKind::Area) {
404        min = min.min(0.0);
405        max = max.max(0.0);
406    }
407    let (low, step, ticks) = nice_ticks(min, max, TICKS);
408    let high = low + step * (ticks.saturating_sub(1)) as f64;
409    let span = (high - low).max(f64::EPSILON);
410    let size = options.size * 0.85;
411    let axis_labels: Vec<String> = (0..ticks).map(|i| (options.format)(low + step * i as f64)).collect();
412    let axis_width =
413        axis_labels.iter().map(|t| t.chars().count()).max().unwrap_or(0) as f32 * size * CHAR_WIDTH + GAP;
414    let bottom_room = size * 1.8;
415    let plot = Rect::new(
416        area.x() + axis_width,
417        area.y() + size / 2.0,
418        (area.width() - axis_width).max(1.0),
419        (area.height() - bottom_room - size / 2.0).max(1.0),
420    );
421    let y_of = |value: f64| plot.max_y() - ((value - low) / span) as f32 * plot.height();
422    for (index, text) in axis_labels.iter().enumerate() {
423        let y = y_of(low + step * index as f64);
424        drawing.shapes.push(rectangle(Rect::new(plot.x(), y - 0.25, plot.width(), 0.5), options.grid));
425        drawing.labels.push(label(
426            text.clone(),
427            Rect::new(area.x(), y - size, axis_width - GAP / 2.0, size * 2.0),
428            LabelAlign::Right,
429            size,
430            options.muted,
431        ));
432    }
433    let count = data.categories.len();
434    let band = plot.width() / count as f32;
435    // Only as many category labels as fit, evenly spaced.
436    let longest =
437        data.categories.iter().map(|c| c.chars().count()).max().unwrap_or(1) as f32 * size * CHAR_WIDTH + 4.0;
438    let every = (longest / band.max(1.0)).ceil().max(1.0) as usize;
439    if options.kind != ChartKind::Scatter {
440        for (index, name) in data.categories.iter().enumerate().step_by(every) {
441            let width = band * every as f32;
442            let center = plot.x() + band * (index as f32 + 0.5);
443            drawing.labels.push(label(
444                name.clone(),
445                Rect::new(center - width / 2.0, plot.max_y() + 2.0, width, bottom_room - 2.0),
446                LabelAlign::Center,
447                size,
448                options.muted,
449            ));
450        }
451    }
452    match options.kind {
453        ChartKind::Bar => {
454            let series_count = data.series.len().max(1) as f32;
455            let bar = (band * 0.72 / series_count).min(style.max_bar_width);
456            let group = bar * series_count;
457            let zero = y_of(0.0_f64.clamp(low, high));
458            for (s, series) in data.series.iter().enumerate() {
459                for (index, value) in series.values.iter().enumerate() {
460                    if !value.is_finite() {
461                        continue;
462                    }
463                    let x = plot.x() + band * index as f32 + (band - group) / 2.0 + bar * s as f32;
464                    let y = y_of(*value);
465                    let (top, height) = if y < zero { (y, zero - y) } else { (zero, y - zero) };
466                    let rect = Rect::new(x, top, (bar - 0.5).max(0.5), height.max(0.25));
467                    drawing
468                        .shapes
469                        .push(Shape::filled(vector::rectangle(rect, style.bar_radius), series.color));
470                }
471            }
472        }
473        ChartKind::Scatter => {
474            let (x_low, x_step, x_ticks) = {
475                let (a, b) = data
476                    .positions
477                    .iter()
478                    .fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), v| (a.min(*v), b.max(*v)));
479                nice_ticks(if a.is_finite() { a } else { 0.0 }, if b.is_finite() { b } else { 1.0 }, TICKS)
480            };
481            let x_span = (x_step * x_ticks.saturating_sub(1) as f64).max(f64::EPSILON);
482            // Inset by a dot's radius, so a dot at either end stays inside the plot.
483            let x_of = |value: f64| {
484                plot.x()
485                    + SCATTER_DOT
486                    + ((value - x_low) / x_span) as f32 * (plot.width() - 2.0 * SCATTER_DOT)
487            };
488            for index in 0..x_ticks {
489                let value = x_low + x_step * index as f64;
490                drawing.labels.push(label(
491                    (options.format)(value),
492                    Rect::new(x_of(value) - 40.0, plot.max_y() + 2.0, 80.0, bottom_room - 2.0),
493                    LabelAlign::Center,
494                    size,
495                    options.muted,
496                ));
497            }
498            for series in &data.series {
499                for (position, value) in data.positions.iter().zip(&series.values) {
500                    if value.is_finite() {
501                        drawing.shapes.push(dot(
502                            Point::new(x_of(*position), y_of(*value)),
503                            SCATTER_DOT,
504                            series.color,
505                        ));
506                    }
507                }
508            }
509        }
510        _ => {
511            let filled = options.kind == ChartKind::Area;
512            for series in &data.series {
513                let points: Vec<Point> = series
514                    .values
515                    .iter()
516                    .enumerate()
517                    .filter(|(_, v)| v.is_finite())
518                    .map(|(i, v)| Point::new(plot.x() + band * (i as f32 + 0.5), y_of(*v)))
519                    .collect();
520                if filled && let (Some(first), Some(last)) = (points.first(), points.last()) {
521                    let zero = y_of(0.0_f64.clamp(low, high));
522                    let mut path = vec![PathSegment::Move(Point::new(first.x, zero))];
523                    path.extend(points.iter().map(|p| PathSegment::Line(*p)));
524                    path.push(PathSegment::Line(Point::new(last.x, zero)));
525                    path.push(PathSegment::Close);
526                    drawing.shapes.push(Shape {
527                        path,
528                        fill: Some(with_alpha(series.color, 0.25)),
529                        stroke: None,
530                    });
531                }
532                drawing.shapes.push(polyline(&points, 1.75, series.color));
533                if band > 8.0 {
534                    drawing.shapes.extend(points.iter().map(|p| dot(*p, 2.25, series.color)));
535                }
536            }
537        }
538    }
539}

541fn polyline(points: &[Point], width: f32, color: Color) -> Shape {
542    let path = points
543        .iter()
544        .enumerate()
545        .map(|(i, p)| if i == 0 { PathSegment::Move(*p) } else { PathSegment::Line(*p) })
546        .collect();
547    Shape { path, fill: None, stroke: Some((width, color)) }
548}
549
550/// A curve through evenly spaced points (Catmull-Rom), without its starting move.
551fn smooth(points: &[Point]) -> Vec<PathSegment> {
552    let last = points.len().saturating_sub(1);
553    (0..last)
554        .map(|i| {
555            let (p0, p1) = (points[i.saturating_sub(1)], points[i]);
556            let (p2, p3) = (points[i + 1], points[(i + 2).min(last)]);
557            PathSegment::Cubic(
558                Point::new(p1.x + (p2.x - p0.x) / 6.0, p1.y + (p2.y - p0.y) / 6.0),
559                Point::new(p2.x - (p3.x - p1.x) / 6.0, p2.y - (p3.y - p1.y) / 6.0),
560                p2,
561            )
562        })
563        .collect()
564}

566fn sparkline(
567    data: &ChartData,
568    rect: Rect,
569    options: &ChartOptions<'_>,
570    style: &ChartStyle<'_>,
571    drawing: &mut ChartDrawing,
572) {
573    let Some(series) = data.series.first() else { return };
574    let values: Vec<f64> = series.values.iter().copied().filter(|v| v.is_finite()).collect();
575    if values.is_empty() {
576        return;
577    }
578    let (min, max) =
579        values.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), v| (a.min(*v), b.max(*v)));
580    let span = (max - min).max(f64::EPSILON);
581    let filled = options.kind == ChartKind::SparkArea;
582    // Room for the last dot.
583    let pad = if filled { 6.0 } else { 2.5 };
584    let (width, height) = ((rect.width() - 2.0 * pad).max(1.0), (rect.height() - 2.0 * pad).max(1.0));
585    let last = values.len().saturating_sub(1).max(1) as f32;
586    let points: Vec<Point> = values
587        .iter()
588        .enumerate()
589        .map(|(i, v)| {
590            Point::new(
591                rect.x() + pad + width * i as f32 / last,
592                rect.y() + pad + height * (1.0 - ((*v - min) / span) as f32),
593            )
594        })
595        .collect();
596    let (Some(first), Some(end)) = (points.first().copied(), points.last().copied()) else { return };
597    if filled {
598        let curve = smooth(&points);
599        let mut area = vec![PathSegment::Move(Point::new(first.x, rect.max_y())), PathSegment::Line(first)];
600        area.extend(curve.iter().copied());
601        area.extend([PathSegment::Line(Point::new(end.x, rect.max_y())), PathSegment::Close]);
602        drawing.shapes.push(Shape::filled(area, with_alpha(series.color, 0.2)));
603        let mut line = vec![PathSegment::Move(first)];
604        line.extend(curve);
605        drawing.shapes.push(Shape { path: line, fill: None, stroke: Some((2.0, series.color)) });
606        let mut last = dot(end, 4.0, series.color);
607        last.stroke = Some((2.0, style.separator));
608        drawing.shapes.push(last);
609    } else {
610        drawing.shapes.push(polyline(&points, 1.25, series.color));
611        drawing.shapes.push(dot(end, 2.0, series.color));
612    }
613}
