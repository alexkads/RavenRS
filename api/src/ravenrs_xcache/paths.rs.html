1//! The directories that differ between worktrees, and how they are taken out of a compilation.
2//!
3//! rustc writes the paths it sees into what it produces, and hashes them into the crate's identity,
4//! so the same crate built under two directories is two different artifacts that do not even mix
5//! (error E0460). With `--remap-path-prefix` for the build directory and the working directory the
6//! artifacts come out byte for byte the same. Whatever still carries the real path afterwards is
7//! data the crate embedded, such as `env!("OUT_DIR")` in a constant, and cannot be shared.
8
9use std::path::Path;
10
11/// One directory that changes from worktree to worktree.
12#[derive(Debug, Clone)]
13pub(crate) struct Root {
14    /// The path on this machine.
15    pub real: String,
16    /// What stands for it in a cache key.
17    pub marker: &'static str,
18    /// What rustc is told to write in its place.
19    pub remapped: &'static str,
20}
21
22/// The roots of one compilation, longest first.
23#[derive(Debug, Clone, Default)]
24pub(crate) struct Roots(pub Vec<Root>);
25
26impl Roots {
27    /// The build directory `out_dir` is in, and the working directory unless it is inside Cargo's
28    /// home, where registry crates live at the same path for every worktree.
29    ///
30    /// Cargo's layout gives the build directory: `<build dir>/[<triple>/]<profile>/deps`. The
31    /// `CACHEDIR.TAG` Cargo writes is not a sign to go by: Cargo only writes it in a directory it
32    /// creates, and a script that made the directory first leaves it without one.
33    pub(crate) fn find(out_dir: &Path, triple: Option<&str>, cwd: &Path, cargo_home: Option<&Path>) -> Roots {
34        let mut roots = Vec::new();
35        let in_profile =
36            matches!(out_dir.file_name().and_then(|name| name.to_str()), Some("deps" | "examples"));
37        let profile = out_dir.parent().filter(|_| in_profile);
38        let above = profile.and_then(Path::parent);
39        let target = match (above, triple) {
40            (Some(above), Some(triple)) if above.file_name().is_some_and(|name| name == triple) => {
41                above.parent().unwrap_or(above)
42            }
43            (Some(above), _) => above,
44            (None, _) => out_dir,
45        };
46        if let Some(real) = target.to_str() {
47            roots.push(Root { real: real.to_owned(), marker: "<xcache:target>", remapped: "/xcache/target" });
48        }
49        let in_cargo_home = cargo_home.is_some_and(|home| cwd.starts_with(home));
50        if let (false, Some(real)) = (in_cargo_home, cwd.to_str())
51            && roots.iter().all(|root| root.real != real)
52        {
53            roots.push(Root { real: real.to_owned(), marker: "<xcache:cwd>", remapped: "/xcache/cwd" });
54        }
55        roots.sort_by_key(|root| std::cmp::Reverse(root.real.len()));
56        Roots(roots)
57    }
58
59    /// `text` with each root replaced by its marker, where the root is a whole path prefix.
60    pub(crate) fn norm(&self, text: &str) -> String {
61        let mut text = text.to_owned();
62        for root in &self.0 {
63            text = replace_prefix(&text, &root.real, root.marker);
64        }
65        text
66    }
67
68    /// The reverse of [`Roots::norm`], for the roots of this compilation.
69    pub(crate) fn denorm(&self, text: &str) -> String {
70        let mut text = text.to_owned();
71        for root in &self.0 {
72            text = text.replace(root.marker, &root.real);
73        }
74        text
75    }
76
77    /// The `--remap-path-prefix` arguments, or why they cannot be given. rustc lets the last match
78    /// win, so the longest root goes last.
79    pub(crate) fn remap_args(&self) -> Result<Vec<String>, String> {
80        let mut args = Vec::new();
81        for root in self.0.iter().rev() {
82            if root.real.contains('=') {
83                return Err(format!("the path {} has an = in it", root.real));
84            }
85            args.push(format!("--remap-path-prefix={}={}", root.real, root.remapped));
86        }
87        Ok(args)
88    }
89
90    /// The real paths, as one line: what a result that embeds them is tied to.
91    pub(crate) fn raw(&self) -> String {
92        self.0.iter().map(|root| root.real.as_str()).collect::<Vec<_>>().join("|")
93    }
94
95    /// The first root whose real path is in `bytes`.
96    pub(crate) fn leaked_in(&self, bytes: &[u8]) -> Option<&str> {
97        self.0.iter().find(|root| contains(bytes, root.real.as_bytes())).map(|root| root.real.as_str())
98    }
99}
100
101/// Replaces `prefix` by `with` where what follows cannot be more of the same path component, so
102/// `/work/app` does not match inside `/work/app2`.
103fn replace_prefix(text: &str, prefix: &str, with: &str) -> String {
104    if prefix.is_empty() {
105        return text.to_owned();
106    }
107    let mut out = String::with_capacity(text.len());
108    let mut rest = text;
109    while let Some(at) = rest.find(prefix) {
110        let after = &rest[at + prefix.len()..];
111        let whole =
112            after.chars().next().is_none_or(|c| !(c.is_alphanumeric() || matches!(c, '-' | '_' | '.')));
113        out.push_str(&rest[..at]);
114        out.push_str(if whole { with } else { prefix });
115        rest = after;
116    }
117    out.push_str(rest);
118    out
119}

121/// Whether `needle` is in `haystack` (Boyer-Moore-Horspool: artifacts are large, paths are long).
122pub(crate) fn contains(haystack: &[u8], needle: &[u8]) -> bool {
123    let size = needle.len();
124    if size == 0 || haystack.len() < size {
125        return size == 0;
126    }
127    let mut skip = [size; 256];
128    for (index, byte) in needle[..size - 1].iter().enumerate() {
129        skip[usize::from(*byte)] = size - 1 - index;
130    }
131    let mut at = 0;
132    while at + size <= haystack.len() {
133        if &haystack[at..at + size] == needle {
134            return true;
135        }
136        at += skip[usize::from(haystack[at + size - 1])];
137    }
138    false
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144
145    fn roots() -> Roots {
146        Roots(vec![
147            Root { real: "/work/app/target".into(), marker: "<xcache:target>", remapped: "/xcache/target" },
148            Root { real: "/work/app".into(), marker: "<xcache:cwd>", remapped: "/xcache/cwd" },
149        ])
150    }
151
152    #[test]
153    fn roots_become_markers_and_come_back() {
154        let roots = roots();
155        let text = "OUT_DIR=/work/app/target/debug/build/x/out src=/work/app/crates/x/src/lib.rs";
156        let normal = roots.norm(text);
157        assert_eq!(normal, "OUT_DIR=<xcache:target>/debug/build/x/out src=<xcache:cwd>/crates/x/src/lib.rs");
158        assert_eq!(roots.denorm(&normal), text);
159    }
160
161    #[test]
162    fn a_longer_folder_name_is_another_folder() {
163        let roots = roots();
164        assert_eq!(roots.norm("/work/app2/src"), "/work/app2/src");
165        assert_eq!(roots.norm("/work/app"), "<xcache:cwd>");
166        assert_eq!(roots.norm("\"/work/app\""), "\"<xcache:cwd>\"");
167    }
168
169    #[test]
170    fn the_longest_root_is_remapped_last_so_it_wins() {
171        let args = roots().remap_args().unwrap_or_default();
172        assert_eq!(
173            args,
174            [
175                "--remap-path-prefix=/work/app=/xcache/cwd",
176                "--remap-path-prefix=/work/app/target=/xcache/target",
177            ]
178        );
179        let odd = Roots(vec![Root { real: "/a=b".into(), marker: "<xcache:cwd>", remapped: "/xcache/cwd" }]);
180        assert!(odd.remap_args().is_err());
181    }
182
183    #[test]
184    fn a_path_left_in_the_bytes_is_found() {
185        let roots = roots();
186        assert_eq!(roots.leaked_in(b"\0\0/work/app/target/debug/build/x/out\0"), Some("/work/app/target"));
187        assert_eq!(roots.leaked_in(b"\0/xcache/target/debug\0/xcache/cwd/src\0"), None);
188        assert!(contains(b"abcabcabd", b"abcabd"));
189        assert!(!contains(b"abcabcabc", b"abcabd"));
190        assert!(!contains(b"ab", b"abc"));
191    }
192
193    #[test]
194    fn registry_crates_keep_their_working_directory() {
195        let folder = std::env::temp_dir().join(format!("xcache-roots-{}", std::process::id()));
196        let out = folder.join("target/debug/deps");
197        std::fs::create_dir_all(&out).unwrap_or_default();
198        let home = folder.join("cargo-home");
199        let member = Roots::find(&out, None, &folder.join("ws"), Some(&home));
200        assert_eq!(member.0.len(), 2);
201        let registry = Roots::find(&out, None, &home.join("registry/src/x/serde-1.0.0"), Some(&home));
202        assert_eq!(registry.0.len(), 1);
203        assert_eq!(registry.0[0].real, folder.join("target").to_string_lossy());
204        std::fs::remove_dir_all(&folder).unwrap_or_default();
205    }
206
207    #[test]
208    fn the_build_directory_is_found_by_its_layout_with_or_without_a_triple() {
209        let cwd = Path::new("/work/app");
210        let root_of = |out: &str, triple: Option<&str>| {
211            let roots = Roots::find(Path::new(out), triple, cwd, None);
212            roots
213                .0
214                .iter()
215                .find(|root| root.marker == "<xcache:target>")
216                .map(|root| root.real.clone())
217                .unwrap_or_default()
218        };
219        assert_eq!(root_of("/b/t/debug/deps", None), "/b/t");
220        assert_eq!(
221            root_of("/b/t/wasm32-unknown-unknown/wasm-release/deps", Some("wasm32-unknown-unknown")),
222            "/b/t"
223        );
224        // A profile directory is not mistaken for a triple.
225        assert_eq!(root_of("/b/t/debug/deps", Some("aarch64-apple-darwin")), "/b/t");
226        assert_eq!(root_of("/b/t/debug/examples", None), "/b/t");
227        // Not Cargo's layout: only the output directory itself is taken out.
228        assert_eq!(root_of("/b/out", None), "/b/out");
229    }
