1//! Where compiled crates are kept: files named by the SHA-256 of their content, results that list
2//! them, manifests that say which inputs lead to which result, and a log of every request.
3//!
4//! Every write goes to a temporary file first and is renamed into place, so a build that is stopped
5//! halfway leaves nothing a later one could read as whole. Two builds writing the same thing write
6//! the same bytes, and the last rename wins.
7
8use std::fs;
9use std::io::{self, Read, Write};
10use std::path::{Path, PathBuf};
11use std::time::{SystemTime, UNIX_EPOCH};
12
13use sha2::{Digest, Sha256};

15/// The cache on disk.
16#[derive(Debug, Clone)]
17pub(crate) struct Store {
18    pub root: PathBuf,
19}

21/// One file a compilation produced.
22#[derive(Debug, Clone, PartialEq, Eq)]
23pub(crate) struct Output {
24    /// `dep-info`, `metadata` or `link`.
25    pub emit: String,
26    pub sha: String,
27    pub size: u64,
28    /// Its name inside the output directory.
29    pub name: String,
30}
31
32/// What was kept of one compilation.
33#[derive(Debug, Clone, Default, PartialEq, Eq)]
34pub(crate) struct Stored {
35    pub outputs: Vec<Output>,
36    /// What rustc wrote to stderr, in order: `D <line>` for a diagnostic as it was, `A <emit> <name>`
37    /// where it reported an output file.
38    pub stderr: Vec<String>,
39}

41/// One way a command line was compiled before: the inputs it read, and the result they gave.
42#[derive(Debug, Clone, Default, PartialEq, Eq)]
43pub(crate) struct Entry {
44    pub result: String,
45    /// The real roots the result embeds, when it embeds them; it only serves under the same ones.
46    pub pinned: Option<String>,
47    /// Source files: content hash and path, with markers for the roots.
48    pub files: Vec<(String, String)>,
49    /// Environment variables rustc said it read: name, and the value with markers, or `None` if unset.
50    pub env: Vec<(String, Option<String>)>,
51}

53/// SHA-256 of some bytes, in hex.
54pub(crate) fn sha(bytes: &[u8]) -> String {
55    hex(&Sha256::digest(bytes))
56}
57
58fn hex(bytes: &[u8]) -> String {
59    bytes.iter().map(|byte| format!("{byte:02x}")).collect()
60}
61
62/// SHA-256 of a file, read in pieces.
63pub(crate) fn sha_of_file(path: &Path) -> io::Result<String> {
64    let mut file = fs::File::open(path)?;
65    let mut hasher = Sha256::new();
66    let mut buffer = vec![0_u8; 1 << 16];
67    loop {
68        let read = file.read(&mut buffer)?;
69        if read == 0 {
70            break;
71        }
72        hasher.update(&buffer[..read]);
73    }
74    Ok(hex(&hasher.finalize()))
75}

77fn stamp(path: &Path) -> io::Result<(u64, u128)> {
78    let meta = fs::metadata(path)?;
79    let modified = meta.modified()?.duration_since(UNIX_EPOCH).map(|d| d.as_nanos()).unwrap_or(0);
80    Ok((meta.len(), modified))
81}

83impl Store {
84    /// The store named by `RAVENRS_XCACHE_DIR`, or `~/.cache/ravenrs/xcache`.
85    pub(crate) fn open() -> io::Result<Store> {
86        let root = match std::env::var_os("RAVENRS_XCACHE_DIR") {
87            Some(dir) if !dir.is_empty() => PathBuf::from(dir),
88            _ => {
89                let home = std::env::var_os("HOME")
90                    .or_else(|| std::env::var_os("USERPROFILE"))
91                    .ok_or_else(|| io::Error::other("no home directory; set RAVENRS_XCACHE_DIR"))?;
92                PathBuf::from(home).join(".cache").join("ravenrs").join("xcache")
93            }
94        };
95        fs::create_dir_all(&root)?;
96        Ok(Store { root })
97    }
98
99    fn fanned(&self, kind: &str, key: &str) -> PathBuf {
100        self.root.join(kind).join(&key[..key.len().min(2)]).join(key)
101    }
102
103    /// Writes a file whole or not at all.
104    pub(crate) fn write_atomic(path: &Path, bytes: &[u8]) -> io::Result<()> {
105        if let Some(folder) = path.parent() {
106            fs::create_dir_all(folder)?;
107        }
108        let nanos = SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_nanos()).unwrap_or(0);
109        let temporary = path.with_extension(format!("tmp-{}-{nanos}", std::process::id()));
110        let written = fs::File::create(&temporary).and_then(|mut file| file.write_all(bytes));
111        match written.and_then(|()| fs::rename(&temporary, path)) {
112            Ok(()) => Ok(()),
113            Err(error) => {
114                let _ = fs::remove_file(&temporary);
115                Err(error)
116            }
117        }
118    }
119
120    /// Where the file with this content hash is.
121    pub(crate) fn blob(&self, sha: &str) -> PathBuf {
122        self.fanned("blobs", sha)
123    }
124
125    /// Keeps a file under its content hash, copy-on-write where the file system can.
126    pub(crate) fn keep_file(&self, from: &Path, sha: &str) -> io::Result<()> {
127        let to = self.blob(sha);
128        if to.is_file() {
129            return Ok(());
130        }
131        if let Some(folder) = to.parent() {
132            fs::create_dir_all(folder)?;
133        }
134        let nanos = SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_nanos()).unwrap_or(0);
135        let temporary = to.with_extension(format!("tmp-{}-{nanos}", std::process::id()));
136        match fs::copy(from, &temporary).and_then(|_| fs::rename(&temporary, &to)) {
137            Ok(()) => Ok(()),
138            Err(error) => {
139                let _ = fs::remove_file(&temporary);
140                Err(error)
141            }
142        }
143    }
144
145    /// Keeps some bytes under their content hash.
146    pub(crate) fn keep_bytes(&self, bytes: &[u8]) -> io::Result<String> {
147        let sha = sha(bytes);
148        let to = self.blob(&sha);
149        if !to.is_file() {
150            Self::write_atomic(&to, bytes)?;
151        }
152        Ok(sha)
153    }

155    /// The identity of a file a compilation depends on: what this cache recorded for it when it
156    /// wrote or restored it, as long as its size and time are the same, else its content hash.
157    pub(crate) fn identity(&self, path: &Path) -> io::Result<String> {
158        let (size, modified) = stamp(path)?;
159        let record = self.fanned("identities", &sha(path.to_string_lossy().as_bytes()));
160        if let Ok(text) = fs::read_to_string(&record) {
161            let mut parts = text.split_whitespace();
162            let known = (parts.next(), parts.next(), parts.next());
163            if let (Some(known_size), Some(known_time), Some(identity)) = known
164                && known_size == size.to_string()
165                && known_time == modified.to_string()
166            {
167                return Ok(identity.to_owned());
168            }
169        }
170        let identity = sha_of_file(path)?;
171        let _ = Self::write_atomic(&record, format!("{size} {modified} {identity}").as_bytes());
172        Ok(identity)
173    }
174
175    /// Records the identity of a file this cache just wrote, so nothing has to read it again.
176    pub(crate) fn remember(&self, path: &Path, identity: &str) {
177        if let Ok((size, modified)) = stamp(path) {
178            let record = self.fanned("identities", &sha(path.to_string_lossy().as_bytes()));
179            let _ = Self::write_atomic(&record, format!("{size} {modified} {identity}").as_bytes());
180        }
181    }

183    /// The ways this command line was compiled before, newest first.
184    pub(crate) fn entries(&self, request: &str) -> Vec<Entry> {
185        let Ok(text) = fs::read_to_string(self.fanned("manifests", request)) else {
186            return Vec::new();
187        };
188        let mut entries: Vec<Entry> = Vec::new();
189        for line in text.lines() {
190            let (tag, rest) = line.split_once(' ').unwrap_or((line, ""));
191            if tag == "R" {
192                entries.push(Entry { result: rest.to_owned(), ..Entry::default() });
193                continue;
194            }
195            let Some(entry) = entries.last_mut() else { continue };
196            match tag {
197                "P" => entry.pinned = Some(rest.to_owned()),
198                "F" => {
199                    if let Some((sha, path)) = rest.split_once(' ') {
200                        entry.files.push((sha.to_owned(), path.to_owned()));
201                    }
202                }
203                "V" => {
204                    if let Some((name, value)) = rest.split_once('=') {
205                        entry.env.push((name.to_owned(), Some(value.to_owned())));
206                    }
207                }
208                "U" => entry.env.push((rest.to_owned(), None)),
209                _ => {}
210            }
211        }
212        entries
213    }

215    /// Puts an entry first among the ways this command line was compiled, keeping the last twenty.
216    pub(crate) fn add_entry(&self, request: &str, entry: &Entry) -> io::Result<()> {
217        let mut entries = self.entries(request);
218        entries.retain(|known| known.result != entry.result);
219        entries.insert(0, entry.clone());
220        entries.truncate(20);
221        let mut text = String::new();
222        for entry in &entries {
223            text.push_str(&entry_text(entry));
224        }
225        Self::write_atomic(&self.fanned("manifests", request), text.as_bytes())
226    }

228    /// What was kept under a result key.
229    pub(crate) fn stored(&self, result: &str) -> Option<Stored> {
230        let text = fs::read_to_string(self.fanned("results", result)).ok()?;
231        let mut stored = Stored::default();
232        for line in text.lines() {
233            match line.split_once(' ') {
234                Some(("O", rest)) => {
235                    let mut parts = rest.splitn(4, ' ');
236                    let (emit, sha, size, name) =
237                        (parts.next()?, parts.next()?, parts.next()?, parts.next()?);
238                    stored.outputs.push(Output {
239                        emit: emit.to_owned(),
240                        sha: sha.to_owned(),
241                        size: size.parse().ok()?,
242                        name: name.to_owned(),
243                    });
244                }
245                Some(("S", rest)) => stored.stderr.push(rest.to_owned()),
246                _ => {}
247            }
248        }
249        (!stored.outputs.is_empty()).then_some(stored)
250    }
251
252    /// Keeps the list of what a compilation produced.
253    pub(crate) fn keep_stored(&self, result: &str, stored: &Stored) -> io::Result<()> {
254        let mut text = String::from("xcache-result 1\n");
255        for output in &stored.outputs {
256            text.push_str(&format!("O {} {} {} {}\n", output.emit, output.sha, output.size, output.name));
257        }
258        for line in &stored.stderr {
259            text.push_str(&format!("S {line}\n"));
260        }
261        Self::write_atomic(&self.fanned("results", result), text.as_bytes())
262    }
263
264    /// Forgets a result that could not be restored.
265    pub(crate) fn drop_stored(&self, result: &str) {
266        let _ = fs::remove_file(self.fanned("results", result));
267    }

269    /// Marks a result as used now, which is what cleaning goes by.
270    pub(crate) fn touch(&self, result: &str) {
271        if let Ok(file) = fs::File::options().write(true).open(self.fanned("results", result)) {
272            let _ = file.set_modified(SystemTime::now());
273        }
274    }

276    /// Adds a line to the log of requests.
277    pub(crate) fn log(&self, line: &str) {
278        let path = self.root.join("events.tsv");
279        if let Ok(mut file) = fs::File::options().create(true).append(true).open(path) {
280            let _ = file.write_all(format!("{line}\n").as_bytes());
281        }
282    }

285/// An entry as text. The result key is made from this, without its first line.
286pub(crate) fn entry_text(entry: &Entry) -> String {
287    let mut text = format!("R {}\n", entry.result);
288    if let Some(roots) = &entry.pinned {
289        text.push_str(&format!("P {roots}\n"));
290    }
291    for (sha, path) in &entry.files {
292        text.push_str(&format!("F {sha} {path}\n"));
293    }
294    for (name, value) in &entry.env {
295        match value {
296            Some(value) => text.push_str(&format!("V {name}={value}\n")),
297            None => text.push_str(&format!("U {name}\n")),
298        }
299    }
300    text
301}

307    fn store(name: &str) -> Store {
308        let root = std::env::temp_dir().join(format!("xcache-store-{name}-{}", std::process::id()));
309        let _ = fs::remove_dir_all(&root);
310        fs::create_dir_all(&root).unwrap_or_default();
311        Store { root }
312    }

314    #[test]
315    fn entries_and_results_come_back_as_they_were_kept() {
316        let store = store("roundtrip");
317        let entry = Entry {
318            result: "r1".into(),
319            pinned: Some("/a|/b".into()),
320            files: vec![("abc".into(), "<xcache:cwd>/src/lib with space.rs".into())],
321            env: vec![("OUT_DIR".into(), Some("<xcache:target>/out=x".into())), ("GONE".into(), None)],
322        };
323        store.add_entry("req", &entry).unwrap_or_default();
324        let newer = Entry { result: "r2".into(), ..Entry::default() };
325        store.add_entry("req", &newer).unwrap_or_default();
326        assert_eq!(store.entries("req"), vec![newer, entry]);
327
328        let stored = Stored {
329            outputs: vec![Output {
330                emit: "link".into(),
331                sha: "f00".into(),
332                size: 3,
333                name: "libx-1 2.rlib".into(),
334            }],
335            stderr: vec!["A link libx-1 2.rlib".into(), "D {\"message\":\"x\"}".into()],
336        };
337        store.keep_stored("r1", &stored).unwrap_or_default();
338        assert_eq!(store.stored("r1"), Some(stored));
339        assert_eq!(store.stored("missing"), None);
340        let _ = fs::remove_dir_all(&store.root);
341    }

343    #[test]
344    fn a_file_is_known_by_what_was_recorded_until_it_changes() {
345        let store = store("identity");
346        let file = store.root.join("lib.rmeta");
347        fs::write(&file, b"one").unwrap_or_default();
348        assert_eq!(store.identity(&file).ok(), Some(sha(b"one")));
349        store.remember(&file, "what-it-was-built-from");
350        assert_eq!(store.identity(&file).ok().as_deref(), Some("what-it-was-built-from"));
351        fs::write(&file, b"three").unwrap_or_default();
352        assert_eq!(store.identity(&file).ok(), Some(sha(b"three")));
353        let _ = fs::remove_dir_all(&store.root);
354    }

356    #[test]
357    fn content_is_kept_once_under_its_hash() {
358        let store = store("blobs");
359        let kept = store.keep_bytes(b"hello").unwrap_or_default();
360        assert_eq!(kept, sha(b"hello"));
361        assert_eq!(fs::read(store.blob(&kept)).ok().as_deref(), Some(&b"hello"[..]));
362        assert_eq!(sha_of_file(&store.blob(&kept)).ok(), Some(kept));
363        let _ = fs::remove_dir_all(&store.root);
