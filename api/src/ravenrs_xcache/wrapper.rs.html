32/// Milliseconds spent in each part of one request.
33#[derive(Default)]
34struct Times {
35    key: u128,
36    lookup: u128,
37    restore: u128,
38    compile: u128,
39}
40
41/// What became of one request, for the log.
42struct Outcome<'a> {
43    kind: &'a str,
44    crate_name: &'a str,
45    reason: &'a str,
46    bytes: u64,
47}
48
49fn log(store: &Store, outcome: &Outcome<'_>, times: &Times) {
50    let now = SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_secs()).unwrap_or(0);
51    store.log(&format!(
52        "{now}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}",
53        outcome.kind,
54        outcome.crate_name,
55        outcome.reason.replace(['\t', '\n'], " "),
56        times.key,
57        times.lookup,
58        times.restore,
59        times.compile,
60        outcome.bytes
61    ));
62}

64/// Runs rustc as it was asked, with nothing in between.
65fn untouched(compiler: &OsStr, rest: &[OsString]) -> i32 {
66    match Command::new(compiler).args(rest).status() {
67        Ok(status) => status.code().unwrap_or(1),
68        Err(error) => {
69            eprintln!("ravenrs-xcache: cannot run {}: {error}", compiler.to_string_lossy());
70            1
71        }
72    }
73}

75/// The wrapper: `ravenrs-xcache <rustc> <arguments>`.
76pub(crate) fn run(compiler: &OsStr, rest: &[OsString]) -> i32 {
77    let started = Instant::now();
78    let off = std::env::var_os("RAVENRS_XCACHE_DISABLE").is_some_and(|v| v == "1" || v == "true");
79    let Ok(store) = Store::open() else {
80        return untouched(compiler, rest);
81    };
82    if off {
83        return untouched(compiler, rest);
84    }
85    let pass = |crate_name: &str, reason: &str| {
86        let code_started = Instant::now();
87        let code = untouched(compiler, rest);
88        let times = Times { compile: code_started.elapsed().as_millis(), ..Times::default() };
89        let kind = if reason.contains("question") { "probe" } else { "pass" };
90        log(&store, &Outcome { kind, crate_name, reason, bytes: 0 }, &times);
91        code
92    };
93    let Some(args) = rest.iter().map(|arg| arg.to_str().map(str::to_owned)).collect::<Option<Vec<_>>>()
94    else {
95        return pass("?", "an argument is not UTF-8");
96    };
97    let name_of = |args: &[String]| {
98        let at = args.iter().position(|a| a == "--crate-name");
99        at.and_then(|at| args.get(at + 1)).cloned().unwrap_or_else(|| "?".to_owned())
100    };
101    if Path::new(compiler).file_stem().is_none_or(|stem| stem != "rustc") {
102        // `ravenrs-xcache clippy-driver rustc …`: a workspace wrapper compiles, not rustc.
103        return pass(&name_of(&args), "another tool compiles in rustc's place (clippy-driver)");
104    }
105    let invocation = match parse(&args) {
106        Ok(invocation) => invocation,
107        Err(reason) => return pass(&name_of(&args), &reason),
108    };
109    let Ok(cwd) = std::env::current_dir() else {
110        return pass(&invocation.crate_name, "the working directory cannot be read");
111    };
112    let cargo_home = std::env::var_os("CARGO_HOME")
113        .map(PathBuf::from)
114        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".cargo")));
115    let roots = Roots::find(&invocation.out_dir, invocation.target.as_deref(), &cwd, cargo_home.as_deref());
116    let remaps = match roots.remap_args() {
117        Ok(remaps) => remaps,
118        Err(reason) => return pass(&invocation.crate_name, &reason),
119    };
120    let request = match request_key(&store, compiler, &invocation, &roots, &cwd) {
121        Ok(request) => request,
122        Err(reason) => return pass(&invocation.crate_name, &reason),
123    };
124    let mut times = Times { key: started.elapsed().as_millis(), ..Times::default() };
125    let job = Job {
126        store: &store,
127        compiler,
128        rest,
129        invocation: &invocation,
130        roots: &roots,
131        remaps: &remaps,
132        cwd: &cwd,
133    };
134
135    if invocation.kind == Kind::ProcMacro {
136        return job.proc_macro(&request, &mut times);
137    }
138    let looking = Instant::now();
139    let found = job.look_up(&request);
140    times.lookup = looking.elapsed().as_millis();
141    let reason = match found {
142        Ok(result) => {
143            let restoring = Instant::now();
144            match job.restore(&result) {
145                Ok(bytes) => {
146                    times.restore = restoring.elapsed().as_millis();
147                    store.touch(&result);
148                    let outcome =
149                        Outcome { kind: "hit", crate_name: &invocation.crate_name, reason: "", bytes };
150                    log(&store, &outcome, &times);
151                    return 0;
152                }
153                Err(error) => {
154                    store.drop_stored(&result);
155                    format!("the kept result could not be restored ({error}); compiled again")
156                }
157            }
158        }
159        Err(reason) => reason,
160    };
161    job.compile_and_keep(&request, &reason, &mut times)
162}

164/// What identifies the compiler: what `rustc -vV` says, read once for each binary.
165fn compiler_identity(store: &Store, compiler: &OsStr) -> Result<String, String> {
166    let meta = fs::metadata(compiler).map_err(|e| format!("rustc cannot be read: {e}"))?;
167    let modified = meta.modified().ok().and_then(|t| t.duration_since(UNIX_EPOCH).ok()).map(|d| d.as_nanos());
168    let toolchain = std::env::var("RUSTUP_TOOLCHAIN").unwrap_or_default();
169    let which =
170        sha(format!("{}|{}|{modified:?}|{toolchain}", compiler.to_string_lossy(), meta.len()).as_bytes());
171    let record = store.root.join("compilers").join(which);
172    if let Ok(known) = fs::read_to_string(&record) {
173        return Ok(known);
174    }
175    let output = Command::new(compiler).arg("-vV").output().map_err(|e| format!("rustc -vV failed: {e}"))?;
176    if !output.status.success() {
177        return Err("rustc -vV failed".to_owned());
178    }
179    let version = String::from_utf8_lossy(&output.stdout).into_owned();
180    let _ = Store::write_atomic(&record, version.as_bytes());
181    Ok(version)
182}

184/// The key of a command line: everything known before compiling that can change the result.
185fn request_key(
186    store: &Store,
187    compiler: &OsStr,
188    invocation: &Invocation,
189    roots: &Roots,
190    cwd: &Path,
191) -> Result<String, String> {
192    let mut text = format!("{KEY_VERSION}\n{}\n", compiler_identity(store, compiler)?);
193    text.push_str(&format!("kind {:?}\ncwd {}\n", invocation.kind, roots.norm(&cwd.to_string_lossy())));
194    for arg in &invocation.key_args {
195        text.push_str(&format!("arg {}\n", roots.norm(arg)));
196    }
197    for (spec, path) in &invocation.externs {
198        let identity = match path {
199            Some(path) => store
200                .identity(&cwd.join(path))
201                .map_err(|e| format!("the dependency {path} cannot be read: {e}"))?,
202            None => String::new(),
203        };
204        text.push_str(&format!("extern {spec} {identity}\n"));
205    }
206    let mut environment: Vec<(String, String)> = std::env::vars_os()
207        .filter_map(|(name, value)| Some((name.into_string().ok()?, value.into_string().ok()?)))
208        .filter(|(name, _)| name.starts_with("CARGO_PKG_") || ENVIRONMENT.contains(&name.as_str()))
209        .collect();
210    environment.sort();
211    for (name, value) in environment {
212        text.push_str(&format!("env {name}={}\n", roots.norm(&value)));
213    }
214    let key = sha(text.as_bytes());
215    // RAVENRS_XCACHE_TRACE=<folder>: what each key was made of, to compare two builds that should
216    // have shared a crate and did not.
217    if let Some(folder) = std::env::var_os("RAVENRS_XCACHE_TRACE") {
218        let file = Path::new(&folder).join(format!("{}-{}.key", invocation.crate_name, &key[..12]));
219        let _ = Store::write_atomic(&file, text.as_bytes());
220    }
221    Ok(key)
222}

224/// One library or proc macro to produce.
225struct Job<'a> {
226    store: &'a Store,
227    compiler: &'a OsStr,
228    rest: &'a [OsString],
229    invocation: &'a Invocation,
230    roots: &'a Roots,
231    remaps: &'a [String],
232    cwd: &'a Path,
233}

235/// What rustc did: its exit code, and what it wrote to stderr.
236struct Compiled {
237    code: i32,
238    lines: Vec<String>,
239    began: SystemTime,
240}
241
242impl Job<'_> {
243    /// The result kept for this command line whose recorded inputs are all as they are now, or why
244    /// there is none.
245    fn look_up(&self, request: &str) -> Result<String, String> {
246        let entries = self.store.entries(request);
247        if entries.is_empty() {
248            return Err("first time: the compiler, the flags or a dependency are new to the cache".to_owned());
249        }
250        let mut why = String::new();
251        for entry in &entries {
252            match self.still_holds(entry) {
253                Ok(()) if self.store.stored(&entry.result).is_some() => return Ok(entry.result.clone()),
254                Ok(()) => why = "its entry was cleaned from the cache".to_owned(),
255                Err(reason) if why.is_empty() => why = reason,
256                Err(_) => {}
257            }
258        }
259        Err(why)
260    }
261
262    fn still_holds(&self, entry: &Entry) -> Result<(), String> {
263        if let Some(pinned) = &entry.pinned
264            && *pinned != self.roots.raw()
265        {
266            return Err("it embeds the paths it was built under, and these are other paths".to_owned());
267        }
268        for (name, value) in &entry.env {
269            let now = std::env::var(name).ok().map(|value| self.roots.norm(&value));
270            if now != *value {
271                return Err(format!("the environment variable {name} changed"));
272            }
273        }
274        for (known, path) in &entry.files {
275            let real = self.cwd.join(self.roots.denorm(path));
276            match self.store.identity(&real) {
277                Ok(now) if now == *known => {}
278                Ok(_) => return Err(format!("the source changed: {path}")),
279                Err(_) => return Err(format!("the source is gone: {path}")),
280            }
281        }
282        Ok(())
283    }
284
285    /// Puts the kept files in the output directory and says to Cargo what rustc said. Returns the
286    /// bytes restored.
287    fn restore(&self, result: &str) -> Result<u64, String> {
288        let stored = self.store.stored(result).ok_or("the result is gone")?;
289        let verify = std::env::var_os("RAVENRS_XCACHE_VERIFY").is_none_or(|v| v != "0");
290        let mut bytes = 0;
291        for output in &stored.outputs {
292            let blob = self.store.blob(&output.sha);
293            let size = fs::metadata(&blob).map_err(|e| format!("{}: {e}", output.name))?.len();
294            if size != output.size || (verify && sha_of_file(&blob).ok().as_deref() != Some(&output.sha)) {
295                let _ = fs::remove_file(&blob);
296                return Err(format!("{} is not what was kept", output.name));
297            }
298            let to = self.invocation.out_dir.join(&output.name);
299            let _ = fs::remove_file(&to);
300            if output.emit == "dep-info" {
301                // Cargo reads this file to know what to watch: it has to name this checkout's files.
302                let text = fs::read_to_string(&blob).map_err(|e| e.to_string())?;
303                fs::write(&to, self.roots.denorm(&text)).map_err(|e| e.to_string())?;
304            } else {
305                fs::copy(&blob, &to).map_err(|e| format!("{}: {e}", output.name))?;
306                // A copy keeps the time of the kept file. Cargo takes a crate whose output is older
307                // than a dependency's for stale, and would compile again what was just restored
308                // next to something compiled now: a restored file is as new as a compiled one.
309                if let Ok(file) = fs::File::options().write(true).open(&to) {
310                    let _ = file.set_modified(SystemTime::now());
311                }
312                self.store.remember(&to, &output.sha);
313            }
314            bytes += size;
315        }
316        let mut stderr = std::io::stderr().lock();
317        for line in &stored.stderr {
318            let text = match line.split_once(' ') {
319                Some(("A", rest)) => {
320                    let (emit, name) = rest.split_once(' ').unwrap_or((rest, ""));
321                    artifact_line(&self.invocation.out_dir.join(name), emit)
322                }
323                Some(("D", rest)) => rest.to_owned(),
324                _ => continue,
325            };
326            let _ = writeln!(stderr, "{text}");
327        }
328        Ok(bytes)
329    }

331    /// Runs rustc with the roots remapped, passing on what it says as it says it.
332    fn compile(&self) -> Compiled {
333        let began = SystemTime::now();
334        let mut command = Command::new(self.compiler);
335        command.args(self.rest).args(self.remaps).stderr(Stdio::piped());
336        let mut lines = Vec::new();
337        let code = match command.spawn() {
338            Ok(mut child) => {
339                if let Some(pipe) = child.stderr.take() {
340                    let mut stderr = std::io::stderr();
341                    for line in BufReader::new(pipe).split(b'\n').map_while(Result::ok) {
342                        let _ = stderr.write_all(&line);
343                        let _ = stderr.write_all(b"\n");
344                        lines.push(String::from_utf8_lossy(&line).into_owned());
345                    }
346                }
347                child.wait().ok().and_then(|status| status.code()).unwrap_or(1)
348            }
349            Err(error) => {
350                eprintln!("ravenrs-xcache: cannot run {}: {error}", self.compiler.to_string_lossy());
351                1
352            }
353        };
354        Compiled { code, lines, began }
355    }

357    /// The files rustc reported, each inside the output directory: `(emit, file name)`.
358    fn reported(&self, lines: &[String]) -> Result<Vec<(String, String)>, String> {
359        let mut outputs = Vec::new();
360        for line in lines {
361            let Some((path, emit)) = artifact_of(line) else { continue };
362            let path = PathBuf::from(path);
363            let inside = path.parent() == Some(self.invocation.out_dir.as_path());
364            match path.file_name().and_then(|name| name.to_str()) {
365                Some(name) if inside => outputs.push((emit, name.to_owned())),
366                _ => return Err("rustc wrote a file outside the output directory".to_owned()),
367            }
368        }
369        Ok(outputs)
370    }
371
372    /// What the compilation read, from the dep-info rustc wrote: the entry without its result.
373    fn inputs(&self, outputs: &[(String, String)], began: SystemTime) -> Result<Entry, String> {
374        let (_, name) = outputs
375            .iter()
376            .find(|(emit, _)| emit == "dep-info")
377            .ok_or("rustc wrote no dep-info, so its inputs are not known")?;
378        let text = fs::read_to_string(self.invocation.out_dir.join(name)).map_err(|e| e.to_string())?;
379        let (files, variables) = dep_info(&text);
380        let mut entry = Entry::default();
381        for file in files {
382            let real = self.cwd.join(&file);
383            let written =
384                fs::metadata(&real).and_then(|meta| meta.modified()).map_err(|e| format!("{file}: {e}"))?;
385            if written >= began {
386                return Err(format!("{file} was written while it compiled"));
387            }
388            let known = self.store.identity(&real).map_err(|e| format!("{file}: {e}"))?;
389            entry.files.push((known, self.roots.norm(&file)));
390        }
391        for name in variables {
392            let value = std::env::var(&name).ok().map(|value| self.roots.norm(&value));
393            entry.env.push((name, value));
394        }
395        Ok(entry)
396    }

398    fn compile_and_keep(&self, request: &str, reason: &str, times: &mut Times) -> i32 {
399        let compiling = Instant::now();
400        let compiled = self.compile();
401        times.compile = compiling.elapsed().as_millis();
402        let crate_name = self.invocation.crate_name.as_str();
403        if compiled.code != 0 {
404            log(self.store, &Outcome { kind: "failed", crate_name, reason, bytes: 0 }, times);
405            return compiled.code;
406        }
407        match self.keep(request, &compiled) {
408            Ok(bytes) => log(self.store, &Outcome { kind: "miss", crate_name, reason, bytes }, times),
409            Err(why) => {
410                let reason = format!("{reason}; not kept: {why}");
411                log(self.store, &Outcome { kind: "miss", crate_name, reason: &reason, bytes: 0 }, times);
412            }
413        }
414        compiled.code
415    }

417    /// Keeps what a compilation produced. Returns the bytes kept.
418    fn keep(&self, request: &str, compiled: &Compiled) -> Result<u64, String> {
419        let outputs = self.reported(&compiled.lines)?;
420        let mut entry = self.inputs(&outputs, compiled.began)?;
421        let mut stored = Stored::default();
422        let mut embeds = !self.invocation.plain_objects;
423        let mut bytes = 0;
424        for (emit, name) in &outputs {
425            let path = self.invocation.out_dir.join(name);
426            let content = fs::read(&path).map_err(|e| format!("{name}: {e}"))?;
427            let hash = if emit == "dep-info" {
428                let text = String::from_utf8(content).map_err(|_| "the dep-info is not text")?;
429                self.store.keep_bytes(self.roots.norm(&text).as_bytes()).map_err(|e| e.to_string())?
430            } else {
431                // The proof: with the roots remapped, a real path still in the bytes is data the
432                // crate embedded, and the artifact only serves where it was built.
433                embeds = embeds || self.roots.leaked_in(&content).is_some();
434                let hash = sha(&content);
435                self.store.keep_file(&path, &hash).map_err(|e| e.to_string())?;
436                self.store.remember(&path, &hash);
437                hash
438            };
439            let size =
440                fs::metadata(self.store.blob(&hash)).map(|meta| meta.len()).map_err(|e| e.to_string())?;
441            bytes += size;
442            stored.outputs.push(Output { emit: emit.clone(), sha: hash, size, name: name.clone() });
443        }
444        for line in &compiled.lines {
445            match artifact_of(line) {
446                Some((path, emit)) => {
447                    let name =
448                        Path::new(&path).file_name().and_then(|n| n.to_str()).unwrap_or_default().to_owned();
449                    stored.stderr.push(format!("A {emit} {name}"));
450                }
451                None => stored.stderr.push(format!("D {line}")),
452            }
453        }
454        if embeds {
455            entry.pinned = Some(self.roots.raw());
456        }
457        entry.result = sha(format!("{request}\n{}", entry_text(&entry)).as_bytes());
458        self.store.keep_stored(&entry.result, &stored).map_err(|e| e.to_string())?;
459        self.store.add_entry(request, &entry).map_err(|e| e.to_string())?;
460        Ok(bytes)
461    }

463    /// A proc macro links, so it is compiled here every time. Compiled with the roots remapped it is
464    /// the same macro in every worktree, so the crates that use it are told apart by what it was
465    /// built from, not by the bytes of a library the linker stamped with this build directory.
466    fn proc_macro(&self, request: &str, times: &mut Times) -> i32 {
467        let compiling = Instant::now();
468        let compiled = self.compile();
469        times.compile = compiling.elapsed().as_millis();
470        let crate_name = self.invocation.crate_name.as_str();
471        let mut reason = "a proc macro links; compiled with the paths remapped".to_owned();
472        if compiled.code == 0 {
473            let known = self.reported(&compiled.lines).and_then(|outputs| {
474                let entry = self.inputs(&outputs, compiled.began)?;
475                Ok((outputs, sha(format!("proc-macro {request}\n{}", entry_text(&entry)).as_bytes())))
476            });
477            match known {
478                Ok((outputs, identity)) => {
479                    for (emit, name) in &outputs {
480                        if emit == "link" {
481                            self.store.remember(&self.invocation.out_dir.join(name), &identity);
482                        }
483                    }
484                }
485                Err(why) => reason = format!("{reason}; its users are told apart by its bytes: {why}"),
486            }
487        }
488        log(self.store, &Outcome { kind: "pass", crate_name, reason: &reason, bytes: 0 }, times);
489        compiled.code
490    }
491}

493/// The path and kind of an output file, from the line rustc writes for it with `--json=artifacts`.
494fn artifact_of(line: &str) -> Option<(String, String)> {
495    if !line.starts_with("{\"$message_type\":\"artifact\"") {
496        return None;
497    }
498    Some((json_string(line, "artifact")?, json_string(line, "emit")?))
499}

501/// The string value of a key in a line of JSON written by rustc.
502fn json_string(line: &str, key: &str) -> Option<String> {
503    let start = line.find(&format!("\"{key}\":\""))? + key.len() + 4;
504    let mut value = String::new();
505    let mut characters = line[start..].chars();
506    while let Some(character) = characters.next() {
507        match character {
508            '"' => return Some(value),
509            '\\' => value.push(characters.next()?),
510            other => value.push(other),
511        }
512    }
513    None
514}

516/// The line rustc writes for an output file.
517fn artifact_line(path: &Path, emit: &str) -> String {
518    let path = path.to_string_lossy().replace('\\', "\\\\").replace('"', "\\\"");
519    format!("{{\"$message_type\":\"artifact\",\"artifact\":\"{path}\",\"emit\":\"{emit}\"}}")
520}

522/// The source files and the environment variables in a dep-info file, as rustc writes it: Makefile
523/// rules with spaces escaped, then `# env-dep:NAME=VALUE` comments.
524fn dep_info(text: &str) -> (Vec<String>, Vec<String>) {
525    let mut files: Vec<String> = Vec::new();
526    let mut variables: Vec<String> = Vec::new();
527    for line in text.lines() {
528        if let Some(variable) = line.strip_prefix("# env-dep:") {
529            let name = variable.split('=').next().unwrap_or(variable).to_owned();
530            if !variables.contains(&name) {
531                variables.push(name);
532            }
533            continue;
534        }
535        let Some((target, sources)) = line.split_once(": ") else { continue };
536        if target.starts_with('#') {
537            continue;
538        }
539        let mut current = String::new();
540        let mut characters = sources.chars().peekable();
541        while let Some(character) = characters.next() {
542            match character {
543                '\\' if characters.peek() == Some(&' ') => {
544                    current.push(' ');
545                    characters.next();
546                }
547                ' ' => {
548                    if !current.is_empty() && !files.contains(&current) {
549                        files.push(std::mem::take(&mut current));
550                    }
551                    current.clear();
552                }
553                other => current.push(other),
554            }
555        }
556        if !current.is_empty() && !files.contains(&current) {
557            files.push(current);
558        }
559    }
560    (files, variables)
561}
