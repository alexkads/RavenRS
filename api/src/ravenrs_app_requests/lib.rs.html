1//! Requests decided step by step, as an optional `ravenrs-app` package.
2//!
3//! The application says which steps a request goes through and who may decide each one; the
4//! package keeps the request, every decision and an event for each decision. A decision is
5//! recorded once however many times it is sent, one taken on an old version of the request is
6//! refused, and its event is written **in the same commit**: when the decision is refused, no event
7//! exists to be sent.
8//!
9//! [`Dispatcher`] hands the events to an [`EventSink`] and never sends blindly again: an event sent
10//! with no answer is checked with [`EventSink::status`] first, and stays to be reconciled when the
11//! sink cannot tell.
12//!
13//! Nothing here belongs to a sector. Steps, roles and texts are the application's, and two
14//! applications can install the same package with different ones.
15

40/// The state of a request no step is deciding anymore because it was approved.
41pub const APPROVED: &str = "approved";

45/// A step a request goes through, and the role that decides it.
46#[derive(Clone, Debug, PartialEq, Eq)]
47pub struct Step {
48    /// What the state of a request on this step is called.
49    pub code: String,
50    /// What people read.
51    pub caption: String,
52    /// The role whose members decide it. Nobody else does, whatever other role they have, unless
53    /// a member lent it with [`RequestsModule::delegate`].
54    pub decided_by: String,
55    /// An expression of `ravenrs-criteria` over the request that has to hold for a decision, with
56    /// `?CurrentUserName` and `?CurrentUserId` as the person deciding.
57    pub when: Option<String>,
58    /// How long the step has, in seconds from when its task opens.
59    pub due_in: Option<i64>,
60}
61
62impl Step {
63    /// A step with this code and caption, decided by members of this role.
64    pub fn new(code: impl Into<String>, caption: impl Into<String>, decided_by: impl Into<String>) -> Self {
65        Self {
66            code: code.into(),
67            caption: caption.into(),
68            decided_by: decided_by.into(),
69            when: None,
70            due_in: None,
71        }
72    }
73
74    /// Only decided when this expression holds over the request, such as
75    /// `[requester] <> ?CurrentUserName`.
76    pub fn when(mut self, expression: impl Into<String>) -> Self {
77        self.when = Some(expression.into());
78        self
79    }
80
81    /// Gives the step this many seconds from when its task opens.
82    pub fn due_in(mut self, seconds: i64) -> Self {
83        self.due_in = Some(seconds);
84        self
85    }
86}

88/// A rule of the application about one decision, beyond the role of the step: an error says why it
89/// is refused.
90pub type Policy = Rc<dyn Fn(&Identity, &Object, &Step) -> Result<(), String>>;
91
92/// A policy that refuses decisions taken by whoever made the request.
93pub fn forbid_self_decision() -> Policy {
94    Rc::new(|identity, request, _| {
95        if request.get("requester") == Value::Text(identity.name.clone()) {
96            Err("Whoever made a request does not decide it.".to_owned())
97        } else {
98            Ok(())
99        }
100    })
101}

139    /// The entities it declares: requests, decisions, tasks and delegations. Events are the
140    /// framework's, from [`OutboxModule`], which this package depends on.
141    ///
142    /// The list of tasks shows the open ones: with [`task_reader`] on each role, it is everyone's
143    /// own inbox.
144    pub fn declared_entities(&self) -> Vec<EntityInfo> {
145        let mut states: Vec<EnumOption> =
146            self.steps.iter().map(|step| EnumOption::new(step.code.clone(), step.caption.clone())).collect();
147        states.push(EnumOption::new(APPROVED, "Approved"));
148        states.push(EnumOption::new(REJECTED, "Rejected"));
149        let request = EntityInfo::new(REQUEST)
150            .field(FieldInfo::new("id", FieldKind::Integer).key())
151            .field(FieldInfo::new("title", FieldKind::Text).default_property())
152            .field(FieldInfo::new("requester", FieldKind::Text))
153            .field(FieldInfo::new("unit", FieldKind::Text))
154            .field(FieldInfo::new("state", FieldKind::Enum(states)))
155            .field(FieldInfo::new("version", FieldKind::Integer))
156            .field(FieldInfo::new("document", FieldKind::Text));
157        let outcomes = vec![EnumOption::new("approve", "Approve"), EnumOption::new("reject", "Reject")];
158        let decision = EntityInfo::new(DECISION)
159            .field(FieldInfo::new("id", FieldKind::Text).key())
160            .field(FieldInfo::new("request", FieldKind::Integer))
161            .field(FieldInfo::new("step", FieldKind::Text))
162            .field(FieldInfo::new("decider", FieldKind::Text))
163            .field(FieldInfo::new("outcome", FieldKind::Enum(outcomes)))
164            .field(FieldInfo::new("reason", FieldKind::LongText))
165            .field(FieldInfo::new("at", FieldKind::Integer));
166        let task_states = vec![EnumOption::new("open", "Open"), EnumOption::new("done", "Done")];
167        let task = EntityInfo::new(TASK)
168            .field(FieldInfo::new("id", FieldKind::Text).key().in_list(false))
169            .field(FieldInfo::new("title", FieldKind::Text).default_property())
170            .field(FieldInfo::new("request", FieldKind::Integer))
171            .field(FieldInfo::new("step", FieldKind::Text))
172            .field(FieldInfo::new("role", FieldKind::Text))
173            .field(FieldInfo::new("state", FieldKind::Enum(task_states)))
174            .field(FieldInfo::new("due", FieldKind::Integer))
175            .field(FieldInfo::new("decided_by", FieldKind::Text))
176            .list_criteria("[state] = 'open'");
177        let delegation = EntityInfo::new(DELEGATION)
178            .field(FieldInfo::new("id", FieldKind::Text).key())
179            .field(FieldInfo::new("delegator", FieldKind::Text))
180            .field(FieldInfo::new("delegate", FieldKind::Text).default_property())
181            .field(FieldInfo::new("role", FieldKind::Text))
182            .field(FieldInfo::new("until", FieldKind::Integer));
183        vec![request, decision, task, delegation]
184    }

186    fn step(&self, code: &str) -> Option<(usize, &Step)> {
187        self.steps.iter().enumerate().find(|(_, step)| step.code == code)
188    }

190    /// Makes a request on its first step, as this person, and opens the task of that step.
191    pub fn submit(
192        &self,
193        space: &mut dyn ObjectSpace,
194        identity: &Identity,
195        id: i64,
196        title: &str,
197        unit: &str,
198        at: i64,
199    ) -> Result<Object, SpaceError> {
200        let mut request = space.new_object(REQUEST)?;
201        request.set("id", id as f64);
202        request.set("title", title);
203        request.set("requester", identity.name.clone());
204        request.set("unit", unit);
205        request.set("state", self.steps[0].code.clone());
206        request.set("version", 1.0);
207        let written =
208            space.save(request.clone()).and_then(|()| open_task(space, &request, &self.steps[0], at));
209        if let Err(error) = written {
210            space.rollback();
211            return Err(error);
212        }
213        commit_or_rollback(space)?;
214        Ok(request)
215    }

217    /// Decides the step a request is on, as this person.
218    ///
219    /// `decision.id` makes it idempotent: sending the same decision again changes nothing and says
220    /// so. `organization` is whose event it is, until the organization comes from the context of the
221    /// space (#225). `decision.version` is the version of the request the person saw; another one means
222    /// someone decided in between, and the decision is refused as a conflict. The decision, the
223    /// request moving on and its event are one commit.
224    pub fn decide(
225        &self,
226        space: &mut dyn ObjectSpace,
227        organization: &str,
228        identity: &Identity,
229        decision: &Decision,
230        at: i64,
231    ) -> Result<Decided, SpaceError> {
232        if space.find(DECISION, &Value::Text(decision.id.clone()))?.is_some() {
233            return Ok(Decided::AlreadyRecorded);
234        }
235        let not_there =
236            || SpaceError::Denied("This request is not there, or you cannot decide it.".to_owned());
237        let mut request =
238            space.find(REQUEST, &Value::Number(decision.request as f64))?.ok_or_else(not_there)?;
239        let state = text(&request.get("state"));
240        let Some((index, step)) = self.step(&state) else {
241            return Err(SpaceError::Refused("This request is not waiting for a decision.".to_owned()));
242        };
243        if request.get("version") != Value::Number(decision.version as f64) {
244            return Err(SpaceError::Conflict("Someone decided this request in the meantime.".to_owned()));
245        }
246        if !self.roles_of(space, identity, at)?.contains(&step.decided_by) {
247            return Err(SpaceError::Denied(format!("Only {} decides this step.", step.decided_by)));
248        }
249        if let Some(expression) = &step.when {
250            let criteria = Criteria::parse(expression).map_err(|e| SpaceError::BadQuery(e.to_string()))?;
251            let environment = ravenrs_app::environment()
252                .parameter("CurrentUserName", identity.name.clone())
253                .parameter("CurrentUserId", identity.key.clone());
254            match Scope::new(&environment).row(&request).evaluate(&criteria) {
255                Ok(Value::Bool(true)) => {}
256                _ => {
257                    return Err(SpaceError::Denied(format!(
258                        "The rule of {} does not allow this decision.",
259                        step.caption
260                    )));
261                }
262            }
263        }
264        for policy in &self.policies {
265            policy(identity, &request, step).map_err(SpaceError::Denied)?;
266        }
267        let next = match decision.outcome {
268            Outcome::Reject => REJECTED.to_owned(),
269            Outcome::Approve => {
270                self.steps.get(index + 1).map_or_else(|| APPROVED.to_owned(), |step| step.code.clone())
271            }
272        };
273        let mut record = space.new_object(DECISION)?;
274        record.set("id", decision.id.clone());
275        record.set("request", decision.request as f64);
276        record.set("step", step.code.clone());
277        record.set("decider", identity.name.clone());
278        record.set("outcome", decision.outcome.code());
279        record.set("reason", decision.reason.clone());
280        record.set("at", at as f64);
281        request.set("state", next.clone());
282        request.set("version", (decision.version + 1) as f64);
283        let payload = format!(
284            "request={};step={};outcome={};state={next}",
285            decision.request,
286            step.code,
287            decision.outcome.code()
288        );
289        let event = Event::new(organization, "request.decided", format!("decision/{}", decision.id), payload);
290        let task_key = Value::Text(format!("{}/{}", decision.request, step.code));
291        let mut task = space.find(TASK, &task_key)?.ok_or_else(not_there)?;
292        task.set("state", "done");
293        task.set("decided_by", identity.name.clone());
294        let following = match decision.outcome {
295            Outcome::Approve => self.steps.get(index + 1),
296            Outcome::Reject => None,
297        };
298        let written = space
299            .insert(record)
300            .and_then(|()| space.save(task))
301            .and_then(|()| following.map_or(Ok(()), |next| open_task(space, &request, next, at)))
302            .and_then(|()| space.save(request))
303            .and_then(|()| space.emit(&event).map(drop));
304        if let Err(error) = written {
305            space.rollback();
306            return Err(error);
307        }
308        match commit_or_rollback(space) {
309            Ok(()) => Ok(Decided::Recorded { state: next }),
310            Err(SpaceError::DuplicateKey { entity, .. }) if entity == DECISION => {
311                Ok(Decided::AlreadyRecorded)
312            }
313            Err(error) => Err(error),
314        }
315    }
316}

318impl RequestsModule {
319    /// The roles this person decides with at this moment: their own, and those lent to them by a
320    /// delegation still running.
321    pub fn roles_of(
322        &self,
323        space: &dyn ObjectSpace,
324        identity: &Identity,
325        at: i64,
326    ) -> Result<Vec<String>, SpaceError> {
327        let mut roles: Vec<String> = identity.roles.iter().map(|role| role.name().to_owned()).collect();
328        let name = identity.name.replace('\'', "''");
329        let mine = Criteria::parse(&format!("[delegate] = '{name}' And [until] > {at}"))
330            .map_err(|e| SpaceError::BadQuery(e.to_string()))?;
331        for delegation in space.query(DELEGATION, &Query::all().filter(mine))? {
332            let role = text(&delegation.get("role"));
333            if !roles.contains(&role) {
334                roles.push(role);
335            }
336        }
337        Ok(roles)
338    }
339
340    /// Lends a role this person has to someone else until a moment, after which the other person
341    /// no longer decides with it. `id` makes it idempotent.
342    #[allow(clippy::too_many_arguments)]
343    pub fn delegate(
344        &self,
345        space: &mut dyn ObjectSpace,
346        identity: &Identity,
347        id: &str,
348        to: &str,
349        role: &str,
350        until: i64,
351        at: i64,
352    ) -> Result<(), SpaceError> {
353        if !identity.roles.iter().any(|own| own.name() == role) {
354            return Err(SpaceError::Denied(format!("Only a member of {role} lends it.")));
355        }
356        if until <= at {
357            return Err(SpaceError::Refused("A delegation has to end later than now.".to_owned()));
358        }
359        if space.find(DELEGATION, &Value::Text(id.to_owned()))?.is_some() {
360            return Ok(());
361        }
362        let mut delegation = space.new_object(DELEGATION)?;
363        delegation.set("id", id);
364        delegation.set("delegator", identity.name.clone());
365        delegation.set("delegate", to);
366        delegation.set("role", role);
367        delegation.set("until", until as f64);
368        space.insert(delegation)?;
369        commit_or_rollback(space)
370    }
371
372    /// Roles that let this person read the tasks of the roles lent to them and still running, to
373    /// add to their identity when they sign in. Deciding checks the end of each delegation again,
374    /// so a role kept past it reads nothing new and decides nothing.
375    pub fn delegated_roles(
376        &self,
377        space: &dyn ObjectSpace,
378        identity: &Identity,
379        at: i64,
380    ) -> Result<Vec<Role>, SpaceError> {
381        let own: Vec<&str> = identity.roles.iter().map(Role::name).collect();
382        Ok(self
383            .roles_of(space, identity, at)?
384            .into_iter()
385            .filter(|role| !own.contains(&role.as_str()))
386            .map(|role| Role::new(format!("{role} (lent)")).reads_tasks_of(&role))
387            .collect())
388    }
389
390    /// The open tasks this person decides at this moment, by their roles and the delegations
391    /// still running.
392    pub fn tasks_for(
393        &self,
394        space: &dyn ObjectSpace,
395        identity: &Identity,
396        at: i64,
397    ) -> Result<Vec<Object>, SpaceError> {
398        let roles = self.roles_of(space, identity, at)?;
399        let open = Criteria::parse("[state] = 'open'").map_err(|e| SpaceError::BadQuery(e.to_string()))?;
400        let tasks = space.query(TASK, &Query::all().filter(open).sort("id", true))?;
401        Ok(tasks.into_iter().filter(|task| roles.contains(&text(&task.get("role")))).collect())
402    }
403}

405/// Lets members of a role read the tasks of the steps that role decides, so the generated list of
406/// tasks is their inbox. Give it to each role that decides a step.
407pub fn task_reader(role: Role) -> Role {
408    let name = role.name().to_owned();
409    role.reads_tasks_of(&name)
410}
411
412/// Reading the tasks of a role, added to another role.
413trait ReadsTasks {
414    fn reads_tasks_of(self, role: &str) -> Role;
415}
416
417impl ReadsTasks for Role {
418    fn reads_tasks_of(self, role: &str) -> Role {
419        let name = role.replace('\'', "''");
420        self.object_permission(TASK, &format!("[role] = '{name}'"), [Operation::Read], Access::Allow)
421    }
422}
423
424fn open_task(space: &mut dyn ObjectSpace, request: &Object, step: &Step, at: i64) -> Result<(), SpaceError> {
425    let key = match request.get("id") {
426        Value::Number(n) => n as i64,
427        _ => 0,
428    };
429    let mut task = space.new_object(TASK)?;
430    task.set("id", format!("{key}/{}", step.code));
431    task.set("title", request.get("title"));
432    task.set("request", key as f64);
433    task.set("step", step.code.clone());
434    task.set("role", step.decided_by.clone());
435    task.set("state", "open");
436    if let Some(due_in) = step.due_in {
437        task.set("due", (at + due_in) as f64);
438    }
439    space.insert(task)
440}

475/// What a decision says.
476#[derive(Clone, Copy, Debug, PartialEq, Eq)]
477pub enum Outcome {
478    /// The request moves to the next step, or is approved after the last one.
479    Approve,
480    /// The request stops, rejected.
481    Reject,
482}
483
484impl Outcome {
485    fn code(self) -> &'static str {
486        match self {
487            Self::Approve => "approve",
488            Self::Reject => "reject",
489        }
490    }
491}

493/// A decision someone sends.

495pub struct Decision {

497    pub id: String,
498    /// The key of the request.
499    pub request: i64,
500    /// The version of the request the person saw when deciding.
501    pub version: i64,
502    /// Approve or reject.

504    /// Why, in their words.
505    pub reason: String,
506}
