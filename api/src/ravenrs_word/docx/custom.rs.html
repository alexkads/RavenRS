1//! Custom document properties (`docProps/custom.xml`, ECMA-376 Part 1 §22.3). Citation
2//! managers keep their document preferences here (Zotero: `ZOTERO_PREF_1`…), so they must
3//! survive a round trip intact.
4
5use crate::doc::CustomProp;
6
7use crate::docx::xml::{El, MAX_DEPTH, Node, W, esc};
8
9/// The format identifier every custom property carries (§22.3.2.2).
10const FMTID: &str = "{D5CDD505-2E9C-101B-9397-08002B2CF9AE}";
11/// Most properties we keep from one file.
12const MAX_PROPS: usize = 10_000;
13
14/// Simple variant types kept as text.
15const SIMPLE: &[&str] = &[
16    "lpwstr", "lpstr", "bstr", "i1", "i2", "i4", "i8", "int", "ui1", "ui2", "ui4", "ui8", "uint", "r4", "r8", "decimal", "bool", "date", "filetime",
17    "cy", "error", "clsid",
18];
19
20/// Properties of a parsed `docProps/custom.xml` root.
21pub fn read(root: &El) -> Vec<CustomProp> {
22    let mut out = Vec::new();
23    for p in root.els().filter(|e| e.local() == "property").take(MAX_PROPS) {
24        let Some(name) = p.attr("name").filter(|n| !n.is_empty()) else { continue };
25        let Some(v) = p.els().next() else { continue };
26        let kind = v.local();
27        let prop = if SIMPLE.contains(&kind) {
28            CustomProp {
29                name: name.to_string(),
30                kind: kind.to_string(),
31                value: v.text(),
32                link_target: p.attr("linkTarget").map(str::to_string),
33                odf_user_field: false,
34            }
35        } else {
36            let mut xml = String::new();
37            el_xml(v, &mut xml, 0);
38            CustomProp {
39                name: name.to_string(),
40                kind: "raw".into(),
41                value: xml,
42                link_target: p.attr("linkTarget").map(str::to_string),
43                odf_user_field: false,
44            }
45        };
46        out.push(prop);
47    }
48    out
49}

51/// `docProps/custom.xml` for `props`.
52pub fn write(props: &[CustomProp]) -> Vec<u8> {
53    let mut w = W::new();
54    w.open(
55        "Properties",
56        &[
57            ("xmlns", "http://schemas.openxmlformats.org/officeDocument/2006/custom-properties"),
58            ("xmlns:vt", "http://schemas.openxmlformats.org/officeDocument/2006/docPropsVTypes"),
59        ],
60    );
61    // Property ids start at 2 (§22.3.2.2).
62    for (pid, p) in (2u32..).zip(props.iter().filter(|p| !p.name.is_empty())) {
63        if p.kind == "raw" && !raw_is_wellformed(&p.value) {
64            continue;
65        }
66        let pid = pid.to_string();
67        let mut attrs = vec![("fmtid", FMTID), ("pid", &pid), ("name", &p.name)];
68        if let Some(target) = &p.link_target {
69            attrs.push(("linkTarget", target));
70        }
71        w.open("property", &attrs);
72        if p.kind == "raw" {
73            w.raw(&p.value);
74        } else {
75            let kind = if SIMPLE.contains(&p.kind.as_str()) { p.kind.as_str() } else { "lpwstr" };
76            w.leaf(&format!("vt:{kind}"), &[], &p.value);
77        }
78        w.close("property");
79    }
80    w.close("Properties");
81    w.into_bytes()
82}

84/// A raw value is written only if it parses inside a `vt`-namespaced wrapper.
85fn raw_is_wellformed(xml: &str) -> bool {
86    let doc = format!(r#"<x xmlns:vt="http://schemas.openxmlformats.org/officeDocument/2006/docPropsVTypes">{xml}</x>"#);
87    crate::docx::xml::parse(doc.as_bytes()).is_ok()
88}

90/// Serialise an element with its canonical prefixes.
91fn el_xml(e: &El, out: &mut String, depth: usize) {
92    if depth > MAX_DEPTH {
93        return;
94    }
95    out.push('<');
96    out.push_str(&e.name);
97    for (k, v) in &e.attrs {
98        if k.starts_with("xmlns") {
99            continue;
100        }
101        out.push(' ');
102        out.push_str(k);
103        out.push_str("=\"");
104        out.push_str(&esc(v));
105        out.push('"');
106    }
107    if e.kids.is_empty() {
108        out.push_str("/>");
109        return;
110    }
111    out.push('>');
112    for k in &e.kids {
113        match k {
114            Node::El(c) => el_xml(c, out, depth + 1),
115            Node::Text(t) => out.push_str(&esc(t)),
116        }
117    }
118    out.push_str("</");
119    out.push_str(&e.name);
120    out.push('>');
121}
