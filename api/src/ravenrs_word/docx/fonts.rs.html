1//! Fonts a file embeds through its font table (#1171).
2//!
3//! The reader keeps the font table and its font parts as opaque passthrough parts, for writing
4//! back. [`embedded_fonts`] hands their font files to whoever lays the document out, undoing the
5//! obfuscation Word applies to them (ECMA-376 Part 1, 17.8.1).
6
7use std::collections::HashSet;
8
9use crate::doc::Document;
10
11use crate::docx::package::{EMBEDDED_PARTS, EmbeddedManifest};
12
13/// Most font files taken from one document.
14const MAX_FONTS: usize = 64;
15
16/// Most bytes of font files copied out of one document: one part's limit, however many
17/// references the font table makes (each reference is a copy; 64 of them to one large part
18/// were 64 copies, #3692).
19const MAX_FONT_BYTES: u64 = crate::docx::package::MAX_PART;
20
21/// The font table's references to embedded font files, one per style.
22const EMBEDS: [&str; 4] = ["w:embedRegular", "w:embedBold", "w:embedItalic", "w:embedBoldItalic"];
23
24/// A font file a document embeds, and the name its font table gives the font: the name the text
25/// asks for, which a subset Word embeds doesn't carry inside (#2240).
26#[derive(Clone, Debug, PartialEq)]
27pub struct EmbeddedFont {
28    /// `w:font/@w:name`.
29    pub name: String,
30    /// The font file, de-obfuscated.
31    pub data: Vec<u8>,
32}
33
34/// The font files `doc`'s font table embeds, de-obfuscated, in table order. Not checked to be
35/// fonts: that is up to the font database. Empty for a document read from anything but a package
36/// with a font table.
37pub fn embedded_fonts(doc: &Document) -> Vec<EmbeddedFont> {
38    fonts_within(doc, MAX_FONT_BYTES)
39}
40
41/// [`embedded_fonts`], copying at most `budget` bytes of font files.
42fn fonts_within(doc: &Document, mut budget: u64) -> Vec<EmbeddedFont> {
43    let budget_total = budget;
44    let manifest = doc.passthrough.get(EMBEDDED_PARTS).map(|m| EmbeddedManifest::parse(m)).unwrap_or_default();
45    // The table is kept under the spelling the package stores it as (#3872).
46    let Some(key) = crate::docx::package::font_table_key(&doc.passthrough) else {
47        return Vec::new();
48    };
49    let Some(table) = doc.passthrough.get(key).and_then(|b| crate::docx::xml::parse(b).ok()) else {
50        return Vec::new();
51    };
52    let Some(part) = manifest.parts.get(key) else {
53        return Vec::new();
54    };
55    let mut out = Vec::new();
56    // A reference repeating one already taken (same file, key and name) gives the same font.
57    let mut taken: HashSet<(&str, [u8; 16], &str)> = HashSet::new();
58    let mut over_budget = 0usize;
59    let embeds = table
60        .children("w:font")
61        .flat_map(|f| f.els().filter(|e| EMBEDS.contains(&e.name.as_str())).map(move |e| (f.attr("w:name").unwrap_or(""), e)));
62    for (name, embed) in embeds {
63        if out.len() >= MAX_FONTS {
64            log::warn!("docx: more than {MAX_FONTS} embedded fonts; the rest are not loaded");
65            break;
66        }
67        let Some(rel) = embed.attr("r:id").and_then(|id| part.rels.iter().find(|r| r.id == id && !r.external)) else {
68            continue;
69        };
70        let Some(bytes) = doc.passthrough.get(&rel.target) else {
71            continue;
72        };
73        // No key: the file is stored as is, as with the all-zero key.
74        let key = match embed.attr("w:fontKey") {
75            Some(k) => match font_key(k) {
76                Some(key) => key,
77                None => {
78                    log::warn!("docx: embedded font {} has an unreadable key {k:?}", rel.target);
79                    continue;
80                }
81            },
82            None => [0; 16],
83        };
84        if !taken.insert((rel.target.as_str(), key, name)) {
85            continue;
86        }
87        // Checked before the copy is made.
88        let Some(left) = budget.checked_sub(bytes.len() as u64) else {
89            over_budget += 1;
90            continue;
91        };
92        budget = left;
93        let mut data = bytes.to_vec();
94        for (i, b) in data.iter_mut().take(32).enumerate() {
95            *b ^= key.get(i % 16).copied().unwrap_or(0);
96        }
97        out.push(EmbeddedFont { name: name.to_string(), data });
98    }
99    if over_budget > 0 {
100        // Once for the document, not once per font.
101        log::warn!("docx: embedded fonts past the {budget_total} byte budget; {over_budget} of them are not loaded");
102    }
103    out
104}

106/// The obfuscation key of a `w:fontKey` GUID (`{XXXXXXXX-XXXX-…}`): its 16 bytes, reversed.
107fn font_key(guid: &str) -> Option<[u8; 16]> {
108    let digits: Vec<u8> = guid.bytes().filter(|c| !matches!(c, b'{' | b'}' | b'-')).collect();
109    if digits.len() != 32 {
110        return None;
111    }
112    let mut key = [0u8; 16];
113    for (i, pair) in digits.chunks(2).enumerate() {
114        let byte = u8::from_str_radix(std::str::from_utf8(pair).ok()?, 16).ok()?;
115        *key.get_mut(15 - i)? = byte;
116    }
117    Some(key)
118}

120#[cfg(test)]
121mod tests {
122    use std::sync::Arc;
123
124    use super::*;
125    use crate::docx::package::FONT_TABLE_PART;
126
127    /// A document whose font table makes `refs` references (`(font name, part)`) to font parts
128    /// `parts` (`(name, size)`).
129    fn doc_with(refs: &[(&str, &str)], parts: &[(&str, usize)]) -> Document {
130        let mut table = String::from(
131            r#"<w:fonts xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">"#,
132        );
133        let mut manifest = format!("P\t{FONT_TABLE_PART}\tapplication/xml\n");
134        for (i, (font, part)) in refs.iter().enumerate() {
135            table.push_str(&format!(r#"<w:font w:name="{font}"><w:embedRegular r:id="rId{i}"/></w:font>"#));
136            manifest.push_str(&format!("R\t{FONT_TABLE_PART}\trId{i}\tfont\t{part}\tI\n"));
137        }
138        table.push_str("</w:fonts>");
139        let mut doc = Document::default();
140        doc.passthrough.insert(FONT_TABLE_PART.into(), Arc::new(table.into_bytes()));
141        doc.passthrough.insert(EMBEDDED_PARTS.into(), Arc::new(manifest.into_bytes()));
142        for (part, size) in parts {
143            doc.passthrough.insert((*part).into(), Arc::new(vec![7u8; *size]));
144        }
145        doc
146    }
147
148    #[test]
149    fn a_font_referenced_again_is_copied_once() {
150        // Each reference was its own copy: the 64 references allowed to one 256 MB part were
151        // 16 GB of copies (#3692).
152        let refs = vec![("Body", "word/fonts/a.odttf"); MAX_FONTS];
153        let fonts = embedded_fonts(&doc_with(&refs, &[("word/fonts/a.odttf", 1000)]));
154        assert_eq!(fonts.len(), 1);
155    }
156
157    #[test]
158    fn font_copies_stop_at_the_byte_budget() {
159        let refs: Vec<(String, &str)> = (0..MAX_FONTS).map(|i| (format!("Font {i}"), "word/fonts/a.odttf")).collect();
160        let refs: Vec<(&str, &str)> = refs.iter().map(|(n, p)| (n.as_str(), *p)).collect();
161        let doc = doc_with(&refs, &[("word/fonts/a.odttf", 1000)]);
162        let (fonts, warned) = crate::docx::test_log::warnings(|| fonts_within(&doc, 3500));
163        assert_eq!(fonts.len(), 3, "three copies fit");
164        let budget: Vec<_> = warned.iter().filter(|w| w.contains("byte budget")).collect();
165        assert_eq!(budget.len(), 1, "one warning for the document, not one per font: {warned:?}");
166        assert!(budget.iter().all(|w| w.contains(&format!("{} of them", MAX_FONTS - 3))), "{budget:?}");
167        assert_eq!(embedded_fonts(&doc).len(), MAX_FONTS, "well within the real budget");
168    }
