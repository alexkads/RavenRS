131/// Read a `.docx` package.
132pub fn read(bytes: &[u8]) -> Result<Document, DocxError> {
133    // A password-protected document is a compound file, not a zip: say so rather than failing
134    // as a broken zip (`read_with_password` opens it).
135    if crate::docx::is_encrypted(bytes) {
136        return crate::docx::read_with_password(bytes, None);
137    }
138    read_package(bytes, None)
139}

141/// [`read`] for bytes known not to be an encrypted package; `chunks` imports alternative format
142/// chunks in formats other than WordprocessingML.
143pub(crate) fn read_package(bytes: &[u8], chunks: Option<ChunkImporter>) -> Result<Document, DocxError> {
144    read_opened(&Package::open(bytes)?, chunks)
145}

147/// Read a package whose parts are already unpacked (from a zip, a Flat OPC file or a converted
148/// Word 2003 XML document).
149pub(crate) fn read_opened(pkg: &Package, chunks: Option<ChunkImporter>) -> Result<Document, DocxError> {
150    read_opened_in(pkg, ChunkState::new(chunks)).map(|(doc, ..)| doc)
151}

153/// [`read_opened`] with what chunks may still bring in; also gives the chunks, chunk bytes and
154/// graphic work left of that.
155fn read_opened_in(pkg: &Package, chunks: ChunkState) -> Result<(Document, usize, u64, usize), DocxError> {
156    let root_rels = pkg.rels("");
157    let main = root_rels.by_type(rt::OFFICE_DOC).filter(|r| !r.external).map(|r| r.target.clone()).unwrap_or_else(|| "word/document.xml".to_string());
158    let Some(root) = pkg.xml_strict(&main)? else {
159        return Err(DocxError::MissingPart(main));
160    };
161    if root.name != "w:document" {
162        return Err(DocxError::NotWord(format!("root element is {}", root.name)));
163    }
164    let rels = pkg.rels(&main);
165    let mut doc = Document::new();
166    doc.body.clear();
167    let mut r = Reader {
168        pkg,
169        doc,
170        pc: PropCtx::default(),
171        revs: HashMap::new(),
172        moves: MoveCtx::default(),
173        media_by_path: HashMap::new(),
174        hf_by_path: HashMap::new(),
175        hf_depth: 0,
176        footnotes: HashMap::new(),
177        endnotes: HashMap::new(),
178        current_note: None,
179        in_note_sep: false,
180        comment_map: HashMap::new(),
181        comments_ended: HashSet::new(),
182        bookmarks: HashMap::new(),
183        perms: HashMap::new(),
184        next_perm: 0,
185        graphic_parts: HashMap::new(),
186        graphics: HashMap::new(),
187        glued: HashSet::new(),
188        shape_ids: false,
189        main: main.clone(),
190        story: main.as_str().into(),
191        content_types: None,
192        embedded: EmbeddedManifest::default(),
193        txbx_links: Vec::new(),
194        theme_fills: Default::default(),
195        group_fill: None,
196        chunks,
197    };
198    r.doc.missing_pictures.main = main.clone();
199    r.pc.major_font = r.doc.settings.major_font.clone();
200    r.pc.minor_font = r.doc.settings.minor_font.clone();
201    let part = |t: &str| rels.by_type(t).filter(|x| !x.external).map(|x| x.target.clone());
202
203    // The font table and the fonts embedded through it ride along as opaque parts.
204    if let Some(p) = part(rt::FONT_TABLE) {
205        r.keep_part_tree(&p);
206        r.read_font_families(&p);
207    }
208    if let Some(p) = part(rt::THEME) {
209        r.read_theme(&p);
210    }
211    // Secondary parts are optional: a broken one is skipped (logged), not fatal.
212    let lenient = |what: &str, res: Result<(), DocxError>| {
213        if let Err(e) = res {
214            log::warn!("docx: ignoring unreadable {what}: {e}");
215        }
216    };
217    // Styles come before settings, but their theme colours already follow its colour mapping, and
218    // their `…Bidi` theme fonts its theme font language.
219    if let Some(p) = part(rt::SETTINGS) {
220        r.read_color_mapping(&p);
221    }
222    if let Some(p) = part(rt::STYLES) {
223        lenient("styles", r.read_styles(&p));
224    }
225    if let Some(p) = part(rt::NUMBERING) {
226        lenient("numbering", r.read_numbering(&p));
227    }
228    // Without a stated mode (or settings at all) Word lays a document out as Word 2007 did.
229    r.doc.settings.compat_mode = crate::doc::LEGACY_COMPAT_MODE;
230    if let Some(p) = part(rt::SETTINGS) {
231        lenient("settings", r.read_settings(&p));
232    }
233    // Comments before notes: a comment range inside a footnote or endnote resolves its id against
234    // the comments already read, and was dropped when the notes came first.
235    if let Some(p) = part(rt::COMMENTS) {
236        let ex = part(rt::COMMENTS_EX);
237        let ids = part(rt::COMMENTS_IDS);
238        let cex = part(rt::COMMENTS_EXTENSIBLE);
239        r.story = p.as_str().into();
240        lenient("comments", r.read_comments(&p, ex.as_deref(), ids.as_deref(), cex.as_deref()));
241    }
242    if let Some(p) = part(rt::FOOTNOTES) {
243        r.story = p.as_str().into();
244        lenient("footnotes", r.read_notes(&p, true));
245    }
246    if let Some(p) = part(rt::ENDNOTES) {
247        r.story = p.as_str().into();
248        lenient("endnotes", r.read_notes(&p, false));
249    }
250    r.story = main.as_str().into();
251    // Bibliography sources live in a custom XML data store (`b:Sources`), related from the
252    // package root or the main part; Word's own files sometimes leave the relationship out, so
253    // the conventional `customXml/itemN.xml` names are tried too. Each part is read at most once.
254    let mut bib_parts: Vec<String> = Vec::new();
255    for rel in root_rels.list.iter().chain(rels.list.iter()) {
256        if rel_is(&rel.kind, rt::CUSTOM_XML) && !rel.external && !bib_parts.contains(&rel.target) {
257            bib_parts.push(rel.target.clone());
258            if bib_parts.len() >= MAX_BIBLIOGRAPHY_PARTS {
259                break;
260            }
261        }
262    }
263    for i in 1..=16 {
264        let path = format!("customXml/item{i}.xml");
265        if bib_parts.len() < MAX_BIBLIOGRAPHY_PARTS && pkg.get(&path).is_some() && !bib_parts.contains(&path) {
266            bib_parts.push(path);
267        }
268    }
269    let mut seen_sources = Seen::default();
270    for path in &bib_parts {
271        if r.doc.sources.len() >= MAX_SOURCES {
272            break;
273        }
274        lenient("customXml bibliography", r.read_bibliography(path, &mut seen_sources));
275    }
276
277    if let Some(bg) = root.child("w:background").and_then(|b| b.attr("w:color")).and_then(Rgb::parse) {
278        r.doc.settings.page_color = Some(bg);
279    }
280    // A picture or texture fill: `w:background/v:background/v:fill` with a relationship id.
281    if let Some(fill) = root.child("w:background").and_then(|b| b.child("v:background")).and_then(|b| b.child("v:fill")) {
282        let rid = fill.attr("r:id").or_else(|| fill.attr("r:embed")).or_else(|| fill.attr("o:relid"));
283        if let Some(media) = rid.and_then(|id| r.media_for(&rels, id)) {
284            r.doc.settings.page_picture = Some(PagePicture {
285                media,
286                fill_type: fill.attr("type").unwrap_or("").chars().take(32).collect(),
287                title: fill.attr("o:title").unwrap_or("").chars().take(255).collect(),
288            });
289        }
290    }
291    if let Some(body) = root.child("w:body") {
292        let mut sc = StoryCtx::default();
293        let mut blocks = Blocks::new();
294        r.read_blocks(&mut sc, body, &rels, &mut blocks, r.chunks.depth);
295        r.flush_pending(&mut sc, &mut blocks);
296        r.doc.body = blocks;
297        unstyle_toc_links(&mut r.doc);
298        toc_control_to_toc(&mut r.doc.body);
299        toc_field_to_heading(&mut r.doc.body);
300        if let Some(s) = body.child("w:sectPr") {
301            r.doc.last_section = r.read_section(s, &rels);
302        }
303    }
304    // A watermark in the box Word's Design › Watermark gives it for the page keeps no box of its
305    // own: it follows the page, as when it was made.
306    let (pw, ph) = (r.doc.last_section.page_w, r.doc.last_section.page_h);
307    if let Some(wm) = r.doc.settings.watermark.as_mut()
308        && let Some([w, h]) = wm.size
309    {
310        let [dw, dh] = crate::doc::Watermark { size: None, ..wm.clone() }.frame(pw, ph);
311        if (w - dw).abs() < 0.06 && (h - dh).abs() < 0.06 {
312            wm.size = None;
313        }
314    }
315
316    let core = root_rels.by_type(rt::CORE).map(|x| x.target.clone()).unwrap_or_else(|| "docProps/core.xml".into());
317    r.read_core(&core);
318    if let Some(app) = root_rels.by_type(rt::APP).map(|x| x.target.clone()) {
319        r.read_app(&app);
320    }
321    if let Some(c) = root_rels.by_type(rt::CUSTOM)
322        && let Ok(Some(x)) = r.xml(&c.target)
323    {
324        r.doc.custom_props = crate::docx::custom::read(&x);
325    }
326    // A macro project and every part it relates to (VBA data, signatures…) ride along as opaque
327    // bytes, never parsed or run, so a .docm/.dotm saved again carries exactly what the file had.
328    // Only the relationships and content types are read.
329    if let Some(v) = part(rt::VBA_PROJECT)
330        && let Some(b) = pkg.get_shared(&v).filter(|b| !b.is_empty())
331    {
332        r.doc.passthrough.insert(VBA_PROJECT_PART.into(), b.clone());
333        r.read_vba_related(&v);
334    }
335    r.read_glossary(&rels);
336    r.keep_custom_xml(&root_rels, &rels);
337    r.keep_unmodelled(&root, &root_rels, &rels);
338    if !r.embedded.parts.is_empty() {
339        let manifest = r.embedded.to_bytes();
340        r.doc.passthrough.insert(EMBEDDED_PARTS.into(), Arc::new(manifest));
341    }
342    r.finish_moves();
343    let links = std::mem::take(&mut r.txbx_links);
344    link_text_boxes(&mut r.doc, links);
345    if r.shape_ids {
346        keep_glued_ids(&mut r.doc, &r.glued);
347    }
348    let (left, bytes, graphic) = (r.chunks.left, r.chunks.bytes, r.chunks.graphic);
349    let mut doc = r.doc;
350    doc.ensure_nonempty();
351    Ok((doc, left, bytes, graphic))
352}

354/// Most move ranges open at once, and most move names tracked, per file.
355const MAX_OPEN_MOVES: usize = 64;
356const MAX_MOVES: usize = 100_000;
357
358/// The move ranges of a file (ECMA-376 §17.13.5.22–27): a `w:moveFrom` / `w:moveTo` is one end
359/// of a move when it sits inside a range of that kind; the range's name links both ends.
360#[derive(Default)]
361pub(crate) struct MoveCtx {
362    /// Open source and destination ranges: (`w:id`, `w:name`).
363    open: [Vec<(String, String)>; 2],
364    /// Per name: a source range started / ended, a destination range started / ended (bits 0–3).
365    seen: HashMap<String, u8>,
366    /// The range of each kind that ended last in the paragraph being read (for its mark).
367    pub closed_here: [Option<String>; 2],
368}
369
370impl Reader<'_> {
371    /// A `w:moveFromRangeStart/End` or `w:moveToRangeStart/End`.
372    pub fn move_range(&mut self, e: &El) {
373        let (side, start) = match e.name.as_str() {
374            "w:moveFromRangeStart" => (0, true),
375            "w:moveFromRangeEnd" => (0, false),
376            "w:moveToRangeStart" => (1, true),
377            "w:moveToRangeEnd" => (1, false),
378            _ => return,
379        };
380        let id = e.attr("w:id").unwrap_or("").chars().take(64).collect::<String>();
381        let m = &mut self.moves;
382        let Some(open) = m.open.get_mut(side) else { return };
383        if start {
384            let name = e.attr("w:name").unwrap_or("").chars().take(256).collect::<String>();
385            if name.is_empty() || open.len() >= MAX_OPEN_MOVES || (m.seen.len() >= MAX_MOVES && !m.seen.contains_key(&name)) {
386                return;
387            }
388            *m.seen.entry(name.clone()).or_default() |= 1 << (side * 2);
389            open.push((id, name));
390        } else if let Some(i) = open.iter().rposition(|(x, _)| *x == id) {
391            let (_, name) = open.remove(i);
392            *m.seen.entry(name.clone()).or_default() |= 2 << (side * 2);
393            if let Some(c) = m.closed_here.get_mut(side) {
394                *c = Some(name);
395            }
396        }
397    }
398
399    /// The revision of a `w:moveFrom` (`to` false) or `w:moveTo`: one end of the innermost open
400    /// range of its kind, else (no range) a plain deletion or insertion.
401    pub fn move_revision(&mut self, to: bool, e: &El, mark: bool) -> u32 {
402        let side = usize::from(to);
403        let open = self.moves.open.get(side).and_then(|o| o.last()).map(|(_, n)| n.clone());
404        // A paragraph mark comes after its paragraph's text: a range that ended in it counts.
405        let name = if mark { open.or_else(|| self.moves.closed_here.get(side).cloned().flatten()) } else { open };
406        let kind = if to { RevisionKind::Insert } else { RevisionKind::Delete };
407        self.revision_named(kind, e, name)
408    }
409
410    /// Moves missing an end (a source or destination range, or its text) are plain deletions
411    /// and insertions.
412    fn finish_moves(&mut self) {
413        let mut ends: HashMap<&str, u8> = HashMap::new();
414        for r in &self.doc.revisions {
415            if let Some(n) = r.move_name.as_deref() {
416                *ends.entry(n).or_default() |= if r.kind == RevisionKind::Delete { 1 } else { 2 };
417            }
418        }
419        let ok: HashSet<String> =
420            ends.into_iter().filter(|(n, e)| *e == 3 && self.moves.seen.get(*n) == Some(&0b1111)).map(|(n, _)| n.to_string()).collect();
421        for r in &mut self.doc.revisions {
422            if r.move_name.as_ref().is_some_and(|n| !ok.contains(n)) {
423                r.move_name = None;
424            }
425        }
426    }
427}

429/// Link the text boxes read as (id, seq, part) into chains (see [`Reader::txbx_links`]): each
430/// id's boxes in `seq` order after the box holding the text. A repeated seq, a chain without its
431/// first box or more than [`crate::doc::MAX_LINKED_BOXES`] boxes: the extra boxes stay apart.
432fn link_text_boxes(doc: &mut Document, mut links: Vec<(u32, u32, u32)>) {
433    links.sort_by_key(|l| (l.0, l.1));
434    links.dedup_by_key(|l| (l.0, l.1));
435    for group in links.chunk_by(|a, b| a.0 == b.0) {
436        if group.first().is_none_or(|l| l.1 != 0) {
437            continue;
438        }
439        let chain: Vec<u32> = group.iter().take(crate::doc::MAX_LINKED_BOXES).map(|l| l.2).collect();
440        for pair in chain.windows(2) {
441            if let [a, b] = pair
442                && let Some(p) = doc.parts.get_mut(a)
443            {
444                p.body.next = Some(*b);
445            }
446        }
447    }
448}

450/// Drop the drawing ids read with shapes, except those connectors are glued to (so a shape
451/// without connectors reads back as it was written).
452fn keep_glued_ids(doc: &mut Document, glued: &HashSet<u32>) {
453    let has_id = |o: &InlineObject| match o {
454        InlineObject::Shape { extra, .. } => extra.id != 0,
455        InlineObject::Group { children, .. } => children.iter().any(|c| matches!(&c.obj, InlineObject::Shape { extra, .. } if extra.id != 0)),
456        _ => false,
457    };
458    let clear = |o: &mut InlineObject| {
459        if let InlineObject::Shape { extra, .. } = o
460            && !glued.contains(&extra.id)
461        {
462            extra.id = 0;
463        }
464    };
465    doc.objects_mut(&has_id, &mut |o| match o {
466        InlineObject::Group { children, .. } => children.iter_mut().for_each(|c| clear(&mut c.obj)),
467        o => clear(o),
468    });
