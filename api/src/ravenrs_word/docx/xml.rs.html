1//! A small, bounded XML tree on top of `quick-xml`, plus a string-building XML writer.
2//!
3//! Element and attribute names are rewritten to *canonical prefixes* by namespace URI (`w:p`,
4//! `r:id`, `a:blip`…), so the reader doesn't care which prefixes a producer chose. Names in
5//! unknown namespaces keep their local name behind a `?:` prefix.
6

85fn prefix_for(uri: &[u8], extra: &[(&'static str, &str)]) -> &'static str {
86    if let Some((p, _)) = extra.iter().find(|(_, u)| u.as_bytes() == uri) {
87        return p;
88    }
89    for (p, u) in NAMESPACES.iter().chain(OTHER_NAMESPACES).chain(STRICT) {
90        if u.as_bytes() == uri {
91            return p;
92        }
93    }
94    "?"
95}

97/// Word's 2006 namespace (`wne`). A Strict document from Word keeps a text box's content in
98/// `wne:txbxContent`; a Transitional one uses `w:txbxContent`.
99const WNE: &[u8] = b"http://schemas.microsoft.com/office/word/2006/wordml";
100
101/// The canonical prefix of an element named `local` in namespace `uri`, inside an element
102/// named `parent` (canonical, `None` at the root). This is [`prefix_for`] with two exceptions
103/// for Strict documents (#3548):
104///
105/// - Shapes, groups and canvases (`wsp`, `wgp`, `wpc`) and their parts are elements of the
106///   WordprocessingDrawing namespace in ECMA-376 Part 1, 5th edition (`dml-wordprocessingDrawing.xsd`,
107///   Strict and Transitional alike). A Strict document from Word puts them there; a Transitional
108///   one uses Word 2010's own namespaces, whose names the reader knows. So they read as `wps:`,
109///   `wpg:` and `wpc:` (see [`drawing_prefix`]).
110/// - A shape's text box content reads as `w:txbxContent` from the drawing namespace (the
111///   schema's name) and from Word's `wne:txbxContent`.
112fn element_prefix(uri: &[u8], local: &str, parent: Option<&str>, extra: &[(&'static str, &str)]) -> &'static str {
113    match prefix_for(uri, extra) {
114        "wp" => drawing_prefix(local, parent),
115        "?" if uri == WNE && local == "txbxContent" && parent == Some("wps:txbx") => "w",
116        p => p,
117    }
118}
119
120/// The prefix for `local`, an element of the WordprocessingDrawing namespace inside `parent`.
121/// Word 2010's namespaces split what this one holds: a shape and its parts are `wps:`, a group
122/// (`wgp`, a nested `grpSp`) and a graphic frame in it are `wpg:`, and a canvas is `wpc:`. Its
123/// members are named for what they are. A text box's content is `w:txbxContent`, and anything
124/// else (`inline`, `anchor`, `docPr`…) stays `wp:`.
125fn drawing_prefix(local: &str, parent: Option<&str>) -> &'static str {
126    match (parent, local) {
127        (_, "wsp") => "wps",
128        (_, "wgp" | "grpSp" | "graphicFrame") => "wpg",
129        (_, "wpc") => "wpc",
130        (Some("wps:wsp"), "cNvPr" | "cNvSpPr" | "cNvCnPr" | "spPr" | "style" | "extLst" | "txbx" | "linkedTxbx" | "bodyPr") => "wps",
131        (Some("wps:txbx"), "txbxContent") => "w",
132        (Some("wps:txbx" | "wps:linkedTxbx"), "extLst") => "wps",
133        (Some("wpg:wgp" | "wpg:grpSp"), "cNvPr" | "cNvGrpSpPr" | "grpSpPr" | "extLst") => "wpg",
134        (Some("wpg:graphicFrame"), "cNvPr" | "cNvFrPr" | "xfrm" | "extLst") => "wpg",
135        (Some("wpc:wpc"), "bg" | "whole" | "extLst") => "wpc",
136        _ => "wp",
137    }
138}

140/// Compatibility feature namespaces whose elements are not otherwise handled by this tree.
141/// Keep the reader's existing feature set when canonicalizing `mc:Choice/@Requires` aliases.
142fn requirement_prefix(uri: &[u8]) -> &'static str {
143    let prefix = prefix_for(uri, &[]);
144    if prefix != "?" {
145        return prefix;
146    }
147    match uri {
148        b"http://schemas.microsoft.com/office/drawing/2010/main" => "a14",
149        b"http://schemas.microsoft.com/office/word/2015/wordml/symex" => "w16se",
150        b"http://schemas.microsoft.com/office/word/2016/wordml/cid" => "w16cid",
151        b"http://schemas.microsoft.com/office/word/2018/wordml" => "w16",
152        b"http://schemas.microsoft.com/office/word/2018/wordml/cex" => "w16cex",
153        b"http://schemas.microsoft.com/office/word/2020/wordml/sdtdatahash" => "w16sdtdh",
154        _ => "?",
155    }
156}

164/// Elements whose markup names a relationship: the id is in an attribute of the relationship
165/// namespace (`r:id`, `r:embed`, `r:link`, `r:pict`, `r:dm`…) or VML's `o:relid`, and — from a
166/// producer that left the prefix out — in the unqualified attribute of the same local name, the
167/// alias [`El::attr`] accepts. On any other element an unqualified `id` is the element's own
168/// (a drawing's `wp:docPr`, a picture's `pic:cNvPr`, a VML shape, a page background) and names
169/// no part.
170const REL_ELEMENTS: &[&str] = &[
171    "a:blip",
172    "a:hlinkClick",
173    "a:hlinkHover",
174    "a:audioFile",
175    "a:videoFile",
176    "c:chart",
177    "cx:chart",
178    "dgm:relIds",
179    "o:OLEObject",
180    "v:fill",
181    "v:imagedata",
182    "w:attachedTemplate",
183    "w:control",
184    "w:embedBold",
185    "w:embedBoldItalic",
186    "w:embedItalic",
187    "w:embedRegular",
188    "w:footerReference",
189    "w:headerReference",
190    "w:hyperlink",
191    "w:object",
192    "w:subDoc",
193    "w14:contentPart",
194];
195
196/// The local names a relationship attribute has, so an unqualified one is recognised as the
197/// relationship id it names.
198const REL_ALIASES: &[&str] = &["id", "embed", "link", "pict", "dm", "lo", "qs", "cs", "relid"];
199
200/// Is the attribute `k` of the element named `element` a relationship id?
201///
202/// The relationship namespace (`r:`) and VML's `o:relid` always are. An unqualified attribute
203/// is one only on an element that carries a relationship ([`REL_ELEMENTS`]) and only with the
204/// local name of a relationship attribute ([`REL_ALIASES`]) — the alias [`El::attr`] accepts.
205/// Everywhere else an unqualified `id` is the element's own and names no part.
206pub fn is_rel_attr(element: &str, k: &str) -> bool {
207    k.starts_with("r:") || k == "o:relid" || (REL_ELEMENTS.contains(&element) && REL_ALIASES.contains(&k))
208}

210/// An element with canonical-prefixed name and attributes.
211#[derive(Clone, Debug, PartialEq, Default)]
212pub struct El {
213    pub name: String,
214    pub attrs: Vec<(String, String)>,
215    pub kids: Vec<Node>,
216}
217
218impl El {
219    /// Attribute by canonical name (`w:val`); falls back to an unqualified attribute with the
220    /// same local name (some producers omit the prefix).
221    pub fn attr(&self, name: &str) -> Option<&str> {
222        if let Some((_, v)) = self.attrs.iter().find(|(k, _)| k == name) {
223            return Some(v);
224        }
225        let local = name.rsplit(':').next().unwrap_or(name);
226        if local != name {
227            return self.attrs.iter().find(|(k, _)| k == local).map(|(_, v)| v.as_str());
228        }
229        None
230    }
231    /// Child elements.
232    pub fn els(&self) -> impl Iterator<Item = &El> {
233        self.kids.iter().filter_map(|n| match n {
234            Node::El(e) => Some(e),
235            Node::Text(_) => None,
236        })
237    }
238    pub fn child(&self, name: &str) -> Option<&El> {
239        self.els().find(|e| e.name == name)
240    }
241    pub fn children<'a>(&'a self, name: &'a str) -> impl Iterator<Item = &'a El> + 'a {
242        self.els().filter(move |e| e.name == name)
243    }
244    /// `w:val` of a child.
245    pub fn child_val(&self, name: &str) -> Option<&str> {
246        self.child(name).and_then(|c| c.attr("w:val"))
247    }
248    /// Direct text content.
249    pub fn text(&self) -> String {
250        let mut s = String::new();
251        for n in &self.kids {
252            if let Node::Text(t) = n {
253                s.push_str(t);
254            }
255        }
256        s
257    }
258    /// All descendant text (depth-bounded by the parser).
259    pub fn deep_text(&self) -> String {
260        let mut s = String::new();
261        fn go(e: &El, s: &mut String, depth: usize) {
262            if depth > MAX_DEPTH {
263                return;
264            }
265            for n in &e.kids {
266                match n {
267                    Node::Text(t) => s.push_str(t),
268                    Node::El(c) => go(c, s, depth + 1),
269                }
270            }
271        }
272        go(self, &mut s, 0);
273        s
274    }
275    /// First descendant with this name (depth-first, bounded).
276    pub fn find(&self, name: &str) -> Option<&El> {
277        fn go<'a>(e: &'a El, name: &str, depth: usize) -> Option<&'a El> {
278            if depth > MAX_DEPTH {
279                return None;
280            }
281            for c in e.els() {
282                if c.name == name {
283                    return Some(c);
284                }
285                if let Some(f) = go(c, name, depth + 1) {
286                    return Some(f);
287                }
288            }
289            None
290        }
291        go(self, name, 0)
292    }
293    /// The element as XML with canonical prefixes (for re-emitting kept content). Elements and
294    /// attributes in unknown namespaces are dropped.
295    pub fn to_xml(&self) -> String {
296        fn go(e: &El, s: &mut String, depth: usize) {
297            if depth > MAX_DEPTH || e.name.starts_with("?:") {
298                return;
299            }
300            s.push('<');
301            s.push_str(&e.name);
302            for (k, v) in &e.attrs {
303                if k.starts_with("?:") {
304                    continue;
305                }
306                s.push(' ');
307                s.push_str(k);
308                s.push_str("=\"");
309                s.push_str(&esc(v));
310                s.push('"');
311            }
312            if e.kids.is_empty() {
313                s.push_str("/>");
314                return;
315            }
316            s.push('>');
317            for n in &e.kids {
318                match n {
319                    Node::Text(t) => s.push_str(&esc(t)),
320                    Node::El(c) => go(c, s, depth + 1),
321                }
322            }
323            s.push_str("</");
324            s.push_str(&e.name);
325            s.push('>');
326        }
327        let mut s = String::new();
328        go(self, &mut s, 0);
329        s
330    }
331    /// Local part of the name.
332    pub fn local(&self) -> &str {
333        self.name.rsplit(':').next().unwrap_or(&self.name)
334    }
335}

337/// UTF-16 LE/BE text with a byte-order mark as UTF-8 (`None` without one). Lossy: an unpaired
338/// surrogate becomes U+FFFD and a trailing odd byte is dropped. The result is at most 1.5× the input.
339pub(crate) fn utf16_to_utf8(bytes: &[u8]) -> Option<String> {
340    let (rest, le) = match bytes.get(..2) {
341        Some([0xFF, 0xFE]) => (bytes.get(2..)?, true),
342        Some([0xFE, 0xFF]) => (bytes.get(2..)?, false),
343        _ => return None,
344    };
345    let units = rest.as_chunks::<2>().0.iter().map(|c| if le { u16::from_le_bytes(*c) } else { u16::from_be_bytes(*c) });
346    Some(char::decode_utf16(units).map(|c| c.unwrap_or(char::REPLACEMENT_CHARACTER)).collect())
347}

354/// [`parse`] with `extra` namespaces (prefix, URI) mapped to canonical prefixes first, so a
355/// related vocabulary (WordprocessingML 2003) reads with the prefixes the reader knows.
356pub fn parse_ns(bytes: &[u8], extra: &[(&'static str, &str)]) -> Result<El, DocxError> {
357    parse_with(bytes, extra, false)

360/// [`parse`] for a part whose loss would be silent data loss (the main document): a mismatched end
361/// tag or an element left open at the end of the data is an error, not a cue to stop reading.
362pub fn parse_strict(bytes: &[u8]) -> Result<El, DocxError> {
363    parse_with(bytes, &[], true)
364}

366fn parse_with(bytes: &[u8], extra: &[(&'static str, &str)], strict: bool) -> Result<El, DocxError> {
367    // UTF-16 (with a BOM) is transcoded to UTF-8; a UTF-8 BOM is stripped.
368    let transcoded = utf16_to_utf8(bytes);
369    let bytes = match &transcoded {
370        Some(s) => s.as_bytes(),
371        None => bytes.strip_prefix(&[0xEF, 0xBB, 0xBF][..]).unwrap_or(bytes),
372    };
373    let mut r = NsReader::from_reader(bytes);
374    {
375        let c = r.config_mut();
376        c.trim_text_start = false;
377        c.trim_text_end = false;
378        c.expand_empty_elements = false;
379        c.check_end_names = strict;
380    }
381    let mut stack: Vec<El> = Vec::new();
382    let mut root: Option<El> = None;
383    let mut count = 0usize;
384    let mut buf = Vec::new();
385    loop {
386        buf.clear();
387        let (res, ev) = r.read_resolved_event_into(&mut buf).map_err(|e| DocxError::Xml(e.to_string()))?;
388        let el_ns = match &res {
389            ResolveResult::Bound(ns) => Some(ns.as_ref()),
390            _ => None,
391        };
392        match ev {
393            Event::Start(ref e) | Event::Empty(ref e) => {
394                if strict && root.is_some() && stack.is_empty() {
395                    return Err(DocxError::Xml("more than one root element".into()));
396                }
397                count += 1;
398                if count > MAX_ELEMENTS {
399                    return Err(DocxError::Limit("too many XML elements".into()));
400                }
401                let local = String::from_utf8_lossy(e.local_name().as_ref()).into_owned();
402                let name = match el_ns {
403                    Some(ns) => format!("{}:{local}", element_prefix(ns, &local, stack.last().map(|p| p.name.as_str()), extra)),
404                    None => local,
405                };
406                let mut el = El { name, attrs: Vec::new(), kids: Vec::new() };
407                for a in e.attributes().with_checks(false) {
408                    if el.attrs.len() >= MAX_ATTRS {
409                        return Err(DocxError::Limit("too many attributes on one element".into()));
410                    }
411                    let Ok(a) = a else { continue };
412                    let key = a.key;
413                    let raw_key = key.as_ref();
414                    if raw_key == b"xmlns" || raw_key.starts_with(b"xmlns:") {
415                        continue;
416                    }
417                    let (ares, alocal) = r.resolver_mut().resolve_attribute(key);
418                    let alocal = String::from_utf8_lossy(alocal.as_ref()).into_owned();
419                    let aname = match ares {
420                        ResolveResult::Bound(ns) => format!("{}:{alocal}", prefix_for(ns.as_ref(), extra)),
421                        _ => alocal,
422                    };
423                    let val = match a.normalized_value(XmlVersion::Explicit1_0) {
424                        Ok(v) => v.into_owned(),
425                        Err(_) => String::from_utf8_lossy(&a.value).into_owned(),
426                    };
427                    // Requires contains namespace prefixes, not feature names. Resolve them while
428                    // the namespace scope still exists; declarations are omitted from this tree.
429                    let val = if el.name == "mc:Choice" && aname == "Requires" {
430                        let requirements: Vec<_> = val.split_whitespace().take(65).collect();
431                        if requirements.len() > 64 || requirements.iter().any(|prefix| prefix.len() > 128) {
432                            "?".to_string()
433                        } else {
434                            requirements
435                                .into_iter()
436                                .map(|prefix| {
437                                    let qualified = format!("{prefix}:requirement");
438                                    match r.resolver_mut().resolve_attribute(QName(qualified.as_bytes())).0 {
439                                        ResolveResult::Bound(ns) => requirement_prefix(ns.as_ref()),
440                                        _ => "?",
441                                    }
442                                })
443                                .collect::<Vec<_>>()
444                                .join(" ")
445                        }
446                    } else {
447                        val
448                    };
449                    el.attrs.push((aname, val));
450                }
451                if matches!(ev, Event::Start(_)) {
452                    if stack.len() >= MAX_DEPTH {
453                        return Err(DocxError::Limit("XML nesting too deep".into()));
454                    }
455                    stack.push(el);
456                } else {
457                    attach(&mut stack, &mut root, el);
458                }
459            }
460            Event::End(_) => {
461                if let Some(el) = stack.pop() {
462                    attach(&mut stack, &mut root, el);
463                }
464            }
465            // Line ends are normalised as XML 1.0 says (§2.11): CR LF and a lone CR are read as
466            // LF, so a file saved with Windows line ends has no CR in its text. A CR written as
467            // `&#13;` is a character reference and stays.
468            Event::Text(t) => {
469                if let Some(top) = stack.last_mut() {
470                    let s = t.xml_content(XmlVersion::Explicit1_0).map_err(|e| DocxError::Xml(e.to_string()))?;
471                    push_text(top, &s);
472                }
473            }
474            Event::CData(t) => {
475                if let Some(top) = stack.last_mut() {
476                    // CDATA that isn't UTF-8 is decoded lossily, with its line ends normalised the same way.
477                    let s = t
478                        .xml_content(XmlVersion::Explicit1_0)
479                        .map(|s| s.into_owned())
480                        .unwrap_or_else(|_| String::from_utf8_lossy(&t).replace("\r\n", "\n").replace('\r', "\n"));
481                    push_text(top, &s);
482                }
483            }
484            Event::GeneralRef(g) => {
485                if let Some(top) = stack.last_mut() {
486                    let ch = match g.resolve_char_ref() {
487                        Ok(Some(c)) => Some(c.to_string()),
488                        Ok(None) => {
489                            let n = g.decode().map_err(|e| DocxError::Xml(e.to_string()))?;
490                            quick_xml::escape::resolve_predefined_entity(&n).map(str::to_string)
491                        }
492                        Err(_) => None,
493                    };
494                    match ch {
495                        Some(c) => push_text(top, &c),
496                        None if strict => return Err(DocxError::Xml("an entity is not defined".into())),
497                        None => {}
498                    }
499                }
500            }
501            Event::Eof => break,
502            _ => {}
503        }
504        if !strict && root.is_some() && stack.is_empty() {
505            break;
506        }
507    }
508    if strict && !stack.is_empty() {
509        return Err(DocxError::Xml("an element is not closed".into()));
510    }
511    // Unclosed elements at EOF: close them (lenient).
512    while let Some(el) = stack.pop() {
513        attach(&mut stack, &mut root, el);
514    }
515    root.ok_or_else(|| DocxError::Xml("no root element".into()))
516}
517
518fn push_text(top: &mut El, s: &str) {
519    if let Some(Node::Text(prev)) = top.kids.last_mut() {
520        prev.push_str(s);
521    } else {
522        top.kids.push(Node::Text(s.to_string()));
523    }
524}
525
526fn attach(stack: &mut [El], root: &mut Option<El>, el: El) {
527    match stack.last_mut() {
528        Some(parent) => parent.kids.push(Node::El(el)),
529        None => {
530            if root.is_none() {
531                *root = Some(el);
532            }
533        }
534    }
535}

537/// Is `c` allowed in XML 1.0 text?
538fn xml_char_ok(c: char) -> bool {
539    matches!(c, '\t' | '\n' | '\r' | '\u{20}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..='\u{10FFFF}')
540}

542/// Escape text for element content or attribute values (drops characters XML can't carry).
543pub fn esc(s: &str) -> String {
544    let mut o = String::with_capacity(s.len());
545    for c in s.chars() {
546        match c {
547            '&' => o.push_str("&amp;"),
548            '<' => o.push_str("&lt;"),
549            '>' => o.push_str("&gt;"),
550            '"' => o.push_str("&quot;"),
551            '\'' => o.push_str("&apos;"),
552            '\t' => o.push_str("&#9;"),
553            '\n' => o.push_str("&#10;"),
554            '\r' => o.push_str("&#13;"),
555            c if xml_char_ok(c) => o.push(c),
556            _ => {}
557        }
558    }
559    o
560}

562/// A string-building XML writer.
563#[derive(Default)]
564pub struct W {
565    pub s: String,
566}
567
568impl W {
569    pub fn new() -> W {
570        W { s: String::from("<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n") }
571    }
572    fn attrs(&mut self, attrs: &[(&str, &str)]) {
573        for (k, v) in attrs {
574            self.s.push(' ');
575            self.s.push_str(k);
576            self.s.push_str("=\"");
577            self.s.push_str(&esc(v));
578            self.s.push('"');
579        }
580    }
581    pub fn open(&mut self, name: &str, attrs: &[(&str, &str)]) {
582        self.s.push('<');
583        self.s.push_str(name);
584        self.attrs(attrs);
585        self.s.push('>');
586    }
587    pub fn empty(&mut self, name: &str, attrs: &[(&str, &str)]) {
588        self.s.push('<');
589        self.s.push_str(name);
590        self.attrs(attrs);
591        self.s.push_str("/>");
592    }
593    pub fn close(&mut self, name: &str) {
594        self.s.push_str("</");
595        self.s.push_str(name);
596        self.s.push('>');
597    }
598    pub fn text(&mut self, t: &str) {
599        self.s.push_str(&esc(t));
600    }
601    /// `<name>text</name>`.
602    pub fn leaf(&mut self, name: &str, attrs: &[(&str, &str)], t: &str) {
603        self.open(name, attrs);
604        self.text(t);
605        self.close(name);
606    }
607    /// `<name w:val="v"/>`.
608    pub fn val(&mut self, name: &str, v: &str) {
609        self.empty(name, &[("w:val", v)]);
610    }
611    pub fn raw(&mut self, s: &str) {
612        self.s.push_str(s);
613    }
614    pub fn into_bytes(self) -> Vec<u8> {
615        self.s.into_bytes()
616    }
617}
