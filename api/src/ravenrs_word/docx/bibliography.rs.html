1//! Word bibliography sources: a `b:Sources` document — the custom XML part of a document, or
2//! Word's own `Sources.xml` master list, imported through Manage Sources (#2640).
3
4use std::collections::HashSet;
5
6use crate::doc::Source;
7
8use crate::docx::DocxError;
9use crate::docx::xml::El;
10
11/// Bibliography sources kept from one file (Word's own source manager holds far fewer).
12pub const MAX_SOURCES: usize = 10_000;
13/// Characters kept of a bibliography's citation style name.
14pub const MAX_STYLE_NAME: usize = 64;
15
16/// A `b:Sources` document read into the document model: its sources in file order, and the
17/// citation style it names (`b:Sources/@StyleName`), when it has one.
18#[derive(Clone, Debug, Default, PartialEq)]
19pub struct Sources {
20    pub sources: Vec<Source>,
21    pub style: Option<String>,
22}
23
24/// The bibliography namespace Word's own `Sources.xml` files may carry instead of OOXML's (the
25/// VBA `Source.XML` format, `schemas.microsoft.com/office/word/2004/10/bibliography`).
26const WORD_2004_BIBLIOGRAPHY: &str = "http://schemas.microsoft.com/office/word/2004/10/bibliography";
27
28/// Read a Word bibliography sources file — Word's `Sources.xml`, or any `b:Sources` document in
29/// one of the bibliography namespaces (ECMA-376's, or Word's own 2004/10) — into [`Sources`].
30pub fn read_sources(bytes: &[u8]) -> Result<Sources, DocxError> {
31    // `parse_ns` maps Word's 2004/10 namespace onto the canonical `b` prefix as well, so both
32    // formats read; the root check then refuses an un-namespaced or unrelated `<Sources>`.
33    let root = crate::docx::xml::parse_ns(bytes, &[("b", WORD_2004_BIBLIOGRAPHY)])?;
34    if root.name != "b:Sources" {
35        return Err(DocxError::NotWord(format!("root element is {}", root.name)));
36    }
37    Ok(parse(&root, &mut Seen::default(), MAX_SOURCES))
38}
39
40/// Bibliography sources already read: tags, and (title, author) of untagged ones.
41#[derive(Default)]
42pub(crate) struct Seen {
43    tags: HashSet<String>,
44    untagged: HashSet<(String, String)>,
45}
46
47/// The fresh sources and style of a parsed `b:Sources` root element: `seen` carries the tags and
48/// untagged (title, author) keys already taken, so repeated parts collapse into one source, and
49/// at most `room` fresh sources are built (later ones are left unread, bounding hostile files).
50pub(crate) fn parse(root: &El, seen: &mut Seen, room: usize) -> Sources {
51    let style = root
52        .attrs
53        .iter()
54        .find(|(k, _)| k.rsplit(':').next() == Some("StyleName"))
55        .map(|(_, v)| v.trim())
56        .filter(|name| !name.is_empty())
57        .map(|name| name.chars().take(MAX_STYLE_NAME).collect());
58    let child_text = |el: &El, local: &str| el.els().find(|e| e.local() == local).map(|e| e.text()).unwrap_or_default();
59    let parse_persons = |parent: &El| -> String {
60        let mut names = Vec::new();
61        let person_els: Vec<&El> = {
62            let in_lists: Vec<&El> =
63                parent.els().filter(|e| e.local() == "NameList").flat_map(|n| n.els().filter(|e| e.local() == "Person")).collect();
64            if in_lists.is_empty() { parent.els().filter(|e| e.local() == "Person").collect() } else { in_lists }
65        };
66        for name_el in person_els {
67            let last = child_text(name_el, "Last");
68            let first = child_text(name_el, "First");
69            let middle = child_text(name_el, "Middle");
70            let name = if !first.is_empty() && !middle.is_empty() {
71                format!("{last}, {first} {middle}")
72            } else if !first.is_empty() {
73                format!("{last}, {first}")
74            } else {
75                last
76            };
77            if !name.is_empty() {
78                names.push(name);
79            }
80        }
81        names.join("; ")
82    };
83
84    let mut out = Sources { sources: Vec::new(), style };
85    for src_el in root.els().filter(|e| e.local() == "Source") {
86        if out.sources.len() >= room {
87            log::warn!("docx: more than {MAX_SOURCES} bibliography sources; the rest are dropped");
88            break;
89        }
90        let tag = child_text(src_el, "Tag");
91        let raw_kind = child_text(src_el, "SourceType");
92        let kind = crate::docx::write::from_source_type(&raw_kind);
93        let (author, corporate, editor) = match src_el.els().find(|e| e.local() == "Author") {
94            Some(a) => {
95                let role = a.els().find(|e| e.local() == "Author");
96                let auth = role.map(parse_persons).filter(|s| !s.is_empty()).unwrap_or_else(|| parse_persons(a));
97                // Word's Corporate Author: one name, kept whole (`b:Corporate`, #4848).
98                let corporate = role.map(|r| child_text(r, "Corporate").trim().to_string()).unwrap_or_default();
99                let ed = a.els().find(|e| e.local() == "Editor").map(parse_persons).unwrap_or_default();
100                (auth, corporate, ed)
101            }
102            None => (String::new(), String::new(), String::new()),
103        };
104        let editor = if editor.is_empty() { src_el.els().find(|e| e.local() == "Editor").map(parse_persons).unwrap_or_default() } else { editor };
105        let title = child_text(src_el, "Title");
106        let year = child_text(src_el, "Year");
107        let month = child_text(src_el, "Month");
108        let day = child_text(src_el, "Day");
109        let city = child_text(src_el, "City");
110        let publisher = child_text(src_el, "Publisher");
111        let journal = child_text(src_el, "JournalName");
112        let edition = child_text(src_el, "Edition");
113        let volume = child_text(src_el, "Volume");
114        let issue = child_text(src_el, "Issue");
115        let pages = child_text(src_el, "Pages");
116        let url = child_text(src_el, "URL");
117        let accessed = match child_text(src_el, "Accessed") {
118            a if !a.is_empty() => a,
119            _ => {
120                let y = child_text(src_el, "YearAccessed");
121                let m = child_text(src_el, "MonthAccessed");
122                let d = child_text(src_el, "DayAccessed");
123                if !y.is_empty() && !m.is_empty() && !d.is_empty() {
124                    format!("{y}-{m}-{d}")
125                } else if !y.is_empty() && !m.is_empty() {
126                    format!("{y}-{m}")
127                } else {
128                    y
129                }
130            }
131        };
132        let doi = match child_text(src_el, "DOI") {
133            d if !d.is_empty() => d,
134            _ => child_text(src_el, "StandardNumber"),
135        };
136
137        // Keep sources with some content, once each (by tag, or title and author untagged).
138        if tag.is_empty() && title.is_empty() && author.is_empty() && corporate.is_empty() && editor.is_empty() {
139            continue;
140        }
141        let fresh = if tag.is_empty() {
142            let who = [&author, &corporate, &editor].into_iter().find(|w| !w.is_empty()).cloned().unwrap_or_default();
143            seen.untagged.insert((title.clone(), who))
144        } else {
145            seen.tags.insert(tag.clone())
146        };
147        if !fresh {
148            continue;
149        }
150        out.sources.push(Source {
151            tag,
152            kind,
153            author,
154            corporate,
155            title,
156            year,
157            city,
158            publisher,
159            journal,
160            volume,
161            pages,
162            url,
163            editor,
164            edition,
165            issue,
166            month,
167            day,
168            accessed,
169            doi,
170        });
171    }
172    out
173}

235    /// A Word `Sources.xml` (`b:Sources`): its style, the `b:Source` elements with their names,
236    /// kinds and fallback fields; a repeated tag and an entry with no content at all are dropped.
237    #[test]
238    fn a_word_sources_file_reads_into_sources() {
239        let f = read_sources(WORD_SOURCES.as_bytes()).unwrap();
240        assert_eq!(f.style.as_deref(), Some("MLA"), "the citation style comes from StyleName");
241        assert_eq!(
242            f.sources.len(),
243            3,
244            "the repeated tag and the empty b:Source are dropped: {:?}",
245            f.sources.iter().map(|s| &s.tag).collect::<Vec<_>>()
246        );
247        let book = &f.sources[0];
248        assert_eq!((book.tag.as_str(), book.kind.as_str()), ("Riv21", "book"));
249        assert_eq!(book.author, "Rivera, Alex; Chen, Mei L.");
250        assert_eq!(book.title, "Shared Spaces", "the first source of a repeated tag wins");
251        assert_eq!(
252            (book.title.as_str(), book.year.as_str(), book.city.as_str(), book.publisher.as_str()),
253            ("Shared Spaces", "2021", "Portland", "Harbor Books")
254        );
255        let web = &f.sources[1];
256        assert_eq!(web.kind, "website");
257        assert_eq!(web.author, "Okafor, Ada");
258        assert_eq!(web.accessed, "2024-May-17", "assemble the accessed date from its parts");
259        assert_eq!(web.doi, "10.1234/x", "StandardNumber stands in for a missing DOI");
260        let notes = &f.sources[2];
261        assert_eq!((notes.kind.as_str(), notes.journal.as_str(), notes.volume.as_str()), ("article", "Field Notes", "4"));
262    }

264    /// Anything whose root isn't `b:Sources` is refused, not read as an empty bibliography.
265    #[test]
266    fn a_file_that_is_not_b_sources_is_an_error() {
267        assert!(read_sources(b"<html><body>hi</body></html>").is_err());
268        assert!(matches!(read_sources(b"not xml at all"), Err(DocxError::Xml(_))));
269        assert!(read_sources(b"<Sources><Source><Tag>A</Tag></Source></Sources>").is_err(), "no namespace");
270        assert!(read_sources(br#"<o:Sources xmlns:o="urn:example:other"><o:Source/></o:Sources>"#).is_err(), "an unrelated namespace");
271    }

273    /// #4848: a corporate author (Word's Corporate Author box, `b:Corporate`) is read whole, and
274    /// an untagged source with only a corporate author is still kept, once.
275    #[test]
276    fn a_corporate_author_is_read_whole() {
277        let xml = br#"<b:Sources xmlns:b="http://schemas.openxmlformats.org/officeDocument/2006/bibliography">
278  <b:Source><b:Tag>Wor20</b:Tag><b:SourceType>Report</b:SourceType>
279    <b:Author><b:Author><b:Corporate> World Health Organization </b:Corporate></b:Author></b:Author>
280    <b:Title>World report on hearing</b:Title><b:Year>2020</b:Year></b:Source>
281  <b:Source><b:SourceType>Book</b:SourceType>
282    <b:Author><b:Author><b:Corporate>Harbor Guild, Inc.</b:Corporate></b:Author></b:Author></b:Source>
283  <b:Source><b:SourceType>Book</b:SourceType>
284    <b:Author><b:Author><b:Corporate>Harbor Guild, Inc.</b:Corporate></b:Author></b:Author></b:Source>
285</b:Sources>"#;
286        let f = read_sources(xml).unwrap();
287        assert_eq!(f.sources.len(), 2, "{:?}", f.sources);
288        assert_eq!((f.sources[0].corporate.as_str(), f.sources[0].author.as_str()), ("World Health Organization", ""));
289        assert_eq!(f.sources[1].corporate, "Harbor Guild, Inc.", "a comma doesn't split a corporate name");
290    }
