5/// Largest absolute length we accept, points (beyond any real page: 200 inches).
6pub const MAX_LEN_PT: f32 = 14_400.0;
7
8/// Parse a number, accepting decimals and ISO 29500 universal measures (`1in`, `2.5cm`, `12pt`,
9/// `3mm`, `1pc`, `1pi`). `per_pt` is how many file units make a point when there's no unit
10/// suffix (20 for twips, 2 for half-points, 8 for eighths, 12700 for EMU). Returns points.
11pub fn measure(s: &str, per_pt: f32) -> Option<f32> {
12    let t = s.trim();
13    if t.is_empty() || t.len() > 40 {
14        return None;
15    }
16    let bytes = t.as_bytes();
17    let mut split = t.len();
18    let mut in_exp = false;
19    for (i, &b) in bytes.iter().enumerate() {
20        if b.is_ascii_digit() || b == b'.' {
21            in_exp = false;
22            continue;
23        }
24        if (b == b'-' || b == b'+') && (i == 0 || in_exp) {
25            in_exp = false;
26            continue;
27        }
28        if b == b'e' || b == b'E' {
29            let rest = &bytes[i + 1..];
30            let after_sign = match rest.first() {
31                Some(&s) if s == b'+' || s == b'-' => &rest[1..],
32                _ => rest,
33            };
34            if after_sign.first().is_some_and(u8::is_ascii_digit) {
35                in_exp = true;
36                continue;
37            }
38        }
39        split = i;
40        break;
41    }
42    let (num, unit) = t.split_at(split);
43    let v: f64 = num.parse().ok()?;
44    if !v.is_finite() {
45        return None;
46    }
47    let unit_lower = unit.trim().to_ascii_lowercase();
48    let pt = match unit_lower.as_str() {
49        "" => v / per_pt as f64,
50        "pt" => v,
51        "in" => v * 72.0,
52        "cm" => v * 72.0 / 2.54,
53        "mm" => v * 72.0 / 25.4,
54        "pc" | "pi" => v * 12.0,
55        "px" => v * 0.75,
56        "emu" => v / 12_700.0,
57        _ => return None,
58    };
59    let pt = pt as f32;
60    pt.is_finite().then_some(pt.clamp(-MAX_LEN_PT, MAX_LEN_PT))
61}

63/// `w:line` under `lineRule="auto"`: the multiple of single line spacing, in 240ths (240 =
64/// single). Without a unit the value already is that — the transitional form. A universal
65/// measure is a height (strict OOXML writes `w:line="13.90pt"`, ISO 29500
66/// `ST_SignedTwipsMeasure`), so it goes through twips as Word reads it: × 20 ÷ 240 (13.90 pt =
67/// 278 twips, the 1.1583 multiple Word renders).
68pub fn line_multiple(s: &str) -> Option<f32> {
69    let t = s.trim();
70    let measured = t.ends_with(|c: char| c.is_ascii_alphabetic());
71    measure(t, 240.0).map(|v| if measured { v * 20.0 / 240.0 } else { v })
72}

74/// Plain integer (accepts decimals, rounds; clamps into i64 range).
75pub fn int(s: &str) -> Option<i64> {
76    let t = s.trim();
77    if t.is_empty() || t.len() > 40 {
78        return None;
79    }
80    if let Ok(v) = t.parse::<i64>() {
81        return Some(v);
82    }
83    let v: f64 = t.parse().ok()?;
84    v.is_finite().then(|| v.round().clamp(i64::MIN as f64, i64::MAX as f64) as i64)
85}

87/// A VML number (`gain`, `blacklevel`), unclamped: 16.16 fixed point with an `f` suffix
88/// (`19661f` is 0.3) or a plain decimal (`0.3`). None for anything else or non-finite.
89pub fn vml_number(s: &str) -> Option<f64> {
90    let t = s.trim();
91    if t.is_empty() || t.len() > 40 {
92        return None;
93    }
94    let v = match t.strip_suffix('f') {
95        Some(fixed) => fixed.trim().parse::<f64>().ok()? / 65_536.0,
96        None => t.parse::<f64>().ok()?,
97    };
98    v.is_finite().then_some(v)
99}

101/// A picture's brightness, contrast and transparency: `a:lum` (ECMA-376 §20.1.8.42) and
102/// `a:alphaModFix` (§20.1.8.4) in its `a:blip` (#1766).
103pub fn blip_fx(blip: &El) -> crate::doc::picfx::PicFx {
104    let val = |el: Option<&El>, k: &str| el.and_then(|e| e.attr(k)).and_then(int).map(|v| v.clamp(-1_000_000, 1_000_000) as i32);
105    let lum = blip.child("a:lum");
106    let amt = val(blip.child("a:alphaModFix"), "amt").unwrap_or(100_000).clamp(0, 100_000);
107    crate::doc::picfx::PicFx::new(val(lum, "bright").unwrap_or(0), val(lum, "contrast").unwrap_or(0), 100_000 - amt)
108}

110pub fn u32_of(s: &str) -> Option<u32> {
111    int(s).map(|v| v.clamp(0, u32::MAX as i64) as u32)
112}

114/// Twips attribute → points.
115pub fn tw(e: &El, attr: &str) -> Option<f32> {
116    e.attr(attr).and_then(|v| measure(v, 20.0))
117}

119/// Signed twips Word keeps in 16 bits (paragraph and list indents, ECMA-376 Part 1 §17.3.1.12):
120/// a plain integer outside `i16` keeps its low 16 bits, two's complement, as Word reads it
121/// (66256 → 720). Values with a unit or a fraction read as [`tw`].
122pub fn tw_i16(e: &El, attr: &str) -> Option<f32> {
123    let v = e.attr(attr)?;
124    match v.trim().parse::<i64>() {
125        Ok(i) => Some(f32::from(i as i16) / 20.0),
126        Err(_) => measure(v, 20.0),
127    }
128}
129
130/// Unsigned twips Word keeps in 16 bits (paragraph spacing, §17.3.1.33): a plain integer above
131/// 65535 keeps its low 16 bits (65537 → 1). Other values read as [`tw`].
132pub fn tw_u16(e: &El, attr: &str) -> Option<f32> {
133    let v = e.attr(attr)?;
134    match v.trim().parse::<i64>() {
135        Ok(i) if i > 0xFFFF => Some(f32::from((i & 0xFFFF) as u16) / 20.0),
136        _ => measure(v, 20.0),
137    }
138}

140/// ST_OnOff: absent `w:val` = on; `0`, `false`, `off` = off.
141pub fn on_off(e: &El) -> bool {
142    match e.attr("w:val") {
143        None => true,
144        Some(v) => !matches!(v.trim(), "0" | "false" | "off" | "False" | "FALSE"),
145    }
146}
147
148/// An on/off child of `parent`, if present.
149pub fn flag(parent: &El, name: &str) -> Option<bool> {
150    parent.child(name).map(on_off)
151}

172    #[test]
173    fn twips_beyond_16_bits_wrap() {
174        let el = |a: &str| El { attrs: vec![("w:v".into(), a.into())], ..El::default() };
175        let s = |a: &str| tw_i16(&el(a), "w:v");
176        let u = |a: &str| tw_u16(&el(a), "w:v");
177        assert_eq!(s("66256"), Some(36.0));
178        assert_eq!(s("-65516"), Some(1.0));
179        assert_eq!(s("98304"), Some(-1638.4));
180        assert_eq!(s("32767"), Some(1638.35));
181        assert_eq!(s("-32768"), Some(-1638.4));
182        assert_eq!(s("1in"), Some(72.0));
183        assert_eq!(s("x"), None);
184        assert_eq!(u("65537"), Some(0.05));
185        assert_eq!(u("65535"), Some(3276.75));
186        assert_eq!(u("12pt"), Some(12.0));
187        assert_eq!(u("-20"), Some(-1.0));
188    }

190    #[test]
191    fn measures() {
192        assert_eq!(measure("1440", 20.0), Some(72.0));
193        assert_eq!(measure("1in", 20.0), Some(72.0));
194        assert_eq!(measure("12pt", 2.0), Some(12.0));
195        assert_eq!(measure("24.0", 2.0), Some(12.0));
196        assert_eq!(measure("nan", 20.0), None);
197        assert_eq!(measure("1e400", 20.0), None);
198        assert_eq!(measure("99999999999999", 20.0), Some(MAX_LEN_PT));
199        assert_eq!(measure("x", 20.0), None);
200        // `w:line` auto: the multiple in 240ths without a unit, through twips with one (#2329).
201        assert_eq!(line_multiple("240"), Some(1.0));
202        assert_eq!(line_multiple("276"), Some(1.15));
203        assert_eq!(line_multiple("0"), Some(0.0));
204        let strict = line_multiple("13.90pt").expect("strict line height");
205        assert!((strict - 278.0 / 240.0).abs() < 1e-5, "{strict}");
206        let inches = line_multiple("1.2in").expect("inches");
207        assert!((inches - 7.2).abs() < 1e-5, "{inches}");
208        assert_eq!(line_multiple("junk"), None);
209        assert_eq!(line_multiple(""), None);
210        assert_eq!(measure("12700emu", 20.0), Some(1.0));
211        assert_eq!(measure("25400EMU", 20.0), Some(2.0));
212        assert_eq!(measure("1IN", 20.0), Some(72.0));
213        assert_eq!(measure("12PT", 2.0), Some(12.0));

215        assert_eq!(measure("25.4MM", 20.0), Some(72.0));
