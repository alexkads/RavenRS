1//! Password-protected Word documents: ECMA-376 document encryption with the *agile* method
2//! (Word 2010 and later), written from Microsoft's public Open Specification [MS-OFFCRYPTO]
3//! §2.3.4.4 and §2.3.4.10–§2.3.4.15, with the data-spaces streams of §2.1 / §2.3.4.1–§2.3.4.3.
4//!
5//! An encrypted package is not a zip but an OLE compound file ([MS-CFB]) holding
6//! `EncryptionInfo` (how the key is derived from the password, as XML) and `EncryptedPackage`
7//! (the original zip, AES-CBC encrypted in 4096-byte segments). A random *intermediate key*
8//! encrypts the package; the password, hashed `spinCount` times with a salt, encrypts that key and
9//! a verifier that tells a wrong password apart; an HMAC over the encrypted stream checks that it
10//! hasn't been damaged or tampered with.
11//!
12//! Older packages (Word 2007) use the *Standard* method (§2.3.4.5–§2.3.4.9): a binary
13//! `EncryptionInfo`, one AES key derived from the password with SHA-1 and 50,000 spins, and the
14//! whole package AES-ECB encrypted. [`decrypt`] reads both; [`encrypt`] writes agile only.
15//!
16//! Files are hostile: every stream read is capped, every length is checked before slicing, and the
17//! spin count is bounded by the specification's maximum. Passwords and keys are wiped from memory
18//! when dropped ([`Zeroizing`]) and never logged.
19

31/// The spin count Word uses, and WordCraft writes.
32pub const DEFAULT_SPIN_COUNT: u32 = 100_000;
33/// The largest spin count the specification allows (§2.3.4.10 `ST_SpinCount`): it bounds the
34/// work a hostile file can ask for.
35const MAX_SPIN_COUNT: u32 = 10_000_000;
36/// Word's limit on password length, in characters.
37pub const MAX_PASSWORD_CHARS: usize = 255;
38/// `EncryptedPackage` is encrypted in segments of this many bytes (§2.3.4.15).
39const SEGMENT: usize = 4096;
40/// Largest `EncryptionInfo` stream read (the XML is a few kilobytes).
41const MAX_INFO: u64 = 1 << 20;
42/// AES's block size.
43const BLOCK: usize = 16;
44/// Salt length written.
45const SALT: usize = 16;
46
47/// Block keys (§2.3.4.13, §2.3.4.14).
48const BK_VERIFIER_INPUT: [u8; 8] = [0xfe, 0xa7, 0xd2, 0x76, 0x3b, 0x4b, 0x9e, 0x79];
49const BK_VERIFIER_VALUE: [u8; 8] = [0xd7, 0xaa, 0x0f, 0x6d, 0x30, 0x61, 0x34, 0x4e];
50const BK_KEY_VALUE: [u8; 8] = [0x14, 0x6e, 0x0b, 0xe7, 0xab, 0xac, 0xd0, 0xd6];
51const BK_HMAC_KEY: [u8; 8] = [0x5f, 0xb2, 0xad, 0x01, 0x0c, 0xb9, 0xe1, 0xf6];
52const BK_HMAC_VALUE: [u8; 8] = [0xa0, 0x67, 0x7f, 0x02, 0xb2, 0x2c, 0x84, 0x33];
53
54/// Standard encryption (§2.3.2, §2.3.4.5–§2.3.4.7): `EncryptionHeader.Flags` bits, the AES and
55/// SHA-1 algorithm ids, the provider type, the fixed spin count and the SHA-1 size.
56const F_CRYPTO_API: u32 = 0x04;
57const F_EXTERNAL: u32 = 0x10;
58const F_AES: u32 = 0x20;
59const STANDARD_FLAGS: u32 = F_CRYPTO_API | F_AES;
60const ALG_AES_128: u32 = 0x660E;
61const ALG_AES_192: u32 = 0x660F;
62const ALG_AES_256: u32 = 0x6610;
63const ALG_SHA1: u32 = 0x8004;
64const PROV_RSA_AES: u32 = 0x18;
65const STANDARD_SPIN_COUNT: u32 = 50_000;
66const SHA1_SIZE: usize = 20;

68/// An OLE compound file starts with these bytes ([MS-CFB] §2.2).
69const CFB_MAGIC: [u8; 8] = [0xD0, 0xCF, 0x11, 0xE0, 0xA1, 0xB1, 0x1A, 0xE1];

71const NS_ENCRYPTION: &str = "http://schemas.microsoft.com/office/2006/encryption";
72const NS_PASSWORD: &str = "http://schemas.microsoft.com/office/2006/keyEncryptor/password";

74/// Is this an encrypted Office package (a compound file with `EncryptionInfo` and
75/// `EncryptedPackage` streams) rather than a zip?
76pub fn is_encrypted(bytes: &[u8]) -> bool {
77    if bytes.get(..8) != Some(&CFB_MAGIC[..]) || crate::doc::cfb_guard::check(bytes).is_err() {
78        return false;
79    }
80    match cfb::CompoundFile::open(Cursor::new(bytes)) {
81        Ok(comp) => comp.is_stream("/EncryptionInfo") && comp.is_stream("/EncryptedPackage"),
82        Err(_) => false,
83    }
84}

86/// The zip package inside an encrypted Office file. Without a password (`None`) this says
87/// [`DocxError::PasswordRequired`] once it knows the file is one it can decrypt; a password that
88/// doesn't fit gives [`DocxError::WrongPassword`].
89pub fn decrypt(bytes: &[u8], password: Option<&str>) -> Result<Vec<u8>, DocxError> {
90    crate::doc::cfb_guard::check(bytes).map_err(|e| bad(format!("not a readable compound file ({e})")))?;
91    let mut comp = cfb::CompoundFile::open(Cursor::new(bytes)).map_err(|e| bad(format!("not a readable compound file ({e})")))?;
92    let info = stream(&mut comp, "/EncryptionInfo", MAX_INFO)?;
93    let package = stream(&mut comp, "/EncryptedPackage", bytes.len() as u64)?;
94    if matches!(info.get(..4), Some([2..=4, 0, 2, 0])) {
95        return decrypt_standard(&info, &package, password);
96    }
97    let info = parse_info(&info)?;
98    let Some(password) = password else { return Err(DocxError::PasswordRequired) };
99    let pe = &info.password;
100    let hn = iterate(pe.hash, &pe.salt, password, pe.spin_count);
101    // The verifier (§2.3.4.13): a random value and its hash, both encrypted with keys derived
102    // from the password. They match only for the right password.
103    let key_input = derive_key(pe.hash, &hn, &BK_VERIFIER_INPUT, pe.key_bytes);
104    let key_value = derive_key(pe.hash, &hn, &BK_VERIFIER_VALUE, pe.key_bytes);
105    let iv = make_iv(pe.hash, &pe.salt, None, pe.block_size);
106    let input = Zeroizing::new(cbc_decrypt(&key_input, &iv, &pe.verifier_input)?);
107    let input = input.get(..pe.salt_size).ok_or_else(|| bad("the password verifier is too short"))?;
108    let value = Zeroizing::new(cbc_decrypt(&key_value, &iv, &pe.verifier_hash)?);
109    let value = value.get(..pe.hash.size()).ok_or_else(|| bad("the password verifier hash is too short"))?;
110    if !same(&pe.hash.hash(&[input]), value) {
111        return Err(DocxError::WrongPassword);
112    }
113    // The intermediate key that encrypts the package.
114    let key_key = derive_key(pe.hash, &hn, &BK_KEY_VALUE, pe.key_bytes);
115    let key = Zeroizing::new(cbc_decrypt(&key_key, &iv, &pe.key_value)?);
116    let key = Zeroizing::new(key.get(..info.key_bytes).ok_or_else(|| bad("the encrypted key is too short"))?.to_vec());
117    // Data integrity (§2.3.4.14): an HMAC of the whole EncryptedPackage stream. It is required:
118    // a file that could drop it would let altered ciphertext, or an altered StreamSize, through
119    // unauthenticated (`parse_info` refuses a file without it).
120    let (enc_key, enc_value) = &info.integrity;
121    let hmac_key = Zeroizing::new(cbc_decrypt(&key, &make_iv(info.hash, &info.salt, Some(&BK_HMAC_KEY), info.block_size), enc_key)?);
122    // The key is the salt, zero-padded to the block size; HMAC pads short keys with zeros
123    // anyway, so the padded key gives the same HMAC.
124    let hmac_key = hmac_key.get(..hmac_key.len().min(info.hash.size())).unwrap_or(&[]);
125    let expected = cbc_decrypt(&key, &make_iv(info.hash, &info.salt, Some(&BK_HMAC_VALUE), info.block_size), enc_value)?;
126    let expected = expected.get(..info.hash.size()).ok_or_else(|| bad("the integrity check value is too short"))?;
127    if !same(&info.hash.hmac(hmac_key, &package)?, expected) {
128        return Err(bad("the document is damaged (its integrity check failed)"));
129    }
130    // The package itself (§2.3.4.15): StreamSize, then 4096-byte segments, each with its own IV.
131    let size = package
132        .get(..8)
133        .and_then(|b| <[u8; 8]>::try_from(b).ok())
134        .map(u64::from_le_bytes)
135        .ok_or_else(|| bad("the encrypted package is truncated"))?;
136    let data = package.get(8..).unwrap_or(&[]);
137    let size = usize::try_from(size).ok().filter(|s| *s <= data.len()).ok_or_else(|| bad("the encrypted package is truncated"))?;
138    let needed = size.div_ceil(BLOCK).saturating_mul(BLOCK);
139    let data = data.get(..needed).ok_or_else(|| bad("the encrypted package is truncated"))?;
140    let mut out = Vec::with_capacity(needed);
141    for (i, seg) in data.chunks(SEGMENT).enumerate() {
142        let i = u32::try_from(i).map_err(|_| bad("the encrypted package is too large"))?;
143        let iv = make_iv(info.hash, &info.salt, Some(&i.to_le_bytes()), info.block_size);
144        out.extend_from_slice(&cbc_decrypt(&key, &iv, seg)?);
145    }
146    out.truncate(size);
147    Ok(out)
148}

150/// [`decrypt`] for Standard encryption (§2.3.4.5–§2.3.4.9): one AES key from the password,
151/// checked with the verifier, and the whole package AES-ECB encrypted after its 8-byte size.
152fn decrypt_standard(info: &[u8], package: &[u8], password: Option<&str>) -> Result<Vec<u8>, DocxError> {
153    let s = parse_standard(info)?;
154    let Some(password) = password else { return Err(DocxError::PasswordRequired) };
155    let key = standard_key(s.salt, password, s.key_bytes);
156    // The verifier (§2.3.4.9): a random value and its SHA-1, both encrypted with the key.
157    let verifier = Zeroizing::new(ecb(&key, s.verifier, false)?);
158    let hash = Zeroizing::new(ecb(&key, s.verifier_hash, false)?);
159    let hash = hash.get(..SHA1_SIZE).ok_or_else(|| bad("the password verifier hash is too short"))?;
160    if !same(&Hash::Sha1.hash(&[&verifier]), hash) {
161        return Err(DocxError::WrongPassword);
162    }
163    // The package (§2.3.4.4): StreamSize, then the data; bytes past the last block are ignored.
164    let size = package
165        .get(..8)
166        .and_then(|b| <[u8; 8]>::try_from(b).ok())
167        .map(u64::from_le_bytes)
168        .ok_or_else(|| bad("the encrypted package is truncated"))?;
169    let data = package.get(8..).unwrap_or(&[]);
170    let size = usize::try_from(size).ok().filter(|s| *s <= data.len()).ok_or_else(|| bad("the encrypted package is truncated"))?;
171    let needed = size.div_ceil(BLOCK).saturating_mul(BLOCK);
172    let data = data.get(..needed).ok_or_else(|| bad("the encrypted package is truncated"))?;
173    let mut out = ecb(&key, data, false)?;
174    out.truncate(size);
175    Ok(out)
176}

178/// What a Standard `EncryptionInfo` stream says (§2.3.4.5): the key size, and the verifier's salt
179/// and encrypted values.
180struct Standard<'a> {
181    key_bytes: usize,
182    salt: &'a [u8],
183    verifier: &'a [u8],
184    verifier_hash: &'a [u8],
185}
186
187fn parse_standard(info: &[u8]) -> Result<Standard<'_>, DocxError> {
188    let short = || bad("EncryptionInfo is truncated");
189    let u32_at = |at: usize| -> Result<u32, DocxError> {
190        let b = at.checked_add(4).and_then(|end| info.get(at..end)).ok_or_else(short)?;
191        <[u8; 4]>::try_from(b).map(u32::from_le_bytes).map_err(|_| short())
192    };
193    let flags = u32_at(4)?;
194    if flags & F_EXTERNAL != 0 {
195        return Err(bad("this file uses extensible encryption, which WordCraft can't open"));
196    }
197    if flags & STANDARD_FLAGS != STANDARD_FLAGS {
198        return Err(bad("this file's encryption isn't AES, which WordCraft can't open"));
199    }
200    // EncryptionHeader (§2.3.2): Flags, SizeExtra, AlgID, AlgIDHash, KeySize, ProviderType, two
201    // reserved words, then the CSP name, which isn't needed.
202    let header_size = usize::try_from(u32_at(8)?).map_err(|_| short())?;
203    if header_size < 32 {
204        return Err(bad("the encryption header is too short"));
205    }
206    let verifier_at = header_size.checked_add(12).filter(|&v| v <= info.len()).ok_or_else(short)?;
207    // AlgID 0 and AlgIDHash 0 mean "set by the flags": AES-128 and SHA-1.
208    let bits = match u32_at(20)? {
209        0 | ALG_AES_128 => 128,
210        ALG_AES_192 => 192,
211        ALG_AES_256 => 256,
212        id => return Err(bad(format!("encryption algorithm 0x{id:X} isn't supported"))),
213    };
214    let hash_id = u32_at(24)?;
215    if !matches!(hash_id, 0 | ALG_SHA1) {
216        return Err(bad(format!("hash algorithm 0x{hash_id:X} isn't supported")));
217    }
218    if u32_at(28)? != bits {
219        return Err(bad("the key size doesn't match the encryption algorithm"));
220    }
221    // EncryptionVerifier (§2.3.3): SaltSize (16), Salt, EncryptedVerifier, VerifierHashSize (20
222    // for SHA-1), EncryptedVerifierHash (32 bytes for AES).
223    if u32_at(verifier_at)? != SALT as u32 {
224        return Err(bad("the salt isn't 16 bytes"));
225    }
226    let field = |at: usize, len: usize| verifier_at.checked_add(at).and_then(|a| info.get(a..a.checked_add(len)?)).ok_or_else(short);
227    let salt = field(4, SALT)?;
228    let verifier = field(4 + SALT, BLOCK)?;
229    if u32_at(verifier_at.saturating_add(4 + SALT + BLOCK))? != SHA1_SIZE as u32 {
230        return Err(bad("the password verifier hash isn't SHA-1"));
231    }
232    let verifier_hash = field(8 + SALT + BLOCK, 2 * BLOCK)?;
233    Ok(Standard { key_bytes: bits as usize / 8, salt, verifier, verifier_hash })
234}

236/// Encrypt a zip package with `password` (agile encryption: AES-256-CBC, SHA-512, Word's spin
237/// count), as Word's "Encrypt with Password" does.
238pub fn encrypt(package: &[u8], password: &str) -> Result<Vec<u8>, DocxError> {
239    encrypt_with_spin_count(package, password, DEFAULT_SPIN_COUNT)
240}

242/// [`encrypt`] with a chosen spin count (tests use a small one; files should use the default).
243pub fn encrypt_with_spin_count(package: &[u8], password: &str, spin_count: u32) -> Result<Vec<u8>, DocxError> {
244    check_password(password)?;
245    let spin_count = spin_count.min(MAX_SPIN_COUNT);
246    let hash = Hash::Sha512;
247    let key_bytes = 32;
248    let key_data_salt = random(SALT)?;
249    let password_salt = random(SALT)?;
250    let key = random(key_bytes)?;
251    let verifier = random(SALT)?;
252    // The HMAC key is a salt as long as the hash (MS-OFFCRYPTO §2.3.4.14).
253    let hmac_salt = random(hash.size())?;
254
255    // The package, segment by segment; the last segment is zero-padded to the block size.
256    let mut stream = Vec::with_capacity(package.len().saturating_add(8 + BLOCK));
257    stream.extend_from_slice(&(package.len() as u64).to_le_bytes());
258    for (i, seg) in package.chunks(SEGMENT).enumerate() {
259        let i = u32::try_from(i).map_err(|_| bad("the document is too large to encrypt"))?;
260        let iv = make_iv(hash, &key_data_salt, Some(&i.to_le_bytes()), BLOCK);
261        stream.extend_from_slice(&cbc_encrypt(&key, &iv, &padded(seg))?);
262    }
263
264    // The password key encryptor (§2.3.4.13).
265    let hn = iterate(hash, &password_salt, password, spin_count);
266    let iv = make_iv(hash, &password_salt, None, BLOCK);
267    let enc_verifier_input = cbc_encrypt(&derive_key(hash, &hn, &BK_VERIFIER_INPUT, key_bytes), &iv, &padded(&verifier))?;
268    let enc_verifier_hash = cbc_encrypt(&derive_key(hash, &hn, &BK_VERIFIER_VALUE, key_bytes), &iv, &padded(&hash.hash(&[&verifier])))?;
269    let enc_key_value = cbc_encrypt(&derive_key(hash, &hn, &BK_KEY_VALUE, key_bytes), &iv, &padded(&key))?;
270
271    // Data integrity (§2.3.4.14).
272    let enc_hmac_key = cbc_encrypt(&key, &make_iv(hash, &key_data_salt, Some(&BK_HMAC_KEY), BLOCK), &padded(&hmac_salt))?;
273    let hmac = hash.hmac(&hmac_salt, &stream)?;
274    let enc_hmac_value = cbc_encrypt(&key, &make_iv(hash, &key_data_salt, Some(&BK_HMAC_VALUE), BLOCK), &padded(&hmac))?;
275
276    let b64 = |b: &[u8]| base64::engine::general_purpose::STANDARD.encode(b);
277    let params = format!(
278        r#"saltSize="{SALT}" blockSize="{BLOCK}" keyBits="{}" hashSize="{}" cipherAlgorithm="AES" cipherChaining="ChainingModeCBC" hashAlgorithm="{}""#,
279        key_bytes * 8,
280        hash.size(),
281        hash.name(),
282    );
283    let xml = format!(
284        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\r\n\
285         <encryption xmlns=\"{NS_ENCRYPTION}\" xmlns:p=\"{NS_PASSWORD}\">\
286         <keyData {params} saltValue=\"{}\"/>\
287         <dataIntegrity encryptedHmacKey=\"{}\" encryptedHmacValue=\"{}\"/>\
288         <keyEncryptors><keyEncryptor uri=\"{NS_PASSWORD}\">\
289         <p:encryptedKey spinCount=\"{spin_count}\" {params} saltValue=\"{}\" encryptedVerifierHashInput=\"{}\" encryptedVerifierHashValue=\"{}\" encryptedKeyValue=\"{}\"/>\
290         </keyEncryptor></keyEncryptors></encryption>",
291        b64(&key_data_salt),
292        b64(&enc_hmac_key),
293        b64(&enc_hmac_value),
294        b64(&password_salt),
295        b64(&enc_verifier_input),
296        b64(&enc_verifier_hash),
297        b64(&enc_key_value),
298    );
299    // EncryptionInfo (§2.3.4.10): version 4.4, reserved 0x40, the XML.
300    let mut info = vec![4, 0, 4, 0, 0x40, 0, 0, 0];
301    info.extend_from_slice(xml.as_bytes());
302
303    let io = |e: std::io::Error| bad(format!("couldn't write the compound file ({e})"));
304    let mut comp = cfb::CompoundFile::create_with_version(cfb::Version::V3, Cursor::new(Vec::new())).map_err(io)?;
305    for (path, data) in [
306        ("/\u{6}DataSpaces/Version", data_space_version()),
307        ("/\u{6}DataSpaces/DataSpaceMap", data_space_map()),
308        ("/\u{6}DataSpaces/DataSpaceInfo/StrongEncryptionDataSpace", data_space_definition()),
309        ("/\u{6}DataSpaces/TransformInfo/StrongEncryptionTransform/\u{6}Primary", primary_transform()),
310        ("/EncryptionInfo", info),
311        ("/EncryptedPackage", stream),
312    ] {
313        if let Some((parent, _)) = path.rsplit_once('/')
314            && !parent.is_empty()
315        {
316            comp.create_storage_all(parent).map_err(io)?;
317        }
318        comp.create_stream(path).map_err(io)?.write_all(&data).map_err(io)?;
319    }
320    comp.flush().map_err(io)?;
321    Ok(comp.into_inner().into_inner())
322}

324/// A package encrypted with the *Standard* method (§2.3.4.5–§2.3.4.9: AES-ECB, SHA-1, 50,000
325/// spins), as Word 2007 wrote it, for tests only: WordCraft saves with the agile method
326/// ([`encrypt`]). `key_bits` is 128, 192 or 256. The compound file holds just the two streams
327/// [`decrypt`] reads.
328#[doc(hidden)]
329pub fn encrypt_standard_for_tests(package: &[u8], password: &str, key_bits: u32) -> Result<Vec<u8>, DocxError> {
330    check_password(password)?;
331    let alg_id = match key_bits {
332        128 => ALG_AES_128,
333        192 => ALG_AES_192,
334        256 => ALG_AES_256,
335        _ => return Err(bad(format!("unsupported AES key size {key_bits}"))),
336    };
337    let salt = random(SALT)?;
338    let verifier = random(SALT)?;
339    let key = standard_key(&salt, password, key_bits as usize / 8);
340    let mut header = Vec::new();
341    for v in [STANDARD_FLAGS, 0, alg_id, ALG_SHA1, key_bits, PROV_RSA_AES, 0, 0] {
342        header.extend_from_slice(&v.to_le_bytes());
343    }
344    // CSPName: an empty null-terminated UTF-16 string.
345    header.extend_from_slice(&[0, 0]);
346    let mut info = vec![3, 0, 2, 0];
347    info.extend_from_slice(&STANDARD_FLAGS.to_le_bytes());
348    info.extend_from_slice(&(header.len() as u32).to_le_bytes());
349    info.extend_from_slice(&header);
350    info.extend_from_slice(&(SALT as u32).to_le_bytes());
351    info.extend_from_slice(&salt);
352    info.extend_from_slice(&ecb(&key, &verifier, true)?);
353    info.extend_from_slice(&(SHA1_SIZE as u32).to_le_bytes());
354    info.extend_from_slice(&ecb(&key, &padded(&Hash::Sha1.hash(&[&verifier])), true)?);
355    let mut stream = (package.len() as u64).to_le_bytes().to_vec();
356    stream.extend_from_slice(&ecb(&key, &padded(package), true)?);
357    let io = |e: std::io::Error| bad(format!("couldn't write the compound file ({e})"));
358    let mut comp = cfb::CompoundFile::create_with_version(cfb::Version::V3, Cursor::new(Vec::new())).map_err(io)?;
359    comp.create_stream("/EncryptionInfo").map_err(io)?.write_all(&info).map_err(io)?;
360    comp.create_stream("/EncryptedPackage").map_err(io)?.write_all(&stream).map_err(io)?;
361    comp.flush().map_err(io)?;
362    Ok(comp.into_inner().into_inner())
363}

365/// Refuse a password Word wouldn't accept: empty, or longer than [`MAX_PASSWORD_CHARS`].
366pub fn check_password(password: &str) -> Result<(), DocxError> {
367    if password.is_empty() {
368        return Err(bad("the password is empty"));
369    }
370    if password.chars().count() > MAX_PASSWORD_CHARS {
371        return Err(bad(format!("passwords are at most {MAX_PASSWORD_CHARS} characters")));
372    }
373    Ok(())
374}

376/// Restrict Editing's password (`w:documentProtection`, ECMA-376 Part 1 §17.15.1.29): the
377/// attributes that store its hash — SHA-512 of a random 16-byte salt and the password as UTF-16LE,
378/// hashed again `spin_count` times with the iteration number appended ([MS-OFFCRYPTO] §2.4.2.4,
379/// the ISO write-protection method). Protection isn't encryption: the document stays readable,
380/// the password only guards turning protection off.
381pub fn protection_hash(password: &str, spin_count: u32) -> Result<Vec<(String, String)>, DocxError> {
382    check_password(password)?;
383    let spin_count = spin_count.min(MAX_SPIN_COUNT);
384    let salt = random(SALT)?;
385    let hash = iterate_appended(Hash::Sha512, &salt, password, spin_count);
386    let b64 = |b: &[u8]| base64::engine::general_purpose::STANDARD.encode(b);
387    Ok(vec![
388        ("w:algorithmName".into(), "SHA-512".into()),
389        ("w:hashValue".into(), b64(&hash)),
390        ("w:saltValue".into(), b64(&salt)),
391        ("w:spinCount".into(), spin_count.to_string()),
392    ])
393}

395/// What checking a Restrict Editing password found ([`check_protection_password`]).
396#[derive(Clone, Copy, Debug, PartialEq, Eq)]
397pub enum PasswordCheck {
398    /// The protection has no password: anyone may stop it.
399    NoPassword,
400    /// The password matches the stored hash.
401    Correct,
402    /// The password doesn't match.
403    Incorrect,
404    /// The hash uses an algorithm or form WordCraft can't check, or is damaged. The protection
405    /// must stay on: guessing would let any password through.
406    Unsupported,
407}
408
409/// Does `password` match a Restrict Editing hash (attributes as [`protection_hash`] makes them, or
410/// as read from a file)? Two forms are checked:
411///
412/// - `w:algorithmName` / `w:hashValue` / `w:saltValue` / `w:spinCount`: the password itself is
413///   hashed ([`iterate_appended`]).
414/// - `w:cryptAlgorithmSid` / `w:hash` / `w:salt` / `w:cryptSpinCount` (ECMA-376 Part 4 §14.8.1):
415///   the password first becomes Word's legacy 32-bit key ([`legacy_key`]), whose bytes, reversed
416///   and written as an upper-case hex string, are hashed the same way.
417///
418/// Anything else — an algorithm other than SHA-1/SHA-256/SHA-384/SHA-512, an application-defined
419/// algorithm (`w:algIdExt`), a class other than `hash`, a spin count above the 10,000,000 Word
420/// accepts, damaged values — is [`PasswordCheck::Unsupported`], never a match.
421pub fn check_protection_password(attrs: &[(String, String)], password: &str) -> PasswordCheck {
422    let get = |k: &str| attrs.iter().find(|(n, _)| n == k).map(|(_, v)| v.trim());
423    if get("w:hashValue").is_some() {
424        let Some(hash) = get("w:algorithmName").and_then(Hash::parse) else { return PasswordCheck::Unsupported };
425        return verify(hash, get("w:hashValue"), get("w:saltValue"), get("w:spinCount"), password);
426    }
427    if get("w:hash").is_none() {
428        // §17.15.1.29: without a hash no password is associated with the protection.
429        return PasswordCheck::NoPassword;
430    }
431    let class_ok = get("w:cryptAlgorithmClass").is_none_or(|c| c == "hash");
432    let type_ok = get("w:cryptAlgorithmType").is_none_or(|t| t == "typeAny");
433    if !class_ok || !type_ok || get("w:algIdExt").is_some() {
434        return PasswordCheck::Unsupported;
435    }
436    // Part 4 §14.8.1 lists 1 MD2, 2 MD4, 3 MD5, 4 SHA-1, 12 SHA-256, 13 SHA-384, 14 SHA-512.
437    let hash = match get("w:cryptAlgorithmSid").and_then(|v| v.parse::<u32>().ok()) {
438        Some(4) => Hash::Sha1,
439        Some(12) => Hash::Sha256,
440        Some(13) => Hash::Sha384,
441        Some(14) => Hash::Sha512,
442        _ => return PasswordCheck::Unsupported,
443    };
444    verify(hash, get("w:hash"), get("w:salt"), get("w:cryptSpinCount"), &legacy_key_hex(password))
445}

447/// Hash `input` with the stored salt and spin count and compare with the stored hash. Damaged or
448/// out-of-range values are [`PasswordCheck::Unsupported`].
449fn verify(hash: Hash, want: Option<&str>, salt: Option<&str>, spin_count: Option<&str>, input: &str) -> PasswordCheck {
450    let dec = |v: &str| base64::engine::general_purpose::STANDARD.decode(v).ok();
451    let Some(want) = want.and_then(dec) else { return PasswordCheck::Unsupported };
452    // §17.15.1.29: without a salt nothing is prepended.
453    let salt = match salt {
454        Some(v) => match dec(v) {
455            Some(s) => s,
456            None => return PasswordCheck::Unsupported,
457        },
458        None => Vec::new(),
459    };
460    // Word accepts spin counts from 0 to 10,000,000 ([MS-OI29500] 2.1.402 d); a missing one is 0.
461    let spin_count = match spin_count.map(str::parse::<u32>) {
462        None => 0,
463        Some(Ok(n)) if n <= MAX_SPIN_COUNT => n,
464        Some(_) => return PasswordCheck::Unsupported,
465    };
466    if want.len() != hash.size() || salt.len() > 1024 {
467        return PasswordCheck::Unsupported;
468    }
469    if same(&iterate_appended(hash, &salt, input, spin_count), &want) { PasswordCheck::Correct } else { PasswordCheck::Incorrect }
470}

497/// Word's legacy 32-bit password key (ECMA-376 Part 4 §14.8.1): the password as UTF-16 without a
498/// leading BOM, cut to 15 characters, each character reduced to one byte (its low byte, or its
499/// high byte when the low byte is 0); the high-order word comes from the initial code array and
500/// the encryption matrix, the low-order word from rotating and XOR-ing the bytes backwards.
501fn legacy_key(password: &str) -> u32 {
502    let mut units = password.encode_utf16().peekable();
503    units.next_if_eq(&0xFEFF);
504    let bytes: Zeroizing<Vec<u8>> =
505        Zeroizing::new(units.take(LEGACY_INITIAL_CODE.len()).map(|u| if u & 0xFF != 0 { (u & 0xFF) as u8 } else { (u >> 8) as u8 }).collect());
506    let Some(&initial) = bytes.len().checked_sub(1).and_then(|i| LEGACY_INITIAL_CODE.get(i)) else { return 0 };
507    let mut high = initial;
508    let first_row = LEGACY_MATRIX.len() - bytes.len();
509    for (row, &c) in LEGACY_MATRIX.iter().skip(first_row).zip(bytes.iter()) {
510        for (bit, &code) in row.iter().enumerate() {
511            if c & (1 << bit) != 0 {
512                high ^= code;
513            }
514        }
515    }
516    let rotate = |w: u16| ((w >> 14) & 0x0001) | ((w << 1) & 0x7FFF);
517    let mut low = 0u16;
518    for &c in bytes.iter().rev() {
519        low = rotate(low) ^ u16::from(c);
520    }
521    // `bytes` holds at most 15 values, so its length fits a word.
522    low = rotate(low) ^ (bytes.len() as u16) ^ 0xCE4B;
523    (u32::from(high) << 16) | u32::from(low)
524}

526/// What the legacy form hashes: the key's bytes in reversed order (Part 4 §14.8.1: 0x64CEED7E
527/// becomes 7EEDCE64), as an upper-case hex string — hashed as UTF-16LE like a password, as Word
528/// does ([MS-OE376] 2.1.410 a).
529fn legacy_key_hex(password: &str) -> Zeroizing<String> {
530    let key = legacy_key(password);
531    Zeroizing::new(key.to_le_bytes().iter().map(|b| format!("{b:02X}")).collect())
532}

534/// The ISO write-protection hash ([MS-OFFCRYPTO] §2.4.2.4): H0 = H(salt + password as UTF-16LE),
535/// Hn = H(Hn-1 + iterator) — the iterator after the hash, unlike encryption's [`iterate`].
536fn iterate_appended(hash: Hash, salt: &[u8], password: &str, spin_count: u32) -> Zeroizing<Vec<u8>> {
537    let pw: Zeroizing<Vec<u8>> = Zeroizing::new(password.encode_utf16().flat_map(u16::to_le_bytes).collect());
538    let mut h = Zeroizing::new(hash.hash(&[salt, &pw]));
539    for i in 0..spin_count.min(MAX_SPIN_COUNT) {
540        *h = hash.hash(&[&h, &i.to_le_bytes()]);
541    }
542    h
543}

549/// A root-level stream, at most `cap` bytes.
550fn stream<F: Read + std::io::Seek>(comp: &mut cfb::CompoundFile<F>, path: &str, cap: u64) -> Result<Vec<u8>, DocxError> {
551    let s = comp.open_stream(path).map_err(|e| bad(format!("can't read {} ({e})", path.trim_start_matches('/'))))?;
552    let mut v = Vec::new();
553    s.take(cap.saturating_add(1)).read_to_end(&mut v).map_err(|e| bad(format!("can't read {} ({e})", path.trim_start_matches('/'))))?;
554    if v.len() as u64 > cap {
555        return Err(DocxError::Limit(format!("{} is too large", path.trim_start_matches('/'))));
556    }
557    Ok(v)
558}

560/// The hash algorithms agile encryption names that WordCraft supports.
561#[derive(Clone, Copy, Debug, PartialEq, Eq)]
562enum Hash {
563    Sha1,
564    Sha256,
565    Sha384,
566    Sha512,
567}
568
569impl Hash {
570    fn parse(s: &str) -> Option<Hash> {
571        match s.to_ascii_uppercase().replace('-', "").as_str() {
572            "SHA1" => Some(Hash::Sha1),
573            "SHA256" => Some(Hash::Sha256),
574            "SHA384" => Some(Hash::Sha384),
575            "SHA512" => Some(Hash::Sha512),
576            _ => None,
577        }
578    }
579
580    fn name(self) -> &'static str {
581        match self {
582            Hash::Sha1 => "SHA-1",
583            Hash::Sha256 => "SHA256",
584            Hash::Sha384 => "SHA384",
585            Hash::Sha512 => "SHA512",
586        }
587    }
588
589    fn size(self) -> usize {
590        match self {
591            Hash::Sha1 => 20,
592            Hash::Sha256 => 32,
593            Hash::Sha384 => 48,
594            Hash::Sha512 => 64,
595        }
596    }
597
598    /// The hash of the concatenated `parts`.
599    fn hash(self, parts: &[&[u8]]) -> Vec<u8> {
600        fn go<D: Digest>(parts: &[&[u8]]) -> Vec<u8> {
601            let mut d = D::new();
602            for p in parts {
603                d.update(p);
604            }
605            d.finalize().to_vec()
606        }
607        match self {
608            Hash::Sha1 => go::<sha1::Sha1>(parts),
609            Hash::Sha256 => go::<sha2::Sha256>(parts),
610            Hash::Sha384 => go::<sha2::Sha384>(parts),
611            Hash::Sha512 => go::<sha2::Sha512>(parts),
612        }
613    }
614
615    fn hmac(self, key: &[u8], msg: &[u8]) -> Result<Vec<u8>, DocxError> {
616        fn go<M: Mac + KeyInit>(key: &[u8], msg: &[u8]) -> Result<Vec<u8>, DocxError> {
617            let mut m = <M as KeyInit>::new_from_slice(key).map_err(|_| bad("bad integrity key"))?;
618            m.update(msg);
619            Ok(m.finalize().into_bytes().to_vec())
620        }
621        match self {
622            Hash::Sha1 => go::<hmac::Hmac<sha1::Sha1>>(key, msg),
623            Hash::Sha256 => go::<hmac::Hmac<sha2::Sha256>>(key, msg),
624            Hash::Sha384 => go::<hmac::Hmac<sha2::Sha384>>(key, msg),
625            Hash::Sha512 => go::<hmac::Hmac<sha2::Sha512>>(key, msg),
626        }
627    }
628}

630/// The password hash after `spin_count` iterations (§2.3.4.11): H0 = H(salt + password as
631/// UTF-16LE), Hn = H(iterator + Hn-1).
632fn iterate(hash: Hash, salt: &[u8], password: &str, spin_count: u32) -> Zeroizing<Vec<u8>> {
633    let pw: Zeroizing<Vec<u8>> = Zeroizing::new(password.encode_utf16().flat_map(u16::to_le_bytes).collect());
634    let mut h = Zeroizing::new(hash.hash(&[salt, &pw]));
635    for i in 0..spin_count.min(MAX_SPIN_COUNT) {
636        *h = hash.hash(&[&i.to_le_bytes(), &h]);
637    }
638    h
639}
640
641/// A key from the iterated password hash and a block key (§2.3.4.11): H(Hn + blockKey),
642/// truncated to `len`, or padded with 0x36.
643fn derive_key(hash: Hash, hn: &[u8], block_key: &[u8], len: usize) -> Zeroizing<Vec<u8>> {
644    let mut k = Zeroizing::new(hash.hash(&[hn, block_key]));
645    k.resize(len, 0x36);
646    k
647}

649/// Standard encryption's key (§2.3.4.7): the SHA-1 password hash after 50,000 iterations, hashed
650/// with block number 0; then X1 = SHA-1 of that hash XOR-ed into 64 bytes of 0x36, X2 the same
651/// with 0x5C, and the key is the first `len` bytes of X1 + X2.
652fn standard_key(salt: &[u8], password: &str, len: usize) -> Zeroizing<Vec<u8>> {
653    let hn = iterate(Hash::Sha1, salt, password, STANDARD_SPIN_COUNT);
654    let hfinal = Zeroizing::new(Hash::Sha1.hash(&[&hn, &0u32.to_le_bytes()]));
655    let x = |fill: u8| {
656        let mut buf = Zeroizing::new([fill; 64]);
657        for (b, h) in buf.iter_mut().zip(hfinal.iter()) {
658            *b ^= h;
659        }
660        Zeroizing::new(Hash::Sha1.hash(&[&buf[..]]))
661    };
662    let mut key = Zeroizing::new(x(0x36).to_vec());
663    key.extend_from_slice(&x(0x5C));
664    key.truncate(len);
665    key
666}

668/// AES-ECB without padding (Standard encryption, §2.3.4.7–§2.3.4.9): `data` must be whole blocks.
669fn ecb(key: &[u8], data: &[u8], encrypt: bool) -> Result<Vec<u8>, DocxError> {
670    let aes = Aes::new(key)?;
671    let (blocks, rest) = data.as_chunks::<BLOCK>();
672    if !rest.is_empty() {
673        return Err(bad("encrypted data isn't a whole number of blocks"));
674    }
675    let mut out = Vec::with_capacity(data.len());
676    for b in blocks {
677        let mut x = aes::Block::from(*b);
678        if encrypt {
679            aes.encrypt(&mut x);
680        } else {
681            aes.decrypt(&mut x);
682        }
683        out.extend_from_slice(&x);
684    }
685    Ok(out)
686}

688/// An initialization vector (§2.3.4.12): H(salt + blockKey), or the salt itself, truncated to
689/// the block size or padded with 0x36.
690fn make_iv(hash: Hash, salt: &[u8], block_key: Option<&[u8]>, block_size: usize) -> Vec<u8> {
691    let mut iv = match block_key {
692        Some(b) => hash.hash(&[salt, b]),
693        None => salt.to_vec(),
694    };
695    iv.resize(block_size, 0x36);
696    iv
697}

699/// `data` zero-padded to a whole number of blocks.
700fn padded(data: &[u8]) -> Zeroizing<Vec<u8>> {
701    let mut v = Zeroizing::new(data.to_vec());
702    v.resize(data.len().div_ceil(BLOCK).saturating_mul(BLOCK), 0);
703    v
704}

712/// Compare without stopping at the first difference.
713fn same(a: &[u8], b: &[u8]) -> bool {
714    a.len() == b.len() && a.iter().zip(b).fold(0u8, |acc, (x, y)| acc | (x ^ y)) == 0

717/// AES with a 128-, 192- or 256-bit key.
718enum Aes {
719    A128(aes::Aes128),
720    A192(aes::Aes192),
721    A256(aes::Aes256),
722}
723
724impl Aes {
725    fn new(key: &[u8]) -> Result<Aes, DocxError> {
726        let err = |_| bad("bad AES key");
727        Ok(match key.len() {
728            16 => Aes::A128(aes::Aes128::new_from_slice(key).map_err(err)?),
729            24 => Aes::A192(aes::Aes192::new_from_slice(key).map_err(err)?),
730            32 => Aes::A256(aes::Aes256::new_from_slice(key).map_err(err)?),
731            n => return Err(bad(format!("unsupported AES key length {} bits", n * 8))),
732        })
733    }
734
735    fn encrypt(&self, b: &mut aes::Block) {
736        match self {
737            Aes::A128(c) => c.encrypt_block(b),
738            Aes::A192(c) => c.encrypt_block(b),
739            Aes::A256(c) => c.encrypt_block(b),
740        }
741    }
742
743    fn decrypt(&self, b: &mut aes::Block) {
744        match self {
745            Aes::A128(c) => c.decrypt_block(b),
746            Aes::A192(c) => c.decrypt_block(b),
747            Aes::A256(c) => c.decrypt_block(b),
748        }
749    }
750}

752/// AES-CBC without padding: `data` must be whole blocks.
753fn cbc_encrypt(key: &[u8], iv: &[u8], data: &[u8]) -> Result<Vec<u8>, DocxError> {
754    let aes = Aes::new(key)?;
755    let (blocks, rest) = data.as_chunks::<BLOCK>();
756    let mut prev: [u8; BLOCK] = iv.try_into().map_err(|_| bad("bad initialization vector"))?;
757    if !rest.is_empty() {
758        return Err(bad("encrypted data isn't a whole number of blocks"));
759    }
760    let mut out = Vec::with_capacity(data.len());
761    for b in blocks {
762        let mut x = aes::Block::default();
763        for ((o, p), v) in x.iter_mut().zip(b).zip(&prev) {
764            *o = p ^ v;
765        }
766        aes.encrypt(&mut x);
767        prev.copy_from_slice(&x);
768        out.extend_from_slice(&x);
769    }
770    Ok(out)
771}
772
773/// AES-CBC decryption without padding.
774fn cbc_decrypt(key: &[u8], iv: &[u8], data: &[u8]) -> Result<Vec<u8>, DocxError> {
775    let aes = Aes::new(key)?;
776    let (blocks, rest) = data.as_chunks::<BLOCK>();
777    if !rest.is_empty() {
778        return Err(bad("encrypted data isn't a whole number of blocks"));
779    }
780    let mut prev: [u8; BLOCK] = iv.try_into().map_err(|_| bad("bad initialization vector"))?;
781    let mut out = Vec::with_capacity(data.len());
782    for b in blocks {
783        let mut x = aes::Block::default();
784        x.copy_from_slice(b);
785        aes.decrypt(&mut x);
786        for (o, v) in x.iter_mut().zip(&prev) {
787            *o ^= v;
788        }
789        prev = *b;
790        out.extend_from_slice(&x);
791    }
792    Ok(out)
793}

827fn parse_info(info: &[u8]) -> Result<Info, DocxError> {
828    let (major, minor) = match info.get(..4) {
829        Some([a, b, c, d]) => (u16::from_le_bytes([*a, *b]), u16::from_le_bytes([*c, *d])),
830        _ => return Err(bad("EncryptionInfo is truncated")),
831    };
832    match (major, minor) {
833        (4, 4) => {}
834        (3 | 4, 3) => return Err(bad("this file uses extensible encryption, which WordCraft can't open")),
835        _ => return Err(bad(format!("unknown encryption version {major}.{minor}"))),
836    }
837    let root = xml::parse(info.get(8..).unwrap_or(&[]))?;
838    if root.local() != "encryption" {
839        return Err(bad("EncryptionInfo has no <encryption> element"));
840    }
841    let key_data = root.els().find(|e| e.local() == "keyData").ok_or_else(|| bad("EncryptionInfo has no keyData"))?;
842    let kd = params(key_data)?;
843    // The integrity check is mandatory (§2.3.4.14): a file is not trusted to opt out of it.
844    let di = root
845        .els()
846        .find(|e| e.local() == "dataIntegrity")
847        .ok_or_else(|| bad("EncryptionInfo has no dataIntegrity, so the document cannot be checked for tampering"))?;
848    let integrity = (b64_attr(di, "encryptedHmacKey")?, b64_attr(di, "encryptedHmacValue")?);
849    // The password key encryptor is the `encryptedKey` with a spin count (a certificate key
850    // encryptor has none).
851    let encryptors = root.els().find(|e| e.local() == "keyEncryptors").ok_or_else(|| bad("EncryptionInfo has no keyEncryptors"))?;
852    let ek = encryptors
853        .els()
854        .filter(|e| e.local() == "keyEncryptor")
855        .flat_map(|e| e.els())
856        .find(|e| e.local() == "encryptedKey" && e.attr("spinCount").is_some())
857        .ok_or_else(|| bad("this file isn't protected with a password (only certificate encryption is listed)"))?;
858    let pe = params(ek)?;
859    let spin_count: u32 = ek.attr("spinCount").and_then(|s| s.trim().parse().ok()).ok_or_else(|| bad("bad spinCount"))?;
860    if spin_count > MAX_SPIN_COUNT {
861        return Err(bad(format!("spinCount {spin_count} is over the allowed maximum")));
862    }
863    Ok(Info {
864        hash: kd.hash,
865        salt: kd.salt,
866        block_size: kd.block_size,
867        key_bytes: kd.key_bytes,
868        integrity,
869        password: PasswordEncryptor {
870            hash: pe.hash,
871            salt: pe.salt,
872            salt_size: pe.salt_size,
873            block_size: pe.block_size,
874            key_bytes: pe.key_bytes,
875            spin_count,
876            verifier_input: b64_attr(ek, "encryptedVerifierHashInput")?,
877            verifier_hash: b64_attr(ek, "encryptedVerifierHashValue")?,
878            key_value: b64_attr(ek, "encryptedKeyValue")?,
879        },
880    })
881}
882
883fn params(e: &xml::El) -> Result<Params, DocxError> {
884    let num = |k: &str| -> Result<usize, DocxError> { e.attr(k).and_then(|s| s.trim().parse::<usize>().ok()).ok_or_else(|| bad(format!("bad {k}"))) };
885    let cipher = e.attr("cipherAlgorithm").unwrap_or("");
886    if !cipher.eq_ignore_ascii_case("AES") {
887        return Err(bad(format!("the {cipher:?} cipher isn't supported")));
888    }
889    let chaining = e.attr("cipherChaining").unwrap_or("");
890    if chaining != "ChainingModeCBC" {
891        return Err(bad(format!("the {chaining:?} chaining mode isn't supported")));
892    }
893    let name = e.attr("hashAlgorithm").unwrap_or("");
894    let hash = Hash::parse(name).ok_or_else(|| bad(format!("the {name:?} hash algorithm isn't supported")))?;
895    if num("hashSize")? != hash.size() {
896        return Err(bad("hashSize doesn't match the hash algorithm"));
897    }
898    let block_size = num("blockSize")?;
899    if block_size != BLOCK {
900        return Err(bad(format!("AES needs a block size of {BLOCK}, not {block_size}")));
901    }
902    let key_bits = num("keyBits")?;
903    if !matches!(key_bits, 128 | 192 | 256) {
904        return Err(bad(format!("unsupported AES key size {key_bits}")));
905    }
906    let salt = b64_attr(e, "saltValue")?;
907    let salt_size = num("saltSize")?;
908    if salt.is_empty() || salt.len() != salt_size {
909        return Err(bad("the salt doesn't match saltSize"));
910    }
911    Ok(Params { hash, salt, salt_size, block_size, key_bytes: key_bits / 8 })
912}

914fn b64_attr(e: &xml::El, k: &str) -> Result<Vec<u8>, DocxError> {
915    let s = e.attr(k).ok_or_else(|| bad(format!("{k} is missing")))?;
916    let s: String = s.chars().filter(|c| !c.is_ascii_whitespace()).collect();
917    base64::engine::general_purpose::STANDARD.decode(s).map_err(|_| bad(format!("{k} isn't valid base64")))
918}

920/// A UNICODE-LP-P4 string (§2.1.2): byte length, UTF-16LE, zero-padded to 4 bytes.
921fn lp_p4(out: &mut Vec<u8>, s: &str) {
922    let units: Vec<u8> = s.encode_utf16().flat_map(u16::to_le_bytes).collect();
923    out.extend_from_slice(&(units.len() as u32).to_le_bytes());
924    out.extend_from_slice(&units);
925    while !out.len().is_multiple_of(4) {
926        out.push(0);
927    }
928}

930/// `\x06DataSpaces\Version` (§2.1.5).
931fn data_space_version() -> Vec<u8> {

933    lp_p4(&mut v, "Microsoft.Container.DataSpaces");
934    for _ in 0..3 {
935        v.extend_from_slice(&[1, 0, 0, 0]);
936    }

940/// `\x06DataSpaces\DataSpaceMap` (§2.1.6, §2.3.4.1): EncryptedPackage → StrongEncryptionDataSpace.
941fn data_space_map() -> Vec<u8> {
942    let mut entry = Vec::new();
943    entry.extend_from_slice(&1u32.to_le_bytes());
944    entry.extend_from_slice(&0u32.to_le_bytes());
945    lp_p4(&mut entry, "EncryptedPackage");
946    lp_p4(&mut entry, "StrongEncryptionDataSpace");
947    let mut v = Vec::new();
948    v.extend_from_slice(&8u32.to_le_bytes());
949    v.extend_from_slice(&1u32.to_le_bytes());
950    v.extend_from_slice(&((entry.len() + 4) as u32).to_le_bytes());
951    v.extend_from_slice(&entry);
952    v
953}

959    v.extend_from_slice(&1u32.to_le_bytes());

964/// `\x06DataSpaces\TransformInfo\StrongEncryptionTransform\x06Primary` (§2.1.8, §2.1.9,
965/// §2.3.4.3): the encryption transform, with a null EncryptionName as agile encryption requires.
966fn primary_transform() -> Vec<u8> {
967    let mut id = Vec::new();
968    lp_p4(&mut id, "{FF9A3F03-56EF-4613-BDD5-5A41C1D07246}");
969    let mut v = Vec::new();
970    // TransformLength: the bytes before TransformName.
971    v.extend_from_slice(&((8 + id.len()) as u32).to_le_bytes());
972    v.extend_from_slice(&1u32.to_le_bytes());
973    v.extend_from_slice(&id);
974    lp_p4(&mut v, "Microsoft.Container.EncryptionTransform");
975    for _ in 0..3 {
976        v.extend_from_slice(&[1, 0, 0, 0]);
977    }
978    // EncryptionTransformInfo: null name, block size, cipher mode 0, reserved 4.
979    v.extend_from_slice(&0u32.to_le_bytes());
980    v.extend_from_slice(&(BLOCK as u32).to_le_bytes());
981    v.extend_from_slice(&0u32.to_le_bytes());
982    v.extend_from_slice(&4u32.to_le_bytes());
983    v
984}
