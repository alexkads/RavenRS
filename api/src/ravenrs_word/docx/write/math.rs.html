11/// Write an equation: `m:oMath`, inside `m:oMathPara` for a display equation. The source OMML
12/// is written back unchanged when the equation came from a file. `color` (the equation's run
13/// colour) goes on every math run without its own: OMML has no equation-level colour (#1467).
14pub fn write_equation(w: &mut W, linear: &str, display: bool, math: &Math, color: Option<Rgb>) {
15    let tinted;
16    let math = if color == omml_color(&math.omml) {
17        math
18    } else {
19        // The kept source XML has the old colour baked in: write from the nodes instead.
20        let mut m = math.clone();
21        if m.nodes.is_empty() {
22            m.nodes = parse_linear(linear);
23        }
24        if let Some(c) = color {
25            crate::doc::math_edit::each_run_mut(&mut m.nodes, &mut |r| {
26                r.color.get_or_insert(c);
27            });
28        }
29        m.omml.clear();
30        tinted = m;
31        &tinted
32    };
33    if display {
34        w.open("m:oMathPara", &[]);
35        let jc = match math.jc {
36            MathJc::CenterGroup => None,
37            MathJc::Center => Some("center"),
38            MathJc::Left => Some("left"),
39            MathJc::Right => Some("right"),
40        };
41        if let Some(jc) = jc {
42            w.open("m:oMathParaPr", &[]);
43            w.empty("m:jc", &[("m:val", jc)]);
44            w.close("m:oMathParaPr");
45        }
46    }
47    if !math.omml.is_empty() && math.omml.starts_with("<m:oMath") && crate::docx::xml::parse(wrap_ns(&math.omml).as_bytes()).is_ok() {
48        w.raw(&math.omml);
49    } else {
50        let parsed;
51        let nodes: &Arg = if math.nodes.is_empty() {
52            parsed = parse_linear(linear);
53            &parsed
54        } else {
55            &math.nodes
56        };
57        w.open("m:oMath", &[]);
58        arg_inner(w, nodes, 0);
59        w.close("m:oMath");
60    }
61    if display {
62        w.close("m:oMathPara");
63    }
64}

66/// Count the equation's numbered rows; when some use an automatic number (`#` alone), a copy
67/// with the numbers written out as `(n)`.
68pub fn resolve_numbers(math: &Math, display: bool, counter: &mut u32) -> Option<Math> {
69    if !display {
70        return None;
71    }
72    let [MNode::EqArr { rows }] = math.nodes.as_slice() else { return None };
73    let mut out = rows.clone();
74    let mut changed = false;
75    for row in out.iter_mut() {
76        if !crate::doc::math::has_number_mark(row) {
77            continue;
78        }
79        *counter = counter.saturating_add(1);
80        // Nothing after the `#`: an automatic number.
81        let mut after = false;
82        let mut empty = true;
83        for n in row.iter() {
84            match n {
85                MNode::Run(r) if !after => {
86                    if let Some(k) = r.text.find('#') {
87                        after = true;
88                        if !r.text.get(k + 1..).unwrap_or("").is_empty() {
89                            empty = false;
90                        }
91                    }
92                }
93                _ if after => empty = false,
94                _ => {}
95            }
96        }
97        if empty {
98            let n = vec![MNode::Run(MRun::new(counter.to_string()))];
99            row.push(MNode::Delim { beg: Some('('), end: Some(')'), sep: None, grow: true, shp_match: false, elems: vec![n] });
100            changed = true;
101        }
102    }
103    changed.then(|| Math { nodes: vec![MNode::EqArr { rows: out }], omml: String::new(), ..math.clone() })
104}

106/// The equation colour the kept source OMML carries, as the reader lifts it.
107fn omml_color(omml: &str) -> Option<Rgb> {
108    if omml.is_empty() {
109        return None;
110    }
111    let root = crate::docx::xml::parse(wrap_ns(omml).as_bytes()).ok()?;
112    let e = root.child("m:oMath")?;
113    crate::docx::read::math::lift_color(&mut crate::docx::read::math::read_omath(e, MathJc::default()))
114}

116/// The kept XML with namespace declarations, for validating it on its own.
117fn wrap_ns(xml: &str) -> String {
118    let decl: String = crate::docx::xml::body_ns().iter().filter(|(k, _)| k.starts_with("xmlns:")).map(|(k, v)| format!(" {k}=\"{v}\"")).collect();
119    format!("<w:x{decl}>{xml}</w:x>")
120}

122/// The settings part's `m:mathPr` (CT_MathPr order; `m:wrapIndent` and `m:wrapRight` are a choice).
123pub fn math_pr(w: &mut W, m: &MathProps) {
124    w.open("m:mathPr", &[]);
125    let bin = match m.brk_bin {
126        BrkBin::Before => "before",
127        BrkBin::After => "after",
128        BrkBin::Repeat => "repeat",
129    };
130    val(w, "m:brkBin", bin);
131    let sub = match m.brk_bin_sub {
132        BrkBinSub::MinusMinus => "--",
133        BrkBinSub::MinusPlus => "-+",
134        BrkBinSub::PlusMinus => "+-",
135    };
136    val(w, "m:brkBinSub", sub);
137    if m.wrap_right {
138        w.empty("m:wrapRight", &[]);
139    } else {
140        let indent = if m.wrap_indent.is_finite() { m.wrap_indent.clamp(0.0, 1584.0) } else { crate::doc::math::DEFAULT_WRAP_INDENT };
141        val(w, "m:wrapIndent", &crate::docx::units::twips(indent));
142    }
143    w.close("m:mathPr");
144}

150fn arg(w: &mut W, name: &str, a: &[MNode], depth: usize) {
151    if a.is_empty() {
152        w.empty(name, &[]);
153        return;
154    }
155    w.open(name, &[]);
156    arg_inner(w, a, depth + 1);
157    w.close(name);
158}

160fn arg_inner(w: &mut W, a: &[MNode], depth: usize) {
161    if depth > MAX_DEPTH {
162        return;
163    }
164    for n in a {
165        node(w, n, depth);
166    }
167}

169fn ctrl(w: &mut W) {
170    w.open("m:ctrlPr", &[]);
171    w.open("w:rPr", &[]);
172    w.empty("w:rFonts", &[("w:ascii", "Cambria Math"), ("w:hAnsi", "Cambria Math")]);
173    w.empty("w:i", &[]);
174    w.close("w:rPr");
175    w.close("m:ctrlPr");

183fn write_matrix(w: &mut W, n: &MNode, depth: usize) {
184    let MNode::Matrix { rows, col_jc } = n else { return };
185    // Every row is padded to the widest one, so a ragged matrix is clamped to the reader's
186    // caps rather than written as rows × widest cells (#3643).
187    let rows = rows.get(..rows.len().min(MAX_MATRIX_ROWS)).unwrap_or_default();
188    let cols = rows.iter().map(|r| r.len()).max().unwrap_or(0).clamp(1, MAX_MATRIX_COLS);
189    w.open("m:m", &[]);
190    w.open("m:mPr", &[]);
191    w.open("m:mcs", &[]);
192    for c in 0..cols {
193        let jc = match col_jc.get(c).copied().unwrap_or_default() {
194            ColJc::Center => "center",
195            ColJc::Left => "left",
196            ColJc::Right => "right",
197        };
198        w.open("m:mc", &[]);
199        w.open("m:mcPr", &[]);
200        val(w, "m:count", "1");
201        val(w, "m:mcJc", jc);
202        w.close("m:mcPr");
203        w.close("m:mc");
204    }
205    w.close("m:mcs");
206    ctrl(w);
207    w.close("m:mPr");
208    for row in rows {
209        w.open("m:mr", &[]);
210        for c in 0..cols {
211            arg(w, "m:e", row.get(c).map(|v| v.as_slice()).unwrap_or(&[]), depth);
212        }
213        w.close("m:mr");
214    }
215    w.close("m:m");
216}

218fn write_nary(w: &mut W, n: &MNode, depth: usize) {
219    let MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide, sub, sup, e } = n else { return };
220    w.open("m:nary", &[]);
221    w.open("m:naryPr", &[]);
222    if *chr != '∫' {
223        char_prop(w, "m:chr", Some(*chr));
224    }
225    match lim_loc {
226        Some(LimLoc::UndOvr) => val(w, "m:limLoc", "undOvr"),
227        Some(LimLoc::SubSup) => val(w, "m:limLoc", "subSup"),
228        None => {}
229    }
230    if *grow {
231        val(w, "m:grow", "1");
232    }
233    if *sub_hide {
234        val(w, "m:subHide", "1");
235    }
236    if *sup_hide {
237        val(w, "m:supHide", "1");
238    }
239    ctrl(w);
240    w.close("m:naryPr");
241    arg(w, "m:sub", sub, depth);
242    arg(w, "m:sup", sup, depth);
243    arg(w, "m:e", e, depth);
244    w.close("m:nary");
245}

247fn node(w: &mut W, n: &MNode, depth: usize) {
248    match n {
249        MNode::Run(r) => run(w, r),
250        MNode::Frac { kind, num, den } => {
251            w.open("m:f", &[]);
252            w.open("m:fPr", &[]);
253            match kind {
254                FracKind::Bar => {}
255                FracKind::Skewed => val(w, "m:type", "skw"),
256                FracKind::Linear => val(w, "m:type", "lin"),
257                FracKind::NoBar => val(w, "m:type", "noBar"),
258            }
259            ctrl(w);
260            w.close("m:fPr");
261            arg(w, "m:num", num, depth);
262            arg(w, "m:den", den, depth);
263            w.close("m:f");
264        }
265        MNode::Script { kind, base, sub, sup } => {
266            let (el, pr) = match kind {
267                ScriptKind::Sup => ("m:sSup", "m:sSupPr"),
268                ScriptKind::Sub => ("m:sSub", "m:sSubPr"),
269                ScriptKind::SubSup => ("m:sSubSup", "m:sSubSupPr"),
270                ScriptKind::Pre => ("m:sPre", "m:sPrePr"),
271            };
272            w.open(el, &[]);
273            w.open(pr, &[]);
274            ctrl(w);
275            w.close(pr);
276            if *kind == ScriptKind::Pre {
277                arg(w, "m:sub", sub, depth);
278                arg(w, "m:sup", sup, depth);
279                arg(w, "m:e", base, depth);
280            } else {
281                arg(w, "m:e", base, depth);
282                if matches!(kind, ScriptKind::Sub | ScriptKind::SubSup) {
283                    arg(w, "m:sub", sub, depth);
284                }
285                if matches!(kind, ScriptKind::Sup | ScriptKind::SubSup) {
286                    arg(w, "m:sup", sup, depth);
287                }
288            }
289            w.close(el);
290        }
291        MNode::Rad { deg, deg_hide, e } => {
292            w.open("m:rad", &[]);
293            w.open("m:radPr", &[]);
294            if *deg_hide {
295                val(w, "m:degHide", "1");
296            }
297            ctrl(w);
298            w.close("m:radPr");
299            arg(w, "m:deg", deg, depth);
300            arg(w, "m:e", e, depth);
301            w.close("m:rad");
302        }
303        MNode::Nary { .. } => write_nary(w, n, depth),
304        MNode::Delim { beg, end, sep, grow, shp_match, elems } => {
305            w.open("m:d", &[]);
306            w.open("m:dPr", &[]);
307            if *beg != Some('(') {
308                char_prop(w, "m:begChr", *beg);
309            }
310            if *sep != Some('|') {
311                char_prop(w, "m:sepChr", *sep);
312            }
313            if *end != Some(')') {
314                char_prop(w, "m:endChr", *end);
315            }
316            if !grow {
317                val(w, "m:grow", "0");
318            }
319            if *shp_match {
320                val(w, "m:shp", "match");
321            }
322            ctrl(w);
323            w.close("m:dPr");
324            for e in elems {
325                arg(w, "m:e", e, depth);
326            }
327            w.close("m:d");
328        }
329        MNode::Func { name, e } => {
330            w.open("m:func", &[]);
331            w.open("m:funcPr", &[]);
332            ctrl(w);
333            w.close("m:funcPr");
334            arg(w, "m:fName", name, depth);
335            arg(w, "m:e", e, depth);
336            w.close("m:func");
337        }
338        MNode::Lim { upper, e, lim } => {
339            let (el, pr) = if *upper { ("m:limUpp", "m:limUppPr") } else { ("m:limLow", "m:limLowPr") };
340            w.open(el, &[]);
341            w.open(pr, &[]);
342            ctrl(w);
343            w.close(pr);
344            arg(w, "m:e", e, depth);
345            arg(w, "m:lim", lim, depth);
346            w.close(el);
347        }
348        MNode::Acc { chr, e } => {
349            w.open("m:acc", &[]);
350            w.open("m:accPr", &[]);
351            char_prop(w, "m:chr", Some(*chr));
352            ctrl(w);
353            w.close("m:accPr");
354            arg(w, "m:e", e, depth);
355            w.close("m:acc");
356        }
357        MNode::Bar { top, e } => {
358            w.open("m:bar", &[]);
359            w.open("m:barPr", &[]);
360            val(w, "m:pos", if *top { "top" } else { "bot" });
361            ctrl(w);
362            w.close("m:barPr");
363            arg(w, "m:e", e, depth);
364            w.close("m:bar");
365        }
366        MNode::BorderBox { hide, strike, e } => {
367            w.open("m:borderBox", &[]);
368            w.open("m:borderBoxPr", &[]);
369            for (on, name) in hide.iter().zip(["m:hideTop", "m:hideBot", "m:hideLeft", "m:hideRight"]) {
370                if *on {
371                    val(w, name, "1");
372                }
373            }
374            for (on, name) in strike.iter().zip(["m:strikeH", "m:strikeV", "m:strikeBLTR", "m:strikeTLBR"]) {
375                if *on {
376                    val(w, name, "1");
377                }
378            }
379            ctrl(w);
380            w.close("m:borderBoxPr");
381            arg(w, "m:e", e, depth);
382            w.close("m:borderBox");
383        }
384        MNode::Boxed { e } => {
385            w.open("m:box", &[]);
386            w.open("m:boxPr", &[]);
387            ctrl(w);
388            w.close("m:boxPr");
389            arg(w, "m:e", e, depth);
390            w.close("m:box");
391        }
392        MNode::GroupChr { chr, top, e } => {
393            w.open("m:groupChr", &[]);
394            w.open("m:groupChrPr", &[]);
395            char_prop(w, "m:chr", Some(*chr));
396            val(w, "m:pos", if *top { "top" } else { "bot" });
397            val(w, "m:vertJc", if *top { "bot" } else { "top" });
398            ctrl(w);
399            w.close("m:groupChrPr");
400            arg(w, "m:e", e, depth);
401            w.close("m:groupChr");
402        }
403        MNode::EqArr { rows } => {
404            w.open("m:eqArr", &[]);
405            w.open("m:eqArrPr", &[]);
406            // Numbered rows (`#`) stretch across the line so the number reaches the margin.
407            if rows.iter().any(|r| crate::doc::math::has_number_mark(r)) {
408                val(w, "m:maxDist", "1");
409            }
410            ctrl(w);
411            w.close("m:eqArrPr");
412            for r in rows {
413                arg(w, "m:e", r, depth);
414            }
415            w.close("m:eqArr");
416        }
417        MNode::Matrix { .. } => write_matrix(w, n, depth),
418        MNode::Phant { show, zero_wid, zero_asc, zero_desc, e } => {
419            w.open("m:phant", &[]);
420            w.open("m:phantPr", &[]);
421            val(w, "m:show", if *show { "1" } else { "0" });
422            for (on, name) in [(*zero_wid, "m:zeroWid"), (*zero_asc, "m:zeroAsc"), (*zero_desc, "m:zeroDesc")] {
423                if on {
424                    val(w, name, "1");
425                }
426            }
427            ctrl(w);
428            w.close("m:phantPr");
429            arg(w, "m:e", e, depth);
430            w.close("m:phant");
431        }
432    }
433}

435fn run(w: &mut W, r: &MRun) {
436    w.open("m:r", &[]);
437    let sty = r.sty.map(|s| match s {
438        MSty::Plain => "p",
439        MSty::Bold => "b",
440        MSty::Italic => "i",
441        MSty::BoldItalic => "bi",
442    });
443    let scr = match r.scr {
444        MScr::Roman => None,
445        MScr::Script => Some("script"),
446        MScr::Fraktur => Some("fraktur"),
447        MScr::DoubleStruck => Some("double-struck"),
448        MScr::SansSerif => Some("sans-serif"),
449        MScr::Monospace => Some("monospace"),
450    };
451    if sty.is_some() || scr.is_some() || r.nor || r.lit || r.brk.is_some() {
452        w.open("m:rPr", &[]);
453        if r.lit {
454            w.empty("m:lit", &[]);
455        }
456        if r.nor {
457            w.empty("m:nor", &[]);
458        }
459        if let Some(s) = scr {
460            val(w, "m:scr", s);
461        }
462        if let Some(s) = sty {
463            val(w, "m:sty", s);
464        }
465        match r.brk {
466            Some(0) => w.empty("m:brk", &[]),
467            Some(n) => w.empty("m:brk", &[("m:alnAt", &n.to_string())]),
468            None => {}
469        }
470        w.close("m:rPr");
471    }
472    w.open("w:rPr", &[]);
473    let font = r.font.as_deref().filter(|_| r.nor).unwrap_or("Cambria Math");
474    w.empty("w:rFonts", &[("w:ascii", font), ("w:hAnsi", font)]);
475    if let Some(c) = r.color {
476        w.val("w:color", &c.hex());
477    }
478    if let Some(sz) = r.size {
479        let hp = (sz * 2.0).round().clamp(2.0, 3276.0) as i64;
480        w.val("w:sz", &hp.to_string());
481        w.val("w:szCs", &hp.to_string());
482    }
483    w.close("w:rPr");
484    w.leaf("m:t", &[("xml:space", "preserve")], &r.text);
485    w.close("m:r");
