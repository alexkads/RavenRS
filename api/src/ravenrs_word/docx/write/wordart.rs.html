1//! WordArt text effects as `w14:` run properties ([MS-DOCX] §2.6.1), in the schema's order:
2//! glow, shadow, reflection, text outline, text fill. Lengths are EMUs, angles 60000ths of a
3//! degree, percentages 1000ths of a percent.
4
5use crate::doc::props::Rgb;
6use crate::doc::wordart::{TextEffects, TextFill};
7
8use crate::docx::units::{emu, n};
9use crate::docx::xml::W;
10
11fn pct(v: f32) -> String {
12    n((crate::geom::finite(v).clamp(0.0, 100.0) * 1000.0).round() as i64)
13}
14
15fn ang(deg: f32) -> String {
16    n((crate::geom::finite(deg).rem_euclid(360.0) * 60_000.0).round().clamp(0.0, 21_599_999.0) as i64)
17}
18
19fn len(pt: f32) -> String {
20    emu(crate::geom::finite(pt).clamp(0.0, 2000.0))
21}
22
23/// `w14:srgbClr` with its transparency (`w14:alpha` is the transparency, not the opacity).
24fn color(w: &mut W, c: Rgb, transparency: f32) {
25    if transparency > 0.0 {
26        w.open("w14:srgbClr", &[("w14:val", &c.hex())]);
27        w.empty("w14:alpha", &[("w14:val", &pct(transparency))]);
28        w.close("w14:srgbClr");
29    } else {
30        w.empty("w14:srgbClr", &[("w14:val", &c.hex())]);
31    }
32}
33
34fn fill(w: &mut W, f: &TextFill) {
35    match f {
36        TextFill::None => w.empty("w14:noFill", &[]),
37        TextFill::Solid { color: c, transparency } => {
38            w.open("w14:solidFill", &[]);
39            color(w, *c, *transparency);
40            w.close("w14:solidFill");
41        }
42        TextFill::Gradient { stops, angle } => {
43            w.open("w14:gradFill", &[]);
44            w.open("w14:gsLst", &[]);
45            for s in stops {
46                w.open("w14:gs", &[("w14:pos", &pct(s.pos))]);
47                color(w, s.color, s.transparency);
48                w.close("w14:gs");
49            }
50            w.close("w14:gsLst");
51            w.empty("w14:lin", &[("w14:ang", &ang(*angle)), ("w14:scaled", "0")]);
52            w.close("w14:gradFill");
53        }
54    }
55}

57/// The effects of `fx` (sanitized first).
58pub fn rpr_effects(w: &mut W, fx: &TextEffects) {
59    let fx = fx.sanitized();
60    if let Some(g) = fx.glow {
61        w.open("w14:glow", &[("w14:rad", &len(g.size))]);
62        color(w, g.color, g.transparency);
63        w.close("w14:glow");
64    }
65    if let Some(s) = fx.shadow {
66        w.open(
67            "w14:shadow",
68            &[
69                ("w14:blurRad", &len(s.blur)),
70                ("w14:dist", &len(s.distance)),
71                ("w14:dir", &ang(s.angle)),
72                ("w14:sx", "100000"),
73                ("w14:sy", "100000"),
74                ("w14:kx", "0"),
75                ("w14:ky", "0"),
76                ("w14:algn", "ctr"),
77            ],
78        );
79        color(w, s.color, s.transparency);
80        w.close("w14:shadow");
81    }
82    if let Some(r) = fx.reflection {
83        w.empty(
84            "w14:reflection",
85            &[
86                ("w14:blurRad", &len(r.blur)),
87                ("w14:stA", &pct(100.0 - r.transparency)),
88                ("w14:stPos", "0"),
89                ("w14:endA", "0"),
90                ("w14:endPos", &pct(r.size)),
91                ("w14:dist", &len(r.distance)),
92                ("w14:dir", "5400000"),
93                ("w14:fadeDir", "5400000"),
94                ("w14:sx", "100000"),
95                ("w14:sy", "-100000"),
96                ("w14:kx", "0"),
97                ("w14:ky", "0"),
98                ("w14:algn", "bl"),
99            ],
100        );
101    }
102    if let Some(o) = fx.outline {
103        w.open("w14:textOutline", &[("w14:w", &len(o.width)), ("w14:cap", "flat"), ("w14:cmpd", "sng"), ("w14:algn", "ctr")]);
104        match o.color {
105            Some(c) => {
106                w.open("w14:solidFill", &[]);
107                color(w, c, o.transparency);
108                w.close("w14:solidFill");
109            }
110            None => w.empty("w14:noFill", &[]),
111        }
112        w.empty("w14:prstDash", &[("w14:val", "solid")]);
113        w.empty("w14:round", &[]);
114        w.close("w14:textOutline");
115    }
116    if let Some(f) = &fx.fill {
117        w.open("w14:textFill", &[]);
118        fill(w, f);
119        w.close("w14:textFill");
120    }
121}
