1//! Run direction for mixed right-to-left and left-to-right text (#1543).
2//!
3//! WordCraft lays a paragraph out with the Unicode Bidirectional Algorithm (UAX #9) from its
4//! text and direction alone, so a run need not say its direction. Word orders a run's characters
5//! by the run's `w:rtl` (ECMA-376 Part 1 §17.3.2.30): text in a run without it is laid out left
6//! to right, so Hebrew words on either side of a Latin name, and the punctuation between them,
7//! change places. A run that doesn't say its direction is therefore split where the resolved
8//! level changes parity, and the right-to-left pieces are written `w:rtl`, as Word writes text
9//! typed in a right-to-left paragraph. Runs that say their direction, and paragraphs without
10//! right-to-left characters, are written as they are.
11//!
12//! Word gives right-to-left text without a bidi language (`w:lang w:bidi`, §17.3.2.20) its
13//! default complex-script language, Arabic, so Hebrew would proof and show as Arabic. The pieces
14//! marked `w:rtl` therefore also say their language, unless the run or its styles already do.
15
16use std::borrow::Cow;
17use std::ops::Range;

22/// The language Word expects for right-to-left text of `c`'s script, for the scripts that have
23/// one usual language.
24fn script_lang(c: char) -> Option<&'static str> {
25    match u32::from(c) {
26        0x0590..=0x05FF | 0xFB1D..=0xFB4F => Some("he-IL"),
27        0x0600..=0x06FF | 0x0750..=0x077F | 0x0870..=0x08FF | 0xFB50..=0xFDFF | 0xFE70..=0xFEFF => Some("ar-SA"),
28        0x0700..=0x074F => Some("syr-SY"),
29        0x0780..=0x07BF => Some("dv-MV"),
30        _ => None,
31    }
32}
33
34/// The language of the first right-to-left letter of `text` with a usual language.
35fn text_lang(text: &str) -> Option<&'static str> {
36    text.chars().find_map(script_lang)
37}
38
39/// A language tag of a right-to-left language (`he-IL`, `yi`, `fa-IR`, `ar-EG`…)?
40fn is_rtl_lang(tag: &str) -> bool {
41    let primary = tag.split(['-', '_']).next().unwrap_or("").to_ascii_lowercase();
42    matches!(primary.as_str(), "he" | "iw" | "yi" | "ar" | "fa" | "ur" | "ps" | "sd" | "ug" | "ckb" | "syr" | "dv" | "ku")
43}

45/// Does a paragraph of style `para_style` inherit a bidi language from the document defaults or
46/// its style chain? (Resolved once per paragraph.)
47pub(super) fn para_inherits_lang_bidi(styles: &StyleSheet, para_style: Option<&str>) -> bool {
48    styles.para_style_props(para_style.unwrap_or("Normal")).1.lang_bidi.is_some()
49}
50
51/// Does `c`'s character style chain give it a bidi language?
52pub(super) fn char_style_has_lang_bidi(styles: &StyleSheet, c: &CharProps) -> bool {
53    c.style.as_deref().is_some_and(|s| styles.chain(s).iter().any(|st| st.chr.lang_bidi.is_some()))
54}

56/// Add `rtl` and, when given, the bidi language `lang` to `c` where it doesn't say them.
57fn direct(c: &mut CharProps, lang: Option<&String>) {
58    c.rtl.get_or_insert(true);
59    if let Some(l) = lang
60        && c.lang_bidi.is_none()
61    {
62        c.lang_bidi = Some(l.clone());
63    }
64}

66/// `runs` (byte ranges of `text` with their props), split where the bidi level of `text` in a
67/// paragraph of direction `rtl_para` changes parity; right-to-left pieces get `rtl: Some(true)`
68/// and, unless they have one or `inherits_lang` says it comes from a style, a bidi language.
69pub(super) fn direction_spans<'a>(
70    text: &str,
71    rtl_para: bool,
72    runs: impl Iterator<Item = (Range<usize>, &'a CharProps)>,
73    inherits_lang: impl Fn(&CharProps) -> bool,
74) -> Vec<(Range<usize>, Cow<'a, CharProps>)> {
75    if !crate::doc::bidi::has_rtl(text) {
76        return runs.map(|(r, p)| (r, Cow::Borrowed(p))).collect();
77    }
78    let level = if rtl_para { unicode_bidi::Level::rtl() } else { unicode_bidi::Level::ltr() };
79    let info = unicode_bidi::ParagraphBidiInfo::new(text, Some(level));
80    let odd = |at: usize| info.levels.get(at).is_some_and(|l| l.is_rtl());
81    let para_lang = text_lang(text);
82    let mut out = Vec::new();
83    for (range, props) in runs {
84        let Some(run) = text.get(range.clone()).filter(|_| props.rtl.is_none()) else {
85            out.push((range, Cow::Borrowed(props)));
86            continue;
87        };
88        let named = props.lang_bidi.is_some() || inherits_lang(props);
89        let piece = |r: Range<usize>, rtl: bool| {
90            if !rtl {
91                return (r, Cow::Borrowed(props));
92            }
93            let lang = if named {
94                None
95            } else {
96                let own = props.lang.clone().filter(|l| is_rtl_lang(l));
97                own.or_else(|| text.get(r.clone()).and_then(text_lang).or(para_lang).map(str::to_string))
98            };
99            let mut p = props.clone();
100            direct(&mut p, lang.as_ref());
101            // A tracked formatting change's "before" copy gets the same, so the change lists
102            // nothing the author didn't do.
103            if let Some(ch) = p.fmt_change.as_mut() {
104                direct(&mut ch.old, lang.as_ref());
105            }
106            (r, Cow::Owned(p))
107        };
108        let mut start = range.start;
109        let mut cur: Option<bool> = None;
110        for (i, _) in run.char_indices() {
111            let at = range.start.saturating_add(i);
112            let rtl = odd(at);
113            match cur {
114                Some(c) if c != rtl => {
115                    out.push(piece(start..at, c));
116                    start = at;
117                }
118                _ => {}
119            }
120            cur = Some(rtl);
121        }
122        out.push(piece(start..range.end, cur.unwrap_or(false)));
123    }
124    out
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    #[test]
132    fn hostile_ranges_never_panic() {
133        let p = CharProps::default();
134        let text = "עברית WordCraft";
135        let runs = vec![(0..3, &p), (3..1000, &p), (usize::MAX - 1..usize::MAX, &p), (2..text.len(), &p)];
136        let out = direction_spans(text, true, runs.into_iter(), |_| false);
137        assert_eq!(out.len(), 5, "bad ranges pass through unsplit: {out:?}");
138    }
139
140    #[test]
141    fn languages_by_script() {
142        assert_eq!(text_lang("(שלום"), Some("he-IL"));
143        assert_eq!(text_lang("سلام"), Some("ar-SA"));
144        assert_eq!(text_lang("ܫܠܡܐ"), Some("syr-SY"));
145        assert_eq!(text_lang(". ,"), None);
146        assert!(is_rtl_lang("he-IL") && is_rtl_lang("yi") && is_rtl_lang("FA_ir") && !is_rtl_lang("en-US") && !is_rtl_lang(""));
147    }
