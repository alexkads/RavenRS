1//! Writing the glossary document (ECMA-376 Part 1 §17.12) from `Document::glossary`, once its
2//! entries differ from the file's (an untouched glossary is written back verbatim: see
3//! `read::glossary`).
4
5use crate::doc::{Document, QUICK_PARTS_GALLERY};
6
7use super::{CT_WML, PartRels, Writer, numbering_xml, push_part, styles_xml};
8use crate::docx::package::{GLOSSARY_BEHAVIORS, GLOSSARY_CONTENT_TYPE, GLOSSARY_READ, GLOSSARY_TYPES, glossary_fingerprint, rt};
9use crate::docx::xml::{self, W};
10
11/// `values` known to `known`, as `parent` with one `tag` child each (nothing when none are).
12fn listed(w: &mut W, parent: &str, tag: &str, values: &[String], known: &[&str]) {
13    let mut values = values.iter().filter(|v| known.contains(&v.as_str())).peekable();
14    if values.peek().is_none() {
15        return;
16    }
17    w.open(parent, &[]);
18    for v in values {
19        w.empty(tag, &[("w:val", v)]);
20    }
21    w.close(parent);
22}
23
24/// Whether the glossary differs from the one read from the file (none read: an empty one).
25pub(super) fn edited(doc: &Document) -> bool {
26    let print = glossary_fingerprint(&doc.glossary);
27    match doc.passthrough.get(GLOSSARY_READ) {
28        Some(read) => read.as_slice() != print.as_bytes(),
29        None => print != glossary_fingerprint(&[]),
30    }
31}

33impl Writer<'_> {
34    /// Write the glossary part from the model, with a styles part of its own (the document's
35    /// styles), related from the main document's `rels`. Its parts live in `word/`, beside the
36    /// main document, so the relationships its stories add resolve as theirs do.
37    pub(super) fn glossary(&mut self, entries: &mut Vec<(String, Vec<u8>)>, overrides: &mut Vec<(String, String)>, rels: &mut PartRels) {
38        let doc = self.doc;
39        if doc.glossary.is_empty() {
40            return;
41        }
42        let names = self.embeds.fresh_set(&["word/glossary", "word/glossaryStyles", "word/glossaryNumbering"], ".xml");
43        let [part, styles, numbering] = names.as_slice() else { return };
44        let ns = xml::body_ns();
45        let ns_refs: Vec<(&str, &str)> = ns.iter().map(|(k, v)| (k.as_str(), v.as_str())).collect();
46        let mut prels = PartRels::default();
47        let mut w = W::new();
48        w.open("w:glossaryDocument", &ns_refs);
49        w.open("w:docParts", &[]);
50        for e in &doc.glossary {
51            w.open("w:docPart", &[]);
52            w.open("w:docPartPr", &[]);
53            w.empty("w:name", &[("w:val", &e.name)]);
54            if !e.style.is_empty() {
55                w.empty("w:style", &[("w:val", &e.style)]);
56            }
57            w.open("w:category", &[]);
58            w.empty("w:name", &[("w:val", if e.category.is_empty() { "General" } else { &e.category })]);
59            w.empty("w:gallery", &[("w:val", if e.gallery.is_empty() { QUICK_PARTS_GALLERY } else { &e.gallery })]);
60            w.close("w:category");
61            listed(&mut w, "w:types", "w:type", &e.types, &GLOSSARY_TYPES);
62            listed(&mut w, "w:behaviors", "w:behavior", &e.behaviors, &GLOSSARY_BEHAVIORS);
63            if !e.description.is_empty() {
64                w.empty("w:description", &[("w:val", &e.description)]);
65            }
66            if !e.guid.is_empty() {
67                w.empty("w:guid", &[("w:val", &e.guid)]);
68            }
69            w.close("w:docPartPr");
70            w.open("w:docPartBody", &[]);
71            if e.blocks.is_empty() {
72                w.empty("w:p", &[]);
73            } else {
74                self.blocks(&mut w, &e.blocks, &mut prels, false, 0);
75            }
76            w.close("w:docPartBody");
77            w.close("w:docPart");
78        }
79        w.close("w:docParts");
80        w.close("w:glossaryDocument");
81        let file = |p: &str| p.strip_prefix("word/").unwrap_or(p).to_string();
82        prels.add(rt::STYLES, &file(styles), false);
83        // Lists in entries need their definitions within the glossary's reach.
84        let lists = !doc.numbering.nums.is_empty() || !doc.numbering.abstracts.is_empty();
85        if lists {
86            prels.add(rt::NUMBERING, &file(numbering), false);
87        }
88        rels.add(rt::GLOSSARY, &file(part), false);
89        push_part(entries, overrides, part, w.into_bytes(), GLOSSARY_CONTENT_TYPE, prels);
90        push_part(entries, overrides, styles, styles_xml(self), &format!("{CT_WML}styles+xml"), PartRels::default());
91        if lists {
92            push_part(entries, overrides, numbering, numbering_xml(doc), &format!("{CT_WML}numbering+xml"), PartRels::default());
93        }
94    }
95}
