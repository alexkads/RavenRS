4/// The file id of the first footnote or endnote: after the separators (-1, 0) and the
5/// continuation notice (1), when the document has one.
6pub(super) fn note_id_base(doc: &Document, foot: bool) -> usize {
7    let seps = if foot { &doc.footnote_separators } else { &doc.endnote_separators };
8    if seps.continuation_notice.is_some() { 2 } else { 1 }
9}

11/// Whether the footnotes (or endnotes) part is written: there are notes, or separators of the
12/// document's own to keep.
13pub(super) fn notes_part(doc: &Document, wr: &Writer<'_>, foot: bool) -> bool {
14    let (ids, seps) = if foot { (&wr.footnotes, &doc.footnote_separators) } else { (&wr.endnotes, &doc.endnote_separators) };
15    !ids.is_empty() || !seps.is_empty()
16}

18/// `w:settings`, its children in the schema's order (CT_Settings is a sequence). `template_rel`
19/// is the id of the attached template's relationship, when there is one.
20pub(super) fn settings_xml(doc: &Document, footnotes: bool, endnotes: bool, template_rel: Option<&str>) -> Vec<u8> {
21    let s = &doc.settings;
22    let s_notes = (&doc.footnote_separators, &doc.endnote_separators);
23    let mut w = W::new();
24    let ns = xml::body_ns();
25    let ns_refs: Vec<(&str, &str)> = ns.iter().map(|(k, v)| (k.as_str(), v.as_str())).collect();
26    w.open("w:settings", &ns_refs);
27    let zoom = s.zoom.unwrap_or(crate::doc::Zoom { percent: 100, fit: None });
28    let percent = zoom.percent.clamp(10, 500).to_string();
29    match zoom.fit {
30        Some(fit) => {
31            let val = match fit {
32                crate::doc::ZoomFit::OnePage => "fullPage",
33                crate::doc::ZoomFit::PageWidth => "bestFit",
34                crate::doc::ZoomFit::TextWidth => "textFit",
35            };
36            w.empty("w:zoom", &[("w:val", val), ("w:percent", &percent)]);
37        }
38        None => w.empty("w:zoom", &[("w:percent", &percent)]),
39    }
40    // §17.15.1.70 / §17.15.1.69, right after `w:zoom` in CT_Settings.
41    if s.remove_personal_info {
42        w.empty("w:removePersonalInformation", &[]);
43    }
44    if s.remove_date_time {
45        w.empty("w:removeDateAndTime", &[]);
46    }
47    if s.page_color.is_some() || s.page_picture.is_some() {
48        w.empty("w:displayBackgroundShape", &[]);
49    }
50    if s.mirror_margins {
51        w.empty("w:mirrorMargins", &[]);
52    }
53    if let Some(id) = template_rel {
54        w.empty("w:attachedTemplate", &[("r:id", id)]);
55    }
56    if s.link_styles {
57        w.empty("w:linkStyles", &[]);
58    }
59    if s.track_changes {
60        w.empty("w:trackRevisions", &[]);
61    }
62    if !s.track_moves {
63        w.empty("w:doNotTrackMoves", &[]);
64    }
65    if let Some(p) = &s.protection {
66        let mut a: Vec<(&str, &str)> = Vec::new();
67        if p != "none" {
68            a.push(("w:edit", p));
69        }
70        if s.protect_formatting {
71            a.push(("w:formatting", "1"));
72        }
73        a.push(("w:enforcement", "1"));
74        // Only the hash attributes the reader keeps, each once.
75        for name in crate::docx::read::PROTECTION_HASH_ATTRS {
76            if let Some((_, v)) = s.protect_hash.iter().find(|(k, _)| k == name) {
77                a.push((name, v));
78            }
79        }
80        w.empty("w:documentProtection", &a);
81    }
82    w.val("w:defaultTabStop", &crate::docx::units::twips(s.default_tab.clamp(1.0, 1584.0)));
83    if s.auto_hyphenation {
84        w.empty("w:autoHyphenation", &[]);
85    }
86    // Schema order: consecutiveHyphenLimit, hyphenationZone, doNotHyphenateCaps.
87    if s.consecutive_hyphen_limit > 0 {
88        w.val("w:consecutiveHyphenLimit", &s.consecutive_hyphen_limit.min(32_767).to_string());
89    }
90    if s.hyphenation_zone.is_finite() && (s.hyphenation_zone - crate::doc::DEFAULT_HYPHENATION_ZONE).abs() > 0.01 {
91        w.val("w:hyphenationZone", &crate::docx::units::twips(s.hyphenation_zone.clamp(0.0, 1584.0)));
92    }
93    if !s.hyphenate_caps {
94        w.empty("w:doNotHyphenateCaps", &[]);
95    }
96    if s.even_odd_headers {
97        w.empty("w:evenAndOddHeaders", &[]);
98    }
99    for (tag, v) in [("w:drawingGridHorizontalSpacing", s.grid_h), ("w:drawingGridVerticalSpacing", s.grid_v)] {
100        if v.is_finite() && (v - crate::doc::DEFAULT_GRID).abs() > 0.01 {
101            w.val(tag, &crate::docx::units::twips(v.clamp(0.5, 1584.0)));
102        }
103    }
104    w.val("w:characterSpacingControl", "doNotCompress");
105    write_note_settings(&mut w, s, s_notes, footnotes, endnotes);
106    write_compatibility_settings(&mut w, doc);
107    if let Some(m) = &s.math {
108        math::math_pr(&mut w, m);
109    }
110    w.close("w:settings");
111    w.into_bytes()
112}

114fn write_note_settings(
115    w: &mut W,
116    settings: &crate::doc::Settings,
117    separators: (&crate::doc::NoteSeparators, &crate::doc::NoteSeparators),
118    footnotes: bool,
119    endnotes: bool,
120) {
121    for (tag, fmt, pr, used, el) in [
122        ("w:footnotePr", settings.footnote_format, &settings.footnote_pr, footnotes, "w:footnote"),
123        ("w:endnotePr", settings.endnote_format, &settings.endnote_pr, endnotes, "w:endnote"),
124    ] {
125        w.open(tag, &[]);
126        story::note_pr_inner(w, &crate::doc::section::NoteProps { num_fmt: Some(fmt), ..*pr });
127        if used {
128            w.empty(el, &[("w:id", "-1")]);
129            w.empty(el, &[("w:id", "0")]);
130            let seps = if el == "w:footnote" { separators.0 } else { separators.1 };
131            if seps.continuation_notice.is_some() {
132                w.empty(el, &[("w:id", "1")]);
133            }
134        }
135        w.close(tag);
136    }
137}

139fn write_compatibility_settings(w: &mut W, doc: &Document) {
140    w.open("w:compat", &[]);
141    // CT_Compat: the older switches come before the `w:compatSetting`s.
142    if doc.settings.do_not_expand_shift_return {
143        w.empty("w:doNotExpandShiftReturn", &[]);
144    }
145    if doc.settings.sum_paragraph_spacing {
146        w.empty("w:doNotUseHTMLParagraphAutoSpacing", &[]);
147    }
148    if doc.settings.allow_space_of_same_style_in_table {
149        w.empty("w:allowSpaceOfSameStyleInTable", &[]);
150    }
151    let mode = doc.settings.compat_mode.clamp(11, 15).to_string();
152    w.empty("w:compatSetting", &[("w:name", "compatibilityMode"), ("w:uri", "http://schemas.microsoft.com/office/word"), ("w:val", &mode)]);
153    w.close("w:compat");
154}

156/// Is `d` a usable `w16cid:durableId`: 1–8 hex digits, below 0x7FFFFFFF, not 0?
157pub(crate) fn durable_id_ok(d: &str) -> bool {
158    d.len() <= 8 && u32::from_str_radix(d, 16).is_ok_and(|v| v != 0 && v < 0x7FFF_FFFF)
159}

161/// `commentsIds.xml` (paragraph id → durable id) and `commentsExtensible.xml` (durable id → UTC
162/// date) for the comments that got paragraph ids. Durable ids read from the file are kept (unique
163/// ones); comments without one get the next free id. Word uses them to keep tracking a comment
164/// across saves and for its UTC time.
165pub(super) fn write_comment_ids(
166    doc: &Document,
167    comment_paras: &[(u32, String)],
168    entries: &mut Vec<(String, Vec<u8>)>,
169    overrides: &mut Vec<(String, String)>,
170    rels: &mut PartRels,
171) {
172    if comment_paras.is_empty() {
173        return;
174    }
175    // Keep the existing exact-string reservation rule, including comments without a written part.
176    let reserved: std::collections::HashSet<&str> = doc.comments.values().filter_map(|c| c.durable_id.as_deref()).collect();
177    let mut used: std::collections::HashSet<String> = std::collections::HashSet::new();
178    let mut next = 0x1000_0000u32;
179    let mut rows: Vec<(String, String, Option<String>)> = Vec::new();
180    for (cid, pid) in comment_paras {
181        let Some(c) = doc.comments.get(cid) else { continue };
182        let kept = c.durable_id.as_deref().filter(|d| durable_id_ok(d)).map(|d| format!("{:08X}", u32::from_str_radix(d, 16).unwrap_or(0)));
183        let durable = match kept.filter(|d| !used.contains(d)) {
184            Some(d) => d,
185            None => loop {
186                next = next.saturating_add(1);
187                let d = format!("{next:08X}");
188                if !used.contains(&d) && !reserved.contains(d.as_str()) {
189                    break d;
190                }
191            },
192        };
193        used.insert(durable.clone());
194        // Only a known UTC time: `w:date` is local time, often with a `Z` it doesn't mean.
195        // Remove date and time on save (#1062): no comment date here either.
196        let utc = if doc.settings.remove_date_time { None } else { c.date_utc.clone() };
197        rows.push((pid.clone(), durable, utc));
198    }
199    let mc = "http://schemas.openxmlformats.org/markup-compatibility/2006";
200    let mut w = W::new();
201    w.open(
202        "w16cid:commentsIds",
203        &[("xmlns:mc", mc), ("xmlns:w16cid", "http://schemas.microsoft.com/office/word/2016/wordml/cid"), ("mc:Ignorable", "w16cid")],
204    );
205    for (pid, durable, _) in &rows {
206        w.empty("w16cid:commentId", &[("w16cid:paraId", pid), ("w16cid:durableId", durable)]);
207    }
208    w.close("w16cid:commentsIds");
209    rels.add(rt::COMMENTS_IDS, "commentsIds.xml", false);
210    push_part(entries, overrides, "word/commentsIds.xml", w.into_bytes(), &format!("{CT_WML}commentsIds+xml"), PartRels::default());
211
212    let mut w = W::new();
213    w.open(
214        "w16cex:commentsExtensible",
215        &[("xmlns:mc", mc), ("xmlns:w16cex", "http://schemas.microsoft.com/office/word/2018/wordml/cex"), ("mc:Ignorable", "w16cex")],
216    );
217    for (_, durable, utc) in &rows {
218        match utc {
219            Some(d) => w.empty("w16cex:commentExtensible", &[("w16cex:durableId", durable), ("w16cex:dateUtc", d)]),
220            None => w.empty("w16cex:commentExtensible", &[("w16cex:durableId", durable)]),
221        }
222    }
223    w.close("w16cex:commentsExtensible");
224    rels.add(rt::COMMENTS_EXTENSIBLE, "commentsExtensible.xml", false);
225    push_part(entries, overrides, "word/commentsExtensible.xml", w.into_bytes(), &format!("{CT_WML}commentsExtensible+xml"), PartRels::default());
226}

228pub(super) fn theme_xml(doc: &Document) -> Vec<u8> {
229    let s = &doc.settings;
230    let names = ["dk1", "lt1", "dk2", "lt2", "accent1", "accent2", "accent3", "accent4", "accent5", "accent6", "hlink", "folHlink"];
231    let mut w = W::new();
232    let name = if s.theme_name.is_empty() { "Craft" } else { s.theme_name.as_str() };
233    w.open("a:theme", &[("xmlns:a", "http://schemas.openxmlformats.org/drawingml/2006/main"), ("name", name)]);
234    w.open("a:themeElements", &[]);
235    w.open("a:clrScheme", &[("name", name)]);
236    for (i, n) in names.iter().enumerate() {
237        let c = s.theme_colors.get(i).or_else(|| crate::doc::THEME_COLORS.get(i)).copied().unwrap_or_default();
238        w.open(&format!("a:{n}"), &[]);
239        w.empty("a:srgbClr", &[("val", &c.hex())]);
240        w.close(&format!("a:{n}"));
241    }
242    w.close("a:clrScheme");
243    w.open("a:fontScheme", &[("name", name)]);
244    for (tag, face) in [("a:majorFont", &s.major_font), ("a:minorFont", &s.minor_font)] {
245        w.open(tag, &[]);
246        w.empty("a:latin", &[("typeface", face)]);
247        w.empty("a:ea", &[("typeface", "")]);
248        w.empty("a:cs", &[("typeface", "")]);
249        w.close(tag);
250    }
251    w.close("a:fontScheme");
252    // A minimal format scheme (three entries per list, as the schema requires).
253    w.open("a:fmtScheme", &[("name", name)]);
254    w.open("a:fillStyleLst", &[]);
255    for _ in 0..3 {
256        w.raw("<a:solidFill><a:schemeClr val=\"phClr\"/></a:solidFill>");
257    }
258    w.close("a:fillStyleLst");
259    w.open("a:lnStyleLst", &[]);
260    for width in ["6350", "12700", "19050"] {
261        w.raw(&format!(
262            "<a:ln w=\"{width}\" cap=\"flat\" cmpd=\"sng\" algn=\"ctr\"><a:solidFill><a:schemeClr val=\"phClr\"/></a:solidFill><a:prstDash val=\"solid\"/><a:miter lim=\"800000\"/></a:ln>"
263        ));
264    }
265    w.close("a:lnStyleLst");
266    w.open("a:effectStyleLst", &[]);
267    for _ in 0..3 {
268        w.raw("<a:effectStyle><a:effectLst/></a:effectStyle>");
269    }
270    w.close("a:effectStyleLst");
271    w.open("a:bgFillStyleLst", &[]);
272    for _ in 0..3 {
273        w.raw("<a:solidFill><a:schemeClr val=\"phClr\"/></a:solidFill>");
274    }
275    w.close("a:bgFillStyleLst");
276    w.close("a:fmtScheme");
277    w.close("a:themeElements");
278    w.empty("a:objectDefaults", &[]);
279    w.empty("a:extraClrSchemeLst", &[]);
280    w.close("a:theme");
281    w.into_bytes()
282}

284/// The author the saved file gives a comment or tracked change when the document removes personal
285/// information on save (`w:removePersonalInformation`), as Word does.
286pub(crate) const ANONYMOUS_AUTHOR: &str = "Author";
287
288pub(crate) const ANONYMOUS_INITIALS: &str = "A";

290/// The `w:author` written for an annotation by `author`: "Author" when the document removes
291/// personal information on save, "Unknown" when it has none.
292pub(crate) fn annotation_author<'a>(doc: &Document, author: &'a str) -> &'a str {
293    if doc.settings.remove_personal_info {
294        ANONYMOUS_AUTHOR
295    } else if author.is_empty() {
296        "Unknown"
297    } else {
298        author
299    }
300}

302pub(super) fn core_xml(doc: &Document) -> Vec<u8> {
303    let c = &doc.core;
304    // Remove personal information (§17.15.1.70): no creator or last-modified-by in the file.
305    let personal = |v: &String| if doc.settings.remove_personal_info { String::new() } else { v.clone() };
306    let (creator, last_modified_by) = (personal(&c.creator), personal(&c.last_modified_by));
307    let mut w = W::new();
308    w.open(
309        "cp:coreProperties",
310        &[
311            ("xmlns:cp", "http://schemas.openxmlformats.org/package/2006/metadata/core-properties"),
312            ("xmlns:dc", "http://purl.org/dc/elements/1.1/"),
313            ("xmlns:dcterms", "http://purl.org/dc/terms/"),
314            ("xmlns:dcmitype", "http://purl.org/dc/dcmitype/"),
315            ("xmlns:xsi", "http://www.w3.org/2001/XMLSchema-instance"),
316        ],
317    );
318    for (tag, v) in [
319        ("dc:title", &c.title),
320        ("dc:subject", &c.subject),
321        ("dc:creator", &creator),
322        ("cp:keywords", &c.keywords),
323        ("dc:description", &c.description),
324        ("cp:lastModifiedBy", &last_modified_by),
325        ("cp:category", &c.category),
326    ] {
327        if !v.is_empty() {
328            w.leaf(tag, &[], v);
329        }
330    }
331    if c.revision > 0 {
332        w.leaf("cp:revision", &[], &c.revision.to_string());
333    }
334    for (tag, v) in [("dcterms:created", &c.created), ("dcterms:modified", &c.modified)] {
335        if !v.is_empty() {
336            w.leaf(tag, &[("xsi:type", "dcterms:W3CDTF")], v);
337        }
338    }
339    w.close("cp:coreProperties");
340    w.into_bytes()
341}

343pub(super) fn app_xml(doc: &Document) -> Vec<u8> {
344    let mut w = W::new();
345    w.open(
346        "Properties",
347        &[
348            ("xmlns", "http://schemas.openxmlformats.org/officeDocument/2006/extended-properties"),
349            ("xmlns:vt", "http://schemas.openxmlformats.org/officeDocument/2006/docPropsVTypes"),
350        ],
351    );
352    w.leaf("Application", &[], "WordCraft");
353    w.leaf("Words", &[], &doc.word_count().to_string());
354    w.leaf("Paragraphs", &[], &doc.paragraph_count().to_string());
355    w.leaf("DocSecurity", &[], "0");
356    // Kept properties, except the personal ones when the file asks Word to remove personal
357    // information on save.
358    let private = doc.settings.remove_personal_info;
359    if let Some(kept) = doc.passthrough.get(crate::docx::package::APP_KEPT) {
360        for line in String::from_utf8_lossy(kept).lines().take(crate::docx::package::APP_KEPT_PROPS.len()) {
361            let Some((name, value)) = line.split_once('\t') else { continue };
362            let Some(name) = crate::docx::package::APP_KEPT_PROPS.iter().find(|n| **n == name) else { continue };
363            if private && matches!(*name, "Company" | "Manager") {
364                continue;
365            }
366            w.leaf(name, &[], value);
367        }
368    }
369    w.close("Properties");
370    w.into_bytes()
