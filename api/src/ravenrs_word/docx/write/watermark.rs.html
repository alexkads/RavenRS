1//! The document's text watermark as a VML text shape in the headers (ECMA-376 Part 4, VML:
2//! `shape`, `shapetype`, `path`, `textpath`, `fill`), which is how a WordprocessingML consumer
3//! draws text behind every page. The reader (`read::story::vml_watermark`) maps it back.
4
5use crate::doc::{SectionProps, Watermark};
6
7use crate::docx::xml::esc;
8
9/// The shape type the watermark shape uses: two straight subpaths, the box's top and bottom
10/// edges, between which the `v:textpath` text is stretched to fit the shape (`fitshape`). A single
11/// path gives the text no height, and Word draws nothing.
12const SHAPETYPE_ID: &str = "_x0000_t136";
13const SHAPETYPE_PATH: &str = "m0,0l21600,0m0,21600l21600,21600e";
14
15/// The watermark text as written: what the reader accepts (non-blank, at most 200 characters).
16pub(super) fn text(wm: &Watermark) -> Option<String> {
17    let t: String = wm.text.chars().take(200).collect();
18    (!t.trim().is_empty()).then_some(t)
19}
20
21/// A `w:r` holding the watermark shape (and its shape type), for the first paragraph of header
22/// part number `n`, in its box ([`Watermark::frame`]: its own, or Word's default for the page).
23pub(super) fn run(wm: &Watermark, n: usize, page: &SectionProps) -> Option<String> {
24    let text = text(wm)?;
25    let [width, height] = wm.frame(page.page_w, page.page_h);
26    // A font name can't break out of the textpath's CSS-like style or make it malformed (#3994):
27    // the characters a quoted CSS string can't carry raw (`\22 ` is `"`, `\3b ` `;`, `\3a ` `:`,
28    // `\5c ` `\`) are escaped and control characters are dropped (no font name carries one); the
29    // reader (`read::story::css_string`) maps the escapes back. An apostrophe is legal inside a
30    // double-quoted string, so it is kept.
31    let mut font = String::with_capacity(wm.font.len());
32    for c in wm.font.chars().filter(|c| !c.is_control()).take(100) {
33        match c {
34            '"' => font.push_str(r"\22 "),
35            ';' => font.push_str(r"\3b "),
36            ':' => font.push_str(r"\3a "),
37            '\\' => font.push_str(r"\5c "),
38            c => font.push(c),
39        }
40    }
41    let font = if font.trim().is_empty() { Watermark::default().font } else { font };
42    let rotation = if wm.diagonal { "rotation:315;" } else { "" };
43    let style = format!(
44        "position:absolute;margin-left:0;margin-top:0;width:{width:.1}pt;height:{height:.1}pt;{rotation}z-index:-251654144;\
45         mso-position-horizontal:center;mso-position-horizontal-relative:margin;\
46         mso-position-vertical:center;mso-position-vertical-relative:margin"
47    );
48    let fill = if wm.semitransparent { r#"<v:fill opacity=".5"/>"# } else { "" };
49    Some(format!(
50        concat!(
51            r#"<w:r><w:rPr><w:noProof/></w:rPr><w:pict>"#,
52            r#"<v:shapetype id="{st}" coordsize="21600,21600" o:spt="136" path="{path}">"#,
53            r#"<v:path textpathok="t"/><v:textpath on="t" fitshape="t"/></v:shapetype>"#,
54            r##"<v:shape id="PowerPlusWaterMarkObject{n}" type="#{st}" style="{style}" o:allowincell="f" fillcolor="#{color}" stroked="f">"##,
55            r#"{fill}<v:textpath style="{tstyle}" string="{text}"/></v:shape></w:pict></w:r>"#
56        ),
57        st = SHAPETYPE_ID,
58        path = SHAPETYPE_PATH,
59        n = n,
60        style = esc(&style),
61        color = wm.color.hex(),
62        fill = fill,
63        tstyle = esc(&format!("font-family:\"{font}\";font-size:1pt")),
64        text = esc(&text),
65    ))
66}
