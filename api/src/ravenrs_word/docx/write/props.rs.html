11fn toggle(w: &mut W, name: &str, v: Option<bool>) {
12    match v {
13        Some(true) => w.empty(name, &[]),
14        Some(false) => w.val(name, "0"),
15        None => {}
16    }
17}

24/// `w:rPr` content (without the wrapper). Link and revision marks are not run properties.
25pub fn rpr_inner(w: &mut W, c: &CharProps) {
26    if let Some(s) = &c.style {
27        w.val("w:rStyle", s);
28    }
29    // Emit only explicit properties so absent values keep inheriting from the style chain.
30    let mut fonts: Vec<(&str, &str)> = Vec::new();
31    if let Some(f) = &c.font {
32        fonts.extend([("w:ascii", f.as_str()), ("w:hAnsi", f.as_str())]);
33    }
34    if let Some(ea) = &c.font_ea {
35        fonts.push(("w:eastAsia", ea));
36    }
37    if let Some(cs) = &c.font_cs {
38        fonts.push(("w:cs", cs));
39    }
40    if !fonts.is_empty() {
41        w.empty("w:rFonts", &fonts);
42    }
43    toggle(w, "w:b", c.bold);
44    toggle(w, "w:bCs", c.bold_cs);
45    toggle(w, "w:i", c.italic);
46    toggle(w, "w:iCs", c.italic_cs);
47    toggle(w, "w:caps", c.caps);
48    toggle(w, "w:smallCaps", c.small_caps);
49    toggle(w, "w:strike", c.strike);
50    toggle(w, "w:dstrike", c.double_strike);
51    toggle(w, "w:outline", c.outline);
52    toggle(w, "w:shadow", c.shadow);
53    toggle(w, "w:emboss", c.emboss);
54    toggle(w, "w:imprint", c.engrave);
55    toggle(w, "w:noProof", c.no_proof);
56    toggle(w, "w:snapToGrid", c.snap_to_grid);
57    toggle(w, "w:vanish", c.hidden);
58    toggle(w, "w:webHidden", c.web_hidden);
59    match c.color {
60        Some(TextColor::Auto) => w.val("w:color", "auto"),
61        Some(TextColor::Rgb(rgb)) => w.val("w:color", &rgb.hex()),
62        None => {}
63    }
64    if let Some(s) = c.spacing {
65        w.val("w:spacing", &twips(s.clamp(-1584.0, 1584.0)));
66    }
67    if let Some(s) = c.scale {
68        w.val("w:w", &n(round(s.clamp(1.0, 600.0))));
69    }
70    if let Some(k) = c.kern {
71        w.val("w:kern", &n(crate::geom::to_half_points(k.clamp(0.0, 1638.0))));
72    }
73    if let Some(p) = c.position {
74        w.val("w:position", &n(crate::geom::to_half_points(p.clamp(-1584.0, 1584.0))));
75    }
76    let half = |s: f32| n(crate::geom::to_half_points(s.clamp(1.0, 1638.0)).max(2));
77    if let Some(s) = c.size {
78        w.val("w:sz", &half(s));
79    }
80    if let Some(s) = c.size_cs {
81        w.val("w:szCs", &half(s));
82    }
83    if let Some(h) = c.highlight {
84        w.val("w:highlight", if h == Highlight::None { "none" } else { h.ooxml() });
85    }
86    match (c.underline, c.underline_color) {
87        (Some(u), Some(col)) => w.empty("w:u", &[("w:val", u.ooxml()), ("w:color", &col.hex())]),
88        (Some(u), None) => w.val("w:u", u.ooxml()),
89        // A colour-only w:u, as read: keep it without inventing an underline
90        (None, Some(col)) => w.empty("w:u", &[("w:color", &col.hex())]),
91        (None, None) => {}
92    }
93    if let Some(b) = &c.border {
94        border_el(w, "w:bdr", b);
95    }
96    if let Some(s) = c.shading {
97        w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &s.hex())]);
98    }
99    if let Some(v) = c.vert_align {
100        w.val(
101            "w:vertAlign",
102            match v {
103                VertAlign::Baseline => "baseline",
104                VertAlign::Superscript => "superscript",
105                VertAlign::Subscript => "subscript",
106            },
107        );
108    }
109    toggle(w, "w:rtl", c.rtl);
110    toggle(w, "w:cs", c.cs);
111    // Emphasis mark (§17.3.2.12), after `w:cs` and before `w:lang`.
112    if let Some(m) = c.emphasis {
113        w.val("w:em", m.ooxml());
114    }
115    let mut lang_attrs: Vec<(&str, &str)> = Vec::with_capacity(3);
116    if let Some(l) = &c.lang {
117        lang_attrs.push(("w:val", l));
118    }
119    if let Some(ea) = &c.lang_ea {
120        lang_attrs.push(("w:eastAsia", ea));
121    }
122    if let Some(b) = &c.lang_bidi {
123        lang_attrs.push(("w:bidi", b));
124    }
125    if !lang_attrs.is_empty() {
126        w.empty("w:lang", &lang_attrs);
127    }
128    // After w:lang (CT_RPr's sequence).
129    if let Some(l) = &c.east_asian_layout {
130        east_asian_layout(w, l);
131    }
132    if let Some(fx) = &c.text_effects {
133        super::wordart::rpr_effects(w, fx);
134    }
135    if let Some(l) = c.ligatures {
136        w.empty("w14:ligatures", &[("w14:val", &l.ooxml())]);
137    }
138}

140/// `w:eastAsianLayout` with the attributes that are set (ISO/IEC 29500-1 §17.3.2.10).
141fn east_asian_layout(w: &mut W, l: &EastAsianLayout) {
142    let on = |v: bool| if v { "1" } else { "0" };
143    let id = l.id.map(|v| v.to_string());
144    let mut a: Vec<(&str, &str)> = Vec::new();
145    if let Some(id) = &id {
146        a.push(("w:id", id));
147    }
148    if let Some(v) = l.combine {
149        a.push(("w:combine", on(v)));
150    }
151    if let Some(b) = l.combine_brackets {
152        a.push(("w:combineBrackets", b.ooxml()));
153    }
154    if let Some(v) = l.vert {
155        a.push(("w:vert", on(v)));
156    }
157    if let Some(v) = l.vert_compress {
158        a.push(("w:vertCompress", on(v)));
159    }
160    w.empty("w:eastAsianLayout", &a);
161}

171/// Open a tracked change element (`w:rPrChange`, `w:pPrChange`, `w:tblPrChange`…).
172pub fn open_change(w: &mut W, tag: &str, a: &ChangeAttrs) {
173    let refs: Vec<(&str, &str)> = a.iter().map(|(k, v)| (*k, v.as_str())).collect();
174    w.open(tag, &refs);
175}

177/// A tracked change element holding the properties before the change: `<tag attrs><inner>…</inner></tag>`.
178fn change_el(w: &mut W, tag: &str, inner: &str, a: &ChangeAttrs, body: impl FnOnce(&mut W)) {
179    open_change(w, tag, a);
180    w.open(inner, &[]);
181    body(w);
182    w.close(inner);
183    w.close(tag);
184}

215fn border_el(w: &mut W, name: &str, b: &Border) {
216    let sz = n(round(b.width.clamp(0.0, 12.0) * 8.0));
217    let space = n(round(b.space.clamp(0.0, 31.0)));
218    let color = b.color.map(|c| c.hex()).unwrap_or_else(|| "auto".into());
219    let style = if b.style == BorderStyle::None { "nil" } else { b.style.ooxml() };
220    w.empty(name, &[("w:val", style), ("w:sz", &sz), ("w:space", &space), ("w:color", &color)]);
221}

223/// Borders element. `inner` names the between/insideH element (`w:between` for paragraphs,
224/// `w:insideH` for tables/cells; `None` for page borders).
225pub fn borders(w: &mut W, tag: &str, b: &Borders, inner: Option<&str>, extra: &[(&str, &str)]) {
226    w.open(tag, extra);
227    if let Some(x) = &b.top {
228        border_el(w, "w:top", x);
229    }
230    if let Some(x) = &b.left {
231        border_el(w, "w:left", x);
232    }
233    if let Some(x) = &b.bottom {
234        border_el(w, "w:bottom", x);
235    }
236    if let Some(x) = &b.right {
237        border_el(w, "w:right", x);
238    }
239    if let Some(name) = inner {
240        if let Some(x) = &b.between {
241            border_el(w, name, x);
242        }
243        if name == "w:insideH"
244            && let Some(x) = &b.inside_v
245        {
246            border_el(w, "w:insideV", x);
247        }
248    }
249    w.close(tag);
250}

252/// `w:pPr` content except the paragraph mark `w:rPr` and `w:sectPr`, which the caller appends.
253/// `w:pPr` content (without the wrapper); `num_change` are the attributes of a tracked list
254/// numbering change (`w:numberingChange`, written in `w:numPr`).
255pub fn ppr_inner(w: &mut W, p: &ParaProps, framed: bool, num_change: Option<&ChangeAttrs>) {
256    if let Some(s) = &p.style {
257        w.val("w:pStyle", s);
258    }
259    toggle(w, "w:keepNext", p.keep_next);
260    toggle(w, "w:keepLines", p.keep_lines);
261    toggle(w, "w:pageBreakBefore", p.page_break_before);
262    write_frame(w, p, framed);
263    toggle(w, "w:widowControl", p.widow_control);
264    if let Some(nr) = p.numbering {
265        w.open("w:numPr", &[]);
266        w.val("w:ilvl", &n(nr.level.min(8) as i64));
267        w.val("w:numId", &n(nr.num as i64));
268        if let Some(a) = num_change {
269            let refs: Vec<(&str, &str)> = a.iter().map(|(k, v)| (*k, v.as_str())).collect();
270            w.empty("w:numberingChange", &refs);
271        }
272        w.close("w:numPr");
273    }
274    toggle(w, "w:suppressLineNumbers", p.suppress_line_numbers);
275    if let Some(b) = &p.borders {
276        borders(w, "w:pBdr", b, Some("w:between"), &[]);
277    }
278    if let Some(s) = p.shading {
279        w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &s.hex())]);
280    }
281    write_tabs(w, p);
282    write_text_flow(w, p);
283    write_spacing(w, p);
284    write_indentation(w, p);
285    toggle(w, "w:contextualSpacing", p.contextual_spacing);
286    toggle(w, "w:mirrorIndents", p.mirror_indents);
287    if let Some(a) = p.align {
288        w.val("w:jc", align_val(a));
289    }
290    if let Some(direction) = p.text_direction {
291        w.val("w:textDirection", direction.ooxml());
292    }
293    if let Some(t) = p.text_alignment {
294        w.val("w:textAlignment", t.ooxml());
295    }
296    if let Some(l) = p.outline_level {
297        w.val("w:outlineLvl", &n(l.min(9) as i64));
298    }
299}

301fn write_frame(w: &mut W, p: &ParaProps, framed: bool) {
302    if framed && let Some(lines) = p.drop_cap.filter(|l| *l > 0) {
303        // ECMA-376 §17.3.1.11: the drop cap's frame — dropped or in the margin, the lines it
304        // spans and its distance from the text (`w:hSpace`, twips).
305        let kind = if p.drop_cap_margin == Some(true) { "margin" } else { "drop" };
306        let lines = n(lines.min(10) as i64);
307        let space = p.drop_cap_distance.filter(|d| d.is_finite() && *d > 0.0).map(|d| twips(d.min(crate::doc::resolve::MAX_DROP_CAP_DISTANCE)));
308        let mut a: Vec<(&str, &str)> = vec![("w:dropCap", kind), ("w:lines", &lines)];
309        if let Some(h) = &space {
310            a.push(("w:hSpace", h));
311        }
312        a.extend([("w:wrap", "around"), ("w:vAnchor", "text"), ("w:hAnchor", "text")]);
313        w.empty("w:framePr", &a);
314    } else if let Some(f) = &p.frame {
315        let (fw, fh, fx, fy) = (f.w.map(twips), f.h.map(twips), twips(f.x), twips(f.y));
316        let (hs, vs) = (twips(f.h_space), twips(f.v_space));
317        let mut a: Vec<(&str, &str)> = Vec::new();
318        if let Some(v) = &fw {
319            a.push(("w:w", v));
320        }
321        if let Some(v) = &fh {
322            a.push(("w:h", v));
323            a.push(("w:hRule", if f.h_exact { "exact" } else { "atLeast" }));
324        }
325        if f.h_space > 0.0 {
326            a.push(("w:hSpace", &hs));
327        }
328        if f.v_space > 0.0 {
329            a.push(("w:vSpace", &vs));
330        }
331        a.push(("w:wrap", &f.wrap));
332        if let Some(anchor_lock) = f.anchor_lock {
333            a.push(("w:anchorLock", if anchor_lock { "1" } else { "0" }));
334        }
335        a.push(("w:vAnchor", &f.v_anchor));
336        a.push(("w:hAnchor", &f.h_anchor));
337        match &f.x_align {
338            Some(x) => a.push(("w:xAlign", x)),
339            None => a.push(("w:x", &fx)),
340        }
341        match &f.y_align {
342            Some(y) => a.push(("w:yAlign", y)),
343            None => a.push(("w:y", &fy)),
344        }
345        w.empty("w:framePr", &a);
346    }
347}

349fn write_tabs(w: &mut W, p: &ParaProps) {
350    if let Some(tabs) = &p.tabs
351        && !tabs.is_empty()
352    {
353        w.open("w:tabs", &[]);
354        for t in tabs.iter().take(64) {
355            let val = match t.align {
356                TabAlign::Left => "left",
357                TabAlign::Center => "center",
358                TabAlign::Right => "right",
359                TabAlign::Decimal => "decimal",
360                TabAlign::Bar => "bar",
361                TabAlign::Clear => "clear",
362            };
363            let pos = twips(t.pos);
364            match t.leader {
365                TabLeader::None => w.empty("w:tab", &[("w:val", val), ("w:pos", &pos)]),
366                l => w.empty("w:tab", &[("w:val", val), ("w:leader", tab_leader(l)), ("w:pos", &pos)]),
367            }
368        }
369        w.close("w:tabs");
370    }
371}

386fn write_spacing(w: &mut W, p: &ParaProps) {
387    if p.space_before.is_some()
388        || p.space_after.is_some()
389        || p.line_spacing.is_some()
390        || p.space_before_auto.is_some()
391        || p.space_after_auto.is_some()
392    {
393        let before = p.space_before.map(|v| twips(v.max(0.0)));
394        let after = p.space_after.map(|v| twips(v.max(0.0)));
395        let line = p.line_spacing.map(|l| match l {
396            LineSpacing::Multiple(m) => (n(round(m.clamp(0.06, 132.0) * 240.0)), "auto"),
397            LineSpacing::AtLeast(v) => (twips(v.abs()), "atLeast"),
398            LineSpacing::Exactly(v) => (twips(v.abs()), "exact"),
399        });
400        let mut a: Vec<(&str, &str)> = Vec::new();
401        if let Some(b) = &before {
402            a.push(("w:before", b));
403        }
404        if let Some(on) = p.space_before_auto {
405            a.push(("w:beforeAutospacing", if on { "1" } else { "0" }));
406        }
407        if let Some(x) = &after {
408            a.push(("w:after", x));
409        }
410        if let Some(on) = p.space_after_auto {
411            a.push(("w:afterAutospacing", if on { "1" } else { "0" }));
412        }
413        if let Some((l, r)) = &line {
414            a.push(("w:line", l));
415            a.push(("w:lineRule", r));
416        }
417        w.empty("w:spacing", &a);
418    }
419}

421fn write_indentation(w: &mut W, p: &ParaProps) {
422    let chars = [p.indent_left_chars, p.indent_right_chars, p.indent_first_chars];
423    if p.indent_left.is_some() || p.indent_right.is_some() || p.indent_first.is_some() || chars.iter().any(Option::is_some) {
424        let l = p.indent_left.map(twips);
425        let r = p.indent_right.map(twips);
426        let f = p.indent_first.map(|v| (v < 0.0, twips(v.abs())));
427        // Character units in hundredths of a character.
428        let hundredths = |v: f32| ((v.clamp(-1584.0, 1584.0) * 100.0).round() as i64).to_string();
429        let [lc, rc, fc] = chars.map(|c| c.map(|v| (v < 0.0, hundredths(v.abs()), hundredths(v))));
430        let mut a: Vec<(&str, &str)> = Vec::new();
431        if let Some((_, _, v)) = &lc {
432            a.push(("w:leftChars", v));
433        }
434        if let Some(l) = &l {
435            a.push(("w:left", l));
436        }
437        if let Some((_, _, v)) = &rc {
438            a.push(("w:rightChars", v));
439        }
440        if let Some(r) = &r {
441            a.push(("w:right", r));
442        }
443        if let Some((hang, v, _)) = &fc {
444            a.push((if *hang { "w:hangingChars" } else { "w:firstLineChars" }, v));
445        }
446        if let Some((hang, v)) = &f {
447            a.push((if *hang { "w:hanging" } else { "w:firstLine" }, v));
448        }
449        w.empty("w:ind", &a);
450    }
451}

457fn margin_sides(w: &mut W, tag: &str, m: &[Option<f32>; 4]) {
458    w.open(tag, &[]);
459    for (name, value) in ["w:top", "w:left", "w:bottom", "w:right"].iter().zip(m.iter()) {
460        if let Some(v) = value {
461            w.empty(name, &[("w:w", &twips(v.clamp(0.0, 1584.0))), ("w:type", "dxa")]);
462        }
463    }
464    w.close(tag);
465}

467pub fn look_attrs(l: &TableLook) -> Vec<(&'static str, String)> {
468    let mut bits = 0u32;
469    if l.header_row {
470        bits |= 0x0020;
471    }
472    if l.total_row {
473        bits |= 0x0040;
474    }
475    if l.first_column {
476        bits |= 0x0080;
477    }
478    if l.last_column {
479        bits |= 0x0100;
480    }
481    if !l.banded_rows {
482        bits |= 0x0200;
483    }
484    if !l.banded_columns {
485        bits |= 0x0400;
486    }
487    let b = |v: bool| if v { "1".to_string() } else { "0".to_string() };
488    vec![
489        ("w:val", format!("{bits:04X}")),
490        ("w:firstRow", b(l.header_row)),
491        ("w:lastRow", b(l.total_row)),
492        ("w:firstColumn", b(l.first_column)),
493        ("w:lastColumn", b(l.last_column)),
494        ("w:noHBand", b(!l.banded_rows)),
495        ("w:noVBand", b(!l.banded_columns)),
496    ]
497}

499/// `w:tblPr`, with the properties before a tracked change last when `change` gives its attributes.
500pub fn tblpr(w: &mut W, t: &TableProps, change: Option<&ChangeAttrs>) {
501    w.open("w:tblPr", &[]);
502    tblpr_inner(w, t);
503    if let (Some(a), Some(ch)) = (change, &t.fmt_change) {
504        change_el(w, "w:tblPrChange", "w:tblPr", a, |w| tblpr_inner(w, &ch.old));
505    }
506    w.close("w:tblPr");
507}

509fn tblpr_inner(w: &mut W, t: &TableProps) {
510    if let Some(s) = &t.style {
511        w.val("w:tblStyle", s);
512    }
513    if let Some(f) = &t.float {
514        table_float(w, f);
515    }
516    if t.bidi_visual {
517        w.empty("w:bidiVisual", &[]);
518    }
519    if let Some(p) = t.width_pct {
520        w.empty("w:tblW", &[("w:w", &n(round(p.clamp(0.0, 1000.0) * 50.0))), ("w:type", "pct")]);
521    } else if let Some(v) = t.width {
522        w.empty("w:tblW", &[("w:w", &twips(v.max(0.0))), ("w:type", "dxa")]);
523    } else {
524        w.empty("w:tblW", &[("w:w", "0"), ("w:type", "auto")]);
525    }
526    if let Some(a) = t.align {
527        w.val("w:jc", align_val(if a == Align::Justify || a == Align::Distribute { Align::Left } else { a }));
528    }
529    if let Some(i) = t.indent {
530        w.empty("w:tblInd", &[("w:w", &twips(i)), ("w:type", "dxa")]);
531    }
532    if let Some(b) = &t.borders {
533        borders(w, "w:tblBorders", b, Some("w:insideH"), &[]);
534    }
535    if let Some(s) = t.shading {
536        w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &s.hex())]);
537    }
538    if t.fixed {
539        w.empty("w:tblLayout", &[("w:type", "fixed")]);
540    }
541    if let Some(m) = &t.cell_margins {
542        margins(w, "w:tblCellMar", m);
543    }
544    let look = look_attrs(&t.look);
545    let refs: Vec<(&str, &str)> = look.iter().map(|(k, v)| (*k, v.as_str())).collect();
546    w.empty("w:tblLook", &refs);
547    if let Some(c) = &t.caption {
548        w.val("w:tblCaption", c);
549    }
550    if let Some(d) = &t.description {
551        w.val("w:tblDescription", d);
552    }
553}

555/// `w:trPr` (when the row has any), with a tracked change last as for [`tblpr`].
556pub fn trpr(w: &mut W, r: &RowProps, change: Option<&ChangeAttrs>) {
557    if let Some(x) = &r.exceptions {
558        tblprex(w, x);
559    }
560    let change = change.zip(r.fmt_change.as_ref());
561    let gaps = r.grid_before > 0 || r.grid_after > 0 || r.w_before.is_some() || r.w_after.is_some();
562    if r.height.is_none() && !r.header && !r.cant_split && !gaps && change.is_none() {
563        return;
564    }
565    w.open("w:trPr", &[]);
566    trpr_inner(w, r);
567    if let Some((a, ch)) = change {
568        change_el(w, "w:trPrChange", "w:trPr", a, |w| trpr_inner(w, &ch.old));
569    }
570    w.close("w:trPr");
571}

573/// `w:tblPrEx`: only the properties the row overrides (it takes no style, float or look).
574fn tblprex(w: &mut W, t: &TableProps) {
575    w.open("w:tblPrEx", &[]);
576    if let Some(p) = t.width_pct {
577        w.empty("w:tblW", &[("w:w", &n(round(p.clamp(0.0, 1000.0) * 50.0))), ("w:type", "pct")]);
578    } else if let Some(v) = t.width {
579        w.empty("w:tblW", &[("w:w", &twips(v.max(0.0))), ("w:type", "dxa")]);
580    }
581    if let Some(a) = t.align {
582        w.val("w:jc", align_val(if a == Align::Justify || a == Align::Distribute { Align::Left } else { a }));
583    }
584    if let Some(i) = t.indent {
585        w.empty("w:tblInd", &[("w:w", &twips(i)), ("w:type", "dxa")]);
586    }
587    if let Some(b) = &t.borders {
588        borders(w, "w:tblBorders", b, Some("w:insideH"), &[]);
589    }
590    if let Some(s) = t.shading {
591        w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &s.hex())]);
592    }
593    if t.fixed {
594        w.empty("w:tblLayout", &[("w:type", "fixed")]);
595    }
596    if let Some(m) = &t.cell_margins {
597        margins(w, "w:tblCellMar", m);
598    }
599    w.close("w:tblPrEx");
600}
601
602fn trpr_inner(w: &mut W, r: &RowProps) {
603    // The grid gaps go first, in the schema's order (as Word writes them).
604    if r.grid_before > 0 {
605        w.val("w:gridBefore", &n(r.grid_before.into()));
606    }
607    if r.grid_after > 0 {
608        w.val("w:gridAfter", &n(r.grid_after.into()));
609    }
610    for (name, width) in [("w:wBefore", r.w_before), ("w:wAfter", r.w_after)] {
611        if let Some(v) = width {
612            w.empty(name, &[("w:w", &twips(v)), ("w:type", "dxa")]);
613        }
614    }
615    if r.cant_split {
616        w.empty("w:cantSplit", &[]);
617    }
618    if let Some(h) = r.height {
619        let rule = match r.height_rule {
620            HeightRule::Auto => "auto",
621            HeightRule::AtLeast => "atLeast",
622            HeightRule::Exact => "exact",
623        };
624        w.empty("w:trHeight", &[("w:val", &twips(h.abs())), ("w:hRule", rule)]);
625    }
626    if r.header {
627        w.empty("w:tblHeader", &[]);
628    }
629}

631/// `w:tcPr`, with a tracked change last as for [`tblpr`].
632pub fn tcpr(w: &mut W, c: &CellProps, change: Option<&ChangeAttrs>) {
633    w.open("w:tcPr", &[]);
634    tcpr_inner(w, c);
635    if let (Some(a), Some(ch)) = (change, &c.fmt_change) {
636        change_el(w, "w:tcPrChange", "w:tcPr", a, |w| tcpr_inner(w, &ch.old));
637    }
638    w.close("w:tcPr");
639}
640
641fn tcpr_inner(w: &mut W, c: &CellProps) {
642    match (c.width, c.width_pct) {
643        (Some(v), _) => w.empty("w:tcW", &[("w:w", &twips(v.max(0.0))), ("w:type", "dxa")]),
644        (None, Some(p)) => w.empty("w:tcW", &[("w:w", &n((p.clamp(0.0, 100.0) * 50.0).round() as i64)), ("w:type", "pct")]),
645        (None, None) => w.empty("w:tcW", &[("w:w", "0"), ("w:type", "auto")]),
646    }
647    if c.span > 1 {
648        w.val("w:gridSpan", &n(c.span.min(63) as i64));
649    }
650    match c.vmerge {
651        VMerge::Restart => w.val("w:vMerge", "restart"),
652        VMerge::Continue => w.empty("w:vMerge", &[]),
653        VMerge::None => {}
654    }
655    if let Some(b) = &c.borders {
656        borders(w, "w:tcBorders", b, Some("w:insideH"), &[]);
657    }
658    if let Some(s) = c.shading {
659        w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &s.hex())]);
660    }
661    if c.no_wrap {
662        w.empty("w:noWrap", &[]);
663    }
664    if let Some(m) = &c.margins {
665        margins(w, "w:tcMar", m);
666    } else if let Some(m) = &c.partial_margins {
667        margin_sides(w, "w:tcMar", m);
668    }
669    if c.text_direction.is_turned() || c.text_direction_set {
670        w.val("w:textDirection", c.text_direction.ooxml());
671    }
672    match c.valign {
673        None => {}
674        Some(VAlign::Top) => w.val("w:vAlign", "top"),
675        Some(VAlign::Center) => w.val("w:vAlign", "center"),
676        Some(VAlign::Bottom) => w.val("w:vAlign", "bottom"),
677    }
678    if c.hide_mark {
679        w.empty("w:hideMark", &[]);
680    }
681}

683/// `w:tblpPr` and `w:tblOverlap` for a floating table.
684fn table_float(w: &mut W, f: &TableFloat) {
685    use crate::doc::para::Anchor;
686    let rel = |a: Anchor, text: &'static str| match a {
687        Anchor::Page => "page",
688        Anchor::Margin => "margin",
689        _ => text,
690    };
691    let [left, top, right, bottom] = f.dist_from_text().map(twips);
692    let fin = |v: f32| twips(if v.is_finite() { v.clamp(-31_680.0, 31_680.0) } else { 0.0 });
693    let (x, y) = (fin(f.x), fin(f.y));
694    let mut a: Vec<(&str, &str)> = vec![
695        ("w:leftFromText", &left),
696        ("w:rightFromText", &right),
697        ("w:topFromText", &top),
698        ("w:bottomFromText", &bottom),
699        ("w:vertAnchor", rel(f.v_rel, "text")),
700        ("w:horzAnchor", rel(f.h_rel, "text")),
701    ];
702    match f.h_align {
703        Some(h) => a.push(("w:tblpXSpec", h.ooxml(true))),
704        None => a.push(("w:tblpX", &x)),
705    }
706    match f.v_align {
707        Some(v) => a.push(("w:tblpYSpec", v.ooxml(false))),
708        None => a.push(("w:tblpY", &y)),
709    }
710    w.empty("w:tblpPr", &a);
711    if !f.overlap {
712        w.val("w:tblOverlap", "never");
713    }
714}
