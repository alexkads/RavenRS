11/// The parts of the package [`write_as`] zips, in order, `[Content_Types].xml` first: (part name
12/// without the leading `/`, bytes).
13pub(crate) fn write_parts(doc: &Document, flavor: Flavor) -> Result<Vec<(String, Vec<u8>)>, DocxError> {
14    // A range field must be whole in the file: drop markers that lost their partner in editing.
15    let balanced;
16    let doc = if doc.has_unbalanced_field_ranges() {
17        let mut d = doc.clone();
18        d.balance_field_ranges();
19        balanced = d;
20        &balanced
21    } else {
22        doc
23    };
24    let mut wr = Writer {
25        eq_number: 0,
26        doc,
27        media_files: BTreeMap::new(),
28        bookmarks: HashMap::new(),
29        next_rev_id: 0,
30        open_moves: Vec::new(),
31        // Fresh drawing ids start past the shapes' own (connectors refer to those).
32        docpr: doc.next_shape_id().saturating_sub(1),
33        shape_ids: std::collections::HashSet::new(),
34        z: 251_658_240,
35        footnotes: Vec::new(),
36        endnotes: Vec::new(),
37        hf: Vec::new(),
38        linked_headers: Default::default(),
39        wm_header: (0..=u32::MAX).rev().find(|id| !doc.parts.contains_key(id)).unwrap_or(u32::MAX),
40        pending_watermark: None,
41        para_ids: HashMap::new(),
42        pending_mark: None,
43        current_note: None,
44        toc_hold_end: false,
45        toc_field: None,
46        toc_begin_here: false,
47        toc_end_here: false,
48        inline_controls: Vec::new(),
49        used_media: Default::default(),
50        boxes: crate::doc::BoxBudget::default(),
51        written_boxes: std::collections::HashSet::new(),
52        embeds: embed::EmbedWriter::new(doc),
53        txbx_links: doc
54            .text_box_chains()
55            .iter()
56            .enumerate()
57            .flat_map(|(n, c)| c.iter().enumerate().map(move |(seq, id)| (*id, (n as u32 + 1, seq as u32))))
58            .collect(),
59    };
60    wr.assign_media();
61    wr.embeds.reserve_media(wr.media_files.values());
62    let mut entries: Vec<(String, Vec<u8>)> = Vec::new();
63    let mut overrides: Vec<(String, String)> = Vec::new();
64    let mut rels = PartRels::default();
65    let mut root = PartRels::default();
66    let ns = xml::body_ns();
67    let ns_refs: Vec<(&str, &str)> = ns.iter().map(|(k, v)| (k.as_str(), v.as_str())).collect();
68
69    // Comments get paragraph ids before any story is written.
70    let mut comment_paras: Vec<(u32, String)> = Vec::new();
71    for (i, (cid, c)) in doc.comments.iter().enumerate() {
72        if let Some(part) = doc.parts.get(&c.part)
73            && let Some(last) = part.blocks.iter().rev().find_map(|b| b.as_para())
74        {
75            let pid = format!("{:08X}", 0x1000_0000u32 + i as u32 + 1);
76            wr.para_ids.insert(last as *const _ as usize, pid.clone());
77            comment_paras.push((*cid, pid));
78        }
79    }
80
81    let body = write_document_body(doc, &mut wr, &mut rels, &ns_refs);
82
83    // Headers and footers (discovered while writing sections).
84    let mut done = 0;
85    while let Some((id, file, footer)) = wr.hf.get(done).cloned() {
86        done += 1;
87        let mut prels = PartRels::default();
88        let mut w = W::new();
89        let tag = if footer { "w:ftr" } else { "w:hdr" };
90        w.open(tag, &ns_refs);
91        let blocks: Blocks = doc.parts.get(&id).map(|p| p.blocks.clone()).unwrap_or_default();
92        // The watermark goes in every header, at the start of its first paragraph.
93        wr.pending_watermark = match &doc.settings.watermark {
94            Some(wm) if !footer => watermark::run(wm, done, &doc.last_section),
95            _ => None,
96        };
97        wr.blocks(&mut w, &blocks, &mut prels, false, 0);
98        // Always taken: the story ends with a top-level paragraph.
99        wr.pending_watermark = None;
100        w.close(tag);
101        push_part(
102            &mut entries,
103            &mut overrides,
104            &format!("word/{file}"),
105            w.into_bytes(),
106            &format!("{CT_WML}{}+xml", if footer { "footer" } else { "header" }),
107            prels,
108        );
109    }
110
111    // The glossary: kept verbatim while its entries are as read and its parts were kept, else
112    // written from the model (before the notes: references its entries hold are collected too).
113    // When the privacy settings ask for changes, its kept parts are written without the names and
114    // dates of its tracked changes and comments (#3692), or from the model if they can't be.
115    let (names, dates) = (doc.settings.remove_personal_info, doc.settings.remove_date_time);
116    if glossary::edited(doc)
117        || (!doc.glossary.is_empty() && !wr.embeds.kept_glossary())
118        || ((names || dates) && !wr.embeds.scrub_glossary(names, dates))
119    {
120        wr.embeds.drop_glossary = true;
121        wr.glossary(&mut entries, &mut overrides, &mut rels);
122    }
123
124    write_notes(doc, &ns_refs, &mut wr, &mut entries, &mut overrides, &mut rels);
125
126    write_comments(doc, &comment_paras, &ns_refs, &mut wr, &mut entries, &mut overrides, &mut rels);
127
128    rels.add(rt::STYLES, "styles.xml", false);
129    push_part(&mut entries, &mut overrides, "word/styles.xml", styles_xml(&mut wr), &format!("{CT_WML}styles+xml"), PartRels::default());
130    if !doc.numbering.nums.is_empty() || !doc.numbering.abstracts.is_empty() {
131        rels.add(rt::NUMBERING, "numbering.xml", false);
132        push_part(&mut entries, &mut overrides, "word/numbering.xml", numbering_xml(doc), &format!("{CT_WML}numbering+xml"), PartRels::default());
133    }
134    rels.add(rt::SETTINGS, "settings.xml", false);
135    // The attached template is an external relationship of the settings part (§17.15.1.6).
136    let mut settings_rels = PartRels::default();
137    let template = doc.settings.attached_template.as_deref().map(str::trim).filter(|t| !t.is_empty());
138    let template_rel = template.map(|t| settings_rels.add(rt::ATTACHED_TEMPLATE, t, true));
139    push_part(
140        &mut entries,
141        &mut overrides,
142        "word/settings.xml",
143        {
144            let generated = settings_xml(doc, notes_part(doc, &wr, true), notes_part(doc, &wr, false), template_rel.as_deref());
145            match doc.passthrough.get(crate::docx::package::SETTINGS_ORIG) {
146                Some(orig) => settings_merge::merge(orig, &generated).unwrap_or(generated),
147                None => generated,
148            }
149        },
150        &format!("{CT_WML}settings+xml"),
151        settings_rels,
152    );
153    rels.add(rt::THEME, "theme/theme1.xml", false);
154    push_part(
155        &mut entries,
156        &mut overrides,
157        "word/theme/theme1.xml",
158        theme_xml(doc),
159        "application/vnd.openxmlformats-officedocument.theme+xml",
160        PartRels::default(),
161    );
162
163    wr.embeds.font_table(&mut rels);
164
165    // Media referenced by some drawing (orphan parts would make the package suspect).
166    let mut media_types: BTreeMap<String, &str> = BTreeMap::new();
167    for (key, file) in &wr.media_files {
168        if !wr.used_media.contains(key) {
169            continue;
170        }
171        if let Some(bytes) = doc.media.get(key) {
172            let ext = file.rsplit('.').next().unwrap_or("bin").to_ascii_lowercase();
173            media_types.insert(ext, content_type_for(file));
174            entries.push((format!("word/media/{file}"), bytes.to_vec()));
175        }
176    }
177
178    wr.embeds.custom_xml_rels("D", &mut rels);
179    wr.embeds.custom_xml_rels("P", &mut root);
180    wr.embeds.unmodelled_rels("D", &mut rels);
181    wr.embeds.unmodelled_rels("P", &mut root);
182    // Parts of embedded objects and custom XML stores kept from the file.
183    for (name, bytes, ct, prels) in std::mem::take(&mut wr.embeds.parts) {
184        push_part(&mut entries, &mut overrides, &name, bytes, &ct, prels);
185    }
186
187    // The macro project and the parts it relates to (VBA data, signatures…), verbatim. A
188    // macro-free package can't hold them (Word drops them too).
189    if let Some(vba) = doc.passthrough.get(VBA_PROJECT_PART).filter(|b| !b.is_empty()) {
190        if flavor.macros() {
191            let mut vba_rels = PartRels::default();
192            let manifest = doc.passthrough.get(VBA_RELATED).map(|m| String::from_utf8_lossy(m).into_owned()).unwrap_or_default();
193            let mut written: Vec<String> = Vec::new();
194            for line in manifest.lines().take(MAX_VBA_RELATED) {
195                let mut f = line.split('\t');
196                let (Some(kind), Some(path), Some(ct), None) = (f.next(), f.next(), f.next(), f.next()) else { continue };
197                let Some(target) = path.strip_prefix("word/").filter(|t| !t.is_empty()) else { continue };
198                let Some(bytes) = doc.passthrough.get(path) else { continue };
199                let ours = written.iter().any(|w| w.eq_ignore_ascii_case(path));
200                // Never shadow a part this writer produces (a hostile relationship may point at one).
201                let clash = path.eq_ignore_ascii_case("word/document.xml")
202                    || path.eq_ignore_ascii_case(VBA_PROJECT_PART)
203                    || path.to_ascii_lowercase().ends_with(".rels")
204                    || entries.iter().any(|(n, _)| n.eq_ignore_ascii_case(path));
205                if clash && !ours {
206                    continue;
207                }
208                // A relative reference whose first segment has a colon would read as a URI scheme.
209                let target = if target.split('/').next().is_some_and(|seg| seg.contains(':')) { format!("./{target}") } else { target.to_string() };
210                vba_rels.add(kind, &target, false);
211                if !ours {
212                    push_part(&mut entries, &mut overrides, path, bytes.to_vec(), ct, PartRels::default());
213                    written.push(path.to_string());
214                }
215            }
216            rels.add(rt::VBA_PROJECT, "vbaProject.bin", false);
217            push_part(&mut entries, &mut overrides, VBA_PROJECT_PART, vba.to_vec(), "application/vnd.ms-office.vbaProject", vba_rels);
218        } else {
219            log::warn!("docx: {flavor:?} can't hold macros; the VBA project is left out");
220        }
221    }
222
223    // Main part goes first in the zip after content types.
224    entries.insert(0, ("word/document.xml".into(), body));
225    overrides.insert(0, ("/word/document.xml".into(), flavor.main_content_type().into()));
226    entries.insert(1, ("word/_rels/document.xml.rels".into(), rels.xml()));
227
228    // Package-level parts.
229    root.add(rt::OFFICE_DOC, "word/document.xml", false);
230    root.add(rt::CORE, "docProps/core.xml", false);
231    root.add(rt::APP, "docProps/app.xml", false);
232    entries.push(("docProps/core.xml".into(), core_xml(doc)));
233    overrides.push(("/docProps/core.xml".into(), "application/vnd.openxmlformats-package.core-properties+xml".into()));
234    entries.push(("docProps/app.xml".into(), app_xml(doc)));
235    overrides.push(("/docProps/app.xml".into(), "application/vnd.openxmlformats-officedocument.extended-properties+xml".into()));
236    if !doc.custom_props.is_empty() {
237        root.add(rt::CUSTOM, "docProps/custom.xml", false);
238        entries.push(("docProps/custom.xml".into(), crate::docx::custom::write(&doc.custom_props)));
239        overrides.push(("/docProps/custom.xml".into(), "application/vnd.openxmlformats-officedocument.custom-properties+xml".into()));
240    }
241    if !doc.sources.is_empty() {
242        root.add(rt::CUSTOM_XML, "customXml/item1.xml", false);
243        let mut item_rels = PartRels::default();
244        item_rels.add(rt::CUSTOM_XML_PROPS, "itemProps1.xml", false);
245        push_part(
246            &mut entries,
247            &mut overrides,
248            "customXml/item1.xml",
249            bibliography_xml(doc),
250            "application/vnd.openxmlformats-officedocument.customXml+xml",
251            item_rels,
252        );
253        push_part(
254            &mut entries,
255            &mut overrides,
256            "customXml/itemProps1.xml",
257            bibliography_props_xml(),
258            "application/vnd.openxmlformats-officedocument.customXmlProperties+xml",
259            PartRels::default(),
260        );
261    }
262    entries.insert(0, ("_rels/.rels".into(), root.xml()));
263
264    // Content types.
265    let mut w = W::new();
266    w.open("Types", &[("xmlns", "http://schemas.openxmlformats.org/package/2006/content-types")]);
267    w.empty("Default", &[("Extension", "rels"), ("ContentType", "application/vnd.openxmlformats-package.relationships+xml")]);
268    w.empty("Default", &[("Extension", "xml"), ("ContentType", "application/xml")]);
269    for (ext, ct) in &media_types {
270        if ext != "xml" && ext != "rels" {
271            w.empty("Default", &[("Extension", ext), ("ContentType", ct)]);
272        }
273    }
274    for (part, ct) in &overrides {
275        w.empty("Override", &[("PartName", part), ("ContentType", ct)]);
276    }
277    w.close("Types");
278    entries.insert(0, ("[Content_Types].xml".into(), w.into_bytes()));
279    Ok(entries)
280}

282fn write_document_body(doc: &Document, wr: &mut Writer<'_>, rels: &mut PartRels, ns_refs: &[(&str, &str)]) -> Vec<u8> {
283    let mut w = W::new();
284    w.open("w:document", ns_refs);
285    let pic = doc.settings.page_picture.as_ref().filter(|p| wr.media_files.contains_key(&p.media) && doc.media.contains_key(&p.media));
286    if pic.is_some() || doc.settings.page_color.is_some() {
287        let color = doc.settings.page_color.map(|c| c.hex());
288        let attrs: Vec<(&str, &str)> = color.iter().map(|c| ("w:color", c.as_str())).collect();
289        w.open("w:background", &attrs);
290        if let Some(p) = pic
291            && let Some(file) = wr.media_files.get(&p.media)
292        {
293            let rid = rels.add(rt::IMAGE, &format!("media/{file}"), false);
294            wr.used_media.insert(p.media.clone());
295            w.open("v:background", &[("id", "_x0000_s1025"), ("o:bwmode", "white")]);
296            let mut fa: Vec<(&str, &str)> = vec![("r:id", &rid)];
297            // VML has no centred fill (an ODT page background's): stretched to the page.
298            let kind = if p.fill_type.eq_ignore_ascii_case("center") { "frame" } else { p.fill_type.as_str() };
299            if !kind.is_empty() {
300                fa.push(("type", kind));
301            }
302            if !p.title.is_empty() {
303                fa.push(("o:title", &p.title));
304            }
305            w.empty("v:fill", &fa);
306            w.close("v:background");
307        }
308        w.close("w:background");
309    }
310    w.open("w:body", &[]);
311    wr.blocks(&mut w, &doc.body, rels, true, 0);
312    let last = doc.last_section.clone();
313    wr.sectpr(&mut w, &last, rels);
314    w.close("w:body");
315    w.close("w:document");
316    w.into_bytes()
317}

319fn write_notes(
320    doc: &Document,
321    ns_refs: &[(&str, &str)],
322    wr: &mut Writer<'_>,
323    entries: &mut Vec<(String, Vec<u8>)>,
324    overrides: &mut Vec<(String, String)>,
325    rels: &mut PartRels,
326) {
327    for foot in [true, false] {
328        let seps = if foot { &doc.footnote_separators } else { &doc.endnote_separators };
329        if !notes_part(doc, wr, foot) {
330            continue;
331        }
332        let (root, tag, file) = if foot { ("w:footnotes", "w:footnote", "footnotes.xml") } else { ("w:endnotes", "w:endnote", "endnotes.xml") };
333        let mut prels = PartRels::default();
334        let mut w = W::new();
335        w.open(root, ns_refs);
336        // The separators and continuation notice: the document's own, else the defaults (no
337        // notice). Word numbers them -1, 0 and 1.
338        for (id, kind, own) in [
339            ("-1", "separator", &seps.separator),
340            ("0", "continuationSeparator", &seps.continuation_separator),
341            ("1", "continuationNotice", &seps.continuation_notice),
342        ] {
343            match own {
344                Some(blocks) => {
345                    w.open(tag, &[("w:type", kind), ("w:id", id)]);
346                    if blocks.is_empty() {
347                        w.empty("w:p", &[]);
348                    } else {
349                        wr.blocks(&mut w, blocks, &mut prels, false, 0);
350                    }
351                    w.close(tag);
352                }
353                None if kind != "continuationNotice" => {
354                    w.open(tag, &[("w:type", kind), ("w:id", id)]);
355                    w.open("w:p", &[]);
356                    w.open("w:pPr", &[]);
357                    w.empty("w:spacing", &[("w:after", "0"), ("w:line", "240"), ("w:lineRule", "auto")]);
358                    w.close("w:pPr");
359                    w.open("w:r", &[]);
360                    w.empty(if kind == "separator" { "w:separator" } else { "w:continuationSeparator" }, &[]);
361                    w.close("w:r");
362                    w.close("w:p");
363                    w.close(tag);
364                }
365                None => {}
366            }
367        }
368        let base = note_id_base(doc, foot);
369        let mut k = 0;
370        while let Some(pid) = (if foot { &wr.footnotes } else { &wr.endnotes }).get(k).copied() {
371            let nid = (k + base).to_string();
372            k += 1;
373            w.open(tag, &[("w:id", &nid)]);
374            let blocks: Blocks = doc.parts.get(&pid).map(|p| p.blocks.clone()).unwrap_or_default();
375            wr.note_blocks(&mut w, &blocks, &mut prels, foot, pid);
376            w.close(tag);
377        }
378        w.close(root);
379        let kind = if foot { rt::FOOTNOTES } else { rt::ENDNOTES };
380        rels.add(kind, file, false);
381        push_part(
382            entries,
383            overrides,
384            &format!("word/{file}"),
385            w.into_bytes(),
386            &format!("{CT_WML}{}+xml", if foot { "footnotes" } else { "endnotes" }),
387            prels,
388        );
389    }
390}

392fn write_comments(
393    doc: &Document,
394    comment_paras: &[(u32, String)],
395    ns_refs: &[(&str, &str)],
396    wr: &mut Writer<'_>,
397    entries: &mut Vec<(String, Vec<u8>)>,
398    overrides: &mut Vec<(String, String)>,
399    rels: &mut PartRels,
400) {
401    if !doc.comments.is_empty() {
402        let mut prels = PartRels::default();
403        let mut w = W::new();
404        w.open("w:comments", ns_refs);
405        for (cid, c) in &doc.comments {
406            let id = cid.to_string();
407            let author = annotation_author(doc, &c.author);
408            let mut a = vec![("w:id", id.as_str()), ("w:author", author)];
409            if !c.date.is_empty() && !doc.settings.remove_date_time {
410                a.push(("w:date", c.date.as_str()));
411            }
412            if doc.settings.remove_personal_info {
413                a.push(("w:initials", ANONYMOUS_INITIALS));
414            } else if !c.initials.is_empty() {
415                a.push(("w:initials", c.initials.as_str()));
416            }
417            w.open("w:comment", &a);
418            let blocks: Blocks = doc.parts.get(&c.part).map(|p| p.blocks.clone()).unwrap_or_default();
419            wr.blocks(&mut w, &blocks, &mut prels, false, 0);
420            w.close("w:comment");
421        }
422        w.close("w:comments");
423        rels.add(rt::COMMENTS, "comments.xml", false);
424        push_part(entries, overrides, "word/comments.xml", w.into_bytes(), &format!("{CT_WML}comments+xml"), prels);
425
426        let mut w = W::new();
427        w.open(
428            "w15:commentsEx",
429            &[
430                ("xmlns:w15", "http://schemas.microsoft.com/office/word/2012/wordml"),
431                ("xmlns:mc", "http://schemas.openxmlformats.org/markup-compatibility/2006"),
432            ],
433        );
434        // Assigned in BTreeMap comment-ID order above, including gaps for missing parts.
435        let pid_of =
436            |cid: u32| comment_paras.binary_search_by_key(&cid, |(c, _)| *c).ok().and_then(|i| comment_paras.get(i)).map(|(_, p)| p.as_str());
437        for (cid, c) in &doc.comments {
438            let Some(pid) = pid_of(*cid) else { continue };
439            let done = if c.resolved { "1" } else { "0" };
440            match c.parent.and_then(pid_of) {
441                Some(parent) => w.empty("w15:commentEx", &[("w15:paraId", pid), ("w15:paraIdParent", parent), ("w15:done", done)]),
442                None => w.empty("w15:commentEx", &[("w15:paraId", pid), ("w15:done", done)]),
443            }
444        }
445        w.close("w15:commentsEx");
446        rels.add(rt::COMMENTS_EX, "commentsExtended.xml", false);
447        push_part(entries, overrides, "word/commentsExtended.xml", w.into_bytes(), &format!("{CT_WML}commentsExtended+xml"), PartRels::default());
448        write_comment_ids(doc, comment_paras, entries, overrides, rels);
449    }
450}

452pub(super) fn push_part(
453    entries: &mut Vec<(String, Vec<u8>)>,
454    overrides: &mut Vec<(String, String)>,
455    path: &str,
456    bytes: Vec<u8>,
457    ct: &str,
458    rels: PartRels,
459) {
460    if !rels.is_empty() {
461        let (dir, file) = path.rsplit_once('/').unwrap_or(("", path));
462        entries.push((format!("{dir}/_rels/{file}.rels"), rels.xml()));
463    }
464    entries.push((path.to_string(), bytes));
465    overrides.push((format!("/{path}"), ct.to_string()));
466}

468/// Content type by file extension (or sniffed name).
469pub(super) fn content_type_for(file: &str) -> &'static str {
470    match file.rsplit('.').next().unwrap_or("").to_ascii_lowercase().as_str() {
471        "png" => "image/png",
472        "jpg" | "jpeg" | "jpe" => "image/jpeg",
473        "gif" => "image/gif",
474        "bmp" => "image/bmp",
475        "webp" => "image/webp",
476        "tif" | "tiff" => "image/tiff",
477        "svg" => "image/svg+xml",
478        "emf" => "image/x-emf",
479        "wmf" => "image/x-wmf",
480        "ico" => "image/x-icon",
481        _ => "application/octet-stream",
482    }
483}

485/// Guess an image extension from magic bytes.
486pub(super) fn sniff(bytes: &[u8]) -> &'static str {
487    if bytes.starts_with(b"\x89PNG") {
488        "png"
489    } else if bytes.starts_with(&[0xFF, 0xD8]) {
490        "jpeg"
491    } else if bytes.starts_with(b"GIF8") {
492        "gif"
493    } else if bytes.starts_with(b"BM") {
494        "bmp"
495    } else if bytes.starts_with(b"RIFF") && bytes.get(8..12) == Some(b"WEBP") {
496        "webp"
497    } else if bytes.starts_with(b"II*\0") || bytes.starts_with(b"MM\0*") {
498        "tiff"
499    } else if bytes.starts_with(&[0xD7, 0xCD, 0xC6, 0x9A]) {
500        "wmf"
501    } else if bytes.get(40..44) == Some(b" EMF") {
502        "emf"
503    } else if bytes.starts_with(b"<svg") || bytes.starts_with(b"<?xml") {
504        "svg"
505    } else {
506        "bin"
507    }
