4impl Writer<'_> {
5    fn write_trailing_para(&mut self, w: &mut W, depth: usize) {
6        // A header with no paragraph of its own before: the watermark goes in this one.
7        if depth == 0
8            && let Some(run) = self.pending_watermark.take()
9        {
10            w.raw("<w:p>");
11            w.raw(&run);
12            w.raw("</w:p>");
13        } else {
14            w.raw("<w:p/>");
15        }
16    }

18    /// Write a block list. `top` = body level (sections allowed).
19    pub fn blocks(&mut self, w: &mut W, bl: &Blocks, rels: &mut PartRels, top: bool, depth: usize) {
20        if depth > MAX_DEPTH {
21            w.raw("<w:p/>");
22            return;
23        }
24        let plan = crate::docx::write::sdt::plan(bl);
25        let toc = if top { written_toc_span(bl) } else { None };
26        // Whether the heading's TOC field was written and still needs its end.
27        let mut toc_field_open = false;
28        for (i, b) in bl.iter().enumerate() {
29            for c in plan.opens.get(i).into_iter().flatten() {
30                crate::docx::write::sdt::open(w, c);
31            }
32            self.update_toc_before_block(i, toc, toc_field_open, w);
33            match &**b {
34                Block::Para(p) => {
35                    self.inline_controls = plan.inline_in(i);
36                    self.para(w, p, rels, top, depth);
37                }
38                Block::Table(t) => self.table(w, t, rels, depth),
39            }
40            self.update_toc_after_block(i, toc, &mut toc_field_open, w);
41            for _ in 0..plan.closes.get(i).copied().unwrap_or(0) {
42                crate::docx::write::sdt::close(w);
43            }
44        }
45        if depth == 0 {
46            self.move_ranges(w, &[]);
47        }
48        if bl.last().is_none_or(|b| matches!(**b, Block::Table(_))) {
49            self.write_trailing_para(w, depth);
50        }
51    }

53    /// The moves (revision, destination?) in `props`.
54    pub(super) fn moves_of(&self, props: &CharProps) -> Vec<(u32, bool)> {
55        [(props.del, false), (props.ins, true)]
56            .into_iter()
57            .filter_map(|(r, to)| r.filter(|r| self.doc.move_name(Some(*r)).is_some()).map(|r| (r, to)))
58            .collect()
59    }

61    /// Bring the open move ranges to `want` (ECMA-376 §17.13.5.22–27): end the ranges of moves
62    /// that stop here, start those that begin. A range is written around each stretch of a
63    /// move's text (and paragraph marks), with the move's name, author and date.
64    pub(super) fn move_ranges(&mut self, w: &mut W, want: &[(u32, bool)]) {
65        let mut k = 0;
66        while k < self.open_moves.len() {
67            match self.open_moves.get(k) {
68                Some((rev, id, to)) if !want.contains(&(*rev, *to)) => {
69                    w.empty(if *to { "w:moveToRangeEnd" } else { "w:moveFromRangeEnd" }, &[("w:id", id)]);
70                    self.open_moves.remove(k);
71                }
72                _ => k += 1,
73            }
74        }
75        for &(rev, to) in want {
76            if self.open_moves.iter().any(|(r, _, t)| *r == rev && *t == to) {
77                continue;
78            }
79            let name = self.doc.move_name(Some(rev)).unwrap_or("").to_string();
80            let (id, author, date) = self.rev_attrs(rev);
81            let tag = if to { "w:moveToRangeStart" } else { "w:moveFromRangeStart" };
82            if date.is_empty() {
83                w.empty(tag, &[("w:id", &id), ("w:name", &name), ("w:author", &author)]);
84            } else {
85                w.empty(tag, &[("w:id", &id), ("w:name", &name), ("w:author", &author), ("w:date", &date)]);
86            }
87            self.open_moves.push((rev, id, to));
88        }
89    }

91    fn table(&mut self, w: &mut W, t: &Table, rels: &mut PartRels, depth: usize) {
92        let rows: Vec<_> = t.rows.iter().filter(|r| !r.cells.is_empty()).collect();
93        if rows.is_empty() {
94            return;
95        }
96        w.open("w:tbl", &[]);
97        let chg = t.props.fmt_change.as_ref().map(|c| self.change_attrs(c.rev));
98        tblpr(w, &t.props, chg.as_ref());
99        w.open("w:tblGrid", &[]);
100        let cols = t.cols().clamp(1, 63);
101        for g in 0..cols {
102            let gw = t.grid.get(g).copied().unwrap_or(72.0);
103            w.empty("w:gridCol", &[("w:w", &twips(gw.max(0.0)))]);
104        }
105        w.close("w:tblGrid");
106        // Controls around rows and cells (repeating sections): opens and closes balanced
107        // whatever editing left, the ones still open closed at the end.
108        let mut open_rows = 0u32;
109        for r in rows {
110            for c in &r.controls.open {
111                crate::docx::write::sdt::open(w, c);
112                open_rows += 1;
113            }
114            w.open("w:tr", &[]);
115            let chg = r.props.fmt_change.as_ref().map(|c| self.change_attrs(c.rev));
116            trpr(w, &r.props, chg.as_ref());
117            let mut open_cells = 0u32;
118            for c in &r.cells {
119                for ctl in &c.controls.open {
120                    crate::docx::write::sdt::open(w, ctl);
121                    open_cells += 1;
122                }
123                w.open("w:tc", &[]);
124                let chg = c.props.fmt_change.as_ref().map(|c| self.change_attrs(c.rev));
125                tcpr(w, &c.props, chg.as_ref());
126                self.blocks(w, &c.blocks, rels, false, depth + 1);
127                w.close("w:tc");
128                let n = c.controls.close.min(open_cells);
129                for _ in 0..n {
130                    crate::docx::write::sdt::close(w);
131                }
132                open_cells -= n;
133            }
134            for _ in 0..open_cells {
135                crate::docx::write::sdt::close(w);
136            }
137            w.close("w:tr");
138            let n = r.controls.close.min(open_rows);
139            for _ in 0..n {
140                crate::docx::write::sdt::close(w);
141            }
142            open_rows -= n;
143        }
144        for _ in 0..open_rows {
145            crate::docx::write::sdt::close(w);
146        }
147        w.close("w:tbl");
148    }
