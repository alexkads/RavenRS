3impl Writer<'_> {
4    /// The section's headers, plus the header holding only the watermark in each slot the
5    /// pages show (first page with a title page, even pages with even/odd headers) that has no
6    /// header of its own or linked from an earlier section. Sections are written in body order.
7    fn watermark_headers(&mut self, s: &SectionProps) -> crate::doc::section::HeaderSet {
8        let mut set = s.headers;
9        let linked = &mut self.linked_headers;
10        if self.doc.settings.watermark.as_ref().and_then(crate::docx::write::watermark::text).is_some() {
11            let id = self.wm_header;
12            if set.default.or(linked.default).is_none() {
13                set.default = Some(id);
14            }
15            if s.title_page && set.first.or(linked.first).is_none() {
16                set.first = Some(id);
17            }
18            if self.doc.settings.even_odd_headers && set.even.or(linked.even).is_none() {
19                set.even = Some(id);
20            }
21        }
22        linked.default = set.default.or(linked.default);
23        linked.first = set.first.or(linked.first);
24        linked.even = set.even.or(linked.even);
25        set
26    }

28    pub fn sectpr(&mut self, w: &mut W, s: &SectionProps, rels: &mut PartRels) {
29        w.open("w:sectPr", &[]);
30        let headers = self.watermark_headers(s);
31        for (set, footer) in [(&headers, false), (&s.footers, true)] {
32            for (kind, id) in [("default", set.default), ("first", set.first), ("even", set.even)] {
33                let Some(id) = id else { continue };
34                if let Some(rid) = self.hf_rel(id, footer, rels) {
35                    w.empty(if footer { "w:footerReference" } else { "w:headerReference" }, &[("w:type", kind), ("r:id", &rid)]);
36                }
37            }
38        }
39        sectpr_body(w, s);
40        // The properties before a tracked change come last (§17.13.5.32).
41        if let Some(ch) = &s.fmt_change {
42            let a = self.change_attrs(ch.rev);
43            open_change(w, "w:sectPrChange", &a);
44            w.open("w:sectPr", &[]);
45            sectpr_body(w, &ch.old);
46            w.close("w:sectPr");
47            w.close("w:sectPrChange");
48        }
49        w.close("w:sectPr");
50    }
51}

53/// `w:sectPr` content after the header and footer references.
54fn sectpr_body(w: &mut W, s: &SectionProps) {
55    {
56        // The section's own note options come before `w:type` (ECMA-376 §17.6.17).
57        for (tag, pr) in [("w:footnotePr", &s.footnote_pr), ("w:endnotePr", &s.endnote_pr)] {
58            if !pr.is_empty() {
59                w.open(tag, &[]);
60                note_pr_inner(w, pr);
61                w.close(tag);
62            }
63        }
64        let start = match s.start {
65            SectionStart::NextPage => "nextPage",
66            SectionStart::Continuous => "continuous",
67            SectionStart::EvenPage => "evenPage",
68            SectionStart::OddPage => "oddPage",
69            SectionStart::NextColumn => "nextColumn",
70        };
71        w.val("w:type", start);
72        let (pw, ph) = (s.page_w.clamp(36.0, crate::docx::units::MAX_LEN_PT), s.page_h.clamp(36.0, crate::docx::units::MAX_LEN_PT));
73        if s.landscape {
74            w.empty("w:pgSz", &[("w:w", &twips(pw)), ("w:h", &twips(ph)), ("w:orient", "landscape")]);
75        } else {
76            w.empty("w:pgSz", &[("w:w", &twips(pw)), ("w:h", &twips(ph))]);
77        }
78        w.empty(
79            "w:pgMar",
80            &[
81                ("w:top", &twips(s.margin_top)),
82                ("w:right", &twips(s.margin_right.max(0.0))),
83                ("w:bottom", &twips(s.margin_bottom)),
84                ("w:left", &twips(s.margin_left.max(0.0))),
85                ("w:header", &twips(s.header.max(0.0))),
86                ("w:footer", &twips(s.footer.max(0.0))),
87                ("w:gutter", &twips(s.gutter.max(0.0))),
88            ],
89        );
90        paper_source(w, s);
91        if let Some(b) = &s.page_borders {
92            borders(w, "w:pgBorders", b, None, &[("w:offsetFrom", "text")]);
93        }
94        if let Some(l) = &s.line_numbers {
95            let restart = match l.restart {
96                LineNumberRestart::Page => "newPage",
97                LineNumberRestart::Section => "newSection",
98                LineNumberRestart::Continuous => "continuous",
99            };
100            let mut a = vec![
101                ("w:countBy", n(l.count_by.clamp(1, 100) as i64)),
102                ("w:start", n(l.start.saturating_sub(1).min(32_767) as i64)),
103                ("w:restart", restart.to_string()),
104            ];
105            if l.distance > 0.0 {
106                a.push(("w:distance", twips(l.distance)));
107            }
108            let refs: Vec<(&str, &str)> = a.iter().map(|(k, v)| (*k, v.as_str())).collect();
109            w.empty("w:lnNumType", &refs);
110        }
111        page_number_type(w, s);
112        let c = &s.columns;
113        let count = c.count.clamp(1, 45);
114        let mut ca = vec![("w:space", twips(c.space.max(0.0))), ("w:num", n(count as i64))];
115        if c.separator {
116            ca.push(("w:sep", "1".into()));
117        }
118        let unequal = !c.widths.is_empty() && c.widths.len() == count as usize;
119        if unequal {
120            ca.push(("w:equalWidth", "0".into()));
121            let refs: Vec<(&str, &str)> = ca.iter().map(|(k, v)| (*k, v.as_str())).collect();
122            w.open("w:cols", &refs);
123            for (cw, sp) in &c.widths {
124                w.empty("w:col", &[("w:w", &twips(cw.max(0.0))), ("w:space", &twips(sp.max(0.0)))]);
125            }
126            w.close("w:cols");
127        } else {
128            let refs: Vec<(&str, &str)> = ca.iter().map(|(k, v)| (*k, v.as_str())).collect();
129            w.empty("w:cols", &refs);
130        }
131        if let Some(count) = s.footnote_columns.filter(|count| *count > 0) {
132            w.empty("w15:footnoteColumns", &[("w:val", &count.clamp(1, 45).to_string())]);
133        }
134        if let Some(on) = s.form_prot {
135            w.empty("w:formProt", if on { &[] } else { &[("w:val", "0")] });
136        }
137        match s.valign {
138            crate::doc::section::PageVAlign::Top => {}
139            crate::doc::section::PageVAlign::Center => w.val("w:vAlign", "center"),
140            crate::doc::section::PageVAlign::Justify => w.val("w:vAlign", "both"),
141            crate::doc::section::PageVAlign::Bottom => w.val("w:vAlign", "bottom"),
142        }
143        if let Some(on) = s.no_endnote {
144            w.empty("w:noEndnote", if on { &[] } else { &[("w:val", "0")] });
145        }
146        if s.title_page {
147            w.empty("w:titlePg", &[]);
148        }
149        if s.text_direction.is_turned() || s.text_direction_set {
150            w.val("w:textDirection", s.text_direction.ooxml());
151        }
152        if s.rtl {
153            w.empty("w:bidi", &[]);
154        }
155        if let Some(g) = &s.doc_grid {
156            let mut a = vec![("w:linePitch", twips(g.line_pitch()))];
157            if g.kind != crate::doc::section::DocGridType::Default {
158                a.insert(0, ("w:type", g.kind.ooxml().to_string()));
159            }
160            if g.char_space != 0 {
161                let cs = g.char_space.clamp(-crate::doc::section::MAX_CHAR_SPACE, crate::doc::section::MAX_CHAR_SPACE);
162                a.push(("w:charSpace", n(cs as i64)));
163            }
164            let refs: Vec<(&str, &str)> = a.iter().map(|(k, v)| (*k, v.as_str())).collect();
165            w.empty("w:docGrid", &refs);
166        }
167    }
168}

170fn paper_source(w: &mut W, s: &SectionProps) {
171    let mut attrs = Vec::new();
172    if let Some(first) = s.paper_src_first {
173        attrs.push(("w:first", first.to_string()));
174    }
175    if let Some(other) = s.paper_src_other {
176        attrs.push(("w:other", other.to_string()));
177    }
178    if !attrs.is_empty() {
179        let refs: Vec<(&str, &str)> = attrs.iter().map(|(k, v)| (*k, v.as_str())).collect();
180        w.empty("w:paperSrc", &refs);
181    }
182}

184fn page_number_type(w: &mut W, s: &SectionProps) {
185    if !(s.page_num_set
186        || s.page_num_start.is_some()
187        || s.page_num_format != crate::doc::section::NumFormat::Decimal
188        || s.chapter_style.is_some()
189        || s.chapter_sep.is_some())
190    {
191        return;
192    }
193    let mut attrs = vec![("w:fmt", s.page_num_format.ooxml().to_string())];
194    if let Some(start) = s.page_num_start {
195        attrs.push(("w:start", start.to_string()));
196    }
197    if let Some(style) = s.chapter_style {
198        attrs.push(("w:chapStyle", style.to_string()));
199    }
200    if let Some(separator) = s.chapter_sep {
201        attrs.push(("w:chapSep", separator.ooxml().to_string()));
202    }
203    let refs: Vec<(&str, &str)> = attrs.iter().map(|(k, v)| (*k, v.as_str())).collect();
204    w.empty("w:pgNumType", &refs);
205}

207/// The children of a `w:footnotePr` / `w:endnotePr` in schema order (`w:pos`, `w:numFmt`,
208/// `w:numStart`, `w:numRestart`; ECMA-376 §17.11.11), for the fields that are set.
209pub(in super::super) fn note_pr_inner(w: &mut W, pr: &NoteProps) {
210    if let Some(p) = pr.pos {
211        w.val("w:pos", p.ooxml());
212    }
213    if let Some(f) = pr.num_fmt {
214        w.val("w:numFmt", f.ooxml());
215    }
216    if let Some(n) = pr.num_start {
217        w.val("w:numStart", &n.clamp(1, crate::doc::section::MAX_NOTE_START).to_string());
218    }
219    if let Some(r) = pr.num_restart {
220        w.val("w:numRestart", r.ooxml());
221    }
222}

224/// The character style for comment reference marks: the document's "annotation reference" style
225/// (Word's built-in name; localized documents give it another id, e.g. Polish
226/// `Odwoaniedokomentarza`), else `CommentReference`.
227pub(super) fn comment_reference_style(doc: &crate::doc::Document) -> &str {
228    doc.styles
229        .styles
230        .iter()
231        .find(|s| s.kind == crate::doc::styles::StyleKind::Character && s.name.eq_ignore_ascii_case("annotation reference"))
232        .map_or("CommentReference", |s| s.id.as_str())
