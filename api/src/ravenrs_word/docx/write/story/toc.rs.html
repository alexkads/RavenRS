3impl Writer<'_> {
4    pub(super) fn update_toc_before_block(&mut self, i: usize, toc: Option<(usize, usize)>, toc_field_open: bool, w: &mut W) {
5        if let Some((start, end)) = toc {
6            if i == start {
7                w.raw(TOC_SDT_OPEN);
8                self.toc_hold_end = true;
9            }
10            self.toc_begin_here = toc_field_open && i == start + 1;
11            self.toc_end_here = toc_field_open && i == end;
12        }
13    }
14
15    pub(super) fn update_toc_after_block(&mut self, i: usize, toc: Option<(usize, usize)>, toc_field_open: &mut bool, w: &mut W) {
16        if let Some((start, end)) = toc {
17            if i == start {
18                // `field` holds the heading's TOC field back for the first entry.
19                self.toc_hold_end = false;
20                *toc_field_open = self.toc_field.is_some();
21            }
22            if i == end {
23                w.raw("</w:sdtContent></w:sdt>");
24            }
25        }
26    }
27}

29/// Word's Table of Contents content control, which gives the TOC its frame and Update Table.
30const TOC_SDT_OPEN: &str =
31    r#"<w:sdt><w:sdtPr><w:docPartObj><w:docPartGallery w:val="Table of Contents"/><w:docPartUnique/></w:docPartObj></w:sdtPr><w:sdtContent>"#;

33pub(super) fn is_toc(instr: &str) -> bool {
34    instr.split_whitespace().next().is_some_and(|k| k.eq_ignore_ascii_case("TOC"))
35}

37/// The body TOC the writer wraps in Word's Table of Contents control (`toc_span`): not one whose
38/// heading is inside a content control already (read from a file).
39pub(crate) fn written_toc_span(bl: &Blocks) -> Option<(usize, usize)> {
40    toc_span(bl).filter(|(start, _)| !crate::docx::write::sdt::plan(bl).covers(*start))
41}

43/// The engine's TOC: a paragraph holding an empty `TOC` field, then its TOC-styled entries.
44/// Returns (heading, last entry) so the field can be written spanning the entries, as Word does.
45/// The heading may end a section: entries in columns follow a continuous break (#3889).
46fn toc_span(bl: &Blocks) -> Option<(usize, usize)> {
47    let start = bl.iter().position(|b| {
48        b.as_para()
49            .is_some_and(|p| p.objects.iter().any(|o| matches!(o, InlineObject::Field { instr, result, .. } if is_toc(instr) && result.is_empty())))
50    })?;
51    let entries = bl[start + 1..]
52        .iter()
53        .take_while(|b| {
54            b.as_para().is_some_and(|p| p.section.is_none() && p.props.style.as_deref().is_some_and(|st| st.starts_with("TOC") && st != "TOCHeading"))
55        })
56        .count();
57    (entries > 0).then_some((start, start + entries))
58}
