5/// Built-in names Word keeps in lower case in styles.xml.
6fn file_style_name(name: &str) -> String {
7    if let Some((f, _)) = crate::docx::read::LOWER_NAMES.iter().find(|(_, ui)| ui.eq_ignore_ascii_case(name)) {
8        return (*f).to_string();
9    }
10    for (pre, f) in [("Heading ", "heading "), ("TOC ", "toc "), ("Index ", "index ")] {
11        if let Some(d) = name.strip_prefix(pre)
12            && d.len() == 1
13            && d.chars().all(|c| c.is_ascii_digit())
14        {
15            return format!("{f}{d}");
16        }
17    }
18    name.to_string()
19}

21pub(super) fn styles_xml(wr: &mut Writer<'_>) -> Vec<u8> {
22    let doc = wr.doc;
23    let s = &doc.styles;
24    let mut w = W::new();
25    let ns = xml::body_ns();
26    let ns_refs: Vec<(&str, &str)> = ns.iter().map(|(k, v)| (k.as_str(), v.as_str())).collect();
27    w.open("w:styles", &ns_refs);
28    w.open("w:docDefaults", &[]);
29    w.open("w:rPrDefault", &[]);
30    w.open("w:rPr", &[]);
31    if s.default_chr.size.is_none() {
32        // The reader reads an omitted `w:sz` as 10 pt, but the resolver falls back to
33        // DEFAULT_FONT_SIZE, so writing nothing would change the size on reopening.
34        let mut chr = s.default_chr.clone();
35        chr.size = Some(crate::doc::props::DEFAULT_FONT_SIZE);
36        props::rpr_inner(&mut w, &chr);
37    } else {
38        props::rpr_inner(&mut w, &s.default_chr);
39    }
40    w.close("w:rPr");
41    w.close("w:rPrDefault");
42    if s.default_para_present || s.default_para != crate::doc::props::ParaProps::default() {
43        w.open("w:pPrDefault", &[]);
44        w.open("w:pPr", &[]);
45        props::ppr_inner(&mut w, &s.default_para, false, None);
46        w.close("w:pPr");
47        w.close("w:pPrDefault");
48    }
49    w.close("w:docDefaults");
50    if let Some(latent) = s.latent_styles.as_deref() {
51        w.raw(latent);
52    }
53    // Kinds whose default style the file named (`w:default`); the English ids stand in otherwise.
54    let flagged: Vec<StyleKind> = s.styles.iter().filter(|st| st.default).map(|st| st.kind).collect();
55    for st in &s.styles {
56        style_xml(wr, &mut w, st, &flagged);
57    }
58    w.close("w:styles");
59    w.into_bytes()
60}

62/// One `w:style` element. A tracked change of its paragraph/run properties is nested in the
63/// corresponding property element as `w:pPrChange` / `w:rPrChange`.
64fn style_xml(wr: &mut Writer<'_>, w: &mut W, st: &Style, flagged: &[StyleKind]) {
65    let ty = match st.kind {
66        StyleKind::Paragraph => "paragraph",
67        StyleKind::Character => "character",
68        StyleKind::Table => "table",
69        StyleKind::Numbering => "numbering",
70    };
71    let english_default = matches!(
72        (st.kind, st.id.as_str()),
73        (StyleKind::Paragraph, "Normal")
74            | (StyleKind::Character, "DefaultParagraphFont")
75            | (StyleKind::Table, "TableNormal")
76            | (StyleKind::Numbering, "NoList")
77    );
78    let is_default = st.default || (english_default && !flagged.contains(&st.kind));
79    let mut a = vec![("w:type", ty), ("w:styleId", st.id.as_str())];
80    if is_default {
81        a.push(("w:default", "1"));
82    }
83    if !st.builtin {
84        a.push(("w:customStyle", "1"));
85    }
86    w.open("w:style", &a);
87    w.val("w:name", &file_style_name(if st.name.is_empty() { &st.id } else { &st.name }));
88    if let Some(b) = &st.based_on {
89        w.val("w:basedOn", b);
90    }
91    if let Some(b) = &st.next {
92        w.val("w:next", b);
93    }
94    if let Some(b) = &st.linked {
95        w.val("w:link", b);
96    }
97    if let Some(p) = st.priority {
98        w.val("w:uiPriority", &p.to_string());
99    }
100    if st.hidden {
101        w.empty("w:semiHidden", &[]);
102        w.empty("w:unhideWhenUsed", &[]);
103    }
104    if st.quick {
105        w.empty("w:qFormat", &[]);
106    }
107    if st.locked {
108        w.empty("w:locked", &[]);
109    }
110    style_property_xml(wr, w, st);
111    if let Some(t) = &st.table {
112        if t.borders.is_some() || t.cell_margins.is_some() || t.band_size.is_some() || t.col_band_size.is_some() {
113            w.open("w:tblPr", &[]);
114            if let Some(n) = t.band_size {
115                w.val("w:tblStyleRowBandSize", &n.min(1000).to_string());
116            }
117            if let Some(n) = t.col_band_size {
118                w.val("w:tblStyleColBandSize", &n.min(1000).to_string());
119            }
120            if let Some(b) = &t.borders {
121                props::borders(w, "w:tblBorders", b, Some("w:insideH"), &[]);
122            }
123            if let Some(m) = &t.cell_margins {
124                props::margins(w, "w:tblCellMar", m);
125            }
126            w.close("w:tblPr");
127        }
128        // The conditional formatting parts (ECMA-376 §17.7.6): paragraph and run properties,
129        // then cell borders, shading and vertical alignment. The whole table's text formatting
130        // is the style's own (above).
131        for part in TablePart::ALL {
132            let f = t.part(part);
133            let ppr = ParaProps { align: f.align, ..ParaProps::default() };
134            let has_rpr = props::has_rpr(&f.chr);
135            // The whole table's borders are its table borders (`w:tblPr` above).
136            let borders = f.borders.filter(|_| part != TablePart::WholeTable);
137            let has_tc = f.fill.is_some() || borders.is_some() || f.v_align.is_some();
138            if !has_rpr && !has_tc && f.align.is_none() {
139                continue;
140            }
141            w.open("w:tblStylePr", &[("w:type", part.ooxml())]);
142            if f.align.is_some() {
143                w.open("w:pPr", &[]);
144                props::ppr_inner(w, &ppr, false, None);
145                w.close("w:pPr");
146            }
147            if has_rpr {
148                w.open("w:rPr", &[]);
149                props::rpr_inner(w, &f.chr);
150                w.close("w:rPr");
151            }
152            if has_tc {
153                w.open("w:tcPr", &[]);
154                if let Some(bs) = borders {
155                    props::borders(w, "w:tcBorders", &bs, Some("w:insideH"), &[]);
156                }
157                if let Some(c) = f.fill {
158                    w.empty("w:shd", &[("w:val", "clear"), ("w:color", "auto"), ("w:fill", &c.hex())]);
159                }
160                if let Some(v) = f.v_align {
161                    w.val(
162                        "w:vAlign",
163                        match v {
164                            VAlign::Top => "top",
165                            VAlign::Center => "center",
166                            VAlign::Bottom => "bottom",
167                        },
168                    );
169                }
170                w.close("w:tcPr");
171            }
172            w.close("w:tblStylePr");
173        }
174    }
175    w.close("w:style");
176}

178fn style_property_xml(wr: &mut Writer<'_>, w: &mut W, st: &Style) {
179    let para_change = st.para_change.as_deref().filter(|ch| ch.old.formatting() != st.para.formatting());
180    if (!st.para.is_empty() || para_change.is_some()) && st.kind != StyleKind::Character {
181        w.open("w:pPr", &[]);
182        props::ppr_inner(w, &st.para, false, None);
183        if let Some(ch) = para_change {
184            let a = wr.change_attrs(ch.rev);
185            open_change(w, "w:pPrChange", &a);
186            w.open("w:pPr", &[]);
187            props::ppr_inner(w, &ch.old, false, None);
188            w.close("w:pPr");
189            w.close("w:pPrChange");
190        }
191        w.close("w:pPr");
192    }
193    let run_change = st.chr_change.as_deref().filter(|ch| ch.old.formatting() != st.chr.formatting());
194    if (props::has_rpr(&st.chr) || run_change.is_some()) && st.kind != StyleKind::Numbering {
195        w.open("w:rPr", &[]);
196        props::rpr_inner(w, &st.chr);
197        if let Some(ch) = run_change {
198            let a = wr.change_attrs(ch.rev);
199            open_change(w, "w:rPrChange", &a);
200            w.open("w:rPr", &[]);
201            props::rpr_inner(w, &ch.old);
202            w.close("w:rPr");
203            w.close("w:rPrChange");
204        }
205        w.close("w:rPr");
206    }
207}

209pub(super) fn numbering_xml(doc: &Document) -> Vec<u8> {
210    let mut w = W::new();
211    let ns = xml::body_ns();
212    let ns_refs: Vec<(&str, &str)> = ns.iter().map(|(k, v)| (k.as_str(), v.as_str())).collect();
213    w.open("w:numbering", &ns_refs);
214    let mut nsids: std::collections::HashSet<&str> = std::collections::HashSet::new();
215    for a in &doc.numbering.abstracts {
216        w.open("w:abstractNum", &[("w:abstractNumId", &a.id.to_string())]);
217        // CT_AbstractNum order: nsid, multiLevelType, tmpl, name… A copied definition repeats
218        // its source's nsid; only the first keeps it, so Word doesn't take them for one list.
219        if let Some(n) = a.nsid.as_deref().filter(|n| nsids.insert(n)) {
220            w.val("w:nsid", n);
221        }
222        let derived = if a.levels.len() > 1 { "multilevel" } else { "singleLevel" };
223        w.val("w:multiLevelType", a.multi_level_type.as_deref().unwrap_or(derived));
224        if let Some(t) = &a.tmpl {
225            w.val("w:tmpl", t);
226        }
227        if let Some(n) = &a.name {
228            w.val("w:name", n);
229        }
230        for (i, l) in a.levels.iter().take(9).enumerate() {
231            write_level(&mut w, i, l);
232        }
233        w.close("w:abstractNum");
234    }
235    for n in &doc.numbering.nums {
236        w.open("w:num", &[("w:numId", &n.id.to_string())]);
237        w.val("w:abstractNumId", &n.abstract_id.to_string());
238        for lvl in 0..9u8 {
239            let start = n.start_overrides.iter().find(|(l, _)| (*l).min(8) == lvl).map(|(_, s)| *s);
240            let level = n.level_overrides.iter().find(|(l, _)| (*l).min(8) == lvl).map(|(_, l)| l);
241            if start.is_none() && level.is_none() {
242                continue;
243            }
244            w.open("w:lvlOverride", &[("w:ilvl", &lvl.to_string())]);
245            if let Some(start) = start {
246                w.val("w:startOverride", &start.to_string());
247            }
248            if let Some(level) = level {
249                write_level(&mut w, lvl as usize, level);
250            }
251            w.close("w:lvlOverride");
252        }
253        w.close("w:num");
254    }
255    w.close("w:numbering");
256    w.into_bytes()
257}

259/// One `w:lvl` (list level `i`).
260fn write_level(w: &mut W, i: usize, l: &Level) {
261    w.open("w:lvl", &[("w:ilvl", &i.to_string())]);
262    w.val("w:start", &l.start.to_string());
263    w.val("w:numFmt", l.format.ooxml());
264    if !l.restart {
265        w.val("w:lvlRestart", "0");
266    } else if let Some(k) = l.restart_after.filter(|k| (1..=9).contains(k)) {
267        w.val("w:lvlRestart", &k.to_string());
268    }
269    if let Some(s) = &l.style {
270        w.val("w:pStyle", s);
271    }
272    if l.legal {
273        w.empty("w:isLgl", &[]);
274    }
275    match l.suffix {
276        LevelSuffix::Tab => {}
277        LevelSuffix::Space => w.val("w:suff", "space"),
278        LevelSuffix::Nothing => w.val("w:suff", "nothing"),
279    }
280    w.val("w:lvlText", &l.text);
281    w.val("w:lvlJc", props::align_val(l.align));
282    w.open("w:pPr", &[]);
283    if let Some(t) = l.tab.filter(|t| t.is_finite()) {
284        w.open("w:tabs", &[]);
285        w.empty("w:tab", &[("w:val", "num"), ("w:pos", &crate::docx::units::twips(t.clamp(-1584.0, 1584.0)))]);
286        w.close("w:tabs");
287    }
288    let ind = crate::docx::units::twips(l.indent);
289    if l.hanging >= 0.0 {
290        w.empty("w:ind", &[("w:left", &ind), ("w:hanging", &crate::docx::units::twips(l.hanging))]);
291    } else {
292        w.empty("w:ind", &[("w:left", &ind), ("w:firstLine", &crate::docx::units::twips(-l.hanging))]);
293    }
294    w.close("w:pPr");
295    props::rpr(w, &l.chr);
296    w.close("w:lvl");
