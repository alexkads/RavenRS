1//! `word/settings.xml` merged into the file's original settings part.
2//!
3//! The model owns only some settings (`super::settings_xml`). Regenerating the part from the
4//! model alone drops everything else Word stores there: compatibility options, rsids, document
5//! variables, proofing state, the colour scheme mapping, "remove personal information on save"…
6//! So when a file brought settings the model doesn't hold ([`has_unowned`]; the reader keeps its
7//! bytes under `SETTINGS_ORIG`), the writer merges instead ([`merge`]): the original root start
8//! tag (namespace declarations, `mc:Ignorable`) and every child the model doesn't own are kept
9//! byte for byte; owned children come from the generated part (or go, when the model turned
10//! them off); owned children the original lacks are inserted at their CT_Settings position; in
11//! `w:compat` only the `compatibilityMode` setting is replaced. Kept children that refer to a
12//! relationship (`r:id`, e.g. a mail merge data source) are dropped: their relationships aren't
13//! carried. Anything unexpected returns `None`, and the generated part is written as before.
14

145fn rank(name: &str) -> usize {
146    ORDER.iter().position(|n| *n == name).unwrap_or(ORDER.len())
147}

149/// A child element of the root: its qualified name and byte span.
150struct Child {
151    name: String,
152    start: usize,
153    end: usize,
154}
155
156/// Byte index of the `>` that ends the tag at the start of `s`, skipping quoted attribute values.
157fn tag_end(s: &str) -> Option<usize> {
158    let mut quote = None;
159    for (i, c) in s.char_indices() {
160        match (quote, c) {
161            (Some(q), c) if c == q => quote = None,
162            (Some(_), _) => {}
163            (None, '"' | '\'') => quote = Some(c),
164            (None, '>') => return Some(i),
165            _ => {}
166        }
167    }
168    None
169}

171/// The root start tag's span end and the root's direct children, or `None` for anything the
172/// scanner doesn't expect (CDATA, DOCTYPE, a different root).
173fn children(xml: &str, root: &str) -> Option<(usize, usize, Vec<Child>)> {
174    let open = xml.find(&format!("<{root}"))?;
175    let open_end = open + tag_end(xml.get(open..)?)? + 1;
176    if xml.get(open_end - 2..open_end)? == "/>" {
177        return None;
178    }
179    let close = xml.rfind(&format!("</{root}>"))?;
180    let mut out = Vec::new();
181    let mut depth = 0usize;
182    let mut i = open_end;
183    let mut cur: Option<(String, usize)> = None;
184    while i < close {
185        let rest = xml.get(i..close)?;
186        let Some(lt) = rest.find('<') else { break };
187        let at = i + lt;
188        let tail = xml.get(at..close)?;
189        if tail.starts_with("<!--") {
190            i = at + tail.find("-->")? + 3;
191            continue;
192        }
193        if tail.starts_with("<![CDATA[") || tail.starts_with("<!") {
194            return None;
195        }
196        if tail.starts_with("<?") {
197            i = at + tail.find("?>")? + 2;
198            continue;
199        }
200        let gt = at + tag_end(tail)? + 1;
201        let tag = xml.get(at..gt)?;
202        if tag.starts_with("</") {
203            depth = depth.checked_sub(1)?;
204            if depth == 0 {
205                let (name, s) = cur.take()?;
206                out.push(Child { name, start: s, end: gt });
207            }
208        } else {
209            let name: String = tag.get(1..)?.chars().take_while(|c| !c.is_whitespace() && *c != '>' && *c != '/').collect();
210            let selfclose = tag.ends_with("/>");
211            if depth == 0 {
212                if selfclose {
213                    out.push(Child { name, start: at, end: gt });
214                } else {
215                    cur = Some((name, at));
216                    depth = 1;
217                }
218            } else if !selfclose {
219                depth += 1;
220            }
221        }
222        i = gt;
223    }
224    if depth != 0 {
225        return None;
226    }
227    Some((open_end, close, out))
228}

230/// The original `w:compat` with its `compatibilityMode` taken from the generated one.
231fn merge_compat(orig: &str, generated: &str) -> Option<String> {
232    let mode = |s: &str| -> Option<(usize, usize)> {
233        let at = s.find("w:name=\"compatibilityMode\"")?;
234        let tag_start = s.get(..at)?.rfind('<')?;
235        let tag_end = tag_start + s.get(tag_start..)?.find('>')? + 1;
236        Some((tag_start, tag_end))
237    };
238    match (mode(orig), mode(generated)) {
239        (Some((a, b)), Some((c, d))) => Some(format!("{}{}{}", orig.get(..a)?, generated.get(c..d)?, orig.get(b..)?)),
240        (None, Some((c, d))) => {
241            // No mode in the original: add the generated one before `</w:compat>`, or expand a self-closing `<w:compat/>`.
242            if let Some(close) = orig.rfind("</w:compat>") {
243                Some(format!("{}{}{}", orig.get(..close)?, generated.get(c..d)?, orig.get(close..)?))
244            } else if let Some(slash) = orig.rfind("/>") {
245                let open = orig.get(..slash)?.trim_end();
246                Some(format!("{open}>{}</w:compat>", generated.get(c..d)?))
247            } else {
248                None
249            }
250        }
251        _ => Some(orig.to_string()),
252    }
253}

255/// Does `original` (a settings part) hold anything the model doesn't write? Then the reader keeps
256/// it for [`merge`]. False for parts WordCraft wrote itself, and for anything it can't scan.
257pub fn has_unowned(original: &[u8]) -> bool {
258    let Some(xml) = std::str::from_utf8(original).ok() else { return false };
259    let Some((_, _, kids)) = children(xml, "w:settings") else { return false };
260    kids.iter().any(|k| match k.name.as_str() {
261        "w:zoom" => false,
262        // Anything in `w:compat` beyond its own tag and the compatibility mode.
263        "w:compat" => xml.get(k.start..k.end).is_some_and(|c| {
264            let mode = usize::from(c.contains("w:name=\"compatibilityMode\""));
265            c.matches("<w:").count().saturating_sub(1 + mode) > 0
266        }),
267        n => !OWNED.contains(&n),
268    })
269}

271/// The merged settings part, or `None` to fall back to `generated`.
272pub fn merge(original: &[u8], generated: &[u8]) -> Option<Vec<u8>> {
273    let orig = std::str::from_utf8(original).ok()?;
274    let gen_s = std::str::from_utf8(generated).ok()?;
275    let (o_open, o_close, o_kids) = children(orig, "w:settings")?;
276    let (_, _, g_kids) = children(gen_s, "w:settings")?;
277    // Prefixes the generated children use must be declared by the original root.
278    let root_tag = orig.get(..o_open)?;
279    for p in ["w", "m", "r"] {
280        if g_kids.iter().any(|k| k.name.starts_with(&format!("{p}:"))) && !root_tag.contains(&format!("xmlns:{p}=")) {
281            return None;
282        }
283    }
284    let g_text = |k: &Child| gen_s.get(k.start..k.end).map(str::to_string);
285    let mut used = vec![false; g_kids.len()];
286    let mark = |name: &str, used: &mut Vec<bool>| -> Option<usize> {
287        let gi = g_kids.iter().enumerate().position(|(i, g)| g.name == name && !used.get(i).copied().unwrap_or(true))?;
288        if let Some(u) = used.get_mut(gi) {
289            *u = true;
290        }
291        Some(gi)
292    };
293    let mut out: Vec<(String, String)> = Vec::new();
294    for k in &o_kids {
295        let text = orig.get(k.start..k.end)?;
296        if KEEP_ORIGINAL.contains(&k.name.as_str()) {
297            // The model reads these only partly (math settings) or writes them only in one case
298            // (the background shape flag, with a page colour): the original wins.
299            mark(&k.name, &mut used);
300            out.push((k.name.clone(), text.to_string()));
301        } else if OWNED.contains(&k.name.as_str()) {
302            if let Some(gi) = mark(&k.name, &mut used) {
303                out.push((k.name.clone(), g_text(g_kids.get(gi)?)?));
304            }
305        } else if k.name == "w:compat" {
306            let merged = match mark("w:compat", &mut used) {
307                Some(gi) => merge_compat(text, &g_text(g_kids.get(gi)?)?)?,
308                None => text.to_string(),
309            };
310            out.push((k.name.clone(), merged));
311        } else if !text.contains("r:id=") {
312            out.push((k.name.clone(), text.to_string()));
313        }
314    }
315    for (gi, g) in g_kids.iter().enumerate() {
316        let owned = OWNED.contains(&g.name.as_str()) || g.name == "w:compat";
317        if used.get(gi).copied().unwrap_or(true) || !owned {
318            continue;
319        }
320        let r = rank(&g.name);
321        let at = out.iter().position(|(n, _)| rank(n) > r).unwrap_or(out.len());
322        out.insert(at, (g.name.clone(), g_text(g)?));
323    }
324    let mut s = String::with_capacity(orig.len() + 256);
325    s.push_str(root_tag);
326    for (_, t) in &out {
327        s.push_str(t);
328    }
329    s.push_str(orig.get(o_close..)?);
330    Some(s.into_bytes())
331}

339    #[test]
340    fn odd_or_hostile_originals_fall_back_or_merge_without_panicking() {
341        let odd = [
342            "",
343            "<w:settings",
344            "<w:settings/>",
345            "<x:other/>",
346            r#"<w:settings xmlns:w="w"><w:a><w:b></w:a></w:settings>"#,
347            r#"<w:settings xmlns:w="w"><![CDATA[x]]></w:settings>"#,
348            r#"<w:settings xmlns:w="w"></w:x></w:settings>"#,
349            "<w:settings xmlns:w=\"w\">\u{1F600}<w:rsids/></w:settings>",
350            r#"<w:settings xmlns:w="w"><!-- c --><?pi x?><w:rsids/></w:settings>"#,
351        ];
352        for o in odd {
353            let _ = merge(o.as_bytes(), GEN.as_bytes());
354            let _ = has_unowned(o.as_bytes());
355        }
356        assert!(merge(&[0xff, 0xfe], GEN.as_bytes()).is_none());
357        // Missing declarations for a prefix the generated part uses: fall back.
358        assert!(merge(br#"<w:settings xmlns:x="x"><w:rsids/></w:settings>"#, GEN.as_bytes()).is_none());
359    }

361    #[test]
362    fn a_gt_inside_an_attribute_value_does_not_hide_the_kept_settings() {
363        for v in [r#"w:val="x>y""#, "w:val='x>y'"] {
364            let orig = format!(
365                r#"<w:settings xmlns:w="w" xmlns:m="m" xmlns:r="r"><w:docVar w:name="a" {v}/><w:docVar w:name="b" {v}></w:docVar></w:settings>"#
366            );
367            assert!(has_unowned(orig.as_bytes()), "{v}");
368            let out = String::from_utf8(merge(orig.as_bytes(), GEN.as_bytes()).unwrap()).unwrap();
369            assert!(out.contains(r#"w:name="a""#) && out.contains(r#"w:name="b""#), "{v}");
370        }
371    }
