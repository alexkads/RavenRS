1//! Content controls → `w:sdt` (ECMA-376 §17.5.2; check boxes and repeating sections as the
2//! [MS-DOCX] `w14`/`w15` extensions).
3//!
4//! Controls are markers in the model ([`crate::doc::control`]); the file wants them as
5//! elements around runs (inline) or around whole paragraphs and tables (block level). [`plan`]
6//! decides, per block list, which controls are written where so that the XML always nests.
7
8use crate::doc::Blocks;
9use crate::doc::control::{ContentControl, ControlEvent, ControlKind, control_events, pair_events};
10use crate::doc::para::OBJ;
11
12use crate::docx::xml::W;

14/// Where a block list's controls are written.
15#[derive(Default)]
16pub struct Plan<'a> {
17    /// Controls opening before block `i` (outermost first).
18    pub opens: Vec<Vec<&'a ContentControl>>,
19    /// Controls closing after block `i`.
20    pub closes: Vec<u32>,
21    /// The paragraph markers written inline, by block: the byte offsets in block `i`, in order.
22    ///
23    /// Grouped by block rather than one set of (block, offset) pairs, so that asking for a
24    /// paragraph's markers is a lookup and not a search of every marker in the document (#13).
25    pub inline: Vec<Vec<usize>>,
26}
27
28impl Plan<'_> {
29    /// Whether a block-level control is around block `i`.
30    pub fn covers(&self, i: usize) -> bool {
31        let mut open = 0i64;
32        for k in 0..=i {
33            open += self.opens.get(k).map(|v| v.len() as i64).unwrap_or(0);
34            if k < i {
35                open -= i64::from(self.closes.get(k).copied().unwrap_or(0));
36            }
37        }
38        open > 0
39    }
40    /// The byte offsets of paragraph `i`'s markers that are written inline.
41    pub fn inline_in(&self, i: usize) -> Vec<usize> {
42        self.inline.get(i).cloned().unwrap_or_default()
43    }
44}
45
46#[derive(Clone, Copy, PartialEq)]
47enum Mode {
48    Inline,
49    Block,
50}
51
52/// Plan the controls of a block list (not looking into cells: they have their own lists).
53/// Markers without a partner are not written.
54pub fn plan(bl: &Blocks) -> Plan<'_> {
55    let n = bl.len();
56    let mut out = Plan { opens: vec![Vec::new(); n], closes: vec![0; n], inline: vec![Vec::new(); n] };
57    let events = control_events(bl);
58    if events.is_empty() {
59        return out;
60    }
61    let (pairs, _) = pair_events(&events);
62    // Ranges in the order they open (parents before children).
63    let mut ranges: Vec<(usize, usize)> = pairs.iter().map(|(o, c, _)| (*o, *c)).collect();
64    ranges.sort_unstable();
65    // Parent of each range: the innermost one opened before it and closed after it.
66    let mut parent: Vec<Option<usize>> = vec![None; ranges.len()];
67    let mut stack: Vec<usize> = Vec::new();
68    for (k, (o, _)) in ranges.iter().enumerate() {
69        while let Some(&top) = stack.last() {
70            if ranges.get(top).is_some_and(|(_, tc)| tc < o) {
71                stack.pop();
72            } else {
73                break;
74            }
75        }
76        if let Some(slot) = parent.get_mut(k) {
77            *slot = stack.last().copied();
78        }
79        stack.push(k);
80    }
81    let para = |b: usize| bl.get(b).and_then(|x| x.as_para());
82    // Only control starts before `off` in the paragraph / only control ends after it.
83    let at_start = |b: usize, off: usize| {
84        para(b).is_some_and(|p| {
85            p.text.get(..off).is_some_and(|t| t.chars().all(|c| c == OBJ))
86                && (0..off).step_by(OBJ.len_utf8()).all(|o| matches!(p.object_at(o), Some(crate::doc::InlineObject::ControlStart { .. })))
87        })
88    };
89    // Bookmark, comment and permission markers after the end may stay there: a body-level
90    // bookmark after a block control (Word's `_GoBack`) is read onto the end of its paragraph,
91    // and the control still covers the whole paragraph.
92    let at_end = |b: usize, off: usize| {
93        use crate::doc::InlineObject as O;
94        para(b).is_some_and(|p| {
95            let from = off + OBJ.len_utf8();
96            p.text.get(from..).is_some_and(|t| t.chars().all(|c| c == OBJ))
97                && (from..p.len()).step_by(OBJ.len_utf8()).all(|o| {
98                    matches!(
99                        p.object_at(o),
100                        Some(
101                            O::ControlEnd
102                                | O::BookmarkStart { .. }
103                                | O::BookmarkEnd { .. }
104                                | O::CommentStart { .. }
105                                | O::CommentEnd { .. }
106                                | O::PermStart { .. }
107                                | O::PermEnd { .. }
108                        )
109                    )
110                })
111        })
112    };
113    let mut mode = vec![Mode::Inline; ranges.len()];
114    let mut span: Vec<(usize, usize)> = vec![(0, 0); ranges.len()];
115    for (k, (o, c)) in ranges.iter().enumerate() {
116        let (Some(ControlEvent::Open { block: sb, off: so, control }), Some(e)) = (events.get(*o), events.get(*c)) else { continue };
117        let (eb, eo) = (e.block(), e.off());
118        let m = match (so, eo) {
119            (Some(so), Some(eo)) if *sb == eb => {
120                let parent_inline = parent.get(k).copied().flatten().and_then(|p| mode.get(p)).is_some_and(|m| *m == Mode::Inline);
121                if !parent_inline && control.block && at_start(*sb, *so) && at_end(eb, eo) { Mode::Block } else { Mode::Inline }
122            }
123            _ => Mode::Block,
124        };
125        if let Some(slot) = mode.get_mut(k) {
126            *slot = m;
127        }
128        if let Some(slot) = span.get_mut(k) {
129            *slot = (*sb, eb);
130        }
131    }
132    // Block-level controls nest by block: one that opens in the block another (not its
133    // ancestor) closes after starts with the next block instead.
134    //
135    // Such a clash is with a control that closes before this one opens and ends in the block
136    // this one starts in. The block-level controls are grouped by the block they end in, each
137    // group keeping the two earliest closes in it: the earliest close that is not this
138    // control's own is the one the test is about (#13).
139    let mut first_close: Vec<usize> = vec![usize::MAX; n + 1];
140    let mut second_close: Vec<usize> = vec![usize::MAX; n + 1];
141    let mut first_owner: Vec<usize> = vec![usize::MAX; n + 1];
142    for (q, (_, qc)) in ranges.iter().enumerate() {
143        if mode.get(q) != Some(&Mode::Block) {
144            continue;
145        }
146        let Some(&(_, qe)) = span.get(q) else { continue };
147        let e = qe.min(n);
148        if *qc < first_close[e] {
149            second_close[e] = first_close[e];
150            first_close[e] = *qc;
151            first_owner[e] = q;
152        } else if *qc < second_close[e] {
153            second_close[e] = *qc;
154        }
155    }
156    for k in 0..ranges.len() {
157        if mode.get(k) != Some(&Mode::Block) {
158            continue;
159        }
160        let Some(&(o, _)) = ranges.get(k) else { continue };
161        let Some(&(sb, eb)) = span.get(k) else { continue };
162        let e = sb.min(n);
163        let close = if first_owner[e] == k { second_close[e] } else { first_close[e] };
164        if close < o {
165            let to = (sb + 1).min(eb);
166            if let Some(s) = span.get_mut(k) {
167                s.0 = to;
168            }
169        }
170    }
171    for (k, (o, c)) in ranges.iter().enumerate() {
172        let Some(ControlEvent::Open { control, .. }) = events.get(*o) else { continue };
173        match mode.get(k) {
174            Some(Mode::Block) => {
175                let Some(&(sb, eb)) = span.get(k) else { continue };
176                if let Some(v) = out.opens.get_mut(sb) {
177                    v.push(control);
178                }
179                if let Some(x) = out.closes.get_mut(eb) {
180                    *x += 1;
181                }
182            }
183            _ => {
184                for e in [events.get(*o), events.get(*c)].into_iter().flatten() {
185                    if let Some(off) = e.off()
186                        && let Some(v) = out.inline.get_mut(e.block())
187                    {
188                        v.push(off);
189                    }
190                }
191            }
192        }
193    }
194    for v in out.inline.iter_mut() {
195        v.sort_unstable();
196        v.dedup();
197    }
198    out
199}

201fn on(w: &mut W, name: &str, v: bool) {
202    if v {
203        w.empty(name, &[]);
204    }
205}

207/// `w:sdtPr` (and `w:sdtEndPr`), then open `w:sdtContent`.
208pub fn open(w: &mut W, c: &ContentControl) {
209    w.open("w:sdt", &[]);
210    w.open("w:sdtPr", &[]);
211    w.raw(&c.rpr_xml);
212    if !c.title.is_empty() {
213        w.val("w:alias", &c.title);
214    }
215    if !c.tag.is_empty() {
216        w.val("w:tag", &c.tag);
217    }
218    if let Some(id) = c.id {
219        w.val("w:id", &id.to_string());
220    }
221    if let Some(l) = c.lock.ooxml() {
222        w.val("w:lock", l);
223    }
224    if !c.placeholder.is_empty() {
225        w.open("w:placeholder", &[]);
226        w.val("w:docPart", &c.placeholder);
227        w.close("w:placeholder");
228    }
229    on(w, "w:temporary", c.temporary);
230    on(w, "w:showingPlcHdr", c.showing_placeholder);
231    // Kept elements of the schema's sequence before the type, then the type, then the rest.
232    let early = |x: &&String| {
233        ["<w:dataBinding", "<w:label", "<w:tabIndex"].iter().any(|n| x.strip_prefix(n).is_some_and(|rest| rest.starts_with([' ', '/', '>'])))
234    };
235    for x in c.extra.iter().filter(early) {
236        w.raw(x);
237    }
238    kind(w, &c.kind);
239    for x in c.extra.iter().filter(|x| !early(x)) {
240        w.raw(x);
241    }
242    w.close("w:sdtPr");
243    w.raw(&c.end_pr_xml);
244    w.open("w:sdtContent", &[]);
245}

247/// Close `w:sdtContent` and `w:sdt`.
248pub fn close(w: &mut W) {
249    w.raw("</w:sdtContent></w:sdt>");
250}

252fn kind(w: &mut W, k: &ControlKind) {
253    match k {
254        ControlKind::RichText => {}
255        ControlKind::Text { multi_line } => {
256            if *multi_line {
257                w.empty("w:text", &[("w:multiLine", "1")]);
258            } else {
259                w.empty("w:text", &[]);
260            }
261        }
262        ControlKind::CheckBox { checked, checked_char, checked_font, unchecked_char, unchecked_font } => {
263            w.open("w14:checkbox", &[]);
264            w.empty("w14:checked", &[("w14:val", if *checked { "1" } else { "0" })]);
265            for (tag, ch, font) in [("w14:checkedState", checked_char, checked_font), ("w14:uncheckedState", unchecked_char, unchecked_font)] {
266                let hex = format!("{:04X}", u32::from(*ch));
267                if font.is_empty() {
268                    w.empty(tag, &[("w14:val", &hex)]);
269                } else {
270                    w.empty(tag, &[("w14:val", &hex), ("w14:font", font)]);
271                }
272            }
273            w.close("w14:checkbox");
274        }
275        ControlKind::ComboBox { items, last_value } | ControlKind::DropDown { items, last_value } => {
276            let tag = if matches!(k, ControlKind::ComboBox { .. }) { "w:comboBox" } else { "w:dropDownList" };
277            if last_value.is_empty() {
278                w.open(tag, &[]);
279            } else {
280                w.open(tag, &[("w:lastValue", last_value)]);
281            }
282            for i in items {
283                w.empty("w:listItem", &[("w:displayText", &i.display), ("w:value", i.value())]);
284            }
285            w.close(tag);
286        }
287        ControlKind::Date { full_date, format, lid, calendar, store_as } => {
288            if full_date.is_empty() {
289                w.open("w:date", &[]);
290            } else {
291                w.open("w:date", &[("w:fullDate", full_date)]);
292            }
293            for (tag, v) in [("w:dateFormat", format), ("w:lid", lid), ("w:storeMappedDataAs", store_as), ("w:calendar", calendar)] {
294                if !v.is_empty() {
295                    w.val(tag, v);
296                }
297            }
298            w.close("w:date");
299        }
300        ControlKind::Picture => w.empty("w:picture", &[]),
301        ControlKind::Gallery { list, gallery, category, unique } => {
302            let tag = if *list { "w:docPartList" } else { "w:docPartObj" };
303            w.open(tag, &[]);
304            if !gallery.is_empty() {
305                w.val("w:docPartGallery", gallery);
306            }
307            if !category.is_empty() {
308                w.val("w:docPartCategory", category);
309            }
310            on(w, "w:docPartUnique", *unique);
311            w.close(tag);
312        }
313        ControlKind::RepeatingSection { title, no_insert_delete } => {
314            w.open("w15:repeatingSection", &[]);
315            if !title.is_empty() {
316                w.val("w15:sectionTitle", title);
317            }
318            on(w, "w15:doNotAllowInsertDeleteSection", *no_insert_delete);
319            w.close("w15:repeatingSection");
320        }
321        ControlKind::RepeatingSectionItem => w.empty("w15:repeatingSectionItem", &[]),
322    }
