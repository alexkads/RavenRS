1//! Writing back charts, SmartArt diagrams, native ink and OLE objects read from a file (see `read::embed`).
2//!
3//! Each written object gets its kept markup with fresh relationship ids, pointing at copies of the
4//! parts it needs. The first object to use a part writes it under its original name (unless that
5//! name is one this writer uses for something else: then it's renamed). An object written again
6//! (copied and pasted in WordCraft) gets its own renamed copies of every part, so the copies stay
7//! independent in Word; past [`MAX_COPY_BYTES`] of such copies, later ones share the first's parts.
8

22/// Most bytes written for copies of parts (objects pasted more than once).
23const MAX_COPY_BYTES: u64 = 256 * 1024 * 1024;
24/// Most bytes of kept markup written for all objects (objects can share one kept markup).
25const MAX_MARKUP_BYTES: u64 = 128 * 1024 * 1024;
26/// What one part copy costs at least against [`MAX_COPY_BYTES`] (its relationships, entry…).
27const PART_COST: u64 = 4096;
28
29/// Where the stories' parts live: every story this writer makes is a part in `word/`.
30const STORY_PART: &str = "word/document.xml";
31
32/// The embedded-object parts of a package being written.
33pub(crate) struct EmbedWriter<'d> {
34    manifest: EmbeddedManifest,
35    passthrough: &'d BTreeMap<String, Arc<Vec<u8>>>,
36    /// Part names in use (lower case): this writer's own, and parts written so far.
37    taken: HashSet<String>,
38    /// Next suffix to try when renaming a part.
39    next: HashMap<String, u32>,
40    /// Original part name → name written, for parts written by an object's first copy.
41    first: HashMap<String, String>,
42    /// Budget left for copies.
43    copy_bytes: u64,
44    /// Budget left for kept markup.
45    markup_bytes: u64,
46    /// Parts to write: (name, bytes, content type, relationships).
47    pub parts: Vec<(String, Vec<u8>, String, PartRels)>,
48    /// Leave out the main document's kept glossary (it's written from the model instead).
49    pub drop_glossary: bool,
50    /// Kept parts written with these bytes instead of their own (see [`Self::scrub_glossary`]).
51    scrubbed: HashMap<String, Vec<u8>>,
52    /// A chart past the chart limits has been written (and the loss logged, once per save).
53    chart_cut: bool,
54}

57    pub fn new(doc: &'d crate::doc::Document) -> EmbedWriter<'d> {
58        let manifest = doc.passthrough.get(EMBEDDED_PARTS).map(|m| EmbeddedManifest::parse(m)).unwrap_or_default();
59        let mut taken: HashSet<String> = [
60            "word/document.xml",
61            "word/styles.xml",
62            "word/numbering.xml",
63            "word/settings.xml",
64            "word/theme/theme1.xml",
65            "word/footnotes.xml",
66            "word/endnotes.xml",
67            "word/comments.xml",
68            "word/commentsextended.xml",
69            "word/commentsids.xml",
70            "word/commentsextensible.xml",
71            "[content_types].xml",
72        ]
73        .iter()
74        .map(ToString::to_string)
75        .collect();
76        taken.insert(VBA_PROJECT_PART.to_ascii_lowercase());
77        // The bibliography writer uses these names when sources are present (#532).
78        if !doc.sources.is_empty() {
79            taken.extend(["customxml/item1.xml".into(), "customxml/itemprops1.xml".into()]);
80        }
81        // The macro project's parts keep their names (see `write_as`).
82        if let Some(m) = doc.passthrough.get(VBA_RELATED) {
83            for line in String::from_utf8_lossy(m).lines() {
84                if let Some(path) = line.split('\t').nth(1) {
85                    taken.insert(path.to_ascii_lowercase());
86                }
87            }
88        }
89        EmbedWriter {
90            manifest,
91            passthrough: &doc.passthrough,
92            taken,
93            next: HashMap::new(),
94            first: HashMap::new(),
95            copy_bytes: MAX_COPY_BYTES,
96            markup_bytes: MAX_MARKUP_BYTES,
97            parts: Vec::new(),
98            drop_glossary: false,
99            scrubbed: HashMap::new(),
100            chart_cut: false,
101        }
102    }

120    /// Names this writer may give its own parts later (headers, footers, package parts).
121    fn reserved(lower: &str) -> bool {
122        lower.ends_with(".rels")
123            || (lower.starts_with("docprops/") && !lower.starts_with("docprops/thumbnail."))
124            || lower.starts_with("_rels/")
125            || ((lower.starts_with("word/header") || lower.starts_with("word/footer")) && lower.ends_with(".xml"))
126    }

128    /// A free part name for (a copy of) `orig`: `orig` itself if free, else `stem_N.ext`.
129    fn alloc(&mut self, orig: &str) -> String {
130        let clean = valid_part_name(orig);
131        let orig = clean.as_str();
132        // Suffixes cannot escape reserved directories or a .rels extension. Such a kept
133        // part needs a different base name, otherwise every candidate would be reserved.
134        let orig = if Self::reserved(&orig.to_ascii_lowercase()) { "word/embeddings/part.xml" } else { orig };
135        let lower = orig.to_ascii_lowercase();
136        if !Self::reserved(&lower) && self.taken.insert(lower) {
137            return orig.to_string();
138        }
139        let file_at = orig.rfind('/').map_or(0, |i| i + 1);
140        let (stem, ext) = match orig.rfind('.').filter(|i| *i > file_at) {
141            Some(i) => (orig.get(..i).unwrap_or(orig), orig.get(i..).unwrap_or("")),
142            None => (orig, ""),
143        };
144        loop {
145            let n = self.next.entry(orig.to_string()).or_insert(1);
146            *n = n.saturating_add(1);
147            let name = format!("{stem}_{n}{ext}");
148            let lower = name.to_ascii_lowercase();
149            if !Self::reserved(&lower) && self.taken.insert(lower) {
150                return name;
151            }
152        }
153    }

155    /// A number `n` for which every `{stem}{n}{ext}` is a free part name, with those names taken.
156    pub(super) fn fresh_set(&mut self, stems: &[&str], ext: &str) -> Vec<String> {
157        let mut n: u32 = 1;
158        loop {
159            let names: Vec<String> = stems.iter().map(|s| format!("{s}{n}{ext}")).collect();
160            if names.iter().all(|name| {
161                let lower = name.to_ascii_lowercase();
162                !Self::reserved(&lower) && !self.taken.contains(&lower)
163            }) {
164                for name in &names {
165                    self.taken.insert(name.to_ascii_lowercase());
166                }
167                return names;
168            }
169            n = n.saturating_add(1);
170        }
171    }

173    /// A free part name `{stem}{n}{ext}` for a new part, from n = 1 up.
174    fn fresh(&mut self, stem: &str, ext: &str) -> String {
175        let mut n: u32 = 1;
176        loop {
177            let name = format!("{stem}{n}{ext}");
178            let lower = name.to_ascii_lowercase();
179            if !Self::reserved(&lower) && self.taken.insert(lower) {
180                return name;
181            }
182            n = n.saturating_add(1);
183        }
184    }

191    /// Reattach implicit custom XML roots, sharing their kept parts across both owners.
192    pub fn custom_xml_rels(&mut self, owner: &str, rels: &mut PartRels) {
193        let Some(roots) = self.passthrough.get(CUSTOM_XML_ROOTS).cloned() else { return };
194        let mut memo = std::mem::take(&mut self.first);
195        let mut written = Vec::new();
196        for line in String::from_utf8_lossy(&roots).lines().take(MAX_EMBEDDED_RELS * 2) {
197            let fields: Vec<&str> = line.split('\t').collect();
198            let [source, kind, target] = fields.as_slice() else { continue };
199            if *source != owner || !rel_is(kind, rt::CUSTOM_XML) {
200                continue;
201            }
202            if let Some(name) = self.copy_tree(target, &mut memo, &mut written) {
203                let source_part = if owner == "P" { "" } else { STORY_PART };
204                rels.add(kind, &relative(source_part, &name), false);
205            }
206        }
207        self.first = memo;
208    }

210    /// Reattach the parts kept by `keep_unmodelled` (`UNMODELLED_ROOTS`) to their owner: the
211    /// package (`P`) or the main document (`D`).
212    pub fn unmodelled_rels(&mut self, owner: &str, rels: &mut PartRels) {
213        let Some(roots) = self.passthrough.get(crate::docx::package::UNMODELLED_ROOTS).cloned() else { return };
214        let mut memo = std::mem::take(&mut self.first);
215        let mut written = Vec::new();
216        for line in String::from_utf8_lossy(&roots).lines().take(MAX_EMBEDDED_RELS * 2) {
217            let fields: Vec<&str> = line.split('\t').collect();
218            let [source, kind, target] = fields.as_slice() else { continue };
219            if *source != owner || (self.drop_glossary && owner == "D" && rel_is(kind, rt::GLOSSARY)) {
220                continue;
221            }
222            if let Some(name) = self.copy_tree(target, &mut memo, &mut written) {
223                let source_part = if owner == "P" { "" } else { STORY_PART };
224                rels.add(kind, &relative(source_part, &name), false);
225            }
226        }
227        self.first = memo;
228    }

230    /// Whether `unmodelled_rels("D", …)` can write the main document's kept glossary back.
231    pub fn kept_glossary(&self) -> bool {
232        let Some(roots) = self.passthrough.get(crate::docx::package::UNMODELLED_ROOTS) else { return false };
233        String::from_utf8_lossy(roots).lines().take(MAX_EMBEDDED_RELS * 2).any(|line| {
234            let fields: Vec<&str> = line.split('\t').collect();
235            matches!(fields.as_slice(), [source, kind, target] if *source == "D" && rel_is(kind, rt::GLOSSARY) && (self.first.contains_key(*target) || self.available(target)))
236        })
237    }

239    /// Have the main document's kept glossary, and the parts it reaches, written without the
240    /// personal information (`names`) and dates the privacy settings remove (#3692). False when a
241    /// part's markup can't be rewritten: the glossary is then written from the model.
242    pub fn scrub_glossary(&mut self, names: bool, dates: bool) -> bool {
243        let Some(roots) = self.passthrough.get(crate::docx::package::UNMODELLED_ROOTS) else { return true };
244        let mut todo: Vec<String> = String::from_utf8_lossy(roots)
245            .lines()
246            .take(MAX_EMBEDDED_RELS * 2)
247            .filter_map(|line| match line.split('\t').collect::<Vec<_>>().as_slice() {
248                [source, kind, target] if *source == "D" && rel_is(kind, rt::GLOSSARY) => Some(target.to_string()),
249                _ => None,
250            })
251            .collect();
252        let mut seen: HashSet<String> = HashSet::new();
253        while let Some(p) = todo.pop() {
254            if seen.len() >= MAX_EMBEDDED_PARTS || !seen.insert(p.clone()) {
255                continue;
256            }
257            let (Some(info), Some(bytes)) = (self.manifest.parts.get(&p), self.passthrough.get(&p)) else { continue };
258            todo.extend(info.rels.iter().filter(|r| !r.external).map(|r| r.target.clone()));
259            if !(info.content_type.ends_with("xml") || p.to_ascii_lowercase().ends_with(".xml")) {
260                continue;
261            }
262            match scrub_private(bytes, names, dates) {
263                Some(b) => {
264                    self.scrubbed.insert(p, b);
265                }
266                None => {
267                    log::warn!(
268                        "docx: the glossary's part {p} can't be cleaned of personal information; the glossary is written from its entries, without what WordCraft doesn't model"
269                    );
270                    return false;
271                }
272            }
273        }
274        true
275    }

277    /// What copying the parts reached from `roots` costs against the copy budget.
278    fn tree_cost(&self, roots: &[&str]) -> u64 {
279        let mut seen: HashSet<&str> = HashSet::new();
280        let mut todo: Vec<&str> = roots.to_vec();
281        let mut cost = 0u64;
282        while let Some(p) = todo.pop() {
283            if !seen.insert(p) || seen.len() > MAX_EMBEDDED_PARTS {
284                continue;
285            }
286            cost = cost.saturating_add(self.passthrough.get(p).map_or(0, |b| b.len() as u64)).saturating_add(PART_COST);
287            if let Some(info) = self.manifest.parts.get(p) {
288                todo.extend(info.rels.iter().filter(|r| !r.external).map(|r| r.target.as_str()));
289            }
290        }
291        cost
292    }

294    /// Write part `root` and the parts it reaches, naming them through `memo` (original → written
295    /// name; parts already in it aren't written again). Returns `root`'s written name and adds
296    /// the indices of the parts written to `written`.
297    fn copy_tree(&mut self, root: &str, memo: &mut HashMap<String, String>, written: &mut Vec<usize>) -> Option<String> {
298        if let Some(n) = memo.get(root) {
299            return Some(n.clone());
300        }
301        if !self.available(root) {
302            return None;
303        }
304        let name = self.alloc(root);
305        memo.insert(root.to_string(), name.clone());
306        let mut todo = vec![root.to_string()];
307        while let Some(orig) = todo.pop() {
308            let (Some(info), Some(bytes)) = (self.manifest.parts.get(&orig).cloned(), self.passthrough.get(&orig)) else { continue };
309            let Some(new) = memo.get(&orig).cloned() else { continue };
310            let mut rels = PartRels::default();
311            for r in &info.rels {
312                if r.external {
313                    rels.add_with_id(&r.id, &r.kind, &r.target, true);
314                    continue;
315                }
316                let target = match memo.get(&r.target) {
317                    Some(t) => t.clone(),
318                    None if self.available(&r.target) => {
319                        let t = self.alloc(&r.target);
320                        memo.insert(r.target.clone(), t.clone());
321                        todo.push(r.target.clone());
322                        t
323                    }
324                    // A part that wasn't kept: leave the relationship out.
325                    None => continue,
326                };
327                rels.add_with_id(&r.id, &r.kind, &relative(&new, &target), false);
328            }
329            let ct = if info.content_type.is_empty() { "application/octet-stream".to_string() } else { info.content_type };
330            let bytes = self.scrubbed.get(&orig).cloned().unwrap_or_else(|| bytes.to_vec());
331            written.push(self.parts.len());
332            self.parts.push((new, bytes, ct, rels));
333        }
334        Some(name)
335    }

339    /// The markup for an object kept from a file, with its relationships added to `rels` (the
340    /// story part's) and the parts it needs queued for writing. `resize`, for an OLE object, writes
341    /// the object's current size and position into its own markup (a chart or diagram's frame is
342    /// written by the caller). `None` when it can't be written back (its parts are gone).
343    pub(super) fn embedded_xml(&mut self, src: &Embedded, rels: &mut PartRels, resize: Option<(f32, f32, &Float, &str)>) -> Option<String> {
344        self.embedded_tree(src, rels, resize).map(|root| root.to_xml())
345    }

347    /// The same bounded preservation operation, retaining the parsed tree for a caller that
348    /// must rebuild an ink AlternateContent drawing frame without reparsing unbound prefixes.
349    pub(super) fn embedded_tree(&mut self, src: &Embedded, rels: &mut PartRels, resize: Option<(f32, f32, &Float, &str)>) -> Option<El> {
350        let Some(left) = self.embeds.markup_bytes.checked_sub(src.xml.len() as u64) else {
351            log::warn!("docx: too much embedded-object markup; the rest is left out");
352            return None;
353        };
354        self.embeds.markup_bytes = left;
355        let mut root = xml::parse(src.xml.as_bytes()).ok()?;
356        let mut used = HashSet::new();
357        rel_attr_values(&root, &mut used, 0);
358        // Every relationship the markup uses must be writable, or the object would be broken.
359        let writable: HashSet<&str> = src.refs.iter().filter(|r| r.external || self.embeds.available(&r.target)).map(|r| r.id.as_str()).collect();
360        if !used.iter().all(|id| writable.contains(id.as_str())) {
361            log::warn!("docx: an embedded object's parts are missing; it is left out");
362            return None;
363        }
364        let roots: Vec<&str> = src.refs.iter().filter(|r| !r.external).map(|r| r.target.as_str()).collect();
365        // Written before (a copy of this object): give this one its own parts while the budget lasts.
366        let again = roots.iter().any(|r| self.embeds.first.contains_key(*r));
367        let cost = if again { self.embeds.tree_cost(&roots) } else { 0 };
368        let fresh = again && cost <= self.embeds.copy_bytes;
369        if fresh {
370            self.embeds.copy_bytes -= cost;
371        }
372        let mut memo = if fresh { HashMap::new() } else { std::mem::take(&mut self.embeds.first) };
373        let mut written = Vec::new();
374        let mut ids: HashMap<String, String> = HashMap::new();
375        for r in &src.refs {
376            let new_id = if r.external {
377                Some(rels.add(&r.kind, &r.target, true))
378            } else {
379                self.embeds.copy_tree(&r.target, &mut memo, &mut written).map(|name| rels.add(&r.kind, &relative(STORY_PART, &name), false))
380            };
381            if let Some(id) = new_id {
382                ids.insert(r.id.clone(), id);
383            }
384        }
385        if !fresh {
386            self.embeds.first = memo;
387        }
388        if used.iter().any(|u| !ids.contains_key(u)) {
389            log::warn!("docx: an embedded object lost a part; it is left out");
390            return None;
391        }
392        // A SmartArt data part names its drawing's relationship (of the story part) by id.
393        for i in written {
394            if let Some((_, bytes, ct, _)) = self.embeds.parts.get_mut(i)
395                && ct.contains("diagramData")
396            {
397                *bytes = patch_rel_ids(bytes, &ids);
398            }
399        }
400        set_rel_ids(&mut root, &ids, 0);
401        if let Some((w, h, float, alt)) = resize {
402            self.resize_object(&mut root, w, h, float, alt, 0);
403        }
404        Some(root)
405    }

407    /// The `a:graphic` of a SmartArt graphic written from its model, laid out `w` × `h` points:
408    /// new diagram parts (`word/diagrams/{data,layout,quickStyle,colors,drawing}N.xml`) related
409    /// from the story part through `rels`.
410    pub(super) fn smart_art_graphic(&mut self, spec: &crate::doc::smart_art::SmartArtSpec, w: f32, h: f32, rels: &mut PartRels) -> String {
411        use crate::docx::package::rt;
412        use crate::docx::smart_art as sa;
413        let names = self.embeds.fresh_set(
414            &["word/diagrams/data", "word/diagrams/layout", "word/diagrams/quickStyle", "word/diagrams/colors", "word/diagrams/drawing"],
415            ".xml",
416        );
417        let [data, layout, style, colors, drawing] = names.as_slice() else { return String::new() };
418        // The drawing first: the data part names it by the story part's relationship id.
419        let drawing_id = rels.add(rt::DIAGRAM_DRAWING, &relative(STORY_PART, drawing), false);
420        let parts = [
421            (data, sa::data_xml(spec, &drawing_id), sa::DATA_CT, rt::DIAGRAM_DATA),
422            (layout, sa::layout_xml(spec.layout), sa::LAYOUT_CT, rt::DIAGRAM_LAYOUT),
423            (style, sa::quick_style_xml(), sa::STYLE_CT, rt::DIAGRAM_QUICK_STYLE),
424            (colors, sa::colors_xml(spec.colors), sa::COLORS_CT, rt::DIAGRAM_COLORS),
425            (drawing, sa::drawing_xml(spec, w, h), sa::DRAWING_CT, rt::DIAGRAM_DRAWING),
426        ];
427        let mut ids = Vec::new();
428        for (name, xml, ct, kind) in parts {
429            self.embeds.parts.push((name.clone(), xml.into_bytes(), ct.to_string(), PartRels::default()));
430            ids.push(rels.add(kind, &relative(STORY_PART, name), false));
431        }
432        let id = |i: usize| ids.get(i).cloned().unwrap_or_default();
433        let mut x = xml::W::default();
434        x.open("a:graphic", &[("xmlns:a", "http://schemas.openxmlformats.org/drawingml/2006/main")]);
435        x.open("a:graphicData", &[("uri", sa::DIAGRAM_URI)]);
436        x.empty(
437            "dgm:relIds",
438            &[
439                ("xmlns:dgm", "http://schemas.openxmlformats.org/drawingml/2006/diagram"),
440                ("xmlns:r", "http://schemas.openxmlformats.org/officeDocument/2006/relationships"),
441                ("r:dm", &id(0)),
442                ("r:lo", &id(1)),
443                ("r:qs", &id(2)),
444                ("r:cs", &id(3)),
445            ],
446        );
447        x.close("a:graphicData");
448        x.close("a:graphic");
449        x.s
450    }

452    /// The `a:graphic` of a chart written from its model: a new chart part (`word/charts/chartN.xml`)
453    /// related from the story part through `rels`.
454    pub(super) fn chart_graphic(&mut self, spec: &crate::doc::chart::ChartSpec, rels: &mut PartRels) -> String {
455        let name = self.embeds.fresh("word/charts/chart", ".xml");
456        let (xml, cut) = crate::docx::chart_spec::chart_part(spec);
457        if cut && !self.embeds.chart_cut {
458            self.embeds.chart_cut = true;
459            log::warn!("docx: a chart has more series or points than a chart part holds; the rest are left out of the file");
460        }
461        let bytes = xml.into_bytes();
462        self.embeds.parts.push((name.clone(), bytes, crate::docx::chart_spec::CHART_CT.to_string(), PartRels::default()));
463        let id = rels.add(crate::docx::package::rt::CHART, &relative(STORY_PART, &name), false);
464        let mut w = xml::W::default();
465        w.open("a:graphic", &[("xmlns:a", "http://schemas.openxmlformats.org/drawingml/2006/main")]);
466        w.open("a:graphicData", &[("uri", crate::docx::chart_spec::CHART_URI)]);
467        w.empty(
468            "c:chart",
469            &[
470                ("xmlns:c", "http://schemas.openxmlformats.org/drawingml/2006/chart"),
471                ("xmlns:r", "http://schemas.openxmlformats.org/officeDocument/2006/relationships"),
472                ("r:id", &id),
473            ],
474        );
475        w.close("a:graphicData");
476        w.close("a:graphic");
477        w.s
478    }

480    /// Write an OLE object's size (and position, when floating), rotation and flips into its VML
481    /// shape and DrawingML picture, with the effect extent covering the rotated bounds as for
482    /// other drawings, its alt text into both, and give its drawing a fresh `wp:docPr` id.
483    fn resize_object(&mut self, e: &mut El, w: f32, h: f32, float: &Float, alt: &str, depth: usize) {
484        if depth > MAX_DEPTH {
485            return;
486        }
487        match e.name.as_str() {
488            "v:shape" => {
489                if let Some((_, style)) = e.attrs.iter_mut().find(|(k, _)| k == "style") {
490                    *style = vml_style_with(style, w, h, float);
491                }
492                set_attr(e, "alt", alt);
493            }
494            "wp:extent" => set_size_attrs(e, w, h),
495            // A turned object needs an effect extent (filled in below) right after its extent.
496            "wp:inline" | "wp:anchor" if !e.els().any(|c| c.name == "wp:effectExtent") && float.spin().deg != 0.0 => {
497                if let Some(at) = e.kids.iter().position(|k| matches!(k, xml::Node::El(c) if c.name == "wp:extent")) {
498                    e.kids.insert(at + 1, xml::Node::El(El { name: "wp:effectExtent".to_string(), attrs: Vec::new(), kids: Vec::new() }));
499                }
500            }
501            "wp:effectExtent" => {
502                // Word's effect extent also covers a rotated object's overhang (its rotated bounds),
503                // unless the file it came from kept none.
504                let (px, py) = if float.is_bare_turn() { (0.0, 0.0) } else { float.spin_pad(w, h) };
505                let [el, et, er, eb] = float.effect_extent();
506                let ext = [el + px, et + py, er + px, eb + py].map(emu);
507                e.attrs.retain(|(k, _)| !matches!(k.as_str(), "l" | "t" | "r" | "b"));
508                for (k, v) in ["l", "t", "r", "b"].into_iter().zip(ext) {
509                    e.attrs.push((k.to_string(), v));
510                }
511            }
512            // The picture's own frame (`pic:spPr`): its size and the object's rotation and flips.
513            "a:xfrm" => {
514                e.attrs.retain(|(k, _)| !matches!(k.as_str(), "rot" | "flipH" | "flipV"));
515                for (k, v) in super::story::spin_attrs(float.spin()) {
516                    e.attrs.push((k.to_string(), v));
517                }
518                for k in e.kids.iter_mut() {
519                    if let xml::Node::El(c) = k
520                        && c.name == "a:ext"
521                    {
522                        set_size_attrs(c, w, h);
523                    }
524                }
525                return;
526            }
527            "wp:docPr" => {
528                let id = self.next_docpr();
529                if let Some((_, v)) = e.attrs.iter_mut().find(|(k, _)| k == "id") {
530                    *v = id;
531                }
532                set_attr(e, "descr", alt);
533            }
534            _ => {}
535        }
536        for k in e.kids.iter_mut() {
537            if let xml::Node::El(c) = k {
538                self.resize_object(c, w, h, float, alt, depth + 1);
539            }
540        }
541    }

543    /// Ink's own transform need not fill its outer drawing frame. Scale its original offsets
544    /// and extents proportionally; an unrelated edit must not stretch the preserved strokes.
545    pub(super) fn ink_transform(e: &mut El, from: (f32, f32), to: (f32, f32), float: &Float, depth: usize) {
546        if depth > MAX_DEPTH {
547            return;
548        }
549        if e.name == "w14:xfrm" {
550            e.attrs.retain(|(k, _)| !matches!(k.as_str(), "rot" | "flipH" | "flipV"));
551            for (k, v) in super::story::spin_attrs(float.spin()) {
552                e.attrs.push((k.to_string(), v));
553            }
554            let scale = |old: f32, new: f32| if old.is_finite() && old > 0.0 && new.is_finite() { (new.max(0.0) / old) as f64 } else { 1.0 };
555            let (sx, sy) = (scale(from.0, to.0), scale(from.1, to.1));
556            let max = MAX_LEN_PT as f64 * 12_700.0;
557            for c in e.kids.iter_mut().filter_map(|k| match k {
558                xml::Node::El(c) => Some(c),
559                _ => None,
560            }) {
561                if !matches!(c.name.as_str(), "a:off" | "a:ext") {
562                    continue;
563                }
564                for (key, value) in &mut c.attrs {
565                    let factor = match key.as_str() {
566                        "x" | "cx" => sx,
567                        "y" | "cy" => sy,
568                        _ => continue,
569                    };
570                    if factor != 1.0
571                        && let Some(n) = int(value)
572                    {
573                        let min = if c.name == "a:ext" { 0.0 } else { -max };
574                        *value = ((n as f64 * factor).clamp(min, max).round() as i64).to_string();
575                    }
576                }
577            }
578            return;
579        }
580        for c in e.kids.iter_mut().filter_map(|k| match k {
581            xml::Node::El(c) => Some(c),
582            _ => None,
583        }) {
584            Self::ink_transform(c, from, to, float, depth + 1);
585        }
586    }

588    /// Put a group or canvas member kept from a file (native ink and its picture fallback) at
589    /// `frame` ([x, y, w, h] in points, in the group's space, as the group is written): its
590    /// `w14:xfrm` and the fallback's `a:xfrm` get that offset and extent and the member's turn.
591    /// It rewrites every `w14:xfrm` and `a:xfrm` under `e` and doesn't descend into them: right
592    /// for the forms Word writes (the content part's own transform and one `pic:pic` fallback's
593    /// `pic:spPr/a:xfrm`); a fallback holding several transforms (a group) would get one frame.
594    pub(super) fn place_member(e: &mut El, frame: [f32; 4], float: &Float, depth: usize) {
595        if depth > MAX_DEPTH {
596            return;
597        }
598        if matches!(e.name.as_str(), "w14:xfrm" | "a:xfrm") {
599            e.attrs.retain(|(k, _)| !matches!(k.as_str(), "rot" | "flipH" | "flipV"));
600            for (k, v) in super::story::spin_attrs(float.spin()) {
601                e.attrs.push((k.to_string(), v));
602            }
603            let [x, y, w, h] = frame.map(|v| v.clamp(-MAX_LEN_PT, MAX_LEN_PT));
604            e.kids.retain(|k| !matches!(k, xml::Node::El(c) if matches!(c.name.as_str(), "a:off" | "a:ext")));
605            let el = |name: &str, attrs: [(&str, String); 2]| {
606                xml::Node::El(El { name: name.to_string(), attrs: attrs.map(|(k, v)| (k.to_string(), v)).to_vec(), kids: Vec::new() })
607            };
608            e.kids.insert(0, el("a:ext", [("cx", emu(w.max(0.0))), ("cy", emu(h.max(0.0)))]));
609            e.kids.insert(0, el("a:off", [("x", emu(x)), ("y", emu(y))]));
610            return;
611        }
612        for c in e.kids.iter_mut().filter_map(|k| match k {
613            xml::Node::El(c) => Some(c),
614            _ => None,
615        }) {
616            Self::place_member(c, frame, float, depth + 1);
617        }
618    }
619}

641/// The distinct relationship ids `e` uses: a set, so a large tree (a native document's markup
642/// never went through the reader's id limit) costs no scan per attribute.
643fn rel_attr_values(e: &El, out: &mut HashSet<String>, depth: usize) {
644    if depth > MAX_DEPTH {
645        return;
646    }
647    for (k, v) in &e.attrs {
648        if xml::is_rel_attr(&e.name, k) && !out.contains(v) {
649            out.insert(v.clone());
650        }
651    }
652    for c in e.els() {
653        rel_attr_values(c, out, depth + 1);
654    }
655}

657fn set_rel_ids(e: &mut El, ids: &HashMap<String, String>, depth: usize) {
658    if depth > MAX_DEPTH {
659        return;
660    }
661    let name = e.name.clone();
662    for (k, v) in e.attrs.iter_mut() {
663        if xml::is_rel_attr(&name, k)
664            && let Some(n) = ids.get(v.as_str())
665        {
666            *v = n.clone();
667        }
668    }
669    for k in e.kids.iter_mut() {
670        if let xml::Node::El(c) = k {
671            set_rel_ids(c, ids, depth + 1);
672        }
673    }

676/// `bytes` with each `relId="old"` naming a remapped relationship rewritten to its new id.
677fn patch_rel_ids(bytes: &[u8], ids: &HashMap<String, String>) -> Vec<u8> {
678    const KEY: &[u8] = b"relId=\"";
679    let mut out = Vec::with_capacity(bytes.len());
680    let mut rest = bytes;
681    while let Some(at) = rest.windows(KEY.len()).position(|w| w == KEY) {
682        let (head, tail) = rest.split_at(at + KEY.len());
683        out.extend_from_slice(head);
684        let end = tail.iter().position(|b| *b == b'"').unwrap_or(tail.len());
685        let (old, after) = tail.split_at(end);
686        match std::str::from_utf8(old).ok().and_then(|o| ids.get(o)) {
687            Some(new) => out.extend_from_slice(new.as_bytes()),
688            None => out.extend_from_slice(old),
689        }
690        rest = after;
691    }
692    out.extend_from_slice(rest);
693    out
694}

696/// `bytes` (kept XML) without the personal information and dates the privacy settings remove, as
697/// the writer leaves them out of the parts it writes: annotation authors become "Author" and
698/// comment initials "A" (`names`); annotation dates and UTC dates go (`dates`); and with `names`,
699/// a people part's authors become "Author" and their presence information goes. Attributes are
700/// matched by namespace, so any prefix is caught. `None` when the markup can't be rewritten
701/// (malformed, or not UTF-8).
702fn scrub_private(bytes: &[u8], names: bool, dates: bool) -> Option<Vec<u8>> {
703    use std::borrow::Cow;
704
705    use quick_xml::events::Event;
706    use quick_xml::events::attributes::Attribute;
707    use quick_xml::name::{QName, ResolveResult};
708    use quick_xml::reader::NsReader;
709    use quick_xml::writer::Writer;
710
711    const WML: &[u8] = b"http://schemas.openxmlformats.org/wordprocessingml/2006/main";
712    const WML_STRICT: &[u8] = b"http://purl.oclc.org/ooxml/wordprocessingml/main";
713    const W15: &[u8] = b"http://schemas.microsoft.com/office/word/2012/wordml";
714    // UTF-16 (or other wide encodings) would pass through unread.
715    if bytes.contains(&0) || bytes.starts_with(&[0xFF, 0xFE]) || bytes.starts_with(&[0xFE, 0xFF]) {
716        return None;
717    }
718    let ns_is = |r: &ResolveResult, ns: &[u8]| matches!(r, ResolveResult::Bound(n) if n.as_ref() == ns);
719    let wml = |r: &ResolveResult| ns_is(r, WML) || ns_is(r, WML_STRICT);
720    let mut r = NsReader::from_reader(bytes);
721    r.config_mut().check_end_names = false;
722    let mut out = Writer::new(Vec::with_capacity(bytes.len()));
723    let mut buf = Vec::new();
724    let mut skip = Vec::new();
725    loop {
726        buf.clear();
727        let (res, ev) = r.read_resolved_event_into(&mut buf).ok()?;
728        let (in_wml, in_w15) = (wml(&res), ns_is(&res, W15));
729        let ev = match ev {
730            Event::Eof => break,
731            Event::Start(ref e) | Event::Empty(ref e) if names && in_w15 && e.local_name().as_ref() == b"presenceInfo" => {
732                if matches!(ev, Event::Start(_)) {
733                    let end = e.name().as_ref().to_vec();
734                    r.read_to_end_into(QName(&end), &mut skip).ok()?;
735                    skip.clear();
736                }
737                continue;
738            }
739            Event::Start(ref e) | Event::Empty(ref e) => {
740                let mut el = e.to_owned();
741                el.clear_attributes();
742                for a in e.attributes().with_checks(false) {
743                    let a = a.ok()?;
744                    let (ares, local) = r.resolver_mut().resolve_attribute(a.key);
745                    // An unprefixed attribute of a WordprocessingML element is read as its own.
746                    let wml_attr = wml(&ares) || (matches!(ares, ResolveResult::Unbound) && in_wml);
747                    let value: Option<&[u8]> = match local.as_ref() {
748                        b"author" if names && (wml_attr || ns_is(&ares, W15)) => Some(super::ANONYMOUS_AUTHOR.as_bytes()),
749                        b"initials" if names && wml_attr => Some(super::ANONYMOUS_INITIALS.as_bytes()),
750                        b"date" if dates && wml_attr => None,
751                        b"dateUtc" if dates => None,
752                        _ => {
753                            // `push_attribute` writes the value as given, inside `"`. In well-formed
754                            // markup the raw value holds no `<` and every `&` starts a valid
755                            // reference, so only a `"` from a single-quoted value needs escaping;
756                            // references such as `&#10;` stay as written (#3872).
757                            let raw = a.value.as_ref();
758                            let value = if raw.contains(&b'"') {
759                                let mut v = Vec::with_capacity(raw.len() + 8);
760                                for &c in raw {
761                                    if c == b'"' {
762                                        v.extend_from_slice(b"&quot;");
763                                    } else {
764                                        v.push(c);
765                                    }
766                                }
767                                Cow::Owned(v)
768                            } else {
769                                Cow::Borrowed(raw)
770                            };
771                            el.push_attribute(Attribute { key: a.key, value });
772                            continue;
773                        }
774                    };
775                    if let Some(v) = value {
776                        el.push_attribute(Attribute { key: a.key, value: Cow::Borrowed(v) });
777                    }
778                }
779                if matches!(ev, Event::Start(_)) { Event::Start(el) } else { Event::Empty(el) }
780            }
781            ev => ev,
782        };
783        out.write_event(ev).ok()?;
784    }
785    Some(out.into_inner())
786}

788/// A VML `style` with the object's size, rotation and flips, and its offset when it floats
789/// (`position:absolute`).
790fn vml_style_with(style: &str, w: f32, h: f32, float: &Float) -> String {
791    let absolute = style
792        .split(';')
793        .filter_map(|d| d.split_once(':'))
794        .any(|(k, v)| k.trim().eq_ignore_ascii_case("position") && v.trim().eq_ignore_ascii_case("absolute"));
795    let mut out: Vec<String> = style
796        .split(';')
797        .filter(|d| {
798            let key = d.split_once(':').map_or("", |(k, _)| k).trim().to_ascii_lowercase();
799            !(d.trim().is_empty()
800                || matches!(key.as_str(), "width" | "height" | "rotation" | "flip")
801                || (absolute && matches!(key.as_str(), "left" | "top" | "margin-left" | "margin-top")))
802        })
803        .map(|d| d.trim().to_string())
804        .collect();
805    if absolute {
806        out.push(format!("margin-left:{}pt", pt(float.x)));
807        out.push(format!("margin-top:{}pt", pt(float.y)));
808    }
809    out.push(format!("width:{}pt", pt(w.max(0.0))));
810    out.push(format!("height:{}pt", pt(h.max(0.0))));
811    // VML `rotation`: degrees clockwise; `flip`: `x` and/or `y` (as `read::story::vml_float` reads them).
812    let spin = float.spin();
813    if spin.deg != 0.0 {
814        out.push(format!("rotation:{}", pt(spin.deg)));
815    }
816    let flip: Vec<&str> = [(spin.flip_h, "x"), (spin.flip_v, "y")].into_iter().filter(|(on, _)| *on).map(|(_, f)| f).collect();
817    if !flip.is_empty() {
818        out.push(format!("flip:{}", flip.join(" ")));
819    }
820    out.join(";")
821}

823/// Points for a VML style: finite, bounded, at most two decimals.
824fn pt(v: f32) -> String {
825    let v = crate::geom::finite(v).clamp(-crate::docx::units::MAX_LEN_PT, crate::docx::units::MAX_LEN_PT);
826    let s = format!("{v:.2}");
827    s.trim_end_matches('0').trim_end_matches('.').to_string()
828}

830/// `name` as a valid OPC part name: no backslashes, empty or dot segments, segments ending in a
831/// dot, or encoded separators.
832fn valid_part_name(name: &str) -> String {
833    let mut out = Vec::new();
834    for seg in name.split('/') {
835        let mut seg = seg.replace('\\', "_");
836        for enc in ["%2f", "%5c"] {
837            while let Some(i) = seg.to_ascii_lowercase().find(enc) {
838                seg.replace_range(i..i + 3, "_");
839            }
840        }
841        if seg.is_empty() || seg == "." || seg == ".." {
842            continue;
843        }
844        // A segment can't end in a dot either (`word/a.`).
845        if seg.ends_with('.') {
846            seg = format!("{}_", seg.trim_end_matches('.'));
847        }
848        out.push(seg);
849    }
850    if out.is_empty() { "word/embeddings/part.xml".to_string() } else { out.join("/") }
851}
