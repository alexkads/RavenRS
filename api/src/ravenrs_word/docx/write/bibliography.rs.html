3fn to_source_type(kind: &str) -> &str {
4    match kind.to_ascii_lowercase().as_str() {
5        "book" => "Book",
6        "booksection" | "chapter" => "BookSection",
7        "article" | "journal" | "journalarticle" => "JournalArticle",
8        "periodical" | "articleinaperiodical" => "ArticleInAPeriodical",
9        "website" | "site" | "internetsite" => "InternetSite",
10        "documentfrominternetsite" => "DocumentFromInternetSite",
11        "electronicsource" => "ElectronicSource",
12        "report" => "Report",
13        "conference" | "proceedings" | "conferenceproceedings" => "ConferenceProceedings",
14        "case" => "Case",
15        "art" => "Art",
16        "film" => "Film",
17        "interview" => "Interview",
18        "patent" => "Patent",
19        "performance" => "Performance",
20        "soundrecording" => "SoundRecording",
21        "standard" => "Standard",
22        "thesis" => "Thesis",
23        "other" | "misc" => "Misc",
24        _ => "Book",
25    }
26}

28pub(crate) fn from_source_type(st: &str) -> String {

53/// The document's citation style, APA when it has none, and a style sheet file name made from
54/// it (`\APA.XSL`, `\MLA.XSL`).
55fn bib_style(doc: &Document) -> (&str, String) {
56    let name = doc.settings.bib_style.as_deref().map(str::trim).filter(|n| !n.is_empty()).unwrap_or("APA");
57    let file: String = name.chars().filter(char::is_ascii_alphanumeric).collect();
58    (name, format!("\\{}.XSL", if file.is_empty() { "APA" } else { &file }))
59}

61fn write_name_list(w: &mut W, names: &str) {
62    w.open("b:NameList", &[]);
63    let persons: Vec<_> = names.split(';').map(str::trim).filter(|s| !s.is_empty()).collect();
64    for a in persons {
65        w.open("b:Person", &[]);
66        // Parse "Last, First" or "Last, First Middle"
67        let (last, first_middle) = match a.split_once(',') {
68            Some((l, f)) => (l.trim(), f.trim()),
69            None => (a, ""),
70        };
71        w.leaf("b:Last", &[], last);
72        if !first_middle.is_empty() {
73            let parts: Vec<&str> = first_middle.split_whitespace().collect();
74            if let Some(first) = parts.first() {
75                w.leaf("b:First", &[], first);
76            }
77            if let Some(slice) = parts.get(1..).filter(|s| !s.is_empty()) {
78                let mid = slice.join(" ");
79                w.leaf("b:Middle", &[], &mid);
80            }
81        }
82        w.close("b:Person");
83    }
84    w.close("b:NameList");
85}

87fn split_accessed_date(s: &str) -> (Option<&str>, Option<&str>, Option<&str>) {
88    let s = s.trim();
89    if s.is_empty() {
90        return (None, None, None);
91    }
92    let parts: Vec<&str> = if s.contains('-') {
93        s.split('-').collect()
94    } else if s.contains('/') {
95        s.split('/').collect()
96    } else {
97        Vec::new()
98    };
99    if let [y, m, d] = parts.as_slice() {
100        (Some(y.trim()), Some(m.trim()), Some(d.trim()))
101    } else if let [y, m] = parts.as_slice() {
102        (Some(y.trim()), Some(m.trim()), None)
103    } else {
104        (Some(s), None, None)
105    }
106}

108pub(super) fn bibliography_xml(doc: &Document) -> Vec<u8> {
109    let (style_name, selected) = bib_style(doc);
110    let mut w = W::new();
111    w.open(
112        "b:Sources",
113        &[
114            ("SelectedStyle", selected.as_str()),
115            ("StyleName", style_name),
116            ("xmlns:b", "http://schemas.openxmlformats.org/officeDocument/2006/bibliography"),
117            ("xmlns", "http://schemas.openxmlformats.org/officeDocument/2006/bibliography"),
118        ],
119    );
120    for src in &doc.sources {
121        w.open("b:Source", &[]);
122        w.leaf("b:Tag", &[], &src.tag);
123        w.leaf("b:SourceType", &[], to_source_type(&src.kind));
124        // Author & Editor. A role holds a name list or a corporate name, not both (ECMA-376
125        // `CT_NameOrCorporateType`), so a corporate author takes the author's place (#4848).
126        let corporate = src.corporate.trim();
127        if !src.author.is_empty() || !corporate.is_empty() || !src.editor.is_empty() {
128            w.open("b:Author", &[]);
129            if !corporate.is_empty() {
130                w.open("b:Author", &[]);
131                w.leaf("b:Corporate", &[], corporate);
132                w.close("b:Author");
133            } else if !src.author.is_empty() {
134                w.open("b:Author", &[]);
135                write_name_list(&mut w, &src.author);
136                w.close("b:Author");
137            }
138            if !src.editor.is_empty() {
139                w.open("b:Editor", &[]);
140                write_name_list(&mut w, &src.editor);
141                w.close("b:Editor");
142            }
143            w.close("b:Author");
144        }
145        if !src.title.is_empty() {
146            w.leaf("b:Title", &[], &src.title);
147        }
148        if !src.year.is_empty() {
149            w.leaf("b:Year", &[], &src.year);
150        }
151        if !src.month.is_empty() {
152            w.leaf("b:Month", &[], &src.month);
153        }
154        if !src.day.is_empty() {
155            w.leaf("b:Day", &[], &src.day);
156        }
157        if !src.city.is_empty() {
158            w.leaf("b:City", &[], &src.city);
159        }
160        if !src.publisher.is_empty() {
161            w.leaf("b:Publisher", &[], &src.publisher);
162        }
163        if !src.journal.is_empty() {
164            w.leaf("b:JournalName", &[], &src.journal);
165        }
166        if !src.edition.is_empty() {
167            w.leaf("b:Edition", &[], &src.edition);
168        }
169        if !src.volume.is_empty() {
170            w.leaf("b:Volume", &[], &src.volume);
171        }
172        if !src.issue.is_empty() {
173            w.leaf("b:Issue", &[], &src.issue);
174        }
175        if !src.pages.is_empty() {
176            w.leaf("b:Pages", &[], &src.pages);
177        }
178        if !src.url.is_empty() {
179            w.leaf("b:URL", &[], &src.url);
180        }
181        if !src.accessed.is_empty() {
182            let (y, m, d) = split_accessed_date(&src.accessed);
183            if let Some(y) = y.filter(|s| !s.is_empty()) {
184                w.leaf("b:YearAccessed", &[], y);
185            }
186            if let Some(m) = m.filter(|s| !s.is_empty()) {
187                w.leaf("b:MonthAccessed", &[], m);
188            }
189            if let Some(d) = d.filter(|s| !s.is_empty()) {
190                w.leaf("b:DayAccessed", &[], d);
191            }
192            w.leaf("b:Accessed", &[], &src.accessed);
193        }
194        if !src.doi.is_empty() {
195            w.leaf("b:DOI", &[], &src.doi);
196            w.leaf("b:StandardNumber", &[], &src.doi);
197        }
198        w.close("b:Source");
199    }
200    w.close("b:Sources");
201    w.into_bytes()
202}

204pub(super) fn bibliography_props_xml() -> Vec<u8> {
205    let mut w = W::new();
206    w.open(
207        "ds:datastoreItem",
208        &[("xmlns:ds", "http://schemas.openxmlformats.org/officeDocument/2006/customXml"), ("ds:itemID", "{3315B9A8-2323-4554-B8B2-1234567890AB}")],
209    );
210    w.open("ds:schemaRefs", &[]);
211    w.empty("ds:schemaRef", &[("ds:uri", "http://schemas.openxmlformats.org/officeDocument/2006/bibliography")]);
212    w.close("ds:schemaRefs");
213    w.close("ds:datastoreItem");
214    w.into_bytes()
