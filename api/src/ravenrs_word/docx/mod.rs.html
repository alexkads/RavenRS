1//! WordCraft DOCX reader and writer (ECMA-376 / ISO/IEC 29500 WordprocessingML).
2//!
3//! [`read`] turns a `.docx` package into a [`crate::doc::Document`]; [`write`](fn@write) does the
4//! reverse. Both are written from the public specification. The reader is lenient (unknown
5//! markup is skipped, bad numbers fall back to defaults) and bounded (zip entry sizes, total
6//! decompressed size, XML depth and element counts are capped), so hostile input yields an
7//! error or a best-effort document, never a panic.
8#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]

40/// Largest `.xml` file read as a Word XML document.
41pub const MAX_XML_FILE: usize = 1 << 30;
42
43/// Read a Word XML file (`.xml`): a Word XML Document (Flat OPC, the whole package in one XML
44/// file) or a Word 2003 XML Document (WordprocessingML 2003), told apart by the root element.
45/// Any other XML gives [`DocxError::NotWordXml`].
46pub fn read_xml(bytes: &[u8]) -> Result<crate::doc::Document, DocxError> {
47    if bytes.len() > MAX_XML_FILE {
48        return Err(DocxError::Limit(format!("XML file of {} bytes", bytes.len())));
49    }
50    match sniff_xml(bytes) {
51        XmlKind::FlatOpc => flat::read(bytes),
52        XmlKind::WordMl2003 => wordml2003::read(bytes),
53        XmlKind::Other => Err(DocxError::NotWordXml),
54    }
55}

57/// The title (`dc:title`) in a package's core properties part (`docProps/core.xml`, ECMA-376
58/// Part 2 §11), trimmed; `None` when the part can't be read or has no title. Only the package's
59/// directory and at most `max_part` bytes of that part are inflated: for listing templates by
60/// name without unpacking the whole package. A package over 64 MiB, or one declaring more entries
61/// than `MAX_ENTRIES` (20,000), has none (#3660).
62pub fn package_title<R: std::io::Read>(package: R, max_part: u64) -> Option<String> {
63    use std::io::Read;
64    const MAX_PACKAGE: u64 = 64 << 20;
65    let mut bytes = Vec::new();
66    package.take(MAX_PACKAGE + 1).read_to_end(&mut bytes).ok()?;
67    if bytes.len() as u64 > MAX_PACKAGE {
68        return None;
69    }
70    crate::doc::zip_guard::check(&bytes, package::MAX_ENTRIES as u64).ok()?;
71    let mut zip = zip::ZipArchive::new(std::io::Cursor::new(bytes)).ok()?;
72    let part = zip.by_name("docProps/core.xml").ok()?;
73    let mut xml = Vec::new();
74    part.take(max_part).read_to_end(&mut xml).ok()?;
75    let root = xml::parse(&xml).ok()?;
76    let title = root.child("dc:title")?.text();
77    let title = title.trim();
78    (!title.is_empty()).then(|| title.to_string())
79}

91/// Read a package that may be password-protected ([`is_encrypted`]): it is decrypted with
92/// `password` first. An encrypted package without a password gives
93/// [`DocxError::PasswordRequired`]; a wrong one, [`DocxError::WrongPassword`].
94pub fn read_with_password(bytes: &[u8], password: Option<&str>) -> Result<crate::doc::Document, DocxError> {
95    read_with(bytes, &ReadOptions { password, ..Default::default() })
96}
97
98/// Read a package with options: the password of an encrypted one ([`read_with_password`]) and
99/// an importer for `w:altChunk` content in formats other than WordprocessingML (see
100/// [`ChunkImporter`]).
101pub fn read_with(bytes: &[u8], opts: &ReadOptions) -> Result<crate::doc::Document, DocxError> {
102    if is_encrypted(bytes) {
103        let package = zeroize::Zeroizing::new(decrypt(bytes, opts.password)?);
104        return read::read_package(&package, opts.chunks);
105    }
106    read::read_package(bytes, opts.chunks)
107}

109/// Which kind of WordprocessingML package to write. The main part's content type differs per
110/// kind, and Word refuses a file whose content type doesn't match its extension.
111#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
112pub enum Flavor {
113    /// `.docx`
114    #[default]
115    Document,
116    /// `.docm`: may carry a VBA project.
117    MacroDocument,
118    /// `.dotx`
119    Template,
120    /// `.dotm`: may carry a VBA project.
121    MacroTemplate,
122}
123
124impl Flavor {
125    /// The flavour a file extension (without the dot, any case) names.
126    pub fn from_ext(ext: &str) -> Option<Flavor> {
127        match ext.to_ascii_lowercase().as_str() {
128            "docx" => Some(Flavor::Document),
129            "docm" => Some(Flavor::MacroDocument),
130            "dotx" => Some(Flavor::Template),
131            "dotm" => Some(Flavor::MacroTemplate),
132            _ => None,
133        }
134    }
135
136    /// Content type of the main document part.
137    pub fn main_content_type(self) -> &'static str {
138        match self {
139            Flavor::Document => "application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml",
140            Flavor::MacroDocument => "application/vnd.ms-word.document.macroEnabled.main+xml",
141            Flavor::Template => "application/vnd.openxmlformats-officedocument.wordprocessingml.template.main+xml",
142            Flavor::MacroTemplate => "application/vnd.ms-word.template.macroEnabledTemplate.main+xml",
143        }
144    }
145
146    /// Can this package hold macros?
147    pub fn macros(self) -> bool {
148        matches!(self, Flavor::MacroDocument | Flavor::MacroTemplate)
149    }
150}

152/// Errors from reading or writing a DOCX package.
153#[derive(Debug, thiserror::Error, Clone, PartialEq)]
154pub enum DocxError {
155    /// The bytes are not a readable zip archive.
156    #[error("not a valid zip package: {0}")]
157    Zip(String),
158    /// A required part is absent.
159    #[error("missing part: {0}")]
160    MissingPart(String),
161    /// A part is not well-formed XML.
162    #[error("malformed XML: {0}")]
163    Xml(String),
164    /// The input exceeds a safety limit (size, nesting, element count).
165    #[error("limit exceeded: {0}")]
166    Limit(String),
167    /// The package is readable but isn't a WordprocessingML document.
168    #[error("not a Word document: {0}")]
169    NotWord(String),
170    /// An `.xml` file that is neither a Word XML Document (Flat OPC) nor Word 2003 XML.
171    #[error("not a Word XML document")]
172    NotWordXml,
173    /// The package is encrypted and no password was given.
174    #[error("{}", PASSWORD_REQUIRED)]
175    PasswordRequired,
176    /// The password doesn't open the package.
177    #[error("{}", WRONG_PASSWORD)]
178    WrongPassword,
179    /// The package is encrypted in a way WordCraft can't read, or its encryption data is damaged
180    /// (or encrypting failed).
181    #[error("encrypted document: {0}")]
182    Encryption(String),
