40/// Relationships of one part.
41#[derive(Default)]
42pub struct PartRels {
43    list: Vec<(String, String, String, bool)>,
44    index: HashMap<(String, String, bool), String>,
45}
46
47impl PartRels {
48    /// Add (or reuse) a relationship; returns its id.
49    pub fn add(&mut self, kind: &str, target: &str, external: bool) -> String {
50        let key = (kind.to_string(), target.to_string(), external);
51        if let Some(id) = self.index.get(&key) {
52            return id.clone();
53        }
54        let id = format!("rId{}", self.list.len() + 1);
55        self.list.push((id.clone(), kind.to_string(), target.to_string(), external));
56        self.index.insert(key, id.clone());
57        id
58    }
59    /// Add a relationship with a given id (the id a kept part's markup uses); a repeated id is
60    /// ignored. Not to be mixed with [`PartRels::add`] in one part.
61    pub fn add_with_id(&mut self, id: &str, kind: &str, target: &str, external: bool) {
62        if self.list.iter().any(|(i, ..)| i == id) {
63            return;
64        }
65        self.list.push((id.to_string(), kind.to_string(), target.to_string(), external));
66    }
67    pub(crate) fn is_empty(&self) -> bool {
68        self.list.is_empty()
69    }
70    pub(crate) fn xml(&self) -> Vec<u8> {
71        let mut w = W::new();
72        w.open("Relationships", &[("xmlns", "http://schemas.openxmlformats.org/package/2006/relationships")]);
73        for (id, kind, target, external) in &self.list {
74            if *external {
75                w.empty("Relationship", &[("Id", id), ("Type", kind), ("Target", target), ("TargetMode", "External")]);
76            } else {
77                w.empty("Relationship", &[("Id", id), ("Type", kind), ("Target", target)]);
78            }
79        }
80        w.close("Relationships");
81        w.into_bytes()
82    }
83}

143/// Write a `.docx` package.
144pub fn write(doc: &Document) -> Result<Vec<u8>, DocxError> {
145    write_as(doc, Flavor::Document)
146}

148/// Write a package of the given flavour (`.docx`, `.docm`, `.dotx`, `.dotm`).
149pub fn write_as(doc: &Document, flavor: Flavor) -> Result<Vec<u8>, DocxError> {
150    zip_entries(&write_parts(doc, flavor)?)
151}

153impl Writer<'_> {
154    fn assign_media(&mut self) {
155        let mut used: std::collections::HashSet<String> = std::collections::HashSet::new();
156        for (key, bytes) in &self.doc.media {
157            let clean: String = key.chars().filter(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-')).take(80).collect();
158            let has_known_ext = clean.rsplit_once('.').is_some_and(|(s, _)| !s.is_empty()) && content_type_for(&clean) != "application/octet-stream";
159            let mut base = if has_known_ext && !clean.starts_with('.') {
160                clean
161            } else {
162                let stem = clean.trim_matches('.');
163                let stem = if stem.is_empty() { "image" } else { stem };
164                format!("{stem}.{}", sniff(bytes))
165            };
166            let (stem, ext) = base.rsplit_once('.').map(|(s, e)| (s.to_string(), e.to_string())).unwrap_or((base.clone(), "bin".into()));
167            let mut i = 1;
168            while used.contains(&base.to_ascii_lowercase()) {
169                i += 1;
170                base = format!("{stem}_{i}.{ext}");
171            }
172            used.insert(base.to_ascii_lowercase());
173            self.media_files.insert(key.clone(), base);
174        }
175    }
176
177    fn next_docpr(&mut self) -> String {
178        self.docpr = self.docpr.saturating_add(1);
179        self.docpr.to_string()
180    }
181
182    /// The drawing id of a shape: its own (once; a copy gets a fresh one), else a fresh one.
183    fn shape_docpr(&mut self, own: u32) -> String {
184        if own != 0 && self.shape_ids.insert(own) { own.to_string() } else { self.next_docpr() }
185    }
186
187    fn bookmark_id(&mut self, name: &str) -> String {
188        let n = self.bookmarks.len() as u32;
189        self.bookmarks.entry(name.to_string()).or_insert(n).to_string()
190    }
191
192    fn note_id(&mut self, part: u32, foot: bool) -> String {
193        let base = note_id_base(self.doc, foot);
194        let list = if foot { &mut self.footnotes } else { &mut self.endnotes };
195        let idx = match list.iter().position(|p| *p == part) {
196            Some(i) => i,
197            None => {
198                list.push(part);
199                list.len() - 1
200            }
201        };
202        (idx + base).to_string()
203    }
204
205    fn hf_rel(&mut self, part: u32, footer: bool, rels: &mut PartRels) -> Option<String> {
206        let kind = if part == self.wm_header && !footer { PartKind::Header } else { self.doc.parts.get(&part)?.kind };
207        if !matches!(kind, PartKind::Header | PartKind::Footer) {
208            return None;
209        }
210        let file = match self.hf.iter().find(|(p, _, f)| *p == part && *f == footer) {
211            Some((_, f, _)) => f.clone(),
212            None => {
213                let n = self.hf.iter().filter(|(_, _, f)| *f == footer).count() + 1;
214                let f = format!("{}{n}.xml", if footer { "footer" } else { "header" });
215                self.hf.push((part, f.clone(), footer));
216                f
217            }
218        };
219        Some(rels.add(if footer { rt::FOOTER } else { rt::HEADER }, &file, false))
220    }
221
222    fn take_para_id(&self, p: &crate::doc::Paragraph) -> Option<String> {
223        self.para_ids.get(&(p as *const _ as usize)).cloned()
224    }
225
226    /// Whether `p` holds a reference to the note being written.
227    fn holds_own_ref(&self, p: &crate::doc::Paragraph) -> bool {
228        let Some((foot, id)) = self.current_note else { return false };
229        p.objects.iter().any(|o| match o {
230            crate::doc::InlineObject::NoteRef { kind, id: nid, .. } => *nid == id && (*kind == crate::doc::para::NoteKind::Footnote) == foot,
231            _ => false,
232        })
233    }
234
235    /// Note stories: the first paragraph starts with the note's own reference mark.
236    fn note_blocks(&mut self, w: &mut W, blocks: &Blocks, rels: &mut PartRels, foot: bool, part: u32) {
237        self.pending_mark = Some(if foot { "w:footnoteRef" } else { "w:endnoteRef" });
238        self.current_note = Some((foot, part));
239        self.blocks(w, blocks, rels, false, 0);
240        self.current_note = None;
241        self.pending_mark = None;
242    }
