3/// Word draws a table of contents' entry links without the Hyperlink character style (no blue,
4/// no underline) though the entries carry it (measured in Word for Windows: a Hyperlink-styled run
5/// in a hyperlink inside a TOC field's result is plain; one outside a hyperlink, another style,
6/// direct formatting, or a hyperlink in another field keep their look). The style is dropped from
7/// those runs so they look the same here; the links stay. The style is found by its built-in
8/// name: localized Word and older files give it another id (`Hipervnculo`, `Hyperlnk`).
9pub(super) fn unstyle_toc_links(doc: &mut Document) {
10    let ids: HashSet<String> = doc
11        .styles
12        .styles
13        .iter()
14        .filter(|s| s.kind == StyleKind::Character && s.name.eq_ignore_ascii_case("Hyperlink"))
15        .map(|s| s.id.clone())
16        .chain(["Hyperlink".to_string()])
17        .collect();
18    for f in doc.field_ranges(crate::doc::StoryRef::Body) {
19        if is_toc(&f.instr) {
20            let _ = doc.format_range(&f.start, &f.end, &|c| {
21                if c.link.is_some() && c.style.as_deref().is_some_and(|s| ids.contains(s)) {
22                    c.style = None;
23                }
24            });
25        }
26    }
27}

29/// Word's Table of Contents content control belongs to the TOC: the writer puts it back around
30/// the TOC heading and its entries (`toc_span`), which References › Update Table rebuilds. It is
31/// handed to the TOC only when the writer would put it back around the same paragraphs; any other
32/// is kept as read, or saving would drop it (#2427).
33pub(super) fn toc_control_to_toc(body: &mut Blocks) {
34    let span = {
35        let events = control_events(body);
36        let (pairs, _) = pair_events(&events);
37        pairs.iter().find_map(|&(o, c, depth)| {
38            let (ControlEvent::Open { block: s, off: Some(0), control }, ControlEvent::Close { block: e, off: Some(off) }) =
39                (events.get(o)?, events.get(c)?)
40            else {
41                return None;
42            };
43            let toc = matches!(&control.kind, ControlKind::Gallery { list: false, gallery, .. } if gallery == "Table of Contents");
44            let at_end = body.get(*e)?.as_para()?.len() == off.checked_add(OBJ.len_utf8())?;
45            (depth == 0 && toc && at_end).then_some((*s, *e, *off))
46        })
47    };
48    let Some((s, e, off)) = span else { return };
49    let mut toc = body.clone();
50    // The end first: both markers may be in the same paragraph.
51    if let Some(Block::Para(p)) = toc.get_mut(e).map(Arc::make_mut) {
52        let _ = p.delete(off, p.len());
53    }
54    if let Some(Block::Para(p)) = toc.get_mut(s).map(Arc::make_mut) {
55        let _ = p.delete(0, OBJ.len_utf8());
56    }
57    toc_field_to_heading(&mut toc);
58    if crate::docx::write::written_toc_span(&toc) == Some((s, e)) {
59        *body = toc;
60        return;
61    }
62    // TOCs the writer rebuilds in the same paragraphs once Update Table has turned them into the
63    // engine's TOC (`range_toc_to_field`) are handed over as the range read, as without the control:
64    // - Word's layout, the field's end alone in a paragraph after the entries (it goes with them);
65    // - ours with a heading that isn't a `TOCHeading`, written empty and the field opening in the
66    //   first entry. Kept in the control, that paragraph wouldn't be empty and each save and update
67    //   would add a blank line above the entries (#2182).
68    let mut engine = toc.clone();
69    let mut last = e;
70    // The entry before Word's paragraph holding nothing but the field's end, and that end's format.
71    let end_only = e.checked_sub(1).filter(|&l| l > s).and_then(|l| {
72        let p = engine.get(e)?.as_para()?;
73        let only = p.section.is_none() && p.len() == OBJ.len_utf8() && matches!(p.object_at(0), Some(InlineObject::FieldEnd));
74        only.then(|| (l, p.props_of_char(0).clone()))
75    });
76    if let Some((l, props)) = end_only {
77        engine.remove(e);
78        last = l;
79        if let Some(Block::Para(p)) = engine.get_mut(l).map(Arc::make_mut) {
80            let len = p.len();
81            let _ = p.insert_object(len, InlineObject::FieldEnd, &props);
82        }
83    }
84    // An empty paragraph opening the control takes the field. It may end a section, as above
85    // entries in columns (#3889).
86    let empty_heading = engine.get(s).and_then(|b| b.as_para()).is_some_and(|p| p.is_empty());
87    if empty_heading {
88        field_to_heading(&mut engine, s + 1);
89    } else {
90        toc_field_to_heading(&mut engine);
91    }
92    if crate::docx::write::written_toc_span(&engine) == Some((s, last)) {
93        *body = if empty_heading { toc } else { engine };
94    }
95}

97/// Normalize only a complete TOC range that the engine's heading/entry representation can
98/// reproduce exactly. Other imported TOCs keep their explicit boundaries, independent of styles.
99pub(super) fn toc_field_to_heading(body: &mut Blocks) {
100    for i in 1..body.len() {
101        if body.get(i - 1).and_then(|b| b.as_para()).is_some_and(|p| p.props.style.as_deref() == Some("TOCHeading")) {
102            field_to_heading(body, i);
103        }
104    }
105}

107/// Moves the TOC field that opens paragraph `i` (a complete range over the entries, or an empty
108/// field) to the end of paragraph `i - 1`, the engine's TOC heading, unless that one has a TOC
109/// field already.
110fn field_to_heading(body: &mut Blocks, i: usize) {
111    let has_toc = |o: &InlineObject| matches!(o, InlineObject::Field { instr, .. } if is_toc(instr));
112    if i == 0 || body.get(i - 1).and_then(|b| b.as_para()).is_none_or(|h| h.objects.iter().any(has_toc)) {
113        return;
114    }
115    let Some((field, props, end)) = body.get(i).and_then(|b| b.as_para()).and_then(|p| {
116        let (field, end) = match p.object_at(0) {
117            Some(InlineObject::FieldStart { instr, locked, dirty }) if is_toc(instr) => (
118                InlineObject::Field { instr: instr.clone(), result: String::new(), locked: *locked, dirty: *dirty, code: false, form: String::new() },
119                Some(toc_range_end(body, i)?),
120            ),
121            Some(f @ InlineObject::Field { result, .. }) if result.is_empty() && has_toc(f) => (f.clone(), None),
122            _ => return None,
123        };
124        Some((field, p.props_of_char(0).clone(), end))
125    }) else {
126        return;
127    };
128    // Delete the end first: the two markers may occupy the same paragraph.
129    if let Some(Block::Para(p)) = end.and_then(|end| body.get_mut(end)).map(Arc::make_mut) {
130        let off = p.len().saturating_sub(OBJ.len_utf8());
131        let _ = p.delete(off, p.len());
132    }
133    if let Some(Block::Para(p)) = body.get_mut(i).map(Arc::make_mut) {
134        let _ = p.delete(0, OBJ.len_utf8());
135    }
136    if let Some(Block::Para(h)) = body.get_mut(i - 1).map(Arc::make_mut) {
137        let end = h.len();
138        let _ = h.insert_object(end, field, &props);
139    }
140}

142pub(super) fn is_toc(instr: &str) -> bool {
143    instr.split_whitespace().next().is_some_and(|k| k.eq_ignore_ascii_case("TOC"))
144}

146/// The actual matching end must also be the end the styled writer would choose: at the end
147/// of the last contiguous TOC entry, with no section break or trailing content outside it.
148fn toc_range_end(body: &Blocks, start: usize) -> Option<usize> {
149    let entry =
150        |p: &crate::doc::Paragraph| p.section.is_none() && p.props.style.as_deref().is_some_and(|s| s.starts_with("TOC") && s != "TOCHeading");
151    let mut depth = 0_usize;
152    for (i, b) in body.iter().enumerate().skip(start) {
153        let p = b.as_para().filter(|p| entry(p))?;
154        for (off, o) in p.object_offsets().into_iter().zip(&p.objects) {
155            match o {
156                InlineObject::FieldStart { .. } => depth = depth.checked_add(1)?,
157                InlineObject::FieldEnd => {
158                    depth = depth.checked_sub(1)?;
159                    if depth == 0 {
160                        let following_entry = body.get(i + 1).and_then(|b| b.as_para()).is_some_and(entry);
161                        return (off.checked_add(OBJ.len_utf8()) == Some(p.len()) && !following_entry).then_some(i);
162                    }
163                }
164                _ => {}
165            }
166        }
167    }
168    None
