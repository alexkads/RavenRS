1//! Objects kept for writing back: charts, SmartArt diagrams and OLE objects.
2//!
3//! WordCraft draws charts and diagrams and shows an OLE object's picture, but doesn't edit them.
4//! So that saving doesn't lose them, the reader keeps each one's own markup (an `a:graphic` or a
5//! `w:object`) as a [`crate::doc::graphic::Embedded`] on the object, and every part that markup
6//! reaches through relationships (chart, embedded workbook, colours and style; diagram data, layout,
7//! quick style, colours and drawing; OLE embedding and its picture…) as opaque bytes in
8//! `Document::passthrough`, with their content types and relationships in an
9//! [`EmbeddedManifest`](crate::docx::package::EmbeddedManifest). The font table and the fonts it embeds
10//! are kept the same way.

65/// Most relationship ids one object's markup may use.
66const MAX_IDS: usize = 64;

68/// The relationship ids the main document's markup refers to: the ids it names in a
69/// relationship attribute (`r:id="rId5"`, VML's `o:relid`) — the same rule the embedded objects
70/// use ([`rel_ids`]).
71///
72/// One pass over the markup the reader has already parsed, not one search per id: the ids are
73/// looked up as they are met, so the cost is the size of the document rather than its size
74/// times the number of relationships (#12). A name that is not a relationship is ignored, and a
75/// name that is not one of the document's relationship ids is not collected. An id that only
76/// appears as text is not a reference: the text does not name a part, and treating it as one
77/// lost the part on save.
78fn referenced_ids<'a>(root: &El, document: &'a Rels) -> HashSet<&'a [u8]> {
79    // The ids the loop in `keep_unmodelled` can ask about: the first `MAX_EMBEDDED_RELS`
80    // non-external relationships, and no more, so the set stays bounded however many the file
81    // carries.
82    let ids: HashSet<&[u8]> = document.list.iter().filter(|r| !r.external).take(MAX_EMBEDDED_RELS).map(|r| r.id.as_bytes()).collect();
83    let mut found = HashSet::new();
84    if ids.is_empty() {
85        return found;
86    }
87    collect_rel_ids(root, &ids, &mut found, 0);
88    found
89}
90
91/// The ids `e` names in a relationship attribute that are in `ids`, collected into `found`.
92///
93/// The attribute is a relationship attribute by [`is_rel_attr`]: one in the relationship
94/// namespace, VML's `o:relid`, or the unqualified alias of one on an element that carries a
95/// relationship — the same attribute the reader resolves a relationship from, so a part the
96/// markup names is not carried over a second time.
97fn collect_rel_ids<'a>(e: &El, ids: &HashSet<&'a [u8]>, found: &mut HashSet<&'a [u8]>, depth: usize) {
98    if depth > MAX_DEPTH {
99        return;
100    }
101    for (k, v) in &e.attrs {
102        if is_rel_attr(&e.name, k)
103            && let Some(id) = ids.get(v.as_bytes())
104        {
105            found.insert(*id);
106        }
107    }
108    for c in e.els() {
109        collect_rel_ids(c, ids, found, depth + 1);
110    }
111}

113impl Reader<'_> {
114    /// Keep custom XML stores reached from the main document or package, without evaluating
115    /// bindings. A bibliography already loaded into the source model belongs to its writer.
116    pub(super) fn keep_custom_xml(&mut self, package: &Rels, document: &Rels) {
117        let mut roots = String::new();
118        for (owner, rels) in [("P", package), ("D", document)] {
119            for rel in rels.list.iter().filter(|r| !r.external && rel_is(&r.kind, rt::CUSTOM_XML)).take(MAX_EMBEDDED_RELS) {
120                if ![&rel.kind, &rel.target].iter().all(|s| manifest_field_ok(s)) {
121                    continue;
122                }
123                if !self.doc.sources.is_empty() && self.xml(&rel.target).ok().flatten().is_some_and(|r| r.local() == "Sources") {
124                    continue;
125                }
126                // Native ink shares the custom XML relationship type, but its part belongs to the
127                // drawing that refers to it: kept with it, or dropped with it (no orphan part).
128                let pkg = self.pkg;
129                if self.content_types.get_or_insert_with(|| ContentTypes::read(pkg)).of(&rel.target) == Some("application/inkml+xml") {
130                    continue;
131                }
132                if self.keep_part_tree(&rel.target) {
133                    roots.push_str(&format!("{owner}\t{}\t{}\n", rel.kind, rel.target));
134                }
135            }
136        }
137        if !roots.is_empty() {
138            self.doc.passthrough.insert(CUSTOM_XML_ROOTS.into(), Arc::new(roots.into_bytes()));
139        }
140    }

142    /// Keep the parts the package and the main document relate to that WordCraft neither models
143    /// nor writes itself (web settings, styles-with-effects, add-in task panes, the thumbnail, a
144    /// glossary…), with every part they reach, so saving carries them unchanged. Skipped: the
145    /// relationship types the writer produces or the reader models ([`HANDLED_RELS`]) and any
146    /// relationship the main document's markup refers to by id (its target goes with the
147    /// element, or not at all).
148    pub(super) fn keep_unmodelled(&mut self, root: &El, package: &Rels, document: &Rels) {
149        // The ids the main document's markup refers to, in one pass over the markup: an id is
150        // named in a relationship attribute (`r:id="rId5"`). One scan for all of them, not one
151        // search per relationship — a large document with many relationships took minutes to
152        // open (#12).
153        let referenced = referenced_ids(root, document);
154        let mut roots = String::new();
155        // One glossary per main document (its first relationship; see `read_glossary`).
156        let mut glossary = false;
157        for (owner, rels) in [("P", package), ("D", document)] {
158            for rel in rels.list.iter().filter(|r| !r.external).take(MAX_EMBEDDED_RELS) {
159                let tail = rel.kind.rsplit('/').next().unwrap_or("");
160                if HANDLED_RELS.contains(&tail)
161                    || (owner == "D" && referenced.contains(rel.id.as_bytes()))
162                    || (owner == "D" && tail == "glossaryDocument" && std::mem::replace(&mut glossary, true))
163                    || ![&rel.kind, &rel.target].iter().all(|s| manifest_field_ok(s))
164                {
165                    continue;
166                }
167                if self.keep_part_tree(&rel.target) {
168                    roots.push_str(&format!("{owner}\t{}\t{}\n", rel.kind, rel.target));
169                }
170            }
171        }
172        if !roots.is_empty() {
173            self.doc.passthrough.insert(crate::docx::package::UNMODELLED_ROOTS.into(), Arc::new(roots.into_bytes()));
174        }
175    }

177    /// Keep `el` (an object's `a:graphic` or `w:object`, read with `rels`) and the parts it refers
178    /// to, so the object can be written back. `extra` names relationships the object's parts use
179    /// by id although its markup doesn't (a SmartArt drawing, named in the diagram's data part).
180    /// `None` when it can't be written back faithfully: a relationship it uses is missing, a part
181    /// is missing, or it's too big.
182    pub(super) fn keep_embedded(&mut self, el: &El, rels: &Rels, extra: &[String]) -> Option<Arc<Embedded>> {
183        let mut ids = Vec::new();
184        rel_ids(el, &mut ids, 0)?;
185        let mut refs = Vec::new();
186        for id in &ids {
187            let rel = rels.by_id(id)?;
188            if !rel.external && !self.keep_part_tree(&rel.target) {
189                return None;
190            }
191            refs.push(EmbeddedRef { id: id.clone(), kind: rel.kind.clone(), target: rel.target.clone(), external: rel.external });
192        }
193        for id in extra.iter().filter(|id| !ids.contains(id)) {
194            if let Some(rel) = rels.by_id(id).filter(|r| !r.external)
195                && self.keep_part_tree(&rel.target)
196            {
197                refs.push(EmbeddedRef { id: id.clone(), kind: rel.kind.clone(), target: rel.target.clone(), external: false });
198            }
199        }
200        let xml = standalone_xml(el)?;
201        Some(Arc::new(Embedded { xml, refs }))
202    }
203
204    /// Keep part `root` and every part it reaches through its relationships. False when `root`
205    /// itself can't be kept.
206    pub(super) fn keep_part_tree(&mut self, root: &str) -> bool {
207        if self.embedded.parts.contains_key(root) {
208            return true;
209        }
210        if !self.keep_part(root) {
211            return false;
212        }
213        let mut todo = vec![root.to_string()];
214        while let Some(part) = todo.pop() {
215            let mut kept: Vec<Rel> = Vec::new();
216            for r in self.pkg.rels(&part).list {
217                if kept.len() >= MAX_EMBEDDED_RELS {
218                    log::warn!("docx: part {part} has more than {MAX_EMBEDDED_RELS} relationships; the rest are dropped");
219                    break;
220                }
221                if ![&r.id, &r.kind, &r.target].iter().all(|f| manifest_field_ok(f)) || kept.iter().any(|k| k.id == r.id) {
222                    continue;
223                }
224                if !r.external && !self.embedded.parts.contains_key(&r.target) {
225                    if !self.keep_part(&r.target) {
226                        continue;
227                    }
228                    todo.push(r.target.clone());
229                }
230                kept.push(r);
231            }
232            if let Some(p) = self.embedded.parts.get_mut(&part) {
233                p.rels = kept;
234            }
235        }
236        true
237    }
238
239    /// Keep one part's bytes and content type (not its relationships).
240    fn keep_part(&mut self, part: &str) -> bool {
241        let lower = part.to_ascii_lowercase();
242        // `wordcraft:` names are WordCraft's own bookkeeping in `passthrough`, never a file's part.
243        if !manifest_field_ok(part)
244            || lower.ends_with(".rels")
245            || lower == "[content_types].xml"
246            || lower.starts_with("wordcraft:")
247            || part.eq_ignore_ascii_case(&self.main)
248        {
249            log::warn!("docx: not carrying part {part:?} for an embedded object");
250            return false;
251        }
252        if self.embedded.parts.len() >= MAX_EMBEDDED_PARTS {
253            log::warn!("docx: more than {MAX_EMBEDDED_PARTS} parts for embedded objects; the rest are dropped");
254            return false;
255        }
256        let Some(bytes) = self.pkg.get_shared(part) else {
257            log::warn!("docx: part {part} of an embedded object is missing");
258            return false;
259        };
260        let pkg = self.pkg;
261        let ct = self.content_types.get_or_insert_with(|| ContentTypes::read(pkg)).of(part).unwrap_or("application/octet-stream");
262        let content_type = if manifest_field_ok(ct) { ct.to_string() } else { "application/octet-stream".to_string() };
263        self.doc.passthrough.entry(part.to_string()).or_insert_with(|| bytes.clone());
264        self.embedded.parts.insert(part.to_string(), crate::docx::package::EmbeddedPart { content_type, rels: Vec::new() });
265        true
266    }
267}

269/// The relationship ids `e` uses (attributes in the relationships namespace, VML's `o:relid`,
270/// and the unqualified alias of one on an element that carries a relationship), in document
271/// order. `None` past [`MAX_IDS`].
272fn rel_ids(e: &El, out: &mut Vec<String>, depth: usize) -> Option<()> {
273    if depth > MAX_DEPTH {
274        return Some(());
275    }
276    for (k, v) in &e.attrs {
277        if is_rel_attr(&e.name, k) && !out.contains(v) {
278            if out.len() >= MAX_IDS {
279                return None;
280            }
281            out.push(v.clone());
282        }
283    }
284    for c in e.els() {
285        rel_ids(c, out, depth + 1)?;
286    }
287    Some(())
288}

290/// `e` as a standalone element: its XML with the namespace declarations its canonical prefixes
291/// need. `None` when too long, or in a namespace we can't name.
292fn standalone_xml(e: &El) -> Option<String> {
293    let xml = e.to_xml();
294    let rest = xml.get(1 + e.name.len()..).filter(|_| !xml.is_empty())?;
295    let mut s = format!("<{}", e.name);
296    for (p, u) in NAMESPACES {
297        s.push_str(&format!(" xmlns:{p}=\"{}\"", esc(u)));
298    }
299    s.push_str(rest);
300    (s.len() <= MAX_EMBEDDED_XML).then_some(s)
301}

313    /// A package whose main part holds `body` (the XML inside `w:body`), and which relates to
314    /// `rels` parts of a type the reader doesn't model, so each one is looked for in the part.
315    fn package(body: &str, rels: &[(&str, &str)], parts: &[(&str, &[u8])]) -> Vec<u8> {
316        let doc = format!(r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?><w:document {W_NS}><w:body>{body}</w:body></w:document>"#);
317        let mut r = String::from(r#"<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">"#);
318        for (id, target) in rels {
319            r.push_str(&format!(
320                r#"<Relationship Id="{id}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/printerSettings" Target="{target}"/>"#
321            ));
322        }
323        r.push_str("</Relationships>");
324
325        let mut zw = zip::ZipWriter::new(std::io::Cursor::new(Vec::new()));
326        let opts = zip::write::SimpleFileOptions::default();
327        zw.start_file("[Content_Types].xml", opts).unwrap();
328        zw.write_all(CONTENT_TYPES.as_bytes()).unwrap();
329        zw.start_file("_rels/.rels", opts).unwrap();
330        zw.write_all(ROOT_RELS.as_bytes()).unwrap();
331        zw.start_file("word/document.xml", opts).unwrap();
332        zw.write_all(doc.as_bytes()).unwrap();
333        zw.start_file("word/_rels/document.xml.rels", opts).unwrap();
334        zw.write_all(r.as_bytes()).unwrap();
335        for (name, bytes) in parts {
336            zw.start_file(name, opts).unwrap();
337            zw.write_all(bytes).unwrap();
338        }
339        zw.finish().unwrap().into_inner()
340    }

347    /// Opening a document costs the size of its main part, not that size times the number of its
348    /// relationships (#12). Eight megabytes of main part with a thousand relationships used to
349    /// take half a minute; the scan is now one pass over the markup.
350    #[test]
351    fn a_document_with_many_relationships_opens_in_the_time_its_part_takes() {
352        let rels: Vec<(String, String)> = (0..1024).map(|i| (format!("rIdT{i}"), format!("file{i}.bin"))).collect();
353        let rels: Vec<(&str, &str)> = rels.iter().map(|(a, b)| (a.as_str(), b.as_str())).collect();
354        let body = format!(r#"<w:p><w:r><w:t xml:space="preserve">{}</w:t></w:r></w:p>"#, "a".repeat(8 << 20));
355        let bytes = package(&body, &rels, &[]);
356        let t = std::time::Instant::now();
357        let doc = crate::docx::read(&bytes).unwrap();
358        let took = t.elapsed();
359        assert_eq!(doc.plain_text(crate::doc::StoryRef::Body).len(), 8 << 20, "the text is there");
360        assert!(took.as_secs() < 15, "opening it took {took:?}");
361    }

363    /// The scan finds the relationships the markup names, so a part the document already refers
364    /// to is not carried over a second time (#12).
365    #[test]
366    fn a_relationship_the_markup_names_is_not_carried_over_again() {
367        // The markup names the relationship, so it is not carried over a second time as an
368        // unmodelled part. (The part itself survives only if the writer writes it with the
369        // element that names it; the reader does not carry it on its own.)
370        let named = package(
371            r#"<w:p><w:r><w:t>before</w:t></w:r></w:p><w:sectPr><w:printerSettings r:id="rIdDoc1"/></w:sectPr>"#,
372            &[("rIdDoc1", "keep.bin")],
373            &[("word/keep.bin", b"kept")],
374        );
375        let roots = unmodelled_roots(&crate::docx::read(&named).unwrap());
376        assert!(!roots.contains("keep.bin"), "the part is carried twice: {roots}");
377
378        // A part the markup doesn't name is carried over, as before.
379        let unnamed = package(r#"<w:p><w:r><w:t>no mention here</w:t></w:r></w:p>"#, &[("rIdDoc1", "keep.bin")], &[("word/keep.bin", b"kept")]);
380        let roots = unmodelled_roots(&crate::docx::read(&unnamed).unwrap());
381        assert!(roots.contains("keep.bin"), "the part is dropped: {roots}");
382    }
