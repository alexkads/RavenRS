14/// Theme colour `i` (0 dk1, 1 lt1, 2 dk2, 3 lt2, 4..9 accent1..6, 10 hlink, 11 folHlink), falling
15/// back to the default theme.
16pub(crate) fn slot(theme: &[Rgb], i: usize) -> Rgb {
17    theme.get(i).or_else(|| THEME_COLORS.get(i)).copied().unwrap_or(Rgb::BLACK)
18}

20/// The colour of a fill or line element: a solid colour, the first stop of a gradient, or the
21/// foreground of a pattern. `None` for no fill and picture fills.
22pub(crate) fn paint(e: &El, theme: &[Rgb]) -> Option<Rgb> {
23    match e.name.as_str() {
24        "a:gradFill" => e.child("a:gsLst").and_then(|l| l.child("a:gs")).and_then(|g| color(g, theme)),
25        "a:pattFill" => e.child("a:fgClr").and_then(|c| color(c, theme)),
26        "a:noFill" | "a:blipFill" => None,
27        _ => color(e, theme),
28    }
29}

31/// The fill of a `…Pr` or `a:ln` element (its first solid, gradient or pattern fill), as [`paint`].
32pub(crate) fn fill_colour(e: &El, theme: &[Rgb]) -> Option<Rgb> {
33    let f = e.els().find(|c| matches!(c.name.as_str(), "a:solidFill" | "a:gradFill" | "a:pattFill"))?;
34    paint(f, theme)
35}

37fn is_color(e: &El) -> bool {
38    matches!(e.name.as_str(), "a:srgbClr" | "a:schemeClr" | "a:sysClr" | "a:prstClr" | "a:scrgbClr" | "a:hslClr")
39}

41/// A colour element (`a:srgbClr`, `a:schemeClr`, `a:sysClr`), or the one inside `e`, with its
42/// `lumMod` / `lumOff` / `tint` / `shade` / `satMod` applied in document order.
43pub(crate) fn color(e: &El, theme: &[Rgb]) -> Option<Rgb> {
44    color_alpha(e, theme, None).map(|(c, _)| c)
45}

47/// [`color`] and its transparency, percent (`a:alpha` is opacity, so 30% opaque is 70%
48/// transparent). `ph` is the colour and transparency `phClr` stands for (in a theme's fill
49/// styles: the colour of the shape's style reference).
50pub(crate) fn color_alpha(e: &El, theme: &[Rgb], ph: Option<(Rgb, f32)>) -> Option<(Rgb, f32)> {
51    let c = if is_color(e) { e } else { e.els().find(|c| is_color(c))? };
52    let mut opacity = 100.0;
53    let base = match c.name.as_str() {
54        "a:srgbClr" => Rgb::parse(c.attr("val")?)?,
55        "a:schemeClr" if c.attr("val") == Some("phClr") => {
56            let (rgb, t) = ph?;
57            opacity -= t;
58            rgb
59        }
60        "a:schemeClr" => *theme.get(scheme_slot(c.attr("val")?)?)?,
61        "a:sysClr" => match c.attr("lastClr").and_then(Rgb::parse) {
62            Some(rgb) => rgb,
63            None => sys_color(c.attr("val")?)?,
64        },
65        "a:prstClr" => preset_color(c.attr("val")?)?,
66        "a:scrgbClr" => {
67            // Linear 0–100 000 per channel, to sRGB.
68            let ch = |n: &str| -> u8 {
69                let l = c.attr(n).and_then(int).map_or(0.0, |v| (v as f64 / 100_000.0).clamp(0.0, 1.0));
70                let s = if l <= 0.003_130_8 { l * 12.92 } else { 1.055 * l.powf(1.0 / 2.4) - 0.055 };
71                (s * 255.0).round().clamp(0.0, 255.0) as u8
72            };
73            Rgb(ch("r"), ch("g"), ch("b"))
74        }
75        "a:hslClr" => {
76            let v = |n: &str| c.attr(n).and_then(int).unwrap_or(0) as f64;
77            let hue = (v("hue") / 60_000.0).rem_euclid(360.0) / 360.0;
78            Rgb::from_hsl(hue, (v("sat") / 100_000.0).clamp(0.0, 1.0), (v("lum") / 100_000.0).clamp(0.0, 1.0))
79        }
80        _ => return None,
81    };
82    let val = |m: &El| m.attr("val").and_then(int).map_or(0.0, |v| (v as f64 / 100_000.0).clamp(-10.0, 10.0));
83    let mut rgb = base;
84    for m in c.els() {
85        rgb = match m.name.as_str() {
86            n @ ("a:lumMod" | "a:lumOff" | "a:tint" | "a:shade" | "a:satMod") => modified(rgb, n.trim_start_matches("a:"), val(m)),
87            "a:alpha" => {
88                opacity = val(m) as f32 * 100.0;
89                rgb
90            }
91            "a:alphaMod" => {
92                opacity *= val(m) as f32;
93                rgb
94            }
95            "a:alphaOff" => {
96                opacity += val(m) as f32 * 100.0;
97                rgb
98            }
99            _ => rgb,
100        };
101    }
102    Some((rgb, (100.0 - opacity).clamp(0.0, 100.0)))
103}

105/// Fill element `e` (`a:solidFill`, `a:gradFill`; a pattern as its foreground colour) as a
106/// shape's fill; `None` for no fill, picture and group fills. `ph` as in [`color_alpha`].
107pub(crate) fn fill(e: &El, theme: &[Rgb], ph: Option<(Rgb, f32)>) -> Option<Fill> {
108    let solid = |c: Option<&El>| c.and_then(|c| color_alpha(c, theme, ph)).map(|(color, transparency)| Fill::Solid { color, transparency });
109    match e.name.as_str() {
110        "a:solidFill" => solid(Some(e)),
111        "a:pattFill" => solid(e.child("a:fgClr")),
112        "a:gradFill" => {
113            let pct = |v: Option<&str>| v.and_then(int).map_or(0.0, |v| v.clamp(-100_000_000, 100_000_000) as f32 / 1000.0);
114            let rect = |r: Option<&El>| r.map_or([0.0; 4], |r| ["l", "t", "r", "b"].map(|k| pct(r.attr(k))));
115            let stops = e
116                .child("a:gsLst")
117                .into_iter()
118                .flat_map(|l| l.children("a:gs"))
119                .take(MAX_STOPS * 4)
120                .filter_map(|g| color_alpha(g, theme, ph).map(|(color, transparency)| GradStop { pos: pct(g.attr("pos")), color, transparency }))
121                .collect();
122            let on = |v: Option<&str>| v.is_some_and(|v| v == "1" || v == "true");
123            let kind = match (e.child("a:path"), e.child("a:lin")) {
124                (Some(p), _) => {
125                    let shape = match p.attr("path") {
126                        Some("rect") => PathShape::Rect,
127                        Some("shape") => PathShape::Shape,
128                        _ => PathShape::Circle,
129                    };
130                    GradientKind::Path { shape, focus: rect(p.child("a:fillToRect")) }
131                }
132                (None, l) => GradientKind::Linear {
133                    angle: l.and_then(|l| l.attr("ang")).and_then(int).map_or(0.0, |a| (a % 21_600_000) as f32 / 60_000.0),
134                    scaled: on(l.and_then(|l| l.attr("scaled"))),
135                },
136            };
137            let flip = match e.attr("flip") {
138                Some("x") => GradientFlip::X,
139                Some("y") => GradientFlip::Y,
140                Some("xy") => GradientFlip::Xy,
141                _ => GradientFlip::None,
142            };
143            let tile = e.child("a:tileRect").map(|r| rect(Some(r)));
144            Fill::Gradient(Arc::new(Gradient { stops, kind, tile, rot_with_shape: on(e.attr("rotWithShape")), flip })).sanitized()
145        }
146        _ => None,
147    }
148}

150/// A preset colour (`a:prstClr/@val`, ECMA-376 Part 1 §20.1.10.48): the common ones; others are
151/// unknown (`None`).
152pub(crate) fn preset_color(name: &str) -> Option<Rgb> {
153    let lower = name.to_ascii_lowercase();
154    Some(match lower.as_str() {
155        "black" => Rgb(0, 0, 0),
156        "white" => Rgb(0xFF, 0xFF, 0xFF),
157        "red" => Rgb(0xFF, 0, 0),
158        "green" => Rgb(0, 0x80, 0),
159        "lime" => Rgb(0, 0xFF, 0),
160        "blue" => Rgb(0, 0, 0xFF),
161        "yellow" => Rgb(0xFF, 0xFF, 0),
162        "cyan" | "aqua" => Rgb(0, 0xFF, 0xFF),
163        "magenta" | "fuchsia" => Rgb(0xFF, 0, 0xFF),
164        "gray" | "grey" => Rgb(0x80, 0x80, 0x80),
165        "silver" => Rgb(0xC0, 0xC0, 0xC0),
166        "ltgray" | "ltgrey" | "lightgray" | "lightgrey" => Rgb(0xD3, 0xD3, 0xD3),
167        "dkgray" | "dkgrey" | "darkgray" | "darkgrey" => Rgb(0xA9, 0xA9, 0xA9),
168        "maroon" => Rgb(0x80, 0, 0),
169        "navy" => Rgb(0, 0, 0x80),
170        "olive" => Rgb(0x80, 0x80, 0),
171        "purple" => Rgb(0x80, 0, 0x80),
172        "teal" => Rgb(0, 0x80, 0x80),
173        "orange" => Rgb(0xFF, 0xA5, 0),
174        "brown" => Rgb(0xA5, 0x2A, 0x2A),
175        "pink" => Rgb(0xFF, 0xC0, 0xCB),
176        "gold" => Rgb(0xFF, 0xD7, 0),
177        "darkblue" | "dkblue" => Rgb(0, 0, 0x8B),
178        "darkred" | "dkred" => Rgb(0x8B, 0, 0),
179        "darkgreen" | "dkgreen" => Rgb(0, 0x64, 0),
180        "lightblue" | "ltblue" => Rgb(0xAD, 0xD8, 0xE6),
181        "coral" => Rgb(0xFF, 0x7F, 0x50),
182        "salmon" => Rgb(0xFA, 0x80, 0x72),
183        "tomato" => Rgb(0xFF, 0x63, 0x47),
184        "violet" => Rgb(0xEE, 0x82, 0xEE),
185        "indigo" => Rgb(0x4B, 0, 0x82),
186        "beige" => Rgb(0xF5, 0xF5, 0xDC),
187        "ivory" => Rgb(0xFF, 0xFF, 0xF0),
188        _ => return None,
189    })
190}

192/// A system colour without its `lastClr`: the usual light-theme values of the common ones.
193fn sys_color(name: &str) -> Option<Rgb> {
194    Some(match name {
195        "windowText" | "btnText" | "menuText" | "captionText" | "infoText" => Rgb(0, 0, 0),
196        "window" | "menu" | "highlightText" => Rgb(0xFF, 0xFF, 0xFF),
197        "btnFace" | "3dLight" => Rgb(0xF0, 0xF0, 0xF0),
198        "grayText" | "btnShadow" => Rgb(0xA0, 0xA0, 0xA0),
199        "highlight" => Rgb(0x00, 0x78, 0xD7),
200        _ => return None,
201    })
202}

204/// Index into `Document::settings.theme_colors` (dk1, lt1, dk2, lt2, accent1..6, hlink, folHlink).
205fn scheme_slot(name: &str) -> Option<usize> {
206    Some(match name {
207        "dk1" | "tx1" => 0,
208        "lt1" | "bg1" => 1,
209        "dk2" | "tx2" => 2,
210        "lt2" | "bg2" => 3,
211        "accent1" => 4,
212        "accent2" => 5,
213        "accent3" => 6,
214        "accent4" => 7,
215        "accent5" => 8,
216        "accent6" => 9,
217        "hlink" => 10,
218        "folHlink" => 11,
219        _ => return None,
220    })
221}

223/// Theme colour `name` (`accent1`, `tx1`…) of `theme`.
224pub(crate) fn scheme(name: &str, theme: &[Rgb]) -> Option<Rgb> {
225    scheme_slot(name).map(|i| slot(theme, i))
226}

228/// `c` with its luminance changed by `f`.
229pub(crate) fn lum(c: Rgb, f: impl Fn(f64) -> f64) -> Rgb {
230    adjust_lum(c, f)
231}

233/// `c` with the DrawingML colour modifier `name` at `v` (see [`Rgb::modified`]).
234fn modified(c: Rgb, name: &str, v: f64) -> Rgb {
235    c.modified(name, v)
236}

238/// `c` with its HSL luminance changed by `f`.
239pub(crate) fn adjust_lum(c: Rgb, f: impl Fn(f64) -> f64) -> Rgb {
240    c.with_lum(f)
241}

243#[cfg(test)]
244mod tests {

249    const NS: &str = r#"xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main""#;

251    /// The element `name` with `attrs` and `body`, parsed.
252    fn el(name: &str, attrs: &str, body: &str) -> El {
253        crate::docx::xml::parse(format!(r#"<{name} {NS} {attrs}>{body}</{name}>"#).as_bytes()).unwrap()
254    }
255
256    #[test]
257    fn gradient_and_pattern_fills_take_their_first_colour() {
258        let theme = THEME_COLORS.to_vec();
259        let grad = el("a:gradFill", "", r#"<a:gsLst><a:gs pos="0"><a:srgbClr val="FF0000"/></a:gs></a:gsLst>"#);
260        assert_eq!(paint(&grad, &theme), Some(Rgb(255, 0, 0)));
261        let patt =
262            el("a:pattFill", r#"prst="dkDnDiag""#, r#"<a:fgClr><a:srgbClr val="00FF00"/></a:fgClr><a:bgClr><a:srgbClr val="0000FF"/></a:bgClr>"#);
263        assert_eq!(paint(&patt, &theme), Some(Rgb(0, 255, 0)));
264        let ln = el("a:ln", "", r#"<a:gradFill><a:gsLst><a:gs pos="0"><a:srgbClr val="FF0000"/></a:gs></a:gsLst></a:gradFill>"#);
265        assert_eq!(fill_colour(&ln, &theme), Some(Rgb(255, 0, 0)));
266        assert_eq!(fill_colour(&el("a:ln", "", "<a:noFill/>"), &theme), None);
267    }
268
269    #[test]
270    fn tint_and_shade_move_towards_white_and_black() {
271        let theme = THEME_COLORS.to_vec();
272        let near = |c: Option<Rgb>| c.is_some_and(|Rgb(r, g, b)| [r, g, b].iter().all(|x| x.abs_diff(128) <= 1));
273        assert!(near(color(&el("a:srgbClr", r#"val="000000""#, r#"<a:tint val="50000"/>"#), &theme)));
274        assert!(near(color(&el("a:srgbClr", r#"val="FFFFFF""#, r#"<a:shade val="50000"/>"#), &theme)));
275    }
276
277    #[test]
278    fn modifiers_apply_in_the_order_written() {
279        let theme = THEME_COLORS.to_vec();
280        let a = el("a:srgbClr", r#"val="808080""#, r#"<a:lumMod val="50000"/><a:lumOff val="50000"/>"#);
281        let b = el("a:srgbClr", r#"val="808080""#, r#"<a:lumOff val="50000"/><a:lumMod val="50000"/>"#);
282        assert_ne!(color(&a, &theme), color(&b, &theme));
283    }

285    #[test]
286    fn preset_colors_are_case_insensitive_and_include_common_names() {
287        assert_eq!(preset_color("red"), Some(Rgb(255, 0, 0)));
288        assert_eq!(preset_color("Red"), Some(Rgb(255, 0, 0)));
289        assert_eq!(preset_color("darkBlue"), Some(Rgb(0, 0, 0x8B)));
290        assert_eq!(preset_color("dkBlue"), Some(Rgb(0, 0, 0x8B)));
291        assert_eq!(preset_color("lightGray"), Some(Rgb(0xD3, 0xD3, 0xD3)));
292        assert_eq!(preset_color("ltGray"), Some(Rgb(0xD3, 0xD3, 0xD3)));
293        assert_eq!(preset_color("gold"), Some(Rgb(0xFF, 0xD7, 0)));
294        assert_eq!(preset_color("unknownColor"), None);
295    }
