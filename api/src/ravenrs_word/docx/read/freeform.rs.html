9/// Straight pieces a Bézier curve is flattened into.
10const CURVE_STEPS: usize = 8;
11/// Points read from one path segment. A conformant `a:lnTo`/`a:moveTo` has one, `a:quadBezTo`
12/// two and `a:cubicBezTo` three; the cap bounds the de Casteljau control polygon so a segment
13/// stuffed with points can't make curve flattening quadratic in the point count.
14const MAX_SEG_POINTS: usize = 8;
15/// Largest path coordinate kept, in EMUs.
16const MAX_EMU: f64 = 1.0e12;
17
18/// The paths of the custom geometry in `sppr` (`wps:spPr`), mapped onto a `w` × `h` shape
19/// (points): each `a:path` in its own `w` × `h` space (EMUs of the shape's size when it gives
20/// none). Lines and Bézier curves (flattened) are kept; arcs are drawn straight to their end.
21/// `None` when there is no `a:custGeom` or it has no points.
22pub fn cust_geom(sppr: &El, w: f32, h: f32) -> Option<Freeform> {
23    let list = sppr.child("a:custGeom")?.child("a:pathLst")?;
24    let mut paths = Vec::new();
25    for p in list.children("a:path").take(MAX_PATHS) {
26        // Subpaths count too: stop once the shape has as many paths as it may keep.
27        if paths.len() >= MAX_PATHS {
28            break;
29        }
30        let dim = |n: &str, shape: f32| p.attr(n).and_then(int).filter(|v| *v > 0).map_or(shape as f64 * 12_700.0, |v| v as f64);
31        let (pw, ph) = (dim("w", w), dim("h", h));
32        let scale = |[x, y]: [f64; 2]| -> [f32; 2] {
33            let sx = if pw > 0.0 { w as f64 / pw } else { 1.0 / 12_700.0 };
34            let sy = if ph > 0.0 { h as f64 / ph } else { 1.0 / 12_700.0 };
35            [(x * sx) as f32, (y * sy) as f32]
36        };
37        let pt = |e: &El| -> Option<[f64; 2]> {
38            let c = |n: &str| e.attr(n).and_then(int).map(|v| (v as f64).clamp(-MAX_EMU, MAX_EMU));
39            Some([c("x")?, c("y")?])
40        };
41        // Each subpath (from an `a:moveTo`) becomes a path of its own.
42        let mut cur: Vec<[f64; 2]> = Vec::new();
43        let mut done: Vec<(Vec<[f64; 2]>, bool)> = Vec::new();
44        let fill_none = p.attr("fill") == Some("none");
45        for seg in p.els() {
46            if cur.len() >= MAX_POINTS || done.len() >= MAX_PATHS {
47                break;
48            }
49            let pts: Vec<[f64; 2]> = seg.children("a:pt").filter_map(pt).take(MAX_SEG_POINTS).collect();
50            match seg.local() {
51                "moveTo" => {
52                    if !cur.is_empty() {
53                        done.push((std::mem::take(&mut cur), false));
54                    }
55                    cur.extend(pts.first());
56                }
57                "lnTo" => cur.extend(pts.first()),
58                "cubicBezTo" | "quadBezTo" => {
59                    let Some(&start) = cur.last() else {
60                        cur.extend(pts.last());
61                        continue;
62                    };
63                    let ctrl: Vec<[f64; 2]> = std::iter::once(start).chain(pts.iter().copied()).collect();
64                    for i in 1..=CURVE_STEPS {
65                        cur.push(bezier(&ctrl, i as f64 / CURVE_STEPS as f64));
66                    }
67                }
68                "close" if !cur.is_empty() => done.push((std::mem::take(&mut cur), true)),
69                _ => {}
70            }
71        }
72        if !cur.is_empty() {
73            // A path drawn back to where it started is closed, though it has no `a:close` (#743:
74            // Word's cover pages), so it's filled.
75            let back = cur.len() > 2 && cur.first() == cur.last();
76            done.push((cur, back));
77        }
78        for (pts, closed) in done.into_iter().take(MAX_PATHS.saturating_sub(paths.len())) {
79            paths.push(FreePath { pts: pts.into_iter().take(MAX_POINTS).map(scale).collect(), closed: closed && !fill_none });
80        }
81    }
82    let f = Freeform { w, h, paths, alpha: 1.0, ink: None, order: None }.sanitized();
83    (!f.paths.is_empty()).then_some(f)
84}

86/// Largest VML path coordinate kept (coordinate-space units).
87const MAX_VML: f64 = 1.0e12;
88/// Characters of a VML `path` attribute read; the rest is ignored.
89const MAX_VML_PATH: usize = 1 << 20;
90/// Parameters of a VML path kept, all commands together.
91const MAX_VML_PARAMS: usize = MAX_POINTS * 8;

93/// A legacy VML freeform (`v:shape` with a `path`, ECMA-376 Part 4 §14.1.2.19).
94pub struct VmlPath {
95    pub freeform: Freeform,
96    /// `nf` in the path: no fill.
97    pub no_fill: bool,
98    /// `ns` in the path: no outline.
99    pub no_stroke: bool,
100}

102/// The VML path commands that are two letters long.
103const VML_TWO: [&str; 11] = ["nf", "ns", "ae", "al", "at", "ar", "wa", "wr", "qx", "qy", "qb"];
104
105/// The VML `path` of a shape, mapped onto its `w` × `h` frame (points) through its `coordsize`
106/// (default `1000,1000`) and `coordorigin` (default `0,0`). Commands: `m`/`t` move (absolute /
107/// relative), `l`/`r` line, `c`/`v` cubic Bézier (flattened), `x` close, `e` end, `nf`/`ns` no
108/// fill / no outline; arcs (`at`, `ar`, `wa`, `wr`) and quadrants (`qx`, `qy`) are drawn straight
109/// to their end, `qb` straight to its last point, and ellipses (`ae`, `al`) are skipped. An empty
110/// parameter is 0. `None` when the path has no points, or uses formulas (`@n`, from `v:formulas`,
111/// which aren't evaluated): such a shape isn't drawn rather than drawn wrong.
112pub fn vml_path(path: &str, coordsize: Option<&str>, coordorigin: Option<&str>, w: f32, h: f32) -> Option<VmlPath> {
113    if path.contains('@') {
114        return None;
115    }
116    let pair = |v: Option<&str>| -> Option<(f64, f64)> {
117        let v = v?;
118        let (a, b) = v.split_once(',').unwrap_or((v, ""));
119        let n = |s: &str| int(s).map(|i| (i as f64).clamp(-MAX_VML, MAX_VML));
120        Some((n(a)?, n(b).unwrap_or(0.0)))
121    };
122    let size = |v: f64| if v > 0.0 { v } else { 1000.0 };
123    let (cw, ch) = pair(coordsize).map_or((1000.0, 1000.0), |(a, b)| (size(a), size(b)));
124    let (ox, oy) = pair(coordorigin).unwrap_or((0.0, 0.0));
125    let (sx, sy) = (w as f64 / cw, h as f64 / ch);
126    let scale = |[x, y]: [f64; 2]| -> [f32; 2] { [((x - ox) * sx) as f32, ((y - oy) * sy) as f32] };
127
128    let mut d = Draw::default();
129    let (mut no_fill, mut no_stroke) = (false, false);
130    for (cmd, params) in vml_commands(path) {
131        if d.full() {
132            break;
133        }
134        let pts: Vec<[f64; 2]> = params.as_chunks::<2>().0.to_vec();
135        match cmd.as_str() {
136            "m" | "t" => {
137                for (i, &p) in pts.iter().enumerate() {
138                    let p = if cmd == "t" { d.rel(p) } else { p };
139                    if i == 0 {
140                        d.move_to(p);
141                    } else {
142                        d.line_to(p);
143                    }
144                }
145            }
146            "l" | "r" | "qx" | "qy" => {
147                for &p in &pts {
148                    let p = if cmd == "r" { d.rel(p) } else { p };
149                    d.line_to(p);
150                }
151            }
152            "c" | "v" => {
153                for c in pts.as_chunks::<3>().0 {
154                    let start = d.at;
155                    let ctrl: Vec<[f64; 2]> = std::iter::once(start).chain(c.iter().map(|&p| if cmd == "v" { d.rel(p) } else { p })).collect();
156                    for i in 1..=CURVE_STEPS {
157                        d.line_to(bezier(&ctrl, i as f64 / CURVE_STEPS as f64));
158                    }
159                }
160            }
161            "qb" => {
162                if let Some(&p) = pts.last() {
163                    d.line_to(p);
164                }
165            }
166            // Bounding box, then start and end: `at`/`wa` draw a line to the start, `ar`/`wr` move there.
167            "at" | "ar" | "wa" | "wr" => {
168                for [_, _, s, e] in pts.as_chunks::<4>().0 {
169                    if matches!(cmd.as_str(), "ar" | "wr") {
170                        d.move_to(*s);
171                    } else {
172                        d.line_to(*s);
173                    }
174                    d.line_to(*e);
175                }
176            }
177            "x" => d.end(true),
178            "e" => d.end(false),
179            "nf" => no_fill = true,
180            "ns" => no_stroke = true,
181            _ => {}
182        }
183    }
184    d.end(false);
185    let paths = d.done.into_iter().map(|(pts, closed)| FreePath { pts: pts.into_iter().map(scale).collect(), closed }).collect();
186    let freeform = Freeform { w, h, paths, alpha: 1.0, ink: None, order: None }.sanitized();
187    (!freeform.paths.is_empty()).then_some(VmlPath { freeform, no_fill, no_stroke })
188}

190/// The pen while a VML path is drawn: where it is, where the subpath began, the subpath being
191/// drawn and those done (closed or not).
192#[derive(Default)]
193struct Draw {
194    at: [f64; 2],
195    start: [f64; 2],
196    cur: Vec<[f64; 2]>,
197    done: Vec<(Vec<[f64; 2]>, bool)>,
198}
199
200impl Draw {
201    fn full(&self) -> bool {
202        self.done.len() >= MAX_PATHS
203    }
204
205    fn rel(&self, [x, y]: [f64; 2]) -> [f64; 2] {
206        [(self.at[0] + x).clamp(-MAX_VML, MAX_VML), (self.at[1] + y).clamp(-MAX_VML, MAX_VML)]
207    }
208
209    fn move_to(&mut self, p: [f64; 2]) {
210        self.end(false);
211        self.at = p;
212        self.start = p;
213        self.cur.push(p);
214    }
215
216    fn line_to(&mut self, p: [f64; 2]) {
217        if self.cur.is_empty() {
218            self.cur.push(self.at);
219            self.start = self.at;
220        }
221        if self.cur.len() < MAX_POINTS {
222            self.cur.push(p);
223        }
224        self.at = p;
225    }
226
227    /// Finish the subpath being drawn, closed back to its start or not.
228    fn end(&mut self, closed: bool) {
229        if !self.cur.is_empty() && !self.full() {
230            self.done.push((std::mem::take(&mut self.cur), closed));
231        }
232        self.cur.clear();
233        if closed {
234            self.at = self.start;
235        }
236    }
237}

239/// The commands of a VML path with their parameters: a command is a letter (or one of
240/// [`VML_TWO`]) followed by numbers separated by commas or spaces. An empty parameter (nothing
241/// before a comma) is 0, and so is a formula reference (`@n`, not expected here: see
242/// [`vml_path`]); fractions are dropped. Anything else is skipped.
243fn vml_commands(path: &str) -> Vec<(String, Vec<f64>)> {
244    let mut out: Vec<(String, Vec<f64>)> = Vec::new();
245    let mut chars = path.chars().take(MAX_VML_PATH).peekable();
246    // Whether a number was read since the command or the last comma.
247    let mut has_num = true;
248    let mut kept = 0usize;
249    let mut push = |out: &mut Vec<(String, Vec<f64>)>, v: f64| {
250        if kept < MAX_VML_PARAMS
251            && let Some((_, p)) = out.last_mut()
252        {
253            p.push(v);
254            kept += 1;
255        }
256    };
257    while let Some(c) = chars.next() {
258        if c.is_ascii_alphabetic() {
259            // Cap the commands too, so a path of bare letters can't build a million entries.
260            if out.len() >= MAX_VML_PARAMS {
261                break;
262            }
263            if !has_num {
264                push(&mut out, 0.0);
265            }
266            has_num = false;
267            let lower = c.to_ascii_lowercase();
268            let mut cmd = lower.to_string();
269            if let Some(&n) = chars.peek() {
270                let two: String = [lower, n.to_ascii_lowercase()].iter().collect();
271                if VML_TWO.contains(&two.as_str()) {
272                    chars.next();
273                    cmd = two;
274                }
275            }
276            out.push((cmd, Vec::new()));
277        } else if c == ',' {
278            if !has_num {
279                push(&mut out, 0.0);
280            }
281            has_num = false;
282        } else if matches!(c, '@' | '-' | '+' | '.') || c.is_ascii_digit() {
283            let mut v = c.to_digit(10).map_or(0.0, f64::from);
284            let mut frac = c == '.';
285            while let Some(&n) = chars.peek() {
286                if n == '.' {
287                    frac = true;
288                } else if n.is_ascii_digit() {
289                    if !frac {
290                        v = (v * 10.0 + n.to_digit(10).map_or(0.0, f64::from)).min(MAX_VML);
291                    }
292                } else {
293                    break;
294                }
295                chars.next();
296            }
297            let v = match c {
298                '@' => 0.0,
299                '-' => -v,
300                _ => v,
301            };
302            has_num = true;
303            push(&mut out, v);
304        }
305        // Whitespace and anything else only separate numbers.
306    }
307    if !has_num {
308        push(&mut out, 0.0);
309    }
310    out
311}

313/// The pen of an ink stroke written by WordCraft, from its drawing's name (`Ink Pen 3`,
314/// `Ink Highlighter 4`).
315pub fn ink_tool(name: &str) -> Option<InkTool> {
316    let mut words = name.split_whitespace();
317    if words.next() != Some("Ink") {
318        return None;
319    }
320    words.next().and_then(InkTool::parse)
321}

323/// The drawing order of an ink stroke written by WordCraft, from its drawing's name
324/// (`Ink Pen 3 #12`: the twelfth stroke drawn), for Ink Replay.
325pub fn ink_order(name: &str) -> Option<u32> {
326    name.split_whitespace().skip(3).find_map(|w| w.strip_prefix('#')?.parse().ok())
327}

329/// The outline opacity of `ln` (`a:ln`): its colour's `a:alpha`, 0–1.
330pub fn line_alpha(ln: Option<&El>) -> f32 {
331    ln.and_then(|l| l.child("a:solidFill"))
332        .and_then(|f| f.els().next())
333        .and_then(|c| c.child("a:alpha"))
334        .and_then(|a| a.attr("val"))
335        .and_then(int)
336        .map_or(1.0, |v| (v.clamp(0, 100_000) as f32) / 100_000.0)
337}

339/// The point at `t` on the Bézier curve with control points `c` (de Casteljau).
340fn bezier(c: &[[f64; 2]], t: f64) -> [f64; 2] {
341    let mut p: Vec<[f64; 2]> = c.to_vec();
342    while p.len() > 1 {
343        p = p
344            .windows(2)
345            .filter_map(|w| match w {
346                [a, b] => Some([a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t]),
347                _ => None,
348            })
349            .collect();
350    }
351    p.first().copied().unwrap_or([0.0, 0.0])
352}

359        let xml = format!(
360            "<wps:spPr xmlns:a=\"http://schemas.openxmlformats.org/drawingml/2006/main\" \

363             <a:moveTo><a:pt x=\"0\" y=\"0\"/></a:moveTo>{seg}</a:path></a:pathLst></a:custGeom></wps:spPr>"

368    #[test]
369    fn a_vml_path_of_bare_letters_builds_a_bounded_command_list() {
370        let path = "x".repeat(MAX_VML_PATH);
371        let n = vml_commands(&path).len();
372        assert!(n <= MAX_VML_PARAMS, "{n} commands");

375    #[test]
376    fn a_bezier_segment_with_many_points_is_bounded() {
377        // One cubicBezTo stuffed with 50 000 points must not make de Casteljau quadratic: the
378        // control polygon is capped, so flattening stays a handful of points and returns at once.
379        let pts = "<a:pt x=\"1\" y=\"1\"/>".repeat(50_000);
380        let sppr = sppr(&format!("<a:cubicBezTo>{pts}</a:cubicBezTo>"));
381        let f = cust_geom(&sppr, 100.0, 100.0).expect("a path");
382        assert!(f.paths.iter().all(|p| p.pts.len() <= 1 + CURVE_STEPS), "{:?}", f.paths.iter().map(|p| p.pts.len()).collect::<Vec<_>>());
383    }
384
385    #[test]
386    fn a_conformant_cubic_still_flattens_to_a_curve() {
387        let sppr = sppr("<a:cubicBezTo><a:pt x=\"0\" y=\"100\"/><a:pt x=\"100\" y=\"100\"/><a:pt x=\"100\" y=\"0\"/></a:cubicBezTo>");
388        let f = cust_geom(&sppr, 100.0, 100.0).expect("a path");
389        assert_eq!(f.paths.len(), 1);
390        assert!(f.paths[0].pts.len() >= 2 && f.paths[0].pts.len() <= 1 + CURVE_STEPS);
391    }
