86/// The last paragraph of a block list: its last block, or inside a final table the last
87/// paragraph of its last cell (nested tables to [`MAX_TABLE_DEPTH`]).
88fn last_para_mut(out: &mut Blocks, depth: usize) -> Option<&mut Paragraph> {
89    match Arc::make_mut(out.last_mut()?) {
90        Block::Para(p) => Some(p),
91        Block::Table(t) if depth < MAX_TABLE_DEPTH => last_para_mut(&mut t.rows.last_mut()?.cells.last_mut()?.blocks, depth + 1),
92        Block::Table(_) => None,
93    }
94}

96/// The field that buffers content now: the innermost one that isn't a range field, unless it
97/// has spilled (then content goes straight into the paragraph).
98fn sink(sc: &mut StoryCtx) -> Option<&mut FieldState> {
99    match sc.fields.iter_mut().rev().find(|f| !f.range) {
100        Some(f) if !f.spilled => Some(f),
101        _ => None,
102    }
103}

118/// Paragraph content under construction.
119#[derive(Default)]
120struct PB {
121    text: String,
122    runs: Vec<Run>,
123    objects: Vec<InlineObject>,
124}
125
126impl PB {
127    fn push_text(&mut self, s: &str, props: &CharProps) {
128        if s.is_empty() {
129            return;
130        }
131        self.text.push_str(s);
132        self.push_run(s.len(), props);
133    }
134    fn push_obj(&mut self, o: InlineObject, props: &CharProps) {
135        self.text.push(crate::doc::para::OBJ);
136        self.objects.push(o);
137        self.push_run(crate::doc::para::OBJ.len_utf8(), props);
138    }
139    fn push_run(&mut self, len: usize, props: &CharProps) {
140        match self.runs.last_mut() {
141            Some(r) if r.props == *props => r.len += len,
142            _ => self.runs.push(Run { len, props: props.clone() }),
143        }
144    }
145}

147/// Clean text from `w:t`: drop characters with special meaning in the model and control
148/// characters XML producers shouldn't have put there.
149fn clean_text(s: &str) -> String {
150    s.chars()
151        .filter_map(|c| match c {
152            '\n' | '\r' => Some(' '),
153            '\t' => Some('\t'),
154            crate::doc::para::OBJ => None,
155            c if (c as u32) < 0x20 => None,
156            c => Some(c),
157        })
158        .collect()
159}

161fn children_of_choice(ac: &El) -> Option<&El> {
162    // Prefer a Choice whose requirements we understand; else the Fallback; else any Choice.
163    // `m`: Office Math (a producer may guard an equation with a picture fallback, #1273).
164    const KNOWN: &[&str] = &["wps", "wpg", "wpc", "wpi", "w14", "w15", "wp14", "a14", "w16se", "w16cid", "w16", "w16cex", "w16sdtdh", "v", "m"];
165    let ok = |c: &El| c.attr("Requires").unwrap_or("").split_whitespace().all(|r| KNOWN.contains(&r));
166    ac.children("mc:Choice").find(|c| ok(c)).or_else(|| ac.child("mc:Fallback")).or_else(|| ac.child("mc:Choice"))
167}

169/// Gather `name` children (rows, cells), looking through content controls, custom XML and the
170/// selected alternate-content branch; each with the content controls that open before it and how
171/// many close after it. `pending` holds controls opened but not yet given to an element.
172fn collect<'a>(e: &'a El, name: &str, out: &mut Vec<(&'a El, ControlWrap)>, pending: &mut Vec<ContentControl>, depth: usize) {
173    if depth > 8 {
174        return;
175    }
176    for k in e.els() {
177        if k.name == name {
178            out.push((k, ControlWrap { open: std::mem::take(pending), close: 0 }));
179        } else if k.name == "w:sdt" {
180            if let Some(c) = k.child("w:sdtContent") {
181                let mut ctl = super::sdt::sdt_pr(k.child("w:sdtPr"), true);
182                ctl.end_pr_xml = super::sdt::sdt_end_pr(k.child("w:sdtEndPr"));
183                let before = out.len();
184                pending.push(ctl);
185                collect(c, name, out, pending, depth + 1);
186                let grew = out.len() > before;
187                match out.last_mut() {
188                    Some((_, w)) if grew => w.close = w.close.saturating_add(1),
189                    // Nothing inside: the control goes (as before).
190                    _ => {
191                        pending.pop();
192                    }
193                }
194            }
195        } else if k.name == "w:customXml" {
196            collect(k, name, out, pending, depth + 1);
197        } else if k.name == "mc:AlternateContent"
198            && let Some(branch) = children_of_choice(k)
199        {
200            collect(branch, name, out, pending, depth + 1);
201        }
202    }
203}

205/// Placeholder text as read from a control showing it: one line, bounded.
206fn placeholder_of(s: &str) -> String {
207    s.chars().map(|c| if c.is_control() { ' ' } else { c }).take(crate::doc::control::MAX_CONTROL_TEXT).collect::<String>().trim().to_string()
