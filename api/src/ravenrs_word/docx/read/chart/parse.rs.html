7    pub(super) fn parse(space: &El, theme: &[Rgb]) -> Option<Chart> {
8        let chart = space.child("c:chart")?;
9        let plot = chart.child("c:plotArea")?;
10        let ty = plot.els().find(|e| Kind::of(&e.name).is_some())?;
11        let kind = Kind::of(&ty.name)?;
12        let scatter = kind == Kind::Scatter;
13        let date1904 = space.child("c:date1904").is_some_and(|d| d.attr("val").is_none_or(truthy));
14        let mut series = read_series(ty, scatter, theme);
15        if series.is_empty() {
16            return None;
17        }
18        let cats = ty.children("c:ser").find_map(|s| s.child("c:cat")).map(|c| labels(Some(c), MAX_POINTS, date1904)).unwrap_or_default();
19        let n = if kind == Kind::Radar {
20            series.iter().map(|s| s.values.len()).max().unwrap_or(0).max(cats.len()).min(MAX_SPOKES)
21        } else if matches!(kind, Kind::Pie | Kind::Doughnut) {
22            series.first().map_or(0, |s| s.values.len()).max(cats.len())
23        } else {
24            series.iter().map(|s| s.values.len()).max().unwrap_or(0).max(cats.len())
25        };
26        // Every series is padded to `n` points, so keep only as many as the cell budget holds.
27        if let Some(keep) = MAX_CELLS.checked_div(n) {
28            series.truncate(keep.max(1));
29        }
30        if kind == Kind::Radar {
31            series.truncate(MAX_RADAR_SERIES);
32        }
33        let grouping = match ty.child("c:grouping").and_then(|g| g.attr("val")) {
34            Some("stacked") if matches!(kind, Kind::Bar | Kind::Area | Kind::Line) => Grouping::Stacked,
35            Some("percentStacked") if matches!(kind, Kind::Bar | Kind::Area | Kind::Line) => Grouping::Percent,
36            _ => Grouping::Clustered,
37        };
38        let mut cats = cats;
39        cats.resize(n, String::new());
40        let radar_style = ty.child("c:radarStyle").and_then(|r| r.attr("val"));
41        let type_labels = show_values(ty.child("c:dLbls"));
42        for s in &mut series {
43            s.values.resize(n, None);
44            s.xs.resize(n, None);
45            s.labels = Some(s.labels.or(type_labels).unwrap_or(false));
46            if s.labels == Some(true) {
47                s.raw = s.values.clone();
48            }
49        }
50        if grouping == Grouping::Percent {
51            to_percent(&mut series, n);
52        }
53        let (cat, val) = chart_axes(plot, scatter);
54        let val_fmt = if grouping == Grouping::Percent {
55            "0%".to_string()
56        } else {
57            val.fmt.clone().or_else(|| series.iter().find_map(|s| s.fmt.clone())).unwrap_or_else(|| "General".to_string())
58        };
59        let num = |n: &str| ty.child(n).and_then(|g| g.attr("val")).and_then(int);
60        Some(Chart {
61            kind,
62            horizontal: kind == Kind::Bar && ty.child("c:barDir").and_then(|d| d.attr("val")) == Some("bar"),
63            grouping,
64            gap: num("c:gapWidth").unwrap_or(150).clamp(0, 500) as f32 / 100.0,
65            overlap: num("c:overlap").unwrap_or(0).clamp(-100, 100) as f32 / 100.0,
66            hole: num("c:holeSize").unwrap_or(50).clamp(10, 90) as f32 / 100.0,
67            first_angle: num("c:firstSliceAng").unwrap_or(0).clamp(0, 360) as f64,
68            radar_fill: radar_style == Some("filled"),
69            radar_marks: radar_style == Some("marker"),
70            vary: matches!(kind, Kind::Pie | Kind::Doughnut) && ty.child("c:varyColors").and_then(|v| v.attr("val")).is_none_or(truthy),
71            n,
72            cats,
73            title: title_of(chart, &series),
74            series,
75            accents: (4..10).map(|i| slot(theme, i)).collect(),
76            legend: legend_pos(chart),
77            cat,
78            val,
79            text_size: text_size(space),
80            val_fmt,
81            date1904,
82            plot_border: plot.child("c:spPr").and_then(|p| p.child("a:ln")).and_then(|l| fill_colour(l, theme)),
83        })
84    }

94    /// Colour of series `j`: its own, else the accent its `c:idx` picks (as Word does; `c:order`
95    /// doesn't matter), else the one its place in the file picks.
96    pub(super) fn series_colour(&self, j: usize) -> Rgb {
97        let s = self.series.get(j);
98        s.and_then(|s| s.color).unwrap_or_else(|| self.accent(s.and_then(|s| s.idx).unwrap_or(j)))
99    }
100
101    /// Colour of point `i` of series `j`: its `c:dPt` override, else the varied accent (pie) or
102    /// the series colour.
103    pub(super) fn point_colour(&self, j: usize, i: usize) -> Rgb {
104        if let Some(Some(c)) = self.series.get(j).and_then(|s| s.points.get(i)) {
105            return *c;
106        }
107        if self.vary { self.accent(i) } else { self.series_colour(j) }
108    }
109
110    pub(super) fn legend_entries(&self) -> Vec<(String, Rgb)> {
111        if matches!(self.kind, Kind::Pie | Kind::Doughnut) {
112            return self
113                .cats
114                .iter()
115                .enumerate()
116                .map(|(i, c)| (if c.is_empty() { (i + 1).to_string() } else { c.clone() }, self.point_colour(0, i)))
117                .collect();
118        }
119        self.series
120            .iter()
121            .enumerate()
122            .map(|(j, s)| (if s.name.is_empty() { format!("Series {}", j + 1) } else { s.name.clone() }, self.series_colour(j)))
123            .collect()
124    }

127/// The series of a chart type element, within the chart's cell budget.
128pub(super) fn read_series(ty: &El, scatter: bool, theme: &[Rgb]) -> Vec<Series> {
129    let mut budget = MAX_CELLS;
130    let mut series = Vec::new();
131    for ser in ty.children("c:ser").take(MAX_SERIES) {
132        let s = Series::parse(ser, scatter, theme, budget.min(MAX_POINTS));
133        budget = budget.saturating_sub(s.values.len().max(s.xs.len()));
134        series.push(s);
135    }
136    series
137}

139/// Values as their share of each category's total (100% stacked charts).
140fn to_percent(series: &mut [Series], n: usize) {
141    for i in 0..n {
142        let total: f64 = series.iter().map(|s| s.values.get(i).copied().flatten().map_or(0.0, f64::abs)).sum();
143        for s in series.iter_mut() {
144            if let Some(Some(v)) = s.values.get_mut(i) {
145                *v = if total > 0.0 { *v / total } else { 0.0 };
146            }
147        }
148    }
149}

151/// The category axis (scatter: the horizontal value axis) and the value axis of a plot area.
152fn chart_axes(plot: &El, scatter: bool) -> (Axis, Axis) {
153    let vals_ax: Vec<&El> = plot.children("c:valAx").collect();
154    let horiz_pos = |a: &&El| matches!(a.child("c:axPos").and_then(|p| p.attr("val")), Some("b" | "t"));
155    let (val_el, x_el) = if scatter {
156        let x = vals_ax.iter().copied().find(horiz_pos).or_else(|| vals_ax.first().copied());
157        let y = vals_ax.iter().copied().find(|a| !horiz_pos(a)).or_else(|| vals_ax.iter().copied().find(|a| Some(*a) != x));
158        (y, x)
159    } else {
160        (vals_ax.first().copied(), None)
161    };
162    let cat_el = plot.els().find(|e| matches!(e.name.as_str(), "c:catAx" | "c:dateAx"));
163    (axis(if scatter { x_el } else { cat_el }), axis(val_el))
164}

166/// The chart's title. An empty `c:title` shows the series name (Word's automatic title of a
167/// one-series chart) unless the automatic title is deleted.
168pub(super) fn title_of(chart: &El, series: &[Series]) -> Option<String> {
169    let auto_deleted = chart.child("c:autoTitleDeleted").and_then(|a| a.attr("val")).is_some_and(truthy);
170    let auto = if !auto_deleted && series.len() == 1 { series.first().map(|s| s.name.clone()).filter(|s| !s.is_empty()) } else { None };
171    chart.child("c:title").and_then(|t| title_text(t).or(auto))
172}

174/// Whether a `c:dLbls` shows the values (`None` when there's none, or it doesn't say).

202impl Series {
203    fn parse(ser: &El, scatter: bool, theme: &[Rgb], cap: usize) -> Series {
204        let name = ser.child("c:tx").and_then(|t| t.find("c:v")).map(|v| label(v.text().trim())).unwrap_or_default();
205        let idx = ser.child("c:idx").and_then(|e| e.attr("val")).and_then(|v| v.trim().parse::<u32>().ok()).and_then(|v| usize::try_from(v).ok());
206        let sp = ser.child("c:spPr");
207        let ln = sp.and_then(|p| p.child("a:ln"));
208        let marker = ser.child("c:marker").and_then(|m| m.child("c:spPr")).and_then(|p| fill_colour(p, theme));
209        let color = sp.and_then(|p| fill_colour(p, theme)).or_else(|| ln.and_then(|l| fill_colour(l, theme))).or(marker);
210        let line_w = ln.and_then(|l| l.attr("w")).and_then(|v| measure(v, 12_700.0)).map_or(LINE_W, |w| w.clamp(0.0, 20.0));
211        let line = ln.is_none_or(|l| l.child("a:noFill").is_none());
212        let (values, fmt) = numbers(ser.child(if scatter { "c:yVal" } else { "c:val" }), cap);
213        let (mut xs, x_fmt) = if scatter { numbers(ser.child("c:xVal"), cap) } else { (Vec::new(), None) };
214        if scatter && xs.is_empty() {
215            xs = (1..=values.len()).map(|i| Some(i as f64)).collect();
216        }
217        let mut points = vec![None; values.len()];
218        for dpt in ser.children("c:dPt") {
219            let i = dpt.child("c:idx").and_then(|e| e.attr("val")).and_then(int).and_then(|i| usize::try_from(i).ok());
220            let c = dpt.child("c:spPr").and_then(|p| fill_colour(p, theme));
221            if let Some(slot) = i.and_then(|i| points.get_mut(i)) {
222                *slot = c;
223            }
224        }
225        let labels = show_values(ser.child("c:dLbls"));
226        Series { name, idx, color, line_w, line, values, xs, points, fmt, x_fmt, labels, raw: Vec::new() }
227    }
228}

230/// Longest number-format code honoured (the limit spreadsheet apps use); longer ones are ignored.
231const MAX_FORMAT_CODE: usize = 255;
232
233/// Category labels of a `c:cat` (string or number cache), capped. Numbers show in the cache's
234/// format code (dates included); text and `General` numbers as they are stored.
235pub(super) fn labels(el: Option<&El>, cap: usize, date1904: bool) -> Vec<String> {
236    let Some(el) = el else { return Vec::new() };
237    let Some(cache) = el.find("c:strCache").or_else(|| el.find("c:numCache")).or_else(|| el.find("c:strLit")).or_else(|| el.find("c:numLit")) else {
238        return Vec::new();
239    };
240    let code =
241        cache.child("c:formatCode").map(|f| f.text()).filter(|f| !f.is_empty() && f.len() <= MAX_FORMAT_CODE && !f.eq_ignore_ascii_case("General"));
242    pts(cache, cap)
243        .into_iter()
244        .map(|p| {
245            let text = p.unwrap_or_default();
246            match (code.as_deref(), text.trim().parse::<f64>()) {
247                (Some(code), Ok(v)) if v.is_finite() => label(&category_text(v, code, 1.0, date1904)),
248                _ => label(&text),
249            }
250        })
251        .collect()
252}

254/// A number in its format code: a date or time for date codes (serial days, from 1904 when
255/// `date1904`), else as [`format_value`] with tick step `step`.
256pub(super) fn category_text(v: f64, code: &str, step: f64, date1904: bool) -> String {
257    match excel_stamp(v, date1904) {
258        Some(t) if is_date_code(code) => date_text(t, code),
259        _ => format_value(v, code, step),
260    }
261}

263/// Values of a `c:val`-style element (number cache or literal), capped; with its format code.
264pub(super) fn numbers(el: Option<&El>, cap: usize) -> (Vec<Option<f64>>, Option<String>) {
265    let Some(cache) = el.and_then(|e| e.find("c:numCache").or_else(|| e.find("c:numLit"))) else {
266        return (Vec::new(), None);
267    };
268    let fmt = cache.child("c:formatCode").map(|f| f.text()).filter(|f| !f.is_empty() && f.len() <= MAX_FORMAT_CODE);
269    let vals = pts(cache, cap).into_iter().map(|p| p.and_then(|t| t.trim().parse::<f64>().ok()).filter(|v| v.is_finite())).collect();
270    (vals, fmt)
271}

273/// The `c:pt` texts of a cache by index. Indexes and counts from the file are capped.
274fn pts(cache: &El, cap: usize) -> Vec<Option<String>> {
275    let declared = cache.child("c:ptCount").and_then(|c| c.attr("val")).and_then(int).unwrap_or(0).clamp(0, cap as i64) as usize;
276    let mut out: Vec<Option<String>> = vec![None; declared];
277    for p in cache.children("c:pt") {
278        let Some(i) = p.attr("idx").and_then(int).and_then(|i| usize::try_from(i).ok()) else { continue };
279        if i >= cap {
280            continue;
281        }
282        if out.len() <= i {
283            out.resize(i + 1, None);
284        }
285        if let Some(slot) = out.get_mut(i) {
286            *slot = p.child("c:v").map(|v| v.text());
287        }
288    }
289    out
290}

292pub(super) fn axis(a: Option<&El>) -> Axis {
293    let Some(a) = a else { return Axis { deleted: true, ..Default::default() } };
294    let scaling = a.child("c:scaling");
295    let num =
296        |n: &str| scaling.and_then(|s| s.child(n)).and_then(|e| e.attr("val")).and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite());
297    Axis {
298        deleted: a.child("c:delete").and_then(|d| d.attr("val")).is_some_and(truthy),
299        grid: a.child("c:majorGridlines").is_some(),
300        line: a.child("c:spPr").and_then(|p| p.child("a:ln")).and_then(|l| l.child("a:noFill")).is_none(),
301        mid: a.child("c:crossBetween").and_then(|c| c.attr("val")) == Some("midCat"),
302        min: num("c:min"),
303        max: num("c:max"),
304        fmt: a
305            .child("c:numFmt")
306            .filter(|f| f.attr("sourceLinked") != Some("1"))
307            .and_then(|f| f.attr("formatCode"))
308            .filter(|f| f.len() <= MAX_FORMAT_CODE)
309            .map(str::to_string),
310    }
311}

313fn title_text(t: &El) -> Option<String> {
314    let tx = t.child("c:tx")?;
315    let s = match tx.child("c:rich") {
316        Some(rich) => {
317            let mut s = String::new();
318            rich_into(rich, &mut s);
319            s
320        }
321        None => tx.find("c:v").map(|v| v.text()).unwrap_or_default(),
322    };
323    let s = label(s.trim());
324    (!s.is_empty()).then_some(s)
325}

327/// The `a:t` runs under an element, in order.
328fn rich_into(e: &El, s: &mut String) {
329    for c in e.els() {
330        if c.name == "a:t" {
331            s.push_str(&c.text());
332        } else {
333            rich_into(c, s);
334        }
335    }
