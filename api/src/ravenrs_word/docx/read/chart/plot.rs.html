12/// Chart area: the chart space's fill and border (white and borderless by default).
13pub(super) fn backdrop(out: &mut Vec<GraphicItem>, space: &El, theme: &[Rgb], w: f32, h: f32) {
14    let sp = space.child("c:spPr");
15    let fill = match sp {
16        Some(p) if p.child("a:noFill").is_some() => None,
17        Some(p) => fill_colour(p, theme).or(Some(Rgb::WHITE)),
18        None => Some(Rgb::WHITE),
19    };
20    let ln = sp.and_then(|p| p.child("a:ln"));
21    let stroke = ln.and_then(|l| fill_colour(l, theme));
22    let stroke_width = ln.and_then(|l| l.attr("w")).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.75).clamp(0.0, 20.0);
23    if fill.is_some() || stroke.is_some() {
24        out.push(GraphicItem::Shape { rect: [0.0, 0.0, w, h], kind: ShapeKind::Rectangle, fill, stroke, stroke_width });
25    }
26}

28/// An axis scale: `lo` to `hi` in steps of `step`.
29#[derive(Clone, Copy, Debug, PartialEq)]
30pub(super) struct Scale {
31    pub(super) lo: f64,
32    pub(super) hi: f64,
33    pub(super) step: f64,
34}
35
36impl Scale {
37    pub(super) fn ticks(&self) -> Vec<f64> {
38        let n = (((self.hi - self.lo) / self.step + 1e-9).floor().max(0.0) as usize).min(MAX_TICKS);
39        (0..=n).map(|k| self.lo + k as f64 * self.step).filter(|t| *t <= self.hi + 1e-9).collect()
40    }
41}

43/// A scale for data `dlo..dhi` (zero included when `zero`), as Excel and Word pick it: an end
44/// away from zero sits 5% of the data span beyond the data (an end at zero stays there), rounded
45/// out to a step of 1, 2, 2.5, 5 or 10 times a power of ten (about seven steps). `min`/`max`
46/// from the file replace the ends.
47pub(super) fn nice_scale(dlo: f64, dhi: f64, zero: bool, min: Option<f64>, max: Option<f64>) -> Scale {
48    let (dlo, dhi) = if dhi > dlo { (dlo, dhi) } else { (dlo, dlo + 1.0) };
49    let span = dhi - dlo;
50    // With zero, the ends reach it: all-negative data still starts its bars at zero.
51    let (ends_lo, ends_hi) = if zero { (dlo.min(0.0), dhi.max(0.0)) } else { (dlo, dhi) };
52    let lo = min.unwrap_or(if ends_lo < 0.0 { ends_lo - 0.05 * span } else { ends_lo });
53    let hi = max.unwrap_or(if ends_hi > 0.0 { ends_hi + 0.05 * span } else { ends_hi });
54    if !(hi > lo && hi.is_finite() && lo.is_finite()) {
55        return Scale { lo: 0.0, hi: 1.0, step: 0.2 };
56    }
57    let raw = (hi - lo) / 7.5;
58    let mag = 10f64.powf(raw.log10().floor());
59    let step = [1.0, 2.0, 2.5, 5.0, 10.0].iter().map(|m| m * mag).find(|s| *s >= raw).unwrap_or(10.0 * mag);
60    let lo_r = if min.is_some() { lo } else { ((lo / step) + 1e-9).floor() * step };
61    let hi_r = if max.is_some() { hi } else { ((hi / step) - 1e-9).ceil() * step };
62    if !(step.is_finite() && step > 0.0 && lo_r.is_finite() && hi_r.is_finite() && hi_r > lo_r) {
63        return Scale { lo: 0.0, hi: 1.0, step: 0.2 };
64    }
65    Scale { lo: lo_r, hi: hi_r, step }
66}

68/// The plot rectangle and the scales that map values and categories onto it.
69struct Frame {
70    plot: Rect,
71    horiz: bool,
72    n: usize,
73    vs: Scale,
74    xs: Option<Scale>,
75    mid: bool,
76}
77
78impl Frame {
79    /// Width of one category band.
80    fn band(&self) -> f32 {
81        let len = if self.horiz { self.plot.h } else { self.plot.w };
82        len / self.n.max(1) as f32
83    }
84
85    /// Start of category `i`'s band: its left edge, or for bars its top edge (category 0 at the
86    /// bottom).
87    fn band_start(&self, i: usize) -> f32 {
88        let k = i as f32;
89        if self.horiz { self.plot.bottom() - (k + 1.0) * self.band() } else { self.plot.x + k * self.band() }
90    }
91
92    /// A value's position along the value direction: y for columns, x for bars.
93    fn val(&self, v: f64) -> f32 {
94        let t = ((v - self.vs.lo) / (self.vs.hi - self.vs.lo)).clamp(0.0, 1.0) as f32;
95        if self.horiz { self.plot.x + t * self.plot.w } else { self.plot.bottom() - t * self.plot.h }
96    }
97
98    /// x of category `i` for line and area points.
99    fn cat_x(&self, i: usize) -> f32 {
100        if !self.mid {
101            return self.plot.x + (i as f32 + 0.5) * self.band();
102        }
103        if self.n <= 1 { self.plot.x + self.plot.w / 2.0 } else { self.plot.x + self.plot.w * i as f32 / (self.n - 1) as f32 }
104    }
105
106    /// x of a scatter point.
107    fn scat_x(&self, x: f64) -> f32 {
108        match self.xs {
109            Some(s) => self.plot.x + ((x - s.lo) / (s.hi - s.lo)).clamp(0.0, 1.0) as f32 * self.plot.w,
110            None => self.plot.x,
111        }
112    }
113}

116pub(super) struct Axes {

123impl Axes {
124    pub(super) fn of(c: &Chart) -> Axes {
125        let (dlo, dhi) = extent(c);
126        let (vmin, vmax) = if c.grouping == Grouping::Percent { (Some(0.0), Some(1.0)) } else { (c.val.min, c.val.max) };
127        // Lines and scatters don't need zero for a narrow range (1000..1050 stays on its own scale).
128        let zero = !matches!(c.kind, Kind::Line | Kind::Scatter) || !narrow(dlo, dhi);
129        let vs = nice_scale(dlo, dhi, zero, vmin, vmax);
130        let vticks = vs.ticks().into_iter().map(|t| (t, format_value(t, &c.val_fmt, vs.step))).collect();
131        let xs = (c.kind == Kind::Scatter).then(|| {
132            let (xl, xh) = x_extent(c);
133            nice_scale(xl, xh, false, c.cat.min, c.cat.max)
134        });
135        // Scatter x values show in the axis's own format, else their cache's (dates included).
136        let xfmt = c.cat.fmt.clone().or_else(|| c.series.iter().find_map(|s| s.x_fmt.clone())).unwrap_or_else(|| "General".to_string());
137        let xticks = xs.map(|s| s.ticks().into_iter().map(|t| (t, category_text(t, &xfmt, s.step, c.date1904))).collect()).unwrap_or_default();
138        Axes { vs, vticks, xs, xticks }
139    }
140}

142/// The room around the plot for axis labels, and the category labels as they'll be drawn.
143struct Margins {
144    left: f32,
145    top: f32,
146    bottom: f32,
147    /// Category labels in lines (none for scatter charts or a deleted category axis).
148    wrapped: Vec<Vec<String>>,
149    /// Every `skip`-th category label is shown under a column chart.
150    skip: usize,
151}

153/// The margins the axis labels need around the plot in `area`. Category labels under a column
154/// chart wrap to the band; labels that still don't fit show every k-th category.
155fn axis_margins(c: &Chart, ax: &Axes, area: Rect) -> Margins {
156    let fs = c.text_size;
157    let lh = fs * LINE;
158    let scatter = c.kind == Kind::Scatter;
159    let (cat_on, val_on) = (!c.cat.deleted, !c.val.deleted);
160    let left = if c.horizontal {
161        if cat_on { widest(&c.cats, fs).min(area.w * 0.4) + GAP } else { 0.0 }
162    } else if val_on {
163        ax.vticks.iter().map(|(_, s)| text_w(s, fs)).fold(0.0, f32::max) + GAP
164    } else {
165        0.0
166    };
167    let band_w = (area.w - left).max(1.0) / c.n as f32;
168    let wrapped: Vec<Vec<String>> = if cat_on && !scatter {
169        let width = if c.horizontal { left - GAP } else { band_w };
170        c.cats.iter().map(|s| wrap(s, width, fs)).collect()
171    } else {
172        Vec::new()
173    };
174    let widest_line = wrapped.iter().flatten().map(|s| text_w(s, fs)).fold(0.0, f32::max);
175    let skip = ((widest_line / band_w).ceil() as usize).max(1);
176    let label_lines = wrapped.iter().map(Vec::len).max().unwrap_or(1) as f32;
177    let bottom = if cat_on && !c.horizontal {
178        if scatter { lh + GAP } else { label_lines * lh + GAP }
179    } else if c.horizontal && val_on {
180        lh + GAP
181    } else {
182        0.0
183    };
184    let top = if !c.horizontal && val_on { lh / 2.0 } else { 0.0 };
185    Margins { left, top, bottom, wrapped, skip }
186}

188/// A bar, column, line, area or scatter chart in `area`: gridlines, series, axes, then labels.
189pub(super) fn cartesian(out: &mut Vec<GraphicItem>, c: &Chart, area: Rect) {
190    if c.n == 0 {
191        return;
192    }
193    let ax = Axes::of(c);
194    let m = axis_margins(c, &ax, area);
195    let plot = Rect { x: area.x + m.left, y: area.y + m.top, w: area.w - m.left, h: area.h - m.top - m.bottom };
196    if plot.w < 1.0 || plot.h < 1.0 {
197        return;
198    }
199    let g = Frame { plot, horiz: c.horizontal, n: c.n, vs: ax.vs, xs: ax.xs, mid: c.val.mid };
200    if c.val.grid {
201        draw_grid(out, &g, &ax.vticks);
202    }
203    let mut labels = Vec::new();
204    match c.kind {
205        Kind::Bar => bars(out, c, &g, &mut labels),
206        Kind::Line => lines(out, c, &g, &mut labels),
207        Kind::Area => areas(out, c, &g, &mut labels),
208        Kind::Scatter => draw_scatter(out, c, &g, &mut labels),
209        Kind::Pie | Kind::Doughnut | Kind::Radar => {}
210    }
211    draw_axes(out, c, &g);
212    draw_data_labels(out, &labels, c.text_size, g.horiz && c.kind == Kind::Bar);
213    if !c.val.deleted {
214        draw_value_labels(out, c, &g, &ax.vticks, Rect { w: m.left, ..area });
215    }
216    if !c.cat.deleted {
217        draw_category_labels(out, c, &g, &ax.xticks, &m, area);
218    }
219}

247/// Value tick labels: under the plot for bars, else right-aligned in `margin` (the area's left
248/// margin).
249fn draw_value_labels(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, ticks: &[(f64, String)], margin: Rect) {
250    let (fs, lh) = (c.text_size, c.text_size * LINE);
251    for (t, s) in ticks {
252        let p = g.val(*t);
253        if g.horiz {
254            text_at(out, p, g.plot.bottom() + GAP / 2.0, s, fs);
255        } else {
256            text(out, Rect { x: margin.x, y: p - lh / 2.0, w: margin.w - GAP, h: lh }, s, fs, TextAlign::Right, false);
257        }
258    }

261/// Category labels: scatter x ticks under the plot, bar labels in the left margin, column
262/// labels under their bands (every `m.skip`-th).
263fn draw_category_labels(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, xticks: &[(f64, String)], m: &Margins, area: Rect) {
264    let (fs, lh) = (c.text_size, c.text_size * LINE);
265    let plot = g.plot;
266    if c.kind == Kind::Scatter {
267        for (t, s) in xticks {
268            text_at(out, g.scat_x(*t), plot.bottom() + GAP / 2.0, s, fs);
269        }
270    } else if g.horiz {
271        for (i, lines) in m.wrapped.iter().enumerate() {
272            let top = g.band_start(i) + g.band() / 2.0 - lines.len() as f32 * lh / 2.0;
273            for (k, s) in lines.iter().enumerate() {
274                text(out, Rect { x: area.x, y: top + k as f32 * lh, w: m.left - GAP, h: lh }, s, fs, TextAlign::Right, false);
275            }
276        }
277    } else {
278        let band = g.band();
279        for (i, lines) in m.wrapped.iter().enumerate().filter(|(i, _)| i % m.skip == 0) {
280            for (k, s) in lines.iter().enumerate() {
281                let y = plot.bottom() + GAP / 2.0 + k as f32 * lh;
282                text(out, Rect { x: g.cat_x(i) - band / 2.0, y, w: band, h: lh }, s, fs, TextAlign::Center, false);
283            }
284        }
285    }
286}

288/// Whether data `lo..hi` is a narrow band away from zero: its span is under a sixth of the end
289/// farther from zero (the mirror image for negative data).
290pub(super) fn narrow(lo: f64, hi: f64) -> bool {
291    if lo > 0.0 {
292        (hi - lo) / hi < 1.0 / 6.0
293    } else if hi < 0.0 {
294        (hi - lo) / lo.abs() < 1.0 / 6.0
295    } else {
296        false
297    }
298}

300/// Lowest and highest value of the plot; stacked series add up per category.
301fn extent(c: &Chart) -> (f64, f64) {
302    let mut pts: Vec<f64> = Vec::new();
303    if c.stacked() {
304        for i in 0..c.n {
305            let (mut p, mut q) = (0.0, 0.0);
306            for s in &c.series {
307                match s.values.get(i).copied().flatten() {
308                    Some(v) if v >= 0.0 => p += v,
309                    Some(v) => q += v,
310                    None => {}
311                }
312            }
313            pts.push(p);
314            pts.push(q);
315        }
316    } else {
317        pts.extend(c.series.iter().flat_map(|s| s.values.iter().flatten().copied()));
318    }
319    let lo = pts.iter().copied().fold(f64::INFINITY, f64::min);
320    let hi = pts.iter().copied().fold(f64::NEG_INFINITY, f64::max);
321    if lo.is_finite() && hi.is_finite() { (lo, hi) } else { (0.0, 0.0) }
322}

324/// Range of scatter x values (index + 1 where missing).
325fn x_extent(c: &Chart) -> (f64, f64) {
326    let xs: Vec<f64> = c.series.iter().flat_map(|s| s.xs.iter().flatten().copied()).collect();
327    let lo = xs.iter().copied().fold(f64::INFINITY, f64::min);
328    let hi = xs.iter().copied().fold(f64::NEG_INFINITY, f64::max);
329    if lo.is_finite() && hi.is_finite() { (lo, hi) } else { (0.0, 1.0) }
330}

332fn bars(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, labels: &mut Vec<DataLabel>) {
333    let stacked = c.stacked();
334    let slots = if stacked { 1 } else { c.series.len().max(1) };
335    let ov = if stacked { 0.0 } else { c.overlap };
336    let ns = slots as f32;
337    let band = g.band();
338    let thick = band / (ns - (ns - 1.0) * ov + c.gap);
339    let group = thick * (ns - (ns - 1.0) * ov);
340    let mut pos = vec![0.0f64; g.n];
341    let mut neg = vec![0.0f64; g.n];
342    for (j, s) in c.series.iter().enumerate().filter(|(_, s)| has_values(s)) {
343        let slot = if stacked { 0 } else { j };
344        let slot = (if g.horiz { slots.saturating_sub(1 + slot) } else { slot }) as f32;
345        for i in 0..g.n {
346            let Some(v) = s.values.get(i).copied().flatten() else { continue };
347            let (v0, v1) = if !stacked {
348                (0.0, v)
349            } else if v >= 0.0 {
350                let Some(p) = pos.get_mut(i) else { continue };
351                let b = *p;
352                *p += v;
353                (b, *p)
354            } else {
355                let Some(q) = neg.get_mut(i) else { continue };
356                let b = *q;
357                *q += v;
358                (b, *q)
359            };
360            let a = g.band_start(i) + (band - group) / 2.0 + slot * thick * (1.0 - ov);
361            let (p0, p1) = (g.val(v0), g.val(v1));
362            let (p0, p1) = (p0.min(p1), p0.max(p1));
363            let segs = if g.horiz { rect_segs(p0, a, p1, a + thick) } else { rect_segs(a, p0, a + thick, p1) };
364            fill(out, segs, c.point_colour(j, i));
365            if let Some(text) = label_of(s, i) {
366                let mid = a + thick / 2.0;
367                let (x, y) = match (g.horiz, stacked) {
368                    (true, true) => ((p0 + p1) / 2.0, mid),
369                    (false, true) => (mid, (p0 + p1) / 2.0),
370                    // Beyond the bar's end: right of a positive bar, above a positive column.
371                    (true, false) => (if v >= 0.0 { p1 } else { p0 }, mid),
372                    (false, false) => (mid, if v >= 0.0 { p0 } else { p1 }),
373                };
374                labels.push(DataLabel { x, y, text, inside: stacked });
375            }
376        }
377    }
378}

385/// Line series; stacked ones are drawn at the running total of the series up to them.
386fn lines(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, labels: &mut Vec<DataLabel>) {
387    let mut total = vec![0.0f64; g.n];
388    for (j, s) in c.series.iter().enumerate().filter(|(_, s)| has_values(s)) {
389        let pts: Vec<Option<(f32, f32)>> = (0..g.n)
390            .map(|i| {
391                let v = s.values.get(i).copied().flatten()?;
392                let y = match total.get_mut(i) {
393                    Some(t) if c.stacked() => {
394                        *t += v;
395                        *t
396                    }
397                    _ => v,
398                };
399                Some((g.cat_x(i), g.val(y)))
400            })
401            .collect();
402        for (i, p) in pts.iter().enumerate() {
403            if let (Some((x, y)), Some(text)) = (p, label_of(s, i)) {
404                labels.push(DataLabel { x: *x, y: *y, text, inside: false });
405            }
406        }
407        stroke(out, runs(pts.into_iter()), c.series_colour(j), s.line_w);
408    }
409}

411fn areas(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, labels: &mut Vec<DataLabel>) {
412    let mut base = vec![0.0f64; g.n];
413    for (j, s) in c.series.iter().enumerate().filter(|(_, s)| has_values(s)) {
414        let vals = (0..g.n).map(|i| s.values.get(i).copied().flatten().unwrap_or(0.0));
415        let top: Vec<f64> = base.iter().zip(vals).map(|(b, v)| b + v).collect();
416        let mut segs = Vec::with_capacity(2 * g.n + 1);
417        for (i, t) in top.iter().enumerate() {
418            let (x, y) = (g.cat_x(i), g.val(*t));
419            segs.push(if i == 0 { PathSeg::Move(x, y) } else { PathSeg::Line(x, y) });
420        }
421        for (i, b) in base.iter().enumerate().rev() {
422            segs.push(PathSeg::Line(g.cat_x(i), g.val(*b)));
423        }
424        segs.push(PathSeg::Close);
425        fill(out, segs, c.series_colour(j));
426        for (i, (t, b)) in top.iter().zip(&base).enumerate() {
427            if let Some(text) = label_of(s, i) {
428                labels.push(DataLabel { x: g.cat_x(i), y: g.val((t + b) / 2.0), text, inside: true });
429            }
430        }
431        if c.stacked() {
432            base = top;
433        }
434    }
435}

437/// Scatter series as lines, or as small markers when the series has no line.
438fn draw_scatter(out: &mut Vec<GraphicItem>, c: &Chart, g: &Frame, labels: &mut Vec<DataLabel>) {
439    const MARKER: f32 = 6.0;
440    for (j, s) in c.series.iter().enumerate() {
441        let pts: Vec<Option<(f32, f32)>> =
442            s.xs.iter()
443                .zip(&s.values)
444                .map(|(x, y)| match (x, y) {
445                    (Some(x), Some(y)) => Some((g.scat_x(*x), g.val(*y))),
446                    _ => None,
447                })
448                .collect();
449        for (i, p) in pts.iter().enumerate() {
450            if let (Some((x, y)), Some(text)) = (p, label_of(s, i)) {
451                labels.push(DataLabel { x: *x, y: *y - MARKER / 2.0, text, inside: false });
452            }
453        }
454        if s.line {
455            stroke(out, runs(pts.iter().copied()), c.series_colour(j), s.line_w);
456            continue;
457        }
458        for (x, y) in pts.iter().flatten() {
459            let rect = [x - MARKER / 2.0, y - MARKER / 2.0, MARKER, MARKER];
460            out.push(GraphicItem::Shape { rect, kind: ShapeKind::Ellipse, fill: Some(c.series_colour(j)), stroke: None, stroke_width: 0.0 });
461        }
462    }
