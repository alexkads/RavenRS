1/// A piece of a format section: literal text, a number character (`#`, `0`, `,`, `.`), or the
2/// exponent of a scientific format (`E+` when true, `E-` when false).
3enum Tok {
4    Lit(String),
5    Num(char),
6    Exp(bool),
7}
8
9/// The sections of a format code (positive; negative; zero; text), split outside quotes.
10fn sections(code: &str) -> Vec<&str> {
11    let mut out = Vec::new();
12    let (mut start, mut quoted, mut escaped) = (0, false, false);
13    for (i, c) in code.char_indices() {
14        match c {
15            _ if escaped => escaped = false,
16            '\\' if !quoted => escaped = true,
17            '"' => quoted = !quoted,
18            ';' if !quoted => {
19                out.push(code.get(start..i).unwrap_or(""));
20                start = i + 1;
21            }
22            _ => {}
23        }
24    }
25    out.push(code.get(start..).unwrap_or(""));
26    out
27}

29/// The first section of a format code.
30fn first_section(code: &str) -> &str {
31    sections(code).first().copied().unwrap_or("")
32}

34/// The pieces of one format section. Quoted text, escapes and `[$€-407]` currency symbols are
35/// literal; other brackets (`[Red]`) are dropped; `_x` (room for `x`) is a space; `*x` is dropped.
36fn format_toks(section: &str) -> Vec<Tok> {
37    let mut toks = Vec::new();
38    let mut chars = section.chars().peekable();
39    while let Some(c) = chars.next() {
40        match c {
41            '"' => toks.push(Tok::Lit(chars.by_ref().take_while(|q| *q != '"').collect())),
42            '[' => {
43                let inner: String = chars.by_ref().take_while(|q| *q != ']').collect();
44                if let Some(sym) = currency(&inner) {
45                    toks.push(Tok::Lit(sym.to_string()));
46                }
47            }
48            '\\' => toks.extend(chars.next().map(|n| Tok::Lit(n.to_string()))),
49            '_' => {
50                chars.next();
51                toks.push(Tok::Lit(" ".to_string()));
52            }
53            '*' => {
54                chars.next();
55            }
56            'E' | 'e' if matches!(chars.peek(), Some('+' | '-')) => toks.push(Tok::Exp(chars.next() == Some('+'))),
57            '#' | '0' | ',' | '.' => toks.push(Tok::Num(c)),
58            c => toks.push(Tok::Lit(c.to_string())),
59        }
60    }
61    toks
62}

64/// The symbol of a `$€-407` bracket content (`[$€-407]`), if it is one.
65fn currency(inner: &str) -> Option<&str> {
66    inner.strip_prefix('$').map(|rest| rest.split('-').next().unwrap_or(""))
67}

69/// Tick text for `v` in a number format code (`#,##0`, `0.0`, `0%`, `0.00E+00`, and literal text
70/// such as `"€"0.00` or `#,##0.00 [$€-407]`). A negative number takes the second section when
71/// there is one, which shows it without a minus (`#,##0;(#,##0)`). `General` shows as many
72/// decimals as the step needs.
73pub(super) fn format_value(v: f64, code: &str, step: f64) -> String {
74    let v = if v.abs() < step * 1e-9 { 0.0 } else { v };
75    let secs = sections(code);
76    match secs.get(1) {
77        Some(neg) if v < 0.0 => format_section(-v, neg, step),
78        _ => format_section(v, secs.first().copied().unwrap_or(""), step),
79    }
80}

82/// `v` in one section of a format code.
83fn format_section(v: f64, section: &str, step: f64) -> String {
84    let toks = format_toks(section);
85    let is_num = |t: &Tok| matches!(t, Tok::Num(_) | Tok::Exp(_));
86    let lit = |r: &[Tok]| -> String {
87        r.iter()
88            .filter_map(|t| match t {
89                Tok::Lit(l) => Some(l.as_str()),
90                _ => None,
91            })
92            .collect()
93    };
94    let (Some(s), Some(e)) = (toks.iter().position(is_num), toks.iter().rposition(is_num)) else { return general(v, step) };
95    let body = toks.get(s..=e).unwrap_or(&[]);
96    let (pre, suf) = (lit(toks.get(..s).unwrap_or(&[])), lit(toks.get(e + 1..).unwrap_or(&[])));
97    let digits = |r: &[Tok]| -> String { r.iter().filter_map(|t| if let Tok::Num(c) = t { Some(*c) } else { None }).collect() };
98    if let Some(k) = body.iter().position(|t| matches!(t, Tok::Exp(_))) {
99        let plus = matches!(body.get(k), Some(Tok::Exp(true)));
100        let mantissa = digits(body.get(..k).unwrap_or(&[]));
101        let exp_digits = digits(body.get(k + 1..).unwrap_or(&[])).len();
102        return format!("{pre}{}{suf}", scientific(v, decimals(&mantissa), exp_digits, plus));
103    }
104    let body = digits(body);
105    let pct = pre.contains('%') || suf.contains('%');
106    let v = if pct { v * 100.0 } else { v };
107    let digits = format!("{:.*}", decimals(&body), v.abs());
108    let (ip, fp) = match digits.split_once('.') {
109        Some((a, b)) => (a.to_string(), format!(".{b}")),
110        None => (digits.clone(), String::new()),
111    };
112    let mut grouped = String::new();
113    for (k, ch) in ip.chars().enumerate() {
114        if body.contains(',') && k > 0 && (ip.len() - k) % 3 == 0 {
115            grouped.push(',');
116        }
117        grouped.push(ch);
118    }
119    let sign = if v < 0.0 && digits.chars().any(|c| c != '0' && c != '.') { "-" } else { "" };
120    format!("{sign}{pre}{grouped}{fp}{suf}")
121}

123/// Decimal places of a number body (`#,##0.00` has 2).
124fn decimals(body: &str) -> usize {
125    body.split_once('.').map_or(0, |(_, d)| d.chars().filter(|c| matches!(c, '0' | '#')).count()).min(12)
126}

128/// `v` with no number format (`General`): as many decimals as the tick step needs.
129fn general(v: f64, step: f64) -> String {
130    let mut d = 0;
131    while d < 6 && (step * 10f64.powi(d)).fract().abs() > 1e-9 {
132        d += 1;
133    }
134    let s = format!("{:.*}", d as usize, v);
135    if s.trim_start_matches(['-', '0', '.']).is_empty() { s.trim_start_matches('-').to_string() } else { s }
136}

138/// `v` in scientific notation: `decimals` mantissa decimals, at least `exp_digits` exponent
139/// digits, and a `+` on positive exponents when `plus` (`0.00E+00`: 1.23E+04).
140fn scientific(v: f64, decimals: usize, exp_digits: usize, plus: bool) -> String {
141    let s = format!("{:.*e}", decimals, v.abs());
142    let (mantissa, exp) = s.split_once('e').unwrap_or((s.as_str(), "0"));
143    let exp: i32 = if v == 0.0 { 0 } else { exp.parse().unwrap_or(0) };
144    let sign = if exp < 0 {
145        "-"
146    } else if plus {
147        "+"
148    } else {
149        ""
150    };
151    let minus = if v < 0.0 { "-" } else { "" };
152    format!("{minus}{mantissa}E{sign}{:0w$}", exp.unsigned_abs(), w = exp_digits.clamp(1, 4))
153}

155/// A moment as Excel shows a serial day number.
156#[derive(Clone, Copy, Debug, PartialEq)]
157pub(super) struct Stamp {
158    pub(super) y: i64,
159    pub(super) m: u32,
160    pub(super) d: u32,
161    /// Day of the week, 0 = Sunday.
162    weekday: u32,
163    /// Seconds into the day.
164    secs: u32,
165}

167/// The moment of Excel serial day `serial` as Excel shows it, from 0 ("January 0, 1900") to
168/// 31 December 9999. In the 1900 system serial 60 is Excel's phantom 29 February 1900, and
169/// serials before it count from 31 December 1899; in the 1904 system (`date1904`) serial 0 is
170/// 1 January 1904. `None` outside that range.
171pub(super) fn excel_stamp(serial: f64, date1904: bool) -> Option<Stamp> {
172    let serial = if date1904 { serial + 1462.0 } else { serial };
173    if !(0.0..2_958_466.0).contains(&serial) {
174        return None;
175    }
176    let whole = serial.floor();
177    let secs = (((serial - whole) * 86_400.0).round() as u32).min(86_399);
178    let day = whole as i64;
179    let (y, m, d) = match day {
180        0 => (1900, 1, 0),
181        60 => (1900, 2, 29),
182        1..=59 => civil(day + 1),
183        _ => civil(day),
184    };
185    // Excel's weekdays count serial 1 as a Sunday.
186    Some(Stamp { y, m, d, weekday: (day - 1).rem_euclid(7) as u32, secs })
187}

189/// Calendar date (year, month, day) of day `n` counted from 30 December 1899 (an Excel serial
190/// day number). Serial `n` is `n - 25_569` days from 1970-01-01.
191///
192/// The caller's serial guard keeps `n` inside chrono's range, but the Gregorian calendar repeats
193/// every 400 years (146_097 days), so this stays exact for any `n` without hand-rolled calendar
194/// math: chrono converts the day within its cycle and whole cycles go back into the year (an
195/// i128 so the extremes can't overflow). 1970-01-01 is `num_days_from_ce` 719_163, so serial `n`
196/// is CE day `n - 25_569 + 719_163`.
197pub(super) fn civil(n: i64) -> (i64, u32, u32) {
198    use chrono::Datelike;
199    // CE day of serial `n`, in i128 so no `n` overflows.
200    let from_ce = i128::from(n) - 25_569 + 719_163;
201    let (cycles, rest) = (from_ce.div_euclid(146_097), from_ce.rem_euclid(146_097));
202    match i32::try_from(rest).ok().and_then(chrono::NaiveDate::from_num_days_from_ce_opt) {
203        Some(date) => {
204            let y = i128::from(date.year()) + cycles * 400;
205            (i64::try_from(y).unwrap_or(if y < 0 { i64::MIN } else { i64::MAX }), date.month(), date.day())
206        }
207        // Unreachable: every `rest` in 0..146_097 is a valid CE day.
208        None => (1900, 1, 1),
209    }
210}

216/// Whether a format code (outside its quoted and bracketed parts) has date or time fields.
217pub(super) fn is_date_code(code: &str) -> bool {
218    format_toks_raw(code).any(|c| matches!(c.to_ascii_lowercase(), 'y' | 'd' | 'm' | 'h' | 's'))
219}

221/// The characters of a format code's first section outside quotes and brackets.
222fn format_toks_raw(code: &str) -> impl Iterator<Item = char> + '_ {
223    let mut quoted = false;
224    let mut bracket = false;
225    first_section(code).chars().filter(move |c| match c {
226        '"' => {
227            quoted = !quoted;
228            false
229        }
230        '[' if !quoted => {
231            bracket = true;
232            false
233        }
234        ']' if bracket => {
235            bracket = false;
236            false
237        }
238        _ => !quoted && !bracket,
239    })
240}

242/// A piece of a date format: literal text, a field (`y`, `m`, `d`, `h`, `s`, or `n` for minutes)
243/// with its length, or AM/PM (`AM/PM` when true, `A/P` when false).
244enum DTok {
245    Lit(String),
246    Field(char, usize),
247    AmPm(bool),
248}
249
250/// The pieces of a date format's first section. `m` or `mm` right after an hour or right before
251/// a seconds field is minutes.
252fn date_toks(code: &str) -> Vec<DTok> {
253    let section: Vec<char> = first_section(code).chars().collect();
254    let mut toks = Vec::new();
255    let mut i = 0;
256    while let Some(&c) = section.get(i) {
257        let at = i;
258        i += 1;
259        match c {
260            '"' => {
261                let lit: String = section.get(i..).unwrap_or(&[]).iter().take_while(|q| **q != '"').collect();
262                i += lit.chars().count() + 1;
263                toks.push(DTok::Lit(lit));
264            }
265            '[' => {
266                let inner: String = section.get(i..).unwrap_or(&[]).iter().take_while(|q| **q != ']').collect();
267                i += inner.chars().count() + 1;
268                toks.extend(currency(&inner).map(|s| DTok::Lit(s.to_string())));
269            }
270            '\\' | '_' | '*' => {
271                let next = section.get(i).copied();
272                i += 1;
273                match (c, next) {
274                    ('\\', Some(n)) => toks.push(DTok::Lit(n.to_string())),
275                    ('_', _) => toks.push(DTok::Lit(" ".to_string())),
276                    _ => {}
277                }
278            }
279            'a' | 'A' if starts_ci(&section, at, "AM/PM") => {
280                i += 4;
281                toks.push(DTok::AmPm(true));
282            }
283            'a' | 'A' if starts_ci(&section, at, "A/P") => {
284                i += 2;
285                toks.push(DTok::AmPm(false));
286            }
287            _ if matches!(c.to_ascii_lowercase(), 'y' | 'm' | 'd' | 'h' | 's') => {
288                let mut n = 1;
289                while section.get(i).is_some_and(|q| q.eq_ignore_ascii_case(&c)) {
290                    i += 1;
291                    n += 1;
292                }
293                toks.push(DTok::Field(c.to_ascii_lowercase(), n));
294            }
295            c => toks.push(DTok::Lit(c.to_string())),
296        }
297    }
298    mark_minutes(&mut toks);
299    toks
300}

302/// Whether `chars` from `at` on start with `pat`, ignoring ASCII case.
303fn starts_ci(chars: &[char], at: usize, pat: &str) -> bool {
304    pat.chars().enumerate().all(|(k, p)| chars.get(at + k).is_some_and(|c| c.eq_ignore_ascii_case(&p)))
305}

307/// Turns the `m`/`mm` fields that mean minutes (after an hour, or before seconds) into `n`.
308fn mark_minutes(toks: &mut [DTok]) {
309    let fields: Vec<usize> = toks.iter().enumerate().filter(|(_, t)| matches!(t, DTok::Field(..))).map(|(k, _)| k).collect();
310    let kind = |k: Option<&usize>, toks: &[DTok]| match k.and_then(|k| toks.get(*k)) {
311        Some(DTok::Field(c, _)) => Some(*c),
312        _ => None,
313    };
314    for (j, k) in fields.iter().enumerate() {
315        let after_hour = j.checked_sub(1).and_then(|p| kind(fields.get(p), toks)) == Some('h');
316        let before_secs = kind(fields.get(j + 1), toks) == Some('s');
317        if let Some(DTok::Field(c @ 'm', n)) = toks.get_mut(*k)
318            && *n <= 2
319            && (after_hour || before_secs)
320        {
321            *c = 'n';
322        }
323    }
324}

326/// A moment in a date format: day, month, year, weekday (`ddd`, `dddd`), hour, minute and
327/// second fields replaced (12-hour with AM/PM); quoted text and currency kept, other brackets
328/// dropped.
329pub(super) fn date_text(t: Stamp, code: &str) -> String {
330    let toks = date_toks(code);
331    let twelve = toks.iter().any(|k| matches!(k, DTok::AmPm(_)));
332    let hour = t.secs / 3600;
333    let mut out = String::new();
334    for tok in &toks {
335        match tok {
336            DTok::Lit(s) => out.push_str(s),
337            DTok::AmPm(long) => out.push_str(match (hour < 12, long) {
338                (true, true) => "AM",
339                (false, true) => "PM",
340                (true, false) => "A",
341                (false, false) => "P",
342            }),
343            DTok::Field(c, n) => out.push_str(&field_text(t, *c, *n, twelve)),
344        }
345    }
346    out
347}

349/// One date or time field (`c` as in [`DTok::Field`], `n` letters long).
350fn field_text(t: Stamp, c: char, n: usize, twelve: bool) -> String {
351    let month = MONTHS.get((t.m as usize).clamp(1, 12) - 1).copied().unwrap_or("");
352    let weekday = WEEKDAYS.get(t.weekday as usize).copied().unwrap_or("");
353    let pad = |v: u32| if n >= 2 { format!("{v:02}") } else { v.to_string() };
354    let hour = t.secs / 3600;
355    match c {
356        'y' if n <= 2 => format!("{:02}", t.y.rem_euclid(100)),
357        'y' => format!("{:04}", t.y),
358        'd' if n <= 2 => pad(t.d),
359        'd' if n == 3 => weekday.get(..3).unwrap_or(weekday).to_string(),
360        'd' => weekday.to_string(),
361        'm' if n <= 2 => pad(t.m),
362        'm' if n == 3 => month.get(..3).unwrap_or(month).to_string(),
363        'm' if n == 4 => month.to_string(),
364        'm' => month.get(..1).unwrap_or(month).to_string(),
365        'h' if twelve => pad(if hour.is_multiple_of(12) { 12 } else { hour % 12 }),
366        'h' => pad(hour),
367        'n' => pad(t.secs / 60 % 60),
368        _ => pad(t.secs % 60),
369    }
