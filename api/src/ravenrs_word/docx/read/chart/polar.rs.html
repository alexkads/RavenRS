7pub(super) fn pie(out: &mut Vec<GraphicItem>, c: &Chart, area: Rect) {
8    let Some(s) = c.series.first() else { return };
9    let vals: Vec<f64> = s.values.iter().map(|v| v.unwrap_or(0.0).max(0.0)).collect();
10    let total: f64 = vals.iter().sum();
11    let r = (area.w.min(area.h) / 2.0 - 2.0) as f64;
12    if !(total > 0.0 && total.is_finite() && r >= 1.0) {
13        return;
14    }
15    let (cx, cy) = ((area.x + area.w / 2.0) as f64, (area.y + area.h / 2.0) as f64);
16    let inner = if c.kind == Kind::Doughnut { r * c.hole as f64 } else { 0.0 };
17    let mut a = c.first_angle.to_radians();
18    let mut labels = Vec::new();
19    for (i, v) in vals.iter().enumerate() {
20        if *v <= 0.0 {
21            continue;
22        }
23        let b = a + v / total * TAU;
24        let mut segs = Vec::new();
25        if inner > 0.0 {
26            segs.push(pt(polar(cx, cy, r, a)));
27            arc(&mut segs, (cx, cy, r), a, b);
28            segs.push(line_to(polar(cx, cy, inner, b)));
29            arc(&mut segs, (cx, cy, inner), b, a);
30        } else {
31            segs.push(pt((cx, cy)));
32            segs.push(line_to(polar(cx, cy, r, a)));
33            arc(&mut segs, (cx, cy, r), a, b);
34        }
35        segs.push(PathSeg::Close);
36        fill(out, segs, c.point_colour(0, i));
37        if let Some(text) = label_of(s, i) {
38            // Halfway across the ring (pie: two thirds out from the centre).
39            let at = if inner > 0.0 { (r + inner) / 2.0 } else { r * 0.65 };
40            let (x, y) = polar(cx, cy, at, (a + b) / 2.0);
41            labels.push(DataLabel { x: x as f32, y: y as f32, text, inside: true });
42        }
43        a = b;
44    }
45    draw_data_labels(out, &labels, c.text_size, false);
46}

48/// A radar chart in `area`: category spokes from the centre, concentric grid polygons at the value
49/// ticks, one polygon per series (filled, or outlined with markers) and the category labels at the
50/// spoke ends. The lowest tick is the centre.
51pub(super) fn radar(out: &mut Vec<GraphicItem>, c: &Chart, area: Rect) {
52    const MARKER: f32 = 5.0;
53    if c.n == 0 {
54        return;
55    }
56    let fs = c.text_size;
57    let lh = fs * LINE;
58    let labelled = c.cats.iter().any(|s| !s.is_empty());
59    let side = if labelled { widest(&c.cats, fs).min(area.w * 0.25) + GAP + 2.0 } else { 0.0 };
60    let vert = if labelled { lh + GAP } else { 0.0 };
61    let r = ((area.w / 2.0 - side).min(area.h / 2.0 - vert) - 2.0) as f64;
62    if r.is_nan() || r < 8.0 {
63        return;
64    }
65    let (cx, cy) = ((area.x + area.w / 2.0) as f64, (area.y + area.h / 2.0) as f64);
66    let vs = Axes::of(c).vs;
67    let span = vs.hi - vs.lo;
68    let angle = |i: usize| i as f64 / c.n as f64 * TAU;
69    let at = |v: f64, i: usize| {
70        let t = if span > 0.0 { ((v - vs.lo) / span).clamp(0.0, 1.0) } else { 0.0 };
71        polar(cx, cy, r * t, angle(i))
72    };
73    let ticks = vs.ticks();
74    if c.val.grid {
75        for t in ticks.iter().skip(1) {
76            let mut segs: Vec<PathSeg> = (0..c.n).map(|i| if i == 0 { pt(at(*t, i)) } else { line_to(at(*t, i)) }).collect();
77            segs.push(PathSeg::Close);
78            stroke(out, segs, GRID, 0.75);
79        }
80    }
81    for i in 0..c.n {
82        let end = polar(cx, cy, r, angle(i));
83        stroke(out, vec![pt((cx, cy)), line_to(end)], GRID, 0.75);
84    }
85    let mut labels = Vec::new();
86    for (j, s) in c.series.iter().enumerate().filter(|(_, s)| has_values(s)) {
87        let col = c.series_colour(j);
88        // A missing value counts as the centre, as Word draws it.
89        let ring: Vec<(f64, f64)> = (0..c.n).map(|i| at(s.values.get(i).copied().flatten().unwrap_or(vs.lo), i)).collect();
90        let mut segs: Vec<PathSeg> = ring.iter().enumerate().map(|(i, p)| if i == 0 { pt(*p) } else { line_to(*p) }).collect();
91        segs.push(PathSeg::Close);
92        if c.radar_fill {
93            fill(out, segs.clone(), col);
94        }
95        stroke(out, segs, col, s.line_w);
96        for (i, p) in ring.iter().enumerate() {
97            if c.radar_marks && s.values.get(i).copied().flatten().is_some() {
98                let rect = [p.0 as f32 - MARKER / 2.0, p.1 as f32 - MARKER / 2.0, MARKER, MARKER];
99                out.push(GraphicItem::Shape { rect, kind: ShapeKind::Ellipse, fill: Some(col), stroke: None, stroke_width: 0.0 });
100            }
101            if let Some(text) = label_of(s, i) {
102                labels.push(DataLabel { x: p.0 as f32, y: p.1 as f32, text, inside: false });
103            }
104        }
105    }
106    draw_data_labels(out, &labels, fs, false);
107    if !c.val.deleted {
108        for t in &ticks {
109            let y = at(*t, 0).1 as f32;
110            let s = format_value(*t, &c.val_fmt, vs.step);
111            let w = text_w(&s, fs) + 4.0;
112            text(out, Rect { x: cx as f32 - GAP / 2.0 - w, y: y - lh / 2.0, w, h: lh }, &s, fs, TextAlign::Right, false);
113        }
114    }
115    if labelled {
116        for (i, s) in c.cats.iter().enumerate().filter(|(_, s)| !s.is_empty()) {
117            let a = angle(i);
118            let (x, y) = polar(cx, cy, r + GAP as f64, a);
119            let (x, y) = (x as f32, y as f32);
120            let w = side - GAP;
121            let txt = clip_text(s, w, fs);
122            let (sx, sy) = (a.sin(), a.cos());
123            let (rect, align) = if sx > 0.2 {
124                (Rect { x, y: y - lh / 2.0, w, h: lh }, TextAlign::Left)
125            } else if sx < -0.2 {
126                (Rect { x: x - w, y: y - lh / 2.0, w, h: lh }, TextAlign::Right)
127            } else {
128                let cw = (text_w(&txt, fs) + 4.0).max(1.0);
129                (Rect { x: x - cw / 2.0, y: if sy > 0.0 { y - lh } else { y }, w: cw, h: lh }, TextAlign::Center)
130            };
131            text(out, rect, &txt, fs, align, false);
132        }
133    }
134}

136/// The data label of point `i` of series `s`, when it shows one: the value as the file has it,
137/// in the series' number format.
138pub(super) fn label_of(s: &Series, i: usize) -> Option<String> {
139    if s.labels != Some(true) {
140        return None;
141    }
142    let v = s.raw.get(i).copied().flatten()?;
143    let fmt = s.fmt.as_deref().filter(|f| !f.eq_ignore_ascii_case("General"));
144    Some(match fmt {
145        Some(f) => format_value(v, f, 1.0),
146        None => general_value(v),
147    })
148}

150/// A value in the `General` format: whole numbers without decimals, others with up to six.
151fn general_value(v: f64) -> String {
152    if v.fract() == 0.0 && v.abs() < 1e15 {
153        return format!("{v:.0}");
154    }
155    let s = format!("{v:.6}");
156    let s = s.trim_end_matches('0').trim_end_matches('.');
157    if s == "-0" { "0".to_string() } else { s.to_string() }
158}

160/// Data labels over the drawn chart. Beside labels sit above their point, or right of it for
161/// horizontal bars (`beside_right`).
162pub(super) fn draw_data_labels(out: &mut Vec<GraphicItem>, labels: &[DataLabel], fs: f32, beside_right: bool) {
163    let lh = fs * LINE;
164    for l in labels {
165        let w = text_w(&l.text, fs) + 4.0;
166        let (rect, align) = if l.inside {
167            (Rect { x: l.x - w / 2.0, y: l.y - lh / 2.0, w, h: lh }, TextAlign::Center)
168        } else if beside_right {
169            (Rect { x: l.x + GAP / 2.0, y: l.y - lh / 2.0, w, h: lh }, TextAlign::Left)
170        } else {
171            (Rect { x: l.x - w / 2.0, y: l.y - lh, w, h: lh }, TextAlign::Center)
172        };
173        let color = if l.inside { Rgb::WHITE } else { TEXT };
174        out.push(GraphicItem::Text { rect: rect.arr(), text: l.text.clone(), size: fs, color, bold: false, align, font: None });
175    }
176}

178/// A point at clockwise angle `t` (radians from 12 o'clock) on a circle of radius `r`.
179fn polar(cx: f64, cy: f64, r: f64, t: f64) -> (f64, f64) {
180    (cx + r * t.sin(), cy - r * t.cos())
181}
182
183fn pt(p: (f64, f64)) -> PathSeg {
184    PathSeg::Move(p.0 as f32, p.1 as f32)
185}
186
187fn line_to(p: (f64, f64)) -> PathSeg {
188    PathSeg::Line(p.0 as f32, p.1 as f32)
189}

191/// Cubic segments along a circle from angle `a` to `b` (the current point is at `a`). Each piece
192/// spans at most a quarter turn.
193fn arc(segs: &mut Vec<PathSeg>, (cx, cy, r): (f64, f64, f64), a: f64, b: f64) {
194    let n = (((b - a).abs() / FRAC_PI_2).ceil() as usize).clamp(1, 16);
195    let d = (b - a) / n as f64;
196    let k = 4.0 / 3.0 * (d / 4.0).tan();
197    for j in 0..n {
198        let t0 = a + d * j as f64;
199        let t1 = t0 + d;
200        let p0 = polar(cx, cy, r, t0);
201        let p1 = polar(cx, cy, r, t1);
202        let c1 = (p0.0 + k * r * t0.cos(), p0.1 + k * r * t0.sin());
203        let c2 = (p1.0 - k * r * t1.cos(), p1.1 - k * r * t1.sin());
204        segs.push(PathSeg::Cubic(c1.0 as f32, c1.1 as f32, c2.0 as f32, c2.1 as f32, p1.0 as f32, p1.1 as f32));
205    }
