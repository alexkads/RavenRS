1//! Content control properties (`w:sdtPr`, ECMA-376 §17.5.2; check boxes and repeating
2//! sections from the [MS-DOCX] `w14`/`w15` extensions).
3
4use crate::doc::control::{CHECKED, ContentControl, ControlKind, ControlLock, ListItem, MAX_CONTROL_TEXT, MAX_LIST_ITEMS, UNCHECKED};
5
6use crate::docx::xml::El;
7
8/// Most bytes of `w:sdtPr` children kept verbatim per control.
9const MAX_EXTRA: usize = 64 * 1024;
10
11fn text(v: Option<&str>) -> String {
12    let v = v.unwrap_or("");
13    v.chars().filter(|c| *c != crate::doc::para::OBJ && !c.is_control()).take(MAX_CONTROL_TEXT).collect()
14}
15
16/// An on/off element: on unless its value (`w:val`, or the extension's own) says off.
17fn on(e: &El) -> bool {
18    e.attr("w:val").or_else(|| e.attr("w14:val")).or_else(|| e.attr("w15:val")).is_none_or(|v| !matches!(v, "0" | "false" | "off"))
19}
20
21fn symbol(e: Option<&El>, default: char) -> (char, String) {
22    let Some(e) = e else { return (default, String::new()) };
23    let ch = e
24        .attr("w14:val")
25        .and_then(|h| u32::from_str_radix(h.trim(), 16).ok())
26        .and_then(char::from_u32)
27        .filter(|c| *c as u32 >= 0x20 && *c != crate::doc::para::OBJ);
28    (ch.unwrap_or(default), text(e.attr("w14:font")))
29}
30
31fn items(e: &El) -> Vec<ListItem> {
32    e.children("w:listItem")
33        .take(MAX_LIST_ITEMS)
34        .map(|i| ListItem { display: text(i.attr("w:displayText")), value: text(i.attr("w:value")) })
35        .collect()
36}

38/// The control described by `w:sdtPr` (`None`: an `w:sdt` without one, a rich text control).
39pub fn sdt_pr(pr: Option<&El>, block: bool) -> ContentControl {
40    let mut c = ContentControl { block, ..Default::default() };
41    let Some(pr) = pr else { return c };
42    let mut extra_len = 0;
43    for k in pr.els() {
44        match k.name.as_str() {
45            "w:rPr" => c.rpr_xml = k.to_xml(),
46            "w:alias" => c.title = text(k.attr("w:val")),
47            "w:tag" => c.tag = text(k.attr("w:val")),
48            "w:id" => c.id = k.attr("w:val").and_then(|v| v.trim().parse::<i64>().ok()),
49            "w:lock" => c.lock = ControlLock::from_ooxml(k.attr("w:val").unwrap_or("")),
50            "w:placeholder" => c.placeholder = text(k.child("w:docPart").and_then(|d| d.attr("w:val"))),
51            "w:temporary" => c.temporary = on(k),
52            "w:showingPlcHdr" => c.showing_placeholder = on(k),
53            "w:richText" => c.kind = ControlKind::RichText,
54            "w:text" => c.kind = ControlKind::Text { multi_line: k.attr("w:multiLine").is_some_and(|v| !matches!(v, "0" | "false" | "off")) },
55            "w14:checkbox" => {
56                let checked = k.child("w14:checked").is_some_and(on);
57                let (checked_char, checked_font) = symbol(k.child("w14:checkedState"), CHECKED);
58                let (unchecked_char, unchecked_font) = symbol(k.child("w14:uncheckedState"), UNCHECKED);
59                c.kind = ControlKind::CheckBox { checked, checked_char, checked_font, unchecked_char, unchecked_font };
60            }
61            "w:comboBox" => c.kind = ControlKind::ComboBox { items: items(k), last_value: text(k.attr("w:lastValue")) },
62            "w:dropDownList" => c.kind = ControlKind::DropDown { items: items(k), last_value: text(k.attr("w:lastValue")) },
63            "w:date" => {
64                let v = |n: &str| text(k.child_val(n));
65                c.kind = ControlKind::Date {
66                    full_date: text(k.attr("w:fullDate")),
67                    format: v("w:dateFormat"),
68                    lid: v("w:lid"),
69                    calendar: v("w:calendar"),
70                    store_as: v("w:storeMappedDataAs"),
71                };
72            }
73            "w:picture" => c.kind = ControlKind::Picture,
74            "w:docPartObj" | "w:docPartList" => {
75                c.kind = ControlKind::Gallery {
76                    list: k.name == "w:docPartList",
77                    gallery: text(k.child_val("w:docPartGallery")),
78                    category: text(k.child_val("w:docPartCategory")),
79                    unique: k.child("w:docPartUnique").is_some_and(on),
80                };
81            }
82            "w15:repeatingSection" => {
83                c.kind = ControlKind::RepeatingSection {
84                    title: text(k.child("w15:sectionTitle").and_then(|t| t.attr("w:val").or_else(|| t.attr("w15:val")))),
85                    no_insert_delete: k.child("w15:doNotAllowInsertDeleteSection").is_some_and(on),
86                };
87            }
88            "w15:repeatingSectionItem" => c.kind = ControlKind::RepeatingSectionItem,
89            _ => {
90                let x = k.to_xml();
91                if !x.is_empty() && extra_len + x.len() <= MAX_EXTRA {
92                    extra_len += x.len();
93                    c.extra.push(x);
94                }
95            }
96        }
97    }
98    c
99}

101/// `w:sdtEndPr`, kept verbatim.
102pub fn sdt_end_pr(e: Option<&El>) -> String {
103    e.map(|e| e.to_xml()).filter(|x| x.len() <= MAX_EXTRA).unwrap_or_default()
