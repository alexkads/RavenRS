4/// Enclosed characters (`eq \o\ac(○,字)`), with their style from the code's font sizes: Word
5/// enlarges the symbol (bigger than the text, which keeps the field's size) or shrinks the text
6/// (smaller than the symbol).
7fn enclosure(f: &FieldState) -> Option<crate::doc::eq::Enclosure> {
8    use crate::doc::eq::{EncloseShape, EncloseStyle, Enclosure};
9    let mut e = Enclosure::parse(&f.instr)?;
10    let size_at = |at: usize| f.sizes.iter().rev().find(|(start, _)| *start <= at).and_then(|(_, s)| *s);
11    let sym = f.instr.char_indices().find(|(_, c)| EncloseShape::from_symbol(&c.to_string()) == Some(e.shape)).map(|(i, _)| i)?;
12    let txt = e.text.chars().next().and_then(|c| f.instr.get(sym..)?.find(c).map(|i| i + sym)).unwrap_or(sym);
13    let base = f.props.size;
14    e.style = match (size_at(sym), size_at(txt)) {
15        (Some(s), Some(t)) if s > t + 0.01 && base.is_none_or(|b| t >= b - 0.01) => EncloseStyle::Enlarge,
16        (Some(s), None) if base.is_none_or(|b| s > b + 0.01) => EncloseStyle::Enlarge,
17        _ => EncloseStyle::Shrink,
18    };
19    Some(e)
20}

22/// Fields kept as ranges: those whose result another program writes, with formatting and often
23/// several paragraphs (TOCs and citation managers' `ADDIN` fields).
24fn is_range_instr(instr: &str) -> bool {
25    crate::docx::read::is_toc(instr) || instr.trim_start().get(..5).is_some_and(|k| k.eq_ignore_ascii_case("ADDIN"))
26}

31    pub(super) fn begin_field(&mut self, sc: &mut StoryCtx, props: CharProps, instr: String, locked: bool, dirty: bool, form: String) {
32        if sc.fields.len() >= MAX_FIELD_DEPTH || sc.ignored_begins > 0 {
33            sc.ignored_begins += 1;
34            return;
35        }
36        // Fields nested deeper than the cap are ignored (their content flows to the enclosing field).
37        sc.fields.push(FieldState {
38            instr,
39            phase: Phase::Instr,
40            spilled: false,
41            buf: Vec::new(),
42            props,
43            locked,
44            dirty,
45            range: false,
46            sizes: Vec::new(),
47            form,
48        });
49    }
50
51    /// Whether the field below the innermost one is still collecting its code (a field nested
52    /// in another's code contributes text to that code, never content).
53    fn parent_in_instr(sc: &StoryCtx) -> bool {
54        let n = sc.fields.len();
55        n >= 2 && sc.fields.get(n - 2).is_some_and(|p| !p.spilled && !p.range && p.phase == Phase::Instr)
56    }

58    /// `separate`: the code is complete; a range field starts its result here.
59    pub(super) fn separate_field(&mut self, sc: &mut StoryCtx, pb: &mut PB) {
60        if sc.ignored_begins > 0 {
61            return;
62        }
63        let nested_in_instr = Self::parent_in_instr(sc);
64        let Some(f) = sc.fields.last_mut() else { return };
65        if f.range {
66            return;
67        }
68        f.phase = Phase::Result;
69        if !f.spilled && !nested_in_instr && is_range_instr(&f.instr) {
70            f.range = true;
71            let (obj, props) = (InlineObject::FieldStart { instr: f.instr.trim().to_string(), locked: f.locked, dirty: f.dirty }, f.props.clone());
72            self.emit_obj(sc, pb, obj, &props);
73        }
74    }
75
76    pub(super) fn end_field(&mut self, sc: &mut StoryCtx, pb: &mut PB) {
77        if sc.ignored_begins > 0 {
78            sc.ignored_begins -= 1;
79            return;
80        }
81        let nested_in_instr = Self::parent_in_instr(sc);
82        let Some(f) = sc.fields.pop() else { return };
83        if f.range {
84            self.emit_obj(sc, pb, InlineObject::FieldEnd, &f.props);
85            return;
86        }
87        if f.spilled {
88            return;
89        }
90        if f.phase == Phase::Instr && !nested_in_instr && is_range_instr(&f.instr) {
91            // A range field with no result yet.
92            self.emit_obj(sc, pb, InlineObject::FieldStart { instr: f.instr.trim().to_string(), locked: f.locked, dirty: f.dirty }, &f.props);
93            self.emit_obj(sc, pb, InlineObject::FieldEnd, &f.props);
94            return;
95        }
96        // A `HYPERLINK` field is a link, as a `w:hyperlink` is: its result stays ordinary content
97        // carrying the link, so a tab in it reaches its tab stop and the `PAGEREF` in a table of
98        // contents entry (Word for the web writes them this way) stays a field of its own (#2174).
99        if !nested_in_instr && let Some(target) = hyperlink_target(&f.instr) {
100            for (it, mut p) in f.buf {
101                p.link.get_or_insert_with(|| target.clone());
102                match it {
103                    Item::Text(t) => self.emit_text(sc, pb, &t, &p),
104                    Item::Obj(o) => self.emit_obj(sc, pb, *o, &p),
105                }
106            }
107            return;
108        }
109        // A cached result holding a picture, chart or shape (`INCLUDEPICTURE`'s embedded
110        // picture) can't be a text result: keep the field as a range around it so the picture
111        // stays in the document and is saved again. The linked source is never read.
112        // A tracked insertion or deletion inside the result (one the field itself isn't wrapped
113        // in) keeps the field a range too: a cached text result has one revision at most, so
114        // each piece stays an ordinary run with its own `w:ins` / `w:del` (#1037).
115        let tracked_inside = f.buf.iter().any(|(_, p)| (p.ins, p.del) != (f.props.ins, f.props.del));
116        if !nested_in_instr && (tracked_inside || f.buf.iter().any(|(it, _)| matches!(it, Item::Obj(o) if o.frame().is_some()))) {
117            self.emit_obj(sc, pb, InlineObject::FieldStart { instr: f.instr.trim().to_string(), locked: f.locked, dirty: f.dirty }, &f.props);
118            for (it, p) in f.buf {
119                match it {
120                    Item::Text(t) => self.emit_text(sc, pb, &t, &p),
121                    Item::Obj(o) => self.emit_obj(sc, pb, *o, &p),
122                }
123            }
124            self.emit_obj(sc, pb, InlineObject::FieldEnd, &f.props);
125            return;
126        }
127        let mut result = String::new();
128        for (it, _) in &f.buf {
129            match it {
130                Item::Text(t) => result.push_str(t),
131                Item::Obj(o) => result.push_str(o.plain_text()),
132            }
133        }
134        let (instr, result) = match enclosure(&f) {
135            // Word writes no result for enclosed characters: the text is their plain text.
136            Some(e) => (e.instr(), if result.is_empty() { e.text } else { result }),
137            None => (f.instr.trim().to_string(), result),
138        };
139        // The visible result owns the field's formatting; empty results keep the begin's.
140        let props = f
141            .buf
142            .iter()
143            .find_map(|(it, p)| match it {
144                Item::Text(t) if !t.is_empty() => Some(p),
145                Item::Obj(o) if !o.plain_text().is_empty() => Some(p),
146                _ => None,
147            })
148            .unwrap_or(&f.props)
149            .clone();
150        let obj = InlineObject::Field { instr, result, locked: f.locked, dirty: f.dirty, code: false, form: f.form };
151        if let Some(parent) = sc.fields.last_mut()
152            && !parent.spilled
153            && parent.phase == Phase::Instr
154        {
155            parent.instr.push_str(obj.plain_text());
156            return;
157        }
158        self.emit_obj(sc, pb, obj, &props);
159    }

161    /// At a paragraph end, open fields can't keep buffering: emit them (empty result) and
162    /// let the rest of their result flow as ordinary text.
163    pub(super) fn spill_fields(&mut self, sc: &mut StoryCtx, pb: &mut PB) {
164        for f in sc.fields.iter_mut() {
165            if f.spilled || f.range {
166                continue;
167            }
168            f.spilled = true;
169            let form = std::mem::take(&mut f.form);
170            pb.push_obj(
171                InlineObject::Field { instr: f.instr.trim().to_string(), result: String::new(), locked: f.locked, dirty: f.dirty, code: false, form },
172                &f.props,
173            );
174            for (it, p) in f.buf.drain(..) {
175                match it {
176                    Item::Text(t) => pb.push_text(&t, &p),
177                    Item::Obj(o) => pb.push_obj(*o, &p),
178                }
179            }
180        }
181    }
