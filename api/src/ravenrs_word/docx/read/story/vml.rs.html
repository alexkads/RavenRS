14    /// Every legacy VML drawing in one `w:pict`, in document order: a `w:pict` may hold any
15    /// number of shapes (ECMA-376 Part 4, VML), and each one is its own object. A `v:group`
16    /// becomes an [`InlineObject::Group`].
17    pub(super) fn read_vml_all(&mut self, sc: &mut StoryCtx, pict: &El, rels: &Rels) -> Vec<InlineObject> {
18        let mut out = Vec::new();
19        for e in pict.els().filter(|e| e.name.starts_with("v:") && e.local() != "shapetype") {
20            if out.len() >= crate::doc::para::MAX_GROUP_CHILDREN {
21                break;
22            }
23            if sc.in_hf
24                && let Some(wm) = vml_watermark(e)
25            {
26                // The document's watermark (the first one found); drawn on every page, not as
27                // a shape, so it is not one of the objects this picture contributes.
28                self.doc.settings.watermark.get_or_insert(wm);
29                continue;
30            }
31            let o = if e.local() == "group" { self.read_vml_group(sc, pict, e, rels) } else { self.read_vml_shape(sc, pict, e, rels, None) };
32            out.extend(o);
33        }
34        out
35    }

37    /// The first VML drawing in `pict`, for the places that show a single one: an OLE object's
38    /// picture and a `mc:Fallback` preview.
39    pub(super) fn read_vml(&mut self, sc: &mut StoryCtx, pict: &El, rels: &Rels) -> Option<InlineObject> {
40        self.read_vml_all(sc, pict, rels).into_iter().next()
41    }

43    /// A `v:group`: its own box from its `style`, its members in the space `coordsize` and
44    /// `coordorigin` give it, nested groups flattened into it as [`Self::group_members`] does for
45    /// DrawingML. `None` when it has no member we can show. `pict` is the picture the group sits
46    /// in, where a member's `v:shapetype` is written.
47    fn read_vml_group(&mut self, sc: &mut StoryCtx, pict: &El, grp: &El, rels: &Rels) -> Option<InlineObject> {
48        let style = grp.attr("style").unwrap_or("");
49        let len = |k: &str| vml_style(style, k).and_then(|v| measure(v, 1.0)).unwrap_or(0.0).clamp(0.0, crate::docx::units::MAX_LEN_PT);
50        let (w, h) = (len("width"), len("height"));
51        let float = vml_float(grp, style, sc.in_hf);
52        // The members' space; without a usable `coordsize` it is the group's own box.
53        let (ch_w, ch_h) = vml_coordpair(grp, "coordsize").filter(|(cw, ch)| *cw > 0.0 && *ch > 0.0).unwrap_or((w.max(1.0), h.max(1.0)));
54        let (cx, cy) = vml_coordpair(grp, "coordorigin").unwrap_or((0.0, 0.0));
55        let mut children = Vec::new();
56        self.vml_group_members(sc, pict, grp, rels, VmlGroupSpace { origin: (-cx, -cy), scale: (1.0, 1.0), depth: 0 }, &mut children);
57        if children.is_empty() {
58            return None;
59        }
60        Some(InlineObject::Group { w, h, float, ch_w, ch_h, children, canvas: None })
61    }

63    /// The members of VML group `g` into `out`, mapped into the outermost group's space by `t`
64    /// (x0, y0, sx, sy: outer = x0 + inner × sx). A member is placed by `left`/`top` in that
65    /// space, except a `v:line`, which carries its own offset in `from`.
66    fn vml_group_members(
67        &mut self,
68        sc: &mut StoryCtx,
69        pict: &El,
70        g: &El,
71        rels: &Rels,
72        space: VmlGroupSpace,
73        out: &mut Vec<crate::doc::para::GroupChild>,
74    ) {
75        let (x0, y0) = space.origin;
76        let (sx, sy) = space.scale;
77        let max = crate::docx::units::MAX_LEN_PT;
78        let fit = |v: f32, lo: f32| crate::geom::finite(v).clamp(lo, max);
79        for e in g.els() {
80            if out.len() >= crate::doc::para::MAX_GROUP_CHILDREN {
81                return;
82            }
83            if !e.name.starts_with("v:") || e.local() == "shapetype" {
84                continue;
85            }
86            let style = e.attr("style").unwrap_or("");
87            let num = |k: &str| vml_style(style, k).and_then(|v| measure(v, 1.0)).map_or(0.0, crate::geom::finite);
88            let ((ox, oy), (ew, eh)) = ((num("left"), num("top")), (num("width"), num("height")));
89            if e.local() == "group" {
90                if space.depth >= MAX_VML_GROUP_DEPTH {
91                    continue;
92                }
93                // Its members' space maps onto its own box.
94                let (cw, ch) = vml_coordpair(e, "coordsize").filter(|(cw, ch)| *cw > 0.0 && *ch > 0.0).unwrap_or((ew.max(1.0), eh.max(1.0)));
95                let (cox, coy) = vml_coordpair(e, "coordorigin").unwrap_or((0.0, 0.0));
96                let (kx, ky) = (ew / cw, eh / ch);
97                let nt = (x0 + (ox - cox * kx) * sx, y0 + (oy - coy * ky) * sy, sx * kx, sy * ky);
98                self.vml_group_members(sc, pict, e, rels, VmlGroupSpace { origin: (nt.0, nt.1), scale: (nt.2, nt.3), depth: space.depth + 1 }, out);
99                continue;
100            }
101            let Some(mut obj) = self.read_vml_shape(sc, pict, e, rels, Some((sx, sy))) else { continue };
102            // A member is placed by `x`/`y`, so a line's own offset comes off the member itself.
103            let (dx, dy) = match obj.float_mut() {
104                Some(f) => {
105                    let d = (f.x, f.y);
106                    (f.x, f.y) = (0.0, 0.0);
107                    d
108                }
109                None => (0.0, 0.0),
110            };
111            out.push(crate::doc::para::GroupChild { x: fit(x0 + (ox + dx) * sx, -max), y: fit(y0 + (oy + dy) * sy, -max), obj });
112        }
113    }

115    /// One legacy VML picture, text box, line, rectangle, oval or preset shape (best effort).
116    /// `pict` is the picture it sits in, where its `v:shapetype` is written -- for a group's
117    /// member too, since Word writes each shapetype once, beside the group rather than inside
118    /// it. `grp` scales a member from its group's space, and leaves placing it to the caller.
119    fn read_vml_shape(&mut self, sc: &mut StoryCtx, pict: &El, shape: &El, rels: &Rels, grp: Option<(f32, f32)>) -> Option<InlineObject> {
120        let style = shape.attr("style").unwrap_or("");
121        let len = |k: &str| vml_style(style, k).and_then(|v| measure(v, 1.0)).unwrap_or(0.0).clamp(0.0, crate::docx::units::MAX_LEN_PT);
122        let (sx, sy) = grp.unwrap_or((1.0, 1.0));
123        let (mut w, mut h) = ((len("width") * sx).min(crate::docx::units::MAX_LEN_PT), (len("height") * sy).min(crate::docx::units::MAX_LEN_PT));
124        // A member of a group never floats: the group does, and places it.
125        let mut float = match grp {
126            Some(_) => {
127                let mut f = Float { hidden: vml_style(style, "visibility").is_some_and(|v| v.eq_ignore_ascii_case("hidden")), ..Float::default() };
128                f.set_spin(vml_spin(style));
129                f
130            }
131            None => vml_float(shape, style, sc.in_hf),
132        };
133        if let Some(img) = shape.find("v:imagedata") {
134            let Some(id) = img.attr("r:id").or_else(|| img.attr("r:pict")) else {
135                // `r:href` names a file outside the package: never fetched.
136                if let Some(href) = img.attr("r:href") {
137                    let target = rels.by_id(href).map_or_else(|| href.to_string(), |r| r.target.clone());
138                    self.note_missing_picture(&target, crate::doc::MissingPictureReason::LinkOnly);
139                }
140                return None;
141            };
142            let media = self.media_for(rels, id)?;
143            let alt = shape.attr("alt").or_else(|| img.attr("o:title")).unwrap_or("").to_string();
144            // Brightness and contrast (Word's picture watermark "Washout" sets them, #1711).
145            let frac = |k: &str| img.attr(k).and_then(crate::docx::units::vml_number);
146            let fx = crate::doc::picfx::PicFx::from_vml(frac("gain").unwrap_or(1.0), frac("blacklevel").unwrap_or(0.0));
147            return Some(InlineObject::Image { media, w, h, alt, float, crop: [0.0; 4], fx, ole: None });
148        }
149        // The shapetype the shape is drawn from (its `type`, ECMA-376 Part 4 VML), when written in
150        // the same picture: what the shape leaves out, it takes from there.
151        let type_id = shape.attr("type").map(|t| t.trim().trim_start_matches('#'));
152        let shapetype = type_id.and_then(|id| pict.els().find(|e| e.name == "v:shapetype" && e.attr("id") == Some(id)));
153        let attr = |name: &str| shape.attr(name).or_else(|| shapetype.and_then(|t| t.attr(name)));
154        let on = |name: &str| {
155            attr(name).is_none_or(|v| {
156                let t = v.trim();
157                !t.eq_ignore_ascii_case("f") && !t.eq_ignore_ascii_case("false") && t != "0"
158            })
159        };
160        let stroke = on("stroked").then(|| attr("strokecolor").map_or(Some(Rgb::BLACK), vml_color)).flatten();
161        let stroke_width = attr("strokeweight").and_then(|v| measure(v, 1.0)).unwrap_or(0.75).clamp(0.0, 100.0);
162        let fill = vml_fill(shape, shapetype);
163        let endcap = shape
164            .child("v:stroke")
165            .and_then(|s| s.attr("endcap"))
166            .or_else(|| shapetype.and_then(|t| t.child("v:stroke")).and_then(|s| s.attr("endcap")));
167        let cap = match endcap {
168            Some("square") => crate::doc::connector::LineCap::Square,
169            Some("round") => crate::doc::connector::LineCap::Round,
170            _ => crate::doc::connector::LineCap::Flat,
171        };
172        let kind = match shape.local() {
173            "line" => Some(ShapeKind::Line),
174            "rect" => Some(ShapeKind::Rectangle),
175            "roundrect" => Some(ShapeKind::RoundedRectangle),
176            "oval" => Some(ShapeKind::Ellipse),
177            // A preset, unless the shape's own path replaces its shapetype's geometry.
178            "shape" if shape.attr("path").is_none() && shape.child("v:path").and_then(|p| p.attr("v")).is_none() => {
179                vml_spt(shape, shapetype, type_id).and_then(vml_preset_kind)
180            }
181            _ => None,
182        };
183        // Word 2007 WordArt: a `v:shape` whose `v:textpath` draws its `string` (the header watermark
184        // is taken before this). Shown as a text box holding that text, in the textpath's font and the
185        // shape's `fillcolor`; the box itself has neither fill nor outline, as the letters are the art.
186        if shape.local() == "shape"
187            && let Some(tp) = shape.child("v:textpath")
188            && !tp.attr("on").is_some_and(|v| matches!(v, "f" | "false" | "0"))
189            && let Some(text) = tp.attr("string").map(|t| t.chars().take(200).collect::<String>()).filter(|t| !t.trim().is_empty())
190            && sc.story_depth < MAX_STORY_DEPTH
191        {
192            let font = tp
193                .attr("style")
194                .and_then(|s| vml_style(s, "font-family"))
195                .map(|f| css_string(f).chars().take(100).collect::<String>())
196                .filter(|f| !f.is_empty());
197            // The shapetype's WordArt shape (`o:spt` 136-175) bends the text like DrawingML's warp.
198            let warp = vml_spt(shape, shapetype, type_id)
199                .and_then(|spt| spt.checked_sub(136))
200                .and_then(|i| VML_TEXT_WARPS.get(i as usize))
201                .copied()
202                .and_then(TextWarp::new);
203            // The letters fill the box (`fitshape`): sized to its height.
204            let props = CharProps {
205                font,
206                size: Some((h * 0.75).clamp(8.0, 200.0)),
207                color: attr("fillcolor").and_then(vml_color).map(TextColor::Rgb),
208                ..CharProps::default()
209            };
210            let id = self.doc.add_part(PartKind::TextBox, vec![para_block(Paragraph::with_text(&text, props))]);
211            if let Some(part) = self.doc.parts.get_mut(&id) {
212                part.body.anchor = crate::doc::props::VAlign::Center;
213                part.body.inset = Some([0; 4]);
214            }
215            return Some(InlineObject::Shape {
216                kind: ShapeKind::TextBox,
217                w,
218                h,
219                fill: None,
220                stroke: None,
221                stroke_width,
222                float,
223                story: Some(id),
224                freeform: None,
225                effects: Default::default(),
226                extra: ShapeExtra { warp, ..ShapeExtra::default() },
227            });
228        }
229        if let Some((tb, t)) = vml_textbox(shape, 0)
230            && sc.story_depth < MAX_STORY_DEPTH
231        {
232            let id = self.read_textbox(sc, t, rels);
233            if let Some(part) = self.doc.parts.get_mut(&id) {
234                // Text direction (#5193), as Word reads it: `mso-layout-flow-alt:bottom-to-top` is
235                // `vert270` whatever `layout-flow` says; otherwise `layout-flow:vertical…` (also
236                // `vertical-ideographic`, our `eaVert`) is `vert`, and anything else horizontal.
237                let tb_style = tb.attr("style").unwrap_or("");
238                if vml_style(tb_style, "mso-layout-flow-alt").is_some_and(|v| v.eq_ignore_ascii_case("bottom-to-top")) {
239                    part.body.vert = crate::doc::props::TextVert::Vert270;
240                } else if vml_style(tb_style, "layout-flow").is_some_and(|v| v.to_ascii_lowercase().starts_with("vertical")) {
241                    part.body.vert = crate::doc::props::TextVert::Vert;
242                }
243                // Internal margins `inset="l,t,r,b"` (ECMA-376 Part 4 §19.1.2.22), each one left
244                // out keeping Word's default; a bare number is in EMUs, VML's base unit.
245                // `o:insetmode="auto"` leaves them to the application.
246                let mut inset = crate::doc::TEXT_BOX_INSET;
247                let custom = tb.attr("o:insetmode").is_none_or(|m| !m.eq_ignore_ascii_case("auto"));
248                for (slot, v) in inset.iter_mut().zip(tb.attr("inset").filter(|_| custom).unwrap_or("").split(',')) {
249                    if let Some(pt) = measure(v, 12_700.0) {
250                        *slot = (pt.max(0.0) * 12_700.0).round().min(crate::doc::MAX_TEXT_BOX_INSET as f32) as u32;
251                    }
252                }
253                part.body.inset = (inset != crate::doc::TEXT_BOX_INSET).then_some(inset);
254                // Resize shape to fit text (DrawingML's `a:spAutoFit`).
255                let fit = vml_style(tb.attr("style").unwrap_or(""), "mso-fit-shape-to-text").or_else(|| vml_style(style, "mso-fit-shape-to-text"));
256                part.body.auto_fit = fit.is_some_and(|v| v.eq_ignore_ascii_case("t") || v.eq_ignore_ascii_case("true"));
257                // Where the text sits between the box's top and bottom (DrawingML's `anchor`).
258                if let Some(a) = vml_style(tb.attr("style").unwrap_or(""), "v-text-anchor").or_else(|| vml_style(style, "v-text-anchor")) {
259                    let al = a.trim().to_ascii_lowercase();
260                    part.body.anchor = if al.starts_with("middle") || al == "center" {
261                        crate::doc::props::VAlign::Center
262                    } else if al.starts_with("bottom") {
263                        crate::doc::props::VAlign::Bottom
264                    } else {
265                        crate::doc::props::VAlign::Top
266                    };
267                }
268            }
269            let story = Some(id);
270            // Text in a preset shape keeps the shape around it; otherwise it is a text box.
271            let kind = kind.filter(|k| *k != ShapeKind::Rectangle && !k.is_open()).unwrap_or(ShapeKind::TextBox);
272            return Some(InlineObject::Shape {
273                kind,
274                w,
275                h,
276                fill,
277                stroke,
278                stroke_width,
279                float,
280                story,
281                freeform: None,
282                effects: Default::default(),
283                extra: ShapeExtra { cap, ..Default::default() },
284            });
285        }
286        // A freeform: its `path`, in its `coordsize` space; a `v:fill` with no opacity
287        // or `nf` in the path leaves it unfilled. Not WordArt (`v:textpath`).
288        if shape.local() == "shape"
289            && shape.child("v:textpath").is_none()
290            && let Some(path) = shape.attr("path")
291            && let Some(vp) = crate::docx::read::freeform::vml_path(path, shape.attr("coordsize"), shape.attr("coordorigin"), w, h)
292        {
293            let opacity = |el: &str| shape.child(el).and_then(|e| e.attr("opacity")).and_then(vml_opacity).unwrap_or(1.0);
294            let fill = fill.filter(|_| !vp.no_fill && opacity("v:fill") > 0.0);
295            let stroke = stroke.filter(|_| !vp.no_stroke);
296            if stroke.is_none() && fill.is_none() {
297                return None;
298            }
299            let mut f = vp.freeform;
300            f.alpha = opacity("v:stroke");
301            return Some(InlineObject::Shape {
302                kind: ShapeKind::Freeform,
303                w,
304                h,
305                fill,
306                stroke,
307                stroke_width,
308                float,
309                story: None,
310                freeform: Some(Arc::new(f)),
311                effects: Default::default(),
312                extra: ShapeExtra { cap, ..Default::default() },
313            });
314        }
315        let kind = kind?;
316        let mut extra = ShapeExtra { cap, ..Default::default() };
317        // A horizontal rule (`o:hr`, a pasted `<hr>`): across `o:hrpct` tenths of a percent of its
318        // line, or its own width (`o:hrpct="0"`); a width of 0 with no percentage is the whole
319        // line. `o:hralign` places a shorter one (Word, measured: 50 % centred, left, right).
320        // Only a rule in the text, shown: a positioned or hidden one stays a plain shape, which
321        // keeps its placement and visibility when saved.
322        if attr("o:hr").is_some_and(|v| matches!(v.trim(), "t" | "true" | "1")) && float.wrap == Wrap::Inline && !float.hidden {
323            let tenths = attr("o:hrpct").and_then(|v| v.trim().parse::<f32>().ok()).filter(|p| p.is_finite() && *p > 0.0);
324            let pct = tenths.map_or(if w > 0.0 { 0.0 } else { 100.0 }, |t| (t / 10.0).min(100.0));
325            let align = match attr("o:hralign").map(|a| a.trim().to_ascii_lowercase()).as_deref() {
326                Some("center") => crate::doc::props::Align::Center,
327                Some("right") => crate::doc::props::Align::Right,
328                _ => crate::doc::props::Align::Left,
329            };
330            let standard = attr("o:hrstd").is_some_and(|v| matches!(v.trim(), "t" | "true" | "1"));
331            extra.hr = Some(crate::doc::connector::HorizontalRule { pct, align, standard });
332        }
333        if kind == ShapeKind::Line {
334            // Arrowheads (`v:stroke` startarrow / endarrow, with their width and length): any but
335            // `none` is drawn.
336            use crate::doc::connector::{ArrowKind, ArrowSize, ArrowStyle};
337            let stroke = shape.child("v:stroke");
338            let arrow = |end: &str| {
339                let s = stroke?;
340                let kind = s.attr(&format!("{end}arrow")).filter(|v| *v != "none")?;
341                let size = |what: &str| ArrowSize::from_vml(s.attr(&format!("{end}arrow{what}")).unwrap_or("medium"));
342                Some(ArrowStyle { kind: ArrowKind::from_vml(kind), width: size("width"), length: size("length") })
343            };
344            if let Some(style) = arrow("start") {
345                extra.arrow_start = true;
346                extra.start_style = style;
347            }
348            if let Some(style) = arrow("end") {
349                extra.arrow_end = true;
350                extra.end_style = style;
351            }
352        }
353        if shape.local() == "line" {
354            // A line runs `from` → `to` (points, offsets from where its style places it). A preset
355            // line (`v:shape`) runs corner to corner of the box its style gives it.
356            let pt = |name: &str| -> (f32, f32) {
357                let v = shape.attr(name).unwrap_or("0,0");
358                let (x, y) = v.split_once(',').unwrap_or((v, "0"));
359                let m =
360                    |s: &str| measure(s, 1.0).map_or(0.0, crate::geom::finite).clamp(-crate::docx::units::MAX_LEN_PT, crate::docx::units::MAX_LEN_PT);
361                (m(x), m(y))
362            };
363            let ((x0, y0), (x1, y1)) = (pt("from"), pt("to"));
364            let max = crate::docx::units::MAX_LEN_PT;
365            (w, h) = (((x1 - x0).abs() * sx).min(max), ((y1 - y0).abs() * sy).min(max));
366            float.x = (float.x + x0.min(x1)).clamp(-max, max);
367            float.y = (float.y + y0.min(y1)).clamp(-max, max);
368            // Drawn top-left to bottom-right; flipped for the other directions.
369            float.flip_h ^= x1 < x0;
370            float.flip_v ^= y1 < y0;
371        }
372        if stroke.is_none() && (fill.is_none() || kind == ShapeKind::Line) {
373            return None;
374        }
375        Some(InlineObject::Shape {
376            kind,
377            w,
378            h,
379            fill: fill.filter(|_| kind != ShapeKind::Line),
380            stroke,
381            stroke_width,
382            float,
383            story: None,
384            freeform: None,
385            effects: Default::default(),
386            extra,
387        })
388    }

390    /// An OLE object (`w:object`, ECMA-376 §17.3.3.19): shown as its picture (VML, or a DrawingML
391    /// picture in `w:drawing`), and kept whole so saving writes the object back.
392    pub(super) fn read_object(&mut self, sc: &mut StoryCtx, obj: &El, rels: &Rels) -> Option<InlineObject> {
393        let mut o = match self.read_vml(sc, obj, rels) {
394            Some(o) => o,
395            None => {
396                let d = obj.child("w:drawing")?;
397                self.read_drawing(sc, d, rels)?
398            }
399        };
400        if let InlineObject::Image { ole, .. } = &mut o {
401            *ole = self.keep_embedded(obj, rels, &[]);
402        }
403        Some(o)
404    }

406    /// A Word signature line (Insert › Signature Line): a `w:pict` whose one VML picture carries
407    /// an `o:signatureline` (the suggested signer, title, e-mail and instructions). Shown as its
408    /// picture, and kept whole like an OLE object so saving writes the signature line back,
409    /// still signable in Word, not a plain picture. `None` for any other `w:pict`.
410    pub(super) fn read_signature_line(&mut self, sc: &mut StoryCtx, pict: &El, rels: &Rels) -> Option<InlineObject> {
411        let mut drawings = pict.els().filter(|e| e.name.starts_with("v:") && e.local() != "shapetype");
412        let shape = drawings.next().filter(|s| s.name == "v:shape")?;
413        if drawings.next().is_some() || shape.child("o:signatureline").is_none() {
414            return None;
415        }
416        let mut o = self.read_vml_shape(sc, pict, shape, rels, None)?;
417        if let InlineObject::Image { ole, .. } = &mut o {
418            *ole = self.keep_embedded(pict, rels, &[]);
419        }
420        Some(o)
421    }
422}

480/// The value of property `key` in a VML `style` attribute (CSS declarations; the last one wins).
481/// The first `v:textbox` under `e` holding text, with its `w:txbxContent` (depth-first, bounded
482/// like `El::find`): in a group, a member's margins go with that member's text.
483/// A VML `coordsize="w,h"` or `coordorigin="x,y"` pair (ECMA-376 Part 4 §14.1.2): the member
484/// space of a group, in its own unitless coordinates.
485fn vml_coordpair(e: &El, name: &str) -> Option<(f32, f32)> {
486    let v = e.attr(name)?;
487    let (x, y) = v.split_once(',')?;
488    let n = |s: &str| measure(s.trim(), 1.0).map(crate::geom::finite);
489    Some((n(x)?, n(y)?))
490}
491
492fn vml_textbox(e: &El, depth: usize) -> Option<(&El, &El)> {
493    if depth > crate::docx::xml::MAX_DEPTH {
494        return None;
495    }
496    e.els().find_map(|c| match c.child("w:txbxContent") {
497        Some(t) if c.name == "v:textbox" => Some((c, t)),
498        _ => vml_textbox(c, depth + 1),
499    })
500}

502fn vml_style<'a>(style: &'a str, key: &str) -> Option<&'a str> {
503    style.rsplit(';').filter_map(|d| d.split_once(':')).find(|(k, _)| k.trim().eq_ignore_ascii_case(key)).map(|(_, v)| v.trim())
504}

506/// A `v:textpath` style value with one pair of surrounding quotes removed and the writer's CSS
507/// escapes mapped back: `\` + up to six hex digits + one optional space is that character, and
508/// `\x` is `x` (#3994; `write::watermark::run` writes them). Only the surrounding pair goes, so a
509/// font name that starts or ends with an apostrophe keeps it. Word writes font names raw, and a
510/// lone trailing `\` none of CSS would accept stays literal.
511fn css_string(v: &str) -> String {
512    let v = v.trim();
513    let v = ['"', '\''].into_iter().find_map(|q| v.strip_prefix(q)?.strip_suffix(q)).unwrap_or(v).trim();
514    let mut out = String::with_capacity(v.len());
515    let mut chars = v.chars();
516    while let Some(c) = chars.next() {
517        if c != '\\' {
518            out.push(c);
519            continue;
520        }
521        let mut hex = String::new();
522        while hex.len() < 6 {
523            match chars.clone().next() {
524                Some(h) if h.is_ascii_hexdigit() => {
525                    hex.push(h);
526                    chars.next();
527                }
528                _ => break,
529            }
530        }
531        if hex.is_empty() {
532            match chars.next() {
533                Some(e) => out.push(e),
534                None => out.push('\\'),
535            }
536        } else {
537            if let Some(d) = u32::from_str_radix(&hex, 16).ok().and_then(char::from_u32) {
538                out.push(d);
539            }
540            // One whitespace after the escape is its terminator, not part of the value.
541            if chars.clone().next().is_some_and(char::is_whitespace) {
542                chars.next();
543            }
544        }
545    }
546    out
547}

549/// Placement of a VML shape (ECMA-376 Part 4, VML): `position:absolute` in its style makes it
550/// float, offset by `left`/`margin-left` and `top`/`margin-top` from the area named by
551/// `mso-position-horizontal-relative` / `mso-position-vertical-relative` (unless aligned with
552/// `mso-position-horizontal` / `mso-position-vertical`), wrapped as its `w10:wrap` says. Without
553/// wrapping (`none`, or no `w10:wrap`) it's in front of the text, or behind it at a negative
554/// `z-index`. Anything else stays inline.
555fn vml_float(shape: &El, style: &str, in_hf: bool) -> Float {
556    let get = |k: &str| vml_style(style, k);
557    let hidden = get("visibility").is_some_and(|v| v.eq_ignore_ascii_case("hidden"));
558    if !get("position").is_some_and(|p| p.eq_ignore_ascii_case("absolute")) {
559        // Inline: only the turn (an OLE object turned in WordCraft is saved so, #319).
560        let mut f = Float { hidden, ..Float::default() };
561        f.set_spin(vml_spin(style));
562        return f;
563    }
564    let off = |k: &str| get(k).and_then(|v| measure(v, 1.0)).unwrap_or(0.0);
565    let max = crate::docx::units::MAX_LEN_PT;
566    let behind = get("z-index").and_then(int).is_some_and(|z| z < 0);
567    let wrap_el = shape.child("w10:wrap");
568    let wrap_type = wrap_el.and_then(|w| w.attr("type")).unwrap_or("none");
569    let wrap = match wrap_type {
570        t if t.eq_ignore_ascii_case("square") => Wrap::Square,
571        t if t.eq_ignore_ascii_case("tight") => Wrap::Tight,
572        t if t.eq_ignore_ascii_case("through") => Wrap::Through,
573        t if t.eq_ignore_ascii_case("topAndBottom") || t.eq_ignore_ascii_case("top-and-bottom") => Wrap::TopAndBottom,
574        _ if behind => Wrap::BehindText,
575        _ => Wrap::InFrontOfText,
576    };
577    let wrap_text = wrap_el.and_then(|w| w.attr("side")).and_then(crate::doc::para::WrapText::from_ooxml).unwrap_or_default();
578    let h_rel_str = get("mso-position-horizontal-relative").unwrap_or("text");
579    let h_rel = match h_rel_str {
580        s if s.eq_ignore_ascii_case("page") => Anchor::Page,
581        s if s.eq_ignore_ascii_case("margin") => Anchor::Margin,
582        s if s.eq_ignore_ascii_case("char") => Anchor::Character,
583        s if s.eq_ignore_ascii_case("left-margin-area") => Anchor::LeftMargin,
584        s if s.eq_ignore_ascii_case("right-margin-area") => Anchor::RightMargin,
585        s if s.eq_ignore_ascii_case("inner-margin-area") => Anchor::InsideMargin,
586        s if s.eq_ignore_ascii_case("outer-margin-area") => Anchor::OutsideMargin,
587        _ => Anchor::Column,
588    };
589    let v_rel_str = get("mso-position-vertical-relative").unwrap_or("text");
590    let v_rel = match v_rel_str {
591        s if s.eq_ignore_ascii_case("page") => Anchor::Page,
592        s if s.eq_ignore_ascii_case("margin") => Anchor::Margin,
593        s if s.eq_ignore_ascii_case("line") => Anchor::Line,
594        s if s.eq_ignore_ascii_case("top-margin-area") => Anchor::TopMargin,
595        s if s.eq_ignore_ascii_case("bottom-margin-area") => Anchor::BottomMargin,
596        s if s.eq_ignore_ascii_case("inner-margin-area") => Anchor::InsideMargin,
597        s if s.eq_ignore_ascii_case("outer-margin-area") => Anchor::OutsideMargin,
598        _ => Anchor::Paragraph,
599    };
600    let dist = |k: &str| get(k).and_then(|v| measure(v, 1.0)).unwrap_or(0.0).clamp(0.0, 1584.0);
601    let mut f = Float {
602        wrap,
603        wrap_text,
604        h_rel,
605        v_rel,
606        x: (off("left") + off("margin-left")).clamp(-max, max),
607        y: (off("top") + off("margin-top")).clamp(-max, max),
608        h_align: get("mso-position-horizontal").and_then(FloatAlign::from_ooxml),
609        v_align: get("mso-position-vertical").and_then(FloatAlign::from_ooxml),
610        dist: dist("mso-wrap-distance-left").max(dist("mso-wrap-distance-right")),
611        dist_top: dist("mso-wrap-distance-top"),
612        dist_bottom: dist("mso-wrap-distance-bottom"),
613        hidden,
614        // Word for Mac draws a VML shape anchored in a header table cell on the page whatever
615        // `o:allowincell` says (user-checked), as for Word's picture and text watermarks.
616        outside_cell: in_hf,
617        ..Float::default()
618    };
619    f.set_spin(vml_spin(style));
620    f
621}

623/// Word's text watermark (Design › Watermark): a VML shape in a header whose `v:textpath` draws
624/// its `string` as text (ECMA-376 Part 4, VML `textpath`), in the textpath's `font-family`, filled
625/// with the shape's `fillcolor`, semitransparent when its `v:fill` has an `opacity` below 1, and
626/// diagonal when the shape is turned. Only Word's watermark shape counts (its `id`), or text
627/// placed absolutely and centred both ways; other header WordArt (a logo or a title) is None.
628fn vml_watermark(shape: &El) -> Option<crate::doc::Watermark> {
629    let tp = shape.child("v:textpath")?;
630    if tp.attr("on").is_some_and(|v| matches!(v, "f" | "false" | "0")) {
631        return None;
632    }
633    let style = shape.attr("style").unwrap_or("");
634    let centred = |k: &str| vml_style(style, k).is_some_and(|v| v.eq_ignore_ascii_case("center"));
635    let word = shape.attr("id").is_some_and(|id| id.starts_with("PowerPlusWaterMarkObject"));
636    let placed = vml_style(style, "position").is_some_and(|p| p.eq_ignore_ascii_case("absolute"))
637        && centred("mso-position-horizontal")
638        && centred("mso-position-vertical");
639    if !word && !placed {
640        return None;
641    }
642    let text: String = tp.attr("string")?.chars().take(200).collect();
643    if text.trim().is_empty() {
644        return None;
645    }
646    let default = crate::doc::Watermark::default();
647    let font = tp
648        .attr("style")
649        .and_then(|s| vml_style(s, "font-family"))
650        .map(|f| css_string(f).chars().take(100).collect::<String>())
651        .filter(|f| !f.is_empty())
652        .unwrap_or(default.font);
653    let color = shape.attr("fillcolor").and_then(vml_color).unwrap_or(default.color);
654    let opacity = shape.child("v:fill").and_then(|f| f.attr("opacity")).and_then(vml_opacity).unwrap_or(1.0);
655    // The box the text is stretched to (`fitshape`).
656    let len = |k: &str| vml_style(style, k).and_then(|v| measure(v, 1.0)).filter(|v| v.is_finite() && *v > 0.0).map(|v| v.min(4000.0));
657    let size = len("width").zip(len("height")).map(|(w, h)| [w, h]);
658    Some(crate::doc::Watermark { text, font, color, diagonal: vml_spin(style).deg != 0.0, semitransparent: opacity < 1.0, size })
659}

661/// A VML `opacity`, 0..=1.
662fn vml_opacity(v: &str) -> Option<f32> {
663    vml_fraction(v).map(|f| f.clamp(0.0, 1.0))
664}

666/// A VML shape's turn: `rotation` in degrees, or 65536ths of one with an `fd` suffix; `flip`: `x`
667/// and/or `y`.
668fn vml_spin(style: &str) -> crate::geom::Spin {
669    let get = |k: &str| vml_style(style, k);
670    let rot = get("rotation").and_then(|v| match v.strip_suffix("fd") {
671        Some(fd) => fd.trim().parse::<f32>().ok().map(|f| f / 65_536.0),
672        None => v.trim().parse::<f32>().ok(),
673    });
674    let flip = get("flip").unwrap_or("");
675    crate::geom::Spin::new(rot.unwrap_or(0.0), flip.contains('x'), flip.contains('y'))
676}
