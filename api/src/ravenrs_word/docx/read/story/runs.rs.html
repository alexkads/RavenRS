9    pub(super) fn read_inline_children(&mut self, sc: &mut StoryCtx, pb: &mut PB, e: &El, rels: &Rels, ctx: &RunCtx, depth: usize) {
10        if depth > crate::docx::xml::MAX_DEPTH {

12        }
13        for k in e.els() {

18    /// Read an inline child element (`w:r`, `m:oMath`, `m:oMathPara`, `w:ins`, etc.).
19    fn read_inline(&mut self, sc: &mut StoryCtx, pb: &mut PB, k: &El, rels: &Rels, ctx: &RunCtx, depth: usize) {
20        match k.name.as_str() {
21            "w:r" => self.read_run(sc, pb, k, rels, ctx, depth),
22            "w:hyperlink" => {
23                let mut c = ctx.clone();
24                let url = k.attr("r:id").and_then(|id| rels.by_id(id)).map(|r| r.target.trim().to_string()).filter(|u| !u.is_empty());
25                let anchor = k.attr("w:anchor").filter(|a| !a.is_empty());
26                c.link = match (url, anchor) {
27                    (Some(u), Some(a)) => Some(format!("{u}#{a}")),
28                    (Some(u), None) => Some(u),
29                    (None, Some(a)) => Some(format!("#{a}")),
30                    (None, None) => ctx.link.clone(),
31                };
32                self.read_inline_children(sc, pb, k, rels, &c, depth + 1);
33            }
34            "w:ins" | "w:moveTo" => {
35                let mut c = ctx.clone();
36                c.ins = Some(if k.name == "w:moveTo" { self.move_revision(true, k, false) } else { self.revision(RevisionKind::Insert, k) });
37                self.read_inline_children(sc, pb, k, rels, &c, depth + 1);
38            }
39            "w:del" | "w:moveFrom" => {
40                let mut c = ctx.clone();
41                c.del = Some(if k.name == "w:moveFrom" { self.move_revision(false, k, false) } else { self.revision(RevisionKind::Delete, k) });
42                self.read_inline_children(sc, pb, k, rels, &c, depth + 1);
43            }
44            "w:moveFromRangeStart" | "w:moveFromRangeEnd" | "w:moveToRangeStart" | "w:moveToRangeEnd" => self.move_range(k),
45            "w:fldSimple" => {
46                let instr = k.attr("w:instr").unwrap_or("").to_string();
47                let props = k.child("w:r").and_then(|r| r.child("w:rPr")).map(|p| self.run_props(p, ctx)).unwrap_or_else(|| self.run_props_none(ctx));
48                self.begin_field(sc, props, instr, on_off_attr(k, "w:fldLock"), on_off_attr(k, "w:dirty"), String::new());
49                self.separate_field(sc, pb);
50                self.read_inline_children(sc, pb, k, rels, ctx, depth + 1);
51                self.end_field(sc, pb);
52            }
53            "w:bookmarkStart" | "w:bookmarkEnd" | "w:commentRangeStart" | "w:commentRangeEnd" | "w:permStart" | "w:permEnd" => {
54                if let Some(o) = self.marker(k) {
55                    let props = self.run_props_none(ctx);
56                    // Markers go straight into the paragraph (never into a field's cached result).
57                    pb.push_obj(o, &props);
58                }
59            }
60            "w:sdt" => {
61                let Some(c) = k.child("w:sdtContent") else { return };
62                let mut ctl = crate::docx::read::sdt::sdt_pr(k.child("w:sdtPr"), false);
63                ctl.end_pr_xml = crate::docx::read::sdt::sdt_end_pr(k.child("w:sdtEndPr"));
64                ctl.run_props = k.child("w:sdtPr").and_then(|pr| pr.child("w:rPr")).map(|r| self.pc.rpr(r));
65                // The markers take the formatting of the content's first run (typing at the
66                // start of the content continues it).
67                let props = c.find("w:rPr").map(|p| self.run_props(p, ctx)).unwrap_or_else(|| self.run_props_none(ctx));
68                let (at, k0) = (pb.text.len(), pb.objects.len());
69                pb.push_obj(InlineObject::ControlStart { control: Box::new(ctl) }, &props);
70                self.read_inline_children(sc, pb, c, rels, ctx, depth + 1);
71                if let Some(InlineObject::ControlStart { control }) = pb.objects.get_mut(k0)
72                    && control.showing_placeholder
73                {
74                    let shown: String = pb.text.get(at..).unwrap_or("").chars().filter(|c| *c != crate::doc::para::OBJ).collect();
75                    control.placeholder_text = placeholder_of(&shown);
76                }
77                let end_props = pb.runs.last().map(|r| r.props.clone()).unwrap_or(props);
78                pb.push_obj(InlineObject::ControlEnd, &end_props);
79            }
80            "w:smartTag" | "w:customXml" | "w:dir" | "w:bdo" => self.read_inline_children(sc, pb, k, rels, ctx, depth + 1),
81            "m:oMath" => {
82                let props = self.run_props_none(ctx);
83                self.emit_omath(sc, pb, k, Default::default(), false, &props);
84            }
85            "m:oMathPara" => {
86                let base_props = self.run_props_none(ctx);
87                let jc = crate::docx::read::math::para_jc(k);
88                for c in k.els() {
89                    match c.name.as_str() {
90                        "m:oMath" => self.emit_omath(sc, pb, c, jc, true, &base_props),
91                        "w:ins" | "w:moveTo" => {
92                            let mut sub_ctx = ctx.clone();
93                            sub_ctx.ins =
94                                Some(if c.name == "w:moveTo" { self.move_revision(true, c, false) } else { self.revision(RevisionKind::Insert, c) });
95                            let sub_props = self.run_props_none(&sub_ctx);
96                            for m in c.children("m:oMath") {
97                                self.emit_omath(sc, pb, m, jc, true, &sub_props);
98                            }
99                        }
100                        "w:del" | "w:moveFrom" => {
101                            let mut sub_ctx = ctx.clone();
102                            sub_ctx.del = Some(if c.name == "w:moveFrom" {
103                                self.move_revision(false, c, false)
104                            } else {
105                                self.revision(RevisionKind::Delete, c)
106                            });
107                            let sub_props = self.run_props_none(&sub_ctx);
108                            for m in c.children("m:oMath") {
109                                self.emit_omath(sc, pb, m, jc, true, &sub_props);
110                            }
111                        }
112                        _ => {}
113                    }
114                }
115            }
116            "mc:AlternateContent" => {
117                if let Some((o, link)) = self.read_alternate(sc, k, rels) {
118                    let props = self.run_props_none(ctx);
119                    self.emit_linked_obj(sc, pb, o, &props, link);
120                } else if let Some(c) = children_of_choice(k) {
121                    self.read_inline_children(sc, pb, c, rels, ctx, depth + 1);
122                }
123            }
124            // Some generators write a break directly under the paragraph instead of inside a run.
125            "w:br" | "w:cr" => {
126                let props = self.run_props_none(ctx);
127                self.read_run_child(sc, pb, k, (&props, rels), &mut None, depth + 1);
128            }
129            _ => {}
130        }
131    }

133    fn run_props(&mut self, rpr: &El, ctx: &RunCtx) -> CharProps {
134        let mut p = self.pc.rpr(rpr);
135        p.link = ctx.link.clone();
136        p.ins = ctx.ins;
137        p.del = ctx.del;
138        p.fmt_change = self.rpr_change(rpr);
139        p
140    }

142    fn run_props_none(&self, ctx: &RunCtx) -> CharProps {
143        CharProps { link: ctx.link.clone(), ins: ctx.ins, del: ctx.del, ..Default::default() }
144    }

146    fn read_run(&mut self, sc: &mut StoryCtx, pb: &mut PB, r: &El, rels: &Rels, ctx: &RunCtx, depth: usize) {
147        let props = match r.child("w:rPr") {
148            Some(p) => self.run_props(p, ctx),
149            None => self.run_props_none(ctx),
150        };
151        let mut note: Option<(NoteKind, u32, Option<String>)> = None;
152        for k in r.els() {
153            self.read_run_child(sc, pb, k, (&props, rels), &mut note, depth + 1);
154        }
155        if let Some((kind, id, custom)) = note.take() {
156            self.emit_obj(sc, pb, InlineObject::NoteRef { kind, id, custom: custom.unwrap_or_default() }, &props);
157        }
158    }

160    fn read_run_child(
161        &mut self,
162        sc: &mut StoryCtx,
163        pb: &mut PB,
164        k: &El,
165        (props, rels): (&CharProps, &Rels),
166        note: &mut Option<(NoteKind, u32, Option<String>)>,
167        depth: usize,
168    ) {
169        if depth > crate::docx::xml::MAX_DEPTH {
170            return;
171        }
172        // A custom footnote mark: the text right after the reference is the mark.
173        if let Some((_, _, Some(custom))) = note.as_mut()
174            && k.name == "w:t"
175        {
176            custom.push_str(&clean_text(&k.text()));
177            return;
178        }
179        if k.name != "w:rPr"
180            && let Some((kind, id, custom)) = note.take()
181        {
182            self.emit_obj(sc, pb, InlineObject::NoteRef { kind, id, custom: custom.unwrap_or_default() }, props);
183        }
184        match k.name.as_str() {
185            "w:t" | "w:delText" => {
186                let t = clean_text(&k.text());
187                self.emit_text(sc, pb, &t, props);
188            }
189            "w:instrText" | "w:delInstrText" => {
190                if let Some(f) = sc.fields.last_mut()
191                    && f.phase == Phase::Instr
192                    && !f.spilled
193                    && f.instr.len() < MAX_INSTR
194                {
195                    if f.sizes.len() < 64 {
196                        f.sizes.push((f.instr.len(), props.size));
197                    }
198                    f.instr.push_str(&k.text());
199                }
200            }
201            "w:tab" => self.emit_text(sc, pb, "\t", props),
202            // Alignment and base are required (ECMA-376 §17.3.3.23); without them it stays a plain tab.
203            "w:ptab" => match (k.attr("w:alignment").and_then(PTabAlign::from_ooxml), k.attr("w:relativeTo").and_then(PTabBase::from_ooxml)) {
204                (Some(align), Some(relative_to)) => {
205                    let leader = crate::docx::read::props::tab_leader(k.attr("w:leader").unwrap_or("none"));
206                    self.emit_obj(sc, pb, InlineObject::PositionalTab { align, relative_to, leader }, props)
207                }
208                _ => self.emit_text(sc, pb, "\t", props),
209            },
210            "w:br" => {
211                if sc.fields.iter().any(|f| f.phase == Phase::Instr) {
212                    return;
213                }
214                let ty = k.attr("w:type").map(|t| t.trim().to_ascii_lowercase());
215                let c = match ty.as_deref() {
216                    Some("page") => "\u{000C}",
217                    Some("column") => "\u{000E}",
218                    _ => "\n",
219                };
220                self.emit_text(sc, pb, c, props);
221            }
222            "w:cr" if !sc.fields.iter().any(|f| f.phase == Phase::Instr) => {
223                self.emit_text(sc, pb, "\n", props);
224            }
225            "w:cr" => {}
226            "w:noBreakHyphen" => self.emit_text(sc, pb, "\u{2011}", props),
227            "w:softHyphen" => self.emit_text(sc, pb, "\u{00AD}", props),
228            "w:sym" => {
229                let h_str = k.attr("w:char").map(|h| {
230                    let s = h.trim();
231                    let s = s.strip_prefix("0x").or_else(|| s.strip_prefix("0X")).unwrap_or(s);
232                    let s = s.strip_prefix("U+").or_else(|| s.strip_prefix("u+")).unwrap_or(s);
233                    s.strip_prefix("#x").or_else(|| s.strip_prefix("#X")).unwrap_or(s)
234                });
235                let ch = h_str.and_then(|h| u32::from_str_radix(h, 16).ok()).and_then(char::from_u32).filter(|c| *c as u32 >= 0x20);
236                if let Some(ch) = ch {
237                    let mut p = props.clone();
238                    // The symbol's own font, kept apart from the run's (ECMA-376 §17.3.3.30).
239                    if let Some(f) = k.attr("w:font").filter(|f| !f.trim().is_empty()) {
240                        p.sym_font = Some(f.to_string());
241                    }
242                    self.emit_text(sc, pb, &ch.to_string(), &p);
243                }
244            }
245            "w:fldChar" => match k.attr("w:fldCharType") {
246                Some("begin") => {
247                    let form = k.child("w:ffData").map(|f| f.to_xml()).filter(|x| x.len() <= MAX_FORM).unwrap_or_default();
248                    self.begin_field(sc, props.clone(), String::new(), on_off_attr(k, "w:fldLock"), on_off_attr(k, "w:dirty"), form)
249                }
250                Some("separate") => self.separate_field(sc, pb),
251                Some("end") => self.end_field(sc, pb),
252                _ => {}
253            },
254            "w:drawing" => {
255                if let Some(o) = self.read_drawing(sc, k, rels) {
256                    self.emit_linked_obj(sc, pb, o, props, Self::drawing_link(k, rels));
257                }
258            }
259            "w:pict" => {
260                if let Some(o) = self.read_signature_line(sc, k, rels) {
261                    self.emit_obj(sc, pb, o, props);
262                } else {
263                    for o in self.read_vml_all(sc, k, rels) {
264                        self.emit_obj(sc, pb, o, props);
265                    }
266                }
267            }
268            "w:object" => {
269                if let Some(o) = self.read_object(sc, k, rels) {
270                    self.emit_obj(sc, pb, o, props);
271                }
272            }
273            "w:contentPart" => {
274                if let Some(o) = self.read_ink_content_part(k, rels) {
275                    self.emit_obj(sc, pb, o, props);
276                }
277            }
278            "w:footnoteReference" | "w:endnoteReference" => {
279                let foot = k.local() == "footnoteReference";
280                let fid = k.attr("w:id").and_then(int).unwrap_or(i64::MIN);
281                let map = if foot { &self.footnotes } else { &self.endnotes };
282                if let Some(id) = map.get(&fid).copied() {
283                    let kind = if foot { NoteKind::Footnote } else { NoteKind::Endnote };
284                    let custom = on_off_attr(k, "w:customMarkFollows").then(String::new);
285                    *note = Some((kind, id, custom));
286                }
287            }
288            // The note's own number at the start of its text: a reference to the note being read.
289            "w:footnoteRef" | "w:endnoteRef" => {
290                let kind = if k.local() == "footnoteRef" { NoteKind::Footnote } else { NoteKind::Endnote };
291                if let Some((nk, id)) = self.current_note
292                    && nk == kind
293                {
294                    *note = Some((kind, id, None));
295                }
296            }
297            // The lines of a note separator (only there).
298            "w:separator" | "w:continuationSeparator" if self.in_note_sep => {
299                self.emit_obj(sc, pb, InlineObject::note_separator(k.name == "w:continuationSeparator"), props);
300            }
301            "w:commentReference" => {
302                if let Some(cid) = k.attr("w:id").and_then(|i| self.comment_id(i))
303                    && !self.comments_ended.contains(&cid)
304                {
305                    self.comments_ended.insert(cid);
306                    pb.push_obj(InlineObject::CommentEnd { id: cid }, props);
307                }
308            }
309            "w:pgNum" => self.emit_obj(
310                sc,
311                pb,
312                InlineObject::Field { instr: "PAGE".into(), result: String::new(), locked: false, dirty: false, code: false, form: String::new() },
313                props,
314            ),
315            "w:ruby" => self.read_ruby(sc, pb, k, props),
316            "mc:AlternateContent" => {
317                if let Some((o, link)) = self.read_alternate(sc, k, rels) {
318                    self.emit_linked_obj(sc, pb, o, props, link);
319                } else if let Some(c) = children_of_choice(k) {
320                    for x in c.els() {
321                        self.read_run_child(sc, pb, x, (props, rels), note, depth + 1);
322                    }
323                }
324            }
325            _ => {}
326        }
327    }

329    /// `w:ruby` (ECMA-376 §17.3.3.25): ruby text (`w:rt`) over base text (`w:rubyBase`), kept as
330    /// one [`InlineObject::Ruby`] in the base text's formatting (the first base run's).
331    fn read_ruby(&mut self, sc: &mut StoryCtx, pb: &mut PB, k: &El, props: &CharProps) {
332        const MAX: usize = 4096;
333        // The text of a part's runs and the formatting of its first run.
334        let part = |name: &str| -> (String, Option<CharProps>) {
335            let mut text = String::new();
336            let mut first = None;
337            for r in k.child(name).map(|e| e.children("w:r").collect::<Vec<_>>()).unwrap_or_default() {
338                if first.is_none() {
339                    first = Some(r.child("w:rPr").map(|p| self.pc.rpr(p)).unwrap_or_default());
340                }
341                for t in r.els().filter(|t| matches!(t.name.as_str(), "w:t" | "w:delText")) {
342                    if text.len() < MAX {
343                        text.push_str(&clean_text(&t.text()).replace('\t', " "));
344                    }
345                }
346            }
347            (text.chars().take(MAX).collect(), first)
348        };
349        let (base, base_props) = part("w:rubyBase");
350        let (ruby, ruby_props) = part("w:rt");
351        let mut bp = base_props.unwrap_or_default();
352        (bp.link, bp.ins, bp.del) = (props.link.clone(), props.ins, props.del);
353        if base.is_empty() {
354            return;
355        }
356        if ruby.is_empty() {
357            self.emit_text(sc, pb, &base, &bp);
358            return;
359        }
360        let pr = k.child("w:rubyPr");
361        let val = |n: &str| pr.and_then(|p| p.child(n)).and_then(|e| e.attr("w:val"));
362        let pt = |n: &str| val(n).and_then(|v| measure(v, 2.0)).filter(|v| v.is_finite() && *v >= 0.0);
363        let mut rp = ruby_props.unwrap_or_default();
364        let base_size = pt("w:hpsBaseText").filter(|v| *v > 0.0).or(bp.size).unwrap_or(0.0).min(1584.0);
365        let size = pt("w:hps").filter(|v| *v > 0.0).or(rp.size).unwrap_or(if base_size > 0.0 { base_size / 2.0 } else { 5.0 }).clamp(1.0, 1584.0);
366        let raise = pt("w:hpsRaise").unwrap_or_else(|| crate::doc::para::ruby_raise(0.0, base_size.max(size), size)).min(1584.0);
367        // The ruby's size is `size`; its run keeps what it sets beyond the base text's formatting
368        // (writers repeat the base's).
369        (rp.size, rp.size_cs) = (None, None);
370        let rp = rp.minus(&bp);
371        let obj = InlineObject::Ruby {
372            base,
373            ruby,
374            align: val("w:rubyAlign").and_then(crate::doc::para::RubyAlign::from_ooxml).unwrap_or_default(),
375            size,
376            raise,
377            base_size,
378            lang: val("w:lid").map(|l| l.chars().take(32).collect()).unwrap_or_default(),
379            props: Box::new(rp),
380        };
381        self.emit_obj(sc, pb, obj, &bp);
382    }

384    pub(super) fn emit_text(&mut self, sc: &mut StoryCtx, pb: &mut PB, s: &str, props: &CharProps) {
385        match sink(sc) {
386            Some(f) => {
387                if f.phase == Phase::Result && f.buf.len() < 100_000 {
388                    f.buf.push((Item::Text(s.to_string()), props.clone()));
389                }
390            }
391            _ => pb.push_text(s, props),
392        }
393    }

395    /// [`Self::emit_obj`] for a drawing that carries its own `link`: a link around the run wins.
396    fn emit_linked_obj(&mut self, sc: &mut StoryCtx, pb: &mut PB, o: InlineObject, props: &CharProps, link: Option<String>) {
397        match link.filter(|_| props.link.is_none()) {
398            Some(link) => self.emit_obj(sc, pb, o, &CharProps { link: Some(link), ..props.clone() }),
399            None => self.emit_obj(sc, pb, o, props),
400        }
401    }

403    /// Read an `m:oMath` element and emit its [`InlineObject::Equation`].
404    fn emit_omath(&mut self, sc: &mut StoryCtx, pb: &mut PB, m: &El, jc: MathJc, display: bool, props: &CharProps) {
405        let mut math = crate::docx::read::math::read_omath(m, jc);
406        let mut props = props.clone();
407        if let Some(color) = crate::docx::read::math::lift_color(&mut math) {
408            props.color = Some(TextColor::Rgb(color));
409        }
410        let linear = crate::doc::math::to_linear(&math.nodes);
411        self.emit_obj(sc, pb, InlineObject::Equation { linear, display, math }, &props);
412    }

414    pub(super) fn emit_obj(&mut self, sc: &mut StoryCtx, pb: &mut PB, o: InlineObject, props: &CharProps) {
415        match sink(sc) {
416            Some(f) => {
417                if f.phase == Phase::Result && f.buf.len() < 100_000 {
418                    f.buf.push((Item::Obj(Box::new(o)), props.clone()));
419                }
420            }
421            _ => pb.push_obj(o, props),
422        }
423    }
