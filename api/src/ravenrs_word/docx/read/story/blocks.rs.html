8    /// Read a story's blocks from a container element (`w:body`, `w:hdr`, `w:tc`, `w:txbxContent`…).
9    pub fn read_blocks(&mut self, sc: &mut StoryCtx, parent: &El, rels: &Rels, out: &mut Blocks, depth: usize) {
10        if depth > crate::docx::xml::MAX_DEPTH {
11            return;
12        }
13        for e in parent.els() {
14            match e.name.as_str() {
15                "w:p" => {
16                    let p = self.read_para(sc, e, rels, depth);
17                    // A framed drop-cap paragraph merges into the paragraph it starts.
18                    if let Some(Block::Para(prev)) = out.last().map(|b| &**b)
19                        && prev.props.drop_cap.is_some()
20                        && prev.section.is_none()
21                        && prev.text.len() <= 16
22                        && p.props.drop_cap.is_none()
23                        && let Some(Block::Para(mut first)) = out.pop().map(|b| (*b).clone())
24                    {
25                        let mut props = p.props.clone();
26                        props.drop_cap = first.props.drop_cap;
27                        props.drop_cap_margin = first.props.drop_cap_margin;
28                        props.drop_cap_distance = first.props.drop_cap_distance;
29                        // A letter in a font of its own: the drop cap's font, the letter itself
30                        // in the text's (so taking the drop cap away gives the letter its font back).
31                        let cap_font = first.runs.first().and_then(|r| r.props.font.clone());
32                        let text_font = p.runs.first().and_then(|r| r.props.font.clone());
33                        if cap_font.is_some() && cap_font != text_font {
34                            for r in first.runs.iter_mut().filter(|r| r.props.font == cap_font) {
35                                r.props.font = text_font.clone();
36                            }
37                            props.drop_cap_font = cap_font;
38                        }
39                        first.append(p);
40                        first.props = props;
41                        out.push(para_block(first));
42                        continue;
43                    }
44                    out.push(para_block(p));
45                }
46                "w:tbl" => {
47                    if depth / 4 >= MAX_TABLE_DEPTH {
48                        out.push(para_block(Paragraph::with_text(&clean_text(&e.deep_text()), CharProps::default())));
49                    } else if let Some(t) = self.read_table(sc, e, rels, depth) {
50                        out.push(Arc::new(Block::Table(t)));
51                    }
52                }
53                "w:sdt" => self.read_block_sdt(sc, e, rels, out, depth),
54                "w:altChunk" => self.read_alt_chunk(sc, e, rels, out, depth),
55                "w:customXml" | "w:smartTag" | "w:ins" | "w:moveTo" => self.read_blocks(sc, e, rels, out, depth + 1),
56                // Math is allowed straight in a story (EG_BlockLevelElts takes EG_RunLevelElements,
57                // which takes EG_MathContent): it gets a paragraph of its own (#1273).
58                "m:oMathPara" | "m:oMath" => {
59                    let wrapped = El { name: "w:p".into(), attrs: Vec::new(), kids: vec![Node::El(e.clone())] };
60                    let p = self.read_para(sc, &wrapped, rels, depth);
61                    out.push(para_block(p));
62                }
63                "mc:AlternateContent" => {
64                    if let Some(c) = children_of_choice(e) {
65                        self.read_blocks(sc, c, rels, out, depth + 1);
66                    }
67                }
68                "w:bookmarkStart" | "w:bookmarkEnd" | "w:commentRangeStart" | "w:commentRangeEnd" | "w:permStart" | "w:permEnd" => {
69                    if let Some(o) = self.marker(e) {
70                        sc.pending.push(o);
71                    }
72                }
73                "w:moveFromRangeStart" | "w:moveFromRangeEnd" | "w:moveToRangeStart" | "w:moveToRangeEnd" => self.move_range(e),
74                _ => {}
75            }
76        }
77    }

79    /// A block-level content control: its content's blocks, between a start marker at the start
80    /// of the first paragraph and an end marker at the end of the last (a table at either end
81    /// carries the control instead).
82    fn read_block_sdt(&mut self, sc: &mut StoryCtx, e: &El, rels: &Rels, out: &mut Blocks, depth: usize) {
83        let Some(content) = e.child("w:sdtContent") else { return };
84        let mut ctl = crate::docx::read::sdt::sdt_pr(e.child("w:sdtPr"), true);
85        ctl.end_pr_xml = crate::docx::read::sdt::sdt_end_pr(e.child("w:sdtEndPr"));
86        ctl.run_props = e.child("w:sdtPr").and_then(|pr| pr.child("w:rPr")).map(|r| self.pc.rpr(r));
87        let mut inner = Blocks::new();
88        self.read_blocks(sc, content, rels, &mut inner, depth + 1);
89        // A table of contents' control is handed to the TOC later, when the writer puts it back
90        // (`toc_control_to_toc`).
91        if inner.is_empty() {
92            out.extend(inner);
93            return;
94        }
95        if ctl.showing_placeholder
96            && let Some(Block::Para(p)) = inner.first().map(|b| &**b)
97        {
98            ctl.placeholder_text = placeholder_of(&p.plain_text());
99        }
100        if let Some(first) = inner.first_mut() {
101            match Arc::make_mut(first) {
102                Block::Para(p) => {
103                    let props = p.props_of_char(0).clone();
104                    let _ = p.insert_object(0, InlineObject::ControlStart { control: Box::new(ctl) }, &props);
105                }
106                Block::Table(t) => t.controls.open.insert(0, ctl),
107            }
108        }
109        if let Some(last) = inner.last_mut() {
110            match Arc::make_mut(last) {
111                Block::Para(p) => {
112                    let (at, props) = (p.len(), p.runs.last().map(|r| r.props.clone()).unwrap_or_else(|| p.mark.clone()));
113                    let _ = p.insert_object(at, InlineObject::ControlEnd, &props);
114                }
115                Block::Table(t) => t.controls.close = t.controls.close.saturating_add(1),
116            }
117        }
118        out.extend(inner);
119    }

121    /// Markers left over at the end of a story go to its last paragraph (after a final table:
122    /// the last paragraph of its last cell, so a bookmark around the table keeps its end).
123    pub fn flush_pending(&mut self, sc: &mut StoryCtx, out: &mut Blocks) {
124        if sc.pending.is_empty() {
125            return;
126        }
127        if let Some(p) = last_para_mut(out, 0) {
128            for o in sc.pending.drain(..) {
129                let props = p.runs.last().map(|r| r.props.clone()).unwrap_or_default();
130                p.text.push(crate::doc::para::OBJ);
131                p.objects.push(o);
132                p.runs.push(Run { len: crate::doc::para::OBJ.len_utf8(), props });
133            }
134            p.normalize();
135        }
136        sc.pending.clear();
137    }

139    /// Markers read before content placed whole (a chunk's `next`) stay before it: at the end of
140    /// the paragraph before, or after a table or with none, at the start of the content's first
141    /// paragraph, as a paragraph read there would take them (with a table first, the next
142    /// paragraph read takes them).
143    pub fn flush_pending_before(&mut self, sc: &mut StoryCtx, out: &mut Blocks, next: &mut [Block]) {
144        if out.last().is_some_and(|b| matches!(**b, Block::Para(_))) {
145            return self.flush_pending(sc, out);
146        }
147        if let Some(Block::Para(p)) = next.first_mut() {
148            for o in sc.pending.drain(..).rev() {
149                let _ = p.insert_object(0, o, &CharProps::default());
150            }
151        }
152    }

154    pub(super) fn marker(&mut self, e: &El) -> Option<InlineObject> {
155        let id = e.attr("w:id").map(str::trim).unwrap_or("").to_string();
156        match e.name.as_str() {
157            "w:bookmarkStart" => {
158                if self.bookmarks.len() >= 100_000 {
159                    return None;
160                }
161                let name = e.attr("w:name").map(str::trim).unwrap_or("").to_string();
162                if name.is_empty() || name.len() > 256 {
163                    return None;
164                }
165                self.bookmarks.insert(id, name.clone());
166                Some(InlineObject::BookmarkStart { name })
167            }
168            "w:bookmarkEnd" => self
169                .bookmarks
170                .get(&id)
171                .cloned()
172                .or_else(|| e.attr("w:name").map(str::trim).filter(|n| !n.is_empty() && n.len() <= 256).map(str::to_string))
173                .map(|name| InlineObject::BookmarkEnd { name }),
174            "w:commentRangeStart" => self.comment_id(&id).map(|id| InlineObject::CommentStart { id }),
175            "w:permStart" => {
176                // Bounded: a hostile file can't grow the map without end.
177                if self.perms.len() >= 100_000 {
178                    return None;
179                }
180                let id = match self.perms.get(&id) {
181                    Some(n) => *n,
182                    None => {
183                        let n = self.next_perm;
184                        self.next_perm = n.saturating_add(1);
185                        self.perms.insert(id, n);
186                        n
187                    }
188                };
189                let text = |a: &str| e.attr(a).filter(|v| v.len() <= 256).unwrap_or("").to_string();
190                let (group, editor) = (text("w:edGrp"), text("w:ed"));
191                if group.is_empty() && editor.is_empty() {
192                    return None;
193                }
194                Some(InlineObject::PermStart { id, group, editor })
195            }
196            "w:permEnd" => self.perms.get(&id).map(|id| InlineObject::PermEnd { id: *id }),
197            "w:commentRangeEnd" => {
198                let cid = self.comment_id(&id)?;
199                self.comments_ended.insert(cid);
200                Some(InlineObject::CommentEnd { id: cid })
201            }
202            _ => None,
203        }
204    }

210    fn read_para(&mut self, sc: &mut StoryCtx, e: &El, rels: &Rels, depth: usize) -> Paragraph {
211        let mut pb = PB::default();
212        let mut para = Paragraph::new();
213        let mut mark_moves: [Option<&El>; 2] = [None, None];
214        self.moves.closed_here = [None, None];
215        for o in std::mem::take(&mut sc.pending) {
216            pb.push_obj(o, &CharProps::default());
217        }
218        if let Some(ppr) = e.child("w:pPr") {
219            let (mut props, mark) = self.pc.ppr(ppr);
220            crate::docx::read::props::resolve_level_only(&self.doc.styles, &mut props, "Normal");
221            para.props = props;
222            para.mark = mark;
223            // A tracked paragraph mark (ECMA-376 §17.13.5.16 / §17.13.5.15): `w:ins` / `w:del`
224            // in the mark's `w:rPr`.
225            if let Some(rpr) = ppr.child("w:rPr") {
226                for k in rpr.els() {
227                    match k.name.as_str() {
228                        "w:ins" => para.mark.ins = Some(self.revision(RevisionKind::Insert, k)),
229                        "w:del" => para.mark.del = Some(self.revision(RevisionKind::Delete, k)),
230                        // A moved paragraph mark (§17.13.5.21, §17.13.5.24): its move range
231                        // usually starts in the paragraph's content, so it is read after that.
232                        "w:moveTo" => mark_moves[1] = Some(k),
233                        "w:moveFrom" => mark_moves[0] = Some(k),
234                        _ => {}
235                    }
236                }
237                para.mark.fmt_change = self.rpr_change(rpr);
238            }
239            // Tracked formatting changes (§17.13.5.29, §17.13.5.19): the paragraph properties
240            // before the change, and a change of the list numbering.
241            if let Some(ch) = ppr.child("w:pPrChange") {
242                let old = ch
243                    .child("w:pPr")
244                    .map(|o| {
245                        let mut old = self.pc.ppr(o).0;
246                        crate::docx::read::props::resolve_level_only(&self.doc.styles, &mut old, "Normal");
247                        old.formatting()
248                    })
249                    .unwrap_or_default();
250                para.props.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, ch), old);
251            }
252            if para.props.numbering.is_some()
253                && let Some(ch) = ppr.child("w:numPr").and_then(|n| n.child("w:numberingChange"))
254            {
255                let original = ch.attr("w:original").unwrap_or("").chars().take(256).collect();
256                para.props.num_change = Some(Box::new(NumChange { rev: self.revision(RevisionKind::Format, ch), original }));
257            }
258            if let Some(s) = ppr.child("w:sectPr") {
259                para.section = Some(Box::new(self.read_section(s, rels)));
260            }
261        }
262        let ctx = RunCtx::default();
263        self.read_inline_children(sc, &mut pb, e, rels, &ctx, depth + 1);
264        self.spill_fields(sc, &mut pb);
265        if let Some(k) = mark_moves[0] {
266            para.mark.del = Some(self.move_revision(false, k, true));
267        }
268        if let Some(k) = mark_moves[1] {
269            para.mark.ins = Some(self.move_revision(true, k, true));
270        }
271        para.text = pb.text;
272        para.runs = pb.runs;
273        para.objects = pb.objects;
274        // Paragraph boundaries inside an instruction are hidden with the field code.
275        // The result paragraph still takes a line, even when its cached text is empty.
276        if sc.fields.iter().any(|f| f.phase == Phase::Instr) && para.plain_text().is_empty() {
277            para.mark.hidden = Some(true);
278        }
279        para.normalize();
280        para
281    }

283    /// A tracked run formatting change in `rpr` (`w:rPrChange`, §17.13.5.31): the formatting
284    /// before it. A change nested inside it is ignored.
285    pub(super) fn rpr_change(&mut self, rpr: &El) -> Option<Box<PropChange<CharProps>>> {
286        let ch = rpr.child("w:rPrChange")?;
287        let old = ch.child("w:rPr").map(|o| self.pc.rpr(o).formatting()).unwrap_or_default();
288        PropChange::boxed(self.revision(RevisionKind::Format, ch), old)
289    }

291    pub fn read_section(&mut self, s: &El, rels: &Rels) -> crate::doc::SectionProps {
292        let (mut sp, refs) = sectpr(&self.pc, s);
293        // §17.13.5.32: the section's properties before a tracked change (no header references).
294        if let Some(ch) = s.child("w:sectPrChange") {
295            let old = ch.child("w:sectPr").map(|o| sectpr(&self.pc, o).0).unwrap_or_default();
296            sp.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, ch), old);
297        }
298        for r in refs {
299            let Some(rel) = rels.by_id(&r.rid) else { continue };
300            if rel.external {
301                continue;
302            }
303            let target = rel.target.clone();
304            let Some(id) = self.load_header_footer(&target, r.footer) else { continue };
305            let set = if r.footer { &mut sp.footers } else { &mut sp.headers };
306            match r.kind.as_str() {
307                "first" => set.first = Some(id),
308                "even" => set.even = Some(id),
309                _ => set.default = Some(id),
310            }
311        }
312        sp
313    }

317    fn read_table(&mut self, sc: &mut StoryCtx, t: &El, rels: &Rels, depth: usize) -> Option<Table> {
318        let mut table = Table::default();
319        if let Some(p) = t.child("w:tblPr") {
320            table.props = self.pc.tblpr(p);
321            if let Some(ch) = p.child("w:tblPrChange") {
322                let old = ch.child("w:tblPr").map(|o| self.pc.tblpr(o)).unwrap_or_default();
323                table.props.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, ch), old);
324            }
325        }
326        if let Some(g) = t.child("w:tblGrid") {
327            table.grid = g
328                .children("w:gridCol")
329                .take(MAX_COLS)
330                .map(|c| crate::docx::units::tw(c, "w:w").unwrap_or(0.0).clamp(0.0, crate::docx::units::MAX_LEN_PT))
331                .collect();
332        }
333        let mut rows = Vec::new();
334        collect(t, "w:tr", &mut rows, &mut Vec::new(), 0);
335        for (tr, row_controls) in rows.into_iter().take(MAX_ROWS) {
336            let mut row = Row { props: tr.child("w:trPr").map(trpr).unwrap_or_default(), cells: Vec::new(), controls: row_controls };
337            row.props.exceptions = tr.child("w:tblPrEx").map(|e| Box::new(self.pc.tblpr(e)));
338            if let Some(ch) = tr.child("w:trPr").and_then(|p| p.child("w:trPrChange")) {
339                let old = ch.child("w:trPr").map(trpr).unwrap_or_default();
340                row.props.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, ch), old);
341            }
342            let mut tcs = Vec::new();
343            collect(tr, "w:tc", &mut tcs, &mut Vec::new(), 0);
344            for (tc, cell_controls) in tcs.into_iter().take(MAX_COLS) {
345                let (mut props, hcont) = tc
346                    .child("w:tcPr")
347                    .map(|p| tcpr(&self.pc, p))
348                    .unwrap_or_else(|| (crate::doc::props::CellProps { span: 1, ..Default::default() }, false));
349                if let Some(ch) = tc.child("w:tcPr").and_then(|p| p.child("w:tcPrChange")) {
350                    let old = ch.child("w:tcPr").map(|o| tcpr(&self.pc, o).0).unwrap_or_default();
351                    props.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, ch), old);
352                }
353                if hcont && let Some(prev) = row.cells.last_mut() {
354                    prev.props.span = (prev.props.span + props.span.max(1)).min(63);
355                    continue;
356                }
357                let mut blocks = Blocks::new();
358                self.read_blocks(sc, tc, rels, &mut blocks, depth + 4);
359                self.flush_pending(sc, &mut blocks);
360                row.cells.push(Cell { props, blocks, controls: cell_controls });
361            }
362            if !row.cells.is_empty() {
363                table.rows.push(row);
364            }
365        }
366        if table.rows.is_empty() {
367            return None;
368        }
369        if table.grid.is_empty() {
370            let cols = table.cols().clamp(1, MAX_COLS);
371            table.grid = vec![72.0; cols];
372        }
373        Some(table)
374    }
