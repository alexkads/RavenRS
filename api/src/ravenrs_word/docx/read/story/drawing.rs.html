10    /// Read a `w:drawing` frame and dispatch its payload to the picture, shape or graphic reader.
11    /// Frame dimensions are converted from EMUs to bounded point values; placement, alternative
12    /// text and drawing ids are carried into the object. Returns `None` when the frame or payload
13    /// cannot be read. Ink written by WordCraft keeps its tool's caps rather than an outline cap.
14    pub(super) fn read_drawing(&mut self, sc: &mut StoryCtx, d: &El, rels: &Rels) -> Option<InlineObject> {
15        let (c, anchored) = match d.child("wp:inline") {
16            Some(c) => (c, false),
17            None => (d.child("wp:anchor")?, true),
18        };
19        let ext = c.child("wp:extent");
20        let dim = |n: &str| ext.and_then(|e| e.attr(n)).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.0).clamp(0.0, crate::docx::units::MAX_LEN_PT);
21        let (w, h) = (dim("cx"), dim("cy"));
22        let (alt, meta) = nv_alt(c.child("wp:docPr"));
23        let mut float = if anchored { anchor_float(c) } else { Float::default() };
24        float.alt = meta;
25        float.hidden = c.child("wp:docPr").is_some_and(|p| on_off_attr(p, "hidden"));
26        if let Some(e) = c.child("wp:effectExtent") {
27            for (slot, n) in float.effect.iter_mut().zip(["l", "t", "r", "b"]) {
28                // Negative for a rotated object narrower than its frame; settled below.
29                *slot = e.attr(n).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.0).clamp(-1584.0, 1584.0);
30            }
31        }
32        let gd = c.child("a:graphic").and_then(|g| g.child("a:graphicData"))?;
33        // Checked by URI before the blip search: a diagram's data can hold a blip further down.
34        let uri = gd.attr("uri").unwrap_or("");
35        if uri == INK_URI {
36            let content = gd.child("w14:contentPart")?;
37            let part = crate::docx::read::part_of(rels, content.attr("r:id")?, INK_REL)?;
38            let pkg = self.pkg;
39            if self.content_types.get_or_insert_with(|| ContentTypes::read(pkg)).of(&part) != Some("application/inkml+xml") {
40                return None;
41            }
42            let source = self.keep_embedded(c.child("a:graphic")?, rels, &[])?;
43            float.set_spin(xfrm_spin(content.child("w14:xfrm")));
44            float.bare_turn = !anchored && float.rot != 0.0 && float.effect == [0.0; 4];
45            let (px, py) = float.spin_pad(w, h);
46            for (e, p) in float.effect.iter_mut().zip([px, py, px, py]) {
47                if *e != 0.0 {
48                    *e = (*e - p).clamp(0.0, 1584.0);
49                }
50            }
51            // The renderer's existing missing-picture box marks ink without a usable preview.
52            // InkML stays opaque: no approximation of its strokes or conversion to freeforms.
53            let items = vec![GraphicItem::Image { rect: [0.0, 0.0, w, h], media: String::new() }];
54            let graphic = Arc::new(Graphic { kind: GraphicKind::Ink, items, w, h, source: Some(source), ..Default::default() });
55            return Some(InlineObject::Graphic { w, h, alt, float, graphic });
56        }
57        let graphic_kind = if uri.ends_with("/chart") {
58            Some(GraphicKind::Chart)
59        } else if uri.ends_with("/diagram") {
60            Some(GraphicKind::Diagram)
61        } else {
62            None
63        };
64        if let Some(kind) = graphic_kind {
65            // Kept so saving writes the chart or diagram back: the frame around it is written
66            // from the object's (possibly moved or resized) size and position.
67            let extra: Vec<String> = match kind {
68                GraphicKind::Chart | GraphicKind::Ink => Vec::new(),
69                GraphicKind::Diagram => self.diagram_drawing_rel(gd, rels).into_iter().collect(),
70            };
71            let source = c.child("a:graphic").and_then(|g| self.keep_embedded(g, rels, &extra));
72            let graphic = self.graphic(kind, gd, rels, w, h, source);
73            float.effect = float.effect.map(|e| e.max(0.0));
74            return Some(InlineObject::Graphic { w, h, alt, float, graphic });
75        }
76        // A textbox may contain a picture; its descendant blip is not the outer payload.
77        let direct_shape = gd.child("wps:wsp").filter(|shape| {
78            // Picture fills still use the image fallback; the shape model has no image fill.
79            shape.child("wps:spPr").and_then(|props| props.child("a:blipFill")).is_none()
80        });
81        let mut obj = if let Some(c) = gd.child("wpc:wpc") {
82            self.read_canvas(sc, c, rels, w, h, float)
83        } else if let Some(g) = gd.child("wpg:wgp") {
84            self.read_group(sc, g, rels, w, h, float)?
85        } else if direct_shape.is_none() && gd.find("a:blip").is_some() {
86            self.read_pic(gd, rels, w, h, alt.clone(), float)?
87        } else {
88            let wsp = direct_shape.or_else(|| gd.find("wps:wsp"))?;
89            let mut obj = self.read_wsp(sc, wsp, rels, w, h, float);
90            // A freeform: whether it's ink WordCraft wrote.
91            if let InlineObject::Shape { freeform: Some(f), extra, .. } = &mut obj
92                && let Some(name) = c.child("wp:docPr").and_then(|p| p.attr("name"))
93                && let Some(tool) = crate::docx::read::freeform::ink_tool(name)
94            {
95                let f = Arc::make_mut(f);
96                f.ink = Some(tool);
97                f.order = crate::docx::read::freeform::ink_order(name);
98                // Ink's round ends belong to its tool, not the optional shape outline. Keep
99                // its canonical model unchanged when reading the round cap the writer adds.
100                extra.cap = Default::default();
101            }
102            obj
103        };
104        if !alt.is_empty() {
105            obj.set_alt_text(&alt);
106        }
107        self.set_shape_id(&mut obj, c.child("wp:docPr").and_then(|p| p.attr("id")));
108        // Word's effect extent also covers a rotated object's overhang; the model keeps that
109        // apart (it follows from the angle), so take it back out.
110        // Shadow or glow room the file didn't record: saving records it (and the turn's room).
111        let fx_room = matches!(&obj, InlineObject::Shape { effects, .. } if effects.extent() != [0.0; 4]);
112        if let Some(f) = obj.float_mut() {
113            // None recorded: Word keeps an inline one's line to the frame (see `Float::bare_turn`).
114            f.bare_turn = !anchored && !fx_room && f.rot != 0.0 && f.effect == [0.0; 4];
115            let (px, py) = f.spin_pad(w, h);
116            for (e, p) in f.effect.iter_mut().zip([px, py, px, py]) {
117                // Zero adds no effects room when rotation contracts this axis.
118                if *e != 0.0 {
119                    *e = (*e - p).clamp(0.0, 1584.0);
120                }
121            }
122        }
123        Some(obj)
124    }

126    /// The hyperlink Word keeps on a drawing itself: its `wp:docPr`'s `a:hlinkClick` (ECMA-376
127    /// Part 1 §20.4.2.5, §21.1.2.3.5) naming an external hyperlink relationship. Word stores a
128    /// picture's link there rather than in a `w:hyperlink` around its run (#5267). Its `#name`
129    /// part, when the address has one, stays part of the address.
130    pub(super) fn drawing_link(d: &El, rels: &Rels) -> Option<String> {
131        let frame = d.child("wp:inline").or_else(|| d.child("wp:anchor"))?;
132        let id = frame.child("wp:docPr")?.child("a:hlinkClick")?.attr("r:id")?;
133        let rel = rels.by_id(id).filter(|r| r.external && crate::docx::package::rel_is(&r.kind, rt::HYPERLINK))?;
134        Some(rel.target.trim().to_string()).filter(|u| !u.is_empty())
135    }

137    /// Native ink or an Office 2016 chart kept with its fallback (`mc:AlternateContent`), and the
138    /// [`Self::drawing_link`] of the choice it was read from, else of its fallback: another
139    /// choice's link is not this object's.
140    pub(super) fn read_alternate(&mut self, sc: &mut StoryCtx, ac: &El, rels: &Rels) -> Option<(InlineObject, Option<String>)> {
141        let (object, choice) = match self.read_ink_alternate(sc, ac, rels) {
142            Some(o) => (o, children_of_choice(ac).filter(|e| e.name == "mc:Choice")),
143            None => (self.read_chartex_alternate(sc, ac, rels)?, chartex_choice(ac)),
144        };
145        let link =
146            [choice, ac.child("mc:Fallback")].into_iter().flatten().filter_map(|e| e.child("w:drawing")).find_map(|d| Self::drawing_link(d, rels));
147        Some((object, link))
148    }

150    /// Keep the selected native-ink branch and its picture fallback as one object. Reading only
151    /// the fallback flattens the ink; reading only the choice loses its preview on the next save.
152    fn read_ink_alternate(&mut self, sc: &mut StoryCtx, ac: &El, rels: &Rels) -> Option<InlineObject> {
153        let choice = children_of_choice(ac).filter(|e| e.name == "mc:Choice")?;
154        let drawing = choice.child("w:drawing")?;
155        if drawing.find("a:graphicData")?.attr("uri") != Some(INK_URI) {
156            return None;
157        }
158        let fallback = ac.child("mc:Fallback");
159        let preview = fallback.and_then(|f| f.child("w:pict")).and_then(|p| self.read_vml(sc, p, rels));
160        let Some(mut object) = self.read_drawing(sc, drawing, rels) else { return preview };
161        if let InlineObject::Graphic { w, h, graphic, .. } = &mut object {
162            let graphic = Arc::make_mut(graphic);
163            if let Some(InlineObject::Image { media, .. }) = preview {
164                graphic.items = vec![GraphicItem::Image { rect: [0.0, 0.0, *w, *h], media }];
165            }
166            if let Some(fallback) = fallback {
167                let kept =
168                    El { name: "mc:AlternateContent".into(), attrs: Vec::new(), kids: vec![Node::El(choice.clone()), Node::El(fallback.clone())] };
169                if let Some(source) = self.keep_embedded(&kept, rels, &[]) {
170                    graphic.source = Some(source);
171                }
172                // A malformed or over-budget fallback must not discard a valid native part.
173            }
174        }
175        Some(object)
176    }

178    /// An Office 2016 chart (`cx:chart`: waterfall, sunburst, box & whisker…) drawn from its
179    /// fallback picture, keeping both branches so saving writes the chart back rather than only
180    /// the picture. `None` (read the fallback alone) when it can't be kept whole.
181    fn read_chartex_alternate(&mut self, sc: &mut StoryCtx, ac: &El, rels: &Rels) -> Option<InlineObject> {
182        let choice = chartex_choice(ac)?;
183        // A requirement in a namespace we can't name couldn't be written back.
184        let requires = choice.attr("Requires").unwrap_or("");
185        if requires.split_whitespace().next().is_none() || requires.split_whitespace().any(|p| p == "?") {
186            return None;
187        }
188        let fallback = ac.child("mc:Fallback")?;
189        let kept = El { name: "mc:AlternateContent".into(), attrs: Vec::new(), kids: vec![Node::El(choice.clone()), Node::El(fallback.clone())] };
190        let source = self.keep_embedded(&kept, rels, &[])?;
191        let Some(InlineObject::Image { media, w, h, alt, float, .. }) = self.read_drawing(sc, fallback.child("w:drawing")?, rels) else {
192            return None;
193        };
194        let items = vec![GraphicItem::Image { rect: [0.0, 0.0, w, h], media }];
195        let graphic = Arc::new(Graphic { kind: GraphicKind::Chart, items, w, h, source: Some(source), ..Default::default() });
196        Some(InlineObject::Graphic { w, h, alt, float, graphic })
197    }

199    /// A group or canvas member that is native ink (`content`, a `w14:contentPart`, inside
200    /// `alternate` when it has a picture fallback): what to write back (the whole alternate when
201    /// it can be kept, else the ink alone) and its fallback picture (a `Document::media` key).
202    /// `None` when the ink's part is missing or isn't InkML.
203    fn ink_member(&mut self, content: &El, alternate: Option<&El>, rels: &Rels) -> Option<(Arc<Embedded>, Option<String>)> {
204        let InlineObject::Graphic { graphic, .. } = self.read_ink_content_part(content, rels)? else { return None };
205        let mut source = graphic.source.clone()?;
206        let mut preview = None;
207        let choice = alternate.and_then(children_of_choice).filter(|c| c.name == "mc:Choice");
208        if let (Some(choice), Some(fallback)) = (choice, alternate.and_then(|a| a.child("mc:Fallback"))) {
209            let kept = El { name: "mc:AlternateContent".into(), attrs: Vec::new(), kids: vec![Node::El(choice.clone()), Node::El(fallback.clone())] };
210            // A malformed or over-budget fallback must not discard a valid native part.
211            if let Some(whole) = self.keep_embedded(&kept, rels, &[]) {
212                source = whole;
213                preview = match fallback.child("pic:pic").and_then(|p| self.read_pic(p, rels, 0.0, 0.0, String::new(), Float::default())) {
214                    Some(InlineObject::Image { media, .. }) => Some(media),
215                    _ => None,
216                };
217            }
218        }
219        Some((source, preview))
220    }

233    /// The chart or SmartArt diagram in an `a:graphicData`, as items in points from its top-left
234    /// corner. Charts are drawn from their cached data, SmartArt diagrams from the drawing Word
235    /// stores beside them. Built once per part and size; once the file's graphic budget is spent,
236    /// later ones are empty.
237    fn graphic(&mut self, kind: GraphicKind, gd: &El, rels: &Rels, w: f32, h: f32, source: Option<Arc<Embedded>>) -> Arc<Graphic> {
238        let part = match kind {
239            GraphicKind::Chart => gd.child("c:chart").and_then(|c| c.attr("r:id")).and_then(|id| crate::docx::read::part_of(rels, id, rt::CHART)),
240            GraphicKind::Diagram => self.diagram_part(gd, rels),
241            GraphicKind::Ink => None,
242        };
243        let Some(path) = part else { return Arc::new(Graphic { kind, items: Vec::new(), w, h, source, chart: None, smart_art: None }) };
244        let key = (kind, path, w.to_bits(), h.to_bits());
245        if let Some(g) = self.graphics.get(&key) {
246            return g.clone();
247        }
248        let items = if self.chunks.graphic == 0 { Vec::new() } else { self.graphic_items(kind, &key.1, w, h) };
249        self.chunks.graphic = self.chunks.graphic.saturating_sub(graphic_work(&items));
250        // A chart or SmartArt that is exactly what WordCraft writes for its model can be edited
251        // (and is written from the model); any other stays as it was read.
252        let chart = match kind {
253            GraphicKind::Chart if !items.is_empty() => {
254                self.graphic_part(&key.1).and_then(|space| crate::docx::chart_spec::spec_of(&space)).map(Arc::new)
255            }
256            _ => None,
257        };
258        let smart_art = match kind {
259            GraphicKind::Diagram if !items.is_empty() => self.smart_art_spec(gd, rels).map(Arc::new),
260            _ => None,
261        };
262        let source = if chart.is_some() || smart_art.is_some() { None } else { source };
263        let g = Arc::new(Graphic { kind, items, w, h, source, chart, smart_art });
264        self.graphics.insert(key, g.clone());
265        g
266    }

280    /// A picture: the first `a:blip` in `pic` and its crop.
281    fn read_pic(&mut self, pic: &El, rels: &Rels, w: f32, h: f32, alt: String, mut float: Float) -> Option<InlineObject> {
282        let blip = pic.find("a:blip")?;
283        float.set_spin(xfrm_spin(pic.find("pic:spPr")));
284        let Some(embed) = blip.attr("r:embed") else {
285            // A linked picture holds no copy: nothing to draw, and nothing is fetched.
286            if let Some(link) = blip.attr("r:link") {
287                let target = rels.by_id(link).map_or_else(|| link.to_string(), |r| r.target.clone());
288                self.note_missing_picture(&target, crate::doc::MissingPictureReason::LinkOnly);
289            }
290            return None;
291        };
292        let media = self.media_for(rels, embed)?;
293        let mut crop = [0.0f32; 4];
294        if let Some(sr) = pic.find("a:srcRect") {
295            for (i, n) in ["l", "t", "r", "b"].iter().enumerate() {
296                if let Some(v) = sr.attr(n).and_then(int)
297                    && let Some(slot) = crop.get_mut(i)
298                {
299                    *slot = (v.clamp(0, 100_000) as f32 / 100_000.0).clamp(0.0, 1.0);
300                }
301            }
302        }
303        Some(InlineObject::Image { media, w, h, alt, float, crop, fx: crate::docx::units::blip_fx(blip), ole: None })
304    }

306    /// A `wpg:wgp` group (`w` × `h`): its pictures, shapes and text boxes, nested groups
307    /// flattened into it. `None` when it has no member we can show.
308    fn read_group(&mut self, sc: &mut StoryCtx, g: &El, rels: &Rels, w: f32, h: f32, mut float: Float) -> Option<InlineObject> {
309        let x = group_xfrm(g.child("wpg:grpSpPr"));
310        float.set_spin(xfrm_spin(g.child("wpg:grpSpPr")));
311        // The members' space; without `a:chExt` it is the group's own size.
312        let (ch_w, ch_h) = match x.ch_ext {
313            Some((cw, ch)) if cw > 0.0 && ch > 0.0 => (cw, ch),
314            _ => (w.max(1.0), h.max(1.0)),
315        };
316        let (cx, cy) = x.ch_off.unwrap_or((0.0, 0.0));
317        let mut children = Vec::new();
318        let outer = self.group_fill.take();
319        self.group_fill = own_fill(g.child("wpg:grpSpPr"), &self.doc.settings.theme_colors, None).flatten();
320        self.group_members(sc, g, rels, (-cx, -cy, 1.0, 1.0), 0, &mut children);
321        self.group_fill = outer;
322        if children.is_empty() {
323            return None;
324        }
325        Some(InlineObject::Group { w, h, float, ch_w, ch_h, children, canvas: None })
326    }

328    /// A `wpc:wpc` Drawing Canvas (`w` × `h`, ECMA-376 Part 1 §20.4 and the Word 2010 canvas
329    /// schema): its `wpc:bg` fill, `wpc:whole` outline and members, which sit in the canvas's
330    /// own space (offsets from its top-left corner, unscaled). An empty canvas is kept.
331    /// The canvas schema has no `a:xfrm` of its own (nor does the anchor), so a turned or
332    /// flipped canvas keeps its turn in our `wpc:extLst` extension ([`CANVAS_SPIN_EXT`]).
333    fn read_canvas(&mut self, sc: &mut StoryCtx, c: &El, rels: &Rels, w: f32, h: f32, mut float: Float) -> InlineObject {
334        let spin_ext = c.child("wpc:extLst").and_then(|l| l.els().find(|e| e.name == "a:ext" && e.attr("uri") == Some(crate::docx::CANVAS_SPIN_EXT)));
335        float.set_spin(xfrm_spin(spin_ext));
336        let solid = |e: Option<&El>| {
337            e.and_then(|f| f.child("a:solidFill")).and_then(|f| f.child("a:srgbClr")).and_then(|c| c.attr("val")).and_then(Rgb::parse)
338        };
339        let ln = c.child("wpc:whole").and_then(|e| e.child("a:ln"));
340        let style = crate::doc::para::CanvasStyle {
341            fill: own_fill(c.child("wpc:bg"), &self.doc.settings.theme_colors, None).flatten(),
342            stroke: solid(ln),
343            stroke_width: ln.and_then(|l| l.attr("w")).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.0).clamp(0.0, 100.0),
344        };
345        let mut children = Vec::new();
346        self.group_members(sc, c, rels, (0.0, 0.0, 1.0, 1.0), 0, &mut children);
347        InlineObject::Group { w, h, float, ch_w: w.max(1.0), ch_h: h.max(1.0), children, canvas: Some(style) }
348    }

350    /// The members of group `g` into `out`, mapped into the outermost group's space by `t`
351    /// (x0, y0, sx, sy: outer = x0 + inner × sx).
352    fn group_members(
353        &mut self,
354        sc: &mut StoryCtx,
355        g: &El,
356        rels: &Rels,
357        t: (f32, f32, f32, f32),
358        depth: usize,
359        out: &mut Vec<crate::doc::para::GroupChild>,
360    ) {
361        let (x0, y0, sx, sy) = t;
362        for e in g.els() {
363            if out.len() >= crate::doc::para::MAX_GROUP_CHILDREN {
364                return;
365            }
366            // Native ink (`w14:contentPart`), alone or with a picture fallback: kept whole, in its
367            // place among the members. Ink that can't be kept gives way to its fallback.
368            let (e, ink) = match e.name.as_str() {
369                "w14:contentPart" => (e, self.ink_member(e, None, rels)),
370                "mc:AlternateContent" => {
371                    let choice = children_of_choice(e);
372                    match choice.filter(|c| c.name == "mc:Choice").and_then(|c| c.els().next()).filter(|c| c.name == "w14:contentPart") {
373                        Some(content) => match self.ink_member(content, Some(e), rels) {
374                            Some(ink) => (content, Some(ink)),
375                            None => match e.child("mc:Fallback").and_then(|f| f.els().next()) {
376                                Some(c) => (c, None),
377                                None => continue,
378                            },
379                        },
380                        None => match choice.and_then(|c| c.els().next()) {
381                            Some(c) => (c, None),
382                            None => continue,
383                        },
384                    }
385                }
386                _ => (e, None),
387            };
388            let sppr = match e.name.as_str() {
389                "wps:wsp" => e.child("wps:spPr"),
390                "pic:pic" => e.child("pic:spPr"),
391                "w14:contentPart" if ink.is_some() => Some(e),
392                // A group inside a canvas is a whole `wpg:wgp`.
393                "wpg:grpSp" | "wpg:wgp" => e.child("wpg:grpSpPr"),
394                _ => continue,
395            };
396            let x = group_xfrm(sppr);
397            let ((ox, oy), (ew, eh)) = (x.off, x.ext);
398            if e.name == "wpg:grpSp" || e.name == "wpg:wgp" {
399                if depth >= 8 {
400                    continue;
401                }
402                // Its members' space maps onto its own box.
403                let (cox, coy) = x.ch_off.unwrap_or((0.0, 0.0));
404                let (cw, ch) = x.ch_ext.filter(|(cw, ch)| *cw > 0.0 && *ch > 0.0).unwrap_or((ew.max(1.0), eh.max(1.0)));
405                let (kx, ky) = (ew / cw, eh / ch);
406                let nt = (x0 + (ox - cox * kx) * sx, y0 + (oy - coy * ky) * sy, sx * kx, sy * ky);
407                // Its own fill, for its members' `a:grpFill` (#1500).
408                // One without a fill of its own passes its group's on.
409                let own = own_fill(sppr, &self.doc.settings.theme_colors, self.group_fill.as_ref()).unwrap_or_else(|| self.group_fill.clone());
410                let outer = std::mem::replace(&mut self.group_fill, own);
411                self.group_members(sc, e, rels, nt, depth + 1, out);
412                self.group_fill = outer;
413                continue;
414            }
415            let max = crate::docx::units::MAX_LEN_PT;
416            let fit = |v: f32, lo: f32| crate::geom::finite(v).clamp(lo, max);
417            let (w, h) = (fit(ew * sx, 0.0), fit(eh * sy, 0.0));
418            let nv = match e.name.as_str() {
419                "pic:pic" => e.find("pic:cNvPr"),
420                "w14:contentPart" => e.child("w14:nvContentPartPr").and_then(|p| p.child("w14:cNvPr")),
421                _ => e.child("wps:cNvPr"),
422            };
423            let (alt, meta) = nv_alt(nv);
424            let float = Float { alt: meta, hidden: nv.is_some_and(|p| on_off_attr(p, "hidden")), ..Float::default() };
425            let obj = if let Some((source, preview)) = ink {
426                let mut float = float;
427                float.set_spin(xfrm_spin(e.child("w14:xfrm")));
428                // As drawing-level ink: its fallback picture, else the missing-picture box.
429                let items = vec![GraphicItem::Image { rect: [0.0, 0.0, w, h], media: preview.unwrap_or_default() }];
430                let graphic = Arc::new(Graphic { kind: GraphicKind::Ink, items, w, h, source: Some(source), ..Default::default() });
431                Some(InlineObject::Graphic { w, h, alt: alt.clone(), float, graphic })
432            } else if e.name == "pic:pic" {
433                self.read_pic(e, rels, w, h, alt.clone(), float)
434            } else {
435                let mut o = self.read_wsp(sc, e, rels, w, h, float);
436                self.set_shape_id(&mut o, e.child("wps:cNvPr").and_then(|p| p.attr("id")));
437                Some(o)
438            };
439            let obj = obj.map(|mut o| {
440                o.set_alt_text(&alt);
441                o
442            });
443            if let Some(obj) = obj {
444                out.push(crate::doc::para::GroupChild { x: fit(x0 + ox * sx, -max), y: fit(y0 + oy * sy, -max), obj });
445            }
446        }
447    }

449    /// Build a shape from `wps:wsp`, using frame dimensions `w` and `h` in points and the caller's
450    /// placement. Reads geometry, paint, effects and connector ends, and creates a text-box story
451    /// when present within the nesting limit. Explicit outline caps are preserved; missing or
452    /// unknown caps use flat ends, without resolving theme-inherited caps.
453    fn read_wsp(&mut self, sc: &mut StoryCtx, wsp: &El, rels: &Rels, w: f32, h: f32, mut float: Float) -> InlineObject {
454        let sppr = wsp.child("wps:spPr");
455        float.set_spin(xfrm_spin(sppr));
456        let prst = sppr.and_then(|s| s.child("a:prstGeom")).and_then(|g| g.attr("prst")).unwrap_or("rect");
457        let txbx = wsp.child("wps:txbx").and_then(|t| t.child("w:txbxContent"));
458        let mut kind = preset_kind(prst);
459        let is_tb = wsp.child("wps:cNvSpPr").and_then(|c| c.attr("txBox")).is_some_and(|v| v == "1" || v == "true");
460        if is_tb || (txbx.is_some() && kind == ShapeKind::Rectangle) {
461            kind = ShapeKind::TextBox;
462        }
463        // Explicit paint, else the theme's through the style references (#611).
464        let paint = shape_paint(sppr, wsp.child("wps:style"), &self.doc.settings.theme_colors, &self.theme_fills, self.group_fill.as_ref());
465        let (stroke, stroke_width) = (paint.stroke, paint.stroke_width);
466        let fill = if kind.is_open() { None } else { paint.fill };
467        let ln = sppr.and_then(|s| s.child("a:ln"));
468        let effects = sppr.and_then(|s| s.child("a:effectLst")).map(effect_list).unwrap_or_default();
469        let story = match txbx {
470            Some(t) if sc.story_depth < MAX_STORY_DEPTH => Some(self.read_textbox(sc, t, rels)),
471            _ => None,
472        };
473        // A box continuing another's text (Create Link): its own, empty story, linked up once the
474        // whole document is read.
475        let linked = wsp.child("wps:linkedTxbx").filter(|_| txbx.is_none());
476        let story = match linked {
477            Some(l) => {
478                kind = ShapeKind::TextBox;
479                let id = l.attr("id").and_then(|v| v.trim().parse::<u32>().ok());
480                let seq = l.attr("seq").and_then(|v| v.trim().parse::<u32>().ok());
481                match (id, seq) {
482                    (Some(id), Some(seq)) if seq > 0 => {
483                        let part = self.doc.add_part(PartKind::TextBox, Blocks::new());
484                        self.txbx_links.push((id, seq, part));
485                        Some(part)
486                    }
487                    _ => story,
488                }
489            }
490            None => {
491                if let (Some(part), Some(id)) = (story, wsp.child("wps:txbx").and_then(|t| t.attr("id")).and_then(|v| v.trim().parse::<u32>().ok())) {
492                    self.txbx_links.push((id, 0, part));
493                }
494                story
495            }
496        };
497        // Text direction and alignment (`wps:bodyPr`).
498        if let (Some(id), Some(bp)) = (story, wsp.child("wps:bodyPr"))
499            && let Some(part) = self.doc.parts.get_mut(&id)
500        {
501            if let Some(v) = bp.attr("vert") {
502                part.body.vert = crate::doc::props::TextVert::from_ooxml(v);
503            }
504            if let Some(a) = bp.attr("anchor") {
505                part.body.anchor = crate::doc::props::valign_from_anchor(a);
506            }
507            // Internal margins, each defaulting to Word's (#743).
508            let mut inset = crate::doc::TEXT_BOX_INSET;
509            for (slot, n) in inset.iter_mut().zip(["lIns", "tIns", "rIns", "bIns"]) {
510                if let Some(v) = bp.attr(n).and_then(int) {
511                    *slot = u32::try_from(v.clamp(0, i64::from(crate::doc::MAX_TEXT_BOX_INSET))).unwrap_or(0);
512                }
513            }
514            part.body.inset = (inset != crate::doc::TEXT_BOX_INSET).then_some(inset);
515            part.body.auto_fit = bp.child("a:spAutoFit").is_some();
516        }
517        // The shape style's text colour (`a:fontRef`, often `lt1`: white on the accent fill).
518        if let Some(id) = story
519            && let Some(c) = wsp.child("wps:style").and_then(|s| s.child("a:fontRef"))
520            && let Some(rgb) = crate::docx::read::drawing_color::color(c, &self.doc.settings.theme_colors)
521            && let Some(part) = self.doc.parts.get_mut(&id)
522        {
523            part.body.auto_color = Some(rgb);
524        }
525        let mut extra = ShapeExtra {
526            dash: crate::docx::read::shape_style::line_dash(ln),
527            cap: crate::doc::connector::LineCap::from_ooxml(ln.and_then(|e| e.attr("cap")).unwrap_or("flat")),
528            outline_fill: paint.outline_fill,
529            ..ShapeExtra::default()
530        };
531        // A rounded rectangle's own corner (`adj`, ECMA-376 §20.1.10.56); other presets drawn as
532        // rounded rectangles keep the default.
533        if prst == "roundRect" {
534            extra.corner = sppr
535                .and_then(|s| s.child("a:prstGeom"))
536                .and_then(|g| preset_adj(g, "adj"))
537                .and_then(|v| u32::try_from(v.clamp(0, i64::from(crate::doc::connector::MAX_CORNER))).ok());
538        }
539        // Arrowheads: any end type but `none` (ECMA-376 §20.1.10.33) is drawn as a triangle of its
540        // width and length.
541        let arrow = |tag: &str| {
542            let e = ln.and_then(|l| l.child(tag))?;
543            let kind = e.attr("type").filter(|t| *t != "none")?;
544            let size = |name: &str| crate::doc::connector::ArrowSize::from_ooxml(e.attr(name).unwrap_or("med"));
545            Some(crate::doc::connector::ArrowStyle {
546                kind: crate::doc::connector::ArrowKind::from_ooxml(kind),
547                width: size("w"),
548                length: size("len"),
549            })
550        };
551        if let Some(style) = arrow("a:headEnd") {
552            extra.arrow_start = true;
553            extra.start_style = style;
554        }
555        if let Some(style) = arrow("a:tailEnd") {
556            extra.arrow_end = true;
557            extra.end_style = style;
558        }
559        // A connector's glued ends (ECMA-376 §20.1.2.2.36, §20.1.2.2.13).
560        if let Some(cn) = wsp.child("wps:cNvCnPr") {
561            let end = |tag: &str| {
562                let e = cn.child(tag)?;
563                let id = e.attr("id").and_then(u32_of).filter(|v| *v != 0)?;
564                let site = e.attr("idx").and_then(u32_of).unwrap_or(0).min(crate::doc::connector::MAX_SITE);
565                Some(ConnEnd { id, site })
566            };
567            extra.start = end("a:stCxn");
568            extra.end = end("a:endCxn");
569            self.glued.extend(extra.start.iter().chain(extra.end.iter()).map(|e| e.id));
570        }
571        extra.warp = wsp.child("wps:bodyPr").and_then(|b| b.child("a:prstTxWarp")).and_then(text_warp);
572        // A freeform: its custom geometry (#743: in a group too).
573        let freeform = sppr.and_then(|s| crate::docx::read::freeform::cust_geom(s, w, h)).map(|mut f| {
574            f.alpha = crate::docx::read::freeform::line_alpha(ln);
575            kind = ShapeKind::Freeform;
576            Arc::new(f)
577        });
578        InlineObject::Shape { kind, w, h, fill, stroke, stroke_width, float, story, freeform, effects, extra }
579    }

581    /// Keep drawing id `id` on a shape read from a file (see [`Reader::glued`]).
582    fn set_shape_id(&mut self, obj: &mut InlineObject, id: Option<&str>) {
583        if let InlineObject::Shape { extra, .. } = obj
584            && let Some(id) = id.and_then(u32_of).filter(|v| *v != 0)
585        {
586            extra.id = id;
587            self.shape_ids = true;
588        }
589    }

591    pub(super) fn read_textbox(&mut self, sc: &StoryCtx, content: &El, rels: &Rels) -> u32 {
592        let mut inner = StoryCtx { story_depth: sc.story_depth + 1, ..Default::default() };
593        let mut blocks = Blocks::new();
594        // Shapes in the text aren't members of a group the box may be in.
595        let group = self.group_fill.take();
596        self.read_blocks(&mut inner, content, rels, &mut blocks, self.chunks.depth);
597        self.group_fill = group;
598        self.flush_pending(&mut inner, &mut blocks);
599        self.doc.add_part(PartKind::TextBox, blocks)
600    }

625fn anchor_float(c: &El) -> Float {
626    // `relativeHeight`: the stacking order Word draws floating objects in (higher on top), which
627    // need not be their order in the file.
628    let z = c.attr("relativeHeight").and_then(|v| v.trim().parse::<u32>().ok()).unwrap_or(0);
629    let mut f = Float { wrap: Wrap::Square, z, ..Float::default() };
630    let behind = on_off_attr(c, "behindDoc");
631    // Word wraps a position relative to the page's size (`wp14:pctPosVOffset`) in
632    // `mc:AlternateContent`; its Fallback gives the same position as an offset (#743).
633    let kids = c.els().flat_map(|k| match k.name.as_str() {
634        "mc:AlternateContent" => k.child("mc:Fallback").map(|f| f.els().collect()).unwrap_or_default(),
635        _ => vec![k],
636    });
637    for k in kids {
638        match k.name.as_str() {
639            "wp:wrapSquare" => {
640                f.wrap = Wrap::Square;
641                f.wrap_text = k.attr("wrapText").and_then(WrapText::from_ooxml).unwrap_or_default();
642            }
643            "wp:wrapTight" | "wp:wrapThrough" => {
644                f.wrap = if k.name == "wp:wrapTight" { Wrap::Tight } else { Wrap::Through };
645                f.wrap_text = k.attr("wrapText").and_then(WrapText::from_ooxml).unwrap_or_default();
646                f.wrap_polygon = k.child("wp:wrapPolygon").and_then(wrap_polygon).map(Arc::new);
647            }
648            "wp:wrapTopAndBottom" => f.wrap = Wrap::TopAndBottom,
649            "wp:wrapNone" => f.wrap = if behind { Wrap::BehindText } else { Wrap::InFrontOfText },
650            "wp:positionH" | "wp:positionV" => {
651                let horiz = k.name == "wp:positionH";
652                let rel = match (k.attr("relativeFrom").unwrap_or(""), horiz) {
653                    ("page", _) => Anchor::Page,
654                    ("margin", _) => Anchor::Margin,
655                    ("leftMargin", true) => Anchor::LeftMargin,
656                    ("rightMargin", true) => Anchor::RightMargin,
657                    ("insideMargin", _) => Anchor::InsideMargin,
658                    ("outsideMargin", _) => Anchor::OutsideMargin,
659                    ("character", true) => Anchor::Character,
660                    ("topMargin", false) => Anchor::TopMargin,
661                    ("bottomMargin", false) => Anchor::BottomMargin,
662                    ("line", false) => Anchor::Line,
663                    ("paragraph", false) => Anchor::Paragraph,
664                    (_, true) => Anchor::Column,
665                    (_, false) => Anchor::Paragraph,
666                };
667                let align = k.child("wp:align").and_then(|a| FloatAlign::from_ooxml(a.text().trim()));
668                let off = k.child("wp:posOffset").and_then(|o| measure(&o.text(), 12_700.0)).unwrap_or(0.0);
669                if horiz {
670                    f.h_rel = rel;
671                    f.h_align = align;
672                    f.x = off;
673                } else {
674                    f.v_rel = rel;
675                    f.v_align = align;
676                    f.y = off;
677                }
678            }
679            _ => {}
680        }
681    }
682    let dist = |n: &str| c.attr(n).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.0).clamp(0.0, 1584.0);
683    f.dist = dist("distL").max(dist("distR"));
684    f.dist_top = dist("distT");
685    f.dist_bottom = dist("distB");
686    // Required by the schema; a file leaving it out lays out in the cell, as before.
687    f.outside_cell = !on_off_attr(c, "layoutInCell") && c.attr("layoutInCell").is_some();
688    f
689}

691/// A `wp:wrapPolygon`: `wp:start`, then the `wp:lineTo` points (ECMA-376 Part 1, §20.4.2.16).
692fn wrap_polygon(e: &El) -> Option<crate::doc::wrap::WrapPolygon> {
693    let pts = e
694        .els()
695        .filter(|k| k.name == "wp:start" || k.name == "wp:lineTo")
696        .take(crate::doc::wrap::MAX_WRAP_POINTS + 1)
697        .filter_map(|k| Some((int(k.attr("x")?)?, int(k.attr("y")?)?)));
698    let mut pts: Vec<(i64, i64)> = pts.collect();
699    // The last point usually repeats the start to close the polygon.
700    if pts.len() > 3 && pts.first() == pts.last() {
701        pts.pop();
702    }
703    let edited = on_off_attr(e, "edited");
704    // An unedited polygon round the whole frame says nothing the object doesn't (it's what we
705    // write when there's no outline): keep deriving the outline.
706    let s = i64::from(crate::doc::wrap::WRAP_SPACE);
707    let mut corners = pts.clone();
708    corners.sort_unstable();
709    if !edited && corners == [(0, 0), (0, s), (s, 0), (s, s)] {
710        return None;
711    }
712    crate::doc::wrap::WrapPolygon::new(edited, pts)
713}

720/// A shape's `a:effectLst` (ECMA-376 Part 1 §20.1.8.26): its outer shadow, glow and soft edges.
721/// Other effects are dropped.
722/// A preset text warp (`a:prstTxWarp`, ECMA-376 §20.1.9.19) and its adjust values
723/// (`a:avLst/a:gd` with `fmla="val N"`). Unknown presets are dropped.
724fn text_warp(e: &El) -> Option<TextWarp> {
725    let mut w = TextWarp::new(e.attr("prst")?)?;
726    for gd in e.child("a:avLst").into_iter().flat_map(|l| l.children("a:gd")) {
727        let (Some(name), Some(f)) = (gd.attr("name"), gd.attr("fmla")) else { continue };
728        if let Some(v) = f.trim().strip_prefix("val").and_then(|v| v.trim().parse::<i64>().ok()) {
729            w.set_adj(name, v);
730        }
731    }
732    Some(w)
733}

741fn effect_list(l: &El) -> ShapeEffects {
742    let pt = |e: &El, n: &str| e.attr(n).and_then(|v| measure(v, 12_700.0)).unwrap_or(0.0);
743    let shadow = l.child("a:outerShdw").map(|e| {
744        let (color, transparency) = effect_color(e);
745        // `dir`: 60000ths of a degree, clockwise.
746        let angle = e.attr("dir").and_then(int).map(|v| (v.rem_euclid(21_600_000) as f32) / 60_000.0).unwrap_or(0.0);
747        // `rotWithShape` is true unless it says otherwise (ECMA-376 §20.1.8.49).
748        let rot_with_shape = e.attr("rotWithShape").is_none_or(|v| !matches!(v, "0" | "false" | "off"));
749        Shadow { color: color.unwrap_or(Rgb::BLACK), transparency, blur: pt(e, "blurRad"), distance: pt(e, "dist"), angle, rot_with_shape }
750    });
751    let glow = l.child("a:glow").map(|e| {
752        let (color, transparency) = effect_color(e);
753        Glow { color: color.unwrap_or(Glow::default().color), size: pt(e, "rad"), transparency }
754    });
755    let soft_edge = l.child("a:softEdge").map(|e| pt(e, "rad"));
756    ShapeEffects { shadow, glow, soft_edge }.sanitized()
757}

759/// The colour inside an effect (`a:srgbClr`, `a:prstClr`, `a:sysClr`; theme colours aren't
760/// resolved here) and its transparency, percent (from `a:alpha`, opaque when absent).
761fn effect_color(e: &El) -> (Option<Rgb>, f32) {
762    let Some(c) = e.els().find(|c| matches!(c.name.as_str(), "a:srgbClr" | "a:prstClr" | "a:sysClr" | "a:schemeClr" | "a:scrgbClr" | "a:hslClr"))
763    else {
764        return (None, 0.0);
765    };
766    let rgb = match c.name.as_str() {
767        "a:srgbClr" => c.attr("val").and_then(Rgb::parse),
768        "a:sysClr" => c.attr("lastClr").and_then(Rgb::parse),
769        "a:prstClr" => match c.attr("val").unwrap_or("") {
770            "black" => Some(Rgb::BLACK),
771            "white" => Some(Rgb::WHITE),
772            "gray" | "grey" => Some(Rgb(0x80, 0x80, 0x80)),
773            _ => None,
774        },
775        _ => None,
776    };
777    // `a:alpha`: opacity in 1000ths of a percent.
778    let opacity = c.child("a:alpha").and_then(|a| a.attr("val")).and_then(int).map(|v| v.clamp(0, 100_000) as f32 / 1000.0).unwrap_or(100.0);
779    (rgb, 100.0 - opacity)
780}

791/// The rotation and flips of the `a:xfrm` in shape properties `sppr` (ECMA-376 §20.1.7.6:
792/// `rot` in 60000ths of a degree, clockwise; `flipH`, `flipV`). None when missing or junk.
793fn xfrm_spin(sppr: Option<&El>) -> crate::geom::Spin {
794    let Some(x) = sppr.and_then(|s| if matches!(s.name.as_str(), "a:xfrm" | "w14:xfrm") { Some(s) } else { s.child("a:xfrm") }) else {
795        return crate::geom::Spin::default();
796    };
797    let rot = x.attr("rot").and_then(int).map(|v| v.rem_euclid(21_600_000) as f32 / 60_000.0).unwrap_or(0.0);
798    crate::geom::Spin::new(rot, on_off_attr(x, "flipH"), on_off_attr(x, "flipV"))
799}

801/// The `a:xfrm` in shape properties `sppr` (zeros when missing).
802fn group_xfrm(sppr: Option<&El>) -> Xfrm {
803    // A content part (native ink) has its own `w14:xfrm`.
804    let x = sppr.and_then(|s| s.child("a:xfrm").or_else(|| s.child("w14:xfrm")));
805    let pair = |name: &str, a: &str, b: &str, lim: f32| {
806        let e = x.and_then(|x| x.child(name))?;
807        let v = |n: &str| e.attr(n).and_then(|v| measure(v, 12_700.0)).map(|v| v.clamp(-lim, lim)).unwrap_or(0.0);
808        Some((v(a), v(b)))
809    };
810    let max = crate::docx::units::MAX_LEN_PT;
811    let pos = |p: Option<(f32, f32)>| p.map(|(a, b)| (a.max(0.0), b.max(0.0)));
812    Xfrm {
813        off: pair("a:off", "x", "y", max).unwrap_or((0.0, 0.0)),
814        ext: pos(pair("a:ext", "cx", "cy", max)).unwrap_or((0.0, 0.0)),
815        ch_off: pair("a:chOff", "x", "y", max),
816        ch_ext: pos(pair("a:chExt", "cx", "cy", max)),
817    }
818}
