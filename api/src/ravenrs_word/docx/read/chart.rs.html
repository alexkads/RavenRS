1//! Charts (ECMA-376 Part 1 §21.2): the cached data of a `c:chartSpace` drawn as items in points.
2//!
3//! Bar and column (clustered, stacked or 100% stacked), line, area, pie, doughnut and scatter
4//! charts are drawn from the first supported chart type of the plot area, with their values as
5//! data labels where `c:dLbls` shows them (`c:showVal`). Radar charts (standard, with markers or
6//! filled) draw as spokes, concentric grid polygons and one polygon per series. A chart of only other types (stock, bubble…)
7//! draws a placeholder: its frame, its title and the type it is. Text widths are estimated ([`CHAR_W`] em a character), so label places are approximate.

76/// Items for a chart part, sized `w` x `h` points: a placeholder when only unsupported chart
77/// types are in it, empty when there's nothing to draw.
78pub(crate) fn chart_items(space: &El, theme: &[Rgb], w: f32, h: f32) -> Vec<GraphicItem> {
79    let mut out = Vec::new();
80    if !(w.is_finite() && h.is_finite() && w > 0.0 && h > 0.0) {
81        return out;
82    }
83    let Some(chart) = Chart::parse(space, theme) else {
84        placeholder(&mut out, space, theme, w, h);
85        return out;
86    };
87    backdrop(&mut out, space, theme, w, h);
88    let mut area = Rect { x: PAD, y: PAD, w: (w - 2.0 * PAD).max(1.0), h: (h - 2.0 * PAD).max(1.0) };
89    if let Some(t) = &chart.title {
90        area = title_block(&mut out, t, area);
91    }
92    if let Some(pos) = chart.legend {
93        // Word lists bar series in the order they are drawn: reversed for clustered horizontal bars
94        // and for stacked columns with a vertical legend.
95        let vertical = matches!(pos, Pos::Left | Pos::Right);
96        let reverse = chart.kind == Kind::Bar && if chart.horizontal { !chart.stacked() } else { chart.stacked() && vertical };
97        let mut entries = chart.legend_entries();
98        if reverse {
99            entries.reverse();
100        }
101        area = legend(&mut out, &entries, pos, area, chart.text_size);
102    }
103    match chart.kind {
104        Kind::Pie | Kind::Doughnut => pie(&mut out, &chart, area),
105        Kind::Radar => radar(&mut out, &chart, area),
106        _ => cartesian(&mut out, &chart, area),
107    }
108    out
109}

111/// The title at the top of `area`; returns the room left under it.
112fn title_block(out: &mut Vec<GraphicItem>, t: &str, area: Rect) -> Rect {
113    let lh = TITLE_SIZE * LINE;
114    text(out, Rect { x: area.x, y: area.y, w: area.w, h: lh }, t, TITLE_SIZE, TextAlign::Center, true);
115    Rect { y: area.y + lh + PAD, h: (area.h - lh - PAD).max(1.0), ..area }
116}

118/// A chart whose plot area holds only chart types WordCraft doesn't draw (stock, bubble…): its
119/// frame with a light border, its title and a line saying which type it is, so it isn't blank
120/// space. Nothing when the plot area has no chart type at all, or a supported one with nothing
121/// to draw.
122fn placeholder(out: &mut Vec<GraphicItem>, space: &El, theme: &[Rgb], w: f32, h: f32) {
123    let Some(chart) = space.child("c:chart") else { return };
124    let Some(plot) = chart.child("c:plotArea") else { return };
125    if plot.els().any(|e| Kind::of(&e.name).is_some()) {
126        return;
127    }
128    let Some(ty) = plot.els().find(|e| e.name.ends_with("Chart")) else { return };
129    backdrop(out, space, theme, w, h);
130    out.push(GraphicItem::Shape { rect: [0.0, 0.0, w, h], kind: ShapeKind::Rectangle, fill: None, stroke: Some(AXIS), stroke_width: 0.75 });
131    let mut area = Rect { x: PAD, y: PAD, w: (w - 2.0 * PAD).max(1.0), h: (h - 2.0 * PAD).max(1.0) };
132    if let Some(t) = title_of(chart, &read_series(ty, false, theme)) {
133        area = title_block(out, &t, area);
134    }
135    let size = text_size(space);
136    let lh = size * LINE;
137    let msg = format!("{} (not supported yet)", unsupported_name(&ty.name));
138    let line = Rect { x: area.x, y: area.y + ((area.h - lh) / 2.0).max(0.0), w: area.w, h: lh };
139    text(out, line, &clip_text(&msg, area.w, size), size, TextAlign::Center, false);
140}

142/// What a placeholder calls a chart type element WordCraft doesn't draw.
143fn unsupported_name(name: &str) -> &'static str {
144    match name {
145        "c:stockChart" => "Stock chart",
146        "c:bubbleChart" => "Bubble chart",
147        "c:surfaceChart" | "c:surface3DChart" => "Surface chart",
148        "c:ofPieChart" => "Pie of pie chart",
149        "c:bar3DChart" => "3-D bar chart",
150        "c:line3DChart" => "3-D line chart",
151        "c:area3DChart" => "3-D area chart",
152        "c:pie3DChart" => "3-D pie chart",
153        _ => "Chart",
154    }
155}

157struct Chart {
158    kind: Kind,
159    horizontal: bool,
160    grouping: Grouping,
161    /// Gap between groups as a fraction of a bar's width.
162    gap: f32,
163    /// Overlap of clustered bars, -1..1 (1 = all on top of each other).
164    overlap: f32,
165    /// Doughnut hole, fraction of the outer radius.
166    hole: f32,
167    first_angle: f64,
168    /// Radar: series polygons are filled (`c:radarStyle` = `filled`).
169    radar_fill: bool,
170    /// Radar: points carry markers (`c:radarStyle` = `marker`).
171    radar_marks: bool,
172    vary: bool,
173    /// Number of categories (scatter: points per series at most).
174    n: usize,
175    cats: Vec<String>,
176    series: Vec<Series>,
177    accents: Vec<Rgb>,
178    title: Option<String>,
179    legend: Option<Pos>,
180    /// Category axis (scatter: the x axis).
181    cat: Axis,
182    val: Axis,
183    text_size: f32,
184    val_fmt: String,
185    /// Serial days count from 1904 (`c:date1904`), not 1900.
186    date1904: bool,
187    /// Outline of the plot area, when the file draws one.
188    plot_border: Option<Rgb>,
189}
