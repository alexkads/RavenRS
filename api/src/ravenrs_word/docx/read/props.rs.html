20/// A text frame (`w:framePr` without a drop cap, ECMA-376 §17.3.1.11), sizes and offsets in
21/// points (clamped), names kept as OOXML writes them.
22fn text_frame(k: &El) -> crate::doc::props::TextFrame {
23    let len = |name: &str| tw(k, name).filter(|v| v.is_finite()).map(|v| v.clamp(-1584.0, 1584.0));
24    let word = |name: &str, allowed: &[&str], default: &str| k.attr(name).filter(|v| allowed.contains(v)).unwrap_or(default).to_string();
25    let align = |name: &str, allowed: &[&str]| k.attr(name).filter(|v| allowed.contains(v)).map(str::to_string);
26    crate::doc::props::TextFrame {
27        w: len("w:w").filter(|v| *v >= 0.0),
28        h: len("w:h").map(f32::abs),
29        h_exact: k.attr("w:hRule") == Some("exact"),
30        x: len("w:x").unwrap_or(0.0),
31        y: len("w:y").unwrap_or(0.0),
32        x_align: align("w:xAlign", &["left", "center", "right", "inside", "outside"]),
33        y_align: align("w:yAlign", &["top", "center", "bottom", "inside", "outside"]),
34        h_anchor: word("w:hAnchor", &["text", "margin", "page"], "page"),
35        v_anchor: word("w:vAnchor", &["text", "margin", "page"], "page"),
36        wrap: word("w:wrap", &["around", "auto", "notBeside", "none", "tight", "through"], "auto"),
37        anchor_lock: k.attr("w:anchorLock").map(|v| !matches!(v.trim(), "0" | "false" | "off" | "False" | "FALSE")),
38        h_space: len("w:hSpace").unwrap_or(0.0).max(0.0),
39        v_space: len("w:vSpace").unwrap_or(0.0).max(0.0),
40    }
41}

79/// The script a language tag is written in, as a theme's `a:font script` names it (ISO 15924, and
80/// Office's own `Uigh` for Uyghur and `Viet` for Vietnamese): an explicit script subtag
81/// (`ku-Arab`), else the language's usual script. Languages not listed are taken as Latin.
82pub(crate) fn lang_script(tag: &str) -> &'static str {
83    const SCRIPTS: &[&str] = &[
84        "Arab", "Armn", "Beng", "Cans", "Cher", "Cyrl", "Deva", "Ethi", "Geor", "Grek", "Gujr", "Guru", "Hang", "Hans", "Hant", "Hebr", "Jpan",
85        "Khmr", "Knda", "Laoo", "Latn", "Mlym", "Mong", "Mymr", "Orya", "Sinh", "Syrc", "Taml", "Telu", "Tfng", "Thaa", "Thai", "Tibt", "Uigh",
86        "Viet", "Yiii",
87    ];
88    let mut parts = tag.split(['-', '_']).filter(|p| !p.is_empty());
89    let lang = parts.next().unwrap_or_default().to_ascii_lowercase();
90    let rest: Vec<&str> = parts.take(8).collect();
91    if let Some(s) = rest.iter().find_map(|p| SCRIPTS.iter().find(|s| s.eq_ignore_ascii_case(p))) {
92        return s;
93    }
94    match lang.as_str() {
95        "ar" | "fa" | "ur" | "ps" | "sd" | "ckb" | "prs" => "Arab",
96        "ug" => "Uigh",
97        "vi" => "Viet",
98        "he" | "yi" => "Hebr",
99        "syr" => "Syrc",
100        "dv" => "Thaa",
101        "th" => "Thai",
102        "hi" | "mr" | "ne" | "sa" | "kok" => "Deva",
103        "bn" | "as" => "Beng",
104        // Measured in Word: Punjabi takes the Arabic-script font in Pakistan, Gurmukhi elsewhere.
105        "pa" if rest.iter().any(|p| p.eq_ignore_ascii_case("pk")) => "Arab",
106        "pa" => "Guru",
107        "gu" => "Gujr",
108        "or" => "Orya",
109        "ta" => "Taml",
110        "te" => "Telu",
111        "kn" => "Knda",
112        "ml" => "Mlym",
113        "si" => "Sinh",
114        "lo" => "Laoo",
115        "km" => "Khmr",
116        "bo" => "Tibt",
117        "my" => "Mymr",
118        "am" | "ti" => "Ethi",
119        "chr" => "Cher",
120        "iu" => "Cans",
121        "ii" => "Yiii",
122        "ka" => "Geor",
123        "hy" => "Armn",
124        "el" => "Grek",
125        "ja" => "Jpan",
126        "ko" => "Hang",
127        "zh" if rest.iter().any(|p| ["tw", "hk", "mo"].iter().any(|r| r.eq_ignore_ascii_case(p))) => "Hant",
128        "zh" => "Hans",
129        _ => "Latn",
130    }
131}

133/// The list a level-only `numPr` refers to: the nearest numbering in style `style`'s based-on chain
134/// (its own included) that names a list. `None` when no style in the chain has one.
135pub(crate) fn chain_list(sheet: &crate::doc::styles::StyleSheet, style: &str) -> Option<u32> {
136    sheet.chain(style).iter().rev().filter_map(|s| s.para.numbering).map(|n| n.num).find(|n| *n != LEVEL_ONLY)
137}

139/// Resolve level-only numbering in `p` against its style chain (`style` when it has none).
140pub(crate) fn resolve_level_only(sheet: &crate::doc::styles::StyleSheet, p: &mut ParaProps, style: &str) {
141    if let Some(n) = p.numbering.as_mut().filter(|n| n.num == LEVEL_ONLY) {
142        match chain_list(sheet, p.style.as_deref().unwrap_or(style)) {
143            Some(num) => n.num = num,
144            None => p.numbering = None,
145        }
146    }
147}

149/// A `w:rFonts` value as the document keeps it: `None` when it names no family, else its leading
150/// families up to [`MAX_FONT_NAME`] bytes, cut at a `;` so no name is split. A LibreOffice file
151/// names its Latin font as a semicolon-separated fallback list, and a real list names a few
152/// families; a longer value is corrupt or hostile, and every piece of text drawn in it would pay
153/// for its whole length as the list is classified and resolved. Keeping the leading families keeps
154/// the family such a list picks, and the rest of the value is dropped like `w:cs`/`w:eastAsia`
155/// drop an over-long name.
156fn bounded_font_list(v: &str) -> Option<String> {
157    if v.len() <= MAX_FONT_NAME {
158        return (!v.is_empty()).then(|| v.to_string());
159    }
160    let mut end = 0usize;
161    for part in v.split_inclusive(';') {
162        let next = end + part.len();
163        if next > MAX_FONT_NAME {
164            break;
165        }
166        end = next;
167    }
168    let kept = v.get(..end)?.trim_end_matches(';');
169    (!kept.is_empty()).then(|| kept.to_string())
170}

177    fn theme_font(&self, t: &str) -> Option<String> {
178        let major = t.starts_with("major");
179        let latin = if major { &self.major_font } else { &self.minor_font };
180        if t.ends_with("Bidi") {
181            return Some(self.bidi_theme_font(major));
182        }
183        (!latin.is_empty()).then(|| latin.clone())
184    }

186    /// The font a `majorBidi`/`minorBidi` theme slot names. Measured in Word for Mac: the theme's
187    /// font for the script of the settings' `w:themeFontLang/@w:bidi` (its `a:latin` for a Latin
188    /// language), else the theme's `a:cs`, else Times New Roman; with no bidi language there,
189    /// Times New Roman whatever the theme says. The run's own `w:lang` plays no part.
190    fn bidi_theme_font(&self, major: bool) -> String {
191        let (latin, cs, scripts) =
192            if major { (&self.major_font, &self.major_cs, &self.major_scripts) } else { (&self.minor_font, &self.minor_cs, &self.minor_scripts) };
193        let named = |f: &String| (!f.is_empty()).then(|| f.clone());
194        let Some(script) = self.bidi_script else { return BIDI_THEME_FALLBACK_FONT.to_string() };
195        let by_script = if script == "Latn" { named(latin) } else { scripts.iter().find(|(s, _)| s == script).and_then(|(_, f)| named(f)) };
196        by_script.or_else(|| named(cs)).unwrap_or_else(|| BIDI_THEME_FALLBACK_FONT.to_string())
197    }

199    /// The theme colour `e` names in `attr` (`w:themeColor`, `w:themeFill`), through the colour
200    /// scheme mapping, with its `tint` or else its `shade` (hex bytes) applied to the HSL
201    /// luminance as Word does: given both, Word applies the tint alone (MS-OI29500 17.3.2.6 b,
202    /// 17.4.32 a). When given it supersedes the fixed colour; `None` leaves the fixed one.
203    fn theme_color(&self, e: &El, attr: &str, tint: &str, shade: &str) -> Option<Rgb> {
204        let logical = |i: usize| self.color_map.and_then(|m| m.get(i).copied()).unwrap_or(i);
205        let slot = match e.attr(attr)? {
206            "dark1" => 0,
207            "light1" => 1,
208            "dark2" => 2,
209            "light2" => 3,
210            "text1" => logical(0),
211            "background1" => logical(1),
212            "text2" => logical(2),
213            "background2" => logical(3),
214            "accent1" => logical(4),
215            "accent2" => logical(5),
216            "accent3" => logical(6),
217            "accent4" => logical(7),
218            "accent5" => logical(8),
219            "accent6" => logical(9),
220            "hyperlink" => logical(10),
221            "followedHyperlink" => logical(11),
222            _ => return None,
223        };
224        let mut c = *self.theme_colors.get(slot)?;
225        let byte = |a: &str| e.attr(a).and_then(|v| u8::from_str_radix(v.trim(), 16).ok()).map(|v| f64::from(v) / 255.0);
226        if let Some(t) = byte(tint) {
227            c = adjust_lum(c, |l| l * t + (1.0 - t));
228        } else if let Some(s) = byte(shade) {
229            c = adjust_lum(c, |l| l * s);
230        }
231        Some(c)
232    }

234    /// `w:shd`: its fill colour (a theme fill first), or a solid pattern's colour.
235    pub fn shd_fill(&self, e: &El) -> Option<Rgb> {
236        // `nil` counts as no shading, whatever fill it carries (MS-OI29500 17.18.78).
237        if e.attr("w:val") == Some("nil") {
238            return None;
239        }
240        if let Some(c) = self.theme_color(e, "w:themeFill", "w:themeFillTint", "w:themeFillShade") {
241            return Some(c);
242        }
243        // A solid pattern without a fill is painted in its pattern colour, a theme one first.
244        if e.attr("w:val") == Some("solid")
245            && e.attr("w:fill").and_then(Rgb::parse).is_none()
246            && let Some(c) = self.theme_color(e, "w:themeColor", "w:themeTint", "w:themeShade")
247        {
248            return Some(c);
249        }
250        shd_fill(e)
251    }

253    /// The character properties `e` (a `w:rPr`) sets: direct formatting, plus the fonts, size,
254    /// weight and posture of the complex-script and East Asian text beside them.
255    pub fn rpr(&self, e: &El) -> CharProps {
256        let mut c = CharProps::default();
257        for k in e.els() {
258            match k.name.as_str() {
259                "w:rStyle" => c.style = k.attr("w:val").filter(|v| !v.is_empty()).map(|v| self.style_id(v)),
260                "w:rFonts" => {
261                    // A theme font supersedes the fixed font beside it (`w:asciiTheme` over
262                    // `w:ascii`, `w:hAnsiTheme` over `w:hAnsi`).
263                    // A LibreOffice file names the Latin font as a semicolon-separated fallback
264                    // list. Names are bounded like `w:cs`/`w:eastAsia` below: a longer one is
265                    // corrupt or hostile, and every piece of text drawn in it would pay for its
266                    // whole length as the list is classified (see `bounded_font_list`).
267                    let named = |a: &str| k.attr(a).and_then(bounded_font_list);
268                    let themed = |a: &str| k.attr(a).and_then(|t| self.theme_font(t));
269                    // `w:eastAsia` and `w:cs` name fonts for other scripts and stay in `font_ea` /
270                    // `font_cs`: `w:hint` only picks between them for ambiguous characters, never for
271                    // Latin letters. A Normal style with only `w:eastAsia`/`w:cs` keeps the
272                    // `docDefaults` font, and so does a hinted run, as in Word.
273                    c.font = themed("w:asciiTheme").or_else(|| named("w:ascii")).or_else(|| themed("w:hAnsiTheme")).or_else(|| named("w:hAnsi"));
274                    // The complex-script font (Persian, Arabic, Hebrew text). A `w:cstheme` font is
275                    // left to inherit: the theme's per-script fonts aren't read.
276                    c.font_cs = k.attr("w:cs").filter(|f| !f.is_empty() && f.len() < 256).map(str::to_string);
277                    // The East Asian font (Chinese, Japanese, Korean text). A `w:eastAsiaTheme` font
278                    // is left to inherit, like `w:cstheme`.
279                    c.font_ea = k.attr("w:eastAsia").filter(|f| !f.is_empty() && f.len() < 256).map(str::to_string);
280                }
281                "w:b" => c.bold = Some(on_off(k)),
282                "w:bCs" => c.bold_cs = Some(on_off(k)),
283                "w:i" => c.italic = Some(on_off(k)),
284                "w:iCs" => c.italic_cs = Some(on_off(k)),
285                "w:cs" => c.cs = Some(on_off(k)),
286                "w:caps" => c.caps = Some(on_off(k)),
287                "w:smallCaps" => c.small_caps = Some(on_off(k)),
288                "w:em" => c.emphasis = Some(k.attr("w:val").map(crate::doc::props::EmphasisMark::from_ooxml).unwrap_or_default()),
289                "w:strike" => c.strike = Some(on_off(k)),
290                "w:dstrike" => c.double_strike = Some(on_off(k)),
291                "w:outline" => c.outline = Some(on_off(k)),
292                "w:shadow" => c.shadow = Some(on_off(k)),
293                "w:emboss" => c.emboss = Some(on_off(k)),
294                "w:imprint" => c.engrave = Some(on_off(k)),
295                "w:noProof" => c.no_proof = Some(on_off(k)),
296                "w:vanish" => c.hidden = Some(on_off(k)),
297                "w:webHidden" => c.web_hidden = Some(on_off(k)),
298                "w:snapToGrid" => c.snap_to_grid = Some(on_off(k)),
299                "w:rtl" => c.rtl = Some(on_off(k)),
300                "w:color" => {
301                    c.color = match (self.theme_color(k, "w:themeColor", "w:themeTint", "w:themeShade"), k.attr("w:val")) {
302                        (Some(t), _) => Some(TextColor::Rgb(t)),
303                        (None, Some(v)) if v.eq_ignore_ascii_case("auto") => Some(TextColor::Auto),
304                        (None, Some(v)) => Rgb::parse(v).map(TextColor::Rgb),
305                        (None, None) => None,
306                    }
307                }
308                "w:spacing" => c.spacing = tw(k, "w:val").map(|v| v.clamp(-1584.0, 1584.0)),
309                "w:w" => c.scale = k.attr("w:val").and_then(|v| measure(v.trim_end_matches('%'), 1.0)).map(|v| v.clamp(1.0, 600.0)),
310                "w:kern" => c.kern = k.attr("w:val").and_then(|v| measure(v, 2.0)).map(|v| v.clamp(0.0, 1638.0)),
311                "w:position" => c.position = k.attr("w:val").and_then(|v| measure(v, 2.0)).map(|v| v.clamp(-1584.0, 1584.0)),
312                "w:sz" => c.size = k.attr("w:val").and_then(|v| measure(v, 2.0)).filter(|v| *v > 0.0).map(|v| v.clamp(1.0, 1638.0)),
313                "w:szCs" => c.size_cs = k.attr("w:val").and_then(|v| measure(v, 2.0)).filter(|v| *v > 0.0).map(|v| v.clamp(1.0, 1638.0)),
314                "w:highlight" => c.highlight = k.attr("w:val").map(Highlight::from_ooxml),
315                "w:u" => {
316                    // w:val has no default in the schema: without it Word draws no underline
317                    c.underline = k.attr("w:val").map(Underline::from_ooxml);
318                    c.underline_color =
319                        self.theme_color(k, "w:themeColor", "w:themeTint", "w:themeShade").or_else(|| k.attr("w:color").and_then(Rgb::parse));
320                }
321                "w:bdr" => c.border = Some(self.border(k)),
322                "w:shd" => c.shading = self.shd_fill(k),
323                "w:vertAlign" => {
324                    c.vert_align = k.attr("w:val").map(|v| match v {
325                        "superscript" => VertAlign::Superscript,
326                        "subscript" => VertAlign::Subscript,
327                        _ => VertAlign::Baseline,
328                    })
329                }
330                "w:eastAsianLayout" => {
331                    let l = east_asian_layout(k);
332                    c.east_asian_layout = (l != EastAsianLayout::default()).then_some(l);
333                }
334                "w:lang" => {
335                    c.lang = k.attr("w:val").map(str::trim).filter(|v| !v.is_empty() && v.len() < 64).map(str::to_string);
336                    c.lang_bidi = k.attr("w:bidi").map(str::trim).filter(|v| !v.is_empty() && v.len() < 64).map(str::to_string);
337                    c.lang_ea = k.attr("w:eastAsia").map(str::trim).filter(|v| !v.is_empty() && v.len() < 64).map(str::to_string);
338                }
339                "w14:ligatures" => c.ligatures = k.attr("w14:val").and_then(crate::doc::props::Ligatures::from_ooxml),
340                n if n.starts_with("w14:") => {
341                    let fx = c.text_effects.get_or_insert_with(Default::default);
342                    super::wordart::read(k, &self.theme, fx);
343                    if fx.is_empty() {
344                        c.text_effects = None;
345                    }
346                }
347                _ => {}
348            }
349        }
350        c
351    }

353    /// `w:pPr` (without `w:sectPr`; `w:rPr` is the paragraph mark, returned separately).
354    pub fn ppr(&self, e: &El) -> (ParaProps, CharProps) {
355        // (`w:numPr` with a level and no list sets `LEVEL_ONLY`, resolved by the caller.)
356        let mut p = ParaProps::default();
357        let mut mark = CharProps::default();
358        for k in e.els() {
359            match k.name.as_str() {
360                "w:pStyle" => p.style = k.attr("w:val").filter(|v| !v.is_empty()).map(|v| self.style_id(v)),
361                "w:keepNext" => p.keep_next = Some(on_off(k)),
362                "w:keepLines" => p.keep_lines = Some(on_off(k)),
363                "w:pageBreakBefore" => p.page_break_before = Some(on_off(k)),
364                "w:widowControl" => p.widow_control = Some(on_off(k)),
365                "w:contextualSpacing" => p.contextual_spacing = Some(on_off(k)),
366                "w:mirrorIndents" => p.mirror_indents = Some(on_off(k)),
367                "w:adjustRightInd" => p.adjust_right_indent = Some(on_off(k)),
368                "w:suppressAutoHyphens" => p.suppress_hyphens = Some(on_off(k)),
369                "w:suppressLineNumbers" => p.suppress_line_numbers = Some(on_off(k)),
370                "w:bidi" => p.bidi = Some(on_off(k)),
371                "w:textDirection" => p.text_direction = k.attr("w:val").and_then(ParagraphTextDirection::from_ooxml),
372                "w:kinsoku" => p.kinsoku = Some(on_off(k)),
373                "w:wordWrap" => p.word_wrap = Some(on_off(k)),
374                "w:overflowPunct" => p.overflow_punct = Some(on_off(k)),
375                "w:topLinePunct" => p.top_line_punct = Some(on_off(k)),
376                "w:autoSpaceDE" => p.auto_space_de = Some(on_off(k)),
377                "w:autoSpaceDN" => p.auto_space_dn = Some(on_off(k)),
378                "w:snapToGrid" => p.snap_to_grid = Some(on_off(k)),
379                "w:textAlignment" => p.text_alignment = k.attr("w:val").and_then(TextAlignment::from_ooxml),
380                "w:framePr" => {
381                    if let Some(kind @ ("drop" | "margin")) = k.attr("w:dropCap") {
382                        p.drop_cap = Some(k.attr("w:lines").and_then(u32_of).unwrap_or(3).clamp(1, 10) as u8);
383                        p.drop_cap_margin = (kind == "margin").then_some(true);
384                        p.drop_cap_distance =
385                            tw(k, "w:hSpace").filter(|d| d.is_finite() && *d > 0.0).map(|d| d.min(crate::doc::resolve::MAX_DROP_CAP_DISTANCE));
386                    } else {
387                        p.frame = Some(text_frame(k));
388                    }
389                }
390                "w:numPr" => {
391                    let num = k.child_val("w:numId").and_then(u32_of);
392                    let ilvl = k.child_val("w:ilvl").and_then(u32_of);
393                    let level = ilvl.unwrap_or(0).min(8) as u8;
394                    match (num, ilvl) {
395                        (Some(num), _) => p.numbering = Some(NumRef { num, level }),
396                        // A level without a list: the list comes from the based-on style, as in
397                        // a Heading 3 that only says "level 3" of Heading 2's list. Resolved once
398                        // the styles are read (`resolve_list_levels`).
399                        (None, Some(_)) => p.numbering = Some(NumRef { num: LEVEL_ONLY, level }),
400                        (None, None) => {}
401                    }
402                }
403                "w:pBdr" => p.borders = Some(self.borders(k)),
404                "w:shd" => p.shading = self.shd_fill(k),
405                "w:tabs" => {
406                    let tabs: Vec<TabStop> = k.children("w:tab").take(64).filter_map(tab_stop).collect();
407                    p.tabs = Some(tabs);
408                }
409                "w:spacing" => {
410                    if let Some(v) = tw_u16(k, "w:before") {
411                        p.space_before = Some(v.max(0.0));
412                    }
413                    if let Some(v) = tw_u16(k, "w:after") {
414                        p.space_after = Some(v.max(0.0));
415                    }
416                    // HTML-style automatic spacing; "0" switches an inherited one off.
417                    let auto = |name| k.attr(name).map(|v| matches!(v.trim(), "1" | "true" | "on"));
418                    if let Some(a) = auto("w:beforeAutospacing") {
419                        p.space_before_auto = Some(a);
420                    }
421                    if let Some(a) = auto("w:afterAutospacing") {
422                        p.space_after_auto = Some(a);
423                    }
424                    if let Some(line) = k.attr("w:line") {
425                        // Word reads the rule case-insensitively (`atleast` is seen in the wild).
426                        let rule = k.attr("w:lineRule").unwrap_or("auto").to_ascii_lowercase();
427                        p.line_spacing = match rule.as_str() {
428                            "exact" => measure(line, 20.0).map(|v| LineSpacing::Exactly(v.abs())),
429                            "atleast" => measure(line, 20.0).map(|v| LineSpacing::AtLeast(v.abs())),
430                            // Strict OOXML writes the height with a unit (`13.90pt`, #2329).
431                            _ => line_multiple(line).map(|v| LineSpacing::Multiple(v.abs().clamp(0.06, 132.0))),
432                        };
433                    }
434                }
435                "w:ind" => {
436                    if let Some(v) = tw_i16(k, "w:left").or_else(|| tw_i16(k, "w:start")) {
437                        p.indent_left = Some(v);
438                    }
439                    if let Some(v) = tw_i16(k, "w:right").or_else(|| tw_i16(k, "w:end")) {
440                        p.indent_right = Some(v);
441                    }
442                    if let Some(v) = tw_i16(k, "w:hanging") {
443                        p.indent_first = Some(-v);
444                    } else if let Some(v) = tw_i16(k, "w:firstLine") {
445                        p.indent_first = Some(v);
446                    }
447                    // Character units (hundredths of a character), which win when non-zero.
448                    let chars = |a: &str| k.attr(a).and_then(int).map(|v| v.clamp(-158_400, 158_400) as f32 / 100.0);
449                    if let Some(v) = chars("w:leftChars").or_else(|| chars("w:startChars")) {
450                        p.indent_left_chars = Some(v);
451                    }
452                    if let Some(v) = chars("w:rightChars").or_else(|| chars("w:endChars")) {
453                        p.indent_right_chars = Some(v);
454                    }
455                    match (chars("w:hangingChars"), chars("w:firstLineChars")) {
456                        (Some(h), f) if h != 0.0 || f.is_none() => p.indent_first_chars = Some(-h),
457                        (_, Some(f)) => p.indent_first_chars = Some(f),
458                        _ => {}
459                    }
460                }
461                "w:jc" => p.align = k.attr("w:val").map(align),
462                "w:outlineLvl" => p.outline_level = k.attr("w:val").and_then(u32_of).map(|v| v.min(9) as u8),
463                "w:rPr" => mark = self.rpr(k),
464                _ => {}
465            }
466        }
467        (p, mark)
468    }
469}

505/// `w:shd/@w:fill` (a solid fill colour; `auto` = none), ignoring theme fills ([`PropCtx::shd_fill`]).
506fn shd_fill(e: &El) -> Option<Rgb> {
507    let fill = e.attr("w:fill").and_then(Rgb::parse);
508    if fill.is_some() {
509        return fill;
510    }
511    // A solid pattern uses the pattern colour.
512    if e.attr("w:val") == Some("solid") {
513        return e.attr("w:color").and_then(Rgb::parse);
514    }
515    None
516}

519    /// One border (`CT_Border`); its theme colour supersedes `w:color`, as for text.
520    pub fn border(&self, e: &El) -> Border {
521        let style = BorderStyle::from_ooxml(e.attr("w:val").unwrap_or("single"));
522        let width = e.attr("w:sz").and_then(|v| measure(v, 8.0)).unwrap_or(0.5).clamp(0.0, 12.0);
523        let color = self.theme_color(e, "w:themeColor", "w:themeTint", "w:themeShade").or_else(|| e.attr("w:color").and_then(Rgb::parse));
524        let space = e.attr("w:space").and_then(|v| measure(v, 1.0)).unwrap_or(0.0).clamp(0.0, 31.0);
525        Border { style, width, color, space }
526    }

537                "between" | "insideH" => b.between = Some(self.border(k)),
538                "insideV" => b.inside_v = Some(self.border(k)),

540            }
541        }

551fn margin_sides(e: &El) -> [Option<f32>; 4] {
552    let get = |names: &[&str]| names.iter().find_map(|n| e.child(n)).and_then(|c| tw(c, "w:w")).map(|v| v.clamp(0.0, 1584.0));

554}

558        let mut t = TableProps::default();
559        for k in e.els() {
560            match k.name.as_str() {

569                                .filter(|v| *v > 0.0)
570                        }
571                        "auto" | "nil" => {}
572                        _ => t.width = measure(w, 20.0).filter(|v| *v > 0.0),
573                    }
574                }
575                "w:jc" => t.align = k.attr("w:val").map(align),
576                "w:tblInd" => t.indent = tw(k, "w:w"),
577                "w:tblBorders" => t.borders = Some(self.borders(k)),
578                "w:shd" => t.shading = self.shd_fill(k),
579                "w:tblLayout" => t.fixed = k.attr("w:type") == Some("fixed"),
580                "w:bidiVisual" => t.bidi_visual = on_off(k),
581                "w:tblCellMar" => t.cell_margins = Some(margins(k)),
582                "w:tblLook" => t.look = look(k),
583                "w:tblpPr" => t.float = Some(table_float(k)),
584                "w:tblOverlap" => {
585                    if let Some(f) = t.float.as_mut() {
586                        f.overlap = k.attr("w:val") != Some("never");
587                    }
588                }
589                "w:tblCaption" => t.caption = k.attr("w:val").map(cap_alt),
590                "w:tblDescription" => t.description = k.attr("w:val").map(cap_alt),
591                _ => {}
592            }
593        }
594        t
595    }

598/// `w:tblLook` in either the attribute form or the legacy hex bitmask form.
599pub fn look(e: &El) -> TableLook {
600    let mut l = TableLook::default();
601    if let Some(v) = e.attr("w:val") {
602        let bits = u32::from_str_radix(v.trim(), 16).unwrap_or(0x04A0);
603        l.header_row = bits & 0x0020 != 0;
604        l.total_row = bits & 0x0040 != 0;
605        l.first_column = bits & 0x0080 != 0;
606        l.last_column = bits & 0x0100 != 0;
607        l.banded_rows = bits & 0x0200 == 0;
608        l.banded_columns = bits & 0x0400 == 0;
609    }
610    let b = |name: &str| e.attr(name).map(|v| !matches!(v, "0" | "false" | "off"));
611    if let Some(v) = b("w:firstRow") {
612        l.header_row = v;
613    }
614    if let Some(v) = b("w:lastRow") {
615        l.total_row = v;
616    }
617    if let Some(v) = b("w:firstColumn") {
618        l.first_column = v;
619    }
620    if let Some(v) = b("w:lastColumn") {
621        l.last_column = v;
622    }
623    if let Some(v) = b("w:noHBand") {
624        l.banded_rows = !v;
625    }
626    if let Some(v) = b("w:noVBand") {
627        l.banded_columns = !v;
628    }
629    l
630}

664/// `w:tcPr`; returns the props and whether the cell is a legacy `hMerge` continuation.
665pub fn tcpr(pc: &PropCtx, e: &El) -> (CellProps, bool) {
666    let mut c = CellProps { span: 1, ..Default::default() };
667    let mut hcont = false;
668    for k in e.els() {
669        match k.name.as_str() {
670            "w:tcW" => match k.attr("w:type") {
671                None | Some("dxa") => c.width = tw(k, "w:w").filter(|v| *v > 0.0),
672                Some("pct") => {
673                    let w = k.attr("w:w").unwrap_or("0");
674                    c.width_pct = if let Some(p) = w.strip_suffix('%') { measure(p, 1.0) } else { measure(w, 50.0) }
675                        .map(|v| v.clamp(0.0, 100.0))
676                        .filter(|v| *v > 0.0);
677                }
678                _ => {}
679            },
680            "w:gridSpan" => c.span = k.attr("w:val").and_then(u32_of).unwrap_or(1).clamp(1, 63),
681            "w:hMerge" => hcont = k.attr("w:val") != Some("restart"),
682            "w:vMerge" => c.vmerge = if k.attr("w:val") == Some("restart") { VMerge::Restart } else { VMerge::Continue },
683            "w:tcBorders" => c.borders = Some(pc.borders(k)),
684            "w:shd" => c.shading = pc.shd_fill(k),
685            "w:noWrap" => c.no_wrap = on_off(k),
686            "w:hideMark" => c.hide_mark = on_off(k),
687            "w:tcMar" => {
688                let sides = margin_sides(k);
689                if sides.iter().all(Option::is_some) {
690                    c.margins = Some(sides.map(|side| side.unwrap_or(0.0)));
691                } else {
692                    c.partial_margins = Some(sides);
693                }
694            }
695            "w:textDirection" => {
696                c.text_direction = TextDirection::from_ooxml(k.attr("w:val").unwrap_or(""));
697                c.text_direction_set = k.attr("w:val") == Some("lrTb");
698            }
699            "w:vAlign" => {
700                c.valign = Some(match k.attr("w:val") {
701                    Some("center") => VAlign::Center,
702                    Some("bottom") => VAlign::Bottom,
703                    _ => VAlign::Top,
704                })
705            }
706            _ => {}
707        }
708    }
709    (c, hcont)
710}

712/// A header/footer reference found in a `w:sectPr`: (is_footer, type, relationship id).
713pub struct HfRef {
714    pub footer: bool,
715    pub kind: String,
716    pub rid: String,
717}

719/// Reads `w:sectPr` element properties (header/footer references are returned for the caller to resolve).
720///
721/// Parses section page size, margins, columns, line numbering (`w:lnNumType`), page numbering,
722/// vertical alignment, borders, and document grid settings.
723pub fn sectpr(pc: &PropCtx, e: &El) -> (SectionProps, Vec<HfRef>) {
724    let mut s = SectionProps::default();
725    let mut refs = Vec::new();
726    for k in e.els() {
727        match k.name.as_str() {
728            "w:headerReference" | "w:footerReference" => {
729                if let Some(rid) = k.attr("r:id") {
730                    refs.push(HfRef {
731                        footer: k.name == "w:footerReference",
732                        kind: k.attr("w:type").unwrap_or("default").to_string(),
733                        rid: rid.to_string(),
734                    });
735                }
736            }
737            "w:type" => {
738                s.start = match k.attr("w:val") {
739                    Some("continuous") => SectionStart::Continuous,
740                    Some("evenPage") => SectionStart::EvenPage,
741                    Some("oddPage") => SectionStart::OddPage,
742                    Some("nextColumn") => SectionStart::NextColumn,
743                    _ => SectionStart::NextPage,
744                }
745            }
746            "w:pgSz" => {
747                if let Some(w) = tw(k, "w:w").filter(|v| *v >= 36.0) {
748                    s.page_w = w;
749                }
750                if let Some(h) = tw(k, "w:h").filter(|v| *v >= 36.0) {
751                    s.page_h = h;
752                }
753                s.landscape = match k.attr("w:orient") {
754                    Some("landscape") => true,
755                    Some(_) => false,
756                    None => s.page_w > s.page_h,
757                };
758            }
759            "w:pgMar" => {
760                let m = |n: &str, d: f32| tw(k, n).map(f32::abs).unwrap_or(d).min(s.page_w.max(s.page_h));
761                s.margin_top = m("w:top", s.margin_top);
762                s.margin_bottom = m("w:bottom", s.margin_bottom);
763                s.margin_left = m("w:left", s.margin_left);
764                s.margin_right = m("w:right", s.margin_right);
765                s.header = m("w:header", s.header);
766                s.footer = m("w:footer", s.footer);
767                s.gutter = m("w:gutter", s.gutter);
768            }
769            "w:cols" => {
770                let mut c = Columns { count: k.attr("w:num").and_then(u32_of).unwrap_or(1).clamp(1, 45), ..Columns::default() };
771                if let Some(sp) = tw(k, "w:space") {
772                    c.space = sp.max(0.0);
773                }
774                c.separator = k.attr("w:sep").is_some_and(|v| !matches!(v, "0" | "false" | "off"));
775                let equal = k.attr("w:equalWidth").is_none_or(|v| !matches!(v, "0" | "false" | "off"));
776                if !equal {
777                    c.widths = k
778                        .children("w:col")
779                        .take(45)
780                        .map(|col| (tw(col, "w:w").unwrap_or(0.0).max(0.0), tw(col, "w:space").unwrap_or(0.0).max(0.0)))
781                        .collect();
782                    if !c.widths.is_empty() {
783                        c.count = c.widths.len() as u32;
784                    }
785                }
786                s.columns = c;
787            }
788            "w:footnotePr" => s.footnote_pr = note_pr(k, true),
789            "w:endnotePr" => s.endnote_pr = note_pr(k, true),
790            "w:formProt" => s.form_prot = Some(on_off(k)),
791            "w:titlePg" => s.title_page = on_off(k),
792            "w:bidi" => s.rtl = on_off(k),
793            "w:textDirection" => {
794                s.text_direction = crate::doc::props::TextDirection::from_ooxml(k.attr("w:val").unwrap_or(""));
795                s.text_direction_set = k.attr("w:val") == Some("lrTb");
796            }
797            "w:pgNumType" => {
798                s.page_num_start = k.attr("w:start").and_then(u32_of);
799                if let Some(f) = k.attr("w:fmt") {
800                    s.page_num_format = NumFormat::from_ooxml(f);
801                }
802                s.chapter_style = k.attr("w:chapStyle").and_then(u32_of).filter(|v| *v <= u8::MAX as u32).map(|v| v as u8);
803                s.chapter_sep = k.attr("w:chapSep").and_then(crate::doc::section::ChapterSep::from_ooxml);
804                s.page_num_set = s.page_num_start.is_none() && s.page_num_format == NumFormat::Decimal;
805            }
806            "w:paperSrc" => {
807                let tray = |name| {
808                    let value = k.attr(name)?.trim();
809                    let digits = value.strip_prefix('+').unwrap_or(value);
810                    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) {
811                        return None;
812                    }
813                    u32_of(digits).filter(|v| *v <= u16::MAX as u32).map(|v| v as u16)
814                };
815                s.paper_src_first = tray("w:first");
816                s.paper_src_other = tray("w:other");
817            }
818            "w:noEndnote" => s.no_endnote = Some(on_off(k)),
819            "w15:footnoteColumns" => {
820                s.footnote_columns = k.attr("w:val").and_then(u32_of).filter(|v| (1..=45).contains(v));
821            }
822            "w:lnNumType" => {
823                // Word leaves line numbering off unless `w:countBy` gives an interval above 0,
824                // whatever `w:start`, `w:distance` and `w:restart` say (#6209, observed in Word;
825                // the RTF-to-DOCX converter there writes the element without `w:countBy`).
826                // Falling back to `LineNumbering::default()` would number every line instead.
827                if let Some(count_by) = k.attr("w:countBy").and_then(u32_of).filter(|&v| v > 0) {
828                    let mut l = LineNumbering { count_by: count_by.clamp(1, 100), ..LineNumbering::default() };
829                    if let Some(v) = k.attr("w:start").and_then(int) {
830                        l.start = (v.clamp(0, 32_767) + 1) as u32;
831                    }
832                    l.distance = tw(k, "w:distance").unwrap_or(0.0).max(0.0);
833                    l.restart = match k.attr("w:restart") {
834                        Some("newSection") => LineNumberRestart::Section,
835                        Some("continuous") => LineNumberRestart::Continuous,
836                        _ => LineNumberRestart::Page,
837                    };
838                    s.line_numbers = Some(l);
839                }
840            }
841            "w:vAlign" => {
842                s.valign = match k.attr("w:val") {
843                    Some("center") => PageVAlign::Center,
844                    Some("both") => PageVAlign::Justify,
845                    Some("bottom") => PageVAlign::Bottom,
846                    _ => PageVAlign::Top,
847                }
848            }
849            "w:pgBorders" => s.page_borders = Some(pc.borders(k)),
850            "w:docGrid" => {
851                let mut g = DocGrid::default();
852                if let Some(t) = k.attr("w:type").and_then(DocGridType::from_ooxml) {
853                    g.kind = t;
854                }
855                if let Some(p) = tw(k, "w:linePitch").filter(|v| v.is_finite() && *v > 0.0) {
856                    g.line_pitch = p.min(MAX_GRID_PITCH);
857                }
858                if let Some(c) = k.attr("w:charSpace").and_then(int) {
859                    g.char_space = c.clamp(-(MAX_CHAR_SPACE as i64), MAX_CHAR_SPACE as i64) as i32;
860                }
861                s.doc_grid = Some(g);
862            }
863            _ => {}
864        }
865    }
866    (s, refs)
867}

869/// `w:footnotePr` / `w:endnotePr` (ECMA-376 §17.11): placement, number format (only kept when
870/// `fmt`: the document's goes to its own setting), start and restart. A start out of range is

876        num_start: e.child_val("w:numStart").and_then(int).map(|v| v.clamp(1, crate::doc::section::MAX_NOTE_START as i64) as u32),
877        num_restart: e.child_val("w:numRestart").and_then(NoteRestart::from_ooxml),
878    }

881/// `w:tblpPr`: a floating table's anchors, offsets or alignments and distances from text.
882fn table_float(e: &El) -> TableFloat {
883    use crate::doc::para::{Anchor, FloatAlign};
884    let pt = |n: &str| e.attr(n).and_then(|v| measure(v, 20.0)).map(|v| v.clamp(-31_680.0, 31_680.0)).unwrap_or(0.0);
885    let dist = |n: &str| pt(n).clamp(0.0, 1584.0);
886    let rel = |n: &str, text: Anchor| match e.attr(n) {
887        Some("page") => Anchor::Page,
888        Some("margin") => Anchor::Margin,
889        _ => text,
890    };
891    let align = |n: &str| e.attr(n).and_then(FloatAlign::from_ooxml);
892    TableFloat {
893        h_rel: rel("w:horzAnchor", Anchor::Column),
894        v_rel: rel("w:vertAnchor", Anchor::Paragraph),
895        x: pt("w:tblpX"),
896        y: pt("w:tblpY"),
897        h_align: align("w:tblpXSpec"),
898        v_align: align("w:tblpYSpec"),
899        dist: [dist("w:leftFromText"), dist("w:topFromText"), dist("w:rightFromText"), dist("w:bottomFromText")],
900        overlap: true,
901    }
902}

904/// `w:eastAsianLayout` (ISO/IEC 29500-1 §17.3.2.10). Attributes the schema doesn't allow (an id
905/// out of `xsd:int`, an unknown bracket, an on/off that isn't one) are left unset.
906fn east_asian_layout(e: &El) -> EastAsianLayout {
907    let flag = |name: &str| {
908        e.attr(name).and_then(|v| match v.trim() {
909            "1" | "true" | "on" => Some(true),
910            "0" | "false" | "off" => Some(false),
911            _ => None,
912        })
913    };
914    EastAsianLayout {
915        id: e.attr("w:id").and_then(int).and_then(|v| i32::try_from(v).ok()),
916        combine: flag("w:combine"),
917        combine_brackets: e.attr("w:combineBrackets").and_then(|v| CombineBrackets::from_ooxml(v.trim())),
918        vert: flag("w:vert"),
919        vert_compress: flag("w:vertCompress"),
920    }
