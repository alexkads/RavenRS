7    pub(super) fn read_settings(&mut self, path: &str) -> Result<(), DocxError> {
8        let Some(root) = self.xml(path)? else { return Ok(()) };
9        if let Some(b) = self.pkg.get(path).filter(|b| crate::docx::write::settings_has_unowned(b)) {
10            self.doc.passthrough.insert(crate::docx::package::SETTINGS_ORIG.into(), Arc::new(b.to_vec()));
11        }
12        let rels = self.pkg.rels(path);
13        let s = &mut self.doc.settings;
14        for k in root.els() {
15            match k.name.as_str() {
16                // §17.15.1.6: the template, by a relationship (usually external) of the settings part.
17                "w:attachedTemplate" => {
18                    s.attached_template = k
19                        .attr("r:id")
20                        .and_then(|id| rels.by_id(id))
21                        .map(|r| r.target.trim().to_string())
22                        .filter(|t| !t.is_empty() && t.len() <= MAX_TEMPLATE_PATH);
23                }
24                "w:linkStyles" => s.link_styles = on_off(k),
25                "w:removePersonalInformation" => s.remove_personal_info = on_off(k),
26                "w:removeDateAndTime" => s.remove_date_time = on_off(k),
27                "w:compat" => {
28                    s.allow_space_of_same_style_in_table = k.child("w:allowSpaceOfSameStyleInTable").is_some_and(on_off);
29                    // Only Word's own setting: another `w:uri` names a different application's.
30                    let mode = k
31                        .children("w:compatSetting")
32                        .find(|c| {
33                            c.attr("w:name") == Some("compatibilityMode")
34                                && c.attr("w:uri").is_none_or(|u| u == "http://schemas.microsoft.com/office/word")
35                        })
36                        .and_then(|c| c.attr("w:val"));
37                    if let Some(m) = mode.and_then(u32_of) {
38                        s.compat_mode = m.clamp(11, 99);
39                    }
40                    s.sum_paragraph_spacing = k.children("w:doNotUseHTMLParagraphAutoSpacing").any(on_off);
41                    s.do_not_expand_shift_return = k.children("w:doNotExpandShiftReturn").any(on_off);
42                }
43                "w:trackRevisions" => s.track_changes = on_off(k),
44                "w:doNotTrackMoves" => s.track_moves = !on_off(k),
45                "w:defaultTabStop" => {
46                    if let Some(v) = tw(k, "w:val").filter(|v| *v > 0.0) {
47                        s.default_tab = v.clamp(1.0, 1584.0);
48                    }
49                }
50                "w:evenAndOddHeaders" => s.even_odd_headers = on_off(k),
51                "w:drawingGridHorizontalSpacing" | "w:drawingGridVerticalSpacing" => {
52                    if let Some(v) = tw(k, "w:val").filter(|v| *v > 0.0) {
53                        let v = v.clamp(0.5, 1584.0);
54                        if k.name == "w:drawingGridHorizontalSpacing" { s.grid_h = v } else { s.grid_v = v }
55                    }
56                }
57                "m:mathPr" => s.math = Some(math::read_math_pr(k)),
58                "w:mirrorMargins" => s.mirror_margins = on_off(k),
59                "w:autoHyphenation" => s.auto_hyphenation = on_off(k),
60                "w:doNotHyphenateCaps" => s.hyphenate_caps = !on_off(k),
61                "w:hyphenationZone" => {
62                    if let Some(v) = tw(k, "w:val").filter(|v| v.is_finite() && *v >= 0.0) {
63                        s.hyphenation_zone = v.min(1584.0);
64                    }
65                }
66                "w:consecutiveHyphenLimit" => {
67                    if let Some(v) = k.attr("w:val").and_then(u32_of) {
68                        s.consecutive_hyphen_limit = v.min(32_767);
69                    }
70                }
71                "w:footnotePr" => {
72                    if let Some(f) = k.child_val("w:numFmt") {
73                        s.footnote_format = NumFormat::from_ooxml(f);
74                    }
75                    s.footnote_pr = props::note_pr(k, false);
76                }
77                "w:endnotePr" => {
78                    if let Some(f) = k.child_val("w:numFmt") {
79                        s.endnote_format = NumFormat::from_ooxml(f);
80                    }
81                    s.endnote_pr = props::note_pr(k, false);
82                }
83                "w:documentProtection" => read_protection(k, s),
84                "w:zoom" => {
85                    let percent = k.attr("w:percent").and_then(|v| u32_of(v.trim_end_matches('%'))).unwrap_or(100).clamp(10, 500);
86                    let fit = match k.attr("w:val") {
87                        Some("fullPage") => Some(ZoomFit::OnePage),
88                        Some("bestFit") => Some(ZoomFit::PageWidth),
89                        Some("textFit") => Some(ZoomFit::TextWidth),
90                        _ => None,
91                    };
92                    s.zoom = Some(Zoom { percent, fit }).filter(|z| z.percent != 100 || z.fit.is_some());
93                }
94                _ => {}
95            }
96        }
97        Ok(())
98    }

100    pub(super) fn read_notes(&mut self, path: &str, foot: bool) -> Result<(), DocxError> {
101        let Some(root) = self.xml(path)? else { return Ok(()) };
102        let rels = self.pkg.rels(path);
103        let tag = if foot { "w:footnote" } else { "w:endnote" };
104        for n in root.children(tag) {
105            match n.attr("w:type") {
106                None | Some("normal") => {}
107                Some(t @ ("separator" | "continuationSeparator" | "continuationNotice")) => {
108                    self.read_note_sep(n, &rels, foot, t);
109                    continue;
110                }
111                Some(_) => continue,
112            }
113            let Some(fid) = n.attr("w:id").and_then(int) else { continue };
114            if self.doc.parts.len() >= MAX_PARTS {
115                break;
116            }
117            // Reserve the id first: the note's own reference mark points at it.
118            let (pk, kind) = if foot { (PartKind::Footnote, NoteKind::Footnote) } else { (PartKind::Endnote, NoteKind::Endnote) };
119            let id = self.doc.add_part(pk, Blocks::new());
120            self.doc.missing_pictures.part_names.insert(id, self.story.clone());
121            let mut sc = StoryCtx::default();
122            let mut blocks = Blocks::new();
123            self.current_note = Some((kind, id));
124            self.read_blocks(&mut sc, n, &rels, &mut blocks, self.chunks.depth);
125            self.flush_pending(&mut sc, &mut blocks);
126            self.current_note = None;
127            if let Some(p) = self.doc.parts.get_mut(&id)
128                && !blocks.is_empty()
129            {
130                p.blocks = blocks;
131            }
132            if foot {
133                self.footnotes.insert(fid, id);
134            } else {
135                self.endnotes.insert(fid, id);
136            }
137        }
138        Ok(())
139    }

141    /// A document's own separator, continuation separator or continuation notice (`kind`, the
142    /// note's `w:type`) for its footnotes or endnotes. The first of each kind counts.
143    fn read_note_sep(&mut self, n: &El, rels: &Rels, foot: bool, kind: &str) {
144        let seps = if foot { &self.doc.footnote_separators } else { &self.doc.endnote_separators };
145        let taken = match kind {
146            "separator" => seps.separator.is_some(),
147            "continuationSeparator" => seps.continuation_separator.is_some(),
148            _ => seps.continuation_notice.is_some(),
149        };
150        if taken {
151            return;
152        }
153        let mut sc = StoryCtx::default();
154        let mut blocks = Blocks::new();
155        self.in_note_sep = true;
156        self.read_blocks(&mut sc, n, rels, &mut blocks, self.chunks.depth);
157        self.flush_pending(&mut sc, &mut blocks);
158        self.in_note_sep = false;
159        if blocks.is_empty() {
160            blocks.push(para_block(Paragraph::new()));
161        }
162        let seps = if foot { &mut self.doc.footnote_separators } else { &mut self.doc.endnote_separators };
163        let slot = match kind {
164            "separator" => &mut seps.separator,
165            "continuationSeparator" => &mut seps.continuation_separator,
166            _ => &mut seps.continuation_notice,
167        };
168        *slot = Some(blocks);
169    }

171    /// The extended properties a save carries over ([`crate::docx::package::APP_KEPT_PROPS`]).
172    pub(super) fn read_app(&mut self, path: &str) {
173        let Ok(Some(root)) = self.xml(path) else { return };
174        let mut kept = String::new();
175        for e in root.els() {
176            let name = e.local();
177            let value = e.text();
178            if crate::docx::package::APP_KEPT_PROPS.contains(&name) && value.len() <= 2048 && !value.contains(['\t', '\n']) {
179                kept.push_str(&format!("{name}\t{value}\n"));
180            }
181        }
182        if !kept.is_empty() {
183            self.doc.passthrough.insert(crate::docx::package::APP_KEPT.into(), Arc::new(kept.into_bytes()));
184        }
185    }

187    pub(super) fn read_comments(&mut self, path: &str, ext: Option<&str>, ids: Option<&str>, cex: Option<&str>) -> Result<(), DocxError> {
188        let Some(root) = self.xml(path)? else { return Ok(()) };
189        let rels = self.pkg.rels(path);
190        let mut by_para: HashMap<String, u32> = HashMap::new();
191        let mut next_id = 0u32;
192        // Straight into the document, beside the comments chunks in notes brought in.
193        for c in root.children("w:comment") {
194            if self.doc.parts.len() >= MAX_PARTS {
195                break;
196            }
197            let file_id = c.attr("w:id").unwrap_or("").to_string();
198            if self.comment_map.contains_key(&file_id) {
199                continue;
200            }
201            let id = match file_id.parse::<u32>() {
202                Ok(v) if !self.doc.comments.contains_key(&v) => v,
203                _ => {
204                    // Every id handed out so far is a key of `doc.comments` (inserted below).
205                    while self.doc.comments.contains_key(&next_id) {
206                        next_id += 1;
207                    }
208                    next_id
209                }
210            };
211            // Taken before its text is read: a chunk there brings comments of its own.
212            self.doc.comments.insert(id, Comment::default());
213            let mut sc = StoryCtx::default();
214            let mut blocks = Blocks::new();
215            self.read_blocks(&mut sc, c, &rels, &mut blocks, self.chunks.depth);
216            self.flush_pending(&mut sc, &mut blocks);
217            let part = self.doc.add_part(PartKind::Comment, blocks);
218            self.doc.missing_pictures.part_names.insert(part, self.story.clone());
219            if let Some(pid) = c.children("w:p").last().and_then(|p| p.attr("w14:paraId")) {
220                by_para.insert(pid.to_string(), id);
221            }
222            self.doc.comments.insert(
223                id,
224                Comment {
225                    author: c.attr("w:author").unwrap_or("").to_string(),
226                    initials: c.attr("w:initials").unwrap_or("").to_string(),
227                    date: c.attr("w:date").unwrap_or("").to_string(),
228                    parent: None,
229                    resolved: false,
230                    part,
231                    ..Default::default()
232                },
233            );
234            self.comment_map.insert(file_id, id);
235        }
236        if let Some(ext) = ext
237            && let Ok(Some(x)) = self.xml(ext)
238        {
239            for e in x.els().filter(|e| e.local() == "commentEx") {
240                let Some(cid) = e.attr("w15:paraId").and_then(|p| by_para.get(p)).copied() else { continue };
241                let parent = e.attr("w15:paraIdParent").and_then(|p| by_para.get(p)).copied().filter(|p| *p != cid);
242                let done = e.attr("w15:done").is_some_and(|v| v == "1" || v == "true");
243                if let Some(c) = self.doc.comments.get_mut(&cid) {
244                    c.parent = parent;
245                    c.resolved = done;
246                }
247            }
248        }
249        // Durable ids (by paragraph id) and UTC dates (by durable id), which the writer
250        // carries over (`write_comment_ids`). Their namespaces (w16cid, w16cex) have no canonical
251        // prefix here, so attributes are found by local name.
252        let local = |e: &crate::docx::xml::El, name: &str| -> Option<String> {
253            e.attrs.iter().find(|(k, _)| k.rsplit(':').next() == Some(name)).map(|(_, v)| v.clone())
254        };
255        let mut by_durable: HashMap<String, u32> = HashMap::new();
256        if let Some(ids) = ids
257            && let Ok(Some(x)) = self.xml(ids)
258        {
259            for e in x.els().filter(|e| e.local() == "commentId") {
260                let Some(cid) = local(e, "paraId").and_then(|p| by_para.get(&p).copied()) else { continue };
261                let Some(durable) = local(e, "durableId").filter(|d| crate::docx::write::durable_id_ok(d)) else { continue };
262                if let Some(c) = self.doc.comments.get_mut(&cid) {
263                    c.durable_id = Some(durable.to_ascii_uppercase());
264                    by_durable.insert(durable.to_ascii_uppercase(), cid);
265                }
266            }
267        }
268        if let Some(cex) = cex
269            && let Ok(Some(x)) = self.xml(cex)
270        {
271            for e in x.els().filter(|e| e.local() == "commentExtensible") {
272                let Some(cid) = local(e, "durableId").and_then(|d| by_durable.get(&d.to_ascii_uppercase()).copied()) else { continue };
273                let date = local(e, "dateUtc").filter(|d| d.len() <= 40 && d.chars().all(|c| c.is_ascii_alphanumeric() || ":-.+".contains(c)));
274                if let (Some(c), Some(date)) = (self.doc.comments.get_mut(&cid), date) {
275                    c.date_utc = Some(date);
276                }
277            }
278        }
279        Ok(())
280    }

282    /// Sources from a `b:Sources` custom XML part (other custom XML is left alone); at most
283    /// [`MAX_SOURCES`] in all, and the document's citation style from `StyleName`.
284    pub(super) fn read_bibliography(&mut self, path: &str, seen: &mut Seen) -> Result<(), DocxError> {
285        let Some(root) = self.xml(path)? else { return Ok(()) };
286        if root.local() != "Sources" {
287            return Ok(());
288        }
289        let room = MAX_SOURCES.saturating_sub(self.doc.sources.len());
290        let bib = crate::docx::bibliography::parse(&root, seen, room);
291        if self.doc.settings.bib_style.is_none() {
292            self.doc.settings.bib_style = bib.style;
293        }
294        self.doc.sources.extend(bib.sources);
295        Ok(())
296    }

298    pub(super) fn read_core(&mut self, path: &str) {
299        let Ok(Some(root)) = self.xml(path) else { return };
300        let c = &mut self.doc.core;
301        for k in root.els() {
302            let t = k.text().trim().to_string();
303            match k.name.as_str() {
304                "dc:title" => c.title = t,
305                "dc:subject" => c.subject = t,
306                "dc:creator" => c.creator = t,
307                "cp:keywords" => c.keywords = t,
308                "dc:description" => c.description = t,
309                "cp:lastModifiedBy" => c.last_modified_by = t,
310                "cp:revision" => c.revision = u32_of(&t).unwrap_or(0),
311                "dcterms:created" => c.created = t,
312                "dcterms:modified" => c.modified = t,
313                "cp:category" => c.category = t,
314                _ => {}
315            }
316        }
317    }

320/// The password hash attributes of `w:documentProtection` that are kept (ECMA-376 §17.15.1.29:
321/// the current `algorithmName`/`hashValue`/`saltValue`/`spinCount`, and the older transitional
322/// `crypt*`/`hash`/`salt` set).
323pub(crate) const PROTECTION_HASH_ATTRS: [&str; 16] = [
324    "w:algorithmName",
325    "w:hashValue",
326    "w:saltValue",
327    "w:spinCount",
328    "w:cryptProviderType",
329    "w:cryptAlgorithmClass",
330    "w:cryptAlgorithmType",

333    "w:cryptProvider",

336    "w:cryptProviderTypeExt",

342/// Restrict Editing (`w:documentProtection`, §17.15.1.29): the editing restriction, whether
343/// formatting is limited to unlocked styles, and the password hash. Only enforced protection is
344/// kept; a hostile attribute value can't be longer than a hash needs.
345fn read_protection(k: &El, s: &mut crate::doc::Settings) {
346    let on = |v: &str| !matches!(v, "0" | "false" | "off");
347    if !k.attr("w:enforcement").is_some_and(on) {
348        return;
349    }
350    let edit = k.attr("w:edit").filter(|e| matches!(*e, "readOnly" | "comments" | "trackedChanges" | "forms"));
351    let formatting = k.attr("w:formatting").is_some_and(on);
352    if edit.is_none() && !formatting {
353        return;
354    }
355    s.protection = Some(edit.unwrap_or("none").to_string());
356    s.protect_formatting = formatting;
357    s.protect_hash = PROTECTION_HASH_ATTRS
358        .iter()
359        .filter_map(|n| k.attr(n).filter(|v| !v.is_empty() && v.len() <= 1024).map(|v| (n.to_string(), v.to_string())))
360        .collect();
