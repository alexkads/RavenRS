1//! `w:altChunk` (ECMA-376 §17.17.2.1): content kept in another format (HTML, MHT, RTF, plain
2//! text, another WordprocessingML document) that Word brings into the document where the chunk
3//! stands when it opens the file. A nested WordprocessingML package is read here; other formats
4//! go to the caller's [`ChunkImporter`] (the format crates sit beside this one), and without one
5//! the chunk is left out.
6

16/// Most chunks brought in from one file (nested documents' chunks included).
17pub(crate) const MAX_CHUNKS: usize = 256;
18/// Deepest a WordprocessingML chunk may nest in another.
19pub(crate) const MAX_CHUNK_NESTING: u8 = 4;
20
21/// An alternative format chunk's content, as a [`ChunkImporter`] gets it.
22pub struct AltChunk<'a> {
23    /// The part's content type (`application/xhtml+xml`, `message/rfc822`, `application/rtf`,
24    /// `text/plain`…), empty when the package doesn't give one.
25    pub content_type: &'a str,
26    /// The part's name (`word/afchunk.mht`): its extension tells the format when the content type
27    /// doesn't.
28    pub part: &'a str,
29    pub bytes: &'a [u8],
30}
31
32/// Turns a chunk the docx reader can't read itself into a document whose body takes the chunk's
33/// place; `None` leaves the chunk out.
34pub type ChunkImporter<'a> = &'a dyn Fn(&AltChunk) -> Option<Document>;
35
36/// How [`crate::docx::read_with`] reads a package.
37#[derive(Clone, Copy, Default)]
38pub struct ReadOptions<'a> {
39    /// The password of an encrypted package.
40    pub password: Option<&'a str>,
41    /// Imports `w:altChunk` content in formats other than WordprocessingML.
42    pub chunks: Option<ChunkImporter<'a>>,
43}
44
45impl Reader<'_> {
46    /// Bring the chunk `e` names into `out` where it stands (`depth`: of the XML around it).
47    pub(crate) fn read_alt_chunk(&mut self, sc: &mut StoryCtx, e: &El, rels: &Rels, out: &mut Blocks, depth: usize) {
48        let Some(part) = e.attr("r:id").and_then(|id| part_of(rels, id, rt::AF_CHUNK)) else { return };
49        let pkg = self.pkg;
50        let Some(bytes) = pkg.get(&part) else {
51            log::warn!("docx: alternative format chunk {part} is missing");
52            return;
53        };
54        // Every chunk's bytes count against one budget for the whole file, however often the same
55        // part is named and however deep chunks nest.
56        if self.chunks.left == 0 || bytes.len() as u64 > self.chunks.bytes {
57            log::warn!("docx: more than {MAX_CHUNKS} alternative format chunks or {MAX_TOTAL} bytes of them; the rest are left out");
58            return;
59        }
60        self.chunks.left -= 1;
61        self.chunks.bytes -= bytes.len() as u64;
62        let ct = self.content_types.get_or_insert_with(|| ContentTypes::read(pkg)).of(&part).unwrap_or("").to_string();
63        let chunk = AltChunk { content_type: &ct, part: &part, bytes };
64        // A WordprocessingML chunk is a package; one that isn't Word's fails to read (logged).
65        let is_docx = bytes.starts_with(b"PK\x03\x04");
66        let doc = if is_docx { self.read_nested(bytes, depth) } else { self.chunks.import.and_then(|f| f(&chunk)) };
67        let Some(mut doc) = doc else {
68            log::warn!("docx: alternative format chunk {part} ({ct}) is in a format WordCraft can't bring in");
69            return;
70        };
71        // The chunk flows in the section it stands in; its own section breaks don't come along.
72        for b in &mut doc.body {
73            if let Block::Para(p) = &**b
74                && p.section.is_some()
75                && let Block::Para(p) = Arc::make_mut(b)
76            {
77                p.section = None;
78            }
79        }
80        // `w:matchSrc`: a Word document's content keeps its own styles (other formats' formatting
81        // is their own already).
82        let keep = is_docx && e.child("w:altChunkPr").and_then(|p| p.child("w:matchSrc")).is_some_and(crate::docx::units::on_off);
83        // The pictures the chunk's own reader dropped are the file's too (strict export, #6005),
84        // each named by the chunk and its part in it (`word/afchunk1.docx > word/header1.xml`).
85        // A chunk in another format brings none: those importers record nothing yet (#6007).
86        let nested = std::mem::take(&mut doc.missing_pictures);
87        let list_len = nested.list.len() as u64;
88        for mut item in nested.list {
89            let inner = if item.story.is_empty() { nested.main.as_str() } else { item.story.as_str() };
90            item.story = format!("{part} > {inner}");
91            self.doc.missing_pictures.push(item);
92        }
93        // The nested records past its list are the file's too; their stories carry the chunk's
94        // part, so none is one the outer list already counted.
95        let unlisted = nested.total.saturating_sub(list_len);
96        self.doc.missing_pictures.add_unlisted(&part, unlisted);
97        self.doc.missing_pictures.truncated |= nested.truncated;
98        let mut blocks = self.doc.adopt_blocks(doc, keep, &mut self.next_perm);
99        // Bookmarks and comment marks read before the chunk stay before it.
100        self.flush_pending_before(sc, out, &mut blocks);
101        out.extend(blocks.into_iter().map(Arc::new));
102    }

104    /// A WordprocessingML chunk: its document, read with the same importer and what is left of
105    /// the budgets, its XML depth counting on from `depth`.
106    fn read_nested(&mut self, bytes: &[u8], depth: usize) -> Option<Document> {
107        if self.chunks.nesting >= MAX_CHUNK_NESTING {
108            log::warn!("docx: alternative format chunks nest deeper than {MAX_CHUNK_NESTING}; the inner ones are left out");
109            return None;
110        }
111        let pkg =
112            Package::open_limited(bytes, MAX_PART, self.chunks.bytes).map_err(|e| log::warn!("docx: unreadable nested document chunk: {e}")).ok()?;
113        self.chunks.bytes = self.chunks.bytes.saturating_sub(pkg.size());
114        let state = ChunkState { nesting: self.chunks.nesting + 1, depth: depth + 1, ..self.chunks };
115        let (doc, left, bytes, graphic) =
116            super::read_opened_in(&pkg, state).map_err(|e| log::warn!("docx: unreadable nested document chunk: {e}")).ok()?;
117        self.chunks.left = left;
118        self.chunks.bytes = bytes;
119        self.chunks.graphic = graphic;
120        Some(doc)
121    }
122}

124/// What a reader may still bring in from chunks.
125#[derive(Clone, Copy)]
126pub(crate) struct ChunkState<'a> {
127    /// Chunks left in the file's budget ([`MAX_CHUNKS`]).
128    pub left: usize,
129    /// Chunk bytes (nested packages unpacked) left in the file's budget.
130    pub bytes: u64,
131    /// Graphic work left in the file's budget ([`super::MAX_GRAPHIC_WORK`]): nested documents'
132    /// charts and diagrams count against it too.
133    pub graphic: usize,
134    /// How deep in nested WordprocessingML chunks this reader is.
135    pub nesting: u8,
136    /// The XML depth its stories start at (a nested document's: that of the chunk naming it).
137    pub depth: usize,
138    pub import: Option<ChunkImporter<'a>>,
139}
140
141impl<'a> ChunkState<'a> {
142    pub fn new(import: Option<ChunkImporter<'a>>) -> Self {
143        ChunkState { left: MAX_CHUNKS, bytes: MAX_TOTAL, graphic: super::MAX_GRAPHIC_WORK, nesting: 0, depth: 0, import }
144    }
145}

156    /// A package whose body is `body`, with the document relationship `rel` (id, type, target) to
157    /// the part `(name, bytes)`.
158    fn package(body: &str, (id, ty, target): (&str, &str, &str), (name, bytes): (&str, &[u8])) -> Vec<u8> {
159        let ns = r#"xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:wp="http://schemas.openxmlformats.org/drawingml/2006/wordprocessingDrawing" xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main""#;
160        let rels =
161            |inner: &str| format!(r#"<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">{inner}</Relationships>"#);
162        let rel = |id: &str, ty: &str, target: &str| {
163            format!(r#"<Relationship Id="{id}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/{ty}" Target="{target}"/>"#)
164        };
165        let files: [(&str, Vec<u8>); 4] = [
166            ("_rels/.rels", rels(&rel("rId1", "officeDocument", "word/document.xml")).into_bytes()),
167            ("word/document.xml", format!(r#"<w:document {ns}><w:body>{body}</w:body></w:document>"#).into_bytes()),
168            ("word/_rels/document.xml.rels", rels(&rel(id, ty, target)).into_bytes()),
169            (name, bytes.to_vec()),
170        ];
171        let mut zw = zip::ZipWriter::new(std::io::Cursor::new(Vec::new()));
172        for (name, bytes) in files {
173            zw.start_file(name, zip::write::SimpleFileOptions::default()).unwrap();
174            zw.write_all(&bytes).unwrap();
175        }
176        zw.finish().unwrap().into_inner()
177    }
178
179    /// Whether the body's first paragraph's chart is drawn.
180    fn drawn(d: &Document) -> bool {
181        let Some(Block::Para(p)) = d.body.first().map(|b| &**b) else { panic!("no paragraph") };
182        p.objects.iter().any(|o| matches!(o, InlineObject::Graphic { graphic, .. } if !graphic.items.is_empty()))
183    }

185    #[test]
186    fn a_nested_documents_charts_count_against_the_files_graphic_budget() {
187        let chart = r#"<c:chartSpace xmlns:c="http://schemas.openxmlformats.org/drawingml/2006/chart"><c:chart><c:plotArea><c:barChart><c:ser><c:val><c:numLit><c:ptCount val="2"/><c:pt idx="0"><c:v>1</c:v></c:pt><c:pt idx="1"><c:v>2</c:v></c:pt></c:numLit></c:val></c:ser></c:barChart></c:plotArea></c:chart></c:chartSpace>"#;
188        let drawing = r#"<w:p><w:r><w:drawing><wp:inline><wp:extent cx="2743200" cy="1828800"/><wp:docPr id="5" name="G"/><a:graphic><a:graphicData uri="http://schemas.openxmlformats.org/drawingml/2006/chart"><c:chart xmlns:c="http://schemas.openxmlformats.org/drawingml/2006/chart" r:id="rIdChart"/></a:graphicData></a:graphic></wp:inline></w:drawing></w:r></w:p>"#;
189        let inner = package(drawing, ("rIdChart", "chart", "charts/chart1.xml"), ("word/charts/chart1.xml", chart.as_bytes()));
190        let outer = package(r#"<w:altChunk r:id="rIdC"/>"#, ("rIdC", "aFChunk", "afchunk.docx"), ("word/afchunk.docx", &inner));
191        let pkg = Package::open(&outer).unwrap();
192        let read = |graphic| {
193            let (d, .., left) = super::super::read_opened_in(&pkg, ChunkState { graphic, ..ChunkState::new(None) }).unwrap();
194            (drawn(&d), left)
195        };
196        let (shown, left) = read(super::super::MAX_GRAPHIC_WORK);
197        assert!(shown);
198        assert!(left < super::super::MAX_GRAPHIC_WORK, "the nested chart's work is the file's");
199        // With the file's budget spent, the nested document doesn't get one of its own.
200        assert_eq!(read(0), (false, 0));
201    }
