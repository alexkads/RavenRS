11/// An `m:oMath` element as an equation (structure plus its source XML).
12pub fn read_omath(e: &El, jc: MathJc) -> Math {
13    Math { nodes: merge_runs(arg(e, 0)), omml: e.to_xml(), jc, linear: false }
14}

16/// A colour every math run shares, taken off the runs: it is the equation's own colour, as
17/// [`write_equation`](crate::docx::write::math::write_equation) spreads it (#1467). The source OMML
18/// is kept: the writer replaces it only when the colour changes.
19pub fn lift_color(math: &mut Math) -> Option<Rgb> {
20    let (mut shared, mut uniform) = (None, true);
21    crate::doc::math_edit::each_run_mut(&mut math.nodes, &mut |r| match shared {
22        None => shared = Some(r.color),
23        Some(c) => uniform &= c == r.color,
24    });
25    let c = shared.flatten().filter(|_| uniform)?;
26    crate::doc::math_edit::each_run_mut(&mut math.nodes, &mut |r| r.color = None);
27    Some(c)
28}

30/// `m:oMathParaPr/m:jc`.
31pub fn para_jc(para: &El) -> MathJc {
32    match para.child("m:oMathParaPr").and_then(|p| p.child("m:jc")).and_then(|j| j.attr("m:val")) {
33        Some("left") => MathJc::Left,
34        Some("right") => MathJc::Right,
35        Some("center") => MathJc::Center,
36        _ => MathJc::CenterGroup,
37    }
38}

40/// The settings part's `m:mathPr`: how display equations wrap.
41pub fn read_math_pr(e: &El) -> MathProps {
42    let mut m = MathProps::default();
43    let pr = Some(e);
44    m.brk_bin = match val(pr, "m:brkBin") {
45        Some("after") => BrkBin::After,
46        Some("repeat") => BrkBin::Repeat,
47        _ => BrkBin::Before,
48    };
49    m.brk_bin_sub = match val(pr, "m:brkBinSub") {
50        Some("-+") => BrkBinSub::MinusPlus,
51        Some("+-") => BrkBinSub::PlusMinus,
52        _ => BrkBinSub::MinusMinus,
53    };
54    if let Some(v) = e.child("m:wrapIndent").and_then(|c| crate::docx::units::tw(c, "m:val")).filter(|v| v.is_finite() && *v >= 0.0) {
55        m.wrap_indent = v.min(1584.0);
56    }
57    m.wrap_right = prop(pr, "m:wrapRight", false);
58    m
59}

61/// OMML on/off property: absent `m:val` = on.
62fn on(e: &El) -> bool {
63    match e.attr("m:val") {
64        None => true,
65        Some(v) => !matches!(v.trim(), "0" | "false" | "off" | "False" | "FALSE"),
66    }
67}

69/// An on/off property `name` in the properties element `pr` (absent = `default`).
70fn prop(pr: Option<&El>, name: &str, default: bool) -> bool {
71    pr.and_then(|p| p.child(name)).map(on).unwrap_or(default)
72}

74/// A character property (`m:chr`, `m:begChr`…): `None` when absent, `Some(None)` when empty.
75fn chr(pr: Option<&El>, name: &str) -> Option<Option<char>> {
76    let c = pr?.child(name)?;
77    match c.attr("m:val") {
78        Some(v) => Some(v.chars().next()),
79        None => Some(None),
80    }
81}

83fn val<'a>(pr: Option<&'a El>, name: &str) -> Option<&'a str> {
84    pr.and_then(|p| p.child(name)).and_then(|c| c.attr("m:val"))

92/// Read the math content of an element (`m:oMath`, `m:e`, `m:num`…).
93fn arg(e: &El, depth: usize) -> Arg {
94    let mut out = Vec::new();
95    if depth > MAX_DEPTH {
96        return out;
97    }
98    for k in e.els() {
99        node(k, depth, &mut out);
100    }
101    out
102}

104fn node(k: &El, depth: usize, out: &mut Arg) {
105    let d = depth + 1;
106    match k.name.as_str() {
107        "m:r" => {
108            if let Some(r) = run(k) {
109                out.push(MNode::Run(r));
110            }
111        }
112        // Word puts ordinary runs in math zones for normal text in some files.
113        "w:r" => {
114            let text = run_text(k);
115            if !text.is_empty() {
116                let mut r = MRun { text, nor: true, sty: Some(MSty::Plain), ..Default::default() };
117                wrpr(k.child("w:rPr"), &mut r);
118                out.push(MNode::Run(r));
119            }
120        }
121        "m:f" => {
122            let pr = k.child("m:fPr");
123            let kind = match val(pr, "m:type") {
124                Some("skw") => FracKind::Skewed,
125                Some("lin") => FracKind::Linear,
126                Some("noBar") => FracKind::NoBar,
127                _ => FracKind::Bar,
128            };
129            out.push(MNode::Frac { kind, num: sub_arg(k, "m:num", d), den: sub_arg(k, "m:den", d) });
130        }
131        "m:sSup" => out.push(MNode::Script { kind: ScriptKind::Sup, base: sub_arg(k, "m:e", d), sub: Vec::new(), sup: sub_arg(k, "m:sup", d) }),
132        "m:sSub" => out.push(MNode::Script { kind: ScriptKind::Sub, base: sub_arg(k, "m:e", d), sub: sub_arg(k, "m:sub", d), sup: Vec::new() }),
133        "m:sSubSup" => {
134            out.push(MNode::Script { kind: ScriptKind::SubSup, base: sub_arg(k, "m:e", d), sub: sub_arg(k, "m:sub", d), sup: sub_arg(k, "m:sup", d) })
135        }
136        "m:sPre" => {
137            out.push(MNode::Script { kind: ScriptKind::Pre, base: sub_arg(k, "m:e", d), sub: sub_arg(k, "m:sub", d), sup: sub_arg(k, "m:sup", d) })
138        }
139        "m:rad" => {
140            let pr = k.child("m:radPr");
141            out.push(MNode::Rad { deg: sub_arg(k, "m:deg", d), deg_hide: prop(pr, "m:degHide", false), e: sub_arg(k, "m:e", d) });
142        }
143        "m:nary" => {
144            let pr = k.child("m:naryPr");
145            let chr = chr(pr, "m:chr").map(|c| c.unwrap_or('∫')).unwrap_or('∫');
146            let lim_loc = match val(pr, "m:limLoc") {
147                Some("undOvr") => Some(LimLoc::UndOvr),
148                Some("subSup") => Some(LimLoc::SubSup),
149                _ => None,
150            };
151            out.push(MNode::Nary {
152                chr,
153                lim_loc,
154                grow: prop(pr, "m:grow", false),
155                sub_hide: prop(pr, "m:subHide", false),
156                sup_hide: prop(pr, "m:supHide", false),
157                sub: sub_arg(k, "m:sub", d),
158                sup: sub_arg(k, "m:sup", d),
159                e: sub_arg(k, "m:e", d),
160            });
161        }
162        "m:d" => {
163            let pr = k.child("m:dPr");
164            let beg = chr(pr, "m:begChr").unwrap_or(Some('('));
165            let end = chr(pr, "m:endChr").unwrap_or(Some(')'));
166            let sep = chr(pr, "m:sepChr").unwrap_or(Some('|'));
167            let elems: Vec<Arg> = k.children("m:e").map(|e| arg(e, d)).collect();
168            out.push(MNode::Delim {
169                beg,
170                end,
171                sep,
172                grow: prop(pr, "m:grow", true),
173                shp_match: val(pr, "m:shp") == Some("match"),
174                elems: if elems.is_empty() { vec![Vec::new()] } else { elems },
175            });
176        }
177        "m:func" => out.push(MNode::Func { name: sub_arg(k, "m:fName", d), e: sub_arg(k, "m:e", d) }),
178        "m:limLow" => out.push(MNode::Lim { upper: false, e: sub_arg(k, "m:e", d), lim: sub_arg(k, "m:lim", d) }),
179        "m:limUpp" => out.push(MNode::Lim { upper: true, e: sub_arg(k, "m:e", d), lim: sub_arg(k, "m:lim", d) }),
180        "m:acc" => {
181            let c = chr(k.child("m:accPr"), "m:chr").map(|c| c.unwrap_or('\u{302}')).unwrap_or('\u{302}');
182            out.push(MNode::Acc { chr: c, e: sub_arg(k, "m:e", d) });
183        }
184        "m:bar" => out.push(MNode::Bar { top: val(k.child("m:barPr"), "m:pos") == Some("top"), e: sub_arg(k, "m:e", d) }),
185        "m:borderBox" => {
186            let pr = k.child("m:borderBoxPr");
187            out.push(MNode::BorderBox {
188                hide: [prop(pr, "m:hideTop", false), prop(pr, "m:hideBot", false), prop(pr, "m:hideLeft", false), prop(pr, "m:hideRight", false)],
189                strike: [
190                    prop(pr, "m:strikeH", false),
191                    prop(pr, "m:strikeV", false),
192                    prop(pr, "m:strikeBLTR", false),
193                    prop(pr, "m:strikeTLBR", false),
194                ],
195                e: sub_arg(k, "m:e", d),
196            });
197        }
198        "m:box" => out.push(MNode::Boxed { e: sub_arg(k, "m:e", d) }),
199        "m:groupChr" => {
200            let pr = k.child("m:groupChrPr");
201            let c = chr(pr, "m:chr").map(|c| c.unwrap_or('\u{23DF}')).unwrap_or('\u{23DF}');
202            out.push(MNode::GroupChr { chr: c, top: val(pr, "m:pos") == Some("top"), e: sub_arg(k, "m:e", d) });
203        }
204        "m:eqArr" => out.push(MNode::EqArr { rows: k.children("m:e").map(|e| arg(e, d)).collect() }),
205        "m:m" => {
206            let mut col_jc = Vec::new();
207            if let Some(mcs) = k.child("m:mPr").and_then(|p| p.child("m:mcs")) {
208                for mc in mcs.children("m:mc") {
209                    let pr = mc.child("m:mcPr");
210                    let jc = match val(pr, "m:mcJc") {
211                        Some("left") => ColJc::Left,
212                        Some("right") => ColJc::Right,
213                        _ => ColJc::Center,
214                    };
215                    let count = val(pr, "m:count").and_then(|c| c.trim().parse::<usize>().ok()).unwrap_or(1).clamp(1, 64);
216                    for _ in 0..count {
217                        if col_jc.len() < 64 {
218                            col_jc.push(jc);
219                        }
220                    }
221                }
222            }
223            let rows = k.children("m:mr").take(1000).map(|r| r.children("m:e").take(64).map(|e| arg(e, d)).collect()).collect();
224            out.push(MNode::Matrix { rows, col_jc });
225        }
226        "m:phant" => {
227            let pr = k.child("m:phantPr");
228            out.push(MNode::Phant {
229                show: prop(pr, "m:show", true),
230                zero_wid: prop(pr, "m:zeroWid", false),
231                zero_asc: prop(pr, "m:zeroAsc", false),
232                zero_desc: prop(pr, "m:zeroDesc", false),
233                e: sub_arg(k, "m:e", d),
234            });
235        }
236        // Tracked insertions show; deletions don't. Hyperlinks, simple fields and direction
237        // wrappers (EG_PContentMath, §22.1.2.77) keep their runs (#1273).
238        "w:ins" | "w:moveTo" | "w:smartTag" | "w:customXml" | "w:hyperlink" | "w:fldSimple" | "w:dir" | "w:bdo" | "m:oMath" if depth <= MAX_DEPTH => {
239            for c in k.els() {
240                node(c, d, out);
241            }
242        }
243        // Markup compatibility: the math choice (we read math), else the fallback.
244        "mc:AlternateContent" if depth <= MAX_DEPTH => {
245            if let Some(c) = k.child("mc:Choice").or_else(|| k.child("mc:Fallback")) {
246                for c in c.els() {
247                    node(c, d, out);
248                }
249            }
250        }
251        "w:sdt" => {
252            if let Some(c) = k.child("w:sdtContent") {
253                for c in c.els() {
254                    node(c, d, out);
255                }
256            }
257        }
258        _ => {}
259    }
260}

262/// `m:t` (and `w:t`) text of a run, with symbol characters (`w:sym`) and special hyphens.
263/// Text in a symbol font (Symbol, Wingdings…) becomes its Unicode equivalent: the math font has
264/// no glyphs at those private-use codes (#1273).
265fn run_text(r: &El) -> String {
266    let font = r.child("w:rPr").and_then(|p| p.child("w:rFonts")).and_then(|f| f.attr("w:ascii").or_else(|| f.attr("w:hAnsi"))).unwrap_or("");
267    let mut s = String::new();
268    for c in r.els() {
269        match c.name.as_str() {
270            "m:t" | "w:t" => s.push_str(&symbol_fonts::map_text(font, &c.text())),
271            "w:tab" => s.push('\t'),
272            "w:noBreakHyphen" => s.push('\u{2011}'),
273            "w:sym" => {
274                let ch = c.attr("w:char").and_then(|h| u32::from_str_radix(h.trim(), 16).ok()).and_then(char::from_u32).filter(|c| *c as u32 >= 0x20);
275                if let Some(ch) = ch {
276                    let f = c.attr("w:font").unwrap_or(font);
277                    s.push(symbol_fonts::symbol_char(f, ch).unwrap_or(ch));
278                }
279            }
280            _ => {}
281        }
282    }
283    s
284}

286fn run(r: &El) -> Option<MRun> {
287    let text = run_text(r);
288    if text.is_empty() {
289        return None;
290    }
291    let mut out = MRun { text, ..Default::default() };
292    if let Some(pr) = r.child("m:rPr") {
293        out.sty = match pr.child("m:sty").and_then(|s| s.attr("m:val")) {
294            Some("p") => Some(MSty::Plain),
295            Some("b") => Some(MSty::Bold),
296            Some("i") => Some(MSty::Italic),
297            Some("bi") => Some(MSty::BoldItalic),
298            _ => None,
299        };
300        out.scr = match pr.child("m:scr").and_then(|s| s.attr("m:val")) {
301            Some("script") => MScr::Script,
302            Some("fraktur") => MScr::Fraktur,
303            Some("double-struck") => MScr::DoubleStruck,
304            Some("sans-serif") => MScr::SansSerif,
305            Some("monospace") => MScr::Monospace,
306            _ => MScr::Roman,
307        };
308        out.nor = pr.child("m:nor").is_some_and(on);
309        out.lit = pr.child("m:lit").is_some_and(on);
310        if let Some(b) = pr.child("m:brk") {
311            let aln = b.attr("m:alnAt").and_then(|v| v.trim().parse::<u32>().ok()).unwrap_or(0);
312            out.brk = Some(aln.min(255) as u8);
313        }
314    }
315    wrpr(r.child("w:rPr"), &mut out);
316    Some(out)
317}

319/// The run formatting a math run takes from `w:rPr`: size, colour, normal-text font and weight.
320fn wrpr(pr: Option<&El>, out: &mut MRun) {
321    let Some(pr) = pr else { return };
322    if let Some(sz) = pr.child("w:sz").and_then(|s| s.attr("w:val")).and_then(|v| v.trim().parse::<f32>().ok()) {
323        let pt = sz / 2.0;
324        if pt.is_finite() && (1.0..=1638.0).contains(&pt) {
325            out.size = Some(pt);
326        }
327    }
328    if let Some(c) = pr.child("w:color").and_then(|c| c.attr("w:val")).and_then(Rgb::parse) {
329        out.color = Some(c);
330    }
331    if out.nor {
332        // Symbol-font text was mapped to Unicode: it takes the text font.
333        if let Some(f) =
334            pr.child("w:rFonts").and_then(|f| f.attr("w:ascii").or_else(|| f.attr("w:hAnsi"))).filter(|f| !symbol_fonts::is_symbol_font(f))
335        {
336            out.font = Some(f.to_string());
337        }
338        let bold = pr.child("w:b").is_some_and(crate::docx::units::on_off);
339        let italic = pr.child("w:i").is_some_and(crate::docx::units::on_off);
340        if bold || italic {
341            out.sty = Some(match (bold, italic) {
342                (true, true) => MSty::BoldItalic,
343                (true, false) => MSty::Bold,
344                _ => MSty::Italic,
345            });
346        }
347    }
348}
