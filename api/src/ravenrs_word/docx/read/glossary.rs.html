1//! The glossary document (ECMA-376 Part 1 §17.12): the document's own building blocks (Quick
2//! Parts, AutoText…), read into `Document::glossary`.
3//!
4//! The glossary part and everything it reaches are also kept verbatim (`keep_unmodelled`), with a
5//! fingerprint of the entries as read ([`GLOSSARY_READ`]): a save writes the kept part while the
6//! entries are unchanged, so whatever the model doesn't hold survives, and writes the glossary from
7//! the model once they change.
8
9use std::sync::Arc;
10
11use crate::doc::{Blocks, BuildingBlock};
12
13use super::Reader;
14use super::story::StoryCtx;
15use crate::docx::package::{
16    ContentTypes, GLOSSARY_BEHAVIORS, GLOSSARY_CONTENT_TYPE, GLOSSARY_READ, GLOSSARY_TYPES, MAX_GLOSSARY_ENTRIES, Rels, glossary_fingerprint, rel_is,
17    rt,
18};
19use crate::docx::xml::El;
20
21/// Longest name, gallery, category, description, guid or behaviour value kept (characters).
22const MAX_FIELD: usize = 1024;
23
24/// The known values of `parent`'s `tag` children, each once, in file order.
25fn listed(parent: Option<&El>, tag: &str, known: &[&str]) -> Vec<String> {
26    let mut out: Vec<String> = Vec::new();
27    for v in parent.into_iter().flat_map(|p| p.children(tag)).filter_map(|e| e.attr("w:val")) {
28        if known.contains(&v) && !out.iter().any(|o| o == v) {
29            out.push(v.to_string());
30        }
31    }
32    out
33}
34
35impl Reader<'_> {
36    /// Read the glossary the main document relates to (`document`: its relationships). The first
37    /// internal glossary relationship counts; one pointing at the main document, a part that isn't a
38    /// glossary, or a missing one gives no entries. Entries without a name or a body are skipped.
39    pub(super) fn read_glossary(&mut self, document: &Rels) {
40        // The first internal one, as `keep_unmodelled` keeps.
41        let Some(rel) = document.list.iter().find(|r| !r.external && rel_is(&r.kind, rt::GLOSSARY)) else { return };
42        let mut entries = Vec::new();
43        if !rel.target.eq_ignore_ascii_case(&self.main) {
44            let pkg = self.pkg;
45            let ct = self.content_types.get_or_insert_with(|| ContentTypes::read(pkg)).of(&rel.target).map(str::to_string);
46            let root = match self.xml(&rel.target) {
47                Ok(r) => r,
48                Err(e) => {
49                    log::warn!("docx: ignoring unreadable glossary {}: {e}", rel.target);
50                    None
51                }
52            };
53            // Without content types (a converted package) the root element alone decides.
54            if let Some(root) = root.filter(|r| r.name == "w:glossaryDocument" && ct.as_deref().is_none_or(|t| t == GLOSSARY_CONTENT_TYPE)) {
55                let rels = self.pkg.rels(&rel.target);
56                // Pictures in a building block belong to the glossary part, not the main story.
57                let outer = std::mem::replace(&mut self.story, rel.target.as_str().into());
58                entries = self.doc_parts(&root, &rels);
59                self.story = outer;
60            }
61        }
62        self.doc.glossary = entries;
63        let print = glossary_fingerprint(&self.doc.glossary);
64        self.doc.passthrough.insert(GLOSSARY_READ.into(), Arc::new(print.into_bytes()));
65    }
66
67    /// The building blocks of a `w:glossaryDocument`, read with the glossary's relationships.
68    fn doc_parts(&mut self, root: &El, rels: &Rels) -> Vec<BuildingBlock> {
69        // The glossary's notes, comments, bookmarks, permissions and moves are its own: what the
70        // main document's ids mean doesn't apply to it, and its ids mustn't leak into the body's.
71        let footnotes = std::mem::take(&mut self.footnotes);
72        let endnotes = std::mem::take(&mut self.endnotes);
73        let comments = std::mem::take(&mut self.comment_map);
74        let ended = std::mem::take(&mut self.comments_ended);
75        let bookmarks = std::mem::take(&mut self.bookmarks);
76        let perms = std::mem::take(&mut self.perms);
77        let moves = std::mem::take(&mut self.moves);
78        let mut out = Vec::new();
79        for part in root.child("w:docParts").into_iter().flat_map(|d| d.children("w:docPart")) {
80            if out.len() >= MAX_GLOSSARY_ENTRIES {
81                log::warn!("docx: glossary has more than {MAX_GLOSSARY_ENTRIES} entries; the rest are dropped");
82                break;
83            }
84            let pr = part.child("w:docPartPr");
85            let val = |e: Option<&El>| e.and_then(|e| e.attr("w:val")).unwrap_or("").chars().take(MAX_FIELD).collect::<String>();
86            let name = val(pr.and_then(|p| p.child("w:name")));
87            let Some(body) = part.child("w:docPartBody").filter(|_| !name.is_empty()) else { continue };
88            let category = pr.and_then(|p| p.child("w:category"));
89            let mut sc = StoryCtx::default();
90            let mut blocks = Blocks::new();
91            self.read_blocks(&mut sc, body, rels, &mut blocks, 0);
92            self.flush_pending(&mut sc, &mut blocks);
93            if blocks.is_empty() {
94                continue;
95            }
96            out.push(BuildingBlock {
97                name,
98                gallery: val(category.and_then(|c| c.child("w:gallery"))),
99                category: val(category.and_then(|c| c.child("w:name"))),
100                description: val(pr.and_then(|p| p.child("w:description"))),
101                guid: val(pr.and_then(|p| p.child("w:guid"))),
102                style: val(pr.and_then(|p| p.child("w:style"))),
103                types: listed(pr.and_then(|p| p.child("w:types")), "w:type", &GLOSSARY_TYPES),
104                behaviors: listed(pr.and_then(|p| p.child("w:behaviors")), "w:behavior", &GLOSSARY_BEHAVIORS),
105                blocks,
106            });
107        }
108        self.footnotes = footnotes;
109        self.endnotes = endnotes;
110        self.comment_map = comments;
111        self.comments_ended = ended;
112        self.bookmarks = bookmarks;
113        self.perms = perms;
114        self.moves = moves;
115        out
116    }
117}
