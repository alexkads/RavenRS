19/// The shape kind drawn for DrawingML preset geometry `prst` (ECMA-376 Part 1 §20.1.10.56).
20/// Presets WordCraft has no shape for are drawn as the nearest one it has: callouts as the box,
21/// round or cloud they point from, flowchart boxes as rectangles.
22pub(crate) fn preset_kind(prst: &str) -> ShapeKind {
23    match prst {
24        "roundRect"
25        | "round1Rect"
26        | "round2SameRect"
27        | "round2DiagRect"
28        | "snipRoundRect"
29        | "wedgeRoundRectCallout"
30        | "flowChartAlternateProcess"
31        | "flowChartTerminator"
32        | "plaque" => ShapeKind::RoundedRectangle,
33        "ellipse" | "wedgeEllipseCallout" | "cloudCallout" | "cloud" | "flowChartConnector" | "donut" | "flowChartOnlineStorage" => {
34            ShapeKind::Ellipse
35        }
36        "flowChartOr" => ShapeKind::FlowchartOr,
37        "triangle" | "flowChartExtract" | "flowChartMerge" => ShapeKind::Triangle,
38        "rtTriangle" => ShapeKind::RightTriangle,
39        "diamond" | "flowChartDecision" => ShapeKind::Diamond,
40        "line" | "lineInv" => ShapeKind::Line,
41        "straightConnector1" => ShapeKind::StraightConnector,
42        "bentConnector2" | "bentConnector3" | "bentConnector4" | "bentConnector5" => ShapeKind::ElbowConnector,
43        "curvedConnector2" | "curvedConnector3" | "curvedConnector4" | "curvedConnector5" => ShapeKind::CurvedConnector,
44        "rightArrow" | "leftRightArrow" | "upDownArrow" | "notchedRightArrow" | "stripedRightArrow" => ShapeKind::Arrow,
45        "leftArrow" => ShapeKind::ArrowLeft,
46        "upArrow" => ShapeKind::ArrowUp,
47        "downArrow" => ShapeKind::ArrowDown,
48        "star4" | "star5" | "star6" | "star7" | "star8" | "star10" | "star12" | "star16" | "star24" | "star32" => ShapeKind::Star,
49        "heart" => ShapeKind::Heart,
50        _ => ShapeKind::Rectangle,
51    }
52}

124/// The theme's fill styles (`a:fmtScheme`): `a:fillStyleLst` and `a:bgFillStyleLst`, which a
125/// shape's `a:fillRef` 1–999 and 1001+ point to.
126#[derive(Debug, Default)]
127pub(crate) struct ThemeFills {
128    pub fill: Vec<El>,
129    pub bg: Vec<El>,
130}

132impl ThemeFills {
133    pub fn read(fmt: Option<&El>) -> ThemeFills {
134        let list = |n: &str| fmt.and_then(|f| f.child(n)).map(|l| l.els().take(MAX_THEME_FILLS).cloned().collect()).unwrap_or_default();
135        ThemeFills { fill: list("a:fillStyleLst"), bg: list("a:bgFillStyleLst") }
136    }
137    /// Style `idx` (1-based; 1001+ the background styles; 1000 names none).
138    fn get(&self, idx: i64) -> Option<&El> {
139        let (list, i) = if idx > 1000 { (&self.bg, idx - 1001) } else { (&self.fill, idx - 1) };
140        list.get(usize::try_from(i).ok()?)
141    }
142}

144/// Most fill styles kept per theme list: every one a `a:fillRef` can reach (1–999 into
145/// `a:fillStyleLst`, 1001–1999 into `a:bgFillStyleLst`, ECMA-376 Part 1 §20.1.4.2.10), and no
146/// more, so a hostile theme with millions of entries keeps a bounded list.
147const MAX_THEME_FILLS: usize = 999;

149/// The fill a `…spPr` element states itself: `Some(None)` for none (`a:noFill`, picture fills),
150/// `group` for `a:grpFill`, `None` when it leaves the fill to the shape's style.
151pub(crate) fn own_fill(sppr: Option<&El>, theme: &[Rgb], group: Option<&Fill>) -> Option<Option<Fill>> {
152    let f = sppr?.els().find(|c| matches!(c.name.as_str(), "a:noFill" | "a:blipFill" | "a:grpFill" | "a:solidFill" | "a:gradFill" | "a:pattFill"))?;
153    Some(match f.name.as_str() {
154        "a:grpFill" => group.cloned(),
155        _ => read_fill(f, theme, None),
156    })
157}

159/// The paint of a shape from its properties `sppr`, falling back on its style `style` (theme
160/// colours `theme`, fill styles `styles`) for what they leave out: an explicit `a:noFill` is no
161/// fill or outline, an explicit colour wins, `a:grpFill` is the enclosing group's fill `group`,
162/// and a style reference with index 0 is none.
163pub(crate) fn shape_paint(sppr: Option<&El>, style: Option<&El>, theme: &[Rgb], styles: &ThemeFills, group: Option<&Fill>) -> ShapePaint {
164    let style_ref = |name: &str| -> Option<(i64, &El)> {
165        let r = style?.child(name)?;
166        Some((r.attr("idx").and_then(int).unwrap_or(0), r))
167    };
168    let fill = own_fill(sppr, theme, group).unwrap_or_else(|| {
169        let (idx, r) = style_ref("a:fillRef").filter(|(idx, _)| *idx > 0)?;
170        // The theme style in the reference's colour (`phClr`), else that colour: also for a style
171        // the model can't hold (a picture fill), so the shape doesn't vanish.
172        let ph = color_alpha(r, theme, None)?;
173        let solid = Fill::Solid { color: ph.0, transparency: ph.1 };
174        match styles.get(idx) {
175            Some(st) if st.name == "a:noFill" => None,
176            Some(st) => read_fill(st, theme, Some(ph)).or(Some(solid)),
177            None => Some(solid),
178        }
179    });
180    let ln = sppr.and_then(|s| s.child("a:ln"));
181    let line_ref = style_ref("a:lnRef").filter(|(idx, _)| *idx > 0);
182    let outline_fill = ln.and_then(|l| l.child("a:gradFill")).and_then(|g| read_fill(g, theme, None)).filter(|f| matches!(f, Fill::Gradient(_)));
183    let stroke = match ln {
184        Some(l) if l.child("a:noFill").is_some() => None,
185        _ if outline_fill.is_some() => outline_fill.as_ref().map(Fill::main_color),
186        Some(l) if l.els().any(|c| matches!(c.name.as_str(), "a:solidFill" | "a:gradFill" | "a:pattFill")) => fill_colour(l, theme),
187        _ => line_ref.and_then(|(_, r)| color(r, theme)),
188    };
189    let themed_width = line_ref.and_then(|(idx, _)| THEME_LINE_WIDTHS.get(usize::try_from(idx.clamp(1, 3) - 1).unwrap_or(0)).copied());
190    let stroke_width = ln
191        .and_then(|l| l.attr("w"))
192        .and_then(|v| measure(v, 12_700.0))
193        .or(themed_width)
194        // Word's default outline when the file names none (no outline: no width).
195        .unwrap_or(if stroke.is_some() { 0.75 } else { 0.0 })
196        .clamp(0.0, 100.0);
197    ShapePaint { fill, stroke, stroke_width, outline_fill }
198}

200/// A VML shape's fill: `fillcolor` (white by default) and its `v:fill`: opacity, and a linear
201/// (`type="gradient"`) or circular (`gradientRadial`) gradient to `color2`, or through `colors`.
202/// A gradient's `focus` (−100%–100%) and `angle` are approximated: 0 and 100% run `color` to
203/// `color2`, −100% the other way, and a focus between mirrors the gradient there (`color2`
204/// there for a positive focus, `color` for a negative one). `None` when not filled.
205pub(crate) fn vml_fill(shape: &El, shapetype: Option<&El>) -> Option<Fill> {
206    // What the shape leaves out comes from its shapetype (#1377).
207    let own = |n: &str| shape.attr(n).or_else(|| shapetype.and_then(|t| t.attr(n)));
208    let f = shape.child("v:fill").or_else(|| shapetype.and_then(|t| t.child("v:fill")));
209    let attr = |n: &str| f.and_then(|f| f.attr(n));
210    let off = |v: Option<&str>| {
211        v.is_some_and(|v| {
212            let t = v.trim();
213            t.eq_ignore_ascii_case("f") || t.eq_ignore_ascii_case("false") || t == "0"
214        })
215    };
216    if off(own("filled")) || off(attr("on")) {
217        return None;
218    }
219    let c1 = attr("color").or(own("fillcolor")).map_or(Some(Rgb::WHITE), vml_color)?;
220    let t1 = 100.0 - attr("opacity").and_then(vml_fraction).unwrap_or(1.0) * 100.0;
221    let radial = match attr("type") {
222        Some("gradient") => false,
223        Some("gradientRadial") => true,
224        _ => return Fill::Solid { color: c1, transparency: t1 }.sanitized(),
225    };
226    let c2 = attr("color2").and_then(vml_color).unwrap_or(Rgb::WHITE);
227    let t2 = 100.0 - attr("o:opacity2").and_then(vml_fraction).unwrap_or(1.0) * 100.0;
228    let stop = |pos: f32, (color, transparency): (Rgb, f32)| GradStop { pos, color, transparency };
229    let (a, b) = ((c1, t1), (c2, t2));
230    let focus = attr("focus").and_then(vml_fraction).unwrap_or(0.0).clamp(-1.0, 1.0) * 100.0;
231    let colors: Vec<GradStop> = attr("colors")
232        .into_iter()
233        .flat_map(|v| v.split(';'))
234        .take(MAX_STOPS)
235        .filter_map(|s| {
236            let (pos, c) = s.trim().split_once(' ')?;
237            Some(stop(vml_fraction(pos)? * 100.0, (vml_color(c)?, t1)))
238        })
239        .collect();
240    let stops = if colors.len() >= 2 {
241        colors
242    } else if focus <= -100.0 {
243        vec![stop(0.0, b), stop(100.0, a)]
244    } else if focus >= 100.0 || focus == 0.0 {
245        vec![stop(0.0, a), stop(100.0, b)]
246    } else if focus > 0.0 {
247        vec![stop(0.0, a), stop(focus, b), stop(100.0, a)]
248    } else {
249        vec![stop(0.0, b), stop(-focus, a), stop(100.0, b)]
250    };
251    let kind = if radial {
252        // `focusposition` "x,y" (fractions of the shape) is the centre.
253        let fp = attr("focusposition").unwrap_or("0.5,0.5");
254        let (x, y) = fp.split_once(',').unwrap_or((fp, fp));
255        let (x, y) = (vml_fraction(x).unwrap_or(0.5) * 100.0, vml_fraction(y).unwrap_or(0.5) * 100.0);
256        GradientKind::Path { shape: PathShape::Circle, focus: [x, y, 100.0 - x, 100.0 - y] }
257    } else {
258        // VML's angle is counter-clockwise from a top-to-bottom gradient.
259        let ang = attr("angle").and_then(|v| v.trim().parse::<f32>().ok()).filter(|v| v.is_finite()).unwrap_or(0.0);
260        GradientKind::Linear { angle: 90.0 - ang, scaled: false }
261    };
262    Fill::Gradient(Arc::new(Gradient { stops, kind, tile: None, rot_with_shape: true, flip: GradientFlip::None })).sanitized()
263}

265/// A VML fraction (`opacity`, `focus`, a `colors` position): `0.5`, `50%`, or 65536ths
266/// (`32768f`). Not clamped: a focus runs from -1 to 1.
267pub(crate) fn vml_fraction(v: &str) -> Option<f32> {
268    let v = v.trim();
269    let n = if let Some(f) = v.strip_suffix('f') {
270        f.trim().parse::<f32>().ok()? / 65_536.0
271    } else if let Some(p) = v.strip_suffix('%') {
272        p.trim().parse::<f32>().ok()? / 100.0
273    } else {
274        v.parse::<f32>().ok()?
275    };
276    n.is_finite().then_some(n)
277}

279/// Explicit DrawingML dash stops override the solid default.
280pub(crate) fn line_dash(ln: Option<&El>) -> crate::doc::line_dash::LineDash {
281    use crate::doc::line_dash::{LineDash, MAX_DASH_STOPS};
282    let Some(ln) = ln else { return LineDash::Solid };
283    if let Some(custom) = ln.child("a:custDash") {
284        let stops: Vec<_> =
285            custom.children("a:ds").take(MAX_DASH_STOPS).filter_map(|e| Some((dash_ratio(e.attr("d")?)?, dash_ratio(e.attr("sp")?)?))).collect();
286        return if stops.is_empty() { LineDash::Solid } else { LineDash::Custom(stops) };
287    }
288    ln.child("a:prstDash").and_then(|e| e.attr("val")).map(LineDash::from_ooxml).unwrap_or_default()
289}

291/// Strict OOXML may spell percentages with a percent sign instead of integer units.
292fn dash_ratio(value: &str) -> Option<u32> {
293    const PERCENT_UNITS: f64 = 1000.0;
294    let value = value.trim();
295    let Some(percent) = value.strip_suffix('%') else { return value.parse().ok() };
296    let units = percent.parse::<f64>().ok()? * PERCENT_UNITS;
297    if !units.is_finite() || !(0.0..=f64::from(u32::MAX)).contains(&units) {
298        return None;
299    }
300    Some(units.round() as u32)
301}

303/// A VML colour (`strokecolor`, `fillcolor`): `#rrggbb`, `rrggbb`, `#rgb` or `rgb` (Word follows it
304/// with the theme slot, as in `#4472c4 [3204]` or `4472c4 [3204]`), an `rgb(...)` function, or a
305/// colour name.
306pub(crate) fn vml_color(v: &str) -> Option<Rgb> {
307    let s = v.trim();
308    if let Some(inner) = s.strip_prefix("rgb(").or_else(|| s.strip_prefix("rgba(")) {
309        let nums: Vec<u8> = inner
310            .trim_end_matches(')')
311            .split(',')
312            .take(3)
313            .filter_map(|x| {
314                let x = x.trim();
315                if let Some(p) = x.strip_suffix('%') {
316                    p.trim().parse::<f32>().ok().map(|f| (f.clamp(0.0, 100.0) * 2.55).round() as u8)
317                } else {
318                    x.parse::<f32>().ok().map(|f| f.clamp(0.0, 255.0).round() as u8)
319                }
320            })
321            .collect();
322        if let [r, g, b] = nums[..] {
323            return Some(Rgb(r, g, b));
324        }
325    }
326    let tok = s.split_whitespace().next()?;
327    let hex = tok.strip_prefix('#').unwrap_or(tok);
328    if hex.len() == 3 && hex.chars().all(|c| c.is_ascii_hexdigit()) {
329        let long: String = hex.chars().flat_map(|c| [c, c]).collect();
330        return Rgb::parse(&long);
331    }
332    if hex.len() == 6 && hex.chars().all(|c| c.is_ascii_hexdigit()) {
333        return Rgb::parse(hex);
334    }
335    super::drawing_color::preset_color(tok)
336}

502    #[test]
503    fn vml_gradients_focus_and_colours() {
504        let v = |attrs: &str, fill: &str| {
505            let ns = r#"xmlns:v="urn:schemas-microsoft-com:vml" xmlns:o="urn:schemas-microsoft-com:office:office""#;
506            vml_fill(&crate::docx::xml::parse(format!(r#"<v:rect {ns} {attrs}>{fill}</v:rect>"#).as_bytes()).unwrap(), None)
507        };
508        let stops = |f: Option<Fill>| match f {
509            Some(Fill::Gradient(g)) => g.stops.iter().map(|s| (s.pos, s.color)).collect::<Vec<_>>(),
510            o => panic!("{o:?}"),
511        };
512        let (red, blue) = (Rgb(255, 0, 0), Rgb(0, 0, 255));
513        // A focus of 50% mirrors the gradient in the middle; 100% runs as 0%, −100% reverses it.
514        assert_eq!(
515            stops(v(r##"fillcolor="red""##, r##"<v:fill type="gradient" color2="blue" focus="50%"/>"##)),
516            [(0.0, red), (50.0, blue), (100.0, red)]
517        );
518        assert_eq!(stops(v(r##"fillcolor="red""##, r##"<v:fill type="gradient" color2="blue" focus="100%"/>"##)), [(0.0, red), (100.0, blue)]);
519        assert_eq!(stops(v(r##"fillcolor="red""##, r##"<v:fill type="gradient" color2="blue" focus="-100%"/>"##)), [(0.0, blue), (100.0, red)]);
520        assert_eq!(v("", r##"<v:fill opacity="-2"/>"##), Some(Fill::Solid { color: Rgb::WHITE, transparency: 100.0 }));
521        // `colors` lists the stops (fractions or 65536ths).
522        assert_eq!(stops(v("", r##"<v:fill type="gradient" colors="0 red;32768f blue;1 red"/>"##)), [(0.0, red), (50.0, blue), (100.0, red)]);
523        let Some(Fill::Gradient(g)) = v(r##"fillcolor="red""##, r##"<v:fill type="gradientRadial" color2="blue" focusposition=".25,.75"/>"##) else {
524            panic!()
525        };
526        assert_eq!(g.kind, GradientKind::Path { shape: PathShape::Circle, focus: [25.0, 75.0, 75.0, 25.0] });
527        assert_eq!(v(r##"filled="f""##, ""), None);
528        assert_eq!(v("", r##"<v:fill opacity="32768f"/>"##), Some(Fill::Solid { color: Rgb::WHITE, transparency: 50.0 }));
529        // Hostile values don't panic.
530        assert!(v("", r##"<v:fill type="gradient" colors="x;;1 #zz;NaN red" focus="NaN" angle="inf" opacity="1e99f"/>"##).is_some());
531    }
