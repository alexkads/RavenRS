15/// Deepest `dsp:grpSp` nesting followed.
16const MAX_GROUP_DEPTH: usize = 8;
17/// Most shapes read from one drawing.
18const MAX_SHAPES: usize = 10_000;
19/// Most items emitted for one diagram.
20const MAX_ITEMS: usize = 50_000;
21/// EMU per point.
22const EMU_PT: f64 = 12_700.0;
23/// Largest coordinate or extent accepted, EMU (about 78,000 pt).
24const MAX_EMU: f64 = 1.0e9;
25/// Line height as a multiple of the font size.
26const LINE: f32 = 1.2;
27
28/// An affine map per axis (`x' = kx·x + ox`, `y' = ky·y + oy`).
29#[derive(Clone, Copy, Debug)]
30struct Xf {
31    kx: f64,
32    ox: f64,
33    ky: f64,
34    oy: f64,
35}
36
37impl Xf {
38    const ID: Xf = Xf { kx: 1.0, ox: 0.0, ky: 1.0, oy: 0.0 };
39
40    /// `self` applied after `inner`.
41    fn then(self, inner: Xf) -> Xf {
42        Xf { kx: self.kx * inner.kx, ox: self.kx * inner.ox + self.ox, ky: self.ky * inner.ky, oy: self.ky * inner.oy + self.oy }
43    }
44
45    /// The box `[x, y, w, h]` mapped to corners `[x0, y0, x1, y1]`.
46    fn rect(self, b: [f64; 4]) -> [f64; 4] {
47        let (x0, x1) = (self.kx * b[0] + self.ox, self.kx * (b[0] + b[2]) + self.ox);
48        let (y0, y1) = (self.ky * b[1] + self.oy, self.ky * (b[1] + b[3]) + self.oy);
49        [x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)]
50    }
51}

61/// Points, clamped into the range layout accepts (NaN, from a degenerate transform, is 0).
62fn pt(v: f64) -> f32 {
63    if v.is_nan() { 0.0 } else { v.clamp(-MAX_LEN_PT as f64, MAX_LEN_PT as f64) as f32 }
64}

66/// An EMU attribute of `e`, finite and bounded.
67fn emu(e: &El, attr: &str) -> Option<f64> {
68    int(e.attr(attr)?).map(|v| (v as f64).clamp(-MAX_EMU, MAX_EMU))
69}

71/// `a:off` / `a:ext` of an element (a shape's `a:xfrm`, `dsp:txXfrm`) as `[x, y, w, h]` in EMU.
72fn xfrm_box(x: &El) -> Option<[f64; 4]> {
73    let (off, ext) = (x.child("a:off")?, x.child("a:ext")?);
74    Some([emu(off, "x")?, emu(off, "y")?, emu(ext, "cx")?.max(0.0), emu(ext, "cy")?.max(0.0)])
75}

77/// Child-space to parent-space scale of a group, from its `a:xfrm` (`off`/`ext` against `chOff`/`chExt`).
78fn group_xf(grp: &El) -> Xf {
79    let Some(x) = grp.child("dsp:grpSpPr").and_then(|p| p.child("a:xfrm")) else { return Xf::ID };
80    let v = |tag: &str, a: &str| x.child(tag).and_then(|e| emu(e, a)).unwrap_or(0.0);
81    let scale = |e: f64, c: f64| if c > 0.0 && e >= 0.0 { e / c } else { 1.0 };
82    let kx = scale(v("a:ext", "cx"), v("a:chExt", "cx"));
83    let ky = scale(v("a:ext", "cy"), v("a:chExt", "cy"));
84    Xf { kx, ox: v("a:off", "x") - v("a:chOff", "x") * kx, ky, oy: v("a:off", "y") - v("a:chOff", "y") * ky }
85}

87/// Shapes in `tree` (depth-first through groups), each with its transform into the drawing.
88fn collect<'a>(tree: &'a El, xf: Xf, depth: usize, out: &mut Vec<(Xf, &'a El)>) {
89    for e in tree.els() {
90        match e.name.as_str() {
91            "dsp:sp" if out.len() < MAX_SHAPES => out.push((xf, e)),
92            "dsp:grpSp" if depth < MAX_GROUP_DEPTH => collect(e, xf.then(group_xf(e)), depth + 1, out),
93            _ => {}
94        }
95    }
96}

98/// The items of a SmartArt drawing, scaled so its extent fills the `w` × `h` (points) frame.
99/// `image` maps a relationship id of the drawing part to a `Document::media` key.
100pub(crate) fn diagram_items(drawing: &El, theme: &[Rgb], w: f32, h: f32, image: &mut dyn FnMut(&str) -> Option<String>) -> Vec<GraphicItem> {
101    let Some(tree) = drawing.find("dsp:spTree") else { return Vec::new() };
102    let mut shapes = Vec::new();
103    collect(tree, Xf::ID, 0, &mut shapes);
104    let placed: Vec<(Xf, &El, [f64; 4])> = shapes
105        .into_iter()
106        .filter_map(|(xf, sp)| {
107            let b = sp.child("dsp:spPr").and_then(|p| p.child("a:xfrm")).and_then(xfrm_box)?;
108            Some((xf, sp, b))
109        })
110        .collect();
111    // The drawing's extent (from its origin) against the frame. One scale for both axes, never
112    // above 1:1: Word draws the drawing at its own size when it fits the frame.
113    let (ext_x, ext_y) = placed.iter().fold((0.0f64, 0.0f64), |(mx, my), (xf, _, b)| {
114        let r = xf.rect(*b);
115        (mx.max(r[2]), my.max(r[3]))
116    });
117    let frame = |f: f32| if f.is_finite() && f > 0.0 { f as f64 * EMU_PT } else { 0.0 };
118    let fit_of = |frame: f64, ext: f64| if frame > 0.0 && ext > 0.0 { frame / ext } else { 1.0 };
119    let fit = [1.0, fit_of(frame(w), ext_x), fit_of(frame(h), ext_y)].into_iter().fold(f64::INFINITY, f64::min);
120    let root = Xf { kx: fit / EMU_PT, ox: 0.0, ky: fit / EMU_PT, oy: 0.0 };
121    let mut out = Vec::new();
122    for (xf, sp, b) in placed {
123        if out.len() >= MAX_ITEMS {
124            break;
125        }
126        shape(sp, root.then(xf), b, theme, fit as f32, image, &mut out);
127    }
128    out
129}

131/// One `dsp:sp`: its picture or outline, then its text.
132/// `fit` is the drawing's scale into the frame, which text sizes follow.
133fn shape(sp: &El, xf: Xf, b: [f64; 4], theme: &[Rgb], fit: f32, image: &mut dyn FnMut(&str) -> Option<String>, out: &mut Vec<GraphicItem>) {
134    let [x0, y0, x1, y1] = xf.rect(b);
135    if ![x0, y0, x1, y1].iter().all(|v| v.is_finite()) {
136        return;
137    }
138    let (w, h) = (x1 - x0, y1 - y0);
139    let sppr = sp.child("dsp:spPr");
140    let xfrm = sppr.and_then(|p| p.child("a:xfrm"));
141    let attr = |n: &str| xfrm.and_then(|x| x.attr(n));
142    let rot = attr("rot").and_then(int).unwrap_or(0).rem_euclid(21_600_000) as f64 / 60_000.0;
143    let flip_h = attr("flipH").is_some_and(on);
144    let flip_v = attr("flipV").is_some_and(on);
145    let turned = rot != 0.0 || flip_h || flip_v;
146    let rect = [pt(x0), pt(y0), pt(w), pt(h)];
147    let filled;
148    if let Some(blip) = sppr.and_then(|p| p.child("a:blipFill")).and_then(|f| f.find("a:blip")) {
149        // A picture fill stretches into the shape's box; no clipping to the outline.
150        if let Some(media) = blip.attr("r:embed").and_then(image) {
151            out.push(GraphicItem::Image { rect, media });
152        }
153        filled = false;
154    } else {
155        let fill = fill_of(sp, sppr, theme);
156        let (stroke, stroke_width) = line_of(sp, sppr, theme);
157        let prst = sppr.and_then(|p| p.child("a:prstGeom")).and_then(|g| g.attr("prst")).unwrap_or("rect");
158        let geom = sppr.and_then(|p| p.child("a:prstGeom"));
159        let geo = match kind_of(prst) {
160            Some(k) if !turned || prst == "ellipse" => Geo::Kind(k),
161            _ => match outline(prst, w, h, geom) {
162                Some((pts, closed)) => Geo::Poly(pts, closed),
163                None => Geo::Kind(kind_of(prst).unwrap_or(ShapeKind::Rectangle)),
164            },
165        };
166        match geo {
167            Geo::Kind(kind) => out.push(GraphicItem::Shape { rect, kind, fill, stroke, stroke_width }),
168            Geo::Poly(pts, closed) if !pts.is_empty() => {
169                // Flip in the box, then turn about its centre (clockwise, y down).
170                let (s, c) = rot.to_radians().sin_cos();
171                let (cx, cy) = (w / 2.0, h / 2.0);
172                let map = |(px, py): (f64, f64)| {
173                    let px = if flip_h { w - px } else { px };
174                    let py = if flip_v { h - py } else { py };
175                    let (dx, dy) = (px - cx, py - cy);
176                    (x0 + cx + dx * c - dy * s, y0 + cy + dx * s + dy * c)
177                };
178                let mut segs = Vec::with_capacity(pts.len() + 1);
179                for (i, p) in pts.into_iter().enumerate() {
180                    let (x, y) = map(p);
181                    segs.push(if i == 0 { PathSeg::Move(pt(x), pt(y)) } else { PathSeg::Line(pt(x), pt(y)) });
182                }
183                if closed {
184                    segs.push(PathSeg::Close);
185                }
186                let fill = if closed { fill } else { None };
187                out.push(GraphicItem::Path { segs, fill, stroke, stroke_width });
188            }
189            Geo::Poly(..) => {}
190        }
191        filled = fill.is_some();
192    }
193    text(sp, xf, b, theme, filled, fit, out);
194}

200/// The colour of a shape's style reference (`a:fillRef`, `a:lnRef`); index 0 means none.
201fn style_colour(sp: &El, which: &str, theme: &[Rgb]) -> Option<Rgb> {
202    let r = sp.child("dsp:style")?.child(which)?;
203    if r.attr("idx") == Some("0") {
204        return None;
205    }
206    color(r, theme)
207}

209/// A shape's fill: its own `a:*Fill`, else the style's fill colour.
210fn fill_of(sp: &El, sppr: Option<&El>, theme: &[Rgb]) -> Option<Rgb> {
211    let own = sppr.and_then(|p| p.els().find(|e| matches!(e.name.as_str(), "a:solidFill" | "a:noFill" | "a:gradFill" | "a:pattFill")));
212    match own {
213        Some(e) => paint(e, theme),
214        None => style_colour(sp, "a:fillRef", theme),
215    }
216}

218/// A shape's outline colour and width in points (`a:ln`, else the style's line at 0.75 pt).
219fn line_of(sp: &El, sppr: Option<&El>, theme: &[Rgb]) -> (Option<Rgb>, f32) {
220    let ln = sppr.and_then(|p| p.child("a:ln"));
221    let width = ln.and_then(|l| l.attr("w")).and_then(int).map_or(0.75, |v| (v as f64 / EMU_PT).clamp(0.0, 100.0) as f32);
222    let own = ln.and_then(|l| l.els().find(|e| matches!(e.name.as_str(), "a:solidFill" | "a:noFill" | "a:gradFill")));
223    match own {
224        Some(e) if e.name == "a:noFill" => (None, 0.0),
225        Some(e) => (paint(e, theme), width),
226        None => (style_colour(sp, "a:lnRef", theme), width),
227    }
228}

230/// The preset's own [`ShapeKind`], for the presets that have one. `None` means a polygon.
231fn kind_of(prst: &str) -> Option<ShapeKind> {
232    Some(match prst {
233        "roundRect" => ShapeKind::RoundedRectangle,
234        "ellipse" => ShapeKind::Ellipse,
235        "triangle" => ShapeKind::Triangle,
236        "diamond" => ShapeKind::Diamond,
237        // A line runs from its box's top-left to its bottom-right corner (a path; the Line shape
238        // kind draws the other diagonal).
239        "line" | "straightConnector1" => return None,
240        "rightArrow" => ShapeKind::Arrow,
241        "star5" | "star4" | "star6" => ShapeKind::Star,
242        "heart" => ShapeKind::Heart,
243        "leftArrow" | "upArrow" | "downArrow" | "leftRightArrow" | "upDownArrow" | "chevron" | "homePlate" | "hexagon" | "pentagon" | "trapezoid"
244        | "parallelogram" | "rtTriangle" | "pie" | "blockArc" => return None,
245        _ => ShapeKind::Rectangle,
246    })
247}

249/// A preset's adjust value (`a:avLst` `a:gd name`, `fmla="val N"`) in its raw units, `def` when absent.
250fn raw_adj(geom: Option<&El>, name: &str, def: f64) -> f64 {
251    geom.and_then(|g| g.child("a:avLst"))
252        .and_then(|l| l.children("a:gd").find(|d| d.attr("name") == Some(name)))
253        .and_then(|d| d.attr("fmla"))
254        .and_then(|f| f.strip_prefix("val "))
255        .and_then(|v| v.trim().parse::<f64>().ok())
256        .filter(|v| v.is_finite())
257        .map_or(def, |v| v.clamp(-1.0e9, 1.0e9))
258}

260/// A preset's adjust value as a fraction of the box (raw / 100000), `def` (a fraction) when absent.
261fn adj(geom: Option<&El>, name: &str, def: f64) -> f64 {
262    raw_adj(geom, name, def * 100_000.0) / 100_000.0
263}

265/// A pie slice (`pie`) or block arc (`blockArc`) from `adj1` to `adj2` (60000ths of a degree,
266/// clockwise from 3 o'clock); the block arc's band is `adj3` of the short side thick.
267fn sector(block: bool, w: f64, h: f64, geom: Option<&El>) -> Poly {
268    let deg = |n: &str, def: f64| raw_adj(geom, n, def) / 60_000.0;
269    let (st, en) = if block { (deg("adj1", 10_800_000.0), deg("adj2", 0.0)) } else { (deg("adj1", 0.0), deg("adj2", 16_200_000.0)) };
270    let sweep = (en - st).rem_euclid(360.0);
271    let (hw, hh) = (w / 2.0, h / 2.0);
272    let arc = move |rx: f64, ry: f64| {
273        (0..=36).map(move |k| {
274            let t = (st + sweep * k as f64 / 36.0).to_radians();
275            (hw + rx * t.cos(), hh + ry * t.sin())
276        })
277    };
278    if block {
279        let band = lim(w.min(h) * adj(geom, "adj3", 0.25), w.min(h) / 2.0);
280        let pts = arc(hw, hh).chain(arc(hw - band, hh - band).rev()).collect();
281        (pts, true)
282    } else {
283        let mut pts = vec![(hw, hh)];
284        pts.extend(arc(hw, hh));
285        (pts, true)
286    }
287}

289/// `v` limited to `[0, hi]` (never panics, unlike `clamp` with `hi < 0`).
290fn lim(v: f64, hi: f64) -> f64 {
291    v.max(0.0).min(hi.max(0.0))
292}

294/// The outline of a polygon preset in a `w` × `h` box (points), or `None` for presets with no
295/// polygon (ellipse, heart, stars). Unknown presets are rectangles.
296fn outline(prst: &str, w: f64, h: f64, geom: Option<&El>) -> Option<Poly> {
297    let ss = w.min(h);
298    let a = |n: &str, d: f64| adj(geom, n, d);
299    let (hw, hh) = (w / 2.0, h / 2.0);
300    let pts = match prst {
301        "line" | "straightConnector1" => return Some((vec![(0.0, 0.0), (w, h)], false)),
302        "pie" | "blockArc" => return Some(sector(prst == "blockArc", w, h, geom)),
303        "ellipse" | "heart" | "star5" | "star4" | "star6" => return None,
304        "roundRect" => return Some((rounded(w, h, lim(ss * a("adj", 0.16667), ss / 2.0)), true)),
305        "triangle" => {
306            let apex = lim(w * a("adj", 0.5), w);
307            vec![(apex, 0.0), (w, h), (0.0, h)]
308        }
309        "rtTriangle" => vec![(0.0, 0.0), (0.0, h), (w, h)],
310        "diamond" => vec![(hw, 0.0), (w, hh), (hw, h), (0.0, hh)],
311        "rightArrow" | "leftArrow" | "upArrow" | "downArrow" => {
312            let (len, thick) = if prst == "upArrow" || prst == "downArrow" { (h, w) } else { (w, h) };
313            arrow(len, thick, a("adj1", 0.5), a("adj2", 0.5), false)
314                .into_iter()
315                .map(|(u, v)| match prst {
316                    "leftArrow" => (w - u, v),
317                    "upArrow" => (v, h - u),
318                    "downArrow" => (v, u),
319                    _ => (u, v),
320                })
321                .collect()
322        }
323        "leftRightArrow" | "upDownArrow" => {
324            let (len, thick) = if prst == "upDownArrow" { (h, w) } else { (w, h) };
325            arrow(len, thick, a("adj1", 0.5), a("adj2", 0.5), true)
326                .into_iter()
327                .map(|(u, v)| if prst == "upDownArrow" { (v, u) } else { (u, v) })
328                .collect()
329        }
330        "chevron" => {
331            let x1 = lim(ss * a("adj", 0.5), w);
332            vec![(0.0, 0.0), (w - x1, 0.0), (w, hh), (w - x1, h), (0.0, h), (x1, hh)]
333        }
334        "homePlate" => {
335            let x1 = w - lim(ss * a("adj", 0.5), w);
336            vec![(0.0, 0.0), (x1, 0.0), (w, hh), (x1, h), (0.0, h)]
337        }
338        "hexagon" => {
339            let x1 = lim(ss * a("adj", 0.25), w / 2.0);
340            vec![(0.0, hh), (x1, 0.0), (w - x1, 0.0), (w, hh), (w - x1, h), (x1, h)]
341        }
342        "pentagon" => (0..5)
343            .map(|k| {
344                let t = (-90.0 + 72.0 * k as f64).to_radians();
345                (hw + hw * t.cos(), hh + hh * t.sin())
346            })
347            .collect(),
348        "trapezoid" => {
349            let x2 = lim(ss * a("adj", 0.25), w / 2.0);
350            vec![(0.0, h), (x2, 0.0), (w - x2, 0.0), (w, h)]
351        }
352        "parallelogram" => {
353            let x2 = lim(ss * a("adj", 0.25), w);
354            vec![(0.0, h), (x2, 0.0), (w, 0.0), (w - x2, h)]
355        }
356        _ => vec![(0.0, 0.0), (w, 0.0), (w, h), (0.0, h)],
357    };
358    Some((pts, true))
359}

361/// A block arrow along `len` (tip at `len`) across `thick`. `a1` is the shaft thickness and `a2`
362/// the head length, as fractions of the box; `double` puts a head on both ends.
363fn arrow(len: f64, thick: f64, a1: f64, a2: f64, double: bool) -> Vec<(f64, f64)> {
364    let head = lim(len.min(thick) * a2, len / 2.0);
365    let half = lim(thick * a1 / 2.0, thick / 2.0);
366    let c = thick / 2.0;
367    if double {
368        vec![
369            (0.0, c),
370            (head, 0.0),
371            (head, c - half),
372            (len - head, c - half),
373            (len - head, 0.0),
374            (len, c),
375            (len - head, thick),
376            (len - head, c + half),
377            (head, c + half),
378            (head, thick),
379        ]
380    } else {
381        vec![(0.0, c - half), (len - head, c - half), (len - head, 0.0), (len, c), (len - head, thick), (len - head, c + half), (0.0, c + half)]
382    }
383}

385/// A `w` × `h` rectangle with corners of radius `r`, each corner in six steps.
386fn rounded(w: f64, h: f64, r: f64) -> Vec<(f64, f64)> {
387    let mut pts = Vec::with_capacity(28);
388    for (cx, cy, start) in [(w - r, r, -90.0), (w - r, h - r, 0.0), (r, h - r, 90.0), (r, r, 180.0)] {
389        for k in 0..=6 {
390            let t = (start + 15.0 * k as f64).to_radians();
391            pts.push((cx + r * t.cos(), cy + r * t.sin()));
392        }
393    }
394    pts
395}

407/// The text of a `dsp:txBody`: one item per (wrapped) line, stacked and centred in the text box.
408fn text(sp: &El, xf: Xf, b: [f64; 4], theme: &[Rgb], filled: bool, fit: f32, out: &mut Vec<GraphicItem>) {
409    let Some(tb) = sp.child("dsp:txBody") else { return };
410    let body = tb.child("a:bodyPr");
411    let scale = body
412        .and_then(|p| p.child("a:normAutofit"))
413        .and_then(|n| n.attr("fontScale"))
414        .and_then(int)
415        .map_or(1.0, |v| (v as f32 / 100_000.0).clamp(0.1, 1.0));
416    let inset = |n: &str, def: f64| body.and_then(|p| p.attr(n)).and_then(int).map_or(def, |v| (v as f64).clamp(0.0, MAX_EMU));
417    let text_box = match sp.child("dsp:txXfrm").and_then(xfrm_box) {
418        Some(t) => t,
419        None => {
420            let (l, t, r, bo) = (inset("lIns", 91_440.0), inset("tIns", 45_720.0), inset("rIns", 91_440.0), inset("bIns", 45_720.0));
421            [b[0] + l, b[1] + t, (b[2] - l - r).max(0.0), (b[3] - t - bo).max(0.0)]
422        }
423    };
424    let [x0, y0, x1, y1] = xf.rect(text_box);
425    let (x0, y0, tw, th) = (pt(x0), pt(y0), pt(x1 - x0), pt(y1 - y0));
426    // Each line is one item: stop at the diagram's item limit, however much text there is.
427    let room = MAX_ITEMS.saturating_sub(out.len());
428    let fallback = theme.get(if filled { 1 } else { 0 }).copied().unwrap_or(if filled { Rgb::WHITE } else { Rgb::BLACK });
429    let font_ref = sp.child("dsp:style").and_then(|s| s.child("a:fontRef")).and_then(|f| color(f, theme));
430    let mut lines: Vec<Line> = Vec::new();
431    for p in tb.children("a:p") {
432        if lines.len() >= room {
433            break;
434        }
435        let text: String = p.children("a:r").filter_map(|r| r.child("a:t")).map(|t| t.text()).collect();
436        let text: String = text.chars().map(|c| if c.is_control() { ' ' } else { c }).collect();
437        let text = text.trim();
438        if text.is_empty() {
439            continue;
440        }
441        let run = p.children("a:r").find(|r| r.child("a:t").is_some());
442        let rpr = run.and_then(|r| r.child("a:rPr"));
443        // The file's size (18 pt when unset), reduced by the autofit and scaled with the drawing.
444        let base = rpr.and_then(|r| r.attr("sz")).and_then(int).map_or(18.0, |v| (v as f32 / 100.0).clamp(1.0, 400.0));
445        let size = base * scale * fit;
446        let color = rpr.and_then(|r| r.child("a:solidFill")).and_then(|f| color(f, theme)).or(font_ref).unwrap_or(fallback);
447        let bold = rpr.and_then(|r| r.attr("b")).is_some_and(on);
448        let align = match p.child("a:pPr").and_then(|q| q.attr("algn")) {
449            Some("l" | "just" | "dist") => TextAlign::Left,
450            Some("r") => TextAlign::Right,
451            _ => TextAlign::Center,
452        };
453        let font = rpr
454            .and_then(|r| r.child("a:latin"))
455            .and_then(|l| l.attr("typeface"))
456            .filter(|t| !t.is_empty() && !t.starts_with('+'))
457            .map(str::to_string);
458        for line in wrap(text, size, tw, room - lines.len()) {
459            lines.push(Line { text: line, size, color, bold, align, font: font.clone() });
460        }
461    }
462    let total: f32 = lines.iter().map(|l| l.size * LINE).sum();
463    let mut y = y0 + (th - total) / 2.0;
464    for l in lines {
465        let lh = l.size * LINE;
466        out.push(GraphicItem::Text { rect: [x0, y, tw, lh], text: l.text, size: l.size, color: l.color, bold: l.bold, align: l.align, font: l.font });
467        y += lh;
468    }
469}

471/// How wide `s` is at `size` points by the estimate chart and diagram text uses.
472pub(crate) fn text_width(s: &str, size: f32) -> f32 {
473    text_w(s, size)
474}
475
476/// Splits `text` into at most `max` lines of whole words that fit `width` points (by the shared
477/// text width estimate); a word wider than that keeps its own line.
478pub(crate) fn wrap(text: &str, size: f32, width: f32, max: usize) -> Vec<String> {
479    let limit = if width > 0.0 && size > 0.0 { width } else { f32::INFINITY };
480    let space = text_w(" ", size);
481    let mut lines = Vec::new();
482    let (mut cur, mut cur_w) = (String::new(), 0.0f32);
483    for word in text.split_whitespace() {
484        let w = text_w(word, size);
485        if !cur.is_empty() && cur_w + space + w > limit {
486            if lines.len() + 1 >= max {
487                break;
488            }
489            lines.push(std::mem::take(&mut cur));
490            cur_w = 0.0;
491        }
492        if !cur.is_empty() {
493            cur.push(' ');
494            cur_w += space;
495        }
496        cur.push_str(word);
497        cur_w += w;
498    }
499    if !cur.is_empty() && lines.len() < max {
500        lines.push(cur);
501    }
502    lines
503}

506    /// The drawing part of a SmartArt diagram (`a:graphicData` with `dgm:relIds`): the one its
507    /// data part names, else the drawing numbered like the data part. `None` when there's none.
508    pub(super) fn diagram_part(&mut self, gd: &El, rels: &Rels) -> Option<String> {
509        let dm = gd.child("dgm:relIds")?.attr("r:dm")?;
510        let data = super::part_of(rels, dm, rt::DIAGRAM_DATA)?;
511        let ext_id = self.graphic_part(&data).and_then(|d| d.find("dsp:dataModelExt").and_then(|e| e.attr("relId")).map(str::to_string));
512        let is_drawing = |r: &&Rel| !r.external && rel_is(&r.kind, rt::DIAGRAM_DRAWING);
513        let drawing = ext_id.as_deref().and_then(|id| rels.by_id(id)).filter(is_drawing).or_else(|| {
514            // No usable relationship id: the drawing whose number matches the data part's.
515            let num = |t: &str| t.rsplit('/').next().unwrap_or(t).chars().filter(char::is_ascii_digit).collect::<String>();
516            let n = num(&data);
517            rels.list.iter().filter(is_drawing).find(|r| !n.is_empty() && num(&r.target) == n)
518        });
519        drawing.map(|r| r.target.clone())
520    }
521
522    /// The id of the relationship to a SmartArt diagram's drawing part, as its data part names it
523    /// (`dsp:dataModelExt relId`): the id the parts use, not the markup, so it's kept with them.
524    pub(super) fn diagram_drawing_rel(&mut self, gd: &El, rels: &Rels) -> Option<String> {
525        let dm = gd.child("dgm:relIds")?.attr("r:dm")?;
526        let data = super::part_of(rels, dm, rt::DIAGRAM_DATA)?;
527        let id = self.graphic_part(&data)?.find("dsp:dataModelExt")?.attr("relId")?.to_string();
528        rels.by_id(&id).is_some_and(|r| !r.external && rel_is(&r.kind, rt::DIAGRAM_DRAWING)).then_some(id)
529    }
530
531    /// The SmartArt model of a diagram WordCraft wrote (its data part exactly as WordCraft writes
532    /// it); `None` for any other.
533    pub(super) fn smart_art_spec(&mut self, gd: &El, rels: &Rels) -> Option<crate::doc::smart_art::SmartArtSpec> {
534        let dm = gd.child("dgm:relIds")?.attr("r:dm")?;
535        let data = super::part_of(rels, dm, rt::DIAGRAM_DATA)?;
536        crate::docx::smart_art::spec_of(self.graphic_part(&data)?.as_ref())
537    }

539    /// The items of the SmartArt drawing part at `path`, fitted to `w` × `h` points.
540    pub(super) fn diagram_drawing(&mut self, path: &str, w: f32, h: f32) -> Vec<GraphicItem> {
541        let Some(root) = self.graphic_part(path) else { return Vec::new() };
542        let drels = self.pkg.rels(path);
543        let theme = self.doc.settings.theme_colors.clone();
544        let mut image = |id: &str| self.media_for(&drels, id);
545        diagram_items(&root, &theme, w, h, &mut image)
546    }
