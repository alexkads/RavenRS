1//! Charts WordCraft makes ([`ChartSpec`]) as chart parts (ECMA-376 Part 1 §21.2) and back.
2//!
3//! [`chart_xml`] writes a `c:chartSpace` with the data as literals in the part itself
4//! (`c:strLit`, `c:numLit`): no embedded workbook (`c:externalData`) and no external references.
5//! [`chart_items`] draws a chart through the same code that draws chart parts read from files.
6//! [`spec_of`] turns a chart part back into the model only when writing that model again gives
7//! the same markup, so a chart with anything the model leaves out stays as it was read.
8
9use crate::doc::chart::{ChartSeries, ChartSpec, ChartType, LegendPos, MAX_POINTS, MAX_SERIES};
10use crate::doc::graphic::GraphicItem;
11use crate::doc::props::Rgb;
12
13use crate::docx::xml::{self, El, MAX_DEPTH, Node, W};

15const NS_C: &str = "http://schemas.openxmlformats.org/drawingml/2006/chart";
16const NS_A: &str = "http://schemas.openxmlformats.org/drawingml/2006/main";
17const NS_R: &str = "http://schemas.openxmlformats.org/officeDocument/2006/relationships";
18/// Content type of a chart part.
19pub(crate) const CHART_CT: &str = "application/vnd.openxmlformats-officedocument.drawingml.chart+xml";
20/// `a:graphicData` URI of a chart.
21pub(crate) const CHART_URI: &str = "http://schemas.openxmlformats.org/drawingml/2006/chart";
22/// Axis ids (any pair of distinct numbers will do; Word keeps them).
23const CAT_AX: &str = "500000001";
24const VAL_AX: &str = "500000002";

26/// The chart part for `spec`: a complete XML document, written as [`ChartSpec::sanitize`] leaves
27/// it (a value that isn't finite is a gap).
28pub fn chart_xml(spec: &ChartSpec) -> String {
29    chart_part(spec).0
30}
31
32/// [`chart_xml`], and whether the chart limits left series or points of `spec` out of it.
33pub(crate) fn chart_part(spec: &ChartSpec) -> (String, bool) {
34    let kept = bounded(spec);
35    let points = spec.series.iter().map(|s| s.values.len()).max().unwrap_or(0).max(spec.categories.len());
36    let cut = kept.series.len() < spec.series.len() || kept.points() < points;
37    (write_chart(&kept), cut)
38}

40fn write_chart(spec: &ChartSpec) -> String {
41    let mut w = W::new();
42    w.open("c:chartSpace", &[("xmlns:c", NS_C), ("xmlns:a", NS_A), ("xmlns:r", NS_R)]);
43    w.empty("c:roundedCorners", &[("val", "0")]);
44    w.open("c:chart", &[]);
45    if let Some(t) = &spec.title {
46        w.open("c:title", &[]);
47        w.open("c:tx", &[]);
48        w.open("c:rich", &[]);
49        w.empty("a:bodyPr", &[]);
50        w.empty("a:lstStyle", &[]);
51        w.open("a:p", &[]);
52        w.open("a:r", &[]);
53        w.leaf("a:t", &[], t);
54        w.close("a:r");
55        w.close("a:p");
56        w.close("c:rich");
57        w.close("c:tx");
58        w.empty("c:overlay", &[("val", "0")]);
59        w.close("c:title");
60    }
61    w.empty("c:autoTitleDeleted", &[("val", if spec.title.is_some() { "0" } else { "1" })]);
62    w.open("c:plotArea", &[]);
63    w.empty("c:layout", &[]);
64    plot(&mut w, spec);
65    w.close("c:plotArea");
66    if let Some(pos) = spec.legend {
67        w.open("c:legend", &[]);
68        let p = match pos {
69            LegendPos::Right => "r",
70            LegendPos::Left => "l",
71            LegendPos::Top => "t",
72            LegendPos::Bottom => "b",
73        };
74        w.empty("c:legendPos", &[("val", p)]);
75        w.empty("c:overlay", &[("val", "0")]);
76        w.close("c:legend");
77    }
78    w.empty("c:plotVisOnly", &[("val", "1")]);
79    w.empty("c:dispBlanksAs", &[("val", "gap")]);
80    w.close("c:chart");
81    // A white chart area without a border, in the document's own background colour.
82    w.open("c:spPr", &[]);
83    w.open("a:solidFill", &[]);
84    w.empty("a:schemeClr", &[("val", "bg1")]);
85    w.close("a:solidFill");
86    w.open("a:ln", &[]);
87    w.empty("a:noFill", &[]);
88    w.close("a:ln");
89    w.close("c:spPr");
90    w.close("c:chartSpace");
91    w.s
92}

94/// `spec` held to the chart limits. A model deserialised as is (a `.json` document) never went
95/// through [`ChartSpec::sanitize`], and every series repeats every category, so its size would
96/// multiply (#3648). Only what the limits keep is copied before sanitizing.
97fn bounded(spec: &ChartSpec) -> ChartSpec {
98    ChartSpec {
99        kind: spec.kind,
100        categories: spec.categories.iter().take(MAX_POINTS).cloned().collect(),
101        series: spec
102            .series
103            .iter()
104            .take(MAX_SERIES)
105            .map(|s| ChartSeries { name: s.name.clone(), values: s.values.iter().take(MAX_POINTS).copied().collect() })
106            .collect(),
107        title: spec.title.clone(),
108        legend: spec.legend,
109        data_labels: spec.data_labels,
110    }
111    .sanitized()
112}

123/// The chart type element and the series in it, then the axes.
124fn plot(w: &mut W, spec: &ChartSpec) {
125    let k = spec.kind;
126    let el = match k {
127        ChartType::Column | ChartType::ColumnStacked | ChartType::Bar | ChartType::BarStacked => "c:barChart",
128        ChartType::Line => "c:lineChart",
129        ChartType::Area => "c:areaChart",
130        ChartType::Pie => "c:pieChart",
131        ChartType::Doughnut => "c:doughnutChart",
132        ChartType::Scatter => "c:scatterChart",
133    };
134    let stacked = matches!(k, ChartType::ColumnStacked | ChartType::BarStacked);
135    let horizontal = matches!(k, ChartType::Bar | ChartType::BarStacked);
136    w.open(el, &[]);
137    match k {
138        ChartType::Column | ChartType::ColumnStacked | ChartType::Bar | ChartType::BarStacked => {
139            w.empty("c:barDir", &[("val", if horizontal { "bar" } else { "col" })]);
140            w.empty("c:grouping", &[("val", if stacked { "stacked" } else { "clustered" })]);
141        }
142        ChartType::Line | ChartType::Area => w.empty("c:grouping", &[("val", "standard")]),
143        ChartType::Scatter => w.empty("c:scatterStyle", &[("val", "lineMarker")]),
144        ChartType::Pie | ChartType::Doughnut => {}
145    }
146    w.empty("c:varyColors", &[("val", if k.is_round() { "1" } else { "0" })]);
147    // Pie and doughnut charts show the first series; the others are kept for other types.
148    for (i, s) in spec.series.iter().enumerate() {
149        series(w, spec, i, s);
150    }
151    if spec.data_labels {
152        w.open("c:dLbls", &[]);
153        for (name, on) in [
154            ("c:showLegendKey", false),
155            ("c:showVal", true),
156            ("c:showCatName", false),
157            ("c:showSerName", false),
158            ("c:showPercent", false),
159            ("c:showBubbleSize", false),
160        ] {
161            w.empty(name, &[("val", if on { "1" } else { "0" })]);
162        }
163        w.close("c:dLbls");
164    }
165    match k {
166        ChartType::Column | ChartType::ColumnStacked | ChartType::Bar | ChartType::BarStacked => {
167            w.empty("c:gapWidth", &[("val", "150")]);
168            if stacked {
169                w.empty("c:overlap", &[("val", "100")]);
170            }
171        }
172        ChartType::Line => w.empty("c:marker", &[("val", "1")]),
173        ChartType::Pie => w.empty("c:firstSliceAng", &[("val", "0")]),
174        ChartType::Doughnut => {
175            w.empty("c:firstSliceAng", &[("val", "0")]);
176            w.empty("c:holeSize", &[("val", "50")]);
177        }
178        ChartType::Area | ChartType::Scatter => {}
179    }
180    if !k.is_round() {
181        w.empty("c:axId", &[("val", CAT_AX)]);
182        w.empty("c:axId", &[("val", VAL_AX)]);
183    }
184    w.close(el);
185    match k {
186        ChartType::Pie | ChartType::Doughnut => {}
187        ChartType::Scatter => {
188            value_axis(w, CAT_AX, VAL_AX, "b", false, "midCat");
189            value_axis(w, VAL_AX, CAT_AX, "l", true, "midCat");
190        }
191        _ => {
192            category_axis(w, if horizontal { "l" } else { "b" });
193            value_axis(w, VAL_AX, CAT_AX, if horizontal { "b" } else { "l" }, true, "between");
194        }
195    }
196}

198/// One `c:ser` (ECMA-376 element order: idx, order, tx, spPr, marker, cat/xVal, val/yVal, smooth).
199fn series(w: &mut W, spec: &ChartSpec, i: usize, s: &ChartSeries) {
200    let k = spec.kind;
201    let i = i.to_string();
202    w.open("c:ser", &[]);
203    w.empty("c:idx", &[("val", &i)]);
204    w.empty("c:order", &[("val", &i)]);
205    if !s.name.is_empty() {
206        w.open("c:tx", &[]);
207        w.leaf("c:v", &[], &s.name);
208        w.close("c:tx");
209    }
210    match k {
211        // Markers only: the series has no line.
212        ChartType::Scatter => {
213            w.open("c:spPr", &[]);
214            w.open("a:ln", &[("w", "19050")]);
215            w.empty("a:noFill", &[]);
216            w.close("a:ln");
217            w.close("c:spPr");
218        }
219        ChartType::Column | ChartType::ColumnStacked | ChartType::Bar | ChartType::BarStacked => {
220            w.empty("c:invertIfNegative", &[("val", "0")]);
221        }
222        ChartType::Line => {
223            w.open("c:marker", &[]);
224            w.empty("c:symbol", &[("val", "none")]);
225            w.close("c:marker");
226        }
227        _ => {}
228    }
229    if k == ChartType::Scatter {
230        let xs: Option<Vec<Option<f64>>> = spec
231            .categories
232            .iter()
233            .map(|c| if c.is_empty() { Some(None) } else { c.parse::<f64>().ok().filter(|v| v.is_finite()).map(Some) })
234            .collect();
235        w.open("c:xVal", &[]);
236        match xs {
237            Some(xs) => num_lit(w, &xs),
238            None => str_lit(w, &spec.categories),
239        }
240        w.close("c:xVal");
241        w.open("c:yVal", &[]);
242        num_lit(w, &s.values);
243        w.close("c:yVal");
244    } else {
245        w.open("c:cat", &[]);
246        str_lit(w, &spec.categories);
247        w.close("c:cat");
248        w.open("c:val", &[]);
249        num_lit(w, &s.values);
250        w.close("c:val");
251    }
252    if matches!(k, ChartType::Line | ChartType::Scatter) {
253        w.empty("c:smooth", &[("val", "0")]);
254    }
255    w.close("c:ser");
256}

258fn str_lit(w: &mut W, items: &[String]) {
259    w.open("c:strLit", &[]);
260    w.empty("c:ptCount", &[("val", &items.len().to_string())]);
261    for (i, s) in items.iter().enumerate().filter(|(_, s)| !s.is_empty()) {
262        w.open("c:pt", &[("idx", &i.to_string())]);
263        w.leaf("c:v", &[], s);
264        w.close("c:pt");
265    }
266    w.close("c:strLit");
267}
268
269fn num_lit(w: &mut W, values: &[Option<f64>]) {
270    w.open("c:numLit", &[]);
271    w.leaf("c:formatCode", &[], "General");
272    w.empty("c:ptCount", &[("val", &values.len().to_string())]);
273    for (i, v) in values.iter().enumerate() {
274        if let Some(v) = v.filter(|v| v.is_finite()) {
275            w.open("c:pt", &[("idx", &i.to_string())]);
276            w.leaf("c:v", &[], &num(v));
277            w.close("c:pt");
278        }
279    }
280    w.close("c:numLit");
281}

283/// A value as the shortest text that reads back as the same number.
284fn num(v: f64) -> String {
285    if v == 0.0 { "0".to_string() } else { format!("{v}") }
286}

288fn category_axis(w: &mut W, pos: &str) {
289    w.open("c:catAx", &[]);
290    w.empty("c:axId", &[("val", CAT_AX)]);
291    w.open("c:scaling", &[]);
292    w.empty("c:orientation", &[("val", "minMax")]);
293    w.close("c:scaling");
294    w.empty("c:delete", &[("val", "0")]);
295    w.empty("c:axPos", &[("val", pos)]);
296    w.empty("c:majorTickMark", &[("val", "none")]);
297    w.empty("c:minorTickMark", &[("val", "none")]);
298    w.empty("c:tickLblPos", &[("val", "nextTo")]);
299    w.empty("c:crossAx", &[("val", VAL_AX)]);
300    w.empty("c:crosses", &[("val", "autoZero")]);
301    w.empty("c:auto", &[("val", "1")]);
302    w.empty("c:lblAlgn", &[("val", "ctr")]);
303    w.empty("c:lblOffset", &[("val", "100")]);
304    w.empty("c:noMultiLvlLbl", &[("val", "0")]);
305    w.close("c:catAx");
306}

308/// A value axis: with gridlines and no axis line (`grid`), else with its line and no gridlines.
309fn value_axis(w: &mut W, id: &str, cross: &str, pos: &str, grid: bool, between: &str) {
310    w.open("c:valAx", &[]);
311    w.empty("c:axId", &[("val", id)]);
312    w.open("c:scaling", &[]);
313    w.empty("c:orientation", &[("val", "minMax")]);
314    w.close("c:scaling");
315    w.empty("c:delete", &[("val", "0")]);
316    w.empty("c:axPos", &[("val", pos)]);
317    if grid {
318        w.empty("c:majorGridlines", &[]);
319    }
320    w.empty("c:majorTickMark", &[("val", "none")]);
321    w.empty("c:minorTickMark", &[("val", "none")]);
322    w.empty("c:tickLblPos", &[("val", "nextTo")]);
323    if grid {
324        w.open("c:spPr", &[]);
325        w.open("a:ln", &[]);
326        w.empty("a:noFill", &[]);
327        w.close("a:ln");
328        w.close("c:spPr");
329    }
330    w.empty("c:crossAx", &[("val", cross)]);
331    w.empty("c:crosses", &[("val", "autoZero")]);
332    w.empty("c:crossBetween", &[("val", between)]);
333    w.close("c:valAx");
334}

336/// The chart a chart part holds, when the part is exactly what [`chart_xml`] writes for it (so
337/// editing and saving it loses nothing). `None` for any other chart.
338pub(crate) fn spec_of(space: &El) -> Option<ChartSpec> {
339    let chart = space.child("c:chart")?;
340    let plot = chart.child("c:plotArea")?;
341    let ty = plot.els().find(|e| e.name.ends_with("Chart"))?;
342    let val = |e: &El, n: &str| e.child(n).and_then(|c| c.attr("val")).map(str::to_string);
343    let bar = |stacked_kind, clustered_kind| if val(ty, "c:grouping").as_deref() == Some("stacked") { stacked_kind } else { clustered_kind };
344    let kind = match ty.name.as_str() {
345        "c:barChart" if val(ty, "c:barDir").as_deref() == Some("bar") => bar(ChartType::BarStacked, ChartType::Bar),
346        "c:barChart" => bar(ChartType::ColumnStacked, ChartType::Column),
347        "c:lineChart" => ChartType::Line,
348        "c:areaChart" => ChartType::Area,
349        "c:pieChart" => ChartType::Pie,
350        "c:doughnutChart" => ChartType::Doughnut,
351        "c:scatterChart" => ChartType::Scatter,
352        _ => return None,
353    };
354    let scatter = kind == ChartType::Scatter;
355    let sers: Vec<&El> = ty.children("c:ser").take(crate::doc::chart::MAX_SERIES + 1).collect();
356    // More than the model holds can't be the same chart: don't build it to find out.
357    let count = |s: &&El| s.find("c:ptCount").and_then(|c| c.attr("val")).and_then(|v| v.parse::<usize>().ok()).unwrap_or(0);
358    let longest = sers.iter().map(count).max().unwrap_or(0);
359    if sers.len() > crate::doc::chart::MAX_SERIES
360        || longest > crate::doc::chart::MAX_POINTS
361        || sers.len().saturating_mul(longest) > crate::doc::chart::MAX_CELLS
362    {
363        return None;
364    }
365    let categories = sers.first().and_then(|s| s.child(if scatter { "c:xVal" } else { "c:cat" })).map(literal).unwrap_or_default();
366    let series = sers
367        .iter()
368        .map(|s| ChartSeries {
369            name: s.child("c:tx").and_then(|t| t.child("c:v")).map(|v| v.text()).unwrap_or_default(),
370            values: s
371                .child(if scatter { "c:yVal" } else { "c:val" })
372                .map(literal)
373                .unwrap_or_default()
374                .iter()
375                .map(|v| v.parse::<f64>().ok().filter(|v| v.is_finite()))
376                .collect(),
377        })
378        .collect();
379    let title = chart.child("c:title").and_then(|t| t.find("a:t")).map(|t| t.text());
380    let legend = chart.child("c:legend").map(|l| match val(l, "c:legendPos").as_deref() {
381        Some("l") => LegendPos::Left,
382        Some("t") => LegendPos::Top,
383        Some("b") => LegendPos::Bottom,
384        _ => LegendPos::Right,
385    });
386    let data_labels = ty.child("c:dLbls").and_then(|d| d.child("c:showVal")).and_then(|v| v.attr("val")) == Some("1");
387    let spec = ChartSpec { kind, categories, series, title, legend, data_labels }.sanitized();
388    // Editable only when nothing would be lost: writing the model again gives the same part.
389    let again = xml::parse(chart_xml(&spec).as_bytes()).ok()?;
390    same(space, &again, 0).then_some(spec)
391}

393/// The points of a `c:strLit` / `c:numLit` (or a cache) as texts by index, `ptCount` long.
394fn literal(e: &El) -> Vec<String> {
395    let Some(lit) = e.els().find(|c| matches!(c.name.as_str(), "c:strLit" | "c:numLit")) else { return Vec::new() };
396    let n =
397        lit.child("c:ptCount").and_then(|c| c.attr("val")).and_then(|v| v.parse::<usize>().ok()).unwrap_or(0).min(crate::doc::chart::MAX_POINTS + 1);
398    let mut out = vec![String::new(); n];
399    for p in lit.children("c:pt") {
400        if let Some(slot) = p.attr("idx").and_then(|i| i.parse::<usize>().ok()).and_then(|i| out.get_mut(i)) {
401            *slot = p.child("c:v").map(|v| v.text()).unwrap_or_default();
402        }
403    }
404    out
405}

407/// Whether two elements are the same markup: names, attributes (in any order), child elements in
408/// order and text (whitespace-only text between elements ignored).
409fn same(a: &El, b: &El, depth: usize) -> bool {
410    if depth > MAX_DEPTH || a.name != b.name || a.attrs.len() != b.attrs.len() {
411        return false;
412    }
413    if !a.attrs.iter().all(|(k, v)| b.attr(k) == Some(v.as_str()) && b.attrs.iter().any(|(bk, _)| bk == k)) {
414        return false;
415    }
416    let text = |e: &El| -> String {
417        e.kids
418            .iter()
419            .filter_map(|n| match n {
420                Node::Text(t) if !t.trim().is_empty() => Some(t.as_str()),
421                _ => None,
422            })
423            .collect()
424    };
425    if text(a) != text(b) {
426        return false;
427    }
428    let (mut x, mut y) = (a.els(), b.els());
429    loop {
430        match (x.next(), y.next()) {
431            (None, None) => return true,
432            (Some(p), Some(q)) if same(p, q, depth + 1) => {}
433            _ => return false,
434        }
435    }
436}

452    #[test]
453    fn a_chart_past_the_limits_is_written_within_them() {
454        // A model deserialised as is (a `.json` document) never went through `sanitize`, and every
455        // series repeats every category: 300 series of 4,500 points wrote 2.7 million points (#3648).
456        use crate::doc::chart::MAX_CELLS;
457        let spec = ChartSpec {
458            kind: ChartType::Column,
459            categories: (0..4500).map(|i| i.to_string()).collect(),
460            series: (0..300).map(|i| ChartSeries { name: format!("S{i}"), values: vec![Some(1.0); 4500] }).collect(),
461            title: None,
462            legend: None,
463            data_labels: false,
464        };
465        let x = chart_xml(&spec);
466        let series = x.matches("<c:ser>").count();
467        let points = x.matches("<c:pt ").count();
468        assert!((1..=MAX_SERIES).contains(&series), "{series} series");
469        assert!(points <= 2 * MAX_CELLS, "{points} points");
470        assert!(!x.contains(&format!("<c:pt idx=\"{MAX_POINTS}\"")), "past the point limit");
