66/// A fingerprint of `glossary` (FNV-1a over its debug form): stable across runs and builds that
67/// keep the model's shape, and cheap (nothing is allocated).
68pub fn glossary_fingerprint(glossary: &[crate::doc::BuildingBlock]) -> String {
69    struct Fnv(u64);
70    impl std::fmt::Write for Fnv {
71        fn write_str(&mut self, s: &str) -> std::fmt::Result {
72            for b in s.bytes() {
73                self.0 = (self.0 ^ u64::from(b)).wrapping_mul(0x0100_0000_01b3);
74            }
75            Ok(())
76        }
77    }
78    let mut h = Fnv(0xcbf2_9ce4_8422_2325);
79    // Writing into `Fnv` never fails.
80    let _ = std::fmt::Write::write_fmt(&mut h, format_args!("{glossary:?}"));
81    format!("{:016x}", h.0)

150/// Can `s` be a field of a manifest line?
151pub fn manifest_field_ok(s: &str) -> bool {
152    !s.is_empty() && !s.contains(['\t', '\r', '\n'])
153}
154
155impl EmbeddedManifest {
156    pub fn parse(bytes: &[u8]) -> EmbeddedManifest {
157        let mut m = EmbeddedManifest::default();
158        for line in String::from_utf8_lossy(bytes).lines() {
159            let f: Vec<&str> = line.split('\t').collect();
160            match f.as_slice() {
161                ["P", part, ct] if m.parts.len() < MAX_EMBEDDED_PARTS => {
162                    m.parts.entry((*part).to_string()).or_insert_with(|| EmbeddedPart { content_type: (*ct).to_string(), rels: Vec::new() });
163                }
164                ["R", part, id, kind, target, mode] => {
165                    if let Some(p) = m.parts.get_mut(*part)
166                        && p.rels.len() < MAX_EMBEDDED_RELS
167                        && !p.rels.iter().any(|r| r.id == *id)
168                    {
169                        p.rels.push(Rel { id: (*id).to_string(), kind: (*kind).to_string(), target: (*target).to_string(), external: *mode == "E" });
170                    }
171                }
172                _ => {}
173            }
174        }
175        m
176    }
177
178    pub fn to_bytes(&self) -> Vec<u8> {
179        let mut s = String::new();
180        for (part, p) in &self.parts {
181            if !manifest_field_ok(part) || !manifest_field_ok(&p.content_type) {
182                continue;
183            }
184            s.push_str(&format!("P\t{part}\t{}\n", p.content_type));
185        }
186        for (part, p) in &self.parts {
187            for r in p.rels.iter().filter(|r| [&r.id, &r.kind, &r.target].iter().all(|f| manifest_field_ok(f))) {
188                s.push_str(&format!("R\t{part}\t{}\t{}\t{}\t{}\n", r.id, r.kind, r.target, if r.external { "E" } else { "I" }));
189            }
190        }
191        s.into_bytes()
192    }
193}

201/// The decompressed parts of a package (bounded). Shared, so a part the document keeps as it is (a
202/// picture, an embedded object) is handed over rather than copied.
203pub struct Package {
204    files: BTreeMap<String, Arc<Vec<u8>>>,
205    /// Lowercased part name → its stored spelling (the first in name order): part names are
206    /// case-insensitive (ECMA-376 Part 2 §6.2.2.3), so every spelling is one part.
207    lower: HashMap<String, String>,
208}
209
210impl Package {
211    pub fn open(bytes: &[u8]) -> Result<Package, DocxError> {
212        Package::open_limited(bytes, MAX_PART, MAX_TOTAL)
213    }
214
215    pub fn open_limited(bytes: &[u8], max_part: u64, max_total: u64) -> Result<Package, DocxError> {
216        // Before `zip` parses every entry the end record declares (#3660).
217        crate::doc::zip_guard::check(bytes, MAX_ENTRIES as u64).map_err(DocxError::Limit)?;
218        let mut zip = zip::ZipArchive::new(Cursor::new(bytes)).map_err(|e| DocxError::Zip(e.to_string()))?;
219        if zip.len() > MAX_ENTRIES {
220            return Err(DocxError::Limit(format!("{} zip entries", zip.len())));
221        }
222        let mut files = BTreeMap::new();
223        let mut total: u64 = 0;
224        for i in 0..zip.len() {
225            let mut f = match zip.by_index(i) {
226                Ok(f) => f,
227                Err(e) => {
228                    log::warn!("docx: skipping unreadable zip entry {i}: {e}");
229                    continue;
230                }
231            };
232            if f.is_dir() {
233                continue;
234            }
235            // RavenRS: zip 9 says when an entry name can't be read; such an entry is skipped.
236            let Ok(name) = f.name().map(|n| n.trim_start_matches('/').to_string()) else { continue };
237            if f.size() > max_part {
238                return Err(DocxError::Limit(format!("part {name} declares {} bytes", f.size())));
239            }
240            let room = max_part.min(max_total.saturating_sub(total));
241            let mut buf = Vec::with_capacity(f.size().min(16 * 1024 * 1024) as usize);
242            let read = (&mut f).take(room + 1).read_to_end(&mut buf);
243            // Charged whether or not the read succeeds: an entry inflates in full before its CRC
244            // is checked, so failed entries would otherwise inflate without limit (#3723).
245            total = total.saturating_add(buf.len() as u64);
246            if buf.len() as u64 > room {
247                return Err(DocxError::Limit(format!("part {name} decompresses past the size limit")));
248            }
249            if let Err(e) = read {
250                // A corrupt entry: keep going unless it's a part we can't do without.
251                log::warn!("docx: zip entry {name} unreadable: {e}");
252                continue;
253            }
254            files.insert(name, Arc::new(buf));
255        }
256        Ok(Package::new(files))
257    }
258
259    /// The bytes of all its parts.
260    pub fn size(&self) -> u64 {
261        self.files.values().map(|f| f.len() as u64).sum()
262    }
263
264    /// A package from parts already in memory (a Flat OPC file, a converted document), by part
265    /// name without the leading `/`.
266    pub fn from_parts(files: BTreeMap<String, Vec<u8>>) -> Package {
267        Package::new(files.into_iter().map(|(name, bytes)| (name, Arc::new(bytes))).collect())
268    }
269
270    fn new(mut files: BTreeMap<String, Arc<Vec<u8>>>) -> Package {
271        let mut lower: HashMap<String, String> = HashMap::new();
272        let mut collides = false;
273        for name in files.keys() {
274            match lower.entry(name.to_ascii_lowercase()) {
275                Entry::Occupied(_) => collides = true,
276                Entry::Vacant(v) => {
277                    v.insert(name.clone());
278                }
279            }
280        }
281        if !collides {
282            return Package { files, lower };
283        }
284        // Entries whose names differ only in case are one part, so every spelling of the name
285        // reaches the same stored entry (#3704). The spelling kept is the one a relationship
286        // names, so the part the document points at is the one read; the first in byte order only
287        // when no relationship names any of them.
288        let named = rel_targets(&files);
289        for name in files.keys().rev().filter(|n| named.contains(n.as_str())) {
290            lower.insert(name.to_ascii_lowercase(), name.clone());
291        }
292        let before = files.len();
293        files.retain(|name, _| lower.get(&name.to_ascii_lowercase()) == Some(name));
294        let dropped = before.saturating_sub(files.len());
295        log::warn!("docx: {dropped} zip entries are other spellings of parts already read; they are left out");
296        Package { files, lower }
297    }

330    /// Relationships of `part` (from `dir/_rels/name.rels`).
331    pub fn rels(&self, part: &str) -> Rels {
332        let (dir, file) = split_dir(part);
333        let path = if dir.is_empty() { format!("_rels/{file}.rels") } else { format!("{dir}/_rels/{file}.rels") };
334        let mut rels = Rels::default();
335        let root = match self.xml(&path) {
336            Ok(Some(root)) => root,
337            Ok(None) => return rels,
338            Err(e) => {
339                log::warn!("docx: ignoring unreadable relationships {path}: {e}");
340                return rels;
341            }
342        };
343        for r in root.els().filter(|e| e.local() == "Relationship") {
344            let (Some(id), Some(target)) = (r.attr("Id"), r.attr("Target")) else { continue };
345            let external = r.attr("TargetMode").is_some_and(|m| m.eq_ignore_ascii_case("External"));
346            // Internal targets name the part as stored, so a part reached under several spellings
347            // is kept, cached and written once (#3704).
348            let resolved = if external {
349                target.to_string()
350            } else {
351                let path = resolve(dir, target);
352                self.canonical(&path).map_or(path, str::to_string)
353            };
354            rels.ids.entry(id.to_string()).or_insert(rels.list.len());
355            rels.list.push(Rel { id: id.to_string(), kind: r.attr("Type").unwrap_or("").to_string(), target: resolved, external });
356        }
357        rels
358    }
359}

361/// `[Content_Types].xml`: Override by part name, then Default by extension (both case-insensitive).
362#[derive(Default)]
363pub struct ContentTypes {
364    overrides: HashMap<String, String>,
365    defaults: HashMap<String, String>,
366}
367
368impl ContentTypes {
369    pub fn read(pkg: &Package) -> ContentTypes {
370        let mut ct = ContentTypes::default();
371        let root = match pkg.xml("[Content_Types].xml") {
372            Ok(Some(root)) => root,
373            Ok(None) => return ct,
374            Err(e) => {
375                log::warn!("docx: ignoring unreadable [Content_Types].xml: {e}");
376                return ct;
377            }
378        };
379        for e in root.els() {
380            let (Some(key), Some(t)) = (e.attr("PartName").or_else(|| e.attr("Extension")), e.attr("ContentType")) else { continue };
381            let map = if e.local() == "Override" { &mut ct.overrides } else { &mut ct.defaults };
382            map.entry(key.trim_start_matches('/').to_ascii_lowercase()).or_insert_with(|| t.to_string());
383        }
384        ct
385    }
386
387    /// Content type of `part` (a package path without the leading `/`).
388    pub fn of(&self, part: &str) -> Option<&str> {
389        let part = part.trim_start_matches('/').to_ascii_lowercase();
390        if let Some(t) = self.overrides.get(&part) {
391            return Some(t);
392        }
393        let ext = part.rsplit_once('.').map(|(_, e)| e)?;
394        self.defaults.get(ext).map(String::as_str)
395    }
396}

423fn split_dir(part: &str) -> (&str, &str) {
424    let p = part.trim_start_matches('/');
425    match p.rfind('/') {
426        Some(i) => (p.get(..i).unwrap_or(""), p.get(i + 1..).unwrap_or(p)),
427        None => ("", p),
428    }
429}

431/// Every internal part name the relationship parts in `files` target, spelled as they spell it.
432fn rel_targets(files: &BTreeMap<String, Arc<Vec<u8>>>) -> HashSet<String> {
433    let mut named = HashSet::new();
434    for (name, bytes) in files {
435        // `dir/_rels/file.rels` relates `dir/file`, so its targets resolve against `dir`.
436        let lower = name.to_ascii_lowercase();
437        let Some(at) = lower.rfind("_rels/").filter(|_| lower.ends_with(".rels")) else { continue };
438        let dir = name.get(..at).unwrap_or("").trim_end_matches('/');
439        let Ok(root) = xml::parse(bytes) else { continue };
440        for r in root.els().filter(|e| e.local() == "Relationship") {
441            let external = r.attr("TargetMode").is_some_and(|m| m.eq_ignore_ascii_case("External"));
442            if let (Some(target), false) = (r.attr("Target"), external) {
443                named.insert(resolve(dir, target));
444            }
445        }
446    }
447    named
448}

450/// Resolve a relative target against a source directory into a package path.
451pub fn resolve(dir: &str, target: &str) -> String {
452    let target = target.split('#').next().unwrap_or(target);
453    let mut parts: Vec<&str> = if target.starts_with('/') { Vec::new() } else { dir.split('/').filter(|s| !s.is_empty()).collect() };
454    for seg in target.split('/') {
455        match seg {
456            "" | "." => {}
457            ".." => {
458                parts.pop();
459            }
460            s => parts.push(s),
461        }
462    }
463    parts.join("/")
464}

466/// The relative reference from part `from` to part `to` (both package paths), as a
467/// relationship target: `word/document.xml` → `word/charts/chart1.xml` is `charts/chart1.xml`.
468pub fn relative(from: &str, to: &str) -> String {
469    let from_dir: Vec<&str> = from.split('/').filter(|s| !s.is_empty()).collect();
470    let from_dir = from_dir.get(..from_dir.len().saturating_sub(1)).unwrap_or(&[]);
471    let to: Vec<&str> = to.split('/').filter(|s| !s.is_empty()).collect();
472    let to_dir = to.get(..to.len().saturating_sub(1)).unwrap_or(&[]);
473    let common = from_dir.iter().zip(to_dir).take_while(|(a, b)| a == b).count();
474    let mut out: Vec<&str> = vec![".."; from_dir.len() - common];
475    out.extend(to.get(common..).unwrap_or(&[]));
476    let s = out.join("/");
477    // A relative reference whose first segment has a colon would read as a URI scheme.
478    if s.split('/').next().is_some_and(|seg| seg.contains(':')) { format!("./{s}") } else { s }
479}

481/// Build a zip from (name, bytes) entries.
482pub fn zip_entries(entries: &[(String, Vec<u8>)]) -> Result<Vec<u8>, DocxError> {
483    let mut zw = zip::ZipWriter::new(Cursor::new(Vec::new()));
484    let opts = zip::write::SimpleFileOptions::default().compression_method(zip::CompressionMethod::Deflated).compression_level(Some(6));
485    let stored = zip::write::SimpleFileOptions::default().compression_method(zip::CompressionMethod::Stored);
486    for (name, bytes) in entries {
487        // Already-compressed media gain nothing from deflate.
488        let lower = name.to_ascii_lowercase();
489        let o = if [".png", ".jpg", ".jpeg", ".gif", ".webp"].iter().any(|e| lower.ends_with(e)) { stored } else { opts };
490        zw.start_file(name.as_str(), o).map_err(|e| DocxError::Zip(e.to_string()))?;
491        zw.write_all(bytes).map_err(|e| DocxError::Zip(e.to_string()))?;
492    }
493    let c = zw.finish().map_err(|e| DocxError::Zip(e.to_string()))?;
494    Ok(c.into_inner())
495}

579    #[test]
580    fn entries_that_fail_their_check_still_count_against_the_total() {
581        // Each entry inflates in full before its CRC is checked; a failed one was dropped without
582        // being charged, so any number of them inflated (#3723).
583        let names: Vec<String> = (0..20).map(|i| format!("p{i}.bin")).collect();
584        let entries: Vec<(&str, Vec<u8>)> = names.iter().map(|n| (n.as_str(), vec![0u8; 1_000_000])).collect();
585        let mut z = zip_of(&entries);
586        let mut patched = 0;
587        for sig in [[0x50u8, 0x4B, 0x01, 0x02], [0x50, 0x4B, 0x03, 0x04]] {
588            let crc_at = if sig[2] == 1 { 16 } else { 14 };
589            let found: Vec<usize> = z.windows(4).enumerate().filter(|(_, w)| *w == sig).map(|(i, _)| i).collect();
590            for pos in found {
591                z[pos + crc_at] ^= 0xFF;
592                patched += 1;
593            }
594        }
595        assert_eq!(patched, 40);
596        assert!(matches!(Package::open_limited(&z, 2_000_000, 4_000_000), Err(DocxError::Limit(_))));
597    }
