1//! Word 2003 XML Document (WordprocessingML 2003, root `w:wordDocument`, from the public Office
2//! 2003 XML Reference Schemas). Import; the writer is `wordml2003_write`.
3//!
4//! ECMA-376 kept most of the 2003 vocabulary (`w:p`, `w:r`, `w:pPr`, `w:rPr`, `w:tbl`,
5//! `w:sectPr`, `w:styles`…), so the file is opened by rewriting it as the package it corresponds
6//! to and reading that with the DOCX reader: one reader, one set of property rules. The rewrite
7//! renames what changed (hyphenated names, `w:hlink`, `w:listPr`, `w:nfc`, `w:vmerge`…), moves
8//! what 2003 keeps inline into parts (lists into numbering, headers and footers, footnotes,
9//! endnotes, comments, `w:binData` pictures into media) and turns `aml:annotation` bookmarks,
10//! comments and tracked insertions/deletions into their ECMA-376 elements. Bounded: the parser
11//! caps depth and element count, pictures are capped in number and size.
12

24/// Auxiliary hints (`wx:`): display hints Word writes alongside the real markup.
25const AUX_NS: &str = "http://schemas.microsoft.com/office/word/2003/auxHint";
26/// Annotation markup (`aml:`): bookmarks, comments, tracked changes.
27const AML_NS: &str = "http://schemas.microsoft.com/aml/2001/core";
28
29/// Most pictures (`w:binData`) taken from one file.
30const MAX_MEDIA: usize = 4096;
31/// Most header, footer, footnote, endnote and comment stories.
32const MAX_STORIES: usize = 50_000;
33
34/// Read a Word 2003 XML Document.
35pub fn read(bytes: &[u8]) -> Result<Document, DocxError> {
36    crate::docx::read::read_opened(&convert(bytes)?, None)
37}
38
39/// Parts the converted package gets relationships for.
40const DOC: usize = 0;
41const FOOTNOTES: usize = 1;
42const ENDNOTES: usize = 2;
43const COMMENTS: usize = 3;
44
45struct Conv {
46    /// `w:binData` name (`wordml://…`) → media target relative to `word/`.
47    media: HashMap<String, String>,
48    files: BTreeMap<String, Vec<u8>>,
49    /// Relationships per part: document, footnotes, endnotes, comments, then one per header or
50    /// footer (index in `stories` + 4).
51    rels: Vec<PartRels>,
52    footnotes: Vec<Node>,
53    endnotes: Vec<Node>,
54    comments: Vec<Node>,
55    /// Header and footer parts: (part name under `word/`, root element).
56    stories: Vec<(String, El)>,
57    next_note: i64,
58    story_count: usize,
59}

61/// Rewrite a Word 2003 XML Document as a WordprocessingML package.
62pub(crate) fn convert(bytes: &[u8]) -> Result<Package, DocxError> {
63    let root = xml::parse_ns(bytes, &[("w", W2003_NS), ("wx", AUX_NS), ("aml", AML_NS)])?;
64    if root.name != "w:wordDocument" {
65        return Err(DocxError::NotWord(format!("root element is {}", root.name)));
66    }
67    let mut c = Conv {
68        media: HashMap::new(),
69        files: BTreeMap::new(),
70        rels: (0..4).map(|_| PartRels::default()).collect(),
71        footnotes: Vec::new(),
72        endnotes: Vec::new(),
73        comments: Vec::new(),
74        stories: Vec::new(),
75        next_note: 1,
76        story_count: 0,
77    };
78    let mut total = 0u64;
79    c.collect_media(&root, 0, &mut total)?;
80
81    // Body: `wx:sect` / `wx:sub-section` wrappers unwrapped; the last section's properties last.
82    let mut body = El { name: "w:body".into(), ..El::default() };
83    if let Some(b) = root.child("w:body") {
84        body.kids = c.kids(b, DOC, 1);
85    }
86    let last_sect = body.kids.iter().rposition(|n| matches!(n, Node::El(e) if e.name == "w:sectPr"));
87    if let Some(i) = last_sect {
88        let s = body.kids.remove(i);
89        body.kids.retain(|n| !matches!(n, Node::El(e) if e.name == "w:sectPr"));
90        body.kids.push(s);
91    }
92    let mut document = El { name: "w:document".into(), attrs: xml::body_ns(), kids: Vec::new() };
93    document.kids.push(Node::El(body));
94
95    let mut overrides: Vec<(String, &str)> = Vec::new();
96    let wml = |s: &str| format!("application/vnd.openxmlformats-officedocument.wordprocessingml.{s}+xml");
97    let styles = c.styles(&root);
98    c.put("word/styles.xml", &styles);
99    c.rels[DOC].add(rt::STYLES, "styles.xml", false);
100    overrides.push(("/word/styles.xml".into(), "styles"));
101    if let Some(n) = root.child("w:lists").map(|l| c.numbering(l)) {
102        c.put("word/numbering.xml", &n);
103        c.rels[DOC].add(rt::NUMBERING, "numbering.xml", false);
104        overrides.push(("/word/numbering.xml".into(), "numbering"));
105    }
106    for (part, root_name, kids, rel, ct) in [
107        (FOOTNOTES, "w:footnotes", std::mem::take(&mut c.footnotes), rt::FOOTNOTES, "footnotes"),
108        (ENDNOTES, "w:endnotes", std::mem::take(&mut c.endnotes), rt::ENDNOTES, "endnotes"),
109        (COMMENTS, "w:comments", std::mem::take(&mut c.comments), rt::COMMENTS, "comments"),
110    ] {
111        if kids.is_empty() {
112            continue;
113        }
114        let file = format!("{ct}.xml");
115        c.put(&format!("word/{file}"), &El { name: root_name.into(), attrs: xml::body_ns(), kids });
116        c.put_rels(&file, part);
117        c.rels[DOC].add(rel, &file, false);
118        overrides.push((format!("/word/{file}"), ct));
119    }
120    for (i, (file, story)) in std::mem::take(&mut c.stories).into_iter().enumerate() {
121        c.put(&format!("word/{file}"), &story);
122        c.put_rels(&file, i + 4);
123        overrides.push((format!("/word/{file}"), if story.name == "w:ftr" { "footer" } else { "header" }));
124    }
125    c.put("word/document.xml", &document);
126    c.put_rels("document.xml", DOC);
127
128    let mut root_rels = PartRels::default();
129    root_rels.add(rt::OFFICE_DOC, "word/document.xml", false);
130    if let Some(props) = root.child("o:DocumentProperties") {
131        c.files.insert("docProps/core.xml".into(), core_xml(props));
132        root_rels.add(rt::CORE, "docProps/core.xml", false);
133    }
134    c.files.insert("_rels/.rels".into(), root_rels.xml());
135
136    let mut w = W::new();
137    w.open("Types", &[("xmlns", "http://schemas.openxmlformats.org/package/2006/content-types")]);
138    w.empty("Default", &[("Extension", "rels"), ("ContentType", "application/vnd.openxmlformats-package.relationships+xml")]);
139    w.empty("Default", &[("Extension", "xml"), ("ContentType", "application/xml")]);
140    let mut exts: Vec<&str> = c.media.values().filter_map(|t| t.rsplit_once('.').map(|(_, e)| e)).collect();
141    exts.sort_unstable();
142    exts.dedup();
143    for ext in exts {
144        w.empty("Default", &[("Extension", ext), ("ContentType", media_type(ext))]);
145    }
146    w.empty("Override", &[("PartName", "/word/document.xml"), ("ContentType", &wml("document.main"))]);
147    for (part, kind) in &overrides {
148        w.empty("Override", &[("PartName", part), ("ContentType", &wml(kind))]);
149    }
150    if c.files.contains_key("docProps/core.xml") {
151        w.empty("Override", &[("PartName", "/docProps/core.xml"), ("ContentType", "application/vnd.openxmlformats-package.core-properties+xml")]);
152    }
153    w.close("Types");
154    c.files.insert("[Content_Types].xml".into(), w.into_bytes());
155    Ok(Package::from_parts(c.files))
156}

158impl Conv {
159    /// Store an element as a part.
160    fn put(&mut self, path: &str, root: &El) {
161        let mut s = String::from("<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n");
162        s.push_str(&root.to_xml());
163        self.files.insert(path.into(), s.into_bytes());
164    }
165
166    /// Store the relationships of `word/{file}` (if it has any).
167    fn put_rels(&mut self, file: &str, part: usize) {
168        if let Some(r) = self.rels.get(part).filter(|r| !r.is_empty()) {
169            let x = r.xml();
170            self.files.insert(format!("word/_rels/{file}.rels"), x);
171        }
172    }
173
174    /// Every `w:binData` in the file, decoded into a media part.
175    fn collect_media(&mut self, e: &El, depth: usize, total: &mut u64) -> Result<(), DocxError> {
176        if depth > MAX_DEPTH {
177            return Ok(());
178        }
179        for k in e.els() {
180            if k.name == "w:binData" {
181                let Some(name) = k.attr("w:name").filter(|n| !self.media.contains_key(*n)) else { continue };
182                if self.media.len() >= MAX_MEDIA {
183                    continue;
184                }
185                let text = k.text();
186                let b64: String = text.chars().filter(|c| !c.is_ascii_whitespace()).collect();
187                if (b64.len() / 4 * 3) as u64 > MAX_PART {
188                    return Err(DocxError::Limit(format!("picture {name} is larger than the size limit")));
189                }
190                let Ok(bytes) = base64::engine::general_purpose::STANDARD_NO_PAD.decode(b64.trim_end_matches('=')) else {
191                    log::warn!("wordml2003: skipping picture {name}: invalid base64");
192                    continue;
193                };
194                *total = total.saturating_add(bytes.len() as u64);
195                if *total > MAX_TOTAL {
196                    return Err(DocxError::Limit("pictures larger than the size limit".into()));
197                }
198                let ext = picture_ext(&bytes, name);
199                let target = format!("media/image{}.{ext}", self.media.len() + 1);
200                self.files.insert(format!("word/{target}"), bytes);
201                self.media.insert(name.to_string(), target);
202            } else {
203                self.collect_media(k, depth + 1, total)?;
204            }
205        }
206        Ok(())
207    }
208
209    /// The converted children of `e`.
210    fn kids(&mut self, e: &El, part: usize, depth: usize) -> Vec<Node> {
211        let mut out = Vec::with_capacity(e.kids.len());
212        for n in &e.kids {
213            match n {
214                Node::Text(t) => out.push(Node::Text(t.clone())),
215                Node::El(k) => out.extend(self.node(k, part, depth + 1)),
216            }
217        }
218        out
219    }
220
221    /// `e` converted: nothing (dropped), its children (unwrapped) or one element.
222    fn node(&mut self, e: &El, part: usize, depth: usize) -> Vec<Node> {
223        if depth > MAX_DEPTH {
224            return Vec::new();
225        }
226        match e.name.as_str() {
227            "wx:sect" | "wx:sub-section" | "wx:pBdrGroup" => return self.kids(e, part, depth),
228            "w:binData" | "w:fonts" | "w:docPr" | "w:lists" | "w:styles" | "w:latentStyles" | "w:versionOfBuiltInStyleNames" => return Vec::new(),
229            "w:hlink" => return vec![Node::El(self.hyperlink(e, part, depth))],
230            "w:listPr" => return vec![Node::El(list_pr(e))],
231            "w:footnote" | "w:endnote" => return self.note(e, depth).into_iter().map(Node::El).collect(),
232            "aml:annotation" => return self.annotation(e, part, depth),
233            "w:sectPr" => return vec![Node::El(self.section(e, part, depth))],
234            "w:nfc" => return vec![Node::El(num_fmt(e))],
235            _ => {}
236        }
237        if !keep(&e.name) {
238            return Vec::new();
239        }
240        let mut el = El { name: rename(&e.name), attrs: attrs(e), kids: self.kids(e, part, depth) };
241        if el.name == "v:imagedata"
242            && let Some(target) = e.attr("src").and_then(|s| self.media.get(s)).cloned()
243            && let Some(r) = self.rels.get_mut(part)
244        {
245            let id = r.add(rt::IMAGE, &target, false);
246            el.attrs.retain(|(k, _)| k != "r:id");
247            el.attrs.push(("r:id".into(), id));
248        }
249        if el.name == "w:style" {
250            for (k, v) in &mut el.attrs {
251                if k == "w:default" && v == "on" {
252                    *v = "1".into();
253                }
254            }
255        }
256        vec![Node::El(el)]
257    }
258
259    /// `w:hlink` → `w:hyperlink` (an external relationship for `w:dest`).
260    fn hyperlink(&mut self, e: &El, part: usize, depth: usize) -> El {
261        let mut attrs = Vec::new();
262        if let Some(dest) = e.attr("w:dest").filter(|d| !d.is_empty())
263            && let Some(r) = self.rels.get_mut(part)
264        {
265            attrs.push(("r:id".into(), r.add(rt::HYPERLINK, dest, true)));
266        }
267        for (from, to) in [("w:bookmark", "w:anchor"), ("w:screenTip", "w:tooltip"), ("w:target", "w:tgtFrame")] {
268            if let Some(v) = e.attr(from) {
269                attrs.push((to.into(), v.to_string()));
270            }
271        }
272        El { name: "w:hyperlink".into(), attrs, kids: self.kids(e, part, depth) }
273    }

275    /// A footnote or endnote written inline in its run → a reference, the note in its part.
276    fn note(&mut self, e: &El, depth: usize) -> Option<El> {
277        // Separator definitions (`w:type`) live in document settings, not in the text.
278        if e.attr("w:type").is_some() || self.story_count >= MAX_STORIES {
279            return None;
280        }
281        self.story_count += 1;
282        let foot = e.name == "w:footnote";
283        let id = self.next_note.to_string();
284        self.next_note = self.next_note.saturating_add(1);
285        let part = if foot { FOOTNOTES } else { ENDNOTES };
286        let kids = self.kids(e, part, depth);
287        let note = El { name: e.name.clone(), attrs: vec![("w:id".into(), id.clone())], kids };
288        if foot {
289            self.footnotes.push(Node::El(note));
290        } else {
291            self.endnotes.push(Node::El(note));
292        }
293        let name = if foot { "w:footnoteReference" } else { "w:endnoteReference" };
294        Some(El { name: name.into(), attrs: vec![("w:id".into(), id)], kids: Vec::new() })
295    }
296
297    /// `aml:annotation`: bookmarks, comments and tracked insertions/deletions.
298    fn annotation(&mut self, e: &El, part: usize, depth: usize) -> Vec<Node> {
299        let id = e.attr("aml:id").filter(|v| v.parse::<i64>().is_ok()).unwrap_or("0").to_string();
300        let el = |name: &str, attrs: Vec<(String, String)>| vec![Node::El(El { name: name.into(), attrs, kids: Vec::new() })];
301        let author_date = |e: &El| {
302            let mut a = vec![("w:id".to_string(), id.clone())];
303            if let Some(v) = e.attr("aml:author") {
304                a.push(("w:author".into(), v.to_string()));
305            }
306            if let Some(v) = e.attr("aml:createdate") {
307                a.push(("w:date".into(), v.to_string()));
308            }
309            a
310        };
311        let content = |c: &mut Conv, part: usize| e.child("aml:content").map(|k| c.kids(k, part, depth + 1)).unwrap_or_default();
312        match e.attr("w:type").unwrap_or("") {
313            "Word.Bookmark.Start" => {
314                el("w:bookmarkStart", vec![("w:id".into(), id.clone()), ("w:name".into(), e.attr("w:name").unwrap_or("").to_string())])
315            }
316            "Word.Bookmark.End" => el("w:bookmarkEnd", vec![("w:id".into(), id)]),
317            "Word.Comment.Start" => el("w:commentRangeStart", vec![("w:id".into(), id)]),
318            "Word.Comment.End" => el("w:commentRangeEnd", vec![("w:id".into(), id)]),
319            "Word.Comment" => {
320                if self.story_count >= MAX_STORIES {
321                    return Vec::new();
322                }
323                self.story_count += 1;
324                let mut attrs = author_date(e);
325                if let Some(v) = e.attr("w:initials") {
326                    attrs.push(("w:initials".into(), v.to_string()));
327                }
328                let kids = content(self, COMMENTS);
329                self.comments.push(Node::El(El { name: "w:comment".into(), attrs, kids }));
330                el("w:commentReference", vec![("w:id".into(), id)])
331            }
332            t @ ("Word.Insertion" | "Word.Deletion") => {
333                let name = if t == "Word.Insertion" { "w:ins" } else { "w:del" };
334                vec![Node::El(El { name: name.into(), attrs: author_date(e), kids: content(self, part) })]
335            }
336            _ => Vec::new(),
337        }
338    }

340    /// `w:sectPr`: inline `w:hdr` / `w:ftr` become header and footer parts.
341    fn section(&mut self, e: &El, part: usize, depth: usize) -> El {
342        let mut refs = Vec::new();
343        let mut rest = Vec::new();
344        for n in &e.kids {
345            let Node::El(k) = n else { continue };
346            if (k.name == "w:hdr" || k.name == "w:ftr") && self.story_count < MAX_STORIES {
347                self.story_count += 1;
348                let footer = k.name == "w:ftr";
349                let kind = match k.attr("w:type") {
350                    Some("even") => "even",
351                    Some("first") => "first",
352                    _ => "default",
353                };
354                let idx = self.stories.len() + 4;
355                self.rels.push(PartRels::default());
356                let kids = self.kids(k, idx, depth + 1);
357                let file = format!("{}{}.xml", if footer { "footer" } else { "header" }, self.stories.len() + 1);
358                self.stories.push((file.clone(), El { name: k.name.clone(), attrs: xml::body_ns(), kids }));
359                let id = self.rels.get_mut(part).map(|r| r.add(if footer { rt::FOOTER } else { rt::HEADER }, &file, false)).unwrap_or_default();
360                let name = if footer { "w:footerReference" } else { "w:headerReference" };
361                refs.push(Node::El(El { name: name.into(), attrs: vec![("w:type".into(), kind.into()), ("r:id".into(), id)], kids: Vec::new() }));
362            } else if k.name != "w:hdr" && k.name != "w:ftr" {
363                rest.extend(self.node(k, part, depth + 1));
364            }
365        }
366        refs.extend(rest);
367        El { name: "w:sectPr".into(), attrs: attrs(e), kids: refs }
368    }

370    /// `w:styles` → `styles.xml`, with the document's default fonts (`w:fonts/w:defaultFonts`).
371    fn styles(&mut self, root: &El) -> El {
372        let mut kids = Vec::new();
373        if let Some(f) = root.child("w:fonts").and_then(|f| f.child("w:defaultFonts")) {
374            let fonts = El { name: "w:rFonts".into(), attrs: attrs(f), kids: Vec::new() };
375            let rpr = El { name: "w:rPr".into(), attrs: Vec::new(), kids: vec![Node::El(fonts)] };
376            let rpr_default = El { name: "w:rPrDefault".into(), attrs: Vec::new(), kids: vec![Node::El(rpr)] };
377            kids.push(Node::El(El { name: "w:docDefaults".into(), attrs: Vec::new(), kids: vec![Node::El(rpr_default)] }));
378        }
379        if let Some(s) = root.child("w:styles") {
380            for st in s.children("w:style") {
381                kids.extend(self.node(st, DOC, 2));
382            }
383        }
384        El { name: "w:styles".into(), attrs: xml::body_ns(), kids }
385    }
386
387    /// `w:lists` → `numbering.xml`: `w:listDef` → `w:abstractNum`, `w:list` → `w:num`.
388    fn numbering(&mut self, lists: &El) -> El {
389        let mut kids = Vec::new();
390        for def in lists.children("w:listDef") {
391            let Some(id) = def.attr("w:listDefId") else { continue };
392            let mut a = El { name: "w:abstractNum".into(), attrs: vec![("w:abstractNumId".into(), id.to_string())], kids: Vec::new() };
393            for k in def.els() {
394                let converted = match k.name.as_str() {
395                    "w:plt" => {
396                        let v = match k.attr("w:val").unwrap_or("") {
397                            "SingleLevel" => "singleLevel",
398                            "HybridMultilevel" => "hybridMultilevel",
399                            _ => "multilevel",
400                        };
401                        vec![Node::El(val_el("w:multiLevelType", v))]
402                    }
403                    "w:lsid" => vec![Node::El(val_el("w:nsid", k.attr("w:val").unwrap_or("")))],
404                    "w:listStyleLink" => vec![Node::El(val_el("w:numStyleLink", k.attr("w:val").unwrap_or("")))],
405                    _ => self.node(k, DOC, 2),
406                };
407                a.kids.extend(converted);
408            }
409            kids.push(Node::El(a));
410        }
411        for list in lists.children("w:list") {
412            let Some(id) = list.attr("w:ilfo") else { continue };
413            let Some(abs) = list.child_val("w:ilst") else { continue };
414            let mut n =
415                El { name: "w:num".into(), attrs: vec![("w:numId".into(), id.to_string())], kids: vec![Node::El(val_el("w:abstractNumId", abs))] };
416            for o in list.children("w:lvlOverride") {
417                n.kids.extend(self.node(o, DOC, 2));
418            }
419            kids.push(Node::El(n));
420        }
421        El { name: "w:numbering".into(), attrs: xml::body_ns(), kids }
422    }
423}

425/// `<name w:val="v"/>`.
426fn val_el(name: &str, v: &str) -> El {
427    El { name: name.into(), attrs: vec![("w:val".into(), v.into())], kids: Vec::new() }
428}

430/// `w:listPr` (`w:ilvl`, `w:ilfo`) → `w:numPr` (`w:ilvl`, `w:numId`).
431fn list_pr(e: &El) -> El {
432    let mut kids = Vec::new();
433    if let Some(v) = e.child_val("w:ilvl") {
434        kids.push(Node::El(val_el("w:ilvl", v)));
435    }
436    if let Some(v) = e.child_val("w:ilfo") {
437        kids.push(Node::El(val_el("w:numId", v)));
438    }
439    El { name: "w:numPr".into(), attrs: Vec::new(), kids }
440}

442/// `w:nfc` (a number format code) → `w:numFmt`.
443fn num_fmt(e: &El) -> El {
444    let f = match e.attr("w:val").and_then(|v| v.trim().parse::<u32>().ok()).unwrap_or(0) {
445        1 => "upperRoman",
446        2 => "lowerRoman",
447        3 => "upperLetter",
448        4 => "lowerLetter",
449        5 => "ordinal",
450        6 => "cardinalText",
451        7 => "ordinalText",
452        9 => "chicago",
453        14 | 19 => "decimalFullWidth",
454        15 => "decimalHalfWidth",
455        18 => "decimalEnclosedCircle",
456        22 => "decimalZero",
457        23 => "bullet",
458        24 => "ganada",
459        25 => "chosung",
460        26 => "decimalEnclosedFullstop",
461        27 => "decimalEnclosedParen",
462        255 => "none",
463        _ => "decimal",
464    };
465    val_el("w:numFmt", f)
466}

468/// Is an element or attribute with this name kept (a namespace the package knows, or none)?
469fn keep(name: &str) -> bool {
470    match name.split_once(':') {
471        None => true,
472        Some(("xml", _)) => true,
473        Some((p, _)) => NAMESPACES.iter().any(|(q, _)| *q == p),
474    }
475}

477/// A WordprocessingML 2003 name in its ECMA-376 spelling: `w:sz-cs` → `w:szCs`, `w:first-line`
478/// → `w:firstLine`, `w:fareast` → `w:eastAsia`, `w:vmerge` → `w:vMerge`.
479fn rename(name: &str) -> String {
480    let Some(local) = name.strip_prefix("w:") else { return name.to_string() };
481    match local {
482        "fareast" => return "w:eastAsia".into(),
483        "vmerge" => return "w:vMerge".into(),
484        "hmerge" => return "w:hMerge".into(),
485        _ => {}
486    }
487    let mut out = String::from("w:");
488    let mut up = false;
489    for c in local.chars() {
490        if c == '-' {
491            up = true;
492        } else if up {
493            out.extend(c.to_uppercase());
494            up = false;
495        } else {
496            out.push(c);
497        }
498    }
499    out
500}
501
502/// The attributes of `e` that the package keeps, renamed.
503fn attrs(e: &El) -> Vec<(String, String)> {
504    e.attrs.iter().filter(|(k, _)| keep(k)).map(|(k, v)| (rename(k), v.clone())).collect()
505}

507/// A picture's file extension: from its bytes, else from its `wordml://` name.
508fn picture_ext(b: &[u8], name: &str) -> String {
509    let sniffed = if b.starts_with(b"\x89PNG") {
510        "png"
511    } else if b.starts_with(&[0xFF, 0xD8]) {
512        "jpeg"
513    } else if b.starts_with(b"GIF8") {
514        "gif"
515    } else if b.starts_with(b"BM") {
516        "bmp"
517    } else if b.get(40..44) == Some(b" EMF") {
518        "emf"
519    } else if b.starts_with(&[0xD7, 0xCD, 0xC6, 0x9A]) || b.starts_with(&[0x01, 0x00, 0x09, 0x00]) {
520        "wmf"
521    } else {
522        ""
523    };
524    if !sniffed.is_empty() {
525        return sniffed.into();
526    }
527    let ext = name.rsplit_once('.').map(|(_, e)| e.to_ascii_lowercase()).unwrap_or_default();
528    if !ext.is_empty() && ext.len() <= 5 && ext.chars().all(|c| c.is_ascii_alphanumeric()) && ext != "xml" && ext != "rels" {
529        ext
530    } else {
531        "bin".into()
532    }
533}

535fn media_type(ext: &str) -> &'static str {
536    match ext {
537        "png" => "image/png",
538        "jpeg" | "jpg" => "image/jpeg",
539        "gif" => "image/gif",
540        "bmp" => "image/bmp",
541        "emf" => "image/x-emf",
542        "wmf" => "image/x-wmf",
543        _ => "application/octet-stream",
544    }
545}

547/// `o:DocumentProperties` → `docProps/core.xml`.
548fn core_xml(props: &El) -> Vec<u8> {
549    let mut w = W::new();
550    w.open(
551        "cp:coreProperties",
552        &[
553            ("xmlns:cp", "http://schemas.openxmlformats.org/package/2006/metadata/core-properties"),
554            ("xmlns:dc", "http://purl.org/dc/elements/1.1/"),
555            ("xmlns:dcterms", "http://purl.org/dc/terms/"),
556        ],
557    );
558    for (from, to) in [
559        ("o:Title", "dc:title"),
560        ("o:Subject", "dc:subject"),
561        ("o:Author", "dc:creator"),
562        ("o:Keywords", "cp:keywords"),
563        ("o:Description", "dc:description"),
564        ("o:LastAuthor", "cp:lastModifiedBy"),
565        ("o:Revision", "cp:revision"),
566        ("o:Category", "cp:category"),
567        ("o:Created", "dcterms:created"),
568        ("o:LastSaved", "dcterms:modified"),
569    ] {
570        if let Some(t) = props.child(from).map(El::text).filter(|t| !t.trim().is_empty()) {
571            w.leaf(to, &[], t.trim());
572        }
573    }
574    w.close("cp:coreProperties");
575    w.into_bytes()
576}
