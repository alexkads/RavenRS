1//! The File Information Block ([MS-DOC] §2.5): `FibBase` plus the variable sections
2//! (`FibRgW97`, `FibRgLw97`, the `RgFcLcb` blob, `FibRgCswNew`) at the start of the
3//! WordDocument stream.
4
5use crate::docbin::DocbinError;
6
7/// `wIdent` of every Word 97-2003 file (Word 6/95 wrote 0xA5DC).
8const W_IDENT: u16 = 0xA5EC;
9/// Offsets within the FIB, in bytes (see the spec's Fib layout).
10const OFF_FLAGS: usize = 0x0A;
11const OFF_CSW: usize = 0x20;
12const OFF_CSLW: usize = 0x3E;
13const OFF_RGLW: usize = 0x40;
14const OFF_CB_RG_FCLCB: usize = 0x98;
15/// `csw` (count of 16-bit values in `FibRgW97`) — fixed by the spec.
16const CSW: u16 = 0x000E;
17/// `cslw` (count of 32-bit values in `FibRgLw97`) — fixed by the spec.
18const CSLW: u16 = 0x0016;
19/// `cbRgFcLcb` for each known `nFib` ([MS-DOC] Fib.cbRgFcLcb table).
20const CB_RG_FCLCB: &[(u16, u16)] = &[(0x00C1, 0x005D), (0x00D9, 0x006C), (0x0101, 0x0088), (0x010C, 0x00A4), (0x0112, 0x00B7)];
21

52/// Character-position ranges of the subdocuments, from `FibRgLw97`.
53#[derive(Clone, Copy, Debug, Default, PartialEq)]
54pub(crate) struct SubDocs {
55    pub(crate) text: u32,
56    pub(crate) ftn: u32,
57    pub(crate) hdd: u32,
58    pub(crate) atn: u32,
59    pub(crate) edn: u32,
60    pub(crate) txbx: u32,
61    pub(crate) hdr_txbx: u32,
62}
63
64impl SubDocs {
65    /// Total CPs across all subdocuments, checked against a cap.
66    pub(crate) fn total(&self) -> Result<u64, DocbinError> {
67        let t = u64::from(self.text)
68            + u64::from(self.ftn)
69            + u64::from(self.hdd)
70            + u64::from(self.atn)
71            + u64::from(self.edn)
72            + u64::from(self.txbx)
73            + u64::from(self.hdr_txbx);
74        // Word's own documents stay far below this; a larger total means a corrupt FIB
75        // (the ccp fields are signed and "MUST be 0, 1, or greater").
76        if t > 100_000_000 {
77            return Err(DocbinError::Limit(format!("document claims {t} characters")));
78        }
79        Ok(t)
80    }
81}

83/// What the unencrypted head of a password-protected file says (see [`peek_encryption`]).
84#[derive(Clone, Copy, Debug)]
85pub(crate) struct Protection {
86    pub(crate) flags: u16,
87    /// `FibBase.lKey`: the XOR verifier, or the size of the RC4 header in the Table stream.
88    pub(crate) lkey: u32,
89    pub(crate) which_tbl_stm: bool,
90}
91
92/// Is the WordDocument stream that of an encrypted Word 97-2003 file? Only the first 68 bytes are
93/// stored in the clear; this reads nothing else. `None` for anything else, including streams that
94/// are too short or have a foreign magic (the full parse reports those).
95pub(crate) fn peek_encryption(word: &[u8]) -> Option<Protection> {
96    if word.len() < crate::docbin::crypt::FIB_PLAIN || word.get(0..2)? != W_IDENT.to_le_bytes() {
97        return None;
98    }
99    let flags = u16::from_le_bytes([*word.get(OFF_FLAGS)?, *word.get(OFF_FLAGS + 1)?]);
100    if flags & crate::docbin::crypt::F_ENCRYPTED == 0 {
101        return None;
102    }
103    let lkey = u32::from_le_bytes([*word.get(0x0E)?, *word.get(0x0F)?, *word.get(0x10)?, *word.get(0x11)?]);
104    Some(Protection { flags, lkey, which_tbl_stm: flags & 0x0200 != 0 })
105}

107/// The parsed FIB.
108#[derive(Clone, Debug)]
109pub(crate) struct Fib {
110    /// Selects 1Table (true) or 0Table (false).
111    pub(crate) which_tbl_stm: bool,
112    pub(crate) ccp: SubDocs,
113    /// The `RgFcLcb` pairs flattened as fc₀, lcb₀, fc₁, lcb₁, …
114    pub(crate) pairs: Vec<u32>,
115}
116
117impl Fib {
118    /// The (fc, lcb) pair at `FibRgFcLcb97` index `i` (later blobs follow contiguously).
119    pub(crate) fn pair(&self, i: usize) -> Option<(u32, u32)> {
120        let fc = *self.pairs.get(i * 2)?;
121        let lcb = *self.pairs.get(i * 2 + 1)?;
122        Some((fc, lcb))
123    }
124
125    /// Parse the FIB from the head of the WordDocument stream.
126    pub(crate) fn parse(word: &[u8]) -> Result<Fib, DocbinError> {
127        let u16le = |at: usize| -> u16 { u16::from_le_bytes([word.get(at).copied().unwrap_or(0), word.get(at + 1).copied().unwrap_or(0)]) };
128        let u32le = |at: usize| -> u32 {
129            u32::from_le_bytes([
130                word.get(at).copied().unwrap_or(0),
131                word.get(at + 1).copied().unwrap_or(0),
132                word.get(at + 2).copied().unwrap_or(0),
133                word.get(at + 3).copied().unwrap_or(0),
134            ])
135        };
136        if word.len() < OFF_RGLW + 88 {
137            return Err(DocbinError::NotWord(format!("WordDocument stream is {} bytes", word.len())));
138        }
139        let w_ident = u16le(0);
140        if w_ident == 0xA5DC {
141            return Err(DocbinError::NotWord("Word 6.0/95 format, only Word 97-2003 is supported".into()));
142        }
143        if w_ident != W_IDENT {
144            return Err(DocbinError::NotWord(format!("magic 0x{w_ident:04X}, expected 0x{W_IDENT:04X}")));
145        }
146        let flags = u16le(OFF_FLAGS);
147        let nfib = u16le(2);
148        if u16le(OFF_CSW) != CSW {
149            return Err(DocbinError::NotWord(format!("csw is 0x{:04X}", u16le(OFF_CSW))));
150        }
151        if u16le(OFF_CSLW) != CSLW {
152            return Err(DocbinError::NotWord(format!("cslw is 0x{:04X}", u16le(OFF_CSLW))));
153        }
154        // rglw u32 indices per FibRgLw97: cbMac, r1, r2, ccpText, ccpFtn, ccpHdd, r3, ccpAtn,
155        // ccpEdn, ccpTxbx, ccpHdrTxbx, …
156        let lw = |i: usize| -> u32 { u32le(OFF_RGLW + i * 4) };
157        // The ccp fields are signed in the spec; negative values are corrupt input, clamped.
158        let ccp = |i: usize| -> u32 { (lw(i) as i32).max(0) as u32 };
159        let sub = SubDocs { text: ccp(3), ftn: ccp(4), hdd: ccp(5), atn: ccp(7), edn: ccp(8), txbx: ccp(9), hdr_txbx: ccp(10) };
160        sub.total()?;
161
162        let cb_rg_fclcb = u16le(OFF_CB_RG_FCLCB) as usize;
163        if cb_rg_fclcb == 0 {
164            return Err(DocbinError::NotWord("empty RgFcLcb".into()));
165        }
166        if let Some(&(_, expected)) = CB_RG_FCLCB.iter().find(|(n, _)| *n == nfib)
167            && cb_rg_fclcb < expected as usize
168        {
169            // Word itself always writes the full blob for its version; anything smaller is
170            // corrupt, but a *larger* blob (newer writer) is fine — we just read what's there.
171            return Err(DocbinError::NotWord(format!("cbRgFcLcb is 0x{cb_rg_fclcb:04X} for nFib 0x{nfib:04X}")));
172        }
173        let blob_off = OFF_CB_RG_FCLCB + 2;
174        let blob_len = cb_rg_fclcb.saturating_mul(8);
175        if word.len() < blob_off + blob_len {
176            return Err(DocbinError::NotWord("truncated RgFcLcb".into()));
177        }
178        let mut pairs = Vec::with_capacity(cb_rg_fclcb * 2);
179        for i in 0..cb_rg_fclcb * 2 {
180            pairs.push(u32le(blob_off + i * 4));
181        }
182
183        Ok(Fib { which_tbl_stm: flags & 0x0200 != 0, ccp: sub, pairs })
184    }
