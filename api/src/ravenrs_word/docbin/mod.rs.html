1//! WordCraft reader for the Word 97-2003 binary file format (`.doc`, [MS-DOC]).
2//!
3//! [`read`] turns the bytes of a `.doc` file into a [`crate::doc::Document`]. There is no
4//! writer: saving always goes through the other formats. The reader is written from the public
5//! specification ([MS-DOC], Word (.doc) Binary File Format, Microsoft Open Specifications) and,
6//! like every WordCraft parser, is lenient by design (unknown structures are skipped, bad numbers
7//! fall back to defaults) and bounded (stream sizes, piece counts and property counts are
8//! capped), so hostile input yields an error or a best-effort document, never a panic.
9#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]

46/// Largest stream we materialise (bytes); the engine already caps whole files at 2 GB.
47const MAX_STREAM: u64 = 1 << 30;
48/// Most header/footer parts we create (matches the docx reader's cap).
49const MAX_PARTS: usize = 50_000;
50/// Hard ceiling on characters materialised across every story walk of one document (body,
51/// headers, notes, text boxes), whatever the file's size. The working budget is tied to the
52/// WordDocument stream (see [`read`]); this caps even a maximal stream.
53const MAX_TOTAL_CHARS: u64 = 64 << 20;
54/// Most paragraph and cell marks across every story walk of one document. The character
55/// budget does not see them: a mark is one byte of the file but about 1 KB of document, so a
56/// file of nothing but marks would grow memory about a thousand times faster than its size.
57/// The cap bounds a flood to about 2 GB while reading (about 4 GB at the very cap, when the
58/// blocks are assembled). Past it the read fails with [`DocbinError::Limit`]. Real documents stay far below it:
59/// 50 paragraphs on each of 10,000 pages is 500,000, and a 10,000-page one-column table (46
60/// rows, each a cell mark plus a row mark) is about 920,000.
61const MAX_PARAGRAPHS: usize = 2_000_000;
62/// Bytes of repeated work per document beyond the text itself: piece Prm grpprls (Prcs, up
63/// to 64 KB each) interpreted (#3692), hyperlink targets (#3708), picture alt text (#3709)
64/// and floating objects (#3710) copied into every run or anchor. Past it Prms are skipped,
65/// copied strings left empty and floating objects dropped, each kind logged once a document
66/// ([`Work::log`]).
67const MAX_WORK: u64 = 64 << 20;
68
69/// What the [`MAX_WORK`] budget pays for, and leaves out once it is spent.
70#[derive(Clone, Copy, Debug, PartialEq, Eq)]
71enum Cost {
72    /// A piece's Prm grpprl: past the budget the piece keeps its CHPX formatting only.
73    Formatting,
74    /// A hyperlink target copied into a run of its result: past it the text is kept, unlinked.
75    LinkTarget,
76    /// A picture's alt text copied into an anchor: past it the picture has none.
77    AltText,
78    /// A floating object copied into its anchor: past it the object is left out.
79    FloatingObject,
80}
81
82impl Cost {
83    const ALL: [Cost; 4] = [Cost::Formatting, Cost::LinkTarget, Cost::AltText, Cost::FloatingObject];
84
85    /// What is left out, for the log.
86    fn left_out(self) -> &'static str {
87        match self {
88            Cost::Formatting => "some text keeps only part of its formatting",
89            Cost::LinkTarget => "some hyperlinks' text is kept without its link",
90            Cost::AltText => "some pictures are left without their alt text",
91            Cost::FloatingObject => "some floating shapes and pictures are left out",
92        }
93    }
94}

96/// One document's [`MAX_WORK`] budget, and what it has left out (each kind logged once when
97/// the document has been read, [`Work::log`]).
98struct Work {
99    left: std::cell::Cell<u64>,
100    dropped: std::cell::Cell<[bool; 4]>,
101}
102
103impl Work {
104    fn new(budget: u64) -> Self {
105        Work { left: std::cell::Cell::new(budget), dropped: std::cell::Cell::new([false; 4]) }
106    }
107
108    /// Spends `cost` bytes on `what`: false (and `what` noted as left out) when they aren't left.
109    fn charge(&self, cost: u64, what: Cost) -> bool {
110        match self.left.get().checked_sub(cost) {
111            Some(left) => {
112                self.left.set(left);
113                true
114            }
115            None => {
116                let mut dropped = self.dropped.get();
117                if let Some(d) = Cost::ALL.iter().position(|c| *c == what).and_then(|i| dropped.get_mut(i)) {
118                    *d = true;
119                }
120                self.dropped.set(dropped);
121                false
122            }
123        }
124    }
125
126    /// The kinds of work left out so far, in [`Cost::ALL`] order.
127    fn dropped(&self) -> Vec<Cost> {
128        Cost::ALL.iter().zip(self.dropped.get()).filter(|(_, d)| *d).map(|(c, _)| *c).collect()
129    }
130
131    /// Logs one warning for each kind left out.
132    fn log(&self) {
133        let dropped = self.dropped();
134        for c in &dropped {
135            log::warn!(".doc: the document repeats more than WordCraft copies ({MAX_WORK} bytes): {}", c.left_out());
136        }
137        #[cfg(test)]
138        LAST_DROPPED.with(|d| *d.borrow_mut() = dropped);
139    }
140}

146}

156/// Errors from reading a Word 97-2003 binary file.
157#[derive(Debug, thiserror::Error, Clone, PartialEq)]
158pub enum DocbinError {
159    /// The bytes are not a readable compound file (OLE2/CFB container).
160    #[error("not a Word document (compound file): {0}")]
161    Container(String),
162    /// The compound file is readable but is not a Word 97-2003 document.
163    #[error("not a Word 97-2003 document: {0}")]
164    NotWord(String),
165    /// The document is password-protected and no password was given.
166    #[error("this document is protected with a password")]
167    PasswordRequired,
168    /// The password does not open the document.
169    #[error("the password is incorrect")]
170    WrongPassword,
171    /// The document is encrypted in a way WordCraft can't read (AES, an external provider, an
172    /// unknown header).
173    #[error("encrypted document: {0}")]
174    Encryption(String),
175    /// The input exceeds a safety limit (stream size, piece count…).
176    #[error("limit exceeded: {0}")]
177    Limit(String),
178    /// A structure is malformed beyond recovery.
179    #[error("malformed document: {0}")]
180    Malformed(String),
181}

183/// Is this `.doc` password-protected (XOR obfuscation, RC4 or RC4 CryptoAPI)? False for
184/// anything that is not a readable Word 97-2003 file.
185pub fn is_encrypted(bytes: &[u8]) -> bool {
186    let Ok(mut comp) = open_container(bytes) else { return false };
187    stream(&mut comp, "WordDocument").is_some_and(|w| fib::peek_encryption(&w).is_some())
188}

190/// Open the compound file around a `.doc`, after [`crate::doc::cfb_guard::check`].
191fn open_container(bytes: &[u8]) -> Result<cfb::CompoundFile<std::io::Cursor<&[u8]>>, DocbinError> {
192    crate::doc::cfb_guard::check(bytes).map_err(DocbinError::Container)?;
193    cfb::CompoundFile::open(std::io::Cursor::new(bytes)).map_err(|e| DocbinError::Container(e.to_string()))
194}

196/// Read a `.doc` (Word 97-2003 binary) file into a [`Document`]. A password-protected file
197/// gives [`DocbinError::PasswordRequired`]; see [`read_with_password`].
198pub fn read(bytes: &[u8]) -> Result<Document, DocbinError> {
199    read_with_password(bytes, None)
200}

202/// [`read`] for a file that may be password-protected: it is checked against the file's stored
203/// verifier ([`DocbinError::WrongPassword`] when it does not match) and then decrypted. The
204/// password is ignored for an unprotected file.
205pub fn read_with_password(bytes: &[u8], password: Option<&str>) -> Result<Document, DocbinError> {
206    read_limited(bytes, password, MAX_PARAGRAPHS)
207}

209/// [`read_with_password`] with the paragraph cap as a parameter, so tests need not build a
210/// file of [`MAX_PARAGRAPHS`] marks.
211fn read_limited(bytes: &[u8], password: Option<&str>, max_paragraphs: usize) -> Result<Document, DocbinError> {
212    let mut comp = open_container(bytes)?;
213    let mut word = stream(&mut comp, "WordDocument").ok_or_else(|| DocbinError::NotWord("no WordDocument stream".into()))?;
214    if word.len() as u64 > MAX_STREAM {
215        return Err(DocbinError::Limit(format!("WordDocument stream is {} bytes", word.len())));
216    }
217    let enc = fib::peek_encryption(&word);
218    // The FIB names the Table stream (0Table or 1Table); fall back to whichever exists,
219    // since files written by other writers sometimes carry the wrong bit.
220    let which = if let Some(e) = enc { e.which_tbl_stm } else { fib::Fib::parse(&word)?.which_tbl_stm };
221    let named = if which { "1Table" } else { "0Table" };
222    let other = if which { "0Table" } else { "1Table" };
223    let mut table = stream(&mut comp, named).or_else(|| stream(&mut comp, other)).unwrap_or_default();
224    if table.len() as u64 > MAX_STREAM {
225        return Err(DocbinError::Limit(format!("Table stream is {} bytes", table.len())));
226    }
227    let mut data = stream(&mut comp, "Data").unwrap_or_default();
228    if data.len() as u64 > MAX_STREAM {
229        return Err(DocbinError::Limit(format!("Data stream is {} bytes", data.len())));
230    }
231    if let Some(e) = enc {
232        // Verify the password first; only then is anything decrypted.
233        let password = password.ok_or(DocbinError::PasswordRequired)?;
234        let cipher = crypt::Cipher::open(e.flags, e.lkey, &table, password)?;
235        cipher.decrypt_word(&mut word);
236        cipher.decrypt_table(&mut table);
237        cipher.decrypt_data(&mut data);
238    }
239    let fib = fib::Fib::parse(&word)?;
240
241    // The piece table locates every character in the WordDocument stream.
242    let (fc_clx, lcb_clx) =
243        fib.pair(fib::pair::CLX).filter(|(_, lcb)| *lcb > 0).ok_or_else(|| DocbinError::Malformed("the FIB has no piece table (Clx)".into()))?;
244    let pieces = piece::PieceTable::parse(&table, fc_clx, lcb_clx)?;
245
246    // Styles, fonts and the formatting bin tables; each is optional in broken files.
247    let fonts = fib.pair(fib::pair::STTBF_FFN).map(|(f, l)| stsh::fonts(&table, f, l)).unwrap_or_default();
248    let raw_styles = match fib.pair(fib::pair::STSHF) {
249        Some((f, l)) => stsh::parse(&table, f, l)?,
250        None => Vec::new(),
251    };
252    let standard_fonts = fib.pair(fib::pair::STSHF).map(|(f, l)| stsh::standard_fonts(&table, f, l)).unwrap_or_default();
253    let chpx_bins = match fib.pair(fib::pair::PLCF_BTE_CHPX) {
254        Some((f, l)) => Bins::parse(&table, f, l)?,
255        None => Bins::parse(&[], 0, 0)?,
256    };
257    let papx_bins = match fib.pair(fib::pair::PLCF_BTE_PAPX) {
258        Some((f, l)) => Bins::parse(&table, f, l)?,
259        None => Bins::parse(&[], 0, 0)?,
260    };
261
262    let mut doc = Document::new();
263    // Word opens Word 97-2003 files in compatibility mode 11 (Word 2003); the model's default
264    // 15 would apply Word 2013+ justification and line-breaking rules to them.
265    doc.settings.compat_mode = 11;
266    if let Some(core) = metadata::read(&mut comp) {
267        doc.core = core;
268    }
269    doc.body.clear();
270    doc.styles = Arc::new(fmt::stylesheet(&raw_styles, &fonts, standard_fonts));
271    doc.numbering = Arc::new(list::parse(&table, &fib, &fonts));
272    let sheet: StyleSheet = StyleSheet::clone(&doc.styles);
273    let sects = sections::parse(&word, &table, &fib);
274    let stories = sections::header_stories(&table, &fib);
275    // Subdocument layout in the CP space: main document, footnotes, headers, comments,
276    // endnotes, text boxes, header text boxes.
277    // Story ranges from the Table stream are hostile: each is clamped to its subdocument's
278    // length here and to the last piece's CP inside `walk`.
279    let ftn_base = fib.ccp.text;
280    let hdd_base = fib.ccp.text.saturating_add(fib.ccp.ftn);
281    let edn_base = hdd_base.saturating_add(fib.ccp.hdd).saturating_add(fib.ccp.atn);
282    let mut pics = media::Pictures::for_input(word.len().saturating_add(data.len()));
283    // Every materialised character comes from at least one byte of the WordDocument stream, so a
284    // legitimate document's total text (body plus every header, footnote and text box, each read
285    // once) cannot exceed its stream. A hostile piece or story table that aliases the same bytes
286    // many times can, so the walk is held to twice the stream's size — generous for real files,
287    // tight against aliasing — and never past the absolute ceiling.
288    let budget = std::cell::Cell::new((word.len() as u64).saturating_mul(2).min(MAX_TOTAL_CHARS));
289    let work = Work::new(MAX_WORK);
290    let paragraphs = ParaBudget { left: std::cell::Cell::new(max_paragraphs), exceeded: std::cell::Cell::new(false) };
291
292    let ctx = WalkCtx {
293        word: &word,
294        data: &data,
295        pieces: &pieces,
296        chpx_bins: &chpx_bins,
297        papx_bins: &papx_bins,
298        fonts: &fonts,
299        raw_styles: &raw_styles,
300        sheet: &sheet,
301        notes: &[],
302        marks: &[],
303        drawn: &[],
304        sect_ends: &[],
305        budget: &budget,
306        paragraphs: &paragraphs,
307        work: &work,
308    };
309
310    // Drawings: text box stories become parts first, then the floating shapes and pictures
311    // anchored in the main document and the headers, found by their anchor CP while walking.
312    let txbx_base = edn_base.saturating_add(fib.ccp.edn);
313    let hdr_txbx_base = txbx_base.saturating_add(fib.ccp.txbx);
314    let art = fib
315        .pair(fib::pair::DGG_INFO)
316        .and_then(|(f, l)| table.get(f as usize..(f as usize).saturating_add(l as usize)))
317        .map(officeart::content)
318        .unwrap_or_default();
319    let mut tb_stories = std::collections::HashMap::new();
320    for (pair, base, len) in [(fib::pair::PLCF_TXBX_TXT, txbx_base, fib.ccp.txbx), (fib::pair::PLCF_HDR_TXBX_TXT, hdr_txbx_base, fib.ccp.hdr_txbx)] {
321        for (spid, ab) in drawing::textbox_stories(&table, fib.pair(pair)) {
322            let Some((a, b)) = story(base, len, ab) else { continue };
323            if doc.parts.len() >= MAX_PARTS || tb_stories.contains_key(&spid) {
324                continue;
325            }
326            let blocks = story_blocks(&ctx, a, b, &mut pics, max_paragraphs)?;
327            tb_stories.insert(spid, doc.add_part(PartKind::TextBox, blocks));
328        }
329    }
330    let build = drawing::Build { bstore: &art.bstore, delay: &word, stories: &tb_stories };
331    let mut drawn = drawing::anchors(&table, fib.pair(fib::pair::PLC_SPA_MOM), 0, &art.main, &build, &mut pics);
332    drawn.extend(drawing::anchors(&table, fib.pair(fib::pair::PLC_SPA_HDR), hdd_base, &art.header, &build, &mut pics));
333    drawn.sort_by_key(|(cp, _)| *cp);
334    let ctx = WalkCtx { drawn: &drawn, ..ctx };
335
336    // Header and footer parts: skip the six separator stories, then six stories per section
337    // in the order even header, odd (default) header, even footer, odd (default) footer,
338    // first header, first footer. Empty stories inherit from the previous section.
339    let mut hf_ids: Vec<(usize, usize, u32)> = Vec::new();
340    for si in 0..sects.len() {
341        for k in 0..6usize {
342            let Some(&ab) = stories.get(6 + si * 6 + k) else { break };
343            let Some((a, b)) = story(hdd_base, fib.ccp.hdd, ab) else { continue };
344            if doc.parts.len() >= MAX_PARTS {
345                continue;
346            }
347            let kind = match k {
348                0 | 1 | 4 => PartKind::Header,
349                _ => PartKind::Footer,
350            };
351            let blocks = header_story_blocks(&ctx, a, b, &mut pics, max_paragraphs)?;
352            if !blocks.is_empty() {
353                let id = doc.add_part(kind, blocks);
354                hf_ids.push((si, k, id));
355            }
356        }
357    }
358
359    // Footnote and endnote stories: each text range becomes a part; the reference CPs then
360    // link to the part ids during the main walk.
361    let mut note_links: Vec<(u32, NoteKind, u32)> = Vec::new();
362    for (cp, ab) in notes::parse_notes(&table, &fib, fib::pair::PLCF_FND_REF, fib::pair::PLCF_FND_TXT) {
363        let Some((a, b)) = story(ftn_base, fib.ccp.ftn, ab) else { continue };
364        let blocks = story_blocks(&ctx, a, b, &mut pics, max_paragraphs)?;
365        if !blocks.is_empty() && doc.parts.len() < MAX_PARTS {
366            let id = doc.add_part(PartKind::Footnote, blocks);
367            note_links.push((cp, NoteKind::Footnote, id));
368        }
369    }
370    for (cp, ab) in notes::parse_notes(&table, &fib, fib::pair::PLCF_END_REF, fib::pair::PLCF_END_TXT) {
371        let Some((a, b)) = story(edn_base, fib.ccp.edn, ab) else { continue };
372        let blocks = story_blocks(&ctx, a, b, &mut pics, max_paragraphs)?;
373        if !blocks.is_empty() && doc.parts.len() < MAX_PARTS {
374            let id = doc.add_part(PartKind::Endnote, blocks);
375            note_links.push((cp, NoteKind::Endnote, id));
376        }
377    }
378    // Sorted so the main walk finds each reference by binary search.
379    note_links.sort_by_key(|(cp, _, _)| *cp);
380    let marks = notes::parse_bookmarks(&table, &fib);
381    let mut sect_ends: Vec<u32> = sects.iter().map(|s| s.end_cp).collect();
382    sect_ends.sort_unstable();
383    let ctx = WalkCtx { notes: &note_links, marks: &marks, sect_ends: &sect_ends, ..ctx };
384
385    // Sections: each section's properties attach to the paragraph that ends it (its story of
386    // header/footer parts included); the section reaching the end of the text is the final one.
387    let mut paras = walk(&ctx, 0, fib.ccp.text, &mut pics);
388    // The first paragraph ending at each CP, found once rather than by a scan per section.
389    let mut ending: std::collections::HashMap<u32, usize> = std::collections::HashMap::new();
390    for (i, p) in paras.iter().enumerate() {
391        ending.entry(p.end_cp).or_insert(i);
392    }
393    let last = sects.last().map(|s| s.props.clone());
394    for (si, sec) in sects.iter().enumerate() {
395        let mut props = sec.props.clone();
396        let id_of = |k: usize| hf_ids.iter().find(|(s, kk, _)| *s == si && *kk == k).map(|(_, _, id)| *id);
397        props.headers = HeaderSet { even: id_of(0), default: id_of(1), first: id_of(4) };
398        props.footers = HeaderSet { even: id_of(2), default: id_of(3), first: id_of(5) };
399        let is_final = si + 1 == sects.len() || sec.end_cp >= fib.ccp.text;
400        if is_final {
401            continue; // handled below through `last`
402        }
403        if let Some(p) = ending.get(&sec.end_cp).and_then(|&i| paras.get_mut(i)) {
404            p.para.section = Some(Box::new(props));
405        }
406    }
407    if let Some(mut l) = last {
408        let si = sects.len().saturating_sub(1);
409        let id_of = |k: usize| hf_ids.iter().find(|(s, kk, _)| *s == si && *kk == k).map(|(_, _, id)| *id);
410        l.headers = HeaderSet { even: id_of(0), default: id_of(1), first: id_of(4) };
411        l.footers = HeaderSet { even: id_of(2), default: id_of(3), first: id_of(5) };
412        doc.last_section = l;
413    }
414
415    if paragraphs.exceeded.get() {
416        return Err(paragraph_limit(max_paragraphs));
417    }
418    doc.media.extend(pics.media);
419    doc.body = table::assemble(paras);
420    doc.ensure_nonempty();
421    work.log();
422    Ok(doc)
423}

425fn paragraph_limit(max_paragraphs: usize) -> DocbinError {
426    DocbinError::Limit(format!("the document has more than {max_paragraphs} paragraphs and table cells"))
427}

429/// Walks one story into blocks. A walk that spent the shared paragraph budget is refused before
430/// its partial list is assembled, so a flood does not build blocks it will throw away.
431fn story_blocks(ctx: &WalkCtx<'_>, a: u32, b: u32, pics: &mut media::Pictures, max_paragraphs: usize) -> Result<crate::doc::Blocks, DocbinError> {
432    let paras = walk(ctx, a, b, pics);
433    if ctx.paragraphs.exceeded.get() {
434        return Err(paragraph_limit(max_paragraphs));
435    }
436    Ok(table::assemble(paras))
437}

439/// A header or footer story's blocks without its guard paragraph mark. [MS-DOC] 2.3.3: a
440/// non-empty header story ends with a paragraph mark that only separates it from the next story
441/// and is not part of its contents, so a one-line header is stored as `text\r\r`. Laid out, the
442/// guard would add an empty paragraph to every header and footer and push the body down when
443/// the header reaches past the top margin (#5922).
444///
445/// Only an empty last paragraph after an ordinary one is dropped: a story that is nothing but a
446/// mark stays a part (an empty header, not one linked to the previous section), and a mark right
447/// after a table row stays the paragraph that closes the table.
448fn header_story_blocks(
449    ctx: &WalkCtx<'_>,
450    a: u32,
451    b: u32,
452    pics: &mut media::Pictures,
453    max_paragraphs: usize,
454) -> Result<crate::doc::Blocks, DocbinError> {
455    let mut paras = walk(ctx, a, b, pics);
456    if ctx.paragraphs.exceeded.get() {
457        return Err(paragraph_limit(max_paragraphs));
458    }
459    let outside_table = |p: &table::ParaOut| p.row.as_ref().is_none_or(|r| !r.in_table && !r.ttp && !r.cell_mark);
460    if let [.., prev, last] = paras.as_slice() {
461        let guard = outside_table(last) && last.para.text.is_empty() && last.para.objects.is_empty() && last.para.section.is_none();
462        if guard && outside_table(prev) {
463            paras.pop();
464        }
465    }
466    Ok(table::assemble(paras))
467}

469/// A story's CP range `[a, b)`, relative to a subdocument starting at `base` with `len`
470/// CPs, as absolute CPs clamped to that subdocument; `None` when empty.
471fn story(base: u32, len: u32, (a, b): (u32, u32)) -> Option<(u32, u32)> {
472    let (a, b) = (a.min(len), b.min(len));
473    if b <= a {
474        return None;
475    }
476    Some((base.checked_add(a)?, base.checked_add(b)?))
477}

479/// Everything the walker needs besides its CP range.
480struct WalkCtx<'a> {
481    word: &'a [u8],
482    data: &'a [u8],
483    pieces: &'a PieceTable,
484    chpx_bins: &'a Bins,
485    papx_bins: &'a Bins,
486    fonts: &'a [String],
487    raw_styles: &'a [stsh::RawStyle],
488    sheet: &'a StyleSheet,
489    /// (reference CP in the main document, kind, part id) for notes.
490    notes: &'a [(u32, NoteKind, u32)],
491    /// Bookmark boundaries (main-document CPs), sorted for a sequential walk.
492    marks: &'a [(u32, notes::Mark)],
493    /// Floating shapes and pictures by anchor CP (absolute), sorted.
494    drawn: &'a [(u32, InlineObject)],
495    /// Where sections end (main-document CPs, sorted): the section mark (0x0C) just before
496    /// each ends the section's last paragraph, it isn't a page break inside it.
497    sect_ends: &'a [u32],
498    /// Characters still allowed across every story walk of this document. Story ranges (headers,
499    /// footnotes, text boxes) are hostile and can each span a whole subdocument, so thousands of
500    /// them could otherwise materialise the same text thousands of times; this bounds the total.
501    budget: &'a std::cell::Cell<u64>,
502    /// Paragraph and cell marks still allowed across every story walk ([`MAX_PARAGRAPHS`]).
503    paragraphs: &'a ParaBudget,
504    /// Repeated-work bytes left ([`MAX_WORK`]), and what they left out.
505    work: &'a Work,
506}
507
508/// The paragraph and cell marks still allowed in this document, and whether a walk ran into
509/// the end of them (the read then fails).
510struct ParaBudget {
511    left: std::cell::Cell<usize>,
512    exceeded: std::cell::Cell<bool>,
513}
514
515impl ParaBudget {
516    /// Spends one mark; false (and the overrun noted) when none are left.
517    fn spend(&self) -> bool {
518        match self.left.get().checked_sub(1) {
519            Some(left) => {
520                self.left.set(left);
521                true
522            }
523            None => {
524                self.exceeded.set(true);
525                false
526            }
527        }
528    }
529}

531/// State while assembling one paragraph from per-CP formatting.
532#[derive(Default)]
533struct ParaBuild {
534    text: String,
535    runs: Vec<Run>,
536    props: crate::doc::props::ParaProps,
537    mark: CharProps,
538    /// Which resolved formatting ([`FmtState::generation`]) the last character came from, while
539    /// the last run is still the one it extended: the same generation means equal properties.
540    run_gen: Option<u64>,
541    /// Inline objects, each anchored at a U+FFFC pushed into `text`.
542    objects: Vec<InlineObject>,
543}
544
545impl ParaBuild {
546    fn push(&mut self, c: char, props: CharProps) {
547        self.run_gen = None;
548        let len = c.len_utf8();
549        match self.runs.last_mut() {
550            Some(r) if r.props == props => r.len += len,
551            _ => self.runs.push(Run { len, props }),
552        }
553        self.text.push(c);
554    }
555
556    /// [`ParaBuild::push`] for properties that are resolved once and shared by many characters:
557    /// they are cloned only when a run starts, and not even compared while `generation` stays
558    /// the same.
559    fn push_shared(&mut self, c: char, props: &CharProps, generation: u64) {
560        let len = c.len_utf8();
561        match self.runs.last_mut() {
562            Some(r) if self.run_gen == Some(generation) || r.props == *props => r.len += len,
563            _ => self.runs.push(Run { len, props: props.clone() }),
564        }
565        self.run_gen = Some(generation);
566        self.text.push(c);
567    }
568
569    /// Append `c` to the current run, whose properties the caller knows it matches.
570    fn extend(&mut self, c: char) {
571        if let Some(r) = self.runs.last_mut() {
572            r.len += c.len_utf8();
573        }
574        self.text.push(c);
575    }
576}

578/// Walk a CP range, splitting paragraphs at 0x0D/0x07 marks, applying the direct character
579/// formatting of each CP and the paragraph formatting of each mark. Works for the main
580/// document and for any subdocument range (headers, footers, notes…). Field characters,
581/// note references, bookmarks and inline pictures become inline objects; pictures go into
582/// `pics`, which shares one media entry per Data-stream location.
583///
584/// The range is clamped to the last piece's CP, and unreadable stretches (gaps, truncated
585/// pieces) are skipped a whole piece at a time, so the work is bounded by the text that
586/// really exists, not by the CP numbers a file claims.
587fn walk<'a>(ctx: &WalkCtx<'a>, cp_start: u32, cp_end: u32, pics: &mut media::Pictures) -> Vec<ParaOut> {
588    let cp_end = cp_end.min(ctx.pieces.cp_end());
589    let mut out = Vec::new();
590    let mut pb = ParaBuild::default();
591    let mut cp = cp_start;
592    let mut marks = ctx.marks.iter().peekable();
593    // Field assembly state: instruction characters are swallowed, the result is kept until
594    // the end character, then emitted as one object (hyperlinks keep their result text).
595    let mut field: Option<FieldBuild> = None;
596    let mut fmt = FmtState::default();
597    while cp < cp_end {
598        // Bookmark boundaries at this CP (main-document walks only; subdocument walks get
599        // an empty table and `cp` never matches the main-document CPs).
600        // `<=` so a boundary inside a skipped stretch is still emitted (late) rather than
601        // blocking every later one.
602        while marks.peek().is_some_and(|&(mcp, _)| *mcp <= cp) {
603            if let Some((_, m)) = marks.next() {
604                let obj = match m {
605                    notes::Mark::Start(name) => InlineObject::BookmarkStart { name: name.clone() },
606                    notes::Mark::End(name) => InlineObject::BookmarkEnd { name: name.clone() },
607                };
608                pb.push('\u{FFFC}', CharProps::default());
609                pb.objects.push(obj);
610            }
611        }
612        let (c, units) = match ctx.pieces.char_at(ctx.word, cp) {
613            Ok(found) => found,
614            Err(next) => {
615                cp = next;
616                continue;
617            }
618        };
619        // Every real character counts against the document-wide budget; once it is spent, stop
620        // (a hostile file with thousands of story ranges over the same text can't multiply on).
621        if ctx.budget.get() == 0 {
622            break;
623        }
624        ctx.budget.set(ctx.budget.get() - 1);
625        // A section mark (0x0C) at a section's end is that section's last paragraph mark.
626        let section_mark = c == '\u{C}' && ctx.sect_ends.binary_search(&cp.saturating_add(1)).is_ok();
627        match c {
628            '\r' | '\u{7}' | '\u{C}' if c != '\u{C}' || section_mark => {
629                if !ctx.paragraphs.spend() {
630                    break;
631                }
632                // Paragraph mark: its PAPX covers the FC range of the paragraph it ends, so
633                // querying at the mark's own FC finds that paragraph's properties.
634                let fc_mark = ctx.pieces.fc_of_cp(cp).unwrap_or(0);
635                let (istd, papx) = ctx.papx_bins.papx(ctx.word, fc_mark);
636                if let Some(st) = ctx.raw_styles.get(istd as usize).filter(|s| !s.name.is_empty()) {
637                    pb.props.style = Some(fmt::style_id(&st.name));
638                }
639                let mut row = table::decode(papx);
640                row.cell_mark = c == '\u{7}';
641                // Only table marks keep their table context; a plain paragraph would carry
642                // a `RowInfo` (about 300 bytes) it never uses.
643                let row = row.is_table().then(|| Box::new(row));
644                let mut ilvl = 0u8;
645                let mut ilfo = 0i32;
646                for prl in sprm::iter(papx) {
647                    if prl.op == P_ILVL {
648                        ilvl = prl.operand.first().copied().unwrap_or(0).min(8);
649                        continue;
650                    }
651                    if prl.op == P_ILFO {
652                        ilfo = match prl.operand {
653                            [b0, b1, ..] => i16::from_le_bytes([*b0, *b1]) as i32,
654                            _ => 0,
655                        };
656                        continue;
657                    }
658                    if prl.sgc() == 1 {
659                        fmt::apply_para(&mut pb.props, &prl);
660                    }
661                }
662                if let Some(num) = num_of(ilfo) {
663                    pb.props.numbering = Some(NumRef { num, level: ilvl });
664                }
665                let fc = ctx.pieces.fc_of_cp(cp).unwrap_or(0);
666                pb.mark = char_props(ctx.chpx_bins.chpx(ctx.word, fc), ctx.fonts, &CharProps::default());
667                let done = std::mem::take(&mut pb);
668                out.push(ParaOut {
669                    para: Paragraph {
670                        text: done.text,
671                        runs: done.runs,
672                        props: done.props,
673                        mark: done.mark,
674                        objects: done.objects,
675                        ..Default::default()
676                    },
677                    end_cp: cp.saturating_add(1),
678                    row,
679                });
680            }
681            // Field begin/separator/end ([MS-DOC] §2.9.84 Plcfld).
682            '\u{13}' => {
683                match field.as_mut() {
684                    Some(f) => {
685                        f.depth = f.depth.saturating_add(1);
686                    }
687                    None => field = Some(FieldBuild { depth: 1, instr: String::new(), sep: None }),
688                }
689                cp = cp.saturating_add(1);
690                continue;
691            }
692            '\u{14}' => {
693                if let Some(f) = field.as_mut()
694                    && f.depth == 1
695                    && f.sep.is_none()
696                {
697                    f.sep = Some(FieldResult::default());
698                }
699                cp = cp.saturating_add(1);
700                continue;
701            }
702            '\u{15}' => {
703                if let Some(mut f) = field.take() {
704                    if f.depth > 1 {
705                        f.depth -= 1;
706                        field = Some(f);
707                    } else {
708                        finish_field(&mut pb, f, ctx.work);
709                    }
710                }
711                cp = cp.saturating_add(1);
712                continue;
713            }
714            '\u{1}' => {
715                // Inline picture: the CHPX of the anchor carries sprmCPicLocation.
716                let fc = ctx.pieces.fc_of_cp(cp).unwrap_or(0);
717                let fc_pic = pic_location(ctx.chpx_bins.chpx(ctx.word, fc));
718                match fc_pic.and_then(|at| pics.get(ctx.data, at)) {
719                    Some(p) => {
720                        // Every anchor of a picture gets its own copy of the alt text; the
721                        // copies are charged to the work budget (#3709).
722                        let alt = if ctx.work.charge(p.alt.len() as u64, Cost::AltText) { p.alt.clone() } else { String::new() };
723                        pb.push('\u{FFFC}', CharProps::default());
724                        pb.objects.push(InlineObject::Image {
725                            media: p.key.clone(),
726                            w: p.w,
727                            h: p.h,
728                            alt,
729                            float: Default::default(),
730                            crop: p.crop,
731                            fx: Default::default(),
732                            ole: None,
733                        });
734                    }
735                    None => pics.dropped(cp),
736                }
737            }
738            '\u{8}' => {
739                // Drawn object anchor: the floating shape or picture anchored at this CP.
740                let at = ctx.drawn.partition_point(|(c, _)| *c < cp);
741                // Each copy (a story can be walked more than once) is charged to the work budget
742                // by what a clone copies: a picture's alt text, a group with all its members (#3710).
743                for (_, obj) in ctx.drawn.get(at..).unwrap_or_default().iter().take_while(|(c, _)| *c == cp) {
744                    if !ctx.work.charge(object_cost(obj), Cost::FloatingObject) {
745                        continue;
746                    }
747                    pb.push('\u{FFFC}', CharProps::default());
748                    pb.objects.push(obj.clone());
749                }
750            }
751            '\u{2}' => {
752                // Footnote or endnote reference; custom symbols share the same character.
753                let at = ctx.notes.partition_point(|(c, _, _)| *c < cp);
754                if let Some((_, kind, id)) = ctx.notes.get(at).filter(|(c, _, _)| *c == cp) {
755                    pb.push('\u{FFFC}', CharProps::default());
756                    pb.objects.push(InlineObject::NoteRef { kind: *kind, id: *id, custom: String::new() });
757                }
758            }
759            '\t' => push_formatted(&mut pb, cp, '\t', ctx, field.as_mut(), &mut fmt),
760            '\u{B}' => push_formatted(&mut pb, cp, '\n', ctx, field.as_mut(), &mut fmt),
761            '\u{C}' => push_formatted(&mut pb, cp, '\u{C}', ctx, field.as_mut(), &mut fmt),
762            '\u{E}' => push_formatted(&mut pb, cp, '\u{E}', ctx, field.as_mut(), &mut fmt),
763            '\u{1E}' => push_formatted(&mut pb, cp, '\u{2011}', ctx, field.as_mut(), &mut fmt),
764            '\u{1F}' => push_formatted(&mut pb, cp, '\u{AD}', ctx, field.as_mut(), &mut fmt),
765            // Object anchors, comment references and other special characters are dropped
766            // (or handled by later passes).
767            c if (c as u32) < 0x20 => {}
768            c => push_formatted(&mut pb, cp, c, ctx, field.as_mut(), &mut fmt),
769        }
770        cp = cp.saturating_add(units);
771    }
772    if !pb.text.is_empty() {
773        let done = pb;
774        out.push(ParaOut {
775            para: Paragraph { text: done.text, runs: done.runs, props: done.props, mark: done.mark, objects: done.objects, ..Default::default() },
776            end_cp: cp_end,
777            row: None,
778        });
779    }
780    out
781}

783/// Field assembly: `instr` collects everything before the separator (nested fields
784/// flattened), `sep` holds the cached result once the separator was seen.
785struct FieldBuild {
786    depth: u32,
787    instr: String,
788    sep: Option<FieldResult>,
789}
790
791/// A field's cached result: its text and, as runs, the formatting of that text. A run is added
792/// only when the formatting changes, so a result of millions of equally formatted characters
793/// costs one `CharProps`, not one per character (365 bytes each).
794#[derive(Default)]
795struct FieldResult {
796    text: String,
797    /// (byte length in `text`, formatting) in order; adjacent runs differ.
798    runs: Vec<(usize, CharProps)>,
799}
800
801impl FieldResult {
802    fn push(&mut self, c: char, props: CharProps) {
803        let len = c.len_utf8();
804        match self.runs.last_mut() {
805            Some((n, last)) if *last == props => *n += len,
806            _ => self.runs.push((len, props)),
807        }
808        self.text.push(c);
809    }
810}
811
812/// Bytes a clone of `obj` copies: the object itself plus a picture's alt text, a group with all
813/// its members (#3710).
814fn object_cost(obj: &InlineObject) -> u64 {
815    let shallow = std::mem::size_of::<InlineObject>() as u64;
816    match obj {
817        InlineObject::Image { alt, .. } => shallow.saturating_add(alt.len() as u64),
818        InlineObject::Group { children, .. } => children.iter().fold(shallow, |n, c| n.saturating_add(object_cost(&c.obj))),
819        _ => shallow,
820    }
821}
822
823/// Emit a finished field: HYPERLINK keeps its result text as a linked run; every other
824/// field becomes one inline object with its cached result.
825///
826/// The target is copied once per run of equally formatted result text, not per character,
827/// and each copy is charged to `work` ([`MAX_WORK`]); once that is spent the rest of the
828/// result is kept as plain text (#3708).
829fn finish_field(pb: &mut ParaBuild, f: FieldBuild, work: &Work) {
830    let instr = f.instr.trim().to_string();
831    let result = f.sep.unwrap_or_default();
832    if let Some(target) = hyperlink_target(&instr) {
833        let cost = target.len() as u64;
834        let mut chars = result.text.chars();
835        for (len, props) in result.runs {
836            let mut linked = props;
837            if work.charge(cost, Cost::LinkTarget) {
838                linked.link = Some(target.clone());
839            }
840            let mut left = len;
841            let Some(first) = chars.next() else { break };
842            pb.push(first, linked);
843            left = left.saturating_sub(first.len_utf8());
844            while left > 0 {
845                let Some(c) = chars.next() else { break };
846                pb.extend(c);
847                left = left.saturating_sub(c.len_utf8());
848            }
849        }
850        return;
851    }
852    pb.push('\u{FFFC}', CharProps::default());
853    pb.objects.push(InlineObject::Field { instr, result: result.text, locked: false, dirty: false, code: false, form: String::new() });
854}

856/// `sprmCPicLocation` from a CHPX grpprl: the fc of the picture in the Data stream.
857fn pic_location(chpx: &[u8]) -> Option<u32> {
858    for prl in sprm::iter(chpx) {
859        if prl.op == C_PIC_LOCATION {
860            return match prl.operand {
861                [b0, b1, b2, b3, ..] => Some(u32::from_le_bytes([*b0, *b1, *b2, *b3])),
862                _ => None,
863            };
864        }
865    }
866    None
867}

869/// `sprmCSymbol` from a CHPX grpprl: the font table index and the character code of a symbol
870/// character ([MS-DOC] §2.6.1, CSymbolOperand).
871fn symbol_operand(chpx: &[u8]) -> Option<(u16, u16)> {
872    for prl in sprm::iter(chpx) {
873        if prl.op == C_SYMBOL
874            && let [f0, f1, c0, c1, ..] = prl.operand
875        {
876            return Some((u16::from_le_bytes([*f0, *f1]), u16::from_le_bytes([*c0, *c1])));
877        }
878    }
879    None
880}
881
882/// `sprmPIlfo` operand (a signed 16-bit value, [MS-DOC] §2.6.2) → the num id, if the
883/// paragraph is in a list. 0x0001-0x07FE is a 1-based LFO index; 0xF802-0xFFFF (-2046..=-1)
884/// is the negation of one and keeps the paragraph's own indents, which we do by leaving the
885/// level's indents out of the paragraph (they never enter `ParaProps` anyway). 0 and 0xF801
886/// mean "not in a list".
887fn num_of(ilfo: i32) -> Option<u32> {
888    match ilfo {
889        0x0001..=0x07FE | -0x07FE..=-1 => Some(ilfo.unsigned_abs()),
890        _ => None,
891    }
892}

894/// Resolve the character properties for the character at `cp`: the CHPX grpprl of its FC,
895/// plus the piece's Prm, over the paragraph style's base character props. Inside a field,
896/// the character is recorded into the instruction or the cached result instead of (or as
897/// well as) the visible text.
898fn push_formatted<'a>(pb: &mut ParaBuild, cp: u32, c: char, ctx: &WalkCtx<'a>, field: Option<&mut FieldBuild>, st: &mut FmtState<'a>) {
899    // The piece and the CHPX run of the previous character usually cover this one too.
900    let span = match st.span.filter(|s| s.contains(cp)) {
901        Some(s) => Some(s),
902        None => {
903            st.span = ctx.pieces.span(cp);
904            st.span
905        }
906    };
907    let fc = span.and_then(|s| s.fc_at(cp)).unwrap_or(0);
908    let run = match st.chpx.filter(|r| r.contains(fc)) {
909        Some(r) => r,
910        None => {
911            let r = ctx.chpx_bins.chpx_run(ctx.word, fc);
912            st.chpx = Some(r);
913            r
914        }
915    };
916    let grpprl = run.grpprl;
917    let prm = span.map_or(0, |s| s.prm());
918    let style = pb.props.style.as_deref();
919    let hit = st.memo.as_ref().is_some_and(|m| std::ptr::eq(m.chpx, grpprl) && m.prm == prm && m.style.as_deref() == style);
920    if !hit {
921        let default;
922        let base = match style.and_then(|id| ctx.sheet.get(id)) {
923            Some(sty) => &sty.chr,
924            None => {
925                default = CharProps::default();
926                &default
927            }
928        };
929        let mut props = char_props(grpprl, ctx.fonts, base);
930        apply_prm(&mut props, ctx.pieces.piece_prm(cp), ctx.fonts, base, ctx.work);
931        st.memo = Some(FmtMemo { chpx: grpprl, prm, style: style.map(str::to_owned), props });
932        st.generation += 1;
933    }
934    let Some(m) = st.memo.as_ref() else { return };
935    // A symbol character (Insert › Symbol) is stored as the placeholder U+0028 with
936    // sprmCSymbol carrying the font and the character code ([MS-DOC] §2.6.1).
937    let symbol = if c == '(' { symbol_operand(grpprl) } else { None };
938    if field.is_none() && symbol.is_none() {
939        pb.push_shared(c, &m.props, st.generation);
940        return;
941    }
942    let mut props = m.props.clone();
943    let mut c = c;
944    if let Some((ftc, xchar)) = symbol {
945        if let Some(font) = ctx.fonts.get(usize::from(ftc)) {
946            props.font = Some(font.clone());
947        }
948        if let Some(sym) = char::from_u32(u32::from(xchar)) {
949            c = sym;
950        }
951    }
952    match field {
953        Some(f) => match f.sep.as_mut() {
954            // Inside a field, characters are recorded for the field object instead of shown;
955            // `finish_field` emits the visible text (hyperlink runs or one object anchor).
956            Some(result) => result.push(c, props),
957            None => f.instr.push(c),
958        },
959        None => pb.push(c, props),
960    }
961}

963/// What the walk carries from one character to the next while it resolves formatting: the piece
964/// and the CHPX run the last character was in (so most characters need no search), and the last
965/// formatting resolved.
966#[derive(Default)]
967struct FmtState<'a> {
968    span: Option<piece::PieceSpan<'a>>,
969    chpx: Option<fkp::ChpxRun<'a>>,
970    memo: Option<FmtMemo<'a>>,
971    /// Counts the memo's replacements, so equal values mean the same properties.
972    generation: u64,
973}
974
975/// The last character formatting resolved, reused while the CHPX, the piece's Prm and the
976/// paragraph style stay the same, so a Prm grpprl (up to 64 KB) is interpreted once per
977/// formatting change, not once per character (#3692).
978struct FmtMemo<'a> {
979    /// The CHPX grpprl, compared by address: it is a slice of the WordDocument stream.
980    chpx: &'a [u8],
981    prm: u16,
982    style: Option<String>,
983    props: CharProps,
984}

986/// Apply a piece's Prm (Prm0 short sprm or Prm1 grpprl) to character properties. A Prm1
987/// grpprl is charged to `work`; once that is spent the grpprl is skipped.
988fn apply_prm(props: &mut CharProps, prm: PrmRef<'_>, fonts: &[String], base: &CharProps, work: &Work) {
989    match prm {
990        PrmRef::None => {}
991        PrmRef::Sprm0 { isprm, val } => {
992            if let Some(op) = sprm::from_prm0(isprm) {
993                let operand = [val];
994                let prl = Prl { op, operand: &operand };
995                if prl.sgc() == 2 {
996                    fmt::apply_char(props, &prl, base, fonts);
997                }
998            }
999        }
1000        PrmRef::Grpprl(Some(g)) => {
1001            if !work.charge(g.len() as u64, Cost::Formatting) {
1002                return;
1003            }
1004            for prl in sprm::iter(g) {
1005                if prl.sgc() == 2 {
1006                    fmt::apply_char(props, &prl, base, fonts);
1007                }
1008            }
1009        }
1010        PrmRef::Grpprl(None) => {}
1011    }
1012}

1014/// Character props from a CHPX grpprl applied over `base`.
1015fn char_props(grpprl: &[u8], fonts: &[String], base: &CharProps) -> CharProps {
1016    let mut p = base.clone();
1017    for prl in sprm::iter(grpprl) {
1018        if prl.sgc() == 2 {
1019            fmt::apply_char(&mut p, &prl, base, fonts);
1020        }
1021    }
1022    p
1023}

1025/// Read a root-level stream fully, or `None` if it is absent.
1026fn stream<F: Read + Seek>(comp: &mut cfb::CompoundFile<F>, name: &str) -> Option<Vec<u8>> {
1027    let mut s = comp.open_stream(format!("/{name}")).ok()?;
1028    let mut v = Vec::new();
1029    // Read with a cap one byte above the limit so oversized streams are still caught by the
1030    // caller instead of exhausting memory here.
1031    s.by_ref().take(MAX_STREAM + 1).read_to_end(&mut v).ok()?;
1032    Some(v)
1033}
