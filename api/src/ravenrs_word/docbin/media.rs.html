1//! Inline pictures: `sprmCPicLocation` → a `PICF` in the Data stream → embedded image bytes
2//! ([MS-DOC] §2.9.190 PICF, §1.3.5 Pictures), and the shared store of every picture the
3//! document reads (inline and floating).
4
5use std::collections::{BTreeMap, HashMap};
6use std::ops::Range;
7use std::sync::Arc;
8
9use crate::docbin::officeart;

11/// `PICF.cbHeader` — the fixed size of the PICF structure.
12const CB_HEADER: usize = 0x44;
13/// Offsets inside PICF: `PICMID.dxaGoal/dyaGoal/mx/my` start at 28.
14const OFF_PICMID: usize = 28;
15
16/// Most distinct pictures one document materialises from the Data stream.
17const MAX_PICTURES: usize = 5_000;
18/// Most picture bytes one document copies out of the Data stream, all pictures together.
19pub(crate) const MAX_MEDIA_BYTES: usize = 256 << 20;
20/// Most bytes one document decodes pictures into (metafiles inflated, DIB pixels), charged
21/// per attempt whether it succeeds or not: retained media is capped separately, by
22/// `MAX_MEDIA_BYTES` (#3711, #3712).
23const MAX_DECODE_WORK: usize = 512 << 20;
24/// The decode budget grows with the input: up to this many bytes per byte of the WordDocument
25/// and Data streams. An uncompressed DIB declares at most 32 bytes of pixels per file byte (1
26/// bit per pixel, 4 bytes charged each), so 64 passes any such picture. RLE8/RLE4 bitmaps
27/// reach about 510 times that and rely on the `MAX_DECODE_WORK` floor instead; a decompression
28/// bomb (ratios around 1e6) still hits the cap.
29const DECODE_WORK_PER_INPUT_BYTE: usize = 64;
30/// The most the decode budget grows to, whatever the input.
31const MAX_DECODE_WORK_CEILING: usize = 2 << 30;
32
33/// The decode budget for a document whose pictures live in `input` stream bytes, never less
34/// than `floor`.
35pub(crate) fn decode_work(input: usize, floor: usize) -> usize {
36    input.saturating_mul(DECODE_WORK_PER_INPUT_BYTE).clamp(floor, floor.max(MAX_DECODE_WORK_CEILING))
37}

39/// The document's pictures, keyed by their location in the Data stream: anchors that share a
40/// `fc_pic` share one media entry (or one failure), and the total bytes copied are capped, so a
41/// file with thousands of anchors on one huge picture costs one copy, not thousands. Pictures of
42/// the drawing layer's BLIP store are shared the same way, by store index.
43pub(crate) struct Pictures {
44    /// fc_pic → the inline picture already materialised (`None`: unreadable or over the cap,
45    /// not retried).
46    by_fc: HashMap<u32, Option<Inline>>,
47    /// BLIP store index → media key (`None`: unreadable, not retried).
48    by_bse: HashMap<u32, Option<String>>,
49    /// (stream, foDelay, record length) of a BLIP record outside the store → media key: store
50    /// entries and inline pictures that alias one record share one decode (`None`: failed, not
51    /// retried). The stream is told apart by its address.
52    by_delay: HashMap<(usize, usize, usize), Option<String>>,
53    /// The media entries, merged into the document at the end.
54    pub(crate) media: BTreeMap<String, Arc<Vec<u8>>>,
55    total: usize,
56    /// The byte cap (`MAX_MEDIA_BYTES`; smaller in tests).
57    pub(crate) limit: usize,
58    /// Decode work left (`MAX_DECODE_WORK`; smaller in tests).
59    pub(crate) work: officeart::Decode,
60    /// Anchors whose picture was dropped (only the first few are logged).
61    dropped: usize,
62    /// Pictures kept as placeholders (formats we can't draw).
63    placeholders: usize,
64}
65
66impl Default for Pictures {
67    fn default() -> Self {
68        Pictures {
69            by_fc: HashMap::new(),
70            by_bse: HashMap::new(),
71            by_delay: HashMap::new(),
72            media: BTreeMap::new(),
73            total: 0,
74            limit: MAX_MEDIA_BYTES,
75            work: officeart::Decode::new(MAX_DECODE_WORK),
76            dropped: 0,
77            placeholders: 0,
78        }
79    }
80}

82/// An inline picture: its media key, display size (points), crop fractions and alt text.
83#[derive(Clone, Debug)]
84pub(crate) struct Inline {
85    pub(crate) key: String,
86    pub(crate) w: f32,
87    pub(crate) h: f32,
88    pub(crate) crop: [f32; 4],
89    pub(crate) alt: String,
90}
91
92impl Pictures {
93    /// An empty store whose decode budget scales with the `input` stream bytes (WordDocument
94    /// plus Data) the pictures can come from.
95    pub(crate) fn for_input(input: usize) -> Self {
96        Pictures { work: officeart::Decode::new(decode_work(input, MAX_DECODE_WORK)), ..Pictures::default() }
97    }
98
99    /// The inline picture at `fc_pic` in the Data stream, copying its bytes the first time.
100    /// `None` when the header is malformed, the picture can't be read, or a picture
101    /// count/byte cap is reached.
102    pub(crate) fn get(&mut self, data: &[u8], fc_pic: u32) -> Option<&Inline> {
103        if !self.by_fc.contains_key(&fc_pic) {
104            if self.by_fc.len().saturating_add(self.by_bse.len()) >= MAX_PICTURES {
105                return None;
106            }
107            let pic = self.read(data, fc_pic);
108            self.by_fc.insert(fc_pic, pic);
109        }
110        self.by_fc.get(&fc_pic)?.as_ref()
111    }
112
113    /// Read the inline picture at `fc_pic`: nothing is decoded or copied beyond what is left of
114    /// the byte cap, and nothing at all once it is used up.
115    fn read(&mut self, data: &[u8], fc_pic: u32) -> Option<Inline> {
116        let room = self.limit.saturating_sub(self.total);
117        if room == 0 {
118            return None;
119        }
120        match locate(data, fc_pic, self)? {
121            Found::Range(pic) => {
122                let bytes = data.get(pic.range.clone()).filter(|b| b.len() <= room)?.to_vec();
123                let key = self.insert(bytes, pic.ext, false)?;
124                Some(Inline { key, w: pic.w, h: pic.h, crop: [0.0; 4], alt: String::new() })
125            }
126            Found::Shape { w, h, key, props } => Some(Inline { key: key?, w, h, crop: props.crop, alt: props.alt }),
127        }
128    }
129
130    /// The media key of BLIP store entry `pib` (1-based), decoding it the first time; `delay`
131    /// is the stream that holds BLIPs not embedded in the store (the WordDocument stream).
132    pub(crate) fn bse(&mut self, pib: u32, bstore: &[&[u8]], delay: &[u8]) -> Option<String> {
133        if let Some(hit) = self.by_bse.get(&pib) {
134            return hit.clone();
135        }
136        if self.by_fc.len().saturating_add(self.by_bse.len()) >= MAX_PICTURES {
137            return None;
138        }
139        let fbse = bstore.get((pib as usize).checked_sub(1)?)?;
140        let key = self.decode_bse(fbse, delay);
141        self.by_bse.insert(pib, key.clone());
142        key
143    }
144
145    /// Decode a store entry's BLIP, once per delay-stream record however many entries name it.
146    fn decode_bse(&mut self, fbse: &[u8], delay: &[u8]) -> Option<String> {
147        let (rec, at) = officeart::fbse_record(fbse, delay)?;
148        let at = at.map(|(fo, used)| (delay.as_ptr() as usize, fo, used));
149        if let Some(hit) = at.and_then(|at| self.by_delay.get(&at)) {
150            return hit.clone();
151        }
152        let room = self.limit.saturating_sub(self.total);
153        let key = officeart::blip(&rec, room, &mut self.work).and_then(|b| self.insert(b.bytes, b.ext, b.placeholder));
154        if let Some(at) = at {
155            self.by_delay.insert(at, key.clone());
156        }
157        key
158    }
159
160    /// Add picture bytes as a new media entry, within the byte cap.
161    fn insert(&mut self, bytes: Vec<u8>, ext: &str, placeholder: bool) -> Option<String> {
162        let total = self.total.checked_add(bytes.len()).filter(|&t| t <= self.limit)?;
163        let key = format!("image{}.{ext}", self.media.len() + 1);
164        if placeholder {
165            if self.placeholders < 10 {
166                log::warn!("docbin: picture `{key}` is in a format WordCraft can't draw ({ext}); it shows as a placeholder");
167            }
168            self.placeholders = self.placeholders.saturating_add(1);
169        }
170        self.media.insert(key.clone(), Arc::new(bytes));
171        self.total = total;
172        Some(key)
173    }
174
175    /// Note a picture anchor that could not be materialised.
176    pub(crate) fn dropped(&mut self, cp: u32) {
177        if self.dropped < 10 {
178            log::warn!("docbin: picture at CP {cp} could not be read (or a picture cap was reached) and was dropped");
179        }
180        self.dropped = self.dropped.saturating_add(1);
181    }
182}

184/// Where a picture's image bytes sit in the Data stream, and its display size in points.
185struct Located {
186    w: f32,
187    h: f32,
188    range: Range<usize>,
189    ext: &'static str,
190}
191
192/// What a `PICFAndOfficeArtData` holds: raw image bytes (older writers), or an
193/// `OfficeArtInlineSpContainer` (Word 97 and later) whose BLIP is decoded.
194enum Found {
195    Range(Located),
196    Shape { w: f32, h: f32, key: Option<String>, props: officeart::Props },
197}
198
199/// `PICF.mfpf.mm` values of pictures stored as a shape (`MM_SHAPE`) or a shape with a file
200/// name (`MM_SHAPEFILE`).
201const MM_SHAPE: u16 = 0x0064;
202const MM_SHAPEFILE: u16 = 0x0066;
203
204/// Locate the picture at `fc_pic` in the Data stream ([MS-DOC] §2.9.192 PICFAndOfficeArtData);
205/// the BLIP is decoded once per record and charged to the decode budget.
206fn locate(data: &[u8], fc_pic: u32, pics: &mut Pictures) -> Option<Found> {
207    let base = fc_pic as usize;
208    let pic = data.get(base..)?;
209    let lcb = u32::from_le_bytes(pic.get(0..4)?.try_into().ok()?) as usize;
210    let cb_header = u16::from_le_bytes(pic.get(4..6)?.try_into().ok()?) as usize;
211    let cb_header = if (0x20..=0x100).contains(&cb_header) { cb_header } else { CB_HEADER };
212    let mm = u16::from_le_bytes(pic.get(6..8)?.try_into().ok()?);
213    let m = pic.get(OFF_PICMID..OFF_PICMID + 8)?;
214    let dxa_goal = i16::from_le_bytes([m[0], m[1]]) as f32;
215    let dya_goal = i16::from_le_bytes([m[2], m[3]]) as f32;
216    let mx = u16::from_le_bytes([m[4], m[5]]) as f32;
217    let my = u16::from_le_bytes([m[6], m[7]]) as f32;
218    let scale = |r: f32| if r > 0.0 { (r / 1000.0).clamp(0.01, 100.0) } else { 1.0 };
219    let end = lcb.min(pic.len());
220    if mm == MM_SHAPE || mm == MM_SHAPEFILE {
221        let mut at = cb_header;
222        if mm == MM_SHAPEFILE {
223            at = at.checked_add(1 + *pic.get(at)? as usize)?;
224        }
225        let (key, props) = inline_shape(pic.get(at..end)?, data, pics)?;
226        // The goal size is before cropping; the crop fractions take their share off it.
227        let [cl, ct, cr, cb] = props.crop;
228        let w = (dxa_goal / 20.0 * (1.0 - cl - cr).max(0.01) * scale(mx)).clamp(1.0, 1584.0);
229        let h = (dya_goal / 20.0 * (1.0 - ct - cb).max(0.01) * scale(my)).clamp(1.0, 1584.0);
230        return Some(Found::Shape { w, h, key, props });
231    }
232    let w = (dxa_goal / 20.0 * scale(mx)).clamp(1.0, 1584.0);
233    let h = (dya_goal / 20.0 * scale(my)).clamp(1.0, 1584.0);
234    let body = pic.get(cb_header..end)?;
235    let at = sniff(body)?;
236    let image = body.get(at..)?;
237    let start = base.checked_add(cb_header)?.checked_add(at)?;
238    Some(Found::Range(Located { w, h, range: start..start.checked_add(image.len())?, ext: ext_of(image) }))
239}

241/// An `OfficeArtInlineSpContainer`: the picture shape, then the `OfficeArtFBSE` records its
242/// `pib` indexes (BLIPs embedded, or at `foDelay` in the Data stream). The BLIP is decoded
243/// once per record, however many pictures (and FBSE sizes) name it.
244fn inline_shape(buf: &[u8], data: &[u8], pics: &mut Pictures) -> Option<(Option<String>, officeart::Props)> {
245    let mut budget = 10_000usize;
246    let recs = officeart::children(buf, &mut budget);
247    let sp = recs.iter().find(|r| r.typ == officeart::rt::SP_CONTAINER).map(|r| officeart::shape(r.body, &mut budget))?;
248    let fbses: Vec<&[u8]> = recs.iter().filter(|r| r.typ == officeart::rt::FBSE).map(|r| r.body).collect();
249    let fbse = sp.props.pib.and_then(|p| fbses.get((p as usize).checked_sub(1)?)).or(fbses.first())?;
250    let key = pics.decode_bse(fbse, data);
251    Some((key, sp.props))
252}

254/// Offset of the first embedded raster image inside the payload, skipping the small
255/// metafile header some writers put in front of PNG/JPEG data. `None` for metafiles.
256fn sniff(body: &[u8]) -> Option<usize> {
257    for at in [0usize, 8, 12, 16, 20, 22, 24, 32] {
258        if let Some(b) = body.get(at..)
259            && (b.starts_with(b"\x89PNG") || b.starts_with(b"\xFF\xD8\xFF") || b.starts_with(b"GIF8") || b.starts_with(b"BM"))
260        {
261            return Some(at);
262        }
263    }
264    None
265}
266
267fn ext_of(b: &[u8]) -> &'static str {
268    if b.starts_with(b"\x89PNG") {
269        "png"
270    } else if b.starts_with(b"\xFF\xD8\xFF") {
271        "jpeg"
272    } else if b.starts_with(b"GIF8") {
273        "gif"
274    } else {
275        "bmp"
276    }
