10/// Most sections we read.
11const MAX_SECTIONS: usize = 1024;
12
13// Section sprms ([MS-DOC] §2.6.4).
14const S_BKC: u16 = 0x3009;
15const S_TITLE_PAGE: u16 = 0x300A; // sprmSFTitlePage
16const S_CCOLUMNS: u16 = 0x500B;
17const S_PGN_START: u16 = 0x300C; // sprmSPgnStart
18const S_ORIENTATION: u16 = 0x301D; // sprmSBOrientation
19const S_HDR_TOP: u16 = 0xB017;
20const S_HDR_BOTTOM: u16 = 0xB018;
21const S_XA_PAGE: u16 = 0xB01F;
22const S_YA_PAGE: u16 = 0xB020;
23const S_DXA_LEFT: u16 = 0xB021;
24const S_DXA_RIGHT: u16 = 0xB022;
25const S_DYA_TOP: u16 = 0x9023;
26const S_DYA_BOTTOM: u16 = 0x9024;
27const S_DXA_GUTTER: u16 = 0xB025;

29/// One section: covers paragraphs up to (excluding) `end_cp`, with its properties.
30pub(crate) struct Section {
31    pub(crate) end_cp: u32,
32    pub(crate) props: SectionProps,
33}
34
35/// Parse `PlcfSed` and each section's `Sepx` grpprl.
36pub(crate) fn parse(word: &[u8], table: &[u8], fib: &Fib) -> Vec<Section> {
37    let Some((fc, lcb)) = fib.pair(pair::PLCF_SED) else { return Vec::new() };
38    let Some(plc) = table.get(fc as usize..(fc as usize).saturating_add(lcb as usize)) else { return Vec::new() };
39    // PlcSed: aCp[n+1] u32s + aSed[n] (Sed = fn 2, fcSepx 4, fnMpr 2, fcMpr 4).
40    if plc.len() < 4 || (plc.len() - 4) % 16 != 0 {
41        return Vec::new();
42    }
43    let n = (plc.len() - 4) / 16;
44    let cp_at = |i: usize| -> Option<u32> { plc.get(i * 4..i * 4 + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) };
45    let mut out = Vec::with_capacity(n.min(MAX_SECTIONS));
46    for i in 0..n.min(MAX_SECTIONS) {
47        let Some(end_cp) = cp_at(i + 1) else { break };
48        let sed_at = (n + 1) * 4 + i * 12;
49        let fc_sepx = plc.get(sed_at + 2..sed_at + 6).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])).unwrap_or(0xFFFF_FFFF);
50        let mut props = SectionProps::default();
51        // Sepx: cb u16, then a grpprl of section sprms.
52        let mut explicit_orient = false;
53        if let Some(grpprl) =
54            word.get(fc_sepx as usize..).and_then(|sepx| sepx.get(..2).map(|b| u16::from_le_bytes([b[0], b[1]]) as usize)).and_then(|cb| {
55                let at = (fc_sepx as usize).checked_add(2)?;
56                word.get(at..at.checked_add(cb)?)
57            })
58        {
59            for prl in sprm::iter(grpprl) {
60                // Only a defined operand counts as explicit; anything else leaves the
61                // orientation to the page shape.
62                if prl.op == S_ORIENTATION && matches!(prl.operand.first(), Some(1 | 2)) {
63                    explicit_orient = true;
64                }
65                apply_section(&mut props, &prl);
66            }
67        }
68        // Without an explicit orientation sprm, infer it from the page shape.
69        if !explicit_orient {
70            if props.page_w < props.page_h {
71                props.landscape = false;
72            } else if props.page_w > props.page_h {
73                props.landscape = true;
74            }
75        }
76        out.push(Section { end_cp, props });
77    }
78    out
79}

81/// Apply a section sprm to the model's section properties. All the measurement sprms here
82/// carry 16-bit twip operands ([MS-DOC] §2.6.4).
83fn apply_section(s: &mut SectionProps, prl: &Prl) {
84    let tw = |v: i32| (v as f32 / 20.0).clamp(0.0, 1584.0);
85    let s16 = |prl: &Prl| -> Option<i16> {
86        match prl.operand {
87            [b0, b1, ..] => Some(i16::from_le_bytes([*b0, *b1])),
88            _ => None,
89        }
90    };
91    match prl.op {
92        S_BKC => {
93            s.start = match prl.operand.first().copied().unwrap_or(2) {
94                // SBkcOperand ([MS-DOC] sprmSBkc): bkcContinuous, bkcNewColumn, bkcNewPage,
95                // bkcEvenPage, bkcOddPage; anything else falls back to the default (new page).
96                0 => SectionStart::Continuous,
97                1 => SectionStart::NextColumn,
98                3 => SectionStart::EvenPage,
99                4 => SectionStart::OddPage,
100                _ => SectionStart::NextPage,
101            };
102        }
103        S_TITLE_PAGE => s.title_page = prl.operand.first().is_some_and(|&v| v != 0),
104        S_CCOLUMNS => {
105            if let Some(c) = s16(prl) {
106                s.columns = Columns { count: (c as u32).saturating_add(1).min(44), ..Default::default() };
107            }
108        }
109        S_PGN_START => {
110            if let Some(v) = s16(prl) {
111                s.page_num_start = Some(v.max(0) as u32);
112                s.page_num_format = NumFormat::Decimal;
113            }
114        }
115        // SBOrientationOperand ([MS-DOC] §2.9.236): 1 portrait, 2 landscape.
116        S_ORIENTATION => match prl.operand.first() {
117            Some(1) => s.landscape = false,
118            Some(2) => s.landscape = true,
119            _ => {}
120        },
121        S_HDR_TOP => {
122            if let Some(v) = s16(prl) {
123                s.header = tw(v as i32);
124            }
125        }
126        S_HDR_BOTTOM => {
127            if let Some(v) = s16(prl) {
128                s.footer = tw(v as i32);
129            }
130        }
131        S_XA_PAGE => {
132            if let Some(v) = s16(prl) {
133                s.page_w = tw(v as i32).max(7.2);
134            }
135        }
136        S_YA_PAGE => {
137            if let Some(v) = s16(prl) {
138                s.page_h = tw(v as i32).max(7.2);
139            }
140        }
141        S_DXA_LEFT => {
142            if let Some(v) = s16(prl) {
143                s.margin_left = tw(v as i32);
144            }
145        }
146        S_DXA_RIGHT => {
147            if let Some(v) = s16(prl) {
148                s.margin_right = tw(v as i32);
149            }
150        }
151        S_DYA_TOP => {
152            if let Some(v) = s16(prl) {
153                s.margin_top = tw(v as i32);
154            }
155        }
156        S_DYA_BOTTOM => {
157            if let Some(v) = s16(prl) {
158                s.margin_bottom = tw(v as i32);
159            }
160        }
161        S_DXA_GUTTER => {
162            if let Some(v) = s16(prl) {
163                s.gutter = tw(v as i32);
164            }
165        }
166        _ => {}
167    }
168}

170/// The header stories' CP ranges from `Plcfhdd` (a PLC of bare CPs): story i spans
171/// `[cps[i], cps[i+1])`. Stories 0-5 are footnote/endnote separators; every section then owns
172/// six stories in the order even header, odd (default) header, even footer, odd (default)
173/// footer, first header, first footer. The CPs are relative to the header subdocument.
174///
175/// The CPs of a well-formed PLC ascend, so the stories are disjoint. A story starting before
176/// the end of an earlier one (a hostile table pointing many sections at the same text, each
177/// read into its own part) is left empty, like a slot linked to the previous section's:
178/// every subdocument CP becomes at most one story.
179pub(crate) fn header_stories(table: &[u8], fib: &Fib) -> Vec<(u32, u32)> {
180    let Some((fc, lcb)) = fib.pair(pair::PLCF_HDD) else { return Vec::new() };
181    let Some(plc) = table.get(fc as usize..(fc as usize).saturating_add(lcb as usize)) else { return Vec::new() };
182    // Only the six separator stories and six per section (at most 1024) are ever used.
183    let n = (lcb as usize / 4).min(6 + 6 * MAX_SECTIONS + 1);
184    let cp_at = |i: usize| -> Option<u32> { plc.get(i * 4..i * 4 + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) };
185    let mut stories = Vec::with_capacity(n);
186    let mut kept_to = 0u32;
187    for i in 0..n.saturating_sub(1) {
188        if let (Some(a), Some(b)) = (cp_at(i), cp_at(i + 1)) {
189            let b = if a < kept_to { a } else { b.max(a) };
190            kept_to = kept_to.max(b);
191            stories.push((a, b));
192        }
193    }
194    stories
195}
