1//! Password-protected Word 97-2003 files ([MS-DOC] §2.2.6, [MS-OFFCRYPTO] §2.3.5-2.3.7).
2//!
3//! Three schemes exist for `.doc`:
4//!
5//! * **XOR obfuscation** (`fObfuscated`): a 32-bit password verifier sits in `FibBase.lKey` and
6//!   the streams are XORed with a 16-byte array derived from the password (method 2).
7//! * **RC4** (header version 1.1): `EncryptionHeader` at the start of the Table stream; the key
8//!   per 512-byte block comes from MD5.
9//! * **RC4 CryptoAPI** (header version 2/3/4 . 2): like RC4 but with a SHA-1 key derivation, a
10//!   key size of 40-128 bits and a variable-size header.
11//!
12//! In all three the WordDocument stream (past its first 68 bytes), the Table stream (past its
13//! header, for RC4) and the whole Data stream are transformed. The password is checked against
14//! the stored verifier before anything is decrypted, so a wrong password is reported and never
15//! yields a garbled document. Everything is written from the public specifications.
16
17use sha1::{Digest, Sha1};
18
19use crate::docbin::DocbinError;

21/// Bytes at the start of the WordDocument stream that are stored untransformed.
22pub(crate) const FIB_PLAIN: usize = 68;
23/// RC4 re-keys every this many bytes of a stream.
24const BLOCK: usize = 512;
25/// Longest password considered (the specifications' limit for the RC4 schemes).
26const MAX_PASSWORD_CHARS: usize = 255;
27/// XOR obfuscation uses at most this many single-byte password characters.
28const MAX_XOR_PASSWORD: usize = 15;
29/// `FibBase.flags`: the document is encrypted (`fEncrypted`) / obfuscated (`fObfuscated`).
30pub(crate) const F_ENCRYPTED: u16 = 0x0100;
31pub(crate) const F_OBFUSCATED: u16 = 0x8000;
32/// `EncryptionHeaderFlags`: `fCryptoAPI`, `fExternal`, `fAES`.
33const H_CRYPTO_API: u32 = 0x04;
34const H_EXTERNAL: u32 = 0x10;
35const H_AES: u32 = 0x20;

37fn unsupported(what: &str) -> DocbinError {
38    DocbinError::Encryption(what.into())
39}

44/// MD5 ([RFC 1321]); the RC4 scheme is defined on it.
45pub(crate) fn md5(data: &[u8]) -> [u8; 16] {
46    const S: [u32; 64] = [
47        7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22, 5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20, 4, 11, 16, 23, 4, 11, 16,
48        23, 4, 11, 16, 23, 4, 11, 16, 23, 6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21,
49    ];
50    // K[i] = floor(2^32 * abs(sin(i + 1))), as the RFC defines it.
51    let k: Vec<u32> = (0..64u32).map(|i| (f64::from(i + 1).sin().abs() * 4_294_967_296.0) as u32).collect();
52    let (mut a0, mut b0, mut c0, mut d0) = (0x6745_2301u32, 0xefcd_ab89u32, 0x98ba_dcfeu32, 0x1032_5476u32);
53    let mut msg = Vec::with_capacity(data.len() + 72);
54    msg.extend_from_slice(data);
55    msg.push(0x80);
56    while msg.len() % 64 != 56 {
57        msg.push(0);
58    }
59    msg.extend_from_slice(&(data.len() as u64).wrapping_mul(8).to_le_bytes());
60    for chunk in msg.as_chunks::<64>().0 {
61        let mut m = [0u32; 16];
62        let words = chunk.as_chunks::<4>().0;
63        for (w, b) in m.iter_mut().zip(words) {
64            *w = u32::from_le_bytes(*b);
65        }
66        let (mut a, mut b, mut c, mut d) = (a0, b0, c0, d0);
67        for i in 0..64usize {
68            let (f, g) = match i / 16 {
69                0 => ((b & c) | (!b & d), i),
70                1 => ((d & b) | (!d & c), (5 * i + 1) % 16),
71                2 => (b ^ c ^ d, (3 * i + 5) % 16),
72                _ => (c ^ (b | !d), (7 * i) % 16),
73            };
74            let f = f.wrapping_add(a).wrapping_add(k[i]).wrapping_add(m[g]);
75            a = d;
76            d = c;
77            c = b;
78            b = b.wrapping_add(f.rotate_left(S[i]));
79        }
80        a0 = a0.wrapping_add(a);
81        b0 = b0.wrapping_add(b);
82        c0 = c0.wrapping_add(c);
83        d0 = d0.wrapping_add(d);
84    }
85    let mut out = [0u8; 16];
86    let (cells, _) = out.as_chunks_mut::<4>();
87    for (o, v) in cells.iter_mut().zip([a0, b0, c0, d0]) {
88        *o = v.to_le_bytes();
89    }
90    out
91}

93/// SHA-1, as [MS-OFFCRYPTO] §2.3.5 specifies for the RC4 CryptoAPI key derivation.
94/// The digest is part of the file format, not a choice made here: a stronger hash would
95/// make password-protected `.doc` files written by Word unreadable.
96fn sha1(parts: &[&[u8]]) -> Vec<u8> {
97    let mut h = Sha1::new(); // NOSONAR: required by the MS-OFFCRYPTO RC4 CryptoAPI format
98    for p in parts {
99        h.update(p);
100    }
101    h.finalize().to_vec()
102}

104/// The RC4 stream cipher.
105pub(crate) struct Rc4 {
106    s: [u8; 256],
107    i: u8,
108    j: u8,
109}
110
111impl Rc4 {
112    pub(crate) fn new(key: &[u8]) -> Rc4 {
113        let mut s = [0u8; 256];
114        for (n, v) in s.iter_mut().enumerate() {
115            *v = n as u8;
116        }
117        let kl = key.len().max(1);
118        let mut j = 0u8;
119        for i in 0..256usize {
120            let k = key.get(i % kl).copied().unwrap_or(0);
121            j = j.wrapping_add(s[i]).wrapping_add(k);
122            s.swap(i, usize::from(j));
123        }
124        Rc4 { s, i: 0, j: 0 }
125    }
126
127    fn next(&mut self) -> u8 {
128        self.i = self.i.wrapping_add(1);
129        self.j = self.j.wrapping_add(self.s[usize::from(self.i)]);
130        self.s.swap(usize::from(self.i), usize::from(self.j));
131        self.s[usize::from(self.s[usize::from(self.i)].wrapping_add(self.s[usize::from(self.j)]))]
132    }
133
134    pub(crate) fn apply(&mut self, data: &mut [u8]) {
135        for b in data {
136            *b ^= self.next();
137        }
138    }
139}

168/// `CreateXorKey_Method1`. `pw` is 1 to 15 bytes.
169fn xor_key(pw: &[u8]) -> u16 {
170    let mut key = INITIAL_CODE.get(pw.len().wrapping_sub(1)).copied().unwrap_or(0);
171    let mut current = 0x68i32;
172    for &ch in pw.iter().rev() {
173        let mut ch = ch;
174        for _ in 0..7 {
175            if ch & 0x40 != 0
176                && let Some(m) = usize::try_from(current).ok().and_then(|c| XOR_MATRIX.get(c))
177            {
178                key ^= m;
179            }
180            ch <<= 1;
181            current -= 1;
182        }
183    }
184    key
185}

187/// `CreatePasswordVerifier_Method1`.
188fn verifier_method1(pw: &[u8]) -> u16 {
189    let mut v = 0u16;
190    for &b in pw.iter().rev().chain(std::iter::once(&(pw.len() as u8))) {
191        let i1 = u16::from(v & 0x4000 != 0);
192        let i2 = v.wrapping_mul(2) & 0x7FFF;
193        v = (i1 | i2) ^ u16::from(b);
194    }
195    v ^ 0xCE4B
196}

198/// `CreatePasswordVerifier_Method2`: the 32-bit value stored in `FibBase.lKey`.
199pub(crate) fn xor_verifier(pw: &[u8]) -> u32 {
200    (u32::from(xor_key(pw)) << 16) | u32::from(verifier_method1(pw))
201}
202
203/// `CreateXorArray_Method2`.
204pub(crate) fn xor_array(pw: &[u8]) -> [u8; 16] {
205    let key = (xor_verifier(pw) >> 16) as u16;
206    let (high, low) = ((key >> 8) as u8, key as u8);
207    let mut arr = [0u8; 16];
208    for (i, slot) in arr.iter_mut().enumerate() {
209        *slot = pw.get(i).copied().unwrap_or_else(|| PAD_ARRAY.get(i.saturating_sub(pw.len())).copied().unwrap_or(0));
210    }
211    let (pairs, _) = arr.as_chunks_mut::<2>();
212    for pair in pairs {
213        pair[0] = (pair[0] ^ low).rotate_right(1);
214        pair[1] = (pair[1] ^ high).rotate_right(1);
215    }
216    arr
217}

219/// The single-byte forms of a password the XOR scheme tries: the system-codepage form (Latin-1
220/// here) and the "low byte, or high byte when the low one is zero" form. Each is cut to 15.
221fn xor_candidates(password: &str) -> Vec<Vec<u8>> {
222    let mut out: Vec<Vec<u8>> = Vec::new();
223    let latin: Option<Vec<u8>> = password.chars().take(MAX_XOR_PASSWORD).map(|c| u8::try_from(u32::from(c)).ok()).collect();
224    if let Some(l) = latin {
225        out.push(l);
226    }
227    let folded: Vec<u8> = password.encode_utf16().take(MAX_XOR_PASSWORD).map(|u| if u & 0xFF == 0 { (u >> 8) as u8 } else { u as u8 }).collect();
228    if !out.contains(&folded) {
229        out.push(folded);
230    }
231    out.retain(|p| !p.is_empty());
232    out
233}

238/// How the streams of one file are transformed, once the password has been verified.
239pub(crate) enum Cipher {
240    Xor { array: [u8; 16] },
241    Rc4 { base: Vec<u8>, crypto_api: bool, key_len: usize, table_skip: usize },
242}
243
244impl Cipher {
245    /// Check `password` against the file's verifier and set up the cipher. `flags` and `lkey` come
246    /// from the FIB base, `table` is the (still encrypted) Table stream.
247    pub(crate) fn open(flags: u16, lkey: u32, table: &[u8], password: &str) -> Result<Cipher, DocbinError> {
248        if flags & F_OBFUSCATED != 0 {
249            for pw in xor_candidates(password) {
250                if xor_verifier(&pw) == lkey {
251                    return Ok(Cipher::Xor { array: xor_array(&pw) });
252                }
253            }
254            return Err(DocbinError::WrongPassword);
255        }
256        let short = || DocbinError::Malformed("the encryption header is truncated".into());
257        let major = u16_at(table, 0).ok_or_else(short)?;
258        let minor = u16_at(table, 2).ok_or_else(short)?;
259        match (major, minor) {
260            (1, 1) => Self::open_rc4(table, password),
261            (2..=4, 2) => Self::open_crypto_api(table, password),
262            _ => Err(unsupported(&format!("unknown encryption version {major}.{minor}"))),
263        }
264    }
265
266    fn open_rc4(table: &[u8], password: &str) -> Result<Cipher, DocbinError> {
267        let short = || DocbinError::Malformed("the encryption header is truncated".into());
268        let salt = table.get(4..20).ok_or_else(short)?;
269        let enc_verifier = table.get(20..36).ok_or_else(short)?;
270        let enc_hash = table.get(36..52).ok_or_else(short)?;
271        let h0 = md5(&utf16le(password));
272        let mut buf = Vec::with_capacity(336);
273        for _ in 0..16 {
274            buf.extend_from_slice(&h0[..5]);
275            buf.extend_from_slice(salt);
276        }
277        let base = md5(&buf)[..5].to_vec();
278        let cipher = Cipher::Rc4 { base, crypto_api: false, key_len: 16, table_skip: 52 };
279        let mut rc4 = Rc4::new(&cipher.block_key(0));
280        let mut verifier = [0u8; 16];
281        verifier.copy_from_slice(enc_verifier);
282        let mut hash = [0u8; 16];
283        hash.copy_from_slice(enc_hash);
284        rc4.apply(&mut verifier);
285        rc4.apply(&mut hash);
286        if md5(&verifier) != hash {
287            return Err(DocbinError::WrongPassword);
288        }
289        Ok(cipher)
290    }
291
292    fn open_crypto_api(table: &[u8], password: &str) -> Result<Cipher, DocbinError> {
293        let short = || DocbinError::Malformed("the encryption header is truncated".into());
294        let hflags = u32_at(table, 4).ok_or_else(short)?;
295        if hflags & H_EXTERNAL != 0 {
296            return Err(unsupported("the document uses an external encryption provider"));
297        }
298        // [MS-OFFCRYPTO] defines AES only for the encrypted packages of Word 2007 and later
299        // (which open through the .docx reader whatever their extension); a Word 97-2003 file
300        // may only use XOR, RC4 or RC4 CryptoAPI, so there is no public scheme to follow here.
301        if hflags & H_AES != 0 {
302            return Err(unsupported(
303                "this Word 97-2003 file claims AES encryption, which is only defined for the newer encrypted-package format; it can't be opened",
304            ));
305        }
306        if hflags & H_CRYPTO_API == 0 {
307            return Err(unsupported("unknown encryption header"));
308        }
309        let hsize = usize::try_from(u32_at(table, 8).ok_or_else(short)?).unwrap_or(usize::MAX);
310        // EncryptionHeader: flags, sizeExtra, algId, algIdHash, keySize, providerType, 2 reserved
311        // words and the CSP name; it starts at byte 12.
312        if hsize < 32 || hsize > table.len() {
313            return Err(DocbinError::Malformed("the encryption header has a bad size".into()));
314        }
315        let alg_id = u32_at(table, 20).ok_or_else(short)?;
316        let alg_hash = u32_at(table, 24).ok_or_else(short)?;
317        if !matches!(alg_id, 0 | 0x6801) {
318            return Err(unsupported(&format!("encryption algorithm 0x{alg_id:X}")));
319        }
320        if !matches!(alg_hash, 0 | 0x8004) {
321            return Err(unsupported(&format!("hash algorithm 0x{alg_hash:X}")));
322        }
323        let mut bits = u32_at(table, 28).ok_or_else(short)?;
324        if bits == 0 {
325            bits = 40;
326        }
327        if !(40..=128).contains(&bits) || bits % 8 != 0 {
328            return Err(unsupported(&format!("a {bits}-bit key")));
329        }
330        let key_len = (bits / 8) as usize;
331        // EncryptionVerifier: saltSize, salt[16], encryptedVerifier[16], verifierHashSize,
332        // encryptedVerifierHash[20].
333        let v = 12 + hsize;
334        if u32_at(table, v) != Some(16) {
335            return Err(DocbinError::Malformed("the encryption salt has a bad size".into()));
336        }
337        let salt = table.get(v + 4..v + 20).ok_or_else(short)?;
338        let enc_verifier = table.get(v + 20..v + 36).ok_or_else(short)?;
339        if u32_at(table, v + 36) != Some(20) {
340            return Err(unsupported("a verifier hash that is not SHA-1"));
341        }
342        let enc_hash = table.get(v + 40..v + 60).ok_or_else(short)?;
343        let mut pw = utf16le(password);
344        let base = sha1(&[salt, &pw]);
345        pw.fill(0);
346        let cipher = Cipher::Rc4 { base, crypto_api: true, key_len, table_skip: v + 60 };
347        let mut rc4 = Rc4::new(&cipher.block_key(0));
348        let mut verifier = enc_verifier.to_vec();
349        let mut hash = enc_hash.to_vec();
350        rc4.apply(&mut verifier);
351        rc4.apply(&mut hash);
352        if sha1(&[&verifier]) != hash {
353            return Err(DocbinError::WrongPassword);
354        }
355        Ok(cipher)
356    }

358    /// The RC4 key of stream block `block`.
359    fn block_key(&self, block: u32) -> Vec<u8> {
360        match self {
361            Cipher::Xor { .. } => Vec::new(),
362            Cipher::Rc4 { base, crypto_api: false, .. } => md5(&[base.as_slice(), &block.to_le_bytes()].concat()).to_vec(),
363            Cipher::Rc4 { base, crypto_api: true, key_len, .. } => {
364                let mut k = sha1(&[base, &block.to_le_bytes()]);
365                k.truncate(*key_len);
366                // A 40-bit key is zero-padded to 128 bits.
367                if *key_len == 5 {
368                    k.resize(16, 0);
369                }
370                k
371            }
372        }
373    }

375    /// Decrypt `data` in place; its first `skip` bytes are stored untransformed.
376    fn decrypt(&self, data: &mut [u8], skip: usize) {
377        match self {
378            Cipher::Xor { array } => {
379                for (i, b) in data.iter_mut().enumerate().skip(skip) {
380                    let a = array[i % 16];
381                    // Zero bytes, and bytes equal to the array element, were left as they were.
382                    if *b != 0 && *b != a {
383                        *b ^= a;
384                    }
385                }
386            }
387            Cipher::Rc4 { .. } => {
388                for (n, chunk) in data.chunks_mut(BLOCK).enumerate() {
389                    let Ok(block) = u32::try_from(n) else { return };
390                    let mut rc4 = Rc4::new(&self.block_key(block));
391                    let start = n * BLOCK;
392                    // The keystream runs over the untransformed bytes too.
393                    let plain = skip.saturating_sub(start).min(chunk.len());
394                    let head = chunk.get(..plain).map(<[u8]>::to_vec).unwrap_or_default();
395                    rc4.apply(chunk);
396                    if let Some(dst) = chunk.get_mut(..plain) {
397                        dst.copy_from_slice(&head);
398                    }
399                }
400            }
401        }
402    }
403
404    pub(crate) fn decrypt_word(&self, data: &mut [u8]) {
405        self.decrypt(data, FIB_PLAIN);
406    }
407
408    pub(crate) fn decrypt_table(&self, data: &mut [u8]) {
409        let skip = match self {
410            Cipher::Xor { .. } => 0,
411            Cipher::Rc4 { table_skip, .. } => *table_skip,
412        };
413        self.decrypt(data, skip);
414    }
415
416    pub(crate) fn decrypt_data(&self, data: &mut [u8]) {
417        self.decrypt(data, 0);
418    }
