1//! Sprm decoding ([MS-DOC] §2.6). A Prl is a 2-byte opcode followed by an operand whose
2//! size comes from the opcode's `spra` field (with two variable-length exceptions).
3
4/// `sprmPChgTabs`: custom tab stops added and removed ([MS-DOC] §2.6.3, PChgTabsOperand).
5pub(crate) const P_CHG_TABS: u16 = 0xC615;
6/// `sprmPChgTabsPapx`: the same for paragraph and style formatting (PChgTabsPapxOperand).
7pub(crate) const P_CHG_TABS_PAPX: u16 = 0xC60D;
8/// `sprmTDefTable`: table cell borders and widths.
9const T_DEF_TABLE: u16 = 0xD608;
10
11/// One decoded Prl: an opcode plus its operand bytes.
12#[derive(Clone, Copy, Debug)]
13pub(crate) struct Prl<'a> {
14    pub(crate) op: u16,
15    pub(crate) operand: &'a [u8],
16}
17
18impl Prl<'_> {
19    /// Which kind of property this sprm modifies: 1 paragraph, 2 character, 3 picture,
20    /// 4 section, 5 table ([MS-DOC] Sprm.sgc).
21    pub(crate) fn sgc(&self) -> u8 {
22        ((self.op >> 10) & 7) as u8
23    }
24}
25
26/// Iterate the Prls of a grpprl. Malformed tails are dropped.
27pub(crate) fn iter(grpprl: &[u8]) -> impl Iterator<Item = Prl<'_>> {
28    let mut rest = grpprl;
29    std::iter::from_fn(move || {
30        if rest.len() < 2 {
31            return None;
32        }
33        let op = u16::from_le_bytes([rest[0], rest[1]]);
34        let after = &rest[2..];
35        // Operand size per spra, with the exceptions whose size byte lies.
36        let len = match (op >> 13, op) {
37            // PChgTabsOperand.cb is one byte, the size after it; 255 means "work it out" from the
38            // two counts: cb, cTabs, 4 bytes per removed stop, cTabs, 3 bytes per added stop.
39            (_, P_CHG_TABS) => match after.first().copied() {
40                Some(255) => after.get(1).and_then(|&del| {
41                    let add = after.get(2 + 4 * del as usize).copied()?;
42                    Some(1 + 1 + 4 * del as usize + 1 + 3 * add as usize)
43                }),
44                Some(cb) => Some(1 + cb as usize),
45                None => None,
46            },
47            // TDefTableOperand.cb counts the bytes after cb, incremented by 1, so the
48            // operand is one byte longer than cb.
49            (_, T_DEF_TABLE) => after.get(..2).map(|b| 1 + u16::from_le_bytes([b[0], b[1]]) as usize),
50            (0, _) | (1, _) => Some(1),
51            (2, _) | (4, _) | (5, _) => Some(2),
52            (3, _) => Some(4),
53            (7, _) => Some(3),
54            (6, _) => after.first().map(|&n| 1 + n as usize),
55            _ => None,
56        }?;
57        let operand = after.get(..len)?;
58        rest = &after[len.min(after.len())..];
59        Some(Prl { op, operand })
60    })
61}

63/// A ToggleOperand value: 0 = off, 1 = on, 0x80 = leave unchanged, 0x81 = invert.
64pub(crate) enum Toggle {
65    Off,
66    On,
67    Keep,
68    Invert,
69}
70
71pub(crate) fn toggle(p: &Prl) -> Toggle {
72    match p.operand.first().copied().unwrap_or(0) {
73        0x81 => Toggle::Invert,
74        0x80 => Toggle::Keep,
75        0x00 => Toggle::Off,
76        _ => Toggle::On,
77    }
78}

80/// The isprm indices of Prm0 that map to sprms we understand ([MS-DOC] §2.9.215 table).
81pub(crate) fn from_prm0(isprm: u8) -> Option<u16> {
82    Some(match isprm {
83        0x05 => 0x2461, // sprmPJc
84        0x07 => 0x2405, // sprmPFKeep
85        0x08 => 0x2406, // sprmPFKeepFollow
86        0x09 => 0x2407, // sprmPFPageBreakBefore
87        0x33 => 0x2435, // sprmPWidowControl (low byte of the operand is the value)
88        0x78 => 0x2640, // sprmPOutLvl
89        0x4D => 0x2A0C, // sprmCHighlight
90        0x55 => 0x0835, // sprmCFBold
91        0x56 => 0x0836, // sprmCFItalic
92        0x57 => 0x0837, // sprmCFStrike
93        0x58 => 0x0838, // sprmCFOutline
94        0x59 => 0x0839, // sprmCFShadow
95        0x5A => 0x083A, // sprmCFSmallCaps
96        0x5B => 0x083B, // sprmCFCaps
97        0x5C => 0x083C, // sprmCFVanish
98        0x5E => 0x2A3E, // sprmCKul
99        0x62 => 0x2A42, // sprmCIco
100        0x74 => 0x0854, // sprmCFImprint
101        0x75 => 0x0855, // sprmCFSpec
102        _ => return None,
103    })
