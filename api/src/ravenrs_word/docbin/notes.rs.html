1//! Footnotes, endnotes and bookmarks: `PlcffndRef`/`PlcffndTxt`, `PlcfendRef`/`PlcfendTxt`,
2//! and `SttbfBkmk` + `Plcfbkf`/`Plcfbkl` ([MS-DOC] §2.9).
3
4/// Cap on notes and bookmarks per document.
5const MAX: usize = 10_000;
6
7/// Parse the note reference/story PLCs. `ref_pair`/`txt_pair` are the FIB pair indices
8/// (`(2, 3)` for footnotes, `(46, 47)` for endnotes); story ranges are relative to the
9/// note subdocument.
10///
11/// The story CPs of a well-formed PLC ascend, so the stories are disjoint. A hostile table
12/// that steps back would make many notes read the same text again (each one materialised
13/// as its own part), so a story starting before the end of the last one kept is dropped:
14/// every subdocument CP becomes at most one note.
15pub(crate) fn parse_notes(table: &[u8], fib: &crate::docbin::fib::Fib, ref_pair: usize, txt_pair: usize) -> Vec<(u32, (u32, u32))> {
16    let refs = fib.pair(ref_pair).and_then(|(fc, lcb)| plc_of(table, fc, lcb, 2)).unwrap_or_default();
17    let txts = fib.pair(txt_pair).and_then(|(fc, lcb)| plc_of(table, fc, lcb, 0)).unwrap_or_default();
18    if txts.len() < 2 {
19        return Vec::new();
20    }
21    let mut out = Vec::new();
22    let mut kept_to = 0u32;
23    // The reference PLC's aFtnIdx order matches the story order; pair them by index.
24    for (i, &cp) in refs.iter().enumerate() {
25        if let (Some(&a), Some(&b)) = (txts.get(i), txts.get(i + 1))
26            && b > a
27            && a >= kept_to
28        {
29            out.push((cp, (a, b)));
30            kept_to = b;
31        }
32    }
33    out
34}

36/// The CPs of a PLC (the trailing element is a limit and is kept so callers can pair
37/// range ends) plus the data-element size check. Returns the CPs only.
38fn plc_of(table: &[u8], fc: u32, lcb: u32, data_size: usize) -> Option<Vec<u32>> {
39    if lcb == 0 {
40        return None;
41    }
42    let plc = table.get(fc as usize..(fc as usize).saturating_add(lcb as usize))?;
43    if plc.len() < 4 {
44        return None;
45    }
46    // aCP has one more element than aData: n = (len - data_size) / (4 + data_size).
47    let n = (plc.len() - data_size) / (4 + data_size);
48    if n == 0 || n > MAX {
49        return None;
50    }
51    let mut cps = Vec::with_capacity(n.min(MAX));
52    for i in 0..=n.min(MAX) {
53        match plc.get(i * 4..i * 4 + 4) {
54            Some(b) => cps.push(u32::from_le_bytes([b[0], b[1], b[2], b[3]])),
55            None => break,
56        }
57    }
58    Some(cps)
59}

61/// A bookmark boundary to insert while walking.
62pub(crate) enum Mark {
63    Start(String),
64    End(String),
65}
66
67/// Parse `SttbfBkmk` + `Plcfbkf` + `Plcfbkl` into (cp, event) pairs sorted for a sequential
68/// walk. Hidden bookmarks (leading `_`) are kept: the engine can decide.
69pub(crate) fn parse_bookmarks(table: &[u8], fib: &crate::docbin::fib::Fib) -> Vec<(u32, Mark)> {
70    let names = fib.pair(crate::docbin::fib::pair::STTBF_BKMK).and_then(|(fc, lcb)| sttb_strings(table, fc, lcb)).unwrap_or_default();
71    let starts = fib.pair(crate::docbin::fib::pair::PLCF_BKF).and_then(|(fc, lcb)| plc_of(table, fc, lcb, 4)).unwrap_or_default();
72    let ends = fib.pair(crate::docbin::fib::pair::PLCF_BKL).and_then(|(fc, lcb)| plc_of(table, fc, lcb, 2)).unwrap_or_default();
73    let n = starts.len().saturating_sub(1).min(ends.len().saturating_sub(1)).min(names.len()).min(MAX);
74    let mut out = Vec::with_capacity(n * 2);
75    for i in 0..n {
76        let (Some(&a), Some(&b)) = (starts.get(i), ends.get(i)) else { break };
77        if b < a {
78            continue;
79        }
80        if let Some(name) = names.get(i) {
81            if name.is_empty() {
82                continue;
83            }
84            out.push((a, Mark::Start(name.clone())));
85            out.push((b, Mark::End(name.clone())));
86        }
87    }
88    out.sort_by_key(|(cp, m)| (*cp, matches!(m, Mark::End(_))));
89    out
90}

92/// Strings of an STTB: extended (fExtend = 0xFFFF, 2-byte length and UTF-16 chars) or not
93/// (1-byte length and Windows-1252 chars). `cbExtra` bytes after each string are skipped.
94/// Names are kept to 40 characters, but the reader always advances by the stored length.
95fn sttb_strings(table: &[u8], fc: u32, lcb: u32) -> Option<Vec<String>> {
96    let b = table.get(fc as usize..(fc as usize).saturating_add(lcb as usize))?;
97    let extended = b.get(0..2)? == [0xFF, 0xFF];
98    let head = if extended { 2 } else { 0 };
99    let h = b.get(head..head + 4)?;
100    let c_data = u16::from_le_bytes([h[0], h[1]]) as usize;
101    let cb_extra = u16::from_le_bytes([h[2], h[3]]) as usize;
102    if c_data > MAX {
103        return None;
104    }
105    let mut at = head + 4;
106    let mut out = Vec::with_capacity(c_data);
107    for _ in 0..c_data {
108        let (cch, unit) = if extended {
109            match b.get(at..at + 2) {
110                Some(x) => (u16::from_le_bytes([x[0], x[1]]) as usize, 2),
111                None => break,
112            }
113        } else {
114            match b.get(at) {
115                Some(&x) => (x as usize, 1),
116                None => break,
117            }
118        };
119        at += unit;
120        let Some(chars) = b.get(at..at + cch * unit) else { break };
121        at += cch * unit;
122        // The string's extra data must be there too, or the table is cut short.
123        if b.get(at..at + cb_extra).is_none() {
124            break;
125        }
126        at += cb_extra;
127        let s: String = if extended {
128            chars.as_chunks::<2>().0.iter().take(40).map(|c| char::from_u32(u16::from_le_bytes(*c) as u32).unwrap_or('\u{FFFD}')).collect()
129        } else {
130            chars.iter().take(40).map(|&c| crate::doc::encoding::cp1252(c)).collect()
131        };
132        out.push(s);
133    }
134    Some(out)
135}

137#[cfg(test)]
138mod tests {
139    use super::sttb_strings;
140
141    fn ext_sttb(names: &[&str], extra: &[u8]) -> Vec<u8> {
142        let mut b = vec![0xFF, 0xFF];
143        b.extend_from_slice(&(names.len() as u16).to_le_bytes());
144        b.extend_from_slice(&(extra.len() as u16).to_le_bytes());
145        for n in names {
146            let units: Vec<u16> = n.encode_utf16().collect();
147            b.extend_from_slice(&(units.len() as u16).to_le_bytes());
148            for u in units {
149                b.extend_from_slice(&u.to_le_bytes());
150            }
151            b.extend_from_slice(extra);
152        }
153        b
154    }
155
156    #[test]
157    fn long_bookmark_name_does_not_desync_table() {
158        let long = "A".repeat(50);
159        let b = ext_sttb(&[&long, "bm2"], &[]);
160        let got = sttb_strings(&b, 0, b.len() as u32).unwrap();
161        assert_eq!(got, vec!["A".repeat(40), "bm2".to_string()]);
162    }
163
164    #[test]
165    fn sttb_extra_data_is_skipped() {
166        let b = ext_sttb(&["first", "bm2"], &[1, 2, 3, 4]);
167        let got = sttb_strings(&b, 0, b.len() as u32).unwrap();
168        assert_eq!(got, vec!["first".to_string(), "bm2".to_string()]);
169    }
170
171    #[test]
172    fn a_name_without_its_extra_data_is_dropped() {
173        let mut b = ext_sttb(&["first", "bm2"], &[1, 2, 3, 4]);
174        b.truncate(b.len() - 2);
175        let got = sttb_strings(&b, 0, b.len() as u32).unwrap();
176        assert_eq!(got, vec!["first".to_string()]);
177    }
178
179    #[test]
180    fn non_extended_sttb_reads_byte_strings() {
181        let mut b = Vec::new();
182        b.extend_from_slice(&2u16.to_le_bytes());
183        b.extend_from_slice(&2u16.to_le_bytes());
184        for n in ["one", "caf\u{e9}"] {
185            let bytes: Vec<u8> = n.chars().map(|c| c as u8).collect();
186            b.push(bytes.len() as u8);
187            b.extend_from_slice(&bytes);
188            b.extend_from_slice(&[9, 9]);
189        }
190        let got = sttb_strings(&b, 0, b.len() as u32).unwrap();
191        assert_eq!(got, vec!["one".to_string(), "caf\u{e9}".to_string()]);
192    }
