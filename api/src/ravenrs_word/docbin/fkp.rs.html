1//! Character and paragraph formatting lookup: the bin tables (`PlcBteChpx`/`PlcBtePapx`) in
2//! the Table stream point at 512-byte FKP pages in the WordDocument stream, which map FC
3//! ranges to CHPX (grpprls) and PAPX (istd + grpprl) ([MS-DOC] §2.4.6.1/2.4.6.2).
4
5use crate::docbin::DocbinError;
6
7/// Largest bin table (entries) we accept.
8const MAX_BINS: usize = 100_000;
9
10/// A Chpx grpprl and the FCs `[lo, hi)` for which it is the one [`Bins::chpx`] returns.
11#[derive(Clone, Copy)]
12pub(crate) struct ChpxRun<'w> {
13    pub(crate) grpprl: &'w [u8],
14    lo: u32,
15    hi: u32,
16}
17
18impl ChpxRun<'_> {
19    pub(crate) fn contains(&self, fc: u32) -> bool {
20        self.lo <= fc && fc < self.hi
21    }
22}
23
24/// A parsed `PlcBte*`: FC breakpoints and the FKP page each range lives in.
25pub(crate) struct Bins {
26    fcs: Vec<u32>,
27    /// `fcs` ascends, so the bin of an FC holds every FC up to the next breakpoint (a corrupt
28    /// table that does not gets no cached ranges, [`ChpxRun`]).
29    sorted: bool,
30    /// Page number (× 512 = byte offset in the WordDocument stream).
31    pns: Vec<u32>,
32}
33
34impl Bins {
35    /// Parse the Plc at `table[fc..fc+lcb]` (aFc[n+1] u32s + aPn[n] u32s).
36    pub(crate) fn parse(table: &[u8], fc: u32, lcb: u32) -> Result<Bins, DocbinError> {
37        let fc = fc as usize;
38        let lcb = lcb as usize;
39        if lcb < 4 || !(lcb - 4).is_multiple_of(8) {
40            return Ok(Bins { fcs: Vec::new(), sorted: true, pns: Vec::new() });
41        }
42        let end = fc.checked_add(lcb).ok_or_else(|| DocbinError::Malformed("bin table overflow".into()))?;
43        let plc = table.get(fc..end).ok_or_else(|| DocbinError::Malformed("bin table outside the Table stream".into()))?;
44        let n = (lcb - 4) / 8;
45        if n > MAX_BINS {
46            return Err(DocbinError::Limit(format!("bin table has {n} entries")));
47        }
48        let at = |i: usize| -> Option<u32> { plc.get(i * 4..i * 4 + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) };
49        let mut fcs = Vec::with_capacity(n + 1);
50        for i in 0..=n {
51            fcs.push(at(i).ok_or_else(|| DocbinError::Malformed("truncated bin aFc".into()))?);
52        }
53        let mut pns = Vec::with_capacity(n);
54        for i in 0..n {
55            // PnBte* keeps the page number in its low bits; the rest must be zero.
56            pns.push(at(n + 1 + i).ok_or_else(|| DocbinError::Malformed("truncated bin aPn".into()))? & 0x003F_FFFF);
57        }
58        let sorted = fcs.windows(2).all(|w| w.first() <= w.get(1));
59        Ok(Bins { fcs, sorted, pns })
60    }
61
62    /// The page containing `fc`, as a slice of the WordDocument stream, and the FC range
63    /// `[lo, hi)` of its bin (`hi` is `u32::MAX` for the last; an empty range when the table
64    /// is not ascending, so nothing is cached from it).
65    fn page<'w>(&self, word: &'w [u8], fc: u32) -> Option<(&'w [u8], u32, u32)> {
66        let i = self.fcs.partition_point(|&f| f <= fc).checked_sub(1)?;
67        let pn = *self.pns.get(i)?;
68        let at = pn.checked_mul(512)? as usize;
69        let page = word.get(at..at.checked_add(512)?)?;
70        if !self.sorted {
71            return Some((page, 1, 0));
72        }
73        let lo = self.fcs.get(i).copied().unwrap_or(0);
74        let hi = self.fcs.get(i + 1).copied().unwrap_or(u32::MAX);
75        Some((page, lo, hi))
76    }
77
78    /// The grpprl of the Chpx covering `fc` ([MS-DOC] ChpxFkp: rgfc u32s, rgb offsets ×2,
79    /// crun in the last byte).
80    pub(crate) fn chpx<'w>(&self, word: &'w [u8], fc: u32) -> &'w [u8] {
81        self.chpx_run(word, fc).grpprl
82    }
83
84    /// [`Bins::chpx`] with the FC range over which the same grpprl is the answer, so a walk
85    /// over consecutive characters looks it up once per run, not once per character.
86    pub(crate) fn chpx_run<'w>(&self, word: &'w [u8], fc: u32) -> ChpxRun<'w> {
87        let none = ChpxRun { grpprl: &[], lo: 1, hi: 0 };
88        let Some((page, page_lo, page_hi)) = self.page(word, fc) else { return none };
89        let Some(&crun) = page.get(511) else { return none };
90        let crun = crun.min(0x65) as usize;
91        let rgfc = |i: usize| -> Option<u32> { page.get(i * 4..i * 4 + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) };
92        // rgb array sits right after rgfc (crun+1 u32s); each byte is a half-offset into the
93        // region between rgb's end and crun.
94        let rgb_at = (crun + 1) * 4;
95        // The last run starting at or before `fc`. It stays the answer for every FC from its own
96        // start up to the smallest start after it (also in a page whose starts do not ascend).
97        let j = (0..crun).rev().find(|&j| rgfc(j).is_some_and(|f| f <= fc));
98        let first_after = j.map_or(0, |j| j + 1);
99        let hi = (first_after..crun).filter_map(rgfc).min().map_or(page_hi, |m| m.min(page_hi));
100        let lo = j.and_then(rgfc).map_or(page_lo, |f| f.max(page_lo));
101        let run = |grpprl: &'w [u8]| ChpxRun { grpprl, lo, hi };
102        let Some(j) = j else { return run(&[]) };
103        let Some(&off) = page.get(rgb_at + j) else { return run(&[]) };
104        if off == 0 {
105            return run(&[]);
106        }
107        let at = off as usize * 2;
108        let Some(&cb) = page.get(at) else { return run(&[]) };
109        run(page.get(at + 1..at + 1 + cb as usize).unwrap_or(&[]))
110    }
111
112    /// The (istd, grpprl) of the Papx of the paragraph whose mark is just before `fc`
113    /// ([MS-DOC] PapxFkp: rgfc u32s — each the FC *following* a paragraph mark —, rgbx BX
114    /// entries (1 offset byte + 12 PHE bytes), crun in the last byte; the PAPX itself is a
115    /// PapxInFkp with the cb/cb' size quirk). Query with the FC of the character after the mark.
116    pub(crate) fn papx<'w>(&self, word: &'w [u8], fc: u32) -> (u16, &'w [u8]) {
117        let Some((page, _, _)) = self.page(word, fc) else { return (0, &[]) };
118        let Some(&crun) = page.get(511) else { return (0, &[]) };
119        // BX is 13 bytes; cap crun at what fits before byte 511.
120        let max_run = (511usize.saturating_sub((crun as usize + 1) * 4)) / 13;
121        let crun = (crun as usize).min(max_run);
122        let rgfc = |i: usize| -> Option<u32> { page.get(i * 4..i * 4 + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) };
123        let rgbx_at = (crun + 1) * 4;
124        let j = (0..crun).rev().find(|&j| rgfc(j).is_some_and(|f| f <= fc));
125        let Some(j) = j else { return (0, &[]) };
126        let Some(&off) = page.get(rgbx_at + j * 13) else { return (0, &[]) };
127        if off == 0 {
128            return (0, &[]);
129        }
130        let at = off as usize * 2;
131        let u16at = |o: usize| -> Option<u16> { page.get(o..o + 2).map(|b| u16::from_le_bytes([b[0], b[1]])) };
132        let Some(cb) = page.get(at).copied() else { return (0, &[]) };
133        // PapxInFkp: if cb is nonzero the GrpPrlAndIstd (istd u16 + grpprl) is 2×cb-1 bytes;
134        // if cb is zero, the next byte cb' (1 byte) gives 2×cb' bytes of it.
135        let (data_at, data_len) = if cb != 0 {
136            (at + 1, 2 * cb as usize - 1)
137        } else {
138            let cb2 = page.get(at + 1).copied().unwrap_or(0) as usize;
139            (at + 2, 2 * cb2)
140        };
141        if data_len < 2 {
142            return (0, &[]);
143        }
144        let istd = u16at(data_at).unwrap_or(0);
145        (istd, page.get(data_at + 2..data_at + data_len).unwrap_or(&[]))
146    }
