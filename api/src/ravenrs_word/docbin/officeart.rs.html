1//! OfficeArt (Escher) drawing records, as Word 97-2003 stores them ([MS-ODRAW], public Microsoft
2//! Open Specifications): the record header, the BLIP store and its pictures, shape containers with
3//! their property tables, and the `OfficeArtContent` of the Table stream ([MS-DOC] §2.9.172).
4//!
5//! Everything here walks hostile bytes: record lengths are clamped to their parent, nesting depth
6//! and record counts are capped, and decompressed metafiles are size-limited.
7

15/// Deepest container nesting walked (groups inside groups…).
16const MAX_DEPTH: usize = 12;
17/// Most records walked in one drawing tree or BLIP store.
18const MAX_RECORDS: usize = 200_000;
19/// Most BLIP store entries kept.
20const MAX_BSE: usize = 10_000;

24const MAX_METAFILE: usize = 64 << 20;
25/// Largest DIB converted to PNG, per side (pixels).
26const MAX_DIB_SIDE: u32 = 16_384;

53/// One record: header fields and its body (clamped to what the parent holds).
54#[derive(Clone, Copy, Debug)]
55pub(crate) struct Rec<'a> {
56    pub(crate) inst: u16,
57    pub(crate) typ: u16,
58    pub(crate) body: &'a [u8],
59}
60
61/// The record at the start of `buf` and the bytes it spans (header included). A body longer
62/// than `buf` is cut to what is there.
63pub(crate) fn record(buf: &[u8]) -> Option<(Rec<'_>, usize)> {
64    let h = buf.get(0..8)?;
65    let ver_inst = u16::from_le_bytes([h[0], h[1]]);
66    let typ = u16::from_le_bytes([h[2], h[3]]);
67    let len = u32::from_le_bytes([h[4], h[5], h[6], h[7]]) as usize;
68    let rest = buf.get(8..)?;
69    let len = len.min(rest.len());
70    let body = rest.get(..len)?;
71    Some((Rec { inst: ver_inst >> 4, typ, body }, 8 + len))
72}
73
74/// The sibling records of a container body, at most `budget` of them (the budget is shared
75/// across a whole walk, so a deep or wide tree cannot multiply the work).
76pub(crate) fn children<'a>(body: &'a [u8], budget: &mut usize) -> Vec<Rec<'a>> {
77    let mut out = Vec::new();
78    let mut at = 0usize;
79    while *budget > 0 {
80        let Some((rec, used)) = body.get(at..).and_then(record) else { break };
81        *budget -= 1;
82        out.push(rec);
83        at = at.saturating_add(used);
84    }
85    out
86}
87
88fn u16_at(b: &[u8], at: usize) -> Option<u16> {
89    Some(u16::from_le_bytes(b.get(at..at.checked_add(2)?)?.try_into().ok()?))
90}
91
92fn u32_at(b: &[u8], at: usize) -> Option<u32> {
93    Some(u32::from_le_bytes(b.get(at..at.checked_add(4)?)?.try_into().ok()?))
94}
95
96fn i32_at(b: &[u8], at: usize) -> Option<i32> {
97    u32_at(b, at).map(|v| v as i32)
98}

112/// The picture of an `OfficeArtFBSE` body: its embedded BLIP, or the BLIP stored at `foDelay`
113/// in `delay` (the WordDocument stream for the document's BLIP store, the Data stream for
114/// inline pictures). `limit` caps the bytes produced.
115#[cfg(test)]
116pub(crate) fn fbse_blip(fbse: &[u8], delay: &[u8], limit: usize, work: &mut Decode) -> Option<Blip> {
117    let (rec, _) = fbse_record(fbse, delay)?;
118    blip(&rec, limit, work)
119}

121/// The document's decode budget: bytes of inflating and pixel decoding left, and how many
122/// attempts it has refused (only the first few are logged).
123pub(crate) struct Decode {
124    pub(crate) left: usize,
125    refused: usize,
126}
127
128impl Decode {
129    pub(crate) fn new(left: usize) -> Self {
130        Decode { left, refused: 0 }
131    }
132
133    /// Take `cost` off the budget; `false` (and a log line) when it doesn't fit.
134    fn charge(&mut self, cost: usize, what: &str) -> bool {
135        match self.left.checked_sub(cost) {
136            Some(left) if cost > 0 => {
137                self.left = left;
138                true
139            }
140            _ => {
141                if self.refused < 10 {
142                    log::warn!(
143                        "docbin: a {what} needing {cost} bytes of decode work is over the decode budget ({} left) and was not decoded",
144                        self.left
145                    );
146                }
147                self.refused = self.refused.saturating_add(1);
148                false
149            }
150        }
151    }
152}

154/// The BLIP record of an `OfficeArtFBSE`: embedded after it, or in the delay stream, where
155/// its (`foDelay`, record length) location is returned too (entries can share one record,
156/// whatever sizes their FBSEs claim: the record's own header is what counts).
157pub(crate) fn fbse_record<'a>(fbse: &'a [u8], delay: &'a [u8]) -> Option<(Rec<'a>, Option<(usize, usize)>)> {
158    // btWin32, btMacOS, rgbUid[16], tag, size, cRef, foDelay, unused, cbName, unused×2.
159    let fo_delay = u32_at(fbse, 28)? as usize;
160    let cb_name = *fbse.get(33)? as usize;
161    let at = 36usize.checked_add(cb_name)?;
162    if let Some((rec, _)) = fbse.get(at..).and_then(record) {
163        return Some((rec, None));
164    }
165    // Not embedded: the record sits in the delay stream.
166    let (rec, used) = record(delay.get(fo_delay..)?)?;
167    Some((rec, Some((fo_delay, used))))
168}

170/// Decode one `OfficeArtBlip*` record ([MS-ODRAW] §2.2.23-§2.2.31). `work` is the
171/// document's decode budget: inflating a metafile or decoding a DIB is charged to it before
172/// it runs, whether it succeeds or not, and nothing is decoded once it is spent.
173pub(crate) fn blip(rec: &Rec, limit: usize, work: &mut Decode) -> Option<Blip> {
174    // Each type has two instance values; the odd one carries a second 16-byte UID.
175    let uids = if rec.inst & 1 == 1 { 32 } else { 16 };
176    match rec.typ {
177        rt::BLIP_PNG | rt::BLIP_JPEG | rt::BLIP_JPEG_CMYK | rt::BLIP_DIB | rt::BLIP_TIFF => {
178            let data = rec.body.get(uids + 1..)?; // + tag byte
179            if data.is_empty() || data.len() > limit {
180                return None;
181            }
182            match rec.typ {
183                rt::BLIP_PNG => Some(Blip { bytes: data.to_vec(), ext: "png", placeholder: false }),
184                rt::BLIP_JPEG | rt::BLIP_JPEG_CMYK => Some(Blip { bytes: data.to_vec(), ext: "jpeg", placeholder: false }),
185                rt::BLIP_TIFF => Some(Blip { bytes: data.to_vec(), ext: "tiff", placeholder: false }),
186                _ => dib_to_png(data, limit, work),
187            }
188        }
189        rt::BLIP_EMF | rt::BLIP_WMF | rt::BLIP_PICT => {
190            // OfficeArtMetafileHeader: cbSize, rcBounds, ptSize, cbSave, compression, filter.
191            let hdr = rec.body.get(uids..uids + 34)?;
192            let cb_size = u32_at(hdr, 0)? as usize;
193            let cb_save = u32_at(hdr, 28)? as usize;
194            let compression = *hdr.get(32)?;
195            let raw = rec.body.get(uids + 34..)?;
196            let raw = raw.get(..cb_save.min(raw.len()))?;
197            let cap = limit.min(MAX_METAFILE);
198            let bytes = if compression == 0 {
199                let want = if cb_size > 0 { cb_size.min(cap) } else { cap };
200                // The compressed stream is read whole whatever it inflates to: a record that
201                // claims to inflate to nothing still costs its input (#3871).
202                if !work.charge(want.max(raw.len()), "metafile") {
203                    return None;
204                }
205                miniz_oxide::inflate::decompress_to_vec_zlib_with_limit(raw, want).ok()?
206            } else {
207                if raw.len() > cap {
208                    return None;
209                }
210                raw.to_vec()
211            };
212            if bytes.is_empty() {
213                return None;
214            }
215            match rec.typ {
216                rt::BLIP_EMF => Some(Blip { bytes, ext: "emf", placeholder: false }),
217                rt::BLIP_WMF => Some(Blip { bytes, ext: "wmf", placeholder: false }),
218                _ => Some(Blip { bytes, ext: "pict", placeholder: true }),
219            }
220        }
221        _ => None,
222    }
223}

225/// A device-independent bitmap (BITMAPINFOHEADER + palette + pixels) as a PNG: the DIB gets
226/// the 14-byte BMP file header it lacks, is decoded with size limits and re-encoded. When the
227/// bitmap can't be decoded, or its declared pixels (4 bytes each) don't fit in `work`, the
228/// BMP bytes are kept (and render as a grey placeholder).
229fn dib_to_png(dib: &[u8], limit: usize, work: &mut Decode) -> Option<Blip> {
230    let header = u32_at(dib, 0)? as usize;
231    if !(12..=124).contains(&header) {
232        return None;
233    }
234    let (bits, used, compression) =
235        if header == 12 { (u16_at(dib, 10)? as usize, 0, 0) } else { (u16_at(dib, 14)? as usize, u32_at(dib, 32)? as usize, u32_at(dib, 16)?) };
236    let entry = if header == 12 { 3 } else { 4 };
237    let colors = if used > 0 {
238        used.min(256)
239    } else if bits <= 8 {
240        1usize << bits
241    } else {
242        0
243    };
244    // BI_BITFIELDS masks follow a 40-byte header.
245    let masks = if header == 40 && (compression == 3 || compression == 6) { if compression == 3 { 12 } else { 16 } } else { 0 };
246    let off_bits = 14usize.checked_add(header)?.checked_add(colors * entry)?.checked_add(masks)?;
247    let file_size = 14usize.checked_add(dib.len()).filter(|&size| size <= limit)?;
248    // The declared size is what decoding allocates, however small the (RLE) data is.
249    let (w, h) = if header == 12 {
250        (u64::from(u16_at(dib, 4)?), u64::from(u16_at(dib, 6)?))
251    } else {
252        (u64::from(u32_at(dib, 4)?), u64::from((u32_at(dib, 8)? as i32).unsigned_abs()))
253    };
254    let cost = w.saturating_mul(h).saturating_mul(4).try_into().unwrap_or(usize::MAX);
255    // Copying the DIB into a BMP is work too, however few pixels it declares: it is charged
256    // before the copy is made (#3871). A bitmap whose pixels don't fit still keeps its BMP
257    // bytes as a placeholder, if the copy fits.
258    let decode = work.charge(cost.max(dib.len()), "bitmap");
259    if !decode && !work.charge(dib.len(), "bitmap") {
260        return None;
261    }
262    let mut bmp = Vec::with_capacity(file_size);
263    bmp.extend_from_slice(b"BM");
264    bmp.extend_from_slice(&u32::try_from(file_size).ok()?.to_le_bytes());
265    bmp.extend_from_slice(&[0, 0, 0, 0]);
266    bmp.extend_from_slice(&u32::try_from(off_bits).ok()?.to_le_bytes());
267    bmp.extend_from_slice(dib);
268    let placeholder = |bmp| Some(Blip { bytes: bmp, ext: "bmp", placeholder: true });
269    if !decode {
270        return placeholder(bmp);
271    }
272    let mut reader = image::ImageReader::new(std::io::Cursor::new(&bmp));
273    reader.set_format(image::ImageFormat::Bmp);
274    let mut limits = image::Limits::default();
275    limits.max_image_width = Some(MAX_DIB_SIDE);
276    limits.max_image_height = Some(MAX_DIB_SIDE);
277    limits.max_alloc = Some((256 << 20).min(limit as u64));
278    reader.limits(limits);
279    let mut png = PictureWriter { bytes: Vec::new(), limit };
280    match reader.decode() {
281        Ok(img)
282            if image::codecs::png::PngEncoder::new(&mut png).write_image(img.as_bytes(), img.width(), img.height(), img.color().into()).is_ok() =>
283        {
284            Some(Blip { bytes: png.bytes, ext: "png", placeholder: false })
285        }
286        _ => placeholder(bmp),
287    }
288}

290/// Cap encoded bytes while the PNG encoder writes, before the buffer can grow.
291struct PictureWriter {
292    bytes: Vec<u8>,
293    limit: usize,
294}
295
296impl Write for PictureWriter {
297    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
298        let Some(end) = self.bytes.len().checked_add(buf.len()).filter(|&end| end <= self.limit) else {
299            return Err(std::io::Error::new(std::io::ErrorKind::InvalidData, "picture exceeds remaining media budget"));
300        };
301        if end > self.bytes.capacity() {
302            let new_capacity = self.bytes.capacity().saturating_mul(2).max(end).min(self.limit);
303            self.bytes.try_reserve_exact(new_capacity - self.bytes.len()).map_err(std::io::Error::other)?;
304        }
305        self.bytes.extend_from_slice(buf);
306        Ok(buf.len())
307    }
308
309    fn flush(&mut self) -> std::io::Result<()> {
310        Ok(())
311    }
312}

317/// The shape properties we use, from `OfficeArtFOPT` and its secondary/tertiary tables
318/// ([MS-ODRAW] §2.3).
319#[derive(Clone, Debug, Default, PartialEq)]
320pub(crate) struct Props {
321    /// `pib`: 1-based index of the picture in the BLIP store.
322    pub(crate) pib: Option<u32>,
323    /// Rotation, degrees clockwise.
324    pub(crate) rot: f32,
325    /// Crop fractions (left, top, right, bottom).
326    pub(crate) crop: [f32; 4],
327    pub(crate) fill_color: Option<Rgb>,
328    pub(crate) filled: Option<bool>,
329    pub(crate) line_color: Option<Rgb>,
330    /// Line width in EMUs.
331    pub(crate) line_width: Option<u32>,
332    pub(crate) line: Option<bool>,
333    /// An arrowhead at either end of a line.
334    pub(crate) arrowhead: bool,
335    /// `posh`/`posrelh`/`posv`/`posrelv`: alignment and reference frame (Word 2000 and later).
336    pub(crate) posh: Option<u32>,
337    pub(crate) posrelh: Option<u32>,
338    pub(crate) posv: Option<u32>,
339    pub(crate) posrelv: Option<u32>,
340    /// Distance from wrapped text (left, top, right, bottom), EMUs.
341    pub(crate) wrap_dist: [Option<u32>; 4],
342    /// `wzDescription`: the alternative text.
343    pub(crate) alt: String,
344}
345
346impl Props {
347    /// Apply one property table (`OfficeArtFOPT` or a secondary/tertiary one): `count` 6-byte
348    /// entries, then the complex data of the entries flagged complex, in order.
349    pub(crate) fn apply(&mut self, body: &[u8], count: u16) {
350        let n = (count as usize).min(body.len() / 6);
351        let mut complex_at = n * 6;
352        for i in 0..n {
353            let Some(opid) = u16_at(body, i * 6) else { break };
354            let Some(op) = u32_at(body, i * 6 + 2) else { break };
355            let pid = opid & 0x3FFF;
356            if opid & 0x8000 != 0 {
357                let start = complex_at;
358                complex_at = complex_at.saturating_add(op as usize);
359                if pid == 0x0381 {
360                    let end = complex_at.min(body.len());
361                    if let Some(data) = body.get(start..end) {
362                        let units: Vec<u16> =
363                            data.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).take_while(|u| *u != 0).take(4096).collect();
364                        self.alt = String::from_utf16_lossy(&units);
365                    }
366                }
367                continue;
368            }
369            let fixed = |v: u32| (v as i32) as f32 / 65_536.0;
370            match pid {
371                0x0004 => self.rot = fixed(op),
372                0x0100..=0x0103 => {
373                    // cropFromTop, cropFromBottom, cropFromLeft, cropFromRight.
374                    let slot = [1, 3, 0, 2][(pid - 0x0100) as usize];
375                    let v = fixed(op);
376                    self.crop[slot] = if v.is_finite() { v.clamp(0.0, 0.99) } else { 0.0 };
377                }
378                0x0104 => self.pib = Some(op),
379                0x0181 => self.fill_color = color(op),
380                // fFilled (bit 4) is valid when fUsefFilled (bit 20) is set.
381                0x01BF if op & (1 << 20) != 0 => self.filled = Some(op & (1 << 4) != 0),
382                0x01BF => {}
383                0x01C0 => self.line_color = color(op),
384                0x01CB => self.line_width = Some(op),
385                0x01D0 | 0x01D1 => self.arrowhead |= op != 0,
386                // fLine (bit 3) is valid when fUsefLine (bit 19) is set.
387                0x01FF if op & (1 << 19) != 0 => self.line = Some(op & (1 << 3) != 0),
388                0x01FF => {}
389                0x0384..=0x0387 => self.wrap_dist[(pid - 0x0384) as usize] = Some(op),
390                0x038F => self.posh = Some(op),
391                0x0390 => self.posrelh = Some(op),
392                0x0391 => self.posv = Some(op),
393                0x0392 => self.posrelv = Some(op),
394                _ => {}
395            }
396        }
397    }
398}

400/// An `OfficeArtCOLORREF` holding a literal RGB value (palette, scheme and system colour
401/// references are not resolved).
402fn color(v: u32) -> Option<Rgb> {
403    let [r, g, b, flags] = v.to_le_bytes();
404    // fPaletteIndex, fSchemeIndex, fSysIndex.
405    if flags & 0x19 != 0 {
406        return None;
407    }
408    Some(Rgb(r, g, b))
409}

411/// One shape (`OfficeArtSpContainer`).
412#[derive(Clone, Debug, Default)]
413pub(crate) struct Sp {
414    pub(crate) spid: u32,
415    /// Shape type (`MSOSPT`), from the FSP's instance.
416    pub(crate) spt: u16,
417    /// FSP flags: fGroup, fChild, fPatriarch, fDeleted, fOleShape, fHaveMaster, fFlipH, fFlipV…
418    pub(crate) flags: u32,
419    pub(crate) props: Props,
420    /// `OfficeArtChildAnchor` (left, top, right, bottom) in the parent group's space.
421    pub(crate) child: Option<[i32; 4]>,
422    /// `OfficeArtFSPGR`: a group's own coordinate space.
423    pub(crate) group_space: Option<[i32; 4]>,
424    /// Has an `OfficeArtClientTextbox` (its text is in the text box story).
425    pub(crate) textbox: bool,
426}
427
428impl Sp {
429    pub(crate) fn flip_h(&self) -> bool {
430        self.flags & (1 << 6) != 0
431    }
432    pub(crate) fn flip_v(&self) -> bool {
433        self.flags & (1 << 7) != 0
434    }
435    fn deleted(&self) -> bool {
436        self.flags & (1 << 3) != 0
437    }
438}

440/// A shape or a group of shapes.
441#[derive(Clone, Debug)]
442pub(crate) enum Node {
443    Shape(Sp),
444    /// A group: its own shape record (FSPGR, anchor) and members.
445    Group(Sp, Vec<Node>),
446}
447
448impl Node {
449    pub(crate) fn sp(&self) -> &Sp {
450        match self {
451            Node::Shape(sp) | Node::Group(sp, _) => sp,
452        }
453    }
454}
455
456/// Parse one `OfficeArtSpContainer`.
457pub(crate) fn shape(body: &[u8], budget: &mut usize) -> Sp {
458    let mut sp = Sp::default();
459    for rec in children(body, budget) {
460        match rec.typ {
461            rt::FSP => {
462                sp.spt = rec.inst;
463                sp.spid = u32_at(rec.body, 0).unwrap_or(0);
464                sp.flags = u32_at(rec.body, 4).unwrap_or(0);
465            }
466            rt::FOPT | rt::SECONDARY_FOPT | rt::TERTIARY_FOPT => sp.props.apply(rec.body, rec.inst),
467            rt::FSPGR | rt::CHILD_ANCHOR => {
468                let r = [0, 4, 8, 12].map(|o| i32_at(rec.body, o));
469                if let [Some(l), Some(t), Some(r), Some(b)] = r {
470                    if rec.typ == rt::FSPGR {
471                        sp.group_space = Some([l, t, r, b]);
472                    } else {
473                        sp.child = Some([l, t, r, b]);
474                    }
475                }
476            }
477            rt::CLIENT_TEXTBOX => sp.textbox = true,
478            _ => {}
479        }
480    }
481    sp
482}

484/// Parse an `OfficeArtSpgrContainer`: the first shape is the group itself, the rest its members.
485fn group(body: &[u8], depth: usize, budget: &mut usize) -> Option<Node> {
486    let recs = children(body, budget);
487    let mut it = recs.iter();
488    let head = it.find(|r| r.typ == rt::SP_CONTAINER).map(|r| shape(r.body, budget))?;
489    let mut members = Vec::new();
490    for rec in it {
491        if let Some(n) = node(rec, depth + 1, budget) {
492            members.push(n);
493        }
494    }
495    Some(Node::Group(head, members))
496}
497
498/// A shape or group record as a node (`None` for other records, deleted shapes or past the
499/// depth cap).
500fn node(rec: &Rec, depth: usize, budget: &mut usize) -> Option<Node> {
501    match rec.typ {
502        rt::SP_CONTAINER => Some(shape(rec.body, budget)).filter(|s| !s.deleted()).map(Node::Shape),
503        rt::SPGR_CONTAINER if depth < MAX_DEPTH => group(rec.body, depth, budget),
504        _ => None,
505    }
506}

508/// The document's drawings: the BLIP store and the top-level shapes of each drawing by spid.
509#[derive(Default)]
510pub(crate) struct Content<'a> {
511    /// `OfficeArtFBSE` bodies, in BLIP store order (`pib` 1 = the first).
512    pub(crate) bstore: Vec<&'a [u8]>,
513    /// The main document's shapes (`dgglbl` 0) and the headers' (`dgglbl` 1), by spid.
514    pub(crate) main: HashMap<u32, Node>,
515    pub(crate) header: HashMap<u32, Node>,
516}
517
518/// Parse `OfficeArtContent` (the `fcDggInfo` blob of the Table stream): an
519/// `OfficeArtDggContainer` followed by (`dgglbl`, `OfficeArtDgContainer`) drawings.
520pub(crate) fn content(blob: &[u8]) -> Content<'_> {
521    let mut out = Content::default();
522    let mut budget = MAX_RECORDS;
523    let Some((dgg, mut at)) = record(blob) else { return out };
524    if dgg.typ == rt::DGG_CONTAINER {
525        for rec in children(dgg.body, &mut budget) {
526            if rec.typ == rt::BSTORE_CONTAINER {
527                out.bstore = children(rec.body, &mut budget).into_iter().filter(|r| r.typ == rt::FBSE).map(|r| r.body).take(MAX_BSE).collect();
528            }
529        }
530    }
531    for _ in 0..MAX_DRAWINGS {
532        let Some(&label) = blob.get(at) else { break };
533        let Some((dg, used)) = blob.get(at + 1..).and_then(record) else { break };
534        at = at.saturating_add(1 + used);
535        if dg.typ != rt::DG_CONTAINER {
536            continue;
537        }
538        let into = if label == 0 { &mut out.main } else { &mut out.header };
539        for rec in children(dg.body, &mut budget) {
540            if rec.typ != rt::SPGR_CONTAINER {
541                continue;
542            }
543            // The top group: its first shape is the patriarch; the members are the drawing's shapes.
544            if let Some(Node::Group(_, members)) = group(rec.body, 0, &mut budget) {
545                for n in members {
546                    into.entry(n.sp().spid).or_insert(n);
547                }
548            }
549        }
550    }
551    out
552}
