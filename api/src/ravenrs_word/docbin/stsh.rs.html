1//! The stylesheet (`STSH`) and the font table (`SttbfFfn`) from the Table stream
2//! ([MS-DOC] §2.9.271 and §2.9.247).
3
4use crate::docbin::DocbinError;
5
6/// Largest stylesheet we accept (styles are 0x0FFE max by the spec; stay stricter).
7const MAX_STYLES: usize = 4096;
8
9/// One raw style record (`STD`).
10#[derive(Clone, Debug, Default)]
11pub(crate) struct RawStyle {
12    /// 1 paragraph, 2 character, 3 table, 4 numbering.
13    pub(crate) stk: u8,
14    /// Zero-based index of the parent style; 0x0FFF = none.
15    pub(crate) istd_base: u16,
16    pub(crate) name: String,
17    /// Paragraph-formatting UPX grpprl (paragraph styles).
18    pub(crate) papx: Vec<u8>,
19    /// Character-formatting UPX grpprl (paragraph and character styles).
20    pub(crate) chpx: Vec<u8>,
21}
22
23/// The stylesheet's standard fonts, the ones text takes when neither its style nor its own
24/// formatting names a font: font-table indexes of the ASCII, East Asian and other
25/// (non-East-Asian, non-ASCII) fonts (`Stshif.rgftcStandardChpStsh`, [MS-DOC] §2.9.273) and the
26/// complex-script font (`Stshi.ftcBi`, §2.9.272). `None` where the header is too short for one.
27#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
28pub(crate) struct StandardFonts {
29    pub(crate) ascii: Option<u16>,
30    pub(crate) east_asian: Option<u16>,
31    pub(crate) other: Option<u16>,
32    pub(crate) bidi: Option<u16>,
33}
34
35/// `StshiLsd.cbLSD`: the size of a latent style's `LSD` ([MS-DOC] `StshiLsd`).
36const LSD_SIZE: u16 = 4;
37
38/// The [`StandardFonts`] in the header (`Stshi`) of the `STSH` at `table[fc..fc+lcb]`: Word 365
39/// writes Aptos there while the font table's first entry is still Times New Roman (#5066).
40pub(crate) fn standard_fonts(table: &[u8], fc: u32, lcb: u32) -> StandardFonts {
41    let Some(stsh) = (fc as usize).checked_add(lcb as usize).and_then(|end| table.get(fc as usize..end)) else { return StandardFonts::default() };
42    let cb_stshi = stsh.get(0..2).map_or(0, |b| u16::from_le_bytes([b[0], b[1]]) as usize);
43    let stshi = stsh.get(2..).map_or(&[][..], |s| s.get(..cb_stshi.min(s.len())).unwrap_or(&[]));
44    // Stshif: cstd, cbSTDBaseInFile, flags, stiMaxWhenSaved, istdMaxFixedWhenSaved,
45    // nVerBuiltInNamesWhenSaved (2 bytes each), then rgftcStandardChpStsh (3 × 2); ftcBi follows.
46    let at = |o: usize| stshi.get(o..o + 2).map(|b| u16::from_le_bytes([b[0], b[1]]));
47    // ftcBi is optional, and the latent style data (`StshiLsd`, whose `cbLSD` MUST be 4) follows
48    // it: a 4 right after it says it is there, a 4 in its place with more data after says it was
49    // left out. A header that ends right after the field (as Word 97's) has it, a 4 included.
50    let bidi = match (at(18), at(20)) {
51        (b, Some(LSD_SIZE) | None) => b,
52        (Some(LSD_SIZE), Some(_)) => None,
53        (b, _) => b,
54    };
55    StandardFonts { ascii: at(12), east_asian: at(14), other: at(16), bidi }
56}

58/// Parse the `STSH` at `table[fc..fc+lcb]`.
59pub(crate) fn parse(table: &[u8], fc: u32, lcb: u32) -> Result<Vec<RawStyle>, DocbinError> {
60    let fc = fc as usize;
61    let lcb = lcb as usize;
62    let end = fc.checked_add(lcb).ok_or_else(|| DocbinError::Malformed("STSH overflow".into()))?;
63    let stsh = table.get(fc..end).ok_or_else(|| DocbinError::Malformed("STSH outside the Table stream".into()))?;
64    if stsh.len() < 4 {
65        return Ok(Vec::new());
66    }
67    // LPStshi: cbStshi u16 + Stshi; the style count and the size of each Stdf are what we need.
68    let cb_stshi = u16::from_le_bytes([stsh[0], stsh[1]]) as usize;
69    let stshi = stsh.get(2..2 + cb_stshi).unwrap_or(&[]);
70    if stshi.len() < 4 {
71        return Ok(Vec::new());
72    }
73    let cstd = u16::from_le_bytes([stshi[0], stshi[1]]) as usize;
74    let cb_std_base = u16::from_le_bytes([stshi[2], stshi[3]]) as usize;
75    // 10 (Word 97) or 18 (with StdfPost2000); anything else is corrupt but tolerated at 10.
76    let cb_std_base = if cb_std_base == 0x12 { 18 } else { 10 };
77    let cstd = cstd.min(MAX_STYLES);
78    let mut rest = stsh.get(2 + cb_stshi..).unwrap_or(&[]);
79    let mut out = Vec::with_capacity(cstd);
80    for _ in 0..cstd {
81        if rest.len() < 2 || out.len() >= MAX_STYLES {
82            break;
83        }
84        let cb_std = u16::from_le_bytes([rest[0], rest[1]]) as usize;
85        let (std, next) = rest.split_at(rest.len().min(2 + cb_std));
86        rest = next;
87        if cb_std == 0 {
88            out.push(RawStyle::default()); // empty style: keeps istd numbering aligned
89            continue;
90        }
91        let Some(stdf) = std.get(2..2 + cb_std_base.min(cb_std.saturating_sub(2))) else {
92            out.push(RawStyle::default());
93            continue;
94        };
95        // StdfBase ([MS-DOC] §2.9.260, 10 bytes): sti(12)+flags(4); stk(4)+istdBase(12);
96        // cupx(4)+istdNext(12); bchUpe(16) (= cbStd); grfstd(16). istdBase is a zero-based
97        // index into the stylesheet, 0x0FFF meaning "no parent".
98        let stk = (stdf.get(2).copied().unwrap_or(1) & 0x0F).max(1);
99        let w1 = stdf.get(2..4).map(|b| u16::from_le_bytes([b[0], b[1]])).unwrap_or(0xFFF1);
100        let istd_base = w1 >> 4;
101        let w2 = stdf.get(4..6).map(|b| u16::from_le_bytes([b[0], b[1]])).unwrap_or(0);
102        let cupx = (w2 & 0x0F) as usize;
103        let _istd_next = w2 >> 4; // next-paragraph style; unused for now
104        let mut at = 2 + cb_std_base;
105        // xstzName: Xst (cch u16 + UTF-16LE) + terminator u16.
106        let mut name = String::new();
107        if let Some(cch) = std.get(at..at + 2).map(|b| u16::from_le_bytes([b[0], b[1]])).map(|c| (c as usize).min(256)) {
108            if let Some(units) = std.get(at + 2..at + 2 + cch * 2) {
109                let chars: Vec<u16> = units.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).collect();
110                name = String::from_utf16_lossy(&chars);
111            }
112            at += 2 + cch * 2 + 2;
113        }
114        // grLPUpxSw: cupx entries, each cbUpx u16 + data (+ pad to even).
115        let mut upxs: Vec<Vec<u8>> = Vec::new();
116        for _ in 0..cupx.min(3) {
117            let Some(cb) = std.get(at..at + 2).map(|b| u16::from_le_bytes([b[0], b[1]])).map(|v| v as usize) else { break };
118            let data = std.get(at + 2..at + 2 + cb).unwrap_or(&[]).to_vec();
119            upxs.push(data);
120            at += 2 + cb + (cb & 1); // padded to even; cb in the header excludes the pad
121        }
122        // Paragraph styles: UPX0 is a PAPX (istd u16 + grpprl), UPX1 a raw grpprl of
123        // character sprms; character styles: UPX0 is the character grpprl.
124        let mut papx = Vec::new();
125        let mut chpx = Vec::new();
126        if stk == 1 {
127            if let Some(u) = upxs.first() {
128                papx = u.get(2..).unwrap_or(&[]).to_vec(); // skip the istd of GrpPrlAndIstd
129            }
130            if let Some(u) = upxs.get(1) {
131                chpx = u.clone();
132            }
133        } else if stk == 2
134            && let Some(u) = upxs.first()
135        {
136            chpx = u.clone();
137        }
138        out.push(RawStyle { stk, istd_base, name, papx, chpx });
139    }
140    Ok(out)
141}

143/// Font names from the `SttbfFfn` at `table[fc..fc+lcb]` ([MS-DOC] §2.9.286): an STTB
144/// (§2.2.4) whose entries are FFN records (§2.9.82). Each entry is a length (`cchData`: a
145/// one-byte byte count in the non-extended table the spec requires, a two-byte count of
146/// 16-bit units in an extended one) followed by the FFN, and that length alone decides where
147/// the next entry starts: an FFN ends with its primary name and an optional alternate name
148/// (`xszAlt`), so scanning for the primary name's terminator would land inside the latter.
149pub(crate) fn fonts(table: &[u8], fc: u32, lcb: u32) -> Vec<String> {
150    let end = match (fc as usize).checked_add(lcb as usize) {
151        Some(e) if e <= table.len() => e,
152        _ => return Vec::new(),
153    };
154    let Some(sttb) = table.get(fc as usize..end) else { return Vec::new() };
155    let u16at = |o: usize| -> Option<u16> { sttb.get(o..o.checked_add(2)?).map(|b| u16::from_le_bytes([b[0], b[1]])) };
156    let Some(first) = u16at(0) else { return Vec::new() };
157    // FFN: ffid(1) wWeight(2) chs(1) ixchSzAlt(1) panose(10) fs(24) = 39 bytes, then the
158    // primary name, NUL-terminated, as UTF-16LE in the Word 97 files we accept (8-bit names
159    // from older writers fall back to Windows-1252).
160    let name_of = |ffn: &[u8]| -> String {
161        let tail = ffn.get(FFN_HEADER.min(ffn.len())..).unwrap_or(&[]);
162        let units: Vec<u16> = tail.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).take_while(|&u| u != 0).take(64).collect();
163        let s = String::from_utf16_lossy(&units);
164        if s.chars().any(|c| c as u32 > 0x7F || c.is_ascii_graphic() || c == ' ') || s.is_empty() {
165            s
166        } else {
167            // An 8-bit name reads as garbage through UTF-16; fall back to Windows-1252.
168            tail.iter().take_while(|&&b| b != 0).take(64).map(|&b| crate::doc::encoding::cp1252(b)).collect()
169        }
170    };
171    // STTB header: [fExtend 0xFFFF] cData(2) cbExtra(2); each entry is followed by cbExtra
172    // bytes (0 for font tables, but skip them if present).
173    let (extended, cdata, cb_extra, mut at) =
174        if first == 0xFFFF { (true, u16at(2).unwrap_or(0), u16at(4).unwrap_or(0), 6usize) } else { (false, first, u16at(2).unwrap_or(0), 4usize) };
175    let mut out = Vec::new();
176    for _ in 0..cdata.min(1024) {
177        let (len_bytes, cb) = if extended {
178            let Some(cb) = u16at(at) else { break };
179            (2usize, usize::from(cb).saturating_mul(2))
180        } else {
181            let Some(&cb) = sttb.get(at) else { break };
182            (1usize, usize::from(cb))
183        };
184        let start = at.saturating_add(len_bytes);
185        let stop = start.saturating_add(cb);
186        // A record cut short by the end of the table still yields what it holds.
187        out.push(name_of(sttb.get(start..stop.min(sttb.len())).unwrap_or(&[])));
188        at = stop.saturating_add(usize::from(cb_extra));
189    }
190    out
191}

193/// Bytes of an FFN before its primary name ([MS-DOC] §2.9.82).
194const FFN_HEADER: usize = 39;
