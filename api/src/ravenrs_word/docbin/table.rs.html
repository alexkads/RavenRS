15// Paragraph sprms for table membership.
16const P_IN_TABLE: u16 = 0x2416;
17const P_TTP: u16 = 0x2417;
18/// `sprmPItap`: table depth (we only model depth 1; deeper rows fall back to paragraphs).
19const P_ITAP: u16 = 0x6649;
20
21// Table sprms ([MS-DOC] §2.6.3).
22const T_JC: u16 = 0x5400;
23const T_JC_90: u16 = 0x548A;
24const T_DXA_LEFT: u16 = 0x9601;
25const T_DXA_GAP_HALF: u16 = 0x9602;
26const T_TABLE_HEADER: u16 = 0x3404;
27const T_TABLE_BORDERS_80: u16 = 0xD605;
28const T_ROW_HEIGHT: u16 = 0x9407;
29const T_DEF_TABLE: u16 = 0xD608;
30const T_DEF_TABLE_SHD_80: u16 = 0xD609;
31/// `sprmTDefTableShd` / `Shd2nd` / `Shd3rd`: 10-byte `Shd`s for cells 0-21, 22-43, 44-62.
32const T_DEF_TABLE_SHD: u16 = 0xD612;
33const T_DEF_TABLE_SHD_2ND: u16 = 0xD616;
34const T_DEF_TABLE_SHD_3RD: u16 = 0xD60C;
35/// `sprmTDefTableShdRaw` / `Raw2nd` / `Raw3rd`: the same `Shd` arrays over the same cell
36/// ranges, which Word 2007+ writes next to the above.
37const T_DEF_TABLE_SHD_RAW: u16 = 0xD670;
38const T_DEF_TABLE_SHD_RAW_2ND: u16 = 0xD671;
39const T_DEF_TABLE_SHD_RAW_3RD: u16 = 0xD672;
40const T_CANT_SPLIT_90: u16 = 0x3403;
41const T_MERGE: u16 = 0x5624;
42const T_VERT_MERGE: u16 = 0xD62B;
43const T_VERT_ALIGN: u16 = 0xD62C;
44const T_SET_SHD: u16 = 0xD62D;
45const T_CANT_SPLIT: u16 = 0x3466;
46const T_CELL_WIDTH: u16 = 0xD635;

48/// Most grid columns we build from a `sprmTDefTable` (the model's own cap is 63; one more
49/// edge is fine but merged rows can sum spans slightly over, so clamp when assembling).
50const MAX_COLS: usize = 63;

52/// What a paragraph mark says about its table context, from its PAPX.
53#[derive(Default)]
54pub(crate) struct RowInfo {
55    pub(crate) in_table: bool,
56    pub(crate) ttp: bool,
57    /// The mark itself was 0x07 (a cell end), set by the walker.
58    pub(crate) cell_mark: bool,
59    /// Table depth from `sprmPItap` (0 or 1 supported; deeper tables degrade to paragraphs).
60    pub(crate) depth: u32,
61    /// Cell-edge positions (points) from `sprmTDefTable` — [MS-DOC] defines them per row,
62    /// so each row keeps its own; ascending in well-formed rows, the first edge usually 0
63    /// (the row origin). Rows without a `sprmTDefTable` keep this empty.
64    pub(crate) edges: Vec<f32>,
65    height: Option<f32>,
66    height_rule: HeightRule,
67    header: bool,
68    cant_split: bool,
69    align: Option<Align>,
70    indent: Option<f32>,
71    borders: Option<Borders>,
72    /// Horizontal merges: (itcFirst, itcLim) cell-index ranges.
73    merges: Vec<(u8, u8)>,
74    /// Per-cell overrides keyed by (pre-merge) cell index.
75    vmerge: Vec<(u8, VMerge)>,
76    /// Later entries win: the `Shd` arrays override the `Shd80` fallback before them, and an
77    /// automatic `Shd` (`None`) clears it.
78    shading: Vec<(u8, Option<Rgb>)>,
79    widths: Vec<(u8, f32)>,
80    width_pcts: Vec<(u8, f32)>,
81    valign: Vec<(u8, VAlign)>,
82}

119    }

122/// Set cell `c`'s override to `v`, replacing an earlier one: one slot per cell, however often
123/// a PAPX repeats the sprm, so a shared PAPX can't pile up copies on every paragraph mark.
124fn set<T>(slots: &mut Vec<(u8, T)>, c: u8, v: T) {
125    match slots.iter_mut().find(|(k, _)| *k == c) {
126        Some(slot) => slot.1 = v,
127        None => slots.push((c, v)),
128    }
129}

131pub(crate) fn decode(papx: &[u8]) -> RowInfo {
132    let mut r = RowInfo::default();
133    let mut gap_half: Option<i16> = None;
134    let mut dxa_left: Option<i16> = None;
135    for prl in sprm::iter(papx) {
136        // The membership sprms are paragraph sprms (sgc 1); the rest here are table sprms
137        // (sgc 5), so match opcodes directly.
138        match prl.op {
139            P_IN_TABLE => r.in_table = prl.operand.first().copied().unwrap_or(0) != 0,
140            P_TTP => r.ttp = prl.operand.first().copied().unwrap_or(0) != 0,
141            P_ITAP => r.depth = i32_of(&prl).unwrap_or(0).max(0) as u32,
142            T_DEF_TABLE => r.edges = def_table_edges(prl.operand),
143            T_TABLE_HEADER => r.header = prl.operand.first().copied().unwrap_or(0) != 0,
144            T_CANT_SPLIT | T_CANT_SPLIT_90 => r.cant_split = prl.operand.first().copied().unwrap_or(0) != 0,
145            T_ROW_HEIGHT => {
146                if let Some(v) = s16(&prl) {
147                    if v < 0 {
148                        r.height_rule = HeightRule::Exact;
149                        r.height = Some((-(v as f32) / 20.0).clamp(0.0, 1584.0));
150                    } else {
151                        r.height_rule = HeightRule::AtLeast;
152                        r.height = Some((v as f32 / 20.0).clamp(0.0, 1584.0));
153                    }
154                }
155            }
156            T_JC | T_JC_90 => {
157                r.align = match u16_of(&prl).unwrap_or(0) {
158                    1 => Some(Align::Center),
159                    2 => Some(Align::Right),
160                    _ => Some(Align::Left),
161                };
162            }
163            T_DXA_LEFT => dxa_left = s16(&prl),
164            T_DXA_GAP_HALF => gap_half = s16(&prl),
165            T_TABLE_BORDERS_80 => r.borders = table_borders(prl.operand),
166            T_MERGE => {
167                // Rows apply only the first MAX_COLS merges.
168                if let [a, b, ..] = prl.operand
169                    && b > a
170                    && r.merges.len() < MAX_COLS
171                {
172                    r.merges.push((*a, *b));
173                }
174            }
175            T_VERT_MERGE => {
176                // cb, itc, flag: 1 = continues the merge from above, 3 = restarts it.
177                if let [_, itc, flag, ..] = prl.operand {
178                    let m = match flag {
179                        3 => VMerge::Restart,
180                        1 => VMerge::Continue,
181                        _ => VMerge::None,
182                    };
183                    // The first override for a cell wins.
184                    if m != VMerge::None && !r.vmerge.iter().any(|(k, _)| k == itc) {
185                        r.vmerge.push((*itc, m));
186                    }
187                }
188            }
189            T_SET_SHD => {
190                if let Some(rgb) = prl.operand.get(3..).and_then(shd_rgb)
191                    && let [a, b, ..] = prl.operand.get(1..3).unwrap_or(&[])
192                {
193                    for c in *a..(*b).min(MAX_COLS as u8) {
194                        set(&mut r.shading, c, Some(rgb));
195                    }
196                }
197            }
198            T_DEF_TABLE_SHD_80 => {
199                let cells = prl.operand.get(1..).unwrap_or(&[]);
200                for (i, b) in cells.as_chunks::<2>().0.iter().take(MAX_COLS).enumerate() {
201                    if let Some(rgb) = shd80_rgb(b) {
202                        set(&mut r.shading, i as u8, Some(rgb));
203                    }
204                }
205            }
206            T_DEF_TABLE_SHD | T_DEF_TABLE_SHD_RAW | T_DEF_TABLE_SHD_2ND | T_DEF_TABLE_SHD_RAW_2ND | T_DEF_TABLE_SHD_3RD | T_DEF_TABLE_SHD_RAW_3RD => {
207                // Each array covers its own cell range (0-21, 22-43, 44-62); extra entries
208                // are ignored so an over-long array can't override the next range.
209                let (first, lim) = match prl.op {
210                    T_DEF_TABLE_SHD_2ND | T_DEF_TABLE_SHD_RAW_2ND => (22, 44),
211                    T_DEF_TABLE_SHD_3RD | T_DEF_TABLE_SHD_RAW_3RD => (44, MAX_COLS),
212                    _ => (0, 22),
213                };
214                let cells = prl.operand.get(1..).unwrap_or(&[]);
215                let raw = matches!(prl.op, T_DEF_TABLE_SHD_RAW | T_DEF_TABLE_SHD_RAW_2ND | T_DEF_TABLE_SHD_RAW_3RD);
216                for (i, b) in cells.as_chunks::<10>().0.iter().take(lim - first).enumerate() {
217                    // In the Raw arrays a nil Shd (ipat 0xFFFF) means "as the table style
218                    // shades it", so it leaves any earlier fill alone.
219                    if raw && b.get(8..10) == Some(&[0xFF, 0xFF]) {
220                        continue;
221                    }
222                    set(&mut r.shading, (first + i) as u8, shd_rgb(b));
223                }
224            }
225            T_CELL_WIDTH => {
226                // Operand: cb (1 byte), itcFirst (1 byte), itcLim (1 byte), ftsWidth (1 byte), wWidth (2 bytes).
227                // [MS-DOC] §2.9.290 sprmTCellWidth (0xD635).
228                // ftsWidth: 3 = twips (dxa), 2 = percentage (50ths of a percent).
229                if let [_, a, b, fts, w0, w1, ..] = prl.operand {
230                    if *fts == 3 {
231                        let w = (i16::from_le_bytes([*w0, *w1]) as f32 / 20.0).clamp(0.0, 1584.0);
232                        for c in *a..(*b).min(MAX_COLS as u8) {
233                            set(&mut r.widths, c, w);
234                        }
235                    } else if *fts == 2 {
236                        let pct = (u16::from_le_bytes([*w0, *w1]) as f32 / 50.0).clamp(0.0, 100.0);
237                        for c in *a..(*b).min(MAX_COLS as u8) {
238                            set(&mut r.width_pcts, c, pct);
239                        }
240                    }
241                }
242            }
243            T_VERT_ALIGN => {
244                if let [_, a, b, va, ..] = prl.operand {
245                    let v = match va {
246                        1 => VAlign::Center,
247                        2 => VAlign::Bottom,
248                        _ => VAlign::Top,
249                    };
250                    for c in *a..(*b).min(MAX_COLS as u8) {
251                        set(&mut r.valign, c, v);
252                    }
253                }
254            }
255            _ => {}
256        }
257    }
258    // The table indent is the row origin minus half the cell gap.
259    if let Some(l) = dxa_left {
260        let gap = gap_half.unwrap_or(0) as f32 / 20.0;
261        r.indent = Some(((l as f32 / 20.0) - gap).clamp(-1584.0, 1584.0));
262    }
263    r
264}

275/// Group consecutive in-table paragraphs into [`Table`] blocks; the rest pass through.
276/// Rows that never close (or tables nested deeper than we model) degrade to paragraphs so
277/// no text is lost.
278pub(crate) fn assemble(paras: Vec<ParaOut>) -> Blocks {
279    let mut out: Blocks = Vec::new();
280    let mut rows: Vec<Row> = Vec::new();
281    let mut cells: Vec<Cell> = Vec::new();
282    let mut cell: Blocks = Vec::new();
283    let mut grid: Vec<f32> = Vec::new();
284    let mut props = TableProps::default();
285    let mut open = false;
286    let mut depth_seen = 0u32;
287
288    fn flush_cell(cells: &mut Vec<Cell>, cell: &mut Blocks) {
289        let mut blocks = std::mem::take(cell);
290        if blocks.is_empty() {
291            blocks.push(para_block(Paragraph::new()));
292        }
293        cells.push(Cell { props: CellProps { span: 1, ..Default::default() }, blocks, controls: Default::default() });
294    }
295    fn build_row(mut cells: Vec<Cell>, info: &RowInfo) -> Row {
296        // Per-cell overrides first (sprm cell indices count pre-merge cells), then merges.
297        for (i, c) in cells.iter_mut().enumerate() {
298            let i = i as u8;
299            if let Some((_, m)) = info.vmerge.iter().find(|(k, _)| *k == i) {
300                c.props.vmerge = *m;
301            }
302            if let Some((_, w)) = info.widths.iter().rfind(|(k, _)| *k == i) {
303                c.props.width = Some(*w);
304            }
305            if let Some((_, pct)) = info.width_pcts.iter().rfind(|(k, _)| *k == i) {
306                c.props.width_pct = Some(*pct);
307            }
308            if let Some((_, s)) = info.shading.iter().rfind(|(k, _)| *k == i) {
309                c.props.shading = *s;
310            }
311            if let Some((_, v)) = info.valign.iter().rfind(|(k, _)| *k == i) {
312                c.props.valign = Some(*v);
313            }
314        }
315        for &(a, b) in info.merges.iter().take(MAX_COLS) {
316            let a = a as usize;
317            let b = (b as usize).min(cells.len());
318            if a < b && b <= cells.len() && b - a > 1 {
319                let removed: Vec<Cell> = cells.drain(a + 1..b).collect();
320                if let Some(first) = cells.get_mut(a) {
321                    first.props.span = ((b - a) as u32).min(63);
322                    for c in removed {
323                        first.blocks.extend(c.blocks.into_iter().filter(|x| x.as_para().is_none_or(|p| !p.is_empty())));
324                    }
325                    if first.blocks.len() > 1 && first.blocks.first().and_then(|x| x.as_para()).is_some_and(Paragraph::is_empty) {
326                        first.blocks.remove(0);
327                    }
328                }
329            }
330        }
331        Row {
332            props: RowProps {
333                height: info.height,
334                height_rule: info.height_rule,
335                header: info.header,
336                cant_split: info.cant_split,
337                ..Default::default()
338            },
339            cells,
340            controls: Default::default(),
341        }
342    }
343    fn flush_table(out: &mut Blocks, rows: &mut Vec<Row>, cells: &mut Vec<Cell>, cell: &mut Blocks, grid: &[f32], props: &TableProps, nested: bool) {
344        if !cells.is_empty() || !cell.is_empty() {
345            flush_cell(cells, cell);
346            // A stray trailing row (no TTP of its own): size its cells from the grid.
347            let mut stray = build_row(std::mem::take(cells), &RowInfo::default());
348            for (i, c) in stray.cells.iter_mut().enumerate() {
349                if c.props.width.is_none() && c.props.span == 1 {
350                    c.props.width = grid.get(i).copied();
351                }
352            }
353            rows.push(stray);
354        }
355        if rows.is_empty() {
356            return;
357        }
358        if nested {
359            // Unclosed or nested deeper than we model: keep the text, drop the table shape.
360            for r in rows.iter() {
361                for c in &r.cells {
362                    out.extend(c.blocks.iter().cloned());
363                }
364            }
365        } else {
366            out.push(Arc::new(Block::Table(Table {
367                props: props.clone(),
368                grid: grid.to_vec(),
369                rows: std::mem::take(rows),
370                controls: Default::default(),
371            })));
372        }
373        rows.clear();
374    }
375
376    for p in paras {
377        let Some(info) = p.row else {
378            if open {
379                flush_table(&mut out, &mut rows, &mut cells, &mut cell, &grid, &props, depth_seen > 1);
380                grid.clear();
381                // Table props are per-table (first row carrying the sprm wins within the
382                // table): reset the accumulator so one table's borders, align, indent and
383                // cell margins never leak into the next (the repro's first table wrote
384                // Word's default 108-twip gap, and every later table — including the
385                // change history with its own 40-twip gap — inherited 5.4 pt margins).
386                props = TableProps::default();
387                open = false;
388                depth_seen = 0;
389            }
390            out.push(para_block(p.para));
391            continue;
392        };
393        if !open {
394            open = true;
395        }
396        depth_seen = depth_seen.max(info.depth);
397        if grid.is_empty() {
398            // Column widths from the table's first `sprmTDefTable`.
399            grid = info.edges.windows(2).take(MAX_COLS).map(|w| (w[1] - w[0]).max(0.0)).collect();
400        }
401        if info.borders.is_some() && props.borders.is_none() {
402            props.borders = info.borders;
403        }
404        if props.align.is_none() {
405            props.align = info.align;
406        }
407        if props.indent.is_none() {
408            props.indent = info.indent;
409        }
410        if info.cell_mark {
411            // A row-end (TTP) mark terminates the row. Word writes rows as N content
412            // cells each ended by their own cell mark, followed by an empty row-end mark
413            // that carries the row's TDefTable (the 3GPP change history: 8 content marks
414            // + an empty TTP whose TDefTable defines itcMac = 8). When that mark ends no
415            // content of its own and the TDefTable already has a column per flushed cell,
416            // it is a pure row terminator, not the end of a phantom (N+1)th cell — such a
417            // cell has no column (width None) and shifts every real cell's span one grid
418            // column over. A TTP mark with content, or one short of the TDefTable's
419            // columns, still ends a real cell (rows without a TDefTable keep today's
420            // shape: the count is unknowable, so the mark always flushes).
421            let row_end_only = info.ttp && p.para.text.is_empty() && cell.is_empty() && !info.edges.is_empty() && cells.len() + 1 >= info.edges.len();
422            // ...but a row-end mark with no cells before it still makes a (one-cell) row.
423            let row_end_only = row_end_only && !cells.is_empty();
424            if !p.para.text.is_empty() {
425                cell.push(para_block(p.para));
426            }
427            if !row_end_only {
428                flush_cell(&mut cells, &mut cell);
429            }
430            if info.ttp {
431                let row = build_row(std::mem::take(&mut cells), &info);
432                rows.push(row);
433            }
434        } else {
435            cell.push(para_block(p.para));
436        }
437    }
438    if open {
439        flush_table(&mut out, &mut rows, &mut cells, &mut cell, &grid, &props, depth_seen > 1);
440    }
441    out
442}

444/// Cell-edge positions (points) from a `sprmTDefTable` operand: cb u16, itcMac u8, then
445/// itcMac+1 twip boundaries followed by the TC80s (skipped). Edges are returned as stored:
446/// per-row, in operand order; sorting and capping happen when the table grid is assembled.
447fn def_table_edges(operand: &[u8]) -> Vec<f32> {
448    let itc = operand.get(2).copied().unwrap_or(0) as usize;
449    if itc == 0 {
450        return Vec::new();
451    }
452    // itcMac is a u8, so this allocation stays tiny; the loop stops at the operand's end.
453    let mut edges = Vec::with_capacity((itc + 1).min(256));
454    for i in 0..=itc {
455        match operand.get(3 + i * 2..3 + i * 2 + 2) {
456            Some(b) => edges.push(i16::from_le_bytes([b[0], b[1]]) as f32 / 20.0),
457            None => break,
458        }
459    }
460    edges
461}

463/// Table borders from `sprmTTableBorders80`: cb (0x18) then six Brc80s in the order top,
464/// left, bottom, right, inside-horizontal, inside-vertical, read as paragraph borders are.
465fn table_borders(operand: &[u8]) -> Option<Borders> {
466    let b = |i: usize| operand.get(1 + i * 4..1 + i * 4 + 4).and_then(fmt::brc80);
467    Some(Borders { top: b(0), left: b(1), bottom: b(2), right: b(3), between: b(4), inside_v: b(5) })
468}

470/// Background colour from a 10-byte `Shd`: cvFore, cvBack (COLORREFs), ipat. Solid patterns
471/// show the foreground; clear (0) and the other patterns show the background (Word writes a
472/// plain cell fill as clear with cvBack); nil (0xFFFF) and automatic colours are no fill.
473fn shd_rgb(b: &[u8]) -> Option<Rgb> {
474    let cv = |b: &[u8]| -> Option<Rgb> {
475        match b {
476            [_, _, _, 0xFF] => None, // fAuto
477            [r, g, bl, _] => Some(Rgb(*r, *g, *bl)),
478            _ => None,
479        }
480    };
481    let ipat = b.get(8..10).map(|x| u16::from_le_bytes([x[0], x[1]]))?;
482    match ipat {
483        0xFFFF => None,
484        1 => cv(b.get(0..4)?),
485        _ => cv(b.get(4..8)?),
486    }
487}

489/// Background colour from a 2-byte `Shd80` (icoFore, icoBack, ipat bit-fields). As in
490/// [`shd_rgb`], solid shows the foreground and clear (0) the background, which is how Word
491/// 97 writes a plain fill; ico 0 is automatic (no fill).
492fn shd80_rgb(b: &[u8]) -> Option<Rgb> {
493    let v = u16::from_le_bytes([*b.first()?, *b.get(1)?]);
494    let fore = v & 0x1F;
495    let back = (v >> 5) & 0x1F;
496    let ipat = v >> 10;
497    if fore == 0x1F && back == 0x1F {
498        return None;
499    }
500    fmt::ico(if ipat == 1 { fore as u8 } else { back as u8 })
501}

503fn u16_of(p: &Prl) -> Option<u16> {
504    match p.operand {
505        [b0, b1, ..] => Some(u16::from_le_bytes([*b0, *b1])),
506        _ => None,
507    }
508}
509
510fn s16(p: &Prl) -> Option<i16> {
511    u16_of(p).map(|v| v as i16)
512}

516        [b0, b1, b2, b3, ..] => Some(i32::from_le_bytes([*b0, *b1, *b2, *b3])),
517        _ => None,
518    }
