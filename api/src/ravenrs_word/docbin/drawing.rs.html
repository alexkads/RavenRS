14/// Most anchors (and text box stories) read per PLC.
15const MAX_ANCHORS: usize = 10_000;
16/// `Spa` and `FTXBXS` record sizes.
17const CB_SPA: usize = 26;
18const CB_FTXBXS: usize = 22;
19/// Largest frame or offset kept (points; 22 inches, Word's own limit for sizes).
20const MAX_PT: f32 = 1584.0;
21/// Deepest group nesting flattened.
22const MAX_GROUP_DEPTH: usize = 8;
23
24/// The (CP, data) entries of a PLC with `cb`-byte data elements, the trailing CP dropped.
25fn plc(table: &[u8], (fc, lcb): (u32, u32), cb: usize) -> Vec<(u32, &[u8])> {
26    let Some(blob) = table.get(fc as usize..(fc as usize).saturating_add(lcb as usize)) else { return Vec::new() };
27    let n = blob.len().saturating_sub(4) / (4 + cb);
28    let n = n.min(MAX_ANCHORS);
29    let data_at = (n + 1) * 4;
30    (0..n)
31        .filter_map(|i| {
32            let cp = u32::from_le_bytes(blob.get(i * 4..i * 4 + 4)?.try_into().ok()?);
33            let at = data_at + i * cb;
34            Some((cp, blob.get(at..at + cb)?))
35        })
36        .collect()
37}

39/// The text box stories of a `PlcftxbxTxt` (or `PlcfHdrtxbxTxt`): (shape spid, story CP range
40/// relative to the text box subdocument).
41///
42/// The CPs of a well-formed PLC ascend, so the stories are disjoint. A story starting before
43/// the end of the last one kept (a hostile table pointing many text boxes at the same text,
44/// each read into its own part) is dropped: every subdocument CP becomes at most one story.
45pub(crate) fn textbox_stories(table: &[u8], pair: Option<(u32, u32)>) -> Vec<(u32, (u32, u32))> {
46    let Some(pair) = pair else { return Vec::new() };
47    let Some(blob) = table.get(pair.0 as usize..(pair.0 as usize).saturating_add(pair.1 as usize)) else { return Vec::new() };
48    let mut out = Vec::new();
49    let mut kept_to = 0u32;
50    for (i, (cp, ftxbxs)) in plc(table, pair, CB_FTXBXS).iter().enumerate() {
51        // The story runs to the next entry's CP (the PLC's last CP for the last entry).
52        let Some(end) = blob.get((i + 1) * 4..(i + 1) * 4 + 4).and_then(|b| b.try_into().ok()).map(u32::from_le_bytes) else { continue };
53        let Some(lid) = ftxbxs.get(14..18).and_then(|b| b.try_into().ok()).map(u32::from_le_bytes) else { continue };
54        if end > *cp && *cp >= kept_to && lid != 0 {
55            out.push((lid, (*cp, end)));
56            kept_to = end;
57        }
58    }
59    out
60}

62/// What building the floating objects needs.
63pub(crate) struct Build<'a> {

67    /// Text box part ids by shape spid.
68    pub(crate) stories: &'a HashMap<u32, u32>,
69}

71/// The floating objects anchored by `PlcfSpa` (at `pair`), as (absolute anchor CP, object),
72/// sorted by CP. `cp_base` turns the PLC's CPs into absolute ones (the header subdocument's
73/// start for `PlcSpaHdr`).
74pub(crate) fn anchors(
75    table: &[u8],
76    pair: Option<(u32, u32)>,
77    cp_base: u32,
78    shapes: &HashMap<u32, Node>,
79    b: &Build,
80    pics: &mut Pictures,
81) -> Vec<(u32, InlineObject)> {
82    let Some(pair) = pair else { return Vec::new() };
83    let mut out = Vec::new();
84    let mut seen = std::collections::HashSet::new();
85    for (cp, spa) in plc(table, pair, CB_SPA) {
86        let i32_at = |o: usize| spa.get(o..o + 4).and_then(|b| b.try_into().ok()).map(i32::from_le_bytes).unwrap_or(0);
87        let spid = i32_at(0) as u32;
88        let [xl, yt, xr, yb] = [4, 8, 12, 16].map(i32_at);
89        let flags = spa.get(20..22).map(|b| u16::from_le_bytes([b[0], b[1]])).unwrap_or(0);
90        let Some(node) = shapes.get(&spid) else { continue };
91        // Word anchors each shape once. A repeated spid would build another copy of the shape,
92        // and of every member of a group, per anchor (#3710).
93        if !seen.insert(spid) {
94            continue;
95        }
96        let tw = |v: i64| (v as f32 / 20.0).clamp(-2.0 * MAX_PT, 2.0 * MAX_PT);
97        let (mut x, mut y) = (tw(xl as i64), tw(yt as i64));
98        let (mut w, mut h) = (tw(xr as i64 - xl as i64).clamp(0.0, MAX_PT), tw(yb as i64 - yt as i64).clamp(0.0, MAX_PT));
99        let sp = node.sp();
100        let rot = crate::geom::normalize_degrees(if sp.props.rot.is_finite() { sp.props.rot } else { 0.0 });
101        // A shape turned by 45°–135° (or 225°–315°) is anchored by its turned bounds: the
102        // unrotated frame has the other aspect, about the same centre.
103        if (45.0..135.0).contains(&rot) || (225.0..315.0).contains(&rot) {
104            x += (w - h) / 2.0;
105            y += (h - w) / 2.0;
106            std::mem::swap(&mut w, &mut h);
107        }
108        let mut float = spa_float(flags, &sp.props);
109        float.x = x;
110        float.y = y;
111        float.rot = rot;
112        float.flip_h = sp.flip_h();
113        float.flip_v = sp.flip_v();
114        let obj = match node {
115            Node::Shape(sp) => leaf(sp, w, h, float, b, pics),
116            Node::Group(sp, members) => group(sp, members, w, h, float, b, pics),
117        };
118        match obj {
119            Some(obj) => out.push((cp_base.saturating_add(cp), obj)),
120            None => pics.dropped(cp_base.saturating_add(cp)),
121        }
122    }
123    out.sort_by_key(|(cp, _)| *cp);
124    out
125}

127/// Placement from a `Spa`'s flags (fHdr, bx, by, wr, wrk, fRcaSimple, fBelowText, fAnchorLock)
128/// and the shape's position properties, which take precedence when present.
129fn spa_float(flags: u16, p: &crate::docbin::officeart::Props) -> Float {
130    let bx = (flags >> 1) & 3;
131    let by = (flags >> 3) & 3;
132    let wr = (flags >> 5) & 0xF;
133    let below = flags & (1 << 14) != 0;
134    let wrap = match wr {
135        1 => Wrap::TopAndBottom,
136        3 => {
137            if below {
138                Wrap::BehindText
139            } else {
140                Wrap::InFrontOfText
141            }
142        }
143        4 => Wrap::Tight,
144        5 => Wrap::Through,
145        _ => Wrap::Square,
146    };
147    let h_rel = match p.posrelh.unwrap_or(bx as u32) {
148        0 => Anchor::Margin,
149        1 => Anchor::Page,
150        3 => Anchor::Character,
151        _ => Anchor::Column,
152    };
153    let v_rel = match p.posrelv.unwrap_or(by as u32) {
154        0 => Anchor::Margin,
155        1 => Anchor::Page,
156        3 => Anchor::Line,
157        _ => Anchor::Paragraph,
158    };
159    let align = |v: Option<u32>| match v {
160        Some(1) => Some(FloatAlign::Start),
161        Some(2) => Some(FloatAlign::Center),
162        Some(3) => Some(FloatAlign::End),
163        Some(4) => Some(FloatAlign::Inside),
164        Some(5) => Some(FloatAlign::Outside),
165        _ => None,
166    };
167    // Distances from text, EMUs; Word's defaults are 0.125" at the sides, none above and below.
168    let emu = |v: Option<u32>, def: f32| v.map_or(def, |e| (e as f32 / 12_700.0).clamp(0.0, MAX_PT));
169    let [dl, dt, dr, db] = p.wrap_dist;
170    Float {
171        wrap,
172        h_rel,
173        v_rel,
174        h_align: align(p.posh),
175        v_align: align(p.posv),
176        dist: emu(dl, 9.0).max(emu(dr, 9.0)),
177        dist_top: emu(dt, 0.0),
178        dist_bottom: emu(db, 0.0),
179        ..Float::default()
180    }
181}

183/// One shape as a picture or a drawn shape, `w` × `h` points.
184fn leaf(sp: &Sp, w: f32, h: f32, float: Float, b: &Build, pics: &mut Pictures) -> Option<InlineObject> {
185    let p = &sp.props;
186    if let Some(pib) = p.pib {
187        let media = pics.bse(pib, b.bstore, b.delay)?;
188        return Some(InlineObject::Image { media, w, h, alt: p.alt.clone(), float, crop: p.crop, fx: Default::default(), ole: None });
189    }
190    let story = b.stories.get(&sp.spid).copied().filter(|_| sp.textbox || sp.spt == 202);
191    let mut kind = match sp.spt {
192        1 => ShapeKind::Rectangle,
193        2 => ShapeKind::RoundedRectangle,
194        3 => ShapeKind::Ellipse,
195        4 => ShapeKind::Diamond,
196        5 => ShapeKind::Triangle,
197        6 => ShapeKind::RightTriangle,
198        12 | 187 => ShapeKind::Star,
199        13 | 69 => ShapeKind::Arrow,
200        66 => ShapeKind::ArrowLeft,
201        67 => ShapeKind::ArrowDown,
202        68 => ShapeKind::ArrowUp,
203        20 | 32 => ShapeKind::Line,
204        74 => ShapeKind::Heart,
205        202 => ShapeKind::TextBox,
206        // A freeform's path is not read: better left out than drawn as a box.
207        0 => return None,
208        _ => ShapeKind::Rectangle,
209    };
210    if story.is_some() && kind == ShapeKind::Rectangle {
211        kind = ShapeKind::TextBox;
212    }
213    let line = kind == ShapeKind::Line;
214    // Brackets and braces (85 to 88, bracket and brace pairs 185, 186) are unfilled unless
215    // `fFilled` says otherwise, as Word's VML shapetypes for them say (`filled="f"`). They are
216    // drawn as their box here, which a default fill would make an opaque block over the text.
217    let unfilled = p.filled == Some(false) || (matches!(sp.spt, 85..=88 | 185 | 186) && p.filled != Some(true));
218    let fill = if line || unfilled { None } else { Some(Fill::solid(p.fill_color.unwrap_or(Rgb::WHITE))) };
219    let stroke = if p.line == Some(false) { None } else { Some(p.line_color.unwrap_or(Rgb::BLACK)) };
220    // Line widths are EMUs; Word's default is 0.75 pt.
221    let stroke_width = p.line_width.map_or(0.75, |e| (e as f32 / 12_700.0).clamp(0.0, 100.0));
222    Some(InlineObject::Shape {
223        kind,
224        w,
225        h,
226        fill,
227        stroke,
228        stroke_width,
229        float,
230        story,
231        freeform: None,
232        effects: Default::default(),
233        extra: Default::default(),
234    })
235}

237/// A group: its members flattened into the group's own coordinate space (nested groups mapped
238/// through their child anchors).
239fn group(sp: &Sp, members: &[Node], w: f32, h: f32, float: Float, b: &Build, pics: &mut Pictures) -> Option<InlineObject> {
240    let [l, t, r, btm] = sp.group_space.unwrap_or([0, 0, 0, 0]);
241    let (cw, ch) = ((r as f64 - l as f64) as f32, (btm as f64 - t as f64) as f32);
242    let (ch_w, ch_h) = if cw > 0.0 && ch > 0.0 { (cw, ch) } else { (w.max(1.0), h.max(1.0)) };
243    let mut children = Vec::new();
244    // Map: outer = x0 + inner × s; the top group's space starts at its FSPGR origin.
245    members_into(members, (-(l as f32), -(t as f32), 1.0, 1.0), 0, b, pics, &mut children);
246    if children.is_empty() {
247        return None;
248    }
249    Some(InlineObject::Group { w, h, float, ch_w, ch_h, children, canvas: None })
250}

252fn members_into(nodes: &[Node], t: (f32, f32, f32, f32), depth: usize, b: &Build, pics: &mut Pictures, out: &mut Vec<GroupChild>) {
253    let (x0, y0, sx, sy) = t;
254    let fit = |v: f32| if v.is_finite() { v.clamp(-1e7, 1e7) } else { 0.0 };
255    for n in nodes {
256        if out.len() >= crate::doc::para::MAX_GROUP_CHILDREN {
257            return;
258        }
259        let sp = n.sp();
260        let Some([l, t, r, btm]) = sp.child else { continue };
261        let (ox, oy) = (l as f32, t as f32);
262        let (ew, eh) = ((r as f64 - l as f64) as f32, (btm as f64 - t as f64) as f32);
263        match n {
264            Node::Group(_, members) => {
265                if depth >= MAX_GROUP_DEPTH {
266                    continue;
267                }
268                let [gl, gt, gr, gb] = sp.group_space.unwrap_or([l, t, r, btm]);
269                let (gw, gh) = ((gr as f64 - gl as f64) as f32, (gb as f64 - gt as f64) as f32);
270                let (kx, ky) = if gw > 0.0 && gh > 0.0 { (ew / gw, eh / gh) } else { (1.0, 1.0) };
271                let nt = (fit(x0 + (ox - gl as f32 * kx) * sx), fit(y0 + (oy - gt as f32 * ky) * sy), fit(sx * kx), fit(sy * ky));
272                members_into(members, nt, depth + 1, b, pics, out);
273            }
274            Node::Shape(sp) => {
275                let float = Float { rot: sp.props.rot, flip_h: sp.flip_h(), flip_v: sp.flip_v(), ..Float::default() };
276                let (w, h) = (fit(ew * sx).max(0.0), fit(eh * sy).max(0.0));
277                if let Some(obj) = leaf(sp, w, h, float, b, pics) {
278                    out.push(GroupChild { x: fit(x0 + ox * sx), y: fit(y0 + oy * sy), obj });
279                }
280            }
281        }
282    }
