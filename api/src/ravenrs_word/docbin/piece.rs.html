1//! The piece table ([MS-DOC] `Clx`/`Pcdt`/`PlcPcd`): where every character position lives in
2//! the WordDocument stream, and how it is encoded.
3
4use crate::doc::encoding::cp1252;
5
6use crate::docbin::DocbinError;
7
8/// Most pieces any real file has; more means a corrupt `PlcPcd`.
9const MAX_PIECES: usize = 1_000_000;
10
11/// One piece: text at FcCompressed `fc` (30-bit offset + compression bit).
12#[derive(Clone, Copy, Debug)]
13struct Pcd {
14    fc: u32,
15    compressed: bool,
16    /// `Prm`: either a single sprm (Prm0) or an index into the Clx's Prc grpprls (Prm1).
17    prm: u16,
18}
19
20/// The parsed piece table plus the grpprls of the Prcs that Prm1 references.
21pub(crate) struct PieceTable {
22    /// n+1 character-position starts.
23    cps: Vec<u32>,
24    pieces: Vec<Pcd>,
25    prc_grpprls: Vec<Vec<u8>>,
26}
27
28impl PieceTable {
29    /// Parse the `Clx` at `table[fc .. fc+lcb]`.
30    pub(crate) fn parse(table: &[u8], fc: u32, lcb: u32) -> Result<PieceTable, DocbinError> {
31        let fc = fc as usize;
32        let lcb = lcb as usize;
33        if lcb == 0 {
34            return Err(DocbinError::Malformed("empty Clx".into()));
35        }
36        let end = fc.checked_add(lcb).ok_or_else(|| DocbinError::Malformed("Clx offset overflow".into()))?;
37        let clx = table.get(fc..end).ok_or_else(|| DocbinError::Malformed(format!("Clx at {fc}+{lcb} is outside the Table stream")))?;
38        // A Clx is an array of Prc (clxt 0x01, each holding a grpprl that Prm1 values can
39        // reference) followed by the Pcdt (clxt 0x02) that holds the piece table.
40        let mut prc_grpprls = Vec::new();
41        let mut i = 0usize;
42        let plc = loop {
43            let Some(&clxt) = clx.get(i) else {
44                return Err(DocbinError::Malformed("Clx ends before the Pcdt".into()));
45            };
46            match clxt {
47                0x01 => {
48                    let cb = clx
49                        .get(i + 1..i + 3)
50                        .map(|b| u16::from_le_bytes([b[0], b[1]]) as usize)
51                        .ok_or_else(|| DocbinError::Malformed("truncated Prc".into()))?;
52                    let stop = i.checked_add(3).and_then(|v| v.checked_add(cb)).ok_or_else(|| DocbinError::Malformed("Prc size overflow".into()))?;
53                    if let Some(g) = clx.get(i + 3..stop)
54                        && prc_grpprls.len() < MAX_PIECES
55                    {
56                        prc_grpprls.push(g.to_vec());
57                    }
58                    i = stop;
59                }
60                0x02 => {
61                    let lcb_plc = clx
62                        .get(i + 1..i + 5)
63                        .map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]]) as usize)
64                        .ok_or_else(|| DocbinError::Malformed("truncated Pcdt".into()))?;
65                    let start = i.checked_add(5).ok_or_else(|| DocbinError::Malformed("Pcdt offset overflow".into()))?;
66                    let stop = start.checked_add(lcb_plc).ok_or_else(|| DocbinError::Malformed("Pcdt size overflow".into()))?;
67                    break clx.get(start..stop).ok_or_else(|| DocbinError::Malformed("PlcPcd is outside the Clx".into()))?;
68                }
69                other => return Err(DocbinError::Malformed(format!("unexpected Clx item 0x{other:02X}"))),
70            }
71        };
72        // PlcPcd: aCp[n+1] u32s followed by aPcd[n] Pcds (8 bytes each).
73        if plc.len() < 4 || (plc.len() - 4) % 12 != 0 {
74            return Err(DocbinError::Malformed(format!("PlcPcd is {} bytes", plc.len())));
75        }
76        let n = (plc.len() - 4) / 12;
77        if n > MAX_PIECES {
78            return Err(DocbinError::Limit(format!("piece table has {n} pieces")));
79        }
80        let mut cps = Vec::with_capacity(n + 1);
81        for j in 0..=n {
82            let at = j * 4;
83            let Some(b) = plc.get(at..at + 4) else {
84                return Err(DocbinError::Malformed("truncated aCp".into()));
85            };
86            // Keep the CPs non-decreasing so binary search over them is sound: an inverted
87            // piece in a corrupt file becomes an empty one.
88            let c = u32::from_le_bytes([b[0], b[1], b[2], b[3]]);
89            cps.push(c.max(cps.last().copied().unwrap_or(0)));
90        }
91        let mut pieces = Vec::with_capacity(n);
92        for j in 0..n {
93            let at = (n + 1) * 4 + j * 8 + 2;
94            let Some(fc_raw) = plc.get(at..at + 4).map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]])) else {
95                return Err(DocbinError::Malformed("truncated aPcd".into()));
96            };
97            let prm = plc.get(at + 4..at + 6).map(|b| u16::from_le_bytes([b[0], b[1]])).unwrap_or(0);
98            pieces.push(Pcd { fc: fc_raw & 0x3FFF_FFFF, compressed: fc_raw & 0x4000_0000 != 0, prm });
99        }
100        Ok(PieceTable { cps, pieces, prc_grpprls })
101    }
102
103    /// The Prls this piece's `Prm` adds. Bit 0 (`fComplex`) discriminates: a Prm0
104    /// ([MS-DOC] §2.9.215) holds a 7-bit `isprm` in bits 1-7 and its one-byte operand in
105    /// bits 8-15; a Prm1 (§2.9.216) holds a zero-based index into the Clx's Prcs in bits
106    /// 1-15.
107    pub(crate) fn piece_prm(&self, cp: u32) -> PrmRef<'_> {
108        let Some(p) = self.piece_of(cp) else { return PrmRef::None };
109        if p.prm & 1 == 0 {
110            let isprm = ((p.prm >> 1) & 0x7F) as u8;
111            if isprm == 0 {
112                return PrmRef::None;
113            }
114            PrmRef::Sprm0 { isprm, val: (p.prm >> 8) as u8 }
115        } else {
116            PrmRef::Grpprl(self.prc_grpprls.get((p.prm >> 1) as usize).map(Vec::as_slice))
117        }
118    }
119
120    fn piece_of(&self, cp: u32) -> Option<&Pcd> {
121        let i = self.cps.partition_point(|&c| c <= cp).checked_sub(1)?;
122        if i >= self.pieces.len() {
123            return None;
124        }
125        self.pieces.get(i)
126    }
127
128    /// The FC of a character position in the space the FKP bin tables use: true byte
129    /// offsets into the WordDocument stream. A compressed piece stores its base `fc`
130    /// doubled (`byte = fc/2`), so the doubling unwinds here; an uncompressed piece's `fc`
131    /// already is its byte offset and each character covers two bytes.
132    pub(crate) fn fc_of_cp(&self, cp: u32) -> Option<u32> {
133        let p = self.piece_of(cp)?;
134        let i = self.cps.partition_point(|&c| c <= cp).checked_sub(1)?;
135        let cp0 = *self.cps.get(i)?;
136        let step = cp.checked_sub(cp0)?.checked_mul(2)?;
137        if p.compressed { p.fc.checked_add(step)?.checked_div(2) } else { p.fc.checked_add(step) }
138    }
139
140    /// The piece holding `cp`, with the CP range it covers: for every CP in that range the
141    /// answers of [`PieceTable::fc_of_cp`] and the piece's Prm come from this span alone,
142    /// so a sequential walk searches the piece table once per piece.
143    pub(crate) fn span(&self, cp: u32) -> Option<PieceSpan<'_>> {
144        let i = self.cps.partition_point(|&c| c <= cp).checked_sub(1)?;
145        let pcd = self.pieces.get(i)?;
146        let cp_lo = *self.cps.get(i)?;
147        let cp_hi = *self.cps.get(i + 1)?;
148        Some(PieceSpan { cp_lo, cp_hi, pcd })
149    }
150
151    /// The CP just past the last piece: no text exists at or beyond it.
152    pub(crate) fn cp_end(&self) -> u32 {
153        self.cps.last().copied().unwrap_or(0)
154    }
155
156    /// The character stored at `cp` (control characters included) and how many CPs it
157    /// covers (2 for a surrogate pair), found by binary search over the pieces.
158    /// `Err(next)` when there is no readable character at `cp` (a gap between pieces, an
159    /// inverted piece, or piece bytes past the end of the stream): `next` is the CP to
160    /// resume at, always greater than `cp` (`u32::MAX` past the last piece), so callers
161    /// skip an unreadable range in one step instead of one CP at a time.
162    pub(crate) fn char_at(&self, word: &[u8], cp: u32) -> Result<(char, u32), u32> {
163        let i = self.cps.partition_point(|&c| c <= cp);
164        // The next piece boundary above `cp`, or the end of the CP space.
165        let next = self.cps.get(i).copied().filter(|&n| n > cp).unwrap_or(u32::MAX);
166        let Some(i) = i.checked_sub(1) else { return Err(next) };
167        let (Some(p), Some(&cp0)) = (self.pieces.get(i), self.cps.get(i)) else { return Err(u32::MAX) };
168        let Some(off) = cp.checked_sub(cp0) else { return Err(next) };
169        let off = off as usize;
170        if p.compressed {
171            // 8-bit, Windows-1252, one byte per character, stored at fc/2.
172            let at = ((p.fc / 2) as usize).checked_add(off);
173            return match at.and_then(|a| word.get(a)) {
174                Some(&b) => Ok((cp1252(b), 1)),
175                None => Err(next),
176            };
177        }
178        // UTF-16LE, one 16-bit unit per character, stored at fc.
179        let unit = |k: usize| -> Option<u16> {
180            let at = (p.fc as usize).checked_add(k.checked_mul(2)?)?;
181            word.get(at..at.checked_add(2)?).map(|b| u16::from_le_bytes([b[0], b[1]]))
182        };
183        let Some(u) = unit(off) else { return Err(next) };
184        if (0xD800..0xDC00).contains(&u)
185            && cp.saturating_add(1) < next
186            && let Some(lo) = unit(off.saturating_add(1)).filter(|lo| (0xDC00..0xE000).contains(lo))
187            && let Some(Ok(c)) = char::decode_utf16([u, lo]).next()
188        {
189            return Ok((c, 2));
190        }
191        Ok((char::from_u32(u as u32).unwrap_or('\u{FFFD}'), 1))
192    }
193}

195/// One piece and the CPs `[cp_lo, cp_hi)` it covers ([`PieceTable::span`]).
196#[derive(Clone, Copy)]
197pub(crate) struct PieceSpan<'a> {
198    cp_lo: u32,
199    cp_hi: u32,
200    pcd: &'a Pcd,
201}
202
203impl PieceSpan<'_> {
204    pub(crate) fn contains(&self, cp: u32) -> bool {
205        self.cp_lo <= cp && cp < self.cp_hi
206    }
207
208    /// [`PieceTable::fc_of_cp`] for a CP inside the span.
209    pub(crate) fn fc_at(&self, cp: u32) -> Option<u32> {
210        let step = cp.checked_sub(self.cp_lo)?.checked_mul(2)?;
211        if self.pcd.compressed { self.pcd.fc.checked_add(step)?.checked_div(2) } else { self.pcd.fc.checked_add(step) }
212    }
213
214    /// The raw Prm of the piece: two CPs with the same value get the same [`PrmRef`].
215    pub(crate) fn prm(&self) -> u16 {
216        self.pcd.prm
217    }
218}
219
220/// What a piece's `Prm` refers to.
221pub(crate) enum PrmRef<'a> {
222    None,
223    /// Prm0: a short sprm index into the isprm table, with a 1-byte operand.
224    Sprm0 {
225        isprm: u8,
226        val: u8,
227    },
228    /// Prm1: a grpprl from a Prc (absent if out of range).
229    Grpprl(Option<&'a [u8]>),
230}
