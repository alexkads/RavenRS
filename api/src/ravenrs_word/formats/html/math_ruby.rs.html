4/// A `<math>` element (#816): its tokens up to the matching end tag, as a tree. Nesting and size
5/// are bounded; an unclosed element ends with the input.
6pub(super) fn read_math(lx: &mut Lexer, name: String, attrs: Vec<(String, String)>) -> crate::formats::mathml::El {
7    use crate::formats::mathml::{El, Item, MAX_NODES, local};
8    let mut stack: Vec<El> = vec![El { name, attrs, children: Vec::new() }];
9    let mut count = 0usize;
10    let close = |stack: &mut Vec<El>| {
11        if stack.len() > 1
12            && let Some(done) = stack.pop()
13            && let Some(parent) = stack.last_mut()
14        {
15            parent.children.push(Item::El(done));
16        }
17    };
18    while let Some(t) = lx.next() {
19        count += 1;
20        if count > MAX_NODES * 4 {
21            // Over the cap: drop the rest of this element so its tokens aren't read as HTML text.
22            skip_math(lx, t);
23            break;
24        }
25        match t {
26            Tok::Text(t) => {
27                if let Some(top) = stack.last_mut()
28                    && !t.is_empty()
29                {
30                    top.children.push(Item::Text(t));
31                }
32            }
33            Tok::Start { name, attrs, self_close } => {
34                let el = El { name, attrs, children: Vec::new() };
35                if self_close || stack.len() > crate::doc::math::MAX_DEPTH {
36                    if let Some(top) = stack.last_mut() {
37                        top.children.push(Item::El(el));
38                    }
39                } else {
40                    stack.push(el);
41                }
42            }
43            Tok::End(name) => {
44                let name = local(&name).to_string();
45                // Close up to the matching open element; a stray end tag is ignored.
46                if let Some(k) = stack.iter().rposition(|e| local(&e.name) == name) {
47                    if k == 0 {
48                        break;
49                    }
50                    while stack.len() > k {
51                        close(&mut stack);
52                    }
53                }
54            }
55        }
56    }
57    while stack.len() > 1 {
58        close(&mut stack);
59    }
60    stack.pop().unwrap_or_default()
61}

63/// Skip tokens (starting with `first`) up to and including the `</math>` that closes the element
64/// being read; nested `<math>` elements are counted. An unclosed element ends with the input.
65fn skip_math(lx: &mut Lexer, first: Tok) {
66    use crate::formats::mathml::local;
67    let mut depth = 0usize;
68    let mut t = Some(first);
69    while let Some(tok) = t {
70        match tok {
71            Tok::Start { name, self_close: false, .. } if local(&name) == "math" => depth = depth.saturating_add(1),
72            Tok::End(name) if local(&name) == "math" => {
73                if depth == 0 {
74                    return;
75                }
76                depth = depth.saturating_sub(1);
77            }
78            _ => {}
79        }
80        t = lx.next();
81    }
82}

84/// A `<ruby>` element's base text and reading (#1681).
85#[derive(Default)]
86pub(super) struct RubyText {
87    pub(super) base: String,
88    pub(super) ruby: String,
89}
90
91/// Which part of a `<ruby>` element the text being read belongs to.
92#[derive(Clone, Copy, PartialEq, Eq)]
93enum RubyPart {
94    /// The text being annotated: an `<rb>` child, or the text outside `<rt>`/`<rp>`.
95    Base,
96    /// The reading: an `<rt>` child.
97    Ruby,
98    /// `<rp>` fallback parentheses, which are not text: browsers show them only where ruby isn't
99    /// supported.
100    Fallback,
101}
102
103/// A block-level tag ends an open `<ruby>` implicitly, as it does in a browser.
104fn ends_ruby(t: &Tok) -> bool {
105    match t {
106        Tok::Start { name, .. } | Tok::End(name) => block_kind(name).is_some(),
107        Tok::Text(_) => false,
108    }
109}

111/// Read a `<ruby>` element's content: its base text from `<rb>` children (or the text outside
112/// them), its reading from `<rt>` children, and `<rp>` fallback parentheses dropped. Tolerant,
113/// like the rest of the reader: tags inside are ignored, an unclosed element ends at the next
114/// block-level tag (handed back to the caller) or with the input, and the tokens read are capped.
115pub(super) fn read_ruby(lx: &mut Lexer) -> RubyText {
116    /// Most tokens one `<ruby>` element is read from: a missing end tag must not scan the rest of
117    /// a hostile file here.
118    const MAX_TOKENS: usize = 4096;
119    let mut out = RubyText::default();
120    let mut part = RubyPart::Base;
121    let mut nested = 0usize;
122    let mut count = 0usize;
123    while count < MAX_TOKENS {
124        count += 1;
125        let start = lx.pos;
126        let Some(t) = lx.next() else { break };
127        if nested == 0 && ends_ruby(&t) {
128            // The boundary ends the element: give the token back to the main reader.
129            lx.pos = start;
130            break;
131        }
132        match t {
133            Tok::Text(t) => match part {
134                RubyPart::Base => out.base.push_str(&t),
135                RubyPart::Ruby => out.ruby.push_str(&t),
136                RubyPart::Fallback => {}
137            },
138            // The self-closing slash is ignored on HTML elements (`<rt/>` opens an `rt`), as in a
139            // browser's parser.
140            Tok::Start { name, self_close, .. } => match name.as_str() {
141                "rb" => part = RubyPart::Base,
142                "rt" => part = RubyPart::Ruby,
143                "rp" => part = RubyPart::Fallback,
144                "ruby" if !self_close => nested += 1,
145                _ => {}
146            },
147            Tok::End(name) => match name.as_str() {
148                "rb" | "rt" | "rp" => part = RubyPart::Base,
149                "ruby" if nested > 0 => nested -= 1,
150                "ruby" => break,
151                _ => {}
152            },
153        }
154    }
155    out
156}
