5/// A CSS length (or font-size keyword) in points; `None` when it is not one.
6pub fn css_len_pt(v: &str) -> Option<f32> {
7    let v = v.trim().to_ascii_lowercase();
8    let num = |s: &str| s.trim().parse::<f32>().ok().filter(|x| x.is_finite());
9    if let Some(x) = v.strip_suffix("pt") {
10        num(x)
11    } else if let Some(x) = v.strip_suffix("px") {
12        num(x).map(|p| p * 0.75)
13    } else if let Some(x) = v.strip_suffix("rem").or_else(|| v.strip_suffix("em")) {
14        num(x).map(|p| p * 12.0)
15    } else if let Some(x) = v.strip_suffix("in") {
16        num(x).map(|p| p * 72.0)
17    } else if let Some(x) = v.strip_suffix("cm") {
18        num(x).map(|p| p * 72.0 / 2.54)
19    } else if let Some(x) = v.strip_suffix("mm") {
20        num(x).map(|p| p * 72.0 / 25.4)
21    } else if let Some(x) = v.strip_suffix('%') {
22        num(x).map(|p| p * 0.12)
23    } else {
24        match v.as_str() {
25            "xx-small" => Some(7.0),
26            "x-small" => Some(7.5),
27            "small" | "smaller" => Some(10.0),
28            "medium" => Some(12.0),
29            "large" | "larger" => Some(13.5),
30            "x-large" => Some(18.0),
31            "xx-large" => Some(24.0),
32            "xxx-large" => Some(36.0),
33            _ => num(&v).map(|p| p * 0.75),
34        }
35    }
36}

38pub(super) fn strip_important(v: &str) -> &str {
39    let s = v.trim();
40    if let Some(pos) = s.rfind('!') {
41        let after = s[pos + 1..].trim();
42        if after.eq_ignore_ascii_case("important") {
43            return s[..pos].trim();
44        }
45    }
46    s
47}

49/// Decode CSS escapes (CSS Syntax §4.3.7): `\` and one to six hex digits, with one whitespace
50/// after them, is that code point; `\` and any other character is that character. CR LF, CR and
51/// form feed count as one newline, as CSS reads them (§3.3).
52pub(super) fn css_unescape(s: &str) -> String {
53    let mut o = String::with_capacity(s.len());
54    let mut it = s.chars().peekable();
55    // Takes a newline (CR LF as one) or other whitespace off the front, if there is one.
56    let skip_space = |it: &mut std::iter::Peekable<std::str::Chars>| match it.peek() {
57        Some('\r') => {
58            it.next();
59            it.next_if_eq(&'\n');
60        }
61        Some(' ' | '\t' | '\n' | '\u{C}') => {
62            it.next();
63        }
64        _ => {}
65    };
66    while let Some(c) = it.next() {
67        if c != '\\' {
68            o.push(c);
69            continue;
70        }
71        let mut hex = String::new();
72        while hex.len() < 6 && it.peek().is_some_and(char::is_ascii_hexdigit) {
73            hex.extend(it.next());
74        }
75        if hex.is_empty() {
76            // Before a newline it continues the line; at the end it is dropped.
77            match it.peek() {
78                Some('\r' | '\n' | '\u{C}') => skip_space(&mut it),
79                Some(_) => o.extend(it.next()),
80                None => {}
81            }
82            continue;
83        }
84        skip_space(&mut it);
85        let cp = u32::from_str_radix(&hex, 16).unwrap_or(0);
86        o.push(char::from_u32(cp).filter(|c| *c != '\0').unwrap_or('\u{FFFD}'));
87    }
88    o
89}

91/// Apply a `style` attribute to formatting / block properties.
92pub(super) fn apply_style(
93    css: &str,
94    f: &mut Fmt,
95    align: &mut Option<Align>,
96    preserve: &mut bool,
97    page_break: &mut bool,
98    page_break_after: &mut bool,
99) {
100    for (k, v) in decls(css) {
101        let lv = v.to_ascii_lowercase();
102        match k.as_str() {
103            "font-weight" => f.bold = lv.starts_with("bold") || lv.parse::<u32>().is_ok_and(|w| w >= 600),
104            "font-style" => f.italic = lv == "italic" || lv == "oblique",
105            "text-decoration" | "text-decoration-line" => {
106                if lv.contains("none") {
107                    f.underline = false;
108                    f.strike = false;
109                }
110                if lv.contains("underline") {
111                    f.underline = true;
112                }
113                if lv.contains("line-through") {
114                    f.strike = true;
115                }
116            }
117            "color" => {
118                if let Some(c) = parse_color(v) {
119                    f.color = Some(c);
120                }
121            }
122            "background-color" | "background" => {
123                if let Some(c) = v.split_whitespace().find_map(parse_color) {
124                    f.background = Some(c);
125                }
126            }
127            "font-size" => {
128                if let Some(p) = css_len_pt(v).filter(|p| *p >= 1.0 && *p <= 1638.0) {
129                    f.size = Some(p);
130                }
131            }
132            "font-family" => {
133                let fam = css_unescape(v.split(',').next().unwrap_or("").trim().trim_matches(['"', '\'']));
134                if model::is_mono(&fam) || fam.eq_ignore_ascii_case("monospace") {
135                    f.code = true;
136                } else if !fam.is_empty() && !matches!(fam.to_ascii_lowercase().as_str(), "serif" | "sans-serif" | "inherit" | "initial") {
137                    f.font = model::kept_font(&fam).or(f.font.take());
138                } else if f.font.is_some() {
139                    // A generic family replaces a face the text has (a page's Times New Roman);
140                    // text without one keeps the document's font.
141                    match fam.to_ascii_lowercase().as_str() {
142                        "serif" => f.font = Some(PAGE_FONT.into()),
143                        "sans-serif" => f.font = Some("Arial".into()),
144                        _ => {}
145                    }
146                }
147            }
148            "vertical-align" => {
149                f.sup = lv == "super";
150                f.sub = lv == "sub";
151            }
152            "text-align" => {
153                *align = match lv.as_str() {
154                    "center" => Some(Align::Center),
155                    "right" | "end" => Some(Align::Right),
156                    "justify" => Some(Align::Justify),
157                    "left" | "start" => Some(Align::Left),
158                    _ => *align,
159                }
160            }
161            "white-space" => *preserve = lv.starts_with("pre"),
162            "page-break-before" | "break-before" => *page_break = lv == "always" || lv == "page",
163            "page-break-after" | "break-after" => *page_break_after = lv == "always" || lv == "page",
164            _ => {}
165        }
166    }
