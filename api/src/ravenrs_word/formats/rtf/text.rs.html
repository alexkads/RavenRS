5/// The encoding of an East Asian multibyte code page (932 Shift-JIS, 936 GBK, 949 UHC, 950 Big5),
6/// which [`Reader::finish_multibyte`] decodes a run with; `None` for the single-byte code pages.
7fn multibyte_encoding(code_page: u16) -> Option<&'static encoding_rs::Encoding> {
8    let label = crate::doc::encoding::multibyte_encoding_name(code_page)?;
9    encoding_rs::Encoding::for_label(label.as_bytes())
10}

13    pub(super) fn commit_font(&mut self) {
14        let name = std::mem::take(&mut self.font_name).trim().to_string();
15        if let Some(name) = model::kept_font(&name).filter(|n| !n.is_empty())
16            && self.fonts.len() < 4096
17        {
18            self.fonts.insert(self.font_cur, name);
19            if let Some(cp) = self.font_cp_cur.take() {
20                self.font_cps.insert(self.font_cur, cp);
21            }
22        }
23    }

25    /// The code page 8-bit text is read in: the current font's, else the default font's, else the
26    /// document's (Windows-1252 when none is named).
27    fn code_page(&self) -> u16 {
28        self.st.cp.or_else(|| self.font_cps.get(&self.deff).copied()).unwrap_or(if self.ansicpg == 0 { 1252 } else { self.ansicpg })
29    }

31    /// A byte of 8-bit text (`\'hh` or raw): a single-byte code page decodes it at once; a
32    /// multibyte one is fed to a streaming decoder ([`Reader::finish_multibyte`]), so a lead byte decodes
33    /// together with its trail byte.
34    pub(super) fn byte(&mut self, b: u8) {
35        if self.skip_chars > 0 {
36            self.skip_chars -= 1;
37            return;
38        }
39        // `\'3b` in a font name is a `;` in every code page, and part of the name (see `ch`).
40        if b == b';' && self.st.dest == Dest::FontTbl {
41            self.ch(';');
42            return;
43        }
44        let cp = self.code_page();
45        let Some(encoding) = multibyte_encoding(cp) else {
46            self.finish_multibyte();
47            let c = crate::doc::encoding::code_page_byte(cp, b);
48            self.ch(c);
49            return;
50        };
51        // A code page switch ends the previous run (a lead byte left then becomes U+FFFD, as Word
52        // writes it) before the new decoder starts.
53        if self.mb_cp != cp || self.mb_decoder.is_none() {
54            self.finish_multibyte();
55            self.mb_cp = cp;
56            self.mb_decoder = Some(encoding.new_decoder_without_bom_handling());
57        }
58        self.decode_run(&[b]);
59    }

61    /// Feed 8-bit text bytes to the open multibyte decoder and append what it emits. A single
62    /// `\'hh` byte decodes into a stack buffer (no heap allocation per byte of escaped text); a
63    /// raw run decodes in one call, so a long ASCII run in a Japanese, Chinese or Korean document
64    /// costs one append instead of one per byte. The destination's spare capacity is all
65    /// `decode_to_*` writes into, so a full buffer is drained and decoding continues.
66    fn decode_run(&mut self, bytes: &[u8]) {
67        // The decoder is taken out so the text it produced can be appended (which borrows the
68        // reader) before it goes back.
69        let Some(mut decoder) = self.mb_decoder.take() else { return };
70        let mut read = 0;
71        if bytes.len() <= 16 {
72            // A character is at most 4 UTF-8 bytes, and a byte may also replace a pending lead
73            // byte (3 more), so one escaped byte always fits the stack buffer.
74            let mut buf = [0u8; 16];
75            let (result, got, written, _) = decoder.decode_to_utf8(bytes, &mut buf, false);
76            read = got;
77            let full = matches!(result, encoding_rs::CoderResult::OutputFull);
78            if written > 0
79                && let Ok(s) = std::str::from_utf8(&buf[..written])
80            {
81                self.text(s);
82            }
83            if !full {
84                self.mb_decoder = Some(decoder);
85                return;
86            }
87        }
88        let mut out = String::with_capacity(bytes.len().saturating_mul(2));
89        let mut rest = bytes.get(read..).unwrap_or_default();
90        loop {
91            let (result, got, _) = decoder.decode_to_string(rest, &mut out, false);
92            rest = rest.get(got..).unwrap_or_default();
93            if !matches!(result, encoding_rs::CoderResult::OutputFull) {
94                break;
95            }
96            out.reserve(16.max(rest.len().saturating_mul(2)));
97        }
98        self.mb_decoder = Some(decoder);
99        if !out.is_empty() {
100            self.text(&out);
101        }
102    }

104    /// A run of raw 8-bit text from the file (not `\'hh` escapes): a single-byte code page maps
105    /// each byte to a char; a multibyte one decodes the whole run in one pass, so a long ASCII run
106    /// in a Japanese, Chinese or Korean document costs one decode and one append instead of one
107    /// per byte.
108    ///
109    /// The decoder is created without BOM handling: the code page comes from `\ansicpg`, `\cpg` or
110    /// `\fcharset` and must not change. With BOM sniffing, a run that starts `EF BB BF` in GBK,
111    /// Big5 or UHC (a valid character pair followed by a lead byte) would be dropped as a UTF-8
112    /// BOM and the rest decoded as UTF-8.
113    pub(super) fn run(&mut self, chunk: &[u8]) {
114        // The fallback bytes after `\uN` are skipped; the rest of the run is text like any other.
115        let skip = self.skip_chars.min(chunk.len());
116        self.skip_chars -= skip;
117        let chunk = chunk.get(skip..).unwrap_or_default();
118        if chunk.is_empty() {
119            return;
120        }
121        let cp = self.code_page();
122        let Some(encoding) = multibyte_encoding(cp) else {
123            self.finish_multibyte();
124            let mut out = String::with_capacity(chunk.len());
125            for b in chunk {
126                out.push(crate::doc::encoding::code_page_byte(cp, *b));
127            }
128            if !out.is_empty() {
129                self.text(&out);
130            }
131            return;
132        };
133        if self.mb_cp != cp || self.mb_decoder.is_none() {
134            self.finish_multibyte();
135            self.mb_cp = cp;
136            self.mb_decoder = Some(encoding.new_decoder_without_bom_handling());
137        }
138        self.decode_run(chunk);
139    }

141    /// Ends the current multibyte run: the streaming decoder emits whatever it still holds (a lone
142    /// lead byte becomes U+FFFD, as Word writes it) and is dropped. A code page switch, a single-byte
143    /// run, a group end and the end of the document finish the run this way.
144    pub(super) fn finish_multibyte(&mut self) {
145        let Some(mut decoder) = self.mb_decoder.take() else { return };
146        let mut out = String::with_capacity(16);
147        let _ = decoder.decode_to_string(&[], &mut out, true);
148        if !out.is_empty() {
149            self.text(&out);
150        }
151    }

153    pub(super) fn finish_pict(&mut self) {
154        if self.pict_half.take().is_some() {
155            // Odd hex digit count: drop the dangling nibble.
156        }
157        let data = std::mem::take(&mut self.pict_hex);
158        if self.pict_kind == 0 || data.is_empty() {
159            return;
160        }
161        let sx = if self.pict_sx > 0.0 { self.pict_sx / 100.0 } else { 1.0 };
162        let sy = if self.pict_sy > 0.0 { self.pict_sy / 100.0 } else { 1.0 };
163        let w = self.pict_w.map(|v| v * sx);
164        let h = self.pict_h.map(|v| v * sy);
165        // The control word only says "a metafile": the bytes decide what it is (#4259).
166        let img = if matches!(self.pict_kind, 3 | 4) && crate::metafile::is_metafile(&data) {
167            // An EMF starts with its header record, type 1.
168            let ext = if data.get(..4) == Some(&[1, 0, 0, 0]) { "emf" } else { "wmf" };
169            // `\picw`/`\pich` are in the units of the `\wmetafileN` mapping mode; 0.01 mm for the
170            // isotropic and anisotropic modes and for EMF.
171            let unit = match (self.pict_kind, self.pict_mm) {
172                (3, 1) => 0.75,         // MM_TEXT: pixels
173                (3, 2) => 72.0 / 254.0, // MM_LOMETRIC: 0.1 mm
174                (3, 4) => 0.72,         // MM_LOENGLISH: 0.01 in
175                (3, 5) => 0.072,        // MM_HIENGLISH: 0.001 in
176                (3, 6) => 0.05,         // MM_TWIPS
177                _ => 72.0 / 2540.0,     // MM_HIMETRIC, MM_ISOTROPIC, MM_ANISOTROPIC
178            };
179            let pt = |v: i32| (v > 0).then(|| v.min(1_000_000) as f32 * unit);
180            // `\picscalex`/`\picscaley` scale the natural size too, when no goal size is given.
181            let natural = pt(self.pict_nat.0).zip(pt(self.pict_nat.1)).map(|(a, b)| (a * sx, b * sy));
182            Some(model::sized_img(Arc::new(data), ext, natural, w, h, ""))
183        } else {
184            make_img(data, w, h, "")
185        };
186        if let Some(img) = img {
187            self.para.inlines.push(Inline::Image(img));
188            self.para_has = true;
189        }
190    }

192    pub(super) fn text(&mut self, s: &str) {
193        match self.st.dest {
194            Dest::Body => {
195                if self.st.hidden {
196                    return;
197                }
198                let mut f = self.st.fmt.clone();
199                if s.contains([MARK_OPEN, MARK_CLOSE]) {
200                    self.para.push_text(&s.replace([MARK_OPEN, MARK_CLOSE], "\u{FFFD}"), &f);
201                    self.para_has = true;
202                    return;
203                }
204                let r = self.st.rev;
205                if r.ins {
206                    f.ins = Some(self.rev_mark(r.auth, r.dttm));
207                }
208                if r.del {
209                    f.del = Some(self.rev_mark(r.authdel, r.dttmdel));
210                }
211                let c = &self.st.chg_c;
212                if let Some(words) = c.old.clone() {
213                    let mark = self.rev_mark(c.auth, c.date);
214                    let old = self.old_fmt(&words);
215                    f.fmt_change = Some(Box::new(model::FmtChange { mark, old }));
216                }
217                self.para.push_text(s, &f);
218                self.para_has = true;
219            }
220            Dest::FontTbl => {
221                for c in s.chars() {
222                    if c == ';' {
223                        self.commit_font();
224                    } else if self.font_name.len() <= model::MAX_FONT_NAME {
225                        // One past the limit is enough for `commit_font` to drop the name.
226                        self.font_name.push(c);
227                    }
228                }
229            }
230            Dest::ColorTbl => {
231                for c in s.chars() {
232                    if c == ';' {
233                        let (r, g, b, set) = self.color_cur;
234                        if self.colors.len() < 4096 {
235                            self.colors.push(if set { Some(Rgb(r, g, b)) } else { None });
236                        }
237                        self.color_cur = (0, 0, 0, false);
238                    }
239                }
240            }
241            Dest::StyleSheet => {
242                for c in s.chars() {
243                    if c == ';' {
244                        self.commit_style();
245                    } else if self.style_name.len() < model::MAX_FONT_NAME {
246                        self.style_name.push(c);
247                    } else {
248                        model::dropped(model::Dropped::StyleName);
249                    }
250                }
251            }
252            Dest::Info(_) => self.info_text.push_str(s),
253            Dest::Atn(_) => {
254                if self.atn_text.len() < 256 {
255                    self.atn_text.push_str(s);
256                }
257            }
258            Dest::FldInst => {
259                if let Some(f) = self.st.field.and_then(|i| self.fields.get_mut(i))
260                    && f.len() < 4096
261                {
262                    f.push_str(s);
263                }
264            }
265            Dest::Pict => {
266                for b in s.bytes() {
267                    let v = match b {
268                        b'0'..=b'9' => b - b'0',
269                        b'a'..=b'f' => b - b'a' + 10,
270                        b'A'..=b'F' => b - b'A' + 10,
271                        _ => continue,
272                    };
273                    match self.pict_half.take() {
274                        Some(h) => {
275                            if self.pict_hex.len() < MAX_PICT {
276                                self.pict_hex.push((h << 4) | v);
277                            }
278                        }
279                        None => self.pict_half = Some(v),
280                    }
281                }
282            }
283            Dest::Bookmark => {
284                if let Some((_, n)) = &mut self.bkmk
285                    && n.len() < 256
286                {
287                    n.push_str(s);
288                }
289            }
290            Dest::RevTbl => {
291                for c in s.chars() {
292                    if c == ';' {
293                        self.commit_author();
294                    } else if self.rev_name.len() < 256 {
295                        self.rev_name.push(c);
296                    }
297                }
298            }
299            Dest::Skip | Dest::ListTable | Dest::ListOverride | Dest::OldProps(_) => {}
300        }
301    }

303    /// The formatting before a tracked character change: the current run's, with the
304    /// `\oldcprops` words applied (a `\plain` among them starts from none).
305    fn old_fmt(&mut self, words: &[(String, Option<i32>)]) -> Fmt {
306        let fmt = Fmt { link: None, ins: None, del: None, fmt_change: None, ..self.st.fmt.clone() };
307        let saved = std::mem::replace(&mut self.st, St { dest: Dest::OldProps(false), fmt, ..Default::default() });
308        for (w, p) in words {
309            self.word(w, *p, false, false);
310        }
311        let old = std::mem::replace(&mut self.st, saved).fmt;
312        Fmt { link: None, ..old }
313    }

315    pub(super) fn ch(&mut self, c: char) {
316        if self.skip_chars > 0 {
317            self.skip_chars -= 1;
318            return;
319        }
320        self.finish_multibyte();
321        // Only a literal `;` ends a font name: an escaped one (`\'3b`, `\u59?`) belongs to it.
322        if c == ';' && self.st.dest == Dest::FontTbl {
323            // Held to the limit as other characters are; one past it drops the name.
324            if self.font_name.len() <= model::MAX_FONT_NAME {
325                self.font_name.push(c);
326            }
327            return;
328        }
329        let mut buf = [0u8; 4];
330        self.text(c.encode_utf8(&mut buf));
331    }
