5/// Parse RTF bytes into the flow model.
6pub fn parse(bytes: &[u8]) -> Result<Flow, String> {
7    parse_with_page(bytes).map(|(f, _, _)| f)
8}

10/// The flow, the page, and the default font (`\deff`, font 0 when the file names none), which
11/// is the font of all text that doesn't select one.
12fn parse_with_page(bytes: &[u8]) -> Result<(Flow, PageGeom, Option<String>), String> {
13    parse_full(bytes, false).map(|r| (r.flow, r.page, r.default_font))
14}

16/// Everything an import takes from an RTF file.
17struct Parsed {
18    flow: Flow,
19    page: PageGeom,
20    /// The last section's properties (the others are on the paragraphs that end them).
21    last_section: crate::doc::SectionProps,
22    default_font: Option<String>,
23    /// Each section's headers and footers, in order, the last section's last.
24    headers_footers: Vec<Vec<(HeaderFooter, Vec<FBlock>)>>,
25    notes: Vec<NoteData>,
26    comments: Vec<CommentData>,
27}

29/// `marks`: also read footnotes, endnotes and comments (as place markers in the flow's text, for
30/// [`import`] to turn into objects); without it they are skipped.
31fn parse_full(bytes: &[u8], marks: bool) -> Result<Parsed, String> {
32    let start = bytes.iter().position(|b| !b.is_ascii_whitespace()).unwrap_or(0);
33    if !bytes.get(start..).is_some_and(|b| b.starts_with(b"{\\rtf")) {
34        return Err("not an RTF document (it doesn't start with {\\rtf)".into());
35    }
36    let mut r = Reader::default();
37    r.marks_on = marks;
38    let mut i = start;
39    let mut first_in_group = false;
40    let mut starred = false;
41    while let Some(&c) = bytes.get(i) {
42        match c {
43            b'{' => {
44                r.open();
45                first_in_group = true;
46                starred = false;
47                i += 1;
48                continue;
49            }
50            b'}' => {
51                r.close();
52                if r.info_depth.is_some_and(|d| r.depth() < d) {
53                    r.info_depth = None;
54                }
55                first_in_group = false;
56                starred = false;
57                i += 1;
58                continue;
59            }
60            b'\\' => {
61                let Some(&n) = bytes.get(i + 1) else { break };
62                if n.is_ascii_alphabetic() {
63                    let mut j = i + 1;
64                    while bytes.get(j).is_some_and(u8::is_ascii_alphabetic) && j - i <= 32 {
65                        j += 1;
66                    }
67                    let word = std::str::from_utf8(bytes.get(i + 1..j).unwrap_or(b"")).unwrap_or("");
68                    let mut k = j;
69                    let neg = bytes.get(k) == Some(&b'-');
70                    if neg {
71                        k += 1;
72                    }
73                    let d0 = k;
74                    while bytes.get(k).is_some_and(u8::is_ascii_digit) && k - d0 < 10 {
75                        k += 1;
76                    }
77                    let param = if k > d0 {
78                        let v = std::str::from_utf8(bytes.get(d0..k).unwrap_or(b"0")).ok().and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
79                        let v = if neg { -v } else { v };
80                        // A date (DTTM) is a 32-bit pattern some writers put unsigned.
81                        let v = if matches!(word, "revdttm" | "revdttmdel" | "crdate" | "prdate")
82                            && v > i64::from(i32::MAX)
83                            && v <= i64::from(u32::MAX)
84                        {
85                            v - (1i64 << 32)
86                        } else {
87                            v
88                        };
89                        Some(v.clamp(i64::from(i32::MIN), i64::from(i32::MAX)) as i32)
90                    } else {
91                        if neg {
92                            k -= 1;
93                        }
94                        None
95                    };
96                    if bytes.get(k) == Some(&b' ') {
97                        k += 1;
98                    }
99                    i = k;
100                    if word == "bin" {
101                        let n = param.unwrap_or(0).max(0) as usize;
102                        let end = i.saturating_add(n).min(bytes.len());
103                        if r.st.dest == Dest::Pict
104                            && let Some(data) = bytes.get(i..end)
105                            && r.pict_hex.len() + data.len() <= MAX_PICT
106                        {
107                            r.pict_hex.extend_from_slice(data);
108                        }
109                        i = end;
110                        continue;
111                    }
112                    if r.skip_chars > 0
113                        && matches!(
114                            r.st.dest,
115                            Dest::Body | Dest::Info(_) | Dest::FldInst | Dest::Bookmark | Dest::FontTbl | Dest::RevTbl | Dest::Atn(_)
116                        )
117                    {
118                        r.skip_chars -= 1;
119                        first_in_group = false;
120                        continue;
121                    }
122                    if word == "info" {
123                        r.info_depth = Some(r.depth());
124                    }
125                    r.word(word, param, starred, first_in_group);
126                    first_in_group = false;
127                    starred = false;
128                    continue;
129                }
130                if r.skip_chars > 0 && n != b'\'' {
131                    r.skip_chars -= 1;
132                    i += 2;
133                    first_in_group = false;
134                    continue;
135                }
136                match n {
137                    b'\'' => {
138                        // `from_str_radix` takes a leading `+`: a pair that isn't both hex digits
139                        // is not an escape at all.
140                        let h = bytes
141                            .get(i + 2..i + 4)
142                            .and_then(|h| std::str::from_utf8(h).ok())
143                            .filter(|h| h.bytes().all(|b| b.is_ascii_hexdigit()))
144                            .and_then(|h| u8::from_str_radix(h, 16).ok());
145                        match h {
146                            Some(v) => {
147                                if r.st.dest == Dest::Pict {
148                                    // Hex digits, two at a time, without a String per picture byte.
149                                    const HEX: &[u8; 16] = b"0123456789abcdef";
150                                    let h = [HEX[usize::from(v >> 4)], HEX[usize::from(v & 0x0f)]];
151                                    if let Ok(s) = std::str::from_utf8(&h) {
152                                        r.text(s);
153                                    }
154                                } else {
155                                    r.byte(v);
156                                }
157                                i += 4;
158                            }
159                            // Not two hex digits: the escape is malformed. Consume the escape and
160                            // any hex digits of a truncated pair, and leave every other byte,
161                            // `}` or a control word's `\`, to be read as the structure it is.
162                            None => {
163                                i += 2;
164                                let mut n = 0;
165                                while n < 2 && bytes.get(i).is_some_and(u8::is_ascii_hexdigit) {
166                                    i += 1;
167                                    n += 1;
168                                }
169                            }
170                        }
171                    }
172                    b'*' => {
173                        starred = true;
174                        i += 2;
175                        continue;
176                    }
177                    b'~' | b' ' => {
178                        r.ch('\u{00A0}');
179                        i += 2;
180                    }
181                    b'-' => {
182                        r.ch('\u{00AD}');
183                        i += 2;
184                    }
185                    b'_' => {
186                        r.ch('\u{2011}');
187                        i += 2;
188                    }
189                    b'\n' | b'\r' => {
190                        r.word("par", None, false, false);
191                        i += 2;
192                    }
193                    b'\\' | b'{' | b'}' => {
194                        r.ch(n as char);
195                        i += 2;
196                    }
197                    _ => i += 2,
198                }
199                first_in_group = false;
200            }
201            b'\r' | b'\n' | 0 => i += 1,
202            _ => {
203                // A run of plain text.
204                let mut j = i;
205                while bytes.get(j).is_some_and(|b| !matches!(b, b'{' | b'}' | b'\\' | b'\r' | b'\n' | 0)) {
206                    j += 1;
207                }
208                let chunk = bytes.get(i..j).unwrap_or(b"");
209                if r.st.dest == Dest::Pict && r.skip_chars == 0 && chunk.is_ascii() {
210                    // Picture hex is not text: it needs no decoder, so neither a pending lead byte
211                    // nor the code page's single-byte path applies to it.
212                    if let Ok(s) = std::str::from_utf8(chunk) {
213                        r.text(s);
214                    }
215                } else if r.skip_chars == 0 && chunk.is_ascii() && r.mb_decoder.is_none() {
216                    // Plain ASCII with no multibyte run open: append it as it is.
217                    if let Ok(s) = std::str::from_utf8(chunk) {
218                        r.text(s);
219                    }
220                } else {
221                    r.run(chunk);
222                }
223                i = j;
224                first_in_group = false;
225            }
226        }
227    }
228    Ok(finalize(r))
229}

231fn finalize(mut r: Reader) -> Parsed {
232    // A multibyte run ends here, and so does a note, comment, header or footer left open by a
233    // truncated file.
234    r.finish_multibyte();
235    r.end_sub();
236    r.end_header_footer();
237    if r.para_has || !r.para.inlines.is_empty() {
238        r.end_para(false);
239    }
240    r.close_table();
241    if let Some((y, mo, d, h, mi)) = r.created
242        && y > 0
243    {
244        r.meta.created = format!("{y:04}-{:02}-{:02}T{:02}:{:02}:00Z", mo.clamp(1, 12), d.clamp(1, 31), h.min(23), mi.min(59));
245    }
246    let mut blocks = std::mem::take(&mut r.body);
247    // A file that ends with `\sect` ends its last section there: no empty one follows, and
248    // header or footer groups after that `\sect` (a section without text) are dropped.
249    let mut last_section = r.sec.props(&r.page);
250    let mut headers_footers = std::mem::take(&mut r.section_parts);
251    if let Some(FBlock::Para(p)) = blocks.last_mut()
252        && let Some(s) = p.section.take()
253    {
254        last_section = *s;
255    } else {
256        headers_footers.push(std::mem::take(&mut r.hf_parts));
257    }
258    if blocks.is_empty() {
259        blocks.push(FBlock::Para(Para::default()));
260    }
261    let default_font = r.fonts.get(&r.deff).filter(|n| !model::is_mono(n)).map(|n| n.to_string());
262    Parsed { flow: Flow { blocks, meta: r.meta }, page: r.page, last_section, default_font, headers_footers, notes: r.notes, comments: r.comments }
263}

265/// The column grid of a table from its rows' cell right edges: every distinct edge starts a grid
266/// column, and each cell spans the grid columns between its left and right edges, as Word lays
267/// out RTF rows whose cells don't line up. Returns the widths in points, or none (equal columns)
268/// when the edges don't make a grid.
269pub(super) fn column_grid(rows: &mut [Vec<Cell>], rights: &[Vec<i32>]) -> Vec<f32> {
270    let mut edges: Vec<i32> = rights.iter().flatten().copied().filter(|&x| x > 0).collect();
271    edges.sort_unstable();
272    edges.dedup();
273    let ok = rights.len() == rows.len()
274        && rows
275            .iter()
276            .zip(rights)
277            .all(|(r, x)| r.len() == x.len() && x.windows(2).all(|w| w.first() < w.get(1)) && x.first().is_some_and(|&v| v > 0))
278        && !edges.is_empty()
279        && edges.len() <= 63;
280    if !ok {
281        return Vec::new();
282    }
283    for (row, x) in rows.iter_mut().zip(rights) {
284        let mut left = 0;
285        for (c, &right) in row.iter_mut().zip(x) {
286            let span = edges.iter().filter(|&&e| e > left && e <= right).count().max(1);
287            c.colspan = span as u32;
288            left = right;
289        }
290    }
291    let mut prev = 0;
292    edges
293        .iter()
294        .map(|&e| {
295            let w = (e - prev) as f32 / 20.0;
296            prev = e;
297            w
298        })
299        .collect()
300}

302/// Parse RTF bytes into a document.
303pub fn import(bytes: &[u8]) -> Result<Document, String> {
304    model::begin_import();
305    let Parsed { flow, last_section, default_font, headers_footers, notes, comments, .. } = parse_full(bytes, true)?;
306    let mut doc = model::to_doc_in(&flow, last_section);
307    // Each section's own headers and footers; those it leaves out stay linked to the previous.
308    let mut sets = Vec::with_capacity(headers_footers.len());
309    for parts in &headers_footers {
310        let (mut headers, mut footers) = (crate::doc::section::HeaderSet::default(), crate::doc::section::HeaderSet::default());
311        for (which, blocks) in parts {
312            let blocks = model::part_blocks(&mut doc, blocks);
313            let id = doc.add_part(if which.footer { crate::doc::PartKind::Footer } else { crate::doc::PartKind::Header }, blocks);
314            let set = if which.footer { &mut footers } else { &mut headers };
315            match which.pages {
316                HfPages::Default => set.default = Some(id),
317                HfPages::First => set.first = Some(id),
318                HfPages::Even => set.even = Some(id),
319            }
320        }
321        sets.push((headers, footers));
322    }
323    let mut sets = sets.into_iter();
324    for b in &mut doc.body {
325        if let crate::doc::Block::Para(p) = &**b
326            && p.section.is_some()
327            && let crate::doc::Block::Para(p) = std::sync::Arc::make_mut(b)
328            && let Some(s) = p.section.as_deref_mut()
329            && let Some((h, f)) = sets.next()
330        {
331            (s.headers, s.footers) = (h, f);
332        }
333    }
334    if let Some((h, f)) = sets.next() {
335        (doc.last_section.headers, doc.last_section.footers) = (h, f);
336    }
337    place_notes_and_comments(&mut doc, notes, comments);
338    // RTF's paragraph defaults are single spacing and no space around paragraphs; what a file
339    // wants beyond that it states on each paragraph.
340    let dp = &mut Arc::make_mut(&mut doc.styles).default_para;
341    dp.space_before = Some(0.0);
342    dp.space_after = Some(0.0);
343    dp.line_spacing = Some(crate::doc::props::LineSpacing::Multiple(1.0));
344    if default_font.is_some() {
345        Arc::make_mut(&mut doc.styles).default_chr.font = default_font;
346    }
347    Ok(doc)
348}

350/// Turn the place markers the reader left in the text into note references and comment ranges,
351/// and add the notes' and comments' stories to `doc`.
352fn place_notes_and_comments(doc: &mut Document, notes: Vec<NoteData>, comments: Vec<CommentData>) {
353    use crate::doc::para::NoteKind;
354    use crate::doc::{Comment, PartKind};
355    if notes.is_empty() && comments.is_empty() {
356        return;
357    }
358    // (kind, part id) per note, by index.
359    let mut note_parts: Vec<(NoteKind, u32)> = Vec::new();
360    for n in notes {
361        let (pk, nk, style) =
362            if n.endnote { (PartKind::Endnote, NoteKind::Endnote, "EndnoteText") } else { (PartKind::Footnote, NoteKind::Footnote, "FootnoteText") };
363        let mut blocks = model::part_blocks(doc, &n.blocks);
364        for b in &mut blocks {
365            if let Block::Para(p) = Arc::make_mut(b)
366                && p.props.style.is_none()
367            {
368                p.props.style = Some(style.to_string());
369            }
370        }
371        note_parts.push((nk, doc.add_part(pk, blocks)));
372    }
373    // RTF range id to the comment's id in the document.
374    let mut by_rtf: HashMap<i64, u32> = HashMap::new();
375    for (k, c) in comments.into_iter().enumerate() {
376        let id = k as u32;
377        let blocks = model::part_blocks(doc, &c.blocks);
378        let part = doc.add_part(PartKind::Comment, blocks);
379        doc.comments
380            .insert(id, Comment { author: c.author, initials: c.initials, date: c.date, parent: None, resolved: false, part, ..Default::default() });
381        by_rtf.insert(i64::from(c.rtf_id), id);
382    }
383    let ctx = Placer { note_parts, by_rtf };
384    let mut body = std::mem::take(&mut doc.body);
385    ctx.blocks(&mut body, None, 0);
386    doc.body = body;
387    let ids: Vec<u32> = doc.parts.keys().copied().collect();
388    for id in ids {
389        let Some(part) = doc.parts.get_mut(&id) else { continue };
390        let own = ctx.note_parts.iter().find(|(_, p)| *p == id).copied();
391        ctx.blocks(&mut part.blocks, own, 0);
392        // A note starts with its own number, as Word's do.
393        if let Some((kind, pid)) = own
394            && let Some(first) = part.blocks.first_mut()
395            && let Block::Para(p) = Arc::make_mut(first)
396            && !p.objects.iter().any(|o| matches!(o, InlineObject::NoteRef { id, .. } if *id == pid))
397        {
398            let props = Placer::ref_props(kind);
399            if p.insert_object(0, InlineObject::NoteRef { kind, id: pid, custom: String::new() }, &props).is_err() {
400                // The paragraph can't take it; the note simply has no number mark.
401            }
402        }
403    }
404}

406struct Placer {
407    note_parts: Vec<(crate::doc::para::NoteKind, u32)>,
408    by_rtf: HashMap<i64, u32>,
409}
410
411impl Placer {
412    fn ref_props(kind: crate::doc::para::NoteKind) -> crate::doc::CharProps {
413        let style = if kind == crate::doc::para::NoteKind::Endnote { "EndnoteReference" } else { "FootnoteReference" };
414        crate::doc::CharProps { style: Some(style.into()), ..Default::default() }
415    }
416
417    fn blocks(&self, bl: &mut Blocks, own: Option<(crate::doc::para::NoteKind, u32)>, depth: usize) {
418        for b in bl.iter_mut() {
419            let needs = match &**b {
420                Block::Para(p) => p.text.contains(MARK_OPEN),
421                Block::Table(_) => depth < 8,
422            };
423            if !needs {
424                continue;
425            }
426            match Arc::make_mut(b) {
427                Block::Para(p) => self.para(p, own),
428                Block::Table(t) => {
429                    for r in &mut t.rows {
430                        for c in &mut r.cells {
431                            self.blocks(&mut c.blocks, own, depth + 1);
432                        }
433                    }
434                }
435            }
436        }
437    }
438
439    fn para(&self, p: &mut Paragraph, own: Option<(crate::doc::para::NoteKind, u32)>) {
440        // Every pass removes one marker, so this ends; the bound is only a seat belt.
441        for _ in 0..100_000 {
442            let Some(a) = p.text.find(MARK_OPEN) else { return };
443            let Some(len) = p.text.get(a..).and_then(|t| t.find(MARK_CLOSE)) else { return };
444            let end = a + len + MARK_CLOSE.len_utf8();
445            let inner = p.text.get(a + MARK_OPEN.len_utf8()..a + len).unwrap_or("").to_string();
446            let mut chars = inner.chars();
447            let tag = chars.next().unwrap_or(' ');
448            let n = chars.as_str().parse::<i64>().unwrap_or(i64::MIN);
449            let (obj, props) = match tag {
450                'N' => match usize::try_from(n).ok().and_then(|i| self.note_parts.get(i)) {
451                    Some(&(kind, id)) => (Some(InlineObject::NoteRef { kind, id, custom: String::new() }), Self::ref_props(kind)),
452                    None => (None, Default::default()),
453                },
454                'S' => match own {
455                    Some((kind, id)) => (Some(InlineObject::NoteRef { kind, id, custom: String::new() }), Self::ref_props(kind)),
456                    None => (None, Default::default()),
457                },
458                'B' => (self.by_rtf.get(&n).map(|&id| InlineObject::CommentStart { id }), Default::default()),
459                'E' => (self.by_rtf.get(&n).map(|&id| InlineObject::CommentEnd { id }), Default::default()),
460                _ => (None, Default::default()),
461            };
462            if p.delete(a, end).is_err() {
463                return;
464            }
465            if let Some(o) = obj
466                && p.insert_object(a, o, &props).is_err()
467            {
468                return;
469            }
470        }
471    }
