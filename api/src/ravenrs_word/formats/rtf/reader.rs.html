120    /// Set the output aside (a header or footer starts) or put it back (it ended), returning the
121    /// other one.
122    fn swap_output(&mut self, mut o: Output) -> Output {
123        std::mem::swap(&mut o.body, &mut self.body);
124        std::mem::swap(&mut o.para, &mut self.para);
125        std::mem::swap(&mut o.para_has, &mut self.para_has);
126        std::mem::swap(&mut o.table, &mut self.table);
127        std::mem::swap(&mut o.row, &mut self.row);
128        std::mem::swap(&mut o.cell, &mut self.cell);
129        std::mem::swap(&mut o.defs, &mut self.defs);
130        std::mem::swap(&mut o.def_cur, &mut self.def_cur);
131        std::mem::swap(&mut o.row_header, &mut self.row_header);
132        std::mem::swap(&mut o.rule_pending, &mut self.rule_pending);
133        std::mem::swap(&mut o.table_rights, &mut self.table_rights);
134        o
135    }

159    /// The header or footer group closed: keep what it held and go back to the body.
160    pub(super) fn end_header_footer(&mut self) {
161        let Some((_, which, body)) = self.hf.take() else { return };
162        if self.para_has || !self.para.inlines.is_empty() {
163            self.end_para(false);
164        }
165        self.close_table();
166        let part = self.swap_output(*body);
167        // A section names each one once; a repeat replaces it.
168        self.hf_parts.retain(|(w, _)| *w != which);
169        if self.hf_parts.len() < 16 {
170            self.hf_parts.push((which, part.body));
171        }
172    }

174    /// A footnote, endnote or comment group starts: read it into an output of its own.
175    pub(super) fn start_sub(&mut self, comment: bool) {
176        if self.overflow > 0 || self.notes.len() >= MAX_NOTES || self.comments.len() >= MAX_NOTES {
177            self.st.dest = Dest::Skip;
178            return;
179        }
180        let saved = self.swap_output(Output::default());
181        self.sub = Some(Sub { depth: self.depth(), comment, endnote: false, saved: Box::new(saved) });
182        self.st.intbl = false;
183    }

191    /// The footnote or comment group closed: keep its text and leave a marker in the body.
192    pub(super) fn end_sub(&mut self) {
193        let Some(sub) = self.sub.take() else { return };
194        if self.para_has || !self.para.inlines.is_empty() {
195            self.end_para(false);
196        }
197        self.close_table();
198        let fmt = self.stack.last().map(|s| s.fmt.clone()).unwrap_or_default();
199        let mut inner = self.swap_output(*sub.saved);
200        inner.body.truncate(MAX_NOTE_BLOCKS);
201        if !sub.comment {
202            self.push_mark('N', self.notes.len() as i64, fmt);
203            self.notes.push(NoteData { endnote: sub.endnote, blocks: inner.body });
204            return;
205        }
206        let want = self.atn_ref.take().or(self.last_end.take());
207        let id = match want {
208            Some(i) if self.range_starts.contains(&i) && self.range_ends.contains(&i) && self.claimed.insert(i) => i,
209            _ => {
210                // No range of its own: the comment sits at this point.
211                self.synth = self.synth.saturating_sub(1);
212                let id = self.synth;
213                self.push_mark('B', i64::from(id), fmt.clone());
214                self.push_mark('E', i64::from(id), fmt);
215                id
216            }
217        };
218        let date =
219            if !self.atn_date.is_empty() { std::mem::take(&mut self.atn_date) } else { self.atn_parts.take().map(iso_date).unwrap_or_default() };
220        self.atn_parts = None;
221        self.comments.push(CommentData {
222            rtf_id: id,
223            author: std::mem::take(&mut self.atn_author),
224            initials: std::mem::take(&mut self.atn_initials),
225            date,
226            blocks: inner.body,
227        });
228    }

230    /// A comment word's text group closed (`\atrfstart 3`, `\atnauthor Ann`…).
231    pub(super) fn commit_atn(&mut self, kind: u8) {
232        let text = std::mem::take(&mut self.atn_text);
233        let text = text.trim();
234        let id = text.parse::<i32>().ok();
235        let fmt = self.stack.last().map(|s| s.fmt.clone()).unwrap_or_default();
236        match kind {
237            1 => {
238                if let Some(id) = id {
239                    self.range_starts.insert(id);
240                    self.push_mark('B', i64::from(id), fmt);
241                }
242            }
243            2 => {
244                if let Some(id) = id {
245                    self.range_ends.insert(id);
246                    self.last_end = Some(id);
247                    self.push_mark('E', i64::from(id), fmt);
248                }
249            }
250            3 => self.atn_initials = text.to_string(),
251            4 => self.atn_author = text.to_string(),
252            5 => {
253                // Either `\yr\mo\dy\hr\min` words (read as they came) or a packed date-time number.
254                if let Some(parts) = self.atn_parts.take() {
255                    self.atn_date = iso_date(parts);
256                } else if let Some(n) = text.parse::<u32>().ok().filter(|n| *n != 0) {
257                    self.atn_date = iso_date((1900 + ((n >> 20) & 0x1FF), (n >> 16) & 0xF, (n >> 11) & 0x1F, (n >> 6) & 0x1F, n & 0x3F));
258                }
259            }
260            6 => self.atn_ref = id,
261            _ => {}
262        }
263    }

265    /// `\sect`: end the paragraph, and with it the section, which takes its formatting and the
266    /// headers and footers read since the last break. Inside a header, footer or table, and past
267    /// [`MAX_SECTIONS`], it only ends the paragraph.
268    pub(super) fn end_section(&mut self) {
269        self.end_para_mark();
270        if self.hf.is_some() || self.sub.is_some() || self.st.intbl || self.section_parts.len() + 1 >= MAX_SECTIONS {
271            return;
272        }
273        let props = self.sec.props(&self.page);
274        if let Some(FBlock::Para(p)) = self.body.last_mut() {
275            p.section = Some(Box::new(props));
276            self.section_parts.push(std::mem::take(&mut self.hf_parts));
277        }
278    }

280    /// A section formatting word: applied to the current section, unless it's inside a header
281    /// or footer (whose `\sectd` isn't the body's). False when `w` isn't one.
282    pub(super) fn section_word(&mut self, w: &str, p: Option<i32>) -> bool {
283        use crate::doc::section::{NumFormat, SectionStart};
284        let on = p != Some(0);
285        let pv = p.unwrap_or(0);
286        // Inside a header or footer the word is read into a scratch copy and dropped.
287        let mut scratch;
288        let sec = if self.hf.is_some() || self.sub.is_some() {
289            scratch = self.sec.clone();
290            &mut scratch
291        } else {
292            &mut self.sec
293        };
294        match w {
295            "sectd" => *sec = SecFmt::default(),
296            "titlepg" => sec.title_page = on,
297            "pgwsxn" => sec.page.w = Some(pv),
298            "pghsxn" => sec.page.h = Some(pv),
299            "marglsxn" => sec.page.left = Some(pv),
300            "margrsxn" => sec.page.right = Some(pv),
301            "margtsxn" => sec.page.top = Some(pv),
302            "margbsxn" => sec.page.bottom = Some(pv),
303            "headery" => sec.page.header = Some(pv),
304            "footery" => sec.page.footer = Some(pv),
305            "lndscpsxn" => sec.page.landscape = on,
306            "sbknone" => sec.start = SectionStart::Continuous,
307            "sbkcol" => sec.start = SectionStart::NextColumn,
308            "sbkpage" => sec.start = SectionStart::NextPage,
309            "sbkeven" => sec.start = SectionStart::EvenPage,
310            "sbkodd" => sec.start = SectionStart::OddPage,
311            "cols" => sec.cols = Some(pv),
312            "colsx" => sec.colsx = Some(pv),
313            "linebetcol" => sec.linebetcol = on,
314            "pgnrestart" => sec.pgn_restart = true,
315            "pgncont" => sec.pgn_restart = false,
316            "pgnstarts" => sec.pgn_start = Some(pv),
317            "pgndec" => sec.pgn_fmt = NumFormat::Decimal,
318            "pgnucrm" => sec.pgn_fmt = NumFormat::UpperRoman,
319            "pgnlcrm" => sec.pgn_fmt = NumFormat::LowerRoman,
320            "pgnucltr" => sec.pgn_fmt = NumFormat::UpperLetter,
321            "pgnlcltr" => sec.pgn_fmt = NumFormat::LowerLetter,
322            _ => return false,
323        }
324        true
325    }

327    pub(super) fn open(&mut self) {
328        self.skip_chars = 0;
329        if self.stack.len() >= MAX_GROUPS {
330            self.overflow += 1;
331            return;
332        }
333        self.stack.push(self.st.clone());
334    }

336    pub(super) fn close(&mut self) {
337        // A group end decodes the multibyte bytes read in it, before the state is restored.
338        self.finish_multibyte();
339        self.skip_chars = 0;
340        if self.overflow > 0 {
341            self.overflow -= 1;
342            return;
343        }
344        if self.sub.as_ref().is_some_and(|s| s.depth == self.depth()) {
345            self.end_sub();
346        }
347        let closing = self.st.dest;
348        // Destination-level commits.
349        match closing {
350            Dest::FontTbl if !self.font_name.is_empty() => self.commit_font(),
351            Dest::StyleSheet if !self.style_name.is_empty() => self.commit_style(),
352            Dest::RevTbl if !self.rev_name.trim().is_empty() => self.commit_author(),
353            Dest::Info(k) => {
354                let t = std::mem::take(&mut self.info_text).trim().to_string();
355                match k {
356                    1 => self.meta.title = t,
357                    2 => self.meta.author = t,
358                    3 => self.meta.subject = t,
359                    4 => self.meta.keywords = t,
360                    5 => self.meta.description = t,
361                    _ => {}
362                }
363                self.in_creatim = false;
364            }
365            Dest::Atn(k) => self.commit_atn(k),
366            _ => {}
367        }
368        let Some(mut prev) = self.stack.pop() else { return };
369        // The end of old formatting read in the body: the formatting it holds is the change's
370        // old formatting (nested old-formatting groups only change their parent's copy).
371        if let Dest::OldProps(para) = closing
372            && prev.dest == Dest::Body
373        {
374            if para {
375                prev.chg_p.old = Some(Box::new(self.para_props(&self.st)));
376            } else {
377                prev.chg_c.old = Some(Box::new(std::mem::take(&mut self.st.old_words)));
378                if let Some(a) = self.st.old_rev.0 {
379                    prev.chg_c.auth = a;
380                }
381                if let Some(d) = self.st.old_rev.1 {
382                    prev.chg_c.date = d;
383                }
384            }
385        }
386        if self.pict_depth.is_some_and(|d| self.depth() < d) {
387            self.pict_depth = None;
388            self.finish_pict();
389        }
390        if let Some((d, _)) = &self.bkmk
391            && self.depth() < *d
392            && let Some((_, name)) = self.bkmk.take()
393        {
394            let name = name.trim().to_string();
395            if !name.is_empty() {
396                self.para.inlines.push(Inline::Anchor(name));
397                self.para_has = true;
398            }
399        }
400        self.st = prev;
401        if self.hf.as_ref().is_some_and(|(d, _, _)| self.depth() < *d) {
402            self.end_header_footer();
403        }
404    }

429    /// End the current paragraph (`\par`, `\cell`).
430    pub(super) fn end_para(&mut self, in_cell: bool) {
431        let read = std::mem::take(&mut self.para);
432        let props = self.para_props(&self.st);
433        let mut p = Para { inlines: read.inlines, mark_ins: read.mark_ins, mark_del: read.mark_del, ..props };
434        if p.kind == Kind::Normal && p.inlines.is_empty() && self.rule_pending {
435            p.kind = Kind::Rule;
436        }
437        self.rule_pending = false;
438        let c = &self.st.chg_p;
439        if let Some(old) = &c.old {
440            p.fmt_change = Some(Box::new(model::FmtChange { mark: self.rev_mark(c.auth, c.date), old: (**old).clone() }));
441        }
442        // Headings, code, titles and quotes keep their style formatting implicit, in the text's
443        // formatting and in the formatting before a tracked change alike. The style is the
444        // file's when its name gave the kind (not an outline level on another style).
445        let base = if self.style_kind(&self.st) == Some(p.kind) { self.style_base(p.kind, self.st.style) } else { style_fmt(p.kind) };
446        for i in &mut p.inlines {
447            if let Inline::Text(_, f) = i {
448                drop_style_formatting(p.kind, &base, f);
449                if let Some(c) = f.fmt_change.as_deref_mut() {
450                    drop_style_formatting(p.kind, &base, &mut c.old);
451                }
452            }
453        }
454        self.para_has = false;
455        if self.st.intbl || in_cell {
456            self.cell.push(FBlock::Para(p));
457        } else {
458            self.close_table();
459            self.body.push(FBlock::Para(p));
460        }
461    }

471    pub(super) fn end_row(&mut self) {
472        if self.para_has {
473            self.end_cell();
474        }
475        let mut cells = std::mem::take(&mut self.row);
476        let defs = self.defs.clone();
477        let mut out: Vec<Cell> = Vec::new();
478        let mut rights: Vec<i32> = Vec::new();
479        for (i, mut c) in cells.drain(..).enumerate() {
480            let d = defs.get(i).cloned().unwrap_or_default();
481            // A merged-in cell widens the cell it joins.
482            if d.hmerge == 2
483                && let Some(last) = rights.last_mut()
484            {
485                *last = d.right;
486            } else {
487                rights.push(d.right);
488            }
489            c.shading = d.shading;
490            c.borders = d.borders.any_visible().then_some(d.borders);
491            c.header = self.row_header;
492            c.valign = d.valign;
493            match d.vmerge {
494                1 => c.rowspan = 2,
495                2 => c.covered = true,
496                _ => {}
497            }
498            if d.hmerge == 2
499                && let Some(prev) = out.last_mut()
500            {
501                prev.colspan = (prev.colspan + 1).min(63);
502                continue;
503            }
504            out.push(c);
505        }
506        if !out.is_empty() && self.table.len() < crate::doc::table::MAX_ROWS {
507            rights.truncate(out.len());
508            self.table.push(out);
509            self.table_rights.push(rights);
510        }
511    }

513    pub(super) fn close_table(&mut self) {
514        if !self.row.is_empty() || !self.cell.is_empty() {
515            if !self.cell.is_empty() {
516                let blocks = std::mem::take(&mut self.cell);
517                self.row.push(Cell { blocks, ..Default::default() });
518            }
519            self.end_row();
520        }
521        if !self.table.is_empty() {
522            let mut rows = std::mem::take(&mut self.table);
523            let rights = std::mem::take(&mut self.table_rights);
524            let widths = column_grid(&mut rows, &rights);
525            // Word draws only the borders the cells name: no grid style, each cell its own.
526            self.body.push(FBlock::Table(FTable { rows, widths, borderless: true, extra: None, kept: None, wrap: None }));
527        }
528    }
