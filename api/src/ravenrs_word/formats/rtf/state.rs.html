221/// Page size and margins in twips (`\paperw`, `\margl`… and their `…sxn` section forms).
222#[derive(Clone, Copy, Debug, Default, PartialEq)]
223pub(super) struct PageGeom {
224    pub(super) w: Option<i32>,
225    pub(super) h: Option<i32>,
226    pub(super) left: Option<i32>,
227    pub(super) right: Option<i32>,
228    pub(super) top: Option<i32>,
229    pub(super) bottom: Option<i32>,
230    pub(super) header: Option<i32>,
231    pub(super) footer: Option<i32>,
232    pub(super) landscape: bool,
233}
234
235impl PageGeom {
236    /// The section properties these give, Word's defaults where the file names none.
237    fn section(&self) -> crate::doc::SectionProps {
238        let mut s = crate::doc::SectionProps::default();
239        // Twips to points, ignoring nonsense (zero, negative, or beyond Word's 22").
240        let pt = |v: Option<i32>, lo: f32, hi: f32| v.map(|t| t as f32 / 20.0).filter(|v| v.is_finite() && (lo..=hi).contains(v));
241        if let Some(w) = pt(self.w, 72.0, 1584.0) {
242            s.page_w = w;
243        }
244        if let Some(h) = pt(self.h, 72.0, 1584.0) {
245            s.page_h = h;
246        }
247        if self.landscape && s.page_w < s.page_h {
248            std::mem::swap(&mut s.page_w, &mut s.page_h);
249        }
250        s.landscape = s.page_w > s.page_h;
251        for (v, f) in [
252            (self.left, &mut s.margin_left),
253            (self.right, &mut s.margin_right),
254            (self.top, &mut s.margin_top),
255            (self.bottom, &mut s.margin_bottom),
256            (self.header, &mut s.header),
257            (self.footer, &mut s.footer),
258        ] {
259            if let Some(v) = pt(v, 0.0, 720.0) {
260                *f = v;
261            }
262        }
263        // Margins that leave no room for text fall back to Word's.
264        let d = crate::doc::SectionProps::default();
265        if s.page_w - s.margin_left - s.margin_right < 72.0 {
266            (s.margin_left, s.margin_right) = (d.margin_left, d.margin_right);
267        }
268        if s.page_h - s.margin_top - s.margin_bottom < 72.0 {
269            (s.margin_top, s.margin_bottom) = (d.margin_top, d.margin_bottom);
270        }
271        s
272    }
273}

275/// Section formatting (`\sectd` resets it): the section's page words (`\pgwsxn`…, over the
276/// document's `\paperw`…), its break, columns, page numbering and title page.
277#[derive(Clone, Debug, Default)]
278pub(super) struct SecFmt {
279    pub(super) page: PageGeom,
280    pub(super) start: crate::doc::section::SectionStart,
281    pub(super) cols: Option<i32>,
282    pub(super) colsx: Option<i32>,
283    pub(super) linebetcol: bool,
284    pub(super) pgn_restart: bool,
285    pub(super) pgn_start: Option<i32>,
286    pub(super) pgn_fmt: crate::doc::section::NumFormat,
287    pub(super) title_page: bool,
288}
289
290impl SecFmt {
291    /// The section's properties over the document's page (`doc`); headers and footers unset.
292    pub(super) fn props(&self, doc: &PageGeom) -> crate::doc::SectionProps {
293        let p = &self.page;
294        let page = PageGeom {
295            w: p.w.or(doc.w),
296            h: p.h.or(doc.h),
297            left: p.left.or(doc.left),
298            right: p.right.or(doc.right),
299            top: p.top.or(doc.top),
300            bottom: p.bottom.or(doc.bottom),
301            header: p.header.or(doc.header),
302            footer: p.footer.or(doc.footer),
303            // A section that gives its own page size or orientation has its own orientation;
304            // the document's `\landscape` only applies to sections that give neither.
305            landscape: if p.w.is_some() || p.h.is_some() || p.landscape { p.landscape } else { doc.landscape },
306        };
307        let mut s = page.section();
308        s.start = self.start;
309        if let Some(n) = self.cols {
310            s.columns.count = n.clamp(1, 45) as u32;
311        }
312        if let Some(x) = self.colsx {
313            s.columns.space = (x as f32 / 20.0).clamp(0.0, 1584.0);
314        }
315        s.columns.separator = self.linebetcol;
316        if self.pgn_restart {
317            s.page_num_start = Some(self.pgn_start.unwrap_or(1).clamp(0, 32_767) as u32);
318        }
319        s.page_num_format = self.pgn_fmt;
320        s.title_page = self.title_page;
321        s
322    }
