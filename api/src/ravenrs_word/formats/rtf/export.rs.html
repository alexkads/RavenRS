11/// RTF-escape text (non-ASCII as `\uN?`).
12pub fn esc(s: &str) -> String {
13    let mut o = String::with_capacity(s.len() + 8);
14    for c in s.chars() {
15        match c {
16            '\\' => o.push_str("\\\\"),
17            '{' => o.push_str("\\{"),
18            '}' => o.push_str("\\}"),
19            '\t' => o.push_str("\\tab "),
20            '\n' => o.push_str("\\line "),
21            '\u{000C}' => o.push_str("\\page "),
22            '\u{000E}' => o.push_str("\\column "),
23            '\u{00A0}' => o.push_str("\\~"),
24            '\u{00AD}' => o.push_str("\\-"),
25            '\u{2011}' => o.push_str("\\_"),
26            c if c.is_ascii_control() => {}
27            c if c.is_ascii() => o.push(c),
28            c => {
29                let mut buf = [0u16; 2];
30                for u in c.encode_utf16(&mut buf) {
31                    o.push_str(&format!("\\u{}?", *u as i16));
32                }
33            }
34        }
35    }
36    o
37}

39/// [`esc`] for a `\fonttbl` name: `;` would end the entry, so it is written `\'3b`, and control
40/// characters, which `esc` writes as `\line` or `\tab`, are dropped.
41fn esc_font(name: &str) -> String {
42    esc(&name.replace(char::is_control, "")).replace(';', "\\'3b")
43}

110pub(super) fn has_marks(bl: &Blocks, depth: usize) -> bool {
111    bl.iter().any(|b| match &**b {
112        Block::Para(p) => {
113            p.objects.iter().any(|o| matches!(o, InlineObject::NoteRef { .. } | InlineObject::CommentStart { .. } | InlineObject::CommentEnd { .. }))
114        }
115        Block::Table(t) => depth < 8 && t.rows.iter().any(|r| r.cells.iter().any(|c| has_marks(&c.blocks, depth + 1))),
116    })
117}

119/// A copy of `doc` whose note references and comment range ends are place markers in the text
120/// (see [`MARK_OPEN`]), with what each stands for; none when the body has neither.
121fn mark_doc(doc: &Document) -> Option<(Document, Vec<Mark>)> {
122    if !has_marks(&doc.body, 0) {
123        return None;
124    }
125    let mut m = Marker { doc, marks: Vec::new(), annotated: HashSet::new() };
126    let body = m.blocks(&doc.body, None, 0);
127    let mut d = doc.clone();
128    d.body = body;
129    Some((d, m.marks))
130}

132/// The story id standing for "inside a note or comment": no nested notes, no comment ranges.
133const IN_STORY: u32 = u32::MAX;

135struct Marker<'a> {
136    doc: &'a Document,
137    marks: Vec<Mark>,
138    annotated: HashSet<u32>,
139}

141impl Marker<'_> {
142    fn token(&mut self, m: Mark) -> String {
143        self.marks.push(m);
144        self.marks.len().saturating_sub(1).to_string()
145    }

147    /// The flow of a story (a note's or comment's text); tables are flattened to their paragraphs.
148    fn story(&mut self, bl: &Blocks, note: u32, depth: usize, out: &mut Vec<FBlock>) {
149        for b in bl.iter().take(MAX_NOTE_BLOCKS) {
150            match &**b {
151                Block::Para(p) => {
152                    let q = self.para(p, Some(note));
153                    out.extend(model::flow_paras(self.doc, &q).into_iter().map(FBlock::Para));
154                }
155                Block::Table(t) if depth < 4 => {
156                    for c in t.rows.iter().flat_map(|r| r.cells.iter()) {
157                        self.story(&c.blocks, note, depth + 1, out);
158                    }
159                }
160                Block::Table(_) => {}
161            }
162        }
163    }
164
165    fn blocks(&mut self, bl: &Blocks, note: Option<u32>, depth: usize) -> Blocks {
166        bl.iter()
167            .map(|b| match &**b {
168                Block::Para(p) if p.objects.is_empty() => b.clone(),
169                Block::Para(p) => Arc::new(Block::Para(self.para(p, note))),
170                Block::Table(t) if depth < 8 => {
171                    let mut t = t.clone();
172                    for c in t.rows.iter_mut().flat_map(|r| r.cells.iter_mut()) {
173                        c.blocks = self.blocks(&c.blocks, note, depth + 1);
174                    }
175                    Arc::new(Block::Table(t))
176                }
177                Block::Table(_) => b.clone(),
178            })
179            .collect()
180    }
181
182    /// `p` with its note references and comment range ends replaced by place markers. `note` is
183    /// the note being written, when `p` is part of one.
184    fn para(&mut self, p: &Paragraph, note: Option<u32>) -> Paragraph {
185        let mut q = p.clone();
186        for off in p.object_offsets().into_iter().rev() {
187            let Some(obj) = q.object_at(off).cloned() else { continue };
188            let tok = match obj {
189                InlineObject::NoteRef { id, .. } if note == Some(id) => Some(String::from("S")),
190                InlineObject::NoteRef { .. } if note.is_some() => Some(String::new()),
191                InlineObject::NoteRef { id, kind, .. } => self.doc.parts.get(&id).map(|part| {
192                    let mut blocks = Vec::new();
193                    self.story(&part.blocks, id, 0, &mut blocks);
194                    self.token(Mark::Note { endnote: kind == crate::doc::para::NoteKind::Endnote, blocks })
195                }),
196                InlineObject::CommentStart { id } if note.is_none() => Some(self.token(Mark::RangeStart(id))),
197                InlineObject::CommentEnd { id } if note.is_none() => {
198                    let ann = if self.annotated.insert(id) { self.annotation(id) } else { None };
199                    Some(self.token(Mark::RangeEnd { id, ann }))
200                }
201                _ => None,
202            };
203            let Some(tok) = tok else { continue };
204            let end = off + crate::doc::para::OBJ.len_utf8();
205            let text = if tok.is_empty() { tok } else { format!("{MARK_OPEN}{tok}{MARK_CLOSE}") };
206            if q.delete(off, end).is_err() {
207                continue;
208            }
209            if !text.is_empty() && q.insert_text(off, &text, &Default::default()).is_err() {
210                continue;
211            }
212        }
213        q
214    }
215
216    fn annotation(&mut self, id: u32) -> Option<Annotation> {
217        let c = self.doc.comments.get(&id)?;
218        let part = self.doc.parts.get(&c.part)?;
219        let mut blocks = Vec::new();
220        self.story(&part.blocks, IN_STORY, 0, &mut blocks);
221        Some(Annotation { author: c.author.clone(), initials: c.initials.clone(), date: c.date.clone(), blocks })
222    }
223}

225/// `\sectd` and a section's formatting, in the words the reader reads back.
226fn sect_words(s: &crate::doc::SectionProps) -> String {
227    use crate::doc::section::{NumFormat, SectionStart};
228    let tw = |v: f32| if v.is_finite() { ((v * 20.0).round() as i64).max(0) } else { 0 };
229    let mut o = String::from("\\sectd");
230    o.push_str(match s.start {
231        SectionStart::Continuous => "\\sbknone",
232        SectionStart::NextColumn => "\\sbkcol",
233        SectionStart::NextPage => "\\sbkpage",
234        SectionStart::EvenPage => "\\sbkeven",
235        SectionStart::OddPage => "\\sbkodd",
236    });
237    o.push_str(&format!("\\pgwsxn{}\\pghsxn{}", tw(s.page_w), tw(s.page_h)));
238    if s.landscape {
239        o.push_str("\\lndscpsxn");
240    }
241    o.push_str(&format!(
242        "\\marglsxn{}\\margrsxn{}\\margtsxn{}\\margbsxn{}\\headery{}\\footery{}",
243        tw(s.margin_left),
244        tw(s.margin_right),
245        tw(s.margin_top),
246        tw(s.margin_bottom),
247        tw(s.header),
248        tw(s.footer)
249    ));
250    if s.columns.count > 1 {
251        o.push_str(&format!("\\cols{}\\colsx{}", s.columns.count, tw(s.columns.space)));
252        if s.columns.separator {
253            o.push_str("\\linebetcol");
254        }
255    }
256    if let Some(n) = s.page_num_start {
257        o.push_str(&format!("\\pgnrestart\\pgnstarts{n}"));
258    }
259    match s.page_num_format {
260        NumFormat::UpperRoman => o.push_str("\\pgnucrm"),
261        NumFormat::LowerRoman => o.push_str("\\pgnlcrm"),
262        NumFormat::UpperLetter => o.push_str("\\pgnucltr"),
263        NumFormat::LowerLetter => o.push_str("\\pgnlcltr"),
264        _ => {}
265    }
266    if s.title_page {
267        o.push_str("\\titlepg");
268    }
269    o.push('\n');
270    o
271}

376    /// A run of text; place markers in it become notes and comment ranges.
377    fn run(&mut self, t: &str, f: &Fmt, out: &mut String) {
378        if !t.contains(MARK_OPEN) {
379            return self.run_plain(t, f, out);
380        }
381        let mut rest = t;
382        while !rest.is_empty() {
383            let Some((pre, after)) = rest.split_once(MARK_OPEN) else {
384                self.run_plain(rest, f, out);
385                break;
386            };
387            if !pre.is_empty() {
388                self.run_plain(pre, f, out);
389            }
390            let Some((tok, tail)) = after.split_once(MARK_CLOSE) else {
391                self.run_plain(after, f, out);
392                break;
393            };
394            self.mark(tok, out);
395            rest = tail;
396        }
397    }

399    /// The RTF for one place marker: `S` is a note's own number, a number indexes `marks`.
400    fn mark(&mut self, tok: &str, out: &mut String) {
401        if tok == "S" {
402            out.push_str("{\\super\\chftn}");
403            return;
404        }
405        let Some(m) = tok.parse::<usize>().ok().and_then(|i| self.marks.get_mut(i)) else { return };
406        match std::mem::replace(m, Mark::Done) {
407            Mark::Note { endnote, blocks } => {
408                out.push_str("{\\super\\chftn}{\\footnote");
409                if endnote {
410                    out.push_str("\\ftnalt");
411                }
412                out.push(' ');
413                self.blocks(&blocks, false, "\\par", out, 0);
414                out.push('}');
415            }
416            Mark::RangeStart(id) => out.push_str(&format!("{{\\*\\atrfstart {id}}}")),
417            Mark::RangeEnd { id, ann } => {
418                out.push_str(&format!("{{\\*\\atrfend {id}}}"));
419                if let Some(a) = ann {
420                    out.push_str(&format!(
421                        "{{\\*\\atnid {}}}{{\\*\\atnauthor {}}}\\chatn{{\\*\\annotation{{\\*\\atnref {id}}}",
422                        esc(&a.initials),
423                        esc(&a.author)
424                    ));
425                    if let Some(d) = rtf_date(&a.date) {
426                        out.push_str(&format!("{{\\*\\atndate {d}}}"));
427                    }
428                    out.push('\n');
429                    self.blocks(&a.blocks, false, "\\par", out, 0);
430                    out.push('}');
431                }
432            }
433            Mark::Done => {}
434        }
435    }

437    fn run_plain(&mut self, t: &str, f: &Fmt, out: &mut String) {
438        let mut ctl = String::new();
439        if let Some(c) = &f.fmt_change {
440            // RTF 1.9.1, as Word writes it: the properties the change altered, as they were,
441            // then who changed it and when, all in the group. The group goes first so a control
442            // word, not the group, ends the run's properties.
443            let old = self.char_diff(&c.old, f);
444            let words = self.rev_words("\\crauth", "\\crdate", &c.mark);
445            ctl.push_str(&format!("{{\\*\\oldcprops {old}{words}}}"));
446        }
447        ctl.push_str(&self.rev_ctl(f.ins.as_ref(), f.del.as_ref()));
448        ctl.push_str(&self.char_ctl(f));
449        let body = if ctl.is_empty() { esc(t) } else { format!("{{{ctl} {}}}", esc(t)) };
450        // A script, `file:`, UNC or other non-web target is written as plain text, as the HTML and
451        // Markdown exporters do: Word follows a HYPERLINK field's target on click, and a UNC one
452        // leaks the reader's NTLM credentials.
453        match f.link.as_deref().filter(|l| crate::doc::link::is_safe_href(l)) {
454            Some(l) => {
455                // In a field's quotes `\\` is one backslash; a quote is stored as `%22`.
456                let target = l.replace('\\', "\\\\").replace('"', "%22");
457                out.push_str(&format!("{{\\field{{\\*\\fldinst{{HYPERLINK \"{}\"}}}}{{\\fldrslt{{{}}}}}}}", esc(&target), body));
458            }
459            None => out.push_str(&body),
460        }
461    }

463    /// The character formatting control words that turn `cur` back into `old`: only the
464    /// properties that differ, with explicit "off" values (`\b0`, `\ulnone`, `\cf0`…).
465    fn char_diff(&mut self, old: &Fmt, cur: &Fmt) -> String {
466        // A link's underline and colour are the link's, before the change and after; what a run
467        // doesn't set directly is its paragraph style's, as Word writes resolved values.
468        let old = &styled(&self.base, &Fmt { link: cur.link.clone(), ..old.clone() });
469        let cur = &styled(&self.base, cur);
470        let mut ctl = String::new();
471        let underline = |f: &Fmt| f.underline || f.link.is_some();
472        let flags = [
473            (old.bold, cur.bold, "\\b", "\\b0"),
474            (old.italic, cur.italic, "\\i", "\\i0"),
475            (underline(old), underline(cur), "\\ul", "\\ulnone"),
476            (old.strike, cur.strike, "\\strike", "\\strike0"),
477        ];
478        for (o, c, on, off) in flags {
479            if o != c {
480                ctl.push_str(if o { on } else { off });
481            }
482        }
483        if (old.sup, old.sub) != (cur.sup, cur.sub) {
484            ctl.push_str(if old.sup {
485                "\\super"
486            } else if old.sub {
487                "\\sub"
488            } else {
489                "\\nosupersub"
490            });
491        }
492        fn font(f: &Fmt) -> Option<&str> {
493            if f.code { Some(model::MONO_FONT) } else { f.font.as_deref() }
494        }
495        let old_font = font(old);
496        if old_font != font(cur) {
497            // No font is the document's default font, font 0 (Normal's).
498            let i = old_font.map_or(0, |n| self.t.font(n));
499            ctl.push_str(&format!("\\f{i}"));
500        }
501        let half_points = |s: Option<f32>| s.map(|s| (s * 2.0).round().clamp(2.0, 3276.0) as i32);
502        let old_size = half_points(old.size);
503        if old_size != half_points(cur.size) {
504            // No size is Normal's, RTF's default 12 pt.
505            ctl.push_str(&format!("\\fs{}", old_size.unwrap_or(24)));
506        }
507        let color = |f: &Fmt| f.color.or(if f.link.is_some() { Some(Rgb(0x05, 0x63, 0xC1)) } else { None });
508        let old_color = color(old);
509        if old_color != color(cur) {
510            let i = old_color.map_or(0, |c| self.t.color(c));
511            ctl.push_str(&format!("\\cf{i}"));
512        }
513        if old.background != cur.background {
514            let i = old.background.map_or(0, |c| self.t.color(c));
515            ctl.push_str(&format!("\\chcbpat{i}"));
516        }
517        ctl
518    }

520    /// Character formatting control words of `f` (no revision marks).
521    fn char_ctl(&mut self, f: &Fmt) -> String {
522        let mut ctl = String::new();
523        if f.bold {
524            ctl.push_str("\\b");
525        }
526        if f.italic {
527            ctl.push_str("\\i");
528        }
529        if f.underline || f.link.is_some() {
530            ctl.push_str("\\ul");
531        }
532        if f.strike {
533            ctl.push_str("\\strike");
534        }
535        if f.sup {
536            ctl.push_str("\\super");
537        } else if f.sub {
538            ctl.push_str("\\sub");
539        }
540        if f.code {
541            let i = self.t.font(model::MONO_FONT);
542            ctl.push_str(&format!("\\f{i}"));
543        } else if let Some(fam) = &f.font {
544            let i = self.t.font(fam);
545            ctl.push_str(&format!("\\f{i}"));
546        }
547        if let Some(s) = f.size {
548            ctl.push_str(&format!("\\fs{}", (s * 2.0).round().clamp(2.0, 3276.0) as i32));
549        }
550        let color = f.color.or(if f.link.is_some() { Some(Rgb(0x05, 0x63, 0xC1)) } else { None });
551        if let Some(c) = color {
552            let i = self.t.color(c);
553            ctl.push_str(&format!("\\cf{i}"));
554        }
555        if let Some(c) = f.background {
556            let i = self.t.color(c);
557            ctl.push_str(&format!("\\chcbpat{i}"));
558        }
559        ctl
560    }

562    fn picture(&mut self, img: &model::Img, out: &mut String) {
563        // A metafile's `\picw`/`\pich` are its size in 0.01 mm (#4259).
564        let hundredths = ((img.w * 2540.0 / 72.0).round() as i64, (img.h * 2540.0 / 72.0).round() as i64);
565        let (data, blip) = match img.ext.as_str() {
566            "png" => (img.data.to_vec(), "\\pngblip"),
567            "jpeg" | "jpg" => (img.data.to_vec(), "\\jpegblip"),
568            // The bytes decide, as on reading: an EMF starts with its header record, type 1 (a
569            // DOCX may store one under `.wmf`).
570            "emf" | "wmf" if img.data.starts_with(&[1, 0, 0, 0]) => (img.data.to_vec(), "\\emfblip"),
571            // RTF holds a WMF without its 22-byte placeable header.
572            "emf" | "wmf" => {
573                let data = if img.data.starts_with(&[0xD7, 0xCD, 0xC6, 0x9A]) { img.data.get(22..).unwrap_or_default() } else { img.data.as_slice() };
574                (data.to_vec(), "\\wmetafile8")
575            }
576            _ => {
577                // Re-encode other formats as PNG, bounding the decode so a tiny file declaring
578                // huge dimensions can't exhaust memory on export.
579                let Some(dec) = model::decode_bounded(&img.data) else { return };
580                let mut buf = Vec::new();
581                if dec.write_to(&mut std::io::Cursor::new(&mut buf), image::ImageFormat::Png).is_err() {
582                    return;
583                }
584                (buf, "\\pngblip")
585            }
586        };
587        // A placeable header with nothing after it leaves no picture to write.
588        if data.is_empty() {
589            return;
590        }
591        let (pw, ph) = match blip {
592            "\\emfblip" | "\\wmetafile8" => hundredths,
593            _ => model::image_px(&data).map(|(a, b)| (i64::from(a), i64::from(b))).unwrap_or((1, 1)),
594        };
595        out.push_str(&format!(
596            "{{\\pict{blip}\\picw{pw}\\pich{ph}\\picwgoal{}\\pichgoal{}\n",
597            (img.w * 20.0).round() as i64,
598            (img.h * 20.0).round() as i64
599        ));
600        for chunk in data.chunks(64) {
601            for b in chunk {
602                out.push_str(&format!("{b:02x}"));
603            }
604            out.push('\n');
605        }
606        out.push('}');
607    }

609    fn inlines(&mut self, inl: &[Inline], out: &mut String) {
610        let inl = model::equations_as_text(inl);
611        for i in inl.iter() {
612            match i {
613                Inline::Text(t, f) => self.run(t, f, out),
614                Inline::Image(img) => self.picture(img, out),
615                Inline::Anchor(a) => {
616                    let a = esc(a);
617                    out.push_str(&format!("{{\\*\\bkmkstart {a}}}{{\\*\\bkmkend {a}}}"));
618                }
619                // RTF keeps no ruby yet: its base text, in the run's formatting.
620                Inline::Ruby { base, fmt, .. } => self.run(base, fmt, out),
621                // ASK and SET show nothing (see `model::shows_nothing`).
622                Inline::Field { instr, .. } if model::shows_nothing(instr) => {}
623                Inline::Field { result, fmt, .. } | Inline::Control { text: result, fmt, .. } => self.run(result, fmt, out),
624                Inline::Figure(_)
625                | Inline::Chart(_)
626                | Inline::Equation { .. }
627                | Inline::Note { .. }
628                | Inline::Comment { .. }
629                | Inline::CommentEnd(_)
630                | Inline::Shape { .. }
631                | Inline::Group { .. }
632                | Inline::FieldStart(_)
633                | Inline::FieldEnd
634                | Inline::AnchorEnd(_) => {}
635            }
636        }
637    }

639    pub(super) fn para(&mut self, p: &Para, intbl: bool, ls: Option<(usize, String)>, end: &str, out: &mut String) {
640        let ctl = self.para_ctl(p, intbl);
641        out.push_str(&ctl);
642        if let Some(c) = &p.fmt_change {
643            // RTF 1.9.1: the paragraph formatting before the change, then who changed it and when.
644            let old = self.para_ctl(&c.old, intbl);
645            // The old list is the same list instance (only its level may differ).
646            let list = match (c.old.list, &ls) {
647                (Some(li), Some((n, _))) => format!("\\ls{n}\\ilvl{}", li.level.min(8)),
648                _ => String::new(),
649            };
650            out.push_str(&format!("{{\\*\\oldpprops {old}{list}}}"));
651            let words = self.rev_words("\\prauth", "\\prdate", &c.mark);
652            out.push_str(&words);
653        }
654        if p.kind != Kind::Rule
655            && let Some(b) = &p.borders
656        {
657            for (side, opt) in [('t', &b.top), ('l', &b.left), ('b', &b.bottom), ('r', &b.right)] {
658                if let Some(border) = opt.filter(|x| x.is_visible()) {
659                    self.write_border("brdr", side, border, out);
660                }
661            }
662        }
663        if let (Some(li), Some((n, label))) = (p.list, &ls) {
664            let lv = li.level.min(8);
665            out.push_str(&format!("\\ls{n}\\ilvl{lv}\\fi-360\\li{}", 720 * (u32::from(lv) + 1)));
666            // The closing brace ends the label; a space after it would be text.
667            out.push_str(&format!(" {{\\listtext\\pard\\plain {}\\tab}}", esc(label)));
668        } else {
669            out.push(' ');
670        }
671        self.base = style_fmt(p.kind);
672        self.inlines(&p.inlines, out);
673        self.base = Fmt::default();
674        // A tracked paragraph mark is the `\par` inside the revision's group.
675        let marks = self.rev_ctl(p.mark_ins.as_deref(), p.mark_del.as_deref());
676        if end == "\\par" && !marks.is_empty() {
677            out.push_str(&format!("{{{marks} {end}}}"));
678        } else {
679            out.push_str(end);
680        }
681        out.push('\n');
682    }

684    /// The paragraph formatting control words of `p`, from `\pard\plain` (its list left out).
685    fn para_ctl(&mut self, p: &Para, intbl: bool) -> String {
686        let mut out = String::from("\\pard\\plain");
687        if intbl {
688            out.push_str("\\intbl");
689        }
690        if let Some(s) = style_num(p.kind) {
691            out.push_str(&format!("\\s{s}"));
692        }
693        if let Kind::Heading(n) = p.kind {
694            out.push_str(&format!("\\outlinelevel{}", n.clamp(1, 6) - 1));
695        }
696        match p.align {
697            Some(Align::Center) => out.push_str("\\qc"),
698            Some(Align::Right) => out.push_str("\\qr"),
699            Some(Align::Justify) => out.push_str("\\qj"),
700            Some(Align::Distribute) => out.push_str("\\qd"),
701            _ => out.push_str("\\ql"),
702        }
703        if p.page_break {
704            out.push_str("\\pagebb");
705        }
706        match p.kind {
707            Kind::Quote => out.push_str("\\li720\\ri720\\i"),
708            Kind::Code => {
709                let f = self.t.font(model::MONO_FONT);
710                out.push_str(&format!("\\sa0\\f{f}\\fs20"));
711            }
712            Kind::Rule => out.push_str("\\brdrb\\brdrs\\brdrw15\\brsp20"),
713            Kind::Heading(n) => {
714                let sz = HEAD_SIZES.get(usize::from(n.clamp(1, 6)) - 1).copied().unwrap_or(12.0);
715                out.push_str(&format!("\\sb240\\sa80\\keepn\\b\\fs{}", (sz * 2.0) as i32));
716            }
717            Kind::Title => out.push_str("\\sa80\\fs56"),
718            Kind::Normal => {}
719        }
720        out
721    }

723    fn blocks(&mut self, blocks: &[FBlock], intbl: bool, last_end: &str, out: &mut String, depth: usize) {
724        let mut counter = model::ListCounter::default();
725        let mut cur_list: Option<usize> = None;
726        let mut top_ordered = false;
727        let n = blocks.len();
728        for (i, b) in blocks.iter().enumerate() {
729            let mut end = if i + 1 == n { last_end.to_string() } else { "\\par".to_string() };
730            match b {
731                FBlock::Para(p) => {
732                    // A paragraph that ends a section ends it with `\sect`; the next one begins.
733                    if p.section.is_some() && !intbl && depth == 0 && self.sect_i + 1 < self.sections.len() {
734                        self.sect_i += 1;
735                        end = format!("\\sect\n{}", self.sections.get(self.sect_i).map(sect_words).unwrap_or_default());
736                        // A body that ends on the break keeps its last section: an empty paragraph holds it.
737                        if i + 1 == n {
738                            end.push_str("\\pard\\plain \\par");
739                        }
740                    }
741                    let ls = match p.list {
742                        Some(li) => {
743                            // A top-level item of the other kind starts a new list.
744                            if li.level == 0 && cur_list.is_some() && top_ordered != li.ordered {
745                                cur_list = None;
746                            }
747                            if li.level == 0 {
748                                top_ordered = li.ordered;
749                            }
750                            let id = match cur_list {
751                                Some(id) => id,
752                                None => {
753                                    // Level kinds from the whole run of list paragraphs.
754                                    let mut kinds = [li.ordered; 9];
755                                    let mut seen = [false; 9];
756                                    for q in blocks.iter().skip(i) {
757                                        let FBlock::Para(q) = q else { break };
758                                        let Some(ql) = q.list else { break };
759                                        let l = ql.level.min(8) as usize;
760                                        if l == 0 && seen[0] && kinds[0] != ql.ordered {
761                                            break;
762                                        }
763                                        if let (Some(s), Some(k)) = (seen.get_mut(l), kinds.get_mut(l))
764                                            && !*s
765                                        {
766                                            *s = true;
767                                            *k = ql.ordered;
768                                        }
769                                    }
770                                    self.lists.push(kinds);
771                                    counter.reset();
772                                    self.lists.len()
773                                }
774                            };
775                            cur_list = Some(id);
776                            Some((id, counter.label(li)))
777                        }
778                        None => {
779                            cur_list = None;
780                            None
781                        }
782                    };
783                    self.para(p, intbl, ls, &end, out);
784                }
785                FBlock::Table(t) => {
786                    cur_list = None;
787                    if intbl || depth > 0 {
788                        // Nested table: its cells' paragraphs flow into this cell.
789                        let inner: Vec<FBlock> = t.rows.iter().flatten().flat_map(|c| c.blocks.clone()).collect();
790                        if depth < model::MAX_DEPTH {
791                            self.blocks(&inner, intbl, &end, out, depth + 1);
792                        }
793                    } else {
794                        self.table(t, out, depth);
795                    }
796                }
797            }
798        }
799        if blocks.is_empty() && !last_end.is_empty() {
800            out.push_str(&format!("\\pard\\plain{} {last_end}\n", if intbl { "\\intbl" } else { "" }));
801        }
802    }

804    fn table(&mut self, t: &FTable, out: &mut String, depth: usize) {
805        let cols = t.cols().max(1);
806        let total = 9360.0f32; // 6.5" in twips
807        let widths: Vec<f32> = if t.widths.len() == cols && t.widths.iter().all(|w| w.is_finite() && *w > 0.0) {
808            t.widths.iter().map(|w| w * 20.0).collect()
809        } else {
810            vec![total / cols as f32; cols]
811        };
812        for row in &t.rows {
813            out.push_str("\\trowd\\trgaph108\\trleft0");
814            if !row.is_empty() && row.iter().all(|c| c.header) {
815                out.push_str("\\trhdr");
816            }
817            let mut g = 0usize;
818            let mut x = 0.0f32;
819            for c in row {
820                let span = c.colspan.clamp(1, 63) as usize;
821                for k in g..(g + span).min(widths.len()) {
822                    x += widths.get(k).copied().unwrap_or(0.0);
823                }
824                g += span;
825                if c.covered {
826                    out.push_str("\\clvmrg");
827                } else if c.rowspan > 1 {
828                    out.push_str("\\clvmgf");
829                }
830                match c.valign {
831                    Some(VAlign::Top) => out.push_str("\\clvertalt"),
832                    Some(VAlign::Center) => out.push_str("\\clvertalc"),
833                    Some(VAlign::Bottom) => out.push_str("\\clvertalb"),
834                    None => {}
835                }
836                self.cell_borders(c.borders.as_ref(), t.borderless, out);
837                if let Some(s) = c.shading {
838                    let i = self.t.color(s);
839                    out.push_str(&format!("\\clcbpat{i}"));
840                }
841                out.push_str(&format!("\\cellx{}", x.round() as i64));
842            }
843            out.push('\n');
844            for c in row {
845                if c.covered {
846                    out.push_str("\\pard\\plain\\intbl \\cell\n");
847                } else {
848                    let blocks: Vec<FBlock> = c
849                        .blocks
850                        .iter()
851                        .cloned()
852                        .map(|b| match b {
853                            FBlock::Para(mut p) if c.header => {
854                                for i in &mut p.inlines {
855                                    if let Inline::Text(_, f) = i {
856                                        f.bold = true;
857                                    }
858                                }
859                                FBlock::Para(p)
860                            }
861                            other => other,
862                        })
863                        .collect();
864                    self.blocks(&blocks, true, "\\cell", out, depth);
865                }
866            }
867            out.push_str("\\row\n");
868        }
869    }
870}

872fn level_def(ordered: bool, l: usize) -> String {
873    let indent = 720 * (l + 1);
874    if ordered {
875        let nfc = [0, 4, 2].get(l % 3).copied().unwrap_or(0);
876        format!(
877            "{{\\listlevel\\levelnfc{nfc}\\levelnfcn{nfc}\\leveljc0\\leveljcn0\\levelfollow0\\levelstartat1\\levelspace0\\levelindent0{{\\leveltext\\'02\\'{l:02x}.;}}{{\\levelnumbers\\'01;}}\\fi-360\\li{indent}\\lin{indent}}}"
878        )
879    } else {
880        let ch = ['\u{2022}', '\u{25E6}', '\u{25AA}'].get(l % 3).copied().unwrap_or('\u{2022}');
881        format!(
882            "{{\\listlevel\\levelnfc23\\levelnfcn23\\leveljc0\\leveljcn0\\levelfollow0\\levelstartat1\\levelspace0\\levelindent0{{\\leveltext\\'01\\u{}?;}}{{\\levelnumbers;}}\\fi-360\\li{indent}\\lin{indent}}}",
883            ch as u32 as i16
884        )
885    }

888/// RTF text of a document.
889pub fn export(doc: &Document) -> String {
890    let (marked, marks) = match mark_doc(doc) {
891        Some((d, m)) => (Some(d), m),
892        None => (None, Vec::new()),
893    };
894    let doc = marked.as_ref().unwrap_or(doc);
895    let flow = model::from_doc_tracked(doc);
896    let base_font = doc.styles.default_chr.font.clone().unwrap_or_else(|| crate::doc::styles::BODY_FONT.to_string());
897    let mut w = Writer {
898        marks,
899        t: Tables { fonts: vec![base_font], colors: Vec::new() },
900        lists: Vec::new(),
901        authors: None,
902        base: Fmt::default(),
903        sections: Vec::new(),
904        sect_i: 0,
905    };
906    let mut body = String::new();
907    // Every section starts with its own `\sectd`; `\sect` breaks only where there are several.
908    let secs: Vec<crate::doc::SectionProps> = doc.sections().into_iter().map(|(_, s)| s.clone()).collect();
909    if let Some(first) = secs.first() {
910        body.push_str(&sect_words(first));
911    }
912    if secs.len() > 1 {
913        w.sections = secs;
914    }
915    w.blocks(&flow.blocks, false, "\\par", &mut body, 0);
916    export_with(&flow.meta, &w, &body, &doc.last_section)
917}

919fn export_with(m: &Meta, w: &Writer, body: &str, sec: &crate::doc::SectionProps) -> String {
920    let mut out = String::from("{\\rtf1\\ansi\\ansicpg1252\\deff0\\deflang1033\\uc1\n{\\fonttbl");
921    for (i, f) in w.t.fonts.iter().enumerate() {
922        let fam = if model::is_mono(f) { "\\fmodern" } else { "\\fswiss" };
923        out.push_str(&format!("{{\\f{i}{fam}\\fcharset0 {};}}", esc_font(f)));
924    }
925    out.push_str("}\n{\\colortbl;");
926    for c in &w.t.colors {
927        out.push_str(&format!("\\red{}\\green{}\\blue{};", c.0, c.1, c.2));
928    }
929    out.push_str("}\n{\\*\\generator WordCraft;}\n{\\stylesheet{\\s0\\snext0 Normal;}");
930    for n in 1..=6u32 {
931        let sz = HEAD_SIZES.get(n as usize - 1).copied().unwrap_or(12.0);
932        out.push_str(&format!("{{\\s{n}\\sbasedon0\\snext0\\outlinelevel{}\\keepn\\b\\fs{} heading {n};}}", n - 1, (sz * 2.0) as i32));
933    }
934    out.push_str(
935        "{\\s7\\sbasedon0\\snext0\\fs56 Title;}{\\s8\\sbasedon0\\snext0\\li720\\ri720\\i Quote;}{\\s9\\sbasedon0\\snext9\\sa0\\fs20 Code;}}\n",
936    );
937    if !w.lists.is_empty() {
938        out.push_str("{\\*\\listtable");
939        for (i, kinds) in w.lists.iter().enumerate() {
940            let id = i + 1;
941            out.push_str(&format!("\n{{\\list\\listtemplateid{id}\\listhybrid"));
942            for (l, k) in kinds.iter().enumerate() {
943                out.push_str(&level_def(*k, l));
944            }
945            out.push_str(&format!("{{\\listname ;}}\\listid{id}}}"));
946        }
947        out.push_str("}\n{\\*\\listoverridetable");
948        for i in 1..=w.lists.len() {
949            out.push_str(&format!("{{\\listoverride\\listid{i}\\listoverridecount0\\ls{i}}}"));
950        }
951        out.push_str("}\n");
952    }
953    if let Some(authors) = &w.authors {
954        out.push_str("{\\*\\revtbl {Unknown;}");
955        for a in authors {
956            out.push_str(&format!("{{{};}}", esc(&a.replace(';', ","))));
957        }
958        out.push_str("}\n");
959    }
960    out.push_str("{\\info");
961    for (k, v) in [("title", &m.title), ("author", &m.author), ("subject", &m.subject), ("keywords", &m.keywords), ("doccomm", &m.description)] {
962        if !v.is_empty() {
963            out.push_str(&format!("{{\\{k} {}}}", esc(v)));
964        }
965    }
966    if let Some(d) = rtf_date(&m.created) {
967        out.push_str(&format!("{{\\creatim{d}}}"));
968    }
969    out.push_str("}\n");
970    let tw = |v: f32| (v * 20.0).round() as i64;
971    let landscape = if sec.landscape { "\\landscape" } else { "" };
972    out.push_str(&format!(
973        "\\paperw{}\\paperh{}\\margl{}\\margr{}\\margt{}\\margb{}{landscape}\\viewkind4\n",
974        tw(sec.page_w),
975        tw(sec.page_h),
976        tw(sec.margin_left),
977        tw(sec.margin_right),
978        tw(sec.margin_top),
979        tw(sec.margin_bottom)
980    ));
981    out.push_str(body);
982    out.push_str("}\n");
983    out
