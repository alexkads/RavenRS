1/// An ISO 8601 date-time from year, month, day, hour and minute.
2pub(super) fn iso_date((y, mo, d, h, mi): (u32, u32, u32, u32, u32)) -> String {
3    format!("{y:04}-{:02}-{:02}T{:02}:{:02}:00Z", mo.clamp(1, 12), d.clamp(1, 31), h.min(23), mi.min(59))
4}

6pub(super) fn rtf_date(s: &str) -> Option<String> {
7    let num = |a: usize, b: usize| s.get(a..b).and_then(|x| x.parse::<u32>().ok());
8    let (y, mo, d) = (num(0, 4)?, num(5, 7)?, num(8, 10)?);
9    let (h, mi) = (num(11, 13).unwrap_or(0), num(14, 16).unwrap_or(0));
10    Some(format!("\\yr{y}\\mo{mo}\\dy{d}\\hr{h}\\min{mi}"))
11}

13fn days_in_month(y: u32, m: u32) -> u32 {
14    match m {
15        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
16        4 | 6 | 9 | 11 => 30,
17        2 if (y.is_multiple_of(4) && !y.is_multiple_of(100)) || y.is_multiple_of(400) => 29,
18        2 => 28,
19        _ => 0,
20    }
21}

23/// Day of the week, 0 = Sunday (Sakamoto's method).
24fn weekday(y: u32, m: u32, d: u32) -> u32 {
25    const T: [u32; 12] = [0, 3, 2, 5, 0, 3, 5, 1, 4, 6, 2, 4];
26    let y = if m < 3 { y.saturating_sub(1) } else { y };
27    (y + y / 4 - y / 100 + y / 400 + T.get(m.saturating_sub(1) as usize).copied().unwrap_or(0) + d) % 7
28}

30/// An RTF revision date (DTTM, RTF 1.9.1 "Revision Marks": minute in bits 0–5, hour 6–10, day
31/// 11–15, month 16–19, year − 1900 20–28, weekday 29–31) as ISO 8601 (`2024-03-15T10:30:00Z`).
32/// Zero or an impossible date is no date. DTTM has no time zone or seconds: times are read as
33/// UTC (`Z`).
34pub fn dttm_to_iso(v: u32) -> Option<String> {
35    let (mi, h, d, mo, y) = (v & 0x3F, (v >> 6) & 0x1F, (v >> 11) & 0x1F, (v >> 16) & 0xF, 1900 + ((v >> 20) & 0x1FF));
36    if v == 0 || mi > 59 || h > 23 || d == 0 || d > days_in_month(y, mo) {
37        return None;
38    }
39    Some(format!("{y:04}-{mo:02}-{d:02}T{h:02}:{mi:02}:00Z"))
40}

42/// An ISO 8601 date (`YYYY-MM-DD`, optionally `THH:MM…`; seconds and zone ignored) as an RTF
43/// DTTM; `None` when it isn't one or falls outside DTTM's years (1900–2411). DTTM has no time
44/// zone or seconds: the clock time is written as given (WordCraft keeps UTC, `Z`), and seconds
45/// and any offset are dropped.
46pub fn iso_to_dttm(s: &str) -> Option<u32> {
47    let num = |a: usize, b: usize| s.get(a..b).filter(|t| t.bytes().all(|c| c.is_ascii_digit())).and_then(|t| t.parse::<u32>().ok());
48    let sep = |i: usize, c: &[&str]| s.get(i..i + 1).is_some_and(|t| c.contains(&t));
49    let (y, mo, d) = (num(0, 4)?, num(5, 7)?, num(8, 10)?);
50    if !sep(4, &["-"]) || !sep(7, &["-"]) {
51        return None;
52    }
53    let (h, mi) = if s.len() > 10 {
54        if !sep(10, &["T", " "]) || !sep(13, &[":"]) {
55            return None;
56        }
57        (num(11, 13)?, num(14, 16)?)
58    } else {
59        (0, 0)
60    };
61    if !(1900..=2411).contains(&y) || mi > 59 || h > 23 || d == 0 || d > days_in_month(y, mo) {
62        return None;
63    }
64    Some(mi | (h << 6) | (d << 11) | (mo << 16) | ((y - 1900) << 20) | (weekday(y, mo, d) << 29))
65}
