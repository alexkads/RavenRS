1//! Drawing shapes (ODF 1.3 §10.3): `draw:rect`, `draw:ellipse`, `draw:line` and
2//! `draw:custom-shape`, whose `draw:enhanced-geometry` (§10.6.2) names its predefined geometry
3//! in `draw:type`. WordCraft's shape kinds both ways; LibreOffice rebuilds a predefined
4//! geometry from its type alone, so only the type is written.

9/// Largest size or offset read, points (22 inches).
10const MAX_LEN: f32 = 1584.0;

12/// Whether `q` is a drawing shape element read as a shape.
13pub(super) fn is_shape(q: &str) -> bool {
14    matches!(
15        q,
16        "draw:rect"
17            | "draw:ellipse"
18            | "draw:circle"
19            | "draw:line"
20            | "draw:custom-shape"
21            | "draw:polygon"
22            | "draw:polyline"
23            | "draw:path"
24            | "draw:regular-polygon"
25            | "draw:connector"
26    )
27}

29/// The kind of shape element `name` (local name); a custom shape's geometry says more.
30pub(super) fn element_kind(name: &str) -> ShapeKind {
31    match name {
32        "ellipse" | "circle" => ShapeKind::Ellipse,
33        "line" => ShapeKind::Line,
34        "connector" => ShapeKind::ElbowConnector,
35        "polygon" | "polyline" | "path" | "regular-polygon" => ShapeKind::Freeform,
36        _ => ShapeKind::Rectangle,
37    }
38}

40/// The kind of a custom shape of `draw:type` `t`: LibreOffice's own type names, and the
41/// DrawingML preset names it writes as `ooxml-…` for shapes from Word documents. A geometry
42/// WordCraft has no shape for is a rectangle.
43/// A preset geometry type WordCraft has no kind for: drawn as a rectangle (see [`kind_of`]), its
44/// geometry kept to be written back.
45pub(super) fn unknown_preset(t: &str) -> bool {
46    let t = t.trim();
47    kind_of(t) == ShapeKind::Rectangle
48        && !matches!(t.strip_prefix("ooxml-").unwrap_or(t), "" | "rectangle" | "rect" | "flowchart-process" | "flowChartProcess" | "non-primitive")
49}

51pub(super) fn kind_of(t: &str) -> ShapeKind {
52    let t = t.trim();
53    match t.strip_prefix("ooxml-").unwrap_or(t) {
54        "round-rectangle" | "roundRect" => ShapeKind::RoundedRectangle,
55        "ellipse" | "flowchart-connector" | "flowChartConnector" => ShapeKind::Ellipse,
56        "flowchart-or" | "flowChartOr" => ShapeKind::FlowchartOr,
57        "isosceles-triangle" | "triangle" => ShapeKind::Triangle,
58        "right-triangle" | "rtTriangle" => ShapeKind::RightTriangle,
59        "diamond" | "flowchart-decision" | "flowChartDecision" => ShapeKind::Diamond,
60        "right-arrow" | "rightArrow" => ShapeKind::Arrow,
61        "left-arrow" | "leftArrow" => ShapeKind::ArrowLeft,
62        "up-arrow" | "upArrow" => ShapeKind::ArrowUp,
63        "down-arrow" | "downArrow" => ShapeKind::ArrowDown,
64        "star5" => ShapeKind::Star,
65        "heart" => ShapeKind::Heart,
66        "line" | "straightConnector1" => ShapeKind::Line,
67        _ => ShapeKind::Rectangle,
68    }
69}

71/// A connector's kind by its `draw:type` (§19.229): `line` straight, `curve` curved, `standard`
72/// (the default) and `lines` with bends.
73pub(super) fn connector_kind(t: Option<&str>) -> ShapeKind {
74    match t.map(str::trim) {
75        Some("line") => ShapeKind::StraightConnector,
76        Some("curve") => ShapeKind::CurvedConnector,
77        _ => ShapeKind::ElbowConnector,
78    }
79}

85        ShapeKind::CurvedConnector => "curve",

90/// WordCraft's connection site of a `kind` shape for ODF glue point `g` (§9.2.19: a shape's
91/// default glue points are 0 top, 1 right, 2 bottom, 3 left; see
92/// [`crate::doc::connector::sites`]). `None` for a glue point of its own (4 on) or a shape
93/// without sides.
94pub(super) fn site_of_glue(kind: ShapeKind, g: u32) -> Option<u32> {
95    let sides: [u32; 4] = match kind {
96        ShapeKind::Ellipse => [0, 6, 4, 2],
97        ShapeKind::Triangle => [0, 5, 3, 1],
98        k if k.is_open() || k == ShapeKind::RightTriangle => return None,
99        _ => [0, 3, 2, 1],
100    };
101    sides.get(g as usize).copied()
102}
103
104/// The ODF glue point of site `site` of a `kind` shape (the inverse of [`site_of_glue`]).
105pub(super) fn glue_of_site(kind: ShapeKind, site: u32) -> Option<u32> {
106    (0..4).find(|g| site_of_glue(kind, *g) == Some(site))
107}

109/// How shape `kind` is written: its element, and a custom shape's `draw:type`.
110pub(super) fn element_of(kind: ShapeKind) -> (&'static str, Option<&'static str>) {
111    let custom = |t| ("draw:custom-shape", Some(t));
112    match kind {
113        ShapeKind::Ellipse => ("draw:ellipse", None),
114        ShapeKind::Line => ("draw:line", None),
115        ShapeKind::StraightConnector | ShapeKind::ElbowConnector | ShapeKind::CurvedConnector => ("draw:connector", None),
116        ShapeKind::RoundedRectangle => custom("round-rectangle"),
117        ShapeKind::FlowchartOr => custom("flowchart-or"),
118        ShapeKind::Triangle => custom("isosceles-triangle"),
119        ShapeKind::RightTriangle => custom("right-triangle"),
120        ShapeKind::Diamond => custom("diamond"),
121        ShapeKind::Arrow => custom("right-arrow"),
122        ShapeKind::ArrowLeft => custom("left-arrow"),
123        ShapeKind::ArrowUp => custom("up-arrow"),
124        ShapeKind::ArrowDown => custom("down-arrow"),
125        ShapeKind::Star => custom("star5"),
126        ShapeKind::Heart => custom("heart"),
127        ShapeKind::Freeform => ("draw:path", None),
128        ShapeKind::Rectangle | ShapeKind::TextBox => ("draw:rect", None),
129    }
130}

132/// A shape's size and offset (`svg:width`, `svg:height`, `svg:x`, `svg:y`), points, and which
133/// way it is drawn (from right to left, from bottom to top). A line's come from its two ends
134/// (`svg:x1`…): its box spans them, and it starts at the first end, so a line drawn leftwards
135/// or upwards is flipped that way (its arrowheads stay at its ends).
136pub(super) fn geometry(kind: ShapeKind, a: &Attrs) -> (f32, f32, Option<f32>, Option<f32>, (bool, bool)) {
137    let len = |k: &str| get(a, k).and_then(length).filter(|v| v.is_finite()).map(|v| v.clamp(-MAX_LEN, MAX_LEN));
138    if kind.is_open() {
139        let [x1, y1, x2, y2] = ["svg:x1", "svg:y1", "svg:x2", "svg:y2"].map(|k| len(k).unwrap_or(0.0));
140        let size = |d: f32| d.abs().min(MAX_LEN);
141        return (size(x2 - x1), size(y2 - y1), Some(x1.min(x2)), Some(y1.min(y2)), (x2 < x1, y2 < y1));
142    }
143    // A circle may be given by its centre and radius (§19.546 `svg:cx`, §19.554 `svg:r`).
144    if get(a, "svg:width").is_none()
145        && let Some(r) = len("svg:r").map(f32::abs)
146    {
147        let (cx, cy) = (len("svg:cx").unwrap_or(r), len("svg:cy").unwrap_or(r));
148        let d = (r * 2.0).clamp(1.0, MAX_LEN);
149        return (d, d, Some(cx - r), Some(cy - r), (false, false));
150    }
151    let size = |k: &str| len(k).map_or(72.0, |v| v.clamp(1.0, MAX_LEN));
152    (size("svg:width"), size("svg:height"), len("svg:x"), len("svg:y"), (false, false))
153}

155/// The paths of a polygon, polyline, path or regular polygon `name` (local) with attributes
156/// `a`, `w` × `h` points: in its view box (§19.570 `svg:viewBox`; none: around its points).
157pub(super) fn freeform(name: &str, a: &Attrs, w: f32, h: f32) -> Option<crate::doc::freeform::Freeform> {
158    use super::path;
159    let vb = path::view_box(get(a, "svg:viewBox"));
160    match name {
161        "polygon" | "polyline" => path::points(get(a, "draw:points")?, name == "polygon", vb),
162        "path" => path::path(get(a, "svg:d")?, vb),
163        "regular-polygon" => {
164            let corners = get(a, "draw:corners").and_then(|c| c.trim().parse::<u32>().ok()).unwrap_or(3);
165            let concave = get(a, "draw:concave").is_some_and(|c| c.trim() == "true");
166            let sharpness = get(a, "draw:sharpness").and_then(|v| v.trim().strip_suffix('%')?.trim().parse::<f32>().ok()).map_or(0.0, |p| p / 100.0);
167            path::regular(corners, concave, sharpness, w, h)
168        }
169        _ => None,
170    }
