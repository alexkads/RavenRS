31impl FmtDelta {
32    /// Whether the properties read so far underline the text.
33    fn fmt_underline_on(&self) -> bool {
34        self.underline == Some(true)
35    }
36
37    fn apply(&self, f: &mut Fmt) {
38        macro_rules! set {
39            ($($k:ident),*) => { $( if let Some(v) = self.$k.clone() { f.$k = v; } )* };
40        }
41        set!(bold, italic, underline, strike, sup, sub, code);
42        // What this turns off stays said on the text, or the style of its paragraph or span would
43        // show through; turned back on, it no longer is.
44        if self.bold.is_some() || self.italic.is_some() || self.underline.is_some() || self.strike.is_some() {
45            use crate::doc::props::Underline;
46            let mut x = f.extra.take().map(|b| *b).unwrap_or_default();
47            if self.strike == Some(false) {
48                x.double_strike = None;
49            }
50            for (state, said) in [(self.bold, &mut x.bold), (self.italic, &mut x.italic), (self.strike, &mut x.strike)] {
51                if let Some(on) = state {
52                    *said = (!on).then_some(false);
53                }
54            }
55            match self.underline {
56                Some(false) => x.underline = Some(Underline::None),
57                Some(true) if x.underline == Some(Underline::None) => x.underline = None,
58                _ => {}
59            }
60            f.extra = (x != crate::doc::props::CharProps::default()).then(|| Box::new(x));
61        }
62        if self.color.is_some() {
63            f.color = self.color;
64        }
65        if self.background.is_some() {
66            f.background = self.background;
67        }
68        if self.size.is_some() {
69            f.size = self.size;
70        }
71        if let Some(p) = self.size_pct {
72            f.size = Some((f.size.unwrap_or(12.0) * p / 100.0).clamp(1.0, 1638.0));
73        }
74        if let Some(fam) = &self.font {
75            f.font = Some(fam.clone());
76        }
77        if self.extra != crate::doc::props::CharProps::default() || self.raise.is_some() {
78            let mut x = f.extra.take().map(|b| *b).unwrap_or_default();
79            x.overlay(&self.extra);
80            if let Some(r) = self.raise {
81                let size = f.size.unwrap_or(12.0);
82                x.position = Some((r / 100.0 * size).clamp(-1584.0, 1584.0)).filter(|p| *p != 0.0);
83            }
84            // An explicit "off" only cancels what a parent style set (capitals are kept: a
85            // paragraph's or span's style may set them too).
86            if self.strike == Some(false) {
87                x.double_strike = None;
88            }
89            for b in [&mut x.double_strike, &mut x.hidden, &mut x.outline, &mut x.shadow, &mut x.emboss, &mut x.engrave] {
90                if *b == Some(false) {
91                    *b = None;
92                }
93            }
94            x.spacing = x.spacing.filter(|s| *s != 0.0);
95            x.scale = x.scale.filter(|s| *s != 100.0);
96            f.extra = (x != crate::doc::props::CharProps::default()).then(|| Box::new(x));
97        }
98    }
99}

101/// The underline `style:text-underline-style`, `-type`, `-width` and `-mode` describe (ODF 1.3
102/// §20.390–20.393), when underlined.
103fn underline_kind(a: &Attrs) -> Underline {
104    let style = get(a, "style:text-underline-style").map(str::trim).unwrap_or("solid");
105    let double = get(a, "style:text-underline-type").is_some_and(|t| t.trim() == "double");
106    let bold = get(a, "style:text-underline-width").is_some_and(|w| matches!(w.trim(), "bold" | "thick" | "medium"));
107    let words = get(a, "style:text-underline-mode").is_some_and(|m| m.trim() == "skip-white-space");
108    match style {
109        "dotted" => Underline::Dotted,
110        "dash" | "long-dash" => Underline::Dash,
111        "dot-dash" => Underline::DotDash,
112        "dot-dot-dash" => Underline::DotDotDash,
113        "wave" if double => Underline::DoubleWave,
114        "wave" => Underline::Wave,
115        _ if double => Underline::Double,
116        _ if bold => Underline::Thick,
117        _ if words => Underline::Words,
118        _ => Underline::Single,
119    }
120}

122/// The rest of a `style:text-properties` element: what only [`Fmt::extra`] holds.
123/// The Asian and complex-script fonts, complex-script size, weight and posture and both
124/// languages of a `style:text-properties` element (ODF 1.3 §20.2xx `-asian`/`-complex`
125/// attributes), into `x`.
126fn script_properties(a: &Attrs, x: &mut crate::doc::props::CharProps, faces: &HashMap<String, String>) {
127    let family = |fam: &str, name: &str| {
128        let v = get(a, fam).or_else(|| get(a, name).map(|n| faces.get(n).map_or(n, String::as_str)))?;
129        let v = v.split(',').next().unwrap_or("").trim().trim_matches(['\'', '"']);
130        (!v.is_empty() && v.len() <= model::MAX_FONT_NAME).then(|| v.to_string())
131    };
132    if let Some(f) = family("style:font-family-asian", "style:font-name-asian") {
133        x.font_ea = Some(f);
134    }
135    if let Some(f) = family("style:font-family-complex", "style:font-name-complex") {
136        x.font_cs = Some(f);
137    }
138    if let Some(s) = get(a, "style:font-size-complex").and_then(length).filter(|s| (1.0..=1638.0).contains(s)) {
139        x.size_cs = Some(s);
140    }
141    if let Some(w) = get(a, "style:font-weight-complex") {
142        let w = w.trim();
143        x.bold_cs = Some(w.eq_ignore_ascii_case("bold") || w.parse::<u32>().is_ok_and(|n| n >= 600));
144    }
145    if let Some(s) = get(a, "style:font-style-complex") {
146        x.italic_cs = Some(matches!(s.trim(), "italic" | "oblique"));
147    }
148    if let Some(l) = language_tag(a, "style:language-asian", "style:script-asian", "style:country-asian") {
149        x.lang_ea = Some(l);
150    }
151    if let Some(l) = language_tag(a, "style:language-complex", "style:script-complex", "style:country-complex") {
152        x.lang_bidi = Some(l);
153    }
154}

156fn extra_text_properties(a: &Attrs, d: &mut FmtDelta) {
157    let underlined = d.fmt_underline_on();
158    let x = &mut d.extra;
159    if underlined {
160        x.underline = Some(underline_kind(a));
161        if let Some(c) = get(a, "style:text-underline-color").filter(|c| c.trim() != "font-color").and_then(model::parse_color) {
162            x.underline_color = Some(c);
163        }
164    }
165    if let Some(t) = get(a, "style:text-line-through-type") {
166        x.double_strike = Some(t.trim() == "double");
167    }
168    // A border around the text (LibreOffice's `loext:border`, or its sides): Word's text border,
169    // one line all round (the first side with a line), its padding the border's distance.
170    let side = ["loext:border", "loext:border-top", "loext:border-left", "loext:border-bottom", "loext:border-right"]
171        .into_iter()
172        .find_map(|k| get(a, k).and_then(parse_border).filter(|b| b.is_visible()));
173    if let Some(mut b) = side {
174        let pad = ["loext:padding", "loext:padding-top", "loext:padding-left"].into_iter().find_map(|k| get(a, k).and_then(length));
175        b.space = pad.filter(|p| p.is_finite()).map_or(0.0, |p| p.clamp(0.0, 31.0));
176        x.border = Some(b);
177    }
178    if let Some(t) = get(a, "fo:text-transform") {
179        x.caps = Some(t.trim() == "uppercase");
180    }
181    if let Some(v) = get(a, "fo:font-variant") {
182        x.small_caps = Some(v.trim() == "small-caps");
183    }
184    if let Some(v) = get(a, "text:display") {
185        x.hidden = Some(v.trim() == "none");
186    }
187    // Pair kerning (§20.317): on as Word's at every size, or off.
188    if let Some(v) = get(a, "style:letter-kerning") {
189        x.kern = Some(if v.trim() == "true" { 1.0 } else { 0.0 });
190    }
191    if let Some(v) = get(a, "fo:letter-spacing") {
192        x.spacing = Some(if v.trim() == "normal" { 0.0 } else { length(v).unwrap_or(0.0).clamp(-1584.0, 1584.0) });
193    }
194    if let Some(v) = get(a, "style:text-scale")
195        .and_then(|v| v.trim().strip_suffix('%'))
196        .and_then(|v| v.trim().parse::<f32>().ok())
197        .filter(|v| v.is_finite() && *v > 0.0)
198    {
199        x.scale = Some(v.clamp(1.0, 600.0));
200    }
201    // Emphasis marks (§20.370 `style:text-emphasize`): Word's dot, comma (an accent), circle
202    // (a disc is a dot) and dot below.
203    if let Some(v) = get(a, "style:text-emphasize") {
204        use crate::doc::props::EmphasisMark as E;
205        let mut words = v.split_whitespace();
206        let below = v.split_whitespace().nth(1) == Some("below");
207        x.emphasis = Some(match words.next() {
208            Some("dot" | "disc") if below => E::UnderDot,
209            Some("dot" | "disc") => E::Dot,
210            Some("accent") => E::Comma,
211            Some("circle") => E::Circle,
212            _ => E::None,
213        });
214    }
215    if let Some(v) = get(a, "style:text-outline") {
216        x.outline = Some(v.trim() == "true");
217    }
218    if let Some(v) = get(a, "fo:text-shadow") {
219        x.shadow = Some(v.trim() != "none");
220    }
221    if let Some(v) = get(a, "style:font-relief") {
222        (x.emboss, x.engrave) = (Some(v.trim() == "embossed"), Some(v.trim() == "engraved"));
223    }
224    if let Some(l) = language(a) {
225        x.lang = Some(l);
226    }
227}

234/// The language the attributes `language`, `script` and `country` state, as a tag (`de-DE`,
235/// `sr-Latn-RS`); `None` when they state none, or no language at all (`none`, `zxx`). A script of
236/// any other shape than a script's is left out: written back, it would read as a country.
237fn language_tag(a: &Attrs, language: &str, script: &str, country: &str) -> Option<String> {
238    fn tag(v: Option<&str>) -> Option<&str> {
239        v.map(str::trim).filter(|v| (1..=8).contains(&v.len()) && v.chars().all(|c| c.is_ascii_alphanumeric()) && *v != "none")
240    }
241    let language = tag(get(a, language)).filter(|l| *l != "zxx")?;
242    let script = tag(get(a, script)).filter(|s| super::writer::is_script(s));
243    Some([Some(language), script, tag(get(a, country))].into_iter().flatten().collect::<Vec<_>>().join("-"))
244}

336/// A `style:tab-stop` (ODF 1.3 §17.8), its position from where the style measures; `None` when
337/// it has no usable position.
338fn tab_stop(a: &Attrs) -> Option<TabStop> {
339    let pos = get(a, "style:position").and_then(length).filter(|p| p.is_finite())?.clamp(-MAX_LEN, MAX_LEN);
340    let align = match get(a, "style:type").map(str::trim) {
341        Some("center") => TabAlign::Center,
342        Some("right") => TabAlign::Right,
343        Some("char") => TabAlign::Decimal,
344        _ => TabAlign::Left,
345    };
346    let style = get(a, "style:leader-style").map(str::trim).unwrap_or("none");
347    let leader = if style == "none" {
348        TabLeader::None
349    } else {
350        match get(a, "style:leader-text").and_then(|t| t.chars().next()) {
351            Some('.') => TabLeader::Dot,
352            Some('-') => TabLeader::Hyphen,
353            Some('_') => TabLeader::Underscore,
354            Some('·') => TabLeader::MiddleDot,
355            _ if style == "dotted" => TabLeader::Dot,
356            _ if matches!(style, "dash" | "long-dash") => TabLeader::Hyphen,
357            _ => TabLeader::Underscore,
358        }
359    };
360    Some(TabStop { pos, align, leader })
361}

363/// Paragraph spacing, indents and line height of a `style:paragraph-properties` element (ODF 1.3
364/// §20.204–209, §20.225, §20.317–318) over `sp`. Percentages of the parent's margins aren't
365/// followed (left as they were); spacing is clamped to 0..=1584 pt, indents to ±1584 pt.
366fn paragraph_spacing(a: &Attrs, sp: &mut Spacing) {
367    let len = |k: &str| get(a, k).and_then(length);
368    let space = |k: &str| len(k).map(|v| v.clamp(0.0, MAX_LEN));
369    let indent = |k: &str| len(k).map(|v| v.clamp(-MAX_LEN, MAX_LEN));
370    if let Some(v) = get(a, "fo:margin").and_then(length) {
371        sp.before = Some(v.clamp(0.0, MAX_LEN));
372        sp.after = sp.before;
373        sp.left = Some(v.clamp(-MAX_LEN, MAX_LEN));
374        sp.right = sp.left;
375    }
376    sp.before = space("fo:margin-top").or(sp.before);
377    sp.after = space("fo:margin-bottom").or(sp.after);
378    sp.left = indent("fo:margin-left").or(sp.left);
379    sp.right = indent("fo:margin-right").or(sp.right);
380    sp.first = indent("fo:text-indent").or(sp.first);
381    if let Some(v) = get(a, "fo:line-height").map(str::trim) {
382        if v.eq_ignore_ascii_case("normal") {
383            sp.line = Some(LineSpacing::Multiple(1.0));
384        } else if let Some(pc) = v.strip_suffix('%').and_then(|n| n.trim().parse::<f32>().ok()).filter(|n| n.is_finite() && *n > 0.0) {
385            // Word's range: a sixteenth of a line up to 132 lines.
386            sp.line = Some(LineSpacing::Multiple((pc / 100.0).clamp(0.06, 132.0)));
387        } else if let Some(h) = length(v).filter(|h| *h > 0.0) {
388            sp.line = Some(LineSpacing::Exactly(h.clamp(0.7, MAX_LEN)));
389        }
390    }
391    if let Some(h) = len("style:line-height-at-least").filter(|h| *h >= 0.0) {
392        sp.line = Some(LineSpacing::AtLeast(h.clamp(0.7, MAX_LEN)));
393    }
394    // Leading (§20.336: extra space between lines, above each line as LibreOffice draws it) has
395    // no WordCraft mode: at least a 12 pt line and the leading, which also puts the extra space
396    // above the text.
397    // LibreOffice also leaves the leading after the last line: space after the paragraph.
398    if let Some(l) = len("style:line-spacing").filter(|l| *l >= 0.0) {
399        let l = l.min(MAX_LEN);
400        sp.line = Some(LineSpacing::AtLeast((LEADING_LINE + l).clamp(0.7, MAX_LEN)));
401        sp.after = Some((sp.after.unwrap_or(0.0) + l).clamp(0.0, MAX_LEN));
402    }
403}

408/// Height of the single line `style:line-spacing` leading is added to (12 pt text).

411/// `over` laid on `base`: what `over` sets wins.
412fn overlay_spacing(base: &mut Spacing, over: &Spacing) {
413    base.before = over.before.or(base.before);
414    base.after = over.after.or(base.after);
415    base.line = over.line.or(base.line);
416    base.left = over.left.or(base.left);
417    base.right = over.right.or(base.right);
418    base.first = over.first.or(base.first);
419}

421/// The paragraph kind a style name (lower case, `_20_` as spaces) stands for.
422fn kind_named(n: &str) -> Option<Kind> {
423    if let Some(d) = n.strip_prefix("heading ").and_then(|r| r.trim().parse::<u8>().ok()) {
424        Some(Kind::Heading(d.clamp(1, 6)))
425    } else {
426        match n {
427            "title" => Some(Kind::Title),
428            "quotations" | "quote" => Some(Kind::Quote),
429            "preformatted text" | "source text" | "code" => Some(Kind::Code),
430            _ => None,
431        }
432    }

496/// Reads date and time styles (`number:date-style`, `number:time-style`) as Word pictures.
497#[derive(Default)]
498struct DateStyles {
499    /// The style being read: its name and parts.
500    data: Option<(String, Vec<super::datastyle::Part>)>,
501    /// It orders its parts as the locale does (§19.341 `number:automatic-order`).
502    auto_order: bool,
503    /// In its `number:text`.
504    in_text: bool,
505}
506
507impl DateStyles {
508    /// Event `ev` of a style file, into `out` (name → picture); false when it isn't one of a
509    /// date or time style.
510    fn event(&mut self, ev: &Ev, out: &mut HashMap<String, String>) -> bool {
511        use super::datastyle::{Part, part, picture};
512        match ev {
513            Ev::Start(q, a) => match local(q) {
514                "date-style" | "time-style" => {
515                    self.data = get(a, "style:name").map(|n| (n.to_string(), Vec::new()));
516                    self.auto_order = local(q) == "date-style" && get(a, "number:automatic-order").is_some_and(|v| v.trim() == "true");
517                }
518                "text" => self.in_text = self.data.is_some(),
519                name => {
520                    let on = |k: &str, v: &str| get(a, k).is_some_and(|x| x.trim() == v);
521                    if let (Some((_, parts)), Some(p)) = (&mut self.data, part(name, on("number:style", "long"), on("number:textual", "true")))
522                        && parts.len() < 64
523                    {
524                        parts.push(p);
525                    }
526                }
527            },
528            Ev::End(q) => match local(q) {
529                "text" => self.in_text = false,
530                "date-style" | "time-style" => {
531                    if let Some((name, parts)) = self.data.take()
532                        && out.len() < MAX_DATA_STYLES
533                    {
534                        let parts = if std::mem::take(&mut self.auto_order) { super::datastyle::auto_ordered(&parts) } else { parts };
535                        out.insert(name, picture(&parts));
536                    }
537                }
538                _ => {}
539            },
540            Ev::Text(t) => {
541                if !self.in_text {
542                    return false;
543                }
544                if let Some((_, parts)) = &mut self.data {
545                    let n = parts.len();
546                    if let Some(Part::Text(s)) = parts.last_mut()
547                        && s.len() < 200
548                    {
549                        s.push_str(t);
550                    } else if n < 64 {
551                        parts.push(Part::Text(t.chars().take(200).collect()));
552                    }
553                }
554            }
555        }
556        true
557    }
558}

577/// An ODF border (`0.5pt solid #000000`, `none`; ODF 1.3 §20.183): width, style and colour in any
578/// order. `none` and `hidden` are an explicit no line; `None` when nothing in it reads.
579pub(super) fn parse_border(v: &str) -> Option<Border> {
580    let mut b = Border { style: BorderStyle::Single, width: 0.75, color: None, space: 0.0 };
581    let mut any = false;
582    for t in v.split_whitespace() {
583        let t = t.to_ascii_lowercase();
584        match t.as_str() {
585            "none" | "hidden" => return Some(Border { style: BorderStyle::None, width: 0.0, color: None, space: 0.0 }),
586            "solid" | "groove" | "ridge" | "inset" | "outset" => b.style = BorderStyle::Single,
587            "double" => b.style = BorderStyle::Double,
588            "dotted" => b.style = BorderStyle::Dotted,
589            "dashed" => b.style = BorderStyle::Dashed,
590            "dot-dash" | "dot-dot-dash" => b.style = BorderStyle::DotDash,
591            "wave" => b.style = BorderStyle::Wave,
592            "thin" => b.width = 0.5,
593            "medium" => b.width = 1.0,
594            "thick" => b.width = 2.0,
595            _ => {
596                if let Some(c) = model::parse_color(&t) {
597                    b.color = Some(c);
598                } else if let Some(w) = length(&t).filter(|w| w.is_finite()) {
599                    b.width = w.clamp(0.0, 12.0);
600                } else {
601                    continue;
602                }
603            }
604        }
605        any = true;
606    }
607    any.then_some(b)
608}

632fn style_child(name: &str, q: &str, a: &Attrs, s: &mut StyleInfo) -> bool {
633    match name {
634        "paragraph-properties" => paragraph_properties(a, s),
635        "tab-stops" => s.tabs = Some(Vec::new()),
636        // One line drops nothing.
637        "drop-cap" => {
638            let lines = get(a, "style:lines").and_then(|l| l.trim().parse::<u32>().ok()).unwrap_or(1);
639            let distance = get(a, "style:distance").and_then(length).filter(|d| d.is_finite()).map_or(0.0, |d| d.clamp(0.0, 1584.0));
640            s.drop_cap = Some(((lines.min(10) as u8), distance)).filter(|(l, _)| *l > 1);
641        }
642        "tab-stop" => {
643            if let Some(t) = s.tabs.as_mut().filter(|t| t.len() < MAX_TABS)
644                && let Some(stop) = tab_stop(a)
645            {
646                t.push(stop);
647            }
648        }
649        "table-properties" => {
650            // A table starting a page (§20.191 `fo:break-before`).
651            if get(a, "fo:break-before").is_some_and(|b| b.trim().eq_ignore_ascii_case("page")) {
652                s.page_break = true;
653            }
654            if let Some(v) = get(a, "style:page-number") {
655                s.page_number = Some(v.trim().parse::<u32>().ok().map(|n| n.min(32_767)));
656            }
657            // §20.300 `table:align`: `margins` fills the text width, as no alignment does.
658            s.table_background = table_background_of(a).or(s.table_background);
659            s.table_align = match get(a, "table:align").map(str::trim) {
660                Some("left") => Some(Align::Left),
661                Some("center") => Some(Align::Center),
662                Some("right") => Some(Align::Right),
663                _ => s.table_align,
664            };
665            if let Some(w) = get(a, "style:width").and_then(length).filter(|w| w.is_finite() && *w > 0.0) {
666                s.table_width = Some(w.min(MAX_LEN));
667            }
668            if let Some(p) =
669                get(a, "style:rel-width").and_then(|v| v.trim().strip_suffix('%')?.trim().parse::<f32>().ok()).filter(|p| p.is_finite() && *p > 0.0)
670            {
671                s.table_rel_width = Some(p.min(100.0));
672            }
673            for (i, k) in ["fo:margin-top", "fo:margin-bottom"].iter().enumerate() {
674                if let (Some(v), Some(slot)) = (get(a, k).and_then(length).filter(|v| v.is_finite()), s.table_space.get_mut(i)) {
675                    *slot = Some(v.clamp(0.0, MAX_LEN));
676                }
677            }
678        }
679        "table-row-properties" => {
680            s.table_background = table_background_of(a).or(s.table_background);
681            if let Some(h) = get(a, "style:row-height").and_then(length).filter(|h| h.is_finite() && *h > 0.0) {
682                s.row_height = Some((h.min(MAX_LEN), true));
683            } else if let Some(h) = get(a, "style:min-row-height").and_then(length).filter(|h| h.is_finite() && *h > 0.0) {
684                s.row_height = Some((h.min(MAX_LEN), false));
685            }
686            if let Some(k) = get(a, "fo:keep-together") {
687                s.row_keep = Some(k.trim() == "always");
688            }
689        }
690        "table-column-properties" => {
691            if let Some(w) = get(a, "style:column-width").and_then(length) {
692                s.column_width = Some(w);
693            }
694            if let Some(r) = column_share_of(a) {
695                s.column_share = Some(r);
696            }
697        }
698        "table-cell-properties" => table_cell_properties(a, s),
699        "section-properties" => {
700            s.dont_balance = get(a, "text:dont-balance-text-columns").map(|v| v.trim() == "true");
701        }
702        "columns" if q.starts_with("style:") => {
703            s.columns = Some(columns_of(a));
704        }
705        "column" if q.starts_with("style:") && s.columns.is_some() => {
706            if let Some(spec) = column_spec(a).filter(|_| s.column_specs.len() < MAX_COLUMN_SPECS) {
707                s.column_specs.push(spec);
708            }
709        }
710        "column-sep" => {
711            if let Some(c) = &mut s.columns {
712                c.separator = column_sep(a);
713            }
714        }
715        "graphic-properties" => s.graphic = Some(super::frames::Graphic::read(a)),
716        "background-image" if q.starts_with("style:") && s.family.as_deref() == Some("table-cell") => {
717            if let Some(href) = get(a, "xlink:href").filter(|h| !h.trim().is_empty() && h.len() < 1024) {
718                let attr = |k: &str| get(a, k).unwrap_or_default().trim().to_string();
719                s.cell_picture = Some((href.trim().to_string(), attr("style:repeat"), attr("style:position")));
720            }
721        }
722        _ => return false,
723    }
724    true
725}

805/// `c` with the widths its `style:column`s give, in a text `tw` points wide: each column's
806/// share of the width less its indents, and the space after it its end indent with the next
807/// column's start indent. Equal columns keep equal widths and take the gap between them.
808pub(super) fn with_widths(mut c: Columns, specs: &[ColumnSpec], tw: f32) -> Columns {
809    let total: f32 = specs.iter().map(|s| s.0).sum();
810    if c.count < 2 || specs.len() != c.count as usize || !(total.is_finite() && total > 0.0) || !(tw.is_finite() && tw > 0.0) {
811        return c;
812    }
813    let boxes: Vec<(f32, f32)> = specs
814        .iter()
815        .enumerate()
816        .map(|(i, (rel, start, end))| {
817            let next_start = specs.get(i + 1).map_or(0.0, |s| s.1);
818            let after = if i + 1 < specs.len() { end + next_start } else { 0.0 };
819            ((rel / total * tw - start - end).max(18.0), after)
820        })
821        .collect();
822    let (w0, gap) = boxes.first().copied().unwrap_or_default();
823    let equal = boxes.iter().all(|(w, _)| (w - w0).abs() < 0.5) && boxes.iter().rev().skip(1).all(|(_, a)| (a - gap).abs() < 0.5);
824    if equal {
825        c.space = gap;
826        c.widths.clear();
827    } else {
828        c.widths = boxes;
829    }
830    c
831}

850fn text_properties(a: &Attrs, style: &mut StyleInfo, faces: &HashMap<String, String>) {
851    let fmt = &mut style.fmt;
852    let is_bold = |weight: &str| {
853        let weight = weight.trim();
854        weight.eq_ignore_ascii_case("bold") || weight.parse::<u32>().is_ok_and(|n| n >= 600)
855    };
856    // Latin attributes take precedence; Asian and complex-script attributes are fallbacks, to turn
857    // it on: one of those saying normal says nothing of Latin text.
858    if let Some(weight) = get(a, "fo:font-weight") {
859        fmt.bold = Some(is_bold(weight));
860    } else if get(a, "style:font-weight-asian").or_else(|| get(a, "style:font-weight-complex")).is_some_and(is_bold) {
861        fmt.bold = Some(true);
862    }
863    let is_italic = |value: &str| value.trim().eq_ignore_ascii_case("italic") || value.trim().eq_ignore_ascii_case("oblique");
864    if let Some(value) = get(a, "fo:font-style") {
865        fmt.italic = Some(is_italic(value));
866    } else if get(a, "style:font-style-asian").or_else(|| get(a, "style:font-style-complex")).is_some_and(is_italic) {
867        fmt.italic = Some(true);
868    }
869    let is_not_none = |value: &str| !value.trim().eq_ignore_ascii_case("none");
870    if let Some(value) = get(a, "style:text-underline-style").or_else(|| get(a, "style:text-underline-type")) {
871        fmt.underline = Some(is_not_none(value));
872    }
873    if let Some(value) = get(a, "style:text-line-through-style").or_else(|| get(a, "style:text-line-through-type")) {
874        fmt.strike = Some(is_not_none(value));
875    }
876    if let Some(value) = get(a, "style:text-position") {
877        // `super`/`sub` or a percentage raising the text, then its size (§20.374): raised text
878        // at full size is a position, smaller text superscript or subscript.
879        let mut parts = value.split_whitespace();
880        let first = parts.next().unwrap_or("");
881        let pct = first.trim_end_matches('%').parse::<f32>().ok().filter(|p| p.is_finite()).unwrap_or(0.0);
882        let full = parts.next().and_then(|s| s.trim_end_matches('%').parse::<f32>().ok()).is_some_and(|s| s >= 100.0);
883        let keyword = first.eq_ignore_ascii_case("super") || first.eq_ignore_ascii_case("sub");
884        if full && !keyword {
885            fmt.sup = Some(false);
886            fmt.sub = Some(false);
887            fmt.raise = Some(pct.clamp(-10_000.0, 10_000.0));
888        } else {
889            fmt.sup = Some(first.eq_ignore_ascii_case("super") || pct > 0.0);
890            fmt.sub = Some(first.eq_ignore_ascii_case("sub") || pct < 0.0);
891            fmt.raise = Some(0.0);
892        }
893    }
894    if let Some(color) = get(a, "fo:color").and_then(model::parse_color) {
895        fmt.color = Some(color);
896    }
897    if let Some(color) = get(a, "fo:background-color").and_then(model::parse_color) {
898        fmt.background = Some(color);
899    }
900    if let Some(pct) = get(a, "fo:font-size")
901        .and_then(|v| v.trim().strip_suffix('%'))
902        .and_then(|v| v.trim().parse::<f32>().ok())
903        .filter(|p| p.is_finite() && *p > 0.0)
904    {
905        fmt.size_pct = Some(pct.min(10_000.0));
906    } else if let Some(size) = get(a, "fo:font-size").and_then(length).filter(|size| (1.0..=1638.0).contains(size)) {
907        fmt.size = Some(size);
908    }
909    if let Some(family) = family_of(a, faces) {

976/// The most padding a side without a line keeps, points (22 inches).
977const MAX_PADDING: f32 = 1584.0;
978
979/// `fo:border`, `fo:border-top`… of `a` (a side's own value wins over the shorthand), each side's
980/// `fo:padding*` as its distance from the text (Word's 31 pt at most beside a line); a side
981/// with padding and no line is an invisible border holding it. `None` when none is stated.
982fn border_sides(a: &Attrs) -> Option<Borders> {
983    let all = get(a, "fo:border").and_then(parse_border);
984    let pad_all = get(a, "fo:padding").and_then(length);
985    let side = |b: &str, p: &str| {
986        let pad = get(a, p).and_then(length).or(pad_all).filter(|v| v.is_finite() && *v > 0.0);
987        // Padding without a line still takes room (LibreOffice): a side with no line holds it.
988        let Some(mut line) = get(a, b).and_then(parse_border).or(all) else {
989            return pad.map(|sp| Border { style: BorderStyle::None, width: 0.0, color: None, space: sp.min(MAX_PADDING) });
990        };
991        if let Some(sp) = pad {
992            line.space = if line.is_visible() { sp.min(31.0) } else { sp.min(MAX_PADDING) };
993        }
994        Some(line)
995    };
996    let b = Borders {
997        top: side("fo:border-top", "fo:padding-top"),
998        left: side("fo:border-left", "fo:padding-left"),
999        bottom: side("fo:border-bottom", "fo:padding-bottom"),
1000        right: side("fo:border-right", "fo:padding-right"),
1001        ..Default::default()
1002    };
1003    (b.top.is_some() || b.left.is_some() || b.bottom.is_some() || b.right.is_some()).then_some(b)
1004}

1384        let chain = self.chain(name);
1385        let Some(at) = chain.iter().position(|(n, _)| self.is_named_text(n)) else { return self.text_fmt(name, base) };
1386        let mut f = base.clone();

1393                if d.bold.is_some() && x.bold == Some(false) {
1394                    x.bold = None;
1395                }
1396                if d.italic.is_some() && x.italic == Some(false) {
1397                    x.italic = None;
1398                }
1399                if d.underline.is_some() && x.underline == Some(Underline::None) {
1400                    x.underline = None;
1401                }
1402                if (d.strike.is_some() || e.double_strike.is_some()) && x.strike == Some(false) {
1403                    (x.strike, x.double_strike) = (None, None);
1404                }
1405                if e.caps.is_some() && x.caps == Some(false) {
1406                    x.caps = None;
1407                }
1408                if e.small_caps.is_some() && x.small_caps == Some(false) {
1409                    x.small_caps = None;
1410                }
1411            }
1412        }
1413        for (_, s) in chain.iter().take(at).rev() {
1414            s.fmt.apply(&mut f);
1415        }
1416        if let Some((n, _)) = chain.get(at) {
1417            f.extra.get_or_insert_with(Default::default).style = Some(n.to_string());
1418        }
1419        f

1548    /// Paragraph kind, alignment, page break, rule, keep-with-next and base formatting. The text formatting of
1549    /// every style in the chain applies (automatic and common, ODF 1.3 §16.2 inheritance), except
1550    /// the style that gives the paragraph its kind and that style's ancestors: their formatting
1551    /// is the kind's own look (a heading's size and colour), not direct formatting. `Standard`,
1552    /// the default paragraph style, holds document defaults and is left out too.
1553    /// The look the file gives each style that maps to a paragraph kind (Heading 1–6, Title,
1554    /// Quotations, Preformatted Text): its text formatting, spacing, alignment and keep with
1555    /// next, over the whole chain but Standard (the document defaults). A style named for the
1556    /// kind wins over one only displayed so.
1557    pub(super) fn kind_looks(&self) -> Vec<(Kind, ParaLook)> {
1558        let mut found: Vec<(Kind, bool, &str)> = Vec::new();
1559        for (n, s) in self.map.iter().filter(|(_, s)| !s.auto) {
1560            let own = kind_named(&n.to_ascii_lowercase().replace("_20_", " "));
1561            let shown = s.display.as_deref().and_then(|d| kind_named(&d.to_ascii_lowercase()));
1562            let Some((kind, canonical)) = own.map(|k| (k, true)).or(shown.map(|k| (k, false))) else { continue };
1563            match found.iter_mut().find(|(k, ..)| *k == kind) {
1564                Some(slot) if !slot.1 && canonical => *slot = (kind, canonical, n),
1565                Some(slot) if slot.1 == canonical && n.as_str() < slot.2 => *slot = (kind, canonical, n),
1566                Some(_) => {}
1567                None => found.push((kind, canonical, n)),
1568            }
1569        }
1570        found
1571            .into_iter()
1572            .map(|(kind, _, name)| {
1573                let chain: Vec<_> = self.chain(name).into_iter().filter(|(n, _)| *n != "Standard").collect();
1574                let mut fmt = Fmt::default();
1575                let mut spacing = Spacing::default();
1576                for (_, s) in chain.iter().rev() {
1577                    s.fmt.apply(&mut fmt);
1578                    overlay_spacing(&mut spacing, &s.spacing);
1579                }
1580                let align = chain.iter().find_map(|(_, s)| s.align);
1581                let keep_next = chain.iter().find_map(|(_, s)| s.keep_next).unwrap_or(false);
1582                let contextual = chain.iter().find_map(|(_, s)| s.contextual);
1583                (kind, ParaLook { kind, align, keep_next, fmt, spacing, contextual, ..Default::default() })
1584            })
1585            .collect()
1586    }

1606    pub(super) fn named_style(&self, odt: &str) -> Option<NamedStyle> {
1607        if !self.is_named(odt) {
1608            return None;
1609        }
1610        let s = self.map.get(odt)?;
1611        let lower = |n: &str| n.to_ascii_lowercase().replace("_20_", " ");
1612        let parent = match s.parent.as_deref() {
1613            None | Some("Standard") => Parent::Normal,
1614            Some(p) => match self.map.get(p) {
1615                None => Parent::Normal,
1616                Some(ps) => match kind_named(&lower(p)).or_else(|| ps.display.as_deref().and_then(|d| kind_named(&lower(d)))) {
1617                    Some(k) => Parent::Kind(k),
1618                    None if ps.auto => Parent::Normal,
1619                    None => Parent::Named(p.to_string()),
1620                },
1621            },
1622        };
1623        let mut fmt = Fmt::default();
1624        s.fmt.apply(&mut fmt);
1625        // A percent is of what's under it: the parent's size, or with none the default style's.
1626        if s.fmt.size_pct.is_some() {
1627            fmt.size = self.named_size(odt);
1628        }
1629        // A style with no parent inherits the default style, not Standard (the document's
1630        // defaults here): its size, when Standard's differs.
1631        if s.parent.is_none() && fmt.size.is_none() && self.doc_size() != self.default_size {
1632            fmt.size = self.default_size;
1633        }
1634        // And its language, likewise.
1635        if s.parent.is_none()
1636            && let Some(lang) = self.default_lang.as_ref().filter(|l| self.document_lang().as_ref() != Some(*l))
1637            && fmt.extra.as_ref().is_none_or(|x| x.lang.is_none())
1638        {
1639            fmt.extra.get_or_insert_with(Default::default).lang = Some(lang.clone());
1640        }
1641        let look = ParaLook {
1642            align: s.align,
1643            keep_next: s.keep_next == Some(true),
1644            fmt,
1645            spacing: s.spacing,
1646            tabs: s.tabs.clone(),
1647            borders: s.para_borders.filter(|b| b.takes_room()),
1648            shading: s.para_shading.flatten(),
1649            contextual: s.contextual,
1650            rtl: s.rtl,
1651            ..Default::default()
1652        };
1653        let name = s.display.clone().unwrap_or_else(|| odt.replace("_20_", " "));
1654        let off = [s.fmt.bold, s.fmt.italic, s.fmt.underline, s.fmt.strike].map(|v| v == Some(false));
1655        let outline = s.outline.filter(|o| (1..=10).contains(o));
1656        Some(NamedStyle { name: name.chars().take(super::super::model::MAX_FONT_NAME).collect(), parent, look, off, outline })
1657    }

1659    pub(super) fn para(&self, name: &str) -> ParaLook {
1660        let chain = self.chain(name);
1661        let mut kind = Kind::Normal;
1662        let mut kind_at = chain.len();
1663        // The nearest outline level; a style with one is no kind's style: its look is its own.
1664        let mut outline = None;
1665        for (i, (n, s)) in chain.iter().enumerate() {
1666            kind_at = i;
1667            let dn = s.display.as_deref().unwrap_or(n).to_ascii_lowercase().replace("_20_", " ");
1668            let nn = n.to_ascii_lowercase().replace("_20_", " ");
1669            for cand in [dn.as_str(), nn.as_str()] {
1670                if let Some(d) = cand.strip_prefix("heading ").and_then(|r| r.trim().parse::<u8>().ok()) {
1671                    kind = Kind::Heading(d.clamp(1, 6));
1672                } else if cand == "title" {
1673                    kind = Kind::Title;
1674                } else if cand == "quotations" || cand == "quote" {
1675                    kind = Kind::Quote;
1676                } else if cand == "preformatted text" || cand == "source text" || cand == "code" {
1677                    kind = Kind::Code;
1678                }
1679            }
1680            if kind != Kind::Normal {
1681                break;
1682            }
1683            outline = outline.or(s.outline.filter(|o| (1..=10).contains(o)));
1684            kind_at = chain.len();
1685        }
1686        if let Some(o) = outline {
1687            kind = Kind::Heading(o.min(6));
1688        }
1689        // The nearest custom style: its formatting stays in it; the automatic styles over it give
1690        // the paragraph's own.
1691        let named_at = chain.iter().take(kind_at).position(|(n, _)| self.is_named(n));
1692        let named = named_at.and_then(|i| chain.get(i)).map(|(n, _)| n.to_string()).or_else(|| {
1693            // An index style the file names but doesn't define (WordCraft has its own).
1694            let undefined = chain.last().map_or(Some(name), |(_, s)| s.parent.as_deref());
1695            undefined.filter(|n| super::writer::generating_style(n).is_some()).map(String::from)
1696        });
1697        let kind_at = named_at.unwrap_or(kind_at);
1698        let align = if named.is_some() { chain.iter().take(kind_at).find_map(|(_, s)| s.align) } else { chain.iter().find_map(|(_, s)| s.align) };
1699        let pb = chain.iter().take_while(|(_, s)| s.auto).any(|(_, s)| s.page_break);
1700        let column_break = !pb && chain.iter().take_while(|(_, s)| s.auto).any(|(_, s)| s.column_break);
1701        let rule = chain.iter().any(|(_, s)| s.border_bottom);
1702        let mut f = Fmt::default();
1703        let mut spacing = Spacing::default();
1704        for (_, s) in chain.iter().take(kind_at).rev().filter(|(n, s)| s.auto || *n != "Standard") {
1705            s.fmt.apply(&mut f);
1706            overlay_spacing(&mut spacing, &s.spacing);
1707        }
1708        // Like the text formatting, the kind style's own `fo:keep-with-next` (a heading's) is the kind's look, not direct formatting.
1709        let keep_next = chain.iter().take(kind_at).find_map(|(_, s)| s.keep_next).unwrap_or(false);
1710        // A style's tab stops replace its parent's; the kind's style and Standard give none.
1711        let tabs = chain.iter().take(kind_at).filter(|(n, s)| s.auto || *n != "Standard").find_map(|(_, s)| s.tabs.clone());
1712        // Borders and background: the nearest in the chain, the kind's style and Standard aside.
1713        let own = || chain.iter().take(kind_at).filter(|(n, s)| s.auto || *n != "Standard");
1714        let borders = own().find_map(|(_, s)| s.para_borders).filter(|b| b.takes_room());
1715        let shading = own().find_map(|(_, s)| s.para_shading).flatten();
1716        let contextual = own().find_map(|(_, s)| s.contextual);
1717        let rtl = own().find_map(|(_, s)| s.rtl);
1718        let drop_cap = own().find_map(|(_, s)| s.drop_cap);
1719        let keep_lines = own().find_map(|(_, s)| s.keep_lines);
1720        let widow_control = own().find_map(|(_, s)| s.widow_control);
1721        // A page break after the paragraph: its own style's (automatic or common).
1722        let break_after = own().take(1).any(|(_, s)| s.break_after);
1723        let hyphenate = chain.iter().find_map(|(_, s)| s.hyphenate);
1724        // Where LibreOffice measures the tab stops from: the indent the paragraph ends up with,
1725        // whichever style in the chain gives it.
1726        let tab_origin = chain.iter().find_map(|(_, s)| s.spacing.left).filter(|v| v.is_finite()).unwrap_or(0.0);
1727        ParaLook {
1728            kind,
1729            align,
1730            page_break: pb,
1731            column_break,
1732            rule,
1733            keep_next,
1734            fmt: f,
1735            spacing,
1736            tabs,
1737            borders,
1738            shading,
1739            named,
1740            contextual,
1741            rtl,
1742            tab_origin,
1743            drop_cap,
1744            keep_lines,
1745            widow_control,
1746            break_after,
1747            hyphenate,
1748        }
1749    }
