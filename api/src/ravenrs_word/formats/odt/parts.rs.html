1//! Package parts WordCraft doesn't read (ODF 1.3 Part 2 §3: any file a package lists), kept as
2//! they are and written back: macros and dialogs (`Basic/`, `Dialogs/`, `Scripts/`),
3//! LibreOffice's interface configuration (`Configurations2/`), metadata (`manifest.rdf`) and
4//! files of other applications.
5//!
6//! Not kept: what WordCraft reads and writes itself (content, styles, meta, settings, pictures,
7//! embedded objects), the thumbnail (it would show the document as it was), and digital
8//! signatures (`META-INF/*signatures*.xml`): they sign the package as it was, so a saved, changed
9//! document can't carry them.
10
11use std::sync::Arc;
12
13use crate::doc::Document;
14
15/// `Document::passthrough` key prefix of a kept part (its path follows), and the key of the list
16/// of kept paths with their media types (`path\tmedia type` lines; directories end in `/`).
17pub(super) const PREFIX: &str = "odf-part/";
18pub(super) const LIST: &str = "odf-parts";
19
20/// Most parts kept, and most bytes in all.
21const MAX_PARTS: usize = 1000;
22const MAX_BYTES: u64 = 64 << 20;
23
24/// Whether package path `name` is one WordCraft reads or writes itself, or drops.
25fn own(name: &str, manifest: &std::collections::HashMap<String, String>) -> bool {
26    const OWN: [&str; 6] = ["mimetype", "content.xml", "styles.xml", "meta.xml", "settings.xml", "META-INF/manifest.xml"];
27    if OWN.contains(&name) || name.ends_with('/') {
28        return true;
29    }
30    if ["Pictures/", "Thumbnails/", "ObjectReplacements/"].iter().any(|d| name.starts_with(d)) {
31        return true;
32    }
33    if name.starts_with("META-INF/") && name.contains("signatures") {
34        return true;
35    }
36    // An embedded object's files: its directory is listed as an OpenDocument.
37    let mut dir = name;
38    while let Some((up, _)) = dir.rsplit_once('/') {
39        if manifest
40            .get(&format!("{up}/"))
41            .is_some_and(|m| m.starts_with("application/vnd.oasis.opendocument.") || m.starts_with("application/vnd.sun.xml.math"))
42        {
43            return true;
44        }
45        dir = up;
46    }
47    false
48}

50/// Whether `name` is a path safe to write back: relative, without `.` or `..` steps or empty
51/// ones, and of no more than 512 bytes.
52fn safe(name: &str) -> bool {
53    !name.is_empty()
54        && name.len() <= 512
55        && !name.starts_with('/')
56        && !name.contains('\\')
57        && name.split('/').all(|s| !s.is_empty() && s != "." && s != "..")
58        && !name.chars().any(char::is_control)
59}

61/// The parts of a package WordCraft doesn't read itself: each file's path and bytes, and the
62/// list of their paths with media types (see [`LIST`]).
63#[derive(Default)]
64pub(super) struct Kept {
65    files: Vec<(String, Vec<u8>)>,
66    list: String,
67}
68
69impl Kept {
70    /// Keeps them in `doc`, for the next save.
71    pub(super) fn store(self, doc: &mut Document) {
72        for (name, bytes) in self.files {
73            doc.passthrough.insert(format!("{PREFIX}{name}"), Arc::new(bytes));
74        }
75        if !self.list.is_empty() {
76            doc.passthrough.insert(LIST.into(), Arc::new(self.list.into_bytes()));
77        }
78    }
79}

81/// Reads the parts of package `z` (entries `entries`) WordCraft doesn't read itself, with their
82/// media types from `manifest`.
83pub(super) fn read(
84    z: &mut zip::ZipArchive<std::io::Cursor<&[u8]>>,
85    entries: &[String],
86    manifest: &std::collections::HashMap<String, String>,
87) -> Kept {
88    use std::io::Read;
89    let mut kept = Kept::default();
90    let mut budget = MAX_BYTES;
91    let mut dirs = std::collections::BTreeSet::new();
92    let clean = |m: &str| m.replace(['\t', '\n', '\r'], " ");
93    for name in entries {
94        if kept.files.len() >= MAX_PARTS || own(name, manifest) || !safe(name) {
95            continue;
96        }
97        let Ok(f) = z.by_name(name) else { continue };
98        if f.size() > budget {
99            continue;
100        }
101        let mut bytes = Vec::new();
102        if f.take(budget).read_to_end(&mut bytes).is_err() {
103            continue;
104        }
105        budget = budget.saturating_sub(bytes.len() as u64);
106        let mime = manifest.get(name.as_str()).map_or("", String::as_str);
107        kept.list.push_str(&format!("{name}\t{}\n", clean(mime)));
108        // The directories it is in that the manifest lists (a Basic library, a configuration).
109        let mut dir = name.as_str();
110        while let Some((up, _)) = dir.rsplit_once('/') {
111            let d = format!("{up}/");
112            if let Some(m) = manifest.get(&d) {
113                dirs.insert((d, clean(m)));
114            }
115            dir = up;
116        }
117        kept.files.push((name.clone(), bytes));
118    }
119    for (d, m) in dirs {
120        kept.list.push_str(&format!("{d}\t{m}\n"));
121    }
122    kept
123}

125/// The parts kept in `doc` to write back: (path, bytes) of each file, and (path, media type) of
126/// each manifest entry (directories too). Paths WordCraft writes itself are left out.
127pub(super) fn kept(doc: &Document) -> (Vec<(String, Arc<Vec<u8>>)>, Vec<(String, String)>) {
128    let mut files = Vec::new();
129    let mut entries = Vec::new();
130    let Some(list) = doc.passthrough.get(LIST).and_then(|l| std::str::from_utf8(l).ok()) else { return (files, entries) };
131    let empty = std::collections::HashMap::new();
132    for line in list.lines().take(MAX_PARTS * 2) {
133        let (path, mime) = line.split_once('\t').unwrap_or((line, ""));
134        let dir = path.ends_with('/');
135        let name = path.trim_end_matches('/');
136        if !safe(name) || own(path, &empty) && !dir {
137            continue;
138        }
139        if dir {
140            entries.push((path.to_owned(), mime.to_owned()));
141        } else if let Some(b) = doc.passthrough.get(&format!("{PREFIX}{path}")) {
142            // Every file has its entry (ODF 1.3 Part 2 §4.3; LibreOffice refuses a package with a
143            // file its manifest doesn't list), its media type empty when the file's was.
144            files.push((path.to_owned(), b.clone()));
145            entries.push((path.to_owned(), mime.to_owned()));
146        }
147    }
148    (files, entries)
149}
