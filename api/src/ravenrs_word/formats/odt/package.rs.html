4/// `style:columns` (ODF 1.3 §17.12) of equal columns with their gap and, when on, the line between.
5fn columns_xml(c: &Columns) -> String {
6    let sep = if c.separator { "<style:column-sep style:width=\"0.5pt\" style:style=\"solid\" style:color=\"#000000\"/>" } else { "" };
7    let count = c.count.clamp(1, 45);
8    // Unequal columns (§17.12): each one's share of the width with the half gaps beside it as
9    // its indents, as LibreOffice writes them.
10    let mut each = String::new();
11    if c.widths.len() == count as usize && count > 1 {
12        let clean = |v: f32| if v.is_finite() { v.clamp(0.0, 1584.0) } else { 0.0 };
13        let mut start = 0.0;
14        for (w, after) in &c.widths {
15            let end = clean(*after) / 2.0;
16            let rel = ((clean(*w) + start + end) * 100.0).round().max(1.0) as u32;
17            each.push_str(&format!("<style:column style:rel-width=\"{rel}*\" fo:start-indent=\"{}\" fo:end-indent=\"{}\"/>", pt(start), pt(end)));
18            start = end;
19        }
20    }
21    // LibreOffice ignores a page's column widths beside a gap: the gaps are in the indents.
22    let gap = if each.is_empty() { format!(" fo:column-gap=\"{}\"", pt(c.space)) } else { String::new() };
23    format!("<style:columns fo:column-count=\"{count}\"{gap}>{sep}{each}</style:columns>")
24}

26/// One master page per entry of `pages` (name, setup), each with its own page layout `pm1`, `pm2`…
27/// (§16.5, §16.9); the first is the default one, "Standard".
28/// The styles WordCraft's paragraph kinds are written in (Heading 1–6, Title, Quotations,
29/// Preformatted Text), each with the look of the document's own style: font, size, bold,
30/// italic, colour and the rest of its text formatting, spacing, indents, alignment and keep with
31/// next, written out in full so other programs show them as WordCraft does.
32fn kind_styles(doc: &Document, on_outline: &std::collections::BTreeSet<String>) -> String {
33    let mut s = String::new();
34    let mut kinds: Vec<(String, String, String, Option<u8>, &str)> =
35        (1..=6).map(|n| (format!("Heading_20_{n}"), format!("Heading {n}"), format!("Heading{n}"), Some(n), "text")).collect();
36    kinds.push(("Title".into(), "Title".into(), "Title".into(), None, "chapter"));
37    kinds.push(("Quotations".into(), "Quotations".into(), "Quote".into(), None, "html"));
38    kinds.push(("Preformatted_20_Text".into(), "Preformatted Text".into(), model::CODE_STYLE.into(), None, "html"));
39    for (name, display, id, outline, class) in kinds {
40        let (pp, cp) = match doc.styles.get(&id) {
41            Some(_) => doc.styles.para_style_props(&id),
42            // Preformatted text in a document without its style: monospaced, no spacing.
43            None => (
44                crate::doc::props::ParaProps { space_before: Some(0.0), space_after: Some(0.0), ..Default::default() },
45                crate::doc::props::CharProps { font: Some(model::MONO_FONT.into()), size: Some(10.0), ..Default::default() },
46            ),
47        };
48        let mut para = spacing_props(&model::Spacing {
49            before: pp.space_before,
50            after: pp.space_after,
51            line: pp.line_spacing,
52            // Zero indents are the defaults; stating them would override the indents of the
53            // heading numbering in LibreOffice (its number then sits against the text).
54            left: pp.indent_left.filter(|v| *v != 0.0),
55            right: pp.indent_right.filter(|v| *v != 0.0),
56            first: pp.indent_first.filter(|v| *v != 0.0),
57            ..Default::default()
58        });
59        para.push_str(align_attr(pp.align));
60        if pp.keep_next == Some(true) {
61            para.push_str(" fo:keep-with-next=\"always\"");
62        }
63        if let Some(c) = pp.contextual_spacing {
64            para.push_str(&format!(" style:contextual-spacing=\"{c}\""));
65        }
66        let outline = outline.map(|n| outline_attrs(doc, n, &pp, on_outline.contains(&id))).unwrap_or_default();
67        s.push_str(&format!(
68            "<style:style style:name=\"{name}\" style:display-name=\"{display}\" style:family=\"paragraph\" style:parent-style-name=\"Standard\" style:next-style-name=\"Standard\"{outline} style:class=\"{class}\"><style:paragraph-properties{para}/><style:text-properties{}/></style:style>",
69            props_text(&cp)
70        ));
71    }
72    s
73}

75/// The styles of the headings in `blocks` (and their tables, to 16 levels deep) with a number by
76/// their own outline level.
77fn styles_numbered_by_own_level(blocks: &[std::sync::Arc<crate::doc::Block>], depth: usize, out: &mut std::collections::BTreeSet<String>) {
78    for b in blocks {
79        match &**b {
80            crate::doc::Block::Para(p) => {
81                if p.props.outline_level.is_some()
82                    && p.props.numbering.is_some_and(|n| n.num != 0)
83                    && let Some(s) = &p.props.style
84                {
85                    out.insert(s.clone());
86                }
87            }
88            crate::doc::Block::Table(t) if depth < 16 => {
89                for cell in t.rows.iter().flat_map(|r| r.cells.iter()) {
90                    styles_numbered_by_own_level(&cell.blocks, depth + 1, out);
91                }
92            }
93            crate::doc::Block::Table(_) => {}
94        }
95    }
96}

98/// A paragraph style's outline level `n` (`style:default-outline-level`), its properties being `pp`.
99fn outline_attrs(doc: &Document, n: u8, pp: &crate::doc::props::ParaProps, on_outline: bool) -> String {
100    // A level the outline doesn't number names no list style: LibreOffice would otherwise give
101    // it the outline's (unnumbered) level, and its indent. Unless a heading in the style has a
102    // number by its own level (`on_outline`), which that would take from it.
103    let numbered = pp
104        .numbering
105        .and_then(|r| doc.numbering.level(r.num, r.level))
106        .is_some_and(|l| !matches!(l.format, crate::doc::section::NumFormat::None | crate::doc::section::NumFormat::Bullet));
107    let detach = if numbered || on_outline { "" } else { " style:list-style-name=\"\"" };
108    format!(" style:default-outline-level=\"{n}\"{detach}")
109}

111/// The ODF parent of document style `id`: Standard for Normal, the kinds' styles for theirs, a
112/// custom style by its id.
113fn odf_parent(doc: &Document, id: Option<&str>) -> String {
114    match id {
115        None | Some("Normal") => "Standard".into(),
116        Some("Title") => "Title".into(),
117        Some("Quote") => "Quotations".into(),
118        Some(id) if id == model::CODE_STYLE => "Preformatted_20_Text".into(),
119        Some(id) => match id.strip_prefix("Heading").and_then(|n| n.parse::<u8>().ok()).filter(|n| (1..=9).contains(n)) {

121            None if model::is_custom_style(doc, id) => super::writer::odf_name(id),
122            None => "Standard".into(),
123        },
124    }
125}

127/// The custom paragraph styles in `named` and the custom styles they are based on, as common
128/// styles (ODF 1.3 §16.2): named by their document id, shown by their name, each with its own
129/// paragraph and text formatting over its parent's.
130fn custom_styles(doc: &Document, named: &std::collections::BTreeSet<String>, on_outline: &std::collections::BTreeSet<String>) -> String {
131    let mut todo: Vec<String> = named.iter().cloned().collect();
132    let mut done = std::collections::BTreeSet::new();
133    let mut s = String::new();
134    while let Some(id) = todo.pop() {
135        if !done.insert(id.clone()) || done.len() > 10_000 {
136            continue;
137        }
138        let Some(st) = doc.styles.get(&id) else { continue };
139        if let Some(b) = st.based_on.as_deref().filter(|b| model::is_custom_style(doc, b)) {
140            todo.push(b.to_string());
141        }
142        let pp = &st.para;
143        let mut attrs = spacing_props(&model::Spacing {
144            before: pp.space_before,
145            after: pp.space_after,
146            line: pp.line_spacing,
147            left: pp.indent_left,
148            right: pp.indent_right,
149            first: pp.indent_first,
150            ..Default::default()
151        });
152        attrs.push_str(align_attr(pp.align));
153        if pp.keep_next == Some(true) {
154            attrs.push_str(" fo:keep-with-next=\"always\"");
155        }
156        if let Some(c) = pp.contextual_spacing {
157            attrs.push_str(&format!(" style:contextual-spacing=\"{c}\""));
158        }
159        if pp.bidi == Some(true) {
160            attrs.push_str(" style:writing-mode=\"rl-tb\"");
161        }
162        attrs.push_str(&super::writer::border_attrs(pp.borders));
163        if let Some(bg) = pp.shading {
164            attrs.push_str(&format!(" fo:background-color=\"#{}\"", bg.hex().to_ascii_lowercase()));
165        }
166        let tabs = pp
167            .tabs
168            .as_deref()
169            .map(|tabs| {
170                // From the style's indent where the document measures them so (as they are read).
171                let indent = if model::tabs_from_indent(doc) { pp.indent_left.unwrap_or(0.0) } else { 0.0 };
172                let tabs: Vec<_> = tabs.iter().map(|t| crate::doc::props::TabStop { pos: (t.pos - indent).clamp(-1584.0, 1584.0), ..*t }).collect();
173                super::writer::tab_stops(&tabs)
174            })
175            .unwrap_or_default();
176        let parent = odf_parent(doc, st.based_on.as_deref());
177        let outline = pp.outline_level.filter(|l| *l < 9).map(|l| outline_attrs(doc, l + 1, pp, on_outline.contains(&id))).unwrap_or_default();
178        s.push_str(&format!(
179            "<style:style style:name=\"{}\" style:display-name=\"{}\" style:family=\"paragraph\" style:parent-style-name=\"{parent}\"{outline}><style:paragraph-properties{attrs}>{tabs}</style:paragraph-properties><style:text-properties{}/></style:style>",
180            super::writer::odf_name(&id),
181            x(&super::writer::generated_style(&id).map_or_else(|| st.name.clone(), |(_, d)| d)),
182            props_text(&with_complex_size(doc, &id, &st.chr))
183        ));
184    }
185    s
186}

189/// states only a size: a parent's complex size where one sets it (the model takes it from there,

227/// `text:outline-style` (ODF 1.3 §16.34): the numbering of Heading 1–6, from the list level each
228/// heading style numbers its paragraphs with; levels without one show no number.
229fn outline_style(doc: &Document) -> String {
230    let mut s = String::from("<text:outline-style style:name=\"Outline\">");
231    for n in 1..=10u8 {
232        let level = (n <= 6)
233            .then(|| doc.styles.para_style_props(&format!("Heading{n}")).0.numbering)
234            .flatten()
235            .and_then(|r| doc.numbering.level(r.num, r.level).cloned().map(|l| (r.level, l)));
236        let (indent, hanging) = level.as_ref().map(|(_, l)| (l.indent, l.hanging)).unwrap_or((0.0, 0.0));
237        let attrs = match level {
238            Some((lv, l)) if !matches!(l.format, crate::doc::section::NumFormat::None | crate::doc::section::NumFormat::Bullet) => {
239                let label = model::Label::of_level(l.format, &l.text);
240                let mut a = format!(" style:num-format=\"{}\"", super::writer::odf_num_format(l.format));
241                for (k, v) in [("style:num-prefix", label.prefix.as_str()), ("style:num-suffix", label.suffix.as_str())] {
242                    if !v.is_empty() {
243                        a.push_str(&format!(" {k}=\"{}\"", x(v)));
244                    }
245                }
246                if label.shown > 1 {
247                    a.push_str(&format!(" text:display-levels=\"{}\"", label.shown.min(lv.saturating_add(1))));
248                }
249                if l.start != 1 {
250                    a.push_str(&format!(" text:start-value=\"{}\"", l.start));
251                }
252                a
253            }
254            _ => " style:num-format=\"\"".into(),
255        };
256        // The number at the level's indent, followed by a tab (to the next default tab stop
257        // when the level has no indent), as Word lays out numbered headings.
258        let fit = |v: f32| if v.is_finite() { v.clamp(0.0, 1584.0) } else { 0.0 };
259        // A level with no indent states none, so LibreOffice uses its own (a tab after the number).
260        let place = if fit(indent) > 0.0 || fit(hanging) > 0.0 {
261            format!(
262                " text:list-tab-stop-position=\"{}\" fo:text-indent=\"{}\" fo:margin-left=\"{}\"",
263                pt(fit(indent)),
264                pt(-fit(hanging)),
265                pt(fit(indent))
266            )
267        } else {
268            String::new()
269        };
270        s.push_str(&format!(
271            "<text:outline-level-style text:level=\"{n}\"{attrs}><style:list-level-properties text:list-level-position-and-space-mode=\"label-alignment\"><style:list-level-label-alignment text:label-followed-by=\"listtab\"{place}/></style:list-level-properties></text:outline-level-style>"
272        ));
273    }
274    s.push_str("</text:outline-style>");
275    s
276}

287/// The headers and footers section `sec` shows, written by `w` (its styles go in `styles.xml`):
288/// the default one for every page, the even one as left pages' when the document has different
289/// odd and even pages, the first one with a different first page. A slot Word leaves empty is an
290/// empty element, so the page shows nothing there rather than the default one.
291fn page_parts(doc: &Document, sec: &SectionProps, w: &mut Writer) -> PageParts {
292    let mut out = PageParts::default();
293    let even = doc.settings.even_odd_headers;
294    // A watermark is a shape in every header (one written for it alone when there is none).
295    let wm = doc.settings.watermark.as_ref().filter(|w| !w.text.trim().is_empty());
296    let area = (sec.page_w - sec.margin_left - sec.margin_right, sec.page_h - sec.margin_top - sec.margin_bottom);
297    for (set, name) in [(&sec.headers, "header"), (&sec.footers, "footer")] {
298        let wanted =
299            set.default.is_some() || (even && set.even.is_some()) || (sec.title_page && set.first.is_some()) || (name == "header" && wm.is_some());
300        if !wanted {
301            continue;
302        }
303        for (id, el, on) in
304            [(set.default, name.to_string(), true), (set.even, format!("{name}-left"), even), (set.first, format!("{name}-first"), sec.title_page)]
305        {
306            if !on {
307                continue;
308            }
309            let mut body = String::new();
310            if let Some(part) = id.and_then(|i| doc.parts.get(&i)) {
311                let blocks = model::part_flow(
312                    doc,
313                    &part.blocks,
314                    // Pictures, shapes, text boxes, charts and objects too, as in the body.
315                    model::FlowOpts { fields: true, lists: true, chars: true, paras: true, boxes: true, ..Default::default() },
316                );
317                w.blocks(&blocks, &mut body, 0);
318            }
319            if body.is_empty() {
320                body.push_str("<text:p/>");
321            }
322            if name == "header"
323                && let Some(wm) = wm
324            {
325                body = with_watermark(&body, &w.watermark(wm, (sec.page_w, sec.page_h), area));
326            }
327            out.xml.push_str(&format!("<style:{el}>{body}</style:{el}>"));
328        }
329        if name == "header" {
330            out.header = true;
331        } else {
332            out.footer = true;
333        }
334    }
335    out
336}

338/// A header's content with `shape` (anchored to a character) at the start of its first
339/// paragraph, or in a paragraph of its own before a table that starts it.
340fn with_watermark(body: &str, shape: &str) -> String {
341    if let Some(rest) = body.strip_prefix("<text:p/>") {
342        return format!("<text:p>{shape}</text:p>{rest}");
343    }
344    if body.starts_with("<text:p") || body.starts_with("<text:h") {
345        // Into the first paragraph's element, after its start tag (or in place of an empty one).
346        if let Some(end) = body.find('>') {
347            let (tag, rest) = body.split_at(end + 1);
348            if let Some(open) = tag.strip_suffix("/>") {
349                let name = if tag.starts_with("<text:h") { "text:h" } else { "text:p" };
350                return format!("{open}>{shape}</{name}>{rest}");
351            }
352            return format!("{tag}{shape}{rest}");
353        }
354    }
355    format!("<text:p>{shape}</text:p>{body}")
356}

358/// `styles.xml` of `doc`: the default style/// `styles.xml` of `doc`: the default style from the document defaults (Normal's spacing, the
359/// body font, pair kerning, the tab interval), the paragraph kinds' styles, and one master page
360/// per entry of `pages` with `parts`, its headers and footers, whose automatic styles are
361/// `part_styles`.
362fn styles_xml(
363    doc: &Document,
364    pages: &[(String, crate::doc::SectionProps)],
365    parts: &[PageParts],
366    part_styles: &str,
367    named: &std::collections::BTreeSet<String>,
368) -> String {
369    // WordCraft's page layouts take another prefix next to the file's own (see `super::kept`).
370    let pm = if doc.kept.of("odt").is_some() { "wcpm" } else { "pm" };
371    // The document defaults: Normal's font, spacing and indents, pair kerning and tab interval.
372    let body_font = doc.styles.default_chr.font.clone().unwrap_or_else(|| crate::doc::styles::BODY_FONT.to_string());
373    let normal = doc.styles.resolve_para(&Default::default());
374    let defaults = &model::Spacing {
375        before: Some(normal.space_before),
376        after: Some(normal.space_after),
377        line: Some(normal.line_spacing),
378        left: Some(normal.indent_left),
379        right: Some(normal.indent_right),
380        first: Some(normal.indent_first),
381        ..Default::default()
382    };
383    let kern = doc.styles.default_chr.kern.is_some_and(|k| k > 0.0);
384    // Normal's font size (Standard, written empty, takes the default style's).
385    let size = doc.styles.resolve_char(None, &Default::default()).size;
386    let size = if size.is_finite() { size.clamp(1.0, 1638.0) } else { 12.0 };
387    let tab = doc.settings.default_tab.clamp(1.0, 1584.0);
388    let mut s = format!("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:document-styles {NS}><office:styles>");
389    s.push_str(&format!(
390        "<style:default-style style:family=\"paragraph\"><style:paragraph-properties{}{widows} style:tab-stop-distance=\"{tab}\"/><style:text-properties fo:font-family=\"'{}'\" fo:font-size=\"{size}\" style:font-size-asian=\"{size}\"{size_cs} style:letter-kerning=\"{kern}\"{color}{lang}{scripts}/></style:default-style>",
391        spacing_props(defaults),
392        x(&body_font),
393        // Widow and orphan control as Normal has it, said either way (the standard leaves it open
394        // when unsaid, and readers differ: docs/odf-interpretation.md).
395        widows = if normal.widow_control { " fo:widows=\"2\" fo:orphans=\"2\"" } else { " fo:widows=\"0\" fo:orphans=\"0\"" },
396        size = pt(size),
397        size_cs = if doc.styles.default_chr.size_cs.is_none() { format!(" style:font-size-complex=\"{}\"", pt(size)) } else { String::new() },
398        // The document's text colour (a colour, not automatic).
399        color = match doc.styles.default_chr.color {
400            Some(crate::doc::props::TextColor::Rgb(c)) => format!(" fo:color=\"#{}\"", c.hex().to_ascii_lowercase()),
401            _ => String::new(),
402        },
403        tab = pt(tab),
404        // The document's language, without which each reader uses its own default: Normal's, as
405        // WordCraft has it (a Word file's Normal may state another than its defaults).
406        lang = doc.styles.resolve_char(None, &Default::default()).lang.as_deref().map(|l| super::writer::lang_attrs(l, "")).unwrap_or_default(),
407        scripts = super::writer::script_attrs(&doc.styles.default_chr)
408    ));
409    s.push_str("<style:style style:name=\"Standard\" style:family=\"paragraph\" style:class=\"text\"/>");
410    let mut on_outline = std::collections::BTreeSet::new();
411    styles_numbered_by_own_level(&doc.body, 0, &mut on_outline);
412    s.push_str(&kind_styles(doc, &on_outline));
413    s.push_str(&outline_style(doc));
414    s.push_str(&custom_styles(doc, named, &on_outline));
415    // The note text styles, as WordCraft's Footnote Text and Endnote Text: 10 pt, no spacing.
416    for (name, class) in [("Footnote", "extra"), ("Endnote", "extra")] {
417        s.push_str(&format!(
418            "<style:style style:name=\"{name}\" style:family=\"paragraph\" style:parent-style-name=\"Standard\" style:class=\"{class}\"><style:paragraph-properties fo:margin-top=\"0pt\" fo:margin-bottom=\"0pt\" fo:line-height=\"100%\"/><style:text-properties fo:font-size=\"10pt\"/></style:style>"
419        ));
420    }
421    s.push_str(&notes_configuration(doc));
422    if let Some(l) = pages.first().and_then(|(_, sec)| sec.line_numbers.as_ref()) {
423        s.push_str(&line_numbering_xml(l));
424    }
425    s.push_str("</office:styles><office:automatic-styles>");
426    s.push_str(part_styles);
427    let none = PageParts::default();
428    for (i, (_, sec)) in pages.iter().enumerate() {
429        let pp = parts.get(i).unwrap_or(&none);
430        // A page of several columns carries them in its layout.
431        let cols = if sec.columns.count > 1 { columns_xml(&sec.columns) } else { String::new() };
432        // Word measures the header from the page edge and the text from the page edge; ODF
433        // starts the header at the page margin and the text after the header's height. So with
434        // a header the page margin is the header's distance, and the header is tall enough to
435        // reach the text (it grows with its content, as Word's pushes the text down).
436        let (top, header_h) = if pp.header { (sec.header, (sec.margin_top - sec.header).max(0.0)) } else { (sec.margin_top, 0.0) };
437        let (bottom, footer_h) = if pp.footer { (sec.footer, (sec.margin_bottom - sec.footer).max(0.0)) } else { (sec.margin_bottom, 0.0) };
438        // A page border is at ODF's margin, the text (or header, footer) inside its padding.
439        let (borders, [top, bottom, left, right]) = page_borders(sec.page_borders, [top, bottom, sec.margin_left, sec.margin_right]);
440        // The band's height takes in its spacing (see the reader), which content taller than
441        // the band keeps below it.
442        let gap = |g: f32, h: f32| if g.is_finite() { g.clamp(0.0, h) } else { 0.0 };
443        // Spacing kept below a taller header says so, as the standard leaves the default open.
444        let keeps = |g: f32| if g.is_finite() && g > 0.0 { " style:dynamic-spacing=\"false\"" } else { "" };
445        let header = if pp.header {
446            format!(
447                "<style:header-style><style:header-footer-properties fo:min-height=\"{}\" fo:margin-bottom=\"{}\"{}/></style:header-style>",
448                pt(header_h),
449                pt(gap(sec.header_gap, header_h)),
450                keeps(sec.header_gap)
451            )
452        } else {
453            String::new()
454        };
455        let footer = if pp.footer {
456            format!(
457                "<style:footer-style><style:header-footer-properties fo:min-height=\"{}\" fo:margin-top=\"{}\"{}/></style:footer-style>",
458                pt(footer_h),
459                pt(gap(sec.footer_gap, footer_h)),
460                keeps(sec.footer_gap)
461            )
462        } else {
463            String::new()
464        };
465        s.push_str(&format!(
466            "<style:page-layout style:name=\"{pm}{}\"><style:page-layout-properties fo:page-width=\"{}\" fo:page-height=\"{}\" fo:margin-top=\"{}\" fo:margin-bottom=\"{}\" fo:margin-left=\"{}\" fo:margin-right=\"{}\" style:print-orientation=\"{}\"{}>{cols}</style:page-layout-properties>{header}{footer}</style:page-layout>",
467            i + 1,
468            pt(sec.page_w),
469            pt(sec.page_h),
470            pt(top),
471            pt(bottom),
472            pt(left),
473            pt(right),
474            if sec.page_w > sec.page_h { "landscape" } else { "portrait" },
475            page_look(doc, sec) + &borders
476        ));
477    }
478    s.push_str("</office:automatic-styles><office:master-styles>");
479    for (i, (name, _)) in pages.iter().enumerate() {
480        let xml = parts.get(i).map(|p| p.xml.as_str()).unwrap_or("");
481        s.push_str(&format!("<style:master-page style:name=\"{}\" style:page-layout-name=\"{pm}{}\">{xml}</style:master-page>", x(name), i + 1));
482    }
483    s.push_str("</office:master-styles></office:document-styles>\n");
484    s
485}

620/// `settings.xml` (ODF 1.3 §3.10) for `doc`: an ODT's own, kept from opening it, or LibreOffice's
621/// switches for Word's layout. Either way tab stops are measured from the margin, as WordCraft
622/// writes them (LibreOffice's own default measures them from the paragraph's indent).
623/// `settings.xml`: LibreOffice's layout settings for a document laid out as WordCraft lays it
624/// out. The set's name, `ooo:configuration-settings`, is a qualified name: without the `ooo`
625/// namespace declared LibreOffice ignores the whole set (observed).
626fn settings_xml(doc: &Document) -> String {
627    // What WordCraft's own layout of the document needs LibreOffice to do the same: tab stops
628    // from the margin (as written), and the space between paragraphs as WordCraft lays it out
629    // (ODF leaves it open; LibreOffice's `AddParaTableSpacing` false takes the larger of the two).
630    let ours = [("TabsRelativeToIndent", model::tabs_from_indent(doc)), ("AddParaTableSpacing", doc.settings.sum_paragraph_spacing)];
631    let item = |name: &str, on: bool| format!("<config:config-item config:name=\"{name}\" config:type=\"boolean\">{on}</config:config-item>");
632    if let Some(own) = doc.passthrough.get(SETTINGS_PART).filter(|b| b.len() <= MAX_SETTINGS).and_then(|b| std::str::from_utf8(b).ok()) {
633        let mut out = own.to_string();
634        let mut ok = true;
635        for (name, on) in ours {
636            match with_setting(&out, name, on, &item(name, on)) {
637                Some(next) => out = next,
638                None => ok = false,
639            }
640        }
641        if ok {
642            return out;
643        }
644    }
645    let mut s = String::from(
646        "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:document-settings xmlns:office=\"urn:oasis:names:tc:opendocument:xmlns:office:1.0\" xmlns:config=\"urn:oasis:names:tc:opendocument:xmlns:config:1.0\" xmlns:ooo=\"http://openoffice.org/2004/office\" office:version=\"1.3\"><office:settings><config:config-item-set config:name=\"ooo:configuration-settings\">",
647    );
648    // LibreOffice's defaults; Word's layout rules for a document WordCraft lays out as Word does
649    // (one opened from ODF is laid out by ODF's rules, `Settings::odf`, as LibreOffice's
650    // defaults have it); and what WordCraft's layout of this document needs.
651    let mut all: Vec<(&str, bool)> = LO_DEFAULTS.to_vec();
652    let word = if doc.settings.odf.is_none() { WORD_COMPAT } else { &[] };
653    for (name, on) in word.iter().copied().chain(ours) {
654        match all.iter_mut().find(|(n, _)| *n == name) {
655            Some(slot) => slot.1 = on,
656            None => all.push((name, on)),
657        }
658    }
659    for (name, on) in all {
660        s.push_str(&item(name, on));
661    }
662    // Metrics independent of any printer, as LibreOffice's own documents.
663    s.push_str("<config:config-item config:name=\"PrinterIndependentLayout\" config:type=\"string\">high-resolution</config:config-item>");
664    s.push_str("</config:config-item-set></office:settings></office:document-settings>\n");
665    s
666}

668/// A file's own `settings.xml` (`own`) with LibreOffice's setting `name` made `on`: its item's
669/// value changed, or `item` added at the start of the configuration settings. `None` when it has
670/// neither.
671fn with_setting(own: &str, name: &str, on: bool, item: &str) -> Option<String> {
672    let key = format!("config:name=\"{name}\"");
673    if let Some(at) = own.find(&key) {
674        let rest = own.get(at..)?;
675        let gt = rest.find('>')?;
676        // A self-closed item (another application's, or a hand-edited file's) is replaced whole:
677        // its close tag would be the next item's.
678        if rest.get(..gt).is_some_and(|t| t.ends_with('/')) {
679            let start = own.get(..at)?.rfind('<')?;
680            return Some(format!("{}{item}{}", own.get(..start).unwrap_or(""), own.get(at + gt + 1..).unwrap_or("")));
681        }
682        let lt = rest.find("</config:config-item>")?;
683        return (gt < lt).then(|| format!("{}{on}{}", own.get(..at + gt + 1).unwrap_or(""), own.get(at + lt..).unwrap_or("")));
684    }
685    let set = "config:name=\"ooo:configuration-settings\"";
686    let at = own.find(set).and_then(|at| own.get(at..)?.find('>').map(|gt| at + gt + 1))?;
687    Some(format!("{}{item}{}", own.get(..at).unwrap_or(""), own.get(at..).unwrap_or("")))
688}

690fn meta_xml(m: &Meta, core: &crate::doc::CoreProps, props: &[crate::doc::CustomProp]) -> String {
691    let mut s =
692        format!("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:document-meta {NS}><office:meta><meta:generator>WordCraft</meta:generator>");
693    if !m.title.is_empty() {
694        s.push_str(&format!("<dc:title>{}</dc:title>", x(&m.title)));
695    }
696    if !m.subject.is_empty() {
697        s.push_str(&format!("<dc:subject>{}</dc:subject>", x(&m.subject)));
698    }
699    if !m.description.is_empty() {
700        s.push_str(&format!("<dc:description>{}</dc:description>", x(&m.description)));
701    }
702    // Who made it, and who changed it last (§4.3.2.6 `dc:creator`, §4.3.2.10 `meta:initial-creator`).
703    let last_by = if core.last_modified_by.trim().is_empty() { m.author.as_str() } else { core.last_modified_by.as_str() };
704    if !m.author.is_empty() {
705        s.push_str(&format!("<meta:initial-creator>{}</meta:initial-creator>", x(&m.author)));
706    }
707    if !last_by.is_empty() {
708        s.push_str(&format!("<dc:creator>{}</dc:creator>", x(last_by)));
709    }
710    // When it was last changed (§4.3.2.7 `dc:date`), and how often it was saved (§4.3.2.12
711    // `meta:editing-cycles`).
712    let modified = core.modified.trim().trim_end_matches(['Z', 'z']);
713    if !modified.is_empty() && modified.len() < 64 {
714        s.push_str(&format!("<dc:date>{}</dc:date>", x(modified)));
715    }
716    if core.revision > 0 {
717        s.push_str(&format!("<meta:editing-cycles>{}</meta:editing-cycles>", core.revision));
718    }
719    for k in m.keywords.split([',', ';']).map(str::trim).filter(|k| !k.is_empty()) {
720        s.push_str(&format!("<meta:keyword>{}</meta:keyword>", x(k)));
721    }
722    if !m.created.is_empty() {
723        s.push_str(&format!("<meta:creation-date>{}</meta:creation-date>", x(m.created.trim_end_matches('Z'))));
724    }
725    // Custom document properties (§4.3.2.14); ones kept as raw XML from a Word file aren't.
726    // (User fields go in the content instead: `user_field_decls`.)
727    for p in props.iter().filter(|p| p.kind != "raw" && !p.odf_user_field) {
728        let (t, value) = odf_value(p);
729        s.push_str(&format!("<meta:user-defined meta:name=\"{}\" meta:value-type=\"{t}\">{}</meta:user-defined>", x(&p.name), x(&value)));
730    }
731    s.push_str("</office:meta></office:document-meta>\n");
732    s
733}

735/// A custom property's ODF value type and value.
736fn odf_value(p: &crate::doc::CustomProp) -> (&'static str, String) {
737    let t = match p.kind.as_str() {
738        "r4" | "r8" | "i1" | "i2" | "i4" | "i8" | "int" | "ui1" | "ui2" | "ui4" | "ui8" | "uint" | "decimal" | "cy" => "float",
739        "bool" => "boolean",
740        "filetime" | "date" => "date",
741        _ => "string",
742    };
743    let value = if t == "boolean" { if matches!(p.value.trim(), "true" | "1") { "true" } else { "false" }.to_string() } else { p.value.clone() };
744    (t, value)
745}

747/// The document's user fields (§7.4.7 `text:user-field-decls`), declared at the start of the text.
748fn user_field_decls(props: &[crate::doc::CustomProp]) -> String {
749    let decls: String = props
750        .iter()
751        .filter(|p| p.odf_user_field && p.kind != "raw")
752        .map(|p| {
753            let (t, value) = odf_value(p);
754            let attr = match t {
755                "float" => "office:value",
756                "boolean" => "office:boolean-value",
757                "date" => "office:date-value",

762        })
763        .collect();

765}

768/// carries a section's properties ends it; the document's last section follows the last such

783    }

787/// Write an `.odt` package.
788/// The kinds of the shapes with drawing ids in `blocks` (paragraphs, tables, groups), by id.
789fn conn_kinds(blocks: &[FBlock]) -> super::HashMap<u32, crate::doc::para::ShapeKind> {
790    fn walk(blocks: &[FBlock], depth: usize, out: &mut super::HashMap<u32, crate::doc::para::ShapeKind>) {
791        if depth > model::MAX_DEPTH {
792            return;
793        }
794        for b in blocks {
795            match b {
796                FBlock::Para(p) => p.inlines.iter().for_each(|i| inline(i, 0, out)),
797                FBlock::Table(t) => t.rows.iter().flatten().for_each(|c| walk(&c.blocks, depth + 1, out)),
798            }
799        }
800    }
801    fn inline(i: &super::Inline, depth: usize, out: &mut super::HashMap<u32, crate::doc::para::ShapeKind>) {
802        match i {
803            super::Inline::Shape { kind, look, .. } => {
804                if let Some(c) = look.conn.as_deref().filter(|c| c.id != 0) {
805                    out.insert(c.id, *kind);
806                }
807            }
808            super::Inline::Group { members, .. } if depth < model::MAX_GROUP_DEPTH => members.iter().for_each(|m| inline(&m.inline, depth + 1, out)),
809            _ => {}
810        }
811    }
812    let mut out = super::HashMap::new();
813    walk(blocks, 0, &mut out);
814    out
815}

830/// The package files the file's own definitions refer to, and those the markup written as read
831/// links to (`used`), written as read with them (added to `files` and `manifest` where no file
832/// `taken` has the path). A picture only an edited paragraph showed isn't written.
833fn with_kept_files(
834    kept: Option<&crate::doc::kept::KeptXml>,
835    used: &std::collections::BTreeSet<String>,
836    taken: &dyn Fn(&str) -> bool,
837    files: &mut Vec<(String, std::sync::Arc<Vec<u8>>)>,
838    manifest: &mut String,
839) {
840    let Some(k) = kept else { return };
841    let mut wanted = used.clone();
842    for d in k.defs.values() {
843        super::kept::hrefs(d, &mut wanted);
844    }
845    // A file linked, or one in a folder linked (an object's).
846    let linked = |p: &str| wanted.contains(p) || wanted.iter().any(|w| p.strip_prefix(w.trim_end_matches('/')).is_some_and(|r| r.starts_with('/')));
847    let entry = |manifest: &mut String, path: &str, mime: &str| {
848        let e = format!("manifest:full-path=\"{}\"", x(path));
849        if !manifest.contains(&e) {
850            manifest.push_str(&format!("<manifest:file-entry {e} manifest:media-type=\"{}\"/>\n", x(mime)));
851        }
852    };
853    for (path, bytes) in &k.files {
854        if linked(path) && !taken(path) && !files.iter().any(|(p, _)| p == path) {
855            // As the file's manifest had it (the folder too), else by its extension.
856            if let Some((dir, _)) = path.rsplit_once('/')
857                && let Some(mime) = k.defs.get(&format!("manifest:{dir}/"))
858            {
859                entry(manifest, &format!("{dir}/"), mime);
860            }
861            let ext = path.rsplit('.').next().unwrap_or("");
862            let mime = k.defs.get(&format!("manifest:{path}")).map_or_else(|| model::mime_of(ext).to_string(), |m| m.to_string());
863            entry(manifest, path, &mime);
864            files.push((path.clone(), bytes.clone()));
865        }
866    }
867}

889                        let e = seen.entry(id).or_insert((0, size));

891                    }
892                }
893                FBlock::Table(t) if depth < model::MAX_DEPTH => {
894                    for c in t.rows.iter().flatten() {
895                        count(&c.blocks, depth + 1, seen);
896                    }
897                }
898                FBlock::Table(_) => {}
899            }
900        }
901    }
902    let mut seen = std::collections::HashMap::new();
903    count(blocks, 0, &mut seen);
904    seen.into_iter().filter(|(_, (n, size))| n == size).map(|(id, _)| id).collect()
905}

908/// read (theirs keep the names they were read with).

911/// `content` and `styles` with the definitions kept paragraphs need (`used`: automatic, common,
912/// font faces), each where it goes and only where its name isn't defined already.
913fn with_kept_defs(
914    mut content: String,
915    mut styles: String,
916    used: &super::kept::Needed,
917    defs: &std::collections::BTreeMap<String, std::sync::Arc<str>>,
918) -> (String, String) {
919    let (auto, common, fonts) = used;
920    if auto.is_empty() && common.is_empty() && fonts.is_empty() && defs.is_empty() {
921        return (content, styles);
922    }
923    // The namespaces the file declared (kept markup may use any), where they aren't declared.
924    for (xml, root) in [(&mut content, "<office:document-content "), (&mut styles, "<office:document-styles ")] {
925        let Some(at) = xml.find(root).map(|i| i + root.len()) else { continue };
926        let head_end = xml.get(at..).and_then(|r| r.find('>')).map_or(at, |e| at + e);
927        let head = xml.get(at..head_end).unwrap_or("").to_string();
928        let missing: String = defs
929            .iter()
930            .filter_map(|(k, uri)| k.strip_prefix("ns:").map(|p| (p, uri)))
931            .filter(|(p, _)| !head.contains(&format!("xmlns:{p}=")))
932            .map(|(p, uri)| format!("xmlns:{p}=\"{}\" ", x(uri)))
933            .collect();
934        xml.insert_str(at, &missing);
935    }
936    let add = |xml: &mut String, close: &str, defs: &std::collections::BTreeMap<String, std::sync::Arc<str>>| {
937        let missing: String = defs.iter().filter(|(_, d)| !super::kept::defined(xml, d)).map(|(_, d)| &**d).collect();
938        if missing.is_empty() {
939            return;
940        }
941        let empty = close.replacen("</", "<", 1).replacen('>', "/>", 1);
942        if let Some(at) = xml.find(close) {
943            xml.insert_str(at, &missing);
944        } else if let Some(at) = xml.find(&empty) {
945            xml.replace_range(at..at + empty.len(), &format!("{}{missing}{close}", close.replacen("</", "<", 1)));
946        }
947    };
948    add(&mut content, "</office:automatic-styles>", auto);
949    add(&mut styles, "</office:styles>", common);
950    // A font the file declares is declared as the file does: a reader without the font picks its
951    // substitute by this, and a different one would reflow every page.
952    let faces: Vec<&str> = defs.iter().filter(|(k, _)| k.starts_with("font:")).map(|(_, d)| &**d).collect();
953    with_file_faces(&mut content, &faces);
954    with_file_faces(&mut styles, &faces);
955    for (xml, before) in [(&mut content, "<office:automatic-styles"), (&mut styles, "<office:styles")] {
956        // No declarations yet: the element they go in, where it belongs (before the styles).
957        if !fonts.is_empty()
958            && !xml.contains("<office:font-face-decls")
959            && let Some(at) = xml.find(before)
960        {
961            xml.insert_str(at, "<office:font-face-decls></office:font-face-decls>");
962        }
963        add(xml, "</office:font-face-decls>", fonts);
964    }
965    (content, styles)
966}

968/// `xml` (a part WordCraft wrote) with each font it declares that `faces` declare too (the file's
969/// own `style:font-face` elements) as `faces` has it: under that name, or else of that family (a
970/// face named `F1` of the family `Mystery`, which WordCraft names by its family), then keeping
971/// WordCraft's name for it.
972fn with_file_faces(xml: &mut String, faces: &[&str]) {
973    use super::kept::{attr_text, raw_attr};
974    let mut named = std::collections::HashMap::new();
975    let mut of_family = std::collections::HashMap::new();
976    for face in faces {
977        let Some(head) = face.split('>').next() else { continue };
978        if let Some(name) = raw_attr(head, "style:name") {
979            named.entry(attr_text(name).into_owned()).or_insert(*face);
980        }
981        if let Some(family) = raw_attr(head, "svg:font-family") {
982            let family = attr_text(family);
983            let first = family.split(',').next().unwrap_or("").trim().trim_matches(['\'', '"']);
984            if let Some(font) = model::kept_font(first) {
985                of_family.entry(font).or_insert(*face);
986            }
987        }
988    }
989    // WordCraft's own declarations are empty elements.
990    let (kids, _) = super::kept::children(xml, "office:font-face-decls", "font");
991    let mut out = String::with_capacity(xml.len());
992    let mut done = 0;
993    for (_, (from, to)) in kids {
994        let (Some(before), Some(tag)) = (xml.get(done..from), xml.get(from..to)) else { continue };
995        out.push_str(before);
996        let name = raw_attr(tag, "style:name").map(attr_text);
997        let same = name.as_deref().and_then(|n| named.get(n)).map(|face| face.to_string());
998        let family = || name.as_deref().and_then(|n| of_family.get(n).and_then(|face| face_named(face, n)));
999        let empty = tag.find('>').is_some_and(|end| end + 1 == tag.len());
1000        out.push_str(same.or_else(family).filter(|_| empty).as_deref().unwrap_or(tag));
1001        done = to;
1002    }
1003    out.push_str(xml.get(done..).unwrap_or(""));
1004    *xml = out;
1005}

1007/// `face` (a `style:font-face` element) with the name `name`.
1008fn face_named(face: &str, name: &str) -> Option<String> {
1009    let head = face.split('>').next()?;
1010    let was = super::kept::raw_attr(head, "style:name")?;

1014}

1302/// Page borders `b` as ODF writes them, and the margins (top, bottom, left, right) of the page
1303/// they go round: each border at its distance from the edge (WordCraft's), which becomes the
1304/// margin, with the rest of the way to the text as its padding.
1305fn page_borders(b: Option<crate::doc::props::Borders>, margins: [f32; 4]) -> (String, [f32; 4]) {
1306    let Some(b) = b.filter(|b| b.any_visible()) else { return (String::new(), margins) };
1307    let mut out = margins;
1308    let mut attrs = String::new();
1309    for (i, (side, edge)) in [("top", b.top), ("bottom", b.bottom), ("left", b.left), ("right", b.right)].into_iter().enumerate() {
1310        let (Some(e), Some(m)) = (edge.filter(|e| e.is_visible()), out.get_mut(i)) else { continue };
1311        // A hostile margin (negative or not a number) leaves no room, so clamp cannot panic.
1312        let room = if m.is_finite() { m.max(0.0) } else { 0.0 };
1313        let at = if e.space.is_finite() { e.space.clamp(0.0, room) } else { 0.0 };
1314        let pad = (*m - at - e.width).max(0.0);
1315        *m = at;
1316        attrs.push_str(&format!(" fo:border-{side}=\"{}\" fo:padding-{side}=\"{}\"", super::writer::odf_border(Some(e)), pt(pad)));
1317    }
1318    (attrs, out)
1319}

1396/// `styles` with the page background picture (Design › Page Color › Fill Effects › Picture) in
1397/// each page layout (§17.3 `style:background-image`, first in `style:page-layout-properties`),
1398/// the picture written into the package by `w`: tiled, stretched, or centred at its own size
1399/// (§20.249 `style:repeat`).
1400fn page_background(doc: &Document, w: &mut Writer, styles: String) -> String {
1401    let Some(pic) = doc.settings.page_picture.as_ref() else { return styles };
1402    let Some(data) = doc.media.get(&pic.media) else { return styles };
1403    let Some(ext) = model::sniff_image(data) else { return styles };
1404    let name = format!("Pictures/page-background.{ext}");
1405    if !w.pictures.iter().any(|(n, _, _)| *n == name) {
1406        w.pictures.push((name.clone(), data.clone(), model::mime_of(ext).to_string()));
1407    }
1408    let repeat = match pic.fill_type.to_ascii_lowercase().as_str() {
1409        "tile" => "repeat",
1410        "center" => "no-repeat",
1411        _ => "stretch",
1412    };
1413    let el = format!(
1414        "<style:background-image xlink:href=\"{}\" xlink:type=\"simple\" xlink:actuate=\"onLoad\" style:position=\"center center\" style:repeat=\"{repeat}\"/>",
1415        x(&name)
1416    );
1417    // After each layout's properties' opening tag.
1418    let mut out = String::with_capacity(styles.len() + el.len() * 4);
1419    let mut rest = styles.as_str();
1420    while let Some(at) = rest.find("<style:page-layout-properties") {
1421        let Some(end) = rest.get(at..).and_then(|r| r.find('>')).map(|e| at + e) else { break };
1422        let self_closing = rest.get(end.saturating_sub(1)..end) == Some("/");
1423        if self_closing {
1424            out.push_str(rest.get(..end.saturating_sub(1)).unwrap_or_default());
1425            out.push('>');
1426            out.push_str(&el);
1427            out.push_str("</style:page-layout-properties>");
1428        } else {
1429            out.push_str(rest.get(..=end).unwrap_or_default());
1430            out.push_str(&el);
1431        }
1432        rest = rest.get(end + 1..).unwrap_or_default();
1433    }
1434    out.push_str(rest);
1435    out
1436}

1438/// `xml` (content.xml or styles.xml) with each font family it uses declared once (§16.21
1439/// `style:font-face`: its family, generic family and pitch) and named where it is used
1440/// (`style:font-name` beside `fo:font-family`, and the Asian and complex ones): a reader without
1441/// the font picks a like one.
1442fn with_font_faces(xml: &str) -> String {
1443    const ATTRS: [(&str, &str); 3] = [
1444        ("fo:font-family=\"", "style:font-name"),
1445        ("style:font-family-asian=\"", "style:font-name-asian"),
1446        ("style:font-family-complex=\"", "style:font-name-complex"),
1447    ];
1448    let mut families: Vec<String> = Vec::new();
1449    let mut out = String::with_capacity(xml.len() + 1024);
1450    let mut rest = xml;
1451    loop {
1452        // The next font family attribute.
1453        let next = ATTRS.iter().filter_map(|(a, n)| rest.find(a).map(|at| (at, *a, *n))).min_by_key(|(at, ..)| *at);
1454        let Some((at, attr, name_attr)) = next else { break };
1455        let value_at = at + attr.len();
1456        let Some(end) = rest.get(value_at..).and_then(|r| r.find('"')).map(|e| value_at + e) else { break };
1457        let value = rest.get(value_at..end).unwrap_or("");
1458        let family = value.replace("&apos;", "'").trim().trim_matches(['\'', '"']).trim().to_string();
1459        out.push_str(rest.get(..at).unwrap_or(""));
1460        if !family.is_empty() && family.len() <= 200 && !family.contains(',') {
1461            if !families.contains(&family) && families.len() < 1000 {
1462                families.push(family.clone());
1463            }
1464            out.push_str(&format!("{name_attr}=\"{}\" ", super::writer::x(&family)));
1465        }
1466        out.push_str(rest.get(at..=end).unwrap_or(""));
1467        rest = rest.get(end + 1..).unwrap_or("");
1468    }
1469    out.push_str(rest);
1470    if families.is_empty() {
1471        return out;
1472    }
1473    let decls: String = families
1474        .iter()
1475        .map(|f| {
1476            let (generic, pitch) = font_kind(f);
1477            format!(
1478                "<style:font-face style:name=\"{0}\" svg:font-family=\"'{0}'\" style:font-family-generic=\"{generic}\" style:font-pitch=\"{pitch}\"/>",
1479                super::writer::x(f)
1480            )
1481        })
1482        .collect();
1483    let decls = format!("<office:font-face-decls>{decls}</office:font-face-decls>");
1484    // Before the styles: first in office:document-styles, after the scripts in the content.
1485    for before in ["<office:styles>", "<office:automatic-styles>", "<office:automatic-styles/>", "<office:master-styles>", "<office:body>"] {
1486        if let Some(at) = out.find(before) {
1487            out.insert_str(at, &decls);
1488            return out;
1489        }
1490    }
1491    out
1492}

1494/// A font family's generic family and pitch (§20.270, §20.271), from its name: monospaced
1495/// families are `modern` and fixed, serif ones `roman`, the rest `swiss`.
1496fn font_kind(family: &str) -> (&'static str, &'static str) {
1497    let f = family.to_ascii_lowercase();
1498    if model::is_mono(family) {
1499        return ("modern", "fixed");
1500    }
1501    const SERIF: [&str; 16] = [
1502        "serif",
1503        "times",
1504        "georgia",
1505        "cambria",
1506        "caladea",
1507        "garamond",
1508        "palatino",
1509        "book",
1510        "roman",
1511        "libertine",
1512        "minion",
1513        "baskerville",
1514        "bodoni",
1515        "didot",
1516        "charter",
1517        "constantia",
1518    ];
1519    if SERIF.iter().any(|s| f.contains(s)) && !f.contains("sans") {
1520        return ("roman", "variable");
1521    }
1522    ("swiss", "variable")
1523}
