1//! A rough size for an equation written as a formula object: LibreOffice scales a formula to its
2//! frame, so the frame should be about the size the formula takes. A box per node (width, height
3//! above and below the baseline, in ems of 12 pt text), sized by its structure.
4
5use crate::doc::math::{MAX_DEPTH, MNode};
6
7/// Width, ascent and descent, in ems.
8#[derive(Clone, Copy, Default)]
9struct Box3 {
10    w: f32,
11    asc: f32,
12    desc: f32,
13}
14
15impl Box3 {
16    fn h(self) -> f32 {
17        self.asc + self.desc
18    }
19    fn scaled(self, k: f32) -> Box3 {
20        Box3 { w: self.w * k, asc: self.asc * k, desc: self.desc * k }
21    }
22}
23
24/// Script and limit size relative to the base.
25const SMALL: f32 = 0.7;
26
27fn row(nodes: &[MNode], depth: usize) -> Box3 {
28    if depth > MAX_DEPTH {
29        return Box3::default();
30    }
31    nodes.iter().map(|n| node(n, depth + 1)).fold(Box3 { w: 0.0, asc: 0.75, desc: 0.25 }, |a, b| Box3 {
32        w: a.w + b.w,
33        asc: a.asc.max(b.asc),
34        desc: a.desc.max(b.desc),
35    })
36}
37
38fn text(s: &str) -> Box3 {
39    // Operators take room on both sides.
40    let w: f32 = s
41        .chars()
42        .map(|c| {
43            if "+-−=<>≤≥±×÷→←≠≈".contains(c) {
44                1.0
45            } else if c.is_whitespace() {
46                0.25
47            } else {
48                0.55
49            }
50        })
51        .sum();
52    Box3 { w, asc: 0.75, desc: 0.25 }
53}
54
55fn node(n: &MNode, depth: usize) -> Box3 {
56    let r = |a: &[MNode]| row(a, depth);
57    match n {
58        MNode::Run(run) => text(&run.text),
59        MNode::Frac { num, den, .. } => {
60            let (a, b) = (r(num), r(den));
61            Box3 { w: a.w.max(b.w) + 0.3, asc: a.h() + 0.35, desc: b.h() - 0.15 }
62        }
63        MNode::Script { base, sub, sup, .. } => {
64            let b = r(base);
65            let (sb, sp) = (r(sub).scaled(SMALL), r(sup).scaled(SMALL));
66            let asc = if sup.is_empty() { b.asc } else { b.asc.max(sp.h() + 0.35) };
67            let desc = if sub.is_empty() { b.desc } else { b.desc.max(sb.h() - 0.1) };
68            Box3 { w: b.w + sb.w.max(sp.w) + 0.1, asc, desc }
69        }
70        MNode::Rad { deg, e, .. } => {
71            let b = r(e);
72            Box3 { w: b.w + 0.8 + r(deg).scaled(0.5).w, asc: b.asc + 0.2, desc: b.desc }
73        }
74        MNode::Nary { sub, sup, e, .. } => {
75            let b = r(e);
76            let (sb, sp) = (r(sub).scaled(SMALL), r(sup).scaled(SMALL));
77            Box3 {
78                w: 1.0 + sb.w.max(sp.w) + b.w + 0.2,
79                asc: b.asc.max(1.0 + if sup.is_empty() { 0.0 } else { sp.h() }),
80                desc: b.desc.max(0.4 + if sub.is_empty() { 0.0 } else { sb.h() }),
81            }
82        }
83        MNode::Delim { elems, .. } => {
84            let inner = elems.iter().map(|a| r(a)).fold(Box3 { w: 0.0, asc: 0.75, desc: 0.25 }, |a, b| Box3 {
85                w: a.w + b.w + 0.3,
86                asc: a.asc.max(b.asc),
87                desc: a.desc.max(b.desc),
88            });
89            Box3 { w: inner.w + 0.6, ..inner }
90        }
91        MNode::Func { name, e } => {
92            let (a, b) = (r(name), r(e));
93            Box3 { w: a.w + b.w + 0.2, asc: a.asc.max(b.asc), desc: a.desc.max(b.desc) }
94        }
95        MNode::Lim { upper, e, lim } => {
96            let (b, l) = (r(e), r(lim).scaled(SMALL));
97            let w = b.w.max(l.w);
98            if *upper { Box3 { w, asc: b.asc + l.h(), desc: b.desc } } else { Box3 { w, asc: b.asc, desc: b.desc + l.h() } }
99        }
100        MNode::Acc { e, .. } => {
101            let b = r(e);
102            Box3 { asc: b.asc + 0.3, ..b }
103        }
104        MNode::Bar { top, e } | MNode::GroupChr { top, e, .. } => {
105            let b = r(e);
106            if *top { Box3 { asc: b.asc + 0.3, ..b } } else { Box3 { desc: b.desc + 0.3, ..b } }
107        }
108        MNode::BorderBox { e, .. } | MNode::Boxed { e } => {
109            let b = r(e);
110            Box3 { w: b.w + 0.4, asc: b.asc + 0.2, desc: b.desc + 0.2 }
111        }
112        MNode::EqArr { rows } => stack(rows.iter().map(|a| r(a))),
113        MNode::Matrix { rows, .. } => {
114            let cols = rows.iter().map(Vec::len).max().unwrap_or(0);
115            let w: f32 = (0..cols).map(|c| rows.iter().filter_map(|row| row.get(c)).map(|a| r(a).w).fold(0.0, f32::max)).sum::<f32>()
116                + 0.6 * cols.saturating_sub(1) as f32;
117            let s =
118                stack(rows.iter().map(|row| {
119                    row.iter().map(|a| r(a)).fold(Box3::default(), |a, b| Box3 { w: 0.0, asc: a.asc.max(b.asc), desc: a.desc.max(b.desc) })
120                }));
121            Box3 { w, ..s }
122        }
123        MNode::Phant { show, zero_wid, zero_asc, zero_desc, e } => {
124            let b = r(e);
125            if !*show && *zero_wid && *zero_asc && *zero_desc {
126                return Box3::default();
127            }
128            Box3 { w: if *zero_wid { 0.0 } else { b.w }, asc: if *zero_asc { 0.0 } else { b.asc }, desc: if *zero_desc { 0.0 } else { b.desc } }
129        }
130    }
131}
132
133/// Rows one under another, centred on the math axis.
134fn stack(rows: impl Iterator<Item = Box3>) -> Box3 {
135    let (mut w, mut h, mut n) = (0.0f32, 0.0f32, 0usize);
136    for b in rows {
137        w = w.max(b.w);
138        h += b.h();
139        n += 1;
140    }
141    let h = h + 0.2 * n.saturating_sub(1) as f32;
142    Box3 { w, asc: h / 2.0 + 0.25, desc: h / 2.0 - 0.25 }
143}

145/// Width and height in points of the equation `linear` set in 12 pt text.
146pub(super) fn size(linear: &str) -> (f32, f32) {
147    let b = row(&crate::doc::math::parse_linear(linear), 0);
148    let fit = |v: f32| if v.is_finite() { v } else { 1.0 };
149    ((fit(b.w) * 12.0 + 2.0).clamp(4.0, 1584.0), (fit(b.h()) * 12.0 + 2.0).clamp(8.0, 1584.0))
150}

152#[cfg(test)]
153mod tests {
154    use super::size;
155
156    #[test]
157    fn structure_makes_room() {
158        let (xw, xh) = size("x");
159        let (fw, fh) = size("(a+b)/c");
160        assert!(fh > xh * 1.6, "a fraction is taller: {fh} vs {xh}");
161        assert!(fw > xw * 2.0);
162        let (sw, sh) = size("x^2");
163        assert!(sw > xw && sh > xh);
164        let (_, mh) = size("■(a&b@c&d)");
165        assert!(mh > xh * 1.6, "{mh}");
166        // Hostile: deep nesting and long text stay bounded.
167        let deep = "(".repeat(5000) + &")".repeat(5000);
168        let (dw, dh) = size(&deep);
169        assert!(dw.is_finite() && dh.is_finite() && dw <= 1584.0 && dh <= 1584.0);
170        assert!(size(&"x".repeat(100_000)).0 <= 1584.0);
171    }
