6/// An embedded ODF formula sub-document's `content.xml`, read for its `math:math` element
7/// (its root, ODF 1.3 Part 3 §2.2.9 B; some writers wrap it in an `office:math`, which isn't
8/// it): the first MathML element named `math` in document order, with its
9/// subtree, passed to [`crate::formats::mathml::read`]. `None` when there's no such element, it holds
10/// nothing, or the XML is malformed. Nesting and element count are bounded like other XML here.
11pub(in crate::formats::odt) fn read_formula(xml: &[u8]) -> Option<(String, bool)> {
12    use crate::formats::mathml::{El, Item, MAX_NODES, local};
13    let mut stack: Vec<El> = Vec::new();
14    let mut math: Option<El> = None;
15    let mut over = 0usize;
16    let mut count = 0usize;
17    events(xml, |ev| {
18        if math.is_some() {
19            return;
20        }
21        count += 1;
22        if count > MAX_NODES * 4 {
23            math = stack.drain(..).next();
24            return;
25        }
26        match ev {
27            Ev::Start(q, a) => {
28                if stack.is_empty() && (local(&q) != "math" || q.starts_with("office:")) {
29                    return;
30                }
31                if stack.len() < crate::doc::math::MAX_DEPTH {
32                    stack.push(El { name: q, attrs: a, children: Vec::new() });
33                } else {
34                    over += 1;
35                }
36            }
37            Ev::End(_) => {
38                if over > 0 {
39                    over -= 1;
40                } else if stack.len() > 1 {
41                    if let Some(done) = stack.pop()
42                        && let Some(parent) = stack.last_mut()
43                    {
44                        parent.children.push(Item::El(done));
45                    }
46                } else if stack.len() == 1 {
47                    math = stack.pop();
48                }
49            }
50            Ev::Text(t) => {
51                if over == 0
52                    && let Some(top) = stack.last_mut()
53                {
54                    top.children.push(Item::Text(t));
55                }
56            }
57        }
58    });
59    crate::formats::mathml::read(&math?)
60}

74/// A picture's crop as fractions of its size (left, top, right, bottom) from its `fo:clip`
75/// (top, right, bottom, left, in lengths of its original size: its pixels at 96 per inch). No crop
76/// when the picture's size can't be read or the crop would leave nothing.
77pub(in crate::formats::odt) fn crop_fractions([t, r, b, l]: [f32; 4], data: &[u8]) -> [f32; 4] {
78    let Some((pw, ph)) = model::image_px(data).filter(|(w, h)| *w > 0 && *h > 0) else { return [0.0; 4] };
79    let (w, h) = (pw as f32 * 0.75, ph as f32 * 0.75);
80    let c = [l / w, t / h, r / w, b / h].map(|v| if v.is_finite() { v.clamp(0.0, 0.99) } else { 0.0 });
81    if c[0] + c[2] < 0.99 && c[1] + c[3] < 0.99 { c } else { [0.0; 4] }
82}

91    /// End a frame or drawing shape: its picture, equation, text box or shape goes into the
92    /// paragraph.
93    pub(in crate::formats::odt) fn end_frame(&mut self) {
94        if let Some((w, h, img, alt, eq, chart, object)) = self.frame.take() {
95            let boxed = self.frame_box.take();
96            let mut float = self.frame_float.take();
97            if boxed.is_some()
98                && self.shape_kind.is_some()
99                && let Some(f) = &mut float
100            {
101                let (h, v) = std::mem::take(&mut self.shape_flip);
102                f.flip_h |= h;
103                f.flip_v ^= v;
104            }
105            // Kept when it floats, or turns or mirrors the object in line.
106            let float = float
107                .filter(|f| f.wrap != crate::doc::para::Wrap::Inline || crate::formats::odt::frames::turned(f) || f.flip_h || f.flip_v || f.y != 0.0);
108            if let Some(blocks) = &boxed
109                && self.take_watermark(blocks)
110            {
111                self.frame = None;
112                return;
113            }
114            // In line with no stated size, a picture comes at its own: fitted to the text, as
115            // other formats' pictures are. A stated size is the file's.
116            let room = self.sec.text_width();
117            let says = |v: Option<f32>| v.is_some_and(|v| v.is_finite() && v > 0.0);
118            // (Still in line when raised, turned or mirrored.)
119            let in_line = float.as_ref().is_none_or(|f| f.wrap == crate::doc::para::Wrap::Inline);
120            let fit = (!says(w) && !says(h) && in_line && room.is_finite() && room > 0.0).then_some(room);
121            let inline = match (boxed, eq) {
122                (Some(blocks), _) => {
123                    let mut look = std::mem::take(&mut self.frame_look);
124                    // Without a height it grows from nothing to its text.
125                    let kind = self.shape_kind.take().unwrap_or(crate::doc::para::ShapeKind::TextBox);
126                    // A custom shape's geometry as read, kept with the path worked out from it.
127                    let geometry = self.shape_geometry.take();
128                    if kind == crate::doc::para::ShapeKind::Freeform || kind == crate::doc::para::ShapeKind::Rectangle {
129                        look.geometry = geometry;
130                    }
131                    // A shape whose paragraphs are all empty (LibreOffice writes `<text:p/>` in
132                    // each) holds no text.
133                    let empty = blocks.iter().all(|b| matches!(b, FBlock::Para(p) if p.inlines.is_empty()));
134                    let blocks = if kind != crate::doc::para::ShapeKind::TextBox && empty { Vec::new() } else { blocks };
135                    let path = self.shape_path.take().filter(|_| kind == crate::doc::para::ShapeKind::Freeform);
136                    let (w, h) = (w.unwrap_or(72.0), h.unwrap_or(1.0));
137                    self.note_place(kind, (w, h), &look, float.as_ref());
138                    Some(Inline::Shape { kind, w, h, float: float.map(Box::new), blocks, look, path })
139                }
140                (None, Some((linear, display))) => Some(Inline::Equation { linear, display }),
141                (None, None) if let Some(spec) = chart => Some(Inline::Chart(model::Chart {
142                    spec: std::sync::Arc::new(spec),
143                    w: w.unwrap_or(360.0),
144                    h: h.unwrap_or(216.0),
145                    alt: alt.trim().to_string(),
146                    float: float.map(Box::new),
147                    source: object.map(std::sync::Arc::new),
148                })),
149                // An object kept whole: shown by its replacement picture when WordCraft can show
150                // it, else by a placeholder.
151                (None, None) if let Some(mut o) = object => {
152                    if o.replacement.is_none() {
153                        o.replacement = img.clone().map(std::sync::Arc::new);
154                    }
155                    let shown = img
156                        .and_then(|data| picture(data, w, h, alt.trim()))
157                        .or_else(|| crate::formats::odt::objects::placeholder(&o.mime).and_then(|p| make_img(p, w, h, alt.trim())));
158                    shown.map(|mut img| {
159                        fit_to(&mut img, fit);
160                        img.float = float.map(Box::new);
161                        img.object = Some(std::sync::Arc::new(o));
162                        Inline::Image(img)
163                    })
164                }
165                (None, None) => img.and_then(|data| picture(data, w, h, alt.trim())).map(|mut img| {
166                    fit_to(&mut img, fit);
167                    img.float = float.map(Box::new);
168                    if let Some(clip) = self.frame_clip.take() {
169                        img.crop = crop_fractions(clip, &img.data);
170                    }
171                    Inline::Image(img)
172                }),
173            };
174            if let Some(inline) = inline {
175                // A group's member goes into the group, at its place in it.
176                if let Some(((x, y), g)) = self.member_at.take().zip(self.group.as_mut().filter(|_| self.note.is_none())) {
177                    if g.members.len() < crate::doc::para::MAX_GROUP_CHILDREN {
178                        g.members.push(model::Member { x, y, inline });
179                    }
180                } else {
181                    self.push_inline(inline);
182                }
183            }
184        }
185        self.frame = None;
186    }

188    /// The shape ending is a watermark (see [`Self::start_shape`]) holding `blocks`: the
189    /// document's watermark, from its text, font, colour, size and turn, and not a shape.
190    fn take_watermark(&mut self, blocks: &[FBlock]) -> bool {
191        let Some(((w, h), rot, color, see_through, true)) = self.watermark_shape.take() else { return false };
192        let mut text = String::new();
193        let mut font = None;
194        for b in blocks {
195            if let FBlock::Para(p) = b {
196                for i in &p.inlines {
197                    if let Inline::Text(t, f) = i {
198                        text.push_str(t);
199                        font = font.or_else(|| f.font.as_deref().map(str::to_string));
200                    }
201                }
202            }
203        }
204        let text: String = text.trim().chars().take(200).collect();
205        if text.is_empty() {
206            return false;
207        }
208        let turned = rot.rem_euclid(360.0);
209        let base = crate::doc::Watermark::default();
210        self.watermark = Some(crate::doc::Watermark {
211            text,
212            font: font.unwrap_or(base.font),
213            color: color.unwrap_or(base.color),
214            diagonal: turned > 1.0 && turned < 359.0,
215            semitransparent: see_through,
216            size: Some([w, h]).filter(|[w, h]| w.is_finite() && h.is_finite() && *w > 0.0 && *h > 0.0),
217        });
218        true
219    }

221    /// A shape's drawing id (its `xml:id` or `draw:id`) and, for a connector, its glued ends (§19.228
222    /// `draw:start-shape`, §19.229 `draw:start-glue-point`…), their glue points as written until
223    /// the paragraph ends (see [`Self::glue_connectors`]). `None` for a shape with neither.
224    fn conn_of(&mut self, a: &Attrs) -> Option<model::Conn> {
225        let mut id_of = |name: &str| -> u32 {
226            let next = u32::try_from(self.conn_ids.len()).unwrap_or(u32::MAX).saturating_add(1);
227            match self.conn_ids.get(name) {
228                Some(id) => *id,
229                None if self.conn_ids.len() < MAX_CONN_IDS => {
230                    self.conn_ids.insert(name.to_string(), next);
231                    next
232                }
233                None => 0,
234            }
235        };
236        let id = get(a, "xml:id").or_else(|| get(a, "draw:id")).map(str::trim).filter(|n| !n.is_empty()).map_or(0, &mut id_of);
237        let mut end = |shape: &str, glue: &str| {
238            let target = get(a, shape).map(str::trim).filter(|n| !n.is_empty()).map(&mut id_of).filter(|id| *id != 0)?;
239            let site = get(a, glue).and_then(|g| g.trim().parse::<u32>().ok()).filter(|g| *g <= crate::doc::connector::MAX_SITE)?;
240            Some(crate::doc::connector::ConnEnd { id: target, site })
241        };
242        let (start, finish) = (end("draw:start-shape", "draw:start-glue-point"), end("draw:end-shape", "draw:end-glue-point"));
243        (id != 0 || start.is_some() || finish.is_some()).then_some(model::Conn { id, start, end: finish })
244    }

246    /// Where a shape with a drawing id (`look`) sits, for connectors glued to it.
247    pub(in crate::formats::odt) fn note_place(
248        &mut self,
249        kind: crate::doc::para::ShapeKind,
250        (w, h): (f32, f32),
251        look: &model::BoxLook,
252        float: Option<&crate::doc::para::Float>,
253    ) {
254        let Some(c) = look.conn.as_deref().filter(|c| c.id != 0) else { return };
255        if self.shape_places.len() >= MAX_CONN_IDS {
256            return;
257        }
258        // A group's member (placed in the group, not on the paragraph) or a shape in line: known
259        // by its kind only, never in the paragraph's coordinates.
260        let seq = if self.member_at.is_some() { usize::MAX } else { self.para_seq };
261        let f = float.cloned().unwrap_or_default();
262        self.shape_places.insert(c.id, (kind, [f.x, f.y, w, h], f, if float.is_some() { seq } else { usize::MAX }));
263    }

265    /// The connectors of paragraph `p` glued to shapes (see [`Self::conn_of`]): each end's glue
266    /// point made WordCraft's site of the shape it is glued to (an end glued to a shape not read,
267    /// or to a glue point of the shape's own, is unglued); and a connector glued at both ends to
268    /// shapes anchored in this paragraph, placed alike, runs between their sites. (A connector's
269    /// own coordinates are in LibreOffice's drawing layer, not its anchor's.)
270    pub(in crate::formats::odt) fn glue_connectors(&self, p: &mut model::Para) {
271        for i in &mut p.inlines {
272            self.glue_connector(i, 0);
273        }
274    }

276    /// Connector `i` of a paragraph or a group (`depth` deep in groups) glued to shapes (see
277    /// [`Self::glue_connectors`]).
278    fn glue_connector(&self, i: &mut Inline, depth: usize) {
279        use crate::formats::odt::shapes::site_of_glue;
280        if let Inline::Group { members, .. } = i {
281            if depth < model::MAX_GROUP_DEPTH {
282                for m in members {
283                    self.glue_connector(&mut m.inline, depth + 1);
284                }
285            }
286            return;
287        }
288        let Inline::Shape { kind, w, h, float, look, .. } = i else { return };
289        let Some(c) = look.conn.as_deref_mut().filter(|_| kind.is_connector()) else { return };
290        let resolve = |e: Option<crate::doc::connector::ConnEnd>| {
291            let e = e?;
292            let place = self.shape_places.get(&e.id)?;
293            site_of_glue(place.0, e.site).map(|site| (crate::doc::connector::ConnEnd { id: e.id, site }, place))
294        };
295        let (start, end) = (resolve(c.start), resolve(c.end));
296        (c.start, c.end) = (start.map(|(e, _)| e), end.map(|(e, _)| e));
297        let (Some((s, sp)), Some((e, ep)), Some(f)) = (start, end, float.as_deref_mut()) else { return };
298        let same = |pl: &(crate::doc::para::ShapeKind, [f32; 4], crate::doc::para::Float, usize)| {
299            pl.3 == self.para_seq && pl.2.h_rel == f.h_rel && pl.2.v_rel == f.v_rel
300        };
301        if !(same(sp) && same(ep)) {
302            return;
303        }
304        let point = |pl: &(crate::doc::para::ShapeKind, [f32; 4], crate::doc::para::Float, usize), site| {
305            crate::doc::connector::site_point(pl.0, pl.1, pl.2.spin(), site)
306        };
307        let (Some((x1, y1)), Some((x2, y2))) = (point(sp, s.site), point(ep, e.site)) else { return };
308        (f.x, f.y, f.flip_h, f.flip_v) = (x1.min(x2), y1.min(y2), x2 < x1, y2 < y1);
309        (*w, *h) = ((x2 - x1).abs().min(MAX_LEN), (y2 - y1).abs().min(MAX_LEN));
310    }

312    /// A frame or shape `w` × `h` (attributes `a`) begins: in a group (not in a shape's text), where
313    /// it sits in it, from its `svg:x`/`svg:y` or `draw:transform` (a line's from its ends, `line`).
314    pub(in crate::formats::odt) fn member_start(&mut self, a: &Attrs, w: f32, h: f32, line: Option<(Option<f32>, Option<f32>)>) {
315        self.member_rot = None;
316        if self.group.is_none() || self.note.is_some() {
317            return;
318        }
319        let len = |k: &str| get(a, k).and_then(length).filter(|v| v.is_finite()).map(|v| v.clamp(-MAX_LEN, MAX_LEN));
320        let at = match (line, get(a, "draw:transform").and_then(|t| crate::formats::odt::frames::transform(t, w, h))) {
321            (Some((x, y)), _) => (x.unwrap_or(0.0), y.unwrap_or(0.0)),
322            (None, Some((x, y, rot))) => {
323                self.member_rot = Some(rot);
324                (x, y)
325            }
326            (None, None) => (len("svg:x").unwrap_or(0.0), len("svg:y").unwrap_or(0.0)),
327        };
328        self.member_at = Some(at);
329    }

331    /// The group being read ends: its members, placed from the corner of the box around them,
332    /// become one object anchored where the group is.
333    pub(in crate::formats::odt) fn end_group(&mut self) {
334        let Some(g) = self.group.take() else { return };
335        let size = |i: &Inline| match i {
336            Inline::Shape { w, h, .. } => (*w, *h),
337            Inline::Image(img) => (img.w, img.h),
338            Inline::Group { w, h, .. } => (*w, *h),
339            _ => (0.0, 0.0),
340        };
341        let fin = |v: f32| if v.is_finite() { v.clamp(-MAX_LEN, MAX_LEN) } else { 0.0 };
342        let (mut x0, mut y0, mut x1, mut y1) = (f32::MAX, f32::MAX, f32::MIN, f32::MIN);
343        for m in &g.members {
344            let (w, h) = size(&m.inline);
345            (x0, y0) = (x0.min(fin(m.x)), y0.min(fin(m.y)));
346            (x1, y1) = (x1.max(fin(m.x + w)), y1.max(fin(m.y + h)));
347        }
348        if g.members.is_empty() {
349            return;
350        }
351        let members = g.members.into_iter().map(|m| model::Member { x: fin(m.x) - x0, y: fin(m.y) - y0, ..m }).collect();
352        let graphic = self.styles.shape_graphic(g.style.as_deref());
353        let mut float = crate::formats::odt::frames::float_of(
354            g.anchor.as_deref(),
355            Some(&crate::formats::odt::writer::pt(x0)),
356            Some(&crate::formats::odt::writer::pt(y0)),
357            &graphic,
358        );
359        float.page = crate::formats::odt::frames::page_of(g.anchor.as_deref(), g.page.as_deref(), &graphic);
360        let float = (float.wrap != crate::doc::para::Wrap::Inline).then(|| Box::new(float));
361        let fit = |v: f32| if v.is_finite() { v.clamp(1.0, MAX_LEN) } else { 1.0 };
362        self.push_inline(Inline::Group { w: fit(x1 - x0), h: fit(y1 - y0), float, members });
363    }

365    /// Begin a drawing shape (ODF 1.3 §10.3): its size, placement and look, and its text read
366    /// as a text box's. A custom shape's kind comes from its geometry, read inside it.
367    pub(in crate::formats::odt) fn start_shape(&mut self, q: &str, a: &Attrs) {
368        let mut kind = crate::formats::odt::shapes::element_kind(local(q));
369        if kind.is_connector() {
370            kind = crate::formats::odt::shapes::connector_kind(get(a, "draw:type"));
371        }
372        let g = self.styles.shape_graphic(get(a, "draw:style-name"));
373        let (w, h, x, y, (flip_h, flip_v)) = crate::formats::odt::shapes::geometry(kind, a);
374        let mut float = if kind.is_open() {
375            // A line is placed by its ends.
376            let at = |v: Option<f32>| v.map(crate::formats::odt::writer::pt);
377            let mut f = crate::formats::odt::frames::float_of(get(a, "text:anchor-type"), at(x).as_deref(), at(y).as_deref(), &g);
378            f.z = crate::formats::odt::frames::z_of(a);
379            f.page = crate::formats::odt::frames::page_of(get(a, "text:anchor-type"), get(a, "text:anchor-page-number"), &g);
380            f
381        } else {
382            crate::formats::odt::frames::place(a, w, h, &g, self.text_size())
383        };
384        (float.flip_h, float.flip_v) = (flip_h, flip_v);
385        self.shape_flip = (false, false);
386        self.member_start(a, w, h, kind.is_open().then_some((x, y)));
387        if let Some(rot) = self.member_rot.take() {
388            float.rot = rot;
389        }
390        self.frame_look = g.shape_look(kind, &self.styles.defs);
391        self.frame_look.conn = self.conn_of(a).map(Box::new);
392        // Word's and LibreOffice's watermark: a shape of this name whose geometry follows its text.
393        self.watermark_shape = get(a, "draw:name").filter(|n| n.starts_with(crate::formats::odt::writer::WATERMARK_NAME)).map(|_| {
394            let (color, see_through) = g.watermark_look();
395            ((w, h), float.rot, color, see_through, false)
396        });
397        self.shape_path = crate::formats::odt::shapes::freeform(local(q), a, w, h).map(std::sync::Arc::new);
398        self.shape_kind = Some(kind);
399        let frame = (Some((Some(w), Some(h), None, String::new(), None, None, None)), Some(float), std::mem::take(&mut self.frame_look));
400        self.start_note(a);
401        if let Some(n) = &mut self.note {
402            n.text_box = Some(frame);
403        }
404    }
405}

412/// A picture from a package's bytes: a raster image, or a Windows metafile (WMF, EMF: LibreOffice
413/// keeps clip art and logos so), sized `w` × `h` (points) when the frame says.
414pub(in crate::formats::odt) fn picture(data: Vec<u8>, w: Option<f32>, h: Option<f32>, alt: &str) -> Option<model::Img> {
415    // What the frame states, if it could be a size (far past any page, kept from overflowing).
416    let stated = |v: Option<f32>| v.filter(|v| v.is_finite() && *v > 0.0).map(|v| v.min(1e6));
417    let (w, h) = (stated(w), stated(h));
418    let mut img = if crate::metafile::is_metafile(&data) {
419        // An EMF starts with its header record, type 1.
420        let ext = if data.get(..4) == Some(&[1, 0, 0, 0]) { "emf" } else { "wmf" };
421        // Its own size, when the frame gives none: a placeable WMF's or an EMF's frame, points.
422        let natural = if w.is_some() && h.is_some() { None } else { metafile_size(&data) };
423        model::sized_img(std::sync::Arc::new(data), ext, natural, w, h, alt)
424    } else {
425        make_img(data, w, h, alt)?
426    };
427    // A size the frame states is the file's, however large (a picture across a whole page): only
428    // a picture with none is shrunk to a column. Stated on one side, the other follows the
429    // picture's proportions.
430    let (w, h) = match (w, h) {
431        (Some(w), Some(h)) => (w, h),
432        (Some(w), None) => (w, w * height_per_width(&img.data)),
433        (None, Some(h)) => (h / height_per_width(&img.data), h),
434        (None, None) => return Some(img),
435    };
436    // Past what a document holds, it shrinks in proportion.
437    let (w, h) = match w.max(h) {
438        most if most <= 1584.0 => (w, h),
439        _ if w >= h => (1584.0, h * 1584.0 / w),
440        _ => (w * 1584.0 / h, 1584.0),
441    };
442    (img.w, img.h) = (w.max(1.0), h.max(1.0));
443    Some(img)
444}
445
446/// `img` shrunk in proportion to `room` points wide, when it's given and the picture is wider.
447fn fit_to(img: &mut model::Img, room: Option<f32>) {
448    if let Some(room) = room.filter(|room| img.w > *room) {
449        img.h = (img.h * room / img.w).max(1.0);
450        img.w = room.max(1.0);
451    }
452}
453
454/// A picture's height for each point of its width, by its own size, before anything caps it (1
455/// when it doesn't say).
456fn height_per_width(data: &[u8]) -> f32 {
457    let own = if crate::metafile::is_metafile(data) { metafile_size(data) } else { model::image_px(data).map(|(w, h)| (w as f32, h as f32)) };
458    own.filter(|(w, h)| w.is_finite() && h.is_finite() && *w > 0.0 && *h > 0.0).map_or(1.0, |(w, h)| h / w)
459}

461/// A metafile's own size, points, from its header: a placeable WMF's bounding box in its units
462/// per inch, or an EMF's frame (`rclFrame`, 0.01 mm). `None` when the header doesn't say.
463fn metafile_size(d: &[u8]) -> Option<(f32, f32)> {
464    let i16_at = |o: usize| d.get(o..o + 2).map(|b| f32::from(i16::from_le_bytes([b[0], b[1]])));
465    let i32_at = |o: usize| d.get(o..o + 4).map(|b| i32::from_le_bytes([b[0], b[1], b[2], b[3]]) as f32);
466    let (w, h) = if d.starts_with(&[0xd7, 0xcd, 0xc6, 0x9a]) {
467        let inch = d.get(14..16).map(|b| f32::from(u16::from_le_bytes([b[0], b[1]]))).filter(|v| *v > 0.0)?;
468        let (l, t, r, b) = (i16_at(6)?, i16_at(8)?, i16_at(10)?, i16_at(12)?);
469        ((r - l).abs() / inch * 72.0, (b - t).abs() / inch * 72.0)
470    } else if d.get(..4) == Some(&[1, 0, 0, 0]) {
471        let (l, t, r, b) = (i32_at(24)?, i32_at(28)?, i32_at(32)?, i32_at(36)?);
472        let pt = 72.0 / 2540.0;
473        ((r - l).abs() * pt, (b - t).abs() * pt)
474    } else {
475        return None;
476    };
477    (w.is_finite() && h.is_finite() && w > 0.0 && h > 0.0).then_some((w.min(1584.0), h.min(1584.0)))
478}
