1//! Tracked changes (ODF 1.3 §5.5): the changed regions, deleted content and the insertions open
2//! around the text.

4use super::{Attrs, Body, FBlock, HashMap, Inline, MAX_OPEN_INSERTIONS, MAX_REGIONS, Para, RevMark, RevisionKind, get, local};

6/// A `text:changed-region` read from `text:tracked-changes`.
7#[derive(Default)]
8pub(in crate::formats::odt) struct Region {
9    pub(in crate::formats::odt) kind: Option<RevisionKind>,
10    /// Sequence number: tells regions by the same author at the same time apart.
11    pub(in crate::formats::odt) id: u32,
12    pub(in crate::formats::odt) author: String,
13    pub(in crate::formats::odt) date: String,
14    /// A deletion's deleted paragraphs.
15    pub(in crate::formats::odt) blocks: Vec<FBlock>,
16}
17
18impl Region {
19    pub(in crate::formats::odt) fn mark(&self) -> RevMark {
20        RevMark { author: self.author.trim().to_string(), date: self.date.trim().to_string(), id: Some(self.id) }
21    }
22}

24/// Reader state saved while a deletion's content is read into its own container.
25pub(in crate::formats::odt) struct DelState {
26    pub(in crate::formats::odt) containers: usize,
27    pub(in crate::formats::odt) tables: usize,
28    pub(in crate::formats::odt) cells: usize,
29    pub(in crate::formats::odt) lists_saved: usize,
30    pub(in crate::formats::odt) fmt: usize,
31    pub(in crate::formats::odt) lists: Vec<(Option<String>, bool)>,
32    pub(in crate::formats::odt) item_used: Vec<bool>,
33    /// Insertions open inside the deleted content (text inserted, then deleted).
34    pub(in crate::formats::odt) ins: Vec<(String, RevMark)>,
35}

37/// A `text:change-start` (`start`) or `text:change-end` (attributes `a`) among the changes of
38/// `kind` open (insertions, or format changes): unknown ids, other kinds' marks, ends without a
39/// start and starts past [`MAX_OPEN_INSERTIONS`] are ignored.
40pub(in crate::formats::odt) fn change_mark(
41    regions: &HashMap<String, Region>,
42    open: &mut Vec<(String, RevMark)>,
43    kind: RevisionKind,
44    start: bool,
45    a: &Attrs,
46) {
47    let Some(id) = get(a, "text:change-id") else { return };
48    if !start {
49        if let Some(k) = open.iter().position(|(o, _)| o == id) {
50            open.remove(k);
51        }
52        return;
53    }
54    if let Some(r) = regions.get(id).filter(|r| r.kind == Some(kind))
55        && open.len() < MAX_OPEN_INSERTIONS
56        && !open.iter().any(|(o, _)| o == id)
57    {
58        open.push((id.to_string(), r.mark()));
59    }
60}

62impl Body<'_> {
63    /// An element inside `text:tracked-changes`, outside deleted content.
64    pub(in crate::formats::odt) fn tracked_start(&mut self, q: &str, a: &Attrs) {
65        match local(q) {
66            "changed-region" if self.region.is_none() && self.region_count < MAX_REGIONS => {
67                // ODF 1.3 names it by `xml:id`; `text:id` is the older name, kept beside it.
68                let id = get(a, "text:id").or_else(|| get(a, "xml:id")).unwrap_or_default().to_string();
69                let r = Region { id: self.region_count as u32, ..Default::default() };
70                self.region_count += 1;
71                self.region = Some((id, r));
72            }
73            "insertion" | "deletion" | "format-change" if self.region.as_ref().is_some_and(|(_, r)| r.kind.is_none()) => {
74                let kind = match local(q) {
75                    "insertion" => RevisionKind::Insert,
76                    "deletion" => RevisionKind::Delete,
77                    _ => RevisionKind::Format,
78                };
79                if let Some((_, r)) = &mut self.region {
80                    r.kind = Some(kind);
81                }
82                if kind == RevisionKind::Delete {
83                    self.containers.push(Vec::new());
84                    self.del = Some(DelState {
85                        containers: self.containers.len(),
86                        tables: self.tables.len(),
87                        cells: self.cells.len(),
88                        lists_saved: self.lists_saved.len(),
89                        fmt: self.fmt.len(),
90                        lists: std::mem::take(&mut self.lists),
91                        item_used: std::mem::take(&mut self.item_used),
92                        ins: Vec::new(),
93                    });
94                }
95            }
96            "change-info" if self.region.is_some() => self.info_at = Some(self.depth),
97            _ => self.skip = 1,
98        }
99    }

101    /// End a deletion's content: its paragraphs go to the region being read.
102    pub(in crate::formats::odt) fn close_deletion(&mut self) {
103        if self.del.is_none() {
104            return;
105        }
106        self.end_para();
107        let Some(d) = self.del.take() else { return };
108        let base = d.containers.saturating_sub(1).min(self.containers.len());
109        let blocks: Vec<FBlock> = self.containers.split_off(base).into_iter().flatten().filter(|b| matches!(b, FBlock::Para(_))).collect();
110        if self.containers.is_empty() {
111            self.containers.push(Vec::new());
112        }
113        self.tables.truncate(d.tables);
114        self.cells.truncate(d.cells);
115        self.formulas.truncate(d.cells);
116        self.cell_tops.truncate(d.cells);
117        self.lists_saved.truncate(d.lists_saved);
118        self.fmt.truncate(d.fmt);
119        self.lists = d.lists;
120        self.item_used = d.item_used;
121        self.frame = None;
122        self.in_desc = false;
123        self.para_cont = false;
124        if let Some((_, r)) = &mut self.region {
125            r.blocks = blocks;
126        }
127    }

129    /// Put a deletion's paragraphs back at a `text:change` point, as deleted text. Inside a
130    /// paragraph, each deleted paragraph mark splits it; between paragraphs, each deleted
131    /// paragraph stands alone with its mark deleted.
132    pub(in crate::formats::odt) fn place_deletion(&mut self, blocks: Vec<FBlock>, mark: RevMark) {
133        let mut paras: Vec<Para> = Vec::new();
134        for b in blocks {
135            let FBlock::Para(mut p) = b else { continue };
136            // Pictures and other objects carry no revision mark: leave them out.
137            p.inlines.retain(|i| matches!(i, Inline::Text(..) | Inline::Anchor(_)));
138            for i in &mut p.inlines {
139                if let Inline::Text(_, f) = i {
140                    f.del = Some(mark.clone());
141                }
142            }
143            paras.push(p);
144        }
145        let append = |dst: &mut Para, src: Vec<Inline>| {
146            for i in src {
147                match i {
148                    Inline::Text(t, f) => dst.push_text(&t, &f),
149                    other => dst.inlines.push(other),
150                }
151            }
152        };
153        let Some(mut cur) = self.para.take() else {
154            self.para_cont = false;
155            for mut p in paras {
156                p.mark_del = Some(Box::new(mark.clone()));
157                self.finish_para(p);
158            }
159            return;
160        };
161        let n = paras.len();
162        let mut it = paras.into_iter();
163        let mut first_mark_ins = None;
164        if let Some(first) = it.next() {
165            first_mark_ins = first.mark_ins;
166            append(&mut cur, first.inlines);
167        }
168        if n < 2 {
169            self.para = Some(cur);
170            return;
171        }
172        let mut rest = Para { kind: cur.kind, list: cur.list, align: cur.align, ..Default::default() };
173        // The host paragraph now ends at the first deleted paragraph's mark.
174        cur.mark_del = Some(Box::new(mark.clone()));
175        cur.mark_ins = first_mark_ins;
176        self.finish_para(cur);
177        for (k, mut p) in it.enumerate() {
178            if k + 2 == n {
179                // The last deleted paragraph runs on into the rest of the body paragraph.
180                rest.kind = p.kind;
181                rest.align = p.align.or(rest.align);
182                append(&mut rest, p.inlines);
183            } else {
184                p.mark_del = Some(Box::new(mark.clone()));
185                self.finish_para(p);
186            }
187        }
188        self.para = Some(rest);
189    }

191    pub(in crate::formats::odt) fn tracked_context(&mut self, q: &str, a: &Attrs) -> bool {
192        if !self.in_tc {
193            return false;
194        }
195        if self.del.is_none() {
196            self.tracked_start(q, a);
197            return true;
198        }
199        // Inside deleted content: only insertions (text inserted, then deleted) nest.
200        if matches!(local(q), "change-start" | "change-end") {
201            let regions = &self.regions;
202            if let Some(d) = &mut self.del {
203                change_mark(regions, &mut d.ins, RevisionKind::Insert, local(q) == "change-start", a);
204            }
205            return true;
206        }
207        if matches!(local(q), "tracked-changes" | "changed-region" | "insertion" | "deletion" | "format-change" | "change") {
208            self.skip = 1;
209            return true;
210        }
211        if local(q) == "change-info" && self.region.is_some() {
212            self.info_at = Some(self.depth);
213            return true;
214        }
215        false
216    }
