1//! Tables being read (ODF 1.3 §9.1): rows, cells and their shared edges.

3use super::{Borders, Cell, FTable};

5/// Give each edge two cells share the line either cell draws: LibreOffice gives a shared edge to
6/// one of them only (the left cell's right side is `none`, the right cell's left side draws it),
7/// while WordCraft's cells each draw their own four sides. The cell below is the one in the next
8/// row starting at the same column.
9pub(in crate::formats::odt) fn share_edges(rows: &mut [Vec<Cell>]) {
10    let visible = |b: Option<crate::doc::props::Border>| b.filter(|b| b.is_visible());
11    for r in 0..rows.len() {
12        let starts: Vec<usize> = rows
13            .get(r)
14            .map(|row| {
15                row.iter()
16                    .scan(0usize, |g, c| {
17                        let s = *g;
18                        *g += c.colspan.max(1) as usize;
19                        Some(s)
20                    })
21                    .collect()
22            })
23            .unwrap_or_default();
24        for k in 0..starts.len() {
25            // The cell to the right.
26            if let Some(row) = rows.get_mut(r)
27                && let [Some(a), Some(b)] = [k, k + 1].map(|i| row.get(i).map(|c| c.borders))
28                && (a.is_some() || b.is_some())
29            {
30                let (a, b) = (a.unwrap_or_default(), b.unwrap_or_default());
31                let line = visible(a.right).or(visible(b.left));
32                if let Some(line) = line {
33                    if let Some(c) = row.get_mut(k) {
34                        c.borders = Some(Borders { right: Some(line), ..a });
35                    }
36                    if let Some(c) = row.get_mut(k + 1) {
37                        c.borders = Some(Borders { left: Some(line), ..b });
38                    }
39                }
40            }
41            // The cell below.
42            let start = starts.get(k).copied().unwrap_or(0);
43            let below = rows.get(r + 1).and_then(|row| {
44                let mut g = 0;
45                row.iter().position(|c| {
46                    let s = g;
47                    g += c.colspan.max(1) as usize;
48                    s == start
49                })
50            });
51            let (Some(j), Some(a)) = (below, rows.get(r).and_then(|row| row.get(k)).map(|c| c.borders)) else { continue };
52            let Some(b) = rows.get(r + 1).and_then(|row| row.get(j)).map(|c| c.borders) else { continue };
53            if a.is_none() && b.is_none() {
54                continue;
55            }
56            let (a, b) = (a.unwrap_or_default(), b.unwrap_or_default());
57            if let Some(line) = visible(a.bottom).or(visible(b.top)) {
58                if let Some(c) = rows.get_mut(r).and_then(|row| row.get_mut(k)) {
59                    c.borders = Some(Borders { bottom: Some(line), ..a });
60                }
61                if let Some(c) = rows.get_mut(r + 1).and_then(|row| row.get_mut(j)) {
62                    c.borders = Some(Borders { top: Some(line), ..b });
63                }
64            }
65        }
66    }
67}

91/// The narrowest column the document model keeps (a narrower one would lose the table its grid).
92const MIN_COLUMN: f32 = 6.0;
93
94/// Each column's width out of `room` points, as LibreOffice shares it out: by its share if it has
95/// one, else its stated width, which comes off the room first; none narrower than a column can be
96/// (6 pt). Empty if a column has neither, or no room is left to share.
97fn widths_of_shares(widths: &[f32], shares: &[f32], room: f32) -> Vec<f32> {
98    let columns = || widths.iter().zip(shares);
99    let usable = |v: f32| v.is_finite() && v > 0.0;
100    let left = room - columns().filter(|(_, s)| **s <= 0.0).map(|(w, _)| *w).sum::<f32>();
101    let total: f32 = shares.iter().filter(|s| **s > 0.0).sum();
102    if widths.len() != shares.len() || !usable(left) || !usable(total) || columns().any(|(w, s)| *w <= 0.0 && *s <= 0.0) {
103        return Vec::new();
104    }
105    columns().map(|(w, s)| if *s > 0.0 { (s / total * left).max(MIN_COLUMN) } else { *w }).collect()
106}

108impl TableB {
109    pub(in crate::formats::odt) fn finish(self) -> FTable {
110        // Columns stating their shares (LibreOffice's tables filling the margins, §20.334) divide
111        // the table's width by them; a column with neither leaves the whole grid to the equal split.
112        // An aligned table's shares go before the widths its columns state too, as LibreOffice
113        // reads them. (A table filling the margins is fitted to them whatever its widths say, and
114        // files give it both in proportion; in something narrower than the page, whose width isn't
115        // known here, stated widths are the best guess.)
116        let all_widths = self.widths.iter().all(|w| *w > 0.0);
117        let all_shares = !self.shares.is_empty() && self.shares.iter().all(|s| *s > 0.0);
118        let aligned = self.extra.align.is_some() && !self.contained;
119        let by_shares = self
120            .room
121            .filter(|_| all_shares && (aligned || !all_widths))
122            .map(|room| widths_of_shares(&self.widths, &self.shares, room))
123            .filter(|w| !w.is_empty());
124        let widths = if let Some(w) = by_shares {
125            w
126        } else if all_widths {
127            self.widths
128        } else {
129            // With nothing to divide, or in something narrower than the page with widths stated
130            // too, the shares can't be worked out here: the equal split, as before.
131            let mixed = self.shares.iter().any(|s| *s <= 0.0);
132            match self.room.filter(|_| !(self.contained && mixed)) {
133                Some(room) => widths_of_shares(&self.widths, &self.shares, room),
134                None => Vec::new(),
135            }
136        };
137        let mut rows = self.rows;
138        share_edges(&mut rows);
139        // A table whose cells all say "no line" draws none.
140        let stated = rows.iter().flatten().filter(|c| !c.covered).all(|c| c.borders.is_some());
141        let borderless = stated && rows.iter().flatten().all(|c| !c.borders.is_some_and(|b| b.any_visible()));
142        let extra = self.extra;
143        let kept = extra.fixed
144            || extra.align.is_some()
145            || extra.width_pct.is_some()
146            || extra.heights.iter().any(Option::is_some)
147            || extra.keep_rows.contains(&true);
148        FTable { rows, widths, borderless, extra: kept.then(|| Box::new(extra)), kept: None, wrap: None }
149    }
