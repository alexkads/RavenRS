39    use crate::doc::section::NumFormat as F;
40    match f {
41        F::LowerRoman => "i",
42        F::UpperRoman => "I",
43        F::LowerLetter => "a",
44        F::UpperLetter => "A",
45        F::None => "",
46        _ => "1",
47    }

50pub(super) fn x(s: &str) -> String {
51    let mut o = String::with_capacity(s.len());
52    for c in s.chars() {
53        match c {
54            '&' => o.push_str("&amp;"),
55            '<' => o.push_str("&lt;"),
56            '>' => o.push_str("&gt;"),
57            '"' => o.push_str("&quot;"),
58            '\'' => o.push_str("&apos;"),
59            '\u{FFFE}' | '\u{FFFF}' => {}
60            c if c.is_control() && !matches!(c, '\t' | '\n') => {}
61            _ => o.push(c),
62        }
63    }
64    o
65}

96/// The run of list items written as one list from `start` in `blocks`: where it ends, and each
97/// level's kind as its first item there says. An item of another list, or (lists not told apart)
98/// a top-level item of the other kind, starts a new list; with section wrappers (`wraps`), so
99/// does an item opening a section, and one closing a section ends it.
100fn list_run(blocks: &[FBlock], start: usize, wraps: bool) -> (usize, [ListOut; 9]) {
101    let p = match blocks.get(start) {
102        Some(FBlock::Para(p)) => p,
103        _ => return (start.saturating_add(1), [ListOut { ordered: false, label: None, start: 1 }; 9]),
104    };
105    let id = p.list.map_or(0, |l| l.list);
106    let first = ListOut { ordered: p.list.is_some_and(|l| l.ordered), label: None, start: 1 };
107    let mut kinds = [first; 9];
108    let mut seen = [false; 9];
109    let mut i = start;
110    while let Some(FBlock::Para(q)) = blocks.get(i) {
111        let Some(li) = q.list else { break };
112        let l = li.level.min(8) as usize;
113        if li.list != id || (id == 0 && l == 0 && seen[0] && kinds[0].ordered != li.ordered) {
114            break;
115        }
116        let wrap = q.wrap.as_deref().filter(|_| wraps);
117        if i > start && wrap.is_some_and(|w| !w.0.is_empty()) {
118            break;
119        }
120        if let (Some(s), Some(k)) = (seen.get_mut(l), kinds.get_mut(l))
121            && !*s
122        {
123            *s = true;
124            *k = ListOut { ordered: li.ordered, label: li.label, start: li.start };
125        }
126        i += 1;
127        if wrap.is_some_and(|w| w.1 > 0) {
128            break;
129        }
130    }
131    (i.max(start + 1), kinds)
132}

147pub(super) fn pt(v: f32) -> String {
148    let s = format!("{:.2}", v);
149    let s = s.trim_end_matches('0').trim_end_matches('.');

299    pub(in crate::formats::odt) fn text_style(&mut self, f: &Fmt) -> Option<String> {
300        let mut props = text_props(f);
301        // A custom character style: the automatic style's parent (after a separator in the key).

312        }
313        let n = format!("{}T{}", self.prefix, self.text_styles.len() + 1);
314        self.text_styles.push((props, n.clone()));
315        Some(n)
316    }

322    /// The paragraph style of `p`; `start` makes it begin a page of that master page (ODF 1.3
323    /// §19.471 `style:master-page-name`), restarting page numbering where asked (§20.322).
324    fn para_style_at(&mut self, p: &Para, start: Option<&SecStart>) -> String {
325        let named = p.extra.as_ref().and_then(|x| x.style.as_deref());
326        let parent = match (p.kind, self.note_parent, named) {
327            // A custom style: written as a common style of the same name (see `custom_styles`).
328            (_, _, Some(id)) => {
329                self.named.insert(id.to_string());
330                odf_name(id)
331            }
332            (Kind::Normal, Some(n), None) => n.to_string(),
333            _ => parent_style(p.kind).to_string(),
334        };
335        let mut props = String::new();
336        match p.align {
337            Some(Align::Center) => props.push_str(" fo:text-align=\"center\""),
338            Some(Align::Right) => props.push_str(" fo:text-align=\"end\""),
339            Some(Align::Justify) | Some(Align::Distribute) => props.push_str(" fo:text-align=\"justify\""),
340            Some(Align::Left) => props.push_str(" fo:text-align=\"start\""),
341            None => {}
342        }
343        // The master page starts the page itself: no page break besides.
344        if p.page_break && start.is_none() {
345            props.push_str(" fo:break-before=\"page\"");
346        } else if !p.page_break && starts_column(p) {
347            // A column break first (dropped from the text): the paragraph starts a column.
348            props.push_str(" fo:break-before=\"column\"");
349        }
350        if let Some(n) = start.and_then(|(_, n)| *n) {
351            props.push_str(&format!(" style:page-number=\"{n}\""));
352        }
353
354        if p.keep_next {
355            props.push_str(" fo:keep-with-next=\"always\"");
356        }
357        props.push_str(&spacing_props(&p.spacing));
358        // Child elements of the paragraph properties: tab stops (§17.8), a drop cap (§17.9).
359        let mut children = String::new();
360        if let Some(tabs) = p.extra.as_ref().and_then(|x| x.tabs.as_ref()) {
361            children.push_str(&tab_stops(tabs));
362        }
363        if let Some((lines, distance)) = p.extra.as_ref().and_then(|x| x.drop_cap.filter(|l| *l > 1).map(|l| (l.min(10), x.drop_cap_distance))) {
364            let d = distance.filter(|d| d.is_finite()).map_or(0.0, |d| d.clamp(0.0, 1584.0));
365            children.push_str(&format!("<style:drop-cap style:lines=\"{lines}\" style:length=\"1\" style:distance=\"{}\"/>", pt(d)));
366        }
367        if p.kind == Kind::Rule {
368            props.push_str(" fo:border-bottom=\"0.75pt solid #a0a0a0\" fo:padding-bottom=\"1pt\"");
369        } else {
370            props.push_str(&border_attrs(p.borders));
371        }
372        if let Some(bg) = p.extra.as_ref().and_then(|x| x.shading) {
373            props.push_str(&format!(" fo:background-color=\"#{}\"", bg.hex().to_ascii_lowercase()));
374        }
375        if let Some(c) = p.extra.as_ref().and_then(|x| x.contextual_spacing) {
376            props.push_str(&format!(" style:contextual-spacing=\"{c}\""));
377        }
378        if p.extra.as_ref().and_then(|x| x.bidi) == Some(true) {
379            props.push_str(" style:writing-mode=\"rl-tb\"");
380        }
381        // How its lines break across pages (§20.247, §20.251, §20.250).
382        if p.extra.as_ref().and_then(|x| x.keep_lines) == Some(true) {
383            props.push_str(" fo:keep-together=\"always\"");
384        }
385        if p.extra.as_ref().and_then(|x| x.widow_control) == Some(false) {
386            props.push_str(" fo:widows=\"0\" fo:orphans=\"0\"");
387        }
388        // The children go after a separator in the key; text properties (hyphenation, a text
389        // property in ODF, §20.217) after another.
390        if !children.is_empty() {
391            props.push('\u{1}');
392            props.push_str(&children);
393        }
394        // A paragraph without text: its mark formatting, its size giving its height.
395        let mark = p.mark.as_deref().filter(|_| p.textless()).map(super::writer::props::props_text).unwrap_or_default();
396        let hyphenate = p.extra.as_ref().and_then(|x| x.suppress_hyphens) == Some(false);
397        if hyphenate || !mark.is_empty() {
398            props.push('\u{2}');
399            props.push_str(&mark);
400        }
401        if hyphenate {
402            props.push_str(" fo:hyphenate=\"true\"");
403        }
404        let master = start.map(|(m, _)| m.clone()).unwrap_or_default();
405        if props.is_empty() && master.is_empty() {
406            return parent;
407        }
408        let key = (parent, props, master);
409        if let Some((_, n)) = self.para_styles.iter().find(|(k, _)| *k == key) {
410            return n.clone();
411        }
412        let n = format!("{}P{}", self.prefix, self.para_styles.len() + 1);
413        self.para_styles.push((key, n.clone()));
414        n
415    }

417    fn plain_inlines(&mut self, inl: &[Inline], out: &mut String) {
418        let mut prev_space = true;
419        let last = inl.len().saturating_sub(1);
420        for (idx, i) in inl.iter().enumerate() {
421            match i {
422                Inline::Text(t, f) => {
423                    let mut body = String::new();
424                    let mut spaces = 0usize;
425                    let flush = |body: &mut String, spaces: &mut usize, prev_space: bool| {
426                        if *spaces == 0 {
427                            return;
428                        }
429                        let (lit, extra) = if prev_space { (0, *spaces) } else { (1, *spaces - 1) };
430                        if lit == 1 {
431                            body.push(' ');
432                        }
433                        if extra > 0 {
434                            body.push_str(&format!("<text:s text:c=\"{extra}\"/>"));
435                        }
436                        *spaces = 0;
437                    };
438                    for c in t.chars() {
439                        if c == ' ' {
440                            spaces += 1;
441                            continue;
442                        }
443                        flush(&mut body, &mut spaces, prev_space);
444                        prev_space = false;
445                        match c {
446                            '\t' => {
447                                // The reader collapses a literal space after a tab, so spaces
448                                // that follow are written as `<text:s/>`.
449                                body.push_str("<text:tab/>");
450                                prev_space = true;
451                            }
452                            '\n' => {
453                                body.push_str("<text:line-break/>");
454                                prev_space = true;
455                            }
456                            '\u{000C}' | '\u{000E}' => {}
457                            c => body.push_str(&x(&c.to_string())),
458                        }
459                    }
460                    if spaces > 0 {
461                        // The reader drops a literal space at the end of a paragraph.
462                        flush(&mut body, &mut spaces, prev_space || idx == last);
463                        prev_space = true;
464                    }
465                    // A script, `file:`, UNC or other non-web target is written as plain text, as
466                    // the HTML and Markdown exporters do (LibreOffice follows `file:`/UNC hrefs).
467                    let link = f.link.as_deref().filter(|l| crate::doc::link::is_safe_href(l));
468                    // A tracked formatting change (§5.5.7): the text between its marks.
469                    let change = f.fmt_change.as_ref().map(|c| self.region("format-change", &c.mark, ""));
470                    if let Some(ct) = &change {
471                        out.push_str(&format!("<text:change-start text:change-id=\"{ct}\"/>"));
472                    }
473                    if let Some(l) = link {
474                        // An index entry's link looks like the entry, not like a web link.
475                        let look = if self.in_index { " text:style-name=\"Index_20_Link\" text:visited-style-name=\"Index_20_Link\"" } else { "" };
476                        out.push_str(&format!("<text:a xlink:type=\"simple\" xlink:href=\"{}\"{look}>", x(l)));
477                    }
478                    let fs = Fmt { link: None, ..f.clone() };
479                    match self.text_style(&fs) {
480                        Some(n) => out.push_str(&format!("<text:span text:style-name=\"{n}\">{body}</text:span>")),
481                        None => out.push_str(&body),
482                    }
483                    if link.is_some() {
484                        out.push_str("</text:a>");
485                    }
486                    if let Some(ct) = change {
487                        out.push_str(&format!("<text:change-end text:change-id=\"{ct}\"/>"));
488                    }
489                }
490                Inline::Image(img) => {
491                    self.image(img, Place::Text(img.float.as_deref()), out);
492                    prev_space = false;
493                }
494                Inline::Group { w, h, float, members } => {
495                    self.group((*w, *h), float.as_deref(), members, out);
496                    prev_space = false;
497                }
498                // A bookmark (§6.2.1): over the text up to its end, or a point; each name once.
499                Inline::Anchor(a) if self.bookmarks.insert(a.clone()) => {
500                    // One ending right where it starts is a point.
501                    let empty = matches!(inl.get(idx + 1), Some(Inline::AnchorEnd(e)) if e == a);
502                    if empty {
503                        self.bookmark_ends.insert(a.clone());
504                    }
505                    let el = if self.ranged.contains(a) && !empty { "bookmark-start" } else { "bookmark" };
506                    out.push_str(&format!("<text:{el} text:name=\"{}\"/>", x(a)));
507                    self.reference_mark(a, inl.get(idx + 1..).unwrap_or(&[]), empty, out);
508                }
509                Inline::Anchor(_) => {}
510                Inline::AnchorEnd(a) => self.anchor_end(a, out),
511                Inline::Note { endnote, custom, blocks, fmt } => {
512                    self.styled_note(*endnote, custom, blocks, fmt, out);
513                    prev_space = false;
514                }
515                Inline::Comment { id, comment } => {
516                    self.comment(*id, comment, out);
517                    prev_space = false;
518                }
519                Inline::Shape { kind, w, h, float, blocks, look, path } => {
520                    self.drawing_shape(*kind, (*w, *h), Place::Text(float.as_deref()), (look, path.as_deref()), blocks, out);
521                    prev_space = false;
522                }
523                // A field around content: an index's is its element (see `index_at`).
524                Inline::FieldStart(_) | Inline::FieldEnd => {}
525                Inline::CommentEnd(id) => out.push_str(&format!("<office:annotation-end office:name=\"{}\"/>", annotation_name(self.prefix, *id))),
526                Inline::Field { instr, result, fmt, locked } => {
527                    match self
528                        .field_element(instr, result)
529                        .and_then(|el| if *locked { super::fields::locked(&el, instr, result) } else { Some(super::fields::unfixed(&el)) })
530                    {
531                        Some(el) => {
532                            match self.text_style(&Fmt { link: None, ..fmt.clone() }) {
533                                Some(n) => out.push_str(&format!("<text:span text:style-name=\"{n}\">{el}</text:span>")),
534                                None => out.push_str(&el),
535                            }
536                            prev_space = false;
537                        }
538                        // No ODF field for it: what it shows, as text.
539                        None => {
540                            self.plain_inlines(&[Inline::Text(result.clone(), fmt.clone())], out);
541                            prev_space = result.ends_with(' ');
542                        }
543                    }
544                }
545                // Inside a tracked change: its base text, which the text path marks as changed.
546                Inline::Ruby { base, fmt, .. } if fmt.ins.is_some() || fmt.del.is_some() => {
547                    self.plain_inlines(&[Inline::Text(base.clone(), fmt.clone())], out);
548                    prev_space = false;
549                }
550                // Ruby (§6.4): the base text in its formatting, the ruby text over it.
551                Inline::Ruby { base, ruby, fmt } => {
552                    self.ruby_used = true;
553                    let base_xml = match self.text_style(&Fmt { link: None, ..fmt.clone() }) {
554                        Some(n) => format!("<text:span text:style-name=\"{n}\">{}</text:span>", x(base)),
555                        None => x(base),
556                    };
557                    out.push_str(&format!(
558                        "<text:ruby text:style-name=\"{}Ru1\"><text:ruby-base>{base_xml}</text:ruby-base><text:ruby-text>{}</text:ruby-text></text:ruby>",
559                        self.prefix,
560                        x(ruby)
561                    ));
562                    prev_space = false;
563                }
564                Inline::Equation { linear, display } => {
565                    self.formula(linear, *display, out);
566                    prev_space = false;
567                }
568                Inline::Figure(_) => {}
569                Inline::Chart(c) => {
570                    self.chart(c, Place::Text(c.float.as_deref()), out);
571                    prev_space = false;
572                }
573                // A form control (in the body: forms belong to the text, ODF 1.3 §13.2), placed
574                // here; elsewhere, or of a kind forms don't have, its text.
575                // A control read from a placeholder (§7.7.2), in the body or anywhere else: a
576                // placeholder while it shows it, else what was typed in its place.
577                Inline::Control { control, text, fmt } if super::forms::is_placeholder(control) => {
578                    if control.showing_placeholder && control.kind == (ControlKind::Text { multi_line: false }) {
579                        // Its description: the control's title.
580                        let start = match control.form_xml.strip_suffix('>') {
581                            Some(head) if !control.title.is_empty() => format!("{head} text:description=\"{}\">", x(&control.title)),
582                            _ => control.form_xml.clone(),
583                        };
584                        let el = format!("{start}{}</text:placeholder>", x(control.placeholder_shown()));
585                        match self.text_style(&Fmt { link: None, ..fmt.clone() }) {
586                            Some(n) => out.push_str(&format!("<text:span text:style-name=\"{n}\">{el}</text:span>")),
587                            None => out.push_str(&el),
588                        }
589                        prev_space = false;
590                    } else {
591                        self.plain_inlines(&[Inline::Text(text.clone(), fmt.clone())], out);
592                        prev_space = text.ends_with(' ');
593                    }
594                }
595                Inline::Control { control, text, fmt } => {
596                    let id = format!("control{}", self.forms.len() + 1);
597                    match super::forms::form_xml(control, text, &id, self.forms.len() + 1).filter(|_| self.forms_allowed) {
598                        Some((xml, w, h)) => {
599                            // Text boxes with a flat border (`fo:border="solid"`, LibreOffice's for
600                            // its own text fields), not its sunken 3D one.
601                            let flat = if xml.starts_with("<form:text ") || xml.starts_with("<form:textarea") { " fo:border=\"solid\"" } else { "" };
602                            self.forms.push(xml);
603                            // Placed by a style, standing on the baseline or raised as it was read
604                            // (LibreOffice puts a control with no placement at the line's top).
605                            let raise = control.form_raise.filter(|r| r.is_finite()).map(|r| r.clamp(-1584.0, 1584.0)).unwrap_or(0.0);
606                            let float = crate::doc::para::Float { wrap: crate::doc::para::Wrap::Inline, y: raise, ..Default::default() };
607                            let (_, props, xy) = drawing::placement(drawing::Place::Text(Some(&float)), w, h);
608                            let style = self.graphic_style(format!("{props}{flat}"));
609                            out.push_str(&format!(
610                                "<draw:control draw:style-name=\"{style}\" text:anchor-type=\"as-char\"{xy} draw:z-index=\"0\" svg:width=\"{}\" svg:height=\"{}\" draw:control=\"{id}\"/>",
611                                pt(w),
612                                pt(h)
613                            ));
614                            prev_space = false;
615                        }
616                        None => {
617                            self.plain_inlines(&[Inline::Text(text.clone(), fmt.clone())], out);
618                            prev_space = text.ends_with(' ');
619                        }
620                    }
621                }
622            }
623        }
624    }

655    /// Markup kept as read (a paragraph's or a table's), with the definitions it names; false
656    /// when one of them can't be written as read (see `super::kept::needed`).
657    pub(in crate::formats::odt) fn kept_markup(&mut self, xml: &str, out: &mut String) -> bool {
658        if !self.kept_defs_of(xml) {
659            return false;
660        }
661        self.close_ins(out);
662        out.push_str(xml);
663        // With the pictures it shows. Its bookmarks' names are taken: the model's don't repeat them.
664        super::kept::hrefs(xml, &mut self.kept_files_used);
665        let (starts, ends) = super::kept::bookmarks(xml);
666        self.bookmarks.extend(starts);
667        self.bookmark_ends.extend(ends);
668        true
669    }

671    /// The definitions markup kept as read names, added to those written; false when one of
672    /// them can't be written as read.
673    fn kept_defs_of(&mut self, xml: &str) -> bool {
674        // The tracked changes it marks, as read, and those their deleted content marks (text
675        // inserted, then deleted); with the styles they name. Not one without its date, which
676        // ODF requires (§4.3.2.10 in `office:change-info`): it is written from the model, dated.
677        let mut todo: Vec<String> = super::kept::changes(xml).into_iter().collect();
678        let mut regions: Vec<(String, std::sync::Arc<str>)> = Vec::new();
679        while let Some(id) = todo.pop() {
680            if self.kept_changes.contains(&id) || regions.iter().any(|(r, _)| *r == id) {
681                continue;
682            }
683            match self.kept_defs.get(&format!("change:{id}")).filter(|r| r.contains("<dc:date>")) {
684                Some(r) if regions.len() < 1000 => {
685                    todo.extend(super::kept::changes(r));
686                    regions.push((id, r.clone()));
687                }
688                _ => return false,
689            }
690        }
691        let mut names = super::kept::names(xml);
692        for (_, r) in &regions {
693            names.extend(super::kept::names(r));
694        }
695        let Some((auto, common, fonts)) = super::kept::needed(&self.kept_defs, names, self.kept_pages) else { return false };
696        // Those their content marks first, as a reader meets them (and as the file had them).
697        for (id, r) in regions.into_iter().rev() {
698            self.kept_changes.insert(id);
699            self.regions.push(r.to_string());
700        }
701        self.kept_used.0.extend(auto);
702        self.kept_used.1.extend(common);
703        self.kept_used.2.extend(fonts);
704        self.sequences.extend(super::kept::sequences(xml));
705        true
706    }

708    fn para_in(&mut self, p: &Para, chain: Option<&Vec<Para>>, out: &mut String) {
709        if chain.is_none() && (self.master.is_none() || self.kept_pages) && self.kept_para(p, out) {
710            // Its own page start, if any, is in its markup.
711            self.master = None;
712            return;
713        }
714        let paras: &[Para] = chain.map(Vec::as_slice).unwrap_or(std::slice::from_ref(p));
715        let start = self.master.take();
716        self.chain = chain.map(|c| c.iter().map(|q| self.para_style(q)).collect()).unwrap_or_default();
717        self.part = 0;
718        let style = self.para_style_at(p, start.as_ref());
719        self.cur_style = style.clone();
720        let tag = match p.kind {
721            Kind::Heading(n) => {
722                // Its own outline level when it sets one apart from its style's.
723                let level = p.extra.as_ref().and_then(|x| x.outline_level).map_or(n.into(), |l| u32::from(l) + 1);
724                out.push_str(&format!("<text:h text:style-name=\"{style}\" text:outline-level=\"{}\">", level.clamp(1, 10)));
725                "</text:h>"
726            }
727            _ => {
728                out.push_str(&format!("<text:p text:style-name=\"{style}\">"));
729                "</text:p>"
730            }
731        };
732        let tracked = paras.len() > 1
733            || p.inlines.iter().any(|i| matches!(i, Inline::Text(_, f) | Inline::Ruby { fmt: f, .. } | Inline::Field { fmt: f, .. } if f.ins.is_some() || f.del.is_some()));
734        if tracked {
735            let mut items = Vec::new();
736            for (k, q) in paras.iter().enumerate() {
737                items.extend(q.inlines.iter().map(Item::Inl));
738                if k + 1 < paras.len() {
739                    items.push(Item::Mark(q));
740                }
741            }
742            self.items(&items, out);
743        } else {
744            self.close_ins(out);
745            self.plain_inlines(&p.inlines, out);
746        }
747        // An inserted paragraph mark: the insertion runs on past the paragraph's end.
748        match paras.last().and_then(|q| q.mark_ins.as_deref()) {
749            Some(m) => self.open_ins(m, out),
750            None => self.close_ins(out),
751        }
752        out.push_str(tag);
753        self.chain.clear();
754    }

780    /// The run of top-level paragraphs `p` begins kept as read whole (an index's, a list's),
781    /// written as it was, when it can be: how many blocks it took.
782    fn kept_run(&mut self, blocks: &[FBlock], i: usize, out: &mut String) -> Option<usize> {
783        let Some(FBlock::Para(p)) = blocks.get(i) else { return None };
784        let ix = p.index.as_deref()?;
785        // Its paragraphs' groups whole (one tied to a paragraph since edited writes it anew).
786        let run = blocks.get(i..i + ix.1.max(1)).unwrap_or(&[]);
787        if !run.iter().all(|b| !matches!(b, FBlock::Para(q) if !self.group_whole(q))) {
788            return None;
789        }
790        if !(self.master.is_none() || self.kept_pages) {
791            return None;
792        }
793        // The sections kept as read its first paragraph opens and its last closes, around it.
794        let mut markup = String::new();
795        if !self.kept_markup(&ix.0, &mut markup) {
796            return None;
797        }
798        // A list without its own `xml:id` (older files' lists) is given one, for a list going on
799        // with it to name.
800        if p.list.is_some()
801            && list_xml_id(&ix.0).is_none()
802            && let Some(at) = markup.find("<text:list").map(|i| i + "<text:list".len())
803        {
804            self.kept_lists += 1;
805            markup.insert_str(at, &format!(" xml:id=\"{}lstk{}\"", self.prefix, self.kept_lists));
806        }
807        let last = match blocks.get(i + ix.1.max(1) - 1) {
808            Some(FBlock::Para(q)) => q.wrap.as_deref().map(|w| (Vec::new(), w.1)),
809            _ => None,
810        };
811        let _ = self.open_wrap(p.wrap.as_deref().map(|w| (w.0.clone(), 0)).as_ref(), out);
812        out.push_str(&markup);
813        self.close_wrap(last.as_ref(), 0, out);
814        self.master = None;
815        // A list kept so: a list continuing it, written from the model, names it.
816        if let Some(l) = p.list
817            && let Some(id) = markup.find("<text:list").and_then(|i| markup.get(i..)).and_then(list_xml_id)
818        {
819            if l.list != 0 {
820                self.list_xml.entry(l.list).or_insert_with(|| id.clone());
821            }
822            self.last_list = Some(id);
823        }
824        out.push('\n');
825        Some(ix.1.max(1))
826    }

854    /// The sections kept as read that open before a top-level block (`wrap`, see `Para::wrap`),
855    /// each once (a copy of the paragraph opening one shares its markup): how many were
856    /// skipped so.
857    fn open_wrap(&mut self, wrap: Option<&(Vec<std::sync::Arc<str>>, u32)>, out: &mut String) -> u32 {
858        let mut skipped = 0u32;
859        for tag in wrap.map(|w| w.0.as_slice()).unwrap_or(&[]) {
860            if !self.wraps_opened.insert(std::sync::Arc::as_ptr(tag).cast::<u8>() as usize) {
861                skipped = skipped.saturating_add(1);
862                continue;
863            }
864            self.close_ins(out);
865            // With the section style it names.
866            self.kept_defs_of(tag);
867            out.push_str(tag);
868            self.sections_open += 1;
869        }
870        skipped
871    }

885    pub(super) fn blocks(&mut self, blocks: &[FBlock], out: &mut String, depth: usize) {
886        let joined = join_deleted_marks(blocks);
887        let (blocks, chains) = match &joined {
888            Some((b, c)) => (b.as_slice(), Some(c)),
889            None => (blocks, None),
890        };
891        let chain = |i: usize| chains.and_then(|c| c.get(&i));
892        let mut i = 0;
893        while let Some(b) = blocks.get(i) {
894            // Only the first top-level block of a section starts its page.
895            if depth == 0 {
896                self.master = self.pending_start.take();
897                // An index or a list kept as read (its paragraphs all as read): as it was.
898                if chains.is_none()
899                    && let Some(n) = self.kept_run(blocks, i, out)
900                {
901                    i += n;
902                    continue;
903                }
904                // A table of contents or another index, with its entries.
905                if let Some((n, kind, titled)) = index_at(blocks, i) {
906                    let joined = chains.is_some_and(|c| (i..i + n).any(|k| c.contains_key(&k)));
907                    if !joined {
908                        self.index(blocks.get(i..i + n).unwrap_or(&[]), &kind, titled, out);
909                        i += n;
910                        continue;
911                    }
912                }
913            }
914            match b {
915                FBlock::Table(t) => {
916                    if depth < model::MAX_DEPTH {
917                        let wrap = t.wrap.as_deref().filter(|_| self.wraps_at(depth));
918                        let skipped = self.open_wrap(wrap, out);
919                        self.table(t, out, depth);
920                        self.close_wrap(wrap, skipped, out);
921                    }
922                    i += 1;
923                }
924                FBlock::Para(p) if p.list.is_some() => {
925                    let start = i;
926                    let id = p.list.map_or(0, |l| l.list);
927                    let (end, kinds) = list_run(blocks, start, self.wraps_at(depth));
928                    i = end;
929                    // The sections kept as read its first item opens and its last closes, around it.
930                    let wrap_of = |k: usize| match blocks.get(k) {
931                        Some(FBlock::Para(q)) => q.wrap.as_deref().filter(|_| self.wraps_at(depth)),
932                        _ => None,
933                    };
934                    let (opens, closes) = (wrap_of(start).map(|w| (w.0.clone(), 0)), wrap_of(end.saturating_sub(1)).map(|w| (Vec::new(), w.1)));
935                    let _ = self.open_wrap(opens.as_ref(), out);
936                    self.list_styles.push(kinds);
937                    let lname = format!("{}L{}", self.prefix, self.list_styles.len());
938                    let id_of_list = id;
939                    // Open lists: per level, whether a list-item is open.
940                    let mut stack: Vec<bool> = Vec::new();
941                    for (qi, q) in blocks.get(start..i).unwrap_or(&[]).iter().enumerate() {
942                        let FBlock::Para(q) = q else { continue };
943                        let lv = q.list.map(|l| l.level.min(8) as usize).unwrap_or(0);
944                        while stack.len() > lv + 1 {
945                            if stack.pop() == Some(true) {
946                                out.push_str("</text:list-item>");
947                            }
948                            out.push_str("</text:list>");
949                        }
950                        while stack.len() < lv + 1 {
951                            if let Some(top) = stack.last_mut()
952                                && !*top
953                            {
954                                out.push_str("<text:list-item>");
955                                *top = true;
956                            }
957                            if stack.is_empty() {
958                                let id = format!("{}lst{}", self.prefix, self.list_styles.len());
959                                // A list cut by a section boundary goes on numbering: continue-list names the
960                                // preceding list by its xml:id, whatever its style name (ODF 1.3 §19.743).
961                                let prev = self.last_list.clone();
962                                let cont = match prev {
963                                    Some(prev) if depth == 0 && start == 0 && std::mem::take(&mut self.continue_list) => {
964                                        format!(" text:continue-list=\"{prev}\"")
965                                    }
966                                    // A list that showed up before goes on with it.
967                                    _ => match self.list_xml.get(&id_of_list) {
968                                        Some(x) if id_of_list != 0 => format!(" text:continue-list=\"{x}\""),
969                                        _ => String::new(),
970                                    },
971                                };
972                                if id_of_list != 0 {
973                                    self.list_xml.entry(id_of_list).or_insert_with(|| id.clone());
974                                }
975                                if depth == 0 {
976                                    self.last_list = Some(id.clone());
977                                }
978                                out.push_str(&format!("<text:list xml:id=\"{id}\" text:style-name=\"{lname}\"{cont}>"));
979                            } else {
980                                out.push_str("<text:list>");
981                            }
982                            stack.push(false);
983                        }
984                        if let Some(top) = stack.last_mut() {
985                            if *top {
986                                out.push_str("</text:list-item>");
987                            }
988                            *top = true;
989                        }
990                        out.push_str("<text:list-item>");
991                        self.para_in(q, chain(start + qi), out);
992                    }
993                    while let Some(open) = stack.pop() {
994                        if open {
995                            out.push_str("</text:list-item>");
996                        }
997                        out.push_str("</text:list>");
998                    }
999                    self.close_wrap(closes.as_ref(), 0, out);
1000                    out.push('\n');
1001                }
1002                FBlock::Para(p) => {
1003                    self.wrapped_para(p, chain(i), out, depth);
1004                    out.push('\n');
1005                    i += 1;
1006                }
1007            }
1008        }
1009    }

1011    /// The ODF element for field `instr` showing `result`, with the date style and sequence
1012    /// declaration it needs; `None` when ODF has none.
1013    fn field_element(&mut self, instr: &str, result: &str) -> Option<String> {
1014        // A citation of a source the document has (§8.1.24).
1015        if let Some(s) = instr.trim_start().strip_prefix("CITATION").and_then(|r| r.split_whitespace().next()).and_then(|t| self.sources.get(t)) {
1016            return Some(super::fields::bib_mark(s, &x(result)));
1017        }
1018        // A custom document property's value (§7.5.6).
1019        // A variable given a value, hidden (§7.4.5): a REF after it shows the value.
1020        if let Some((name, value)) = super::fields::set_of(instr).filter(|(n, _)| self.variables.contains(*n)) {
1021            return Some(format!(
1022                "<text:variable-set text:name=\"{}\" text:display=\"none\" office:value-type=\"string\" office:string-value=\"{}\"/>",
1023                x(name),
1024                x(&value)
1025            ));
1026        }
1027        // A variable's value (§7.4.6).
1028        if let Some(name) = instr.trim_start().strip_prefix("REF").and_then(|r| r.split_whitespace().next()).filter(|n| self.variables.contains(*n)) {
1029            return Some(format!("<text:variable-get text:name=\"{}\">{}</text:variable-get>", x(name), x(result)));
1030        }
1031        // A user field's value (§7.4.9), or a document property's (§7.5.6).
1032        if let Some(n) = super::fields::docproperty_name(instr).filter(|n| self.user_fields.contains(*n)) {
1033            return Some(format!("<text:user-field-get text:name=\"{}\">{}</text:user-field-get>", x(n), x(result)));
1034        }
1035        if let Some(n) = super::fields::docproperty_name(instr).filter(|n| self.custom_props.contains(*n)) {
1036            return Some(format!("<text:user-defined text:name=\"{}\">{}</text:user-defined>", x(n), x(result)));
1037        }
1038        let ds = self.data_style(instr);
1039        if let Some(seq) = super::fields::sequence_name(instr) {
1040            self.sequences.insert(seq.to_string());
1041        }
1042        super::fields::element(instr, &x(result), ds.as_deref())
1043    }

1046    fn data_style(&mut self, instr: &str) -> Option<String> {
1047        let pic = super::datastyle::picture_of(instr)?;
1048        if let Some((_, name, _)) = self.data_styles.iter().find(|(p, ..)| *p == pic) {
1049            return Some(name.clone());
1050        }
1051        let name = format!("{}N{}", self.prefix, self.data_styles.len() + 1);

1053        self.data_styles.push((pic, name.clone(), xml));
1054        Some(name)

1077    /// The automatic styles made so far, without their `office:automatic-styles` element.
1078    pub(super) fn style_defs(&self) -> String {
1079        let mut s: String = self.data_styles.iter().map(|(.., xml)| xml.as_str()).collect();
1080        for (props, n) in &self.text_styles {
1081            let (props, parent) = match props.split_once('\u{1}') {
1082                Some((p, parent)) => (p, format!(" style:parent-style-name=\"{parent}\"")),
1083                None => (props.as_str(), String::new()),
1084            };
1085            s.push_str(&format!("<style:style style:name=\"{n}\" style:family=\"text\"{parent}><style:text-properties{props}/></style:style>"));
1086        }
1087        for ((parent, props, master), n) in &self.para_styles {
1088            let master = if master.is_empty() { String::new() } else { format!(" style:master-page-name=\"{}\"", x(master)) };
1089            let (props, text) =
1090                props.split_once('\u{2}').map_or((props.as_str(), String::new()), |(p, t)| (p, format!("<style:text-properties{t}/>")));
1091            let (attrs, children) = props.split_once('\u{1}').unwrap_or((props, ""));
1092            s.push_str(&format!(
1093                "<style:style style:name=\"{n}\" style:family=\"paragraph\" style:parent-style-name=\"{parent}\"{master}><style:paragraph-properties{attrs}>{children}</style:paragraph-properties>{text}</style:style>"
1094            ));
1095        }
1096        for (i, kinds) in self.list_styles.iter().enumerate() {
1097            s.push_str(&format!("<text:list-style style:name=\"{}L{}\">", self.prefix, i + 1));
1098            for (l, out) in kinds.iter().enumerate() {
1099                let ordered = &out.ordered;
1100                let lv = l + 1;
1101                // The level's own place, else WordCraft's default: 36 pt a level, hanging 18 pt.
1102                let (indent, hanging) = out.label.and_then(|lb| lb.position()).unwrap_or((36.0 * lv as f32, 18.0));
1103                let align = match out.label.and_then(|lb| lb.align) {
1104                    Some(Align::Right) => " fo:text-align=\"end\"",
1105                    Some(Align::Center) => " fo:text-align=\"center\"",
1106                    _ => "",
1107                };
1108                let props = format!(
1109                    "<style:list-level-properties text:list-level-position-and-space-mode=\"label-alignment\"{align}><style:list-level-label-alignment text:label-followed-by=\"listtab\" text:list-tab-stop-position=\"{}\" fo:text-indent=\"{}\" fo:margin-left=\"{}\"/></style:list-level-properties>",
1110                    pt(indent),
1111                    pt(-hanging),
1112                    pt(indent)
1113                );
1114                if *ordered {
1115                    let fmt = ["1", "a", "i"].get(l % 3).copied().unwrap_or("1");
1116                    // The level's own label, else WordCraft's default for the level.
1117                    let attrs = match out.label {
1118                        Some(lb) if lb.format != crate::doc::section::NumFormat::Bullet => {
1119                            let mut a = format!(" style:num-format=\"{}\"", odf_num_format(lb.format));
1120                            for (k, v) in [("style:num-prefix", lb.prefix.as_str()), ("style:num-suffix", lb.suffix.as_str())] {
1121                                if !v.is_empty() {
1122                                    a.push_str(&format!(" {k}=\"{}\"", x(v)));
1123                                }
1124                            }
1125                            if lb.shown > 1 {
1126                                a.push_str(&format!(" text:display-levels=\"{}\"", lb.shown.min(lv as u8)));
1127                            }
1128                            a
1129                        }
1130                        _ => format!(" style:num-suffix=\".\" style:num-format=\"{fmt}\""),
1131                    };
1132                    let start = if out.start != 1 { format!(" text:start-value=\"{}\"", out.start) } else { String::new() };
1133                    s.push_str(&format!("<text:list-level-style-number text:level=\"{lv}\"{attrs}{start}>{props}</text:list-level-style-number>"));
1134                } else {
1135                    let own = out.label.and_then(|lb| lb.bullet()).map(|c| c.to_string());
1136                    let ch = own.as_deref().unwrap_or_else(|| ["•", "◦", "▪"].get(l % 3).copied().unwrap_or("•"));
1137                    let ch = x(ch);
1138                    s.push_str(&format!(
1139                        "<text:list-level-style-bullet text:level=\"{lv}\" text:bullet-char=\"{ch}\">{props}</text:list-level-style-bullet>"
1140                    ));
1141                }
1142            }
1143            s.push_str("</text:list-style>");
1144        }
1145        for (i, widths) in self.tables.iter().enumerate() {
1146            let tn = format!("{}{}", self.prefix, i + 1);
1147            let total: f32 = widths.iter().sum();
1148            let (master, number) = match self.table_starts.get(i).and_then(Option::as_ref) {
1149                Some((m, n)) => {
1150                    (format!(" style:master-page-name=\"{}\"", x(m)), n.map(|n| format!(" style:page-number=\"{n}\"")).unwrap_or_default())
1151                }
1152                None => (String::new(), String::new()),
1153            };
1154            // Aligned, or narrower than the text: as wide as its columns, with its relative width.
1155            let (align, pct) = self.table_looks.get(i).copied().unwrap_or_default();
1156            let brk = if self.table_breaks.get(i).copied().unwrap_or(false) { " fo:break-before=\"page\"" } else { "" };
1157            let align = match align {
1158                Some(Align::Left) => "left",
1159                Some(Align::Center) => "center",
1160                Some(Align::Right) => "right",
1161                _ if pct.is_some() => "left",
1162                _ => "margins",
1163            };
1164            let rel = pct.map(|p| format!(" style:rel-width=\"{}%\"", (p * 100.0).round() / 100.0)).unwrap_or_default();
1165            let space: String = ["fo:margin-top", "fo:margin-bottom"]
1166                .iter()
1167                .zip(self.table_spaces.get(i).copied().unwrap_or_default())
1168                .filter_map(|(k, v)| v.filter(|v| v.is_finite() && *v > 0.0).map(|v| format!(" {k}=\"{}\"", pt(v.min(1584.0)))))
1169                .collect();
1170            s.push_str(&format!(
1171                "<style:style style:name=\"Table{tn}\" style:family=\"table\"{master}><style:table-properties style:width=\"{}\"{rel} table:align=\"{align}\"{space}{number}{brk}/></style:style>",
1172                pt(total)
1173            ));
1174            for (c, w) in widths.iter().enumerate() {
1175                // A table filling the margins gives its columns' shares too (§20.334
1176                // `style:rel-column-width`), so on a narrower page they keep their proportions.
1177                // (LibreOffice lays out an aligned table with shares otherwise: only these.)
1178                let share = if align == "margins" {
1179                    let rel = if w.is_finite() { (w.clamp(0.0, 1584.0) * 100.0).round().max(1.0) as u32 } else { 1 };
1180                    format!(" style:rel-column-width=\"{rel}*\"")
1181                } else {
1182                    String::new()
1183                };
1184                s.push_str(&format!(
1185                    "<style:style style:name=\"Table{tn}.C{}\" style:family=\"table-column\"><style:table-column-properties style:column-width=\"{}\"{share}/></style:style>",
1186                    c + 1,
1187                    pt(*w)
1188                ));
1189            }
1190        }
1191        if self.ruby_used {
1192            s.push_str(&format!(
1193                "<style:style style:name=\"{}Ru1\" style:family=\"ruby\"><style:ruby-properties style:ruby-position=\"above\" style:ruby-align=\"center\"/></style:style>",
1194                self.prefix
1195            ));
1196        }
1197        for (i, props) in self.row_styles.iter().enumerate() {
1198            s.push_str(&format!(
1199                "<style:style style:name=\"{}Row{}\" style:family=\"table-row\"><style:table-row-properties{props}/></style:style>",
1200                self.prefix,
1201                i + 1
1202            ));
1203        }
1204        for (i, props) in self.graphic_styles.iter().enumerate() {
1205            // A text box's frame derives from LibreOffice's Frame style, without which
1206            // LibreOffice ignores its border (`fo:border`) and outlines it as a drawing shape.
1207            let (parent, props) = if let Some(rest) = props.strip_prefix(FRAME_PARENT) {
1208                (" style:parent-style-name=\"Frame\"", rest)
1209            } else if let Some(rest) = props.strip_prefix(GRAPHICS_PARENT) {
1210                (" style:parent-style-name=\"Graphics\"", rest)
1211            } else {
1212                ("", props.as_str())
1213            };
1214            // Laid out where WordCraft places it, not held in the text's area: said, as the
1215            // standard gives no default (§20.267) and a file's default style may say otherwise.
1216            let flow = if props.contains("style:flow-with-text") { "" } else { " style:flow-with-text=\"false\"" };
1217            s.push_str(&format!(
1218                "<style:style style:name=\"{}fr{}\" style:family=\"graphic\"{parent}><style:graphic-properties{props}{flow}/></style:style>",
1219                self.prefix,
1220                i + 1
1221            ));
1222        }
1223        s.push_str(&self.cell_style_defs());
1224        for st in &self.section_styles {
1225            s.push_str(st);
1226        }
1227        s
1228    }
