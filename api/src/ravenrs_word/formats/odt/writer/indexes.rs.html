1//! Tables of contents and other generated indexes written (ODF 1.3 §8): a WordCraft table — a
2//! `TOC`, `INDEX` or `BIBLIOGRAPHY` field in its heading followed by its entries, or the field
3//! around its entries — becomes the ODF index with its title, its source (what LibreOffice
4//! regenerates it from) and its entries as last generated.
5
6use super::{FBlock, Inline, Para, Writer, x};

8/// What a Word index field stands for in ODF.
9#[derive(Clone, Debug, PartialEq)]
10pub(in crate::formats::odt) enum IndexKind {
11    /// A table of contents of headings down to this outline level (`TOC \o "1-n"`).
12    Contents(u8),
13    /// A table of figures (or of any captioned thing): `TOC \c "Figure"`.
14    Captions(String),
15    /// An alphabetical index (`INDEX`).
16    Alphabetical,
17    /// A bibliography (`BIBLIOGRAPHY`).
18    Bibliography,
19}

21/// The quoted (or bare) argument of switch `\sw` in field code `instr`.
22fn switch_arg(instr: &str, sw: char) -> Option<String> {
23    let at = instr.find(&format!("\\{sw}"))?;
24    let rest = instr.get(at + 2..)?.trim_start();
25    let arg = match rest.strip_prefix('"') {
26        Some(q) => q.split('"').next()?,
27        None => rest.split_whitespace().next()?,
28    };
29    (!arg.starts_with('\\')).then(|| arg.chars().take(64).collect())
30}
31
32/// The index field code `instr` is, if it is one.
33pub(in crate::formats::odt) fn index_kind(instr: &str) -> Option<IndexKind> {
34    let name = instr.split_whitespace().next()?.to_ascii_uppercase();
35    match name.as_str() {
36        "TOC" => match switch_arg(instr, 'c').or_else(|| switch_arg(instr, 'C')) {
37            Some(c) => Some(IndexKind::Captions(c)),
38            None => {
39                // `\o "1-3"`: the deepest level shown (3 when none is given, as Word's default).
40                let deepest =
41                    switch_arg(instr, 'o').and_then(|r| r.rsplit('-').next().and_then(|n| n.trim().parse::<u8>().ok())).unwrap_or(3).clamp(1, 10);
42                Some(IndexKind::Contents(deepest))
43            }
44        },
45        "INDEX" => Some(IndexKind::Alphabetical),
46        "BIBLIOGRAPHY" => Some(IndexKind::Bibliography),
47        _ => None,
48    }
49}

51/// The field code an ODF index stands for in Word (the reverse of [`index_kind`]).
52pub(in crate::formats::odt) fn index_instr(kind: &IndexKind) -> String {
53    match kind {
54        IndexKind::Contents(n) => format!("TOC \\o \"1-{n}\" \\h \\z \\u"),
55        IndexKind::Captions(c) => format!("TOC \\h \\z \\c \"{}\"", c.replace('"', "")),
56        IndexKind::Alphabetical => "INDEX".into(),
57        IndexKind::Bibliography => "BIBLIOGRAPHY".into(),
58    }
59}

61/// Whether `p`'s paragraph style is an index entry's (WordCraft's TOC, table of figures, index
62/// and bibliography styles).
63fn is_entry(p: &Para) -> bool {
64    let style = p.extra.as_ref().and_then(|x| x.style.as_deref()).unwrap_or("");
65    (style.starts_with("TOC") && style != "TOCHeading") || style == "TableofFigures" || style.starts_with("Index") || style == "Bibliography"
66}

68/// An index starting at `blocks[i]`: how many blocks it takes, its kind, and whether its first
69/// block is its title. Either a paragraph holding an index field followed by its entries (as
70/// WordCraft inserts a table), or a field starting a paragraph and ending in a later one.
71pub(in crate::formats::odt) fn index_at(blocks: &[FBlock], i: usize) -> Option<(usize, IndexKind, bool)> {
72    let Some(FBlock::Para(p)) = blocks.get(i) else { return None };
73    // A heading holding the field, anywhere in it.
74    if let Some(kind) = p.inlines.iter().find_map(|inl| match inl {
75        Inline::Field { instr, .. } => index_kind(instr),
76        _ => None,
77    }) {
78        let entries =
79            blocks.get(i + 1..).unwrap_or(&[]).iter().take_while(|b| matches!(b, FBlock::Para(q) if is_entry(q) && q.list.is_none())).count();
80        return Some((entries + 1, kind, true));
81    }
82    // A field starting the paragraph.
83    for inline in &p.inlines {
84        match inline {
85            Inline::FieldStart(instr) => {
86                let kind = index_kind(instr)?;
87                // Its end: where the fields opened from here close.
88                let mut open = 0usize;
89                for (k, b) in blocks.get(i..).unwrap_or(&[]).iter().enumerate() {
90                    let FBlock::Para(q) = b else { return None };
91                    for inl in &q.inlines {
92                        match inl {
93                            Inline::FieldStart(_) => open += 1,
94                            Inline::FieldEnd => {
95                                open = open.saturating_sub(1);
96                                if open == 0 {
97                                    return Some((k + 1, kind, false));
98                                }
99                            }
100                            _ => {}
101                        }
102                    }
103                }
104                return None;
105            }
106            Inline::Text(t, _) if t.trim().is_empty() => {}
107            Inline::Anchor(_) => {}
108            _ => return None,
109        }
110    }
111    None
112}

114impl Writer {
115    /// The index `blocks` of `kind` (its first block its title when `titled`).
116    pub(in crate::formats::odt) fn index(&mut self, blocks: &[FBlock], kind: &IndexKind, titled: bool, out: &mut String) {
117        self.indexes += 1;
118        let n = self.indexes;
119        let name = format!("{}Index{n}", self.prefix);
120        let (element, source) = match kind {
121            IndexKind::Contents(levels) => {
122                // An entry per level as LibreOffice makes it: linked text, a right tab with dots
123                // and the page number.
124                let templates: String = (1..=*levels)
125                    .map(|l| {
126                        format!(
127                            "<text:table-of-content-entry-template text:outline-level=\"{l}\" text:style-name=\"Contents_20_{l}\"><text:index-entry-link-start/><text:index-entry-chapter/><text:index-entry-text/><text:index-entry-tab-stop style:type=\"right\" style:leader-char=\".\"/><text:index-entry-page-number/><text:index-entry-link-end/></text:table-of-content-entry-template>"
128                        )
129                    })
130                    .collect();
131                (
132                    "text:table-of-content",
133                    format!("<text:table-of-content-source text:outline-level=\"{levels}\">{templates}</text:table-of-content-source>"),
134                )
135            }
136            IndexKind::Captions(c) => (
137                "text:illustration-index",
138                format!("<text:illustration-index-source text:caption-sequence-name=\"{}\" text:caption-sequence-format=\"text\"/>", x(c)),
139            ),
140            IndexKind::Alphabetical => ("text:alphabetical-index", "<text:alphabetical-index-source/>".to_string()),
141            IndexKind::Bibliography => ("text:bibliography", "<text:bibliography-source/>".to_string()),
142        };
143        out.push_str(&format!("<{element} text:name=\"{name}\">{source}<text:index-body>"));
144        let (title, entries) = if titled { blocks.split_at(blocks.len().min(1)) } else { (&[][..], blocks) };
145        if !title.is_empty() {
146            out.push_str(&format!("<text:index-title text:name=\"{name}_Head\">"));
147            self.blocks(title, out, 1);
148            out.push_str("</text:index-title>");
149        }
150        self.in_index = true;
151        self.blocks(entries, out, 1);
152        self.in_index = false;
153        self.close_ins(out);
154        out.push_str(&format!("</text:index-body></{element}>\n"));
155    }
