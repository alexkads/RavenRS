1//! Tracked changes written (ODF 1.3 §5.5): changed regions, insertions open around the text, and
2//! deleted content, paragraph marks included.
3
4use super::{FBlock, HashMap, Inline, Para, RevMark, Writer, x};

6/// One piece of a paragraph being written: an inline, or the deleted mark of a paragraph
7/// joined to the next one.
8pub(in crate::formats::odt) enum Item<'a> {
9    Inl(&'a Inline),
10    Mark(&'a Para),
11}
12
13impl Item<'_> {
14    /// Its tracked insertion and deletion.
15    pub(in crate::formats::odt) fn marks(&self) -> (Option<&RevMark>, Option<&RevMark>) {
16        match self {
17            Item::Inl(Inline::Text(_, f) | Inline::Ruby { fmt: f, .. } | Inline::Field { fmt: f, .. }) => (f.ins.as_ref(), f.del.as_ref()),
18            Item::Inl(_) => (None, None),
19            Item::Mark(p) => (p.mark_ins.as_deref(), p.mark_del.as_deref()),
20        }
21    }
22
23    /// What a run of content is written as: deleted content is one region whatever was
24    /// inserted inside it, so only its deletion counts.
25    pub(in crate::formats::odt) fn segment(&self) -> (Option<&RevMark>, Option<&RevMark>) {
26        match self.marks() {
27            (_, Some(del)) => (None, Some(del)),
28            k => k,
29        }
30    }
31
32    pub(in crate::formats::odt) fn inline(&self) -> Option<&Inline> {
33        match self {
34            Item::Inl(i) => Some(i),
35            Item::Mark(_) => None,
36        }
37    }
38}

40/// Close the insertion open in `st`.
41pub(in crate::formats::odt) fn ins_close(st: &mut Option<(RevMark, String)>, out: &mut String) {
42    if let Some((_, ct)) = st.take() {
43        out.push_str(&format!("<text:change-end text:change-id=\"{ct}\"/>"));
44    }
45}

47/// The blocks after joining at deleted marks, and each joined run's paragraphs by the first one's index.
48pub(in crate::formats::odt) type JoinedMarks = (Vec<FBlock>, HashMap<usize, Vec<Para>>);
49
50/// `blocks` with each run of paragraphs joined at tracked deleted paragraph marks (ODF keeps
51/// such a deletion inside one body paragraph): the first paragraph stands for the run, and the
52/// whole run is returned by its index. `None` when nothing is joined. A deleted mark with no
53/// paragraph after it (before a table, at the end) is written as a plain mark.
54pub(in crate::formats::odt) fn join_deleted_marks(blocks: &[FBlock]) -> Option<JoinedMarks> {
55    if !blocks.iter().any(|b| matches!(b, FBlock::Para(p) if p.mark_del.is_some())) {
56        return None;
57    }
58    let mut out = Vec::with_capacity(blocks.len());
59    let mut chains = HashMap::new();
60    let mut i = 0;
61    while let Some(b) = blocks.get(i) {
62        i += 1;
63        let FBlock::Para(p) = b else {
64            out.push(b.clone());
65            continue;
66        };
67        let mut chain = vec![p.clone()];
68        while chain.last().is_some_and(|q| q.mark_del.is_some())
69            && let Some(FBlock::Para(q)) = blocks.get(i)
70        {
71            chain.push(q.clone());
72            i += 1;
73        }
74        if chain.len() > 1 {
75            chains.insert(out.len(), chain);
76        }
77        out.push(FBlock::Para(p.clone()));
78    }
79    Some((out, chains))
80}

82/// Now, as an `xsd:dateTime` (UTC).
83pub(in crate::formats::odt) fn now_iso() -> String {
84    let t = now_unix();
85    let (days, secs) = (t.div_euclid(86_400), t.rem_euclid(86_400));
86    // Reuse the crate's 400-year-cycle converter (chrono within a cycle, cycles carried in the
87    // year) so a clock past chrono's range keeps the proleptic Gregorian date the old hand-rolled
88    // code gave, never the epoch. `{y:04}` rather than `%Y`, which writes a `+` before year 10000.
89    let (y, m, d) = crate::formats::sheet::civil(days);
90    format!("{y:04}-{m:02}-{d:02}T{:02}:{:02}:{:02}Z", secs / 3600, secs / 60 % 60, secs % 60)
91}

93/// Seconds since the Unix epoch: the system clock, or the browser's on the web.
94pub(in crate::formats::odt) fn now_unix() -> i64 {
95    #[cfg(not(all(target_arch = "wasm32", target_os = "unknown")))]
96    {
97        use std::time::{SystemTime, UNIX_EPOCH};
98        SystemTime::now().duration_since(UNIX_EPOCH).ok().and_then(|d| i64::try_from(d.as_secs()).ok()).unwrap_or(0)
99    }
100    // `SystemTime::now()` panics on wasm32-unknown-unknown, so ask the browser's clock.
101    #[cfg(all(target_arch = "wasm32", target_os = "unknown"))]
102    {
103        let ms = js_sys::Date::now();
104        if ms.is_finite() && ms > 0.0 { (ms / 1000.0) as i64 } else { 0 }
105    }
106}

108impl Writer {
109    /// A new `text:changed-region` of `kind` by `m` with `body` after its change info; its id.
110    pub(in crate::formats::odt) fn region(&mut self, kind: &str, m: &RevMark, body: &str) -> String {
111        // (With the writer's prefix: a file's own regions, written as read, keep their ids.)
112        let id = format!("{}ct{}", self.prefix, self.regions.len() + 1);
113        // `dc:date` is required: a change without a date is dated now.
114        let date = if m.date.trim().is_empty() { now_iso() } else { m.date.clone() };
115        self.regions.push(format!(
116            "<text:changed-region xml:id=\"{id}\" text:id=\"{id}\"><text:{kind}><office:change-info><dc:creator>{}</dc:creator><dc:date>{}</dc:date></office:change-info>{body}</text:{kind}></text:changed-region>",
117            x(&m.author),
118            x(&date)
119        ));
120        id
121    }

123    /// Make `m` the insertion open in `st` (closing another one first).
124    pub(in crate::formats::odt) fn ins_open(&mut self, st: &mut Option<(RevMark, String)>, m: &RevMark, out: &mut String) {
125        if st.as_ref().is_some_and(|(o, _)| o == m) {
126            return;
127        }
128        ins_close(st, out);
129        let ct = self.region("insertion", m, "");
130        out.push_str(&format!("<text:change-start text:change-id=\"{ct}\"/>"));
131        *st = Some((m.clone(), ct));
132    }
133
134    pub(in crate::formats::odt) fn open_ins(&mut self, m: &RevMark, out: &mut String) {
135        let mut st = self.open_ins.take();
136        self.ins_open(&mut st, m, out);
137        self.open_ins = st;
138    }
139
140    pub(in crate::formats::odt) fn close_ins(&mut self, out: &mut String) {
141        ins_close(&mut self.open_ins, out);
142    }
143
144    /// Inlines with their insertions as change ranges kept in `st`.
145    pub(in crate::formats::odt) fn ins_inlines(&mut self, items: &[Item<'_>], st: &mut Option<(RevMark, String)>, out: &mut String) {
146        let mut k = 0;
147        while k < items.len() {
148            let ins = items.get(k).and_then(|i| i.marks().0);
149            let mut j = k + 1;
150            while j < items.len() && items.get(j).and_then(|i| i.marks().0) == ins {
151                j += 1;
152            }
153            match ins {
154                Some(m) => self.ins_open(st, m, out),
155                None => ins_close(st, out),
156            }
157            let seg: Vec<Inline> = items.get(k..j).unwrap_or(&[]).iter().filter_map(|i| i.inline().cloned()).collect();
158            self.plain_inlines(&seg, out);
159            k = j;
160        }
161    }

163    /// Deleted content: a region holding it (one paragraph per deleted paragraph mark plus
164    /// one; text inserted before it was deleted as change ranges inside) and a `text:change`
165    /// point in the body.
166    pub(in crate::formats::odt) fn deletion(&mut self, seg: &[Item<'_>], m: &RevMark, out: &mut String) {
167        let mut body = String::new();
168        let mut st: Option<(RevMark, String)> = None;
169        let mut start = 0;
170        let mut first = true;
171        loop {
172            let end = seg.iter().skip(start).position(|i| matches!(i, Item::Mark(_))).map(|e| start + e);
173            if !first {
174                self.part += 1;
175            }
176            first = false;
177            let style = self.chain.get(self.part).cloned().unwrap_or_else(|| self.cur_style.clone());
178            body.push_str(&format!("<text:p text:style-name=\"{style}\">"));
179            self.ins_inlines(seg.get(start..end.unwrap_or(seg.len())).unwrap_or(&[]), &mut st, &mut body);
180            // A deleted paragraph mark that had been inserted: the range runs on past it.
181            match end.and_then(|e| seg.get(e)).and_then(|i| i.marks().0) {
182                Some(ins) => self.ins_open(&mut st, ins, &mut body),
183                None => ins_close(&mut st, &mut body),
184            }
185            body.push_str("</text:p>");
186            match end {
187                Some(e) => start = e + 1,
188                None => break,
189            }
190        }
191        let ct = self.region("deletion", m, &body);
192        out.push_str(&format!("<text:change text:change-id=\"{ct}\"/>"));
193    }

195    /// Paragraph content, with tracked insertions as change ranges and deletions as change points.
196    pub(in crate::formats::odt) fn items(&mut self, items: &[Item<'_>], out: &mut String) {
197        let mut k = 0;
198        while k < items.len() {
199            let key = items.get(k).map(Item::segment).unwrap_or_default();
200            let mut j = k + 1;
201            while j < items.len() && items.get(j).map(Item::segment).unwrap_or_default() == key {
202                j += 1;
203            }
204            let seg = items.get(k..j).unwrap_or(&[]);
205            match key {
206                (_, Some(del)) => self.deletion(seg, del, out),
207                (Some(ins), None) => {
208                    self.open_ins(ins, out);
209                    let inl: Vec<Inline> = seg.iter().filter_map(|i| i.inline().cloned()).collect();
210                    self.plain_inlines(&inl, out);
211                }
212                (None, None) => {
213                    self.close_ins(out);
214                    let inl: Vec<Inline> = seg.iter().filter_map(|i| i.inline().cloned()).collect();
215                    self.plain_inlines(&inl, out);
216                }
217            }
218            k = j;
219        }
220    }
