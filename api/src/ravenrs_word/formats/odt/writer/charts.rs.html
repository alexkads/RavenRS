1//! Charts written as ODF chart objects (ODF 1.3 §10.4.3 `draw:object` holding an `office:chart`
2//! sub-document, Part 3 §11), shaped like LibreOffice's: the chart's data in its own table
3//! (`local-table`, labels in the first row and column), each series pointing at its column, its
4//! colour the theme accent WordCraft draws it in.

13/// The table's column `i` (from 0) as a cell address names it: `A`, `B`… `Z`, `AA`…
14fn column(i: usize) -> String {
15    let mut n = i + 1;
16    let mut s = Vec::new();
17    while n > 0 {
18        let d = (n - 1) % 26;
19        s.push(b'A' + d as u8);
20        n = (n - 1) / 26;
21    }
22    s.iter().rev().map(|&b| char::from(b)).collect()
23}
24
25/// `local-table.$C$2:.$C$5`: column `col`'s rows 2 to `n + 1` (the data under its label).
26fn data_range(col: usize, n: usize) -> String {
27    let c = column(col);
28    format!("local-table.${c}$2:.${c}${}", n + 1)
29}
30
31/// Most slices given their own colour (the rest take the chart's own colours).
32const MAX_SLICE_STYLES: usize = 64;
33
34/// The `i`th series' (or slice's) colour, `#rrggbb`: the accents in turn.
35fn accent(accents: &[Rgb], i: usize) -> String {
36    match accents.get(i.checked_rem(accents.len()).unwrap_or(0)) {
37        Some(c) => format!("#{}", c.hex().to_ascii_lowercase()),
38        None => "#4472c4".into(),
39    }
40}
41
42/// A number as a table cell's `office:value`: gaps are `NaN`, as LibreOffice writes them.
43fn number(v: Option<f64>) -> String {
44    match v.filter(|v| v.is_finite()) {
45        Some(v) => format!("{v}"),
46        None => "NaN".into(),
47    }
48}

50/// A chart sub-document's `content.xml` for `spec`, `w` × `h` points, series in `accents`.
51pub(in crate::formats::odt) fn chart_xml(spec: &ChartSpec, w: f32, h: f32, accents: &[Rgb]) -> String {
52    let spec = spec.clone().sanitized();
53    let (class, series_class) = match spec.kind {
54        ChartType::Column | ChartType::ColumnStacked | ChartType::Bar | ChartType::BarStacked => ("bar", "bar"),
55        ChartType::Line => ("line", "line"),
56        ChartType::Area => ("area", "area"),
57        ChartType::Pie => ("circle", "circle"),
58        ChartType::Doughnut => ("ring", "circle"),
59        ChartType::Scatter => ("scatter", "scatter"),
60    };
61    let vertical = matches!(spec.kind, ChartType::Bar | ChartType::BarStacked);
62    let stacked = matches!(spec.kind, ChartType::ColumnStacked | ChartType::BarStacked);
63    let labels = if spec.data_labels { " chart:data-label-number=\"value\"" } else { " chart:data-label-number=\"none\"" };
64    let n = spec.series.iter().map(|s| s.values.len()).max().unwrap_or(0).max(spec.categories.len());
65    // Scatter charts are points without lines.
66    let points = if spec.kind == ChartType::Scatter { " chart:lines=\"false\" chart:symbol-type=\"automatic\"" } else { "" };
67    // A white wall, labelled axes in grey, a legend without a frame: as WordCraft draws them.
68    let mut styles = format!(
69        "<style:style style:name=\"plot\" style:family=\"chart\"><style:chart-properties chart:vertical=\"{vertical}\" chart:stacked=\"{stacked}\"{labels}{points}/></style:style>\
70         <style:style style:name=\"wall\" style:family=\"chart\"><style:graphic-properties draw:stroke=\"none\" draw:fill=\"none\"/></style:style>\
71         <style:style style:name=\"axis\" style:family=\"chart\"><style:chart-properties chart:display-label=\"true\"/><style:graphic-properties svg:stroke-color=\"#b3b3b3\"/><style:text-properties fo:font-size=\"8pt\"/></style:style>\
72         <style:style style:name=\"round\" style:family=\"chart\"><style:chart-properties chart:reverse-direction=\"true\"/></style:style>\
73         <style:style style:name=\"grid\" style:family=\"chart\"><style:graphic-properties svg:stroke-color=\"#d9d9d9\"/></style:style>\
74         <style:style style:name=\"legend\" style:family=\"chart\"><style:graphic-properties draw:stroke=\"none\" draw:fill=\"none\"/><style:text-properties fo:font-size=\"8pt\"/></style:style>"
75    );
76    // A pie's slices take the accents in turn.
77    if spec.kind.is_round() {
78        for k in 0..n.min(MAX_SLICE_STYLES) {
79            styles.push_str(&format!(
80                "<style:style style:name=\"p{k}\" style:family=\"chart\"><style:graphic-properties draw:stroke=\"none\" draw:fill-color=\"{}\"/></style:style>",
81                accent(accents, k)
82            ));
83        }
84    }
85    let mut series = String::new();
86    for i in 0..spec.series.len() {
87        let col = i + 1;
88        let fill = accent(accents, i);
89        let line = match spec.kind {
90            ChartType::Line => format!(" draw:stroke=\"solid\" svg:stroke-color=\"{fill}\" svg:stroke-width=\"0.06cm\""),
91            _ => " draw:stroke=\"none\"".to_string(),
92        };
93        styles.push_str(&format!(
94            "<style:style style:name=\"s{col}\" style:family=\"chart\"><style:chart-properties{labels}/><style:graphic-properties draw:fill-color=\"{fill}\"{line}/></style:style>"
95        ));
96        // A scatter series' x values are the first column's.
97        let domain = if spec.kind == ChartType::Scatter && n > 0 {
98            format!("<chart:domain table:cell-range-address=\"{}\"/>", data_range(0, n))
99        } else {
100            String::new()
101        };
102        let values = if n > 0 { format!(" chart:values-cell-range-address=\"{}\"", data_range(col, n)) } else { String::new() };
103        let slices: String = if spec.kind.is_round() {
104            (0..n.min(MAX_SLICE_STYLES)).map(|k| format!("<chart:data-point chart:style-name=\"p{k}\"/>")).collect()
105        } else {
106            String::new()
107        };
108        series.push_str(&format!(
109            "<chart:series chart:style-name=\"s{col}\"{values} chart:label-cell-address=\"local-table.${}$1\" chart:class=\"chart:{series_class}\">{domain}{slices}</chart:series>",
110            column(col)
111        ));
112    }
113    let title = spec.title.as_deref().map(|t| format!("<chart:title><text:p>{}</text:p></chart:title>", x(t))).unwrap_or_default();
114    let legend = spec
115        .legend
116        .map(|p| {
117            let pos = match p {
118                LegendPos::Right => "end",
119                LegendPos::Left => "start",
120                LegendPos::Top => "top",
121                LegendPos::Bottom => "bottom",
122            };
123            format!("<chart:legend chart:legend-position=\"{pos}\" chart:style-name=\"legend\"/>")
124        })
125        .unwrap_or_default();
126    let categories = if n > 0 { format!("<chart:categories table:cell-range-address=\"{}\"/>", data_range(0, n)) } else { String::new() };
127    let axes = format!(
128        "<chart:axis chart:dimension=\"x\" chart:name=\"primary-x\" chart:style-name=\"axis\">{categories}</chart:axis><chart:axis chart:dimension=\"y\" chart:name=\"primary-y\" chart:style-name=\"{}\">{}</chart:axis>",
129        // A pie runs clockwise from the top, as WordCraft draws it.
130        if spec.kind.is_round() { "round" } else { "axis" },
131        if spec.kind.is_round() { "" } else { "<chart:grid chart:class=\"major\" chart:style-name=\"grid\"/>" }
132    );
133    // The table: the series' names over their columns, a row per category.
134    let mut table =
135        String::from("<table:table table:name=\"local-table\"><table:table-header-columns><table:table-column/></table:table-header-columns>");
136    if !spec.series.is_empty() {
137        table.push_str(&format!(
138            "<table:table-columns><table:table-column table:number-columns-repeated=\"{}\"/></table:table-columns>",
139            spec.series.len()
140        ));
141    }
142    table.push_str("<table:table-header-rows><table:table-row><table:table-cell><text:p/></table:table-cell>");
143    for s in &spec.series {
144        table.push_str(&format!("<table:table-cell office:value-type=\"string\"><text:p>{}</text:p></table:table-cell>", x(&s.name)));
145    }
146    table.push_str("</table:table-row></table:table-header-rows><table:table-rows>");
147    for k in 0..n {
148        let cat = spec.categories.get(k).map(String::as_str).unwrap_or("");
149        table.push_str("<table:table-row>");
150        // A scatter chart's x values are numbers.
151        match cat.trim().parse::<f64>().ok().filter(|v| v.is_finite() && spec.kind == ChartType::Scatter) {
152            Some(v) => table.push_str(&format!(
153                "<table:table-cell office:value-type=\"float\" office:value=\"{v}\"><text:p>{}</text:p></table:table-cell>",
154                x(cat)
155            )),
156            None => table.push_str(&format!("<table:table-cell office:value-type=\"string\"><text:p>{}</text:p></table:table-cell>", x(cat))),
157        }
158        for s in &spec.series {
159            let v = number(s.values.get(k).copied().flatten());
160            table.push_str(&format!("<table:table-cell office:value-type=\"float\" office:value=\"{v}\"><text:p>{v}</text:p></table:table-cell>"));
161        }
162        table.push_str("</table:table-row>");
163    }
164    table.push_str("</table:table-rows></table:table>");
165    format!(
166        "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:document-content xmlns:office=\"urn:oasis:names:tc:opendocument:xmlns:office:1.0\" xmlns:style=\"urn:oasis:names:tc:opendocument:xmlns:style:1.0\" xmlns:text=\"urn:oasis:names:tc:opendocument:xmlns:text:1.0\" xmlns:table=\"urn:oasis:names:tc:opendocument:xmlns:table:1.0\" xmlns:draw=\"urn:oasis:names:tc:opendocument:xmlns:drawing:1.0\" xmlns:chart=\"urn:oasis:names:tc:opendocument:xmlns:chart:1.0\" xmlns:svg=\"urn:oasis:names:tc:opendocument:xmlns:svg-compatible:1.0\" xmlns:fo=\"urn:oasis:names:tc:opendocument:xmlns:xsl-fo-compatible:1.0\" xmlns:xlink=\"http://www.w3.org/1999/xlink\" office:version=\"1.3\"><office:automatic-styles>{styles}</office:automatic-styles><office:body><office:chart><chart:chart svg:width=\"{}\" svg:height=\"{}\" chart:class=\"chart:{class}\">{title}{legend}<chart:plot-area chart:style-name=\"plot\" chart:data-source-has-labels=\"both\">{axes}{series}<chart:wall chart:style-name=\"wall\"/></chart:plot-area>{table}</chart:chart></office:chart></office:body></office:document-content>\n",
167        pt(w),
168        pt(h)
169    )
170}

172/// Chart object `o` (as read) still reads as chart `spec`: the chart hasn't been edited.
173fn unchanged(o: &model::OdfObject, spec: &ChartSpec) -> bool {
174    let Some((_, content)) = o.files.iter().find(|(p, _)| p == "content.xml") else { return false };
175    let Some(mut read) = super::super::chart::read_chart(content) else { return false };
176    read.sanitize();
177    let mut spec = spec.clone();
178    spec.sanitize();
179    read == spec
180}

182impl Writer {
183    /// A chart: its frame here, placed as its float says, and its chart object in the package.
184    pub(in crate::formats::odt) fn chart(&mut self, c: &model::Chart, place: Place<'_>, out: &mut String) {
185        // Unchanged since it was read: the chart object as it was, with all its look.
186        let kept = c.source.as_deref().filter(|o| unchanged(o, &c.spec));
187        let (n, inner) = match kept {
188            Some(o) => {
189                let (n, _, link) = self.kept_object_files(o);
190                (n, link)
191            }
192            None => {
193                let (n, name) = self.object_name();
194                self.objects.push((name.clone(), chart_xml(&c.spec, c.w, c.h, &self.accents), "application/vnd.oasis.opendocument.chart"));
195                let link =
196                    format!("<draw:object xlink:href=\"./{}\" xlink:type=\"simple\" xlink:show=\"embed\" xlink:actuate=\"onLoad\"/>", x(&name));
197                (n, link)
198            }
199        };
200        // Always placed by a style (see `image_frame`).
201        let (anchor, props, xy) = placement(place, c.w, c.h);
202        let (anchor, style) = (anchor, format!(" draw:style-name=\"{}\"{xy}", self.graphic_style(props)));
203        let z = place.z_index();
204        out.push_str(&format!(
205            "<draw:frame draw:name=\"{}Chart{n}\"{style}{anchor} svg:width=\"{}\" svg:height=\"{}\" draw:z-index=\"{z}\">{inner}",
206            self.prefix,
207            pt(c.w),
208            pt(c.h),
209        ));
210        if !c.alt.is_empty() {
211            out.push_str(&format!("<svg:desc>{}</svg:desc>", x(&c.alt)));
212        }
213        out.push_str("</draw:frame>");
214    }
