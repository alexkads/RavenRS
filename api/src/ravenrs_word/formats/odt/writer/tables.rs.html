5/// Word's default cell margins (top, left, bottom, right), points.
6pub(in crate::formats::odt) const WORD_CELL_MARGINS: [f32; 4] = [0.0, 5.4, 0.0, 5.4];
7
8/// The cell of `row` that starts at grid column `g`.
9pub(in crate::formats::odt) fn cell_starting_at(row: &[model::Cell], g: usize) -> Option<&model::Cell> {
10    let mut s = 0usize;
11    row.iter().find(|x| {
12        let here = s == g;
13        s += x.colspan.clamp(1, 63) as usize;
14        here
15    })
16}

18/// The `style:table-cell-properties` attributes of cell `c` of row `ri` of `t`, starting at grid
19/// column `g0` of `cols`: its padding, shading, vertical alignment and borders. Each edge two
20/// cells share is written once, as LibreOffice writes it (every cell its left and bottom, the
21/// first row its top, the last column its right), so lines don't double; a side the cell
22/// doesn't state gets the table's grid line, or none in a table without lines.
23pub(in crate::formats::odt) fn cell_props(t: &FTable, ri: usize, g0: usize, c: &model::Cell, cols: usize) -> String {
24    let grid = (!t.borderless).then_some(Border { style: BorderStyle::Single, width: 0.5, color: Some(Rgb(0, 0, 0)), space: 0.0 });
25    let line = |c: &model::Cell, pick: fn(&Borders) -> Option<Border>| match c.borders.as_ref().and_then(pick) {
26        Some(b) => Some(b).filter(|b| b.is_visible()),
27        None => grid,
28    };
29    let row = t.rows.get(ri).map(Vec::as_slice).unwrap_or(&[]);
30    let left_cell = g0.checked_sub(1).and_then(|g| {
31        let mut s = 0usize;
32        row.iter().find(|x| {
33            let end = s + x.colspan.clamp(1, 63) as usize;
34            let hit = s <= g && g < end;
35            s = end;
36            hit
37        })
38    });
39    let below = t.rows.get(ri + 1).and_then(|r| cell_starting_at(r, g0));
40    let left = line(c, |b| b.left).or_else(|| left_cell.and_then(|l| line(l, |b| b.right)));
41    let bottom = line(c, |b| b.bottom).or_else(|| below.and_then(|b| line(b, |x| x.top)));
42    let top = if ri == 0 { line(c, |b| b.top) } else { None };
43    let right = if g0 + c.colspan.clamp(1, 63) as usize >= cols { line(c, |b| b.right) } else { None };
44    let m = c.margins.filter(|m| m.iter().all(|v| v.is_finite())).unwrap_or(WORD_CELL_MARGINS);
45    let mut p = format!(
46        " fo:padding-top=\"{}\" fo:padding-left=\"{}\" fo:padding-bottom=\"{}\" fo:padding-right=\"{}\"",
47        pt(m[0].max(0.0)),
48        pt(m[1].max(0.0)),
49        pt(m[2].max(0.0)),
50        pt(m[3].max(0.0))
51    );
52    p.push_str(&format!(
53        " fo:border-left=\"{}\" fo:border-right=\"{}\" fo:border-top=\"{}\" fo:border-bottom=\"{}\"",
54        odf_border(left),
55        odf_border(right),
56        odf_border(top),
57        odf_border(bottom)
58    ));
59    if let Some(bg) = c.shading {
60        p.push_str(&format!(" fo:background-color=\"#{}\"", bg.hex().to_ascii_lowercase()));
61    }
62    p.push_str(match c.valign {
63        Some(VAlign::Top) => " style:vertical-align=\"top\"",
64        Some(VAlign::Center) => " style:vertical-align=\"middle\"",
65        Some(VAlign::Bottom) => " style:vertical-align=\"bottom\"",
66        None => "",
67    });
68    // Which way its text runs (§20.404; upward is LibreOffice's own `bt-lr`).
69    p.push_str(match c.direction {
70        crate::doc::props::TextDirection::Horizontal => "",
71        crate::doc::props::TextDirection::Down => " style:writing-mode=\"tb-rl\"",
72        crate::doc::props::TextDirection::Up => " loext:writing-mode=\"bt-lr\"",
73    });
74    p
75}

77/// Separates a cell style's `style:table-cell-properties` attributes from its content (a
78/// background picture) in [`Writer::cell_style`]'s key.
79pub(in crate::formats::odt) const CELL_CHILD: char = '\u{1}';
80
81impl Writer {
82    /// A cell's background picture (§17.3 `style:background-image`), the picture written into
83    /// the package (once for the same bytes).
84    fn cell_picture(&mut self, pic: &model::CellPic) -> String {
85        // The same picture in many cells is one file.
86        let hash = pic.img.data.iter().fold(0xcbf2_9ce4_8422_2325_u64, |h, b| (h ^ u64::from(*b)).wrapping_mul(0x0100_0000_01b3));
87        let name = format!("Pictures/cell-{hash:016x}.{}", pic.img.ext);
88        if !self.pictures.iter().any(|(n, _, _)| *n == name) {
89            self.pictures.push((name.clone(), pic.img.data.clone(), super::mime_of(&pic.img.ext).to_string()));
90        }
91        let repeat = match pic.repeat {
92            crate::doc::props::PictureRepeat::None => "no-repeat",
93            crate::doc::props::PictureRepeat::Tile => "repeat",
94            crate::doc::props::PictureRepeat::Stretch => "stretch",
95        };
96        let pos = |a: f32, words: [&'static str; 3]| {
97            if a < 0.25 {
98                words[0]
99            } else if a > 0.75 {
100                words[2]
101            } else {
102                words[1]
103            }
104        };
105        format!(
106            "<style:background-image xlink:href=\"{}\" xlink:type=\"simple\" xlink:actuate=\"onLoad\" style:position=\"{} {}\" style:repeat=\"{repeat}\"/>",
107            x(&name),
108            pos(pic.align[1], ["top", "center", "bottom"]),
109            pos(pic.align[0], ["left", "center", "right"]),
110        )
111    }

113    /// The automatic style of a cell with `props` (its `style:table-cell-properties` attributes).
114    pub(in crate::formats::odt) fn cell_style(&mut self, props: String) -> String {
115        let k = match self.cell_styles.iter().position(|s| *s == props) {
116            Some(k) => k,
117            None => {
118                self.cell_styles.push(props);
119                self.cell_styles.len() - 1
120            }
121        };
122        format!("{}Cell{}", self.prefix, k + 1)
123    }

125    pub(in crate::formats::odt) fn table(&mut self, t: &FTable, out: &mut String, depth: usize) {
126        // As read, while it is what was read.
127        if let Some(xml) = &t.kept
128            && (self.master.is_none() || self.kept_pages)
129            && self.kept_markup(xml, out)
130        {
131            self.master = None;
132            out.push('\n');
133            return;
134        }
135        self.close_ins(out);
136        let cols = t.cols().max(1);
137        let widths: Vec<f32> = if t.widths.len() == cols && t.widths.iter().all(|w| w.is_finite() && *w > 0.0) {
138            t.widths.clone()
139        } else {
140            vec![468.0 / cols as f32; cols]
141        };
142        self.tables.push(widths);
143        self.table_looks.push(t.extra.as_ref().map(|x| (x.align, x.width_pct.filter(|p| *p < 100.0))).unwrap_or_default());
144        // Its first paragraph starting a page: the table does (§20.191).
145        let first = t.rows.first().and_then(|r| r.first()).and_then(|c| c.blocks.first());
146        self.table_spaces.push(t.extra.as_ref().map(|x| x.space).unwrap_or_default());
147        self.table_breaks.push(matches!(first, Some(FBlock::Para(p)) if p.page_break));
148        self.table_starts.push(self.master.take());
149        let tn = format!("{}{}", self.prefix, self.tables.len());
150        out.push_str(&format!("<table:table table:name=\"Table{tn}\" table:style-name=\"Table{tn}\">"));
151        for c in 0..cols {
152            out.push_str(&format!("<table:table-column table:style-name=\"Table{tn}.C{}\"/>", c + 1));
153        }
154        // The rows repeated on each page (a table style's header row alone repeats nothing).
155        let header_rows = match &t.extra {
156            Some(x) => x.header_rows.min(t.rows.len()),
157            None => t.rows.iter().take_while(|r| !r.is_empty() && r.iter().all(|c| c.header)).count(),
158        };
159        for (ri, row) in t.rows.iter().enumerate() {
160            if ri == 0 && header_rows > 0 {
161                out.push_str("<table:table-header-rows>");
162            }
163            // A row of a set height: exact (`style:row-height`) or at least (`style:min-row-height`).
164            let height = match t.extra.as_ref().and_then(|x| x.heights.get(ri).copied().flatten()) {
165                Some((h, true)) => format!(" style:row-height=\"{}\"", pt(h)),
166                Some((h, false)) => format!(" style:min-row-height=\"{}\"", pt(h)),
167                None => String::new(),
168            };
169            // A row that doesn't break across pages (§20.247 on a row). One that may says so where
170            // the file's default row style, which stays, would keep it together.
171            let keep = if t.extra.as_ref().is_some_and(|x| x.keep_rows.get(ri) == Some(&true)) {
172                " fo:keep-together=\"always\""
173            } else if self.kept_defs.get("common~style:default-style~table-row").is_some_and(|d| d.contains("keep-together") && d.contains("always"))
174            {
175                " fo:keep-together=\"auto\""
176            } else {
177                ""
178            };
179            let props = format!("{height}{keep}");
180            if props.is_empty() {
181                out.push_str("<table:table-row>");
182            } else {
183                let k = match self.row_styles.iter().position(|s| *s == props) {
184                    Some(k) => k,
185                    None => {
186                        self.row_styles.push(props);
187                        self.row_styles.len() - 1
188                    }
189                };
190                out.push_str(&format!("<table:table-row table:style-name=\"{}Row{}\">", self.prefix, k + 1));
191            }
192            let mut g = 0usize;
193            for c in row {
194                let span = c.colspan.clamp(1, 63) as usize;
195                g += span;
196                if c.covered {
197                    for _ in 0..span {
198                        out.push_str("<table:covered-table-cell/>");
199                    }
200                    continue;
201                }
202                let mut props = cell_props(t, ri, g - span, c, cols);
203                if let Some(pic) = &c.picture {
204                    props.push(CELL_CHILD);
205                    props.push_str(&self.cell_picture(pic));
206                }
207                let st = self.cell_style(props);
208                // A formula field: the cell's formula, and the value it shows (§19.646, §19.388).
209                let formula = crate::formats::odt::cell_formula::of_cell(&c.blocks).and_then(|(instr, result)| {
210                    Some((crate::formats::odt::cell_formula::odf_of(instr, (ri, g - span), (t.rows.len(), cols))?, result))
211                });
212                let value = match formula {
213                    Some((f, result)) => match result.trim().parse::<f64>().ok().filter(|v| v.is_finite()) {
214                        Some(v) => format!(" table:formula=\"{}\" office:value-type=\"float\" office:value=\"{v}\"", x(&f)),
215                        None => format!(" table:formula=\"{}\"", x(&f)),
216                    },
217                    None => match crate::formats::odt::cell_formula::number_of(&c.blocks) {
218                        // A number: formulas calculate with it.
219                        Some(v) => format!(" office:value-type=\"float\" office:value=\"{v}\""),
220                        None => " office:value-type=\"string\"".to_string(),
221                    },
222                };
223                let mut attrs = format!(" table:style-name=\"{st}\"{value}");
224                if span > 1 {
225                    attrs.push_str(&format!(" table:number-columns-spanned=\"{span}\""));
226                }
227                if c.rowspan > 1 {
228                    attrs.push_str(&format!(" table:number-rows-spanned=\"{}\"", c.rowspan));
229                }
230                out.push_str(&format!("<table:table-cell{attrs}>"));
231                if c.blocks.is_empty() {
232                    out.push_str("<text:p text:style-name=\"Standard\"/>");
233                }
234                // With the sections kept around its blocks, closed in it.
235                self.story_blocks(&c.blocks, out, depth + 1);
236                self.close_ins(out);
237                out.push_str("</table:table-cell>");
238                for _ in 1..span {
239                    out.push_str("<table:covered-table-cell/>");
240                }
241            }
242            for _ in g..cols {
243                out.push_str("<table:table-cell office:value-type=\"string\"><text:p/></table:table-cell>");
244            }
245            out.push_str("</table:table-row>");
246            if ri + 1 == header_rows {
247                out.push_str("</table:table-header-rows>");
248            }
249        }
250        out.push_str("</table:table>\n");
251    }
