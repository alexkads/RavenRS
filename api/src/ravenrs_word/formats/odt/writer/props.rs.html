6/// `style:text-properties` attributes for what only [`Fmt::extra`] holds (underline and strike
7/// are written with the rest); `size` is the text's size, which a raised position is a share of.
8pub(in crate::formats::odt) fn extra_props(x: &crate::doc::props::CharProps, size: Option<f32>) -> String {
9    let mut p = String::new();
10    let on = |v: Option<bool>| v == Some(true);
11    if on(x.caps) {
12        p.push_str(" fo:text-transform=\"uppercase\"");
13    } else if x.caps == Some(false) {
14        p.push_str(" fo:text-transform=\"none\"");
15    }
16    if on(x.small_caps) {
17        p.push_str(" fo:font-variant=\"small-caps\"");
18    } else if x.small_caps == Some(false) {
19        p.push_str(" fo:font-variant=\"normal\"");
20    }
21    if on(x.hidden) {
22        p.push_str(" text:display=\"none\"");
23    }
24    if let Some(k) = x.kern.filter(|k| k.is_finite()) {
25        p.push_str(&format!(" style:letter-kerning=\"{}\"", k > 0.0));
26    }
27    if let Some(s) = x.spacing.filter(|s| s.is_finite() && *s != 0.0) {
28        p.push_str(&format!(" fo:letter-spacing=\"{}\"", pt(s)));
29    }
30    if let Some(s) = x.scale.filter(|s| s.is_finite() && *s > 0.0 && *s != 100.0) {
31        p.push_str(&format!(" style:text-scale=\"{}%\"", s.round()));
32    }
33    if let Some(m) = x.emphasis {
34        use crate::doc::props::EmphasisMark as E;
35        p.push_str(match m {
36            E::None => " style:text-emphasize=\"none\"",
37            E::Dot => " style:text-emphasize=\"dot above\"",
38            E::Comma => " style:text-emphasize=\"accent above\"",
39            E::Circle => " style:text-emphasize=\"circle above\"",
40            E::UnderDot => " style:text-emphasize=\"dot below\"",
41        });
42    }
43    if let Some(r) = x.position.filter(|r| r.is_finite() && *r != 0.0) {
44        let size = size.filter(|s| s.is_finite() && *s >= 1.0).unwrap_or(12.0);
45        let pct = r / size * 100.0;
46        p.push_str(&format!(" style:text-position=\"{}% 100%\"", (pct * 100.0).round() / 100.0));
47    }
48    if on(x.outline) {
49        p.push_str(" style:text-outline=\"true\"");
50    }
51    if on(x.shadow) {
52        p.push_str(" fo:text-shadow=\"1pt 1pt\"");
53    }
54    if on(x.emboss) {
55        p.push_str(" style:font-relief=\"embossed\"");
56    } else if on(x.engrave) {
57        p.push_str(" style:font-relief=\"engraved\"");
58    }
59    p.push_str(&script_attrs(x));
60    if let Some(l) = x.lang.as_deref() {
61        p.push_str(&lang_attrs(l, ""));
62    }
63    p
64}

66/// `style:text-properties` attributes for character properties `cp` (a style's look).
67pub(in crate::formats::odt) fn props_text(cp: &crate::doc::props::CharProps) -> String {
68    use crate::doc::props::{TextColor, Underline, VertAlign};
69    let f = Fmt {
70        bold: cp.bold == Some(true),
71        italic: cp.italic == Some(true),
72        underline: cp.underline.is_some_and(|u| u != Underline::None),
73        strike: cp.strike == Some(true),
74        sup: cp.vert_align == Some(VertAlign::Superscript),
75        sub: cp.vert_align == Some(VertAlign::Subscript),
76        color: match cp.color {
77            Some(TextColor::Rgb(c)) => Some(c),
78            _ => None,
79        },
80        size: cp.size,
81        font: cp.font.as_deref().map(std::sync::Arc::from),
82        extra: Some(Box::new(cp.clone())),
83        ..Default::default()
84    };
85    text_props(&f)
86}

88/// `fo:text-align` for `a` (ODF 1.3 §20.223).
89pub(in crate::formats::odt) fn align_attr(a: Option<Align>) -> &'static str {
90    match a {
91        Some(Align::Center) => " fo:text-align=\"center\"",
92        Some(Align::Right) => " fo:text-align=\"end\"",
93        Some(Align::Justify) | Some(Align::Distribute) => " fo:text-align=\"justify\"",
94        Some(Align::Left) => " fo:text-align=\"start\"",
95        None => "",
96    }
97}

99/// `fo:language`, `fo:script` and `fo:country` (with `suffix`: `-asian`, `-complex` on `style:`)
100/// for BCP 47 tag `tag`. A script is the four-letter subtag after the language.
101pub(in crate::formats::odt) fn lang_attrs(tag: &str, suffix: &str) -> String {
102    let mut parts = tag.split(['-', '_']);
103    let ok = |s: &&str| (1..=8).contains(&s.len()) && s.chars().all(|c| c.is_ascii_alphanumeric());
104    let ns = if suffix.is_empty() { "fo" } else { "style" };
105    let mut p = String::new();
106    if let Some(lang) = parts.next().filter(ok) {
107        p.push_str(&format!(" {ns}:language{suffix}=\"{lang}\""));
108        let mut next = parts.next().filter(ok);
109        if let Some(script) = next.filter(|s| is_script(s)) {
110            p.push_str(&format!(" {ns}:script{suffix}=\"{script}\""));
111            next = parts.next().filter(ok);
112        }
113        if let Some(country) = next {
114            p.push_str(&format!(" {ns}:country{suffix}=\"{country}\""));
115        }
116    }
117    p
118}

120/// Whether a language tag's subtag is a script (BCP 47: four letters).
121pub(in crate::formats::odt) fn is_script(subtag: &str) -> bool {
122    subtag.len() == 4 && subtag.chars().all(|c| c.is_ascii_alphabetic())
123}

125/// The Asian and complex-script attributes of `x`: fonts, complex size, weight, posture and the
126/// two languages.
127pub(in crate::formats::odt) fn script_attrs(x: &crate::doc::props::CharProps) -> String {
128    let mut p = String::new();
129    if let Some(f) = x.font_ea.as_deref() {
130        p.push_str(&format!(" style:font-family-asian=\"'{}'\"", crate::formats::odt::writer::x(&f.replace('\'', ""))));
131    }
132    if let Some(f) = x.font_cs.as_deref() {
133        p.push_str(&format!(" style:font-family-complex=\"'{}'\"", crate::formats::odt::writer::x(&f.replace('\'', ""))));
134    }
135    if let Some(s) = x.size_cs.filter(|s| s.is_finite()) {
136        p.push_str(&format!(" style:font-size-complex=\"{}\"", pt(s)));
137    }
138    if let Some(b) = x.bold_cs {
139        p.push_str(if b { " style:font-weight-complex=\"bold\"" } else { " style:font-weight-complex=\"normal\"" });
140    }
141    if let Some(i) = x.italic_cs {
142        p.push_str(if i { " style:font-style-complex=\"italic\"" } else { " style:font-style-complex=\"normal\"" });
143    }
144    if let Some(l) = x.lang_ea.as_deref() {
145        p.push_str(&lang_attrs(l, "-asian"));
146    }
147    if let Some(l) = x.lang_bidi.as_deref() {
148        p.push_str(&lang_attrs(l, "-complex"));
149    }
150    p
151}

153pub(in crate::formats::odt) fn text_props(f: &Fmt) -> String {
154    let mut p = String::new();
155    // Complex-script weight and posture of their own are written with the other script extras.
156    let own_cs = |pick: fn(&crate::doc::props::CharProps) -> Option<bool>| f.extra.as_deref().and_then(pick).is_some();
157    let extra = f.extra.as_deref();
158    // What it turns off that its style, or the style that is based on, may turn on: said so, or
159    // that bold, italic, underline or strike-through would show through.
160    let off = |pick: fn(&crate::doc::props::CharProps) -> Option<bool>| extra.and_then(pick) == Some(false);
161    if f.bold {
162        p.push_str(" fo:font-weight=\"bold\" style:font-weight-asian=\"bold\"");
163        if !own_cs(|x| x.bold_cs) {
164            p.push_str(" style:font-weight-complex=\"bold\"");
165        }
166    } else if off(|x| x.bold) {
167        p.push_str(" fo:font-weight=\"normal\" style:font-weight-asian=\"normal\"");
168        if !own_cs(|x| x.bold_cs) {
169            p.push_str(" style:font-weight-complex=\"normal\"");
170        }
171    }
172    if f.italic {
173        p.push_str(" fo:font-style=\"italic\" style:font-style-asian=\"italic\"");
174        if !own_cs(|x| x.italic_cs) {
175            p.push_str(" style:font-style-complex=\"italic\"");
176        }
177    } else if off(|x| x.italic) {
178        p.push_str(" fo:font-style=\"normal\" style:font-style-asian=\"normal\"");
179        if !own_cs(|x| x.italic_cs) {
180            p.push_str(" style:font-style-complex=\"normal\"");
181        }
182    }
183    if f.underline {
184        use crate::doc::props::Underline as U;
185        let (style, kind, width, mode) = match extra.and_then(|e| e.underline) {
186            Some(U::Double) => ("solid", "double", "auto", ""),
187            Some(U::Thick) => ("solid", "single", "bold", ""),
188            Some(U::Dotted) => ("dotted", "single", "auto", ""),
189            Some(U::Dash) => ("dash", "single", "auto", ""),
190            Some(U::DotDash) => ("dot-dash", "single", "auto", ""),
191            Some(U::DotDotDash) => ("dot-dot-dash", "single", "auto", ""),
192            Some(U::Wave) => ("wave", "single", "auto", ""),
193            Some(U::DoubleWave) => ("wave", "double", "auto", ""),
194            Some(U::Words) => ("solid", "single", "auto", " style:text-underline-mode=\"skip-white-space\""),
195            _ => ("solid", "single", "auto", ""),
196        };
197        let color =
198            extra.and_then(|e| e.underline_color).map(|c| format!("#{}", c.hex().to_ascii_lowercase())).unwrap_or_else(|| "font-color".into());
199        p.push_str(&format!(
200            " style:text-underline-style=\"{style}\" style:text-underline-type=\"{kind}\" style:text-underline-width=\"{width}\" style:text-underline-color=\"{color}\"{mode}"
201        ));
202    }
203    if !f.underline && extra.is_some_and(|e| e.underline == Some(crate::doc::props::Underline::None)) {
204        p.push_str(" style:text-underline-style=\"none\"");
205    }
206    if f.strike || extra.is_some_and(|e| e.double_strike == Some(true)) {
207        p.push_str(" style:text-line-through-style=\"solid\"");
208        if extra.is_some_and(|e| e.double_strike == Some(true)) {
209            p.push_str(" style:text-line-through-type=\"double\"");
210        }
211    } else if off(|x| x.strike) {
212        p.push_str(" style:text-line-through-style=\"none\"");
213    }
214    if let Some(e) = extra {
215        if (f.sup || f.sub) && e.position.is_some() {
216            // One `style:text-position` only (a repeated attribute isn't well-formed XML):
217            // superscript or subscript, written below, wins over a raised or lowered position.
218            let e = crate::doc::props::CharProps { position: None, ..e.clone() };
219            p.push_str(&extra_props(&e, f.size));
220        } else {
221            p.push_str(&extra_props(e, f.size));
222        }
223    }
224    // A border around the text (Word's `w:bdr`): LibreOffice's text border, which ODF 1.3 has no
225    // attribute of its own for (`loext:border`, `loext:padding`).
226    if let Some(b) = extra.and_then(|e| e.border).filter(|b| b.is_visible()) {
227        p.push_str(&format!(" loext:border=\"{}\" loext:padding=\"{}\"", odf_border(Some(b)), pt(b.space.clamp(0.0, 31.0))));
228    }
229    if f.sup {
230        p.push_str(" style:text-position=\"super 58%\"");
231    } else if f.sub {
232        p.push_str(" style:text-position=\"sub 58%\"");
233    }
234    if let Some(c) = f.color {
235        p.push_str(&format!(" fo:color=\"#{}\"", c.hex().to_ascii_lowercase()));
236    }
237    if let Some(c) = f.background {
238        p.push_str(&format!(" fo:background-color=\"#{}\"", c.hex().to_ascii_lowercase()));
239    }
240    // One size in WordCraft's model is every script's in ODF's, as bold is: without these,
241    // LibreOffice shows Asian and complex-script text at the default size whatever was chosen.
242    if let Some(s) = f.size {
243        p.push_str(&format!(" fo:font-size=\"{0}\" style:font-size-asian=\"{0}\"", pt(s)));
244        if f.extra.as_deref().and_then(|x| x.size_cs).is_none() {
245            p.push_str(&format!(" style:font-size-complex=\"{}\"", pt(s)));
246        }
247    }
248    if f.code {
249        p.push_str(&format!(" fo:font-family=\"'{}'\" style:font-family-generic=\"modern\" style:font-pitch=\"fixed\"", model::MONO_FONT));
250    } else if let Some(fam) = &f.font {
251        p.push_str(&format!(" fo:font-family=\"'{}'\"", x(&fam.replace('\'', ""))));
252    }
253    p
254}

256/// `style:paragraph-properties` attributes for the spacing, indents and line spacing `sp` sets
257/// (ODF 1.3 §20.204–209, §20.225, §20.317).
258pub(in crate::formats::odt) fn spacing_props(sp: &model::Spacing) -> String {
259    let mut p = String::new();
260    for (k, v) in [
261        ("fo:margin-top", sp.before),
262        ("fo:margin-bottom", sp.after),
263        ("fo:margin-left", sp.left),
264        ("fo:margin-right", sp.right),
265        ("fo:text-indent", sp.first),
266    ] {
267        if let Some(v) = v.filter(|v| v.is_finite()) {
268            p.push_str(&format!(" {k}=\"{}\"", pt(v)));
269        }
270    }
271    match sp.line {
272        Some(LineSpacing::Multiple(m)) if m.is_finite() => p.push_str(&format!(" fo:line-height=\"{}%\"", (m * 100.0).round().max(1.0))),
273        Some(LineSpacing::Exactly(h)) if h.is_finite() => p.push_str(&format!(" fo:line-height=\"{}\"", pt(h))),
274        Some(LineSpacing::AtLeast(h)) if h.is_finite() => p.push_str(&format!(" style:line-height-at-least=\"{}\"", pt(h))),
275        _ => {}
276    }
277    p
278}

280/// `style:tab-stops` (ODF 1.3 §17.7) for `tabs`, measured from the margin (the package's
281/// settings say so); bar and clearing stops have no ODF form and are left out.
282pub(in crate::formats::odt) fn tab_stops(tabs: &[crate::doc::props::TabStop]) -> String {
283    use crate::doc::props::{TabAlign, TabLeader};
284    let mut s = String::from("<style:tab-stops>");
285    for t in tabs.iter().filter(|t| t.pos.is_finite()) {
286        let kind = match t.align {
287            TabAlign::Left => "",
288            TabAlign::Center => " style:type=\"center\"",
289            TabAlign::Right => " style:type=\"right\"",
290            TabAlign::Decimal => " style:type=\"char\" style:char=\".\"",
291            TabAlign::Bar | TabAlign::Clear => continue,
292        };
293        let leader = match t.leader {
294            TabLeader::None => "",
295            TabLeader::Dot => " style:leader-style=\"dotted\" style:leader-text=\".\"",
296            TabLeader::Hyphen => " style:leader-style=\"solid\" style:leader-text=\"-\"",
297            TabLeader::Underscore => " style:leader-style=\"solid\" style:leader-text=\"_\"",
298            TabLeader::MiddleDot => " style:leader-style=\"dotted\" style:leader-text=\"·\"",
299        };
300        s.push_str(&format!("<style:tab-stop style:position=\"{}\"{kind}{leader}/>", pt(t.pos)));
301    }
302    s.push_str("</style:tab-stops>");
303    s
304}

306/// An ODF style name (an XML NCName, ODF 1.3 §19.75) for document style id `id`: letters,
307/// digits and `-`, `.` kept, anything else as `_xHH_` (its code point in hex), and a leading
308/// digit, `-` or `.` escaped too.
309pub(in crate::formats::odt) fn odf_name(id: &str) -> String {
310    if let Some((name, _)) = generated_style(id) {
311        return name;
312    }
313    let mut out = String::new();
314    for (i, c) in id.chars().enumerate() {
315        let ok = c.is_alphanumeric() || (i > 0 && (c == '-' || c == '.' || c.is_ascii_digit()));
316        if ok && !(i == 0 && c.is_ascii_digit()) {
317            out.push(c);
318        } else {
319            out.push_str(&format!("_x{:X}_", c as u32));
320        }
321    }
322    if out.is_empty() { "Custom".into() } else { out }
323}

325/// `fo:border-*` and `fo:padding-*` attributes for paragraph borders `b`, when a side shows.
326pub(in crate::formats::odt) fn border_attrs(b: Option<Borders>) -> String {
327    let mut p = String::new();
328    // A side without a line may still hold padding (read from ODF), which keeps its room.
329    let padded = |l: &Border| !l.is_visible() && l.space.is_finite() && l.space > 0.0;
330    let Some(b) = b.filter(|b| b.any_visible() || [b.top, b.left, b.bottom, b.right].iter().flatten().any(padded)) else { return p };
331    for (side, line) in [("top", b.top), ("left", b.left), ("bottom", b.bottom), ("right", b.right)] {
332        let shown = line.filter(|l| l.is_visible());
333        p.push_str(&format!(" fo:border-{side}=\"{}\"", odf_border(shown)));
334        if let Some(l) = shown {
335            p.push_str(&format!(" fo:padding-{side}=\"{}\"", pt(l.space.clamp(0.0, 31.0))));
336        } else if let Some(l) = line.filter(padded) {
337            p.push_str(&format!(" fo:padding-{side}=\"{}\"", pt(l.space.min(1584.0))));
338        }
339    }
340    p
341}

343/// An ODF border value (ODF 1.3 §20.183) for `b`; `none` for no line.
344pub(in crate::formats::odt) fn odf_border(b: Option<Border>) -> String {
345    let Some(b) = b.filter(|b| b.is_visible() && b.width.is_finite()) else { return "none".into() };
346    let style = match b.style {
347        BorderStyle::Double | BorderStyle::Triple => "double",
348        BorderStyle::Dotted => "dotted",
349        BorderStyle::Dashed => "dashed",
350        BorderStyle::DotDash => "dot-dash",
351        _ => "solid",
352    };
353    let color = b.color.unwrap_or(Rgb(0, 0, 0)).hex().to_ascii_lowercase();
354    format!("{} {style} #{color}", pt(b.width.clamp(0.05, 12.0)))
355}

372/// The LibreOffice style (name and display name) standing for WordCraft's table of contents,
373/// table of figures and index heading styles, which LibreOffice's indexes are generated in:
374/// TOC 1–9 are Contents 1–9, TOC Heading is Contents Heading, Table of Figures is Figure Index 1.
375pub(in crate::formats::odt) fn generated_style(id: &str) -> Option<(String, String)> {
376    let n = id.strip_prefix("TOC").and_then(|n| n.parse::<u8>().ok()).filter(|n| (1..=9).contains(n));
377    Some(match (id, n) {
378        (_, Some(n)) => (format!("Contents_20_{n}"), format!("Contents {n}")),
379        ("TOCHeading", _) => ("Contents_20_Heading".into(), "Contents Heading".into()),
380        ("TableofFigures", _) => ("Figure_20_Index_20_1".into(), "Figure Index 1".into()),
381        _ => return None,
382    })
383}

385/// The WordCraft style (id and name) a LibreOffice index style `odt` stands for (the reverse
386/// of [`generated_style`]).
387pub(in crate::formats::odt) fn generating_style(odt: &str) -> Option<(String, String)> {
388    let n = odt.strip_prefix("Contents_20_").and_then(|n| n.parse::<u8>().ok()).filter(|n| (1..=10).contains(n));
389    Some(match (odt, n) {
390        (_, Some(n)) => (format!("TOC{}", n.min(9)), format!("TOC {}", n.min(9))),
391        ("Contents_20_Heading", _) => ("TOCHeading".into(), "TOC Heading".into()),
392        ("Figure_20_Index_20_1", _) => ("TableofFigures".into(), "Table of Figures".into()),
393        _ => return None,
394    })
395}
