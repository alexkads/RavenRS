2//! `office:document` element, its pictures inline (`office:binary-data`, base64) and its

12/// The top-level parts of a flat document, in the order a package's files take them.

14const SETTINGS: &str = "office:settings";
15const FONTS: &str = "office:font-face-decls";
16const STYLES: &str = "office:styles";
17const AUTOMATIC: &str = "office:automatic-styles";

21/// Most pictures and formulas taken out of a flat document (hostile files).
22const MAX_INLINE: usize = 10_000;

24/// Element `name` in `xml` (its first one, start to end, or self-closed); "" when there is none.
25fn element<'a>(xml: &'a str, name: &str) -> &'a str {
26    let open = format!("<{name}");
27    let Some(start) = xml
28        .match_indices(&open)
29        .map(|(i, _)| i)
30        .find(|&i| xml.get(i + open.len()..).and_then(|r| r.chars().next()).is_some_and(|c| c == '>' || c == '/' || c.is_whitespace()))
31    else {
32        return "";
33    };
34    let Some(tag_end) = xml.get(start..).and_then(|r| r.find('>')).map(|e| start + e) else { return "" };
35    if xml.get(..tag_end).is_some_and(|t| t.ends_with('/')) {
36        return xml.get(start..=tag_end).unwrap_or("");
37    }
38    // Its end: the close tag at its own depth (a chart object inside holds an `office:body`,
39    // an `office:document` of its own).
40    let close = format!("</{name}>");
41    let mut depth = 0usize;
42    let mut at = tag_end;
43    // Every step moves past a tag, so this ends; nesting is bounded like any XML here (deeper is
44    // malformed), however many elements of the name sit side by side within.
45    loop {
46        let rest = xml.get(at..).unwrap_or("");
47        let next_close = rest.find(&close);
48        // Only opens before that close count (and only they are looked for).
49        let before = next_close.and_then(|c| rest.get(..c)).unwrap_or(rest);
50        let next_open = before.match_indices(&open).map(|(i, _)| i).find(|&i| {
51            let after = rest.get(i + open.len()..).unwrap_or("");
52            after.chars().next().is_some_and(|c| c == '>' || c == '/' || c.is_whitespace())
53                // Not self-closed.
54                && after.find('>').and_then(|e| after.get(..e)).is_some_and(|t| !t.ends_with('/'))
55        });
56        match (next_open, next_close) {
57            (Some(o), Some(c)) if o < c => {
58                depth += 1;
59                if depth > super::limits::MAX_XML_DEPTH {
60                    return xml.get(start..).unwrap_or("");
61                }
62                at += o + open.len();
63            }
64            (_, Some(c)) if depth > 0 => {
65                depth -= 1;
66                at += c + close.len();
67            }
68            (_, Some(c)) => return xml.get(start..at + c + close.len()).unwrap_or(""),
69            // Unclosed (malformed): to the end.
70            (_, None) => return xml.get(start..).unwrap_or(""),
71        }
72    }
73}

75/// What is inside element `el` (an element string as [`element`] gives).
76fn inner(el: &str) -> &str {
77    let Some(tag_end) = el.find('>') else { return "" };
78    if el.get(..tag_end).is_some_and(|t| t.ends_with('/')) {
79        return "";
80    }
81    let end = el.rfind("</").unwrap_or(el.len());
82    el.get(tag_end + 1..end.max(tag_end + 1)).unwrap_or("")
83}

85/// The extension of picture `data`, by its first bytes.
86fn picture_ext(data: &[u8]) -> &'static str {
87    if data.starts_with(b"\x89PNG") {
88        "png"
89    } else if data.starts_with(b"\xff\xd8") {
90        "jpg"
91    } else if data.starts_with(b"GIF8") {
92        "gif"
93    } else if data.starts_with(b"<svg") || data.starts_with(b"<?xml") {
94        "svg"
95    } else {
96        "bin"
97    }
98}

100/// Where the first `draw:object` holding its object inline (no `xlink:href`, not self-closed)
101/// starts in `xml`.
102fn inline_object(xml: &str) -> Option<usize> {
103    let mut from = 0;
104    while let Some(i) = xml.get(from..).and_then(|r| r.find("<draw:object")).map(|i| from + i) {
105        let tag = xml.get(i..).and_then(|r| r.find('>').and_then(|e| r.get(..e))).unwrap_or("");
106        let next = tag.get("<draw:object".len()..).and_then(|r| r.chars().next());
107        if next.is_none_or(|c| c.is_whitespace()) && !tag.contains("xlink:href") && !tag.ends_with('/') {
108            return Some(i);
109        }
110        from = i + "<draw:object".len();
111    }
112    None
113}

115/// `xml` with its inline pictures and formulas moved into `files` (named as a package names
116/// them) and pointed at there, as a package's content holds them.
117fn take_out(xml: &str, files: &mut Vec<(String, Vec<u8>)>) -> String {
118    let mut out = String::with_capacity(xml.len());
119    let mut rest = xml;
120    loop {
121        let pic = rest.find("<office:binary-data>");
122        let obj = inline_object(rest);
123        let (at, picture) = match (pic, obj) {
124            (Some(p), Some(o)) if o < p => (o, false),
125            (Some(p), _) => (p, true),
126            (None, Some(o)) => (o, false),
127            (None, None) => break,
128        };
129        let (before, from) = rest.split_at(at);
130        let close = if picture { "</office:binary-data>" } else { "</draw:object>" };
131        let Some(end) = from.find(close) else {
132            out.push_str(before);
133            rest = from;
134            break;
135        };
136        let open_len = if picture { "<office:binary-data>".len() } else { from.find('>').map_or(0, |e| e + 1) };
137        let body = from.get(open_len..end).unwrap_or("");
138        let after = from.get(end + close.len()..).unwrap_or("");
139        let n = files.len() + 1;
140        if files.len() >= MAX_INLINE {
141            out.push_str(before);
142            rest = after;
143            continue;
144        }
145        if picture {
146            // The picture's `draw:image` start tag (just before) gets the package's reference.
147            let b64: String = body.chars().filter(|c| !c.is_whitespace()).collect();
148            let data = STANDARD.decode(b64.as_bytes()).ok().filter(|d| d.len() <= MAX_ENTRY as usize).unwrap_or_default();
149            // The element holding the data: the start tag just before it.
150            let holder = before.rfind('<').filter(|_| !data.is_empty()).and_then(|i| {
151                let tag = before.get(i..).unwrap_or("");
152                ["<draw:image", "<draw:object-ole"]
153                    .into_iter()
154                    .find(|el| tag.starts_with(el) && tag.get(el.len()..).is_some_and(|r| r.starts_with([' ', '>', '\n', '\t', '\r'])))
155                    .map(|el| (i, el))
156            });
157            match holder {
158                Some((at, el)) => {
159                    let name = if el == "<draw:image" { format!("Pictures/flat{n}.{}", picture_ext(&data)) } else { format!("Object flat{n}") };
160                    let (head, tag) = before.split_at(at);
161                    out.push_str(head);
162                    out.push_str(&format!("{el} xlink:href=\"{}\"", if el == "<draw:image" { name.clone() } else { format!("./{name}") }));
163                    out.push_str(tag.get(el.len()..).unwrap_or(""));
164                    files.push((name, data));
165                }
166                None => out.push_str(before),
167            }
168        } else {
169            let dir = format!("Object flat{n}");
170            out.push_str(before);
171            out.push_str(&format!("<draw:object xlink:href=\"./{dir}\"/>"));
172            // A sub-document (a chart, a spreadsheet) goes back into its files; a formula is
173            // its `content.xml`.
174            match package_parts(body.trim()) {
175                Some(parts) => {
176                    for (name, xml) in ["content.xml", "styles.xml", "meta.xml", "settings.xml"].into_iter().zip(parts) {
177                        files.push((format!("{dir}/{name}"), xml.into_bytes()));
178                    }
179                }
180                None => files.push((format!("{dir}/content.xml"), body.as_bytes().to_vec())),
181            }
182        }
183        rest = after;
184    }
185    out.push_str(rest);
186    out
187}

192}

194/// [`import_flat`] onto `page` (see [`super::import_on`]).
195pub fn import_flat_on(bytes: &[u8], page: &crate::doc::SectionProps) -> Result<Document, String> {
196    let xml = std::str::from_utf8(bytes).map_err(|_| "FODT: not UTF-8 text".to_string())?;
197    let root = element(xml, "office:document");
198    let [content, styles, meta, settings] = package_parts(root).ok_or("FODT: no office:document")?;
199    let mut files: Vec<(String, Vec<u8>)> = Vec::new();
200    let content = take_out(&content, &mut files);
201    let styles = take_out(&styles, &mut files);
202    let entries: Vec<(String, Vec<u8>)> = [
203        ("mimetype".to_string(), b"application/vnd.oasis.opendocument.text".to_vec()),
204        ("content.xml".to_string(), content.into_bytes()),
205        ("styles.xml".to_string(), styles.into_bytes()),
206        ("meta.xml".to_string(), meta.into_bytes()),
207        ("settings.xml".to_string(), settings.into_bytes()),
208    ]
209    .into_iter()
210    .chain(files)
211    .collect();
212    let mut z = zip::ZipWriter::new(Cursor::new(Vec::new()));
213    let stored = zip::write::SimpleFileOptions::default().compression_method(zip::CompressionMethod::Stored);
214    for (name, data) in &entries {
215        z.start_file(name.as_str(), stored).map_err(|e| format!("FODT: {e}"))?;
216        z.write_all(data).map_err(|e| format!("FODT: {e}"))?;
217    }
218    let package = z.finish().map_err(|e| format!("FODT: {e}"))?.into_inner();
219    super::import_on(&package, page)
220}

222/// Save `doc` as flat OpenDocument text: its package's files in one `office:document`, the
223/// automatic styles of its content and its styles together, pictures and formulas inline.
224pub fn export_flat(doc: &Document) -> Result<Vec<u8>, String> {
225    let package = super::export(doc)?;
226    let mut z = zip::ZipArchive::new(Cursor::new(package)).map_err(|e| format!("FODT: {e}"))?;
227    let mut read = |name: &str| -> String {
228        let mut s = String::new();
229        if let Ok(mut f) = z.by_name(name) {
230            let _ = f.read_to_string(&mut s);
231        }
232        s
233    };
234    let (content, styles, meta, settings) = (read("content.xml"), read("styles.xml"), read("meta.xml"), read("settings.xml"));
235    let mut bytes_of = |name: &str| -> Vec<u8> {
236        let mut v = Vec::new();
237        if let Ok(mut f) = z.by_name(name) {
238            let _ = f.read_to_end(&mut v);
239        }
240        v
241    };
242    let manifest = super::objects::manifest(&bytes_of("META-INF/manifest.xml"));
243    let body = flat_parts(&content, &styles, &meta, &settings);
244    // Pictures and formulas in the document, not beside it.
245    let mut flat = String::with_capacity(body.len());
246    let mut rest = body.as_str();
247    while let Some(at) = rest.find("xlink:href=\"") {
248        let (before, from) = rest.split_at(at);
249        let target_start = "xlink:href=\"".len();
250        let Some(len) = from.get(target_start..).and_then(|r| r.find('"')) else { break };
251        let target = from.get(target_start..target_start + len).unwrap_or("");
252        let after = from.get(target_start + len + 1..).unwrap_or("");
253        let tag = before.rfind('<').and_then(|i| before.get(i..)).unwrap_or("");
254        // A picture, or an embedded object's replacement picture.
255        let picture = target.starts_with("Pictures/") || target.trim_start_matches("./").starts_with("ObjectReplacements/");
256        // The rest of the linking element's start tag, whether it closes itself, and what follows
257        // the tag (markup kept as read may have `<draw:image …></draw:image>`).
258        let gt = after.find('>').unwrap_or(after.len());
259        let closed = after.get(..gt).is_some_and(|t| t.ends_with('/'));
260        let tag_rest = after.get(..gt).unwrap_or("").trim_end_matches('/');
261        let past_tag = after.get(gt + 1..).unwrap_or("");
262        // What follows the element: past its tag, or past its end tag.
263        let past = |el: &str| {
264            if closed { past_tag } else { past_tag.find(&format!("</{el}>")).and_then(|e| past_tag.get(e + el.len() + 3..)).unwrap_or("") }
265        };
266        // The link's other attributes go with it.
267        let attrs: String = tag_rest.split_whitespace().filter(|a| !a.starts_with("xlink:")).map(|a| format!(" {a}")).collect();
268        if tag.starts_with("<draw:image") && picture {
269            // `<draw:image xlink:href="…" …/>` → `<draw:image …><office:binary-data>…`, before
270            // anything else it holds.
271            let data = bytes_of(target.trim_start_matches("./"));
272            flat.push_str(before.trim_end());
273            flat.push_str(&attrs);
274            flat.push_str(&format!("><office:binary-data>{}</office:binary-data>", STANDARD.encode(&data)));
275            if closed {
276                flat.push_str("</draw:image>");
277            }
278            rest = past_tag;
279        } else if tag.starts_with("<draw:object-ole") && target.starts_with("./Object ") {
280            // An OLE object's file inline (§10.4.7 allows `office:binary-data` in it).
281            let data = bytes_of(target.trim_start_matches("./"));
282            flat.push_str(before.rfind("<draw:object-ole").and_then(|i| before.get(..i)).unwrap_or(before));
283            flat.push_str(&format!("<draw:object-ole><office:binary-data>{}</office:binary-data></draw:object-ole>", STANDARD.encode(&data)));
284            rest = past("draw:object-ole");
285        } else if tag.starts_with("<draw:object") && target.starts_with("./Object ") {
286            let dir = target.trim_start_matches("./");
287            let mut text = |name: &str| String::from_utf8_lossy(&bytes_of(&format!("{dir}/{name}"))).into_owned();
288            let content = text("content.xml");
289            // A sub-document (a chart, a spreadsheet) goes in as an `office:document` (§3.1.2)
290            // of its media type, its files together; a formula as its `math:math`.
291            let math = if content.contains("<office:document-content") {
292                let mime = manifest.get(&format!("{dir}/")).map_or("application/vnd.oasis.opendocument.graphics", String::as_str);
293                let (styles, meta, settings) = (text("styles.xml"), text("meta.xml"), text("settings.xml"));
294                flat_document(mime, &content, &styles, &meta, &settings)
295            } else {
296                content.split_once("?>").map_or(content.as_str(), |(_, m)| m).trim().to_string()
297            };
298            // A plain `draw:object` around it.
299            flat.push_str(before.rfind("<draw:object").and_then(|i| before.get(..i)).unwrap_or(before));
300            flat.push_str(&format!("<draw:object>{math}</draw:object>"));
301            rest = past("draw:object");
302        } else {
303            flat.push_str(before);
304            flat.push_str(from.get(..target_start + len + 1).unwrap_or(""));
305            rest = after;
306        }
307    }
308    flat.push_str(rest);
309    Ok(format!(
310        "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:document {} office:version=\"1.3\" office:mimetype=\"application/vnd.oasis.opendocument.text\">{flat}</office:document>\n",
311        merged_namespaces(&[&content, &styles, &meta, &settings])
312    )
313    .into_bytes())
314}

316/// Every namespace the parts declare (the settings' `config`, say), each prefix once.
317fn merged_namespaces(parts: &[&str]) -> String {
318    let mut ns: Vec<&str> = Vec::new();
319    for part in parts {
320        for decl in namespaces(part) {
321            let prefix = decl.split('=').next().unwrap_or("");
322            if !ns.iter().any(|d| d.split('=').next() == Some(prefix)) {
323                ns.push(decl);
324            }
325        }
326    }
327    ns.join(" ")
328}

330/// What a flat document holds, from a package's parts in the order §3.1.2 gives them: one
331/// font declaration list and one automatic styles list for its content and its styles together.
332fn flat_parts(content: &str, styles: &str, meta: &str, settings: &str) -> String {
333    let both = |name: &str| {
334        let (a, b) = (element(styles, name), element(content, name));
335        if a.is_empty() && b.is_empty() { String::new() } else { format!("<{name}>{}{}</{name}>", inner(a), inner(b)) }
336    };
337    [
338        element(meta, META).to_string(),
339        element(settings, SETTINGS).to_string(),
340        both(FONTS),
341        element(styles, STYLES).to_string(),
342        both(AUTOMATIC),
343        element(styles, MASTERS).to_string(),
344        element(content, BODY).to_string(),
345    ]
346    .concat()
347}

349/// A sub-document's parts as one `office:document` of media type `mime`.
350fn flat_document(mime: &str, content: &str, styles: &str, meta: &str, settings: &str) -> String {
351    format!(
352        "<office:document {} office:version=\"1.3\" office:mimetype=\"{}\">{}</office:document>",
353        merged_namespaces(&[content, styles, meta, settings]),
354        super::writer::x(mime),
355        flat_parts(content, styles, meta, settings)
356    )
357}

359/// An `office:document` element taken apart into a package's `content.xml`, `styles.xml`,
360/// `meta.xml` and `settings.xml` (each part's root with the document's attributes but its
361/// media type). `None` when `root` is no `office:document`.
362fn package_parts(root: &str) -> Option<[String; 4]> {
363    if !root.starts_with("<office:document") || root.starts_with("<office:document-") {
364        return None;
365    }
366    let attrs = root.find('>').and_then(|e| root.get("<office:document".len()..e)).map(|a| a.trim_end_matches('/'))?;
367    // Without its media type, which only the flat root has.
368    let attrs = match attrs.find(" office:mimetype=\"") {
369        Some(i) => {
370            let from = i + " office:mimetype=\"".len();
371            let end = attrs.get(from..).and_then(|r| r.find('"')).map_or(attrs.len(), |e| from + e + 1);
372            format!("{}{}", attrs.get(..i).unwrap_or(""), attrs.get(end..).unwrap_or(""))
373        }
374        None => attrs.to_string(),
375    };
376    let doc = |name: &str, parts: &[&str]| -> String {
377        let body: String = parts.iter().map(|p| element(root, p)).collect();
378        format!("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<office:{name}{attrs}>{body}</office:{name}>")
379    };
380    Some([
381        doc("document-content", &[FONTS, AUTOMATIC, BODY]),
382        doc("document-styles", &[FONTS, STYLES, AUTOMATIC, MASTERS]),
383        doc("document-meta", &[META]),
384        doc("document-settings", &[SETTINGS]),
385    ])
386}

388/// The namespace declarations (`xmlns:p="…"`) on the root element of `xml`.
389fn namespaces(xml: &str) -> Vec<&str> {
390    let start = xml.find("<office:").unwrap_or(0);
391    let tag = xml.get(start..).and_then(|r| r.find('>').and_then(|e| r.get(..e))).unwrap_or("");
392    let mut out = Vec::new();
393    let mut rest = tag;
394    while let Some(i) = rest.find("xmlns:") {
395        let from = rest.get(i..).unwrap_or("");
396        // `xmlns:p="…"`: up to the closing quote.
397        let Some(q1) = from.find('"') else { break };
398        let Some(q2) = from.get(q1 + 1..).and_then(|r| r.find('"')) else { break };
399        out.push(from.get(..q1 + 1 + q2 + 1).unwrap_or(""));
400        rest = from.get(q1 + 1 + q2 + 1..).unwrap_or("");
401    }
402    out
403}

405#[cfg(test)]
406mod element_tests {
407    use super::element;
408
409    /// An element ends at its own close tag however many elements of its name it holds side by
410    /// side (inline objects' own `office:body`s), not after a count of tags.
411    #[test]
412    fn an_element_holding_many_of_its_name_ends_at_its_own_close() {
413        let inner = "<office:body><text:p>x</text:p></office:body>".repeat(1200);
414        let outer = format!("<office:body>{inner}</office:body>");
415        let xml = format!("<office:document>{outer}<office:settings/></office:document>");
416        assert_eq!(element(&xml, "office:body"), outer);
417        // Nested deeper than any XML here: malformed, to the end, as an unclosed one.
418        let deep = format!("{}{}", "<office:body>".repeat(600), "</office:body>".repeat(600));
419        assert_eq!(element(&deep, "office:body"), deep);
420        assert_eq!(element("<office:body><x/>", "office:body"), "<office:body><x/>");
421    }
