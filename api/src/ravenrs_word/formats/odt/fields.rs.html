1//! Text fields (ODF 1.3 §7.3–7.7) and the Word field codes WordCraft keeps them as.
2//!
3//! Read: a field element becomes a field with that code and the element's text as its result:
4//! a date or time with its date style as a `\@` picture, a sequence numbering figures or
5//! tables as `SEQ`, a chapter field as `STYLEREF` to its heading style, references to
6//! bookmarks, reference marks and sequences as `REF`/`PAGEREF`. A fixed date or time
7//! (`text:fixed="true"`) is plain text, as is a field with no Word equivalent (variables, a
8//! reference to a caption's text…), so nothing it shows is lost. Written: a field whose code
9//! has an ODF element is that element around its result; any other field (or one with a code
10//! ODF can't hold) is its result as text.
11

60/// `el`, a field element, fixed at what it shows (§19.807 `text:fixed`), for the fields that can
61/// be.
62/// Field element `el` for locked field `instr` showing `result`, fixed (see [`fixed`]). A date
63/// or time is fixed at the value its text shows under its picture (§19.871, §19.880); `None`
64/// when that can't be read, as a reader shows a fixed date without a value as today's (then it
65/// is written as its text).
66pub(super) fn locked(el: &str, instr: &str, result: &str) -> Option<String> {
67    let code = instr.split_whitespace().next().unwrap_or("").to_ascii_uppercase();
68    let attr = match code.as_str() {
69        "DATE" => "text:date-value",
70        "TIME" => "text:time-value",
71        _ => return Some(fixed(el)),
72    };
73    let value = super::datastyle::picture_of(instr).and_then(|pic| super::datastyle::value_shown(&pic, result))?;
74    let el = fixed(el);
75    let (tag, rest) = el.find(['>', ' ']).and_then(|at| Some((el.get(..at)?, el.get(at..)?)))?;
76    Some(format!("{tag} {attr}=\"{value}\"{rest}"))
77}

79pub(super) fn fixed(el: &str) -> String {
80    with_fixed(el, true)

83/// `el`, a field element, an author field said not to be fixed (§19.807's default, said:
84/// LibreOffice reads an author field without `text:fixed` as fixed, showing what it holds, and
85/// writes `false` itself). Other fields are written as LibreOffice writes them, without it.
86pub(super) fn unfixed(el: &str) -> String {
87    if el.starts_with("<text:author-") { with_fixed(el, false) } else { el.to_string() }
88}

90fn with_fixed(el: &str, fixed: bool) -> String {
91    const FIXABLE: [&str; 6] = ["<text:date", "<text:time", "<text:author-name", "<text:author-initials", "<text:sender-", "<text:file-name"];
92    if !FIXABLE.iter().any(|f| el.starts_with(f)) {
93        return el.to_string();
94    }
95    match el.find(['>', ' ']).and_then(|at| Some((el.get(..at)?, el.get(at..)?))) {
96        Some((tag, rest)) => format!("{tag} text:fixed=\"{fixed}\"{rest}"),
97        None => el.to_string(),
98    }
99}

101/// The Word field code for the `text:` field element `local` with attributes `a`, or `None`
102/// when it is no field WordCraft keeps (then its text is read as text).
103pub(super) fn instr_of(local: &str, a: &Attrs, picture: impl Fn(&str) -> Option<String>) -> Option<String> {
104    // Fixed, unless it holds no value to be fixed at (LibreOffice's templates: a date that shows
105    // the day the document is made from them).
106    // A name a field code can hold as one word.
107    let name = |k: &str| {
108        get(a, k)
109            .map(str::trim)
110            .filter(|n| !n.is_empty() && n.len() <= MAX_REF_NAME && !n.contains(|c: char| c.is_whitespace() || c == '"' || c == '\\'))
111    };
112    match local {
113        // A variable's value (§7.4.5, §7.4.6): what a SET gave its name.
114        "variable-get" | "variable-set" => Some(format!("REF {}", name("text:name")?)),
115        // A document property's value (§7.5.6 `text:user-defined`, a `meta:user-defined`) or a
116        // user field's (§7.4.9 `text:user-field-get`, declared once for the document): Word's
117        // `DOCPROPERTY`, the property kept as a custom document property.
118        "user-defined" | "user-field-get" => {
119            let n = clean_prop_name(get(a, "text:name")?)?;
120            Some(format!("DOCPROPERTY \"{n}\""))
121        }
122        // A citation (§8.1.24 `text:bibliography-mark`): Word's `CITATION` of the source it
123        // carries (see [`bib_source`]).
124        "bibliography-mark" => bib_tag(get(a, "text:identifier")?).map(|t| format!("CITATION {t}")),
125        // An index entry at a point (§8.1.10): Word's `XE`, its keys the levels above it.
126        "alphabetical-index-mark" => xe_instr(get(a, "text:key1"), get(a, "text:key2"), get(a, "text:string-value")?),
127        // Numbering figures, tables and the like (§7.4.13): Word's `SEQ` and its number format.
128        "sequence" => {
129            let seq = name("text:name")?;
130            let fmt = match get(a, "style:num-format").map(str::trim) {
131                Some("i") => "roman",
132                Some("I") => "ROMAN",
133                Some("a") => "alphabetic",
134                Some("A") => "ALPHABETIC",
135                _ => "ARABIC",
136            };
137            Some(format!("SEQ {seq} \\* {fmt}"))
138        }
139        // The chapter a page is in (§7.3.8): the nearest heading of its level, by name or number.
140        "chapter" => {
141            let level = get(a, "text:outline-level").and_then(|l| l.trim().parse::<u32>().ok()).unwrap_or(1).clamp(1, 9);
142            let number = matches!(get(a, "text:display").map(str::trim), Some("number" | "plain-number"));
143            Some(format!("STYLEREF \"Heading {level}\"{}", if number { " \\n" } else { "" }))
144        }
145        // References to reference marks (§7.7.5) and to sequences (§7.7.7): to the text or the
146        // page; a sequence's label and number is the bookmark over them.
147        "reference-ref" | "sequence-ref" => {
148            let target = name("text:ref-name")?;
149            match (local, get(a, "text:reference-format").map(str::trim)) {
150                (_, Some("page")) => Some(format!("PAGEREF {target} \\h")),
151                ("reference-ref", Some("text") | None) | ("sequence-ref", Some("category-and-value")) => Some(format!("REF {target} \\h")),
152                _ => None,
153            }
154        }
155        // Only the current page's number: "previous" and "next" have no Word equivalent.
156        "page-number" => (get(a, "text:select-page").is_none_or(|v| v.trim() == "current")).then(|| {
157            // Its own number format, over the page style's (Word's `\*` format switch).
158            match get(a, "style:num-format").map(str::trim) {
159                Some("1") => "PAGE \\* Arabic".to_string(),
160                Some("i") => "PAGE \\* roman".to_string(),
161                Some("I") => "PAGE \\* ROMAN".to_string(),
162                Some("a") => "PAGE \\* alphabetic".to_string(),
163                Some("A") => "PAGE \\* ALPHABETIC".to_string(),
164                _ => "PAGE".to_string(),
165            }
166        }),
167        "bookmark-ref" => {
168            let name =
169                get(a, "text:ref-name").map(str::trim).filter(|n| !n.is_empty() && n.len() <= MAX_REF_NAME && !n.contains(char::is_whitespace))?;
170            Some(match get(a, "text:reference-format").map(str::trim) {
171                Some("page") => format!("PAGEREF {name} \\h"),
172                Some("number" | "number-all-superior" | "number-no-superior") => format!("REF {name} \\r \\h"),
173                _ => format!("REF {name} \\h"),
174            })
175        }
176        // The sender's details (§7.3.6–7.3.7), from the application's user data: WordCraft's
177        // own `SENDER` field code, shown as its last result.
178        _ if local.strip_prefix("sender-").is_some_and(|p| SENDER_PARTS.contains(&p)) => local.strip_prefix("sender-").map(|p| format!("SENDER {p}")),
179        _ => {
180            let code = SIMPLE.iter().find(|(e, _)| *e == local).map(|(_, code)| code.to_string())?;
181            // A date or time in its date style (§19.473 `style:data-style-name`).
182            let pic = get(a, "style:data-style-name").and_then(&picture).filter(|_| code.contains("DATE") || code == "TIME");
183            Some(match pic {
184                Some(p) => format!("{code} \\@ \"{p}\""),
185                None => code,
186            })
187        }
188    }
189}

191/// The ODF field element for Word field code `instr` around `result` (already escaped), or
192/// `None` when ODF has none for it.
193pub(super) fn element(instr: &str, result: &str, data_style: Option<&str>) -> Option<String> {
194    let mut words = instr.split_whitespace();
195    let code = words.next()?.to_ascii_uppercase();
196    match code.as_str() {
197        "SENDER" => {
198            let part = words.next().filter(|p| SENDER_PARTS.contains(p))?;
199            Some(format!("<text:sender-{part}>{result}</text:sender-{part}>"))
200        }
201        "SEQ" => {
202            let seq = words.next().filter(|n| !n.starts_with('\\') && n.len() <= MAX_REF_NAME)?;
203            let fmt = match switch_arg(instr, '*').as_deref() {
204                Some("roman") => "i",
205                Some("ROMAN") => "I",
206                Some("alphabetic") => "a",
207                Some("ALPHABETIC") => "A",
208                _ => "1",
209            };
210            let seq = x(seq);
211            Some(format!("<text:sequence text:name=\"{seq}\" text:formula=\"ooow:{seq}+1\" style:num-format=\"{fmt}\">{result}</text:sequence>"))
212        }
213        // `STYLEREF "Heading 2"` or `STYLEREF 2`: the chapter of that level.
214        "STYLEREF" => {
215            let target = instr.trim_start().get(code.len()..)?.trim_start();
216            let target = target.strip_prefix('"').and_then(|t| t.split('"').next()).unwrap_or_else(|| target.split_whitespace().next().unwrap_or(""));
217            let level = target.trim().strip_prefix("Heading").unwrap_or(target).trim().parse::<u32>().ok().filter(|l| (1..=10).contains(l))?;
218            let number = [" \\n", " \\r", " \\w"].iter().any(|s| instr.contains(s));
219            let display = if number { "number" } else { "name" };
220            Some(format!("<text:chapter text:display=\"{display}\" text:outline-level=\"{level}\">{result}</text:chapter>"))
221        }
222        "PAGE" => {
223            let fmt = match switch_arg(instr, '*').as_deref() {
224                Some(f) if f.eq_ignore_ascii_case("arabic") => " style:num-format=\"1\"",
225                Some("roman") => " style:num-format=\"i\"",
226                Some("ROMAN") => " style:num-format=\"I\"",
227                Some("alphabetic") => " style:num-format=\"a\"",
228                Some("ALPHABETIC") => " style:num-format=\"A\"",
229                _ => "",
230            };
231            Some(format!("<text:page-number{fmt} text:select-page=\"current\">{result}</text:page-number>"))
232        }
233        // An index entry: a mark at its place, its first levels the keys (§8.1.10).
234        "XE" => {
235            let entry = instr.trim_start().get(code.len()..)?.trim_start();
236            let entry = entry.strip_prefix('"').and_then(|t| t.split('"').next())?;
237            let mut levels: Vec<&str> = entry.split(':').map(str::trim).filter(|l| !l.is_empty()).collect();
238            let value = levels.pop()?;
239            let keys: String = levels.iter().take(2).enumerate().map(|(i, k)| format!(" text:key{}=\"{}\"", i + 1, x(k))).collect();
240            Some(format!("<text:alphabetical-index-mark text:string-value=\"{}\"{keys}/>", x(value)))
241        }
242        "REF" | "PAGEREF" => {
243            let name = words.next().filter(|n| !n.starts_with('\\') && n.len() <= MAX_REF_NAME)?;
244            let rest: Vec<&str> = words.collect();
245            let format = if code == "PAGEREF" {
246                "page"
247            } else if rest.iter().any(|s| s.eq_ignore_ascii_case("\\r") || s.eq_ignore_ascii_case("\\n") || s.eq_ignore_ascii_case("\\w")) {
248                "number"
249            } else {
250                "text"
251            };
252            Some(format!("<text:bookmark-ref text:reference-format=\"{format}\" text:ref-name=\"{}\">{result}</text:bookmark-ref>", x(name)))
253        }
254        _ => {
255            let (el, _) = SIMPLE.iter().find(|(_, c)| *c == code)?;
256            let ds = data_style.map(|d| format!(" style:data-style-name=\"{}\"", x(d))).unwrap_or_default();
257            Some(format!("<text:{el}{ds}>{result}</text:{el}>"))
258        }
259    }
260}

262/// A custom property's name as a field code can quote it: no quotes or backslashes, no control
263/// characters, at most 255 characters. `None` when nothing is left.
264pub(super) fn clean_prop_name(n: &str) -> Option<String> {
265    let n: String = n.chars().filter(|c| !matches!(c, '"' | '\\') && !c.is_control()).take(255).collect();
266    let n = n.trim();
267    (!n.is_empty()).then(|| n.to_string())
268}

270/// The property a `DOCPROPERTY` field code names, if it is one.
271pub(super) fn docproperty_name(instr: &str) -> Option<&str> {
272    let rest = instr.trim_start().strip_prefix("DOCPROPERTY")?.trim_start();
273    match rest.strip_prefix('"') {
274        Some(q) => q.split('"').next(),
275        None => rest.split_whitespace().next(),
276    }
277    .map(str::trim)
278    .filter(|n| !n.is_empty() && !n.starts_with('\\'))
279}

286/// A custom property of ODF value type `t` (`office:value-type`, `meta:value-type`) and value
287/// `v`: as a document's custom property (OOXML variant type, value).
288pub(super) fn custom_prop(name: &str, t: Option<&str>, v: &str) -> Option<crate::doc::CustomProp> {
289    let name = clean_prop_name(name)?;
290    let kind = match t.map(str::trim) {
291        Some("float" | "percentage" | "currency") => "r8",
292        Some("boolean") => "bool",
293        Some("date") => "filetime",
294        _ => "lpwstr",
295    };
296    let mut value: String = v.chars().filter(|c| !c.is_control() || *c == '\n').take(MAX_PROP_VALUE).collect();
297    let mut kind = kind;
298    if kind == "filetime" {
299        // A date and time in UTC, as Word's `filetime` holds it: a bare date at midnight; one
300        // that isn't a date is kept as text.
301        let v = value.trim();
302        let date_ok = v.len() >= 10
303            && v.is_char_boundary(10)
304            && v.get(..10).is_some_and(|d| d.bytes().enumerate().all(|(i, b)| if i == 4 || i == 7 { b == b'-' } else { b.is_ascii_digit() }));
305        value = match (date_ok, v.get(10..)) {
306            (true, Some("")) => format!("{v}T00:00:00Z"),
307            (true, Some(rest)) if rest.ends_with('Z') || rest.contains('+') || rest.rfind('-').is_some() => v.to_string(),
308            (true, Some(_)) => format!("{v}Z"),
309            _ => {
310                kind = "lpwstr";
311                v.to_string()
312            }
313        };
314    }
315    Some(crate::doc::CustomProp { name, kind: kind.into(), value, link_target: None, odf_user_field: false })
316}

318/// The custom properties of a package's `meta.xml` (§4.3.2.14 `meta:user-defined`).
319pub(super) fn meta_props(xml: &[u8]) -> Vec<crate::doc::CustomProp> {
320    let mut out: Vec<crate::doc::CustomProp> = Vec::new();
321    let mut open: Option<(String, Option<String>, String)> = None;
322    super::events(xml, |ev| match ev {
323        super::Ev::Start(q, a) if q == "meta:user-defined" => {
324            open = get(&a, "meta:name").map(|n| (n.to_string(), get(&a, "meta:value-type").map(String::from), String::new()));
325        }
326        super::Ev::Text(t) => {
327            if let Some((_, _, v)) = &mut open
328                && v.len() < MAX_PROP_VALUE * 4
329            {
330                v.push_str(&t);
331            }
332        }
333        super::Ev::End(q) if q == "meta:user-defined" => {
334            if let Some((n, t, v)) = open.take()
335                && out.len() < MAX_PROPS
336                && let Some(p) = custom_prop(&n, t.as_deref(), &v)
337                && !out.iter().any(|o| o.name == p.name)
338            {
339                out.push(p);
340            }
341        }
342        _ => {}
343    });
344    out
345}

347/// The user fields a package's `content.xml` declares (§7.4.7 `text:user-field-decl`), as custom
348/// properties.
349pub(super) fn user_fields(xml: &[u8]) -> Vec<crate::doc::CustomProp> {
350    let mut out: Vec<crate::doc::CustomProp> = Vec::new();
351    super::events(xml, |ev| {
352        if let super::Ev::Start(q, a) = ev
353            && q == "text:user-field-decl"
354            && out.len() < MAX_PROPS
355        {
356            let t = get(&a, "office:value-type");
357            let v = match t.map(str::trim) {
358                Some("float" | "percentage" | "currency") => get(&a, "office:value"),
359                Some("boolean") => get(&a, "office:boolean-value"),
360                Some("date") => get(&a, "office:date-value"),
361                _ => get(&a, "office:string-value"),
362            }
363            .unwrap_or("");
364            if let Some(p) = get(&a, "text:name").and_then(|n| custom_prop(n, t, v))
365                && !out.iter().any(|o| o.name == p.name)
366            {
367                out.push(crate::doc::CustomProp { odf_user_field: true, ..p });
368            }
369        }
370    });
371    out
372}

396/// The source a `text:bibliography-mark` carries: its identifier as the tag, its type as the
397/// kind, and its author, title, year, publisher, address, journal, volume, pages, URL, editor,
398/// edition, number (issue) and month. `None` without an identifier.
399pub(super) fn bib_source(a: &Attrs) -> Option<crate::doc::Source> {
400    let tag = bib_tag(get(a, "text:identifier")?)?;
401    let v = |k: &str| get(a, k).map(|s| s.chars().filter(|c| !c.is_control()).take(1024).collect::<String>()).unwrap_or_default();
402    let t = get(a, "text:bibliography-type").map(str::trim).unwrap_or("");
403    let kind = BIB_KINDS.iter().find(|(o, _)| *o == t).map_or("other", |(_, k)| k).to_string();
404    Some(crate::doc::Source {
405        tag,
406        kind,
407        author: v("text:author"),
408        title: v("text:title"),
409        year: v("text:year"),
410        publisher: v("text:publisher"),
411        city: v("text:address"),
412        journal: v("text:journal"),
413        volume: v("text:volume"),
414        pages: v("text:pages"),
415        url: v("text:url"),
416        editor: v("text:editor"),
417        edition: v("text:edition"),
418        issue: v("text:number"),
419        month: v("text:month"),
420        ..Default::default()
421    })
422}

424/// A source as a `text:bibliography-mark` around `result` (already escaped).
425pub(super) fn bib_mark(s: &crate::doc::Source, result: &str) -> String {
426    let t = BIB_KINDS.iter().find(|(_, k)| *k == s.kind).map_or("misc", |(o, _)| o);
427    let author = if s.corporate.is_empty() { &s.author } else { &s.corporate };
428    let mut a = format!(" text:identifier=\"{}\" text:bibliography-type=\"{t}\"", x(&s.tag));
429    for (k, v) in [
430        ("author", author),
431        ("title", &s.title),
432        ("year", &s.year),
433        ("publisher", &s.publisher),
434        ("address", &s.city),
435        ("journal", &s.journal),
436        ("volume", &s.volume),
437        ("pages", &s.pages),
438        ("url", &s.url),
439        ("editor", &s.editor),
440        ("edition", &s.edition),
441        ("number", &s.issue),
442        ("month", &s.month),
443    ] {
444        if !v.is_empty() {
445            a.push_str(&format!(" text:{k}=\"{}\"", x(v)));
446        }
447    }
448    format!("<text:bibliography-mark{a}>{result}</text:bibliography-mark>")
449}

451/// Most characters of an index entry's level.
452const MAX_ENTRY_LEVEL: usize = 255;

454/// The `XE` field code of an index entry `value` under keys `key1` and `key2` (its levels, top
455/// first). Quotes, which end the entry, and colons, which split levels, are left out of each.
456/// `None` when the entry is empty.
457pub(super) fn xe_instr(key1: Option<&str>, key2: Option<&str>, value: &str) -> Option<String> {
458    let clean = |s: &str| {
459        s.chars().filter(|c| !matches!(c, '"' | ':' | '\\') && !c.is_control()).take(MAX_ENTRY_LEVEL).collect::<String>().trim().to_string()
460    };
461    let value = clean(value);
462    if value.is_empty() {
463        return None;
464    }
465    let levels: Vec<String> = [key1, key2].into_iter().flatten().map(clean).filter(|k| !k.is_empty()).chain([value]).collect();
466    Some(format!("XE \"{}\"", levels.join(":")))
467}

469/// The argument of switch `\sw` in field code `instr` (quoted or one word).
470fn switch_arg(instr: &str, sw: char) -> Option<String> {
471    let at = instr.find(&format!("\\{sw}"))?;
472    let rest = instr.get(at + 2..)?.trim_start();
473    let arg = match rest.strip_prefix('"') {
474        Some(q) => q.split('"').next()?,
475        None => rest.split_whitespace().next()?,
476    };
477    Some(arg.chars().take(64).collect())
478}

480/// The sequence (`SEQ`) field code `instr` counts, if it is one.
481pub(super) fn sequence_name(instr: &str) -> Option<&str> {
482    let mut words = instr.split_whitespace();
483    words.next().filter(|c| c.eq_ignore_ascii_case("SEQ"))?;
484    words.next().filter(|n| !n.starts_with('\\') && n.len() <= MAX_REF_NAME)
485}

487/// The SET field (ECMA-376 §17.16.5.58) a `text:variable-set` with attributes `a` is: its name and
488/// value (§19.383 `office:string-value`, else the number, else the formula without its prefix).
489pub(super) fn set_instr(a: &Attrs) -> Option<String> {
490    let name = get(a, "text:name")
491        .map(str::trim)
492        .filter(|n| !n.is_empty() && n.len() <= MAX_REF_NAME && !n.contains(|c: char| c.is_whitespace() || c == '"' || c == '\\'))?;
493    let value = get(a, "office:string-value")
494        .or_else(|| get(a, "office:value"))
495        .or_else(|| get(a, "text:formula").map(|f| f.trim().strip_prefix("ooow:").unwrap_or(f.trim())))
496        .unwrap_or_default();
497    let value: String = value.chars().take(MAX_FIELD_VALUE).collect();
498    Some(format!("SET {name} \"{}\"", value.replace('\\', "\\\\").replace('"', "\\\"")))
499}
500
501/// Most characters of a variable's value kept.
502const MAX_FIELD_VALUE: usize = 10_000;
503
504/// A SET field's name and value, when `instr` is one.
505pub(super) fn set_of(instr: &str) -> Option<(&str, String)> {
506    let rest = instr.trim_start().strip_prefix("SET")?.strip_prefix(char::is_whitespace)?.trim_start();
507    let end = rest.find(char::is_whitespace).unwrap_or(rest.len());
508    let (name, value) = (rest.get(..end)?, rest.get(end..)?.trim());
509    if name.is_empty() {
510        return None;
511    }
512    // A quoted value, its \" and \\ undone; else the words as they are.
513    let value = match value.strip_prefix('"') {
514        Some(q) => {
515            let mut out = String::new();
516            let mut it = q.chars();
517            while let Some(c) = it.next() {
518                match c {
519                    '"' => break,
520                    '\\' => out.extend(it.next()),
521                    c => out.push(c),
522                }
523            }
524            out
525        }
526        None => value.to_string(),
527    };
528    Some((name, value))
529}
