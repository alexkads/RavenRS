20/// A dash or space length: a fraction of the line width, or an absolute length in points.
21#[derive(Clone, Copy, Debug, PartialEq)]
22enum Len {
23    Width(f32),
24    Points(f32),
25}
26
27impl Len {
28    fn read(v: Option<&str>) -> Option<Len> {
29        let v = v?.trim();
30        match v.strip_suffix('%') {
31            Some(p) => p.trim().parse::<f32>().ok().filter(|p| p.is_finite()).map(|p| Len::Width((p / 100.0).clamp(0.0, 100.0))),
32            None => length(v).filter(|l| l.is_finite()).map(|l| Len::Points(l.clamp(0.0, 1584.0))),
33        }
34    }
35
36    /// In 100000ths of a line `width` points wide.
37    fn of(self, width: f32) -> u32 {
38        let ratio = match self {
39            Len::Width(r) => r,
40            Len::Points(p) => {
41                let w = if width.is_finite() && width > 0.0 { width } else { 0.75 };
42                p / w
43            }
44        };
45        (ratio.clamp(0.0, 1000.0) * DASH_PERCENT as f32).round() as u32
46    }
47}

49/// A `draw:stroke-dash`: `dots1` dashes of one length then `dots2` of another, each followed by
50/// the same space.
51#[derive(Clone, Debug, PartialEq)]
52pub(super) struct Dash {
53    dots1: u32,
54    len1: Len,
55    dots2: u32,
56    len2: Len,
57    space: Len,
58}
59
60impl Dash {
61    pub(super) fn read(a: &Attrs) -> Dash {
62        let count = |k: &str| get(a, k).and_then(|v| v.trim().parse::<u32>().ok()).unwrap_or(0).min(MAX_DASH_STOPS as u32);
63        // A dash without a length is a dot as long as the line is wide.
64        let len = |k: &str| Len::read(get(a, k)).unwrap_or(Len::Width(1.0));
65        Dash {
66            dots1: count("draw:dots1"),
67            len1: len("draw:dots1-length"),
68            dots2: count("draw:dots2"),
69            len2: len("draw:dots2-length"),
70            space: len("draw:distance"),
71        }
72    }
73
74    /// The pattern for a line `width` points wide: one of WordCraft's named dashes when it is one.
75    pub(super) fn pattern(&self, width: f32) -> LineDash {
76        let space = self.space.of(width);
77        let pairs: Vec<(u32, u32)> = std::iter::repeat_n((self.len1.of(width), space), self.dots1 as usize)
78            .chain(std::iter::repeat_n((self.len2.of(width), space), self.dots2 as usize))
79            .take(MAX_DASH_STOPS)
80            .collect();
81        if pairs.is_empty() {
82            return LineDash::Solid;
83        }
84        PRESETS.iter().find(|p| ratios(p) == pairs).cloned().unwrap_or(LineDash::Custom(pairs))
85    }
86}
87
88/// WordCraft's named dashes.
89const PRESETS: [LineDash; 10] = [
90    LineDash::Dot,
91    LineDash::Dash,
92    LineDash::LongDash,
93    LineDash::DashDot,
94    LineDash::LongDashDot,
95    LineDash::LongDashDotDot,
96    LineDash::SystemDash,
97    LineDash::SystemDot,
98    LineDash::SystemDashDot,
99    LineDash::SystemDashDotDot,
100];
101
102/// A dash's (dash, space) pairs in 100000ths of the line width.
103fn ratios(d: &LineDash) -> Vec<(u32, u32)> {
104    if let LineDash::Custom(pairs) = d {
105        return pairs.iter().take(MAX_DASH_STOPS).copied().collect();
106    }
107    let p: Vec<u32> = d.pattern(1.0).iter().map(|v| (v * DASH_PERCENT as f32).round() as u32).collect();
108    p.chunks(2).map(|c| (c.first().copied().unwrap_or(0), c.get(1).copied().unwrap_or(0))).collect()
109}

111/// The `draw:stroke-dash` named `name` for `dash`, or `None` for a solid line. ODF's dash has
112/// at most two kinds of dash and one space: a pattern with more is written as its first two.
113pub(super) fn dash_def(name: &str, dash: &LineDash) -> Option<String> {
114    let pairs = ratios(dash);
115    let (&(l1, space), rest) = pairs.split_first()?;
116    let dots1 = 1 + rest.iter().take_while(|p| **p == (l1, space)).count();
117    let second = rest.get(dots1 - 1).map(|p| p.0);
118    let dots2 = second.map_or(0, |l2| rest.iter().skip(dots1 - 1).take_while(|p| p.0 == l2).count());
119    let pc = |v: u32| format!("{}%", v as f32 * 100.0 / DASH_PERCENT as f32);
120    let mut s = format!("<draw:stroke-dash draw:name=\"{name}\" draw:style=\"rect\" draw:dots1=\"{dots1}\" draw:dots1-length=\"{}\"", pc(l1));
121    if let Some(l2) = second {
122        s.push_str(&format!(" draw:dots2=\"{dots2}\" draw:dots2-length=\"{}\"", pc(l2)));
123    }
124    s.push_str(&format!(" draw:distance=\"{}\"/>", pc(space)));
125    Some(s)
126}

128/// An angle (§18.3.1): degrees, radians or grads by its unit, tenths of a degree without one
129/// (as ODF 1.2 wrote a gradient's).
130/// Whether a part of ODF `version` (its `office:version`; none: before 1.2) has its gradients'
131/// and hatches' unitless angles in tenths of a degree.
132///
133/// **A deviation from the standard, out of necessity.** ODF's angle (§18.3.1, since ODF 1.2)
134/// is in degrees when it has no unit, so `draw:angle="300"` (§19.112) and `draw:rotation`
135/// (§19.209) mean 300°. But LibreOffice and OpenOffice wrote these angles in tenths of a degree
136/// in their files before ODF 1.3 (`300` for 30°), and LibreOffice still reads such files so
137/// (observed: the same gradient turns 30° in an ODF 1.2 file and 300° in an ODF 1.3 one). Read
138/// by the standard, the gradients of every older LibreOffice document would turn the wrong
139/// way. So for files older than 1.3 WordCraft reads them as written; from 1.3 on, as the
140/// standard says. WordCraft writes them with a unit (`30deg`), which both read the same.
141pub(super) fn legacy_angles(version: Option<&str>) -> bool {
142    version.is_none_or(|v| v.trim() < "1.3")
143}
144
145fn angle(v: &str, legacy: bool) -> Option<f32> {
146    let v = v.trim();
147    let (n, scale) = if let Some(n) = v.strip_suffix("deg") {
148        (n, 1.0)
149    } else if let Some(n) = v.strip_suffix("grad") {
150        (n, 0.9)
151    } else if let Some(n) = v.strip_suffix("rad") {
152        (n, 180.0 / std::f32::consts::PI)
153    } else if legacy {
154        // A deviation from the standard, out of necessity: see `legacy_angles`.
155        (v, 0.1)
156    } else {
157        // §18.3.1: without a unit, degrees.
158        (v, 1.0)
159    };
160    n.trim().parse::<f32>().ok().map(|a| a * scale).filter(|a| a.is_finite()).map(|a| a.rem_euclid(360.0))
161}

163/// A percentage (`50%`), 0–100.
164fn percent(v: Option<&str>) -> Option<f32> {
165    v?.trim().strip_suffix('%')?.trim().parse::<f32>().ok().filter(|p| p.is_finite()).map(|p| p.clamp(0.0, 100.0))
166}

168/// A `draw:gradient` (§16.40) as a fill. Linear and axial gradients run along their angle (ODF's
169/// 0° runs top to bottom, turning anticlockwise; WordCraft's 0° left to right, clockwise);
170/// radial, ellipsoid, square and rectangular ones run from their end colour in the middle to
171/// their start colour at the edge. The border is where the start colour stays plain; intensity
172/// darkens a colour.
173pub(super) fn gradient(a: &Attrs, legacy: bool) -> Option<Fill> {
174    let color = |k: &str, intensity: &str, default: Rgb| {
175        let c = get(a, k).and_then(parse_color).unwrap_or(default);
176        let i = percent(get(a, intensity)).unwrap_or(100.0) / 100.0;
177        let dim = |v: u8| (f32::from(v) * i).round().clamp(0.0, 255.0) as u8;
178        Rgb(dim(c.0), dim(c.1), dim(c.2))
179    };
180    let start = color("draw:start-color", "draw:start-intensity", Rgb::BLACK);
181    let end = color("draw:end-color", "draw:end-intensity", Rgb::WHITE);
182    let border = percent(get(a, "draw:border")).unwrap_or(0.0);
183    let stop = |pos: f32, color: Rgb| GradStop { pos, color, transparency: 0.0 };
184    let along = (90.0 - get(a, "draw:angle").and_then(|v| angle(v, legacy)).unwrap_or(0.0)).rem_euclid(360.0);
185    let centre = || {
186        let (cx, cy) = (percent(get(a, "draw:cx")).unwrap_or(50.0), percent(get(a, "draw:cy")).unwrap_or(50.0));
187        [cx, cy, 100.0 - cx, 100.0 - cy]
188    };
189    let (kind, stops) = match get(a, "draw:style").map(str::trim).unwrap_or("linear") {
190        "axial" => (GradientKind::Linear { angle: along, scaled: false }, vec![stop(0.0, start), stop(50.0, end), stop(100.0, start)]),
191        style @ ("radial" | "ellipsoid" | "square" | "rectangular") => {
192            let shape = if matches!(style, "square" | "rectangular") { PathShape::Rect } else { PathShape::Circle };
193            let mut stops = vec![stop(0.0, end)];
194            if border > 0.0 {
195                stops.push(stop(100.0 - border, start));
196            }
197            stops.push(stop(100.0, start));
198            (GradientKind::Path { shape, focus: centre() }, stops)
199        }
200        _ => {
201            let mut stops = vec![stop(0.0, start)];
202            if border > 0.0 {
203                stops.push(stop(border, start));
204            }
205            stops.push(stop(100.0, end));
206            (GradientKind::Linear { angle: along, scaled: false }, stops)
207        }
208    };
209    Fill::Gradient(Arc::new(Gradient { stops, kind, tile: None, rot_with_shape: true, flip: Default::default() })).sanitized()
210}

212/// The `draw:gradient` named `name` for gradient fill `g` (the reverse of [`gradient`]): a
213/// gradient running out from its middle and back is axial; more colours than ODF's two keep
214/// the first and last.
215pub(super) fn gradient_def(name: &str, g: &Gradient) -> Option<String> {
216    let hex = |c: Rgb| format!("#{}", c.hex().to_ascii_lowercase());
217    let (first, last) = (g.stops.first()?, g.stops.last()?);
218    let body = match g.kind {
219        GradientKind::Linear { angle, .. } => {
220            let odf = (90.0 - if angle.is_finite() { angle } else { 0.0 }).rem_euclid(360.0);
221            let axial = g.stops.len() == 3 && first.color == last.color && g.stops.get(1).is_some_and(|m| (m.pos - 50.0).abs() < 0.5);
222            let (style, end) = match g.stops.get(1) {
223                Some(mid) if axial => ("axial", mid.color),
224                _ => ("linear", last.color),
225            };
226            format!(
227                "draw:style=\"{style}\" draw:start-color=\"{}\" draw:end-color=\"{}\" draw:angle=\"{}deg\"",
228                hex(first.color),
229                hex(end),
230                odf.round()
231            )
232        }
233        GradientKind::Path { shape, focus: [l, t, r, b] } => {
234            let style = if shape == PathShape::Rect { "rectangular" } else { "radial" };
235            let mid = |a: f32, b: f32| if a.is_finite() && b.is_finite() { ((a + 100.0 - b) / 2.0).clamp(0.0, 100.0) } else { 50.0 };
236            format!(
237                "draw:style=\"{style}\" draw:cx=\"{}%\" draw:cy=\"{}%\" draw:start-color=\"{}\" draw:end-color=\"{}\"",
238                mid(l, r),
239                mid(t, b),
240                hex(last.color),
241                hex(first.color)
242            )
243        }
244    };
245    Some(format!("<draw:gradient draw:name=\"{name}\" {body} draw:start-intensity=\"100%\" draw:end-intensity=\"100%\" draw:border=\"0%\"/>"))
246}

248/// The width an arrowhead is drawn at on a line `width` points wide: three times as wide, and
249/// at least 5 points.
250pub(super) fn arrow_width(width: f32) -> String {

254/// Most named dashes and gradients kept (hostile files).
255const MAX_DEFS: usize = 10_000;
256
257/// The named dashes and gradients a document's styles define.
258#[derive(Default)]
259pub(super) struct Defs {
260    dashes: std::collections::HashMap<String, Dash>,
261    gradients: std::collections::HashMap<String, Fill>,
262    /// The part read is older than ODF 1.3: unitless angles are tenths of a degree (see
263    /// [`legacy_angles`]).
264    pub(super) legacy_angles: bool,
265}
266
267impl Defs {
268    /// Element `q` with attributes `a`: a dash or gradient definition, or a part's root, whose
269    /// version says how unitless gradient angles are read (see [`legacy_angles`]).
270    pub(super) fn element(&mut self, q: &str, a: &Attrs) {
271        match q {
272            "office:document-styles" | "office:document-content" | "office:document" => self.legacy_angles = legacy_angles(get(a, "office:version")),
273            "draw:stroke-dash" => self.read("stroke-dash", a),
274            "draw:gradient" => self.read("gradient", a),
275            _ => {}
276        }
277    }
278
279    /// Element `name` (local) with attributes `a`, when it is a definition.
280    pub(super) fn read(&mut self, name: &str, a: &Attrs) {
281        let Some(id) = get(a, "draw:name").map(String::from) else { return };
282        if self.dashes.len() + self.gradients.len() >= MAX_DEFS {
283            return;
284        }
285        match name {
286            "stroke-dash" => {
287                self.dashes.insert(id, Dash::read(a));
288            }
289            "gradient" => {
290                if let Some(g) = gradient(a, self.legacy_angles) {
291                    self.gradients.insert(id, g);
292                }
293            }
294            _ => {}
295        }
296    }
297
298    pub(super) fn dash(&self, name: &str) -> Option<&Dash> {
299        self.dashes.get(name)
300    }
301
302    pub(super) fn gradient(&self, name: &str) -> Option<&Fill> {
303        self.gradients.get(name)
304    }
305}

307/// The named definitions a writer's graphic styles use, for `styles.xml`.
308#[derive(Default)]
309pub(super) struct DrawDefs {
310    /// Each definition's name by its content (one name for equal definitions).
311    names: std::collections::BTreeMap<String, String>,
312    /// Name → definition.
313    pub(super) defs: std::collections::BTreeMap<String, String>,
314}
315
316impl DrawDefs {
317    /// The name of the definition `make` writes for `kind` (with this writer's `prefix`).
318    fn named(&mut self, kind: &str, prefix: &str, make: impl Fn(&str) -> Option<String>) -> Option<String> {
319        let key = make("")?;
320        if let Some(n) = self.names.get(&key) {
321            return Some(n.clone());
322        }
323        let name = format!("WordCraft_20_{kind}_20_{prefix}{}", self.names.len() + 1);
324        let def = make(&name)?;
325        self.names.insert(key, name.clone());
326        self.defs.insert(name.clone(), def);
327        Some(name)
328    }
329
330    /// The `draw:stroke-dash` name for `dash`; `None` for a solid line.
331    pub(super) fn dash(&mut self, dash: &LineDash, prefix: &str) -> Option<String> {
332        if dash.is_solid() {
333            return None;
334        }
335        self.named("Dash", prefix, |n| dash_def(n, dash))
336    }
337
338    /// The `draw:gradient` name for `g`.
339    pub(super) fn gradient(&mut self, g: &Gradient, prefix: &str) -> Option<String> {
340        self.named("Gradient", prefix, |n| gradient_def(n, g))
341    }
342
343    /// The arrowhead's `draw:marker` name.
344    pub(super) fn arrow(&mut self) -> &'static str {
345        self.defs.entry(ARROW.to_string()).or_insert_with(|| ARROW_DEF.to_string());
346        ARROW
347    }
